import numpy as np
from scipy.optimize import minimize

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and full gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # conv(a, p) gives dL/da
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a[::-1], M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def softmax_score_grad(f, beta):
    """
    Score and gradient in log-space. a = softmax(f) = exp(f)/sum(exp(f)).
    da/df_j = a_j * (delta_jk - a_k), so grad_f = grad_a / a * a - a * sum(grad_a * a)
    Actually: d score / df_j = sum_k (d score / da_k) * da_k / df_j
    da_k / df_j = a_k * (delta_{kj} - a_j)
    So grad_f_j = d_score/da_j * a_j - a_j * sum_k d_score/da_k * a_k
    = a_j * (grad_a_j - a @ grad_a)
    """
    n = len(f)
    f_shifted = f - f.max()
    exp_f = np.exp(f_shifted)
    a = exp_f / exp_f.sum()
    score_a, grad_a = lse_score_grad(a, beta)
    # grad_f_j = a_j * (grad_a_j - sum(grad_a * a))
    mean_grad = np.dot(grad_a, a)
    grad_f = a * (grad_a - mean_grad)
    return score_a, grad_f


def softmax_lbfgs(f0, beta, maxiter=5000, maxfun=50000, time_budget=None, t_start=None):
    """L-BFGS-B in log-space with softmax parameterization."""
    n = len(f0)
    f0 = f0.astype(np.float64).copy()

    def obj_and_grad(x):
        return softmax_score_grad(x, beta)

        return softmax_score_grad(x, beta)

        sigma = fft_len // 20
        target_mag = np.exp(-0.5 * ((k - freq_center) / sigma) ** 2)
        target_mag += 0.3 * np.exp(-0.5 * (k / (fft_len // 10)) ** 2)
    elif target_type == 'multi':
        # Multiple frequency bands
        target_mag = 0.5 + 0.3 * np.sin(2 * np.pi * k / fft_len * 3)
        target_mag = np.maximum(target_mag, 0.2)
    elif target_type == 'tapered':
        # Tapered: less energy at high frequencies
        target_mag = 1.0 / (1.0 + (normalized_k * fft_len / n) ** 0.5)
    else:
        target_mag = np.ones(fft_len // 2 + 1)

    # Initialize with random phase
    phase = rng.uniform(0, 2 * np.pi, fft_len // 2 + 1)
    phase[0] = 0  # DC component real
    A = target_mag * np.exp(1j * phase)

    a = np.ones(n) / n
    for it in range(n_iter):
        # Time domain: project onto non-negative, fixed-sum
        a_full = np.fft.irfft(A, fft_len)
        a = a_full[:n]
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 1e-12:
            a = a / s

        # Frequency domain: project onto target magnitude
        A = np.fft.rfft(a, fft_len)
        mag = np.abs(A)
        mag_safe = np.maximum(mag, 1e-12)
        A = target_mag * (A / mag_safe)

    return a


def multi_bump_init(n, rng, n_bumps=3, sigma_range=(0.05, 0.3)):
    """Create a sequence with multiple Gaussian bumps at random locations."""
    centers = rng.uniform(0, 1, n_bumps)
    sigmas = rng.uniform(sigma_range[0], sigma_range[1], n_bumps)
    weights = rng.uniform(0.5, 1.5, n_bumps)
    x = np.linspace(0, 1, n)
    a = np.zeros(n)
    for c, s, w in zip(centers, sigmas, weights):
        a += w * np.exp(-0.5 * ((x - c) / s) ** 2)
    a = np.maximum(a, 0)
    s = a.sum()
    if s > 1e-12:
        a = a / s
    return a


def random_smooth_init(n, rng, n_freq_bands=3):
    """Random smooth initialization using sum of random sinusoids."""
    x = np.linspace(0, 1, n)
    a = np.ones(n) * 0.5
    for _ in range(n_freq_bands):
        freq = rng.uniform(1, 20)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4)
        a += amp * np.sin(2 * np.pi * freq * x + phase)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def parent_resample(parent, n_target, rng=None):
    """Resample parent sequence to target length."""
    n_parent = len(parent)
    x_old = np.linspace(0, 1, n_parent)
    x_new = np.linspace(0, 1, n_target)
    a_new = np.interp(x_new, x_old, parent)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-12:
        a_new = a_new / s
    return a_new


def peak_directed_coord_descent(a, n_sweeps=5, top_k=30, time_budget=60.0, t_start=None):
    """
    Coordinate descent focusing on coordinates that most affect the current peak.
    Uses golden-section search for 1D optimization.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S0 = a.sum()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (t_start is not None) else np.inf
    gr = (math.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        S = a.sum()
        score = 2.0 * n * c.max() / (S * S)
        if score < best_score:
            best_score = score
            if best_score < 1.503:
                print(f"    Peak-directed CD: {best_score:.8f} at sweep {sweep}")

        # Find all "significant" peaks (within 5% of max)
        c_max = c.max()
        peak_threshold = 0.95 * c_max
        peak_lags = np.where(c >= peak_threshold)[0]

        # Compute subgradient contributions from each significant peak
        S2 = S * S
        # For each peak lag k, subgrad contribution is a[flipped] at that lag
        # Total subgrad of max is sum over active peaks (approximately)
        sg = np.zeros(n)
        for k in peak_lags[:10]:  # Limit to top 10 peaks
            js = max(0, int(k) - n + 1)
            je = min(int(k), n - 1)
            if je >= js:
                idx = int(k) - np.arange(js, je + 1)
                sg[js:je + 1] += a[idx]

        # Score gradient
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / (S2 * S))

        # Select top-k by |gradient|
        order = np.argsort(np.abs(grad))[::-1][:top_k]

        for j in order:
            if time.monotonic() > deadline:
                break
            j = int(j)
            a_old = a[j]
            if a_old < 1e-15:
                continue
            # Golden section in [0, a_old * 3]
            lo = 0.0
            hi = a_old * 3.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            a[j] = c1
            f1 = compute_score(a)
            a[j] = c2
            f2 = compute_score(a)
            for _ in range(12):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    a[j] = c1
                    f1 = compute_score(a)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    a[j] = c2
                    f2 = compute_score(a)
            best_val = c1 if f1 < f2 else c2
            a[j] = best_val
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc

    # Renormalize
    s = a.sum()
    if s > 1e-12:
        a = a / s
    return best_score, a


def adam_log_space(f0, lr=0.01, n_steps=50000, beta_lse=200,
                   time_budget=None, t_start=None):
    """Adam optimizer in log-space (softmax parameterization)."""
    n = len(f0)
    f = f0.astype(np.float64).copy()

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    a_init = np.exp(f - f.max())
    a_init = a_init / a_init.sum()
    best_a = a_init.copy()
    best_score = compute_score(a_init)
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf

    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break

        lr_t = lr / (1.0 + 1e-5 * t)
        score, grad_f = softmax_score_grad(f, beta_lse)

        m = b1 * m + (1.0 - b1) * grad_f
        v = b2 * v + (1.0 - b2) * grad_f * grad_f
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        f = f - lr_t * m_hat / (np.sqrt(v_hat) + eps)

        if t % 5000 == 0:
            a = np.exp(f - f.max())
            a = a / a.sum()
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    a = np.exp(f - f.max())
    a = a / a.sum()
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_score, best_a


def continuation_optimize(a0, betas, time_budget, t_start, use_log_space=True):
    """Continuation method with increasing beta."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S0 = a.sum()
    if S0 < 1e-12:
        a[:] = 1.0 / n
        S0 = 1.0
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget

    if use_log_space:
        f = np.log(a + 1e-12)
        # Center f
        f = f - f.mean()
    else:
        f = None

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - 5:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        time_alloc = min(remaining * 0.35, 80.0)

        if use_log_space:
            sc, a_out = softmax_lbfgs(f, beta, maxiter=3000, maxfun=20000,
                                      time_budget=time_alloc, t_start=t_start)
            if sc < best_score:
                best_score = sc
                best_a = a_out.copy()
                print(f"    Cont. log-space beta={beta}: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
            # Update f from a_out
            f = np.log(np.maximum(a_out, 1e-12)) - np.log(np.maximum(a_out, 1e-12)).mean()
        else:
            # Direct L-BFGS in a-space
            bounds = [(0.0, 1000.0)] * n
            def obj(x):
                return lse_score_grad(x, beta)
            def timed_obj(x):
                if time.monotonic() > t_start + deadline - t_start:
                    raise TimeoutError
                return lse_score_grad(x, beta)
            try:
                res = minimize(timed_obj, a, method='L-BFGS-B', jac=True, bounds=bounds,
                               options={'ftol': 1e-14, 'gtol': 1e-11,
                                        'maxiter': 3000, 'maxfun': 20000, 'maxls': 50})
                a_out = np.clip(res.x, 0, 1000)
            except (TimeoutError, Exception):
                a_out = a
            sc = compute_score(a_out)
            if sc < best_score:
                best_score = sc
                best_a = a_out.copy()
                print(f"    Cont. direct beta={beta}: {sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
            a = a_out

    return best_score, best_a


def spectral_perturbation_v2(a, rng, strength=0.2):
    """Enhanced spectral perturbation with amplitude and phase modifications."""
    n = len(a)
    fft_len = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, fft_len)

    # Modify amplitudes
    amp_factors = 1.0 + strength * rng.standard_normal(len(A))
    amp_factors = np.clip(amp_factors, 0.5, 1.5)

    # Modify phases
    phase_shifts = strength * 0.3 * rng.standard_normal(len(A))

    A_new = A * amp_factors * np.exp(1j * phase_shifts)
    a_new = np.fft.irfft(A_new, fft_len)[:n]
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() < 1e-10:
        return a.copy()
    return a_new * (a.sum() / a_new.sum())


def block_restructure(a, rng, n_swaps=2):
    """Restructure by swapping and randomly permuting blocks."""
    n = len(a)
    a_new = a.copy()
    for _ in range(n_swaps):
        block_len = int(rng.integers(n // 30, n // 8))
        i = int(rng.integers(0, n - block_len))
        j = int(rng.integers(0, n - block_len))
        if i != j:
            a_new[i:i+block_len], a_new[j:j+block_len] = a_new[j:j+block_len].copy(), a_new[i:i+block_len].copy()
    return a_new


def refine_candidate(a_init, time_budget, t_start):
    """
    Multi-stage refinement pipeline.
    Stage 1: Adam in log-space (30%)
    Stage 2: Continuation L-BFGS (40%)
    Stage 3: Peak-directed coord descent (20%)
    Stage 4: Final L-BFGS push (10%)
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / S
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget

    def try_update(sc, a_cand):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a_cand.copy()
            return True
        return False

    # Stage 1: Adam in log-space
    if time.monotonic() < deadline - 10:
        adam_time = min(deadline - time.monotonic(), time_budget * 0.3)
        f = np.log(np.maximum(a, 1e-12))
        f = f - f.mean()
        sc, a_opt = adam_log_space(f, lr=0.005, n_steps=50000, beta_lse=150,
                                   time_budget=adam_time, t_start=t_start)
        try_update(sc, a_opt)

    # Stage 2: Continuation L-BFGS
    if time.monotonic() < deadline - 10:
        cont_time = min(deadline - time.monotonic() - 5, time_budget * 0.4)
        sc, a_opt = continuation_optimize(best_a, [100, 500, 2000, 8000],
                                          cont_time, t_start, use_log_space=True)
        try_update(sc, a_opt)

    # Stage 3: Peak-directed coordinate descent
    if time.monotonic() < deadline - 5:
        coord_time = min(deadline - time.monotonic() - 3, time_budget * 0.2)
        sc, a_opt = peak_directed_coord_descent(best_a, n_sweeps=8, top_k=40,
                                                time_budget=coord_time, t_start=t_start)
        try_update(sc, a_opt)

    # Stage 4: Final high-beta L-BFGS
    if time.monotonic() < deadline - 3:
        final_time = min(deadline - time.monotonic() - 1, time_budget * 0.1)
        f = np.log(np.maximum(best_a, 1e-12))
        f = f - f.mean()
        sc, a_opt = softmax_lbfgs(f, 20000, maxiter=5000, maxfun=30000,
                                  time_budget=final_time, t_start=t_start)
        try_update(sc, a_opt)

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: Fienup initialization + multi-stage optimization.
    Strategy:
    1. Generate diverse candidates via frequency-domain design
    2. Quick-screen with short Adam runs
    3. Intensive optimization of top candidates
    4. Final polish
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 30.0  # Leave 30s buffer

    # Load parent sequence
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        parent_score = compute_score(parent)
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, score={parent_score:.8f}")
    else:
        parent = np.ones(4096) / 4096
        parent_score = compute_score(parent)
        n_parent = len(parent)
        print(f"No parent, using uniform n=4096, score={parent_score:.8f}")

    best_score = parent_score
    best_a = parent.copy()

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # ================================================================
    # Phase 1: Generate diverse candidates (20% of budget)
    # ================================================================
    p1_budget = effective_budget * 0.20
    print(f"\n{'='*60}")
    print(f"Phase 1: Generate diverse candidates (budget={p1_budget:.0f}s)")
    print(f"{'='*60}")

    n_base = n_parent
    candidates = []

    # Candidate 0: parent itself
    candidates.append(('parent', parent.copy()))

    # Fienup inits with different target spectra
    fienup_targets = ['flat', 'lowpass', 'ring', 'multi', 'tapered']
    for i, target in enumerate(fienup_targets):
        s_per = (i + 1) * 11
        a_f = fienup_init(n_base, target_type=target, n_iter=300,
                          rng=np.random.default_rng(s_per), lowpass_width=1.0)
        candidates.append((f'fienup_{target}', a_f))

    # Multi-bump inits
    for i in range(3):
        a_mb = multi_bump_init(n_base, np.random.default_rng(100 + i * 7),
                               n_bumps=2 + i, sigma_range=(0.05, 0.25))
        candidates.append((f'multi_bump_{i}', a_mb))

    # Random smooth inits
    for i in range(3):
        a_rs = random_smooth_init(n_base, np.random.default_rng(200 + i * 13),
                                  n_freq_bands=4 + i)
        candidates.append((f'random_smooth_{i}', a_rs))

    # Resampled parent at different lengths
    for factor in [0.75, 1.25, 1.5]:
        n_new = int(n_base * factor)
        if 512 <= n_new <= 20000:
            a_r = parent_resample(parent, n_new)
            candidates.append((f'parent_n{n_new}', a_r))

    # Spectral perturbation of parent
    for i in range(2):
        a_sp = spectral_perturbation_v2(parent, np.random.default_rng(300 + i * 17),
                                        strength=0.25 + 0.1 * i)
        candidates.append((f'spectral_pert_{i}', a_sp))

    print(f"Generated {len(candidates)} candidates")

    # ================================================================
    # Phase 2: Quick screening (25% of budget)
    # ================================================================
    p2_budget = effective_budget * 0.25
    print(f"\n{'='*60}")
    print(f"Phase 2: Quick screening (budget={p2_budget:.0f}s)")
    print(f"{'='*60}")

    p2_deadline = t_start + p1_budget + p2_budget
    screen_results = []
    n_screen = len(candidates)
    per_screen_time = p2_budget / n_screen

    for idx, (name, a_cand) in enumerate(candidates):
        if time.monotonic() > p2_deadline - 5:
            break
        remaining = p2_deadline - time.monotonic()
        alloc = min(remaining * 0.9, per_screen_time * 1.5)
        if alloc < 5:
            break

        # Quick Adam screening (short)
        n_c = len(a_cand)
        a_norm = a_cand / a_cand.sum() if a_cand.sum() > 1e-12 else np.ones(n_c) / n_c
        f_init = np.log(np.maximum(a_norm, 1e-12))
        f_init = f_init - f_init.mean()

        sc, a_opt = adam_log_space(f_init, lr=0.01, n_steps=20000, beta_lse=100,
                                   time_budget=min(alloc, 15.0), t_start=t_start)
        screen_results.append((sc, name, a_opt))
        print(f"    {name:25s}: {sc:.8f} ({time.monotonic()-t_start:.0f}s)")

    # Sort by score
    screen_results.sort(key=lambda x: x[0])
    print(f"\n  Top 5 screened:")
    for sc, name, _ in screen_results[:5]:
        print(f"    {sc:.8f}: {name}")

    # ================================================================
    # Phase 3: Intensive optimization of top candidates (40% of budget)
    # ================================================================
    p3_budget = effective_budget * 0.40
    print(f"\n{'='*60}")
    print(f"Phase 3: Intensive optimization (budget={p3_budget:.0f}s)")
    print(f"{'='*60}")

    p3_start = max(time.monotonic(), t_start + p1_budget)
    p3_deadline = p3_start + p3_budget
    n_top = min(5, len(screen_results))
    per_refine_time = p3_budget / n_top

    for idx in range(n_top):
        if time.monotonic() > p3_deadline - 10:
            break
        sc_init, name, a_cand = screen_results[idx]

        # If already better than best, update
        if sc_init < best_score:
            try_update(sc_init, a_cand, f"screen_{name}")

        remaining = p3_deadline - time.monotonic()
        alloc = min(remaining * 0.95, per_refine_time * 1.5)
        if alloc < 10:
            break

        print(f"\n  Refining #{idx}: {name} (init={sc_init:.8f}, alloc={alloc:.0f}s)")
        sc, a_opt = refine_candidate(a_cand, alloc, t_start)
        try_update(sc, a_opt, f"refine_{name}")

    # ================================================================
    # Phase 4: Try different strategy on best (10% of budget)
    # ================================================================
    p4_budget = effective_budget * 0.10
    print(f"\n{'='*60}")
    print(f"Phase 4: Alternative strategies on best (budget={p4_budget:.0f}s)")
    print(f"{'='*60}")

    p4_deadline = t_start + effective_budget * 0.95

    if time.monotonic() < p4_deadline - 10:
        # Try peak-directed CD on best with more sweeps
        alloc = min(p4_deadline - time.monotonic() - 5, p4_budget * 0.5)
        if alloc > 5:
            sc, a_opt = peak_directed_coord_descent(best_a, n_sweeps=15, top_k=50,
                                                    time_budget=alloc, t_start=t_start)
            try_update(sc, a_opt, "CD_phase4")

    # Try a fresh Fienup with more iterations on best's spectrum
    if time.monotonic() < p4_deadline - 10:
        n_best = len(best_a)
        # Use the current best's spectrum as target
        fft_len = 1 << (2 * n_best - 1).bit_length()
        A_best = np.fft.rfft(best_a, fft_len)
        target_mag = np.abs(A_best)
        # Slightly modify the target to explore
        target_mod = target_mag * (1.0 + 0.1 * rng.standard_normal(len(target_mag)))
        target_mod = np.maximum(target_mod, np.min(target_mag) * 0.5)

        # Fienup with modified target
        phase = rng.uniform(0, 2 * np.pi, len(target_mag))
        A = target_mod * np.exp(1j * phase)
        a_f2 = np.ones(n_best) / n_best
        for it in range(500):
            a_full = np.fft.irfft(A, fft_len)
            a_f2 = np.maximum(a_full[:n_best], 0)
            s = a_f2.sum()
            if s > 1e-12:
                a_f2 = a_f2 / s
            A = np.fft.rfft(a_f2, fft_len)
            mag = np.maximum(np.abs(A), 1e-12)
            A = target_mod * (A / mag)

        alloc = min(p4_deadline - time.monotonic() - 5, 30.0)
        if alloc > 5:
            sc, a_opt = refine_candidate(a_f2, alloc, t_start)
            try_update(sc, a_opt, "fienup_retarget")

    # ================================================================
    # Phase 5: Final polish (remaining time)
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    final_deadline = t_start + effective_budget - 5.0

    # High-beta L-BFGS in log-space
    if time.monotonic() < final_deadline - 10:
        f = np.log(np.maximum(best_a, 1e-12))
        f = f - f.mean()
        alloc = min(final_deadline - time.monotonic() - 5, 60.0)
        sc, a_opt = softmax_lbfgs(f, 50000, maxiter=10000, maxfun=50000,
                                  time_budget=alloc, t_start=t_start)
        try_update(sc, a_opt, "final_lbfgs_50k")

    # Final coordinate descent
    if time.monotonic() < final_deadline - 3:
        alloc = min(final_deadline - time.monotonic() - 1, 30.0)
        if alloc > 3:
            sc, a_opt = peak_directed_coord_descent(best_a, n_sweeps=10, top_k=50,
                                                    time_budget=alloc, t_start=t_start)
            try_update(sc, a_opt, "final_cd")

    # One more L-BFGS if time
    if time.monotonic() < final_deadline - 3:
        f = np.log(np.maximum(best_a, 1e-12))
        f = f - f.mean()
        alloc = min(final_deadline - time.monotonic() - 1, 20.0)
        if alloc > 3:
            sc, a_opt = softmax_lbfgs(f, 100000, maxiter=5000, maxfun=20000,
                                      time_budget=alloc, t_start=t_start)
            try_update(sc, a_opt, "final_lbfgs_100k")

    # ================================================================
    # Final verification and return
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
