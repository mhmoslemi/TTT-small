def solve_lp_cutting_plane(a, M, max_iter=50):
    """Solve max sum(b) s.t. b>=0, (a*b)_k <= M using cutting plane."""
    n = len(a)
    # Start with just the peak constraint
    c = fast_conv(a)  # a*a, but we need a*b
    # Actually, we need constraints of the form (a*b)_k <= M
    # Start with the constraint at k = argmax(a*a) as a proxy
    
    active_k = [int(c.argmax())]
    
    b = np.ones(n)  # Initial guess
    
    for it in range(max_iter):
        # Solve LP with current active constraints
        # max sum(b) s.t. (a*b)_k <= M for k in active_k, b >= 0
        
        # Build constraint matrix
        A = np.zeros((len(active_k), n))
        for i, k in enumerate(active_k):
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            for j in range(js, je + 1):
                A[i, j] = a[k - j]
        
        # Solve LP
        # ...
        
        # Check violated constraints
        ab = fast_conv(a, b)  # Actually need to compute a*b
        violated = np.where(ab > M + 1e-10)[0]
        if len(violated) == 0:
            break
        # Add the most violated
        worst = violated[np.argmax(ab[violated])]
        if worst not in active_k:
            active_k.append(int(worst))
    
    return b
```

def fast_cross_conv(a, b):
    """Linear cross-convolution a*b via FFT."""
    n = max(len(a), len(b))
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

```python
"""
LP-based peak reduction with cutting plane method for minimizing
2n * max(self-convolution) / (sum)^2 over non-negative sequences.

Key innovation: Instead of smooth approximations (LSE) + gradient methods,
we use the exact LP formulation alternately: fix one sequence, solve LP for
the other subject to cross-convolution bound. The cutting plane method makes
the LP tractable by only maintaining the binding constraints.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp
import time
import math

def fast_conv(a, b=None):
    """Linear convolution via FFT. If b is None, compute a*a."""
    if b is None:
        b = a
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def build_constraint_matrix(a, active_k):
    """Build sparse matrix A where A[i,j] = a[active_k[i] - j] if valid."""
    n = len(a)
    m = len(active_k)
    rows = np.zeros(m * n, dtype=np.int32)  # oversize
    cols = np.zeros(m * n, dtype=np.int32)
    vals = np.zeros(m * n)
    count = 0
    for i, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        width = je - js + 1
        if width > 0:
            rows[count:count+width] = i
            cols[count:count+width] = np.arange(js, je+1)
            vals[count:count+width] = a[k - np.arange(js, je+1)]
            count += width
    return sp.csc_matrix((vals[:count], (rows[:count], cols[:count])), shape=(m, n))

def solve_lp_conv(a, M, max_cp_iters=30, verbose=False):
    """
    Solve: max sum(b) s.t. b >= 0, (a*b)_k <= M for all k.
    Uses cutting plane: start with peak constraint, add violated ones.
    Returns optimal b.
    """
    n = len(a)
    
    # Initialize active constraints with the peak of a*a
    c_aa = fast_conv(a)
    active_k = [int(c_aa.argmax())]
    
    b = np.ones(n) / n  # Fallback
    
    for cp_iter in range(max_cp_iters):
        m = len(active_k)
        
        # Build constraint matrix
        A = build_constraint_matrix(a, active_k)
        
        # Solve LP using HiGHS
        c_obj = -np.ones(n)
        b_ub = np.full(m, M)
        bounds = [(0, None)] * n
        
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            break
        
        if not res.success:
            break
        
        b = res.x
        b = np.maximum(b, 0)
        
        # Check for violated constraints
        ab = fast_conv(a, b)
        violated = ab > M + M * 1e-8
        if not violated.any():
            if verbose:
                print(f"    LP converged in {cp_iter+1} CP iterations, {m} active constraints")
            break
        
        # Add the most violated constraint
        worst_k = int(np.argmax(ab))
        if worst_k not in active_k:
            active_k.append(worst_k)
        else:
            # All worst are already active but not satisfied?
            # This can happen numerically; just break.
            break
    
    return b

def lp_iteration(a, M, max_cp_iters=30):
    """
    One LP-based iteration:
    1. Solve LP: max sum(b) s.t. b >= 0, (a*b)_k <= M
    2. Return b (unnormalized)
    """
    b = solve_lp_conv(a, M, max_cp_iters=max_cp_iters)
    return b

def main_lp_search(a_init, time_budget, t_start, n_iters=None, verbose=True):
    """
    Main LP-based search: iteratively apply LP updates.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    a_norm = a / S  # Normalize to sum=1
    
    best_score = compute_score(a_norm)
    best_a = a_norm.copy()
    deadline = t_start + time_budget
    
    if n_iters is None:
        n_iters = 10000
    
    for it in range(n_iters):
        if time.monotonic() > deadline - 10:
            break
        
        S_cur = a_norm.sum()
        c = fast_conv(a_norm)
        M = float(c.max())
        cur_score = 2.0 * n * M / (S_cur * S_cur)
        
        if cur_score < best_score:
            best_score = cur_score
            best_a = a_norm.copy()
        
        if verbose and it % 10 == 0:
            elapsed = time.monotonic() - t_start
            print(f"  LP iter {it}: score={cur_score:.8f}, M={M:.6f}, n_active=? (t={elapsed:.0f}s)")
        
        # Solve LP
        b = solve_lp_conv(a_norm, M, max_cp_iters=30)
        
        if b is None or b.sum() < 1e-12:
            break
        
        S_b = b.sum()
        b_norm = b / S_b  # Normalize to sum=1
        
        # Try different step sizes
        best_t = 0
        best_new_score = cur_score
        for t in [1.0, 0.8, 0.6, 0.4, 0.2]:
            a_new = (1 - t) * a_norm + t * b_norm
            S_new = a_new.sum()
            if S_new < 1e-12:
                continue
            a_new = a_new / S_new  # Re-normalize
            sc = compute_score(a_new)
            if sc < best_new_score:
                best_new_score = sc
                best_t = t
        
        if best_t == 0:
            # No improvement; try smaller steps or break
            if it > 50:
                break
            continue
        
        a_norm = (1 - best_t) * a_norm + best_t * b_norm
        a_norm = a_norm / a_norm.sum()
    
    # Final check
    final_score = compute_score(best_a)
    return final_score, best_a

def coord_descent(a, n_sweeps=5, top_k=50, time_budget=60.0, t_start=None):
    """Coordinate descent with golden-section search."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (t_start is not None) else np.inf
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        k_star = int(c.argmax())
        M = float(c[k_star])
        S = float(a.sum())
        
        # Subgradient
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        sg[js:je+1] = a[k_star - np.arange(js, je+1)]
        
        grad = (2.0 * n) * (sg / (S*S) - 2.0 * M / (S**3))
        
        order = np.argsort(np.abs(grad))[::-1][:top_k]
        
        for j in order:
            if time.monotonic() > deadline:
                break
            j = int(j)
            a_old = a[j]
            lo = 0.0
            hi = max(a_old * 4.0, a.max() * 0.5)
            
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            
            a[j] = c1
            f1 = compute_score(a)
            a[j] = c2
            f2 = compute_score(a)
            
            for _ in range(12):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    a[j] = c1
                    f1 = compute_score(a)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    a[j] = c2
                    f2 = compute_score(a)
            
            a[j] = c1 if f1 < f2 else c2
            if compute_score(a) < best_score:
                best_score = compute_score(a)
    
    return best_score, a

def perturbation_spectral(a, rng, strength=0.15):
    """Perturb in frequency domain."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    amp = 1.0 + strength * rng.standard_normal(len(A))
    amp = np.clip(amp, 0.3, 2.0)
    phase = strength * 0.3 * rng.standard_normal(len(A))
    A_new = A * amp * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.clip(a_new, 0.0, 1e10)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new

def perturbation_gaussian(a, rng, n_bumps=5):

def perturbation_block_swap(a, rng, n_blocks=3):

"""
LP-based peak reduction with cutting plane method for minimizing
2n * max(self-convolution) / (sum)^2 over non-negative sequences.

Key innovation: Alternating LP formulation - fix sequence a, solve LP for b
maximizing sum(b) subject to cross-convolution (a*b)_k <= M. The cutting
plane method keeps the LP tractable by only maintaining binding constraints.
Combined with spectral perturbations, coordinate descent polishing, and
adaptive step sizes for robust convergence.
"""

import numpy as np
from scipy.optimize import minimize, linprog
import scipy.sparse as sp
import time
import math


def fast_conv(a, b=None):
    """Linear convolution via FFT. If b is None, compute a*a."""
    if b is None:
        b = a
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def build_constraint_matrix(a, active_k):
    """Build sparse matrix where A[i,j] = a[active_k[i]-j] if valid."""
    n = len(a)
    m = len(active_k)
    rows_list = []
    cols_list = []
    vals_list = []
    for i, k in enumerate(active_k):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        width = je - js + 1
        if width > 0:
            rows_list.extend([i] * width)
            cols_list.extend(range(js, je + 1))
            vals_list.extend(a[k - np.arange(js, je + 1)].tolist())
    if not rows_list:
        return sp.csc_matrix((m, n))
    return sp.csc_matrix((vals_list, (rows_list, cols_list)), shape=(m, n))


def solve_lp_conv(a, M, max_cp_iters=40):
    """
    Solve: max sum(b) s.t. b >= 0, (a*b)_k <= M for all k.
    Uses cutting plane: start with peak lag, add most-violated constraints.
    Returns optimal b (or uniform fallback).
    """
    n = len(a)
    c_aa = fast_conv(a)
    active_k = [int(c_aa.argmax())]
    # Also add second and third peak for faster convergence
    c_sorted = np.sort(c_aa)[::-1]
    top_lags = np.argsort(c_aa)[::-1][:5]
    active_k = [int(k) for k in top_lags]

    b = np.ones(n) / n

    for cp_iter in range(max_cp_iters):
        m = len(active_k)
        A = build_constraint_matrix(a, active_k)

        c_obj = -np.ones(n)
        b_ub = np.full(m, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                          method='highs', options={"presolve": True, "time_limit": 60.0})
        except Exception:
            break

        if not res.success or res.x is None:
            break

        b = np.maximum(res.x, 0.0)

        # Check for violated constraints
        ab = fast_conv(a, b)
        tol = M * 1e-7
        violated = ab > M + tol
        if not violated.any():
            break

        worst_k = int(np.argmax(ab))
        if worst_k not in active_k:
            active_k.append(worst_k)
        else:
            # Numerical issue; try next worst
            sorted_violated = np.argsort(ab)[::-1]
            added = False
            for vk in sorted_violated:
                vk = int(vk)
                if vk not in active_k:
                    active_k.append(vk)
                    added = True
                    break
            if not added:
                break

    return b


def gradient_lse(a, beta):
    """Compute LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled - scaled.max())
    log_sum = np.log(exp_s.sum()) + scaled.max()
    L = c_max + (log_sum - scaled.max()) / beta
    p = exp_s / exp_s.sum()

    # Gradient: g[j] = 2 * sum_k p[k] * a[k-j]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a[::-1], M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_ap = np.fft.irfft(RA * P, M2_fft)
    g = 2.0 * conv_ap[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def adam_polish(a_init, lr=0.003, n_steps=100000, beta_lse=300,
                time_budget=60.0, t_start=None):
    """Adam optimizer with LSE-smoothed gradient."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S0 = a.sum()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = (t_start + time_budget) if t_start is not None else np.inf

    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break
        lr_t = lr / (1.0 + 1e-5 * t)
        score, grad = gradient_lse(a, beta_lse)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)
        s = a.sum()
        if s > 1e-12:
            a = a * (S0 / s)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_score, best_a


def coord_descent_golden(a, n_sweeps=5, top_k=50, time_budget=60.0, t_start=None):
    """Coordinate descent with golden-section 1D search."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if t_start is not None else np.inf
    gr = (math.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        k_star = int(c.argmax())
        M = float(c[k_star])
        S = float(a.sum())
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        sg[js:je + 1] = a[k_star - np.arange(js, je + 1)]
        grad = (2.0 * n) * (sg / (S * S) - 2.0 * M / (S ** 3))
        order = np.argsort(np.abs(grad))[::-1][:top_k]

        for j in order:
            if time.monotonic() > deadline:
                break
            j = int(j)
            a_old = a[j]
            lo = 0.0
            hi = max(a_old * 4.0, 0.5 * a.max())
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            a[j] = c1
            f1 = compute_score(a)
            a[j] = c2
            f2 = compute_score(a)
            for _ in range(14):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    a[j] = c1
                    f1 = compute_score(a)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    a[j] = c2
                    f2 = compute_score(a)
            a[j] = c1 if f1 < f2 else c2
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc

    return best_score, a


def perturbation_spectral(a, rng, strength=0.15):
    """Perturb in DFT domain."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    amp = 1.0 + strength * rng.standard_normal(len(A))
    amp = np.clip(amp, 0.3, 2.0)
    phase = strength * 0.3 * rng.standard_normal(len(A))
    A_new = A * amp * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.clip(a_new, 0.0, 1e10)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def perturbation_gaussian(a, rng, n_bumps=5):
    """Add random Gaussian bumps."""
    n = len(a)
    a_new = a.copy()
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 60), max(2, n // 8)))
        amp = float(rng.uniform(-0.12, 0.12)) * a.max()
        x = np.arange(n, dtype=np.float64)
        a_new += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
    a_new = np.clip(a_new, 0.0, 1e10)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def perturbation_block_swap(a, rng, n_blocks=3):
    """Swap random blocks."""
    n = len(a)
    a_new = a.copy()
    for _ in range(n_blocks):
        blen = int(rng.integers(max(1, n // 40), max(2, n // 6)))
        i = int(rng.integers(0, max(1, n - blen)))
        j = int(rng.integers(0, max(1, n - blen)))
        if i != j:
            tmp = a_new[i:i + blen].copy()
            a_new[i:i + blen] = a_new[j:j + blen]
            a_new[j:j + blen] = tmp
    return a_new


def perturbation_anti_pal(a, rng, strength=0.1):
    """Add anti-palindromic component to reduce center peak."""
    n = len(a)
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    tilt = (x - center) / max(1.0, center)
    freq = float(rng.uniform(1, 4))
    tilt += 0.5 * np.sin(2.0 * np.pi * freq * x / n)
    tilt_anti = 0.5 * (tilt - tilt[::-1])
    a_new = a + strength * tilt_anti * a.max()
    a_new = np.clip(a_new, 0.0, 1e10)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new

def lp_search_loop(a_init, time_budget, t_start, verbose=True):

    stall_count = 0

    while time.monotonic() < deadline - 15:
        it += 1
        S_cur = a_norm.sum()
        c = fast_conv(a_norm)
        M = float(c.max())
        cur_score = 2.0 * n * M / (S_cur * S_cur)

        if cur_score < best_score:
            best_score = cur_score
            best_a = a_norm.copy()
            stall_count = 0
        else:
            stall_count += 1

        if verbose and it % 5 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    LP iter {it}: score={cur_score:.8f}, best={best_score:.8f} (t={elapsed:.0f}s)")

        if stall_count > 40:
            break

        # Solve LP
        b = solve_lp_conv(a_norm, M, max_cp_iters=30)
        if b is None or b.sum() < 1e-12:
            break

        S_b = b.sum()
        b_norm = b / S_b

        # Adaptive step size: try multiple t values
        best_t = 0.0
        best_new_score = cur_score
        for t in [1.0, 0.7, 0.5, 0.3, 0.15]:
            a_cand = (1 - t) * a_norm + t * b_norm
            s_cand = a_cand.sum()
            if s_cand < 1e-12:
                continue
            a_cand = a_cand / s_cand
            sc = compute_score(a_cand)
            if sc < best_new_score:
                best_new_score = sc
                best_t = t

        if best_t == 0.0:
            # No improvement; try very small step
            a_cand = (1 - 0.05) * a_norm + 0.05 * b_norm
            s_cand = a_cand.sum()
            if s_cand > 1e-12:
                a_cand = a_cand / s_cand
                sc = compute_score(a_cand)
                if sc < cur_score:
                    a_norm = a_cand
                    best_t = 0.05
            if best_t == 0.0:
                if it > 30:

                continue

        a_norm = (1 - best_t) * a_norm + best_t * b_norm
        s_final = a_norm.sum()
        if s_final > 1e-12:
            a_norm = a_norm / s_final

    final_sc = compute_score(best_a)
    return final_sc, best_a


def refine_with_lp_and_polish(a_init, time_budget, t_start, verbose=True):
    """
    Full refinement: LP search + Adam polish + coordinate descent.
    """
    a = a_init.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget

    def try_update(sc, a_cand, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_cand, 0.0, 1000.0).copy()
            return True
        return False

    # Stage 1: LP-based search (55% of budget)
    if time.monotonic() < deadline - 20:
        lp_time = min((deadline - time.monotonic()) * 0.55, time_budget * 0.55)
        sc, a_opt = lp_search_loop(a, lp_time, t_start, verbose=verbose)
        try_update(sc, a_opt, "LP")
        a = best_a

    # Stage 2: Adam polish (25% of budget)
    if time.monotonic() < deadline - 15:
        adam_time = min((deadline - time.monotonic()) * 0.25, time_budget * 0.25)
        sc, a_opt = adam_polish(a, lr=0.002, n_steps=100000, beta_lse=300,
                                time_budget=adam_time, t_start=t_start)
        try_update(sc, a_opt, "Adam")
        a = best_a

    # Stage 3: L-BFGS with increasing beta (10% of budget)
    if time.monotonic() < deadline - 10:
        for beta in [500, 2000, 10000]:
            if time.monotonic() > deadline - 8:
                break
            b_time = min(deadline - time.monotonic() - 8, 30.0)
            if b_time < 5:
                break
            sc, a_opt = lbfgs_lse(a, beta, maxiter=2000, maxfun=10000,
                                  time_budget=b_time, t_start=t_start)
            try_update(sc, a_opt, f"LBFGS b={beta}")
            a = best_a

    # Stage 4: Coordinate descent (10% of budget)
    if time.monotonic() < deadline - 5:
        cd_time = min(deadline - time.monotonic() - 5, time_budget * 0.10)
        if cd_time > 5:
            sc, a_opt = coord_descent_golden(a, n_sweeps=5, top_k=40,
                                             time_budget=cd_time, t_start=t_start)
            try_update(sc, a_opt, "CoordDesc")

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: LP cutting-plane iteration + diverse restarts + polishing.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(60.0, float(budget_s)) - 20.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        n_parent = len(parent)
        parent_score = compute_score(parent)
        print(f"Parent: n={n_parent}, score={parent_score:.8f}")
    else:
        parent = np.ones(4096)
        parent_score = compute_score(parent)
        n_parent = len(parent)
        print(f"No parent, using uniform n={n_parent}, score={parent_score:.8f}")

    best_score = parent_score
    best_a = parent.copy()

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # ================================================================
    # Phase 1: Quick Adam to escape bad local structure (10% budget)
    # ================================================================
    p1_budget = effective_budget * 0.10
    print(f"\n{'='*60}")
    print(f"Phase 1: Quick Adam polish (budget={p1_budget:.0f}s)")
    print(f"{'='*60}")

    sc, a_opt = adam_polish(parent, lr=0.005, n_steps=50000, beta_lse=150,
                            time_budget=p1_budget, t_start=t_start)
    try_update(sc, a_opt, "Adam initial")
    a_for_lp = best_a.copy()

    # ================================================================
    # Phase 2: Main LP-based search (50% budget)
    # ================================================================
    p2_budget = effective_budget * 0.50
    print(f"\n{'='*60}")
    print(f"Phase 2: LP cutting-plane search (budget={p2_budget:.0f}s)")
    print(f"{'='*60}")

    sc, a_opt = lp_search_loop(a_for_lp, p2_budget, t_start, verbose=True)
    try_update(sc, a_opt, "LP main")

    # Post-LP polish
    if time.monotonic() < t_start + effective_budget * 0.60 - 10:
        polish_time = min(30.0, t_start + effective_budget * 0.60 - 10 - time.monotonic())
        if polish_time > 5:
            sc, a_opt = coord_descent_golden(best_a, n_sweeps=3, top_k=30,
                                             time_budget=polish_time, t_start=t_start)
            try_update(sc, a_opt, "Post-LP CoordDesc")

    # ================================================================
    # Phase 3: Diverse restarts with LP refinement (25% budget)
    # ================================================================
    p3_budget = effective_budget * 0.25
    print(f"\n{'='*60}")
    print(f"Phase 3: Diverse restarts (budget={p3_budget:.0f}s)")
    print(f"{'='*60}")

    p3_deadline = time.monotonic() + p3_budget
    n_restarts = 6
    per_restart = p3_budget / (n_restarts + 1)

    for ri in range(n_restarts):
        if time.monotonic() > p3_deadline - 15:
            break
        t_remain = p3_deadline - time.monotonic()
        time_alloc = min(t_remain * 0.85, per_restart * 1.3)
        if time_alloc < 20:
            break

        # Generate perturbation
        r = ri
        if r == 0:
            a_pert = perturbation_spectral(best_a, rng, strength=0.12)
        elif r == 1:
            a_pert = perturbation_spectral(best_a, rng, strength=0.25)
        elif r == 2:
            a_pert = perturbation_gaussian(best_a, rng, n_bumps=7)
        elif r == 3:
            a_pert = perturbation_block_swap(best_a, rng, n_blocks=4)
        elif r == 4:
            a_pert = perturbation_anti_pal(best_a, rng, strength=0.12)
        else:
            a_pert = perturbation_spectral(
                perturbation_gaussian(best_a, rng, 3), rng, 0.1)

        a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Refine with LP + polish
        sc, a_opt = refine_with_lp_and_polish(a_pert, time_alloc, t_start, verbose=False)
        try_update(sc, a_opt, f"restart#{ri}")

    # ================================================================
    # Phase 4: Intensive optimization of best (10% budget)
    # ================================================================
    p4_budget = effective_budget * 0.10
    print(f"\n{'='*60}")
    print(f"Phase 4: Intensive optimization (budget={p4_budget:.0f}s)")
    print(f"{'='*60}")

    p4_deadline = time.monotonic() + p4_budget
    if time.monotonic() < p4_deadline - 5:
        # More LP iterations on the best
        lp_time = min((p4_deadline - time.monotonic()) * 0.5, p4_budget * 0.5)
        if lp_time > 10:
            sc, a_opt = lp_search_loop(best_a, lp_time, t_start, verbose=False)
            try_update(sc, a_opt, "Intensive LP")

        # L-BFGS at high beta
        if time.monotonic() < p4_deadline - 5:
            for beta in [5000, 20000]:
                if time.monotonic() > p4_deadline - 5:
                    break
                bt = min(p4_deadline - time.monotonic() - 5, 30.0)
                if bt < 5:
                    break
                sc, a_opt = lbfgs_lse(best_a, beta, maxiter=5000, maxfun=30000,
                                      time_budget=bt, t_start=t_start)
                try_update(sc, a_opt, f"Intensive LBFGS b={beta}")

    # ================================================================
    # Phase 5: Final polish (remaining time)
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    final_deadline = t_start + effective_budget - 8.0

    if time.monotonic() < final_deadline - 5:
        cd_time = min(final_deadline - time.monotonic() - 5, 40.0)
        if cd_time > 5:
            sc, a_opt = coord_descent_golden(best_a, n_sweeps=8, top_k=50,
                                             time_budget=cd_time, t_start=t_start)
            try_update(sc, a_opt, "Final CoordDesc")

    # Final LBFGS
    if time.monotonic() < final_deadline - 3:
        ft = min(final_deadline - time.monotonic() - 3, 25.0)
        if ft > 3:
            sc, a_opt = lbfgs_lse(best_a, 50000, maxiter=10000, maxfun=50000,
                                  time_budget=ft, t_start=t_start)
            try_update(sc, a_opt, "Final LBFGS")

    # ================================================================
    # Output
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
