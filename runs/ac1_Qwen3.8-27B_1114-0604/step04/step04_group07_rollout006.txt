"""
Hybrid LSE-Adam optimizer with multi-scale strategy for minimizing
2n*max(self-conv)/(sum)^2 over non-negative sequences.
Key improvements: aggressive beta continuation, multi-scale (4096→8192→16384),
subgradient polish, LP-dual-inspired updates, and diverse restarts.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Linear cross-convolution via FFT."""
    n_a, n_b = len(a), len(b)
    M = n_a + n_b - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a (assumes a >= 0)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    shifted = beta * (c - c_max)
    exp_vals = np.exp(shifted)
    exp_sum = float(exp_vals.sum())
    p = exp_vals / exp_sum
    L = c_max + np.log(exp_sum) / beta

    # grad_L[i] = 2/beta * sum_k p[k] * a[k-i] (for valid k-i in [0,n-1])
    # = 2/beta * cross_corr(p, a)[i] where cross_corr[i] = sum_r p[r+i]*a[r]
    # Compute via FFT: (a * p_rev)[2n-2-i]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    A_fft = np.fft.rfft(a, M2_fft)
    PR_fft = np.fft.rfft(p[::-1], M2_fft)
    conv_apr = np.fft.irfft(A_fft * PR_fft, M2_fft)[:M2]
    cross_corr = conv_apr[n - 1:2 * n - 1][::-1]

    grad_L = (2.0 / beta) * cross_corr
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def subgrad_score(a, K=1):
    """Exact score and (multi-peak) subgradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    # Find top K peaks
    if K <= 1:
        k_star = int(c.argmax())
        ks = [k_star]
    else:
        ks = np.argsort(c)[-K:][::-1].tolist()

    sg = np.zeros(n)
    for k in ks:
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            idx = k - np.arange(js, je + 1)
            sg[js:je + 1] += a[idx]
    sg /= len(ks)

    score = 2.0 * n * M / (S * S)
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
    return score, grad


def adam_lse(a_init, beta_schedule, lr_schedule, n_steps, time_limit_s, t_start):
    """Adam with LSE smoothing and scheduled beta/lr."""
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0.0, None).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
    else:
        a /= S

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    t_global = 0

    best_a = a.copy()
    best_score = compute_score(a)

    n_beta = len(beta_schedule)
    steps_per_beta = max(1, n_steps // n_beta)

    for bi in range(n_beta):
        beta = beta_schedule[bi]
        lr = lr_schedule[bi] if bi < len(lr_schedule) else lr_schedule[-1]
        for t in range(steps_per_beta):
            if time.monotonic() - t_start > time_limit_s:
                break
            t_global += 1
            score, grad = lse_score_grad(a, beta)

            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            m_hat = m / (1.0 - b1 ** t_global)
            v_hat = v / (1.0 - b2 ** t_global)

            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s > 1e-12:
                a /= s

            if t_global % 5000 == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_score, best_a


def subgrad_descent(a_init, n_steps, lr_init, time_limit_s, t_start, K=3):
    """Projected subgradient descent with adaptive step size."""
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0.0, None).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
    else:
        a /= S

    best_a = a.copy()
    best_score = compute_score(a)
    lr = lr_init
    patience = 0
    best_local_score = best_score

    for t in range(n_steps):
        if time.monotonic() - t_start > time_limit_s:
            break
        score, grad = subgrad_score(a, K=K)
        # Adaptive step size
        gnorm = np.sqrt((grad ** 2).sum()) + 1e-12
        step = lr / (1.0 + 0.01 * t)
        a_new = a - step * grad / gnorm
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-12:
            a_new /= s
        else:
            a_new = a.copy()

        sc_new = compute_score(a_new)
        if sc_new < best_local_score:
            best_local_score = sc_new
            patience = 0
        else:
            patience += 1
            if patience > 2000:
                lr *= 0.5
                patience = 0

        a = a_new
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

    return best_score, best_a


def multiplicative_weight_update(a, n_iters=200, lr=0.5):
    """MWU: find g maximizing sum(g) s.t. (a*g)[k] <= M, g>=0."""
    n = len(a)
    M = float(fast_conv(a).max())
    g = np.ones(n) / n

    for it in range(n_iters):
        c = cross_conv(a, g)
        # Penalize violations
        excess = np.maximum(0, c - M)
        # Update g: reduce elements contributing to violated constraints
        log_w = np.zeros(n)
        for k in range(len(excess)):
            if excess[k] > 0:
                js = max(0, k - n + 1)
                je = min(k, n - 1)
                if je >= js:
                    idx = k - np.arange(js, je + 1)
                    log_w[js:je + 1] -= lr * excess[k] * a[idx]
        g = g * np.exp(log_w)
        g = np.maximum(g, 1e-15)
        g /= g.sum()

    return g


def coord_descent_peak(a, n_sweeps=50, time_limit_s=None, t_start=None):
    """Coordinate descent targeting peak reduction."""
    n = len(a)
    a = np.clip(a.astype(np.float64), 0.0, None).copy()
    a /= a.sum()
    best_score = compute_score(a)
    deadline = (t_start + time_limit_s) if (time_limit_s is not None and t_start is not None) else np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        k_star = int(c.argmax())
        score = compute_score(a)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)

        # Try reducing peak contributors
        for j in range(js, je + 1):
            if time.monotonic() > deadline + 5:
                break
            a_old = a[j]
            if a_old < 1e-12:
                continue
            for frac in [0.7, 0.5, 0.3]:
                a[j] = a_old * frac
                removed = a_old - a[j]
                non_peak = np.ones(n, dtype=bool)
                non_peak[js:je + 1] = False
                if non_peak.sum() > 0:
                    a[non_peak] += removed / non_peak.sum()
                sc = compute_score(a)
                if sc < score:
                    score = sc
                    if sc < best_score:
                        best_score = sc
                    break
            else:
                a[j] = a_old

    return best_score, a


def generate_inits(rng, n, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n) / n

    inits.append(np.ones(n) / n)

    g = np.exp(-0.5 * ((x - 0.5) / 0.2) ** 2)
    inits.append(g / g.sum())

    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(rc / rc.sum())

    freq = 3 + 5 * rng.random()
    s = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x + 2 * np.pi * rng.random())
    s = np.maximum(s, 0.01)
    inits.append(s / s.sum())

    f1, f2 = rng.uniform(1, 5), rng.uniform(5, 12)
    m = 1.0 + 0.2 * np.cos(2 * np.pi * f1 * x) + 0.15 * np.cos(2 * np.pi * f2 * x)
    m = np.maximum(m, 0.01)
    inits.append(m / m.sum())

    r = np.abs(rng.standard_normal(n)) + 0.3
    k = max(3, n // 40)
    r = np.convolve(r, np.ones(k) / k, mode='same')
    inits.append(r / r.sum())

    # Paley-like: use quadratic residue pattern
    if n <= 10000:
        q = np.zeros(n)
        for i in range(n):
            q[i] = 1.0 if ((i * i) % n) < n // 2 else 0.1
        q = np.convolve(q, np.ones(max(3, n // 100)) / max(3, n // 100), mode='same')
        q = np.maximum(q, 0.01)
        inits.append(q / q.sum())

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy() / parent.sum())
        else:
            p_resamp = np.interp(np.linspace(0, len(parent) - 1, n), np.arange(len(parent)), parent)
            p_resamp = np.maximum(p_resamp, 0)
            if p_resamp.sum() > 0:
                inits.append(p_resamp / p_resamp.sum())

    return inits


def resample_seq(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    return np.maximum(np.interp(x_new, x_old, a), 0.0)


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B polishing."""
    n = len(a0)
    a0 = np.clip(a0, 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        x_c = np.maximum(x, 0.0)
        sc, gr = lse_score_grad(x_c, beta)
        return sc, gr

    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-14, 'gtol': 1e-11, 'maxiter': maxiter,
                            'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out

    p4_time = effective_budget * 0.10
    print(f"\n{'='*60}")
    print(f"Phase 4: MWU updates (budget={p4_time:.0f}s)")
    print(f"{'='*60}")

    if best_a is not None and len(best_a) <= 16384:
        remaining = t_start + p4_time - time.monotonic()
        if remaining > 5:
            t0 = time.monotonic()
            g = multiplicative_weight_update(best_a, n_iters=100, lr=0.3)
            for t_blend in [0.2, 0.4, 0.6, 0.8]:
                a_blend = (1 - t_blend) * best_a + t_blend * g
                a_blend = np.maximum(a_blend, 0)
                if a_blend.sum() > 0:
                    a_blend /= a_blend.sum()
                sc = compute_score(a_blend)
                try_update(sc, a_blend, f"MWU blend={t_blend}")
            print(f"  MWU done ({time.monotonic()-t0:.1f}s)")

            # Follow up with Adam on best blend
            remaining = t_start + p4_time - time.monotonic()
            if remaining > 5:
                sc, a_opt = adam_lse(best_a, [10000, 30000], [0.001, 0.0005],
                                     n_steps=200000, time_limit_s=remaining, t_start=t_start)
                try_update(sc, a_opt, "Phase4 Adam after MWU")

    # ============================================================
    # Phase 5: Coordinate descent (10% time)
    # ============================================================
    p5_time = effective_budget * 0.10
    print(f"\n{'='*60}")
    print(f"Phase 5: Coord descent (budget={p5_time:.0f}s)")
    print(f"{'='*60}")

    if best_a is not None:
        remaining = t_start + p5_time - time.monotonic()
        if remaining > 5:
            t0 = time.monotonic()
            sc, a_opt = coord_descent_peak(best_a, n_sweeps=50,
                                           time_limit_s=remaining, t_start=t_start)
            try_update(sc, a_opt, "Coord descent")
            print(f"  Coord descent: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ============================================================
    # Phase 6: Diverse restarts (15% time)
    # ============================================================
    p6_time = effective_budget * 0.15
    print(f"\n{'='*60}")
    print(f"Phase 6: Diverse restarts (budget={p6_time:.0f}s)")
    print(f"{'='*60}")

    n_restarts = 6
    if best_a is not None:
        restart_budget = p6_time / n_restarts
        n_cur = len(best_a)

        for r in range(n_restarts):
            if time.monotonic() > t_start + effective_budget - 20:
                break
            strategy = r % 3
            if strategy == 0:
                # Log-space Gaussian perturbation
                log_a = np.log(np.clip(best_a, 1e-15, 1e10))
                noise = 0.03 + 0.07 * rng.random()
                a_pert = np.exp(log_a + noise * rng.standard_normal(n_cur))
                a_pert = np.maximum(a_pert, 0)
                a_pert /= a_pert.sum()
            elif strategy == 1:
                # Peak-avoidance
                c = fast_conv(best_a)
                k_star = int(c.argmax())
                js = max(0, k_star - n_cur + 1)
                je = min(k_star, n_cur - 1)
                a_pert = best_a.copy()
                a_pert[js:je + 1] *= (0.4 + 0.3 * rng.random())
                non_peak = np.ones(n_cur, dtype=bool)
                non_peak[js:je + 1] = False
                a_pert[non_peak] *= (1.0 + 0.2 * rng.random())
                a_pert = np.maximum(a_pert, 0)
                a_pert /= a_pert.sum()
            else:
                # Fourier modulation
                x = np.arange(n_cur) / n_cur
                freqs = rng.uniform(1, 8, size=4)
                phases = rng.uniform(0, 2 * np.pi, size=4)
                amps = rng.uniform(0.03, 0.1, size=4)
                mod = 1.0
                for f, p, a_ in zip(freqs, phases, amps):
                    mod += a_ * np.cos(2 * np.pi * f * x + p)
                a_pert = best_a * np.maximum(mod, 0.05)
                a_pert /= a_pert.sum()

            t_alloc = min(restart_budget, t_start + effective_budget - 20 - time.monotonic())
            if t_alloc < 3:
                break
            sc, a_opt = adam_lse(a_pert, [50, 200, 1000], [0.02, 0.01, 0.005],
                                 n_steps=100000, time_limit_s=t_alloc * 0.7, t_start=t_start)
            try_update(sc, a_opt, f"restart#{r} s{strategy}")

            # Quick subgrad
            t_left = t_start + effective_budget - 20 - time.monotonic()
            if t_left > 3:
                sc2, a_opt2 = subgrad_descent(a_opt, n_steps=50000, lr_init=0.005,
                                              time_limit_s=min(t_left, restart_budget * 0.3),
                                              t_start=t_start, K=2)
                try_update(sc2, a_opt2, f"restart#{r} subgrad")

    # ============================================================
    # Phase 7: Final intensive polish (5% time)
    # ============================================================
    p7_time = effective_budget * 0.05
    print(f"\n{'='*60}")
    print(f"Phase 7: Final polish (budget={p7_time:.0f}s)")
    print(f"{'='*60}")

    if best_a is not None:
        final_deadline = t_start + effective_budget - 5.0
        remaining = final_deadline - time.monotonic()
        if remaining > 5:
            # Very high beta Adam
            n_cur = len(best_a)
            if n_cur <= 16384:
                sc, a_opt = adam_lse(best_a, [50000, 100000], [0.0003, 0.0001],
                                     n_steps=200000, time_limit_s=remaining * 0.6, t_start=t_start)
                try_update(sc, a_opt, "Final Adam high-beta")

                remaining = final_deadline - time.monotonic()
                if remaining > 3:
                    sc, a_opt = subgrad_descent(best_a, n_steps=100000, lr_init=0.003,
                                                time_limit_s=remaining, t_start=t_start, K=5)
                    try_update(sc, a_opt, "Final subgrad")

    # ============================================================
    # Final verification
    # ============================================================
    assert best_a is not None, "No solution found"
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
