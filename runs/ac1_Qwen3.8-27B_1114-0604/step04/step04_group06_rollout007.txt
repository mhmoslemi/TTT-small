import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def solve_lp_direction(f, M):
    """Solve the LP to find the best direction g.
    
    max sum(g)
    s.t. (f * g)[k] <= M for all k
         g >= 0
    """
    n = len(f)
    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Inequality constraints: A_ub @ g <= b_ub
    # For each k in [0, 2n-2]: sum_{j valid} f_j * g_{k-j} <= M
    # The coefficient of g_i in the constraint for k is f_{k-i} (if valid)
    
    # Build A_ub matrix (sparse would be better, but let's use dense for simplicity)
    # Actually for n=4096, A_ub would be (2n-1) x n = 8191 x 4096 ≈ 33M entries
    # That's about 264 MB in float64. Might be too much.
    # Let's use sparse.
    
    from scipy.sparse import csr_matrix
    import scipy.sparse as sp
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            # g_j appears in constraint k with coefficient f_{k-j}
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        return None

def lp_based_optimize(a, n_iter=100, time_limit_s=None, t_start=None):
    """LP-based direction finding optimization."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()  # Normalize to sum = 1
    
    best_score = compute_score(a)
    best_a = a.copy()
    
    for it in range(n_iter):
        if time_limit_s and t_start and (time.monotonic() - t_start) > time_limit_s:
            break
        
        c = fast_conv(a)
        M = float(c.max())
        
        # Solve LP
        g = solve_lp_direction(a, M)
        if g is None or g.sum() < 1e-10:
            break
        
        g = g / g.sum()  # Rescale to sum = 1
        
        # Line search for t
        best_t = 0.0
        best_sc = best_score
        for t in [0.01, 0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            h = (1-t)*a + t*g
            h = np.clip(h, 0, None)
            if h.sum() < 1e-10:
                continue
            h = h / h.sum()
            sc = compute_score(h)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        
        if best_t == 0.0:
            # No improvement found, try smaller t
            for t in [0.001, 0.005]:
                h = (1-t)*a + t*g
                h = np.clip(h, 0, None)
                if h.sum() < 1e-10:
                    continue
                h = h / h.sum()
                sc = compute_score(h)
                if sc < best_sc:
                    best_sc = sc
                    best_t = t
        
        if best_t == 0.0:
            break
        
        a = (1-best_t)*a + best_t*g
        a = np.clip(a, 0, None)
        a = a / a.sum()
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_score, best_a

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Linear cross-convolution a * b via FFT."""
    n = len(a)
    m = len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

Let me think about it from the perspective of the power spectrum. The autocorrelation b is the inverse DFT of |A(ω)|^2 (the power spectrum). For a non-negative sequence, the DFT A(ω) = sum_j a_j e^{-iωj}. We want |A(ω)| to be approximately constant. This is the "minimum PAPR" or "flat spectrum" design problem.

1. Start with a random non-negative sequence a
2. Compute A = DFT(a)
3. Modify |A| to be flat (set |A| = constant)
4. Compute a' = IDFT(A_modified)
5. Project a' onto the non-negative cone (set negative values to 0)
6. Repeat

This should converge to a sequence with approximately flat spectrum, hence approximately flat autocorrelation.
