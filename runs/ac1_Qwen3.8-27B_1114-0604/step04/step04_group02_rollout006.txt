Multi-phase optimizer for step function peak minimization.
Uses LSE-smoothed gradients, averaged subgradients over near-peaks,
mirror descent (multiplicative weights), and adaptive annealing.
"""

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def normalize(a):
    """Normalize to unit sum, clip to [0, 1000]."""
    a = np.clip(a, 0.0, 1000.0)
    S = a.sum()
    if S < 1e-12:
        return a
    return a / S

def lse_obj_grad(a, beta, top_k=None):
    """Compute LSE-smoothed objective and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    if top_k is not None and top_k < len(c):
        # Get top-k threshold
        sorted_c = np.sort(c)[::-1]
        threshold = sorted_c[min(top_k, len(c)) - 1]
        # Create masked version: only top-k terms contribute fully
        # Others are clipped to threshold
        c_masked = np.where(c >= threshold - 1e-10, c, threshold)
    else:
        c_masked = c
    
    c_max = float(c_masked.max())
    scaled = beta * (c_masked - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient via convolution of reversed a with p
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=20000, maxfun=100000, top_k=None):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = normalize(a0).copy()

        func = lambda x: lse_obj_grad(x, beta, top_k)
    else:
        func = lambda x: lse_obj_grad(x, beta)
    
    res = minimize(
        func, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = normalize(res.x)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def averaged_subgrad_step(a, n_iters=10000, f_lb=1.0, near_peak_tol=1.001):
    """Subgradient descent with averaged subgradient over near-peaks + Polyak step."""
    n = len(a)
    a = normalize(a).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc0 = best_sc

        S = float(a.sum())
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find near-peak indices
        threshold = M * (1.0 / near_peak_tol)
        near_peaks = np.where(c >= threshold)[0]
        
        # Average subgradient over near-peaks
        grad_M = np.zeros(n)
        for k in near_peaks:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                idx = np.arange(j_start, j_end + 1)
                grad_M[idx] += 2.0 * a[k - idx]
        grad_M /= len(near_peaks)
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (grad_M / S2 - 2.0 * M / S3)

        if gnorm < 1e-18:
            break

        step = min(step, 0.1 / (1.0 + 0.01 * it))

        a = normalize(a)
    
    return best_a

"""
Multi-phase optimizer for step function peak minimization.
Combines LSE-smoothed L-BFGS with adaptive beta annealing,
averaged subgradient over near-peaks, mirror descent (multiplicative
weights), and iterated local search with diverse perturbations.
Key improvements: adaptive reheating, frequency-domain perturbations,
top-k LSE for sharp optimization, and exact subgradient polishing.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Normalize to unit sum, clip to [0, 1000]."""
    a = np.clip(a, 0.0, 1000.0)
    S = a.sum()
    if S < 1e-12:
        return a
    return a / S


def lse_obj_grad(a, beta, top_k):
    """Compute LSE-smoothed objective and gradient. top_k=0 means no restriction."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    if top_k > 0 and top_k < len(c):
        sorted_c = np.sort(c)[::-1]
        threshold = sorted_c[top_k - 1]
        c_masked = np.where(c >= threshold - 1e-10, c, threshold)
    else:
        c_masked = c

    c_max = float(c_masked.max())
    scaled = beta * (c_masked - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


class LSEFunc:
    """Callable wrapper for LSE objective to use with scipy minimize."""
    def __init__(self, beta, top_k):
        self.beta = beta
        self.top_k = top_k

    def __call__(self, x):
        return lse_obj_grad(x, self.beta, self.top_k)


def run_lbfgs(a0, beta, maxiter, maxfun, top_k):
    """Run L-BFGS-B on the LSE-smoothed objective. Returns (score, array)."""
    n = len(a0)
    a0_norm = normalize(a0).copy()
    bounds = [(0.0, 1000.0)] * n
    func = LSEFunc(beta, top_k)

    res = minimize(
        func, a0_norm,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = normalize(res.x)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def averaged_subgrad(a, n_iters, f_lb, near_tol):
    """Subgradient with averaged direction over near-peaks + Polyak step."""
    n = len(a)
    a = normalize(a).copy()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        threshold = M / near_tol
        near_peaks = np.where(c >= threshold)[0]

        grad_M = np.zeros(n)
        for k in near_peaks:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                idx = np.arange(j_start, j_end + 1)
                grad_M[idx] += 2.0 * a[k - idx]
        grad_M /= max(1, len(near_peaks))

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (grad_M / S2 - 2.0 * M / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-20:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.05 / (1.0 + 0.005 * it))

        a = a - step * grad
        a = normalize(a)

    return best_a


def mirror_descent_step(a, n_iters, eta0, decay, beta0, beta_growth):
    """Mirror descent (multiplicative weights) with LSE gradient."""
    n = len(a)
    a = normalize(a).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0
    beta = beta0

    for it in range(n_iters):
        sc, grad = lse_obj_grad(a, beta, 0)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        log_a = np.log(np.maximum(a, 1e-30))
        log_a = log_a - eta * grad
        log_a = log_a - np.max(log_a)
        a = np.exp(log_a)
        a = a / a.sum()
        a = np.clip(a, 0.0, 1000.0)

        eta *= decay
        beta *= beta_growth

    return best_a

        sc, a_opt = run_lbfgs(best_a, beta, maxiter, maxfun, 0)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  [final beta={beta}] {sc:.10f}")

    # Top-k final
    for tk in [2, 3, 5]:
        if time.monotonic() > deadline - 8:
            break
        remaining = deadline - time.monotonic()
        if remaining < 8:
            break
        maxiter = int(min(30000, max(2000, remaining * 20)))
        sc_k, a_k = run_lbfgs(best_a, 500000, maxiter, maxiter * 3, tk)
        if sc_k < best_score:
            best_score = sc_k
            best_a = a_k.copy()
            print(f"  [final topk={tk}] {sc_k:.10f}")

    # Exact subgradient polish
    if time.monotonic() < deadline - 5:
        a_final = averaged_subgrad(best_a, 5000, 1.0, 1.0001)
        sc = compute_score(a_final)
        if sc < best_score:
            best_score = sc
            best_a = a_final.copy()
            print(f"  [final_subgrad] {sc:.10f}")

    # One more mirror descent pass
    if time.monotonic() < deadline - 5:
        a_md2 = mirror_descent_step(best_a, 2000, 0.005, 0.99995, 5000, 1.0005)
        sc = compute_score(a_md2)
        if sc < best_score:
            best_score = sc
            best_a = a_md2.copy()
            print(f"  [final_mirror] {sc:.10f}")

    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
