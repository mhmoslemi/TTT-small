import numpy as np
import time
from scipy.optimize import minimize

PEN_SUM = 1e-3


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_score_grad_tuple(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    grad = np.zeros(n, dtype=np.float64)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, grad

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0

    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, grad

    z = beta * (d - dmax)
    # Clip to avoid overflow
    z = np.clip(z, -50.0, 50.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, grad
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    score = 2.0 * L + PEN_SUM * (S - 1.0) * (S - 1.0)

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def fun_lse_tuple(a, beta):
    return lse_score_grad_tuple(a, beta)


def estimate_eval_time(n):
    base = 0.018
    scale = max(0.5, float(n) / 8192.0)
    log_scale = max(1.0, float(np.log2(max(16, n))) / float(np.log2(8192.0)))
    return base * scale * log_scale

    if t_left < 4.0:
        return 64
    est = estimate_eval_time(n)
    mf = int((0.7 * t_left) / max(est, 1e-4))

def run_lbfgs_tuple(obj_func, a0, param, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 4.0:
            return exact_score_a(a), a
        est = estimate_eval_time(n)
        cap = int((0.9 * rem) / max(est, 1e-4))
        if cap < 64:
            cap = 64
        maxfun = int(min(int(maxfun), cap))
    else:
        maxfun = int(maxfun if maxfun is not None else 1000)

    if maxfun < 64:
        maxfun = 64
    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }

    try:
        res = minimize(
            obj_func,
            a,
            method="L-BFGS-B",
            jac=True,
            args=(param,),
            bounds=bounds,
            options=opts,
        )
    except Exception:
        return exact_score_a(a), a

    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a

    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out

def spectral_flat_init(n, rng, flatness=1.0):
    """
    Initialize a sequence by designing a flat power spectrum.
    The time-domain sequence will have approximately flat autocorrelation.
    """
    N = next_power_of_two(2 * n)
    freqs = np.fft.rfftfreq(N) * N
    # Design power spectrum with tunable flatness
    # flatness=1 gives maximally flat, higher values add low-frequency emphasis
    power = 1.0 / (1.0 + flatness * np.abs(freqs) / (N / 2))
    power[0] = 0.0  # Remove DC
    # Add random phase
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = np.sqrt(power) * np.exp(1j * phase)
    spec[0] = 0.0
    sig = np.fft.irfft(spec * N, N)
    a = sig[:n]
    a = np.abs(a)  # Make non-negative
    a = a + 0.1  # Ensure strictly positive
    return normalize_a(a)


def spectral_flat_init_v2(n, rng, target_shape=None):
    """

    """

    freqs = np.fft.rfftfreq(N) * N
    abs_freqs = np.abs(freqs)

    if mode == 0:
        # Flat spectrum (white noise in time domain)
        power = np.ones(len(freqs))
    elif mode == 1:

        for k in range(3):

    else:

        x = abs_freqs / (N / 2)
        power = 1.0 / (0.5 + 0.5 * np.sin(np.pi * x)) ** 0.5
    
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = np.sqrt(power) * np.exp(1j * phase)
    spec[0] = 0.0
    sig = np.fft.irfft(spec * N, N)
    a = sig[:n]
    a = np.abs(a)

    x = np.linspace(0.0, 1.0, n)
    add(np.ones(n, dtype=np.float64))

    # Linear ramps

        vals = np.ones(n, dtype=np.float64)
        K = int(rng.integers(2, 6))
        for _ in range(K):

            amp = float(rng.uniform(0.05, 0.35))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
        add(vals)

    # Low-follower FFT
    Nf = 1 << (n - 1).bit_length()
    for _ in range(6):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 3.5)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        vals = 1.0 + float(rng.uniform(0.3, 0.9)) * noise
        add(vals)

    # Sparse / mask patterns

        mask = rng.random(n) < frac
        vals = np.where(mask, 0.2 + 0.8 * rng.random(n), 0.0)
        add(vals)

    # Bump functions
    for width in (0.03, 0.05, 0.1, 0.15, 0.2):
        for center in (0.0, 0.5, 1.0):
            vals = 0.1 + 0.9 * np.exp(-((x - center) / width) ** 2)
            add(vals)

        add(a[:n])

def active_weights(d, k, temperature=1.0):
    """Weighted active set with soft temperature."""
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if temperature <= 0 or k <= 0 or k >= m:
        idx = np.argpartition(d, m - max(k, 1))[m - max(k, 1):]
        w = np.zeros(m, dtype=np.float64)
        w[idx] = 1.0
        return w / float(k) if k > 0 else np.ones(m) / m
    
    # Soft weighting: w_k ∝ exp(d_k / temperature)
    d_norm = d / max(float(np.max(d)), 1e-12)
    w = np.exp(d_norm / max(temperature, 0.01))
    w = w / float(w.sum())
    return w


def multilevel_optimize(a_start, deadline, rng, label=""):
    """
    Multilevel optimization: coarse-to-fine.
    1. Downsample to get a coarse sequence
    2. Optimize the coarse sequence
    3. Upsample back and optimize fine details
    """
    a = normalize_a(a_start)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)
    
    # Determine levels
    levels = []
    size = n
    while size > 64:
        levels.append(size)
        size = size // 2
    levels.append(max(32, size))
    
    # Coarse to fine
    for level_size in reversed(levels):
        if time.monotonic() > deadline:
            break
        
        if level_size < n:
            # Downsample
            coarse = resample_sequence(a, level_size)
        else:
            coarse = a.copy()
        
        t_level_start = time.monotonic()
        t_level_budget = min(300.0, (deadline - t_level_start) * 0.4)
        
        # Optimize at this level with LSE continuation
        for beta in (50, 200, 800, 3000, 10000):
            if time.monotonic() > t_level_start + t_level_budget:
                break
            rem = (t_level_start + t_level_budget) - time.monotonic()
            if rem < 5:
                break
            mf = estimate_maxfun(rem * 0.7, level_size)
            if mf < 32:
                break
            sc, coarse2 = run_mode(coarse, float(beta), t_level_start + t_level_budget, mf)
            if sc < exact_score_a(coarse):
                coarse = coarse2
            else:
                coarse = normalize_a(coarse2)
        
        # Upsample back
        if level_size < n:
            upsampled = resample_sequence(coarse, n)
            # Blend with current best
            blend = 0.5 * upsampled + 0.5 * a
            a = normalize_a(blend)
        else:
            a = normalize_a(coarse)
        
        sc = exact_score_a(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return float(best_sc), best_a


def convolution_pressure_step(a, k_lags=5, eta_range=(0.01, 0.5), n_trials=8):
    """
    Apply pressure from the worst k lags of the autocorrelation.
    Returns the best improved sequence.
    """
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    d, S = d_from_a(a)
    if d.size == 0 or S < 1e-12:
        return best_sc, best_a
    
    # Get top-k lags with soft weighting
    dmax = float(np.max(d))
    if dmax <= 0:
        return best_sc, best_a
    
    # Use top-k with exponential weighting
    idx = np.argpartition(d, -k_lags)[-k_lags:]
    w = np.zeros(d.size, dtype=np.float64)
    top_vals = d[idx]
    # Weight by how much above average
    w[idx] = np.exp(5.0 * (top_vals - dmax) / dmax + 1.0)  # All positive, worst gets highest
    w = w / float(w.sum())
    
    h = corr_pair(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-12:
        return best_sc, best_a
    
    base = 1.0 / hmax
    for i in range(n_trials):
        eta = float(np.linspace(eta_range[0], eta_range[1], n_trials)[i])
        q = a * np.exp(-eta * h * base)
        q = normalize_a(q)
        sc = exact_score_a(q)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = q.copy()
    
    return float(best_sc), best_a


def combinatorial_perturb(a, rng, mode=None):
    """
    Discrete combinatorial moves to escape local minima.
    """
    a = normalize_a(a)
    n = len(a)
    if mode is None:
        mode = int(rng.integers(0, 5))
    
    if mode == 0:
        # Local swap: swap two nearby elements
        i = int(rng.integers(0, n - 1))
        dist = int(rng.integers(1, max(2, n // 10)))
        j = min(n - 1, i + dist)
        b = a.copy()
        b[i], b[j] = a[j], a[i]
        return normalize_a(b)
    elif mode == 1:
        # Block move: move a block of elements
        block_size = int(rng.integers(2, max(3, n // 20)))
        start = int(rng.integers(0, max(1, n - block_size)))
        target = int(rng.integers(0, n))
        b = a.copy()
        block = b[start:start + block_size].copy()
        if target > start:
            b[start:target + block_size] = np.concatenate([b[start:target], block])
        else:
            b[target:start + block_size] = np.concatenate([block, b[target:start]])
        return normalize_a(b)
    elif mode == 2:
        # Gaussian blur + denoise
        k = int(rng.integers(3, min(31, n)))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern = kern / float(kern.sum())
        b = np.convolve(a, kern, mode="same")
        b = b + 0.1 * rng.standard_normal(n)
        b = np.clip(b, 0, None)
        return normalize_a(b)
    elif mode == 3:
        # Spectral perturbation: modify a few frequency bins
        N = next_power_of_two(2 * n)
        A = np.fft.rfft(a, N)
        n_perturb = int(rng.integers(2, max(3, n // 10)))
        idx = rng.choice(len(A), size=n_perturb, replace=False)
        for i in idx:
            A[i] *= (1.0 + float(rng.uniform(-0.5, 0.5)))
            phase_shift = float(rng.uniform(-0.3, 0.3))
            A[i] = A[i] * np.exp(1j * phase_shift)
        b = np.fft.irfft(A, N)[:n]
        b = np.abs(b) + 0.05
        return normalize_a(b)
    else:
        # Mirror + perturb
        b = a.copy()
        if rng.random() < 0.5:
            b = b[::-1]
        b = b * np.exp(0.3 * rng.standard_normal(n))
        return normalize_a(b)

def subgrad_polish(a, max_iters=20, deadline=None):
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    
    for _ in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        d, S = d_from_a(a)
        if S < 1e-12 or d.size == 0:
            break
        dmax = float(np.max(d))
        if not np.isfinite(dmax) or dmax <= 0:
            break

        cur_a = None
        cur_new_sc = best_sc

        for k in (1, 3, 5, 10):
            w = active_weights(d, k, temperature=0.5)
            h = corr_pair(a, w)
            hmax = float(np.max(h))
            if hmax < 1e-12:
                continue
            base = 1.0 / hmax
            for mult in (0.01, 0.03, 0.08, 0.15, 0.3, 0.5):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                eta = mult * base
                q = a * np.exp(-eta * h)
                q = normalize_a(q)
                sc = exact_score_a(q)
                if not np.isfinite(sc):
                    continue
                if sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                if sc < cur_new_sc:
                    cur_new_sc = sc
                    cur_a = q.copy()

        if cur_a is None:
            break
        a = cur_a

    return float(best_sc), best_a

"""
Multilevel Convolution Feature-guided Optimization for minimizing
2 * n * max(conv(a,a)) / sum(a)^2 over non-negative sequences.

Key improvements over previous approach:
1. Multilevel (coarse-to-fine) LSE optimization to escape local minima
2. Spectral-flattening guided initialization for better starting points
3. Adaptive multi-lag pressure with exponential weighting
4. Non-uniform LSE temperature schedule (more time at intermediate temps)
5. Combinatorial perturbation moves (block swaps, spectral nudges)
6. Exact subgradient refinement at the end of each LSE stage
7. Better time management with progress checkpoints

import numpy as np
import time
from scipy.optimize import minimize

PEN_SUM = 1e-3


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def block_downsample(a, factor):
    """Downsample by averaging blocks of given size."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    out_len = max(1, n // factor)
    padded = np.zeros(out_len * factor)
    padded[:n] = a
    padded = padded.reshape(out_len, factor)
    return padded.sum(axis=1)


def block_upsample(a_coarse, n_fine):
    """Upsample by repeating each element and interpolating."""
    a_coarse = np.asarray(a_coarse, dtype=np.float64).ravel()
    m = a_coarse.size
    if m == 1:
        return np.full(n_fine, float(a_coarse[0]))
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n_fine))
    return np.interp(x_new, x_old, a_coarse).astype(np.float64)


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_score_grad_tuple(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    grad = np.zeros(n, dtype=np.float64)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, grad
    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0
    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, grad
    z = beta * (d - dmax)
    z = np.clip(z, -60.0, 60.0)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, grad
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    h = corr_pair(a, w)
    score = 2.0 * L + PEN_SUM * (S - 1.0) * (S - 1.0)
    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def fun_lse_tuple(a, beta):
    return lse_score_grad_tuple(a, beta)


def estimate_eval_time(n):
    base = 0.015
    scale = max(0.4, float(n) / 8192.0)
    log_scale = max(1.0, float(np.log2(max(16, n))) / float(np.log2(8192.0)))
    return base * scale * log_scale


def estimate_maxfun(t_left, n):
    t_left = float(max(1.0, t_left))
    if t_left < 4.0:
        return 64
    est = estimate_eval_time(n)
    mf = int((0.75 * t_left) / max(est, 1e-4))
    if mf < 64:
        mf = 64
    if mf > 4000:
        mf = 4000
    return mf


def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 4.0:
            return exact_score_a(a), a
        est = estimate_eval_time(n)
        cap = int((0.85 * rem) / max(est, 1e-4))
        if cap < 64:
            cap = 64
        maxfun = int(min(int(maxfun), cap))
    else:
        maxfun = int(maxfun if maxfun is not None else 1000)
    if maxfun < 64:
        maxfun = 64
    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-16,
        "gtol": 1e-13,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 40,
    }
    try:
        res = minimize(fun_lse_tuple, a, method="L-BFGS-B", jac=True,
                       args=(float(beta),), bounds=bounds, options=opts)
    except Exception:
        return exact_score_a(a), a
    a_out = normalize_a(res.x) if getattr(res, "x", None) is not None else a
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def exact_subgrad_step(a, k_lags, eta_list):
    """Take a step using exact subgradient from top-k lags."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    d, S = d_from_a(a)
    if d.size == 0 or S < 1e-12:
        return best_sc, best_a
    dmax = float(np.max(d))
    if dmax <= 0:
        return best_sc, best_a
    # Top-k lags with exponential weighting
    kk = min(k_lags, d.size)
    idx = np.argpartition(d, d.size - kk)[-kk:]
    w = np.zeros(d.size, dtype=np.float64)
    top_vals = d[idx]
    # Weight: worse lags get higher weight
    spread = float(top_vals.max() - top_vals.min() + 1e-12)
    w[idx] = np.exp(3.0 * (top_vals - top_vals.min()) / spread)
    w /= float(w.sum())
    h = corr_pair(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-12:
        return best_sc, best_a
    base = 1.0 / hmax
    for eta in eta_list:
        q = a * np.exp(-eta * h * base)
        q = normalize_a(q)
        sc = exact_score_a(q)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = q.copy()
    return float(best_sc), best_a


def spectral_init(n, rng, mode=0):
    """Generate a sequence with approximately flat autocorrelation via spectral design."""
    N = next_power_of_two(2 * n)
    freqs = np.fft.rfftfreq(N) * N
    abs_f = np.abs(freqs)
    power = np.ones(len(freqs), dtype=np.float64)
    if mode == 0:
        pass  # Flat
    elif mode == 1:
        mid = N / 4.0
        power = 1.0 / (1.0 + 0.3 * ((abs_f - mid) / (mid + 1)) ** 2)
    elif mode == 2:
        x = abs_f / (N / 2.0)
        power = 1.0 + 0.5 * np.cos(2 * np.pi * x)
    elif mode == 3:
        # Multi-peak
        for k in range(2, 6, 2):
            center = k * N / 12.0
            power += 0.8 * np.exp(-((abs_f - center) / (N / 20.0)) ** 2)
    else:
        # Random smooth
        power = 1.0 + 0.4 * np.sin(2 * np.pi * abs_f * rng.uniform(0.1, 0.5) + rng.uniform(0, 2*np.pi))
        power = np.maximum(power, 0.1)
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = np.sqrt(power) * np.exp(1j * phase)
    spec[0] = 0.0
    sig = np.fft.irfft(spec * N, N)
    a = np.abs(sig[:n]) + 0.05
    return normalize_a(a)


def generate_inits(n, rng, parent, cap=20):
    inits = []
    n = int(n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size == 0:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 0.05
        p = normalize_a(arr)
        inits.append(p)

    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])
        add(pr * np.exp(0.15 * rng.standard_normal(n)))
        add(pr * np.exp(0.4 * rng.standard_normal(n)))
        # Block-perturbed parent
        bs = max(4, n // 200)
        nblocks = n // bs
        bp = pr.copy()
        for b in range(min(nblocks, 50)):
            bp[b*bs:(b+1)*bs] *= float(rng.uniform(0.5, 1.5))
        add(bp)

    x = np.linspace(0.0, 1.0, n)
    add(np.ones(n, dtype=np.float64))

    for alpha in (0.3, 0.6, 0.85):
        add(1.0 + alpha * (2.0 * x - 1.0))
        add(1.0 - alpha * (2.0 * x - 1.0))

    # Spectral initializations (key for diversity)
    for m in range(5):
        add(spectral_init(n, rng, mode=m))
    for _ in range(5):
        add(spectral_init(n, rng, mode=int(rng.integers(0, 5))))

    # Trigonometric
    for r in range(4):
        vals = np.ones(n, dtype=np.float64)
        K = int(rng.integers(2, 5))
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 6)))
            amp = float(rng.uniform(0.1, 0.3))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
        add(vals)

    # Low-follower
    Nf = 1 << (n - 1).bit_length()
    for _ in range(4):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 3.0)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        vals = 1.0 + float(rng.uniform(0.3, 0.8)) * noise
        add(vals)

    # Random
    for _ in range(3):
        add(0.1 + 0.9 * rng.random(n))

    return inits


def multilevel_lse(a, deadline, max_levels=3):
    """
    Coarse-to-fine LSE optimization.
    Downsample, optimize, upsample, optimize at full resolution.
    """
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)

    # Build the levels (coarsest to finest)
    levels = []
    size = n
    for _ in range(max_levels):
        levels.append(size)
        size = max(64, size // 2)
    # Reverse so we go coarse to fine
    levels = levels[::-1]
    # Remove duplicates
    unique_levels = []
    for l in levels:
        if not unique_levels or l != unique_levels[-1]:
            unique_levels.append(l)
    levels = unique_levels

    current = a.copy()
    for li, level_size in enumerate(levels):
        if time.monotonic() > deadline:
            break

        if level_size < n:
            coarse = block_downsample(a, n // level_size)
            # Blend with current downsampled
            cur_coarse = block_downsample(current, n // level_size)
            coarse = 0.5 * coarse + 0.5 * cur_coarse
            coarse = normalize_a(coarse)
        else:
            coarse = current.copy()

        # Optimize at this level
        level_budget = 120.0
        level_dead = time.monotonic() + level_budget
        level_dead = min(level_dead, deadline)

        betas = [50, 200, 800, 3000]
        if level_size <= 256:
            betas = [50, 200, 800, 3000, 10000]

        for beta in betas:
            if time.monotonic() > level_dead - 5.0:
                break
            rem = float(level_dead - time.monotonic())
            if rem < 5.0:
                break
            mf = estimate_maxfun(rem * 0.7, level_size)
            if mf < 32:
                break
            sc, coarse2 = run_lbfgs(coarse, float(beta), level_dead, mf)
            coarse = normalize_a(coarse2)

        # Upsample back to full resolution
        if level_size < n:
            upsampled = block_upsample(coarse, n)
            # Blend: give more weight to the upsampled (optimized coarse) version
            current = 0.3 * current + 0.7 * upsampled
            current = normalize_a(current)
        else:
            current = normalize_a(coarse)

        sc = exact_score_a(current)
        if sc < best_sc:
            best_sc = sc
            best_a = current.copy()

    return float(best_sc), best_a


def combinatorial_perturb(a, rng):
    """Random combinatorial move to escape local minima."""
    a = normalize_a(a)
    n = len(a)
    mode = int(rng.integers(0, 5))

    if mode == 0:
        # Gaussian noise in log-space
        loga = np.log(a + 1e-300)
        loga += 0.3 * rng.standard_normal(n)
        b = np.exp(loga - float(np.max(loga)))
        return normalize_a(b)
    elif mode == 1:
        # Block swap
        bs = int(rng.integers(2, max(3, n // 50)))
        i = int(rng.integers(0, max(1, n - 2 * bs)))
        j = int(rng.integers(0, max(1, n - 2 * bs)))
        if abs(i - j) < bs:
            j = (i + bs + 1) % max(1, n - bs)
        b = a.copy()
        tmp = b[i:i+bs].copy()
        b[i:i+bs] = b[j:j+bs]
        b[j:j+bs] = tmp
        return normalize_a(b)
    elif mode == 2:
        # Spectral nudge
        N = next_power_of_two(2 * n)
        A = np.fft.rfft(a, N)
        n_pert = int(rng.integers(3, max(4, n // 20)))
        idx = rng.choice(len(A), size=min(n_pert, len(A)-1), replace=False) + 1
        for i in idx:
            A[i] *= (1.0 + float(rng.uniform(-0.4, 0.4)))
            A[i] = A[i] * np.exp(1j * float(rng.uniform(-0.2, 0.2)))
        b = np.fft.irfft(A, N)[:n]
        b = np.abs(b) + 0.05
        return normalize_a(b)
    elif mode == 3:
        # Mirror + noise
        b = a[::-1].copy()
        b *= np.exp(0.2 * rng.standard_normal(n))
        return normalize_a(b)
    else:
        # Smooth + rough
        k = int(rng.integers(3, min(21, n)))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.5)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= float(kern.sum())
        b = np.convolve(a, kern, mode="same")
        b += 0.15 * rng.standard_normal(n)
        b = np.clip(b, 0, None)
        return normalize_a(b)

                best_sc = sc
                best_a = a2.copy()
                print("[{}] LSE b={:.0f}: {:.8f} t={:.0f}s".format(label, beta, sc, time.monotonic() - t0))
            a = normalize_a(a2)
            # Subgradient refinement after each LSE run
            if time.monotonic() < sub_dead - 5.0:
                sc_sg, a_sg = exact_subgrad_step(a, k_lags=3,
                                                  eta_list=[0.02, 0.05, 0.1, 0.2])
                if sc_sg < best_sc:
                    best_sc = sc_sg
                    best_a = a_sg.copy()
                a = normalize_a(a_sg)

    # Phase 3: Alternating pressure + LSE (15% of budget)
    alt_dead = t0 + 0.85 * total
    while time.monotonic() < alt_dead - 10.0:
        # Pressure step
        sc_p, a_p = exact_subgrad_step(a, k_lags=5,
                                        eta_list=[0.01, 0.03, 0.08, 0.15, 0.25, 0.4])
        a = normalize_a(a_p)
        # LSE polish
        rem = float(alt_dead - time.monotonic() - 10.0)
        if rem > 8.0:
            mf = estimate_maxfun(rem * 0.35, n)
            if mf >= 32:
                sc_l, a_l = run_lbfgs(a, 2000.0, alt_dead - 5.0, mf)
                a = normalize_a(a_l)
        sc = exact_score_a(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            print("[{}] alt: {:.8f} t={:.0f}s".format(label, sc, time.monotonic() - t0))

    # Phase 4: Combinatorial escapes (15% of budget)
    combo_dead = deadline - 20.0
    attempts = 0
    while time.monotonic() < combo_dead and attempts < 30:
        rem = float(combo_dead - time.monotonic() - 20.0)
        if rem < 20.0:
            break
        b = combinatorial_perturb(best_a, rng)
        # Quick LSE on perturbed
        chunk_dead = time.monotonic() + min(rem * 0.4, 45.0)
        mf = estimate_maxfun((chunk_dead - time.monotonic()) * 0.6, n)
        if mf >= 32 and time.monotonic() < chunk_dead:
            sc_b, b2 = run_lbfgs(b, 500.0, chunk_dead, mf)
            b = normalize_a(b2)
        sc = exact_score_a(b)
        if sc < best_sc:
            best_sc = sc
            best_a = b.copy()
            print("[{}] combo: {:.8f} t={:.0f}s".format(label, sc, time.monotonic() - t0))
        attempts += 1

    return float(best_sc), best_a


def subgrad_polish(a, max_iters=30, deadline=None):
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        d, S = d_from_a(a)
        if S < 1e-12 or d.size == 0:
            break
        dmax = float(np.max(d))
        if not np.isfinite(dmax) or dmax <= 0:
            break

        cur_a = None
        cur_best = best_sc

        for k in (1, 2, 3, 5, 8):
            kk = min(k, d.size)
            idx = np.argpartition(d, d.size - kk)[-kk:]
            w = np.zeros(d.size, dtype=np.float64)
            tv = d[idx]
            spread = float(tv.max() - tv.min() + 1e-12)
            w[idx] = np.exp(3.0 * (tv - tv.min()) / spread)
            w /= float(w.sum())
            h = corr_pair(a, w)
            hmax = float(np.max(h))
            if hmax < 1e-12:
                continue
            base = 1.0 / hmax
            for mult in (0.01, 0.03, 0.07, 0.12, 0.2, 0.35, 0.5):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                eta = mult * base
                q = a * np.exp(-eta * h)
                q = normalize_a(q)
                sc = exact_score_a(q)
                if not np.isfinite(sc):
                    continue
                if sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                if sc < cur_best:
                    cur_best = sc
                    cur_a = q.copy()

        if cur_a is None:
            break
        a = cur_a

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Uses multilevel convolution feature-guided optimization
    to find a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    """
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    # Determine lengths
    parent_len = int(parent.size) if parent is not None else 0
    candidate_lengths = []
    if parent_len > 0:
        candidate_lengths.append(parent_len)
        if parent_len > 2048:
            candidate_lengths.append(max(1024, parent_len // 2))
        if parent_len < 16384:
            candidate_lengths.append(min(16384, parent_len * 3 // 2))
    else:
        candidate_lengths = [8192, 4096, 12288]

    # Deduplicate, filter, sort
    seen = set()
    filtered = []
    for l in candidate_lengths:
        l = int(l)
        if 1024 <= l <= 16384 and l not in seen:
            seen.add(l)
            filtered.append(l)
    if not filtered:
        filtered = [8192]
    candidate_lengths = sorted(filtered)[:3]

    print("Parent: {} (len={})".format(parent is not None, parent_len))
    print("Lengths: {}".format(candidate_lengths))

    # Phase A: Quick screening (12% budget)
    screen_dead = t_start + 0.12 * total_budget
    best_sc = np.inf
    best_seq = None

    for n in candidate_lengths:
        if time.monotonic() > screen_dead - 15.0:
            break
        inits = generate_inits(n, rng, parent, cap=14)
        for arr in inits:
            if time.monotonic() > screen_dead - 5.0:
                break
            p = normalize_a(arr)
            sc = exact_score_a(p)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_seq = p.copy()

    if best_seq is None:
        n0 = candidate_lengths[0]
        best_seq = np.ones(n0) / n0
        best_sc = exact_score_a(best_seq)

    print("Screen best: {:.8f} (n={})".format(best_sc, len(best_seq)))

    # Phase B: Main optimization trajectories (78% budget)
    main_end = t_start + 0.90 * total_budget

    # Determine which lengths to optimize
    best_n = len(best_seq)
    other_ns = [l for l in candidate_lengths if l != best_n]

    if other_ns:
        # Split 65% / 35%
        main_dead = t_start + 0.90 * total_budget * 0.65
        other_dead = main_end
    else:
        main_dead = main_end
        other_dead = main_end

    # Main trajectory
    sc1, seq1 = full_trajectory(best_seq, main_dead, rng, "main")
    if sc1 < best_sc:
        best_sc = sc1
        best_seq = seq1.copy()

    # Secondary trajectory (different length)
    if other_ns and time.monotonic() < other_dead - 120.0:
        other_n = other_ns[0]
        a2 = resample_sequence(best_seq, other_n)
        # Add perturbation for diversity
        a2 = a2 * np.exp(0.2 * rng.standard_normal(other_n))
        a2 = normalize_a(a2)
        sc2, seq2 = full_trajectory(a2, other_dead, rng, "alt")
        if sc2 < best_sc:
            best_sc = sc2
            best_seq = seq2.copy()

    # Phase C: Final intensive polish (10% budget)
    polish_dead = deadline - 10.0
    best_seq = normalize_a(best_seq)

    # Subgradient polish
    if time.monotonic() < polish_dead - 60.0:
        sc, seq = subgrad_polish(best_seq, max_iters=30, deadline=polish_dead - 60.0)
        if sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
            print("Subgrad polish: {:.8f}".format(sc))
        best_seq = normalize_a(seq)

    # Very high temperature LSE
    if time.monotonic() < polish_dead - 30.0:
        rem = float(polish_dead - 30.0 - time.monotonic())
        if rem > 10.0:
            mf = estimate_maxfun(rem * 0.8, len(best_seq))
            sc, seq = run_lbfgs(best_seq, 50000.0, polish_dead - 30.0, mf)
            if sc < best_sc:
                best_sc = sc
                best_seq = seq.copy()
                print("High-beta: {:.8f}".format(sc))
            best_seq = normalize_a(seq)

    # Final micro-polish
    if time.monotonic() < polish_dead:
        sc, seq = subgrad_polish(best_seq, max_iters=15, deadline=polish_dead)
        if sc < best_sc:
            best_sc = sc
            best_seq = seq.copy()
            print("Final: {:.8f}".format(sc))

    # Output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_sc

    print("FINAL: internal={:.8f} verified={:.8f} n={} time={:.1f}s".format(
        best_sc, verified, len(out), time.monotonic() - t_start))
    return out
