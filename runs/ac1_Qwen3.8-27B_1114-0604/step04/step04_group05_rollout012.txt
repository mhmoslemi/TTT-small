import numpy as np
from scipy.optimize import minimize, linprog
import time
import math

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Compute the score 2n*max(conv)/sum^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def conv_peak_info(a):
    """Return convolution, max value, argmax, sum."""
    n = len(a)
    c = fast_conv(a)
    M = float(c.max())
    k_star = int(c.argmax())
    S = float(a.sum())
    return c, k_star, M, S

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -500, 500))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + np.log(exp_scaled.sum()) / beta
    # r_i = sum_k p_k * a[k-i] for valid k
    # Using FFT cross-correlation
    N = 1 << (3 * n - 2 - 1).bit_length()
    A_fft = np.fft.rfft(a, N)
    P_fft = np.fft.rfft(p, N)
    cross = np.fft.irfft(P_fft * np.conj(A_fft), N)
    r = cross[:n]
    S2 = S * S
    grad = 4.0 * n * (r / S2 - L / (S2 * S))
    score = 2.0 * n * L / S2
    return score, grad

def solve_lp_direction(a, active_set, M):
    """
    Solve LP: max sum(g) subject to (a*g)[k] <= M for k in active_set, g >= 0.
    Returns the direction g (normalized to sum=1) or None.
    """
    n = len(a)
    m = len(active_set)
    
    # Build constraint matrix A (m x n), sparse representation
    # A[i, j] = a[j] if 0 <= active_set[i] - j < n
    # For HiGHS, we can pass dense if m*n is not too large
    # For n=7000, m=300: 2.1M entries, which is ~16MB for float64. OK.
    
    A = np.zeros((m, n))
    b = np.full(m, M)
    
    for i, k in enumerate(active_set):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            A[i, js:je+1] = a[js:je+1]
    
    c_obj = -np.ones(n)  # minimize -sum(g)
    
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})
    
    if res.status != 0:
        return None
    
    g = res.x
    S_g = g.sum()
    if S_g < 1e-10:
        return None
    
    g = g / S_g
    return g

def line_search_mix(a, g, t_max=0.5, n_evals=15):
    """
    Find best t in [0, t_max] for a_new = (1-t)*a + t*g.
    Uses golden section search.
    """
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, t_max
    
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    
    f1 = compute_score((1.0 - c1) * a + c1 * g)
    f2 = compute_score((1.0 - c2) * a + c2 * g)
    
    for _ in range(n_evals - 2):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = compute_score((1.0 - c1) * a + c1 * g)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = compute_score((1.0 - c2) * a + c2 * g)
    
    best_t = c1 if f1 < f2 else c2
    best_f = min(f1, f2)
    return best_f, best_t

def lp_iteration(a, n_iters=50, active_frac=0.95, max_active=300):
    """
    Iterative LP-based direction finding.
    """
    a = a.astype(np.float64).copy()
    n = len(a)
    S0 = a.sum()
    best_score = compute_score(a)
    best_a = a.copy()
    
    for it in range(n_iters):
        c, k_star, M, S = conv_peak_info(a)
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Find active set
        threshold = M * active_frac
        active_mask = c >= threshold
        active_indices = np.where(active_mask)[0]
        
        # Limit to top max_active
        if len(active_indices) > max_active:
            vals = c[active_indices]
            top_idx = np.argsort(vals)[::-1][:max_active]
            active_indices = active_indices[top_idx]
        
        # Solve LP
        g = solve_lp_direction(a, active_indices.tolist(), M)
        if g is None:
            break
        
        # Line search for mixing parameter
        f_opt, t_opt = line_search_mix(a, g, t_max=0.3, n_evals=12)
        
        if f_opt < best_score:
            a = (1.0 - t_opt) * a + t_opt * g
            # Renormalize
            a = np.clip(a, 0.0, 1000.0)
            if a.sum() > 1e-12:
                a = a * (S0 / a.sum())
        else:
            # Try smaller t
            for t_try in [0.01, 0.05, 0.1]:
                a_try = (1.0 - t_try) * a + t_try * g
                a_try = np.clip(a_try, 0.0, 1000.0)
                if a_try.sum() > 1e-12:
                    a_try = a_try * (S0 / a_try.sum())
                f_try = compute_score(a_try)
                if f_try < best_score:
                    a = a_try
                    break
            else:
                break
    
    return best_score, best_a

def adam_optimize(a_init, lr=0.005, n_steps=50000, beta_lse=300,
                  time_budget=None, t_start=None):
    """Adam optimizer with LSE-smoothed gradient."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    S0 = a.sum()
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf
    
    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break
        
        lr_t = lr / (1.0 + 1e-5 * t)
        score, grad = lse_score_grad(a, beta_lse)
        
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)
        
        s = a.sum()
        if s > 1e-12:
            a = a * (S0 / s)
        
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a

def run_lbfgs(a0, beta, maxiter=3000, maxfun=30000, time_budget=None, t_start=None):
    """L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def obj_and_grad(x):
        return lse_score_grad(x, beta)
    
    if time_budget is not None and t_start is not None:
        deadline = t_start + time_budget
        def timed_obj(x):
            if time.monotonic() > deadline:
                raise TimeoutError
            return lse_score_grad(x, beta)
        try:
            res = minimize(timed_obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                          options={'ftol': 1e-14, 'gtol': 1e-11,
                                   'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
        except (TimeoutError, Exception):
            return compute_score(a0), a0
    else:
        res = minimize(obj_and_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                      options={'ftol': 1e-14, 'gtol': 1e-11,
                               'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out

def spectral_perturb(a, rng, strength=0.15):
    """Perturb in DFT domain."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    amp = 1.0 + strength * rng.standard_normal(len(A))
    amp = np.clip(amp, 0.3, 2.0)
    phase = strength * 0.5 * rng.standard_normal(len(A))
    A_new = A * amp * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() < 1e-10:
        return a.copy()
    return a_new * (a.sum() / a_new.sum())

def pairwise_coord_descent(a, n_sweeps=2, top_pairs=20, time_budget=60.0, t_start=None):
    """Optimize pairs of nearby coordinates."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (t_start is not None) else np.inf
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        c, k_star, M, S = conv_peak_info(a)
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score
        
        # Find gradient (subgradient at peak)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        sg = np.zeros(n)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je+1] = a[idx]
        
        S2 = S * S
        grad = 2.0 * n * (2.0 * sg / S2 - 2.0 * M / (S2 * S))
        
        # Select top coordinates, then form pairs of neighbors
        order = np.argsort(np.abs(grad))[::-1][:top_pairs * 3]
        
        # Form pairs: for each top coordinate, pair with nearest top neighbor
        pairs = []
        used = set()
        for j in order:
            j = int(j)
            if j in used:
                continue
            # Find nearest top neighbor
            for k in order:
                k = int(k)
                if k != j and k not in used and 0 < abs(k - j) <= 50:
                    pairs.append((j, k))
                    used.add(j)
                    used.add(k)
                    break
        
        for (j, k) in pairs[:top_pairs]:
            if time.monotonic() > deadline:
                break
            # 2D optimization on (a[j], a[k])
            a_j0 = a[j]
            a_k0 = a[k]
            
            def obj_2d(params):
                a[j] = max(0.0, params[0])
                a[k] = max(0.0, params[1])
                return compute_score(a)
            
            res = minimize(obj_2d, [a_j0, a_k0], method='Nelder-Mead',
                          options={'maxiter': 200, 'xatol': 1e-10, 'fatol': 1e-12})
            
            a[j] = max(0.0, res.x[0])
            a[k] = max(0.0, res.x[1])
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
    
    return best_score, a
