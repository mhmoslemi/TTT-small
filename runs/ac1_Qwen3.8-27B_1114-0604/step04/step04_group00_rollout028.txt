    score = 2n * LSE(c, beta) / S^2
    """
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -700.0, 700.0)
    exp_scaled = np.exp(scaled)
    sum_exp = float(exp_scaled.sum())
    if sum_exp < 1e-300:
        return 1e10, np.zeros(n)
    L = c_max + float(np.log(sum_exp)) / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / sum_exp  # softmax weights, length 2n-1

    # grad_j = 2*(conv(a,p)[j] + conv(a,p_rev)[2n-2-j]) / S^2 - 4n*L*a_j/S^3
    p_rev = p[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    A = np.fft.rfft(x, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    PR = np.fft.rfft(p_rev, M2_fft)
    conv_ap = np.fft.irfft(A * P, M2_fft)
    conv_apr = np.fft.irfft(A * PR, M2_fft)

    first_part = conv_ap[:n]
    second_part = conv_apr[n - 1:2 * n - 1][::-1]

    grad = 2.0 * (first_part + second_part) / (S * S)
    grad -= (4.0 * n * L / (S * S * S)) * x

    return score, grad


def run_lbfgs_lse(a0, beta, maxiter=4000, maxfun=60000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0 = np.ones(n) / n
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.maximum(res.x, 0.0)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0
    else:
        a_out = a_out / S_out
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_shave_reweight(a, n_iters=200, alpha_init=0.002, decay=0.999):
    """
    Iterative reweighting to reduce the peak of the autocorrelation.
    At each step, scale down elements that contribute most to the peak.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    alpha = alpha_init

    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        M = float(c[k_star])
        S = float(a.sum())
        sc = 2.0 * n * M / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Compute contribution of each a_j to c[k_star]
        # c[k_star] = sum_j a_j * a[k_star - j]
        # contrib_j = a[k_star - j] (if valid) + a[j - k_star] (if valid)
        j_idx = np.arange(n)
        k_minus_j = k_star - j_idx
        j_minus_k = j_idx - k_star

        contrib = np.zeros(n)
        valid1 = (k_minus_j >= 0) & (k_minus_j < n)
        valid2 = (j_minus_k >= 0) & (j_minus_k < n)
        contrib[valid1] += a[k_minus_j[valid1]]
        contrib[valid2] += a[j_minus_k[valid2]]

        max_contrib = float(contrib.max())
        if max_contrib < 1e-15:
            break

        # Scale down high-contributing elements
        scale = 1.0 - alpha * (contrib / max_contrib)
        scale = np.maximum(scale, 0.0)
        a = a * scale
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        alpha *= decay

    return best_a


def coord_descent_peak(a, n_sweeps=3, n_top=40):
    """
    Coordinate descent: for each top-contributing element at the peak,
    try scaling it up or down to reduce the max.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))

        # Find top contributing elements
        j_idx = np.arange(n)
        k_minus_j = k_star - j_idx
        j_minus_k = j_idx - k_star
        contrib = np.zeros(n)
        valid1 = (k_minus_j >= 0) & (k_minus_j < n)
        valid2 = (j_minus_k >= 0) & (j_minus_k < n)
        contrib[valid1] += a[k_minus_j[valid1]]
        contrib[valid2] += a[j_minus_k[valid2]]

        top_count = min(n_top, n)
        top_idx = np.argsort(contrib)[::-1][:top_count]

        for j in top_idx:
            a_old = a[j]
            if a_old < 1e-12:
                continue
            cur_sc = compute_score(a)
            best_val = a_old
            best_local = cur_sc
            for factor in [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.5]:
                a[j] = a_old * factor
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old
                    continue
                a_norm = a / S_tmp
                sc = compute_score(a_norm)
                if sc < best_local:
                    best_local = sc
                    best_val = a_old * factor
                a[j] = a_old

            a[j] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lse_fienup_hybrid(a, n_iters=150, beta_iter_start=50, beta_iter_end=500):
    """
    Hybrid Fienup + LSE: alternate between frequency-domain flattening
    and time-domain LSE gradient step.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        beta = beta_iter_start + (beta_iter_end - beta_iter_start) * progress

        # Fienup step: flatten power spectrum
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        thresh = P_mean * (1.2 + 0.3 * (1.0 - progress))
        above = P > thresh
        if above.any():
            scale = np.ones_like(P)
            scale[above] = np.sqrt(thresh / P[above]) * (0.8 + 0.2 * progress)
            A_mod = A * scale
            a_f = np.fft.irfft(A_mod, M_fft)[:n]
            a_f = np.maximum(a_f, 0.0)
            S_f = a_f.sum()
            if S_f > 1e-12:
                a_f = a_f / S_f
                # Blend with current
                t = 0.3 + 0.5 * progress
                a = (1.0 - t) * a + t * a_f
                a = np.maximum(a, 0.0)
                S = a.sum()
                if S > 1e-12:
                    a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_error_reduction(a, n_iters=200):
    """
    Fienup with error reduction: compute error in power spectrum,
    map back to time domain, and apply as correction.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    step = 0.1

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_target = float(P.mean()) * np.ones_like(P)

        # Error in power spectrum
        E = P - P_target
        # Map error to amplitude correction
        # dP = d(|A|^2) = 2*Re(A * conj(dA))
        # We want to reduce P where P > target
        # dA = -step * A * conj(E) / (2*|A|^2 + eps)
        eps = 1e-10 * P.mean()
        dA = -step * A * np.conj(E) / (2.0 * P + eps)
        A_new = A + dA
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        step = max(step * 0.999, 0.01)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def build_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    if parent is not None and len(parent) == n:
        inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(np.pi * x))  # half-cosine bump
    inits.append(np.maximum(0.5 + 0.3 * np.cos(4.0 * np.pi * x), 0.01))
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(0.2 + 0.6 * x * (1.0 - x) * 4.0)  # parabolic
    inits.append(np.maximum(0.3 + 0.5 * np.cos(2.0 * np.pi * 2 * x), 0.01))
    # Exponential decay from center
    center = n // 2
    dist = np.abs(np.arange(n) - center) / center
    inits.append(np.exp(-dist * 2.0) * 0.5)
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Determine working length
    if parent is not None:
        n_work = len(parent)
        start_a = parent / parent.sum()
    else:
        n_work = 4096
        start_a = np.ones(n_work) * 0.5

    print(f"Working length: n={n_work}")
    print(f"Parent: {'yes' if parent is not None else 'no'}")
    sc_start = compute_score(start_a)
    print(f"Initial score: {sc_start:.8f}")
    update_best(sc_start, start_a, "START")

    # ================================================================
    # Phase 1: LSE-L-BFGS annealing (50% of time)
    # ================================================================
    phase1_end = t_start + 0.50 * (total_budget - 30.0)
    print(f"\n=== Phase 1: LSE-L-BFGS Annealing (deadline t={phase1_end - t_start:.0f}s) ===")

    beta_schedule = [10, 30, 100, 300, 1000, 3000, 10000, 30000, 100000]

    for i, beta in enumerate(beta_schedule):
        if time.monotonic() > phase1_end - 10:
            break
        print(f"  Beta={beta}...", end=" ", flush=True)
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs_lse(best_a, beta, maxiter=3000, maxfun=40000)
        dt = time.monotonic() - t0
        print(f"score={sc:.8f}, time={dt:.1f}s")
        update_best(sc, a_opt, f"LBFGS b={beta}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ================================================================
    # Phase 2: Peak shaving (15% of time)
    # ================================================================
    phase2_end = t_start + 0.65 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Peak Shaving (deadline t={phase2_end - t_start:.0f}s) ===")

    # Iterative reweighting
    a_ps = peak_shave_reweight(best_a, n_iters=300, alpha_init=0.003)
    sc = compute_score(a_ps)
    update_best(sc, a_ps, "PEAK-SHAVE")

    # L-BFGS polish after peak shaving
    if time.monotonic() < phase2_end - 15:
        sc, a_opt = run_lbfgs_lse(a_ps, 10000, maxiter=2000, maxfun=30000)
        update_best(sc, a_opt, "PS+LBFGS")

    # Coordinate descent
    if time.monotonic() < phase2_end - 10:
        a_cd = coord_descent_peak(best_a, n_sweeps=4, n_top=50)
        sc = compute_score(a_cd)
        update_best(sc, a_cd, "COORD-DESCENT")

    # Another peak shave round
    if time.monotonic() < phase2_end - 10:
        a_ps2 = peak_shave_reweight(best_a, n_iters=200, alpha_init=0.001)
        sc = compute_score(a_ps2)
        update_best(sc, a_ps2, "PEAK-SHAVE-2")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ================================================================
    # Phase 3: Perturbation + mini-annealing (25% of time)
    # ================================================================
    phase3_end = t_start + 0.90 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Perturbation Cycles (deadline t={phase3_end - t_start:.0f}s) ===")

    n_cycles = 3
    for cyc in range(n_cycles):
        if time.monotonic() > phase3_end - 30:
            break
        print(f"\n  Cycle {cyc + 1}/{n_cycles}:", end=" ", flush=True)

        # Create perturbed version
        if cyc == 0:
            # Small perturbation
            a_pert = best_a * (1.0 + 0.02 * rng.standard_normal(n_work))
        elif cyc == 1:
            # Medium perturbation + clip some values to zero
            a_pert = best_a * (1.0 + 0.05 * rng.standard_normal(n_work))
            mask = rng.random(n_work) < 0.1
            a_pert[mask] = 0.0
        else:
            # Larger perturbation with structural change
            a_pert = best_a.copy()
            # Flip a random segment
            seg_start = rng.integers(0, n_work // 2)
            seg_len = rng.integers(n_work // 8, n_work // 4)
            a_pert[seg_start:seg_start + seg_len] *= 0.5
            a_pert = a_pert * (1.0 + 0.03 * rng.standard_normal(n_work))

        a_pert = np.maximum(a_pert, 0.0)
        S_pert = a_pert.sum()
        if S_pert < 1e-6:
            continue
        a_pert = a_pert / S_pert

        # Mini-annealing on perturbed version
        mini_betas = [100, 1000, 10000]
        for beta in mini_betas:
            if time.monotonic() > phase3_end - 15:
                break
            sc, a_opt = run_lbfgs_lse(a_pert, beta, maxiter=2000, maxfun=25000)
            a_pert = a_opt  # chain the results
            update_best(sc, a_opt, f"PERT-{cyc} b={beta}")

        # Peak shave the result
        if time.monotonic() < phase3_end - 10:
            a_ps = peak_shave_reweight(a_pert, n_iters=100, alpha_init=0.002)
            sc = compute_score(a_ps)
            update_best(sc, a_ps, f"PERT-{cyc}-PS")

        print(f"done. Best so far: {best_score:.8f}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ================================================================
    # Phase 4: Diverse initializations (try different structures)
    # ================================================================
    if time.monotonic() < deadline - 60:
        print(f"\n=== Phase 4: Diverse Inits ===")
        inits = build_diverse_inits(n_work, rng, None)
        for i, a0 in enumerate(inits):
            if time.monotonic() > deadline - 30:
                break
            a0 = np.maximum(a0, 0.0)
            S0 = a0.sum()
            if S0 < 1e-6:
                continue
            a0 = a0 / S0

            # Quick L-BFGS with medium beta
            sc, a_opt = run_lbfgs_lse(a0, 500, maxiter=1500, maxfun=20000)
            update_best(sc, a_opt, f"INIT-{i} b=500")

            # If promising, do more
            if sc < best_score + 0.01 and time.monotonic() < deadline - 20:
                sc2, a_opt2 = run_lbfgs_lse(a_opt, 5000, maxiter=2000, maxfun=30000)
                update_best(sc2, a_opt2, f"INIT-{i} b=5000")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ================================================================
    # Phase 5: Final refinement (remaining time)
    # ================================================================
    print(f"\n=== Phase 5: Final Refinement ===")

    if time.monotonic() < deadline - 20:
        # High-beta L-BFGS
        sc, a_opt = run_lbfgs_lse(best_a, 200000, maxiter=3000, maxfun=50000)
        update_best(sc, a_opt, "FINAL-LBFGS-200k")

    if time.monotonic() < deadline - 15:
        # Peak shave
        a_ps = peak_shave_reweight(best_a, n_iters=200, alpha_init=0.001)
        sc = compute_score(a_ps)
        update_best(sc, a_ps, "FINAL-PEAKSHAVE")

    if time.monotonic() < deadline - 10:
        # Final L-BFGS polish
        sc, a_opt = run_lbfgs_lse(best_a, 500000, maxiter=2000, maxfun=30000)
        update_best(sc, a_opt, "FINAL-POLISH")

    # ================================================================
    # Try different length (if time permits)
    # ================================================================
    if time.monotonic() < deadline - 60:
        print(f"\n=== Phase 6: Try n={n_work * 2 // 1} ===")
        n_try = n_work  # same length, but try from a different start
        # Actually try a slightly different length
        n_alt = n_work + 16 if n_work % 32 == 0 else n_work
        if n_alt != n_work:
            # Resample best to new length
            x_old = np.linspace(0, 1, n_work)
            x_new = np.linspace(0, 1, n_alt)
            a_alt = np.interp(x_new, x_old, best_a)
            a_alt = np.maximum(a_alt, 0.0)
            S_alt = a_alt.sum()
            if S_alt > 1e-6:
                a_alt = a_alt / S_alt
                sc, a_opt = run_lbfgs_lse(a_alt, 10000, maxiter=2000, maxfun=30000)
                update_best(sc, a_opt, f"N-ALT n={n_alt}")

    # ================================================================
    # Final output
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
