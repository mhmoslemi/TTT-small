```python
"""
Multi-phase optimizer for step function peak minimization.
Uses LSE-smoothed objective with aggressive beta schedule,
subgradient polishing, and iterated local search.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -500.0))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + np.log(exp_scaled.sum()) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient computation
    # dL/d(a_j) = 2 * r[j] where r[j] = sum_m a[m] * p[j+m]
    p_rev = p[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    A_fft = np.fft.rfft(a, M2_fft)
    PR_fft = np.fft.rfft(p_rev, M2_fft)
    conv_res = np.fft.irfft(A_fft * PR_fft, M2_fft)
    r = conv_res[n-1:2*n-1][::-1]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (2.0 * r / S2 - 2.0 * L / S3)
    
    return score, grad


def lse_obj_grad_topk(a, beta, top_k):
    """LSE-smoothed score with top-k restriction."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    # Get top-k values
    if top_k < len(c):
        threshold = np.partition(c, -top_k)[-top_k]
        c_masked = c.copy()
        c_masked[c_masked < threshold] = threshold  # clip below threshold
    
    c_max = float(np.max(c_masked))
    scaled = beta * (c_masked - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -500.0))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + np.log(exp_scaled.sum()) / beta
    score = 2.0 * n * L / (S * S)
    
    p_rev = p[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    A_fft = np.fft.rfft(a, M2_fft)
    PR_fft = np.fft.rfft(p_rev, M2_fft)
    conv_res = np.fft.irfft(A_fft * PR_fft, M2_fft)
    r = conv_res[n-1:2*n-1][::-1]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (2.0 * r / S2 - 2.0 * L / S3)
    
    return score, grad


def run_lbfgs(a0, beta, maxiter=30000, maxfun=100000, top_k=None):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    if top_k is not None:
        func = lambda x: lse_obj_grad_topk(x, beta, top_k)
    else:
        func = lambda x: lse_obj_grad(x, beta)
    
    res = minimize(
        func, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-30, 'gtol': 1e-18,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 500}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    Sc = compute_score(a_out)
    return Sc, a_out


def subgrad_polish(a, n_iters=2000, f_lb=1.0):
    """Subgradient method with Polyak step size for the exact max objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S_a = float(a.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (S_a * S_a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of 2n*max(c)/S^2 at the peak
        # d(max(c))/d(a_j) = 2*a_{k_star - j} (for valid j)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg[idx] = 2.0 * a[k_star - idx]
        
        S2 = S_a * S_a
        S3 = S2 * S_a
        grad = (2.0 * n) * (sg / S2 - 2.0 * Mak / S3)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        # Polyak step size
        step = max(1e-10, (sc - f_lb) / (gnorm * gnorm + 1e-30))
        step = min(step, 0.5)
        
        a_new = a - step * grad
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
    
    return best_a


def peak_shave(a, n_iters=100):
    """Targeted peak shaving: reduce the top peaks specifically."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S_a = float(a.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (S_a * S_a)
        
        if sc >= best_sc:
            # Try a smaller step
            pass
        
        # Identify contributors to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        seg = a[j_start:j_end+1]
        contrib = seg[::-1]  # contrib[i] = a[k_star - (j_start+i)] = a[k_star - j_start - i]
        # Actually: c[k_star] = sum_j a[j] * a[k_star-j]
        # For j in [j_start, j_end]: term is a[j] * a[k_star-j]
        # contrib[j - j_start] = a[j] * a[k_star - j]
        
        # Find the top contributors
        n_top = min(20, len(contrib))
        top_local = np.argsort(contrib)[::-1][:n_top]
        top_global = j_start + top_local
        
        # Reduce top contributors and add to bottom
        n_bottom = min(20, len(contrib) - n_top)
        bottom_local = np.argsort(contrib)[:-n_bottom] if n_bottom > 0 else np.array([0])
        bottom_global = j_start + bottom_local
        
        total_reduction = 0.0
        for j in top_global:
            old_val = a[j]
            new_val = old_val * 0.7  # reduce by 30%
            a[j] = new_val
            total_reduction += old_val - new_val
        
        # Redistribute to bottom contributors (and spread across all if needed)
        if total_reduction > 1e-15:
            if n_bottom > 0:
                per = total_reduction / n_bottom
                for j in bottom_global:
                    a[j] += per
            else:
                # Spread across all
                per = total_reduction / n
                a += per
        
        a = np.clip(a, 0.0, 1000.0)
        Sa = a.sum()
        if Sa > 1e-12:
            a = a / Sa
        
        sc_new = compute_score(a)
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a.copy()
        else:
            a = best_a.copy()  # revert if it got worse
    
    return best_a


def perturb(a, rng, scale=0.05):
    """Random perturbation of a."""
    n = len(a)
    strategy = rng.integers(0, 4)
    
    if strategy == 0:
        # Gaussian noise
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        # Sinusoidal
        freq = 1.0 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        a_p = a * (1.0 + scale * np.cos(2.0 * np.pi * freq * np.arange(n)/n + phase))
    elif strategy == 2:
        # Block perturbation
        n_blocks = 2 + int(rng.integers(0, 6))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    else:
        # Sparse perturbation
        n_sel = n // 10
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 2.0 * (rng.random(n_sel) - 0.5))
    
    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p
```

```python
"""
Multi-phase optimizer for step function peak minimization.
Combines LSE-smoothed gradient descent with aggressive beta scheduling,
subgradient polishing, peak-aware perturbations, and iterated local search.
Key improvements: adaptive top-k smoothing, Nesterov-accelerated subgradient,
spectral-shaping initialization, and multi-length exploration.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -500.0))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + np.log(exp_scaled.sum()) / beta
    score = 2.0 * n * L / (S * S)

    p_rev = p[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    A_fft = np.fft.rfft(a, M2_fft)
    PR_fft = np.fft.rfft(p_rev, M2_fft)
    conv_res = np.fft.irfft(A_fft * PR_fft, M2_fft)
    r = conv_res[n - 1:2 * n - 1][::-1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (2.0 * r / S2 - 2.0 * L / S3)

    return score, grad


def lse_obj_grad_topk(a, beta, top_k):
    """LSE-smoothed score with top-k restriction for smoother optimization."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    if top_k < len(c):
        threshold = np.partition(c, len(c) - top_k)[len(c) - top_k]
        c_eff = np.maximum(c, threshold)
    else:
        c_eff = c

    c_max = float(np.max(c_eff))
    scaled = beta * (c_eff - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -500.0))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + np.log(exp_scaled.sum()) / beta
    score = 2.0 * n * L / (S * S)

    p_rev = p[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    A_fft = np.fft.rfft(a, M2_fft)
    PR_fft = np.fft.rfft(p_rev, M2_fft)
    conv_res = np.fft.irfft(A_fft * PR_fft, M2_fft)
    r = conv_res[n - 1:2 * n - 1][::-1]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (2.0 * r / S2 - 2.0 * L / S3)

    return score, grad


def run_lbfgs(a0, beta, maxiter=30000, maxfun=100000, top_k=None):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    if top_k is not None:
        func = _make_lbfgs_func_topk(beta, top_k)
    else:
        func = _make_lbfgs_func(beta)

    res = minimize(
        func, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-30, 'gtol': 1e-18,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 500}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    Sc = compute_score(a_out)
    if not np.isfinite(Sc):
        return np.inf, a0
    return Sc, a_out


def _make_lbfgs_func(beta):
    def func(x):
        return lse_obj_grad(x, beta)
    return func


def _make_lbfgs_func_topk(beta, top_k):
    def func(x):
        return lse_obj_grad_topk(x, beta, top_k)
    return func


def nesterov_subgrad(a, n_iters=5000, f_lb=1.0, verbose=False):
    """Nesterov-accelerated subgradient on the exact objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = max(1e-10, (sc - f_lb) / (gnorm * gnorm + 1e-30))
        step = min(step, 0.3)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)

        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    return best_a


def subgrad_polish(a, n_iters=2000, f_lb=1.0):
    """Heavy-ball momentum subgradient with decaying step."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    step0 = 0.05
    decay = 0.9995

    for it in range(n_iters):
        c = fast_conv(a)
        S_a = float(a.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (S_a * S_a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * a[k_star - idx]

        S2 = S_a * S_a
        S3 = S2 * S_a
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = step0 * (decay ** it)
        step = min(step, 2.0 * max(0.0, sc - f_lb) / (gnorm + 1e-30))

        v = 0.95 * v - step * grad
        a_new = a + v
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa

    return best_a


def peak_shave(a, n_sweeps=3, n_top=30):
    """Targeted coordinate descent on peak contributors."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib) // 2)
        bot_count = min(n_top, len(contrib) - top_count)
        if bot_count < 1:
            bot_count = 1

        top_local = np.argsort(contrib)[::-1][:top_count]
        bot_local = np.argsort(contrib)[:bot_count]

        total_red = 0.0
        for j in top_local:
            old = a[j_start + j]
            new = old * 0.6
            a[j_start + j] = new
            total_red += old - new

        if total_red > 1e-15:
            per = total_red / bot_count
            for j in bot_local:
                a[j_start + j] += per

        a = np.clip(a, 0.0, 1000.0)
        Sa = a.sum()
        if Sa > 1e-12:
            a = a / Sa

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        else:
            a = best_a.copy()

    return best_a


def perturb(a, rng, scale=0.03):
    """Generate a perturbed version of a."""
    n = len(a)
    strategy = rng.integers(0, 5)
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 15.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (0.5 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 8))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.5 * (rng.random() - 0.5))
    elif strategy == 3:
        # Peak-targeted: reduce top contributors in the dominant peak
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(5, n // 100)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.15 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            deficit = a.sum() - S
            bot_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            per = deficit / len(bot_local)
            for li in bot_local:
                a_p[j_start + li] += per
    else:
        n_sel = max(10, n // 8)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 3.0 * (rng.random(n_sel) - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(0.1 + 0.9 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.05, np.cos(np.pi * x)))
    # Broadband-like: combination of multiple frequencies
    a_bb = np.zeros(n)
    for k in range(1, 6):
        a_bb += np.abs(np.cos(2.0 * np.pi * k * x + rng.random() * np.pi))
    inits.append(a_bb / a_bb.sum())
    return inits


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a non-negative sequence minimizing 2n*max(a*a)/(sum(a))^2.

    Multi-phase strategy:
    Phase 1: LSE optimization with increasing beta schedule (55% time)
    Phase 2: Subgradient polishing + Nesterov (10% time)
    Phase 3: Iterated local search with peak-aware perturbations (20% time)
    Phase 4: Final high-precision refinement (10% time)
    Phase 5: Multi-length exploration (5% time)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(30.0, float(budget_s)) - 30.0
    deadline = t_start + total_budget

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        parent_norm = parent / parent.sum()
        sc_p = compute_score(parent_norm)
        print(f"Parent: n={len(parent)}, score={sc_p:.10f}")
        report(sc_p, parent_norm, "parent")
        n_work = len(parent)
    else:
        n_work = 8192

    print(f"Working length: {n_work}, Time budget: {total_budget:.0f}s")

    # Initialize working sequence
    if parent is not None and len(parent) == n_work:
        a_cur = parent_norm.copy()
    else:
        a_cur = np.interp(np.linspace(0, 1, n_work),
                          np.linspace(0, 1, len(best_a)), best_a)
        a_cur = np.clip(a_cur, 0, 1000.0)
        S = a_cur.sum()
        if S > 0:
            a_cur = a_cur / S

    # ============================================================
    # Phase 1: LSE Optimization with increasing beta (55% time)
    # ============================================================
    phase1_end = t_start + 0.55 * total_budget
    print(f"\n--- Phase 1: LSE Optimization (deadline {phase1_end - t_start:.0f}s) ---")

    beta_schedule = [20, 50, 100, 300, 1000, 3000, 10000, 30000, 100000]

    for beta in beta_schedule:
        if time.monotonic() > phase1_end:
            break
        remaining = phase1_end - time.monotonic()
        if remaining < 45:
            break

        print(f"  Beta={beta}, remaining={remaining:.0f}s, current_best={best_score:.8f}")
        maxiter = int(min(80000, max(5000, remaining * 40)))
        maxfun = int(min(300000, max(20000, remaining * 150)))

        sc, a_opt = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=maxfun)
        improved = report(sc, a_opt, f"LSE b={beta}")
        if improved:
            a_cur = a_opt.copy()

        # Try top-k variant for smoother landscape
        if time.monotonic() < phase1_end - 30:
            for tk in [2, 3]:
                if time.monotonic() > phase1_end - 20:
                    break
                sc_k, a_k = run_lbfgs(a_cur, beta, maxiter=maxiter // 3,
                                       maxfun=maxfun // 3, top_k=tk)
                report(sc_k, a_k, f"LSE b={beta} tk={tk}")
                if sc_k < best_score:
                    a_cur = a_k.copy()

    print(f"After Phase 1: best = {best_score:.10f}")

    # ============================================================
    # Phase 2: Subgradient polishing (10% time)
    # ============================================================
    phase2_end = t_start + 0.65 * total_budget
    print(f"\n--- Phase 2: Subgradient Polishing ---")

    # Nesterov
    a_ns = nesterov_subgrad(a_cur, n_iters=5000, f_lb=max(1.0, best_score * 0.99))
    sc_ns = compute_score(a_ns)
    report(sc_ns, a_ns, "nesterov")
    if sc_ns < best_score:
        a_cur = a_ns.copy()

    # Heavy-ball
    if time.monotonic() < phase2_end:
        a_hb = subgrad_polish(a_cur, n_iters=3000, f_lb=max(1.0, best_score * 0.99))
        sc_hb = compute_score(a_hb)
        report(sc_hb, a_hb, "heavyball")
        if sc_hb < best_score:
            a_cur = a_hb.copy()

    # Peak shaving
    if time.monotonic() < phase2_end:
        a_sh = peak_shave(a_cur, n_sweeps=5, n_top=50)
        sc_sh = compute_score(a_sh)
        report(sc_sh, a_sh, "peak_shave")
        if sc_sh < best_score:
            a_cur = a_sh.copy()

    print(f"After Phase 2: best = {best_score:.10f}")

    # ============================================================
    # Phase 3: Iterated Local Search (20% time)
    # ============================================================
    phase3_end = t_start + 0.85 * total_budget
    print(f"\n--- Phase 3: Iterated Local Search ---")

    n_trials = 40
    for trial in range(n_trials):
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 45:
            break

        # Peak-aware perturbation
        scale = 0.015 + 0.04 * rng.random()
        a_pert = perturb(best_a, rng, scale=scale)

        # Optimize with random beta
        beta = 500 + int(20000 * rng.random())
        maxiter = int(min(40000, max(3000, remaining * 25)))
        maxfun = int(min(150000, max(10000, remaining * 80)))

        sc, a_opt = run_lbfgs(a_pert, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"ILS #{trial}")

        # Follow-up with higher beta if improved
        if sc < best_score + 0.001 and time.monotonic() < phase3_end - 30:
            beta2 = min(beta * 5, 200000)
            sc2, a_opt2 = run_lbfgs(a_opt, beta2, maxiter=maxiter // 2,
                                     maxfun=maxfun // 2)
            report(sc2, a_opt2, f"ILS #{trial} refine")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ============================================================
    # Phase 4: Final high-precision refinement (10% time)
    # ============================================================
    phase4_end = t_start + 0.95 * total_budget
    print(f"\n--- Phase 4: Final Refinement ---")

    # Very high beta LSE
    for beta in [100000, 300000, 500000]:
        if time.monotonic() > phase4_end - 10:
            break
        maxiter = int(min(60000, max(10000, (phase4_end - time.monotonic()) * 50)))
        maxfun = int(min(200000, max(50000, (phase4_end - time.monotonic()) * 200)))
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"final b={beta}")

    # Top-k final
    if time.monotonic() < phase4_end - 10:
        for tk in [1, 2]:
            if time.monotonic() > phase4_end - 5:
                break
            sc_k, a_k = run_lbfgs(best_a, 200000, maxiter=30000,
                                   maxfun=100000, top_k=tk)
            report(sc_k, a_k, f"final tk={tk}")

    # Final subgradient
    if time.monotonic() < phase4_end - 5:
        a_sg = subgrad_polish(best_a, n_iters=2000, f_lb=max(1.0, best_score * 0.999))
        sc_sg = compute_score(a_sg)
        report(sc_sg, a_sg, "final subgrad")

    # Final peak shave
    if time.monotonic() < phase4_end - 5:
        a_sh = peak_shave(best_a, n_sweeps=3, n_top=30)
        sc_sh = compute_score(a_sh)
        report(sc_sh, a_sh, "final shave")

    # ============================================================
    # Phase 5: Multi-length exploration (5% time)
    # ============================================================
    if time.monotonic() < deadline - 90 and parent is not None:
        print(f"\n--- Phase 5: Length Exploration ---")
        n_refs = [int(n_work * 1.25), int(n_work * 1.5), int(n_work * 0.75)]
        for n_try in n_refs:
            if time.monotonic() > deadline - 45:
                break
            if n_try < 1024 or n_try > 40000:
                continue
            print(f"  Trying n={n_try}")
            a_interp = np.interp(np.linspace(0, 1, n_try),
                                  np.linspace(0, 1, len(best_a)), best_a)
            a_interp = np.clip(a_interp, 0, 1000.0)
            S = a_interp.sum()
            if S > 0:
                a_interp = a_interp / S

            for beta in [200, 2000, 20000]:
                if time.monotonic() > deadline - 20:
                    break
                maxiter = int(min(20000, max(2000, (deadline - time.monotonic()) * 20)))
                sc, a_opt = run_lbfgs(a_interp, beta, maxiter=maxiter,
                                       maxfun=maxiter * 3)
                report(sc, a_opt, f"n={n_try} b={beta}")

    # ============================================================
    # Final output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
```
