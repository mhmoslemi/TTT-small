"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
- FFT-based exact scoring
- Aggressive multiplicative subgradient with line search
- Adaptive active-set size and soft weighting
- LSE-LBFGS refinement at key points
- Perturbation restarts to escape local minima
"""

import numpy as np
import time
from scipy.optimize import minimize

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    return a * (float(target) / S)

def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)

def exact_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (S * S))

def lse_smooth(d, beta):
    dmax = float(np.max(d))
    z = beta * (d - dmax)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        return dmax
    return dmax + np.log(s) / beta

def lse_obj_grad(a, beta, pen=1e-4):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    dmax = float(np.max(d))
    
    z = beta * (d - dmax)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        return 1e10, np.zeros(n)
    w = e / s
    
    L = dmax + np.log(s) / beta
    LDS = float(np.dot(w, d))
    
    h = corr_pair(a, w)
    score = 2.0 * L + pen * (S - 1.0) ** 2
    
    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * pen * (S - 1.0)
    
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad

def soft_pressure(a, width):
    """Compute pressure using a Gaussian-weighted active set."""
    c = fast_self_conv(a)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(n)
    
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    dmax = float(np.max(d))
    if dmax < 1e-15:
        return np.zeros(n)
    
    # Soft weight based on distance from max
    w = np.exp(-width * (dmax - d) / dmax)
    w = np.maximum(w, 0)
    S_w = float(w.sum())
    if S_w < 1e-15:
        # Fall back to hard top-1
        idx = int(np.argmax(d))
        w = np.zeros_like(d)
        w[idx] = 1.0
    else:
        w = w / S_w
    
    return corr_pair(a, w)

def subgrad_step_with_line_search(a, eta_range=None, deadline=None):
    """One subgradient step with line search over eta values."""
    if eta_range is None:
        eta_range = [0.01, 0.03, 0.1, 0.2, 0.4, 0.8, 1.2, 2.0]
    
    h = soft_pressure(a, width=5.0)
    hmax = float(np.max(h))
    if hmax < 1e-15:
        h = soft_pressure(a, width=1.0)
        hmax = float(np.max(h))
        if hmax < 1e-15:
            return a.copy(), exact_score(a)
    
    cur_score = exact_score(a)
    best_seq = a.copy()
    best_score = cur_score
    
    for eta in eta_range:
        if deadline is not None and time.monotonic() > deadline:
            break
        q = a * np.exp(-eta * h / hmax)
        q = normalize_a(q)
        sc = exact_score(q)
        if sc < best_score:
            best_score = sc
            best_seq = q.copy()
    
    return best_seq, best_score

def subgrad_step_topk(a, k, etas=None, deadline=None):
    """Subgradient step using top-k peaks."""
    if etas is None:
        etas = [0.01, 0.05, 0.1, 0.2, 0.4, 0.8]
    
    c = fast_self_conv(a)
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return a.copy(), exact_score(a)
    
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    dmax = float(np.max(d))
    if dmax < 1e-15:
        return a.copy(), exact_score(a)
    
    # Top-k weights
    if k <= 1:
        idx = int(np.argmax(d))
        w = np.zeros_like(d)
        w[idx] = 1.0
    else:
        idx = np.argpartition(d, -k)[-k:]
        w = np.zeros_like(d)
        w[idx] = 1.0
        w = w / k
    
    h = corr_pair(a, w)
    hmax = float(np.max(h))
    if hmax < 1e-15:
        return a.copy(), exact_score(a)
    
    cur_score = exact_score(a)
    best_seq = a.copy()
    best_score = cur_score
    
    for eta in etas:
        if deadline is not None and time.monotonic() > deadline:
            break
        q = a * np.exp(-eta * h / hmax)
        q = normalize_a(q)
        sc = exact_score(q)
        if sc < best_score:
            best_score = sc
            best_seq = q.copy()
    
    return best_seq, best_score

def lse_lbfgs(a, beta, deadline, maxfun=500):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    a = normalize_a(a)
    n = len(a)
    
    def obj(x):
        x = np.maximum(x, 0.0)  # Project for safety
        return lse_obj_grad(float(x).sum() > 0 and x or a, beta)
    
    # Actually use the function directly
    def fun(x):
        x_clip = np.clip(x, 0, None)
        S = float(x_clip.sum())
        if S < 1e-10:
            return 1e10, np.zeros_like(x)
        x_norm = x_clip / S
        val, grad = lse_obj_grad(x_norm, beta)
        # Chain rule: d/dx f(x/S) * 1/S + gradient of normalization
        # Actually, let's just work with the normalized variable
        return val, grad / S
    
    def fun2(x):
        # Work directly, with the penalty handling the normalization
        return lse_obj_grad(x, beta)
    
    bounds = [(0.0, 1.0)] * n
    opts = {
        "maxiter": max(20, maxfun // 3),
        "maxfun": maxfun,
        "ftol": 1e-14,
        "gtol": 1e-10,
        "maxls": 30,
    }
    
    try:
        res = minimize(fun2, a, method="L-BFGS-B", jac=True, bounds=bounds, options=opts)
        if res.x is not None:
            a_out = normalize_a(res.x)
        else:
            a_out = a
    except Exception:
        a_out = a
    
    return a_out, exact_score(a_out)

def perturb(a, rng, strength=0.3):
    """Random perturbation of the sequence."""
    n = len(a)
    mode = int(rng.integers(0, 5))
    
    if mode == 0:
        q = a * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.3))
        q = (1 - t) * a + t * (np.ones(n) / n)
    elif mode == 2:
        q = a[::-1].copy()
    elif mode == 3:
        # Smooth noise in log domain
        log_a = np.log(a + 1e-300)
        noise = rng.standard_normal(n)
        # Smooth the noise
        k = min(21, max(3, n // 100))
        if k % 2 == 0:
            k += 1
        if k >= 3 and k < n:
            sig = max(1.0, k / 4.0)
            kern = np.exp(-((np.arange(k) - k//2)**2) / (2*sig*sig))
            kern /= kern.sum()
            noise = np.convolve(noise, kern, mode='same')
        log_a = log_a + strength * 0.5 * noise
        q = np.exp(log_a)
    else:
        # Block perturbation
        nb = int(rng.integers(4, min(32, n)))
        block_len = max(1, n // nb)
        multipliers = 0.5 + rng.random(nb)
        q = np.repeat(multipliers, block_len * nb)[:n]
        q = a * q
    
    return normalize_a(q)

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
- FFT-based exact scoring at each step
- Aggressive multiplicative subgradient with adaptive line search
- Soft Gaussian-weighted pressure that widens when stuck
- Periodic LSE-LBFGS refinement interleaved with subgradient
- Perturbation restarts to escape local minima
- Focus on the parent sequence length for maximum efficiency
"""

import numpy as np
import time
from scipy.optimize import minimize


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    """Full linear self-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    """h_j = sum_k w_k * a_{k-j}, the middle slice of conv(a[::-1], w)."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    return a * (float(target) / S)


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def exact_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (S * S))


def compute_d(a):
    """d = n*conv(a,a)/sum(a)^2, so score = 2*max(d)."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def soft_pressure(a, width=3.0):
    """Pressure using Gaussian-weighted active set around the max of d."""
    d, S = compute_d(a)
    n = len(a)
    dmax = float(np.max(d))
    if dmax < 1e-15:
        return np.zeros(n)
    w = np.exp(-width * (dmax - d) / dmax)
    w = np.maximum(w, 0.0)
    S_w = float(w.sum())
    if S_w < 1e-15:
        idx = int(np.argmax(d))
        w = np.zeros_like(d)
        w[idx] = 1.0
    else:
        w = w / S_w
    return corr_pair(a, w)


def topk_pressure(a, k=3):
    """Pressure using hard top-k active set."""
    d, S = compute_d(a)
    n = len(a)
    dmax = float(np.max(d))
    if dmax < 1e-15:
        return np.zeros(n)
    if k <= 1:
        idx = int(np.argmax(d))
        w = np.zeros_like(d)
        w[idx] = 1.0
    else:
        kv = min(k, len(d))
        idx = np.argpartition(d, len(d) - kv)[len(d) - kv:]
        w = np.zeros_like(d)
        w[idx] = 1.0 / kv
    return corr_pair(a, w)


def subgrad_line_search(a, pressure_fn, etas=None, deadline=None):
    """
    One subgradient step with line search.
    pressure_fn(a) -> h array (the pressure direction)
    Returns (best_sequence, best_score).
    """
    if etas is None:
        etas = [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4, 0.7, 1.0, 1.5]

    h = pressure_fn(a)
    hmax = float(np.max(np.abs(h)))
    if hmax < 1e-15:
        h = topk_pressure(a, 1)
        hmax = float(np.max(np.abs(h)))
        if hmax < 1e-15:
            return a.copy(), exact_score(a)

    cur_score = exact_score(a)
    best_seq = a.copy()
    best_score = cur_score

    for eta in etas:
        if deadline is not None and time.monotonic() > deadline:
            break
        q = a * np.exp(-eta * h / hmax)
        q = normalize_a(q)
        sc = exact_score(q)
        if sc < best_score:
            best_score = sc
            best_seq = q.copy()

    return best_seq, best_score


def lse_obj_grad(a, beta, pen=1e-4):
    """LSE-smoothed objective and gradient."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)

    d, S = compute_d(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (d - dmax)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / s

    L = dmax + np.log(s) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    score = 2.0 * L + pen * (S - 1.0) ** 2

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * pen * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def run_lse_lbfgs(a, beta, deadline, maxfun=200):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    a = normalize_a(a)
    n = len(a)

    def fun(x):
        x_clip = np.maximum(x, 0.0)
        S = float(x_clip.sum())
        if S < 1e-10:
            return 1e10, np.zeros_like(x)
        val, grad = lse_obj_grad(x_clip, beta)
        return val, grad

    bounds = [(0.0, 2.0)] * n
    maxiter = max(10, maxfun // 2)
    opts = {
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "ftol": 1e-14,
        "gtol": 1e-11,
        "maxls": 25,
    }

    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True, bounds=bounds, options=opts)
        if getattr(res, "x", None) is not None:
            a_out = normalize_a(res.x)
        else:
            a_out = a
    except Exception:
        a_out = a

    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return a_out, float(sc)


def smooth_noise(n, rng, strength):
    noise = rng.standard_normal(n)
    k = min(31, max(3, int(n) // 100))
    if k % 2 == 0:
        k += 1
    if k >= 3 and k < n:
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = np.convolve(noise, kern, mode="same")
    noise -= float(noise.mean())
    return noise * strength


def perturb(a, rng, strength=0.3):
    n = len(a)
    mode = int(rng.integers(0, 6))

    if mode == 0:
        q = a * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.03, 0.25))
        q = (1.0 - t) * a + t * (np.ones(n) / n)
    elif mode == 2:
        q = a[::-1].copy()
        q = q * np.exp(0.08 * rng.standard_normal(n))
    elif mode == 3:
        log_a = np.log(a + 1e-300)
        log_a = log_a + smooth_noise(n, rng, strength * 0.5)
        q = np.exp(log_a)
    elif mode == 4:
        h = topk_pressure(a, 4)
        hmax = float(np.max(h))
        if hmax < 1e-12:
            q = a.copy()
        else:
            q = a * np.exp(-strength * h / hmax)
    else:
        nb = int(rng.integers(4, min(24, n)))
        block_len = max(1, n // nb)
        multipliers = 0.5 + rng.random(nb)
        q = np.repeat(multipliers, block_len * nb)[:n]
        q = a * q

    return normalize_a(q)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.

    Strategy:
    1. Start from parent (height_sequence_1) or uniform
    2. Main loop: interleaved subgradient (with adaptive pressure width and line search)
       and periodic LSE-LBFGS refinement
    3. Perturbation restarts when stuck
    4. Final high-temperature LSE polish
    """
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 18.0
    rng = np.random.default_rng(seed)

    # Get parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    if parent is not None:
        a = normalize_a(parent)
        n = len(a)
        print("Using parent, n={}".format(n))
    else:
        n = 8192
        a = np.ones(n, dtype=np.float64) / n
        print("No parent, uniform n={}".format(n))

    best_a = a.copy()
    best_score = exact_score(a)
    print("Initial score: {:.10f}".format(best_score))

    # Main optimization loop
    stuck = 0
    width = 3.0
    lse_counter = 0
    total_iters = 0
    last_lse_time = 0.0
    last_print = t_start

    while time.monotonic() < deadline - 60.0:
        total_iters += 1
        rem = deadline - 60.0 - time.monotonic()
        if rem < 3.0:
            break

        # Decide what to do this iteration
        frac_done = (time.monotonic() - t_start) / (float(budget_s) - 20.0)

        # Periodic LSE-LBFGS refinement (roughly every 200 subgradient steps
        # or every few seconds, whichever is appropriate)
        time_since_lse = time.monotonic() - last_lse_time
        should_do_lse = False
        lse_beta = 100.0

        if total_iters % 150 == 0 and time_since_lse > 1.0:
            should_do_lse = True
            # Temperature increases with progress

                lse_beta = 50.0
            elif frac_done < 0.5:
                lse_beta = 200.0
            elif frac_done < 0.7:
                lse_beta = 1000.0
            elif frac_done < 0.85:
                lse_beta = 5000.0
            else:
                lse_beta = 20000.0

        if should_do_lse:
            maxfun = min(300, max(30, int(rem / 5.0)))
            a2, sc = run_lse_lbfgs(a, lse_beta, deadline - 30.0, maxfun=maxfun)
            last_lse_time = time.monotonic()
            if sc < best_score:
                best_score = sc
                best_a = a2.copy()
                stuck = 0
                print("  LSE beta={:.0f} score={:.10f} iter={}".format(lse_beta, sc, total_iters))
            a = normalize_a(a2)
            continue

        # Subgradient step with adaptive pressure
        if stuck < 30:
            pfn = lambda x: soft_pressure(x, width)
        elif stuck < 80:
            k_val = min(2 ** (stuck // 20), 16)
            pfn = lambda x: topk_pressure(x, k_val)
        else:
            # Perturb and use soft pressure
            a = perturb(a, rng, strength=float(rng.uniform(0.1, 0.5)))
            stuck = max(0, stuck - 20)
            pfn = lambda x: soft_pressure(x, width)

        step_deadline = time.monotonic() + min(rem, 3.0)
        seq, sc = subgrad_line_search(a, pfn, deadline=step_deadline)

        if sc < best_score:
            improvement = best_score - sc
            best_score = sc
            best_a = seq.copy()
            stuck = 0
        else:
            stuck += 1
            # Adapt width
            if stuck % 10 == 0:
                width = min(width * 1.3, 20.0)
                if stuck > 100:
                    width = 3.0  # Reset

        # Always move toward the improved sequence if it's not much worse

            a = seq
        else:
            a = best_a.copy()
            # Small random nudge to escape
            a = normalize_a(a * (1.0 + 0.005 * rng.standard_normal(n)))

        # Print progress
        if time.monotonic() - last_print > 10.0:
            print("  iter={} score={:.10f} stuck={} width={:.1f} t={:.0f}s".format(
                total_iters, best_score, stuck, width, time.monotonic() - t_start))
            last_print = time.monotonic()

    print("Main loop done: {} iters, score={:.10f}".format(total_iters, best_score))

    # === Final refinement phase ===
    a = best_a.copy()

    # High-temperature LSE runs
    for beta in [5000, 10000, 20000, 50000]:
        if time.monotonic() > deadline - 15.0:
            break
        rem = deadline - 15.0 - time.monotonic()
        if rem < 10.0:
            break
        maxfun = min(400, max(30, int(rem / 3.0)))
        a2, sc = run_lse_lbfgs(a, beta, deadline - 10.0, maxfun=maxfun)
        if sc < best_score:
            best_score = sc
            best_a = a2.copy()
            print("  Final LSE beta={}".format(beta), "score={:.10f}".format(sc))
        a = normalize_a(a2)

    # Final subgradient polish
    a = best_a.copy()
    for k in [1, 2, 3, 5, 8]:
        if time.monotonic() > deadline - 5.0:
            break

            if time.monotonic() > deadline - 5.0:
                break
            pfn = lambda x: topk_pressure(x, k)
            seq, sc = subgrad_line_search(a, pfn, etas=[0.003, 0.008, 0.02, 0.05, 0.1, 0.2])
            if sc < best_score:
                best_score = sc
                best_a = seq.copy()
            a = seq if sc < exact_score(a) else a

            seq, sc = subgrad_line_search(a, pfn, etas=[0.003, 0.01, 0.03, 0.1, 0.3])
            if sc < best_score:
                best_score = sc
                best_a = seq.copy()
            a = seq if sc < exact_score(a) else a

    print("After polish: score={:.10f}".format(best_score))

    # === Output ===
    out_arr = np.asarray(best_a, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.10f} verified={:.10f} n={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
