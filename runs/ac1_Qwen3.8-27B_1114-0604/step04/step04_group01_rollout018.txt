"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Strategy:
- Phase 1: Fast multiplicative subgradient (mirror descent) for rapid basin finding
- Phase 2: LSE-smoothed objective with Adam optimizer for refinement
- Phase 3: High-beta LSE polish + subgradient cleanup
- Multi-start with different lengths and initializations

import numpy as np
import time
import math


def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fft_conv(a):
    """Full linear self-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def fft_correlate(a, w):
    """
    Compute h_j = sum_k w_k * a_{(k-j) mod valid} for j=0..n-1.
    This is the middle slice of conv(a[::-1], w).
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    m = int(w.size)
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    M = n + m - 1
    N = next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize(a):
    """Clip to non-negative and normalize to sum=1."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-15:
        return np.full(a.size, 1.0 / max(1, a.size), dtype=np.float64)
    return a / S


def exact_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fft_conv(a)
    cmax = float(np.max(c))
    return float(2.0 * n * cmax / (S * S))


def lse_obj_grad(a, beta):
    """
    LSE-smoothed objective and gradient.
    F(a) = 2*LSE_beta(d) + pen*(sum(a)-1)^2
    d_k = n * (a*a)_k / S^2
    """
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-10:
        return 1e10, np.zeros(n, dtype=np.float64)

    c = fft_conv(a)
    d = n * c / (S * S)
    dmax = float(np.max(d))
    if not np.isfinite(dmax):
        return 1e10, np.zeros(n, dtype=np.float64)

    z = beta * (d - dmax)
    # Clip z for numerical stability
    z = np.clip(z, -500, 500)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e < 1e-30:
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = fft_correlate(a, w)
    pen = 1e-3
    score = 2.0 * L + pen * (S - 1.0) ** 2

    grad = (2.0 * n * h) / (S * S) - (4.0 * LDS) / S
    grad += 2.0 * pen * (S - 1.0)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(score), grad


def multiplicative_subgrad(a, n_steps, deadline, adapt=True):
    """
    Fast multiplicative (mirror descent) subgradient method.
    a_{t+1} = normalize(a_t * exp(-eta * g_t))
    where g is the subgradient of max_k (a*a)_k.
    """
    a = normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    t0 = time.monotonic()

    # Initial step size
    c = fft_conv(a)
    cmax = float(np.max(c))
    eta = 2.0 * cmax / max(a.max(), 1e-10)
    eta = min(eta, 5.0)

    for step in range(int(n_steps)):
        if time.monotonic() > deadline:
            break

        c = fft_conv(a)
        cmax = float(np.max(c))
        sc = 2.0 * n * cmax  # since sum(a)=1

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Find active set (top-lag lags)
        # Use top-k for better direction
        k_active = min(8, len(c))
        idx = np.argpartition(c, -k_active)[-k_active:]
        w = np.zeros(len(c), dtype=np.float64)
        w[idx] = 1.0 / k_active

        # Compute subgradient: g_j = sum_k w_k * a_{k-j}
        h = fft_correlate(a, w)
        hmax = float(np.max(h))
        if hmax < 1e-15:
            break

        # Adaptive step size
        if adapt:
            eta = min(eta * 0.999, 2.0 * cmax / max(hmax, 1e-10))
            eta = max(eta, 0.01)

        # Multiplicative update
        log_a = np.log(a + 1e-300)
        log_a -= eta * h / hmax
        log_a -= np.max(log_a)  # for numerical stability
        a_new = np.exp(log_a)
        a = normalize(a_new)

        # Periodic exact score check
        if (step + 1) % 50 == 0:
            sc_check = exact_score(a)
            if sc_check < best_sc:
                best_sc = sc_check
                best_a = a.copy()

    return best_sc, best_a


def adam_lse(a, beta, deadline, max_iters=500):
    """
    Adam optimizer for the LSE-smoothed objective.
    """
    a = normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    best_lse = np.inf

    # Adam hyperparameters
    lr = 0.01
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    t0 = time.monotonic()

    for step in range(int(max_iters)):
        if time.monotonic() > deadline:
            break
        t += 1

        obj, grad = lse_obj_grad(a, beta)
        if not np.isfinite(obj):
            break

        if obj < best_lse:
            best_lse = obj
            # Check exact score periodically
            if step % 20 == 0 or obj < best_lse + 1e-10:
                sc = exact_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()

        # Adam update
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)

        # Learning rate schedule
        lr_t = lr * 0.5 ** (step // 200)

        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        # Soft normalization (don't hard-normalize to preserve Adam dynamics)
        S = float(a.sum())
        if S > 1e-10:
            a = a / S

    # Final exact check
    sc = exact_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    return best_sc, best_a


def subgrad_polish(a, deadline, max_iterations=100):
    """
    Rapid subgradient polish with multiple active-set sizes.
    """
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = exact_score(a)

    for it in range(int(max_iterations)):
        if time.monotonic() > deadline:
            break

        c = fft_conv(a)
        cmax = float(np.max(c))
        sc = 2.0 * len(a) * cmax

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        improved = False
        for k in (1, 2, 4, 8, 16):
            if k >= len(c):
                k = len(c)
            idx = np.argpartition(c, -k)[-k:]
            w = np.zeros(len(c), dtype=np.float64)
            w[idx] = 1.0 / k

            h = fft_correlate(a, w)
            hmax = float(np.max(h))
            if hmax < 1e-15:
                continue

            for mult in (0.5, 1.0, 2.0, 4.0):
                eta = mult * cmax / hmax
                eta = min(eta, 3.0)
                log_a = np.log(a + 1e-300) - eta * h / hmax
                log_a -= np.max(log_a)
                a_new = normalize(np.exp(log_a))
                sc_new = exact_score(a_new)
                if sc_new < best_sc:
                    best_sc = sc_new
                    best_a = a_new.copy()
                    a = a_new
                    improved = True
                    break
        if not improved:
            # Try smaller perturbation
            h = fft_correlate(a, np.ones(len(c)) / len(c))
            hmax = float(np.max(h))
            if hmax > 1e-15:
                eta = 0.1 * cmax / hmax
                log_a = np.log(a + 1e-300) - eta * h / hmax
                log_a -= np.max(log_a)
                a = normalize(np.exp(log_a))

    return best_sc, best_a


def resample_to_length(a, n):
    """Resample sequence to new length using linear interpolation."""
    a = np.asarray(a, dtype=np.float64).ravel()
    m = len(a)
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def make_initializations(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []

    def add(arr):
        arr = np.asarray(arr, dtype=np.float64).ravel()
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        arr = np.maximum(arr, 0.0)
        S = float(arr.sum())
        if S > 1e-10:
            inits.append(arr / S)

    # Parent (if available)
    if parent is not None and len(parent) > 10:
        p = resample_to_length(parent, n)
        add(p)
        add(p[::-1])

    # Uniform
    add(np.ones(n))

    # Linear ramps
    x = np.linspace(0, 1, n)
    add(1.0 + 0.5 * x)
    add(1.0 - 0.5 * x)
    add(1.0 + 0.8 * x)
    add(1.0 - 0.8 * x)

    # Beta-like distributions
    for alpha in (0.5, 1.0, 2.0, 3.0, 0.3, 5.0):
        vals = x ** alpha * (1 - x) ** alpha
        add(vals + 0.1)

    # Cosine windows
    for width_mult in (0.5, 1.0, 2.0):
        cx = n / 2
        vals = np.exp(-((np.arange(n) - cx) / (width_mult * n / 4)) ** 2)
        add(vals)

    # Edge-concentrated
    for w in (0.02, 0.05, 0.1, 0.15):
        vals = np.exp(-((np.arange(n) / n) / w) ** 2) + np.exp(-(((1 - np.arange(n) / n)) / w) ** 2)
        add(vals)

    # Random smooth
    for _ in range(5):
        freqs = [int(rng.integers(1, 8)) for _ in range(4)]
        amps = [float(rng.uniform(0.1, 0.4)) for _ in range(4)]
        phases = [float(rng.uniform(0, 2 * np.pi)) for _ in range(4)]
        vals = np.ones(n)
        for f, a_, ph in zip(freqs, amps, phases):
            vals += a_ * np.cos(2 * np.pi * f * x + ph)
        add(vals)

    # Random with smoothing
    for _ in range(3):
        raw = rng.random(n) + 0.1
        # Smooth with Gaussian kernel
        k = max(3, n // 100)
        if k % 2 == 0:
            k += 1
        kern = np.exp(-((np.arange(k) - k // 2) ** 2) / (2 * (k // 4 + 1) ** 2))
        kern /= kern.sum()
        smoothed = np.convolve(raw, kern, mode='same')
        add(smoothed)

    return inits


def optimize_single(a, deadline):
    """Full optimization pipeline for a single sequence."""
    a = normalize(a.copy())
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    elapsed = 0.0

    # Phase 1: Multiplicative subgradient
    t1_deadline = time.monotonic() + min(120.0, 0.25 * (deadline - time.monotonic()))
    sc, seq = multiplicative_subgrad(a, n_steps=10000, deadline=t1_deadline)
    elapsed = time.monotonic() - (deadline - t1_deadline + 120.0)
    if sc < best_sc:
        best_sc = sc
        best_a = seq.copy()

    # Phase 2: LSE + Adam with increasing beta
    remaining = deadline - time.monotonic()
    betas = [50, 100, 200, 500, 1000, 2000, 5000, 10000]
    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - 30:
            break
        # Allocate time for this beta stage
        frac = max(0.1, 0.6 * (remaining - 60.0) / (len(betas) - i))
        stage_deadline = min(time.monotonic() + min(frac, 180.0), deadline - 30)
        if stage_deadline - time.monotonic() < 5:
            continue

        sc, seq = adam_lse(best_a, beta, stage_deadline, max_iters=300)
        if sc < best_sc:
            best_sc = sc
            best_a = seq.copy()
        else:
            # Try from multiplicative result too
            sc2, seq2 = adam_lse(seq, beta, stage_deadline, max_iters=200)
            if sc2 < best_sc:
                best_sc = sc2
                best_a = seq2.copy()

    # Phase 3: Subgradient polish
    if time.monotonic() < deadline - 20:
        sc, seq = subgrad_polish(best_a, deadline - 15, max_iterations=50)
        if sc < best_sc:
            best_sc = sc
            best_a = seq.copy()

    # Phase 4: Final high-beta Adam
    if time.monotonic() < deadline - 10:
        sc, seq = adam_lse(best_a, 20000.0, deadline - 5, max_iters=200)
        if sc < best_sc:
            best_sc = sc
            best_a = seq.copy()

    # Final subgradient
    if time.monotonic() < deadline - 5:
        sc, seq = subgrad_polish(best_a, deadline - 2, max_iterations=30)
        if sc < best_sc:
            best_sc = sc
            best_a = seq.copy()

    return best_sc, best_a


def perturb(a, rng, strength=0.3):
    """Perturb a sequence for exploration."""
    a = normalize(a.copy())
    n = len(a)
    mode = int(rng.integers(0, 5))

    if mode == 0:
        # Multiplicative noise
        q = a * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        # Blend with uniform
        t = float(rng.uniform(0.1, 0.4))
        q = (1 - t) * a + t / n
    elif mode == 2:
        # Reverse + noise
        q = a[::-1] * np.exp(0.1 * rng.standard_normal(n))
    elif mode == 3:
        # Pressure-based
        c = fft_conv(a)
        k = min(4, len(c))
        idx = np.argpartition(c, -k)[-k:]
        w = np.zeros(len(c), dtype=np.float64)
        w[idx] = 1.0 / k
        h = fft_correlate(a, w)
        hmax = float(np.max(h))
        if hmax > 1e-15:
            q = a * np.exp(-strength * h / hmax)
        else:
            q = a.copy()
    else:
        # Smooth noise in log domain
        log_a = np.log(a + 1e-300)
        noise = rng.standard_normal(n)
        k = min(31, max(5, n // 100))
        if k % 2 == 0:
            k += 1
        if k >= 3:
            sig = max(1.0, k / 6.0)
            kern = np.exp(-((np.arange(k) - k // 2) ** 2) / (2 * sig ** 2))
            kern /= kern.sum()
            noise = np.convolve(noise, kern, mode='same')
        log_a += 0.2 * strength * (noise - noise.mean())
        q = np.exp(log_a - log_a.max())

    return normalize(q)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Returns the best non-negative sequence found.
    """
    t_start = time.monotonic()
    total_time = float(budget_s)
    deadline = t_start + total_time - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64:
                parent = p
    except Exception:
        parent = None

    print("Parent: len={}".format(len(parent) if parent is not None else 0))

    # Determine lengths to try
    lengths = []
    if parent is not None:
        parent_len = int(parent.size)
        if parent_len >= 1024:
            lengths.append(min(parent_len, 8192))
        if parent_len < 8192:
            lengths.append(8192)
        if parent_len > 4096:
            lengths.append(4096)
    else:
        lengths = [4096, 8192]

    # Remove duplicates, keep order
    seen = set()
    unique_lengths = []
    for L in lengths:
        L = int(L)
        if 1024 <= L <= 16384 and L not in seen:
            unique_lengths.append(L)
            seen.add(L)
    if not unique_lengths:
        unique_lengths = [4096]
    lengths = unique_lengths[:3]

    print("Lengths to try: {}".format(lengths))

    # Evaluate initializations for all lengths
    all_candidates = []  # (score, sequence, length)
    for n in lengths:
        inits = make_initializations(n, rng, parent)
        for arr in inits:
            p = normalize(arr)
            sc = exact_score(p)
            if np.isfinite(sc) and sc < 2.5:
                all_candidates.append((sc, p, n))

    # Sort by score
    all_candidates.sort(key=lambda x: x[0])
    print("Best initial: score={:.6f}, len={}".format(
        all_candidates[0][0] if all_candidates else float('inf'),
        all_candidates[0][2] if all_candidates else 0
    ))

    # Select up to 3 diverse starting points
    starts = []
    used_lengths = set()
    for sc, arr, n in all_candidates:
        if n not in used_lengths:
            starts.append(arr.copy())
            used_lengths.add(n)
        if len(starts) >= 3:
            break

    if len(starts) < 2:
        # Add perturbed versions
        base = starts[0] if starts else np.ones(4096) / 4096
        starts.append(perturb(base, rng, 0.5))
    if len(starts) < 3:
        starts.append(perturb(starts[0], rng, 0.7))

    # Track global best
    best_score = min(exact_score(s) for s in starts)
    best_seq = starts[int(np.array([exact_score(s) for s in starts]).argmin())].copy()
    print("Starting best: {:.8f}, len={}".format(best_score, len(best_seq)))

    # Time allocation
    remaining = deadline - time.monotonic()
    n_starts = len(starts)
    # Give more time to the best start
    if n_starts == 1:
        fracs = [1.0]
    elif n_starts == 2:
        fracs = [0.65, 0.35]
    else:
        fracs = [0.50, 0.30, 0.20]

    cum_time = 0.0
    for i, st in enumerate(starts):
        if i > 0:
            # For later starts, use the current best as base
            if len(starts[0]) != len(st):
                base_best = resample_to_length(best_seq, len(st))
            else:
                base_best = best_seq.copy()
            # Blend start with best
            if i == 1:
                st = normalize(0.5 * normalize(st) + 0.5 * base_best)
            else:
                st = perturb(base_best, rng, 0.4)

        sub_deadline = t_start + (cum_time + fracs[i]) * remaining
        sub_deadline = min(sub_deadline, deadline - 100.0)
        cum_time += fracs[i]

        print("[start {}] len={}, score={:.8f}, budget={:.1f}s".format(
            i, len(st), exact_score(st), sub_deadline - time.monotonic()))

        sc, seq = optimize_single(st, sub_deadline)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("  -> new global best: {:.8f}".format(best_score))

    print("After main phases: best={:.8f}, len={}".format(best_score, len(best_seq)))

    # Final global optimization on the best sequence
    if time.monotonic() < deadline - 60:
        # Try different lengths for the best
        for target_len in [len(best_seq)]:
            if time.monotonic() >= deadline - 30:
                break
            sc, seq = optimize_single(resample_to_length(best_seq, target_len), deadline - 30)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()

    # Final perturbation + quick optimize
    if time.monotonic() < deadline - 40:
        for _ in range(3):
            if time.monotonic() >= deadline - 15:
                break
            p = perturb(best_seq, rng, 0.2)
            quick_deadline = min(time.monotonic() + 30, deadline - 10)
            sc, seq = multiplicative_subgrad(p, n_steps=5000, deadline=quick_deadline)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("Perturb restart: {:.8f}".format(best_score))

    # Final subgradient polish
    if time.monotonic() < deadline - 10:
        sc, seq = subgrad_polish(best_seq, deadline - 5, max_iterations=30)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()

    # Prepare output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / S

    out = [float(x) for x in out_arr]

    # Verify
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))

    return out
