Let me now think about the LP approach mentioned in the strategy. The idea is:
- Given current a with max convolution M, solve: max sum(b) subject to b >= 0, conv(a, b)[k] <= M for all k.
- This gives a direction to move that keeps the convolution bounded.
- Then update a = (1-t)*a + t*b_rescaled.

This is interesting because it's a linear program (the constraint conv(a,b)[k] <= M is linear in b). We can solve it efficiently with scipy or cvxpy.

For large n, the LP has n variables and 2n-1 constraints, which is very tractable.

Let me design the algorithm:

1. **Initialization**: Try multiple starting points (parent, linear ramp, etc.)
2. **Main loop** (within time budget):
   a. Compute current convolution and identify the "active set" (entries near max)
   b. Solve the LP to find the best direction
   c. Line search along the direction
   d. Occasionally do a "jump" (perturbation + re-optimization)
3. **Final polish**: Coordinate descent on the exact objective

The LP at each step:
- Variables: b[0], ..., b[n-1] >= 0
- Maximize: sum(b)
- Subject to: for k = 0, ..., 2n-2: sum_j a[j] * b[k-j] <= M (where b[k-j] = 0 if k-j out of range)

This is a standard LP with n variables and 2n-1 constraints. We can solve it with scipy.optimize.linprog.

For n = 4096, this is 4096 variables and 8191 constraints - very tractable.

The solution b* gives a direction. We then do:
- new_a = (1-t)*a + t * b* * (sum(a)/sum(b*))
- Choose t by line search

This should give a non-increasing sequence of max-convolution values (after rescaling).

```python
import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2*n - 1
    M_fft = 1 << (M-1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A*A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = a.sum()
    if S < 1e-12: return np.inf
    return 2.0 * n * c.max() / (S*S)

def solve_lp_direction(a, M, top_k=None):
    """
    Solve LP: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k (or top-k active).
    """
    n = len(a)
    # conv(a, b)[k] = sum_j a[j] * b[k-j]
    # For each k, the constraint is: sum_{j: 0<=k-j<n} a[j] * b[k-j] <= M
    # Let i = k-j, so j = k-i: sum_{i: 0<=i<n, 0<=k-i<n} a[k-i] * b[i] <= M
    # i.e., for each k, the coefficients of b[i] are a[k-i] for valid i.
    
    # Build constraint matrix A_eq (sparse)
    # For constraint k (k=0,...,2n-2):
    #   sum_{i=max(0,k-n+1)}^{min(k,n-1)} a[k-i] * b[i] <= M
    
    # A_ub[i, k] = a[k-i] if 0 <= k-i < n, else 0
    # b_ub = M for all constraints
    
    # This is a banded matrix! A_ub[i,k] = a[k-i] means A_ub is a Hankel-like matrix.
    
    # For efficiency, use sparse matrix
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        i_start = max(0, k-n+1)
        i_end = min(k, n-1)
        for i in range(i_start, i_end+1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k-i])
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b_ub = np.full(2*n-1, M)
    
    c_obj = -np.ones(n)  # minimize -sum(b)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    else:
        return None

def lp_iteration(a, t=0.3):
    """One LP-based iteration."""
    n = len(a)
    c = fast_conv(a)
    M = c.max()
    S = a.sum()
    
    b = solve_lp_direction(a, M)
    if b is None:
        return a, compute_score(a)
    
    b_sum = b.sum()
    if b_sum < 1e-12:
        return a, compute_score(a)
    
    # Rescale b to have same sum as a
    b_scaled = b * (S / b_sum)
    
    # Line search
    best_a = a.copy()
    best_score = compute_score(a)
    for tt in [t*0.5, t, t*1.5]:
        candidate = (1-tt)*a + tt*b_scaled
        candidate = np.maximum(candidate, 0)
        sc = compute_score(candidate)
        if sc < best_score:
            best_score = sc
            best_a = candidate
    
    return best_a, best_score
```

```python
"""
LP-guided optimization for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
Uses linear programming to find optimal improvement directions, combined with
Adam gradient refinement and coordinate descent polish.
Key innovation: LP finds the best direction b maximizing sum(b) subject to conv(a,b) <= current_max,
giving a structured, globally-informed update rather than local gradient steps.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time
import math


def fast_conv(a):
    """Fast linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def precompute_lp_structure(n):
    """Precompute row/col indices and offsets for the LP constraint matrix."""
    rows_list = []
    cols_list = []
    for i in range(n):
        length = min(n, 2 * n - 1 - i)
        rows_list.append(np.arange(i, i + length))
        cols_list.append(np.full(length, i))
    rows = np.concatenate(rows_list)
    cols = np.concatenate(cols_list)
    offsets = rows - cols
    return rows, cols, offsets


def solve_lp_for_direction(a, M, rows, cols, offsets, time_limit=10.0):
    """
    Solve LP: max sum(b) s.t. conv(a,b)[k] <= M for all k, b >= 0.
    Returns the optimal b vector, or None if failed.
    """
    n = len(a)
    vals = a.astype(np.float64)[offsets]
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                  bounds=[(0.0, None)] * n, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})

    if res.success:
        return res.x
    return None


def lp_iteration(a, rows, cols, offsets, time_limit=10.0, t_candidates=None):
    """
    One LP-based iteration: find direction, rescale, line search.
    Returns (new_a, new_score, improved).
    """
    n = len(a)
    c = fast_conv(a)
    M = float(c.max())
    S = float(a.sum())
    current_score = 2.0 * n * M / (S * S)

    b = solve_lp_for_direction(a, M, rows, cols, offsets, time_limit)
    if b is None:
        return a, current_score, False

    b_sum = float(b.sum())
    if b_sum < 1e-12:
        return a, current_score, False

    # Check if b is meaningfully different from a
    b_normed = b / b_sum
    a_normed = a / S
    diff = np.abs(b_normed - a_normed).sum()
    if diff < 1e-8:
        return a, current_score, False

    # Rescale b to have same sum as a
    b_rescaled = b * (S / b_sum)

    # Line search
    if t_candidates is None:
        t_candidates = [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]

    best_a = a.copy()
    best_score = current_score
    for tt in t_candidates:
        candidate = np.maximum((1.0 - tt) * a + tt * b_rescaled, 0.0)
        if candidate.sum() < 1e-12:
            continue
        sc = compute_score(candidate)
        if sc < best_score:
            best_score = sc
            best_a = candidate.copy()

    improved = best_score < current_score - 1e-12
    return best_a, best_score, improved


def adam_lse(a_init, beta, lr, n_steps, time_budget, t_start):
    """Adam optimizer with LSE-smoothed objective."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8

    best_a = a.copy()
    best_score = compute_score(a)
    deadline = t_start + time_budget

    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break

        # Compute LSE-smoothed score and gradient
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        c_max = float(c.max())
        scaled = beta * (c - c_max)
        exp_scaled = np.exp(scaled)
        exp_sum = exp_scaled.sum()
        L = c_max + np.log(exp_sum) / beta
        p = exp_scaled / exp_sum

        # Gradient of LSE w.r.t. a
        # dL/da[j] = sum_k p[k] * d(conv(a,a)[k])/da[j] = sum_k p[k] * a[k-j] (valid)
        # This is conv(a, p)[n-1 : 2n-1] (the "correlation" part)
        n2 = 3 * n - 2
        M2_fft = 1 << (n2 - 1).bit_length()
        RA = np.fft.rfft(a[::-1], M2_fft)
        P = np.fft.rfft(p, M2_fft)
        conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
        dL_da = 2.0 * conv_res

        # Full gradient of score = 2n*L/S^2
        grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))

        # Adam update
        lr_t = lr / (1.0 + 1e-5 * t)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e6)

        # Normalize
        s = a.sum()
        if s > 1e-12:
            a = a * (a_init.sum() / s)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def subgrad_descent(a_init, n_steps, lr_init, time_budget, t_start):
    """Subgradient descent on the exact max-convolution objective."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S0 = a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = t_start + time_budget

    # Track all near-max indices for smoother subgradient
    active_weights = np.zeros(2 * n - 1)
    decay = 0.95

    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        score = 2.0 * n * M / (S * S)

        # Update active weights (softmax over near-max entries)
        delta = 0.01 * M
        near_max = c >= (M - delta)
        active_weights = decay * active_weights + (1 - decay) * (near_max.astype(np.float64))
        aw_sum = active_weights.sum()
        if aw_sum > 0:
            p = active_weights / aw_sum
        else:
            p = np.zeros(2 * n - 1)
            p[int(c.argmax())] = 1.0

        # Subgradient: d(max_c)/da[j] = sum_k p[k] * a[k-j] (valid)
        n2 = 3 * n - 2
        M2_fft = 1 << (n2 - 1).bit_length()
        RA = np.fft.rfft(a[::-1], M2_fft)
        P = np.fft.rfft(p, M2_fft)
        conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
        dM_da = 2.0 * conv_res

        grad = (2.0 * n) * (dM_da / (S * S) - 2.0 * M / (S * S * S))

        lr_t = lr_init / (1.0 + 0.01 * t)
        a = a - lr_t * grad
        a = np.clip(a, 0.0, 1e6)

        s = a.sum()
        if s > 1e-12:
            a = a * (S0 / s)

        if t % 1000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def coordinate_descent(a_init, n_sweeps, top_k, time_budget, t_start):
    """Coordinate descent with golden-section 1D search on top-k coordinates."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget
    gr = (math.sqrt(5.0) + 1.0) / 2.0

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        # Compute subgradient at peak
        k_star = int(c.argmax())
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = a[idx]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)

        # Select top-k by |gradient|
        order = np.argsort(np.abs(grad))[::-1][:top_k]

        for j in order:
            if time.monotonic() > deadline:
                break
            j = int(j)
            a_old = a[j]
            if a_old < 1e-15:
                candidates = [0.0, 0.01 * a.max(), 0.05 * a.max()]
                best_val = a_old
                best_sc = compute_score(a)
                for v in candidates:
                    a[j] = v
                    sc = compute_score(a)
                    if sc < best_sc:
                        best_sc = sc
                        best_val = v
                a[j] = best_val
            else:
                lo = 0.0
                hi = a_old * 4.0
                c1 = hi - ((gr - 1.0)) * (hi - lo)
                c2 = lo + (gr - 1.0) * (hi - lo)
                a[j] = c1
                f1 = compute_score(a)
                a[j] = c2
                f2 = compute_score(a)
                for _ in range(12):
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - (gr - 1.0) * (hi - lo)
                        a[j] = c1
                        f1 = compute_score(a)
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + (gr - 1.0) * (hi - lo)
                        a[j] = c2
                        f2 = compute_score(a)
                a[j] = c1 if f1 < f2 else c2

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()

    return best_score, best_a


def spectral_perturbation(a, rng, strength=0.15):
    """Perturb in DFT domain for diversity."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    amp_factors = 1.0 + strength * rng.standard_normal(len(A))
    amp_factors = np.clip(amp_factors, 0.3, 2.0)
    phase_shifts = strength * 0.5 * rng.standard_normal(len(A))
    A_new = A * amp_factors * np.exp(1j * phase_shifts)
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() < 1e-10:
        a_new = a.copy()
    else:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def anti_palindromic_init(n, rng, smoothness=0.5):
    """Create an anti-palindromic initial guess."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    # Base: linear ramp (anti-palindromic component)
    ramp = (x - center) / center + 1.0
    # Add some frequency structure
    n_modes = 5
    for f in range(1, n_modes + 1):
        ramp += 0.1 / f * np.sin(2.0 * np.pi * f * x / n + rng.uniform(0, 2 * np.pi))
    ramp = np.clip(ramp, 0.1, None)
    # Smooth
    if smoothness > 0:
        kernel_size = max(3, int(n * smoothness * 0.01))
        if kernel_size % 2 == 0:
            kernel_size += 1
        kernel = np.exp(-0.5 * (np.arange(-kernel_size // 2, kernel_size // 2 + 1) / (kernel_size // 4)) ** 2)
        kernel /= kernel.sum()
        from scipy.ndimage import uniform_filter1d
        ramp = uniform_filter1d(ramp, size=kernel_size, mode='nearest')
        ramp = np.clip(ramp, 0.01, None)
    return ramp


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: LP-guided optimization + Adam + subgradient + coord descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    # Determine working length
    if parent is not None and 1024 <= len(parent) <= 16384:
        n = len(parent)
    else:
        n = 4096

    print(f"Working length: n={n}")

    # Phase 1: Multiple initializations (5% time)
    p1_time = effective_budget * 0.05
    print(f"\nPhase 1: Initialization (budget={p1_time:.0f}s)")

    candidates = []
    if parent is not None:
        candidates.append(("parent", parent[:n].copy() if len(parent) >= n else np.pad(parent, (0, n - len(parent)))))
    candidates.append(("ramp", anti_palindromic_init(n, rng, 0.3)))

    # Linear ramp
    ramp = np.linspace(0.5, 2.0, n)
    candidates.append(("linear_ramp", ramp))

    # Random with bias
    rand_a = np.abs(rng.standard_normal(n)) + 0.1
    rand_a += 0.3 * np.linspace(0, 1, n)
    candidates.append(("random", rand_a))

    # Evaluate all
    best_score = np.inf
    best_a = None
    for name, cand in candidates:
        cand = np.clip(cand, 0.0, 1000.0)
        sc = compute_score(cand)
        print(f"  {name}: score={sc:.8f}")
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()

    print(f"  Best initial: {best_score:.8f}")

    # Precompute LP structure
    print("Precomputing LP structure...")
    t0 = time.monotonic()
    rows, cols, offsets = precompute_lp_structure(n)
    print(f"  Done in {time.monotonic()-t0:.1f}s, nnz={len(rows)}")

    # Phase 2: LP-based optimization (55% time)
    p2_time = effective_budget * 0.55
    print(f"\nPhase 2: LP optimization (budget={p2_time:.0f}s)")

    a = best_a.copy()
    current_score = best_score
    p2_deadline = t_start + effective_budget * 0.62  # Leave time for other phases
    n_lp_iters = 0
    no_improve_count = 0

    while time.monotonic() < p2_deadline:
        remaining = p2_deadline - time.monotonic()
        if remaining < 5:
            break

        # Adaptive LP time limit
        lp_time = min(remaining * 0.3, 15.0)
        if remaining < 60:
            lp_time = min(lp_time, remaining * 0.5)

        new_a, new_score, improved = lp_iteration(
            a, rows, cols, offsets, time_limit=lp_time,
            t_candidates=[0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
        )

        n_lp_iters += 1
        if improved:
            a = new_a
            current_score = new_score
            no_improve_count = 0
            if n_lp_iters % 50 == 0:
                print(f"  LP iter {n_lp_iters}: score={current_score:.8f} (t={time.monotonic()-t_start:.0f}s)")
        else:
            no_improve_count += 1
            if no_improve_count >= 20:
                # Try a perturbation
                print(f"  Stuck at iter {n_lp_iters}, perturbing...")
                a_pert = spectral_perturbation(a, rng, strength=0.1)
                # Quick LP step from perturbed
                new_a2, new_sc2, _ = lp_iteration(
                    a_pert, rows, cols, offsets, time_limit=min(lp_time, 10.0)
                )
                if new_sc2 < current_score:
                    a = new_a2
                    current_score = new_sc2
                else:
                    # Try stronger perturbation
                    a_pert2 = spectral_perturbation(a, rng, strength=0.3)
                    new_a3, new_sc3, _ = lp_iteration(
                        a_pert2, rows, cols, offsets, time_limit=min(lp_time, 10.0)
                    )
                    if new_sc3 < current_score:
                        a = new_a3
                        current_score = new_sc3
                no_improve_count = 0

        # Update best
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()

    print(f"  Phase 2 done: {n_lp_iters} LP iters, score={current_score:.8f}")

    # Phase 3: Adam refinement (20% time)
    p3_time = effective_budget * 0.20
    print(f"\nPhase 3: Adam refinement (budget={p3_time:.0f}s)")

    adam_deadline = t_start + effective_budget * 0.82
    if time.monotonic() < adam_deadline - 10:
        t_adam = min(adam_deadline - time.monotonic(), p3_time)
        # Multiple beta values
        for beta in [200, 500, 2000]:
            if time.monotonic() > adam_deadline - 5:
                break
            t_rem = adam_deadline - time.monotonic()
            sc, a_opt = adam_lse(best_a, beta=beta, lr=0.003,
                                 n_steps=50000, time_budget=t_rem, t_start=t_start)
            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()
                print(f"  Adam beta={beta}: {sc:.8f}")

    # Subgradient descent
    if time.monotonic() < adam_deadline - 5:
        t_sub = min(adam_deadline - time.monotonic(), 30.0)
        sc, a_opt = subgrad_descent(best_a, n_steps=20000, lr_init=0.001,
                                    time_budget=t_sub, t_start=t_start)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  Subgrad: {sc:.8f}")

    # Phase 4: More LP iterations on best (10% time)
    p4_time = effective_budget * 0.10
    print(f"\nPhase 4: Final LP on best (budget={p4_time:.0f}s)")

    p4_deadline = t_start + effective_budget * 0.92
    a = best_a.copy()
    n_lp2 = 0
    while time.monotonic() < p4_deadline:
        remaining = p4_deadline - time.monotonic()
        if remaining < 5:
            break
        lp_time = min(remaining * 0.4, 10.0)
        new_a, new_score, improved = lp_iteration(
            a, rows, cols, offsets, time_limit=lp_time,
            t_candidates=[0.03, 0.05, 0.1, 0.2, 0.3, 0.5]
        )
        n_lp2 += 1
        if improved:
            a = new_a
            if new_score < best_score:
                best_score = new_score
                best_a = a.copy()
                if n_lp2 % 20 == 0:
                    print(f"  Final LP iter {n_lp2}: {best_score:.8f}")
        elif n_lp2 % 10 == 0:
            # Small perturbation to escape
            a = spectral_perturbation(a, rng, strength=0.05)

    print(f"  Phase 4 done: {n_lp2} iters, best={best_score:.8f}")

    # Phase 5: Coordinate descent polish (8% time)
    p5_deadline = t_start + effective_budget * 0.98
    print(f"\nPhase 5: Coordinate descent polish")

    if time.monotonic() < p5_deadline - 5:
        t_cd = min(p5_deadline - time.monotonic(), effective_budget * 0.08)
        sc, a_opt = coordinate_descent(best_a, n_sweeps=10, top_k=40,
                                       time_budget=t_cd, t_start=t_start)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  CoordDesc: {sc:.8f}")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")

    return out
```
