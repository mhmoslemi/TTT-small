"""
Improved optimizer using hybrid LP (FISTA + exact LP on active constraints),
multilevel initialization, and LSE-smoothed L-BFGS polish.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix
import time


def cross_conv(a, b):
    """Full linear convolution via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    # Gradient of L w.r.t. a
    rev_a = a[::-1]
    # cross_corr(a, p) = conv(rev_a, p), we need indices [n-1, 2n-2]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def func_and_grad(a):
        return lse_score_grad(a, beta)
    
    res = minimize(
        func_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fista_lp(a, M, n_iters=2000, beta_start=5.0, beta_end=100.0):
    """
    Solve LP: max sum(x) s.t. conv(a,x)[k] <= M for all k, x >= 0
    using FISTA on the LSE-smoothed surrogate.
    """
    n = len(a)
    x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0
    
    sum_a2 = float(np.sum(a * a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n
    
    c_len = 2 * n - 1
    
    for it in range(n_iters):
        frac = min(1.0, it / max(1, n_iters * 0.4))
        beta = beta_start + (beta_end - beta_start) * frac
        
        L = beta * n * sum_a2 + 1e-12
        step = 1.0 / L
        
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(c_len) / c_len
        else:
            p = exp_z / exp_sum
        
        # C^T @ p: for each i, sum_k p[k] * a[k-i] where valid
        # = cross-correlation of a and p, specifically (a * p)[i] for i=0..n-1
        # Actually: (C^T p)[i] = sum_k C[k,i] * p[k] = sum_k a[k-i] * p[k] (where 0<=k-i<n)
        # = sum_j a[j] * p[i+j] where j = k-i, so k = i+j, j=0..n-1
        # This is the cross-correlation <a, shift_i(p)>.
        # Can compute via FFT: it's the "anti-convolution" or cross-correlation.
        # (a * p_pad)[i + n - 1] where p_pad is p zero-padded... 
        # Actually: sum_j a[j] * p[i+j] = (a_reversed * p)[i + n - 1]
        # where a_reversed[j] = a[n-1-j]
        
        # Simpler: use sliding window sum
        # (C^T p)[i] = sum_{j=0}^{n-1} a[j] * p[i+j]
        # This is a convolution of a with a portion of p.
        
        # Use FFT: let a_pad = a (length n), p needs to be at least 2n-1 long (which it is)
        # sum_j a[j] * p[i+j] = (a * p)[i + n - 1] ... no wait
        # conv(a, p)[k] = sum_j a[j] * p[k-j]
        # We want sum_j a[j] * p[i+j]. Let k = i+j, j = k-i.
        # = sum_j a[j] * p[i+j] = sum_k a[k-i] * p[k] (where 0<=k-i<n, 0<=k<2n-1)
        # This is the cross-correlation: <a, p shifted by i>
        # = conv(a_rev, p)[i + n - 1] where a_rev = a[::-1]
        
        a_rev = a[::-1]
        M3 = n + (2*n-1) - 1
        M3_fft = 1 << (M3 - 1).bit_length()
        AR = np.fft.rfft(a_rev, M3_fft)
        P = np.fft.rfft(p, M3_fft)
        conv_ap = np.fft.irfft(AR * P, M3_fft)[:M3]
        grad = conv_ap[n-1:n-1+n] - 1.0  # subtract the linear objective gradient
        
        # FISTA update
        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new
    
    return x


def solve_lp_active(a, M, active_constraints, n):
    """
    Solve the LP with only the active constraints.
    max sum(x) s.t. conv(a,x)[k] <= M for k in active_constraints, x >= 0
    Returns x or None if infeasible.
    """
    # Build constraint matrix for active constraints only
    K = len(active_constraints)
    if K == 0:
        return np.ones(n)
    
    # For constraint k: sum_i C[k,i] * x[i] <= M
    # C[k,i] = a[k-i] if 0 <= k-i < n, else 0
    
    rows = []
    cols = []
    vals = []
    
    for k in active_constraints:
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(k)  # We'll map to 0..K-1 later
            cols.append(i)
            vals.append(a[k - i])
    
    # Better: build in COO format directly
    row_idx = []
    col_idx = []
    data = []
    
    for ci, k in enumerate(active_constraints):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            row_idx.append(ci)
            col_idx.append(i)
            data.append(a[k - i])
    
    if not data:
        return np.ones(n)
    
    A_ub = coo_matrix((data, (row_idx, col_idx)), shape=(K, n)).tocsr()
    b_ub = np.full(K, M)
    
    # Objective: maximize sum(x) = minimize -sum(x)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': 30})
    
    if res.success:
        return np.maximum(res.x, 0)
    return None


def hybrid_lp(a, M, fista_iters=500, time_limit=None):
    """
    Hybrid LP solving: FISTA to get approximate solution, then exact LP on active constraints.
    """
    n = len(a)
    
    # Step 1: FISTA for approximate solution
    x_approx = fista_lp(a, M, n_iters=fista_iters, beta_start=5.0, beta_end=50.0)
    Sx = float(x_approx.sum())
    if Sx < 1e-12:
        return x_approx
    
    # Step 2: Identify active constraints
    c = cross_conv(a, x_approx)
    # Active: c[k] > 0.85 * M (close to the bound)
    threshold = 0.85 * M
    active = np.where(c > threshold)[0]
    
    # Also always include the peak
    k_peak = int(np.argmax(c))
    if k_peak not in active:
        active = np.append(active, k_peak)
    
    # Limit to at most 500 active constraints (most violated)
    if len(active) > 500:
        c_active = c[active]
        top_idx = np.argsort(c_active)[::-1][:500]
        active = active[top_idx]
    
    # Step 3: Solve reduced LP
    x_lp = solve_lp_active(a, M, active, n)
    
    if x_lp is not None:
        # Verify against all constraints
        c_check = cross_conv(a, x_lp)
        if c_check.max() <= M * (1 + 1e-6):
            return x_lp
        else:
            # Add violated constraints and re-solve
            violated = np.where(c_check > M)[0]
            active = np.unique(np.concatenate([active, violated]))
            if len(active) > 1000:
                c_act = c_check[active]
                top_idx = np.argsort(c_act)[::-1][:1000]
                active = active[top_idx]
            x_lp2 = solve_lp_active(a, M, active, n)
            if x_lp2 is not None:
                c_check2 = cross_conv(a, x_lp2)
                if c_check2.max() <= M * (1 + 1e-6):
                    return x_lp2
    
    # Fallback to FISTA solution
    return x_approx


def lp_alternating(a, max_outer_iters=50, time_limit=None):
    """
    LP-based alternating method with hybrid LP solving and line search for mixing.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    
    for outer in range(max_outer_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        
        M = float(self_conv(a).max())
        
        # Adaptive FISTA iterations based on remaining time
        fista_iter = 800 if outer < 10 else 500
        
        x = hybrid_lp(a, M, fista_iters=fista_iter)
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx
        
        # Line search for mixing parameter t
        best_t = 0.3
        best_sc_t = best_sc
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
            a_try = (1.0 - t) * a + t * x
            a_try = np.maximum(a_try, 0.0)
            St = float(a_try.sum())
            if St < 1e-12:
                continue
            a_try = a_try / St
            sc_try = compute_score(a_try)
            if sc_try < best_sc_t:
                best_sc_t = sc_try
                best_t = t
        
        a_new = (1.0 - best_t) * a + best_t * x
        a_new = np.maximum(a_new, 0.0)
        Sn = float(a_new.sum())
        if Sn < 1e-12:
            break
        a_new = a_new / Sn
        
        sc = compute_score(a_new)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 10:
                break
        
        a = a_new
    
    return best_a, best_sc


def error_reduction(a, n_iters=100, alpha=0.7):
    """Newton-like error reduction in frequency domain."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]
        
        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break
        
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)
        
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0
        
        damp = 0.3 + 0.7 * (it / max(1, n_iters))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def subgradient_polish(a, total_iters=2000):
    """Projected subgradient descent on the exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(total_iters):
        c = self_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = seg[::-1]
        
        # The subgradient of M = max conv at k* is:
        # ∂M/∂a[i] = a[k*-i] + a[2i-k*] (for valid indices)
        # But we need to be more careful.
        # M = sum_j a[j] * a[k*-j]
        # ∂M/∂a[i] = a[k*-i] (from the first a) + a[2i-k*] (from the second a)
        # where both terms are valid if the index is in [0, n-1]
        
        # Actually, let me recompute:
        # M = conv(a,a)[k*] = sum_j a[j]*a[k*-j]
        # ∂M/∂a[i] = sum_j [δ(j,i)*a[k*-j] + a[j]*δ(k*-j,i)]
        # = a[k*-i] + a[k*-i]  (when k*-i is valid, both terms give a[k*-i])
        # Wait no:
        # = a[k*-i] (from first term, j=i) + a[k*-i] (from second term, k*-j=i => j=k*-i)
        # But the second term: sum_j a[j]*δ(k*-j,i) = a[k*-i] (when k*-i is valid)
        # So ∂M/∂a[i] = a[k*-i] + a[k*-i] = 2*a[k*-i] (when k*-i is valid)
        # But wait, if i = k*-i (i.e., k* = 2i), then it's just a[i] (counted once in the sum)
        # Actually no: M = sum_j a[j]*a[k*-j]. If we differentiate w.r.t. a[i]:
        # The term where j=i gives a[k*-i]
        # The term where k*-j=i gives a[i] (i.e., j=k*-i)
        # If i != k*-i, these are two different terms, each contributing a[k*-i] and a[i]... 
        # wait, a[j]*a[k*-j] with j = k*-i gives a[k*-i]*a[i]
        # So ∂M/∂a[i] = a[k*-i] + a[k*-i] = 2*a[k*-i] (when i != k*-i)
        # And ∂M/∂a[i] = a[i] (when i = k*-i, i.e., k* = 2i)
        
        # Hmm, let me just use the simpler formula:
        # ∂M/∂a[i] = 2 * a[k*-i] (for k*-i in [0,n-1])
        # Special case: if k* = 2i, then ∂M/∂a[i] = a[i] (not 2*a[i])
        # Actually wait: M = sum_j a[j]*a[k*-j]
        # If k* = 2i, then the term j=i gives a[i]*a[i] = a[i]^2, whose derivative is 2*a[i]
        # And there's no other term that contributes to ∂/∂a[i].
        # So ∂M/∂a[i] = 2*a[i] when k* = 2i.
        # When k* != 2i: ∂M/∂a[i] = a[k*-i] (from j=i term) + a[k*-i] (from j=k*-i term) = 2*a[k*-i]
        # Wait that's the same! Because a[k*-i] when j=k*-i: the term is a[k*-i]*a[i], derivative w.r.t. a[i] is a[k*-i].
        # And from j=i: the term is a[i]*a[k*-i], derivative w.r.t. a[i] is a[k*-i].
        # So total: 2*a[k*-i] always (when k*-i is in [0,n-1]).
        # And when k* = 2i: k*-i = i, so it's 2*a[i]. Same thing!
        
        # So the formula is simply: ∂M/∂a[i] = 2*a[k*-i] if 0 <= k*-i < n, else 0.
        
        sg = np.zeros(n)
        idx = k_star - np.arange(n)  # k_star - i for i=0..n-1
        valid = (idx >= 0) & (idx < n)
        sg[valid] = 2.0 * a[idx[valid]]
        
        # Objective: F = 2n * M / S^2
        # ∂F/∂a[i] = 2n * (∂M/∂a[i] * S^2 - M * 2*S * ∂S/∂a[i]) / S^4
        # = 2n * (sg[i] / S^2 - 2*M / S^3)
        # (since ∂S/∂a[i] = 1)
        
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M_val / S3)
        
        # Projected gradient step (project onto {a >= 0, sum(a) = S})
        # Simple: subtract gradient, clip at 0, renormalize
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Adaptive step size
        step = 2.0 * (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.02)
        
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    return best_a
