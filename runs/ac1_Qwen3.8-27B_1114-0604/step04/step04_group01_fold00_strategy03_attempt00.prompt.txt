def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

import numpy as np

from scipy.optimize import minimize

PEN_SUM = 1e-3


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    """Full linear self-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    """
    h_j = sum_k w_k * a_{k-j}, for j = 0,...,n-1,
    computed as a middle slice of conv(a[::-1], w).
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    """Clip to non-negative, normalize to target sum."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    S2 = float(a.sum())
    if S2 < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    return a * (float(target) / S2)


def resample_sequence(a, n):
    """Linearly resample a non-negative sequence to length n."""
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def d_from_a(a):
    """
    d = n * conv(a,a) / sum(a)^2.
    The exact score is 2 * max(d).
    """
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def exact_score_a(a):
    """Exact score 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_score_grad_tuple(a, beta):
    """
    LSE-smooth objective and gradient for d = n*conv(a,a)/sum(a)^2.
    Objective:
        F = 2 * LSE_beta(d) + PEN * (sum(a)-1)^2
    Gradient is exact for this objective.
    """
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    grad = np.zeros(n, dtype=np.float64)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, grad

    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0

    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, grad

    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, grad
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))

    h = corr_pair(a, w)
    score = 2.0 * L + PEN_SUM * (S - 1.0) * (S - 1.0)

    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def fun_lse_tuple(a, beta):
    return lse_score_grad_tuple(a, beta)


def estimate_eval_time(n):
    """Conservative estimate of seconds per LSE objective+gradient evaluation."""
    base = 0.018
    scale = max(0.5, float(n) / 8192.0)
    log_scale = max(1.0, float(np.log2(max(16, n))) / float(np.log2(8192.0)))
    return base * scale * log_scale


def estimate_maxfun(t_left, n):
    t_left = float(max(1.0, t_left))
    if t_left < 4.0:
        return 64
    est = estimate_eval_time(n)
    mf = int((0.7 * t_left) / max(est, 1e-4))
    if mf < 64:
        mf = 64
    if mf > 3000:
        mf = 3000
    return mf


def run_lbfgs_tuple(obj_func, a0, param, deadline, maxfun):
    """
    Run L-BFGS-B with an objective/gradient function returning (value, grad).
    """
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 4.0:
            return exact_score_a(a), a
        est = estimate_eval_time(n)
        cap = int((0.9 * rem) / max(est, 1e-4))
        if cap < 64:
            cap = 64
        maxfun = int(min(int(maxfun), cap))
    else:
        maxfun = int(maxfun if maxfun is not None else 1000)

    if maxfun < 64:
        maxfun = 64
    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-12,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 50,
    }

    try:
        res = minimize(
            obj_func,
            a,
            method="L-BFGS-B",
            jac=True,
            args=(param,),
            bounds=bounds,
            options=opts,
        )
    except Exception:
        return exact_score_a(a), a

    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a

    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out

def generate_inits(n, rng, parent, cap=None):
    if cap is None:
        cap = 24 if n > 6000 else 32
    inits = []
    n = int(n)

    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add_to_inits(inits, pr, cap)
        add_to_inits(inits, pr[::-1], cap)

    x = np.linspace(0.0, 1.0, n)
    add_to_inits(inits, np.ones(n, dtype=np.float64), cap)

    for alpha in (0.3, 0.6, 0.85):
        add_to_inits(inits, 1.0 + alpha * (2.0 * x - 1.0), cap)
        add_to_inits(inits, 1.0 - alpha * (2.0 * x - 1.0), cap)

    max_trig = 10 if n > 6000 else 14
    for r in range(max_trig):
        vals = np.ones(n, dtype=np.float64)
        K = int(rng.integers(2, 6))
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 8)))
            amp = float(rng.uniform(0.05, 0.35))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
            if r % 2 == 1:
                vals += 0.5 * amp * np.sin(2.0 * np.pi * f * x + ph)
        add_to_inits(inits, vals, cap)

    max_low = 5 if n > 6000 else 8
    Nf = 1 << (n - 1).bit_length()
    for _ in range(max_low):
        freqs = np.fft.rfftfreq(Nf) * Nf
        power = 1.0 / ((np.abs(freqs) + 1.0) ** float(rng.uniform(1.5, 3.5)))
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        noise = np.fft.irfft(spec, Nf)[:n]
        noise = noise - float(noise.mean())
        vals = 1.0 + float(rng.uniform(0.3, 0.9)) * noise
        add_to_inits(inits, vals, cap)

    for frac in (0.50, 0.65, 0.80):
        mask = rng.random(n) < frac
        vals = np.where(mask, 0.2 + 0.8 * rng.random(n), 0.0)
        add_to_inits(inits, vals, cap)

    for width in (0.05, 0.10, 0.20):
        vals = 0.2 + 0.8 * (
            np.exp(-((x / width) ** 2)) + np.exp(-(((1.0 - x) / width) ** 2))
        )
        add_to_inits(inits, vals, cap)

    for nb in (8, 16, 32, 64):
        if nb > n:
            nb = n
        block_vals = 0.1 + 0.9 * rng.random(nb)
        per = max(1, n // nb)
        a = np.repeat(block_vals, per)
        if a.size < n:
            a = np.concatenate([a, np.repeat(block_vals, n - a.size)])
        add_to_inits(inits, a[:n], cap)

    for _ in range(5):
        add_to_inits(inits, 0.1 + 0.9 * rng.random(n), cap)

    return inits


def active_weights(d, k):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / float(k)


def pressure_direction(p, k=3):
    d, S = d_from_a(p)
    n = len(p)
    if S < 1e-12 or d.size == 0:
        return np.zeros(n, dtype=np.float64)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return np.zeros(n, dtype=np.float64)
    w = active_weights(d, k)
    return corr_pair(p, w)


def smooth_noise(n, rng, strength):
    noise = rng.standard_normal(int(n))
    k = min(31, max(3, int(n) // 200))
    if k % 2 == 0:
        k += 1
    if k > int(n):
        k = int(n) if int(n) % 2 == 1 else max(3, int(n) - 1)
    if k >= 3:
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern = kern / float(kern.sum())
        noise = np.convolve(noise, kern, mode="same")
    noise = noise - float(noise.mean())
    return noise * float(strength)


def perturb_best(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.15, 0.70))
    mode = int(rng.integers(0, 6))

    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.10, 0.50))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        q = p[::-1].copy()
        q = q * np.exp(0.1 * rng.standard_normal(n))
    elif mode == 3:
        h = pressure_direction(p, 4)
        hmax = float(np.max(h))
        if hmax < 1e-12:
            q = p.copy()
        else:
            q = p * np.exp(-strength * h / hmax)
    elif mode == 4:
        logp = np.log(p + 1e-300)
        logp = logp + smooth_noise(n, rng, strength * 0.5)
        q = np.exp(logp - float(np.max(logp)))
    else:
        k = int(rng.integers(3, 15))
        if k % 2 == 0:
            k += 1
        if k > n:
            k = n if n % 2 == 1 else max(3, n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        knorm = float(kern.sum())
        if knorm > 0.0:
            kern = kern / knorm
        q = np.convolve(p, kern, mode="same")

    return normalize_a(q)

def subgrad_polish(a, max_iters=20, deadline=None):
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    cur_sc = best_sc

    for _ in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        d, S = d_from_a(a)
        if S < 1e-12 or d.size == 0:
            break
        dmax = float(np.max(d))
        if not np.isfinite(dmax) or dmax <= 0.0:
            break

        cur_a = None
        cur_new_sc = cur_sc

        for k in (1, 3, 8):
            w = active_weights(d, k)
            h = corr_pair(a, w)
            hmax = float(np.max(h))
            if hmax < 1e-12:
                continue
            base = 1.0 / hmax
            for mult in (0.02, 0.08, 0.20, 0.50):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                eta = mult * base
                q = a * np.exp(-eta * h)
                q = normalize_a(q)
                sc = exact_score_a(q)
                if not np.isfinite(sc):
                    continue
                if sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                if sc < cur_new_sc:
                    cur_new_sc = sc
                    cur_a = q.copy()

        if cur_a is None:
            break
        a = cur_a
        cur_sc = cur_new_sc

    return float(best_sc), best_a
