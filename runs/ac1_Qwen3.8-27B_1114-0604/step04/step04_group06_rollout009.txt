def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_cross_conv(a, b):
    """Cross-convolution a * b via FFT."""
    n_a = len(a)
    n_b = len(b)
    M = n_a + n_b - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:

    return 2.0 * n * float(c.max()) / (S * S)

    conv_ap = np.fft.irfft(A_flip * P, M2_fft)[:M2]
    h = conv_ap[n - 1:2 * n - 1]
    grad_L = 2.0 * h
    S2 = S * S
    S3 = S2 * S
    grad_score = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad_score


def subgrad_score(a):
    """Exact score and subgradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    score = 2.0 * n * M / (S * S)
    # Subgradient: for each i, d/da_i max_k conv = 2*sum_{k: binding} a[k-i]
    # Use all argmax positions
    k_max = np.where(c >= M - 1e-12)[0]
    grad = np.zeros(n)
    for k in k_max:
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            idx = k - np.arange(js, je + 1)
            grad[js:je + 1] += 2.0 * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad / S2 - 2.0 * M / S3)
    return score, grad


def lp_direction(f, M, maxiter_lp=5000):
    """
    Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0.
    Returns the optimal g (unnormalized).
    """
    n = len(f)
    n_cons = 2 * n - 1
    # Build sparse constraint matrix A: A[k, j] = f[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for k in range(n_cons):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        for j in range(js, je + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    A = csr_matrix((vals, (rows, cols)), shape=(n_cons, n))
    # Maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    b_ub = np.full(n_cons, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'maxiter': maxiter_lp})
    if res.status == 0:
        return np.clip(res.x, 0, None)
    return None


def adam_lse(a_init, lr=0.01, n_steps=30000, beta_lse=200,
             time_limit_s=None, t_start=None):
    """Adam optimizer with LSE-smoothed gradient, normalized to sum=1."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / S
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    best_a = a.copy()
    best_score = compute_score(a)
    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        lr_t = lr / (1.0 + 1e-4 * t)
        score, grad = lse_score_grad(a, beta_lse)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 1e-12:
            a = a / s
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def lbfgs_lse(a0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B in a-space with LSE smoothing, normalized."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, None).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S

    def objective(a_flat):
        a = np.clip(a_flat, 0.0, None)
        s = a.sum()
        if s < 1e-12:
            return 1e10, np.zeros(n)
        a = a / s
        sc, g = lse_score_grad(a, beta)
        return sc, g / s  # account for normalization

    res = minimize(objective, a0, method='L-BFGS-B', jac=True,
                   bounds=[(0.0, None)] * n,
                   options={'ftol': 1e-14, 'gtol': 1e-10,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30})
    a_out = np.clip(res.x, 0.0, None)
    s = a_out.sum()
    if s > 1e-12:
        a_out = a_out / s
    return compute_score(a_out), a_out


def coord_descent(a, n_sweeps=5, time_limit_s=60, t_start=None):
    """Coordinate descent on top-K gradient coordinates."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if t_start is not None else np.inf
    top_k = min(50, n)
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        score, grad = subgrad_score(a)
        order = np.argsort(np.abs(grad))[::-1][:top_k]
        for j in order:
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            if a_old < 1e-12:
                candidates = np.array([a.mean() * 0.5, a.mean(), a.mean() * 2.0])
            elif grad[j] > 1e-10:
                candidates = a_old * np.array([0.3, 0.5, 0.7, 0.85, 0.95])
            elif grad[j] < -1e-10:
                candidates = np.concatenate([[0.0], a_old * np.array([1.5, 2.0, 3.0, 5.0])])
            else:
                continue
            candidates = np.clip(candidates, 0.0, 1000.0)
            # Normalize after each change
            best_x = a_old
            best_sc = compute_score(a / a.sum()) if a.sum() > 1e-12 else np.inf
            for x in candidates:
                a[j] = x
                s = a.sum()
                if s < 1e-12:
                    a[j] = a_old
                    continue
                sc = compute_score(a / s)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x
    a = a / a.sum()
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a


def lp_frank_wolfe(f, n_iters=30, time_limit_s=120, t_start=None):
    """
    LP-based Frank-Wolfe iterations.
    f is normalized to sum=1.
    """
    n = len(f)
    f = f.astype(np.float64).copy()
    S = f.sum()
    if S < 1e-12:
        f[:] = 1.0 / n
    else:
        f = f / S
    best_f = f.copy()
    best_score = compute_score(f)
    deadline = t_start + time_limit_s if t_start is not None else np.inf
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        M = float(fast_conv(f).max())
        t0 = time.monotonic()
        g = lp_direction(f, M, maxiter_lp=3000)
        lp_time = time.monotonic() - t0
        if g is None or g.sum() < 1e-12:
            break
        # Rescale g to have same sum as f
        g = g / g.sum()
        # Line search: try different t values
        best_t = 0.0
        best_sc = best_score
        for t in [1.0, 0.75, 0.5, 0.35, 0.25, 0.15, 0.1]:
            f_try = (1 - t) * f + t * g
            sc = compute_score(f_try)
            if sc < best_sc:
                best_sc = sc
                best_t = t
        if best_t > 0:
            f = (1 - best_t) * f + best_t * g
            if best_sc < best_score:
                best_score = best_sc
                best_f = f.copy()
                elapsed = time.monotonic() - t_start
                print(f"    LP-FW iter {it}: score={best_score:.8f}, t={best_t:.2f}, lp_time={lp_time:.1f}s, total={elapsed:.0f}s")
        else:
            # No improvement, try smaller steps
            improved = False
            for t in [0.05, 0.03, 0.01]:
                f_try = (1 - t) * f + t * g
                sc = compute_score(f_try)
                if sc < best_score:
                    f = f_try
                    best_score = sc
                    best_f = f.copy()
                    improved = True
                    break
            if not improved:
                break
    return best_score, best_f


def make_initialization(n, kind='uniform'):
    """Create various initializations."""
    if kind == 'uniform':
        return np.ones(n) / n
    elif kind == 'cosine':
        x = np.cos(np.pi * np.arange(n) / (n - 1)) ** 2
        x = np.clip(x, 0.01, None)
        return x / x.sum()
    elif kind == 'gaussian':
        x = np.exp(-0.5 * ((np.arange(n) - n / 2) / (n / 4)) ** 2)
        return x / x.sum()
    elif kind == 'triangle':
        x = 1.0 - 2.0 * np.abs(np.arange(n) - (n - 1) / 2) / n
        x = np.clip(x, 0.01, None)
        return x / x.sum()
    elif kind == 'bump':
        x = np.exp(-0.5 * ((np.arange(n) - n * 0.3) / (n * 0.15)) ** 2) + \
            np.exp(-0.5 * ((np.arange(n) - n * 0.7) / (n * 0.15)) ** 2)
        return x / x.sum()
    else:
        return np.ones(n) / n


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for minimum of 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LSE-Adam + LP Frank-Wolfe + coordinate descent.
    """
    global global_best_a, global_best_score
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = float(budget_s) - 20.0

    best_score = np.inf
    best_a = None

    def try_update(sc, a, label):
        nonlocal best_score, best_a, global_best_score, global_best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            global_best_score = best_score
            global_best_a = best_a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.10f} | {label} | n={len(a)} | t={elapsed:.0f}s")
        return sc < best_score

    # Check parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64 and p.sum() > 0.01:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        ps = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={ps:.8f}")
        try_update(ps, parent, "parent")

    # Phase 1: Quick scan of n values and initializations
    print(f"\n{'='*60}")
    print("Phase 1: Quick scan")
    print(f"{'='*60}")
    n_candidates = [2048, 4096, 6962, 8192, 10240, 12288]
    inits = ['uniform', 'cosine', 'gaussian', 'triangle']
    for n in n_candidates:
        for init in inits:
            if time.monotonic() - t_start > effective_budget * 0.05:
                break
            a = make_initialization(n, init)
            sc = compute_score(a)
            try_update(sc, a, f"init {init} n={n}")

    print(f"  After Phase 1: best={best_score:.8f} at n={len(best_a)}")

    # Phase 2: LSE-Adam optimization on best candidates
    print(f"\n{'='*60}")
    print("Phase 2: LSE-Adam optimization")
    print(f"{'='*60}")
    # Sort candidates by score and pick top 3 n values
    unique_ns = sorted(set([len(best_a)] + [2048, 4096, 8192]))
    unique_ns = [n for n in unique_ns if n <= 16384]

    for n_opt in unique_ns[:4]:
        if time.monotonic() - t_start > effective_budget * 0.35:
            break
        # Start from best available or fresh initialization
        if best_a is not None and len(best_a) == n_opt:
            a_init = best_a / best_a.sum()
        else:
            a_init = make_initialization(n_opt, 'uniform')
        for beta in [100, 300, 800]:
            remaining = effective_budget * 0.35 - (time.monotonic() - t_start)
            if remaining < 10:
                break
            time_alloc = min(remaining, 180.0)
            t0 = time.monotonic()
            sc, a_opt = adam_lse(a_init, lr=0.008, n_steps=200000,
                                 beta_lse=beta, time_limit_s=time_alloc, t_start=t_start)
            try_update(sc, a_opt, f"Adam n={n_opt} beta={beta}")
            a_init = a_opt

    # Phase 3: L-BFGS polish
    print(f"\n{'='*60}")
    print("Phase 3: L-BFGS polish")
    print(f"{'='*60}")
    if best_a is not None:
        for beta in [500, 1500, 3000]:
            remaining = effective_budget * 0.5 - (time.monotonic() - t_start)
            if remaining < 15:
                break
            t0 = time.monotonic()
            sc, a_opt = lbfgs_lse(best_a, beta, maxiter=2000, maxfun=30000)
            try_update(sc, a_opt, f"L-BFGS beta={beta}")
            print(f"    L-BFGS beta={beta}: {sc:.10f} ({time.monotonic()-t0:.1f}s)")

    # Phase 4: LP-based Frank-Wolfe (key improvement)
    print(f"\n{'='*60}")
    print("Phase 4: LP Frank-Wolfe refinement")
    print(f"{'='*60}")
    if best_a is not None:
        n_lp = len(best_a)
        if n_lp <= 8192:
            remaining = effective_budget - (time.monotonic() - t_start)
            lp_time = min(remaining * 0.5, 900.0)
            if lp_time > 60:
                t0 = time.monotonic()
                sc, a_opt = lp_frank_wolfe(best_a, n_iters=50,
                                           time_limit_s=lp_time, t_start=t_start)
                try_update(sc, a_opt, f"LP-FW n={n_lp}")
                print(f"    LP-FW done: {sc:.10f} ({time.monotonic()-t0:.1f}s)")
                best_a = a_opt
        else:
            # For large n, try LP on a trimmed version
            print(f"    n={n_lp} too large for LP, skipping")

    # Phase 5: Second round of LP on best result
    print(f"\n{'='*60}")
    print("Phase 5: Second LP round")
    print(f"{'='*60}")
    if best_a is not None and len(best_a) <= 8192:
        remaining = effective_budget - (time.monotonic() - t_start)
        if remaining > 60:
            lp_time = min(remaining * 0.4, 600.0)
            t0 = time.monotonic()
            sc, a_opt = lp_frank_wolfe(best_a, n_iters=30,
                                       time_limit_s=lp_time, t_start=t_start)
            try_update(sc, a_opt, "LP-FW round 2")
            best_a = a_opt

    # Phase 6: Coordinate descent polish
    print(f"\n{'='*60}")
    print("Phase 6: Coordinate descent")
    print(f"{'='*60}")
    if best_a is not None:
        remaining = effective_budget - (time.monotonic() - t_start)
        if remaining > 10:
            cd_time = min(remaining * 0.3, 120.0)
            t0 = time.monotonic()
            sc, a_opt = coord_descent(best_a, n_sweeps=10,
                                      time_limit_s=cd_time, t_start=t_start)
            try_update(sc, a_opt, "Coord descent")
            best_a = a_opt

    # Phase 7: Final LSE-Adam with very high beta (nearly exact)
    print(f"\n{'='*60}")
    print("Phase 7: High-beta final polish")
    print(f"{'='*60}")
    if best_a is not None:
        remaining = effective_budget - (time.monotonic() - t_start)
        if remaining > 15:
            for beta in [2000, 5000]:
                if time.monotonic() - t_start > effective_budget - 10:
                    break
                time_alloc = min(remaining * 0.2, 60.0)
                sc, a_opt = adam_lse(best_a, lr=0.002, n_steps=50000,
                                     beta_lse=beta, time_limit_s=time_alloc, t_start=t_start)
                try_update(sc, a_opt, f"final Adam beta={beta}")
                best_a = a_opt

    # Final output
    if best_a is None:
        out = [1.0] * 64
    else:
        out = [float(max(0.0, min(1000.0, x))) for x in best_a]
        if sum(out) < 0.01:
            out = [1.0] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
