```python
import numpy as np
from scipy.optimize import linprog
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    # Variables: g[0], ..., g[n-1]
    # Objective: maximize sum(g) → minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: for each k = 0, ..., 2n-2:
    # sum_{i: valid} a[i]*g[k-i] <= M
    # This is a matrix A_ub @ g <= b_ub
    
    # Build the constraint matrix
    # For each k, the coefficient of g[j] in constraint k is:
    # a[k-j] if 0 <= k-j < n, else 0
    # i.e., a[j' ] where j' = k-j, if 0 <= k-j < n
    
    # A_ub[k, j] = a[k-j] if 0 <= k-j < n else 0
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = a[idx]
    
    b_ub = np.full(2*n-1, M)
    
    # Bounds: g[j] >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
        return np.ones(n) / n  # fallback

def lp_iteration(a, t=0.1, max_lp_time=30):
    """One LP-based iteration."""
    n = len(a)
    b = fast_conv(a)
    M = float(b.max())
    
    g = solve_lp(a, M)
    g_sum = g.sum()
    if g_sum < 1e-12:
        return a, M
    
    g = g / g_sum  # normalize to sum 1
    a = a / a.sum()  # ensure a sums to 1
    
    a_new = (1 - t) * a + t * g
    a_new = np.clip(a_new, 0, None)
    if a_new.sum() < 1e-12:
        return a, M
    a_new = a_new / a_new.sum()
    
    b_new = fast_conv(a_new)
    M_new = float(b_new.max())
    
    return a_new, M_new
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses LP-based iterations (max sum(g) s.t. conv(a,g) <= M) with multiple
initializations, adaptive step sizes, and subgradient polishing.
Theoretical lower bound is 2n/(2n-1) ~ 1, but non-negativity constrains
how flat the self-convolution can be. Uniform gives score 2.
The LP approach exploits the bilinear structure to find flatter convolutions.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_cross_conv(a, b):
    """Linear cross-convolution via FFT."""
    n1, n2 = len(a), len(b)
    M = n1 + n2 - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Score = 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def build_conv_matrix(a):
    """Build (2n-1) x n matrix A where constraint is A@g <= M element-wise."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = a
    return A


def solve_lp_for_g(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    if res.success:
        return np.clip(res.x, 0.0, None)
    return None


def subgrad_score(a):
    """Exact score and subgradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(c.argmax())
    score = 2.0 * n * M / (S * S)
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad


def lse_smoothed_grad(a, beta):
    """LSE-smoothed objective and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def adam_lse_optimize(a_init, beta, lr, n_steps, time_limit, t_start):
    """Adam with LSE smoothing."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    a = a / a.sum()

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, n_steps + 1):
        if time.monotonic() - t_start > time_limit:
            break
        lr_t = lr / (1.0 + 1e-5 * t)
        _, grad = lse_smoothed_grad(a, beta)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 1e-12:
            a = a / s

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def coordinate_descent(a, n_sweeps, top_k, time_limit, t_start):
    """Coordinate descent on top-gradient coordinates."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_score = compute_score(a)

    for sweep in range(n_sweeps):
        if time.monotonic() - t_start > time_limit:
            break
        score, grad = subgrad_score(a)
        order = np.argsort(np.abs(grad))[::-1][:top_k]
        improved = False
        for j in order:
            if time.monotonic() - t_start > time_limit:
                break
            a_old = a[j]
            if grad[j] > 1e-12:
                candidates = a_old * np.array([0.3, 0.5, 0.7, 0.85, 0.95])
            elif grad[j] < -1e-12:
                candidates = np.concatenate([[0.0], a_old * np.array([1.3, 1.7, 2.5, 4.0])])
            else:
                continue
            for x in candidates:
                a[j] = x
                sc = compute_score(a)
                if sc < score:
                    score = sc
                    improved = True
            if not improved:
                a[j] = a_old

        if not improved:
            break
        if score < best_score:
            best_score = score

    return best_score, a

            sc, grad = subgrad_score(a)
            a = a - lr * grad
            a = np.clip(a, 0.0, None)
            if a.sum() < 1e-12:
                break
            a = a / a.sum()
            if it % 500 == 0:
                sc_check = compute_score(a)
                if sc_check < cur_sc:
                    cur_sc = sc_check
                    best_a = a.copy()
                    try_update(cur_sc, a, f"subgrad_it{it}")
                else:
                    lr *= 0.9

    # === Final output ===
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [1.0] * 1024

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
```
