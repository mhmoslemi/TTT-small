import numpy as np
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cv - 1))[0][-1]
    theta = (cv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def cross_corr(p, a):
    """Compute cross-correlation (p ⊛ a)[m] = sum_k p[k] * a[k-m]."""
    n = len(a)
    len_p = len(p)
    M = n + len_p - 1
    M_fft = 1 << (M - 1).bit_length()

def fast_conv(a):
    n = len(a)
    M = 2*n - 1
    M_fft = 1 << (M-1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A*A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0*n*float(c.max())/(S*S)

def project_simplex(v):
    n = len(v)
    if n <= 1:
        return np.array([1.0]) if n == 1 else v.copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cv - 1)
    rho = np.nonzero(cond)[0][-1]
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lse_grad(a, beta):
    """Return (LSE_value, gradient) for LSE-smoothed max of conv(a,a)."""
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    scaled = beta * (b - b_max)
    # For numerical stability
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = b_max + log_sum / beta
    p = exp_scaled / exp_scaled.sum()

    P = np.fft.rfft(p, L_fft)
    A = np.fft.rfft(a, L_fft)
    cc = np.fft.irfft(P * np.conj(A), L_fft)
    g = 2.0 * cc[:n]
    return L, g

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Strategy: Start from parent, use LSE-smoothed L-BFGS-B with increasing beta,
FISTA with Nesterov momentum, mirror descent, and coordinate refinement.
Key insight: score is scale-invariant, so normalize to simplex (sum=1, a>=0)
and minimize 2*n*max(b) where b=conv(a,a).
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def project_simplex(v):
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n <= 1:
        return np.ones_like(v) if n == 1 else v.copy()
    u = np.sort(v)[::-1]
    cv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lse_obj_and_grad(a, beta):
    """
    Compute LSE-smoothed max of conv(a,a) and its gradient w.r.t. a.
    Returns (objective_value, gradient_vector).
    """
    n = len(a)
    b = fast_conv(a)
    b_max = float(b.max())
    scaled = beta * (b - b_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = b_max + log_sum / beta
    p = exp_scaled / exp_scaled.sum()
    # Gradient: 2 * cross_correlation(p, a)
    # cross_corr[m] = sum_k p[k] * a[k-m]
    L_fft = 1 << (n + len(b) - 2).bit_length()
    P = np.fft.rfft(p, L_fft)
    A = np.fft.rfft(a, L_fft)
    cc = np.fft.irfft(P * np.conj(A), L_fft)
    g = 2.0 * cc[:n]
    return L, g

def fista_phase(a0, max_time, t_start, beta_start=20.0, beta_end=5000.0):
    """FISTA (Nesterov accelerated) with LSE smoothing on the simplex."""
    n = len(a0)
    a = project_simplex(a0.astype(np.float64).copy())
    y = a.copy()
    t_mom = 1.0
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = t_start + max_time

    # Adaptive step size based on beta
    # Lipschitz constant is roughly 4*beta*max(a) + 4
    # Start conservative
    step = 1.0 / (4.0 * beta_start * 2.0)

    it = 0
    while time.monotonic() < deadline:
        it += 1
        frac = min(1.0, it / 100000.0)
        beta = beta_start * (beta_end / beta_start) ** frac
        step = 1.0 / (4.0 * beta * max(a.max(), 0.01) + 4.0)

        L_val, g = lse_obj_and_grad(y, beta)

        # Gradient step
        a_new = y - step * g
        a_new = project_simplex(a_new)

        # FISTA momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = a_new + ((t_mom - 1.0) / t_new) * (a_new - a)
        a = a_new
        t_mom = t_new

        # Periodic exact check
        if it % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_score, best_a, it

def subgrad_coord_descent(a0, max_time, t_start, top_k=50):
    """
    Coordinate descent using subgradients of the exact max function.
    Optimizes the top-k coordinates with the largest |subgradient|.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = t_start + max_time

    b = fast_conv(a)
    k_star = int(np.argmax(b))
    max_b = float(b[k_star])

    while time.monotonic() < deadline:
        # Compute subgradient at k_star
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = 2.0 * a[idx]

        # Normalize subgradient (gradient of max_b / (sum_a)^2 w.r.t. a)
        # For normalized a (sum=1), just use sg directly
        order = np.argsort(np.abs(sg))[::-1][:top_k]

        improved = False
        for j in order:
            if time.monotonic() > deadline:
                break
            old_val = a[j]
            if old_val < 1e-15:
                continue
            # Try decreasing a[j] (since sg[j] > 0 means increasing a[j] increases b[k_star])
            # But we also need to keep sum = 1, so distribute the decrease
            for frac in [0.5, 0.7, 0.85, 0.95]:
                new_val = old_val * frac
                delta = old_val - new_val
                if delta < 1e-15:
                    continue
                # Distribute delta to other coordinates proportionally
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                weights = a[mask]
                if weights.sum() < 1e-15:
                    break
                a_trial = a.copy()
                a_trial[j] = new_val
                a_trial[mask] = a[mask] + delta * weights / weights.sum()
                a_trial = np.maximum(a_trial, 0.0)
                sc = compute_score(a_trial)
                if sc < best_score - 1e-12:
                    a = a_trial
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
                # Also try increasing a[j] (if sg[j] < 0)
            if sg[j] < -1e-12:
                for frac in [1.3, 1.5, 2.0]:
                    new_val = old_val * frac
                    delta = new_val - old_val
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    weights = a[mask]
                    if weights.sum() < 1e-15:
                        break
                    a_trial = a.copy()
                    a_trial[j] = new_val
                    a_trial[mask] = a[mask] - delta * weights / weights.sum()
                    a_trial = np.maximum(a_trial, 0.0)
                    if a_trial.sum() < 1e-12:
                        break
                    sc = compute_score(a_trial)
                    if sc < best_score - 1e-12:
                        a = a_trial
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break

        # Update b and k_star
        b = fast_conv(a)
        k_star = int(np.argmax(b))
        max_b = float(b[k_star])

        if not improved:
            # No improvement, break
            break

    return best_score, best_a
