def build_active_V(a,c,K):
    n=len(a); L=len(c)
    K=min(int(K),L)

        r0=n-1-k+start

        V[start:end+1, kk] = rev[r0:r1+1]
    return idx,V

import numpy as np
import time
from scipy.optimize import minimize, lsq_linear

_DEADLINE = 0.0
_GOAL_SCORE = 1.5005


def next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def clean_normalized(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.ones(1)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12:
        return np.full_like(a, 1.0 / float(a.size))
    return a / s


def conv_self_with(a, N):
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)
    L = 2 * a.size - 1
    out = out[:L]
    np.maximum(out, 0.0, out=out)
    return out


def score_normalized(a):
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    if a.size == 0:
        return np.inf
    s = float(a.sum())
    if s <= 1e-12:
        return np.inf
    a = a / s
    N = next_pow2(2 * a.size - 1)
    c = conv_self_with(a, N)
    P = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(P) or P < 0.0:
        P = 0.0
    return float(2.0 * a.size * P)


def crossconv_weights(p, a, N2):
    n = a.size
    rev = a[::-1]
    RA = np.fft.rfft(rev, N2)
    P = np.fft.rfft(p, N2)
    out = np.fft.irfft(RA * P, N2)
    return out[n - 1:2 * n - 1]


def softmax(x):
    x = np.asarray(x, dtype=np.float64).reshape(-1)
    m = float(np.max(x)) if x.size else 0.0
    e = np.exp(np.clip(x - m, -745.0, 745.0))
    z = float(e.sum())
    if z <= 0.0 or not np.isfinite(z):
        return np.full_like(x, 1.0 / float(x.size))
    return e / z


def perturb_sequence(a, rng, scale=0.01, smooth=10):
    a = clean_normalized(a)
    n = a.size
    if scale <= 0.0 or rng is None:
        return a.copy()
    z = rng.standard_normal(n)
    if smooth > 1:
        k = max(1, int(smooth))
        kernel = np.ones(k + 1) / float(k + 1)
        z = np.convolve(z, kernel, mode="same")
        z -= float(z.mean())
    a2 = a * (1.0 + scale * z)
    return clean_normalized(a2)


def interpolate_seq(parent, n):
    parent = np.asarray(parent, dtype=np.float64).reshape(-1)
    if parent.size == n:
        return clean_normalized(parent)
    if parent.size < 2:
        x = np.linspace(0.0, 1.0, n)
        out = np.interp(x, [0.0], [float(np.sum(parent))] if parent.size else [1.0])
    else:
        x_old = np.linspace(0.0, 1.0, parent.size)
        x_new = np.linspace(0.0, 1.0, n)
        out = np.interp(x_new, x_old, parent)
    return clean_normalized(out)


def simple_initializations(n, rng):
    t = np.linspace(0.0, 1.0, n)
    raw = []
    raw.append(np.ones(n) * 0.5)
    raw.append(np.maximum(0.5 + 0.3 * np.cos(np.pi * t), 0.01))
    raw.append(np.maximum(0.5 + 0.3 * np.cos(2.0 * np.pi * t), 0.01))
    raw.append(np.maximum(0.5 + 0.3 * np.cos(np.pi * (t - 0.25)), 0.01))
    raw.append(np.maximum(0.5 + 0.3 * np.sin(np.pi * t), 0.01))
    raw.append(np.maximum(0.5 + 0.3 * np.cos(np.pi * t) + 0.1 * np.sin(2.0 * np.pi * t), 0.01))
    for _ in range(4):
        z = rng.standard_normal(n)
        z = np.convolve(z, np.ones(33) / 33.0, mode="same")
        z -= float(z.mean())
        raw.append(np.maximum(0.5 + 0.25 * z, 0.01))
    out = []
    for a in raw:
        a = clean_normalized(a)
        sc = score_normalized(a)
        if np.isfinite(sc):
            out.append(a)
    return out


def make_target(c, tau, gamma=1.0, flat_mix=0.0):
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    t = c.copy()
    tau = max(float(tau), 1.0 / L)

    if gamma is not None and abs(float(gamma) - 1.0) > 1e-12:
        t = np.power(np.maximum(t, 0.0), float(gamma))
        s = float(t.sum())
        if s > 0.0:
            t = t / s

    if flat_mix is not None and 0.0 < float(flat_mix) < 1.0:
        uniform = np.full(L, 1.0 / L)
        t = (1.0 - float(flat_mix)) * t + float(flat_mix) * uniform
        s = float(t.sum())
        if s > 0.0:
            t = t / s

    excess = np.maximum(t - tau, 0.0)
    total_excess = float(excess.sum())
    if total_excess > 0.0:
        t = np.minimum(t, tau)
        slack = np.maximum(tau - t, 0.0)
        total_slack = float(slack.sum())
        if total_slack > 0.0:
            add_total = min(total_excess, total_slack)
            t = t + add_total * (slack / total_slack)

    s = float(t.sum())
    if s <= 0.0:
        return np.ones(L) / L
    if abs(s - 1.0) > 1e-12:
        if float(np.max(t)) * s <= tau * 1.05:
            t = t / s
    return t


def square_spectrum_step(a, target, alpha=0.5, N=None):
    a = clean_normalized(a)
    n = a.size
    L = 2 * n - 1
    if N is None:
        N = next_pow2(L)
    A = np.fft.rfft(a, N)
    C = A * A
    T = np.fft.rfft(target, N)
    magC = np.abs(C)
    magT = np.abs(T)
    denom = np.where(magC > 1e-300, magC, 1.0)
    ratio = magT / denom
    ratio = np.clip(ratio, 1e-6, 1e6)
    scale = np.power(ratio, alpha)
    A2 = A * scale
    a2 = np.fft.irfft(A2, N)[:n]
    return clean_normalized(a2)


def square_spectrum_optimize(a, target_peak, n_iters=100, gamma=1.0, flat_mix=0.0, stop_time=0.0):
    a = clean_normalized(a)
    best_a = a.copy()
    best_sc = score_normalized(a)
    n = a.size
    L = 2 * n - 1
    N = next_pow2(L)

    for it in range(n_iters):
        if stop_time > 0.0 and time.monotonic() > stop_time:
            break
        c = conv_self_with(a, N)
        P = float(np.max(c)) if c.size else 0.0
        if P <= 0.0 or not np.isfinite(P):
            break
        sc = float(2.0 * n * P)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if target_peak is None:
            tau = P * 0.999
        else:
            if P <= float(target_peak):
                tau = P * 0.9995
            else:
                prog = it / max(1, n_iters - 1)
                rel = max(1e-9, float(target_peak) / P)
                taurel = rel + (1.0 - rel) * (1.0 - min(1.0, prog * 1.25))
                taurel = min(1.0, max(taurel, rel))
                tau = P * taurel

        tau = max(tau, (1.0 / L) * 1.0001)
        if tau >= P:
            tau = P * 0.999

        target = make_target(c, tau, gamma=gamma, flat_mix=flat_mix)
        alpha = 0.35 + 0.45 * (it / max(1, n_iters - 1))
        a = square_spectrum_step(a, target, alpha=alpha, N=N)

    return best_a, best_sc


def log_lse_objective(x, beta, N, N2):
    x = np.asarray(x, dtype=np.float64).reshape(-1)
    n = x.size
    if not np.all(np.isfinite(x)):
        return 1e10, np.zeros(n)
    a = softmax(x)
    c = conv_self_with(a, N)
    cmax = float(np.max(c)) if c.size else 0.0
    if not np.isfinite(cmax):
        return 1e10, np.zeros(n)
    scaled = beta * (c - cmax)
    scaled = np.clip(scaled, -745.0, 0.0)
    expv = np.exp(scaled)
    ess = float(expv.sum())
    if ess <= 0.0 or not np.isfinite(ess):
        return 1e10, np.zeros(n)
    lse = cmax + float(np.log(ess)) / beta
    p = expv / ess
    g = 2.0 * crossconv_weights(p, a, N2)
    gs = (2.0 * n) * g
    mean = float(np.dot(gs, a))
    grad_x = a * (gs - mean)
    val = float(2.0 * n * lse)
    if not np.all(np.isfinite(grad_x)):
        grad_x = np.zeros(n)
    if not np.isfinite(val):
        val = 1e10
    return val, grad_x


def run_lbfgs_log(x0, beta, maxiter, maxfun, N, N2):
    x0 = np.asarray(x0, dtype=np.float64).reshape(-1).copy()
    x0 = np.clip(x0, -60.0, 60.0)
    n = x0.size
    bounds = [(-60.0, 60.0)] * n
    try:
        res = minimize(
            log_lse_objective,
            x0,
            args=(beta, N, N2),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options={
                "ftol": 1e-12,
                "gtol": 1e-8,
                "maxiter": int(maxiter),
                "maxfun": int(maxfun),
                "maxls": 50
            }
        )
        out = softmax(res.x)
    except Exception:
        out = softmax(x0)
    return clean_normalized(out)

def build_active_V(a, c, K):
    n = a.size
    L = c.size
    K = int(max(1, min(K, L)))
    idx = np.argsort(c)[-K:][::-1]
    V = np.zeros((n, K), dtype=np.float64)
    rev = a[::-1]
    for kk, k in enumerate(idx):
        k = int(k)
        start = max(0, k - n + 1)
        end = min(k, n - 1)
        if end < start:
            continue
        r0 = n - 1 - k + start
        r1 = n - 1 - k + end
        V[start:end + 1, kk] = rev[r0:r1 + 1]
    return idx, V

        lb_budget = rem / max(1, n_lb - gi)
        lb_budget = min(max(20.0, lb_budget), 220.0)
        maxfun = int(min(4500, max(1200, lb_budget * 350)))
        maxiter = int(min(5000, max(500, maxfun // 3)))
        try:
            a2 = run_lbfgs_log(x0, beta, maxiter=maxiter, maxfun=maxfun, N=N, N2=N2)
        except Exception:
            a2 = a
        sc2 = score_normalized(a2)
        if sc2 < best_sc:
            best_sc = sc2
            best_a = a2.copy()
        a = best_a.copy()

    if active_time > 0 and time.monotonic() < t_end:
        stop3 = min(t_end, time.monotonic() + float(active_time))
        if stop3 > time.monotonic() + 5.0:
            a1, sc1 = log_active_ls(
                a, K=active_K, n_outer=active_outer,
                step_init=0.12, stop_time=stop3
            )
            if sc1 < best_sc:
                best_sc = sc1
                best_a = a1.copy()
            a = best_a.copy()

    if extra_sq_time > 0 and time.monotonic() < t_end:
        stop4 = min(t_end, time.monotonic() + float(extra_sq_time))
        if stop4 > time.monotonic() + 5.0:
            a1, sc1 = square_spectrum_optimize(
                a, target_peak, n_iters=150,
                gamma=1.0, flat_mix=0.0, stop_time=stop4
            )
            if sc1 < best_sc:
                best_sc = sc1
                best_a = a1.copy()
            a = best_a.copy()

    if extra_fo_time > 0 and time.monotonic() < t_end:
        stop5 = min(t_end, time.monotonic() + float(extra_fo_time))
        if stop5 > time.monotonic() + 5.0:
            a1, sc1 = log_first_order(
                a, top_m=0, n_steps=20000, step_init=0.04, stop_time=stop5
            )
            if sc1 < best_sc:
                best_sc = sc1
                best_a = a1.copy()
            a = best_a.copy()

    return best_a, best_sc


def try_trim(a, best_score):
    a = clean_normalized(a)
    best_a = a.copy()
    if not np.isfinite(best_score):
        best_sc = score_normalized(a)
    else:
        best_sc = float(best_score)

    maxv = float(np.max(a)) if a.size else 0.0
    if maxv <= 0.0:
        return best_a, best_sc

    for tol in (1e-8, 1e-9, 1e-10):
        mask = a > tol * maxv
        if not mask.any():
            continue
        s = int(np.argmax(mask))
        e = len(a) - int(np.argmax(mask[::-1]))
        if e - s < 16:
            continue
        a2 = clean_normalized(a[s:e])
        sc = score_normalized(a2)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    global _DEADLINE
    t0 = time.monotonic()
    _DEADLINE = t0 + float(budget_s) - 40.0
    rng = np.random.default_rng(seed)

    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        p = np.clip(p, 0.0, 1000.0)
        if p.size >= 32 and float(p.sum()) > 0.01:
            parent = p
    except Exception:
        parent = None

    goal = _GOAL_SCORE

    if parent is not None:
        best_a = clean_normalized(parent)
        best_sc = score_normalized(best_a)
        print(f"Parent loaded: n={len(best_a)}, score={best_sc:.9f}")
    else:
        best_a = None
        best_sc = np.inf
        print("No parent available; using random/sinusoid initializations.")
        for n in (4096, 6144):
            if time.monotonic() > _DEADLINE - 300.0:
                break
            inits = simple_initializations(n, rng)
            for idx, a in enumerate(inits):
                sc = score_normalized(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
        if best_a is None:
            best_a = clean_normalized(np.ones(4096))
            best_sc = score_normalized(best_a)
        print(f"Best initial (no parent): n={len(best_a)}, score={best_sc:.9f}")

    # Primary long polish on the best known sequence.
    rem = _DEADLINE - time.monotonic()
    if rem > 250.0:
        budget = min(750.0, max(80.0, rem * 0.35))
        score_goal = min(goal, max(1.0, best_sc * 0.999))
        target_peak = score_goal / (2.0 * len(best_a))
        target_peak = max(target_peak, (1.0 / (2 * len(best_a) - 1)) * 1.0001)
        print(f"\nPrimary polish: budget={budget:.0f}s, target_peak={target_peak:.10f}")
        a2, sc2 = polish_sequence(
            best_a, target_peak,
            time_budget=budget,
            sq_time=140, fo_time=140,
            lbfgs_gammas=(40, 200),
            active_time=100, active_K=32, active_outer=20,
            extra_sq_time=60, extra_fo_time=80,
            noise_scale=0.003, rng=rng
        )
        if sc2 < best_sc:
            best_sc = sc2
            best_a = a2.copy()
            print(f"Primary polish improved: {best_sc:.9f}")
        else:
            print(f"Primary polish no improvement: best={best_sc:.9f}")

    # Alternate length: often 8192 is a useful power-of-two target.
    rem = _DEADLINE - time.monotonic()
    if rem > 400.0 and len(best_a) != 8192:
        budget = min(520.0, max(80.0, rem * 0.28))
        cand = interpolate_seq(best_a, 8192)
        cand = perturb_sequence(cand, rng, 0.002, smooth=120)
        sc_cand = score_normalized(cand)
        score_goal = min(goal, max(1.0, sc_cand * 0.999))
        target_peak = score_goal / (2.0 * len(cand))
        target_peak = max(target_peak, (1.0 / (2 * len(cand) - 1)) * 1.0001)
        print(f"\nAlt n=8192 polish: budget={budget:.0f}s, candidate score={sc_cand:.9f}")
        a2, sc2 = polish_sequence(
            cand, target_peak,
            time_budget=budget,
            sq_time=90, fo_time=90,
            lbfgs_gammas=(100,),
            active_time=60, active_K=32, active_outer=12,
            extra_sq_time=40, extra_fo_time=50,
            noise_scale=0.001, rng=rng
        )
        if sc2 < best_sc:
            best_sc = sc2
            best_a = a2.copy()
            print(f"Alt n=8192 improved: {best_sc:.9f}")
        else:
            print(f"Alt n=8192 no improvement: best={best_sc:.9f}")

    # Controlled perturbations of the current best, polished briefly.
    for scale in (0.004, 0.010, 0.020):
        rem = _DEADLINE - time.monotonic()
        if rem < 260.0:
            break
        budget = min(260.0, max(60.0, rem * 0.22))
        cand = perturb_sequence(best_a, rng, scale, smooth=max(12, len(best_a) // 100))
        sc_cand = score_normalized(cand)
        score_goal = min(goal, max(1.0, sc_cand * 0.999))
        target_peak = score_goal / (2.0 * len(cand))
        target_peak = max(target_peak, (1.0 / (2 * len(cand) - 1)) * 1.0001)
        print(f"\nPerturb polish scale={scale:.4f}, budget={budget:.0f}s, candidate score={sc_cand:.9f}")
        a2, sc2 = polish_sequence(
            cand, target_peak,
            time_budget=budget,
            sq_time=40, fo_time=70,
            lbfgs_gammas=(150,),
            active_time=40, active_K=32, active_outer=8,
            extra_sq_time=20, extra_fo_time=30,
            noise_scale=0.0, rng=rng
        )
        if sc2 < best_sc:
            best_sc = sc2
            best_a = a2.copy()
            print(f"Perturb polish improved: {best_sc:.9f}")
            break
        else:
            print(f"Perturb polish no improvement: best={best_sc:.9f}")

    # Final concentrated refinement on best sequence, with sharper smoothed objectives.
    rem = _DEADLINE - time.monotonic()
    if rem > 160.0:
        budget = min(300.0, max(60.0, rem - 70.0))
        score_goal = min(goal, max(1.0, best_sc * 0.999))
        target_peak = score_goal / (2.0 * len(best_a))
        target_peak = max(target_peak, (1.0 / (2 * len(best_a) - 1)) * 1.0001)
        print(f"\nFinal polish: budget={budget:.0f}s, target_peak={target_peak:.10f}")
        a2, sc2 = polish_sequence(
            best_a, target_peak,
            time_budget=budget,
            sq_time=50, fo_time=80,
            lbfgs_gammas=(500, 1500),
            active_time=70, active_K=48, active_outer=12,
            extra_sq_time=30, extra_fo_time=50,
            noise_scale=0.002, rng=rng
        )
        if sc2 < best_sc:
            best_sc = sc2
            best_a = a2.copy()
            print(f"Final polish improved: {best_sc:.9f}")
        else:
            print(f"Final polish no improvement: best={best_sc:.9f}")

    # Opportunistic cycling until near the deadline.
    cycle = 0
    while time.monotonic() < _DEADLINE - 130.0 and cycle < 8:
        rem = _DEADLINE - time.monotonic()
        if rem < 250.0:
            break
        budget = min(200.0, max(50.0, rem - 90.0))
        if cycle % 2 == 1:
            scale = 0.003 * (cycle + 1)
            cand = perturb_sequence(best_a, rng, scale, smooth=max(10, len(best_a) // 120))
        else:
            cand = best_a.copy()
        sc_cand = score_normalized(cand)
        score_goal = min(goal, max(1.0, sc_cand * 0.999))
        target_peak = score_goal / (2.0 * len(cand))
        target_peak = max(target_peak, (1.0 / (2 * len(cand) - 1)) * 1.0001)
        print(f"\nExtra cycle {cycle + 1}: budget={budget:.0f}s, candidate score={sc_cand:.9f}")
        a2, sc2 = polish_sequence(
            cand, target_peak,
            time_budget=budget,
            sq_time=30, fo_time=60,
            lbfgs_gammas=(150,),
            active_time=40, active_K=32, active_outer=8,
            extra_sq_time=20, extra_fo_time=30,
            noise_scale=0.0, rng=rng
        )
        if sc2 < best_sc - 1e-12:
            best_sc = sc2
            best_a = a2.copy()
            cycle = 0
            print(f"Extra cycle improved: {best_sc:.9f}")
        else:
            cycle += 1
            print(f"Extra cycle no improvement: best={best_sc:.9f}")

    # Trim true/near-zero boundary slack if it helps.
    best_a, best_sc = try_trim(best_a, best_sc)

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if not out or sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = float(best_sc)

    print("=" * 60)
    print(f"FINAL internal score = {float(best_sc):.9f}")
    print(f"FINAL verified score = {verified:.9f}")
    print(f"Output length = {len(out)}")
    print(f"Elapsed = {time.monotonic() - t0:.1f} s")
    print("=" * 60)
    return out
