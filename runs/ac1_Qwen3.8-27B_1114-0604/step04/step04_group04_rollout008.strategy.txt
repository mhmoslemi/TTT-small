def evaluate_sequence(sequence):
    """Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid."""
    n = len(sequence)
    # Check sequence is a list or array
    if not isinstance(sequence, (list, np.ndarray)):
        return np.inf
    # Reject empty sequence
    if not sequence:
        return np.inf
    
    # Convert to float32 for efficiency
    seq = np.asarray(sequence, dtype=np.float32)
    
    # Check each element
    for x in seq:
        if isinstance(x, bool):
            return np.inf
        if not isinstance(x, (int, float)) or np.isnan(x) or np.isinf(x) or x < 0:
            return np.inf
    
    # Defensive clip to [0, 1000] explicitly with float checks
    seq = np.clip(seq, 0.0, 1000.0)
    # Recalculate normalized sequence if needed
    s = np.sum(seq)
    if s < 0.01:
        return np.inf
    
    # Compute linear convolution (full mode) via FFT
    len_fft = 2**np.ceil(np.log2(2*n-1)).astype(np.int32)
    a_fft = np.fft.rfft(seq, len_fft)
