"""
Improved flat-autocorrelation optimizer for non-negative sequences.
Key improvements:

2. LP alternating with line-search mixing for guaranteed monotonic improvement
3. Extended LSE-LBFGS with geometric beta schedule and multiple restarts

5. Frequency-domain error reduction as global initializer
6. Better time management and parallel strategy exploration

import numpy as np
from scipy.optimize import minimize
import time


def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def fista_lp_solve(a, M, n_iters=15000, beta=30.0):
    """
    Solve LP: max sum(x) s.t. conv(a,x) <= M, x >= 0
    using FISTA on LSE-smoothed surrogate.
    Returns x and dual_gap estimate.
    """
    n = len(a)
    x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    sum_a2 = float(np.sum(a * a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n

    L = beta * n * sum_a2 + 1e-12
    step = 0.9 / L

    prev_obj = None

    for it in range(n_iters):
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum

        # Objective tracking
        if it % 1000 == 0:
            obj = -float(x.sum()) + (z_max + np.log(exp_sum)) / beta

        # C^T @ p via sliding window: (C^T p)[j] = sum_k a_{k-j} p_k
        # = cross-correlation of a and p, specifically:
        # (a * p)_k at the relevant range
        # Actually: (C^T p)[j] = sum_k C_{k,j} p_k = sum_k a_{k-j} p_k
        # This is the cross-correlation of a and p evaluated at j
        # = (a reversed) convolved with p, at specific indices
        # More precisely: sum_k a_{k-j} p_k where 0<=k-j<n and 0<=k<2n-1
        # Let m = k-j, then k = m+j, sum_m a_m p_{m+j} = (a * p_rev)[j] ... 
        # Actually simpler: use the fact that (C^T p)[j] = sum_{k} a_{k-j} p_k
        # = sum_{m=0}^{n-1} a_m p_{m+j} (where p is defined on [0, 2n-2])
        # This is a sliding dot product of a with p starting at offset j.
        M2 = 3 * n - 2
        M2_fft = 1 << (M2 - 1).bit_length()
        a_rev = a[::-1]
        RA = np.fft.rfft(a_rev, M2_fft)
        P = np.fft.rfft(p, M2_fft)
        conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
        Ct_p = conv_res[n - 1:2 * n - 1]

        grad = Ct_p - 1.0

        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    Sx = float(x.sum())
    return x


def fista_lp_solve_v2(a, M, n_iters=20000, beta_start=10.0, beta_end=50.0):
    """
    Improved FISTA LP solver with adaptive beta ramp.
    """
    n = len(a)
    x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    sum_a2 = float(np.sum(a * a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n

    ramp_len = int(n_iters * 0.25)

    for it in range(n_iters):
        frac = min(1.0, it / max(1, ramp_len))
        beta = beta_start + (beta_end - beta_start) * frac

        L = beta * n * sum_a2 + 1e-12
        step = 0.9 / L

        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum

        M2 = 3 * n - 2
        M2_fft = 1 << (M2 - 1).bit_length()
        a_rev = a[::-1]
        RA = np.fft.rfft(a_rev, M2_fft)
        P = np.fft.rfft(p, M2_fft)
        conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
        Ct_p = conv_res[n - 1:2 * n - 1]

        grad = Ct_p - 1.0

        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    return x


def lp_alternating_v2(a, max_outer=100, fista_iters=15000, time_limit=None):
    """
    LP alternating method with adaptive mixing and line search.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    t_factor = 1.0

    for outer in range(max_outer):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        M = float(self_conv(a).max())
        x = fista_lp_solve_v2(a, M, n_iters=fista_iters)
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx

        # Line search for mixing parameter
        t_candidates = [t_factor * min(1.0, 1.0 - 0.3 * outer / max(1, max_outer))]
        t_candidates += [t_candidates[0] * 0.5, t_candidates[0] * 0.25, t_candidates[0] * 0.1]

        accepted = False
        for t in t_candidates:
            a_new = (1.0 - t) * a + t * x
            a_new = np.maximum(a_new, 0.0)
            Sn = float(a_new.sum())
            if Sn < 1e-12:
                continue
            a_new = a_new / Sn
            sc = compute_score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                no_improve = 0
            if sc < compute_score(a):
                a = a_new
                accepted = True
                break

        if not accepted:
            no_improve += 1
            t_factor *= 0.5
            if t_factor < 0.01:
                break
            # Force accept with very small step
            t = 0.01
            a_new = (1.0 - t) * a + t * x
            a_new = np.maximum(a_new, 0.0)
            Sn = float(a_new.sum())
            if Sn > 1e-12:
                a = a_new / Sn

    return best_a, best_sc


def lse_obj_grad(x, beta):
    """LSE-smoothed objective and gradient for L-BFGS."""
    n = len(x)
    c = self_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()

    # Gradient of L w.r.t. x: 2 * cross_correlate(x, p) at relevant indices
    rev_x = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RX = np.fft.rfft(rev_x, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RX * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=8000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def error_reduction(a, n_iters=150, alpha=0.75):
    """Frequency-domain error reduction."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        conv = np.fft.ifft(A * A).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.7 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M_fft, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        damp = 0.4 + 0.6 * (it / max(1, n_iters // 4))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def peak_shave(a, n_iters=300, frac=0.02):
    """
    Peak shaving: identify peak of autocorrelation, reduce contributing
    elements, redistribute mass to low-contribution elements.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = self_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)

        if j_end < j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg * seg[::-1]
        total_contrib = float(contrib.sum())
        if total_contrib < 1e-15:
            break

        # Identify top contributors
        top_count = max(3, n // 200)
        sorted_idx = np.argsort(contrib)[::-1]
        top_local = sorted_idx[:top_count]
        bottom_count = min(top_count, len(seg) - top_count)
        bottom_local = sorted_idx[len(seg) - bottom_count:]

        if bottom_count <= 0:
            break

        # Reduce top contributors
        total_red = 0.0
        for li in top_local:
            gi = j_start + li
            red = a[gi] * frac
            a[gi] = max(a[gi] - red, 0.0)
            total_red += red

        if total_red < 1e-15:
            break

        # Redistribute to bottom contributors
        per = total_red / len(bottom_local)
        for li in bottom_local:
            a[j_start + li] += per

        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def subgradient_polish(a, total_iters=5000):
    """Projected subgradient descent on exact max-conv objective on simplex."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    # Nesterov momentum
    t_mom = 1.0
    y = a.copy()
    avg_a = a.copy()
    lr_scale = 2.0 / (best_sc - 1.0 + 0.1)

    for it in range(total_iters):
        c = self_conv(y)
        M_val = float(c.max())
        k_star = int(np.argmax(c))

        # Subgradient: 2 * P_{k*} y (where P_{k*} y is the cross-correlation)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = y[j_end - (np.arange(j_end - j_start + 1))][::-1]
            # Actually: (P_k y)_i = y_{k-i} for valid i
            # So sg[i] = y[k_star - i] for i in [j_start, j_end]
            for i in range(j_start, j_end + 1):
                sg[i] = y[k_star - i]

        # Gradient of f = 2n * max / S^2 on the simplex (S=1 constraint)
        # Projected gradient: a - lr * grad, then project to simplex
        gn = float(np.linalg.norm(sg))
        if gn < 1e-15:
            break
        step = lr_scale / (np.sqrt(it + 1))
        step = min(step, 0.1)

        v = y - step * (2.0 * sg)
        # Project to simplex {x: sum(x)=1, x>=0}
        v_proj = _project_simplex(v)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = v_proj + (t_mom - 1.0) / t_new * (v_proj - y)
        y = np.maximum(y, 0.0)
        Sy = float(y.sum())
        if Sy > 1e-12:
            y = y / Sy
        avg_a = (it / (it + 1)) * avg_a + (1.0 / (it + 1)) * v_proj

        if it % 500 == 0:
            sc_y = compute_score(y)
            if sc_y < best_sc:
                best_sc = sc_y
                best_a = y.copy()
            sc_avg = compute_score(avg_a)
            if sc_avg < best_sc:
                best_sc = sc_avg
                best_a = avg_a.copy()

        t_mom = t_new

    return best_a


def _project_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    mu = np.sort(v)[::-1]
    cumsum = np.cumsum(mu)
    rho = np.nonzero(mu * np.arange(1, n + 1) > (cumsum - 1))[0][-1]
    theta = (cumsum[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))
    noise = rng.standard_normal(n)
    from scipy.ndimage import uniform_filter1d
    smooth = uniform_filter1d(noise, size=max(3, n // 50))
    inits.append(np.clip(0.5 + 0.3 * smooth, 0.01, 1000.0))
    inits.append(0.5 + 0.15 * np.cos(4.0 * np.pi * x) + 0.05 * np.cos(10.0 * np.pi * x))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-phase strategy: LP alternating (main), LSE-LBFGS, peak shaving, subgradient.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 25.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent available: {parent is not None}, len={0 if parent is None else len(parent)}")
    if parent is not None:
        p_sc = compute_score(parent)
        print(f"Parent score: {p_sc:.8f}")
        report(p_sc, parent, "parent")

    # Determine lengths
    n_main = 6962
    if parent is not None:
        n_main = len(parent)

    print(f"Main length: {n_main}")

    # ==================================================================
    # Phase 1: Global search with error reduction (15% time)
    # ==================================================================
    phase1_deadline = t_start + 0.15 * (float(budget_s) - 25.0)
    print(f"\n--- Phase 1: Global Search (Error Reduction) ---")

    lengths = [n_main]
    for n_try in [4096, 8192]:
        if n_try != n_main:
            lengths.append(n_try)

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 20:
            break
        inits = build_initializations(n, rng, parent)
        max_inits = max(2, min(len(inits), int(remaining / 15.0)))
        for i in range(min(max_inits, len(inits))):
            if time.monotonic() > phase1_deadline:
                break
            a0 = inits[i].copy()
            S0 = float(a0.sum())
            if S0 > 1e-12:
                a0 = a0 / S0
            a_er = error_reduction(a0, n_iters=120, alpha=0.75)
            sc = compute_score(a_er)
            report(sc, a_er, f"ER n={n} i={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: LP Alternating (55% time) - MAIN
    # ==================================================================
    phase2_deadline = t_start + 0.70 * (float(budget_s) - 25.0)
    print(f"\n--- Phase 2: LP Alternating ---")

    n_use = len(best_a)
    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 1e-12:
        a_work = a_work / S

    # Progressive LP alternating with increasing FISTA budget
    fista_budgets = [8000, 12000, 18000, 25000]
    max_outers = [60, 50, 40, 30]

    for fi, (f_it, mx_out) in enumerate(zip(fista_budgets, max_outers)):
        if time.monotonic() > phase2_deadline - 60:
            break
        remaining = phase2_deadline - time.monotonic()
        print(f"  LP round {fi}: fista={f_it}, max_outer={mx_out}, remaining={remaining:.0f}s")

        a_out, sc_out = lp_alternating_v2(
            a_work, max_outer=mx_out, fista_iters=f_it,
            time_limit=phase2_deadline - 30
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP r{fi}")
        a_work = a_out.copy()

    # Try other lengths if time permits
    for n in lengths:
        if n == n_use or time.monotonic() > phase2_deadline - 90:
            continue
        print(f"  LP alt length: n={n}")
        a_test = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(best_a)), best_a)
        a_test = np.maximum(a_test, 0.0)
        S = float(a_test.sum())
        if S > 1e-12:
            a_test = a_test / S
        a_out, sc_out = lp_alternating_v2(
            a_test, max_outer=30, fista_iters=10000,
            time_limit=phase2_deadline - 15
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP n={n}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: LSE-LBFGS refinement (15% time)
    # ==================================================================
    phase3_deadline = t_start + 0.85 * (float(budget_s) - 25.0)
    print(f"\n--- Phase 3: LSE-LBFGS Refinement ---")

    n_best = len(best_a)
    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 1e-12:
        a_norm = a_norm / S

    beta_schedule = [20, 50, 100, 200, 500, 1000, 2000]
    for beta in beta_schedule:
        if time.monotonic() > phase3_deadline:
            break
        sc, a_opt, _ = run_lbfgs(a_norm, beta, maxiter=6000, maxfun=150000)
        report(sc, a_opt, f"LBFGS b={beta}")
        a_norm = a_opt.copy()
        S = float(a_norm.sum())
        if S > 1e-12:
            a_norm = a_norm / S

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Peak shaving + local search (10% time)
    # ==================================================================
    print(f"\n--- Phase 4: Peak Shaving & Local Search ---")

    n_best = len(best_a)
    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 1e-12:
        a_norm = a_norm / S

    # Peak shaving
    if time.monotonic() < deadline - 120:
        a_ps = peak_shave(a_norm, n_iters=500, frac=0.015)
        sc = compute_score(a_ps)
        report(sc, a_ps, "peak_shave")

    # Perturbation strategies followed by LBFGS
    n_trials = 6
    for trial in range(n_trials):
        if time.monotonic() > deadline - 90:
            break
        strategy = trial % 3
        a_pert = best_a.copy()
        S = float(a_pert.sum())
        if S > 1e-12:
            a_pert = a_pert / S

        if strategy == 0:
            # Gaussian perturbation
            scale = 0.01 + 0.02 * rng.random()
            a_pert = a_pert * (1.0 + scale * rng.standard_normal(n_best))
        elif strategy == 1:
            # Targeted: reduce peak contributors
            c = self_conv(a_pert)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            seg = a_pert[j_start:j_end + 1]
            contrib = seg * seg[::-1]
            top_count = max(10, n_best // 100)
            top_local = np.argsort(contrib)[::-1][:top_count]
            total_red = 0.0
            for li in top_local:
                gi = j_start + li
                red = a_pert[gi] * 0.02
                a_pert[gi] = max(a_pert[gi] - red, 0.0)
                total_red += red
            if total_red > 0:
                bottom_count = len(seg) - top_count
                if bottom_count > 0:
                    bottom_local = np.argsort(contrib)[:bottom_count]
                    per = total_red / len(bottom_local)
                    for li in bottom_local:
                        a_pert[j_start + li] += per
        else:
            # Sinusoidal modulation
            freq = 1.0 + 5.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.02 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = a_pert * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        Sp = float(a_pert.sum())
        if Sp > 1e-12:
            a_pert = a_pert / Sp
            sc, a_opt, _ = run_lbfgs(a_pert, 500, maxiter=4000, maxfun=100000)
            report(sc, a_opt, f"local #{trial} s={strategy}")

    # Subgradient polish
    if time.monotonic() < deadline - 150:
        print("  Subgradient polish...")
        a_sp = subgradient_polish(best_a, total_iters=5000)
        sc = compute_score(a_sp)
        report(sc, a_sp, "subgrad")

    # Final LP polish
    if time.monotonic() < deadline - 100:
        print("  Final LP polish...")
        a_fin = best_a.copy()
        S = float(a_fin.sum())
        if S > 1e-12:
            a_fin = a_fin / S
        a_final, sc_final = lp_alternating_v2(
            a_fin, max_outer=25, fista_iters=20000,
            time_limit=deadline - 40
        )
        sc = compute_score(a_final)
        report(sc, a_final, "final LP")

    # Final LBFGS with very high beta
    if time.monotonic() < deadline - 30:
        a_norm = best_a.copy()
        S = float(a_norm.sum())
        if S > 1e-12:
            a_norm = a_norm / S
        sc, a_opt, _ = run_lbfgs(a_norm, 5000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "final LBFGS b=5000")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
