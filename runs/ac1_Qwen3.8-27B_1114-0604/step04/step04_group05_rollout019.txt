    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def conv_info(a):
    """Return convolution, max value, argmax, sum."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(c.argmax())
    M = float(c[k_star])
    S = float(a.sum())
    return c, k_star, M, S


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a: dL/da_j = sum_k p_k * d(d a*a)_k / da_j
    # (a*a)_k = sum_j a_j * a_{k-j}, so d(a*a)_k / da_j = a_{k-j} + a_{k-j} (if j != k-j)
    # Actually: d(a*a)_k / da_j = a_{k-j} (treating one slot) + a_{j} * 0 ... 
    # More carefully: (a*a)_k = sum_i a_i * a_{k-i}
    # d/d a_j (a*a)_k = a_{k-j} + a_{k-j} if j appears twice, else a_{k-j}
    # Actually for the sum: sum_i a_i * a_{k-i}, the derivative w.r.t. a_j is:
    #   a_{k-j} (from the term where i=j) + a_j (from the term where k-i=j, i.e., i=k-j)
    # But if i=j and k-i=j simultaneously (i.e., 2j=k), then it's 2*a_j.
    # In general: d(a*a)_k / da_j = a_{k-j} + (a_j if 2j==k else 0)
    # Wait no. Let me redo. (a*a)_k = sum_{i=0}^{n-1} a_i * a[k-i] where a[m]=0 if m notin [0,n-1].
    # d/d a_j: only terms where i=j or k-i=j contribute.
    # i=j: a_j * a[k-j] -> derivative is a[k-j]
    # k-i=j => i=k-j: a[k-j] * a[j] -> derivative is a[j] (if k-j != j, i.e., k != 2j)
    # If k == 2j: the term is a_j * a_j, derivative is 2*a_j.
    # So d(a*a)_k / da_j = a[k-j] + [a[j] if k != 2j else a[j]] = a[k-j] + a[j] if k != 2j
    # Wait that's not right either. Let me be very careful.
    # (a*a)_k = sum_i a_i * a[k-i]
    # When we differentiate w.r.t. a_j:
    #   Term i=j: a_j * a[k-j] contributes a[k-j]
    #   Term i=k-j (if k-j >= 0 and k-j != j): a[k-j] * a[j] contributes a[k-j]  ... NO
    #   Term i=k-j: the term is a[k-j] * a[k-(k-j)] = a[k-j] * a[j]. Derivative w.r.t. a_j is a[k-j].
    #   But wait, if k-j == j (i.e., k=2j), then the term is a_j * a_j and derivative is 2*a_j.
    #   If k-j != j, then we have two separate terms: i=j gives a[k-j], i=k-j gives a[j].
    #   So total: a[k-j] + a[j] if k != 2j, and 2*a[j] if k == 2j.
    #   But a[k-j] when k=2j is a[j], so it's a[j] + a[j] = 2*a[j]. Consistent!
    #   So in all cases: d(a*a)_k / da_j = a[k-j] + a[j] if k != 2j, else 2*a[j].
    #   Equivalently: a[k-j] + a[j] - a[j]*(1 if k==2j else 0) ... no that's getting complicated.
    #   Simpler: d(a*a)_k/da_j = a[k-j] + (a[j] if k != 2j else a[j])
    #   Hmm, let me just use the formula: d(a*a)_k/da_j = a[k-j] + a[k-j] ... NO.
    #   
    # Let me restart. (a*a)_k = sum_i a_i * a_{k-i}.
    # d/d a_j of (a*a)_k:
    #   For each i in the sum, the term is a_i * a_{k-i}.
    #   d/d a_j of a_i * a_{k-i} is nonzero when i=j or k-i=j.
    #   Case 1: i=j. Term is a_j * a_{k-j}. Derivative: a_{k-j}.
    #   Case 2: k-i=j, i.e., i=k-j. Term is a_{k-j} * a_j. Derivative: a_{k-j}.
    #   If both cases apply (i=j AND i=k-j, i.e., 2j=k): the term is a_j*a_j, derivative 2*a_j.
    #   If only case 1 applies (j != k-j, i.e., j != k/2): derivative is a_{k-j} (from case 1) + a_{k-j} (from case 2) = 2*a_{k-j}?
    #   NO WAIT. Case 2 gives derivative of a_{k-j}*a_j w.r.t. a_j = a_{k-j}.
    #   So total when 2j != k: a_{k-j} + a_{k-j} = 2*a_{k-j}? That doesn't seem right either.
    #   
    #   Let me verify with a simple example. a = [a0, a1], n=2.
    #   (a*a)_0 = a0*a0 = a0^2. d/da0 = 2*a0. d/da1 = 0.
    #   (a*a)_1 = a0*a1 + a1*a0 = 2*a0*a1. d/da0 = 2*a1. d/da1 = 2*a0.
    #   (a*a)_2 = a1*a1 = a1^2. d/da0 = 0. d/da1 = 2*a1.
    #
    #   For k=1, j=0: 2j=0 != k=1. Formula gives a[k-j] = a[1] = a1. But actual is 2*a1.
    #   So my formula is wrong!
    #   
    #   The issue: in the sum sum_i a_i * a_{k-i}, when i=j and i=k-j are DIFFERENT indices,
    #   both terms contribute. So the derivative is a_{k-j} (from i=j) + a_{k-j} (from i=k-j) 
    #   = 2*a_{k-j} when j != k-j.
    #   When j = k-j (i.e., k=2j), it's a single term a_j^2 with derivative 2*a_j = 2*a_{k-j}.
    #   So in ALL cases: d(a*a)_k/da_j = 2*a_{k-j}!
    #   
    #   Wait, that can't be right for all cases. Let me check k=0, j=0:
    #   (a*a)_0 = sum_i a_i * a_{-i} = a_0 * a_0 (only i=0 works since a is zero outside [0,n-1]).
    #   d/da_0 = 2*a_0. Formula: 2*a_{0-0} = 2*a_0. Correct!
    #   
    #   k=0, j=1: (a*a)_0 only has term i=0 (a_0*a_0). j=1 doesn't appear. Derivative = 0.
    #   Formula: 2*a_{0-1} = 2*a_{-1} = 0 (out of range). Correct!
    #   
    #   Great! So d(a*a)_k/da_j = 2*a_{k-j} where a[m] = 0 for m < 0 or m >= n.
    #   Actually wait, I need to double-check. The convolution is (a*a)_k = sum_i a_i * a_{k-i}.
    #   The derivative w.r.t. a_j: we need i=j OR k-i=j (i.e., i=k-j).
    #   If i=j is a valid index (0 <= j < n) AND k-j is also a valid index (0 <= k-j < n):
    #     Contribution from i=j: a_{k-j}
    #     Contribution from i=k-j (if k-j != j): a_j  ... WAIT NO.
    #     
    #     The term for i=k-j is: a_{k-j} * a_{k-(k-j)} = a_{k-j} * a_j.
    #     Derivative of this term w.r.t. a_j is a_{k-j}.
    #     
    #     So total = a_{k-j} + a_{k-j} = 2*a_{k-j} when j != k-j.
    #     When j = k-j: single term a_j * a_j, derivative = 2*a_j = 2*a_{k-j}.
    #
    #   So yes, d(a*a)_k/da_j = 2*a_{k-j} (with a[m]=0 for invalid m).
    #
    #   Therefore: dL/da_j = 2 * sum_k p_k * a_{k-j} = 2 * (p * a_rev)_{j} ... let me think.
    #   sum_k p_k * a_{k-j}: let m = k-j, so k = m+j. sum_m p_{m+j} * a_m.
    #   This is the cross-correlation of p and a, evaluated at offset j.
    #   (p * a_rev)(j) where a_rev(i) = a[n-1-i]? Not quite.
    #   Let's define: (p conv a)(j) = sum_k p_k * a_{k-j}. This is what we want.
    #   We can compute this via FFT: FFT of the result = FFT(p) * FFT(a) (with appropriate zero-padding).
    #   But we need the linear cross-correlation, not circular.
    #   Actually, sum_k p_k * a[k-j] for k=0..2n-2 and a[m]=0 outside [0,n-1]
    #   = sum_{m=0}^{n-1} a[m] * p[m+j] (where p has length 2n-1, padded with zeros)
    #   This is the cross-correlation of p and a, which can be computed via FFT.
    
    # Use the formula: dL/da_j = 2 * sum_k p_k * a[k-j]
    # Compute via FFT: this is the convolution of p_rev with a, or equivalently
    # the cross-correlation. Let me just compute it directly using FFT.
    # sum_k p_k * a[k-j] = (p * a_flip)(j + offset) ... let me use a different approach.
    # 
    # Actually, the simplest: d(a*a)/da is a "Jacobian-vector product" structure.
    # d(a*a)_k = 2 * sum_i (da_i) * a_{k-i} (by the formula above with da being a unit vector)
    # So dL = sum_k p_k * 2 * sum_i da_i * a_{k-i} = 2 * sum_i da_i * sum_k p_k * a_{k-i}
    # So the gradient component at i is: 2 * sum_k p_k * a_{k-i}
    # Let u_i = sum_k p_k * a_{k-i}. This is the cross-correlation of p and a.
    # u = correlate(p, a) = ifft(fft(p) * conj(fft(a))) ... or we can compute directly.
    # u_i = sum_k p_k * a_{k-i}. Let m = k-i, k = m+i: u_i = sum_m p_{m+i} * a_m.
    # This is (a * p_rev)_i where p_rev is the reverse of p... 
    # Actually u = ifft(fft(a) * fft(p)) shifted... let me just use the convolution structure.
    # u_i = sum_m a_m * p_{i+m} = (a * p)_i if we define convolution as (a*p)_i = sum_m a_m * p_{i+m}
    # But standard convolution is (a*p)_i = sum_m a_m * p_{i-m}.
    # So u_i = (a * p_rev)_i where p_rev(k) = p_{max_k - k}... this is getting confusing.
    #
    # Let me just compute it via FFT directly:
    # We want u_i = sum_{k} p_k * a_{k-i} for i = 0, ..., n-1
    # where a_m = 0 for m < 0 or m >= n, and p_k = 0 for k < 0 or k >= 2n-1.
    # 
    # Define: U = [u_0, u_1, ..., u_{n-1}]
    # u_i = sum_{k=0}^{2n-2} p_k * a_{k-i}
    # For a_{k-i} to be nonzero, we need 0 <= k-i < n, i.e., i <= k < i+n.
    # 
    # This is the cross-correlation: u = cross_correlate(p, a) evaluated at lags 0 to n-1.
    # cross_correlate(p, a)[i] = sum_k p[k+i] * a[k] = sum_k p[k+i] * a[k]
    # Let m = k: u_i = sum_m p[m+i] * a[m]
    # 
    # Via FFT: cross_correlate(p, a) = ifft(fft(p, N) * conj(fft(a, N))) where N >= 3n-2
    # Wait, I need to be careful about the definition.
    # numpy.correlate(p, a, mode='full')[i] = sum_k p[k+i-(n_a-1)] * a[k] ... it's complicated.
    #
    # Let me just use the direct FFT approach:
    # u_i = sum_m a[m] * p[m+i]
    # Define q[m] = p[m] for m in [0, 2n-2], 0 otherwise. q has length 2n-1.
    # u_i = sum_m a[m] * q[m+i] for i in [0, n-1]
    # This is the "anti-convolution" or we can write it as:
    # Let a_rev[m] = a[n-1-m]. Then:
    # u_i = sum_m a[m] * q[m+i]
    # Let j = n-1-m: u_i = sum_j a[n-1-j] * q[n-1-j+i]
    # = sum_j a_rev[j] * q[i + n - 1 - j]
    # = (a_rev * q)[i + n - 1]  (standard convolution)
    # So u = (a_rev * q)[n-1 : 2n-1] (taking the appropriate slice)
    
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    A_REV = np.fft.rfft(a_rev, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_ap = np.fft.irfft(A_REV * P, M2_fft)
    # u_i = conv_ap[i + n - 1] for i in [0, n-1]
    u = conv_ap[n - 1: 2 * n - 1]
    
    g_L = 2.0 * u  # gradient of L w.r.t. a
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def spectral_shape(a, n_iters=50, alpha=0.5, rng=None):
    """Spectral equalization to flatten the power spectrum."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S0 = float(a.sum())
    if S0 < 1e-12:
        return a
    M_fft = 1 << (2 * n - 2).bit_length()
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        mag2 = np.abs(A) ** 2
        # Target: equalize non-DC magnitudes
        if len(mag2) > 1:
            non_dc = mag2[1:]
            target = float(np.mean(non_dc))
            if target > 1e-20:
                ratio = np.ones(len(A))
                ratio[1:] = (target / np.maximum(mag2[1:], 1e-20)) ** (alpha / 2.0)
                # Clip ratio to avoid extreme changes
                ratio[1:] = np.clip(ratio[1:], 0.5, 2.0)
                A = A * ratio
        a_new = np.fft.irfft(A, M_fft)[:n]
        a_new = np.clip(a_new, 0.0, None)
        s = float(a_new.sum())
        if s < 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new * (S0 / s)
        a = a_new
    return a


def spectral_shape_grad(a, n_iters=30, alpha=0.3):
    """Spectral equalization with decreasing alpha (annealing)."""
    result = a.copy()
    for i in range(n_iters):
        al = alpha * (1.0 - i / n_iters) + 0.1
        result = spectral_shape(result, n_iters=1, alpha=al)
    return result


def multiplicative_weights(a, n_steps=5000, beta_lse=200, lr=0.01, 
                           time_budget=None, t_start=None):
    """Multiplicative weights with LSE gradient."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S0 = float(a.sum())
    if S0 < 1e-12:
        a[:] = 1.0 / n
        S0 = 1.0
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf
    
    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break
        score, grad = lse_score_grad(a, beta_lse)
        # Normalized gradient (for multiplicative update)
        g_norm = grad / (np.max(np.abs(grad)) + 1e-15)
        # Adaptive learning rate
        lr_t = lr / (1.0 + 0.001 * t)
        a = a * np.exp(-lr_t * g_norm)
        a = np.clip(a, 0.0, 1e10)
        # Renormalize to keep sum constant
        s = float(a.sum())
        if s > 1e-12:
            a = a * (S0 / s)
        if t % 1000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def run_lbfgs_lse(a0, beta, maxiter=3000, maxfun=30000, time_budget=None, t_start=None):
    """L-BFGS-B optimization with LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def objective_and_grad(x):
        return lse_score_grad(x, beta)
    
    try:
        res = minimize(
            objective_and_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0.copy()
    
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def continuation_lbfgs(a0, betas, time_budget, t_start):
    """Continuation method: increasing LSE beta."""
    a = a0.copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget
    
    for beta in betas:
        if time.monotonic() > deadline - 5:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break
        time_alloc = min(remaining * 0.5, 80.0)
        sc, a_out = run_lbfgs_lse(a, beta, maxiter=3000, maxfun=30000,
                                   time_budget=time_alloc, t_start=t_start)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            print(f"    Continuation beta={beta:g}: score={sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
        a = a_out
    
    return best_score, best_a


def lp_direction_heuristic(f, M, n_top=200):
    """
    Heuristic LP direction: for each coordinate j, compute how much we can
    increase b_j before the peak constraint is violated (assuming b_j is the
    only nonzero). Then set b proportional to this slack.
    """
    n = len(f)
    c, k_star, M_actual, S = conv_info(f)
    M = max(M, M_actual)
    
    # Find all active peaks (within 5% of max)
    threshold = 0.95 * M_actual
    active_peaks = np.where(c >= threshold)[0]
    
    # For each coordinate j, the constraint is:
    # (f*b)_k = sum_i f[k-i] * b_i <= M for all k
    # If only b_j is nonzero: (f*b)_k = f[k-j] * b_j
    # So b_j <= M / max_k f[k-j] = M / f[k-j] for the k that maximizes f[k-j]
    # Actually: (f*b)_k = f[k-j] * b_j (only one term since b has only one nonzero)
    # We need f[k-j] * b_j <= M for ALL k.
    # So b_j <= M / max_k f[k-j] = M / max(f[0:n] shifted by j)
    # max_k f[k-j] = max(f[max(0,j):min(n, j+2n-1)]) but f is only defined on [0,n-1]
    # So max_k f[k-j] = max of f over indices [max(0, k-j) for k in range(2n-1)]
    # = max of f over [0, n-1] (since as k ranges over [0, 2n-2], k-j ranges over [-j, 2n-2-j]
    #   and we take the overlap with [0, n-1])
    # = f.max() (since the range always includes the full [0,n-1] range... wait no)
    # Actually for a fixed j, the values f[k-j] for k in [0, 2n-2] are f[-j], f[1-j], ..., f[2n-2-j]
    # The valid ones (in [0, n-1]) are f[max(0,-j+0)] ... hmm this is getting complicated.
    # 
    # Simpler: for fixed j, (f*b)_k = f[k-j]*b_j where f[m]=0 for m not in [0,n-1].
    # So (f*b)_k is nonzero only when k-j is in [0,n-1], i.e., k in [j, j+n-1].
    # The max over k is: b_j * max_{k in [j, j+n-1]} f[k-j] = b_j * max_{m in [0,n-1]} f[m] = b_j * f.max()
    # Wait, that's just f.max() regardless of j! So the constraint is b_j * f.max() <= M,
    # giving b_j <= M / f.max().
    # 
    # That's for a SINGLE nonzero coordinate. For the full LP, the constraints interact.
    # The heuristic is: set b_j proportional to M / (f.max()) for all j, which is
    # just a constant vector. Not useful.
    #
    # Better heuristic: consider the constraint at the PEAK only.
    # (f*b)_{k_star} = sum_j f[k_star - j] * b_j <= M
    # We want to maximize sum b_j subject to this single constraint and b >= 0.
    # Solution: b_j = t * (1 / f[k_star - j]) where t is chosen so the constraint is tight.
    # sum_j t / f[k_star - j] ... no, that's not right either.
    # Max sum b_j s.t. sum_j f[k_star-j] b_j <= M, b_j >= 0.
    # This is: b_j = M / f[k_star-j] * (1 / sum_i 1/f[k_star-i]) ... no.
    # By LP duality or just inspection: the optimal is b_j = M * w_j where
    # w = argmax sum w_j s.t. sum f[k*-j] w_j <= 1, w >= 0.
    # Solution: put all mass on the j with smallest f[k*-j].
    # b_{j*} = M / f[k_star - j*] where j* = argmin_j f[k_star - j].
    # 
    # This is too aggressive. Let me use a softer approach:
    # b_j = M / (f[k_star - j] + eps) normalized, for "active" j only.
    
    # Better approach: use multiple peaks
    n_peaks = min(len(active_peaks), n_top)
    peak_indices = active_peaks[:n_peaks]
    
    # For each j, compute the "bottleneck": min over active peaks of M / f[k_j - j]
    # (i.e., the maximum b_j allowed by the tightest constraint)
    slacks = np.full(n, np.inf)
    for k in peak_indices:
        # Constraint: sum_j f[k-j] * b_j <= M
        # Slack for b_j: M / f[k-j] (if we only set b_j)
        # f[k-j] is valid when 0 <= k-j < n
        valid = (k >= 0) & (k < 2 * n - 1)
        for j in range(n):
            m = k - j
            if 0 <= m < n:
                f_val = f[m]
                if f_val > 1e-15:
                    slacks[j] = min(slacks[j], M / f_val)
                else:
                    slacks[j] = np.inf  # no constraint from this peak
    
    # Set b proportional to slack (but this might be too large)
    max_slack = np.max(slacks[slacks < np.inf]) if np.any(slacks < np.inf) else 1.0
    b = np.minimum(slacks, max_slack * 0.1)
    b = np.maximum(b, 0.0)
    
    s = float(b.sum())
    if s < 1e-12:
        b = np.ones(n) / n
    else:
        b = b / s
    
    return b


def lp_direction_simple(f, M):
    """
    Simple LP direction: b_j = 1/(f at the peak offset from j), normalized.
    This puts more weight where f is small at the peak's offset.
    """
    n = len(f)
    c, k_star, M_actual, S = conv_info(f)
    M = max(M, M_actual)
    
    # b_j proportional to 1 / f[k_star - j] (where valid)
    b = np.zeros(n)
    for j in range(n):
        m = k_star - j
        if 0 <= m < n:
            f_val = f[m]
            if f_val > 1e-10:
                b[j] = 1.0 / f_val
            else:
                b[j] = 1e10
        else:
            b[j] = 0.0
    
    # Also consider: we want to REDUCE the peak, so we should move mass
    # from positions that contribute to the peak to positions that don't.
    # The gradient of (f*f)_{k_star} w.r.t. f_j is 2*f[k_star-j].
    # To reduce the peak, decrease f where f[k_star-j] is large.
    # To maintain the sum, increase f where f[k_star-j] is small.
    # So direction: b_j = 1 (increase where f[k_star-j] is small)
    
    s = float(b.sum())
    if s < 1e-12:
        b = np.ones(n) / n
    else:
        b = b / s
    return b


def refine_with_lbfgs(a0, time_budget, t_start):
    """Full continuation L-BFGS refinement."""
    a = a0.copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget
    
    # Stage 1: Spectral shaping
    if time.monotonic() < deadline - 10:
        a_spectral = spectral_shape(a, n_iters=30, alpha=0.5)
        sc = compute_score(a_spectral)
        if sc < best_score:
            best_score = sc
            best_a = a_spectral.copy()
            print(f"    Spectral: {sc:.8f}")
        a = best_a
    
    # Stage 2: LSE continuation
    if time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        sc, a_opt = continuation_lbfgs(a, [50, 200, 800, 3000, 10000, 30000],
                                        remaining, t_start)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
    
    # Stage 3: Multiplicative weights polish
    if time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic()
        sc, a_opt = multiplicative_weights(best_a, n_steps=20000, beta_lse=5000,
                                            lr=0.005, time_budget=min(remaining, 30.0),
                                            t_start=t_start)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"    MW polish: {sc:.8f}")
    
    return best_score, best_a


def perturbation_spectral(a, rng, strength=0.15):
    """Spectral domain perturbation for diversity."""
    n = len(a)
    M_fft = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M_fft)
    amp_factors = 1.0 + strength * rng.standard_normal(len(A))
    amp_factors = np.clip(amp_factors, 0.3, 2.5)
    phase_shifts = strength * 0.3 * rng.standard_normal(len(A))
    A_new = A * amp_factors * np.exp(1j * phase_shifts)
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() < 1e-10:
        a_new = a.copy()
    else:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def perturbation_anti_palin(a, rng, strength=0.12):
    """Anti-palindromic perturbation to reduce middle peak."""
    n = len(a)
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    tilt = (x - center) / center
    freq = float(rng.uniform(2, 8))
    tilt += 0.5 * np.sin(2.0 * np.pi * freq * x / n)
    tilt_anti = 0.5 * (tilt - tilt[::-1])
    a_new = a + strength * tilt_anti * a.max()
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def perturbation_modulated(a, rng):
    """Modulate with a random carrier."""
    n = len(a)
    x = np.arange(n, dtype=np.float64)
    freq = float(rng.uniform(1, 10))
    phase = float(rng.uniform(0, 2 * np.pi))
    mod = 0.1 * np.sin(2.0 * np.pi * freq * x / n + phase)
    a_new = a * (1.0 + mod)
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def perturbation_gaussian(a, rng, n_bumps=4):
    """Gaussian bump perturbation."""
    n = len(a)
    a_new = a.copy()
    S0 = float(a.sum())
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.integers(max(1, n // 100), max(2, n // 15)))
        amplitude = float(rng.uniform(-0.1, 0.1)) * S0 / n
        x = np.arange(n, dtype=np.float64)
        bump = amplitude * np.exp(-0.5 * ((x - center) / width) ** 2)
        a_new += bump
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (S0 / a_new.sum())
    return a_new


def golden_section_1d(a, j, n_evals=12):
    """Golden-section search for optimal value of a[j]."""
    old_val = a[j]
    if old_val < 1e-12:
        lo = 0.0
        hi = a.max() * 0.5
    else:
        lo = 0.0
        hi = old_val * 3.0
        if hi < 1e-10:
            hi = a.max() * 0.5
        if hi > 50.0:
            hi = 50.0
    
    if hi <= lo:
        hi = lo + 1e-3
    
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    
    a[j] = c1
    f1 = compute_score(a)
    a[j] = c2
    f2 = compute_score(a)
    
    for _ in range(n_evals - 2):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            a[j] = c1
            f1 = compute_score(a)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            a[j] = c2
            f2 = compute_score(a)
    
    best_val = c1 if f1 < f2 else c2
    a[j] = best_val
    return compute_score(a)


def pair_descent(a, n_pairs=50, n_evals=10, time_budget=30.0, t_start=None):
    """
    Pair coordinate descent: for each pair (i, j), try to optimally
    redistribute mass between them (keeping sum constant).
    """
    n = len(a)
    a = a.copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if t_start is not None else np.inf
    
    c, k_star, M, S = conv_info(a)
    # Gradient-like quantity: positions that contribute most to peak
    contrib = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        contrib[js:je + 1] = a[idx]
    
    # Sort by contribution (high contributors should be decreased)
    order = np.argsort(contrib)[::-1]
    
    for _ in range(n_pairs):
        if time.monotonic() > deadline:
            break
        # Pick a pair: one high-contributor and one low-contributor
        i = int(order[int(np.random.randint(0, min(n//2, n)))])
        j = int(np.random.randint(0, n))
        if i == j:
            j = (j + 1) % n
        
        # Try to move mass from i to j
        a_old_i = a[i]
        a_old_j = a[j]
        total = a_old_i + a_old_j
        if total < 1e-15:
            continue
        
        # Search for optimal split: a[i] = t * total, a[j] = (1-t) * total
        best_t = a_old_i / total
        best_sc = best_score
        
        # Golden section on t in [0, 1]
        lo, hi = 0.0, 1.0
        gr = (math.sqrt(5.0) - 1.0) / 2.0
        c1 = lo + gr * (hi - lo)
        c2 = hi - gr * (hi - lo)
        
        a[i] = c1 * total
        a[j] = c2 * total
        f1 = compute_score(a)
        a[i] = c2 * total
        a[j] = c1 * total
        f2 = compute_score(a)
        
        for _ in range(n_evals - 2):
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = lo + gr * (hi - lo)
                a[i] = c1 * total
                a[j] = c2 * total
                f1 = compute_score(a)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = hi - gr * (hi - lo)
                a[i] = c2 * total
                a[j] = c1 * total
                f2 = compute_score(a)
        
        best_t = c1 if f1 < f2 else c2
        a[i] = best_t * total
        a[j] = (1 - best_t) * total
        a = np.clip(a, 0.0, 1000.0)
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
        else:
            # Revert if worse
            a = np.clip(a, 0.0, 1000.0)
    
    return best_score, a


def coord_descent(a, n_sweeps=5, top_k=40, time_budget=60.0, t_start=None):
    """Coordinate descent on top-k coordinates by subgradient magnitude."""
    n = len(a)
    a = a.copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if t_start is not None else np.inf
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        c, k_star, M, S = conv_info(a)
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score
        
        # Subgradient at peak
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = a[idx]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        
        order = np.argsort(np.abs(grad))[::-1][:top_k]
        for j in order:
            if time.monotonic() > deadline:
                break
            sc = golden_section_1d(a, int(j), n_evals=10)
            if sc < best_score:
                best_score = sc
    
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: Spectral shaping + LSE continuation + LP directions +
    multiplicative weights + coordinate descent.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 20.0
    
    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    if parent is not None:
        parent_score = compute_score(parent)
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, score={parent_score:.8f}")
    else:
        parent = np.ones(4096)
        parent_score = compute_score(parent)
        n_parent = len(parent)
        print(f"No parent, using uniform n={n_parent}, score={parent_score:.8f}")
    
    best_score = parent_score
    best_a = parent.copy()
    
    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False
    
    # ============================================================
    # Phase 1: Spectral shaping + quick LSE (20% budget)
    # ============================================================
    p1_budget = effective_budget * 0.20
    print(f"\n{'='*60}")
    print(f"Phase 1: Spectral shaping + LSE (budget={p1_budget:.0f}s)")
    print(f"{'='*60}")
    
    # Spectral shaping
    a_spec = spectral_shape(parent, n_iters=50, alpha=0.6)
    sc = compute_score(a_spec)
    try_update(sc, a_spec, "Spectral")
    
    # Spectral + anti-palindromic
    a_spec_anti = perturbation_anti_palin(a_spec, rng, strength=0.08)
    a_spec_anti = spectral_shape(a_spec_anti, n_iters=30, alpha=0.4)
    sc = compute_score(a_spec_anti)
    try_update(sc, a_spec_anti, "Spectral+Anti")
    
    # Quick LSE from best
    if time.monotonic() < t_start + p1_budget:
        remaining = t_start + p1_budget - time.monotonic()
        if remaining > 5:
            sc, a_opt = continuation_lbfgs(best_a, [100, 500, 2000], remaining, t_start)
            try_update(sc, a_opt, "Quick LSE")
    
    # ============================================================
    # Phase 2: Intensive LSE continuation (35% budget)
    # ============================================================
    p2_budget = effective_budget * 0.35
    print(f"\n{'='*60}")
    print(f"Phase 2: Intensive LSE continuation (budget={p2_budget:.0f}s)")
    print(f"{'='*60}")
    
    p2_start = time.monotonic()
    p2_deadline = p2_start + p2_budget
    
    # Multi-start LSE from different perturbations
    perturbations = [
        ("none", best_a.copy()),
        ("spectral_weak", perturbation_spectral(best_a, rng, strength=0.08)),
        ("anti_palin", perturbation_anti_palin(best_a, rng, strength=0.1)),
        ("modulated", perturbation_modulated(best_a, rng)),
        ("gaussian", perturbation_gaussian(best_a, rng, n_bumps=3)),
    ]
    
    time_per = p2_budget / len(perturbations)
    for name, a_init in perturbations:
        if time.monotonic() > p2_deadline - 10:
            break
        remaining = p2_deadline - time.monotonic()
        time_alloc = min(remaining * 0.9, time_per * 1.2)
        if time_alloc < 15:
            break
        sc, a_opt = refine_with_lbfgs(a_init, time_alloc, t_start)
        try_update(sc, a_opt, f"LSE: {name}")
    
    # ============================================================
    # Phase 3: Pair descent + coordinate descent (20% budget)
    # ============================================================
    p3_budget = effective_budget * 0.20
    print(f"\n{'='*60}")
    print(f"Phase 3: Pair + Coord descent (budget={p3_budget:.0f}s)")
    print(f"{'='*60}")
    
    p3_start = time.monotonic()
    p3_deadline = p3_start + p3_budget
    
    # Pair descent
    if time.monotonic() < p3_deadline - 10:
        remaining = p3_deadline - time.monotonic()
        time_alloc = min(remaining * 0.5, 60.0)
        sc, a_opt = pair_descent(best_a, n_pairs=200, n_evals=10,
                                  time_budget=time_alloc, t_start=t_start)
        try_update(sc, a_opt, "PairDescent")
    
    # Coordinate descent
    if time.monotonic() < p3_deadline - 10:
        remaining = p3_deadline - time.monotonic()
        time_alloc = min(remaining * 0.8, 60.0)
        sc, a_opt = coord_descent(best_a, n_sweeps=10, top_k=50,
                                   time_budget=time_alloc, t_start=t_start)
        try_update(sc, a_opt, "CoordDescent")
    
    # More pair descent
    if time.monotonic() < p3_deadline - 5:
        remaining = p3_deadline - time.monotonic()
        time_alloc = min(remaining, 30.0)
        sc, a_opt = pair_descent(best_a, n_pairs=100, n_evals=8,
                                  time_budget=time_alloc, t_start=t_start)
        try_update(sc, a_opt, "PairDescent2")

    # Phase 4: Diverse restarts with refinement (15% budget)
    # ============================================================
    p4_budget = effective_budget * 0.15

        ("strong_anti", lambda a, r: perturbation_anti_palin(a, r, strength=0.2)),
        ("combined", None),  # handled below
        ("mod+gauss", None),  # handled below
        ("weak_multi", lambda a, r: spectral_shape(
            perturbation_gaussian(perturbation_spectral(a, r, 0.1), r, 3), r, 20, 0.3)),
    ]
    
    for ri in range(n_restarts):
        if time.monotonic() > p4_deadline - 15:
            break
        remaining = p4_deadline - time.monotonic()
        time_alloc = min(remaining * 0.9, time_per * 1.3)
        if time_alloc < 20:
            break
        
        name, fn = strat_fns[ri % len(strat_fns)]
        if fn is not None:
            a_pert = fn(best_a, rng)
        elif name == "combined":
            a_pert = perturbation_anti_palin(perturbation_spectral(best_a, rng, 0.15), rng, 0.15)
        else:
            a_pert = perturbation_gaussian(perturbation_modulated(best_a, rng), rng, 3)
        
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        sc, a_opt = refine_with_lbfgs(a_pert, time_alloc, t_start)
        try_update(sc, a_opt, f"Restart#{ri}: {name}")
    
    # ============================================================
    # Phase 5: Final intensive optimization (10% budget)
    # ============================================================
    p5_budget = effective_budget * 0.10
    print(f"\n{'='*60}")
    print(f"Phase 5: Final optimization (budget={p5_budget:.0f}s)")
    print(f"{'='*60}")
    
    p5_deadline = t_start + effective_budget - 10.0
    
    # High-beta LSE
    if time.monotonic() < p5_deadline - 10:
        remaining = p5_deadline - time.monotonic()
        sc, a_opt = continuation_lbfgs(best_a, [5000, 20000, 50000, 100000],
                                         min(remaining, 120.0), t_start)
        try_update(sc, a_opt, "Final high-beta LSE")
    
    # Coordinate descent
    if time.monotonic() < p5_deadline - 5:
        remaining = p5_deadline - time.monotonic()
        time_alloc = min(remaining, 30.0)
        sc, a_opt = coord_descent(best_a, n_sweeps=8, top_k=60,
                                   time_budget=time_alloc, t_start=t_start)
        try_update(sc, a_opt, "Final CoordDescent")
    
    # Pair descent
    if time.monotonic() < p5_deadline - 3:
        remaining = p5_deadline - time.monotonic()
        time_alloc = min(remaining, 20.0)
        sc, a_opt = pair_descent(best_a, n_pairs=100, n_evals=10,
                                  time_budget=time_alloc, t_start=t_start)
        try_update(sc, a_opt, "Final PairDescent")
    
    # ============================================================
    # Phase 6: Try different sequence lengths (if time allows)
    # ============================================================
    if time.monotonic() < t_start + effective_budget - 30:
        print(f"\n{'='*60}")
        print(f"Phase 6: Different lengths")
        print(f"{'='*60}")
        
        n_best = len(best_a)
        for factor in [0.8, 1.2, 1.5]:
            if time.monotonic() > t_start + effective_budget - 20:
                break
            tl = int(n_best * factor)
            if tl < 256 or tl > 50000:
                continue
            x_old = np.arange(n_best, dtype=np.float64)
            x_new = np.linspace(0, n_best - 1, tl)
            a_res = np.interp(x_new, x_old, best_a)
            a_res = np.clip(a_res, 0.0, 1000.0)
            
            time_left = t_start + effective_budget - 20 - time.monotonic()
            if time_left < 15:
                break
            time_alloc = min(time_left * 0.7, 40.0)
            sc, a_opt = refine_with_lbfgs(a_res, time_alloc, t_start)
            try_update(sc, a_opt, f"n={tl}")
    
    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
