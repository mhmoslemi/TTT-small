def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

        self.m = np.zeros(dim)
        self.v = np.zeros(dim)
        self.t = 0
    
    def step(self, x, grad):
        self.t += 1
        self.m = self.beta1 * self.m + (1 - self.beta1) * grad
        self.v = self.beta2 * self.v + (1 - self.beta2) * grad**2
        m_hat = self.m / (1 - self.beta1**self.t)
        v_hat = self.v / (1 - self.beta2**self.t)
        self.lr_t = self.lr / (1 + (self.t / 2000))  # decay learning rate
        x_new = x - self.lr_t * m_hat / (np.sqrt(v_hat) + self.eps)
        return x_new


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a:
    # dL/da_j = sum_k p_k * d(conv(a,a)_k)/da_j
    # d(conv(a,a)_k)/da_j = a_{k-j} + a_{k-j} (if k-j is in [0,n-1])
    # Actually d(conv(a,a)_k)/da_j = a_{k-j} for j in [max(0,k-n+1), min(k,n-1)]
    # So dL/da_j = sum_{k: j in range} p_k * a_{k-j}
    # This is the cross-correlation of p with a, evaluated at lag j.
    # conv(p, rev_a)[n-1+j] = sum_k p_k * a_{k-j}... 
    # Let me think more carefully.
    # conv(a,a)_k = sum_{j} a_j * a_{k-j} (with 0<=j<=n-1, 0<=k-j<=n-1)
    # d(conv(a,a)_k)/da_j = a_{k-j} + a_{k-j} = 2*a_{k-j} (if we consider both terms vary)
    # Actually no: d/da_j (a_j * a_{k-j} + a_{k-j} * a_j + ...) 
    # If j != k-j: d/da_j (a_j * a_{k-j}) = a_{k-j}, and d/da_j (a_{k-j} * a_j) = a_{k-j}
    # But these are the same term, so d/da_j conv_k = 2 * a_{k-j} if j != k-j
    # If j == k-j (i.e., k = 2j): d/da_j (a_j^2) = 2*a_j
    # So in all cases: d(conv(a,a)_k)/da_j = 2 * a_{k-j} (when 0<=k-j<=n-1)
    # Wait, that's not right either. Let me be very careful.
    # conv(a,a)_k = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a_j * a_{k-j}
    # d(conv(a,a)_k)/da_i:
    #   The term a_i * a_{k-i} contributes a_{k-i} (if i != k-i) or 2*a_i (if i = k-i)
    #   The term a_{k-i} * a_i is the same term (just written differently)
    #   So: if i != k-i: contribution is a_{k-i}
    #         if i == k-i: contribution is 2*a_i = 2*a_{k-i}
    # In both cases, the contribution is a_{k-i} * (2 if i==k-i else 1)
    # Hmm, actually for i != k-i, there's only one term involving a_i, which is a_i * a_{k-i}
    # (or equivalently a_{k-i} * a_i, same thing). So the derivative is a_{k-i}.
    # For i == k-i, the term is a_i * a_i = a_i^2, derivative is 2*a_i.
    # So: d(conv(a,a)_k)/da_i = a_{k-i} * (1 + delta_{i, k-i})
    # where delta is 1 if i = k-i, else 0.
    # In most cases (i != k-i), this is just a_{k-i}.
    # The case i = k-i only happens when k is even and i = k/2.
    # For simplicity, let's just use a_{k-i} (the error is O(1/n) per peak).
    
    # Actually, for the LSE gradient, the standard approach is:
    # dL/da_j = sum_k p_k * d(conv_k)/da_j = sum_k p_k * a_{k-j} (approximately)
    # This is the cross-correlation of p and a (reversed).
    # Specifically: (p cross-correlate a)[j] = sum_k p_k * a_{k-j}
    # Which equals conv(p, a_rev)[n-1+j] where a_rev = a[::-1]
    # Wait, let me think again.
    # sum_k p_k * a_{k-j} = sum_m p_{m+j} * a_m (substituting m = k-j)
    # = (conv(p, a))[j+n-1]... no.
    # Actually, sum_k p_k * a_{k-j}: let m = k-j, so k = m+j
    # = sum_m p_{m+j} * a_m
    # This is the cross-correlation of p with a at lag j.
    # In terms of convolution: cross_correlation(p, a)[j] = (p * a_rev)[j + len(a) - 1]
    # where a_rev = a[::-1]
    # So we need conv(p, a_rev) evaluated at positions n-1+0, n-1+1, ..., n-1+(n-1)
    # i.e., positions n-1 to 2n-2 in the full convolution of length len(p)+len(a)-1 = (2n-1)+n-1 = 3n-2
    
    rev_a = a[::-1]
    M2 = 3 * n - 2  # length of conv(p, rev_a)
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_res = np.fft.irfft(P * RA, M2_fft)[n-1:2*n-1]
    
    # dL/da_j ≈ conv_res[j] (the factor of 2 from the diagonal is a small correction)
    # For a more accurate version, we'd add the diagonal correction, but let's skip it.
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (conv_res / S2 - 2.0 * L / S3)
    return score, grad


def run_adam(a0, lr=0.01, n_iters=5000, beta=100):
    """Adam optimization on LSE objective."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a.sum()
    if S < 1e-6:
        a[:] = 0.5
    a = a / a.sum()
    
    adam = Adam(n, lr=lr)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        sc, grad = lse_score_grad(a, beta)
        x_new = adam.step(a, grad)
        x_new = np.clip(x_new, 0.0, 1000.0)
        S_new = x_new.sum()
        if S_new < 1e-6:
            break
        x_new = x_new / S_new
        
        # Periodically check exact score
        if it % 100 == 0 or it == n_iters - 1:
            exact_sc = compute_score(x_new)
            if exact_sc < best_sc:
                best_sc = exact_sc
                best_a = x_new.copy()
        
        a = x_new
    
    return best_sc, best_a


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B on LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    
    def func(x):
        s, g = lse_score_grad(x, beta)
        return s, g
    
    res = minimize(
        func, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def subgradient_all_peaks(a, n_iters=5000, threshold_mult=1.0, step_init=0.03):
    """
    Subgradient using all peaks above threshold (not just top-M).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        c_mean = float(c.mean())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Weighted subgradient: weight each peak by how much it exceeds threshold
        threshold = c_mean * threshold_mult
        weights = np.maximum(c - threshold, 0)
        if weights.sum() < 1e-30:
            break
        
        # Build subgradient
        sg = np.zeros(n)
        active = weights > 1e-10
        active_idx = np.where(active)[0]
        
        for k in active_idx:
            k = int(k)
            w = float(weights[k])
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += w * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        # Normalize the subgradient
        w_sum = float(weights.sum())
        grad = (2.0 * n) * (sg / (w_sum * S2) - 2.0 * M_val / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step_eff = min(step, (sc - 1.0) / (gn * gn + 1e-30))
        step_eff = min(step_eff, 0.05)
        
        a -= step_eff * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= 0.9999
    
    return best_a

"""
Hybrid optimizer for minimizing 2n * max(conv(a,a)) / (sum(a))^2.
Strategy: Long L-BFGS runs with large-beta LSE smoothing, Gerchberg-Saxton
spectral perturbation for escaping local minima, and all-peak subgradient.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for optimization."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: dL/da_j = sum_k p_k * a_{k-j} (cross-correlation of p with a)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    RA = np.fft.rfft(rev_a, M2_fft)
    conv_res = np.fft.irfft(P * RA, M2_fft)[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (conv_res / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def func(x):
        s, g = lse_score_grad(x, beta)
        return s, g

    res = minimize(
        func, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def gs_partial(a, n_iters=300, strength=0.4):
    """
    Gerchberg-Saxton with partial spectral flattening.
    Reduces peaks in the power spectrum proportionally.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        # Adaptive strength: start stronger, get gentler
        s = strength * (0.5 + 0.5 * (1.0 - it / n_iters))

        A = np.fft.rfft(a, M_fft)
        mag = np.sqrt(A.real**2 + A.imag**2)
        phase = np.arctan2(A.imag, A.real)
        mag_mean = mag.mean()

        above = mag > mag_mean
        if not above.any():
            break

        # Reduce peaks: new_mag = mag*(1-s) + mag_mean*s
        mag_new = mag.copy()
        mag_new[above] = mag[above] * (1.0 - s) + mag_mean * s
        # Don't go too far below mean
        mag_new[above] = np.maximum(mag_new[above], mag_mean * 0.3)

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def gs_aggressive(a, n_iters=200, strength=0.7):
    """Aggressive GS: stronger peak reduction."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        s = strength

        A = np.fft.rfft(a, M_fft)
        mag = np.sqrt(A.real**2 + A.imag**2)
        phase = np.arctan2(A.imag, A.real)
        mag_mean = mag.mean()

        # Only target top 50% of bins
        thresh = np.quantile(mag, 0.5)
        above = mag > thresh

        mag_new = mag.copy()
        mag_new[above] = mag[above] * (1.0 - s) + mag_mean * s
        mag_new[above] = np.maximum(mag_new[above], mag_mean * 0.2)

        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def subgradient_all(a, n_iters=5000, threshold_mult=1.01, step_init=0.02):
    """
    Subgradient using all peaks above threshold.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        c_mean = float(c.mean())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        threshold = c_mean * threshold_mult
        weights = np.maximum(c - threshold, 0)
        w_sum = float(weights.sum())
        if w_sum < 1e-30:
            break

        # Build subgradient from all weighted peaks
        sg = np.zeros(n)
        active_idx = np.where(weights > 1e-10)[0]

        for k in active_idx:
            k = int(k)
            w = float(weights[k])
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += w * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / (w_sum * S2) - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step_eff = min(step, max((sc - 1.0), 0.01) / (gn * gn + 1e-30))
        step_eff = min(step_eff, 0.04)

        a -= step_eff * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= 0.9999

    return best_a


def coordinate_descent(a, n_sweeps=3, n_top=50):
    """
    Coordinate descent: try perturbation factors on top peak contributors.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    factors = [0.5, 0.7, 0.85, 0.92, 1.08, 1.15, 1.3, 1.5]

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg * seg[::-1]
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-12:
                continue
            cur_sc = compute_score(a)
            best_val = a_old_j
            best_local = cur_sc
            for f in factors:
                cv = a_old_j * f
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old_j) < 1e-12:
                    continue
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                sc = compute_score(a / S_tmp)
                if sc < best_local:
                    best_local = sc
                    best_val = cv
                a[j] = a_old_j
            a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.maximum(np.interp(x_new, x_old, parent), 0.01).copy())

    x = np.arange(n, dtype=np.float64) / n

    # Flat-top window (designed for flat passband)
    at_coeff = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    flat_top = np.zeros(n)
    for k, ak in enumerate(at_coeff):
        flat_top += ak * np.cos(2 * np.pi * k * (x - 0.5))
    flat_top = np.maximum(flat_top, 0.001)
    inits.append(flat_top / flat_top.sum())

    # Various structured inits
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x)) + 0.05)
    inits.append(np.maximum(0.3 + 0.7 * np.cos(np.pi * (x - 0.5))**2, 0.01))
    inits.append(np.maximum(0.4 + 0.4 * np.cos(2.0 * np.pi * 2.0 * x) + 0.1 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.maximum(0.3 + 0.5 * (1 - 2*x)**4, 0.01))

    return [np.maximum(a, 0.0).copy() / max(np.sum(np.maximum(a, 0.0)), 1e-12) for a in inits]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n * max(conv(a,a)) / (sum(a))^2.
    Returns the best sequence found within the time budget.
    """

    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:

            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine working length
    n_work = 6962
    if parent is not None:
        n_work = len(parent)

    print(f"Working length: {n_work}, Parent: {'yes' if parent is not None else 'no'}")

    # ================================================================
    # Phase 1: GS spectral flattening from diverse initializations (25%)
    # ================================================================
    phase1_end = t_start + 0.25 * (total_budget - 30.0)
    print(f"\n=== Phase 1: GS Spectral Flattening ===")

    inits = build_initializations(n_work, rng, parent)

    for i, a0 in enumerate(inits):
        if time.monotonic() > phase1_end:
            break

        # Partial GS
        a_gs = gs_partial(a0, n_iters=200, strength=0.4)
        sc = compute_score(a_gs)
        update_best(sc, a_gs, f"GS-p i={i}")

        # Aggressive GS
        if time.monotonic() < phase1_end:
            a_ag = gs_aggressive(a0, n_iters=150, strength=0.6)
            sc2 = compute_score(a_ag)
            update_best(sc2, a_ag, f"GS-a i={i}")

        # Chain: partial then aggressive
        if time.monotonic() < phase1_end:
            a_ch = gs_aggressive(a_gs, n_iters=100, strength=0.5)
            sc3 = compute_score(a_ch)
            update_best(sc3, a_ch, f"GS-c i={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # If parent available, make sure we at least have its score
    if parent is not None:
        p_sc = compute_score(parent)
        if p_sc < best_score:
            best_score = p_sc
            best_a = parent.copy()
            print(f"  [PARENT] Using parent: {p_sc:.8f}")

    if best_a is None:
        best_a = np.ones(n_work) / n_work
        best_score = compute_score(best_a)

    # ================================================================
    # Phase 2: L-BFGS with large beta (main workhorse, 45%)
    # ================================================================
    phase2_end = t_start + 0.70 * (total_budget - 30.0)
    print(f"\n=== Phase 2: L-BFGS (main) ===")

    # First pass: medium beta for quick improvement
    for beta in [500, 1000, 2000]:
        if time.monotonic() > phase2_end:
            break
        remaining_budget = phase2_end - time.monotonic()

        sc, a_opt = run_lbfgs(best_a, beta, maxiter=20000, maxfun=maxfun)
        update_best(sc, a_opt, f"LBFGS b={beta}")

    # Second pass: high beta for fine polishing
    for beta in [3000, 5000, 8000]:
        if time.monotonic() > phase2_end:
            break
        remaining_budget = phase2_end - time.monotonic()
        maxfun = int(min(100000, remaining_budget * 2000))
        if maxfun < 1000:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=15000, maxfun=maxfun)
        update_best(sc, a_opt, f"LBFGS-2 b={beta}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ================================================================
    # Phase 3: GS perturbation + L-BFGS restart (20%)
    # ================================================================
    phase3_end = t_start + 0.90 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Perturb + Restart ===")

    # Try GS perturbation to escape potential local minimum
    if time.monotonic() < phase3_end:
        a_pert = gs_partial(best_a, n_iters=50, strength=0.15)

            if time.monotonic() > phase3_end:

            noise_scale = 0.02 + 0.02 * trial

    print(f"After Phase 3: best = {best_score:.8f}")

    # ================================================================
    # Phase 4: Final polish (remaining 5%)
    # ================================================================
    print(f"\n=== Phase 4: Final Polish ===")

    # Coordinate descent
    if time.monotonic() < deadline - 15:
        a_cd = coordinate_descent(best_a, n_sweeps=2, n_top=40)
        sc = compute_score(a_cd)
        update_best(sc, a_cd, "CoordDesc")

    # Very high beta L-BFGS

        sc, a_opt = run_lbfgs(best_a, 10000, maxiter=5000, maxfun=50000)
        update_best(sc, a_opt, "LBFGS-final-b10000")

        sc = compute_score(a_fgs)
        update_best(sc, a_fgs, "GS-final")

    # ================================================================
    # Final output
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))

    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
