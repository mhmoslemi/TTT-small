def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:

    return 2.0 * n * float(np.max(c)) / (S * S)

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_obj_grad(x, beta):
    n = len(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c = fast_conv(x)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    sum_exp = float(exp_s.sum())
    if sum_exp < 1e-300:
        return 1e10, np.zeros(n)
    L = c_max + float(np.log(sum_exp)) / beta
    score = 2.0 * n * L / (S * S)
    
    p = exp_s / sum_exp
    # Cross-correlation: corr[i] = sum_k p(k+i) * a(k)
    M_corr = 3 * n - 1
    M_corr_fft = 1 << (M_corr - 1).bit_length()
    P = np.fft.rfft(p, M_corr_fft)
    X = np.fft.rfft(x, M_corr_fft)
    corr = np.fft.irfft(P * np.conj(X), M_corr_fft)[:n]
    dL = 2.0 * corr
    
    # d(score)/dx_i = 2n * (dL_i * S - 2*L) / S^3
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL * S - 2.0 * L) / S3
    return score, grad

def run_lbfgs(a0, beta, maxiter, maxfun):
    n = len(a0)
    a0 = np.maximum(a0, 0.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12, 'maxiter': maxiter,
                 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.maximum(res.x, 0.0)
    sc = compute_score(a_out)
    return sc, a_out

Adaptive LSE-LBFGS with spectral deflation and subgradient polish.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key ideas:
1. LSE (log-sum-exp) smoothing of the max function for differentiable optimization
2. Adaptive beta schedule: start smooth, sharpen over time
3. Spectral deflation (Fienup-style) between optimization phases
4. Top-M subgradient for final polish on exact objective
5. Multi-length exploration with warm starts
"""

from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed score and its gradient, for scipy minimize."""
    n = len(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    exp_s = np.exp(scaled)

    if sum_exp < 1e-300:
        sc = 2.0 * n * c_max / (S * S)
        return sc, np.zeros(n)

    S3 = S2 * S

    return score, grad
