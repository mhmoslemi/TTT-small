"""
LP-based active-set optimization for minimizing 2n*max(self-conv)/(sum)^2.
Uses linear programming to find descent directions, with adaptive active set
growth and line search. Combined with coordinate descent for polishing.
"""

import numpy as np
from scipy.optimize import linprog, minimize
from scipy.sparse import csr_matrix
import time
import math


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Linear cross-convolution via FFT."""
    n1, n2 = len(a), len(b)
    M = n1 + n2 - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def normalize(a, target=None):
    """Normalize so sum = target (default sqrt(2n))."""
    S = a.sum()
    if S < 1e-15:
        return a.copy()
    if target is None:
        target = math.sqrt(2.0 * len(a))
    return a * (target / S)


def build_lp_matrix(a, active_set):
    """
    Build sparse constraint matrix for LP.
    Row k (for k in active_set): (a*b)_k = sum_i A[k,i] * b_i
    where A[k,i] = a[k-i] for valid i.
    """
    n = len(a)
    K = len(active_set)
    rows = []
    cols = []
    data = []
    
    for idx, k in enumerate(active_set):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        for j in range(js, je + 1):
            i = k - j
            rows.append(idx)
            cols.append(i)
            data.append(a[j])
    
    return csr_matrix((data, (rows, cols)), shape=(K, n))


def solve_lp_direction(a, M, active_set, max_iters_lp=5000):
    """
    Solve: max sum(b) s.t. (a*b)_k <= M for k in active_set, b >= 0.
    Returns optimal b or None.
    """
    n = len(a)
    K = len(active_set)
    
    # Build constraint matrix
    A_mat = build_lp_matrix(a, active_set)
    
    # Objective: minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    # RHS
    b_rhs = np.full(K, M)
    
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_rhs, bounds=bounds,
                      method='highs', options={'maxiter': max_iters_lp})
        if res.success:
            return res.x
        else:
            # Try with more iterations
            res = linprog(c_obj, A_ub=A_mat, b_ub=b_rhs, bounds=bounds,
                          method='highs', options={'maxiter': max_iters_lp * 3})
            if res.success:
                return res.x
    except Exception as e:
        pass
    
    return None


def line_search_convex(a, b, ca, cab, cb, n_iter=40):
    """
    Golden section search for t in [0,1] minimizing max_k of:
    (1-t)^2 * ca_k + 2t(1-t) * cab_k + t^2 * cb_k
    """
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    
    def peak(t):
        return ((1-t)**2 * ca + 2.0*t*(1-t) * cab + t**2 * cb).max()
    
    f1 = peak(c1)
    f2 = peak(c2)
    
    for _ in range(n_iter):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = peak(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = peak(c2)
    
    return c1 if f1 < f2 else c2


def lp_iteration(a, K, n_ls_iter=40):
    """
    One LP-based iteration.
    a: current sequence (will be normalized internally)
    K: size of active set
    Returns: new_a, score_before, score_after
    """
    n = len(a)
    a_norm = normalize(a)
    S_target = math.sqrt(2.0 * n)
    
    c = fast_conv(a_norm)
    M = float(c.max())
    score_before = M  # since sum = sqrt(2n), score = 2n*M/(2n) = M
    
    # Active set: top K positions
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:K]
    
    # Solve LP
    b = solve_lp_direction(a_norm, M * (1 + 1e-10), active)
    
    if b is None or b.sum() < 1e-10:
        return a, score_before, score_before
    
    # Normalize b
    b = normalize(b)
    
    # Precompute convolutions for line search
    cb = fast_conv(b)
    cab = cross_conv(a_norm, b)
    
    # Line search
    t = line_search_convex(a_norm, b, c, cab, cb, n_ls_iter)
    
    # Update
    a_new = (1 - t) * a_norm + t * b
    a_new = np.clip(a_new, 0.0, 1000.0)
    
    # Renormalize
    a_new = normalize(a_new)
    
    score_after = float(fast_conv(a_new).max())
    
    a_out = a_new * (a.sum() / a_new.sum()) if a_new.sum() > 1e-12 else a
    
    return a_out, score_before, score_after


def coord_descent_polish(a, n_sweeps=3, top_k=30, time_budget=None, t_start=None):
    """Coordinate descent with golden section on top-k coordinates."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = np.inf
    if t_start is not None and time_budget is not None:
        deadline = t_start + time_budget
    
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        M = c.max()
        S = a.sum()
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score
        
        # Subgradient
        k_star = int(c.argmax())
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        sg = np.zeros(n)
        idx = k_star - np.arange(js, je + 1)
        sg[js:je+1] = a[idx]
        
        S2 = S * S

        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        
        order = np.argsort(np.abs(grad))[::-1][:top_k]

            if time.monotonic() > deadline:

def spectral_perturb(a, rng, strength=0.15):
    """Perturb in DFT domain."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    
    amp_factors = 1.0 + strength * rng.standard_normal(len(A))
    amp_factors = np.clip(amp_factors, 0.3, 2.0)
    phase_shifts = strength * 0.5 * rng.standard_normal(len(A))
    
    A_new = A * amp_factors * np.exp(1j * phase_shifts)
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.clip(a_new, 0.0, 1000.0)
    
    if a_new.sum() < 1e-10:
        a_new = a.copy()
    else:
        a_new = a_new * (a.sum() / a_new.sum())
    
    return a_new

- Actually, for non-negative sequences, anti-palindromic means a_j = C - a_{n-1-j} for some C. This would make $(a*a)_{n-1} = \sum_j a_j(C - a_j) = C \cdot S - \sum a_j^2$, which can be small if C is chosen well.

Wait, let me compute: if $a_j = C - a_{n-1-j}$, then $\sum_j a_j = Cn - \sum_j a_{n-1-j} = Cn - S$, so $S = Cn/2$, giving $C = 2S/n$.

Then $(a*a)_{n-1} = \sum_j a_j a_{n-1-j} = \sum_j a_j (C - a_j) = C S - \sum a_j^2 = C \cdot Cn/2 - \sum a_j^2 = C^2 n/2 - \sum a_j^2$.

With $C = 2S/n$: $= (4S^2/n^2) \cdot n/2 - \sum a_j^2 = 2S^2/n - \sum a_j^2$.

By Cauchy-Schwarz, $\sum a_j^2 \geq S^2/n$, so $(a*a)_{n-1} \leq 2S^2/n - S^2/n = S^2/n$.

For the uniform sequence, $(a*a)_{n-1} = S^2/n$. For the anti-palindromic sequence, we can potentially get $(a*a)_{n-1} < S^2/n$ by making $\sum a_j^2$ larger (more concentrated).

But we also need to check other positions of the convolution. The anti-palindromic constraint only helps at position $n-1$. At other positions, it might not help or could even hurt.

"""
LP-based active-set optimization for minimizing 2n*max(self-conv)/(sum)^2.
Uses linear programming to find descent directions with adaptive active set growth,
combined with line search, coordinate descent polishing, and spectral perturbations.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time
import math


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Linear cross-convolution via FFT."""
    n1, n2 = len(a), len(b)
    M = n1 + n2 - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def normalize(a, target=None):
    """Normalize so sum = target (default sqrt(2n))."""
    S = a.sum()
    if S < 1e-15:
        return a.copy()
    if target is None:
        target = math.sqrt(2.0 * len(a))
    return a * (target / S)


def build_lp_matrix(a, active_set):
    """Build sparse constraint matrix: row k has A[k,i] = a[k-i] for valid i."""
    n = len(a)
    K = len(active_set)
    rows = []
    cols = []
    data = []
    for idx, k in enumerate(active_set):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        for j in range(js, je + 1):
            i = k - j
            rows.append(idx)
            cols.append(i)
            data.append(a[j])
    return csr_matrix((data, (rows, cols)), shape=(K, n))


def solve_lp_direction(a, M, active_set):
    """Solve: max sum(b) s.t. (a*b)_k <= M for k in active_set, b >= 0."""
    n = len(a)
    K = len(active_set)
    A_mat = build_lp_matrix(a, active_set)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    b_rhs = np.full(K, M)
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_rhs, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def line_search_convex(ca, cab, cb, n_iter=45):
    """Golden section for t in [0,1] minimizing max of quadratics."""
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    lo, hi = 0.0, 1.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)

    def peak(t):
        return ((1.0 - t) ** 2 * ca + 2.0 * t * (1.0 - t) * cab + t ** 2 * cb).max()

    f1 = peak(c1)
    f2 = peak(c2)
    for _ in range(n_iter):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            f1 = peak(c1)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = peak(c2)
    return c1 if f1 < f2 else c2


def lp_iteration(a, K):
    """One LP-based iteration. Returns (new_a, score_before, score_after)."""
    n = len(a)
    a_norm = normalize(a)
    c = fast_conv(a_norm)
    M = float(c.max())
    score_before = M

    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:K]

    b = solve_lp_direction(a_norm, M * (1.0 + 1e-9), active)
    if b is None or b.sum() < 1e-10:
        return a, score_before, score_before

    b = normalize(b)
    cb = fast_conv(b)
    cab = cross_conv(a_norm, b)
    t = line_search_convex(c, cab, cb, n_iter=45)

    if t < 1e-6:
        return a, score_before, score_before

    a_new = (1.0 - t) * a_norm + t * b
    a_new = np.clip(a_new, 0.0, 1000.0)
    a_new = normalize(a_new)
    score_after = float(fast_conv(a_new).max())
    a_out = a_new * (a.sum() / a_new.sum()) if a_new.sum() > 1e-12 else a
    return a_out, score_before, score_after


def coord_descent_polish(a, n_sweeps=3, top_k=30, time_budget=None, t_start=None):
    """Coordinate descent with golden section on top-k coordinates."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = np.inf
    if t_start is not None and time_budget is not None:
        deadline = t_start + time_budget
    gr = (math.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        M = c.max()
        S = a.sum()
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score
        k_star = int(c.argmax())
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        sg = np.zeros(n)
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = a[idx]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        order = np.argsort(np.abs(grad))[::-1][:top_k]
        for j in order:
            if time.monotonic() > deadline:
                break
            j = int(j)
            a_old = a[j]
            lo = 0.0
            hi = a_old * 4.0 if a_old > 1e-10 else a.max() * 2.0
            if hi < 1e-10:
                hi = 1.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            a[j] = c1
            f1 = compute_score(a)
            a[j] = c2
            f2 = compute_score(a)
            for _ in range(15):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    a[j] = c1
                    f1 = compute_score(a)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    a[j] = c2
                    f2 = compute_score(a)
            a[j] = c1 if f1 < f2 else c2
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
    return best_score, a


def spectral_perturb(a, rng, strength=0.15):
    """Perturb in DFT domain for diversity."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    amp_factors = 1.0 + strength * rng.standard_normal(len(A))
    amp_factors = np.clip(amp_factors, 0.3, 2.0)
    phase_shifts = strength * 0.5 * rng.standard_normal(len(A))
    A_new = A * amp_factors * np.exp(1j * phase_shifts)
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() < 1e-10:
        a_new = a.copy()
    else:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def run_lp_method(a_init, t_start, time_budget, max_K=500, start_K=20,
                  K_growth=1.25, stuck_limit=8):
    """Main LP-based iterative method with adaptive active set."""
    a = a_init.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = t_start + time_budget
    K = start_K
    stuck = 0
    iters = 0
    print(f"  LP: n={len(a)}, init_score={best_score:.8f}, K_max={max_K}")

    while time.monotonic() < deadline - 3:
        a_new, sc_b, sc_a = lp_iteration(a, K)
        iters += 1
        exact = compute_score(a_new)
        if exact < best_score - 1e-12:
            best_score = exact
            best_a = a_new.copy()
            a = a_new
            stuck = 0
        else:
            a = a_new
            stuck += 1
        if iters % 100 == 0:
            print(f"    iter={iters}, K={K}, score={best_score:.8f}, t={time.monotonic()-t_start:.0f}s")
        if stuck >= stuck_limit:
            if K < max_K:
                K = min(int(K * K_growth), max_K)
                stuck = 0
            else:
                a = spectral_perturb(a, np.random.default_rng(iters), 0.1)
                K = start_K
                stuck = 0
    return best_score, best_a
