"""
Hybrid optimizer for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
Key improvements:
1. FFT-based spectral flattening (novel) - directly targets flat power spectrum
2. Palindrome constraint with half-space optimization
3. LSE continuation with Adam for smooth optimization
4. Direct subgradient method for exact max (no smoothing bias)
5. L-BFGS polishing at high beta
6. Multi-n exploration to find optimal length
7. Adaptive step-size with momentum

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Linear cross-convolution via FFT."""
    n_a, n_b = len(a), len(b)
    M = n_a + n_b - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

    # Actually the cleanest: d(conv(a,a))[k]/da[j] = a[k-j] + a[j-k] (zero-padded)

def lse_grad_pal(half_a, beta):

    n = len(a)
    grad_half = grad_full[:n_half] + grad_full[n - 1: n_half - 1: -1]

def subgrad_score_grad(a):
    """Exact score and subgradient at the max lag."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(c.argmax())
    score = 2.0 * n * M / (S * S)
    
    # Subgradient of max at k_star:
    # d(conv(a,a))[k_star]/da[j] = a[k_star-j] + a[j-k_star] (zero-padded)
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = a[idx]
        # Also add the symmetric part
        js2 = max(0, n - 1 - k_star)
        je2 = min(n - 1, n - 1 - k_star + (je - js))
        # Actually for self-conv, d/da[j] of sum_i a[i]*a[k-i] = a[k-j] + a[j-k]
        # The term a[j-k] is non-zero when 0 <= j-k < n, i.e., k <= j < k+n
        # Since k = k_star < 2n-1, j-k is in [0,n-1] when j in [k_star, k_star+n-1] ∩ [0,n-1]
        # = [k_star, n-1] if k_star < n
        # So: sg[j] += a[j-k_star] for j in [k_star, n-1]
        if k_star < n:
            sg[k_star:n] += a[:n - k_star]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad


def fft_spectral_flatten(a, n_iters=2000, beta_start=0.1, beta_end=0.5, time_budget=30.0, t_start=None):
    """
    Iterative frequency-domain spectral flattening.
    Makes the power spectrum |FFT(a)|^2 as flat as possible,
    which corresponds to flattening the autocorrelation.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return np.inf, a
    a = a / S
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    M_fft = 1 << (2 * n - 1).bit_length()
    
    t0 = time.monotonic() if t_start is None else t_start
    for it in range(n_iters):
        if time.monotonic() - t0 > time_budget:
            break
        
        # Animate beta
        frac = it / max(n_iters, 1)
        beta = beta_start + (beta_end - beta_start) * frac
        
        A = np.fft.rfft(a, M_fft)
        mag = np.abs(A)
        mag_sq = mag ** 2
        mag_mean = mag_sq.mean()
        
        if mag_mean < 1e-20:
            break
        
        # Flatten: A_new = A * (mag_mean / mag_sq)^(beta/2)
        # This reduces dynamic range of |A|
        ratio = mag_mean / np.maximum(mag_sq, 1e-20)
        scale = ratio ** (beta / 2.0)
        A_new = A * scale
        
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-12:
            break
        a_new = a_new / sn
        
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        # If no improvement in a while, reduce beta
        a = a_new
    
    return best_score, best_a


def fft_flatten_pal(half_a, n_iters=2000, beta_start=0.1, beta_end=0.5, time_budget=30.0, t_start=None):
    """Spectral flattening preserving palindrome structure."""
    n_half = len(half_a)
    a = np.concatenate([half_a, half_a[::-1]])
    sc, a_flat = fft_spectral_flatten(a, n_iters, beta_start, beta_end, time_budget, t_start)
    
    # Re-project to palindrome
    n = len(a_flat)
    half_flat = (a_flat[:n_half] + a_flat[::-1][:n_half]) / 2.0
    sf = half_flat.sum()
    if sf < 1e-12:
        return sc, half_a
    half_flat = half_flat / sf
    
    a_pal = np.concatenate([half_flat, half_flat[::-1]])
    sc_pal = compute_score(a_pal)
    
    if sc_pal < sc:
        return sc_pal, half_flat
    return sc, half_flat


def adam_lse(a_init, beta_schedule, lr, n_steps, time_budget, t_start=None, normalize=True):
    """Adam with LSE smoothing."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    n_beta = len(beta_schedule)
    steps_per_beta = max(1, n_steps // n_beta)
    
    t0 = t_start if t_start is not None else time.monotonic()
    
    for bi, beta in enumerate(beta_schedule):
        steps = min(steps_per_beta, n_steps - bi * steps_per_beta)
        for t in range(steps):
            if time.monotonic() - t0 > time_budget:
                return best_score, best_a
            
            lr_t = lr / (1.0 + 1e-5 * t)
            score, grad = lse_score_grad(a, beta)
            
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            t_global = bi * steps_per_beta + t + 1
            m_hat = m / (1.0 - b1 ** t_global)
            v_hat = v / (1.0 - b2 ** t_global)
            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, 1e10)
            
            if normalize:
                s = a.sum()
                if s > 1e-12:
                    a = a / s
            
            if t % 2000 == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a


def adam_lse_pal(half_init, beta_schedule, lr, n_steps, time_budget, t_start=None):
    """Adam with LSE in palindrome half-space."""
    n_half = len(half_init)
    h = half_init.astype(np.float64).copy()
    if h.sum() < 1e-12:
        h[:] = 1.0 / n_half
    
    m = np.zeros(n_half)
    v = np.zeros(n_half)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_h = h.copy()
    best_score = compute_score(np.concatenate([h, h[::-1]]))
    
    n_beta = len(beta_schedule)
    steps_per_beta = max(1, n_steps // n_beta)
    t0 = t_start if t_start is not None else time.monotonic()
    
    for bi, beta in enumerate(beta_schedule):
        steps = min(steps_per_beta, n_steps - bi * steps_per_beta)
        for t in range(steps):
            if time.monotonic() - t0 > time_budget:
                return best_score, best_h
            
            lr_t = lr / (1.0 + 1e-5 * t)
            a_full = np.concatenate([h, h[::-1]])
            sc, grad_full = lse_score_grad(a_full, beta)
            grad_half = grad_full[:n_half] + grad_full[n_half: n_half * 2][::-1]
            # Actually: grad_half[i] = grad_full[i] + grad_full[2*n_half-1-i]
            n = 2 * n_half
            grad_half = grad_full[:n_half] + grad_full[n-1: n_half-1: -1]
            
            m = b1 * m + (1.0 - b1) * grad_half
            v = b2 * v + (1.0 - b2) * grad_half * grad_half
            t_global = bi * steps_per_beta + t + 1
            m_hat = m / (1.0 - b1 ** t_global)
            v_hat = v / (1.0 - b2 ** t_global)
            h = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            h = np.clip(h, 0.0, 1e10)
            
            s = h.sum()
            if s > 1e-12:
                h = h / s
            
            if t % 2000 == 0:
                a_f = np.concatenate([h, h[::-1]])
                sc = compute_score(a_f)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()
    
    a_full = np.concatenate([h, h[::-1]])
    sc_final = compute_score(a_full)
    if sc_final < best_score:
        best_score = sc_final
        best_h = h.copy()
    
    return best_score, best_h


def subgrad_momentum(a_init, lr, n_iters, time_budget, t_start=None, momentum=0.9):
    """Subgradient method with momentum for the exact max."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        S = a.sum()
    a = a / S
    
    vel = np.zeros(n)
    best_a = a.copy()
    best_score = compute_score(a)
    
    t0 = t_start if t_start is not None else time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_budget:
            break
        
        score, grad = subgrad_score_grad(a)
        
        # Adaptive learning rate
        lr_t = lr / (1.0 + 0.01 * it)
        
        vel = momentum * vel - lr_t * grad
        a = a + vel
        a = np.clip(a, 0.0, 1e10)
        
        s = a.sum()
        if s > 1e-12:
            a = a / s
        
        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a


def lbfgs_optimize(a0, beta, maxiter=5000, maxfun=80000):
    """L-BFGS-B with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        x_c = np.clip(x, 0.0, 1e10)
        sc, gr = lse_score_grad(x_c, beta)
        return sc, gr
    
    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def lbfgs_pal(half0, beta, maxiter=5000, maxfun=80000):
    """L-BFGS-B in palindrome half-space."""
    n_half = len(half0)
    h0 = np.clip(half0.astype(np.float64), 0.0, 1000.0).copy()
    if h0.sum() < 1e-6:
        h0[:] = 1.0 / n_half
    bounds = [(0.0, 1000.0)] * n_half
    
    def obj(h):
        h_c = np.clip(h, 0.0, 1e10)
        a_full = np.concatenate([h_c, h_c[::-1]])
        sc, grad_full = lse_score_grad(a_full, beta)
        n = len(a_full)
        grad_half = grad_full[:n_half] + grad_full[n-1: n_half-1: -1]
        return sc, grad_half
    
    res = minimize(obj, h0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    h_out = np.clip(res.x, 0.0, 1000.0)
    a_full = np.concatenate([h_out, h_out[::-1]])
    exact_sc = compute_score(a_full)
    return exact_sc, h_out


def coord_descent(a, n_sweeps=50, time_budget=30.0, t_start=None):
    """Coordinate descent targeting peak reduction."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    t0 = t_start if t_start is not None else time.monotonic()
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_budget:
            break
        
        c = fast_conv(a)
        k_star = int(c.argmax())
        score = compute_score(a)
        
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        
        # Try reducing peak contributors
        for j in range(js, je + 1):
            if time.monotonic() - t0 > time_budget:
                break
            a_old = a[j]
            if a_old < 1e-10:
                continue
            for frac in [0.8, 0.6, 0.4]:
                a[j] = a_old * frac
                removed = a_old - a[j]
                non_peak = np.ones(n, dtype=bool)
                non_peak[js:je + 1] = False
                if non_peak.sum() > 0:
                    a[non_peak] += removed / non_peak.sum()
                sc = compute_score(a)
                if sc < score:
                    score = sc
                    if sc < best_score:
                        best_score = sc
                        a_best = a.copy()
                    break
            else:
                a[j] = a_old
    
    a[best_score < compute_score(a)] = 0  # noop, just to keep linter happy
    if best_score < compute_score(a):
        pass  # a_best was saved
    return best_score, a


def generate_inits(rng, n, n_half=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n) / n
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Edge-concentrated (important for score < 2)
    edge = np.exp(-((x - 0.5) / 0.3) ** 2) * 0.3 + 0.7
    # Actually we want mass at edges:
    edge = np.exp(-0.5 * ((x - 0.0) / 0.15) ** 2) + np.exp(-0.5 * ((x - 1.0) / 0.15) ** 2)
    inits.append(edge / edge.sum())
    
    # Two-edge with gap
    gap = 0.1 + 0.2 * rng.random()
    e2 = np.exp(-0.5 * ((x - gap) / 0.1) ** 2) + np.exp(-0.5 * ((x - 1 + gap) / 0.1) ** 2)
    inits.append(e2 / e2.sum())
    
    # Raised cosine (centered)
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(rc / rc.sum())
    
    # Gaussian centered
    sigma = 0.1 + 0.1 * rng.random()
    g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
    inits.append(g / g.sum())
    
    # Multi-edge
    n_edges = 2 + int(2 * rng.random())
    centers = np.sort(rng.uniform(0, 1, n_edges))
    me = np.zeros(n)
    for c in centers:
        me += np.exp(-0.5 * ((x - c) / 0.08) ** 2)
    inits.append(me / me.sum())
    
    # Sinusoidal
    freq = 2 + 6 * rng.random()
    s = 1.0 + 0.4 * np.cos(2 * np.pi * freq * x)
    s = np.maximum(s, 0.01)
    inits.append(s / s.sum())
    
    # Random smooth
    r = np.abs(rng.standard_normal(n)) + 0.5
    k = max(3, n // 50)
    r_s = np.convolve(r, np.ones(k) / k, mode='same')
    inits.append(r_s / r_s.sum())
    
    return [np.clip(x_i, 1e-10, 1000.0) for x_i in inits]


def make_palindromic(a):
    """Make a sequence palindromic by averaging with its reverse."""
    n = len(a)
    if n % 2 == 1:
        a = a[:-1]
        n = len(a)
    n_half = n // 2
    half = (a[:n_half] + a[::-1][:n_half]) / 2.0
    return np.concatenate([half, half[::-1]])


def resample(a, new_n):
    """Resample sequence to new length."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400):

        return False

    print(f"Phase 1: Initialization & Screening (budget={p1_time:.0f}s)")
    print(f"{'='*60}")
    
    # Choose working n values
    if parent is not None:
        n_parent = len(parent)
        print(f"Parent: n={n_parent}, score={compute_score(parent):.8f}")
        try_update(compute_score(parent), parent, "parent")
        # Try palindromic version
        pal_parent = make_palindromic(parent)
        sc_pal = compute_score(pal_parent)
        print(f"  Palindromized: {sc_pal:.8f}")
        try_update(sc_pal, pal_parent, "parent-pal")
        
        n_candidates = [n_parent]
        # Also try nearby sizes
        for delta in [-512, -256, 256, 512, 1024]:

            if nc >= 1024 and nc <= 32768:

    else:

    print(f"  Best after Phase 1: {best_score:.8f}")
    
    # ==========================================
    # Phase 2: FFT Spectral Flattening (25% time)
    # ==========================================
    p2_time = effective_budget * 0.25
    print(f"\n{'='*60}")
    print(f"Phase 2: FFT Spectral Flattening (budget={p2_time:.0f}s)")
    print(f"{'='*60}")
