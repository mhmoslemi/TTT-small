    # grad[j] = 2 * sum_{m=0}^{n-1} a[m] * p[m+j]
    # 
    # I'll compute this using FFT by padding both to length 3n-1 (or next power of 2)
    # Let a_pad = [a, 0, 0, ..., 0] (length 3n-1)
    # Let p_pad = [0, 0, ..., 0, p, 0, 0, ..., 0] ... no this doesn't work directly.
    # 
    # OK let me just use the approach from the previous code, which was:
    # rev_a = a[::-1]
    # conv_res = cross_conv(rev_a, p)  # length (n-1) + (2n-1) - 1 = 3n-3
    # grad_L = 2.0 * conv_res[n-1:2*n-1]  # length n
    # 
    # Let me verify: cross_conv(rev_a, p)[k] = sum_i rev_a[i] * p[k-i] = sum_i a[n-1-i] * p[k-i]
    # Let j' = n-1-i, so i = n-1-j': sum_{j'} a[j'] * p[k-n+1+j']
    # = sum_{j'} a[j'] * p[j' + (k-n+1)]
    # So cross_conv(rev_a, p)[k] = sum_j a[j] * p[j + k - n + 1]
    # We want grad[j] = 2 * sum_m a[m] * p[m+j]
    # Setting k - n + 1 = j, i.e., k = j + n - 1:
    # cross_conv(rev_a, p)[j + n - 1] = sum_m a[m] * p[m + j]
    # So grad[j] = 2 * cross_conv(rev_a, p)[n-1 + j]
    # And grad_L = 2 * cross_conv(rev_a, p)[n-1 : 2n-1]
    # This matches the previous code. Good.
    
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    rev_a = a[::-1]
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad

def adam_lbfgs_hybrid(a0, beta, maxiter=5000, time_limit=None):
    """
    Optimize LSE-smoothed objective using a combination of Adam-like 
    heavy-ball gradient descent and occasional L-BFGS steps.
    """
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = float(a.sum())
    if S < 1e-6:
        a[:] = 1.0 / n
        S = 1.0
    a = a / S  # Normalize to sum = 1
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.01  # Initial learning rate
    
    # Heavy ball momentum
    prev_a = a.copy()
    mu = 0.3  # Momentum
    
    t_start = time.monotonic()
    
    for it in range(maxiter):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        
        sc, grad = lse_score_and_grad(a, beta)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Check convergence
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        
        # Adam update
        t = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        # Adaptive step size
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Heavy ball
        step = step + mu * (a - prev_a)
        
        # Limit step size
        step_norm = np.linalg.norm(step)
        max_step = 0.1 / (S + 1e-12)
        if step_norm > max_step:
            step *= max_step / step_norm
        
        prev_a = a.copy()
        a = a - step
        a = np.clip(a, 0.0, 1000.0)
        
        # Renormalize
        S = float(a.sum())
        if S < 1e-12:
            a = best_a.copy()
            break
        a = a / S
        
        # Occasionally reduce learning rate
        if it % 500 == 0 and it > 0:
            lr *= 0.95
    
    return compute_score(best_a), best_a

def spectral_flatten(a, n_iters=200, alpha=0.5):
    """
    Iteratively flatten the power spectrum by reducing peaks in the 
    convolution and redistributing energy to valleys.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
        S = 1.0
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    M_fft = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        power = np.abs(A) ** 2
        
        # The power spectrum is related to the circular autocorrelation
        # which for a zero-padded sequence equals the linear autocorrelation
        circ_corr = np.fft.ifft(power).real[:2*n-1]
        
        c_max = float(circ_corr.max())
        c_mean = float(circ_corr.mean() * (2*n-1) / (2*n-1))  # Just mean
        # Actually, mean of circ_corr = (sum a)^2 / M_fft = 1/M_fft (since sum a = 1)
        # But we want the mean of the 2n-1 values:
        c_mean_actual = float(circ_corr.sum()) / (2*n-1)
        
        if c_max - c_mean_actual < 1e-15:
            break
        
        # Create a target: reduce peaks, boost valleys
        # Use a soft threshold
        thresh = c_mean_actual + alpha * (c_max - c_mean_actual)
        excess = np.maximum(circ_corr - thresh, 0.0)
        deficit = np.maximum(thresh - circ_corr, 0.0)
        
        if excess.max() < 1e-15:
            break
        
        # Modify the correlation
        target_corr = circ_corr - 0.7 * excess + 0.7 * deficit * (excess.sum() / max(deficit.sum(), 1e-30))
        
        # The error in correlation space
        err = target_corr - circ_corr
        err_pad = np.zeros(M_fft, dtype=np.complex128)
        err_pad[:2*n-1] = err
        err_spec = np.fft.fft(err_pad)
        
        # Newton step in frequency domain:
        # We want to change A such that |A|^2 changes appropriately
        # d|A|^2 = A* dA + A dA* = 2 Re(A* dA)
        # So dA = err_spec / (2 * A) (approximately)
        with np.errstate(divide='ignore', invalid='ignore'):
            dA = err_spec / (2.0 * A + 1e-15)
        dA[np.abs(A) < 1e-10] = 0.0
        
        # Damping
        damp = 0.3 + 0.3 * (it / max(1, n_iters))
        A_new = A + damp * dA
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a, best_sc

def peak_targeting_refinement(a, n_iters=500):
    """
    Directly target the peaks in the convolution.
    At each step, identify the worst peak and apply a targeted 
    modification to reduce it.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n, 2.0
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = self_conv(a)
        c_max = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * c_max
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # The peak is at lag k_star, contributed by:
        # (a*a)[k_star] = sum_i a[i] * a[k_star - i]
        # We want to reduce this.
        # The "contributors" are pairs (i, k_star - i) where both are in [0, n-1]
        i_min = max(0, k_star - (n-1))
        i_max = min(n-1, k_star)
        
        if i_max < i_min:
            break
        
        # Contribution of each a[j] to the peak:
        # a[j] appears in the peak if j = i or j = k_star - i for some valid i
        # Specifically, a[j] contributes a[j] * a[k_star - j] to the peak (times 2 if j != k_star - j)
        contrib = np.zeros(n)
        for j in range(i_min, i_max + 1):
            partner = k_star - j
            if 0 <= partner < n:
                contrib[j] = a[partner]
        
        # Reduce the elements that contribute most to the peak
        # and redistribute to elements that don't
        total_contrib = float(contrib.sum())
        if total_contrib < 1e-15:
            break
        
        # Find the top contributors
        top_frac = 0.1  # Top 10% of contributors
        n_top = max(1, int(n_top_frac * (i_max - i_min + 1)))
        indices = np.arange(i_min, i_max + 1)
        top_indices = indices[np.argsort(contrib[i_min:i_max+1])[::-1][:n_top]]
        
        # Amount to reduce
        reduction_factor = 0.02 + 0.03 * (it / n_iters)  # Start small, increase
        total_reduced = 0.0
        for j in top_indices:
            red = a[j] * reduction_factor
            a[j] -= red
            total_reduced += red
            a[j] = max(a[j], 0.0)
        
        # Redistribute to non-contributing elements (or low-contributing)
        if total_reduced > 1e-15:
            # Find elements that don't contribute to this peak
            non_contrib_mask = contrib < np.percentile(contrib[contrib > 0], 25) if (contrib > 0).any() else (contrib == 0)
            non_contrib_indices = np.where(non_contrib_mask)[0]
            if len(non_contrib_indices) > 0:
                per = total_reduced / len(non_contrib_indices)
                a[non_contrib_indices] += per
        
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
    
    return best_a, best_sc / (1.0)  # Score assuming sum=1, so 2n*max_c


def multi_resolution_optimize(a_init, target_n, beta_schedule=None, time_limit=None):
    """
    Multi-resolution optimization: start with a shorter sequence,
    optimize it, then upsample and continue.
    """
    if beta_schedule is None:
        beta_schedule = [20, 50, 100, 200, 400]
    
    if time_limit is None:
        time_limit = time.monotonic() + 600
    
    # Start with a shorter version
    n_coarse = max(64, target_n // 16)
    
    # Downsample the initial sequence
    if len(a_init) > n_coarse:
        x_old = np.linspace(0, 1, len(a_init))
        x_new = np.linspace(0, 1, n_coarse)
        a = np.interp(x_new, x_old, a_init)
    else:
        a = a_init.copy()
        if len(a) < n_coarse:
            x_old = np.linspace(0, 1, len(a))
            x_new = np.linspace(0, 1, n_coarse)
            a = np.interp(x_new, x_old, a)
    
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S > 0:
        a = a / S
    
    print(f"  Multi-res: starting with n={n_coarse}")
    sc = compute_score(a)
    print(f"  Initial coarse score: {sc:.8f}")
    
    for beta in beta_schedule:
        if time.monotonic() > time_limit - 60:
            break
        
        # Optimize at current resolution
        time_sub = min(120, max(30, (time_limit - time.monotonic()) / (len(beta_schedule) * 2)))
        a_opt, sc_opt = adam_lbfgs_hybrid(a, beta, maxiter=2000, time_limit=time.monotonic() + time_sub)
        print(f"  Coarse beta={beta}: score={sc_opt:.8f}")
        
        # Spectral flattening
        a_flat, sc_flat = spectral_flatten(a_opt, n_iters=100, alpha=0.6)
        if sc_flat < sc_opt:
            a_opt = a_flat
            sc_opt = sc_flat
        
        # Upsample to next resolution
        n_next = min(target_n, n_coarse * 4)
        if n_next == n_coarse and n_coarse == target_n:
            a = a_opt
            n_coarse = target_n
        elif n_next > n_coarse:
            x_old = np.linspace(0, 1, n_coarse)
            x_new = np.linspace(0, 1, n_next)
            a = np.interp(x_new, x_old, a_opt)
            n_coarse = n_next
            a = np.maximum(a, 0.0)
            S = float(a.sum())
            if S > 0:
                a = a / S
            print(f"  Upsampled to n={n_coarse}")
    
    return a_opt if n_coarse == target_n else a, sc_opt


def diverse_initializations(n, rng, parent=None, num_inits=10):
    """Generate diverse initializations."""
    x = np.arange(n, dtype=np.float64) / n
    inits = []
    
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            inits.append(np.interp(x, x_old, parent).copy())
        # Perturbed parent
        inits.append(np.clip(parent_resampled * (1 + 0.05 * rng.standard_normal(n)), 0, 1000))
    
    # Various parametric families
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 * (1 + np.cos(2 * np.pi * x)))  # Single bump
    inits.append(0.5 * (1 - np.cos(2 * np.pi * x)))  # Single dip
    inits.append(x ** 0.5)  # Ramp up
    inits.append((1 - x) ** 0.5)  # Ramp down
    inits.append(np.exp(-2 * (x - 0.3) ** 2) + 0.3 * np.exp(-2 * (x - 0.7) ** 2))  # Two bumps
    inits.append(0.5 + 0.3 * np.cos(4 * np.pi * x) + 0.2 * np.cos(9 * np.pi * x))
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000))
    
    # Smoothed noise
    from scipy.ndimage import uniform_filter1d
    for sz in [max(3, n//100), max(5, n//30)]:
        noise = rng.standard_normal(n)
        smooth = uniform_filter1d(noise, size=sz, mode='wrap')
        inits.append(np.clip(0.5 + 0.3 * smooth, 0.01, 1000))
    
    return inits[:num_inits]


def lse_objective_wrapper(a, beta):
    return lse_score_and_grad(a, beta)


def run_lbfgs_refined(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE objective with careful parameter settings."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = float(a0.sum())
    if S < 1e-6:
        a0[:] = 1.0 / n
        S = 1.0
    a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_wrapper, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = float(a_out.sum())
    if S_out > 1e-12:
        a_out = a_out / S_out
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimizer combining:
    1. Multi-resolution LSE optimization (Adam + L-BFGS)
    2. Spectral flattening in frequency domain
    3. Peak-targeting refinement
    4. Diverse local search
    
    Key improvements over previous:
    - Multi-resolution strategy (coarse-to-fine)
    - Adam-based gradient descent for better convergence on LSE
    - More aggressive spectral flattening
    - Targeted peak reduction
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(60.0, float(budget_s)) - 30.0
    deadline = t_start + effective_budget
    
    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in dir() or 'height_sequence_1' in globals():
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 32:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    best_score = np.inf
    best_a = np.ones(1024) * 0.5
    
    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False
    
    print(f"Parent available: {parent is not None}, len={0 if parent is None else len(parent)}")
    
    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n_try in [4096, 6962, 8192, 10240, 12288]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))
    
    # ============================================================
    # Phase 1: Quick exploration (20% time)
    # ============================================================
    phase1_end = t_start + 0.20 * effective_budget
    print(f"\n{'='*50}")
    print(f"Phase 1: Exploration (until t={phase1_end - t_start:.0f}s)")
    print(f"{'='*50}")
    
    for n in lengths[:4]:  # Try at most 4 lengths
        if time.monotonic() > phase1_end:
            break
        remaining = phase1_end - time.monotonic()
        if remaining < 30:
            break
        
        print(f"\n  n={n}:")
        inits = diverse_initializations(n, rng, parent, num_inits=6)
        
        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break
            a0 = np.clip(a0, 0, 1000)
            S = float(a0.sum())
            if S < 1e-12:
                continue
            a0 = a0 / S
            
            # Quick LSE optimization
            sc, a_opt = adam_lbfgs_hybrid(a0, beta=50, maxiter=1500)
            report(sc, a_opt, f"P1 n={n} i={i}")
    
    print(f"\nAfter Phase 1: best={best_score:.8f}")
    
    # ============================================================
    # Phase 2: Multi-resolution deep optimization (35% time)
    # ============================================================
    phase2_end = t_start + 0.55 * effective_budget
    print(f"\n{'='*50}")
    print(f"Phase 2: Multi-Resolution Optimization")
    print(f"{'='*50}")
    
    # Focus on the best length found so far
    n_main = len(best_a)
    if parent is not None and len(parent) in lengths:
        n_main = len(parent)
    
    a_start = best_a.copy()
    S = float(a_start.sum())
    if S > 0:
        a_start = a_start / S
    
    # Multi-resolution for the main length
    print(f"\n  Multi-res on n={n_main}:")
    a_mr, sc_mr = multi_resolution_optimize(
        a_start, n_main,
        beta_schedule=[15, 30, 60, 120, 250],
        time_limit=phase2_end - 30
    )
    report(sc_mr, a_mr, f"MultiRes n={n_main}")
    
    # Try a couple other lengths
    for n in lengths:
        if n == n_main or time.monotonic() > phase2_end - 60:
            continue
        a_try = best_a.copy()
        if len(a_try) != n:
            x_old = np.linspace(0, 1, len(a_try))
            a_try = np.interp(np.linspace(0, 1, n), x_old, a_try)
        a_try = np.maximum(a_try, 0)
        S = float(a_try.sum())
        if S > 0:
            a_try = a_try / S
        a_mr2, sc_mr2 = multi_resolution_optimize(
            a_try, n,
            beta_schedule=[20, 50, 100, 200],
            time_limit=phase2_end - 30
        )
        report(sc_mr2, a_mr2, f"MultiRes n={n}")
    
    print(f"\nAfter Phase 2: best={best_score:.8f}")
    
    # ============================================================
    # Phase 3: Intensive L-BFGS refinement (25% time)
    # ============================================================
    phase3_end = t_start + 0.80 * effective_budget
    print(f"\n{'='*50}")
    print(f"Phase 3: L-BFGS Refinement")
    print(f"{'='*50}")
    
    n_best = len(best_a)
    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 0:
        a_norm = a_norm / S
    
    # Progressive beta increase
    beta_schedule = [30, 60, 100, 200, 400, 800, 1500]
    for beta in beta_schedule:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        if remaining < 30:
            break
        maxiter = min(5000, int(remaining * 50))  # Scale iterations to time
        sc, a_opt = run_lbfgs_refined(a_norm, beta, maxiter=maxiter)
        report(sc, a_opt, f"LBFGS beta={beta}")
        a_norm = a_opt
    
    print(f"\nAfter Phase 3: best={best_score:.8f}")
    
    # ============================================================
    # Phase 4: Peak targeting + local search (15% time)
    # ============================================================
    print(f"\n{'='*50}")
    print(f"Phase 4: Peak Targeting & Local Search")
    print(f"{'='*50}")
    
    n_best = len(best_a)
    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 0:
        a_norm = a_norm / S
    
    # Peak targeting
    a_pt, sc_pt = peak_targeting_refinement(a_norm, n_iters=300)
    # Convert to proper score
    S_pt = float(a_pt.sum())
    sc_pt_proper = 2.0 * n_best * float(self_conv(a_pt).max()) / (S_pt * S_pt)
    report(sc_pt_proper, a_pt, "PeakTarget")
    
    # Local search with various perturbations
    n_trials = 6
    for trial in range(n_trials):
        if time.monotonic() > deadline - 30:
            break
        
        strategy = trial % 3
        if strategy == 0:
            # Gaussian perturbation
            scale = 0.01 + 0.02 * rng.random()
            a_pert = a_norm * (1.0 + scale * rng.standard_normal(n_best))
        elif strategy == 1:
            # Frequency-domain perturbation
            M_fft = 1 << (2 * n_best - 1).bit_length()
            a_pad = np.zeros(M_fft, dtype=np.complex128)
            a_pad[:n_best] = a_norm
            A = np.fft.fft(a_pad)
            A *= (1.0 + 0.05 * rng.standard_normal(M_fft // 2 + 1))
            A = A[:M_fft // 2 + 1]  # Keep only positive frequencies
            # Reconstruct
            A_full = np.zeros(M_fft, dtype=np.complex128)
            A_full[:M_fft // 2 + 1] = A
            A_full[M_fft // 2 + 1:] = np.conj(A[1:M_fft // 2][::-1])
            a_pert = np.fft.ifft(A_full).real[:n_best]
        else:

            amp = 0.02 + 0.04 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = a_norm * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
