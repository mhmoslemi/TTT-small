"""
Multi-phase optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Combines L-BFGS on LSE-smoothed objective with subgradient polishing.
Focuses on precise local optimization of the parent solution.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed max(conv(x,x))/(sum(x))^2 objective and gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta  # LSE-smoothed max
    
    score = 2.0 * n * L / (S * S)
    
    # p[k] = softmax weights
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient of LSE max: sum_k p[k] * d(c[k])/d(x[i])
    # d(c[k])/d(x[i]) = x[k-i] + x[i] (with x[j]=0 for j out of range)
    # sum_k p[k] * x[k-i] = cross-correlation of p and x at lag i
    
    # Compute cross = IFFT(conj(FFT(x_pad)) * FFT(p_pad))[:n]
    M_conv = 2 * n - 1  # length of c
    L_fft = 1 << (M_conv + n - 1 - 1).bit_length()  # >= M_conv + n - 1 = 3n - 2
    
    x_pad = np.zeros(L_fft)
    x_pad[:n] = x
    X = np.fft.rfft(x_pad)
    
    p_pad = np.zeros(L_fft)
    p_pad[:M_conv] = p
    P = np.fft.rfft(p_pad)
    
    cross_full = np.fft.irfft(np.conj(X) * P, L_fft)[:n]
    
    # grad_LSE[i] = cross_full[i] + x[i] * sum(p) = cross_full[i] + x[i]
    # (since sum(p) = 1)
    grad_LSE = cross_full + x.copy()
    
    # Full gradient of score = 2n * L / S^2
    # d(score)/d(x[i]) = 2n * (dL/dx[i] * S^2 - L * 2 * S * dx_S/dx[i]) / S^4
    # = 2n * (grad_LSE[i] * S - 2 * L * x[i]) / S^3
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_LSE * S - 2.0 * L * x) / S3
    
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, None).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
        S = 1.0
    a0 = a0 / S
    
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, None)
    S_out = a_out.sum()
    if S_out < 1e-6:
        a_out = a0 * 0 + 1.0 / n
    else:
        a_out = a_out / S_out
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def subgradient_polish(a, n_iters=5000, top_m=1):
    """Polish using exact subgradient of max."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    # Adaptive step size
    step = 0.02
    
    for it in range(n_iters):
        c = fast_conv(a)
        M_val = float(c.max())
        S = float(a.sum())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc - 1e-15:
            best_sc = sc
            best_a = a.copy()
        
        # Find top-m peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
        
        # Subgradient
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Step size: cap by (sc - lb) / ||grad||^2
        step = min(step, (sc - sc_lb) / (gn * gn + 1e-30))
        step = min(step, 0.05)
        step = max(step, 1e-10)
        
        a -= step * grad
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= 0.999
        
    return best_a


def fienup_quick(a, n_iters=100, threshold_mult=1.2):
    """Quick Fienup iterations for global shaping."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        
        thresh = P_mean * threshold_mult
        above = P > thresh
        if not above.any():
            break
        
        scale = np.ones_like(P)
        scale[above] = np.sqrt(thresh / P[above])
        
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a
