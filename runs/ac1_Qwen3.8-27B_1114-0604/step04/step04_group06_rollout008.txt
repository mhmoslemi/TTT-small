"""
Minimize 2*n*max(self-convolution)/(sum)^2 for non-negative sequences.
Uses LSE-smoothed gradient descent with annealing, Nesterov momentum,
backtracking line search, multiple restarts, and L-BFGS polishing.
Key improvement over previous: adaptive beta annealing, better line search,
and exploitation of the parent sequence with targeted perturbation.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and gradient w.r.t. a.
    Score = 2n * LSE_beta(conv(a,a)) / (sum a)^2
    Assumes a >= 0, sum(a) > 0.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -500.0, 500.0)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    p = exp_scaled / exp_scaled.sum()
    
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * conv_res
    
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def exact_subgrad(a):
    """Exact score and subgradient at the maximizing index."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_idx = int(c.argmax())
    score = 2.0 * n * M / (S * S)
    sg = np.zeros(n)
    # db[k]/da[i] = 2*a[k-i] (with a[m]=0 outside [0,n-1])
    # Only need the component from the single k = k_idx
    js = max(0, k_idx - n + 1)
    je = min(k_idx, n - 1)
    if je >= js:
        idx = k_idx - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)
    return score, grad


def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 1e-12:
        a = a / s
    else:
        a = np.ones_like(a) / len(a)
    return a


def lse_anneal(a_init, time_limit_s, t_start, seed=0):

    LSE-smoothed gradient descent with annealing and Nesterov momentum.
    Beta increases from 20 to 5000 over the course of optimization.

    beta_start = 20.0

    lr_decay = 1e-5
    
    momentum = 0.9

    v = np.zeros(n)  # momentum velocity

        # Anneal beta logarithmically
        progress = min(1.0, t / max_iterations)
        beta = beta_start * (beta_end / beta_start) ** progress

def clip_nonneg(a):
    return np.where(a > 0, a, 0.0)


def lbfgs_polish(a0, beta, maxiter=5000, maxfun=50000):
    """L-BFGS-B polish with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0, 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0 = np.ones(n) / n
    bounds = [(0.0, 1000.0)] * n
    
    def func_and_grad(x):
        x = np.clip(x, 0.0, None)
        if x.sum() < 1e-12:
            return 1e10, np.zeros(n)
        s, g = lse_score_grad(x, beta)
        return s, g
    
    res = minimize(
        func_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

        # Sort by impact (gradient magnitude)
        score, grad = exact_subgrad(a)
        order = sorted(affected, key=lambda i: abs(grad[i]), reverse=True)
        
        for j in order[:50]:
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            # Try reducing (if it contributes to peak) or increasing
            if grad[j] > 1e-10:
                candidates = [a_old * f for f in [0.1, 0.3, 0.5, 0.7, 0.9]]
                candidates.append(0.0)
            elif grad[j] < -1e-10:
                candidates = [a_old * f for f in [1.5, 2.0, 3.0, 5.0]]
            else:
                candidates = [a_old]
            
            best_x = a_old
            best_sc = score
            for x in candidates:
                a[j] = max(0.0, x)
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = a[j]
            a[j] = best_x
        
        final_sc = compute_score(a)
        if final_sc < best_score:
            best_score = final_sc
    
    return best_score, a


def random_perturbation(a, sigma, seed=0):
    """Add random perturbation and renormalize."""
    rng = np.random.default_rng(seed)
    n = len(a)
    perturbed = a + sigma * rng.standard_normal(n)
    return normalize(perturbed)


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for minimum of 2*n*max(conv(a,a))/(sum(a))^2.
    Strategy: Start from parent, apply LSE annealing with multiple configs,
    then polish with L-BFGS and coordinate descent.
    """
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 20.0
    rng = np.random.default_rng(seed)
    
    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 16:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    
    best_score = np.inf
    best_a = None
    
    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.10f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False
    
    if parent is not None and parent.sum() > 0.01:
        n0 = len(parent)
        parent_score = compute_score(parent)
        print(f"Parent: n={n0}, score={parent_score:.10f}")
        best_score = parent_score
        best_a = parent.copy()
    else:
        n0 = 4096
        print(f"No parent available, starting fresh with n={n0}")
        a = np.ones(n0) / n0
        best_score = compute_score(a)
        best_a = a.copy()
    
    # Phase 1: LSE Annealing from parent
    print(f"\n{'='*60}")
    print("Phase 1: LSE Annealing from parent")
    print(f"{'='*60}")
    
    n_opt = len(best_a)
    # Truncate to reasonable size if too large
    if n_opt > 8192:
        # Take the best 8192 elements
        a_resized = best_a[:8192]
        sc = compute_score(a_resized)
        try_update(sc, a_resized, "truncated parent")
        n_opt = 8192
    
    a_work = normalize(best_a[:n_opt].copy())
    
    # Run annealing with different time allocations
    time_per_run = min(300.0, effective_budget * 0.2)
    
    for run_idx in range(3):
        remaining = effective_budget - (time.monotonic() - t_start)
        if remaining < 30:
            break
        alloc = min(time_per_run, remaining * 0.3)
        
        if run_idx == 0:
            a_init = a_work.copy()
        else:
            a_init = random_perturbation(a_work, 0.05, seed=seed + run_idx)
        
        t0 = time.monotonic()
        sc, a_opt = lse_anneal(a_init, alloc, t_start, seed=seed + run_idx)
        try_update(sc, a_opt, f"anneal run {run_idx}")
        print(f"    Run {run_idx}: {sc:.10f} ({time.monotonic()-t0:.0f}s)")
    
    # Phase 2: L-BFGS Polish
    print(f"\n{'='*60}")
    print("Phase 2: L-BFGS Polish")
    print(f"{'='*60}")
    
    for beta in [200, 1000, 3000]:
        remaining = effective_budget - (time.monotonic() - t_start)
        if remaining < 15:
            break
        t0 = time.monotonic()
        sc, a_opt, _ = lbfgs_polish(best_a, beta, maxiter=3000, maxfun=30000)
        try_update(sc, a_opt, f"L-BFGS beta={beta}")
        print(f"    L-BFGS beta={beta}: {sc:.10f} ({time.monotonic()-t0:.0f}s)")
    
    # Phase 3: Coordinate Descent
    print(f"\n{'='*60}")
    print("Phase 3: Coordinate Descent")
    print(f"{'='*60}")
    
    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 30:
        t0 = time.monotonic()
        cd_time = min(remaining * 0.25, 300.0)
        sc, a_opt = coordinate_descent_polish(best_a, cd_time, t_start, n_sweeps=10)
        try_update(sc, a_opt, "coord descent")
        print(f"    Coord descent: {sc:.10f} ({time.monotonic()-t0:.0f}s)")
    
    # Phase 4: More LSE annealing with higher starting beta
    print(f"\n{'='*60}")
    print("Phase 4: Fine-tuning LSE Annealing")
    print(f"{'='*60}")
    
    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 30:
        a_work = normalize(best_a.copy())
        alloc = min(remaining * 0.2, 200.0)
        sc, a_opt = lse_anneal(a_work, alloc, t_start, seed=seed + 100)
        try_update(sc, a_opt, "fine-tune anneal")
    
    # Phase 5: Final L-BFGS with very high beta
    print(f"\n{'='*60}")
    print("Phase 5: Final High-Beta L-BFGS")
    print(f"{'='*60}")
    
    for beta in [5000, 10000]:
        remaining = effective_budget - (time.monotonic() - t_start)
        if remaining < 15:
            break
        t0 = time.monotonic()
        sc, a_opt, _ = lbfgs_polish(best_a, beta, maxiter=5000, maxfun=50000)
        try_update(sc, a_opt, f"final L-BFGS beta={beta}")
        print(f"    Final L-BFGS beta={beta}: {sc:.10f} ({time.monotonic()-t0:.0f}s)")
    
    # Phase 6: Try shorter sequences (might be easier to optimize)
    print(f"\n{'='*60}")
    print("Phase 6: Alternative lengths")
    print(f"{'='*60}")
    
    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 60:
        for n_try in [2048, 4096]:
            if time.monotonic() - t_start > effective_budget - 30:
                break
            # Resize parent by interpolation
            a_resized = np.interp(
                np.linspace(0, len(best_a)-1, n_try),
                np.arange(len(best_a)),
                best_a
            )
            a_resized = normalize(a_resized)
            sc_init = compute_score(a_resized)
            
            # Quick anneal
            alloc = min(100.0, remaining * 0.15)
            sc, a_opt = lse_anneal(a_resized, alloc, t_start, seed=seed + 200)
            try_update(sc, a_opt, f"alt length n={n_try}")
            
            # L-BFGS polish
            if remaining > 30:
                sc2, a_opt2, _ = lbfgs_polish(a_opt, 2000, maxiter=2000, maxfun=20000)
                try_update(sc2, a_opt2, f"alt n={n_try} + L-BFGS")
    
    # Phase 7: Final coordinate descent on best
    print(f"\n{'='*60}")
    print("Phase 7: Final Coordinate Descent")
    print(f"{'='*60}")
    
    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 20:
        cd_time = min(remaining * 0.4, 200.0)
        t0 = time.monotonic()
        sc, a_opt = coordinate_descent_polish(best_a, cd_time, t_start, n_sweeps=5)
        try_update(sc, a_opt, "final coord descent")
        print(f"    Final CD: {sc:.10f} ({time.monotonic()-t0:.0f}s)")
    
    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [1.0] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
