def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def newton_flatten(a, max_iters=200, tol=1e-12):
    """Newton-type iteration to flatten conv(a,a)."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    a_pad = np.zeros(M_fft)
    a_pad[:n] = a

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(max_iters):
        A = np.fft.rfft(a_pad)
        conv_spec = A * A
        conv = np.fft.irfft(conv_spec, M_fft)[:M]

        target = S * S / M
        error = target - conv

        err_pad = np.zeros(M_fft)
        err_pad[:M] = error
        E_spec = np.fft.rfft(err_pad)

        denom = 2.0 * A
        max_denom = float(np.max(np.abs(denom))) if np.max(np.abs(denom)) > 0 else 1.0
        mask = np.abs(denom) > 1e-10 * max_denom
        Delta_spec = np.zeros_like(denom)
        Delta_spec[mask] = E_spec[mask] / denom[mask]

        delta = np.fft.irfft(Delta_spec, M_fft)[:n]

        # Adaptive damping
        damp = 0.3 if it < 20 else (0.5 if it < 50 else 0.8)
        a_new = a + damp * delta
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new * S / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        if np.max(np.abs(delta)) < tol * S:
            break

        a = a_new
        a_pad[:n] = a

    return best_a


def error_reduction(a, n_iters=200, alpha=0.5):
    """Adaptive threshold error reduction in frequency domain."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    a_pad = np.zeros(M_fft)
    a_pad[:n] = a

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a_pad)
        conv_spec = A * A
        conv = np.fft.irfft(conv_spec, M_fft)[:M]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())

        if conv_max - conv_mean < 1e-12 * max(conv_mean, 1e-12):
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)

        if excess.max() < 1e-15:
            break

        scale = 0.5 + 0.4 * (it / max(1, n_iters // 2))
        conv_mod = conv - scale * excess
        err = conv_mod - conv

        err_pad = np.zeros(M_fft)
        err_pad[:M] = err
        err_spec = np.fft.rfft(err_pad)

        max_A = float(np.max(np.abs(A))) if np.max(np.abs(A)) > 0 else 1.0
        with np.errstate(divide='ignore', invalid='ignore'):
            d_spec = err_spec / (2.0 * A + 1e-15)
        d_spec[np.abs(A) < 1e-8 * max_A] = 0.0

        A_new = A + d_spec
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new * S / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

        a = a_new
        a_pad[:n] = a

    return best_a


def lse_gradient_descent(a, beta, n_iters=5000, lr=None):
    """Simple projected gradient descent on LSE objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0

    best_a = a.copy()
    best_sc = compute_score(a)
    prev_sc = best_sc

    if lr is None:
        lr = 0.01

    for it in range(n_iters):
        sc, grad = lse_score_grad(a, beta)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step size
        step = lr * S / (gn + 1e-15)
        step = min(step, 0.1)

        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new * S / S_new

        sc_new = compute_score(a_new)
        if sc_new > prev_sc * (1 + 0.1):
            # Backtrack
            step *= 0.5
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            S_new = float(a_new.sum())
            if S_new < 1e-12:
                break
            a_new = a_new * S / S_new
            sc_new = compute_score(a_new)

        a = a_new
        prev_sc = sc_new

    return best_a


def multi_scale_newton(a, max_rounds=5, iters_per_round=100):
    """Newton with decreasing damping (aggressive -> conservative)."""
    n = len(a)
    S0 = float(a.sum())
    best_a = a.copy()
    best_sc = compute_score(a)

    for r in range(max_rounds):
        damp_factor = 0.2 + 0.15 * r
        # Run modified newton with specific damping
        M = 2 * n - 1
        M_fft = 1 << (M - 1).bit_length()
        a_work = a.copy()
        a_pad = np.zeros(M_fft)
        a_pad[:n] = a_work

        for it in range(iters_per_round):
            A = np.fft.rfft(a_pad)
            conv_spec = A * A
            conv = np.fft.irfft(conv_spec, M_fft)[:M]

            target = S0 * S0 / M
            error = target - conv

            err_pad = np.zeros(M_fft)
            err_pad[:M] = error
            E_spec = np.fft.rfft(err_pad)

            denom = 2.0 * A
            max_denom = float(np.max(np.abs(denom))) if np.max(np.abs(denom)) > 0 else 1.0
            mask = np.abs(denom) > 1e-10 * max_denom
            Delta_spec = np.zeros_like(denom)
            Delta_spec[mask] = E_spec[mask] / denom[mask]

            delta = np.fft.irfft(Delta_spec, M_fft)[:n]
            a_work = a_work + damp_factor * delta
            a_work = np.maximum(a_work, 0.0)
            S_w = float(a_work.sum())
            if S_w < 1e-12:
                break
            a_work = a_work * S0 / S_w

            a_pad[:n] = a_work

        sc = compute_score(a_work)
        if sc < best_sc:
            best_sc = sc
            best_a = a_work.copy()
            a = a_work.copy()

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses binomial-sequence initialization, Newton flattening, LSE-LBFGS,
    and multi-scale refinement.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 15.0

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Try to load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    print(f"Parent available: {parent is not None}, len={0 if parent is None else len(parent)}")

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n_try in [4096, 6962, 8192, 10000, 12000]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))[:5]
    print(f"Lengths to explore: {lengths}")

    # ==================================================================
    # Phase 1: Initialize with binomial sequences + quick Newton (20% time)
    # ==================================================================
    phase1_deadline = t_start + 0.20 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 1: Binomial Init + Newton (20% time) ---")

    candidates = []
    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 15:
            break

        # Binomial initialization
        a_bin = make_binomial_seq(n)
        a_bin = a_bin / a_bin.sum()
        sc_init = compute_score(a_bin)
        print(f"  n={n}: binomial init score = {sc_init:.8f}")

        # Quick Newton
        a_newton = newton_flatten(a_bin, max_iters=80)
        sc_newton = compute_score(a_newton)
        report(sc_newton, a_newton, f"Newton n={n}")
        candidates.append((sc_newton, a_newton, n))

        # Error reduction
        a_er = error_reduction(a_bin, n_iters=60, alpha=0.5)
        sc_er = compute_score(a_er)
        report(sc_er, a_er, f"ER n={n}")
        candidates.append((sc_er, a_er, n))

    # If parent is good, include it
    if parent is not None:
        sc_p = compute_score(parent)
        print(f"  Parent score: {sc_p:.8f}")
        report(sc_p, parent, "parent")
        candidates.append((sc_p, parent.copy(), len(parent)))

    candidates.sort(key=lambda x: x[0])
    print(f"Phase 1 best: {best_score:.8f}")

    # ==================================================================
    # Phase 2: Multi-scale Newton refinement (25% time)
    # ==================================================================
    phase2_deadline = t_start + 0.45 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 2: Multi-Scale Newton (25% time) ---")

    for sc_c, a_c, n_c in candidates[:3]:
        if time.monotonic() > phase2_deadline:
            break
        print(f"  Refining n={n_c}, current score={sc_c:.8f}")
        a_ms = multi_scale_newton(a_c, max_rounds=5, iters_per_round=80)
        sc_ms = compute_score(a_ms)
        report(sc_ms, a_ms, f"MS-Newton n={n_c}")
        # Also try error reduction on the result
        a_er2 = error_reduction(a_ms, n_iters=80, alpha=0.4)
        sc_er2 = compute_score(a_er2)
        report(sc_er2, a_er2, f"ER2 n={n_c}")

    print(f"Phase 2 best: {best_score:.8f}")

    # ==================================================================
    # Phase 3: LSE-smoothed L-BFGS (30% time)
    # ==================================================================
    phase3_deadline = t_start + 0.75 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 3: LSE L-BFGS (30% time) ---")

    n_best = len(best_a)
    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 1e-12:
        a_norm = a_norm / S

    beta_schedule = [20, 50, 100, 200, 400, 800, 1500]
    for beta in beta_schedule:
        if time.monotonic() > phase3_deadline:
            break
        elapsed = time.monotonic() - t_start
        remaining = phase3_deadline - t_start
        if remaining < 30:
            break
        print(f"  L-BFGS beta={beta}, remaining={remaining:.0f}s")
        sc, a_opt, _ = run_lbfgs(a_norm, beta, maxiter=4000, maxfun=150000)
        report(sc, a_opt, f"LBFGS b={beta}")
        if sc <= best_score:
            a_norm = a_opt.copy()
            S = float(a_norm.sum())
            if S > 1e-12:
                a_norm = a_norm / S

    # Try L-BFGS on other lengths
    for n_c in lengths:
        if n_c == n_best:
            continue
        if time.monotonic() > phase3_deadline - 30:
            break
        print(f"  L-BFGS on n={n_c}")
        a_test = np.interp(np.linspace(0, 1, n_c),
                           np.linspace(0, 1, len(best_a)), best_a)
        a_test = np.maximum(a_test, 0.0)
        S_t = float(a_test.sum())
        if S_t > 1e-12:
            a_test = a_test / S_t
        sc, a_opt, _ = run_lbfgs(a_test, 100, maxiter=3000, maxfun=80000)
        report(sc, a_opt, f"LBFGS n={n_c}")

    print(f"Phase 3 best: {best_score:.8f}")

    # ==================================================================
    # Phase 4: Hybrid Newton + LSE refinement (15% time)
    # ==================================================================
    phase4_deadline = t_start + 0.90 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 4: Hybrid Refinement (15% time) ---")

    n_best = len(best_a)
    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 1e-12:
        a_work = a_work / S

    # Alternate Newton and LSE-LBFGS
    for round_i in range(6):
        if time.monotonic() > phase4_deadline:
            break

        # Newton step
        a_newton = newton_flatten(a_work, max_iters=50)
        sc_n = compute_score(a_newton)
        if sc_n < compute_score(a_work):
            a_work = a_newton
            report(sc_n, a_work, f"hybrid-N {round_i}")

        # LSE-LBFGS step
        beta = 200 + 200 * round_i
        if time.monotonic() > phase4_deadline - 20:
            break
        sc, a_opt, _ = run_lbfgs(a_work, beta, maxiter=3000, maxfun=80000)
        report(sc, a_opt, f"hybrid-L {round_i}")
        if sc <= best_score:
            a_work = a_opt.copy()
            S = float(a_work.sum())
            if S > 1e-12:
                a_work = a_work / S

    print(f"Phase 4 best: {best_score:.8f}")

    # ==================================================================
    # Phase 5: Final polish with high-beta LSE + targeted Newton (10% time)
    # ==================================================================
    print(f"\n--- Phase 5: Final Polish ---")

    n_best = len(best_a)
    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 1e-12:
        a_norm = a_norm / S

    # Very high beta L-BFGS
    if time.monotonic() < deadline - 60:
        print("  Final L-BFGS beta=3000...")
        sc, a_opt, _ = run_lbfgs(a_norm, 3000, maxiter=5000, maxfun=200000)
        report(sc, a_opt, "final LBFGS")
        if sc <= best_score:
            a_norm = a_opt.copy()

    # Final Newton
    if time.monotonic() < deadline - 30:
        print("  Final Newton...")
        a_fn = newton_flatten(a_norm, max_iters=100)
        sc_fn = compute_score(a_fn)
        report(sc_fn, a_fn, "final Newton")

    # Try a few perturbations + quick optimization
    if time.monotonic() < deadline - 20:
        for trial in range(5):
            if time.monotonic() > deadline - 10:
                break
            scale = 0.005 + 0.01 * rng.random()
            a_pert = a_norm * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            S_p = float(a_pert.sum())
            if S_p > 1e-12:
                a_pert = a_pert / S_p
            a_pert = newton_flatten(a_pert, max_iters=30)
            sc_p = compute_score(a_pert)
            report(sc_p, a_pert, f"perturb {trial}")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
