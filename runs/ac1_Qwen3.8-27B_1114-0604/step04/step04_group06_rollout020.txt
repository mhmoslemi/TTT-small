def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):

    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def subgrad_score(a):
    """Exact score and subgradient at the unique max position."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    # Find all positions achieving the max (within numerical tolerance)
    max_val = c.max()
    mask = c >= max_val - 1e-10 * max_val
    k_positions = np.where(mask)[0]
    score = 2.0 * n * M / (S * S)
    
    # Subgradient: average over all max-achieving positions
    sg = np.zeros(n)
    for k in k_positions:
        # d conv[k] / d a[i] = 2*a[k-i] if 0<=k-i<n
        valid = np.arange(n)
        idx = k - valid
        valid_mask = (idx >= 0) & (idx < n)
        sg[valid_mask] += 2.0 * a[idx[valid_mask]]
    sg /= len(k_positions)
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad


def adam_lse_schedule(a_init, time_limit_s, t_start, initial_beta=30, final_beta=20000,
                      lr=0.01, check_interval=2000):
    """Adam optimizer with LSE-smoothed gradient, beta schedule from initial to final."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        S = a.sum()
    a = a / S  # Normalize to sum=1

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)
    n_steps_done = 0
    max_steps = 500000

    for t in range(1, max_steps + 1):
        if time.monotonic() - t_start > time_limit_s:
            break
        
        # Beta schedule: exponential from initial to final
        progress = min(1.0, t / (max_steps * 0.7))
        beta = initial_beta * (final_beta / initial_beta) ** progress
        
        lr_t = lr / (1.0 + 5e-6 * t)
        score, grad = lse_score_grad(a, beta)
        
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)
        s = a.sum()
        if s > 1e-12:
            a = a / s
        
        n_steps_done = t
        
        if t % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a, n_steps_done


def peak_equalization(a, n_rounds=50, time_limit_s=120, t_start=0):
    """Iteratively reduce peaks by transferring mass from peak-contributing coords."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s

            break
        
        # Find top-K peak positions
        top_k = 10
        peak_positions = np.argsort(c)[::-1][:top_k]
        
        # For each peak position, compute contribution vector
        # contrib[i] = weighted sum of a[peak_pos - i] over peaks
        # We want to decrease a[i] where it contributes to many peaks
        contrib = np.zeros(n)
        for k in peak_positions:
            weight = c[k] / c_max  # Weight by how close to max
            valid = np.arange(n)
            idx = k - valid
            valid_mask = (idx >= 0) & (idx < n)
            contrib[valid_mask] += weight * a[idx[valid_mask]]
        
        # Normalize contrib to [0, 1]
        c_max contrib = contrib.max()
        if c_maxc < 1e-15:
            break
        contrib_norm = contrib / c_maxc
        
        # Transfer mass: decrease high-contrib, increase low-contrib
        # Find the "worst" 20% and "best" 20% of coordinates
        q_high = np.percentile(contrib_norm, 80)
        q_low = np.percentile(contrib_norm, 20)
        
        high_mask = (contrib_norm >= q_high) & (a > 1e-10)
        low_mask = contrib_norm <= q_low
        
        if not np.any(high_mask) or not np.any(low_mask):
            break
        
        # Transfer a small amount
        transfer_rate = 0.05
        high_sum = a[high_mask].sum()
        low_count = low_mask.sum()
        if low_count == 0:

        transfer_amount = high_sum * transfer_rate
        a[high_mask] -= transfer_amount / high_mask.sum() * a[high_mask]
        a[low_mask] += transfer_amount / low_count
        
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 1e-12:
            a = a / s
        
        sc = compute_score(a)

def coord_descent_refine(a, n_sweeps=20, time_limit_s=120, t_start=0):
    """Coordinate descent on coordinates with highest gradient impact."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        score, grad = subgrad_score(a)
        if np.isinf(score):
            break
        # Sort by |gradient|, take top coordinates
        order = np.argsort(np.abs(grad))[::-1][:50]
        for j in order:
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            if grad[j] > 1e-12:
                candidates = a_old * np.array([0.5, 0.7, 0.85, 0.95])
            elif grad[j] < -1e-12:
                candidates = np.array([0.0, a_old * 1.5, a_old * 2.0, a_old * 3.0])
            else:
                continue
            candidates = np.clip(candidates, 0.0, 1000.0)
            best_x = a_old
            best_sc = score
            for x in candidates:
                a[j] = x
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x
    
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a


def make_anti_palindromic(n, style='sine'):
    """Create anti-palindromic initializations (small conv[n-1])."""
    if style == 'ramp':
        a = np.linspace(1.0, 0.0, n)
        a = np.clip(a, 0.01, None)
    elif style == 'sine':
        a = (1.0 + np.cos(2.0 * np.pi * np.arange(n) / n)) ** 2
        a = np.clip(a, 0.0, None)
        # Ensure non-negative and add small floor
        a = np.maximum(a, 0.001)
    elif style == 'exp':
        a = np.exp(-2.0 * np.arange(n) / n)
        a = np.maximum(a, 0.001)
    elif style == 'two_peaks':
        x = np.arange(n)
        a = np.exp(-4.0 * (x / n)) + np.exp(-4.0 * ((n - 1 - x) / n))
        # This peaks at both ends (anti-palindromic in a sense)
        a = np.maximum(a, 0.001)
    elif style == 'shifted_gauss':
        x = np.arange(n)
        center = n * 0.25
        a = np.exp(-0.5 * ((x - center) / (n * 0.1)) ** 2)
        a = np.maximum(a, 0.001)
    else:
        a = np.ones(n)
    a = a / a.sum()
    return a


def make_structured(n, style='bilaterally_anti'):
    """Create structured initializations that should have low central convolution."""
    x = np.arange(n)
    if style == 'bilaterally_anti':
        # High at edges, low in middle: a[j] ~ min(j, n-1-j)
        a = np.minimum(x, n - 1 - x).astype(float) + 1.0
    elif style == 'quarter_wave':
        a = np.sin(np.pi * x / (2 * n)) ** 2 + 0.01
    elif style == 'asymmetric_two':
        a = np.exp(-3.0 * x / n) * 0.7 + np.exp(-3.0 * (n - 1 - x) / n) * 0.3
    elif style == 'comb':
        a = np.ones(n)
        a[::3] = 0.3
    else:
        a = np.ones(n)
    a = np.maximum(a, 0.001)
    a = a / a.sum()
    return a

    scored_inits.sort(key=lambda x: x[0])
    top_inits = scored_inits[:min(5, len(scored_inits))]
    
    # Allocate time: give more time to better initializations
    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining < 30:
        remaining = 30
    
    # Phase 2a: Quick exploration (30s per init, up to 5 inits = 150s)
    for idx, (sc, name, a) in enumerate(top_inits):
        elapsed = time.monotonic() - t_start
        if elapsed > effective_budget - 120:
            break
        time_alloc = min(30.0, effective_budget - elapsed - 60)
        if time_alloc < 5:
            break
        print(f"  Exploring {name} (alloc={time_alloc:.0f}s)...")
        sc2, a2, steps = adam_lse_schedule(a, time_limit_s=time_alloc, t_start=t_start,
                                            initial_beta=20, final_beta=5000, lr=0.01)
        try_update(sc2, a2, f"explore_{name}")
    
    # Phase 2b: Deep optimization from the best found so far
    if best_a is not None:
        elapsed = time.monotonic() - t_start
        remaining = effective_budget - elapsed
        if remaining > 60:
            print(f"\n  Deep optimization (alloc={remaining - 60:.0f}s)...")
            sc3, a3, steps = adam_lse_schedule(best_a, time_limit_s=remaining - 60,
                                                t_start=t_start,
                                                initial_beta=100, final_beta=30000, lr=0.005)
            try_update(sc3, a3, "deep_opt")
    
    # Phase 3: Multiple restarts with different betas from best
    print(f"\n{'='*60}")
    print(f"Phase 3: Multi-beta refinement")
    print(f"{'='*60}")
    
    if best_a is not None:
        elapsed = time.monotonic() - t_start
        remaining = effective_budget - elapsed
        if remaining > 60:
            time_per = min(40.0, remaining / 4.0)
            for beta_start, beta_end, lr_val in [(50, 5000, 0.008), (200, 20000, 0.004),
                                                   (500, 50000, 0.002), (100, 10000, 0.006)]:
                elapsed = time.monotonic() - t_start
                if elapsed > effective_budget - 60:
                    break
                time_alloc = min(time_per, effective_budget - elapsed - 30)
                if time_alloc < 5:
                    break
                sc4, a4, _ = adam_lse_schedule(best_a, time_limit_s=time_alloc, t_start=t_start,
                                                initial_beta=beta_start, final_beta=beta_end, lr=lr_val)
                try_update(sc4, a4, f"multi_beta_{beta_start}_{beta_end}")
    
    # Phase 4: Peak equalization
    print(f"\n{'='*60}")
    print(f"Phase 4: Peak equalization")
    print(f"{'='*60}")
    
    if best_a is not None:
        elapsed = time.monotonic() - t_start
        remaining = effective_budget - elapsed
        if remaining > 30:
            time_alloc = min(60.0, remaining * 0.5)
            print(f"  Peak equalization (alloc={time_alloc:.0f}s)...")
            sc5, a5 = peak_equalization(best_a, n_rounds=200, time_limit_s=time_alloc,
                                        t_start=time.monotonic())
            try_update(sc5, a5, "peak_equal")
            if a5 is not None and best_a is not None and sc5 < best_score:
                best_a = a5
    
    # Phase 5: Coordinate descent polish
    print(f"\n{'='*60}")
    print(f"Phase 5: Coordinate descent")
    print(f"{'='*60}")
    
    if best_a is not None:
        elapsed = time.monotonic() - t_start
        remaining = effective_budget - elapsed
        if remaining > 20:
            time_alloc = min(60.0, remaining * 0.5)
            print(f"  Coord descent (alloc={time_alloc:.0f}s)...")
            sc6, a6 = coord_descent_refine(best_a, n_sweeps=50, time_limit_s=time_alloc,
                                           t_start=time.monotonic())
            try_update(sc6, a6, "coord_desc")
    
    # Phase 6: Final short LSE-Adam polish
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish")
    print(f"{'='*60}")
    
    if best_a is not None:
        elapsed = time.monotonic() - t_start
        remaining = effective_budget - elapsed
        if remaining > 20:
            time_alloc = min(remaining - 10, 120.0)
            if time_alloc > 10:
                sc7, a7, _ = adam_lse_schedule(best_a, time_limit_s=time_alloc, t_start=t_start,
                                                initial_beta=5000, final_beta=50000, lr=0.001)
                try_update(sc7, a7, "final_polish")
    
    # Final: ensure valid output
    if best_a is None:
        best_a = np.ones(64) / 64
    
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [1.0] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
