"""
Multi-phase optimizer combining frequency-domain phase retrieval (HIO/ERR),
LP-based alternating method, and LSE-smoothed L-BFGS for minimizing
the peak-to-mean ratio of the aperiodic autocorrelation.
"""

import numpy as np
from scipy.optimize import minimize
import time

# Precompute FFT size for given n
def next_pow2(n):
    return 1 << (n - 1).bit_length()

def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = next_pow2(M)
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def hio_iteration(a, target_mag, gamma=0.9):
    """One HIO iteration."""
    n = len(a)
    M = 2 * n - 1
    M_fft = next_pow2(M)
    
    A = np.fft.rfft(a, M_fft)
    phase = np.angle(A)
    
    # New magnitude
    A_new = target_mag * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    
    # HIO: in the "forbidden" region (where a should be 0), replace
    # In the "known" region (where a >= 0), keep
    # For our case: all positions are "known" (a >= 0)
    # So we just clip
    a_clipped = np.maximum(a_new, 0.0)
    
    # HIO combination:
    # a = gamma * a_clipped + (1-gamma) * a_new (in known region)
    # But since all positions are known, we just use:
    a_out = gamma * a_clipped + (1.0 - gamma) * a_new
    a_out = np.maximum(a_out, 0.0)
    
    return a_out

def err_iteration(a, gamma=0.9):
    """One ERR (Error Reduction) iteration for flat spectrum."""
    n = len(a)
    M_fft = next_pow2(2 * n - 1)
    
    A = np.fft.rfft(a, M_fft)
    mag = np.abs(A)
    phase = np.angle(A)

    target = np.mean(mag)  # or median for robustness
    A_target = target * np.exp(1j * phase)

    err = a_clipped - a_err
    err_fft = np.fft.rfft(err, M_fft)

    a_out = np.fft.irfft(A_new, M_fft)[:n]

"""
Multi-phase optimizer combining HIO phase retrieval, LP-based alternating,
and LSE-smoothed L-BFGS for minimizing peak-to-mean aperiodic autocorrelation.
Key innovations: proper HIO with support+non-negativity constraints,
adaptive multi-length exploration, and improved FISTA LP sub-solver.
"""

import numpy as np
from scipy.optimize import minimize
import time


def next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = next_pow2(M)
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = next_pow2(M)
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def hio_phase_retrieval(a0, n_iters=3000, gamma_base=0.82):
    """HIO phase retrieval: find non-negative seq with flat magnitude spectrum."""
    n = len(a0)
    L = 2 * n - 1
    L_fft = next_pow2(L)

    a = a0.copy().astype(np.float64)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    a_pad = np.zeros(L, dtype=np.float64)
    a_pad[:n] = a

    for block in range(0, n_iters, 250):
        block_size = min(250, n_iters - block)
        # Update target magnitude
        A = np.fft.rfft(a_pad, L_fft)
        mags = np.abs(A)
        target_mag = np.full(np.size(mags), np.sqrt(np.mean(mags * mags)))

        for it in range(block_size):
            frac = (block + it) / n_iters
            g = gamma_base - 0.15 * frac

            A = np.fft.rfft(a_pad, L_fft)
            phase = np.angle(A)
            A_mod = target_mag * np.exp(1j * phase)
            x = np.fft.irfft(A_mod, L_fft)[:L]

            # HIO: project onto constraint set (support + non-negativity)
            x_proj = np.zeros(L, dtype=np.float64)
            x_proj[:n] = np.maximum(x[:n] - x[n:], 0.0)

            a_pad = g * x_proj + (1.0 - g) * np.maximum(x, 0.0)
            a_pad[n:] = 0.0

            if (it % 50 == 0) or (it == block_size - 1):
                sc = compute_score(a_pad[:n])
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_pad[:n].copy()

    return best_a


def err_phase_retrieval(a0, n_iters=2000, gamma=0.8):
    """Error reduction phase retrieval."""
    n = len(a0)
    L = 2 * n - 1
    L_fft = next_pow2(L)

    a = a0.copy().astype(np.float64)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    a_pad = np.zeros(L, dtype=np.float64)
    a_pad[:n] = a

    for it in range(n_iters):
        A = np.fft.rfft(a_pad, L_fft)
        mags = np.abs(A)
        phase = np.angle(A)
        target = np.full(np.size(mags), np.sqrt(np.mean(mags * mags)))
        A_target = target * np.exp(1j * phase)

        x = np.fft.irfft(A_target, L_fft)[:L]
        x_clipped = np.maximum(x, 0.0)
        x_clipped[n:] = 0.0

        # Error in constraint region
        err = x_clipped - x
        err_pad = np.zeros(L, dtype=np.float64)
        err_pad[:n] = err[:n]
        err_fft = np.fft.rfft(err_pad, L_fft)

        A_new = A - gamma * err_fft
        a_pad = np.fft.irfft(A_new, L_fft)[:L]
        a_pad = np.maximum(a_pad, 0.0)
        a_pad[n:] = 0.0

        if it % 50 == 0 or it == n_iters - 1:
            sc = compute_score(a_pad[:n])
            if sc < best_sc:
                best_sc = sc
                best_a = a_pad[:n].copy()

    return best_a


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()

    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = next_pow2(M2)
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def fista_lp(a, M, n_iters=5000, beta_start=10.0, beta_end=200.0):
    """FISTA for: max sum(x) s.t. conv(a,x) <= M, x >= 0."""
    n = len(a)
    x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    sum_a2 = float(np.sum(a * a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n

    for it in range(n_iters):
        frac = min(1.0, it / (n_iters * 0.4))
        beta = beta_start + (beta_end - beta_start) * frac

        L = beta * n * sum_a2 + 1e-12
        step = 1.0 / L

        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum

        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(p) - n + 1]
        grad = a * sw - 1.0

        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    return x


def lp_alternating(a, max_outer_iters=50, fista_iters=4000,
                   t_start=0.4, t_min=0.05, time_limit=None):
    """LP-based alternating method."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0

    for outer in range(max_outer_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        M = float(self_conv(a).max())
        x = fista_lp(a, M, n_iters=fista_iters)
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx

        t = max(t_min, t_start * (1.0 + 1.0 / (outer + 1)))
        t = min(t, 1.0)

        a_new = (1.0 - t) * a + t * x
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 20:
                t = t_min * 0.5
                if t < 0.005:
                    break
                no_improve = 0
        a = a_new

    return best_a, best_sc


def frequency_domain_polish(a, n_iters=300, alpha=0.8):
    """Frequency domain error reduction polish."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = next_pow2(2 * n - 1)
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        conv = np.fft.ifft(A * A).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-16:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-16:
            break

        conv_mod = conv.copy()
        conv_mod -= 0.7 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-10] = 0.0

        damp = 0.4 + 0.4 * (it / max(1, n_iters // 3))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def build_diverse_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    # Various structured initializations
    inits.append(0.5 * np.ones(n))
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(np.clip(0.5 + 0.3 * np.sin(4.0 * np.pi * x), 0.01, 2.0))
    inits.append(0.3 + 0.7 * (1.0 - 4.0 * (x - 0.5) ** 2))
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 2.0))

    # Raised cosine
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x)))
    # Hanning-like
    inits.append(0.5 - 0.5 * np.cos(2.0 * np.pi * x))
    # Multi-frequency
    inits.append(np.clip(0.5 + 0.15 * np.cos(6.0 * np.pi * x) + 0.1 * np.cos(10.0 * np.pi * x), 0.01, 2.0))

    for init in inits:
        S = float(init.sum())
        if S > 1e-6:
            init = init / S
    return inits

            # ERR on best of HIO
            if time.monotonic() < phase1_end - 30:
                a_err = err_phase_retrieval(a_hio, n_iters=n_iters // 2, gamma=0.75)
                sc = compute_score(a_err)
                report(sc, a_err, f"ERR n={n} i={i}")

    print(f"  After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Frequency Domain Polish + LP Alternating - 35% time
    # ============================================================
    phase2_end = t_start + 0.70 * (float(budget_s) - 30.0)
    print(f"\n=== Phase 2: Freq Polish + LP Alternating ===")

    if best_a is not None:
        n_main = len(best_a)
    else:
        n_main = 6962
        best_a = np.ones(n_main) * 0.5

    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    # Step 2a: Frequency domain polish
    if time.monotonic() < phase2_end - 60:
        print(f"  Freq polish on n={n_main}...")
        a_fp = frequency_domain_polish(a_work, n_iters=200, alpha=0.8)
        sc = compute_score(a_fp)
        report(sc, a_fp, "freq_polish")
        a_work = best_a.copy()
        S = float(a_work.sum())
        if S > 0:
            a_work = a_work / S

    # Step 2b: LP Alternating (main optimization)
    print(f"  LP Alternating on n={n_main}...")
    fista_budgets = [3000, 4000, 5000, 6000]
    t_schedules = [(0.5, 0.1), (0.4, 0.08), (0.3, 0.05), (0.2, 0.03)]

    for fi, (fista_it, (t_s, t_m)) in enumerate(zip(fista_budgets, t_schedules)):
        if time.monotonic() > phase2_end - 30:
            break
        remaining = phase2_end - time.monotonic()
        max_outer = max(5, int(remaining / max(5.0, fista_it * n_main / 500000.0)))
        max_outer = min(max_outer, 40)
        print(f"    LP round {fi}: fista={fista_it}, t=[{t_s},{t_m}], outer~{max_outer}")

        a_in = best_a.copy()
        S = float(a_in.sum())
        if S > 0:
            a_in = a_in / S
        a_out, sc_out = lp_alternating(
            a_in, max_outer_iters=max_outer, fista_iters=fista_it,
            t_start=t_s, t_min=t_m,
            time_limit=phase2_end - 10
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP r{fi}")

    # Try other lengths if time permits
    for n_try in lengths:
        if n_try == n_main:
            continue
        if time.monotonic() > phase2_end - 120:
            break
        print(f"  LP alt length: n={n_try}")
        a_test = np.interp(
            np.linspace(0, 1, n_try),
            np.linspace(0, 1, len(best_a)),
            best_a
        )
        a_test = np.maximum(a_test, 0.0)
        S = float(a_test.sum())
        if S > 1e-12:
            a_test = a_test / S
        a_out, sc_out = lp_alternating(
            a_test, max_outer_iters=20, fista_iters=3000,
            t_start=0.4, t_min=0.1, time_limit=phase2_end - 20
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP n={n_try}")

    print(f"  After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: L-BFGS with progressive beta - 20% time
    # ============================================================
    phase3_end = t_start + 0.90 * (float(budget_s) - 30.0)
    print(f"\n=== Phase 3: L-BFGS Refinement ===")

    n_best = len(best_a)
    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 1e-12:
        a_norm = a_norm / S

    beta_schedule = [20, 50, 100, 200, 500, 1000, 2000]
    for beta in beta_schedule:
        if time.monotonic() > phase3_end - 10:
            break
        print(f"    L-BFGS beta={beta}...")
        sc, a_opt, _ = run_lbfgs(a_norm, beta, maxiter=8000, maxfun=200000)
        report(sc, a_opt, f"LBFGS b={beta}")
        a_norm = best_a.copy()
        S = float(a_norm.sum())
        if S > 1e-12:
            a_norm = a_norm / S

    # Multi-restart L-BFGS with perturbations
    if time.monotonic() < phase3_end - 30:
        print("    Multi-restart L-BFGS...")
        for trial in range(4):
            if time.monotonic() > phase3_end - 10:
                break
            scale = 0.005 + 0.01 * trial
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            S_p = float(a_pert.sum())
            if S_p > 1e-12:
                a_pert = a_pert / S_p
                sc, a_opt, _ = run_lbfgs(a_pert, 500, maxiter=5000, maxfun=150000)
                report(sc, a_opt, f"LBFGS restart {trial}")

    print(f"  After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Final Polish - 10% time
    # ============================================================
    print(f"\n=== Phase 4: Final Polish ===")

    # Subgradient-like targeted reduction
    if time.monotonic() < deadline - 60:
        print("  Targeted peak reduction...")
        a_target = best_a.copy()
        S = float(a_target.sum())
        if S > 1e-12:
            a_target = a_target / S
        n_b = len(a_target)

        for inner in range(50):
            if time.monotonic() > deadline - 30:
                break
            c = self_conv(a_target)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_b + 1)
            j_end = min(k_star, n_b - 1)

            # Reduce top contributors, add to bottom
            seg = a_target[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            top_k = max(10, n_b // 100)
            top_idx = np.argsort(contrib)[::-1][:top_k]
            bot_idx = np.argsort(contrib)[:top_k]

            total_red = 0.0
            for li in top_idx:
                gi = j_start + li
                red = a_target[gi] * 0.02
                a_target[gi] = max(a_target[gi] - red, 0.0)
                total_red += red

            if total_red > 0:
                per = total_red / len(bot_idx)
                for li in bot_idx:
                    a_target[j_start + li] += per

            S_t = float(a_target.sum())
            if S_t < 1e-12:
                break
            a_target = a_target / S_t
            sc = compute_score(a_target)
            report(sc, a_target, f"targeted {inner}")

    # Final LP
    if time.monotonic() < deadline - 60:
        print("  Final LP polish...")
        a_final_in = best_a.copy()
        S = float(a_final_in.sum())
        if S > 1e-12:
            a_final_in = a_final_in / S
        a_final, sc_final = lp_alternating(
            a_final_in, max_outer_iters=20, fista_iters=5000,
            t_start=0.15, t_min=0.02, time_limit=deadline - 15
        )
        sc = compute_score(a_final)
        report(sc, a_final, "final_LP")

    # Final L-BFGS
    if time.monotonic() < deadline - 20:
        a_norm = best_a.copy()
        S = float(a_norm.sum())
        if S > 1e-12:
            a_norm = a_norm / S
        sc, a_opt, _ = run_lbfgs(a_norm, 3000, maxiter=3000, maxfun=100000)
        report(sc, a_opt, "final_LBFGS")

    # Final frequency polish
    if time.monotonic() < deadline - 10:
        a_fp2 = frequency_domain_polish(best_a, n_iters=100, alpha=0.9)
        sc = compute_score(a_fp2)
        report(sc, a_fp2, "final_freq")

    # ============================================================
    # Final output
    # ============================================================
    if best_a is None:
        best_a = np.ones(1024) * 0.5

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
