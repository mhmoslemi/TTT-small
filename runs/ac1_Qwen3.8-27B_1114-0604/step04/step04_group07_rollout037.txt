```python
Hybrid optimizer using Gerchberg-Saxton (error reduction) + LSE-Adam + L-BFGS
to minimize 2n*max(self-conv)/(sum)^2 over non-negative sequences.

import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def gs_iterate(a, n_iters=1000, time_limit=None, t_start=None):
    """Gerchberg-Saxton error reduction for flat spectrum."""
    n = len(a)
    a = a.copy()
    best_score = compute_score(a)
    best_a = a.copy()
    
    for it in range(n_iters):
        if time_limit and time.monotonic() - t_start > time_limit:
            break
        
        # Zero-pad
        a_padded = np.zeros(2 * n)
        a_padded[:n] = a
        
        # FFT
        A = np.fft.fft(a_padded)
        mag = np.abs(A)
        
        # Target: constant magnitude (use current mean to preserve energy)
        mean_mag = np.mean(mag)
        target_mag = np.full_like(mag, mean_mag)
        
        # Keep phase, replace magnitude
        phase = np.angle(A)
        A_new = target_mag * np.exp(1j * phase)
        
        # Inverse FFT
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        
        if a_new.sum() < 1e-12:
            break
        a = a_new
        
        if it % 100 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_score, best_a
```

```python
"""
Optimizer using Gerchberg-Saxton error reduction + LSE-Adam + L-BFGS
to minimize 2n*max(self-conv)/(sum)^2 over non-negative sequences.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    
    # Compute cross = [sum_j a[j]*p[i+j] for i=0..n-1]
    N_fft = 1 << (3 * n - 2).bit_length()
    A_fft = np.fft.rfft(a, N_fft)
    P_fft = np.fft.rfft(p, N_fft)
    cross_full = np.fft.irfft(A_fft * np.conj(P_fft), N_fft)
    cross = cross_full[:n]
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (2.0 * cross / S2 - 2.0 * L / S3)
    
    return score, grad


def adam_lse(a_init, beta_schedule, lr, n_steps, time_limit_s, t_start, verbose_every=2000):
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    best_a = a.copy()
    best_score = compute_score(a)
    
    n_beta = len(beta_schedule)
    steps_per_beta = max(1, n_steps // n_beta)
    
    for bi, beta in enumerate(beta_schedule):
        steps = min(steps_per_beta, n_steps - bi * steps_per_beta)
        for t in range(steps):
            if time.monotonic() - t_start > time_limit_s:
                return best_score, best_a
            
            lr_t = lr / (1.0 + 1e-5 * t)
            score, grad = lse_score_grad(a, beta)
            
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            t_global = bi * steps_per_beta + t + 1
            m_hat = m / (1.0 - b1 ** t_global)
            v_hat = v / (1.0 - b2 ** t_global)
            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, 1e10)
            
            s = a.sum()
            if s > 1e-12:
                a /= s
            
            if (bi * steps_per_beta + t) % verbose_every == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def gs_optimize(a_init, n_iters=100000, relaxation=1.0, time_limit_s=60.0, t_start=None):
    """Gerchberg-Saxton error reduction."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    
    N_fft = 2 * n
    a_padded = np.zeros(N_fft, dtype=np.float64)
    
    if t_start is None:
        t_start = time.monotonic()
    
    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit_s:
            break
        
        a_padded[:n] = a
        a_padded[n:] = 0.0
        
        A = np.fft.rfft(a_padded)
        mag = np.abs(A)
        
        # Target: constant magnitude (mean of current)
        mean_mag = np.mean(mag)
        new_mag = (1.0 - relaxation) * mag + relaxation * mean_mag

        A_new = new_mag * np.exp(1j * phase)
        
        a_padded = np.fft.irfft(A_new, N_fft)
        a = np.maximum(a_padded[:n], 0.0)
        
        if a.sum() < 1e-12:
            break
        
        if it % 5000 == 0:

            if sc < best_score:

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_score, best_a


def ks_optimize(a_init, time_limit_s=60.0, t_start=None):
    """
    Katsaggelos-Serrano (KS) generalized projection.
    At each step, identify the lag k* where convolution peaks,
    then adjust the sequence to reduce the peak at k*.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    
    if t_start is None:
        t_start = time.monotonic()
    
    it = 0
    while time.monotonic() - t_start < time_limit_s:
        c = fast_conv(a)
        k_star = int(c.argmax())
        M = c[k_star]
        
        # Elements contributing to the peak:
        # c[k_star] = sum_j a[j] * a[k_star - j] for j in [max(0, k_star-n+1), min(k_star, n-1)]
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        
        if je < js:
            break
        
        # KS update: reduce elements with high "amplitude" at the peak
        # The idea: elements that are part of the peak and have large values should be reduced
        # More specifically, for each j in [js, je], the contribution is a[j] * a[k_star - j]
        # We want to reduce the product, so we reduce the larger of the pair
        
        # Simple approach: scale down all peak-contributing elements
        contrib = a[js:je+1].copy()
        # Weight by their partner's value
        partners = a[k_star - np.arange(js, je+1)]
        weights = np.maximum(contrib, partners)
        
        if weights.max() > 1e-15:
            # Reduce: a[j] *= (1 - step * weight[j] / max_weight)
            step = 0.1
            max_w = weights.max()
            reduction = step * weights / max_w
            a[js:je+1] *= (1.0 - reduction)
            a = np.maximum(a, 0.0)
            
            # Renormalize
            s = a.sum()
            if s > 1e-12:
                a /= s
        
        it += 1
        if it % 1000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_score, best_a


def run_lbfgs(a0, beta, maxiter=5000, maxfun=80000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        x_c = np.clip(x, 0.0, 1e10)
        sc, gr = lse_score_grad(x_c, beta)
        return sc, gr
    
    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def gen_init_kaiser(n, beta_param):
    """Kaiser window."""
    x = np.linspace(-1, 1, n)
    # Kaiser Bessel window
    beta_sq = beta_param * beta_param
    i0_num = np.ones(n)
    i0_den = np.ones(n)
    for k in range(1, 200):
        term_num = (beta_param * np.sqrt(np.maximum(1 - x**2, 0))) ** (2*k) / (np.prod(np.arange(1, k+1)) ** 2 if k > 0 else 1)
        # This is getting complicated, let me use scipy
    from scipy.special import i0
    w = i0(beta_param * np.sqrt(np.maximum(1 - x**2, 0))) / i0(beta_param)
    return w / w.sum()


def gen_init_raised_cos(n, power=2):
    """Raised cosine to the given power."""
    x = np.arange(n) / n
    w = (0.5 * (1 + np.cos(2 * np.pi * x))) ** power
    if w.sum() < 1e-12:
        w = np.ones(n)
    return w / w.sum()


def gen_init_flat_top(n):
    """Flat-top window (for flat frequency response)."""
    # Standard flat-top window coefficients
    a_coeffs = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
    x = np.arange(n)
    w = np.ones(n)
    for k, ak in enumerate(a_coeffs):
        w -= ak * np.cos(2 * np.pi * k * x / (n - 1))
    w = np.abs(w)
    return w / w.sum()


def gen_init_random_smooth(n, rng, smooth_k=10):
    """Random smooth initialization."""
    r = np.abs(rng.standard_normal(n)) + 0.5
    kernel = np.ones(smooth_k) / smooth_k
    r_smooth = np.convolve(r, kernel, mode='same')
    return r_smooth / r_smooth.sum()


def gen_init_u_shaped(n, strength=1.0):
    """U-shaped: large at edges, small in middle."""
    x = np.arange(n) / n
    w = 1.0 + strength * np.cos(np.pi * x)  # peaks at edges
    w = np.maximum(w, 0.01)
    return w / w.sum()


def gen_init_v_shaped(n, strength=1.0):
    """V-shaped: small at edges, large in middle."""
    x = np.arange(n) / n
    w = 1.0 - strength * np.cos(np.pi * x)  # valley in middle
    w = np.maximum(w, 0.01)
    return w / w.sum()
```
