```python
import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed score and its exact gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum()) + scaled.max() / beta
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # dL/da[i] = 2 * sum_j p[i+j] * a[j]
    # Using FFT: dL/da = 2 * (a_rev * p)[n-1 : 2n-1]
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad

        if abs(M_new - M) / (M + 1e-15) < 1e-8:
            break
    
    sc = compute_score(f)
    if sc < best_sc:
        best_sc = sc
        best_f = f.copy()

def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5

    def obj(x):
        s, g = lse_obj_grad(x, beta)
        return s, g

    res = minimize(
        obj, a0,

def error_reduction(a, n_iters=300):
    """Frequency-domain error reduction."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        v = A * A
        conv = np.fft.irfft(v, M)[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-16:
            break

        alpha = 0.3 + 0.6 * (it / max(1, n_iters - 1))
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-16:
            break

        conv_mod = conv - 0.5 * excess
        err = conv_mod - conv

        err_pad = np.zeros(M, dtype=np.float64)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_pad, M)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-10] = 0.0

        A_new = A + d
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def coordinate_descent_top_peaks(a, n_sweeps=3, n_top=40):
    """Reduce top peak contributors and redistribute."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            continue

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        # Reduce top contributors
        for j in top_global:
            old = a[j]
            for factor in [0.5, 0.7, 0.85]:
                a[j] = old * factor
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                else:
                    a[j] = old
                    break

    return best_a


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None and len(parent) > 10:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    # Hann window
    inits.append(0.5 * (1 + np.cos(2 * np.pi * x)))
    # Triangular
    inits.append(1.0 - np.abs(2.0 * x - 1.0))
    # Raised cosine variations
    for alpha in [0.3, 0.5, 0.7]:
        inits.append(0.5 * (1 + np.cos(2 * np.pi * (x - 0.5) / alpha)) ** alpha)
    # Gaussian-like
    inits.append(np.exp(-((x - 0.5) ** 2) / (2 * 0.1 ** 2)))
    inits.append(np.exp(-((x - 0.5) ** 2) / (2 * 0.2 ** 2)))
    # Optimal continuously: prolate-like
    inits.append(0.5 + 0.5 * np.cos(np.pi * (2 * x - 1) ** 2))
    # Flat with tapered edges
    t = 0.1
    a = np.ones(n)
    m = int(t * n)
    a[:m] = 0.5 * (1 - np.cos(np.pi * np.arange(m) / m))
    a[-m:] = a[:m][::-1]
    inits.append(a)
    # Random smooth
    from scipy.interpolate import interp1d
    x_coarse = rng.random(20)
    y_coarse = 0.3 + 0.7 * rng.random(20)
    f_interp = interp1d(x_coarse, y_coarse, kind='cubic', fill_value='extrapolate')
    inits.append(np.maximum(f_interp(x), 0.01))

    return inits
```

```python
"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 using:
1. LP-based iterative direction finding (solving a linear program at each step)
2. LSE-smoothed L-BFGS with aggressive progressive beta schedule
3. Targeted peak reduction via active set analysis
4. Multi-length exploration with interleaved optimization
5. Starting from provided parent sequence
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed score and its exact gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    exp_sum = exp_s.sum()
    if exp_sum < 1e-300:
        return 2.0 * n * c_max / (S * S), np.zeros(n)
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_sum
    # dL/da[i] = 2 * sum_j p[i+j] * a[j] via FFT
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_fn(x):
        return lse_obj_grad(x, beta)

    res = minimize(
        obj_fn, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def build_conv_matrix(f):
    """Build the convolution constraint matrix A where (A @ g)[k] = (f*g)[k]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_end >= i_start:
            A[k, i_start:i_end + 1] = f[i_end - i_start + 1 - 1:-1 + i_end - i_start + 1][::-1] if i_end > i_start else f[k - i_end]
    # Simpler construction
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A[k, i] = f[j]
    return A


def solve_lp_direction(f, M, timeout_hint=60.0):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns optimal g or None.
    """
    n = len(f)
    c_obj = -np.ones(n)

    # Build constraint matrix
    A_ub = np.zeros((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            A_ub[k, i] = f[k - i]

    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={"time_limit": timeout_hint})
        if res.success and res.x is not None:
            return np.clip(res.x, 0.0, None)
    except Exception as e:
        print(f"    LP error: {e}")
    return None


def lp_iteration(f, max_iters=15, t_init=0.15, t_final=0.03, lp_timeout=45.0):
    """
    LP-based iterative refinement:
    At each step, solve LP to find optimal g, then blend f with g.
    """
    n = len(f)
    f = np.maximum(f.astype(np.float64), 0.0).copy()
    S = float(f.sum())
    if S < 1e-12:
        return f
    f = f / S

    best_f = f.copy()
    best_sc = compute_score(f)
    print(f"    LP iteration start: score={best_sc:.8f}")

    for it in range(max_iters):
        c = fast_conv(f)
        M = float(c.max())

        # Small t for stability
        t = t_init * (t_final / t_init) ** (it / max(1, max_iters - 1))

        g = solve_lp_direction(f, M, timeout_hint=lp_timeout)
        if g is None:
            print(f"    LP failed at iter {it}, stopping")
            break

        g_sum = float(g.sum())
        if g_sum < 1e-12:
            break
        g = g / g_sum

        f_new = (1.0 - t) * f + t * g
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-12:
            break
        f_new = f_new / s

        sc = compute_score(f_new)
        if sc < best_sc:
            best_sc = sc
            best_f = f_new.copy()
            print(f"    LP iter {it}: score={sc:.8f} (improved)")
        else:
            print(f"    LP iter {it}: score={sc:.8f} (no improvement)")
            # Try smaller step
            f_new2 = (1.0 - t * 0.3) * f + t * 0.3 * g
            f_new2 = np.maximum(f_new2, 0.0)
            s2 = f_new2.sum()
            if s2 > 1e-12:
                f_new2 = f_new2 / s2
                sc2 = compute_score(f_new2)
                if sc2 < best_sc:
                    best_sc = sc2
                    best_f = f_new2.copy()
                    f = f_new2
                else:
                    break
            else:
                break
        f = f_new

    print(f"    LP iteration end: score={best_sc:.8f}")
    return best_f


def error_reduction(a, n_iters=200):
    """Frequency-domain error reduction to flatten convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M_fft)
        v = A * A
        conv = np.fft.irfft(v, M_fft)[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        alpha = 0.2 + 0.7 * (it / max(1, n_iters - 1))
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break

        conv_mod = conv - 0.6 * excess
        err = conv_mod - conv

        err_pad = np.zeros(M_fft, dtype=np.float64)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_pad, M_fft)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-10] = 0.0

        damp = 0.5 + 0.3 * np.sin(np.pi * it / max(1, n_iters))
        A_new = A + damp * d
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def targeted_peak_reduction(a, n_rounds=50, n_peaks=5):
    """
    Identify top peaks in convolution and apply targeted reduction.
    For each peak, reduce contributing elements and redistribute to low contributors.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for rd in range(n_rounds):
        c = fast_conv(a)
        # Get top peaks
        peak_idxs = np.argsort(c)[::-1][:n_peaks]

        improved = False
        for k_star in peak_idxs:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end < j_start:
                continue

            seg = a[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            total_contrib = float(contrib.sum())
            if total_contrib < 1e-15:
                continue

            # Find top and bottom contributors
            n_top = max(3, len(contrib) // 10)
            top_local = np.argsort(contrib)[::-1][:n_top]
            bot_local = np.argsort(contrib)[n_top:]

            if len(bot_local) == 0:
                continue

            # Try reducing top contributors by small amounts
            for frac in [0.01, 0.02, 0.05]:
                a_trial = a.copy()
                total_red = 0.0
                for li in top_local:
                    gi = j_start + li
                    red = a_trial[gi] * frac
                    a_trial[gi] -= red
                    total_red += red

                # Redistribute to bottom contributors
                per = total_red / len(bot_local)
                for li in bot_local:
                    gi = j_start + li
                    a_trial[gi] += per

                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s < 1e-12:
                    continue
                a_trial = a_trial / s

                sc = compute_score(a_trial)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
                    break
        if not improved:
            break

    return best_a


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None and len(parent) > 10:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    # Hann window
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x)))
    # Triangular
    inits.append(np.maximum(1.0 - np.abs(2.0 * x - 1.0), 0.01))
    # Flat top with cosine taper
    t_edge = 0.15
    a = np.ones(n)
    m = max(1, int(t_edge * n))
    ramp = 0.5 * (1.0 - np.cos(np.pi * np.arange(m) / m))
    a[:m] = ramp
    a[-m:] = ramp[::-1]
    inits.append(a)
    # Gaussian bump
    inits.append(np.exp(-((x - 0.5) ** 2) / (2.0 * 0.15 ** 2)))
    inits.append(np.exp(-((x - 0.5) ** 2) / (2.0 * 0.25 ** 2)))
    # Slepian-like (Doppler SIDELOBES)
    inits.append(0.5 + 0.5 * np.cos(np.pi * (2.0 * x - 1.0) ** 2))
    # Super-Gaussian
    inits.append(np.exp(-np.abs(2.0 * x - 1.0) ** 4 / 0.3))
    # Cosine power
    inits.append(np.abs(np.cos(np.pi * (2.0 * x - 1.0))) ** 0.5)
    inits.append(np.abs(np.cos(np.pi * (2.0 * x - 1.0))) ** 2.0)

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses LP-based iteration, progressive LSE-LBFGS, targeted peak reduction,
    and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                print(f"Loaded parent of length {len(parent)}")
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def report(sc, arr, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(arr.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(arr)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Choose lengths
    lengths = [2048, 4096, 6962, 8192]
    if parent is not None:
        pl = len(parent)
        if pl not in lengths:
            lengths.append(pl)
        lengths = sorted(set(lengths))[:5]

    print(f"Lengths to try: {lengths}")
    print(f"Budget: {total_budget}s, deadline at t={deadline - t_start:.0f}s")

    # ============================================================
    # Phase 1: Quick exploration with error reduction (20% time)
    # ============================================================
    phase1_end = t_start + 0.20 * (total_budget - 30)
    print(f"\n{'='*60}")
    print(f"Phase 1: Error Reduction Exploration (until t={phase1_end-t_start:.0f}s)")
    print(f"{'='*60}")

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"  n={n}")
        inits = build_initializations(n, rng, parent)
        remaining = phase1_end - time.monotonic()
        max_inits = max(2, int(remaining / 10.0))
        inits = inits[:min(len(inits), max_inits)]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break
            a_er = error_reduction(a0, n_iters=150)
            sc = compute_score(a_er)
            report(sc, a_er, f"ER n={n} i={i}")

    print(f"  After Phase 1: best = {best_score:.10f}")

    # ============================================================
    # Phase 2: LP-based iteration (20% time)
    # ============================================================
    phase2_end = t_start + 0.40 * (total_budget - 30)
    print(f"\n{'='*60}")
    print(f"Phase 2: LP-Based Iteration (until t={phase2_end-t_start:.0f}s)")
    print(f"{'='*60}")

    # Use moderate n for LP (matrix construction is O(n^2))
    lp_lengths = [n for n in lengths if 1024 <= n <= 4096]
    if not lp_lengths:
        lp_lengths = [2048]

    for n_lp in lp_lengths:
        if time.monotonic() > phase2_end:
            break
        print(f"  LP on n={n_lp}")

        # Best starting point for this length
        if best_a is not None:
            if len(best_a) == n_lp:
                a_start = best_a.copy()
            else:
                a_start = np.interp(
                    np.linspace(0, 1, n_lp),
                    np.linspace(0, 1, len(best_a)),
                    best_a
                )
        else:
            a_start = 0.5 * (1.0 + np.cos(2.0 * np.pi * np.arange(n_lp) / n_lp))

        # First do error reduction to get a good start
        a_start = error_reduction(a_start, n_iters=100)
        sc_start = compute_score(a_start)
        report(sc_start, a_start, f"ER pre-LP n={n_lp}")

        # LP iterations
        remaining_lp = phase2_end - time.monotonic()
        max_lp_iters = max(3, int(remaining_lp / 60.0))
        a_lp = lp_iteration(a_start, max_iters=min(max_lp_iters, 20),
                           t_init=0.1, t_final=0.02, lp_timeout=40.0)
        sc = compute_score(a_lp)
        report(sc, a_lp, f"LP n={n_lp}")

    # LP on the best length if it's moderate
    if best_a is not None and time.monotonic() < phase2_end:
        n_best = len(best_a)
        if 1024 <= n_best <= 4096 and n_best not in lp_lengths:
            print(f"  LP on best length n={n_best}")
            a_lp2 = lp_iteration(best_a, max_iters=10, t_init=0.08, t_final=0.02, lp_timeout=35.0)
            sc = compute_score(a_lp2)
            report(sc, a_lp2, f"LP best_n={n_best}")

    print(f"  After Phase 2: best = {best_score:.10f}")

    # ============================================================
    # Phase 3: Progressive LSE-LBFGS (35% time)
    # ============================================================
    phase3_end = t_start + 0.75 * (total_budget - 30)
    print(f"\n{'='*60}")
    print(f"Phase 3: Progressive LSE-LBFGS (until t={phase3_end-t_start:.0f}s)")
    print(f"{'='*60}")

    if best_a is not None:
        # Progressive beta: start moderate, go very high
        beta_schedule = [200, 500, 1000, 3000, 10000, 30000, 100000, 300000]
        for beta in beta_schedule:
            if time.monotonic() > phase3_end:
                break
            remaining = phase3_end - time.monotonic()
            if remaining < 30:
                break
            maxiter = min(15000, int(remaining * 3))
            maxfun = min(100000, int(remaining * 10))
            print(f"  L-BFGS beta={beta} (maxiter={maxiter}, maxfun={maxfun})")
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
            report(sc, a_opt, f"LBFGS b={beta}")

    print(f"  After Phase 3: best = {best_score:.10f}")

    # ============================================================
    # Phase 4: Targeted peak reduction + fine L-BFGS (15% time)
    # ============================================================
    phase4_end = t_start + 0.90 * (total_budget - 30)
    print(f"\n{'='*60}")
    print(f"Phase 4: Targeted Peak Reduction (until t={phase4_end-t_start:.0f}s)")
    print(f"{'='*60}")

    if best_a is not None:
        # Targeted peak reduction
        remaining = phase4_end - time.monotonic()
        n_rounds = max(10, int(remaining / 2.0))
        a_tpr = targeted_peak_reduction(best_a, n_rounds=n_rounds, n_peaks=5)
        sc = compute_score(a_tpr)
        report(sc, a_tpr, "TPR")

        # Fine L-BFGS after TPR
        if time.monotonic() < phase4_end:
            sc2, a_opt = run_lbfgs(a_tpr, 50000, maxiter=10000, maxfun=100000)
            report(sc2, a_opt, "TPR+LBFGS b=50k")

    print(f"  After Phase 4: best = {best_score:.10f}")

    # ============================================================
    # Phase 5: Diversification (7% time)
    # ============================================================
    phase5_end = t_start + 0.97 * (total_budget - 30)
    print(f"\n{'='*60}")
    print(f"Phase 5: Diversification (until t={phase5_end-t_start:.0f}s)")
    print(f"{'='*60}")

    if best_a is not None:
        n_best = len(best_a)
        n_trials = int((phase5_end - time.monotonic()) / 20.0)
        n_trials = max(1, min(n_trials, 8))

        for trial in range(n_trials):
            if time.monotonic() > phase5_end:
                break

            if trial == 0:
                # Gaussian perturbation
                a_pert = best_a * (1.0 + 0.03 * rng.standard_normal(n_best))
                a_pert = np.clip(a_pert, 0.0, 1000.0)
            elif trial == 1:
                # Sinusoidal perturbation
                freq = 3.0 + 10.0 * rng.random()
                phase = 2.0 * np.pi * rng.random()
                amp = 0.02 + 0.03 * rng.random()
                x = np.arange(n_best, dtype=np.float64) / n_best
                a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
                a_pert = np.clip(a_pert, 0.0, 1000.0)
            elif trial == 2:
                # Edge reduction
                a_pert = best_a.copy()
                m = max(1, n_best // 15)
                a_pert[:m] *= 0.4
                a_pert[-m:] *= 0.4
            elif trial == 3:
                # Middle boost
                a_pert = best_a.copy()
                m = n_best // 8
                a_pert[m:2*m] *= 1.2
                a_pert[-2*m:-m] *= 1.2
            elif trial == 4:
                # Broad perturbation
                a_pert = best_a * (1.0 + 0.05 * rng.standard_normal(n_best))
                a_pert = np.clip(a_pert, 0.0, 1000.0)
            elif trial == 5:
                # Frequency perturbation
                M_fft = 1 << (2 * n_best - 1).bit_length()
                A = np.fft.rfft(np.concatenate([best_a, np.zeros(M_fft - n_best)]), M_fft)
                # Dampen high frequencies slightly
                n_freq = len(A)
                for fi in range(n_freq // 3, n_freq):
                    A[fi] *= 0.95
                a_pert = np.fft.irfft(A, M_fft)[:n_best]
                a_pert = np.clip(a_pert, 0.0, 1000.0)
            else:
                # Combine best with a fresh ER result
                a_fresh = error_reduction(
                    best_a * (1.0 + 0.02 * rng.standard_normal(n_best)),
                    n_iters=80
                )
                a_pert = 0.5 * best_a + 0.5 * a_fresh
                a_pert = np.clip(a_pert, 0.0, 1000.0)

            s = a_pert.sum()
            if s < 1e-12:
                continue

            sc, a_opt = run_lbfgs(a_pert, 20000, maxiter=8000, maxfun=80000)
            report(sc, a_opt, f"div#{trial}")

    print(f"  After Phase 5: best = {best_score:.10f}")

    # ============================================================
    # Phase 6: Final squeeze
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Final Squeeze")
    print(f"{'='*60}")

    if best_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            sc, a_opt = run_lbfgs(best_a, 500000, maxiter=20000, maxfun=400000)
            report(sc, a_opt, "final b=500k")
        elif remaining > 30:
            sc, a_opt = run_lbfgs(best_a, 200000, maxiter=10000, maxfun=200000)
            report(sc, a_opt, "final b=200k")

        if remaining > 20 and best_score > 1.503:
            a_tpr2 = targeted_peak_reduction(best_a, n_rounds=30, n_peaks=8)
            sc = compute_score(a_tpr2)
            report(sc, a_tpr2, "final TPR")
            if remaining > 40:
                sc2, a_opt = run_lbfgs(a_tpr2, 100000, maxiter=10000, maxfun=100000)
                report(sc2, a_opt, "final TPR+LBFGS")

    # ============================================================
    # Final verification
    # ============================================================
    if best_a is None:
        best_a = np.ones(64) * 0.5

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL RESULT:")
    print(f"  Internal score: {best_score:.10f}")
    print(f"  Verified score: {verified:.10f}")
    print(f"  Sequence length: {len(out)}")
    print(f"  Time used: {elapsed_total:.1f}s / {total_budget}s")
    print(f"{'='*60}")
    return out
```
