"""
LSE-Softmax Multiplicative Weight Optimizer for minimum-peak autocorrelation.
Key innovations over previous approach:
1. Softmax (log-space) parametrization for automatic positivity/normalization
2. Clean gradient structure df/dx_i = 4n*a_i*(q_i - r) enabling efficient MWM updates
3. Cross-correlation via FFT for O(n log n) gradient computation
4. Adaptive top-k peak weighting for aggressive peak reduction
5. Multi-n exploration with upsample/downsample
6. L-BFGS in x-space for second-order convergence
7. Peak-migration coordinate descent as a complementary local search
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_corr(a, p):
    """
    Linear cross-correlation: r[m] = sum_j a[j]*p[j+m]
    a has length n, p has length 2n-1. Returns first n values.
    """
    n = len(a)
    N = n + len(p) - 1
    N_fft = 1 << (N - 1).bit_length()
    A_f = np.fft.rfft(a, N_fft)
    P_f = np.fft.rfft(p, N_fft)
    r = np.fft.irfft(np.conj(A_f) * P_f, N_fft)
    return r[:n]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_peak_distribution(c, beta):
    """Compute the LSE-smoothed peak distribution p_k."""
    c_max = c.max()
    scaled = beta * (c - c_max)
    # Numerical stability
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    p = exp_s / exp_s.sum()
    L = c_max + c_max + np.log(exp_s.sum()) / beta  # LSE value
    return p, L


def lse_mwm_step(a, beta, lr):
    """
    One LSE-Softmax MWM step.
    Returns updated a and the smoothed score.
    """
    n = len(a)
    c = fast_conv(a)
    p, L = lse_peak_distribution(c, beta)
    S = a.sum()
    score = 2.0 * n * L / (S * S)
    
    # q_i = sum_k p[k] * a[k-i] = cross_corr(a, p)[i]
    q = cross_corr(a, p)
    r = float(np.dot(p, c))
    
    # Gradient in x-space: df/dx_i = 4n * a_i * (q_i - r) / S^2
    # For the multiplicative update: a_i <- a_i * exp(-lr * 4n * (q_i - r) / S^2)
    h = q - r
    factor = -lr * 4.0 * n * h / (S * S)
    # Clip for numerical stability
    factor = np.clip(factor, -50.0, 50.0)
    a_new = a * np.exp(factor)
    a_new = np.maximum(a_new, 1e-30)
    a_new = a_new / a_new.sum()
    return a_new, score


def lse_adam_step(x, a, beta, lr, m, v, t, b1=0.9, b2=0.999, eps=1e-8):
    """One Adam step in x-space with LSE smoothing."""
    n = len(a)
    c = fast_conv(a)
    p, L = lse_peak_distribution(c, beta)
    S = a.sum()
    
    q = cross_corr(a, p)
    r = float(np.dot(p, c))
    grad = 4.0 * n * a * (q - r) / (S * S)
    
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * grad * grad
    m_hat = m / (1 - b1 ** t)
    v_hat = v / (1 - b2 ** t)
    x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
    
    # Numerical safety
    x_new = x_new - x_new.max()
    a_new = np.exp(x_new)
    a_new = a_new / a_new.sum()
    
    score = 2.0 * n * L / (S * S)
    return x_new, a_new, m, v, score


def lse_soft_score_grad_x(x, beta):
    """
    LSE-smoothed score and gradient w.r.t. x (where a = softmax(x)).
    For use with scipy L-BFGS-B.
    """
    x_safe = x - x.max()
    ex = np.exp(x_safe)
    a = ex / ex.sum()
    n = len(a)
    
    c = fast_conv(a)
    c_max = c.max()
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    p = exp_s / exp_s.sum()
    L = c_max + np.log(exp_s.max() + exp_s.sum() - exp_s.max()) / beta
    # More stable: L = c_max + logsumexp(scaled) / beta
    # Already did: scaled = beta*(c - c_max), then subtracted max for stability
    # logsumexp(scaled) = max(scaled) + log(sum(exp(scaled - max(scaled))))
    # But we already computed exp_s from the stabilized scaled
    L = c_max + float(np.log(exp_s.sum())) / beta
    
    S = a.sum()
    score = 2.0 * n * L / (S * S)
    
    q = cross_corr(a, p)
    r = float(np.dot(p, c))
    grad = 4.0 * n * a * (q - r) / (S * S)
    
    return score, grad


def lse_soft_score_grad_x_bound(x, beta):
    """Version with bounded x for L-BFGS."""
    x_c = np.clip(x, -50.0, 50.0)
    return lse_soft_score_grad_x(x_c, beta)


def run_lbfgs_x(x0, beta, maxiter=3000, maxfun=50000, time_limit=60.0):
    """L-BFGS-B in x-space."""
    x0 = x0.copy()
    x0 = x0 - x0.max()
    
    def obj(x):
        sc, gr = lse_soft_score_grad_x_bound(x, beta)
        return sc, gr
    
    res = minimize(obj, x0, method='L-BFGS-B', jac=True,
                   options={'ftol': 1e-14, 'gtol': 1e-10,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30})
    x_out = np.clip(res.x, -50.0, 50.0)
    x_out = x_out - x_out.max()
    ex = np.exp(x_out)
    a_out = ex / ex.sum()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res.nit


def run_lbfgs_a(a0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B in a-space with LSE smoothing and non-negativity constraint."""
    n = len(a0)
    a0 = np.clip(a0, 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        x_c = np.maximum(x, 0.0)
        n_c = len(x_c)
        c = fast_conv(x_c)
        c_max = c.max()
        scaled = beta * (c - c_max)
        scaled -= scaled.max()
        exp_s = np.exp(scaled)
        p = exp_s / exp_s.sum()
        L = c_max + float(np.log(exp_s.sum())) / beta
        S = x_c.sum()
        score = 2.0 * n_c * L / (S * S)
        
        # Gradient w.r.t. a (not x)
        # df/da_i = 2n * sum_k p_k * (2*a_{k-i}) / S^2 - 4n*L*a_i/S^3
        n_c = len(x_c)
        sg = np.zeros(n_c)
        for k_idx in range(len(p)):
            if p[k_idx] < 1e-15:
                continue
            js = max(0, k_idx - n_c + 1)
            je = min(k_idx, n_c - 1)
            if je >= js:
                idx = k_idx - np.arange(js, je + 1)
                sg[js:je+1] += p[k_idx] * 2.0 * x_c[idx]
        
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n_c * sg / S2 - 4.0 * n_c * L / S3
        return score, grad
    
    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-14, 'gtol': 1e-10,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30})
    a_out = np.clip(res.x, 0.0, 1000.0)
    a_out = a_out / a_out.sum()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_coord_descent(a, n_sweeps=50, deadline=np.inf):
    """
    Coordinate descent targeting peak reduction.
    For each coordinate, try to optimally adjust it to reduce the peak.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        k_star = int(c.argmax())
        score = best_score
        
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        
        # Try reducing peak contributors
        for j in range(js, je + 1):
            if time.monotonic() > deadline:
                break
            if a[j] < 1e-12:
                continue
            a_old = a[j]
            # Try several reduction fractions
            improved = False
            for frac in [0.7, 0.5, 0.3, 0.1]:
                a[j] = a_old * frac
                # Redistribute to non-peak elements
                removed = a_old - a[j]
                non_peak = np.ones(n, dtype=bool)
                non_peak[js:je+1] = False
                if non_peak.sum() > 0:
                    a[non_peak] += removed / non_peak.sum()
                sc = compute_score(a)
                if sc < score:
                    score = sc
                    improved = True
                    if sc < best_score:
                        best_score = sc
                        a_best = a.copy()
                    break
                else:
                    a[j] = a_old
                    non_peak = np.ones(n, dtype=bool)
                    non_peak[js:je+1] = False
                    if non_peak.sum() > 0:
                        a[non_peak] -= removed / non_peak.sum()
            if not improved:
                a[j] = a_old
        
        # Also try a random non-peak element increase
        non_peak = np.ones(n, dtype=bool)
        non_peak[js:je+1] = False
        nc_idx = np.where(non_peak)[0]
        if len(nc_idx) > 0 and np.random.randn() > 0:
            j = nc_idx[np.random.randint(len(nc_idx))]
            a_old = a[j]

                a[j] = a_old * mult
                sc = compute_score(a)
                if sc < score:
                    score = sc
                    if sc < best_score:
                        best_score = sc
                    break
                a[j] = a_old
    
    return best_score, a


def resample_sequence(a, new_n):
    """Resample sequence to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.maximum(a_new, 0.0)
    if a_new.sum() > 0:
        a_new = a_new / a_new.sum()
    return a_new


def generate_diverse_inits(rng, n, parent=None):
    """Generate diverse initializations."""
    inits = []
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Gaussian centered
    x = np.arange(n) / n
    for sigma in [0.1, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        inits.append(g / g.sum())
    
    # Raised cosine (Hann)
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(rc / rc.sum())
    
    # Sine (exclude endpoints to keep positive)
    s = np.sin(np.pi * x)
    s = np.maximum(s, 0.001)
    inits.append(s / s.sum())
    
    # Cosine modulated
    for freq in [1.0, 2.0, 3.0]:
        for amp in [0.3, 0.5, 0.7]:
            c = 1.0 + amp * np.cos(2 * np.pi * freq * x)
            c = np.maximum(c, 0.001)
            inits.append(c / c.sum())
    
    # Two-bump
    for c1, c2, s1, s2 in [(0.25, 0.75, 0.1, 0.1), (0.2, 0.8, 0.15, 0.15),
                            (0.3, 0.7, 0.08, 0.08)]:
        b = np.exp(-0.5 * ((x - c1) / s1) ** 2) + np.exp(-0.5 * ((x - c2) / s2) ** 2)
        inits.append(b / b.sum())
    
    # Random smooth
    r = np.abs(rng.standard_normal(n)) + 0.3
    k = max(3, n // 30)
    kernel = np.ones(k) / k
    r_sm = np.convolve(r, kernel, mode='same')
    inits.append(r_sm / r_sm.sum())
    
    # Add parent if available
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.copy())
        else:
            inits.append(resample_sequence(parent, n))
    
    return inits

        t0 = time.monotonic()
        sc, a_opt = optimize_with_mwm(init, screen_betas, screen_lrs,
                                       min(time_per_init, 5.0), t0)
        candidates.append((sc, a_opt))
        print(f"  Init {i}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    candidates.sort(key=lambda x: x[0])
    for sc, a_opt in candidates[:3]:
        try_update(sc, a_opt, "screening")
    
    # Also quick MWM on parent
    if parent is not None:
        t0 = time.monotonic()
        sc, a_opt = optimize_with_mwm(parent, [100, 500], [0.08, 0.04],
                                       min(5.0, p1_time * 0.3), t0)
        try_update(sc, a_opt, "parent MWM quick")
    
    print(f"  Best after screening: {best_score:.8f}")
    
    # ==========================================
    # Phase 2: Intensive MWM optimization (35% time)
    # ==========================================
    p2_time = effective_budget * 0.35
    print(f"\n{'='*60}")
    print(f"Phase 2: Intensive MWM (budget={p2_time:.0f}s)")
    print(f"{'='*60}")
    
    # Beta and lr schedules for MWM
    mwm_betas = [20, 50, 100, 300, 1000, 3000]
    mwm_lrs = [0.15, 0.12, 0.10, 0.08, 0.05, 0.03]
    
    t0 = time.monotonic()
    sc, a_opt = optimize_with_mwm(best_a, mwm_betas, mwm_lrs, p2_time * 0.6, t0)
    try_update(sc, a_opt, "MWM full schedule")
    
    # Continue from best with finer schedule
    remaining_p2 = (t_start + effective_budget * 0.45) - time.monotonic()
    if remaining_p2 > 5:
        fine_betas = [1000, 3000, 10000]
        fine_lrs = [0.03, 0.02, 0.01]
        t0 = time.monotonic()
        sc, a_opt = optimize_with_mwm(best_a, fine_betas, fine_lrs, remaining_p2, t0)
        try_update(sc, a_opt, "MWM fine schedule")
    
    print(f"  Best after MWM: {best_score:.8f}")
    
    # ==========================================
    # Phase 3: Adam in x-space (20% time)
    # ==========================================
    p3_time = effective_budget * 0.20
    print(f"\n{'='*60}")
    print(f"Phase 3: Adam-x optimization (budget={p3_time:.0f}s)")
    print(f"{'='*60}")
    
    # Convert to x-space
    x_init = np.log(np.maximum(best_a, 1e-30))
    x_init = x_init - x_init.max()
    
    adam_betas = [100, 500, 2000]
    adam_lrs = [0.01, 0.005, 0.002]
    
    t0 = time.monotonic()
    sc, a_opt = optimize_with_adam_x(x_init, adam_betas, adam_lrs, p3_time * 0.5, t0)
    try_update(sc, a_opt, "Adam-x")
    
    if a_opt is not None and compute_score(a_opt) < best_score:
        x_init = np.log(np.maximum(a_opt, 1e-30))
        x_init = x_init - x_init.max()
    
    # Fine Adam
    remaining_p3 = (t_start + effective_budget * 0.60) - time.monotonic()
    if remaining_p3 > 5:
        fine_adam_betas = [2000, 10000]
        fine_adam_lrs = [0.002, 0.0005]
        t0 = time.monotonic()
        sc, a_opt = optimize_with_adam_x(x_init, fine_adam_betas, fine_adam_lrs,
                                          min(remaining_p3, p3_time * 0.5), t0)
        try_update(sc, a_opt, "Adam-x fine")
    
    print(f"  Best after Adam-x: {best_score:.8f}")
    
    # ==========================================
    # Phase 4: L-BFGS polishing (20% time)
    # ==========================================
    p4_time = effective_budget * 0.20
    print(f"\n{'='*60}")
    print(f"Phase 4: L-BFGS polishing (budget={p4_time:.0f}s)")
    print(f"{'='*60}")
    
    # L-BFGS in x-space
    x_init = np.log(np.maximum(best_a, 1e-30))
    x_init = x_init - x_init.max()
    
    n_lbfgs = len(best_a)
    # Scale maxfun based on n
    maxfun_x = max(10000, min(50000, 200 * n_lbfgs))
    
    for beta in [500, 2000, 8000]:
        remaining = (t_start + effective_budget * 0.80) - time.monotonic()
        if remaining < 3:
            break
        t0 = time.monotonic()
        sc, a_opt, nit = run_lbfgs_x(x_init, beta, maxiter=2000, maxfun=maxfun_x,
                                       time_limit=min(remaining, p4_time * 0.3))
        try_update(sc, a_opt, f"LBFGS-x beta={beta} ({nit} iters)")
        if a_opt is not None and compute_score(a_opt) < best_score:
            x_init = np.log(np.maximum(a_opt, 1e-30))
            x_init = x_init - x_init.max()
    
    # L-BFGS in a-space
    remaining = (t_start + effective_budget * 0.80) - time.monotonic()
    if remaining > 5:
        for beta in [1000, 5000]:
            remaining = (t_start + effective_budget * 0.80) - time.monotonic()
            if remaining < 3:
                break
            t0 = time.monotonic()
            sc, a_opt = run_lbfgs_a(best_a, beta, maxiter=2000, maxfun=maxfun_x)
            try_update(sc, a_opt, f"LBFGS-a beta={beta}")
    
    print(f"  Best after L-BFGS: {best_score:.8f}")
    
    # ==========================================
    # Phase 5: Peak-targeting coordinate descent (10% time)
    # ==========================================
    p5_time = effective_budget * 0.10
    print(f"\n{'='*60}")
    print(f"Phase 5: Peak coord descent (budget={p5_time:.0f}s)")
    print(f"{'='*60}")
    
    t0 = time.monotonic()
    deadline = t_start + effective_budget * 0.90
    sc, a_opt = peak_coord_descent(best_a, n_sweeps=100, deadline=deadline)
    try_update(sc, a_opt, "peak coord descent")
    
    print(f"  Best after coord descent: {best_score:.8f}")
    
    # ==========================================
    # Phase 6: Multi-n exploration (5% time)
    # ==========================================
    p6_time = effective_budget * 0.05
    print(f"\n{'='*60}")
    print(f"Phase 6: Multi-n exploration (budget={p6_time:.0f}s)")
    print(f"{'='*60}")
    
    for n_try in candidate_ns:
        if n_try == n_parent:
            continue
        if time.monotonic() > t_start + effective_budget - 30:
            break
        if n_try < 128 or n_try > 32768:
            continue
        
        a_resampled = resample_sequence(best_a, n_try)
        t0 = time.monotonic()
        sc, a_opt = optimize_with_mwm(a_resampled, [100, 500, 2000],
                                       [0.08, 0.05, 0.03],
                                       min(p6_time * 0.5, 15.0), t0)
        try_update(sc, a_opt, f"multi-n={n_try}")
    
    # ==========================================
    # Phase 7: Final intensive polish (remaining time)
    # ==========================================
    p7_time = max(5.0, effective_budget - (time.monotonic() - t_start) - 10.0)
    print(f"\n{'='*60}")
    print(f"Phase 7: Final polish (budget={p7_time:.0f}s)")
    print(f"{'='*60}")
    
    final_deadline = t_start + effective_budget - 5.0
    
    # Very high beta MWM
    if time.monotonic() < final_deadline:
        t0 = time.monotonic()
        sc, a_opt = optimize_with_mwm(best_a, [5000, 20000], [0.02, 0.01],
                                       min(p7_time * 0.4, final_deadline - time.monotonic()), t0)
        try_update(sc, a_opt, "final MWM high-beta")
    
    # L-BFGS with very high beta
    if time.monotonic() < final_deadline - 2:
        x_init = np.log(np.maximum(best_a, 1e-30))
        x_init = x_init - x_init.max()
        for beta in [10000, 30000]:
            if time.monotonic() > final_deadline - 2:
                break
            t0 = time.monotonic()
            sc, a_opt, nit = run_lbfgs_x(x_init, beta, maxiter=3000, maxfun=maxfun_x,
                                           time_limit=min(30.0, final_deadline - time.monotonic() - 2))
            try_update(sc, a_opt, f"final LBFGS-x beta={beta}")
            if a_opt is not None and compute_score(a_opt) < best_score:
                x_init = np.log(np.maximum(a_opt, 1e-30))
                x_init = x_init - x_init.max()
    
    # Final coord descent
    if time.monotonic() < final_deadline - 2:
        sc, a_opt = peak_coord_descent(best_a, n_sweeps=30,
                                        deadline=final_deadline - 2)
        try_update(sc, a_opt, "final coord")
    
    # ==========================================
    # Final verification
    # ==========================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified
    
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
