def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def power_sum_smoothed(c, p, beta=1.0):
    """Compute (sum c^p / len(c))^(1/p) as a smoothed max approximation."""

    c_clipped = np.maximum(c, 1e-30)

    exp_scaled = np.exp(scaled)
    L = c_max + np.log(exp_scaled.sum()) / beta
    return L

def fienup_hybrid(a, n_iters=500, relax=0.3, target_flat=True):
    """
    Hybrid Fienup (IO + ER combined).
    IO: enforce target magnitude (flat spectrum)
    ER: enforce non-negativity
    Combined with relaxation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / n_iters
        # Adaptive relaxation: start conservative, become more aggressive
        t_relax = relax * (0.5 + 0.5 * progress)

        A = np.fft.rfft(a, M_fft)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)
        mag2 = mag * mag  # = |A|^2

        # Target: flat spectrum (or slightly shaped)
        if target_flat:
            target_mag2 = np.ones_like(mag2) * float(mag2.mean())
        else:
            # Slightly raised cosine target (emphasizes low frequencies)
            freqs = np.arange(len(mag2)) / len(mag2)
            target_mag2 = 1.0 + 0.3 * np.cos(np.pi * freqs)
            target_mag2 *= float(mag2.sum()) / float(target_mag2.sum())

        # IO step: replace magnitude with target, keep phase
        target_mag = np.sqrt(np.maximum(target_mag2, 1e-30))
        A_io = target_mag * phase
        a_io = np.fft.irfft(A_io, M_fft)[:n]

        # ER step: just enforce a >= 0 on current
        a_er = np.maximum(a, 0.0)

        # Hybrid: blend
        a_new = (1 - t_relax) * a_er + t_relax * a_io
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            a_new = a_er
            S_new = S
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_spectral_shaping(a, n_iters=400, shape_power=2.0):
    """
    Fienup with shaped target spectrum.
    Instead of flat, use a spectrum shaped to reduce the peak of inverse DFT.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / n_iters
        # Shape parameter: start flat, become more shaped
        shape = shape_power * progress

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Target spectrum: raised to a power (flattening)
        # P_target = P_mean * (P / P_mean)^(-shape) = P_mean * (P_mean/P)^(shape)
        # For shape=0: target = P (no change)
        # For shape=1: target = P_mean (flat)
        # For shape>1: even flatter (suppresses peaks more)
        ratio = P / P_mean
        target_P = P_mean * np.power(np.maximum(ratio, 1e-10), -shape)
        
        # Blend target with current
        t_blend = 0.5 + 0.3 * progress
        P_new = (1 - t_blend) * P + t_blend * target_P
        P_new = np.maximum(P_new, 1e-30)

        # Update magnitude, keep phase
        target_mag = np.sqrt(P_new)
        current_mag = np.sqrt(np.maximum(P, 1e-30))
        phase = A / (current_mag + 1e-30)
        A_new = target_mag * phase
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            continue
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def nesterov_gradient(a, n_iters=5000, beta=500.0, momentum=0.9, lr=0.01):
    """
    Nesterov-accelerated gradient descent on LSE-smoothed objective.
    With adaptive learning rate.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    # Nesterov: y = a + momentum * (a - a_prev)
    a_prev = a.copy()
    y = a.copy()
    v = np.zeros(n)  # velocity (optional, for heavy ball)

    for it in range(n_iters):
        # Evaluate at y (Nesterov look-ahead)
        sc_y, grad_y = lse_score_and_grad(y, beta)
        
        # Check if good
        sc_exact = compute_score(y)
        if sc_exact < best_sc:
            best_sc = sc_exact
            best_a = y.copy()

        # Adaptive step size
        # Use the gap between smoothed and exact as a guide
        gap = sc_y - 2.0 * n * float(np.max(fast_conv(y))) / (float(y.sum()) ** 2)
        # Actually this is always >= 0 and should be small for large beta
        
        # Gradient step with Nesterov momentum
        a_new = y - lr * grad_y
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            a_new = a_prev
            S_new = S
        a_new = a_new / S_new

        # Update momentum
        y = a_new + momentum * (a_new - a_prev)
        y = np.maximum(y, 0.0)
        S_y = float(y.sum())
        if S_y < 1e-12:
            y = a_new
            S_y = S_new
        y = y / S_y

        a_prev = a_new

        # Periodically reduce learning rate
        if it % 1000 == 999:
            lr *= 0.8
            momentum = min(momentum + 0.01, 0.95)

    return best_a

def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))

    inits.append(np.maximum(0.3 + 0.3 * np.cos(6.0 * np.pi * x) + 0.2 * np.cos(2.0 * np.pi * x), 0.01))
    return inits

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """
    LSE-smoothed score and its gradient.
    Score = 2n * LSE_beta(c) / S^2 where c = a*a (linear conv).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 2.0 * n * c_max / (S * S), np.zeros(n)
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)

    # p_k = weight of each conv element
    p = exp_scaled / exp_sum

    # Compute sum_k p_k * dc_k/da_j = 2*cross[j] - p[2j]*a[j] (diagonal)
    # cross[j] = sum_k p_k * a[k-j] = conv(p, a_rev)[j + n - 1]
    rev_a = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    cross = conv_res[n - 1:2 * n - 1]

    # Diagonal correction: where 2j < 2n-1, subtract p[2j]*a[j]
    dc_da = 2.0 * cross
    valid_diag = 2 * np.arange(n) < 2 * n - 1
    diag_idx = 2 * np.arange(n)[valid_diag]
    dc_da[valid_diag] -= p[diag_idx] * a[valid_diag]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dc_da / S2 - 2.0 * L / S3)

    return score, grad


def lse_objective_top(x, beta):
    """Top-level wrapper for scipy minimize (no closures)."""
    return lse_score_and_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
        S = a0.sum()
    a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_top, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fienup_hybrid_io_er(a, n_iters=500, relax=0.3):
    """
    Hybrid Fienup combining IO (Input-Output) and ER (Error Reduction).
    IO: replace magnitude with target (flat), keep phase.
    ER: enforce non-negativity directly.
    Blend: a_new = (1-t)*ER + t*IO, then clip to non-negative.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        # Adaptive relaxation: start with more ER, move toward IO
        t_relax = relax * (0.4 + 0.6 * progress)

        A = np.fft.rfft(a, M_fft)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)

        # Target: flat power spectrum
        target_P = float((mag * mag).mean()) * np.ones_like(mag)
        target_mag = np.sqrt(target_P)

        # IO step
        A_io = target_mag * phase
        a_io = np.fft.irfft(A_io, M_fft)[:n]

        # ER step (just enforce non-negativity on current)
        a_er = np.maximum(a, 0.0)

        # Hybrid blend
        a_new = (1.0 - t_relax) * a_er + t_relax * a_io
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            a_new = a_er
            S_new = S
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_shaped(a, n_iters=400, shape_param=1.0):
    """
    Fienup with boundary-aware shaped target spectrum.
    The linear convolution boundary means the "natural" spectrum isn't perfectly flat.
    We use a slightly raised target at low frequencies to compensate.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    n_bins = M_fft // 2 + 1
    best_a = a.copy()
    best_sc = compute_score(a)

    # Build target spectrum: slightly raised at edges (accounts for linear conv boundary)
    freqs = np.arange(n_bins, dtype=np.float64) / n_bins
    # Raised cosine shape: 1 + shape_param * cos(2*pi*freqs)
    target_shape = 1.0 + shape_param * np.cos(2.0 * np.pi * freqs)
    target_shape = np.maximum(target_shape, 0.1)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        # Blend factor: start with more target, keep more current later
        t_blend = 0.4 + 0.3 * progress

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Target power spectrum (normalized to same total energy)
        target_P = P_mean * target_shape
        target_P *= P.sum() / target_P.sum()

        # Blend in frequency domain
        P_new = (1.0 - t_blend) * P + t_blend * target_P
        P_new = np.maximum(P_new, 1e-30)

        # Update magnitude, preserve phase
        target_mag = np.sqrt(P_new)
        current_mag = np.sqrt(np.maximum(P, 1e-30))
        phase = A / (current_mag + 1e-30)
        A_new = target_mag * phase
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            continue
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_adaptive_multi(a, n_iters=300):
    """
    Adaptive multi-level Fienup: progressively tighten the spectrum flattening.
    Uses multiple threshold levels that tighten over iterations.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        n_levels = 2 + int(5 * progress)
        clip_str = 0.3 + 0.6 * progress

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        scale = np.ones_like(P)
        for lv in range(n_levels):
            thresh = P_mean * (1.0 + 0.2 * (lv + 1) / n_levels)
            above = P > thresh
            if above.any():
                red = np.sqrt(np.maximum(thresh / P[above], 0.001))
                scale[above] *= (1.0 - clip_str) + clip_str * red

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def nesterov_gradient(a, n_iters=5000, beta=500.0, momentum_init=0.85, lr_init=0.01):
    """
    Nesterov-accelerated gradient descent on LSE-smoothed objective.
    Key difference from plain gradient: uses momentum to build up velocity,
    helping escape shallow local minima.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    # Nesterov state
    a_curr = a.copy()
    a_prev = a.copy()
    momentum = momentum_init
    lr = lr_init

    for it in range(n_iters):
        # Nesterov look-ahead
        if it == 0:
            y = a_curr.copy()
        else:
            y = a_curr + momentum * (a_curr - a_prev)
            y = np.maximum(y, 0.0)
            S_y = float(y.sum())
            if S_y < 1e-12:
                y = a_curr
                S_y = float(a_curr.sum())
            else:
                y = y / S_y

        # Evaluate gradient at look-ahead point
        sc_y, grad_y = lse_score_and_grad(y, beta)

        # Check exact score at look-ahead point
        sc_exact = compute_score(y)
        if sc_exact < best_sc:
            best_sc = sc_exact
            best_a = y.copy()

        # Gradient step from current (not y) for stability
        sc_curr, grad_curr = lse_score_and_grad(a_curr, beta)

        # Adaptive step: use gap between smoothed and exact
        c_curr = fast_conv(a_curr)
        S_curr = float(a_curr.sum())
        exact_sc_curr = 2.0 * n * float(np.max(c_curr)) / (S_curr * S_curr)
        gap = max(sc_curr - exact_sc_curr, 0.0)
        # Scale learning rate inversely with gradient norm
        gn = float(np.linalg.norm(grad_curr))
        if gn > 1e-15:
            adaptive_lr = lr * min(1.0, 0.5 / (gn + 1e-10))
        else:
            adaptive_lr = lr

        # Take step
        a_new = a_curr - adaptive_lr * grad_curr
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            a_new = a_curr
            S_new = S_curr
        else:
            a_new = a_new / S_new

        # Update Nesterov state
        a_prev = a_curr.copy()
        a_curr = a_new

        # Schedule: slowly increase momentum, decrease lr
        momentum = min(momentum + 0.0001, 0.95)
        if it % 500 == 499:
            lr *= 0.85

    # Final check
    sc_final = compute_score(a_curr)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a_curr.copy()

    return best_a


def active_set_equalize(a, n_top=20, n_iters=2000, temp=50.0):
    """
    Active-set method: identify top peaks, compute weighted subgradient,
    take steps to reduce them. Uses softmax weighting over peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = 0.02

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Get top peaks with softmax weighting
        top_idx = np.argsort(c)[::-1][:n_top]
        top_vals = c[top_idx]
        # Softmax weights (hotter = more focused on the very top)
        w_scaled = temp * (top_vals - max(top_vals))
        weights = np.exp(w_scaled)
        weights = weights / weights.sum()

        # Compute weighted subgradient
        # sg[j] = sum_k w_k * dc_k/da_j
        # dc_k/da_j = 2*a[k-j] (if j != k-j and 0<=k-j<n), a[j] (if k=2j)
        sg = np.zeros(n)
        for w_k, k in zip(weights, top_idx):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue
            js = np.arange(j_start, j_end + 1)
            is_ = k - js
            vals = 2.0 * a[is_]
            diag_mask = (js == is_)
            vals[diag_mask] = a[is_[diag_mask]]
            sg[js] += w_k * vals

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step
        effective_step = min(step, 0.5 / (gn + 1e-10))
        a_new = a - effective_step * grad
        a_new = np.clip(a_new, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        step *= 0.9995

    return best_a

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Active-set peak equalization (15% of time)
    # ============================================================
    phase3_end = t_start + 0.70 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Active Set (deadline t={phase3_end - t_start:.0f}s) ===")

    for n_top in [10, 30, 60]:
        if time.monotonic() > phase3_end - 3:
            break
        print(f"  Active-set top={n_top}")
        a_as = active_set_equalize(best_a, n_top=n_top, n_iters=2000, temp=50.0)
        sc = compute_score(a_as)
        update_best(sc, a_as, f"ACTSET top={n_top}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: L-BFGS with increasing beta (20% of time)
    # ============================================================
    phase4_end = t_start + 0.90 * (total_budget - 30.0)
    print(f"\n=== Phase 4: L-BFGS (deadline t={phase4_end - t_start:.0f}s) ===")

    for beta in [500, 1500, 4000, 10000]:
        if time.monotonic() > phase4_end - 5:
            break
        print(f"  L-BFGS beta={beta}")
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=8000, maxfun=150000)
        update_best(sc, a_opt, f"LBFGS beta={beta}")

    # Try at parent length if different
    if parent is not None and len(parent) != len(best_a):
        if time.monotonic() < phase4_end - 5:
            for beta in [1000, 5000]:
                if time.monotonic() > phase4_end - 3:
                    break
                sc, a_opt = run_lbfgs(parent, beta, maxiter=5000, maxfun=100000)
                update_best(sc, a_opt, f"LBFGS-parent beta={beta}")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: Perturbation restarts + final polish (10% of time)
    # ============================================================
    print(f"\n=== Phase 5: Perturbation + Final Polish ===")

    # Perturb and re-optimize
    if time.monotonic() < deadline - 30:
        for sigma in [0.01, 0.03, 0.05]:
            if time.monotonic() > deadline - 15:
                break
            a_pert = best_a * (1.0 + sigma * rng.standard_normal(len(best_a)))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            S_p = a_pert.sum()
            if S_p < 1e-12:
                continue
            a_pert = a_pert / S_p

            # Quick Fienup on perturbed
            a_pf = fienup_hybrid_io_er(a_pert, n_iters=80, relax=0.3)
            sc = compute_score(a_pf)
            update_best(sc, a_pf, f"PERT sigma={sigma}")

            # Quick L-BFGS on perturbed
            if time.monotonic() < deadline - 10:
                sc2, a_opt = run_lbfgs(a_pf, 2000, maxiter=3000, maxfun=50000)
                update_best(sc2, a_opt, f"PERT+LBFGS sigma={sigma}")

    # Final Fienup polish
    if time.monotonic() < deadline - 15:
        a_final = fienup_hybrid_io_er(best_a, n_iters=100, relax=0.2)
        sc = compute_score(a_final)
        update_best(sc, a_final, "FINAL-FIENUP")

    # Final L-BFGS
    if time.monotonic() < deadline - 8:
        sc, a_opt = run_lbfgs(best_a, 8000, maxiter=3000, maxfun=40000)
        update_best(sc, a_opt, "FINAL-LBFGS")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
