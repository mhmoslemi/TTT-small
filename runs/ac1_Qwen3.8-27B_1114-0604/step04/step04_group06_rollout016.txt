Minimize 2*n*max(self-convolution)/(sum)^2 using LP-based iterative improvement.
The key method: given current sequence f with peak M, solve an LP to find the best
direction g (maximizing sum(g) subject to (f*g)[k] <= M), then mix f and g.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_smoothed_score(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)

    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

            prob.solve(solver=cp.CBC, verbose=verbose)
        except:
            return None
    
    if g.value is None:
        return None
    return np.clip(g.value, 0, None)

def lp_improve(f, M, n_inner=10, verbose=False):
    """Solve LP with cutting planes."""
    n = len(f)
    f_conv_f = fast_conv(f)  # f*f for reference
    
    # Start with active constraints
    active_k = set(np.where(f_conv_f >= 0.99 * M)[0].tolist())
    # Add a few random ones
    for _ in range(20):
        active_k.add(int(np.random.randint(0, 2*n-1)))
    
    g = None
    for inner in range(n_inner):
        # Build and solve LP
        n = len(f)
        var = cp.Variable(n, non_negative=True)
        objective = cp.Maximize(cp.sum(var))
        constraints = []
        for k in sorted(active_k):
            i_start = max(0, k - (n-1))
            i_end = min(k, n-1)
            idx = np.arange(i_start, i_end+1)
            g_idx = k - idx
            constraints.append(cp.sum(f[idx] * var[g_idx]) <= M)
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.GLOP)
        except:
            prob.solve(solver=cp.CBC)
        
        if var.value is None:
            break
        g = np.clip(var.value, 0, None)
        
        # Check for violated constraints
        fg = np.convolve(f, g)  # (f*g)
        violated = np.where(fg > M * (1 + 1e-6))[0]
        
        if len(violated) == 0:
            break
        
        # Add the most violated constraints
        worst = violated[np.argsort(fg[violated])[::-1][:10]]
        active_k.update(worst.tolist())
    
    return g

def mix_and_check(f, g, t=1.0):
    """Mix f and g, normalize, check score."""
    n = len(f)
    mixed = (1-t) * f + t * g
    S = mixed.sum()
    if S < 1e-12:
        return None, np.inf
    mixed_norm = mixed / S * f.sum()  # normalize to same sum as f
    sc = compute_score(mixed_norm)
    return mixed_norm, sc

def lse_adam(a, lr=0.01, n_steps=50000, beta=100, time_limit=None, t_start=None):
    """Adam with LSE-smoothed gradient."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0/n
        S = 1.0
    a = a / S
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for t in range(1, n_steps+1):
        if time_limit is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit:
                break
        
        lr_t = lr / (1 + 1e-5*t)
        score, grad = lse_smoothed_score(a, beta)
        m = b1*m + (1-b1)*grad
        v = b2*v + (1-b2)*grad*grad
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0, None)
        S = a.sum()
        if S > 1e-12:
            a = a / S
        
        if t % 10000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_score, best_a

Actually, let me think about this problem from the perspective of **signal processing**. We want a sequence a >= 0 such that its aperiodic autocorrelation (well, self-convolution) has a low peak-to-mean ratio.

The peak-to-mean ratio of the self-convolution is: max_k (a*a)[k] / (sum a)^2 / (2n-1)

For a "random" sequence (entries drawn from an exponential distribution, normalized), the self-convolution values are approximately chi-squared distributed, and the peak is about O(log n) times the mean. So the score would be about 2 * O(log n), which is much worse than 2.

For a "well-designed" sequence, we can get the peak close to the mean, giving a score close to 2 * 1 = 2 (since mean * (2n-1) ≈ (sum a)^2, so peak/mean ≈ 1 gives score ≈ 2).

But we're achieving 1.5, which means the peak is only 0.75 times the "uniform" value. This requires a very specific structure.

I wonder if the optimal sequence has some kind of **Fourier structure**. If a has a DFT that is "flat" (all magnitudes equal), then the circular autocorrelation is a delta function. But we care about the aperiodic (linear) autocorrelation/convolution, not the circular one.

For the linear convolution, if we think in terms of the zero-padded DFT:
Let A = DFT(a padded to length N >= 2n-1). Then the linear convolution c = a*a has DFT C = A^2.
So c_k = IDFT(A^2)_k.

To minimize max c_k, we want |A| to be as "flat" as possible (since C = A^2, and if A is flat, C is a delta function... but that's for the circular case).

Actually, for the linear convolution, the relationship is:
c = IDFT_N(A * A) truncated to 2n-1 elements.

If A is flat (|A_k| = constant for all k), then A*A has all entries of magnitude constant^2, and IDFT of a constant is a delta function. So c would be concentrated at k=0, with c_0 = constant^2 * N and c_k = 0 for k > 0. But this is the circular convolution! For the linear convolution, we get the "unwrapped" version.

Actually, I realize I should also think about the **scaling** of the problem. The score is `2n * max(conv) / sum^2`. This is scale-invariant (if we multiply a by c, both max(conv) and sum^2 scale by c^2). So we can normalize a to have sum = 1 without loss of generality.

max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.

Since f has sum 1, M = max(f*f) is on the order of 1/n (for a "flat" sequence). The LP variables g are on the order of 1/n as well.

"""

1. LP-based iterative improvement (cutting-plane method)
2. LSE-smoothed Adam optimization
3. Random restarts with different initializations and lengths

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n-1:2*n-1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def solve_lp_cutting_plane(f, M, max_inner=20, slack=1e-8):
    """Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M using cutting planes."""
    n = len(f)
    rng = np.random.default_rng(42)
    
    # Initial active set: constraints that are "tight" for f*f
    f_conv = fast_conv(f)
    active_keys = set(np.where(f_conv >= 0.95 * M)[0].tolist())
    # Add some random constraints
    for _ in range(min(30, 2*n-1)):
        active_keys.add(int(rng.integers(0, 2*n-1)))
    
    g = None
    for inner in range(max_inner):
        keys = sorted(active_keys)
        
        # Build constraint matrix
        A_ub = np.zeros((len(keys), n))
        b_ub = np.full(len(keys), M * (1 + slack))
        
        for i, k in enumerate(keys):
            i_start = max(0, k - (n-1))
            i_end = min(k, n-1)
            if i_start > i_end:
                continue
            idx = np.arange(i_start, i_end + 1)
            g_idx = k - idx
            A_ub[i, g_idx] = f[idx]
        
        # Solve LP
        c_obj = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        
        if not res.success:
            break
        g = res.x
        
        # Check for violated constraints
        fg = np.convolve(f, g)
        violated = np.where(fg > M * (1 + 1e-6))[0]
        
        if len(violated) == 0:
            break
        
        # Add most violated constraints
        worst_idx = violated[np.argsort(fg[violated])[::-1][:5]]
        active_keys.update(worst_idx.tolist())
    
    return g
