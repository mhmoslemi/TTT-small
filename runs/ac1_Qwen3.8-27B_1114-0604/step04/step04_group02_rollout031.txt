We want to minimize f(a) = max_k (a*a)[k] subject to sum(a) = 1, a >= 0.

At a point a with peak at k*:

- So the subgradient vector g has g[j] = 2*a[k*-j] (shifted/reversed).

The Polyak step size (for minimizing f over the convex set {sum(a)=1, a>=0}):
step = (f(a) - f*) / ||g||^2

where f* is a lower bound on the optimum. We use f* = 1/(2n-1) (the trivial lower bound).

The update: a_new = a - step * g, then project onto the feasible set.

But wait: we're on the simplex (sum=1, a>=0). The subgradient method on a constrained problem uses the projection. The projection onto the simplex is:
P_simplex(a) = argmin_{x: sum(x)=1, x>=0} ||x - a||^2

This is computed by:
1. Sort a: a_(1) >= ... >= a_(n)
2. Find m = max{j: a_(j) - (a_(1)+...+a_(j) - 1)/j > 0}  [equivalently, the largest j such that a_(j) > lambda where lambda = (sum_{i<=j} a_(i) - 1)/j]
3. lambda = (sum_{i=1}^m a_(i) - 1) / m
4. P(a)_i = max(a_i - lambda, 0)

This is O(n log n) per step. For n=8000, that's about 8000 * 13 ≈ 100,000 operations, which is fast.

"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def get_subgradient(a, k_star):
    """Compute subgradient of (a*a)[k_star] w.r.t. a. Result is 2*a[k_star - j] for valid j."""
    n = len(a)
    g = np.zeros(n)
    j_min = max(0, k_star - n + 1)
    j_max = min(n - 1, k_star)
    if j_max >= j_min:
        js = np.arange(j_min, j_max + 1)
        g[js] = 2.0 * a[k_star - js]
    return g


def subgrad_nesterov(a_init, n_iters, f_lb=1.0, verbose=False, max_step=0.5):
    """Nesterov-accelerated subgradient descent on the exact objective (sum-normalized)."""
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64).copy(), 0, None)
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    a = a / S

    y = a.copy()
    a_old = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        if Sy < 1e-12:
            break
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)

        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()

        if verbose and it % 10000 == 0:
            print(f"    subgrad iter {it}: score={sc:.10f}, k*={k_star}")

        g = get_subgradient(y, k_star)
        gnorm2 = float(np.dot(g, g))
        if gnorm2 < 1e-30:
            break

        # Polyak step size (using f* lower bound = 1/(2n-1) for normalized problem)
        f_star_lb = 1.0 / (2.0 * n - 1.0)
        gap = Mak - f_star_lb
        if gap < 1e-15:
            break
        step = gap / (gnorm2 + 1e-30)
        step = min(step, max_step / max(Sy, 1e-12))

        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - step * g
        a_new = y_new + theta * (y_new - a)

        a = np.clip(a_new, 0.0, None)
        Sa = a.sum()
        if Sa < 1e-12:
            a = a_old.copy()
            y = a.copy()
            continue
        a = a / Sa
        a_old = a.copy()
        y = a.copy()

    return best_a


def lse_score_and_grad(a, beta):

    # grad_L = sum_k p[k] * d(a*a)[k]/da = sum_k p[k] * 2*a_shifted(k)

    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    grad_L = 2.0 * conv_res

    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L * grad_L.sum() / S3)

    # score = 2n * L / S^2
    # d(score)/da = 2n * (dL/da * S^2 - L * 2*S * dS/da) / S^4

    # dS/da = 1 (vector of ones)
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L * np.ones(n) / S3)

def polish_lbfgs(a0, beta, maxiter=30000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64).copy(), 0.0, 1000.0)
    if a0.sum() < 1e-6:
        a0[:] = 1.0
    bounds = [(0.0, 1000.0)] * n

    def func(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        func, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_coordinate_descent(a_init, n_sweeps=10, n_top=50):
    """
    Targeted coordinate descent: for the current peak, reduce top contributors
    and redistribute to bottom contributors.
    """
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64).copy(), 0, None)
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_min = max(0, k_star - n + 1)
        j_max = min(n - 1, k_star)
        if j_max < j_min:
            break

        js = np.arange(j_min, j_max + 1)
        partners = k_star - js
        products = a[js] * a[partners]

        top_count = min(n_top, len(js) // 2)
        bottom_count = max(1, len(js) - top_count)

        sorted_idx = np.argsort(products)
        top_local = sorted_idx[::-1][:top_count]
        bottom_local = sorted_idx[:bottom_count]

        for li in top_local:
            j1 = js[li]
            j2 = partners[li]
            a_j1 = a[j1]
            if a_j1 < 1e-12:
                continue
            # Try reducing a[j1] and distributing to bottom
            for frac in [0.1, 0.3, 0.5, 0.7]:
                delta = frac * a_j1
                a[j1] = a_j1 - delta
                # Distribute to bottom elements
                per = delta / len(bottom_local)
                old_a = a.copy()
                for bl in bottom_local:
                    jb = js[bl]
                    a[jb] += per
                a = np.clip(a, 0, None)
                S_new = a.sum()
                if S_new < 1e-12:
                    a = old_a.copy()
                    continue
                a = a / S_new
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                else:
                    a = old_a.copy()
                    S = a.sum()
                    a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def generate_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.linspace(0, 1, n)

    if parent is not None and len(parent) >= 16:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            inits.append(np.interp(x, x_old, parent).copy())

    # Constant
    inits.append(np.ones(n) * 0.5)

    # U-shaped
    inits.append(0.2 + 1.8 * (2.0 * x - 1.0) ** 2)

    # Inverted U (bell)
    inits.append(np.maximum(0.1, 1.0 - 2.0 * (2.0 * x - 1.0) ** 2))

    # Cosine window
    inits.append(np.maximum(0.05, 0.5 * (1.0 + np.cos(2.0 * np.pi * x))))

    # Random non-negative
    r = np.abs(rng.standard_normal(n))
    inits.append(r / r.max() * 0.5 + 0.1)

    # Multi-scale random
    r1 = 0.3 * np.abs(rng.standard_normal(n))
    smooth = np.convolve(r1, np.ones(max(1, n // 50)) / max(1, n // 50), mode='same')
    inits.append(0.1 + smooth)

    # Chirp-like
    inits.append(0.3 + 0.7 * np.abs(np.sin(np.pi * x * (3 + 5 * rng.random()))))

    for inp in inits:
        inp = np.clip(inp, 0, 1000)
        S = inp.sum()
        if S > 1e-12:
            inp = inp / S

    return inits

            best_score = sc

            elapsed = time.monotonic() - t_start
            print(f"  ** GLOBAL BEST: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s, {label})")
            return True
        return False

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0, 1000)
                parent = parent / parent.sum()
    except Exception:
        pass

    if parent is not None:
        sc_parent = compute_score(parent)
        print(f"Parent: n={len(parent)}, score={sc_parent:.10f}")
        global_update(sc_parent, parent, "parent")

    # Determine target lengths to try
    if parent is not None:
        n_parent = len(parent)
        # Try parent length and nearby powers of 2
        target_lengths = [n_parent]
        for n_try in [4096, 8192, 16384, 32768]:
            if n_try != n_parent and 256 <= n_try <= 50000:
                target_lengths.append(n_try)
        # Sort by proximity to parent length (optimize parent length first)
        target_lengths.sort(key=lambda n: abs(n - n_parent))
    else:
        target_lengths = [8192, 4096, 16384]

    # Time allocation: give more time to the first (parent) length
    # Reserve 15% of time for the last length exploration
    n_lengths = len(target_lengths)
    if n_lengths == 1:
        budgets = [float(budget_s) - 25.0]
    else:
        # 70% to first, 15% to each remaining (if >2 lengths)
        remaining = (float(budget_s) - 25.0) * 0.25 // max(1, n_lengths - 1)
        first_budget = (float(budget_s) - 25.0) - remaining * (n_lengths - 1)
        budgets = [first_budget] + [remaining] * (n_lengths - 1)

    print(f"\nTarget lengths: {target_lengths}")
    print(f"Budgets: {[f'{b:.0f}s' for b in budgets]}")

    for idx, n_target in enumerate(target_lengths):
        if time.monotonic() > deadline:
            break
        budget = min(budgets[idx], deadline - time.monotonic())
        if budget < 30:
            break
        print(f"\n{'='*60}")
        print(f"Optimizing at n={n_target} (budget={budget:.0f}s)")
        print(f"{'='*60}")

        # Upscale parent to target length
        parent_for_len = None
        if parent is not None:
            if len(parent) == n_target:
                parent_for_len = parent
            else:
                parent_for_len = np.interp(np.linspace(0, 1, n_target),
                                           np.linspace(0, 1, len(parent)), parent)
                parent_for_len = np.clip(parent_for_len, 0, None)
                S = parent_for_len.sum()
                if S > 0:
                    parent_for_len = parent_for_len / S

        sc_len, a_len = optimize_at_length(n_target, parent_for_len, rng, budget)
        global_update(sc_len, a_len, f"n={n_target}")

    # Final phase: ultra-fine polishing on the best found
    print(f"\n{'='*60}")
    print(f"FINAL POLISH (remaining={deadline - time.monotonic():.0f}s)")
    print(f"{'='*60}")

    n_final = len(best_a)
    remaining = deadline - time.monotonic()

    if remaining > 30:
        # High-beta L-BFGS
        for beta in [100000, 200000, 500000]:
            if time.monotonic() > deadline - 10:
                break
            rem = deadline - time.monotonic()
            if rem < 10:
                break
            maxiter = int(min(50000, max(5000, rem * 25)))
            maxfun = int(min(200000, max(10000, rem * 80)))
            sc, a_opt = polish_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
            global_update(sc, a_opt, f"final_b{beta}")

    # One more coordinate descent pass
    if time.monotonic() < deadline - 10:
        a_cd = peak_coordinate_descent(best_a, n_sweeps=30, n_top=100)
        sc = compute_score(a_cd)
        global_update(sc, a_cd, "final_cd")

    # Verify with evaluate_sequence
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
