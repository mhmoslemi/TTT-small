import numpy as np
from scipy.optimize import linprog, minimize
import time

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def solve_lp_constraint_gen(a, M, max_constraints=200, time_limit=30):
    """
    Solve: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0.
    Uses constraint generation.
    Returns b* and the set of active constraints.
    """
    n = len(a)
    t_start = time.monotonic()
    
    # Start with the most "active" constraint (k = argmax of a*a)
    c = fast_conv(a)
    k_most_active = int(c.argmax())
    
    active_k = [k_most_active]
    b = None
    
    for outer_iter in range(max_constraints):
        if time.monotonic() - t_start > time_limit:
            break
        
        # Build constraint matrix for active constraints
        A_ub = np.zeros((len(active_k), n))
        for idx, k in enumerate(active_k):
            # (a*b)[k] = sum_i a[i] * b[k-i]
            # For b[j] to appear in (a*b)[k], we need j = k-i, i.e., i = k-j
            # So (a*b)[k] = sum_{j=0}^{n-1} a[k-j] * b[j] (where k-j is valid)
            js = np.arange(n)
            valid = (js >= 0) & (js < n) & (k - js >= 0) & (k - js < n)
            A_ub[idx, js[valid]] = a[k - js[valid]]
        
        # Solve LP: max sum(b) s.t. A_ub * b <= M, b >= 0
        c_obj = -np.ones(n)  # minimize -sum(b)
        res = linprog(c_obj, A_ub=A_ub, b_ub=np.full(len(active_k), M),
                      bounds=[(0, None)] * n, method='highs')
        
        if not res.success:
            break
        
        b = res.x
        b = np.clip(b, 0, None)
        
        # Check for violated constraints
        if b.sum() < 1e-10:
            break
        
        # Compute (a*b)[k] for all k
        ab = np.convolve(a, b)  # This is O(n^2), might be slow for large n
        
        # Find violated constraints
        violated = np.where(ab > M + 1e-8)[0]
        
        if len(violated) == 0:
            break
        
        # Add the most violated constraint
        worst_k = violated[np.argmax(ab[violated])]
        if worst_k not in active_k:
            active_k.append(worst_k)
    
    return b if b is not None else np.ones(n) / n

"""
Minimize 2*n*max(self-convolution)/(sum)^2 using LP-based constraint generation
and LSE-smoothed gradient polishing. Uses the zero-sum game interpretation where
player 1 chooses a and player 2 chooses b, with payoff max_k (a*b)[k].
The LP finds the best response b* to current a, guiding the update direction.
"""

import numpy as np
from scipy.optimize import linprog, minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv_cross(a, b):
    """Linear cross-convolution a * b via FFT."""
    total_len = len(a) + len(b) - 1
    N = 1 << (total_len - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    return np.fft.irfft(A * B, N)[:total_len]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def build_constraint_row(k, a, n):
    """Build row of constraint matrix for constraint k: (a*b)[k] = sum_j a[k-j]*b[j]."""
    row = np.zeros(n)
    j_min = max(0, k - n + 1)
    j_max = min(k, n - 1)
    if j_max >= j_min:
        js = np.arange(j_min, j_max + 1)
        row[js] = a[k - js]
    return row


def lp_constraint_generation(a, M, max_constraints=300, lp_time_limit=20, total_time_limit=120):
    """
    Solve: max sum(b) s.t. (a*b)[k] <= M for all k, b >= 0.
    Uses constraint generation to keep LP small.
    """
    n = len(a)
    t_start = time.monotonic()

    # Start with the most active constraint
    c = fast_conv(a)
    k_star = int(c.argmax())
    active_k = [k_star]
    # Also add a few nearby constraints
    for dk in [1, -1, 2, -2, 5, -5]:
        k2 = k_star + dk
        if 0 <= k2 < 2 * n - 1 and k2 not in active_k:
            active_k.append(k2)

    b = None
    for it in range(max_constraints):
        if time.monotonic() - t_start > total_time_limit:
            break

        # Build constraint matrix
        A_ub = np.array([build_constraint_row(k, a, n) for k in active_k])
        b_ub = np.full(len(active_k), M)
        c_obj = -np.ones(n)

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': lp_time_limit, 'presolve': True})

        if not res.success:
            # If LP fails, fall back to current a as direction
            b = a / a.sum()
            break

        b = np.clip(res.x, 0, None)
        b_sum = float(b.sum())
        if b_sum < 1e-10:
            b = a / a.sum()
            break

        # Check all constraints for violation
        ab = fast_conv_cross(a, b)
        max_ab = float(ab.max())

        if max_ab <= M * (1.0 + 1e-6) + 1e-8:
            break

        # Add most violated constraint
        k_viol = int(np.argmax(ab))
        if k_viol not in active_k:
            active_k.append(k_viol)
        elif len(active_k) < max_constraints:
            # Add a random other violated constraint
            violated = np.where(ab > M * 1.01 + 1e-8)[0]
            if len(violated) > 0:
                k_rand = violated[np.random.randint(len(violated))]
                if k_rand not in active_k:
                    active_k.append(k_rand)

    return b if b is not None else a / a.sum()


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient for score = 2n*max_c/sum^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()

    # Gradient: dL/da[i] = 2 * sum_k p[k] * a[k-i] (valid k)
    # = 2 * (cross-correlation of p and a)
    N_fft = 1 << (2 * n - 2).bit_length()
    P = np.fft.rfft(p, N_fft)
    A = np.fft.rfft(a, N_fft)
    # h[i] = sum_j p[j+i]*a[j] = IFFT(conj(P) * A)[i]
    h = np.fft.irfft(np.conj(P) * A, N_fft)[:n]

    S2 = S * S
    S3 = S2 * S
    grad = (4.0 * n * h / S2) - (8.0 * n * L * a / S3)
    return float(score), grad


def adam_lse_optimize(a_init, lr=0.01, n_steps=50000, beta_lse=200,
                      time_limit_s=None, t_start=None):
    """Adam optimizer with LSE-smoothed gradient, normalized to sum=1."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S0 = a.sum()
    if S0 < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / S0

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break

        lr_t = lr / (1.0 + 2e-5 * t)
        score, grad = lse_score_and_grad(a, beta_lse)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 1e-12:
            a = a / s

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def lbfgs_polish(a0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B polish with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S0 = a0.sum()
    if S0 < 1e-6:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / S0

    bounds = [(0.0, 10.0)] * n

    def func_and_grad(x):
        sc, g = lse_score_and_grad(x, beta)
        return sc, g

    res = minimize(func_and_grad, a0, method='L-BFGS-B', jac=True,
                   bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-10,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out > 1e-12:
        a_out = a_out / S_out
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

    for beta in [3000, 5000, 10000]:
        if time.monotonic() > t_end - 15:
            break
        sc, a_opt, _ = lbfgs_polish(best_a, beta, maxiter=2000, maxfun=30000)
        try_update(sc, a_opt, f"Final L-BFGS beta={beta}")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [1.0] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
