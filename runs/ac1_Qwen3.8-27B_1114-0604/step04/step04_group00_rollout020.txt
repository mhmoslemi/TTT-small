"""
Multi-scale warm start + Nesterov-accelerated LSE + momentum Fienup optimizer.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key innovations over previous approach:
1. Multi-scale warm start (n=64 -> 256 -> 1024 -> 2048 -> 4096 -> target)
2. Nesterov-accelerated projected gradient on LSE-smoothed objective with adaptive beta
3. Momentum-based Fienup with cosine-annealed threshold
4. Peak-tracking coordinate descent for final micro-optimization
5. Symmetry-exploiting initialization (a[i] = a[n-1-i])

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """
    Compute LSE-smoothed score and its gradient.
    Uses FFT for efficient computation.
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    shifted = beta * (c - c_max)  # all <= 0, max is 0
    exp_shifted = np.exp(shifted)
    log_sum = float(np.log(exp_shifted.sum()))
    L = c_max + log_sum / beta  # LSE approximation to max(c)
    p = exp_shifted / exp_shifted.sum()  # weighting for gradient

    # dL/da_i = 2 * sum_{m=0}^{n-1} a[m] * p[m+i]
    # Computed via FFT: d[i] = 2 * (a_pad * p_rev_pad)[2n-2-i]
    conv_len = 3 * n - 2
    M_fft = 1 << (conv_len - 1).bit_length()
    A_pad = np.zeros(M_fft)
    A_pad[:n] = a
    p_rev = np.zeros(M_fft)
    p_rev[:2*n-1] = p[::-1]
    Aft = np.fft.rfft(A_pad)
    Prft = np.fft.rfft(p_rev)
    conv_res = np.fft.irfft(Aft * Prft)[:conv_len]
    # d[i] = 2 * conv_res[2n-2-i] for i=0..n-1
    grad_L = 2.0 * conv_res[n-1:2*n-1][::-1]

    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def fienup_momentum(a, n_iters=300, momentum_init=0.0, momentum_end=0.5,
                    thresh_init=2.0, thresh_end=1.05, clip_init=0.4, clip_end=0.8):
    """
    Fienup with momentum and cosine-annealed parameters.
    Flattens the power spectrum by reducing high-amplitude bins.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    v = np.zeros(n)  # momentum buffer
    A_prev = None

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        # Cosine annealing
        momentum = momentum_init + (momentum_end - momentum_init) * 0.5 * (1 - np.cos(np.pi * t))
        thresh = thresh_init + (thresh_end - thresh_init) * t
        clip = clip_init + (clip_end - clip_init) * t

        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Identify bins above threshold
        bin_thresh = P_mean * thresh
        above = P > bin_thresh

        if above.any():
            # Reduce amplitude of high bins
            scale = np.ones_like(P)
            # Bring amplitudes down to threshold (with clip_strength)
            target_amp = np.sqrt(bin_thresh)
            current_amp = np.sqrt(P[above])
            # Interpolate: full reduction at clip=1, no reduction at clip=0
            factor = 1.0 - clip + clip * (target_amp / np.maximum(current_amp, 1e-10))
            factor = np.clip(factor, 0.01, 1.0)
            scale[above] = factor

        A_new = A * scale
        a_new = np.fft.irfft(A_new, M_fft)[:n]

        # Linear combination with momentum
        a_update = a_new - a
        v = momentum * v + a_update
        a = a + v

        # Project to non-negative and normalize
        a = np.maximum(a, 0.0)
        S_new = float(a.sum())
        if S_new < 1e-12:
            a = best_a.copy()
            break
        a = a / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def nesterov_lse(a, beta_start=20.0, beta_end=5000.0, n_iters=2000,
                 lr_start=0.02, lr_end=0.002, momentum_init=0.9, momentum_end=0.95):
    """
    Nesterov-accelerated projected gradient on LSE-smoothed objective.
    Adaptive beta schedule: exponential increase from beta_start to beta_end.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)  # Nesterov velocity

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        # Exponential beta schedule
        beta = beta_start * (beta_end / beta_start) ** t
        # Linear LR schedule
        lr = lr_start + (lr_end - lr_start) * t
        # Momentum schedule
        mu = momentum_init + (momentum_end - momentum_init) * t

        # Nesterov lookahead
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = a_look.sum()
        if S_look < 1e-12:
            break
        a_look = a_look / S_look

        # Compute gradient at lookahead point
        _, grad = lse_score_and_grad(a_look, beta)

        # Update velocity
        v = mu * v - lr * grad

        # Take step
        a = a + v

        # Project to non-negative and normalize
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        # Track best (using exact score periodically)
        if it % 50 == 0 or it == n_iters - 1:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a


def nesterov_lse_adaptive(a, max_time=100.0, beta_start=10.0, beta_end=10000.0):
    """
    Nesterov LSE with time-adaptive beta and learning rate.
    Runs until max_time elapsed.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t_start = time.monotonic()

    it = 0
    beta = beta_start
    beta_growth = 1.005  # multiply by this each iteration
    lr = 0.015
    lr_decay = 0.999
    mu = 0.9

    while time.monotonic() - t_start < max_time:
        it += 1
        mu = min(0.96, 0.9 + 0.00005 * it)
        lr *= lr_decay

        # Nesterov lookahead
        a_look = a + mu * v
        a_look = np.maximum(a_look, 0.0)
        S_look = a_look.sum()
        if S_look < 1e-12:
            break
        a_look = a_look / S_look

        # Compute gradient
        _, grad = lse_score_and_grad(a_look, beta)

        # Gradient norm for adaptive step
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Adaptive step: limit the step size relative to function value
        # The score is at least 2n/(2n-1), so (score - lower_bound) gives room
        sc_actual = compute_score(a) if it % 20 == 0 else best_sc
        lb = 2.0 * n / (2.0 * n - 1.0)
        max_step = max(0.01, (sc_actual - lb) * 0.5)
        effective_lr = min(lr, max_step / (gnorm + 1e-10))

        # Update
        v = mu * v - effective_lr * grad
        a = a + v
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        # Increase beta
        beta = min(beta * beta_growth, beta_end)

        # Track best
        if it % 100 == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc

    return best_a


def fienup_simple(a, n_iters=200, thresh=1.3, clip=0.5):
    """Simple Fienup without momentum (for comparison/backup)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        above = P > P_mean * thresh
        if not above.any():
            break

        scale = np.ones_like(P)
        scale[above] = ((P_mean * thresh) / P[above]) ** (clip * 0.5)

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def objective(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        objective, a0, args=(),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_coordinate_descent(a, n_sweeps=5, n_top=40):
    """
    Coordinate descent targeting the top peak contributors.
    For each peak, perturb the coordinates that contribute most to it.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        # Get top peaks
        top_k = min(n_top, len(c))
        peak_indices = np.argsort(c)[::-1][:top_k]

        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end < j_start:
                continue

            # Contribution of each a[j] to c[k] is a[k-j]
            contrib = np.array([a[k - j] for j in range(j_start, j_end + 1)])
            # Find top contributors
            n_top_local = min(20, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_top_local]
            top_global = j_start + top_local

            for j in top_global:
                j = int(j)
                old_val = a[j]
                if old_val < 1e-10:
                    continue

                # Try scaling the value
                for factor in [0.5, 0.7, 0.85, 1.15, 1.3]:
                    a[j] = old_val * factor
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j] = old_val
                        continue
                    a_scaled = a / S_tmp
                    sc = compute_score(a_scaled)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a_scaled.copy()
                        a = a_scaled
                        break
                else:
                    a[j] = old_val
                    continue
                # If we improved, the value is already updated
                # Re-normalize
                S = a.sum()
                if S > 1e-12:
                    a = a / S

    return best_a


def upsample(a, new_n):
    """Upsample a sequence to a new length using linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def build_symmetric_init(n, rng):
    """Build a symmetric initialization."""
    half = n // 2
    x = np.linspace(0, 1, half + 1)
    # Raised cosine window
    a_half = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a_half = np.maximum(a_half, 0.01)
    # Build full symmetric sequence
    if n % 2 == 0:
        a = np.concatenate([a_half[-1:][::-1], a_half[1:]])
    else:
        a = np.concatenate([a_half[-1:][::-1], a_half])
    a = a[:n]
    return np.maximum(a, 0.01)


def build_chirp_init(n, rng):
    """Build a chirp-like initialization (non-negative version)."""
    x = np.arange(n, dtype=np.float64) / n
    # Absolute value of a chirp, shifted to be non-negative
    a = np.abs(np.sin(2 * np.pi * x * x * 2.0)) + 0.1
    return a / a.sum()


def build_gaussian_init(n, rng):
    """Gaussian-shaped initialization."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = n / 4.0
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    return a / a.sum()


def build_mixed_init(n, rng):
    """Mixed initialization combining several shapes."""
    x = np.linspace(0, 1, n)
    a1 = 0.5 + 0.3 * np.cos(2 * np.pi * x) + 0.2 * np.cos(4 * np.pi * x)
    a1 = np.maximum(a1, 0.01)
    return a1 / a1.sum()


def multistage_optimize_at_scale(a, time_budget):
    """Run a sequence of optimizers on a single scale."""
    n = len(a)
    best_a = a.copy()
    best_sc = compute_score(a)

    def try_update(a_new, label):
        nonlocal best_a, best_sc
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            print(f"    {label}: {sc:.8f}")
        return best_a

    # Fienup with momentum
    if time_budget > 10:
        a_f = fienup_momentum(best_a, n_iters=150,
                              momentum_init=0.0, momentum_end=0.3,
                              thresh_init=2.0, thresh_end=1.1,
                              clip_init=0.3, clip_end=0.7)
        try_update(a_f, "fienup-mom")

    # Nesterov LSE (short)
    if time_budget > 20:
        a_nl = nesterov_lse(best_a, beta_start=20.0, beta_end=1000.0,
                            n_iters=500, lr_start=0.01, lr_end=0.003,
                            momentum_init=0.9, momentum_end=0.94)
        try_update(a_nl, "nesterov-short")

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Try to load parent
    parent = None
    try:
        p = globals().get('height_sequence_1', None)
        if p is not None:
            p = np.asarray(p, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # ============================================================
    # Phase 1: Multi-scale warm start (20% of time)
    # ============================================================
    phase1_end = t_start + 0.20 * (total_budget - 30.0)
    print(f"=== Phase 1: Multi-scale warm start (t < {phase1_end - t_start:.0f}s) ===")

    # Build up from small to large
    n_target = len(parent) if parent is not None else 6962
    scales = [64, 128, 256, 512, 1024, 2048, 4096]
    scales = [s for s in scales if s < n_target] + [n_target]
    # Ensure no duplicates
    scales = list(dict.fromkeys(scales))

    current_a = None
    for n in scales:
        if time.monotonic() > phase1_end:
            break
        t_scale_start = time.monotonic()
        scale_budget = phase1_end - t_scale_start
        if scale_budget < 5:
            break

        print(f"  Scale n={n} (budget {scale_budget:.0f}s)")

        # Initialize
        if current_a is None:
            a = build_symmetric_init(n, rng)
        else:
            if len(current_a) == n:
                a = current_a.copy()
            else:
                a = upsample(current_a, n)

        # Optimize at this scale
        time_left = phase1_end - time.monotonic()
        if time_left > 10:
            a = multistage_optimize_at_scale(a, time_left)

        current_a = a
        sc = compute_score(a)
        print(f"    Score at n={n}: {sc:.8f}")

    if current_a is not None:
        # Upsample to target if needed
        if len(current_a) != n_target:
            current_a = upsample(current_a, n_target)
        update_best(compute_score(current_a), current_a, "multi-scale")

    # Also try from parent at full scale
    if parent is not None:
        a_parent = parent.copy()
        sc_p = compute_score(a_parent)
        update_best(sc_p, a_parent, "parent")
        # Quick Fienup on parent
        if time.monotonic() < phase1_end - 5:
            a_pf = fienup_momentum(a_parent, n_iters=100,
                                    momentum_init=0.0, momentum_end=0.3,
                                    thresh_init=1.8, thresh_end=1.1,
                                    clip_init=0.3, clip_end=0.7)
            update_best(compute_score(a_pf), a_pf, "parent+fienup")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Nesterov LSE with adaptive parameters (35% of time)
    # ============================================================
    phase2_end = t_start + 0.55 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Nesterov LSE (t < {phase2_end - t_start:.0f}s) ===")

    # Run from best with different parameter settings
    configs = [
        {'beta_start': 10.0, 'beta_end': 5000.0},
        {'beta_start': 30.0, 'beta_end': 10000.0},
        {'beta_start': 5.0, 'beta_end': 3000.0},
    ]

    for cfg in configs:
        if time.monotonic() > phase2_end:
            break
        time_left = phase2_end - time.monotonic()
        if time_left < 30:
            break
        print(f"  Nesterov LSE: beta [{cfg['beta_start']}, {cfg['beta_end']}], time_budget={time_left:.0f}s")
        a_nl = nesterov_lse_adaptive(best_a, max_time=time_left * 0.9,
                                      beta_start=cfg['beta_start'],
                                      beta_end=cfg['beta_end'])
        update_best(compute_score(a_nl), a_nl, "nesterov-adapt")

    # Also try from a perturbed version
    if time.monotonic() < phase2_end - 30:
        a_pert = best_a * (1.0 + 0.02 * rng.standard_normal(len(best_a)))
        a_pert = np.clip(a_pert, 0.0, None)
        a_pert = a_pert / a_pert.sum()
        time_left = phase2_end - time.monotonic()
        a_nl2 = nesterov_lse_adaptive(a_pert, max_time=time_left * 0.8,
                                       beta_start=20.0, beta_end=5000.0)
        update_best(compute_score(a_nl2), a_nl2, "nesterov-pert")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Fienup refinement (15% of time)
    # ============================================================
    phase3_end = t_start + 0.70 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Fienup refinement (t < {phase3_end - t_start:.0f}s) ===")

    fienup_configs = [
        {'n_iters': 300, 'momentum_init': 0.0, 'momentum_end': 0.4,
         'thresh_init': 2.0, 'thresh_end': 1.05, 'clip_init': 0.3, 'clip_end': 0.9},
        {'n_iters': 400, 'momentum_init': 0.1, 'momentum_end': 0.5,
         'thresh_init': 1.8, 'thresh_end': 1.02, 'clip_init': 0.5, 'clip_end': 0.95},
        {'n_iters': 200, 'momentum_init': 0.0, 'momentum_end': 0.2,
         'thresh_init': 2.5, 'thresh_end': 1.2, 'clip_init': 0.2, 'clip_end': 0.6},
    ]

    for i, cfg in enumerate(fienup_configs):
        if time.monotonic() > phase3_end:
            break
        a_f = fienup_momentum(best_a, **cfg)
        update_best(compute_score(a_f), a_f, f"fienup-{i}")

    # Simple Fienup as backup
    if time.monotonic() < phase3_end - 10:
        a_sf = fienup_simple(best_a, n_iters=300, thresh=1.2, clip=0.6)
        update_best(compute_score(a_sf), a_sf, "fienup-simple")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: L-BFGS polish (20% of time)
    # ============================================================
    phase4_end = t_start + 0.90 * (total_budget - 30.0)
    print(f"\n=== Phase 4: L-BFGS polish (t < {phase4_end - t_start:.0f}s) ===")

    lbfgs_betas = [200, 500, 1000, 2000, 5000, 10000]
    for beta in lbfgs_betas:
        if time.monotonic() > phase4_end:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=3000, maxfun=50000)
        update_best(sc, a_opt, f"lbfgs-beta={beta}")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: Peak coordinate descent (5% of time)
    # ============================================================
    phase5_end = t_start + 0.95 * (total_budget - 30.0)
    print(f"\n=== Phase 5: Peak coordinate descent (t < {phase5_end - t_start:.0f}s) ===")

    if time.monotonic() < phase5_end - 10:
        a_cd = peak_coordinate_descent(best_a, n_sweeps=4, n_top=30)
        update_best(compute_score(a_cd), a_cd, "peak-cd")

        # Quick LSE after CD
        if time.monotonic() < phase5_end - 5:
            a_cd2 = nesterov_lse(a_cd, beta_start=500.0, beta_end=5000.0,
                                  n_iters=500, lr_start=0.005, lr_end=0.001,
                                  momentum_init=0.9, momentum_end=0.95)
            update_best(compute_score(a_cd2), a_cd2, "cd+nesterov")

    # ============================================================
    # Final: one more L-BFGS with highest beta
    # ============================================================
    if time.monotonic() < deadline - 5:
        sc, a_opt = run_lbfgs(best_a, 15000, maxiter=2000, maxfun=30000)
        update_best(sc, a_opt, "final-lbfgs-hibeta")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
