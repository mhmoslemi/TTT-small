"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS on LSE-smoothed objective, top-M subgradient, coordinate descent,
and multi-start exploration with adaptive scheduling.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_obj_and_grad(x, beta):
    """LSE-smoothed objective and gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a: sum_k p[k] * d(a*a)[k]/da
    # d(a*a)[k]/da[j] = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
    # = (a + flip(a)) convolved with delta, restricted
    # More precisely: d(a*a)/da is a "convolution template"
    # For LSE: grad_L[j] = sum_k p[k] * (a[(k-j) mod ...]) 
    # Actually: L = log-sum-exp(beta*(c - c_max))/beta + c_max
    # dL/da[j] = sum_k p[k] * dc[k]/da[j]
    # dc[k]/da[j] = a[k-j] if 0 <= k-j < n, else 0
    # So dL/da[j] = sum_{k: 0<=k-j<n} p[k] * a[k-j]
    # = sum_{m=0}^{n-1} p[j+m] * a[m] (where m = k-j, k = j+m)
    # This is a "cross-correlation" of p and a
    # More precisely: dL/da[j] = sum_{m=max(0, -(n-1-j))}^{min(n-1-j, n-1)} ... 
    # Let me use FFT: dL/da = ifft(fft(p) * fft(rev_a)) ... 
    # Actually, dc[k]/da[j] = a[k-j] for 0 <= k-j <= n-1
    # dL/da[j] = sum_k p[k] * a[k-j] * 1[0 <= k-j <= n-1]
    # Let m = k-j: dL/da[j] = sum_m p[j+m] * a[m] * 1[0 <= m <= n-1] * 1[0 <= j+m <= 2n-2]
    # = sum_{m=0}^{n-1} p[j+m] * a[m] (where p is zero-extended)
    # This is the cross-correlation of p and a: (p ⊛ a)[j] where ⊛ denotes cross-correlation
    # = conv(flip(p), a)[j + (2n-2) - (n-1)] ... hmm this is getting complicated
    # Let me just use the fact that:
    # dL/da = IFFT(FFT(p) * conj(FFT(a_shifted))) ... 
    # Actually the simplest way: use the fact that d(a*a)/da = 2 * (a * r) where r[k] = a[-k] (reversed)
    # No wait, for the self-convolution: d(a*a)[k]/da[j] = a[k-j] + a[j-k] (symmetric)
    # So dL/da[j] = sum_k p[k] * (a[k-j] + a[j-k]) * 1[valid indices]
    # = 2 * sum_k p[k] * a[k-j] * 1[0 <= k-j < n] (by symmetry, since p is symmetric)
    # Wait, p is NOT necessarily symmetric. p[k] is the softmax over all 2n-1 values of c.
    # But c[k] = c[2n-2-k] (self-convolution is symmetric), so if beta is the same, p[k] = p[2n-2-k].
    # So p IS symmetric!
    # Then dL/da[j] = sum_k p[k] * a[k-j] * 1[0 <= k-j < n]
    # Let me compute this via FFT:
    # dL/da = cross-correlation of p and a (with appropriate handling)
    # Using FFT: ifft(fft(p) * conj(fft(rev_a))) ... 
    # Actually, let me just compute it directly for clarity:
    # dL/da[j] = sum_{k=max(j,0)}^{min(j+n-1, 2n-2)} p[k] * a[k-j]
    # = sum_{m=0}^{n-1} p[j+m] * a[m] (with p zero-extended to length 2n-1)
    # This is a sliding dot product, which can be computed as a convolution:
    # dL/da = conv(a, p_reversed)[n-1 : 2n-1] where p is padded
    
    # Let me use a clean FFT approach:
    # dL/da[j] = sum_{m=0}^{n-1} a[m] * p[j+m] for j = 0, ..., n-1
    # (with p defined for indices 0 to 2n-2, zero elsewhere)
    # This is: (a convolved with flip(p))[j + n - 1] ... 
    # Let's use: result = ifft(fft(a, N) * fft(flip(p), N)) where N = 3n-2
    # Then result[n-1 : 2n-1] = dL/da
    
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    A_fft = np.fft.rfft(a, M2_fft)
    P_rev_fft = np.fft.rfft(p[::-1], M2_fft)
    dL_da = np.fft.irfft(A_fft * P_rev_fft, M2_fft)[n-1:2*n-1]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 0.5
        S = a0.sum()
    a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def top_m_subgradient(a, n_iters=3000, top_m=10, step_init=0.05, decay=0.9995):
    """Subgradient method using top-M peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Top-M peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
        
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        step = min(step, (sc - sc_lb) / (gn * gn + 1e-30))
        step = min(step, 0.08)
        
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay
    
    return best_a


def coordinate_descent_peak(a, n_sweeps=5, n_top=60):
    """Coordinate descent on peak-contributing elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        # Find all near-peak positions
        threshold = 0.98 * c.max()
        peak_idx = np.where(c >= threshold)[0]
        
        for k_star in peak_idx:
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                continue
            
            seg = a[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            top_count = min(n_top, len(contrib))
            top_local = np.argsort(contrib)[::-1][:top_count]
            top_global = j_start + top_local
            
            for j in top_global:
                a_old_j = a[j]
                if a_old_j < 1e-12:
                    continue
                cur_sc = compute_score(a)
                best_val = a_old_j
                best_local_sc = cur_sc
                for factor in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3]:
                    cv = a_old_j * factor
                    cv = max(0.0, min(1000.0, cv))
                    if abs(cv - a_old_j) < 1e-10:
                        continue
                    a[j] = cv
                    S_tmp = a.sum()
                    if S_tmp < 1e-12:
                        a[j] = a_old_j
                        continue
                    a_scaled = a / S_tmp
                    sc = compute_score(a_scaled)
                    if sc < best_local_sc:
                        best_local_sc = sc
                        best_val = cv
                    a[j] = a_old_j
                a[j] = best_val
        
        S = a.sum()
        if S > 1e-12:
            a = a / S
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
    
    return best_a


def fienup_concentrate(a, n_iters=200):
    """

    This should flatten the convolution (opposite of the previous program's approach).

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P[0] = 0  # Exclude DC from modification
        
        if P.sum() < 1e-15:
            break
        
        # We want to reduce non-DC power
        # Use a soft threshold: reduce bins proportionally to their power
        progress = it / n_iters
        strength = 0.1 + 0.3 * progress  # Start gentle, get stronger
        
        # Reduce non-DC bins
        scale = np.ones_like(P)
        # Reduce non-DC bins by a factor
        scale[1:] = 1.0 - strength * (P[1:] / max(P[1:].sum(), 1e-15))
        scale[1:] = np.maximum(scale[1:], 0.1)
        
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)

        if S_new < 1e-12:

        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def fienup_flat_conv(a, n_iters=200):
    """
    Fienup that directly targets flattening the convolution.
    In frequency domain, this means making |A|^2 closer to a delta at DC.
    Equivalently, we reduce the variance of the convolution in time domain.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        c_mean = c.sum() / len(c)
        
        # Compute "error" spectrum: FFT of (c - c_mean)
        # We want to reduce the high-frequency components of c
        M_fft = 1 << (2 * n - 1).bit_length()
        C = np.fft.rfft(c, M_fft)
        
        # Reduce high-frequency components of the convolution
        # This corresponds to making c smoother (flatter)
        freq = np.arange(len(C))
        # Keep low frequencies, reduce high frequencies
        cutoff = len(C) * (0.3 + 0.3 * (it / n_iters))
        mask = np.ones(len(C))
        mask[int(cutoff):] = 0.5 + 0.3 * (it / n_iters)
        
        C_mod = C * mask
        c_new = np.fft.irfft(C_mod, M_fft)[:2*n-1]
        
        # Now we need to find a such that a*a ≈ c_new
        # Use one step of "gradient descent" in the a-space
        # dc/da = 2 * (template convolution)
        # Alternatively, use the frequency domain: A^2 ≈ C_mod, so A ≈ sqrt(C_mod)
        A = np.fft.rfft(a, M_fft)
        # Phase recovery: keep the phase of A, use magnitude from C_mod
        P_target = C_mod.real  # Should be real and non-negative
        P_target = np.maximum(P_target, 0)
        A_mag_target = np.sqrt(P_target)
        A_phase = A / np.maximum(np.abs(A), 1e-15)
        A_new = A_mag_target * A_phase
        
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

            update_best(sc2, a_opt, "FINAL-CD+LBFGS")

    return out
