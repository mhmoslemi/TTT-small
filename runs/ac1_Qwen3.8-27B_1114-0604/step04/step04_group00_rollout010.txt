"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses improved Fienup spectral flattening, adaptive L-BFGS with LSE smoothing,
simulated annealing restarts, and multi-length exploration.
Key innovations: log-domain parameterization, adaptive beta schedule,
momentum-based subgradient, and structured perturbations.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    log_sum = float(np.log(exp_sum))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # dLSE/da_i = sum_k p_k * a_{k-i} for valid indices
    # This is cross-correlation: (a cross p)[i] = sum_k a[k] * p[k+i]
    rev_a = a[::-1].copy()
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g_lse = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n / S3) * (S * g_lse - 2.0 * L)
    return score, grad


def _lbfgs_obj(x, beta):
    """Top-level objective for scipy."""
    return lse_score_and_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 1e-10, 1000.0).copy()
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        _lbfgs_obj, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fienup_refined(a, n_iters=300, clip_power=0.5, n_bands=3):
    """
    Improved Fienup: multi-band spectral flattening with adaptive strength.
    Instead of a single threshold, divide spectrum into bands and flatten each.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    n_bins = M_fft // 2 + 1
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        # Adaptive: start with more aggressive flattening, ease off
        strength = 0.3 + 0.4 * progress

        # Multi-band approach: flatten each band toward its mean
        band_size = max(1, n_bins // n_bands)
        scale = np.ones(n_bins)
        for b in range(n_bands):
            lo = b * band_size
            hi = min(lo + band_size, n_bins)
            if hi <= lo:
                continue
            band = P[lo:hi]
            band_mean = float(band.mean())
            if band_mean < 1e-20:
                continue
            ratio = P[lo:hi] / band_mean
            # Reduce peaks, boost valleys (but conservatively)
            above = ratio > 1.0
            if above.any():
                reduce = (1.0 / ratio[above]) ** strength
                scale[lo:hi][above] = reduce
            below = ratio < 1.0
            if below.any():
                boost = (1.0 / ratio[below]) ** (strength * 0.3)
                scale[lo:hi][below] = boost

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_quantile(a, n_iters=300, top_frac=0.2, reduction=0.5):
    """
    Fienup using quantile-based thresholding.
    Reduce the top fraction of spectrum bins.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        # Adaptive top fraction
        frac = top_frac * (1.0 + 0.5 * it / n_iters)
        thresh = np.quantile(P, 1.0 - frac)

        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            # Soft reduction
            scale[above] = (thresh / P[above]) ** reduction

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def anneal_subgradient(a, n_iters=5000, top_m=5, step_init=0.03,
                       noise_scale=0.01, cooling=0.9997):
    """
    Subgradient method with simulated annealing noise.
    Uses top-M peaks for a more robust subgradient.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    rng = np.random.default_rng(12345)
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    temp = noise_scale

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Top-M peaks for subgradient
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, (sc - 1.0) / (gn * gn + 1e-30) * 0.5)
        step = min(step, 0.05)

        # Add annealing noise
        noise = temp * rng.standard_normal(n)
        a -= step * grad
        a += noise
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        temp *= cooling
        step *= 0.9999

    return best_a


def log_domain_lbfgs(a0, beta, maxiter=5000):
    """
    L-BFGS in log domain: a_i = exp(x_i). Ensures non-negativity automatically.
    """
    n = len(a0)
    a0 = np.maximum(a0, 1e-10).copy()
    a0 = a0 / a0.sum()
    x0 = np.log(a0)

    def log_obj(x):
        a = np.exp(x)
        a = a / a.sum()  # normalize
        sc, grad_a = lse_score_and_grad(a, beta)
        # Chain rule: dx/dx_i = 1/a_i, so grad_x = grad_a * a
        # But we normalized a, so it's more complex. Use quotient rule:
        # a_i = exp(x_i) / sum(exp(x))
        # da_i/dx_j = a_i * (delta_ij - a_j)
        # grad_x_j = sum_i grad_a_i * da_i/dx_j = sum_i grad_a_i * a_i * (delta_ij - a_j)
        #           = a_j * grad_a_j - a_j * sum_i grad_a_i * a_i
        ga = grad_a * a
        grad_x = ga - a * float(ga.sum())
        return sc, grad_x

    res = minimize(log_obj, x0, method='L-BFGS-B', jac=True,
                   options={'ftol': 1e-16, 'gtol': 1e-14,
                            'maxiter': maxiter, 'maxfun': maxiter * 30})
    a_out = np.exp(res.x)
    a_out = a_out / a_out.sum()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def structured_perturb(a, rng, alpha=0.05):
    """Perturb with structured noise (low-frequency components)."""
    n = len(a)

    noise_f = np.fft.rfft(noise)

    a_pert = np.clip(a_pert, 0.0, 1000.0)
    S = a_pert.sum()
    if S > 1e-12:

    inits.append(np.maximum(0.5 + 0.2 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(6.0 * np.pi * x), 0.01))
    return inits


def get_parent():
    """Try to load parent sequence from global scope."""
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                return np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase search for optimal non-negative sequence.
    Phase 1: Fienup from diverse initializations (30% time)
    Phase 2: L-BFGS with adaptive beta (25% time)
    Phase 3: Annealing + perturbation restarts (30% time)
    Phase 4: Multi-length exploration (10% time)
    Phase 5: Final polish (5% time)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    parent = get_parent()

    best_score = np.inf
    best_a = np.ones(6962) * (1.0 / 6962)

    def _update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent: {'loaded (n=' + str(len(parent)) + ')' if parent is not None else 'none'}")

    # Determine lengths to try
    lengths = [6962, 8192, 4096]
    if parent is not None and len(parent) not in lengths:
        lengths.insert(0, len(parent))
    lengths = sorted(set(lengths))
    print(f"Lengths: {lengths}")

    # ============================================================
    # Phase 1: Fienup from diverse inits (30% of time)
    # ============================================================
    p1_end = t_start + 0.30 * (budget_s - 30.0)
    print(f"\n=== Phase 1: Fienup (until t={p1_end - t_start:.0f}s) ===")

    for n in lengths:
        if time.monotonic() > p1_end:
            break
        print(f"  n={n}")
        inits = build_inits(n, rng, parent)

        for i, a0 in enumerate(inits):
            if time.monotonic() > p1_end:
                break
            a0 = np.maximum(a0, 0.01)
            a0 = a0 / a0.sum()

            # Refined Fienup
            a_f = fienup_refined(a0, n_iters=200, clip_power=0.5, n_bands=4)
            _update(compute_score(a_f), a_f, f"Fienup-ref n={n} i={i}")

            # Quantile Fienup
            if time.monotonic() < p1_end:
                a_q = fienup_quantile(a0, n_iters=150, top_frac=0.15, reduction=0.6)
                _update(compute_score(a_q), a_q, f"Fienup-quant n={n} i={i}")

    print(f"Phase 1 done: best={best_score:.8f}")

    # ============================================================
    # Phase 2: L-BFGS with adaptive beta (25% of time)
    # ============================================================
    p2_end = t_start + 0.55 * (budget_s - 30.0)
    print(f"\n=== Phase 2: L-BFGS (until t={p2_end - t_start:.0f}s) ===")

    betas = [50, 100, 200, 500, 1000, 2000, 5000, 10000]
    for beta in betas:
        if time.monotonic() > p2_end:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=100000)
        _update(sc, a_opt, f"LBFGS beta={beta}")

    # Log-domain L-BFGS
    if time.monotonic() < p2_end:
        sc, a_opt = log_domain_lbfgs(best_a, beta=500, maxiter=3000)
        _update(sc, a_opt, "LogLBFGS")

    print(f"Phase 2 done: best={best_score:.8f}")

    # ============================================================
    # Phase 3: Annealing + perturbation restarts (30% of time)
    # ============================================================
    p3_end = t_start + 0.85 * (budget_s - 30.0)
    print(f"\n=== Phase 3: Anneal + Perturb (until t={p3_end - t_start:.0f}s) ===")

    n_restarts = 8
    for r in range(n_restarts):
        if time.monotonic() > p3_end:
            break
        print(f"  Restart {r+1}/{n_restarts}")

        # Create perturbed starting point
        alpha = 0.03 + 0.02 * r
        if r % 2 == 0:
            a_start = structured_perturb(best_a, rng, alpha=alpha)
        else:
            a_start = random_perturb(best_a, rng, alpha=alpha)

        # Annealing subgradient
        a_ann = anneal_subgradient(a_start, n_iters=3000, top_m=5,
                                   step_init=0.02, noise_scale=0.005,
                                   cooling=0.9998)
        sc = compute_score(a_ann)
        _update(sc, a_ann, f"Anneal r={r}")

        # Quick Fienup on the result
        if time.monotonic() < p3_end:
            a_f = fienup_refined(a_ann, n_iters=100, clip_power=0.4, n_bands=3)
            sc = compute_score(a_f)
            _update(sc, a_f, f"Fienup-after-anneal r={r}")

        # Short L-BFGS polish
        if time.monotonic() < p3_end:
            sc, a_opt = run_lbfgs(a_ann, beta=1000, maxiter=2000, maxfun=40000)
            _update(sc, a_opt, f"LBFGS-polish r={r}")

    print(f"Phase 3 done: best={best_score:.8f}")

    # ============================================================
    # Phase 4: Multi-length exploration (10% of time)
    # ============================================================
    p4_end = deadline - 60.0
    print(f"\n=== Phase 4: Multi-length (until t={p4_end - t_start:.0f}s) ===")

    for n_try in [8192, 12288, 16384]:
        if time.monotonic() > p4_end:
            break
        print(f"  Trying n={n_try}")
        # Interpolate best to new length
        x_old = np.linspace(0, 1, len(best_a))
        x_new = np.linspace(0, 1, n_try)
        a_interp = np.interp(x_new, x_old, best_a)
        a_interp = np.maximum(a_interp, 0.01)
        a_interp = a_interp / a_interp.sum()

        # Fienup at new length
        a_new = fienup_refined(a_interp, n_iters=150, clip_power=0.5, n_bands=4)
        sc = compute_score(a_new)
        _update(sc, a_new, f"Fienup n={n_try}")

        # L-BFGS at new length
        if time.monotonic() < p4_end:
            sc, a_opt = run_lbfgs(a_interp, beta=500, maxiter=3000, maxfun=60000)
            _update(sc, a_opt, f"LBFGS n={n_try}")

    print(f"Phase 4 done: best={best_score:.8f}")

    # ============================================================
    # Phase 5: Final polish (5% of time)
    # ============================================================
    print(f"\n=== Phase 5: Final Polish ===")

    # Long L-BFGS with high beta
    if time.monotonic() < deadline - 20:
        sc, a_opt = run_lbfgs(best_a, beta=5000, maxiter=8000, maxfun=150000)
        _update(sc, a_opt, "Final-LBFGS-5000")

    # Another Fienup pass
    if time.monotonic() < deadline - 10:
        a_f = fienup_refined(best_a, n_iters=200, clip_power=0.6, n_bands=5)
        sc = compute_score(a_f)
        _update(sc, a_f, "Final-Fienup")

    # Quick anneal
    if time.monotonic() < deadline - 5:
        a_ann = anneal_subgradient(best_a, n_iters=2000, top_m=3,
                                   step_init=0.01, noise_scale=0.002,
                                   cooling=0.9999)
        sc = compute_score(a_ann)
        _update(sc, a_ann, "Final-Anneal")

    # Final L-BFGS
    if time.monotonic() < deadline - 2:
        sc, a_opt = run_lbfgs(best_a, beta=10000, maxiter=5000, maxfun=100000)
        _update(sc, a_opt, "Final-LBFGS-10000")

    # ============================================================
    # Final output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
