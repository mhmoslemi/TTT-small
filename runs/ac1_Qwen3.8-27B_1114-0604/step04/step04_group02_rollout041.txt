```python
"""
Search for a non-negative sequence minimizing:

    2*n*max(convolve(a,a)) / (sum(a)^2)

The program uses:
  * softmax-parameterized L-BFGS on a top-peak log-sum-exp surrogate,
  * convex "best response" LPs (minimize peak of a*b for sum b=1),
  * cross-convolution L-BFGS best responses,
  * occasional length resampling and controlled perturbations.

The previous parent sequence (global height_sequence_1, if present) is used
as the start point. The best sequence found is returned before the deadline.
"""

import time

import numpy as np
from scipy.optimize import minimize, linprog, Bounds


# ----------------------------------------------------------------------------
# Basic numerical helpers
# ----------------------------------------------------------------------------

def next_pow2(m):
    """Smallest power of two >= m."""
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def safe_normalize(a):
    """Clip to non-negative finite values and normalize sum to 1."""
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a[~np.isfinite(a)] = 0.0
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def exact_score(a):
    """Exact internal score, equivalent to evaluate_sequence (except IO)."""
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12:
        return np.inf
    c = fast_self_conv(a)
    m = float(np.max(c)) if c.size else 0.0
    return float(2.0 * len(a) * m / (s * s))


def fast_self_conv(a):
    """Linear self-convolution via FFT, clipped to non-negative."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    m = 2 * n - 1
    size = next_pow2(m)
    A = np.fft.rfft(a, size)
    c = np.fft.irfft(A * A, size)[:m]
    return np.maximum(c, 0.0)


def fast_cross_conv(a, b):
    """Linear convolution of two same-length sequences via FFT."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    if len(b) != n:
        # Fallback, should not happen in this program.
        out = np.convolve(a, b)
        return np.maximum(out, 0.0)
    m = 2 * n - 1
    size = next_pow2(m)
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    c = np.fft.irfft(A * B, size)[:m]
    return np.maximum(c, 0.0)


def softmax(x):
    """Stable softmax."""
    x = np.asarray(x, dtype=np.float64)
    if x.size == 0:
        return x
    if not np.all(np.isfinite(x)):
        x = np.nan_to_num(x, nan=0.0, posinf=50.0, neginf=-50.0)
    x = x - np.max(x)
    e = np.exp(x)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(x.shape, 1.0 / float(x.size), dtype=np.float64)
    return e / s


def log_from_a(a):
    """Safe log for positive sequences."""
    a = np.asarray(a, dtype=np.float64)
    return np.log(np.maximum(a, 1e-12))


# ----------------------------------------------------------------------------
# LSE / gradient machinery
# ----------------------------------------------------------------------------

def lse_weights(c, beta, topk):
    """
    Return (L, p), where L is an LSE approximation to max(c) and p are
    normalized softmax weights. If topk is given, only the largest topk
    coordinates receive weight.
    """
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return 0.0, np.array([], dtype=np.float64)
    cmax = float(np.max(c))
    if topk is None or topk >= c.size:
        scaled = beta * (c - cmax)
        e = np.exp(np.maximum(scaled, -100.0))
        s = float(e.sum())
        p = np.zeros(c.size, dtype=np.float64)
        if s <= 0.0 or not np.isfinite(s):
            p[int(np.argmax(c))] = 1.0
            return cmax, p
        p = e / s
        L = cmax + float(np.log(s)) / beta
        return L, p

    topk = int(max(1, topk))
    idx = np.argpartition(c, -topk)[-topk:]
    # Ensure the all-time maximum is represented.
    amax = int(np.argmax(c))
    if amax not in idx:
        idx[int(np.argmin(c[idx]))] = amax
    csel = c[idx]
    scaled = beta * (csel - cmax)
    e = np.exp(np.maximum(scaled, -100.0))
    s = float(e.sum())
    p = np.zeros(c.size, dtype=np.float64)
    if s <= 0.0 or not np.isfinite(s):
        p[amax] = 1.0
        return cmax, p
    p[idx] = e / s
    L = cmax + float(np.log(s)) / beta
    return L, p


def conv_grad_weights(p, q, factor):
    """
    For c_k = sum_i q_i r_{k-i}, return factor * gradient wrt r:
      grad_j = factor * sum_k p_k q_{k-j}
    (with valid-index handling). This is the same routine used for self
    gradient (factor=2) and cross-gradient (factor=1).
    """
    n = len(q)
    need = 2 * n - 1
    p = np.asarray(p, dtype=np.float64)
    if p.size != need:
        tmp = np.zeros(need, dtype=np.float64)
        m = min(p.size, need)
        if m > 0:
            tmp[:m] = p[:m]
        p = tmp

    m = n + need - 1
    size = next_pow2(m)
    q_rev = q[::-1]
    Q = np.fft.rfft(q_rev, size)
    P = np.fft.rfft(p, size)
    out = np.fft.irfft(Q * P, size)[n - 1:n - 1 + n]
    out = np.maximum(out, 0.0)
    return float(factor) * out


class SelfLSEObjective:
    """
    Softmax-parameterized LSE objective for self-convolution peak.
    x -> a = softmax(x), score = 2n * LSE(a*a).
    """

    def __init__(self, beta, topk, reg):
        self.beta = float(beta)
        self.topk = topk
        self.reg = float(reg)

    def __call__(self, x):
        x = np.asarray(x, dtype=np.float64)
        if not np.all(np.isfinite(x)):
            x = np.nan_to_num(x, nan=0.0, posinf=50.0, neginf=-50.0)
        xc = x - float(np.mean(x))
        a = softmax(xc)
        c = fast_self_conv(a)
        L, p = lse_weights(c, self.beta, self.topk)
        n = a.size
        f = 2.0 * n * L + self.reg * float(np.dot(xc, xc))
        g = conv_grad_weights(p, a, 2.0)
        dot = float(np.dot(a, g))
        grad = a * (g - dot) + 2.0 * self.reg * xc
        grad -= float(np.mean(grad))
        return float(f), grad


class CrossLSEObjective:
    """
    Softmax-parameterized LSE objective for cross-convolution a_fixed * b.
    Used to compute a best-response / valley direction b.
    """

    def __init__(self, a_fixed, beta, topk, reg):
        self.a = safe_normalize(a_fixed)
        self.beta = float(beta)
        self.topk = topk
        self.reg = float(reg)

    def __call__(self, y):
        y = np.asarray(y, dtype=np.float64)
        if not np.all(np.isfinite(y)):
            y = np.nan_to_num(y, nan=0.0, posinf=50.0, neginf=-50.0)
        yc = y - float(np.mean(y))
        b = softmax(yc)
        z = fast_cross_conv(self.a, b)
        L, p = lse_weights(z, self.beta, self.topk)
        n = b.size
        f = 2.0 * n * L + self.reg * float(np.dot(yc, yc))
        g = conv_grad_weights(p, self.a, 1.0)
        dot = float(np.dot(b, g))
        grad = b * (g - dot) + 2.0 * self.reg * yc
        grad -= float(np.mean(grad))
        return float(f), grad


def estimate_fun_budget(deadline, n):
    """Rough number of L-BFGS function evaluations affordable."""
    rem = float(deadline) - time.monotonic() - 15.0
    if rem <= 5.0:
        return 0
    n = max(1, int(n))
    per_sec = max(10.0, 120000.0 / float(n))
    return int(np.clip(rem * per_sec, 400, 10000))


def run_self_lbfgs(a0, beta, topk, maxfun, maxiter):
    """Polish a sequence using softmax L-BFGS plus exact score check."""
    a0 = safe_normalize(a0)
    sc0 = exact_score(a0)
    if maxfun <= 0:
        return sc0, a0
    x0 = log_from_a(a0)
    x0 -= float(np.mean(x0))
    obj = SelfLSEObjective(beta, topk, 1e-12)
    try:
        res = minimize(
            obj,
            x0,
            method="L-BFGS-B",
            options={
                "maxiter": max(20, int(maxiter)),
                "maxfun": int(maxfun),
                "ftol": 1e-15,
                "gtol": 1e-8,
                "maxls": 30,
            },
        )
    except Exception:
        return sc0, a0

    xout = getattr(res, "x", x0)
    if xout is None or not np.all(np.isfinite(xout)):
        xout = x0
    aout = safe_normalize(softmax(xout))
    sc = exact_score(aout)
    if sc < sc0 + 1e-9:
        return sc, aout
    return sc0, a0


def run_cross_lbfgs(a_fixed, beta, topk, maxfun, maxiter):
    """Compute a best-response direction b minimizing LSE(a_fixed*b)."""
    a_fixed = safe_normalize(a_fixed)
    if maxfun <= 0:
        return None, np.inf
    b0 = safe_normalize(a_fixed)
    y0 = log_from_a(b0)
    y0 -= float(np.mean(y0))
    obj = CrossLSEObjective(a_fixed, beta, topk, 1e-12)
    try:

            obj,
            y0,
            method="L-BFGS-B",
            options={
                "maxiter": max(20, int(maxiter)),
                "maxfun": int(maxfun),
                "ftol": 1e-15,
                "gtol": 1e-8,
                "maxls": 30,
            },
        )
    except Exception:
        return None, np.inf

    yout = getattr(res, "x", y0)
    if yout is None or not np.all(np.isfinite(yout)):
        return None, np.inf
    b = safe_normalize(softmax(yout))
    zmax = float(np.max(fast_cross_conv(a_fixed, b)))
    return b, zmax

def make_conv_matrix(a, lags):
    """
    Build rows A_k such that (A b)_k = (a*b)_k for selected lag indices k.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    lags = np.asarray(lags, dtype=np.int64)
    A = np.zeros((lags.size, n), dtype=np.float64)
    max_k = 2 * n - 2
    for r, kk in enumerate(lags):
        k = int(kk)
        if k < 0 or k > max_k:
            continue
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if je >= js:
            A[r, js:je + 1] = a[k - je:k - js + 1][::-1]
    return A


def lp_min_t_best_response(a, K_init=80, K_max=220, lp_iters=4):
    """
    Solve, with lazy constraints:
        min t  s.t. (a*b)_k <= t for selected k, sum b = 1, b >= 0.
    """
    a = safe_normalize(a)
    n = a.size
    if n <= 0:
        return None, np.inf
    c = fast_self_conv(a)
    M = float(np.max(c))
    lags = np.argsort(c, kind="stable")[::-1][:int(min(K_init, 2 * n - 1))]
    if lags.size == 0:
        return None, np.inf

    b = a.copy()
    t = M
    for _ in range(int(lp_iters)):
        try:
            A = make_conv_matrix(a, lags)
            K = A.shape[0]
            if K == 0:
                break
            A_ub = np.empty((K, n + 1), dtype=np.float64)
            A_ub[:, :n] = A
            A_ub[:, -1] = -1.0
            b_ub = np.zeros(K, dtype=np.float64)

            A_eq = np.zeros((1, n + 1), dtype=np.float64)
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0], dtype=np.float64)

            c_obj = np.zeros(n + 1, dtype=np.float64)
            c_obj[-1] = 1.0

            lb = np.zeros(n + 1, dtype=np.float64)
            ub = np.ones(n + 1, dtype=np.float64)
            ub[-1] = np.inf
            bounds = Bounds(lb, ub)

            res = linprog(
                c_obj,
                A_ub=A_ub,
                b_ub=b_ub,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=bounds,
                method="highs",
            )
        except Exception:
            break

        if res is None or not getattr(res, "success", False):
            continue
        x = getattr(res, "x", None)
        if x is None or len(x) != n + 1:
            continue
        b = np.asarray(x[:n], dtype=np.float64)
        tt = float(x[-1])
        b[~np.isfinite(b)] = 0.0
        b = np.maximum(b, 0.0)
        sb = float(b.sum())
        if sb <= 1e-10:
            break
        b = b / sb
        t = max(0.0, tt)

        z = fast_cross_conv(a, b)
        viol = z - t
        max_viol = float(np.max(viol)) if viol.size else 0.0
        tol = 1e-8 + 1e-6 * max(1.0, abs(t))
        if max_viol <= tol:
            return b, t

        if lags.size >= int(K_max):
            break
        add = int(min(K_init // 2, 2 * n - 1))
        if add > 0:
            bad = np.argpartition(viol, -add)[-add:]
            new_lags = np.unique(np.concatenate((lags, bad)))
            if new_lags.size > 0:
                scores = c[new_lags] + 0.5 * np.maximum(z[new_lags] - t, 0.0)
                order = np.argsort(scores, kind="stable")[::-1]
                lags = new_lags[order]
                lags = lags[:int(K_max)]

    return b, t


def lp_maxsum_best_response(a, K=120, B=4.0):
    """
    Similar to previous SOTA:
        max sum b  s.t. (a*b)_k <= M for selected peaks, 0 <= b <= 1,
        sum b <= B.
    The solution is normalized afterwards.
    """
    a = safe_normalize(a)
    n = a.size
    if n <= 0:
        return None, np.inf
    c = fast_self_conv(a)
    M = float(np.max(c))
    lags = np.argsort(c, kind="stable")[::-1][:int(min(K, 2 * n - 1))]
    if lags.size == 0:
        return None, np.inf
    try:
        A = make_conv_matrix(a, lags)
        K = A.shape[0]
        A_full = np.vstack((A, np.ones((1, n), dtype=np.float64)))
        b_full = np.concatenate((np.full(K, M, dtype=np.float64),
                                 np.array([B], dtype=np.float64)))
        c_obj = -np.ones(n, dtype=np.float64)
        lb = np.zeros(n, dtype=np.float64)
        ub = np.ones(n, dtype=np.float64)
        bounds = Bounds(lb, ub)
        res = linprog(
            c_obj,
            A_ub=A_full,
            b_ub=b_full,
            bounds=bounds,
            method="highs",
        )
    except Exception:
        return None, np.inf

    if res is None or not getattr(res, "success", False):
        return None, np.inf
    b = getattr(res, "x", None)
    if b is None or len(b) != n:
        return None, np.inf
    b = np.asarray(b, dtype=np.float64)
    b[~np.isfinite(b)] = 0.0
    b = np.clip(b, 0.0, 1.0)
    sb = float(b.sum())
    if sb <= 1e-10:
        return None, np.inf
    return b / sb, M

def consider_with_score(state, a, sc, label):
    """Update the global best state if needed."""
    if sc is None or not np.isfinite(sc):
        return
    a = safe_normalize(a)
    if sc < state.best_score - 1e-15:
        state.best_score = float(sc)
        state.best_a = a.copy()
        print(f"[{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic():.0f}s)")
    return sc


def resize_sequence(a, n):
    """Linear resizing to length n."""
    a = safe_normalize(a)
    n = int(n)
    if n <= 0:
        return a
    if a.size == n:
        return a
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, n)
    b = np.interp(x_new, x_old, a)
    return safe_normalize(b)


def perturb_sequence(a, rng, scale=0.05):
    """Positive-preserving multiplicative perturbations."""
    a = safe_normalize(a)
    n = a.size
    x = np.arange(n, dtype=np.float64) / float(n)
    strat = int(rng.integers(0, 5))

    if strat == 0:
        b = a * (1.0 + scale * rng.standard_normal(n))
    elif strat == 1:
        freq = 1.0 + 10.0 * float(rng.random())
        phase = 2.0 * np.pi * float(rng.random())
        amp = scale * (1.0 + float(rng.random()))
        b = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strat == 2:
        b = a.copy()
        nblocks = 3 + int(rng.integers(0, 8))
        block = max(1, n // max(1, nblocks))
        for s in range(0, n, block):
            e = min(n, s + block)
            b[s:e] *= (1.0 + 2.0 * scale * (float(rng.random()) - 0.5))
    elif strat == 3:
        nsel = max(1, n // 4)
        sel = rng.choice(n, size=nsel, replace=False)
        b = a.copy()
        b[sel] *= (1.0 + 3.0 * scale * (rng.random(nsel) - 0.5))
    else:
        b = a * (1.0 + 0.5 * scale * rng.random(n))
    return safe_normalize(b)


def blend_search(a, b):
    """Search for a good convex/geometric mixture between a and b."""
    a = safe_normalize(a)
    b = safe_normalize(b)
    best_sc = exact_score(a)
    best_a = a.copy()

    alphas = (0.05, 0.10, 0.15, 0.20, 0.30, 0.40, 0.50, 0.65, 0.80, 0.95)
    for alpha in alphas:
        m = (1.0 - alpha) * a + alpha * b
        sc = exact_score(m)
        if sc < best_sc:
            best_sc = sc
            best_a = m

    m = np.sqrt(np.maximum(a, 1e-14) * np.maximum(b, 1e-14))
    m = safe_normalize(m)
    sc = exact_score(m)
    if sc < best_sc:
        best_sc = sc
        best_a = m

    m = b.copy()
    sc = exact_score(m)
    if sc < best_sc:
        best_sc = sc
        best_a = m

    return best_sc, best_a

def local_refine_seq(a, state, sub_deadline, rng, tag):
    """
    Improve one branch by:
      - self LSE L-BFGS,
      - LP/cross best-response alternation,
      - final high-beta self LSE.
    """
    a = safe_normalize(a)
    local_sc = exact_score(a)
    local_a = a.copy()
    consider_with_score(state, local_a, local_sc, tag + ":start")
    n = local_a.size

    # Self LSE L-BFGS schedule.
    self_schedule = [
        (150, 12),
        (600, 8),
        (2500, 5),
        (10000, 3),
        (40000, 2),
    ]
    for beta, topk in self_schedule:
        if time.monotonic() > sub_deadline - 40.0:
            break
        mf = estimate_fun_budget(sub_deadline, n)
        if mf <= 0:
            break
        sc2, a2 = run_self_lbfgs(local_a, beta, topk, mf, max(50, mf // 2))
        if sc2 < local_sc - 1e-13:
            local_sc = sc2
            local_a = a2
            consider_with_score(state, local_a, local_sc,
                                tag + f":self beta={beta} topk={topk}")

    # Best-response alternation.
    for alt in range(3):
        if time.monotonic() > sub_deadline - 110.0:
            break
        b = None
        zmax = np.inf

        if alt == 0 and n <= 13000:
            b, t = lp_min_t_best_response(local_a,
                                          K_init=80, K_max=220, lp_iters=4)
            if b is not None:
                zmax = t
        elif alt == 1 and n <= 20000:
            mf = estimate_fun_budget(sub_deadline, n)
            if mf > 0:
                b, zmax = run_cross_lbfgs(local_a, 70, None, mf, max(50, mf // 2))
        elif alt == 2 and n <= 13000:
            b, zmax = lp_maxsum_best_response(local_a, K=120, B=4.0)

        if b is None:
            continue

        b = safe_normalize(b)
        # If the LP response is too sparse, blend it with uniform for stability.
        support = int(np.count_nonzero(b > max(1e-9, 1.0 / n)))
        if support < max(32, n // 64):
            b = safe_normalize(0.75 * b + 0.25 * (np.ones(n) / float(n)))

        scb = exact_score(b)
        consider_with_score(state, b, scb, tag + f":response {alt}")

        sc_blend, best_blend = blend_search(local_a, b)
        if sc_blend < local_sc - 1e-13:
            local_sc = sc_blend
            local_a = best_blend
            consider_with_score(state, local_a, local_sc, tag + f":blend {alt}")

            # Short polish after a successful blend.
            mf = estimate_fun_budget(sub_deadline, n)
            if mf > 300:
                sc2, a2 = run_self_lbfgs(local_a, 2500, 8, max(300, mf // 2),
                                         max(50, mf // 4))
                if sc2 < local_sc - 1e-13:
                    local_sc = sc2
                    local_a = a2
                    consider_with_score(state, local_a, local_sc,
                                        tag + f":polish after blend {alt}")

    # Final high-precision self LSE.
    for beta, topk in [(20000, 3), (80000, 2)]:
        if time.monotonic() > sub_deadline - 25.0:
            break
        mf = estimate_fun_budget(sub_deadline, n)
        if mf <= 0:
            break
        sc2, a2 = run_self_lbfgs(local_a, beta, topk, mf, max(50, mf // 2))
        if sc2 < local_sc - 1e-13:
            local_sc = sc2
            local_a = a2
            consider_with_score(state, local_a, local_sc,
                                tag + f":final self beta={beta} topk={topk}")

    return local_a

class State:
    """Global best found so far."""
    def __init__(self):
        self.best_score = np.inf
        self.best_a = None


def pair_first(pair):
    return pair[0]


def get_parent():
    """Load previous construction through global height_sequence_1."""
    p = globals().get("height_sequence_1", None)
    if p is None:
        return None
    try:
        arr = np.asarray(p, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr[~np.isfinite(arr)] = 0.0
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s <= 1e-8:
        return None
    return arr / s


def make_initial_candidates(parent, rng):
    """Create a small pool of starting candidates."""
    cands = []
    if parent is not None:
        cands.append(parent)
        cands.append(perturb_sequence(parent, rng, 0.03))
        cands.append(perturb_sequence(parent, rng, 0.08))
        np_ = parent.size
        for factor in (1.25, 1.5, 0.8):
            n2 = int(round(np_ * factor))
            if 512 <= n2 <= 50000:
                cands.append(resize_sequence(parent, n2))
    else:
        n = 8192
        x = np.arange(n, dtype=np.float64) / float(n)
        cands.append(np.ones(n, dtype=np.float64))
        cands.append(np.abs(rng.normal(1.0, 0.6, n)) + 0.05)
        cands.append(0.2 + 0.8 * np.abs(np.sin(2.0 * np.pi * 5.0 * x)) ** 2.0)
        cands.append(0.2 + 0.8 * np.exp(-4.0 * (x - 0.5) ** 2))
        mode = (rng.standard_normal(n) > 0.0).astype(np.float64)
        cands.append(0.1 + mode)

    scored = []
    for cand in cands:
        try:
            cand = safe_normalize(cand)
            if cand.size < 8:
                continue
            sc = exact_score(cand)
            if np.isfinite(sc):
                scored.append((float(sc), cand))
        except Exception:
            pass
    if not scored:
        scored.append((exact_score(np.ones(1024)), np.ones(1024)))
    scored.sort(key=pair_first)
    return scored[:4]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search entry point. Returns the best sequence found."""
    try:
        bs = float(budget_s)
    except Exception:
        bs = 2400.0
    if not np.isfinite(bs) or bs <= 0.0:
        bs = 2400.0
    bs = max(60.0, bs)

    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    work_budget = bs - 40.0
    deadline = t0 + max(30.0, work_budget)

    state = State()
    parent = get_parent()

    if parent is not None:
        sc = exact_score(parent)
        consider_with_score(state, parent, sc, "parent")
        # A few direct evaluations of a quick mix with uniform are cheap.
        for mix in (0.02, 0.05):
            cand = safe_normalize((1.0 - mix) * parent + mix * (np.ones(parent.size) / parent.size))
            sc = exact_score(cand)
            consider_with_score(state, cand, sc, "parent-mix")

    scored = make_initial_candidates(parent, rng)
    for sc, cand in scored:
        consider_with_score(state, cand, sc, "init")

    if state.best_a is None:
        state.best_a = safe_normalize(np.ones(1024))
        state.best_score = exact_score(state.best_a)

    a_start = state.best_a

    # Phase 1: deep refine the strongest initial branch.
    d1 = t0 + 0.38 * work_budget
    a1 = local_refine_seq(a_start, state, d1, rng, "P1")

    # Phase 2: perturb the global best and refine.
    d2 = t0 + 0.60 * work_budget
    p1 = perturb_sequence(state.best_a, rng, scale=0.035)
    local_refine_seq(p1, state, d2, rng, "P2a")
    p2 = perturb_sequence(state.best_a, rng, scale=0.07)
    local_refine_seq(p2, state, d2, rng, "P2b")

    # Phase 3: try a nearby length from the best sequence.
    d3 = t0 + 0.80 * work_budget
    curr = state.best_a
    nc = curr.size
    for factor in (1.25, 1.5, 0.8):
        n2 = int(round(nc * factor))
        if 512 <= n2 <= 50000 and n2 != nc:
            cand = resize_sequence(curr, n2)
            sc = exact_score(cand)
            consider_with_score(state, cand, sc, f"P3 len={n2}")
            if time.monotonic() < d3 - 30:
                local_refine_seq(cand, state, d3, rng, f"P3 len={n2}")

    # Phase 4: final polish of global best.
    final_a = local_refine_seq(state.best_a, state, deadline, rng, "Final")

    # Ensure return type is a plain Python list of finite non-negative floats.
    out = state.best_a if state.best_a is not None else final_a
    out = np.asarray(out, dtype=np.float64).ravel()
    if out.size == 0 or not np.all(np.isfinite(out)):
        out = np.ones(64, dtype=np.float64)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s <= 0.01:
        out = np.ones(64, dtype=np.float64)
        s = 64.0
    out = out / s  # normalization does not change the score

    ret = [float(v) for v in out]

    # Optional final verification.
    try:
        verified = float(evaluate_sequence(ret))
        if np.isfinite(verified) and verified < state.best_score - 1e-9:
            print(f"Final verification improved bound: {verified:.10f}")
    except Exception:
        verified = np.nan

    print("=" * 70)
    print(f"FINAL internal best: {state.best_score:.10f}, length={len(ret)}")
    if np.isfinite(verified):
        print(f"FINAL evaluated: {verified:.10f}")
    print(f"Time used: {time.monotonic() - t0:.1f}s / budget {bs:.0f}s")
    print("=" * 70)
    return ret
```
