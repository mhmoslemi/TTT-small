```python
"""
Optimizer for sequences minimizing
    2*n*max(np.convolve(a,a)) / (sum(a))^2

Method:
- Work with non-negative sequences, normalized internally.
- Use a differentiable log-sum-exp (LSE) surrogate for the maximum convolution
  coefficient, optimized in log-softmax variables with L-BFGS-B.
- Use a first-order spectral "convolution steering" step that directly shapes
  the convolution by updating the DFT through Delta A = -Delta C / (2A).
- Use local coordinate moves on active convolution peaks and diverse restarts.
"""

import numpy as np
import time
from scipy.optimize import minimize, Bounds

# Globals used by the top-level LSE objective for L-BFGS.
_LS_N = 0
_LS_M = 1
_LS_BETA = 1.0


def fft_len(n):
    m = 2 * int(n) - 1
    if m < 1:
        m = 1
    return 1 << (m - 1).bit_length()


def conv_full(a):
    n = int(a.size)
    M = fft_len(n)
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def normalize_nonnegative(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if (not np.isfinite(s)) or s < 1e-12:
        return np.ones(a.size, dtype=np.float64) / max(1, a.size)
    return a / s


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return float('inf')
    a = np.clip(a, 0.0, 1000.0)
    S = float(a.sum())
    if (not np.isfinite(S)) or S < 1e-12:
        return float('inf')
    n = int(a.size)
    M = fft_len(n)
    A = np.fft.rfft(a, M)
    c = np.fft.irfft(A * A, M)[:2 * n - 1]
    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def interpolate_sequence(a, m):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = int(m)
    if a.size == 0:
        return np.ones(m, dtype=np.float64) / max(1, m)
    if a.size == m:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, a.size)
    x_new = np.linspace(0.0, 1.0, m)
    b = np.interp(x_new, x_old, a)
    return normalize_nonnegative(b)


def seq_to_log(a):
    a = normalize_nonnegative(a)
    eps = 1e-15
    z = np.log(np.maximum(a, eps))
    z -= float(z.mean())
    return np.clip(z, -30.0, 30.0).astype(np.float64)


def log_to_seq(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - float(z.max())
    e = np.exp(z)
    s = float(e.sum())
    if (not np.isfinite(s)) or s <= 0.0:
        return np.ones(z.size, dtype=np.float64) / max(1, z.size)
    return e / s


def lse_z_objective(z):
    """Objective and exact gradient for LSE-smoothed score in softmax variables."""
    n = int(_LS_N)
    if n <= 0:
        return 1e100, np.array([], dtype=np.float64)

    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n:
        return 1e100, np.zeros(n, dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return 1e100, np.zeros(n, dtype=np.float64)

    z = z - float(z.max())
    e = np.exp(z)
    s = float(e.sum())
    if (not np.isfinite(s)) or s <= 0.0:
        return 1e100, np.zeros(n, dtype=np.float64)

    a = e / s
    M = int(_LS_M)
    L = 2 * n - 1

    A = np.fft.rfft(a, M)
    c = np.fft.irfft(A * A, M)[:L]
    cmax = float(c.max())
    if (not np.isfinite(cmax)) or cmax <= 0.0:
        return 1e100, np.zeros(n, dtype=np.float64)

    beta = float(_LS_BETA)
    scaled = beta * (c - cmax)
    scaled = np.maximum(scaled, -700.0)
    w = np.exp(scaled)
    wsum = float(w.sum())
    if (not np.isfinite(wsum)) or wsum <= 0.0:
        return 1e100, np.zeros(n, dtype=np.float64)

    # Use unnormalized w in the FFT, then normalize the resulting q.
    P = np.fft.rfft(w, M)
    q_un = np.fft.irfft(P * np.conj(A), M)[:n]
    q = q_un / wsum

    m = float(np.dot(c, w) / wsum)
    Lse = cmax + float(np.log(wsum)) / beta
    score = 2.0 * float(n) * Lse

    grad = 4.0 * float(n) * a * (q - m)
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=np.float64)
    if (not np.isfinite(score)) or score < 0.0:
        return 1e100, np.zeros(n, dtype=np.float64)

    return float(score), np.asarray(grad, dtype=np.float64)


def run_z_lbfgs(a, beta, maxiter=2000, maxfun=8000):
    a = normalize_nonnegative(a)
    n = int(a.size)
    global _LS_N, _LS_M, _LS_BETA
    _LS_N = n
    _LS_M = fft_len(n)
    _LS_BETA = float(beta)

    z0 = seq_to_log(a)
    bounds = [(-30.0, 30.0)] * n
    try:
        res = minimize(
            lse_z_objective,
            z0,
            method='L-BFGS-B',
            jac=True,
            bounds=bounds,
            options={
                'maxiter': int(maxiter),
                'maxfun': int(maxfun),
                'ftol': 1e-13,
                'gtol': 1e-9,
                'maxls': 25,
                'disp': False,
            }
        )
        a_out = log_to_seq(res.x)
        sc = compute_score(a_out)
    except Exception:
        a_out = a.copy()
        sc = compute_score(a_out)

    if (not np.isfinite(sc)) or not np.all(np.isfinite(a_out)):
        a_out = a.copy()
        sc = compute_score(a_out)
    return float(sc), a_out


def lbfgs_schedule(a, deadline, factors=(1, 4, 15, 50)):
    """Run L-BFGS with increasing LSE sharpness until deadline."""
    a = normalize_nonnegative(a)
    t_end = deadline if deadline is not None else float('inf')
    best_score = compute_score(a)
    best_a = a.copy()
    n = int(a.size)

    if n <= 5000:
        base_fun = 12000
    elif n <= 8192:
        base_fun = 9000
    elif n <= 12288:
        base_fun = 5500
    else:
        base_fun = 3500

    for fac in factors:
        if time.monotonic() + 20.0 > t_end:
            break
        beta = max(1000.0, float(n) * float(fac))
        maxfun = max(800, base_fun)
        sc, a2 = run_z_lbfgs(best_a, beta, maxiter=2000, maxfun=maxfun)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()
    return float(best_score), best_a


def spectral_steering(a, deadline=None, beta_start=None, beta_end=None,
                      max_steps=5000, start_gamma=0.15, stall_max=80):
    """
    Frequency-domain first-order step.

    If C = A*A is the DFT of the convolution, then to a first order
    C_new ~= C - gamma*E  is obtained by A_new ~= A - gamma*E/(2*A),
    where E is the DFT of the desired convolution change.
    """
    a = normalize_nonnegative(a)
    n = int(a.size)
    M = fft_len(n)
    t_end = deadline if deadline is not None else float('inf')

    if beta_start is None:
        beta_start = max(1000.0, float(n))
    if beta_end is None:
        beta_end = max(30.0 * float(n), 30.0 * float(beta_start))
    beta_start = float(beta_start)
    beta_end = max(float(beta_start), float(beta_end))

    best_score = compute_score(a)
    best_a = a.copy()
    cur = a.copy()
    cur_score = best_score

    gamma = float(start_gamma)
    stall = 0
    max_steps = int(max_steps)
    ratio = beta_end / beta_start

    for step in range(max_steps):
        if time.monotonic() > t_end:
            break

        if max_steps > 1:
            frac = float(step) / float(max_steps - 1)
            beta = beta_start * (ratio ** frac)
        else:
            beta = beta_end

        A = np.fft.rfft(cur, M)
        c = np.fft.irfft(A * A, M)[:2 * n - 1]
        cmax = float(c.max())
        if (not np.isfinite(cmax)) or cmax <= 0.0:
            break

        scaled = beta * (c - cmax)
        scaled = np.maximum(scaled, -700.0)
        w = np.exp(scaled)
        wsum = float(w.sum())
        if (not np.isfinite(wsum)) or wsum <= 0.0:
            break

        p = w / wsum
        m = float(np.dot(c, p))
        e = p * (c - m)
        emax = float(np.max(np.abs(e)))
        if (not np.isfinite(emax)) or emax < 1e-18 / max(1.0, float(n)):
            break

        E = np.fft.rfft(e, M)
        den = 2.0 * A
        amax = float(np.max(np.abs(A))) if A.size else 1.0
        tol = 1e-12 * max(1.0, amax)

        d = np.zeros(M, dtype=np.complex128)
        mask = np.abs(den) > tol
        d[mask] = E[mask] / den[mask]
        d[0] = 0.0
        delta = np.fft.irfft(d, M)[:n]

        if not np.all(np.isfinite(delta)):
            continue

        max_abs = float(np.max(np.abs(delta)))
        if (not np.isfinite(max_abs)) or max_abs <= 0.0:
            continue

        cap = 5.0 / float(n)
        if max_abs > cap:
            delta = delta * (cap / max_abs)

        improved = False
        for scale in (1.0, 0.5, 0.25, 0.125, 0.0625):
            g = gamma * scale
            trial = cur - g * delta
            trial = np.maximum(trial, 0.0)
            S = float(trial.sum())
            if (not np.isfinite(S)) or S < 1e-12:
                continue
            trial = trial / S
            sc = compute_score(trial)
            if np.isfinite(sc) and sc < cur_score - 1e-15:
                cur = trial.copy()
                cur_score = sc
                if sc < best_score:
                    best_score = sc
                    best_a = trial.copy()
                improved = True
                break

        if improved:
            gamma = min(0.8, gamma * 1.15)
            stall = 0
        else:
            gamma = max(0.02, gamma * 0.65)
            stall += 1
            if stall > stall_max:
                break

    return float(best_score), best_a


def coordinate_polish(a, deadline=None, n_sweeps=3, top_lags=3,
                      top_count=10, bottom_count=10):
    """Local moves that reduce top convolution anti-diagonals."""
    a = normalize_nonnegative(a)
    t_end = deadline if deadline is not None else float('inf')
    n = int(a.size)
    best_score = compute_score(a)
    best_a = a.copy()

    for _ in range(int(n_sweeps)):
        if time.monotonic() > t_end:
            break

        c = conv_full(a)
        if c.size == 0:
            break
        peaks = np.argsort(c)[::-1][:int(top_lags)]
        stall = True

        for k in peaks:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je < js:
                continue

            idx = np.arange(js, je + 1)
            weights = 2.0 * a[k - idx]
            if weights.size == 0 or np.max(weights) <= 0.0:
                continue

            tcount = min(int(top_count), weights.size)
            bcount = min(int(bottom_count), weights.size)
            if tcount < 1 or bcount < 1:
                continue

            order = np.argsort(weights)
            top_local = order[-tcount:]
            bottom_local = order[:bcount]

            top_global = js + top_local
            bottom_global = js + bottom_local
            overlap = np.intersect1d(top_global, bottom_global)
            if overlap.size:
                bottom_global = np.setdiff1d(bottom_global, overlap)
                if bottom_global.size == 0:
                    continue

            improved_here = False
            for mult in (0.85, 0.70, 0.50, 0.30):
                trial = a.copy()
                old = trial[top_global]
                new = old * float(mult)
                transfer = float(old.sum() - new.sum())
                if transfer < 0.0:
                    continue

                trial[top_global] = new
                trial[bottom_global] += transfer / float(bottom_global.size)
                trial = np.maximum(trial, 0.0)
                S = float(trial.sum())
                if (not np.isfinite(S)) or S < 1e-12:
                    continue
                trial = trial / S
                sc = compute_score(trial)
                if np.isfinite(sc) and sc < best_score - 1e-15:
                    best_score = sc
                    best_a = trial.copy()
                    a = trial
                    stall = False
                    improved_here = True
                    break

            if improved_here:
                break

        if stall:
            break

    return float(best_score), best_a


def cosine_init(n, rng, variant=0):
    """Asymmetric smooth non-negative initial densities."""
    n = int(n)
    t = np.linspace(0.0, 1.0, n)
    pi = np.pi

    if variant == 0:
        f = 1.0 + 0.35 * np.cos(pi * t)
    elif variant == 1:
        f = 1.0 + 0.30 * np.cos(pi * t) + 0.20 * np.cos(2.0 * pi * t)
    elif variant == 2:
        f = 1.0 + 0.25 * np.cos(3.0 * pi * t) + 0.18 * np.cos(2.0 * pi * t)
    else:
        f = np.ones(n, dtype=np.float64)
        for m in (1, 2, 3, 4):
            amp = 0.12 + 0.34 * float(rng.random())
            phase = 0.0 if float(rng.random()) < 0.5 else pi
            f += amp * np.cos(float(m) * pi * t + phase)

    f = np.maximum(f, 1e-6)
    if not np.all(np.isfinite(f)):
        f = np.ones(n, dtype=np.float64)
    return normalize_nonnegative(f)


def perturb_sequence(a, rng, mode=0):
    """Random smooth perturbations for restarts."""
    a = normalize_nonnegative(a)
    n = int(a.size)
    t = np.linspace(0.0, 1.0, n)

    if mode == 0:
        sd = 0.03 + 0.05 * float(rng.random())
        mult = 1.0 + sd * rng.standard_normal(n)
        b = a * mult
    elif mode == 1:
        f = 1.0 + 2.0 * float(rng.random())
        ph = 2.0 * np.pi * float(rng.random())
        amp = 0.05 + 0.18 * float(rng.random())
        b = a * (1.0 + amp * np.cos(2.0 * np.pi * f * t + ph))
    elif mode == 2:
        nb = int(4 + rng.integers(0, 9))
        per = max(1, n // nb)
        b = a.copy()
        for s in range(0, n, per):
            e = min(n, s + per)
            b[s:e] *= (0.65 + 0.70 * float(rng.random()))
    else:
        alpha = 0.08 + 0.25 * float(rng.random())
        if float(rng.random()) < 0.5:
            alpha = -alpha
        b = a * (1.0 + alpha * (2.0 * t - 1.0))

    b = np.clip(b, 0.0, None)
    return normalize_nonnegative(b)


def lbfgs_schedule(a, deadline, factors):
    """Thin wrapper preserving best across multiple beta levels."""
    a = normalize_nonnegative(a)
    t_end = deadline if deadline is not None else float('inf')
    best_score = compute_score(a)
    best_a = a.copy()
    n = int(a.size)

    if n <= 5000:
        base_fun = 12000
    elif n <= 8192:
        base_fun = 9000
    elif n <= 12288:
        base_fun = 5500
    else:
        base_fun = 3500

    for fac in factors:
        if time.monotonic() + 15.0 > t_end:
            break
        beta = max(1000.0, float(n) * float(fac))
        maxfun = max(800, base_fun)
        sc, a2 = run_z_lbfgs(best_a, beta, maxiter=2000, maxfun=maxfun)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a2.copy()

    return float(best_score), best_a


def polish_candidate(a, budget_s, do_lbfgs=True, lbfgs_factors=(1, 4, 15)):
    """
    High-level polishing pipeline:
      1) spectral steering
      2) L-BFGS with increasing LSE sharpness
      3) coordinate polish
    """
    a = normalize_nonnegative(a)
    budget = max(1.0, float(budget_s))
    t0 = time.monotonic()
    end = t0 + budget

    best_score = compute_score(a)
    best_a = a.copy()

    # 1) Spectral shaping.
    spec_deadline = t0 + 0.50 * budget
    if time.monotonic() + 5.0 < end and spec_deadline > time.monotonic():
        s, a2 = spectral_steering(
            best_a,
            deadline=spec_deadline,
            max_steps=3000,
            start_gamma=0.12
        )
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_a = a2.copy()

    # 2) L-BFGS refinement.
    if do_lbfgs and time.monotonic() + 30.0 < end:
        lb_deadline = end - 0.05 * budget
        s, a2 = lbfgs_schedule(best_a, lb_deadline, lbfgs_factors)
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_a = a2.copy()

    # 3) Coordinate polish.
    if time.monotonic() + 10.0 < end:
        s, a2 = coordinate_polish(
            best_a,
            deadline=end - 2.0,
            n_sweeps=2,
            top_lags=3,
            top_count=10,
            bottom_count=10
        )
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_a = a2.copy()

    return float(best_score), best_a


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for the best non-negative sequence within the time budget.
    Returns a Python list of floats.
    """
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 25.0
    rng = np.random.default_rng(int(seed))

    # ------------------------------------------------------------------
    # Load a parent construction if available. Keep length moderate because
    # the official evaluator uses direct convolution.
    # ------------------------------------------------------------------
    best_a = None
    parent = globals().get('height_sequence_1', None)
    if parent is not None:
        try:
            p = np.asarray(parent, dtype=np.float64).ravel()
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 32 and float(p.sum()) > 1e-8:
                if p.size > 12288:
                    p = interpolate_sequence(p, 12288)
                best_a = normalize_nonnegative(p)
        except Exception:
            best_a = None

    if best_a is None:
        best_a = normalize_nonnegative(np.ones(8192, dtype=np.float64))

    best_score = compute_score(best_a)
    print(f"[init] best={best_score:.10f}, n={len(best_a)}, t={time.monotonic()-t_start:.1f}s")

    # ------------------------------------------------------------------
    # Phase 1: strengthen the given / best known construction.
    # ------------------------------------------------------------------
    alloc1 = min(360.0, 0.30 * (deadline - t_start))
    if alloc1 > 30.0:
        s, a2 = polish_candidate(
            best_a,
            alloc1,
            do_lbfgs=True,
            lbfgs_factors=(1, 4, 15, 50)
        )
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_a = a2.copy()
            print(f"[phase1] best={best_score:.10f}, n={len(best_a)}, t={time.monotonic()-t_start:.1f}s")

    # ------------------------------------------------------------------
    # Phase 2: Try a slightly longer length with the same FFT support level
    # when possible (8192 is usually cheaper than 12288).
    # ------------------------------------------------------------------
    if time.monotonic() < deadline - 260.0 and best_score > 1.5050:
        if len(best_a) <= 8192:
            long_n = 8192
        else:
            long_n = min(12288, int(len(best_a)))
        a_long = interpolate_sequence(best_a, long_n)
        sc0 = compute_score(a_long)
        if np.isfinite(sc0) and sc0 < best_score + 0.01:
            alloc2 = min(320.0, 0.25 * (deadline - time.monotonic()))
            if alloc2 > 30.0:
                s, a2 = polish_candidate(
                    a_long,
                    alloc2,
                    do_lbfgs=True,
                    lbfgs_factors=(1, 4, 15)
                )
                if np.isfinite(s) and s < best_score:
                    best_score = s
                    best_a = a2.copy()
                    print(f"[long] best={best_score:.10f}, n={len(best_a)}, t={time.monotonic()-t_start:.1f}s")

    # ------------------------------------------------------------------
    # Phase 3: Diversification / restarts from the current best.
    # ------------------------------------------------------------------
    restart_deadline = deadline - 300.0
    modes = [0, 1, 2, 3, 0, 1, 2]
    for i, mode in enumerate(modes):
        if time.monotonic() > restart_deadline:
            break
        if i == 2:
            trial = cosine_init(len(best_a), rng, variant=1)
        else:
            trial = perturb_sequence(best_a, rng, mode=mode)

        remaining = restart_deadline - time.monotonic()
        budget_i = min(110.0, max(20.0, remaining / 2.0))
        if budget_i < 20.0:
            break

        s, a2 = polish_candidate(
            trial,
            budget_i,
            do_lbfgs=True,
            lbfgs_factors=(4, 15)
        )
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_a = a2.copy()
            print(f"[restart {i}] best={best_score:.10f}, n={len(best_a)}, t={time.monotonic()-t_start:.1f}s")

    # ------------------------------------------------------------------
    # Phase 4: Final deep polish on the best sequence found so far.
    # ------------------------------------------------------------------
    remaining = deadline - time.monotonic()
    if remaining > 60.0:
        s, a2 = polish_candidate(
            best_a,
            remaining - 10.0,
            do_lbfgs=True,
            lbfgs_factors=(15, 50, 150)
        )
        if np.isfinite(s) and s < best_score:
            best_score = s
            best_a = a2.copy()
            print(f"[final] best={best_score:.10f}, n={len(best_a)}, t={time.monotonic()-t_start:.1f}s")

    # ------------------------------------------------------------------
    # Prepare output and verify with the official evaluator if available.
    # ------------------------------------------------------------------
    best_a = normalize_nonnegative(best_a)
    if len(best_a) > 20000:
        best_a = interpolate_sequence(best_a, 12288)

    out = [float(x) for x in best_a]
    if len(out) == 0 or sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = float(best_score)

    if (not np.isfinite(verified)) or verified > max(1.0, best_score + 1e-6):
        # Fallback to a known safe non-negative sequence.
        out = [0.5] * 64
        try:
            verified = float(evaluate_sequence(out))
        except Exception:
            verified = 2.0

    elapsed = time.monotonic() - t_start
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}, "
          f"n={len(out)}, elapsed={elapsed:.1f}s/{budget_s:.1f}s")
    return out
```
