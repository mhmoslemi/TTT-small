        parent_input = kwargs.pop('parent', None)
        # Execute the main optimized search
        optimal_score, best_sequence = hybrid_optimize(parent_input or uniform_candidate(1024), maxits=200)
        print("Optimal score:", optimal_score)
        print("Best sequence length:", len(best_sequence))
        result = best_sequence.tolist()
    except Exception as e:
        print("Error during hybrid optimization:", e)
        # Use fallback sequence of appropriate length
        length = 1024
        result = uniform_candidate(length).tolist()
    finally:
        # Ensure the sequence is returned even if parent wasn't utilizable
        print("Returning candidate sequence")
        assert all(any(isinstance(x, float)) for x in result), "Candidates must be floats"
        assert 0.0 <= float(result[0]) <= 1000.0, "Invalid float value in sequence"
