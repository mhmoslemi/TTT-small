As we computed: `J[k,j] = 2*a[k-j]` (with a[m]=0 for m outside [0,n-1]).

`J^T J` is n×n with `(J^T J)[i,j] = 4 * sum_k a[k-i]*a[k-j]`.

Let `autocorr[m] = sum_k a[k]*a[k+m]` (the autocorrelation of a, which is `b[m + n - 1]` for m in range... actually let me just use the convolution).

The autocorrelation `R[m] = sum_k a[k]*a[k+m]` for m = -(n-1), ..., n-1. Note `R[m] = R[-m]`.

`(J^T J)[i,j] = 4 * R[j-i]` (checking: `(J^T J)[i,j] = sum_k 2*a[k-i]*2*a[k-j] = 4*sum_k a[k-i]*a[k-j]. Let m = k-i, then = 4*sum_m a[m]*a[m+j-i] = 4*R[j-i]`.)

So `J^T J = 4 * Toeplitz(R)` where `R` is the autocorrelation of `a`.

Since `J^T J` is a symmetric Toeplitz matrix, we can use the **Levinson-Durbin recursion** or **conjugate gradient with FFT-based matvec**.

For CG: we need to compute `v_new = (J^T J) * v = 4 * Toeplitz(R) * v`.

`[Toeplitz(R) * v][i] = sum_j R[j-i] * v[j] = sum_m R[m] * v[i+m]` (substituting m=j-i).

This is a convolution! Specifically, if we define `R_ext[m] = R[m]` for m in [-(n-1), n-1] and 0 otherwise, then:
`[Toeplitz(R) * v][i] = sum_m R_ext[m] * v[i+m]`

This is the cross-correlation of `R_ext` and `v`, which can be computed via FFT.

So each CG iteration costs O(n log n) for the FFT-based matvec, plus O(n) for the vector operations.

The number of CG iterations needed for convergence depends on the condition number of `J^T J`. In practice, we might need O(sqrt(kappa)) iterations, which could be a few hundred for well-conditioned problems.

Step 2: Compute `b = a * a`.
Step 3: Compute the "correction factor": `r[k] = target[k] / b[k]` (or `target[k] - b[k]`).

The challenge is step 4: how to update `a` given that `b = a * a`?

If we think of it as a fixed-point iteration: we want `a * a = target`. Taking "square root" in the convolution algebra: `a = sqrt(target)` in the convolution algebra. The "square root" can be computed in the Fourier domain: if `A(f) = FFT(a)` and `B(f) = FFT(b) = A(f)^2`, then `A(f) = sqrt(B(f))` (choosing the correct branch).

1. Compute `B(f) = FFT(b)` = `FFT(a * a)` = `FFT(a)^2`.
2. Compute `A_new(f) = B_target(f) / A(f)` (Newton's method in the Fourier domain for the equation `A^2 = B_target`).
3. `a_new = IFFT(A_new(f))`.
4. Project to `a_new >= 0` and `sum a_new = 1`.

```python
"""
Minimize 2*n*max(self-convolution)/(sum)^2 for non-negative sequences.
Uses Fourier-domain iterative method (flattening the convolution spectrum),
LSE-smoothed Adam optimization, and Gauss-Newton polishing with CG.
Key insight: flat convolution => score approaches 1 as n grows.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a (normalized sum=1)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled - scaled.max())
    log_sum = np.log(exp_scaled.sum())
    L = c_max + (log_sum + scaled.max()) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a: (p convolved with a, shifted)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def adam_lse(a_init, lr=0.003, n_steps=200000, beta_lse=200,
             time_limit_s=None, t_start=None):
    """Adam optimizer with LSE-smoothed gradient, normalized to sum=1."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0 / n
        s = 1.0
    a = a / s

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.8, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break

        lr_t = lr / (1.0 + 0.0001 * t)
        score, grad = lse_score_grad(a, beta_lse)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e6)
        s = a.sum()
        if s > 1e-12:
            a = a / s

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def jTJ_matvec(b_conv, v):
    """Compute (J^T J) @ v using FFT. b_conv is self-convolution (length 2n-1)."""
    n = len(v)
    M_needed = 3 * n - 2
    M_fft = 1 << (M_needed - 1).bit_length()
    W = np.fft.fft(b_conv, M_fft)
    V = np.fft.fft(v, M_fft)
    C = np.fft.ifft(np.conj(W) * V).real
    start = M_fft - n + 1
    y = np.empty(n)
    y[:-1] = C[start:M_fft]
    y[-1] = C[0]
    return 4.0 * y

        p_cg = r_cg.copy()
        r_norm_sq = np.dot(r_cg, r_cg)
        if r_norm_sq < 1e-30:
            break

        for cg_it in range(cg_maxiter):
            if time.monotonic() > deadline:
                break
            Ap = matvec(p_cg)
            pAp = np.dot(p_cg, Ap)
            if pAp < 1e-30:
                break
            alpha_cg = r_norm_sq / pAp
            da += alpha_cg * p_cg
            r_cg -= alpha_cg * Ap
            r_new_sq = np.dot(r_cg, r_cg)
            if r_new_sq < cg_tol * r_norm_sq:
                break
            p_cg = r_cg + (r_new_sq / r_norm_sq) * p_cg
            r_norm_sq = r_new_sq

        # Line search
        obj_val = np.dot(residual, residual)
        best_alpha = 0.01
        best_obj = obj_val
        alphas = [0.01, 0.03, 0.1, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]
        for alpha in alphas:
            a_trial = a + alpha * da
            a_trial = np.clip(a_trial, 0, 1e6)
            s_tr = a_trial.sum()
            if s_tr < 1e-12:
                continue
            a_trial = a_trial / s_tr
            b_trial = fast_conv(a_trial)
            obj_trial = np.dot(b_trial - target, b_trial - target)
            if obj_trial < best_obj:
                best_obj = obj_trial
                best_alpha = alpha

        a = a + best_alpha * da
        a = np.clip(a, 0, 1e6)
        s = a.sum()
        if s > 1e-12:
            a = a / s

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Convergence check
        if it > 0 and abs(obj_val - best_obj) / max(obj_val, 1e-10) < 1e-10:
            break

    return best_score, best_a


def fourier_iterate(a_init, gamma=0.5, n_max_iters=10000, time_limit_s=300.0,
                    t_start=None, blend=0.0):
    """
    Fourier-domain iterative method to flatten self-convolution.
    Updates A(f) *= (T/B)^gamma where B=|A|^2, T=FFT(target).
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0 / n
        s = 1.0
    a = a / s

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

    target = np.ones(M) / M
    T = np.fft.rfft(target, M_fft)

    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_limit_s if (t_start is not None and time_limit_s is not None) else np.inf

    # Precompute FFT of initial a
    A = np.fft.rfft(a, M_fft)

    for it in range(n_max_iters):
        if time.monotonic() > deadline:
            break

        # Compute B = |A|^2
        B = A.real * A.real + A.imag * A.imag

        # Guard against zeros
        B = np.maximum(B, 1e-30)

        # Update: A *= (T/B)^gamma
        ratio = T / B
        # Handle complex ratio: use magnitude and phase
        ratio_mag = np.abs(ratio)
        ratio_phase = np.angle(ratio)
        A_mag = np.sqrt(B)
        A_phase = np.angle(A)

        # New magnitude: A_mag * ratio_mag^gamma
        new_A_mag = A_mag * (ratio_mag ** gamma)
        # New phase: A_phase + gamma * ratio_phase (for complex, this is the phase of ratio^gamma)
        new_A_phase = A_phase + gamma * ratio_phase

        A = new_A_mag * np.exp(1j * new_A_phase)

        # Enforce Hermitian symmetry (rfft already handles this)
        # Inverse FFT
        a_new = np.fft.irfft(A, M_fft)[:n]

        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-12:
            # Fallback: keep old a
            continue
        a_new = a_new / sn

        # Optional: blend with old a for stability
        if blend > 0:
            a = (1.0 - blend) * a + blend * a_new
        else:
            a = a_new

        # Update FFT
        A = np.fft.rfft(a, M_fft)

        # Periodic score check
        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def fourier_iterate_newton(a_init, n_max_iters=5000, time_limit_s=300.0,
                           t_start=None, damping=1.0):
    """
    Newton-style Fourier iteration: A_new = T / conj(A) (damped).
    This is gamma=1 in the Fourier iteration.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0 / n
        s = 1.0
    a = a / s

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

    target = np.ones(M) / M
    T = np.fft.rfft(target, M_fft)

    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_limit_s if (t_start is not None and time_limit_s is not None) else np.inf

    A = np.fft.rfft(a, M_fft)

    for it in range(n_max_iters):
        if time.monotonic() > deadline:
            break

        # Newton step: A_new = T / conj(A) (i.e., solve A^2 = T)
        Ac = np.conj(A)
        Ac = np.where(np.abs(Ac) > 1e-30, Ac, 1e-30)
        A_new = T / Ac

        # Damped update
        A = (1.0 - damping) * A + damping * A_new

        # Inverse FFT
        a_new = np.fft.irfft(A, M_fft)[:n]

        # Project
        a_new = np.maximum(a_new, 0.0)
        sn = a_new.sum()
        if sn < 1e-12:
            continue
        a_new = a_new / sn

        a = a_new
        A = np.fft.rfft(a, M_fft)

        if it % 500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def subgradient_momentum(a_init, n_steps=50000, step_size=0.01, momentum=0.9,
                         time_limit_s=120.0, t_start=None):
    """Subgradient descent with momentum on the exact max-convolution objective."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    s = a.sum()
    a = a / s

    vel = np.zeros(n)
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_limit_s if (t_start is not None and t_start is not None) else np.inf

    for t in range(n_steps):
        if time.monotonic() > deadline:
            break

        b = fast_conv(a)
        k_star = int(b.argmax())
        M_val = float(b[k_star])

        # Subgradient of max_b w.r.t. a: sg[j] = 2*a[k_star - j] (with a[m]=0)
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = 2.0 * a[idx]

        # Score gradient (with sum(a)=1 constraint via projection)
        # score = 2n * M / (sum_a)^2, with sum_a=1: grad = 2n * sg
        grad = 2.0 * n * sg

        # Momentum
        vel = momentum * vel - step_size * grad
        a = a + vel

        # Project to simplex {a >= 0, sum a = 1}
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a[:] = 1.0 / n
        else:
            a = a / s

        # Decay step size
        step_size = 0.01 / (1.0 + 0.001 * t)

        if t % 1000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def random_perturb(a, strength=0.05, seed=0):
    """Add random perturbation and re-project."""
    rng = np.random.default_rng(seed)
    n = len(a)
    noise = rng.normal(0, strength, n)
    a_new = a + noise
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-12:
        return a.copy()
    return a_new / s
