"""
Peak-focused optimization for minimizing 2n*max(self-conv)/(sum)^2.
Improvements over previous: peak-aware coordinate descent, aggressive
high-beta continuation, adaptive coordinate selection targeting peak-contributing
coordinates, and simulated annealing for escaping local minima.
"""

import numpy as np
from scipy.optimize import minimize
import time
import math


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # dL/da[j] = sum_k p[k] * a[k-j] (cross-correlation of p and a at lag j)
    N_fft = 1 << (2 * n - 1).bit_length()
    A_full = np.fft.rfft(a, N_fft)
    P_full = np.fft.rfft(p, N_fft)
    corr = np.fft.irfft(np.conj(A_full) * P_full, N_fft)[:n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (corr / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000, time_budget=None, t_start=None):
    """L-BFGS-B with LSE smoothing and time budget."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_grad(x, beta)

    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf

    def timed_obj(x):
        if time.monotonic() > deadline:
            raise Exception("timeout")
        return lse_score_grad(x, beta)

    try:
        res = minimize(
            timed_obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def golden_section_1d(a, j, n_evals=20, search_range=None):
    """Golden-section search for optimal a[j]. Returns new score."""
    n = len(a)
    a_old = a[j]
    if search_range is not None:
        lo, hi = search_range
    elif a_old < 1e-12:
        lo, hi = 0.0, 2.0 * max(a.max(), 1.0 / n)
    else:
        lo, hi = 0.0, a_old * 5.0
        if hi < 1e-10:
            hi = 3.0 * a.max()

    gr = (math.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    a[j] = c1
    f1 = compute_score(a)
    a[j] = c2
    f2 = compute_score(a)

    for _ in range(n_evals - 2):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            a[j] = c1
            f1 = compute_score(a)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            a[j] = c2
            f2 = compute_score(a)

    best_val = c1 if f1 < f2 else c2
    a[j] = best_val
    return compute_score(a)


def peak_focused_coord_descent(a, n_sweeps=5, n_evals_per_coord=18,
                               time_budget=60.0, t_start=None, top_k=80):
    """
    Coordinate descent that focuses on peak-contributing coordinates.
    At each sweep: identify peak lag, find contributing coords, optimize them.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = (t_start + time_budget) if (t_start is not None) else np.inf
    gr = (math.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c, k_star, M, S = fast_conv(a), int(fast_conv(a).argmax()), 0, 0
        c = fast_conv(a)
        k_star = int(c.argmax())
        M = float(c[k_star])
        S = float(a.sum())
        cur_score = 2.0 * n * M / (S * S)
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        # Identify coordinates contributing to the peak
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        peak_coords = []
        if je >= js:
            for j in range(js, je + 1):
                partner = k_star - j
                if 0 <= partner < n and a[j] > 1e-12 and a[partner] > 1e-12:
                    contribution = a[j] * a[partner]
                    peak_coords.append((j, contribution))

        # Sort by contribution (largest first) - these have the most impact
        peak_coords.sort(key=lambda x: -x[1])

        # Also add some random non-peak coordinates for diversity
        all_coords = set(range(js, je + 1)) if je >= js else set()
        n_random = min(10, n - len(all_coords))
        if n_random > 0:
            candidates = [j for j in range(n) if j not in all_coords]
            rand_coords = np.random.default_rng(sweep * 1000).choice(
                candidates, size=min(n_random, len(candidates)), replace=False)
            for j in rand_coords:
                peak_coords.append((int(j), 0.0))

        # Select top_k coordinates to optimize
        coords_to_opt = [j for j, _ in peak_coords[:top_k]]

        # Optimize each coordinate
        for j in coords_to_opt:
            if time.monotonic() > deadline:
                break
            sc = golden_section_1d(a, j, n_evals=n_evals_per_coord)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    final_sc = compute_score(best_a)
    if final_sc < best_score:
        best_score = final_sc

    return best_score, best_a


def adam_refine(a_init, lr=0.004, n_steps=200000, beta_lse=500,
                time_budget=None, t_start=None):
    """Adam optimizer with LSE smoothing."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf

    t = 0
    while t < n_steps:
        if time.monotonic() > deadline:
            break
        t += 1

        lr_t = lr / (1.0 + 1e-5 * t)
        score, grad = lse_score_grad(a, beta_lse)

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e6)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def continuation_lbfgs(a0, betas, time_budget, t_start, maxiter_per_stage=3000):
    """Continuation method with geometrically increasing beta."""
    a = a0.copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget

    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - 5:
            break
        remaining = deadline - time.monotonic()
        # Allocate more time to later (higher beta) stages
        frac = 0.3 + 0.7 * (i / max(1, len(betas) - 1))
        time_alloc = remaining * frac * 0.8
        time_alloc = max(time_alloc, 5.0)

        sc, a_out = run_lbfgs(a, beta, maxiter=maxiter_per_stage,
                              maxfun=30000, time_budget=time_alloc, t_start=t_start)
        if sc < best_score:
            best_score = sc
            best_a = a_out.copy()
            print(f"    Continuation beta={beta}: score={sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
        a = a_out

    return best_score, best_a


def spectral_perturbation(a, rng, strength=0.15):
    """Perturb in DFT domain for diversity."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    amp_factors = 1.0 + strength * rng.standard_normal(len(A))
    amp_factors = np.clip(amp_factors, 0.5, 1.8)
    A_new = A * amp_factors
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() < 1e-10:
        a_new = a.copy()
    else:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def antipalindromic_perturbation(a, rng, strength=0.12):
    """Add anti-palindromic component to reduce center peak."""
    n = len(a)
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    tilt = (x - center) / max(center, 1.0)
    freq = float(rng.uniform(1, 4))
    tilt += 0.3 * np.sin(2.0 * np.pi * freq * x / n)
    tilt_anti = 0.5 * (tilt - tilt[::-1])
    a_new = a + strength * tilt_anti * a.max()
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def annealing_perturbation(a, rng, temp=0.05):
    """Simulated annealing style random perturbation."""
    n = len(a)
    a_new = a.copy()
    n_changes = max(1, int(n * temp * 0.1))
    for _ in range(n_changes):
        j = int(rng.integers(0, n))
        delta = float(rng.uniform(-1, 1)) * 0.3 * a[j]
        a_new[j] = max(0.0, a_new[j] + delta)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def refine_candidate(a_init, time_budget, t_start, tag=""):

    a = a_init.copy()

    def try_update(sc, a_cand):

        if sc < best_score:

            best_a = np.clip(a_cand, 0.0, 1000.0).copy()
            return True
        return False

    # Stage 1: Adam warmup (30%)
    if time.monotonic() < deadline - 10:
        adam_time = min((deadline - time.monotonic()) * 0.3, time_budget * 0.35)
        sc, a_opt = adam_refine(a, lr=0.004, n_steps=200000, beta_lse=300,
                                time_budget=adam_time, t_start=t_start)
        if try_update(sc, a_opt):
            print(f"      [Adam] {best_score:.8f} {tag}")

    # Stage 2: Continuation L-BFGS (45%)

        if try_update(sc, a_opt):
