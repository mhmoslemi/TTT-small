"""
Minimize 2*n*max(self-convolution)/(sum)^2 for non-negative sequences.
Combines LP-based iteration with LSE-smoothed gradient descent and coordinate descent.
"""

import numpy as np
from scipy.optimize import linprog
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def cross_conv(a, b):
    n, m = len(a), len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -50, 0)
    exp_scaled = np.exp(scaled)
    L = c_max + np.log(exp_scaled.sum()) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    
    # Compute cross-correlation q[j] = sum_i a[i] * p[i+j] via FFT
    rev_a = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n-1:2*n-1]
    
    # Correction for double-counting at k=2j
    correction = np.zeros(n)
    for j in range(n):
        if 2 * j < 2 * n - 1:
            correction[j] = p[2*j] * a[j]
    
    grad_inner = 2.0 * conv_res - correction
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_inner / S2 - 2.0 * L * a / S3)
    
    return score, grad

def lp_based_step(f, M):
    n = len(f)
    # Constraint matrix: A[k, j] = f[k-j] if 0 <= k-j < n
    # Build as sparse matrix
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k-j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: max 1^T g s.t. A g <= M, g >= 0
    # linprog minimizes, so: min -1^T g s.t. A g <= M, g >= 0
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None

- For T iterations:

  - Find the most violated constraint: j* = argmin (A^T y)[j]
  - Update: y[k] *= exp(-eta * f[k-j*]) for k where f[k-j*] > 0
  - (or: y[k] += eta * f[k-j*] for the projected gradient method)

This is the multiplicative weights method for LP. It converges to the optimal solution in O(1/epsilon^2) iterations.

The optimal g for the LP will have the property that the "active" constraints (where h[k] = M) form a "hitting set" that covers all of g. In other words, for each j with g[j] > 0, there's at least one k with h[k] = M and f[k-j] > 0.

This suggests that the optimal g is supported on a "complementary" set to the active constraints.

Actually, here's another idea: instead of the full LP, I can use a "block LP" approach:
1. Divide g into blocks of size B (say B=64)
2. For each block, solve a small LP (64 variables, 2n-1 constraints)
3. Cycle through all blocks

This reduces the LP size dramatically. Each block LP has 64 variables and ~14000 constraints, which is very fast.

```python
def block_lp_step(f, M, g, block_size=64):
    """Do one pass of block LP: for each block, solve LP to optimize that block."""
    n = len(f)
    g_new = g.copy()
    
    for start in range(0, n, block_size):
        end = min(start + block_size, n)
        idx = np.arange(start, end)
        b_size = end - start
        
        # The constraint (f*g)[k] <= M becomes:
        # sum_{j in block} f[k-j] * g[j] + sum_{j not in block} f[k-j] * g[j] <= M
        # Let residual[k] = M - sum_{j not in block} f[k-j] * g[j]
        # Then: sum_{j in block} f[k-j] * g[j] <= residual[k]
        
        # Compute residual
        h = cross_conv(f, g_new)  # length 2n-1
        # subtract contribution of the block
        h_block = cross_conv(f, np.zeros(n))  # this is 0, need to do manually
        # Actually: h[k] = sum_j f[k-j]*g[j]. The block contribution is sum_{j in block} f[k-j]*g[j]
        # residual[k] = h[k] - sum_{j in block} f[k-j]*g[j_old]
        
        # For the LP, we need: sum_{j in block} f[k-j] * g_new[j] <= h[k] - sum_{j in block} f[k-j]*g[j_old] + M - h[k]
        # = M - sum_{j not in block} f[k-j]*g_new[j]
        
        # Let's just compute: for each k, the "remaining budget" after fixing all non-block variables
        # remaining[k] = M - sum_{j not in block} f[k-j] * g_new[j]
        # = M - (h[k] - sum_{j in block} f[k-j] * g_new[j])
        # But we're about to change g_new in the block, so this is circular.
        
        # Simpler: compute the contribution of non-block variables to each h[k]
        non_block_g = g_new.copy()
        non_block_g[start:end] = 0
        h_non_block = cross_conv(f, non_block_g)  # O(n log n)
        
        # LP for the block:
        # max sum(g_block) s.t. sum_{j in block} f[k-j] * g_block[j] <= M - h_non_block[k] for all k
        #                          g_block >= 0
        
        # The constraint matrix for this LP:
        # A_block[k, j_in_block] = f[k - (start + j_in_block)] if valid
        # b[k] = M - h_non_block[k]
        
        # Build A_block (sparse)
        rows = []
        cols = []
        vals = []
        for k in range(2*n-1):
            for jj in range(b_size):
                j_global = start + jj
                diff = k - j_global
                if 0 <= diff < n:
                    rows.append(k)
                    cols.append(jj)
                    vals.append(f[diff])
        
        from scipy.sparse import csr_matrix
        A_block = csr_matrix((vals, (rows, cols)), shape=(2*n-1, b_size))
        b_vec = M - h_non_block
        b_vec = np.clip(b_vec, 0, None)  # can't have negative RHS
        
        res = linprog(-np.ones(b_size), A_ub=A_block, b_ub=b_vec, 
                      bounds=[(0, None)]*b_size, method='highs',
                      options={'presolve': True})
        
        if res.success:
            g_new[start:end] = res.x
    
    return g_new
```

This is O(n/block_size * (n log n + LP_time)) per pass. For n=7000, block_size=64, that's about 110 LPs per pass, each very fast. The cross_conv is the bottleneck at O(n log n) each.

Total per pass: 110 * (n log n + small LP) ≈ 110 * 7000 * 13 ≈ 10 million ops. Very fast!

```python
"""
Minimize 2*n*max(self-convolution)/(sum)^2 for non-negative sequences.
Strategy: Block LP iteration + LSE-smoothed Adam + coordinate descent polish.
The block LP approach solves the state-of-the-art LP approximately by 
optimizing small blocks of variables at a time.
"""

import numpy as np
from scipy.sparse import csr_matrix
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Linear cross-convolution of a and b via FFT."""
    n, m = len(a), len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = np.clip(beta * (c - c_max), -50, 0)
    exp_scaled = np.exp(scaled)
    L = c_max + np.log(exp_scaled.sum()) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient via cross-correlation
    rev_a = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n-1:2*n-1]
    
    # Correction for diagonal terms
    j_idx = np.arange(n)
    two_j = 2 * j_idx
    valid = two_j < 2 * n - 1
    correction = np.zeros(n)
    correction[valid] = p[two_j[valid]] * a[valid]
    
    grad_inner = 2.0 * conv_res - correction
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_inner / S2 - 2.0 * L * a / S3)
    
    return score, grad


def block_lp_iteration(f, M, g, block_size=128, n_blocks_per_pass=None):
    """One pass of block LP optimization."""
    n = len(f)
    g_copy = g.copy()
    n_blocks = (n + block_size - 1) // block_size
    
    if n_blocks_per_pass is not None:
        n_blocks = min(n_blocks, n_blocks_per_pass)
    
    for bi in range(n_blocks):
        start = bi * block_size
        end = min(start + block_size, n)
        b_size = end - start
        
        # Compute residual: M - contribution of non-block variables
        non_block = g_copy.copy()
        non_block[start:end] = 0
        h_nb = cross_conv(f, non_block)
        
        # LP: max sum(g_block) s.t. A_block @ g_block <= M - h_nb, g_block >= 0
        # A_block[k, jj] = f[k - (start + jj)] if 0 <= k - (start+jj) < n
        
        # Build sparse matrix
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            k_min = k - (end - 1)  # min j_global for which k - j_global >= 0
            k_max = k - start      # max j_global for which k - j_global < n
            j_start = max(0, k_min - start)
            j_end = min(b_size, k_max - start + 1)
            for jj in range(j_start, j_end):
                jj_global = start + jj
                diff = k - jj_global
                if 0 <= diff < n:
                    rows.append(k)
                    cols.append(jj)
                    vals.append(f[diff])
        
        A_blk = csr_matrix((vals, (rows, cols)), shape=(2*n-1, b_size))
        b_vec = np.maximum(M - h_nb, 1e-15)
        
        try:
            res = linprog(-np.ones(b_size), A_ub=A_blk, b_ub=b_vec,
                          bounds=[(0, None)] * b_size, method='highs',
                          options={'presolve': True, 'time_limit': 5})
            if res.success:
                g_copy[start:end] = res.x
        except:
            pass
    
    return g_copy


def lp_based_optimization(f0, n_lp_iters=30, block_size=128, time_limit=None, t_start=None):
    """Run the LP-based iteration for several rounds."""
    n = len(f0)
    f = f0.copy()
    target_sum = float(f.sum())
    
    def check_time():
        if time_limit is not None and t_start is not None:
            return time.monotonic() - t_start > time_limit
        return False
    
    for it in range(n_lp_iters):
        if check_time():
            break
        
        M = float(fast_conv(f).max())
        
        # Initialize g as current f (warm start idea)
        g = f.copy()
        
        # Do one pass of block LP
        g = block_lp_iteration(f, M, g, block_size=block_size)
        
        # Rescale g to match target_sum
        g_sum = float(g.sum())
        if g_sum < 1e-12:
            break
        g = g * (target_sum / g_sum)
        
        # Update: convex combination
        t = 0.3  # mixing parameter
        f = (1 - t) * f + t * g
        f = np.maximum(f, 0)
        
        if it % 5 == 0:
            sc = compute_score(f)
            print(f"  LP iter {it}: score={sc:.8f}")
    
    return f
