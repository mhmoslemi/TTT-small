For a sequence a of length n, zero-padded to length M (M ≥ 2n-1):
- A(k) = sum_{j=0}^{n-1} a[j] * exp(-2πi*j*k/M) for k = 0, ..., M-1
- The linear convolution c = a * a has length 2n-1
- The circular convolution of the zero-padded a with itself has length M, and equals c padded with zeros (if M ≥ 2n-1)
- So |A(k)|^2 = C(k) where C = FFT(c_padded) is the DFT of the zero-padded convolution

We want c[k] = 1/(2n-1) for k = 0, ..., 2n-2 (perfectly flat).
Then C(k) = sum_{j=0}^{2n-2} (1/(2n-1)) * exp(-2πi*j*k/M) = (1/(2n-1)) * D_M(k)

where D_M(k) = sum_{j=0}^{2n-2} exp(-2πi*j*k/M) is the Dirichlet kernel.

So we want |A(k)|^2 = (1/(2n-1)) * D_M(k) for all k.
Thus |A(k)| = sqrt(D_M(k) / (2n-1)).

Note: D_M(k) = exp(-2πi*(2n-2)*k/(2M)) * sin(π*(2n-1)*k/M) / sin(π*k/M)

So |D_M(k)| = |sin(π*(2n-1)*k/M) / sin(π*k/M)|

And |A(k)| should be proportional to sqrt(|sin(π*(2n-1)*k/M) / sin(π*k/M)|).

Note that |A(0)| = sum(a) = 1 (after normalization), and D_M(0) = 2n-1. So |A(0)| = 1 = sqrt((2n-1)/(2n-1)) = 1. ✓

    Frequency-domain error reduction targeting the ideal amplitude spectrum.
    """
    n = len(a)
    M = 1 << (2*n - 1).bit_length()  # FFT size
    a = np.maximum(a, 0).astype(np.float64)
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Compute target amplitude
    k = np.arange(M)
    # |D(k)| = |sin(pi*(2n-1)*k/M) / sin(pi*k/M)|
    with np.errstate(divide='ignore', invalid='ignore'):
        target = np.abs(np.sin(np.pi * (2*n-1) * k / M) / np.sin(np.pi * k / M + 1e-30))
    target[0] = 2*n - 1  # D(0) = 2n-1
    target_amp = np.sqrt(target / (2*n - 1))  # |A(k)| target
    # Normalize so that target_amp[0] = 1
    target_amp = target_amp / target_amp[0]
    
    for it in range(n_iters):
        # Compute FFT
        a_pad = np.zeros(M)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)  # Only real frequencies: 0 to M/2
        
        # Compute |A| for real frequencies
        absA = np.abs(A)
        
        # Target for real frequencies
        k_r = np.arange(len(A))  # 0 to M//2
        target_r = np.abs(np.sin(np.pi * (2*n-1) * k_r / M) / np.sin(np.pi * k_r / M + 1e-30))
        target_r[0] = 2*n - 1
        target_r = np.sqrt(target_r / (2*n - 1))
        target_r = target_r / target_r[0]
        
        # Soft amplitude replacement
        with np.errstate(divide='ignore', invalid='ignore'):
            ratio = np.where(absA > 1e-10, target_r / absA, 1.0)
        ratio = np.clip(ratio, 0.1, 10.0)
        A_new = A * np.power(ratio, alpha)
        
        # Inverse FFT
        a_new_full = np.fft.irfft(A_new, M)
        a_new = a_new_full[:n]
        
        # Project onto non-negative
        a_new = np.maximum(a_new, 0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def fast_conv(a):
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2*n-1]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_obj_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled_shift = scaled - scaled.max()
    exp_s = np.exp(scaled_shift)
    log_sum = np.log(exp_s.sum()) + scaled.max() / beta
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # dL/da[i] = 2 * cross_correlation(a, p)[i]
    M2 = 1 << (3 * n).bit_length()
    P = np.fft.rfft(p, M2)
    A = np.fft.rfft(a, M2)
    corr = np.fft.irfft(np.conj(P) * A, M2)
    dL_da = 2.0 * corr[:n]
    grad = (2.0 * n / (S * S)) * dL_da - (4.0 * n * L) / (S**3)
    return score, grad

        a = a_new

def dual_subgradient(a, n_iters=2000, K=5, step_scale=0.1):
    """Dual subgradient using multiple active lags."""
    n = len(a)
    a = np.maximum(a, 0).astype(np.float64)
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        c_max = float(c.max())
        sc = 2.0 * n * c_max / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top K lags
        top_K = np.argsort(c)[::-1][:K]
        # Weights: softmax of top K conv values
        c_top = c[top_K]
        w = np.exp(c_top - c_top.max())
        w = w / w.sum()
        
        # Weighted subgradient: sg[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
        sg = np.zeros(n)
        for idx, k in enumerate(top_K):
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end+1] += w[idx] * 2.0 * a[j_start:j_end+1][::-1]
        
        # Normalize sg
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        
        # Step size
        step = step_scale / (1.0 + 0.01 * it)
        
        a = a - step * sg / (S * np.max(sg) + 1e-30)
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
    
    return best_a
