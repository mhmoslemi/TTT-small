    n = len(f)

from scipy.sparse import csr_matrix
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

    c_obj = -np.ones(n)
    
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)

    if res.success:
        return res.x
    return None

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient: d/d a_j of sum_k p_k * b_k where b_k = sum_i a_i a_{k-i}
    # db_k / da_j = sum_i [delta_{k-i,j} * a_i + delta_{i,j} * a_{k-i}]
    #             = a_{k-j} + a_{k-j} = 2*a_{k-j} (if 0 <= k-j < n)
    # So grad_j = 2 * sum_k p_k * a_{k-j} (where 0 <= k-j < n)
    #           = 2 * (p * a_reversed)[j + n - 1] ... let me be more careful
    
    # sum_k p_k * a_{k-j} where k ranges over [0, 2n-2] and k-j ranges over [0, n-1]
    # Let i = k-j, so k = i+j. i ranges over [0, n-1], k ranges over [j, j+2n-2] ∩ [0, 2n-2]
    # = sum_i p_{i+j} * a_i for valid i,j
    # This is a cross-correlation of p and a, evaluated at lag j.
    # cross[j] = sum_i a[i] * p[i+j] for i = 0, ..., n-1-j (if j >= 0)
    
    # Actually, let me just use the FFT approach:
    # grad_j = 2 * sum_k p[k] * a[k-j] (where 0 <= k-j < n, 0 <= k < 2n-1)
    # Let i = k-j: grad_j = 2 * sum_i a[i] * p[i+j] (where 0 <= i < n, 0 <= i+j < 2n-1)
    # This is the cross-correlation of a and p at lag j.
    
    # Using FFT: cross-correlation of a (length n) and p (length 2n-1)
    M2 = 3 * n - 2  # length of a * p
    M2_fft = 1 << (M2 - 1).bit_length()
    A_corr = np.fft.rfft(a, M2_fft)
    P_corr = np.fft.rfft(p, M2_fft)
    # cross[j] = sum_i a[i] * p[i+j] = (a * p_flipped)[j + n - 1] ... 
    # Actually, let me use a different approach
    # cross[j] = sum_{i=0}^{n-1} a[i] p[i+j] 
    # This is the "anti-convolution" or cross-correlation
    
    # np.correlate(p, a, mode='full')[n-1 + j] gives sum_i a[i] p[i+j]
    # But that's O(n^2). Let me use FFT:
    # cross[j] = sum_i a[i] p[i+j] = (a * flip(p)) evaluated at position (n-1) + j
    # where flip(p) is p reversed
    
    # Actually, a simpler approach:
    # grad = 2 * (convolution of p and reversed-a, shifted)
    # (p * a_rev)[k] = sum_i p[i] a_rev[k-i] = sum_i p[i] a[n-1-(k-i)] = sum_i p[i] a[n-1-k+i]
    # Let j = n-1-k: = sum_i p[i] a[j+i] = cross[j]
    # So cross[j] = (p * a_rev)[n-1-j]
    # And grad[j] = 2 * cross[j] = 2 * (p * a_rev)[n-1-j]
    
    a_rev = a[::-1]
    M2_fft = 1 << (3 * n - 2 - 1).bit_length()
    P_fft = np.fft.rfft(p, M2_fft)
    AR_fft = np.fft.rfft(a_rev, M2_fft)
    conv_pa = np.fft.irfft(P_fft * AR_fft, M2_fft)
    # cross[j] = conv_pa[n-1-j]
    cross = conv_pa[n-1::-1][:n]  # cross[0] = conv_pa[n-1], cross[1] = conv_pa[n-2], ...
    # Wait: cross[j] = (p * a_rev)[n-1-j]
    # For j=0: cross[0] = (p*a_rev)[n-1]
    # For j=n-1: cross[n-1] = (p*a_rev)[0]
    # So cross = conv_pa[n-1::-1][:n] ... that gives [conv_pa[n-1], conv_pa[n-2], ..., conv_pa[0]]
    # Which is cross[0]=conv_pa[n-1], cross[1]=conv_pa[n-2], ..., cross[n-1]=conv_pa[0]. Correct!
    
    grad_conv = 2.0 * cross
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_conv / S2 - 2.0 * L / S3)
    
    return score, grad

def lse_optimize(a_init, beta, n_steps=50000, lr=0.005, time_limit_s=None, t_start=None):
    """Adam optimizer with LSE-smoothed gradient."""
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0, None).copy()
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0/n
    else:
        a = a / s
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        
        lr_t = lr / (1.0 + 1e-5 * t)
        score_lse, grad = lse_score_grad(a, beta)
        
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 1e-12:
            a = a / s
        
        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a

def coord_descent(a, top_k=50, n_sweeps=5, time_limit_s=None, t_start=None):
    """Coordinate descent on the exact (non-smooth) objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_limit_s) if (time_limit_s is not None and t_start is not None) else np.inf
    
    for sweep in range(n_sweps):
        if time.monotonic() > deadline:
            break
        
        # Find the k that gives the max of the convolution
        conv = fast_conv(a)
        S = a.sum()
        M = conv.max()
        k_star = int(conv.argmax())
        
        # The subgradient w.r.t. a_j is:
        # d(M)/d(a_j) = 2 * a[k_star - j] (if 0 <= k_star - j < n)
        # d(score)/d(a_j) = 2n * (subgrad_j / S^2 - 2*M / S^3)
        
        active_js = []
        for j in range(min(k_star, n-1) + 1):
            if k_star - j >= 0 and k_star - j < n:
                active_js.append(j)
        active_js = [j for j in active_js if a[j] > 1e-15 or True]
        # Actually all j in [max(0, k_star-n+1), min(k_star, n-1)] are active
        j_min = max(0, k_star - n + 1)
        j_max = min(k_star, n - 1)
        active_js = list(range(j_min, j_max + 1))
        
        # Sort by |gradient| descending
        S2 = S * S
        S3 = S2 * S
        grads = np.zeros(n)
        for j in active_js:
            subgrad = 2.0 * a[k_star - j]
            grads[j] = 2.0 * n * (subgrad / S2 - 2.0 * M / S3)
        
        # Sort active coordinates by |grad|
        order = np.argsort(np.abs(grads))[::-1][:top_k]
        
        for j in order:
            if time.monotonic() > deadline:
                break
            if j < 0 or j >= n:
                continue
            a_old = a[j]
            if grads[j] > 1e-10:
                # Decrease a[j]
                candidates = a_old * np.array([0.5, 0.7, 0.85, 0.95, 0.99])
            elif grads[j] < -1e-10:
                # Increase a[j]
                candidates = np.concatenate([[0.0], a_old * np.array([1.3, 1.5, 2.0, 3.0])])
            else:
                continue
            
            candidates = np.clip(candidates, 0.0, 1000.0)
            best_x = a_old
            best_sc = compute_score(a)
            for x in candidates:
                a[j] = x
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x
    
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a

"""
Minimize 2*n*max(self-convolution)/(sum)^2 for non-negative sequences.
Strategy: LP-based multiplicative update (primary), LSE-smoothed Adam polish,
coordinate descent. The LP finds optimal improvement direction g maximizing
sum(g) subject to cross-correlation (a*g)[k] <= M for all k.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def solve_lp_full(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(a[k - i])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True, 'time_limit': 120})
    if res.status == 0:
        result = res.x.copy()
        result[result < 1e-15] = 0.0
        return result
    return None


def solve_lp_sampled(a, M, n_samples=300):
    n = len(a)
    c_obj = -np.ones(n)
    conv = fast_conv(a)
    top_k = np.argsort(conv)[::-1][:n_samples]
    rows = []
    cols = []
    vals = []
    for idx, k_val in enumerate(top_k):
        k = int(k_val)
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            rows.append(idx)
            cols.append(i)
            vals.append(a[k - i])
    A = csr_matrix((vals, (rows, cols)), shape=(len(top_k), n))
    b = np.full(len(top_k), M)
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.status == 0:
        result = res.x.copy()
        result[result < 1e-15] = 0.0
        return result
    return None


def multiplicative_update(a_init, n_iters, t_init, time_limit_s, t_start, use_full, n_samples):
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0, None).copy()
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / s
    best_a = a.copy()
    best_score = compute_score(a)
    t = t_init
    no_improve_count = 0
    for it in range(n_iters):
        elapsed = time.monotonic() - t_start
        if elapsed > time_limit_s:
            break
        conv = fast_conv(a)
        M = float(conv.max())
        current_score = 2.0 * n * M / (a.sum() ** 2)
        t0 = time.monotonic()
        if use_full:
            g = solve_lp_full(a, M)
        else:
            g = solve_lp_sampled(a, M, n_samples)
        if g is None and use_full:
            g = solve_lp_sampled(a, M, n_samples)
        lp_time = time.monotonic() - t0
        if g is None:
            break
        g_sum = float(g.sum())
        if g_sum < 1e-10:
            break
        g = g / g_sum
        a_new = None
        for step in [t, 0.7*t, 0.5*t, 0.3*t, 0.1*t, 0.05*t]:
            a_trial = (1.0 - step) * a + step * g
            a_trial = np.clip(a_trial, 0.0, None)
            st = a_trial.sum()
            if st < 1e-12:
                continue
            a_trial = a_trial / st
            sc = compute_score(a_trial)
            if sc < current_score * (1.0 - 1e-10):
                a_new = a_trial
                break
        if a_new is None:
            for step in [0.01, 0.005, 0.001]:
                a_trial = (1.0 - step) * a + step * g
                a_trial = np.clip(a_trial, 0.0, None)
                st = a_trial.sum()
                if st < 1e-12:
                    continue
                a_trial = a_trial / st
                sc = compute_score(a_trial)
                if sc < current_score * (1.0 - 1e-10):
                    a_new = a_trial
                    break
        if a_new is None:
            no_improve_count += 1
            t = max(0.05, t * 0.7)
            if no_improve_count > 20:
                break
            continue
        no_improve_count = 0
        a = a_new
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            if it % 10 == 0 or it < 5:
                print(f"    MU iter {it}: score={sc:.8f} LP={lp_time:.1f}s cum={time.monotonic()-t_start:.0f}s")
    print(f"    MU done: best_score={best_score:.8f} iters={it} time={time.monotonic()-t_start:.0f}s")
    return best_score, best_a


def lse_grad_func(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    max_scaled = scaled.max()
    exp_scaled = np.exp(scaled - max_scaled)
    log_sum = max_scaled + np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    a_rev = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    P_fft = np.fft.rfft(p, M2_fft)
    AR_fft = np.fft.rfft(a_rev, M2_fft)
    conv_val = np.fft.irfft(P_fft * AR_fft, M2_fft)
    cross = conv_val[n-1:2*n-1]
    grad_conv = 2.0 * cross
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_conv / S2 - 2.0 * L / S3)
    return score, grad


def adam_lse(a_init, beta, n_steps, lr, time_limit_s, t_start):
    n = len(a_init)
    a = np.clip(a_init.astype(np.float64), 0, None).copy()
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / s
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    best_a = a.copy()
    best_score = compute_score(a)
    for t in range(1, n_steps + 1):
        if time.monotonic() - t_start > time_limit_s:
            break
        lr_t = lr / (1.0 + 1e-5 * t)
        _, grad = lse_grad_func(a, beta)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 1e-12:
            a = a / s
        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def coord_descent(a_init, top_k, n_sweeps, time_limit_s, t_start):
    n = len(a_init)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        conv = fast_conv(a)
        S = float(a.sum())
        M = float(conv.max())
        k_star = int(conv.argmax())
        j_min = max(0, k_star - n + 1)
        j_max = min(k_star, n - 1)
        S2 = S * S
        S3 = S2 * S
        active_js = list(range(j_min, j_max + 1))
        if len(active_js) > top_k:
            S2 = S * S
            S3 = S2 * S
            grads = np.zeros(len(active_js))
            for idx, j in enumerate(active_js):
                subgrad = 2.0 * a[k_star - j]
                grads[idx] = abs(2.0 * n * (subgrad / S2 - 2.0 * M / S3))
            top_idx = np.argsort(grads)[::-1][:top_k]
            active_js = [active_js[i] for i in top_idx]
        for j in active_js:
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            subgrad = 2.0 * a[k_star - j]
            gval = 2.0 * n * (subgrad / S2 - 2.0 * M / S3)
            if gval > 1e-10:
                candidates = a_old * np.array([0.3, 0.5, 0.7, 0.85, 0.95, 0.99])
            elif gval < -1e-10:
                candidates = np.concatenate([[0.0], a_old * np.array([1.3, 1.7, 2.2, 3.0])])
            else:
                continue
            candidates = np.clip(candidates, 0.0, 1000.0)
            best_x = a_old
            best_sc = compute_score(a)
            for x in candidates:
                a[j] = x
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a[j] = best_x
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a


def try_update(best_score, best_a, sc, a_arr):
    if sc < best_score:
        return sc, np.clip(a_arr, 0.0, 1000.0).copy()
    return best_score, best_a
