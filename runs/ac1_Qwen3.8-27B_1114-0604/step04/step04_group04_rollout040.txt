```
def cap_refill(b, M_target, water=False):
   total=float(b.sum()); L=len(b); avg=total/L
   if M_target <= avg: return None
   bnew=b.astype(np.float64).copy()
   # in case sum slightly diff, target? We want preserve total.
   for _ in range(3):
      over=bnew > M_target
      if not over.any(): break
      count=int(np.count_nonzero(over))
      removed=float(bnew[over].sum()) - M_target*count
      bnew[over]=M_target
      if removed <= 1e-12: continue
      if water:
         order=np.argsort(bnew)
         slack_total = M_target* order.size - bnew.sum()
         if slack_total < removed: break
         cap = M_target - bnew[order]
         cum=np.cumsum(cap)
         idx=np.searchsorted(cum, removed)
         if idx < order.size:
             bnew[order[:idx]] += cap[:idx]
             rem = removed - (cum[idx-1] if idx>0 else 0.0)
             if rem > 0: bnew[order[idx]] += min(rem, cap[idx])
         else:
             bnew[order] += cap
      else:
         slack = M_target - bnew
         ss = float(slack.sum())
         if ss <= 1e-12: break
         bnew += removed * slack / ss
   # enforce bounds and total
   bnew = np.clip(bnew, 0.0, M_target)
   s=float(bnew.sum())
   if s <= 1e-12: return None
   bnew *= total/s
   # final clip with small redistribution
   over=bnew > M_target
   if over.any():
       # repeat proportional
       for _ in range(2):
          removed=...; bnew[over]=M_target; slack=...; bnew += removed*slack/ss
   bnew = np.clip(bnew, 0.0, M_target)
   return bnew
```

```
def recover_sqrt(a,b_target):
    n=len(a); L=2*n-1; M=next_pow2(L)
    A=np.fft.fft(a, M).astype(np.complex128)
    B=np.fft.fft(b_target, M).astype(np.complex128)
    S=np.sqrt(B)
    # align signs
    if np.real(S[0]*np.conj(A[0])) < 0: S[0] = -S[0]
    if M % 2 == 0:
        if np.real(S[M//2]*np.conj(A[M//2])) < 0: S[M//2] = -S[M//2]
    if M > 2:
        pos=np.arange(1, M//2)
        flip=np.real(S[pos]*np.conj(A[pos])) < 0
        S[pos]=np.where(flip, -S[pos], S[pos])
    S=enforce_hermitian(S)
    out=np.fft.ifft(S)
```

```
def recover_taylor(a,b_target, eta=1.0):
    n=len(a); L=2*n-1; M=next_pow2(L)
    A=np.fft.fft(a,M).astype(np.complex128)
```

```
                delta=min(delta, max(min_delta, (bestsc-floor)*0.05))
```

```python
"""
Spectral flat-square refinement, multiplied by LP best-response alternating
and softmax/LSE polish for nonnegative sequences minimizing
2*n*max(conv(a,a))/(sum(a))^2.

The optimizer works mainly in the scale-invariant normalization
sum(a)^2 = 2*n, so the objective equals max(conv(a,a)).  It alternates:
  1) project the current square coefficients to a lower peak in coefficient
     space, invert by a phase-aligned polynomial square root, and clip/normalize;
  2) LP best-response mixing;
  3) LSE-smoothed L-BFGS in a softmax/tanh parametrization;
  4) peak-subgradient polishing and short exploratory length changes.
"""

import numpy as np
import time
import math
from scipy.optimize import minimize

GLOBAL_TARGET_SCORE = 1.5025
SOFT_C = 40.0


def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def uniform_sequence(n):
    return np.full(n, math.sqrt(2.0 / max(1, n)), dtype=np.float64)


def clean_sequence(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    t = math.sqrt(2.0 * n)
    if s < 1e-12:
        return np.full(n, t / n, dtype=np.float64)
    return a * (t / s)


def resample_sequence(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return uniform_sequence(n_new)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if a.size == 1:
        out = np.full(n_new, float(a[0]), dtype=np.float64)
    else:
        x_old = np.linspace(0.0, 1.0, a.size)
        x_new = np.linspace(0.0, 1.0, n_new)
        out = np.interp(x_new, x_old, a)
    return clean_sequence(out)


def self_conv(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    M = next_pow2(L)
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:L]


def cross_conv(a, b):
    n1 = len(a)
    n2 = len(b)
    L = n1 + n2 - 1
    M = next_pow2(L)
    A = np.fft.rfft(a, M)
    B = np.fft.rfft(b, M)
    return np.fft.irfft(A * B, M)[:L]


def corr_weights_single(a, w):
    # h_i = sum_k w_k * a_{k-i}, valid indices only.
    n = len(a)
    m = len(w)
    if m <= 0:
        return np.zeros(n, dtype=np.float64)
    M = next_pow2(n + m - 1)
    ar = np.fft.rfft(a[::-1], M)
    wr = np.fft.rfft(w, M)
    c = np.fft.irfft(ar * wr, M)
    # We need the slice corresponding to valid i=0..n-1.
    start = n - 1
    stop = start + n
    if stop > c.size:
        stop = c.size
    if start >= c.size:
        return np.zeros(n, dtype=np.float64)
    return c[start:stop]


def grad_self_weights(a, w):
    # d/d a of sum_k w_k * (a*a)_k, including the factor 2 for identical terms.
    return 2.0 * corr_weights_single(a, w)


def score_general(a):
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    if n == 0:
        return np.inf
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 1e-12:
        return np.inf
    b = self_conv(a)
    m = float(np.max(b))
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m / (s * s))


def enforce_hermitian(Z):
    Z = np.asarray(Z, dtype=np.complex128)
    M = len(Z)
    Z[0] = Z[0].real
    if M % 2 == 0:
        Z[M // 2] = Z[M // 2].real
    if M > 2:
        pos = Z[1:M // 2].copy()
        Z[1:M // 2] = pos
        Z[M // 2 + 1:M] = np.conj(pos[::-1])
    return Z


def cap_refill(b, M_target, water=False):
    """
    Preserve sum(b), force max(b) <= M_target by capping peaks and redistributing
    the removed mass to entries with slack.  water=True fills lowest entries first.
    """
    b = np.asarray(b, dtype=np.float64).ravel()
    L = b.size
    if L == 0:
        return None
    total = float(b.sum())
    if total <= 0.0:
        return None
    avg = total / float(L)
    M_target = max(float(M_target), avg)
    if M_target <= avg * (1.0 + 1e-12):
        return None

    bnew = b.copy()
    bnew = np.clip(bnew, 0.0, None)

    for _ in range(3):
        over = bnew > M_target
        if not over.any():
            break
        cnt = int(np.count_nonzero(over))
        removed = float(bnew[over].sum()) - M_target * cnt
        bnew[over] = M_target
        if removed <= 1e-12:
            continue

        if water:
            order = np.argsort(bnew)
            cap = M_target - bnew[order]
            csum = float(cap.sum())
            if csum <= 1e-12:
                break
            if csum < removed:
                bnew[order] += removed * cap / csum
            else:
                cum = np.cumsum(cap)
                idx = int(np.searchsorted(cum, removed, side='left'))
                if idx < order.size:
                    if idx > 0:
                        bnew[order[:idx]] += cap[:idx]
                    rem = removed - (float(cum[idx - 1]) if idx > 0 else 0.0)
                    if rem > 0.0:
                        add = min(rem, float(cap[idx]))
                        if add > 0.0:
                            bnew[order[idx]] += add
                else:
                    bnew[order] += cap
        else:
            slack = M_target - bnew
            ss = float(slack.sum())
            if ss <= 1e-12:
                break
            bnew += removed * slack / ss

    bnew = np.clip(bnew, 0.0, M_target)
    s = float(bnew.sum())
    if s <= 1e-12:
        return None
    bnew *= total / s

    for _ in range(3):
        over = bnew > M_target
        if not over.any():
            break
        cnt = int(np.count_nonzero(over))
        removed = float(bnew[over].sum()) - M_target * cnt
        bnew[over] = M_target
        if removed <= 1e-12:
            continue
        slack = M_target - bnew
        ss = float(slack.sum())
        if ss <= 1e-12:
            break
        bnew += removed * slack / ss

    bnew = np.clip(bnew, 0.0, M_target)

    # Final tiny sum correction.
    s = float(bnew.sum())
    if s > 0.0:
        diff = total - s
        if diff > 0.0:
            slack = M_target - bnew
            ss = float(slack.sum())
            if ss > 1e-12:
                bnew += diff * slack / ss
        elif diff < 0.0:
            rem = -diff
            idx = np.argsort(bnew)[::-1]
            for ii in idx:
                take = min(rem, float(bnew[ii]))
                if take <= 0.0:
                    continue
                bnew[ii] -= take
                rem -= take
                if rem <= 1e-15:
                    break
    bnew = np.clip(bnew, 0.0, M_target)
    return bnew


def affine_b(b, M_target):
    b = np.asarray(b, dtype=np.float64).ravel()
    L = b.size
    if L == 0:
        return None
    total = float(b.sum())
    if total <= 0.0:
        return None
    mu = total / float(L)
    M_target = float(M_target)
    if M_target <= mu:
        return None
    mb = float(b.max())
    mn = float(b.min())
    if mb <= mu * (1.0 + 1e-14):
        return b.copy()
    lam = (M_target - mu) / (mb - mu)
    if mn < mu:
        lam2 = mu / max(mu - mn, 1e-15)
        lam = min(lam, lam2)
    lam = max(0.0, min(1.0, float(lam)))
    bnew = mu + lam * (b - mu)
    bnew = np.clip(bnew, 0.0, None)
    s = float(bnew.sum())
    if s > 0.0:
        bnew *= total / s
    if bnew.max() <= M_target * (1.0 + 1e-9):
        return bnew
    return cap_refill(b, M_target, water=False)


def make_b_targets(b, M_target, coarse=False):
    out = []
    cr = cap_refill(b, M_target, water=coarse)
    if cr is not None:
        out.append(cr)
    if coarse:
        af = affine_b(b, M_target)
        if af is not None and float(af.max()) <= M_target * (1.0 + 1e-9):
            out.append(af)
    return out


def recover_sqrt(a, b_target):
    n = len(a)
    L = 2 * n - 1
    M = next_pow2(L)
    A = np.fft.fft(a, M).astype(np.complex128)
    B = np.fft.fft(b_target, M).astype(np.complex128)
    S = np.sqrt(B)

    # Choose the square-root branch closest to the current polynomial spectrum.
    if np.real(S[0] * np.conj(A[0])) < 0.0:
        S[0] = -S[0]
    if M % 2 == 0:
        if np.real(S[M // 2] * np.conj(A[M // 2])) < 0.0:
            S[M // 2] = -S[M // 2]
    if M > 2:
        pos = np.arange(1, M // 2)
        flip = np.real(S[pos] * np.conj(A[pos])) < 0.0
        S[pos] = np.where(flip, -S[pos], S[pos])

    S = enforce_hermitian(S)
    out = np.fft.ifft(S)
    return out.real[:n]


def recover_taylor(a, b_target, eta=1.0):
    n = len(a)
    L = 2 * n - 1
    M = next_pow2(L)
    A = np.fft.fft(a, M).astype(np.complex128)
    Bnew = np.fft.fft(b_target, M).astype(np.complex128)
    Bold = A * A
    delta = Bnew - Bold
    dA = np.zeros_like(A)
    mask = np.abs(A) > 1e-10
    dA[mask] = delta[mask] / (2.0 * A[mask])
    S = A + eta * dA
    S = enforce_hermitian(S)
    out = np.fft.ifft(S)
    return out.real[:n]


def spectral_step(a, M_target, coarse=False):
    a = clean_sequence(a)
    b = self_conv(a)
    b = np.maximum(b, 0.0)
    cur = float(np.max(b))
    if not np.isfinite(cur):
        return a, np.inf
    if M_target >= cur:
        return a, cur

    targets = make_b_targets(b, M_target, coarse)
    if len(targets) == 0:
        return a, cur

    best = a
    bestsc = cur
    tot = float(b.sum())
    shape_alphas = [1.0]
    if coarse:
        shape_alphas.append(0.5)

    for bt in targets:
        bt = np.asarray(bt, dtype=np.float64).ravel()
        s = float(bt.sum())
        if s <= 0.0:
            continue
        bt = bt * (tot / s)

        for alpha in shape_alphas:
            btshape = (1.0 - alpha) * b + alpha * bt
            btshape = np.maximum(btshape, 0.0)
            s = float(btshape.sum())
            if s <= 0.0:
                continue
            btshape = btshape * (tot / s)

            try:
                a1 = clean_sequence(recover_sqrt(a, btshape))
                sc = score_general(a1)
                if np.isfinite(sc) and sc < bestsc:
                    best = a1
                    bestsc = float(sc)

                if np.isfinite(sc):
                    for mixal in (0.5, 0.25):
                        am = (1.0 - mixal) * a + mixal * a1
                        am = clean_sequence(am)
                        scm = score_general(am)
                        if np.isfinite(scm) and scm < bestsc:
                            best = am
                            bestsc = float(scm)

                a2 = clean_sequence(recover_taylor(a, btshape, 1.0))
                sc2 = score_general(a2)
                if np.isfinite(sc2) and sc2 < bestsc:
                    best = a2
                    bestsc = float(sc2)
            except Exception:
                pass

    return best, float(bestsc)


def spectral_polish(a, time_limit, global_target=GLOBAL_TARGET_SCORE,
                    min_delta=1e-6, max_delta=0.05, verbose=False):
    a = clean_sequence(a)
    best = a.copy()
    bestsc = score_general(a)
    if not np.isfinite(bestsc):
        return a, np.inf

    n = len(a)
    floor = (2.0 * n) / max(1.0, (2.0 * n - 1.0)) + 1e-12
    floor = max(floor, 1.0)
    if global_target < floor:
        global_target = floor

    gap = bestsc - max(floor, global_target)
    delta = min(max_delta, max(1e-4, 0.1 * gap if gap > 0.0 else 1e-4))
    fail = 0
    cycle = 0
    improvements = 0

    while time.monotonic() < time_limit:
        if bestsc <= floor + 1e-11:
            break

        if bestsc > global_target + 1e-8:
            if cycle % 2 == 0:
                target = max(floor, global_target)
            else:
                target = max(floor, bestsc - delta)
        else:
            target = max(floor, bestsc - delta)

        if target <= floor + 1e-12:
            break

        coarse = (bestsc - target) > 0.015
        try:
            a1, sc1 = spectral_step(a, target, coarse=coarse)
        except Exception:
            a1, sc1 = a, bestsc

        if np.isfinite(sc1) and sc1 < bestsc - 1e-12:
            best = a1
            bestsc = float(sc1)
            fail = 0
            improvements += 1
            if bestsc < global_target:
                delta = min(delta, max(min_delta, (bestsc - floor) * 0.05))
            else:
                delta = min(max_delta, max(delta * 1.25, min_delta))
            if verbose and improvements % 25 == 0:
                print(f"spectral polish: n={n}, score={bestsc:.9f}, "
                      f"delta={delta:.3e}, t={time.monotonic()-time_limit:.0f}/remaining")
        else:
            fail += 1
            delta = max(min_delta, delta * 0.6)
            if bestsc > global_target and target == global_target and fail > 2:
                # Avoid hammering an unattainable hard global target; force cautious steps.
                delta = max(min_delta, 0.1 * max(1e-6, bestsc - global_target))
            if fail > 25 or delta <= min_delta * 1.01:
                break

        cycle += 1

    return best, float(bestsc)


def soft_lse_value_grad(x, beta):
    x = np.asarray(x, dtype=np.float64).ravel()
    n = len(x)
    xc = x - np.mean(x)
    t = np.tanh(xc / SOFT_C)
    y = SOFT_C * t

    ym = y - np.max(y)
    e = np.exp(ym)
    se = float(e.sum())
    if se <= 1e-300 or not np.isfinite(se):
        return 1e12, np.zeros_like(x)

    p = e / se
    tsum = math.sqrt(2.0 * n)
    a = tsum * p
    b = self_conv(a)
    b = np.maximum(b, 0.0)
    mb = float(np.max(b))
    z = beta * (b - mb)
    ez = np.exp(z)
    ezsum = float(ez.sum())
    if ezsum <= 1e-300 or not np.isfinite(ezsum):
        w = np.zeros(len(b))
        w[int(np.argmax(b))] = 1.0
    else:
        w = ez / ezsum

    H = mb + math.log(ezsum) / beta
    ga = grad_self_weights(a, w)
    gp = ga * tsum
    dot = float(np.dot(p, gp))
    gy = p * (gp - dot)

    # Chain through y = SOFT_C * tanh((x - mean(x))/SOFT_C).
    grad_u = gy * SOFT_C * (1.0 - t * t)
    grad_x = grad_u - np.mean(grad_u)

    if not np.all(np.isfinite(grad_x)):
        grad_x = np.nan_to_num(grad_x, nan=0.0, posinf=0.0, neginf=0.0)
    if not np.isfinite(H):
        H = 1e12
    return float(H), grad_x


def logx_to_seq(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    xc = x - np.mean(x)
    t = np.tanh(xc / SOFT_C)
    y = SOFT_C * t
    ym = y - np.max(y)
    e = np.exp(ym)
    se = float(e.sum())
    if se <= 1e-300 or not np.isfinite(se):
        return uniform_sequence(len(x))
    p = e / se
    return clean_sequence(p)


def seq_to_logx(a):
    a = clean_sequence(a)
    n = len(a)
    p = a / math.sqrt(2.0 * n)
    p = np.clip(p, 1e-15, 1e6)
    y = np.log(p)
    y = y - np.mean(y)
    yc = np.clip(y / SOFT_C, -0.999999, 0.999999)
    x = SOFT_C * np.arctanh(yc)
    return x


def run_soft_lbfgs(a, beta, maxfun=8000, x0=None):
    a = clean_sequence(a)
    n = len(a)
    if x0 is None:
        x0 = seq_to_logx(a)
    x0 = np.asarray(x0, dtype=np.float64).ravel()
    x0 = x0 - np.mean(x0)

    options = {
        'maxiter': max(100, maxfun // 4),
        'maxfun': maxfun,
        'ftol': 1e-15,
        'gtol': 1e-12,
        'maxls': 30
    }
    try:
        res = minimize(soft_lse_value_grad, x0, args=(beta,), jac=True,
                       method='L-BFGS-B', options=options)
        x1 = np.asarray(res.x, dtype=np.float64)
    except Exception:
        x1 = x0

    if not np.all(np.isfinite(x1)):
        x1 = x0
    out = logx_to_seq(x1)
    sc = score_general(out)
    return sc, out, x1


def lp_solve_fista(a, M, n_iters=1200, beta=1.0):
    a = clean_sequence(a)
    n = len(a)
    x = a.copy()
    u = x.copy()
    t_mom = 1.0

    Mself = next_pow2(2 * n - 1)
    A_spec = np.fft.rfft(a, Mself)
    sigma2 = float(np.max(np.abs(A_spec) ** 2)) + 1e-12
    step = 1.0 / max(1e-12, beta * sigma2)

    M2 = next_pow2(3 * n - 2)
    rev = np.fft.rfft(a[::-1], M2)

    for it in range(n_iters):
        c = cross_conv(a, x)
        c = c[:2 * n - 1]
        z = beta * (c - M)
        zmax = float(np.max(z))
        ez = np.exp(z - zmax)
        ezsum = float(ez.sum())
        if ezsum <= 1e-300 or not np.isfinite(ezsum):
            p = np.ones(len(c), dtype=np.float64) / float(len(c))
        else:
            p = ez / ezsum

        wr = np.fft.rfft(p, M2)
        h = np.fft.irfft(rev * wr, M2)
        h = h[n - 1:2 * n - 1]
        grad = h - 1.0

        v = u - step * grad
        xnew = np.maximum(v, 0.0)
        xnew = np.minimum(xnew, 1000.0)

        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = xnew + (t_mom - 1.0) / t_new * (xnew - x)
        u = np.maximum(u, 0.0)
        u = np.minimum(u, 1000.0)

        x = xnew
        t_mom = t_new

    return x


def lp_alternating(a, max_outer=10, fista_iters=1200, beta=1.0, time_limit=None):
    a = clean_sequence(a)
    best = a.copy()
    bestsc = score_general(a)

    for _ in range(max_outer):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        M = score_general(a)
        if not np.isfinite(M):
            break

        try:
            x = lp_solve_fista(a, M, fista_iters=fista_iters, beta=beta)
        except Exception:
            break
        x = clean_sequence(x)

        improved = False
        for al in (1.0, 0.75, 0.5, 0.25):
            cand = (1.0 - al) * a + al * x
            cand = clean_sequence(cand)
            sc = score_general(cand)
            if np.isfinite(sc) and sc < bestsc - 1e-12:
                best = cand
                bestsc = float(sc)
                a = cand
                improved = True
        if not improved:
            break

    return best, float(bestsc)


def subgradient_polish(a, n_iters=800, time_limit=None):
    a = clean_sequence(a)
    best = a.copy()
    bestsc = score_general(a)
    n = len(a)
    k = max(8, min(200, int(0.01 * n)))

    for _ in range(n_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        b = self_conv(a)
        b = np.maximum(b, 0.0)
        m = float(np.max(b))
        if not np.isfinite(m):
            break

        idx = np.argsort(b)[-k:]
        w = np.zeros(b.size, dtype=np.float64)
        w[idx] = 1.0 / float(len(idx))

        g = grad_self_weights(a, w)
        g = g - np.mean(g)
        gn = float(np.linalg.norm(g))
        if gn < 1e-12:
            break
        d = g / gn

        improved = False
        for st in (0.03, 0.012, 0.004, 0.0012):
            cand = a - st * d
            cand = clean_sequence(cand)
            sc = score_general(cand)
            if np.isfinite(sc) and sc < bestsc - 1e-12:
                best = cand
                bestsc = float(sc)
                a = cand
                improved = True
                break
        if not improved:
            break

    return best, float(bestsc)


def get_parent():
    try:
        name = 'height_sequence_1'
        if name in globals() and globals()[name] is not None:
            p = np.asarray(globals()[name], dtype=np.float64).ravel()
            if p.size >= 8:
                p = np.where(np.isfinite(p), p, 0.0)
                p = np.clip(p, 0.0, 1000.0)
                if float(p.sum()) > 0.0:
                    return p
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t0 = time.monotonic()
    budget = max(120.0, float(budget_s))
    deadline = t0 + budget - 45.0

    parent = get_parent()
    has_parent = parent is not None
    if has_parent:
        if len(parent) > 20000:
            parent = resample_sequence(parent, 12000)
        best_a = clean_sequence(parent)
    else:
        best_a = uniform_sequence(6962)

    best_score = score_general(best_a)
    fallback_a = best_a.copy()
    fallback_score = float(best_score)
    if not np.isfinite(best_score):
        best_a = uniform_sequence(6962)
        best_score = score_general(best_a)
        fallback_a = best_a.copy()
        fallback_score = float(best_score)

    print(f"initial: n={len(best_a)}, score={best_score:.9f}, parent={has_parent}")

    # Main spectral refinement.
    span = max(60.0, deadline - t0)
    if has_parent:
        phase1_deadline = t0 + 0.30 * span
    else:
        phase1_deadline = t0 + 0.45 * span
    if time.monotonic() < phase1_deadline:
        a1, sc1 = spectral_polish(best_a, phase1_deadline, verbose=True)
        if np.isfinite(sc1) and sc1 < best_score - 1e-12:
            best_score = float(sc1)
            best_a = a1
            print(f"after spectral phase 1: n={len(best_a)}, score={best_score:.9f}")

    # LP best-response alternating.
    phase2_deadline = t0 + 0.50 * span
    if time.monotonic() < phase2_deadline:
        fista_iters = 1000 if len(best_a) <= 10000 else 700
        a2, sc2 = lp_alternating(best_a, max_outer=10, fista_iters=fista_iters,
                                 beta=0.5, time_limit=phase2_deadline - 5.0)
        if np.isfinite(sc2) and sc2 < best_score - 1e-12:
            best_score = float(sc2)
            best_a = a2
            print(f"after LP beta=0.5: n={len(best_a)}, score={best_score:.9f}")

    phase2b_deadline = t0 + 0.68 * span
    if time.monotonic() < phase2b_deadline:
        a3, sc3 = lp_alternating(best_a, max_outer=8, fista_iters=fista_iters,
                                 beta=2.0, time_limit=phase2b_deadline - 5.0)
        if np.isfinite(sc3) and sc3 < best_score - 1e-12:
            best_score = float(sc3)
            best_a = a3
            print(f"after LP beta=2.0: n={len(best_a)}, score={best_score:.9f}")

    # Softmax/LSE polish with warm starts.
    phase3_deadline = t0 + 0.86 * span
    if time.monotonic() < phase3_deadline:
        n = len(best_a)
        time_scale = max(0.25, min(1.0, budget / 2400.0))
        if n <= 8000:
            base_fun = 8000
        elif n <= 15000:
            base_fun = 5000
        else:
            base_fun = 3000
        base_fun = int(base_fun * time_scale)
        x0 = None
        for beta in (50.0, 300.0, 1500.0, 8000.0):
            if time.monotonic() > phase3_deadline - 45.0:
                break
            a4, sc4, x0 = run_soft_lbfgs(best_a, beta, maxfun=base_fun, x0=x0)
            if np.isfinite(sc4) and sc4 < best_score - 1e-12:
                best_score = float(sc4)
                best_a = a4
                x0 = seq_to_logx(a4)
                print(f"after soft LSE beta={beta:g}: n={len(best_a)}, score={best_score:.9f}")

    # Final spectral polish.
    phase4_deadline = t0 + 0.94 * span
    if time.monotonic() < phase4_deadline:
        a5, sc5 = spectral_polish(best_a, phase4_deadline, verbose=True)
        if np.isfinite(sc5) and sc5 < best_score - 1e-12:
            best_score = float(sc5)
            best_a = a5
            print(f"after final spectral: n={len(best_a)}, score={best_score:.9f}")

    # Short subgradient polish.
    if time.monotonic() < deadline - 30.0:
        a6, sc6 = subgradient_polish(best_a, n_iters=400, time_limit=deadline - 20.0)
        if np.isfinite(sc6) and sc6 < best_score - 1e-12:
            best_score = float(sc6)
            best_a = a6
            print(f"after subgradient: n={len(best_a)}, score={best_score:.9f}")

    # Very short exploratory resampling to nearby lengths.
    if time.monotonic() < deadline - 25.0:
        n_best = len(best_a)
        candidates = []
        for nn in (n_best - 512 if n_best > 1024 else None,
                   n_best + 512 if n_best < 15000 else None,
                   8192, 10240, 5000):
            if nn is None:
                continue
            nn = int(nn)
            if 1024 <= nn <= 15000 and nn not in candidates:
                candidates.append(nn)
        for nn in candidates[:3]:
            if time.monotonic() > deadline - 15.0:
                break
            tr = time.monotonic() + max(6.0, min(15.0, deadline - time.monotonic() - 5.0))
            ar = resample_sequence(best_a, nn)
            ar, scr = spectral_polish(ar, tr, verbose=False)
            if np.isfinite(scr) and scr < best_score - 1e-12:
                best_score = float(scr)
                best_a = ar
                print(f"exploration length {nn}: score={best_score:.9f}")

    # Final verification and safe fallback.
    arr = np.asarray(best_a, dtype=np.float64).ravel()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    out = [float(x) for x in arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = None
    try:
        verified = float(evaluate_sequence(out))
    except Exception as e:
        print("evaluate_sequence verification failed:", repr(e))

    if verified is None or not np.isfinite(verified):
        farr = np.asarray(fallback_a, dtype=np.float64).ravel()
        farr = np.where(np.isfinite(farr), farr, 0.0)
        farr = np.clip(farr, 0.0, 1000.0)
        fout = [float(x) for x in farr]
        if sum(fout) < 0.01:
            fout = [0.5] * 64
        try:
            vfin = float(evaluate_sequence(fout))
        except Exception:
            vfin = np.inf
        if (verified is None or not np.isfinite(verified)) or np.isfinite(vfin) and vfin < verified:
            out = fout
            verified = vfin

    # Last-ditch fallback if everything else is nonfinite.
    if not np.isfinite(verified):
        out = [float(x) for x in uniform_sequence(64)]
        try:
            verified = float(evaluate_sequence(out))
        except Exception:
            verified = np.inf

    print(f"FINAL internal={best_score:.9f}, verified={verified:.9f}, n={len(out)}")
    return out
```
