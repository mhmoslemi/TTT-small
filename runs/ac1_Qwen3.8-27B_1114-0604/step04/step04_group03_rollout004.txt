def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_obj_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    shifted = scaled - scaled.max()
    exp_s = np.exp(shifted)
    L = c_max + (shifted.max() + np.log(exp_s.sum())) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5

        return lse_obj_grad(x, beta)
    
    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def fienup_error_reduction(a, n_iters=500, mu_range=(0.5, 1.5)):
    """
    Fienup-style alternating projection for flat convolution.
    Alternates between:
    - Frequency domain: adjust spectrum to flatten convolution
    - Time domain: enforce non-negativity and normalization
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        # Compute current convolution
        a_pad = np.zeros(M, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        conv = np.fft.irfft(A * A, M)[:2 * n - 1]
        
        # Target: flat convolution
        target = 1.0 / (2 * n - 1)
        
        # Error in convolution domain
        err = conv - target
        
        # Try to cancel error in frequency domain
        err_pad = np.zeros(M, dtype=np.float64)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_pad, M)
        
        # DA = err_spec / (2 * A)
        with np.errstate(divide='ignore', invalid='ignore'):
            DA = err_spec / (2.0 * A)
        DA[np.abs(A) < 1e-10] = 0.0
        
        # Adaptive relaxation parameter
        mu = 0.3 + 0.7 * (it / max(1, n_iters * 0.3))
        A_new = A - mu * DA
        
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def active_set_coord_descent(a, n_sweeps=10, n_top_peaks=5, n_top_contrib=30):
    """
    Coordinate descent that targets the top peaks of the convolution.
    For each active peak, redistributes mass from high contributors to low contributors.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        # Find active peaks (top K)
        peak_indices = np.argsort(c)[::-1][:n_top_peaks]
        
        for k_star in peak_indices:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end < j_start:
                continue
            
            seg = a[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            
            # Find top contributors to this peak
            n_active = min(n_top_contrib, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_active]
            top_global = j_start + top_local
            
            # Try reducing top contributors and increasing bottom ones
            for j_idx in top_global:
                old_val = a[j_idx]
                best_val = old_val
                best_local_sc = best_sc
                for factor in [0.5, 0.7, 0.85, 0.95]:
                    new_val = old_val * factor
                    a[j_idx] = new_val
                    sc = compute_score(a)
                    if sc < best_local_sc:
                        best_local_sc = sc
                        best_val = new_val
                a[j_idx] = best_val
            
            S = a.sum()
            if S > 1e-12:
                a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

def subgradient_descent(a, n_iters=5000):
    """Projected subgradient method with adaptive step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

        M = float(c.max())
        k_star = int(np.argmax(c))
        S = float(a.sum())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of 2n*max(conv)/S^2
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        
        grad = (2.0 * n) * (sg / (S * S) - 2.0 * M * np.ones(n) / (S * S * S))
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break
        
        # Polyak step size with lower bound 1.0
        step = (sc - 1.0001) / (gn * gn + 1e-30)
        step = min(step, 0.1)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()

def spectral_shape_opt(a, n_iters=200):
    """
    Optimize the spectral shape |A(f)|^2 to minimize peak autocorrelation.
    Uses water-filling in the frequency domain.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    a = a / a.sum()
    
    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        power = A * np.conj(A).real
        # Actually for real input, A is complex, power = |A|^2 = A.real^2 + A.imag^2
        A_real = A.real
        A_imag = A.imag
        power = A_real**2 + A_imag**2
        
        conv = np.fft.irfft(A * A, M)[:2*n-1]
        conv_max = float(conv.max())
        
        # Flatten the power spectrum (reduce peaks in autocorrelation)
        # The autocorrelation peak is related to the variance of the power spectrum
        # Reduce high-power frequencies and boost low-power ones
        power_mean = power.mean()
        # Soft clipping of power
        clip_level = power_mean * 1.5
        power_clipped = np.minimum(power, clip_level)
        # Renormalize to preserve total energy (which relates to sum(a)^2)
        if power_clipped.sum() > 0:
            scale = np.sqrt(power.sum() / power_clipped.sum())
            power_clipped = power_clipped * scale
        
        # Reconstruct A with modified power (preserving phase)
        A_new_mag = np.sqrt(power_clipped)
        A_new_phase = np.angle(A)
        A_new = A_new_mag * (np.cos(A_new_phase) + 1j * np.sin(A_new_phase))
        
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a

        if time.monotonic() > phase1_deadline:
            break
        print(f"  n={n}")
        inits = build_initializations(n, rng, parent)
        
        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            t0 = time.monotonic()
            a_er = fienup_error_reduction(a0, n_iters=300)
            sc = compute_score(a_er)
            report(sc, a_er, f"Fienup n={n} i={i}")
            print(f"    init {i}: score={sc:.6f} ({time.monotonic()-t0:.1f}s)")
    
    print(f"After Phase 1: best = {best_score:.8f}")
    
    # ============================================================
    # Phase 2: LSE L-BFGS (30% time)
    # ============================================================
    phase2_deadline = t_start + 0.70 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 2: LSE L-BFGS ---")
    
    n_best = len(best_a)
    beta_schedule = [100, 300, 1000, 3000, 10000]
    
    for beta in beta_schedule:
        if time.monotonic() > phase2_deadline:
            break
        print(f"  L-BFGS beta={beta}")
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=8000, maxfun=200000)
        report(sc, a_opt, f"LBFGS b={beta}")
        print(f"    score={sc:.8f} ({time.monotonic()-t0:.1f}s)")
    
    # Try from other starting points at different lengths
    if time.monotonic() < phase2_deadline:
        for n in lengths[:3]:
            if time.monotonic() > phase2_deadline:
                break
            if len(best_a) != n:
                a_interp = np.interp(
                    np.linspace(0, 1, n),
                    np.linspace(0, 1, len(best_a)),
                    best_a
                )
            else:
                a_interp = best_a.copy()
            for beta in [500, 2000]:
                if time.monotonic() > phase2_deadline:
                    break
                sc, a_opt = run_lbfgs(a_interp, beta, maxiter=5000, maxfun=100000)
                report(sc, a_opt, f"LBFGS alt n={n} b={beta}")
    
    print(f"After Phase 2: best = {best_score:.8f}")
    
    # ============================================================
    # Phase 3: Active Set Coord Descent + Perturbation (20% time)
    # ============================================================
    phase3_deadline = t_start + 0.90 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 3: Local Search ---")
    
    n_best = len(best_a)
    
    for trial in range(30):
        if time.monotonic() > phase3_deadline:
            break
        
        # Generate perturbation
        strategy = trial % 4
        if strategy == 0:
            # Gaussian perturbation
            scale = 0.02 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Targeted peak redistribution
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            a_pert = best_a.copy()
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            seg = a_pert[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            top_count = max(10, n_best // 30)
            top_local = np.argsort(contrib)[::-1][:top_count]
            bottom_local = np.argsort(contrib)[:top_count]
            red = a_pert[top_local + j_start].sum() * 0.05
            a_pert[top_local + j_start] *= 0.95
            a_pert[bottom_local + j_start] += red / top_count
            S = a_pert.sum()
            if S > 0:
                a_pert = a_pert / S
        elif strategy == 2:
            # Sinusoidal perturbation
            freq = 1.0 + 20.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.02 + 0.04 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Fienup restart
            a_pert = fienup_error_reduction(
                best_a * (1 + 0.02 * rng.standard_normal(n_best)),
                n_iters=100
            )
        
        # Polish with L-BFGS
        sc, a_opt = run_lbfgs(a_pert, 1000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, f"local#{trial} s={strategy}")
    
    print(f"After Phase 3: best = {best_score:.8f}")
    
    # ============================================================
    # Phase 4: Final Polish (10% time)
    # ============================================================
    print(f"\n--- Phase 4: Final Polish ---")
    remaining = deadline - time.monotonic()
    
    if remaining > 60:
        a_sp = subgradient_descent(best_a, n_iters=int(min(10000, remaining / 0.05)))
        sc = compute_score(a_sp)
        report(sc, a_sp, "subgrad")
    
    if remaining > 30:
        sc, a_opt = run_lbfgs(best_a, 20000, maxiter=10000, maxfun=200000)
        report(sc, a_opt, "final LBFGS b=20000")
    
    if remaining > 20:
        a_er = fienup_error_reduction(best_a, n_iters=200)
        sc = compute_score(a_er)
        report(sc, a_er, "final Fienup")
    
    if remaining > 15 and best_score > 1.504:
        sc, a_opt = run_lbfgs(best_a, 50000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "final LBFGS b=50000")
    
    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
