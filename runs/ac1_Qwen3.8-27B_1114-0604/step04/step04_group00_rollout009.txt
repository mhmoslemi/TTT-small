"""
Continuation L-BFGS + Peak-directed coordinate descent + Fienup hybrid optimizer.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key innovations:
1. LSE continuation method with finely-tuned beta schedule
2. Efficient gradient computation via FFT-based cross-correlation
3. Peak-directed paired coordinate descent for fine polishing
4. Mirror symmetrization exploitation
5. Multi-length exploration around parent
6. Adaptive stopping criteria

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_objective_and_grad(x, beta):
    """
    LSE-smoothed objective and its gradient for L-BFGS.
    Objective: 2n * LSE_beta(conv(x,x)) / (sum(x))^2
    where LSE_beta(c) = (1/beta) * log(sum(exp(beta*(c - c_max))))
    """
    n = len(x)
    c = fast_conv(x)
    S = float(np.sum(x))
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(np.max(c))
    scaled = beta * (c - c_max)
    # Numerical stability: subtract max (already done via c_max)
    exp_scaled = np.exp(np.maximum(scaled, -50))
    sum_exp = float(np.sum(exp_scaled))
    if sum_exp < 1e-30:
        sum_exp = 1e-30
    # LSE value
    L = c_max + np.log(sum_exp) / beta
    score = 2.0 * n * L / (S * S)

    # Weights for LSE gradient
    w = exp_scaled / sum_exp  # length 2n-1

    # Gradient of LSE w.r.t. x:
    # (dLSE/dx)_j = (1/beta) * sum_{m=0}^{n-1} w[j+m] * x[m]
    # This is cross-correlation of w with x, evaluated at lags 0..n-1
    M_corr = 1 << (3 * n - 2 - 1).bit_length()
    W = np.fft.rfft(w, M_corr)
    X = np.fft.rfft(x, M_corr)
    dLSE_dx = np.fft.irfft(np.conj(W) * X, M_corr)[:n] / beta

    # Full gradient of score
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dLSE_dx / S2 - 2.0 * L / S3)

    return score, grad


def run_lbfgs_continuation(a0, betas, maxiter_base=5000, maxfun_base=50000):
    """
    Run L-BFGS with continuation over increasing beta values.
    Each stage uses the previous result as starting point.
    Returns (best_score, best_a).
    """
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a.sum()
    if S < 1e-6:
        a[:] = 0.5
        S = 1.0
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    bounds = [(0.0, 1000.0)] * n

    for i, beta in enumerate(betas):
        # Scale maxiter and maxfun based on beta (larger beta needs more iters)
        scale = 1.0 + 0.5 * i
        res = minimize(
            lse_objective_and_grad, a.copy(), args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': int(maxiter_base * scale),
                     'maxfun': int(maxfun_base * scale), 'maxls': 50}
        )
        a_new = np.clip(res.x, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-6:
            continue
        a_new = a_new / S_new
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Keep a as is if new result is worse
            pass

    return best_sc, best_a


def peak_directed_descent(a, n_sweeps=10, n_top_peaks=20, n_top_elems=40, 
                          multipliers=None):
    """
    Coordinate descent targeting the top peaks and their most responsible elements.
    Uses multiplicative updates for stability.
    """
    if multipliers is None:
        multipliers = [0.5, 0.7, 0.85, 1.15, 1.3, 1.5]

    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        cur_sc = 2.0 * n * float(c.max()) / (S * S)

        # Get top peaks
        n_pk = min(n_top_peaks, len(c))
        peak_idx = np.argsort(c)[::-1][:n_pk]

        improved = False
        for pk in peak_idx:
            pk = int(pk)
            # Compute contribution of each element to this peak
            j_start = max(0, pk - n + 1)
            j_end = min(pk, n - 1)
            if j_end < j_start:
                continue
            seg = a[j_start:j_end + 1]
            contrib = seg * a[pk - j_start:pk - j_start + (j_end - j_start + 1)][::-1]
            # Actually: conv(a,a)(pk) = sum_j a[j]*a[pk-j]
            # Element j contributes a[j]*a[pk-j]
            # For element j in [j_start, j_end], its partner is pk-j
            partners = pk - np.arange(j_start, j_end + 1)
            contrib = a[j_start:j_end + 1] * a[partners]

            # Top contributing elements
            n_te = min(n_top_elems, len(contrib))
            top_local = np.argsort(contrib)[::-1][:n_te]
            top_global = np.arange(j_start, j_end + 1)[top_local]

            for j in top_global:
                a_old = a[j]
                if a_old < 1e-10:
                    continue
                best_local = a_old
                best_local_sc = cur_sc

                for mult in multipliers:
                    a_test = a_old * mult
                    a_test = max(0.0, min(1000.0, a_test))
                    if abs(a_test - a_old) < 1e-10:
                        continue
                    a[j] = a_test
                    S_t = a.sum()
                    if S_t < 1e-12:
                        a[j] = a_old
                        continue
                    a_norm = a / S_t
                    sc_t = compute_score(a_norm)
                    a[j] = a_old
                    if sc_t < best_local_sc:
                        best_local_sc = sc_t
                        best_local = a_test

                if best_local != a_old:
                    a[j] = best_local
                    S = a.sum()
                    if S > 1e-12:
                        a = a / S
                    cur_sc = best_local_sc
                    improved = True

        sc_final = compute_score(a)
        if sc_final < best_sc:
            best_sc = sc_final
            best_a = a.copy()

        if not improved:
            break

    return best_a


def fienup_soft(a, n_iters=200, clip_alpha=5.0):
    """
    Fienup with soft non-negativity via logistic/sigmoid function.
    a_new = a_raw * sigmoid(clip_alpha * a_raw) / sum(...)
    This is smoother than hard clipping.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Adaptive threshold
        thresh = P_mean * (1.2 + 0.3 * (1.0 - progress))
        above = P > thresh
        if not above.any():
            break

        scale = np.ones_like(P)
        strength = 0.4 + 0.4 * progress
        scale[above] = (thresh / P[above]) ** strength

        A_mod = A * scale
        a_raw = np.fft.irfft(A_mod, M_fft)[:n]

        # Soft non-negativity: a_new = a_raw * sigmoid(alpha * a_raw)
        alpha = clip_alpha * (1.0 + progress)
        # Use stable sigmoid
        pos = a_raw > 0
        a_soft = a_raw.copy()
        a_soft[a_raw < 0] = 0.0  # still clip negatives to 0 for safety
        # Apply soft weighting: elements with larger values are kept more
        if alpha > 0:
            # sigmoid(alpha * x) ≈ 1 for x >> 0, ≈ 0 for x << 0
            # We use: weight = x * sigmoid(alpha * x) / (mean of x * sigmoid(alpha * x))
            weights = np.where(a_soft > 0, a_soft * (1.0 / (1.0 + np.exp(-alpha * a_soft))), 0.0)
            w_sum = weights.sum()
            if w_sum > 1e-12:
                a_new = weights / w_sum
            else:
                a_new = a_soft
                s = a_new.sum()
                if s > 1e-12:
                    a_new = a_new / s
                else:
                    break

        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_flat_target(a, n_iters=200, target_flatness=0.0):
    """
    Fienup that targets a specific flatness in the spectrum.
    Tries to make |DFT(a)|^2 as close to uniform as possible.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        # Target: make P close to its mean
        P_target = P.mean() * (1.0 + target_flatness)

        # Compute correction: reduce bins above target, increase below
        diff = P - P_target
        # Scale correction
        corr_strength = 0.3 + 0.3 * (it / n_iters)

        # Only modify bins significantly above target
        threshold = P_target * 1.1
        above = P > threshold
        if above.sum() < 2:
            # Also try to raise bins below
            below = P < P_target * 0.9
            if below.sum() < 2:
                break
            scale = np.ones_like(P)
            scale[below] = 1.0 + corr_strength * 0.5
        else:
            scale = np.ones_like(P)
            scale[above] = 1.0 - corr_strength * (diff[above] / P_target)
            scale[above] = np.maximum(scale[above], 0.3)

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a = a_new / S_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def mirror_try(a):
    """Try the mirror-symmetrized version and return the better one."""
    a_rev = a[::-1].copy()
    a_sym = 0.5 * (a + a_rev)
    sc_orig = compute_score(a)
    sc_sym = compute_score(a_sym)
    if sc_sym < sc_orig:
        return a_sym, sc_sym
    return a, sc_orig


def paired_perturbation(a, n_trials=2000, rng=None):
    """
    Randomly perturb pairs of elements to try to reduce the peak.
    Maintains sum by adjusting pairs.
    """
    if rng is None:
        rng = np.random.default_rng(0)

    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    c = fast_conv(a)
    current_peak = float(c.max())

    for trial in range(n_trials):
        # Pick two random elements
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        if a[i] < 1e-8 or a[j] < 1e-8:
            continue

        # Try to reduce the larger one and increase the smaller
        if a[i] > a[j]:
            big, small = i, j
        else:
            big, small = j, i

        # Transfer a small amount
        delta = a[big] * rng.uniform(0.01, 0.05)
        a[big] -= delta
        a[small] += delta

        if a[big] < 0:
            a[big] = 0
            a[small] -= a[big]
            continue

        S_new = a.sum()
        if S_new < 1e-12:
            a[big] += delta
            a[small] -= delta
            continue

        a_norm = a / S_new
        sc = compute_score(a_norm)
        if sc < best_sc:
            best_sc = sc
            best_a = a_norm.copy()
        else:
            # Revert
            a[big] += delta
            a[small] -= delta

    return best_a


def subgradient_all_peaks(a, n_iters=2000, threshold_frac=0.95, step_init=0.02):
    """
    Subgradient using ALL peaks above a threshold (not just top-M).
    This gives a denser gradient signal.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    decay = 0.9999
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Use all peaks above threshold
        thresh = threshold_frac * M_val
        active = c > thresh
        if not active.any():
            break

        # Weight peaks by how much they exceed threshold
        weights = np.zeros(len(c))
        weights[active] = (c[active] - thresh) / (M_val - thresh + 1e-30)
        W_sum = weights.sum()
        if W_sum < 1e-12:
            break
        weights = weights / W_sum

        # Compute subgradient: for each active peak k, d(conv_k)/da_j = a[k-j]
        # Combined: sg_j = sum_k weights[k] * a[k-j]
        # This is a weighted cross-correlation
        M_corr = 1 << (3 * n - 2 - 1).bit_length()
        Wf = np.fft.rfft(weights, M_corr)
        Af = np.fft.rfft(a, M_corr)
        sg = np.fft.irfft(np.conj(Wf) * Af, M_corr)[:n]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Polyak step size
        step = min(step, max(0.001, (sc - sc_lb)) / (gn * gn + 1e-30))
        step = min(step, 0.05)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a

        print(f"Parent: n={len(a0)}, initial score={best_score:.8f}")

    # Determine target lengths
    n_parent = len(best_a)
    target_lengths = [n_parent]
    # Add nearby lengths for exploration
    for dl in [0, -64, 64, -128, 128, -256, 256]:
        nl = n_parent + dl
        if 1024 <= nl <= 16384 and nl not in target_lengths:
            target_lengths.append(nl)
    target_lengths.sort(key=lambda x: abs(x - n_parent))
    target_lengths = target_lengths[:5]  # Limit to 5 lengths
    print(f"Target lengths: {target_lengths}")

    # ============================================================
    # Phase 1: L-BFGS Continuation on parent (40% of time)
    # ============================================================
    phase1_end = t_start + 0.40 * (total_budget - 30.0)
    print(f"\n=== Phase 1: L-BFGS Continuation (deadline t={phase1_end-t_start:.0f}s) ===")

    n = len(best_a)
    a_work = best_a.copy()

    # First a quick Fienup to get a good starting point
    if time.monotonic() < phase1_end - 60:
        a_f = fienup_soft(a_work, n_iters=150, clip_alpha=8.0)
        sc_f = compute_score(a_f)
        update_best(sc_f, a_f, "FIENUP-soft init")
        if sc_f < best_score:
            a_work = a_f.copy()

    # L-BFGS continuation with increasing betas
    betas = [5, 20, 50, 100, 200, 500, 1000, 3000]
    t_phase1_start = time.monotonic()
    phase1_time = phase1_end - t_phase1_start - 10

    for i, beta in enumerate(betas):
        if time.monotonic() > phase1_end - 5:
            break
        # Allocate remaining time proportionally
        remaining_betas = len(betas) - i
        time_per_beta = (phase1_end - time.monotonic() - 5) / remaining_betas
        maxiter = min(8000, max(500, int(time_per_beta * 50)))
        maxfun = min(50000, max(2000, int(time_per_beta * 300)))

        print(f"  L-BFGS beta={beta} (maxiter={maxiter}, maxfun={maxfun})", flush=True)
        sc, a_opt = run_lbfgs_continuation(a_work, [beta], 
                                             maxiter_base=maxiter, maxfun_base=maxfun)
        update_best(sc, a_opt, f"LBFGS beta={beta}")
        if sc < best_score:
            a_work = a_opt.copy()

        # Try mirror
        a_sym, sc_sym = mirror_try(a_work)
        update_best(sc_sym, a_sym, f"LBFGS-beta{beta}-mirror")
        if sc_sym < best_score:
            a_work = a_sym.copy()

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: Peak-directed coordinate descent (15% of time)
    # ============================================================
    phase2_end = t_start + 0.55 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Peak-directed Descent (deadline t={phase2_end-t_start:.0f}s) ===")

    a_cd = best_a.copy()
    while time.monotonic() < phase2_end - 5:
        a_cd = peak_directed_descent(a_cd, n_sweeps=5, n_top_peaks=15, 
                                      n_top_elems=30,
                                      multipliers=[0.6, 0.8, 0.9, 1.1, 1.2, 1.4])
        sc_cd = compute_score(a_cd)
        update_best(sc_cd, a_cd, "PEAK-DESCENT")
        if sc_cd >= best_score:
            break
        a_cd = best_a.copy()  # Reset to best for next round

    # Try with different parameters
    if time.monotonic() < phase2_end - 10:
        a_cd2 = peak_directed_descent(best_a, n_sweeps=3, n_top_peaks=30,
                                       n_top_elems=50,
                                       multipliers=[0.7, 0.85, 0.95, 1.05, 1.15, 1.3])
        sc_cd2 = compute_score(a_cd2)
        update_best(sc_cd2, a_cd2, "PEAK-DESCENT-wide")

    # Mirror the coordinate descent result
    a_cd_sym, sc_cd_sym = mirror_try(a_cd)
    update_best(sc_cd_sym, a_cd_sym, "PD-mirror")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: Subgradient + Fienup (20% of time)
    # ============================================================
    phase3_end = t_start + 0.75 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Subgradient + Fienup (deadline t={phase3_end-t_start:.0f}s) ===")

    # Subgradient with all peaks
    a_sg = subgradient_all_peaks(best_a, n_iters=2000, threshold_frac=0.90, step_init=0.01)
    sc_sg = compute_score(a_sg)
    update_best(sc_sg, a_sg, "SUBGRAD-all")

    # Fienup soft on current best
    if time.monotonic() < phase3_end - 30:
        a_f2 = fienup_soft(a_sg, n_iters=200, clip_alpha=10.0)
        sc_f2 = compute_score(a_f2)
        update_best(sc_f2, a_f2, "FIENUP-soft-2")

    # Flat-target Fienup
    if time.monotonic() < phase3_end - 30:
        a_ff = fienup_flat_target(a_sg, n_iters=150, target_flatness=0.0)
        sc_ff = compute_score(a_ff)
        update_best(sc_ff, a_ff, "FIENUP-flat")

    # Fienup with different alpha
    if time.monotonic() < phase3_end - 20:
        a_f3 = fienup_soft(best_a, n_iters=250, clip_alpha=3.0)
        sc_f3 = compute_score(a_f3)
        update_best(sc_f3, a_f3, "FIENUP-soft-3")

    # L-BFGS from subgradient result
    if time.monotonic() < phase3_end - 20:
        sc_l, a_l = run_lbfgs_continuation(a_sg, [500], maxiter_base=3000, maxfun_base=30000)
        update_best(sc_l, a_l, "SG+LBFGS")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Multi-length exploration (10% of time)
    # ============================================================
    phase4_end = t_start + 0.85 * (total_budget - 30.0)
    print(f"\n=== Phase 4: Multi-length (deadline t={phase4_end-t_start:.0f}s) ===")

    for nl in target_lengths:
        if time.monotonic() > phase4_end - 10:
            break
        if nl == len(best_a):
            continue

        # Interpolate best_a to new length
        x_old = np.linspace(0, 1, len(best_a))
        x_new = np.linspace(0, 1, nl)
        a_interp = np.interp(x_new, x_old, best_a)
        a_interp = np.maximum(a_interp, 0.01)

        # Quick L-BFGS
        if time.monotonic() < phase4_end - 15:
            sc_i, a_opt_i = run_lbfgs_continuation(a_interp, [100, 500],
                                                    maxiter_base=3000, maxfun_base=30000)
            update_best(sc_i, a_opt_i, f"LEN-{nl}")

            # Quick coordinate descent
            if time.monotonic() < phase4_end - 15:
                a_cd_i = peak_directed_descent(a_opt_i, n_sweeps=3, n_top_peaks=10,
                                               n_top_elems=20)
                sc_cd_i = compute_score(a_cd_i)
                update_best(sc_cd_i, a_cd_i, f"LEN-{nl}-CD")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ============================================================
    # Phase 5: Final polishing (remaining time)
    # ============================================================
    print(f"\n=== Phase 5: Final Polishing ===")

    # Final L-BFGS with high beta
    if time.monotonic() < deadline - 30:
        sc_final, a_final = run_lbfgs_continuation(best_a, [2000, 5000],
                                                    maxiter_base=5000, maxfun_base=50000)
        update_best(sc_final, a_final, "FINAL-LBFGS")

    # Final coordinate descent
    if time.monotonic() < deadline - 20:
        a_cd_final = peak_directed_descent(best_a, n_sweeps=8, n_top_peaks=10,
                                            n_top_elems=25,
                                            multipliers=[0.85, 0.9, 0.95, 1.0, 1.05, 1.1, 1.15])
        sc_cd_final = compute_score(a_cd_final)
        update_best(sc_cd_final, a_cd_final, "FINAL-CD")

    # Mirror
    a_m, sc_m = mirror_try(best_a)
    update_best(sc_m, a_m, "FINAL-MIRROR")

    # One more L-BFGS from the very best
    if time.monotonic() < deadline - 15:
        sc_v, a_v = run_lbfgs_continuation(best_a, [3000], maxiter_base=4000, maxfun_base=40000)
        update_best(sc_v, a_v, "VERY-FINAL-LBFGS")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
