"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses frequency-domain flattening, L-BFGS with LSE smoothing,
coordinate descent on peaks, and simulated annealing for diversification.
Starts from the best of: parent sequence, uniform, and diverse inits.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed score and its exact gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum())
    L = c_max + (log_sum + scaled.max()) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad

    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        v = A * A
        conv = np.fft.irfft(v, M)[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-14:
            break

        alpha = 0.3 + 0.6 * (it / max(1, n_iters - 1))
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-14:
            break

        conv_mod = conv - 0.5 * excess - 0.3 * (conv - conv.mean())
        err = conv_mod - conv

        err_pad = np.zeros(M, dtype=np.float64)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_pad, M)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-10] = 0.0

        damp = 0.3 + 0.5 * min(1.0, it / max(1, n_iters // 2))
        A_new = A + damp * d

        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def peak_coord_descent(a, n_sweeps=10, n_top=40):
    """Coordinate descent targeting top convolution peaks."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        top_k = min(5, len(c))
        peak_idx = np.argsort(c)[::-1][:top_k]

        for k_star in peak_idx:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                continue

            seg = a[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            top_count = min(n_top, len(contrib))
            top_local = np.argsort(contrib)[::-1][:top_count]
            top_global = j_start + top_local

            for j in top_global:
                a_old_j = a[j]
                candidates = [a_old_j * f for f in [0.3, 0.5, 0.7, 0.85, 0.9, 1.1, 1.3]]
                candidates.append(0.0)
                best_val = a_old_j
                cur_sc = compute_score(a)
                saved = a_old_j
                for cv in candidates:
                    cv = max(0.0, min(1000.0, cv))
                    if abs(cv - saved) < 1e-12:
                        continue
                    a[j] = cv
                    sc = compute_score(a)
                    if sc < cur_sc:
                        cur_sc = sc
                        best_val = cv
                a[j] = best_val
                S = a.sum()
                if S > 1e-12:
                    a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def simul_anneal_polish(a, n_iters=2000, T0=0.05, T_end=0.001):
    """Simulated annealing with frequency-domain perturbations."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    cur_sc = best_sc

    rng = np.random.default_rng(12345)

    for it in range(n_iters):
        T = T0 * (T_end / T0) ** (it / n_iters)
        x = np.arange(n, dtype=np.float64) / n

        pert_type = rng.integers(0, 4)
        if pert_type == 0:
            amp = T * rng.random()
            freq = 1.0 + 20.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            a_pert = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
        elif pert_type == 1:
            scale = T * rng.random()
            mask = rng.random(n) < 0.1
            a_pert = a.copy()
            a_pert[mask] *= (1.0 + scale * rng.standard_normal(mask.sum()))
        elif pert_type == 2:
            amp = T * rng.random()
            a_pert = a * (1.0 + amp * rng.standard_normal(n))
        else:
            n_blocks = 3 + int(rng.integers(0, 10))
            block_size = max(1, n // n_blocks)
            a_pert = a.copy()
            for b in range(min(n_blocks, n // block_size)):
                s = b * block_size
                e = min(s + block_size, n)
                a_pert[s:e] *= (1.0 + T * rng.standard_normal())

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        S_pert = a_pert.sum()
        if S_pert < 1e-12:
            continue
        a_pert = a_pert / S_pert
        sc_pert = compute_score(a_pert)

        if sc_pert < cur_sc or rng.random() < np.exp(-(sc_pert - cur_sc) / max(T, 1e-10)):
            a = a_pert
            cur_sc = sc_pert
            if sc_pert < best_sc:
                best_sc = sc_pert
                best_a = a.copy()

    return best_a


def flatten_convolution(a, n_iters=500):
    """Iteratively flatten the convolution by reducing peaks and boosting valleys."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        c_min = float(c.min())
        if c_max - c_min < 1e-13:
            break

        # Target: make conv more uniform
        target = (c_max + c_min) / 2
        alpha = 0.01 + 0.04 * (it / n_iters)

        # Find the peak and reduce its contributors
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            continue

        seg = a[j_start:j_end + 1].copy()
        # Reduce large contributors, increase small ones
        seg_mean = seg.mean()
        for j in range(j_start, j_end + 1):
            local_idx = j - j_start
            if seg[local_idx] > seg_mean:
                a[j] -= alpha * (seg[local_idx] - seg_mean)
            else:
                a[j] += alpha * (seg_mean - seg[local_idx])

        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.cos(4.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 1000.0))
    # Block structure
    for nb in (4, 8, 16, 32):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses multi-phase optimization: initialization, error reduction, L-BFGS,
    coordinate descent, simulated annealing, and final polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 10.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 32:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine candidate lengths
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n in [2048, 4096, 6962, 8192, 12288, 16384]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))[:6]

    print(f"Lengths to try: {lengths}")
    print(f"Parent available: {parent is not None}")

    # Evaluate parent first
    if parent is not None:
        sc_parent = compute_score(parent)
        report(sc_parent, parent, "parent init")
        print(f"  Parent score: {sc_parent:.8f}")

    # Evaluate uniform
    n_uni = lengths[len(lengths) // 2]
    uni = np.ones(n_uni) * 0.5
    sc_uni = compute_score(uni)
    report(sc_uni, uni, "uniform")
    print(f"  Uniform score: {sc_uni:.8f}")

    # Phase 1: Error reduction from diverse inits (25% time)
    phase1_deadline = t_start + 0.25 * (float(budget_s) - 10.0)
    print(f"\n--- Phase 1: Error Reduction ---")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"  n={n}")
        inits = build_initializations(n, rng, parent)
        remaining = phase1_deadline - time.monotonic()
        max_inits = max(2, int(remaining / 10.0))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a_er = error_reduction(a0, n_iters=250)
            sc = compute_score(a_er)
            report(sc, a_er, f"ER n={n} i={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # Phase 2: L-BFGS refinement (35% time)
    phase2_deadline = t_start + 0.60 * (float(budget_s) - 10.0)
    print(f"\n--- Phase 2: L-BFGS Refinement ---")

    n_best = len(best_a)
    beta_schedule = [50, 100, 300, 1000, 3000, 8000]

    for beta in beta_schedule:
        if time.monotonic() > phase2_deadline:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=8000, maxfun=250000)
        report(sc, a_opt, f"LBFGS b={beta}")

    # Try from parent as well
    if parent is not None and time.monotonic() < phase2_deadline:
        for beta in [200, 1000, 3000]:
            if time.monotonic() > phase2_deadline:
                break
            sc, a_opt = run_lbfgs(parent, beta, maxiter=5000, maxfun=150000)
            report(sc, a_opt, f"LBFGS parent b={beta}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # Phase 3: Local search with diverse strategies (25% time)
    phase3_deadline = t_start + 0.85 * (float(budget_s) - 10.0)
    print(f"\n--- Phase 3: Local Search ---")

    n_best = len(best_a)
    n_trials = 25

    for trial in range(n_trials):
        if time.monotonic() > phase3_deadline:
            break
        strategy = trial % 5

        if strategy == 0:
            scale = 0.005 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            a_pert = error_reduction(best_a * (1 + 0.02 * rng.standard_normal(n_best)), n_iters=100)
        elif strategy == 2:
            freq = 1.0 + 15.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.04 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            a_pert = flatten_convolution(best_a, n_iters=200)
        else:
            a_pert = peak_coord_descent(best_a, n_sweeps=5, n_top=30)

        sc, a_opt = run_lbfgs(a_pert, 500, maxiter=4000, maxfun=80000)
        report(sc, a_opt, f"local#{trial} s={strategy}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # Phase 4: Simulated annealing + final polish (15% time)
    remaining = deadline - time.monotonic()
    print(f"\n--- Phase 4: SA + Polish (remaining={remaining:.0f}s) ---")

    if remaining > 60:
        n_sa = int(min(3000, remaining / 0.3))
        a_sa = simul_anneal_polish(best_a, n_iters=n_sa, T0=0.04, T_end=0.001)
        sc = compute_score(a_sa)
        report(sc, a_sa, "SA")

    if remaining > 40:
        a_cd = peak_coord_descent(best_a, n_sweeps=8, n_top=50)
        sc = compute_score(a_cd)
        report(sc, a_cd, "CD final")

    if remaining > 30:
        a_er = error_reduction(best_a, n_iters=200)
        sc = compute_score(a_er)
        report(sc, a_er, "ER final")

    if remaining > 20:
        sc, a_opt = run_lbfgs(best_a, 5000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "LBFGS final b=5000")

    if remaining > 15:
        sc, a_opt = run_lbfgs(best_a, 15000, maxiter=3000, maxfun=60000)
        report(sc, a_opt, "LBFGS final b=15000")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
