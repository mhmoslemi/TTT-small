"""
Multi-stage optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Uses adaptive L-BFGS with LSE smoothing, peak-sculpting local search,
and multi-restart with perturbations.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    N = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(x, beta):
    """LSE-smoothed score and gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clamp for numerical stability
    scaled = np.clip(scaled, -100, 100)
    exp_s = np.exp(scaled)
    exp_sum = float(exp_s.sum())
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S) if S > 0 else 1e10
    p = exp_s / exp_sum
    
    # Gradient of L w.r.t. a: 2 * correlate(p, a)
    # q[i] = sum_k p[k] * a[k-i] = conv(a, p_rev)[2n-2-i] for i=0..n-1
    # i.e., q = conv(a, p_rev)[n-1 : 2n-1]
    p_rev = p[::-1]
    M2 = n + (2 * n - 1) - 1  # = 3n-2
    N2 = 1 << (M2 - 1).bit_length()
    A_fft = np.fft.rfft(x, N2)
    P_fft = np.fft.rfft(p_rev, N2)
    conv_res = np.fft.irfft(A_fft * P_fft, N2)[:M2]
    g_L = 2.0 * conv_res[n - 1:2 * n - 1]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.maximum(a0, 0.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.maximum(res.x, 0.0)
    S = a_out.sum()
    if S < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out
