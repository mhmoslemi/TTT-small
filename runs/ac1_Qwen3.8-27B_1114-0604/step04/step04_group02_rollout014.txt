def proj_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def fast_self_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def exact_score(a):
    """Compute exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def _lisheight_arrangement(a, beta):
    """Full LSE score and gradient (for moderate beta)."""
    n = len(a)

    S = float(a.sum())

        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_vals = np.exp(np.clip(scaled, -50, 50))
    if exp_vals.sum() < 1e-30:
        exp_vals = np.zeros_like(c)
        exp_vals[c == c_max] = 1.0
    p = exp_vals / exp_vals.sum()
    L = c_max + float(np.log(exp_vals.sum())) / beta
    score = 2.0 * n * L / (S * S)
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA = np.fft.rfft(a, M2_fft)
    corr = np.fft.irfft(np.conj(PA) * AA, M2_fft)
    h = 2.0 * corr[:n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (h / S2 - 2.0 * L / S3)
    return score, grad

def proj_simplex(v, s=1.0):
    """Project v onto {x >= 0, sum(x) = s}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - s)
    if not cond.any():
        return np.full(n, s / n)
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fista_topk(a0, beta, k, n_iters, step_init=0.5, verbose=False):
    """FISTA with adaptive step on top-k LSE, projected to simplex."""
    n = len(a0)

    if S < 1e-12:
        a0 = np.ones(n) / n
        S = 1.0
    a = a0 / S
    y = a.copy()
    t = 1.0
    step = step_init
    best_a = a.copy()
    best_sc = exact_score(a)

    for it in range(n_iters):
        sc_y, grad_y = _topk_lse_obj(y, beta, k)
        if not np.isfinite(sc_y):
            break

        # Adaptive step: try current step
        x_new = proj_simplex(y - step * grad_y, 1.0)
        sc_x = exact_score(x_new)

        # Backtracking if not sufficient decrease
        attempts = 0
        while sc_x > sc_y + 1e-8 and attempts < 10:
            step *= 0.5
            x_new = proj_simplex(y - step * grad_y, 1.0)
            sc_x = exact_score(x_new)
            attempts += 1

        if attempts >= 10:
            step = max(step, 1e-4)
            x_new = proj_simplex(y - step * grad_y, 1.0)
            sc_x = exact_score(x_new)

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        a = x_new

            if verbose and (it % 500 == 0):

def nesterov_subgrad(a0, n_iters, f_lb=1.0, verbose=False):
    """Nesterov-accelerated subgradient on exact objective."""
    n = len(a0)
    S = float(a0.sum())
    if S < 1e-12:
        return a0
    a = a0 / S
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = exact_score(a)

    for it in range(n_iters):
        c = fast_self_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()
            if verbose and (it % 500 == 0):
                print(f"    NS it={it} sc={best_sc:.10f}")

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.5)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    return best_a


def targeted_coord_descent(a0, n_sweeps=3, n_top=50):
    """Targeted coordinate descent on peak contributors."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = exact_score(a)

    for sweep in range(n_sweeps):
        c = fast_self_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        n_loc = len(contrib)
        top_count = min(n_top, n_loc)
        top_local = np.argsort(contrib)[::-1][:top_count]
        bottom_local = np.argsort(contrib)[n_top:] if n_loc > n_top else np.array([], dtype=int)

        for j in top_local:
            a_old_j = a[j_start + j]
            if a_old_j < 1e-12:
                continue
            best_frac = 1.0
            best_local_sc = best_sc
            for frac in [0.2, 0.4, 0.6, 0.8]:
                reduction = a_old_j * (1.0 - frac)
                a[j_start + j] = a_old_j * frac
                if len(bottom_local) > 0 and reduction > 1e-15:
                    per = reduction / len(bottom_local)
                    for li in bottom_local:
                        a[j_start + li] += per
                S_tmp = a.sum()
                if S_tmp > 1e-12:
                    a = a / S_tmp
                sc = exact_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_frac = frac
            a[j_start + j] = a_old_j * best_frac
            if best_frac < 1.0 and len(bottom_local) > 0:
                reduction = a_old_j * (1.0 - best_frac)
                per = reduction / len(bottom_local)
                for li in bottom_local:
                    a[j_start + li] += per
            S_tmp = a.sum()
            if S_tmp > 1e-12:
                a = a / S_tmp

        sc = exact_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def make_perturbation(a, rng, scale, strategy):

    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 5.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        a_p = a * (1.0 + scale * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 7))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strategy == 3:
        # Peak-targeted: reduce top contributors to the main peak
        c = fast_self_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(10, n // 100)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - scale * rng.random())
        S_p = a_p.sum()
        if S_p > 1e-12:
            a_p = a_p / S_p
    else:
        # Random subset
        n_sel = max(10, n // 4)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 2.0 * (rng.random(n_sel) - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S_p = float(a_p.sum())
    if S_p > 1e-12:
        a_p = a_p / S_p
    return a_p


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-phase: warmup L-BFGS, escalating beta continuation, FISTA, ILS, final polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 15.0

    # Get parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and p.sum() > 0.01:
                parent = p
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def report(sc, a, label):
        nonlocal best_score, best_a
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize
    if parent is not None:
        n_work = len(parent)
        a_cur = parent.copy()
        S = float(a_cur.sum())
        if S > 1e-12:
            a_cur = a_cur / S
        sc_init = exact_score(a_cur)
        print(f"Parent: n={n_work}, score={sc_init:.10f}")
        report(sc_init, a_cur, "parent")
    else:
        n_work = 8192
        a_cur = np.ones(n_work) * 0.5
        report(exact_score(a_cur), a_cur, "uniform")

    if best_a is None:
        best_a = a_cur.copy()
        best_score = exact_score(best_a)

    print(f"\n=== Phase 1: Quick L-BFGS Warmup (10% time) ===")
    p1_end = t_start + 0.10 * total_budget

    for beta in [500, 2000]:
        if time.monotonic() > p1_end:
            break
        rem = max(10, p1_end - time.monotonic())
        mi = int(min(5000, max(2000, rem * 30)))
        mf = int(min(30000, max(10000, rem * 100)))
        sc, a_opt = run_lbfgs_full(best_a, beta, mi, mf)
        report(sc, a_opt, f"warmup B={beta}")

    print(f"  After Phase 1: {best_score:.10f}")

    print(f"\n=== Phase 2: Escalating Beta L-BFGS Continuation (35% time) ===")
    p2_end = t_start + 0.45 * total_budget

    beta_schedule = [5000, 10000, 20000, 50000, 100000, 200000, 500000, 1000000]
    for beta in beta_schedule:
        if time.monotonic() > p2_end - 30:
            break
        rem = max(20, p2_end - time.monotonic())
        mi = int(min(30000, max(5000, rem * 20)))
        mf = int(min(100000, max(20000, rem * 60)))

        # Determine top-k based on beta (higher beta -> fewer peaks matter)
        if beta <= 10000:
            k = 50
        elif beta <= 50000:
            k = 20
        elif beta <= 200000:
            k = 10
        else:
            k = 5

        sc, a_opt = run_lbfgs_topk(best_a, beta, k, mi, mf)
        report(sc, a_opt, f"cont B={beta} k={k}")

        # If time permits, also try a slightly different k
        if time.monotonic() < p2_end - 60 and k > 3:
            sc2, a_opt2 = run_lbfgs_topk(best_a, beta, max(3, k // 2), mi // 2, mf // 2)
            report(sc2, a_opt2, f"cont B={beta} k={max(3, k//2)}")

    print(f"  After Phase 2: {best_score:.10f}")

    print(f"\n=== Phase 3: FISTA Refinement (20% time) ===")
    p3_end = t_start + 0.65 * total_budget

    for beta in [50000, 100000, 500000]:
        if time.monotonic() > p3_end:
            break
        rem = max(10, p3_end - time.monotonic())
        n_it = int(min(20000, max(2000, rem * 10)))
        k = 10 if beta <= 100000 else 5
        a_f = fista_topk(best_a, beta, k, n_it, step_init=0.3)
        sc = exact_score(a_f)
        report(sc, a_f, f"FISTA B={beta} k={k}")

    print(f"  After Phase 3: {best_score:.10f}")

    print(f"\n=== Phase 4: Nesterov Subgradient (10% time) ===")
    p4_end = t_start + 0.75 * total_budget

    if time.monotonic() < p4_end:
        a_ns = nesterov_subgrad(best_a, n_iters=5000, f_lb=1.0)
        sc = exact_score(a_ns)
        report(sc, a_ns, "nesterov")

    if time.monotonic() < p4_end:
        a_ns2 = nesterov_subgrad(fista_topk(best_a, 100000, 10, 1000, step_init=0.1),
                                  n_iters=3000, f_lb=1.0)
        sc = exact_score(a_ns2)
        report(sc, a_ns2, "fista+nes")

    print(f"  After Phase 4: {best_score:.10f}")

    print(f"\n=== Phase 5: Iterated Local Search (15% time) ===")
    p5_end = t_start + 0.90 * total_budget

    n_trials = 30
    for trial in range(n_trials):
        if time.monotonic() > p5_end:
            break
        rem = max(5, p5_end - time.monotonic())
        if rem < 10:
            break

        strategy = trial % 5
        scale = 0.01 + 0.04 * rng.random()
        a_pert = make_perturbation(best_a, rng, scale, strategy)

        # Quick L-BFGS on perturbation
        beta = 20000 + int(80000 * rng.random())
        k = 10
        mi = int(min(10000, max(2000, rem * 15)))
        mf = int(min(50000, max(5000, rem * 40)))
        sc, a_opt = run_lbfgs_topk(a_pert, beta, k, mi, mf)
        report(sc, a_opt, f"ILS {trial} s={strategy}")

        # If good, try to refine further
        if sc < best_score + 0.001 and time.monotonic() < p5_end - 10:
            sc2, a_opt2 = run_lbfgs_topk(a_opt, 500000, 5,
                                          min(5000, mi // 2), min(20000, mf // 2))
            report(sc2, a_opt2, f"ILS {trial} refine")

    print(f"  After Phase 5: {best_score:.10f}")

    print(f"\n=== Phase 6: Final Polish (5% time) ===")

    # Targeted coordinate descent
    a_tcd = targeted_coord_descent(best_a, n_sweeps=3, n_top=50)
    sc = exact_score(a_tcd)
    report(sc, a_tcd, "coord_desc")

    # Final high-beta L-BFGS
    if time.monotonic() < deadline - 30:
        for beta, k in [(500000, 5), (1000000, 3), (2000000, 2)]:
            if time.monotonic() > deadline - 20:
                break
            sc, a_opt = run_lbfgs_topk(best_a, beta, k, max(2000, int(30 * (deadline - time.monotonic()))),
                                         max(10000, int(100 * (deadline - time.monotonic()))))
            report(sc, a_opt, f"final B={beta} k={k}")

    # One more FISTA pass
    if time.monotonic() < deadline - 20:
        a_f = fista_topk(best_a, 1000000, 3,
                          int(max(1000, 5 * (deadline - time.monotonic()))), step_init=0.1)
        sc = exact_score(a_f)
        report(sc, a_f, "final FISTA")

    # Final verification and output
    out = np.clip(best_a, 0.0, 1000.0)
    out_list = [float(x) for x in out]
    if sum(out_list) < 0.01:
        out_list = [0.5] * 64

    verified = float(evaluate_sequence(out_list))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out_list)}, Time: {elapsed_total:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out_list
