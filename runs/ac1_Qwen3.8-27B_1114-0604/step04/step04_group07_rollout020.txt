def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_cross_conv(a, b):
    """Linear cross-convolution via FFT."""
    n_a, n_b = len(a), len(b)
    M = n_a + n_b - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

    # R[j] = IFFT(FFT(a) * conj(FFT(p)))[j] (circular, need zero-padding for linear)
    
    # What we want:
    # T1[j] = sum_i a[i] * p[i+j] = R[-j] (where R is cross-correlation of a and p)
    # T2[j] = sum_i a[i] * p[j-i] = R[j]
    
    # So grad_L[j] = R[j] + R[-j]
    
    # Compute R via FFT:
    M_corr = 2 * n - 1
    M_corr_fft = 1 << (M_corr - 1).bit_length()
    A_fft = np.fft.rfft(a, M_corr_fft)
    P_fft = np.fft.rfft(p, M_corr_fft)
    R = np.fft.irfft(A_fft * np.conj(P_fft), M_corr_fft)
    # R is circular cross-correlation, but with zero-padding it's linear
    # R[j] = sum_i a[i] * p[(i-j) mod M_corr_fft]
    # With zero-padding, for j in [0, M_corr_fft-1]:
    # R[j] = sum_{i=j}^{j+n-1} a[i-j] * p[i] ... 
    # Actually with the standard definition:
    # (a ⊛ p)[k] = sum_n a[n] * p[n-k] (for linear cross-correlation)
    # Via FFT: IFFT(FFT(a) * conj(FFT(p))) gives circular cross-correlation
    # With sufficient zero-padding, it equals linear cross-correlation
    
    # The result R has length M_corr_fft
    # R[k] for k=0..M_corr-1 gives the linear cross-correlation
    # R[0] = sum a[i]*p[i] (zero lag)
    # R[k] for k>0: sum_i a[i+k]*p[i] = sum_i a[i]*p[i-k] (advance)
    # R[k] for k>M_corr: gives negative lags (wrap around in circular)
    
    # Actually, the standard result is:
    # If we zero-pad a and p to length L >= 2n-1, then:
    # c = IFFT(FFT(a, L) * conj(FFT(p, L)))
    # c[k] for k=0..L-1 gives:
    #   c[k] = sum_i a[i] * p[(i-k) mod L]
    # For k=0: sum a[i]*p[i]
    # For k=1: sum a[i]*p[i-1] (with wrap) = sum_i a[i]*p[i-1]
    #   = a[1]*p[0] + a[2]*p[1] + ... + a[n-1]*p[n-2] + a[0]*p[L-1] + ...
    # With zero-padding, a[0] is the only "wrap" term if L >= 2n-1
    #   = sum_{i=1}^{n-1} a[i]*p[i-1] (the wrap terms are zero)
    
    # Hmm, this is the "delay" convention. Let me use a different approach.
    
    # Let me just compute grad_L directly:
    # grad_L[j] = sum_k p[k] * (a[k-j] + a[j-k]) where indices are valid
    # = sum_{k=j}^{j+n-1} p[k] * a[k-j] + sum_{k=0}^{j} p[k] * a[j-k]
    #   (first sum: k-j ranges from 0 to n-1)
    #   (second sum: j-k ranges from 0 to n-1, so k ranges from 0 to j)
    
    # First sum: sum_i p[i+j] * a[i] for i=0..n-1 (j+i < 2n-1)
    # Second sum: sum_i p[j-i] * a[i] for i=0..j (and j-i < 2n-1, which is always true)
    
    # These can be computed as:
    # First sum = cross-correlation of a and p at "advance" j
    # Second sum = cross-correlation of a and p at "delay" j
    
    # Using the FFT cross-correlation result:
    # c = IFFT(FFT(a, L) * conj(FFT(p, L))) with L >= 2n-1
    # c[k] = sum_i a[i+k] * p[i] (for k = 0, 1, ..., L-1, with a out of range = 0)
    # Wait, I need to be careful about conventions.
    
    # Let me use numpy's approach:
    # np.correlate(a, p, mode='full')[k] = sum_n a[n+k] * p[n] (for the appropriate range)
    # Actually numpy's correlate uses: 
    # np.correlate(a, v, 'full')[k] = sum_n a[n] * conj(v[n-(k-(len(v)-1))])
    
    # This is getting too complicated. Let me just use a direct but efficient method.
    
    # Alternative: compute the gradient via a single FFT operation
    # d(a*a)[k]/da[j] = a[k-j] + a[j-k]
    # grad[j] = sum_k p[k] * (a[k-j] + a[j-k])
    # 
    # Let's define: for fixed j, sum_k p[k]*a[k-j] where 0<=k-j<n
    # = sum_{i=0}^{n-1} p[i+j]*a[i] (substituting i=k-j)
    # This is the dot product of a[0:n] and p[j:j+n]
    # = (a circulant correlation with p) at shift j
    
    # The most efficient way for large n: use FFT
    # Define q = p (length 2n-1, zero-padded to M_fft)
    # Define a_padded = a (length n, zero-padded to M_fft)
    # Then the "cross-correlation" at shift j is:
    # sum_i a[i] * p[i+j] = IFFT(FFT(a_padded) * conj(FFT(p_padded)))[j]
    # Wait, that gives sum_i a[i] * p[i-j] (the "delay" version)
    
    # Ugh, let me just use the formula:
    # c[j] = IFFT(FFT(a, L) * conj(FFT(p, L)))[j] = sum_i a[i] * p[(i-j) mod L]
    # For linear (with L >= n + (2n-1) - 1 = 3n-2 to avoid aliasing):
    # c[j] = sum_i a[i] * p[i-j] (for i-j in [0, 2n-2])
    # = sum_{i=j}^{j+2n-2} a[i] * p[i-j] (with a[i]=0 for i>=n)
    # = sum_{i=j}^{n-1} a[i] * p[i-j] (since p has length 2n-1)
    # = sum_{k=0}^{n-1-j} a[k+j] * p[k] (substituting k=i-j)
    
    # So c[j] = sum_{k=0}^{n-1-j} a[k+j] * p[k] for j = 0, 1, ..., n-1
    # And for j >= n: c[j] = sum_{k=0}^{2n-2-(j)} a[?] ... this is getting messy.
    
    # OK let me just use a completely different approach. I'll compute the gradient
    # using the fact that:
    # grad[j] = sum_k p[k] * (a[k-j] + a[j-k])
    # 
    # Let's split this into two parts:
    # Part 1: sum_k p[k] * a[k-j] = sum_k p[k] * a[k-j] (k-j in [0,n-1])
    #         = sum_i a[i] * p[i+j] (i in [0,n-1], i+j in [0,2n-2])
    # Part 2: sum_k p[k] * a[j-k] = sum_k p[k] * a[j-k] (j-k in [0,n-1])  
    #         = sum_i a[i] * p[j-i] (i in [0,min(j, n-1)])
    
    # Part 1 is a "cross-correlation" (sliding dot product)
    # Part 2 is also a "cross-correlation"
    
    # Part 1[j] = sum_i a[i] * p[i+j] for i=0..n-1
    # This can be computed as: (a * p_reversed_adjusted) ... 
    
    # You know what, let me just use the simplest correct method:
    # Compute the full matrix-free gradient using two cross-correlations
    
    # cross1[j] = sum_i a[i] * p[i+j] = correlation of a and p at lag j (a "leads" p by j)
    # cross2[j] = sum_i a[i] * p[j-i] = correlation of a and p at lag j (a "lags" p by j)
    
    # cross1 can be computed as: cross1[j] = (p padded) dot a, with p shifted by -j
    # = np.correlate(p_full, a, 'full')[something]
    
    # Actually, the cleanest way:
    # cross1[j] = sum_{i=0}^{n-1} a[i] * p[i+j]
    # Let's define r = p (length 2n-1). Then cross1[j] = sum_i a[i] * r[i+j]
    # This is the dot product of a[0:n] and r[j:j+n]
    # = (a * reverse(r)) ... no, it's a cross-correlation
    
    # I'll use the FFT-based cross-correlation:
    # np.correlate(a, r, 'full') gives c[k] = sum_n a[n] * conj(r[n-k]) 
    # Wait no, numpy's correlate:
    # np.correlate(a, v)[k] = sum_n a[n+k] * conj(v[n])
    # In 'full' mode, k ranges from -(len(v)-1) to len(a)-1
    
    # So np.correlate(a, r, 'full')[j + (2n-2)] for j=0..n-1 gives:
    # sum_n a[n + j + (2n-2) - (2n-2)] * r[n] ... I'm confusing myself.
    
    # Let me just use the definition directly with FFT:
    # To compute cross1[j] = sum_i a[i] * p[i+j] for j = 0, ..., n-1:
    # This is equivalent to convolving a with a "shifted" version of p
    # Or: cross1 = convolve(a, p_reversed) ... 
    
    # FINAL APPROACH: I'll compute the gradient as follows:
    # grad[j] = sum_k p[k] * (a[k-j] + a[j-k])
    # 
    # Note that if we define the "reversal" of p as p_rev[k] = p[2n-2-k],
    # then a[j-k] = a_rev[k-j] where a_rev[i] = a[n-1-i]
    # 
    # So: sum_k p[k]*a[k-j] = (p * a_rev)[(n-1)+j] ... no this isn't right either.
    
    # OK I'll just use the most straightforward O(n^2) method for now and optimize later if needed.
    # For n up to ~10000, O(n^2) per gradient step is 10^8 which is too slow.
    # So I need the FFT approach.
    
    # Let me use this clean formulation:
    # grad[j] = sum_k p[k] * a[k-j] + sum_k p[k] * a[j-k]
    # 
    # First term: T1[j] = sum_k p[k] * a[k-j] (valid: 0<=k<2n-1, 0<=k-j<n)
    #   = sum_{k=j}^{j+n-1} p[k] * a[k-j]
    #   = sum_{i=0}^{n-1} p[i+j] * a[i]
    #   This is the cross-correlation of p and a at "advance" j
    #   = (p ⊛ a)[j] where (p ⊛ a)[j] = sum_i p[i+j]*a[i]
    #   
    #   Via FFT: Let P = FFT(p, L), A = FFT(a, L) with L >= 3n-2
    #   Then IFFT(P * conj(A))[j] = sum_i p[i+j]*a[i] ... 
    #   Actually: IFFT(P * conj(A))[j] = sum_n p[n] * a[n-j] (circular, with zero-padding = linear)
    #   = sum_n p[n] * a[n-j] = sum_i p[i+j] * a[i] (substituting i = n-j)
    #   YES! So T1[j] = IFFT(FFT(p) * conj(FFT(a)))[j]
    
    # Second term: T2[j] = sum_k p[k] * a[j-k] (valid: 0<=k<2n-1, 0<=j-k<n)
    #   = sum_{k=max(0,j-n+1)}^{min(j, 2n-2)} p[k] * a[j-k]
    #   = sum_{i=0}^{n-1} p[j-i] * a[i] (substituting i=j-k, so k=j-i)
    #   This is the cross-correlation at "delay" j
    #   = (p ⊛ a)[-j] (negative lag)
    #   
    #   Via FFT: IFFT(conj(FFT(p)) * FFT(a))[j] = sum_n p[n-j] * a[n] = sum_i p[i-j] * a[i]
    #   Hmm, that's not quite T2[j].
    #   
    #   T2[j] = sum_i a[i] * p[j-i]
    #   IFFT(conj(FFT(p)) * FFT(a))[j] = sum_n a[n] * p[n-j] = sum_i a[i] * p[i-j]
    #   That's p[i-j], not p[j-i].
    #   
    #   But: sum_i a[i] * p[j-i] = sum_i a[i] * p[(-i)+j]
    #   If we define p_neg[i] = p[-i mod L] = p[L-i], then...
    #   
    #   Alternatively: T2[j] = sum_i a[i] * p[j-i] = T1 evaluated with p and a swapped and lag negated
    #   T2[j] = sum_i p[j-i] * a[i] = sum_k p[k] * a[j-k] (letting k=j-i)
    #   This is the same as T1 but with the roles swapped: it's (a ⊛ p)[j] = sum_i a[i+j]*p[i]
    #   
    #   Hmm, let me just note that T2[j] = T1_rev[j] where T1_rev is computed with a and p swapped.
    #   T2[j] = sum_i a[i] * p[j-i]
    #   = sum_i p[j-i] * a[i]
    #   Let l = j-i: = sum_l p[l] * a[j-l]
    #   = sum_l a[j-l] * p[l]
    #   This is the "cross-correlation" of a and p at lag j (a delayed by j)
    #   = IFFT(FFT(a) * conj(FFT(p)))[j] (with appropriate zero-padding)
    
    # So: grad[j] = IFFT(FFT(p) * conj(FFT(a)))[j] + IFFT(FFT(a) * conj(FFT(p)))[j]
    # Note these are conjugates of each other (for real signals), so:
    # grad[j] = 2 * Re(IFFT(FFT(p) * conj(FFT(a)))[j])
    
    # Wait, that's only true if both FFTs have the same length and the signals are real.
    # Since a and p are real, FFT(p) * conj(FFT(a)) and FFT(a) * conj(FFT(p)) are conjugates.
    # So their IFFTs are conjugates, and the sum is 2*Re of either one.
    
    # Let me verify with a small example... actually let me just code it up and test.
    
    M_fft = 1 << (3 * n - 2).bit_length()
    A_fft = np.fft.rfft(a, M_fft)
    P_fft = np.fft.rfft(p, M_fft)
    cross = np.fft.irfft(P_fft * np.conj(A_fft), M_fft)
    # cross[j] = sum_i p[i+j]*a[i] (for valid indices, with zero-padding)
    # But we need cross[j] for j = 0, ..., n-1
    # And the "delay" version is cross_delay[j] = sum_i a[i+j]*p[i]
    # = cross computed the other way
    
    # Actually, since a and p are real:
    # cross[j] = sum_i p[i+j]*a[i] = Re(sum_i p[i+j]*a[i])
    # cross_delay[j] = sum_i a[i+j]*p[i] = Re(sum_i a[i+j]*p[i])
    # And cross_delay[j] = cross_conj[-j] ... 
    
    # For real signals: IFFT(FFT(p)*conj(FFT(a)))[j] = sum_i p[i+j]*a[i] (linear, with zero-pad)
    # And IFFT(FFT(a)*conj(FFT(p)))[j] = sum_i a[i+j]*p[i]
    # These are NOT the same in general (they're the same only at j=0).
    # But they ARE related: the second is the first with a and p swapped.
    
    # So I need to compute both:
    cross1 = np.fft.irfft(P_fft * np.conj(A_fft), M_fft)  # T1: sum p[i+j]*a[i]
    cross2 = np.fft.irfft(A_fft * np.conj(P_fft), M_fft)  # T2: sum a[i+j]*p[i]
    
    # grad_L[j] = T1[j] + T2[j] for j = 0, ..., n-1
    grad_L = cross1[:n] + cross2[:n]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def subgrad_score(a):
    """Exact score and one subgradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(c.argmax())
    score = 2.0 * n * M / (S * S)
    
    # Subgradient: d(a*a)[k_star]/da[j] = a[k_star-j] + a[j-k_star] (valid)
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = a[idx]
        # Also add the "reverse" part: a[j - k_star] for valid j
        # a[j - k_star] is valid when 0 <= j - k_star < n, i.e., k_star <= j < k_star + n
        # Since j in [0, n-1]: j >= k_star and j < k_star + n (always true since j < n <= k_star + n)
        # So: for j in [max(0, k_star), n-1], add a[j - k_star]
        js2 = max(0, k_star)
        je2 = min(n - 1, k_star + n - 1)
        if je2 >= js2:
            idx2 = np.arange(js2, je2 + 1) - k_star
            sg[js2:je2 + 1] += a[idx2]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad


def adam_lse(a_init, beta_schedule, lr, n_steps, time_limit_s, t_start, normalize=True):
    """Adam with LSE smoothing."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0 / n
        s = 1.0
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t_global = 0
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    n_beta = len(beta_schedule)
    steps_per_beta = max(1, n_steps // n_beta)
    
    for bi, beta in enumerate(beta_schedule):
        steps = min(steps_per_beta, n_steps - bi * steps_per_beta)
        for t in range(steps):
            if time.monotonic() - t_start > time_limit_s:
                break
            t_global += 1
            lr_t = lr / (1.0 + 1e-5 * t_global)
            score, grad = lse_score_grad(a, beta)
            if not np.isfinite(score):
                break
            
            m = b1 * m + (1.0 - b1)
            v = b2 * v + (1.0 - b2) * grad * grad
            m_hat = m / (1.0 - b1 ** t_global)
            v_hat = v / (1.0 - b2 ** t_global)
            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, 1e10)
            
            if normalize:
                s = a.sum()
                if s > 1e-12:
                    a = a / s
            
            if t_global % 1000 == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
        
        # Check at end of each beta phase
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_score, best_a


def lbfgs_optimize(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B optimization with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        x_c = np.clip(x, 0.0, 1e10)
        sc, gr = lse_score_grad(x_c, beta)
        return sc, gr
    
    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50,
                            'maxfun': maxfun})
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def generate_periodic_init(rng, n, period):
    """Generate a periodic initialization with given period."""
    base = np.abs(rng.standard_normal(period)) + 0.1
    base = base / base.sum() * period  # normalize so sum of one period = period
    a = np.tile(base, n // period + 1)[:n]
    a = a / a.sum()
    return a


def generate_gaussian_init(rng, n):
    """Gaussian-shaped initialization."""
    x = np.arange(n) / n
    center = 0.3 + 0.4 * rng.random()
    sigma = 0.1 + 0.2 * rng.random()
    a = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    a = a / a.sum()
    return a


def generate_random_smooth_init(rng, n):
    """Random smooth initialization."""
    n_fourier = max(3, n // 100)
    freqs = rng.uniform(0.5, 20, size=n_fourier)
    phases = rng.uniform(0, 2 * np.pi, size=n_fourier)
    amps = rng.uniform(0.1, 1.0, size=n_fourier)
    x = np.arange(n) / n
    a = np.ones(n)
    for f, p, amp in zip(freqs, phases, amps):
        a += amp * np.cos(2 * np.pi * f * x + p)
    a = np.maximum(a, 0.01)
    a = a / a.sum()
    return a


def resamp
