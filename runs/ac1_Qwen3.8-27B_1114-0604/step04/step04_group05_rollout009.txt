def spectral_shaping(a, n_iters=50, gamma=0.3, quantile=30):
    """
    Gerchberg-Saxton style spectral shaping to flatten the power spectrum.
    Raises low-power DFT bins to reduce autocorrelation peaks.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S_target = a.sum()
    N_fft = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        A = np.fft.fft(a, N_fft)
        power = np.abs(A) ** 2
        # Only work on the positive frequency half (and DC)
        n_half = N_fft // 2
        
        # Target: raise low-power bins
        target = np.percentile(power[1:n_half], quantile)
        # Only raise bins below target
        raise_mask = power[1:] < target
        new_power = power.copy()
        new_power[1:][raise_mask] += gamma * (target - power[1:][raise_mask])
        # Keep DC unchanged (it's the square of the sum)
        new_A = np.sqrt(new_power) * np.exp(1j * np.angle(A))
        # Enforce conjugate symmetry
        a_new = np.fft.ifft(new_A).real[:n]
        a_new = np.clip(a_new, 0.0, 1000.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new = a_new * (S_target / s)
        a = a_new
    return a


def peak_equalization(a, n_top=3, n_iters=100, step=0.05):
    """
    Try to equalize the top n_top peaks of the convolution.
    At each step, identify the dominant peak and adjust coordinates
    to reduce it while being careful not to increase other peaks too much.
    """

    a = a.astype(np.float64).copy()
    S_target = a.sum()
    best_score = compute_score(a)
    
    for it in range(n_iters):
        c, k_star, M, S = conv_peak_info(a)
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score
        
        # Find the peak lag(s)
        # Subgradient at peak: sg[j] = a[k_star - j] for valid j
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je < js:
            break
        idx = k_star - np.arange(js, je + 1)
        sg = a[idx]
        
        # Compute full subgradient-based update direction
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg * (1.0 / S2) - 2.0 * M / S3)
        
        # Find coordinates to decrease (where grad > 0 means increasing a increases score)
        # We want to decrease a[j] where grad[j] > 0 (reducing a[j] reduces score)
        # and increase a[j] where grad[j] < 0
        decrease_mask = grad > 0
        increase_mask = grad < 0
        
        if not decrease_mask.any():
            break
        
        # Apply a small step: decrease the most problematic coordinates
        n_decrease = max(1, int(0.1 * n))
        top_decrease = np.argsort(grad)[::-1][:n_decrease]
        top_increase = np.argsort(grad)[:max(1, int(0.05 * n))]
        
        # Compute step size adaptively
        max_sg = sg.max() if len(sg) > 0 else 1.0
        step_size = step * max_sg * 0.1
        
        a[top_decrease] -= step_size
        a[top_increase] += step_size * 0.5
        
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-12:
            break
        a = a * (S_target / s)
    
    final_sc = compute_score(a)
    if final_sc < best_score:

        if time.monotonic() > deadline:
            break
        c, k_star, M, S = conv_peak_info(a)
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score
        
        # Which coordinates affect the peak?
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je < js:
            break
        peak_coords = np.arange(js, je + 1)
        
        # Rank by contribution to peak
        idx = k_star - peak_coords
        contributions = a[peak_coords] * a[idx]
        
        # Also consider "influential" coordinates: those where changing a[j]
        # most affects the peak value. This is a[k_star - j] (the paired value)
        paired = a[idx]
        
        # Score each coordinate by its "importance"
        importance = contributions + 0.5 * paired * a[peak_coords].max()
        
        # Select top-k most important coordinates to optimize
        order = np.argsort(importance)[::-1][:top_k]
        selected = peak_coords[order]
        
        for j in selected:
            if time.monotonic() > deadline:
                break
            j = int(j)
            a_old = a[j]
            if a_old < 1e-15 and a.max() < 1e-15:
                continue
            
            # Golden section search for a[j]
            lo = 0.0
            hi = max(a_old * 3.0, a.max() * 0.5)
            
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            
            a[j] = c1
            f1 = compute_score(a)
            a[j] = c2
            f2 = compute_score(a)
            
            for _ in range(12):
                if time.monotonic() > deadline:
                    break
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    a[j] = c1
                    f1 = compute_score(a)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    a[j] = c2
                    f2 = compute_score(a)
            
            best_val = c1 if f1 < f2 else c2
            a[j] = best_val
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
    
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a


def lbfgs_optimize(a0, beta, maxiter=3000, maxfun=20000, time_budget=None, t_start=None):
    """L-BFGS-B optimization with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    def obj_and_grad(x):
        return lse_score_grad(x, beta)
    
    if time_budget is not None and t_start is not None:
        deadline = t_start + time_budget
        def timed_obj(x):
            if time.monotonic() > deadline:
                raise TimeoutError
            return lse_score_grad(x, beta)
        try:
            res = minimize(timed_obj, a0, method='L-BFGS-B', jac=True,
                           bounds=bounds,
                           options={'ftol': 1e-15, 'gtol': 1e-12,
                                    'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
        except (TimeoutError, Exception):
            a_out = a0
            return compute_score(a_out), a_out
    else:
        res = minimize(obj_and_grad, a0, method='L-BFGS-B', jac=True,
                       bounds=bounds,
                       options={'ftol': 1e-15, 'gtol': 1e-12,
                                'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def deflation_move(a, n_steps=50, shrink=0.95, grow=1.1):
    """
    Deflation: iteratively shrink coordinates that contribute to the peak
    and grow coordinates that don't, to redistribute energy away from the peak.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S_target = a.sum()
    best_score = compute_score(a)
    
    for step in range(n_steps):
        c, k_star, M, S = conv_peak_info(a)
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score
        
        # Identify peak-contributing coordinates
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je < js:
            break
        
        peak_coords = np.arange(js, je + 1)
        idx = k_star - peak_coords
        # Contribution of each coordinate to the peak
        contrib = a[peak_coords] * a[idx]
        
        # Shrink high-contribution coordinates, grow low ones
        threshold = contrib.mean()
        shrink_mask = peak_coords[contrib > threshold]
        grow_mask = peak_coords[contrib <= threshold]
        
        if len(shrink_mask) == 0:
            break
        
        a[shrink_mask] *= shrink
        if len(grow_mask) > 0:
            a[grow_mask] *= grow
        
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-12:
            break
        a = a * (S_target / s)
    
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a


defadam_optimize(a_init, lr=0.002, n_steps=50000, beta_lse=500,
                 time_budget=None, t_start=None):
    """Adam optimizer with LSE-smoothed gradient and adaptive learning rate."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S_target = a.sum()
    if S_target < 1e-12:
        a[:] = 1.0 / n
        S_target = a.sum()
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf
    
    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break
        
        lr_t = lr / (1.0 + 1e-5 * t)
        score, grad = lse_score_grad(a, beta_lse)
        
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e6)
        s = a.sum()
        if s > 1e-12:
            a = a * (S_target / s)
        
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def make_window(n, window_type='hann'):
    """Create various window functions as initializations."""
    x = np.arange(n, dtype=np.float64)
    if window_type == 'uniform':
        return np.ones(n)
    elif window_type == 'hann':
        return 0.5 * (1 + np.cos(2 * np.pi * x / (n - 1)))
    elif window_type == 'hamming':
        return 0.54 + 0.46 * np.cos(2 * np.pi * x / (n - 1))
    elif window_type == 'blackman':
        return 0.42 + 0.5 * np.cos(2 * np.pi * x / (n - 1)) + 0.08 * np.cos(4 * np.pi * x / (n - 1))
    elif window_type == 'flat_top':
        w = (0.2156 - 0.4160 * np.cos(2 * np.pi * x / (n - 1))
             + 0.2776 * np.cos(4 * np.pi * x / (n - 1))
             - 0.0836 * np.cos(6 * np.pi * x / (n - 1))
             + 0.0069 * np.cos(8 * np.pi * x / (n - 1)))
        return np.clip(w, 0.0, None)
    elif window_type == 'raised_cos2':
        return np.cos(np.pi * x / (n - 1)) ** 2
    elif window_type == 'sqrt':
        return np.sqrt((x + 1) / n)
    elif window_type == 'bimodal':
        center = (n - 1) / 2.0
        return np.exp(-0.5 * ((x - center) / (n * 0.25)) ** 2) + 0.3 * np.exp(-0.5 * ((x - 0) / (n * 0.1)) ** 2) + 0.3 * np.exp(-0.5 * ((x - (n-1)) / (n * 0.1)) ** 2)
    elif window_type == 'chirp':
        return 1.0 + 0.3 * np.sin(4 * np.pi * x / n)
    elif window_type == 'sinc_abs':
        t = (x - (n-1)/2.0) / (n * 0.3)
        w = np.sinc(t)
        return np.clip(np.abs(w), 0.01, None)
    else:
        return np.ones(n)


def refine_sequence(a_init, time_budget, t_start, use_spectral=True):
    """
    Multi-stage refinement: spectral shaping -> L-BFGS continuation -> 
    peak equalization -> coordinate descent.
    """
    a = a_init.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget
    
    def try_update(sc, a_cand, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_cand, 0.0, 1000.0).copy()
            return True
        return False
    
    # Stage 1: Spectral shaping (if enabled, use ~25% of budget)
    if use_spectral and time.monotonic() < deadline - 10:
        spec_iters = max(20, int(time_budget * 0.5))
        a_spec = spectral_shaping(a, n_iters=spec_iters, gamma=0.2, quantile=25)
        sc = compute_score(a_spec)
        if try_update(sc, a_spec, "spectral"):
            a = best_a.copy()
        
        # Also try with different parameters
        a_spec2 = spectral_shaping(a, n_iters=spec_iters // 2, gamma=0.5, quantile=50)
        sc2 = compute_score(a_spec2)
        if try_update(sc2, a_spec2, "spectral_strong"):
            a = best_a.copy()
    
    # Stage 2: L-BFGS continuation (~40% of budget)
    if time.monotonic() < deadline - 10:
        remaining = deadline - time.monotonic()
        lbfgs_budget = remaining * 0.5
        for beta in [200, 1000, 5000, 20000]:
            if time.monotonic() > t_start + time_budget - 10:
                break
            t_alloc = min(lbfgs_budget / 4, 30.0)
            sc, a_opt = lbfgs_optimize(a, beta, maxiter=3000, maxfun=15000,
                                       time_budget=t_alloc, t_start=t_start)
            if try_update(sc, a_opt, f"LBFGS_beta{beta}"):
                a = best_a.copy()
    
    # Stage 3: Peak equalization + deflation (~20% of budget)
    if time.monotonic() < deadline - 5:
        pe_budget = min(deadline - time.monotonic(), time_budget * 0.15)
        sc, a_pe = peak_equalization(a, n_top=3, n_iters=200, step=0.08)
        if try_update(sc, a_pe, "peak_eq"):
            a = best_a.copy()
        
        sc, a_def = deflation_move(a, n_steps=80, shrink=0.93, grow=1.08)
        if try_update(sc, a_def, "deflation"):
            a = best_a.copy()
    
    # Stage 4: Coordinate descent (~15% of budget)
    if time.monotonic() < deadline - 3:
        cd_budget = min(deadline - time.monotonic() - 2, time_budget * 0.15)
        sc, a_cd = coord_descent_peak(a, n_sweeps=6, top_k=50,
                                      time_budget=cd_budget, t_start=t_start)
        if try_update(sc, a_cd, "coord_desc"):
            a = best_a.copy()
    
    return best_score, best_a


def resample_sequence(a, new_len):
    """Resample sequence to a new length using linear interpolation."""
    old_len = len(a)
    if new_len == old_len:
        return a.copy()
    x_old = np.arange(old_len, dtype=np.float64)
    x_new = np.linspace(0, old_len - 1, new_len)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)
