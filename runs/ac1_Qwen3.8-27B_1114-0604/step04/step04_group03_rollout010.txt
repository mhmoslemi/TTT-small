"""

LP-guided directional updates, LSE-smoothed L-BFGS, and multi-length search.

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def hio_flat(a0, n_iters=2000, relax=0.8):
    """
    Hybrid Input-Output (Gerchberg-Saxton) algorithm.
    Alternates between:
    - Frequency domain: reduce peaks in |A|^2 (= convolution)
    - Time domain: enforce non-negativity and zero-padding
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    L = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        a_pad = np.zeros(L)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, L)
        
        # Frequency domain: reduce peaks in |A|^2
        # |A|^2 corresponds to the circular autocorrelation
        # We want to minimize the max of |A|^2
        mag_sq = np.abs(A) ** 2
        mag_sq_max = mag_sq.max()
        
        if mag_sq_max < 1e-16:
            break
        
        # Soft peak reduction: values above threshold get pushed down
        # Use a "power reduction" approach
        # For the DC component (index 0), |A(0)| = sum(a) = 1, which is fixed
        # For other components, we can reduce them
        threshold = mag_sq.mean() * 1.5  # adaptive threshold
        
        # Create modified magnitude
        A_mag = np.abs(A)
        A_phase = A / (A_mag + 1e-30)
        
        # Reduce peaks: for |A|^2 > threshold, scale down
        # |A_new|^2 = threshold + (|A|^2 - threshold) * 0.5 for |A|^2 > threshold
        # |A_new|^2 = |A|^2 for |A|^2 <= threshold
        new_mag_sq = np.where(mag_sq > threshold,
                              threshold + 0.3 * (mag_sq - threshold),
                              mag_sq)
        # Keep DC fixed
        new_mag_sq[0] = mag_sq[0]
        new_mag = np.sqrt(np.maximum(new_mag_sq, 0))
        
        A_new = new_mag * A_phase
        b = np.fft.irfft(A_new, L)
        
        # Time domain: enforce constraints
        b[n:] = 0
        b[:n] = np.maximum(b[:n], 0)
        
        # Relaxed update
        S_new = b[:n].sum()
        if S_new < 1e-12:
            break
        a_candidate = b[:n] / S_new
        
        # Linear interpolation (relaxed HIO)
        a = (1 - relax) * a + relax * a_candidate
        a = np.maximum(a, 0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def freq_peak_shave(a0, n_iters=3000):
    """
    Frequency-domain peak shaving with adaptive threshold.
    Iteratively reduces the largest |A(k)| values.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    L = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        a_pad = np.zeros(L)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, L)
        
        mag = np.abs(A)
        mag_max = mag.max()
        
        # Adaptive threshold: start high, decrease
        # Target: make all non-DC magnitudes close to the geometric mean
        non_dc_mag = mag[1:]
        if len(non_dc_mag) == 0:
            break
        target = np.sqrt(np.prod(np.maximum(non_dc_mag, 1e-15)) ** (1 / len(non_dc_mag)))
        # Or use geometric mean via log
        log_mags = np.log(np.maximum(non_dc_mag, 1e-15))
        target = np.exp(np.mean(log_mags))
        
        # Blend target with current max
        blend = 0.3 + 0.7 * (it / n_iters)  # increase aggressiveness over time
        thresh = (1 - blend) * mag_max + blend * target
        
        # Clip
        mask = mag > thresh
        if not mask.any():
            # No peaks to shave, try reducing target
            thresh = mag[1:].max() * 0.99 if len(mag) > 1 else mag.max() * 0.99
            mask = mag > thresh
            if not mask.any():
                break
        
        A_phase = A / (mag + 1e-30)
        mag_new = np.where(mask, thresh, mag)
        A_new = mag_new * A_phase
        
        b = np.fft.irfft(A_new, L)
        b[n:] = 0
        b[:n] = np.maximum(b[:n], 0)
        
        S_new = b[:n].sum()
        if S_new < 1e-12:
            break
        a_new = b[:n] / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()

def lse_obj_grad(a, beta):
    """LSE-smoothed score and its exact gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum()) + scaled.max() / beta
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # dL/da[i] = 2 * sum_j p[i+j] * a[j] = 2 * (correlation of rev(a) and p)
    # We need: for each i, sum over j where i+j = k: p(k) * a(j)
    # = sum_k p(k) * a(j) where j = k - i, so a(k-i)
    # = (p * rev_a)(i + n - 1) where rev_a[j] = a[n-1-j]
    rev_a = a[::-1]
    # Compute correlation: (p * rev_a) full convolution, take indices [n-1, 2n-2]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        return lse_obj_grad(x, beta)

    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def error_reduction_v3(a0, n_iters=500, alpha_start=0.3, alpha_end=0.9):
    """
    Improved error reduction with adaptive parameters and non-negative enforcement.
    Uses proper linearization of the convolution.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        frac = it / max(1, n_iters - 1)
        alpha = alpha_start + (alpha_end - alpha_start) * frac

        a_pad = np.zeros(M)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        conv = np.fft.irfft(A * A, M)[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-18:
            break

        # Target: reduce values above (1-alpha)*mean + alpha*max
        thresh = (1.0 - alpha) * conv_mean + alpha * conv_max
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-18:
            break

        # Compute correction in frequency domain
        # First-order: dc_k = 2 * sum_j a_j * (da)_{k-j}
        # We want dc = -gamma * excess
        # In frequency: dc_spec = FFT(dc)
        # dc_spec = 2 * A * da_spec (approximately, for small da)
        # So da_spec = dc_spec / (2*A) = -gamma * FFT(excess) / (2*A)
        excess_pad = np.zeros(M)
        excess_pad[:2 * n - 1] = excess
        err_spec = np.fft.rfft(excess_pad, M)

        with np.errstate(divide='ignore', invalid='ignore'):
            d_spec = -err_spec / (2.0 * A)
        # Zero out where A is too small
        small = np.abs(A) < 1e-10 * np.abs(A).max()
        d_spec[small] = 0.0

        # Apply correction
        gamma = 0.5 + 0.3 * frac  # increase step size over time
        A_new = A + gamma * d_spec
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def lp_direction(f, M, n_active=100, max_rounds=20):
    """
    LP-based direction finding using constraint generation.
    Finds g that maximizes sum(g) subject to g>=0 and (f*g)(k) <= M.
    """
    n = len(f)
    
    # Initialize active set with the most binding constraints
    fg = np.convolve(f, f, mode='full')  # This is f*f which is what we have
    # Wait, we need f*g, not f*f. Let me use the current f as a proxy.
    # Actually, the binding constraints are where f*f is largest (since g starts near f)
    k_active = np.argsort(fg)[::-1][:n_active]
    active_set = set(k_active.tolist())
    
    # Solve LP iteratively
    g = f.copy()
    
    for round_idx in range(max_rounds):
        # Build constraint matrix for active set
        k_list = list(active_set)
        m = len(k_list)
        if m == 0:
            return f.copy()
        
        A_ub = np.zeros((m, n))
        b_ub = np.full(m, M)
        
        for idx, k in enumerate(k_list):
            i_start = max(0, k - n + 1)
            i_end = min(k, n - 1)
            for i in range(i_start, i_end + 1):
                A_ub[idx, k - i] = f[i]
        
        # Solve LP
        try:
            res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub,
                         bounds=[(0, None)] * n, method='highs',
                         options={'presolve': True})
            if res.status != 0:
                return f.copy()
            g_new = res.x
        except Exception:
            return f.copy()
        
        # Check for violated constraints
        fg_new = np.convolve(f, g_new, mode='full')
        violated = np.where(fg_new > M * (1 + 1e-8) + 1e-10)[0]
        
        if len(violated) == 0:
            g = g_new
            break
        
        # Add violated constraints (limit total)
        for v in violated[:50]:
            active_set.add(int(v))
        
        if len(active_set) > 2 * n:
            # Too many constraints, trim
            # Keep the most violated
            k_list = list(active_set)
            fg_vals = fg_new[k_list]
            k_sorted = np.argsort(fg_vals)[::-1][:n]
            active_set = set(k_sorted.tolist())
        
        g = g_new
    
    return g


def mirror_descent_v2(a0, n_iters=3000, eta0=0.05, eta_decay=0.999):
    """
    Mirror descent on the simplex with adaptive step size.
    """
    n = len(a0)
    a = np.maximum(a0.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta0

    for it in range(n_iters):
        c = fast_conv(a)
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val  # since sum(a)=1

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of max(conv) at a
        # grad[i] = 2 * a[k_star - i] if 0 <= k_star - i < n, else 0
        grad = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            grad[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        
        # Add a small regularization to prevent zeroing out
        grad += 1e-10
        
        # Mirror descent update (entropic)
        log_a = np.log(a) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        
        a = a_new
        eta *= eta_decay

    return best_a
