def fast_conv(a):

    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

    eps = 1e-8

    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf

        if time.monotonic() > deadline:
            break
        
        score, grad = lse_score_grad(a, beta_lse)
        
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        
        eta = lr / (1.0 + 1e-6 * t)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)

def subgradient_step(a, eta, n_inner=1):

        S = float(a.sum())

            best_score = score

        sg = np.zeros(n)

        je = min(k_star, n - 1)

        S3 = S2 * S

"""

import numpy as np
from scipy.optimize import minimize

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # Gradient of L w.r.t. a: use FFT convolution
    a_rev = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    AR = np.fft.rfft(a_rev, M2_fft)
    conv_pa = np.fft.irfft(P * AR, M2_fft)
    grad_L = 2.0 * conv_pa[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def adam_optimize(a_init, lr, n_steps, beta_lse, time_budget, t_start):
    """Adam optimizer with LSE smoothing and normalization."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    S0 = a.sum()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = t_start + time_budget

    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break
        score, grad = lse_score_grad(a, beta_lse)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        eta = lr / (1.0 + 1e-6 * t)
        a = a - eta * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)
        s = a.sum()
        if s > 1e-12:
            a = a * (S0 / s)
        if t % 3000 == 0:
            sc = compute_score(a)

                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:

        best_a = a.copy()
    return best_score, best_a


def subgradient_descent(a_init, n_steps, eta_init, time_budget, t_start):
    """Exact subgradient descent on the max-of-convolution function."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S0 = a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = t_start + time_budget
    eta = eta_init

    for t in range(n_steps):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        k_star = int(c.argmax())
        M = float(c[k_star])
        S = float(a.sum())
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        S2 = S * S
        S3 = S2 * S
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = a[idx]
        full_grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)

        # Adaptive step size
        grad_norm = np.sqrt(np.sum(full_grad * full_grad))
        if grad_norm > 1e-15:
            eta_t = eta_init * M / (S * grad_norm + 1e-10) * 0.5
        else:
            eta_t = eta_init
        eta_t = min(eta_t, 1.0)

        a = a - eta_t * full_grad
        a = np.clip(a, 0.0, 1e10)
        s = a.sum()
        if s > 1e-12:
            a = a * (S0 / s)

        # Anneal step size
        eta_init = eta_init * (1.0 - 1e-6 * t)

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def coord_descent(a_init, n_sweeps, top_k, time_budget, t_start):
    """Coordinate descent with golden-section 1D search."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget
    gr = (math.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        k_star = int(c.argmax())
        M = float(c[k_star])
        S = float(a.sum())
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score
            best_a = a.copy()

        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = a[idx]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / S3)

        order = np.argsort(np.abs(grad))[::-1][:top_k]
        for j in order:
            if time.monotonic() > deadline:
                break
            j = int(j)
            a_old = a[j]
            lo = 0.0

            a[j] = c2
            f2 = compute_score(a)

                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)

                    f1 = compute_score(a)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    a[j] = c2
                    f2 = compute_score(a)

            a[j] = best_val

    final_sc = compute_score(a)

def run_lbfgs(a0, beta, maxiter, maxfun, time_budget, t_start):
    """L-BFGS-B with LSE smoothing and time limit."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:

    bounds = [(0.0, 1000.0)] * n

            raise RuntimeError("timeout")
        return lse_score_grad(x, beta)

    try:

            obj_and_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-14, 'gtol': 1e-11,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    sc = compute_score(a_out)
    return sc, a_out


def multi_stage_refine(a_init, time_budget, t_start):
    """Multi-stage: Adam -> Subgrad -> LBFGS -> CoordDescent."""
    a = a_init.copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget

    def try_update(sc, a_cand):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_cand, 0.0, 1000.0).copy()
        return best_score

    # Stage 1: Adam
    if time.monotonic() < deadline - 5:
        tb = min(deadline - time.monotonic() - 5, time_budget * 0.35)
        sc, a_opt = adam_optimize(a, lr=0.005, n_steps=100000, beta_lse=150,
                                  time_budget=tb, t_start=t_start)
        try_update(sc, a_opt)
        a = best_a

                                        time_budget=tb, t_start=t_start)

def antipalindromic_init(n, alpha, rng):
    """Create anti-palindromic initialization."""
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    # Linear anti-palindromic component
    f = (x - center) / center if center > 0 else np.zeros(n)
    # Add some Fourier anti-palindromic modes
    for m in rng.choice([1, 2, 3], size=2, replace=False):
        f += 0.3 * np.sin(np.pi * m * (2 * x + 1) / n) * np.sign(x - center)
    a = 1.0 + alpha * f
    a = np.clip(a, 1e-6, None)
    a = a / a.sum()
    return a


def fourier_perturb(a, rng, strength=0.1):
    """Perturb in the Fourier domain for diversity."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    amp = 1.0 + strength * rng.standard_normal(len(A))
    amp = np.clip(amp, 0.2, 3.0)
    A_new = A * amp
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() < 1e-10:
        a_new = a.copy()
    else:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def block_perturb(a, rng, n_blocks=2):
    """Swap random blocks."""
    n = len(a)
    a_new = a.copy()
    for _ in range(n_blocks):
        bl = max(1, int(rng.integers(n // 30, n // 5)))
        i = int(rng.integers(0, n - bl))
        j = int(rng.integers(0, n - bl))
        if i != j:
            tmp = a_new[i:i + bl].copy()
            a_new[i:i + bl] = a_new[j:j + bl]
            a_new[j:j + bl] = tmp
    return a_new


def heat_perturb(a, rng, strength=0.05):
    """Add small random noise and clip."""
    a_new = a + strength * a.mean() * rng.standard_normal(len(a))
    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (a.sum() / a_new.sum())
    return a_new


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: multi-stage hybrid optimization with structured restarts.
    Uses anti-palindromic initializations, LSE-Adam, subgradient descent,
    L-BFGS continuation, coordinate descent, and multi-length exploration.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 25.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        parent_score = compute_score(parent)
        n0 = len(parent)
        print(f"Parent: n={n0}, score={parent_score:.8f}")
    else:
        n0 = 4096
        parent = np.ones(n0) / n0
        parent_score = compute_score(parent)
        print(f"No parent, uniform n={n0}, score={parent_score:.8f}")

    best_score = parent_score
    best_a = parent.copy()

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # ============================================================
    # Phase 1: Intensive optimization from parent (40% budget)
    # ============================================================
    p1 = effective_budget * 0.40
    print(f"\n{'='*60}")
    print(f"Phase 1: Intensive optimization from parent ({p1:.0f}s)")
    print(f"{'='*60}")

    a_work = parent.copy()

    # 1a: Adam warmup
    tb = min(20.0, p1 * 0.15)
    sc, a_opt = adam_optimize(a_work, lr=0.005, n_steps=50000, beta_lse=100,
                              time_budget=tb, t_start=t_start)
    try_update(sc, a_opt, "Adam warmup")
    a_work = best_a

    # 1b: Subgradient descent
    if time.monotonic() < t_start + p1 - 5:
        tb = min(15.0, p1 * 0.1)
        sc, a_opt = subgradient_descent(a_work, n_steps=30000, eta_init=0.05,
                                        time_budget=tb, t_start=t_start)
        try_update(sc, a_opt, "Subgrad")
        a_work = best_a

    # 1c: L-BFGS continuation
    if time.monotonic() < t_start + p1 - 10:
        remaining = t_start + p1 - 10 - time.monotonic()
        for b in [200, 800, 3000, 10000]:
            if time.monotonic() > t_start + p1 - 10:
                break
            tb_alloc = min(remaining * 0.25, 25.0)
            sc, a_opt = run_lbfgs(a_work, b, maxiter=3000, maxfun=20000,
                                  time_budget=tb_alloc, t_start=t_start)
            try_update(sc, a_opt, f"LBFGS b={b}")
            a_work = best_a

    # 1d: Coordinate descent
    if time.monotonic() < t_start + p1 - 3:
        tb = min(20.0, t_start + p1 - 3 - time.monotonic())
        sc, a_opt = coord_descent(a_work, n_sweeps=5, top_k=50,
                                  time_budget=tb, t_start=t_start)
        try_update(sc, a_opt, "CoordDesc P1")
        a_work = best_a

    print(f"  Phase 1 best: {best_score:.8f}")

    # ============================================================
    # Phase 2: Diverse structured restarts (35% budget)
    # ============================================================
    p2 = effective_budget * 0.35
    print(f"\n{'='*60}")
    print(f"Phase 2: Structured restarts ({p2:.0f}s)")
    print(f"{'='*60}")

    p2_deadline = t_start + p1 + p2
    n_restarts = 8
    per_restart = p2 / n_restarts

    for ri in range(n_restarts):
        if time.monotonic() > p2_deadline - 15:
            break
        time_alloc = min(p2_deadline - time.monotonic() - 10, per_restart * 1.5)
        if time_alloc < 20:
            break

        # Generate diverse perturbation
        r = ri % 4
        if r == 0:
            a_pert = fourier_perturb(best_a, rng, strength=0.12)
            label = f"restart {ri}: fourier"
        elif r == 1:
            a_pert = block_perturb(best_a, rng, n_blocks=3)
            label = f"restart {ri}: block"
        elif r == 2:
            a_pert = heat_perturb(best_a, rng, strength=0.08)
            # Add anti-palindromic component
            n_p = len(a_pert)
            x = np.arange(n_p, dtype=np.float64)
            center = (n_p - 1) / 2.0
            anti = 0.05 * a_pert.max() * (x - center) / max(center, 1)
            a_pert = a_pert + anti
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            if a_pert.sum() > 1e-10:
                a_pert = a_pert * (best_a.sum() / a_pert.sum())
            label = f"restart {ri}: heat+anti"
        else:
            # Stronger spectral perturbation
            a_pert = fourier_perturb(best_a, rng, strength=0.25)
            a_pert = fourier_perturb(a_pert, rng, strength=0.1)
            label = f"restart {ri}: strong spectral"

        # Refine
        sc, a_opt = multi_stage_refine(a_pert, time_alloc, t_start)
        try_update(sc, a_opt, label)

    print(f"  Phase 2 best: {best_score:.8f}")

    # ============================================================
    # Phase 3: Anti-palindromic structured initializations (15% budget)
    # ============================================================
    p3 = effective_budget * 0.15
    print(f"\n{'='*60}")
    print(f"Phase 3: Anti-palindromic inits ({p3:.0f}s)")
    print(f"{'='*60}")

    p3_deadline = t_start + p1 + p2 + p3
    n_best = len(best_a)

    for alpha in [0.3, 0.5, 0.7, 0.9]:
        if time.monotonic() > p3_deadline - 20:
            break
        time_alloc = min(p3_deadline - time.monotonic() - 10, p3 / 4)
        if time_alloc < 15:
            break

        a_anti = antipalindromic_init(n_best, alpha, rng)
        sc_init = compute_score(a_anti)
        print(f"    alpha={alpha}: init score={sc_init:.8f}")

        sc, a_opt = multi_stage_refine(a_anti, time_alloc, t_start)
        try_update(sc, a_opt, f"anti-palin a={alpha}")

    print(f"  Phase 3 best: {best_score:.8f}")

    # ============================================================
    # Phase 4: Length variation (10% budget)
    # ============================================================
    p4 = effective_budget * 0.10
    print(f"\n{'='*60}")
    print(f"Phase 4: Length variation ({p4:.0f}s)")
    print(f"{'='*60}")

    p4_deadline = t_start + p1 + p2 + p3 + p4
    n_best = len(best_a)

    for factor in [0.75, 0.9, 1.1, 1.25, 1.5]:
        if time.monotonic() > p4_deadline - 20:
            break
        tl = int(n_best * factor)
        if tl < 256 or tl > 50000:
            continue
        time_alloc = min(p4_deadline - time.monotonic() - 10, 30.0)
        if time_alloc < 15:
            break

        x_old = np.arange(n_best, dtype=np.float64)
        x_new = np.linspace(0, n_best - 1, tl)
        a_res = np.interp(x_new, x_old, best_a)
        a_res = np.clip(a_res, 0.0, 1000.0)

        sc, a_opt = multi_stage_refine(a_res, time_alloc, t_start)
        try_update(sc, a_opt, f"n={tl} (x{factor})")

    print(f"  Phase 4 best: {best_score:.8f}")

    # ============================================================
    # Phase 5: Final intensive polish (remaining time)
    # ============================================================
    print(f"\n{'='*60}")
    print("Phase 5: Final polish")
    print(f"{'='*60}")

    final_deadline = t_start + effective_budget - 5.0
    a_work = best_a.copy()

    # High-beta L-BFGS
    for b in [5000, 15000, 50000]:
        if time.monotonic() > final_deadline - 10:
            break
        tb = min(30.0, final_deadline - time.monotonic() - 10)
        sc, a_opt = run_lbfgs(a_work, b, maxiter=5000, maxfun=30000,
                              time_budget=tb, t_start=t_start)
        try_update(sc, a_opt, f"Final LBFGS b={b}")
        a_work = best_a

    # Subgradient + coordinate descent
    if time.monotonic() < final_deadline - 5:
        tb = min(10.0, final_deadline - time.monotonic() - 5)
        sc, a_opt = subgradient_descent(a_work, n_steps=20000, eta_init=0.03,
                                        time_budget=tb, t_start=t_start)
        try_update(sc, a_opt, "Final Subgrad")
        a_work = best_a

    if time.monotonic() < final_deadline - 3:
        tb = min(20.0, final_deadline - time.monotonic() - 3)
        sc, a_opt = coord_descent(a_work, n_sweeps=8, top_k=60,
                                  time_budget=tb, t_start=t_start)
        try_update(sc, a_opt, "Final CoordDesc")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
