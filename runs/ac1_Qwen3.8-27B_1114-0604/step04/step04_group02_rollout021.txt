    # More simply: it's the convolution of a with p_full (reversed), 
    # evaluated at appropriate indices.
    # 
    # (a * p_full_rev)[m] = sum_j a[j] * p_full_rev[m-j] = sum_j a[j] * p_full[len(p_full_rev)-1-m+j]
    # 
    # Let me just use FFT:
    # dL/da_i = sum_k p_full[k] * a[k-i] (for valid k-i)
    # = [a convolve p_full_reversed] at index (len(c)-1 - i) ... 
    # 
    # Actually, let me use the approach from the previous code:
    # conv_res = irfft(FFT(a_reversed) * FFT(p])[n-1:2n-1]
    # where a_reversed = a[::-1]
    # Then dL/da = 2 * conv_res
    
    rev_a = a[::-1]
    M2 = len(c) + n - 1  # = 3n-2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p_full, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    
    return score, grad

        return lse_topk_grad(x, beta, top_k)

def nesterov_subgrad(a, n_iters=10000, f_lb=1.0):
    """Nesterov-accelerated subgradient descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)
    patience = 0
    
    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = y.copy()
            patience = 0
        else:
            patience += 1
            if patience > 2000:
                break
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]
        
        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.5 / (np.sqrt(it + 1)))
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new
        
        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()
    
    return best_a


def peel_peaks(a, n_rounds=50, n_top=20):
    """Iteratively reduce the peak by adjusting top contributors."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for _ in range(n_rounds):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()  # contrib[i] = a[j_start + i] * ... 
        # Actually, the contribution of a[j] to c[k_star] is a[k_star - j]
        # So for j in [j_start, j_end], the contribution is a[k_star - j]
        # But we want to reduce c[k_star] = sum_j a[j] * a[k_star - j]
        # The "pair" of a[j] in the peak is a[k_star - j]
        
        # Sort by the value of a[j] (those that are large and paired with large a[k_star-j])
        # Reduce the largest a[j] in the segment
        order = np.argsort(seg)[::-1]
        
        reduced = False
        for idx in order[:n_top]:
            j = j_start + idx
            a_old = a[j]
            if a_old < 1e-10:
                continue
            # Try reducing a[j] by 50% and redistributing
            reduction = a_old * 0.5
            a[j] -= reduction
            
            # Redistribute to positions NOT in the peak segment
            # Find positions far from the peak
            in_peak = np.zeros(n, dtype=bool)
            in_peak[j_start:j_end+1] = True
            out_idx = np.where(~in_peak)[0]
            if len(out_idx) > 0:
                # Redistribute proportionally to current values (or uniformly)
                a[out_idx] += reduction / len(out_idx)
            
            S = a.sum()
            a = a / S
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                reduced = True
            else:
                # Revert
                a[j] += reduction
                if len(out_idx) > 0:
                    a[out_idx] -= reduction / len(out_idx)
                S = a.sum()
                a = a / S
        
        if not reduced:
            break
    
    return best_a


def freq_domain_init(n, rng):
    """Generate initial sequence using frequency-domain shaping."""
    # Start with random phases in frequency domain
    M_fft = 1 << (2 * n - 1).bit_length()
    F = rng.standard_normal(M_fft // 2 + 1) + 1j * rng.standard_normal(M_fft // 2 + 1)
    # Ensure DC component is positive
    F[0] = abs(F[0])
    a = np.fft.irfft(F, M_fft)[:n]
    a = np.abs(a)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    return a


def build_init ngã(n, rng, parent):

"""
Improved optimizer using iterative active-set LSE with adaptive K and beta.
Key innovations over previous approach:
1. Adaptive top-K LSE: start with K=1 (essentially max), gradually increase
2. Iterative active-set re-identification with patience-based stopping
3. Frequency-domain initialization for diverse starting points
4. Multi-length exploration with interpolation
5. Time-bounded L-BFGS-B with exception handling
6. Aggressive tail-end refinement with very high beta
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_topk_grad(a, beta, top_k):
    """
    Compute LSE-smoothed score using only top_k peaks, and its gradient.
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    # Get top-k convolution values
    if top_k >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argsort(c)[::-1][:top_k]

    c_top = c[top_idx]
    c_top_max = float(c_top.max())

    scaled = beta * (c_top - c_top_max)
    exp_scaled = np.exp(np.maximum(scaled, -60))
    p = exp_scaled / exp_scaled.sum()

    L = c_top_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)

    # Build full-length p array
    p_full = np.zeros(len(c))
    p_full[top_idx] = p

    # Gradient: dL/da_i = sum_k p_full[k] * 2 * a[k-i] (for valid 0<=k-i<n)
    # Computed via: 2 * (a[::-1] * p_full)[n-1 : 2n-1]
    rev_a = a[::-1]
    M2 = len(c) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p_full, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]

    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)

    return score, grad


def run_lbfgs_timed(a0, beta, top_k, max_time=60.0, maxiter=50000, maxfun=100000):
    """Run L-BFGS-B on the top-K LSE objective with a hard time limit."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    t0 = time.monotonic()

    def func_and_grad(x):
        if time.monotonic() - t0 > max_time:
            raise TimeoutError("Time limit exceeded")
        return lse_topk_grad(x, beta, top_k)

    try:
        res = minimize(
            func_and_grad, a0.copy(),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0.copy()

    exact_sc = compute_score(a_out)
    elapsed = time.monotonic() - t0
    return exact_sc, a_out, elapsed


def nesterov_subgrad(a, n_iters=10000, f_lb=1.0):
    """Nesterov-accelerated subgradient descent on the exact objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)
    patience = 0

    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_a = y.copy()
            patience = 0
        else:
            patience += 1
            if patience > 3000:
                break

        # Subgradient at the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]

        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Adaptive step size
        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 1.0 / (np.sqrt(it + 1) * 2.0))

        # Nesterov momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new

        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)

        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()

    return best_a


def peel_peaks(a, n_rounds=40, n_top=15):
    """
    Iteratively reduce the peak by adjusting top contributors.
    At each step, identify the peak, find the largest contributors,
    and try redistributing their mass to positions far from the peak.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for _ in range(n_rounds):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)

        seg = a[j_start:j_end + 1]
        order = np.argsort(seg)[::-1]

        # Positions NOT in the peak segment (for redistribution)
        in_peak = np.zeros(n, dtype=bool)
        in_peak[j_start:j_end + 1] = True
        out_idx = np.where(~in_peak)[0]

        reduced_any = False
        for idx in order[:n_top]:
            j = j_start + idx
            a_old = a[j]
            if a_old < 1e-10:
                continue

            # Try reducing by 40%
            reduction = a_old * 0.4
            a[j] -= reduction

            if len(out_idx) > 0:
                # Redistribute proportionally to existing values at out_idx
                out_vals = a[out_idx]
                total_out = out_vals.sum()
                if total_out > 1e-15:
                    a[out_idx] += reduction * out_vals / total_out
                else:
                    a[out_idx] += reduction / len(out_idx)

            S = a.sum()
            a = a / S
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                reduced_any = True
            else:
                # Revert
                a[j] += reduction
                if len(out_idx) > 0:
                    total_out = a[out_idx].sum() - reduction
                    if total_out > 1e-15:
                        pass  # approximate revert (good enough for search)
                    a[out_idx] -= reduction * a[out_idx] / (a[out_idx].sum() + 1e-15)
                S = a.sum()
                if S > 1e-12:
                    a = a / S

        if not reduced_any:
            break

    return best_a


def freq_domain_init(n, rng):
    """Generate initial sequence using frequency-domain shaping."""
    M_fft = 1 << (2 * n).bit_length()
    F_real = rng.standard_normal(M_fft // 2 + 1)
    F_imag = rng.standard_normal(M_fft // 2 + 1)
    F = F_real + 1j * F_imag
    F[0] = abs(F[0]) + 1.0
    a = np.fft.irfft(F, M_fft)[:n]
    a = np.abs(a)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    return a


def subgrad_with_momentum(a, n_iters=5000, f_lb=1.0, decay=0.9995, beta_mom=0.85):
    """Subgradient with heavy-ball momentum."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    step0 = 0.15

    for it in range(n_iters):
        c = fast_conv(a)
        S_a = float(a.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (S_a * S_a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * a[k_star - idx]

        S2 = S_a * S_a
        S3 = S2 * S_a
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = step0 * decay ** it
        step = min(step, 2.0 * (sc - f_lb) / (gnorm + 1e-30))

        v = beta_mom * v - step * grad
        a_new = a + v
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa

    return best_a
