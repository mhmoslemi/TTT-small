One more thought: the "active set" approach. At the optimum, the maximum of the convolution is achieved at multiple lags (otherwise we could reduce the unique maximum). So at the optimum, there should be a set of "active" lags all achieving the same value (the max).

Algorithm:
1. Find the current max lag k*
2. Try to reduce (a*a)[k*] by moving in the direction -grad_k*
3. Check if any other lag has become the new max
4. If so, add it to the active set
5. Repeat until the active set stabilizes and the max value stops decreasing

This is similar to the "embossed" or "equi-ripple" design in filter design.

For the subgradient direction at the active set {k_1, ..., k_m}:
direction = (1/m) * sum_i grad_{k_i}

where grad_{k_i}[j] = 2*a[k_i - j] (for valid k_i - j).

This is the average of the gradients of all active lags. The idea is to reduce ALL active lags simultaneously.

"""
Hybrid optimizer for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
Key improvements over previous:
1. Non-palindromic optimization for full freedom
2. LSE-smoothed L-BFGS with continuation (high beta)
3. Active-set subgradient to equalize top lags
4. Multi-length exploration
5. Projected gradient with adaptive step size
"""

import numpy as np
from scipy.optimize import minimize
import time

TOTAL_BUDGET = 2400

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    x = beta * (c - c_max)
    ex = np.exp(x)
    L = c_max + np.log(ex.sum()) / beta
    score = 2.0 * n * L / (S * S)
    p = ex / ex.sum()
    N = 1 << (2 * n - 1).bit_length()
    X = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    R = np.fft.irfft(X * np.conj(P), N)
    h = R[:n]
    dL_da = (2.0 / beta) * h
    S2 = S * S
    grad = 2.0 * n * (dL_da / S - 2.0 * L / S2)
    return score, grad

def subgrad_score_grad(a, top_k=1):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    score = 2.0 * n * M / (S * S)
    S2 = S * S
    if top_k == 1:
        k_star = int(c.argmax())
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = 2.0 * a[idx]
    else:
        sorted_idx = np.argsort(c)[::-1][:top_k]
        thresh = c[sorted_idx[-1]]
        active = np.where(c >= thresh - 0.01 * M)[0]
        sg = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                idx = k - np.arange(js, je + 1)
                sg[js:je + 1] += 2.0 * a[idx]
        sg /= len(active)
    grad = 2.0 * n * (sg / S2 - 2.0 * M / (S2 * S))
    return score, grad

def simplex_project(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def projected_subgrad_step(a, grad, step):
    a_new = a - step * grad
    return simplex_project(a_new)

def run_projected_subgrad(a0, max_iter=5000, time_limit=None, t_start=None, top_k=1, verbose_every=500):
    a = a0.copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / len(a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    step_init = 0.01
    decay = 0.999
    step = step_init
    n = len(a)
    for it in range(max_iter):
        if time_limit is not None and time.monotonic() - t_start > time_limit:
            break
        score, grad = subgrad_score_grad(a, top_k=top_k)
        a_new = projected_subgrad_step(a, grad, step)
        sc_new = compute_score(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new
            step = min(step * 1.05, 0.1)
        else:
            # Try a smaller step
            step *= 0.8
            if step < 1e-8:
                step = step_init * 0.1
        if verbose_every and it % verbose_every == 0:
            pass
    return best_score, best_a

def run_lbfgs_lse(a0, beta, maxiter=3000, maxfun=50000, time_limit=None, t_start=None):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    call_count = [0]
    def obj(x):
        call_count[0] += 1
        if time_limit is not None and time.monotonic() - t_start > time_limit:
            x_c = np.clip(x, 0.0, 1000.0)
            return compute_score(x_c), np.zeros(n)
        x_c = np.clip(x, 0.0, 1e10)
        return lse_score_grad(x_c, beta)

    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-14,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50,
                            'olsson_width': 1e16})
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def active_set_balance(a, n_iters=200, time_limit=None, t_start=None):
    """Iteratively balance the top lags to equalize them."""
    n = len(a)
    a = a.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    for it in range(n_iters):
        if time_limit is not None and time.monotonic() - t_start > time_limit:
            break
        c = fast_conv(a)
        M = c.max()
        # Find lags within 5% of max
        thresh = 0.95 * M
        active = np.where(c >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(c.argmax())])
        # Compute average gradient over active lags
        sg = np.zeros(n)
        for k in active:
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                idx = k - np.arange(js, je + 1)
                sg[js:je + 1] += 2.0 * a[idx]
        sg /= len(active)
        S = a.sum()
        S2 = S * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M / (S2 * S))
        # Line search
        best_step = 0.001
        best_sc = compute_score(a)
        for step in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]:
            a_try = simplex_project(a - step * grad)
            sc_try = compute_score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_step = step
                a = a_try
        if best_sc < best_score:
            best_score = best_sc
            best_a = a.copy()
    return best_score, best_a

def generate_inits(rng, n):
    inits = []
    x = np.arange(n) / n
    # Uniform
    inits.append(np.ones(n) / n)
    # Gaussian centered
    for sigma in [0.1, 0.15, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        inits.append(g / g.sum())
    # Asymmetric gaussian (skewed right)
    g = np.exp(-0.5 * ((x - 0.4) / 0.2) ** 2)
    inits.append(g / g.sum())
    g = np.exp(-0.5 * ((x - 0.6) / 0.2) ** 2)
    inits.append(g / g.sum())
    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(rc / rc.sum())
    # Sinusoidal
    for freq in [2, 4, 6, 10]:
        s = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x)
        s = np.maximum(s, 0.01)
        inits.append(s / s.sum())
    # Random smooth
    r = np.abs(rng.standard_normal(n)) + 0.3
    k = max(3, n // 100)
    kernel = np.ones(k) / k
    r_s = np.convolve(r, kernel, mode='same')
    inits.append(r_s / r_s.sum())
    # Two bump asymmetric
    c1, c2 = 0.3, 0.7
    s1, s2 = 0.08, 0.12
    b = np.exp(-0.5 * ((x - c1) / s1) ** 2) + 0.7 * np.exp(-0.5 * ((x - c2) / s2) ** 2)
    inits.append(b / b.sum())
    # Linear ramp
    lin = np.linspace(0.1, 1.0, n)
    inits.append(lin / lin.sum())
    # Power law
    pw = (x + 0.01) ** 2
    inits.append(pw / pw.sum())
    return [np.clip(x, 0.0, 1000.0) for x in inits]

def resample_sequence(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)
