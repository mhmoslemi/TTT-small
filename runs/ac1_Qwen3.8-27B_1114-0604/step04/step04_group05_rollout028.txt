"""
Nesterov accelerated gradient with LSE smoothing in log-domain,
spectral-domain projection, and multi-scale restarts for minimizing
the peak self-convolution ratio of non-negative sequences.
"""

import numpy as np
from scipy.optimize import minimize
import time
import math


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_smoothed_obj_grad(a, beta):

    score = 2.0 * n * log_sum / S2

    p = exp_scaled / exp_scaled.sum()
    
    # Gradient of LSE_beta(c) w.r.t. a:

    # This is the cross-correlation of a and p, evaluated at lags 0 to n-1
    # Using FFT:
    L_conv = n + L - 1  # = n + 2n - 1 - 1 = 3n - 2
    L_fft = 1 << (L_conv - 1).bit_length()

def nesterov_lse(a0, betas, time_budget, t_start, lr_init=0.01, momentum=0.9):
    """
    Nesterov accelerated gradient with LSE smoothing.
    Works in the log-domain to handle non-negativity.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    
    # Normalize to sum = 1
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = t_start + time_budget
    
    # Work in log-domain: x = log(a), so a = exp(x)
    # But we need to maintain sum(exp(x)) = 1
    # Alternative: just work in a-domain with projection
    
    # Actually, let's work in a-domain with Nesterov momentum and projection
    # The projection onto {a >= 0, sum a = 1} is: a_proj = clip(a, 0, inf), then normalize
    
    x = a.copy()  # Current point
    v = np.zeros(n)  # Velocity (momentum)
    
    step = 0
    for beta in betas:
        # For each beta, run several Nesterov steps
        n_steps = 5000
        for t in range(n_steps):
            if time.monotonic() > deadline:
                break
            
            # Evaluate gradient at lookahead point
            # Nesterov: y = x + (t/(t+2)) * v (approximately)
            # Simplified: y = x + momentum * v
            y = x + momentum * v
            y = np.clip(y, 0.0, None)
            sy = y.sum()
            if sy < 1e-12:
                y = x.copy()
                sy = x.sum()
            else:
                y = y / sy  # Normalize for evaluation
            
            score, grad = lse_smoothed_obj_grad(y, beta)
            
            # Nesterov update
            # v_new = momentum * v + lr * grad
            # x_new = x - v_new
            # Then project
            lr_t = lr_init / (1.0 + 0.01 * t / 100.0)  # Simple decay
            v = momentum * v + lr_t * grad
            x = x - v
            
            # Project onto {a >= 0, sum a = 1}
            x = np.clip(x, 0.0, None)
            sx = x.sum()
            if sx < 1e-12:
                x[:] = 1.0 / n
            else:
                x = x / sx
            
            step += 1
            if step % 1000 == 0:
                sc = compute_score(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()
        
        if time.monotonic() > deadline:
            break
    
    sc = compute_score(x)
    if sc < best_score:
        best_score = sc
        best_a = x.copy()
    
    return best_score, best_a


def spectral_flattening(a, n_iters=100, t_start=None, time_budget=30.0):
    """
    Iterative spectral flattening: alternate between time and frequency domain.
    In freq domain: try to equalize |A[m]| for m != 0.
    In time domain: enforce non-negativity and sum constraint.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        # Compute DFT
        A = np.fft.rfft(a, M_fft)
        A_abs = np.abs(A)
        
        # Target: |A[m]| should be constant for m != 0
        # A[0] = sum(a) = 1 (DC)
        # Target for m != 0: some value c
        # By Parseval: sum_m |A[m]|^2 = M_fft * sum a[j]^2
        # We want |A[m]|^2 = c^2 for m=1,...,M_fft/2
        # (1 + (M_fft/2 - 1)) * c^2 + 1^2 = M_fft * sum(a^2)
        # For a "flat" spectrum (m != 0), c^2 = (M_fft * sum(a^2) - 1) / (M_fft/2 - 1)
        
        # Simpler: just try to make |A[m]| more equal
        # Use a water-filling approach:
        # target_power = mean(|A[m]|^2 for m != 0)
        powers = A_abs ** 2
        dc_power = powers[0]
        ac_powers = powers[1:]
        target = ac_powers.mean()
        
        # Adjust: scale each |A[m]| to be between current and target
        # A[m] -> A[m] * (target / |A[m]|^2)^alpha for m != 0
        alpha = 0.5  # Mixing parameter
        
        A_new = A.copy()
        for m in range(1, len(A)):
            if A_abs[m] > 1e-15:
                scale = (target / (A_abs[m] ** 2)) ** alpha
                scale = np.clip(scale, 0.5, 2.0)
                A_new[m] = A[m] * scale
        
        # Inverse DFT
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        
        # Project to non-negative, sum=1
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        a = a_new
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    return best_score, best_a

def random_nonnegative_init(n, rng):
    """Generate random non-negative sequence."""
    # Exponential distribution gives a "random" non-negative sequence
    a = rng.exponential(1.0, size=n)
    return a / a.sum()


def raised_cosine_init(n, rng):
    """Raised cosine (Hamming-like) initialization."""
    x = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (x + 0.5) / n) ** 2
    # Add some randomness
    a += 0.1 * rng.uniform(0, 1, size=n)
    return a / a.sum()


def spectral_random_init(n, rng):
    """
    Initialize by choosing random spectral magnitudes and random phases,
    then clips to non-negative.
    """
    M_fft = 1 << (2 * n - 2).bit_length()
    # Random phases
    A = np.fft.rfft(np.zeros(M_fft))
    A[0] = np.sqrt(M_fft)  # DC: sum = 1 after inverse
    for m in range(1, len(A)):
        phase = rng.uniform(0, 2 * np.pi)
        mag = rng.uniform(0.5, 1.5) * 1.0 / np.sqrt(len(A) - 1)
        A[m] = mag * np.exp(1j * phase)
    
    a = np.fft.irfft(A, M_fft)[:n]
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        return np.ones(n) / n
    return a / s


def subgrad_sum_method(a0, time_budget, t_start, n_inner=10000):
    """
    Subgradient method using subgradients from ALL lags (not just argmax).
    This gives a "smoothed" subgradient that's more stable.
    Uses Nesterov acceleration.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = t_start + time_budget
    
    x = a.copy()
    v = np.zeros(n)
    momentum = 0.9
    lr = 0.1
    
    for t in range(n_inner):
        if time.monotonic() > deadline:
            break
        
        # Lookahead
        y = x + momentum * v
        y = np.clip(y, 0.0, None)
        sy = y.sum()
        if sy < 1e-12:
            y = x.copy()
            sy = x.sum()
        else:
            y = y / sy
        
        # Compute "smoothed" subgradient using softmax over conv values
        c = fast_conv(y)
        S = 1.0
        c_max = float(c.max())
        beta = 50.0  # Temperature for softmax
        p = np.exp(beta * (c - c_max))
        p /= p.sum()
        
        # Weighted subgradient
        # For each lag k with weight p[k], subgrad is (2n/S^2)(2*a[k-j] - 2*c[k]*a[j]/S)
        # Sum over k: (2n/S^2)(2*sum_k p[k]*a[k-j] - 2*c[k_weighted]*a[j]/S)
        # where c[k_weighted] = sum_k p[k]*c[k]
        
        c_weighted = float(np.dot(p, c))
        
        # sum_k p[k] * a[k-j] = cross-correlation
        L = 2 * n - 1
        L_conv = 3 * n - 2
        L_fft = 1 << (L_conv - 1).bit_length()
        Y_fft = np.fft.rfft(y, L_fft)
        P_fft = np.fft.rfft(p, L_fft)
        x_corr = np.fft.irfft(np.conj(Y_fft) * P_fft, L_fft)
        
        sg = (2.0 * n) * (2.0 * x_corr[:n] - 2.0 * c_weighted * y)
        
        # Nesterov update
        lr_t = lr / (1.0 + 0.1 * t / 1000.0)
        v = momentum * v + lr_t * sg
        x = x - v
        
        # Project
        x = np.clip(x, 0.0, None)
        sx = x.sum()
        if sx < 1e-12:
            x[:] = 1.0 / n
        else:
            x = x / sx
        
        if (t + 1) % 2000 == 0:
            sc = compute_score(x)
            if sc < best_score:
                best_score = sc
                best_a = x.copy()
            # Adaptive step size
            lr *= 0.95
    
    sc = compute_score(x)
    if sc < best_score:
        best_score = sc
        best_a = x.copy()
    
    return best_score, best_a

Let me also think about the theoretical lower bound. For non-negative sequences:
- The convolution at lag n-1 is `sum_j a[j]*a[n-1-j]`
- By Cauchy-Schwarz: this is <= `sqrt(sum a[j]^2 * sum a[n-1-j]^2) = sum a[j]^2`
- With sum a = 1: `sum a[j]^2 >= 1/n`
- So the center value is >= 1/n, giving score >= 2n * (1/n) = 2

But this bound is only achieved by the uniform sequence. To get below 2, we need the max to be at a different lag, not the center.

Wait, that can't be right. For non-negative a, is the max always at the center?

Not necessarily! Consider a = [1, 0, 0, ..., 0, 1] (two 1s at the ends). Then:
- c[0] = 1
- c[n-2] = 2 (cross term)
- c[n-1] = 0 (center)
- c[n] = 2 (cross term)
- c[2n-2] = 1

So the max is at lags n-2 and n, not at the center! And the score would be `2n * 2 / 4 = n` which is terrible.

What about a = [1, 1, 0, 0, ..., 0]? 
- c[0] = 1, c[1] = 2, c[2] = 1, rest are 0
- max = 2, sum = 2, score = 2n * 2 / 4 = n. Still bad.

What about a evenly spread: a = [1, 0, 1, 0, ..., 1, 0] (alternating)?
- n elements, n/2 ones

"""
Multi-strategy search for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
Uses Nesterov accelerated gradient with LSE smoothing, spectral-domain projection,
subgradient averaging, and multi-scale restarts with diverse initializations.

import numpy as np
import time
import math


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_obj_grad(a, beta):
    """
    LSE-smoothed objective 2n*LSE_beta(c)/S^2 and its gradient.
    c = linear self-convolution of a.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.clip(scaled, -50.0, 50.0))
    exp_sum = float(exp_scaled.sum())
    log_sum = c_max + math.log(exp_sum) / beta
    S2 = S * S
    score = 2.0 * n * log_sum / S2

    p = exp_scaled / exp_sum

    # Gradient: dLSE/da[j] = 2 * sum_m a[m] * p[m+j] (cross-correlation of a and p)
    L = 2 * n - 1
    L_conv = 3 * n - 2
    L_fft = 1 << (L_conv - 1).bit_length()
    A_fft = np.fft.rfft(a, L_fft)
    P_fft = np.fft.rfft(p, L_fft)
    x_corr = np.fft.irfft(np.conj(A_fft) * P_fft, L_fft)
    dLSE_da = 2.0 * x_corr[:n]

    # Full gradient: 2n * (dLSE_da/S^2 - 2*LSE*a/S^3)
    grad = (2.0 * n) * (dLSE_da / S2 - 2.0 * log_sum * a / (S2 * S))
    return score, grad


def nesterov_lse_simplex(a0, betas, time_budget, t_start, lr_init=0.05, momentum=0.9):
    """
    Nesterov accelerated gradient with LSE smoothing, projected onto the simplex.
    Works in a-space with projection onto {a>=0, sum=1}.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 1e-15, None)
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)
    deadline = t_start + time_budget

    x = a.copy()
    v = np.zeros(n)

    total_steps = 0
    for beta in betas:
        n_steps = 4000
        for t in range(n_steps):
            if time.monotonic() > deadline - 1.0:
                break

            # Nesterov lookahead
            y = x + momentum * v
            y = np.clip(y, 0.0, None)
            sy = y.sum()
            if sy < 1e-12:
                y = x.copy()
            else:
                y = y / sy

            score, grad = lse_obj_grad(y, beta)

            # Adaptive learning rate
            lr_t = lr_init / (1.0 + 0.02 * total_steps / 1000.0)

            # Nesterov update
            v = momentum * v + lr_t * grad
            x = x - v

            # Project onto simplex {a >= 0, sum a = 1}
            x = np.clip(x, 0.0, None)
            sx = x.sum()
            if sx < 1e-12:
                x[:] = 1.0 / n
            else:
                x = x / sx

            total_steps += 1
            if total_steps % 800 == 0:
                sc = compute_score(x)
                if sc < best_score:
                    best_score = sc
                    best_a = x.copy()

        # After each beta stage, do a projection
        x = np.clip(x, 0.0, None)
        sx = x.sum()
        if sx > 1e-12:
            x = x / sx

    sc = compute_score(x)
    if sc < best_score:
        best_score = sc
        best_a = x.copy()

    return best_score, best_a


def spectral_water_filling(a0, time_budget, t_start, n_iters=200, alpha=0.3):
    """
    Spectral water-filling: iteratively flatten the power spectrum in frequency domain,
    then project back to non-negative time domain.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a[:] = 1.0 / n
        s = 1.0
    a = a / s

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

    best_a = a.copy()
    best_score = compute_score(a)
    deadline = t_start + time_budget

    for it in range(n_iters):
        if time.monotonic() > deadline - 1.0:
            break

        # Compute DFT
        A = np.fft.rfft(a, M_fft)
        powers = np.abs(A) ** 2

        # Target: equalize AC power
        ac_powers = powers[1:]
        target_power = float(ac_powers.mean())

        if target_power < 1e-20:
            break

        # Water-filling: scale each frequency bin
        A_new = A.copy()
        for m in range(1, len(A)):
            pw = powers[m]
            if pw > 1e-20:
                # Move toward target
                ratio = target_power / pw
                scale = 1.0 + alpha * (math.sqrt(ratio) - 1.0)
                scale = np.clip(scale, 0.5, 2.0)
                A_new[m] = A[m] * scale

        # Inverse DFT
        a_new = np.fft.irfft(A_new, M_fft)[:n]

        # Project to non-negative, sum=1
        a_new = np.clip(a_new, 0.0, None)
        sn = a_new.sum()
        if sn < 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new / sn

        # Some relaxation to avoid oscillation
        a = 0.7 * a + 0.3 * a_new

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_score, best_a


def subgrad_nesterov_multi(a0, time_budget, t_start, beta_smooth=100.0):
    """
    Subgradient method using softmax-averaged subgradients from all lags.
    Nesterov acceleration for faster convergence.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 1e-15, None)
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)
    deadline = t_start + time_budget

    x = a.copy()
    v = np.zeros(n)
    momentum = 0.9
    lr = 0.08

    for t in range(20000):
        if time.monotonic() > deadline - 1.0:
            break

        # Lookahead
        y = x + momentum * v
        y = np.clip(y, 0.0, None)
        sy = y.sum()
        if sy < 1e-12:
            y = x.copy()
            sy = x.sum()
        else:
            y = y / sy

        # Compute softmax-weighted subgradient
        c = fast_conv(y)
        c_max = float(c.max())
        p = np.exp(beta_smooth * (c - c_max))
        p_sum = p.sum()
        if p_sum < 1e-30:
            p = np.zeros(len(c))
            p[c.argmax()] = 1.0
        else:
            p = p / p_sum

        c_weighted = float(np.dot(p, c))

        # Weighted subgradient direction
        L = 2 * n - 1
        L_conv = 3 * n - 2
        L_fft = 1 << (L_conv - 1).bit_length()
        Y_fft = np.fft.rfft(y, L_fft)
        P_fft = np.fft.rfft(p, L_fft)
        x_corr = np.fft.irfft(np.conj(Y_fft) * P_fft, L_fft)

        # Subgradient of 2n*max(c)/S^2 (using weighted max as surrogate)
        sg = (2.0 * n) * (2.0 * x_corr[:n] - 2.0 * c_weighted * y)

        # Nesterov update
        lr_t = lr / (1.0 + 0.05 * t / 1000.0)
        v = momentum * v + lr_t * sg
        x = x - v

        # Project
        x = np.clip(x, 0.0, None)
        sx = x.sum()
        if sx < 1e-12:
            x[:] = 1.0 / n
        else:
            x = x / sx

        if (t + 1) % 1500 == 0:
            sc = compute_score(x)
            if sc < best_score:
                best_score = sc
                best_a = x.copy()
            # Decay step size
            lr *= 0.97

    sc = compute_score(x)
    if sc < best_score:
        best_score = sc
        best_a = x.copy()

    return best_score, best_a


def log_domain_nesterov(a0, betas, time_budget, t_start, lr_init=0.1):
    """
    Nesterov in log-domain: a = softmax(x) parameterization.
    This naturally ensures a > 0 and sum(a) = 1.
    """
    n = len(a0)
    a_init = a0.astype(np.float64).copy()
    a_init = np.clip(a_init, 1e-10, None)
    a_init = a_init / a_init.sum()
    x = np.log(a_init)

    best_a = a_init.copy()
    best_score = compute_score(a_init)
    deadline = t_start + time_budget

    v = np.zeros(n)
    momentum = 0.9

    for beta in betas:
        n_steps = 3000
        for t in range(n_steps):
            if time.monotonic() > deadline - 1.0:
                break

            # Lookahead
            y = x + momentum * v
            # Convert to probability via softmax
            y_shifted = y - y.max()
            a_y = np.exp(y_shifted)
            a_y = a_y / a_y.sum()

            # Compute gradient of smoothed objective w.r.t. x
            c = fast_conv(a_y)
            S = 1.0
            c_max = float(c.max())
            scaled = beta * (c - c_max)
            exp_scaled = np.exp(np.clip(scaled, -50.0, 50.0))
            p = exp_scaled / exp_scaled.sum()
            log_sum = c_max + math.log(float(exp_scaled.sum())) / beta

            # dLSE/da[j] = 2 * cross_corr(a, p)[j]
            L = 2 * n - 1
            L_conv = 3 * n - 2
            L_fft = 1 << (L_conv - 1).bit_length()
            A_fft = np.fft.rfft(a_y, L_fft)
            P_fft = np.fft.rfft(p, L_fft)
            x_corr = np.fft.irfft(np.conj(A_fft) * P_fft, L_fft)
            dLSE_da = 2.0 * x_corr[:n]

            # dF/da[j] = 2n*(dLSE_da[j] - 2*log_sum*a_y[j]) (with S=1)
            grad_a = 2.0 * n * (dLSE_da - 2.0 * log_sum * a_y)

            # Chain rule: dF/dx[j] = sum_k dF/da[k] * da[k]/dx[j]
            # da[k]/dx[j] = a_y[k] * (delta[k,j] - a_y[j])
            # dF/dx[j] = a_y[j] * (grad_a[j] - sum_k a_y[k]*grad_a[k])
            grad_a_mean = float(np.dot(a_y, grad_a))
            grad_x = a_y * (grad_a - grad_a_mean)

            # Nesterov update
            lr_t = lr_init / (1.0 + 0.03 * t / 1000.0)
            v = momentum * v + lr_t * grad_x
            x = x - v

            if (t + 1) % 500 == 0:
                x_shifted = x - x.max()
                a_check = np.exp(x_shifted)
                a_check = a_check / a_check.sum()
                sc = compute_score(a_check)
                if sc < best_score:
                    best_score = sc
                    best_a = a_check.copy()

        if time.monotonic() > deadline - 1.0:
            break

    x_shifted = x - x.max()
    a_final = np.exp(x_shifted)
    a_final = a_final / a_final.sum()
    sc = compute_score(a_final)
    if sc < best_score:
        best_score = sc
        best_a = a_final.copy()

    return best_score, best_a


def random_exp_init(n, rng):
    """Exponential random initialization."""
    a = rng.exponential(1.0, size=n)
    return a / a.sum()


def hamming_window_init(n, rng):
    """Hamming window with random perturbation."""
    x = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * x / (n - 1))
    a += 0.05 * rng.uniform(0, 1, size=n)
    return a / a.sum()


def two_bump_init(n, rng):
    """Two Gaussian bumps at random locations."""
    a = np.zeros(n)
    for _ in range(2):
        center = float(rng.integers(n // 4, 3 * n // 4))
        width = float(rng.uniform(n / 20, n / 5))
        amp = float(rng.uniform(0.5, 1.0))
        x = np.arange(n, dtype=np.float64)
        a += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
    return a / a.sum()


def spectral_flat_init(n, rng):
    """Initialize with roughly flat spectrum (random phases)."""
    M_fft = 1 << (2 * n - 2).bit_length()
    A = np.zeros(M_fft // 2 + 1, dtype=np.complex128)
    A[0] = math.sqrt(M_fft)
    for m in range(1, len(A)):
        phase = rng.uniform(0, 2 * np.pi)
        A[m] = 1.0 / math.sqrt(len(A) - 1) * np.exp(1j * phase)
    a = np.fft.irfft(A, M_fft)[:n]
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        return np.ones(n) / n
    return a / s

def coord_descent_local(a, time_budget, t_start, top_k=30):
    """
    Local coordinate descent: for the top-k coordinates by gradient magnitude,
    do a 1D golden section search.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget
    gr = (math.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(5):
        if time.monotonic() > deadline - 2.0:
            break

        # Compute subgradient direction
        c = fast_conv(a)
        S = a.sum()
        k_star = int(c.argmax())
        M_val = float(c[k_star])
        S2 = S * S

        # Subgradient
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = 2.0 * a[idx]
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val * a / (S2 * S))

        # Select top-k by |gradient|
        order = np.argsort(np.abs(grad))[::-1][:top_k]

        for j in order:
            if time.monotonic() > deadline - 2.0:
                break
            j = int(j)
            a_old = a[j]
            if a_old < 1e-10:
                candidates = [0.0, 0.01 * a.max(), 0.05 * a.max()]
            else:
                lo = 0.0
                hi = a_old * 3.0
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)
                a[j] = c1
                f1 = compute_score(a)
                a[j] = c2
                f2 = compute_score(a)
                for _ in range(12):
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        a[j] = c1
                        f1 = compute_score(a)
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        a[j] = c2
                        f2 = compute_score(a)
                best_val = c1 if f1 < f2 else c2
                a[j] = best_val

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_score, best_a

    def make_perturbation_base(a, rng, method):
        n = len(a)
        if method == 0:  # Spectral perturbation
            M = 2 * n - 1
            M_fft = 1 << (M - 1).bit_length()
            A = np.fft.rfft(a, M_fft)
            amp = 1.0 + 0.2 * rng.standard_normal(len(A))
            amp = np.clip(amp, 0.3, 2.0)
            phase = 0.15 * rng.standard_normal(len(A))
            A_new = A * amp * np.exp(1j * phase)
            a_new = np.fft.irfft(A_new, M_fft)[:n]
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s < 1e-10:
                return a.copy()
            return a_new / s

        elif method == 1:  # Anti-palindromic tilt
            x = np.arange(n, dtype=np.float64)
            center = (n - 1) / 2.0
            freq = float(rng.uniform(0.5, 3.0))
            tilt = np.sin(2.0 * np.pi * freq * x / n)
            tilt_anti = 0.5 * (tilt - tilt[::-1])
            a_new = a + 0.2 * tilt_anti * a.max()
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s < 1e-10:
                return a.copy()
            return a_new / s

        elif method == 2:  # Block permutation
            a_new = a.copy()
            for _ in range(3):
                bl = int(rng.integers(n // 16, n // 4))
                i = int(rng.integers(0, n - bl))
                j = int(rng.integers(0, n - bl))
                if i != j:
                    a_new[i:i+bl], a_new[j:j+bl] = a_new[j:j+bl].copy(), a_new[i:i+bl].copy()
            return a_new / a_new.sum()

        elif method == 3:  # Add random bumps
            a_new = a.copy()
            for _ in range(4):
                c = float(rng.integers(0, n))
                w = float(rng.uniform(n / 30, n / 8))
                amp = float(rng.uniform(-0.1, 0.1)) * a.max()
                x = np.arange(n, dtype=np.float64)
                a_new += amp * np.exp(-0.5 * ((x - c) / w) ** 2)
            a_new = np.clip(a_new, 0.0, None)
            s = a_new.sum()
            if s < 1e-10:
                return a.copy()
            return a_new / s

        else:  # Method 4: Partial randomization
            a_new = a.copy()
            frac = 0.3
            n_rand = int(frac * n)
            indices = rng.choice(n, size=n_rand, replace=False)
            a_new[indices] = rng.exponential(a.mean(), size=n_rand)
            return a_new / a_new.sum()

    for ri in range(n_restarts):
        if time.monotonic() > p2_deadline - 15:
            break

        method = ri % 5
        a_pert = make_perturbation_base(best_a, rng, method)

        time_left = p2_deadline - time.monotonic() - 10
        time_alloc = min(time_left * 0.85, per_restart * 1.5)
        if time_alloc < 10:
            break

        sc, a_opt = refine_sequence(a_pert, time_alloc, t_start)
        try_update(sc, a_opt, f"restart#{ri} method={method}")

    # ================================================================
    # Phase 3: Try different sequence lengths (15% budget)
    # ================================================================
    p3_budget = effective_budget * 0.15
    p3_deadline = t_start + effective_budget * 0.80
    print(f"\n{'='*60}")
    print(f"Phase 3: Different lengths (budget={p3_budget:.0f}s)")
    print(f"{'='*60}")

    n_cur = len(best_a)
    # Try nearby and larger sizes
    size_candidates = []
    for factor in [0.75, 1.25, 1.5, 2.0]:
        tl = int(n_cur * factor)
        if 1024 <= tl <= 32000:
            size_candidates.append(tl)
    # Also try some absolute sizes
    for abs_n in [4096, 8192, 12288, 16384, 20480]:
        if abs_n not in size_candidates and 1024 <= abs_n <= 32000:
            size_candidates.append(abs_n)

    for tl in size_candidates[:6]:
        if time.monotonic() > p3_deadline - 10:
            break
        time_left = p3_deadline - time.monotonic() - 10
        if time_left < 15:
            break

        # Resample from best_a
        x_old = np.arange(n_cur, dtype=np.float64)
        x_new = np.linspace(0, n_cur - 1, tl)
        a_res = np.interp(x_new, x_old, best_a)
        a_res = np.clip(a_res, 0.0, None)
        if a_res.sum() < 1e-10:
            a_res = random_exp_init(tl, rng)

        time_alloc = min(time_left * 0.7, 60.0)
        sc, a_opt = refine_sequence(a_res, time_alloc, t_start)
        try_update(sc, a_opt, f"n={tl}")

    # ================================================================
    # Phase 4: Fresh random initializations at promising sizes (10% budget)
    # ================================================================
    p4_budget = effective_budget * 0.10
    p4_deadline = t_start + effective_budget * 0.90
    print(f"\n{'='*60}")
    print(f"Phase 4: Fresh random inits (budget={p4_budget:.0f}s)")
    print(f"{'='*60}")

    n_fresh = len(best_a)
    inits = [
        ("exp", lambda: random_exp_init(n_fresh, rng)),
        ("hamming", lambda: hamming_window_init(n_fresh, rng)),
        ("2bump", lambda: two_bump_init(n_fresh, rng)),
        ("spectral", lambda: spectral_flat_init(n_fresh, rng)),
    ]

    for name, init_fn in inits:
        if time.monotonic() > p4_deadline - 10:
            break
        time_left = p4_deadline - time.monotonic() - 10
        if time_left < 15:
            break

        a_new = init_fn()
        time_alloc = min(time_left * 0.6, 45.0)
        sc, a_opt = refine_sequence(a_new, time_alloc, t_start)
        try_update(sc, a_opt, f"fresh {name} n={n_fresh}")

    # ================================================================
    # Phase 5: Final intensive refinement (remaining time)
    # ================================================================
    final_deadline = t_start + effective_budget
    print(f"\n{'='*60}")
    print(f"Phase 5: Final intensive refinement")
    print(f"{'='*60}")

    if time.monotonic() < final_deadline - 20:
        t_rem = final_deadline - time.monotonic() - 15

        # Long Nesterov LSE with high beta
        t1 = min(t_rem * 0.4, 120.0)
        if t1 > 10:
            sc, a_opt = nesterov_lse_simplex(best_a, [1000, 5000, 15000, 50000], t1, t_start,
                                              lr_init=0.01, momentum=0.95)
            try_update(sc, a_opt, "Final Nesterov LSE high-beta")

        # Coordinate descent polish
        t2 = min(t_rem * 0.3, 60.0)
        if t2 > 5:
            sc, a_opt = coord_descent_local(best_a, t2, t_start, top_k=50)
            try_update(sc, a_opt, "Final CoordDesc")

        # One more spectral pass
        t3 = min(t_rem * 0.2, 40.0)
        if t3 > 5:
            sc, a_opt = spectral_water_filling(best_a, t3, t_start, n_iters=50, alpha=0.2)
            try_update(sc, a_opt, "Final Spectral")

    # ================================================================
    # Final output
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 128

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
