Combines parametric basis optimization, LSE-smoothed L-BFGS,

# --- Core computation ---

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

# --- LSE objective and gradient ---

def lse_obj(x, beta, n):
    """LSE-smoothed score (scale-invariant) and gradient."""
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    a = x / S
    c = fast_conv(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L
    p = exp_s / exp_s.sum()
    # Gradient: d(score)/dx[i]
    # score = 2n * L(a) where a = x/S
    # dL/dx[i] = (dL/da[i]) * (S - x[i]) / S^2
    # dL/da[i] = 2 * (rev(a) * p)[n-1+i]
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    d_a_dx = (S - x) / (S * S)
    grad = 2.0 * n * dL_da * d_a_dx
    return score, grad

    bounds = [(0.0, 1000.0)] * n

        return lse_obj(x, beta, n)

                   options={'ftol': 1e-18, 'gtol': 1e-14,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out

    """
    LSE-smoothed scale-invariant score and its gradient.
    score(x) = 2n * LSE_beta(conv(x/sum(x)))
    """
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    a = x / S
    c = fast_conv(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L
    p = exp_s / exp_s.sum()
    # dL/da[i] = 2 * sum_{j: i+j valid} p[i+j] * a[j]
    # = 2 * (a_rev * p)[n-1+i] where a_rev[i] = a[n-1-i]
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    # Chain rule: a[i] = x[i]/S, da[i]/dx[i] = (S - x[i])/S^2
    d_a_dx = (S - x) / (S * S)
    grad = 2.0 * n * dL_da * d_a_dx
    return score, grad


def run_lbfgs(a0, beta, maxiter, maxfun):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0_c = np.maximum(a0.astype(np.float64), 1e-10)
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        return lse_obj(x, beta, n)

    res = minimize(obj, a0_c, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-18, 'gtol': 1e-14,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out


def parametric_sequence(params, n, M_cos):
    """
    Generate sequence from parametric representation:
    a(x) = (x*(1-x))^p * (1 + sum_{k=1}^{M} d_k * cos(pi*k*x))
    """
    p = max(params[0], 0.05)
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)
    base = (x * (1.0 - x)) ** p
    if base.sum() < 1e-15:
        return np.ones(n) / n
    perturb = np.zeros(n)
    for k in range(1, M_cos + 1):
        perturb += params[k] * np.cos(np.pi * k * x)
    a = base * (1.0 + perturb)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        return np.ones(n) / n
    return a / a.sum()


def parametric_score(params, n, M_cos):
    a = parametric_sequence(params, n, M_cos)
    return compute_score(a)


def peak_targeting(a, n_iters, n_top_peaks=3, n_top_contrib=30, step_scale=0.01):
    """
    Reduce top convolution peaks by adjusting contributing elements.
    Redistributes mass from high-contributor pairs to low-contributor pairs.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        peak_idx = np.argsort(c)[::-1][:n_top_peaks]

        for k_star in peak_idx:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end < j_start:
                continue

            seg = a[j_start:j_end + 1]
            # Contribution of a[j_start+i] to conv(k*) is seg[i] * seg[len-1-i]
            contrib = seg * seg[::-1]
            total_contrib = contrib.sum()
            if total_contrib < 1e-15:
                continue

            top_count = min(n_top_contrib, len(contrib))
            sorted_idx = np.argsort(contrib)[::-1]

            # Reduce top half
            n_reduce = max(1, top_count // 2)
            total_red = 0.0
            for li in sorted_idx[:n_reduce]:
                gi = j_start + li
                red = a[gi] * step_scale
                a[gi] = max(a[gi] - red, 0.0)
                total_red += red

            # Add to bottom half
            if total_red > 1e-15:
                n_add = max(1, top_count - n_reduce)
                bottom_idx = sorted_idx[n_reduce:n_reduce + n_add]
                per = total_red / len(bottom_idx)
                for li in bottom_idx:
                    gi = j_start + li
                    a[gi] += per

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def spectral_flattening(a, n_iters=100, alpha=0.5):
    """
    Frequency-domain error reduction: flatten the convolution spectrum.
    Adjusts a in the frequency domain to make conv(a) more uniform.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    M = 1 << (2 * n - 1).bit_length()
    a_pad = np.zeros(M, dtype=np.float64)
    a_pad[:n] = a
    A = np.fft.rfft(a_pad, M)

    for it in range(n_iters):
        v = A * A
        conv = np.fft.irfft(v, M)[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-18:
            break

        # Soft threshold: reduce excess above mean
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-18:
            break

        conv_mod = conv - 0.6 * excess
        err = conv_mod - conv

        err_pad = np.zeros(M, dtype=np.float64)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_pad, M)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-12] = 0.0

        damp = 0.3 + 0.5 * (it / max(1, n_iters))
        A = A + damp * d

        a_new = np.fft.irfft(A, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
        # Update A
        a_pad = np.zeros(M, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)

    return best_a


def mirror_descent(a, n_iters=2000, eta0=0.05):
    """Mirror descent (entropic) on the simplex."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))

        grad = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            grad[j_start:j_end + 1] = 2.0 * seg[::-1]

        eta = eta0 / (1.0 + 0.05 * it)
        log_a = np.log(a) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def subgradient_polish(a, total_iters=3000):
    """Projected subgradient with adaptive step size."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(total_iters):
        c = fast_conv(a)
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of max(conv) at a
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]

        # Project to tangent space of simplex
        sg = sg - np.mean(sg)
        gn = float(np.linalg.norm(sg))
        if gn < 1e-18:
            break

        # Polyak step size
        step = max(0.001, min(0.05, 0.5 / (it + 1) / gn))
        a = a - step * sg
        a = np.maximum(a, 0.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def coordinate_descent_1d(a, n_coord_trials=100, n_window=10):
    """
    1D coordinate descent: for random positions, try to optimize a[i]
    by line search, while keeping sum fixed.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    rng = np.random.default_rng(12345)

    for trial in range(n_coord_trials):
        i = rng.integers(0, n)
        # Try different values for a[i], adjusting a random other position to keep sum
        j = rng.integers(0, n)
        if i == j:
            j = (i + 1) % n

        base_sc = best_sc
        best_val = a[i]

        # Try a range of values
        scale = a[i] + 1e-8
        for frac in [0.3, 0.5, 0.7, 1.3, 1.7, 2.0, 0.1, 3.0]:
            new_val_i = a[i] * frac
            new_val_j = a[j] + (a[i] - new_val_i)
            if new_val_i < 0 or new_val_j < 0:
                continue
            a[i] = new_val_i
            a[j] = new_val_j
            sc = compute_score(a)
            if sc < base_sc:
                base_sc = sc
                best_val = new_val_i
                best_j_val = new_val_j
        # Restore if no improvement
        a[i] = best_val
        if base_sc < best_sc:
            best_sc = base_sc
            best_a = a.copy()

    return best_a


def build_diverse_inits(n, rng, parent=None):
    """Generate diverse initialization candidates."""
    inits = []
    x = np.arange(n, dtype=np.float64) / max(n - 1, 1)

    if parent is not None:
        if len(parent) == n:
            inits.append(np.maximum(parent.astype(np.float64), 0.0).copy())
        else:
            inits.append(np.interp(x, np.linspace(0, 1, len(parent)),
                                   np.maximum(parent, 0.0)).copy())

    # Parabolic with various powers
    for p in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        inits.append(np.maximum((x * (1 - x)) ** p, 0.0))

    # Cosine-based
    inits.append(0.5 + 0.4 * np.cos(np.pi * x))
    inits.append(0.3 + 0.6 * np.cos(np.pi * x) ** 2)
    inits.append(0.5 + 0.3 * np.cos(np.pi * x) * np.cos(2 * np.pi * x))
    inits.append(0.4 + 0.5 * (1 - x * x))

    # Random smooth
    for std in [0.1, 0.3, 0.5]:
        raw = 0.5 + std * rng.standard_normal(n)
        # Smooth it
        kernel = np.exp(-0.5 * (np.arange(-20, 21) / 5.0) ** 2)
        kernel = kernel / kernel.sum()
        raw = np.convolve(raw, kernel, mode='same')
        inits.append(np.clip(raw, 0.01, 10.0))

    # Block structure
    for nb in [4, 8, 16]:
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())

    # Normalize all
    inits = [a / max(a.sum(), 1e-15) for a in inits]
    return inits
