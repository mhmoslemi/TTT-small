def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_and_grad(a, beta):
    """
    Compute LSE-smoothed score and its gradient.
    score = 2n * LSE(b) / S^2
    where LSE(b) = max(b) + log(sum(exp(beta*(b - max(b))))) / beta
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    # For numerical stability, shift by c_max
    scaled = beta * (c - c_max)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    
    # p_k = exp(beta*(b[k]-max)) / sum(exp(...)) = softmax(beta*(b-max))
    p = exp_scaled / exp_scaled.sum()
    
    score = 2.0 * n * L / (S * S)
    
    # Gradient of L w.r.t. a[i]:
    # L = c_max + (1/beta) * log(sum_k exp(beta*(c[k]-c_max)))
    # dL/da[i] = sum_k p_k * dc[k]/da[i]
    # dc[k]/da[i] = a[k-i] (if valid) + a[i-k] (if valid, and i != k-i)
    #              = (a[k-i] if 0<=k-i<n else 0) + (a[i-k] if 0<=i-k<n else 0)
    # But if k-i == i-k (i.e. k=2i), we get 2*a[i]

    # Wait, (p*a)[m] = sum_k p[k]*a[m-k]. So (p*a)[i] = sum_k p[k]*a[i-k]. That's the second sum!

    # dc[k]/da[i] = u[k-i] + u[i-k] where u[j] = a[j] if 0<=j<n else 0
    # (with the understanding that if k-i = i-k, i.e., k=2i, the contribution is 2*a[i])
    
    # sum_k p[k] * dc[k]/da[i] = sum_k p[k]*u[k-i] + sum_k p[k]*u[i-k]
    # = sum_j p[j+i]*u[j] + sum_j p[i+j]... 
    # Hmm, let me think differently.
    
    # sum_k p[k] * u[k-i]: let j = k-i, so k = j+i. Sum over j where 0<=j<n and 0<=j+i<2n-1.
    # = sum_j p[j+i] * a[j] (with p indexed from 0 to 2n-2)
    # This is a cross-correlation of p and a.
    
    # sum_k p[k] * u[i-k]: let j = i-k, so k = i-j. Sum over j where 0<=j<n and 0<=i-j<2n-1.
    # = sum_j p[i-j] * a[j]
    # This is a convolution of p and a.
    
    # Using FFT:
    # cross_corr[i] = sum_j p[j+i] * a[j] = (p_rev * a)[n-1+i] where p_rev is reversed
    # Actually, let me just use the direct formula with FFT.
    
    # Define the "response" kernel: for each i, dL/da[i] = sum_k p[k] * (a[k-i] + a[i-k])

    # a[i] appears in: a[j]*a[i-j] when j=i (gives a[i]*a[0]) and when i-j=i, i.e., j=0 (gives a[0]*a[i]).

    # dc[1]/da[1] = d(4)/da[1]: c[1] = a[0]*a[1] + a[1]*a[0] = 2*1*2 = 4
    # dc[1]/da[1] = a[0] + a[0] = 2*a[0] = 2. (k=1, i=1, k≠2i, so a[k-i]+a[i-k] = a[0]+a[0] = 2)

    # So the original code's gradient formula might be WRONG in general!

    # Let me fix this.
    
    # Correct gradient:

    # (a*q)[m] = sum_j a[j]*q[m-j] = sum_j a[j]*p[2n-2-(m-j)]

    # Actually, for real signals, FFT of the time-reversed signal is the conjugate of the FFT:
    # If q[j] = p[N-1-j] (circular reversal), then FFT(q)[k] = conj(FFT(p)[k]) * exp(-i*2πk(N-1)/N)
    # But for linear reversal (q[j] = p[2n-2-j] with zero padding to length M_fft), it's different.
    # Let me just compute it directly.
    
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    
    A_fft = np.fft.rfft(a, M2_fft)
    P_fft = np.fft.rfft(p, M2_fft)
    
    conv1 = np.fft.irfft(A_fft * P_fft, M2_fft)[:M2]  # (a * p)
    
    # flip(p): p has length 2n-1, flip it to get p_flip of length 2n-1
    p_flip = p[::-1]
    P_flip_fft = np.fft.rfft(p_flip, M2_fft)
    conv2 = np.fft.irfft(A_fft * P_flip_fft, M2_fft)[:M2]  # (a * flip(p))
    
    # grad_L[i] = conv1[i] + conv2[2n-2-i] for i = 0, ..., n-1
    grad_L = conv1[:n] + conv2[2*n-2-n+1:2*n-2+1][::-1]
    # Wait, conv2[2n-2-i] for i=0,...,n-1 gives indices 2n-2, 2n-3, ..., n-1
    # That's conv2[n-1 : 2n-1] reversed.
    grad_L = conv1[:n] + conv2[n-1:2*n-1][::-1]
    
    # Now the full score gradient:
    # score = 2n * L / S^2
    # d(score)/da[i] = 2n * (grad_L[i] * S^2 - L * 2S) / S^4
    # = 2n * (grad_L[i] / S^2 - 2L/S^3)
    
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L / S3)
    
    return score, grad


def adam_optimize(a_init, beta_start=50, beta_end=500, lr=0.01, n_steps=20000,
                  time_limit_s=None, t_start=None):
    """Adam optimizer with adaptive LSE beta."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S0 = a.sum()
    if S0 < 1e-12:
        a[:] = 1.0 / n
        S0 = 1.0
    a = a / S0  # Normalize to sum=1
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break
        
        # Adaptive beta: linear interpolation from beta_start to beta_end
        frac = min(1.0, t / (n_steps * 0.8))
        beta = beta_start + (beta_end - beta_start) * frac
        
        # Cyclical learning rate
        cycle = 2000
        lr_t = lr * (0.5 + 0.5 * np.cos(np.pi * (t % cycle) / cycle))
        
        score, grad = lse_score_and_grad(a, beta)
        
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 1e-12:
            a = a / s
        
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a


def mass_redistribution(a, n_iters=100, time_limit_s=None, t_start=None):
    """
    Local search: redistribute mass from high-gradient to low-gradient coordinates.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_limit_s) if (time_limit_s and t_start) else np.inf
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        score, grad = lse_score_and_grad(a, beta=200)
        
        # Find coordinates to adjust
        # High gradient: should have LESS mass (increase a to reduce score? No...)
        # Actually, if grad[i] > 0, increasing a[i] increases score, so we want to DECREASE a[i]
        # If grad[i] < 0, increasing a[i] decreases score, so we want to INCREASE a[i]
        
        # Normalize gradient (subtract mean to make it zero-sum)
        mask = a > 1e-10
        if mask.sum() < 2:
            break
        g_masked = grad[mask]
        g_mean = g_masked.mean()
        g_centered = grad - g_mean  # Zero-mean in the supported set
        
        # Find donor (high grad, has mass) and receiver (low grad)
        donors = np.where(mask & (g_centered > 0))[0]
        receivers = np.where(mask & (g_centered < 0))[0]
        
        if len(donors) == 0 or len(receivers) == 0:
            # Try adding from zero coordinates
            zero_coords = np.where(~mask)[0]
            if len(zero_coords) > 0 and len(donors) == 0:
                break
            if len(zero_coords) > 0 and len(receivers) == 0:
                receivers = zero_coords[np.argsort(grad[zero_coords])[:5]]
            if len(donors) == 0:
                donors = np.argsort(a)[::-1][:5]
            if len(receivers) == 0:
                break
        
        # Take top donors and receivers
        n_move = min(10, len(donors), len(receivers))
        top_donors = donors[np.argsort(-g_centered[donors])[:n_move]]
        top_receivers = receivers[np.argsort(g_centered[receivers])[:n_move]]
        
        moved = False
        for d_idx in top_donors:
            for r_idx in top_receivers:
                if a[d_idx] < 1e-10:
                    continue
                # Try moving a small fraction
                for frac in [0.1, 0.05, 0.02]:
                    delta = a[d_idx] * frac
                    a[d_idx] -= delta
                    a[r_idx] += delta
                    sc = compute_score(a)
                    if sc < best_score:
                        best_score = sc
                        moved = True
                    else:
                        a[d_idx] += delta
                        a[r_idx] -= delta
                        break
                if moved:
                    best_score = compute_score(a)
                    break
    
    return best_score, a


def lbfgs_optimize(a0, beta, maxiter=5000, maxfun=100000):
    """L-BFGS-B optimization."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
        S = 1.0
    a0 = a0 / S  # Normalize
    bounds = [(0.0, None)] * n
    res = minimize(
        lse_score_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    S_out = a_out.sum()
    if S_out < 1e-12:
        a_out = a0.copy()
    else:
        a_out = a_out / S_out
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res
