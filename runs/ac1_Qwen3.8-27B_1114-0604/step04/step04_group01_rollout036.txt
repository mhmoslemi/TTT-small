"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using LSE-smoothed optimization with aggressive continuation, exact subgradient
polishing, mirror descent steps, and multi-length exploration.
Key improvements over previous: higher final beta, more iterations at high beta,
exact subgradient refinement, adaptive active-set detection, and softmax
reparametrization for final polish.
"""

import numpy as np
import time
from scipy.optimize import minimize

PEN_SUM = 1e-3


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    a = a * (float(target) / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    n = int(n)
    if m == 0:
        return np.ones(n, dtype=np.float64) / float(max(1, n))
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def lse_score_grad_tuple(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    grad = np.zeros(n, dtype=np.float64)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, grad
    beta = float(beta)
    if not np.isfinite(beta) or beta <= 1e-6:
        beta = 100.0
    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, grad
    z = beta * (d - dmax)
    np.exp(z, out=z)
    sum_e = float(z.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, grad
    w = z / sum_e
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    h = corr_pair(a, w)
    score = 2.0 * L + PEN_SUM * (S - 1.0) * (S - 1.0)
    grad = (4.0 * float(n) * h) / (S * S)
    grad -= (4.0 * LDS) / S
    grad += 2.0 * PEN_SUM * (S - 1.0)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def fun_lse_tuple(a, beta):
    return lse_score_grad_tuple(a, float(beta))


def estimate_eval_time(n):
    base = 0.02
    scale = max(0.5, float(n) / 8192.0)
    log_scale = max(1.0, float(np.log2(max(16, n))) / float(np.log2(8192.0)))
    return base * scale * log_scale


def estimate_maxfun(t_left, n):
    t_left = float(max(1.0, t_left))
    if t_left < 4.0:
        return 64
    est = estimate_eval_time(n)
    mf = int((0.75 * t_left) / max(est, 1e-4))
    if mf < 64:
        mf = 64
    if mf > 5000:
        mf = 5000
    return mf


def run_lbfgs_tuple(obj_func, a0, param, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 4.0:
            return exact_score_a(a), a
        est = estimate_eval_time(n)
        cap = int((0.9 * rem) / max(est, 1e-4))
        if cap < 64:
            cap = 64
        maxfun = int(min(int(maxfun), cap))
    else:
        maxfun = int(maxfun if maxfun is not None else 1000)
    if maxfun < 64:
        maxfun = 64
    maxiter = max(80, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-18,
        "gtol": 1e-14,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 40,
    }
    try:
        res = minimize(
            obj_func, a,
            method="L-BFGS-B",
            jac=True,
            args=(param,),
            bounds=bounds,
            options=opts,
        )
    except Exception:
        return exact_score_a(a), a
    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def run_mode(a0, beta, deadline, maxfun):
    return run_lbfgs_tuple(fun_lse_tuple, a0, float(beta), deadline, maxfun)


def generate_inits(n, rng, parent, cap=28):
    inits = []
    n = int(n)

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size != n:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0.0:
            arr = arr - mn + 0.05
        p = normalize_a(arr)
        inits.append(p)

    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])

    x = np.linspace(0.0, 1.0, n)
    add(np.ones(n, dtype=np.float64))

    for alpha in (0.2, 0.4, 0.6, 0.8):
        add(1.0 + alpha * (2.0 * x - 1.0))
        add(1.0 - alpha * (2.0 * x - 1.0))

    max_trig = 8
    for r in range(max_trig):
        vals = np.ones(n, dtype=np.float64)
        K = int(rng.integers(2, 5))
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 10)))
            amp = float(rng.uniform(0.05, 0.25))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
        add(vals)

    for frac in (0.4, 0.6, 0.75):
        mask = rng.random(n) < frac
        vals = np.where(mask, 0.3 + 0.7 * rng.random(n), 0.0)
        add(vals)

    for width in (0.05, 0.1, 0.15, 0.25):
        vals = 0.3 + 0.7 * (
            np.exp(-((x / width) ** 2)) + np.exp(-(((1.0 - x) / width) ** 2))
        )
        add(vals)

    for _ in range(4):
        add(0.1 + 0.9 * rng.random(n))

    return inits


def active_weights(d, k):
    m = int(d.size)
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m, dtype=np.float64) / float(m)
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / float(k)


def pressure_direction(p, k=3):
    d, S = d_from_a(p)
    n = len(p)
    if S < 1e-12 or d.size == 0:
        return np.zeros(n, dtype=np.float64)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return np.zeros(n, dtype=np.float64)
    w = active_weights(d, k)
    return corr_pair(p, w)


def perturb_best(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.15, 0.60))
    mode = int(rng.integers(0, 6))
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.30))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        q = p[::-1].copy()
        q = q * np.exp(0.08 * rng.standard_normal(n))
    elif mode == 3:
        h = pressure_direction(p, 4)
        hmax = float(np.max(h))
        if hmax < 1e-12:
            q = p.copy()
        else:
            q = p * np.exp(-strength * h / hmax)
    elif mode == 4:
        logp = np.log(p + 1e-300)
        noise = rng.standard_normal(n)
        k = min(21, max(3, int(n) // 200))
        if k % 2 == 0:
            k += 1
        if k > n:
            k = n if n % 2 == 1 else max(3, n - 1)
        if k >= 3 and k <= n:
            c = k // 2
            sig = max(1.0, c / 2.0)
            kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
            kern = kern / float(kern.sum())
            noise = np.convolve(noise, kern, mode="same")
        noise = noise - float(noise.mean())
        logp = logp + strength * 0.4 * noise
        q = np.exp(logp - float(np.max(logp)))
    else:
        k = int(rng.integers(5, 21))
        if k % 2 == 0:
            k += 1
        if k > n:
            k = n if n % 2 == 1 else max(3, n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        knorm = float(kern.sum())
        if knorm > 0.0:
            kern = kern / knorm
        q = np.convolve(p, kern, mode="same")
    return normalize_a(q)


def subgrad_polish(a, max_iters=30, deadline=None):
    """Polish using exact subgradient of the max convolution lag."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    cur_sc = best_sc
    n = len(a)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        d, S = d_from_a(a)
        if S < 1e-12 or d.size == 0:
            break
        dmax = float(np.max(d))
        if not np.isfinite(dmax) or dmax <= 0.0:
            break

        cur_a = None
        cur_new_sc = cur_sc

        for k in (1, 2, 3, 5, 8):
            w = active_weights(d, k)
            h = corr_pair(a, w)
            hmax = float(np.max(h))
            if hmax < 1e-12:
                continue
            base = 1.0 / hmax
            for mult in (0.005, 0.01, 0.03, 0.08, 0.15, 0.30, 0.50):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                eta = mult * base
                q = a * np.exp(-eta * h)
                q = normalize_a(q)
                sc = exact_score_a(q)
                if not np.isfinite(sc):
                    continue
                if sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                if sc < cur_new_sc:
                    cur_new_sc = sc
                    cur_a = q.copy()

        if cur_a is None:
            break
        a = cur_a
        cur_sc = cur_new_sc

    return float(best_sc), best_a


def mirror_descent_polish(a, max_iters=50, deadline=None):
    """Mirror descent with adaptive step size on the exact objective."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    n = len(a)
    eta = 0.01

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        d, S = d_from_a(a)
        if d.size == 0 or S < 1e-12:
            break
        dmax = float(np.max(d))
        if dmax <= 0:
            break
        # Find active set (within 5% of max)
        threshold = dmax * 0.95
        active = d >= threshold
        k_active = int(active.sum())
        if k_active == 0:
            break
        w = active.astype(np.float64) / float(k_active)
        h = corr_pair(a, w)
        hmax = float(np.max(h))
        if hmax < 1e-12:
            break
        # Mirror descent step
        q = a * np.exp(-eta * h / hmax)
        q = normalize_a(q)
        sc = exact_score_a(q)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_a = q.copy()
            a = q
            eta *= 1.5
        else:
            eta *= 0.6
            if eta < 1e-6:
                break
    return float(best_sc), best_a

        n = int(n)
        if 1024 <= n <= 12288 and n not in unique_lengths:
            unique_lengths.append(n)
    if not unique_lengths:
        unique_lengths = [8192]
    lengths = unique_lengths[:3]
    print("Lengths to try: {}".format(lengths))

    # Generate initial candidates
    all_scores = []
    all_seqs = []
    all_lens = []

    for n in lengths:
        inits = generate_inits(n, rng, parent)
        for arr in inits:
            p = normalize_a(arr)
            sc = exact_score_a(p)
            if np.isfinite(sc):
                all_scores.append(float(sc))
                all_seqs.append(p)
                all_lens.append(int(n))

    if not all_scores:
        n0 = lengths[0]
        all_seqs.append(np.ones(n0) / float(n0))
        all_scores.append(float(exact_score_a(all_seqs[-1])))
        all_lens.append(n0)

    order = np.argsort(np.asarray(all_scores, dtype=np.float64))

    # Select starts
    starts = []
    used_lens = set()

    if parent is not None and parent_len > 0:
        ps = normalize_a(resample_sequence(parent, min(parent_len, 12288)))
        starts.append(ps)
        used_lens.add(min(parent_len, 12288))

    for idx in order:
        nidx = int(all_lens[idx])
        if nidx not in used_lens:
            starts.append(all_seqs[int(idx)].copy())
            used_lens.add(nidx)
            if len(starts) >= 3:
                break

    while len(starts) < 2:
        starts.append(perturb_best(starts[0], rng))

    best_seq = starts[0].copy()
    best_score = exact_score_a(best_seq)
    for i in range(1, len(starts)):
        sc = exact_score_a(starts[i])
        if sc < best_score:
            best_score = sc
            best_seq = starts[i].copy()

    print("Initial best score={:.8f} length={}".format(best_score, len(best_seq)))

    # Main optimization: allocate time across starts
    main_budget = max(120.0, deadline - t_start - 300.0)
    if len(starts) == 1:
        fr = [1.00]
    elif len(starts) == 2:
        fr = [0.65, 0.35]
    else:
        fr = [0.55, 0.30, 0.15]

    cum = 0.0
    for i, st in enumerate(starts):
        cum += fr[i]
        sub_dead = t_start + cum * main_budget
        sub_dead = min(float(sub_dead), float(deadline - 300.0))
        if time.monotonic() > max(t_start + 10.0, sub_dead - 30.0):
            continue

        sq = st
        if i > 0 and np.isfinite(best_score) and best_score < 2.0:
            base = best_seq
            if len(base) != len(sq):
                base = resample_sequence(base, len(sq))
            base = normalize_a(base)
            if i == 1:
                sq = base
            else:
                sq = perturb_best(base, rng)

        sc, seq = optimize_trajectory(sq, sub_dead, rng, "traj{}".format(i + 1))
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Global best after traj{}: {:.8f}".format(i + 1, best_score))

    best_seq = normalize_a(best_seq)
    print("After main trajectories: {:.8f} n={}".format(best_score, len(best_seq)))

    # Subgradient polish
    if time.monotonic() < deadline - 120.0:
        sc, seq = subgrad_polish(best_seq, max_iters=25, deadline=deadline - 120.0)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Subgradient polish: {:.8f}".format(best_score))

    # Mirror descent polish
    if time.monotonic() < deadline - 80.0:
        sc, seq = mirror_descent_polish(best_seq, max_iters=80, deadline=deadline - 80.0)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Mirror descent: {:.8f}".format(best_score))

    # More subgradient polish
    if time.monotonic() < deadline - 60.0:
        sc, seq = subgrad_polish(best_seq, max_iters=20, deadline=deadline - 60.0)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("2nd subgradient polish: {:.8f}".format(best_score))

    # Perturbation restarts
    perturb_done = 0
    while time.monotonic() < deadline - 80.0 and perturb_done < 4:
        left = float(deadline - 80.0 - time.monotonic())
        if left < 60.0:
            break
        chunk = time.monotonic() + left / float(4 - perturb_done)
        p = perturb_best(best_seq, rng, strength=float(rng.uniform(0.05, 0.25)))
        sc, seq = quick_optimize(p, chunk)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Perturb restart {} best: {:.8f}".format(perturb_done + 1, best_score))
        perturb_done += 1

    # Final high-beta LSE
    if time.monotonic() < deadline - 30.0:
        rem = float(deadline - time.monotonic() - 5.0)
        if rem > 15.0:
            mf = estimate_maxfun(rem * 0.85, len(best_seq))
            sc, seq = run_mode(best_seq, 100000.0, deadline, mf)
            if sc < best_score:
                best_score = sc
                best_seq = seq.copy()
                print("High beta final: {:.8f}".format(best_score))

    # Final subgradient
    if time.monotonic() < deadline - 10.0:
        sc, seq = subgrad_polish(best_seq, max_iters=10, deadline=deadline - 10.0)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Final subgradient: {:.8f}".format(best_score))

    # Prepare output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL internal best={:.8f} verified={:.8f} length={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start
    ))
    return out
