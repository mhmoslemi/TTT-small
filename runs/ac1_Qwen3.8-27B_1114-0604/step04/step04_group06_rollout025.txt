"""
Minimize 2*n*max(self-convolution)/(sum)^2 for non-negative sequences.
Strategy: Use POCS spectrum flattening for initialization, then LSE-smoothed
gradient descent with smooth annealing, followed by peak-targeting coordinate
descent. Exploit asymmetry (key to score < 2) via asymmetric initializations.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_smoothed_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def pocs_flatten(a, n_iters=100, n_peaks_to_keep=0):
    """POCS-based spectrum flattening.
    Iterates between time-domain (non-negative projection) and 
    frequency-domain (magnitude flattening)."""
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S  # Normalize to sum=1
    
    for it in range(n_iters):
        # Forward transform
        A = np.fft.rfft(a, M)
        # Flatten magnitude: set all magnitudes to the mean
        mags = np.abs(A)
        target_mag = mags.mean()
        # Create modified spectrum: keep phase, set magnitude to target
        phases = np.angle(A)
        A_new = target_mag * np.exp(1j * phases)
        # Inverse transform
        a_new = np.fft.irfft(A_new, M)
        # Project to non-negative and truncate
        a_new = np.clip(a_new, 0.0, None)[:n]
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        # Check convergence
        if it > 0 and np.max(np.abs(a_new - a)) < 1e-10:
            a = a_new
            break
        a = a_new
    
    return a


def pocs_soft_flatten(a, n_iters=200, alpha=0.5):
    """Softer POCS variant with partial magnitude flattening."""
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M)
        mags = np.abs(A)
        target_mag = mags.mean()
        # Partial flattening: blend current magnitude with target
        mags_new = (1 - alpha) * mags + alpha * target_mag
        phases = np.angle(A)
        A_new = mags_new * np.exp(1j * phases)
        a_new = np.fft.irfft(A_new, M)
        a_new = np.clip(a_new, 0.0, None)[:n]
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
    
    return a


def adam_lse_optimize(a_init, betas, lr_schedule, time_limit_s, t_start):
    """Adam optimization with LSE smoothing and smooth annealing over beta.
    
    betas: list of beta values to use (in order, typically increasing)
    lr_schedule: function of (beta_index, step) -> learning rate
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    a = a / S
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    t_global = 0
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for beta_idx, beta in enumerate(betas):
        if time.monotonic() - t_start > time_limit_s:
            break
        n_steps = 20000
        for t in range(1, n_steps + 1):
            if time.monotonic() - t_start > time_limit_s:
                break
            t_global += 1
            lr = lr_schedule(beta_idx, t)
            score, grad = lse_smoothed_score_grad(a, beta)
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            m_hat = m / (1.0 - b1 ** t_global)
            v_hat = v / (1.0 - b2 ** t_global)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s > 1e-12:
                a = a / s
            
            if t % 5000 == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a


def peak_targeting_descent(a, n_rounds=50, time_limit_s=None, t_start=None):
    """Coordinate descent that targets the peak of the convolution.
    
    At each round:
    1. Find the peak lag k* of the convolution
    2. The peak is sum_j a[j]*a[k*-j]. To reduce it, we should 
       adjust the a[j] values that contribute most to the peak.
    3. For each contributing j, try decreasing a[j] (and compensating 
       elsewhere) to reduce the peak.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return compute_score(a), a
    a = a / S
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf
    
    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        peak_val = c[k_star]
        
        # Find contributing indices
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je < js:
            continue
        
        # The peak is sum_{j=js}^{je} a[j] * a[k_star - j]
        # To reduce it, decrease the largest contributors
        contribs = a[js:je+1] * a[k_star - np.arange(js, je+1)]
        total_contrib = contribs.sum()
        
        # Try reducing top contributors
        n_top = min(10, je - js + 1)
        top_idx = np.argsort(contribs)[::-1][:n_top]
        
        for idx in top_idx:
            if time.monotonic() > deadline:
                break
            j = js + idx
            a_old = a[j]
            if a_old < 1e-10:
                continue
            
            # Try reducing a[j] by various amounts
            for frac in [0.7, 0.5, 0.3]:
                a[j] = a_old * frac
                # Renormalize
                s = a.sum()
                if s < 1e-12:
                    a[j] = a_old
                    break
                a = a / s
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                else:
                    # Revert
                    a = a * s  # Undo normalization
                    a[j] = a_old
                    break
    
    return best_score, a


def topk_coord_descent(a, top_k=20, n_sweeps=20, time_limit_s=None, t_start=None):
    """Coordinate descent on top-K coordinates with gradient information."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return compute_score(a), a
    a = a / S
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        # Compute LSE gradient for coordinate selection
        score, grad = lse_smoothed_score_grad(a, beta=100)
        # Select top-K coordinates by gradient magnitude
        order = np.argsort(np.abs(grad))[::-1][:top_k]
        
        for j in order:
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            
            if grad[j] > 1e-10:
                # Gradient suggests decreasing
                candidates = a_old * np.array([0.1, 0.3, 0.5, 0.7, 0.9])
            elif grad[j] < -1e-10:
                # Gradient suggests increasing
                candidates = np.concatenate([[0.0], a_old * np.array([1.5, 2.0, 4.0, 8.0])])
            else:
                candidates = np.array([a_old])
            
            best_x = a_old
            best_sc = score
            for x in candidates:
                a[j] = x
                s = a.sum()
                if s < 1e-12:
                    a[j] = a_old
                    continue
                a_norm = a / s
                sc = compute_score(a_norm)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
            a = a_norm if best_x != a_old else a
            a[j] = best_x
            s = a.sum()
            if s > 1e-12:
                a = a / s
        
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
    
    return best_score, a


def make_ramp(n, asymmetry=0.5):
    """Create an asymmetric ramp sequence.
    asymmetry controls how much more mass is on the right side.
    asymmetry=0 gives uniform, asymmetry=1 gives a pure ramp."""
    x = np.arange(n, dtype=np.float64)
    # Blend uniform and ramp
    a = (1 - asymmetry) * np.ones(n) + asymmetry * (x / (n - 1))
    a = np.maximum(a, 1e-6)
    a = a / a.sum()
    return a


def make_bimodal_asymmetric(n, ratio=0.3):
    """Create a bimodal asymmetric sequence."""
    x = np.arange(n, dtype=np.float64)
    # Two peaks, one larger
    center1 = n * ratio
    center2 = n * (1 - ratio)
    width1 = n * 0.1
    width2 = n * 0.15
    a = np.exp(-0.5 * ((x - center1) / width1) ** 2) + 0.7 * np.exp(-0.5 * ((x - center2) / width2) ** 2)
    a = a / a.sum()
    return a
