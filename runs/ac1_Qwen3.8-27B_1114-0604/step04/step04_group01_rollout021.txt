"""
Search for non-negative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LSE-smoothed gradient descent with aggressive continuation,
peak-targeted multiplicative updates, simulated annealing, and diversity.
"""

import numpy as np
import time
from scipy.optimize import minimize

def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_conv(a):
    n = len(a)
    if n == 0:
        return np.empty(0)
    M = 2*n - 1
    N = next_pow2(M)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:M]

def score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fast_conv(a)
    return float(2*n*max(0, c.max()) / (S*S))

def normalize(a, target=1.0):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-12:
        return np.full_like(a, target/a.size)
    return a * target / s

def lse_obj_grad(a, beta, pen=0.0):
    """LSE-smoothed objective + gradient for score = 2n*max(conv)/S^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-8:
        return 1e10, np.zeros_like(a)
    c = fast_conv(a)
    d = n * c / (S*S)  # normalized, score = 2*max(d)
    dmax = d.max()
    if dmax <= 0:
        return 0, np.zeros_like(a)
    z = beta * (d - dmax)
    # numerical stability
    z = z - z.max()
    e = np.exp(z)
    se = e.sum()
    w = e / se
    L = dmax + np.log(se) / beta
    
    # gradient of L w.r.t. a
    # L = dmax + log(sum exp(beta*(d-dmax)))/beta
    # dL/da = (1/beta) * sum_j w_j * dd_j/da
    # d_k = n * c_k / S^2
    # dc_k/da_i = a_{k-i} + a_{k-n+1-i} (the two terms in the convolution)
    # Actually: c_k = sum_j a_j * a_{k-j}
    # dc_k/da_i = a_{k-i} (if 0 <= k-i < n) + a_{k-i} (wait, no)
    # c_k = sum_{j: 0<=j<n, 0<=k-j<n} a_j * a_{k-j}
    # dc_k/da_i: i appears as j or as k-j
    # As j=i: contributes a_{k-i} (if valid)
    # As k-j=i, i.e., j=k-i: contributes a_{k-i} (if valid)
    # So dc_k/da_i = a_{k-i} * [0<=k-i<n] + a_{k-i} * [0<=k-i<n]
    # Wait, that's only if k-i is valid AND i is valid (which it is since i in [0,n))
    # If 0 <= k-i < n: dc_k/da_i = 2 * a_{k-i}
    # If i = k-i, i.e., k = 2i: dc_k/da_i = a_i (only one term, not double counted)
    # Hmm, let me think again.
    # c_k = sum_j a_j * a_{k-j}
    # dc_k/da_i: 
    #   Term where j=i: a_{k-i} (valid if 0<=k-i<n)
    #   Term where k-j=i, i.e., j=k-i: a_i (valid if 0<=k-i<n, same condition)
    #   But if i = k-i, i.e., 2i=k, these are the same term, so we count once.
    #   Otherwise (i != k-i), we count twice: a_{k-i} + a_i... wait no.
    #   Let me redo: c_k = sum_j a_j * a_{k-j}
    #   dc_k/da_i = sum_j [delta(j,i)*a_{k-j} + a_j*delta(k-j,i)]
    #             = a_{k-i} (if 0<=k-i<n) + a_{k-i} (if 0<=k-i<n)
    #   Wait, the second term: a_j*delta(k-j,i) = a_j when j=k-i, so = a_{k-i} (if 0<=k-i<n)
    #   So dc_k/da_i = 2*a_{k-i} if 0<=k-i<n, BUT if i=k-i (i.e., k=2i), 
    #   then the two contributions are the same, so it's just a_i.
    #   
    #   Actually no! When k=2i, j=i gives a_i*a_i, and this is ONE term in the sum.
    #   So dc_k/da_i at k=2i: the derivative of a_i*a_i w.r.t. a_i is 2*a_i.
    #   And from the other "side" (j=k-i=i), it's the same term.
    #   So actually dc_k/da_i = 2*a_{k-i} always (when valid), because we're 
    #   differentiating a product where both factors can be a_i.
    #   
    #   Let me verify: c_0 = a_0^2. dc_0/da_0 = 2*a_0. Formula gives 2*a_0. ✓
    #   c_1 = a_0*a_1 + a_1*a_0 = 2*a_0*a_1. dc_1/da_0 = 2*a_1. Formula: k=1, i=0, a_{1-0}=a_1. 2*a_1. ✓
    #   c_1 = 2*a_0*a_1. dc_1/da_1 = 2*a_0. Formula: k=1, i=1, a_{1-1}=a_0. 2*a_0. ✓
    
    # So dc_k/da_i = 2*a_{k-i} if 0<=k-i<n, else 0.
    # 
    # dd_k/da_i = n * (dc_k/da_i * S^2 - c_k * 2*S) / S^4
    #           = n * (2*a_{k-i}*S^2 - 2*c_k*S) / S^4  (when valid)
    #           = 2n * (a_{k-i}*S - c_k) / S^3  (when 0<=k-i<n)
    #
    # dL/da_i = (1/beta) * sum_k w_k * dd_k/da_i
    #          = (2n/(beta*S^3)) * sum_k w_k * (a_{k-i}*S - c_k) * [0<=k-i<n]

    # Actually: h_i = sum_{k: 0<=k-i<n} w_k * a_{k-i} = sum_{j=0}^{n-1} w_{j+i} * a_j (where we need j+i < 2n-1)
    # Since j in [0,n-1] and we need 0 <= j+i < 2n-1 (always true for i in [0,n-1])
    # and k = j+i, so k in [i, i+n-1], which is within [0, 2n-2] for i in [0,n-1]. ✓
    #
    # And g_i = sum_{k: 0<=k-i<n} w_k * c_k = sum_{j=0}^{n-1} w_{j+i} * c_{j+i}
    #
    # So dL/da_i = (2n/(beta*S^3)) * (S*h_i - g_i)
    
    # Compute h and g efficiently using correlation
    # h_i = sum_j w_{j+i} * a_j for j=0..n-1
    # This is a cross-correlation of a and w (reversed)
    # h = corr(a, w) where h[i] = sum_j a[j] * w[j+i]
    # Using FFT: this is the middle slice of conv(a[::-1], w)
    # conv(a[::-1], w)[n-1+i] = sum_j a[n-1-j] * w[n-1+i-j]
    # Let me think differently.
    # h[i] = sum_j a[j] * w[j+i]
    # This is like a "upward" cross-correlation.
    # We can compute it as: reverse a, convolve with w, take appropriate slice.
    # If A_rev = a[::-1], then conv(A_rev, w)[k] = sum_j A_rev[j] * w[k-j] = sum_j a[n-1-j] * w[k-j]
    # Let m = n-1-j, so j = n-1-m:
    # = sum_m a[m] * w[k-n+1+m]
    # We want h[i] = sum_m a[m] * w[m+i], so we need k-n+1+m = m+i, i.e., k = n-1+i.
    # So h[i] = conv(a[::-1], w)[n-1+i]
    
    N_big = next_pow2(2*n - 1)
    Ar = np.fft.rfft(a[::-1], N_big)
    W = np.fft.rfft(w, N_big)
    cross = np.fft.irfft(Ar * W, N_big)
    h = cross[n-1:2*n-1]  # h[i] for i=0..n-1
    
    # g[i] = sum_j w[j+i] * c[j+i] = sum_j w[j+i] * c[j+i]
    # Let k = j+i: g[i] = sum_{k=i}^{i+n-1} w[k] * c[k]
    # This is a cross-correlation of w and c.
    # g[i] = sum_k w[k] * c[k] * [k in [i, i+n-1]]
    # = (w * c) "windowed" sum
    # Alternatively: g = conv(w[::-1], c) at specific indices? Let me think...
    # g[i] = sum_{j=0}^{n-1} w[j+i] * c[j+i]
    # Let's define d[k] = w[k] * c[k]. Then g[i] = sum_{j=0}^{n-1} d[j+i].
    # This is a sliding window sum! We can compute it efficiently.
    d = w * c
    # g[i] = sum_{j=0}^{n-1} d[j+i] for i = 0, ..., n-1
    # Use cumulative sum
    dcum = np.concatenate([[0], np.cumsum(d)])
    g = dcum[n:] - dcum[:n-1]  # g[i] = dcum[n+i] - dcum[i] for i=0..n-1
    # Wait: g[i] = sum_{j=0}^{n-1} d[j+i] = dcum[n-1+i+1] - dcum[i] = dcum[n+i] - dcum[i]
    # dcum has length 2n-1, so dcum[n+i] is valid for i in [0, n-1] (max index = n+n-1 = 2n-1, but dcum has length 2n, indices 0..2n-1). ✓
    
    grad_L = (2.0 * n / (float(beta) * S**3)) * (S * h - g)
    
    # Add penalty: pen * (S - 1)^2
    if pen > 0:
        grad_L = grad_L + 2.0 * pen * (S - 1.0)
    
    val = 2.0 * L + pen * (S - 1.0)**2
    return val, grad_L

def lse_wrap(x, beta, pen):
    x = np.maximum(x, 0)
    v, g = lse_obj_grad(x, beta, pen)
    return v, g

def run_lbfgs(a0, beta, pen, deadline, maxfun=500):
    a0 = normalize(a0.copy())
    n = len(a0)
    rem = deadline - time.monotonic()
    if rem < 5:
        return score(a0), a0
    maxfun = min(maxfun, int(max(50, (0.8*rem) / max(0.02, 0.005*n/4096))))
    maxiter = max(30, maxfun // 3)
    opts = {
        'ftol': 1e-16,
        'gtol': 1e-14,
        'maxiter': int(maxiter),
        'maxfun': int(maxfun),
        'maxls': 30,
    }
    try:
        res = minimize(lse_wrap, a0, jac=True, args=(beta, pen),
                      method='L-BFGS-B', bounds=[(0,10)]*n, options=opts)
        a_out = normalize(res.x)
    except:
        a_out = normalize(a0)
    return score(a_out), a_out

def peak_pressure(a, k_peaks=3):
    """Multiplicative update targeting the top-k peaks of the convolution."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a
    c = fast_conv(a)
    # Get top-k peak locations
    if k_peaks >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argpartition(c, -k_peaks)[-k_peaks:]
    # For each peak at position p, the contributing pairs are (i, p-i) for valid i
    # Build pressure vector
    pressure = np.zeros(n)
    for p in idx:
        for i in range(max(0, p-n+1), min(n, p+1)):
            j = p - i
            if 0 <= j < n:
                pressure[i] += c[p] * a[i] * a[j]
    # Normalize pressure
    pmax = pressure.max()
    if pmax < 1e-15:
        return a
    pressure /= pmax
    return pressure

def multiplicative_update(a, pressure, eta, deadline=None):
    """Apply multiplicative update: a_i *= exp(-eta * pressure_i), then renormalize."""
    q = a * np.exp(-eta * pressure)
    return normalize(q)

def refine_lse(a, deadline, max_stages=8):
    """Run LSE continuation from low to high beta."""
    betas = [30, 80, 200, 500, 1000, 2500, 5000, 10000, 20000, 50000]
    best_a = a.copy()
    best_sc = score(a)
    for beta in betas:
        if time.monotonic() > deadline:
            break
        rem = deadline - time.monotonic()
        if rem < 10:
            break
        # Multiple iterations at this beta for convergence
        for _ in range(3):
            if time.monotonic() > deadline - 5:
                break
            sc, a2 = run_lbfgs(a, beta, 0.0, deadline - 5, maxfun=300)
            if sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
            a = a2
    return best_sc, best_a

def simulate_anneal(a, deadline, n_iter=2000):
    """Simulated annealing with exact score in log-space."""
    n = len(a)
    a = normalize(a.copy())
    best_a = a.copy()
    best_sc = score(a)
    cur_sc = best_sc
    T = best_sc * 0.1  # initial temperature
    T_end = best_sc * 0.001
    
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        # Progress
        frac = it / max(1, n_iter)
        T_cur = T * (T_end / T) ** frac
        
        # Propose perturbation in log-space
        if np.random.random() < 0.7:
            # Local: perturb a few random positions
            k = np.random.randint(1, max(2, n//50))
            idx = np.random.choice(n, k, replace=False)
            a_new = a.copy()
            a_new[idx] *= np.exp(np.random.randn(k) * 0.3)
        else:
            # Global: smooth noise
            noise = np.random.randn(n)
            # Smooth it
            k_smooth = min(21, n)
            if k_smooth > 3:
                sig = k_smooth / 4.0
                kern = np.exp(-((np.arange(k_smooth) - k_smooth//2)**2) / (2*sig*sig))
                kern /= kern.sum()
                noise = np.convolve(noise, kern, 'same')
            a_new = a * np.exp(0.2 * noise)
        
        a_new = np.maximum(a_new, 0)
        sc_new = score(a_new)
        
        if sc_new < cur_sc or np.random.random() < np.exp(-(sc_new - cur_sc) / max(T_cur, 1e-10)):
            a = a_new
            cur_sc = sc_new
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a.copy()
    
    return best_sc, best_a

"""
Search for non-negative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses clean LSE-smoothed gradient optimization, peak-targeted multiplicative
pressure steps, diffusion smoothing, simulated annealing, and diverse starts.
"""

import numpy as np
import time
from scipy.optimize import minimize

def _npow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def _conv(a):
    n = len(a)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _npow2(M)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:M]

def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = _conv(a)
    cm = float(c.max())
    if cm < 0:
        cm = 0.0
    return float(2.0 * n * cm / (S * S))

def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64)
    a = np.maximum(a, 0.0)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    s = a.sum()
    if s < 1e-12:
        return np.full(a.size, target / max(1, a.size))
    return a * (target / s)

def _lse_grad(a, beta, pen=0.0):
    """
    LSE-smoothed max(conv(a,a)) with penalty pen*(sum(a)-1)^2.
    Returns (value, gradient).
    """
    n = len(a)
    S = a.sum()
    if S < 1e-8:
        return 1e10, np.zeros(n)
    c = _conv(a)
    cm = float(c.max())
    if cm <= 0:
        return 0.0, np.zeros(n)
    z = beta * (c - cm)
    z = z - z.max()
    e = np.exp(z)
    se = e.sum()
    if se <= 0 or not np.isfinite(se):
        return 1e10, np.zeros(n)
    w = e / se
    L = cm + np.log(se) / beta

    # h_i = sum_j w[j+i] * a[j]  (cross-correlation)
    Nbig = _npow2(2 * n - 1)
    Ar = np.fft.rfft(a[::-1], Nbig)
    W = np.fft.rfft(w, Nbig)
    cross = np.fft.irfft(Ar * W, Nbig)
    h = cross[n - 1:2 * n - 1]

    # gradient of LSE: dLSE/da_i = 2 * h_i
    grad = 2.0 * h
    if pen > 0:
        grad = grad + 2.0 * pen * (S - 1.0)

    val = L + pen * (S - 1.0) ** 2
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return float(val), grad

def _lse_wrap(x, beta, pen):
    x = np.maximum(x, 0.0)
    v, g = _lse_grad(x, beta, pen)
    return v, g

def _run_lbfgs(a0, beta, pen, deadline, maxfun=400):
    a0 = _norm(a0.copy())
    n = len(a0)
    rem = float(deadline - time.monotonic())
    if rem < 5:
        return _score(a0), a0
    est_per = 0.004 * max(1.0, n / 4096.0) * max(1.0, np.log2(max(16, n)) / 13.0)
    maxfun = int(min(maxfun, max(50, (0.8 * rem) / max(est_per, 0.001))))
    maxfun = min(maxfun, 2000)
    maxiter = max(30, maxfun // 3)
    opts = {'ftol': 1e-15, 'gtol': 1e-13, 'maxiter': int(maxiter),
            'maxfun': int(maxfun), 'maxls': 30}
    try:
        res = minimize(_lse_wrap, a0, jac=True, args=(beta, pen),
                       method='L-BFGS-B', bounds=[(0.0, 10.0)] * n, options=opts)
        a_out = _norm(res.x)
    except Exception:
        a_out = _norm(a0)
    return _score(a_out), a_out

def _peak_pressure(a, k_peaks=5):
    """Compute pressure vector for peak reduction."""
    n = len(a)
    c = _conv(a)
    if len(c) <= k_peaks:
        idx = np.arange(len(c))
    else:
        idx = np.argpartition(c, -k_peaks)[-k_peaks:]
    pressure = np.zeros(n)
    for p in idx:
        lo = max(0, p - n + 1)
        hi = min(n, p + 1)
        for i in range(lo, hi):
            j = p - i
            if 0 <= j < n:
                pressure[i] += c[p] * a[j]
    pm = pressure.max()
    if pm < 1e-15:
        return pressure
    return pressure / pm

def _multiplicative_update(a, pressure, eta):
    q = a * np.exp(-eta * pressure)
    return _norm(q)

def _diffuse(a, strength=0.1):
    """Gaussian diffusion step to spread out clumps."""
    n = len(a)
    k = min(21, n if n % 2 == 1 else n - 1)
    if k < 3:
        return a
    c = k // 2
    sig = max(1.0, c / 2.5)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    smoothed = np.convolve(a, kern, 'same')
    return _norm((1.0 - strength) * a + strength * smoothed)

def _refine_lse(a, deadline, pen=0.01):
    """LSE continuation with multiple betas."""
    betas = [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 25000, 50000]
    best_a = a.copy()
    best_sc = _score(a)
    for beta in betas:
        if time.monotonic() > deadline:
            break
        rem = deadline - time.monotonic()
        if rem < 15:
            break
        for _ in range(3):
            if time.monotonic() > deadline - 10:
                break
            mf = int(min(400, max(50, (rem * 0.4) / max(0.01, 0.004 * len(a) / 4096))))
            sc, a2 = _run_lbfgs(a, float(beta), pen, deadline - 10, maxfun=mf)
            if sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
            a = a2
    return best_sc, best_a

def _peak_polish(a, deadline, max_iters=200):
    """Iterative peak-pressure reduction with LSE polish."""
    best_a = a.copy()
    best_sc = _score(a)
    cur = a.copy()
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        press = _peak_pressure(cur, k_peaks=5)
        improved = False
        for eta in [0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.4]:
            q = _multiplicative_update(cur, press, eta)
            sc_q = _score(q)
            if sc_q < _score(cur):
                cur = q
                improved = True
            if sc_q < best_sc:
                best_sc = sc_q
                best_a = q.copy()
        if not improved:
            # Try diffusion
            q = _diffuse(cur, 0.15)
            sc_q = _score(q)
            if sc_q < _score(cur):
                cur = q
            if sc_q < best_sc:
                best_sc = sc_q
                best_a = q.copy()
        # Periodic LSE refresh
        if it % 15 == 14:
            rem = deadline - time.monotonic()
            if rem > 20:
                sc, a2 = _refine_lse(cur, deadline - 10)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a2.copy()
                cur = a2
                if sc < _score(best_a):
                    best_a = cur.copy()
    return best_sc, best_a

def _simulate_anneal(a, deadline, n_iter=None):
    """Simulated annealing with exact score."""
    n = len(a)
    a = _norm(a.copy())
    best_a = a.copy()
    best_sc = _score(a)
    cur_sc = best_sc
    if n_iter is None:
        n_iter = int(min(3000, max(200, (deadline - time.monotonic()) * 30)))
    T0 = best_sc * 0.05
    T1 = best_sc * 0.001
    rng = np.random.default_rng(777)
    for it in range(n_iter):
        if time.monotonic() > deadline:
            break
        frac = it / max(1, n_iter)
        T = T0 * (T1 / T0) ** frac
        r = rng.random()
        if r < 0.5:
            k = rng.integers(1, max(2, n // 100))
            idx = rng.integers(0, n, size=k)
            a_new = a.copy()
            a_new[idx] *= np.exp(rng.standard_normal(k) * 0.4)
        elif r < 0.8:
            # Smooth perturbation
            noise = rng.standard_normal(n)
            k = min(15, n if n % 2 == 1 else n - 1)
            if k >= 3:
                sig = k / 4.0
                kern = np.exp(-((np.arange(k) - k // 2) ** 2) / (2 * sig * sig))
                kern /= kern.sum()
                noise = np.convolve(noise, kern, 'same')
            a_new = a * np.exp(0.15 * noise)
        else:
            # Peak-targeted
            press = _peak_pressure(a, 3)
            a_new = a * np.exp(-0.1 * press)
        a_new = np.maximum(np.nan_to_num(a_new, nan=0.0), 0.0)
        sc_new = _score(a_new)
        if not np.isfinite(sc_new):
            continue
        if sc_new < cur_sc or rng.random() < np.exp(-(sc_new - cur_sc) / max(T, 1e-10)):
            a = a_new
            cur_sc = sc_new
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
    return best_sc, best_a
