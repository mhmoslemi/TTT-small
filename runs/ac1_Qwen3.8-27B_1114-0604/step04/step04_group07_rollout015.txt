    M_fft = 1 << (M - 1).bit_length()
    A_ft = np.fft.rfft(a, M_fft)
    P_ft = np.fft.rfft(p_rev, M_fft)
    conv_apr = np.fft.irfft(A_ft * P_ft, M_fft)[:M]
    h = conv_apr[n - 1:2 * n - 1][::-1]
    grad = 2.0 * h
    return lse_val, grad


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and its gradient w.r.t. a."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    L, conv_grad = lse_grad_conv(a, beta)
    S2 = S * S
    S3 = S2 * S
    score = 2.0 * n * L / S2
    grad = 2.0 * n * (conv_grad / S2 - 2.0 * L / S3)
    return score, grad


def mirror_descent_optimize(a_init, betas, eta_base, n_steps_per_beta, time_limit_s, t_start,
                            palindrome=True, verbose=True):
    """
    Mirror descent (multiplicative weight) with LSE smoothing and geometric beta schedule.
    Works in half-space if palindrome=True.
    """
    if palindrome:
        n_half = len(a_init)
        h = np.clip(a_init.astype(np.float64).copy(), 1e-15, None)
        h = h / h.sum()
        n = 2 * n_half
    else:
        n_half = len(a_init)
        h = np.clip(a_init.astype(np.float64).copy(), 1e-15, None)
        h = h / h.sum()
        n = n_half

    best_score = np.inf
    if palindrome:
        best_a = np.concatenate([h, h[::-1]])
        best_score = compute_score(best_a)
    else:
        best_a = h.copy()
        best_score = compute_score(h)

    for bi, beta in enumerate(betas):
        for t in range(n_steps_per_beta):
            if time.monotonic() - t_start > time_limit_s:
                return best_score, best_a

            if palindrome:
                a_full = np.concatenate([h, h[::-1]])
                sc, grad_full = lse_score_and_grad(a_full, beta)
                grad_h = grad_full[:n_half] + grad_full[2 * n_half - 1 - np.arange(n_half)]
            else:
                sc, grad_h = lse_score_and_grad(h, beta)

            # Adaptive step size
            eta = eta_base / (np.max(np.abs(grad_h)) + 1e-10)
            eta = min(eta, 1.0)  # Cap to avoid instability

            # Mirror descent update: h_new[j] = h[j] * exp(-eta * grad[j]) / Z
            log_update = -eta * grad_h
            # Clip for numerical stability
            log_update = np.clip(log_update, -50, 50)
            weights = np.exp(log_update)
            h_new = h * weights
            Z = h_new.sum()
            if Z < 1e-15:
                break
            h_new = h_new / Z
            h

            # Check improvement
            if (bi * n_steps_per_beta + t) % 100 == 0:
                if palindrome:
                    a_check = np.concatenate([h_new, h_new[::-1]])
                    sc_check = compute_score(a_check)
                else:
                    a_check = h_new
                    sc_check = compute_score(h_new)
                if sc_check < best_score:
                    best_score = sc_check
                    best_a = a_check.copy()

            h = h_new

    # Final check
    if palindrome:
        a_final = np.concatenate([h, h[::-1]])
        sc_final = compute_score(a_final)
    else:
        a_final = h
        sc_final = compute_score(h)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    return best_score, best_a


def mirror_descent_adaptive(a_init, betas, eta0, n_steps_per_beta, time_limit_s, t_start,
                            palindrome=True):
    """Mirror descent with backtracking line search."""
    if palindrome:
        n_half = len(a_init)
        h = np.clip(a_init.astype(np.float64).copy(), 1e-15, None)
        h = h / h.sum()
        n = 2 * n_half
    else:
        n_half = len(a_init)
        h = np.clip(a_init.astype(np.float64).copy(), 1e-15, None)
        h = h / h.sum()
        n = n_half

    best_score = np.inf
    if palindrome:
        a_full = np.concatenate([h, h[::-1]])
        best_score = compute_score(a_full)
        best_a = a_full.copy()
    else:
        best_score = compute_score(h)
        best_a = h.copy()

    eta = eta0

    total_steps = len(betas) * n_steps_per_beta
    step_count = 0

    for bi, beta in enumerate(betas):
        for t in range(n_steps_per_beta):
            if time.monotonic() - t_start > time_limit_s:
                return best_score, best_a
            step_count += 1

            if palindrome:
                a_full = np.concatenate([h, h[::-1]])
                sc, grad_full = lse_score_and_grad(a_full, beta)
                grad_h = grad_full[:n_half] + grad_full[2 * n_half - 1 - np.arange(n_half)]
            else:
                sc, grad_h = lse_score_and_grad(h, beta)

            # Backtracking line search
            eta_try = eta
            improved = False
            for shrink in range(10):
                log_upd = -eta_try * grad_h
                log_upd = np.clip(log_upd, -80, 80)
                h_new = h * np.exp(log_upd)
                Z = h_new.sum()
                if Z < 1e-15:
                    eta_try *= 0.5
                    continue
                h_new = h_new / Z

                if palindrome:
                    a_new = np.concatenate([h_new, h_new[::-1]])
                    sc_new = compute_score(a_new)
                else:
                    a_new = h_new
                    sc_new = compute_score(h_new)

                if sc_new < sc - 1e-12 * abs(sc):
                    h = h_new
                    eta = min(eta_try * 1.5, 10.0)
                    improved = True
                    if sc_new < best_score:
                        best_score = sc_new
                        best_a = a_new.copy()
                    break
                else:
                    eta_try *= 0.5

            if not improved:
                # No improvement; try a smaller base step
                eta = max(eta * 0.8, 1e-6)

    return best_score, best_a


def lse_lbfgs(a_init, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B with LSE smoothing."""
    n = len(a_init)
    a0 = np.clip(a_init.astype(np.float64), 1e-15, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        x_c = np.clip(x, 1e-15, None)
        sc, gr = lse_score_and_grad(x_c, beta)
        return sc, gr

    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-16, 'gtol': 1e-14,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30})
    a_out = np.clip(res.x, 0.0, 1000.0)
    a_out = a_out / max(a_out.sum(), 1e-15)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def lse_lbfgs_pal(half_init, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B in palindrome half-space."""
    n_half = len(half_init)
    h0 = np.clip(half_init.astype(np.float64), 1e-15, 1000.0).copy()
    if h0.sum() < 1e-6:
        h0[:] = 1.0 / n_half
    bounds = [(0.0, 1000.0)] * n_half

    def obj(h):
        h_c = np.clip(h, 1e-15, None)
        a_full = np.concatenate([h_c, h_c[::-1]])
        sc, grad_full = lse_score_and_grad(a_full, beta)
        grad_h = grad_full[:n_half] + grad_full[2 * n_half - 1 - np.arange(n_half)]
        return sc, grad_h

    res = minimize(obj, h0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-16, 'gtol': 1e-14,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30})
    h_out = np.clip(res.x, 0.0, 1000.0)
    h_out = h_out / max(h_out.sum(), 1e-15)
    a_full = np.concatenate([h_out, h_out[::-1]])
    exact_sc = compute_score(a_full)
    return exact_sc, h_out


def generate_inits(rng, n):
    """Generate diverse initializations of length n (all normalized to sum=1, non-negative)."""
    inits = []
    x = np.arange(n) / n

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian (various widths)
    for sigma in [0.1, 0.2, 0.3, 0.4]:
        g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        inits.append(g / g.sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(rc / rc.sum())

    # Triangular
    tri = np.maximum(0, 1 - np.abs(2 * x - 1))
    if tri.sum() > 1e-10:
        inits.append(tri / tri.sum())

    # Sine-modulated
    for freq in [2, 3, 5, 7]:
        s = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x)
        s = np.maximum(s, 0.01)
        inits.append(s / s.sum())

    # Two-bump
    b = np.exp(-0.5 * ((x - 0.25) / 0.08) ** 2) + np.exp(-0.5 * ((x - 0.75) / 0.08) ** 2)
    inits.append(b / b.sum())

    # Random smooth
    r = np.abs(rng.standard_normal(n)) + 0.5
    k = max(3, n // 20)
    kernel = np.ones(k) / k
    r_sm = np.convolve(r, kernel, mode='same')
    r_sm = np.maximum(r_sm, 1e-6)
    inits.append(r_sm / r_sm.sum())

    # Palindromic random
    half_n = n // 2
    h = np.abs(rng.standard_normal(half_n)) + 0.5
    k2 = max(3, half_n // 20)
    h_sm = np.convolve(h, np.ones(k2) / k2, mode='same')
    h_sm = np.maximum(h_sm, 1e-6)
    pal = np.concatenate([h_sm, h_sm[::-1]])
    inits.append(pal / pal.sum())

    # Fourier random phases
    coeffs = rng.standard_normal(5) * 0.1
    f = 1.0 + np.arange(5) * 2.0
    phases = rng.uniform(0, 2 * np.pi, 5)
    mod = 1.0 + sum(c * np.cos(2 * np.pi * fi * x + p) for c, fi, p in zip(coeffs, f, phases))
    mod = np.maximum(mod, 0.01)
    inits.append(mod / mod.sum())

    return [np.clip(x_i, 1e-10, 1000.0) for x_i in inits]


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def get_palindrome_half(a):
    """Extract palindrome half from a full sequence."""
    n = len(a)
    if n % 2 != 0:
        n = n + 1
        a = resample_sequence(a, n)
    n_half = n // 2
    half = (a[:n_half] + a[::-1][:n_half]) / 2.0
    half = np.maximum(half, 1e-10)
    half = half / half.sum()
    return half


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase optimizer:
    1. Load parent, try multiple lengths
    2. Screen diverse inits with quick mirror descent
    3. Extended mirror descent with geometric beta schedule
    4. L-BFGS polishing
    5. Multi-restart with diverse perturbations
    6. Final verification
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    budget = float(budget_s) - 30.0  # Leave 30s buffer

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 32:
                parent = np.clip(p, 0.0, 1000.0)
                parent = parent / parent.sum()
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def try_update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # Determine working length
    if parent is not None:
        n_work = len(parent)
        print(f"Parent loaded: n={n_work}, score={compute_score(parent):.8f}")
        try_update(compute_score(parent), parent, "parent")
    else:
        n_work = 4096
        print("No parent. Starting with n=4096.")

    # ==========================================
    # Phase 1: Screen diverse inits (15% time)
    # ==========================================
    p1_budget = budget * 0.15
    print(f"\n{'='*60}")
    print(f"Phase 1: Screening diverse inits (budget={p1_budget:.0f}s)")
    print(f"{'='*60}")

    inits = generate_inits(rng, n_work)
    if parent is not None:
        inits.insert(0, parent.copy())

    candidates = []
    per_init_time = p1_budget / len(inits)

    for i, init in enumerate(inits):
        remaining = t_start + p1_budget - time.monotonic()
        if remaining < 2:
            break

        # Quick screening with low-beta mirror descent
        betas_quick = [5, 20]
        sc, a_opt = mirror_descent_optimize(
            get_palindrome_half(init), betas_quick,
            eta_base=0.5, n_steps_per_beta=500,
            time_limit_s=min(per_init_time, remaining - 1),
            t_start=t_start, palindrome=True
        )
        candidates.append((sc, a_opt))
        try_update(sc, a_opt, f"screen init#{i}")

    candidates.sort(key=lambda x: x[0])
    print(f"  Top candidate scores: {[f'{c[0]:.6f}' for c in candidates[:5]]}")

    # ==========================================
    # Phase 2: Extended mirror descent (45% time)
    # ==========================================
    p2_budget = budget * 0.45
    print(f"\n{'='*60}")
    print(f"Phase 2: Extended mirror descent (budget={p2_budget:.0f}s)")
    print(f"{'='*60}")

    # Work with best candidate in palindrome space
    best_half = get_palindrome_half(best_a)

    # Geometric beta schedule: from smooth to sharp
    betas = [5, 15, 50, 150, 500, 1500, 5000, 15000]
    eta_schedule = [1.0, 0.8, 0.6, 0.4, 0.3, 0.2, 0.1, 0.05]
    steps_schedule = [2000, 3000, 3000, 3000, 3000, 3000, 3000, 3000]

    t_p2 = t_start
    for i, (beta, eta, n_steps) in enumerate(zip(betas, eta_schedule, steps_schedule)):
        remaining = t_start + p2_budget - time.monotonic()
        if remaining < 5:
            break

        t0 = time.monotonic()
        sc, a_opt = mirror_descent_optimize(
            best_half, [beta], eta_base=eta, n_steps_per_beta=n_steps,
            time_limit_s=remaining, t_start=t_start, palindrome=True
        )
        elapsed = time.monotonic() - t0
        try_update(sc, a_opt, f"MD beta={beta}")
        best_half = get_palindrome_half(a_opt)
        print(f"    beta={beta}: score={sc:.8f} ({elapsed:.1f}s)")

    # Also try non-palindrome mirror descent
    remaining = t_start + p2_budget - time.monotonic()
    if remaining > 30:
        betas_np = [50, 200, 1000]
        for beta in betas_np:
            remaining = t_start + p2_budget - time.monotonic()
            if remaining < 10:
                break
            t0 = time.monotonic()
            sc, a_opt = mirror_descent_optimize(
                best_a, [beta], eta_base=0.3, n_steps_per_beta=2000,
                time_limit_s=min(remaining - 2, budget * 0.05),
                t_start=t_start, palindrome=False
            )
            try_update(sc, a_opt, f"MD-nopal beta={beta}")
            print(f"    MD-nopal beta={beta}: score={sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==========================================
    # Phase 3: L-BFGS polishing (25% time)
    # ==========================================
    p3_budget = budget * 0.25
    print(f"\n{'='*60}")
    print(f"Phase 3: L-BFGS polishing (budget={p3_budget:.0f}s)")
    print(f"{'='*60}")

    best_half = get_palindrome_half(best_a)

    lbfgs_betas = [100, 300, 1000, 3000, 10000]
    for beta in lbfgs_betas:
        remaining = t_start + p3_budget - time.monotonic()
        if remaining < 5:
            break
        t0 = time.monotonic()
        sc, h_opt = lse_lbfgs_pal(best_half, beta, maxiter=3000, maxfun=40000)
        a_full = np.concatenate([h_opt, h_opt[::-1]])
        elapsed = time.monotonic() - t0
        try_update(sc, a_full, f"LBFGS-pal beta={beta}")
        best_half = h_opt
        print(f"    LBFGS-pal beta={beta}: {sc:.8f} ({elapsed:.1f}s)")

    # Non-palindrome L-BFGS
    for beta in [500, 2000]:
        remaining = t_start + p3_budget - time.monotonic()
        if remaining < 10:
            break
        t0 = time.monotonic()
        sc, a_opt = lse_lbfgs(best_a, beta, maxiter=2000, maxfun=30000)
        try_update(sc, a_opt, f"LBFGS beta={beta}")
        print(f"    LBFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==========================================
    # Phase 4: Multi-restart (10% time)
    # ==========================================
    p4_budget = budget * 0.10
    print(f"\n{'='*60}")
    print(f"Phase 4: Multi-restart (budget={p4_budget:.0f}s)")
    print(f"{'='*60}")

    n_restarts = 6
    per_restart = p4_budget / n_restarts

    for r in range(n_restarts):
        remaining = t_start + p4_budget - time.monotonic()
        if remaining < 5:
            break

        # Generate perturbation
        strategy = r % 3
        if strategy == 0:
            # Log-space Gaussian
            log_a = np.log(np.clip(best_a, 1e-10, 1e10))
            noise = 0.1 * rng.standard_normal(len(best_a))
            a_pert = np.exp(log_a + noise)
            a_pert = a_pert / a_pert.sum()
        elif strategy == 1:
            # Fourier perturbation
            x = np.arange(len(best_a)) / len(best_a)
            freqs = rng.uniform(1, 8, 3)
            phases = rng.uniform(0, 2 * np.pi, 3)
            amps = rng.uniform(0.05, 0.15, 3)
            mod = 1.0 + sum(a_ * np.cos(2 * np.pi * f * x + p)
                            for f, p, a_ in zip(freqs, phases, amps))
            a_pert = best_a * np.maximum(mod, 0.1)
            a_pert = a_pert / a_pert.sum()
        else:
            # Peak reduction
            c = fast_conv(best_a)
            k_star = int(c.argmax())
            n_cur = len(best_a)
            js = max(0, k_star - n_cur + 1)
            je = min(k_star, n_cur - 1)
            a_pert = best_a.copy()
            a_pert[js:je + 1] *= 0.7
            non_peak = np.ones(n_cur, dtype=bool)
            non_peak[js:je + 1] = False
            if non_peak.sum() > 0:
                removed = a_pert.sum() - best_a[non_peak].sum()
                a_pert[non_peak] += removed * 0.5 / non_peak.sum()
            a_pert = np.maximum(a_pert, 1e-10)
            a_pert = a_pert / a_pert.sum()

        # Quick optimization from perturbation
        t_rem = t_start + p4_budget - time.monotonic()
        if t_rem < 3:
            break
        half_pert = get_palindrome_half(a_pert)
        sc, a_opt = mirror_descent_optimize(
            half_pert, [50, 200], eta_base=0.5, n_steps_per_beta=1500,
            time_limit_s=min(per_restart * 0.6, t_rem - 5), t_start=t_start, palindrome=True
        )
        # Quick L-BFGS
        t_rem = t_start + p4_budget - time.monotonic()
        if t_rem > 5:
            h_opt = get_palindrome_half(a_opt)
            sc2, h2 = lse_lbfgs_pal(h_opt, 500, maxiter=1000, maxfun=15000)
            a2 = np.concatenate([h2, h2[::-1]])
            if sc2 < sc:
                sc, a_opt = sc2, a2
        try_update(sc, a_opt, f"restart#{r} s{strategy}")
        print(f"    Restart {r} (s{strategy}): {sc:.8f}")

    # ==========================================
    # Phase 5: Final intensive polish (5% time)
    # ==========================================
    p5_budget = budget * 0.05
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish (budget={p5_budget:.0f}s)")
    print(f"{'='*60}")

    final_deadline = t_start + budget

    # Very high beta L-BFGS
    best_half = get_palindrome_half(best_a)
    for beta in [5000, 20000]:
        remaining = final_deadline - time.monotonic()
        if remaining < 5:
            break
        t0 = time.monotonic()
        sc, h_opt = lse_lbfgs_pal(best_half, beta, maxiter=5000, maxfun=80000)
        a_full = np.concatenate([h_opt, h_opt[::-1]])
        try_update(sc, a_full, f"final LBFGS beta={beta}")
        best_half = h_opt
        print(f"    Final LBFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Final mirror descent with very high beta
    remaining = final_deadline - time.monotonic()
    if remaining > 10:
        t0 = time.monotonic()
        sc, a_opt = mirror_descent_adaptive(
            get_palindrome_half(best_a), [10000, 30000],
            eta0=0.1, n_steps_per_beta=3000,
            time_limit_s=min(remaining - 2, p5_budget), t_start=t_start, palindrome=True
        )
        try_update(sc, a_opt, "final MD-adaptive")
        print(f"    Final MD-adaptive: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==========================================
    # Final verification
    # ==========================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal_best={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
