Flat-autocorrelation sequence optimizer using:
1. Adam gradient descent on LSE-smoothed objective (primary)
2. LP-based alternating with cvxpy (secondary, for smaller n)
3. L-BFGS polish (tertiary)
4. Coordinate descent (final polish)

import numpy as np
from scipy.optimize import minimize

def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def lse_grad(a, beta):
    """Compute LSE-smoothed objective value and gradient (with sum(a)=1)."""
    n = len(a)
    c = self_conv(a)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    exp_sum = exp_z.sum()
    if exp_sum < 1e-300:
        p = np.ones(len(c)) / len(c)
    else:
        p = exp_z / exp_sum
    L_val = c_max + np.log(exp_sum) / beta

    # This is cross-correlation of p and a (reversed)

    # d(c[k])/d(a[j]) = a[k-j] if 0 <= k-j < n

    # Specifically, grad[j] = sum_{m=0}^{n-1} p[j+m] * a[m]

def adam_optimize(a0, beta_start=10.0, beta_rate=1.05, max_iters=200000, 
                  lr=0.01, time_limit=None, check_interval=100):
    """Adam optimization on LSE-smoothed objective."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    beta = beta_start
    
    for it in range(max_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break
        
        lse_val, grad = lse_grad(a, beta)
        
        # Update Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            a = np.ones(n) / n
        else:
            a = a / S
        
        # Periodic check
        if (it + 1) % check_interval == 0:
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                print(f"    Adam iter {it+1}: score={sc:.8f}, beta={beta:.1f}")
        
        # Increase beta
        if (it + 1) % 100 == 0:
            beta *= beta_rate
    
    # Final check
    sc = compute_score(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()
    
    return best_a, best_sc
