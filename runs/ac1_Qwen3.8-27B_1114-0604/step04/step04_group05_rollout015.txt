    # = 2*a_{k-j} if k in [j, j+n-1] (so that k-j in [0, n-1])

    # = (2/beta) * sum_{k-j in [0,n-1]} p_k * a_{k-j}
    # = (2/beta) * conv_res[j] (where conv_res[j] = sum_l a[l] * p[j+l] as computed above)
    
    # Wait, I need to be more careful. sum_k p_k * a_{k-j} where k-j in [0,n-1] means k in [j, j+n-1].
    # conv_res[j] = sum_l a[l] * p[j+l] = sum_l a[l] * p[l+j]
    # Let k = l+j: = sum_k a[k-j] * p[k] (where k-j in [0,n-1], i.e., k in [j, j+n-1])
    # So conv_res[j] = sum_{k=j}^{j+n-1} p[k] * a[k-j] = sum_k p_k * a_{k-j} (with the range restriction)
    # Great, so dL/da_j = (2/beta) * conv_res[j]
    
    # Now the full gradient of score = 2n*L/S^2:

    S3 = S2 * S
    grad_L = (2.0 / beta) * conv_res
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L / S3)

def exponentiated_gradient(a0, n_steps=5000, lr_init=0.1, lr_decay=0.999,
                           time_budget=None, t_start=None):
    """
    Exponentiated gradient (multiplicative weights) for the non-smooth objective.
    a_j <- a_j * exp(-lr * subgrad_j) / normalizer
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    a = np.maximum(a, 1e-10)  # Ensure positive for log
    S0 = a.sum()
    
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf
    
    lr = lr_init
    
    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break
        
        # Compute subgradient
        c = fast_conv(a)
        S = float(a.sum())
        c_max = float(c.max())
        k_star = int(c.argmax())
        
        # Subgradient of max(conv) w.r.t. a
        sg = np.zeros(n)
        # dc_k/da_j = 2*a_{k-j} for k-j in [0,n-1]
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je+1] = 2.0 * a[idx]
        
        # Gradient of score = 2n * M / S^2
        # d(score)/da_j = 2n * (sg_j / S^2 - 2*M / S^3)
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * c_max / S3)
        
        # Multiplicative update
        a = a * np.exp(-lr * grad / max(1.0, np.abs(grad).max()))
        a = np.clip(a, 1e-15, 1000.0)
        
        # Rescale to maintain sum
        S_new = a.sum()
        if S_new > 1e-12:
            a = a * (S0 / S_new)
        
        # Decay learning rate
        lr *= lr_decay
        
        # Check improvement
        if t % 100 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    
    return best_score, best_a

def hessian_based_perturbation(a, rng, n_perturb=5):
    """
    Perturb along random directions in the null space of the active constraints.
    This is more structured than random perturbations.
    """
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    S = float(a.sum())
    
    # Find near-peak indices
    threshold = c_max - 0.01 * c_max
    peak_indices = np.where(c >= threshold)[0]
    
    a_new = a.copy()
    for _ in range(n_perturb):
        # Random direction
        d = rng.standard_normal(n)
        # Make it have zero sum (to preserve normalization)
        d -= d.mean()
        
        # Try to make it orthogonal to the "active constraint gradients"
        # The gradient of conv[k] at a is: (grad_k)[j] = 2*a[k-j] (for valid k-j)
        # We want to minimize the increase in these active conv values
        # Simple approach: project d onto the null space of the peak gradients
        A_peak = np.zeros((len(peak_indices), n))
        for i, k in enumerate(peak_indices):
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                idx = k - np.arange(js, je + 1)
                A_peak[i, js:je+1] = 2.0 * a[idx]
        
        if len(peak_indices) > 0 and len(peak_indices) < n:
            # Project: d <- d - A^T (A A^T)^{-1} A d
            ATA = A_peak @ A_peak.T + 1e-8 * np.eye(len(peak_indices))
            Ad = A_peak @ d
            d = d - A_peak.T @ np.linalg.solve(ATA, Ad)
        
        # Apply small perturbation
        step = 0.01 * a.max() / max(1.0, np.linalg.norm(d))
        a_candidate = a_new + step * d
        a_candidate = np.clip(a_candidate, 0.0, 1000.0)
        if a_candidate.sum() > 1e-10:
            a_candidate = a_candidate * (a_new.sum() / a_candidate.sum())
        
        # Accept if it doesn't worsen the score too much
        sc = compute_score(a_candidate)
        sc_cur = compute_score(a_new)
        if sc < sc_cur * 1.01:  # Accept if within 1%
            a_new = a_candidate
    
    return a_new

def fista_lse(a0, beta, n_steps=20000, time_budget=None, t_start=None):
    """FISTA (accelerated gradient) on the LSE-smoothed objective."""
    n = len(a0)
    x = a0.astype(np.float64).copy()
    z = x.copy()
    t_k = 1.0
    
    best_a = x.copy()

    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf
    
    # Lipschitz constant estimate (rough)
    L = compute_score(x) * n / max(1e-6, a0.max())
    
    for k in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break
        
        score, grad = lse_score_and_grad(z, beta)
        
        # Gradient step
        step_size = 1.0 / L
        # Adaptive: use backtracking if needed
        x_new = z - step_size * grad
        x_new = np.clip(x_new, 0.0, 1000.0)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z = x_new + ((t_k - 1.0) / t_new) * (x_new - x)
        z = np.clip(z, 0.0, 1000.0)
        
        x = x_new
        t_k = t_new
        
        # Decay L (the function gets smoother as beta increases, but within a fixed beta it's the same)
        # Actually L is fixed for a given beta, so no decay needed
        
        if k % 500 == 0:
            sc = compute_score(x)
            if sc < best_score:
                best_score = sc
                best_a = x.copy()
    
    sc_final = compute_score(x)
    if sc_final < best_score:

        best_a = x.copy()

def refine_with_lbfgs(a0, beta, time_budget=30.0, t_start=None):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    deadline = t_start + time_budget
    
    def objective_and_grad(x):
        return lse_score_and_grad(x, beta)
    
    def timed_obj(x):
        if time.monotonic() > deadline:
            raise TimeoutError
        return lse_score_and_grad(x, beta)
    
    try:
        res = minimize(
            timed_obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-12, 'maxiter': 10000, 'maxfun': 100000, 'maxls': 100}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except (TimeoutError, Exception):
        a_out = a0
    
    sc = compute_score(a_out)
    return sc, a_out

def anti_palindromic_refine(a, rng, n_iters=20):
    """
    Iteratively apply anti-palindromic corrections while maintaining sum.
    This is specialized for reducing the central peak.
    """
    n = len(a)
    a_work = a.astype(np.float64).copy()
    S0 = a_work.sum()
    best = a_work.copy()
    best_sc = compute_score(a_work)
    
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    for it in range(n_iters):
        # Compute the "excess" at the peak
        c = fast_conv(a_work)
        c_max = float(c.max())
        
        # Create anti-palindromic direction weighted by the current values
        # The idea: shift mass from the center-outward in an anti-palindromic way
        # a_j * (j - center) is anti-palindromic if a is palindromic
        direction = a_work * (x - center)
        # Make it have zero sum
        direction -= direction.mean()
        
        # Try a small step
        for alpha in [0.01, 0.005, 0.001, -0.001, -0.005]:
            a_try = a_work + alpha * direction / max(1.0, np.abs(direction).max())
            a_try = np.clip(a_try, 0.0, 1000.0)
            if a_try.sum() < 1e-10:
                continue
            a_try = a_try * (S0 / a_try.sum())
            sc = compute_score(a_try)
            if sc < best_sc:
                best_sc = sc
                best = a_try.copy()
                a_work = a_try.copy()
                break
    
    return best_sc, best

"""
Continuation L-BFGS + Exponentiated Gradient + Pairwise Swap + Structured Perturbations
for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
Key innovations: adaptive beta continuation with convergence detection,
exponentiated gradient for non-smooth structure, pairwise value swaps for
escaping coordinate-wise local minima, and Hessian-informed perturbations.
"""

import numpy as np
from scipy.optimize import minimize
import time
import math


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p_sum = exp_scaled.sum()
    L = c_max + np.log(p_sum) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / p_sum
    rev_a = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    grad_L = (2.0 / beta) * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, time_budget=30.0, t_start=None, maxiter=10000, maxfun=100000):
    """L-BFGS-B on LSE-smoothed objective with time limit."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    deadline = (t_start + time_budget) if (t_start is not None and time_budget is not None) else np.inf

    def timed_obj(x):
        if time.monotonic() > deadline:
            raise TimeoutError
        return lse_score_and_grad(x, beta)

    try:
        res = minimize(
            timed_obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-12, 'maxiter': maxiter,
                     'maxfun': maxfun, 'maxls': 100}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except (TimeoutError, Exception):
        a_out = a0
    sc = compute_score(a_out)
    return sc, a_out


def exponentiated_gradient_step(a, lr, k_star, S, c_max):
    """One step of exponentiated gradient (multiplicative weights)."""
    n = len(a)
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (sg / S2 - 2.0 * c_max / S3)
    gradNorm = np.abs(grad).max()
    if gradNorm > 1e-15:
        a = a * np.exp(-lr * grad / gradNorm)
    a = np.clip(a, 1e-15, 1000.0)
    return a


def exp_gradient_optimize(a0, n_steps=20000, lr_init=0.1, lr_end=0.001,
                          time_budget=None, t_start=None):
    """Exponentiated gradient optimization with linear lr schedule."""
    n = len(a0)
    a = np.maximum(a0.astype(np.float64).copy(), 1e-10)
    S0 = a.sum()
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf

    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break
        lr = lr_init + (lr_end - lr_init) * (t / n_steps)

        c = fast_conv(a)
        S = float(a.sum())
        c_max = float(c.max())
        k_star = int(c.argmax())

        a = exponentiated_gradient_step(a, lr, k_star, S, c_max)
        S_new = a.sum()
        if S_new > 1e-12:
            a = a * (S0 / S_new)

        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def pairwise_swap_optimize(a0, n_trials=5000, time_budget=None, t_start=None,
                           rng=None):
    """
    Try random pairwise partial swaps: a[j] and a[k] -> a[j]+t*(a[k]-a[j]), a[k]-t*(a[k]-a[j]).
    This preserves the sum and can escape coordinate-wise local minima.
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    best_a = a.copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf

    if rng is None:
        rng = np.random.default_rng(0)

    # Precompute for efficiency: we evaluate score after each swap
    # Use a coarse-to-fine approach
    for trial in range(n_trials):
        if time.monotonic() > deadline:
            break

        j = int(rng.integers(0, n))
        k = int(rng.integers(0, n))
        if j == k:
            continue

        # Try several step sizes for the swap
        delta = a[k] - a[j]
        if abs(delta) < 1e-12:
            continue

        found_improvement = False
        for t_frac in [0.1, 0.25, 0.5, 0.75, 1.0]:
            a[j] += t_frac * delta
            a[k] -= t_frac * delta
            if a[j] < 0 or a[k] < 0:
                a[j] -= t_frac * delta
                a[k] += t_frac * delta
                continue
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                found_improvement = True
                break
            else:
                a[j] -= t_frac * delta
                a[k] += t_frac * delta

        # Occasionally try a more targeted swap involving the peak
        if trial % 10 == 0:
            c = fast_conv(a)
            k_peak = int(c.argmax())
            # The peak involves a[j]*a[k_peak - j] for valid j
            # Try reducing a[j] where j contributes most to the peak
            js = max(0, k_peak - n + 1)
            je = min(k_peak, n - 1)
            if je >= js:
                contributions = np.zeros(n)
                idx = k_peak - np.arange(js, je + 1)
                contributions[js:je + 1] = a[js:je + 1] * a[idx]
                # Move mass from high-contribution to low-contribution
                top_j = js + int(np.argmax(contributions[js:je + 1]))
                bot_j = int(rng.integers(0, n))
                if bot_j != top_j and a[top_j] > a[bot_j]:
                    delta2 = (a[top_j] - a[bot_j]) * 0.1
                    a2 = a.copy()
                    a2[top_j] -= delta2
                    a2[bot_j] += delta2
                    if a2[top_j] >= 0:
                        sc2 = compute_score(a2)
                        if sc2 < best_score:
                            best_score = sc2
                            best_a = a2.copy()
                            a = a2

    return best_score, best_a


def structured_perturbation(a, rng, mode="spectral"):
    """Generate a structured perturbation of a."""
    n = len(a)
    S0 = a.sum()

    if mode == "spectral":
        M_fft = 1 << (2 * n - 2).bit_length()
        A = np.fft.rfft(a, M_fft)
        amp_factors = 1.0 + 0.15 * rng.standard_normal(len(A))
        amp_factors = np.clip(amp_factors, 0.5, 1.8)
        phase_shifts = 0.1 * rng.standard_normal(len(A))
        A_new = A * amp_factors * np.exp(1j * phase_shifts)
        a_new = np.fft.irfft(A_new, M_fft)[:n]
    elif mode == "anti_palin":
        x = np.arange(n, dtype=np.float64)
        center = (n - 1) / 2.0
        freq = float(rng.uniform(0.5, 3.0))
        tilt = np.sin(2.0 * np.pi * freq * x / n)
        tilt_anti = 0.5 * (tilt - tilt[::-1])
        a_new = a + 0.1 * tilt_anti * a.max()
    elif mode == "block":
        a_new = a.copy()
        for _ in range(2):
            blen = int(rng.integers(n // 30, n // 8))
            i = int(rng.integers(0, n - blen))
            j = int(rng.integers(0, n - blen))
            if i != j:
                a_new[i:i + blen], a_new[j:j + blen] = a_new[j:j + blen].copy(), a_new[i:i + blen].copy()
    elif mode == "gaussian":
        a_new = a.copy()
        for _ in range(3):
            center = float(rng.integers(0, n))
            width = float(rng.integers(n // 40, n // 8))
            amp = float(rng.uniform(-0.1, 0.1)) * S0 / n
            x = np.arange(n, dtype=np.float64)
            a_new += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
    else:
        a_new = a + 0.05 * a.max() * rng.standard_normal(n)

    a_new = np.clip(a_new, 0.0, 1000.0)
    if a_new.sum() > 1e-10:
        a_new = a_new * (S0 / a_new.sum())
    return a_new


def coord_descent_refine(a0, n_sweeps=5, top_k=40, time_budget=None, t_start=None):
    """Coordinate descent with golden-section on top-k gradient coordinates."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf
    gr = (math.sqrt(5.0) - 1.0) / 2.0

    def golden_1d(a_arr, j, n_evals=12):
        a_save = a_arr[j]
        lo = 0.0
        hi = max(a_save * 3.0, a_arr.max() * 0.5)
        if hi < 1e-10:
            hi = a_arr.max() * 2.0
            if hi < 1e-10:
                hi = 1.0
        c1 = hi - gr * (hi - lo)
        c2 = lo + gr * (hi - lo)
        a_arr[j] = c1
        f1 = compute_score(a_arr)
        a_arr[j] = c2
        f2 = compute_score(a_arr)
        for _ in range(n_evals - 2):
            if f1 < f2:
                hi = c2
                c2 = c1
                f2 = f1
                c1 = hi - gr * (hi - lo)
                a_arr[j] = c1
                f1 = compute_score(a_arr)
            else:
                lo = c1
                c1 = c2
                f1 = f2
                c2 = lo + gr * (hi - lo)
                a_arr[j] = c2
                f2 = compute_score(a_arr)
        best_val = c1 if f1 < f2 else c2
        a_arr[j] = best_val
        return compute_score(a_arr)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        c_max = float(c.max())
        k_star = int(c.argmax())
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = 2.0 * a[idx]
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * c_max / S3)
        order = np.argsort(np.abs(grad))[::-1][:top_k]

        for j in order:
            if time.monotonic() > deadline:
                break
            sc = golden_1d(a, int(j), n_evals=10)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_score, best_a


def parametric_init(n, rng, family="cosine"):
    """Generate initial sequences from parametric families."""
    x = np.arange(n, dtype=np.float64)
    if family == "cosine":
        a = 1.0 + 0.3 * np.cos(np.pi * x / (n - 1))
    elif family == "linear_tilt":
        a = 1.0 + 0.3 * (2 * x / (n - 1) - 1)
    elif family == "parabolic":
        a = 1.0 + 0.3 * (x / (n - 1) - 0.5) ** 2 * 4
    elif family == "sinusoidal":
        f = float(rng.uniform(1, 4))
        a = 1.0 + 0.2 * np.sin(2 * np.pi * f * x / n)
    elif family == "bimodal":
        a = 1.0 + 0.4 * np.abs(np.sin(np.pi * x / (n - 1)))
    else:
        a = np.ones(n)
    a = np.clip(a, 0.01, 10.0)
    return a

    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
