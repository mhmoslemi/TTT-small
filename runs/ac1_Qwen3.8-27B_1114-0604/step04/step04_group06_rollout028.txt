    # C has shape (|active_set|, n)
    
    c = np.zeros((len(active_set), n))
    for idx, k in enumerate(active_set):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            j_arr = np.arange(js, je + 1)
            c[idx, j_arr] = f[k - j_arr]

    # We want to maximize sum(g), so minimize -sum(g)
    c_obj = -np.ones(n)
    b_ub = np.full(len(active_set), M)
    
    res = linprog(c_obj, A_ub=c, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if res.status != 0:
        return None
    return res.x

Minimize 2*n*max(self-convolution)/(sum)^2 for non-negative sequences.
Uses a combination of:
1. LSE-smoothed Adam optimization for smooth local optimization
2. LP-based direction finding for global improvement (main innovation)
3. Coordinate descent for final polishing
The LP approach finds descent directions by solving a linear program
that maximizes the sum of the direction subject to the cross-convolution
constraint, providing a more global view than gradient methods.

def fast_conv(a):

    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def solve_lp_direction(f, M, active_set):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_set, g >= 0."""
    n = len(f)
    K = len(active_set)
    
    # Build constraint matrix C (K x n)
    # C[k_idx, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    C = np.zeros((K, n))
    for idx, k in enumerate(active_set):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            j_arr = np.arange(js, je + 1)
            C[idx, j_arr] = f[k - j_arr]
    
    # linprog: minimize -sum(g) s.t. C@g <= M*ones, g >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    
    res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    
    if res.status != 0:
        return None
    return res.x

def lp_descent(f_init, budget_s, t_start):
    """LP-based descent method."""
    f = f_init.astype(np.float64).copy()
    S = f.sum()
    if S < 1e-12:
        f[:] = 1.0 / len(f)
        S = 1.0
    f = f / S  # normalize to sum = 1
    n = len(f)
    
    best_score = compute_score(f)
    best_f = f.copy()
    
    t_step = 0.15  # mixing parameter
    active_set_size = 30
    consecutive_no_improve = 0
    max_no_improve = 20
    
    start_time = time.monotonic()
    
    while time.monotonic() - t_start < budget_s:
        elapsed = time.monotonic() - start_time
        c = fast_conv(f)
        M = float(c.max())
        
        # Select active set: top positions where c is large
        top_idx = np.argsort(c)[-active_set_size:]
        active_set = top_idx.tolist()
        
        # Solve LP
        g = solve_lp_direction(f, M, active_set)
        if g is None or g.sum() < 1e-10:
            # Try fewer constraints
            active_set_small = top_idx[-10:].tolist()
            g = solve_lp_direction(f, M, active_set_small)
            if g is None or g.sum() < 1e-10:
                break
        
        # Check all constraints
        fg = conv_ab(f, g)
        violated = np.where(fg > M + 1e-10)[0]
        if len(violated) > 0:
            # Add violated constraints and re-solve
            extended_set = list(set(active_set + violated.tolist()))
            # Keep at most 3*active_set_size constraints
            if len(extended_set) > 3 * active_set_size:
                # Prioritize most violated
                viol_order = violated[np.argsort(fg[violated])[::-1]]
                extended_set = active_set + viol_order.tolist()[:active_set_size]
            g = solve_lp_direction(f, M, extended_set)
            if g is None or g.sum() < 1e-10:
                break
        
        # Rescale g to sum = 1
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Try update
        f_new = (1 - t_step) * f + t_step * g
        score = compute_score(f_new)
        
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            consecutive_no_improve = 0
            # Maybe increase step size
            if t_step < 0.3:
                t_step = min(0.3, t_step * 1.05)
        else:
            consecutive_no_improve += 1
            t_step = max(0.02, t_step * 0.7)
            
            if consecutive_no_improve > max_no_improve:

                    f = np.clip(f, 0, None)
                    f = f / f.sum()
                    active_set_size = 30
    
    return best_score, best_f

def coordinate_descent(a, top_k=50, max_sweeps=20, time_limit_s=None, t_start=None):
    """Coordinate descent on the exact objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (time.monotonic() + time_limit_s) if (time_limit_s is not None) else (t_start + 2400 if t_start else np.inf)
    
    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        S = a.sum()
        M = c.max()
        k_star = int(c.argmax())
        
        # Subgradient at k_star
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = 2.0 * a[idx]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        
        # Sort by |gradient| and optimize top_k
        order = np.argsort(np.abs(grad))[::-1][:top_k]
        
        for j in order:
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            # Try several values
            base = a_old
            if base < 1e-15:
                base = 1.0 / n
            candidates = np.array([0.0, base*0.3, base*0.6, base*0.8, base, base*1.2, base*1.5, base*2.0])
            best_val = a_old
            best_sc = best_score  # use current global best as reference
            # Actually, use the score with current a
            cur_sc = compute_score(a)
            for x in candidates:
                a[j] = x
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_val = x
            a[j] = best_val
        
        cur_sc = compute_score(a)
        if cur_sc < best_score:
            best_score = cur_sc
    
    return best_score, a

"""
Minimize 2*n*max(self-convolution)/(sum)^2 for non-negative sequences.
Primary innovation: LP-based descent method that solves a linear program
to find globally optimal descent directions, combined with LSE-smoothed
Adam for local optimization and coordinate descent for polishing.
The LP finds direction g maximizing sum(g) subject to cross-convolution
bound f*g <= M, providing a more global view than gradient methods.
"""

import numpy as np
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def conv_ab(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a) + len(b) - 1
    M_fft = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:n]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Compute sum_k p[k]*a[k-j] for j=0..n-1 via cross-correlation
    M3 = 2 * n - 1
    M3_fft = 1 << (M3 - 1).bit_length()
    P = np.fft.rfft(p, M3_fft)
    A_ft = np.fft.rfft(a, M3_fft)
    xcorr = np.fft.irfft(P * np.conj(A_ft), M3_fft)
    grad_b = 2.0 * xcorr[0:n]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_b / S2 - 2.0 * L / S3)
    return score, grad


def adam_lse(a_init, lr, n_steps, beta, time_limit_s, t_start):
    """Adam optimizer with LSE-smoothed gradient."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    a = a / a.sum()

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, n_steps + 1):
        if time.monotonic() - t_start > time_limit_s:
            break
        lr_t = lr / (1.0 + 1e-5 * t)
        score, grad = lse_score_grad(a, beta)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 1e-12:
            a = a / s
        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def solve_lp_direction(f, M, active_set):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in active_set, g >= 0."""
    n = len(f)
    K = len(active_set)
    C = np.zeros((K, n))
    for idx in range(K):
        k = active_set[idx]
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            j_arr = np.arange(js, je + 1)
            C[idx, j_arr] = f[k - j_arr]
    c_obj = -np.ones(n)
    b_ub = np.full(K, M)
    bounds = [(0.0, None)] * n
    res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x


def lp_descent(f_init, budget_s, t_start, verbose=True):
    """LP-based descent: iteratively find optimal directions via LP."""
    f = f_init.astype(np.float64).copy()
    S = f.sum()
    if S < 1e-12:
        f[:] = 1.0 / len(f)
    else:
        f = f / S
    n = len(f)

    best_score = compute_score(f)
    best_f = f.copy()

    t_step = 0.12
    active_set_size = 25
    consecutive_no_improve = 0
    rng = np.random.default_rng(123)
    iteration = 0

    while time.monotonic() - t_start < budget_s:
        iteration += 1
        c = fast_conv(f)
        M = float(c.max())

        # Active set: top positions + always include boundaries
        top_idx = np.argsort(c)[-active_set_size:]
        active_set = list(set(top_idx.tolist() + [0, n - 1, 2 * n - 2]))

        g = solve_lp_direction(f, M, active_set)
        if g is None or g.sum() < 1e-10:
            active_set_small = list(set(top_idx[-5:].tolist() + [0, n - 1]))
            g = solve_lp_direction(f, M, active_set_small)
            if g is None or g.sum() < 1e-10:
                break

        fg = conv_ab(f, g)
        violated = np.where(fg > M + 1e-10)[0]
        if len(violated) > 0:
            viol_sorted = violated[np.argsort(fg[violated])[::-1]]
            extended = list(set(active_set + viol_sorted[:active_set_size].tolist()))
            g = solve_lp_direction(f, M, extended)
            if g is None or g.sum() < 1e-10:
                break

        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        f_new = (1.0 - t_step) * f + t_step * g
        score = compute_score(f_new)

        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            consecutive_no_improve = 0
            if t_step < 0.25:
                t_step = min(0.25, t_step * 1.03)
        else:
            consecutive_no_improve += 1
            t_step = max(0.03, t_step * 0.75)
            if consecutive_no_improve > 30:
                t_step = 0.12
                active_set_size = min(active_set_size + 15, 80)
                consecutive_no_improve = 0
                if active_set_size >= 80:
                    f = f + 0.02 * rng.random(n)
                    f = np.clip(f, 0, None)
                    f = f / f.sum()
                    active_set_size = 25

        if verbose and iteration % 50 == 0:
            elapsed = time.monotonic() - t_start
            print(f"    LP iter {iteration}: score={best_score:.8f}, "
                  f"t_step={t_step:.4f}, active={active_set_size}, elapsed={elapsed:.0f}s")

    return best_score, best_f


def subgrad_coord_descent(a, top_k=50, max_sweeps=30, time_limit_s=None, t_start=None):
    """Coordinate descent using subgradient information."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    if time_limit_s is not None and t_start is not None:
        deadline = t_start + time_limit_s
    else:
        deadline = np.inf

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        k_star = int(c.argmax())
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = 2.0 * a[idx]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        order = np.argsort(np.abs(grad))[::-1][:top_k]
        for j in order:
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            if a_old < 1e-15:
                base = 1.0 / n
            else:
                base = a_old
            candidates = np.array([0.0, base * 0.2, base * 0.5, base * 0.8,
                                   base, base * 1.3, base * 1.7, base * 2.5])
            cur_sc = best_score
            best_val = a_old
            for x in candidates:
                a[j] = x
                sc = compute_score(a)
                if sc < cur_sc:
                    cur_sc = sc
                    best_val = x
            a[j] = best_val
        sc = compute_score(a)
        if sc < best_score:
            best_score = sc

    return best_score, a


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for minimum of 2*n*max(conv(a,a))/(sum(a))^2.
    Uses LP-based descent (main), LSE-Adam (polish), and coordinate descent.
    """
    t_start = time.monotonic()
    effective_budget = max(60.0, float(budget_s)) - 25.0

    # Get parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64 and p.sum() > 0.01:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        parent = parent / parent.sum()
        n0 = len(parent)
        print(f"Parent: n={n0}, score={compute_score(parent):.8f}")
    else:
        n0 = 4096
        parent = np.ones(n0) / n0
        print(f"No parent, starting with uniform n={n0}")

    current_a = parent.copy()
    current_score = compute_score(current_a)

    # Phase 1: Quick LSE-Adam to refine starting point
    print(f"\n{'='*60}")
    print("Phase 1: LSE-Adam initialization")
    print(f"{'='*60}")

    phase1_time = min(250.0, effective_budget * 0.12)
    for beta in [300, 800]:
        remaining = effective_budget - (time.monotonic() - t_start)
        if remaining < 30:
            break
        time_alloc = min(remaining * 0.4, phase1_time / 2.0)
        t0 = time.monotonic()
        sc, a_opt = adam_lse(current_a, lr=0.004, n_steps=200000, beta=beta,
                             time_limit_s=time_alloc, t_start=t_start)
        el = time.monotonic() - t0
        print(f"  Adam beta={beta}: {sc:.10f} ({el:.0f}s)")
        if sc < current_score:
            current_score = sc
            current_a = a_opt

    # Phase 2: LP-based descent (main optimization)
    print(f"\n{'='*60}")
    print("Phase 2: LP-based descent (main)")
    print(f"{'='*60}")

    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 60:
        phase2_time = min(remaining * 0.55, 1400.0)
        t0 = time.monotonic()
        sc, a_opt = lp_descent(current_a, budget_s=phase2_time, t_start=t_start)
        el = time.monotonic() - t0
        print(f"  LP descent: {sc:.10f} ({el:.0f}s)")
        if sc < current_score:
            current_score = sc
            current_a = a_opt

    # Phase 3: LSE-Adam polish
    print(f"\n{'='*60}")
    print("Phase 3: LSE-Adam polish")
    print(f"{'='*60}")

    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 30:
        for beta in [1000, 2000]:
            remaining = effective_budget - (time.monotonic() - t_start)
            if remaining < 20:
                break
            time_alloc = min(remaining * 0.25, 180.0)
            t0 = time.monotonic()
            sc, a_opt = adam_lse(current_a, lr=0.002, n_steps=200000, beta=beta,
                                 time_limit_s=time_alloc, t_start=t_start)
            el = time.monotonic() - t0
            print(f"  Adam polish beta={beta}: {sc:.10f} ({el:.0f}s)")
            if sc < current_score:
                current_score = sc
                current_a = a_opt

    # Phase 4: Coordinate descent
    print(f"\n{'='*60}")
    print("Phase 4: Coordinate descent")
    print(f"{'='*60}")

    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 20:
        t0 = time.monotonic()
        time_alloc = min(remaining * 0.35, 250.0)
        sc, a_opt = subgrad_coord_descent(current_a, top_k=50, max_sweeps=50,
                                          time_limit_s=time_alloc, t_start=t_start)
        el = time.monotonic() - t0
        print(f"  Coord descent: {sc:.10f} ({el:.0f}s)")
        if sc < current_score:
            current_score = sc
            current_a = a_opt

    # Phase 5: Final LP round (shorter)
    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 60:
        print(f"\n{'='*60}")
        print("Phase 5: Final LP refinement")
        print(f"{'='*60}")
        t0 = time.monotonic()
        time_alloc = min(remaining * 0.5, 250.0)
        sc, a_opt = lp_descent(current_a, budget_s=time_alloc, t_start=t_start, verbose=False)
        el = time.monotonic() - t0
        print(f"  Final LP: {sc:.10f} ({el:.0f}s)")
        if sc < current_score:
            current_score = sc
            current_a = a_opt

    # Phase 6: Final coordinate descent
    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 15:
        t0 = time.monotonic()
        time_alloc = min(remaining * 0.5, 100.0)
        sc, a_opt = subgrad_coord_descent(current_a, top_k=30, max_sweeps=20,
                                          time_limit_s=time_alloc, t_start=t_start)
        el = time.monotonic() - t0
        print(f"\n  Final coord descent: {sc:.10f} ({el:.0f}s)")
        if sc < current_score:
            current_score = sc
            current_a = a_opt

    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in current_a]
    if sum(out) < 0.01:
        out = [1.0] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={current_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
