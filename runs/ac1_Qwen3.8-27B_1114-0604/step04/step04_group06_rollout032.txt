```python
def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)

    a_new = np.clip(a_new, 0.0, None)
```

**Coordinate descent with 1D minimization:**

Given the current sequence a and the current convolution b = a*a:

For coordinate m, if I change a[m] from a_old to x:
- The sum changes: S_new = S - a_old + x
- The affected lags are k in [m, m+n-1] ∩ [0, 2n-2]:
  - If k ≠ 2m: b_new[k] = b[k] + 2*(x - a_old)*a[k-m]
  - If k = 2m: b_new[k] = b[k] + (x^2 - a_old^2)
- Unaffected lags: b_new[k] = b[k]
- max_b_new = max(max of affected b_new, max of unaffected b)
- score(x) = 2n * max_b_new(x) / S_new(x)^2

To find the optimal x:
- The function score(x) for x >= 0 is piecewise: the max_b_new(x) is the upper envelope of linear (and one quadratic) functions, and S_new(x)^2 is a quadratic.
- The minimum can be found by checking:
  1. The "breakpoints" where the argmax of b_new changes
  2. The critical points of score(x) within each piece

```python
import numpy as np

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):

    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):

    g = 2.0 * conv_res

def subgradient_adaptive(a_init, time_limit_s=None, t_start=None,
                         init_step=0.01, min_step=1e-6, max_iter=200000):
    """Subgradient method with adaptive step size (polyak-like)."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)
    step = init_step
    best_score_lb = best_score  # running minimum

    for t in range(1, max_iter + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break

        score, grad = exact_subgrad_score(a)

        # Adaptive step: decrease if stuck, use Polyak-like
        gap = score - best_score_lb
        if gap < 1e-10:
            step = max(min_step, step * 0.95)
        else:
            # Polyak step size
            g_norm = np.linalg.norm(grad)
            if g_norm > 1e-12:
                polyak_step = gap / (g_norm * g_norm)
                step = max(min_step, min(step * 1.01, 0.1 * polyak_step + step * 0.99))

        a_new = a - step * grad
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s > 1e-12:
            a_new = a_new / s

        sc_new = compute_score(a_new)
        if sc_new < score:
            a = a_new
            if sc_new < best_score:
                best_score = sc_new
                best_a = a.copy()
                best_score_lb = best_score
        else:
            step = step * 0.8

    return best_score, best_a


def coord_descent_selective(a_init, top_k=50, time_limit_s=None, t_start=None,
                            n_restarts=5):
    """Selective coordinate descent: update top-K |subgrad| coordinates iteratively."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    a = a / a.sum()

    best_score = compute_score(a)
    deadline = (t_start + time_limit_s) if (time_limit_s is not None and t_start is not None) else np.inf

    for restart in range(n_restarts):
        if time.monotonic() > deadline:
            break
        score, grad = exact_subgrad_score(a)
        order = np.argsort(np.abs(grad))[::-1][:top_k]

        for j in order:
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            if abs(grad[j]) < 1e-12:
                continue

            # Generate candidates
            uniform_val = a.sum() / n
            candidates = [0.0, a_old * 0.3, a_old * 0.7, a_old * 1.3, a_old * 1.8,
                          uniform_val, 0.5 * a_old + 0.5 * uniform_val]
            if grad[j] > 0:
                candidates.append(a_old * 0.5)
                candidates.append(a_old * 0.8)
            else:
                candidates.append(a_old * 2.0)
                candidates.append(a_old + 0.5 * (uniform_val - a_old))

            candidates = np.clip(np.array(candidates), 0.0, 1000.0)

            current_score = compute_score(a)
            best_x = a_old
            best_sc = current_score
            for x in candidates:
                if abs(x - a_old) < 1e-15:
                    continue
                a[j] = x
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_x = x
                if time.monotonic() > deadline:
                    break
            a[j] = best_x

        final_sc = compute_score(a)
        if final_sc < best_score:
            best_score = final_sc

    return best_score, a

        print(f"No parent found, using n={n0}")

    # Phase 1: Generate diverse starts and evaluate
    print(f"\n{'='*60}")
    print(f"Phase 1: Diverse initializations (n={n0})")
    print(f"{'='*60}")

    starts = generate_diverse_starts(n0, rng)
    start_scores = []
    for i, s in enumerate(starts):
        sc = compute_score(s)
        start_scores.append((sc, i, s))
        print(f"  Start {i}: score={sc:.10f}")
        try_update(sc, s, f"start {i}")

    if parent is not None:
        # Also add parent-based starts: parent + noise
        p_norm = parent / parent.sum()
        for noise_level in [0.01, 0.05, 0.1]:
            p_noisy = p_norm + noise_level * rng.standard_normal(n0)
            p_noisy = np.clip(p_noisy, 0.0, None)
            if p_noisy.sum() > 0.01:
                p_noisy = p_noisy / p_noisy.sum()
                sc = compute_score(p_noisy)
                print(f"  Parent+noise({noise_level}): score={sc:.10f}")
                try_update(sc, p_noisy, f"parent+noise {noise_level}")
                start_scores.append((sc, 100 + int(noise_level * 1000), p_noisy))

    # Sort starts by score, take top 3 for optimization
    start_scores.sort(key=lambda x: x[0])
    top_starts = [s[2] for s in start_scores[:3]]

    # Phase 2: LSE Adam on top starts
    print(f"\n{'='*60}")
    print(f"Phase 2: LSE Adam optimization")
    print(f"{'='*60}")

    adam_configs = [
        (50, 0.01), (200, 0.005), (500, 0.003), (1000, 0.002)
    ]

    for si, s_init in enumerate(top_starts):
        if remaining() < 30:
            break
        time_per_cfg = min(remaining() * 0.25, 120.0) / max(1, len(adam_configs))
        a_cur = s_init.copy()
        for beta, lr in adam_configs[:2]:  # Use 2 configs per start
            if remaining() < 15:
                break
            sc, a_opt = adam_lse(a_cur, lr=lr, n_steps=200000, beta_lse=beta,
                                  time_limit_s=time_per_cfg, t_start=t_start)
            try_update(sc, a_opt, f"Adam start{si} beta={beta}")
            a_cur = a_opt

    # Phase 3: Subgradient with adaptive step
    print(f"\n{'='*60}")
    print(f"Phase 3: Subgradient adaptive")
    print(f"{'='*60}")

    if remaining() > 30 and best_a is not None:
        time_alloc = min(remaining() * 0.3, 300.0)
        t0 = time.monotonic()
        sc, a_opt = subgradient_adaptive(best_a, time_limit_s=time_alloc, t_start=t_start,
                                          init_step=0.005, n_steps=200000)
        try_update(sc, a_opt, "subgradient adaptive")
        print(f"  Subgradient: {sc:.10f} ({time.monotonic()-t0:.1f}s)")

    # Phase 4: LSE Adam with high beta for finer optimization
    print(f"\n{'='*60}")
    print(f"Phase 4: High-beta LSE Adam")
    print(f"{'='*60}")

    if remaining() > 30 and best_a is not None:
        for beta, lr in [(2000, 0.002), (5000, 0.001)]:
            if remaining() < 15:
                break
            time_alloc = min(remaining() * 0.25, 150.0)
            t0 = time.monotonic()
            sc, a_opt = adam_lse(best_a, lr=lr, n_steps=200000, beta_lse=beta,
                                  time_limit_s=time_alloc, t_start=t_start)
            try_update(sc, a_opt, f"High-beta Adam beta={beta}")
            print(f"  Adam beta={beta}: {sc:.10f} ({time.monotonic()-t0:.1f}s)")

    # Phase 5: Coordinate descent
    print(f"\n{'='*60}")
    print(f"Phase 5: Coordinate descent")
    print(f"{'='*60}")

    if remaining() > 20 and best_a is not None:
        time_alloc = min(remaining() * 0.3, 300.0)
        t0 = time.monotonic()
        sc, a_opt = coord_descent_selective(best_a, top_k=50, time_limit_s=time_alloc,
                                             t_start=t_start, n_restarts=5)
        try_update(sc, a_opt, "coord descent k=50")
        print(f"  Coord descent k=50: {sc:.10f} ({time.monotonic()-t0:.1f}s)")

    # Phase 6: L-BFGS polish
    print(f"\n{'='*60}")
    print(f"Phase 6: L-BFGS polish")
    print(f"{'='*60}")

    if remaining() > 20 and best_a is not None:
        for beta in [1000, 5000, 10000]:
            if remaining() < 15:
                break
            t0 = time.monotonic()
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=80000)
            try_update(sc, a_opt, f"L-BFGS beta={beta}")
            print(f"  L-BFGS beta={beta}: {sc:.10f} ({time.monotonic()-t0:.1f}s)")

    # Phase 7: Final coordinate descent with smaller k
    print(f"\n{'='*60}")
    print(f"Phase 7: Fine coordinate descent")
    print(f"{'='*60}")

    if remaining() > 15 and best_a is not None:
        time_alloc = min(remaining() * 0.4, 200.0)
        t0 = time.monotonic()
        sc, a_opt = coord_descent_selective(best_a, top_k=20, time_limit_s=time_alloc,
                                             t_start=t_start, n_restarts=10)
        try_update(sc, a_opt, "fine coord descent k=20")
        print(f"  Fine coord k=20: {sc:.10f} ({time.monotonic()-t0:.1f}s)")

    # Phase 8: Final L-BFGS with very high beta
    if remaining() > 15 and best_a is not None:
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, 20000, maxiter=8000, maxfun=150000)
        try_update(sc, a_opt, "final L-BFGS beta=20000")
        print(f"  Final L-BFGS: {sc:.10f} ({time.monotonic()-t0:.1f}s)")

    # Final output
    if best_a is None:
        best_a = np.ones(64) / 64

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [1.0] * 64

    verified = float(evaluate_sequence(out))
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed():.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
```
