Multi-phase optimizer for flat-autocorrelation non-negative sequences.
Uses LP-based alternating method with HiGHS solver, frequency-domain error reduction,
LSE-smoothed L-BFGS, and coordinate descent for fine-tuning.
Key improvements: proper LP solving, adaptive peak clipping in frequency domain,
and multi-scale local search.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    # Gradient of L w.r.t. a
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val * a / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        return lse_score_and_grad(x, beta)

    res = minimize(
        obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def freq_domain_reflection(a, n_iters=50, alpha=0.5, clip_frac=0.8):
    """
    Frequency-domain error reduction with peak clipping.
    Modifies the spectrum to reduce convolution peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)
        # Convolution spectrum is A^2 (element-wise)
        C_spec = A * A
        c = np.fft.irfft(C_spec, M)[:2 * n - 1]

        c_max = float(c.max())
        c_mean = float(c.mean())
        if c_max - c_mean < 1e-15:
            break

        # Clip peaks: target is c with peaks reduced
        threshold = c_mean + clip_frac * (c_max - c_mean)
        c_target = np.minimum(c, threshold)
        # Add small flat component to avoid zeroing
        c_target = c_target + 0.05 * c_mean

        C_target_spec = np.fft.rfft(c_target, M)
        # Update A: we want A_new^2 ~ C_target_spec
        # Use: A_new = A + alpha * (sqrt(C_target/A) - A) where A != 0
        # Better: geometric mean approach
        with np.errstate(divide='ignore', invalid='ignore'):
            ratio = C_target_spec / C_spec
        # Take sqrt of ratio to get the multiplicative update on A
        # ratio = (A_new/A)^2, so A_new = A * sqrt(ratio)
        # But ratio is complex, so we handle phase carefully
        mag_ratio = np.abs(ratio)
        phase_A = np.angle(A)
        # New magnitude: sqrt of current mag^2 * ratio_mag... 
        # Actually: |A_new|^2 should be |C_target|, and phase should be halved
        # C_target phase = 2*phase(A_new), so phase(A_new) = phase(C_target)/2
        phase_C_target = np.angle(C_target_spec)
        new_phase = phase_C_target / 2.0
        new_mag = np.sqrt(np.abs(C_target_spec))
        
        A_new_mag = (1 - alpha) * np.abs(A) + alpha * new_mag
        A_new_phase = (1 - alpha) * phase_A + alpha * new_phase
        # Handle phase wrapping
        A_new = A_new_mag * np.exp(1j * A_new_phase)

        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def lp_solve_fista(a, M, n_fista_iters=3000, beta_start=5.0, beta_end=100.0):
    """
    Solve: max sum(x) s.t. conv(a,x)[k] <= M for all k, x >= 0
    using FISTA on the LSE-smoothed surrogate.
    """
    n = len(a)
    x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    sum_a2 = float(np.sum(a * a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n

    for it in range(n_fista_iters):
        frac = min(1.0, it / (n_fista_iters * 0.3))
        beta = beta_start + (beta_end - beta_start) * frac

        L = beta * n * sum_a2 + 1e-12
        step = 1.0 / L

        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum

        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(p) - n + 1]
        grad = a * sw - 1.0

        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    return x


def lp_alternating(a, max_outer_iters=200, fista_iters=3000, t_start=0.4, t_min=0.05,
                   time_limit=None):
    """LP-based alternating method (FISTA sub-solver)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve_count = 0

    for outer in range(max_outer_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        M = float(self_conv(a).max())
        x = lp_solve_fista(a, M, n_fista_iters=fista_iters)
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx

        t = max(t_min, t_start * (1.0 - 0.5 * outer / max(1, max_outer_iters)))
        a_new = (1.0 - t) * a + t * x
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count >= 20:
                t = t * 0.5
                if t < 0.01:
                    break
                no_improve_count = 0

        a = a_new

    return best_a, best_sc


def coordinate_descent(a, n_passes=5, delta_frac=0.05, n_deltas=7):
    """Coordinate descent: optimize one element at a time."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    deltas = np.linspace(-delta_frac, delta_frac, n_deltas)

    for p in range(n_passes):
        for j in range(n):
            if a[j] < 1e-10:
                continue
            a_save = a[j]
            best_local = best_sc
            best_val = a_save
            for d in deltas:
                new_val = a_save * (1.0 + d)
                if new_val < 0:
                    continue
                a[j] = new_val
                S_cur = float(a.sum())
                if S_cur < 1e-12:
                    a[j] = a_save
                    continue
                c = self_conv(a / S_cur)
                sc = 2.0 * n * float(c.max()) / (S_cur * S_cur)
                if sc < best_local:
                    best_local = sc
                    best_val = new_val
            a[j] = best_val
            if best_local < best_sc - 1e-10:
                best_sc = best_local
                best_a = a / float(a.sum())

    return best_a


def subgradient_polish(a, total_iters=2000):
    """Projected subgradient descent on the exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(total_iters):
        c = self_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M_val * a / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min((sc - 1.0) / (gn * gn + 1e-30), 0.03)
        step = max(step, 1e-6)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    # Smooth noise
    noise = rng.standard_normal(n)
    # Simple moving average smoothing
    k = max(3, n // 100)
    kernel = np.ones(k) / k
    smooth = np.convolve(noise, kernel, mode='same')
    inits.append(np.clip(0.5 + 0.4 * smooth, 0.01, 1000.0))
    # Multi-frequency
    inits.append(0.5 + 0.2 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(8.0 * np.pi * x))
    # Exponential decay from edges
    inits.append(np.exp(-2.0 * (x - 0.5) ** 2 / (0.1 + 0.4 * rng.random())))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-phase: freq-domain init, LP alternating, L-BFGS, coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 15.0

    # Try to load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent available: {parent is not None}, len={0 if parent is None else len(parent)}")

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n_try in [4096, 6962, 8192]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))[:4]
    print(f"Lengths to explore: {lengths}")

    # ==================================================================
    # Phase 1: Frequency-domain initialization (15% time)
    # ==================================================================
    phase1_deadline = t_start + 0.15 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 1: Frequency-Domain Init ---")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        remaining = phase1_deadline - t_start
        if remaining < 20:
            break
        print(f"  n={n}")
        inits = build_initializations(n, rng, parent)
        max_inits = max(3, int(remaining / 8.0))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a0 = np.maximum(a0, 0.0)
            S = float(a0.sum())
            if S > 0:
                a0 = a0 / S
            # Quick frequency-domain reflection
            a_fr = freq_domain_reflection(a0, n_iters=40, alpha=0.6, clip_frac=0.7)
            sc = compute_score(a_fr)
            report(sc, a_fr, f"FD n={n} init={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: LP-based alternating (40% time)
    # ==================================================================
    phase2_deadline = t_start + 0.55 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 2: LP Alternating ---")

    n_main = len(best_a)
    if parent is not None and len(parent) in lengths:
        n_main = len(parent)

    a_work = best_a if len(best_a) == n_main else np.interp(
        np.linspace(0, 1, n_main), np.linspace(0, 1, len(best_a)), best_a
    )
    a_work = np.maximum(a_work, 0.0)
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    # Progressive LP alternating with increasing precision
    configs = [
        (2000, 0.5, 0.1),
        (3000, 0.4, 0.08),
        (4000, 0.3, 0.05),
        (5000, 0.2, 0.03),
        (6000, 0.15, 0.02),
    ]

    for fi, (fista_it, t_s, t_m) in enumerate(configs):
        if time.monotonic() > phase2_deadline:
            break
        elapsed = time.monotonic() - t_start
        remaining = phase2_deadline - t_start
        if remaining < 30:
            break
        # Estimate iterations possible
        # Each FISTA iteration is roughly O(n log n) for the FFT
        # Each outer iter does fista_it inner iters
        est_time_per_outer = fista_it * n_main / 5e6  # rough
        max_outer = max(3, int(remaining / max(est_time_per_outer, 0.5)))
        max_outer = min(max_outer, 50)
        print(f"  LP round {fi}: n={n_main}, fista={fista_it}, t=[{t_s},{t_m}], outer~{max_outer}")

        a_out, sc_out = lp_alternating(
            a_work, max_outer_iters=max_outer, fista_iters=fista_it,
            t_start=t_s, t_min=t_m, time_limit=phase2_deadline - 5
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP r{fi}")
        a_work = a_out.copy()

    # Try other lengths if time permits
    for n in lengths:
        if n == n_main or time.monotonic() > phase2_deadline - 40:
            continue
        print(f"  LP alt length: n={n}")
        a_test = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(best_a)), best_a)
        a_test = np.maximum(a_test, 0.0)
        S = float(a_test.sum())
        if S > 1e-12:
            a_test = a_test / S
        a_out, sc_out = lp_alternating(
            a_test, max_outer_iters=20, fista_iters=2000,
            t_start=0.4, t_min=0.1, time_limit=phase2_deadline - 10
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP n={n}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: L-BFGS refinement (25% time)
    # ==================================================================
    phase3_deadline = t_start + 0.80 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 3: L-BFGS Refinement ---")

    n_best = len(best_a)
    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 1e-12:
        a_norm = a_norm / S

    beta_schedule = [20, 50, 100, 200, 500, 1000, 2000]
    for beta in beta_schedule:
        if time.monotonic() > phase3_deadline:
            break
        print(f"  L-BFGS beta={beta}")
        sc, a_opt = run_lbfgs(a_norm, beta, maxiter=4000, maxfun=150000)
        report(sc, a_opt, f"LBFGS b={beta}")
        if sc < best_score:
            a_norm = a_opt.copy()
            S = float(a_norm.sum())
            if S > 1e-12:
                a_norm = a_norm / S

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Local search and polish (20% time)
    # ==================================================================
    print(f"\n--- Phase 4: Local Search & Polish ---")

    n_best = len(best_a)
    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 1e-12:
        a_norm = a_norm / S

    # Coordinate descent
    if time.monotonic() < deadline - 120:
        print("  Coordinate descent...")
        a_cd = coordinate_descent(a_norm, n_passes=3, delta_frac=0.03, n_deltas=5)
        sc = compute_score(a_cd)
        report(sc, a_cd, "coord_desc")

    # Perturbation + L-BFGS
    n_trials = 10
    for trial in range(n_trials):
        if time.monotonic() > deadline - 60:
            break
        strategy = trial % 5
        if strategy == 0:
            # Small Gaussian perturbation
            scale = 0.005 + 0.02 * rng.random()
            a_pert = a_norm * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0,
