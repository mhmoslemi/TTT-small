from scipy.optimize import minimize, linprog

def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def gs_nn(a0, n_iters=500):

    n = len(a0)

Multi-method optimizer for flat-autocorrelation non-negative sequences.
Key innovations: (1) GS-NN (Gerchberg-Saxton with Non-Negativity) for spectral initialization,
(2) Exact LP alternating via HiGHS with line search, (3) Adaptive L-BFGS polish with
multi-beta schedule, (4) Peak-shaving local search targeting top autocorrelation peaks.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import coo_matrix, csr_matrix
import time


def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear cross-convolution via FFT."""
    n1, n2 = len(a), len(b)
    M = n1 + n2 - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

        a = a / S

    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

    best_a = a.copy()
    best_sc = compute_score(a)
    t0 = time.monotonic()

    for it in range(n_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        A = np.fft.rfft(a, M_fft)
        mag = np.abs(A)
        phase = A / (mag + 1e-30)

        # Flatten: blend current magnitude with uniform
        mag_target = np.full_like(mag, mag.mean())
        mag_target[0] = mag[0]  # preserve DC (sum)
        mag_new = flatten_strength * mag_target + (1.0 - flatten_strength) * mag
        A_new = phase * mag_new

        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a, best_sc


def build_conv_sparsity(n):
    """Build row/col arrays for the sparse convolution matrix (2n-1) x n."""
    M = 2 * n - 1
    total_nz = n * n
    rows = np.zeros(total_nz, dtype=np.int32)
    cols = np.zeros(total_nz, dtype=np.int32)

    offset = 0
    for k in range(M):
        s = max(0, k - n + 1)
        e = min(k, n - 1)
        L = e - s + 1
        rows[offset:offset + L] = k
        cols[offset:offset + L] = np.arange(s, e + 1)
        offset += L

    assert offset == total_nz
    return rows, cols


def lp_solve_convolution(a, M_val, rows, cols):
    """
    Solve LP: max sum(x) s.t. conv(a,x)[k] <= M_val for all k, x >= 0.
    Uses scipy HiGHS.
    """
    n = len(a)
    M = 2 * n - 1

    vals = a[rows - cols]
    A_ub = coo_matrix((vals, (rows, cols)), shape=(M, n)).tocsr()
    b_ub = np.full(M, M_val)
    c = -np.ones(n)

    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs',
                  options={'presolve': True, 'time_limit': 60.0})

    if not res.success:
        return None
    return res.x


def lp_alternating_exact(a, max_outer=80, time_limit=None, t_candidates=(0.3, 0.5, 0.7, 0.9, 1.0)):
    """
    LP alternating with exact LP solver.
    For each iteration: compute M, solve LP, line search on blend parameter.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    rows, cols = build_conv_sparsity(n)

    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0

    for outer in range(max_outer):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        t_lp = time.monotonic()
        M_val = float(self_conv(a).max())
        x = lp_solve_convolution(a, M_val, rows, cols)
        lp_time = time.monotonic() - t_lp

        if x is None:
            break

        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx

        # Line search on blending parameter
        best_t = t_candidates[0]
        best_new_sc = np.inf
        for t in t_candidates:
            a_new = (1.0 - t) * a + t * x
            a_new = np.maximum(a_new, 0.0)
            S_new = float(a_new.sum())
            if S_new < 1e-12:
                continue
            a_new = a_new / S_new
            sc = compute_score(a_new)
            if sc < best_new_sc:
                best_new_sc = sc
                best_t = t

        a_new = (1.0 - best_t) * a + best_t * x
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc - 1e-10:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 10:
                # Try smaller steps
                t_candidates = (0.05, 0.1, 0.15, 0.2)
                no_improve = 0
                if no_improve > 20:
                    break

        a = a_new
        elapsed = time.monotonic() - (time_limit - 600 if time_limit else t0)
        print(f"    LP iter {outer}: sc={sc:.8f}, M={M_val:.6f}, t={best_t:.2f}, lp_time={lp_time:.1f}s")

    return best_a, best_sc


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)

    p = exp_z / exp_z.sum()

    # Gradient of L w.r.t. a: 2 * cross-correlation(a, p) centered at n-1
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val * np.ones(n) / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        return lse_score_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_shaving(a, n_iters=500, n_peaks=5):
    """
    Targeted peak reduction: identify top peaks in autocorrelation,
    reduce contributing elements, redistribute mass.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = self_conv(a)
        S = float(a.sum())
        c_mean = float(c.mean())
        c_max = float(c.max())
        if c_max - c_mean < 1e-12:
            break

        # Find top peaks
        top_indices = np.argsort(c)[::-1][:n_peaks]
        for k_star in top_indices:
            if c[k_star] < c_mean + 1e-8:
                continue
            # Elements contributing to this peak: a[i] where i ranges appropriately
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end < j_start:
                continue

            # Contribution of each a[j] to c[k_star] is a[k_star - j]
            contrib = a[j_start:j_end + 1] * a[k_star - j_end:k_star - j_start + 1]
            # Actually: c[k_star] = sum_{j=j_start}^{j_end} a[j] * a[k_star - j]
            # d c[k_star] / d a[j] = a[k_star - j] (for fixed other a)

            # Reduce elements with high a[k_star - j] (they contribute most)
            a_at_mirror = a[k_star - j_end:k_star - j_start + 1]
            # Reduce top contributors
            n_reduce = max(1, (j_end - j_start + 1) // 5)
            top_contributors = np.argsort(a_at_mirror)[::-1][:n_reduce]
            total_reduce = 0.0
            for lc in top_contributors:
                gi = j_start + lc
                reduce_amt = a[gi] * 0.02
                a[gi] = max(a[gi] - reduce_amt, 0.0)
                total_reduce += reduce_amt

            # Redistribute to bottom contributors (or random positions)
            if total_reduce > 1e-12:
                bottom_count = max(1, (j_end - j_start + 1) - n_reduce)
                bottom_local = np.argsort(a_at_mirror)[:bottom_count]
                per = total_reduce / len(bottom_local)
                for lc in bottom_local:
                    a[j_start + lc] += per

        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def subgradient_momentum(a, n_iters=2000, lr=None, momentum=0.9):
    """Projected subgradient with momentum on the exact max-conv objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    t = 0

    for it in range(n_iters):
        t += 1
        c = self_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of max(conv(a,a))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        mirror = a[k_star - j_end:k_star - j_start + 1]
        sg[j_start:j_end + 1] = 2.0 * mirror

        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        if lr is None:
            step = 2.0 * (sc - 1.0) / (gn * gn + 1e-30)
        else:
            step = lr / (1.0 + 0.1 * it)
        step = min(step, 0.1)

        v = momentum * v - step * grad
        a_new = a + v
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            a_new = a
        else:
            a_new = a_new / S_new
        a = a_new

    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing
    2*n*max(conv(a,a)) / sum(a)^2.

    Strategy:
    1. GS-NN spectral initialization (15% time)
    2. Exact LP alternating with HiGHS + line search (50% time)
    3. L-BFGS multi-beta polish (20% time)
    4. Peak shaving + subgradient final polish (15% time)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent: {'yes' if parent is not None else 'no'}, "
          f"len={0 if parent is None else len(parent)}")

    # Determine lengths to try
    n_main = 6962
    if parent is not None:
        n_main = len(parent)
    lengths = sorted(set([n_main, 4096, 8192]))[:3]
    print(f"Lengths: {lengths}")

    # ================================================================
    # Phase 1: GS-NN spectral initialization (15% time)
    # ================================================================
    phase1_deadline = t_start + 0.15 * (total_budget - 30.0)
    print(f"\n=== Phase 1: GS-NN Initialization ===")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"  n={n}")

        inits = []
        if parent is not None:
            if len(parent) == n:
                inits.append(parent.copy())
            else:
                inits.append(np.interp(np.linspace(0, 1, n),
                                       np.linspace(0, 1, len(parent)), parent))
        x = np.arange(n, dtype=np.float64) / n
        inits.append(np.ones(n) * 0.5)
        inits.append(np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, 10.0))
        inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
        inits.append(0.2 + 0.8 * x)  # linear ramp
        inits.append(0.2 + 0.8 * (1.0 - x))  # linear decay

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a_gs, sc_gs = gs_nn(a0, n_iters=200, time_limit=phase1_deadline - 5)
            report(sc_gs, a_gs, f"GS n={n} i={i}")

    print(f"  After Phase 1: best={best_score:.8f}")

    # ================================================================
    # Phase 2: Exact LP alternating (50% time)
    # ================================================================
    phase2_deadline = t_start + 0.65 * (total_budget - 30.0)
    print(f"\n=== Phase 2: LP Alternating (HiGHS) ===")

    n_lp = len(best_a)
    # Use a moderate length for LP (building the matrix is O(n^2))
    # For very large n, downsample for LP then re-interpolate
    if n_lp > 8000:
        n_lp_eff = 6962
        a_lp = np.interp(np.linspace(0, 1, n_lp_eff),
                         np.linspace(0, 1, len(best_a)), best_a)
    else:
        n_lp_eff = n_lp
        a_lp = best_a.copy()

    a_lp = np.maximum(a_lp, 0.0)
    S = float(a_lp.sum())
    if S > 1e-12:
        a_lp = a_lp / S

    print(f"  LP length: {n_lp_eff}")
    a_out, sc_out = lp_alternating_exact(
        a_lp, max_outer=100, time_limit=phase2_deadline,
        t_candidates=(0.3, 0.5, 0.7, 0.9, 1.0)
    )

    # Re-interpolate back to original length if needed
    if n_lp_eff != len(best_a):
        a_out_full = np.interp(np.linspace(0, 1, len(best_a)),
                               np.linspace(0, 1, n_lp_eff), a_out)
    else:
        a_out_full = a_out

    report(sc_out, a_out_full, "LP")

    # Try LP on the main length directly if different
    if n_main != n_lp_eff and time.monotonic() < phase2_deadline - 60:
        print(f"  LP on n={n_main}...")
        a_lp2 = best_a.copy()
        S = float(a_lp2.sum())
        if S > 1e-12:
            a_lp2 = a_lp2 / S
        a_out2, sc_out2 = lp_alternating_exact(
            a_lp2, max_outer=50, time_limit=phase2_deadline,
            t_candidates=(0.3, 0.5, 0.7, 1.0)
        )
        report(sc_out2, a_out2, f"LP n={n_main}")

    print(f"  After Phase 2: best={best_score:.8f}")

    # ================================================================
    # Phase 3: L-BFGS polish (20% time)
    # ================================================================
    phase3_deadline = t_start + 0.85 * (total_budget - 30.0)
    print(f"\n=== Phase 3: L-BFGS Polish ===")

    a_work = best_a.copy()
    S = float(a_work.sum())
    if S > 1e-12:
        a_work = a_work / S

    beta_schedule = [20, 50, 100, 300, 800, 2000]
    for beta in beta_schedule:
        if time.monotonic() > phase3_deadline:
            break
        remaining = phase3_deadline - time.monotonic()
        maxiter = max(500, int(remaining * 20))
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxiter * 20)
        report(sc, a_opt, f"LBFGS b={beta}")
        a_work = a_opt.copy()
        S = float(a_work.sum())
        if S > 1e-12:
            a_work = a_work / S

    print(f"  After Phase 3: best={best_score:.8f}")

    # ================================================================
    # Phase 4: Peak shaving + subgradient (15% time)
    # ================================================================
    print(f"\n=== Phase 4: Peak Shaving + Subgradient ===")

    a_ps = best_a.copy()
    S = float(a_ps.sum())
    if S > 1e-12:
        a_ps = a_ps / S

    for n_peaks in [3, 5, 10]:
        if time.monotonic() > deadline - 60:
            break
        a_ps_out = peak_shaving(a_ps, n_iters=300, n_peaks=n_peaks)
        sc_ps = compute_score(a_ps_out)
        report(sc_ps, a_ps_out, f"peak n={n_peaks}")
        a_ps = a_ps_out.copy()

    # Subgradient with momentum
    if time.monotonic() < deadline - 120:
        print("  Subgradient polish...")
        a_sg = subgradient_momentum(best_a, n_iters=3000, momentum=0.85)
        sc_sg = compute_score(a_sg)
        report(sc_sg, a_sg, "subgrad")

    # Final L-BFGS with very high beta
    if time.monotonic() < deadline - 60:
        a_final = best_a.copy()
        S = float(a_final.sum())
        if S > 1e-12:
            a_final = a_final / S
        sc, a_opt = run_lbfgs(a_final, 5000, maxiter=3000, maxfun=60000)
        report(sc, a_opt, "final LBFGS")

    # Final GS-NN polish
    if time.monotonic() < deadline - 30:
        a_gs_final, sc_gs_final = gs_nn(best_a, n_iters=100, flatten_strength=0.9,
                                        time_limit=deadline - 10)
        report(sc_gs_final, a_gs_final, "final GS")

    # ================================================================
    # Final verification
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
