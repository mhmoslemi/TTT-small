Let me also think about a completely different approach:

**SDP-based approach:**
The problem min 2n*max(conv)/(sum)^2 can be reformulated. Let's fix sum(a) = 1 (by homogeneity). Then we want to minimize max_k conv_k subject to sum(a) = 1, a ≥ 0.

conv_k = a^T C_k a where C_k is a matrix with (C_k)_{ij} = 1 if i+j=k.

This is a quadratic minimization with quadratic constraints (conv_k ≤ t for all k) and a linear constraint (sum = 1). This is a quadratically constrained quadratic program (QCQP), which is NP-hard in general.

However, we can use an SDP relaxation:
min t
s.t. a^T C_k a ≤ t for all k
     sum(a) = 1
     a ≥ 0

The SDP relaxation replaces a^T C_k a ≤ t with [1, a^T; a, aa^T] ⪰ 0 and [C_k, 0; 0, t - 1^T C_k 1^T ... ] - [1, a^T; a, aa^T] ⪰ 0

"""
Optimized step function sequence minimizer.
Strategy: Multi-length LSE + L-BFGS with adaptive beta, peak-targeted coordinate descent,
and multi-scale perturbation. Focuses on intensive refinement over broad exploration.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta, top_k=None):
    """LSE-smoothed score and its exact gradient. Optionally only top-K peaks."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    if top_k is not None and top_k < len(c):
        # Only include top-K in the LSE
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_top = c[idx]
        c_max = float(c_top.max())
        scaled = beta * (c_top - c_max)
        scaled -= scaled.max()
        exp_s = np.exp(scaled)
        log_sum = np.log(exp_s.sum()) + scaled.max() / beta
        L = c_max + log_sum / beta
        p = exp_s / exp_s.sum()
        # Map indices back
        p_full = np.zeros(len(c))
        p_full[idx] = p
    else:
        c_max = float(c.max())
        scaled = beta * (c - c_max)
        shifted = scaled.max()
        scaled -= shifted
        exp_s = np.exp(scaled)
        log_sum = np.log(exp_s.sum()) + shifted / beta
        L = c_max + log_sum / beta
        p = exp_s / exp_s.sum()
        p_full = p
    
    score = 2.0 * n * L / (S * S)
    
    # Gradient: dL/da[i] = 2 * sum_j p[i+j] * a[j]
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p_full, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000, top_k=None):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        s, g = lse_obj_grad(x, beta, top_k=top_k)
        return s, g
    
    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_targeted_coord_descent(a, n_sweeps=10, top_k_peaks=5, verbose=False):
    """
    Coordinate descent targeting top peaks.
    For each peak, reduce the highest-contributing coordinates and
    redistribute to low-contributing ones.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        peak_idx = np.argsort(c)[::-1][:top_k_peaks]
        
        improved = False
        for k_star in peak_idx:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                continue
            
            seg = a[j_start:j_end + 1]
            # Contribution of each coordinate in the segment to conv[k_star]
            # a[j] contributes a[j] * a[k_star - j]
            k_vals = k_star - np.arange(j_start, j_end + 1)
            # Handle the diagonal case (j = k_star - j, i.e., 2j = k_star)
            contrib = seg * np.where(k_vals >= 0, a[np.clip(k_vals, 0, n-1)], 0)
            # For off-diagonal, double it
            off_diag = np.abs(np.arange(j_start, j_end + 1) - k_vals) > 0
            contrib[off_diag] *= 2.0
            
            n_seg = len(contrib)
            top_count = max(3, n_seg // 8)
            top_local = np.argsort(contrib)[::-1][:top_count]
            bottom_local = np.argsort(contrib)[:max(1, n_seg - top_count)]
            
            total_to_redistribute = 0.0
            new_s = S
            for li in top_local:
                gi = j_start + li
                # Try reducing by 5-20%
                red = a[gi] * 0.1
                a[gi] -= red
                new_s -= red
                total_to_redistribute += red
            
            if total_to_redistribute > 1e-15 and new_s > 1e-12:
                per = total_to_redistribute / len(bottom_local)
                for li in bottom_local:
                    gi = j_start + li
                    a[gi] += per
            
            S = a.sum()
            if S < 1e-12:
                a = best_a.copy()
                S = float(best_a.sum())
                break
            a = a / S
            S = 1.0
            
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                improved = True
        
        if not improved and sweep > 3:
            break
    
    return best_a


def adaptive_coord_descent(a, n_iters=5000, verbose=False):
    """
    Adaptive coordinate descent: for each coordinate, find the optimal value
    via a small 1D search (bisection on the derivative of the score).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    c = fast_conv(a)
    
    for it in range(n_iters):
        # Find current peak
        k_star = int(np.argmax(c))
        M = float(c[k_star])
        sc = 2.0 * n * M / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Identify contributing coordinates for the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        
        # Find the coordinate with largest contribution
        seg = a[j_start:j_end + 1]
        k_vals = k_star - np.arange(j_start, j_end + 1)
        contrib = seg * a[np.clip(k_vals, 0, n - 1)]
        off_diag = np.abs(np.arange(j_start, j_end + 1) - k_vals) > 0
        contrib[off_diag] *= 2.0
        
        # Try modifying the top contributor
        top_local = int(np.argmax(contrib))
        gi = j_start + top_local
        
        # Try a few values for a[gi]
        v_old = a[gi]
        best_v = v_old
        best_local = sc
        
        for frac in [0.5, 0.7, 0.85, 0.9, 1.0, 1.1, 1.3]:
            v_new = v_old * frac
            if v_new < 0 or abs(v_new - v_old) < 1e-10:
                continue
            a[gi] = v_new
            # Recompute affected conv values
            # Only need to check if the max decreases
            delta = v_new - v_old
            # Quick check: recompute conv at peak
            new_peak = M + 2 * delta * a[k_star - gi] if k_star - gi != gi else (v_new**2 - v_old**2)
            # Also check if another k becomes the max (approximate)
            if new_peak < M * 0.99:
                # Recompute full score (expensive but we only do it when promising)
                c_new = fast_conv(a)
                new_sc = 2.0 * n * float(c_new.max()) / (a.sum() ** 2)
                if new_sc < best_local:
                    best_local = new_sc
                    best_v = v_new
            else:
                a[gi] = v_old
            a[gi] = v_old
        
        if best_v != v_old:
            a[gi] = best_v
            S = a.sum()
            if S > 1e-12:
                a = a / S
            c = fast_conv(a)
        else:
            # No improvement from top contributor; try a random coordinate
            # that doesn't contribute much to the peak
            low_contrib = np.where(contrib < 0.1 * contrib.max())[0]
            if len(low_contrib) > 0:
                li = low_contrib[np.random.randint(len(low_contrib))]
                gi = j_start + li
                a[gi] *= (1.0 + 0.05 * (np.random.random() - 0.3))
                a[gi] = max(0.0, a[gi])
                S = a.sum()
                if S > 1e-12:
                    a = a / S
                c = fast_conv(a)
        
        if it % 100 == 0 and verbose:
            print(f"    ADC iter {it}: score={sc:.8f}")
    
    return best_a


def mirror_descent_refined(a, n_iters=3000, top_k=3):
    """
    Mirror descent with top-K peak weighting for smoother gradients.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        # Weight the top-K peaks
        idx = np.argpartition(c, -top_k)[-top_k:]
        c_top = c[idx]
        w = c_top / c_top.sum()
        
        grad_max = np.zeros(n)
        for ki, wi in zip(idx, w):
            ki = int(ki)
            j_start = max(0, ki - n + 1)
            j_end = min(ki, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                grad_max[j_start:j_end + 1] += wi * 2.0 * seg[::-1]
        
        eta = 0.3 / (1.0 + 0.05 * it)
        log_a = np.log(a) - eta * grad_max
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def frequency_domain_flatten(a, n_iters=300, top_frac=0.3):
    """
    Frequency-domain approach: modify the phase/amplitude of FFT to flatten
    the convolution (which is IFFT(|FFT|^2)).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    M = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        conv = np.fft.irfft(A * A, M)[:2 * n - 1]
        
        conv_max = float(conv.max())
        conv_mean = float(conv.mean())
        if conv_max - conv_mean < 1e-14:
            break
        
        # Identify high-convolution regions
        thresh = conv_mean + top_frac * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        
        if excess.max() < 1e-14:
            break
        
        # Compute correction in frequency domain
        err = -0.5 * excess
        err_pad = np.zeros(M, dtype=np.float64)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_pad, M)
        
        # Update spectrum: A_new = A + err_spec / (2*A)
        with np.errstate(divide='ignore', invalid='ignore'):
            damp = 0.3 + 0.3 * (1.0 - it / n_iters)
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-10] = 0.0
        A_new = A + damp * d
        
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def build_initializations(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())
    
    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.2 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(6.0 * np.pi * x))
    inits.append(0.3 + 0.7 * (1.0 - (2.0 * x - 1.0) ** 2))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    
    # Windowed: smooth bump in the middle
    sigma = 0.2 + 0.3 * rng.random()
    inits.append(np.exp(-0.5 * ((2.0 * x - 1.0) / sigma) ** 2))
    
    return inits

            a_cur = a_cd.copy()
        
        if time.monotonic() > phase3_deadline:
            break
        
        # Frequency domain flattening
        remaining = phase3_deadline - time.monotonic()
        n_iters = max(50, int(remaining / 2.0))
        a_fd = frequency_domain_flatten(a_cur, n_iters=n_iters, top_frac=0.2)
        sc = compute_score(a_fd)
        improved = report(sc, a_fd, f"P3 FD iters={n_iters}")
        if improved:
            a_cur = a_fd.copy()
        
        if time.monotonic() > phase3_deadline:
            break
        
        # Quick LSE polish
        remaining = phase3_deadline - time.monotonic()
        if remaining > 10:
            sc, a_opt = run_lbfgs(a_cur, 5000, maxiter=3000, maxfun=50000)
            improved = report(sc, a_opt, "P3 LSE polish")
            if improved:
                a_cur = a_opt.copy()
    
    print(f"Phase 3 done: best = {best_score:.8f}")
    
    # ==============================================================
    # Phase 4: Diversification - perturb and re-optimize (10% time)
    # ==============================================================
    phase4_deadline = t_start + 0.92 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 4: Diversification (deadline t={phase4_deadline - t_start:.0f}s)")
    print(f"{'='*60}")
    
    n = len(best_a)
    x = np.arange(n, dtype=np.float64) / n
    
    n_trials = 8
    for trial in range(n_trials):
        if time.monotonic() > phase4_deadline:
            break
        
        strategy = trial % 4
        if strategy == 0:
            # Gaussian perturbation
            scale = 0.02 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Sinusoidal perturbation
            freq = 2.0 + 8.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.01 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 2:
            # Targeted peak reduction + redistribution
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            a_pert = best_a.copy()
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            seg = a_pert[j_start:j_end + 1]
            k_vals = k_star - np.arange(j_start, j_end + 1)
            contrib = seg * a_pert[np.clip(k_vals, 0, n-1)]
            off_diag = np.abs(np.arange(j_start, j_end + 1) - k_vals) > 0
            contrib[off_diag] *= 2.0
            top_count = max(5, n // 40)
            top_idx = np.argsort(contrib)[::-1][:top_count]
            bot_idx = np.argsort(contrib)[:top_count]
            total_red = 0.0
            for li in top_idx:
                gi = j_start + li
                red = a_pert[gi] * 0.05
                a_pert[gi] -= red
                total_red += red
            per = total_red / len(bot_idx)
            for li in bot_idx:
                a_pert[j_start + li] += per
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Block-level perturbation
            a_pert = best_a.copy()
            n_blocks = 4 + int(rng.integers(0, 8))
            block_size = max(1, n // n_blocks)
            for b in range(min(n_blocks, n // block_size)):
                s = b * block_size
                e = min(s + block_size, n)
                factor = 0.8 + 0.4 * rng.random()
                a_pert[s:e] *= factor
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        
        S = a_pert.sum()
        if S < 1e-12:
            continue
        a_pert = a_pert / S
        
        # Optimize the perturbation
        remaining = phase4_deadline - time.monotonic()
        if remaining < 5:
            break
        sc, a_opt = run_lbfgs(a_pert, 2000, maxiter=min(5000, int(remaining*300)),
                             maxfun=min(20000, int(remaining*3000)))
        report(sc, a_opt, f"P4 trial={trial} s={strategy}")
    
    print(f"Phase 4 done: best = {best_score:.8f}")
    
    # ==============================================================
    # Phase 5: Final polish (8% time)
    # ==============================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final Polish")
    print(f"{'='*60}")
    
    remaining = deadline - time.monotonic()
    
    if remaining > 30:
        # High-beta LSE
        sc, a_opt = run_lbfgs(best_a, 50000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "P5 LSE b=50000")
    
    if remaining > 20:
        # Mirror descent
        iters = int(min(3000, remaining / 0.05))
        a_md = mirror_descent_refined(best_a, n_iters=iters, top_k=3)
        sc = compute_score(a_md)
        report(sc, a_md, f"P5 MD iters={iters}")
    
    if remaining > 15:
        # Final coordinate descent
        sweeps = int(min(20, remaining / 2.0))
        a_cd = peak_targeted_coord_descent(best_a, n_sweeps=sweeps, top_k_peaks=3)
        sc = compute_score(a_cd)
        report(sc, a_cd, f"P5 CD sweeps={sweeps}")
    
    if remaining > 10:
        # One more LSE from the best
        sc, a_opt = run_lbfgs(best_a, 10000, maxiter=3000, maxfun=50000)
        report(sc, a_opt, "P5 LSE b=10000 final")
    
    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
