"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses: correct LSE gradient + L-BFGS with adaptive beta, Fienup spectral flattening,

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

    Compute LSE-smoothed score and its gradient.
    score_smooth = 2n * L / S^2 where L = (1/beta)*log(sum(exp(beta*c)))
    Gradient uses the identity: dL/da_j = 2*(p_rev * a)[2n-2-j] where p = softmax(beta*c)
    """
    n = len(a)
    c = fast_conv(a)  # length 2n-1
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)

    # Compute dL/da using FFT
    # dL/da[j] = 2 * conv(p_rev, a)[2n-2-j] for j=0..n-1
    p_rev = p[::-1]  # length 2n-1
    conv_len = (2 * n - 1) + n - 1  # = 3n-2
    M = 1 << (conv_len - 1).bit_length()
    PR = np.fft.rfft(p_rev, M)
    AR = np.fft.rfft(a, M)
    conv = np.fft.irfft(PR * AR, M)[:conv_len]
    # dL/da[j] = 2 * conv[2n-2-j] for j=0..n-1
    # So dL_da = 2 * conv[n-1:2n-1][::-1]
    dL_da = 2.0 * conv[n - 1:2 * n - 1][::-1]

    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective_wrapper(x, beta):
    """Top-level wrapper for scipy minimize."""
    return lse_score_and_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    try:
        res = minimize(
            lse_objective_wrapper, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
        S_out = a_out.sum()
        if S_out < 1e-6:
            a_out = a0.copy()
        exact_sc = compute_score(a_out)
        return exact_sc, a_out
    except Exception:
        return compute_score(a0), a0


def fienup_flat(a, n_iters=300, clip_power=0.5):
    """
    Fienup-style: flatten the power spectrum |DFT(a)|^2 by reducing high bins.
    Since autocorr = ifft(|DFT|^2), this directly attacks the peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Adaptive threshold: tighten over iterations
        thresh_mult = 1.0 + 0.8 * (1.0 - progress) ** 0.7
        thresh = P_mean * thresh_mult

        above = P > thresh
        if not above.any():
            # Try quantile-based
            thresh_q = np.quantile(P, 0.85 + 0.1 * progress)
            above = P > thresh_q
            if not above.any():
                break
            thresh = thresh_q

        scale = np.ones_like(P)
        # Apply reduction to bins above threshold
        ratio = thresh / P[above]
        scale[above] = ratio ** (clip_power * 0.5)

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def fienup_quantile(a, n_iters=200, top_frac=0.25):
    """Fienup targeting top fraction of power bins."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2

        thresh = np.quantile(P, 1.0 - top_frac)
        above = P > thresh
        if not above.any():
            break

        scale = np.ones_like(P)
        # Strong reduction for top bins
        scale[above] = (thresh / P[above]) ** 0.7
        # Gentle reduction for next tier
        thresh2 = np.quantile(P, 1.0 - top_frac * 2)
        above2 = (P > thresh2) & (~above)
        if above2.any():
            scale[above2] = (thresh2 / P[above2]) ** 0.3

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def peak_subgradient(a, n_iters=5000, top_m=8, step_init=0.03, decay=0.9998):
    """
    Subgradient method targeting top-M peaks of the autocorrelation.
    Uses the exact gradient at the peak locations.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            # Occasionally reset step when making progress
            if it % 500 == 0:
                step = step_init * 0.5

        # Compute subgradient using top-M peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
            weights = np.ones(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
            # Weight by how close to max
            w = c[top_idx] / M_val
            weights = w / w.sum()

        # Gradient at a single peak k: dc_k/da_j = 2*a_{k-j} (if valid)
        # Weighted sum over top peaks
        sg = np.zeros(n)
        for idx, w in zip(top_idx, weights):
            k = int(idx)
            # dc_k/da_j = 2*a_{k-j} for valid j
            # j ranges over [max(0, k-(n-1)), min(k, n-1)]
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            for j in range(j_lo, j_hi + 1):
                sg[j] += 2.0 * w * a[k - j]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Step size: limited by function value gap
        # Theoretical lower bound ~ 2n/(2n-1)
        sc_lb = 2.0 * n / (2.0 * n - 1.0)
        step = min(step, (sc - sc_lb) / (gn * gn + 1e-30))
        step = min(step, 0.06)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def coordinate_descent(a, n_sweeps=3, n_top=40):
    """
    Coordinate descent: for each peak-contributing coordinate,
    try multiplicative perturbations and keep the best.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        # Find top peaks
        n_peaks = min(5, len(c))
        peak_idx = np.argsort(c)[::-1][:n_peaks]

        # Collect all coordinates that contribute to top peaks
        coords = set()
        for k in peak_idx:
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            # Add top-contributing coordinates
            contrib = np.array([a[j] * a[k - j] for j in range(j_lo, j_hi + 1)])
            top_local = np.argsort(contrib)[::-1][:n_top]
            for tl in top_local:
                coords.add(j_lo + int(tl))

        for j in sorted(coords):
            a_old = a[j]
            if a_old < 1e-10:
                continue
            cur_sc = best_sc
            best_val = a_old
            # Try multiplicative factors
            for factor in [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 0.3]:
                if factor == 1.0:
                    continue
                new_val = a_old * factor
                new_val = max(0.0, min(1000.0, new_val))
                if abs(new_val - a_old) < 1e-12:
                    continue
                a[j] = new_val
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old
                    continue
                a_norm = a / S_tmp
                sc = compute_score(a_norm)
                if sc < cur_sc:
                    cur_sc = sc
                    best_val = new_val
                a[j] = a_old
            a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def multiplicative_weight_update(a, n_iters=2000, eta=0.02):
    """
    Exponentiated gradient: a_j <- a_j * exp(-eta * grad_j).
    Naturally maintains non-negativity.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-10).copy()
    S = float(a.sum())
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient at max peak
        k_star = int(np.argmax(c))
        j_lo = max(0, k_star - n + 1)
        j_hi = min(k_star, n - 1)
        sg = np.zeros(n)
        for j in range(j_lo, j_hi + 1):
            sg[j] = 2.0 * a[k_star - j]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        # Multiplicative update
        eta_t = eta * (1.0 - 0.5 * it / n_iters)
        a = a * np.exp(-eta_t * grad)
        a = np.clip(a, 1e-10, 1000.0)
        S = a.sum()
        a = a / S

    return best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1000.0).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.maximum(0.5 + 0.3 * np.cos(4.0 * np.pi * x) * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 100.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = [6962]
    if parent is not None:
        lengths.insert(0, len(parent))
    # Add nearby lengths
    base_n = lengths[0]
    for dn in [-200, -100, 100, 200, 400]:
        candidate = base_n + dn
        if candidate >= 4096 and candidate <= 12000:
            lengths.append(candidate)
    lengths = sorted(set(lengths))[:5]
    print(f"Lengths to try: {lengths}")
    print(f"Parent: {'yes (n=' + str(len(parent)) + ')' if parent is not None else 'no'}")

    # ==================================================================
    # Phase 1: Fienup from diverse inits (20% of time)
    # ==================================================================
    phase1_end = t_start + 0.20 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Fienup (deadline t={phase1_end - t_start:.0f}s) ===")

    for n in lengths[:3]:
        if time.monotonic() > phase1_end:
            break
        print(f"  Processing n={n}")
        inits = build_inits(n, rng, parent)

        for i, a0 in enumerate(inits[:4]):  # Limit inits for speed
            if time.monotonic() > phase1_end:
                break
            a_f = fienup_flat(a0, n_iters=200, clip_power=0.5)
            sc = compute_score(a_f)
            update_best(sc, a_f, f"FIENUP n={n} init={i}")

            if time.monotonic() < phase1_end:
                a_q = fienup_quantile(a_f, n_iters=150, top_frac=0.2)
                sc2 = compute_score(a_q)
                update_best(sc2, a_q, f"FIENUP-q n={n} init={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: LSE + L-BFGS with adaptive beta (35% of time)
    # ==================================================================
    phase2_end = t_start + 0.55 * (total_budget - 30.0)
    print(f"\n=== Phase 2: LSE L-BFGS (deadline t={phase2_end - t_start:.0f}s) ===")

    # Start from best found in Phase 1
    a_work = best_a.copy()

    # Adaptive beta schedule: start low (smooth), increase (sharp)
    beta_schedule = [30, 50, 100, 200, 400, 800, 1500, 3000]
    for beta in beta_schedule:
        if time.monotonic() > phase2_end:
            break
        remaining = phase2_end - time.monotonic()
        maxiter = min(8000, int(remaining * 30))
        maxfun = min(150000, maxiter * 20)
        if maxiter < 100:
            break
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=maxiter, maxfun=maxfun)
        update_best(sc, a_opt, f"LBFGS beta={beta}")
        a_work = a_opt.copy()

    # Also try from parent at its native length
    if parent is not None and len(parent) != len(best_a):
        if time.monotonic() < phase2_end:
            for beta in [100, 500, 2000]:
                if time.monotonic() > phase2_end:
                    break
                sc, a_opt = run_lbfgs(parent, beta, maxiter=5000, maxfun=80000)
                update_best(sc, a_opt, f"LBFGS-parent beta={beta}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Peak subgradient (20% of time)
    # ==================================================================
    phase3_end = t_start + 0.75 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Peak Subgradient (deadline t={phase3_end - t_start:.0f}s) ===")

    for top_m in [5, 8, 12, 20]:
        if time.monotonic() > phase3_end:
            break
        a_sg = peak_subgradient(best_a, n_iters=4000, top_m=top_m, step_init=0.03)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, f"SG top_m={top_m}")

    # Multiplicative weight update
    if time.monotonic() < phase3_end:
        a_mwu = multiplicative_weight_update(best_a, n_iters=3000, eta=0.015)
        sc = compute_score(a_mwu)
        update_best(sc, a_mwu, "MWU")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Coordinate descent + refinement (15% of time)
    # ==================================================================
    phase4_end = t_start + 0.90 * (total_budget - 30.0)
    print(f"\n=== Phase 4: Coord Descent (deadline t={phase4_end - t_start:.0f}s) ===")

    a_cd = coordinate_descent(best_a, n_sweeps=4, n_top=60)
    sc = compute_score(a_cd)
    update_best(sc, a_cd, "COORD")

    # L-BFGS from coordinate descent result
    if time.monotonic() < phase4_end:
        sc, a_opt = run_lbfgs(a_cd, 1000, maxiter=5000, maxfun=80000)
        update_best(sc, a_opt, "CD+LBFGS")

    # Another L-BFGS from best with high beta
    if time.monotonic() < phase4_end:
        sc, a_opt = run_lbfgs(best_a, 5000, maxiter=5000, maxfun=80000)
        update_best(sc, a_opt, "LBFGS-high-beta")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Final polish (5% of time)
    # ==================================================================
    print(f"\n=== Phase 5: Final Polish ===")

    if time.monotonic() < deadline - 10:
        # Fienup from best
        a_ff = fienup_flat(best_a, n_iters=100, clip_power=0.3)
        sc = compute_score(a_ff)
        update_best(sc, a_ff, "FINAL-FIENUP")

        # One more L-BFGS
        if time.monotonic() < deadline - 5:
            sc, a_opt = run_lbfgs(best_a, 10000, maxiter=3000, maxfun=50000)
            update_best(sc, a_opt, "FINAL-LBFGS")

    # ==================================================================
    # Final output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
