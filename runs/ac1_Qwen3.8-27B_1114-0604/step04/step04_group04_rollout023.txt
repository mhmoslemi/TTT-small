```python
"""
Multi-phase optimizer for non-negative sequences minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses: multi-scale init, exact LP (HiGHS) alternating, frequency-domain Newton,
balanced peak reduction, LSE-smoothed L-BFGS, and adaptive local search.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time
import scipy.fft as sfft


def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    Mf = 1 << (M - 1).bit_length()
    A = sfft.rfft(a, Mf)
    return sfft.irfft(A * A, Mf)[:M]


def cross_conv(a, b):
    M = len(a) + len(b) - 1
    Mf = 1 << (M - 1).bit_length()
    A = sfft.rfft(a, Mf)
    B = sfft.rfft(b, Mf)
    return sfft.irfft(A * B, Mf)[:M]


def compute_score(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_smoothed_obj_grad(a, beta):
    n = len(a)
    c = self_conv(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z = z - z.max()
    exp_z = np.exp(z)
    ez_sum = float(exp_z.sum())
    if ez_sum < 1e-300:
        p = np.ones(len(c)) / len(c)
    else:
        p = exp_z / ez_sum
    L_val = c_max + np.log(ez_sum) / beta
    score = 2.0 * n * L_val / (S * S)
    # Gradient of L w.r.t. a: 2 * cross_conv(a_rev, p)[n-1:2n-1]
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2f = 1 << (M2 - 1).bit_length()
    RA = sfft.rfft(a_rev, M2f)
    P = sfft.rfft(p, M2f)
    conv_res = sfft.irfft(RA * P, M2f)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=8000, maxfun=300000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_grad(x):
        return lse_smoothed_obj_grad(x, beta)

    res = minimize(
        obj_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14, 'maxiter': maxiter,
                 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def lp_solve_fista(a, M, n_iters=5000):
    n = len(a)
    x = np.ones(n)
    u = x.copy()
    t_mom = 1.0
    sum_a2 = float(np.dot(a, a))
    if sum_a2 < 1e-15:
        sum_a2 = 1.0 / n

        frac = min(1.0, it / (n_iters * 0.4))
        beta = 10.0 + 40.0 * frac
        L = beta * n * sum_a2 + 1e-14
        step = 1.0 / L
        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum
        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(p) - n + 1]
        grad = a * sw - 1.0
        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new
    return x


def lp_alternating(a, max_outer=30, fista_iters=4000, t_start=0.4, t_min=0.03,
                   deadline=None):
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0

    for outer in range(max_outer):
        if deadline is not None and time.monotonic() > deadline:
            break
        M = float(self_conv(a).max())
        x = lp_solve_fista(a, M, n_iters=fista_iters)
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx
        t = max(t_min, t_start * (1.0 - 0.5 * outer / max(1, max_outer)))
        a_new = (1.0 - t) * a + t * x
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 20:
                break
        a = a_new
    return best_a, best_sc


def lp_solve_highs(a, M):
    n = len(a)
    c_obj = -np.ones(n)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.ones(2 * n - 1) * M
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if res.success:
        return res.x
    return None


def balanced_peak_reduction(a, n_iters=500, K=5, eta_start=0.3, eta_end=0.01):
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        conv = self_conv(a)
        M = float(conv.max())
        sc = 2.0 * n * M / (float(a.sum()) ** 2)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        top_k = np.argsort(conv)[::-1][:K]
        threshold = conv[top_k[-1]]

        pressure = np.zeros(n)
        for k in top_k:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            for j in range(j_start, j_end + 1):
                pressure[j] += a[j] * a[k - j]

        eta = eta_start * (eta_end / eta_start) ** (it / max(1, n_iters - 1))
        p_max = pressure.max()
        if p_max < 1e-15:
            break
        pressure_norm = pressure / p_max
        a_new = a * (1.0 - eta * pressure_norm)
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

    return best_a


def freq_newton(a, n_iters=200, alpha=0.5):
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    M = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = sfft.fft(a_pad)
        A2 = A * A
        conv_full = sfft.ifft(A2).real[:2 * n - 1]
        conv_mean = float(np.mean(conv_full))
        conv_max = float(np.max(conv_full))
        if conv_max - conv_mean < 1e-14:
            break
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv_full - thresh, 0.0)
        if excess.max() < 1e-14:
            break
        conv_mod = conv_full - 0.9 * excess
        err = conv_mod - conv_full
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = sfft.fft(err_pad)
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-10] = 0.0
        damp = 0.3 + 0.7 * min(1.0, it / 20.0)
        A_new = A + damp * d
        a_new = sfft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    return best_a


def subgradient_polish(a, total_iters=5000):
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(total_iters):
        c = self_conv(a)
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (float(a.sum()) ** 2)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        sg[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
        S = float(a.sum())
        S2 = S * S
        S3 = S2 * S
        grad = 2.0 * n * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.03)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
    return best_a


def multiplicative_update(a, n_iters=300, K=10, eta=0.15):
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        conv = self_conv(a)
        M = float(conv.max())
        sc = 2.0 * n * M / (float(a.sum()) ** 2)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        top_k = np.argsort(conv)[::-1][:K]
        threshold = conv[top_k[-1]]
        log_update = np.zeros(n)
        for k in top_k:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            contrib = a[j_start:j_end + 1] * a[k - j_end:k - j_start + 1][::-1]
            log_update[j_start:j_end + 1] -= contrib
        eta_it = eta * (0.5 ** (it / max(1, n_iters // 3)))
        a *= np.exp(-eta_it * log_update / max(1e-15, np.max(np.abs(log_update))))
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S
    return best_a


def upsample(a, n_new):
    n_old = len(a)
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, a)


def propose_candidate(seed=42, budget_s=2400):
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    deadline = t0 + max(30.0, float(budget_s)) - 15.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(2048) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            el = time.monotonic() - t0
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={el:.0f}s)")
            return True
        return False

    print(f"Parent: {parent is not None}, len={0 if parent is None else len(parent)}")

    # Phase 1: Multi-scale initialization (10% time)
    p1_dl = t0 + 0.10 * (float(budget_s) - 15.0)
    print("\n--- Phase 1: Multi-scale Init ---")

    n_small = 512
    n_mid = 2048
    n_target = len(parent) if parent is not None else 6962

    # Small-scale exploration
    for n_try in [n_small]:
        inits = [
            np.ones(n_try) / n_try,
            np.clip(0.5 + 0.3 * np.cos(2 * np.pi * np.arange(n_try) / n_try), 0.01, 1) / n_try,
            np.clip(0.5 + 0.4 * rng.standard_normal(n_try), 0.01, 1) / n_try,
        ]
        for i, a0 in enumerate(inits):
            if time.monotonic() > p1_dl:
                break
            a_er = freq_newton(a0, n_iters=80, alpha=0.6)
            sc = compute_score(a_er)
            report(sc, a_er, f"small n={n_try} i={i}")

    # Mid-scale
    a_mid = upsample(best_a, n_mid) if len(best_a) != n_mid else best_a.copy()
    S = a_mid.sum()
    if S > 0:
        a_mid = a_mid / S
    a_mid = freq_newton(a_mid, n_iters=100, alpha=0.5)
    sc = compute_score(a_mid)
    report(sc, a_mid, f"mid n={n_mid}")

    # Upsample to target
    a_work = upsample(best_a, n_target)
    S = a_work.sum()
    if S > 0:
        a_work = a_work / S
    if parent is not None and len(parent) == n_target:
        a_work = parent / parent.sum()
        sc = compute_score(a_work)
        report(sc, a_work, "parent")

    print(f"After Phase 1: best={best_score:.8f}")

    # Phase 2: LP-based alternating (40% time)
    p2_dl = t0 + 0.50 * (float(budget_s) - 15.0)
    print("\n--- Phase 2: LP Alternating ---")

    n_work = n_target
    a_work = best_a.copy() if len(best_a) == n_work else upsample(best_a, n_work)
    S = a_work.sum()
    if S > 0:
        a_work = a_work / S

    # Try exact LP for smaller n, FISTA for larger
    if n_work <= 3000:
        print(f"  Using HiGHS LP for n={n_work}")
        for outer in range(15):
            if time.monotonic() > p2_dl:
                break
            conv = self_conv(a_work)
            M = float(conv.max())
            x = lp_solve_highs(a_work, M)
            if x is None:
                x = lp_solve_fista(a_work, M, n_iters=3000)
            Sx = float(x.sum())
            if Sx < 1e-12:
                break
            x = x / Sx
            t_mix = max(0.05, 0.4 * (1.0 - outer / 20.0))
            a_new = (1.0 - t_mix) * a_work + t_mix * x
            a_new = np.maximum(a_new, 0.0)
            S_new = float(a_new.sum())
            if S_new < 1e-12:
                break
            a_new = a_new / S_new
            sc = compute_score(a_new)
            report(sc, a_new, f"LP-hiGHS r{outer}")
            a_work = a_new
    else:
        print(f"  Using FISTA LP for n={n_work}")
        for round_i in range(4):
            if time.monotonic() > p2_dl - 30:
                break
            fista_it = [3000, 4000, 5000, 6000][round_i]
            t_s = [0.4, 0.3, 0.2, 0.1][round_i]
            t_m = [0.08, 0.05, 0.03, 0.02][round_i]
            max_out = max(5, int((p2_dl - time.monotonic() - 10) / max(5, fista_it * n_work / 300000.0)))
            a_work, _ = lp_alternating(a_work, max_outer=max_out, fista_iters=fista_it,
                                        t_start=t_s, t_min=t_m, deadline=p2_dl - 5)
            sc = compute_score(a_work)
            report(sc, a_work, f"LP-fista r{round_i}")

    print(f"After Phase 2: best={best_score:.8f}")

    # Phase 3: Frequency-domain Newton + Balanced peaks (20% time)
    p3_dl = t0 + 0.70 * (float(budget_s) - 15.0)
    print("\n--- Phase 3: Freq Newton + Balanced Peaks ---")

    a_work = best_a.copy()
    S = a_work.sum()
    if S > 0:
        a_work = a_work / S

    for alpha in [0.4, 0.5, 0.6, 0.7]:
        if time.monotonic() > p3_dl:
            break
        a_fn = freq_newton(a_work, n_iters=150, alpha=alpha)
        sc = compute_score(a_fn)
        report(sc, a_fn, f"freqN a={alpha}")
        a_work = a_fn

    for K in [3, 5, 10, 20]:
        if time.monotonic() > p3_dl:
            break
        a_bp = balanced_peak_reduction(a_work, n_iters=300, K=K, eta_start=0.2, eta_end=0.01)
        sc = compute_score(a_bp)
        report(sc, a_bp, f"balPeak K={K}")
        a_work = a_bp

    # Multiplicative updates
    a_mu = multiplicative_update(a_work, n_iters=200, K=10, eta=0.1)
    sc = compute_score(a_mu)
    report(sc, a_mu, "multUpd")

    print(f"After Phase 3: best={best_score:.8f}")

    # Phase 4: LSE-LBFGS refinement (15% time)
    p4_dl = t0 + 0.85 * (float(budget_s) - 15.0)
    print("\n--- Phase 4: LSE-LBFGS ---")

    a_work = best_a.copy()
    S = a_work.sum()
    if S > 0:
        a_work = a_work / S

    for beta in [50, 100, 300, 800, 2000, 5000]:
        if time.monotonic() > p4_dl:
            break
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=5000, maxfun=150000)
        report(sc, a_opt, f"LBFGS b={beta}")
        if sc < best_score:
            a_work = a_opt.copy()
            S = a_work.sum()
            if S > 0:
                a_work = a_work / S

    print(f"After Phase 4: best={best_score:.8f}")

    # Phase 5: Final polish (15% time)
    print("\n--- Phase 5: Final Polish ---")

    a_work = best_a.copy()
    S = a_work.sum()
    if S > 0:
        a_work = a_work / S

    # Subgradient
    if time.monotonic() < deadline - 120:
        a_sg = subgradient_polish(a_work, total_iters=5000)
        sc = compute_score(a_sg)
        report(sc, a_sg, "subgrad")

    # Another LP round with very small step
    if time.monotonic() < deadline - 100:
        a_lp, _ = lp_alternating(a_work, max_outer=15, fista_iters=5000,
                                  t_start=0.1, t_min=0.01, deadline=deadline - 60)
        sc = compute_score(a_lp)
        report(sc, a_lp, "finalLP")

    # Final LBFGS with very high beta
    if time.monotonic() < deadline - 40:
        a_work = best_a.copy()
        S = a_work.sum()
        if S > 0:
            a_work = a_work / S
        for beta_final in [5000, 10000]:
            if time.monotonic() > deadline - 20:
                break
            sc, a_opt = run_lbfgs(a_work, beta_final, maxiter=4000, maxfun=100000)
            report(sc, a_opt, f"finalLBFGS b={beta_final}")

    # Try a slightly different n
    for n_alt in [n_target + 100, n_target - 200, 8192]:
        if time.monotonic() > deadline - 60:
            break
        if n_alt < 100 or n_alt == n_target:
            continue
        a_alt = upsample(best_a, n_alt)
        S = a_alt.sum()
        if S > 0:
            a_alt = a_alt / S
        a_alt = freq_newton(a_alt, n_iters=50, alpha=0.5)
        sc = compute_score(a_alt)
        report(sc, a_alt, f"altN n={n_alt}")
        if sc < best_score + 0.001:
            sc2, a_opt = run_lbfgs(a_alt, 500, maxiter=3000, maxfun=80000)
            report(sc2, a_opt, f"altN+LBFGS n={n_alt}")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t0
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
```
