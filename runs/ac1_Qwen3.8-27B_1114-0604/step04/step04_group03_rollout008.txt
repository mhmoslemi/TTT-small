"""
Advanced optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses: spectral shaping with adaptive projection, LSE-smoothed L-BFGS continuation,
simulated annealing with peak-targeted structured moves, active set identification,
and multi-scale exploration. Key innovations: continuation method with adaptive beta,
spectral water-filling, and simulated annealing escape mechanism.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, M)
    return np.fft.irfft(A * A, M)[:2 * n - 1]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and its exact gradient.
    Returns (score, grad) for the scale-invariant objective.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum()) + scaled.max() / beta
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # dL/da[i] = 2 * sum_j p[i+j] * a[j]
    rev_a = a[::-1]
    M2 = 1 << (3 * n - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2)
    P = np.fft.rfft(p, M2)
    r = np.fft.irfft(RA * P, M2)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        s, g = lse_score_grad(x, beta)
        return s, g

    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 60}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def spectral_shaping(a, n_iters=500, alpha0=0.5, alpha1=0.95):
    """
    Iterative spectral water-filling to flatten autocorrelation.
    Works in the frequency domain: modifies |A|^2 to reduce peaks in r=ifft(|A|^2),
    then projects back to non-negative time-domain signal.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    M = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        t = it / max(1, n_iters - 1)
        alpha = alpha0 + (alpha1 - alpha0) * t

        a_pad = np.zeros(M, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        v = A * A
        conv = np.fft.irfft(v, M)[:2 * n - 1]

        conv_max = float(conv.max())
        conv_mean = float(conv.mean())
        if conv_max - conv_mean < 1e-18:
            break

        # Compute excess above target level
        target = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - target, 0.0)
        if excess.max() < 1e-18:
            break

        # Modify spectrum to reduce excess
        # Newton step: A_new = A + delta, where |A+delta|^2 - |A|^2 = -excess_spec
        # => 2*Re(A*delta) + |delta|^2 = -excess_spec
        # First order: delta = -excess_spec / (2*A)
        err_pad = np.zeros(M, dtype=np.float64)
        err_pad[:2 * n - 1] = -excess
        err_spec = np.fft.rfft(err_pad, M)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A)
        d[np.abs(A) < 1e-12] = 0.0

        # Adaptive damping
        damp = 0.3 + 0.5 * t
        A_new = A + damp * d

        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def simulated_annealing(a, n_iters=5000, T0=0.05, T1=0.001):
    """
    Simulated annealing with peak-targeted structured moves.
    At each step, target the current peak and generate a move that specifically
    reduces it, with Metropolis acceptance.
    """
    n = len(a)

    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    cur_sc = best_sc

    c = fast_conv(a)
    k_star = int(np.argmax(c))

    rng_local = np.random.default_rng(12345)

    for it in range(n_iters):
        T = T0 * (T1 / T0) ** (it / max(1, n_iters - 1))

        # Get current peak
        if it % 50 == 0 or it == 0:
            c = fast_conv(a)
            k_star = int(np.argmax(c))

        # Generate peak-targeted move
        move_type = rng_local.integers(0, 4)

        a_new = a.copy()

        if move_type == 0:
            # Reduce top contributors to peak, increase bottom
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end > j_start:
                seg = a_new[j_start:j_end + 1].copy()
                contrib = seg[::-1].copy()
                n_top = max(3, n // 40)
                top_idx = np.argsort(contrib)[::-1][:n_top]
                bot_idx = np.argsort(contrib)[:n_top]
                red_total = 0.0
                for ti in top_idx:
                    gi = j_start + ti
                    red = a_new[gi] * (0.02 + 0.08 * rng_local.random())
                    a_new[gi] = max(0.0, a_new[gi] - red)
                    red_total += red
                for bi in bot_idx:
                    a_new[j_start + bi] += red_total / len(bot_idx)

        elif move_type == 1:
            # Swap mass: move mass from high-contribution to low-contribution
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end > j_start:
                seg = a_new[j_start:j_end + 1].copy()
                contrib = seg[::-1].copy()
                idx = rng_local.choice(len(contrib), size=min(10, len(contrib)), replace=False)
                idx_sorted = idx[np.argsort(contrib[idx])]
                i_from = j_start + idx_sorted[-1]
                i_to = j_start + idx_sorted[0]
                amt = a_new[i_from] * (0.05 + 0.15 * rng_local.random())
                a_new[i_from] = max(0.0, a_new[i_from] - amt)
                a_new[i_to] += amt

        elif move_type == 2:

        S_new = float(a_new.sum())
        if S_new < 1e-12:
            continue
        a_new = a_new / S_new
        sc_new = compute_score(a_new)

        # Metropolis acceptance

            a = a_new
            cur_sc = sc_new

def active_set_refine(a, n_sweeps=3):

    """

    if S < 1e-12:
        return a

    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        # Identify near-zero components
        threshold = np.percentile(a[a > 1e-10], 2) if np.sum(a > 1e-10) > 10 else 1e-10
        mask = a > max(threshold, 1e-8)
        n_free = int(mask.sum())
        if n_free < n * 0.3:
            break  # Too many zeros, skip

        a_freereduced = a.copy()
        a_freereduced[~mask] = 0.0
        S_free = float(a_freereduced.sum())
        if S_free < 1e-12:
            break

        # Optimize only free variables
        free_idx = np.where(mask)[0]
        a_free = a_freereduced[free_idx].copy()

        def obj_free(x):
            x = np.maximum(x, 0.0)
            a_full = np.zeros(n)
            a_full[free_idx] = x
            return compute_score(a_full)

        # Use simple gradient stepping on free vars

        bounds_free = [(0.0, 1000.0)] * n_free

        def obj_full(x):
            a_full = np.zeros(n)

            if S < 1e-12:
                return 1e10
            c = fast_conv(a_full)
            return 2.0 * n * float(np.max(c)) / (S * S)

        res = minimize(obj_full, a_free, method='L-BFGS-B', bounds=bounds_free,
                       options={'maxiter': 500, 'maxfun': 20000})

        a_new = np.zeros(n)

            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n in [4096, 6962, 8192, 10240, 12288, 16384]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))[:5]

    print(f"Lengths to try: {lengths}")
    print(f"Parent: {parent is not None}, len={0 if parent is None else len(parent)}")

    # ==================================================================
    # Phase 1: Spectral shaping from diverse inits (25% time)
    # ==================================================================
    phase1_deadline = t_start + 0.25 * (float(budget_s) - 10.0)
    print(f"\n--- Phase 1: Spectral Shaping ---")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"  n={n}")
        inits = build_initializations(n, rng, parent)
        remaining = phase1_deadline - time.monotonic()
        max_inits = max(2, int(remaining / 12.0))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a_ss = spectral_shaping(a0, n_iters=300, alpha0=0.4, alpha1=0.95)
            sc = compute_score(a_ss)
            report(sc, a_ss, f"SS n={n} i={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: L-BFGS continuation (35% time)
    # ==================================================================
    phase2_deadline = t_start + 0.60 * (float(budget_s) - 10.0)
    print(f"\n--- Phase 2: L-BFGS Continuation ---")

    n_best = len(best_a)
    # Continuation: beta from small to large
    beta_schedule = [20, 50, 100, 200, 500, 1000, 2000, 5000]

    for beta in beta_schedule:
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        maxfun = int(min(300000, remaining * 100))
        maxiter = int(min(15000, remaining * 50))
        if maxfun < 1000:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"LBFGS b={beta}")

    # Try from other good starting points at different lengths
    if time.monotonic() < phase2_deadline:
        for n in lengths[:3]:
            if time.monotonic() > phase2_deadline:
                break
            if len(best_a) != n:
                a_interp = np.interp(
                    np.linspace(0, 1, n),
                    np.linspace(0, 1, len(best_a)),
                    best_a
                )
            else:
                a_interp = best_a.copy()
            for beta in [200, 1000]:
                if time.monotonic() > phase2_deadline:
                    break
                sc, a_opt = run_lbfgs(a_interp, beta, maxiter=5000, maxfun=100000)
                report(sc, a_opt, f"LBFGS alt n={n} b={beta}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Simulated annealing + local restarts (25% time)
    # ==================================================================
    phase3_deadline = t_start + 0.85 * (float(budget_s) - 10.0)
    print(f"\n--- Phase 3: Simulated Annealing + Local Search ---")

    n_best = len(best_a)
    n_trials = 15
    for trial in range(n_trials):
        if time.monotonic() > phase3_deadline:
            break
        strategy = trial % 5
        if strategy == 0:
            # SA from best
            a_sa = simulated_annealing(best_a, n_iters=3000, T0=0.03, T1=0.0005)
            sc = compute_score(a_sa)
            report(sc, a_sa, f"SA#{trial}")
        elif strategy == 1:
            # Perturb + L-BFGS
            scale = 0.02 + 0.04 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            sc, a_opt = run_lbfgs(a_pert, 500, maxiter=5000, maxfun=80000)
            report(sc, a_opt, f"pert+LBFGS#{trial}")
        elif strategy == 2:
            # Peak-targeted + L-BFGS
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            a_pert = best_a.copy()
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            seg = a_pert[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            top_count = max(5, n_best // 30)
            top_local = np.argsort(contrib)[::-1][:top_count]
            total_red = 0.0
            for li in top_local:
                gi = j_start + li
                red = a_pert[gi] * 0.05
                a_pert[gi] = max(a_pert[gi] - red, 0.0)
                total_red += red
            if total_red > 0:
                bot_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
                per = total_red / len(bot_local)
                for li in bot_local:
                    a_pert[j_start + li] += per
            S = a_pert.sum()
            if S > 1e-12:
                a_pert = a_pert / S
            sc, a_opt = run_lbfgs(a_pert, 1000, maxiter=5000, maxfun=80000)
            report(sc, a_opt, f"peak+LBFGS#{trial}")
        elif strategy == 3:
            # Spectral shaping from perturbed best
            a_pert = best_a * (1.0 + 0.03 * rng.standard_normal(n_best))
            a_pert = np.maximum(a_pert, 0.0)
            a_ss = spectral_shaping(a_pert, n_iters=200, alpha0=0.5, alpha1=0.9)
            sc = compute_score(a_ss)
            report(sc, a_ss, f"SS_pert#{trial}")
        else:
            # Block rescale + L-BFGS
            a_pert = best_a.copy()
            n_blocks = 3 + int(rng.integers(0, 8))
            block_size = max(1, n_best // n_blocks)
            for b in range(min(n_blocks, n_best // block_size)):
                s = b * block_size
                e = min(s + block_size, n_best)
                factor = 0.7 + 0.6 * rng.random()
                a_pert[s:e] *= factor
            S = a_pert.sum()
            if S > 1e-12:
                a_pert = a_pert / S
            sc, a_opt = run_lbfgs(a_pert, 500, maxiter=4000, maxfun=60000)
            report(sc, a_opt, f"block+LBFGS#{trial}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Active set + final polish (15% time)
    # ==================================================================
    print(f"\n--- Phase 4: Final Polish ---")
    remaining = deadline - time.monotonic()

    if remaining > 60:
        # Active set refinement
        a_as = active_set_refine(best_a, n_sweeps=3)
        sc = compute_score(a_as)
        report(sc, a_as, "active_set")

    if remaining > 40:
        # High-beta L-BFGS polish
        sc, a_opt = run_lbfgs(best_a, 10000, maxiter=10000, maxfun=200000)
        report(sc, a_opt, "LBFGS b=10000")

    if remaining > 30:
        # Final spectral shaping
        a_ss = spectral_shaping(best_a, n_iters=200, alpha0=0.7, alpha1=0.99)
        sc = compute_score(a_ss)
        report(sc, a_ss, "final_SS")

    if remaining > 20 and best_score > 1.505:
        # One more round
        sc, a_opt = run_lbfgs(best_a, 5000, maxiter=10000, maxfun=150000)
        report(sc, a_opt, "final LBFGS b=5000")

    if remaining > 15:
        # Very short SA
        a_sa = simulated_annealing(best_a, n_iters=2000, T0=0.01, T1=0.0001)
        sc = compute_score(a_sa)
        report(sc, a_sa, "final_SA")

    if remaining > 10:
        sc, a_opt = run_lbfgs(best_a, 20000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "LBFGS b=20000")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
