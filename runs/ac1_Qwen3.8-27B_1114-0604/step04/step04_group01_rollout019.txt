        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / se

    L = dmax + np.log(se) / beta
    LDS = float(np.dot(w, d))
    h = _corr_pair(a, w)

    score = 2.0 * L + PEN_SUM * (s - 1.0) ** 2
    grad = (4.0 * float(n) * h) / (s * s)
    grad -= (4.0 * LDS) / s
    grad += 2.0 * PEN_SUM * (s - 1.0)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e8, 1e8)
    return float(score), grad


def _run_lbfgs(a0, beta, deadline, maxfun):
    a = _norm(a0)
    n = len(a)
    t_rem = float(deadline - time.monotonic())
    if t_rem < 5.0:
        return _exact_score(a), a

    N = _npow2(2 * n - 1)
    est_eval_time = 0.008 * float(n) / 4096.0 * (np.log2(max(16, N)) / 12.0)
    cap = int((0.8 * t_rem) / max(est_eval_time, 1e-5))
    maxfun = int(min(maxfun, max(64, cap)))
    maxfun = max(64, min(maxfun, 5000))
    maxiter = max(30, maxfun // 2)

    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-16,
        "gtol": 1e-13,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 40,
    }

    def obj_wrapper(x):
        return _lse_obj_grad(x, beta)

    try:
        res = minimize(
            obj_wrapper, a,
            method="L-BFGS-B", jac=True,
            bounds=bounds, options=opts,
        )
    except Exception:
        return _exact_score(a), a

    a_out = _norm(res.x) if hasattr(res, 'x') and res.x is not None else a
    sc = _exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = _exact_score(a)
    return float(sc), a_out


def _active_weights(d, k):
    m = d.size
    if m <= 0:
        return np.array([], dtype=np.float64)
    if k <= 0 or k >= m:
        return np.ones(m) / m
    idx = np.argpartition(d, m - k)[m - k:]
    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0
    return w / k


def _subgrad_step(a, deadline, iters=30):
    a = _norm(a)
    best_a = a.copy()
    best_sc = _exact_score(a)

    for _ in range(iters):
        if time.monotonic() > deadline:
            break
        d, s = _d_and_s(a)
        if s < 1e-12 or d.size == 0:
            break
        dmax = float(np.max(d))
        if not np.isfinite(dmax) or dmax <= 0:
            break

        best_step = None
        best_step_sc = best_sc

        for k in (1, 2, 4, 8, 16):
            w = _active_weights(d, k)
            h = _corr_pair(a, w)
            hmax = float(np.max(h))
            if hmax < 1e-12:
                continue
            base = 1.0 / hmax
            for mult in (0.01, 0.03, 0.1, 0.3, 0.8, 2.0):
                if time.monotonic() > deadline:
                    break
                eta = mult * base
                q = a * np.exp(-eta * h)
                q = _norm(q)
                sc = _exact_score(q)
                if not np.isfinite(sc) or sc > best_sc * 2:
                    continue
                if sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                if sc < best_step_sc:
                    best_step_sc = sc
                    best_step = q.copy()

        if best_step is None:
            break
        a = best_step

    return float(best_sc), best_a


def _smooth_noise(n, rng, strength):
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 100))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = n if n % 2 == 1 else max(3, n - 1)
    if k >= 3:
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        noise = np.convolve(noise, kern, mode="same")
    noise -= noise.mean()
    return noise * strength


def _perturb(a, rng, strength=None):
    p = _norm(a)
    n = len(p)
    if strength is None:
        strength = float(rng.uniform(0.1, 0.6))
    mode = int(rng.integers(0, 7))

    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.4))
        q = (1.0 - t) * p + t * (np.ones(n) / n)
    elif mode == 2:
        q = p[::-1].copy()
        q *= np.exp(0.08 * rng.standard_normal(n))
    elif mode == 3:
        d, s = _d_and_s(p)
        w = _active_weights(d, 4)
        h = _corr_pair(p, w)
        hmax = float(np.max(h))
        if hmax < 1e-12:
            q = p.copy()
        else:
            q = p * np.exp(-strength * h / hmax)
    elif mode == 4:
        logp = np.log(p + 1e-300)
        logp += _smooth_noise(n, rng, strength * 0.4)
        q = np.exp(logp - logp.max())
    elif mode == 5:
        k = int(rng.integers(5, 25))
        if k % 2 == 0:
            k += 1
        if k > n:
            k = n if n % 2 == 1 else max(5, n - 1)
        if k < 3:
            k = 3
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= kern.sum()
        q = np.convolve(p, kern, mode="same")
    else:
        # Block perturbation
        nb = int(rng.integers(4, 20))
        per = max(1, n // nb)
        blocks = 1.0 + strength * rng.standard_normal(nb)
        blocks = np.maximum(blocks, 0.1)
        q = np.repeat(blocks, per)
        if q.size < n:
            q = np.concatenate([q, np.repeat(blocks, n - q.size)])
        q = p * q[:n]

    return _norm(q)


def _gen_inits(n, rng, parent, cap=30):
    inits = []

    def add(arr):
        if len(inits) >= cap:
            return
        arr = np.asarray(arr, dtype=np.float64).ravel()
        if arr.size == 0:
            return
        arr = np.nan_to_num(arr, nan=0.0, posinf=1e6, neginf=0.0)
        mn = float(np.min(arr))
        if mn < 0:
            arr = arr - mn + 0.05
        inits.append(_norm(arr))

    if parent is not None and parent.size > 0:
        pr = _resample(parent, n)
        add(pr)
        add(pr[::-1])

    x = np.linspace(0.0, 1.0, n)

    # Uniform
    add(np.ones(n))

    # Linear / anti-linear
    for alpha in (0.2, 0.5, 0.8, 1.0):
        add(1.0 + alpha * (2.0 * x - 1.0))
        add(1.0 - alpha * (2.0 * x - 1.0))

    # Raised cosine
    for width in (0.3, 0.5, 0.7, 1.0):
        add(0.5 * (1.0 + np.cos(2.0 * np.pi * x / width)))

    # Gaussian bumps
    for sigma in (0.05, 0.1, 0.2, 0.3):
        add(np.exp(-((x - 0.5) ** 2) / (2.0 * sigma ** 2)))
        add(np.exp(-(x ** 2) / (2.0 * sigma ** 2)) + np.exp(-((x - 1.0) ** 2) / (2.0 * sigma ** 2)))

    # Chirp-based (real part + offset)
    for f0 in (0.1, 0.3, 0.5):
        t = np.arange(n) / float(n)
        chirp = np.cos(2.0 * np.pi * (f0 * t ** 2 / 2.0 + 0.1 * t))
        add(1.0 + 0.5 * chirp)
        add(1.0 + 0.8 * chirp)

    # Random Fourier
    Nf = _npow2(n)
    n_fourier = min(8, max(3, n // 1024 + 3))
    for _ in range(n_fourier):
        K = int(rng.integers(2, 8))
        vals = np.ones(n)
        for _ in range(K):
            f = int(rng.integers(1, max(2, n // 4)))
            amp = float(rng.uniform(0.05, 0.4))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2.0 * np.pi * f * x + ph)
        add(vals)

    # Low-frequency noise
    for _ in range(4):
        spec = np.zeros(Nf // 2 + 1, dtype=np.complex128)
        n_low = max(2, Nf // 100)
        for j in range(1, n_low):
            spec[j] = (rng.standard_normal() + 1j * rng.standard_normal()) * (1.0 / (j ** 0.5))
        sig = np.fft.irfft(spec * (Nf // 2), Nf)[:n]
        sig -= sig.mean()
        add(1.0 + 0.5 * sig)

    # Sparse / blocky
    for frac in (0.3, 0.5, 0.7):
        mask = rng.random(n) < frac
        vals = np.where(mask, 0.3 + 0.7 * rng.random(n), 0.05)
        add(vals)

    # Block structure
    for nb in (4, 8, 16, 32, 64):
        if nb > n:
            nb = n
        bv = 0.2 + 0.8 * rng.random(nb)
        per = max(1, n // nb)
        b = np.repeat(bv, per)
        if b.size < n:
            b = np.concatenate([b, np.repeat(bv, n - b.size)])
        add(b[:n])

    # Pure random
    for _ in range(3):
        add(0.1 + 0.9 * rng.random(n))

    return inits


def _optimize_trajectory(a_start, deadline, rng, label, budget_frac):
    a = _norm(a_start)
    best_a = a.copy()
    best_sc = _exact_score(a)
    n = len(a)
    t0 = time.monotonic()
    total = max(20.0, min(budget_frac, float(deadline - t0 - 10.0)))

    print("  [{}] start: score={:.8f} n={}".format(label, best_sc, n))

    # Beta schedule: progressive increase
    stages = [
        (50.0, 0.06),
        (100.0, 0.08),
        (200.0, 0.10),
        (400.0, 0.10),
        (800.0, 0.12),
        (1500.0, 0.12),
        (3000.0, 0.12),
        (6000.0, 0.12),
        (12000.0, 0.10),
        (25000.0, 0.08),
    ]

    cum = 0.0
    for beta, frac in stages:
        cum += frac
        sub_dead = t0 + cum * total
        sub_dead = min(float(sub_dead), float(deadline - 8.0))

        while time.monotonic() < sub_dead - 3.0:
            rem = float(sub_dead - time.monotonic())
            if rem < 8.0:
                break
            N = _npow2(2 * n - 1)
            est = 0.008 * float(n) / 4096.0 * (np.log2(max(16, N)) / 12.0)
            mf = int((0.7 * rem) / max(est, 1e-5))
            mf = max(64, min(mf, 3000))

            sc, a2 = _run_lbfgs(a, beta, deadline, mf)
            if sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("  [{}] new best {:.8f} beta={:.0f} t={:.0f}s".format(
                    label, best_sc, beta, time.monotonic() - t0))
            a = _norm(a2)

    # Final high-beta push
    if time.monotonic() < deadline - 15.0:
        for beta_final in (40000.0, 80000.0):
            if time.monotonic() > deadline - 12.0:
                break
            rem = float(deadline - time.monotonic() - 5.0)
            if rem < 15.0:
                break
            N = _npow2(2 * n - 1)
            est = 0.008 * float(n) / 4096.0 * (np.log2(max(16, N)) / 12.0)
            mf = int((0.7 * rem) / max(est, 1e-5))
            mf = max(64, min(mf, 2000))
            sc, a2 = _run_lbfgs(a, beta_final, deadline, mf)
            if sc < best_sc:
                best_sc = sc
                best_a = a2.copy()
                print("  [{}] final beta {:.0f}: {:.8f}".format(label, beta_final, best_sc))

    return float(best_sc), best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 20.0
    rng = np.random.default_rng(seed)

    # Get parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and p.sum() > 0.01:
                parent = p
    except Exception:
        parent = None

    print("Parent: len={} sum={:.4f}".format(
        parent.size if parent is not None else 0,
        parent.sum() if parent is not None else 0))

    # Determine lengths to try
    lengths = []
    if parent is not None:
        pl = int(parent.size)
        if pl <= 12288:
            lengths.append(pl)
        for cand in (4096, 8192, 16384):
            if cand not in lengths and 1024 <= cand <= 32768:
                lengths.append(cand)
    else:
        lengths = [4096, 8192, 16384]

    # Deduplicate and sort
    seen = set()
    unique_lengths = []
    for L in lengths:
        L = int(L)
        if 1024 <= L <= 32768 and L not in seen:
            seen.add(L)
            unique_lengths.append(L)
    lengths = unique_lengths[:4]  # Try at most 4 lengths
    print("Lengths: {}".format(lengths))

    # Phase 1: Generate initializations and pick best starts
    phase1_dead = t_start + 0.08 * (deadline - t_start)
    all_cands = []  # (score, seq, length)

    for n in lengths:
        inits = _gen_inits(n, rng, parent, cap=20)
        for arr in inits:
            sc = _exact_score(arr)
            if np.isfinite(sc):
                all_cands.append((sc, arr, n))

    if not all_cands:
        n0 = lengths[0]
        u = np.ones(n0) / n0
        all_cands.append((_exact_score(u), u, n0))

    all_cands.sort(key=lambda x: x[0])
    print("Phase 1: {} candidates, best score={:.8f}".format(len(all_cands), all_cands[0][0]))

    # Pick up to 3 diverse starts (different lengths preferred)
    starts = []
    used_lens = set()
    for sc, seq, n in all_cands:
        if n not in used_lens and len(starts) < 3:
            starts.append(seq.copy())
            used_lens.add(n)
    while len(starts) < min(3, len(lengths)):
        starts.append(all_cands[0][1].copy())
    starts = starts[:3]

    best_seq = starts[0].copy()
    best_score = _exact_score(best_seq)
    for s in starts[1:]:
        sc = _exact_score(s)
        if sc < best_score:
            best_score = sc
            best_seq = s.copy()

    print("Initial best: {:.8f} len={}".format(best_score, len(best_seq)))

    # Phase 2: LSE optimization on each start
    phase2_dead = t_start + 0.65 * (deadline - t_start)
    n_starts = len(starts)
    fracs = [0.5, 0.3, 0.2][:n_starts] if n_starts > 1 else [1.0]

    cum_f = 0.0
    for i, st in enumerate(starts):
        cum_f += fracs[i]
        sub_dead = t_start + cum_f * (phase2_dead - t_start)
        sub_dead = min(float(sub_dead), float(deadline - 60.0))

        if time.monotonic() > sub_dead - 10.0:
            continue

        # Use current best as seed for later starts
        if i > 0 and np.isfinite(best_score) and best_score < 2.0:
            base = _norm(_resample(best_seq, len(st)))
            st = _perturb(base, rng, strength=0.2)

        sc, seq = _optimize_trajectory(st, deadline, rng, "traj{}".format(i + 1), fracs[i])
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("After traj{}: best={:.8f}".format(i + 1, best_score))

    print("After phase 2: best={:.8f} len={}".format(best_score, len(best_seq)))

    # Phase 3: Subgradient polish
    phase3_dead = t_start + 0.82 * (deadline - t_start)
    if time.monotonic() < phase3_dead - 10.0:
        sc, seq = _subgrad_step(best_seq, deadline, iters=40)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("After subgrad: {:.8f}".format(best_score))

    # Phase 4: Perturbation restarts (simulated annealing style)
    phase4_dead = t_start + 0.93 * (deadline - t_start)
    n_perturb = 0
    while time.monotonic() < phase4_dead - 30.0 and n_perturb < 8:
        left = float(phase4_dead - 30.0 - time.monotonic())
        if left < 60.0:
            break
        chunk_dead = time.monotonic() + left / float(max(1, 8 - n_perturb) - n_perturb if n_perturb < 8 else 1)
        chunk_dead = min(chunk_dead, phase4_dead - 30.0)

        # Annealing: decrease perturbation strength over time
        anneal_frac = float(n_perturb) / 8.0
        strength = 0.4 * (1.0 - anneal_frac) + 0.1

        p = _perturb(best_seq, rng, strength=strength)
        sc, seq = _optimize_trajectory(p, deadline, rng, "pert{}".format(n_perturb + 1), 0.15)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Perturb restart {}: {:.8f}".format(n_perturb + 1, best_score))
        n_perturb += 1

    # Phase 5: Final high-beta LSE + subgradient
    if time.monotonic() < deadline - 30.0:
        rem = float(deadline - time.monotonic() - 5.0)
        if rem > 20.0:
            n = len(best_seq)
            N = _npow2(2 * n - 1)
            est = 0.008 * float(n) / 4096.0 * (np.log2(max(16, N)) / 12.0)
            mf = int((0.7 * rem) / max(est, 1e-5))
            mf = max(64, min(mf, 2000))
            sc, seq = _run_lbfgs(best_seq, 50000.0, deadline, mf)
            if sc < best_score:
                best_
