"""
Projected gradient with Nesterov momentum + LSE smoothing + Simulated annealing.
Optimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.
Uses simplex projection and adaptive beta annealing for the LSE proxy of max.

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_grad(a, beta, M_fft=None):
    """
    Compute LSE-smoothed score and its gradient for a on the simplex (sum=1).
    Returns (score_approx, grad) where grad is the gradient of 2n*LSE w.r.t. a.
    """
    n = len(a)
    if M_fft is None:
        M_fft = 1 << (2 * n - 1).bit_length()
    
    A = np.fft.rfft(a, M_fft)
    c = np.fft.irfft(A * A, M_fft)[:2 * n - 1]
    S = float(a.sum())  # Should be ~1 if on simplex
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    L = c_max + np.log(exp_sum) / beta
    p = exp_scaled / exp_sum
    
    # Score
    score = 2.0 * n * L / (S * S)
    
    # Gradient of L w.r.t. a: (grad_L)_j = 2 * sum_k p_k * a_{k-j}
    # = 2 * (p cross-correlation a)_j
    # Using FFT: cross_corr = ifft(conj(FFT(p)) * FFT(a))
    N = 1 << (3 * n - 2).bit_length()
    P = np.fft.rfft(p, N)
    A_full = np.fft.rfft(a, N)
    cross = np.fft.irfft(np.conj(P) * A_full, N)[:n]
    grad_L = 2.0 * cross
    
    # Gradient of full score = 2n * (grad_L / S^2 - 2*L/S^3 * 1)
    # But on the simplex, S=1 and dS/da = 1 (vector of ones)
    # So grad_score = 2n * (grad_L - 2*L * 1) / S^2
    # = 2n * (grad_L - 2*L) / S^2 (since S=1)
    grad = 2.0 * n * (grad_L / (S * S) - 2.0 * L / (S ** 3))
    
    return score, grad


def project_simplex(v):
    """Project vector v onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def projected_gradient_lse(a, beta_schedule, n_steps, lr_init, lr_end, 
                           momentum_init=0.9, momentum_end=0.95,
                           anneal_accept=0.01, seed=42):
    """
    Nesterov-accelerated projected gradient on the simplex with LSE smoothing.
    """
    n = len(a)
    rng = np.random.default_rng(seed)
    
    # Ensure a is on the simplex
    a = np.maximum(a, 0)
    S = a.sum()
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    N_grad = 1 << (3 * n - 2).bit_length()
    
    best_score = compute_score(a)
    best_a = a.copy()
    
    # Nesterov momentum
    y = a.copy()
    mu = momentum_init
    
    for step in range(n_steps):
        progress = step / max(1, n_steps - 1)
        
        # Adaptive learning rate (linear decay)
        lr = lr_init * (1 - progress) + lr_end * progress
        
        # Adaptive momentum (increase over time)
        mu = momentum_init * (1 - progress) + momentum_end * progress
        
        # Get beta from schedule
        if isinstance(beta_schedule, (list, np.ndarray)):
            idx = progress * (len(beta_schedule) - 1)
            i = int(idx)
            frac = idx - i
            if i >= len(beta_schedule) - 1:
                beta = beta_schedule[-1]
            else:
                beta = (1 - frac) * beta_schedule[i] + frac * beta_schedule[i + 1]
        else:
            beta = beta_schedule
        
        # Nesterov: evaluate gradient at y (lookahead point)
        y_proj = project_simplex(y)
        sc, grad = lse_grad(y_proj, beta)
        
        if sc < best_score:
            best_score = sc
            best_a = y_proj.copy()
        
        # Update: y = project(a - lr * grad + mu * (a - y_prev))
        # Standard Nesterov: 
        #   y_{k+1} = project(x_k - lr * grad(x_k))
        #   x_{k+1} = y_{k+1} + mu * (y_{k+1} - x_k)
        # Or equivalently:
        #   y_{k+1} = x_k + mu * (x_k - x_{k-1})  (momentum)
        #   x_{k+1} = project(y_{k+1} - lr * grad(y_{k+1}))
        
        # Let me use the "heavy ball" / Nesterov variant:
        velocity = a - y  # This is the "momentum" direction (a is current, y is lookahead)
        # Actually let me use the standard formulation:
        # x_{k+1} = project( y_k - lr * grad(y_k) ) where y_k = x_k + mu*(x_k - x_{k-1})

        # a_new = project(a + v)
        # This is simpler and works well.

        # and rely on the LSE + annealing for convergence speed.
        a_new = project_simplex(a - lr * grad)
        
        # Simulated annealing: occasionally accept a random perturbation
        if rng.random() < anneal_accept and step > n_steps * 0.1:
            # Add random noise
            noise = 0.01 * rng.standard_normal(n)
            a_perturb = project_simplex(a_new + noise)
            sc_perturb = compute_score(a_perturb)
            # Accept with probability based on how much worse it is

            else:
                a = a_new
        else:
            a = a_new
        
        # Update y for next Nesterov step
        y = a
    
    return best_score, best_a

def fienup_quick(a, n_iters=100):

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        thresh = P_mean * 1.3
        above = P > thresh
        if above.any():
            scale = np.ones_like(P)
            scale[above] = np.sqrt(P_mean / P[above]) * 0.9
            A_mod = A * scale
            a_new = np.fft.irfft(A_mod, M_fft)[:n]
            a_new = np.maximum(a_new, 0.0)
            S_new = float(a_new.sum())
            if S_new > 1e-12:

                a = a_new
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
    
    return best_a
