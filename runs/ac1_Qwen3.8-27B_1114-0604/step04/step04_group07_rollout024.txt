"""
Optimized LSE-LBFGS solver for minimizing 2n*max(self-conv)/(sum)^2.
Uses adaptive beta continuation, palindrome space, and multi-phase strategy.
"""
import numpy as np
from scipy.optimize import minimize
import time

def fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a) + len(b) - 1
    M = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, M)
    B = np.fft.rfft(b, M)
    return np.fft.irfft(A * B, M)[:n]

def self_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def score_exact(a):
    """Compute exact score."""
    n = len(a)
    c = self_conv(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    return 2.0 * n * c.max() / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = a.sum()
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = c.max()
    x = beta * (c - c_max)
    # Numerical stability: subtract max (already done by c_max)
    exp_x = np.exp(x)
    logsumexp = c_max + np.log(exp_x.sum()) / beta
    p = exp_x / exp_x.sum()
    
    # Gradient of LSE w.r.t. a: 2 * (rev_a * p)[n-1:2n-1]
    rev_a = a[::-1]
    M2 = n + len(p) - 1  # n + (2n-1) - 1 = 3n-2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n-1:2*n-1]
    grad_L = 2.0 * conv_res
    
    # Full gradient of score = 2n*L/S^2
    score = 2.0 * n * logsumexp / (S * S)
    grad = (2.0 * n * grad_L) / (S * S) - (4.0 * n * logsumexp) / (S * S * S)
    
    return score, grad

def lse_score_grad_pal(half, beta):
    """LSE score and gradient in palindrome half-space."""
    n_half = len(half)
    n = 2 * n_half
    a = np.concatenate([half, half[::-1]])
    sc, grad_full = lse_score_grad(a, beta)
    # grad_half[i] = grad_full[i] + grad_full[n-1-i]
    grad_half = grad_full[:n_half] + grad_full[n-1:n-1+n_half][::-1]
    # Wait: grad_full[n-1-i] for i=0..n_half-1
    # grad_full[n-1], grad_full[n-2], ..., grad_full[n-n_half]
    # = grad_full[n-half : n][::-1]
    # Since n = 2*n_half: grad_full[n_half : 2*n_half][::-1]
    # = grad_full[::-1][:n_half]
    grad_half = grad_full[:n_half] + grad_full[::-1][:n_half]
    return sc, grad_half

def lbfgs_lse(a0, beta, maxiter, maxfun, bounds_upper=1000.0):
    """Run L-BFGS-B on LSE-smoothed score."""
    n = len(a0)
    x0 = np.clip(a0.copy(), 0.0, bounds_upper)
    if x0.sum() < 1e-12:
        x0 = np.ones(n) / n
    bounds = [(0.0, bounds_upper)] * n
    
    def obj(x):
        x = np.maximum(x, 0.0)  # Clip for safety
        sc, gr = lse_score_grad(x, beta)
        return sc, gr
    
    res = minimize(obj, x0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-18, 'gtol': 1e-14, 'maxiter': maxiter,
                            'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, bounds_upper)
    return score_exact(a_out), a_out

def lbfgs_lse_pal(half0, beta, maxiter, maxfun, bounds_upper=1000.0):
    """Run L-BFGS-B on LSE-smoothed score in palindrome space."""
    n_half = len(half0)
    x0 = np.clip(half0.copy(), 0.0, bounds_upper)
    if x0.sum() < 1e-12:
        x0 = np.ones(n_half) / n_half
    bounds = [(0.0, bounds_upper)] * n_half
    
    def obj(h):
        h = np.maximum(h, 0.0)
        sc, gr = lse_score_grad_pal(h, beta)
        return sc, gr
    
    res = minimize(obj, x0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-18, 'gtol': 1e-14, 'maxiter': maxiter,
                            'maxfun': maxfun, 'maxls': 50})
    h_out = np.clip(res.x, 0.0, bounds_upper)
    a_full = np.concatenate([h_out, h_out[::-1]])
    return score_exact(a_full), h_out

def subgrad_descent(a0, n_steps, step_size, time_limit, t_start):
    """Subgradient descent on the exact max."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    best_score = score_exact(a)
    best_a = a.copy()
    
    for step in range(n_steps):
        if time.monotonic() - t_start > time_limit:
            break
        
        c = self_conv(a)
        k_star = c.argmax()
        M = c[k_star]
        S = a.sum()
        score = 2.0 * n * M / (S * S)
        
        # Subgradient
        # grad_i = 2n * 2*a[k_star-i]*1(0<=k_star-i<n) / S^2 - 4n*M/S^3
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je+1] = 2.0 * a[idx]
        
        grad = (2.0 * n * sg) / (S * S) - (4.0 * n * M) / (S * S * S)
        
        # Step with diminishing step size
        lr = step_size / (1.0 + 0.01 * step)
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        # Normalize (score is scale-invariant, so normalization doesn't change score)
        if a.sum() > 1e-12:
            a = a / a.sum()
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
    
    return best_score, best_a

def generate_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.linspace(0, 1, n)
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Beta-like distributions: (x*(1-x))^alpha
    for alpha in [0.5, 1.0, 2.0, 5.0, 10.0]:
        f = (x * (1 - x)) ** alpha
        f = np.maximum(f, 1e-15)
        inits.append(f / f.sum())
    
    # Gaussian
    for sigma in [0.1, 0.2, 0.3, 0.4]:
        f = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        inits.append(f / f.sum())
    
    # Cosine
    for k in [1, 2, 3]:
        f = np.maximum(np.cos(np.pi * x) ** (2*k), 1e-15)
        inits.append(f / f.sum())
    
    # Random smooth
    r = np.abs(rng.standard_normal(n)) + 0.5
    k_sm = max(3, n // 100)
    kernel = np.ones(k_sm) / k_sm
    r_sm = np.convolve(r, kernel, mode='same')
    inits.append(r_sm / r_sm.sum())
    
    # Multi-frequency
    freqs = rng.uniform(1, 20, size=5)
    amps = rng.uniform(0.05, 0.3, size=5)
    phases = rng.uniform(0, 2*np.pi, size=5)
    mod = np.ones(n)
    for f, a_, p in zip(freqs, amps, phases):
        mod += a_ * np.cos(2*np.pi*f*x + p)
    mod = np.maximum(mod, 1e-3)
    inits.append(mod / mod.sum())
    
    return inits

"""
Optimized LSE-LBFGS solver for minimizing 2n*max(self-conv)/(sum)^2.
Uses adaptive beta continuation in palindrome space, diverse restarts,
subgradient polishing, and coordinate descent for final improvement.
Key innovations: adaptive beta scheduling, spectral-aware initialization,
and hybrid smooth/non-smooth optimization phases.

import numpy as np
from scipy.optimize import minimize
import time


def fft_linear_conv(a, b):
    """Linear convolution of two arrays via FFT."""
    n = len(a) + len(b) - 1
    M = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, M)
    B = np.fft.rfft(b, M)
    return np.fft.irfft(A * B, M)[:n]


def self_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def score_exact(a):
    """Compute exact score: 2n * max(a*a) / (sum(a))^2."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return float(2.0 * n * c.max() / (S * S))


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    x = beta * (c - c_max)
    # Clamp for numerical stability
    x = np.clip(x, -700, 700)
    exp_x = np.exp(x)
    logsumexp_val = c_max + float(np.log(exp_x.sum())) / beta
    p = exp_x / exp_x.sum()

    # Gradient of LSE w.r.t. a: 2 * (rev_a * p)[n-1 : 2n-1]
    rev_a = a[::-1]
    M2 = n + len(p) - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    grad_L = 2.0 * conv_res

    score = 2.0 * n * logsumexp_val / (S * S)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n * grad_L) / S2 - (4.0 * n * logsumexp_val) / S3

    return score, grad


def lse_score_grad_pal(half, beta):
    """LSE score and gradient in palindrome half-space."""
    n_half = len(half)
    n = 2 * n_half
    a = np.concatenate([half, half[::-1]])
    sc, grad_full = lse_score_grad(a, beta)
    grad_half = grad_full[:n_half] + grad_full[::-1][:n_half]
    return sc, grad_half


def lbfgs_on_lse(x0, beta, maxiter, maxfun, use_pal=False):
    """Run L-BFGS-B on LSE-smoothed score."""
    if use_pal:
        n_d = len(x0)
    else:
        n_d = len(x0)

    x0 = np.clip(x0.astype(np.float64), 0.0, 1000.0).copy()
    if x0.sum() < 1e-12:
        x0[:] = 1.0 / n_d
    bounds = [(0.0, 1000.0)] * n_d

    def obj(x):
        x_c = np.maximum(x, 0.0)
        if use_pal:
            sc, gr = lse_score_grad_pal(x_c, beta)
        else:
            sc, gr = lse_score_grad(x_c, beta)
        return sc, gr

    res = minimize(obj, x0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-18, 'gtol': 1e-15, 'maxiter': maxiter,
                            'maxfun': maxfun, 'maxls': 30, 'factr': 1e12})
    x_out = np.clip(res.x, 0.0, 1000.0)
    if use_pal:
        a_full = np.concatenate([x_out, x_out[::-1]])
    else:
        a_full = x_out
    return score_exact(a_full), x_out


def subgrad_descent(a0, n_steps, step_size, time_remaining_fn):
    """Subgradient descent on the exact max score."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    best_score = score_exact(a)
    best_a = a.copy()
    t0 = time.monotonic()

    for step in range(n_steps):
        if time.monotonic() - t0 > time_remaining_fn():
            break

        c = self_conv(a)
        k_star = int(c.argmax())
        M = float(c[k_star])
        S = float(a.sum())
        score = 2.0 * n * M / (S * S)

        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = 2.0 * a[idx]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n * sg) / S2 - (4.0 * n * M) / S3

        lr = step_size / (1.0 + 0.005 * step)
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        if a.sum() > 1e-12:
            a = a / a.sum()

        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_score, best_a


def coord_descent(a0, n_sweeps, time_remaining_fn):
    """Coordinate descent targeting peak reduction."""
    n = len(a0)
    a = a0.astype(np.float64).copy()
    best_score = score_exact(a)
    best_a = a.copy()
    t0 = time.monotonic()

    for sweep in range(n_sweeps):
        if time.monotonic() - t0 > time_remaining_fn():
            break

        c = self_conv(a)
        k_star = int(c.argmax())
        score = 2.0 * len(a) * c[k_star] / (a.sum() ** 2)

        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        non_peak = np.ones(n, dtype=bool)
        non_peak[js:je + 1] = False
        nc_idx = np.where(non_peak)[0]

        for j in range(js, je + 1):
            if time.monotonic() - t0 > time_remaining_fn():
                break
            a_old = a[j]
            if a_old < 1e-10:
                continue
            for frac in [0.7, 0.5, 0.3]:
                a[j] = a_old * frac
                removed = a_old - a[j]
                if len(nc_idx) > 0:
                    a[nc_idx] += removed / len(nc_idx)
                sc = score_exact(a)
                if sc < score:
                    score = sc
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                    break
            else:
                a[j] = a_old
                if len(nc_idx) > 0:
                    a[nc_idx] -= (a_old * (1 - 1.0)) / len(nc_idx)

                    score = sc
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                    break
            else:
                a[j] = a_old

    return best_score, best_a


def make_palindromic(a):

    n = len(a)
    n_half = n // 2
    half = (a[:n_half] + a[n - 1:n - 1 + n_half][::-1]) / 2.0
    half = np.maximum(half, 1e-15)
    return half

    for i, init in enumerate(inits):
        if time.monotonic() > p1_end - 3:
            break
        if init.sum() < 1e-12:
            continue
        sc, a_opt = lbfgs_on_lse(init, beta=30, maxiter=300, maxfun=3000)
        candidates.append((sc, a_opt))
        update(sc, a_opt)

    candidates.sort(key=lambda x: x[0])
    print(f"  Top-3: {[f'{c[0]:.6f}' for c in candidates[:3]]} | t={time.monotonic()-t0:.0f}s")

    # ==========================================
    # Phase 2: LSE continuation in palindrome space (37% time)
    # ==========================================
    p2_end = t_start + TOTAL * 0.45
    print(f"\n--- Phase 2: LSE continuation (palindrome) ---")

    a_work = best_a.copy()
    n_half = n // 2
    half = (a_work[:n_half] + a_work[n - 1:n - 1 + n_half][::-1]) / 2.0
    half = np.maximum(half, 1e-15)
    half = half / half.sum()

    # Adaptive beta schedule: start moderate, increase when progress slows
    betas = [10, 30, 80, 200, 500, 1200, 3000, 7000, 15000, 30000, 60000]
    prev_score = score_exact(np.concatenate([half, half[::-1]]))

    for beta in betas:
        if time.monotonic() > p2_end - 5:
            break
        remaining = p2_end - time.monotonic()
        if remaining < 5:
            break

        # Allocate more time to lower betas (easier to converge)
        frac = max(0.15, 1.0 - 0.06 * betas.index(beta))
        time_alloc = min(remaining * frac, 150.0)
        maxiter = max(2000, int(time_alloc * 40))
        maxfun = max(20000, int(time_alloc * 400))

        t0 = time.monotonic()
        sc, h_opt = lbfgs_on_lse(half, beta=beta, maxiter=maxiter, maxfun=maxfun, use_pal=True)
        elapsed = time.monotonic() - t0
        a_full = np.concatenate([h_opt, h_opt[::-1]])
        update(sc, a_full)

        # Check improvement for adaptive scheduling
        improvement = prev_score - sc
        prev_score = sc
        half = h_opt
        print(f"  beta={beta:6d}: {sc:.8f} (imp={improvement:.2e}, {elapsed:.1f}s)")

        # If converged very fast, we can try a bigger jump
        if elapsed < time_alloc * 0.05 and beta < 30000:
            # Insert an intermediate beta
            next_beta = int(beta * 2.5)
            if next_beta in betas:
                betas.insert(betas.index(beta) + 1, next_beta)

    # ==========================================
    # Phase 3: Non-palindromic LSE-LBFGS (20% time)
    # ==========================================
    p3_end = t_start + TOTAL * 0.65
    print(f"\n--- Phase 3: Non-palindromic polish ---")

    a_cur = best_a.copy()
    for beta in [1000, 5000, 15000, 40000]:
        if time.monotonic() > p3_end - 5:
            break
        remaining = p3_end - time.monotonic()
        if remaining < 5:
            break
        time_alloc = min(remaining * 0.35, 100.0)
        maxiter = max(2000, int(time_alloc * 30))
        maxfun = max(20000, int(time_alloc * 300))

        t0 = time.monotonic()
        sc, a_opt = lbfgs_on_lse(a_cur, beta=beta, maxiter=maxiter, maxfun=maxfun, use_pal=False)
        elapsed = time.monotonic() - t0
        update(sc, a_opt)
        a_cur = a_opt
        print(f"  beta={beta:6d}: {sc:.8f} ({elapsed:.1f}s)")

    # ==========================================
    # Phase 4: Diverse restarts (20% time)
    # ==========================================
    p4_end = t_start + TOTAL * 0.85
    print(f"\n--- Phase 4: Diverse restarts ---")

    n_restarts = 16
    for r in range(n_restarts):
        if time.monotonic() > p4_end - 15:
            break

        a_base = best_a.copy()
        strategy = r % 5
        if strategy == 0:
            # Log-space perturbation
            log_a = np.log(np.maximum(a_base, 1e-15))
            scale = 0.02 + 0.04 * rng.random()
            pert = log_a + scale * rng.standard_normal(n)
            a_pert = np.exp(pert - pert.max())
        elif strategy == 1:
            # Peak-avoidance: suppress elements contributing to peak
            c = self_conv(a_base)
            k_star = int(c.argmax())
            js = max(0, k_star - n + 1)
            je = min(k_star, n - 1)
            a_pert = a_base.copy()
            a_pert[js:je + 1] *= (0.4 + 0.3 * rng.random())
            mask = np.ones(n, dtype=bool)
            mask[js:je + 1] = False
            a_pert[mask] *= (1.0 + 0.05 * rng.random())
        elif strategy == 2:
            # Fourier modulation
            freqs = rng.uniform(1, 20, size=4)
            phases = rng.uniform(0, 2 * np.pi, size=4)
            amps = rng.uniform(0.01, 0.08, size=4)
            x = np.arange(n) / n
            mod = np.ones(n)
            for f, p_, a_ in zip(freqs, phases, amps):
                mod += a_ * np.cos(2 * np.pi * f * x + p_)
            a_pert = a_base * np.maximum(mod, 0.05)
        elif strategy == 3:
            # Block-scale perturbation
            n_blocks = 4 + int(6 * rng.random())
            a_pert = a_base.copy()
            bs = n // n_blocks
            for b in range(n_blocks):
                s = b * bs
                e = min(s + bs, n)
                a_pert[s:e] *= rng.uniform(0.6, 1.4)
        else:
            # Antipodal: invert the perturbation direction
            log_a = np.log(np.maximum(a_base, 1e-15))
            scale = 0.03
            pert = log_a - scale * rng.standard_normal(n)
            a_pert = np.exp(pert - pert.max())

        a_pert = np.maximum(a_pert, 1e-15)
        if a_pert.sum() < 1e-12:
            continue

        time_left = p4_end - time.monotonic() - 15
        if time_left < 3:
            break

        # Optimize in palindrome space first (faster)
        n_half = n // 2
        half_pert = (a_pert[:n_half] + a_pert[n - 1:n - 1 + n_half][::-1]) / 2.0
        half_pert = np.maximum(half_pert, 1e-15)
        sc, h_opt = lbfgs_on_lse(half_pert, beta=2000, maxiter=1500, maxfun=15000, use_pal=True)
        a_full = np.concatenate([h_opt, h_opt[::-1]])
        update(sc, a_full)

        # Also try non-palindromic
        if time.monotonic() < p4_end - 15:
            sc2, a_opt2 = lbfgs_on_lse(a_pert, beta=5000, maxiter=1000, maxfun=10000, use_pal=False)
            update(sc2, a_opt2)

        if r % 4 == 0:
            print(f"  restart {r}: best={best_score:.8f}")

    # ==========================================
    # Phase 5: Final intensive polish (10% time)
    # ==========================================
    p5_end = t_start + TOTAL * 0.96
    print(f"\n--- Phase 5: Final polish ---")

    a_cur = best_a.copy()

    # Subgradient descent (targets exact max)
    subgrad_time = min(60.0, (p5_end - time.monotonic()) * 0.3)
    if subgrad_time > 5:
        t0 = time.monotonic()

        def tr1():
            return subgrad_time

        sc, a_opt = subgrad_descent(a_cur, n_steps=100000, step_size=0.005,
                                     time_remaining_fn=tr1)
        update(sc, a_opt)
        a_cur = best_a.copy()
        print(f"  subgrad: best={best_score:.8f} ({time.monotonic()-t0:.1f}s)")

    # Very high beta LSE-LBFGS
    for beta in [50000, 100000, 200000]:
        if time.monotonic() > p5_end - 10:
            break
        remaining = p5_end - time.monotonic() - 5
        if remaining < 3:
            break
        time_alloc = min(remaining * 0.3, 40.0)
        maxiter = max(1000, int(time_alloc * 30))
        maxfun = max(10000, int(time_alloc * 300))
        t0 = time.monotonic()
        sc, a_opt = lbfgs_on_lse(a_cur, beta=beta, maxiter=maxiter, maxfun=maxfun, use_pal=False)
        elapsed = time.monotonic() - t0
        update(sc, a_opt)
        a_cur = a_opt
        print(f"  LBFGS beta={beta}: {sc:.8f} ({elapsed:.1f}s)")

    # Palindrome final push
    if time.monotonic() < p5_end - 8:
        n_half = n // 2
        half_cur = (a_cur[:n_half] + a_cur[n - 1:n - 1 + n_half][::-1]) / 2.0
        half_cur = np.maximum(half_cur, 1e-15)
        remaining = p5_end - time.monotonic() - 3
        if remaining > 3:
            sc, h_opt = lbfgs_on_lse(half_cur, beta=100000, maxiter=2000,
                                      maxfun=20000, use_pal=True)
            a_full = np.concatenate([h_opt, h_opt[::-1]])
            update(sc, a_full)
            print(f"  final pal: {sc:.8f}")

    # Coordinate descent
    if time.monotonic() < p5_end - 3:
        cd_time = min(10.0, p5_end - time.monotonic() - 1)
        if cd_time > 1:
            t0 = time.monotonic()

            def tr2():
                return cd_time

            sc, a_opt = coord_descent(best_a, n_sweeps=200, time_remaining_fn=tr2)
            update(sc, a_opt)
            print(f"  coord: best={best_score:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==========================================
    # Final verification and return
    # ==========================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / 2400s")
    print(f"{'=' * 60}")
    return out
