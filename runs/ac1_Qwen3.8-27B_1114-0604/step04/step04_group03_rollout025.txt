    # This is fine since all u_j ≤ 0, so exp(u_j) ≤ 1, no overflow.
    # But for underflow: if beta is large, most u_j are very negative, exp(u_j) ≈ 0.
    # The sum is dominated by the terms near the max. This is OK.
    
    # Actually, let me redo the stability:
    # scaled = beta * (c - c_max), so scaled ≤ 0, max(scaled) = 0.
    # exp(scaled) ≤ 1, no overflow.
    # log_sum = log(sum(exp(scaled)))
    # L = c_max + log_sum / beta
    
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()

    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad

def spectral_equalization(a, n_iters=300, alpha=0.5, deadline=None):
    """
    Spectral equalization: iterate to flatten the convolution.
    Works in frequency domain to reduce peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Compute convolution
        A = np.fft.rfft(a, M)
        C = A * np.conj(A)  # = |A|^2
        c = np.fft.irfft(C, M)[:2*n-1]
        
        # Target: flat convolution (approximately)
        c_target = np.ones(2*n-1) / (2*n-1)
        
        # Error: how much c exceeds the target
        excess = np.maximum(c - c_target, 0.0)
        deficit = np.maximum(c_target - c, 0.0)
        
        # Modify C to reduce excess
        # The change in c corresponds to FFT(dC) = d_c
        # We want to reduce c where it's too high
        d_c = -alpha * excess + alpha * deficit  # push toward target
        # Actually, let's just do: d_c = alpha * (c_target - c) but only for the excess part
        # Simpler: d_c = -alpha * excess (reduce the peaks)
        d_c = -alpha * excess
        # Also add back to the deficit to preserve sum? Not necessary since we renormalize.
        
        dC = np.fft.rfft(d_c, M)
        C_new = C - alpha * np.real(dC)
        C_new = np.maximum(C_new, 1e-30)
        
        # Update A preserving phase
        phase = A / np.maximum(np.abs(A), 1e-30)
        A_new = phase * np.sqrt(C_new)
        
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
        
    return best_a

def mirror_descent_v2(a, n_iters=2000, eta0=0.05, deadline=None):
    """
    Improved mirror descent with momentum.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Momentum buffer
    v = np.zeros(n)
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        M = float(c.max())
        k_star = int(np.argmax(c))
        
        # Gradient of max(conv) wrt a[i]
        grad_max = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            grad_max[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        # Use soft-max gradient for smoother updates
        beta = 100.0
        c_scaled = beta * (c - M)
        c_scaled -= c_scaled.max()
        p = np.exp(c_scaled)
        p = p / p.sum()
        
        grad_soft = np.zeros(n)
        # grad_soft[i] = 2 * sum_j p[j] * a[j-i] (where valid)
        # = 2 * (rev_a * p)[n-1+i]
        rev_a = a[::-1]
        M2 = 3 * n - 1
        M2_fft = 1 << (M2 - 1).bit_length()
        RA = np.fft.rfft(rev_a, M2_fft)
        P = np.fft.rfft(p, M2_fft)
        r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
        grad_soft = 2.0 * r
        
        # Momentum
        mu = 0.9
        v = mu * v + (1 - mu) * grad_soft
        
        eta = eta0 / (1.0 + 0.05 * it)
        log_a = np.log(a) - eta * v
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        a_new = a_new / a_new.sum()
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a

def coordinate_descent_v2(a, n_sweeps=10, deadline=None):
    """
    Improved coordinate descent: for each peak lag, redistribute mass
    from high-contributing to low-contributing positions.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        c = fast_conv(a)
        # Process top few peaks
        top_k = min(5, 2*n-1)
        peak_idx = np.argsort(c)[::-1][:top_k]
        
        for k_star in peak_idx:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                continue
            
            seg = a[j_start:j_end+1]
            contrib = seg[::-1].copy()  # contribution of each a[j] to c[k_star]
            
            # Find top and bottom contributors
            n_top = max(3, len(contrib) // 10)
            n_bot = n_top
            
            top_local = np.argsort(contrib)[::-1][:n_top]
            bot_local = np.argsort(contrib)[:n_bot]
            
            # Move mass from top to bottom
            mass_to_move = 0.0
            for li in top_local:
                gi = j_start + li
                frac = 0.02 + 0.03 * (sweep / max(1, n_sweeps))
                move = a[gi] * frac
                a[gi] -= move
                mass_to_move += move
            
            if mass_to_move > 1e-15:
                per_bot = mass_to_move / n_bot
                for li in bot_local:
                    gi = j_start + li
                    a[gi] += per_bot
            
            a = np.maximum(a, 0.0)
            S = a.sum()
            if S > 1e-12:
                a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a

    """Return (lse_value, gradient) for the LSE-smoothed score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    # = c_max + (1/beta) * log(sum exp(beta*(c_j - c_max)))
    c_max = float(c.max())
