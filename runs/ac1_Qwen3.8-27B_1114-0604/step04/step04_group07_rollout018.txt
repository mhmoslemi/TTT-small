Another idea: **subgradient method with step size 1/t**.

The subgradient of $f(a) = 2n \cdot \max_k (a*a)[k] / S^2$ (with $S$ fixed to 1) at the current point is:
$$g_i = \frac{2n}{S^2} \cdot \frac{\partial \max_k (a*a)[k]}{\partial a_i}$$

At the max lag $k^*$, the gradient w.r.t. $a_i$ is:
- $2a_{k^*-i}$ if $0 \leq k^*-i < n$ and $k^* \neq 2i$
- $2a_i$ if $k^* = 2i$
- 0 otherwise

So the subgradient is concentrated at $n$ positions (the positions $i$ where $0 \leq k^*-i < n$).

The subgradient step with step size $\alpha_t = 1/\sqrt{t}$ (or $1/t$) is:
$$a_{t+1} = \Pi_C(a_t - \alpha_t \cdot g_t)$$

where $\Pi_C$ is projection onto the constraint set $\{a \geq 0, \sum a = 1\}$.

This is a standard approach for non-smooth convex optimization, but our problem is NOT convex. However, the subgradient method can still make progress locally.

The key issue is choosing the step size. If it's too large, we might oscillate. If it's too small, we converge slowly.

A practical approach: use a decreasing step size with occasional "restarts" (resetting the step size to a larger value when progress stalls).

            m_hat = m / (1.0 - b1 ** t_global)
            v_hat = v / (1.0 - b2 ** t_global)
            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, 1e10)

            if normalize:
                s = a.sum()
                if s > 1e-12:
                    a = a / s

            if (bi * steps_per_beta + t) % 5000 == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def adam_optimize_pal(half_init, beta_schedule, lr, n_steps, time_limit_s, t_start):
    """Adam with LSE smoothing in palindrome half-space."""
    n_half = len(half_init)
    h = half_init.astype(np.float64).copy()
    if h.sum() < 1e-12:
        h[:] = 1.0 / n_half

    m = np.zeros(n_half)
    v = np.zeros(n_half)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_h = h.copy()
    best_score = compute_score(np.concatenate([h, h[::-1]]))

    n_beta = len(beta_schedule)
    steps_per_beta = max(1, n_steps // n_beta)

    for bi, beta in enumerate(beta_schedule):
        steps = min(steps_per_beta, n_steps - bi * steps_per_beta)
        for t in range(steps):
            if time.monotonic() - t_start > time_limit_s:
                return best_score, best_h

            lr_t = lr / (1.0 + 1e-5 * t)
            score, grad = lse_score_grad_pal(h, beta)
            if not np.all(np.isfinite(grad)):
                break

            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            t_global = bi * steps_per_beta + t + 1
            m_hat = m / (1.0 - b1 ** t_global)
            v_hat = v / (1.0 - b2 ** t_global)
            h = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            h = np.clip(h, 0.0, 1e10)

            s = h.sum()
            if s > 1e-12:
                h = h / s

            if (bi * steps_per_beta + t) % 5000 == 0:
                a_full = np.concatenate([h, h[::-1]])
                sc = compute_score(a_full)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()

    a_full = np.concatenate([h, h[::-1]])
    sc_final = compute_score(a_full)
    if sc_final < best_score:
        best_score = sc_final
        best_h = h.copy()

    return best_score, best_h


def run_lbfgs_a(a0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B in a-space with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        x_c = np.clip(x, 0.0, 1e10)
        sc, gr = lse_score_grad(x_c, beta)
        return sc, gr

    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def run_lbfgs_pal(half0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B in palindrome half-space."""
    n_half = len(half0)
    h0 = np.clip(half0.astype(np.float64), 0.0, 1000.0).copy()
    if h0.sum() < 1e-6:
        h0[:] = 1.0 / n_half
    bounds = [(0.0, 1000.0)] * n_half

    def obj(h):
        h_c = np.clip(h, 0.0, 1e10)
        sc, gr = lse_score_grad_pal(h_c, beta)
        return sc, gr

    res = minimize(obj, h0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    h_out = np.clip(res.x, 0.0, 1000.0)
    a_full = np.concatenate([h_out, h_out[::-1]])
    exact_sc = compute_score(a_full)
    return exact_sc, h_out


def coord_descent(a_init, max_sweeps=None, time_limit_s=None, t_start=None,
                  start_idx=0):
    """
    Exact coordinate descent with golden section search.
    Each coordinate update uses O(n) work via incremental convolution updates.
    The 1D subproblem is convex (max of affine/quadratic-convex functions).
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    a = a / S

    convs = fast_conv(a).copy()
    n_conv = 2 * n - 1
    best_score = 2.0 * n * float(convs.max())
    best_a = a.copy()

    GR = (np.sqrt(5.0) - 1.0) / 2.0
    convs_copy = np.empty(n_conv, dtype=np.float64)
    delta = np.empty(n_conv, dtype=np.float64)

    if max_sweeps is None:
        max_sweeps = 100000

    for sweep in range(max_sweeps):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break

        sweep_obj = 2.0 * n * float(convs.max())
        n_updates = 0

        for i in range(n):
            if time_limit_s is not None and t_start is not None:
                if time.monotonic() - t_start > time_limit_s:
                    break

            a_old = a[i]
            S_minus_i = 1.0 - a_old

            k_lo = max(0, i)
            k_hi = min(n_conv - 1, i + n - 1)
            ks = np.arange(k_lo, k_hi + 1)
            kis = ks - i
            valid = (kis >= 0) & (kis < n) & (ks != 2 * i)
            ks_v = ks[valid]
            kis_v = kis[valid]
            a_kis = a[kis_v]

            def objective(t):
                if t < 0.0:
                    return np.inf
                S_t = S_minus_i + t
                if S_t < 1e-15:
                    return np.inf
                delta.fill(0.0)
                if 2 * i <= n_conv - 1:
                    delta[2 * i] = t * t - a_old * a_old
                if len(ks_v) > 0:
                    delta[ks_v] = 2.0 * (t - a_old) * a_kis
                np.add(convs, delta, out=convs_copy)
                return float(convs_copy.max()) / (S_t * S_t)

            hi = max(a_old * 3.0, S_minus_i * 0.5)
            if hi < 1e-10:
                hi = 0.1
            lo = 0.0

            c = hi - GR * (hi - lo)
            d = lo + GR * (hi - lo)
            fc = objective(c)
            fd = objective(d)

            nf = 2
            for _ in range(45):
                nf += 1
                if fc <= fd:
                    hi = d
                    d = c
                    fd = fc
                    c = hi - GR * (hi - lo)
                    fc = objective(c)
                else:
                    lo = c
                    c = d
                    fc = fd
                    d = lo + GR * (hi - lo)
                    fd = objective(d)

            t_opt = (lo + hi) / 2.0
            t_opt = max(t_opt, 0.0)
            if abs(t_opt - a_old) < 1e-14 * max(1.0, a_old):
                continue

            a[i] = t_opt
            S_new_raw = S_minus_i + t_opt
            scale = 1.0 / S_new_raw
            a *= scale
            a[i] = t_opt * scale

            convs = fast_conv(a)

            cur_obj = 2.0 * n * float(convs.max())
            if cur_obj < best_score - 1e-12:
                best_score = cur_obj
                best_a = a.copy()
            n_updates += 1

        if sweep > 0 and abs(sweep_obj - 2.0 * n * float(convs.max())) < 1e-9:
            break

    return best_score, best_a


def subgradient_step(a, step_size):
    """
    Subgradient step targeting the peak convolution lag.
    Reduces elements contributing to the peak, increases others.
    """
    n = len(a)
    convs = fast_conv(a)
    k_star = int(convs.argmax())
    M = convs[k_star]
    S = a.sum()

    # Gradient of (a*a)[k_star] w.r.t. a
    grad = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        grad[js:je + 1] = 2.0 * a[idx]

    # Projected subgradient step (minimize max, so go opposite to gradient)
    # But we want to MINIMIZE the max, so we want to REDUCE the peak.
    # The subgradient points in the direction of increase of the peak.
    # So we step in the negative direction: a - alpha * grad
    a_new = a - step_size * grad / S
    a_new = np.maximum(a_new, 0.0)
    a_new = a_new / a_new.sum()
    return a_new


def generate_initializations(rng, n):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n) / n

    # Uniform
    inits.append(np.ones(n) / n)

    # Gaussian
    for sigma in [0.1, 0.2, 0.3]:
        g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        inits.append(g / g.sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    rc = np.maximum(rc, 1e-10)
    inits.append(rc / rc.sum())

    # Triangular
    tri = np.maximum(0, 1 - np.abs(2 * x - 1))
    if tri.sum() > 0:
        inits.append(tri / tri.sum())

    # Sinusoidal
    for freq in [2, 4, 6, 8]:
        s = 1.0 + 0.3 * np.cos(2 * np.pi * freq * x)
        s = np.maximum(s, 0.01)
        inits.append(s / s.sum())

    # Two-bump
    b = np.exp(-0.5 * ((x - 0.25) / 0.1) ** 2) + np.exp(-0.5 * ((x - 0.75) / 0.1) ** 2)
    inits.append(b / b.sum())

    # Random smooth
    r = np.abs(rng.standard_normal(n)) + 0.5
    k = max(3, n // 50)
    r_smooth = np.convolve(r, np.ones(k) / k, mode='same')
    inits.append(r_smooth / r_smooth.sum())

    # Palindromic random
    half_n = n // 2
    h = np.abs(rng.standard_normal(half_n)) + 0.5
    h_smooth = np.convolve(h, np.ones(max(3, half_n // 50)) / max(3, half_n // 50), mode='same')
    pal = np.concatenate([h_smooth, h_smooth[::-1]])
    inits.append(pal / pal.sum())

    # Flat with center dip (to reduce center peak)
    flat = np.ones(n)
    center = n // 2
    width = n // 10
    flat[center - width:center + width] *= 0.5
    inits.append(flat / flat.sum())

    # Flat with center bump
    flat2 = np.ones(n)
    flat2[center - width:center + width] *= 1.5
    inits.append(flat2 / flat2.sum())

    return [np.clip(x, 1e-10, 1000.0) for x in inits]


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase hybrid optimizer combining gradient descent and coordinate descent.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(60.0, float(budget_s)) - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 32 and np.all(np.isfinite(p)) and np.all(p >= 0):
                parent = np.clip(p, 0.0, 1000.0)
                if parent.sum() > 0:
                    parent = parent / parent.sum()
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # Determine working length
    if parent is not None:
        n_work = len(parent)
        print(f"Parent: n={n_work}, score={compute_score(parent):.8f}")
    else:
        n_work = 4096
        print("No parent, starting fresh with n=4096.")

    print(f"\n{'=' * 60}")
    print(f"Phase 1: Initialization + Screening (n={n_work})")
    print(f"{'=' * 60}")

    # Phase 1: Generate and screen initializations (10% time)
    p1_time = effective_budget * 0.10
    inits = generate_initializations(rng, n_work)
    if parent is not None:
        if len(parent) == n_work:
            inits.append(parent.copy())
        else:
            inits.append(resample_sequence(parent, n_work))

    candidates = []
    for i, init in enumerate(inits):
        if time.monotonic() - t_start > t_start + p1_time - 3:
            break
        t0 = time.monotonic()
        sc, a_opt = adam_optimize(init, [20, 100], lr=0.01, n_steps=2000,
                                   time_limit_s=2.0, t_start=t_start)
        candidates.append((sc, a_opt))
        print(f"  Init {i}: score={sc:.8f} ({time.monotonic()-t0:.1f}s)")

    candidates.sort(key=lambda x: x[0])
    for sc, a_opt in candidates[:3]:
        try_update(sc, a_opt, "screening")
    print(f"  Best after screening: {best_score:.8f}")

    # Phase 2: LSE continuation Adam - Palindrome (25% time)
    p2_time = effective_budget * 0.25
    print(f"\n{'=' * 60}")
    print(f"Phase 2: LSE Adam - Palindrome ({p2_time:.0f}s)")
    print(f"{'=' * 60}")

    n_best = len(best_a)
    if n_best % 2 != 0:
        n_best += 1
        best_a = resample_sequence(best_a, n_best)
    n_half = n_best // 2
    half_best = (best_a[:n_half] + best_a[::-1][:n_half]) / 2.0
    half_best = half_best / half_best.sum()

    betas_pal = [10, 30, 100, 300, 1000, 3000, 10000, 30000]
    lrs_pal = [0.02, 0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003]

    for i, (beta, lr) in enumerate(zip(betas_pal, lrs_pal)):
        remaining = t_start + p2_time - time.monotonic()
        if remaining < 3:
            break
        t0 = time.monotonic()
        sc, h_opt = adam_optimize_pal(half_best, [beta], lr, n_steps=1000000,
                                       time_limit_s=remaining * 0.9, t_start=t_start)
        a_full = np.concatenate([h_opt, h_opt[::-1]])
        try_update(sc, a_full, f"Adam-pal beta={beta}")
        half_best = h_opt
        print(f"  Adam-pal beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Also try non-palindrome
    remaining = t_start + p2_time - time.monotonic()
    if remaining > 10:
        for beta, lr in [(200, 0.003), (1000, 0.001)]:
            remaining = t_start + p2_time - time.monotonic()
            if remaining < 5:
                break
            t0 = time.monotonic()
            sc, a_opt = adam_optimize(best_a, [beta], lr, n_steps=1000000,
                                       time_limit_s=remaining * 0.8, t_start=t_start)
            try_update(sc, a_opt, f"Adam-direct beta={beta}")

    # Phase 3: L-BFGS Polishing (15% time)
    p3_time = effective_budget * 0.15
    print(f"\n{'=' * 60}")
    print(f"Phase 3: L-BFGS Polish ({p3_time:.0f}s)")
    print(f"{'=' * 60}")

    n_best = len(best_a)
    if n_best % 2 != 0:
        n_best += 1
        best_a = resample_sequence(best_a, n_best)
    n_half = n_best // 2
    half_best = (best_a[:n_half] + best_a[::-1][:n_half]) / 2.0
    half_best = half_best / half_best.sum()

    for beta in [500, 2000, 8000, 30000]:
        remaining = t_start + p3_time - time.monotonic()
        if remaining < 5:
            break
        t0 = time.monotonic()
        sc, h_opt = run_lbfgs_pal(half_best, beta, maxiter=3000, maxfun=50000)
        a_full = np.concatenate([h_opt, h_opt[::-1]])
        try_update(sc, a_full, f"LBFGS-pal beta={beta}")
        half_best = h_opt
        print(f"  LBFGS-pal beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Non-palindrome L-BFGS
    for beta in [1000, 5000]:
        remaining = t_start + p3_time - time.monotonic()
        if remaining < 5:
            break
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs_a(best_a, beta, maxiter=3000, maxfun=50000)
        try_update(sc, a_opt, f"LBFGS-direct beta={beta}")
        print(f"  LBFGS-direct beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Phase 4: Coordinate Descent (40% time) - THE KEY PHASE
    p4_time = effective_budget * 0.40
    print(f"\n{'=' * 60}")
    print(f"Phase 4: Coordinate Descent ({p4_time:.0f}s)")
    print(f"{'=' * 60}")

    t0 = time.monotonic()
    sc, a_opt = coord_descent(best_a, max_sweeps=500,
                               time_limit_s=p4_time, t_start=t_start)
    try_update(sc, a_opt, "Coord Descent")
    print(f"  Coord descent: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Phase 5: Alternating gradient + coordinate descent (8% time)
    p5_time = effective_budget * 0.08
    print(f"\n{'=' * 60}")
    print(f"Phase 5: Alternating Grad+CD ({p5_time:.0f}s)")
    print(f"{'=' * 60}")

    remaining = t_start + p5_time - time.monotonic()
    while remaining > 10:
        # Gradient step
        g_time = min(remaining * 0.4, 30.0)
        remaining = t_start + p5_time - time.monotonic()
        if remaining < 5:
            break
        t0 = time.monotonic()
        sc, a_opt = adam_optimize(best_a, [5000], lr=0.0005, n_steps=100000,
                                   time_limit_s=g_time, t_start=t_start)
        try_update(sc, a_opt, f"Alt-Adam")

        # Coordinate descent step
        remaining = t_start + p5_time - time.monotonic()
        if remaining < 5:
            break
        cd_time = min(remaining * 0.5, 30.0)
        t0 = time.monotonic()
        sc2, a_opt2 = coord_descent(best_a, max_sweeps=50,
                                     time_limit_s=cd_time, t_start=t_start)
        try_update(sc2, a_opt2, f"Alt-CD")
        print(f"  Alt iteration: {best_score:.8f} ({time.monotonic()-t_start:.0f}s)")
        remaining = t_start + p5_time - time.monotonic()

    # Phase 6: Final intensive coordinate descent (2% time)
    p6_time = effective_budget * 0.02
    print(f"\n{'=' * 60}")
    print(f"Phase 6: Final Coord Descent ({p6_time:.0f}s)")
    print(f"{'=' * 60}")

    remaining = t_start + p6_time - time.monotonic()
    if remaining > 2:
        t0 = time.monotonic()
        sc, a_opt = coord_descent(best_a, max_sweeps=50,
                                   time_limit_s=remaining, t_start=t_start)
        try_update(sc, a_opt, "Final CD")
        print(f"  Final CD: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
