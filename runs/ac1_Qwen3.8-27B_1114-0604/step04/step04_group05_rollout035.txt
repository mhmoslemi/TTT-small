with peak-lag-targeted refinement for minimizing 2n*max(self-conv)/(sum)^2.

import numpy as np

import math

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient via cross-correlation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def fienup_algorithm(a, n_iters=200, target_level=None):
    """
    Fienup/Gerchberg-Saxton style algorithm for spectral flattening.
    Alternates between:
    - Time domain: project onto {a >= 0, sum(a) = 1}
    - Frequency domain: flatten |A_k| to target level
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S  # normalize to sum=1
    
    if target_level is None:
        # Target: flat magnitude = 1/sqrt(n) (roughly)
        target_level = 1.0 / math.sqrt(n)
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for it in range(n_iters):
        # Go to frequency domain
        M_fft = 1 << (2 * n - 1 - 1).bit_length()
        A = np.fft.rfft(a, M_fft)
        
        # Flatten magnitude
        mag = np.abs(A)
        phase = np.angle(A)
        # Use a soft flattening: blend current magnitude with target
        new_mag = 0.5 * mag + 0.5 * target_level
        # Actually, for Fienup, we should set magnitude exactly to target
        # But for non-negative constraint, exact flattening might give negative values
        # So use a compromise
        A_new = new_mag * np.exp(1j * phase)
        
        # Back to time domain
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        
        # Project onto {a >= 0, sum = 1}
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new = a_new / s
        
        # Check progress
        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

def tilted_cosine_init(n, tilt=0.0):
    """
    Generate a tilted cosine bump of length n.
    a_j = sin(pi*(j+0.5)/n) * (1 + tilt * (2j/(n-1) - 1))
    Normalized to sum=1, clipped to non-negative.
    """
    x = np.arange(n, dtype=np.float64)
    base = np.sin(np.pi * (x + 0.5) / n)
    if tilt != 0:
        frac = 2.0 * x / (n - 1) - 1.0 if n > 1 else np.zeros(n)
        base = base * (1.0 + tilt * frac)
    base = np.maximum(base, 0)
    s = base.sum()
    if s < 1e-12:
        base = np.ones(n) / n
    else:
        base = base / s
    return base

def run_lbfgs(a0, beta, maxiter=3000, timeout=30.0):
    """Run L-BFGS-B with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0, 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    
    t_start = time.monotonic()
    
    def objective_and_grad(x):
        if time.monotonic() - t_start > timeout:
            raise TimeoutError
        return lse_score_grad(x, beta)
    
    try:
        res = minimize(
            objective_and_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxiter * 3, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except (TimeoutError, Exception):
        a_out = a0
    
    sc = compute_score(a_out)
    return sc, a_out

def peak_targeted_coord_descent(a, n_sweeps=5, top_k=30, timeout=30.0):
    """
    Coordinate descent that targets the peak lag specifically.
    For the lag k* that achieves max convolution, we identify which
    coordinates contribute most and optimize those first.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    t_start = time.monotonic()
    
    for sweep in range(n_sweeps):
        if time.monotonic() - t_start > timeout:
            break
        
        c = fast_conv(a)
        k_star = int(c.argmax())
        M = float(c[k_star])
        S = float(a.sum())
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score
        
        # For the peak lag k_star, the contribution of a_j is a_j * a_{k_star-j}
        # We want to reduce this. The most effective coordinates are those
        # where a_j * a_{k_star-j} is large.
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            contributions = a[js:je+1] * a[k_star - np.arange(js, je+1)]
            # Top contributors
            local_order = np.argsort(contributions)[::-1][:top_k]
            order = (js + local_order).tolist()
        else:
            order = list(range(min(top_k, n)))
        
        # Also add some random coordinates for exploration
        remaining = [j for j in range(n) if j not in set(order)]
        if remaining:
            extra = np.random.default_rng(sweep).choice(remaining, size=min(10, len(remaining)), replace=False)
            order.extend(extra.tolist())
        
        for j in order[:top_k + 10]:
            if time.monotonic() - t_start > timeout:
                break
            # Golden section for a[j]
            a_old = a[j]
            lo = 0.0
            hi = max(a_old * 3.0, a.max() * 0.5)
            
            gr = (math.sqrt(5.0) - 1.0) / 2.0
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            
            a[j] = c1
            f1 = compute_score(a)
            a[j] = c2
            f2 = compute_score(a)
            
            for _ in range(12):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    a[j] = c1
                    f1 = compute_score(a)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    a[j] = c2
                    f2 = compute_score(a)

"""
Tilted-shape initialization + Fienup spectral flattening + KKT multiplicative updates
+ Aggressive LSE L-BFGS + Peak-lag-targeted coordinate descent for minimizing
2n*max(self-conv)/(sum)^2 over non-negative sequences.
Key innovations: (1) tilted cosine base shapes exploit asymmetry to reduce middle
convolution peak, (2) Fienup alternating projections flatten the power spectrum,
(3) KKT-motivated multiplicative updates equalize responsibility across active lags,
(4) aggressive continuation beta schedule with peak-targeted coordinate refinement.

import numpy as np
from scipy.optimize import minimize
import time
import math


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def conv_peak_info(a):
    """Return convolution, max value, argmax, and sum."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(c.argmax())
    M = float(c[k_star])
    S = float(a.sum())
    return c, k_star, M, S


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def tilted_cosine_init(n, tilt=0.3):
    """Generate tilted cosine bump normalized to sum=1, non-negative."""
    x = np.arange(n, dtype=np.float64)
    base = np.sin(np.pi * (x + 0.5) / n)
    if tilt != 0.0 and n > 1:
        frac = 2.0 * x / (n - 1) - 1.0
        base = base * (1.0 + tilt * frac)
    base = np.maximum(base, 0.0)
    s = base.sum()
    if s < 1e-12:
        base = np.ones(n) / n
    else:
        base = base / s
    return base


def gaussian_bump_init(n, center_frac=0.4, width_frac=0.25, tilt=0.0):
    """Gaussian bump with optional tilt, normalized to sum=1."""
    x = np.arange(n, dtype=np.float64)
    center = center_frac * (n - 1)
    width = width_frac * n
    base = np.exp(-0.5 * ((x - center) / width) ** 2)
    if tilt != 0.0 and n > 1:

    s = base.sum()

def fienup_algorithm(a, n_iters=150, blend=0.5, timeout=15.0):
    """
    Fienup/Gerchberg-Saxton spectral flattening.
    Alternates between time-domain non-negativity projection and
    frequency-domain magnitude flattening.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S0 = a.sum()
    if S0 < 1e-12:
        return a
    a = a / S0

    M_fft = 1 << (2 * n - 2).bit_length()
    target_level = S0 / math.sqrt(M_fft)

    best_a = a.copy()
    best_score = compute_score(a * S0)
    t_start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t_start > timeout:
            break

        A = np.fft.rfft(a, M_fft)
        mag = np.abs(A)
        phase = np.angle(A)

        # Blend current magnitude with target
        new_mag = (1.0 - blend) * mag + blend * target_level
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M_fft)[:n]

        # Project onto a >= 0, sum = 1
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new = a_new / s

        sc = compute_score(a_new * S0)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        a = a_new

    return best_a * S0


def kkt_multiplicative_update(a, n_iters=50, tol=1e-10, timeout=10.0):
    """
    KKT-motivated multiplicative update.
    At the optimum, the 'responsibility' vector r_j = sum_{k in active} a_{k-j}
    should be constant for all j with a_j > 0.
    We iteratively rescale a_j to equalize r.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S0 = a.sum()
    if S0 < 1e-12:
        return a
    a_norm = a / S0

    best_a = a.copy()
    best_score = compute_score(a)
    t_start = time.monotonic()

    for it in range(n_iters):
        if time.monotonic() - t_start > timeout:
            break

        c = fast_conv(a_norm)
        M = float(c.max())
        # Active set: lags within 2% of max
        threshold = M * 0.98
        active_mask = c >= threshold
        S_idx = np.where(active_mask)[0]

        if len(S_idx) == 0 or len(S_idx) > 2 * n:
            break

        # Compute responsibility vector
        r = np.zeros(n)
        for k in S_idx:
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                r[js:je + 1] += a_norm[k - np.arange(js, je + 1)]

        # Multiplicative update: a_j <- a_j * (mean(r) / r_j)^alpha
        r_safe = np.maximum(r, 1e-10)
        target = r[a_norm > 1e-10].mean() if (a_norm > 1e-10).any() else 1.0
        alpha = 0.3
        scale = (target / r_safe) ** alpha
        scale = np.clip(scale, 0.5, 2.0)
        a_new = a_norm * scale
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            break
        a_new = a_new / s

        # Damping
        a_norm = 0.5 * a_norm + 0.5 * a_new

        sc = compute_score(a_norm * S0)
        if sc < best_score:
            best_score = sc
            best_a = (a_norm * S0).copy()

    return best_a


def run_lbfgs(a0, beta, maxiter=3000, timeout=30.0):
    """Run L-BFGS-B with LSE smoothing and time limit."""
    n = len(a0)
    a0 = np.clip(a0, 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    t_start = time.monotonic()

    def objective_and_grad(x):
        if time.monotonic() - t_start > timeout:
            raise TimeoutError
        return lse_score_grad(x, beta)

    try:
        res = minimize(
            objective_and_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxiter * 3, 'maxls': 50}
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except (TimeoutError, Exception):
        a_out = a0

    sc = compute_score(a_out)
    return sc, a_out


def adam_optimize(a_init, lr=0.005, n_steps=50000, beta=500, timeout=30.0):
    """Adam optimizer with LSE smoothing."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    S0 = a.sum()

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)
    t_start = time.monotonic()

    for t in range(1, n_steps + 1):
        if time.monotonic() - t_start > timeout:
            break

        lr_t = lr / (1.0 + 1e-5 * t)
        score, grad = lse_score_grad(a, beta)

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)

        s = a.sum()
        if s > 1e-12:
            a = a * (S0 / s)

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a

def spectral_perturbation(a, rng, strength=0.15):

    amp_factors = np.clip(amp_factors, 0.3, 2.0)
    phase_shifts = strength * 0.5 * rng.standard_normal(len(A))

    if a_new.sum() < 1e-10:

    else:
        a_new = a_new * (a.sum() / a_new.sum())

    for ri in range(n_restarts):
        if time.monotonic() > p3_end - 15:
            break
        t_remain = p3_end - time.monotonic()
        if t_remain < 20:
            break
        time_alloc = min(t_remain * 0.85, 40.0)

        # Choose perturbation strategy
        strategy = ri % 4
        if strategy == 0:
            a_pert = spectral_perturbation(best_a, rng, strength=0.2)
            label = "spectral"
        elif strategy == 1:
            a_pert = antipalindromic_perturbation(best_a, rng, strength=0.2)
            label = "anti-palin"
        elif strategy == 2:
            a_pert = chirp_perturbation(best_a, rng)
            label = "chirp"
        else:
            a_pert = spectral_perturbation(best_a, rng, strength=0.1)
            a_pert = antipalindromic_perturbation(a_pert, rng, strength=0.15)
            label = "spectral+anti"

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        sc, a_opt = refinement_pipeline(a_pert, time_alloc, t_start)
        try_update(sc, a_opt, f"restart#{ri}: {label}")

    # ================================================================
    # Phase 4: Peak-targeted coordinate descent (10% budget)
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Peak-targeted coordinate descent")
    print(f"{'='*60}")
    p4_end = t_start + effective_budget * 0.80

    if time.monotonic() < p4_end - 5:
        sc, a_opt = peak_targeted_coord_descent(best_a, n_sweeps=10, top_k=50,
                                                 timeout=p4_end - time.monotonic() - 5)
        try_update(sc, a_opt, "CoordDesc intensive")

    # Another round of KKT + CD
    if time.monotonic() < p4_end - 5:
        sc, a_opt = kkt_multiplicative_update(best_a, n_iters=80, timeout=5.0)
        try_update(sc, a_opt, "KKT before CD")
        sc, a_opt = peak_targeted_coord_descent(best_a, n_sweeps=5, top_k=40,
                                                 timeout=min(10.0, p4_end - time.monotonic() - 3))
        try_update(sc, a_opt, "CD after KKT")

    # ================================================================
    # Phase 5: Different lengths (10% budget)
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Different sequence lengths")
    print(f"{'='*60}")
    p5_end = t_start + effective_budget * 0.90

    n_best = len(best_a)
    for factor in [0.8, 1.2, 1.5, 0.65]:
        if time.monotonic() > p5_end - 20:
            break
        tl = int(n_best * factor)
        if tl < 256 or tl > 25000:
            continue

        x_old = np.arange(n_best, dtype=np.float64)
        x_new = np.linspace(0, n_best - 1, tl)
        a_res = np.interp(x_new, x_old, best_a)
        a_res = np.clip(a_res, 0.0, 1000.0)

        # Apply tilt to resampled
        x_t = np.arange(tl, dtype=np.float64)
        frac = 2.0 * x_t / max(tl - 1, 1) - 1.0
        a_tilted = a_res * (1.0 + 0.3 * frac)
        a_tilted = np.maximum(a_tilted, 0.0)
        if a_tilted.sum() > 1e-10:
            a_tilted = a_tilted / a_tilted.sum() * a_res.sum()

        time_left = p5_end - time.monotonic() - 15
        if time_left < 15:
            continue
        sc, a_opt = refinement_pipeline(a_tilted, min(time_left, 30.0), t_start)
        try_update(sc, a_opt, f"n={tl}")

    # ================================================================
    # Phase 6: Final polish (10% budget)
    # ================================================================
    print(f"\n{'='*60}")
    print(f"Phase 6: Final polish")
    print(f"{'='*60}")
    final_deadline = t_start + effective_budget - 5.0

    # Very high beta L-BFGS
    if time.monotonic() < final_deadline - 15:
        for beta in [30000, 100000]:
            if time.monotonic() > final_deadline - 10:
                break
            t_avail = final_deadline - time.monotonic() - 8
            if t_avail < 5:
                break
            sc, a_opt = run_lbfgs(best_a, beta=beta, maxiter=5000, timeout=min(t_avail, 25.0))
            try_update(sc, a_opt, f"Final LBFGS b={beta}")

    # Final KKT
    if time.monotonic() < final_deadline - 8:
        sc, a_opt = kkt_multiplicative_update(best_a, n_iters=60, timeout=5.0)
        try_update(sc, a_opt, "Final KKT")

    # Final coordinate descent
    remaining = final_deadline - time.monotonic()
    if remaining > 5:
        sc, a_opt = peak_targeted_coord_descent(best_a, n_sweeps=8, top_k=50,
                                                 timeout=min(remaining - 2, 20.0))
        try_update(sc, a_opt, "Final CoordDesc")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
