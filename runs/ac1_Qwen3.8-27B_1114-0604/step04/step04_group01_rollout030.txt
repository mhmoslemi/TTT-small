"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LSE-smoothed L-BFGS-B with adaptive temperature + exact subgradient polish.
"""

import numpy as np
import time
from scipy.optimize import minimize

def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    return a * (float(target) / S)

def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))

def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S

def lse_score_grad(a, beta):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n)
    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    h = corr_pair(a, w)
    score = 2.0 * L
    grad = (4.0 * float(n) * h) / (S * S) - (4.0 * LDS) / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad

def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score_a(a), a
    bounds = [(0.0, 10.0)] * n
    opts = {"ftol": 1e-16, "gtol": 1e-13, "maxiter": max(maxfun//2, 50), "maxfun": int(maxfun), "maxls": 50}
    try:
        res = minimize(lambda x: lse_score_grad(x, beta), a, method="L-BFGS-B", jac=True, bounds=bounds, options=opts)
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    sc = exact_score_a(a_out)
    if not np.isfinite(sc):
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out

"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses LSE-smoothed L-BFGS-B with front-loaded high-beta schedule,
exact subgradient with adaptive line search, and multi-length exploration.

import numpy as np
import time
from scipy.optimize import minimize

PEN_SUM = 0.0


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def corr_pair(a, w):
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = int(a.size)
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + int(w.size) - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, float(target) / float(a.size), dtype=np.float64)
    return a * (float(target) / S)


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64) / float(max(1, n))
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, a).astype(np.float64)


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def d_from_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.zeros(2 * n - 1, dtype=np.float64), S
    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    return d, S


def lse_score_grad(x, beta):
    a = np.asarray(x, dtype=np.float64)
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    d, S = d_from_a(a)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n)
    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e
    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    h = corr_pair(a, w)
    score = 2.0 * L
    grad = (4.0 * float(n) * h) / (S * S) - (4.0 * LDS) / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def run_lbfgs(a0, beta, deadline, maxfun):
    a = normalize_a(a0)
    n = len(a)
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 5.0:
            return exact_score_a(a), a
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-16,
        "gtol": 1e-13,
        "maxiter": max(maxfun // 2, 30),
        "maxfun": int(maxfun),
        "maxls": 40,
    }
    try:
        res = minimize(lse_score_grad, a, method="L-BFGS-B", jac=True,
                       args=(beta,), bounds=bounds, options=opts)
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def estimate_maxfun(t_left, n):
    t_left = float(max(1.0, t_left))
    if t_left < 3.0:
        return 48
    base = 0.015
    scale = max(0.5, float(n) / 8192.0)
    log_scale = max(1.0, float(np.log2(max(16, n))) / 13.0)
    est = base * scale * log_scale
    mf = int((0.75 * t_left) / max(est, 1e-4))
    return max(48, min(mf, 4000))


def subgrad_with_linesearch(a, deadline, max_iters=50):
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    eta = 0.5
    n = len(a)

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break
        d, S = d_from_a(a)
        if S < 1e-12 or d.size == 0:
            break
        dmax = float(np.max(d))
        if not np.isfinite(dmax) or dmax <= 0:
            break

        # Find all near-max indices (within 0.5% of max)
        thresh = dmax * 0.998
        active = np.where(d >= thresh)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(d))])
        w = np.zeros(d.size)
        w[active] = 1.0 / float(len(active))

        h = corr_pair(a, w)
        hmax = float(np.max(h))
        if hmax < 1e-12:
            break

        # Try a range of step sizes with line search
        found_improvement = False
        candidates = []
        base_eta = eta / hmax
        for mult in (1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.06):
            eta_try = base_eta * mult
            q = a * np.exp(-eta_try * h)
            q = normalize_a(q)
            sc = exact_score_a(q)
            if np.isfinite(sc):
                candidates.append((sc, q, eta_try))

        if not candidates:
            break

        candidates.sort(key=lambda x: x[0])
        best_sc_try, best_q, best_eta_try = candidates[0]

        if best_sc_try < best_sc:
            best_sc = best_sc_try
            best_a = best_q.copy()
            a = best_q.copy()
            eta = max(eta * 1.2, best_eta_try)
            found_improvement = True
        else:
            # Try even smaller steps
            for mult in (0.04, 0.02, 0.01):
                eta_try = base_eta * mult
                q = a * np.exp(-eta_try * h)
                q = normalize_a(q)
                sc = exact_score_a(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                    a = q.copy()
                    eta = eta_try
                    found_improvement = True
                    break
            if not found_improvement:
                eta *= 0.7
                if eta < 1e-6:
                    break

def perturb_sequence(a, rng, strength=None):
    p = normalize_a(np.asarray(a, dtype=np.float64).ravel())
    n = len(p)
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))
    mode = int(rng.integers(0, 5))

    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.05, 0.25))
        q = (1.0 - t) * p + t * (np.ones(n) / float(n))
    elif mode == 2:
        d, S = d_from_a(p)
        thresh = float(np.max(d)) * 0.995
        active = np.where(d >= thresh)[0]
        if len(active) > 0:
            w = np.zeros(d.size)
            w[active] = 1.0 / float(len(active))
            h = corr_pair(p, w)
            hmax = float(np.max(h))
            if hmax > 1e-12:
                q = p * np.exp(-strength * h / hmax)
            else:
                q = p * np.exp(0.05 * rng.standard_normal(n))
        else:
            q = p * np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        # Smooth perturbation
        noise = rng.standard_normal(n)
        k = min(21, max(5, n // 300))
        if k % 2 == 0:
            k += 1
        c = k // 2
        sig = max(1.0, c / 2.0)
        kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
        kern /= float(kern.sum())
        noise = np.convolve(noise, kern, mode="same")
        noise = noise - float(noise.mean())
        logp = np.log(p + 1e-300)
        logp += strength * 0.3 * noise
        q = np.exp(logp - np.max(logp))
    else:
        # Block perturbation
        nb = int(rng.integers(4, 16))
        block_size = max(1, n // nb)
        q = p.copy()
        for _ in range(min(3, nb)):
            bi = int(rng.integers(0, nb))
            s = bi * block_size
            e = min(s + block_size, n)
            q[s:e] *= np.exp(strength * rng.standard_normal(e - s))

    return normalize_a(q)

                print("Phase2 subgrad: {:.8f}".format(best_score))

            sc, a2 = subgrad_multipartition(a2, phase2_deadline, max_rounds=15)
            if sc < best_score:
                best_score = sc
                best_seq = a2.copy()
                print("Phase2 multi-subgrad: {:.8f}".format(best_score))

    # Phase 3: Perturbation restarts
    phase3_deadline = deadline - 30.0
    perturb_count = 0
    while time.monotonic() < phase3_deadline and perturb_count < 8:
        left = float(phase3_deadline - time.monotonic())
        if left < 60.0:
            break
        chunk = min(time.monotonic() + left / float(8 - perturb_count), phase3_deadline)

        p = perturb_sequence(best_seq, rng)
        sc, seq = quick_optimize(p, chunk)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Perturb restart {}: {:.8f}".format(perturb_count + 1, best_score))
        perturb_count += 1

    # Final polish
    if time.monotonic() < deadline - 15.0:
        sc, a2 = subgrad_with_linesearch(best_seq, deadline - 15.0, max_iters=25)
        if sc < best_score:
            best_score = sc
            best_seq = a2.copy()
            print("Final subgrad: {:.8f}".format(best_score))

        sc, a2 = subgrad_multipartition(a2, deadline - 15.0, max_rounds=8)
        if sc < best_score:
            best_score = sc
            best_seq = a2.copy()
            print("Final multi-subgrad: {:.8f}".format(best_score))

    # Prepare output
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024, dtype=np.float64) * 0.5
    else:
        out_arr = out_arr / S
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL internal best={:.10f} verified={:.10f} length={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
