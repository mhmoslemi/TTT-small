Let `d[m] = n * c[m] / S^2` and `L = LSE_beta(d[m] for m in active)` with weights `w[m]`.

`dL/da[i] = sum_{m in active} w[m] * dd[m]/da[i]`

`dd[m]/da[i] = n * dc[m]/da[i] / S^2 - n*c[m] * 2 / S^3 = 2*n*a[m-i]/S^2 - 2*n*c[m]/S^3`
`= 2*n*(a[m-i]/S^2 - c[m]/S^3)`
`= 2*n*(a[m-i]*S - c[m])/S^3`
`= 2*d[m]/S * (1 - a[m-i]*S/c[m])` ... this is getting messy.

Let me just write it as:
`dd[m]/da[i] = 2*n*(a[m-i] - c[m]/S) / S^2`

So: `dL/da[i] = sum_{m in active} w[m] * 2*n*(a[m-i] - c[m]/S) / S^2`
`= 2*n/S^2 * (sum_{m in active} w[m] * a[m-i] - (1/S) * sum_{m in active} w[m] * c[m])`
`= 2*n/S^2 * (corr[a, w_symmetric] - (w·c)/S)`

where `corr[a, w_symmetric][i] = sum_m w[m] * a[m-i]` (with out-of-bounds = 0).

Now, `sum_m w[m] * a[m-i]` is the cross-correlation of `a` with `w`, shifted. Specifically, if I define the "correlation" as:
`h[i] = sum_m w[m] * a[m-i]`

This is `h = correlate(a, w)` in my function (with appropriate indexing).

Actually, let me just use the `corr_pair` function from the previous code. It computes:
`h[j] = sum_k w[k] * a[k-j]` for `j = 0, ..., n-1`

So `h[j] = sum_k w[k] * a[k-j]`. If I identify `k` with `m` and `j` with `i`, then `h[i] = sum_m w[m] * a[m-i]`. That's exactly what I need!

So the gradient of the LSE score (without the sum penalty) is:
`grad[i] = 2*n/S^2 * (corr_pair(a, w)[i] - (w·c)/S)`

And the full score is `2*L` (where `L` is the LSE value of `d = n*c/S^2` on the active set).

If I add a penalty `PEN * (S - 1)^2` to keep the sum near 1, the gradient gets an extra `2*PEN*(S-1)`.

```python
    h[j] = sum_k w[k] * a[k-j], for j = 0,...,n-1.
```

```python
"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses active-set LSE optimization with multilevel length scaling,
adaptive K (number of active convolution positions), and subgradient polishing.
"""

import numpy as np
import time
from scipy.optimize import minimize

# --- Core utilities ---

def next_pow2(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()

def self_conv(a):
    """Full linear self-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def cross_corr(a, w):
    """
    Compute h[j] = sum_k w[k] * a[k-j] for j=0,...,n-1.
    Uses FFT: middle slice of conv(a[::-1], w).
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n-1:2*n-1]

def norm(a, target=1.0):
    """Clip to non-negative, normalize to target sum."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)

def score(a):
    """Exact score: 2*n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))

def upsample(a, n):
    """Linearly resample to length n."""
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)

# --- LSE objective with active set ---

def lse_obj(a, beta, k):
    """
    LSE-smoothed score using top-k active positions.
    a is assumed to be approximately normalized (sum ≈ 1).
    Returns (value, gradient).
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)
    
    c = self_conv(a)
    d = float(n) * c / (S * S)  # d[m] = n*c[m]/S^2
    M = 2 * n - 1
    
    k = max(1, min(k, M))
    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)
    
    z = beta * (d_act - dmax)
    z_shift = z - z.max()  # = -beta * (dmax - d_act) <= 0
    e = np.exp(z_shift)
    se = e.sum()
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta
    
    # Build full weight vector
    w_full = np.zeros(M)
    w_full[idx] = w_act
    
    # corr = cross_corr(a, w_full): corr[i] = sum_m w_full[m] * a[m-i]
    corr = cross_corr(a, w_full)
    LDS = float(w_act @ d_act)  # sum of w[m]*d[m] over active
    
    # dL/da[i] = 2*n/S^2 * (corr[i] - LDS)
    # (derived from dd[m]/da[i] = 2*n*(a[m-i] - c[m]/S)/S^2)
    # and sum w[m]*c[m] = LDS/S (since d[m] = n*c[m]/S^2, so c[m] = d[m]*S^2/n)
    # Actually: sum w[m]*c[m] = sum w[m]*d[m]*S^2/n = LDS*S^2/n
    # So: dL/da[i] = 2*n/S^2 * (corr[i] - LDS*S^2/n / S) = 2*n/S^2 * (corr[i] - LDS/S)
    # = 2*n*corr[i]/S^2 - 2*LDS/S
    
    grad = 2.0 * float(n) * corr / (S * S) - 2.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    
    val = 2.0 * L
    return float(val), grad

# --- L-BFGS-B wrapper ---

def run_lbfgs(a0, beta, k, deadline, maxfun):
    """Run L-BFGS-B on the LSE active-set objective."""
    a = norm(a0)
    n = len(a)
    rem = deadline - time.monotonic()
    if rem < 5:
        return score(a), a
    
    # Estimate eval time
    est_t = 0.005 * max(1.0, n / 4096.0)
    mf = int((0.8 * rem) / max(est_t, 1e-4))
    mf = max(50, min(mf, maxfun, 5000))
    
    def obj(x):
        v, g = lse_obj(x, beta, k)
        # Add sum penalty
        S = float(x.sum())
        v += 1e-4 * (S - 1.0) ** 2
        g = g + 2e-4 * (S - 1.0) * np.ones_like(x)
        return v, g
    
    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': max(30, mf//2), 'maxfun': mf, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 30}
        )
        out = norm(res.x) if res.x is not None else a
    except Exception:
        out = a
    
    return score(out), out

# --- Subgradient polishing ---

def subgrad_polish(a, n_iters=30, deadline=None):
    """
    Exact subgradient method on the max convolution objective.
    """
    a = norm(a)
    best_a = a.copy()
    best_sc = score(a)
    
    for it in range(n_iters):
        if deadline and time.monotonic() > deadline:
            break
        
        S = float(a.sum())
        if S < 1e-12:
            break
        c = self_conv(a)
        d = float(len(a)) * c / (S * S)
        dmax = d.max()
        if dmax <= 0:
            break
        
        # Active set: positions where d is within 1e-3 of max
        active = d >= dmax - 1e-4
        if not active.any():
            active[np.argmax(d)] = True
        
        # Subgradient direction
        n_act = active.sum()
        w = np.zeros_like(d)
        w[active] = 1.0 / n_act
        
        corr = cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        # grad = 2*n*(corr - LDS/S) / S^2 ... let me just use the formula
        # The subgradient of max(d)/S^2 at a is:
        # (1/S^2) * sum_t w_t * grad(margin_t) where margin_t = d[t] - second_max
        # More directly: grad_f = (1/S^2) * sum w_t * grad(d[t])
```

```python
def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def _cross_corr(a, w):
    """h[j] = sum_k w[k] * a[k-j] for j=0..n-1 via FFT."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    if n == 0 or w.size == 0:
        return np.empty(n, dtype=np.float64)
    M = n + w.size - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]


def _norm(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.clip(np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0), 0.0, None)
    S = a.sum()
    if S < 1e-12:
        return np.full(a.size, target / a.size)
    return a * (target / S)


def _score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = _self_conv(a)
    return float(2.0 * n * max(0.0, c.max()) / (S * S))


def _upsample(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel())
    m = a.size
    if m == 0:
        return np.ones(n) / n
    if m == n:
        return a.copy()
    x0 = np.linspace(0, 1, m)
    x1 = np.linspace(0, 1, n)
    return np.interp(x1, x0, a).astype(np.float64)


def _lse_obj_grad(a, beta, k):
    """
    LSE-smoothed 2*max(d) using top-k positions of d = n*conv(a,a)/S^2.
    Returns (value, gradient) for L-BFGS-B.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = _self_conv(a)
    d = float(n) * c / (S * S)
    M = 2 * n - 1
    k = max(1, min(k, M))

    idx = np.argpartition(d, M - k)[M - k:]
    d_act = d[idx]
    dmax = float(d_act.max())
    if dmax <= 0 or not np.isfinite(dmax):
        return 1e10, np.zeros(n)

    z = beta * (d_act - dmax)
    e = np.exp(z)  # z <= 0, so e <= 1, numerically safe
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        e = np.ones(k) / k
        se = 1.0
    w_act = e / se
    L = dmax + np.log(se) / beta

    w_full = np.zeros(M)
    w_full[idx] = w_act

    corr = _cross_corr(a, w_full)
    LDS = float(w_act @ d_act)

    # grad of 2*L w.r.t. a:
    # d(2L)/da[i] = 4n*corr[i]/S^2 - 4*LDS/S
    grad = 4.0 * float(n) * corr / (S * S) - 4.0 * LDS / S
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)

    val = 2.0 * L
    return float(val), grad


def _run_lbfgs(a0, beta, k, time_limit, maxfun_cap):
    """Run L-BFGS-B on active-set LSE objective with sum penalty."""
    a = _norm(a0)
    n = len(a)
    t_end = time.monotonic() + time_limit
    if time_limit < 3.0:
        return _score(a), a

    est_eval_t = 0.004 * max(0.5, float(n) / 4096.0)
    mf = int(0.85 * time_limit / max(est_eval_t, 1e-4))
    mf = max(30, min(mf, maxfun_cap, 8000))
    maxiter = max(20, mf // 3)
    PEN = 1e-3

    def obj(x):
        v, g = _lse_obj_grad(x, beta, k)
        Sx = float(x.sum())
        v += PEN * (Sx - 1.0) ** 2
        g = g + 2.0 * PEN * (Sx - 1.0)
        return v, g

    try:
        res = minimize(
            obj, a, method='L-BFGS-B', jac=True,
            bounds=[(0.0, None)] * n,
            options={'maxiter': maxiter, 'maxfun': mf,
                     'ftol': 1e-15, 'gtol': 1e-11, 'maxls': 40}
        )
        out = _norm(res.x) if getattr(res, 'x', None) is not None else a
    except Exception:
        out = a
    return _score(out), out


def _subgrad_polish(a, max_iters=30, deadline=None):
    """Exact subgradient descent with adaptive step size on the max-convolution objective."""
    a = _norm(a)
    best_a = a.copy()
    best_sc = _score(a)

    for it in range(max_iters):
        if deadline and time.monotonic() > deadline:
            break
        S = float(a.sum())
        if S < 1e-12:
            break
        c = _self_conv(a)
        d = float(len(a)) * c / (S * S)
        dmax = float(d.max())
        if dmax <= 0:
            break

        # Active set: positions within 0.1% of max
        active = d >= dmax - 1e-5 * max(dmax, 1.0)
        if not active.any():
            active[np.argmax(d)] = True
        n_act = int(active.sum())

        w = np.zeros_like(d)
        w[active] = 1.0 / n_act

        corr = _cross_corr(a, w)
        LDS = float(w[active] @ d[active])
        grad = 4.0 * float(len(a)) * corr / (S * S) - 4.0 * LDS / S
        grad = np.nan_to_num(grad, nan=0.0)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        best_step = None
        best_new_sc = best_sc
        for scale in (0.0005, 0.002, 0.008, 0.02, 0.08, 0.2, 0.5):
            eta = scale / gnorm
            q = a - eta * grad
            q = np.clip(q, 0.0, None)
            Sq = q.sum()
            if Sq < 1e-8:
                continue
            q = q / Sq
            sc = _score(q)
            if sc < best_new_sc:
                best_new_sc = sc
                best_step = q.copy()

        if best_step is None:
            break
        if best_new_sc < best_sc:
            best_sc = best_new_sc
            best_a = best_step.copy()
        a = best_step

    return best_sc, best_a


def _spectral_init(n, rng):
    """Generate a non-negative sequence with approximately flat power spectrum."""
    Nf = _next_pow2(2 * n - 1)
    freqs = np.fft.rfftfreq(Nf) * Nf
    exponent = rng.uniform(0.3, 1.5)
    power = 1.0 / (1.0 + np.abs(freqs)) ** exponent
    power[0] = 0.0
    phase = rng.uniform(-np.pi, np.pi, size=len(power))
    spec = power * np.exp(1j * phase)
    sig = np.fft.irfft(spec, Nf)[:n]
    sig = sig - sig.min()
    if sig.max() <= 0:
        sig = np.ones(n) / n
    else:
        sig = sig / sig.sum()
    return sig


def _smooth_perturb(a, rng, strength=0.3):
    """Apply smooth random perturbation and renormalize."""
    n = len(a)
    noise = rng.standard_normal(n)
    k = min(31, max(3, n // 200))
    if k % 2 == 0:
        k += 1
    if k > n:
        k = max(3, n if n % 2 == 1 else n - 1)
    c = k // 2
    sig = max(1.0, c / 2.0)
    kern = np.exp(-((np.arange(k) - c) ** 2) / (2.0 * sig * sig))
    kern /= kern.sum()
    noise = np.convolve(noise, kern, mode='same')
    q = a + strength * noise
    return _norm(q)


def _optimize_trajectory(a0, deadline, label, rng):
    """
    Progressive active-set K growth optimization.
    K grows from small to large, beta increases accordingly.
    """
    a = _norm(a0)
    best_a = a.copy()
    best_sc = _score(a)
    n = len(a)
    t0 = time.monotonic()

    print(f"  [{label}] start: score={best_sc:.8f}, n={n}")

    # Stages: (K, beta, time_fraction)
    total_time = max(10.0, deadline - t0 - 10.0)
    stages = [
        (3, 30.0, 0.06),
        (5, 60.0, 0.08),
        (8, 120.0, 0.10),
        (12, 250.0, 0.12),
        (20, 500.0, 0.14),
        (40, 1000.0, 0.16),
        (80, 2500.0, 0.18),
        (150, 5000.0, 0.16),
    ]

    cum_time = 0.0
    for k, beta, frac in stages:
        cum_time += frac
        stage_end = t0 + cum_time * total_time
        stage_end = min(stage_end, deadline - 10.0)
        stage_budget = stage_end - time.monotonic()
        if stage_budget < 5.0:
            break

        # Run L-BFGS-B for this stage
        # Split budget across a few calls for robustness
        n_calls = min(4, max(1, int(stage_budget / 20)))
        per_call = stage_budget / n_calls

        for ci in range(n_calls):
            if time.monotonic() > stage_end - 2.0:
                break
            sc, a_new = _run_lbfgs(a, beta, k, per_call, 400)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
                print(f"    [{label}] K={k}, beta={beta:.0f}, call{ci}: {best_sc:.8f}")
            a = _norm(a_new)

    # Subgradient polish at the end
    if time.monotonic() < deadline - 10.0:
        sc, a_sg = _subgrad_polish(a, max_iters=15, deadline=deadline - 5.0)
        if sc < best_sc:
            best_sc = sc
            best_a = a_sg.copy()
            print(f"    [{label}] subgrad: {best_sc:.8f}")

    print(f"  [{label}] end: score={best_sc:.8f}, elapsed={time.monotonic()-t0:.0f}s")
    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function: active-set LSE with multilevel scaling."""
    t_start = time.monotonic()
    deadline = t_start + budget_s - 25.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.clip(np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0), 0.0, 1000.0)
            if p.size >= 64:
                parent = _norm(p)
    except Exception:
        pass

    # Determine target length
    if parent is not None:
        target_n = len(parent)
        if target_n < 4096:
            target_n = 8192
    else:
        target_n = 8192

    print(f"Parent: {parent is not None}, target_n={target_n}, budget={budget_s}s")

    # --- Generate diverse initializations ---
    inits = []

    # 1. Parent (resampled to target_n)
    if parent is not None:
        inits.append(('parent', _upsample(parent, target_n)))
        inits.append(('parent_rev', _upsample(parent[::-1], target_n)))

    # 2. Spectral inits
    for i in range(6):
        inits.append((f'spec{i}', _spectral_init(target_n, rng)))

    # 3. Parent + smooth noise
    if parent is not None:
        p = _upsample(parent, target_n)
        for amp in (0.05, 0.15, 0.3):
            inits.append((f'pn{amp}', _smooth_perturb(p, rng, amp)))

    # 4. Cosine modulations
    x = np.linspace(0, 1, target_n)
    for f in (1, 2, 3):
        for amp in (0.15, 0.3, 0.5):
            a = 1.0 + amp * np.cos(2 * np.pi * f * x)
            a = np.clip(a, 0, None)
            if a.sum() > 0:
                inits.append((f'cos{f}_{amp}', a / a.sum()))

    # 5. Pure noise (smoothed)
    for i in range(3):
        noise = rng.standard_normal(target_n)
        k = max(5, target_n // 100)
        if k % 2 == 0:
            k += 1
        kern = np.exp(-((np.arange(k) - k // 2) ** 2) / (2 * (k // 4) ** 2))
        kern /= kern.sum()
        noise = np.convolve(noise, kern, mode='same')
        a = np.exp(noise)
        inits.append((f'noise{i}', a / a.sum()))

    # Evaluate and rank
    scored = []
    for name, arr in inits:
        arr = _norm(arr)
        sc = _score(arr)
        if np.isfinite(sc):
            scored.append((sc, name, arr))

    scored.sort(key=lambda x: x[0])
    print(f"Generated {len(scored)} inits. Top scores: {[f'{s:.6f}' for s, _, _ in scored[:5]]}")

    # Select top 3 diverse candidates
    candidates = []
    for sc, name, arr in scored:
        if len(candidates) >= 3:
            break
        candidates.append((sc, name, arr))

    if not candidates:
        candidates = [(2.0, 'uniform', _norm(np.ones(target_n)))]

    best_score = candidates[0][0]
    best_seq = candidates[0][2].copy()

    # --- Main optimization ---
    # Allocate time to each candidate
    main_time = deadline - t_start - 300.0  # reserve 300s for final polish
    n_cands = len(candidates)

    fractions = []
    if n_cands == 1:
        fractions = [1.0]
    elif n_cands == 2:
        fractions = [0.60, 0.40]
    else:
        fractions = [0.50, 0.30, 0.20]

    for ci, (sc_init, name, arr_init) in enumerate(candidates):
        sub_deadline = t_start + (sum(fractions[:ci+1]) * main_time) + 30.0
        sub_deadline = min(sub_deadline, deadline - 300.0)

        if time.monotonic() > sub_deadline:
            continue

        n = len(arr_init)
        print(f"\n=== Candidate {ci+1}: '{name}', n={n}, score={sc_init:.8f} ===")

        # Multilevel warm start: optimize at n//4 first (if n is large)
        a_work = _norm(arr_init)
        if n > 4096 and time.monotonic() < sub_deadline - 60.0:
            n_small = max(512, n // 4)
            a_small = _upsample(a_work, n_small)
            sc_small = _score(a_small)
            print(f"  Multilevel: n_small={n_small}, score={sc_small:.8f}")

            # Quick optimize at small level
            time_per_stage = min(20.0, (sub_deadline - 60.0 - time.monotonic()) / 6.0)
            for k_s, beta_s in [(3, 50), (10, 200), (30, 1000)]:
                if time.monotonic() > sub_deadline - 60.0:
                    break
                sc_s, a_s = _run_lbfgs(a_small, beta_s, k_s, time_per_stage, 200)
                if sc_s < sc_small:
                    sc_small = sc_s
                    a_small = a_s

            # Upscale back
            a_up = _norm(_upsample(a_small, n))
            sc_up = _score(a_up)
            if sc_up < _score(a_work):
                a_work = a_up
                print(f"  Multilevel improved: {sc_up:.8f}")

        # Main trajectory optimization
        sc_traj, seq_traj = _optimize_trajectory(a_work, sub_deadline, name, rng)
        if sc_traj < best_score:
            best_score = sc_traj
            best_seq = seq_traj.copy()
            print(f"  *** Global best updated: {best_score:.8f} ***")

    # --- Final global polish ---
    print(f"\n=== Final polish (score={best_score:.8f}) ===")
    final_deadline = deadline - 20.0

    # Very high beta LSE on the best
    for k_final, beta_final in [(200, 8000), (500, 20000), (1000, 40000)]:
        if time.monotonic() > final_deadline - 30:
            break
        remaining = final_deadline - time.monotonic() - 30
        if remaining < 10:
            break
        sc_f, a_f = _run_lbfgs(best_seq, beta_final, k_final, remaining * 0.4, 600)
        if sc_f < best_score:
            best_score = sc_f
            best_seq = a_f.copy()
            print(f"  Final LSE K={k_final}, beta={beta_final}: {best_score:.8f}")

    # Subgradient polish
    if time.monotonic() < final_deadline - 15:
        sc_sg, a_sg = _subgrad_polish(best_seq, max_iters=20, deadline=final_deadline - 10)
        if sc_sg < best_score:
            best_score = sc_sg
            best_seq = a_sg.copy()
            print(f"  Final subgrad: {best_score:.8f}")

    # Random restart perturbation
    if time.monotonic() < final_deadline - 60:
        for restart in range(3):
            if time.monotonic() > final_deadline - 30:
                break
            p = _smooth_perturb(best_seq, rng, strength=0.1)
            chunk_deadline = min(final_deadline - 30, time.monotonic() + 15)
            sc_r, a_r = _run_lbfgs(p, 5000, 200, max(2, chunk_deadline - time.monotonic()), 200)
            if sc_r < best_score:
                best_score = sc_r
                best_seq = a_r.copy()
                print(f"  Restart {restart}: {best_score:.8f}")

    # Final subgradient
    if time.monotonic() < final_deadline - 5:
        sc_sg2, a_sg2 = _subgrad_polish(best_seq, max_iters=10, deadline=final_deadline)
        if sc_sg2 < best_score:
            best_score = sc_sg2
            best_seq = a_sg2.copy()

    # Prepare output
    out = _norm(best_seq)
    out = np.clip(out, 0.0, 1000.0)
    S = float(out.sum())
    if S < 0.01:
        out = np.ones(1024) * 0.5
    else:
        out = out / S

    result = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(result))
    except Exception:
        verified = best_score

    print(f"\nFINAL: internal={best_score:.8f}, verified={verified:.8f}, "
          f"len={len(result)}, time={time.monotonic()-t_start:.1f}s")
    return result
```
