```python
import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    n_a, n_b = len(a), len(b)
    M = n_a + n_b - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a. Full space."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())

    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dLSE_da / S2 - 2.0 * L / S3)
    return score, grad
```

```python
"""
Full-space LSE-smoothed optimizer for minimizing 2n*max(self_conv)/(sum)^2.
Key insight: Optimal solutions are NOT palindromic (palindrome forces score >= 2).
Uses: Adam with LSE beta continuation, L-BFGS polishing, Frank-Wolfe steps,
and multi-scale exploration in full (non-symmetric) space.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    n_a, n_b = len(a), len(b)
    M = n_a + n_b - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a (full space, no symmetry)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e30, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-30:
        return 1e30, np.zeros(n)
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum

    # Gradient: dLSE/da_j = 2 * q[j], q[j] = sum_i a[i]*p[i+j]
    # q[j] = (a_rev * p)[n-1+j]
    a_rev = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    AR = np.fft.rfft(a_rev, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(AR * P, M2_fft)[:M2]
    q = conv_res[n - 1:2 * n - 1]

    dLSE_da = 2.0 * q
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dLSE_da / S2 - 2.0 * L / S3)
    return score, grad


def lse_subgrad(a):
    """Exact score and subgradient at the peak lag."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(np.argmax(c))
    score = 2.0 * n * M / (S * S)

    # Subgradient: dScore/da_j = 2n * (2*a[k_star-j]*1[S terms]) ...
    # At the peak: dScore/da_j = 2n * (2*a[k_star-j]/S^2 - 2*M*a_j/S^3)
    # where a[k_star-j] is valid when 0 <= k_star-j < n
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]

    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (sg / S2 - 2.0 * M * a / S3)
    return score, grad


def adam_full(a_init, beta_schedule, lr_init, n_steps, time_limit, t_start):
    """Adam optimizer with LSE smoothing in full space."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)
    t_global = 0

    n_beta = len(beta_schedule)
    steps_per_beta = max(1, n_steps // n_beta)

    for bi, beta in enumerate(beta_schedule):
        steps = min(steps_per_beta, n_steps - bi * steps_per_beta)
        for t in range(steps):
            if time.monotonic() - t_start > time_limit:
                return best_score, best_a

            t_global += 1
            lr_t = lr_init / (1.0 + 5e-5 * t_global)
            score, grad = lse_score_grad(a, beta)

            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            m_hat = m / (1.0 - b1 ** t_global)
            v_hat = v / (1.0 - b2 ** t_global)
            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, 1e10)

            # Normalize to sum=1 for stability
            s = a.sum()
            if s > 1e-12:
                a = a / s

            if t % 2000 == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def lbfgs_full(a0, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B in full space with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        x_c = np.clip(x, 0.0, 1e10)
        sc, gr = lse_score_grad(x_c, beta)
        return sc, gr

    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30})
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-12:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def frank_wolfe_nestorov(a_init, n_steps, time_limit, t_start, beta=100):
    """Nesterov-accelerated Frank-Wolfe (conditional gradient) on simplex."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(n_steps):
        if time.monotonic() - t_start > time_limit:
            break
        t1 = t + 1
        gamma = 2.0 / (t1 + 3.0)  # step size

        score, grad = lse_score_grad(a, beta)

        # Frank-Wolfe direction: find j minimizing <grad, e_j>
        j = int(np.argmin(grad))

        # Nesterov: x_k = (1-gamma)*y_k + gamma*e_j
        a = (1.0 - gamma) * a
        a[j] += gamma

        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_score, best_a


def mirror_descent_entropy(a_init, n_steps, time_limit, t_start, beta=50):
    """Mirror descent with entropy proximity operator on simplex."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)
    lr = 0.01

    for t in range(n_steps):
        if time.monotonic() - t_start > time_limit:
            break
        t1 = t + 1
        lr_t = lr / np.sqrt(t1)

        score, grad = lse_score_grad(a, beta)

        # Mirror descent update: a_j <- a_j * exp(-lr * grad_j)
        a = a * np.exp(-lr_t * np.clip(grad, -100, 100))
        a = np.maximum(a, 1e-15)
        a = a / a.sum()

        if t % 1000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    return best_score, best_a


def resample_sequence(a, new_n):
    """Resample sequence to new length via interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def peak_targeting_refine(a, n_sweeps=100, time_limit=None, t_start=None):
    """Coordinate descent targeting the peak convolution lag."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_score = compute_score(a)
    deadline = (t_start + time_limit) if (time_limit is not None and t_start is not None) else np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        score = compute_score(a)
        c = fast_conv(a)
        k_star = int(c.argmax())

        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)

        # Try reducing elements contributing to peak
        for j in range(js, je + 1):
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            if a_old < 1e-12:
                continue
            for frac in [0.7, 0.5, 0.3]:
                a[j] = a_old * frac
                # Redistribute
                removed = a_old - a[j]
                mask = np.ones(n, dtype=bool)
                mask[js:je + 1] = False
                if mask.sum() > 0:
                    a[mask] += removed / mask.sum()
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                sc = compute_score(a)
                if sc < score:
                    score = sc
                    if sc < best_score:
                        best_score = sc
                    break
            else:
                a[j] = a_old  # revert

    return best_score, a
```
