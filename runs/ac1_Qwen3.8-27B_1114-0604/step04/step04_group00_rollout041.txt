import numpy as np

def _clip_clean(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    return a

def normalize_sum(a):
    a = np.asarray(a, dtype=np.float64)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(a.shape, 1.0/max(1,len(a)))
    return a/s

def next_pow2(m):
    return 1 << (max(1, int(m)) - 1).bit_length()

def fast_conv(a):
    n = len(a)
    if n == 1:
        return np.array([float(a[0]*a[0])])
    M = 2*n - 1
    L = next_pow2(M)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A*A, L)[:M]
    # remove tiny imaginary? irfft real
    return c

def score_of(a):
    a = normalize_sum(a)
    c = fast_conv(a)
    s = float(a.sum())
    if s <= 1e-300: return np.inf
    return float(2.0*len(a)*float(np.max(c))/(s*s))

```python
"""
Log-softmax minimax optimizer for nonnegative step-function heights.

The score is scale invariant:
    score(a) = 2*n*max(a*a) / (sum(a))**2.
We normalize a to sum 1 and minimize max convolution directly.

Core idea:
- Parameterize p = softmax(u), so p_i >= 0 and sum(p) = 1.
- Minimize a log-sum-exp surrogate of max(conv(p,p)) with RMSProp-style
  updates in the log variables. The active-set weights are chosen from the
  current convolution peaks, with annealed softmax/top-k mixing.
- Polish with L-BFGS on the same smoothed surrogate in log-softmax space.
- Try a few nearby lengths and structured initializations to avoid bad
  local minima.
"""

import numpy as np
from scipy.optimize import minimize
import time


def _next_pow2(m: int) -> int:
    m = max(1, int(m))
    return 1 << (m - 1).bit_length()


def _clip_clean(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    return np.clip(a, 0.0, 1000.0)


def normalize_sum(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return np.full(a.shape, 1.0 / max(1, a.size))
    return a / s


def fast_conv(a):
    """Linear self-convolution via FFT. Returns length 2n-1."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    if n == 1:
        return np.array([float(a[0] * a[0])], dtype=np.float64)
    m = 2 * n - 1
    L = _next_pow2(m)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:m]
    return c


def score_normalized(p):
    """Score assuming p is already normalized sum=1."""
    n = len(p)
    if n == 0:
        return np.inf
    c = fast_conv(p)
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0.0:
        mx = max(0.0, mx)
    return float(2.0 * n * mx)


def score_of(a):
    p = normalize_sum(_clip_clean(a))
    if p.size == 0:
        return np.inf
    return score_normalized(p)


def _softmax_vec(u):
    u = np.asarray(u, dtype=np.float64).ravel()
    if u.size == 0:
        return u
    if not np.all(np.isfinite(u)):
        u = np.nan_to_num(u, nan=0.0, posinf=1.0, neginf=-100.0)
    u = u - float(np.mean(u))
    m = float(np.max(u))
    e = np.exp(u - m)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        idx = int(np.argmax(u))
        out = np.zeros_like(u, dtype=np.float64)
        out[idx] = 1.0
        return out
    return e / s


def cross_corr_qp(q, p):
    """
    For q length 2n-1 and p length n, return r_i = sum_k q_k p_{k-i},
    with p_j = 0 outside [0,n-1].
    """
    n = len(p)
    if n == 0:
        return np.array([], dtype=np.float64)
    L = _next_pow2(3 * n - 2)
    Q = np.fft.rfft(q, L)
    P = np.fft.rfft(p[::-1], L)
    h = np.fft.irfft(Q * P, L)
    return h[n - 1:n + n - 1]


def make_conv_weights(c, temp_factor, top_frac, mix):
    """Stable weights over convolution coefficients: softmax + optional top-k."""
    Lc = len(c)
    if Lc == 0:
        return c
    cmax = float(np.max(c))
    cmean = float(np.mean(c))
    if not np.isfinite(cmean) or cmean <= 0.0:
        cmean = max(1e-300, cmax if cmax > 0.0 else 1e-300)
    temp = max(1e-300, cmean * max(1e-12, float(temp_factor)))
    z = (c - cmax) / temp
    z = np.clip(z, -745.0, 0.0)
    e = np.exp(z)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        q = np.zeros(Lc, dtype=np.float64)
        q[int(np.argmax(c))] = 1.0
    else:
        q = e / s

    if float(top_frac) > 0.0 and float(mix) < 1.0 and Lc > 1:
        k = int(np.ceil(float(top_frac) * Lc))
        k = min(max(1, k), Lc - 1)
        idx = np.argpartition(c, -k)[-k:]
        top = np.zeros(Lc, dtype=np.float64)
        top[idx] = 1.0 / float(k)
        mix = float(np.clip(mix, 0.0, 1.0))
        if mix <= 0.0:
            q = top
        elif mix >= 1.0:
            pass
        else:
            q = mix * q + (1.0 - mix) * top
            ss = float(np.sum(q))
            if ss > 0.0 and np.isfinite(ss):
                q = q / ss
            else:
                q = top
    return q


def peak_refine(p0, t_deadline, seed=0, tag="peak",
                temp_f_start=0.5, temp_f_end=0.04,
                top_frac=0.06, mix_start=1.0, mix_end=0.3,
                max_delta_start=0.25, max_delta_end=0.03,
                lr_start=0.30, lr_end=0.08,
                restart_after=12000,
                noise_start=0.05, noise_end=0.35):
    """
    Gradient/RMSProp minimax refinement in log-softmax coordinates.
    """
    t0 = time.monotonic()
    total = float(t_deadline - t0)
    if total <= 1.0:
        p = normalize_sum(_clip_clean(p0))
        return p, score_normalized(p) if p.size else np.inf

    p = normalize_sum(_clip_clean(p0))
    n = len(p)
    if n == 0:
        p = np.ones(1024, dtype=np.float64)
        n = p.size

    p = np.maximum(p, 1e-300)
    p = p / float(np.sum(p))
    u = np.log(p)
    u = u - float(np.mean(u))
    u = np.clip(u, -100.0, 100.0)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)

    rng = np.random.default_rng((int(seed) & 0x7fffffff) ^ (n * 1000003))

    c = fast_conv(p)
    best_sc = score_normalized(p)
    best_p = p.copy()

    b1 = 0.90
    b2 = 0.999
    b1_t = 1.0
    b2_t = 1.0
    no_impr = 0
    last_print = t0

    print(f"[{tag}] start n={n} score={best_sc:.9f} budget={total:.1f}s")

    while time.monotonic() < t_deadline - 0.5:
        elapsed = time.monotonic() - t0
        progress = elapsed / max(1e-12, total)
        if progress < 0.0:
            progress = 0.0
        elif progress > 1.0:
            progress = 1.0

        temp_f = temp_f_start * (float(temp_f_end) / float(temp_f_start)) ** progress
        mix = float(mix_start) + (float(mix_end) - float(mix_start)) * progress
        lr = float(lr_start) + (float(lr_end) - float(lr_start)) * progress
        md = float(max_delta_start) + (float(max_delta_end) - float(max_delta_start)) * progress

        adapt = 1.0 + min(2.0, float(no_impr) / 20000.0)
        lr_eff = min(1.4, lr * adapt)
        md_eff = min(1.0, md * adapt)

        if no_impr > int(restart_after):
            noise = float(noise_start) + (float(noise_end) - float(noise_start)) * min(
                1.0, float(no_impr) / (5.0 * float(restart_after))
            )
            ub = np.log(np.maximum(best_p, 1e-300))
            ub = ub - float(np.mean(ub))
            u = ub + rng.normal(0.0, noise, n)
            u = np.clip(u, -100.0, 100.0)
            m.fill(0.0)
            v.fill(0.0)
            b1_t = 1.0
            b2_t = 1.0
            no_impr = 0
            um = u - float(np.max(u))
            e = np.exp(um)
            es = float(np.sum(e))
            if not np.isfinite(es) or es <= 0.0:
                p = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                p = e / es
            c = fast_conv(p)

        q = make_conv_weights(c, temp_f, top_frac, mix)
        r = cross_corr_qp(q, p)
        s_val = float(np.dot(p, r))
        g = p * (r - s_val)

        if not np.all(np.isfinite(g)):
            p = np.maximum(best_p, 1e-300)
            p = p / float(np.sum(p))
            u = np.log(p)
            u = u - float(np.mean(u))
            c = fast_conv(p)
            m.fill(0.0)
            v.fill(0.0)
            b1_t = 1.0
            b2_t = 1.0
            continue

        m *= b1
        m += (1.0 - b1) * g
        v *= b2
        v += (1.0 - b2) * (g * g)
        b1_t *= b1
        b2_t *= b2

        m_hat = m / (1.0 - b1_t)
        v_hat = v / (1.0 - b2_t)
        step = m_hat / (np.sqrt(v_hat) + 1e-8)

        smax = float(np.max(np.abs(step)))
        if smax > 0.0:
            limit = md_eff / (lr_eff * smax)
            if limit < 1.0:
                step *= limit

        u = u - lr_eff * step
        if not np.all(np.isfinite(u)):
            u = np.nan_to_num(u, nan=0.0, posinf=0.0, neginf=-100.0)
            u = np.clip(u, -100.0, 100.0)

        u = u - float(np.mean(u))
        u = np.clip(u, -100.0, 100.0)

        um = u - float(np.max(u))
        e = np.exp(um)
        es = float(np.sum(e))
        if not np.isfinite(es) or es <= 0.0:
            p = np.full(n, 1.0 / n, dtype=np.float64)
            u = np.zeros(n, dtype=np.float64)
        else:
            p = e / es

        pmx = float(np.max(p))
        if pmx > 0.5:
            p = 0.9 * p + 0.1 / n
            p = np.maximum(p, 1e-300)
            p = p / float(np.sum(p))
            u = np.log(p)
            u = u - float(np.mean(u))
            u = np.clip(u, -100.0, 100.0)
            m.fill(0.0)
            v.fill(0.0)
            b1_t = 1.0
            b2_t = 1.0

        c = fast_conv(p)
        mx = float(np.max(c))
        sc = float(2.0 * n * mx)

        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_p = p.copy()
            no_impr = 0
            if time.monotonic() - last_print > 5.0:
                print(f"[{tag}] improved t={time.monotonic()-t0:7.1f}s score={sc:.9f}")
                last_print = time.monotonic()
        else:
            no_impr += 1

        if time.monotonic() - last_print > 30.0:
            print(f"[{tag}] t={time.monotonic()-t0:7.1f}s cur={sc:.9f} best={best_sc:.9f} noimp={no_impr}")
            last_print = time.monotonic()

    return best_p, best_sc


def lse_objective(u, beta):
    """
    Smooth log-sum-exp objective and gradient for p = softmax(u).
    Optimizes LSE(conv(p,p)), which approximates max(conv(p,p)).
    """
    u = np.asarray(u, dtype=np.float64).ravel()
    n = u.size
    if n == 0:
        return 1e100, np.array([], dtype=np.float64)

    if not np.all(np.isfinite(u)):
        u = np.nan_to_num(u, nan=0.0, posinf=0.0, neginf=-100.0)
    u = u - float(np.mean(u))

    p = _softmax_vec(u)
    c = fast_conv(p)
    cmax = float(np.max(c))
    if not np.isfinite(cmax) or cmax <= 0.0:
        return 1e100, np.zeros(n, dtype=np.float64)

    z = float(beta) * (c - cmax)
    z = np.clip(z, -745.0, 0.0)
    ez = np.exp(z)
    zs = float(np.sum(ez))
    if not np.isfinite(zs) or zs <= 0.0:
        ez = np.zeros_like(c)
        ez[int(np.argmax(c))] = 1.0
        zs = 1.0
    q = ez / zs

    val = cmax + float(np.log(zs)) / float(beta)

    r = cross_corr_qp(q, p)
    s_val = float(np.dot(p, r))
    grad = p * (r - s_val)

    # Tiny ridge in the log-softmax gauge direction to help L-BFGS.
    ridge = 1e-12 * float(np.dot(u, u))
    grad = grad + 2e-12 * u

    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=np.float64)

    return float(val + ridge), grad


def lbfgs_refine(p0, t_deadline, beta=20000.0, tag="lbfgs"):
    """
    L-BFGS polish on log-sum-exp surrogate in log-softmax coordinates.
    """
    t_now = time.monotonic()
    p = normalize_sum(_clip_clean(p0))
    if p.size == 0:
        p = np.ones(1024, dtype=np.float64)
    if t_now >= t_deadline - 3.0:
        return p, score_normalized(p)

    u = np.log(np.maximum(p, 1e-300))
    u = u - float(np.mean(u))
    u = np.clip(u, -100.0, 100.0)

    # Calibration for maxfun.
    t_cal0 = time.monotonic()
    for _ in range(3):
        lse_objective(u, beta)
    dt = (time.monotonic() - t_cal0) / 3.0
    dt = max(5e-4, dt)

    remaining = max(1.0, t_deadline - time.monotonic())
    maxfun = int(min(60000, max(800, remaining / (dt * 2.5))))
    maxiter = int(max(100, maxfun // 10))

    options = {
        "maxiter": maxiter,
        "maxfun": maxfun,
        "ftol": 1e-14,
        "gtol": 1e-12,
        "maxls": 20,
        "disp": False,
    }

    try:
        res = minimize(
            lse_objective,
            u,
            args=(float(beta),),
            method="L-BFGS-B",
            jac=True,
            options=options,
        )
        u2 = np.asarray(res.x, dtype=np.float64)
        p2 = _softmax_vec(u2)
        p2 = np.maximum(p2, 0.0)
        s2 = float(np.sum(p2))
        if not np.isfinite(s2) or s2 <= 0.0:
            p2 = p
        else:
            p2 = p2 / s2
        sc = score_normalized(p2)
        if not np.isfinite(sc):
            p2 = p
            sc = score_normalized(p)
        print(f"[{tag}] L-BFGS done beta={beta:g} score={sc:.9f}")
        return p2, sc
    except Exception as exc:
        print(f"[{tag}] L-BFGS failed: {exc}")
        return p, score_normalized(p)


def resample_seq(a, n_new):
    a = np.asarray(a, dtype=np.float64).ravel()
    n_old = len(a)
    if n_old == 0:
        return np.ones(int(n_new), dtype=np.float64)
    if n_old == int(n_new):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, int(n_new))
    return np.interp(x_new, x_old, a)


def _clean_init(a, L):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size != L:
        if a.size == 1:
            a = np.full(L, float(a[0]), dtype=np.float64)
        else:
            a = np.interp(np.linspace(0.0, 1.0, L), np.linspace(0.0, 1.0, a.size), a)
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-300:
        return None
    return a / s


def make_inits(L, parent, rng):
    L = int(L)
    x = np.linspace(0.0, 1.0, L)
    raw = []

    if parent is not None:
        p = resample_seq(parent, L)
        p = _clean_init(p, L)
        if p is None:
            p = np.ones(L, dtype=np.float64)
        raw.append(p)
        raw.append(p * (1.0 + 0.02 * rng.standard_normal(L)))
        raw.append(p * (1.0 + 0.08 * rng.standard_normal(L)))
        raw.append(0.8 * p + 0.2 / L)
        for f in (1, 2, 3):
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            raw.append(p * (1.0 + 0.25 * np.cos(2.0 * np.pi * f * x + ph)))

    raw.append(np.ones(L, dtype=np.float64))
    raw.append(1.0 + 0.8 * x)
    raw.append(1.0 + 0.8 * (1.0 - x))
    raw.append(0.1 + 0.9 * (1.0 - np.cos(np.pi * x)))
    raw.append(0.1 + 0.9 * (1.0 + np.cos(np.pi * x)))
    raw.append(np.maximum(0.2 + 0.8 * np.cos(2.0 * np.pi * x + float(rng.uniform(0.0, 2.0 * np.pi))), 0.0))
    raw.append(np.maximum(0.2 + 0.5 * np.cos(4.0 * np.pi * x + float(rng.uniform(0.0, 6.0))), 0.0))

    inits = []
    for arr in raw:
        a = _clean_init(arr, L)
        if a is not None:
            inits.append(a)
            if len(inits) >= 14:
                break
    if not inits:
        inits.append(np.full(L, 1.0 / L, dtype=np.float64))
    return inits


def _pick_length_lengths(base_len, max_count=3):
    cands = []
    if base_len is not None and base_len > 0:
        base = int(base_len)
        cands.append(base)
        cands.append(_next_pow2(base))
        cands.append(min(25000, base + 1024))
        cands.append(max(2048, base - 1024))
        if base > 2048:
            cands.append(base // 2)
    else:
        cands = [4096, 8192, 6000]

    out = []
    for Lc in cands:
        Lc = int(Lc)
        if 1024 <= Lc <= 25000:
            skip = False
            for L0 in out:
                if abs(Lc - L0) <= max(128, int(0.05 * max(Lc, L0))):
                    skip = True
                    break
            if not skip:
                out.append(Lc)
            if len(out) >= int(max_count):
                break
    if not out:
        out = [4096]
    return out


def _score_key(entry):
    return entry[0]


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for a nonnegative sequence minimizing the evaluation function.
    """
    t_start = time.monotonic()
    total_budget = float(budget_s)
    reserve = 70.0 if total_budget > 300.0 else 15.0
    deadline = t_start + max(15.0, total_budget - reserve)

    rng = np.random.default_rng(int(seed))

    parent = None
    try:
        if "height_sequence_1" in globals():
            raw = globals().get("height_sequence_1", None)
            if raw is not None:
                p = np.asarray(raw, dtype=np.float64).ravel()
                if p.size > 50:
                    p = np.clip(p, 0.0, 1000.0)
                    s = float(np.sum(p))
                    if np.isfinite(s) and s > 0.01:
                        parent = p / s
    except Exception:
        parent = None

    if parent is not None:
        best_seq = parent.copy()
        best_score = score_normalized(parent)
        print(f"Loaded parent length={len(best_seq)}, parent score={best_score:.9f}")
    else:
        best_seq = np.ones(8192, dtype=np.float64)
        best_score = score_normalized(best_seq)
        print(f"No parent loaded. Starting uniform length={len(best_seq)}, score={best_score:.9f}")

    usable = max(1.0, deadline - t_start)

    # ---------------- Phase 0: multi-length / multi-init short scans -------
    ph0_target = min(180.0, 0.10 * usable)
    ph0_deadline = min(deadline - 120.0, t_start + ph0_target)

    if time.monotonic() < ph0_deadline - 1.0:
        lengths = _pick_length_lengths(len(best_seq), max_count=3)
        print("Phase 0 lengths:", lengths)
        for L in lengths:
            if time.monotonic() > ph0_deadline - 15.0:
                break
            inits = make_inits(L, parent, rng)
            scored = []
            for a in inits[:10]:
                sc = score_normalized(a)
                if np.isfinite(sc):
                    scored.append((sc, a))
            if not scored:
                continue
            scored.sort(key=_score_key)
            primary = scored[0][1]
            secondary = primary
            if len(scored) > 1:
                secondary = scored[int(rng.integers(1, len(scored)))][1]

            sub_dead = min(ph0_deadline, time.monotonic() + 35.0)
            if sub_dead > time.monotonic() + 5.0:
                p, sc = peak_refine(primary, sub_dead, seed=int(seed) + 11 * L,
                                    tag=f"quick L={L}")
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_seq = np.asarray(p, dtype=np.float64).copy()
                    print(f"Quick improved at L={L}: {best_score:.9f}")

            if time.monotonic() < ph0_deadline - 10.0 and secondary is not primary:
                sub_dead = min(ph0_deadline, time.monotonic() + 25.0)
                if sub_dead > time.monotonic() + 5.0:
                    p, sc = peak_refine(secondary, sub_dead, seed=int(seed) + 977 * L,
                                        tag=f"quick2 L={L}")
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_seq = np.asarray(p, dtype=np.float64).copy()
                        print(f"Quick2 improved at L={L}: {best_score:.9f}")

    print(f"After quick phase: best={best_score:.9f}: n={len(best_seq)}: t={time.monotonic()-t_start:.0f}s")

    # ---------------- Phase 1: main long refinement ------------------------
    main_deadline = min(t_start + 0.65 * usable, deadline - max(180.0, 0.30 * usable))
    if time.monotonic() < main_deadline - 20.0:
        p, sc = peak_refine(
            best_seq,
            main_deadline,
            seed=int(seed) + 10001,
            tag="main",
            temp_f_start=0.50,
            temp_f_end=0.040,
            top_frac=0.06,
            mix_start=1.0,
            mix_end=0.30,
            max_delta_start=0.25,
            max_delta_end=0.030,
            lr_start=0.30,
            lr_end=0.08,
            restart_after=12000,
            noise_start=0.05,
            noise_end=0.35,
        )
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.asarray(p, dtype=np.float64).copy()
            print(f"Main improved: {best_score:.9f}")

    # ---------------- Phase 2: L-BFGS polish -------------------------------
    lb_dead = min(deadline - 160.0, time.monotonic() + min(240.0, 0.10 * usable))
    if time.monotonic() < lb_dead - 5.0:
        p, sc = lbfgs_refine(best_seq, lb_dead, beta=20000.0, tag="lbfgs1")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.asarray(p, dtype=np.float64).copy()
            print(f"L-BFGS1 improved: {best_score:.9f}")

    # ---------------- Phase 3: harder peak-oriented refinement -------------
    hard_dead = min(deadline - 90.0, time.monotonic() + min(480.0, 0.17 * usable))
    if time.monotonic() < hard_dead - 15.0:
        p, sc = peak_refine(
            best_seq,
            hard_dead,
            seed=int(seed) + 20002,
            tag="hard",
            temp_f_start=0.12,
            temp_f_end=0.015,
            top_frac=0.02,
            mix_start=0.70,
            mix_end=0.0,
            max_delta_start=0.20,
            max_delta_end=0.020,
            lr_start=0.28,
            lr_end=0.05,
            restart_after=8000,
            noise_start=0.05,
            noise_end=0.30,
        )
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.asarray(p, dtype=np.float64).copy()
            print(f"Hard improved: {best_score:.9f}")

    # ---------------- Phase 4: final L-BFGS with larger beta ---------------
    final_lb_dead = min(deadline - 25.0, time.monotonic() + min(150.0, 0.06 * usable))
    if time.monotonic() < final_lb_dead - 5.0:
        p, sc = lbfgs_refine(best_seq, final_lb_dead, beta=60000.0, tag="lbfgs2")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.asarray(p, dtype=np.float64).copy()
            print(f"L-BFGS2 improved: {best_score:.9f}")

    # ---------------- Final tiny peak_refine if time remains --------------
    final_peak_dead = deadline - 5.0
    if time.monotonic() < final_peak_dead - 10.0:
        p, sc = peak_refine(
            best_seq,
            final_peak_dead,
            seed=int(seed) + 30003,
            tag="final",
            temp_f_start=0.08,
            temp_f_end=0.012,
            top_frac=0.015,
            mix_start=0.60,
            mix_end=0.0,
            max_delta_start=0.15,
            max_delta_end=0.012,
            lr_start=0.20,
            lr_end=0.04,
            restart_after=6000,
            noise_start=0.03,
            noise_end=0.25,
        )
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = np.asarray(p, dtype=np.float64).copy()
            print(f"Final peak improved: {best_score:.9f}")

    # ---------------- Prepare and verify output ---------------------------
    out = np.asarray(best_seq, dtype=np.float64).ravel()
    out = np.nan_to_num(out, nan=0.0, posinf=1000.0, neginf=0.0)
    out = np.clip(out, 0.0, 1000.0)

    s = float(np.sum(out))
    if not np.isfinite(s) or s < 0.01:
        out = np.ones(1024, dtype=np.float64)
        s = float(out.size)
    out = out / s

    out_list = [float(x) for x in out]

    tried_verify = False
    try:
        verified = float(evaluate_sequence(out_list))
        tried_verify = True
        print(f"internal score={best_score:.9f}, verified score={verified:.9f}, length={len(out_list)}")
    except Exception as exc:
        print(f"Verification call failed: {exc}")
        print(f"internal score={best_score:.9f}, length={len(out_list)}")

    if tried_verify and not np.isfinite(verified):
        # Very defensive fallback: uniform sequence is always valid.
        out_list = [1.0 / 1024] * 1024
        return out_list

    return out_list
```
