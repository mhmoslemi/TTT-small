    L = c_max + np.log(exp_s.sum()) / beta

    # Compute (a ⊛ p)[i] = sum_{m=0}^{n-1} a[m] * p[i+m]
    # Using: (a * p_rev)[2n-2-i] = sum_j a[j]*p_rev[2n-2-i-j] = sum_j a[j]*p[i+j]
    p_rev = p[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    A_fft = np.fft.rfft(a, M2_fft)
    PR_fft = np.fft.rfft(p_rev, M2_fft)
    conv_ap = np.fft.irfft(A_fft * PR_fft, M2_fft)

    # Extract r[i] = conv_ap[2n-2-i] for i = 0, ..., n-1
    indices = 2 * n - 2 - np.arange(n)
    r = conv_ap[indices]

    dL_da = 2.0 * r
    grad = (4.0 * n / (S * S)) * r - (4.0 * n * L) / (S * S * S)
    return grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        sc, p = lse_score_and_p(x, beta)
        g = lse_gradient(x, beta)
        return sc, g

    res = minimize(
        obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def lp_direction(a, delta=0.1):
    """
    Find optimal direction d via linearized LP:
    minimize T subject to 2*(a*d)[k] - T <= -M for all k,
    sum(d) = 0, -min(a[i],delta) <= d[i] <= delta.
    Returns the direction d (can be added to a).
    """
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.zeros(n)
    a_norm = a / S  # work with normalized

    c = fast_conv(a_norm)
    M = float(c.max())

    # Build constraint matrix A_ub: (2n-1) x (n+1)
    # Row k, col j: 2 * a_norm[k-j] if 0 <= k-j <= n-1, else 0
    # Row k, col n: -1
    # This is a large matrix; build sparsely
    rows = []
    cols = []
    vals = []

    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            js = np.arange(j_start, j_end + 1)
            ks = np.full(len(js), k)
            vals_k = 2.0 * a_norm[k - js]
            rows.extend(ks.tolist())
            cols.extend(js.tolist())
            vals.extend(vals_k.tolist())
        # T column
        rows.append(k)
        cols.append(n)
        vals.append(-1.0)

    from scipy.sparse import csr_matrix
    A_ub = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n + 1))
    b_ub = np.full(2 * n - 1, -M)

    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])

    # Objective: minimize T
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    # Bounds
    lo = np.maximum(-a_norm, -delta)
    hi = np.full(n, delta)
    bounds = list(zip(lo, hi))
    bounds.append((0.0, None))  # T >= 0

    res = linprog(
        c_obj, A_ub=A_ub, b_ub=b_ub,
        A_eq=A_eq, b_eq=b_eq,
        bounds=bounds, method='highs'
    )

    if res.success:
        d = res.x[:n]
        # Normalize direction
        d = d - d.mean()  # ensure sum = 0 (numerical)
        return d
    return np.zeros(n)


def error_reduction(a, n_iters=200):
    """Frequency-domain error reduction to flatten convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        v = A * A
        conv = np.fft.irfft(v, M)[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-16:
            break

        alpha = 0.5 + 0.4 * (it / max(1, n_iters))
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-16:
            break

        conv_mod = conv - 0.6 * excess
        err = conv_mod - conv

        err_pad = np.zeros(M, dtype=np.float64)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_pad, M)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-10] = 0.0

        damp = 0.3 + 0.3 * (it / max(1, n_iters))
        A_new = A + damp * d

        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def peak_shaving(a, n_rounds=50):
    """Iteratively shave the peak of the convolution."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for rnd in range(n_rounds):
        c = fast_conv(a)
        M = float(c.max())
        k_star = int(np.argmax(c))

        # Find contributors to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg * seg[::-1]  # a[j]*a[k-j] for j in range
        # Actually contrib[j_local] = a[j_start+j_local] * a[k_star-(j_start+j_local)]
        # = a[j] * a[k_star - j]

        total_contrib = contrib.sum()
        if total_contrib < 1e-15:
            break

        # Reduce top contributors proportionally
        sort_idx = np.argsort(contrib)[::-1]
        n_top = max(5, n // 20)
        top_idx = sort_idx[:min(n_top, len(sort_idx))]

        reduction = 0.01 + 0.02 * (rnd / n_rounds)
        for li in top_idx:
            gi = j_start + li
            old = a[gi]
            a[gi] = max(0.0, old * (1.0 - reduction))

        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        else:
            # Revert if it got worse
            a = best_a.copy()

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            inits.append(np.interp(x, x_old, parent).copy())

    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x) * np.cos(4.0 * np.pi * x))
    inits.append(0.3 + 0.5 * (2.0 * x - 1.0) ** 2)
    inits.append(0.3 + 0.5 * np.abs(2.0 * x - 1.0))
    inits.append(0.5 + 0.3 * np.cos(np.pi * x))
    inits.append(0.4 + 0.6 * x)  # linear ramp
    inits.append(0.4 + 0.6 * (1.0 - x))  # reverse ramp
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0))
    inits.append(np.clip(0.5 + 0.2 * np.sin(6.0 * np.pi * x + rng.random() * 6.28), 0.01, 10.0))

    for a in inits:
        a = np.maximum(a, 0.01)
        inits[-1] = a
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses LP-based direction finding, LSE gradient descent, and error reduction.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 20.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = [6962, 4096, 8192, 10240]
    if parent is not None:
        lengths.insert(0, len(parent))
    lengths = sorted(set(lengths))
    print(f"Lengths: {lengths}, parent available: {parent is not None}")

    # ============================================================
    # Phase 1: LSE optimization from diverse inits (30% time)
    # ============================================================
    phase1_deadline = t_start + 0.30 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 1: LSE from diverse inits ---")

    for n in lengths[:3]:
        if time.monotonic() > phase1_deadline:
            break
        print(f"  n={n}")
        inits = build_initializations(n, rng, parent)
        for i, a0 in enumerate(inits[:6]):
            if time.monotonic() > phase1_deadline:
                break
            a0 = np.clip(a0, 0.0, 100.0)
            sc, a_opt = run_lbfgs(a0, beta=50, maxiter=3000, maxfun=50000)
            report(sc, a_opt, f"LSE50 n={n} i={i}")
            if time.monotonic() > phase1_deadline:
                break
            sc, a_opt = run_lbfgs(a_opt, beta=500, maxiter=3000, maxfun=50000)
            report(sc, a_opt, f"LSE500 n={n} i={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ============================================================
    # Phase 2: LP-based direction finding (30% time)
    # ============================================================
    phase2_deadline = t_start + 0.60 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 2: LP Direction Finding ---")

    n_best = len(best_a)
    print(f"  Working with n={n_best}")

    a_work = best_a.copy()
    a_work = a_work / a_work.sum()

    # First do some LSE refinement to get close to optimum
    for beta in [200, 1000]:
        if time.monotonic() > phase2_deadline:
            break
        sc, a_opt = run_lbfgs(a_work, beta, maxiter=5000, maxfun=80000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            a_work = a_opt.copy() / a_opt.sum()
            print(f"  [LSE pre-LP b={beta}] {sc:.8f}")

    # Now do LP iterations
    a_lp = best_a.copy()
    a_lp = a_lp / a_lp.sum()
    n_lp = len(a_lp)

    # Use a smaller n for LP (it's expensive)
    if n_lp > 3000:
        n_lp_small = 3000
        a_lp_small = np.interp(
            np.linspace(0, 1, n_lp_small),
            np.linspace(0, 1, n_lp),
            a_lp
        )
    else:
        n_lp_small = n_lp
        a_lp_small = a_lp.copy()

    print(f"  LP with n={n_lp_small}")
    n_lp_iters = 0
    while time.monotonic() < phase2_deadline:
        t_lp_start = time.monotonic()
        d = lp_direction(a_lp_small, delta=0.05)
        t_lp = time.monotonic() - t_lp_start
        n_lp_iters += 1

        if np.all(np.abs(d) < 1e-10):
            print("  LP: zero direction, stopping")
            break

        # Line search
        a_try = a_lp_small + d
        a_try = np.maximum(a_try, 0.0)
        S = a_try.sum()
        if S < 1e-12:
            break
        a_try = a_try / S

        sc_try = compute_score(a_try)
        sc_cur = compute_score(a_lp_small)
        if sc_try < sc_cur:
            a_lp_small = a_try
            # Upscale to full length and report
            a_full = np.interp(
                np.linspace(0, 1, n_lp),
                np.linspace(0, 1, n_lp_small),
                a_lp_small
            )
            sc_full = compute_score(a_full)
            report(sc_full, a_full, f"LP iter={n_lp_iters}")
        else:
            # Try half step
            a_try = a_lp_small + 0.5 * d
            a_try = np.maximum(a_try, 0.0)
            S = a_try.sum()
            if S < 1e-12:
                break
            a_try = a_try / S
            sc_try = compute_score(a_try)
            if sc_try < sc_cur:
                a_lp_small = a_try
                a_full = np.interp(
                    np.linspace(0, 1, n_lp),
                    np.linspace(0, 1, n_lp_small),
                    a_lp_small
                )
                sc_full = compute_score(a_full)
                report(sc_full, a_full, f"LP iter={n_lp_iters} half")
            else:
                print(f"  LP iter {n_lp_iters}: no improvement, stopping")
                break

        print(f"  LP iter {n_lp_iters}: t={t_lp:.1f}s, score={sc_try:.8f}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: High-beta LSE refinement (25% time)
    # ============================================================
    phase3_deadline = t_start + 0.85 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 3: High-beta LSE ---")

    for beta in [3000, 10000, 30000]:
        if time.monotonic() > phase3_deadline:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=100000)
        report(sc, a_opt, f"LSE b={beta}")

    # Error reduction
    if time.monotonic() < phase3_deadline:
        a_er = error_reduction(best_a, n_iters=300)
        sc = compute_score(a_er)
        report(sc, a_er, "ER phase3")

        # LSE after ER
        if time.monotonic() < phase3_deadline:
            sc, a_opt = run_lbfgs(a_er, 5000, maxiter=5000, maxfun=100000)
            report(sc, a_opt, "ER+LSE5000")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ============================================================
    # Phase 4: Final polish and diverse restarts (15% time)
    # ============================================================
    print(f"\n--- Phase 4: Final Polish ---")

    remaining = deadline - time.monotonic()

    if remaining > 60:
        # Peak shaving
        a_ps = peak_shaving(best_a, n_rounds=100)
        sc = compute_score(a_ps)
        report(sc, a_ps, "peak_shave")

    if remaining > 60:
        # Sinusoidal perturbation + LSE
        n_b = len(best_a)
        for freq in [1, 2, 3, 5, 7, 11]:
            if remaining < 30:
                break
            x = np.arange(n_b, dtype=np.float64) / n_b
            for phase_val in [0, np.pi/2, np.pi]:
                if remaining < 20:
                    break
                amp = 0.02 + 0.02 * rng.random()
                a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase_val))
                a_pert = np.clip(a_pert, 0.0, 1000.0)
                if a_pert.sum() < 1e-12:
                    continue
                sc, a_opt = run_lbfgs(a_pert, 2000, maxiter=3000, maxfun=50000)
                report(sc, a_opt, f"pert f={freq} p={phase_val:.2f}")

    if remaining > 30:
        # Final high-beta
        sc, a_opt = run_lbfgs(best_a, 50000, maxiter=3000, maxfun=60000)
        report(sc, a_opt, "final LSE b=50000")

    if remaining > 20:
        # Final error reduction
        a_er = error_reduction(best_a, n_iters=200)
        sc = compute_score(a_er)
        report(sc, a_er, "final ER")
        if remaining > 15:
            sc, a_opt = run_lbfgs(a_er, 20000, maxiter=3000, maxfun=50000)
            report(sc, a_opt, "final ER+LSE")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
