Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 using p-norm smoothing,
multi-peak coordinate descent, and simulated annealing.
"""
import numpy as np
from scipy.optimize import minimize
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 using p-norm smoothing
with adaptive scheduling, multi-peak coordinate descent, frequency-domain
error reduction, and simulated annealing restarts.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def pnorm_obj_grad(a, p):
    """p-norm smoothed objective and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_pos = np.maximum(c, 0.0)
    if p == 2:
        cp = c_pos.copy()
        sum_cp = float(cp.sum())
        if sum_cp < 1e-30:
            return 1e10, np.zeros(n)
        L = np.sqrt(sum_cp)
    else:
        cp = c_pos ** (p - 1)
        sum_cp = float((cp * c_pos).sum())
        if sum_cp < 1e-30:
            return 1e10, np.zeros(n)
        L = sum_cp ** (1.0 / p)
    score = 2.0 * n * L / (S * S)
    W = sum_cp ** ((1.0 - p) / p)
    dL_dc = cp * W
    rev_a = a[::-1]
    M2 = len(dL_dc) + n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    D = np.fft.rfft(dL_dc, M2_fft)
    dL_da = np.fft.irfft(RA * D, M2_fft)[n - 1:2 * n - 1]
    grad = 2.0 * n * (dL_da - 2.0 * L / S) / (S * S)
    return score, grad


def run_lbfgs_pnorm(a0, p, maxiter=3000, maxfun=50000):
    """Run L-BFGS-B on p-norm smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        s, g = pnorm_obj_grad(x, p)
        return s, g

    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-18, 'gtol': 1e-14,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out


def error_reduction(a, n_iters=150):
    """Frequency-domain error reduction to flatten convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        conv = np.fft.irfft(A * A, M)[:2 * n - 1]

        conv_max = float(conv.max())
        conv_mean = float(conv.mean())
        if conv_max - conv_mean < 1e-16:
            break

        alpha = 0.5 + 0.45 * (it / max(1, n_iters - 1))
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-16:
            break

        conv_mod = conv - 0.6 * excess
        err = conv_mod - conv
        err_full = np.zeros(M, dtype=np.float64)
        err_full[:2 * n - 1] = err
        err_spec = np.fft.rfft(err_full, M)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-10] = 0.0

        damp = 0.3 + 0.5 * (it / max(1, n_iters))
        A_new = A + damp * d
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def multi_peak_coord_descent(a, n_sweeps=8, n_peaks=5, n_top=50):
    """Coordinate descent targeting multiple peaks simultaneously."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        n_p = min(n_peaks, len(c))
        peak_idx = np.argsort(c)[::-1][:n_p]
        improved = False

        for k_star in peak_idx:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end <= j_start:
                continue

            seg = a[j_start:j_end + 1]
            contrib = seg * seg[::-1]

            top_count = min(n_top, len(contrib))
            top_local = np.argsort(contrib)[::-1][:top_count]
            top_global = j_start + top_local

            for j in top_global:
                a_old_j = a[j]
                if a_old_j < 1e-10:
                    continue
                best_val = a_old_j
                cur_sc = compute_score(a)
                for factor in [0.6, 0.75, 0.85, 1.1, 1.25]:
                    a[j] = max(0.0, a_old_j * factor)
                    S = a.sum()
                    if S < 1e-12:
                        a[j] = a_old_j
                        continue
                    a_tmp = a / S
                    sc = compute_score(a_tmp)
                    if sc < cur_sc:
                        cur_sc = sc
                        best_val = a_old_j * factor
                a[j] = best_val
                S = a.sum()
                if S > 1e-12:
                    a = a / S
                    improved = True

        if not improved:
            break
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def natural_subgradient(a, n_iters=2000):
    """Natural gradient subgradient descent with momentum."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    v = np.zeros(n)
    momentum = 0.9
    sc_lb = 1.0

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]

        grad = 2.0 * n * (sg - 2.0 * M / S) / (S * S)
        grad = grad - grad.mean()
        gn = float(np.linalg.norm(grad))
        if gn < 1e-18:
            break

        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.03)
        step = max(step, 1e-7)

        v = momentum * v + step * grad
        a = a - v
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def lse_obj_grad(a, beta):
    """Log-sum-exp smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_pos = np.maximum(c, 0.0)
    c_max = float(c_pos.max())
    scaled = beta * (c_pos - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum()) + scaled.max() / beta
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs_lse(a0, beta, maxiter=3000, maxfun=50000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        s, g = lse_obj_grad(x, beta)
        return s, g

    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-18, 'gtol': 1e-14,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out

            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.cos(4.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(0.3 + 0.7 * (1.0 - (2.0 * x - 1.0) ** 2))
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    # Spectrally motivated init
    inits.append(0.5 + 0.2 * np.sin(2.0 * np.pi * x) + 0.1 * np.sin(6.0 * np.pi * x))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses p-norm L-BFGS, frequency-domain error reduction, multi-peak
    coordinate descent, natural gradient, and simulated annealing.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 20.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n in [8192, 10240, 12288]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))[:4]

    print(f"Lengths to try: {lengths}")
    print(f"Parent: {parent is not None}, len={0 if parent is None else len(parent)}")

    # ==================================================================
    # Phase 1: p-norm L-BFGS with adaptive p schedule (35% time)
    # ==================================================================
    phase1_deadline = t_start + 0.35 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 1: p-norm L-BFGS ---")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"  n={n}")
        inits = build_initializations(n, rng, parent)
        remaining = phase1_deadline - time.monotonic()
        max_inits = max(2, min(len(inits), int(remaining / 30.0)))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a_work = a0.copy()
            S = a_work.sum()
            if S > 0:
                a_work = a_work / S

            p_schedule = [4, 8, 16, 32, 64, 128]
            for p in p_schedule:
                if time.monotonic() > phase1_deadline:
                    break
                sc, a_opt = run_lbfgs_pnorm(a_work, p, maxiter=2000, maxfun=30000)
                report(sc, a_opt, f"p={p} n={n} i={i}")
                a_work = best_a.copy() if len(best_a) == n else a_opt

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: LSE L-BFGS refinement (20% time)
    # ==================================================================
    phase2_deadline = t_start + 0.55 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 2: LSE Refinement ---")

    for beta in [100, 300, 1000, 3000, 8000]:
        if time.monotonic() > phase2_deadline:
            break
        sc, a_opt = run_lbfgs_lse(best_a, beta, maxiter=3000, maxfun=50000)
        report(sc, a_opt, f"LSE b={beta}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Error reduction + coordinate descent (20% time)
    # ==================================================================
    phase3_deadline = t_start + 0.75 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 3: ER + Coord Descent ---")

    n_best = len(best_a)
    for trial in range(10):
        if time.monotonic() > phase3_deadline:
            break

        if trial % 3 == 0:
            a_work = error_reduction(best_a, n_iters=150)
            sc = compute_score(a_work)
            report(sc, a_work, f"ER#{trial}")
        elif trial % 3 == 1:
            a_work = multi_peak_coord_descent(best_a, n_sweeps=10, n_peaks=5, n_top=50)
            sc = compute_score(a_work)
            report(sc, a_work, f"CD#{trial}")
        else:
            # Combine: ER then CD
            a_er = error_reduction(best_a, n_iters=80)
            a_cd = multi_peak_coord_descent(a_er, n_sweeps=5, n_peaks=3, n_top=30)
            sc = compute_score(a_cd)
            report(sc, a_cd, f"ER+CD#{trial}")

            # Then L-BFGS polish on the result
            if time.monotonic() < phase3_deadline - 10:
                sc2, a_opt = run_lbfgs_pnorm(a_cd, 256, maxiter=2000, maxfun=30000)
                report(sc2, a_opt, f"ER+CD+LBFGS#{trial}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Simulated annealing restarts (15% time)
    # ==================================================================
    phase4_deadline = t_start + 0.90 * (float(budget_s) - 20.0)
    print(f"\n--- Phase 4: Annealing Restarts ---")

    n_best = len(best_a)
    n_restarts = 15
    for r in range(n_restarts):
        if time.monotonic() > phase4_deadline:
            break

        strategy = r % 5
        if strategy == 0:
            scale = 0.02 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Targeted peak reduction perturbation
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            a_pert = best_a.copy()
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            seg = a_pert[j_start:j_end + 1]
            contrib = seg * seg[::-1]
            top_count = max(10, n_best // 100)
            top_local = np.argsort(contrib)[::-1][:top_count]
            for li in top_local:
                gi = j_start + li
                a_pert[gi] *= 0.9
            S = a_pert.sum()
            if S > 1e-12:
                a_pert = a_pert / S
        elif strategy == 2:
            # Sinusoidal perturbation
            freq = 1.0 + 8.0 * rng.random()
            phase_ph = 2.0 * np.pi * rng.random()
            amp = 0.02 + 0.04 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase_ph))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Block rescale
            a_pert = best_a.copy()
            n_blocks = 4 + int(rng.integers(0, 8))
            block_size = max(1, n_best // n_blocks)
            for b in range(min(n_blocks, n_best // block_size)):
                s = b * block_size
                e = min(s + block_size, n_best)
                factor = 0.7 + 0.6 * rng.random()
                a_pert[s:e] *= factor
            S = a_pert.sum()
            if S > 1e-12:
                a_pert = a_pert / S
        else:
            # High-freq perturbation (targets rapid oscillations in a)
            a_pert = best_a.copy()
            n_hf = max(5, n_best // 20)
            idx = rng.choice(n_best, size=n_hf, replace=False)
            a_pert[idx] *= (0.5 + rng.random(n_hf))
            S = a_pert.sum()
            if S > 1e-12:
                a_pert = a_pert / S

        # L-BFGS refinement from perturbed point
        p_val = 128 + 128 * (r % 3)
        if time.monotonic() < phase4_deadline - 5:
            sc, a_opt = run_lbfgs_pnorm(a_pert, p_val, maxiter=2000, maxfun=30000)
            report(sc, a_opt, f"anneal#{r} s={strategy} p={p_val}")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Final polish (5% time)
    # ==================================================================
    print(f"\n--- Phase 5: Final Polish ---")
    remaining = deadline - time.monotonic()

    if remaining > 60:
        a_er = error_reduction(best_a, n_iters=200)
        sc = compute_score(a_er)
        report(sc, a_er, "final ER")

    if remaining > 40:
        a_cd = multi_peak_coord_descent(best_a, n_sweeps=15, n_peaks=7, n_top=100)
        sc = compute_score(a_cd)
        report(sc, a_cd, "final CD")

    if remaining > 40:
        sc, a_opt = run_lbfgs_pnorm(best_a, 512, maxiter=5000, maxfun=80000)
        report(sc, a_opt, "final p=512")

    if remaining > 30:
        sc, a_opt = run_lbfgs_lse(best_a, 10000, maxiter=5000, maxfun=80000)
        report(sc, a_opt, "final LSE b=10000")

    if remaining > 30:
        a_ng = natural_subgradient(best_a, n_iters=int(min(3000, remaining / 0.05)))
        sc = compute_score(a_ng)
        report(sc, a_ng, "final NG")

    if remaining > 20 and best_score > 1.505:
        # One more aggressive round
        a_er2 = error_reduction(best_a, n_iters=100)
        a_cd2 = multi_peak_coord_descent(a_er2, n_sweeps=10, n_peaks=5, n_top=80)
        sc2, a_opt2 = run_lbfgs_pnorm(a_cd2, 1024, maxiter=3000, maxfun=50000)
        report(sc2, a_opt2, "aggressive")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
