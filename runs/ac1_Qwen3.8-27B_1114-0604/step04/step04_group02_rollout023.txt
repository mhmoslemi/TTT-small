    total_nz = int(lengths.sum())
    
    # Build row and column indices
    row_idx = np.repeat(js, lengths)
    
    # Column indices: for row j, columns are ks[j], ks[j]+1, ..., ke[j]
    # We can build this as: col_idx[i] = ks[j] + offset where offset goes from 0 to lengths[j]-1
    offsets = np.arange(total_nz)

    # Data: for each (j, k), the value is a[k]
    data = a[col_idx]
    
    from scipy.sparse import coo_matrix
    return coo_matrix((data, (row_idx, col_idx)), shape=(n_rows, n)).tocsr()

"""
Convex optimization approach for step function peak minimization.
Uses LP-based active set method combined with LSE smoothing for polishing.
The problem is convex (minimize max of convex quadratics subject to simplex constraint),
so we exploit this structure for guaranteed convergence to the global optimum.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / sum^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -100))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient computation
    # d/d a_k [2n * L / S^2] = 2n * (dL/da_k / S^2 - 2L * da_k/S / S^2)
    # dL/da_k = sum_j p_j * d(c_j)/da_k = sum_j p_j * 2 * a_{j-k} (for valid indices)
    # = 2 * (a reversed) convolved with p, evaluated at appropriate indices
    
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL = 2.0 * conv_res
    
    S2 = S * S
    grad = 2.0 * n * (dL / S2 - 2.0 * L / (S2 * S))
    return score, grad


def lse_objective(x, beta):
    """Objective function for scipy minimize."""
    val, grad = lse_score_and_grad(x, beta)
    return val, grad


def run_lbfgs(a0, beta, maxiter=30000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def build_conv_constraint(a, top_k=None):
    """
    Build sparse constraint matrix for (a*g)_j <= M.
    Returns (A_sparse, rhs_indices) where A_sparse @ g <= M gives the constraints.
    """
    n = len(a)
    n_rows_full = 2 * n - 1
    c = fast_conv(a)
    
    if top_k is not None and top_k < n_rows_full:
        top_idx = np.argsort(c)[::-1][:top_k]
    else:
        top_idx = np.arange(n_rows_full)
    
    n_rows = len(top_idx)
    
    # Vectorized construction
    js = top_idx
    ks_start = np.maximum(0, js - n + 1)
    ks_end = np.minimum(js, n - 1)
    lengths = ks_end - ks_start + 1
    total_nz = int(lengths.sum())
    
    row_idx = np.repeat(js, lengths)
    cum_lengths = np.concatenate([[0], np.cumsum(lengths)[:-1]])
    offsets = np.arange(total_nz) - cum_lengths[np.repeat(np.arange(n_rows), lengths)]
    col_idx = ks_start[np.repeat(np.arange(n_rows), lengths)] + offsets
    data = a[col_idx]
    
    from scipy.sparse import coo_matrix, csr_matrix
    A = coo_matrix((data, (row_idx, col_idx)), shape=(n_rows, n)).tocsr()
    return A, top_idx


def lp_active_set(a, max_iters=50, top_k_init=20, max_k=500, solver='PDLP'):
    """
    LP-based active set method.
    Iteratively solves: max sum(g) s.t. (a*g)_j <= M for active j, g >= 0.
    Adds violated constraints until convergence.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    a = a / S  # Normalize
    best_a = a.copy()
    best_sc = compute_score(a)
    
    c = fast_conv(a)
    M = float(c.max())
    
    # Start with top k constraints
    k = top_k_init
    active_set = np.argsort(c)[::-1][:k]
    
    for it in range(max_iters):
        # Build constraint matrix for active set
        js = active_set
        ks_start = np.maximum(0, js - n + 1)
        ks_end = np.minimum(js, n - 1)
        lengths = ks_end - ks_start + 1
        total_nz = int(lengths.sum())
        
        if total_nz > 50_000_000:
            # Too large, reduce active set
            k = max(10, k // 2)
            active_set = np.argsort(c)[::-1][:k]
            continue
        
        row_idx = np.repeat(js, lengths)
        rows_ar = np.repeat(np.arange(len(js)), lengths)
        cum_lengths = np.concatenate([[0], np.cumsum(lengths)[:-1]])
        offsets = np.arange(total_nz) - cum_lengths[rows_ar]
        col_idx = ks_start[rows_ar] + offsets
        data = a[col_idx]
        
        from scipy.sparse import coo_matrix, csr_matrix
        A_mat = coo_matrix((data, (row_idx, col_idx)), shape=(len(js), n)).tocsr()
        
        # Solve LP
        g_var = cp.Variable(n, non_negative=True)
        obj = cp.Maximize(cp.sum(g_var))
        cons = [A_mat @ g_var <= M]
        prob = cp.Problem(obj, cons)
        
        t0 = time.monotonic()
        try:
            prob.solve(solver=cp.PDLP, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.GLOP, verbose=False)
            except Exception:
                print(f"  LP solve failed at iter {it}")
                break
        t1 = time.monotonic()
        
        if g_var.value is None:
            break
        
        g_val = np.maximum(0, np.array(g_var.value).ravel())
        S_g = float(g_val.sum())
        if S_g < 1e-12:
            break
        
        # Rescale g to match sum of a
        g_rescaled = g_val * S / S_g
        
        # Check violations: compute (a * g_rescaled)_j for all j
        # Actually, we should check (a * g_val / S_g)_j <= 1 (normalized)
        # Or equivalently (a * g_rescaled)_j <= M
        c_cross = np.fft.irfft(
            np.fft.rfft(a, 2*n-1) * np.fft.rfft(g_rescaled, 2*n-1),
            2*n-1
        )[:2*n-1]
        
        # Find violations
        violations = c_cross > M * (1 + 1e-6)
        if np.sum(violations) == 0:
            # No violations, this is the optimal active set
            a_new = g_rescaled / S  # Normalize
            sc = compute_score(a_new)
            if sc < best_sc:
                best_sc = sc
                best_a = a_new.copy()
            a = a_new
            c = fast_conv(a)
            M = float(c.max())
            break
        
        # Add violated constraints to active set
        viol_idx = np.where(violations)[0]
        active_set = np.unique(np.concatenate([active_set, viol_idx]))
        active_set = np.argsort(c_cross[active_set])[::-1][:min(len(active_set), max_k)]
        k = len(active_set)
        
        # Update a with the LP solution
        a_new = g_rescaled / S
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        a = a_new
        c = fast_conv(a)
        M = float(c.max())
        
        print(f"  LP iter {it}: M={M:.8f}, k={k}, viol={len(viol_idx)}, time={t1-t0:.1f}s, score={best_sc:.8f}")
    
    return best_a, best_sc


def nesterov_subgrad(a, n_iters=3000, f_lb=1.0):
    """Nesterov-accelerated subgradient descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]
        
        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.3)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new
        
        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()
    
    return best_a


def multiplicative_weights(a, n_iters=5000, f_lb=1.0):
    """Mirror descent with entropy proximal on the normalized problem."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    a = a / S
    a = np.maximum(a, 1e-10)  # Ensure strictly positive for log
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    eta = 1.0 / np.sqrt(n_iters * n)
    
    for it in range(n_iters):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        Sc = 2.0 * n * Mak  # score when sum = 1
        
        if Sc < best_sc:
            best_sc = Sc
            best_a = a.copy()
        
        # Subgradient of max_j (a*a)_j at the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg[idx] = 2.0 * a[k_star - idx]
        
        # Mirror descent update: a_new = a * exp(-eta * sg) / Z
        log_a = np.log(a)
        log_a_new = log_a - eta * sg
        log_a_new -= np.max(log_a_new)  # For numerical stability
        a_new = np.exp(log_a_new)
        a_new /= a_new.sum()
        
        a = a_new
    
    return best_a


def perturb_and_optimize(a, rng, scale=0.03, beta=500, maxiter=10000):
    """Perturb a and run L-BFGS."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    strategy = rng.integers(0, 4)
    
    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 8))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // max(1,block_size))):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    else:
        # Targeted: reduce peak contributors
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(5, n // 50)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.1 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            a_p = a_p / S
    
    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    
    sc, a_opt = run_lbfgs(a_p, beta, maxiter=maxiter)
    return sc, a_opt

        rows_ar = np.repeat(np.arange(len(js)), lengths)

        offsets = np.arange(total_nz) - cum_lengths[rows_ar]
        col_idx = ks_start[rows_ar] + offsets

        A_mat = coo_matrix((data, (row_idx, col_idx)), shape=(len(js), n)).tocsr()

```python
"""
Convex optimization for step function peak minimization.
Combines LP-based active set method with LSE smoothing and iterated local search.
Exploits the convexity of the problem for guaranteed progress.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
from scipy.sparse import coo_matrix, csr_matrix


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Linear cross-convolution via real FFT."""
    n = len(a)
    m = len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, M_fft) * np.fft.rfft(b, M_fft), M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / sum^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_and_grad(x, beta):
    """LSE-smoothed score and gradient for scipy minimize."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -100))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)
    
    rev_x = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RX = np.fft.rfft(rev_x, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RX * P, M2_fft)[n - 1:2 * n - 1]
    dL = 2.0 * conv_res
    
    S2 = S * S
    grad = 2.0 * n * (dL / S2 - 2.0 * L / (S2 * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=30000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-12:
        a_out = np.clip(a0, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def nesterov_subgrad(a, n_iters=3000, f_lb=1.0):
    """Nesterov-accelerated subgradient descent on normalized problem."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]
        
        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.3)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new
        
        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()
    
    return best_a


def solve_lp_for_g(a, active_idx, M, n, solver='PDLP'):
    """
    Solve LP: max sum(g) s.t. (a*g)_j <= M for j in active_idx, g >= 0.
    Returns g or None if infeasible/failed.
    """
    js = active_idx
    n_constr = len(js)
    
    ks_start = np.maximum(0, js - n + 1)
    ks_end = np.minimum(js, n - 1)
    lengths = ks_end - ks_start + 1
    total_nz = int(lengths.sum())
    
    if total_nz == 0:
        return None
    
    rows_ar = np.repeat(np.arange(n_constr), lengths)
    row_idx = np.repeat(js, lengths)
    cum_lengths = np.concatenate([[0], np.cumsum(lengths)[:-1]])
    offsets = np.arange(total_nz) - cum_lengths[rows_ar]
    col_idx = ks_start[rows_ar] + offsets
    data = a[col_idx]
    
    A_mat = coo_matrix((data, (row_idx, col_idx)), shape=(n_constr, n)).tocsr()
    
    g_var = cp.Variable(n, non_negative=True)
    obj = cp.Maximize(cp.sum(g_var))
    cons = [A_mat @ g_var <= M]
    prob = cp.Problem(obj, cons)
    
    try:
        prob.solve(solver=solver, verbose=False)
    except Exception:
        try:
            prob.solve(solver=cp.GLOP, verbose=False)
        except Exception:
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
            except Exception:
                return None
    
    if g_var.value is None:
        return None
    
    g_val = np.maximum(0, np.array(g_var.value).ravel())
    return g_val


def lp_active_set(a, max_iters=50, top_k_init=20, max_k=300, time_limit=600):
    """
    LP-based active set method for minimizing max self-convolution peak.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    c = fast_conv(a)
    M = float(c.max())
    
    active_set = np.argsort(c)[::-1][:top_k_init]
    t_start = time.monotonic()
    
    for it in range(max_iters):
        if time.monotonic() - t_start > time_limit:
            break
        
        g_val = solve_lp_for_g(a, active_set, M, n)
        if g_val is None:
            print(f"  LP solve failed at iter {it}")
            break
        
        S_g = float(g_val.sum())
        if S_g < 1e-12:
            break
        
        # Rescale g to same sum as a
        g_rescaled = g_val * S / S_g
        
        # Check violations
        c_cross = cross_conv(a, g_rescaled)
        violations = c_cross > M * (1 + 1e-6)
        n_viol = int(np.sum(violations))
        
        # Try different step sizes
        best_t = 1.0
        best_step_sc = np.inf
        for t in [1.0, 0.7, 0.5, 0.3]:
            a_try = (1 - t) * a + t * (g_rescaled / S)
            sc_try = compute_score(a_try)
            if sc_try < best_step_sc:
                best_step_sc = sc_try
                best_t = t
        
        a_new = (1 - best_t) * a + best_t * (g_rescaled / S)
        S_new = float(a_new.sum())
        a_new = a_new / S_new
        sc_new = compute_score(a_new)
        
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
        
        a = a_new
        c = fast_conv(a)
        M = float(c.max())
        
        if n_viol == 0:
            print(f"  LP converged at iter {it}, M={M:.8f}, score={best_sc:.8f}")
            break
        
        # Update active set
        viol_idx = np.where(violations)[0]
        active_set = np.unique(np.concatenate([active_set, viol_idx]))
        if len(active_set) > max_k:
            # Keep top max_k by c_cross value
            active_set = active_set[np.argsort(c_cross[active_set])[::-1][:max_k]]
        
        print(f"  LP iter {it}: M={M:.8f}, k={len(active_set)}, viol={n_viol}, t={best_t:.1f}, score={best_sc:.8f}")
    
    return best_a, best_sc


def perturb_and_optimize(a, rng, scale=0.03, beta=500, maxiter=10000):
    """Perturb a and run L-BFGS on the result."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    strategy = rng.integers(0, 4)
    
    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 8))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    else:
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(5, n // 50)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.1 * rng.random())
        S_tmp = a_p.sum()
        if S_tmp > 1e-12:
            a_p = a_p / S_tmp
    
    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S < 1e-12:
        a_p = np.ones(n) * 0.5
    else:
        a_p = a_p / S
    
    sc, a_opt = run_lbfgs(a_p, beta, maxiter=maxiter)
    return sc, a_opt

    print(f"  LP took {time.monotonic()-t_lp:.1f}s, best={best_score:.10f}")
    
    # ============================================================
    # Phase 3: LSE L-BFGS (30% time)
    # ============================================================
    phase3_deadline = t_start + 0.70 * total_time
    print(f"\n--- Phase 3: LSE L-BFGS ---")
    
    beta_schedule = [100, 300, 1000, 3000, 10000, 30000, 100000]
    
    for beta in beta_schedule:
        if time.monotonic() > phase3_deadline - 30:
            break
        remaining = phase3_deadline - time.monotonic()
        maxiter = int(min(30000, max(2000, remaining * 25)))
        maxfun = int(min(80000, max(5000, remaining * 60)))
        
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"  [LBFGS beta={beta}] NEW BEST: {sc:.10f}")
    
    # ============================================================
    # Phase 4: ILS (20% time)
    # ============================================================
    phase4_deadline = t_start + 0.90 * total_time
