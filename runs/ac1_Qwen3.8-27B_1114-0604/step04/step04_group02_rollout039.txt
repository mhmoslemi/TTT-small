    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

    n = len(a)
    m = len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:

    return 2.0 * n * float(np.max(c)) / (S * S)

    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def full_lse_score_grad(a, beta):
    """Standard LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -100))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)
    
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter, maxfun, top_k=None, bounds=None):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    if bounds is None:
        bounds = [(0.0, 1000.0)] * n
    
    if top_k is not None:
        func = lambda x: topk_lse_score_grad(x, beta, top_k)
    else:
        func = lambda x: full_lse_score_grad(x, beta)
    
    res = minimize(
        func, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 300,
                 'iprint': -1}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out, res


def multi_peak_subgrad(a, n_iters, f_lb, top_k=5, decay=0.9999):
    """Multi-peak subgradient with momentum."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S_a = float(a.sum())
        
        # Get top_k peaks
        k = min(top_k, len(c))
        peak_indices = np.argsort(c)[::-1][:k]
        peak_vals = c[peak_indices]
        
        # Weighted subgradient (weight by how close to max)
        max_val = peak_vals[0]
        weights = np.exp(-0.1 * (max_val - peak_vals) / (max_val + 1e-30))
        weights = weights / weights.sum()
        
        Mak = float(np.dot(weights, peak_vals))
        sc = 2.0 * n * float(c.max()) / (S_a * S_a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Compute weighted subgradient
        sg = np.zeros(n)
        for w, ks in zip(weights, peak_indices):
            j_start = max(0, ks - n + 1)
            j_end = min(ks, n - 1)
            if j_end >= j_start:
                idx = np.arange(j_start, j_end + 1)
                sg[idx] += w * 2.0 * a[ks - idx]
        
        S2 = S_a * S_a
        S3 = S2 * S_a
        grad = (2.0 * n) * (sg / S2 - 2.0 * Mak / S3)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        step = 0.1 * decay ** it
        step = min(step, 2.0 * (sc - f_lb) / (gnorm * gnorm + 1e-30))
        step = max(step, 1e-6)
        
        v = 0.9 * v - step * grad
        a_new = a + v
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
    
    return best_a

def spectral_shaping(a, lowpass_frac=0.3, strength=0.1):
    """Apply a mild spectral low-pass filter and reshape."""
    n = len(a)
    F = np.fft.rfft(a)
    n_freq = len(F)
    cutoff = int(lowpass_frac * n_freq)
    # Smooth rolloff
    for i in range(cutoff, min(cutoff + n_freq // 10, n_freq)):
        F[i] *= (1.0 - (i - cutoff) / (n_freq // 10))
    for i in range(cutoff + n_freq // 10, n_freq):
        F[i] *= (1.0 - strength)
    a_out = np.fft.irfft(F, n)
    a_out = np.clip(a_out, 0.0, 1000.0)
    S = a_out.sum()
    if S > 1e-12:
        a_out = a_out / S
    return a_out

Combines LP-based direction finding, adaptive LSE L-BFGS, multi-peak

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):

    n = len(a)

    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter, maxfun, top_k=None):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    if top_k is not None:
        func = lambda x: lse_grad(x, beta, top_k)
    else:
        func = lambda x: lse_grad(x, beta)

    res = minimize(
        func, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 300,
                 'iprint': -1}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out, res


def multi_peak_subgradient(a, n_iters, f_lb=1.0, top_k=5, momentum=0.9):
    """Multi-peak subgradient with heavy-ball momentum."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    v = np.zeros(n)
    best_a = a.copy()
    best_sc = compute_score(a)
    step_base = 0.05
    decay = 1.0 - 2.0 / max(n_iters, 100)

    for it in range(n_iters):
        c = fast_conv(a)
        S_a = float(a.sum())
        max_val = float(c.max())
        sc = 2.0 * n * max_val / (S_a * S_a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        k = min(top_k, len(c))
        peak_idx = np.argsort(c)[::-1][:k]
        peak_vals = c[peak_idx]
        diff = max_val - peak_vals
        weights = np.exp(-diff / (0.01 * max_val + 1e-30))
        weights /= weights.sum()
        weighted_M = float(np.dot(weights, peak_vals))

        sg = np.zeros(n)
        for w, ks in zip(weights, peak_idx):
            j_start = max(0, ks - n + 1)
            j_end = min(ks, n - 1)
            if j_end >= j_start:
                idx = np.arange(j_start, j_end + 1)
                sg[idx] += w * 2.0 * a[ks - idx]

        S2 = S_a * S_a
        S3 = S2 * S_a
        grad = (2.0 * n) * (sg / S2 - 2.0 * weighted_M / S3)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        step = step_base * (decay ** it)
        step = min(step, 2.0 * max(sc - f_lb, 0.01) / (gnorm * gnorm + 1e-30))
        step = max(step, 1e-8)

        v = momentum * v - step * grad
        a_new = a + v
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa

    return best_a


def lp_direction(f, M, time_limit=30.0):
    """
    Solve LP: max sum(g) s.t. (f*g)(k) <= M, g >= 0.
    Uses cutting plane with cvxpy.
    Returns normalized g (sum=1) or None on failure.
    """
    import cvxpy as cp
    n = len(f)

    first_col = np.zeros(2 * n - 1)
    first_col[:n] = f
    first_row = np.zeros(n)
    first_row[0] = f[0]
    A = sparse_toeplitz(first_col, first_row)

    g_var = cp.Variable(n, non_neg=True)
    obj = cp.Maximize(cp.sum(g_var))

    t_start = time.monotonic()
    num_constraints = 2 * n - 1
    all_rhs = M * np.ones(num_constraints)

    # Cutting plane: start with subset, add violated constraints
    # Initial: use all if small enough, else top ones
    if num_constraints <= 5000:
        active = np.arange(num_constraints)
    else:
        # Start with a random subset + the most likely active ones
        g_init = np.ones(n) / n
        conv_init = A.dot(g_init)
        active = np.argsort(conv_init)[::-1][:2000]

    for cp_iter in range(50):
        if time.monotonic() - t_start > time_limit:
            break
        constraints = [A[active, :] @ g_var <= all_rhs[active]]
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver='CLARABEL')
        except Exception:
            try:
                prob.solve(solver='ECOS')
            except Exception:
                try:
                    prob.solve(solver='SCS')
                except Exception:
                    return None

        if g_var.value is None:
            return None

        g_val = np.clip(g_var.value, 0, 1000)
        # Check for violated constraints
        conv_vals = A.dot(g_val)
        violated = np.where(conv_vals > M + 1e-10)[0]
        if len(violated) == 0:
            break
        # Add most violated constraints
        worst = violated[np.argsort(conv_vals[violated])[::-1][:200]]
        active = np.unique(np.concatenate([active, worst]))

    if g_var.value is None:
        return None
    result = np.clip(g_var.value, 0, 1000)
    S = result.sum()
    if S < 1e-12:
        return None
    return result / S


def perturb(a, rng, scale):
    """Generate diverse perturbation."""
    n = len(a)
    strategy = int(rng.integers(0, 7))
    x = np.arange(n, dtype=np.float64) / n
    a_p = a.copy()

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 30.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        a_p = a * (1.0 + scale * np.sin(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 12))
        block_size = max(1, n // n_blocks)
        for b in range(min(n_blocks, n // block_size + 1)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + 2.0 * scale * (rng.random() - 0.5))
    elif strategy == 3:
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(10, n // 25)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.03 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            a_p = a_p / S
    elif strategy == 4:
        # Frequency domain
        F = np.fft.rfft(a)
        F[1:-1] *= (1.0 + scale * 0.5 * rng.standard_normal(F[1:-1].shape))
        a_p = np.fft.irfft(F, n)
    elif strategy == 5:
        # Localized random rescale
        n_sel = max(50, n // 8)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p[sel] *= (1.0 + 2.0 * scale * (rng.random(n_sel) - 0.5))
    else:
        # Gaussian kernel smoothing (sharpens or blurs)
        sigma = 0.5 + 5.0 * rng.random()
        from scipy.ndimage import gaussian_filter1d
        a_p = gaussian_filter1d(a, sigma, mode='nearest')

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def spectral_shaping(a, strength=0.3):
    """Mild spectral low-pass to remove high-freq oscillations."""
    n = len(a)
    F = np.fft.rfft(a)
    n_freq = len(F)
    # Cosine window rolloff
    window = np.ones(n_freq)
    transition = max(1, n_freq // 15)
    for i in range(n_freq - transition, n_freq):
        t = (i - (n_freq - transition)) / transition
        window[i] = 1.0 - strength * 0.5 * (1 - np.cos(np.pi * t))
    F *= window
    a_out = np.fft.irfft(F, n)
    a_out = np.clip(a_out, 0.0, 1000.0)
    S = a_out.sum()
    if S > 1e-12:
        a_out = a_out / S
    return a_out


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-strategy search: LP direction + adaptive L-BFGS + ILS + spectral shaping.
    Returns best non-negative sequence found within time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_time = max(60.0, float(budget_s) - 40.0)
    deadline = t_start + total_time

    best_score = np.inf
    best_a = None

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 128:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        p_norm = parent / parent.sum()
        sc_p = compute_score(p_norm)
        print(f"Parent: n={len(parent)}, score={sc_p:.10f}")
        report(sc_p, p_norm, "parent")
        n_work = len(parent)
    else:
        n_work = 8192

    n = n_work
    print(f"Working length: {n}")

    # Initial sequence
    if best_a is not None and len(best_a) == n:
        a_cur = best_a.copy()
    elif parent is not None:
        a_cur = p_norm.copy()
    else:
        a_cur = np.ones(n) * 0.5
        a_cur = a_cur / a_cur.sum()

    # ============================================================
    # Phase 1: LP-based direction finding (20% time)
    # ============================================================
    p1_end = t_start + 0.20 * total_time
    print(f"\n=== Phase 1: LP Direction (until t={p1_end - t_start:.0f}s) ===")

    n_lp = 0
    while time.monotonic() < p1_end - 5:
        c = fast_conv(a_cur)
        M = float(c.max())
        S = float(a_cur.sum())
        a_unit = a_cur / S
        g = lp_direction(a_unit, M, time_limit=20.0)
        if g is None:
            break
        # Line search over blend factor
        best_t = 0.0
        best_sc_t = compute_score(a_unit)
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            blend = (1 - t) * a_unit + t * g
            sc_t = compute_score(blend)
            if sc_t < best_sc_t:
                best_sc_t = sc_t
                best_t = t
        if best_t <= 0:
            break
        a_cur = (1 - best_t) * a_unit + best_t * g
        a_cur = a_cur / a_cur.sum()
        n_lp += 1
        report(compute_score(a_cur), a_cur, f"LP#{n_lp} t={best_t:.2f}")

    print(f"Phase 1: {n_lp} LP iters, best={best_score:.10f}")

    # ============================================================
    # Phase 2: Adaptive L-BFGS refinement (40% time)
    # ============================================================
    p2_end = t_start + 0.60 * total_time
    print(f"\n=== Phase 2: L-BFGS (until t={p2_end - t_start:.0f}s) ===")

    beta_schedule = [30, 60, 150, 400, 1000, 3000, 10000, 30000, 100000, 300000]

    for beta in beta_schedule:
        if time.monotonic() > p2_end - 15:
            break
        remaining = p2_end - time.monotonic()
        maxiter = int(min(40000, max(3000, remaining * 40)))
        maxfun = int(min(200000, max(15000, remaining * 200)))

        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter, maxfun)
        report(sc, a_opt, f"LBF b={beta}")

        if time.monotonic() < p2_end - 20:
            for tk in [3, 5]:
                sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter // 3, maxfun // 3, top_k=tk)
                report(sc, a_opt, f"LBF b={beta} tk={tk}")

    print(f"Phase 2: best={best_score:.10f}")

    # ============================================================
    # Phase 3: ILS with diverse perturbations (25% time)
    # ============================================================
    p3_end = t_start + 0.85 * total_time
    print(f"\n=== Phase 3: ILS (until t={p3_end - t_start:.0f}s) ===")

    n_trials = 0
    while time.monotonic() < p3_end - 30:
        remaining = p3_end - time.monotonic()
        if remaining < 30:
            break
        scale = 0.02 + 0.05 * rng.random()
        a_p = perturb(best_a, rng, scale)
        # Spectral shaping on perturbation
        if rng.random() < 0.3:
            a_p = spectral_shaping(a_p, strength=0.2 + 0.3 * rng.random())

        beta = int(200 + 800 * rng.random())
        maxiter = int(min(25000, max(3000, remaining * 25)))
        maxfun = int(min(100000, max(10000, remaining * 100)))
        sc, a_opt, _ = run_lbfgs(a_p, beta, maxiter, maxfun)
        report(sc, a_opt, f"ILS#{n_trials} b={beta} s={scale:.3f}")

        # Small perturbation follow-up
        if time.monotonic() < p3_end - 20:
            a_p2 = perturb(best_a, rng, scale * 0.3)
            sc2, a_opt2, _ = run_lbfgs(a_p2, 1000, maxiter // 3, maxfun // 3)
            report(sc2, a_opt2, f"ILS#{n_trials}sm")

        n_trials += 1

    print(f"Phase 3: {n_trials} trials, best={best_score:.10f}")

    # ============================================================
    # Phase 4: Final polish (15% time)
    # ============================================================
    print(f"\n=== Phase 4: Final Polish ===")

    # Highest beta L-BFGS
    if time.monotonic() < deadline - 30:
        for beta in [200000, 500000, 1000000]:
            if time.monotonic() > deadline - 20:
                break
            sc, a_opt, _ = run_lbfgs(best_a, beta, 50000, 200000)
            report(sc, a_opt, f"FIN b={beta}")

    # Top-k final
    if time.monotonic() < deadline - 20:
        for tk in [2, 3, 5]:
            if time.monotonic() > deadline - 15:
                break
            sc, a_opt, _ = run_lbfgs(best_a, 500000, 30000, 100000, top_k=tk)
            report(sc, a_opt, f"FIN tk={tk}")

    # Multi-peak subgradient polish
    if time.monotonic() < deadline - 15:
        a_sg = multi_peak_subgradient(best_a, n_iters=2000, top_k=7)
        sc = compute_score(a_sg)
        report(sc, a_sg, "FIN subgrad")

    # Spectral shaping + re-optimize
    if time.monotonic() < deadline - 15:
        a_sh = spectral_shaping(best_a, strength=0.5)
        sc, a_opt, _ = run_lbfgs(a_sh, 50000, 20000, 80000)
        report(sc, a_opt, "FIN shape")

    # Multi-length: try slightly different lengths
    if time.monotonic() < deadline - 40 and best_a is not None:
        for n_try in [int(n * 0.8), int(n * 1.2)]:
            if time.monotonic() > deadline - 20:
                break
            if n_try < 512 or n_try > 20000:
                continue
            a_interp = np.interp(np.linspace(0, 1, n_try),
                                  np.linspace(0, 1, len(best_a)), best_a)
            a_interp = np.clip(a_interp, 0, 1000)
            S = a_interp.sum()
            if S > 0:
                a_interp = a_interp / S
            sc, a_opt, _ = run_lbfgs(a_interp, 500, 15000, 60000)
            report(sc, a_opt, f"LEN n={n_try}")

    # ============================================================
    # Final output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
