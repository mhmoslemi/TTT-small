    """Adam optimizer with LSE smoothing."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0/n
    # We work in the space where sum(a) = 1 to make it scale-invariant
    a = a / a.sum()
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.8, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for t in range(1, n_steps + 1):
        if time.monotonic() - t_start > time_limit:
            break
        
        score, grad = lse_smoothed_score_and_grad(a, beta)
        if not np.isfinite(score):
            break
            
        lr_t = lr / (1.0 + 1e-4 * t)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1**t)
        v_hat = v / (1.0 - b2**t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)

        if s < 1e-12:

    sc_final = compute_score(a)

        best_score = sc_final

    return best_score, best_a

def lp_step(f, M):
    """
    LP step: maximize sum(b) subject to b >= 0 and (f*b)_k <= M for all k.
    Returns optimal b normalized to have same sum as f.
    """
    n = len(f)
    # Variables: b_0, ..., b_{n-1} (n variables)
    # Objective: maximize sum(b) = c^T b where c = ones(n)
    # Constraints: A @ b <= M (scalar) for each k
    #   For k = 0, ..., 2n-2: sum_{j=max(0,k-n+1)}^{min(k,n-1)} f[k-j] * b_j <= M
    #   This is equivalent to: for each k, the row is f[k-min(k,n-1):k+1] reversed... 
    #   Actually: (f*b)_k = sum_j f_j * b_{k-j} where 0<=j<n and 0<=k-j<n
    #   = sum_{j: k-n+1<=j<=k, 0<=j<n} f_j * b_{k-j}
    #   Let's reindex: let i = k-j, so j = k-i
    #   = sum_{i: 0<=i<n, 0<=k-i<n} f_{k-i} * b_i
    #   = sum_{i: max(0,k-n+1)<=i<=min(k,n-1)} f_{k-i} * b_i
    # So the constraint matrix A has A[k, i] = f[k-i] if max(0,k-n+1)<=i<=min(k,n-1), else 0
    
    # This is a (2n-1) x n matrix, which is very sparse (tridiagonal-like banded)
    # For scipy linprog, we can use the 'highs' solver with sparse matrix
    
    from scipy.sparse import csr_matrix
    
    # Build A as a sparse matrix
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_vec = np.full(2 * n - 1, M)
    c_vec = -np.ones(n)  # minimize -sum(b)
    
    bounds = [(0, None)] * n
    
    res = linprog(c_vec, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs')
    
    if res.success:
        b_opt = res.x
        s = b_opt.sum()
        if s > 1e-12:
            b_opt = b_opt / s * f.sum()  # rescale to match sum of f
            return b_opt
    return None

    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def lse_score_grad(a, beta):
    """
    LSE-smoothed score and gradient.
    Uses the log-sum-exp trick to smoothly approximate max.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    if exp_sum < 1e-300:
        return 1e10, np.zeros(n)
    
    L = c_max + np.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)
    
    # Probability weights
    p = exp_scaled / exp_sum  # length 2n-1
    
    # Gradient of L w.r.t. a:
    # dL/da_i = (1/beta) * sum_k p_k * dc_k/da_i
    # dc_k/da_i = 2*a_{k-i} if 0 <= k-i < n, else 0
    # So dL/da_i = (2/beta) * sum_k p_k * a_{k-i} (over valid k)
    #            = (2/beta) * (p cross-correlate a)_i
    # where (p ⋆ a)_i = sum_j p_{j+i} * a_j
    
    # Compute cross-correlation via FFT
    L_fft = 1 << (len(p) + n - 1).bit_length()
    P = np.fft.rfft(p, L_fft)
    A_fft = np.fft.rfft(a, L_fft)
    cross = np.fft.irfft(P * np.conj(A_fft), L_fft)
    q = cross[:n]  # q_i = sum_j p[j+i] * a_j
    
    dL_da = (2.0 / beta) * q
    
    # d(score)/da_i = 2n * (dL_da_i / S^2 - 2*L / S^3)
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (dL_da / S2 - 2.0 * L / S3)
    
    return score, grad

def adam_optimize(a_init, beta, lr, n_steps, time_limit, t_start):
    """Adam with LSE-smoothed gradient. Works in normalized space (sum=1)."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / S
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.8, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_score = compute_score(a)
    
    for t in range(1, n_steps + 1):
        if time.monotonic() - t_start > time_limit:
            break
        
        score, grad = lse_score_grad(a, beta)
        if not np.isfinite(score) or not np.all(np.isfinite(grad)):
            break
        
        lr_t = lr / (1.0 + 1e-4 * t)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a[:] = 1.0 / n
        else:
            a = a / s
        
        if t % 5000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a

def lp_mirror_step(f, M, max_n=1024):
    """
    LP step: maximize sum(b) s.t. b >= 0, (f*b)_k <= M for all k.
    Only works for n <= max_n due to LP size.
    """
    n = len(f)
    if n > max_n:
        return None
    
    # Build constraint matrix: A[k, j] = f[k-j] if 0<=k-j<n, else 0
    # Total nonzeros: sum_k min(k+1, n, 2n-1-k) = O(n^2/2)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[k - j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    b_vec = np.full(2 * n - 1, M)
    c_vec = -np.ones(n)  # minimize -sum(b)
    bounds = [(0, None)] * n
    
    from scipy.optimize import linprog
    res = linprog(c_vec, A_ub=A, b_ub=b_vec, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 60})
    
    if res.success:
        b_opt = res.x.copy()
        s = b_opt.sum()
        if s > 1e-12:
            b_opt = b_opt * (f.sum() / s)
            return b_opt
    return None

def subgradient_descent(a_init, n_steps=5000, time_limit=120, t_start=None):
    """Subgradient descent with averaging."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = a / a.sum()
    
    a_avg = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    
    for t in range(1, n_steps + 1):
        if t_start is not None and time.monotonic() - t_start > time_limit:
            break
        
        c = fast_conv(a)
        S = a.sum()
        k_star = int(c.argmax())
        M = float(c[k_star])
        
        # Subgradient of max(conv) w.r.t. a
        sg = np.zeros(n)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            idx = k_star - np.arange(js, je + 1)
            sg[js:je + 1] = 2.0 * a[idx]
        
        # Score function: 2n*M/S^2
        # Subgradient: 2n*(sg/S^2 - 2*M*ones/S^3)
        S2 = S * S
        S3 = S2 * S
        full_sg = 2.0 * n * (sg / S2 - 2.0 * M / S3)
        
        # Step size
        lr_t = 0.5 / np.sqrt(t)
        a = a - lr_t * full_sg
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a[:] = 1.0 / n
        else:
            a = a / s
        
        # Averaged iterate
        a_avg = a_avg + (a - a_avg) / (t + 1)
        
        if t % 1000 == 0:
            for candidate in [a, a_avg]:
                sc = compute_score(candidate)
                if sc < best_score:
                    best_score = sc
                    best_a = candidate.copy()
    
    for candidate in [a, a_avg]:
        sc = compute_score(candidate)
        if sc < best_score:
            best_score = sc
            best_a = candidate.copy()
    
    return best_score, best_a
