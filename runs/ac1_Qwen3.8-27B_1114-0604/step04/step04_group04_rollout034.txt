import numpy as np
from scipy.optimize import minimize
import time


def self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def project_simplex(v):
    n = len(v)
    v_sorted = np.sort(v)[::-1]
    cssv = np.cumsum(v_sorted)
    ind = np.arange(1, n + 1)
    cond = v_sorted - (cssv - 1.0) / ind > 0
    i = np.where(cond)[0]
    if len(i) == 0:
        return np.ones(n) / n
    rho = i[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1)
    return np.maximum(v - theta, 0.0)


def lse_obj_at(a, beta):
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    log_sum_exp = z.max() - z_shift.max() + np.log(exp_z.sum() + 1e-300)
    L_val = c_max + log_sum_exp / beta
    return 2.0 * n * L_val / (S * S)


def lse_grad_at(a, beta):
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    exp_sum = float(exp_z.sum()) + 1e-300
    p = exp_z / exp_sum
    log_sum_exp = z.max() - z_shift.max() + np.log(exp_sum)
    L_val = c_max + log_sum_exp / beta

    M_fft = 1 << (3 * n - 2 - 1).bit_length()
    P = np.fft.rfft(p, M_fft)
    AA = np.fft.rfft(a, M_fft)
    corr = np.fft.irfft(P * np.conj(AA), M_fft)[:n]
    grad_L = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return grad


def lse_obj_grad_wrapped(a, beta):
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shift = z - z.max()
    exp_z = np.exp(z_shift)
    exp_sum = float(exp_z.sum()) + 1e-300
    p = exp_z / exp_sum
    log_sum_exp = z.max() - z_shift.max() + np.log(exp_sum)
    L_val = c_max + log_sum_exp / beta
    score = 2.0 * n * L_val / (S * S)

    M_fft = 1 << (3 * n - 2 - 1).bit_length()
    P = np.fft.rfft(p, M_fft)
    AA = np.fft.rfft(a, M_fft)
    corr = np.fft.irfft(P * np.conj(AA), M_fft)[:n]
    grad_L = 2.0 * corr
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = float(a0.sum())
    if S < 1e-6:
        a0[:] = 0.5
    else:
        a0 = a0 / S
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_obj_grad_wrapped, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    Sc = compute_score(a_out)
    return Sc, a_out, res


def fista_lp(a, M, n_iters=3000, beta_start=5.0, beta_end=100.0):
    n = len(a)
    x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0
    sum_a2 = float(np.sum(a * a)) + 1e-15

    for it in range(n_iters):
        frac = min(1.0, it / max(1, n_iters * 0.3))
        beta = beta_start + (beta_end - beta_start) * frac

        L = beta * n * sum_a2 + 1e-12
        step = 1.0 / L

        c = cross_conv(a, u)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum()) + 1e-300
        p = exp_z / exp_sum

        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(p) - n + 1]
        grad = a * sw - 1.0

        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    return x


def mw_lp_solver(a, M, n_iters=2000, eta=None):
    n = len(a)
    x = np.ones(n) / n
    y = np.zeros(2 * n - 1)
    if eta is None:
        eta = 2.0 / (n * float(np.sum(a * a)) + 1e-10)

    for it in range(n_iters):
        c = cross_conv(a, x)
        excess = np.maximum(c - M, 0.0)
        y = y + eta * excess
        Cy = np.zeros(n)
        cum = np.concatenate([[0.0], np.cumsum(y)])
        sw = cum[n:] - cum[:len(y) - n + 1]
        Cy = a * sw
        x = x * np.exp(-eta * np.maximum(Cy - 1.0, 0.0))
        Sx = float(x.sum())
        if Sx < 1e-12:
            x = np.ones(n) / n
        else:
            x = x / Sx

    return x


def lp_alternating(a, max_outer_iters=200, fista_iters=3000, t_start=0.4, t_min=0.05,
                   time_limit=None, use_mw=False, mw_iters=1000):
    n = len(a)
    a = np.maximum(a.astype(np.float64).copy(), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0

    for outer in range(max_outer_iters):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        M = float(self_conv(a).max())
        if use_mw:
            x = mw_lp_solver(a, M, n_iters=mw_iters)
        else:
            x = fista_lp(a, M, n_iters=fista_iters)
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx

        t = max(t_min, t_start * (1.0 - 0.4 * outer / max(1, max_outer_iters)))
        a_new = (1.0 - t) * a + t * x
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 12:
                t = t * 0.5
                if t < 0.01:
                    break
                no_improve = 0
        a = a_new

    return best_a, best_sc


def error_reduction(a, n_iters=100, alpha=0.7):
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    M_fft = 1 << (2 * n - 1 - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-16:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-16:
            break
        conv_mod = conv - 0.8 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M_fft, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-10] = 0.0

        damp = 0.5 + 0.5 * (it / max(1, n_iters // 4))
        A_new = A + damp * d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def subgradient_polish(a, total_iters=3000):
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(total_iters):
        c = self_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.05)
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a

        maxiter = min(5000, max(100, int(remaining / 0.5)))
        maxfun = max(500, int(remaining * 2))
        sc, a_opt, _ = run_lbfgs(a_norm, beta, maxiter=maxiter, maxfun=maxfun)
        report(sc, a_opt, f"LBFGS b={beta}")
        a_norm = a_opt.copy()
        Sn = float(a_norm.sum())
        if Sn > 1e-12:
            a_norm = a_norm / Sn

    print(f"Phase 3 done: best={best_score:.8f}")

    # Phase 4: Final polish (remaining time)
    print(f"\n--- Phase 4: Final Polish ---")
    remaining_at_p4 = deadline - time.monotonic()

    n_final = len(best_a)
    a_f = best_a.copy()
    S = float(a_f.sum())
    if S > 1e-12:
        a_f = a_f / S

    # Subgradient
    if remaining_at_p4 > 120:
        print(f"  Subgradient...")
        a_sp = subgradient_polish(a_f, total_iters=2000)
        sc = compute_score(a_sp)
        report(sc, a_sp, "subgrad")

    # Perturbation + LBFGS
    for trial in range(6):
        if time.monotonic() > deadline - 60:
            break
        strategy = trial % 3
        if strategy == 0:
            scale = 0.005 + 0.02 * rng.random()
            a_pert = a_f * (1.0 + scale * rng.standard_normal(n_final))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            c = self_conv(a_f)
            k_star = int(np.argmax(c))
            js = max(0, k_star - n_final + 1)
            je = min(k_star, n_final - 1)
            a_pert = a_f.copy()
            seg = a_pert[js:je + 1]
            contrib = seg[::-1].copy()
            top_k = max(10, n_final // 100)
            top_idx = np.argsort(contrib)[::-1][:top_k]
            total_red = 0.0
            for li in top_idx:
                gi = js + li
                red = a_pert[gi] * 0.02
                a_pert[gi] = max(a_pert[gi] - red, 0.0)
                total_red += red
            bot_idx = np.argsort(contrib)[:max(1, len(contrib) - top_k)]
            if len(bot_idx) > 0:
                per = total_red / len(bot_idx)
                for li in bot_idx:
                    a_pert[js + li] += per
        else:
            freq = 1.0 + 4.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.005 + 0.02 * rng.random()
            xv = np.arange(n_final, dtype=np.float64) / n_final
            a_pert = a_f * (1.0 + amp * np.cos(2.0 * np.pi * freq * xv + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        Sp = float(a_pert.sum())
        if Sp > 1e-12:
            a_pert = a_pert / Sp
            sc, a_opt, _ = run_lbfgs(a_pert, 5000, maxiter=3000, maxfun=50000)
            report(sc, a_opt, f"pert #{trial}")

    # Final LP
    if time.monotonic() < deadline - 80:
        print(f"  Final LP...")
        a_lp = best_a.copy()
        S = float(a_lp.sum())
        if S > 1e-12:
            a_lp = a_lp / S
        a_out, _ = lp_alternating(a_lp, max_outer_iters=15, fista_iters=5000,
                                  t_start=0.1, t_min=0.01, time_limit=deadline - 30)
        sc = compute_score(a_out)
        report(sc, a_out, "final LP")

    # Final LBFGS very high beta
    if time.monotonic() < deadline - 30:
        a_vh = best_a.copy()
        S = float(a_vh.sum())
        if S > 1e-12:
            a_vh = a_vh / S
        sc, a_opt, _ = run_lbfgs(a_vh, 50000, maxiter=3000, maxfun=50000)
        report(sc, a_opt, "final LBFGS b=50000")

    # Verify
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
