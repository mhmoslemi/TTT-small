"""
Minimizing 2n*max(self-conv)/(sum)^2 for non-negative sequences.
Uses: (1) LSE-smoothed projected gradient with continuation on beta,
(2) coordinate descent polish with exact max, (3) diverse multi-start.
Key: fix sum=1 via projection, use FFT for fast cross-correlation gradient.
"""

import numpy as np
import time
import math

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)

def lse_crosscorr_grad(a, beta):
    """
    Compute LSE(a⊛a; beta) and its gradient w.r.t. a (assuming sum(a)=1).
    Returns: (lse_value, grad) where grad = 2n * 2 * (a ⋆ p) = 4n * (a ⋆ p)
    Actually, just return (lse_value, cross) where cross[i] = (a⋆p)[i],
    and the caller can compute the full gradient.
    """
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    p = exp_scaled / exp_scaled.sum()
    
    # Compute cross-correlation (a ⋆ p)[i] = sum_m a[m] * p[m+i]
    # Using FFT: rev_a cross-correlated with p
    # (a ⋆ p)[i] = sum_m a[m] * p[m+i]
    # = conv(rev_a, p)[n-1+i] where rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a[::-1], M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n-1:2*n-1]
    
    # cross[i] = conv_res[i] = (a ⋆ p)[i]
    return L, conv_res

def projected_gradient_step(a, beta, lr):
    """One step of projected gradient: a -= lr * grad, project to {a>=0, sum=1}"""
    n = len(a)
    L, cross = lse_crosscorr_grad(a, beta)
    # Score = 2n * L (with sum=1)
    # dScore/da_i = 2n * 2 * cross[i] = 4n * cross[i]
    grad = 4.0 * n * cross
    a_new = a - lr * grad
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        a_new = np.ones(n) / n
    else:
        a_new = a_new / s
    return a_new, L

def adam_lse(a_init, beta, lr=0.01, n_steps=10000, time_budget=None, t_start=None):
    """Adam optimizer for LSE score with sum normalization."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a[:] = 1.0/n
    else:
        a /= s
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    
    best_a = a.copy()
    best_lse = None
    best_score = compute_score(a)
    
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf
    
    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break
        
        L, cross = lse_crosscorr_grad(a, beta)
        grad = 4.0 * n * cross
        
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a
        else:
            a_new = a_new / s
        a = a_new
        
        if t % 500 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_score, best_a

def coordinate_descent_exact(a, n_sweeps=5, top_k=30, time_budget=30.0, t_start=None):
    """
    Coordinate descent with golden-section 1D search using exact score.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (t_start is not None) else np.inf
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    
    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        
        # Compute score and find which coordinates to modify
        c = fast_conv(a)
        k_star = int(c.argmax())
        M = float(c[k_star])
        S = float(a.sum())
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score
        
        # Subgradient of max: the coordinates that "feed into" the peak
        # Peak is at k_star. The coordinates a_i that affect c_{k_star} are
        # those with 0 <= k_star - i < n, i.e., max(0, k_star-n+1) <= i <= min(n-1, k_star)
        # For each such i, the subgradient contribution is a_{k_star - i}
        
        # Select top-k by |a_{k_star - i}| (these are the most impactful)
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            impacts = a[k_star - np.arange(js, je + 1)]
            order_in = np.argsort(np.abs(impacts))[::-1][:top_k]
            candidates = np.arange(js, je + 1)[order_in]
        else:
            candidates = np.arange(n)
        
        for j in candidates:
            j = int(j)
            if time.monotonic() > deadline:
                break
            
            # Golden section search for a[j]
            a_old = a[j]
            lo = 0.0
            hi = a_old * 3.0 + 0.01 * a.max()
            
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            
            a[j] = c1
            f1 = compute_score(a)
            a[j] = c2
            f2 = compute_score(a)
            
            for _ in range(12):
                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = hi - gr * (hi - lo)
                    a[j] = c1
                    f1 = compute_score(a)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = lo + gr * (hi - lo)
                    a[j] = c2
                    f2 = compute_score(a)
            
            best_val = c1 if f1 < f2 else c2
            a[j] = best_val
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
    
    return best_score, a

def continuation_adam(a0, betas, time_budget, t_start):
    """Run Adam with increasing beta values."""
    a = a0.copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget
    
    for i, beta in enumerate(betas):
        if time.monotonic() > deadline - 5:
            break
        remaining = deadline - time.monotonic()
        time_alloc = min(remaining * 0.35, 45.0)
        sc, a_opt = adam_lse(a, beta, lr=0.005, n_steps=20000,
                             time_budget=time_alloc, t_start=t_start)
        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"    Continuation beta={beta}: score={sc:.8f} (t={time.monotonic()-t_start:.0f}s)")
        a = a_opt
    
    return best_score, best_a

def make_hann(n):
    x = np.arange(n, dtype=np.float64)
    return np.sin(np.pi * x / (n - 1)) ** 2

def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((x - center) / sigma) ** 2)

def make_cosine_mod(n, f=3, alpha=0.3):
    x = np.arange(n, dtype=np.float64)
    return 1.0 + alpha * np.cos(2.0 * np.pi * f * x / n)

def make_multi_freq(n, rng, n_freqs=10):
    x = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(n_freqs):
        f = float(rng.integers(1, n // 2))
        phase = float(rng.uniform(0, 2 * np.pi))
        amp = float(rng.uniform(0.02, 0.08))
        a += amp * np.cos(2.0 * np.pi * f * x / n + phase)
    return np.maximum(a, 0.0)

def make_parent_perturbed(parent, rng, strength=0.1):
    n = len(parent)
    a = parent.copy().astype(np.float64)
    x = np.arange(n, dtype=np.float64)
    # Add smooth random perturbation
    n_modes = 5
    for _ in range(n_modes):
        f = float(rng.uniform(1, 10))
        phase = float(rng.uniform(0, 2 * np.pi))
        amp = strength * a.max() * float(rng.uniform(0.3, 1.0))
        a += amp * np.cos(2.0 * np.pi * f * x / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    return a

"""
Minimizing 2n*max(self-conv)/(sum)^2 for non-negative sequences.
Uses: (1) LSE-smoothed Adam + L-BFGS with continuation on beta,
(2) targeted pair-swap coordinate descent exploiting peak structure,
(3) diverse multi-start with anti-symmetric and multi-frequency perturbations.
Key insight: fix sum=1 via projection; anti-symmetric components reduce central peak.
"""

import numpy as np
from scipy.optimize import minimize
import time
import math


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_and_p(a, beta):
    """Compute LSE value and probability distribution p over convolution lags."""
    c = fast_conv(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    p = exp_scaled / exp_scaled.sum()
    return L, p, int(c.argmax())


def crosscorr_a_p(a, p):
    """Compute (a ⋆ p)[i] = sum_m a[m] * p[m+i] for i=0..n-1 via FFT."""
    n = len(a)
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a[::-1], M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    return conv_res


def lse_grad_sum1(a, beta):
    """
    Gradient of 2n*LSE(conv(a,a);beta) w.r.t. a, assuming sum(a)=1.
    Returns: (score=2n*L, grad)
    grad[i] = 4n * (a ⋆ p)[i]
    """
    n = len(a)
    L, p, _ = lse_and_p(a, beta)
    cross = crosscorr_a_p(a, p)
    score = 2.0 * n * L
    grad = 4.0 * n * cross
    return score, grad


def adam_lse(a_init, beta, lr=0.01, n_steps=10000, time_budget=None, t_start=None):
    """Adam optimizer for LSE score with sum=1 constraint."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a[:] = 1.0 / n
    else:
        a /= s

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_score = compute_score(a)
    best_a = a.copy()

    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf

    for t in range(1, n_steps + 1):
        if time.monotonic() > deadline:
            break
        score, grad = lse_grad_sum1(a, beta)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            a_new = a
        else:
            a_new = a_new / s
        a = a_new

        if t % 1000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = compute_score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_score, best_a


def lbfgs_lse_obj(a, beta):
    """Objective and gradient for L-BFGS (sum=1 normalization handled by bounds)."""
    n = len(a)
    a_clip = np.maximum(a, 0.0)
    s = a_clip.sum()
    if s < 1e-10:
        return 1e10, np.zeros(n)
    a_norm = a_clip / s
    score, grad = lse_grad_sum1(a_norm, beta)
    # Chain rule: we pass a directly, normalize internally
    # Grad w.r.t. a (before normalization): grad_norm / s - grad_norm * a / s^2
    # But since we project and normalize after each step, we just pass the normalized grad
    # L-BFGS-B will handle bounds, so we compute grad at a_norm projected back
    grad_out = grad / s - 2.0 * score * a_clip / (s * s)
    # Actually simpler: just use the gradient at the normalized point
    # and let the optimizer deal with it. Since we normalize after each step,
    # the effective gradient is just grad (tangent to the simplex).
    # For L-BFGS-B with bounds [0, inf], the normalization is a constraint.
    # Let's just use the score and grad at a_norm, treating a as the variable.
    return score, grad


def run_lbfgs(a_init, beta, maxiter=3000, time_budget=30.0, t_start=None):
    """Run L-BFGS-B on the LSE-smoothed score."""
    n = len(a_init)
    a0 = a_init.astype(np.float64).copy()
    a0 = np.maximum(a0, 0.0)
    s = a0.sum()
    if s < 1e-10:
        a0[:] = 1.0 / n
    else:
        a0 = a0 / s

    bounds = [(0.0, None)] * n
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf

    def obj_and_grad(x):
        if time.monotonic() > deadline:
            raise TimeoutError
        xn = np.maximum(x, 0.0)
        s = xn.sum()
        if s < 1e-10:
            return 1e10, np.zeros(n)
        xn = xn / s
        score, grad = lse_grad_sum1(xn, beta)
        return score, grad

    try:
        res = minimize(
            obj_and_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-12, 'maxiter': maxiter,
                     'maxfun': maxiter * 10, 'maxls': 50}
        )
        a_out = np.maximum(res.x, 0.0)
        s = a_out.sum()
        if s > 1e-10:
            a_out = a_out / s
    except (TimeoutError, Exception):
        a_out = a0

    sc = compute_score(a_out)
    return sc, a_out

def pair_swap_descent(a, n_sweeps=5, time_budget=30.0, t_start=None):
    """
    Targeted coordinate descent: for the peak lag k*, find pairs (j, k*-j)
    and do golden-section on the transfer parameter t.
    This exploits the structure that c_{k*} involves a_j * a_{k*-j}.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (t_start is not None) else np.inf
    gr = (math.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        k_star = int(c.argmax())
        M = float(c[k_star])
        S = float(a.sum())
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score

        # Find pairs (j, k*-j) that contribute to the peak
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je < js:
            continue

        # Sort pairs by impact (a_j * a_{k*-j})
        pairs = []
        for j in range(js, je + 1):
            l = k_star - j
            if 0 <= l < n and j != l:
                impact = a[j] * a[l]
                pairs.append((impact, j, l))
            elif j == l:
                pairs.append((a[j] * a[j], j, -1))

        pairs.sort(key=lambda x: -x[0])

        # Try top pairs
        max_pairs = min(40, len(pairs))
        for idx in range(max_pairs):
            if time.monotonic() > deadline:
                break
            _, j, l = pairs[idx]

            if l == -1:
                # Diagonal: just optimize a[j]
                _golden_single(a, j, gr, compute_score)
            else:
                # Pair swap: a[j] -> a[j]+t, a[l] -> a[l]-t
                a_j = a[j]
                a_l = a[l]
                # t range: a[j]+t >= 0 and a[l]-t >= 0 => t in [-a_j, a_l]
                lo = -a_j * 0.8
                hi = a_l * 0.8
                if hi <= lo:
                    continue

                # Golden section on t
                c1 = hi - gr * (hi - lo)
                c2 = lo + gr * (hi - lo)

                a[j] = a_j + c1
                a[l] = a_l - c1
                f1 = compute_score(a)

                a[j] = a_j + c2
                a[l] = a_l - c2
                f2 = compute_score(a)

                for _ in range(10):
                    if f1 < f2:
                        hi = c2
                        c2 = c1
                        f2 = f1
                        c1 = hi - gr * (hi - lo)
                        a[j] = a_j + c1
                        a[l] = a_l - c1
                        f1 = compute_score(a)
                    else:
                        lo = c1
                        c1 = c2
                        f1 = f2
                        c2 = lo + gr * (hi - lo)
                        a[j] = a_j + c2
                        a[l] = a_l - c2
                        f2 = compute_score(a)

                best_t = c1 if f1 < f2 else c2
                a[j] = a_j + best_t
                a[l] = a_l - best_t
                a = np.maximum(a, 0.0)
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc

    return best_score, a


def _golden_single(a, j, gr, score_fn):
    """Golden section on a single coordinate a[j]."""
    a_old = a[j]
    lo = 0.0
    hi = a_old * 3.0 + 0.01 * a.max()
    if hi < 1e-10:
        hi = 0.1 * a.mean() + 1e-6

    c1 = hi - gr * (hi - lo)
    c2 = lo + gr * (hi - lo)
    a[j] = c1
    f1 = score_fn(a)
    a[j] = c2
    f2 = score_fn(a)

    for _ in range(12):
        if f1 < f2:
            hi = c2
            c2 = c1
            f2 = f1
            c1 = hi - gr * (hi - lo)
            a[j] = c1
            f1 = score_fn(a)
        else:
            lo = c1
            c1 = c2
            f1 = f2
            c2 = lo + gr * (hi - lo)
            a[j] = c2
            f2 = score_fn(a)

    a[j] = c1 if f1 < f2 else c2


def coord_descent_topk(a, n_sweeps=5, top_k=40, time_budget=30.0, t_start=None):
    """Standard coordinate descent on top-k coordinates by gradient magnitude."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if (t_start is not None) else np.inf
    gr = (math.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        k_star = int(c.argmax())
        M = float(c[k_star])
        S = float(a.sum())
        score = 2.0 * n * M / (S * S)
        if score < best_score:
            best_score = score

        # Subgradient at peak
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            impacts = a[k_star - np.arange(js, je + 1)]
            order = np.argsort(np.abs(impacts))[::-1][:top_k]
            candidates = np.arange(js, je + 1)[order]
        else:
            candidates = np.argsort(np.abs(a))[::-1][:top_k]

        for j in candidates:
            j = int(j)
            if time.monotonic() > deadline:
                break
            _golden_single(a, j, gr, compute_score)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc

    return best_score, a


def make_hann(n):
    x = np.arange(n, dtype=np.float64)
    return np.sin(np.pi * x / (n - 1)) ** 2


def make_gaussian(n, sigma_frac=0.3):
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * ((x - center) / sigma) ** 2)


def make_anti_perturbed(a, rng, strength=0.15):
    """Add anti-symmetric component to reduce central peak."""
    n = len(a)
    x = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    # Multiple anti-symmetric frequencies
    a_new = a.copy()
    for _ in range(3):
        f = float(rng.uniform(0.5, 8.0))
        phase = float(rng.uniform(0, math.pi))  # Only 0 to pi for anti-sym
        # Anti-symmetric: sin(2*pi*f*x/n) is anti-symmetric about center (approx)
        comp = np.sin(2.0 * np.pi * f * (x - center) / n + phase)
        a_new += strength * a.max() * comp * float(rng.uniform(0.3, 1.0))
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s > 1e-10:
        a_new = a_new / s
    return a_new


def make_multi_freq(n, rng, n_freqs=10, amp_range=(0.02, 0.08)):
    """Sum of random cosines + offset for flat auto-correlation."""
    x = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    total_amp = 0.0
    for _ in range(n_freqs):
        f = float(rng.integers(1, max(2, n // 4)))
        phase = float(rng.uniform(0, 2 * np.pi))
        amp = float(rng.uniform(*amp_range)) * float(rng.uniform(0.5, 1.5))
        a += amp * np.cos(2.0 * np.pi * f * x / n + phase)
        total_amp += amp
    # Ensure non-negative by adding offset if needed
    if a.min() < 0:
        a += -a.min() + 0.01
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a


def make_parent_perturbed(parent, rng, strength=0.1):
    """Smooth perturbation of parent."""
    n = len(parent)
    a = parent.copy().astype(np.float64)
    x = np.arange(n, dtype=np.float64)
    for _ in range(5):
        f = float(rng.uniform(0.5, 15.0))
        phase = float(rng.uniform(0, 2 * np.pi))
        amp = strength * a.max() * float(rng.uniform(0.3, 1.0))
        a += amp * np.cos(2.0 * np.pi * f * x / n + phase)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a


def make_two_level(n, rng, density=0.5):
    """Random two-level sequence."""
    mask = rng.random(n) < density
    a = mask.astype(np.float64)
    a += 0.1 * rng.random(n)
    s = a.sum()
    if s > 1e-10:
        a = a / s
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for minimizing 2n*max(self-conv)/(sum)^2.
    Phases: (1) Continuation from parent, (2) Diverse restarts,
    (3) Targeted pair-swap descent, (4) Final polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 20.0

    # Get parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    if parent is None:
        parent = np.ones(4096)

    p_sum = parent.sum()
    if p_sum < 1e-10:
        parent = np.ones(4096)
        p_sum = 4096.0
    parent = parent / p_sum

    n = len(parent)
    parent_score = compute_score(parent)
    print(f"Parent: n={n}, score={parent_score:.8f}")

    best_score = parent_score
    best_a = parent.copy()
    best_a_display = parent_score  # Track for display

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.maximum(a, 0.0).copy()
            s = best_a.sum()
            if s > 1e-10:
                best_a = best_a / s
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # ============================================================
    # Phase 1: Continuation optimization from parent (35% budget)
    # ============================================================
    p1_budget = effective_budget * 0.35
    print(f"\n{'='*60}")
    print(f"Phase 1: Continuation from parent (budget={p1_budget:.0f}s)")
    print(f"{'='*60}")

    # First try anti-symmetric perturbation of parent
    a_anti = make_anti_perturbed(best_a, rng, strength=0.1)
    sc = compute_score(a_anti)
    try_update(sc, a_anti, "Anti-sym init")

    # Continuation from best so far
    sc, a_opt = continuation_optimize(best_a,
                                       [30, 100, 300, 1000, 3000, 10000],
                                       p1_budget, t_start)
    try_update(sc, a_opt, "Continuation P1")

    # Quick pair-swap descent
    remaining = t_start + effective_budget * 0.35 - time.monotonic()
    if remaining > 5:
        sc, a_opt = pair_swap_descent(best_a, n_sweeps=3,
                                       time_budget=min(20.0, remaining - 2), t_start=t_start)
        try_update(sc, a_opt, "PairSwap P1")

    # ============================================================
    # Phase 2: Diverse restarts (40% budget)
    # ============================================================
    p2_budget = effective_budget * 0.40
    print(f"\n{'='*60}")
    print(f"Phase 2: Diverse restarts (budget={p2_budget:.0f}s)")
    print(f"{'='*60}")

    p2_start = time.monotonic()
    p2_deadline = p2_start + p2_budget

    # Generate diverse initializations
    inits = []
    labels = []

    # Parent-based
    for s in [0.05, 0.1, 0.2, 0.3]:
        inits.append(make_parent_perturbed(best_a, rng, strength=s))
        labels.append(f"parent_pert_{s}")
    inits.append(make_anti_perturbed(best_a, rng, strength=0.15))
    labels.append("anti_parent")
    inits.append(make_anti_perturbed(best_a, rng, strength=0.25))
    labels.append("anti_parent_strong")

    # Structural
    inits.append(make_hann(n))
    labels.append("hann")
    inits.append(make_gaussian(n, 0.2))
    labels.append("gauss_0.2")
    inits.append(make_gaussian(n, 0.35))
    labels.append("gauss_0.35")
    inits.append(make_multi_freq(n, rng, n_freqs=8))
    labels.append("multifreq_8")
    inits.append(make_multi_freq(n, rng, n_freqs=20, amp_range=(0.01, 0.05)))
    labels.append("multifreq_20")
    inits.append(make_multi_freq(n, rng, n_freqs=40, amp_range=(0.005, 0.03)))
    labels.append("multifreq_40")
    inits.append(make_two_level(n, rng, density=0.4))
    labels.append("two_level")

    # Select subset based on budget
    n_restarts = len(inits)
    per_restart = p2_budget / (n_restarts + 2)

    bethas_short = [50, 200, 800]

    for idx in range(n_restarts):
        if time.monotonic() > p2_deadline - 15:
            break
        time_left = p2_deadline - time.monotonic()
        time_alloc = min(time_left * 0.85, per_restart * (1 + 0.5 * (n_restarts - idx - 1) / max(1, n_restarts - 1)))
        if time_alloc < 8:
            break

        a_init = np.maximum(inits[idx], 0.0)
        s = a_init.sum()
        if s < 1e-10:
            continue
        a_init = a_init / s

        # Quick continuation
        sc, a_opt = continuation_optimize(a_init, bethas_short, time_alloc * 0.7, t_start,
                                           use_lbfgs=(time_alloc > 30))
        try_update(sc, a_opt, labels[idx])

        # Quick pair-swap
        if time.monotonic() < p2_deadline - 5:
            t_rem = p2_deadline - time.monotonic()
            if t_rem > 3:
                sc, a_opt = pair_swap_descent(a_opt, n_sweeps=2,
                                               time_budget=min(10.0, t_rem - 2), t_start=t_start)
                try_update(sc, a_opt, labels[idx] + "+pair")

    # ============================================================
    # Phase 3: Intensive optimization of best (15% budget)
    # ============================================================
    p3_budget = effective_budget * 0.15
    print(f"\n{'='*60}")
    print(f"Phase 3: Intensive optimization (budget={p3_budget:.0f}s)")
    print(f"{'='*60}")

    remaining = t_start + effective_budget - 40 - time.monotonic()
    if remaining > 10:
        # High-precision continuation
        sc, a_opt = continuation_optimize(best_a, [3000, 10000, 30000],
                                           remaining * 0.4, t_start)
        try_update(sc, a_opt, "High-beta cont")

        # Aggressive pair-swap descent
        remaining2 = t_start + effective_budget - 40 - time.monotonic()
        if remaining2 > 5:
            sc, a_opt = pair_swap_descent(best_a, n_sweeps=8,
                                           time_budget=min(remaining2 * 0.5, 30.0), t_start=t_start)
            try_update(sc, a_opt, "Intensive pair-swap")

        # Coordinate descent
        remaining3 = t_start + effective_budget - 40 - time.monotonic()
        if remaining3 > 5:
            sc, a_opt = coord_descent_topk(best_a, n_sweeps=5, top_k=50,
                                            time_budget=min(remaining3 - 3, 30.0), t_start=t_start)
            try_update(sc, a_opt, "Intensive CD")

    # ============================================================
    # Phase 4: Try different lengths (5% budget)
    # ============================================================
    p4_budget = effective_budget * 0.05
    print(f"\n{'='*60}")
    print(f"Phase 4: Length variation (budget={p4_budget:.0f}s)")
    print(f"{'='*60}")

    n_best = len(best_a)
    for factor in [0.75, 1.0, 1.25, 1.5]:
        if time.monotonic() > t_start + effective_budget - 20:
            break
        tl = int(n_best * factor)
        if not (256 <= tl <= 20000) or tl == n_best:
            continue
        x_old = np.arange(n_best, dtype=np.float64)
        x_new = np.linspace(0, n_best - 1, tl)
        a_res = np.interp(x_new, x_old, best_a)
        a_res = np.maximum(a_res, 0.0)
        s = a_res.sum()
        if s < 1e-10:
            continue
        a_res = a_res / s
        time_alloc = min(15.0, t_start + effective_budget - 20 - time.monotonic())
        if time_alloc < 5:
            break
        sc, a_opt = continuation_optimize(a_res, [100, 500], time_alloc, t_start,
                                           use_lbfgs=False)
        try_update(sc, a_opt, f"n={tl}")

    # ============================================================
    # Phase 5: Final polish with remaining time
    # ============================================================
    print(f"\n{'='*60}")
    print(f"Phase 5: Final polish")
    print(f"{'='*60}")

    final_deadline = t_start + effective_budget - 5.0

    if time.monotonic() < final_deadline - 10:
        # One more pair-swap pass
        t_rem = final_deadline - time.monotonic()
        sc, a_opt = pair_swap_descent(best_a, n_sweeps=5,
                                       time_budget=min(t_rem * 0.5, 20.0), t_start=t_start)
        try_update(sc, a_opt, "Final pair-swap")

    if time.monotonic() < final_deadline - 5:
        # Final coordinate descent
        t_rem = final_deadline - time.monotonic()
        sc, a_opt = coord_descent_topk(best_a, n_sweeps=3, top_k=60,
                                        time_budget=min(t_rem - 3, 20.0), t_start=t_start)
        try_update(sc, a_opt, "Final CD")

    # ============================================================
    # Output
    # ============================================================
    # Denormalize: the score is scale-invariant so we can output as-is,
    # but make sure sum >= 0.01 for the evaluator
    out_a = best_a.copy()
    s = out_a.sum()
    if s < 0.01:
        out_a = out_a * (1.0 / max(s, 1e-10))

    out = [float(max(0.0, min(1000.0, x))) for x in out_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
