def subgrad_step(a, alpha):
    """One subgradient step on the simplex."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(c.argmax())
    M = c[k_star]
    S = a.sum()
    
    # Subgradient of max conv
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    for j in range(js, je + 1):
        if 2*j == k_star:
            sg[j] = a[j]  # center term
        else:
            sg[j] = 2 * a[k_star - j]
    
    # Full subgradient of score = 2n*M/S^2
    # d(score)/da = (2n/S^2)*sg - (4n*M/S^3)*1
    grad = (2*n/S**2) * sg - (4*n*M/S**3) * np.ones(n)
    
    # Multiplicative weight update (mirror descent with entropy)
    # a_new[i] ∝ a[i] * exp(-alpha * grad[i])
    log_a = np.log(np.maximum(a, 1e-30))
    log_a -= alpha * grad
    log_a -= log_a.max()  # for numerical stability
    a_new = np.exp(log_a)
    a_new = a_new / a_new.sum()
    
    return a_new, M, k_star

"""
Optimizer for minimizing 2n*max(self-conv)/(sum)^2 over non-negative sequences.
Key improvements over previous approach:
1. NO palindrome constraint (palindromic sequences are limited to score >= 2)
2. Efficient LSE-smoothed gradient via FFT cross-correlation
3. Adaptive beta continuation with subgradient "nudge" for escape
4. Multiplicative weight (mirror descent) as alternate optimizer
5. Frequency-domain inspired asymmetric initializations
6. Multi-n exploration to find optimal length
7. L-BFGS polishing in full (non-palindromic) space
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and gradient w.r.t. a.
    Score = 2n * LSE_beta(conv) / S^2
    Uses FFT for efficient gradient computation.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    p = exp_scaled / exp_sum
    L = c_max + np.log(exp_sum) / beta

    # Gradient: dF/da[i] = (4n/S^2) * h[i] - 4nL/S^3
    # where h[i] = sum_j a[j] * p[i+j] = cross-correlation computed via FFT
    # h[i] = (a_rev * p)[i + n - 1]
    M_fft = 1 << (3 * n - 2).bit_length()
    A_rev = np.fft.rfft(a[::-1], M_fft)
    P = np.fft.rfft(p, M_fft)
    conv_ap = np.fft.irfft(A_rev * P, M_fft)
    h = conv_ap[n - 1:2 * n - 1]

    S2 = S * S
    S3 = S2 * S
    grad = (4.0 * n / S2) * h - (4.0 * n * L / S3)

    score = 2.0 * n * L / S2
    return score, grad


def subgrad_info(a):
    """Get subgradient of max(conv) and related info."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    M = float(c.max())
    k_star = int(c.argmax())

    # Subgradient of (a*a)[k_star] w.r.t. a
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    for j in range(js, je + 1):
        if 2 * j == k_star:
            sg[j] = a[j]
        else:
            sg[j] = 2.0 * a[k_star - j]

    return M, k_star, sg, c


def adam_lse_optimize(a_init, beta_schedule, lr_schedule, time_limit_s, t_start,
                      normalize=True, check_interval=200):
    """Adam optimizer with LSE smoothing and scheduled beta/lr."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    a = a / a.sum()

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    t_global = 0

    best_a = a.copy()
    best_score = compute_score(a)

    n_beta = len(beta_schedule)
    total_steps_target = 50000
    steps_per_beta = max(1, total_steps_target // n_beta)

    n_evals = 0
    for bi, beta in enumerate(beta_schedule):
        lr = lr_schedule[min(bi, len(lr_schedule) - 1)]
        steps = min(steps_per_beta, total_steps_target - bi * steps_per_beta)
        for t in range(steps):
            if time.monotonic() - t_start > time_limit_s:
                break
            if time.monotonic() > t_start + time_limit_s:
                return best_score, best_a

            lr_t = lr / np.sqrt(1.0 + 0.001 * t)
            score, grad = lse_score_grad(a, beta)

            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            t_global += 1
            m_hat = m / (1.0 - b1 ** t_global)
            v_hat = v / (1.0 - b2 ** t_global)
            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, 1e10)

            if normalize:
                s = a.sum()
                if s > 1e-12:
                    a = a / s

            n_evals += 1
            if n_evals % check_interval == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        if time.monotonic() - t_start > time_limit_s:
            break

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def mw_optimize(a_init, n_iters, lr_schedule, time_limit_s, t_start,
                check_interval=50):
    """
    Multiplicative weight (mirror descent) optimizer.
    Directly targets the peak of the convolution.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit_s:
            break

        lr = lr_schedule[min(it // max(1, n_iters // len(lr_schedule)),
                             len(lr_schedule) - 1)]

        M, k_star, sg, c = subgrad_info(a)
        S = a.sum()

        # Multiplicative weight: a_new[i] ∝ a[i] * exp(-lr * sg[i] / M)
        # Normalize sg by M to make lr scale-invariant
        log_a = np.log(np.maximum(a, 1e-30))
        log_a -= lr * sg / max(M, 1e-30)
        log_a -= log_a.max()
        a = np.exp(log_a)
        a = a / a.sum()

        if (it + 1) % check_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def lbfgs_lse_optimize(a0, beta, maxiter=3000, maxfun=50000, time_limit_s=None, t_start=None):
    """L-BFGS-B with LSE smoothing (full space, no palindrome)."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    eval_count = [0]

    def obj(x):
        eval_count[0] += 1
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                raise TimeExceeded()
        x_c = np.clip(x, 0.0, 1e10)
        sc, gr = lse_score_grad(x_c, beta)
        return sc, gr

    try:
        res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                       options={'ftol': 1e-16, 'gtol': 1e-14,
                                'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30})
        a_out = np.clip(res.x, 0.0, 1000.0)
    except (TimeExceeded, Exception):
        # Use best we can get
        a_out = a0.copy()

    if a_out.sum() < 1e-12:
        a_out[:] = 1.0 / n
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def subgrad_adaptive(a_init, n_iters, lr0, time_limit_s, t_start):
    """
    Adaptive subgradient with geometric decay and occasional restarts.
    Uses the exact max (no smoothing).
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit_s:
            break

        lr = lr0 / (1.0 + 0.01 * it)
        M, k_star, sg, c = subgrad_info(a)
        S = a.sum()

        # Subgradient of score = 2n*M/S^2
        # d(score)/da = (2n/S^2)*sg - (4n*M/S^3)*1
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n / S2) * sg - (4.0 * n * M / S3)

        # Projected gradient step on simplex
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            a_new = a.copy()
        else:
            a_new = a_new / s

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
        a = a_new

    return best_score, best_a


def coord_descent_optimize(a_init, n_sweeps=50, time_limit_s=None, t_start=None):
    """
    Coordinate descent targeting peak reduction.
    Tries to reduce elements contributing to the peak and redistribute.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = a / a.sum()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_limit_s if (time_limit_s is not None and t_start is not None) else np.inf
    rng = np.random.default_rng(12345)

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        k_star = int(c.argmax())
        M = c[k_star]
        S = a.sum()
        score = 2.0 * n * M / (S * S)

        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)

        # Try reducing peak contributors
        indices = list(range(js, je + 1))
        rng.shuffle(indices)

        for j in indices:
            if time.monotonic() > deadline:
                break
            if a[j] < 1e-10:
                continue

            a_old = a[j]
            S_old = S
            # Try different reduction factors
            best_local = score
            best_frac = 1.0
            for frac in [0.7, 0.5, 0.3, 0.1]:
                a[j] = a_old * frac
                removed = a_old - a[j]
                # Redistribute to random non-contributing elements
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je + 1] = False
                nc_idx = np.where(non_contrib)[0]
                if len(nc_idx) > 0:
                    # Distribute to a few random non-contributing elements
                    n_target = min(5, len(nc_idx))
                    targets = rng.choice(nc_idx, size=n_target, replace=False)
                    a[targets] += removed / n_target

                sc = compute_score(a)
                if sc < best_local:
                    best_local = sc
                    best_frac = frac
            if best_frac < 1.0:
                a[j] = a_old * best_frac
                removed = a_old * (1 - best_frac)
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je + 1] = False
                nc_idx = np.where(non_contrib)[0]
                if len(nc_idx) > 0:
                    n_target = min(5, len(nc_idx))
                    targets = rng.choice(nc_idx, size=n_target, replace=False)
                    a[targets] += removed / n_target

        if score < best_score:
            best_score = score
            best_a = a.copy()

    return best_score, best_a


def generate_inits(rng, n):
    """Generate diverse asymmetric initializations."""
    inits = []
    x = np.arange(n) / n

    # Uniform
    inits.append(np.ones(n) / n)

    # Off-center Gaussian (asymmetric)
    for center in [0.3, 0.4, 0.6, 0.7]:
        sigma = 0.12 + 0.08 * rng.random()
        g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
        inits.append(g / g.sum())

    # Exponential decay (strongly asymmetric)
    for tau in [2.0, 4.0, 6.0]:
        e = np.exp(-x / (tau / n * n))  # exp(-j * n / tau)
        e = np.exp(-np.arange(n) / tau)
        inits.append(e / e.sum())

    # Two-bump asymmetric
    c1 = 0.2 + 0.1 * rng.random()
    c2 = 0.6 + 0.15 * rng.random()
    s1 = 0.05 + 0.05 * rng.random()
    s2 = 0.08 + 0.05 * rng.random()
    amp1 = 1.0 + rng.random()
    amp2 = 0.5 + 0.5 * rng.random()
    b = amp1 * np.exp(-0.5 * ((x - c1) / s1) ** 2) + amp2 * np.exp(-0.5 * ((x - c2) / s2) ** 2)
    inits.append(b / b.sum())

    # Raised cosine (shifted)
    for shift in [0.0, 0.1, 0.2]:
        rc = 0.5 * (1 + np.cos(2 * np.pi * (x - shift)))
        rc = np.maximum(rc, 0.01)
        inits.append(rc / rc.sum())

    # Sinusoidal modulation (asymmetric carrier)
    freq = 1 + 3 * rng.random()
    phase = 2 * np.pi * rng.random()
    env = np.exp(-((x - 0.4) / 0.3) ** 2)
    s = env * (1.0 + 0.4 * np.cos(2 * np.pi * freq * x + phase))
    s = np.maximum(s, 0.01)
    inits.append(s / s.sum())

    # Smoothed random (asymmetric)
    r = np.abs(rng.standard_normal(n)) + 0.3
    # Asymmetric smoothing kernel
    k_left = max(3, n // 100)
    k = np.exp(-np.linspace(0, 5, k_left))
    k = k / k.sum()
    r_smooth = np.convolve(r, k, mode='same')
    inits.append(r_smooth / r_smooth.sum())

    # Frequency-domain design: flat magnitude, random phase
    # Generate in frequency domain for "flat autocorrelation" target
    n_fft = 1 << (2 * n - 1).bit_length()
    # Target: |A(w)|^2 proportional to Dirichlet kernel (for flat conv)
    freqs = np.fft.rfftfreq(n_fft) * 2 * np.pi
    # Use a broad low-pass shape
    target_mag = np.exp(-0.5 * (freqs / (2.0 / n)) ** 2)
    # Add some structure
    target_mag *= (1.0 + 0.3 * np.cos(4.0 * np.pi * x[np.minimum(np.abs(np.round(freqs * n_fft / (2*np.pi)).astype(int)), n-1)] + rng.random()))
    target_mag = np.maximum(target_mag, 1e-6)

    phase = rng.uniform(0, 2 * np.pi, size=len(target_mag))
    phase[0] = 0  # DC must be real
    # Enforce Hermitian symmetry for real output
    A = target_mag * np.exp(1j * phase)
    a_fd = np.fft.irfft(A, n_fft)[:n]
    a_fd = np.maximum(a_fd, 0)
    if a_fd.sum() > 1e-6:
        inits.append(a_fd / a_fd.sum())

    # "Barkhausen" sequence: binary-like with specific structure
    # Random binary sequence smoothed
    b_seq = (rng.random(n) > 0.4).astype(float)
    k = max(5, n // 80)
    kernel = np.ones(k) / k
    b_smooth = np.convolve(b_seq, kernel, mode='same')
    b_smooth = np.maximum(b_smooth, 0.01)
    inits.append(b_smooth / b_smooth.sum())

    # Wedge / triangular (asymmetric)
    peak_pos = int(n * (0.3 + 0.4 * rng.random()))
    wedge = np.zeros(n)
    for i in range(n):
        wedge[i] = min(i / peak_pos, (n - 1 - i) / max(1, n - 1 - peak_pos))
    wedge = np.maximum(wedge, 0.01)
    inits.append(wedge / wedge.sum())

    return [np.clip(x, 1e-10, 1000.0) for x in inits]


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)

    if phase1_best:
        best_n = min(phase1_best.keys(), key=lambda k: phase1_best[k][0])
    elif parent is not None:
        best_n = len(parent)
    else:
        best_n = 4096

    print(f"\n  Selected n={best_n} for main optimization")

    # ==========================================
    # Phase 2: Main blended optimization (45% time)
    # ==========================================
    p2_time = effective_budget * 0.45
    print(f"\n{'=' * 60}")
    print(f"Phase 2: Main blended optimization (budget={p2_time:.0f}s)")
    print(f"{'=' * 60}")

    if best_a is not None:
        a_work = best_a.copy()
    else:
        a_work = np.ones(best_n) / best_n

    # Ensure correct length
    if len(a_work) != best_n:
        a_work = resample_sequence(a_work, best_n)

    t0 = time.monotonic()
    sc, a_opt = blended_optimize(a_work, p2_time, t_start)
    try_update(sc, a_opt, "phase2 blended")
    best_a = a_opt

    # ==========================================
    # Phase 3: L-BFGS polishing (20% time)
    # ==========================================
    p3_time = effective_budget * 0.20
    print(f"\n{'=' * 60}")
    print(f"Phase 3: L-BFGS polishing (budget={p3_time:.0f}s)")
    print(f"{'=' * 60}")

    t0 = time.monotonic()
    for beta in [1000, 3000, 8000, 15000]:
        remaining = t_start + p3_time - time.monotonic()
        if remaining < 5:
            break
        t0 = time.monotonic()
        sc, a_opt = lbfgs_lse_optimize(best_a, beta, maxiter=3000, maxfun=40000,
                                        time_limit_s=min(remaining - 2, p3_time / 4),
                                        t_start=t_start)
        try_update(sc, a_opt, f"LBFGS beta={beta}")
        print(f"  LBFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==========================================
    # Phase 4: MW + Coord Descent (10% time)
    # ==========================================
    p4_time = effective_budget * 0.10
    print(f"\n{'=' * 60}")
    print(f"Phase 4: MW + Coord Descent (budget={p4_time:.0f}s)")
    print(f"{'=' * 60}")

    # MW refinement
    t0 = time.monotonic()
    mw_alloc = p4_time * 0.5
    sc, a_opt = mw_optimize(best_a, 3000, [0.3, 0.1, 0.05, 0.02], mw_alloc, t0)
    try_update(sc, a_opt, "phase4 MW")
    print(f"  MW: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Coordinate descent
    t0 = time.monotonic()
    cd_alloc = p4_time * 0.5
    sc, a_opt = coord_descent_optimize(best_a, n_sweeps=30,
                                        time_limit_s=cd_alloc, t_start=t0)
    try_update(sc, a_opt, "phase4 CD")
    print(f"  CD: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==========================================
    # Phase 5: Restart perturbations (4% time)
    # ==========================================
    p5_time = effective_budget * 0.04
    print(f"\n{'=' * 60}")
    print(f"Phase 5: Restarts (budget={p5_time:.0f}s)")
    print(f"{'=' * 60}")

    n_restarts = 4
    restart_budget = p5_time / n_restarts

    for r in range(n_restarts):
        if time.monotonic() > t_start + effective_budget - 20:
            break

        # Peak-avoidance perturbation
        c = fast_conv(best_a)
        k_star = int(c.argmax())
        n_cur = len(best_a)
        js = max(0, k_star - n_cur + 1)
        je = min(k_star, n_cur - 1)

        a_pert = best_a.copy()
        # Reduce peak contributors
        a_pert[js:je + 1] *= (0.5 + 0.3 * rng.random())
        # Boost non-contributors
        non_peak = np.ones(n_cur, dtype=bool)
        non_peak[js:je + 1] = False
        if non_peak.sum() > 0:
            a_pert[non_peak] *= (1.0 + 0.2 * rng.random())
        a_pert = np.maximum(a_pert, 0)
        a_pert = a_pert / a_pert.sum()

        # Quick optimize
        t_rem = t_start + effective_budget - 20 - time.monotonic()
        if t_rem < 3:
            break
        alloc = min(restart_budget, t_rem - 2)
        sc, a_r = adam_lse_optimize(a_pert, [100, 500], [0.005, 0.002],
                                     alloc, time.monotonic())
        try_update(sc, a_r, f"restart#{r}")

    # ==========================================
    # Phase 6: Final intensive polish (1% time + remaining)
    # ==========================================
    print(f"\n{'=' * 60}")
    print(f"Phase 6: Final Polish")
    print(f"{'=' * 60}")

    final_deadline = t_start + effective_budget - 5.0

    # One more L-BFGS with very high beta
    if time.monotonic() < final_deadline:
        t0 = time.monotonic()
        remaining = final_deadline - time.monotonic()
        sc, a_opt = lbfgs_lse_optimize(best_a, 20000, maxiter=5000, maxfun=80000,
                                        time_limit_s=min(remaining - 5, 60),
                                        t_start=t_start)
        try_update(sc, a_opt, "final LBFGS")
        print(f"  Final LBFGS: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Final MW
    if time.monotonic() < final_deadline - 2:
        t0 = time.monotonic()
        remaining = final_deadline - time.monotonic() - 2
        sc, a_opt = mw_optimize(best_a, 2000, [0.05, 0.02, 0.01],
                                 remaining, time.monotonic())
        try_update(sc, a_opt, "final MW")
        print(f"  Final MW: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==========================================
    # Final verification
    # ==========================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
