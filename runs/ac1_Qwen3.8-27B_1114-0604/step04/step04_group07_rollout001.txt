import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Linear cross-convolution via FFT."""
    n_a, n_b = len(a), len(b)
    M = n_a + n_b - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    if exp_sum < 1e-300:
        return 1e10, np.zeros(n)
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # Gradient of LSE w.r.t. a: p convolved with reversed a, windowed
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def subgrad_score(a):
    """Exact score and subgradient at the peak lag."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(c.argmax())
    score = 2.0 * n * M / (S * S)
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
    return score, grad


def lse_pal_grad(half_a, beta):

    n_half = len(half_a)

    sc, grad_full = lse_score_grad(a, beta)
    grad_half = grad_full[:n_half] + grad_full[::-1][:n_half]

def adam_lse(a_init, beta_schedule, lr, time_limit_s, t_start,
             normalize=True, perturb_every=0, perturb_scale=0.0,
             verbose_interval=10000):
    """Adam with LSE smoothing and scheduled beta, with optional perturbation."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    t_count = 0

    best_a = a.copy()
    best_score = compute_score(a)

    n_beta = len(beta_schedule)
    deadlines = []
    total_deadline = t_start + time_limit_s
    # Allocate time proportionally: later betas get more time
    time_weights = np.linspace(0.5, 2.0, n_beta)
    time_weights = time_weights / time_weights.sum()
    cum_time = 0.0
    for w in time_weights:
        cum_time += w * time_limit_s
        deadlines.append(t_start + cum_time)

    for bi, beta in enumerate(beta_schedule):
        beta_deadline = min(deadlines[bi], total_deadline)
        while time.monotonic() < beta_deadline:
            t_count += 1
            lr_t = lr / (1.0 + 1e-4 * t_count)
            score, grad = lse_score_grad(a, beta)

            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            m_hat = m / (1.0 - b1 ** t_count)
            v_hat = v / (1.0 - b2 ** t_count)
            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, 1e10)

            if normalize:
                s = a.sum()
                if s > 1e-12:
                    a = a / s

            # Periodic perturbation
            if perturb_every > 0 and t_count % perturb_every == 0 and perturb_scale > 0:
                noise = perturb_scale * np.random.standard_normal(n)
                a = a + noise
                a = np.clip(a, 0.0, 1e10)
                s = a.sum()
                if s > 1e-12:
                    a = a / s

            if t_count % verbose_interval == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def subgrad_method(a_init, time_limit_s, t_start, lr0=0.1, decay=0.9999,
                   normalize=True, verbose_interval=1000):
    """Subgradient method with step-size decay for the exact (non-smooth) problem."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n

    best_a = a.copy()
    best_score = compute_score(a)
    lr = lr0
    t_count = 0
    deadline = t_start + time_limit_s

    while time.monotonic() < deadline:
        t_count += 1
        score, grad = subgrad_score(a)
        if np.isinf(score):
            break

        a = a - lr * grad
        a = np.clip(a, 0.0, 1e10)

        if normalize:
            s = a.sum()
            if s > 1e-12:
                a = a / s

        lr *= decay

        if t_count % verbose_interval == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a

        deadlines.append(t_start + cum_time)

    for bi, beta in enumerate(beta_schedule):
        beta_deadline = min(deadlines[bi], total_deadline)
        while time.monotonic() < beta_deadline:
            t_count += 1
            lr_t = lr / (1.0 + 1e-4 * t_count)
            score, grad = lse_pal_grad(h, beta)

            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            m_hat = m / (1.0 - b1 ** t_count)
            v_hat = v / (1.0 - b2 ** t_count)
            h = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            h = np.clip(h, 0.0, 1e10)

            s = h.sum()
            if s > 1e-12:
                h = h / s

            if perturb_every > 0 and t_count % perturb_every == 0 and perturb_scale > 0:
                noise = perturb_scale * np.random.standard_normal(n_half)
                h = h + noise
                h = np.clip(h, 0.0, 1e10)
                s = h.sum()
                if s > 1e-12:
                    h = h / s

            if t_count % verbose_interval == 0:
                sc = pal_score(h)
                if sc < best_score:
                    best_score = sc
                    best_h = h.copy()

    sc_final = pal_score(h)
    if sc_final < best_score:
        best_score = sc_final
        best_h = h.copy()

    return best_score, best_h


def lbfgs_lse(a0, beta, maxiter=10000, maxfun=200000):
    """L-BFGS-B with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        x_c = np.clip(x, 0.0, 1e10)
        sc, gr = lse_score_grad(x_c, beta)
        return sc, gr

    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-18, 'gtol': 1e-14,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def lbfgs_lse_pal(half0, beta, maxiter=10000, maxfun=200000):
    """L-BFGS-B in palindrome half-space with LSE smoothing."""
    n_half = len(half0)
    h0 = np.clip(half0.astype(np.float64), 0.0, 1000.0).copy()
    if h0.sum() < 1e-6:
        h0[:] = 1.0 / n_half
    bounds = [(0.0, 1000.0)] * n_half

    def obj(h):
        h_c = np.clip(h, 0.0, 1e10)
        sc, gr = lse_pal_grad(h_c, beta)
        return sc, gr

    res = minimize(obj, h0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-18, 'gtol': 1e-14,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    h_out = np.clip(res.x, 0.0, 1000.0)
    a_full = np.concatenate([h_out, h_out[::-1]])
    exact_sc = compute_score(a_full)
    return exact_sc, h_out


def peak_targeting_cd(a, n_sweeps, time_limit_s, t_start):
    """Coordinate descent targeting the peak convolution lag."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        k_star = int(c.argmax())
        score = compute_score(a)

        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)

        # Try reducing peak contributors
        for j in range(js, je + 1):
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            if a_old < 1e-10:
                continue
            best_val = a_old
            best_sc = score
            for frac in [0.9, 0.8, 0.7, 0.5, 0.3]:
                a[j] = a_old * frac
                removed = a_old - a[j]
                non_contrib = np.ones(n, dtype=bool)
                non_contrib[js:je + 1] = False
                nc_count = non_contrib.sum()
                if nc_count > 0:
                    a[non_contrib] += removed / nc_count
                sc = compute_score(a)
                if sc < best_sc:
                    best_sc = sc
                    best_val = a[j]
                else:
                    a[j] = best_val
                    a[non_contrib] -= removed / nc_count if nc_count > 0 else 0
                    break
            a[j] = best_val
            if best_sc < score:
                score = best_sc
                if sc < best_score:
                    best_score = best_sc

        # Try increasing non-peak elements
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[js:je + 1] = False
        nc_idx = np.where(non_contrib)[0]
        if len(nc_idx) > 0:
            for _ in range(min(5, len(nc_idx))):
                j = nc_idx[np.random.randint(len(nc_idx))]
                a_old = a[j]
                S = a.sum()
                best_val = a_old
                best_sc = score
                for mult in [1.1, 1.3, 1.5, 2.0]:
                    a[j] = a_old * mult
                    sc = compute_score(a)
                    if sc < best_sc:
                        best_sc = sc
                        best_val = a[j]
                    else:
                        a[j] = best_val
                        break
                a[j] = best_val
                if best_sc < score:
                    score = best_sc
                    if best_sc < best_score:
                        best_score = best_sc

    return best_score, a


def log_space_adam(a_init, beta_schedule, lr, time_limit_s, t_start):
    """Adam in log-space: optimize log(a) with LSE smoothing."""
    n = len(a_init)
    log_a = np.log(np.clip(a_init, 1e-10, 1e10)).astype(np.float64)

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    t_count = 0

    a = np.exp(log_a)
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)

    n_beta = len(beta_schedule)
    total_deadline = t_start + time_limit_s
    time_weights = np.linspace(0.5, 2.0, n_beta)
    time_weights = time_weights / time_weights.sum()
    deadlines = []
    cum_time = 0.0
    for w in time_weights:
        cum_time += w * time_limit_s
        deadlines.append(t_start + cum_time)

    for bi, beta in enumerate(beta_schedule):
        beta_deadline = min(deadlines[bi], total_deadline)
        while time.monotonic() < beta_deadline:
            t_count += 1
            lr_t = lr / (1.0 + 1e-4 * t_count)

            # Compute score and grad in a-space
            sc, grad_a = lse_score_grad(a, beta)
            # Chain rule: d/d(log_a) = a * d/da
            grad_log = grad_a * a

            m = b1 * m + (1.0 - b1) * grad_log
            v = b2 * v + (1.0 - b2) * grad_log * grad_log
            m_hat = m / (1.0 - b1 ** t_count)
            v_hat = v / (1.0 - b2 ** t_count)
            log_a = log_a - lr_t * m_hat / (np.sqrt(v_hat) + eps)

            a = np.exp(log_a - np.max(log_a))
            a = a / a.sum()

            if t_count % 10000 == 0:
                sc_exact = compute_score(a)
                if sc_exact < best_score:
                    best_score = sc_exact
                    best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def perturbation_search(a_best, n_perturbs, time_per_perturb, t_start):
    """Try random perturbations of the best solution and re-optimize."""
    n = len(a_best)
    best_score = compute_score(a_best)
    best_a = a_best.copy()
    deadline = t_start + n_perturbs * time_per_perturb

    for p in range(n_perturbs):
        if time.monotonic() > deadline:
            break

        # Generate perturbation
        strategy = p % 4
        if strategy == 0:
            # Gaussian in log-space
            log_a = np.log(np.clip(a_best, 1e-10, 1e10))
            scale = 0.03 + 0.04 * np.random.random()
            log_pert = log_a + scale * np.random.standard_normal(n)
            a_pert = np.exp(log_pert - np.max(log_pert))
            a_pert = a_pert / a_pert.sum()
        elif strategy == 1:
            # Peak avoidance
            c = fast_conv(a_best)
            k_star = int(c.argmax())
            js = max(0, k_star - n + 1)
            je = min(k_star, n - 1)
            a_pert = a_best.copy()
            a_pert[js:je + 1] *= (0.4 + 0.3 * np.random.random())
            non_peak = np.ones(n, dtype=bool)
            non_peak[js:je + 1] = False
            if non_peak.sum() > 0:
                a_pert[non_peak] *= (1.0 + 0.05 * np.random.random())
            a_pert = np.clip(a_pert, 0, 1000)
            a_pert = a_pert / a_pert.sum()
        elif strategy == 2:
            # Fourier modulation
            x = np.arange(n) / n
            freqs = np.random.uniform(1, 8, size=3)
            phases = np.random.uniform(0, 2 * np.pi, size=3)
            amps = np.random.uniform(0.02, 0.06, size=3)
            mod = 1.0
            for f, ph, am in zip(freqs, phases, amps):
                mod += am * np.cos(2 * np.pi * f * x + ph)
            a_pert = a_best * np.maximum(mod, 0.1)
            a_pert = a_pert / a_pert.sum()
        else:
            # Block modification
            n_blocks = 4 + int(4 * np.random.random())
            a_pert = a_best.copy()
            block_size = n // n_blocks
            for b in range(n_blocks):
                s = b * block_size
                e = min(s + block_size, n)
                a_pert[s:e] *= np.random.uniform(0.5, 1.5)
            a_pert = a_pert / a_pert.sum()

        # Quick Adam refinement
        t0 = time.monotonic()
        sc, a_opt = adam_lse(a_pert, [50, 200, 500], lr=0.005,
                             time_limit_s=time_per_perturb * 0.8, t_start=t0)

        if sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
            print(f"    Perturb #{p} (strat {strategy}): NEW BEST {sc:.8f}")

    return best_score, best_a


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase optimizer for minimizing 2n*max(self-conv)/(sum)^2.
    
    Phases:
    1. Quick screening with LSE-Adam from diverse initializations (8% time)
    2. Main LSE-Adam continuation with fine beta schedule (35% time)
    3. Log-space Adam track (10% time)
    4. Subgradient method (8% time)
    5. Peak-targeting coordinate descent (7% time)
    6. Perturbation search / random restarts (15% time)
    7. Final L-BFGS polish (12% time)
    8. Final coordinate descent (5% time)
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 15.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                parent = parent / parent.sum()
    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def try_update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # ==========================================
    # Phase 1: Quick screening (8% time)
    # ==========================================
    p1_time = effective_budget * 0.08
    print(f"\n{'=' * 60}")
    print(f"Phase 1: Quick screening (budget={p1_time:.0f}s)")
    print(f"{'=' * 60}")

    if parent is not None:
        n_work = len(parent)
        print(f"Parent: n={n_work}, score={compute_score(parent):.8f}")
        try_update(compute_score(parent), parent, "parent")
    else:
        n_work = 4096
        parent = np.ones(n_work) / n_work
        print("No parent available, starting fresh with uniform n=4096.")

    # Generate diverse initializations
    n = n_work
    x = np.arange(n) / n
    inits = []
    inits.append(parent.copy())

    # Gaussian
    for sigma in [0.1, 0.15, 0.2, 0.25]:
        g = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        inits.append(g / g.sum())

    # Raised cosine
    rc = 0.5 * (1 + np.cos(2 * np.pi * x))
    inits.append(rc / rc.sum())

    # Triangular
    tri = np.maximum(0, 1 - np.abs(2 * x - 1))
    if tri.sum() > 0:
        inits.append(tri / tri.sum())

    # Sinusoidal
    for freq in [3, 5, 7]:
        s = 1.0 + 0.2 * np.cos(2 * np.pi * freq * x)
        s = np.maximum(s, 0.01)
        inits.append(s / s.sum())

    # Random smooth
    for _ in range(3):
        r = np.abs(rng.standard_normal(n)) + 0.5
        k = max(3, n // 50)
        r_smooth = np.convolve(r, np.ones(k) / k, mode='same')
        inits.append(r_smooth / r_smooth.sum())

    # Palindromic random
    for _ in range(2):
        half_n = n // 2
        h = np.abs(rng.standard_normal(half_n)) + 0.5
        h_s = np.convolve(h, np.ones(max(3, half_n // 30)) / max(3, half_n // 30), mode='same')
        pal = np.concatenate([h_s, h_s[::-1]])
        inits.append(pal / pal.sum())

    # Quick Adam screening
    for i, init in enumerate(inits):
        if time.monotonic() > t_start + p1_time - 2:
            break
        sc, a_opt = adam_lse(init, [20, 80, 200], lr=0.01,
                             time_limit_s=min(3.0, p1_time / len(inits) * 1.5),
                             t_start=time.monotonic() - 0.1)
        try_update(sc, a_opt, f"screen init {i}")

    print(f"  Best after Phase 1: {best_score:.8f}")

    # ==========================================
    # Phase 2: Main LSE-Adam continuation (35% time)
    # ==========================================
    p2_time = effective_budget * 0.35
    print(f"\n{'=' * 60}")
    print(f"Phase 2: LSE-Adam continuation (budget={p2_time:.0f}s)")
    print(f"{'=' * 60}")

    # Track A: Non-palindromic
    # Fine beta schedule with more time at high beta
    beta_schedule_A = [10, 30, 80, 200, 500, 1000, 2000, 4000, 8000]
    lr_A = 0.008

    # Split time: 60% without perturbation, 40% with perturbation
    t_no_pert = p2_time * 0.6
    t_pert = p2_time * 0.4

    # First part: no perturbation
    sc, a_opt = adam_lse(best_a, beta_schedule_A[:6], lr=lr_A,
                         time_limit_s=t_no_pert, t_start=time.monotonic() - 0.1)
    try_update(sc, a_opt, "Adam main no-pert")

    # Second part: with perturbation at high beta
    beta_schedule_A2 = [500, 1000, 2000, 4000, 8000, 12000]
    sc, a_opt = adam_lse(best_a, beta_schedule_A2, lr=0.003,
                         time_limit_s=t_pert, t_start=time.monotonic() - 0.1,
                         perturb_every=20000, perturb_scale=0.002)
    try_update(sc, a_opt, "Adam main pert")

    # Track B: Palindromic
    n_even = len(best_a)
    if n_even % 2 != 0:
        n_even += 1
        best_a = resample_sequence(best_a, n_even)
    n_half = n_even // 2
    half_best = (best_a[:n_half] + best_a[::-1][:n_half]) / 2.0
    half_best = half_best / half_best.sum()

    # Quick check if palindrome is already good
    pal_sc = pal_score(half_best)
    try_update(pal_sc, np.concatenate([half_best, half_best[::-1]]), "palindromized")

    beta_schedule_B = [20, 60, 150, 400, 1000, 2500, 5000, 10000]
    t_pal = min(p2_time * 0.3, effective_budget * 0.1)
    if time.monotonic() < t_start + p1_time + p2_time:
        remaining_pal = t_start + p1_time + p2_time - time.monotonic()
        if remaining_pal > 5:
            sc, h_opt = adam_lse_pal(half_best, beta_schedule_B, lr=0.005,
                                     time_limit_s=remaining_pal, t_start=time.monotonic() - 0.1)
            a_full = np.concatenate([h_opt, h_opt[::-1]])
            try_update(sc, a_full, "Adam pal")

    print(f"  Best after Phase 2: {best_score:.8f}")

    # ==========================================
    # Phase 3: Log-space Adam (10% time)
    # ==========================================
    p3_time = effective_budget * 0.10
    print(f"\n{'=' * 60}")
    print(f"Phase 3: Log-space Adam (budget={p3_time:.0f}s)")
    print(f"{'=' * 60}")

    remaining3 = t_start + p1_time + p2_time + p3_time - time.monotonic()
    if remaining3 > 5:
        sc, a_opt = log_space_adam(best_a, [30, 100, 300, 800, 2000], lr=0.01,
                                   time_limit_s=remaining3, t_start=time.monotonic())
        try_update(sc, a_opt, "Log-space Adam")

    print(f"  Best after Phase 3: {best_score:.8f}")

    # ==========================================
    # Phase 4: Subgradient method (8% time)
    # ==========================================
    p4_time = effective_budget * 0.08
    print(f"\n{'=' * 60}")
    print(f"Phase 4: Subgradient method (budget={p4_time:.0f}s)")
    print(f"{'=' * 60}")

    remaining4 = t_start + p1_time + p2_time + p3_time + p4_time - time.monotonic()
    if remaining4 > 5:
        sc, a_opt = subgrad_method(best_a, time_limit_s=remaining4,
                                   t_start=time.monotonic(), lr0=0.05, decay=0.99995)
        try_update(sc, a_opt, "Subgradient")

    print(f"  Best after Phase 4: {best_score:.8f}")

    # ==========================================
    # Phase 5: Peak-targeting coordinate descent (7% time)
    # ==========================================
    p5_time = effective_budget * 0.07
    print(f"\n{'=' * 60}")
    print(f"Phase 5: Peak-targeting CD (budget={p5_time:.0f}s)")
    print(f"{'=' * 60}")

    remaining5 = t_start + p1_time + p2_time + p3_time + p4_time + p5_time - time.monotonic()
    if remaining5 > 5:
        sc, a_opt = peak_targeting_cd(best_a, n_sweeps=200,
                                      time_limit_s=remaining5, t_start=time.monotonic())
        try_update(sc, a_opt, "Peak CD")

    print(f"  Best after Phase 5: {best_score:.8f}")

    # ==========================================
    # Phase 6: Perturbation search (15% time)
    # ==========================================
    p6_time = effective_budget * 0.15
    print(f"\n{'=' * 60}")
    print(f"Phase 6: Perturbation search (budget={p6_time:.0f}s)")
    print(f"{'=' * 60}")

    n_perturbs = 12
    time_per_perturb = p6_time / n_perturbs
    remaining6 = t_start + p1_time + p2_time + p3_time + p4_time + p5_time + p6_time - time.monotonic()
    if remaining6 > 10:
        sc, a_opt = perturbation_search(best_a, n_perturbs,
                                        time_per_perturb=min(time_per_perturb, remaining6 / n_perturbs),
                                        t_start=time.monotonic())
        try_update(sc, a_opt, "Perturbation search")

    print(f"  Best after Phase 6: {best_score:.8f}")

    # ==========================================
    # Phase 7: Final L-BFGS polish (12% time)
    # ==========================================
    p7_time = effective_budget * 0.12
    print(f"\n{'=' * 60}")
    print(f"Phase 7: L-BFGS polish (budget={p7_time:.0f}s)")
    print(f"{'=' * 60}")

    remaining7 = t_start + p1_time + p2_time + p3_time + p4_time + p5_time + p6_time + p7_time - time.monotonic()

    # L-BFGS on full space
    for beta in [1000, 3000, 8000]:
        if time.monotonic() > t_start + effective_budget - 10:
            break
        remaining = t_start + effective_budget - 10 - time.monotonic()
        if remaining < 3:
            break
        t0 = time.monotonic()
        maxfun = min(50000, int(remaining * 500))
        sc, a_opt = lbfgs_lse(best_a, beta, maxiter=min(3000, maxfun), maxfun=maxfun)
        try_update(sc, a_opt, f"LBFGS beta={beta}")
        print(f"  LBFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # L-BFGS on palindrome space
    n_even2 = len(best_a)
    if n_even2 % 2 != 0:
        n_even2 += 1
        best_a_temp = resample_sequence(best_a, n_even2)
    else:
        best_a_temp = best_a
    n_half2 = n_even2 // 2
    half_best2 = (best_a_temp[:n_half2] + best_a_temp[::-1][:n_half2]) / 2.0
    half_best2 = half_best2 / half_best2.sum()

    for beta in [1000, 5000]:
        if time.monotonic() > t_start + effective_budget - 5:
            break
        t0 = time.monotonic()
        sc, h_opt = lbfgs_lse_pal(half_best2, beta, maxiter=3000, maxfun=50000)
        a_full = np.concatenate([h_opt, h_opt[::-1]])
        try_update(sc, a_full, f"LBFGS-pal beta={beta}")
        print(f"  LBFGS-pal beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s}")

    print(f"  Best after Phase 7: {best_score:.8f}")

    # ==========================================
    # Phase 8: Final coordinate descent (5% time)
    # ==========================================
    p8_time = effective_budget * 0.05
    print(f"\n{'=' * 60}")
    print(f"Phase 8: Final CD (budget={p8_time:.0f}s)")
    print(f"{'=' * 60}")

    remaining8 = t_start + effective_budget - 3.0 - time.monotonic()
    if remaining8 > 3:
        sc, a_opt = peak_targeting_cd(best_a, n_sweeps=100,
                                      time_limit_s=remaining8, t_start=time.monotonic())
        try_update(sc, a_opt, "Final CD")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
