Actually, a cleaner spectral equalization:
1. Compute F = FFT(a, padded to 4n or so)
2. Compute P = |F|^2
3. Find the peaks in P
4. Reduce the peaks in P: P_new = P * (1 - α * (P - P_target) / P_max) for P > P_target
5. Recover a_new = IFFT(sqrt(P_new)) (taking the minimum phase factor)
6. Project to non-negative

But step 5 is tricky because we need the phase. The standard approach is:
- Keep the phase of F from the previous iteration
- Only modify the magnitude: F_new = |F_new| * exp(i*angle(F))
- a_new = IFFT(F_new)
- Project: a_new = max(0, a_new)

This is the "alternating projections" or "iterative amplitude modification" (IAM) method.

def fast_conv(a):

def spectral_equalize(a, n_iters=20, alpha=0.5):
    """Spectral equalization via iterative amplitude modification."""
    n = len(a)
    a = a.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    M_fft = 1 << (2*n-1).bit_length()
    
    for it in range(n_iters):
        # Compute spectrum
        F = np.fft.rfft(a, M_fft)
        P = np.abs(F)**2
        
        # Target: flatten P
        P_mean = P.mean()
        P_max = P.max()
        
        # Reduce peaks: P_new = P * (1 - alpha * max(0, P - P_mean) / P_max)
        excess = np.maximum(0, P - P_mean)
        P_new = P * (1.0 - alpha * excess / P_max)
        
        # Recover with same phase
        magnitude_new = np.sqrt(np.maximum(P_new, 0))
        angle_F = np.angle(F)
        F_new = magnitude_new * np.exp(1j * angle_F)
        
        # Inverse FFT
        a_new = np.fft.irfft(F_new, M_fft)[:n]
        
        # Project to non-negative
        a_new = np.maximum(0, a_new)
        
        # Normalize
        S_new = a_new.sum()
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        a = a_new
    
    return best_a

    n = len(a)

    k_star = int(np.argmax(r))

    # Subgradient direction: reduce a[j] for j contributing to r[k_star]
    # and increase a[j] for j not contributing

    j_end = min(k_star, n - 1)

    # grad[j] = 2n * (2*a[k_star-j] / S^2 - 2*r[k_star]/S^3) for j in [j_start, j_end]
    # grad[j] = 2n * (-2*r[k_star]/S^3) for j not in [j_start, j_end]

    # Direction: d[j] = -a[k_star-j] for j in [j_start, j_end] (reduce contributors)
    #            d[j] = +c for j not in [j_start, j_end] (increase non-contributors)

    d = np.zeros(n)
    for j in range(j_start, j_end + 1):
        d[j] = -a[k_star - j]  # negative for contributors
    
    n_contrib = len(range(j_start, j_end + 1))
    n_noncontrib = n - n_contrib
    sum_contrib = -d[j_start:j_end+1].sum()  # positive
    d[j_start:j_end+1] = sum_contrib / n_contrib  # wait, this should be negative

    #               + 2 * sum_j a[j] * d[k_star-j] for j in [j_start, j_end]

    grad = np.zeros(n)

    best_a = a.copy()

def peak_reduction(a, n_steps=100, verbose=False):
    """Reduce the peak of the autocorrelation via iterative subgradient steps."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for step in range(n_steps):
        r = fast_conv(a)
        k_star = int(np.argmax(r))
        S_cur = a.sum()
        sc = compute_score(a)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Compute subgradient direction
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        S2 = S_cur * S_cur
        S3 = S2 * S_cur
        Mak = r[k_star]
        
        # Direction: -subgradient projected onto sum=0 hyperplane
        d = np.zeros(n)
        for j in range(j_start, j_end + 1):
            d[j] = -(2.0 * a[k_star - j] / S2 - 2.0 * Mak / S3)
        for j in range(n):
            if j < j_start or j > j_end:
                d[j] = 2.0 * Mak / S3
        
        # Project onto sum(d)=0
        d -= d.mean()
        
        # Also ensure non-negativity: limit the step
        # Find the maximum step that keeps a + t*d >= 0
        neg_mask = d < 0
        if neg_mask.any():
            max_t = (a[neg_mask] / (-d[neg_mask])).min()
        else:
            max_t = 1.0
        max_t = min(max_t * 0.5, 0.1)  # Conservative
        
        # Try a few step sizes
        for t in [max_t, max_t*0.5, max_t*0.2]:
            a_try = a + t * d
            if a_try.min() < -1e-15:
                continue
            a_try = np.maximum(0, a_try)
            S_try = a_try.sum()
            if S_try < 1e-12:
                continue
            a_try = a_try / S_try
            sc_try = compute_score(a_try)
            if sc_try < best_sc:
                best_sc = sc_try
                best_a = a_try.copy()
                a = a_try.copy()
                break
    
    return best_a

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()

    return np.fft.irfft(A * A, M_fft)[:M]

    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())

    return 2.0 * n * float(np.max(c)) / (S * S)

        step_est = min(step_est, 0.5)

        candidates_t = [step_est, step_est * 0.5, step_est * 0.2,
                        step_est * 0.1, max(0.001, step_est * 0.05), 0.0005, 0.0001]

        for t in candidates_t:
            a_try = a + t * direction
            if a_try.min() < -1e-12:
                neg_mask = a_try < 0
                if neg_mask.any():
                    t_max = (a[neg_mask] / (-direction[neg_mask])).min()
                    if t_max <= 1e-18:
                        continue
                    t = t_max * 0.7
                    a_try = a + t * direction
            a_try = np.maximum(0, a_try)
            S_try = float(a_try.sum())
            if S_try < 1e-12:
                continue
            a_try = a_try / S_try
            max_r_try = float(fast_conv(a_try).max())
            if max_r_try < max_r * (1.0 - 1e-12):
                a = a_try
                improved = True
                break

        if not improved:
            # Very small step to escape potential stall
            a_try = a + 0.0001 * direction
            a_try = np.maximum(0, a_try)
            S_try = float(a_try.sum())
            if S_try > 1e-12:
                a = a_try / S_try

    return best_a


def peak_shave(a, n_steps=100):
    """
    Aggressive peak shaving: repeatedly identify the peak,
    reduce its top contributors, redistribute to non-contributors.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S

    best_a = a.copy()
    best_max_r = float(fast_conv(a).max())

    for step in range(n_steps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        max_r = float(c[k_star])

        if max_r < best_max_r:
            best_max_r = max_r
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        active = np.arange(j_start, j_end + 1)
        n_active = len(active)
        inactive = np.setdiff1d(np.arange(n), active)
        n_inactive = len(inactive)

        if n_inactive == 0:
            break

        # Contributions to the peak from each active index
        contribs = a[active] * a[k_star - active]
        total_contrib = contribs.sum()
        if total_contrib < 1e-15:
            break

        # Fraction of mass to move (fraction of peak value)
        move_frac = 0.05

        # Reduce top contributors proportionally
        n_top = max(1, n_active // 4)
        top_indices_in_active = np.argsort(contribs)[::-1][:n_top]
        top_active = active[top_indices_in_active]
        top_contrib_sum = contribs[top_indices_in_active].sum()
        if top_contrib_sum < 1e-15:
            break

        reduction = move_frac * max_r
        # Scale reduction: reduce each top contributor by a[j] * reduction / (a[j]*a[k*-j])
        # Actually, reduce a[j] by reduction * (a[j]*a[k*-j]) / top_contrib_sum
        for idx in top_active:
            frac = contribs[np.where(active == idx)[0][0]] / top_contrib_sum
            a[idx] -= reduction * frac

        # Add to inactive (spread evenly to those with smallest current value)
        a[inactive] += reduction / n_inactive

        # Project non-negative and renormalize
        a = np.maximum(0, a)
        S_new = float(a.sum())
        if S_new < 1e-12:
            break
        a = a / S_new

    return best_a


def perturbed_version(a, rng, scale):
    """Generate a perturbed copy of a."""
    n = len(a)
    x = np.arange(n, dtype=np.float64) / n
    strategy = rng.integers(0, 6)

    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 0.5 + 15.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (0.5 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 10))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + 2.0 * scale * (rng.random() - 0.5))
    elif strategy == 3:
        # Targeted: reduce peak contributors
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(5, n // 40)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.05 * rng.random())
    elif strategy == 4:
        # Random subset rescale
        n_sel = max(1, n // 5)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + 2.0 * scale * (rng.random(n_sel) - 0.5))
    else:
        # Smooth bump
        center = rng.random()
        width = 0.05 + 0.3 * rng.random()
        a_p = a * (1.0 + scale * 2.0 * np.exp(-((x - center) / width) ** 2))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = float(a_p.sum())
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def optimize_sequence(a_init, t_budget, seed=42):
    """
    Full multi-phase optimization starting from a_init.
    Returns the best sequence found within t_budget seconds.
    """
    t_start = time.monotonic()
    deadline = t_start + t_budget
    n = len(a_init)
    rng = np.random.default_rng(seed)

    a_init = a_init.astype(np.float64).copy()
    S = float(a_init.sum())
    if S < 1e-12:
        a_init = np.ones(n) * 0.5
        S = n * 0.5
    a_init = a_init / S

    best_a = a_init.copy()
    best_sc = compute_score(a_init)
    print(f"  Start: score={best_sc:.10f} (n={n})")

    def try_update(a_new, label):
        nonlocal best_a, best_sc
        if a_new is not None:
            sc_new = compute_score(a_new)
            if sc_new < best_sc:
                best_sc = sc_new
                best_a = a_new.copy()
                print(f"    [{label}] score={sc_new:.10f}")
            return sc_new
        return np.inf

    # Phase 1: Subgradient (global)
    rem = deadline - time.monotonic()
    if rem > 30:
        n_iter = min(3000, int(rem * 20))
        a_sg = subgrad_method(a_init, n_iters=n_iter, use_momentum=True)
        try_update(a_sg, "subgrad_global")

    # Phase 2: L-BFGS continuation
    beta_schedule = [50, 200, 1000, 5000, 20000, 100000, 500000]
    for beta in beta_schedule:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        maxiter = int(min(30000, max(3000, rem * 40)))
        maxfun = int(min(100000, max(10000, rem * 100)))
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        try_update(a_opt, f"LBFGS_b{beta}")

        # Peak shave after L-BFGS
        rem = deadline - time.monotonic()
        if rem > 20:
            a_ps = peak_shave(a_opt, n_steps=min(100, int(rem * 30)))
            try_update(a_ps, f"peakshave_b{beta}")

    # Phase 3: Interleaved subgradient + L-BFGS (fine tuning)
    for round_idx in range(3):
        rem = deadline - time.monotonic()
        if rem < 60:
            break

        # Subgradient fine-tune
        n_iter = min(2000, int(rem * 10))
        a_sg = subgrad_method(best_a, n_iters=n_iter, use_momentum=True, momentum=0.95)
        try_update(a_sg, f"finetune_sg_{round_idx}")

        # L-BFGS with high beta
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        beta = 50000 * (round_idx + 1)
        maxiter = int(min(20000, max(2000, rem * 30)))
        maxfun = int(min(60000, max(5000, rem * 60)))
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        try_update(a_opt, f"finetune_lbfgs_{round_idx}")

    # Phase 4: Iterated local search
    rem = deadline - time.monotonic()
    if rem > 60:
        n_trials = min(25, int(rem / 15))
        print(f"  ILS: {n_trials} trials, {rem:.0f}s remaining")
        for trial in range(n_trials):
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            scale = 0.005 + 0.02 * rng.random()
            a_pert = perturbed_version(best_a, rng, scale)
            beta = 500 + int(5000 * rng.random())
            maxiter = int(min(15000, max(2000, rem * 25)))
            maxfun = int(min(50000, max(5000, rem * 50)))
            sc, a_opt, _ = run_lbfgs(a_pert, beta, maxiter=maxiter, maxfun=maxfun)
            try_update(a_opt, f"ILS_{trial}")

            if time.monotonic() < deadline - 15:
                a_ps = peak_shave(a_opt, n_steps=30)
                try_update(a_ps, f"ILS_{trial}_ps")

    # Phase 5: Final peak shave
    rem = deadline - time.monotonic()
    if rem > 10:
        a_final = peak_shave(best_a, n_steps=min(200, int(rem * 20)))
        try_update(a_final, "final_shave")

    # Phase 6: Final subgradient
    rem = deadline - time.monotonic()
    if rem > 15:
        a_final2 = subgrad_method(best_a, n_iters=min(2000, int(rem * 15)),
                                   use_momentum=True, momentum=0.95)
        try_update(a_final2, "final_sg")

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    Uses multi-phase optimization: subgradient, L-BFGS continuation,
    peak shaving, and iterated local search.
    """
    t_start = time.monotonic()
    total_deadline = t_start + max(30.0, float(budget_s)) - 25.0

    # Get parent sequence
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64 and np.all(np.isfinite(p)):
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None and parent.sum() > 1e-6:
        n_work = len(parent)
        a_start = parent / parent.sum()
        print(f"Using parent length {n_work}")
    else:
        n_work = 8192
        a_start = np.ones(n_work) * 0.5
        print("No parent, starting fresh with n=", n_work)

    best_score = np.inf
    best_seq = a_start

    # Main optimization at parent length
    print(f"\n=== Main optimization (n={n_work}) ===")
    t_phase_start = time.monotonic()
    t_budget_main = min(total_deadline - t_phase_start, float(budget_s) - 120)
    a_opt, sc_opt = optimize_sequence(a_start, t_budget_main, seed=seed)
    if sc_opt < best_score:
        best_score = sc_opt
        best_seq = a_opt.copy()
        print(f"After main: {best_score:.10f}")

    # Try a different length if time permits
    rem = total_deadline - time.monotonic()
    if rem > 120:
        print(f"\n=== Length exploration ({rem:.0f}s remaining) ===")
        n_alt_options = []
        if parent is not None:
            n_alt_options = [int(n_work * 0.7), int(n_work * 1.3)]
        else:
            n_alt_options = [4096, 8192, 12288]

        for n_alt in n_alt_options:
            if n_alt < 256 or n_alt > 30000:
                continue
            rem = total_deadline - time.monotonic()
            if rem < 60:
                break
            # Interpolate from best
            a_alt = np.interp(np.linspace(0, 1, n_alt),
                              np.linspace(0, 1, len(best_seq)), best_seq)
            a_alt = np.clip(a_alt, 0.0, 1000.0)
            S = float(a_alt.sum())
            if S > 1e-6:
                a_alt = a_alt / S
            t_budget_alt = min(rem - 30, 180.0)
            a_alt_opt, sc_alt = optimize_sequence(a_alt, t_budget_alt,
                                                  seed=seed + n_alt)
            if sc_alt < best_score:
                best_score = sc_alt
                best_seq = a_alt_opt.copy()
                print(f"Length {n_alt} improved to: {best_score:.10f}")

    # Final subgradient polish on the best
    rem = total_deadline - time.monotonic()
    if rem > 20:
        print(f"\n=== Final polish ({rem:.0f}s) ===")
        a_final = subgrad_method(best_seq, n_iters=min(3000, int(rem * 20)),
                                  use_momentum=True, momentum=0.95)
        sc_final = compute_score(a_final)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = a_final.copy()
            print(f"Final subgrad: {best_score:.10f}")

        # One more L-BFGS
        rem = total_deadline - time.monotonic()
        if rem > 30:
            sc_lb, a_lb, _ = run_lbfgs(best_seq, 200000,
                                        maxiter=int(min(10000, rem * 50)),
                                        maxfun=int(min(30000, rem * 100)))
            if sc_lb < best_score:
                best_score = sc_lb
                best_seq = a_lb.copy()
                print(f"Final L-BFGS: {best_score:.10f}")

    # Construct valid output
    out = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    if verified < np.inf:
        return out
    return best_seq
