"""

multi-peak equalization, and iterated local search with diverse perturbations.

import numpy as np
from scipy.optimize import minimize, linprog
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_objective_and_grad(a, beta, top_k=None):
    """
    LSE-smoothed objective and gradient.
    Returns (value, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    if top_k is not None and top_k < len(c):
        # Get the top_k values
        top_vals = np.sort(c)[::-1][:top_k]
        threshold = top_vals[-1]
        # Soft mask: weights are 1 for top peaks, 0 for others
        # But use a smooth version
        mask = np.ones(len(c))
        mask[c < threshold * 0.99] = 0.0
        # Actually, let's just use the top_k restriction differently
        # We'll compute LSE only over the top_k peaks
        idx = np.argsort(c)[::-1][:top_k]
        c_top = c[idx]
        c_max = float(c_top.max())
        scaled = beta * (c_top - c_max)
        exp_scaled = np.exp(scaled - scaled.max())  # stability
        Z = float(exp_scaled.sum())
        p = exp_scaled / Z
        L = c_max + float(np.log(Z)) / beta
        
        # Gradient contribution from top peaks
        # dL/d(a_i) = sum_k p_k * d(c_k)/d(a_i)
        # d(c_k)/d(a_i) = 2 * a[k - i] (for valid i)
        # So grad = 2 * (p * a) correlated appropriately
        rev_a = a[::-1]
        M2 = 3 * n - 2
        M2_fft = 1 << (M2 - 1).bit_length()
        RA = np.fft.rfft(rev_a, M2_fft)
        P = np.fft.rfft(p, M2_fft)
        conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
        g = 2.0 * conv_res
        
        score = 2.0 * n * L / (S * S)
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
        return score, grad
    else:
        c_max = float(c.max())
        scaled = beta * (c - c_max)
        scaled = scaled - scaled.max()  # numerical stability
        exp_scaled = np.exp(scaled)
        Z = float(exp_scaled.sum())
        p = exp_scaled / Z
        L = c_max + float(np.log(Z)) / beta
        
        rev_a = a[::-1]
        M2 = 3 * n - 2
        M2_fft = 1 << (M2 - 1).bit_length()
        RA = np.fft.rfft(rev_a, M2_fft)
        P = np.fft.rfft(p, M2_fft)
        conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
        g = 2.0 * conv_res
        
        score = 2.0 * n * L / (S * S)
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
        return score, grad


def fista_lse(a0, beta, n_iters=5000, top_k=None, init_step=None):
    """
    FISTA (accelerated proximal gradient) on LSE-smoothed objective.
    Handles non-negativity via projection (clipping + renormalization).
    """
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 0.5
        S = n * 0.5
    a = a / S  # normalize to sum = 1
    
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Lipschitz constant estimate
    c = fast_conv(y)
    S_y = float(y.sum())
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled - scaled.max())
    p = exp_scaled / exp_scaled.sum()
    rev_y = y[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_y, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    L_est = 2.0 * n * (float(np.max(np.abs(g))) + 2.0 * c_max)
    
    if init_step is not None:
        step = init_step
    else:
        step = 1.0 / (L_est + 1e-10)
    
    for it in range(n_iters):
        # Gradient step
        val, grad = lse_objective_and_grad(y, beta, top_k)
        gnorm = float(np.linalg.norm(grad))
        
        y_new = y - step * grad
        
        # Proximal step: project onto {a >= 0, sum(a) = 1}
        # First clip to non-negative, then normalize
        a_new = np.clip(y_new, 0.0, 1000.0)
        S_new = a_new.sum()
        if S_new < 1e-12:
            a_new[:] = 1.0 / n
            S_new = 1.0
        a_new = a_new / S_new
        
        # Check improvement
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        
        # Backtracking if needed
        if it > 0 and gnorm > 1e-15:
            # Simple backtracking: if the function increased too much, reduce step
            # (FISTA with backtracking)
            pass  # For now, use fixed step
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new
        y = a_new + theta * (a_new - a)
        y = np.clip(y, 0.0, 1000.0)
        Sy = y.sum()
        if Sy > 1e-12:
            y = y / Sy
        
        a = a_new
    
    return best_a


def fista_adaptive(a0, beta_start=50, beta_end=100000, n_total=20000, top_k=None):
    """
    FISTA with adaptive beta that increases geometrically.
    """
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 0.5
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    n_phases = 8
    iters_per_phase = n_total // n_phases
    log_beta_start = np.log(beta_start)
    log_beta_end = np.log(beta_end)
    
    for phase in range(n_phases):
        beta = float(np.exp(log_beta_start + (log_beta_end - log_beta_start) * phase / max(1, n_phases - 1)))
        
        # Determine iterations for this phase
        n_iters = iters_per_phase
        if phase == 0:
            n_iters = iters_per_phase * 2  # more iterations at low beta
        
        a_ns = fista_lse(a, beta, n_iters=n_iters, top_k=top_k)
        sc = compute_score(a_ns)
        if sc < best_sc:
            best_sc = sc
            best_a = a_ns.copy()
            print(f"    FISTA phase {phase}: beta={beta:.0f}, score={sc:.10f}")
        else:
            print(f"    FISTA phase {phase}: beta={beta:.0f}, score={sc:.10f} (no improvement)")
        a = a_ns
    
    return best_a


def multi_peak_equalization(a, top_k=50, n_inner=1000):
    """
    Try to equalize the top peaks by doing a small optimization
    focused on reducing the spread of the top peaks.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    # Identify top peaks
    c = fast_conv(a)
    top_idx = np.argsort(c)[::-1][:top_k]
    c_top = c[top_idx]
    c_mean = float(c_top.mean())
    c_std = float(c_top.std())
    
    if c_std < 1e-10:
        return best_a
    
    # Use LSE with high beta focused on top peaks
    beta = 1000.0
    a_opt = fista_lse(a, beta, n_iters=n_inner, top_k=top_k)
    sc = compute_score(a_opt)
    if sc < best_sc:
        best_a = a_opt.copy()
        best_sc = sc
    
    return best_a


def run_lbfgs_refine(a0, beta, maxiter=30000, maxfun=100000, top_k=None):
    """Run L-BFGS-B refinement on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    # Reduce bounds for speed
    bounds = [(0.0, 2.0)] * n
    
    func = lambda x: lse_objective_and_grad(x, beta, top_k)
    
    res = minimize(
        func, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    Sc = compute_score(a_out)
    return Sc, a_out


def perturbed_restart(a, rng, scale=0.02, strategy=None):
    """Generate a perturbed version of a."""
    n = len(a)
    if strategy is None:
        strategy = rng.integers(0, 6)
    x = np.arange(n, dtype=np.float64) / n
    
    if strategy == 0:
        # Gaussian noise
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        # Sinusoidal
        freq = 0.5 + 5.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        # Block perturbation
        n_blocks = 3 + int(rng.integers(0, 10))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strategy == 3:
        # Targeted: reduce peak contributors
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(10, n // 30)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.05 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            deficit = a.sum() - S
            bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            if len(bottom_local) > 0:
                per = deficit / len(bottom_local)
                for li in bottom_local:
                    a_p[j_start + li] = max(0, a_p[j_start + li] + per)
    elif strategy == 4:
        # Multi-frequency sinusoidal
        n_freqs = 2 + int(rng.integers(0, 3))
        mod = np.ones(n)
        for _ in range(n_freqs):
            freq = 0.5 + 20.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = scale * 0.5
            mod += amp * np.sin(2.0 * np.pi * freq * x + phase)
        a_p = a * mod
    else:
        # Random subset rescale
        n_sel = max(1, n // 3)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 2.0 * (rng.random(n_sel) - 0.5))
    
    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def coarse_to_fine(a_coarse, n_fine, beta=500, n_iters=5000):
    """
    Interpolate from coarse to fine resolution and optimize.
    """
    n_c = len(a_coarse)
    x_c = np.linspace(0, 1, n_c)
    x_f = np.linspace(0, 1, n_fine)
    a_fine = np.interp(x_f, x_c, a_coarse)
    a_fine = np.clip(a_fine, 0.0, 1000.0)
    S = a_fine.sum()
    if S > 1e-12:
        a_fine = a_fine / S
    
    # Optimize at fine resolution
    a_opt = fista_lse(a_fine, beta, n_iters=n_iters)
    sc = compute_score(a_opt)
    return a_opt, sc

"""
Hybrid optimizer for step function peak minimization.
Combines adaptive FISTA (accelerated proximal gradient) with LSE smoothing,

Key innovations:
1. FISTA with backtracking line search for robust first-order optimization

4. Symmetric initialization to exploit the likely symmetry of the optimum
5. Diverse perturbation strategies for iterated local search

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta, top_k=None):
    """
    LSE-smoothed objective and gradient.
    Objective: 2n * L / S^2 where L = (1/beta)*log(sum exp(beta*(c_j - c_max)))
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)

    if top_k is not None and top_k < len(c):
        idx = np.argsort(c)[::-1][:top_k]
        c_top = c[idx]
        c_max = float(c_top.max())
        scaled = beta * (c_top - c_max)
        scaled = scaled - scaled.max()
        exp_scaled = np.exp(scaled)
        Z = float(exp_scaled.sum())
        p = exp_scaled / Z
        L = c_max + float(np.log(Z)) / beta

        # Gradient: 2n * (g/S^2 - 2L/S^3) where g_i = 2*sum_k p_k a_{k-i}
        rev_a = a[::-1]
        M2 = 3 * n - 2
        M2_fft = 1 << (M2 - 1).bit_length()
        RA = np.fft.rfft(rev_a, M2_fft)
        P = np.fft.rfft(p, M2_fft)
        conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
        g = 2.0 * conv_res

        score = 2.0 * n * L / (S * S)
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
        return score, grad
    else:
        c_max = float(c.max())
        scaled = beta * (c - c_max)
        scaled = scaled - scaled.max()
        exp_scaled = np.exp(scaled)
        Z = float(exp_scaled.sum())
        p = exp_scaled / Z
        L = c_max + float(np.log(Z)) / beta

    S = a.sum()

    t = 1.0

    best_sc = compute_score(a)

    gnorm0 = float(np.linalg.norm(grad0))
    if gnorm0 < 1e-15:
        step = 0.01
    else:

        step = min(step, 0.1)

        z = y - step * grad_y

        # Proximal step: project onto {a >= 0, sum(a) = 1}

        if Sy > 1e-12:
            y = y / Sy

        a = a_new

    return best_a, best_sc


def adaptive_fista(a0, beta_start=30, beta_end=200000, n_total=20000, top_k=None):
    """
    FISTA with geometrically increasing beta.
    Returns the best sequence found.
    """
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    n_phases = 10
    iters_per_phase = max(500, n_total // n_phases)
    log_betas = np.linspace(np.log(beta_start), np.log(beta_end), n_phases + 1)

    for phase in range(n_phases):
        beta = float(np.exp((log_betas[phase] + log_betas[phase + 1]) / 2.0))
        n_iters = iters_per_phase
        if phase == 0:
            n_iters = iters_per_phase * 2

        a_opt, sc = fista_optimize(a, beta, n_iters=n_iters, top_k=top_k)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
            print(f"    FISTA phase {phase}: beta={beta:.0f}, score={sc:.10f} *")
        else:
            print(f"    FISTA phase {phase}: beta={beta:.0f}, score={sc:.10f}")
        a = best_a

    return best_a, best_sc


def lbfgs_refine(a0, beta, maxiter=20000, maxfun=80000, top_k=None):
    """L-BFGS-B refinement on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 2.0)] * n

    res = minimize(
        lse_obj_grad, a0,
        args=(beta, top_k),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    Sc = compute_score(a_out)
    return Sc, a_out


def active_set_equalize(a, top_k=30, n_field_iters=500, beta=5000):
    """
    Active-set peak equalization:
    Identify the top K peaks, then do a focused optimization to flatten them.
    Uses a high-beta LSE restricted to those peaks.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a, compute_score(a)
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    # Multiple rounds of equalization with increasing beta
    for round_idx in range(n_field_iters // 200 + 1):
        current_beta = beta * (round_idx + 1)
        current_topk = max(5, top_k - round_idx * 5)
        a_opt, sc = fista_optimize(a, current_beta, n_iters=200, top_k=current_topk)
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
        a = best_a

    return best_a, best_sc


def perturb_sequence(a, rng, scale=0.02):
    """Generate a diverse perturbation of a."""
    n = len(a)
    strategy = rng.integers(0, 7)
    x = np.arange(n, dtype=np.float64) / n

    if strategy == 0:
        # Gaussian noise
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        # Single sinusoid
        freq = 0.5 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        # Multi-sinusoid
        n_freqs = 2 + int(rng.integers(0, 3))
        mod = np.ones(n)
        for _ in range(n_freqs):
            freq = 0.5 + 15.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = scale * 0.4
            mod += amp * np.sin(2.0 * np.pi * freq * x + phase)
        a_p = a * mod
    elif strategy == 3:
        # Block perturbation
        n_blocks = 4 + int(rng.integers(0, 12))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 1.5 * (rng.random() - 0.5))
    elif strategy == 4:
        # Peak-targeted reduction
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(10, n // 40)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.05 * rng.random())
        S_p = a_p.sum()
        if S_p > 1e-12:
            deficit = a.sum() - S_p
            bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            if len(bottom_local) > 0 and deficit > 0:
                per = deficit / len(bottom_local)
                for li in bottom_local:
                    a_p[j_start + li] = max(0.0, a_p[j_start + li] + per)
    elif strategy == 5:
        # Envelope modulation
        envelope = 1.0 + scale * 2.0 * rng.standard_normal(n)
        # Smooth the envelope
        k = max(3, n // 500)
        kernel = np.ones(k) / k
        envelope = np.convolve(envelope, kernel, mode='same')
        a_p = a * np.clip(envelope, 0.1, 2.0)
    else:
        # Sparse random rescaling
        n_sel = max(1, n // 4)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 2.0 * (rng.random(n_sel) - 0.5))

    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p


def symmetric_enhance(a):
    """
    Try to make the sequence more symmetric, which should reduce
    non-center peaks. Returns the symmetric version if it improves score.
    """
    n = len(a)
    a_sym = (a + a[::-1]) / 2.0
    a_sym = np.clip(a_sym, 0.0, 1000.0)
    S = a_sym.sum()
    if S > 1e-12:
        a_sym = a_sym / S
    sc_orig = compute_score(a)
    sc_sym = compute_score(a_sym)
    if sc_sym < sc_orig:
        return a_sym
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Hybrid optimizer: Adaptive FISTA + Active-set equalization + L-BFGS + ILS.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 12.0

    # Get parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(8192) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    # Initialize from parent
    if parent is not None:
        parent_norm = parent / parent.sum()
        sc_parent = compute_score(parent_norm)
        print(f"Parent: n={len(parent)}, score={sc_parent:.10f}")
        report(sc_parent, parent_norm, "parent")
        n_work = len(parent)
        a_cur = parent_norm.copy()
    else:
        n_work = 8192
        a_cur = np.ones(n_work) * (1.0 / n_work)
        sc_init = compute_score(a_cur)
        report(sc_init, a_cur, "uniform_init")

    print(f"Working length: {n_work}, target: {1.5027:.6f}")

    # ==================================================================
    # Phase 1: Symmetric enhancement + Adaptive FISTA (25% time)
    # ==================================================================
    phase1_deadline = t_start + 0.25 * (float(budget_s) - 12.0)
    print(f"\n--- Phase 1: Symmetric + FISTA (n={n_work}) ---")

    # Try symmetric version
    a_sym = symmetric_enhance(a_cur)
    sc_sym = compute_score(a_sym)
    report(sc_sym, a_sym, "symmetric")
    a_cur = best_a.copy()

    # Adaptive FISTA
    a_opt, sc = adaptive_fista(a_cur, beta_start=30, beta_end=100000,
                                n_total=12000, top_k=None)
    report(sc, a_opt, "fista_adaptive")

    # FISTA with top_k restriction
    if time.monotonic() < phase1_deadline:
        for top_k in [10, 30, 60]:
            if time.monotonic() > phase1_deadline:
                break
            a_tk, sc_tk = fista_optimize(best_a, 1000, n_iters=5000, top_k=top_k)
            report(sc_tk, a_tk, f"fista_tk={top_k}")

    print(f"After Phase 1: best = {best_score:.10f}")

    # ==================================================================
    # Phase 2: L-BFGS refinement with increasing beta (25% time)
    # ==================================================================
    phase2_deadline = t_start + 0.50 * (float(budget_s) - 12.0)
    print(f"\n--- Phase 2: L-BFGS Refinement ---")

    beta_schedule = [200, 1000, 5000, 20000, 100000]
    for beta in beta_schedule:
        if time.monotonic() > phase2_deadline:
            break
        remaining_s = phase2_deadline - time.monotonic()
        if remaining_s < 20:
            break
        maxiter = int(min(30000, max(3000, remaining_s * 25)))
        maxfun = int(min(100000, max(10000, remaining_s * 80)))

        sc, a_opt = lbfgs_refine(best_a, beta, maxiter=maxiter, maxfun=maxfun, top_k=None)
        report(sc, a_opt, f"LBFGS b={beta}")

        # Top-k variant for additional improvement
        if time.monotonic() < phase2_deadline - 30:
            for tk in [5, 15]:
                if time.monotonic() > phase2_deadline - 15:
                    break
                sc_k, a_k = lbfgs_refine(best_a, beta, maxiter=maxiter // 2,
                                         maxfun=maxfun // 2, top_k=tk)
                report(sc_k, a_k, f"LBFGS b={beta} tk={tk}")

    print(f"After Phase 2: best = {best_score:.10f}")

    # ==================================================================
    # Phase 3: Active-set equalization + FISTA (15% time)
    # ==================================================================
    phase3_deadline = t_start + 0.65 * (float(budget_s) - 12.0)
    print(f"\n--- Phase 3: Peak Equalization ---")

    a_eq, sc_eq = active_set_equalize(best_a, top_k=30, n_field_iters=500, beta=3000)
    report(sc_eq, a_eq, "peak_eq_30")

    a_eq2, sc_eq2 = active_set_equalize(best_a, top_k=10, n_field_iters=500, beta=10000)
    report(sc_eq2, a_eq2, "peak_eq_10")

    # Symmetric enhance after equalization
    a_sym2 = symmetric_enhance(best_a)
    sc_sym2 = compute_score(a_sym2)
    report(sc_sym2, a_sym2, "sym_eq")

    # More FISTA with very high beta
    if time.monotonic() < phase3_deadline:
        a_hi, sc_hi = fista_optimize(best_a, 200000, n_iters=8000, top_k=None)
        report(sc_hi, a_hi, "fista_hi_beta")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ==================================================================
    # Phase 4: Iterated Local Search with diverse perturbations (25% time)
    # ==================================================================
    phase4_deadline = t_start + 0.90 * (float(budget_s) - 12.0)
    print(f"\n--- Phase 4: Iterated Local Search ---")

    n_trials = 20
    for trial in range(n_trials):
        if time.monotonic() > phase4_deadline:
            break
        remaining_s = phase4_deadline - time.monotonic()
        if remaining_s < 40:
            break

        # Diverse scales
        scale = 0.008 + 0.025 * rng.random()
        a_pert = perturb_sequence(best_a, rng, scale=scale)

        # Optimize with FISTA
        beta = 300 + int(3000 * rng.random())
        a_opt, sc = fista_optimize(a_pert, beta, n_iters=4000, top_k=None)
        report(sc, a_opt, f"ILS#{trial} fista")

        # Quick LBFGS polish if promising
        if sc < best_score * 1.0005 and time.monotonic() < phase4_deadline - 30:
            sc2, a_opt2 = lbfgs_refine(a_opt, 5000, maxiter=15000, maxfun=60000)
            report(sc2, a_opt2, f"ILS#{trial} polish")

        # If still time, try a smaller perturbation from current best
        if time.monotonic() < phase4_deadline - 20:
            a_pert2 = perturb_sequence(best_a, rng, scale=scale * 0.3)
            a_opt2, sc2 = fista_optimize(a_pert2, 2000, n_iters=2000, top_k=10)
            report(sc2, a_opt2, f"ILS#{trial} small")

    print(f"After Phase 4: best = {best_score:.10f}")

    # ==================================================================
    # Phase 5: Final high-precision polish (10% time)
    # ==================================================================
    print(f"\n--- Phase 5: Final Polish ---")

    # Very high beta L-BFGS
    for beta in [200000, 500000]:
        if time.monotonic() > deadline - 15:
            break
        if time.monotonic() < deadline - 40:
            sc, a_opt = lbfgs_refine(best_a, beta, maxiter=40000, maxfun=120000)
            report(sc, a_opt, f"final b={beta}")

    # Final FISTA
    if time.monotonic() < deadline - 20:
        a_fin, sc_fin = fista_optimize(best_a, 500000, n_iters=10000, top_k=None)
        report(sc_fin, a_fin, "final_fista")

    # One more symmetric check
    a_sym_fin = symmetric_enhance(best_a)
    sc_sym_fin = compute_score(a_sym_fin)
    report(sc_sym_fin, a_sym_fin, "final_sym")

    # ==================================================================
    # Final output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"Target: 1.5027, Improvement: {best_score - 1.5027:+.6f}")
    print(f"{'=' * 60}")
    return out
