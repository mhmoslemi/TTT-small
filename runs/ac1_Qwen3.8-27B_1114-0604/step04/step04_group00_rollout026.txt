```python
"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Core innovation: Iterative Reweighted Peak Reduction (WPR) using FFT-based cross-correlation
for gradient computation, combined with LSE-smoothed L-BFGS and spectral Fienup iterations.
Uses adaptive weighting exponent, Polyak step sizes, and expert iteration for diversity.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def cross_correlation_wpr(w, a):
    """
    Compute r[j] = sum_{i=0}^{n-1} w[j+i] * a[i] for j=0..n-1
    where w has length 2n-1 and a has length n.
    Uses FFT for O(n log n) computation.
    """
    n = len(a)
    N_fft = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, N_fft)
    A = np.fft.rfft(a, N_fft)
    cross = np.fft.irfft(W * np.conj(A), N_fft)[:n]
    return cross

    to minimize sum_k w[k]*c[k], which pushes peaks down.
    Uses momentum and adaptive step size.

    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    vel = np.zeros(n)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)
    patience = 0
    best_patience_sc = best_sc

    for it in range(n_iters):
        progress = it / n_iters
        p = p_init + (p_final - p_init) * progress

        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > 500 and sc > best_patience_sc * 1.001:
                patience = 0
                best_patience_sc = sc

        # Compute weights: emphasize peaks
        c_safe = c + smooth_eps
        w = 1.0 / np.power(c_safe, p)
        # Cap weights to avoid numerical issues
        w_max = w.max()
        if w_max > 1e10:
            w = (w / w_max)

        # Gradient: g[j] = 2 * sum_k w[k] * a[k-j] = 2 * cross_correlation(w, a)[j]
        cross = cross_correlation_wpr(w, a)
        g = 2.0 * cross

        # Project onto tangent space (sum = 0)
        g = g - g.mean()

        # Gradient norm for step size
        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            break

        # Adaptive step: Polyak-like
        step_polyak = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, min(step_polyak, 0.08))
        step = max(step, 1e-12)

        # Momentum update
        vel = momentum * vel + step * g
        a = a - vel
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

        momentum *= momentum_decay

    return best_a


def weighted_peak_reduction_topm(a, n_iters=10000, top_m=10, p=2.0,
                                 step_init=0.05, momentum=0.9,
                                 smooth_eps=1e-10):
    """
    WPR variant: uses only top-M peak locations for weights.
    More focused on the actual peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    vel = np.zeros(n)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Top-M indices
        m_eff = min(top_m, len(c))
        top_idx = np.argsort(c)[::-1][:m_eff]

        # Build weight vector with only top-M peaks weighted
        w = np.zeros_like(c)
        for k in top_idx:
            w[k] = 1.0 / (c[k] + smooth_eps) ** p

        # Normalize weights
        w_sum = w.sum()
        if w_sum < 1e-15:
            break
        w = w / w_sum

        # Gradient via cross-correlation
        cross = cross_correlation_wpr(w, a)
        g = 2.0 * cross
        g = g - g.mean()

        gn = float(np.linalg.norm(g))
        if gn < 1e-15:
            break

        step_polyak = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, min(step_polyak, 0.08))
        step = max(step, 1e-12)

        vel = momentum * vel + step * g
        a = a - vel
        a = np.maximum(a, 0.0)
        S = float(a.sum())
        if S < 1e-12:
            break
        a = a / S

    return best_a


def fienup_spectral(a, n_iters=200, top_frac=0.3, strength=0.7):
    """
    Spectral Fienup: flatten the power spectrum by reducing high-power bins.
    Helps create more uniform autocorrelation.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())

        # Threshold: top fraction of bins
        thresh = np.quantile(P, 1.0 - top_frac)

        scale = np.ones_like(P)
        above = P > thresh
        if above.any():
            # Reduce power in high bins
            scale[above] = np.sqrt(np.maximum(thresh / P[above], 0.0)) * (1.0 - strength * 0.3)

        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def lse_score_grad(x, beta):
    """LSE-smoothed score and gradient."""
    n = len(x)
    x = np.maximum(x, 0.0)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability
    exp_scaled = np.exp(np.maximum(scaled, -50.0))
    log_sum = float(np.log(exp_scaled.sum() + 1e-30))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    p = exp_scaled / (exp_scaled.sum() + 1e-30)
    # Gradient of L w.r.t. x[j]: sum_k p[k] * 2*x[k-j] (valid k-j)
    rev_x = x[::-1]
    # conv(p, rev_x) gives us sum_k p[k] * rev_x[k-j] = sum_k p[k] * x[n-1-(k-j)]
    # We need sum_k p[k] * x[k-j]
    # Let's use: (p * x)[j] = sum_k p[k] x[k-j] = cross_correlation(p, x)
    # But p has length 2n-1 and x has length n
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    P = np.fft.rfft(p, M2_fft)
    X = np.fft.rfft(x, M2_fft)
    cross = np.fft.irfft(P * np.conj(X), M2_fft)[:n]
    g_L = 2.0 * cross

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1e6).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1e6)] * n

    def obj_and_grad(x):
        return lse_score_grad(x, beta)

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1e6)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def expert_iteration(a, n_inner_wpr=2000, n_inner_lbfgs=2000,
                     noise_scale=0.05, seed=0):
    """
    Expert iteration: perturb the current solution, then re-optimize.
    Helps escape local minima.
    """
    rng = np.random.default_rng(seed)
    n = len(a)

    # Perturb
    a_pert = a * (1.0 + noise_scale * rng.standard_normal(n))
    # Also try random multiplicative perturbation
    a_pert2 = a * rng.uniform(0.5, 1.5, n)
    a_pert3 = a + np.maximum(0.0, noise_scale * rng.standard_normal(n))

    candidates = [a_pert, a_pert2, a_pert3]
    best_a = a.copy()
    best_sc = compute_score(a)

    for i, ap in enumerate(candidates):
        ap = np.clip(ap, 0.0, None)
        S = ap.sum()
        if S < 1e-12:
            continue
        ap = ap / S

        # Short WPR
        a_wpr = weighted_peak_reduction(ap, n_iters=n_inner_wpr, p_init=1.5, p_final=4.0,
                                        step_init=0.04, momentum=0.9)
        sc = compute_score(a_wpr)
        if sc < best_sc:
            best_sc = sc
            best_a = a_wpr.copy()

        # Short L-BFGS
        if time.monotonic() < EXPERT_DEADLINE:
            sc2, a_lb = run_lbfgs(a_wpr, 500, maxiter=n_inner_lbfgs, maxfun=30000)
            if sc2 < best_sc:
                best_sc = sc2
                best_a = a_lb.copy()

    return best_a


# Global deadline for expert iteration (set in propose_candidate)
EXPERT_DEADLINE = 0.0


def build_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(np.clip(parent.astype(np.float64), 0.0, 1e6).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(np.maximum(0.5 + 0.4 * np.cos(2.0 * np.pi * x), 0.01))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(np.maximum(0.3 + 0.5 * np.cos(4.0 * np.pi * x), 0.01))
    # Sinusoidal raised window
    inits.append(np.sin(np.pi * x) ** 2)
    # Gaussian window
    inits.append(np.exp(-4.0 * (x - 0.5) ** 2))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses Hybrid WPR + LSE L-BFGS + Spectral Fienup + Expert Iteration.
    """
    global EXPERT_DEADLINE
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1e6)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(4096) * 0.5
    n_target = len(parent) if parent is not None else 6962

    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1e6)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    print(f"Target n={n_target}, parent={'yes' if parent is not None else 'no'}")
    print(f"Budget: {total_budget}s, deadline at t={deadline - t_start:.0f}s")

    # ==================================================================
    # Phase 1: Quick Fienup + WPR from diverse inits (20% time)
    # ==================================================================
    phase1_end = t_start + 0.20 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Quick exploration (deadline t={phase1_end - t_start:.0f}s) ===")

    inits = build_diverse_inits(n_target, rng, parent)

    for i, a0 in enumerate(inits):
        if time.monotonic() > phase1_end:
            break
        a0 = np.maximum(a0, 0.0)
        S = a0.sum()
        if S < 1e-12:
            continue
        a0 = a0 / S

        # Short Fienup
        a_f = fienup_spectral(a0, n_iters=100, top_frac=0.3, strength=0.6)
        sc = compute_score(a_f)
        update_best(sc, a_f, f"Fienup init={i}")

        # Short WPR from Fienup result
        if time.monotonic() < phase1_end:
            a_w = weighted_peak_reduction(a_f, n_iters=2000, p_init=1.0, p_final=3.0,
                                          step_init=0.04, momentum=0.85)
            sc2 = compute_score(a_w)
            update_best(sc2, a_w, f"WPR init={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Extended WPR with various parameters (30% time)
    # ==================================================================
    phase2_end = t_start + 0.50 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Extended WPR (deadline t={phase2_end - t_start:.0f}s) ===")

    # WPR with different p schedules
    wpr_configs = [
        {'p_init': 0.5, 'p_final': 3.0, 'step_init': 0.06, 'momentum': 0.90, 'n_iters': 15000},
        {'p_init': 1.0, 'p_final': 5.0, 'step_init': 0.05, 'momentum': 0.92, 'n_iters': 15000},
        {'p_init': 2.0, 'p_final': 8.0, 'step_init': 0.04, 'momentum': 0.90, 'n_iters': 12000},
        {'p_init': 0.5, 'p_final': 2.0, 'step_init': 0.08, 'momentum': 0.88, 'n_iters': 15000},
    ]

    for ci, cfg in enumerate(wpr_configs):
        if time.monotonic() > phase2_end:
            break
        remaining_phase = phase2_end - time.monotonic()
        iters = min(cfg['n_iters'], max(3000, int(remaining_phase * 100)))

        a_w = weighted_peak_reduction(best_a, n_iters=iters,
                                      p_init=cfg['p_init'], p_final=cfg['p_final'],
                                      step_init=cfg['step_init'], momentum=cfg['momentum'])
        sc = compute_score(a_w)
        update_best(sc, a_w, f"WPR cfg={ci}")

        # Top-M variant
        if time.monotonic() < phase2_end:
            a_tm = weighted_peak_reduction_topm(a_w, n_iters=min(8000, iters // 2),
                                                top_m=10, p=2.0,
                                                step_init=0.04, momentum=0.9)
            sc2 = compute_score(a_tm)
            update_best(sc2, a_tm, f"WPR-topM cfg={ci}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: LSE L-BFGS with beta annealing (25% time)
    # ==================================================================
    phase3_end = t_start + 0.75 * (total_budget - 30.0)
    print(f"\n=== Phase 3: LSE L-BFGS (deadline t={phase3_end - t_start:.0f}s) ===")

    betas = [100, 300, 1000, 3000, 10000]
    for beta in betas:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        maxiter = min(15000, int(remaining * 50))
        maxfun = min(200000, maxiter * 20)

        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxiter, maxfun=maxfun)
        update_best(sc, a_opt, f"LBFGS beta={beta}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Cycling WPR -> L-BFGS -> WPR (15% time)
    # ==================================================================
    phase4_end = t_start + 0.90 * (total_budget - 30.0)
    print(f"\n=== Phase 4: Cycling (deadline t={phase4_end - t_start:.0f}s) ===")

    for cycle in range(3):
        if time.monotonic() > phase4_end:
            break

        # WPR step
        a_c = weighted_peak_reduction(best_a, n_iters=5000, p_init=1.0, p_final=4.0,
                                      step_init=0.05, momentum=0.9)
        sc = compute_score(a_c)
        update_best(sc, a_c, f"Cycle-{cycle} WPR")

        # L-BFGS step
        if time.monotonic() < phase4_end:
            sc2, a_l = run_lbfgs(a_c, 2000, maxiter=5000, maxfun=80000)
            update_best(sc2, a_l, f"Cycle-{cycle} LBFGS")

        # Spectral jump
        if time.monotonic() < phase4_end:
            a_s = fienup_spectral(a_l, n_iters=100, top_frac=0.2, strength=0.8)
            sc3 = compute_score(a_s)
            update_best(sc3, a_s, f"Cycle-{cycle} Fienup")
            # WPR after jump
            a_w2 = weighted_peak_reduction(a_s, n_iters=3000, p_init=1.0, p_final=3.0,
                                           step_init=0.04, momentum=0.85)
            sc4 = compute_score(a_w2)
            update_best(sc4, a_w2, f"Cycle-{cycle} WPR2")

    print(f"After Phase 4: best = {best_score:.8f}")

    # ==================================================================
    # Phase 5: Expert iteration (5% time)
    # ==================================================================
    phase5_end = deadline
    print(f"\n=== Phase 5: Expert iteration (deadline t={phase5_end - t_start:.0f}s) ===")

    EXPERT_DEADLINE = phase5_end
    for ei in range(3):
        if time.monotonic() > phase5_end:
            break
        a_exp = expert_iteration(best_a, n_inner_wpr=3000, n_inner_lbfgs=3000,
                                 noise_scale=0.03 + 0.02 * ei, seed=seed + ei)
        sc = compute_score(a_exp)
        update_best(sc, a_exp, f"Expert-{ei}")

    # ==================================================================
    # Phase 6: Final polish
    # ==================================================================
    print(f"\n=== Phase 6: Final polish ===")
    if time.monotonic() < deadline - 10:
        a_final = weighted_peak_reduction(best_a, n_iters=3000, p_init=2.0, p_final=6.0,
                                          step_init=0.03, momentum=0.9)
        sc = compute_score(a_final)
        update_best(sc, a_final, "Final WPR")

    if time.monotonic() < deadline - 5:
        sc, a_opt = run_lbfgs(best_a, 5000, maxiter=5000, maxfun=60000)
        update_best(sc, a_opt, "Final LBFGS")

    # ==================================================================
    # Final output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
```
