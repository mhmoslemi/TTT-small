    # Using FFT: u = ifft(FFT(p) * conj(FFT(a))) with appropriate padding
    # But since we need the full cross-correlation, let's use:
    # u[i] = sum_k p[k] * a[k-i] = (p ⋆ a)[i] (cross-correlation)
    #
    # In terms of convolution: (p ⋆ a)[i] = conv(p_rev, a)[i + len(p) - 1]
    # where p_rev = p[::-1]
    
    # Let me just use the approach from the previous code which worked:
    rev_a = a[::-1]
    M2 = 3 * n - 2  # length of conv(p, rev_a)
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)
    
    # The gradient of LSE wrt a:
    # g[i] = 2 * sum_k p[k] * a[k-i] (approximately, with the diagonal correction)
    # The conv_res gives us conv(p, rev_a) which relates to cross-correlation
    # conv(p, rev_a)[m] = sum_j p[j] * rev_a[m-j] = sum_j p[j] * a[n-1-(m-j)] = sum_j p[j] * a[n-1-m+j]
    # Let m = n-1-i: conv(p, rev_a)[n-1-i] = sum_j p[j] * a[i-j]... hmm this is getting confusing.
    
    # Let me use a different, cleaner approach:
    # The subgradient of max(c) at the point where c is "flat" (all equal) is uniform.
    # For LSE, the "soft" subgradient is p = softmax(beta*(c - max(c)))
    # The gradient of the LSE wrt a is: sum_k p[k] * (dc[k]/da)
    # 
    # Since dc[k]/da[i] ≈ 2*a[k-i] (with the diagonal correction being negligible when n is large),
    # g[i] ≈ 2 * sum_k p[k] * a[k-i]
    # 
    # This is 2 * (cross_correlation of p and a) at various lags.
    # 
    # Using the FFT approach: 
    # cross_corr[i] = sum_k p[k] * a[k-i] (with bounds)
    # = conv(p, a_rev)[(len(p)-1) + i] where a_rev = a[::-1]
    # Wait, let me verify:
    # conv(p, a_rev)[m] = sum_j p[j] * a_rev[m-j] = sum_j p[j] * a[n-1-(m-j)]
    # For m = (2n-2) + i: sum_j p[j] * a[n-1-2n+2-i+j] = sum_j p[j] * a[j-n+1-i]
    # That doesn't seem right either.
    
    # Let me just do it directly with a smaller FFT:
    # We want: g[i] = 2 * sum_k p[k] * a[k-i] for valid k, i
    # This is equivalent to: g = 2 * ifft(fft(p_pad) * conj(fft(a_pad))) but we need to be careful with the indexing.
    
    # Actually, the simplest correct approach for the gradient:
    # Define the "correlation kernel" Q[i,k] = ∂c[k]/∂a[i]
    # Then g = Q^T p
    # 
    # But Q is large (n x (2n-1)), so we need FFT.
    # 
    # The key identity: if c = conv(a,a), then for the gradient:
    # (Q^T p)[i] = sum_k p[k] * Q[i,k] = sum_k p[k] * (a[k-i] + a[k-i]) [with corrections]
    #            ≈ 2 * sum_k p[k] * a[k-i]
    # 
    # The sum sum_k p[k] * a[k-i] is the cross-correlation (p ⋆ a) evaluated at specific lags.
    # 
    # Let's define b_vec = a (length n) and use:
    # (p ⋆ a)[i] = sum_k p[k] * a[k-i] (cross-correlation, i from -(n-1) to 2n-2)
    # 
    # Using FFT: xcorr = ifft(fft(p, M) * conj(fft(a, M)))
    # where M >= 2n-1 for the full cross-correlation.
    # The result xcorr has the "positive lags" in the first part and "negative lags" in the wrapped part.
    
    M_xc = 1 << (2 * (2 * n - 1) - 1).bit_length()  # large enough
    # Actually, we need M_xc >= len(p) + len(a) - 1 = (2n-1) + n - 1 = 3n - 2
    M_xc = 1 << (3 * n - 2 - 1).bit_length()
    Pf = np.fft.rfft(p, M_xc)
    Af = np.fft.rfft(a, M_xc)
    # Cross-correlation: ifft(fft(p) * conj(fft(a)))
    xc = np.fft.irfft(Pf * np.conj(Af), M_xc)
    # xc[l] = sum_k p[k] * a[k+l] (for l from 0 to M_xc-1, wrapped)
    # Actually: xc[m] = sum_k p[k] * a[(m-k) mod M_xc]
    # For our purpose, we want: sum_k p[k] * a[k-i] for i = 0, ..., n-1
    # Let's set l = -i, so we need xc at negative lags: xc[M_xc - i] for i = 1, ..., n-1, and xc[0] for i=0.
    # Actually: xc[m] = sum_k p[k] * a[(m-k) mod M_xc]
    # We want sum_k p[k] * a[k-i] = sum_k p[k] * a[(-i+k) mod M_xc] ... hmm
    # Let me redefine: let s = k - i, so k = s + i.
    # sum_k p[k] * a[k-i] = sum_s p[s+i] * a[s] (for valid s and s+i)
    # This is the correlation of p (shifted) with a.
    
    # I think the cleanest way is:
    # g_approx[i] = 2 * (conv(p, a[::-1]))[n-1+i]  ... let me just verify numerically for small n.
    
    # For now, let me use the same approach as the previous code (which presumably worked):
    # The previous code computed:
    # conv_res = ifft(fft(rev_a) * fft(p))
    # g = 2 * conv_res[n-1 : 2n-1]
    
    # Let me verify: conv(p, rev_a)[m] = sum_j p[j] * rev_a[m-j] = sum_j p[j] * a[n-1-m+j]
    # For m = n-1+i (i from 0 to n-1):
    # = sum_j p[j] * a[n-1-(n-1+i)+j] = sum_j p[j] * a[j-i]
    # = sum_j p[j] * a[j-i] (for j >= i, so this is sum_{j=i}^{n-1} p[j] * a[j-i])
    # Hmm, this is sum_{j} p[j]*a[j-i] which is the cross-correlation at lag i.
    # But we want sum_k p[k]*a[k-i] which is the same thing! (just rename j to k)
    # So conv_res[n-1+i] = sum_j p[j]*a[j-i] = sum_k p[k]*a[k-i] ✓
    
    # So g[i] ≈ 2 * conv_res[n-1+i] (with the diagonal correction)
    
    # The diagonal correction: when k = 2i, the factor is 1 instead of 2.
    # This correction is: -p[2i]*a[i] for each i where 2i < 2n-1
    # For large n, this is a relatively small correction. Let's include it.
    
    g_approx = 2.0 * conv_res[n - 1:2 * n - 1]
    
    # Diagonal correction: for each i, if 2*i <= 2n-2, subtract p[2*i]*a[i]
    diag_idx = 2 * np.arange(n)
    valid_diag = diag_idx < (2 * n - 1)
    g_approx[valid_diag] -= p[diag_idx[valid_diag]] * a[valid_diag]
    
    # Full gradient of score = 2n * L / S^2:
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_approx / S2 - 2.0 * L / S3)
    
    return score, grad


def lse_objective_and_grad(x, beta):
    """Top-level objective+gradient for scipy.minimize."""
    x = np.maximum(x, 0.0)
    return lse_score_and_grad(x, beta)


def run_homotopy_lbfgs(a0, betas=None, maxiter_per_beta=5000, maxfun_per_beta=100000):
    """
    Run L-BFGS with homotopy: gradually increase beta.
    Returns (best_score, best_a).
    """
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = float(a.sum())
    if S < 1e-6:
        a[:] = 1.0 / n
    a = a / a.sum()
    
    if betas is None:
        betas = [5, 10, 20, 50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000]
    
    best_sc = compute_score(a)
    best_a = a.copy()
    bounds = [(0.0, 1000.0)] * n
    
    for beta in betas:
        res = minimize(
            lse_objective_and_grad, a.copy(), args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter_per_beta, 'maxfun': maxfun_per_beta, 'maxls': 50}
        )
        a_new = np.clip(res.x, 0.0, 1000.0)
        S_new = float(a_new.sum())
        if S_new < 1e-6:
            continue
        a_new = a_new / S_new
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            print(f"    beta={beta}: score={sc:.8f}")
        a = a_new
    
    return best_sc, best_a


def peak_pulling(a, n_iters=200, top_m=20, alpha=0.1):
    """
    Directly reduce the top convolution peaks by pulling down contributing a elements.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find top-m peaks
        top_idx = np.argsort(c)[::-1][:top_m]
        
        # For each peak, find the contributing a elements and reduce them
        grad = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                # c[k] includes terms a[j]*a[k-j] for j in [j_start, j_end]
                # Reduce elements that appear in multiple top peaks
                seg = a[j_start:j_end + 1]
                # Weight by contribution
                grad[j_start:j_end + 1] += c[k] * seg / (seg.sum() + 1e-30)
        
        # Normalize gradient
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Step: reduce elements proportional to their "peak contribution"
        step = alpha / (gn + 1e-10)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        
        # Decay alpha
        alpha *= 0.999
    
    return best_a


def fienup_step(a, n_iters=100, strength=0.5):
    """
    Fienup-style spectral flattening.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        P_mean = float(P.mean())
        
        # Flatten: reduce high-power bins
        thresh = P_mean * 1.2
        above = P > thresh
        if not above.any():
            break
        
        scale = np.ones_like(P)
        scale[above] = np.sqrt(thresh / P[above]) ** strength
        
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def perturbed_restart(a, rng, n_pert=10, max_perturb=0.05):
    """Generate perturbed versions of a for multi-restart."""
    n = len(a)
    results = []
    for i in range(n_pert):
        scale = max_perturb * (1.0 + i * 0.5)
        a_pert = a * (1.0 + scale * rng.standard_normal(n))
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        S = float(a_pert.sum())
        if S > 1e-12:
            a_pert = a_pert / S
        results.append(a_pert)
    return results


def top_m_subgradient(a, n_iters=5000, top_m=15, step_init=0.03):
    """
    Subgradient with top-M peaks and adaptive stepping.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    lb = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        if sc - lb < 1e-10:
            break
        
        # Top-M peaks
        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]
        
        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)
        
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Polyak step size
        step = min(step, (sc - lb) / (gn * gn + 1e-30))
        step = min(step, 0.05)
        
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= 0.9999
    
    return best_a


def coordinate_descent(a, n_sweeps=3, n_top=40):
    """
    Coordinate descent on peak-contributing elements.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local
        
        cur_sc = compute_score(a)
        
        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-12:
                continue
            best_val = a_old_j
            best_local_sc = cur_sc
            for factor in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5]:
                cv = a_old_j * factor
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old_j) < 1e-10:
                    continue
                a[j] = cv
                S_tmp = float(a.sum())
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                sc = compute_score(a / S_tmp)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = cv
                a[j] = a_old_j
            a[j] = best_val
        
        S = float(a.sum())
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv)/sum^2.
    Uses homotopy L-BFGS + peak-pulling + Fienup + subgradient.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 30.0
    
    # Load parent
    parent = None
    try:
        p = np.asarray(globals()['height_sequence_1'], dtype=np.float64).ravel()
        if p.size >= 64:
            parent = np.clip(p, 0.0, 1000.0)
    except (KeyError, NameError, TypeError):
        pass
    
    best_score = np.inf
    best_a = None
    
    def update_best(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.10f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False
    
    n = 6962
    if parent is not None:
        n = len(parent)
        print(f"Parent length: {n}")
        S = float(parent.sum())
        if S > 1e-6:
            parent_norm = parent / S
        else:
            parent_norm = np.ones(n) / n
        sc_parent = compute_score(parent_norm)
        print(f"Parent score: {sc_parent:.10f}")
        update_best(sc_parent, parent_norm, "PARENT")
    else:
        parent_norm = np.ones(n) / n
        print(f"No parent, using uniform n={n}")
    
    # ==========================================================
    # Phase 1: Homotopy L-BFGS from parent (45% of time)
    # ==========================================================
    phase1_end = t_start + 0.45 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Homotopy L-BFGS (deadline t={phase1_end - t_start:.0f}s) ===")
    
    a_work = parent_norm.copy()
    
    # Run homotopy with carefully chosen betas
    betas_list = [3, 5, 10, 20, 40, 80, 160, 320, 640, 1280, 2560]
    
    bounds = [(0.0, 1000.0)] * n
    
    for i, beta in enumerate(betas_list):
        if time.monotonic() > phase1_end:
            break
        print(f"  Homotopy step {i+1}/{len(betas_list)}: beta={beta}")
        
        maxiter = max(200, min(8000, 30000000 // (n * i + n)))
        maxfun = max(500, min(50000, 100000000 // (n * i + n)))
        
        res = minimize(
            lse_objective_and_grad, a_work.copy(), args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-18, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
        a_new = np.clip(res.x, 0.0, 1000.0)
        S = float(a_new.sum())
        if S < 1e-6:
            continue
        a_new = a_new / S
        sc = compute_score(a_new)
        update_best(sc, a_new, f"HOMO beta={beta}")
        a_work = a_new
    
    print(f"After Phase 1: best = {best_score:.10f}")
    
    # ==========================================================
    # Phase 2: Peak-pulling + refinement (20% of time)
    # ==========================================================
    phase2_end = t_start + 0.65 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Peak-pulling (deadline t={phase2_end - t_start:.0f}s) ===")
    
    a_pp = peak_pulling(best_a, n_iters=500, top_m=30, alpha=0.05)
    sc = compute_score(a_pp)
    update_best(sc, a_pp, "PEAK-PULL")
    
    # L-BFGS from peak-pulled result
    if time.monotonic() < phase2_end:
        for beta in [500, 2000, 5000]:
            if time.monotonic() > phase2_end:
                break
            res = minimize(
                lse_objective_and_grad, a_pp.copy(), args=(beta,),
                method='L-BFGS-B', jac=True, bounds=bounds,
                options={'ftol': 1e-18, 'gtol': 1e-14,
                         'maxiter': 5000, 'maxfun': 80000, 'maxls': 50}
            )
            a_new = np.clip(res.x, 0.0, 1000.0)
            S = float(a_new.sum())
            if S > 1e-6:
                a_new = a_new / S
                sc = compute_score(a_new)
                update_best(sc, a_new, f"PP+LBFGS beta={beta}")
    
    # Fienup from best
    if time.monotonic() < phase2_end:
        a_f = fienup_step(best_a, n_iters=200, strength=0.6)
        sc = compute_score(a_f)
        update_best(sc, a_f, "FIENUP")
        
        # L-BFGS from Fienup result
        if time.monotonic() < phase2_end:
            res = minimize(
                lse_objective_and_grad, a_f.copy(), args=(1000,),
                method='L-BFGS-B', jac=True, bounds=bounds,
                options={'ftol': 1e-18, 'gtol': 1e-14,
                         'maxiter': 5000, 'maxfun': 80000, 'maxls': 50}
            )
            a_new = np.clip(res.x, 0.0, 1000.0)
            S = float(a_new.sum())
            if S > 1e-6:
                a_new = a_new / S
                sc = compute_score(a_new)
                update_best(sc, a_new, "FIENUP+LBFGS")
    
    print(f"After Phase 2: best = {best_score:.10f}")
    
    # ==========================================================
    # Phase 3: Multi-restart exploration (20% of time)
    # ==========================================================
    phase3_end = t_start + 0.85 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Multi-restart (deadline t={phase3_end - t_start:.0f}s) ===")
    
    n_restarts = 5
    time_per_restart = (total_budget - 30.0) * 0.20 / n_restarts
    
    for r in range(n_restarts):
        if time.monotonic() > phase3_end:
            break
        
        # Generate perturbation
        scale = 0.02 + 0.03 * r
        a_pert = best_a * (1.0 + scale * rng.standard_normal(n))
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        S = float(a_pert.sum())
        if S < 1e-6:
            continue
        a_pert = a_pert / S
        
        # Short homotopy from perturbation
        t_restart_deadline = time.monotonic() + time_per_restart
        a_cur = a_pert.copy()
        
        for beta in [20, 100, 500]:
            if time.monotonic() > t_restart_deadline:
                break
            res = minimize(
                lse_objective_and_grad, a_cur.copy(), args=(beta,),
                method='L-BFGS-B', jac=True, bounds=bounds,
                options={'ftol': 1e-16, 'gtol': 1e-12,
