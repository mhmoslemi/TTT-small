"""
Improved multi-phase optimizer for non-negative step functions minimizing
2n*max(conv(a,a))/(sum(a))^2. Key improvements over previous: tighter FISTA
Lipschitz via FFT, aggressive L-BFGS with high-beta LSE, adaptive LP-to-LBFGS
transition, coordinate descent polish, and multi-restart strategy.
"""

import numpy as np
from scipy.optimize import minimize
import time


def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = self_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    z = beta * (c - c_max)
    z_shifted = z - z.max()
    exp_z = np.exp(z_shifted)
    log_sum = z.max() - z_shifted.max() + np.log(exp_z.sum())
    L_val = c_max + log_sum / beta
    score = 2.0 * n * L_val / (S * S)
    p = exp_z / exp_z.sum()
    # Gradient: 2n * (2*cross_conv(a,p)[n-1:2n-1]/S^2 - 2*L_val/S^3)
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(a[::-1], M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    grad_L = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_L / S2 - 2.0 * L_val / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=8000, maxfun=300000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * len(a0)
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def fista_lp(a, M, n_iters=5000, beta_start=3.0, beta_end=80.0):
    """
    FISTA for LP: max sum(x) s.t. conv(a,x) <= M, x >= 0.
    Uses tight Lipschitz constant from FFT.
    """
    n = len(a)
    x = np.ones(n) / n
    u = x.copy()
    t_mom = 1.0

    # Compute tight Lipschitz constant: L = beta * ||C||_2^2
    # ||C||_2 = max_omega |A(omega)| where A is DFT of a (zero-padded)
    M_fft = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    C_norm_sq = float(np.max(np.abs(A) ** 2))

    for it in range(n_iters):
        frac = min(1.0, it / (n_iters * 0.4))
        beta = beta_start + (beta_end - beta_start) * frac

        L = beta * C_norm_sq + 1e-12
        step = 1.0 / L

        c = cross_conv(a, x)
        z = beta * (c - M)
        z_max = float(z.max())
        exp_z = np.exp(z - z_max)
        exp_sum = float(exp_z.sum())
        if exp_sum < 1e-300:
            p = np.ones(len(c)) / len(c)
        else:
            p = exp_z / exp_sum

        # C^T p = a * sliding_window_sum(p)
        cum = np.concatenate([[0.0], np.cumsum(p)])
        sw = cum[n:] - cum[:len(p) - n + 1]
        grad = a * sw - 1.0

        v = u - step * grad
        x_new = np.maximum(v, 0.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom ** 2)) / 2.0
        u = x_new + (t_mom - 1.0) / t_new * (x_new - x)
        u = np.maximum(u, 0.0)
        x = x_new
        t_mom = t_new

    return x


def lp_alternating(a, max_outer=50, fista_iters=5000, t_start=0.5, t_min=0.03,
                   time_limit=None):

    S = float(a.sum())
    if S < 1e-12:
        a = np.ones(n) / n
    else:
        a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    no_improve = 0
    t = t_start

    for outer in range(max_outer):
        if time_limit is not None and time.monotonic() > time_limit:
            break

        M = float(self_conv(a).max())
        x = fista_lp(a, M, n_iters=fista_iters)
        Sx = float(x.sum())
        if Sx < 1e-12:
            break
        x = x / Sx

        a_new = (1.0 - t) * a + t * x
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            no_improve = 0
        else:
            no_improve += 1
            t *= 0.7
            if no_improve >= 20:
                break

        a = a_new

    return best_a, best_sc

            if a0 < 0.01:
                lo = 0.0
                hi = 0.1

            gr = (np.sqrt(5.0) - 1.0) / 2.0
            x1 = hi - gr * (hi - lo)
            x2 = lo + gr * (hi - lo)

            a[i] = x1
            s1 = compute_score(a)
            a[i] = x2
            s2 = compute_score(a)

            for _ in range(30):
                if hi - lo < 1e-8:
                    break
                if s1 < s2:
                    hi = x2
                    x2 = x1
                    s2 = s1
                    x1 = hi - gr * (hi - lo)
                    a[i] = x1
                    s1 = compute_score(a)
                else:
                    lo = x1
                    x1 = x2
                    s1 = s2
                    x2 = lo + gr * (hi - lo)
                    a[i] = x2
                    s2 = compute_score(a)

            a[i] = (lo + hi) / 2.0
            sc = compute_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
            else:
                a[i] = best_a[i]  # revert

    return best_a


def error_reduction(a, n_iters=150, alpha=0.7):
    """Newton-like error reduction in frequency domain."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        a_pad = np.zeros(M_fft, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.9 * excess
        err = conv_mod - conv
        err_pad = np.zeros(M_fft, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        A_new = A + d
        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def subgradient_polish(a, total_iters=5000):
    """Projected subgradient with Polyak step size on exact objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    gap_init = best_sc - 1.0  # lower bound is 1 (theoretical)
    if gap_init < 0.01:
        gap_init = 0.01

    for it in range(total_iters):
        c = self_conv(a)
        M_val = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M_val / 1.0  # normalized, sum=1
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient of max conv at a (normalized sum=1)
        sg = np.zeros(n)
        offset = k_star
        j_lo = max(0, offset - n + 1)
        j_hi = min(offset, n - 1)
        seg = a[j_lo:j_hi + 1]
        sg[j_lo:j_hi + 1] = 2.0 * seg[::-1]
        # Account for sum constraint: project gradient
        sg_mean = sg.mean()
        sg -= sg_mean  # tangent to simplex

        gn = float(np.linalg.norm(sg))
        if gn < 1e-15:
            break

        gap = max(0.01, best_sc - 1.0)
        step = gap / (gn * gn + 1e-30)
        step = min(step, 0.1)

        a -= step * sg
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations."""
    inits = []
    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            x_old = np.linspace(0, 1, len(parent))
            x_new = np.linspace(0, 1, n)
            inits.append(np.interp(x_new, x_old, parent).copy())

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)
    # Smooth noise
    noise = rng.standard_normal(n)
    kernel = np.exp(-0.5 * (np.arange(21) - 10) ** 2 / 10.0)
    kernel /= kernel.sum()
    smooth = np.convolve(noise, kernel, mode='same')
    inits.append(np.clip(0.5 + 0.3 * smooth, 0.01, 1000.0))
    inits.append(0.5 + 0.2 * np.cos(4.0 * np.pi * x) + 0.1 * np.cos(6.0 * np.pi * x))
    # Triangular
    inits.append(0.5 + 0.5 * np.abs(1.0 - 2.0 * x))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for non-negative sequence minimizing evaluation score.
    Uses LP alternating (FISTA), LSE L-BFGS, coordinate descent, and
    subgradient polish with adaptive time allocation.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s)) - 30.0

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    print(f"Parent: {parent is not None}, len={0 if parent is None else len(parent)}")

    # Determine lengths
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n_try in [4096, 6962, 8192]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))[:4]
    print(f"Lengths: {lengths}")

    # ============================================================
    # Phase 1: Quick global search (12% time)
    # ============================================================
    p1_end = t_start + 0.12 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 1: Global Search ---")

    for n in lengths:
        if time.monotonic() > p1_end:
            break
        remaining = p1_end - time.monotonic()
        if remaining < 20:
            break
        inits = build_initializations(n, rng, parent)
        max_inits = max(2, int(remaining / 8.0))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > p1_end:
                break
            a_er = error_reduction(a0, n_iters=80, alpha=0.75)
            sc = compute_score(a_er)
            report(sc, a_er, f"ER n={n} i={i}")

    print(f"Phase 1 best: {best_score:.8f}")

    # ============================================================
    # Phase 2: LP Alternating (35% time)
    # ============================================================
    p2_end = t_start + 0.47 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 2: LP Alternating ---")

    n_main = len(best_a)
    if parent is not None and len(parent) in lengths:
        n_main = len(parent)

    # Interpolate best to main length
    if len(best_a) != n_main:
        a_work = np.interp(np.linspace(0, 1, n_main),
                           np.linspace(0, 1, len(best_a)), best_a)
    else:
        a_work = best_a.copy()
    a_work = np.maximum(a_work, 0.0)
    S = float(a_work.sum())
    if S > 0:
        a_work = a_work / S

    # Multiple LP rounds with increasing fista_iters
    configs = [
        (3000, 0.5, 0.1, 15),
        (5000, 0.4, 0.08, 20),
        (7000, 0.3, 0.05, 25),
        (10000, 0.2, 0.03, 30),
    ]

    for fi, (fit, ts, tm, max_out) in enumerate(configs):
        if time.monotonic() > p2_end:
            break
        remaining = p2_end - time.monotonic()
        if remaining < 30:
            break
        # Estimate time per outer iter
        time_per_iter = fit * n_main / 800000.0  # rough estimate
        max_out_eff = max(3, int(remaining / max(1.0, time_per_iter)))
        max_out_eff = min(max_out_eff, max_out)
        print(f"  LP r{fi}: fista={fit}, t=[{ts},{tm}], outer~{max_out_eff}")

        a_out, sc_out = lp_alternating(
            a_work, max_outer=max_out_eff, fista_iters=fit,
            t_start=ts, t_min=tm, time_limit=p2_end - 5
        )
        sc = compute_score(a_out)
        report(sc, a_out, f"LP r{fi}")
        a_work = a_out.copy()

    print(f"Phase 2 best: {best_score:.8f}")

    # ============================================================
    # Phase 3: L-BFGS Refinement (35% time)
    # ============================================================
    p3_end = t_start + 0.82 * (float(budget_s) - 30.0)
    print(f"\n--- Phase 3: L-BFGS Refinement ---")

    n_best = len(best_a)
    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 1e-12:
        a_norm = a_norm / S

    # Aggressive beta schedule with multiple restarts
    beta_schedule = [50, 100, 200, 400, 800, 1500, 3000, 6000]
    for beta in beta_schedule:
        if time.monotonic() > p3_end:
            break
        remaining = p3_end - time.monotonic()
        if remaining < 15:
            break

        # For high beta, do multiple restarts
        n_restarts = 1 if beta < 400 else 2

        for r in range(n_restarts):
            if time.monotonic() > p3_end - 10:
                break
            if r == 0:
                a_in = a_norm.copy()
            else:
                scale = 0.02 + 0.02 * rng.random()
                a_in = a_norm * (1.0 + scale * rng.standard_normal(n_best))
                a_in = np.clip(a_in, 0.0, 1000.0)
                S_in = float(a_in.sum())
                if S_in > 1e-12:
                    a_in = a_in / S_in

            sc, a_opt = run_lbfgs(a_in, beta, maxiter=10000, maxfun=500000)
            report(sc, a_opt, f"LBFGS b={beta} r={r}")
            a_norm = a_opt.copy()
            S = float(a_norm.sum())
            if S > 0:
                a_norm = a_norm / S

    print(f"Phase 3 best: {best_score:.8f}")

    # ============================================================
    # Phase 4: Polish (remaining - 30s)
    # ============================================================
    p4_end = deadline
    print(f"\n--- Phase 4: Polish ---")

    a_norm = best_a.copy()
    S = float(a_norm.sum())
    if S > 1e-12:
        a_norm = a_norm / S

    # Coordinate descent
    if time.monotonic() < p4_end - 120:
        print("  Coordinate descent...")
        a_cd = coordinate_descent(a_norm, n_sweeps=8, n_coords_per_sweep=300,
                                  time_limit=p4_end - 120)
        sc = compute_score(a_cd)
        report(sc, a_cd, "coord")

    # Subgradient polish
    if time.monotonic() < p4_end - 120:
        print("  Subgradient polish...")
        a_sg = subgradient_polish(best_a, total_iters=8000)
        sc = compute_score(a_sg)
        report(sc, a_sg, "subgrad")

    # Final LP alternating with very small step
    if time.monotonic() < p4_end - 60:
        print("  Final LP (small step)...")
        a_final = best_a.copy()
        S = float(a_final.sum())
        if S > 0:
            a_final = a_final / S
        a_out, sc_out = lp_alternating(
            a_final, max_outer=30, fista_iters=8000,
            t_start=0.15, t_min=0.01, time_limit=p4_end - 30
        )
        sc = compute_score(a_out)
        report(sc, a_out, "final LP")

    # Final L-BFGS with very high beta
    if time.monotonic() < p4_end - 30:
        print("  Final L-BFGS...")
        a_norm = best_a.copy()
        S = float(a_norm.sum())
        if S > 0:
            a_norm = a_norm / S
        sc, a_opt = run_lbfgs(a_norm, 10000, maxiter=8000, maxfun=300000)
        report(sc, a_opt, "final LBFGS")

    # One more coordinate descent on the very best
    if time.monotonic() < p4_end - 15:
        print("  Final coord descent...")
        a_cd2 = coordinate_descent(best_a, n_sweeps=5, n_coords_per_sweep=200,
                                   time_limit=p4_end - 5)
        sc = compute_score(a_cd2)
        report(sc, a_cd2, "final coord")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
