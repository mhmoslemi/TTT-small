"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences a.

Algorithm:
1. Generate diverse initial sequences (from parent + structured inits).
2. LSE continuation with corrected gradient (fixes a bug in the previous 
   version where the second gradient term was missing the a_j factor).
3. Mirror descent / exponential updating for final polish.
4. Multi-length exploration to avoid local minima.
"""
import numpy as np
import time
from scipy.optimize import minimize

def next_power_of_two(m):
    m = max(1, int(m))
    return 1 << (m - 1).bit_length()

def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)

def corr_pair(a, w):
    """h_j = sum_k w_k * a_{k-j} for j = 0,...,n-1."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    m = w.size
    if n == 0 or m == 0:
        return np.empty(n, dtype=np.float64)
    M = n + m - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a[::-1], N)
    W = np.fft.rfft(w, N)
    full = np.fft.irfft(A * W, N)[:M]
    return full[n - 1:2 * n - 1]

def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, target / a.size, dtype=np.float64)
    return (a * (target / S)).astype(np.float64)

def resample_seq(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = a.size
    if m == 0:
        return np.ones(n, dtype=np.float64) / n
    if m == n:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, n)
    return np.interp(x_new, x_old, a).astype(np.float64)

def exact_score(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = max(0.0, float(np.max(c)))
    return float(2.0 * n * cmax / (S * S))

def lse_obj_grad(a, beta):
    """
    LSE-smoothed objective and corrected gradient.
    F = 2 * LSE_beta(d) + P*(S-1)^2
    where d_k = n*C(k)/S^2, C = conv(a,a), S = sum(a).
    
    Corrected gradient:
    dF/da_j = 4n*H_j/S^2 - 4*a_j*LDS_d/S + 2*P*(S-1)
    """
    n = len(a)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n, dtype=np.float64)
    
    c = fast_self_conv(a)
    d = n * c / (S * S)
    d = np.maximum(d, 0.0)
    
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0:
        return 1e10, np.zeros(n, dtype=np.float64)
    
    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n, dtype=np.float64)
    w = e / sum_e
    
    L = dmax + np.log(sum_e) / beta
    LDS_d = float(np.dot(w, d))
    
    H = corr_pair(a, w)
    
    P = 1e-2  # slightly larger penalty for better normalization
    score = 2.0 * L + P * (S - 1.0) ** 2
    
    # CORRECTED gradient
    grad = 4.0 * n * H / (S * S) - 4.0 * a * LDS_d / S + 2.0 * P * (S - 1.0)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    
    return float(score), grad

def run_lbfgs(a0, beta, deadline, maxfun=None):
    """Run L-BFGS-B on the LSE objective."""
    a = normalize_a(a0)
    n = len(a)
    
    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 10.0:
            return exact_score(a), a
    
    if maxfun is None:
        maxfun = 500
    
    bounds = [(0.0, None)] * n
    opts = {
        "ftol": 1e-18,
        "gtol": 1e-14,
        "maxiter": max(100, maxfun),
        "maxfun": maxfun,
        "maxls": 100,
    }
    
    def fun(x):
        return lse_obj_grad(x, beta)
    
    try:
        res = minimize(fun, a, method="L-BFGS-B", jac=True, bounds=bounds, options=opts)
        a_out = normalize_a(res.x) if res.x is not None else a
    except Exception:
        a_out = a
    
    sc = exact_score(a_out)
    if not np.isfinite(sc) or sc > 1e6:
        a_out = a
        sc = exact_score(a)
    return sc, a_out

    """
    Mirror descent (exponential updating) for the exact max objective.
    Works on the simplex (sum=1, a>=0).
    """
    a = normalize_a(a)
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    
    eta = 0.1  # initial step size
