def mirror_descent_optimize(a_init, beta_schedule, base_lr, n_steps, time_limit, t_start,
                            normalize=True):
    """Mirror descent (multiplicative weight) with LSE smoothing."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    if normalize:
        a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)

    n_beta = len(beta_schedule)
    steps_per_beta = max(1, n_steps // n_beta)

    for bi, beta in enumerate(beta_schedule):
        steps = min(steps_per_beta, n_steps - bi * steps_per_beta)
        for t in range(steps):
            if time.monotonic() - t_start > time_limit:
                return best_score, best_a

            score, grad = lse_score_grad(a, beta)
            if not np.isfinite(score):
                return best_score, best_a

            # Adaptive learning rate
            grad_inf = np.max(np.abs(grad))
            if grad_inf < 1e-15:
                return best_score, best_a
            lr_t = base_lr / (grad_inf + 1e-10)
            # Dampen for stability
            lr_t = min(lr_t, 0.5 / (grad_inf + 1e-10))

            # Multiplicative update
            log_update = -lr_t * grad
            # Clip to prevent overflow
            log_update = np.clip(log_update, -50, 50)
            a = a * np.exp(log_update)
            a = np.clip(a, 1e-15, 1e10)

            if normalize:
                s = a.sum()
                if s > 1e-12:
                    a = a / s

            if (bi * steps_per_beta + t) % 1000 == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def frank_wolfe_step(a, n_iter=100, time_limit=None, t_start=None):
    """Frank-Wolfe (conditional gradient) method on the simplex."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iter):
        if time_limit is not None and time.monotonic() - t_start > time_limit:
            break

        score, grad = subgrad_score(a)
        if not np.isfinite(score):
            break

        # Linear minimization oracle: pick coordinate with min gradient
        k_star = int(np.argmin(grad))

        # Step size: use standard FW schedule or line search
        t_fw = 2.0 / (it + 2.0)

        # Move toward e_{k_star}
        a_new = (1.0 - t_fw) * a + t_fw * np.zeros(n)
        a_new[k_star] += t_fw
        a_new = np.clip(a_new, 0, None)
        s = a_new.sum()
        if s > 1e-12:
            a_new = a_new / s

        sc_new = compute_score(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            a = a_new

    return best_score, best_a

def lbfgs_optimize(a_init, beta, maxiter=3000, maxfun=50000):
    """L-BFGS-B with LSE smoothing."""
    n = len(a_init)
    a0 = np.clip(a_init.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def obj_and_grad(x):
        x_c = np.clip(x, 0.0, 1e10)
        sc, gr = lse_score_grad(x_c, beta)
        if not np.isfinite(sc):
            return 1e10, np.zeros(n)
        # Gradient of LSE score is not the gradient of the true score,
        # but for optimization purposes it's a smooth approximation
        return sc, gr

    res = minimize(obj_and_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-14, 'gtol': 1e-10,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30})
    a_out = np.clip(res.x, 0.0, 1000.0)
    s = a_out.sum()
    if s > 1e-12:
        a_out = a_out / s
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_shaving(a, n_iters=200, time_limit=None, t_start=None):
    """Directly target and reduce convolution peaks."""
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_score = compute_score(a)

    for it in range(n_iters):
        if time_limit is not None and time.monotonic() - t_start > time_limit:
            break

        c = fast_conv(a)
        k_star = int(c.argmax())
        M = c[k_star]

        # Find elements contributing to the peak
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)

        # Reduce peak contributors
        contrib_with = a[js:je + 1].copy()
        # The contribution of a[j] to c[k_star] is a[k_star - j]
        weights = np.zeros(je - js + 1)
        for j in range(js, je + 1):
            idx = k_star - j
            if 0 <= idx < n:
                weights[j - js] = a[idx]

        # Reduce elements with high weight (they contribute most to the peak)
        if weights.max() > 0:
            reduction = 1.0 - 0.1 * (weights / weights.max())
            a[js:je + 1] *= reduction
            a = np.clip(a, 0, None)
            s = a.sum()
            if s > 1e-12:
                a = a / s

        sc = compute_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

    return best_score, best_a


def generate_inits(rng, n):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64)

    # 1/sqrt profile (theoretically optimal in continuous limit)
    a1 = 1.0 / np.sqrt(x + 1.0)
    inits.append(a1 / a1.sum())

    # 1/sqrt with exponential taper
    for gamma in [0.5, 1.0, 2.0]:
        a2 = (1.0 / np.sqrt(x + 1.0)) * np.exp(-gamma * x / n)
        inits.append(a2 / a2.sum())

    # 1/(i+1)^alpha for various alpha
    for alpha in [0.3, 0.4, 0.6, 0.7]:
        a3 = 1.0 / (x + 1.0) ** alpha
        inits.append(a3 / a3.sum())

    # 1/sqrt symmetric window: 1/sqrt((i+1)*(n-i))
    a4 = 1.0 / np.sqrt((x + 1.0) * (n - x))
    inits.append(a4 / a4.sum())

    # Gaussian centered
    sig = 0.2
    a5 = np.exp(-0.5 * ((x / n - 0.3) / sig) ** 2)
    inits.append(a5 / a5.sum())

    # Two-bump
    a6 = np.exp(-0.5 * ((x / n - 0.25) / 0.1) ** 2) + np.exp(-0.5 * ((x / n - 0.7) / 0.15) ** 2)
    inits.append(a6 / a6.sum())

    # Random smooth
    r = np.abs(rng.standard_normal(n)) + 0.3
    k = max(3, n // 20)
    kernel = np.ones(k) / k
    r_smooth = np.convolve(r, kernel, mode='same')
    inits.append(r_smooth / r_smooth.sum())

    # Fourier-modulated 1/sqrt
    freq = 3.0
    a7 = (1.0 / np.sqrt(x + 1.0)) * (1.0 + 0.3 * np.cos(2 * np.pi * freq * x / n))
    a7 = np.maximum(a7, 1e-6)
    inits.append(a7 / a7.sum())

    return [np.clip(a, 1e-10, 1000.0) for a in inits]


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)
