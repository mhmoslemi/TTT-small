def lse_grad(a, beta):
    """Compute LSE-smoothed log-conv and its gradient."""
    n = len(a)
    c = fast_conv(a)  # length 2n-1
    M_val = c.max()
    scaled = beta * (c - M_val)
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    
    # Compute gradient
    M = 1 << (3*n-2-1).bit_length()  # FFT size >= 3n-2
    A = np.zeros(M)
    A[:n] = a
    W = np.zeros(M)
    W[:2*n-1] = w
    
    A_fft = np.fft.rfft(A)
    W_fft = np.fft.rfft(W)
    
    T1 = np.fft.irfft(np.conj(A_fft) * W_fft, M)[:n]
    T2 = np.fft.irfft(W_fft * A_fft, M)[:n]
    
    grad = T1 + T2
    lse_val = M_val + np.log(exp_scaled.sum()) / beta
    
    return lse_val, grad, w

def optimize_lse(a_init, beta_schedule, lr_schedule, time_limit, t_start):
    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    
    best_score = compute_score(a)
    best_a = a.copy()
    
    for beta, lr in zip(beta_schedule, lr_schedule):
        while time.monotonic() - t_start < time_limit:
            t += 1
            lse_val, grad, w = lse_grad(a, beta)
            
            # Adam update
            m = b1*m + (1-b1)*grad
            v = b2*v + (1-b2)*grad**2
            m_hat = m / (1 - b1**t)
            v_hat = v / (1 - b2**t)
            
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = np.maximum(a, 0)
            a = a / a.sum()
            
            # Check score periodically
            if t % 100 == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
    
    return best_score, best_a
```

Key mathematical insight: The gradient of the LSE-smoothed log-max-conv can be computed as a sum of two cross-correlations, both computable via FFT in O(n log n).

```python
"""
Full-space LSE-Adam optimizer for minimizing 2n*max(self-conv)/(sum)^2.
Key: works in full (non-palindromic) space to break the score>=2 barrier.
Uses FFT-based gradient for LSE-smoothed log-max-convolution objective.
Multi-phase: diverse init -> LSE continuation Adam -> noise exploration -> high-beta polish.
"""

import numpy as np
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    Mf = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, Mf)
    return np.fft.irfft(A * A, Mf)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def lse_grad(a, beta):
    """
    Compute LSE-smoothed log(sum exp(beta*c[k])) and its gradient.
    Returns (lse_value, gradient_array).
    Uses FFT-based cross-correlation for O(n log n) gradient.
    """
    n = len(a)
    c = fast_conv(a)
    M_val = float(c.max())
    scaled = beta * (c - M_val)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    w = exp_scaled / exp_scaled.sum()
    lse_val = M_val + float(np.log(exp_scaled.sum())) / beta

    # FFT size: need M >= 3n-2 for the cross-correlation
    Mf = 1 << (3 * n - 2 - 1).bit_length()
    A = np.zeros(Mf)
    A[:n] = a
    W = np.zeros(Mf)
    W[:2 * n - 1] = w

    A_fft = np.fft.rfft(A)
    W_fft = np.fft.rfft(W)

    # T1[i] = sum_l a[l] * w[l+i] = IFFT(conj(FFT(A)) * FFT(W))[i]
    T1 = np.fft.irfft(np.conj(A_fft) * W_fft, Mf)[:n]
    # T2[i] = sum_l w[l] * a[i-l] = IFFT(FFT(W) * FFT(A))[i] (convolution)
    T2 = np.fft.irfft(W_fft * A_fft, Mf)[:n]

    grad = T1 + T2
    return lse_val, grad


def subgrad_score(a):
    """Exact score and subgradient (for the argmax lag)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(c.argmax())
    score = 2.0 * n * M / (S * S)

    # dc[k_star]/da[i] = a[k_star - i] + a[i - k_star] (with bounds)
    grad_c = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        # From first term: a[k_star - i] for i in [js, je]
        grad_c[js:je + 1] += a[k_star - np.arange(js, je + 1)]
        # From second term: a[i - k_star] for valid i-k_star
        js2 = max(k_star, 0)
        je2 = min(k_star + n - 1, n - 1)
        if je2 >= js2:
            grad_c[js2:je2 + 1] += a[np.arange(js2, je2 + 1) - k_star]

    # d(score)/da = (2n) * (grad_c / S^2 - 2*M*grad_c_sum / S^3)
    # But we need to account for the constraint sum(a)=1
    # Using Lagrange: effective gradient = 2n * grad_c / S^2 - 2n * M * (2/S^3) * ones
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n / S2) * grad_c - (4.0 * n * M / S3) * np.ones(n)
    return score, grad


def adam_lse_full(a_init, beta_schedule, lr_schedule, time_limit, t_start,
                  noise_every=5000, noise_scale=0.0, verbose_every=5000):
    """
    Adam optimizer on the LSE-smoothed objective in full space.
    a is normalized to sum=1 after each step.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0.0).astype(np.float64)
    a = a / a.sum()

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0

    best_score = compute_score(a)
    best_a = a.copy()
    n_improve = 0

    for bi, (beta, lr) in enumerate(zip(beta_schedule, lr_schedule)):
        while time.monotonic() - t_start < time_limit:
            t += 1
            lse_val, grad = lse_grad(a, beta)

            # Noise injection
            if noise_scale > 0 and t % noise_every == 0:
                noise = noise_scale * np.random.standard_normal(n)
                a = a + noise
                a = np.maximum(a, 0.0)
                a = a / a.sum()

            # Adam update
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            t_eff = t + 1
            m_hat = m / (1.0 - b1 ** t_eff)
            v_hat = v / (1.0 - b2 ** t_eff)
            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
            a = np.maximum(a, 0.0)
            a = a / a.sum()

            if t % verbose_every == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    n_improve += 1
                    elapsed = time.monotonic() - t_start
                    print(f"    [imp] score={sc:.8f} beta={beta} t={t} elapsed={elapsed:.0f}s")

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def subgrad_descent(a_init, time_limit, t_start, lr_init=0.01, decay=0.999):
    """
    Projected subgradient descent targeting the argmax lag.
    Normalizes to sum=1, projects to non-negative.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0.0).astype(np.float64)
    a = a / a.sum()

    best_score = compute_score(a)
    best_a = a.copy()
    lr = lr_init
    t = 0

    while time.monotonic() - t_start < time_limit:
        t += 1
        score, grad = subgrad_score(a)

        # Projected subgradient step
        a = a - lr * grad
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.ones(n) / n
        else:
            a = a / s

        lr *= decay

        if t % 100 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                if t % 1000 == 0:
                    print(f"    [subgrad] score={sc:.8f} t={t}")

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def coord_descent_refine(a_init, time_limit, t_start):
    """
    Coordinate descent: try small perturbations of individual elements
    to reduce the max convolution value.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0.0).astype(np.float64)
    a = a / a.sum()
    best_score = compute_score(a)
    best_a = a.copy()
    rng = np.random.default_rng(123)

    while time.monotonic() - t_start < time_limit:
        c = fast_conv(a)
        k_star = int(c.argmax())
        cur_score = 2.0 * n * float(c.max()) / (a.sum() ** 2)

        # Find elements contributing to the peak
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        contrib_idx = np.arange(js, je + 1)
        paired_idx = k_star - contrib_idx

        # Try reducing a contributor and boosting a non-contributor
        if len(contrib_idx) > 0:
            j = contrib_idx[rng.integers(len(contrib_idx))]
            p = paired_idx[rng.integers(len(contrib_idx))]

            # Reduce a[j], increase a[p] (paired element at peak)
            for frac in [0.9, 0.8, 0.7, 0.5]:
                saved = a[j]
                a[j] = saved * frac
                a[p] = a[p] + saved * (1 - frac)
                a = np.maximum(a, 0)
                a = a / a.sum()
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    break
                a[j] = saved
                a[p] = a[p] - saved * (1 - frac)

        # Try random coordinate perturbation
        for _ in range(3):
            j = rng.integers(n)
            saved = a[j]
            if saved < 1e-10:
                continue
            for mult in [0.7, 1.3, 0.5, 1.5]:
                a[j] = saved * mult
                s = a.sum()
                if s < 1e-12:
                    a[j] = saved
                    continue
                a = a / s
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    break
            else:
                a[j] = saved
                a = a / a.sum()

    return best_score, best_a


def generate_initializations(n, rng):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    # Uniform
    inits.append(np.ones(n) / n)

    # Ramp (monotonically increasing)
    r = np.arange(1, n + 1, dtype=np.float64)
    inits.append(r / r.sum())

    # Reverse ramp
    inits.append(r[::-1] / r.sum())

    # Triangle (peak in middle)
    tri = np.minimum(np.arange(n), n - np.arange(n)).astype(np.float64) + 1
    inits.append(tri / tri.sum())

    # Asymmetric triangle
    peak = int(0.3 * n)
    tri_asym = np.zeros(n)
    for i in range(n):
        if i <= peak:
            tri_asym[i] = i + 1
        else:
            tri_asym[i] = max(0.1, (n - i) / (n - peak))
    inits.append(tri_asym / tri_asym.sum())

    # Sinusoidal (single period)
    s = 1.0 + 0.5 * np.sin(2 * np.pi * x)
    inits.append(np.maximum(s, 0.01) / np.maximum(s, 0.01).sum())

    # Two-frequency modulation
    f1, f2 = 2.0, 5.0
    m = 1.0 + 0.3 * np.cos(2 * np.pi * f1 * x) + 0.2 * np.cos(2 * np.pi * f2 * x)
    inits.append(np.maximum(m, 0.01) / np.maximum(m, 0.01).sum())

    # Smooth random
    r = np.abs(rng.standard_normal(n)) + 0.3
    k = max(3, n // 100)
    r = np.convolve(r, np.ones(k) / k, mode='same')
    inits.append(r / r.sum())

    # Gaussian (off-center)
    center = 0.3 + 0.4 * rng.random()
    sigma = 0.1 + 0.15 * rng.random()
    g = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    inits.append(g / g.sum())

    # "Anti-palindrome": high on one side, low on other
    ap = np.maximum(0.1, 1.0 - x)
    inits.append(ap / ap.sum())

    return inits


def smooth_transition(a, b, t):
    """Smooth interpolation between two sequences."""
    return (1 - t) * a + t * b
