        if P_max < 1e-20:
            return a.copy()
        # Power-law flattening

        A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))

def time_perturb(a, rng, scale=0.03):

def nesterov_pass(a, n_iters=3000):
    """Quick Nesterov subgradient pass."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]
        
        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        step = (sc - 1.0) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.3)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new
        
        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()
    
    return best_a

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_and_grad(x, beta):
    """LSE-smoothed score and gradient."""
    n = len(x)
    c = fast_conv(x)
    S = float(x.sum())
    if S < 1e-12:

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -100))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)
    rev_a = x[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()

    g = 2.0 * conv_res

        t_val = t_new
        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = float(a.sum())
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()
    return best_a

def targeted_reduction(a, n_sweeps=3, n_top=50):
    """Reduce peak contributors and redistribute mass."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    for _ in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
        for li in top_local:
            j = j_start + li
            a_old = a[j]
            if a_old < 1e-10:
                continue
            reduction = a_old * 0.5
            a[j] = a_old - reduction
            per = reduction / len(bottom_local)
            for bi in bottom_local:
                a[j_start + bi] += per
        S = float(a.sum())
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a

def interpolate_sequence(src, dst_n):
    """Interpolate a sequence to a new length."""
    n_src = len(src)
    if n_src == dst_n:
        return src.copy()
    x_src = np.linspace(0, 1, n_src)
    x_dst = np.linspace(0, 1, dst_n)
    out = np.interp(x_dst, x_src, src)
    return np.maximum(out, 0.0)

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-length spectral optimizer. Uses frequency-domain moves for global
    exploration, multi-length L-BFGS for diverse local optima, and very
    high-beta LSE smoothing for final precision.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 15.0

    best_score = np.inf
    best_a = np.ones(64) * 0.5

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None and parent.sum() > 0:
        parent_norm = parent / parent.sum()
        sc_p = compute_score(parent_norm)
        print(f"Parent: n={len(parent)}, score={sc_p:.10f}")
        if sc_p < best_score:
            best_score = sc_p
            best_a = parent_norm.copy()
            print(f"  [parent] NEW BEST: {best_score:.10f}")
    else:
        print("No parent found, starting from scratch.")
        best_a = np.ones(4096) * 0.5

    # ==================================================================
    # Phase 0: Quick initial polish (5% time)
    # ==================================================================
    p0_end = t_start + 0.05 * (total_budget - 15.0)
    print(f"\n--- Phase 0: Initial Polish ---")
    sc, a0 = run_lbfgs(best_a, beta=300, maxiter=20000, maxfun=80000)
    if sc < best_score:
        best_score = sc
        best_a = a0
        print(f"  [init] NEW BEST: {sc:.10f} (n={len(a0)})")
    print(f"After Phase 0: best = {best_score:.10f}")

    # ==================================================================
    # Phase 1: Multi-length L-BFGS exploration (45% time)
    # ==================================================================
    p1_end = t_start + 0.50 * (total_budget - 15.0)
    print(f"\n--- Phase 1: Multi-length L-BFGS ---")

    base_n = len(best_a)
    # Candidate lengths to try
    cand_lengths = []
    if base_n >= 64:
        cand_lengths.append(base_n)
    cand_lengths.extend([
        int(base_n * 1.1),
        int(base_n * 1.3),
        int(base_n * 1.5),
        int(base_n * 2.0),
        4096, 8192, 10240, 12288, 16384, 20480,
    ])
    # Deduplicate and filter
    cand_lengths = sorted(set(n for n in cand_lengths if 1024 <= n <= 32768))

    for n_try in cand_lengths:
        if time.monotonic() > p1_end - 30:
            break
        remaining = p1_end - time.monotonic()
        if remaining < 45:
            break
        print(f"  n={n_try} (remaining {remaining:.0f}s)")

        a_curr = interpolate_sequence(best_a, n_try)
        S_c = float(a_curr.sum())
        if S_c > 1e-12:
            a_curr = a_curr / S_c

        # Progressive beta increase
        beta_list = [200, 1000, 5000, 20000, 50000]
        for beta in beta_list:
            if time.monotonic() > p1_end - 10:
                break
            sc, a_opt = run_lbfgs(a_curr, beta=beta, maxiter=30000, maxfun=100000)
            if sc < best_score:
                best_score = sc
                best_a = a_opt
                print(f"  [n={n_try} beta={beta}] NEW BEST: {sc:.10f}")
            a_curr = a_opt

    print(f"After Phase 1: best = {best_score:.10f} (n={len(best_a)})")

    # ==================================================================
    # Phase 2: ILS with spectral and time-domain moves (30% time)
    # ==================================================================
    p2_end = t_start + 0.80 * (total_budget - 15.0)
    print(f"\n--- Phase 2: ILS with Spectral Moves ---")

    n_best = len(best_a)
    trial = 0
    max_trials = 40
    while time.monotonic() < p2_end - 20 and trial < max_trials:
        trial += 1
        remaining = p2_end - time.monotonic()
        if remaining < 25:
            break

        # Choose perturbation strategy
        r = rng.random()
        if r < 0.3:
            a_pert = spectral_flatten_move(best_a, rng)
        elif r < 0.5:
            a_pert = spectral_phase_move(best_a, rng)
        elif r < 0.65:
            a_pert = spectral_mag_move(best_a, rng)
        else:
            a_pert = time_perturb(best_a, rng)

        # L-BFGS polish
        beta = int(300 + 4000 * rng.random())
        sc, a_opt = run_lbfgs(a_pert, beta=beta, maxiter=20000, maxfun=60000)
        improved = False
        if sc < best_score:
            best_score = sc
            best_a = a_opt
            improved = True
            print(f"  [ILS #{trial}] NEW BEST: {sc:.10f} (n={len(a_opt)})")

        # If improved, do a second polish
        if improved and time.monotonic() < p2_end - 20:
            sc2, a_opt2 = run_lbfgs(a_opt, beta=10000, maxiter=20000, maxfun=60000)
            if sc2 < best_score:
                best_score = sc2
                best_a = a_opt2
                print(f"  [ILS #{trial} polish] NEW BEST: {sc2:.10f}")

    print(f"After Phase 2: best = {best_score:.10f} (n={len(best_a)})")

    # ==================================================================
    # Phase 3: Nesterov + targeted reduction (10% time)
    # ==================================================================
    p3_end = t_start + 0.90 * (total_budget - 15.0)
    print(f"\n--- Phase 3: Nesterov + Targeted ---")

    a_ns = nesterov_pass(best_a, n_iters=3000)
    sc_ns = compute_score(a_ns)
    if sc_ns < best_score:
        best_score = sc_ns
        best_a = a_ns
        print(f"  [nesterov] NEW BEST: {sc_ns:.10f}")

    a_tr = targeted_reduction(best_a, n_sweeps=3, n_top=50)
    sc_tr = compute_score(a_tr)
    if sc_tr < best_score:
        best_score = sc_tr
        best_a = a_tr
        print(f"  [targeted] NEW BEST: {sc_tr:.10f}")

    # L-BFGS after Nesterov
    if time.monotonic() < p3_end - 10:
        sc, a_opt = run_lbfgs(best_a, beta=20000, maxiter=30000, maxfun=100000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt
            print(f"  [post-nesterov] NEW BEST: {sc:.10f}")

    print(f"After Phase 3: best = {best_score:.10f}")

    # ==================================================================
    # Phase 4: Final high-precision polish (10% time)
    # ==================================================================
    print(f"\n--- Phase 4: Final Polish ---")

    final_betas = [50000, 100000, 200000, 500000]
    for beta in final_betas:
        if time.monotonic() > deadline - 5:
            break
        sc, a_opt = run_lbfgs(best_a, beta=beta, maxiter=50000, maxfun=200000)
        if sc < best_score:
            best_score = sc
            best_a = a_opt
            print(f"  [final beta={beta}] NEW BEST: {sc:.10f}")

    # Try one more length if time permits
    if time.monotonic() < deadline - 60:
        n_alt = int(len(best_a) * 1.2)
        if 1024 <= n_alt <= 32768 and n_alt != len(best_a):
            a_alt = interpolate_sequence(best_a, n_alt)
            S_alt = float(a_alt.sum())
            if S_alt > 1e-12:
                a_alt = a_alt / S_alt
            sc, a_opt = run_lbfgs(a_alt, beta=50000, maxiter=30000, maxfun=100000)
            if sc < best_score:
                best_score = sc
                best_a = a_opt
                print(f"  [alt_len={n_alt}] NEW BEST: {sc:.10f}")

    # ==================================================================
    # Final output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
