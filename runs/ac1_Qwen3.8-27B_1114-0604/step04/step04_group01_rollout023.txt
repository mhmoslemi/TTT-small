def _next_pow2(m):
    m = max(1, int(m))

def _self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    M = 2 * n - 1
    N = _next_pow2(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.maximum(out, 0.0)


def _corr_vec(a, w):
    """h[i] = sum_k w[k]*a[k-i] for i=0..n-1, via FFT convolution."""
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    mw = w.size
    if n == 0 or mw == 0:
        return np.empty(n)
    w_flip = w[::-1]

    Af = np.fft.rfft(a, N)

    conv = np.fft.irfft(Af * Wf, N)[:M]

    end = 2 * n - 2
    h = conv[start:end + 1][::-1]

    S = float(a.sum())

    grad -= (4.0 * float(n) * LDS) / (S * S * S)
    grad += 2.0 * PEN_SUM * (S - 1.0)

    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def _lbfgs_run(a0, beta, deadline, maxfun):
    a = _normalize(a0)
    n = a.size
    rem = max(1.0, float(deadline - time.monotonic()))
    if rem < 5.0:
        return _score(a), a

    est = 0.015 * max(1.0, n / 4096.0) * max(1.0, np.log2(max(16, n)) / 13.0)
    mf = int(min(maxfun, (0.85 * rem) / max(est, 1e-4)))
    mf = max(32, min(mf, 2000))
    maxiter = max(30, mf // 2)

    def obj(x):
        a_t = np.maximum(x, 0.0)
        a_t = a_t / max(float(a_t.sum()), 1e-14)
        val, g = _lse_obj_grad(a_t, beta)
        g = np.maximum(g, -1e5)
        return val, g

    opts = {"ftol": 1e-14, "gtol": 1e-11, "maxiter": maxiter, "maxfun": mf, "maxls": 40}
    bounds = [(0.0, 10.0)] * n

    try:
        res = minimize(obj, a, method="L-BFGS-B", jac=True, bounds=bounds, options=opts)
        a_out = _normalize(np.maximum(res.x, 0.0)) if res.x is not None else a
    except Exception:
        a_out = a

    sc = _score(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = _score(a)
    return float(sc), a_out


def _adam_run(a0, beta, deadline, max_steps):
    """Adam optimization on the LSE objective with log-barrier for non-negativity."""
    a = _normalize(a0)
    n = a.size
    best_a = a.copy()
    best_sc = _score(a)
    t0 = time.monotonic()

    # Adam state
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    lr = 0.005
    b1, b2, eps = 0.9, 0.999, 1e-8

    for step in range(max_steps):
        if time.monotonic() > deadline:
            break
        if (step % 200 == 0) and (time.monotonic() - t0) > 0:
            # Adaptive LR decay
            lr *= 0.97

        val, grad = _lse_obj_grad(a, beta)
        if not np.isfinite(val):
            break

        # Only update non-zero components effectively
        m_state = b1 * m_state + (1 - b1) * grad
        v_state = b2 * v_state + (1 - b2) * grad ** 2
        m_hat = m_state / (1 - b1 ** (step + 1))
        v_hat = v_state / (1 - b2 ** (step + 1))

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        a = _normalize(a_new)

        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return float(best_sc), best_a


def _pressure_update(a, deadline, max_rounds=50):
    """Multiplicative pressure: reduce a[i] where it contributes most to autocorrelation peaks."""
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = a.size

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _self_conv(a)
        S2 = float(a.sum()) ** 2
        c_norm = c / S2
        cmax = float(c_norm.max())
        mean_c = float(c_norm.mean())

        if cmax <= mean_c * 1.01:
            break

        # Soft weights: exponential of (c - threshold)
        threshold = mean_c
        z = 50.0 * (c_norm - threshold)
        z = np.clip(z, -100.0, 100.0)
        w = np.exp(z)
        w_sum = float(w.sum())
        if w_sum < 1e-10:
            break
        w = w / w_sum

        # Pressure direction: h[i] = sum_k w[k] * a[k-i]
        h = _corr_vec(a, w)
        h = np.maximum(h, 1e-12)
        h_max = float(h.max())
        if h_max < 1e-12:
            break

        # Try multiple step sizes
        best_step_sc = best_sc
        best_step_a = best_a
        for mult in (0.01, 0.03, 0.08, 0.15, 0.30):
            eta = mult / h_max
            q = a * np.exp(-eta * h)
            q = _normalize(q)
            sc = _score(q)
            if np.isfinite(sc) and sc < best_step_sc:
                best_step_sc = sc
                best_step_a = q.copy()

        if best_step_sc < best_sc:
            best_sc = best_step_sc
            best_a = best_step_a.copy()
            a = best_a
        else:
            # Try smaller steps
            improved = False
            for mult in (0.003, 0.005, 0.01):
                eta = mult / h_max
                q = a * np.exp(-eta * h)
                q = _normalize(q)
                sc = _score(q)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                    a = q
                    improved = True
                    break
            if not improved:
                break

    return float(best_sc), best_a


def _exact_subgrad_polish(a, deadline, max_rounds=30):
    """Exact subgradient: push down only the argmax positions of the autocorrelation."""
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = a.size

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _self_conv(a)
        cmax = float(c.max())
        if not np.isfinite(cmax) or cmax <= 0:
            break

        # Find all positions within tolerance of max
        tol = 1e-6 * cmax
        peak_idx = np.where(c >= cmax - tol)[0]

        # Weight: uniform over peaks
        w = np.zeros_like(c)
        w[peak_idx] = 1.0 / len(peak_idx)

        h = _corr_vec(a, w)
        h = np.maximum(h, 1e-12)
        h_max = float(h.max())
        if h_max < 1e-12:
            break

        improved = False
        for mult in (0.005, 0.01, 0.02, 0.05, 0.10):
            eta = mult / h_max
            q = a * np.exp(-eta * h)
            q = _normalize(q)
            sc = _score(q)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = q.copy()
                a = q
                improved = True
                break

        if not improved:
            break

    return float(best_sc), best_a


def _flatten_autocorr(a, deadline, max_rounds=30):
    """Target all supra-mean autocorrelation values, not just the max."""
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score(a)
    n = a.size

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        c = _self_conv(a)
        S2 = float(a.sum()) ** 2
        c_norm = c / S2
        cmax = float(c_norm.max())
        mean_c = float(c_norm.mean())

        if cmax <= mean_c * 1.005:
            break

        # Weight: soft exponential above mean
        z = 30.0 * (c_norm - mean_c)
        z = np.clip(z, -80.0, 80.0)
        w = np.exp(z)
        w_sum = float(w.sum())
        if w_sum < 1e-10:
            break
        w = w / w_sum

        h = _corr_vec(a, w)
        h = np.maximum(h, 1e-12)
        h_max = float(h.max())
        if h_max < 1e-12:
            break

        improved = False
        for mult in (0.01, 0.02, 0.05, 0.10, 0.20):
            eta = mult / h_max
            q = a * np.exp(-eta * h)
            q = _normalize(q)
            sc = _score(q)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_a = q.copy()
                a = q
                improved = True
                break

        if not improved:
            break

    return float(best_sc), best_a


def _generate_inits(n, rng, parent):
    inits = []
    x = np.linspace(0.0, 1.0, n)

    # Parent resampled
    if parent is not None and parent.size > 0:
        pr = _resample(parent, n)
        inits.append(_normalize(pr))
        inits.append(_normalize(pr[::-1]))

    # Uniform
    inits.append(np.ones(n) / n)

    # U-shape (bathtub): larger at endpoints
    for alpha in (0.3, 0.6, 1.0, 1.5, 2.0):
        vals = 1.0 + alpha * (2.0 * x - 1.0) ** 2
        inits.append(_normalize(vals))

    # Inverted U (hill): larger in center
    for alpha in (0.3, 0.6, 1.0):
        vals = 1.0 + alpha * np.exp(-((x - 0.5) ** 2) / (2 * 0.15 ** 2))
        inits.append(_normalize(vals))

    # Double bump
    for w in (0.1, 0.15, 0.25):
        vals = 1.0 + 1.5 * (np.exp(-((x / w) ** 2)) + np.exp(-(((1.0 - x) / w) ** 2)))
        inits.append(_normalize(vals))

    # Sine waves
    for f in (2, 3, 5, 8):
        for amp in (0.2, 0.5, 0.8):
            vals = 1.0 + amp * np.cos(2 * np.pi * f * x)
            inits.append(_normalize(vals))

    # Random smooth (low-pass filtered noise)
    for _ in range(5):
        spec = 1.0 / (1.0 + (np.fft.rfftfreq(n) * n * 0.1) ** 2)
        spec *= np.exp(1j * rng.uniform(-np.pi, np.pi, size=spec.size))
        vals = np.fft.irfft(spec * n, n)
        vals = vals - vals.min() + 0.1
        inits.append(_normalize(vals.astype(np.float64)))

    # Random
    for _ in range(3):
        inits.append(_normalize(0.1 + 0.9 * rng.random(n)))

    # Block-structured
    for nb in (16, 32, 64):
        if nb > n:
            nb = max(2, n // 2)
        bv = rng.random(nb)
        a = np.repeat(bv, n // nb + 1)[:n]
        inits.append(_normalize(0.1 + 0.9 * a))

    # Triangular envelope
    tri = np.minimum(x, 1.0 - x) * 2.0
    inits.append(_normalize(0.3 + 0.7 * tri))
    inits.append(_normalize(0.3 + 0.7 * (1.0 - tri)))

    return inits


def _optimize_sequence(a0, deadline, rng, label=""):
    """Multi-phase optimization of a single sequence."""
    a = _normalize(a0)
    best_a = a.copy()
    best_sc = _score(a)
    n = a.size
    t0 = time.monotonic()

    # Phase 1: Gradual LSE continuation with L-BFGS-B
    betas = [20, 50, 100, 200, 500, 1000, 2000, 5000, 10000]
    for beta in betas:
        rem = float(deadline - time.monotonic())
        if rem < 10.0:
            break
        if rem < 30 and beta > 2000:
            break

        # Allocate time proportional to remaining budget
        frac = min(1.0, 30.0 / rem)
        mf = int(frac * min(500, max(100, int(rem * 0.3 / 0.05))))
        mf = max(50, min(mf, 400))

        sc, a2 = _lbfgs_run(a, float(beta), deadline, mf)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
            if label:
                print("  [{}] beta={:.0f} score={:.8f} t={:.0f}s".format(
                    label, beta, best_sc, time.monotonic() - t0))
        a = _normalize(a2)

    # Phase 2: Adam refinement at moderate beta
    rem = float(deadline - time.monotonic())
    if rem > 30:
        max_steps = int(min(3000, rem * 0.4 / 0.05))
        max_steps = max(200, min(max_steps, 2000))
        sc, a2 = _adam_run(a, 1000.0, t0 + min(rem * 0.4, 120.0), max_steps)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
            print("  [{}] adam score={:.8f}".format(label, best_sc))
        a = _normalize(a2)

    # Phase 3: Flatten autocorrelation
    rem = float(deadline - time.monotonic())
    if rem > 20:
        dl = t0 + min(rem * 0.3, 60.0)
        sc, a2 = _flatten_autocorr(a, dl, max_rounds=40)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
            print("  [{}] flatten score={:.8f}".format(label, best_sc))
        a = _normalize(a2)

    # Phase 4: Exact subgradient polish
    rem = float(deadline - time.monotonic())
    if rem > 10:
        dl = t0 + min(rem * 0.4, 40.0)
        sc, a2 = _exact_subgrad_polish(a, dl, max_rounds=30)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
            print("  [{}] subgrad score={:.8f}".format(label, best_sc))

    return float(best_sc), best_a


def _quick_optimize(a, deadline):
    """Fast multi-beta optimization."""
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score(a)

    for beta in (100, 500, 2000, 8000):
        rem = float(deadline - time.monotonic())
        if rem < 10:
            break
        mf = max(50, int(rem * 0.6 / 0.05))
        mf = min(mf, 300)
        sc, a2 = _lbfgs_run(a, float(beta), deadline, mf)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()
        a = _normalize(a2)

    rem = float(deadline - time.monotonic())
    if rem > 10:
        dl = t0 + min(rem * 0.5, 30.0)  # won't work, fix below
        dl = time.monotonic() + min(rem * 0.5, 30.0)
        sc, a2 = _exact_subgrad_polish(a, dl, max_rounds=20)
        if sc < best_sc:
            best_sc = sc
            best_a = a2.copy()

    return float(best_sc), best_a


def _perturb(a, rng, strength=None):
    """Generate a perturbed version of a for diversity."""
    p = _normalize(a)
    n = p.size
    if n == 0:
        return p
    if strength is None:
        strength = float(rng.uniform(0.1, 0.5))

    mode = int(rng.integers(0, 5))
    if mode == 0:
        q = p * np.exp(strength * rng.standard_normal(n))
    elif mode == 1:
        t = float(rng.uniform(0.1, 0.4))
        q = (1 - t) * p + t * (np.ones(n) / n)
    elif mode == 2:
        q = p[::-1] * np.exp(0.05 * rng.standard_normal(n))
    elif mode == 3:
        # Localized perturbation
        center = int(rng.integers(0, n))
        width = int(rng.integers(max(3, n // 50), max(4, n // 10)))
        mask = np.exp(-((np.arange(n) - center) ** 2) / (2 * width ** 2))
        q = p * (1.0 + strength * mask * rng.standard_normal(n))
    else:
        # Block perturbation
        nblocks = int(rng.integers(4, 16))
        block_size = n // nblocks
        q = p.copy()
        for b in range(nblocks):
            s = b * block_size
            e = min(n, (b + 1) * block_size)
            q[s:e] *= np.exp(0.2 * rng.standard_normal(e - s))
        q = np.maximum(q, 0)

    return _normalize(q)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function. Returns best sequence found within time budget."""
    t_start = time.monotonic()
    deadline = t_start + float(budget_s) - 20.0
    rng = np.random.default_rng(seed)

    # Load parent
    parent = None
    try:
        h = globals().get("height_sequence_1", None)
        if h is not None:
            p = np.asarray(h, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=1e6, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            if p.size >= 64 and p.sum() > 0.01:
                parent = p
    except Exception:
        pass

    print("Parent: {} (len={})".format(
        "available" if parent is not None else "none",
        parent.size if parent is not None else 0))

    # Determine lengths to try
    if parent is not None:
        p_len = int(parent.size)
        # Round to reasonable length
        if p_len > 10000:
            length = 10000
        elif p_len < 2048:
            length = 4096
        else:
            length = p_len
    else:
        length = 8192

    # Generate initializations
    inits = _generate_inits(length, rng, parent)

    # Score all inits
    scored = []
    for arr in inits:
        a = _normalize(arr)
        sc = _score(a)
        if np.isfinite(sc):
            scored.append((sc, a))

    scored.sort(key=lambda x: x[0])
    print("Best init score: {:.8f} (from {} inits)".format(scored[0][0], len(scored)))

    # Select top diverse starts
    starts = []
    used = set()
    for sc, a in scored:
        key = tuple(np.round(a[:32] / a.sum(), 2))
        if key not in used:
            starts.append(a.copy())
            used.add(key)
        if len(starts) >= 4:
            break

    if not starts:
        starts.append(np.ones(length) / length)

    # Global best
    best_seq = starts[0].copy()
    best_score = _score(best_seq)

    print("Starting optimization with {} starts, length={}".format(len(starts), length))
    t_opt_start = time.monotonic()

    # Main optimization: allocate time to each start
    n_starts = len(starts)
    main_end = deadline - 180.0  # Reserve 180s for final polish
    available = max(60.0, main_end - t_opt_start)

    # Allocate more time to the best start
    if n_starts == 1:
        fracs = [1.0]
    elif n_starts == 2:
        fracs = [0.65, 0.35]
    elif n_starts == 3:
        fracs = [0.55, 0.30, 0.15]
    else:
        fracs = [0.50, 0.25, 0.15, 0.10]

    cum = 0.0
    for i, st in enumerate(starts):
        cum += fracs[i] if i < len(fracs) else 0.1
        sub_dead = t_opt_start + cum * available
        sub_dead = min(sub_dead, main_end - 5.0)

        if time.monotonic() > sub_dead:
            continue

        # For starts after the first, use the current best as base
        if i > 0 and np.isfinite(best_score) and best_score < 2.0:
            base = best_seq
            if len(base) != len(st):
                base = _resample(base, len(st))
            base = _normalize(base)
            st = _perturb(base, rng)

        sc, seq = _optimize_sequence(st, sub_dead, rng, "s{}".format(i))
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("New global best: {:.8f} (start {})".format(best_score, i))

    best_seq = _normalize(best_seq)
    print("After main optimization: {:.8f} (t={:.0f}s)".format(
        _score(best_seq), time.monotonic() - t_start))

    # Final polish phase
    rem = float(deadline - time.monotonic())

    # 1. Flatten autocorrelation
    if rem > 30:
        dl = time.monotonic() + min(rem * 0.3, 80.0)
        sc, seq = _flatten_autocorr(best_seq, dl, max_rounds=50)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Flatten polish: {:.8f}".format(best_score))

    # 2. Exact subgradient
    rem = float(deadline - time.monotonic())
    if rem > 20:
        dl = time.monotonic() + min(rem * 0.3, 50.0)
        sc, seq = _exact_subgrad_polish(best_seq, dl, max_rounds=40)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Subgrad polish: {:.8f}".format(best_score))

    # 3. Pressure update
    rem = float(deadline - time.monotonic())
    if rem > 20:
        dl = time.monotonic() + min(rem * 0.3, 50.0)
        sc, seq = _pressure_update(best_seq, dl, max_rounds=40)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Pressure polish: {:.8f}".format(best_score))

    # 4. One more L-BFGS at very high beta
    rem = float(deadline - time.monotonic())
    if rem > 15:
        mf = max(100, int(rem * 0.7 / 0.05))
        mf = min(mf, 500)
        sc, seq = _lbfgs_run(best_seq, 15000.0, time.monotonic() + rem * 0.6, mf)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Final L-BFGS: {:.8f}".format(best_score))

    # 5. Final exact subgradient
    rem = float(deadline - time.monotonic())
    if rem > 10:
        sc, seq = _exact_subgrad_polish(best_seq, time.monotonic() + rem * 0.7, max_rounds=20)
        if sc < best_score:
            best_score = sc
            best_seq = seq.copy()
            print("Final subgrad: {:.8f}".format(best_score))

    # Final output preparation
    out_arr = np.asarray(best_seq, dtype=np.float64).ravel()
    out_arr = np.nan_to_num(out_arr, nan=0.0, posinf=1000.0, neginf=0.0)
    out_arr = np.clip(out_arr, 0.0, 1000.0)
    S = float(out_arr.sum())
    if S < 0.01:
        out_arr = np.ones(1024) * 0.5
    else:
        out_arr = out_arr / (S / float(out_arr.size))  # normalize to sum = n (so mean = 1)
        # Actually normalize so that the score is computed correctly
        # The score is scale-invariant, so any normalization works
        out_arr = out_arr / float(out_arr.sum())  # sum = 1
        out_arr = np.clip(out_arr, 0.0, 1000.0)

    out = [float(x) for x in out_arr]
    if sum(out) < 0.01:
        out = [0.5] * 64

    # Verify
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = best_score

    print("FINAL: internal={:.8f} verified={:.8f} len={} time={:.1f}s".format(
        best_score, verified, len(out), time.monotonic() - t_start))
    return out
