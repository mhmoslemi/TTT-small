"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Strategy: LSE-smoothed L-BFGS with adaptive beta, exact subgradient polish,
and multi-restart with asymmetric perturbations to escape symmetry traps.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    # Compute LSE
    x = beta * (c - c_max)
    x_max = x.max()
    x_shifted = x - x_max
    exp_x = np.exp(x_shifted)
    log_sum_exp = x_max + np.log(exp_x.sum())
    L = log_sum_exp / beta
    score = 2.0 * n * L / (S * S)
    
    # Softmax weights
    p = exp_x / exp_x.sum()
    
    # Gradient: dL/da[i] = 2 * sum_k p[k] * a[k-i]
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    
    # Full gradient of score = 2n * L / S^2
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=500000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    def obj(x):
        s, g = lse_obj_grad(x, beta)
        return s, g
    
    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-20, 'gtol': 1e-16,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def exact_subgradient(a, n_iters=10000, verbose=False):
    """
    Exact projected subgradient with Polyak step and optional momentum.
    Works on normalized sequence (sum = 1).
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    h_lb = 2.0 * n / (2.0 * n - 1.0)  # lower bound for 2n*max_conv when sum=1
    # Actually, the lower bound of 2n*max_conv/(sum)^2 is 2n/(2n-1)
    # When sum=1, the lower bound of 2n*max_conv is 2n/(2n-1)
    
    best_a = a.copy()
    best_h = 2.0 * n * float(fast_conv(a).max())
    
    momentum = 0.0
    v = np.zeros(n)
    
    for it in range(n_iters):
        c = fast_conv(a)
        h_val = 2.0 * n * float(c.max())
        
        if h_val < best_h:
            best_h = h_val
            best_a = a.copy()
        
        gap = h_val - h_lb
        if gap < 1e-10:
            break
        
        k_star = int(np.argmax(c))
        
        # Subgradient of h(a) = 2n*max_k c[k] at peak k*:
        # s[i] = 4n * a[k*-i] for valid i
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        s = np.zeros(n)
        if j_end >= j_start:
            s[j_start:j_end+1] = a[j_end:j_start-1:-1] if j_end > j_start else a[j_start:j_end+1][::-1]
        
        # Project to tangent space of simplex (sum=1): subtract mean
        s = s - s.mean()
        
        s_norm_sq = float(s @ s)
        if s_norm_sq < 1e-30:
            # Stuck (symmetric point), try small perturbation
            a = a + 1e-6 * (np.random.randn(n) - 1.0/n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            continue
        
        # Polyak step
        step = gap / s_norm_sq
        step = min(step, 1.0 / (np.sqrt(s_norm_sq) + 1e-15))
        step = min(step, 0.1)  # cap
        
        # Momentum
        if momentum > 0:
            v = momentum * v + (1 - momentum) * s
            direction = v
            dir_norm_sq = float(direction @ direction)
            if dir_norm_sq > 1e-30:
                step_m = gap / dir_norm_sq
                step_m = min(step_m, 1.0 / (np.sqrt(dir_norm_sq) + 1e-15))
                step_m = min(step_m, 0.1)
                a_new = a - step_m * direction
            else:
                a_new = a - step * s
        else:
            a_new = a - step * s
        
        # Project to simplex
        a_new = np.maximum(a_new, 0.0)
        a_new_sum = a_new.sum()
        if a_new_sum < 1e-12:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / a_new_sum
        
        # Check if improved
        c_new = fast_conv(a_new)
        h_new = 2.0 * n * float(c_new.max())
        
        if h_new <= h_val:
            a = a_new
        else:
            # Try half step
            a_half = a - 0.5 * step * s
            a_half = np.maximum(a_half, 0.0)
            a_half_sum = a_half.sum()
            if a_half_sum > 1e-12:
                a_half = a_half / a_half_sum
                c_half = fast_conv(a_half)
                h_half = 2.0 * n * float(c_half.max())
                if h_half <= h_val:
                    a = a_half
                else:
                    # No improvement, try tiny step
                    a_tiny = a - 0.01 * step * s
                    a_tiny = np.maximum(a_tiny, 0.0)
                    a_tiny_sum = a_tiny.sum()
                    if a_tiny_sum > 1e-12:
                        a_tiny = a_tiny / a_tiny_sum
                        c_tiny = fast_conv(a_tiny)
                        h_tiny = 2.0 * n * float(c_tiny.max())
                        if h_tiny < h_val:
                            a = a_tiny
    
    return best_a
