LP-based alternating optimizer for flat-autocorrelation non-negative sequences.
Uses scipy HiGHS LP solver for exact LP subproblem solutions, combined with
LSE-smoothed L-BFGS for final refinement.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
from scipy.signal import fftconvolve
import time


def self_conv(a):
    """Full linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv(a, b):
    """Full linear convolution via FFT."""
    n = len(a) + len(b) - 1
    M_fft = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:n]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def build_conv_matrix(a):
    """Build sparse matrix A such that A @ x = conv(a, x)."""
    n = len(a)
    # A[k, j] = a[k-j] for 0 <= k-j < n
    # Column j: rows j to j+n-1, values a[0] to a[n-1]
    indices = np.arange(n, dtype=np.int32)
    rows = np.tile(indices, n) + np.repeat(indices, n)
    cols = np.repeat(indices, n)
    vals = np.tile(a, n)
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A


def solve_lp(a, M, n):
    """
    Solve: max sum(x) s.t. conv(a,x) <= M, x >= 0
    Returns x (unnormalized) or None if infeasible.
    """

    A = build_conv_matrix(a)
    b_ub = M * np.ones(2*n - 1)
    c_obj = -np.ones(n)  # minimize -sum(x)
    bounds = [(0, None)] * n

                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    
    if res.status == 0:
        return np.maximum(res.x, 0)
    else:
        return None
