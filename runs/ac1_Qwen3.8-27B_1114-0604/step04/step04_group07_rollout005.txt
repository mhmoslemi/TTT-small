"""
Optimized minimizer for 2n*max(self_conv)/(sum)^2 over non-negative sequences.
Uses LSE-smoothed Adam (non-palindromic), Frank-Wolfe, multi-scale optimization,
and diverse initializations. Focuses on asymmetric sequences to beat the palindrome
lower bound of 2.

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and gradient w.r.t. a.
    score = 2n * LSE_beta(a*a) / (sum a)^2

    S = float(a.sum())
    if S < 1e-12:

    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    log_sum = np.log(exp_sum)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_lse / S2 - 2.0 * L / S3)
    return score, grad


def subgrad_exactly(a):
    """Exact score and a subgradient (at the argmax lag)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    k_star = int(c.argmax())
    score = 2.0 * n * M / (S * S)
    # Subgradient: 2n * (s / S^2 - 2M / S^3) where s[j] = a[k*-j] + a[j-k*]
    s = np.zeros(n)
    js = max(0, k_star - (n - 1))
    je = min(k_star, n - 1)
    if je >= js:
        for j in range(js, je + 1):
            val = 0.0
            if 0 <= k_star - j <= n - 1:
                val += a[k_star - j]
            if 0 <= j - k_star <= n - 1 and j != k_star - j:
                val += a[j - k_star]
            s[j] = val
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (s / S2 - 2.0 * M / S3)
    return score, grad


def adam_lse(a_init, beta_schedule, lr, n_steps, time_limit_s, t_start,
             normalize=True, grad_clip=10.0):
    """Adam with LSE smoothing and scheduled beta."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    n_beta = len(beta_schedule)
    steps_per_beta = max(1, n_steps // n_beta)

    for bi, beta in enumerate(beta_schedule):
        steps = min(steps_per_beta, n_steps - bi * steps_per_beta)
        for t in range(steps):
            if time.monotonic() - t_start > time_limit_s:
                break
            score, grad = lse_score_grad(a, beta)
            if not np.isfinite(score):
                break

            # Gradient clipping
            gnorm = np.linalg.norm(grad)
            if gnorm > grad_clip:
                grad = grad * (grad_clip / gnorm)

            lr_t = lr / (1.0 + 1e-5 * t)
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            t_global = bi * steps_per_beta + t + 1
            m_hat = m / (1.0 - b1 ** t_global)
            v_hat = v / (1.0 - b2 ** t_global)
            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, 1e10)

            if normalize:
                s = a.sum()
                if s > 1e-12:
                    a = a / s

            if t % 2000 == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        if time.monotonic() - t_start > time_limit_s:
            break

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def frank_wolfe(a_init, n_iters, time_limit_s, t_start, decreasing_step=True):
    """Frank-Wolfe (conditional gradient) on the simplex with LSE smoothing."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    a = a / a.sum()

    best_a = a.copy()
    best_score = compute_score(a)

    beta = 50.0

    for it in range(n_iters):
        if time.monotonic() - t_start > time_limit_s:
            break

        score, grad = lse_score_grad(a, beta)
        if not np.isfinite(score):
            grad, _ = subgrad_exactly(a)

        # Linear minimization over simplex: pick index of min gradient
        j_min = int(np.argmin(grad))
        e_min = np.zeros(n)
        e_min[j_min] = 1.0

        # Step size
        if decreasing_step:
            t_step = 2.0 / (it + 2)
        else:
            t_step = 0.1

        a_new = (1.0 - t_step) * a + t_step * e_min
        a_new = np.clip(a_new, 0.0, None)
        a_new = a_new / a_new.sum()

        sc_new = compute_score(a_new)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()

        a = a_new

        # Gradually increase beta
        if it % 500 == 0:
            beta = min(beta * 1.5, 5000.0)

    return best_score, best_a


def lbfgs_lse(a0, beta, maxiter=3000, maxfun=50000, time_limit_s=60.0, t_start=None):
    """L-BFGS-B with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n
    deadline = (t_start + time_limit_s) if t_start is not None else np.inf

    call_count = [0]

    def obj(x):
        call_count[0] += 1
        if time.monotonic() > deadline:
            x_c = np.clip(x, 0.0, 1e10)
            sc, gr = lse_score_grad(x_c, beta)
            return sc, gr
        x_c = np.clip(x, 0.0, 1e10)
        sc, gr = lse_score_grad(x_c, beta)
        return sc, gr

    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-14, 'gtol': 1e-11,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)


def generate_initializations(rng, n):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    # Uniform
    inits.append(np.ones(n) / n)

    # Linear ramp (increasing)
    r = np.linspace(0.1, 1.0, n)
    inits.append(r / r.sum())

    # Linear ramp (decreasing)
    r = np.linspace(1.0, 0.1, n)
    inits.append(r / r.sum())

    # Exponential decay
    e = np.exp(-x * rng.uniform(1, 4))
    inits.append(e / e.sum())

    # Reverse exponential
    e = np.exp(-(1 - x) * rng.uniform(1, 4))
    inits.append(e / e.sum())

    # Two-bump asymmetric
    c1 = 0.2 + 0.2 * rng.random()
    c2 = 0.6 + 0.2 * rng.random()
    s1 = 0.08 + 0.1 * rng.random()
    s2 = 0.08 + 0.1 * rng.random()
    b = np.exp(-0.5 * ((x - c1) / s1) ** 2) + np.exp(-0.5 * ((x - c2) / s2) ** 2)
    inits.append(b / b.sum())

    # Sinusoidal modulation (asymmetric)
    freq = 2 + 6 * rng.random()
    phase = 2 * np.pi * rng.random()
    s = 1.0 + 0.4 * np.cos(2 * np.pi * freq * x + phase)
    s = np.maximum(s, 0.01)
    inits.append(s / s.sum())

    # "Split" structure: high front, low back with smooth transition
    t = np.linspace(0, 1, n)
    split = 1.0 / (1.0 + np.exp(rng.uniform(2, 6) * (t - 0.5)))
    inits.append(split / split.sum())

    # Random smooth
    r = np.abs(rng.standard_normal(n)) + 0.3
    k = max(3, n // 100)
    kernel = np.ones(k) / k
    r_smooth = np.convolve(r, kernel, mode='same')
    inits.append(r_smooth / r_smooth.sum())

    # Cosine window (asymmetric by shifting)
    shift = 0.1 + 0.2 * rng.random()
    cw = 0.5 * (1 + np.cos(2 * np.pi * (x - shift)))
    cw = np.maximum(cw, 0.001)
    inits.append(cw / cw.sum())

    # Power law
    p = rng.uniform(0.3, 2.0)
    pl = (x + 0.01) ** p
    inits.append(pl / pl.sum())

    # Power law reversed
    pl = (1 - x + 0.01) ** p
    inits.append(pl / pl.sum())

    return [np.clip(x, 1e-10, 1000.0) for x in inits]


def coordinate_descent_subgrad(a, n_sweeps=50, time_limit_s=30.0, t_start=None):
    """
    Coordinate descent using exact subgradient information.
    At each step, identify the peak lag and try to reduce contributing elements.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    a = a / a.sum()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = (t_start + time_limit_s) if t_start is not None else np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        k_star = int(c.argmax())
        score = compute_score(a)

        # Elements contributing to the peak: a[j] for j in [k_star-n+1, k_star]
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)

        # Try reducing each contributing element
        for j in np.random.permutation(np.arange(js, je + 1)):
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            if a_old < 1e-10:
                continue
            # Try reducing
            for frac in [0.5, 0.3, 0.1]:
                a[j] = a_old * frac
                S = a.sum()
                if S < 1e-10:
                    a[j] = a_old
                    break
                sc = compute_score(a)
                if sc < score:
                    # Redistribute removed mass to non-contributing elements
                    removed = a_old - a[j]
                    non_contrib = np.ones(n, dtype=bool)
                    non_contrib[js:je + 1] = False
                    nc_count = non_contrib.sum()
                    if nc_count > 0:
                        # Distribute to elements farthest from peak
                        a[non_contrib] += removed / nc_count
                    score = sc
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                    break
                else:
                    a[j] = a_old
                    break

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400):
    """
    Multi-phase optimizer:
    Phase 1: Diverse initializations + quick screening
    Phase 2: LSE Adam on best candidates (non-palindromic)
    Phase 3: Multi-scale optimization (grow n)
    Phase 4: L-BFGS polishing
    Phase 5: Frank-Wolfe + coordinate descent
    Phase 6: Final intensive Adam
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(60.0, float(budget_s)) - 30.0

    best_score = np.inf
    best_a = None

    def try_update(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score and np.isfinite(sc):
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [NEW BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # Try to load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                ps = compute_score(parent)
                print(f"Parent loaded: n={parent.size}, score={ps:.8f}")
                try_update(ps, parent, "parent")
    except Exception:
        pass

    # ==========================================
    # Phase 1: Diverse initializations + screening
    # ==========================================
    p1_time = effective_budget * 0.12
    print(f"\n{'=' * 60}")
    print(f"Phase 1: Diverse init + screening (budget={p1_time:.0f}s)")
    print(f"{'=' * 60}")

    n_screen = 1024  # Start moderate for screening
    inits = generate_initializations(rng, n_screen)
    if parent is not None:
        inits.append(resample_sequence(parent, n_screen))

    candidates = []
    for i, init in enumerate(inits):
        if time.monotonic() - t_start > t_start + p1_time - 10:
            break
        beta_sched = [20, 80]
        sc, a_opt = adam_lse(init, beta_sched, lr=0.02, n_steps=3000,
                             time_limit_s=min(4.0, p1_time / (len(inits) - i)),
                             t_start=time.monotonic() - 0.1)
        candidates.append((sc, a_opt))
        print(f"  Init {i}: score={sc:.8f}")
        try_update(sc, a_opt, f"screen init {i}")

    candidates.sort(key=lambda x: x[0])
    print(f"  Best after screening: {best_score:.8f}")

    # Also try parent at its native length
    if parent is not None and len(parent) >= 2048:
        remaining = t_start + p1_time - time.monotonic()
        if remaining > 5:
            sc, a_opt = adam_lse(parent, [50], lr=0.01, n_steps=2000,
                                  time_limit_s=min(remaining, 15.0),
                                  t_start=t_start)
            try_update(sc, a_opt, "parent quick adam")

    # ==========================================
    # Phase 2: LSE Adam on best (non-palindromic)
    # ==========================================
    p2_time = effective_budget * 0.30
    print(f"\n{'=' * 60}")
    print(f"Phase 2: LSE Adam non-palindromic (budget={p2_time:.0f}s)")
    print(f"{'=' * 60}")

    # Work at the best candidate's length
    n_work = len(best_a)

    # Beta continuation: start low, increase
    beta_schedules = [
        ([10, 30, 80], [0.02, 0.015, 0.01]),
        ([80, 200, 500], [0.01, 0.008, 0.005]),
        ([500, 1500, 4000], [0.005, 0.003, 0.002]),
        ([4000, 10000], [0.002, 0.001]),
    ]

    t_per_config = p2_time / len(beta_schedules)
    for bi, (betas, lrs) in enumerate(beta_schedules):
        remaining = t_start + p2_time - time.monotonic()
        if remaining < 3:
            break
        t_alloc = min(t_per_config, remaining)
        sc, a_opt = adam_lse(best_a, betas, lr=lrs[0] if len(lrs) == 1 else lrs[min(bi, len(lrs)-1)],
                             n_steps=200000, time_limit_s=t_alloc, t_start=t_start)
        try_update(sc, a_opt, f"Adam pal-free beta={betas}")

    # Try from second-best candidate too
    if len(candidates) > 1:
        second_best = candidates[1][1]
        # Resample to same length
        if len(second_best) != n_work:
            second_best = resample_sequence(second_best, n_work)
        remaining = t_start + p2_time - time.monotonic()
        if remaining > 10:
            sc, a_opt = adam_lse(second_best, [100, 500], lr=0.008,
                                  n_steps=100000, time_limit_s=min(remaining, p2_time * 0.15),
                                  t_start=t_start)
            try_update(sc, a_opt, "Adam from 2nd best")

    # ==========================================
    # Phase 3: Multi-scale optimization
    # ==========================================
    p3_time = effective_budget * 0.20
    print(f"\n{'=' * 60}")
    print(f"Phase 3: Multi-scale (budget={p3_time:.0f}s)")
    print(f"{'=' * 60}")

    # Try growing the sequence
    target_ns = [2048, 4096, 8192]
    if len(best_a) > 4096:
        target_ns = [len(best_a), len(best_a) * 2]
        if target_ns[-1] > 16384:
            target_ns = [len(best_a)]

    for n_target in target_ns:
        remaining = t_start + p3_time - time.monotonic()
        if remaining < 10 or n_target > 16384:
            break
        if n_target <= len(best_a):
            continue

        print(f"  Upsampling to n={n_target}...")
        a_up = resample_sequence(best_a, n_target)

        # Optimize at new scale
        t_alloc = min(p3_time * 0.3, remaining)
        sc, a_opt = adam_lse(a_up, [50, 200, 800], lr=0.01,
                             n_steps=200000, time_limit_s=t_alloc, t_start=t_start)
        try_update(sc, a_opt, f"multi-scale n={n_target}")

    # ==========================================
    # Phase 4: L-BFGS polishing
    # ==========================================
    p4_time = effective_budget * 0.15
    print(f"\n{'=' * 60}")
    print(f"Phase 4: L-BFGS polishing (budget={p4_time:.0f}s)")
    print(f"{'=' * 60}")

    n_cur = len(best_a)
    # For very large n, L-BFGS may be too slow; limit to n <= 8192
    if n_cur <= 8192:
        for beta in [500, 2000, 8000]:
            remaining = t_start + p4_time - time.monotonic()
            if remaining < 5:
                break
            t0 = time.monotonic()
            sc, a_opt = lbfgs_lse(best_a, beta, maxiter=3000, maxfun=40000,
                                   time_limit_s=min(remaining * 0.7, 60.0),
                                   t_start=t_start)
            try_update(sc, a_opt, f"LBFGS beta={beta}")
            print(f"  LBFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
    else:
        print(f"  Skipping L-BFGS (n={n_cur} too large), using Adam instead")
        remaining = t_start + p4_time - time.monotonic()
        if remaining > 5:
            sc, a_opt = adam_lse(best_a, [2000, 8000], lr=0.001,
                                  n_steps=100000, time_limit_s=remaining,
                                  t_start=t_start)
            try_update(sc, a_opt, "Adam high-beta (large n)")

    # ==========================================
    # Phase 5: Frank-Wolfe + Coordinate Descent
    # ==========================================
    p5_time = effective_budget * 0.13
    print(f"\n{'=' * 60}")
    print(f"Phase 5: Frank-Wolfe + Coord Descent (budget={p5_time:.0f}s)")
    print(f"{'=' * 60}")

    # Frank-Wolfe
    remaining = t_start + p5_time - time.monotonic()
    if remaining > 5:
        t0 = time.monotonic()
        sc, a_opt = frank_wolfe(best_a, n_iters=20000,
                                 time_limit_s=min(remaining * 0.5, 30.0),
                                 t_start=t_start)
        try_update(sc, a_opt, "Frank-Wolfe")
        print(f"  FW: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Coordinate descent
    remaining = t_start + p5_time - time.monotonic()
    if remaining > 3:
        t0 = time.monotonic()
        sc, a_opt = coordinate_descent_subgrad(best_a, n_sweeps=30,
                                                time_limit_s=min(remaining, 20.0),
                                                t_start=t_start)
        try_update(sc, a_opt, "Coord descent")
        print(f"  CD: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==========================================
    # Phase 6: Diverse restarts
    # ==========================================
    p6_time = effective_budget * 0.12
    print(f"\n{'=' * 60}")
    print(f"Phase 6: Diverse restarts (budget={p6_time:.0f}s)")
    print(f"{'=' * 60}")

    n_restarts = 6
    restart_budget = p6_time / n_restarts
    n_cur = len(best_a)

    for r in range(n_restarts):
        if time.monotonic() > t_start + effective_budget - 30:
            break

        strategy = r % 3
        if strategy == 0:
            # Log-space Gaussian perturbation
            log_a = np.log(np.clip(best_a, 1e-12, 1e10))
            noise_scale = 0.03 + 0.04 * rng.random()
            log_pert = log_a + noise_scale * rng.standard_normal(n_cur)
            a_pert = np.exp(log_pert - np.max(log_pert))
            a_pert = a_pert / a_pert.sum()
        elif strategy == 1:
            # Peak-avoidance perturbation
            c = fast_conv(best_a)
            k_star = int(c.argmax())
            js = max(0, k_star - n_cur + 1)
            je = min(k_star, n_cur - 1)
            a_pert = best_a.copy()
            a_pert[js:je + 1] *= (0.4 + 0.3 * rng.random())
            non_peak = np.ones(n_cur, dtype=bool)
            non_peak[js:je + 1] = False
            a_pert[non_peak] *= (1.0 + 0.2 * rng.random())
            a_pert = a_pert / a_pert.sum()
        else:
            # Block modification
            n_blocks = 6 + int(6 * rng.random())
            a_pert = best_a.copy()
            block_size = n_cur // n_blocks
            for b in range(n_blocks):
                s = b * block_size
                e = min(s + block_size, n_cur)
                a_pert[s:e] *= rng.uniform(0.3, 2.0)
            a_pert = a_pert / a_pert.sum()

        # Quick refinement
        time_left = t_start + effective_budget - 30 - time.monotonic()
        if time_left < 3:
            break
        adam_t = min(restart_budget * 0.7, time_left)
        sc, a_r = adam_lse(a_pert, [50, 200], lr=0.01,
                            n_steps=100000, time_limit_s=adam_t, t_start=t_start)
        try_update(sc, a_r, f"restart#{r} s{strategy}")

    # ==========================================
    # Phase 7: Final intensive polish
    # ==========================================
    p7_time = effective_budget * 0.08
    print(f"\n{'=' * 60}")
    print(f"Phase 7: Final polish (budget={p7_time:.0f}s)")
    print(f"{'=' * 60}")

    final_deadline = t_start + effective_budget - 5.0

    # High-beta Adam
    if time.monotonic() < final_deadline:
        t0 = time.monotonic()
        remaining = final_deadline - time.monotonic()
        beta_final = [5000] if len(best_a) > 4096 else [5000, 20000]
        sc, a_opt = adam_lse(best_a, beta_final, lr=0.0005,
                              n_steps=200000, time_limit_s=min(remaining * 0.6, p7_time),
                              t_start=t_start)
        try_update(sc, a_opt, "final Adam high-beta")
        print(f"  Final Adam: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # One more L-BFGS if n allows
    if len(best_a) <= 4096 and time.monotonic() < final_deadline - 5:
        t0 = time.monotonic()
        remaining = final_deadline - time.monotonic() - 2
        if remaining > 5:
            sc, a_opt = lbfgs_lse(best_a, 10000, maxiter=5000, maxfun=80000,
                                   time_limit_s=min(remaining, 30.0), t_start=t_start)
            try_update(sc, a_opt, "final LBFGS")
            print(f"  Final LBFGS: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Final coordinate descent
    if time.monotonic() < final_deadline - 2:
        sc, a_opt = coordinate_descent_subgrad(best_a, n_sweeps=10,
                                                time_limit_s=min(5.0, final_deadline - time.monotonic() - 1),
                                                t_start=t_start)
        try_update(sc, a_opt, "final coord")

    # ==========================================
    # Final verification and output
    # ==========================================
    if best_a is None:
        best_a = np.ones(64) / 64

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
        best_score = 2.0

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    return out
