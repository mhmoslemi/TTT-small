"""
Symmetric sequence optimizer using multiplicative peak shaving, 
spectral concentration, L-BFGS, and simulated annealing.
Exploits the symmetry of the optimal solution (a[j] = a[n-1-j])
to halve the search space and focus optimization on the center peak.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def symmetrize(a):
    """Enforce symmetry: a[j] = a[n-1-j]."""
    n = len(a)
    half = (n + 1) // 2
    a_sym = np.zeros(n)
    a_sym[:half] = a[:half]
    a_sym[half:] = a[:n - half][::-1]
    return np.maximum(a_sym, 0.0)


def lse_objective_sym(x, beta, n):
    """LSE-smoothed objective for symmetric sequence. x is the half-sequence."""
    a = symmetrize_from_half(x, n)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(len(x))
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient w.r.t. x (the half-sequence)
    p = exp_scaled / exp_scaled.sum()
    # dc/da for each peak position
    rev_a = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    g_full = 2.0 * conv_res
    
    S2 = S * S
    S3 = S2 * S
    grad_full = (2.0 * n) * (g_full / S2 - 2.0 * L / S3)
    
    # Map gradient to half-sequence
    half = (n + 1) // 2
    grad_half = grad_full[:half].copy()
    # For the symmetric part, we need to account for the mirroring
    # a[j] = x[j] for j < half, a[n-1-j] = x[j]
    # da[j]/dx[i] = 1 if j == i (j < half)
    # da[n-1-j]/dx[i] = 1 if n-1-j == i, i.e., j = n-1-i
    # So grad_half[i] = grad_full[i] + grad_full[n-1-i] (for i < half, i != n-1-i)
    half_len = len(x)
    for i in range(half_len):
        j_mirror = n - 1 - i
        if j_mirror != i and j_mirror < n:
            grad_half[i] += grad_full[j_mirror]
    
    return score, grad_half


def symmetrize_from_half(x, n):
    """Create symmetric sequence from half-sequence."""
    a = np.zeros(n)
    half = (n + 1) // 2
    a[:half] = x
    a[half:] = x[:n - half][::-1]
    return np.maximum(a, 0.0)


def peak_shaving_sym(a, n_iters=3000, top_k=3):
    """Multiplicative peak shaving on symmetric sequence."""
    n = len(a)
    half = (n + 1) // 2
    a_half = np.maximum(a[:half].astype(np.float64), 1e-10)
    
    best_a = symmetrize_from_half(a_half, n)
    best_sc = compute_score(best_a)
    
    lr = 0.03
    M_fft = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        a = symmetrize_from_half(a_half, n)
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2*n-1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient at top peaks
        peak_indices = np.argsort(c)[::-1][:top_k]
        sg = np.zeros(n)
        for k in peak_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            sg[j_start:j_end+1] += 2.0 * a[j_end:j_start-1:-1]
        
        # Map to half
        sg_half = np.zeros(half)
        for i in range(half):
            sg_half[i] = sg[i]
            j_mirror = n - 1 - i
            if j_mirror != i:
                sg_half[i] += sg[j_mirror]
        
        sg_half -= sg_half.mean()
        
        # Multiplicative update
        scale = max(M_val, 1e-15)
        log_update = -lr * sg_half / scale
        log_update = np.clip(log_update, -1.0, 1.0)
        a_half = a_half * np.exp(log_update)
        a_half = np.maximum(a_half, 1e-10)
        a_half = a_half / a_half.sum()
        
        lr *= 0.9999
        
    return best_a


def spectral_concentrate(a, n_iters=500, low_pass_frac=0.05):
    """Concentrate power in low frequencies (zero-padded domain)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    a = a / S
    
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        
        # Reduce high-frequency components
        n_bins = len(A)
        cutoff = max(2, int(n_bins * low_pass_frac))
        
        # Smoothly attenuate above cutoff
        for m in range(cutoff, n_bins):
            fade = (m - cutoff) / (n_bins - cutoff)
            attenuation = 1.0 - 0.5 * fade
            A[m] *= np.sqrt(attenuation)
        
        a_new = np.fft.irfft(A, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def simulated_annealing(a, n_iters=10000, t_init=0.1, t_final=0.001):
    """Simulated annealing with structured perturbations on symmetric sequence."""
    n = len(a)
    half = (n + 1) // 2
    a_half = np.maximum(a[:half].astype(np.float64), 1e-10)
    a_half = a_half / a_half.sum()
    
    best_a = symmetrize_from_half(a_half, n)
    best_sc = compute_score(best_a)
    curr_sc = best_sc
    
    t = t_init
    cooling = (t_final / t_init) ** (1.0 / n_iters)
    rng = np.random.default_rng(12345)
    
    for it in range(n_iters):
        # Random perturbation
        a_trial = a_half.copy()
        pert_type = rng.integers(0, 4)
        
        if pert_type == 0:
            # Random rescale of a random block
            i = rng.integers(0, half)
            width = rng.integers(1, max(2, half // 4))
            j_end = min(i + width, half)
            factor = 1.0 + 0.3 * rng.standard_normal()
            a_trial[i:j_end] *= max(factor, 0.1)
        elif pert_type == 1:
            # Swap mass between two random positions
            i1 = rng.integers(0, half)
            i2 = rng.integers(0, half)
            delta = 0.01 * a_half.sum()
            a_trial[i1] = max(a_trial[i1] - delta, 1e-10)
            a_trial[i2] += delta
        elif pert_type == 2:
            # Gaussian perturbation
            a_trial += 0.01 * a_half.sum() * rng.standard_normal(half)
            a_trial = np.maximum(a_trial, 1e-10)
        else:
            # Smooth perturbation (add a low-freq mode)
            freq = rng.integers(1, 5)
            phase = rng.uniform(0, 2 * np.pi)
            x = np.arange(half) / half
            pert = 0.02 * a_half.sum() * np.cos(2 * np.pi * freq * x + phase)
            a_trial += pert
            a_trial = np.maximum(a_trial, 1e-10)
        
        a_trial = a_trial / a_trial.sum()
        a_trial_full = symmetrize_from_half(a_trial, n)
        trial_sc = compute_score(a_trial_full)
        
        # Accept/reject
        if trial_sc < curr_sc or rng.random() < np.exp(-(trial_sc - curr_sc) / t):
            a_half = a_trial
            curr_sc = trial_sc
            if curr_sc < best_sc:
                best_sc = curr_sc
                best_a = a_trial_full.copy()
        
        t *= cooling
        
    return best_a


def run_lbfgs_sym(a, beta, maxiter=5000, maxfun=100000):
    """L-BFGS on symmetric sequence."""
    n = len(a)
    half = (n + 1) // 2
    x0 = np.maximum(a[:half].astype(np.float64), 1e-10)
    x0 = x0 / x0.sum()
    
    bounds = [(1e-10, 1000.0)] * half
    res = minimize(

```python
"""
Symmetric peak-shaving optimizer with L-BFGS and spectral methods.
Exploits the left-right symmetry of the optimal solution.
Uses multiplicative updates targeting convolution peaks,
LSE-smoothed L-BFGS for local optimization, and simulated
annealing for global exploration.
"""
import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def symmetrize(a):
    n = len(a)
    result = 0.5 * (a + a[::-1])
    return np.maximum(result, 0.0)


def make_symmetric_from_half(x, n):
    a = np.zeros(n)
    half = (n + 1) // 2
    a[:half] = x
    a[half:] = x[:n - half][::-1]
    return np.maximum(a, 0.0)


def lse_grad_full(a, beta):
    """Compute LSE-smoothed score and gradient (full sequence)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    exp_sum = float(exp_scaled.sum())
    log_sum = float(np.log(exp_sum))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_sum
    # Gradient: dL/da[i] = sum_k p[k] * dc[k]/da[i]
    # dc[k]/da[i] = a[k-i] + a[i-k] (with valid indices)
    # Efficient computation via FFT
    rev_a = a[::-1]
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[:M2]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lseObjective(x, beta, n):
    """Top-level objective for L-BFGS (non-symmetric)."""
    a = np.maximum(x, 0.0)
    return lse_grad_full(a, beta)


def lseObjectiveSym(x, beta, n):
    """Top-level objective for L-BFGS (symmetric, x is half)."""
    a = make_symmetric_from_half(x, n)
    score, grad_full = lse_grad_full(a, beta)
    half = len(x)
    grad_half = np.zeros(half)
    for i in range(half):
        grad_half[i] = grad_full[i]
        j_m = n - 1 - i
        if j_m != i and j_m < n:
            grad_half[i] += grad_full[j_m]
    return score, grad_half


def peak_shaving(a, n_iters, top_k_start, top_k_end, lr_start, lr_end, momentum=0.5):
    """
    Multiplicative peak shaving with momentum.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 1e-12).copy()
    a = a / a.sum()
    
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    
    v = np.zeros(n)  # momentum buffer
    
    for it in range(n_iters):
        progress = it / max(1, n_iters)
        top_k = int(top_k_start + (top_k_end - top_k_start) * progress)
        lr = lr_start * (lr_end / lr_start) ** progress
        
        a = symmetrize(a)
        A = np.fft.rfft(a, M_fft)
        c = np.fft.irfft(A * A, M_fft)[:2*n-1]
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient
        peak_idx = np.argsort(c)[::-1][:top_k]
        sg = np.zeros(n)
        for k in peak_idx:
            k = int(k)
            js = max(0, k - n + 1)
            je = min(k, n - 1)
            if je >= js:
                sg[js:je+1] += 2.0 * a[je:js-1:-1]
        
        # Momentum
        v = momentum * v + (1 - momentum) * sg
        v -= v.mean()
        
        # Multiplicative update
        scale = max(float(np.max(np.abs(v))), 1e-15)
        log_upd = -lr * v / scale
        log_upd = np.clip(log_upd, -1.5, 1.5)
        a = a * np.exp(log_upd)
        a = np.maximum(a, 1e-12)
        a = a / a.sum()
    
    return best_a


def lbfgs_optimize(a, beta, maxiter=5000, maxfun=80000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-6:
        a[:] = 1.0
    a = a / a.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lseObjective, a, args=(beta, n),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a.copy()
    sc = compute_score(a_out)
    return sc, a_out


def spectral_flat(a, n_iters=200, alpha=0.3):
    """
    Fienup-style: flatten the zero-padded power spectrum.
    This makes the sequence more 'bandlimited' which can help.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    a = a / a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real**2 + A.imag**2
        P_target = np.full_like(P, P.mean())
        
        # Blend: move P toward flat
        P_new = (1 - alpha) * P + alpha * P_target
        # Preserve phase
        phase = A / (np.abs(A) + 1e-30)
        A_new = phase * np.sqrt(np.maximum(P_new, 0))
        
        a_new = np.fft.irfft(A_new, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a

    Opposite of flattening: boost relative weight of low-freq bins.

    n = len(a)
    a = np.maximum(a.astype
