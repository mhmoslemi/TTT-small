"""
Minimize 2*n*max(self-conv)/(sum)^2 for non-negative sequences.
Strategy: LSE-smoothed Adam with adaptive beta annealing, multiple restarts,
block coordinate descent, L-BFGS polishing, and multi-length search.
Builds on parent sequence (score ~1.508) to push below 1.5027.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a (assumes sum(a)=1)."""
    n = len(a)
    c = fast_conv(a)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    p = exp_scaled / exp_scaled.sum()
    # gradient of LSE wrt a: 2 * (reversed_a * p_reversed)
    # d c[k]/d a[i] = a[k-i] + a[i-k] (but we can use: grad = 2 * conv(a_rev, p)[n-1:2n-1])
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    grad_L = 2.0 * conv_res
    # score = 2n * L (since sum=1, sum^2=1)
    score = 2.0 * n * L
    grad = 2.0 * n * grad_L
    return score, grad


def subgrad_score(a):
    """Exact score and subgradient (assumes sum(a)=1)."""
    n = len(a)
    c = fast_conv(a)
    M = float(c.max())
    k_star = int(c.argmax())
    score = 2.0 * n * M
    sg = np.zeros(n)
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    if je >= js:
        idx = k_star - np.arange(js, je + 1)
        sg[js:je + 1] = 2.0 * a[idx]
    return score, sg


def normalize(a):
    """Normalize to sum=1, ensure non-negative."""
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        return np.ones(len(a)) / len(a)
    return a / s


def lse_adam_anneal(a_init, beta_start=20.0, beta_end=5000.0,
                    n_steps=80000, lr=0.003, time_limit=None, t_start=None,
                    anneal_frac=0.6, momentum_decay=0.99):
    """Adam with LSE and adaptive beta annealing."""
    n = len(a_init)
    a = normalize(a_init.copy())
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    best_a = a.copy()
    best_score = compute_score(a)
    n_anneal = int(n_steps * anneal_frac)

    for t in range(1, n_steps + 1):
        if time_limit is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit:
                break

        # Adaptive beta: exponential annealing
        if t <= n_anneal:
            frac = t / max(1, n_anneal)
            beta = beta_start * (beta_end / beta_start) ** frac
        else:
            beta = beta_end

        # Learning rate decay
        lr_t = lr / (1.0 + 0.0001 * t)

        score, grad = lse_score_grad(a, beta)
        m = momentum_decay * m + (1.0 - momentum_decay) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = normalize(a)

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def lbfgs_polish(a0, beta=2000, maxiter=3000, maxfun=60000):
    """L-BFGS-B polish with LSE smoothing."""
    n = len(a0)
    a0 = normalize(a0.copy())
    bounds = [(0.0, None)] * n

    def obj_and_grad(x):
        x = np.clip(x, 0.0, None)
        s = x.sum()
        if s < 1e-12:
            return 1e10, np.zeros(n)
        xn = x / s
        sc, g = lse_score_grad(xn, beta)
        # Chain rule for the normalization
        # d(xn)/dx: xn_i = x_i / sum(x)
        # d sc/dx_i = g_i/s - sc*sum(g)/s^2  (but we're fixing sum=1 via projection)
        # Actually, since we project after each step in L-BFGS, use gradient directly
        return sc, g

    res = minimize(
        obj_and_grad, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-10,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = normalize(res.x)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def block_coordinate_descent(a, block_size=32, n_rounds=200,
                             time_limit=None, t_start=None):
    """Block coordinate descent: optimize blocks of coordinates via L-BFGS."""
    n = len(a)
    a = normalize(a.copy())
    best_score = compute_score(a)
    deadline = (t_start + time_limit) if (time_limit and t_start) else np.inf

    for rnd in range(n_rounds):
        if time.monotonic() > deadline:
            break
        # Random permutation of blocks
        perm = np.random.permutation(n)
        for start in range(0, n, block_size):
            end = min(start + block_size, n)
            block_idx = perm[start:end]
            # Optimize this block
            a_sub = a[block_idx].copy()

            def block_obj(x):
                a_new = a.copy()
                a_new[block_idx] = np.clip(x, 0.0, None)
                a_new = normalize(a_new)
                sc = compute_score(a_new)
                return sc

            from scipy.optimize import minimize as min2
            res = min2(block_obj, a_sub, method='Nelder-Mead',
                       options={'maxiter': 50, 'xatol': 1e-10, 'fatol': 1e-12})
            a[block_idx] = np.clip(res.x, 0.0, None)
            a = normalize(a)
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc

    return best_score, a


def perturb_and_reoptimize(a, n_perturb=100, scale=0.5,
                           beta_start=50.0, beta_end=2000.0, n_steps=20000):
    """Perturb random coordinates and re-optimize."""
    n = len(a)
    a_pert = a.copy()
    idx = np.random.choice(n, size=min(n_perturb, n), replace=False)
    a_pert[idx] *= (1.0 + scale * np.random.randn(len(idx)))
    a_pert = normalize(a_pert)
    return lse_adam_anneal(a_pert, beta_start=beta_start, beta_end=beta_end,
                           n_steps=n_steps, lr=0.005)


def spectral_init(n, seed=0):
    """Initialize using spectral method: flat-ish spectrum with random phases."""
    rng = np.random.default_rng(seed)
    M_fft = 1 << (2 * n - 2).bit_length()
    # Create spectrum with slowly varying magnitude
    k = np.arange(M_fft // 2 + 1)
    mag = 1.0 / (1.0 + 0.1 * k / n)
    mag += 0.3 * rng.random(M_fft // 2 + 1)
    phase = rng.uniform(0, 2 * np.pi, M_fft // 2 + 1)
    phase[0] = 0
    A = mag * np.exp(1j * phase)
    # Make it Hermitian
    if M_fft % 2 == 0:
        A_full = np.concatenate([A, np.conj(A[-2:0:-1])])
    else:
        A_full = np.concatenate([A, np.conj(A[-1:0:-1])])
    a_complex = np.fft.irfft(A_full * 1.0, M_fft)  # inverse of rfft
    # Actually use ifft
    a_complex = np.fft.ifft(A_full)
    a = np.real(a_complex[:n])
    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-12:
        a = np.ones(n) / n
    return a / a.sum()

        if remaining < 60:
            break
        t0 = time.monotonic()
        sc, a_opt = perturb_and_reoptimize(best_a, n_perturb=200, scale=0.8,
                                           beta_start=50.0, beta_end=3000.0,
                                           n_steps=30000)
        try_update(sc, a_opt, f"restart {i+1}")
        print(f"  Restart {i+1}: {sc:.10f} ({time.monotonic()-t0:.1f}s)")

    # === Phase 4: Spectral initialization attempts ===
    print(f"\nPhase 4: Spectral initializations")
    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 60:
        n_spec = len(best_a)
        for s in range(3):
            remaining = effective_budget - (time.monotonic() - t_start)
            if remaining < 30:
                break
            a_spec = spectral_init(n_spec, seed=seed + 100 + s)
            sc = compute_score(a_spec)
            print(f"  Spectral init {s}: score={sc:.10f}")
            if sc < best_score * 1.1:  # Only optimize if competitive
                t0 = time.monotonic()
                sc2, a_opt = lse_adam_anneal(a_spec, beta_start=20.0, beta_end=2000.0,
                                             n_steps=30000, lr=0.003,
                                             time_limit=min(remaining * 0.3, 100.0),
                                             t_start=t_start)
                try_update(sc2, a_opt, f"spectral opt {s}")

    # === Phase 5: Final L-BFGS polishing ===
    print(f"\nPhase 5: Final L-BFGS polishing")
    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 30:
        for beta in [3000, 5000, 8000]:
            remaining = effective_budget - (time.monotonic() - t_start)
            if remaining < 20:
                break
            t0 = time.monotonic()
            sc, a_opt = lbfgs_polish(best_a, beta=beta, maxiter=5000, maxfun=80000)
            try_update(sc, a_opt, f"final L-BFGS beta={beta}")
            print(f"  L-BFGS beta={beta}: {sc:.10f} ({time.monotonic()-t0:.1f}s)")

    # === Phase 6: Subgradient coordinate descent ===
    print(f"\nPhase 6: Subgradient coordinate descent")
    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 30:
        n = len(best_a)
        a = normalize(best_a.copy())
        best_cd_score = compute_score(a)
        steps_without_improvement = 0

        for step in range(50000):
            remaining = effective_budget - (time.monotonic() - t_start)
            if remaining < 15:
                break
            if (step + 1) % 1000 == 0 and steps_without_improvement > 3000:
                break

            sc, sg = subgrad_score(a)
            if sc < best_cd_score:
                best_cd_score = sc
                steps_without_improvement = 0
            else:
                steps_without_improvement += 1

            # Find best coordinate to update (most negative subgradient = most to gain)
            # We want to decrease the max, so we want to reduce contributions to the peak
            k_star = int(fast_conv(a).argmax())
            # The subgradient points in direction of increase; we want to decrease
            # So we decrease coordinates that contribute to the peak
            js = max(0, k_star - n + 1)
            je = min(k_star, n - 1)
            if je >= js:
                idx = k_star - np.arange(js, je + 1)
                # Decrease a[idx] to reduce b[k_star]
                # Try decreasing the largest contributor
                contrib = a[idx] * a[np.arange(js, je + 1)]
                worst_local = idx[np.argmax(contrib)]
                # Try reducing it
                a_old = a[worst_local]
                for frac in [0.7, 0.5, 0.3, 0.1]:
                    a_try = a.copy()
                    a_try[worst_local] = a_old * frac
                    a_try = normalize(a_try)
                    sc_try = compute_score(a_try)
                    if sc_try < best_cd_score:
                        a = a_try
                        best_cd_score = sc_try
                        steps_without_improvement = 0
                        break

        try_update(best_cd_score, a, "coord descent subgrad")
        print(f"  Coord descent: {best_cd_score:.10f}")

    # === Final output ===
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [1.0] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
