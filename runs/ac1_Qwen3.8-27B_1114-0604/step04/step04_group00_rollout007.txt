"""

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def gs_fienup_correct(a, n_iters=500, alpha_init=0.7, alpha_final=0.1, M=None):
    """
    GS-Fienup with correct target: A^2 should approximate DFT(b_target),
    where b_target is the flat convolution target.
    """
    n = len(a)
    if M is None:
        M = 1 << (2 * n - 1).bit_length()

    # Precompute target: A_target^2 = DFT(b_target)
    b_target = np.zeros(M, dtype=np.float64)
    b_target[:2*n-1] = 1.0 / (2*n - 1)
    T = np.fft.fft(b_target)  # Target for A^2

    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)

    A_mag_T = np.abs(T)
    A_phase_T = np.angle(T)
    A_mag_T_safe = np.maximum(A_mag_T, 1e-20)

    for it in range(n_iters):
        progress = it / max(1, n_iters)
        alpha = alpha_init + (alpha_final - alpha_init) * progress

        A = np.fft.fft(a, M)
        A_mag = np.abs(A)
        A_phase = np.angle(A)
        A_mag_safe = np.maximum(A_mag, 1e-20)

        new_mag = A_mag_safe ** (1.0 - alpha) * A_mag_T_safe ** (alpha * 0.5)
        # New phase: (1-alpha)*arg(A) + alpha*arg(T)/2

        a_new = np.fft.ifft(A_new, M).real

        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient for L-BFGS."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(x, beta):
    """Wrapper for scipy.minimize."""
    return lse_score_grad(x, beta)


def run_lbfgs(a0, beta, maxiter=8000, maxfun=150000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def top_m_subgradient(a, n_iters=3000, top_m=10, step_init=0.05, decay=0.9995):
    """Subgradient method using top-M peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    sc_lb = 2.0 * n / (2.0 * n - 1.0)

    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        if top_m >= len(c):
            top_idx = np.arange(len(c))
        else:
            top_idx = np.argsort(c)[::-1][:top_m]

        sg = np.zeros(n)
        for k in top_idx:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M_val / S3)

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = min(step, (sc - sc_lb) / (gn * gn + 1e-30))
        step = min(step, 0.08)

        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S
        step *= decay

    return best_a


def coordinate_descent_peak(a, n_sweeps=5, n_top=60):
    """Coordinate descent on peak-contributing elements."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break

        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-12:
                continue
            candidates = [a_old_j * f for f in [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5]]
            best_val = a_old_j
            cur_sc = compute_score(a)
            for cv in candidates:
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old_j) < 1e-10:
                    continue
                a[j] = cv
                S_tmp = a.sum()
                if S_tmp < 1e-12:
                    a[j] = a_old_j
                    continue
                a_scaled = a / S_tmp
                sc = compute_score(a_scaled)
                if sc < cur_sc:
                    cur_sc = sc
                    best_val = cv
                a[j] = a_old_j
            a[j] = best_val

        S = a.sum()
        if S > 1e-12:
            a = a / S
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a

        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths
    lengths = [4096, 6962, 8192, 10240]
    if parent is not None:
        lengths.insert(0, len(parent))
    lengths = sorted(set(lengths))[:5]
    print(f"Lengths to try: {lengths}")
    print(f"Parent: {'yes (n=' + str(len(parent)) + ')' if parent is not None else 'no'}")

    # ==================================================================
    # Phase 1: GS-Fienup with correct target (45% of time)
    # ==================================================================
    phase1_end = t_start + 0.45 * (total_budget - 30.0)
    print(f"\n=== Phase 1: GS-Fienup correct target (deadline t={phase1_end - t_start:.0f}s) ===")

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"  Processing n={n}")
        inits = build_initializations(n, rng, parent)

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break

            # Correct GS with phase
            a1 = gs_fienup_correct(a0, n_iters=400, alpha_init=0.7, alpha_final=0.1)
            sc = compute_score(a1)
            update_best(sc, a1, f"GS-correct n={n} init={i}")

            # GS magnitude-only (more stable)
            if time.monotonic() < phase1_end:
                a2 = gs_fienup_magnitude_only(a0, n_iters=400, alpha=0.5)
                sc2 = compute_score(a2)
                update_best(sc2, a2, f"GS-mag n={n} init={i}")

            # Adaptive GS
            if time.monotonic() < phase1_end - 10:
                a3 = gs_adaptive(a1, n_iters=300)
                sc3 = compute_score(a3)
                update_best(sc3, a3, f"GS-adapt n={n} init={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: GS with different alpha schedules + more inits (15% of time)
    # ==================================================================
    phase2_end = t_start + 0.60 * (total_budget - 30.0)
    print(f"\n=== Phase 2: GS variants (deadline t={phase2_end - t_start:.0f}s) ===")

    n_best = len(best_a)

    # More GS iterations on best so far
    a_gs2 = gs_fienup_correct(best_a, n_iters=600, alpha_init=0.5, alpha_final=0.05)
    sc = compute_score(a_gs2)
    update_best(sc, a_gs2, "GS-refine-1")

    a_gs3 = gs_fienup_magnitude_only(best_a, n_iters=600, alpha=0.3)
    sc = compute_score(a_gs3)
    update_best(sc, a_gs3, "GS-mag-refine")

    # Try at different lengths
    for n_alt in lengths:
        if time.monotonic() > phase2_end:
            break
        if n_alt == n_best:
            continue
        a_interp = np.interp(np.linspace(0, 1, n_alt),
                             np.linspace(0, 1, n_best), best_a)
        a_interp = np.maximum(a_interp, 0.01)
        a_gs_alt = gs_fienup_correct(a_interp, n_iters=300, alpha_init=0.6, alpha_final=0.1)
        sc = compute_score(a_gs_alt)
        update_best(sc, a_gs_alt, f"GS-alt n={n_alt}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: L-BFGS refinement (25% of time)
    # ==================================================================
    phase3_end = t_start + 0.85 * (total_budget - 30.0)
    print(f"\n=== Phase 3: L-BFGS (deadline t={phase3_end - t_start:.0f}s) ===")

    n_best = len(best_a)
    for beta in [100, 300, 500, 1000, 2000, 5000, 10000]:
        if time.monotonic() > phase3_end:
            break
        t_beta_start = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=10000, maxfun=200000)
        update_best(sc, a_opt, f"LBFGS beta={beta}")
        print(f"    beta={beta}: score={sc:.8f}, time={time.monotonic()-t_beta_start:.1f}s")

    # L-BFGS from GS variant
    if time.monotonic() < phase3_end - 60:
        a_gs4 = gs_fienup_correct(best_a, n_iters=200, alpha_init=0.4, alpha_final=0.1)
        sc_gs4 = compute_score(a_gs4)
        sc_lb, a_lb = run_lbfgs(a_gs4, 2000, maxiter=8000, maxfun=100000)
        update_best(min(sc_gs4, sc_lb), a_lb if sc_lb < sc_gs4 else a_gs4, "GS+LBFGS")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Subgradient + Coordinate descent (10% of time)
    # ==================================================================
    print(f"\n=== Phase 4: Subgradient + Coord Descent ===")

    for top_m in [5, 10, 20, 40, 80]:
        if time.monotonic() > deadline:
            break
        a_sg = top_m_subgradient(best_a, n_iters=2000, top_m=top_m)
        sc = compute_score(a_sg)
        update_best(sc, a_sg, f"SG top_m={top_m}")

    if time.monotonic() < deadline - 40:
        a_cd = coordinate_descent_peak(best_a, n_sweeps=5, n_top=100)
        sc = compute_score(a_cd)
        update_best(sc, a_cd, "coord-descent")

        if time.monotonic() < deadline - 20:
            sc2, a_opt = run_lbfgs(a_cd, 5000, maxiter=3000, maxfun=50000)
            update_best(sc2, a_opt, "CD+LBFGS")

    # ==================================================================
    # Phase 5: Final polish with GS + L-BFGS
    # ==================================================================
    if time.monotonic() < deadline - 20:
        a_final_gs = gs_adaptive(best_a, n_iters=300)
        sc = compute_score(a_final_gs)
        update_best(sc, a_final_gs, "FINAL-GS")

    if time.monotonic() < deadline - 10:
        sc, a_opt = run_lbfgs(best_a, 20000, maxiter=5000, maxfun=80000)
        update_best(sc, a_opt, "FINAL-LBFGS")

    # ==================================================================
    # Final output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
