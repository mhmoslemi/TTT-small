def iterate(f, n_iters=100, t=0.1):
    """Iterative LP-based optimization"""
    for it in range(n_iters):
        M = float(fast_conv(f).max())
        g = solve_lp(f, M)
        if g is None:
            break
        # Normalize
        g_norm = g / g.sum() * f.sum()
        f = (1-t)*f + t*g_norm
        f = np.clip(f, 0, None)
    return f
