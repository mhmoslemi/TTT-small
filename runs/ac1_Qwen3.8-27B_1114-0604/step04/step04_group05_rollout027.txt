Peak = (n-1)/n^2 ≈ 1/n.
Score = 2n * (1/n) = 2.

If a is "bimodal" (two blocks of equal height), the convolution would be more complex.

What if a has a specific periodic structure? For example, a[j] = c * (1 + ε * cos(2πj/p)) for some period p. The convolution of a cosine with itself gives a cosine plus a constant, so the peak would be reduced.

Let me try: a[j] = c * (1 + ε * cos(2πj/p)) where c is chosen so that sum(a) = 1.
sum(a) = c * n + c * ε * sum(cos(2πj/p)) = cn (if p divides n, the sum of cos is 0).
So c = 1/n.
a[j] = (1/n) * (1 + ε * cos(2πj/p))

For this to be non-negative, we need 1 + ε * cos(2πj/p) ≥ 0 for all j, so ε ≤ 1.

The self-convolution:
conv(a)[k] = sum_j a[j] * a[k-j]
= (1/n^2) * sum_j (1 + ε*cos(2πj/p)) * (1 + ε*cos(2π(k-j)/p))
= (1/n^2) * [sum_j 1 + ε * sum_j cos(2π(k-j)/p) + ε * sum_j cos(2πj/p) + ε^2 * sum_j cos(2πj/p)*cos(2π(k-j)/p)]
= (1/n^2) * [n + 0 + 0 + ε^2 * sum_j cos(2πj/p)*cos(2π(k-j)/p)]

The last term: sum_j cos(2πj/p)*cos(2π(k-j)/p)
= (1/2) * sum_j [cos(2πk/p) + cos(4πj/p - 2πk/p)]
= (1/2) * [n*cos(2πk/p) + sum_j cos(4πj/p - 2πk/p)]
= (1/2) * n * cos(2πk/p) (if p doesn't divide 4, the second sum is approximately 0)

So conv(a)[k] ≈ (1/n^2) * [n + (ε^2/2) * n * cos(2πk/p)]
= (1/n) * [1 + (ε^2/2) * cos(2πk/p)]

The peak is at k where cos(2πk/p) = 1: conv_max = (1/n) * (1 + ε^2/2)
The minimum is at cos(2πk/p) = -1: conv_min = (1/n) * (1 - ε^2/2)

For the uniform case (ε=0): peak = 1/n. Score = 2n * (1/n) = 2.
For ε > 0: peak = (1/n)(1 + ε^2/2). Score = 2n * (1/n)(1 + ε^2/2) = 2(1 + ε^2/2).

"""
Multi-peak projected gradient + L-BFGS LSE + LP-guided Frank-Wolfe for minimizing
self-convolution peak ratio. Key innovations: (1) multi-peak subgradient averaging
for smoother descent, (2) momentum-based projected gradient on the simplex, (3) 
adaptive LSE beta schedule, (4) LP-based direction finding for moderate n, (5)
structured restarts with chirp/finite-field initializations.
"""

import numpy as np
from scipy.optimize import minimize
import time
import math

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)

def conv_peak_info(a, top_k=5):
    """Return convolution, sorted peak indices, and sum."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    c_max = float(c.max())
    # Find top_k peaks
    threshold = c_max * 0.99
    active = np.where(c >= threshold)[0]
    if len(active) > top_k:
        # Sort by value and take top_k
        active_vals = c[active]
        top_idx = active[np.argsort(active_vals)[::-1][:top_k]]
    else:
        top_idx = active
    return c, top_idx, c_max, S

def subgradient_multi(a, top_idx):
    """Compute average subgradient over multiple peak positions."""
    n = len(a)
    sg = np.zeros(n)
    for k in top_idx:
        k = int(k)
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            idx = k - np.arange(js, je + 1)
            sg[js:je + 1] += a[idx]
    sg /= len(top_idx)
    return sg

def simplex_project(v, target_sum):
    """Project v onto {x >= 0, sum(x) = target_sum} in O(n log n)."""
    n = len(v)
    if n == 0:
        return v
    # Scale to standard simplex problem
    v_scaled = v / target_sum
    # Sort in descending order
    u = np.sort(v_scaled)[::-1]
    sv = np.cumsum(u)
    # Find rho
    rho_candidate = np.arange(1, n + 1)
    condition = u > (1.0 - sv) / rho_candidate
    if not np.any(condition):
        rho = 1
    else:
        rho = int(np.max(np.where(condition, rho_candidate, 0)))
    theta = (sv[rho - 1] - 1.0) / rho
    proj = np.maximum(v_scaled - theta, 0.0)
    # Ensure exact sum
    s = proj.sum()
    if s > 1e-15:
        proj = proj * (1.0 / s)
    return proj * target_sum

def projected_gradient_optimize(a0, n_iter=200000, lr0=0.01, momentum0=0.9,
                                 time_budget=None, t_start=None, top_k_peaks=5,
                                 lr_decay=1e-6):
    """
    Projected gradient with momentum and multi-peak subgradient.
    Optimizes on the simplex (sum(a) = 1, a >= 0).
    """
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S0 = float(a.sum())
    if S0 < 1e-12:
        a[:] = 1.0 / n
        S0 = 1.0
    a = a / S0  # Normalize to sum = 1

    best_score = compute_score(a)
    best_a = a.copy()
    deadline = (t_start + time_budget) if (time_budget is not None and t_start is not None) else np.inf

    v_mom = np.zeros(n)
    lr = lr0
    momentum = momentum0

    for t in range(1, n_iter + 1):
        if time.monotonic() > deadline:
            break
        if t % 10000 == 0:
            lr = lr0 / (1.0 + lr_decay * t)

        # Compute multi-peak subgradient
        c, top_idx, c_max, S = conv_peak_info(a, top_k=top_k_peaks)
        sg = subgradient_multi(a, top_idx)

        # Momentum step
        v_mom = momentum * v_mom - lr * sg
        a_new = a + v_mom

        # Project onto simplex
        a_proj = simplex_project(a_new, 1.0)

        # Armijo backtracking if score worsened too much
        score_new = 2.0 * n * float(fast_conv(a_proj).max())  # score on simplex
        score_old = 2.0 * n * c_max
        if score_new > score_old * 1.01:
            # Reject step, reduce learning rate
            lr *= 0.5
            momentum = min(momentum * 0.9, 0.99)
            continue

        a = a_proj

        # Update best
        sc = 2.0 * n * float(c.max())  # This is score on simplex (sum=1)
        # Actually need to recompute after update
        if t % 100 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    if exp_sum < 1e-300:
        p = np.zeros(2 * n - 1)
        p[c.argmax()] = 1.0
    else:
        p = exp_scaled / exp_sum
    L = c_max + math.log(exp_sum) / beta
    score = 2.0 * n * L / (S * S)

    # Gradient: dL/da_j = sum_k p_k * a[k-j] (cross-correlation of p and a)
    M2 = 2 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    PA = np.fft.rfft(p, M2_fft)
    AA = np.fft.rfft(a, M2_fft)
    R = np.fft.irfft(np.conj(PA) * AA, M2_fft)
    h = R[:n]

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (h / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=3000, maxfun=30000):
    """L-BFGS-B with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def obj_grad(x):
        return lse_score_grad(x, beta)

    res = minimize(obj_grad, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-15, 'gtol': 1e-12,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50})
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out

def coord_descent(a, n_sweeps=5, top_k=40, time_budget=30.0, t_start=None):
    """Coordinate descent with golden-section 1D search."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = (t_start + time_budget) if t_start is not None else np.inf
    gr = (math.sqrt(5.0) - 1.0) / 2.0

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        c, top_idx, c_max, S = conv_peak_info(a, top_k=3)
        score = 2.0 * n * c_max / (S * S)
        if score < best_score:
            best_score = score

        # Multi-peak gradient
        sg = subgradient_multi(a, top_idx)
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * c_max / S3)

        order = np.argsort(np.abs(grad))[::-1][:top_k]
        for j in order:
            if time.monotonic() > deadline:
                break
            j = int(j)
            a_old = a[j]
            lo = 0.0
            hi = max(a_old * 3.0, a.max() * 0.5)
            c1 = hi - gr * (hi - lo)
            c2 = lo + gr * (hi - lo)
            a[j] = c1
            f1 = compute_score(a)
            a[j] = c2
            f2 = compute_score(a)
            for _ in range(12):
                if f1 < f2:
                    hi = c2; c2 = c1; f2 = f1
                    c1 = hi - gr * (hi - lo)
                    a[j] = c1; f1 = compute_score(a)
                else:
                    lo = c1; c1 = c2; f1 = f2
                    c2 = lo + gr * (hi - lo)
                    a[j] = c2; f2 = compute_score(a)
            a[j] = c1 if f1 < f2 else c2

    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
    return best_score, a

def chirp_initialization(n, rng):
    """Generate a chirp (frequency-modulated) sequence."""
    x = np.arange(n, dtype=np.float64) / n
    f0 = float(rng.uniform(0.5, 2.0))
    f1 = float(rng.uniform(3.0, 8.0))
    phase = float(rng.uniform(0, 2 * np.pi))
    chirp = np.sin(2.0 * np.pi * (f0 * x + 0.5 * (f1 - f0) * x * x + phase))
    a = 0.5 + 0.5 * chirp  # Map to [0, 1]
    a = np.clip(a, 0.01, None)
    a = a / a.sum()
    return a

def random_structured(n, rng):
    """Generate a random sequence with some structure."""
    # Sum of a few Gaussians + noise
    a = np.zeros(n)
    n_bumps = int(rng.integers(3, 10))
    for _ in range(n_bumps):
        center = float(rng.integers(0, n))
        width = float(rng.uniform(n // 50, n // 5))
        amp = float(rng.uniform(0.5, 2.0))
        x = np.arange(n, dtype=np.float64)
        a += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
    a += 0.05 * rng.standard_normal(n)
    a = np.clip(a, 0.0, None)
    if a.sum() < 1e-10:
        a = np.ones(n)
    a = a / a.sum()
    return a

def lp_frank_wolfe(a_current, M, n_lp_vars=None):
    """
    Solve the LP for Frank-Wolfe direction.
    Returns the optimal g or None if infeasible/too large.
    """
    n = len(a_current)
    if n > 3000:
        return None  # Too large for dense LP

    # Build constraint matrix A: A[k,j] = a_current[k-j] if 0<=k-j<n
    from scipy.sparse import lil_matrix, csr_matrix
    A = lil_matrix((2 * n - 1, n))
    for k in range(2 * n - 1):
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            indices = k - np.arange(js, je + 1)
            A[k, js:je + 1] = a_current[indices]
    A = A.tocsr()

    # LP: max 1^T g s.t. A g <= M, g >= 0
    from scipy.optimize import linprog
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.status == 0:
        return res.x
    return None

def full_pipeline(a_init, time_budget, t_start):
    """
    Multi-stage optimization pipeline.
    """
    a = a_init.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_budget

    def try_update(sc, a_cand, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a_cand, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"    [NEW BEST] {sc:.8f} | {label} | t={elapsed:.0f}s")
            return True
        return False

    # Stage 1: Projected gradient with momentum (30% of budget)
    if time.monotonic() < deadline - 10:
        t1 = min(time_budget * 0.30, deadline - time.monotonic() - 10)
        sc, a_opt = projected_gradient_optimize(
            a, n_iter=500000, lr0=0.005, momentum0=0.95,
            time_budget=t1, t_start=t_start, top_k_peaks=5)
        try_update(sc, a_opt, "ProjGrad")

    # Stage 2: L-BFGS with increasing beta (35% of budget)
    if time.monotonic() < deadline - 10:
        t2 = min(time_budget * 0.35, deadline - time.monotonic() - 10)
        betas = [50, 200, 1000, 5000]
        t_per_beta = t2 // len(betas)
        for beta in betas:
            if time.monotonic() > deadline - 10:
                break
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=2000, maxfun=20000)
            try_update(sc, a_opt, f"LBFGS beta={beta}")

    # Stage 3: Coordinate descent (15% of budget)
    if time.monotonic() < deadline - 5:
        t3 = min(time_budget * 0.15, deadline - time.monotonic() - 5)
        sc, a_opt = coord_descent(best_a, n_sweeps=8, top_k=50,
                                   time_budget=t3, t_start=t_start)
        try_update(sc, a_opt, "CoordDesc")

    # Stage 4: Another round of projected gradient (10%)
    if time.monotonic() < deadline - 5:
        t4 = min(time_budget * 0.10, deadline - time.monotonic() - 5)
        sc, a_opt = projected_gradient_optimize(
            best_a, n_iter=300000, lr0=0.002, momentum0=0.98,
            time_budget=t4, t_start=t_start, top_k_peaks=8)
        try_update(sc, a_opt, "ProjGrad2")

    # Stage 5: Final L-BFGS polish (10%)
    if time.monotonic() < deadline - 3:
        t5 = min(time_budget * 0.10, deadline - time.monotonic() - 3)
        for beta in [10000, 50000]:
            if time.monotonic() > deadline - 3:
                break
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=50000)
            try_update(sc, a_opt, f"LBFGS final beta={beta}")

    return best_score, best_a

    tilt = (x - center) / max(center, 1)
    freq = float(rng.uniform(1, 4))
    tilt += 0.5 * np.sin(2 * np.pi * freq * x / n)
    tilt_anti = 0.5 * (tilt - tilt[::-1])
    a_anti = best_a / best_a.sum() + 0.1 * tilt_anti * (best_a / best_a.sum()).max()
    a_anti = np.clip(a_anti, 0.0, None)
    if a_anti.sum() > 1e-10:
        a_anti = a_anti / a_anti.sum()
    initializations.append(("anti_palin", a_anti))

    for ri, (name, a_init) in enumerate(initializations):
        if time.monotonic() > t_start + effective_budget - 30:
            break
        time_left = min(per_restart * 1.5, t_start + effective_budget - 30 - time.monotonic())
        if time_left < 20:
            break
        print(f"  Restart {ri}: {name}")
        sc, a_opt = full_pipeline(a_init, time_left, t_start)
        try_update(sc, a_opt, f"Restart:{name}")

    # Phase 3: Try different lengths (10% of budget)
    p3_budget = effective_budget * 0.10
    print(f"\n{'='*60}")
    print(f"Phase 3: Different lengths ({p3_budget:.0f}s)")
    print(f"{'='*60}")

    n_best = len(best_a)
    for factor in [0.8, 1.2, 1.5]:
        if time.monotonic() > t_start + effective_budget - 20:
            break
        tl = int(n_best * factor)
        if tl < 512 or tl > 30000:
            continue
        x_old = np.arange(n_best, dtype=np.float64)
        x_new = np.linspace(0, n_best - 1, tl)
        a_res = np.interp(x_new, x_old, best_a)
        a_res = np.clip(a_res, 0.0, None)
        if a_res.sum() < 1e-10:
            continue
        a_res = a_res / a_res.sum()
        time_left = min(p3_budget / 3, t_start + effective_budget - 20 - time.monotonic())
        if time_left < 15:
            break
        sc, a_opt = full_pipeline(a_res, time_left, t_start)
        try_update(sc, a_opt, f"n={tl}")

    # Phase 4: Final intensive polish (remaining time)
    print(f"\n{'='*60}")
    print(f"Phase 4: Final polish")
    print(f"{'='*60}")

    final_deadline = t_start + effective_budget - 5.0

    # High-precision L-BFGS
    if time.monotonic() < final_deadline - 15:
        for beta in [20000, 100000]:
            if time.monotonic() > final_deadline - 15:
                break
            sc, a_opt = run_lbfgs(best_a, beta, maxiter=8000, maxfun=80000)
            try_update(sc, a_opt, f"Final LBFGS beta={beta}")

    # Final coordinate descent
    remaining = final_deadline - time.monotonic()
    if remaining > 10:
        sc, a_opt = coord_descent(best_a, n_sweeps=10, top_k=60,
                                   time_budget=min(remaining - 5, 60.0), t_start=t_start)
        try_update(sc, a_opt, "Final CoordDesc")

    # Final projected gradient
    remaining = final_deadline - time.monotonic()
    if remaining > 10:
        sc, a_opt = projected_gradient_optimize(
            best_a, n_iter=200000, lr0=0.001, momentum0=0.99,
            time_budget=min(remaining - 5, 30.0), t_start=t_start, top_k_peaks=10)
        try_update(sc, a_opt, "Final ProjGrad")

    # ==================================================================
    # Final verification and return
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
