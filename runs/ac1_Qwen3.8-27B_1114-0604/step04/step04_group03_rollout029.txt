"""
Optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Key insight: monotone (ramp) initializations give score ~4/3, much better than uniform (score=2).
Uses LSE-smoothed L-BFGS, frequency-domain peak shaving, and multi-start.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()
    exp_s = np.exp(scaled)
    log_sum = np.log(exp_s.sum()) + scaled.max() / beta
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # Gradient computation
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n-1:2*n-1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad


def run_lbfgs(a0, beta, maxiter=20000, maxfun=500000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        s, g = lse_obj_grad(x, beta)
        return s, g

    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-20, 'gtol': 1e-15,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 100})
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def spectral_shaping(a, n_iters=100):
    """Flatten the power spectrum to reduce convolution peaks."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    M = 1 << (2 * n - 1).bit_length()
    for it in range(n_iters):
        a_pad = np.zeros(M)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        v = np.abs(A) ** 2
        conv = np.fft.irfft(A * A, M)[:2*n-1]
        # Try to flatten conv by adjusting |A|^2
        conv_target = conv.mean()
        # Reduce peaks in conv
        excess = np.maximum(conv - conv_target, 0)
        if excess.max() < 1e-15:
            break
        # In freq domain, reduce the components that contribute to the peaks
        # This is approximate: reduce |A|^2 where it's largest
        v_target = v.mean()
        v_excess = np.maximum(v - v_target, 0)
        v_new = v - 0.1 * v_excess
        # Take square root and adjust phase
        mag_new = np.sqrt(v_new)
        phase = np.angle(A)
        A_new = mag_new * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    return best_a


def ramp_init(n):
    return np.arange(1, n+1, dtype=np.float64)
