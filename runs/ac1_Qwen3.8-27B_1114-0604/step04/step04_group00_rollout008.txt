"""

Uses adaptive LSE-LBFGS with progressive sharpness, Fienup spectral conditioning,

import numpy as np

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.maximum(a, 0.0)
    S = float(a.sum())
    if S < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / S


def fienup_soft(a, n_iters=50, threshold_frac=0.85):
    """Soft Fienup spectral flattening. Returns flattened sequence normalized to sum=1."""
    n = len(a)
    a = normalize(a.copy())
    M_fft = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        A = np.fft.rfft(a, M_fft)
        P = A.real ** 2 + A.imag ** 2
        # Clip top fraction of power bins
        thresh = np.quantile(P, threshold_frac)
        above = P > thresh
        if not above.any():
            break
        scale = np.ones_like(P)
        scale[above] = np.sqrt(thresh / P[above])
        A_mod = A * scale
        a_new = np.fft.irfft(A_mod, M_fft)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        a = a_new
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def lse_objective_and_grad(x, beta):
    """LSE-smoothed objective: 2n * LSE(conv, beta) / (sum)^2. Returns (value, gradient)."""
    n = len(x)
    x = np.maximum(x, 0.0)
    S = float(x.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c = fast_conv(x)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = float(np.log(exp_scaled.sum()))
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    
    # Gradient
    p = exp_scaled / exp_scaled.sum()  # softmax weights over convolution positions
    rev_x = x[::-1]
    # conv_p[i] = sum_j p[j] * x[j-i] for valid indices
    # This is a cross-correlation: (rev_x * p)[n-1-i]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RX = np.fft.rfft(rev_x, M2_fft)
    P_fft = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RX * P_fft, M2_fft)
    # conv_res[k] = sum_j rev_x[j] * p[k-j] = sum_j x[n-1-j] * p[k-j]
    # We need: g[i] = sum_k p[k] * x[k-i] = sum_j x[j] * p[i+j] ... 
    # Actually: d(conv[k])/dx[i] = x[k-i] (for valid k-i)
    # grad[k] contribution to x[i]: p[k] * x[k-i] for valid k-i
    # g[i] = sum_{k: 0<=k-i<2n-1} p[k] * x[k-i]
    # Let j = k-i, g[i] = sum_j p[i+j] * x[j] (for valid range)
    # This is a "sliding dot product" - can be computed via FFT
    # g = cross-correlation of p and x with specific indexing
    # g[i] = sum_j p[i+j] * x[j] = (p_rev * x)[2n-2-i] where p_rev[m] = p[2n-2-m]
    p_rev = p[::-1]
    g = np.fft.irfft(np.fft.rfft(p_rev, M2_fft) * np.fft.rfft(x, M2_fft), M2_fft)
    g = g[2 * n - 2 - np.arange(n)]  # g[i] = (p_rev * x)[2n-2-i]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs_adaptive(a0, beta_start=50, beta_end=20000, maxiter_total=200000):
    """
    Adaptive LSE-LBFGS: starts with smooth LSE, progressively sharpens beta.
    Returns (best_score, best_a).
    """
    n = len(a0)
    a = normalize(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    
    bounds = [(0.0, 1000.0)] * n
    beta = beta_start
    iter_budget = maxiter_total
    total_iter = 0
    
    while total_iter < iter_budget:
        chunk = min(5000, iter_budget - total_iter)
        sc, a_opt = run_lbfgs_single(a, beta, maxiter=chunk, maxfun=chunk * 4)
        total_iter += chunk
        if sc < best_sc:
            best_sc = sc
            best_a = a_opt.copy()
        # Increase beta
        beta = min(beta * 1.8, beta_end)
        if beta >= beta_end:

                                          maxfun=(iter_budget - total_iter) * 3)
            if sc < best_sc:
                best_sc = sc
                best_a = a_opt.copy()
            break
    
    return best_sc, best_a


def run_lbfgs_single(a0, beta, maxiter=5000, maxfun=20000):
    """Single L-BFGS-B call with fixed beta."""
    n = len(a0)
    a0 = np.maximum(a0.astype(np.float64), 0.0)
    if a0.sum() < 1e-6:
        a0 = np.ones(n) / n
    a0 = a0 / a0.sum()
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective_and_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-15,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30}
    )
    a_out = np.maximum(res.x, 0.0)
    if a_out.sum() < 1e-6:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, normalize(a_out)


def dual_ascent(a0, n_inners=50, n_outers=5000, step_a=0.1, gamma=None):

    Dual ascent (multiplicative weights) for min max_k conv[k] s.t. a>=0, sum(a)=1.

    Inner loop does projected gradient on QP with matrix Q_p.

        gamma = 2.0 / np.log(n_conv)

        # (Q_p * a)[i] = sum_j p[i+j] * a[j] for valid i+j
        # This equals (p_rev * a)[2n-2-i] where p_rev = p[::-1]
        p_rev = p[::-1].copy()

        p = p * r

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = v.copy()
    n = len(v)
    # Find threshold t such that sum(max(v - t, 0)) = 1
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[0] + 1
    theta = (cssv[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)


def subgradient_topk(a, n_iters=5000, top_k=20, step_init=0.02, decay=0.9999):
    """Subgradient method using top-k convolution peaks."""
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    step = step_init
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        M_val = float(c.max())
        sc = 2.0 * n * M_val / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Top-k peaks
        k = min(top_k, len(c))
        top_idx = np.argsort(c)[::-1][:k]
        
        # Subgradient: weighted sum of gradients of top-k conv values
        # grad of conv[k] at position i: a[k-i] if valid
        # Weights: softmax over top-k values
        c_top = c[top_idx]
        w = np.exp(c_top - c_top.max())
        w = w / w.sum()
        
        grad = np.zeros(n)
        for wi, ki in zip(w, top_idx):
            ki = int(ki)
            j_start = max(0, ki - n + 1)
            j_end = min(ki, n - 1)
            if j_end >= j_start:
                # d(conv[ki])/da[j] = a[ki-j] for j in [j_start, j_end]
                idx = np.arange(j_start, j_end + 1)
                grad[idx] += wi * 2.0 * a[ki - idx]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (grad / S2 - 2.0 * M_val / S3)
        
        a -= step * grad
        a = _project_simplex(a)
        step *= decay
    
    return best_a


def coordinate_descent(a, n_sweeps=3, n_top=40):
    """Coordinate descent on top-contributing variables for the peak."""
    n = len(a)
    a = normalize(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        
        # Find top contributing indices
        contrib = np.zeros(n)
        idx_range = np.arange(j_start, j_end + 1)
        contrib[idx_range] = a[k_star - idx_range]
        
        top_local = np.argsort(contrib)[::-1][:n_top]
        top_global = j_start + top_local
        
        for j in top_global:
            j = int(j)
            a_old = a[j]
            if a_old < 1e-12:
                continue
            # Try multiplicative changes
            cur_sc = compute_score(a)
            best_val = a_old
            for factor in [0.3, 0.5, 0.7, 0.85, 1.2, 1.4]:
                a[j] = a_old * factor
                a[j] = min(1000.0, max(0.0, a[j]))
                a_norm = normalize(a)
                sc = compute_score(a_norm)
                if sc < best_sc:
                    best_sc = sc
                if sc < cur_sc:
                    cur_sc = sc
                    best_val = a_old * factor
            a[j] = best_val
        
        a = normalize(a)
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def spectral_random_init(n, rng):
    """Initialize with random phases on a flat (or shaped) magnitude spectrum."""
    M_fft = 1 << (2 * n - 1).bit_length()
    # Random phases
    phases = rng.uniform(0, 2 * np.pi, M_fft // 2 + 1)
    # Magnitude: roughly flat with some shaping
    mags = np.ones(M_fft // 2 + 1)
    # Add some low-pass shaping to ensure non-negativity is easier
    freq = np.arange(M_fft // 2 + 1) / (M_fft // 2 + 1)
    mags *= (1.0 + 0.5 * np.cos(2 * np.pi * freq))
    mags[mags < 0] = 0
    
    A = mags * np.exp(1j * phases)
    # Ensure A[0] is real positive
    A[0] = np.abs(A[0])
    # Hermitian symmetry for real signal
    a_full = np.fft.irfft(A * 2.0, M_fft) - a_full[0]  # Hmm this isn't right
    # Actually for real signal: a = ifft(A) where A is Hermitian
    # rfft gives M_fft//2+1 bins, irfft reconstructs
    a_full = np.fft.irfft(A, M_fft)
    a = np.maximum(a_full[:n], 0.0)
    S = a.sum()
    if S < 1e-12:
        return np.ones(n) / n
    return a / S


def shape_init(n, rng):

    n_parent = len(parent) if parent is not None else 6962
    lengths = [n_parent]
    # Try nearby lengths
    for delta in [-512, -256, 256, 512, 1024, -1024]:
        nl = n_parent + delta
        if 2048 <= nl <= 16384 and nl not in lengths:
            lengths.append(nl)
    lengths = sorted(set(lengths))[:5]
    print(f"Lengths to try: {lengths}")
    print(f"Parent: {'yes (n=' + str(n_parent) + ')' if parent is not None else 'no'}")
    
    # ==================================================================
    # Phase 1: Fienup pre-processing + initial scoring (10% time)
    # ==================================================================
    phase1_end = t_start + 0.10 * (total_budget - 30.0)
    print(f"\n=== Phase 1: Fienup pre-processing ===")
    
    candidates = []  # List of (a_array, length)
    
    # From parent
    if parent is not None:
        a_p = fienup_soft(parent.astype(np.float64) / parent.sum(), n_iters=30, threshold_frac=0.80)
        candidates.append(a_p)
        sc = compute_score(a_p)
        update_best(sc, a_p, "FIENUP-parent")
    
    # From spectral random inits at parent length
    for i in range(5):
        a_r = spectral_random_init(n_parent, rng)
        a_r = fienup_soft(a_r, n_iters=40, threshold_frac=0.80)
        candidates.append(a_r)
        sc = compute_score(a_r)
        update_best(sc, a_r, f"FIENUP-spectral-{i}")
    
    # From shape inits
    for i, a0 in enumerate(shape_init(n_parent, rng)[:4]):
        a_s = fienup_soft(a0, n_iters=40, threshold_frac=0.80)
        candidates.append(a_s)
        sc = compute_score(a_s)
        update_best(sc, a_s, f"FIENUP-shape-{i}")
    
    # From other lengths
    if time.monotonic() < phase1_end:
        for n_alt in lengths[1:3]:
            a_alt = spectral_random_init(n_alt, rng)
            a_alt = fienup_soft(a_alt, n_iters=30, threshold_frac=0.80)
            candidates.append(a_alt)
            sc = compute_score(a_alt)
            update_best(sc, a_alt, f"FIENUP-n={n_alt}")
    
    print(f"After Phase 1: best = {best_score:.8f}")
    
    # ==================================================================
    # Phase 2: Adaptive LSE-LBFGS on best candidates (50% time)
    # ==================================================================
    phase2_end = t_start + 0.60 * (total_budget - 30.0)
    print(f"\n=== Phase 2: Adaptive LSE-LBFGS ===")
    
    # Sort candidates by score, take top few
    scored = [(compute_score(a), a) for a in candidates if len(a) == n_parent or time.monotonic() < phase2_end]
    scored.sort(key=lambda x: x[0])
    
    n_at_budget = max(1, min(4, int(3.0 * (phase2_end - time.monotonic()) / (total_budget - 30.0))))
    
    for idx, (sc_init, a_init) in enumerate(scored[:n_at_budget]):
        if time.monotonic() > phase2_end:
            break
        
        n_cand = len(a_init)
        remaining_s = phase2_end - time.monotonic()
        if remaining_s < 60:
            break
        
        print(f"  LSE-LBFGS on candidate {idx} (n={n_cand}, init_score={sc_init:.8f}), budget={remaining_s:.0f}s")
        
        # Adaptive LSE-LBFGS
        maxiter = int(remaining_s * 3)  # rough estimate
        sc, a_opt = run_lbfgs_adaptive(a_init, beta_start=30, beta_end=30000, maxiter_total=maxiter)
        update_best(sc, a_opt, f"LBFGS-adapt-{idx}")
        
        # Also try from a Fienup-processed version
        if time.monotonic() < phase2_end - 30:
            a_f2 = fienup_soft(a_opt, n_iters=20, threshold_frac=0.85)
            sc2, a_opt2 = run_lbfgs_adaptive(a_f2, beta_start=100, beta_end=50000,
                                             maxiter_total=int((phase2_end - time.monotonic()) * 2))
            update_best(sc2, a_opt2, f"FIENUP+LBFGS-{idx}")
    
    print(f"After Phase 2: best = {best_score:.8f}")
    
    # ==================================================================
    # Phase 3: Dual ascent (multiplicative weights) (20% time)
    # ==================================================================
    phase3_end = t_start + 0.80 * (total_budget - 30.0)
    print(f"\n=== Phase 3: Dual Ascent ===")
    
    if time.monotonic() < phase3_end:
        a_da = dual_ascent(best_a, n_inners=30, n_outers=3000, step_a=0.05, gamma=1.5)
        # dual_ascent returns (score, a) where score is 2n*max(conv) for sum=1
        sc_da = compute_score(a_da)
        update_best(sc_da, a_da, "DUAL-ASCENT")
        
        # Also try with different parameters
        if time.monotonic() < phase3_end:
            a_da2 = dual_ascent(best_a, n_inners=20, n_outers=2000, step_a=0.03, gamma=2.5)
            sc_da2 = compute_score(a_da2)
            update_best(sc_da2, a_da2, "DUAL-ASCENT-2")
    
    print(f"After Phase 3: best = {best_score:.8f}")
    
    # ==================================================================
    # Phase 4: Subgradient + Coordinate Descent polishing (10% time)
    # ==================================================================
    phase4_end = t_start + 0.92 * (total_budget - 30.0)
    print(f"\n=== Phase 4: Subgradient + Coord Descent ===")
    
    if time.monotonic() < phase4_end:
        for top_k in [10, 30, 100]:
            if time.monotonic() > phase4_end:
                break
            a_sg = subgradient_topk(best_a, n_iters=3000, top_k=top_k, step_init=0.01, decay=0.9999)
            sc = compute_score(a_sg)
            update_best(sc, a_sg, f"SG-top{top_k}")
    
    if time.monotonic() < phase4_end:
        a_cd = coordinate_descent(best_a, n_sweeps=3, n_top=50)
        sc = compute_score(a_cd)
        update_best(sc, a_cd, "COORD-DESCENT")
        
        # L-BFGS from coord descent
        if time.monotonic() < phase4_end - 10:
            sc, a_opt = run_lbfgs_adaptive(a_cd, beta_start=200, beta_end=50000, maxiter_total=30000)
            update_best(sc, a_opt, "CD+LBFGS")
    
    print(f"After Phase 4: best = {best_score:.8f}")
    
    # ==================================================================
    # Phase 5: Final refinement
    # ==================================================================
    print(f"\n=== Phase 5: Final Refinement ===")
    
    if time.monotonic() < deadline:
        # One more Fienup + LSE pass
        a_final = fienup_soft(best_a, n_iters=30, threshold_frac=0.82)
        sc = compute_score(a_final)
        update_best(sc, a_final, "FINAL-FIENUP")
        
        if time.monotonic() < deadline - 10:
            sc, a_opt = run_lbfgs_adaptive(a_final, beta_start=500, beta_end=100000, maxiter_total=20000)
            update_best(sc, a_opt, "FINAL-LBFGS")
    
    # Final coordinate descent
    if time.monotonic() < deadline - 5:
        a_cd_final = coordinate_descent(best_a, n_sweeps=2, n_top=30)
        sc = compute_score(a_cd_final)
        update_best(sc, a_cd_final, "FINAL-CD")
    
    # ==================================================================
    # Output
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64
    
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {total_budget}s")
    print(f"{'=' * 60}")
    return out
