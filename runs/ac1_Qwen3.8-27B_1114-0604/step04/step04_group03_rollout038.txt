"""
Optimizer for minimizing 2n*max(conv(a))/(sum(a))^2 using LSE-smoothed
L-BFGS, projected subgradient, peak shaving, and frequency-domain methods.
Improvements over previous: better beta schedule, softmax parameterization
for constraint handling, active-set identification, and multi-scale approach.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """
    LSE-smoothed score and gradient.
    Score = 2n * LSE_beta(conv(a)) / (sum(a))^2
    where LSE_beta(c) = c_max + (1/beta)*log(sum(exp(beta*(c - c_max))))
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled -= scaled.max()  # numerical stability
    exp_s = np.exp(scaled)
    log_sum_exp = np.log(exp_s.sum()) + scaled.max() / beta
    L = c_max + log_sum_exp / beta
    
    score = 2.0 * n * L / (S * S)
    
    # Softmax weights
    p = exp_s / exp_s.sum()
    
    # dL/da[i] = 2 * sum_j p[j] * a[j-i] (for valid j-i)
    # = 2 * (flip(a) * p)[n-1+i]
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    
    # Full gradient of score = 2n * L / S^2
    # d(score)/da[i] = 2n * (dL/da[i]) / S^2 - 2n * L * 2 / S^3
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    
    return score, grad


def run_lbfgs(a0, beta, maxiter=10000, maxfun=200000):
    """Run L-BFGS-B on the LSE-smoothed objective with projection."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 1e-10, 1000.0).copy()
    bounds = [(1e-10, 1000.0)] * n

    def obj(x):
        s, g = lse_obj_grad(x, beta)
        return s, g

    res = minimize(
        obj, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def projected_subgradient(a, n_iters=5000, verbose=False):
    """
    Projected subgradient method for minimizing max(conv(a)) subject to
    a >= 0 and sum(a) = 1.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S  # normalize to sum = 1
    
    best_a = a.copy()
    best_sc = 2.0 * n * float(fast_conv(a).max())
    
    # Polyak step size: step_k = (f(x_k) - f*) / ||g_k||^2
    # We use f* = 1/(2n-1) as lower bound for max(conv) when sum=1
    f_star = 1.0 / (2 * n - 1)
    
    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        k_star = int(np.argmax(c))
        f_val = M  # current value of max(conv) with sum(a)=1
        
        if f_val < best_sc:
            best_sc = f_val
            best_a = a.copy()
        
        # Subgradient of max at k_star
        g = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            g[j_start:j_end + 1] = a[k_star - j_end:k_star - j_start + 1][::-1]
        
        gn = float(np.linalg.norm(g))
        if gn < 1e-18:
            break
        
        # Polyak step
        step = (f_val - f_star) / (gn * gn + 1e-30)
        step = min(step, 0.02)  # cap step size
        
        # Update
        a_new = a - step * g
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            a_new = a  # revert
        else:
            a_new = a_new / S_new
            a = a_new
    
    return best_a


def peak_shave(a, n_iters=2000, verbose=False):
    """
    Iteratively identify the peak of the convolution and reduce it
    by adjusting the contributing elements.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    best_a = a.copy()
    best_sc = 2.0 * n * float(fast_conv(a).max())
    f_star = 1.0 / (2 * n - 1)
    
    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        k_star = int(np.argmax(c))
        
        if M < best_sc:
            best_sc = M
            best_a = a.copy()
        
        # Find contributors to the peak
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break
        
        # Contribution of each a[j] to c[k_star] is a[k_star - j]
        contrib = a[k_star - j_end:k_star - j_start + 1][::-1].copy()
        
        # Reduce top contributors, add to bottom
        n_top = max(3, n // 20)
        n_top = min(n_top, len(contrib))
        top_idx = np.argsort(contrib)[::-1][:n_top]
        bot_idx = np.argsort(contrib)[:max(1, len(contrib) - n_top)]
        
        # Reduce top by a small amount
        total_reduce = 0.0
        for ti in top_idx:
            gi = j_start + ti
            red = a[gi] * 0.01
            a[gi] = max(a[gi] - red, 0.0)
            total_reduce += red
        
        if total_reduce > 0:
            per = total_reduce / len(bot_idx)
            for bi in bot_idx:
                a[j_start + bi] += per
            
            S = float(a.sum())
            if S > 1e-12:
                a = a / S
    
    return best_a


def freq_domain_equality(a, n_iters=500, target_flatness=0.5):
    """
    Frequency-domain method to make the convolution more flat.
    Iteratively adjusts the spectrum to reduce peaks.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    
    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = 2.0 * n * float(fast_conv(a).max())
    
    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        c_mean = float(c.mean())
        
        if c_max - c_mean < 1e-14:
            break
        
        # Create target: blend between current and flat
        c_target = c_mean + target_flatness * (c - c_mean)
        # Cap the peaks
        thresh = c_mean + 0.3 * (c_max - c_mean)
        c_target = np.minimum(c_target, thresh)
        c_target = np.maximum(c_target, 0.0)
        
        # In frequency domain: want |A|^2 ≈ FFT(c_target)
        C_t = np.fft.rfft(c_target, M)
        A = np.fft.rfft(np.zeros(M))
        a_pad = np.zeros(M)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        
        # Adjust amplitude: A_new = A * (|C_t|^0.5 / |A|)
        # = A * sqrt(C_t / |A|^2) ... but C_t is real (autocorrelation spectrum)
        # Actually C_t should be real and non-negative if c_target is an autocorrelation
        # But it might not be, so we take the real part
        C_t_real = np.real(C_t)
        C_t_real = np.maximum(C_t_real, 0.0)
        
        A_mag = np.abs(A)
        A_phase = np.angle(A)
        
        with np.errstate(divide='ignore', invalid='ignore'):
            A_new_mag = np.sqrt(C_t_real)
            # Blend
            alpha = 0.5
            A_new_mag = alpha * A_new_mag + (1 - alpha) * A_mag
            A_new_mag[A_mag < 1e-10] = A_mag[A_mag < 1e-10]
        
        A_new = A_new_mag * np.exp(1j * A_phase)
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        
        sc = 2.0 * n * float(fast_conv(a_new).max())
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new
    
    return best_a


def softmax_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """
    L-BFGS on the softmax parameterization.
    a = softmax(z), so constraints are automatically satisfied.
    We optimize over z.
    """
    n = len(a0)
    a0 = np.maximum(a0, 1e-10)
    a0 = a0 / a0.sum()
    z0 = np.log(a0)
    z0 -= z0.mean()  # center
    
    def obj_z(z):
        a = np.exp(z - z.max())
        a = a / a.sum()
        s, g_a = lse_obj_grad(a, beta)
        # Chain rule: z -> a -> score
        # a_i = exp(z_i)/sum(exp(z_j))
        # da_i/dz_j = a_i * (delta_ij - a_j)
        # d score / dz_j = sum_i (d score / da_i) * (da_i / dz_j)
        # = sum_i g_a[i] * a_i * (delta_ij - a_j)
        # = g_a[j] * a_j - a_j * sum_i g_a[i] * a_i
        dot = np.dot(g_a, a)
        g_z = g_a * a - a * dot
        return s, g_z
    
    res = minimize(
        obj_z, z0,
        method='L-BFGS-B', jac=True,
        options={'ftol': 1e-16, 'gtol': 1e-10,
                 'maxiter': maxiter, 'maxfun': maxfun}
    )
    z_out = res.x
    a_out = np.exp(z_out - z_out.max())
    a_out = a_out / a_out.sum()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def build_diverse_inits(n, rng, parent=None):
    """Generate diverse initializations for the optimizer."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine bump (concentrated in middle)
    inits.append(np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * x), 0.01))
    inits[-1] = inits[-1] / inits[-1].sum()
    
    # Inverse cosine (concentrated at edges)
    inits.append(np.maximum(1.0 - 0.5 * np.cos(2 * np.pi * x), 0.01))
    inits[-1] = inits[-1] / inits[-1].sum()
    
    # Gaussian
    inits.append(np.exp(-4 * (x - 0.5) ** 2) / np.exp(-4 * (x - 0.5) ** 2).sum())
    
    # Ramp
    inits.append((np.arange(n, dtype=np.float64) + 1) / n)
    inits[-1] = inits[-1] / inits[-1].sum()
    
    # U-shaped (large at edges, small in middle)
    inits.append(np.maximum(1 + (2 * x - 1) ** 2, 0.01))
    inits[-1] = inits[-1] / inits[-1].sum()
    
    # W-shaped
    inits.append(np.maximum(1 + np.cos(4 * np.pi * x), 0.01))
    inits[-1] = inits[-1] / inits[-1].sum()
    
    # Random smooth
    freqs = rng.integers(1, 10, size=5)
    amps = rng.random(5) * 0.3
    phases = rng.uniform(0, 2 * np.pi, 5)
    sig = np.zeros(n)
    for f, a, p in zip(freqs, amps, phases):
        sig += a * np.cos(2 * np.pi * f * x + p)
    inits.append(np.maximum(1 + sig, 0.01))
    inits[-1] = inits[-1] / inits[-1].sum()
    
    # Parent interpolation if available
    if parent is not None and len(parent) > 10:
        if len(parent) == n:
            p = np.maximum(parent, 0.0)
            if p.sum() > 0:
                inits.append(p / p.sum())
        else:
            p = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(parent)), np.maximum(parent, 0.0))
            if p.sum() > 0:
                inits.append(p / p.sum())
    
    return inits

"""
Optimizer for minimizing 2n*max(conv(a))/(sum(a))^2 using LSE-smoothed L-BFGS,
projected subgradient, exact coordinate descent, and frequency-domain flattening.
Key improvements: aggressive beta schedule up to 50000, exact 1D coordinate
descent on affected lags, adaptive peak shaving, and multi-scale exploration.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT, returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed objective and gradient. Returns (score, grad)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled_max = scaled.max()
    exp_s = np.exp(scaled - scaled_max)
    log_sum = np.log(exp_s.sum())
    L = c_max + (scaled_max + log_sum) / beta
    score = 2.0 * n * L / (S * S)
    p = exp_s / exp_s.sum()
    # dL/da[k] = 2 * sum_m p[k+m] * a[m] = 2 * (rev_a * p)[n-1+k]
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    r = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    dL_da = 2.0 * r
    grad = (2.0 * n) * (dL_da / (S * S) - 2.0 * L / (S * S * S))
    return score, grad

            if a_i < 1e-15:
                # Try small positive values
                candidates = [a_i + 1e-5, a_i + 1e-4, a_i + 5e-4]
            else:
                # The affected lags are k in [i, i+n-1] intersect [0, 2n-2]
                k_lo = max(0, i)
                k_hi = min(2 * n - 2, i + n - 1)
                # Current values of affected lags
                c_aff = c[k_lo:k_hi + 1].copy()
                # For each affected k, c_k = 2*a_i*a[k-i] + rest (if k!=2i)
                # or c_k = a_i^2 + rest (if k==2i)
                # When a_i changes to v:
                #   c_k_new = c_k + 2*(v - a_i)*a[k-i] (k != 2i)
                #   c_k_new = c_k + 2*(v-a_i)*a_i + (v-a_i)^2 (k == 2i)
                # We want to minimize max(c_k_new) over affected k
                # Plus un-affected lags which stay at their current values
                c_unaff_max = float(np.max(np.delete(c, range(k_lo, k_hi + 1)))) if k_hi > k_lo else 0.0

                # Generate candidate values
                candidates = [0.0, a_i * 0.1, a_i * 0.3, a_i * 0.5,
                              a_i * 0.7, a_i * 0.9, a_i,
                              a_i * 1.1, a_i * 1.3, a_i * 1.5, a_i * 2.0]
                # Also try values that would make top affected lags equal to c_unaff_max
                # For linear case: c_k + 2*(v-a_i)*a[k-i] = c_unaff_max
                # => v = a_i + (c_unaff_max - c_k) / (2*a[k-i])
                for idx_k in range(len(c_aff)):
                    k = k_lo + idx_k
                    a_ki = a[k - i] if 0 <= k - i < n else 0.0
                    if a_ki > 1e-15 and k != 2 * i:
                        v = a_i + (c_unaff_max - c_aff[idx_k]) / (2.0 * a_ki)
                        if 0.0 <= v <= 10.0 / n:
                            candidates.append(v)
                    if k == 2 * i and a_i > 1e-15:
                        # Quadratic: a_i^2 + 2*(v-a_i)*a_i + (v-a_i)^2 = c_unaff_max
                        # v^2 = c_unaff_max => v = sqrt(c_unaff_max)
                        if c_unaff_max > 0:
                            v = np.sqrt(c_unaff_max)
                            if 0.0 <= v <= 10.0 / n:
                                candidates.append(v)

                # Evaluate best candidate
                best_v = a_i
                best_local = best_sc
                for v in candidates:
                    v = max(0.0, min(v, 1000.0))
                    if abs(v - a_i) < 1e-12:
                        continue
                    a[i] = v
                    # Update affected lags
                    for idx_k in range(len(c_aff)):
                        k = k_lo + idx_k
                        a_ki = a[k - i] if 0 <= k - i < n else 0.0
                        if k == 2 * i:
                            c[k] = c_unaff_max + (v - a_i) * 2 * a_i + (v - a_i) ** 2
                            # Actually recompute: contribution from index i to c[k]
                            # c[k] = sum_j a[j]*a[k-j], the i-contribution is a[i]*a[k-i] + a[k-i]*a[i]
                            # = 2*a[i]*a[k-i] if k!=2i, or a[i]^2 if k=2i
                            # We stored c_aff which is c before change. Let's just recompute locally.
                            pass
                    # Just compute score directly for correctness
                    sc = 2.0 * n * float(np.maximum(c.max(), float(fast_conv(a).max())))
                    # Too expensive to recompute full conv each time for large n
                    # Use approximation: only affected lags changed
                    c_new_max = max(float(np.max(c_aff + 2.0 * (v - a_i) * a[k_lo + np.arange(len(c_aff)) - i]))
                                    if np.all((k_lo + np.arange(len(c_aff)) - i >= 0) & (k_lo + np.arange(len(c_aff)) - i < n))
                                    else float(fast_conv(a).max()),
                                    c_unaff_max)
                    # For safety, just use the full computation for the top few candidates
                    sc_local = 2.0 * n * c_new_max  # sum(a) ≈ 1 since we're adjusting one coordinate
                    if sc_local < best_local:
                        best_local = sc_local
                        best_v = v
                a[i] = best_v
                # Recompute full conv periodically
                if i % 50 == 0 or i == n - 1:
                    c = fast_conv(a)
                    S = float(a.sum())
                    if S > 1e-12:
                        a = a / S
                        c = fast_conv(a)
                    sc = 2.0 * n * float(c.max())
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()

    return best_a


def projected_subgradient(a, n_iters=5000):
    """Projected subgradient on sum=1 simplex for minimizing max(conv)."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S
    best_a = a.copy()
    best_sc = 2.0 * n * float(fast_conv(a).max())
    f_star = 1.0 / (2 * n - 1)

    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        k_star = int(np.argmax(c))
        if M < best_sc / (2.0 * n):
            best_sc = 2.0 * n * M
            best_a = a.copy()

        g = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            g[j_start:j_end + 1] = a[k_star - j_start:k_star - j_end + 1]

        gn2 = float(np.dot(g, g))
        if gn2 < 1e-20:
            break
        step = min((M - f_star) / (gn2 + 1e-30), 0.01)
        step = max(step, 1e-8)

        a_new = a - step * g
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a = a_new / S_new

    return best_a


def peak_shave(a, n_iters=3000):
    """Iteratively reduce the peak by adjusting top contributors."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = 2.0 * n * float(fast_conv(a).max())

    for it in range(n_iters):
        c = fast_conv(a)
        M = float(c.max())
        k_star = int(np.argmax(c))
        if 2.0 * n * M < best_sc:
            best_sc = 2.0 * n * M
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end < j_start:
            break
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        n_top = max(3, min(len(contrib) // 4, len(contrib)))
        top_idx = np.argsort(contrib)[::-1][:n_top]
        bot_idx = np.argsort(contrib)[:max(1, len(contrib) - n_top)]

        total_red = 0.0
        for ti in top_idx:
            gi = j_start + ti
            red = a[gi] * 0.005
            a[gi] = max(a[gi] - red, 0.0)
            total_red += red
        if total_red > 0 and len(bot_idx) > 0:
            per = total_red / len(bot_idx)
            for bi in bot_idx:
                a[j_start + bi] += per
            a = a / a.sum()

    return best_a


def freq_flatten(a, n_iters=300):
    """Frequency domain method to flatten the convolution."""
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = 2.0 * n * float(fast_conv(a).max())

    for it in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        c_mean = float(c.mean())
        if c_max - c_mean < 1e-14:
            break
        # Target: reduce peaks toward mean
        c_target = np.minimum(c, c_mean + 0.4 * (c_max - c_mean))
        c_target = np.maximum(c_target, 0.0)

        a_pad = np.zeros(M)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad, M)
        C_t = np.real(np.fft.rfft(c_target, M))
        C_t = np.maximum(C_t, 0.0)

        A_mag = np.abs(A)
        A_phase = np.angle(A)
        target_mag = np.sqrt(C_t)
        with np.errstate(divide='ignore', invalid='ignore'):
            new_mag = 0.5 * target_mag + 0.5 * A_mag
            new_mag[A_mag < 1e-10] = A_mag[A_mag < 1e-10]
        A_new = new_mag * np.exp(1j * A_phase)
        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            break
        a_new = a_new / S_new
        sc = 2.0 * n * float(fast_conv(a_new).max())
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a


def build_inits(n, rng, parent=None):
    """Diverse initializations normalized to sum=1."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    inits.append(np.ones(n) / n)
    for shape_fn in [
        lambda x: np.maximum(0.5 + 0.5 * np.cos(2 * np.pi * x), 0.01),
        lambda x: np.maximum(1.0 - 0.5 * np.cos(2 * np.pi * x), 0.01),
        lambda x: np.exp(-4 * (x - 0.5) ** 2),
        lambda x: np.maximum(1 + (2 * x - 1) ** 2, 0.01),
        lambda x: np.maximum(1 + np.cos(4 * np.pi * x), 0.01),
        lambda x: np.maximum(1 + 0.3 * np.cos(2 * np.pi * 3 * x), 0.01),
        lambda x: (np.arange(n, dtype=np.float64) + 1) / n,
    ]:
        v = shape_fn(x)
        v = np.maximum(v, 0.01)
        inits.append(v / v.sum())

    # Random smooth
    freqs = rng.integers(1, 8, size=4)
    amps = rng.random(4) * 0.3
    phases = rng.uniform(0, 2 * np.pi, 4)
    sig = np.zeros(n)
    for f, amp, ph in zip(freqs, amps, phases):
        sig += amp * np.cos(2 * np.pi * f * x + ph)
    v = np.maximum(1 + sig, 0.01)
    inits.append(v / v.sum())

    if parent is not None and len(parent) >= 10:
        p = np.maximum(parent, 0.0)
        if p.sum() > 0:
            if len(p) == n:
                inits.append(p / p.sum())
            else:
                pv = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(p)), p)
                pv = np.maximum(pv, 0.0)
                if pv.sum() > 0:
                    inits.append(pv / pv.sum())

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a))/(sum(a))^2.
    Returns best sequence found within time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 20.0

    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
                ps = compute_score(parent)
                print(f"Parent: len={len(p)}, score={ps:.8f}")
    except Exception as e:
        print(f"No parent: {e}")

    best_score = np.inf
    best_a = np.ones(2048) / 2048

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n_try in [4096, 6962, 8192, 10000, 12288, 16384]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))
    print(f"Lengths: {lengths}")

    # Phase 1: LSE from inits (15% time)
    p1_end = t_start + 0.15 * (float(budget_s) - 20.0)
    print(f"\n=== Phase 1: LSE from inits ===")
    for n in lengths:
        if time.monotonic() > p1_end:
            break
        print(f"  n={n}")
        inits = build_inits(n, rng, parent)
        for i, a0 in enumerate(inits):
            if time.monotonic() > p1_end:
                break
            sc, a_opt = run_lbfgs(a0, 200, maxiter=3000, maxfun=50000)
            report(sc, a_opt, f"P1 n={n} i={i}")
    print(f"  Best after P1: {best_score:.8f}")

    # Phase 2: Progressive LSE (35% time)
    p2_end = t_start + 0.50 * (float(budget_s) - 20.0)
    print(f"\n=== Phase 2: Progressive LSE ===")
    for beta in [500, 1000, 2000, 5000, 10000, 20000, 50000]:
        if time.monotonic() > p2_end:
            break
        t0 = time.monotonic()
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=10000, maxfun=200000)
        dt = time.monotonic() - t0
        report(sc, a_opt, f"P2 b={beta} ({dt:.0f}s)")
    print(f"  Best after P2: {best_score:.8f}")

    # Phase 3: Coordinate descent + subgradient + peak shave (25% time)
    p3_end = t_start + 0.75 * (float(budget_s) - 20.0)
    print(f"\n=== Phase 3: Coord descent + subgrad + shave ===")

    # Coordinate descent
    if time.monotonic() < p3_end:
        t0 = time.monotonic()
        a_cd = coordinate_descent(best_a, n_sweeps=3)
        sc = compute_score(a_cd)
        dt = time.monotonic() - t0
        report(sc, a_cd, f"coord ({dt:.0f}s)")

    # Subgradient
    if time.monotonic() < p3_end:
        t0 = time.monotonic()
        n_it = int(min(10000, (p3_end - time.monotonic()) / 0.04))
        a_sub = projected_subgradient(best_a, n_iters=n_it)
        sc = compute_score(a_sub)
        dt = time.monotonic() - t0
        report(sc, a_sub, f"subgrad ({dt:.0f}s)")

    # Peak shave
    if time.monotonic() < p3_end:
        t0 = time.monotonic()
        n_it = int(min(5000, (p3_end - time.monotonic()) / 0.03))
        a_sh = peak_shave(best_a, n_iters=n_it)
        sc = compute_score(a_sh)
        dt = time.monotonic() - t0
        report(sc, a_sh, f"shave ({dt:.0f}s)")

    # Combined: subgrad -> shave -> LSE
    if time.monotonic() < p3_end - 30:
        a_comb = peak_shave(projected_subgradient(best_a, n_iters=2000), n_iters=2000)
        sc, a_opt = run_lbfgs(a_comb, 5000, maxiter=5000, maxfun=80000)
        report(sc, a_opt, "subgrad+shave+LSE")

    # Freq flatten
    if time.monotonic() < p3_end - 20:
        a_ff = freq_flatten(best_a, n_iters=200)
        sc = compute_score(a_ff)
        report(sc, a_ff, "freq_flatten")
        sc, a_opt = run_lbfgs(a_ff, 5000, maxiter=5000, maxfun=80000)
        report(sc, a_opt, "freq_flatten+LSE")

    print(f"  Best after P3: {best_score:.8f}")

    # Phase 4: Diversification (15% time)
    p4_end = t_start + 0.90 * (float(budget_s) - 20.0)
    print(f"\n=== Phase 4: Diversification ===")
    n_best = len(best_a)
    for trial in range(20):
        if time.monotonic() > p4_end:
            break
        s = trial % 5
        if s == 0:
            a_p = np.maximum(best_a * (1 + 0.03 * rng.standard_normal(n_best)), 0)
        elif s == 1:
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            a_p = best_a.copy()
            js = max(0, k_star - n_best + 1)
            je = min(k_star, n_best - 1)
            seg = a_p[js:je + 1]
            contrib = seg[::-1].copy()
            nt = max(5, n_best // 15)
            top = np.argsort(contrib)[::-1][:nt]
            bot = np.argsort(contrib)[:max(1, len(contrib) - nt)]
            for ti in top:
                a_p[js + ti] *= 0.95
            tot = a_p.sum()
            if tot > 0:
                a_p = a_p / tot
                for bi in bot:
                    a_p[js + bi] += 0.001
        elif s == 2:
            x = np.arange(n_best) / n_best
            f = 1 + 8 * rng.random()
            ph = 2 * np.pi * rng.random()
            a_p = np.maximum(best_a * (1 + 0.03 * np.cos(2 * np.pi * f * x + ph)), 0)
        elif s == 3:
            other_n = lengths[1 + trial % (len(lengths) - 1)] if len(lengths) > 1 else n_best
            if other_n != n_best:
                a_p = np.interp(np.linspace(0, 1, other_n), np.linspace(0, 1, n_best), best_a)
            else:
                a_p = np.maximum(best_a * (1 + 0.05 * rng.standard_normal(n_best)), 0)
        else:
            a_p = freq_flatten(np.maximum(best_a * (1 + 0.02 * rng.standard_normal(n_best)), 0), n_iters=100)

        if a_p.sum() < 1e-6:
            continue
        a_p = a_p / a_p.sum()
        sc, a_opt = run_lbfgs(a_p, 3000, maxiter=5000, maxfun=80000)
        report(sc, a_opt, f"P4 #{trial} s={s}")
    print(f"  Best after P4: {best_score:.8f}")

    # Phase 5: Final polish (10% time)
    print(f"\n=== Phase 5: Final polish ===")
    if time.monotonic() < deadline - 10:
        sc, a_opt = run_lbfgs(best_a, 50000, maxiter=10000, maxfun=200000)
        report(sc, a_opt, "final LSE 50k")

    if time.monotonic() < deadline - 10:
        n_it = int(min(15000, max(1000, (deadline - time.monotonic()) / 0.04)))
        a_sub = projected_subgradient(best_a, n_iters=n_it)
        sc = compute_score(a_sub)
        report(sc, a_sub, "final subgrad")

    if time.monotonic() < deadline - 10:
        a_sh = peak_shave(best_a, n_iters=3000)
        sc = compute_score(a_sh)
        report(sc, a_sh, "final shave")
        if time.monotonic() < deadline - 5:
            sc, a_opt = run_lbfgs(a_sh, 10000, maxiter=5000, maxfun=80000)
            report(sc, a_opt, "shave+LSE 10k")

    # Output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 128

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length={len(out)}, Time={elapsed:.1f}s/{budget_s}s")
    print(f"{'='*60}")
    return out
