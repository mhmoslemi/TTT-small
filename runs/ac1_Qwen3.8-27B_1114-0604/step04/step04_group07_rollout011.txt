For random sequences, max/avg is typically O(sqrt(n)) for the sidelobes (the peak at lag 0 is n times the average). But we're looking at the full convolution (not just sidelobes), and the "peak" includes the main lobe.

Actually wait. For a non-negative sequence, c[0] = a[0]², c[1] = 2a[0]a[1], ..., c[n-1] = sum a[j]a[n-1-j]. The "main lobe" of the autocorrelation is the entire thing (since the signal is finite). There's no "sidelobe" distinction here.

The maximum of c is typically at k = n-1 (the center) for "smooth" sequences, because that's where the most terms contribute. For a "spiky" sequence, the max might be elsewhere.

For our target: we need max(c) ≤ 1.5 * S²/(2n-1) ≈ 1.5 * S²/(2n).

At the center (k = n-1): c[n-1] = sum_j a[j]*a[n-1-j]. For this to be ≤ 1.5*S²/(2n), we need the "cross-correlation" of a with its reverse to be small.

One way to achieve this: make a[j] and a[n-1-j] "out of phase" - when one is large, the other is small. This is the opposite of a palindrome!

For example, if a is "left-heavy" (a[j] large for small j, small for large j), then at k=n-1, the products a[j]*a[n-1-j] pair large with small, reducing the sum.

But we also need the other lags to be small. At k=0, c[0] = a[0]² should be small, so a[0] should be small. At k=n-2, c[n-2] = a[0]*a[n-2] + a[1]*a[n-1] + ... + a[n-2]*a[0], which also involves cross-products.

"""
Hybrid optimizer combining Iterative Fourier Method (IFM) with LSE-smoothed
gradient descent and peak shaving for minimizing 2n*max(self-conv)/(sum)^2.
Key innovation: IFM (Gerchberg-Saxton style) provides diverse, high-quality
starting points by targeting flat autocorrelation in the frequency domain,
complementing the gradient-based refinement.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    if n == 0:
        return np.array([])
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv_fast(a, b):
    """Linear cross-convolution via FFT."""
    M = len(a) + len(b) - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    if n == 0:
        return np.inf
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # grad of L w.r.t. a[j]: sum_k p[k] * d(c[k])/da[j]
    # d(c[k])/da[j] = a[k-j] if 0 <= k-j < n, else 0
    # So grad_L[j] = sum_k p[k] * a[k-j] (where valid)
    # This is a cross-correlation of p with a (reversed)
    M2 = n + len(p) - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    A_rev = np.fft.rfft(a[::-1], M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(A_rev * P, M2_fft)
    # conv_res[k] = sum_j a[n-1-j] * p[k-j], we want sum_k p[k]*a[k-j]
    # Let's compute directly: grad_L[j] = sum_k p[k] * a[k-j] for valid k-j
    # = (p * a_rev)[j + n - 1] where a_rev = a[::-1]
    # (p * a_rev)[m] = sum_k p[k] * a_rev[m-k] = sum_k p[k] * a[n-1-m+k]
    # For m = j + n - 1: sum_k p[k] * a[j+k] ... hmm this is getting confusing
    # Let me just do it more carefully.
    # c[k] = sum_{j: 0<=j<n, 0<=k-j<n} a[j]*a[k-j]
    # dc[k]/da[j] = a[k-j] if 0 <= k-j < n (i.e., k-j in [0,n-1])
    # grad_L[j] = sum_k p[k] * a[k-j] where 0 <= k-j <= n-1, i.e., j <= k <= j+n-1
    # Let i = k-j, then: grad_L[j] = sum_{i=0}^{n-1} p[i+j] * a[i] (where i+j < len(p))
    # = cross_correlation(p, a)[j] (appropriate definition)
    # Let's compute: for each j, sum_{i=0}^{n-1} p[j+i]*a[i]
    # This is the "valid" cross-correlation of p with a
    # = (p * a[::-1])[j+n-1:2*n-1] ... let me just compute via FFT
    M3 = 2 * n
    M3_fft = 1 << (M3 - 1).bit_length()
    P3 = np.fft.rfft(p, M3_fft)
    Ar3 = np.fft.rfft(a[::-1], M3_fft)
    corr = np.fft.irfft(P3 * Ar3, M3_fft)
    # corr[m] = sum_k p[k] * a_rev[m-k] = sum_k p[k] * a[n-1-(m-k)]
    # We want grad_L[j] = sum_i p[j+i] * a[i]
    # corr[j+n-1] = sum_k p[k] * a[n-1-(j+n-1-k)] = sum_k p[k] * a[k-j]
    # = sum_k a[k-j]*p[k] which is exactly what we want (with valid range)
    grad_L = corr[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def subgrad_score(a):
    """Exact score and subgradient at the max lag."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf, np.zeros(n)
    M = float(c.max())
    # Use all lags within a small tolerance of max for stability
    threshold = M * (1.0 - 1e-6)
    active = c >= threshold
    k_indices = np.where(active)[0]
    score = 2.0 * n * M / (S * S)
    S2 = S * S
    S3 = S2 * S
    grad_L = np.zeros(n)
    for k in k_indices:
        weight = 1.0 / len(k_indices)
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if je >= js:
            idx = k - np.arange(js, je + 1)
            grad_L[js:je + 1] += weight * a[idx]
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * M / S3)
    return score, grad


def ifm_optimize(a_init, n_iters=2000, target_flatness=1.0, t_start=None, time_limit_s=np.inf):
    """
    Iterative Fourier Method for flat autocorrelation.
    Alternates between:
    1. Frequency domain: adjust |A| to target flat power spectrum
    2. Time domain: enforce non-negativity and support
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    a = a / S  # normalize to sum=1

    N = 1 << (2 * n - 1).bit_length()

    # Target power spectrum: flat (corresponds to delta autocorrelation)
    # But we want flat LINEAR autocorrelation, not circular
    # For the linear case with zero-padding, |A|^2 = FFT(linear_acorr)
    # Target: linear_acorr[k] = 1/(2n-1) for all k
    # So target |A|^2 = FFT(target_acorr_padded)
    target_acorr = np.ones(2 * n - 1) / (2 * n - 1)
    target_acorr_full = np.zeros(N)
    target_acorr_full[:2 * n - 1] = target_acorr
    target_power = np.abs(np.fft.rfft(target_acorr_full)) ** 2
    # Ensure minimum to avoid zeroing
    target_power = np.maximum(target_power, 1e-10)

    best_score = compute_score(a)
    best_a = a.copy()
    # Relaxation parameter for averaging
    w = 0.5

    for it in range(n_iters):
        if t_start is not None and time.monotonic() - t_start > time_limit_s:
            break

        # Forward FFT
        A = np.fft.rfft(a, N)

        # Update magnitude toward target
        mag = np.abs(A)
        phase = np.angle(A)
        # Blend current and target magnitude
        new_mag = np.sqrt((1 - w) * mag ** 2 + w * target_power)
        A_new = new_mag * np.exp(1j * phase)

        # Inverse FFT
        a_raw = np.fft.irfft(A_new, N)
        a_new = a_raw[:n].copy()

        # Enforce non-negativity
        a_new = np.maximum(a_new, 0.0)

        # Renormalize
        S_new = a_new.sum()
        if S_new < 1e-12:
            a_new = 1.0 / n
        else:
            a_new = a_new / S_new

        # Over relaxation to avoid divergence
        a = (1 - 0.5) * a + 0.5 * a_new

        # Compute score periodically
        if it % 100 == 0 or it == n_iters - 1:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def ifm_with_adaptive_target(a_init, n_iters=3000, t_start=None, time_limit_s=np.inf):
    """IFM with adaptive target: gradually tighten from relaxed to flat."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        S = 1.0
    a = a / S

    N = 1 << (2 * n - 1).bit_length()

    best_score = compute_score(a)
    best_a = a.copy()

    # Phase 1: relaxed target (allow some peak), Phase 2: tighter
    n_phase1 = n_iters // 3
    n_phase2 = n_iters // 3
    n_phase3 = n_iters - n_phase1 - n_phase2

    phases = [
        (n_phase1, 2.0, 0.3),   # (iters, peak_mult, relax)
        (n_phase2, 1.5, 0.4),
        (n_phase3, 1.2, 0.5),
    ]

    for n_p, peak_mult, w in phases:
        # Target: allow peak up to peak_mult * average
        avg_val = 1.0 / (2 * n - 1)
        target_acorr = np.ones(2 * n - 1) * avg_val
        # Allow the center to be slightly higher
        center = n - 1
        target_acorr[center] = peak_mult * avg_val
        # Smooth transition
        for k in range(1, min(5, n)):
            target_acorr[center - k] = avg_val + (peak_mult - 1) * avg_val * (1 - k / 5.0)
            target_acorr[center + k] = target_acorr[center - k]
        target_acorr = np.maximum(target_acorr, 1e-10)

        target_acorr_full = np.zeros(N)
        target_acorr_full[:2 * n - 1] = target_acorr
        target_power = np.abs(np.fft.rfft(target_acorr_full)) ** 2
        target_power = np.maximum(target_power, 1e-12)

        for it in range(n_p):
            if t_start is not None and time.monotonic() - t_start > time_limit_s:
                return best_score, best_a

            A = np.fft.rfft(a, N)
            mag = np.abs(A)
            phase = np.angle(A)
            new_mag = np.sqrt((1 - w) * mag ** 2 + w * target_power)
            A_new = new_mag * np.exp(1j * phase)
            a_raw = np.fft.irfft(A_new, N)
            a_new = np.maximum(a_raw[:n], 0.0)
            S_new = a_new.sum()
            if S_new < 1e-12:
                a_new = a.copy()
            else:
                a_new = a_new / S_new
            a = 0.7 * a + 0.3 * a_new

            if it % 200 == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def adam_lse(a_init, beta_schedule, lr_init, time_limit_s, t_start, normalize=True):
    """Adam optimizer with LSE-smoothed objective and scheduled beta."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    t_global = 0

    best_a = a.copy()
    best_score = compute_score(a)

    total_steps = 0
    for beta in beta_schedule:
        # Run until next time checkpoint or a fixed number of steps
        steps = 5000
        for t in range(steps):
            t_global += 1
            if time.monotonic() - t_start > time_limit_s:
                return best_score, best_a
            if t_global % 200000 == 0:
                break

            lr_t = lr_init / (1.0 + 0.1 * t_global / 1000.0)
            score, grad = lse_score_grad(a, beta)

            if not np.all(np.isfinite(grad)):
                return best_score, best_a

            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            m_hat = m / (1.0 - b1 ** t_global)
            v_hat = v / (1.0 - b2 ** t_global)
            a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
            a = np.clip(a, 0.0, 1e10)

            if normalize:
                s = a.sum()
                if s > 1e-12:
                    a = a / s

            if t % 2000 == 0:
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()

    return best_score, best_a


def lbfgs_lse(a0, beta, maxiter=5000, maxfun=50000):
    """L-BFGS-B with LSE smoothing."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = a0.sum()
    if S < 1e-6:
        a0[:] = 1.0 / n
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        x_c = np.clip(x, 0.0, 1e10)
        sc, gr = lse_score_grad(x_c, beta)
        if not np.all(np.isfinite(gr)):
            return sc, np.zeros_like(x)
        return sc, gr

    res = minimize(obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
                   options={'ftol': 1e-16, 'gtol': 1e-13,
                            'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 30})
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = a_out.sum()
    if sc > 1e-12:
        a_out = a_out / sc
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def peak_shaving(a, n_iterations=2000, t_start=None, time_limit_s=np.inf):
    """
    Direct peak shaving: find the max lag, reduce contributing elements,
    redistribute to non-contributing elements.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()

    for it in range(n_iterations):
        if t_start is not None and time.monotonic() - t_start > time_limit_s:
            break

        c = fast_conv(a)
        k_star = int(c.argmax())
        M = c[k_star]
        score = 2.0 * n * M / (a.sum() ** 2)

        # Find contributing elements to peak
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        contrib_indices = np.arange(js, je + 1)
        if len(contrib_indices) == 0:
            break

        # Non-contributing elements
        non_contrib = np.ones(n, dtype=bool)
        non_contrib[js:je + 1] = False
        nc_indices = np.where(non_contrib)[0]

        if len(nc_indices) == 0:
            break

        # Try reducing the largest contributors
        contrib_vals = a[contrib_indices]
        order = np.argsort(contrib_vals)[::-1]
        top_k = min(10, len(order))

        improved = False
        for ki in range(top_k):
            j = contrib_indices[order[ki]]
            if a[j] < 1e-10:
                continue

            a_old = a[j]
            # Try several reduction factors
            for frac in [0.9, 0.7, 0.5]:
                a[j] = a_old * frac
                # Redistribute to random non-contributing elements
                n_redist = min(20, len(nc_indices))
                targets = nc_indices[np.random.choice(len(nc_indices), n_redist, replace=False)]
                removed = a_old - a[j]
                a[targets] += removed / len(targets)
                a = np.maximum(a, 0.0)
                sc = compute_score(a)
                if sc < score - 1e-10:
                    score = sc
                    improved = True
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                    break
            if not improved:
                a[j] = a_old

        if not improved:
            # Try increasing non-contributing elements
            n_try = min(5, len(nc_indices))
            for _ in range(n_try):
                j = nc_indices[np.random.randint(len(nc_indices))]
                a_old = a[j]
                for mult in [1.1, 1.3, 1.5]:
                    a[j] = a_old * mult
                    a = np.maximum(a, 0.0)
                    sc = compute_score(a)
                    if sc < score - 1e-10:
                        score = sc
                        improved = True
                        if sc < best_score:
                            best_score = sc
                            best_a = a.copy()
                        break
                if not improved:
                    a[j] = a_old
            if not improved:
                # Stuck, try a random perturbation
                a += 0.01 * np.random.standard_normal(n)
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-12:
                    a = a / s

    return best_score, best_a


def generate_diverse_inits(rng, n, n_inits=10):
    """Generate diverse initializations."""
    inits = []
    x = np.arange(n) / n

    # Uniform
    inits.append(np.ones(n) / n)

    # Linear ramp (left-heavy)
    ramp = np.linspace(n, 1, n)
    inits.append(ramp / ramp.sum())

    # Linear ramp (right-heavy)
    ramp2 = np.linspace(1, n, n)
    inits.append(ramp2 / ramp2.sum())

    # Anti-palindrome: high on left, low on right, with oscillation
    freq = 1.0 + 4.0 * rng.random()
    anti = np.maximum(0, 0.5 + 0.5 * np.sin(np.pi * x + rng.uniform(0, 2 * np.pi)))
    anti += 0.1 * np.random.random(n)
    inits.append(anti / anti.sum())

    # Two triangular bumps (asymmetric)
    c1 = 0.2 + 0.1 * rng.random()
    c2 = 0.6 + 0.2 * rng.random()
    w1 = 0.1 + 0.1 * rng.random()
    w2 = 0.1 + 0.1 * rng.random()
    b = np.exp(-((x - c1) / w1) ** 2) + 0.7 * np.exp(-((x - c2) / w2) ** 2)
    inits.append(b / b.sum())

    # Frequency-domain random (flat spectrum target)
    N = 1 << (2 * n - 1).bit_length()
    for _ in range(3):
        phase = rng.uniform(0, 2 * np.pi, N // 2 + 1)
        mag = np.ones(N // 2 + 1)
        mag[0] = n  # DC
        A = mag * np.exp(1j * phase)
        a_raw = np.fft.irfft(A, N)[:n]
        a_raw = np.maximum(a_raw, 0)
        s = a_raw.sum()
        if s > 1e-10:
            inits.append(a_raw / s)

    # Random smooth
    r = np.abs(rng.standard_normal(n)) + 0.3
    k = max(3, n // 30)
    kernel = np.exp(-0.5 * (np.arange(-k, k + 1) / (k // 3 + 1)) ** 2)
    kernel /= kernel.sum()
    r_smooth = np.convolve(r, kernel, mode='same')
    inits.append(r_smooth / r_smooth.sum())

    # Cosine-modulated
    f = 1 + 3 * rng.random()
    p = 2 * np.pi * rng.random()
    mod = 0.5 + 0.5 * np.cos(2 * np.pi * f * x + p)
    mod = np.maximum(mod, 0.01)
    inits.append(mod / mod.sum())

    # Quadratic anti-symmetric
    q = x * (1 - x)
    q_anti = q + 0.3 * np.sin(2 * np.pi * 2 * x)
    q_anti = np.maximum(q_anti, 0.01)
    inits.append(q_anti / q_anti.sum())

    return inits[:n_inits]


def resample_sequence(a, new_n):
    """Resample to new length via linear interpolation."""
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return np.clip(a_new, 0.0, 1000.0)

    t_start = time.monotonic()
    effective_budget = max(60.0, float(budget_s)) - 30.0

        if 'height_sequence_1' in globals() and height_sequence_1 is not None:

            if p.size >= 32:

                if ps > 1e-12:

    except Exception:
        pass

    best_score = np.inf
    best_a = None

    def try_update(sc, a, label=""):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = np.clip(a, 0.0, 1000.0).copy()
            elapsed = time.monotonic() - t_start
            print(f"  [BEST] {sc:.8f} | {label} | n={len(a)} | t={elapsed:.0f}s")
            return True
        return False

    # Determine working length
    if parent is not None:
        n_work = len(parent)
        sc_parent = compute_score(parent)
        print(f"Parent: n={n_work}, score={sc_parent:.8f}")
        try_update(sc_parent, parent, "parent")
    else:
        n_work = 4096
        print(f"No parent, using n={n_work}")

    print(f"\n{'=' * 60}")
    print(f"Starting optimization with n={n_work}, budget={effective_budget:.0f}s")
    print(f"{'=' * 60}")

    # ==========================================
    # Phase 1: IFM from diverse initializations (30% time)
    # ==========================================
    p1_time = effective_budget * 0.30
    print(f"\nPhase 1: IFM ({p1_time:.0f}s budget)")

    inits = generate_diverse_inits(rng, n_work, n_inits=8)
    if parent is not None:
        inits.append(parent.copy())

    candidates = []
    time_per_ifm = p1_time / (len(inits) + 1)

    for i, init in enumerate(inits):
        remaining = t_start + p1_time - time.monotonic()
        if remaining < 5:
            break
        t0 = time.monotonic()

        # Run IFM with adaptive target
        sc_ifm, a_ifm = ifm_with_adaptive_target(
            init, n_iters=3000, t_start=t_start, time_limit_s=min(time_per_ifm * 1.5, remaining - 2)
        )
        candidates.append((sc_ifm, a_ifm))
        try_update(sc_ifm, a_ifm, f"IFM init{i}")
        print(f"  IFM {i}: {sc_ifm:.8f} ({time.monotonic()-t0:.1f}s)")

    # Sort candidates
    candidates.sort(key=lambda x: x[0])
    print(f"  Best IFM: {candidates[0][0]:.8f}")

    # ==========================================
    # Phase 2: LSE-Adam refinement (35% time)
    # ==========================================
    p2_time = effective_budget * 0.35
    print(f"\nPhase 2: LSE-Adam ({p2_time:.0f}s budget)")

    # Start from best IFM candidate
    start_a = candidates[0][1].copy()
    S = start_a.sum()
    if S > 1e-12:
        start_a = start_a / S

    # Beta schedule: geometric increase
    beta_schedule = [20, 50, 100, 300, 1000, 3000, 8000, 20000]
    lr = 0.01

    # Split time among beta values, spending more time at higher beta
    time_per_beta = p2_time / len(beta_schedule)
    for bi, beta in enumerate(beta_schedule):
        remaining = t_start + p2_time - time.monotonic()
        if remaining < 3:
            break
        t0 = time.monotonic()
        # Decrease learning rate as beta increases
        lr_b = lr * (0.5 ** bi)
        sc, a_opt = adam_lse(start_a, [beta], lr_init=lr_b,
                             time_limit_s=min(time_per_beta * 1.3, remaining - 1),
                             t_start=t_start)
        start_a = a_opt
        try_update(sc, a_opt, f"Adam beta={beta}")
        print(f"  Adam beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # Also try from second best IFM candidate
    if len(candidates) > 1:
        remaining = t_start + p2_time - time.monotonic()
        if remaining > 10:
            start2 = candidates[1][1].copy()
            S2 = start2.sum()
            if S2 > 1e-12:
                start2 = start2 / S2
            sc2, a2 = adam_lse(start2, [100, 500, 2000], lr_init=0.005,
                               time_limit_s=min(remaining * 0.5, p2_time * 0.15),
                               t_start=t_start)
            try_update(sc2, a2, "Adam from 2nd IFM")
            if sc2 < compute_score(start_a):
                start_a = a2

    # ==========================================
    # Phase 3: L-BFGS polish (15% time)
    # ==========================================
    p3_time = effective_budget * 0.15
    print(f"\nPhase 3: L-BFGS ({p3_time:.0f}s budget)")

    for beta in [500, 2000, 8000, 20000]:
        remaining = t_start + p3_time - time.monotonic()
        if remaining < 5:
            break
        t0 = time.monotonic()
        sc, a_opt = lbfgs_lse(best_a, beta, maxiter=3000, maxfun=40000)
        try_update(sc, a_opt, f"LBFGS beta={beta}")
        print(f"  LBFGS beta={beta}: {sc:.8f} ({time.monotonic()-t0:.1f}s)")
        if sc < best_score:
            best_a = a_opt

    # ==========================================
    # Phase 4: Peak shaving (10% time)
    # ==========================================
    p4_time = effective_budget * 0.10
    print(f"\nPhase 4: Peak Shaving ({p4_time:.0f}s budget)")

    remaining = t_start + p4_time - time.monotonic()
    if remaining > 5:
        t0 = time.monotonic()
        sc, a_opt = peak_shaving(best_a, n_iterations=10000,
                                  t_start=t_start, time_limit_s=min(p4_time, remaining - 2))
        try_update(sc, a_opt, "Peak shaving")
        print(f"  Peak shaving: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

    # ==========================================
    # Phase 5: IFM + gradient alternating (5% time)
    # ==========================================
    p5_time = effective_budget * 0.05
    print(f"\nPhase 5: IFM+Grad alternating ({p5_time:.0f}s budget)")

    remaining = t_start + p5_time - time.monotonic()
    if remaining > 5:
        a_cur = best_a.copy()
        for cycle in range(5):
            remaining = t_start + p5_time - time.monotonic()
            if remaining < 3:
                break
            # IFM step
            sc_ifm, a_ifm = ifm_with_adaptive_target(
                a_cur, n_iters=500, t_start=t_start, time_limit_s=min(remaining * 0.4, p5_time / 10)
            )
            # Gradient step
            remaining = t_start + p5_time - time.monotonic()
            if remaining < 3:
                break
            sc_adam, a_adam = adam_lse(
                a_ifm, [500, 2000], lr_init=0.003,
                time_limit_s=min(remaining * 0.5, p5_time / 10), t_start=t_start
            )
            a_cur = a_adam
            try_update(sc_adam, a_adam, f"IFM+Grad cycle{cycle}")

    # ==========================================
    # Phase 6: Final coordination (5% time)
    # ==========================================
    p6_time = effective_budget * 0.05
    print(f"\nPhase 6: Final ({p6_time:.0f}s budget)")

    remaining = t_start + p6_time - time.monotonic()
    if remaining > 5:
        # Final high-beta LSE Adam
        t0 = time.monotonic()
        sc, a_opt = adam_lse(best_a, [10000, 30000], lr_init=0.001,
                             time_limit_s=min(remaining * 0.5, p6_time * 0.5), t_start=t_start)
        try_update(sc, a_opt, "Final Adam")
        print(f"  Final Adam: {sc:.8f} ({time.monotonic()-t0:.1f}s)")

        # Final peak shaving
        remaining = t_start + p6_time - time.monotonic()
        if remaining > 3:
            sc, a_opt = peak_shaving(best_a, n_iterations=5000,
                                      t_start=t_start, time_limit_s=min(remaining - 2, p6_time * 0.4))
            try_update(sc, a_opt, "Final peak shave")

    # ==========================================
    # Final verification
    # ==========================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    if verified < best_score:
        best_score = verified

    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
