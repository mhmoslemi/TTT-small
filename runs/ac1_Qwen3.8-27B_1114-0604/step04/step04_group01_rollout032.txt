"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using:
- Convolution flattening via smooth quadratic objective (gradient descent with momentum)
- LSE-smoothed max with L-BFGS-B using CORRECT full gradient (fixes asymmetry bug in previous)
- Subgradient pressure polish
- Diverse parametric and random initializations
- Multiple restarts with perturbation

import numpy as np
import time
from scipy.optimize import minimize

PEN_SUM = 1e-4


def next_power_of_two(m):
    m = int(max(1, m))
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64)
    n = int(a.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    M = 2 * n - 1
    N = next_power_of_two(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)[:M]
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def normalize_a(a, target=1.0):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1e6, neginf=0.0)
    a = np.clip(a, 0.0, None)
    S = float(a.sum())
    if S < 1e-12:
        return np.full(a.size, target / float(a.size), dtype=np.float64)
    a = a * (target / S)
    return a


def resample_sequence(a, n):
    a = np.nan_to_num(np.asarray(a, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
    m = int(a.size)
    if m == 0:
        return np.ones(int(n), dtype=np.float64)
    if m == int(n):
        return a.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, int(n))
    return np.interp(x_new, x_old, np.maximum(a, 0)).astype(np.float64)


def exact_score_a(a):
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    cmax = float(np.max(c))
    if cmax < 0.0:
        cmax = 0.0
    return float(2.0 * n * cmax / (S * S))


def full_corr(a, w):
    """
    Compute h[j] = sum_k w[k] * (a[k-j] + a[j-k]) for j=0,...,n-1.
    a has length n, w has length m.
    """
    n = len(a)
    m = len(w)
    total_len = n + m - 1
    N = next_power_of_two(total_len)
    a_rev = a[::-1]
    A = np.fft.rfft(a, N)
    A_rev = np.fft.rfft(a_rev, N)
    W = np.fft.rfft(w, N)

    conv_wa = np.fft.irfft(W * A, N)
    conv_wa_rev = np.fft.irfft(W * A_rev, N)

    h2 = conv_wa[:n]
    h1 = conv_wa_rev[n - 1:n - 1 + n]

    out = h1 + h2
    return np.nan_to_num(out, nan=0.0, posinf=0.0, neginf=0.0)


def quad_obj_grad(a):
    """
    Quadratic flattening objective: F = sum_k (c[k] - avg)^2 + PEN*(S-1)^2
    where c = conv(a,a), avg = 1/(2n-1), S = sum(a).
    Returns (value, gradient).
    """
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = fast_self_conv(a)
    avg = 1.0 / (2 * n - 1)
    d = c - avg
    val = float(np.dot(d, d)) + PEN_SUM * (S - 1.0) ** 2

    # Gradient: 2 * sum_k d[k] * (a[k-i] + a[i-k]) / S^2 factor not needed here
    # Since we're not normalizing by S^2 in this formulation
    # Actually c[k] = sum_i a_i * a[k-i] directly, so:
    # dF/da_i = 2 * sum_k d[k] * dc[k]/da_i + 2*PEN*(S-1)
    # dc[k]/da_i = a[k-i] + a[i-k]
    # So dF/da_i = 2 * sum_k d[k]*(a[k-i]+a[i-k]) + 2*PEN*(S-1)

    grad_corr = full_corr(a, d)
    grad = 2.0 * grad_corr + 2.0 * PEN_SUM * (S - 1.0)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    return val, grad


def lse_obj_grad(a, beta):
    """
    LSE-smoothed objective with CORRECT full gradient.
    F = 2 * LSE_beta(d) + PEN * (S-1)^2
    where d[k] = n * c[k] / S^2.
    """
    a = np.asarray(a, dtype=np.float64)
    n = len(a)
    S = float(a.sum())
    if S < 1e-8:
        return 1e10, np.zeros(n)

    c = fast_self_conv(a)
    d = (float(n) * c) / (S * S)
    d = np.maximum(d, 0.0)
    dmax = float(np.max(d))
    if not np.isfinite(dmax) or dmax <= 0.0:
        return 1e10, np.zeros(n)

    z = beta * (d - dmax)
    e = np.exp(z)
    sum_e = float(e.sum())
    if sum_e <= 0.0 or not np.isfinite(sum_e):
        return 1e10, np.zeros(n)
    w = e / sum_e

    L = dmax + np.log(sum_e) / beta
    LDS = float(np.dot(w, d))
    score = 2.0 * L + PEN_SUM * (S - 1.0) ** 2

    # Full correct gradient:
    # dF/da_i = 2*n/S^2 * sum_k w[k]*(a[k-i]+a[i-k]) - 4*n*dot(w,d)/S^2 * S
    # Wait let me re-derive:
    # d_k = n * c_k / S^2
    # dd_k/da_i = n*(a[k-i]+a[i-k])/S^2 - 2*n*c_k/S^3
    # dLSE/da_i = sum_k w_k * dd_k/da_i
    #   = n/S^2 * sum_k w_k*(a[k-i]+a[i-k]) - 2n/S^3 * sum_k w_k*c_k
    #   = n/S^2 * h[i] - 2n/S^3 * (S^2/n) * dot(w,d)
    #   = n/S^2 * h[i] - 2/S * dot(w,d)
    # dF/da_i = 2*n/S^2 * h[i] - 4/S * dot(w,d) + 2*PEN*(S-1)

    h = full_corr(a, w)
    grad = (2.0 * float(n) / (S * S)) * h - (4.0 / S) * LDS + 2.0 * PEN_SUM * (S - 1.0)
    grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
    grad = np.clip(grad, -1e6, 1e6)
    return float(score), grad


def run_quad_opt(a0, deadline, max_iters=50000):
    """
    Gradient descent with momentum on the quadratic flattening objective.
    Works in the space a >= 0, sum(a) = 1.
    """
    a = normalize_a(a0)
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)

    # Convert to log-space for better conditioning
    # Actually, let's just do projected gradient in original space
    momentum = np.zeros(n, dtype=np.float64)
    mu = 0.9  # momentum coefficient
    lr = 0.01  # initial learning rate
    val, grad = quad_obj_grad(a)
    cur_val = val

    for it in range(int(max_iters)):
        if time.monotonic() > deadline:
            break

        # Momentum update
        momentum = mu * momentum - lr * grad
        a_new = a + momentum

        # Project to non-negative and normalize
        a_new = np.maximum(a_new, 0.0)
        S_new = float(a_new.sum())
        if S_new < 1e-12:
            a_new = np.ones(n) / n
        else:
            a_new = a_new / S_new

        # Evaluate
        val_new, grad_new = quad_obj_grad(a_new)

        if np.isfinite(val_new) and val_new < cur_val * 1.01:
            # Accept step
            a = a_new
            grad = grad_new
            cur_val = val_new

            # Check exact score periodically
            if it % 200 == 0:
                sc = exact_score_a(a)
                if sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()

            # Adaptive learning rate
            if it % 500 == 0 and it > 0:
                lr *= 0.95
        else:
            # Reject step, reduce learning rate
            lr *= 0.5
            momentum *= 0.5
            if lr < 1e-10:
                break

    # Final check
    sc = exact_score_a(a)
    if sc < best_sc:
        best_sc = sc
        best_a = a.copy()

    return float(best_sc), best_a


def run_lbfgs_lse(a0, beta, deadline, maxfun=2000):
    """Run L-BFGS-B on the LSE objective."""
    a = normalize_a(a0)
    n = len(a)

    if deadline is not None:
        rem = float(deadline - time.monotonic())
        if rem < 10.0:
            return exact_score_a(a), a
        # Estimate time per eval
        est_time = 0.002 * max(1.0, n / 8192.0)
        cap = int((0.8 * rem) / max(est_time, 1e-5))
        maxfun = int(min(int(maxfun), max(64, cap)))

    maxiter = max(50, maxfun // 2)
    bounds = [(0.0, 10.0)] * n
    opts = {
        "ftol": 1e-15,
        "gtol": 1e-10,
        "maxiter": int(maxiter),
        "maxfun": int(maxfun),
        "maxls": 40,
    }

    def obj_func(x, b):
        v, g = lse_obj_grad(x, b)
        return v, g

    try:
        res = minimize(
            obj_func, a, method="L-BFGS-B", jac=True,
            args=(float(beta),), bounds=bounds, options=opts
        )
    except Exception:
        return exact_score_a(a), a

    if getattr(res, "x", None) is not None:
        a_out = normalize_a(res.x)
    else:
        a_out = a

    sc = exact_score_a(a_out)
    if not np.isfinite(sc) or sc > 1e8:
        a_out = a
        sc = exact_score_a(a)
    return float(sc), a_out


def subgrad_polish(a, max_iters=30, deadline=None):
    """Subgradient pressure moves on the exact max."""
    a = normalize_a(a)
    best_a = a.copy()
    best_sc = exact_score_a(a)
    cur_sc = best_sc

    for it in range(int(max_iters)):
        if deadline is not None and time.monotonic() > float(deadline):
            break

        c = fast_self_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        d = (float(len(a)) * c) / (S * S)
        dmax = float(np.max(d))
        if not np.isfinite(dmax) or dmax <= 0:
            break

        cur_a = None
        cur_best = cur_sc

        # Try different numbers of active positions
        m = len(d)
        for k_active in (1, 2, 3, 5, 8):
            if k_active >= m:
                w = np.ones(m) / m
            else:
                idx = np.argpartition(d, m - k_active)[m - k_active:]
                w = np.zeros(m)
                w[idx] = 1.0
                w /= float(k_active)

            h = full_corr(a, w)
            hmax = float(np.max(h))
            if hmax < 1e-12:
                continue

            for mult in (0.01, 0.03, 0.08, 0.2, 0.5):
                if deadline is not None and time.monotonic() > float(deadline):
                    break
                eta = mult / hmax
                q = a * np.exp(-eta * h)
                q = normalize_a(q)
                sc = exact_score_a(q)
                if not np.isfinite(sc):
                    continue
                if sc < best_sc:
                    best_sc = sc
                    best_a = q.copy()
                if sc < cur_best:
                    cur_best = sc
                    cur_a = q.copy()

        if cur_a is None:
            break
        a = cur_a
        cur_sc = cur_best

    return float(best_sc), best_a


def generate_inits(n, rng, parent=None):
    """Generate diverse initial sequences of length n."""
    inits = []
    n = int(n)
    x = np.linspace(0.0, 1.0, n)

    def add(arr):
        arr = np.nan_to_num(np.asarray(arr, dtype=np.float64).ravel(), nan=0.0, posinf=0.0, neginf=0.0)
        if arr.size != n:
            arr = resample_sequence(arr, n)
        arr = np.clip(arr, 0.0, None)
        S = float(arr.sum())
        if S > 1e-12:
            inits.append(arr / S)
        else:
            inits.append(np.ones(n) / n)

    # Parent-based
    if parent is not None and parent.size > 0:
        pr = resample_sequence(parent, n)
        add(pr)
        add(pr[::-1])

    # Uniform
    add(np.ones(n))

    # Bulb shapes (high middle, low edges)
    for alpha in (0.5, 1.0, 2.0, 3.0, 5.0):
        add(np.exp(-alpha * (2 * x - 1) ** 2))
    for p in (0.5, 1.0, 2.0, 4.0, 8.0):
        add((1.0 - np.abs(2 * x - 1)) ** p + 0.01)

    # Cosine bulbs
    for k in (1, 2, 3, 4, 6):
        add(0.5 + 0.5 * np.cos(np.pi * x) ** k)
        add(0.3 + 0.7 * np.cos(np.pi * x * 0.5) ** 2)

    # Frequency domain: low-pass shapes
    Nf = 1 << (n - 1).bit_length()
    for _ in range(8):
        freqs = np.fft.rfftfreq(Nf) * Nf
        gamma = float(rng.uniform(1.0, 4.0))
        power = 1.0 / ((np.abs(freqs) + 1.0) ** gamma)
        power[0] = 0.0
        phase = rng.uniform(-np.pi, np.pi, size=len(power))
        spec = power * np.exp(1j * phase)
        spec[0] = 0.0
        sig = np.fft.irfft(spec, Nf)[:n]
        sig = sig - float(sig.min())
        add(0.3 + sig)

    # Random cosine sums
    for _ in range(6):
        vals = np.ones(n)
        ncomp = int(rng.integers(2, 8))
        for _ in range(ncomp):
            f = int(rng.integers(1, max(2, n // 4)))
            amp = float(rng.uniform(0.05, 0.4))
            ph = float(rng.uniform(-np.pi, np.pi))
            vals += amp * np.cos(2 * np.pi * f * x + ph)
        add(vals)

    # Binary-like (subset)
    for frac in (0.4, 0.6, 0.8):
        mask = rng.random(n) < frac
        vals = np.where(mask, 1.0, 0.05)
        add(vals)

    # Block structure
    for nb in (4, 8, 16, 32, 64):
        if nb > n:
            nb = max(2, n // 2)
        block = 0.2 + 0.8 * rng.random(nb)
        # Make it bulb-like: higher in middle blocks
        block *= 0.5 + 0.5 * np.exp(-((np.arange(nb) / max(1, nb) - 0.5) ** 2) / 0.1)
        per = max(1, n // nb)
        arr = np.repeat(block, per)
        if arr.size < n:
            arr = np.concatenate([arr, np.repeat(block, n - arr.size)])
        add(arr[:n])

    # Pure random
    for _ in range(4):
        add(0.1 + 0.9 * rng.random(n))

    # Smooth random
    for _ in range(4):
        raw = rng.standard_normal(n)
        # Smooth with Gaussian kernel
        k = max(3, n // 50)
        if k % 2 == 0:
            k += 1
        kern = np.exp(-(np.arange(k) - k // 2) ** 2 / (2.0 * (k // 4 + 1) ** 2))
        kern /= kern.sum()
        smoothed = np.convolve(raw, kern, mode='same')
        add(0.5 + 0.5 * smoothed / (np.abs(smoothed).max() + 1e-10))

    return inits
