import numpy as np
from scipy.optimize import minimize, linprog
import time

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -100))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)
    
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def solve_lp(f, M, n):
    """
    Solve: maximize sum(b) subject to f*b <= M (elementwise), b >= 0.
    Returns the optimal b.
    """
    # A matrix: A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    # This is a Toeplitz-like matrix
    # For efficiency, we can use the structure
    
    # Build A as a sparse matrix
    from scipy.sparse import csr_matrix, kron, diags
    
    # Actually, for n ~ 7000, a dense matrix is 14000 * 7000 = ~10^8 elements = ~800MB
    # That's too much. Let's use sparse.
    
    # A[k, j] = f[k-j] for max(0, k-n+1) <= j <= min(k, n-1)
    # For row k: non-zeros at columns max(0, k-n+1) to min(k, n-1)
    # Values: f[k - j] for j in that range
    
    row_indices = []
    col_indices = []
    data = []
    
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            js = np.arange(j_start, j_end + 1)
            row_indices.extend([k] * (j_end - j_start + 1))
            col_indices.extend(js.tolist())
            data.extend(f[k - js].tolist())
    
    A = csr_matrix((data, (row_indices, col_indices)), shape=(2*n-1, n))
    
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x
    else:
        # Fallback: return uniform
        return np.ones(n) / n

def lp_refinement(f, n_iters=100, t_init=0.1, t_min=0.01, verbose=True):

def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()

    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def nesterov_subgrad(a, n_iters=3000, f_lb=1.0):
    """Nesterov-accelerated subgradient descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]
        
        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.3)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new
        
        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()
    
    return best_a

"""
Multi-strategy optimizer for step function peak minimization.
Combines LP-based refinement (key innovation), L-BFGS on LSE-smoothed objective,
Nesterov subgradient, and targeted peak reduction.
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(np.maximum(scaled, -100))
    p = exp_scaled / exp_scaled.sum()
    L = c_max + float(np.log(exp_scaled.sum())) / beta
    score = 2.0 * n * L / (S * S)
    
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def build_lp_matrix(n):
    """Build the sparse constraint matrix for the LP.
    A[k, j] = 1 if 0 <= k-j <= n-1 (the actual value is filled from f each time).
    Returns (row_indices, col_indices) for the sparsity pattern.
    """
    row_indices = []
    col_indices = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_end >= j_start:
            js = np.arange(j_start, j_end + 1)
            row_indices.extend([k] * (j_end - j_start + 1))
            col_indices.extend(js.tolist())
    return np.array(row_indices), np.array(col_indices)


def solve_lp(f, M, n, row_idx, col_idx):
    """
    Solve: maximize sum(b) subject to f*b <= M (elementwise), b >= 0.
    Uses the precomputed sparsity pattern.
    """
    # Fill in the values
    data = np.empty(len(row_idx))
    rows = row_idx
    cols = col_idx
    # For each entry (k, j), the value is f[k-j]
    vals = f[rows - cols]
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 60})
    
    if res.status == 0:
        return np.maximum(0, res.x)
    else:
        return np.ones(n) / n


def lp_refinement(f, n_iters=50, t_init=0.1, t_min=0.005, verbose=True, time_limit=600):
    """
    LP-based refinement of the sequence.
    """
    t0 = time.monotonic()
    n = len(f)
    f = f.astype(np.float64).copy()
    S = f.sum()
    if S < 1e-12:
        return f
    f = f / S
    
    # Precompute sparsity pattern
    row_idx, col_idx = build_lp_matrix(n)
    
    best_f = f.copy()
    best_sc = compute_score(f)
    
    for it in range(n_iters):
        if time.monotonic() - t0 > time_limit:
            break
        
        c = fast_conv(f)
        M = float(c.max())
        
        g = solve_lp(f, M, n, row_idx, col_idx)
        Sg = g.sum()
        if Sg < 1e-12:
            break
        g = g / Sg
        
        # Geometric decay of t
        t = t_init * (t_min / t_init) ** (it / max(1, n_iters - 1))
        
        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0, 1000)
        S_new = f_new.sum()
        if S_new < 1e-12:
            break
        f_new = f_new / S_new
        
        sc_new = compute_score(f_new)
        sc_old = compute_score(f)
        
        if sc_new < best_sc:
            best_sc = sc_new
            best_f = f_new.copy()
        
        if sc_new <= sc_old:
            f = f_new
        else:
            # Try smaller t values
            for t_try in [t * 0.5, t * 0.1, t * 0.01]:
                f_try = (1 - t_try) * f + t_try * g
                f_try = np.clip(f_try, 0, 1000)
                S_try = f_try.sum()
                if S_try < 1e-12:
                    continue
                f_try = f_try / S_try
                sc_try = compute_score(f_try)
                if sc_try < sc_old:
                    f = f_try
                    break
        
        if verbose and it % 10 == 0:
            print(f"    LP iter {it}: score={compute_score(f):.10f}, best={best_sc:.10f}, t={time.monotonic()-t0:.0f}s")
    
    return best_f


def run_lbfgs(a0, beta, maxiter=50000, maxfun=300000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-18, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 200}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def nesterov_subgrad(a, n_iters=3000, f_lb=1.0):
    """Nesterov-accelerated subgradient descent."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    
    y = a.copy()
    t = 1.0
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for it in range(n_iters):
        c = fast_conv(y)
        Sy = float(y.sum())
        k_star = int(np.argmax(c))
        Mak = float(c[k_star])
        sc = 2.0 * n * Mak / (Sy * Sy)
        if sc < best_sc:
            best_sc = sc
            best_a = y.copy()
        
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            idx = np.arange(j_start, j_end + 1)
            sg_M[idx] = 2.0 * y[k_star - idx]
        
        S2 = Sy * Sy
        S3 = S2 * Sy
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * Mak / S3)
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        step = (sc - f_lb) / (gnorm * gnorm + 1e-30)
        step = min(step, 0.3)
        
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / t_new
        t = t_new
        
        y_new = y - step * grad
        a_new = y_new + theta * (y_new - a)
        
        a = np.clip(a_new, 0.0, 1000.0)
        Sa = a.sum()
        if Sa < 1e-12:
            break
        a = a / Sa
        y = a.copy()
    
    return best_a


def targeted_peak_reduction(a, n_sweeps=10, n_top=50):
    """Targeted coordinate descent on peak contributors."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    
    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            break
        
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local
        
        for j in top_global:
            a_old_j = a[j]
            if a_old_j < 1e-10:
                continue
            best_val = a_old_j
            best_local_sc = best_sc
            for frac in [0.0, 0.2, 0.4, 0.6, 0.8, 0.95]:
                new_val = a_old_j * frac
                a[j] = new_val
                reduction = a_old_j - new_val
                if reduction > 1e-15:
                    bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
                    per = reduction / len(bottom_local)
                    for li in bottom_local:
                        a[j_start + li] += per
                S = a.sum()
                if S > 1e-12:
                    a = a / S
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = new_val
            a[j] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S
        
        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a


def perturbed_restart(a, rng, scale=0.03):
    """Generate a perturbed version."""
    n = len(a)
    strategy = int(rng.integers(0, 5))
    x = np.arange(n, dtype=np.float64) / n
    
    if strategy == 0:
        a_p = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 1:
        freq = 1.0 + 10.0 * rng.random()
        phase = 2.0 * np.pi * rng.random()
        amp = scale * (1.0 + rng.random())
        a_p = a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
    elif strategy == 2:
        n_blocks = 3 + int(rng.integers(0, 8))
        block_size = max(1, n // n_blocks)
        a_p = a.copy()
        for b in range(min(n_blocks, n // block_size)):
            s = b * block_size
            e = min(s + block_size, n)
            a_p[s:e] *= (1.0 + scale * 2.0 * (rng.random() - 0.5))
    elif strategy == 3:
        # Targeted: reduce peak contributors
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        a_p = a.copy()
        seg = a_p[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        top_count = max(5, n // 50)
        top_local = np.argsort(contrib)[::-1][:top_count]
        for li in top_local:
            a_p[j_start + li] *= (1.0 - 0.1 * rng.random())
        S = a_p.sum()
        if S > 1e-12:
            deficit = a.sum() - S
            bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
            per = deficit / len(bottom_local)
            for li in bottom_local:
                a_p[j_start + li] += per
    else:
        n_sel = max(10, n // 4)
        sel = rng.choice(n, size=n_sel, replace=False)
        a_p = a.copy()
        a_p[sel] *= (1.0 + scale * 3.0 * (rng.random(n_sel) - 0.5))
    
    a_p = np.clip(a_p, 0.0, 1000.0)
    S = a_p.sum()
    if S > 1e-12:
        a_p = a_p / S
    return a_p
