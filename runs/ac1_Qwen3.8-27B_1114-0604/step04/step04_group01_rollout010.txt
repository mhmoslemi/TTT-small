**Targeted Peak Reduction Algorithm:**

Given a sequence a with score S, and the convolution c = conv(a,a):
1. Find the argmax of c, say at position k*.
2. The value c_{k*} = sum_i a_i * a_{k*-i}.
3. We want to reduce c_{k*} while maintaining sum(a) = 1 and a ≥ 0.
4. Strategy: Find a position p where the convolution is low, and transfer mass from positions contributing to the peak to position p.

More specifically:
- The "contribution" of position i to c_{k*} is a_i * a_{k*-i} (if k*-i is in range).
- If we move a small amount δ from position i to position p, the change in c_{k*} is approximately:
  -δ * a_{k*-i} + a_i * (change in a_{k*-p} due to the move)
  - The second term is of second order, so to first order: Δc_{k*} ≈ -δ * a_{k*-i}
- We want to maximize a_{k*-i} / a_i (to get the most reduction per unit of mass moved), but also we need to check that moving to p doesn't increase other peaks.

A simpler heuristic:
- For the peak lag k*, the "pressure" on position m is a_{k*-m} (the amount by which reducing a_m would help).
- Move mass from positions with high pressure to positions with low "total pressure" (considering all peak lags).

        beta_schedule = [50, 100, 200, 500, 1000, 2000, 5000, 10000, 20000, 50000]
    
    a = normalize_nonneg(a)
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    
    for beta in beta_schedule:
        if time.monotonic() > deadline:
            break
        
        # Adaptive step size based on gradient magnitude
        F, grad = lse_objective_grad(a, beta)
        gnorm = float(np.max(np.abs(grad)))
        if gnorm < 1e-12:
            continue
        
        # Initial step: try to reduce F by a fixed fraction
        # For L-smooth function, step = 1/L gives guaranteed convergence
        # L ≈ beta * 2n (rough upper bound on Hessian norm)
        lr = 1.0 / (beta * float(n) * gnorm) * 0.5
        lr = min(lr, 0.1 / gnorm)  # cap the step
        
        for it in range(max_inner_iters):
            if time.monotonic() > deadline:
                break
            
            F, grad = lse_objective_grad(a, beta)
            gnorm = float(np.max(np.abs(grad)))
            if gnorm < 1e-10:
                break
            
            # Backtracking line search
            improved = False
            lr_try = lr
            for backtrack in range(15):
                a_new = simplex_project(a - lr_try * grad)
                F_new, _ = lse_objective_grad(a_new, beta)
                
                # Armijo condition
                if F_new <= F - 1e-4 * lr_try * gnorm * gnorm:
                    a = a_new
                    improved = True
                    # Check exact score
                    sc = exact_score(a)
                    if sc < best_sc:
                        best_sc = sc
                        best_a = a.copy()
                    lr = lr_try * 1.05  # slight increase if it worked
                    break
                lr_try *= 0.5
            
            if not improved:
                lr = lr_try * 0.5
                if lr < 1e-12:
                    break
        
        # Update best
        sc = exact_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc

def mirror_descent_step(a, grad, eta):
    """Exponential weights update on the simplex."""
    log_a = np.log(a + 1e-300)
    log_a -= float(np.max(log_a))  # stability
    log_a -= eta * grad
    a_new = np.exp(log_a)
    return normalize_nonneg(a_new)

def targeted_reduction(a, deadline, num_iter=100):
    """
    Convolution-feature-guided targeted reduction.
    Identify peaks and move mass from contributing positions to safe positions.
    """
    a = normalize_nonneg(a.copy())
    n = len(a)
    best_a = a.copy()
    best_sc = exact_score(a)
    
    for it in range(num_iter):
        if time.monotonic() > deadline:
            break
        
        c = fast_self_conv(a)
        cmax = float(np.max(c))
        S = float(a.sum())
        
        # Find top peak(s)
        peak_idx = int(np.argmax(c))
        
        # Compute pressure: how much each position contributes to the peak
        # c[peak_idx] = sum_i a_i * a_{peak_idx - i}
        # pressure[m] = a_{peak_idx - m} (if valid)
        pressure = np.zeros(n)
        for m in range(n):
            j = peak_idx - m
            if 0 <= j < n:
                pressure[m] = a[j]
        
        pmax = float(pressure.max())
        if pmax < 1e-12:
            break
        
        # Compute "safety" of each position as a target:
        # A position p is safe if moving mass there doesn't increase the peak much
        # safety[p] = a_{|peak_idx - p|} (roughly: how much a_p interacts with the peak)
        safety = np.zeros(n)
        for p in range(n):
            j = peak_idx - p
            if 0 >= j or j >= n:  # Actually, if p is such that peak_idx-p is out of range, it's "safe"
                safety[p] = 0.0  # completely safe
            else:
                safety[p] = a[j]
        
        # We want to move mass from high-pressure, high-value positions
        # to low-safety positions
        # Score for being a source: pressure * a (high pressure × high value)
        src_score = pressure * a
        # Score for being a target: -safety (low safety is good)
        
        # Try multiple source-target pairs
        top_src = np.argsort(src_score)[::-1][:10]  # top 10 sources
        tgt_candidates = np.argsort(safety)[:10]  # bottom 10 safety (most safe)
        
        best_move_sc = best_sc
        best_move_a = None
        
        for s in top_src:
            for t in tgt_candidates:
                if s == t:
                    continue
                if a[s] < 1e-10:
                    continue
                
                # Try moving a small fraction
                fraction = min(0.01, 0.1 * a[s])
                a_try = a.copy()
                a_try[s] -= fraction
                a_try[t] += fraction
                a_try = normalize_nonneg(a_try)
                
                sc = exact_score(a_try)
                if sc < best_move_sc:
                    best_move_sc = sc
                    best_move_a = a_try.copy()
        
        if best_move_a is not None:
            a = best_move_a
            if best_move_sc < best_sc:
                best_sc = best_move_sc
                best_a = a.copy()
        else:
            # No beneficial move found, try smaller fractions
            found = False
            for s in top_src[:5]:
                if a[s] < 1e-10:
                    continue
                for frac in [0.001, 0.005, 0.02]:
                    for t in tgt_candidates[:5]:
                        if s == t:
                            continue
                        a_try = a.copy()
                        a_try[s] -= frac * a[s]
                        a_try[t] += frac * a[s]
                        a_try = normalize_nonneg(a_try)
                        sc = exact_score(a_try)
                        if sc < best_sc:
                            best_sc = sc
                            best_a = a_try.copy()
                            a = a_try
                            found = True
                            break
                    if found:
                        break
                if found:
                    break
            if not found:
                break
    
    return best_a, best_sc

def multilevel_optimize(a_init, n_target, deadline):
    """
    Multilevel: optimize at increasing lengths.
    """
    current_n = min(n_target, 256)
    a = normalize_nonneg(resample_sequence(a_init, current_n))
    
    while current_n < n_target:
        if time.monotonic() > deadline:
            break
        
        # Optimize at current level
        sub_deadline = deadline - max(60.0, 0.3 * (deadline - time.monotonic()))
        a, _ = lse_grad_descent(a, sub_deadline)
        
        # Upsample
        current_n = min(current_n * 2, n_target)
        a = normalize_nonneg(resample_sequence(a, current_n))
    
    # Final optimization at target length
    a, _ = lse_grad_descent(a, deadline)
    return a

def generate_diverse_inits(n, parent, rng):
    """Generate diverse initializations."""
    inits = []
    
    # From parent
    if parent is not None:
        p = normalize_nonneg(resample_sequence(parent, n))
        inits.append(p)
        inits.append(normalize_nonneg(p[::-1]))  # reversed
    
    # Uniform
    inits.append(np.ones(n) / n)
    
    # Cosine windows
    x = np.linspace(0, 1, n)
    inits.append(normalize_nonneg(0.5 + 0.5 * np.cos(2 * np.pi * x)))
    inits.append(normalize_nonneg(0.5 + 0.5 * np.cos(4 * np.pi * x)))
    
    # Gaussian centered
    inits.append(normalize_nonneg(np.exp(-((x - 0.5) ** 2) / (2 * 0.1 ** 2))))
    
    # Two Gaussians
    inits.append(normalize_nonneg(
        np.exp(-((x - 0.3) ** 2) / (2 * 0.1 ** 2)) + np.exp(-((x - 0.7) ** 2) / (2 * 0.1 ** 2))
    ))
    
    # Random smooth
    for _ in range(5):
        freqs = rng.integers(1, 5, size=3)
        amps = rng.uniform(0.1, 0.5, size=3)
        phases = rng.uniform(0, 2 * np.pi, size=3)
        vals = np.ones(n)
        for f, amp, ph in zip(freqs, amps, phases):
            vals += amp * np.cos(2 * np.pi * f * x + ph)
        inits.append(normalize_nonneg(vals))
    
    return inits
