"""
Minimize 2*n*max(self-convolution)/(sum)^2 for non-negative sequences.
Uses corrected LSE-smoothed gradient (T1+T2), cross-correlation LP iteration,
adaptive beta schedule, multiple restarts, and coordinate descent polishing.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import linprog
import time


def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = rfft(a, M_fft)
    return irfft(A * A, M_fft)[:M]


def fast_cross_conv(f, g):
    """Linear cross-convolution f*g via real FFT."""
    n_f = len(f)
    n_g = len(g)
    M = n_f + n_g - 1
    M_fft = 1 << (M - 1).bit_length()
    F = rfft(f, M_fft)
    G = rfft(g, M_fft)
    return irfft(F * G, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def lse_score_grad_corrected(a, beta):
    """
    LSE-smoothed score and CORRECTED gradient w.r.t. a.
    Gradient uses T1 + T2 (both cross-correlation terms), not just 2*T1.
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()

    # Correct gradient: dL/da_j = T1[j] + T2[j]
    # T1[j] = sum_i a[i] * p[i+j] = conv(p, a[::-1])[n-1+j]
    # T2[j] = sum_i a[i] * p[j-i] = conv(a, p)[j]
    M_total = 3 * n - 2
    M_fft = 1 << (M_total - 1).bit_length()

    a_flip = a[::-1]
    P = rfft(p, M_fft)
    A = rfft(a, M_fft)
    AF = rfft(a_flip, M_fft)

    T1 = irfft(P * AF, M_fft)[n - 1:2 * n - 1]
    T2 = irfft(A * P, M_fft)[:n]

    grad_L = T1 + T2

    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_L / S2 - 2.0 * L / S3)
    return score, grad


def adam_lse_optimize(a_init, lr=0.003, n_steps=100000, beta_start=30,
                      beta_end=3000, time_limit_s=None, t_start=None):
    """Adam optimizer with adaptive LSE beta."""
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    if a.sum() < 1e-12:
        a[:] = 1.0 / n
    a = a / a.sum()

    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.85, 0.999
    eps = 1e-8

    best_a = a.copy()
    best_score = compute_score(a)

    for t in range(1, n_steps + 1):
        if time_limit_s is not None and t_start is not None:
            if time.monotonic() - t_start > time_limit_s:
                break

        # Adaptive beta: linear schedule from beta_start to beta_end
        frac = min(1.0, t / max(1, n_steps // 2))
        beta = beta_start + (beta_end - beta_start) * frac

        lr_t = lr / (1.0 + 5e-5 * t)
        score, grad = lse_score_grad_corrected(a, beta)

        # Gradient clipping
        gnorm = np.linalg.norm(grad)
        if gnorm > 100.0:
            grad = grad * (100.0 / gnorm)

        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        a = a - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1e10)
        s = a.sum()
        if s > 1e-12:
            a = a / s

        if t % 2000 == 0:
            sc = compute_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc_final = compute_score(a)
    if sc_final < best_score:
        best_score = sc_final
        best_a = a.copy()
    return best_score, best_a


def cross_corr_matrix(f, n):
    """Build the (2n-1) x n cross-correlation constraint matrix G."""
    m = 2 * n - 1
    G = np.zeros((m, n), dtype=np.float64)
    # G[k, j] = f[k-j] if 0 <= k-j <= n-1
    for k in range(m):
        j_start = max(0, k - (n - 1))
        j_end = min(k, n - 1)
        if j_end >= j_start:
            G[k, j_start:j_end + 1] = f[j_end - (j_end - j_start):0:-1][:j_end - j_start + 1]
            # Actually: G[k, j] = f[k-j] for j in [j_start, j_end]
            j_vals = np.arange(j_start, j_end + 1)
            G[k, j_vals] = f[k - j_vals]
    return G


def lp_cross_correlation(f, M, n, verbose=False):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g and objective value.
    """
    # Constraint: G @ g <= M * ones(2n-1)
    # Objective: maximize -(-1) @ g = 1 @ g
    G = cross_corr_matrix(f, n)
    c_obj = -np.ones(n)  # minimize -sum(g)
    A_ub = G
    b_ub = M * np.ones(2 * n - 1)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            g = np.clip(res.x, 0, None)
            obj = -res.fun
            return g, obj
        else:
            if verbose:
                print(f"  LP status: {res.status}, msg: {res.message}")
    except Exception as e:
        if verbose:
            print(f"  LP error: {e}")

    # Fallback: return f normalized
    return f / f.sum(), 1.0


def lp_iteration(f, n_iter=10, t_start=None, time_limit_s=None, t_step=0.3):
    """
    Iterate: solve cross-correlation LP, update f <- (1-t)*f + t*g.
    """
    n = len(f)
    a = f.astype(np.float64).copy()
    a = a / a.sum()
    best_score = compute_score(a)
    best_a = a.copy()
    deadline = t_start + time_limit_s if (time_limit_s and t_start) else np.inf

    for it in range(n_iter):
        if time.monotonic() > deadline:
            break

        c = fast_conv(a)
        M = float(c.max())

        g, obj = lp_cross_correlation(a, M, n, verbose=(it == 0))
        if g.sum() > 1e-12:
            g = g / g.sum()
        else:
            g = a.copy()

        a_new = (1 - t_step) * a + t_step * g
        a_new = np.clip(a_new, 0, None)
        s = a_new.sum()
        if s > 1e-12:
            a_new = a_new / s

        sc = compute_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
        else:
            # Try smaller step
            a_new2 = (1 - t_step * 0.5) * a + t_step * 0.5 * g
            a_new2 = np.clip(a_new2, 0, None)
            s2 = a_new2.sum()
            if s2 > 1e-12:
                a_new2 = a_new2 / s2
            sc2 = compute_score(a_new2)
            if sc2 < best_score:
                best_score = sc2
                best_a = a_new2.copy()
                a = a_new2
            else:
                # No improvement, reduce step size
                t_step *= 0.7

    return best_score, best_a


def coordinate_descent(a, top_k=50, n_sweeps=5, time_limit_s=None, t_start=None):
    """Selective coordinate descent on the score."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    deadline = t_start + time_limit_s if (time_limit_s and t_start) else np.inf

    for sweep in range(n_sweeps):
        if time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        M = float(c.max())
        score = 2.0 * n * M / (S * S)
        k_star = int(c.argmax())

        # Identify coordinates that contribute to the peak
        js = max(0, k_star - n + 1)
        je = min(k_star, n - 1)
        if je >= js:
            active = list(range(js, je + 1))
        else:
            active = list(range(n))

        # Also include coordinates near other peaks
        top_theta = 0.95 * M
        peak_indices = np.where(c >= top_theta)[0]
        for ki in peak_indices:
            js2 = max(0, ki - n + 1)
            je2 = min(ki, n - 1)
            for j in range(js2, je2 + 1):
                if j not in active:
                    active.append(j)

        active = np.array(active[:top_k * 3])
        rng_local = np.random.default_rng(42 + sweep)
        if len(active) > top_k:
            active = active[rng_local.choice(len(active), size=top_k, replace=False)]

        for j in active:
            if time.monotonic() > deadline:
                break
            a_old = a[j]
            if a_old < 1e-10:
                continue
            # Try reducing this coordinate
            for factor in [0.5, 0.7, 0.85, 0.95]:
                a[j] = a_old * factor
                sc = compute_score(a)
                if sc < best_score:
                    best_score = sc
                    break
            else:
                a[j] = a_old

    final_sc = compute_score(a)
    return best_score, a

        try_update(sc_cd, a_cd, "coord descent")

    # Phase 5: Final short LSE polish on best (remaining time)
    print(f"\n{'=' * 60}")
    print("Phase 5: Final LSE polish")
    print(f"{'=' * 60}")

    remaining = effective_budget - (time.monotonic() - t_start)
    if remaining > 10 and best_a is not None:
        t_p5_start = time.monotonic()
        sc_final, a_final = adam_lse_optimize(
            best_a, lr=0.001, n_steps=50000,
            beta_start=3000, beta_end=10000,
            time_limit_s=remaining, t_start=t_p5_start
        )
        try_update(sc_final, a_final, "final polish")

    # Final output
    if best_a is None:
        best_a = np.ones(64) / 64

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [1.0] * 64

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.10f}, verified={verified:.10f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")

    # Use verified score to decide
    if verified < best_score or np.isinf(best_score):
        best_score = verified

    return out
