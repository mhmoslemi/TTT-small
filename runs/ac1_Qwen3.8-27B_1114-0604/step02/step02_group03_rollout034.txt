"""
Multi-scale adaptive optimizer for minimizing 2*n*max(a*a)/(sum(a))^2.
Uses: multi-scale init, softmax-gradient descent, LP-based direction finding,
and aggressive coordinate descent polish.

import numpy as np

from scipy.optimize import linprog

def _normalize(v):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1)
    v[v < 0] = 0
    s = v.sum()
    if s < 1e-300:
        return np.ones(v.size) / v.size
    return v / s

def _fft_size(m):
    if m <= 2:
        return 2
    return 1 << (m - 1).bit_length()

def _autocorr(f):
    n = f.size
    m = 2 * n - 1
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:m]

def _score(f):
    f = _normalize(f)
    c = _autocorr(f)
    return float(2.0 * f.size * c.max())

def _interp(f, n):
    m = f.size
    if m == n:
        return f.copy()
    x_old = np.linspace(0, m-1, m)
    x_new = np.linspace(0, n-1, n)
    return _normalize(np.interp(x_new, x_old, f))

"""
Multi-scale adaptive optimizer for minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Uses progressive refinement, softmax-gradient descent with adaptive steps,
LP-based direction finding, and aggressive exact coordinate descent.

import numpy as np
import time
import math
from scipy.optimize import linprog

TARGET_SCORE = 1.5027


def _normalize(v):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum.vKV(v, 0.0, out=v)
    s = float(v.sum())
    if s < 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    return v / s


def _fft_size(m):
    if m <= 2:
        return 2
    return 1 << (int(m) - 1).bit_length()


def _autocorr(f):
    n = f.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = 2 * n - 1
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]


def _fast_score(f):
    f = _normalize(f)
    c = _autocorr(f)
    if c.size == 0:
        return np.inf
    return float(2.0 * f.size * c.max())


def _interp(f, n):
    f = _normalize(f)
    m = f.size
    if m == n:
        return f.copy()
    if n == 1:
        return np.array([1.0])
    if m == 1:
        return _normalize(np.full(n, f[0]))
    x_old = np.linspace(0.0, m - 1, m)
    x_new = np.linspace(0.0, n - 1, n)
    return _normalize(np.interp(x_new, x_old, f))


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = out.sum()
    if s < 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return out / s


def _sine_seed(n, rng):
    x = np.linspace(0.0, 1.0, n)
    f = 1.0
    for m in range(1, min(20, n // 2) + 1):
        amp = float(rng.uniform(0.02, 0.15)) / math.sqrt(m)
        phase = float(rng.uniform(0, 2 * math.pi))
        f += amp * np.sin(2.0 * math.pi * m * x + phase)
    f = np.maximum(f, 0.01)
    return _normalize(f)


def _beta_seed(n, rng):
    x = np.linspace(0.0, 1.0, n)
    a = float(rng.uniform(0.3, 2.5))
    b = float(rng.uniform(0.3, 2.5))
    f = (x + 0.01) ** a * (1.0 - x + 0.01) ** b
    return _normalize(f)

    f = _normalize(f)
    n = f.size
    c = _autocorr(f)
    maxc = float(c.max())
    best_f = f.copy()
    best_score = float(2.0 * n * maxc)
    deadline = time.monotonic() + budget
    sigma = 0.15
    no_accept = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        if maxc < 1e-15:
            break
        T = maxc * sigma + 1e-30
        z = (c - maxc) / T

        p = np.exp(z)

        if ps < 1e-30:
            p.fill(1.0 / p.size)
        else:

        g = np.fft.irfft(P * FR, size)

        g = g - g.mean()

                if mc2 < maxc * (1.0 - 1e-10):
                    f = ft2
                    c = ct2
                    maxc = float(mc2)
                    accepted = True
                    break

            no_accept = 0
            sc = float(2.0 * n * maxc)

                    best_f = f.copy()

                sigma = 0.1

def _coord_polish_small(f, c, budget, rng):
    """Quick coordinate polish for small scales."""
    f = _normalize(f)
    n = f.size
    deadline = time.monotonic() + budget
    L = 2 * n - 1
    c = c.copy()
    delta = np.zeros(L)
    total = np.zeros(L)
    maxc = float(c.max())

    while time.monotonic() < deadline:
        # Find top donors and receivers
        g = _quick_gradient(c, f)
        donors = np.argsort(g)[-min(10, n):][::-1]
        receivers = np.argsort(g)[:min(10, n)]
        best_mc = maxc
        best_move = None
        for i in donors:
            i = int(i)
            if f[i] < 1e-12:
                continue
            for j in receivers:
                j = int(j)
                if i == j:
                    continue
                for frac in (0.1, 0.3, 0.5):
                    dv = f[i] * frac
                    mc = _transfer_max(c, f, i, j, dv, delta, total)
                    if mc < best_mc:
                        best_mc = mc
                        best_move = (i, j, dv)
        if best_move is None:
            # Random trial
            for _ in range(20):
                i = int(rng.integers(n))
                j = int(rng.integers(n))
                if i == j or f[i] < 1e-12:
                    continue
                dv = f[i] * float(rng.uniform(0.05, 0.4))
                mc = _transfer_max(c, f, i, j, dv, delta, total)
                if mc < best_mc:
                    best_mc = mc
                    best_move = (i, j, dv)
        if best_move is None:
            break
        i, j, dv = best_move
        f[i] -= dv
        f[j] += dv
        f = _normalize(f)
        c = _autocorr(f)
        maxc = float(c.max())

    return f


def _quick_gradient(c, f):
    """Quick gradient: 2 * sum_k p_k * a[k-i] where p is softmax of c."""
    n = f.size
    maxc = c.max()
    T = maxc * 0.1 + 1e-30
    p = np.exp((c - maxc) / T)
    ps = p.sum()
    p = p / (ps + 1e-30)
    # g[i] = 2 * sum_k p[k] * a[k-i] (valid range)
    size = _fft_size(2 * c.size)
    P = np.fft.rfft(p, size)
    FR = np.fft.rfft(f[::-1], size)
    g = np.fft.irfft(P * FR, size)
    return 2.0 * g[n - 1:n + n - 1]


def _transfer_max(c, f, i, j, dv, delta, total):
    n = f.size
    L = c.size
    delta.fill(0.0)
    delta[i:i + n] -= f
    delta[j:j + n] += f
    delta *= (2.0 * dv)
    q = dv * dv
    delta[2 * j] += q
    delta[2 * i] += q
    delta[i + j] -= 2.0 * q
    np.add(c, delta, out=total)
    return float(np.max(total))


def _lp_direction(f, M, n_lp=None):
    """Solve LP to find direction g that minimizes peak of f*g."""
    if n_lp is None:
        n_lp = f.size
    if n_lp < f.size:
        f_lp = _interp(f, n_lp)
    else:
        f_lp = f.copy()
    n = f_lp.size
    L = 2 * n - 1
    # Build constraint matrix A where A[k,j] = f_lp[k-j] if valid
    # (f*b)_k = sum_j f_lp[k-j] * b[j] for valid (k-j)
    # So A[k,j] = f_lp[k-j] if 0 <= k-j < n, else 0
    # We need (f*b)_k <= M for all k
    # In matrix form: A @ b <= M * ones(L)
    # A is L x n
    # To save memory, build sparsely or use the structure
    
    # For the LP: max sum(b) s.t. A@b <= M, b >= 0
    # c = -ones(n) (minimize -sum(b))
    c_obj = -np.ones(n)
    
    # Build A matrix (L x n)
    # A[k, j] = f_lp[k-j] if 0 <= k-j < n
    # This is a "flipped Toeplitz" structure
    # For k in [0, L-1], j in [0, n-1]:
    #   A[k,j] = f_lp[k-j] if max(0, k-n+1) <= j <= min(n-1, k)
    A = np.zeros((L, n), dtype=np.float64)
    for j in range(n):
        # A[k,j] = f_lp[k-j] for k in [j, j+n-1]
        A[j:j+n, j] = f_lp
    
    b_ub = M * np.ones(L)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if res.status != 0:
            return None
        b_opt = res.x
        s = b_opt.sum()
        if s < 1e-10:
            return None
        g = b_opt / s
        if n_lp < f.size:
            g = _normalize(_interp(g, f.size))
        else:
            g = _normalize(g)
        return g
    except Exception:
        return None

                last_coord = time.monotonic()
                no_accept = 0
                sigma = 0.08
            # LP-based direction (periodic, only for smaller n)
            elif now - last_lp > 60.0 and n <= 5000 and no_accept > 20:
                g_dir = _lp_direction(f, maxc, n_lp=min(n, 2000))
                if g_dir is not None:
                    for t in (0.1, 0.2, 0.3, 0.5):
                        ft = _normalize((1.0 - t) * f + t * g_dir)
                        ct = _autocorr(ft)
                        mc = float(ct.max())
                        if mc < maxc - 1e-14:
                            f = ft
                            c = ct
                            maxc = mc
                            sc = float(2.0 * n * maxc)
                            if sc < best_score:
                                best_score = sc
                                best_f = f.copy()
                            accepted = True
                            no_accept = 0
                            break
                last_lp = time.monotonic()
            # Perturbation if stuck
            elif no_accept > 60:
                r = rng.random()
                if r < 0.5:
                    f = f * np.exp(0.2 * rng.standard_normal(n))
                else:
                    noise = rng.random(n)
                    noise /= noise.sum()
                    f = 0.7 * f + 0.3 * noise
                f = _normalize(f)
                c = _autocorr(f)
                maxc = float(c.max())
                no_accept = 0
                sigma = 0.15
        
        if it % 5000 == 0 or (now - last_print > 60):
            print(f"[{tag}] it={it} best={best_score:.6f} cur={float(2.0*n*maxc):.6f} sigma={sigma:.3f}")
            last_print = time.monotonic()
    
    print(f"[{tag}] end best={best_score:.6f}")
    return best_f, best_score


def _coord_descent_aggressive(f0, c0, budget, rng, tag=""):
    """Aggressive coordinate descent with many donors/receivers."""
    f = _normalize(f0)
    n = f.size
    L = 2 * n - 1
    c = np.asarray(c0, dtype=np.float64).copy()
    if c.size != L:
        c = _autocorr(f)
    maxc = float(c.max())
    best_f = f.copy()
    best_score = float(2.0 * n * maxc)
    deadline = time.monotonic() + budget
    delta = np.zeros(L)
    total = np.zeros(L)
    no_improve = 0
    moves = 0
    
    # Adaptive parameters based on n
    if n > 8000:
        D, R = 30, 30
        n_frac = 4
    elif n > 3000:
        D, R = 50, 50
        n_frac = 6
    else:
        D, R = 80, 80
        n_frac = 8
    
    fracs = np.linspace(0.02, 0.6, n_frac)
    
    print(f"[{tag}] aggressive coord descent: n={n}, D={D}, R={R}, fracs={n_frac}")
    
    while time.monotonic() < deadline and no_improve < 30:
        c = _autocorr(f)
        maxc = float(c.max())
        
        # Active set: top lags
        K = min(L, max(50, n // 5))
        active = np.argpartition(c, L - K)[L - K:]
        p = np.zeros(L)
        T = maxc * 0.15 + 1e-30
        p[active] = np.exp((c[active] - maxc) / T)
        ps = p.sum()
        if ps < 1e-30:
            p[active] = 1.0 / K
        else:
            p /= ps
        
        g = _quick_gradient(c, f)
        
        # Donors: highest gradient (mass should leave)
        donor_candidates = np.argsort(g)[::-1][:D]
        # Receivers: lowest gradient (mass should arrive)
        recv_candidates = np.argsort(g)[:R]
        
        best_mc = maxc
        best_move = None
        
        # Try structured pairs
        for i in donor_candidates:
            i = int(i)
            if f[i] < 1e-12:
                continue
            fi = f[i]
            for j in recv_candidates:
                j = int(j)
                if j == i:
                    continue
                for frac in fracs:
                    dv = fi * frac
                    mc = _transfer_max(c, f, i, j, dv, delta, total)
                    if mc < best_mc:
                        best_mc = mc
                        best_move = (i, j, dv)
        
        # Random pairs
        n_rand = max(20, D * R // 10)
        for _ in range(n_rand):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j or f[i] < 1e-12:
                continue
            dv = f[i] * float(rng.uniform(0.02, 0.5))
            mc = _transfer_max(c, f, i, j, dv, delta, total)
            if mc < best_mc:
                best_mc = mc
                best_move = (i, j, dv)
        
        if best_move is None or best_mc >= maxc - 1e-15:
            no_improve += 1
            continue
        
        i, j, dv = best_move
        f[i] -= dv
        f[j] += dv
        f = _normalize(f)
        moves += 1
        no_improve = 0
        sc = float(2.0 * n * best_mc)
        if sc < best_score:
            best_score = sc
            best_f = f.copy()
        
        # Periodic full recomputation
        if moves % 20 == 0:
            c = _autocorr(f)
            maxc = float(c.max())
            sc = float(2.0 * n * maxc)
            if sc < best_score:
                best_score = sc
                best_f = f.copy()
    
    print(f"[{tag}] aggressive coord done: best={best_score:.6f} moves={moves}")
    return best_f, best_score
