- Result[1] = $a_0 a_1 + a_1 a_0 = 2 a_0 a_1$
- ...
- Result[n-1] = $\sum_{i=0}^{n-1} a_i a_{n-1-i}$

Hmm, that's the "reverse" autocorrelation. Let me just use the FFT approach and not worry about the exact indexing.

For the score computation, I just need $\max(\text{full convolution of } a \text{ with } a)$, which is straightforward with FFT.

For the gradient/pressure computation, let me just compute it directly using FFT:

The "pressure" at position $i$ should be proportional to how much increasing $a_i$ would increase the peak of the convolution. This is:
$\text{pressure}_i \propto \sum_k w_k \frac{\partial c_k}{\partial a_i}$

where $c_k = (\text{convolve}(a,a))_k$ and $w_k$ are the softmax weights.

$\frac{\partial c_k}{\partial a_i} = a_{k-i} + a_{i-k}$ (for the parts where these indices are valid)

This is essentially: $\text{pressure} = w * a + \text{flip}(w) * a$ (convolutions)

where $w$ has length $2n-1$ and $a$ has length $n$.

Using FFT:
- Compute $F_w = \text{FFT}(w, L)$ and $F_a = \text{FFT}(a, L)$ with $L \geq 3n-1$ (to fit the result)
- $\text{pressure} = \text{IFFT}(F_w \cdot F_a)$ (first $n$ elements) $+ \text{IFFT}(F_{\text{flip}(w)} \cdot F_a)$ (first $n$ elements)

Since $w$ is (approximately) symmetric, $F_{\text{flip}(w)} \approx F_w$, so:
$\text{pressure} \approx 2 \cdot \text{IFFT}(F_w \cdot F_a)[:n]$

Actually, I realize I'm overcomplicating this. Let me just use a simpler approach:

1. Compute the autocorrelation $c$ (length $2n-1$)
2. Compute softmax weights $w = \text{softmax}(\beta \cdot c)$
3. Compute the "influence" of each position: for each position $i$, how much does it contribute to each $c_k$?
   - $a_i$ contributes to $c_k$ via $a_i \cdot a_{k-i}$ (for the "forward" part) and $a_i \cdot a_{i-k}$ (for the "backward" part)
   - Total contribution of $a_i$ to the weighted peak: $\sum_k w_k (a_{k-i} + a_{i-k})$ (with appropriate bounds)
4. Update: $a_i \leftarrow a_i \cdot \exp(-\alpha \cdot \text{contribution}_i)$

Step 3 can be computed as a convolution:
$\text{contribution}_i = \sum_k w_k \cdot a_{(k-i) \mod n}$ (circular) + $\sum_k w_k \cdot a_{(i-k) \mod n}$ (circular)

But we want linear, not circular. Using FFT with zero-padding handles this automatically.

Let me code this up properly now. I'll use the approach:
- `pressure = ifft(fft(w, L) * fft(a, L))[:n] + ifft(fft(flip(w), L) * fft(a, L))[:n]`

where $L$ is large enough for the linear convolution.

"""
Optimizer for non-negative sequences minimizing 2*n*max(convolve(a,a))/sum(a)^2.
Uses FFT-based fast evaluation, entropy-regularized multiplicative updates,
adaptive thresholding, and multi-scale local search.
"""

import numpy as np
import time

_rng = np.random.default_rng(42)


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _fast_score(p, L=None):
    n = len(p)
    if n == 0:
        return float('inf'), np.array([])
    if L is None:
        L = _next_pow2(2 * n - 1)
    Fp = np.fft.rfft(p, L)
    c = np.fft.irfft(Fp * Fp, L)
    c = c[:2 * n - 1]
    c = np.maximum(c, 0.0)
    s = p.sum()
    if s < 1e-12:
        return float('inf'), c
    score = 2.0 * n * c.max() / (s * s)
    return score, c


def _sanitize(p, min_p=0.0):
    a = np.asarray(p, dtype=float).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, min_p, 1000.0)
    s = a.sum()
    if s < 1e-12:
        return None
    return a / s


def _resample(p, m):
    a = _sanitize(p)
    if a is None:
        return None
    n = len(a)
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m) / m
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(m) + 0.5) / m
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0) + 1e-12
    return _sanitize(out)


def _smooth(p, sigma):
    a = _sanitize(p)
    if a is None or sigma < 0.1:
        return a
    n = len(a)
    rad = max(1, int(np.ceil(3 * sigma)))
    k = np.arange(-rad, rad + 1, dtype=float)
    kern = np.exp(-0.5 * (k / sigma) ** 2)
    kern /= kern.sum()
    out = np.convolve(a, kern, mode='same')
    return _sanitize(out)


def _noise(p, sigma):
    a = _sanitize(p)
    if a is None or sigma <= 0:
        return a
    r = np.exp(sigma * _rng.normal(0, 1, len(a)))
    b = a * r
    return _sanitize(b, 1e-14)


def _initial_sequence(n, mode=0):
    t = (np.arange(n) + 0.5) / n
    p = np.ones(n)
    if mode == 0:
        z = np.zeros(n)
        for k in range(1, 15):
            amp = _rng.normal(0, 1.0 / (k + 1))
            ph = _rng.uniform(0, 2 * np.pi)
            z += amp * np.cos(2 * np.pi * k * t + ph)
        p = np.log1p(np.exp(np.clip(z, -50, 50)))
    elif mode == 1:
        a = _rng.uniform(0.1, 3.0)
        b = _rng.uniform(0.1, 3.0)
        tt = np.clip(t, 1e-8, 1 - 1e-8)
        p = tt ** a * (1 - tt) ** b
    elif mode == 2:
        K = _rng.integers(1, 5)
        p = np.zeros(n)
        centers = np.sort(_rng.uniform(0, 1, K))
        sigmas = _rng.uniform(0.03, 0.3, K)
        weights = _rng.uniform(0.5, 2.0, K)
        for i in range(K):
            p += weights[i] * np.exp(-((t - centers[i]) / sigmas[i]) ** 2)
    elif mode == 3:
        K = _rng.integers(3, 10)
        knots = np.sort(_rng.uniform(0, 1, K))
        heights = _rng.lognormal(0, 1.0, K)
        p = np.interp(t, knots, heights)
    elif mode == 4:
        K = _rng.integers(4, 16)
        edges = np.sort(_rng.uniform(0, 1, K - 1))
        blocks = np.concatenate([[0], edges, [1.0]])
        heights = _rng.lognormal(0, 1.0, K)
        p = np.zeros(n)
        for i in range(K):
            s = int(round(blocks[i] * n))
            e = max(s + 1, int(round(blocks[i + 1] * n)))
            p[s:min(n, e)] = heights[i]
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0) + 1e-12
    return _sanitize(p)


def _multiplicative_update(p, L, beta, alpha, target_m, eps):
    """One multiplicative update step using softmax-smoothed max."""
    n = len(p)
    Fp = np.fft.rfft(p, L)
    c = np.fft.irfft(Fp * Fp, L)
    c = c[:2 * n - 1]
    c = np.maximum(c, 0.0)
    max_c = c.max()
    score = 2.0 * n * max_c

    if max_c <= target_m * (1 + 1e-10):
        return p, score, max_c

    # Compute softmax weights over the convolution
    shift = max_c
    w = np.exp(beta * (c - shift))
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= ws

    # Compute pressure: for each position i, sum_k w_k * (a_{i-k} + a_{i+k})
    # This is cross-correlation of w with a (and its flip)
    L2 = _next_pow2(3 * n)
    Fw = np.fft.rfft(w, L2)
    Fw_flip = np.fft.rfft(w[::-1], L2)
    Fa = np.fft.rfft(p, L2)
    press1 = np.fft.irfft(Fw * Fa, L2)[:n]
    press2 = np.fft.irfft(Fw_flip * Fa, L2)[:n]
    pressure = press1 + press2
    pressure = np.maximum(pressure, 0.0)

    # Normalize pressure
    pm = pressure.max()
    if pm < 1e-300:
        return p, score, max_c
    pressure = pressure / pm

    # Multiplicative update
    p_new = p * np.exp(-alpha * pressure)
    p_new = np.maximum(p_new, 1e-14)
    p_new /= p_new.sum()

    sc_new, _ = _fast_score(p_new, L)
    return p_new, float(sc_new), max_c


def _local_search(p, time_limit, verbose=False):
    """Block-level local search."""
    p = _sanitize(p, 1e-14)
    if p is None:
        return p, float('inf')
    n = len(p)
    L = _next_pow2(2 * n - 1)
    best = p.copy()
    best_sc = _fast_score(p, L)[0]
    deadline = time.monotonic() + time_limit
    stagnant = 0
    it = 0

    small = max(1, n // 2048)
    mid = max(small + 1, n // 256)
    large = max(mid + 1, min(n, n // 32))
    factors = np.array([0.5, 0.7, 0.85, 1.05, 1.3, 1.8, 0.4, 2.5])

    while time.monotonic() < deadline:
        r = _rng.random()
        if r < 0.30:
            w = 1
        elif r < 0.60:
            w = int(_rng.integers(small, min(n, small * 6 + 1)))
        elif r < 0.85:
            w = int(_rng.integers(mid, min(n, mid * 5 + 1)))
        else:
            w = int(_rng.integers(large, min(n, large * 4 + 1)))
        w = max(1, min(n, w))
        start = int(_rng.integers(0, n - w + 1))

        found = False
        for f in factors:
            cand = p.copy()
            cand[start:start + w] *= f
            cand = np.maximum(cand, 1e-14)
            s = cand.sum()
            if s < 1e-12:
                continue
            cand /= s
            sc, _ = _fast_score(cand, L)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best = cand.copy()
                p = cand.copy()
                found = True
                break

        if found:
            stagnant = 0
            if verbose and it % 200 == 0:
                print(f"  local it={it} best={best_sc:.7f}")
        else:
            stagnant += 1
            if stagnant > 30:
                std = 0.05 + 0.002 * min(stagnant, 100)
                p_try = p * np.exp(_rng.normal(0, std, n))
                p_try = np.maximum(p_try, 1e-14)
                s = p_try.sum()
                if s > 1e-12:
                    p_try /= s
                    sc2, _ = _fast_score(p_try, L)
                    if sc2 < max(best_sc * 1.02, _fast_score(p, L)[0] * 1.05):
                        p = p_try
                    else:
                        p = best.copy()
                stagnant = 0
            if stagnant > 200:
                p = best.copy()
                stagnant = 0
        it += 1

    return best, best_sc


def _optimize_sequence(p0, time_limit, target_score=2.0, beta_start=30.0,
                       beta_end=500.0, alpha_start=0.15, verbose=False):
    """Main optimization loop using multiplicative updates."""
    p = _sanitize(p0, 1e-14)
    if p is None:
        return p0, float('inf')
    n = len(p)
    L = _next_pow2(2 * n - 1)
    norm_sum = 1.0  # p is normalized to sum=1, so score = 2*n*max(c)
    target_m = target_score / (2.0 * n)

    best = p.copy()
    best_sc, _ = _fast_score(p, L)
    cur_sc = best_sc
    beta = beta_start
    alpha = alpha_start
    deadline = time.monotonic() + time_limit
    stall = 0
    it = 0
    last_print = -100

    while time.monotonic() < deadline:
        p_new, sc_new, max_c = _multiplicative_update(
            p, L, beta, alpha, target_m, 0.1
        )

        if np.isfinite(sc_new) and sc_new < cur_sc:
            p = p_new
            cur_sc = sc_new
            stall = 0
            if cur_sc < best_sc:
                best_sc = cur_sc
                best = p.copy()
                if verbose and it - last_print >= 100:
                    print(f"  mmd n={n} it={it} best={best_sc:.7f}")
                    last_print = it
        else:
            # Try smaller alpha
            alpha *= 0.7
            stall += 1
            if stall > 15:
                # Random perturbation
                std = 0.05 + 0.003 * min(stall, 50)
                p_try = p * np.exp(_rng.normal(0, std, n))
                p_try = np.maximum(p_try, 1e-14)
                s = p_try.sum()
                if s > 1e-12:
                    p_try /= s
                    sc2, _ = _fast_score(p_try, L)
                    if np.isfinite(sc2) and sc2 < max(best_sc * 1.03, cur_sc * 1.1):
                        p = p_try
                        cur_sc = sc2
                    else:
                        p = best.copy()
                        cur_sc = best_sc
                else:
                    p = best.copy()
                    cur_sc = best_sc
                stall = 0
                alpha = alpha_start

            if stall > 100:
                p = best.copy()
                cur_sc = best_sc
                alpha = alpha_start
                beta = beta_start
                stall = 0

        if it % 20 == 0:
            beta = min(beta_end, beta * 1.12)
            alpha = min(0.5, alpha * 1.02)

        it += 1

    return best, best_sc


def _try_lengths(parent, cands, n, t_alloc, deadline):
    """Optimize at a specific length."""
    if t_alloc < 10:
        return

    # Build starting candidates
    starts = []
    if parent is not None:
        sp = _resample(parent, n)
        if sp is not None:
            sp = _smooth(sp, 1.0)
            starts.append(sp)

    if cands:
        for item in cands[:2]:
            arr = item[1]
            sp = _resample(arr, n)
            if sp is not None:
                sp = _smooth(sp, 0.5)
                sp = _noise(sp, 0.02)
                if sp is not None:
                    starts.append(sp)

    # Add random starts
    for mode in (0, 1, 2, 3, 4, 0, 1):
        if len(starts) >= 4:
            break
        rp = _initial_sequence(n, mode)
        if rp is not None:
            starts.append(rp)

    # Score and select best starts
    scored = []
    for sp in starts:
        sc, _ = _fast_score(sp)
        if np.isfinite(sc):
            scored.append((sc, sp))
    scored.sort(key=lambda x: x[0])

    if not scored:
        return

    # Allocate time to top candidates
    n_cands = min(3, len(scored))
    t_each = t_alloc / n_cands * 0.7

    for idx in range(n_cands):
        rem = deadline - time.monotonic()
        if rem < 20:
            break
        t_this = min(t_each, rem - 20)
        if t_this < 10:
            break
        sc0 = scored[idx][0]
        eps0 = 0.2 if sc0 < 2.01 else 0.1

        p_opt, sc_opt = _optimize_sequence(
            scored[idx][1], t_this,
            target_score=2.0,
            beta_start=30.0,
            beta_end=400.0,
            alpha_start=0.12,
            verbose=(t_alloc > 100)
        )
        if p_opt is not None and np.isfinite(sc_opt):
            cands.append((sc_opt, p_opt.copy()))

    cands.sort(key=lambda x: x[0])
    if len(cands) > 16:
        del cands[16:]
