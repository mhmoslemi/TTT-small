"""
Step-function optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2
using smooth max approximation (log-sum-exp), Adam optimizer on the
simplex, FFT-based gradients, and coordinate polish.

Key innovations over the previous approach:
- FFT-based gradient of the smooth score (O(n log n) per step)
- Adam optimizer for adaptive per-coordinate learning rates
- Diverse seed pool including plateau, two-peak, and structured seeds
- Multi-phase optimization: explore -> refine -> polish
"""

import numpy as np
import time

TARGET_SCORE = 1.5027


def _normalize(v):
    """Project to non-negative float vector of sum 1."""
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v


def _fft_size(m):
    """Smallest power of two at least m."""
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _autocorr(f):
    """Full linear convolution f*f via FFT."""
    n = f.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = 2 * n - 1
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]


def _fast_score(f):
    """Score = 2*n*max(conv(f,f)) for normalized f."""
    n = f.size
    if n == 0:
        return np.inf
    c = _autocorr(f)
    if c.size == 0:
        return np.inf
    return float(2.0 * n * c.max())


def _project_simplex(v):
    """Euclidean projection onto x >= 0, sum x = 1."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return out / s


def _compute_grad(f, c_temp):
    """
    Compute gradient of smooth score S_smooth = 2n/c * log(sum exp(c*conv_k))
    with respect to f (assuming sum(f)=1, f>=0).
    
    Returns gradient vector of length n.
    """
    n = f.size
    conv = _autocorr(f)
    L = conv.size  # 2n-1
    
    # Softmax weights
    max_conv = float(conv.max())
    z = c_temp * (conv - max_conv)
    np.clip(z, -50.0, 0.0, out=z)
    p = np.exp(z)
    p_sum = float(p.sum())
    if p_sum < 1e-300:
        p = np.ones(L, dtype=np.float64) / L
    else:
        p /= p_sum
    
    # grad[m] = 4n * sum_{j=0}^{n-1} p[m+j] * f[j]  (for m+j < L)
    # This is a cross-correlation, computed via FFT.
    # grad[m] = 4n * (p_rev * f)[2n-2-m] where * is linear convolution
    
    p_rev = p[::-1].copy()
    size = _fft_size(L + n - 1)  # >= 3n-2
    P = np.fft.rfft(p_rev, size)
    F = np.fft.rfft(f, size)
    conv_res = np.fft.irfft(P * F, size)
    
    # We need indices from 2n-2-(n-1)=n-1 to 2n-2-0=2n-2, reversed
    grad = 4.0 * n * conv_res[n - 1:2 * n - 1][::-1]
    
    return grad


def _coordinate_polish(f0, c0, budget_s, rng):
    """Local search by exact mass transfers between coordinates."""
    f = _normalize(f0)
    n = f.size
    if n < 2 or budget_s <= 0:
        return f, _fast_score(f)
    
    L = 2 * n - 1
    try:
        c = np.asarray(c0, dtype=np.float64).copy()
    except Exception:
        c = _autocorr(f)
    if c.size != L or not np.all(np.isfinite(c)):
        c = _autocorr(f)
    
    maxc = float(c.max())
    best_f = f.copy()
    best_score = float(2.0 * n * maxc)
    
    deadline = time.monotonic() + budget_s
    delta = np.zeros(L, dtype=np.float64)
    total = np.zeros(L, dtype=np.float64)
    no_improve = 0
    
    D = 8
    R = 8
    fracs = (0.02, 0.05, 0.1, 0.2, 0.5)
    
    def transfer_max(i, j, dv):
        """Compute new max after f[i]-=dv, f[j]+=dv."""
        delta.fill(0.0)
        delta[i:i + n] -= f
        delta[j:j + n] += f
        delta *= 2.0 * dv
        q = dv * dv
        delta[2 * j] += q
        delta[2 * i] += q
        delta[i + j] -= 2.0 * q
        np.add(c, delta, out=total)
        return float(total.max())
    
    while time.monotonic() < deadline and no_improve < 20:
        maxc = float(c.max())
        
        # Find active lags
        activeK = min(L, max(30, n // 10))
        active = np.argpartition(c, L - activeK)[L - activeK:]
        p = np.zeros(L, dtype=np.float64)
        p[active] = 1.0 / activeK
        
        # Compute weighted gradient for donor/receiver selection
        # g[i] = 2 * sum_k p[k] * a[k-i] (the gradient of weighted sum of conv at active lags)
        # This is similar to _compute_grad but with uniform weights on active lags
        size = _fft_size(L + n - 1)
        P = np.fft.rfft(p[::-1], size)
        F = np.fft.rfft(f, size)
        conv_res = np.fft.irfft(P * F, size)
        g = 2.0 * conv_res[n - 1:2 * n - 1][::-1]
        
        # Donors: high gradient (removing mass helps)
        donor_order = np.argsort(g)[::-1]
        recv_order = np.argsort(g)
        
        donors = []
        for idx in donor_order:
            idx = int(idx)
            if f[idx] > 1e-12:
                donors.append(idx)
                if len(donors) >= D:
                    break
        
        donor_set = set(donors)
        receivers = []
        for idx in recv_order:
            idx = int(idx)
            if idx not in donor_set:
                receivers.append(idx)
                if len(receivers) >= R:
                    break
        
        if not donors or not receivers:
            no_improve += 1
            if no_improve >= 8:
                break
            continue
        
        best_mc = maxc
        best_move = None
        
        for i in donors:
            fi = float(f[i])
            if fi <= 1e-12:
                continue
            for j in receivers:
                for frac in fracs:
                    dv = fi * frac
                    if dv <= 0:
                        continue
                    mc = transfer_max(i, j, dv)
                    if mc < best_mc:
                        best_mc = mc
                        best_move = (i, j, dv)
        
        # Random fallback
        if best_move is None or best_mc >= maxc - 1e-18:
            for _ in range(30):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j or f[i] <= 1e-12:
                    continue
                dv = float(f[i]) * float(rng.uniform(0.01, 0.3))
                mc = transfer_max(i, j, dv)
                if mc < best_mc:
                    best_mc = mc
                    best_move = (i, j, dv)
        
        if best_move is None or best_mc >= maxc - 1e-18:
            no_improve += 1
            continue
        
        i, j, dv = best_move
        f[i] -= dv
        f[j] += dv
        # Update c incrementally
        delta.fill(0.0)
        # Recompute from scratch every few steps to avoid drift
        no_improve = 0
        
        cur_score = float(2.0 * n * best_mc)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()
        
        # Periodic recompute
        if int(rng.integers(0, 10)) == 0:
            f = _normalize(f)
            c = _autocorr(f)
    
    f = _normalize(f)
    c = _autocorr(f)
    sc = float(2.0 * n * c.max()) if c.size else np.inf
    if sc < best_score:
        best_f = f.copy()
        best_score = sc
    
    return _normalize(best_f), best_score


def _fourier_seed(n, rng):
    """Smooth low-frequency perturbation of uniform."""
    x = np.linspace(0.0, 1.0, n)
    u = np.zeros(n)
    for m in (1, 2, 3, 5, 8, 13):
        amp = 0.3 * rng.uniform(0.5, 1.5)
        u += amp * np.cos(2.0 * np.pi * m * x + rng.uniform(0, 2 * np.pi))
    f = np.exp(0.8 * u)
    return _normalize(f)


def _beta_seed(n, rng):
    """Beta-shaped seed."""
    x = np.linspace(0.0, 1.0, n)
    a = rng.uniform(0.3, 4.0)
    b = rng.uniform(0.3, 4.0)
    f = (x + 0.01) ** a * (1.0 - x + 0.01) ** b
    return _normalize(f)


def _plateau_seed(n, rng):
    """Plateau: high in a contiguous block, low elsewhere."""
    frac = rng.uniform(0.1, 0.6)
    width = max(2, int(n * frac))
    start = int(rng.integers(0, n - width))
    f = np.ones(n) * 0.1
    f[start:start + width] = 1.0
    # Smooth the edges
    edge = max(1, width // 10)
    if start >= edge:
        for k in range(edge):
            f[start - edge + k] = 0.1 + 0.9 * k / edge
    if start + width < n:
        for k in range(edge):
            f[start + width + k] = 0.1 + 0.9 * (1 - k / edge)
    return _normalize(f)


def _two_peak_seed(n, rng):
    """Two separated Gaussian peaks."""
    x = np.linspace(0.0, 1.0, n)
    p1 = rng.uniform(0.1, 0.4)
    p2 = rng.uniform(0.6, 0.9)
    w1 = rng.uniform(0.02, 0.15)
    w2 = rng.uniform(0.02, 0.15)
    h1 = rng.uniform(0.5, 2.0)
    h2 = rng.uniform(0.5, 2.0)
    f = h1 * np.exp(-0.5 * ((x - p1) / w1) ** 2) + h2 * np.exp(-0.5 * ((x - p2) / w2) ** 2)
    f += 0.01
    return _normalize(f)


def _ap_seed(n, rng):
    """Arithmetic progression seed: high at positions in an AP."""
    x = np.linspace(0.0, 1.0, n)
    period = rng.uniform(0.05, 0.4)
    width = rng.uniform(0.01, 0.05)
    f = np.exp(-((x / period) % 1.0 - 0.5) ** 2 / (2 * width ** 2))
    f += 0.01
    return _normalize(f)


def _optimize(f0, budget_s, rng, tag=""):
    """
    Main optimizer: Adam on simplex with smooth max, then coordinate polish.
    """
    f = _normalize(f0)
    n = f.size
    if n == 0 or budget_s <= 0:
        return f, np.inf
    
    c = _autocorr(f)
    maxc = float(c.max())
    best_f = f.copy()
    best_score = float(2.0 * n * maxc)
    
    deadline = time.monotonic() + budget_s
    
    # Phase 1: Adam with moderate temperature (exploration)
    # Phase 2: Adam with high temperature (refinement)
    # Phase 3: Coordinate polish
    
    phase1_frac = 0.55
    phase2_frac = 0.25
    # rest is coordinate polish
    
    # Adam state
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    
    print(f"[{tag}] start score={best_score:.6f}")
    
    # Phase 1: exploration
    c_temp1 = 10.0
    lr1 = 0.05
    phase1_end = time.monotonic() + phase1_frac * budget_s
    
    while time.monotonic() < phase1_end:
        t += 1
        grad = _compute_grad(f, c_temp1)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        step = lr1 * m_hat / (np.sqrt(v_hat) + eps)
        f_new = _project_simplex(f - step)
        f_new = _normalize(f_new)
        
        c_new = _autocorr(f_new)
        maxc_new = float(c_new.max())
        score_new = float(2.0 * n * maxc_new)
        
        if score_new < best_score:
            best_score = score_new
            best_f = f_new.copy()
            if best_score < TARGET_SCORE:
                print(f"[{tag}] below target: {best_score:.6f}")
        
        f = f_new
        c = c_new
        maxc = maxc_new
        
        # Decay learning rate
        if t % 100 == 0:
            lr1 *= 0.95
            c_temp1 = min(30.0, c_temp1 * 1.01)
    
    # Phase 2: refinement with higher c
    c_temp2 = 30.0
    lr2 = 0.01
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    phase2_end = time.monotonic() + phase2_frac * budget_s
    
    while time.monotonic() < phase2_end:
        t += 1
        grad = _compute_grad(f, c_temp2)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        step = lr2 * m_hat / (np.sqrt(v_hat) + eps)
        f_new = _project_simplex(f - step)
        f_new = _normalize(f_new)
        
        c_new = _autocorr(f_new)
        maxc_new = float(c_new.max())
        score_new = float(2.0 * n * maxc_new)
        
        if score_new < best_score:
            best_score = score_new
            best_f = f_new.copy()
        
        f = f_new
        c = c_new
        maxc = maxc_new
        
        if t % 100 == 0:
            lr2 *= 0.95
            c_temp2 = min(80.0, c_temp2 * 1.005)
    
    # Phase 3: Coordinate polish
    polish_time = max(0.0, deadline - time.monotonic() - 5.0)
    if polish_time > 5.0:
        f_pol, s_pol = _coordinate_polish(best_f, _autocorr(best_f), polish_time, rng)
        if s_pol < best_score:
            best_score = s_pol
            best_f = f_pol.copy()
    
    print(f"[{tag}] end best={best_score:.6f}")
    return _normalize(best_f), best_score


def _interp_to_length(f, n):
    """Resample f to length n."""
    f = _normalize(f)
    m = f.size
    if n <= 0:
        return np.empty(0)
    if m == n:
        return f
    if n == 1:
        return np.array([1.0])
    if m == 1:
        return _normalize(np.full(n, f[0]))
    x_old = np.linspace(0.0, m - 1, m)
    x_new = np.linspace(0.0, n - 1, n)
    return _normalize(np.interp(x_new, x_old, f))


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative coefficient sequence.
    Returns a Python list of floats.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    total_deadline = total_start + max(60.0, budget_s) - 30.0
    
    # Get initial sequence
    try:
        f0 = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        f0 = np.ones(6121, dtype=np.float64)
    
    if f0.size == 0:
        f0 = np.ones(6121, dtype=np.float64)
    f0 = np.nan_to_num(f0, nan=0.0, posinf=1000.0, neginf=0.0)
    np.clip(f0, 0.0, 1000.0, out=f0)
    n = int(f0.size)
    f0 = _normalize(f0)
    
    # Build seed pool
    seeds = []
    
    def add_seed(name, f):
        f = _interp_to_length(f, n)
        sc = _fast_score(f)
        if np.isfinite(sc):
            seeds.append((sc, name, f))
    
    add_seed("given", f0)
    add_seed("reverse", f0[::-1])
    
    # Smoothed versions
    for win in (5, 11, 31, 101):
        kernel = np.ones(win) / win
        sm = np.convolve(f0, kernel, mode='same')
        add_seed(f"smooth{win}", sm)
    
    # Gamma transforms
    for gamma in (0.5, 0.75, 1.5, 2.0):
        add_seed(f"gamma{gamma}", np.power(f0 + 1e-10, gamma))
    
    # Random seeds
    for i in range(8):
        add_seed(f"fourier{i}", _fourier_seed(n, rng))
    for i in range(6):
        add_seed(f"beta{i}", _beta_seed(n, rng))
    for i in range(4):
        add_seed(f"plateau{i}", _plateau_seed(n, rng))
    for i in range(4):
        add_seed(f"twopeak{i}", _two_peak_seed(n, rng))
    for i in range(4):
        add_seed(f"ap{i}", _ap_seed(n, rng))
    
    # Sort by score
    seeds.sort(key=lambda x: x[0])
    
    # Track global best
    best_f = seeds[0][2].copy()
    best_score = seeds[0][0]
    
    print(f"Seed pool: {len(seeds)} seeds, best initial = {best_score:.6f}")
    
    # Optimize top seeds
    top_k = min(5, len(seeds))
    # Allocate time to top seeds
    remaining = total_deadline - time.monotonic() - 30.0
    if remaining <= 0:
        return [float(x) for x in best_f]
    
    # Allocate: 50% to seed 0, 25% to seed 1, 15% to seed 2, 5% to seed 3, 5% to seed 4
    weights = [0.45, 0.25, 0.15, 0.08, 0.07][:top_k]
    
    for idx in range(top_k):
        sc, name, f = seeds[idx]
        alloc = weights[idx] * remaining
        alloc = max(30.0, alloc)
        
        if time.monotonic() >= total_deadline - 60.0:
            break
        
        print(f"\n=== Optimizing seed #{idx+1}: {name} (score={sc:.6f}, budget={alloc:.1f}s) ===")
        f_opt, s_opt = _optimize(f, alloc, rng, f"s{idx}_{name}")
        
        if s_opt < best_score:
            best_score = s_opt
            best_f = f_opt.copy()
            print(f"  New best: {best_score:.6f}")
    
    # Final polish from the best
    remaining = total_deadline - time.monotonic() - 10.0
    if remaining > 30.0:
        print(f"\n=== Final polish (budget={remaining:.1f}s) ===")
        f_pol, s_pol = _coordinate_polish(best_f, _autocorr(best_f), remaining * 0.6, rng)
        if s_pol < best_score:
            best_score = s_pol
            best_f = f_pol.copy()

        rem2 = total_deadline - time.monotonic() - 10.0
        if rem2 > 60.0:
            f_opt, s_opt = _optimize(best_f, rem2 * 0.7, rng, "final_adam")
            if s_opt < best_score:
                best_score = s_opt
                best_f = f_opt.copy()

    best_f = _normalize(best_f)
    best_f = np.nan_to_num(best_f, nan=0.0, posinf=0.0, neginf=0.0)
    np.clip(best_f, 0.0, 1000.0, out=best_f)

    try:

            print(f"Official score: {official:.6f}")
    except Exception:
        pass
