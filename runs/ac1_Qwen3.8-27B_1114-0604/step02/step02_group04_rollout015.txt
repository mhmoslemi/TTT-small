import numpy as np
import scipy.optimize as sco
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)

def _logsumexp(x, beta=1.0):
    x = beta * x
    x_max = np.max(x)
    if x_max > 700:
        x_max = 700
    return x_max + np.log(np.sum(np.exp(x - x_max))) / beta

def _smooth_max_and_grad(a, beta):
    """Compute smooth max of convolution and its gradient w.r.t. a."""
    c = _fft_convolve(a)
    n = a.size
    
    # Smooth max
    sm = _logsumexp(c, beta)
    
    # Weights
    c_shift = beta * (c - np.max(c))
    w = np.exp(c_shift)
    w /= w.sum()
    
    # Gradient: g[j] = 2 * sum_k w[k] * a[k-j]
    # This is 2 * cross-correlation of w and a
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    # w has length 2n-1, need to pad to N
    w_pad = np.zeros(N, dtype=np.float64)
    w_pad[:len(w)] = w
    fw = np.fft.rfft(w_pad, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    g = 2.0 * xcorr[:n]
    
    return sm, g, c, w

def _objective_and_grad(x, beta, n):
    """Objective and gradient in x-space (softmax parameterization)."""
    # Softmax
    x_shift = x - np.max(x)
    a = np.exp(x_shift)
    a /= a.sum()
    
    sm, g, c, w = _smooth_max_and_grad(a, beta)
    
    # Objective: 2n * sm (since sum(a) = 1)
    obj = 2.0 * n * sm

    a_g = np.dot(a, g)
    grad_x = a * (g - a_g)
    grad_x *= 2.0 * n
    
    return obj, grad_x

def _optimize_lbfgs(x0, beta, n, maxiter=5000, time_limit=300):
    """Optimize using L-BFGS."""
    t0 = time.monotonic()
    
    def fun(x):
        obj, _ = _objective_and_grad(x, beta, n)
        return obj
    
    def fun_and_grad(x):
        return _objective_and_grad(x, beta, n)

            fun_and_grad, x0, method='L-BFGS-B',
            jac=True, maxiter=maxiter,
            options={'maxls': 50, 'gtol': 1e-8, 'ftol': 1e-12}

import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]

def _score_normalized(a):
    """Score assuming sum(a) = 1, a >= 0. Returns 2n*max(c)."""
    c = _fft_convolve(a)
    return 2.0 * a.size * float(np.max(c))

def _smooth_max(c, beta):
    """Compute logsumexp(beta*c)/beta as smooth max."""
    b = beta * c
    b_max = np.max(b)
    if b_max > 500:
        b_max = 500
    return (b_max + np.log(np.sum(np.exp(b - b_max)))) / beta

def _smooth_max_grad(c, a, beta):
    """Compute smooth max and its gradient w.r.t. a."""
    n = a.size
    b = beta * c
    b_max = np.max(b)
    if b_max > 500:
        b_max = 500
    w = np.exp(b - b_max)
    w /= w.sum()
    sm = (b_max + np.log(np.sum(w))) / beta  # = b_max/beta + log(sum(w))/beta
    
    # Gradient w.r.t. a: g[j] = 2 * sum_k w[k] * a[k-j]
    # = 2 * cross-correlation(w, a)[j]
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    w_pad = np.zeros(N, dtype=np.float64)
    w_pad[:len(w)] = w
    fw = np.fft.rfft(w_pad, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    g = 2.0 * xcorr[:n]
    
    return sm, g, w

def _obj_grad_softmax(x, beta):
    """
    Objective and gradient for softmax parameterization.
    x is the parameter vector, a = softmax(x), sum(a) = 1.
    Returns (2n * smooth_max(c), grad_x).
    """
    n = x.size
    x_shift = x - np.max(x)
    a = np.exp(x_shift)
    s = a.sum()
    a /= s
    
    c = _fft_convolve(a)
    sm, g, w = _smooth_max_grad(c, a, beta)
    
    obj = 2.0 * n * sm
    
    # grad_x[j] = a[j] * (g[j] - sum_i a[i]*g[i]) * 2n
    a_dot_g = np.dot(a, g)
    grad = 2.0 * n * a * (g - a_dot_g)
    
    return obj, grad

def _optimize_continuation(x0, n, deadline, beta_start=5.0, beta_end=2000.0, n_betas=10):
    """
    Continuation method: optimize with increasing beta.
    Returns best a found.
    """
    best_a = None
    best_score = np.inf
    
    # Beta schedule (log-spaced)
    betas = np.geomspace(beta_start, beta_end, n_betas)
    
    for beta in betas:
        if time.monotonic() > deadline:
            break
        
        # Optimize with current beta using L-BFGS
        t_start = time.monotonic()
        max_iter = max(50, int(30.0 / (time.monotonic() - t_start + 0.01) * 10) if time.monotonic() > t_start else 200)
        
        try:
            result = sco.minimize(
                _obj_grad_softmax, x0, args=(beta,),
                method='L-BFGS-B', jac=True,
                maxiter=max_iter,
                options={'maxls': 30, 'gtol': 1e-6, 'disp': False}
            )
            x0 = result.x
        except Exception as e:
            print(f"    L-BFGS failed at beta={beta}: {e}")
            break
        
        # Evaluate
        a = np.exp(x0 - np.max(x0))
        a /= a.sum()
        sc = _score_normalized(a)
        
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        print(f"    beta={beta:.1f}: score={sc:.8f}", flush=True)
    
    return best_a, best_score

def _optimize_adam(a0, deadline, beta_start=50.0, beta_end=5000.0, n_steps=5000):
    """
    Custom Adam-like optimizer in a-space (with projection to simplex).
    """
    n = a0.size
    a = a0.copy()
    best_a = a.copy()
    best_score = _score_normalized(a)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    # Learning rate schedule
    lr_start = 0.05 / n
    lr_end = 0.001 / n
    
    for t in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        # Anneal beta and lr
        frac = t / n_steps
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        # Compute gradient in a-space
        c = _fft_convolve(a)
        sm, g, w = _smooth_max_grad(c, a, beta)
        
        # Project gradient to simplex tangent (remove mean)
        g_proj = g - np.mean(g)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * g_proj
        v = beta2 * v + (1 - beta2) * g_proj ** 2
        m_hat = m / (1 - beta1 ** (t + 1))
        v_hat = v / (1 - beta2 ** (t + 1))
        
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0, None)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a /= s
        
        # Evaluate periodically
        if t % 10 == 0:
            sc = _score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score

def _optimize_top_k(a, deadline, K=20):
    """
    Focus on reducing the top K peaks of the convolution.
    Uses targeted gradient steps on the active set.
    """
    n = a.size
    best_a = a.copy()
    best_score = _score_normalized(a)
    
    it = 0
    patience = 500
    last_improve = 0
    lr = 0.1 / n
    rng = np.random.RandomState(42)
    
    while time.monotonic() < deadline:
        it += 1
        
        c = _fft_convolve(a)
        
        # Get top K lag indices
        if c.size <= K:
            top_idx = np.arange(c.size)
        else:
            top_idx = np.argpartition(c, -K)[-K:]
        
        # Soft weights on top K
        top_c = c[top_idx]
        T = 5.0
        top_shift = top_c - top_c.max()
        weights = np.exp(T * top_shift)
        weights /= weights.sum()
        
        # Build full weight vector
        w = np.zeros(c.size)
        for i, idx in enumerate(top_idx):
            w[idx] = weights[i]
        
        # Gradient
        N = _next_pow2(2 * n)
        fa = np.fft.rfft(a, N)
        w_pad = np.zeros(N, dtype=np.float64)
        w_pad[:len(w)] = w
        fw = np.fft.rfft(w_pad, N)
        xcorr = np.fft.irfft(np.conj(fa) * fw, N)
        g = 2.0 * xcorr[:n]
        
        # Project to tangent
        g_proj = g - np.mean(g)
        
        # Update
        a = a - lr * g_proj
        a = np.clip(a, 0, None)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a /= s
        
        # Evaluate
        if it % 5 == 0:
            sc = _score_normalized(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
                lr = min(lr * 1.02, 0.5 / n)
            elif it - last_improve > patience:
                # Restart with perturbation
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.01 * a.mean()
                a = a + noise
                a = np.clip(a, 0, None)
                a /= a.sum()
                patience = int(patience * 1.5)
                lr = max(lr * 0.5, 0.01 / n)
    
    return best_a, best_score

def _make_init_sine(n, amp=0.78):
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + amp * np.sin(2.0 * np.pi * i / n)) / n
    a = np.clip(a, 0, None)
    a /= a.sum()
    return a

def _make_init_random(n, seed=0):
    rng = np.random.RandomState(seed)
    a = rng.uniform(0, 1, n)
    a /= a.sum()
    return a

def _make_init_windowed_random(n, seed=0):
    rng = np.random.RandomState(seed)
    a = rng.uniform(0, 1, n)
    # Apply Hann window
    w = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / n))
    a = a * w
    a = np.clip(a, 0, None)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.ones(n) / n
    return a

def _reshape_sequence(a, target_n):
    """Reshape a sequence to target length by interpolation or truncation."""
    n = a.size
    if n == target_n:
        return a.copy()
    if n > target_n:
        # Downsample by averaging blocks
        n_blocks = n // target_n
        a_trunc = a[:n_blocks * target_n].reshape(target_n, n_blocks)
        return a_trunc.mean(axis=1)
    else:
        # Upsample by linear interpolation
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, target_n)
        return np.interp(x_new, x_old, a)

This is related to the concept of "low-autocorrelation" sequences in radar/communication theory. The optimal sequences for minimizing peak autocorrelation are related to:

- m-sequences (binary, good average autocorrelation)
- Golay complementary pairs (two sequences whose autocorrelations sum to a delta)

"""
Low-peak step-function optimization using softmax parameterization,
continuation method, L-BFGS, Adam refinement, and top-K active set search.
Multi-length and multi-restart strategy to escape local minima.
"""

import numpy as np
import scipy.optimize as sco
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score_norm(a):
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    return float(2.0 * a.size * np.max(c))


def _smooth_max_and_grad(c, a, beta):
    n = a.size
    b = beta * c
    b_max = np.max(b)
    if b_max > 500:
        b = b - b_max
        b_max = 0.0
    w = np.exp(b - b_max)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(c.size)
        w[np.argmax(c)] = 1.0
        sm = float(np.max(c))
    else:
        w /= w_sum
        sm = (b_max + np.log(w_sum)) / beta
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    w_pad = np.zeros(N, dtype=np.float64)
    w_pad[:min(len(w), N)] = w
    fw = np.fft.rfft(w_pad, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    g = 2.0 * xcorr[:n]
    return sm, g, w


def _softmax_obj_grad(x, beta):
    n = x.size
    x_shift = x - np.max(x)
    a = np.exp(x_shift)
    s = a.sum()
    a = a / s
    c = _fft_convolve(a)
    sm, g, w = _smooth_max_and_grad(c, a, beta)
    obj = 2.0 * n * sm
    a_dot_g = float(np.dot(a, g))
    grad = 2.0 * n * a * (g - a_dot_g)
    return obj, grad


def _optimize_lbfgs_continuation(x0, deadline, beta_start=5.0, beta_end=1000.0, n_stages=8):
    n = x0.size
    best_a = None
    best_score = np.inf
    betas = np.geomspace(beta_start, beta_end, n_stages)
    for beta in betas:
        if time.monotonic() > deadline:
            break
        time_left = deadline - time.monotonic()
        if time_left < 5:
            break
        max_iter = max(30, min(500, int(time_left * 10)))
        try:
            res = sco.minimize(
                _softmax_obj_grad, x0, args=(float(beta),),
                method='L-BFGS-B', jac=True, maxiter=max_iter,
                options={'maxls': 30, 'gtol': 1e-5, 'disp': False}
            )
            x0 = res.x
        except Exception:
            break
        a = np.exp(x0 - np.max(x0))
        a /= a.sum()
        sc = _score_norm(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    return best_a, best_score


def _optimize_adam(a0, deadline, beta_start=50.0, beta_end=5000.0, n_steps=5000):
    n = a0.size
    a = a0.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    lr0 = 0.1 / n
    lr_final = 0.002 / n
    for t in range(n_steps):
        if time.monotonic() > deadline:
            break
        frac = t / max(1, n_steps)
        beta = beta_start * (beta_end / beta_start) ** frac
        lr = lr0 * (lr_final / lr0) ** frac
        c = _fft_convolve(a)
        sm, g, w = _smooth_max_and_grad(c, a, beta)
        g_proj = g - np.mean(g)
        m = b1 * m + (1 - b1) * g_proj
        v = b2 * v + (1 - b2) * g_proj * g_proj
        m_hat = m / (1 - b1 ** (t + 1))
        v_hat = v / (1 - b2 ** (t + 1))
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a /= s
        if t % 5 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    return best_a, best_score


def _optimize_topk(a0, deadline, K=20, T_local=5.0):
    n = a0.size
    a = a0.copy()
    best_a = a.copy()
    best_score = _score_norm(a)
    it = 0
    patience = 800
    last_improve = 0
    lr = 0.15 / n
    rng = np.random.RandomState(31415)
    while time.monotonic() < deadline:
        it += 1
        c = _fft_convolve(a)
        if c.size <= K:
            top_idx = np.arange(c.size)
        else:
            top_idx = np.argpartition(c, -K)[-K:]
        top_c = c[top_idx]
        top_shift = top_c - top_c.max()
        weights = np.exp(T_local * top_shift)
        weights /= weights.sum()
        w = np.zeros(c.size, dtype=np.float64)
        for i, idx in enumerate(top_idx):
            w[idx] = weights[i]
        N = _next_pow2(2 * n)
        fa = np.fft.rfft(a, N)
        w_pad = np.zeros(N, dtype=np.float64)
        w_pad[:min(len(w), N)] = w
        fw = np.fft.rfft(w_pad, N)
        xcorr = np.fft.irfft(np.conj(fa) * fw, N)
        g = 2.0 * xcorr[:n]
        g_proj = g - np.mean(g)
        a = a - lr * g_proj
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a /= s
        if it % 3 == 0:
            sc = _score_norm(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
                lr = min(lr * 1.01, 0.5 / n)
            elif it - last_improve > patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.02 * a.mean()
                a = a + noise
                a = np.clip(a, 0.0, None)
                s = a.sum()
                if s > 1e-10:
                    a /= s
                patience = int(patience * 1.4)
                lr = max(lr * 0.6, 0.01 / n)
    return best_a, best_score


def _make_sine(n, amp=0.78, phase=0.0):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.sin(2.0 * np.pi * (i + phase) / n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def _make_multi_sine(n, modes):
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for freq, amp, ph in modes:
        a += amp * np.sin(2.0 * np.pi * freq * i / n + ph)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def _make_random(n, seed=0):
    rng = np.random.RandomState(seed)
    a = rng.uniform(0.1, 1.0, n)
    a /= a.sum()
    return a


def _make_windowed_random(n, seed=0):
    rng = np.random.RandomState(seed)
    a = rng.uniform(0.0, 1.0, n)
    w = 0.5 * (1.0 - np.cos(2.0 * np.pi * np.arange(n, dtype=np.float64) / n))
    a = a * w
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def _reshape_seq(a, target_n):
    n = a.size
    if n == target_n:
        return a.copy()
    if n > target_n:
        nb = n // target_n
        if nb == 0:
            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, target_n)
            return np.interp(x_new, x_old, a)
        a_t = a[:nb * target_n].reshape(target_n, nb)
        return a_t.mean(axis=1)
    else:
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, target_n)
        return np.interp(x_new, x_old, a)


def _coordinate_descent(a, deadline, n_iters_per_elem=3):
    n = a.size
    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.RandomState(7)
    it = 0
    while time.monotonic() < deadline:
        it += 1
        idx = rng.randint(n)
        c = _fft_convolve(best_a)
        g = 2.0 * c[idx] - 2.0 * np.sum(best_a)
        for _ in range(n_iters_per_elem):
            best_a[idx] = max(0.0, best_a[idx] - 0.01 * best_a.mean() * (g / np.mean(c)))
            s = best_a.sum()
            if s < 1e-10:
                best_a[idx] = best_score
                break
            best_a /= s
        sc = _score_norm(best_a)
        if sc < best_score - 1e-10:
            best_score = sc
        else:
            best_a[idx] = best_a[idx] * 1.01
            s = best_a.sum()
            best_a /= s
    return best_a, _score_norm(best_a)
