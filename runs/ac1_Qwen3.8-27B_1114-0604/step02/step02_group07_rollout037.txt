```python
"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum a)^2.
Combines gradient-based optimization with LP-guided direction finding.
Uses FFT for fast convolution, Adam for gradient steps, and HiGHS LP
for periodic high-quality direction updates.
"""

import numpy as np
import scipy.sparse
import scipy.optimize
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, c, S):
    """Objective value given n, convolution array c, and sum S."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def grad_subgrad(a, c, topk, temp_scale=0.1):
    """
    Compute subgradient of the objective.
    Returns (gradient, M, S).
    """
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    if S2 < 1e-20:
        return np.ones(n) / n, 0.0, S
    S3 = S2 * S
    k_act = min(topk, c.size)
    
    part = np.argpartition(c, -k_act)[-k_act:]
    vals = c[part].copy()
    order = np.argsort(vals)[::-1]
    part = part[order]
    vals = vals[order]
    
    top1 = float(vals[0])
    if top1 < 1e-15:
        return np.ones(n) / n, 0.0, S
    
    # Use softmax weights for the top peaks
    temp = max(temp_scale * top1, 1e-15)
    wts = np.exp((vals - top1) / temp)
    wts = wts / wts.sum()
    M = top1  # Use exact max for the objective
    
    grad_c = np.zeros(n)
    for i in range(k_act):
        wi = float(wts[i])
        if wi <= 1e-15:
            continue
        ki = int(part[i])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        grad_c[js:je + 1] += wi * 2.0 * a[js:je + 1][::-1]
    
    grad = (2.0 * n) * (grad_c / S2 - 2.0 * M / S3)
    return grad, M, S


def adam_step(a, g, m, v, t, lr):
    """Adam optimizer step (in-place)."""
    t1 = 0.9 ** t
    t2 = 0.999 ** t
    m[:] = 0.9 * m + 0.1 * g
    v[:] = 0.999 * v + 0.001 * g * g
    a -= lr * (m / (1.0 - t1)) / (np.sqrt(v / (1.0 - t2)) + 1e-8)


def solve_direction_lp(a, M, S_target=None):
    """
    Solve the directional LP:
    max sum(g) s.t. (a*g)(k) <= M for all k, g >= 0.
    Returns the optimal g.
    """
    n = len(a)
    
    # Build the convolution matrix A (2n-1) x n
    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    rows = []
    cols = []
    vals = []
    for i in range(n):
        # k ranges from i to i+n-1 (since k-i ranges from 0 to n-1)
        k_start = i
        k_end = i + n  # exclusive
        a_vals = a[:n]  # a[k-i] for k-i = 0, ..., n-1
        rows.extend(range(k_start, k_end))
        cols.extend([i] * n)
        vals.extend(a_vals.tolist())
    
    A_mat = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: min -sum(g) s.t. A_mat @ g <= M, g >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    res = scipy.optimize.linprog(
        c_obj,
        A_ub=A_mat,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method='highs'
    )
    
    if res.status == 0:
        return res.x
    else:
        return None
```

```python
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def compute_gradient(a, c, topk, temp_frac=0.05):
    n = len(a)
    S = float(a.sum())
    if S < 1e-10:
        return np.ones(n) / n
    S2 = S * S
    S3 = S2 * S
    k_act = min(topk, c.size)
    
    part = np.argpartition(c, -k_act)[-k_act:]
    vals = c[part].copy()
    order = np.argsort(vals)[::-1]
    part = part[order]
    vals = vals[order]
    
    top1 = float(vals[0])
    if top1 < 1e-15:
        return np.ones(n) / n
    
    temp = max(temp_frac * top1, 1e-15)
    wts = np.exp((vals - top1) / temp)
    wts = wts / wts.sum()
    
    M = top1
    grad_c = np.zeros(n)
    for i in range(k_act):
        wi = float(wts[i])
        if wi <= 1e-15:
            continue
        ki = int(part[i])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        grad_c[js:je+1] += wi * 2.0 * a[js:je+1][::-1]
    
    return (2.0 * n) * (grad_c / S2 - 2.0 * M / S3)

def solve_lp_direction(a, M):
    """Solve the directional LP to find a good direction."""
    n = len(a)
    if n > 3000:
        return None  # Too large for LP
    
    # Build convolution matrix
    rows = np.repeat(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    # For column i, the non-zeros are at rows i, i+1, ..., i+n-1
    # values are a[0], a[1], ..., a[n-1]
    # Actually: A[k, i] = a[k-i] for 0 <= k-i < n
    # So for column i: rows = i, i+1, ..., i+n-1, values = a[0], a[1], ..., a[n-1]
    
    row_list = []
    col_list = []
    val_list = []
    for i in range(n):
        k_start = i
        k_end = min(i + n, 2*n - 1)  # Can't exceed 2n-2
        length = k_end - k_start
        row_list.extend(range(k_start, k_end))
        col_list.extend([i] * length)
        val_list.extend(a[:length].tolist())
    
    A_mat = sp.csr_matrix((val_list, (row_list, col_list)), shape=(2*n-1, n))
    
    # Check for all-zero rows (shouldn't happen for non-negative a with positive sum)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    try:
        res = scipy.optimize.linprog(
            c_obj,
            A_ub=A_mat,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method='highs'
        )
        if res.status == 0:
            return res.x
    except Exception as e:
        print(f"LP error: {e}")
    
    return None
```

```python
Uses FFT for fast convolution and HiGHS for LP subproblems.

import scipy.sparse as sp
import scipy.optimize

def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)

def compute_gradient(a, c, topk, temp_frac=0.05):

    n = len(a)

        return np.ones(n) / n

        bc1 = 1.0 - 0.9 ** t
        bc2 = 1.0 - 0.999 ** t
        m[:] = 0.9 * m + 0.1 * g
        v[:] = 0.999 * v + 0.001 * g * g
        a -= lr * (m / bc1) / (np.sqrt(v / bc2) + 1e-8)
        a = np.clip(a, 0.0, 1000.0)

        # Learning rate decay
        if lr_decay is not None and t % lr_decay == 0:
            lr *= 0.85

    return best_score, best_a


def solve_lp_direction(a, M, n_max=1200):
    """
    Solve directional LP: max sum(g) s.t. (a*g)(k) <= M, g >= 0.
    Returns optimal g or None if infeasible/skipped.
    """
    n = len(a)
    if n > n_max:
        return None

    # Build sparse convolution matrix A (2n-1) x n
    # A[k, i] = a[k-i] if 0 <= k-i < n
    row_list = []
    col_list = []
    val_list = []
    for i in range(n):
        k_end = min(i + n, 2 * n - 1)
        length = k_end - i
        if length <= 0:
            continue
        row_list.extend(range(i, k_end))
        col_list.extend([i] * length)
        val_list.extend(a[:length].tolist())

    if not row_list:
        return None

    A_mat = sp.csr_matrix((val_list, (row_list, col_list)), shape=(2 * n - 1, n))

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M + 1e-10)

    try:
        res = scipy.optimize.linprog(
            c_obj, A_ub=A_mat, b_ub=b_ub,
            bounds=[(0, None)] * n, method='highs'
        )
        if res.status == 0:
            g = res.x
            if g is not None and g.sum() > 1e-10:
                return g
    except Exception:
        pass
    return None


def make_init_variants(n, rng, parent=None):
    """Generate diverse initial sequences."""
    inits = []
    x = np.linspace(0, 1, n)

    if parent is not None and len(parent) == n:
        inits.append(np.clip(parent, 0.0, 1000.0))

    # All ones
    inits.append(np.ones(n))
    # Cosine modulation
    inits.append(1.0 + 0.7 * np.cos(2.0 * np.pi * x))
    inits.append(1.0 + 0.7 * np.cos(4.0 * np.pi * x))
    inits.append(1.0 + 0.5 * np.cos(6.0 * np.pi * 3.0 * x))
    # Random
    inits.append(np.clip(0.5 + 0.8 * rng.random(n), 0.01, 1000.0))
    # Gapped (alternating)
    a_g = np.ones(n)
    a_g[1::2] = 0.3
    inits.append(a_g)
    a_g2 = np.ones(n)
    a_g2[1::2] = 0.6
    inits.append(a_g2)
    # Two-period pattern
    a_p = np.ones(n)
    a_p[2::4] = 0.4
    a_p[3::4] = 0.4
    inits.append(a_p)
    # Exponential decay from center
    mid = n // 2
    a_c = np.exp(-0.5 * (np.arange(n) - mid) ** 2 / (n / 4) ** 2)
    inits.append(np.clip(a_c, 0.01, 1000.0))
    # Sine wave (non-negative)
    inits.append(0.5 + 0.5 * np.sin(2.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.sin(4.0 * np.pi * x + np.pi / 4) ** 2)

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum a)^2.
    Uses adaptive subgradient (Adam) with LP-guided steps and diversified restarts.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = float(budget_s)
    deadline = start + total - 5.0

    best_score = np.inf
    best_a = None

    # Determine sequence length from parent
    n = 2048
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            parent_arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if parent_arr.size > 0:
                parent = parent_arr
                n = int(parent_arr.size)
    except Exception:
        pass
    n = min(max(256, n), 16384)
    print(f"n={n}, budget={budget_s:.0f}s, parent={'yes' if parent is not None else 'no'}")

    # ---- Phase 0: Evaluate parent ----
    if parent is not None and parent.size == n:
        a0 = np.clip(parent, 0.0, 1000.0)
        c0 = fast_self_conv(a0)
        s0 = compute_score(n, c0, float(a0.sum()))
        print(f"Parent score: {s0:.6f}")
        best_score = s0
        best_a = a0.copy()
    else:
        a0 = np.ones(n)
        best_a = a0.copy()
        best_score = 2.0  # known for all-ones

    # ---- Phase 1: Explore initializations (10% time) ----
    p1_end = start + 0.10 * (total - 5.0)
    print(f"\n--- Phase 1: Exploration ({p1_end - start:.0f}s) ---")
    variants = make_init_variants(n, rng, parent)
    # Evaluate all variants quickly
    best_init_score = best_score
    best_init_a = best_a.copy()
    for i, a_v in enumerate(variants):
        a_v = np.clip(a_v, 0.0, 1000.0)
        if a_v.sum() < 1e-3:
            continue
        c_v = fast_self_conv(a_v)
        s_v = compute_score(n, c_v, float(a_v.sum()))
        if s_v < best_init_score:
            best_init_score = s_v
            best_init_a = a_v.copy()
        if i == 0 or (i + 1) % 3 == 0:
            print(f"  Init {i}: {s_v:.6f}")
    if best_init_score < best_score:
        best_score = best_init_score
        best_a = best_init_a.copy()

    # Quick gradient refinement on best init
    sc, aa = adam_optimizer(best_a, min(p1_end, start + 0.08 * (total - 5.0)),
                            rng, lr0=0.3, topk=8, patience=100)
    if sc < best_score:
        best_score = sc
        best_a = aa.copy()
    print(f"Phase 1 best: {best_score:.6f}")

    # ---- Phase 2: Main optimization (40% time) ----
    p2_end = start + 0.50 * (total - 5.0)
    print(f"\n--- Phase 2: Main optimization ({p2_end - p1_end:.0f}s) ---")
    a = best_a.copy()

    # Use LP if n is small enough
    use_lp = n <= 1200
    if use_lp:
        print("  Using LP-guided optimization")
        lp_count = 0
        while time.monotonic() < p2_end:
            # Gradient batch
            batch_end = min(p2_end, time.monotonic() + 20)
            sc, aa = adam_optimizer(a, batch_end, rng, lr0=0.04, topk=12, patience=80)
            if sc < best_score:
                best_score = sc
                best_a = aa.copy()
            a = aa.copy()

            # LP step
            if time.monotonic() < p2_end - 15:
                c = fast_self_conv(a)
                M = float(np.max(c))
                S = float(a.sum())
                g = solve_lp_direction(a, M, n_max=n)
                if g is not None:
                    ratio = g.sum() / S
                    if ratio > 1.01:
                        g_scaled = g * S / g.sum()
                        t_blend = 0.25
                        a_new = (1 - t_blend) * a + t_blend * g_scaled
                        a_new = np.clip(a_new, 0.0, 1000.0)
                        c_new = fast_self_conv(a_new)
                        S_new = float(a_new.sum())
                        score_new = compute_score(n, c_new, S_new)
                        if score_new < best_score:
                            best_score = score_new
                            best_a = a_new.copy()
                        a = a_new.copy()
                        lp_count += 1
                        print(f"  LP {lp_count}: ratio={ratio:.3f}, score={score_new:.6f}")
                    else:
                        # LP didn't help much, perturb and retry
                        a = a * (1.0 + 0.02 * rng.standard_normal(n))
                        a = np.clip(a, 0.0, 1000.0)
                else:
                    a = a * (1.0 + 0.03 * rng.standard_normal(n))
                    a = np.clip(a, 0.0, 1000.0)
            print(f"  Current best: {best_score:.6f}")
    else:
        # For large n: multi-config gradient optimization
        print("  Using multi-config gradient optimization (n too large for LP)")
        configs = [
            (0.06, 8, 150, 0.05),
            (0.03, 16, 200, 0.04),
            (0.015, 24, 250, 0.03),
            (0.01, 16, 300, 0.02),
        ]
        for lr_c, tk_c, pat_c, tf_c in configs:
            if time.monotonic() >= p2_end:
                break
            sc, aa = adam_optimizer(a, p2_end, rng, lr0=lr_c, topk=tk_c,
                                    patience=pat_c, temp_frac=tf_c)
            if sc < best_score:
                best_score = sc
                best_a = aa.copy()
            a = aa.copy()
            print(f"  Config (lr={lr_c}, k={tk_c}): score={best_score:.6f}")

    # ---- Phase 3: Diversified restarts (25% time) ----
    p3_end = start + 0.75 * (total - 5.0)
    print(f"\n--- Phase 3: Diversified restarts ({p3_end - p2_end:.0f}s) ---")
    n_restarts = 15
    elapsed_so_far = time.monotonic() - start
    remaining_3 = max(1.0, p3_end - time.monotonic())
    per_restart = remaining_3 / n_restarts

    for r in range(n_restarts):
        if time.monotonic() >= p3_end:
            break
        noise_scale = 0.05 + 0.3 * (r / n_restarts)
        a_pert = best_a * (1.0 + noise_scale * rng.standard_normal(n))
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 1e-3:
            a_pert = best_a + 0.1 * rng.standard_normal(n)
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Vary hyperparameters per restart
        lr_r = 0.04 * (1.0 + 0.5 * rng.random())
        tk_r = int(8 + 16 * rng.random())
        sub_end = min(p3_end, time.monotonic() + per_restart)
        sc, aa = adam_optimizer(a_pert, sub_end, rng, lr0=lr_r, topk=tk_r, patience=80)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
            print(f"  Restart {r}: *** NEW BEST {sc:.6f} ***")
        else:
            print(f"  Restart {r}: {sc:.6f}")

    # ---- Phase 4: LP on shorter sequence + transfer (if time) ----
    if n > 1200 and time.monotonic() < deadline - 300:
        p4_end = deadline - 200
        print(f"\n--- Phase 4: Short-seq LP + transfer ({p4_end - (p3_end if n>1200 else time.monotonic()):.0f}s) ---")
        n_short = min(1024, n)
        # Downsample best sequence
        if n > n_short:
            idx = np.linspace(0, n - 1, n_short).astype(int)
            a_short = best_a[idx].copy()
        else:
            a_short = best_a.copy()
            n_short = n

        c_s = fast_self_conv(a_short)
        s_s = compute_score(n_short, c_s, float(a_short.sum()))
        print(f"  Short seq score (n={n_short}): {s_s:.6f}")

        # Optimize short seq with LP
        a_opt = a_short.copy()
        lp_count = 0
        while time.monotonic() < p4_end - 60:
            sc_s, aa_s = adam_optimizer(a_opt, min(p4_end - 60, time.monotonic() + 10),
                                        rng, lr0=0.03, topk=8, patience=50)
            a_opt = aa_s
            if time.monotonic() < p4_end - 60:
                c_s2 = fast_self_conv(a_opt)
                M_s = float(np.max(c_s2))
                g_s = solve_lp_direction(a_opt, M_s, n_max=n_short)
                if g_s is not None and g_s.sum() > a_opt.sum() * 1.01:
                    g_scaled = g_s * a_opt.sum() / g_s.sum()
                    a_opt = 0.7 * a_opt + 0.3 * g_scaled
                    a_opt = np.clip(a_opt, 0.0, 1000.0)
                    lp_count += 1

        # Upsample back to full length
        if n > n_short:
            from scipy.interpolate import interp1d
            x_short = np.linspace(0, 1, n_short)
            x_full = np.linspace(0, 1, n)
            f_interp = interp1d(x_short, a_opt, kind='cubic', fill_value='extrapolate')
            a_up = np.clip(f_interp(x_full), 0.0, 1000.0)
        else:
            a_up = a_opt.copy()

        # Refine the upsampled sequence
        sub_end = min(p4_end, time.monotonic() + 120)
        sc_up, aa_up = adam_optimizer(a_up, sub_end, rng, lr0=0.02, topk=16, patience=100)
        if sc_up < best_score:
            best_score = sc_up
            best_a = aa_up.copy()
            print(f"  Transfer: NEW BEST {sc_up:.6f}")
        else:
            print(f"  Transfer: {sc_up:.6f}")

    # ---- Phase 5: Final polish (remaining time) ----
    p5_end = deadline
    print(f"\n--- Phase 5: Final polish ({p5_end - (time.monotonic() - start + start):.0f}s) ---")
    if time.monotonic() < p5_end - 30:
        sc, aa = adam_optimizer(best_a, min(p5_end - 60, time.monotonic() + 90),
                                rng, lr0=0.01, topk=24, patience=300, temp_frac=0.03)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"  Polish 1: {best_score:.6f}")

    if time.monotonic() < p5_end - 30:
        sc, aa = adam_optimizer(best_a, min(p5_end - 30, time.monotonic() + 90),
                                rng, lr0=0.005, topk=32, patience=500, temp_frac=0.02)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"  Polish 2: {best_score:.6f}")

    if time.monotonic() < p5_end - 10:
        sc, aa = adam_optimizer(best_a, p5_end - 5,
                                rng, lr0=0.002, topk=48, patience=1000, temp_frac=0.01)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"  Polish 3: {best_score:.6f}")

    # Final verification and output
    if best_a is None:
        best_a = np.ones(n) / n

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    verified = np.inf
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        pass
    print(f"\nFINAL: internal={best_score:.6f}, verified={verified:.6f}, "
          f"n={n}, time={time.monotonic() - start:.1f}s")
    return out
```
