def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def _softmax_weights(d, beta):
    d = np.asarray(d, dtype=float)
    n = int(d.size)
    if n == 0:
        return np.array([], dtype=float)
    if beta <= 1e-8:
        return np.full(n, 1.0 / n, dtype=float)
    shift = float(np.max(d))
    if not np.isfinite(shift):
        return np.full(n, 1.0 / n, dtype=float)
    e = np.exp(float(beta) * (d - shift))
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros(n, dtype=float)
        w[int(np.argmax(d))] = 1.0
        return w
    return e / s

def _local_block_search(p0, time_limit, verbose=False):
    p = _sanitize(p0, _MIN_P, True)
    if p is None:
        return None, float("inf")

    time_limit = float(max(0.0, time_limit))
    if time_limit <= 0.0:
        sc, _ = _score_c_ones(p)
        return p, sc

    n = int(p.size)
    L = _conv_len(n)
    best = p.copy()
    best_sc, _ = _score_c_ones(p, L)
    cur_sc = best_sc

    deadline = time.monotonic() + time_limit
    stagnant = 0
    it = 0
    last_print = -100000

    small = max(1, n // 2048)
    mid = max(small + 1, n // 256)
    large = max(mid + 1, min(n, n // 32))
    factors = (0.5, 0.7, 0.85, 1.2, 1.5, 2.0)

    while time.monotonic() < deadline:
        r = float(_GLOBAL_RNG.random())
        if r < 0.30:
            w = 1
        elif r < 0.55:
            lo = small
            hi = min(n + 1, max(lo + 1, small * 4 + 1))
            w = int(_GLOBAL_RNG.integers(lo, hi))
        elif r < 0.85:
            lo = mid
            hi = min(n + 1, max(lo + 1, int(mid * 4) + 1))
            w = int(_GLOBAL_RNG.integers(lo, hi))
        else:
            lo = large
            hi = min(n + 1, max(lo + 1, int(large * 4) + 1))
            w = int(_GLOBAL_RNG.integers(lo, hi))

        w = max(1, min(n, w))
        start = int(_GLOBAL_RNG.integers(0, n - w + 1))

        current_best = None
        current_score = cur_sc

        for f in factors:
            cand = p.copy()
            cand[start:start + w] = cand[start:start + w] * float(f)
            cand = np.maximum(cand, _MIN_P)
            s = float(cand.sum())
            if s <= 0.0 or not np.isfinite(s):
                continue
            cand = cand / s
            sc, _ = _score_c_ones(cand, L)
            if sc < current_score - 1e-12:
                current_score = sc
                current_best = cand

        if current_best is not None:
            p = current_best
            cur_sc = current_score
            stagnant = 0
            if cur_sc < best_sc:
                best_sc = cur_sc
                best = p.copy()
                if verbose and (it - last_print >= 200):
                    print(f"local n={n} it={it} best={best_sc:.7f}")
                    last_print = it
        else:
            stagnant += 1
            if stagnant > 25:
                std = 0.05 if stagnant < 80 else 0.12
                p_try = p * np.exp(_GLOBAL_RNG.normal(0.0, std, n))
                p_try = np.maximum(p_try, _MIN_P)
                s = float(p_try.sum())
                if s > 0.0 and np.isfinite(s):
                    p_try = p_try / s
                    sc2, _ = _score_c_ones(p_try, L)
                    if np.isfinite(sc2) and sc2 <= max(best_sc * 1.02, cur_sc * 1.05):
                        p = p_try
                        cur_sc = sc2
                    else:
                        p = best.copy()
                        cur_sc = best_sc
                else:
                    p = best.copy()
                    cur_sc = best_sc
                stagnant = 0

            if stagnant > 180:
                p = best.copy()
                cur_sc = best_sc
                stagnant = 0

        it += 1

    return best, best_sc
