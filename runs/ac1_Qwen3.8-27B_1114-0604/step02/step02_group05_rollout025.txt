"""
Multi-strategy optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses log-parameterized Adam, spectral domain shaping, and local search.
Key improvements: Adam optimizer with momentum, adaptive soft-max,
spectral peak shaving, multi-scale optimization, and coordinate refinement.
"""

import numpy as np
import time


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b, L=None):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 1000000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)


def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    pn = p / s
    n = len(pn)
    c = _conv_full(pn, pn)
    return float(2.0 * n * c.max())


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s


def _softmax_weights(x, beta):
    x = np.asarray(x, dtype=float)
    n = len(x)
    if n == 0:
        return np.array([], dtype=float)
    if beta < 1e-8:
        return np.ones(n, dtype=float) / n
    x = x - x.max()
    e = np.exp(beta * x)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones(n, dtype=float) / n
    return e / s


def _grad_score(a, beta, L):
    """
    Compute gradient of 2n * soft_max(conv(a,a)) w.r.t. a.
    Returns (grad, max_c, soft_max_value).
    a should be normalized (sum=1) non-negative.
    """
    n = len(a)
    c = _conv_full(a, a)
    w = _softmax_weights(c, beta)
    max_c = float(c.max())
    sm = float(np.sum(w * c))
    # Gradient: dS/da[i] = 4n * sum_j w[j] * a[j-i] * I(0<=j-i<n)
    # = 4n * cross_corr(w, a)[i]
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    h = np.fft.irfft(W * np.conj(A), L)[:n]
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.maximum(h, 0.0)
    grad = 4.0 * n * h
    return grad, max_c, sm


def _spectral_flatten(a, power, L):
    """
    Flatten magnitude spectrum: |A|^power rescaled back.
    power < 1 reduces dynamic range (flattens).
    Returns new normalized non-negative sequence or None.
    """
    n = len(a)
    A = np.fft.rfft(a, L)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_max = mag.max()
    if mag_max < 1e-15:
        return None
    mag_norm = mag / mag_max
    mag_new = mag_norm ** power * mag_max
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, L)[:n]
    neg_frac = float(np.sum(a_new < 0)) / n
    if neg_frac > 0.4:
        return None
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s


def _spectral_peak_shave(a, ratio, L):
    """
    Shave spectral peaks: reduce |A| values above ratio*max to ratio*max.
    """
    n = len(a)
    A = np.fft.rfft(a, L)
    mag = np.abs(A)
    phase = np.angle(A)
    mag_max = mag.max()
    if mag_max < 1e-15:
        return None
    threshold = ratio * mag_max
    mag_new = np.where(mag > threshold, threshold + 0.3 * (mag - threshold), mag)
    A_new = mag_new * np.exp(1j * phase)
    a_new = np.fft.irfft(A_new, L)[:n]
    neg_frac = float(np.sum(a_new < 0)) / n
    if neg_frac > 0.4:
        return None
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s


def _spectral_blend(a1, a2, L):
    """Blend two sequences in frequency domain (geometric mean of magnitudes)."""
    n = len(a1)
    A1 = np.fft.rfft(a1, L)
    A2 = np.fft.rfft(a2, L)
    mag1 = np.abs(A1)
    mag2 = np.abs(A2)
    phase = np.angle(A1)
    # Geometric mean of magnitudes
    mag_blend = np.sqrt(np.maximum(mag1 * mag2, 0.0))
    A_blend = mag_blend * np.exp(1j * phase)
    a_blend = np.fft.irfft(A_blend, L)[:n]
    a_blend = np.maximum(a_blend, 0.0)
    s = a_blend.sum()
    if s < 1e-10:
        return None
    return a_blend / s


def _log_adam(a0, rng, time_budget, beta_start=50.0, beta_max=20000.0,
              lr_start=0.02, verbose=False):
    """
    Log-parameterized Adam optimization.
    Works with a = exp(b), optimizing b directly.
    """
    a = np.maximum(np.asarray(a0, dtype=float), 1e-30)
    a = a / a.sum()
    b = np.log(a)
    n = len(b)
    L = _next_pow2(4 * n + 10)

    best_a = a.copy()
    best_score = _score(a)
    cur_score = best_score

    beta = float(beta_start)
    lr = float(lr_start)
    M = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2 = 0.85, 0.999

    deadline = time.monotonic() + time_budget
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        # Update beta periodically
        if it % 100 == 0:
            beta = min(beta_max, beta * 1.15)

        # Compute gradient
        a_curr = np.exp(b)
        a_curr = a_curr / a_curr.sum()
        grad_a, max_c, sm = _grad_score(a_curr, beta, L)
        grad_b = a_curr * grad_a
        grad_b = np.where(np.isfinite(grad_b), grad_b, 0.0)

        # Normalize gradient for Adam stability
        gnorm = np.linalg.norm(grad_b)
        if gnorm > 1e-10:
            grad_b = grad_b / gnorm

        t += 1
        M = beta1 * M + (1.0 - beta1) * grad_b
        v = beta2 * v + (1.0 - beta2) * grad_b * grad_b
        M_hat = M / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * M_hat / (np.sqrt(v_hat) + 1e-8)

        # Try step
        b_try = b - step
        a_try = np.exp(b_try)
        a_try = np.maximum(a_try, 1e-30)
        a_try = a_try / a_try.sum()

        sc_try = _score(a_try)
        if sc_try < cur_score - 1e-14:
            b = b_try.copy()
            cur_score = sc_try
            stall = 0
            if sc_try < best_score - 1e-12:
                best_a = a_try.copy()
                best_score = sc_try
                lr = min(lr * 1.01, 0.5)
                if verbose and (it % 1000 == 0):
                    print(f"        logAdam it={it} score={best_score:.7f} beta={beta:.0f}")
        else:
            stall += 1
            if stall % 20 == 0:
                lr = max(lr * 0.9, 0.001)
            if stall > 200:
                b = np.log(np.maximum(best_a, 1e-30))
                cur_score = best_score
                M = np.zeros(n)
                v = np.zeros(n)
                t = 0
                lr = float(lr_start)
                stall = 0

        it += 1

    return best_a, best_score

def _init_uniform(n):
    return np.ones(n, dtype=float) / n


def _init_two_level(n, rng):
    k = int(n * rng.uniform(0.3, 0.7))
    ratio = float(rng.uniform(1.5, 5.0))
    p = np.ones(n, dtype=float)
    p[k:] = ratio
    idx = rng.permutation(n)
    p = p[idx]
    p = p * (1.0 + 0.1 * rng.normal(0, 1, n))
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return _init_uniform(n)
    return p / s


def _init_gaussian_mix(n, rng):
    K = int(rng.integers(2, 8))
    p = np.zeros(n, dtype=float)
    t = np.arange(n, dtype=float) / n
    for _ in range(K):
        c = rng.uniform(0.0, 1.0)
        s = rng.uniform(0.02, 0.3)
        w = rng.uniform(0.5, 2.0)
        x = (t - c) / s
        x = np.minimum(np.abs(x), 1.0 - np.abs(x))
        p += w * np.exp(-0.5 * x * x)
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _init_cosine(n, rng):
    t = np.arange(n, dtype=float) / n
    p = np.ones(n, dtype=float)
    for k in range(1, 15):
        amp = rng.uniform(-0.5, 0.5) / k
        ph = rng.uniform(0, 2 * np.pi)
        p += amp * np.cos(2 * np.pi * k * t + ph)
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return _init_uniform(n)
    return p / s


def _init_block_random(n, rng):
    K = int(rng.integers(4, 20))
    p = np.zeros(n, dtype=float)
    edges = np.sort(rng.uniform(0, 1, K - 1))
    blocks = np.concatenate([[0], edges, [1]])
    heights = rng.lognormal(0, 1, K)
    for i in range(K):
        s = int(round(blocks[i] * n))
        e = max(s + 1, int(round(blocks[i + 1] * n)))
        e = min(n, e)
        p[s:e] = heights[i]
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _resample(p_in, n_new, rng=None):
    p = _sanitize(p_in)
    if p is None:
        return None
    n_old = len(p)
    if n_old == 1:
        return np.ones(n_new, dtype=float)
    if n_old == n_new:
        result = p.copy()
    else:
        x_old = (np.arange(n_old) + 0.5) / n_old
        x_new = (np.arange(n_new) + 0.5) / n_new
        result = np.interp(x_new, x_old, p)
    result = np.maximum(result, 0)
    if rng is not None:
        result = result * np.exp(rng.normal(0, 0.02, n_new))
        result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s


def propose_candidate(seed=12345, budget_s=2400):
    """
    Search for the best non-negative sequence minimizing the evaluation score.
    Multi-strategy: log-Adam + spectral shaping + local refinement.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 2400.0

    start_time = time.monotonic()
    deadline = start_time + budget - 20

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng(iseed % (2**31 - 1))

    # Get parent
    parent = None
    try:
        parent = _sanitize(np.asarray(height_sequence_1, dtype=float))
    except Exception:
        pass

    best_overall = None
    best_score_overall = float('inf')

    def _try_update(p, score_val):
        nonlocal best_overall, best_score_overall
        if p is not None and np.isfinite(score_val) and score_val < best_score_overall:
            best_overall = p.copy()
            best_score_overall = float(score_val)
            if score_val < 1.51:
                print(f"  *** NEW BEST: {score_val:.7f} (n={len(p)}) ***")
            elif score_val < 1.52:
                print(f"  New best: {score_val:.7f} (n={len(p)})")

    # ========== PHASE 1: Exploration (first 12% of time) ==========
    print("=" * 60)
    print(f"Phase 1: Exploration (budget={budget:.0f}s)")
    t1_end = start_time + 0.12 * budget

    if parent is not None:
        sc = _score(parent)
        _try_update(parent, sc)
        print(f"  Parent: score={sc:.7f} (n={len(parent)})")

    # Try various lengths and initializations
    explore_lengths = []
    if parent is not None:
        pn = len(parent)
        explore_lengths.append(pn)
        if pn * 2 <= 12000:
            explore_lengths.append(pn * 2)
        if pn // 2 >= 512:
            explore_lengths.append(pn // 2)
    else:
        explore_lengths = [2048, 4096]

    for n_exp in sorted(set(explore_lengths)):
        if time.monotonic() > t1_end:
            break
        inits = [
            _init_uniform(n_exp),
            _init_two_level(n_exp, rng),
            _init_gaussian_mix(n_exp, rng),
            _init_cosine(n_exp, rng),
            _init_block_random(n_exp, rng),
        ]
        if parent is not None:
            sp = _resample(parent, n_exp, rng)
            if sp is not None:
                inits.append(sp)

        for sp in inits:
            sc = _score(sp)
            _try_update(sp, sc)

    print(f"  Phase 1 done. Best: {best_score_overall:.7f}")

    if best_overall is None:
        best_overall = _init_uniform(3072)
        best_score_overall = _score(best_overall)

    # ========== PHASE 2: Main optimization (next 50%) ==========
    print(f"Phase 2: Main optimization (n={len(best_overall)})")
    t2_end = start_time + 0.62 * budget

    # Primary optimization on the best
    n_main = len(best_overall)
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(0.4 * t_rem, 400.0)
        t_alloc = max(20.0, t_alloc)
        print(f"    Run 1: main hybrid (budget={t_alloc:.0f}s)")
        p, sc = _hybrid_optimize(best_overall, rng, t_alloc, beta_start=30.0, verbose=True)
        _try_update(p, sc)

    # Second run with different parameters
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(0.35 * t_rem, 300.0)
        t_alloc = max(20.0, t_alloc)
        print(f"    Run 2: aggressive beta (budget={t_alloc:.0f}s)")
        p1 = _resample(best_overall, n_main, rng)
        if p1 is None:
            p1 = best_overall.copy()
        p, sc = _hybrid_optimize(p1, rng, t_alloc, beta_start=200.0, verbose=True)
        _try_update(p, sc)

    # Try upscaled version
    t_rem = t2_end - time.monotonic()
    if t_rem > 60 and n_main * 2 <= 12000:
        n_up = n_main * 2
        t_alloc = min(0.3 * t_rem, 250.0)
        t_alloc = max(20.0, t_alloc)
        print(f"    Run 3: upscaled n={n_up} (budget={t_alloc:.0f}s)")
        p_up = _resample(best_overall, n_up, rng)
        if p_up is not None:
            p, sc = _hybrid_optimize(p_up, rng, t_alloc, beta_start=50.0, verbose=True)
            _try_update(p, sc)

    # Local refinement on current best
    t_rem = t2_end - time.monotonic()
    if t_rem > 20:
        t_alloc = min(0.3 * t_rem, 150.0)
        t_alloc = max(10.0, t_alloc)
        print(f"    Run 4: local refine (budget={t_alloc:.0f}s)")
        p, sc = _local_refine(best_overall, rng, t_alloc, verbose=True)
        _try_update(p, sc)

    print(f"  Phase 2 done. Best: {best_score_overall:.7f} (n={len(best_overall)})")

    # ========== PHASE 3: Diverse attempts (next 18%) ==========
    print("Phase 3: Diverse attempts")
    t3_end = start_time + 0.80 * budget

    n_best = len(best_overall)
    for mode in range(6):
        if time.monotonic() > t3_end - 10:
            break
        t_rem = t3_end - time.monotonic()
        if t_rem < 20:
            break
        t_alloc = min(60.0, t_rem * 0.4)
        t_alloc = max(10.0, t_alloc)

        if mode == 0:
            sp = _init_two_level(n_best, rng)
        elif mode == 1:
            sp = _init_gaussian_mix(n_best, rng)
        elif mode == 2:
            sp = _init_cosine(n_best, rng)
        elif mode == 3:
            sp = _init_block_random(n_best, rng)
        elif mode == 4:
            sp = _resample(best_overall, n_best, rng)
            if sp is None:
                sp = _init_uniform(n_best)
        else:
            # Try a completely different length
            n_alt = int(n_best * rng.choice([0.5, 1.5, 2.0]))
            n_alt = max(512, min(n_alt, 12000))
            sp = _resample(best_overall, n_alt, rng)
            if sp is None:
                sp = _init_uniform(n_alt)

        print(f"    Restart {mode}: n={len(sp)}")
        p, sc = _hybrid_optimize(sp, rng, t_alloc, beta_start=50.0, verbose=False)
        _try_update(p, sc)

    print(f"  Phase 3 done. Best: {best_score_overall:.7f}")

    # ========== PHASE 4: Final polishing (last 18%) ==========
    print("Phase 4: Final polishing")

    if best_overall is not None:
        # Extended log-Adam with high beta
        t_rem = deadline - time.monotonic()
        if t_rem > 30:
            t_alloc = min(0.4 * t_rem, 300.0)
            t_alloc = max(10.0, t_alloc)
            print(f"    Polish 1: logAdam high-beta (budget={t_alloc:.0f}s)")
            p, sc = _log_adam(best_overall, rng, t_alloc,
                             beta_start=500.0, beta_max=100000.0,
                             lr_start=0.005, verbose=True)
            _try_update(p, sc)

        # Local refinement
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.3 * t_rem, 200.0)
            t_alloc = max(10.0, t_alloc)
            print(f"    Polish 2: local refine (budget={t_alloc:.0f}s)")
            p, sc = _local_refine(best_overall, rng, t_alloc, verbose=True)
            _try_update(p, sc)

        # One more hybrid pass
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.3 * t_rem, 200.0)
            t_alloc = max(10.0, t_alloc)
            print(f"    Polish 3: final hybrid (budget={t_alloc:.0f}s)")
            p, sc = _hybrid_optimize(best_overall, rng, t_alloc, beta_start=200.0, verbose=True)
            _try_update(p, sc)

    # ========== Final output ==========
    if best_overall is None:
        best_overall = _init_uniform(3072)

    p_final = _sanitize(best_overall)
    if p_final is None:
        p_final = _init_uniform(3072)

    final_score = _score(p_final)
    try:
        eval_score = float(evaluate_sequence([float(x) for x in p_final]))
        if np.isfinite(eval_score):
            final_score = eval_score
    except Exception:
        pass

    print(f"\n{'=' * 60}")
    print(f"FINAL: score={final_score:.7f} (n={len(p_final)})")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    out = np.where(np.isfinite(p_final), p_final, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = _init_uniform(3072)
    else:
        out = out / s

    return [float(x) for x in out]
