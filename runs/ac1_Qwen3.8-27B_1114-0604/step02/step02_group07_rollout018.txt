    """
    Compute (soft-weighted) subgradient of max(c) w.r.t. a.
    Returns gradient vector of same length as a.
    """

    k = min(topk, len(c))

        part = np.arange(len(c))
    else:

        grad[m_start:m_end + 1] += wi * a[idx]

        if S_t < 1e-10:
            return np.inf

        delta = t - a[idx]
        m_start = idx  # c[j] affected for j in [idx, idx+n-1]
        m_end = min(idx + n - 1, 2 * n - 2)
        # c[j] + delta * a[idx-j]... wait
        # c[j] = sum_k a[k] * a[j-k]
        # d(c[j])/d(a[idx]) = a[j-idx] (if 0 <= j-idx < n)

        max_c = c.max()  # baseline
        # Check if any affected c[j] exceeds the baseline max
        for j in range(idx, min(idx + n, 2 * n - 1)):
            cj_t = c[j] + delta * a[j - idx]
            if cj_t > max_c:
                max_c = cj_t
        return 2.0 * n * max_c / (S_t * S_t)
    
    # Golden section
    gr = (math.sqrt(5) - 1) / 2
    for _ in range(30):
        x1 = lo + gr * (hi - lo)
        x2 = hi - gr * (hi - lo)
        f1 = eval_t(x1)
        f2 = eval_t(x2)
        if f1 < f2:

        else:
            lo = x1
    
    t_opt = (lo + hi) / 2
    a_new = a.copy()
    a_new[idx] = t_opt

def get_subgradient(a, c, topk=1, soft_temp=0.0):
    """
    Compute subgradient of max(c) with respect to a.
    Uses top-k peaks with optional soft weighting.
    Returns array of same shape as a.
    """
    n = len(a)
    kc = len(c)
    k = min(topk, kc)
    
    if k == kc:
        part = np.arange(kc)
    else:
        part = np.argpartition(c, -k)[-k:]
    
    vals = c[part]
    
    if soft_temp > 0 and k > 1:
        temp = soft_temp * float(np.max(vals))
        if temp < 1e-30:
            temp = 1e-30
        exponents = (vals - vals.max()) / temp
        wts = np.exp(exponents)
        wts_sum = wts.sum()
        if wts_sum < 1e-30:
            wts = np.zeros(k)
            wts[0] = 1.0
        else:

    log_update = -lr * grad
    log_update = np.clip(log_update, -40, 40)
    
    if entropy_alpha > 0:
        # a_new ∝ a^(1-alpha) * exp(-lr*grad)
        # log(a_new) = (1-alpha)*log(a) - lr*grad
        log_a = np.log(np.maximum(a, 1e-300))
        log_new = (1.0 - entropy_alpha) * log_a + log_update
    else:
        log_a = np.log(np.maximum(a, 1e-300))
        log_new = log_a + log_update
    
    # Normalize (in log space for stability)
    log_new -= log_new.max()
    new = np.exp(log_new)
    S = new.sum()
    if S < 1e-300:
        return a.copy()
    return new / S

def optimize_multiplicative_weights(a, deadline, rng, init_lr=0.5, topk=1, patience=500):
    """
    Main optimization loop using multiplicative weights.
    a is normalized to sum=1.
    Returns (best_score, best_a).
    """
    n = len(a)
    a = a.copy()
    S = float(a.sum())
    a = a / S  # normalize to sum 1
    
    c = fast_self_conv(a)
    score = 2.0 * n * float(np.max(c))  # S=1 so S^2=1
    best_score = score
    best_a = a.copy()
    
    lr = init_lr
    no_improve = 0
    iteration = 0
    last_print = time.monotonic()
    
    while time.monotonic() < deadline:

        grad = get_subgradient(a, c, topk=topk, soft_temp=0.05)
        
        # Multiplicative weights update
        a = mw_update(a, grad, lr, entropy_alpha=0.0)
        
        # Recompute
        c = fast_self_conv(a)
        S = float(a.sum())
        if S < 1e-10:
            a = np.ones(n) / n
            c = fast_self_conv(a)
            S = 1.0
            continue
        a = a / S
        score = 2.0 * n * float(np.max(c))
        
        iteration += 1
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:
            # Restart with perturbation
            perturbation = 0.05 + 0.1 * rng.random()
            a = best_a * (1.0 + perturbation * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            S = a.sum()
            if S < 1e-10:
                a = np.ones(n) / n
            else:
                a = a / S

            no_improve = 0

        elif no_improve > patience // 2:
            # Reduce learning rate
            lr *= 0.7
        
        # Periodic print
        if time.monotonic() - last_print > 30:
            print(f"  MW iter {iteration}: score={score:.6f}, best={best_score:.6f}, lr={lr:.4f}")
            last_print = time.monotonic()
    
    return best_score, best_a

def lp_refinement(a, rng, n_iter=200):
    """
    LP-inspired refinement using multiplicative weights on the LP dual.
    Given current a (sum=1), find direction g that increases sum while
    keeping a*g <= max(a*a).
    """
    n = len(a)
    c = fast_self_conv(a)
    M = float(np.max(c))
    
    if M < 1e-15:
        return a.copy()
    
    # Initialize g
    g = np.ones(n) / n
    
    # Multiplicative weights on the "LP": 
    # We want to maximize sum(g) s.t. (a*g)[j] <= M for all j
    # This is equivalent to: find g that "spreads out" the cross-correlation
    # Using MW: g *= exp(-lr * (a*g)/M) at the most violated constraint
    
    lr = 0.01
    for i in range(n_iter):
        # Compute cross-convolution a*g
        ag = fast_cross_conv(a, g)
        
        # Find most violated constraint
        violations = ag / M
        max_viol = np.max(violations)
        
        if max_viol < 1.0 + 1e-6:
            # Feasible, break
            break
        
        # Get the violated positions
        j_star = int(np.argmax(violations))
        
        # Update: g[m] *= exp(-lr * a[j_star - m]) for valid m
        m_lo = max(0, j_star - n + 1)
        m_hi = min(j_star, n - 1)
        if m_lo <= m_hi:
            indices = j_star - np.arange(m_lo, m_hi + 1)
            g[m_lo:m_hi+1] *= np.exp(-lr * a[indices])
        
        # Renormalize g
        g_sum = g.sum()
        if g_sum < 1e-30:
            g = np.ones(n) / n
        else:
            g = g / g_sum
    
    # Now g is approximately the LP solution
    # Blend into a
    g_normalized = g * (float(a.sum()) / float(g.sum()))
    t = 0.3  # blend parameter
    a_new = (1 - t) * a + t * g_normalized
    a_new = np.maximum(a_new, 0.0)
    S = a_new.sum()
    if S > 1e-10:
        a_new = a_new / S
    return a_new

def interpolate_to_length(a, new_n):
    """Interpolate sequence a to new length new_n using linear interpolation."""
    old_n = len(a)
    if new_n == old_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    return a_new

    A = np.fft.rfft(a, M_fft)

    A = np.fft.rfft(a, M_fft)
