```python
"""
Step function optimizer using soft-max log-sum-exp objective with analytical
gradients, optimized via L-BFGS-B with β-annealing schedule.
Uses softmax parameterization for non-negativity constraint.
"""

import numpy as np
from scipy.optimize import minimize
import time


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def _convolve_fast(a):
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def _score_and_conv(a):
    n = len(a)
    c = _convolve_fast(a)
    s = a.sum()
    if s < 0.01:
        return np.inf, c
    return float(2.0 * n * c.max() / (s * s)), c


def _soft_max_obj_and_grad(z, beta):
    """
    Compute log(sum(exp(beta*c))) and its gradient w.r.t. z.
    where c = conv(softmax(z), softmax(z))
    """
    a = _softmax(z)
    n = len(a)
    
    c = _convolve_fast(a)
    
    # Compute soft-max: log(sum(exp(beta * c)))
    bc = beta * c
    bc_shifted = bc - bc.max()
    log_sum_exp = bc.max() + np.log(np.sum(np.exp(bc_shifted)))
    
    # p[k] = exp(beta*c[k]) / sum(exp(beta*c[k']))
    p = np.exp(bc_shifted)
    p /= p.sum()
    
    # Gradient in a-space: h[m] = 2*beta * sum_j a[j] * p[j+m]
    # = 2*beta * cross_corr(a, p)[m] for m = 0, ..., n-1
    size = 1 << (3 * n - 2).bit_length()
    A = np.fft.rfft(a, size)
    P = np.fft.rfft(p, size)
    cross = np.fft.irfft(np.conj(A) * P, size)[:n]
    
    h = 2.0 * beta * cross
    
    # Convert to z-gradient: grad_z[m] = a[m] * (h[m] - sum_j h[j]*a[j])
    mean_h = np.dot(h, a)
    grad_z = a * (h - mean_h)
    
    return log_sum_exp, grad_z


def _soft_max_obj_wrapper(z, beta):
    val, _ = _soft_max_obj_and_grad(z, beta)
    return val


def _soft_max_grad_wrapper(z, beta):
    _, grad = _soft_max_obj_and_grad(z, beta)
    return grad


def _optimize_from_z(z_init, n, deadline, rng, beta_start=10, beta_end=5000, n_stages=20):
    """
    Optimize z with annealing β schedule.
    """
    best_z = z_init.copy()
    best_a = _softmax(z_init)
    best_score = _score_and_conv(best_a)[0]
    
    # β schedule: geometric from beta_start to beta_end
    betas = np.geomspace(beta_start, beta_end, n_stages)
    
    time_per_stage = None  # will be determined dynamically
    
    for stage, beta in enumerate(betas):
        if time.monotonic() > deadline:
            break
        
        # Optimize with current beta
        # Use L-BFGS-B
        result = minimize(
            _soft_max_obj_wrapper,
            best_z,
            args=(beta,),
            jac=lambda z, b: _soft_max_grad_wrapper(z, b),
            method='L-BFGS-B',
            options={'maxiter': 200, 'maxfun': 500, 'iprint': -1}
        )
        
        new_z = result.x
        new_a = _softmax(new_z)
        new_score = _score_and_conv(new_a)[0]
        
        if new_score < best_score:
            best_score = new_score
            best_z = new_z.copy()
            best_a = new_a.copy()
        
        if stage % 5 == 0:
            print(f"    Stage {stage}: beta={beta:.1f}, score={best_score:.7f}")
    
    return best_z, best_a, best_score
```
