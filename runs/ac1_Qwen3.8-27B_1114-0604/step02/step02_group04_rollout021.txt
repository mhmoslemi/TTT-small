    Returns (grad, lse_value)"""

    c_max = c.max()
    exp_c = np.exp(T * (c - c_max))
    w = exp_c / exp_c.sum()
    # grad[j] = 2 * sum_i a[i] * w[i+j]
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(np.pad(w, (0, N - w.size)), N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    grad = 2.0 * xcorr[:n]
    lse_val = c_max + np.log(exp_c.sum()) / T
    return grad, lse_val

def _adam_update(b, grad, m, v, t, beta1=0.9, beta2=0.999, lr=0.01, eps=1e-8):
    """Adam update in log-space. b = log(a) (unnormalized)."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad ** 2)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    b_new = b - lr * m_hat / (np.sqrt(v_hat) + eps)
    return b_new, m, v

    b_shift = b - b.max()
    a = np.exp(b_shift)
    s = a.sum()
    if s < 1e-300:
        return np.ones(b.size) / b.size
    return a / s

                last_improve = it

                v = np.zeros(n)
                t = 0
                lr = max(0.1, lr * 0.5)
                patience = int(patience * 1.3)

Multi-strategy low-peak step function optimizer.
Combines LSE gradient descent in log-space, active-set subgradient methods,
LP-guided steps, and local search. Uses FFT for O(n log n) convolution.
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _conv_grad_from_w(w, a):
    """Compute grad[j] = 2 * sum_i a[i] * w[i+j] via FFT cross-correlation."""
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    w_padded = np.zeros(N)
    w_padded[:w.size] = w
    fw = np.fft.rfft(w_padded, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]


def _lse_gradient(a, T):
    """Compute LSE-smoothed gradient of max convolution. Returns (grad, lse_val, conv)."""
    c = _fft_convolve(a)
    c_max = float(np.max(c))
    exp_c = np.exp(T * (c - c_max))
    s = float(np.sum(exp_c))
    if s < 1e-300:
        w = np.zeros(c.size)
        w[np.argmax(c)] = 1.0
    else:
        w = exp_c / s
    lse_val = c_max + np.log(s) / T
    grad = _conv_grad_from_w(w, a)
    return grad, lse_val, c


def _adam_step(b, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    m_new = beta1 * m + (1.0 - beta1) * grad
    v_new = beta2 * v + (1.0 - beta2) * (grad * grad)
    m_hat = m_new / (1.0 - beta1 ** t)
    v_hat = v_new / (1.0 - beta2 ** t)
    b_new = b - lr * m_hat / (np.sqrt(v_hat) + eps)
    return b_new, m_new, v_new

    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.sin(theta)) / n
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n) / n

def _optimize_lse(a_init, deadline, seed=0, T_start=5.0, T_end=30.0, lr_init=1.0):
    """LSE gradient descent in log-space with Adam."""
    rng = np.random.RandomState(seed)
    n = a_init.size
    a = a_init / a_init.sum()
    b = np.log(a + 1e-30)
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0

    best_a = a.copy()
    best_score = _score(a)
    it = 0
    last_improve = 0
    patience = 8000
    T = T_start
    lr = lr_init

    while time.monotonic() < deadline:
        it += 1
        t_adam += 1

        grad, lse_val, c = _lse_gradient(a, T)
        grad_proj = grad - np.mean(grad)

        b, m, v = _adam_step(b, grad_proj, m, v, t_adam, lr)
        a = _softmax_norm(b)

        if it % 5 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
                lr = min(lr * 1.005, 5.0)
            else:
                lr = max(lr * 0.999, 0.1)

            progress = it - last_improve
            if progress > patience * 0.7:
                T = min(T * 1.15, T_end)
            else:
                T = max(T * 0.995, T_start)

            if progress > patience:
                a = best_a.copy()
                noise_scale = 0.02 * (progress / patience)
                noise = rng.standard_normal(n) * noise_scale * a.mean()
                a = np.clip(a + noise, 1e-10, None)
                a = a / a.sum()
                b = np.log(a + 1e-30)
                m = np.zeros(n)
                v = np.zeros(n)
                t_adam = 0
                lr = max(0.5, lr * 0.6)
                patience = int(patience * 1.4)

    return best_a, best_score


def _optimize_active(a_init, deadline, seed=0, K=20, lr_init=None):
    """Active-set subgradient with multiplicative updates."""
    rng = np.random.RandomState(seed)
    n = a_init.size
    a = a_init / a_init.sum()
    if lr_init is None:
        lr_init = 0.15 * n

    best_a = a.copy()
    best_score = _score(a)
    it = 0
    last_improve = 0
    patience = 6000
    lr = lr_init

    while time.monotonic() < deadline:
        it += 1
        lr_cur = lr / (1.0 + 0.002 * it)

        c = _fft_convolve(a)
        if c.size <= K:
            active = np.arange(c.size)
        else:
            active = np.argpartition(c, -K)[-K:]

        top_vals = c[active]
        T_local = 5.0
        top_shift = top_vals - top_vals.max()
        weights = np.exp(T_local * top_shift)
        weights /= weights.sum()

        w = np.zeros(c.size)
        for idx, k in enumerate(active):
            w[k] = weights[idx]

        grad = _conv_grad_from_w(w, a)
        grad_proj = grad - np.mean(grad)

        update = -lr_cur * grad_proj
        update = np.clip(update, -5.0, 5.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        if it % 5 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
                lr = min(lr * 1.01, 1.0 * n)
            elif it - last_improve > patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.01 * a.mean()
                a = np.clip(a + noise, 0, None)
                a = a / a.sum()
                patience = int(patience * 1.3)
                lr = max(0.05 * n, lr * 0.7)

    return best_a, best_score


def _lp_guided_step(a, K=30):
    """One LP-guided step: find direction that minimizes max active convolution value."""
    n = a.size
    c = _fft_convolve(a)
    M = float(np.max(c))

    if c.size <= K:
        active = np.arange(c.size)
    else:
        active = np.argpartition(c, -K)[-K:]

    active = active[np.argsort(c[active])[::-1]]

    # Build constraint matrix: A_con[idx, j] = dc[active[idx]]/da[j]
    A_con = np.zeros((len(active), n))
    for idx, k in enumerate(active):
        valid_j = np.arange(max(0, k - n + 1), min(n, k + 1))
        valid_k = k - valid_j
        A_con[idx, valid_j] = 2.0 * a[valid_k]

    # LP: min t s.t. A_con @ d - t <= 0, -d <= a (non-negativity), sum(d) = 0
    # Variables: [d_0, ..., d_{n-1}, t]
    from scipy.optimize import linprog

    c_lp = np.zeros(n + 1)
    c_lp[-1] = 1.0

    n_rows = len(active) + n
    A_ub = np.zeros((n_rows, n + 1))
    b_ub = np.zeros(n_rows)

    A_ub[:len(active), :n] = A_con
    A_ub[:len(active), -1] = -1.0

    A_ub[len(active):, :n] = -np.eye(n)
    b_ub[len(active):] = a

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    res = linprog(c_lp, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(None, None)] * (n + 1), method='highs')

    if res.success:
        d = res.x[:n]
        a_new = a + d
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s > 0.01:
            a_new = a_new / s
            return a_new
    return None


def _local_search(a, n_trials=100, trial_size=20, deadline=None):
    """Local search: try small random perturbations."""
    rng = np.random.RandomState(12345)
    n = a.size
    best_a = a.copy()
    best_score = _score(a)
    step_size = 0.1 * a.mean()

    for trial in range(n_trials):
        if deadline is not None and time.monotonic() > deadline:
            break
        idx = rng.choice(n, size=trial_size, replace=False)
        perturb = rng.standard_normal(trial_size) * step_size
        a_new = best_a.copy()
        a_new[idx] += perturb
        a_new = np.clip(a_new, 0.0, None)
        s = a_new.sum()
        if s < 0.01:
            continue
        a_new = a_new / s
        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            step_size *= 0.9
        else:
            step_size = max(step_size * 0.98, 0.001 * best_a.mean())

    return best_a, best_score


def _scan_initizations(deadline):
    """Scan various initialization families."""
    best_a = None
    best_score = np.inf
    rng = np.random.RandomState(42)

    # Check for height_sequence_1
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, 1000)
        if hs.size > 0 and hs.sum() > 0.01:
            hs = hs / hs.sum()
            sc = _score(hs)
            if sc < best_score:
                best_score = sc
                best_a = hs.copy()
                print(f"  [init] height_sequence_1: n={hs.size}, score={sc:.8f}", flush=True)
    except NameError:
        pass

    lengths = [1024, 2048, 512]
    for n in lengths:
        if time.monotonic() > deadline:
            break
        for amp in [0.5, 0.7, 0.8, 0.9, 1.0, 1.2, 1.5]:
            if time.monotonic() > deadline:
                break
            a = _make_sine(n, amp)
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"  [init] sine n={n} amp={amp}: score={sc:.8f}", flush=True)

        # Multi-harmonic
        for _ in range(3):
            if time.monotonic() > deadline:
                break
            amps = [rng.uniform(0.5, 1.2)]
            modes = [1]
            phases = [rng.uniform(0, 0.5)]
            if rng.random() > 0.5:
                amps.append(rng.uniform(0.1, 0.4))
                modes.append(rng.choice([2, 3]))
                phases.append(rng.uniform(0, 1.0))
            a = _make_multi_sine(n, list(zip(amps, modes, phases)))
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    if best_a is None:
        best_a = _make_sine(1024, 0.8)
        best_score = _score(best_a)

    return best_a, best_score


def _optimize_multi(a_init, deadline, seed=0):
    """
    Combined optimization: alternates between LSE-Adam and active-set methods.
    Includes periodic LP-guided steps.
    """
    rng = np.random.RandomState(seed)
    n = a_init.size
    a = a_init / a_init.sum()

    best_a = a.copy()
    best_score = _score(a)

    it = 0
    last_improve = 0
    patience = 10000
    mode = 0  # 0: LSE-Adam, 1: Active-set

    # Adam state (for LSE mode)
    b = np.log(a + 1e-30)
    m_adam = np.zeros(n)
    v_adam = np.zeros(n)
    t_adam = 0
    T = 5.0
    lr_adam = 1.0

    # Active-set state
    lr_active = 0.15 * n

    while time.monotonic() < deadline:
        it += 1

        if mode == 0:
            # LSE-Adam step
            grad, lse_val, c = _lse_gradient(a, T)
            grad_proj = grad - np.mean(grad)
            b, m_adam, v_adam = _adam_step(b, grad_proj, m_adam, v_adam, t_adam + 1, lr_adam)
            t_adam += 1
            a = _softmax_norm(b)
        else:
            # Active-set step
            lr_cur = lr_active / (1.0 + 0.002 * it)
            c = _fft_convolve(a)
            K = 20
            if c.size <= K:
                active = np.arange(c.size)
            else:
                active = np.argpartition(c, -K)[-K:]
            top_vals = c[active]
            top_shift = top_vals - top_vals.max()
            weights = np.exp(5.0 * top_shift)
            weights /= weights.sum()
            w = np.zeros(c.size)
            for idx, k in enumerate(active):
                w[k] = weights[idx]
            grad = _conv_grad_from_w(w, a)
            grad_proj = grad - np.mean(grad)
            update = -lr_cur * grad_proj
            update = np.clip(update, -5.0, 5.0)
            a = a * np.exp(update)
            a = np.where(np.isfinite(a), a, 0.0)
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s < 1e-10:
                a = best_a.copy()
                b = np.log(a + 1e-30)
                break
            a = a / s

        # Evaluate periodically
        if it % 10 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
                lr_adam = min(lr_adam * 1.01, 5.0)
                lr_active = min(lr_active * 1.01, 1.0 * n)
            else:
                lr_adam = max(lr_adam * 0.998, 0.1)

            # Mode switching
            if it % 500 == 0:
                mode = 1 - mode
                if mode == 0:
                    T = 5.0 + 5.0 * (it / 20000.0)
                    T = min(T, 40.0)

            # LP-guided step every 500 iterations
            if it % 500 == 250 and n <= 2048:
                a_lp = _lp_guided_step(best_a, K=20)
                if a_lp is not None:
                    sc_lp = _score(a_lp)
                    if sc_lp < best_score:
                        best_score = sc_lp
                        best_a = a_lp.copy()
                        a = a_lp.copy()
                        b = np.log(a + 1e-30)
                        last_improve = it
                        print(f"    [LP step it={it}] score={sc_lp:.8f}", flush=True)

            # Restart logic
            progress = it - last_improve
            if progress > patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.03 * a.mean()
                a = np.clip(a + noise, 1e-10, None)
                a = a / a.sum()
                b = np.log(a + 1e-30)
                m_adam = np.zeros(n)
                v_adam = np.zeros(n)
                t_adam = 0
                lr_adam = max(0.5, lr_adam * 0.5)
                lr_active = max(0.05 * n, lr_active * 0.6)
                patience = int(patience * 1.3)
                mode = 1 - mode

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for low-peak nonnegative step function.
    Phase 1: Scan initializations.
    Phase 2: Main optimization with multiple strategies.
    Phase 3: Local search refinement.
    Phase 4: Final validation.
    """
    t0 = time.monotonic()
    hard_deadline = t0 + float(budget_s) - 15.0

    np.random.seed(seed)
    print(f"[Start] Budget: {budget_s}s", flush=True)

    # Phase 1: Find good initialization
    d1 = t0 + 45.0
    print("[Phase 1] Scanning initializations...", flush=True)
    init_a, init_score = _scan_initizations(d1)
    print(f"  Best init: n={init_a.size}, score={init_score:.8f}", flush=True)

    # Phase 2: Main optimization
    results = []

    # Strategy A: Multi-strategy from best init
    d2a = t0 + float(budget_s) * 0.45
    remaining = d2a - time.monotonic()
    if remaining > 30:
        print(f"[Phase 2a] Multi-strategy optimization ({remaining:.0f}s)...", flush=True)
        a_a, s_a = _optimize_multi(init_a, d2a, seed=seed)
        results.append(("multi_main", a_a, s_a))
        print(f"  multi_main: n={a_a.size}, score={s_a:.8f}", flush=True)

    # Strategy B: LSE-Adam with different temperature
    d2b = t0 + float(budget_s) * 0.70
    remaining = d2b - time.monotonic()
    if remaining > 30:
        print(f"[Phase 2b] LSE-Adam high-T ({remaining:.0f}s)...", flush=True)
        a_b, s_b = _optimize_lse(init_a, d2b, seed=seed + 1, T_start=10.0, T_end=60.0, lr_init=2.0)
        results.append(("lse_highT", a_b, s_b))
        print(f"  lse_highT: n={a_b.size}, score={s_b:.8f}", flush=True)

    # Strategy C: Active-set from best so far
    d2c = t0 + float(budget_s) * 0.90
    remaining = d2c - time.monotonic()
    if remaining > 30:
        best_so_far = min(results, key=lambda x: x[2])
        print(f"[Phase 2c] Active-set refinement ({remaining:.0f}s) from {best_so_far[0]}...", flush=True)
        a_c, s_c = _optimize_active(best_so_far[1], d2c, seed=seed + 2, K=15)
        results.append(("active_refine", a_c, s_c))
        print(f"  active_refine: n={a_c.size}, score={s_c:.8f}", flush=True)

    # Strategy D: Try a different length if time permits
    d2d = t0 + float(budget_s) - 60.0
    remaining = d2d - time.monotonic()
    if remaining > 60:
        n_alt = 2048 if init_a.size == 1024 else 1024
        alt_init = _make_sine(n_alt, 0.8)
        print(f"[Phase 2d] Alt length n={n_alt} ({remaining:.0f}s)...", flush=True)
        a_d, s_d = _optimize_multi(alt_init, d2d, seed=seed + 3)
        results.append(("alt_len", a_d, s_d))
        print(f"  alt_len: n={a_d.size}, score={s_d:.8f}", flush=True)

    # Phase 3: Local search on best candidates
    print("[Phase 3] Local search refinement...", flush=True)
    d3 = hard_deadline - 30.0
    results.sort(key=lambda x: x[2])

    for i, (name, arr, sc) in enumerate(results[:3]):
        remaining = d3 - time.monotonic()
        if remaining < 10:
            break
        n_trials = int(remaining * 2)
        n_trials = max(20, min(n_trials, 500))
        a_ls, s_ls = _local_search(arr, n_trials=n_trials, deadline=d3)
        if s_ls < sc:
            results[i] = (name + "_ls", a_ls, s_ls)
            print(f"  {name} -> local search: score={s_ls:.8f} (was {sc:.8f})", flush=True)

    # Phase 4: Final validation
    print("[Phase 4] Validating...", flush=True)
    results.sort(key=lambda x: x[2])

    best_out = None
    best_actual = np.inf

    for name, arr, sc in results[:5]:
        if time.monotonic() > hard_deadline:
            break
        arr_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        try:
            actual = float(evaluate_sequence(arr_list))
        except Exception as e:
            print(f"  Eval error for {name}: {e}", flush=True)
            actual = np.inf
        if np.isfinite(actual) and actual < best_actual:
            best_actual = actual
            best_out = arr_list
            print(f"  {name} (n={arr.size}): actual={actual:.8f}", flush=True)

    if best_out is None:
        best_out = [float(max(0.0, min(1000.0, x))) for x in init_a]
        try:
            best_actual = float(evaluate_sequence(best_out))
        except Exception:
            best_actual = init_score
        print(f"  Fallback: {best_actual:.8f}", flush=True)

    total_time = time.monotonic() - t0
    print(f"[Done] Time: {total_time:.1f}s, Best: {best_actual:.8f}", flush=True)
    return best_out
