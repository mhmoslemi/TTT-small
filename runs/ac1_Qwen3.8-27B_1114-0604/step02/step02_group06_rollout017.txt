Improved step function optimizer using multiplicative minimization descent (MMD),
frequency-domain flattening, and coordinate descent. Targets objective < 1.5027.

Strategy:
1. Multiple length exploration (2048-16384)
2. Softmax-weighted MMD with adaptive temperature and aggressive step sizes

4. Coordinate descent with greedy element selection for final polishing
5. Periodic random restarts to escape local minima

import numpy as np
import time

_RNG = np.random.default_rng(7)
_CONV_CACHE = {}


def _fft_len(n):
    n = int(max(1, n))
    if n in _CONV_CACHE:
        return _CONV_CACHE[n]
    need = 2 * n - 1
    L = 1
    while L < need:
        L <<= 1
    _CONV_CACHE[n] = L
    return L


def _conv_auto(a):
    """Compute linear auto-convolution of a with itself using FFT."""
    n = len(a)
    L = _fft_len(n)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)
    c = c[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score(a):
    """Compute objective: 2*n*max(conv(a,a)) / sum(a)^2."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-12:
        return float('inf')
    c = _conv_auto(a)
    return float(2.0 * n * c.max()) / (s * s)


def _score_fast(a, L):
    """Score using precomputed FFT length."""
    n = len(a)
    s = float(a.sum())
    if s < 1e-12:
        return float('inf')
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)
    c = c[:2 * n - 1]
    cm = float(np.maximum(c, 0.0).max())
    return float(2.0 * n * cm) / (s * s)


def _sanitize(a, normalize=True):
    """Ensure non-negative finite array, optionally normalize to sum=1."""
    a = np.asarray(a, dtype=float).ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if normalize:
        s = float(a.sum())
        if s <= 1e-15:
            return None
        a = a / s
    return a


def _resample(a, m):
    """Resample sequence of length n to length m."""
    a = _sanitize(a, True)
    if a is None:
        return None
    n = len(a)
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m, dtype=float) / m
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 1e-12)
    return _sanitize(out, True)


def _initial_sequence(n, rng, mode=0):
    """Generate an initial sequence of length n."""
    t = (np.arange(n, dtype=float) + 0.5) / float(n)
    if mode == 0:
        # Smooth random
        p = np.ones(n)
        for k in range(1, 15):
            amp = rng.normal(0, 1.0 / (k + 1))
            ph = rng.uniform(0, 2 * np.pi)
            p = p + amp * np.cos(2 * np.pi * k * t + ph)
        p = np.log1p(np.exp(np.clip(p, -50, 50)))
    elif mode == 1:
        # Beta distribution shape
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        tt = np.clip(t, 1e-8, 1 - 1e-8)
        p = tt ** a * (1 - tt) ** b
    elif mode == 2:
        # Gaussian bumps
        K = rng.integers(2, 6)
        p = np.zeros(n)
        for _ in range(K):
            center = rng.uniform(0, 1)
            sigma = rng.uniform(0.05, 0.4)
            w = rng.uniform(0.5, 2.0)
            p = p + w * np.exp(-((t - center) / sigma) ** 2)
    elif mode == 3:
        # Two-block (high edges, low middle)
        h = rng.uniform(1.0, 3.0)
        l = rng.uniform(0.1, 0.8)
        p = np.ones(n)
        mid = n // 2
        p[:mid] = h
        p[mid:] = l
    elif mode == 4:
        # U-shaped
        p = 1.0 / (1.0 + np.exp(-((t - 0.5) * rng.uniform(5, 20))))
    else:
        # Random blocks
        K = rng.integers(4, 12)
        heights = rng.lognormal(0, 1, K)
        edges = np.sort(rng.uniform(0, 1, K - 1))
        p = np.zeros(n)
        for i in range(K):
            s = 0 if i == 0 else int(edges[i-1] * n)
            e = int(edges[i] * n) if i < K - 1 else n
            p[s:e] = heights[i]
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 1e-8, 1000.0)
    return _sanitize(p, True)


def _softmax_weights(c, beta):
    """Compute softmax weights over convolution values."""
    shift = c.max()
    if not np.isfinite(shift):
        return np.ones(len(c)) / len(c)
    e = np.exp(beta * (c - shift))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros(len(c))
        w[c.argmax()] = 1.0
        return w
    return e / s


def _mmd_update(a, L, beta, alpha, threshold_frac):
    """One MMD step. Returns (new_a, conv, max_c)."""
    n = len(a)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    max_c = float(c.max())

    if max_c < 1e-15:
        return a, c, max_c

    # Compute weights: blend softmax and threshold
    w_soft = _softmax_weights(c, beta)

    # Threshold-based weights
    thresh = threshold_frac * max_c
    w_thresh = np.maximum(0, c - thresh)
    if w_thresh.max() > 1e-15:
        w_thresh = w_thresh / w_thresh.sum()
    else:
        w_thresh = w_soft

    # Blend
    w = 0.5 * w_soft + 0.5 * w_thresh

    # Compute gradient direction via cross-correlation
    W = np.fft.rfft(w, L)
    h = np.fft.irfft(W * np.conjugate(F), L)[:n]
    h = np.maximum(h, 0.0)
    hm = h.max()
    if hm < 1e-300:
        return a, c, max_c
    h = h / hm

    # Try multiple step sizes
    best_a = None
    best_sc = float(_score_fast(a, L))
    for mult in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        a_try = a * np.exp(-alpha * mult * h)
        a_try = np.maximum(a_try, 1e-15)
        s = a_try.sum()
        if s < 1e-15:
            continue
        a_try = a_try / s
        sc = _score_fast(a_try, L)
        if sc < best_sc:
            best_sc = sc
            best_a = a_try

    if best_a is not None:
        return best_a, c, max_c
    return a, c, max_c


def _freq_flatten_update(a, L, target_level):
    """Frequency-domain peak shaving: clip the convolution and back-propagate."""
    n = len(a)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)

    # Target: clip c at target_level
    abs_avg = float(c.sum()) / (2 * n - 1)
    clip_level = max(target_level, abs_avg * 1.5)

    c_clipped = np.minimum(c, clip_level)

    # Compute the "correction" in frequency domain
    # We want C_new(f) such that IDFT(C_new) = c_clipped
    # But C_new = |A_new|^2, and we preserve phase of A
    Lc = _fft_len(n)  # For the convolution length
    C_target = np.fft.rfft(c_clipped, L)
    # Current power spectrum
    C_curr = np.abs(F) ** 2

    # Interpolate: A_new = exp(0.5 * log(C_target) * phase(A_curr))
    # To keep it simple and stable, scale the amplitude
    ratio = np.sqrt(np.maximum(C_target, 1e-30) / np.maximum(C_curr, 1e-30))
    ratio = np.clip(ratio, 0.1, 10.0)

    A_new = F * ratio
    a_new = np.fft.irfft(A_new, L)[:n]
    a_new = np.maximum(a_new, 1e-15)
    s = a_new.sum()
    if s < 1e-15:
        return a, c, float(c.max())
    a_new = a_new / s
    return a_new, c, float(c.max())


def _coordinate_descent(a, L, time_limit):
    """Local coordinate descent: adjust individual elements to reduce peak."""
    n = len(a)
    deadline = time.monotonic() + time_limit
    best_a = a.copy()
    best_sc = _score_fast(a, L)
    cur_sc = best_sc
    stagnant = 0

    while time.monotonic() < deadline:
        c = _conv_auto(a)
        k_star = int(np.argmax(c))
        peak_val = c[k_star]

        # Find which j's contribute to c[k_star]
        # c[k_star] = sum_j a[j] * a[k_star - j] (valid j: max(0,k_star-n+1)..min(k_star,n-1))
        j_min = max(0, k_star - n + 1)
        j_max = min(k_star, n - 1)

        # Try reducing the j with largest contribution
        best_j = -1
        best_new_sc = cur_sc

        # Sample a subset of contributing j's
        candidates = np.arange(j_min, j_max + 1)
        if len(candidates) > 50:
            candidates = _RNG.choice(candidates, 50, replace=False)

        for j in candidates:
            partner = k_star - j
            if partner < 0 or partner >= n:
                continue
            contrib = a[j] * a[partner]
            if contrib < 1e-15:
                continue

            # Try reducing a[j] slightly and redistributing
            for frac in (0.01, 0.03, 0.05, 0.1):
                reduction = a[j] * frac
                if reduction < 1e-15:
                    continue
                a_try = a.copy()
                a_try[j] = max(0, a_try[j] - reduction)
                # Add mass to a random non-contributing position
                # Find a position that doesn't contribute to c[k_star]
                mask = np.ones(n, dtype=bool)
                mask[np.arange(max(0, k_star - n + 1), min(k_star, n - 1) + 1)] = False
                mask[j] = False
                if mask.sum() == 0:
                    continue
                j2 = _RNG.choice(np.where(mask)[0])
                a_try[j2] += reduction
                a_try = np.maximum(a_try, 1e-15)
                s = a_try.sum()
                if s < 1e-15:
                    continue
                a_try = a_try / s
                sc = _score_fast(a_try, L)
                if sc < best_new_sc:
                    best_new_sc = sc
                    best_j = j
                    best_a_try = a_try

        if best_j >= 0 and best_new_sc < cur_sc - 1e-12:
            a = best_a_try
            cur_sc = best_new_sc
            stagnant = 0
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
        else:
            stagnant += 1
            if stagnant % 50 == 0:
                # Add small noise
                noise = np.exp(_RNG.normal(0, 0.03, n))
                a_try = a * noise
                a_try = _sanitize(a_try, True)
                if a_try is not None:
                    sc = _score_fast(a_try, L)
                    if sc < cur_sc * 1.02:
                        a = a_try
                        cur_sc = sc
            if stagnant > 300:
                a = best_a.copy()
                cur_sc = best_sc
                stagnant = 0
                break

    return best_a, best_sc


def _optimize_sequence(a, time_limit, target=1.500, verbose=False):
    """Main optimization loop combining MMD and frequency flattening."""
    a = _sanitize(a, True)
    if a is None:
        return None, float('inf')
    n = len(a)
    L = _fft_len(n)
    deadline = time.monotonic() + time_limit

    best_a = a.copy()
    best_sc = _score_fast(a, L)
    cur_sc = best_sc

    beta = 30.0
    alpha = 0.08
    threshold_frac = 0.7
    stall = 0
    it = 0
    use_freq = False
    last_improve = 0

    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 2.0:
            break

        # Occasionally use frequency-domain update
        if it % 30 == 25:
            use_freq = True
        else:
            use_freq = False

        if use_freq:
            c_work = _conv_auto(a)
            abs_avg = float(c_work.sum()) / (2 * n - 1)
            target_level = abs_avg * 1.8
            a_new, c, max_c = _freq_flatten_update(a, L, target_level)
            # Blend with current
            for t_blend in (0.3, 0.5, 0.7):
                a_try = (1 - t_blend) * a + t_blend * a_new
                a_try = _sanitize(a_try, True)
                if a_try is None:
                    continue
                sc_try = _score_fast(a_try, L)
                if sc_try < cur_sc:
                    a = a_try
                    cur_sc = sc_try
                    break

        else:
            a_new, c, max_c = _mmd_update(a, L, beta, alpha, threshold_frac)
            a = a_new
            cur_sc = _score_fast(a, L)

        # Track best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            stall = 0
            last_improve = it
            if verbose and (it - last_improve) < 5 and it % 100 == 0:
                print(f"  n={n} it={it} score={best_sc:.7f}")
        else:
            stall += 1

        # Adapt parameters
        if it % 20 == 0:
            beta = min(3000.0, beta * 1.04)
            threshold_frac = min(0.95, threshold_frac * 1.01)

        # Exploration
        if stall > 15:
            noise = np.exp(_RNG.normal(0, 0.04 + 0.002 * min(stall, 50), n))
            a_try = a * noise
            a_try = _sanitize(a_try, True)
            if a_try is not None:
                sc_try = _score_fast(a_try, L)
                if sc_try < max(best_sc * 1.01, cur_sc * 1.05):
                    a = a_try
                    cur_sc = sc_try
            stall = 0

        if stall > 80:
            a = best_a.copy()
            cur_sc = best_sc
            beta = max(20.0, beta * 0.3)
            alpha = max(0.02, alpha * 0.5)
            threshold_frac = max(0.5, threshold_frac * 0.7)
            stall = 0

        if stall > 200:
            # Bigger restart
            noise = np.exp(_RNG.normal(0, 0.15, n))
            a_try = a * noise
            a_try = _sanitize(a_try, True)
            if a_try is not None:
                a = a_try
                cur_sc = _score_fast(a, L)
            else:
                a = best_a.copy()
                cur_sc = best_sc
            beta = 30.0
            alpha = 0.08
            stall = 0

        it += 1

    return best_a, best_sc


def _try_length(n_init, time_limit, parent, verbose=False):
    """Try optimizing at a specific length."""
    n = int(n_init)
    a_start = _resample(parent, n) if parent is not None else None

    if a_start is None:
        a_start = _initial_sequence(n, _RNG, 0)

    # Try a couple of different initializations
    results = []

    # From parent
    p, sc = _optimize_sequence(a_start, time_limit * 0.5, verbose=verbose)
    if p is not None:
        results.append((sc, p))

    # From random
    a_rand = _initial_sequence(n, _RNG, _RNG.integers(0, 5))
    p2, sc2 = _optimize_sequence(a_rand, time_limit * 0.3, verbose=verbose)
    if p2 is not None:
        results.append((sc2, p2))

    # Coordinate descent on best
    if results:
        results.sort(key=lambda x: x[0])
        best_sc, best_p = results[0]
        # Final coordinate descent polish
        L = _fft_len(n)
        rem = time_limit - (time_limit * 0.8)
        if rem > 5:
            p3, sc3 = _coordinate_descent(best_p, L, rem)
            results.append((sc3, p3))
        results.sort(key=lambda x: x[0])
        return results[0]

    return (float('inf'), None)


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Main search function. Tries multiple sequence lengths and optimization
    strategies to find the best non-negative sequence minimizing
    2*n*max(conv(a,a))/sum(a)^2.
    """
    global _RNG

    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except (TypeError, ValueError):
        budget = 2400.0
    if not np.isfinite(budget) or budget < 120.0:
        budget = 120.0
    total_time = max(60.0, budget - 30.0)
    deadline = start_time + total_time

    try:
        iseed = int(seed)
    except (TypeError, ValueError):
        iseed = 7
    _RNG = np.random.default_rng((iseed % 1000000007) + 1)

    # Get parent sequence
    parent = None
    try:
        parent_raw = height_sequence_1
        p = _sanitize(parent_raw, True)
        if p is not None:
            parent = p
            sc = _score(p)
            print(f"Parent sequence: n={len(p)}, score={sc:.7f}")
    except NameError:
        print("No parent sequence available, using random init.")
    except Exception as e:
        print(f"Error loading parent: {e}")

    if parent is None:
        parent = _initial_sequence(2048, _RNG, 0)
        print(f"Random parent: n={len(parent)}, score={_score(parent):.7f}")

    all_results = []

    # Phase 1: Try the parent's length and nearby lengths
    n_parent = len(parent)
    lengths_to_try = set()

    # Always try parent length
    lengths_to_try.add(n_parent)

    # Add some other lengths based on parent size
    if n_parent <= 2048:
        lengths_to_try.update([2048, 4096, 8192])
    elif n_parent <= 4096:
        lengths_to_try.update([4096, 8192, 16384])
    elif n_parent <= 8192:
        lengths_to_try.update([8192, 16384, 4096])
    else:
        lengths_to_try.update([16384, 8192, 4096])

    # Sort by priority (parent length first, then descending for larger is better)
    def _length_priority(n):
        if n == n_parent:
            return (0, n)
        if n > n_parent:
            return (1, -n)
        return (2, -n)

    lengths_to_try = sorted(lengths_to_try, key=_length_priority)

    # Allocate time to each length
    n_lengths = len(lengths_to_try)
    # Give more time to parent length and larger lengths
    weights = []
    for i, n in enumerate(lengths_to_try):
        if n == n_parent:
            weights.append(0.35)
        elif n > 4096:
            weights.append(0.25)
        elif n > 2048:
            weights.append(0.20)
        else:
            weights.append(0.15)
    total_w = sum(weights)
    weights = [w / total_w for w in weights]

    verbose = total_time > 600

    for i, n in enumerate(lengths_to_try):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = weights[i] * total_time
        t_alloc = min(t_alloc, rem - 20)
        if t_alloc < 15:
            t_alloc = max(10, rem - 10)
        if t_alloc < 5:
            break

        print(f"\nTrying length n={n}, time_alloc={t_alloc:.1f}s")
        result = _try_length(n, t_alloc, parent, verbose=verbose)
        if result[0] < float('inf'):
            all_results.append(result)
            print(f"  Best at n={n}: score={result[0]:.7f}")

    # Phase 2: Focus time on the best result with coordinate descent
    if all_results:
        all_results.sort(key=lambda x: x[0])
        best_sc, best_arr = all_results[0]
        n_best = len(best_arr)
        L_best = _fft_len(n_best)

        print(f"\nBest so far: n={n_best}, score={best_sc:.7f}")

        rem = deadline - time.monotonic()
        if rem > 30:
            # Extended coordinate descent
            t_cd = min(rem - 10, max(30, rem * 0.4))
            print(f"Coordinate descent phase: {t_cd:.1f}s")
            p_cd, sc_cd = _coordinate_descent(best_arr, L_best, t_cd)
            if sc_cd < best_sc:
                best_sc, best_arr = sc_cd, p_cd
                print(f"  After CD: score={best_sc:.7f}")

        # Final MMD polish
        rem = deadline - time.monotonic()
        if rem > 20:
            t_mmd = min(rem - 5, max(15, rem * 0.3))
            print(f"Final MMD polish: {t_mmd:.1f}s")
            p_mmd, sc_mmd = _optimize_sequence(best_arr, t_mmd, verbose=False)
            if sc_mmd < best_sc:
                best_sc, best_arr = sc_mmd, p_mmd
                print(f"  After final MMD: score={best_sc:.7f}")

        # One more coordinate descent
        rem = deadline - time.monotonic()
        if rem > 15:
            p_cd2, sc_cd2 = _coordinate_descent(best_arr, _fft_len(len(best_arr)), rem - 5)
            if sc_cd2 < best_sc:
                best_sc, best_arr = sc_cd2, p_cd2
                print(f"  Final CD: score={best_sc:.7f}")

    # Fallback
    if not all_results or all_results[0][1] is None:
        print("WARNING: No valid result found, returning uniform sequence.")
        best_arr = np.ones(2048, dtype=float) / 2048.0
        best_sc = _score(best_arr)

    # Final validation with evaluate_sequence if available
    final = np.asarray(best_arr, dtype=float).ravel()
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.clip(final, 0.0, 1000.0)
    s = float(final.sum())
    if s < 1e-12:
        final = np.ones(2048, dtype=float) / 2048.0
    else:
        final = final / s

    # Try to verify with the official evaluator (cheap for reasonable n)
    if len(final) <= 16384 and (deadline - time.monotonic()) > 5:
        try:
            official_sc = float(evaluate_sequence([float(x) for x in final]))
            if np.isfinite(official_sc):
                print(f"\nOfficial evaluation: {official_sc:.7f}")
            else:
                print(f"\nOfficial evaluation returned non-finite: {official_sc}")
        except Exception as e:
            print(f"Could not call official evaluator: {e}")

    print(f"\nFinal: n={len(final)}, internal_score={_score(final):.7f}")
    return [float(x) for x in final]
