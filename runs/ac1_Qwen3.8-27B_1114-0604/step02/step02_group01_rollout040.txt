def _pair_candidate(active_vals, active, a, i, j, delta):
    ki = active - i
    valid_i = (ki >= 0) & (ki < len(a))
    gi = np.empty_like(active_vals)
    gi.fill(0.0)
    gi[valid_i] = a[ki[valid_i]]
    kj = active - j
    valid_j = (kj >= 0) & (kj < len(a))
    gj = np.empty_like(active_vals)
    gj.fill(0.0)
    gj[valid_j] = a[kj[valid_j]]
    q = np.zeros_like(active_vals)

    vals = active_vals + 2.0*delta*(gj - gi) + delta*delta*q
    return float(np.max(vals)), vals

```python
"""
Optimizer for minimizing 2*n*max(convolve(a,a))/(sum(a))^2 for nonnegative a.

The program is centered on several complementary local/global search engines:
  1. A cutting-plane linear program that constructs a strong direction g by
     maximizing sum(g) subject to f*g <= ||f*f||_inf.  The active constraint
     set is grown by violated constraints.
  2. Smooth active-set mirror descent using softmax/inverse-gap weighted peaks.
  3. A very fast incremental pairwise transfer polish using exact updates of a
     small set of top convolution lags.
  4. A low-dimensional multiplicative modulation search to escape small basins.

The sequence is optimized in the scale-invariant scaling sum(a) = sqrt(2n), so
the returned official objective is essentially max(a*a).
"""

import numpy as np
import time
import math

try:
    from scipy.optimize import linprog
    HAS_LINPROG = True
except Exception:
    linprog = None
    HAS_LINPROG = False


def _fft_size(need):
    if need <= 1:
        return 1
    return 1 << (need - 1).bit_length()


def convolve_fft(a, b=None):
    """Linear convolution via FFT. For two real sequences this is A(z)B(z)."""
    if b is None:
        b = a
    na = len(a)
    nb = len(b)
    need = na + nb - 1
    if need <= 0:
        return np.zeros(0, dtype=np.float64)
    size = _fft_size(need)
    A = np.fft.rfft(a, size)
    if b is a:
        B = A
    else:
        B = np.fft.rfft(b, size)
    out = np.fft.irfft(A * B, size)
    return out[:need]


def normalize_sum(x, target):
    """Clip to nonnegative finite values and scale to sum target."""
    x = np.asarray(x, dtype=np.float64).copy()
    x = np.nan_to_num(x, nan=0.0, posinf=1000.0, neginf=0.0)
    np.maximum(x, 0.0, out=x)
    np.minimum(x, 1000.0, out=x)
    s = float(x.sum())
    if not math.isfinite(s) or s <= 1e-300:
        x[:] = target / max(1, len(x))
        return x
    x *= target / s
    return x


def score_sequence(a):
    """Official objective, computed with FFT. Scale invariant."""
    n = len(a)
    s = float(np.sum(a))
    if s <= 1e-12:
        return float("inf")
    c = convolve_fft(a)
    m = float(np.max(c))
    if not math.isfinite(m) or m < 0.0:
        m = 0.0
    return float(2.0 * n * m / (s * s))


def top_indices(values, k):
    """Return indices of the k largest values, sorted descending."""
    values = np.asarray(values, dtype=np.float64)
    values = np.nan_to_num(values, nan=-1e300, posinf=1e300, neginf=-1e300)
    k = int(k)
    if k <= 0:
        return np.empty(0, dtype=int)
    if k >= values.size:
        return np.arange(values.size, dtype=int)
    idx = np.argpartition(-values, k - 1)[:k]
    idx = idx[np.argsort(-values[idx])]
    return idx.astype(int, copy=False)


def softmax_weights(vals, beta, gap_mix=0.5):
    """
    Weights on active peaks. A stable softmax is blended with inverse-gap
    weights so that near-active lags are not ignored completely.
    """
    vals = np.asarray(vals, dtype=np.float64)
    if vals.size == 0:
        return np.empty(0, dtype=np.float64)
    vmax = float(vals.max())
    if not math.isfinite(beta) or beta <= 0.0:
        w_soft = np.ones(vals.shape, dtype=np.float64) / vals.size
    else:
        z = beta * (vals - vmax)
        z = np.clip(z, -50.0, 0.0)
        w_soft = np.exp(z, dtype=np.float64)
        s = float(w_soft.sum())
        if s <= 1e-300:
            w_soft = np.ones(vals.shape, dtype=np.float64) / vals.size
        else:
            w_soft /= s
    gap = (vmax - vals) + 1e-30
    w_gap = 1.0 / gap
    s = float(w_gap.sum())
    if s > 1e-300:
        w_gap /= s
    else:
        w_gap = np.ones(vals.shape, dtype=np.float64) / vals.size
    gm = max(0.0, min(1.0, gap_mix))
    w = (1.0 - gm) * w_soft + gm * w_gap
    s = float(w.sum())
    if s <= 1e-300:
        return np.ones(vals.shape, dtype=np.float64) / vals.size
    return w / s


def weighted_gradient(a, active, weights):
    """
    Gradient of sum_k weights[k] * conv(a,a)[active[k]] with respect to a.
    Uses the exact derivative for linear convolution with itself:
        d c_k / d a_i = 2 a_{k-i} when 0 <= k-i < n.
    """
    n = len(a)
    grad = np.zeros(n, dtype=np.float64)
    for kk, ww in zip(active, weights):
        if not math.isfinite(ww) or ww <= 0.0:
            continue
        k = int(kk)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[lo:hi + 1] += (2.0 * ww) * a[k - idx]
    return grad


def mirror_update(a, grad, step, momentum=None, beta=0.0):
    """
    Exponentiated-gradient (mirror) step on the positive simplex, scaled to
    the current target sum.  step controls the maximum change in log-coordinates.
    """
    target = float(a.sum())
    if target <= 1e-300:
        gradient = np.asarray(grad, dtype=np.float64).copy()
        n = len(a)
        new_mom = gradient.copy()
        return np.full(n, target / n), new_mom

    gradient = np.asarray(grad, dtype=np.float64)
    if momentum is None:
        g = gradient
        new_mom = gradient.copy()
    else:
        g = gradient + beta * momentum
        new_mom = gradient + beta * momentum

    span = float(np.max(np.abs(g)))
    if not math.isfinite(span) or span <= 1e-300:
        return normalize_sum(a, target), new_mom

    eta = step / span
    loga = np.log(np.maximum(a, 1e-300))
    loga -= eta * g
    loga -= float(np.max(loga))
    x = np.exp(loga)
    s = float(x.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return normalize_sum(a, target), new_mom
    x *= target / s
    return x, new_mom


def structured_perturb(a, rng, scale=0.1, mode=None):
    """A few harmless nonnegative transformations used for exploration."""
    n = len(a)
    target = float(a.sum())
    if n <= 1:
        return normalize_sum(a, target)
    x = np.maximum(a, 0.0).copy()
    if mode is None:
        mode = int(rng.integers(0, 7))
    mode = int(mode) % 7

    j = np.arange(n, dtype=np.float64)
    t = j / max(1, n - 1)
    sc = max(1e-6, float(scale))

    if mode == 0:
        alpha = rng.uniform(-1.0, 1.0)
        x *= np.exp(alpha * (t - 0.5) * 2.0 * sc)
    elif mode == 1:
        f = int(rng.integers(1, max(2, n // 4) + 1))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.uniform(0.05, 0.35) * max(0.5, sc)
        x *= (1.0 + amp * np.sin(2.0 * np.pi * f * t + ph))
    elif mode == 2:
        c1 = float(rng.uniform(0.1, 0.9))
        c2 = float(rng.uniform(0.1, 0.9))
        bw = float(rng.uniform(0.05, 0.18))
        amp = float(rng.uniform(0.05, 0.35) * max(0.5, sc))
        x *= (1.0 + amp * (np.exp(-((t - c1) / bw) ** 2)
                           + np.exp(-((t - c2) / bw) ** 2)))
    elif mode == 3:
        i1 = int(rng.integers(0, max(1, n // 2)))
        i2 = int(rng.integers(max(1, n // 2), n))
        amt = a[i1] * float(rng.uniform(0.03, 0.25)) * max(0.5, sc)
        x[i1] = max(0.0, x[i1] - amt)
        x[i2] += amt
    elif mode == 4:
        x += a.mean() * sc * rng.normal(0.0, 1.0, size=n)
    elif mode == 5:
        f = int(rng.integers(1, max(2, n // 8) + 1))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        amp = rng.uniform(0.03, 0.2) * max(0.5, sc)
        x *= (1.0 + amp * np.cos(2.0 * np.pi * f * t + ph))
    else:
        # Gentle exponential tilt + small noise.
        alpha = rng.uniform(-0.5, 0.5)
        x *= np.exp(theta * (t - 0.5) * 2.0 * sc)
        x += a.mean() * 0.2 * sc * rng.normal(0.0, 1.0, size=n)

    return normalize_sum(x, target)


def _lp_direction(f, active, m_bound, upper):
    """
    Solve the primal LP:
        maximize sum(g)
        subject to  (f*g)[k] <= m_bound for k in active, 0 <= g <= upper.
    """
    if not HAS_LINPROG:
        return None
    n = len(f)
    q = len(active)
    if q <= 0 or n <= 0:
        return None

    A = np.zeros((q, n), dtype=np.float64)
    for r, kk in enumerate(active):
        k = int(kk)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1, dtype=int)
        A[r, lo:hi + 1] = f[k - idx]

    c_obj = -np.ones(n, dtype=np.float64)
    b = np.full(q, m_bound, dtype=np.float64)
    bounds = np.zeros((n, 2), dtype=np.float64)
    bounds[:, 1] = upper

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    except Exception:
        try:
            res = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds)
        except Exception:
            return None

    if res is None or not getattr(res, "success", False):
        return None
    x = getattr(res, "x", None)
    if x is None:
        return None
    x = np.asarray(x, dtype=np.float64)
    if x.size != n:
        return None
    x = np.nan_to_num(x, nan=0.0, posinf=upper, neginf=0.0)
    x = np.maximum(x, 0.0)
    return x


def _heuristic_direction(f, active, m_bound, upper):
    """Fallback direction if LP fails: large where f is locally small."""
    n = len(f)
    rowmax = np.zeros(n, dtype=np.float64)
    for kk in active:
        k = int(kk)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo > hi:
            continue
        idx = np.arange(lo, hi + 1, dtype=int)
        rowmax[lo:hi + 1] = np.maximum(rowmax[lo:hi + 1], f[k - idx])
    eps = max(1e-30, float(np.max(rowmax)) * 1e-8, m_bound * 1e-12, 1e-12)
    g = 1.0 / (rowmax + eps)
    g = np.minimum(g, upper)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-300:
        g[:] = upper
    return g


def lp_search(a0, time_budget, rng, max_constraints=180, upper=None):
    """
    Iterative LP directions from the public state-of-art idea, using a growing
    active set and line search in the original nonlinear objective.
    """
    time_budget = float(time_budget)
    if time_budget <= 5.0:
        sc = score_sequence(a0)
        return a0, sc

    n = len(a0)
    target = math.sqrt(2.0 * n)
    a = normalize_sum(a0, target)
    c = convolve_fft(a)
    max_c = float(np.max(c))
    a_score = score_sequence(a)
    best = a.copy()
    best_score = a_score

    if upper is None:
        upper = max(20.0, 0.6 * target)

    deadline = time.monotonic() + time_budget
    no_improve = 0

    while time.monotonic() < deadline - 5.0:
        if not HAS_LINPROG:
            break
        if a_score <= 1e-14:
            return best, best_score

        c = convolve_fft(a)
        max_c = float(np.max(c))
        a_score = score_sequence(a)

        active = top_indices(c, min(80, max_constraints))
        best_p = None
        best_raw_sum = target * (1.0 + 1e-9)

        for _ in range(12):
            if time.monotonic() > deadline - 10.0:
                break
            if len(active) > max_constraints:
                active = top_indices(c, max_constraints)

            raw = _lp_direction(a, active, max_c, upper)
            if raw is None:
                raw = _heuristic_direction(a, active, max_c, upper)
            if raw is None:
                break

            raw_sum = float(raw.sum())
            if not math.isfinite(raw_sum) or raw_sum <= target * (1.0 + 1e-7):
                break

            p = raw / raw_sum
            pm = float(p.max())
            if pm > 0.25:
                p = np.minimum(p, 0.25)
                ps = float(p.sum())
                if ps <= 1e-300:
                    break
                p /= ps

            # Check the full constraint set for the unnormalized direction
            # raw = raw_sum * p.  If a constraint is violated, add it to the
            # active set and solve again.
            h = convolve_fft(a, p) * raw_sum
            tol = 1e-8 * max(1.0, max_c)
            viol = np.where(h > max_c + tol)[0]

            if raw_sum > best_raw_sum:
                best_raw_sum = raw_sum
                best_p = p

            if len(viol) == 0 or len(active) >= max_constraints:
                break

            if len(active) + len(viol) <= max_constraints:
                active = np.unique(np.concatenate((active, viol))).astype(int)
            else:
                union = np.unique(np.concatenate((active, viol))).astype(int)
                vals = np.maximum(h[union], c[union])
                order = np.argsort(-vals)[:max_constraints]
                active = union[order]

        if best_p is None or best_raw_sum <= target * (1.0 + 1e-6):
            no_improve += 1
            if no_improve >= 2:
                break
            a = structured_perturb(best, rng, 0.08, int(rng.integers(0, 6)))
            a_score = score_sequence(a)
            continue

        g_target = target * best_p
        cur_score = a_score
        cur = a.copy()
        best_candidate = a.copy()
        best_candidate_score = cur_score

        for tval in (0.02, 0.05, 0.10, 0.15, 0.20, 0.30, 0.50):
            cand = (1.0 - tval) * a + tval * g_target
            cand = normalize_sum(cand, target)
            sc = score_sequence(cand)
            if sc < best_candidate_score - 1e-12:
                best_candidate_score = sc
                best_candidate = cand

        if best_candidate_score < cur_score - 1e-12:
            a = best_candidate
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= 3:
                scale = 0.05 + 0.02 * min(3, no_improve)
                a = structured_perturb(best, rng, scale, int(rng.integers(0, 6)))
            if no_improve >= 6:
                break
        a_score = score_sequence(a)
        if a_score < best_score - 1e-12:
            best_score = a_score
            best = a.copy()

    sc = score_sequence(best)
    return best, sc


def active_mirror_optimize(a0, time_budget, rng, K=48, beta_scale=20.0, step0=0.80, gap_mix=0.5):
    """
    Mirror descent on the positive simplex, using smooth weights over the top K
    convolution peaks.  Adaptive step control keeps only improving moves.
    """
    time_budget = float(time_budget)
    if time_budget <= 5.0:
        return a0, score_sequence(a0)

    n = len(a0)
    target = math.sqrt(2.0 * n)
    a = normalize_sum(a0, target)
    c = convolve_fft(a)
    cur_score = score_sequence(a)
    best = a.copy()
    best_score = cur_score

    deadline = time.monotonic() + time_budget
    step = float(step0)
    momentum = None
    mom_beta = 0.35
    stall = 0
    patience = 180
    accept_count = 0
    K = max(8, min(int(K), 2 * n - 1))

    while time.monotonic() < deadline - 2.0:
        c = convolve_fft(a)
        cm = float(np.max(c))
        cur_score = score_sequence(a)
        if cur_score < best_score:
            best_score = cur_score
            best = a.copy()

        active = top_indices(c, K)
        if active.size == 0:
            break
        active_vals = c[active]
        beta = beta_scale / max(cm, 1e-30)
        w = softmax_weights(active_vals, beta, gap_mix=gap_mix)
        grad = weighted_gradient(a, active, w)

        if float(np.max(np.abs(grad))) <= 1e-300:
            break

        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.125):
            a_try, mom_try = mirror_update(a, grad, step * mult, momentum, mom_beta)
            c_try = convolve_fft(a_try)
            sc_try = score_sequence(a_try)
            if sc_try < cur_score - 1e-14:
                a = a_try
                momentum = mom_try
                cur_score = sc_try
                accepted = True
                break

        if accepted:
            accept_count += 1
            step = min(2.0, step * 1.02)
            stall = 0
            if momentum is not None and accept_count % 120 == 0:
                momentum *= 0.85
            # If the trajectory drifts noticeably above the cached best, reseed.
            if accept_count % 240 == 0 and cur_score > best_score * 1.01 + 1e-8:
                a = structured_perturb(best, rng, 0.06, int(rng.integers(0, 6)))
                momentum = None
                step = step0
                cur_score = score_sequence(a)
                stall = 0
        else:
            step = max(0.03, step * 0.6)
            stall += 1
            if step < 0.05 or stall > patience:
                a = structured_perturb(best, rng, 0.12, int(rng.integers(0, 6)))
                momentum = None
                step = step0
                stall = 0
                cur_score = score_sequence(a)

    sc = score_sequence(a)
    if sc < best_score:
        best_score = sc
        best = a.copy()
    return best, best_score


def _pair_candidate(active_vals, active, a, i, j, delta):
    """
    Exact updated values on the active lags after moving delta mass from i to j.
    Returns (updated_max, updated_active_vals).
    """
    n = len(a)
    k = np.asarray(active, dtype=np.int64)

    ki = k - i
    mi = (ki >= 0) & (ki < n)
    gi = np.zeros(active_vals.shape, dtype=np.float64)
    if np.any(mi):
        gi[mi] = a[ki[mi]]

    kj = k - j
    mj = (kj >= 0) & (kj < n)
    gj = np.zeros(active_vals.shape, dtype=np.float64)
    if np.any(mj):
        gj[mj] = a[kj[mj]]

    q = np.zeros(active_vals.shape, dtype=np.float64)
    q[k == (2 * i)] += 1.0
    q[k == (i + j)] -= 2.0
    q[k == (2 * j)] += 1.0

    new_vals = active_vals + 2.0 * delta * (gj - gi) + delta * delta * q
    return float(np.max(new_vals)), new_vals


def _make_pair_pools(a, active, active_vals, rng):
    """Build donor/receiver pools from weighted derivatives on active peaks."""
    n = len(a)
    if active.size == 0:
        return np.zeros(n), np.empty(0, dtype=int), np.empty(0, dtype=int)

    vmax = float(active_vals.max())
    gap = vmax - active_vals + 1e-12
    w = 1.0 / gap
    ws = float(w.sum())
    if ws > 0.0:
        w /= ws
    else:
        w = np.ones(active.size, dtype=np.float64) / active.size

    r = weighted_gradient(a, active, w)
    # Tiny regularization to avoid completely dead donors.
    r += 1e-12 * np.abs(a)

    ndonor = min(120, n)
    nrecv_low = min(320, n)
    rand_recv = min(120, n)

    donor_pool = np.argsort(-r)[:ndonor].astype(int, copy=False)
    recv_low = np.argsort(r)[:nrecv_low].astype(int, copy=False)
    if nrecv_low < n:
        rand = rng.choice(n, size=rand_recv, replace=False).astype(int, copy=False)
        recv_pool = np.unique(np.concatenate((recv_low, rand))).astype(int, copy=False)
    else:
        recv_pool = recv_low.copy()

    return r, donor_pool, recv_pool


def pair_polish(a0, time_budget, rng, K=64):
    """
    Fast incremental pairwise transfer polish.  It only tracks a small active
    set of top convolution lags and accepts moves that reduce their maximum.
    Full FFT scores are used periodically to validate and refresh best.
    """
    time_budget = float(time_budget)
    if time_budget <= 5.0:
        return a0, score_sequence(a0)

    n = len(a0)
    target = math.sqrt(2.0 * n)
    a = normalize_sum(a0, target)
    c = convolve_fft(a)
    cur_score = score_sequence(a)
    best = a.copy()
    best_score = cur_score
    K = max(16, min(int(K), 2 * n - 1))

    active = top_indices(c, K)
    active_vals = c[active].copy()
    max_active = float(active_vals.max())
    _, donor_pool, recv_pool = _make_pair_pools(a, active, active_vals, rng)

    deadline = time.monotonic() + time_budget
    attempts = 0
    no_accept = 0
    full_check_counter = 0
    last_check = time.monotonic()
    delta_fracs = (0.10, 0.20, 0.30, 0.50, 0.70)

    def _refresh():
        nonlocal c, active, active_vals, max_active, donor_pool, recv_pool, no_accept
        s = float(a.sum())
        if not math.isfinite(s) or s <= 1e-12:
            a[:] = target / n
        else:
            a *= target / s
        c = convolve_fft(a)
        cur_sc = score_sequence(a)
        if cur_sc < best_score:
            best_score = cur_sc
            best = a.copy()
        if cur_sc > best_score + 1e-4 and no_accept > 80:
            a = best.copy()
            c = convolve_fft(a)
            cur_sc = best_score
        active = top_indices(c, K)
        active_vals = c[active].copy()
        max_active = float(active_vals.max())
        _, donor_pool, recv_pool = _make_pair_pools(a, active, active_vals, rng)
        no_accept = 0
        last_check = time.monotonic()

    while time.monotonic() < deadline - 2.0:
        attempts += 1

        if full_check_counter >= 18 or (time.monotonic() - last_check) > 1.2:
            full_check_counter = 0
            _refresh()
            if time.monotonic() > deadline - 2.0:
                break
            if donor_pool.size == 0 or recv_pool.size == 0:
                break

        i = 0
        if donor_pool.size > 0 and a[donor_pool[0]] <= 1e-14:
            # Choose a donor with enough mass.
            i = int(np.argmax(a))
        else:
            i = int(donor_pool[full_check_counter % max(1, donor_pool.size)])
            if a[i] <= 1e-14:
                i = int(np.argmax(a))

        best_new_vals = None
        best_new_max = max_active
        best_i = -1
        best_j = -1
        best_delta = 0.0

        ai = float(a[i])
        if ai > 1e-14 and recv_pool.size > 0:
            # A small set of informed receivers plus a few random receivers.
            n_take = min(8, recv_pool.size)
            start = int((full_check_counter * 3) % max(1, recv_pool.size - n_take + 1))
            recv_slice = recv_pool[start:start + n_take]
            if recv_pool.size >= 2:
                rand_r = rng.choice(recv_pool, size=min(4, recv_pool.size), replace=True)
                recv_candidates = np.unique(np.concatenate((recv_slice, rand_r)))
            else:
                recv_candidates = recv_slice

            for jj in recv_candidates:
                j = int(jj)
                if j == i:
                    continue
                for frac in delta_fracs:
                    d = ai * frac
                    if d <= 1e-15 or d > ai:
                        continue
                    nm, nvals = _pair_candidate(active_vals, active, a, i, j, d)
                    if nm < best_new_max - 1e-15:
                        best_new_max = nm
                        best_new_vals = nvals
                        best_i = i
                        best_j = j
                        best_delta = d

        accepted_move = False
        if best_new_vals is not None:
            a[best_i] -= best_delta
            a[best_j] += best_delta
            active_vals = best_new_vals
            max_active = best_new_max
            no_accept = 0
            full_check_counter += 1
            accepted_move = True
        else:
            # Fallback random micro-move.
            i2 = int(rng.integers(n))
            j2 = int(rng.integers(n))
            if i2 == j2 or a[i2] <= 1e-14:
                i2 = int(np.argmax(a))
                if j2 == i2:
                    j2 = (i2 + 1) % n
            d = a[i2] * float(rng.uniform(0.005, 0.04))
            if d > 1e-15:
                nm, nvals = _pair_candidate(active_vals, active, a, i2, j2, d)
                if nm < max_active - 1e-15:
                    a[i2] -= d
                    a[j2] += d
                    active_vals = nvals
                    max_active = nm
                    no_accept = 0
                    full_check_counter += 1
                    accepted_move = True

        if not accepted_move:
            no_accept += 1
            if no_accept > 1000:
                a = structured_perturb(best, rng, 0.08, int(rng.integers(0, 6)))
                best_score = score_sequence(best)
                _refresh()
                no_accept = 0

    _refresh()
    c = convolve_fft(a)
    cur_score = score_sequence(a)
    if cur_score < best_score:
        best_score = cur_score
        best = a.copy()
    return best, best_score


def _lowdim_make_candidate(a0, bases, theta, target, scale):
    """Multiplicative low-frequency modulation of a base sequence."""
    theta = np.asarray(theta, dtype=np.float64)
    if bases.size == 0:
        return a0.copy()
    mod = np.exp(scale * (bases @ theta))
    x = a0 * mod
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-300:
        return a0.copy()
    x *= target / s
    return x


def lowdim_search(a0, time_budget, rng, nfreq=7):
    """
    Low-dimensional coordinate search over smooth multiplicative modulations.
    This is intentionally short and is used to escape shallow local basins.
    """
    time_budget = float(time_budget)
    if time_budget <= 5.0:
        return a0, score_sequence(a0)

    n = len(a0)
    target = math.sqrt(2.0 * n)
    a0 = normalize_sum(a0, target)

    j = np.arange(n, dtype=np.float64) / float(n)
    bases = []
    for f in range(1, nfreq + 1):
        ph1 = rng.uniform(0.0, 2.0 * np.pi)
        ph2 = rng.uniform(0.0, 2.0 * np.pi)
        bases.append(0.5 * np.sin(2.0 * np.pi * f * j + ph1)
                     + 0.5 * np.cos(2.0 * np.pi * f * j + ph2))
    bases.append(j - 0.5)
    bases.append(np.abs(j - 0.5))
    bases = np.array(bases, dtype=np.float64)
    bases -= bases.mean(axis=1, keepdims=True)
    m = bases.shape[0]

    theta = np.zeros(m, dtype=np.float64)
    cur = _lowdim_make_candidate(a0, bases, theta, target, 0.25)
    cur_score = score_sequence(cur)
    best = cur.copy()
    best_score = cur_score
    best_theta = theta.copy()

    deadline = time.monotonic() + time_budget
    step = 0.20
    stuck = 0
    scale = 0.35

    while time.monotonic() < deadline - 2.0:
        improved = False
        for d in range(m):
            if time.monotonic() > deadline - 2.0:
                break
            for sign in (1.0, -1.0):
                trial = best_theta.copy()
                trial[d] += sign * step * float(rng.uniform(0.4, 1.0))
                trial = np.clip(trial, -1.5, 1.5)
                cand = _lowdim_make_candidate(a0, bases, trial, target, scale)
                sc = score_sequence(cand)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best = cand
                    best_theta = trial
                    improved = True
                    break
            if improved:
                break

        if improved:
            stuck = 0
            step = min(0.6, step * 1.02)
        else:
            stuck += 1
            step = max(0.01, step * 0.75)
            if stuck > 25:
                # Random moderate kick.
                kick = rng.normal(0.0, 0.15, size=m)
                trial = np.clip(best_theta + kick, -1.2, 1.2)
                cand = _lowdim_make_candidate(a0, bases, trial, target, scale)
                sc = score_sequence(cand)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best = cand
                    best_theta = trial
                stuck = 0
                step = 0.15

    return best, best_score


def _phase_budget(usable, frac, deadline, reserve):
    rem = max(0.0, float(deadline - time.monotonic()))
    b = frac * usable
    b = min(b, max(0.0, rem - reserve))
    return max(0.0, b)

        starts.append(("u6", u6))
        starts.append(("sine", sine))
        starts.append(("botched", botched))
        starts.append(("rand", rng.exponential(1.0, size=n) + 0.05))

    scored_starts = []
    for nm, xx in starts:
        xx = normalize_sum(xx, target)
        sc = score_sequence(xx)
        scored_starts.append((sc, nm, xx))
    scored_starts.sort(key=lambda z: z[0])
    if scored_starts[0][0] < best_sc:
        best_sc = scored_starts[0][0]
        best_a = scored_starts[0][2].copy()
        print(f"Better initial start {scored_starts[0][1]}: {best_sc:.9f}")

    # Search phases.  Time budgets are fractions of the usable budget, but
    # because each phase monitors the deadline, unused time cascades forward.
    reserve = 50.0

    b = _phase_budget(usable, 0.28, deadline, reserve)
    if HAS_LINPROG and b > 30.0:
        print(f"\n[Phase LP-1] budget={b:.0f}s, current={best_sc:.9f}")
        t1 = time.monotonic()
        best_a, best_sc = lp_search(best_a, b, rng, max_constraints=180)
        print(f"[Phase LP-1] done in {time.monotonic()-t1:.1f}s, score={best_sc:.9f}")
        sys.stdout.flush()

    b = _phase_budget(usable, 0.20, deadline, reserve)
    if b > 20.0:
        print(f"\n[Phase MIRROR-1] budget={b:.0f}s, current={best_sc:.9f}")
        t1 = time.monotonic()
        best_a, best_sc = active_mirror_optimize(
            best_a, b, rng, K=48, beta_scale=20.0, step0=0.8, gap_mix=0.5
        )
        print(f"[Phase MIRROR-1] done in {time.monotonic()-t1:.1f}s, score={best_sc:.9f}")
        sys.stdout.flush()

    b = _phase_budget(usable, 0.23, deadline, reserve)
    if b > 20.0:
        print(f"\n[Phase PAIR-1] budget={b:.0f}s, current={best_sc:.9f}")
        t1 = time.monotonic()
        best_a, best_sc = pair_polish(best_a, b, rng, K=64)
        print(f"[Phase PAIR-1] done in {time.monotonic()-t1:.1f}s, score={best_sc:.9f}")
        sys.stdout.flush()

    b = _phase_budget(usable, 0.12, deadline, reserve)
    if HAS_LINPROG and b > 30.0:
        print(f"\n[Phase LP-2] budget={b:.0f}s, current={best_sc:.9f}")
        t1 = time.monotonic()
        best_a, best_sc = lp_search(best_a, b, rng, max_constraints=220)
        print(f"[Phase LP-2] done in {time.monotonic()-t1:.1f}s, score={best_sc:.9f}")
        sys.stdout.flush()

    b = _phase_budget(usable, 0.08, deadline, reserve)
    if b > 20.0:
        print(f"\n[Phase MIRROR-2] budget={b:.0f}s, current={best_sc:.9f}")
        t1 = time.monotonic()
        best_a, best_sc = active_mirror_optimize(
            best_a, b, rng, K=32, beta_scale=60.0, step0=1.1, gap_mix=0.35
        )
        print(f"[Phase MIRROR-2] done in {time.monotonic()-t1:.1f}s, score={best_sc:.9f}")
        sys.stdout.flush()

    b = _phase_budget(usable, 0.05, deadline, reserve)
    if b > 15.0:
        print(f"\n[Phase LOWDIM] budget={b:.0f}s, current={best_sc:.9f}")
        t1 = time.monotonic()
        best_a, best_sc = lowdim_search(best_a, b, rng, nfreq=7)
        print(f"[Phase LOWDIM] done in {time.monotonic()-t1:.1f}s, score={best_sc:.9f}")
        sys.stdout.flush()

    # Final polish with whatever remains.
    rem = deadline - 35.0 - time.monotonic()
    if rem > 20.0:
        print(f"\n[Phase PAIR-FINAL] budget={rem:.0f}s, current={best_sc:.9f}")
        t1 = time.monotonic()
        best_a, best_sc = pair_polish(best_a, rem, rng, K=48)
        print(f"[Phase PAIR-FINAL] done in {time.monotonic()-t1:.1f}s, score={best_sc:.9f}")
        sys.stdout.flush()

    # Ensure scale and finiteness; score is invariant to scale, but this keeps
    # the returned official sum in a comfortable range.
    target_final = math.sqrt(2.0 * len(best_a))
    best_a = normalize_sum(best_a, target_final)

    # Small guard: if numerical artifacts produced non-finite values (should not
    # happen), fall back to a uniform sequence.
    if not np.all(np.isfinite(best_a)):
        best_a = np.full(len(best_a), target_final / len(best_a), dtype=np.float64)

    final_list = [float(x) for x in best_a]

    try:
        official_score = evaluate_sequence(final_list)
        print(f"\nOfficial evaluation: {official_score:.9f}")
    except NameError:
        official_score = best_sc
        print(f"\nOfficial evaluator not available; using fast score {best_sc:.9f}")

    if not math.isfinite(official_score) and math.isfinite(best_sc):
        # In a pathological environment, return the best FFT-optimized sequence
        # anyway; it is finite and nonnegative.
        pass

    print(f"Total wall time: {time.monotonic()-t0:.1f}s")
    return final_list
```
