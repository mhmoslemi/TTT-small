import numpy as np

def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 4000000:
        return np.convolve(a, b).astype(float)
    L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)


def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())

def _top_T_subgradient(p, c, T=32):
    """Compute subgradient of max of top-T convolution values w.r.t. p."""
    n = len(p)
    # Find top T peaks (with some margin)
    n_conv = len(c)
    T_eff = min(T, n_conv)
    top_idx = np.argsort(c)[-T_eff:]
    # Average gradient over these peaks
    # dc_k/da_m = a_{k-m} * (2 if k-m != m else 1) for valid k-m
    # We compute: g_m = (1/T_eff) * sum_{k in top_idx} dc_k/da_m
    g = np.zeros(n, dtype=float)
    for k in top_idx:
        # Valid m range: 0 <= k-m < n, i.e., max(0,k-n+1) <= m <= min(k, n-1)
        m_lo = max(0, k - n + 1)
        m_hi = min(k, n - 1)

        vals = p[m_lo:m_hi + 1]

            factors[mid - m_lo] = 1.0
        g[m_lo:m_hi + 1] += vals * factors
    g /= T_eff
    g = np.maximum(g, 0.0)

def _spectral_compress(p, gamma, L=None):

        return None, float('inf')
    n = len(p)
    L_fft = _next_pow2(2 * n - 1)

    best = p.copy()
    best_score = _score(p)
    cur_score = best_score

    alpha = 0.3
    T = max(8, n // 50)
    beta_T = T  # current T
    deadline = time.monotonic() + time_budget
    stall = 0
    it = 0
    last_improve = 0

    while time.monotonic() < deadline:
        c = _conv_full(p, p)
        cmax = c.max()
        if cmax < 1e-15:
            break

        r = rng.random()

        if r < 0.50:
            # Exact subgradient step with adaptive T
            # Adaptive T: increase T when flat, decrease when peaked
            top_vals = np.sort(c)[-min(50, len(c)):]
            frac_above_mean = (top_vals > c.mean() * 1.5).sum() / len(top_vals)
            target_T = max(4, int(n * (0.01 + 0.04 * (1 - frac_above_mean))))
            beta_T = int(0.9 * beta_T + 0.1 * target_T)
            beta_T = max(4, min(beta_T, n // 4))

            # Try multiple step sizes
            best_trial = None
            best_trial_sc = cur_score
            for a_try in (alpha * 0.5, alpha, alpha * 1.5, alpha * 2.0):
                a_try = float(np.clip(a_try, 0.01, 3.0))
                p_try = _gradient_step_exact(p, c, beta_T, a_try)
                sc_try = _score(p_try)
                if sc_try < best_trial_sc - 1e-14:
                    best_trial = p_try
                    best_trial_sc = sc_try

            if best_trial is not None:
                p = best_trial
                cur_score = best_trial_sc
                stall = 0
                if cur_score < best_score - 1e-13:
                    best = p.copy()
                    best_score = cur_score
                    last_improve = it
                    if verbose and (it - last_improve) < 5:
                        print(f"      iter={it}, score={best_score:.7f}, T={beta_T}, alpha={alpha:.4f}")
                alpha = min(alpha * 1.02, 2.0)
            else:
                stall += 1
                alpha = max(0.05, alpha * 0.85)
                if stall > 15:
                    p = best.copy()
                    cur_score = best_score
                    alpha = 0.3
                    stall = 5

        elif r < 0.65:
            # Spectral compression
            gamma = float(rng.uniform(0.6, 0.95))
            p_new = _spectral_compress(p, gamma, L_fft)
            if p_new is not None:
                sc_new = _score(p_new)
                if sc_new < cur_score - 1e-13:
                    p = p_new
                    cur_score = sc_new
                    stall = 0
                    if sc_new < best_score - 1e-13:
                        best = p.copy()
                        best_score = sc_new
                        if verbose:
                            print(f"      SPEC iter={it}, gamma={gamma:.3f}, score={best_score:.7f}")
                else:
                    stall += 1
            else:
                stall += 1

        elif r < 0.75:
            # Spectral shaping (interpolate to flat)
            t_flat = float(rng.uniform(0.05, 0.3))
            p_new = _spectral_shape(p, t_flat, L_fft)
            if p_new is not None:
                sc_new = _score(p_new)
                if sc_new < cur_score - 1e-13:
                    p = p_new
                    cur_score = sc_new
                    stall = 0
                    if sc_new < best_score - 1e-13:
                        best = p.copy()
                        best_score = sc_new
                else:
                    stall += 1
            else:
                stall += 1

        elif r < 0.85:
            # Block scaling
            if stall > 3:
                bs_choices = [4, 8, 16, 32, 64, 128, 256]
                bs_choices = [b for b in bs_choices if b <= n]
                if bs_choices:
                    bs = int(rng.choice(bs_choices))
                    start = int(rng.integers(0, n - bs + 1))
                    factor = float(rng.choice([0.3, 0.5, 0.7, 1.4, 1.8, 2.5, 3.5]))
                    p_try = p.copy()
                    p_try[start:start + bs] *= factor
                    p_try = _sanitize(p_try)
                    if p_try is not None:
                        sc_try = _score(p_try)
                        if sc_try < cur_score - 1e-13:
                            p = p_try
                            cur_score = sc_try
                            stall = 0
                            if sc_try < best_score - 1e-13:
                                best = p.copy()
                                best_score = sc_try
                        else:
                            stall += 1
            else:
                stall += 1

        else:
            # Random perturbation (for escaping local minima)
            if stall > 8:
                std = 0.03 + 0.01 * min(stall, 20)
                p_try = p * np.exp(rng.normal(0, std, n))
                p_try = _sanitize(p_try)
                if p_try is not None:
                    sc_try = _score(p_try)
                    if sc_try < best_score * 1.05:
                        p = p_try
                        cur_score = sc_try
                        stall = 0
            else:
                stall += 1

        # Periodic pullback
        if cur_score > best_score * 1.10 and best_score < float('inf') and stall > 20:
            p = best.copy()
            cur_score = best_score
            stall = 5
            alpha = 0.3

        it += 1

    return best, best_score


def _optimize_sa(p0, rng, time_budget, verbose=False):
    """
    Simulated annealing with structured moves: block scale, block swap,
    spectral compression, and tilt modification.
    """
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    L_fft = _next_pow2(2 * n - 1)

    best = p.copy()
    best_score = _score(p)
    cur_score = best_score

    T_sa = 0.02  # initial temperature
    T_min = 0.001
    deadline = time.monotonic() + time_budget
    it = 0

    while time.monotonic() < deadline:
        r = rng.random()
        p_try = p.copy()

        if r < 0.30:
            # Block scale
            bs = int(rng.choice([4, 8, 16, 32, 64, 128, 256, 512]))
            bs = min(bs, n)
            start = int(rng.integers(0, n - bs + 1))
            factor = float(rng.uniform(0.3, 3.0))
            p_try[start:start + bs] *= factor

        elif r < 0.50:
            # Block swap
            bs = int(rng.choice([8, 16, 32, 64, 128]))
            bs = min(bs, n // 2)
            s1 = int(rng.integers(0, n - 2 * bs))
            s2 = int(rng.integers(s1 + bs, n - bs))
            p_try = p.copy()
            p_try[s1:s1 + bs], p_try[s2:s2 + bs] = p_try[s2:s2 + bs].copy(), p_try[s1:s1 + bs].copy()

        elif r < 0.65:
            # Spectral compression
            gamma = float(rng.uniform(0.5, 0.95))
            p_new = _spectral_compress(p, gamma, L_fft)
            if p_new is not None:
                p_try = p_new
            else:
                stall = 1
                it += 1
                continue

        elif r < 0.80:
            # Tilt modification
            t = np.arange(n, dtype=float) / n
            tilt = float(rng.uniform(-0.5, 0.5))
            p_try = p * np.exp(tilt * (t - 0.5))

        else:
            # Gaussian perturbation
            std = float(rng.uniform(0.02, 0.1))
            p_try = p * np.exp(rng.normal(0, std, n))

        p_try = _sanitize(p_try)
        if p_try is None:
            it += 1
            continue

        sc_try = _score(p_try)

        # Acceptance
        delta = sc_try - cur_score
        if delta < 0:
            accept = True
        else:
            accept = rng.random() < np.exp(-delta / max(T_sa, T_min))

        if accept:
            p = p_try
            cur_score = sc_try
            if cur_score < best_score - 1e-13:
                best = p.copy()
                best_score = cur_score
                if verbose:
                    print(f"      SA iter={it}, score={best_score:.7f}, T={T_sa:.5f}")

        # Cool down
        T_sa *= 0.997
        if T_sa < T_min:
            # Reheat
            T_sa = 0.01
            p = best.copy()
            cur_score = best_score

        it += 1

    return best, best_score


def _block_refine(p0, rng, time_budget):
    """Focused block refinement for final polishing."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    best = p.copy()
    best_score = _score(p)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0
    factors = (0.4, 0.6, 0.8, 1.25, 1.5, 2.0, 3.0)

    while time.monotonic() < deadline and stall < 200:
        # Focus on regions that contribute to the max
        c = _conv_full(p, p)
        cmax_idx = c.argmax()
        # The peak at cmax_idx involves pairs (i, cmax_idx - i)
        # Focus blocks around the "critical" region
        if cmax_idx < n:
            center = cmax_idx // 2
        else:
            center = (cmax_idx - n + 1) // 2
        center = max(0, min(n - 1, center))

        r = rng.random()
        if r < 0.4:
            # Block near the critical region
            bs = int(rng.integers(2, max(3, n // 64)))
            start = int(np.clip(center + rng.integers(-n // 4, n // 4), 0, n - bs))
        else:
            bs = int(rng.integers(2, max(3, min(n // 16, 256))))
            start = int(rng.integers(0, n - bs))
        bs = min	bs, n)
        start = max(0, min(start, n - bs))

        improved = False
        for f in factors:
            p_try = p.copy()
            p_try[start:start + bs] *= f
            p_try = _sanitize(p_try)
            if p_try is None:
                continue
            sc = _score(p_try)
            if sc < cur_score - 1e-14:
                cur_score = sc
                p = p_try
                improved = True
                break

        if improved:
            if cur_score < best_score - 1e-13:
                best = p.copy()
                best_score = cur_score
            stall = 0
        else:
            stall += 1
            if stall > 50:
                p = best.copy()
                cur_score = best_score
                stall = 0

    return best, best_score


def _adaptive_T_optimize(p0, rng, time_budget, verbose=False):
    """
    Optimizer with adaptive T that starts wide (exploratory) and narrows
    (exploitative) over time.
    """
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    L_fft = _next_pow2(2 * n - 1)

    best = p.copy()
    best_score = _score(p)
    cur_score = best_score

    deadline = time.monotonic() + time_budget
    stall = 0
    it = 0
    alpha = 0.4

    while time.monotonic() < deadline:
        progress_frac = (it) / max(1, int((time_budget) * 50))  # rough progress
        T = max(4, int(n * (0.1 * (1 - progress_frac) + 0.005)))

        c = _conv_full(p, p)
        cmax = c.max()
        if cmax < 1e-15:
            break

        r = rng.random()

        if r < 0.60:
            # Subgradient step
            best_trial = None
            best_trial_sc = cur_score
            for a_try in (alpha * 0.5, alpha, alpha * 2.0):
                a_try = float(np.clip(a_try, 0.01, 3.0))
                p_try = _gradient_step_exact(p, c, T, a_try)
                sc_try = _score(p_try)
                if sc_try < best_trial_sc - 1e-14:
                    best_trial = p_try
                    best_trial_sc = sc_try

            if best_trial is not None:
                p = best_trial
                cur_score = best_trial_sc
                stall = 0
                if cur_score < best_score - 1e-13:
                    best = p.copy()
                    best_score = cur_score
                    if verbose and it % 100 == 0:
                        print(f"      AT iter={it}, score={best_score:.7f}, T={T}, alpha={alpha:.4f}")
                alpha = min(alpha * 1.01, 2.5)
            else:
                stall += 1
                alpha = max(0.05, alpha * 0.9)
                if stall > 20:
                    p = best.copy()
                    cur_score = best_score
                    alpha = 0.4
                    stall = 5

        elif r < 0.70:
            # Spectral compression
            gamma = float(rng.uniform(0.65, 0.90))
            p_new = _spectral_compress(p, gamma, L_fft)
            if p_new is not None:
                sc_new = _score(p_new)
                if sc_new < cur_score - 1e-13:
                    p = p_new
                    cur_score = sc_new
                    stall = 0
                    if sc_new < best_score - 1e-13:
                        best = p.copy()
                        best_score = sc_new
                else:
                    stall += 1
            else:
                stall += 1

        elif r < 0.80:
            # Fine block adjustment near peaks
            c = _conv_full(p, p)
            peak_idx = c.argmax()
            if peak_idx < n:
                focus = peak_idx // 2
            else:
                focus = (peak_idx - n + 1) // 2
            focus = max(0, min(n - 1, focus))
            bs = int(rng.integers(2, max(3, min(32, n // 32))))
            start = int(np.clip(focus + rng.integers(-bs * 3, bs * 3), 0, n - bs))
            factor = float(rng.choice([0.5, 0.7, 0.85, 1.15, 1.3, 1.6]))
            p_try = p.copy()
            p_try[start:start + bs] *= factor
            p_try = _sanitize(p_try)
            if p_try is not None:
                sc_try = _score(p_try)
                if sc_try < cur_score - 1e-14:
                    p = p_try
                    cur_score = sc_try
                    stall = 0
                    if sc_try < best_score - 1e-13:
                        best = p.copy()
                        best_score = sc_try
                else:
                    stall += 1
            else:
                stall += 1

        else:
            # Perturbation
            if stall > 10:
                std = 0.02 + 0.005 * min(stall, 20)
                p_try = p * np.exp(rng.normal(0, std, n))
                p_try = _sanitize(p_try)
                if p_try is not None:
                    sc_try = _score(p_try)
                    if sc_try < best_score * 1.03:
                        p = p_try
                        cur_score = sc_try
                        stall = 0
            else:
                stall += 1

        it += 1

    return best, best_score


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing the evaluation score.
    Uses asymmetric tilted initializations, exact subgradient descent,
    spectral compression, simulated annealing, and multi-length search.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 2400.0

    start_time = time.monotonic()
    deadline = start_time + budget - 20

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng(iseed % (2**31 - 1))

    # Get parent
    try:
        parent = _sanitize(np.asarray(height_sequence_1, dtype=float))
    except Exception:
        parent = None

    best_overall = None
    best_score_overall = float('inf')

    def _try_update(p, score):
        nonlocal best_overall, best_score_overall
        if p is not None and np.isfinite(score) and score < best_score_overall:
            best_overall = p.copy()
            best_score_overall = float(score)
            print(f"  *** NEW BEST: {score:.7f} (n={len(p)}) at t={time.monotonic()-start_time:.1f}s ***")

    # ========== PHASE 1: Evaluation and initialization (first 5% of time) ==========
    print("=" * 60)
    print(f"Phase 1: Initialization (budget={budget:.0f}s)")
    t1_end = start_time + 0.05 * budget

    candidates = []

    if parent is not None:
        sc = _score(parent)
        _try_update(parent, sc)
        candidates.append(('parent', parent, sc))
        print(f"  Parent score: {sc:.7f} (n={len(parent)})")

    # Generate diverse asymmetric initializations at the parent length
    n_base = len(parent) if parent is not None else 4096

    n_total = 0
    while n_total < n_base:
        n_total += 1  # just for tracking

    inits_to_try = [
        ('tilt1', _init_tilted(n_base, rng, tilt_param=1.5)),
        ('tilt2', _init_tilted(n_base, rng, tilt_param=-2.0)),
        ('tilt3', _init_tilted(n_base, rng, tilt_param=2.5)),
        ('ramp', _init_ramp_cosine(n_base, rng)),
        ('asym_bump', _init_asymmetric_bump(n_base, rng)),
        ('two_gauss', _init_two_gaussian(n_base, rng)),
    ]

    for name, p_init in inits_to_try:
        if time.monotonic() > t1_end:
            break
        p_init = _sanitize(p_init)
        if p_init is None:
            continue
        sc = _score(p_init)
        _try_update(p_init, sc)
        candidates.append((name, p_init, sc))

    # Also try at a different length
    n_alt = n_base * 2 if n_base <= 4096 else n_base // 2
    n_alt = max(1024, min(8192, n_alt))
    if n_alt != n_base:
        for name2, p_alt in [('tilt_alt', _init_tilted(n_alt, rng, tilt_param=1.8)),
                             ('ramp_alt', _init_ramp_cosine(n_alt, rng))]:
            if time.monotonic() > t1_end:
                break
            p_alt = _sanitize(p_alt)
            if p_alt is not None:
                sc_alt = _score(p_alt)
                _try_update(p_alt, sc_alt)
                candidates.append((name2, p_alt, sc_alt))

    print(f"  Phase 1 done. Best: {best_score_overall:.7f}")

    if best_overall is None:
        best_overall = _init_tilted(4096, rng, tilt_param=1.5)
        best_score_overall = _score(best_overall)

    # ========== PHASE 2: Main optimization on top candidates (50% of time) ==========
    print("Phase 2: Main optimization")
    t2_end = start_time + 0.55 * budget

    # Sort candidates by score, take top 3
    candidates.sort(key=lambda x: x[2])
    top_candidates = candidates[:min(3, len(candidates))]

    for idx, (cname, cp, csc) in enumerate(top_candidates):
        if time.monotonic() > t2_end - 30:
            break
        t_rem = t2_end - time.monotonic()
        if t_rem < 30:
            break
        t_alloc = min(t_rem * 0.8, 300.0)
        t_alloc = max(20.0, t_alloc)

        print(f"  Optimizing candidate '{cname}' (score={csc:.7f}, n={len(cp)})")
        p, sc = _optimize_main(cp, rng, t_alloc, verbose=True)
        _try_update(p, sc)

        # Also try adaptive T on the same candidate
        t_rem = t2_end - time.monotonic()
        if t_rem > 30:
            t_alloc2 = min(t_rem * 0.6, 150.0)
            p2, sc2 = _adaptive_T_optimize(cp, rng, t_alloc2, verbose=False)
            _try_update(p2, sc2)

    # Try the overall best with more time
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(t_rem, 200.0)
        print(f"  Extended optimization on best (score={best_score_overall:.7f})")
        p, sc = _optimize_main(best_overall, rng, t_alloc, verbose=True)
        _try_update(p, sc)

    print(f"  Phase 2 done. Best: {best_score_overall:.7f} (n={len(best_overall)})")

    # ========== PHASE 3: Simulated annealing (15% of time) ==========
    print("Phase 3: Simulated annealing")
    t3_end = start_time + 0.72 * budget

    t_rem = t3_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(t_rem * 0.8, 150.0)
        p, sc = _optimize_sa(best_overall, rng, t_alloc, verbose=True)
        _try_update(p, sc)

    # SA on a perturbed version
    t_rem = t3_end - time.monotonic()
    if t_rem > 30:
        p_pert = best_overall * np.exp(rng.normal(0, 0.05, len(best_overall)))
        p_pert = _sanitize(p_pert)
        if p_pert is not None:
            t_alloc = min(t_rem * 0.7, 100.0)
            p, sc = _optimize_sa(p_pert, rng, t_alloc, verbose=False)
            _try_update(p, sc)

    print(f"  Phase 3 done. Best: {best_score_overall:.7f}")

    # ========== PHASE 4: Multi-length exploration (10% of time) ==========
    print("Phase 4: Multi-length exploration")
    t4_end = start_time + 0.82 * budget

    n_best = len(best_overall)
    for n_try in [n_best, n_best * 2, n_best // 2]:
        if time.monotonic() > t4_end - 10:
            break
        n_try = max(512, min(16384, n_try))
        if n_try == n_best:
            continue
        t_rem = t4_end - time.monotonic()
        if t_rem < 20:
            break

        # Resample best to new length
        p_res = _resample(best_overall, n_try, rng)
        if p_res is None:
            p_res = _init_tilted(n_try, rng, tilt_param=1.5)

        t_alloc = min(t_rem * 0.7, 80.0)
        p, sc = _optimize_main(p_res, rng, t_alloc, verbose=False)
        _try_update(p, sc)

    print(f"  Phase 4 done. Best: {best_score_overall:.7f}")

    # ========== PHASE 5: Final polishing (last 18% of time) ==========
    print("Phase 5: Final polishing")

    if best_overall is not None:
        # Adaptive T optimization
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.4 * t_rem, 150.0)
            p, sc = _adaptive_T_optimize(best_overall, rng, t_alloc, verbose=True)
            _try_update(p, sc)

        # Block refinement
        t_rem = deadline - time.monotonic()
        if t_rem > 15:
            t_alloc = min(0.3 * t_rem, 80.0)
            p, sc = _block_refine(best_overall, rng, t_alloc)
            _try_update(p, sc)

        # One more spectral + gradient pass
        t_rem = deadline - time.monotonic()
        if t_rem > 15:
            t_alloc = min(0.35 * t_rem, 120.0)
            p, sc = _optimize_main(best_overall, rng, t_alloc, verbose=True)
            _try_update(p, sc)

        # Final block refine
        t_rem = deadline - time.monotonic()
        if t_rem > 10:
            t_alloc = min(0.3 * t_rem, 60.0)
            p, sc = _block_refine(best_overall, rng, t_alloc)
            _try_update(p, sc)

    # ========== Final output ==========
    if best_overall is None:
        best_overall = _init_tilted(4096, rng, tilt_param=1.5)

    p_final = _sanitize(best_overall)
    if p_final is None:
        p_final = np.ones(2048, dtype=float) / 2048

    final_score = _score(p_final)

    # Verify with official evaluator
    try:
        eval_score = float(evaluate_sequence([float(x) for x in p_final]))
        if np.isfinite(eval_score):
            final_score = min(final_score, eval_score)
    except Exception:
        pass

    print(f"\n{'=' * 60}")
    print(f"FINAL: score={final_score:.7f} (n={len(p_final)})")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    out = np.where(np.isfinite(p_final), p_final, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = np.ones(2048, dtype=float) / 2048
    else:
        out = out / s

    return [float(x) for x in out]
