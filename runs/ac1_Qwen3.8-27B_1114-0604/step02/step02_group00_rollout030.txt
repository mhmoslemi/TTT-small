import numpy as np

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

"""
Hybrid optimizer: Error Reduction (Newton-like freq-domain) + L-BFGS + Subgradient Polish.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Innovations over previous approach:
1. Newton-like error reduction in frequency domain as primary global search
2. Envelope-modulation initialization strategy
3. Adaptive beta schedule with restarts
4. Coordinate descent with peak-targeted updates
5. Multi-length exploration with parent interpolation
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    """Wrapper for scipy.minimize."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on the LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def error_reduction(a, n_iters=200, alpha=0.7):
    """
    Newton-like error reduction in frequency domain.
    Iteratively flattens the convolution by reducing peak excess.
    """
    n = len(a)
    a = np.maximum(a.astype(np.float64), 0.0)
    S = float(a.sum())
    if S < 1e-12:
        return np.ones(n) / n
    a = a / S

    M = 1 << (2 * n - 1).bit_length()
    best_a = a.copy()
    best_sc = compute_score(a)

    for it in range(n_iters):
        a_pad = np.zeros(M, dtype=np.complex128)
        a_pad[:n] = a
        A = np.fft.fft(a_pad)
        v = A * A
        conv = np.fft.ifft(v).real[:2 * n - 1]

        conv_mean = float(conv.mean())
        conv_max = float(conv.max())
        if conv_max - conv_mean < 1e-15:
            break

        # Soft threshold: compress values above threshold
        thresh = conv_mean + (1.0 - alpha) * (conv_max - conv_mean)
        excess = np.maximum(conv - thresh, 0.0)
        if excess.max() < 1e-15:
            break
        conv_mod = conv - 0.8 * excess

        err = conv_mod - conv
        err_pad = np.zeros(M, dtype=np.complex128)
        err_pad[:2 * n - 1] = err
        err_spec = np.fft.fft(err_pad)

        # Newton step: A_new = A + err_spec / (2*A)
        with np.errstate(divide='ignore', invalid='ignore'):
            d = err_spec / (2.0 * A + 1e-15)
        d[np.abs(A) < 1e-8] = 0.0

        # Damping for stability
        damp = 0.5 + 0.5 * (it / max(1, n_iters // 4))
        A_new = A + damp * d

        a_new = np.fft.ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        if a_new.sum() < 1e-12:
            break
        a_new = a_new / a_new.sum()

        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
        a = a_new

    return best_a

def coordinate_descent_peak(a, n_sweeps=3, n_top=50):
    """
    Coordinate descent targeting the peak-contributing elements.
    For each sweep, find the peak lag, identify top contributors,
    and optimize them via 1D search.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)

    for sweep in range(n_sweeps):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end <= j_start:
            continue

        # Contribution of a[j] to c[k_star] is proportional to a[k_star - j]
        seg = a[j_start:j_end + 1]
        contrib = seg[::-1].copy()
        # Top contributors
        top_count = min(n_top, len(contrib))
        top_local = np.argsort(contrib)[::-1][:top_count]
        top_global = j_start + top_local

        for j in top_global:
            # 1D optimization: vary a[j]
            a_old_j = a[j]
            # Try several values
            candidates = [
                a_old_j * 0.5,
                a_old_j * 0.7,
                a_old_j * 0.8,
                a_old_j * 0.9,
                a_old_j * 1.1,
                a_old_j * 1.3,
                0.0,
                a_old_j * 0.3,
            ]
            best_val = a_old_j
            best_local_sc = compute_score(a)
            for cv in candidates:
                cv = max(0.0, min(1000.0, cv))
                if abs(cv - a_old_j) < 1e-10:
                    continue
                a[j] = cv
                sc = compute_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_val = cv
            a[j] = best_val
            S = a.sum()
            if S > 1e-12:
                a = a / S

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a


def subgradient_polish(a, total_iters=2000):
    """Projected subgradient descent on the exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    S = float(a.sum())
    if S < 1e-12:
        return a
    a = a / S
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0

    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.1)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        S = a.sum()
        if S < 1e-12:
            break
        a = a / S

    return best_a


def build_initializations(n, rng, parent):
    """Generate diverse initializations of length n."""

    x = np.arange(n, dtype=np.float64) / n
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.4 * np.cos(2.0 * np.pi * x))

    inits.append(0.3 + 0.7 * np.cos(2.0 * np.pi * x) ** 2)
    inits.append(0.5 + 0.3 * np.cos(4.0 * np.pi * x) * np.cos(2.0 * np.pi * x))
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))

    # Envelope shape: U-shaped to counteract triangular conv
    inits.append(0.2 + 0.8 * (2.0 * x - 1.0) ** 2)

        if len(a) < n:

            best_score = sc
            best_a = np.clip(a.copy(), 0.0, 1000.0)
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(len(parent))
    for n in [4096, 8192, 12288, 16384]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))[:4]

    print(f"Lengths to try: {lengths}")
    print(f"Parent available: {parent is not None}, len={0 if parent is None else len(parent)}")

    # ==================================================================
    # Phase 1: Error reduction from multiple inits (40% of time)
    # ==================================================================
    phase1_deadline = t_start + 0.40 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 1: Error Reduction ---")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"  n={n}")
        inits = build_initializations(n, rng, parent)
        # Limit inits based on time
        remaining = phase1_deadline - time.monotonic()
        max_inits = max(2, int(remaining / 5.0))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a_er = error_reduction(a0, n_iters=150, alpha=0.7)
            sc = compute_score(a_er)
            report(sc, a_er, f"ER n={n} init={i}")
            # Also try multi-alpha
            a_er2 = error_reduction_multi(a_er, n_total_iters=200)
            sc2 = compute_score(a_er2)
            report(sc2, a_er2, f"ERm n={n} init={i}")

    print(f"After Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: L-BFGS refinement (35% of time)
    # ==================================================================
    phase2_deadline = t_start + 0.75 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 2: L-BFGS Refinement ---")

    beta_schedule = [50, 100, 200, 500, 1000]
    n_best = len(best_a)

    for beta in beta_schedule:
        if time.monotonic() > phase2_deadline:
            break
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=8000, maxfun=150000)
        report(sc, a_opt, f"LBFGS beta={beta}")
        n_best = len(best_a)

    # Try L-BFGS from a few other good solutions
    other_starts = []
    for n in lengths[:2]:
        a_test = best_a if len(best_a) == n else np.interp(
            np.linspace(0, 1, n), np.linspace(0, 1, len(best_a)), best_a
        )
        other_starts.append(a_test)

    for i, a_s in enumerate(other_starts):
        if time.monotonic() > phase2_deadline:
            break
        for beta in [100, 300]:
            if time.monotonic() > phase2_deadline:
                break
            sc, a_opt, _ = run_lbfgs(a_s, beta, maxiter=5000, maxfun=100000)
            report(sc, a_opt, f"LBFGS alt{i} beta={beta}")

    print(f"After Phase 2: best = {best_score:.8f}")

    # ==================================================================
    # Phase 3: Local search with perturbations (15% of time)
    # ==================================================================
    phase3_deadline = t_start + 0.90 * (float(budget_s) - 15.0)
    print(f"\n--- Phase 3: Local Search ---")

    n_best = len(best_a)
    n_trials = 15
    for trial in range(n_trials):
        if time.monotonic() > phase3_deadline:
            break
        strategy = trial % 5
        if strategy == 0:
            scale = 0.02 + 0.05 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Targeted: reduce peak contributors
            c = fast_conv(best_a)
            k_star = int(np.argmax(c))
            j_start = max(0, k_star - n_best + 1)
            j_end = min(k_star, n_best - 1)
            a_pert = best_a.copy()
            seg = a_pert[j_start:j_end + 1]
            contrib = seg[::-1].copy()
            top_count = max(5, n_best // 100)
            top_local = np.argsort(contrib)[::-1][:top_count]
            total_red = 0.0
            for li in top_local:
                gi = j_start + li
                red = a_pert[gi] * 0.05
                a_pert[gi] -= red
                a_pert[gi] = max(a_pert[gi], 0.0)
                total_red += red
            if total_red > 0:
                bottom_local = np.argsort(contrib)[:max(1, len(contrib) - top_count)]
                per = total_red / len(bottom_local)
                for li in bottom_local:
                    a_pert[j_start + li] += per
        elif strategy == 2:
            # Sinusoidal
            freq = 1.0 + 8.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.02 + 0.04 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Error reduction restart
            a_pert = error_reduction(best_a * (1 + 0.05 * rng.standard_normal(n_best)),
                                     n_iters=50, alpha=0.8)
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Random rescale of subsets
            a_pert = best_a.copy()
            n_blocks = 5 + int(rng.integers(0, 10))
            block_size = max(1, n_best // n_blocks)
            for b in range(min(n_blocks, n_best // block_size)):
                s = b * block_size
                e = min(s + block_size, n_best)
                factor = 0.5 + rng.random()
                a_pert[s:e] *= factor
            a_pert = np.clip(a_pert, 0.0, 1000.0)
            S = a_pert.sum()
            if S > 1e-12:
                a_pert = a_pert / S

        sc, a_opt, _ = run_lbfgs(a_pert, 300, maxiter=3000, maxfun=50000)
        report(sc, a_opt, f"local #{trial} s={strategy}")

    print(f"After Phase 3: best = {best_score:.8f}")

    # ==================================================================
    # Phase 4: Coordinate descent + subgradient polish (10% of time)
    # ==================================================================
    print(f"\n--- Phase 4: Polish ---")

    # Coordinate descent
    a_cd = coordinate_descent_peak(best_a, n_sweeps=3, n_top=80)
    sc = compute_score(a_cd)
    report(sc, a_cd, "coord descent")

    # Final L-BFGS from CD result
    if time.monotonic() < deadline - 30:
        sc, a_opt, _ = run_lbfgs(a_cd, 500, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "CD+LBFGS")

    # Subgradient polish
    a_sp = subgradient_polish(best_a, total_iters=2000)
    sc = compute_score(a_sp)
    report(sc, a_sp, "subgrad")

    # Final high-beta L-BFGS
    if time.monotonic() < deadline - 20:
        sc, a_opt, _ = run_lbfgs(best_a, 2000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "final LBFGS beta=2000")

    # One more error reduction from best
    if time.monotonic() < deadline - 15:
        a_er_final = error_reduction(best_a, n_iters=100, alpha=0.85)
        sc = compute_score(a_er_final)
        report(sc, a_er_final, "final ER")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
