"""
Improved multi-phase optimizer for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements over previous:

- Backtracking line search for adaptive step sizes

import numpy as np
import time


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b, L=None):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 4000000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)


def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())


def _score_and_max(p):
    """Return (score, max_conv_value, conv array)."""
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf'), 0.0, None
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    mx = float(c.max())
    return float(2.0 * n * mx), mx, c


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s


def _grad_softmax(p, beta, L=None):
    """Gradient using softmax smoothing of convolution peaks."""
    n = len(p)
    if L is None:
        L = _next_pow2(3 * n + 10)
    c = _conv_full(p, p)
    cm = c.max()
    if cm < 1e-15:
        return np.zeros(n), 0.0
    shift = cm
    e = np.exp(beta * (c - shift))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w = e / s
    W = np.fft.rfft(w, L)
    P = np.fft.rfft(p, L)
    R = np.fft.irfft(W * np.conj(P), L)
    g = 2.0 * R[:n]
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    return g, float(cm)


def _grad_topk(p, k, L=None):
    """Gradient using exact top-k peaks (subgradient)."""
    n = len(p)
    if L is None:
        L = _next_pow2(3 * n + 10)
    c = _conv_full(p, p)
    cm = float(c.max())
    if cm < 1e-15:
        return np.zeros(n), 0.0
    k = min(k, len(c))
    top_idx = np.argpartition(c, -k)[-k:]
    top_vals = c[top_idx]
    w = np.zeros(len(c))
    s = top_vals.sum()
    if s < 1e-15:
        w[top_idx] = 1.0 / k
    else:
        w[top_idx] = top_vals / s
    W = np.fft.rfft(w, L)
    P = np.fft.rfft(p, L)
    R = np.fft.irfft(W * np.conj(P), L)
    g = 2.0 * R[:n]
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    return g, cm


def _grad_mixed(p, beta, k, mix, L=None):
    """Mix of softmax and top-k gradients."""
    if L is None:
        L = _next_pow2(3 * n + 10)
    n = len(p)
    g1, cm1 = _grad_softmax(p, beta, L)
    g2, cm2 = _grad_topk(p, k, L)
    g = mix * g2 + (1.0 - mix) * g1
    gm = max(gm1, gm2)
    return g, gm


def _freq_flatten(p, alpha, L=None):
    """Frequency-domain convolution flattening."""
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(p, L)
    C_freq = A * A
    C = np.fft.irfft(C_freq, L)[:2 * n - 1]
    cmax = C.max()
    if cmax < 1e-15:
        return p.copy()
    C_norm = C / cmax
    C_new = cmax * np.power(np.maximum(C_norm, 0), 1.0 / alpha)
    C_freq_new = np.fft.rfft(C_new, L)
    A_new = np.sqrt(C_freq_new.astype(complex) + 0j)
    a_new = np.fft.irfft(A_new, L)[:n]
    neg_frac = (a_new < -0.01 * p.max()).sum() / max(n, 1)
    if neg_frac > 0.35:
        return None
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s


def _resample(p_in, n_new, rng=None):
    p = _sanitize(p_in)
    if p is None:
        return None
    n_old = len(p)
    if n_old == n_new:
        result = p.copy()
    elif n_old == 1:
        result = np.ones(n_new, dtype=float)
    else:
        x_old = (np.arange(n_old) + 0.5) / n_old
        x_new = (np.arange(n_new) + 0.5) / n_new
        result = np.interp(x_new, x_old, p)
    result = np.maximum(result, 0)
    if rng is not None:
        result = result * np.exp(rng.normal(0, 0.03, n_new))
        result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s


def _optimize_momentum(p0, rng, time_budget, beta_start=50.0, beta_end=5000.0,

def _block_refine_v2(p0, rng, time_budget, verbose=False):
    """Improved block refinement with smarter block selection."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    best = p.copy()
    best_score = _score(p)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0
    factors = (0.5, 0.7, 0.85, 1.0, 1.18, 1.4, 1.8, 2.5)

    if verbose:
        print(f"    Block refine: n={n}, init={best_score:.6f}")

    while time.monotonic() < deadline:
        # Choose block based on where convolution peaks are
        c = _conv_full(p, p)
        peak_pos = int(c.argmax())
        # The peak at position k in conv corresponds to interaction of a[j] and a[k-j]
        # Focus blocks around indices that contribute to the peak
        center = peak_pos // 2
        spread = max(1, n // 50)

        r = rng.random()
        if r < 0.5:
            # Block near the peak-contributing region
            start = max(0, center + int(rng.integers(-spread, spread + 1)))
            bs = int(rng.integers(1, max(2, n // 64)))
        elif r < 0.8:
            start = int(rng.integers(0, n - 1))
            bs = int(rng.integers(1, max(2, n // 32)))
        else:
            start = int(rng.integers(0, max(1, n // 2)))
            bs = int(rng.integers(n // 8, n // 3 + 1))

        bs = min(bs, n - start)
        if bs < 1:
            continue

        improved = False
        for f in factors:
            p_try = p.copy()
            p_try[start:start + bs] *= f
            p_try = _sanitize(p_try)
            if p_try is None:
                continue
            sc = _score(p_try)
            if sc < cur_score - 1e-14:
                cur_score = sc
                p = p_try
                improved = True
                break

        if improved:
            if cur_score < best_score - 1e-13:
                best = p.copy()
                best_score = cur_score
            stall = 0
        else:
            stall += 1
            if stall > 80:
                p = best.copy()
                cur_score = best_score
                stall = 0

        if time.monotonic() > deadline:
            break

    return best, best_score


def _pair_swap_refine(p0, rng, time_budget, verbose=False):
    """Refine by swapping pairs of elements (preserves sum and L2 norm)."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    best = p.copy()
    best_score = _score(p)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0

    if verbose:
        print(f"    Pair swap: n={n}, init={best_score:.6f}")

    while time.monotonic() < deadline:
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        # Try partial swap: blend a[i] and a[j]
        t = float(rng.uniform(0.1, 0.9))
        p_try = p.copy()
        vi, vj = p_try[i], p_try[j]
        p_try[i] = (1 - t) * vi + t * vj
        p_try[j] = (1 - t) * vj + t * vi
        p_try = _sanitize(p_try)
        if p_try is None:
            continue
        sc = _score(p_try)
        if sc < cur_score - 1e-14:
            p = p_try
            cur_score = sc
            stall = 0
            if cur_score < best_score - 1e-13:
                best = p.copy()
                best_score = cur_score
                if verbose and stall == 0:
                    print(f"      swap score={best_score:.6f}")
        else:
            stall += 1
            if stall > 100:
                p = best.copy()
                cur_score = best_score
                stall = 0

    return best, best_score


def _init_uniform(n):
    return np.ones(n, dtype=float) / n


def _init_tapered(n, rng):
    """Tapered (raised cosine) initialization - helps with boundary effects."""
    t = np.arange(n, dtype=float) / n
    # Raised cosine window
    p = 0.5 * (1 + np.cos(2 * np.pi * t - np.pi))
    # Add some randomness
    p = p * (1 + 0.1 * rng.normal(0, 1, n))
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _init_two_level(n, rng):
    k = int(n * rng.uniform(0.3, 0.7))
    ratio = float(rng.uniform(1.5, 4.0))
    p = np.ones(n, dtype=float)
    p[:k] = ratio
    idx = rng.permutation(n)
    p = p[idx]
    p = p * (1.0 + 0.1 * rng.normal(0, 1, n))
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return _init_uniform(n)
    return p / s


def _init_gaussian_mixture(n, rng):
    K = int(rng.integers(2, 6))
    p = np.zeros(n, dtype=float)
    t = np.arange(n, dtype=float) / n
    for _ in range(K):
        c = rng.uniform(0.0, 1.0)
        s = rng.uniform(0.03, 0.25)
        w = rng.uniform(0.5, 2.0)
        x = (t - c) / s
        x = np.minimum(np.abs(x), 1.0 - np.abs(x))
        p += w * np.exp(-0.5 * x * x)
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _init_symmetric(n, rng):
    """Symmetric initialization."""
    half = (n + 1) // 2
    p_half = np.abs(rng.normal(1, 0.5, half))
    p_half = np.maximum(p_half, 0.01)
    p = np.zeros(n)
    p[:half] = p_half
    p[half:] = p_half[:n - half][::-1]
    return p / p.sum()


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing the evaluation score.
    Multi-phase: momentum gradient + direct subgradient + freq flatten + block refine.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 2400.0

    start_time = time.monotonic()
    deadline = start_time + budget - 10

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng(iseed % (2**31 - 1))

    # Get parent
    try:
        parent = _sanitize(np.asarray(height_sequence_1, dtype=float))
    except Exception:
        parent = None

    best_overall = None
    best_score_overall = float('inf')

    def _try_update(p, score):
        nonlocal best_overall, best_score_overall
        if p is not None and np.isfinite(score) and score < best_score_overall:
            best_overall = p.copy()
            best_score_overall = float(score)
            print(f"  *** NEW BEST: {score:.6f} (n={len(p)}) ***")

    # ========== PHASE 1: Exploration (8% time) ==========
    print("=" * 60)
    print(f"Phase 1: Exploration (budget={budget:.0f}s)")
    t1_end = start_time + 0.08 * budget

    # Evaluate parent
    if parent is not None:
        sc = _score(parent)
        _try_update(parent, sc)
        print(f"  Parent score: {sc:.6f} (n={len(parent)})")

    # Try various initializations at multiple lengths
    lengths_to_try = [2048, 3072, 4096, 6144]
    if parent is not None:
        n_par = len(parent)
        if n_par not in lengths_to_try:
            lengths_to_try.append(n_par)
            lengths_to_try.sort()

    for n in lengths_to_try:
        if time.monotonic() > t1_end:
            break
        inits = [
            _init_uniform(n),
            _init_tapered(n, rng),
            _init_two_level(n, rng),
            _init_gaussian_mixture(n, rng),
            _init_symmetric(n, rng),
        ]
        if parent is not None:
            sp = _resample(parent, n, rng)
            if sp is not None:
                inits.append(sp)

        for sp in inits:
            sc = _score(sp)
            _try_update(sp, sc)

    print(f"  Phase 1 done. Best: {best_score_overall:.6f}")

    # ========== PHASE 2: Main optimization - Momentum (30% time) ==========
    print("Phase 2: Momentum gradient optimization")
    t2_end = start_time + 0.38 * budget

    if best_overall is None:
        n = 3072 if parent is None else len(parent)
        best_overall = _init_uniform(n)
        best_score_overall = _score(best_overall)

    # Run momentum optimizer on best
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(t_rem, 0.6 * (t2_end - start_time - 0.08 * budget))
        p, sc = _optimize_momentum(best_overall, rng, t_alloc,
                                   beta_start=50.0, beta_end=5000.0,
                                   k_top=3, mix_start=0.6, verbose=True)
        _try_update(p, sc)

    # Run with different parameters
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(t_rem * 0.7, 200.0)
        p, sc = _optimize_momentum(best_overall, rng, t_alloc,
                                   beta_start=200.0, beta_end=10000.0,
                                   k_top=1, mix_start=0.8, verbose=True)
        _try_update(p, sc)

    print(f"  Phase 2 done. Best: {best_score_overall:.6f} (n={len(best_overall)})")

    # ========== PHASE 3: Direct subgradient (15% time) ==========
    print("Phase 3: Direct subgradient optimization")
    t3_end = start_time + 0.53 * budget

    t_rem = t3_end - time.monotonic()
    if t_rem > 20:
        t_alloc = min(t_rem * 0.8, 300.0)
        p, sc = _optimize_direct_subgrad(best_overall, rng, t_alloc, verbose=True)
        _try_update(p, sc)

    # Try on a resampled version
    t_rem = t3_end - time.monotonic()
    if t_rem > 20:
        n_up = min(6144, len(best_overall) * 2)
        if n_up > len(best_overall):
            p_up = _resample(best_overall, n_up, rng)
            if p_up is not None:
                t_alloc = min(t_rem * 0.6, 200.0)
                p, sc = _optimize_direct_subgrad(p_up, rng, t_alloc, verbose=False)
                _try_update(p, sc)

    print(f"  Phase 3 done. Best: {best_score_overall:.6f}")

    # ========== PHASE 4: Frequency domain reorganization (12% time) ==========
    print("Phase 4: Frequency domain reorganization")
    t4_end = start_time + 0.65 * budget

    t_rem = t4_end - time.monotonic()
    if t_rem > 20:
        n_best = len(best_overall)
        L_fft = _next_pow2(2 * n_best - 1)
        p_cur = best_overall.copy()
        sc_cur = best_score_overall

        for alpha_f in [1.3, 1.5, 1.8, 2.0, 2.5, 3.0, 4.0, 5.0, 7.0, 10.0]:
            if time.monotonic() > t4_end - 5:
                break
            p_try = _freq_flatten(p_cur, alpha_f, L_fft)
            if p_try is not None:
                sc_try = _score(p_try)
                if sc_try < sc_cur - 1e-14:
                    p_cur = p_try
                    sc_cur = sc_try
                    _try_update(p_cur, sc_cur)
                    # Chain: try another flatten
                    for alpha_f2 in [1.5, 2.0, 3.0]:
                        p_try2 = _freq_flatten(p_cur, alpha_f2, L_fft)
                        if p_try2 is not None:
                            sc_try2 = _score(p_try2)
                            if sc_try2 < sc_cur - 1e-14:
                                p_cur = p_try2
                                sc_cur = sc_try2
                                _try_update(p_cur, sc_cur)

        # After frequency reorg, do a short momentum pass
        t_rem = t4_end - time.monotonic()
        if t_rem > 15:
            p, sc = _optimize_momentum(p_cur, rng, min(t_rem, 60.0),
                                       beta_start=200.0, beta_end=3000.0,
                                       k_top=2, mix_start=0.7, verbose=False)
            _try_update(p, sc)

    print(f"  Phase 4 done. Best: {best_score_overall:.6f}")

    # ========== PHASE 5: Block refinement (15% time) ==========
    print("Phase 5: Block refinement")
    t5_end = start_time + 0.80 * budget

    t_rem = t5_end - time.monotonic()
    if t_rem > 20:
        t_alloc = min(t_rem * 0.7, 300.0)
        p, sc = _block_refine_v2(best_overall, rng, t_alloc, verbose=True)
        _try_update(p, sc)

    t_rem = t5_end - time.monotonic()
    if t_rem > 15:
        t_alloc = min(t_rem, 120.0)
        p, sc = _pair_swap_refine(best_overall, rng, t_alloc, verbose=False)
        _try_update(p, sc)

    print(f"  Phase 5 done. Best: {best_score_overall:.6f}")

    # ========== PHASE 6: Diverse restarts (10% time) ==========
    print("Phase 6: Diverse restarts")
    t6_end = start_time + 0.90 * budget

    t_rem = t6_end - time.monotonic()
    if t_rem > 30:
        n_best = len(best_overall)
        n_restarts = 4
        t_each = max(10.0, t_rem / (n_restarts + 1))

        for mode in range(n_restarts):
            if time.monotonic() > t6_end - 5:
                break
            if mode == 0:
                sp = _init_tapered(n_best, rng)
            elif mode == 1:
                sp = _init_symmetric(n_best, rng)
            elif mode == 2:
                sp = _resample(best_overall, n_best, rng)
                if sp is None:
                    sp = _init_uniform(n_best)
            else:
                sp = _init_gaussian_mixture(n_best, rng)

            # Quick momentum pass
            p, sc = _optimize_momentum(sp, rng, t_each,
                                       beta_start=100.0, beta_end=3000.0,
                                       k_top=2, mix_start=0.5, verbose=False)
            _try_update(p, sc)

            # Quick block refine
            if time.monotonic() < t6_end - 5:
                p2, sc2 = _block_refine_v2(p, rng, t_each * 0.5, verbose=False)
                _try_update(p2, sc2)

    print(f"  Phase 6 done. Best: {best_score_overall:.6f}")

    # ========== PHASE 7: Final polishing (10% time) ==========
    print("Phase 7: Final polishing")

    if best_overall is not None:
        # High-beta momentum
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.35 * t_rem, 150.0)
            p, sc = _optimize_momentum(best_overall, rng, t_alloc,
                                       beta_start=500.0, beta_end=15000.0,
                                       k_top=1, mix_start=0.9, verbose=True)
            _try_update(p, sc)

        # Direct subgradient
        t_rem = deadline - time.monotonic()
        if t_rem > 15:
            t_alloc = min(0.3 * t_rem, 100.0)
            p, sc = _optimize_direct_subgrad(best_overall, rng, t_alloc, verbose=False)
            _try_update(p, sc)

        # Block refine
        t_rem = deadline - time.monotonic()
        if t_rem > 15:
            t_alloc = min(0.3 * t_rem, 80.0)
            p, sc = _block_refine_v2(best_overall, rng, t_alloc, verbose=True)
            _try_update(p, sc)

    # ========== Final output ==========
    if best_overall is None:
        best_overall = _init_uniform(3072)

    p_final = _sanitize(best_overall)
    if p_final is None:
        p_final = _init_uniform(3072)

    final_score = _score(p_final)
    try:
        eval_score = float(evaluate_sequence([float(x) for x in p_final]))
        if np.isfinite(eval_score):
            final_score = min(final_score, eval_score)
    except Exception:
        pass

    print(f"\n{'=' * 60}")
    print(f"FINAL: score={final_score:.6f} (n={len(p_final)})")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    out = np.where(np.isfinite(p_final), p_final, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = _init_uniform(3072)
    else:
        out = out / s

    return [float(x) for x in out]
