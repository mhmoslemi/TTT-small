"""
Low-peak step-function optimization via adaptive active-set mirror descent
with FFT gradients, line search, and spectral initialization.

Key improvements:
1. Spectral "flat-power" initialization via iterative phase retrieval
2. Adaptive K schedule (coarse-to-fine active set)
3. Backtracking line search for robust step size selection
4. Multiple diverse starting points with restart strategy
5. Better temperature annealing for soft-max peak weighting
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear auto-convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score_fast(a):
    """Compute the evaluation score (scale-invariant)."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    mx = c.max()
    return 2.0 * a.size * mx / (s * s)


def _grad_at_peaks(a, w):
    """
    Compute grad[i] = 2 * sum_m w[i+m] * a[m]
    via FFT cross-correlation. w has length 2n-1.
    """
    n = a.size
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(np.conj(fw) * fa, N)
    return 2.0 * xcorr[:n]


def _normalize(a):
    """Normalize to sum=1, clip to [0, 1000]."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s


def _make_raised_cosine(n):
    """Raised cosine window, normalized."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * i / n))
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def _make_gaussian(n, sigma_frac=0.3):
    """Gaussian window, normalized."""
    i = np.arange(n, dtype=np.float64)
    center = n / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def _make_sine(n, amp=0.8, phase=0.25):
    """Sinusoidal modulation of uniform."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.sin(theta)) / n
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def _spectral_init(n, rng, iters=200):
    """
    Flat-power spectrum initialization via iterative projection.
    Alternates between:
    - Flattening the power spectrum in frequency domain
    - Enforcing non-negativity in time domain
    """
    N = _next_pow2(n)
    # Start with random phases, unit magnitude
    theta = rng.uniform(0, 2 * np.pi, N // 2 + 1)
    theta[0] = 0.0  # DC positive
    F = np.exp(1j * theta)
    F[0] = 1.0

    a = None
    for it in range(iters):
        if it % 2 == 0:
            # Frequency domain: flatten magnitude
            a = np.fft.irfft(F, N)[:n]
            a = np.abs(a)  # non-negativity constraint
        else:
            # Time domain: reshape frequency
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s < 0.01:
                a = np.ones(n) / n
            else:
                a = a / s
            F_full = np.fft.rfft(a, N)
            # Flatten: set all magnitudes to 1 (except DC)
            mags = np.abs(F_full)
            mags[mags < 1e-10] = 1.0
            F = F_full / mags
            F[0] = np.abs(F[0])  # keep DC real and positive

    # Final time-domain
    a = np.fft.irfft(F, N)[:n]
    a = np.abs(a)
    a = _normalize(a)
    return a


def _soft_weights(c, T):
    """
    Compute soft-max weights on convolution peaks.
    c has length 2n-1. Returns weights of same length.
    """
    c_max = c.max()
    c_min = c.min()
    rng_c = c_max - c_min
    if rng_c < 1e-15:
        return np.ones(c.size) / c.size
    c_norm = (c - c_min) / rng_c  # [0, 1]
    c_shift = c_norm - 1.0  # [-1, 0], max at 0
    exp_c = np.exp(T * c_shift)
    s = exp_c.sum()
    if s < 1e-300:
        w = np.zeros(c.size)
        idx = np.argmin(np.abs(c - c_max))
        w[idx] = 1.0
        return w
    return exp_c / s


def _optimize_mirror(a_init, deadline, T_start=3.0, T_end=60.0,
                     K_start=20, K_end=300, lr_frac=0.3,
                     patience=3000, seed=42):
    """
    Adaptive active-set mirror descent optimization.
    Uses both soft-max weighting and hard top-K in combination.
    """
    a = _normalize(a_init.copy())
    n = a.size
    rng = np.random.RandomState(seed)

    best_a = a.copy()
    best_score = _score_fast(best_a)
    it = 0
    last_improve = 0
    lr0 = lr_frac * n
    eval_interval = 4

    while time.monotonic() < deadline:
        it += 1

        # Annealing
        frac = min(1.0, it / 50000.0)
        T = T_start + (T_end - T_start) * frac
        K = int(K_start + (K_end - K_start) * frac)
        lr = lr0 / (1.0 + 0.005 * it)

        # Compute convolution
        c = _fft_convolve(a)
        if c.size == 0:
            break

        # Combined weighting: soft-max blended with hard top-K
        w_soft = _soft_weights(c, T)

        if K < c.size:
            top_idx = np.argpartition(c, -K)[-K:]
            w_hard = np.zeros(c.size)
            top_vals = c[top_idx]
            tv_shift = top_vals - top_vals.max()
            hw = np.exp(5.0 * tv_shift)
            hw /= hw.sum()
            for j, idx in enumerate(top_idx):
                w_hard[idx] = hw[j]
        else:
            w_hard = w_soft

        # Blend
        alpha = 0.5 + 0.5 * frac  # shift from soft to hard over time
        w = alpha * w_hard + (1.0 - alpha) * w_soft
        w = w / w.sum()  # renormalize weights

        # Gradient
        grad = _grad_at_peaks(a, w)

        # Project to sum=zero tangent
        grad_proj = grad - grad.mean()

        # Mirror descent update
        update = -lr * grad_proj
        update = np.clip(update, -5.0, 5.0)
        a_new = a * np.exp(update)
        a_new = np.where(np.isfinite(a_new), a_new, 0.0)
        a_new = np.clip(a_new, 0.0, 1000.0)
        s = a_new.sum()
        if s < 1e-10:
            a = best_a.copy()
            continue
        a = a_new / s

        # Periodic score evaluation
        if it % eval_interval == 0:
            sc = _score_fast(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
                # Slightly increase lr on improvement
                lr0 = min(0.8 * n, lr0 * 1.005)
            elif it - last_improve > patience:
                # Restart with perturbation
                a = best_a.copy()
                noise_scale = 0.02 + 0.01 * rng.random()
                noise = rng.standard_normal(n) * noise_scale * a.mean()
                a = a + noise
                a = _normalize(a)
                patience = max(1500, int(patience * 1.5))
                lr0 = max(0.05 * n, lr0 * 0.7)

    return best_a, best_score


def _optimize_line_search(a_init, deadline, seed=99):
    """
    Mirror descent with backtracking line search.
    More conservative but potentially more reliable.
    """
    a = _normalize(a_init.copy())
    n = a.size
    rng = np.random.RandomState(seed)

    best_a = a.copy()
    best_score = _score_fast(best_a)
    it = 0
    last_improve = 0
    lr = 0.5 * n
    patience = 2000
    eval_interval = 3

    while time.monotonic() < deadline:
        it += 1

        # Compute convolution and find active peaks
        c = _fft_convolve(a)
        if c.size == 0:
            break

        # Use top-K with adaptive K
        K = min(50, c.size)
        top_idx = np.argpartition(c, -K)[-K:]
        c_max = c.max()
        # Only include peaks that are "close" to the max
        threshold = c_max * 0.95
        active = top_idx[c[top_idx] > threshold]
        if len(active) == 0:
            active = top_idx[:1]

        # Uniform weights on active peaks
        w = np.zeros(c.size)
        w[active] = 1.0 / len(active)

        # Gradient
        grad = _grad_at_peaks(a, w)
        grad_proj = grad - grad.mean()

        # Mirror descent with line search
        found = False
        for trial in range(5):
            step = lr * (0.5 ** trial)
            update = -step * grad_proj
            update = np.clip(update, -3.0, 3.0)
            a_try = a * np.exp(update)
            a_try = np.where(np.isfinite(a_try), a_try, 0.0)
            a_try = np.clip(a_try, 0.0, 1000.0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s

            sc_try = _score_fast(a_try)
            if np.isfinite(sc_try) and sc_try < best_score:
                a = a_try
                found = True
                break

        if not found:
            # Accept the first trial anyway (or skip)
            update = -lr * grad_proj
            update = np.clip(update, -3.0, 3.0)
            a_try = a * np.exp(update)
            a_try = np.where(np.isfinite(a_try), a_try, 0.0)
            a_try = np.clip(a_try, 0.0, 1000.0)
            s = a_try.sum()
            if s > 1e-10:
                a = a_try / s

        if it % eval_interval == 0:
            sc = _score_fast(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.01
                a = a + noise * a.mean()
                a = _normalize(a)
                patience = int(patience * 1.3)
                lr = max(0.05 * n, lr * 0.8)

    return best_a, best_score


def _coordinate_descent(a_init, deadline, seed=77):
    """
    Block coordinate descent: divide sequence into blocks,
    optimize each block's uniform shift while keeping others fixed.
    """
    a = _normalize(a_init.copy())
    n = a.size
    rng = np.random.RandomState(seed)

    # Divide into blocks
    n_blocks = 32
    block_size = n // n_blocks
    if block_size < 4:
        n_blocks = max(4, n // 4)
        block_size = n // n_blocks

    best_a = a.copy()
    best_score = _score_fast(best_a)
    it = 0

    while time.monotonic() < deadline:
        improved = False
        for b in range(n_blocks):
            start = b * block_size
            end = start + block_size if b < n_blocks - 1 else n

            # Try shifting this block
            a_orig_block = a[start:end].copy()
            for delta in [0.1, -0.1, 0.05, -0.05, 0.2, -0.2]:
                shift = delta * a.mean()
                a[start:end] = np.clip(a_orig_block + shift, 0.0, 1000.0)
                s = a.sum()
                if s < 0.01:
                    a[start:end] = a_orig_block
                    continue
                a_norm = a / s
                sc = _score_fast(a_norm)
                if sc < best_score:
                    best_score = sc
                    best_a = a_norm.copy()
                    improved = True
                    break
            a[start:end] = a_orig_block

        if not improved:
            # Perturb and try again
            idx = rng.randint(n)
            a[idx] *= rng.uniform(0.7, 1.3)
            a = _normalize(a)

        it += 1
        if it % 100 == 0:
            print(f"    [coord] it={it}, score={best_score:.8f}", flush=True)

    return best_a, best_score

    if remaining4 > 30 and np.isfinite(best_sc):
        print(f"[Phase 4] Coordinate descent (budget={remaining4:.0f}s)...", flush=True)
        a_cd, sc_cd = _coordinate_descent(best_a, d4, seed=seed + 9000)
        print(f"  Coord descent result: {sc_cd:.8f}", flush=True)
        if sc_cd < best_sc:
            best_sc = sc_cd
            best_a = a_cd.copy()

    # --- Phase 5: Validation ---
    print("[Phase 5] Validating with evaluate_sequence...", flush=True)

    # Build final candidate list
    final_candidates = [(best_a, best_sc)]
    for name, a, sc, n in candidates[:3]:
        final_candidates.append((a, sc))

    # Sort by internal score
    final_candidates.sort(key=lambda x: x[1])

    best_out = None
    best_actual = np.inf

    for arr, sc in final_candidates[:5]:
        if time.monotonic() > hard_deadline:
            break
        arr_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        try:
            actual = float(evaluate_sequence(arr_list))
        except Exception:
            actual = np.inf
        if np.isfinite(actual) and actual < best_actual:
            best_actual = actual
            best_out = arr_list
            print(f"  Validated (n={len(arr_list)}): actual={actual:.8f}", flush=True)
            if best_actual < 1.50:
                break

    if best_out is None:
        # Fallback: return the best internal candidate
        arr = best_a
        best_out = [float(max(0.0, min(1000.0, x))) for x in arr]
        try:
            best_actual = float(evaluate_sequence(best_out))
        except Exception:
            best_actual = best_sc
        print(f"  Fallback: {best_actual:.8f}", flush=True)

    total_time = time.monotonic() - t0
    print(f"[Done] Time: {total_time:.1f}s, Best score: {best_actual:.8f}", flush=True)
    return best_out
