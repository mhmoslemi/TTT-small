def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_objective_and_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -50, 50)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient via cross-correlation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5

        sc, g = lse_objective_and_grad(x, beta)
        return sc, g
    
    res = minimize(
        obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out

def peak_shave(a, n_shaves=100, strength=0.1):
    """Iteratively identify the peak convolution lag and reduce contributing elements."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = compute_score(a)
    best_a = a.copy()
    
    for _ in range(n_shaves):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        M = c[k_star]
        S = float(a.sum())
        sc = 2.0 * n * M / (S * S)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # Find contributing indices
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        # Contribution of a[j] to c[k_star] is a[k_star - j] * 2 (for j != k_star-j)
        # or a[j]^2 (for j == k_star-j)
        # We want to reduce a[j] for the j's that contribute most
        contrib = np.zeros(n)
        for j in range(j_start, j_end + 1):
            k = k_star - j
            contrib[j] = a[k]
        
        # Reduce top contributors, add to bottom contributors
        top_count = max(5, n // 100)
        sorted_idx = np.argsort(contrib)[::-1]
        top_idx = sorted_idx[:top_count]
        bot_idx = sorted_idx[len(sorted_idx) - top_count:]
        
        total_red = 0.0
        for j in top_idx:
            red = a[j] * strength
            a[j] -= red
            a[j] = max(a[j], 0.0)
            total_red += red
        
        if len(bot_idx) > 0 and total_red > 0:
            per = total_red / len(bot_idx)
            for j in bot_idx:
                a[j] += per
        
        a = np.clip(a, 0.0, 1000.0)
    
    final_sc = compute_score(a)
    if final_sc < best_score:
        best_score = final_sc
        best_a = a.copy()
    return best_score, best_a

"""
Multi-strategy optimizer minimizing 2n*max(conv(a,a))/(sum(a))^2.
Key improvements over previous:
  - Frequency-domain-inspired initializations (flat-spectrum targets)
  - Aggressive LSE continuation with geometric beta schedule
  - Multi-restart with diverse perturbations of parent
  - Iterative peak shaving with multi-peak awareness
  - Adam optimizer as complementary gradient method
  - Adaptive time allocation based on observed progress
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_obj_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -60, 60)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Cross-correlation for gradient
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=3000, maxfun=60000):
    """L-BFGS-B on LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj(x):
        return lse_obj_grad(x, beta)

    res = minimize(
        obj, a0, method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 40, 'gtol': 1e-12}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    return compute_score(a_out), a_out


def run_adam(a0, beta, lr=0.005, n_iters=3000, beta1=0.9, beta2=0.999):
    """Adam on LSE objective with decaying LR."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_sc = compute_score(a)
    best_a = a.copy()
    eps = 1e-8

    for t in range(n_iters):
        sc, grad = lse_obj_grad(a, beta)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (t + 1))
        v_hat = v / (1 - beta2 ** (t + 1))
        lr_t = lr * (0.95 ** (t // 500))
        a -= lr_t * m_hat / (np.sqrt(v_hat) + eps)
        a = np.clip(a, 0.0, 1000.0)

        if (t + 1) % 1000 == 0:
            sc_now = compute_score(a)
            if sc_now < best_sc:
                best_sc = sc_now
                best_a = a.copy()

    sc_now = compute_score(a)
    if sc_now < best_sc:
        best_sc = sc_now
        best_a = a.copy()
    return best_sc, best_a


def peak_shave_multi(a, n_iters=500, strength=0.03, peak_frac=0.98):
    """Iteratively shave multiple top convolution peaks."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_sc = compute_score(a)
    best_a = a.copy()

    for _ in range(n_iters):
        c = fast_conv(a)
        c_max = float(c.max())
        threshold = c_max * peak_frac
        peak_lags = np.where(c >= threshold)[0]

        # Accumulate contributions from all nearby peaks
        contrib = np.zeros(n)
        for k in peak_lags:
            k = int(k)
            j_lo = max(0, k - n + 1)
            j_hi = min(k, n - 1)
            for j in range(j_lo, j_hi + 1):
                contrib[j] += a[k - j]

        S = float(a.sum())
        sc = 2.0 * n * c_max / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        top_count = max(10, n // 80)
        order = np.argsort(contrib)[::-1]
        top_idx = order[:top_count]
        bot_idx = order[order.size - top_count:]

        total_red = 0.0
        for j in top_idx:
            r = a[j] * strength
            a[j] = max(a[j] - r, 0.0)
            total_red += r

        if len(bot_idx) > 0 and total_red > 0:
            per = total_red / len(bot_idx)
            for j in bot_idx:
                a[j] += per
        a = np.clip(a, 0.0, 1000.0)

    sc_final = compute_score(a)
    if sc_final < best_sc:
        best_sc = sc_final
        best_a = a.copy()
    return best_sc, best_a


def make_flat_spectrum_init(n, rng):
    """Create a sequence with approximately flat autocorrelation via random phases."""
    # Target: |A(f)|^2 = const => A(f) = sqrt(n/S) * exp(i*phi_f)
    # For real a, need A(-f) = conj(A(f))
    # Build half-spectrum with random phases
    M = n
    phi = rng.uniform(0, 2 * np.pi, M // 2 + 1)
    # Ensure DC is real (for real signal)
    A = np.zeros(M // 2 + 1, dtype=np.complex128)
    A[0] = np.sqrt(n * 0.5)
    A[M // 2:] = np.exp(1j * phi[M // 2:])  # Nyquist must be real for even n
    # Fill conjugate symmetry
    A_full = np.zeros(M, dtype=np.complex128)
    A_full[:M // 2 + 1] = A
    if M % 2 == 0:
        A_full[M // 2 + 1:] = np.conj(A_full[1:M // 2 + 1][::-1])
    else:
        A_full[M // 2 + 1:] = np.conj(A_full[1:M // 2 + 1][::-1])

    a = np.fft.ifft(A_full).real
    a = np.abs(a)  # Make non-negative
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.5
    return a


def build_inits(n, rng, parent=None):
    """Generate diverse initializations."""
    inits = []
    if parent is not None and len(parent) == n:
        inits.append(parent.copy())

    x = np.arange(n, dtype=np.float64) / n

    # Flat spectrum init
    inits.append(make_flat_spectrum_init(n, rng))
    inits.append(make_flat_spectrum_init(n, rng))

    # Uniform
    inits.append(np.ones(n) * 0.5)

    # Cosine modulations
    for k in [1, 2, 3]:
        inits.append(np.clip(0.5 + 0.3 * np.cos(2 * np.pi * k * x), 0.01, 10))

    # Random
    for sig in [0.2, 0.5, 1.0]:
        inits.append(np.clip(0.5 + sig * rng.standard_normal(n), 0.01, 1000.0))

    # Paleotopic (palindromic)
    half = n // 2
    rh = np.clip(0.5 + 0.4 * rng.standard_normal(half), 0.01, 1000.0)
    if n % 2 == 0:
        inits.append(np.concatenate([rh, rh[::-1]]))
    else:
        inits.append(np.concatenate([rh, [0.5], rh[::-1]]))

    # Block
    for nb in [8, 32]:
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())

    return inits
