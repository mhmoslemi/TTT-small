"""
Step-function optimizer: minimizes 2*n*max(conv(a,a))/(sum(a))^2
for non-negative sequences. Key improvements over previous version:
- Nesterov momentum in gradient steps
- Incremental convolution updates in coordinate polish (avoids FFT per move)
- Finer dv grid and more donor/receiver pairs
- Spectral perturbation for escaping local minima
- Multi-length exploration (n=6121 primary, n=10000 secondary)
- Adaptive active-set growth as convergence slows
"""

import numpy as np
import time

TARGET_SCORE = 1.5027


def _normalize(v):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v


def _fft_size(m):
    if m <= 2:
        return 2
    return 1 << (int(m) - 1).bit_length()


def _autocorr(f):
    n = f.size
    m = 2 * n - 1
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]


def _fast_score(f):
    f = _normalize(f)
    if f.size == 0:
        return np.inf
    c = _autocorr(f)
    if c.size == 0:
        return np.inf
    return float(2.0 * f.size * c.max())


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1.0)
    out = v - theta
    np.maximum(out, 0.0, out)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    out /= s
    return out


def _weighted_gradient(p, f):
    n = f.size
    L = p.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = L + n - 1
    size = _fft_size(m)
    P = np.fft.rfft(p, size)
    F = np.fft.rfft(f[::-1], size)
    conv = np.fft.irfft(P * F, size)
    return 2.0 * conv[n - 1:n + n - 1]


def _interp_to_length(f, n):
    f = _normalize(f)
    m = f.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if m == n:
        return f
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    if m == 1:
        return _normalize(np.full(n, f[0]))
    x_old = np.linspace(0.0, m - 1, m, dtype=np.float64)
    x_new = np.linspace(0.0, n - 1, n, dtype=np.float64)
    return _normalize(np.interp(x_new, x_old, f))


def _smooth_with_window(f, win):
    f = _normalize(f)
    n = f.size
    if n <= 2:
        return f
    win = int(win)
    if win < 3:
        win = 3
    if win > n:
        win = max(3, n // 2)
    kernel = np.ones(win, dtype=np.float64) / float(win)
    out = np.convolve(f, kernel, mode="same")
    return _normalize(out)


def _fourier_seed(n, rng):
    if n == 0:
        return np.empty(0, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    u = np.zeros(n, dtype=np.float64)
    u += float(rng.uniform(-1.0, 1.0)) * (x - 0.5)
    for m in (1, 2, 3, 5, 8):
        u += 0.40 * float(rng.uniform(-1.0, 1.0)) * np.cos(
            2.0 * np.pi * m * x + float(rng.uniform(0.0, 2.0 * np.pi))
        )
    u += 0.40 * float(rng.uniform(-1.0, 1.0)) * np.sin(
        2.0 * np.pi * x + float(rng.uniform(0.0, 2.0 * np.pi))
    )
    f = np.exp(0.65 * u)
    return _normalize(f)


def _beta_seed(n, rng):
    if n == 0:
        return np.empty(0, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = float(rng.uniform(0.2, 3.0))
    b = float(rng.uniform(0.2, 3.0))
    f = (x + 0.005) ** a * (1.0 - x + 0.005) ** b
    return _normalize(f)


def _spectral_shake(f, rng, alpha=0.3):
    """Randomize high-frequency phase to escape local minima."""
    n = f.size
    L = 2 * n - 1
    size = _fft_size(L)
    F = np.fft.rfft(f, size)
    # Keep low frequencies, randomize high
    cutoff = size // (2 * 10)  # keep bottom ~5%
    for k in range(cutoff, size // 2 + 1):
        if k < len(F):
            phase = float(rng.uniform(0, 2 * np.pi))
            F[k] = np.abs(F[k]) * (np.cos(phase) + 1j * np.sin(phase))
    f_new = np.fft.irfft(F, size)
    f_new = f_new[:n]
    return _normalize(np.maximum(f_new, 0.0))


def _coordinate_polish(f0, c0, budget_s, rng):
    """
    Local search by exact single mass transfers with incremental c updates.
    Much faster than FFT-per-move approach.
    """
    f = _normalize(f0)
    n = f.size
    if n < 2 or budget_s <= 0.0:
        c = _autocorr(f)
        return f, float(2.0 * n * c.max()), c

    L = 2 * n - 1
    try:
        c = np.asarray(c0, dtype=np.float64).copy()
    except Exception:
        c = _autocorr(f)
    if c.size != L or not np.all(np.isfinite(c)):
        c = _autocorr(f)

    maxc = float(np.max(c))
    best_f = f.copy()
    best_mc = maxc
    best_score = float(2.0 * n * maxc)

    deadline = time.monotonic() + float(budget_s)
    delta = np.zeros(L, dtype=np.float64)
    total = np.zeros(L, dtype=np.float64)
    no_improve = 0
    move_count = 0
    activeK = max(40, min(L, n // 5))

    # Parameters tuned for efficiency
    if n > 10000:
        D = 10
        R = 10
        n_fracs = 8
    elif n > 5000:
        D = 14
        R = 14
        n_fracs = 10
    else:
        D = 16
        R = 16
        n_fracs = 10

    fracs = np.geomspace(0.005, 0.6, n_fracs)

    while time.monotonic() < deadline and no_improve < 20:
        maxc = float(np.max(c))

        # Active lags
        if activeK >= L:
            active = np.arange(L)
        else:
            active = np.argpartition(c, L - activeK)[L - activeK:]

        p = np.zeros(L, dtype=np.float64)
        p[active] = 1.0 / float(active.size)
        g = _weighted_gradient(p, f)

        donor_order = np.argsort(g)[::-1]
        recv_order = np.argsort(g)

        donors = []
        for idx in donor_order:
            idx = int(idx)
            if f[idx] > 1e-12:
                donors.append(idx)
                if len(donors) >= D:
                    break

        donor_set = set(donors)
        receivers = []
        for idx in recv_order:
            idx = int(idx)
            if idx not in donor_set:
                receivers.append(idx)
                if len(receivers) >= R:
                    break

        if not donors or not receivers:
            no_improve += 1
            if no_improve >= 8:
                break
            continue

        best_mc = maxc
        best_move = None

        # Try donor-receiver pairs
        for i in donors:
            fi = float(f[i])
            if fi <= 1e-12:
                continue
            for j in receivers:
                if i == j:
                    continue
                for frac in fracs:
                    dv = fi * frac
                    if dv > fi * 0.95:
                        continue
                    # Compute delta using old f
                    delta.fill(0.0)
                    delta[i:i + n] -= f
                    delta[j:j + n] += f
                    delta *= (2.0 * dv)
                    q = dv * dv
                    if 2 * j < L:
                        delta[2 * j] += q
                    if 2 * i < L:
                        delta[2 * i] += q
                    if i + j < L:
                        delta[i + j] -= 2.0 * q
                    np.add(c, delta, out=total)
                    mc = float(np.max(total))
                    if mc < best_mc - 1e-18:
                        best_mc = mc
                        best_move = (i, j, dv)

        # Random fallback
        if best_move is None:
            for _ in range(20):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j or f[i] <= 1e-12:
                    continue
                frac = float(fracs[3 + int(rng.integers(0, 4))])
                dv = float(f[i]) * frac
                delta.fill(0.0)
                delta[i:i + n] -= f
                delta[j:j + n] += f
                delta *= (2.0 * dv)
                q = dv * dv
                if 2 * j < L:
                    delta[2 * j] += q
                if 2 * i < L:
                    delta[2 * i] += q
                if i + j < L:
                    delta[i + j] -= 2.0 * q
                np.add(c, delta, out=total)
                mc = float(np.max(total))
                if mc < best_mc - 1e-18:
                    best_mc = mc
                    best_move = (i, j, dv)
                    break

        if best_move is None:
            # Gradually increase active set
            activeK = int(min(L, activeK * 1.3))
            no_improve += 1
            continue

        i, j, dv = best_move
        # Apply transfer with incremental update
        delta.fill(0.0)
        delta[i:i + n] -= f  # f is still old
        delta[j:j + n] += f
        delta *= (2.0 * dv)
        q = dv * dv
        if 2 * j < L:
            delta[2 * j] += q
        if 2 * i < L:
            delta[2 * i] += q
        if i + j < L:
            delta[i + j] -= 2.0 * q
        c = c + delta
        f[i] -= dv
        f[j] += dv
        move_count += 1
        no_improve = 0

        # Periodic recomputation to avoid drift
        if move_count % 80 == 0:
            f = _normalize(f)
            c = _autocorr(f)
            maxc = float(np.max(c))

        maxc = float(np.max(c))
        cur_score = float(2.0 * n * maxc)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()

    f = _normalize(f)
    c = _autocorr(f)
    if c.size:
        mc = float(c.max())
        sc = float(2.0 * n * mc)
        if sc <= best_score:
            return f, sc, c
    bf = _normalize(best_f)
    bc = _autocorr(bf)
    bmc = float(bc.max()) if bc.size else 0.0
    return bf, float(2.0 * n * bmc), bc


def _gradient_optimize(f0, budget_s, rng, tag=""):
    """
    Annealed subgradient descent with Nesterov momentum,
    periodic coordinate polish, and spectral perturbation.
    """
    f0 = _normalize(f0)
    n = f0.size
    if n == 0:
        return f0, np.inf
    if budget_s <= 0.0:
        c = _autocorr(f0)
        return f0, float(2.0 * n * c.max())

    f = f0.copy()
    c = _autocorr(f)
    maxc = float(c.max())
    best_f = f.copy()
    best_score = float(2.0 * n * maxc)

    # Nesterov momentum
    v = np.zeros(n, dtype=np.float64)
    beta_mom = 0.9

    deadline = time.monotonic() + float(budget_s)
    sigma_rel = 0.06
    min_sigma_rel = 1e-10
    no_accept = 0
    it = 0
    last_coord = time.monotonic()
    last_shake = time.monotonic()

    print(f"[{tag}] start n={n} score={best_score:.8f}")

    while time.monotonic() < deadline:
        it += 1
        if maxc <= 0.0:
            break

        # Softmax weights over lags
        T = maxc * sigma_rel + 1e-300
        z = (c - maxc) / T
        np.clip(z, -45.0, 0.0, out=z)
        p = np.exp(z)
        ps = float(p.sum())
        if ps < 1e-300:
            p.fill(1.0 / float(c.size))
        else:
            p /= ps

        g = _weighted_gradient(p, f)
        gm = g - float(g.mean())
        scale = float(gm.max() - gm.min())

        accepted = False
        if scale > 1e-300:
            best_mc = maxc
            best_ft = None
            best_ct = None

            # Adaptive step sizes
            if no_accept > 40:
                exps = (0.003, 0.01, 0.05, 0.2)
            elif no_accept > 15:
                exps = (0.01, 0.05, 0.2, 1.0)
            else:
                exps = (0.03, 0.1, 0.5, 2.0)

            for e in exps:
                # Nesterov: look ahead then step back
                y = f + beta_mom * v
                arg = -e * (gm / scale)
                np.clip(arg, -30.0, 30.0, out=arg)
                ft = y * np.exp(arg)
                ft = _normalize(ft)
                ct = _autocorr(ft)
                mc = float(ct.max())
                if mc < best_mc:
                    best_mc = mc
                    best_ft = ft
                    best_ct = ct

                # Also try Euclidean step
                if e in (0.1, 0.5):
                    ft2 = y - e * (gm / scale)
                    ft2 = _project_simplex(ft2)
                    ct2 = _autocorr(ft2)
                    mc2 = float(ct2.max())
                    if mc2 < best_mc:
                        best_mc = mc2
                        best_ft = ft2
                        best_ct = ct2

            if best_ct is not None and best_mc < maxc - 1e-18:
                # Update momentum
                v = beta_mom * v + (best_ft - f)
                f = best_ft
                c = best_ct
                maxc = best_mc
                accepted = True

        if accepted:
            no_accept = 0
            cur_score = float(2.0 * n * maxc)
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()
                if best_score < TARGET_SCORE:
                    print(f"[{tag}] NEW BEST below target: {best_score:.8f}")
            sigma_rel = max(min_sigma_rel, sigma_rel * 0.995)
        else:
            no_accept += 1
            now = time.monotonic()

            # Coordinate polish
            if now - last_coord > 0.8 or no_accept > 25:
                coord_budget = min(1.5, max(0.3, (deadline - now) * 0.01))
                fp, sc, cp = _coordinate_polish(f, c, coord_budget, rng)
                if sc < float(2.0 * n * maxc) - 1e-12:
                    f = _normalize(fp)
                    c = cp
                    maxc = float(c.max())
                    v = np.zeros(n, dtype=np.float64)
                    accepted = True
                    no_accept = 0
                    if sc < best_score:
                        best_score = sc
                        best_f = f.copy()
                        if best_score < TARGET_SCORE:
                            print(f"[{tag}] coord polish below target: {best_score:.8f}")
                else:
                    # Perturbation
                    if no_accept > 40 or (now - last_shake > 15.0 and no_accept > 20):
                        r = float(rng.random())
                        if r < 0.4:
                            f = _spectral_shake(best_f, rng, 0.4)
                        elif r < 0.7:
                            sigma_p = float(rng.uniform(0.1, 0.8))
                            f = best_f * np.exp(sigma_p * rng.standard_normal(n))
                            f = _normalize(f)
                        else:
                            alpha = float(rng.uniform(0.1, 0.5))
                            noise = rng.random(n)
                            s = noise.sum()
                            if s > 1e-300:
                                noise /= s
                            f = (1.0 - alpha) * best_f + alpha * noise
                            f = _normalize(f)
                        c = _autocorr(f)
                        maxc = float(c.max())
                        v = np.zeros(n, dtype=np.float64)
                        sigma_rel = min(0.2, sigma_rel * 2.5 + 1e-4)
                        no_accept = 0
                        last_shake = now
                last_coord = time.monotonic()
            else:
                sigma_rel = max(min_sigma_rel, sigma_rel * 0.997)

        if it % 10000 == 0:
            cur_score = float(2.0 * n * float(c.max()))
            print(f"[{tag}] it={it} best={best_score:.8f} cur={cur_score:.8f} sigma={sigma_rel:.3e}")

    f = _normalize(f)
    c = _autocorr(f)
    maxc = float(c.max())
    cur_score = float(2.0 * n * maxc)
    if cur_score < best_score:
        best_score = cur_score
        best_f = f.copy()

    print(f"[{tag}] end best={best_score:.8f}")
    return _normalize(best_f), float(best_score)


def _build_candidates(f0, n, rng):

    cands = []
    f0n = _interp_to_length(f0, n)
    cands.append(("given", f0n))
    cands.append(("reverse", _normalize(f0n[::-1])))

    for w in (7, 21, 61):
        cands.append((f"sm{w}", _smooth_with_window(f0n, w)))

    for g in (0.5, 0.75, 1.5):
        cands.append((f"pwr{g}", _normalize(np.power(f0n + 1e-12, g))))

    cands.append(("uniform", np.ones(n, dtype=np.float64) / n))

    for i in range(6):
        cands.append((f"fou{i}", _fourier_seed(n, rng)))
    for i in range(4):
        cands.append((f"bet{i}", _beta_seed(n, rng)))

    # Coarse-fine
    if n > 256:
        for K in (128, 256, 512):
            if K < n:
                coarse = _normalize(np.interp(
                    np.linspace(0, K-1, K), np.linspace(0, n-1, n), f0n
                ))
                up = _interp_to_length(coarse, n)
                cands.append((f"cf{K}", up))

    # Evaluate and sort
    scored = []
    for name, f in cands:
        f = _normalize(f)
        try:
            sc = _fast_score(f)
        except Exception:
            sc = np.inf
        if np.isfinite(sc):
            scored.append((sc, name, f))

    scored.sort(key=lambda x: x[0])
    return scored


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative coefficient sequence minimizing evaluate_sequence.
    Returns a Python list of floats.
    """
    rng = np.random.default_rng(seed)
    total_start = time.monotonic()
    total_deadline = total_start + max(120.0, float(budget_s)) - 30.0

    # Load initial sequence
    try:
        f0 = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        f0 = np.ones(6121, dtype=np.float64)

    if f0.size == 0:
        f0 = np.ones(6121, dtype=np.float64)
    f0 = np.nan_to_num(f0, nan=0.0, posinf=1000.0, neginf=0.0)
    np.clip(f0, 0.0, 1000.0, out=f0)
    n0 = int(f0.size)
    if n0 < 3:
        f0 = np.ones(6121, dtype=np.float64)
        n0 = f0.size
    f0 = _normalize(f0)

    best_f = f0.copy()
    best_c = _autocorr(best_f)
    best_score = float(2.0 * n0 * best_c.max()) if best_c.size else np.inf
    print(f"Initial n={n0} score={best_score:.8f}")

    # Strategy: optimize at n0 (primary) and n0*1.6 (secondary)
    n_list = [n0]
    n_alt = int(n0 * 1.6)
    if n_alt != n0 and n_alt <= 20000:
        n_list.append(n_alt)

    # Allocate time: 70% to primary, 25% to secondary, 5% final polish
    time_budget = total_deadline - time.monotonic()
    n_allocs = {}
    if len(n_list) == 1:
        n_allocs[n_list[0]] = time_budget * 0.95
    else:
        n_allocs[n_list[0]] = time_budget * 0.65
        n_allocs[n_list[1]] = time_budget * 0.25
        final_polish_time = time_budget * 0.10

    for ni, n_target in enumerate(n_list):
        if time.monotonic() >= total_deadline - 30:
            break
        remaining = total_deadline - time.monotonic() - 20
        alloc = min(n_allocs.get(n_target, remaining * 0.5), remaining * 0.7)
        if alloc < 30:
            continue

        print(f"\n{'='*60}")
        print(f"Optimizing at n={n_target} (budget={alloc:.0f}s)")
        print(f"{'='*60}")

        # Build candidates for this n
        cands = _build_candidates(f0, n_target, rng)
        if not cands:
            cands = [(best_score, "uniform", _interp_to_length(f0, n_target))]

        # Optimize top candidates
        top_k = min(3, len(cands))
        for ci in range(top_k):
            if time.monotonic() >= total_deadline - 30:
                break
            sc_init, name, f_start = cands[ci]
            sub_deadline = total_deadline - 25
            sub_budget = min(alloc / (top_k - ci) * 0.9, sub_deadline - time.monotonic() - 10)
            if sub_budget < 20:
                break
            print(f"  Seed '{name}' init_score={sc_init:.8f} budget={sub_budget:.0f}s")
            f_opt, sc_opt = _gradient_optimize(f_start, sub_budget, rng, f"n{n_target}_{name}")
            if sc_opt < best_score:
                best_score = sc_opt
                best_f = f_opt.copy()
                best_c = _autocorr(best_f)
                print(f"  *** IMPROVED to {best_score:.8f} ***")

    # Final dedicated coordinate polish on the best
    if time.monotonic() < total_deadline - 30:
        final_budget = min(
            total_deadline - time.monotonic() - 20,
            max(60.0, 0.15 * (budget_s - 2400 + 2400))  # just use remaining
        )
        final_budget = min(final_budget, total_deadline - time.monotonic() - 20)
        if final_budget > 30:
            print(f"\nFinal coordinate polish: budget={final_budget:.0f}s")
            bf, bs, bc = _coordinate_polish(best_f, _autocorr(best_f), final_budget, rng)
            if bs < best_score:
                best_score = bs
                best_f = bf.copy()
                best_c = bc
                print(f"  Polish improved to {best_score:.8f}")

    # One more gradient pass if time allows
    if time.monotonic() < total_deadline - 40:
        extra_budget = min(120.0, total_deadline - time.monotonic() - 30)
        if extra_budget > 20:
            print(f"\nFinal gradient pass: budget={extra_budget:.0f}s")
            f2, s2 = _gradient_optimize(best_f, extra_budget, rng, "final_grad")
            if s2 < best_score:
                best_score = s2
                best_f = f2.copy()
                best_c = _autocorr(best_f)
                print(f"  Final gradient improved to {best_score:.8f}")

    # Ultimate final polish
    if time.monotonic() < total_deadline - 25:
        ultra_budget = total_deadline - time.monotonic() - 20
        if ultra_budget > 15:
            bf, bs, bc = _coordinate_polish(best_f, _autocorr(best_f), ultra_budget, rng)
            if bs < best_score:
                best_score = bs
                best_f = bf.copy()
                best_c = bc

    # Safety
    best_f = _normalize(best_f)
    best_f = np.nan_to_num(best_f, nan=0.0, posinf=0.0, neginf=0.0)
    np.clip(best_f, 0.0, 1000.0, out=best_f)
    s = float(best_f.sum())
    if s > 1e-300:
        best_f /= s

    # Final evaluation
    best_c = _autocorr(best_f)
    best_score = float(2.0 * best_f.size * best_c.max())

    try:
        out = [float(x) for x in best_f]
        official = evaluate_sequence(out)
        if np.isfinite(official) and official < best_score + 1e-12:
            best_score = float(official)
    except Exception:
        pass

    print(f"\n{'='*60}")
    print(f"FINAL score = {best_score:.8f}")
    print(f"Target      = {TARGET_SCORE}")
    print(f"n = {best_f.size}")
    print(f"{'='*60}")

    return [float(x) for x in best_f]
