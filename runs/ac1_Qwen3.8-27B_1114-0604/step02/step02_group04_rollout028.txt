```python
from scipy.optimize import linprog
import time

def solve_lp(f, M):
    """Solve the LP: maximize sum(g) subject to g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    # Build constraint matrix A: A @ g <= M for all k
    # A[k, i] = f[i] if 0 <= i+k <= n-1... wait, no.
    # (f*g)[k] = sum_i f[i] * g[i+k] for valid i, i+k
    # Actually, linear convolution: (f*g)[k] = sum_{i=0}^{n-1} f[i] * g[k-i]
    # where g is defined as 0 outside [0, n-1]
    # So (f*g)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} f[i] * g[k-i]
    # Let j = k-i, then i = k-j, and j ranges from max(0,k-n+1) to min(n-1,k)
    # (f*g)[k] = sum_{j=max(0,k-n+1)}^{min(n-1,k)} f[k-j] * g[j]
    
    # So A[k, j] = f[k-j] if 0 <= k-j <= n-1 and 0 <= j <= n-1, else 0
    # A[k, j] = f[k-j] if max(0, k-n+1) <= j <= min(n-1, k), else 0
    
    # This is a Toeplitz-like matrix but it's actually the full convolution matrix
    # A is (2n-1) x n
    
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            A[k, j] = f[k-j]
    
    # Maximize sum(g) = minimize -sum(g)
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    return None
```

```python
            continue

        if g_sum < 1e-10:

            continue
```

```python
LP-based optimization for minimum-peak nonnegative step functions.
Combines linear programming (from the SOTA paper) with gradient-based polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]

def _build_conv_matrix(f):
    """Build (2n-1) x n matrix A where (A @ g)[k] = (f*g)[k]."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            A[k, lo:hi + 1] = f[k - hi:k - lo + 1][::-1]
    return A


def _lp_iterate(f, deadline, t_init=0.1, patience=300):
    """
    LP-based optimization loop.
    f is normalized to sum=1, non-negative.
    Returns (best_f, best_score).
    """
    f = np.asarray(f, dtype=np.float64).copy()
    f = np.clip(f, 0, 1000)
    s = f.sum()
    if s < 0.01:
        f = np.ones(f.size) / f.size
        s = 1.0
    f = f / s

    n = f.size
    best_f = f.copy()
    best_score = _score(best_f)
    t = t_init
    no_improve = 0
    rng = np.random.RandomState(42)

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # Compute current max
        c = _fft_convolve(f)
        M = float(np.max(c))
        current_score = 2.0 * n * M  # since sum(f) = 1

        if current_score < best_score - 1e-10:
            best_score = current_score
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > patience:
            # Try perturbation
            f = best_f.copy()
            noise = rng.standard_normal(n) * 0.01
            f = f + noise * f.mean()
            f = np.clip(f, 1e-6, None)
            f = f / f.sum()
            no_improve = 0
            t = t_init
            patience = int(patience * 1.3)
            if patience > 1000:
                # Try reducing n? Or just continue
                patience = 1000
            if t < 0.01:
                t = 0.1

        # Build constraint matrix
        A = _build_conv_matrix(f)

        # Solve LP: max sum(g) s.t. A@g <= M, g >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True})
        except Exception:
            t = max(0.01, t * 0.5)
            continue

        if not res.success:
            t = max(0.01, t * 0.5)
            continue

        g = res.x
        g_sum = float(g.sum())

        if g_sum < 1e-10:
            # LP says no improvement possible
            t = max(0.01, t * 0.7)
            if t < 0.02:
                # Perturb and restart
                f = best_f.copy()
                noise = rng.standard_normal(n) * 0.05 * f.mean()
                f = f + noise
                f = np.clip(f, 1e-6, None)
                f = f / f.sum()
                t = t_init
                continue

        # Rescale g
        g = g / g_sum

        # Update f
        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0, 1000)
        s_new = f_new.sum()
        if s_new < 1e-10:
            f = best_f.copy()
            continue
        f_new = f_new / s_new
        f = f_new

    return best_f, best_score
```

```python
LP-based optimization for minimum-peak nonnegative step functions.
Combines the linear programming formulation (from SOTA paper) with
gradient-based mirror descent polishing and adaptive restarts.
The LP provides principled descent directions; gradient polish refines.
"""

import numpy as np
from scipy.optimize import linprog

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    """Compute evaluation score (scale-invariant)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())

        c = _fft_convolve(f)
        M = float(np.max(c))
        current_score = 2.0 * n * M  # since sum(f) = 1

        if current_score < best_score - 1e-12:
            best_score = current_score
            best_f = f.copy()
            no_improve = 0
            # Slightly increase step size on improvement
            t = min(0.3, t * 1.01)
        else:
            no_improve += 1

        # Restart logic
        if no_improve > restart_patience:
            f = best_f.copy()
            # Random perturbation
            scale = 0.02 * (1.0 + 0.5 * (it % 5))
            noise = rng.standard_normal(n) * scale
            f = f + noise * f.mean()
            f = np.clip(f, 1e-8, None)
            f = f / f.sum()
            no_improve = 0
            t = t_init
            restart_patience = min(2000, int(restart_patience * 1.2))
            continue

        # Build constraint matrix
        A = _build_conv_matrix(f)

        # Solve LP: maximize sum(g) subject to A@g <= M, g >= 0
        c_obj = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)

        try:
            res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs',
                          options={'presolve': True, 'time_limit': 5})
        except Exception:
            consecutive_fail += 1
            if consecutive_fail > 10:
                f = best_f.copy()
                noise = rng.standard_normal(n) * 0.03 * f.mean()
                f = np.clip(f + noise, 1e-8, None)
                f = f / f.sum()
                consecutive_fail = 0
            continue

        if not res.success:
            consecutive_fail += 1
            t = max(0.03, t * 0.8)
            if consecutive_fail > 15:
                f = best_f.copy()
                noise = rng.standard_normal(n) * 0.03 * f.mean()
                f = np.clip(f + noise, 1e-8, None)
                f = f / f.sum()
                consecutive_fail = 0
                t = t_init
            continue

        consecutive_fail = 0
        g = res.x
        g_sum = float(g.sum())

        if g_sum < 1e-8:
            # No feasible improvement direction
            t = max(0.03, t * 0.7)
            continue

        # Rescale g to sum = 1
        g = g / g_sum

        # Check how different g is from f
        diff = np.mean(np.abs(g - f))
        if diff < 1e-6:
            # g ≈ f, no progress
            f = best_f.copy()
            noise = rng.standard_normal(n) * 0.02 * f.mean()
            f = np.clip(f + noise, 1e-8, None)
            f = f / f.sum()
            continue

        # Update f
        f_new = (1.0 - t) * f + t * g
        f_new = np.clip(f_new, 0, 1000)
        s_new = f_new.sum()
        if s_new < 1e-10:
            f = best_f.copy()
            continue
        f_new = f_new / s_new
        f = f_new

    return best_f, best_score


def _gradient_polish(f, deadline, T_start=5.0, T_end=60.0):
    """
    Mirror descent (multiplicative weights) polishing using FFT-based gradients.
    Uses log-sum-exp surrogate for the max of auto-convolution.
    """
    f = np.asarray(f, dtype=np.float64).copy()
    f = np.clip(f, 0, 1000)
    s = f.sum()
    if s < 0.01:
        f = np.ones(f.size) / f.size
        s = 1.0
    f = f / s

    n = f.size
    best_f = f.copy()
    best_score = _score(best_f)
    lr = 0.3 * n
    it = 0
    no_improve = 0
    patience = 5000
    rng = np.random.RandomState(31415)

    while time.monotonic() < deadline:
        it += 1
        t_frac = min(1.0, it / 80000.0)
        T = T_start + (T_end - T_start) * t_frac
        lr = 0.3 * n / (1.0 + 0.003 * it)

        c = _fft_convolve(f)
        if c.size == 0:
            break

        # Softmax weights on convolution
        c_max = float(c.max())
        c_shift = c - c_max
        exp_c = np.exp(T * c_shift)
        w_sum = float(exp_c.sum())
        if w_sum < 1e-300:
            w = np.zeros(c.size)
            w[int(np.argmax(c))] = 1.0
        else:
            w = exp_c / w_sum

        # Compute h[i] = sum_k w[k] * f[k-i] via FFT cross-correlation
        # Using: h[i] = convolve(f, w[::-1])[2n-2-i]
        N = _next_pow2(n + 2 * n - 1)
        fw = np.fft.rfft(f, N)
        w_rev = np.zeros(N)
        w_rev[:2 * n - 1] = w[::-1]
        fw_rev = np.fft.rfft(w_rev, N)
        conv_res = np.fft.irfft(fw * fw_rev, N)

        # h[i] = conv_res[2n-2-i] for i=0,...,n-1
        h = conv_res[2 * n - 2: 2 * n - 2 - (n - 1): -1].copy()

        # Gradient (simplified, sum_f=1): proportional to h[i] - max_c * f[i]
        grad = h - c_max * f
        # Project to zero-mean (tangent to simplex)
        grad = grad - np.mean(grad)

        # Mirror descent update (multiplicative weights)
        update = -lr * grad
        update = np.clip(update, -4.0, 4.0)
        f = f * np.exp(update)
        f = np.where(np.isfinite(f), f, 0.0)
        f = np.clip(f, 0.0, 1000.0)
        s = f.sum()
        if s < 1e-10:
            f = best_f.copy()
            break
        f = f / s

        # Periodic evaluation
        if it % 10 == 0:
            sc = _score(f)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_f = f.copy()
                no_improve = 0
                lr = min(0.5 * n, lr * 1.005)
            else:
                no_improve += 10

        if no_improve > patience:
            f = best_f.copy()
            noise = rng.standard_normal(n) * 0.005
            f = f + noise * f.mean()
            f = np.clip(f, 0, None)
            s = f.sum()
            if s > 1e-10:
                f = f / s
            else:
                f = best_f.copy()
            no_improve = 0
            patience = int(min(15000, patience * 1.2))
            lr = max(0.1 * n, lr * 0.8)

    return best_f, best_score


def _make_sine_init(n, amp=0.75, phase=0.0):
    """Sinusoidal initialization."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.sin(2.0 * np.pi * (i + phase) / n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0.01:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def _make_multi_sine_init(n, amps=None, phases=None):
    """Multi-mode sinusoidal initialization."""
    rng = np.random.RandomState(np.random.randint(1000000))
    if amps is None:
        amps = [rng.uniform(0.4, 0.9), rng.uniform(0.1, 0.3)]
    if phases is None:
        phases = [rng.uniform(0, 0.5), rng.uniform(0, 1.0)]
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for amp, ph in zip(amps, phases):
        a += amp * np.sin(2.0 * np.pi * (i + ph) / n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0.01:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def _make_random_init(n, seed=0):
    """Exponential/linear ramp initialization."""
    rng = np.random.RandomState(seed)
    i = np.arange(n, dtype=np.float64) / n
    a = np.exp(rng.uniform(0.5, 2.0) * (2 * i - 1))
    a = a + 0.1 * rng.standard_normal(n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0.01:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for minimum-peak nonnegative step function.
    Phase 1: Diverse initialization scan.
    Phase 2: LP-based optimization (main algorithm).
    Phase 3: Gradient-based polishing.
    Phase 4: Validation and return best.
    """
    t0 = time.monotonic()
    hard_deadline = t0 + float(budget_s) - 20.0
    np.random.seed(seed)

    print(f"[Start] Budget: {budget_s}s", flush=True)

    # Phase 1: Generate and score diverse initializations
    d1 = t0 + 20.0
    print("[Phase 1] Scanning initializations...", flush=True)

    candidates = []
    n_default = 1024

    # Try to load height_sequence_1
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, 1000)
        hs_sum = hs.sum()
        if hs_sum > 0.01:
            candidates.append(hs / hs_sum)
            print(f"  Loaded height_sequence_1: n={hs.size}", flush=True)
    except Exception:
        pass

    # Sinusoidal inits
    for n in [1024, 2048]:
        if time.monotonic() > d1:
            break
        for amp in [0.3, 0.5, 0.6, 0.7, 0.75, 0.8, 0.9, 1.0]:
            a = _make_sine_init(n, amp)
            candidates.append(a)
        # Multi-sine
        for _ in range(3):
            a = _make_multi_sine_init(n)
            candidates.append(a)
        # Random
        a = _make_random_init(n, seed=n)
        candidates.append(a)

    # Constant (baseline)
    candidates.append(np.ones(n_default) / n_default)

    # Score all
    best_init = None
    best_init_score = np.inf
    for a in candidates:
        if time.monotonic() > d1:
            break
        sc = _score(a)
        if np.isfinite(sc) and sc < best_init_score:
            best_init_score = sc
            best_init = a.copy()

    print(f"  Best init: n={best_init.size}, score={best_init_score:.8f}", flush=True)

    # Phase 2: LP optimization (give most time here)
    d2 = hard_deadline - 350.0
    remaining2 = d2 - time.monotonic()
    print(f"[Phase 2] LP optimization ({remaining2:.0f}s budget)...", flush=True)

    all_results = [(best_init, best_init_score)]

    if remaining2 > 60:
        # Main LP run from best init
        a_lp, s_lp = _lp_optimize(best_init, d2, t_init=0.15)
        all_results.append((a_lp, s_lp))
        print(f"  LP from best init: score={s_lp:.8f}", flush=True)

        # Try LP from a different init (for diversity)
        d2b = hard_deadline - 200.0
        remaining2b = d2b - time.monotonic()
        if remaining2b > 60:
            # Pick a different init (not the same as best)
            alt_cands = [c for c in candidates if c is not best_init and c.size == best_init.size]
            if not alt_cands:
                alt_cands = [_make_sine_init(best_init.size, 0.5, 0.3)]
            alt_init = alt_cands[np.random.randint(len(alt_cands))]
            a_lp2, s_lp2 = _lp_optimize(alt_init, d2b, t_init=0.1)
            all_results.append((a_lp2, s_lp2))
            print(f"  LP from alt init: score={s_lp2:.8f}", flush=True)

    # Phase 3: Gradient polishing
    d3 = hard_deadline - 40.0
    remaining3 = d3 - time.monotonic()
    print(f"[Phase 3] Gradient polishing ({remaining3:.0f}s budget)...", flush=True)

    # Sort results by score
    all_results.sort(key=lambda x: x[1])
    best_so_far = all_results[0][0].copy()
    best_so_far_score = all_results[0][1]

    if remaining3 > 30:
        a_polished, s_polished = _gradient_polish(best_so_far, d3, T_start=5.0, T_end=80.0)
        all_results.append((a_polished, s_polished))
        print(f"  Polished: score={s_polished:.8f}", flush=True)

    # Also try polishing the second-best
    d3b = hard_deadline - 20.0
    remaining3b = d3b - time.monotonic()
    if remaining3b > 30 and len(all_results) >= 2:
        second_best_a = all_results[1][0]
        if second_best_a is not best_so_far:
            a_pol2, s_pol2 = _gradient_polish(second_best_a, d3b, T_start=3.0, T_end=50.0)
            all_results.append((a_pol2, s_pol2))
            print(f"  Polished 2nd: score={s_pol2:.8f}", flush=True)

    # Final short LP polish on the best
    d4 = hard_deadline - 15.0
    remaining4 = d4 - time.monotonic()
    if remaining4 > 30:
        all_results.sort(key=lambda x: x[1])
        best_a_current = all_results[0][0]
        a_final, s_final = _lp_optimize(best_a_current, d4, t_init=0.05)
        all_results.append((a_final, s_final))
        print(f"  Final LP: score={s_final:.8f}", flush=True)

    # Phase 4: Validate with evaluate_sequence
    print("[Phase 4] Validating...", flush=True)
    all_results.sort(key=lambda x: x[1])

    best_out = None
    best_actual = np.inf
    seen_sizes = set()

    for arr, sc in all_results[:6]:
        if time.monotonic() > hard_deadline:
            break
        arr_key = arr.size
        arr_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        try:
            actual = float(evaluate_sequence(arr_list))
        except Exception:
            actual = np.inf
        if np.isfinite(actual) and actual < best_actual:
            best_actual = actual
            best_out = arr_list
            print(f"  n={arr.size}: actual={actual:.8f}", flush=True)

    if best_out is None:
        # Fallback
        arr_list = [float(max(0.0, min(1000.0, x))) for x in best_init]
        try:
            best_actual = float(evaluate_sequence(arr_list))
        except Exception:
            best_actual = best_init_score
        best_out = arr_list
        print(f"  Fallback: {best_actual:.8f}", flush=True)

    total_time = time.monotonic() - t0
    print(f"[Done] Time: {total_time:.1f}s, Best: {best_actual:.8f}", flush=True)
    return best_out
```
