"""
Improved optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 using
log-sum-exp smoothing, adaptive temperature, and Adam with cosine schedule.
Focuses on asymmetric sequences which can achieve scores below 2.
"""

import numpy as np
import time


def fast_self_conv(a):
    """Compute self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, c, S):
    """Compute the objective."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def logsumexp(x, T):
    """Compute logsumexp(T*x)/T stably."""
    x_max = np.max(x)
    return x_max + np.log(np.sum(np.exp(T * (x - x_max)))) / T


def grad_logsumexp(a, c, T, topk=64):

    Compute gradient of the smoothed objective using log-sum-exp.

    order = np.argsort(vals)[::-1]

    vals = vals[order]

    # The true logsumexp over all c values would include the non-topk values too
    # For large T, the non-topk values contribute negligibly, so this is fine

def fast_self_conv(a):
    """Compute self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, c, S):
    """Objective value."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def grad_smooth(a, c, T, topk):

def adam_update(a, g, m, v, t, lr):
    """Adam optimizer step (in-place)."""
    b1, b2 = 0.9, 0.999
    m[:] = b1 * m + (1 - b1) * g
    v[:] = b2 * v + (1 - b2) * g * g
    m_hat = m / (1 - b1 ** t)
    v_hat = v / (1 - b2 ** t)
    a -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)


def optimize(a0, t_start, t_end, rng, lr_init, lr_end, T_start, T_end, topk, patience):
    """
    Run optimization from a0 until t_end.
    Returns (best_score, best_a).
    """
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a.sum() < 1e-6:
        a = np.ones(n) * 0.1
    c = fast_self_conv(a)
    S = float(a.sum())
    score = compute_score(n, c, S)
    best_score = score
    best_a = a.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_imp = 0
    cur_pat = patience
    
    total_time = t_end - t_start
    elapsed = 0.0
    
    while time.monotonic() < t_end:
        elapsed = time.monotonic() - t_start
        fraction = min(elapsed / total_time, 1.0)
        
        # Cosine learning rate schedule
        lr = lr_end + 0.5 * (lr_init - lr_end) * (1 + np.cos(np.pi * fraction))
        
        # Temperature schedule (decrease from T_start to T_end)
        T = T_start + (T_end - T_start) * fraction
        
        # Compute convolution and score
        c = fast_self_conv(a)
        score = compute_score(n, c, S)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        # Restart if stuck
        if no_imp >= cur_pat:
            # Structured perturbation
            perturb_type = rng.integers(0, 4)
            if perturb_type == 0:
                # Gaussian noise
                a = best_a * (1.0 + 0.05 * rng.standard_normal(n))
            elif perturb_type == 1:
                # Add ramp
                ramp = np.linspace(-0.1, 0.1, n)
                a = best_a + 0.1 * max(best_a.max(), 1.0) * ramp
            elif perturb_type == 2:
                # Add cosine
                x = np.linspace(0, 2 * np.pi * rng.integers(1, 5), n)
                a = best_a + 0.1 * max(best_a.max(), 1.0) * np.cos(x)
            else:
                # Flip a random segment
                start = rng.integers(0, n // 2)
                length = rng.integers(n // 10, n // 3)
                end = min(start + length, n)
                a = best_a.copy()
                seg = a[start:end].copy()
                a[start:end] = a[end-1:start-1:-1] if length > 1 else seg
            
            a = np.clip(a, 0.0, 1000.0)
            if a.sum() < 1e-3:
                a += 0.1
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            no_imp = 0
            cur_pat = max(20, cur_pat // 2)
            continue
        
        # Compute gradient
        g, M_s, S = grad_smooth(a, c, T, topk)
        t += 1
        adam_update(a, g, m, v, t, lr)
        a = np.clip(a, 0.0, 1000.0)
    
    return best_score, best_a


def init_asymmetric(n, rng):
    """Generate diverse asymmetric initializations."""
    inits = []
    # Linear ramp
    inits.append(np.linspace(0.1, 1.0, n))
    # Quadratic ramp
    x = np.linspace(0, 1, n)
    inits.append(0.1 + 0.9 * x**2)
    # Square root ramp
    inits.append(0.1 + 0.9 * np.sqrt(x))
    # Exponential ramp
    inits.append(0.1 + 0.9 * np.exp(2 * (x - 1)))
    # Two-level step (asymmetric)
    a_step = np.ones(n)
    a_step[:n//3] = 0.3
    a_step[n//2:] = 1.5
    inits.append(a_step)
    # Cosine-modulated ramp
    inits.append(0.3 + 0.7 * x + 0.2 * np.cos(2 * np.pi * x))
    # Random smooth (low-pass filtered noise)
    noise = rng.standard_normal(n)
    # Simple moving average as low-pass
    kernel_size = max(3, n // 50)
    kernel = np.ones(kernel_size) / kernel_size
    smooth_noise = np.convolve(noise, kernel, mode='same')
    inits.append(np.clip(0.5 + 0.5 * smooth_noise, 0.01, 1000))
    # Parent-like: smooth asymmetric
    inits.append(0.2 + 0.8 * x**0.7)
    return inits
