```python
    return None
```

```python
"""
LP-based and gradient-based optimizer for minimizing 2n*max(conv(a,a))/(sum a)^2.
Uses alternating LP projections (fixing one factor and optimizing the other) combined
with subgradient descent for fine-tuning. Multiple initializations and sequence lengths.

import numpy as np
import time
import scipy.optimize
from scipy.sparse import csr_matrix


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, conv, S):
    """Objective: 2*n*max(conv)/(sum a)^2."""
    if S < 0.01:
        return np.inf
    return float(2.0 * n * np.max(conv) / (S * S))


def init_cosine(n):
    """Cosine (Hann-like) taper."""
    x = np.arange(n)
    return np.sin(np.pi * (x + 0.5) / n) ** 2


def init_gaussian(n, sigma_frac=0.4):
    """Gaussian taper."""
    x = np.arange(n) - (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * (x / sigma) ** 2)


def init_super_gaussian(n, p=4):
    """Super-Gaussian taper (flatter top)."""
    x = (2.0 * np.arange(n) / (n - 1) - 1.0)  # in [-1, 1]
    return np.exp(-np.abs(x) ** p)


def init_ones(n):
    return np.ones(n)


def lp_project(f, M, top_k, n):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for top_k peaks, g >= 0.
    Returns g or None on failure.
    """
    # Use the top_k constraint rows
    # Constraint k: sum_j f[k-j] * g[j] <= M where 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    
    # Precompute f for indexing
    # For constraint k: coefficient of g[j] is f[k-j] if 0 <= k-j < n
    for k in top_k:
        k_int = int(k)
        js = max(0, k_int - n + 1)
        je = min(k_int, n - 1)
        if js > je:
            continue
        # g[j] for j in [js, je] has coefficient f[k-j]
        # f[k-j] for j=js..je gives f[k-js], f[k-js-1], ..., f[k-je]
        # which is f[k-je : k-js+1][::-1] ... let me just index directly
        coeffs = np.array([f[k_int - j] for j in range(js, je + 1)])
        rows.extend([len(rows) // max(je - js + 1, 1)] * 0)  # can't do this simply
        # Let me just build it directly
        row_idx = len(rows)  # not right either
    # Actually let me just build the sparse matrix properly
    all_rows = []
    all_cols = []
    all_vals = []
    for ii, k in enumerate(top_k):
        k_int = int(k)
        js = max(0, k_int - n + 1)
        je = min(k_int, n - 1)
        if js > je:
            continue
        for j in range(js, je + 1):
            all_rows.append(ii)
            all_cols.append(j)
            all_vals.append(f[k_int - j])
    
    if not all_rows:
        return None
    
    A = csr_matrix((all_vals, (all_rows, all_cols)), shape=(len(top_k), n))
    
    c = -np.ones(n)
    try:
        res = scipy.optimize.linprog(
            c, A_ub=A, b_ub=np.full(len(top_k), M),
            bounds=[(0, None)] * n, method='highs',
            options={'presolve': True, 'time_limit': 10}
        )
        if res.success:
            return res.x
    except Exception:
        pass
    return None


def gradient_substep(a, conv, topk, temp_scale=0.1):
    """Compute subgradient of the max and return the gradient of the full objective."""
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S
    
    k_act = min(topk, len(conv))
    part = np.argpartition(conv, -k_act)[-k_act:]
    vals = conv[part].copy()
    order = np.argsort(vals)[::-1]
    part = part[order]
    vals = vals[order]
    
    top1 = float(vals[0])
    top2 = float(vals[1]) if k_act > 1 else 0.0
    ratio = top2 / top1 if top1 > 1e-15 else 1.0
    
    # Blend exact and soft-max
    exact_w = float(np.clip((0.9 - ratio) / 0.4, 0.0, 1.0))
    
    if exact_w > 0.99:
        wts = np.zeros(k_act)
        wts[0] = 1.0
        M = top1
    else:
        temp = max(temp_scale * top1, 1e-15)
        wts_smooth = np.exp((vals - top1) / temp)
        wts_smooth = wts_smooth / wts_smooth.sum()
        if exact_w < 0.01:
            wts = wts_smooth
        else:
            wts_exact = np.zeros(k_act)
            wts_exact[0] = 1.0
            wts = exact_w * wts_exact + (1.0 - exact_w) * wts_smooth
            wts = wts / wts.sum()
        M = float(np.dot(vals, wts))
    
    # Compute gradient of max(conv)
    grad_conv = np.zeros(n)
    for i in range(k_act):
        wi = float(wts[i])
        if wi <= 1e-15:
            continue
        ki = int(part[i])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        grad_conv[js:je + 1] += wi * 2.0 * a[js:je + 1][::-1]
    
    # Grad of score = (2n/S^2) * grad_conv - (4n*M/S^3) * ones
    grad = (2.0 * n) * (grad_conv / S2 - 2.0 * M / S3)
    return grad, M


def adam_update(a, g, m, v, t, lr):
    """Adam step."""
    m[:] = 0.9 * m + 0.1 * g
    v[:] = 0.999 * v + 0.001 * g * g
    t1 = 0.9 ** t
    t2 = 0.999 ** t
    a -= lr * (m / (1.0 - t1)) / (np.sqrt(v / (1.0 - t2)) + 1e-8)

    no_imp = 0

            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
```

```python
"""
LP-projection and gradient-based optimizer for minimizing 2n*max(conv(a,a))/(sum a)^2.
Core idea: alternate between LP projections (fix one factor, optimize the other for max
spread subject to convolution bound) and subgradient descent. Uses tapered initializations
and adaptive constraint selection. Multiple phases with different learning rates.
"""

import numpy as np
import time
import scipy.optimize


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score_from_conv(n, conv, S):
    """Objective: 2*n*max(conv)/(sum a)^2."""
    if S < 0.01:
        return np.inf
    return float(2.0 * n * np.max(conv) / (S * S))


def init_cosine_taper(n):
    """Raised-cosine (Hann-like) taper."""
    x = np.arange(n)
    return np.sin(np.pi * (x + 0.5) / n) ** 2


def init_gaussian_taper(n, sigma_frac=0.4):
    """Gaussian taper."""
    x = np.arange(n) - (n - 1) / 2.0
    sigma = sigma_frac * n
    return np.exp(-0.5 * (x / sigma) ** 2)


def init_super_gaussian(n, p=4.0):
    """Super-Gaussian taper (flatter top than Gaussian)."""
    x = (2.0 * np.arange(n) / max(n - 1, 1) - 1.0)
    return np.exp(-np.abs(x) ** p)


def init_parabolic(n):
    """Parabolic (triangular) taper."""
    x = np.arange(n)
    mid = (n - 1) / 2.0
    return np.maximum(0.0, 1.0 - (x - mid) / (mid + 0.5))


def init_ones(n):
    """Constant sequence."""
    return np.ones(n)


def build_lp_matrix(f, peak_idx, n):
    """Build dense constraint matrix for LP. A[ii, j] = f[k-j] if valid."""
    top_k = len(peak_idx)
    A = np.zeros((top_k, n))
    for ii in range(top_k):
        k = int(peak_idx[ii])
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if js <= je:
            lo = k - je
            hi = k - js
            A[ii, js:je + 1] = f[lo:hi + 1][::-1]
    return A


def solve_lp(f, M, peak_idx, n, time_limit=5):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for k in peak_idx, g >= 0.
    Returns g array or None.
    """
    top_k = len(peak_idx)
    A = build_lp_matrix(f, peak_idx, n)
    c = -np.ones(n)
    b = np.full(top_k, M)
    try:
        res = scipy.optimize.linprog(
            c, A_ub=A, b_ub=b,
            bounds=[(0, None)] * n,
            method='highs',
            options={'presolve': True, 'time_limit': time_limit}
        )
        if res.status == 0:
            return res.x
        elif res.status == 1:
            # Infeasible - shouldn't happen since f is feasible
            return None
        else:
            return None
    except Exception:
        return None


def gradient_substep(a, conv, topk, temp_scale=0.1):
    """Compute subgradient of the full objective."""
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S

    k_act = min(topk, len(conv))
    part = np.argpartition(conv, -k_act)[-k_act:]
    vals = conv[part].copy()
    order = np.argsort(vals)[::-1]
    part = part[order]
    vals = vals[order]

    top1 = float(vals[0])
    top2 = float(vals[1]) if k_act > 1 else 0.0
    ratio = top2 / top1 if top1 > 1e-15 else 1.0

    exact_w = float(np.clip((0.9 - ratio) / 0.4, 0.0, 1.0))

    if exact_w > 0.99:
        wts = np.zeros(k_act)
        wts[0] = 1.0
        M = top1
    else:
        temp = max(temp_scale * top1, 1e-15)
        wts_smooth = np.exp((vals - top1) / temp)
        wts_smooth = wts_smooth / wts_smooth.sum()
        if exact_w < 0.01:
            wts = wts_smooth
        else:
            wts_exact = np.zeros(k_act)
            wts_exact[0] = 1.0
            wts = exact_w * wts_exact + (1.0 - exact_w) * wts_smooth
            wts = wts / wts.sum()
        M = float(np.dot(vals, wts))

    grad_conv = np.zeros(n)
    for i in range(k_act):
        wi = float(wts[i])
        if wi <= 1e-15:
            continue
        ki = int(part[i])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        grad_conv[js:je + 1] += wi * 2.0 * a[js:je + 1][::-1]

    grad = (2.0 * n) * (grad_conv / S2 - 2.0 * M / S3)
    return grad, M


def adam_step(a, g, m, v, t, lr):
    """In-place Adam step."""
    m[:] = 0.9 * m + 0.1 * g
    v[:] = 0.999 * v + 0.001 * g * g
    t1 = 0.9 ** t
    t2 = 0.999 ** t
    a -= lr * (m / (1.0 - t1)) / (np.sqrt(v / (1.0 - t2)) + 1e-8)


def optimize_lp_phase(a0, end_time, rng, top_k_start=30, top_k_end=300, blend=0.12):
    """
    LP-based optimization: repeatedly project onto the set {g >= 0 : (f*g) <= M}.
    """
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    conv = fast_self_conv(a)
    S = float(a.sum())
    best_score = compute_score_from_conv(n, conv, S)
    best_a = a.copy()
    no_imp = 0
    it = 0

    total_time = max(end_time - time.monotonic(), 1.0)
    start_time = time.monotonic()

    while time.monotonic() < end_time and it < 5000:
        conv = fast_self_conv(a)
        S = float(a.sum())
        if S < 1e-6:
            a = a0.copy()
            break
        score = compute_score_from_conv(n, conv, S)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        M = float(np.max(conv))

        # Adaptive top_k: grow from top_k_start to top_k_end
        frac = min(1.0, it / 100.0)
        cur_top_k = int(top_k_start + frac * (top_k_end - top_k_start))
        cur_top_k = min(cur_top_k, len(conv))

        peak_idx = np.argpartition(conv, -cur_top_k)[-cur_top_k:]

        g = solve_lp(a, M, peak_idx, n, time_limit=3)

        if g is not None and g.sum() > S * 0.5:
            # Blend: a_new = (1-blend)*a + blend*g
            a_new = (1.0 - blend) * a + blend * g
            a_new = np.clip(a_new, 0.0, 1000.0)
            if a_new.sum() > 1e-6:
                a = a_new
            else:
                a = a0.copy()
        else:
            # Fallback: small gradient step
            grad, _ = gradient_substep(a, conv, min(8, n), 0.1)
            a -= 0.005 * grad
            a = np.clip(a, 0.0, 1000.0)

        it += 1

        if no_imp > 80:
            # Perturb and restart
            sc = 0.02 + 0.04 * rng.random()
            a = best_a * (1.0 + sc * rng.standard_normal(n))
            a = np.clip(a, 0.0, 1000.0)
            if a.sum() < 1e-3:
                a += 0.01
            no_imp = 0

    return best_score, best_a


def optimize_gradient_phase(a0, end_time, rng, init_lr=0.03, topk=12, patience=200):
    """Gradient-based refinement with Adam."""
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    conv = fast_self_conv(a)
    S = float(a.sum())
    best_score = compute_score_from_conv(n, conv, S)
    best_a = a.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = init_lr
    no_imp = 0
    cur_pat = patience

    while time.monotonic() < end_time:
        conv = fast_self_conv(a)
        S = float(a.sum())
        if S < 1e-6:
            a = a0.copy()
            S = float(a.sum())
        score = compute_score_from_conv(n, conv, S)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp >= cur_pat:
            sc = 0.02 + 0.05 * rng.random()
            a = best_a * (1.0 + sc * rng.standard_normal(n))
            a = np.clip(a, 0.0, 1000.0)
            if a.sum() < 1e-3:
                a += 0.05
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            lr = max(0.0002, lr * 0.5)
            no_imp = 0
            cur_pat = max(30, cur_pat // 2)
            continue

        g, M = gradient_substep(a, conv, topk, 0.08)
        t += 1
        adam_step(a, g, m, v, t, lr)
        a = np.clip(a, 0.0, 1000.0)

    return best_score, best_a


def lp_then_grad(a0, end_time, rng, lp_frac=0.6):
    """Combined LP then gradient optimization."""
    lp_end = time.monotonic() + lp_frac * (end_time - time.monotonic())
    sc1, aa1 = optimize_lp_phase(a0, min(lp_end, end_time), rng,
                                  top_k_start=20, top_k_end=200, blend=0.15)
    if time.monotonic() < end_time:
        sc2, aa2 = optimize_gradient_phase(aa1, end_time, rng,
                                            init_lr=0.02, topk=16, patience=150)
        if sc2 < sc1:
            return sc2, aa2
    return sc1, aa1

        pass
    n = min(max(1024, n), 16384)

    print(f"n={n}, budget={budget_s:.0f}s, parent={'yes' if parent is not None else 'no'}")

    # Build list of initializations (no lambdas - use explicit callables)
    # We'll just call them in order
    inits = []
    if parent is not None and parent.size == n:
        inits.append(np.clip(parent, 0, 1000))
    inits.append(init_cosine_taper(n))
    inits.append(init_gaussian_taper(n, 0.35))
    inits.append(init_gaussian_taper(n, 0.45))
    inits.append(init_super_gaussian(n, 3.0))
    inits.append(init_super_gaussian(n, 5.0))
    inits.append(init_parabolic(n))
    inits.append(init_ones(n))
    # Add a random tapered init
    x = np.arange(n)
    taper = np.sin(np.pi * (x + 0.5) / n)
    inits.append(taper * (0.5 + 0.5 * rng.random(n)))

    # Phase 1: Quick scan of inits with short LP (20% time)
    p1_end = start + 0.20 * (total - 15.0)
    per_time = (p1_end - start) / len(inits)
    names = ['parent', 'cosine', 'gauss35', 'gauss45', 'super3', 'super5', 'parab', 'ones', 'rand_taper']

    for idx in range(len(inits)):
        t_end = min(p1_end, start + per_time * (idx + 1))
        if time.monotonic() >= p1_end:
            break
        a0 = inits[idx]
        sc, aa = optimize_lp_phase(a0, t_end, rng, top_k_start=20, top_k_end=100, blend=0.2)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        nm = names[idx] if idx < len(names) else f'init{idx}'
        print(f"Phase 1 [{nm}]: {sc:.6f}")

    # Phase 2: Extended LP from best (30% time)
    p2_end = start + 0.50 * (total - 15.0)
    if best_a is not None and time.monotonic() < p2_end:
        sc, aa = optimize_lp_phase(best_a, p2_end, rng,
                                    top_k_start=50, top_k_end=400, blend=0.10)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 2 (LP): {best_score:.6f}")

    # Phase 3: Gradient refinement (20% time)
    p3_end = start + 0.70 * (total - 15.0)
    if best_a is not None and time.monotonic() < p3_end:
        sc, aa = optimize_gradient_phase(best_a, p3_end, rng,
                                          init_lr=0.015, topk=16, patience=300)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 3 (grad): {best_score:.6f}")

    # Phase 4: LP again from gradient result (15% time)
    p4_end = start + 0.85 * (total - 15.0)
    if best_a is not None and time.monotonic() < p4_end:
        sc, aa = optimize_lp_phase(best_a, p4_end, rng,
                                    top_k_start=100, top_k_end=500, blend=0.08)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 4 (LP2): {best_score:.6f}")

    # Phase 5: Fine gradient polish (remaining time)
    if best_a is not None and time.monotonic() < deadline - 3.0:
        sc, aa = optimize_gradient_phase(best_a, deadline - 3.0, rng,
                                          init_lr=0.003, topk=32, patience=500)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 5 (fine grad): {best_score:.6f}")

    # Final output
    if best_a is None:
        best_a = init_cosine_taper(n)

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    verified = np.inf
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        pass
    print(f"FINAL: internal={best_score:.6f}, verified={verified:.6f}, n={n}, "
          f"time={time.monotonic()-start:.1f}s")
    return out
```
