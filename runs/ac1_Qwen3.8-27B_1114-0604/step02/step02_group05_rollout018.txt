"""
Spectral error-reduction + adaptive gradient hybrid for minimizing
2*n*max(conv(a,a)) / sum(a)^2 with a >= 0.
Novel: Fienup-style alternating projection with correct target power spectrum,
combined with variance-aware multiplicative gradient descent.

import numpy as np
import time

def _npow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv(a, b):
    nout = len(a) + len(b) - 1
    L = _npow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)

def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    n = len(p)
    c = _conv(p, p)
    return float(2.0 * n * c.max() / (s * s))

def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s

def _target_spectrum(n, L):
    """Compute the target power spectrum for flat autocorrelation."""
    c_target = np.zeros(L)
    c_target[:2*n-1] = 1.0 / (2*n - 1)
    C_target = np.fft.rfft(c_target, L)
    # C_target should be real; take real part and clip to 0
    C_target = np.real(C_target)
    C_target = np.maximum(C_target, 0.0)
    return np.sqrt(C_target)

def _fienup(n, rng, n_iter=200, blend=0.2, init=None):
    """
    Fienup error-reduction with correct target power spectrum.
    Alternates between:
      - Frequency domain: project magnitudes towards target
      - Time domain: enforce non-negativity, sum=1, support=n
    """
    L = _npow2(2 * n - 1)
    target_mag = _target_spectrum(n, L)
    
    if init is not None:
        a = init.copy()
    else:
        a = np.maximum(rng.normal(0, 1, n), 0).astype(float)
        s = a.sum()
        if s < 1e-10:
            a = np.ones(n) / n
        else:
            a = a / s
    
    best = a.copy()
    best_sc = _score(a)
    
    for it in range(n_iter):
        A = np.fft.rfft(a, L)
        mags = np.abs(A)
        phases = np.angle(A)
        
        # Blend magnitudes towards target
        new_mags = (1.0 - blend) * mags + blend * target_mag
        A_new = new_mags * np.exp(1j * phases)
        a_new = np.fft.irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            break
        a_new = a_new / s
        
        sc = _score(a_new)
        if sc < best_sc:
            best = a_new.copy()
            best_sc = sc
        a = a_new
    
    return best, best_sc

            # Use a combination of max-focused and variance-focused pressure

            d_std = d.std()

                    w_soft = np.ones(len(d)) / len(d)
                else:
                    shift = d.max()
                    e = np.exp(beta * (d - shift))
                    se = e.sum()
                    w_soft = e / se if se > 1e-300 else np.ones(len(d)) / len(d)
                # Blend: 70% soft max, 30% z-score based

                w_z = w_z / w_z.sum() if w_z.sum() > 1e-300 else np.ones(len(d)) / len(d)

        else:

                e = np.exp(beta * (d - shift))

                w = e / se if se > 1e-300 else np.ones(len(d)) / len(d)

        h = 2.0 * R[:n]
