def _normalize(v):

    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:

    s = float(v.sum())
    if s <= 1e-300:

def _fft_size(m):

    if m <= 1:

    return 1 << (int(m) - 1).bit_length()

def _convolve(a, b):

    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    return np.fft.irfft(A * B, size)[:L]

    u = np.sort(v)[::-1]

    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)

    theta = (cssv[rho] - 1.0) / float(rho + 1.0)

    s = float(out.sum())
    if s <= 1e-300:

    out /= s

def _normalize(v):
    """Project to non-negative float vector of sum 1."""
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v


def _fft_size(m):
    """Smallest power of two at least m."""
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def _convolve_fft(a, b):
    """Full linear convolution via FFT."""
    n, m = a.size, b.size
    L = n + m - 1
    size = _fft_size(L)
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    return np.fft.irfft(A * B, size)[:L]


def _autocorr(a):
    """Full convolution a*a."""
    return _convolve_fft(a, a)


def _fast_score(a):
    """Score for a normalized non-negative vector."""
    a = _normalize(a)
    n = a.size
    c = _autocorr(a)
    return float(2.0 * n * c.max())


def _project_simplex(v):
    """Euclidean projection onto x >= 0, sum x = 1."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1.0)
    out = v - theta
    np.maximum(out, 0.0, out)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    out /= s
    return out


def _shift_sum(a, w):
    """
    Compute h[j] = sum_i a[i] * w[i+j] for j=0,...,n-1.
    a has length n, w has length 2n-1 (the autocorrelation lags).

    A = np.fft.rfft(a, fft_size)
    h_full = np.fft.irfft(W * np.conj(A), fft_size)
    return h_full[:n]


def _softmax_weights(c, T):
    """Softmax weights over lags, emphasizing peaks."""
    c_shifted = c - c.max()
    np.clip(c_shifted, -50.0 * T, 0.0, out=c_shifted)
    w = np.exp(c_shifted / T)
    s = w.sum()
    if s < 1e-300:
        w = np.ones(c.size) / c.size
    else:
        w /= s
    return w


def _hard_active_set(c, K):
    """Return indicator vector for top-K lags."""
    n = c.size
    K = min(K, n)
    if K >= n:
        return np.ones(n, dtype=np.float64)
    idx = np.argpartition(c, n - K)[n - K:]
    ind = np.zeros(n, dtype=np.float64)
    ind[idx] = 1.0 / K
    return ind


def _shift_balance_update(a, c, T, gamma, active_K, use_soft):
    """
    One shift-balancing iteration.
    
    a: current sequence (normalized, length n)
    c: autocorrelation (length 2n-1)
