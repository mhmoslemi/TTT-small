import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score_fast(a):
    """Fast score computation (scale invariant)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    return 2.0 * a.size * float(np.max(c)) / (s * s)


def _compute_gradient(w, a):
    """
    Compute gradient of sum_k w[k]*c_k with respect to a.
    = 2 * (cross-correlation of w and a)
    """
    n = a.size
    N = _next_pow2(max(2 * n - 1, w.size))
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N) if w.size <= N else np.fft.rfft(w[:N], N)
    # irfft(conj(fa) * fw) gives sum_i a[i] * w[i+j]
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]


def _make_sine(n, amp, mode=1, phase=0.0):
    """Sinusoidal initialization."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64) / n
    theta = 2.0 * np.pi * mode * (i + phase) / n
    a = (1.0 + amp * np.sin(theta)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / n
    return a / s


def _make_gaussian(n, sigma_frac):
    """Gaussian-shaped initialization."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / n
    return a / s

    a = rng.standard_normal(n)
    # Apply simple moving average for smoothing
    kernel = np.ones(smooth) / smooth
    a = np.convolve(a, kernel, mode='same')
    a = np.abs(a)  # make non-negative
    s = a.sum()
    if s < 0.01:

    return a / s

def _hard_optimize(a, deadline, K=8):
    """
    Hard-max top-K active set optimization.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = _score_fast(best_a)

    it = 0
    last_improve = 0
    patience = 2000
    rng = np.random.RandomState(777 + n)
    lr = 0.2 * n * 0.01

    while time.monotonic() < deadline:
        it += 1

        c = _fft_convolve(a)
        if c.size == 0:
            break

        # Top-K peaks with soft weighting within
        if c.size <= K:
            top_idx = np.arange(c.size)
        else:
            top_idx = np.argpartition(c, -K)[-K:]

        top_vals = c[top_idx]
        T_local = 2.0
        top_shift = top_vals - top_vals.max()
        weights = np.exp(T_local * top_shift)
        weights /= weights.sum()

        w = np.zeros(c.size, dtype=np.float64)
        for i, k in enumerate(top_idx):
            w[k] = weights[i]

        grad = _compute_gradient(w, a)
        grad_proj = grad - np.mean(grad)

        update = -lr * grad_proj
        update = np.clip(update, -15.0, 15.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0, 1000)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        if it % 5 == 0:
            sc = _score_fast(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
                lr = min(lr * 1.003, 0.1 * n * 0.05)
            elif it - last_improve > patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.02 * a.mean()
                a = np.clip(a + noise, 0, 1000)
                a = a / a.sum()
                patience = max(1000, int(patience * 1.3))
                lr = max(lr * 0.8, 0.01 * n * 0.005)

    return best_a, best_score


def _coordinate_descent(a, deadline, maxpasses=20):
    """
    Exact coordinate descent: for each coordinate, find the optimal value
    that minimizes the max of the self-convolution.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = _score_fast(best_a)
    print(f"    CD start: score = {best_score:.8f}", flush=True)

    c = _fft_convolve(a)  # Current convolution
    c_max_global = np.max(c)

    for pdc in range(maxpasses):
        if time.monotonic() > deadline:
            break
        improved = False

        # Randomize order for diversity
        order = np.random.permutation(n)

        for j in order:
            if time.monotonic() > deadline:
                    break

            aj = a[j]
            if aj < 1e-15:
                continue

            # Compute affected range: k in [j, j+n-1]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)
            affected = np.arange(k_start, k_end + 1)

            # Current c values for affected positions
            c_aff = c[affected].copy()

            # c_k^{(-j)} = c_k - contribution of a_j
            # For k != 2j: c_k = c_k^{(-j)} + 2*a_j*a_{k-j}
            # For k = 2j: c_k = c_k^{(-j)} + a_j^2
            a_kj = a[affected - j]  # a[k-j] for k in affected range
            c_without = c_aff.copy()
            mask_quad = (affected == 2 * j)
            c_without[mask_quad] -= aj * aj
            c_without[~mask_quad] -= 2.0 * aj * a_kj[~mask_quad]

            # The max of non-affected values (constant)
            if k_start > 0 or k_end < 2 * n - 2:
                # Create mask for non-affected
                all_mask = np.ones(2 * n - 1, dtype=bool)
                all_mask[k_start:k_end + 1] = False
                C_const = np.max(c[all_mask]) if all_mask.any() else 0.0
            else:
                C_const = 0.0

            # Now find t >= 0 minimizing max(c_without + 2t*a_kj (linear), c_2j_without + t^2 (quad), C_const)
            # Use golden section search
            alpha = (np.sqrt(5) - 1) / 2  # golden ratio conjugate

            # Initial bracket: [0, 2*aj]
            t_lo = 0.0
            t_hi = max(2.0 * aj, 1e-10)

            def f_of_t(t):
                """Compute max convolution value at a_j = t."""
                vals = c_without + 2.0 * t * a_kj
                if mask_quad.any():
                    idx = np.where(mask_quad)[0][0]
                    vals[idx] = c_without[idx] + t * t
                return max(float(np.max(vals)), C_const)

            # Golden section search
            t1 = t_hi - alpha * (t_hi - t_lo)
            t2 = t_hi - (1 - alpha) * (t_hi - t_lo)
            f1 = f_of_t(t1)
            f2 = f_of_t(t2)

            for _ in range(40):
                if f1 < f2:
                    t_hi = t2
                    t2 = t1
                    f2 = f1
                    t1 = t_hi - alpha * (t_hi - t_lo)
                    f1 = f_of_t(t1)
                else:
                    t_lo = t1
                    t1 = t2
                    f1 = f2
                    t2 = t_hi - (1 - alpha) * (t_hi - t_lo)
                    f2 = f_of_t(t2)

            t_opt = (t_lo + t_hi) / 2.0
            f_opt = f_of_t(t_opt)

            # Also check t=0
            f_zero = f_of_t(0.0)
            if f_zero < f_opt:
                t_opt = 0.0
                f_opt = f_zero

            # Update if improved
            new_max = f_opt
            if new_max < c_max_global - 1e-15:
                # Update a[j] and c
                a[j] = t_opt
                # Update convolution values in affected range
                new_c_aff = c_without + 2.0 * t_opt * a_kj
                if mask_quad.any():
                    idx = np.where(mask_quad)[0][0]
                    new_c_aff[idx] = c_without[idx] + t_opt * t_opt
                c[k_start:k_end + 1] = new_c_aff
                c_max_global = max(float(np.max(new_c_aff)), C_const)
                improved = True

        # Check overall improvement
        sc = _score_fast(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"    CD pass {pdc}: score = {best_score:.8f}", flush=True)
        if not improved:
            break

    return best_a, best_score

2. Backtracking-style adaptive step size in mirror descent

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score_fast(a):
    """Fast scale-invariant score."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    return 2.0 * a.size * float(np.max(c)) / (s * s)


def _compute_gradient(w, a):
    """
    Compute grad[j] = 2 * sum_k w[k] * a[k-j] (cross-correlation).
    This is the gradient of sum_k w[k] * c_k w.r.t. a[j].
    """
    n = a.size
    N = _next_pow2(max(2 * n - 1, w.size))
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]


def _make_sine(n, amp, mode=1, phase=0.0):
    """Sinusoidal initialization."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * mode * (i + phase) / n
    a = (1.0 + amp * np.sin(theta)) / n
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / n
    return a / s


def _make_gaussian(n, sigma_frac):
    """Gaussian-shaped initialization."""
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a = np.clip(a, 1e-10, None)
    s = a.sum()
    return a / s


def _make_multi_sine(n, specs, rng):
    """Multi-harmonic initialization."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64) / n
    for mode, amp, ph in specs:
        theta = 2.0 * np.pi * mode * (i + ph) / n
        a += amp * np.sin(theta) / n
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / n
    return a / s


def _cd_eval(c_without, a_kj, quad_idx, C_const, t):
    """Evaluate max convolution with a_j = t (for coordinate descent)."""
    vals = c_without + 2.0 * t * a_kj
    if quad_idx >= 0:
        vals[quad_idx] = c_without[quad_idx] + t * t
    result = float(np.max(vals))
    if C_const > result:
        result = C_const
    return result


def _soft_optimize(a, deadline, T_start=3.0, T_end=80.0, lr0=None):
    """
    Adaptive mirror descent with softmax gradient and backtracking.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    if lr0 is None:
        lr0 = 0.4

    best_a = a.copy()
    best_score = _score_fast(best_a)

    it = 0
    last_improve = 0
    patience = 4000
    rng = np.random.RandomState(1000 + n)
    T = T_start
    c = _fft_convolve(a)
    current_score = best_score

    while time.monotonic() < deadline:
        it += 1

        # Temperature: geometric cooling then heating (explore-exploit)
        frac = min(1.0, it / 50000.0)
        if frac < 0.6:
            # Cooling phase
            T = T_start * (T_end / T_start) ** (frac / 0.6)
        else:
            # Slight reheating to escape local minima
            T2 = (frac - 0.6) / 0.4
            T = T_end * (1.0 + 0.5 * T2)

        # Compute softmax weights
        c_max = float(np.max(c))
        c_min = float(np.min(c))
        rng_c = c_max - c_min
        if rng_c < 1e-15:
            w = np.ones(c.size) / c.size
        else:
            c_shift = (c - c_max) / rng_c
            exp_c = np.exp(T * c_shift)
            w = exp_c / exp_c.sum()

        # Gradient
        grad = _compute_gradient(w, a)
        grad_mean = float(np.mean(grad))
        grad_proj = grad - grad_mean

        # Backtracking mirror descent
        step_found = False
        for bt in range(8):
            trial_lr = lr0 * (0.5 ** bt)
            update = -trial_lr * grad_proj
            update = np.clip(update, -30.0, 30.0)
            a_cand = a * np.exp(update)
            a_cand = np.where(np.isfinite(a_cand), a_cand, 0.0)
            a_cand = np.clip(a_cand, 0.0, 1000.0)
            s_cand = float(a_cand.sum())
            if s_cand > 1e-10:
                a_cand = a_cand / s_cand
                sc = _score_fast(a_cand)
                if sc < best_score:
                    best_score = sc
                    best_a = a_cand.copy()
                    last_improve = it
                    a = a_cand
                    current_score = sc
                    step_found = True
                    break
                elif bt < 3 and sc < current_score + 0.005:
                    # Accept small degradation for exploration
                    a = a_cand
                    current_score = sc
                    step_found = True
                    break

        if not step_found:
            # No improvement found; adapt
            lr0 = max(lr0 * 0.95, 0.01)

        # Update convolution every few iterations
        if it % 4 == 0:
            c = _fft_convolve(a)

        # Patience-based restart
        if it - last_improve > patience:
            a = best_a.copy()
            # Diversified restart: perturb a random subset
            n_perturb = max(10, n // 10)
            idx = rng.choice(n, size=n_perturb, replace=False)
            a[idx] *= (1.0 + 0.3 * rng.standard_normal(n_perturb))
            a = np.clip(a, 0, 1000)
            s = a.sum()
            if s < 0.01:
                a = best_a.copy()
                s = 1.0
            a = a / s
            current_score = _score_fast(a)
            patience = max(3000, int(patience * 1.4))
            T = T_start

        # Adaptive lr: increase when improving
        if it % 200 == 0 and (it - last_improve) < patience // 3:
            lr0 = min(lr0 * 1.02, 1.0)

    return best_a, best_score


def _hard_optimize(a, deadline, K=6):
    """
    Hard top-K active set mirror descent for final refinement.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = _score_fast(best_a)

    it = 0
    last_improve = 0
    patience = 2000
    rng = np.random.RandomState(55555 + n)
    lr = 0.15

    while time.monotonic() < deadline:
        it += 1

        c = _fft_convolve(a)
        if c.size == 0:
            break

        # Top-K with soft weighting
        if c.size <= K:
            top_idx = np.arange(c.size)
        else:
            top_idx = np.argpartition(c, -K)[-K:]

        top_vals = c[top_idx]
        T_loc = 1.5
        top_shift = top_vals - top_vals.max()
        weights = np.exp(T_loc * top_shift)
        weights /= weights.sum()

        w = np.zeros(c.size, dtype=np.float64)
        for i in range(len(top_idx)):
            w[top_idx[i]] = weights[i]

        grad = _compute_gradient(w, a)
        grad_proj = grad - float(np.mean(grad))

        # Backtracking
        step_ok = False
        for bt in range(6):
            trial_lr = lr * (0.5 ** bt)
            update = -trial_lr * grad_proj
            update = np.clip(update, -20.0, 20.0)
            a_cand = a * np.exp(update)
            a_cand = np.where(np.isfinite(a_cand), a_cand, 0.0)
            a_cand = np.clip(a_cand, 0, 1000)
            s_cand = a_cand.sum()
            if s_cand > 1e-10:
                a_cand = a_cand / s_cand
                sc = _score_fast(a_cand)
                if sc < best_score:
                    best_score = sc
                    best_a = a_cand.copy()
                    last_improve = it
                    a = a_cand
                    step_ok = True
                    break

        if not step_ok:
            lr = max(lr * 0.9, 0.005)

        if it - last_improve > patience:
            a = best_a.copy()
            noise = rng.standard_normal(n) * 0.03 * a.mean()
            a = np.clip(a + noise, 0, 1000)
            a = a / a.sum()
            patience = max(1000, int(patience * 1.3))
            lr = max(lr * 0.7, 0.003)

        if (it - last_improve) < patience // 2 and it % 100 == 0:
            lr = min(lr * 1.01, 0.5)

    return best_a, best_score


def _coordinate_descent(a, deadline, maxpasses=30):
    """
    Exact coordinate descent: minimize max self-convolution per coordinate.
    Works on unnormalized sequence (score is scale-invariant).
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
        s = 1.0

    best_a = a.copy()
    best_score = _score_fast(best_a)
    print(f"    CD start: n={n}, score={best_score:.8f}", flush=True)

    c = _fft_convolve(a)
    alpha_gold = (np.sqrt(5.0) - 1.0) / 2.0

    for pdc in range(maxpasses):
        if time.monotonic() > deadline:
            break
        any_improve = False
        order = np.random.permutation(n)

        for j in order:
            if time.monotonic() > deadline:
                break

            aj = a[j]
            if aj < 1e-15:
                continue

            # Affected range: k in [j, j+n-1]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)
            if k_start > k_end:
                continue

            affected_len = k_end - k_start + 1
            k_indices = np.arange(k_start, k_end + 1)
            a_kj = a[k_indices - j]  # a[k-j] for each k in range
            c_aff = c[k_indices].copy()

            # Compute c without contribution of a_j
            c_without = c_aff.copy()
            # For k = 2j (if in range): contribution is a_j^2
            quad_idx_in_range = -1
            if 2 * j >= k_start and 2 * j <= k_end:
                quad_idx_in_range = 2 * j - k_start

            if quad_idx_in_range >= 0:
                c_without[quad_idx_in_range] -= aj * aj
                # For other indices: linear contribution 2*aj*a[k-j]
                other_mask = np.ones(affected_len, dtype=bool)
                other_mask[quad_idx_in_range] = False
                c_without[other_mask] -= 2.0 * aj * a_kj[other_mask]
            else:
                c_without -= 2.0 * aj * a_kj

            # Constant: max of non-affected convolution values
            if k_start > 0 or k_end < 2 * n - 2:
                # Build mask
                C_const = 0.0
                if k_start > 0:
                    C_const = max(C_const, float(np.max(c[:k_start])))
                if k_end < 2 * n - 2:
                    C_const = max(C_const, float(np.max(c[k_end + 1:])))
            else:
                C_const = 0.0

            # Golden section search for optimal t in [0, t_hi]
            t_hi = max(3.0 * aj, 1e-8)
            # Check if going to 0 is better
            f_zero = _cd_eval(c_without, a_kj, quad_idx_in_range, C_const, 0.0)
            f_current = _cd_eval(c_without, a_kj, quad_idx_in_range, C_const, aj)

            t_lo = 0.0
            # If current is already at boundary, skip
            if f_zero >= f_current and aj >= t_hi:
                continue

            # Golden section search
            if f_zero < f_current:
                # Optimum is likely below current
                t_hi = aj
            else:
                t_lo = aj
                t_hi = max(3.0 * aj, aj + 0.1)

            # Refine bracket
            t1 = t_hi - alpha_gold * (t_hi - t_lo)
            t2 = t_hi - (1.0 - alpha_gold) * (t_hi - t_lo)
            f1 = _cd_eval(c_without, a_kj, quad_idx_in_range, C_const, t1)
            f2 = _cd_eval(c_without, a_kj, quad_idx_in_range, C_const, t2)

            for _ in range(30):
                if f1 < f2:
                    t_hi = t2
                    t2 = t1
                    f2 = f1
                    t1 = t_hi - alpha_gold * (t_hi - t_lo)
                    f1 = _cd_eval(c_without, a_kj, quad_idx_in_range, C_const, t1)
                else:
                    t_lo = t1
                    t1 = t2
                    f1 = f2
                    t2 = t_hi - (1.0 - alpha_gold) * (t_hi - t_lo)
                    f2 = _cd_eval(c_without, a_kj, quad_idx_in_range, C_const, t2)

            t_opt = (t_lo + t_hi) / 2.0
            f_opt = _cd_eval(c_without, a_kj, quad_idx_in_range, C_const, t_opt)

            # Compare with t=0
            if f_zero < f_opt:
                t_opt = 0.0
                f_opt = f_zero

            # Apply update if improved
            c_max_old = max(float(np.max(c)), C_const)
            if f_opt < c_max_old - 1e-14:
                a[j] = t_opt
                # Update convolution
                new_c_aff = c_without + 2.0 * t_opt * a_kj
                if quad_idx_in_range >= 0:
                    new_c_aff[quad_idx_in_range] = c_without[quad_idx_in_range] + t_opt * t_opt
                c[k_start:k_end + 1] = new_c_aff
                any_improve = True

        # Periodic score check
        if pdc % 3 == 0 or not any_improve:
            sc = _score_fast(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                if pdc % 5 == 0:
                    print(f"    CD pass {pdc}: score={best_score:.8f}", flush=True)

        if not any_improve:
            # Try once more with slightly different randomization
            a_noise = a * (1.0 + 0.01 * np.random.standard_normal(n))
            a_noise = np.clip(a_noise, 0, 1000)
            s = a_noise.sum()
            if s > 0.01:
                a = a_noise / s
                c = _fft_convolve(a)
            else:
                break

    # Final normalize and check
    s = a.sum()
    if s > 0.01:
        a_final = a / s
        sc = _score_fast(a_final)
        if sc < best_score:
            best_score = sc
            best_a = a_final.copy()

    return best_a, best_score

    print(f"=== Start | Budget: {budget_s:.0f}s | n_target: 4096 ===", flush=True)

    # Phase 1: Initialization
    d1 = t0 + 50.0
    print("[Phase 1] Scanning initializations...", flush=True)
    init_a, init_score = _scan_init(d1)
    print(f"  Best init: n={init_a.size}, score={init_score:.8f}", flush=True)

    # Pick the best n for main optimization
    main_n = init_a.size
    main_a = init_a.copy()

    # Phase 2: Soft mirror descent (main optimization)
    remaining_for_rest = hard_deadline - time.monotonic()
    frac2 = 0.55
    d2 = time.monotonic() + remaining_for_rest * frac2
    print(f"[Phase 2] Soft mirror descent ({d2 - time.monotonic():.0f}s) n={main_n}...", flush=True)
    a2, s2 = _soft_optimize(main_a, d2, T_start=2.0, T_end=100.0, lr0=0.5)
    print(f"  Soft best: n={a2.size}, score={s2:.8f}", flush=True)

    # Phase 3: Hard top-K refinement
    remaining3 = hard_deadline - time.monotonic()
    frac3 = 0.30
    d3 = time.monotonic() + remaining3 * frac3
    if d3 - time.monotonic() > 60:
        print(f"[Phase 3] Hard top-K ({d3 - time.monotonic():.0f}s) K=6...", flush=True)
        a3, s3 = _hard_optimize(a2, d3, K=6)
        print(f"  Hard best: n={a3.size}, score={s3:.8f}", flush=True)
    else:
        a3, s3 = a2, s2
        print("[Phase 3] Skipped (insufficient time)", flush=True)

    # Phase 4: Coordinate descent polish
    remaining4 = hard_deadline - time.monotonic()
    d4 = time.monotonic() + remaining4 * 0.7
    if d4 - time.monotonic() > 60:
        print(f"[Phase 4] Coordinate desert ({d4 - time.monotonic():.0f}s)...", flush=True)
        a4, s4 = _coordinate_descent(a3, d4, maxpasses=40)
        print(f"  CD best: n={a4.size}, score={s4:.8f}", flush=True)
    else:
        a4, s4 = a3, s3
        print("[Phase 4] Skipped (insufficient time)", flush=True)

    # Phase 5: Validate
    print("[Phase 5] Validating with evaluate_sequence...", flush=True)
    candidates = [(a4, s4), (a3, s3), (a2, s2), (init_a, init_score)]
    # Sort by estimated score
    candidates.sort(key=lambda x: x[1])

    best_out = None
    best_actual = np.inf

    for arr, est_sc in candidates:
        if time.monotonic() > hard_deadline:
            break
        arr_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        try:
            actual = float(evaluate_sequence(arr_list))
        except Exception:
            actual = np.inf
        if np.isfinite(actual) and actual < best_actual:
            best_actual = actual
            best_out = arr_list
            print(f"  Validated: n={arr.size}, est={est_sc:.8f}, actual={actual:.8f}", flush=True)

    if best_out is None:
        best_out = [float(max(0.0, min(1000.0, x))) for x in init_a]
        try:
            best_actual = float(evaluate_sequence(best_out))
        except Exception:
            best_actual = init_score
        print(f"  Fallback score: {best_actual:.8f}", flush=True)

    total_time = time.monotonic() - t0
    print(f"=== Done | Time: {total_time:.0f}s | Best score: {best_actual:.8f} ===", flush=True)
    return best_out
