But we need to be smart about which pairs to try. Let me think about a good strategy:

2. The peak b_{k*} = sum_i a_i * a_{k*-i}
3. The "critical pairs" are (i, k*-i) for i where a_i * a_{k*-i} is large
4. Try to reduce the peak by:

Actually, the most effective move is:
- Take mass from a "hot" coordinate (one that appears in many peak-contributing products)
- Put it in a "cold" coordinate (one that appears in few peak-contributing products)

Let me define:
- For coordinate i, its "heat" is H_i = sum over all active peaks k of (a_{k-i} + a_{i-k}) / 2
  (this is how much i contributes to the peaks)
- Coordinates with high H_i are "hot" and should have mass removed
- Coordinates with low H_i are "cold" and should receive mass

This gives a natural coordinate descent strategy:
1. Compute the autocorrelation and identify peak lags
2. Compute the "heat" of each coordinate
3. For the hottest coordinate i, find the coldest coordinate j
4. Do a line search on the pair (i, j) to find the optimal transfer amount
5. Repeat

This is similar to the "top-k subgradient" approach but with exact line search instead of a fixed step.

```python
def coordinate_descent_step(a, peak_lags):
    """One coordinate descent step: move mass from hottest to coldest coordinate."""
    n = len(a)
    c = autocorr(a)
    current_peak = c.max()
    
    # Identify peak lags (top k)
    top_k = 5
    k_vals = np.argsort(c)[-top_k:]
    
    # Compute heat for each coordinate
    heat = np.zeros(n)
    for k in k_vals:
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                heat[i] += a[j]
            j = i - k
            if 0 <= j < n:
                heat[i] += a[j]
    
    # Find hottest and coldest
    hot = np.argmax(heat)
    cold = np.argmin(heat)
    
    # Line search: find optimal t to move from hot to cold
    # b_k(t) = autocorrelation after moving t from hot to cold
    # Use ternary search on t in [0, a[hot]]
    lo, hi = 0.0, a[hot]
    for _ in range(30):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        a1 = a.copy(); a1[hot] -= m1; a1[cold] += m1
        a2 = a.copy(); a2[hot] -= m2; a2[cold] += m2
        p1 = autocorr(a1).max()
        p2 = autocorr(a2).max()
        if p1 < p2:
            hi = m2
        else:
            lo = m1
    
    t = (lo + hi) / 2
    a[hot] -= t
    a[cold] += t
    return a
```

**Optimization: Analytical peak change**

Instead of recomputing the full autocorrelation after each move, I can compute the change analytically:

b_k(t) = b_k(0) + 2t*(a_{k-j} - a_{k-i}) + t^2 * ([k=2i] + [k=2j] - 2*[k=i+j])

So for a given move (i, j, t), the new autocorrelation is:
new_b[k] = b[k] + 2t*(a[k-j] - a[k-i]) + t^2 * ([k=2i] + [k=2j] - 2*[k=i+j])

where a[k-j] means a[k-j] if 0<=k-j<n else 0, etc.

This can be computed in O(n) without an FFT! So each evaluation is O(n) instead of O(n log n).

For n=6121, O(n) = 6121 operations (just array indexing and addition). This is about 10x faster than the FFT approach.

Let me implement this:

```python
def eval_move(a, b, i, j, t):
    """Compute new autocorrelation after moving t from coordinate i to coordinate j."""
    n = len(a)
    # Linear term: 2t * (a[k-j] - a[k-i])
    shift_j = np.zeros(2*n-1)
    shift_i = np.zeros(2*n-1)
    # a[k-j] for k=0..2n-2, valid when 0<=k-j<n i.e. j<=k<j+n
    if j < 2*n-1:
        lo = max(0, j)
        hi = min(2*n-1, j+n)
        shift_j[lo:hi] = a[lo-j:hi-j]
    if i < 2*n-1:
        lo = max(0, i)
        hi = min(2*n-1, i+n)
        shift_i[lo:hi] = a[lo-i:hi-i]
    
    linear = 2*t*(shift_j - shift_i)
    
    # Quadratic term: t^2 * ([k=2i] + [k=2j] - 2*[k=i+j])
    quad = np.zeros(2*n-1)
    if 2*i < 2*n-1:
        quad[2*i] += 1
    if 2*j < 2*n-1:
        quad[2*j] += 1
    if i+j < 2*n-1:
        quad[i+j] -= 2
    
    quad *= t*t
    
    return b + linear + quad
```

This is O(n) per evaluation. With 6 evaluations per pair and 10 pairs, that's 60 * 6121 ≈ 370K operations per step. Very fast!

```python
def compute_heat(a, b, peak_indices):
    n = len(a)
    heat = np.zeros(n)
    for k in peak_indices:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        indices = np.arange(lo, hi)
        heat[indices] += a[k - indices]
    return heat
```

```python
Step function optimizer minimizing 2*n*max(conv(a,a))/(sum a)^2.
Uses exact coordinate descent with analytical autocorrelation updates,
heat-based coordinate selection, and simulated annealing for exploration.
"""

import numpy as np
import time


def autocorr_fft(a):
    """Compute linear convolution of a with itself via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(a, size)
    result = np.fft.irfft(F * F, size)
    return result[:2 * n - 1]


def apply_move(a, b, i, j, t):
    """
    Compute new autocorrelation after moving mass t from coord i to coord j.
    Uses the analytical formula:
    b_k(t) = b_k + 2t*(a[k-j] - a[k-i]) + t^2*([k=2i]+[k=2j]-2*[k=i+j])
    Returns the new b array.
    """
    n = len(a)
    m = 2 * n - 1  # length of b
    
    # Compute shift_j[k] = a[k-j] if 0<=k-j<n else 0
    shift_j = np.zeros(m)
    lo = max(0, j)
    hi = min(m, j + n)
    if lo < hi:
        shift_j[lo:hi] = a[lo - j:hi - j]
    
    # Compute shift_i[k] = a[k-i] if 0<=k-i<n else 0
    shift_i = np.zeros(m)
    lo = max(0, i)
    hi = min(m, i + n)
    if lo < hi:
        shift_i[lo:hi] = a[lo - i:hi - i]
    
    new_b = b + 2.0 * t * (shift_j - shift_i)
    
    # Quadratic term
    if 2 * i < m:
        new_b[2 * i] += t * t
    if 2 * j < m:
        new_b[2 * j] += t * t
    if i + j < m:
        new_b[i + j] -= 2.0 * t * t
    
    return new_b


def compute_heat(a, peak_lags):
    """Compute heat of each coordinate based on peak lags."""
    n = len(a)
    heat = np.zeros(n)
    for k in peak_lags:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            indices = np.arange(lo, hi)
            heat[indices] += a[k - indices]
    return heat


def coordinate_descent(a, b, deadline, rng, n, verbose=True):
    """
    Main coordinate descent loop with exact line search.
    """
    best_peak = b.max()
    best_a = a.copy()
    
    no_improve_count = 0
    step = 0
    T = best_peak * 0.01  # SA temperature
    
    while time.monotonic() < deadline:
        step += 1
        
        # Compute peaks and heat
        top_k_peaks = 3
        peak_indices = np.argsort(b)[-top_k_peaks:]
        heat = compute_heat(a, peak_indices)
        
        # Select hot and cold coordinates
        num_hot = 15
        num_cold = 15
        hot_idx = np.argsort(heat)[-num_hot:]
        cold_idx = np.argsort(heat)[:num_cold]
        
        cur_peak = b.max()
        best_move = None
        best_new_peak = cur_peak
        
        # Try all (hot, cold) pairs with several t values
        for hi in hot_idx:
            if a[hi] < 1e-15:
                continue
            for ci in cold_idx:
                if hi == ci:
                    continue
                for frac in [0.05, 0.15, 0.3, 0.5]:
                    t = a[hi] * frac
                    if t < 1e-15:
                        continue
                    new_b = apply_move(a, b, hi, ci, t)
                    new_peak = new_b.max()
                    if new_peak < best_new_peak:
                        best_new_peak = new_peak
                        best_move = (hi, ci, t, new_b)
        
        if best_move is not None:
            hi, ci, t, new_b = best_move
            a[hi] -= t
            a[ci] += t
            b = new_b
            no_improve_count = 0
            
            if b.max() < best_peak:
                best_peak = b.max()
                best_a = a.copy()
        else:
            # No improving move found - try SA move
            no_improve_count += 1
            
            # Random move with SA acceptance
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            # Move from i to j (or j to i)
            if a[i] > a[j]:
                src, dst = i, j
            else:
                src, dst = j, i
            frac = rng.choice([0.05, 0.1, 0.2])
            t = a[src] * frac
            if t < 1e-15:
                continue
            
            new_b = apply_move(a, b, src, dst, t)
            new_peak = new_b.max()
            delta = new_peak - cur_peak
            
            if delta <= 0:
                a[src] -= t
                a[dst] += t
                b = new_b
                if b.max() < best_peak:
                    best_peak = b.max()
                    best_a = a.copy()
            elif rng.random() < np.exp(-delta / (T + 1e-30)):
                a[src] -= t
                a[dst] += t
                b = new_b
            else:
                pass
            
            # Cool down
            T = max(best_peak * 1e-10, T * 0.999)
        
        # Periodic perturbation when stuck
        if no_improve_count > 2000:
            mag = min(0.05, 0.005 + 0.002 * (no_improve_count / 5000.0))
            a = a + mag / np.sqrt(n) * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            b = autocorr_fft(a)
            no_improve_count = 0
            T = best_peak * 0.05
        
        # Reheat if stuck for a long time
        if no_improve_count > 5000:
            mag = 0.1
            a = a + mag / np.sqrt(n) * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            b = autocorr_fft(a)
            no_improve_count = 0
            T = best_peak * 0.1
        
        if verbose and step % 5000 == 0:
            print(f"    CD step {step}: peak={cur_peak:.10f}, best={best_peak:.10f}, "
                  f"no_improve={no_improve_count}, T={T:.2e}")
    
    return best_a, 2.0 * n * best_peak


def generate_alt_starts(n, rng):
    """Generate alternative starting points."""
    starts = []
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    # Gaussian
    for alpha in [1.0, 3.0, 10.0]:
        starts.append(np.exp(-alpha * ((j - center) / center) ** 2))
    
    # U-shaped
    starts.append(1.0 + 5.0 * ((j - center) / center) ** 2)
    
    # Sine
    for freq in [2, 3, 5]:

    # Triangle
    starts.append(1.0 - 0.8 * np.abs(j - center) / center)
    
    # Random
    for _ in range(3):
        starts.append(rng.exponential(1.0, n))
    
    return starts
```

```python
def generate_alt_starts(n, rng, count=6):
    """Generate diverse alternative starting points."""
    starts = []
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Gaussian bells
    for alpha in [2.0, 5.0, 15.0]:
        starts.append(np.exp(-alpha * ((j - center) / center) ** 2))

    # U-shaped (high at edges)
    starts.append(0.5 + 3.0 * ((j - center) / center) ** 2)

    # Bimodal
    pos = 0.3
    starts.append(np.exp(-((j - pos * n) / (0.05 * n)) ** 2) +
                  np.exp(-((j - (1 - pos) * n) / (0.05 * n)) ** 2) + 0.01)

    # Sine-modulated
    starts.append(np.maximum(0.05, 1.0 + 0.4 * np.sin(3.0 * 2.0 * np.pi * j / n)))

    # Random exponential
    for _ in range(count - len(starts)):
        starts.append(rng.exponential(1.0, n))

    return starts

        time_per = (remaining - 30) / len(alt_starts)

        for idx, a_start in enumerate(alt_starts):
            if time.monotonic() > deadline - 20:
                break
            a_start = np.maximum(a_start, 0.0)
            s = a_start.sum()
            if s < 1e-10:
                continue
            a_start = a_start / s
            b_start = autocorr_fft(a_start)
            alloc = min(time_per, deadline - time.monotonic() - 15)
            if alloc < 15:
                break

            print(f"  Alt {idx+1}/{len(alt_starts)}: init score={2*n*b_start.max():.6f}, budget={alloc:.0f}s")
            a_opt, score = coordinate_descent(a_start, b_start, time.monotonic() + alloc,
                                              rng, n, verbose=False, log_interval=50000)
            if score < best_score:
                best_score = score
                best_a = a_opt.copy()
                print(f"    NEW BEST: {best_score:.6f}")

    # Final verification with official evaluator
    best_a = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence(best_a.tolist())
    print(f"\nOfficial evaluation: {final_score:.6f}")

    if not np.isfinite(final_score):
        print("Invalid result, using fallback...")
        best_a = np.ones(n)
        final_score = evaluate_sequence(best_a.tolist())
        print(f"Fallback uniform score: {final_score:.6f}")

    if final_score < best_score:
        best_score = final_score

    print(f"=== FINAL SCORE: {best_score:.6f} ===")
    return [float(x) for x in best_a]
```
