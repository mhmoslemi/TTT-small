```python
    """Solve the LP: maximize sum(g) subject to g >= 0 and (a*g)_k <= M for all k."""
    n = len(a)
    # Variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) -> minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: For each k = 0, ..., 2n-2:
    # sum_{j: 0<=k-j<=n-1} a_j * g_{k-j} <= M
    # Let i = k-j, so j = k-i
    # Constraint k: sum_{i=max(0,k-n+1)}^{min(k,n-1)} a_{k-i} * g_i <= M

    # A_ub[k, i] = a_{k-i} if 0 <= k-i <= n-1, else 0
    A_ub = np.zeros((2*n-1, n))
```

```python
import numpy as np
import time
from scipy import optimize
import scipy.sparse

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    """Linear auto-convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]

def _score(a):
    """Compute the evaluation score."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)

def _build_conv_matrix(a):
    """Build the matrix A where (A @ g)[k] = (a*g)[k] for linear convolution."""
    n = len(a)
    # A has shape (2n-1, n)
    # A[k, i] = a[k-i] if 0 <= k-i < n, else 0
    A = np.zeros((2*n-1, n))

        # With coefficient a[k-i] = a[0], a[1], ..., a[n-1]
        A[i:i+n, i] = a
    return A
```

```python
"""
LP best-response optimization for low-peak step functions.
Uses alternating linear programming to find directions that reduce
the peak of the self-convolution, combined with gradient-based polish.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _normalize(a):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.ones_like(a) / a.size
    return a / s


def _make_sine(n, amp, phase=0.5):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.sin(theta)) / n
    a = np.clip(a, 0.0, 1000.0)
    return _normalize(a)


def _make_multi_sine(n, amps, phases):
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n) / n
    for amp, ph in zip(amps, phases):
        theta = 2.0 * np.pi * (i + ph) / n
        a += amp * np.sin(theta) / n
    return _normalize(a)


def _build_conv_matrix(a):
    """Build matrix A where (A @ g)[k] = (a*g)[k] for linear convolution."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        A[i:i + n, i] = a
    return A


def _lp_best_response(a, M):
    """Solve LP: maximize sum(g) s.t. g >= 0, (a*g)_k <= M for all k."""
    n = len(a)
    A = _build_conv_matrix(a)
    b = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b,
                         bounds=[(0, None)] * n, method='highs',
                         options={'presolve': True, 'time_limit': 30})
        if result.status == 0 and result.x is not None:
            return result.x
    except Exception:
        pass
    
    # Fallback
    return a.copy()


def _lp_optimize(a, deadline, verbose=True):
    """LP best-response optimization with adaptive step size and line search."""
    a = _normalize(a.copy())
    n = a.size
    best_a = a.copy()
    best_score = _score(best_a)
    
    t_candidates = [1.0, 0.7, 0.5, 0.4, 0.3, 0.2, 0.1, 0.05]
    rng = np.random.RandomState(42)
    no_improve_count = 0
    max_no_improve = 30
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        c = _fft_convolve(a)
        M = float(np.max(c))
        
        # Solve LP
        g = _lp_best_response(a, M)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum  # normalize to sum = 1
        
        # Check if g is essentially the same as a (fixed point)
        if np.allclose(g, a, atol=1e-6):
            if verbose:
                print(f"    LP iteration {it}: fixed point reached", flush=True)
            break
        
        # Line search
        found_improvement = False
        for t in t_candidates:
            a_new = _normalize((1 - t) * a + t * g)
            sc = _score(a_new)
            if sc < best_score * (1 - 1e-10):  # strict improvement
                best_score = sc
                best_a = a_new.copy()
                a = a_new.copy()
                found_improvement = True
                break
            elif sc < best_score:
                # Small improvement, still take it
                a = a_new.copy()
                sc_check = _score(a)
                if sc_check < best_score:
                    best_score = sc_check
                    best_a = a.copy()
                found_improvement = True
                break
        
        if found_improvement:
            no_improve_count = 0
            if verbose and it % 5 == 0:
                print(f"    LP iter {it}: score={best_score:.8f}, t used", flush=True)
        else:
            no_improve_count += 1
            if no_improve_count >= max_no_improve:
                # Perturb and restart
                if verbose:
                    print(f"    LP iter {it}: no improvement, perturbing", flush=True)
                a = best_a + rng.standard_normal(n) * 0.005 * best_a.mean()
                a = _normalize(a)
                no_improve_count = 0
                max_no_improve = int(max_no_improve * 1.3)
    
    return best_a, best_score


def _grad_optimize(a, deadline, T_start=5.0, T_end=50.0, lr0=None):
    """Soft-max mirror descent (similar to previous approach but tuned)."""
    a = _normalize(a.copy())
    n = a.size
    if lr0 is None:
        lr0 = 0.2 * n
    
    best_a = a.copy()
    best_score = _score(best_a)
    it = 0
    last_improve = 0
    patience = 3000
    rng = np.random.RandomState(777)
    
    def _get_grad(w, a):
        n = a.size
        N = _next_pow2(2 * n)
        fa = np.fft.rfft(a, N)
        fw = np.fft.rfft(w[:N], N) if w.size > N else np.fft.rfft(w, N)
        xcorr = np.fft.irfft(np.conj(fa) * fw, N)
        return 2.0 * xcorr[:n]
    
    while time.monotonic() < deadline:
        it += 1
        t_frac = min(1.0, it / 50000.0)
        T = T_start + (T_end - T_start) * t_frac
        lr = lr0 / (1.0 + 0.01 * it)
        
        c = _fft_convolve(a)
        c_max = c.max()
        c_min = c.min()
        rng_c = c_max - c_min
        if rng_c < 1e-15:
            break
        c_norm = (c - c_min) / rng_c - 1.0
        exp_c = np.exp(T * c_norm)
        w = exp_c / exp_c.sum()
        
        grad = _get_grad(w, a)
        grad_proj = grad - np.mean(grad)
        update = -lr * grad_proj
        update = np.clip(update, -8.0, 8.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = _normalize(a)
        
        if it % 3 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
```

```python
def _lp_minimize_peak(a_old, M_current):
    """
    Solve LP: minimize M subject to:
    - sum(a) = 1
    - a >= 0
    - (a_old * a)_k <= M for all k
    """
    n = len(a_old)
    # Variables: a_0, ..., a_{n-1}, M (total n+1 variables)
    # Objective: minimize M -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0  # minimize M
    
    # Equality constraint: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Inequality constraints: (a_old * a)_k <= M for k = 0, ..., 2n-2
    # sum_i a_old[k-i] * a_i - M <= 0
    # A_ub[k, i] = a_old[k-i] (if valid), A_ub[k, n] = -1
    A_ub = np.zeros((2 * n - 1, n + 1))
    for i in range(n):
        for k in range(i, i + n):
            A_ub[k, i] = a_old[k - i]
    A_ub[:, n] = -1.0  # coefficient of M is -1
    
    b_ub = np.zeros(2 * n - 1)
    
    # Bounds: a_i >= 0, M >= 0
    bounds = [(0, None)] * n + [(0, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status == 0:
        return result.x[:n], result.x[n]
    else:
        return a_old, M_current
```
