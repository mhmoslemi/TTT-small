import numpy as np
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)

def _grad_convolution(w, a):
    """
    Compute the gradient of sum_k w[k]*c[k] with respect to a.
    c = a * a (linear convolution)
    grad[i] = sum_k w[k] * dc[k]/da[i]
    = sum_k w[k] * (a[k-i] + a[k-i]) [with appropriate range checks]
    = 2 * sum_k w[k] * a[k-i] for valid k-i
    This is essentially a cross-correlation of w and a.
    """
    n = a.size
    # c[k] = sum_j a[j]*a[k-j], k = 0..2n-2
    # dc[k]/da[i] = 2*a[k-i] if 0 <= k-i < n (approximately, ignoring diagonal)
    # grad[i] = sum_{k: 0<=k-i<n} 2*w[k]*a[k-i] = 2 * sum_{j: 0<=j<n} w[i+j]*a[j]
    # This is 2 * (w reversed cross-correlated with a) or equivalently a correlation.
    
    # Let's compute: grad[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j] for i=0..n-1
    # (where w has length 2n-1, so i+j ranges from i to i+n-1, all < 2n-1 for i < n)
    
    N = _next_pow2(2 * n)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    w_pad = np.zeros(N)
    w_pad[:min(2*n-1, N)] = w[:min(2*n-1, N)]
    
    # We want: grad[i] = 2 * sum_j a[j] * w[i+j]
    # This is the cross-correlation of a and w, evaluated at lags 0..n-1.
    # Cross-correlation: (a ⊛ w)[i] = sum_j a[j] * w[j+i]
    
    fa = np.fft.rfft(a_pad, N)
    fw = np.fft.rfft(w_pad, N)
    # Cross-correlation via FFT: IFFT(conj(FFT(a)) * FFT(w)) gives sum_j a[j]*w[j+i]
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]

def _optimize_mirror_descent(a_init, deadline, T_start=5.0, T_end=200.0, lr_init=None, top_k=None):
    """Mirror descent optimization with various modes."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size) / a.size
        s = 1.0
    a = a / s
    
    n = a.size
    if lr_init is None:
        lr_init = 0.3 * n
    
    best_a = a.copy()
    best_score = _score(best_a)
    it = 0
    last_improve = 0
    patience = 8000
    rng = np.random.RandomState(42)
    
    while time.monotonic() < deadline:
        it += 1
        t_frac = min(1.0, it / 50000.0)
        T = T_start + (T_end - T_start) * t_frac
        lr = lr_init / (1.0 + 0.005 * it)
        
        c = _fft_convolve(a)
        if c.size == 0:
            break
        
        # Compute weights
        if top_k is not None and c.size > top_k:
            # Use top-K with soft weights
            top_idx = np.argpartition(c, -top_k)[-top_k:]
            w = np.zeros(c.size)
            top_vals = c[top_idx]
            T_loc = 5.0
            weights = np.exp(T_loc * (top_vals - top_vals.max()))
            weights /= weights.sum()
            for i, k in enumerate(top_idx):
                w[k] = weights[i]
        else:
            # Soft-max over all
            c_shift = c - c.max()
            w = np.exp(T * c_shift)
            s_w = w.sum()
            if s_w < 1e-300:
                w = np.zeros(c.size)
                w[np.argmax(c)] = 1.0
            else:
                w = w / s_w
        
        grad = _grad_convolution(w, a)
        grad_proj = grad - grad.mean()
        
        update = -lr * grad_proj
        update = np.clip(update, -5.0, 5.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            s = 1.0
            break
        a = a / s
        
        if it % 10 == 0:

            if np.isfinite(sc) and sc < best_score:

                a = np.maximum(0, a + noise)
                a = a / a.sum()
                patience = int(patience * 1.5)

def _peak_shaving(a, deadline, n_shave=5):
    """
    Peak shaving: iteratively identify the largest peaks and redistribute mass.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size) / a.size
        s = 1.0
    a = a / s
    
    n = a.size
    best_a = a.copy()
    best_score = _score(best_a)
    rng = np.random.RandomState(77)
    
    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        if c.size == 0:
            break
        
        # Find top peaks
        top_k = min(n_shave, c.size)
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        
        improved = False
        for k in top_idx:
            if time.monotonic() > deadline:
                break
            # This peak c[k] = sum_i a[i]*a[k-i]
            # Find the pair (i, k-i) with maximum product
            valid_i = np.arange(max(0, k - n + 1), min(n, k + 1))
            products = a[valid_i] * a[k - valid_i]
            if products.size == 0:
                continue
            i_max = valid_i[np.argmax(products)]
            
            # Try to reduce a[i_max] and a[k-i_max] slightly,
            # and add to a random position that doesn't contribute to top peaks
            # Find a "safe" position to add mass
            # A position p is safe if for all top peaks k', p is not in [k'-n+1, k']
            # (i.e., a[p] doesn't contribute to any top peak)
            
            # Simpler: just try reducing the max contributor and adding to a random small element
            delta = 0.01 * a.mean()
            
            # Try multiple random targets
            for _ in range(20):
                p = rng.randint(n)
                if p == i_max or (0 <= k - i_max < n and p == k - i_max):
                    continue
                a_new = a.copy()
                a_new[i_max] -= delta
                if a_new[i_max] < 0:
                    continue
                a_new[p] += delta
                a_new = a_new / a_new.sum()
                sc = _score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new.copy()
                    improved = True
                    break
        
        if not improved and rng.random() < 0.1:
            # Perturb to escape local minimum
            a = best_a.copy()
            idx = rng.randint(n)
            delta = rng.uniform(0.001, 0.01) * a.mean()
            a[idx] += delta
            a[idx // 2] = max(0, a[idx // 2] - delta)
            a = np.clip(a, 0, 1000)
            a = a / a.sum()
    
    return best_a, best_score

The score is 2*n*max_k c[k] / S^2. Since it's scale-invariant, set S = 1. Then minimize M = max_k c[k].

Note that c[k] = sum_i a[i]*a[k-i] is a quadratic form in a. Specifically, c[k] = a^T (D_k) a where D_k is a specific matrix.

The problem is: min_a max_k a^T D_k a, subject to a >= 0, sum(a) = 1.

This is a non-convex problem (max of quadratic forms). But we can use the following approach:

**Bisection on M**: 
- Given a target M, check if there exists a >= 0, sum(a) = 1 such that a^T D_k a <= M for all k.
- This is a feasibility problem with quadratic constraints.

For a fixed M, the constraints a^T D_k a <= M are (non-convex in general) quadratic constraints. However, if we restrict to a specific structure (like the anti-palindromic family), the problem might become convex.

Actually, let's think about the anti-palindromic family: a[i] = 1/n + d[i] where d[i] = -d[n-1-i] and sum(d) = 0.

In this family, c[n-1] = sum a[i]*a[n-1-i] = sum (1/n + d[i])(1/n - d[i]) = sum (1/n^2 - d[i]^2) = 1/n - sum d[i]^2.

So c[n-1] = 1/n - ||d||^2. To MINIMIZE c[n-1], we want ||d||^2 to be as LARGE as possible. But ||d||^2 is bounded by the non-negativity constraint: a[i] = 1/n + d[i] >= 0, so d[i] >= -1/n.

The maximum ||d||^2 subject to d[i] >= -1/n, d[i] = -d[n-1-i], sum(d) = 0:
If we set d[0] = -1/n (so a[0] = 0) and d[n-1] = 1/n (so a[n-1] = 2/n), and similarly for other pairs...
Actually, the max of sum d[i]^2 subject to d[i] >= -1/n and d[i] = -d[n-1-i] is achieved by making as many |d[i]| as large as possible.

If we set d[i] = -1/n for i = 0, 1, ..., m and d[n-1-i] = 1/n for those i, then:
||d||^2 = 2*m*(1/n)^2 (plus contributions from unassigned pairs which are 0).
The constraint is that we can have at most n/2 such pairs (if n is even).
Max ||d||^2 = 2*(n/2)*(1/n)^2 = 1/n.
So min c[n-1] = 1/n - 1/n = 0.

But if c[n-1] = 0, what about other c[k]? Let's see: if a[0] = a[1] = ... = a[m] = 0 and a[n-1] = a[n-2] = ... = a[n-1-m] = 2/n, and the middle elements are 1/n...

This is getting complicated. The point is that in the anti-palindromic family, we can drive c[n-1] to 0, but other peaks will increase.

Key insight: The function f(a) = max_k a^T D_k a is the maximum of (convex) quadratic functions. It's convex! (The maximum of convex functions is convex.)

Wait, is a^T D_k a convex in a? It is if D_k is positive semidefinite. Let me check.

D_k for the k-th convolution coefficient: c[k] = sum_{i,j: i+j=k} a[i]*a[j] (with the appropriate range).
Actually, c[k] = sum_i a[i]*a[k-i] where the sum is over valid i.
This can be written as a^T Q_k a where Q_k[i,j] = 1/2 if i+j=k (and i≠j), Q_k[i,i] = 1 if 2i=k, Q_k[i,j] = 0 otherwise.

Is Q_k PSD? For k = 0: Q_0[0,0] = 1, and a^T Q_0 a = a[0]^2 >= 0. ✓
For k = 1: Q_1[0,1] = Q_1[1,0] = 1/2. a^T Q_1 a = a[0]*a[1]. This is NOT always non-negative (well, it is since a >= 0, but the quadratic form itself isn't PSD on all of R^n, only on R_+^n).

Actually, for the constraint a >= 0, we only care about PSD on the non-negative orthant. In that case, a^T Q_k a = sum_i a[i]*a[k-i] >= 0, so it is "PSD on the non-negative orthant."

"""
Multi-strategy optimization for low-peak nonnegative step function.
Uses mirror descent with momentum, adaptive active-set tracking,
peak shaving, and length exploration to minimize 2*n*max(conv)/sum^2.
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _grad_convolution(w, a):
    n = a.size
    N = _next_pow2(2 * n)
    a_pad = np.zeros(N)
    a_pad[:n] = a
    w_len = min(w.size, 2 * n - 1)
    w_pad = np.zeros(N)
    w_pad[:w_len] = w[:w_len]
    fa = np.fft.rfft(a_pad, N)
    fw = np.fft.rfft(w_pad, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]


def _make_anti_palindromic(n, amp, mode=1):
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    d = amp * np.sin(2.0 * np.pi * mode * (i - center) / n)
    a = 1.0 / n + d / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def _make_multi_tone(n, amps, modes, phases):
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.zeros(n)
    for amp, mode, ph in zip(amps, modes, phases):
        a += amp * np.sin(2.0 * np.pi * mode * (i - center) / n + ph)
    a = 1.0 / n + a / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def _optimize_mirror_momentum(a_init, deadline, T_start=5.0, T_end=200.0,
                              lr_init=None, momentum=0.7, top_k=0,
                              patience_base=5000):
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size) / a.size
        s = 1.0
    a = a / s

    n = a.size
    if lr_init is None:
        lr_init = 0.25 * n

    best_a = a.copy()
    best_score = _score(best_a)
    velocity = np.zeros(n)
    it = 0
    last_improve = 0
    patience = patience_base
    rng = np.random.RandomState(12345)

    while time.monotonic() < deadline:
        it += 1
        t_frac = min(1.0, it / 80000.0)
        T = T_start + (T_end - T_start) * (t_frac ** 0.5)
        lr = lr_init / (1.0 + 0.003 * it)

        c = _fft_convolve(a)
        if c.size == 0:
            break

        if top_k > 0 and c.size > top_k:
            top_idx = np.argpartition(c, -top_k)[-top_k:]
            w = np.zeros(c.size)
            top_vals = c[top_idx]
            T_loc = 8.0
            weights = np.exp(T_loc * (top_vals - top_vals.max()))
            weights /= weights.sum()
            for i_idx, k in enumerate(top_idx):
                w[k] = weights[i_idx]
        else:
            c_shift = c - c.max()
            w = np.exp(T * c_shift)
            s_w = w.sum()
            if s_w < 1e-300:
                w = np.zeros(c.size)
                w[np.argmax(c)] = 1.0
            else:
                w = w / s_w

        grad = _grad_convolution(w, a)
        grad_proj = grad - grad.mean()

        velocity = momentum * velocity + (1.0 - momentum) * grad_proj
        update = -lr * velocity
        update = np.clip(update, -4.0, 4.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            velocity = np.zeros(n)
            s = 1.0
            break
        a = a / s

        if it % 8 == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                a = best_a.copy()
                velocity = np.zeros(n)
                noise_scale = 0.003 + 0.002 * it / patience
                noise = rng.standard_normal(n) * noise_scale * a.mean()
                a = np.maximum(0, a + noise)
                a = a / a.sum()
                patience = int(patience * 1.4)
                lr_init = max(0.05 * n, lr_init * 0.75)

    return best_a, best_score


def _optimize_adaptive_topk(a_init, deadline, k_start=5, k_end=20):
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size) / a.size
        s = 1.0
    a = a / s

    n = a.size
    lr = 0.3 * n
    best_a = a.copy()
    best_score = _score(best_a)
    it = 0
    last_improve = 0
    patience = 3000
    rng = np.random.RandomState(999)

    while time.monotonic() < deadline:
        it += 1
        t_frac = min(1.0, it / 40000.0)
        K = k_start + int((k_end - k_start) * t_frac)
        lr_cur = lr / (1.0 + 0.002 * it)

        c = _fft_convolve(a)
        if c.size == 0:
            break

        top_idx = np.argpartition(c, -min(K, c.size))[-min(K, c.size):]
        w = np.zeros(c.size)
        top_vals = c[top_idx]
        T_loc = 6.0
        weights = np.exp(T_loc * (top_vals - top_vals.max()))
        weights /= weights.sum()
        for i_idx, k in enumerate(top_idx):
            w[k] = weights[i_idx]

        grad = _grad_convolution(w, a)
        grad_proj = grad - grad.mean()

        update = -lr_cur * grad_proj
        update = np.clip(update, -3.0, 3.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            s = 1.0
            break
        a = a / s

        if it % 6 == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
                lr = min(0.8 * n, lr * 1.003)
            elif it - last_improve > patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.002 * a.mean()
                a = np.maximum(0, a + noise)
                a = a / a.sum()
                patience = int(patience * 1.3)
                lr = max(0.1 * n, lr * 0.8)

    return best_a, best_score


def _coordinate_descent(a_init, deadline, n_steps=50):
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size) / a.size
        s = 1.0
    a = a / s

    n = a.size
    best_a = a.copy()
    best_score = _score(best_a)
    rng = np.random.RandomState(2024)

    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        if c.size == 0:
            break

        top_k = min(8, c.size)
        top_idx = np.argpartition(c, -top_k)[-top_k:]

        improved = False
        for _ in range(n_steps):
            if time.monotonic() > deadline:
                break
            k = top_idx[rng.randint(len(top_idx))]
            valid_i = np.arange(max(0, k - n + 1), min(n, k + 1))
            if valid_i.size == 0:
                continue
            products = a[valid_i] * a[k - valid_i]
            i_max = valid_i[np.argmax(products)]
            j_max = k - i_max

            for _ in range(5):
                p = rng.randint(n)
                if p == i_max or (0 <= j_max < n and p == j_max):
                    continue
                delta = rng.uniform(0.002, 0.02) * a.mean()
                if a[i_max] < delta and a[j_max] < delta:
                    continue
                a_new = a.copy()
                a_new[i_max] = max(0, a_new[i_max] - delta * 0.5)
                if 0 <= j_max < n and j_max != i_max:
                    a_new[j_max] = max(0, a_new[j_max] - delta * 0.5)
                a_new[p] += delta
                s_new = a_new.sum()
                if s_new < 0.01:
                    continue
                a_new = a_new / s_new
                sc = _score(a_new)
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new.copy()
                    improved = True
                    break

        if not improved:
            a = best_a.copy()
            idx = rng.randint(n)
            delta = rng.uniform(0.001, 0.005) * a.mean()
            a[idx] += delta
            q = (idx + 1) % n
            a[q] = max(0, a[q] - delta)
            a = a / a.sum()

    return best_a, best_score


def _scan_initializations(deadline):
    best_a = None
    best_score = np.inf
    rng = np.random.RandomState(555)

    lengths = [1024, 2048, 512, 4096]

    for n_val in lengths:
        if time.monotonic() > deadline:
            break
        for amp in [0.6, 0.7, 0.75, 0.8, 0.85, 0.9, 0.95, 1.0]:
            a = _make_anti_palindromic(n_val, amp, mode=1)
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

        for _ in range(3):
            if time.monotonic() > deadline:
                break
            amps = [rng.uniform(0.5, 0.9), rng.uniform(0.05, 0.2)]
            modes = [1, rng.choice([2, 3])]
            phases = [0.0, rng.uniform(0, 2 * np.pi)]
            a = _make_multi_tone(n_val, amps, modes, phases)
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                print(f"    [multi] n={n_val} score={sc:.8f}", flush=True)

    if best_a is None:
        best_a = _make_anti_palindromic(1024, 0.8)
        best_score = _score(best_a)

    return best_a, best_score


def _try_resample_and_optimize(a_src, new_n, deadline):
    old_n = len(a_src)
    old_x = np.linspace(0, 1, old_n)
    new_x = np.linspace(0, 1, new_n)
    a_new = np.interp(new_x, old_x, a_src)
    a_new = np.clip(a_new, 0, 1000)
    s = a_new.sum()
    if s > 0.01:
        a_new = a_new / s
    sc = _score(a_new)
    if not np.isfinite(sc):
        return a_new, sc
    a_opt, s_opt = _optimize_mirror_momentum(
        a_new, deadline, T_start=5.0, T_end=100.0,
        lr_init=0.25 * new_n, momentum=0.7, top_k=8,
        patience_base=3000)
    return a_opt, s_opt

        print(f"\n[Phase 2b] Top-K from init ({remaining:.0f}s)...", flush=True)
        a2b, s2b = _optimize_adaptive_topk(init_a, d2, k_start=3, k_end=15)
        print(f"  Score: {s2b:.8f}", flush=True)
        if s2b < best_score:
            best_score = s2b
            best_a = a2b.copy()

    # Phase 3: Coordinate descent (polish)
    d3 = t0 + budget_s * 0.80
    remaining = d3 - time.monotonic()
    if remaining > 15:
        print(f"\n[Phase 3] Coordinate descent ({remaining:.0f}s)...", flush=True)
        a3, s3 = _coordinate_descent(best_a, d3, n_steps=30)
        print(f"  Score: {s3:.8f}", flush=True)
        if s3 < best_score:
            best_score = s3
            best_a = a3.copy()

    # Phase 4: Length exploration
    d4 = t0 + budget_s * 0.92
    print(f"\n[Phase 4] Length exploration...", flush=True)
    for new_n in [512, 2048, 4096]:
        if time.monotonic() > d4:
            break
        time_per_trial = min(60, max(20, (d4 - time.monotonic()) / 3))
        d_trial = time.monotonic() + time_per_trial
        a_n, s_n = _try_resample_and_optimize(best_a, new_n, d_trial)
        print(f"  n={new_n}: score={s_n:.8f}", flush=True)
        if s_n < best_score:
            best_score = s_n
            best_a = a_n.copy()

    # Final polish with mirror descent on best
    d5 = t0 + budget_s - 20.0
    remaining = d5 - time.monotonic()
    if remaining > 15:
        print(f"\n[Phase 5] Final polish ({remaining:.0f}s)...", flush=True)
        a5, s5 = _optimize_mirror_momentum(
            best_a, d5, T_start=10.0, T_end=50.0,
            lr_init=0.15 * best_a.size, momentum=0.5, top_k=6,
            patience_base=2000)
        print(f"  Score: {s5:.8f}", flush=True)
        if s5 < best_score:
            best_score = s5
            best_a = a5.copy()

    # Validate
    print(f"\n[Validation] Best internal: {best_score:.8f}", flush=True)
    candidates = [best_a, init_a]
    best_actual = np.inf
    best_out = None

    for cand in candidates:
        if time.monotonic() > t0 + budget_s - 5:
            break
        arr_list = [float(max(0.0, min(1000.0, x))) for x in cand]
        try:
            actual = float(evaluate_sequence(arr_list))
        except Exception:
            actual = np.inf
        print(f"  n={cand.size}: actual={actual:.8f}", flush=True)
        if np.isfinite(actual) and actual < best_actual:
            best_actual = actual
            best_out = arr_list

    if best_out is None:
        best_out = [float(max(0.0, min(1000.0, x))) for x in best_a]
        try:
            best_actual = float(evaluate_sequence(best_out))
        except Exception:
            best_actual = best_score

    total_time = time.monotonic() - t0
    print(f"\n[Done] Time: {total_time:.1f}s, Best: {best_actual:.8f}", flush=True)
    return best_out
