```python
"""
Optimizer for non-negative step-function coefficient sequences.

The returned sequence minimizes (or approximates minimization of)

    score(a) = 2 * n * max_k (a * a)_k / (sum(a)**2),

where `*` is linear convolution.  The optimization is scale invariant, so we
normalize sum(a)=1 and minimize 2*n*max_k (a*a)_k.

Main ingredients:
  - FFT evaluation of a*a.
  - Random/analytic smooth seeds, including a coarse-to-fine stage.
  - A strong local search:
      * active-lag weighted-gradient selection of donor/receiver coordinates,
      * exact one-dimensional mass-transfer candidates,
      * full-convolution verification of top moves,
      * multiplicative / projected Euclidean smoothing moves,
      * careful occasional perturbations from the best point.
  - Optional higher-length refinement, followed by a final adolescent polish.
"""

import numpy as np
import time
import math

TARGET_SCORE = 1.5027


def _fft_size(m):
    """Smallest power of two >= m (and at least 2)."""
    if m <= 2:
        return 2
    return 1 << (int(m) - 1).bit_length()


def _normalize(v):
    """Project to a finite non-negative float vector with sum 1."""
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v


def _autocorr(f):
    """Linear convolution f*f via FFT. f is assumed non-negative/normalized."""
    n = int(f.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    L = 2 * n - 1
    size = _fft_size(L)
    F = np.fft.rfft(f, size)
    c = np.fft.irfft(F * F, size)
    c = c[:L]
    # The exact convolution is non-negative; suppress tiny FFT negatives.
    np.maximum(c, 0.0, out=c)
    return c


def _fast_score(f):
    """Internal fast score for a normalized non-negative vector."""
    f = _normalize(f)
    if f.size == 0:
        return np.inf
    c = _autocorr(f)
    if c.size == 0:
        return np.inf
    return float(2.0 * float(f.size) * float(c.max()))


def _score_from_c(f, c):
    if c.size == 0:
        return np.inf
    return float(2.0 * float(f.size) * float(c.max()))


def _project_simplex(v):
    """Euclidean projection onto x >= 0, sum x = 1."""
    v = np.asarray(v, dtype=np.float64)
    n = int(v.size)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    out = v - theta
    np.maximum(out, 0.0, out)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    out /= s
    return out


def _interp_to_length(f, n):
    """Linearly resample non-negative f to length n and normalize."""
    f = _normalize(f)
    m = int(f.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    if m == 0:
        return np.ones(n, dtype=np.float64)
    if m == 1:
        return np.ones(n, dtype=np.float64)
    if m == n:
        return f
    x_old = np.linspace(0.0, m - 1, m, dtype=np.float64)
    x_new = np.linspace(0.0, n - 1, n, dtype=np.float64)
    return _normalize(np.interp(x_new, x_old, f))


def _smooth_with_window(f, win):
    """Uniform moving-average smoothing."""
    f = _normalize(f)
    n = int(f.size)
    if n < 3:
        return f
    win = int(win)
    win = max(3, min(n, win))
    kernel = np.ones(win, dtype=np.float64) / float(win)
    out = np.convolve(f, kernel, mode="same")
    return _normalize(out)


def _fourier_seed(n, rng, strength=1.0):
    """Smooth random low-frequency positive seed."""
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    u = np.zeros(n, dtype=np.float64)
    for m in (1, 2, 3, 5, 8, 13):
        amp = float(strength) * float(rng.uniform(0.15, 1.0)) / float(np.sqrt(m))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        u += amp * float(rng.uniform(-1.0, 1.0)) * np.cos(2.0 * np.pi * m * x + phase)
        u += amp * float(rng.uniform(-1.0, 1.0)) * np.sin(2.0 * np.pi * m * x + phase)
    f = np.exp(0.65 * u)
    f *= 1.0 + 0.35 * float(rng.uniform(-1.0, 1.0)) * (x - 0.5)
    f = np.maximum(f, 1e-7)
    return _normalize(f)


def _param_seed(n, rng):
    """Low-harmonic asymmetric seed with guaranteed positive envelope."""
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    u = np.zeros(n, dtype=np.float64)
    for m in (1, 2, 3, 4, 6):
        amp = float(rng.uniform(-0.9, 0.9)) / float(m)
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        u += amp * np.cos(2.0 * np.pi * m * x + phase)
    f = 1.0 + u
    f *= np.exp(float(rng.uniform(-0.6, 0.6)) * (x - 0.5))
    f = np.maximum(f, 0.02)
    return _normalize(f)


def _beta_seed(n, rng):
    """Beta-shaped one-sided seed."""
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = float(rng.uniform(0.5, 3.2))
    b = float(rng.uniform(0.5, 3.2))
    f = (x + 1e-4) ** a * (1.0 - x + 1e-4) ** b
    return _normalize(f)


def _random_smoothed_seed(n, rng):
    """Random vector smoothed to avoid pathological spikes."""
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    f = rng.random(n, dtype=np.float64)
    f = np.maximum(f, 1e-6)
    f = _smooth_with_window(f, max(3, n // 100))
    return _normalize(f)


def _add_seed(seeds, name, f, n):
    """Evaluate and append a candidate seed if finite."""
    try:
        f2 = _interp_to_length(f, n)
        sc = _fast_score(f2)
        if np.isfinite(sc):
            seeds.append((float(sc), len(seeds), str(name), f2))
    except Exception:
        pass


def _weighted_gradient(w, f):
    """
    Gradient of sum_k w_k (f*f)_k with respect to f.
    Returns length-n vector g_j = 2 * sum_k w_k f_{k-j}.
    """
    n = int(f.size)
    L = int(w.size)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = L + n - 1
    size = _fft_size(m)
    W = np.fft.rfft(w, size)
    F = np.fft.rfft(f[::-1], size)
    conv = np.fft.irfft(W * F, size)
    return 2.0 * conv[n - 1:n + n - 1]


def _top_weights(c, k):
    """Uniform weight over the top-k lags."""
    L = int(c.size)
    if L == 0:
        return np.empty(0, dtype=np.float64)
    k = int(k)
    if k <= 0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    k = min(k, L)
    w = np.zeros(L, dtype=np.float64)
    if k == L:
        w.fill(1.0 / float(L))
        return w
    idx = np.argpartition(c, L - k)[L - k:]
    w[idx] = 1.0 / float(k)
    return w


def _softmax_weights(c, k, tau):
    """Softmax weights over top-k lags (or all lags if k >= len(c))."""
    L = int(c.size)
    if L == 0:
        return np.empty(0, dtype=np.float64)
    M = float(c.max())
    if M <= 0.0:
        return np.full(L, 1.0 / float(L), dtype=np.float64)
    T = max(float(M) * float(tau), 1e-300)
    k = int(k)
    w = np.zeros(L, dtype=np.float64)

    if 0 < k < L:
        idx = np.argpartition(c, L - k)[L - k:]
        vals = c[idx].astype(np.float64, copy=True)
        z = (vals - M) / T
        np.clip(z, -50.0, 0.0, out=z)
        p = np.exp(z)
        s = float(p.sum())
        if s <= 1e-300:
            w.fill(1.0 / float(L))
        else:
            w[idx] = p / s
        return w

    z = c.astype(np.float64, copy=True)
    z = (z - M) / T
    np.clip(z, -50.0, 0.0, out=z)
    p = np.exp(z)
    s = float(p.sum())
    if s <= 1e-300:
        w.fill(1.0 / float(L))
    else:
        w = p / s
    return w


def _select_directions(g, f, k, high, avoid, rng):
    """
    Select k coordinate indices.
    high=True: largest gradient (donors), and require positive mass.
    high=False: smallest gradient (receivers).
    """
    n = int(g.size)
    if n == 0:
        return []
    avoid_set = set(int(x) for x in avoid) if avoid else set()
    out = []
    seen = set()

    if high:
        if n > k:
            idxs = np.argpartition(g, -k)[-k:]
        else:
            idxs = np.arange(n)
    else:
        if n > k:
            idxs = np.argpartition(g, k)[:k]
        else:
            idxs = np.arange(n)

    for idx in idxs:
        idx = int(idx)
        if idx in seen or idx in avoid_set:
            continue
        if high and f[idx] <= 1e-14:
            continue
        seen.add(idx)
        out.append(idx)
        if len(out) >= k:
            return out

    for _ in range(20 * max(1, k)):
        idx = int(rng.integers(0, n))
        if idx in seen or idx in avoid_set:
            continue
        if high and f[idx] <= 1e-14:
            continue
        seen.add(idx)
        out.append(idx)
        if len(out) >= k:
            break
    return out


def _try_gradient_updates(f, c, Mc, g, deadline):
    """Try several mirror / projected Euclidean smoothing moves."""
    n = int(f.size)
    if n == 0:
        return None
    gm = g - float(g.mean())
    scale = float(gm.max() - gm.min())
    if not np.isfinite(scale) or scale <= 1e-300:
        return None

    best = None
    for eta in (0.02, 0.08, 0.3, 1.0, 3.0):
        if time.monotonic() > deadline:
            break

        arg = -eta * gm / scale
        np.clip(arg, -25.0, 25.0, out=arg)
        ft = f * np.exp(arg)
        ft = _normalize(ft)
        ct = _autocorr(ft)
        mc = float(ct.max())
        if np.isfinite(mc) and mc < Mc - 1e-15:
            if best is None or mc < best[2]:
                best = (ft, ct, mc)

        if eta in (0.3, 1.0):
            v = f - eta * gm / scale
            ft2 = _project_simplex(v)
            ft2 = _normalize(ft2)
            ct2 = _autocorr(ft2)
            mc2 = float(ct2.max())
            if np.isfinite(mc2) and mc2 < Mc - 1e-15:
                if best is None or mc2 < best[2]:
                    best = (ft2, ct2, mc2)
    return best


def _perturb_from_best(best_f, rng):
    """Perturb a known best state multiplicatively or by mixing."""
    best_f = _normalize(best_f)
    n = int(best_f.size)
    r = float(rng.random())

    if r < 0.35:
        sigma = float(rng.uniform(0.04, 0.85))
        f = best_f * np.exp(sigma * rng.standard_normal(n))
    elif r < 0.60:
        alpha = float(rng.uniform(0.05, 0.40))
        noise = rng.random(n, dtype=np.float64)
        s = float(noise.sum())
        if s > 1e-300:
            noise /= s
        f = (1.0 - alpha) * best_f + alpha * noise
    elif r < 0.80:
        alpha = float(rng.uniform(0.15, 0.55))
        other = _fourier_seed(n, rng)
        f = (1.0 - alpha) * best_f + alpha * other
    else:
        f = _smooth_with_window(best_f, max(3, n // 80))
        sigma = float(rng.uniform(0.08, 0.65))
        f = f * np.exp(sigma * rng.standard_normal(n))

    return _normalize(f)


def _coordinate_polish(f0, c0, budget_s, rng, tag=""):
    """
    Main local optimizer: exact single mass-transfer moves, live gradient
    donor/receiver selection, full verification, gradient escapes, and
    periodic perturbations.
    """
    f = _normalize(f0)
    n = int(f.size)
    if n == 0:
        return f, np.inf, np.empty(0, dtype=np.float64)

    L = 2 * n - 1
    try:
        c = np.asarray(c0, dtype=np.float64).copy()
    except Exception:
        c = None
    if c is None or c.size != L or not np.all(np.isfinite(c)):
        c = _autocorr(f)
    else:
        np.maximum(c, 0.0, out=c)

    maxc = float(c.max())
    best_score = float(2.0 * n * maxc)
    best_f = f.copy()

    if budget_s <= 0.0:
        return best_f, best_score, c

    deadline = time.monotonic() + float(budget_s)

    # Hyperparameters by length.
    if n >= 10000:
        D = 8
        R = 8
        Kw = 512
        Ke = 512
        nfr = 16
    elif n >= 5000:
        D = 10
        R = 10
        Kw = 1024
        Ke = 1024
        nfr = 18
    elif n >= 1000:
        D = 14
        R = 14
        Kw = 2048
        Ke = 2048
        nfr = 18
    else:
        D = 20
        R = 20
        Kw = 1024
        Ke = 1024
        nfr = 16

    Ke = min(Ke, L)
    Kw = min(Kw, L)

    if "final" in tag:
        min_frac = 1e-7
        max_frac = 0.25
        nfr = 20
    else:
        min_frac = 1e-6
        max_frac = 0.90
        nfr = max(14, min(24, nfr))

    fracs = np.logspace(float(np.log10(min_frac)), float(np.log10(max_frac)), nfr)
    fracs = np.unique(fracs)

    A_vec = np.empty(L, dtype=np.float64)
    total = np.empty(L, dtype=np.float64)

    maxS = int(D + R + 4)
    shifts_pool = np.empty((maxS, L), dtype=np.float64)
    shifts_act_pool = np.empty((maxS, Ke), dtype=np.float64)
    A_act_vec = np.empty(Ke, dtype=np.float64)

    active_idx = np.empty(0, dtype=np.int64)
    c_act = np.empty(0, dtype=np.float64)
    active_dirty = True

    no_improve = 0
    perturb_count = 0
    move_count = 0
    cycle = 0

    while time.monotonic() < deadline and no_improve < 120:
        cycle += 1

        if active_dirty:
            if Ke >= L:
                active_idx = np.arange(L, dtype=np.int64)
            else:
                active_idx = np.argpartition(c, L - Ke)[L - Ke:]
            c_act = c[active_idx]
            active_dirty = False

        Mc = float(c.max())
        cur_score = float(2.0 * n * Mc)
        if np.isfinite(cur_score) and cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()

        # Choose a lag-weighting mode; rotate to balance sparse/dense peaks.
        mode = cycle % 5
        if mode == 0:
            w = _top_weights(c, max(16, min(L, Kw // 4)))
        elif mode == 1:
            w = _softmax_weights(c, min(L, Kw), 0.15)
        elif mode == 2:
            w = _top_weights(c, max(32, min(L, Kw // 2)))
        elif mode == 3:
            w = _softmax_weights(c, L, 0.80)
        else:
            w = _top_weights(c, Kw)

        g = _weighted_gradient(w, f)
        if not np.all(np.isfinite(g)):
            g = np.zeros(n, dtype=np.float64)

        donors = _select_directions(g, f, D, True, None, rng)
        if len(donors) == 0:
            donors = [int(rng.integers(0, n))]
        receivers = _select_directions(g, f, R, False, set(donors), rng)

        labels = list(donors)
        donor_set = set(donors)
        for j in receivers:
            if j not in donor_set:
                labels.append(j)

        seen_labels = set()
        uniq_labels = []
        for idx in labels:
            idx = int(idx)
            if idx not in seen_labels:
                seen_labels.add(idx)
                uniq_labels.append(idx)
        labels = uniq_labels[:maxS]

        S = len(labels)
        shifts = shifts_pool[:S]
        shifts_act = shifts_act_pool[:S]
        label_pos = {}
        for s_idx, val in enumerate(labels):
            label_pos[int(val)] = s_idx
            row = shifts[s_idx]
            row.fill(0.0)
            row[int(val):int(val) + n] = f
            shifts_act[s_idx] = row[active_idx]

        candidates = []

        # Deterministic donor/receiver transfers, estimated on active lags.
        for i in donors:
            i_int = int(i)
            ii = label_pos.get(i_int)
            if ii is None:
                continue
            ai = float(f[i_int])
            if ai <= 1e-14:
                continue

            T0 = ai * fracs
            T0 = T0[(T0 > 1e-300) & np.isfinite(T0)]
            if T0.size == 0:
                continue

            for j in receivers:
                j_int = int(j)
                jj = label_pos.get(j_int)
                if jj is None or i_int == j_int:
                    continue

                np.subtract(shifts_act[jj], shifts_act[ii], out=A_act_vec)

                vals = c_act[None, :] + 2.0 * (T0[:, None] * A_act_vec[None, :])
                mc_est = vals.max(axis=1)

                b0 = 2 * i_int
                b1 = 2 * j_int
                b2 = i_int + j_int

                A_b = float(shifts[jj, b0] - shifts[ii, b0])
                cb = float(c[b0]) + 2.0 * T0 * A_b + T0 * T0
                np.maximum(mc_est, cb, out=mc_est)

                A_b = float(shifts[jj, b1] - shifts[ii, b1])
                cb = float(c[b1]) + 2.0 * T0 * A_b + T0 * T0
                np.maximum(mc_est, cb, out=mc_est)

                A_b = float(shifts[jj, b2] - shifts[ii, b2])
                cb = float(c[b2]) + 2.0 * T0 * A_b - 2.0 * T0 * T0
                np.maximum(mc_est, cb, out=mc_est)

                ib = int(np.argmin(mc_est))
                est = float(mc_est[ib])
                if est < Mc - 1e-15:
                    candidates.append((est, i_int, j_int, float(T0[ib])))

        # Random full-evaluation fallbacks.
        random_full_count = 10 if len(candidates) >= 3 else 30
        for _ in range(random_full_count):
            i_int = int(rng.integers(0, n))
            j_int = int(rng.integers(0, n))
            if i_int == j_int or f[i_int] <= 1e-14:
                continue
            t = float(f[i_int]) * float(fracs[int(rng.integers(0, fracs.size))])
            if t <= 0.0 or not np.isfinite(t):
                continue

            A_vec.fill(0.0)
            A_vec[i_int:i_int + n] = -f
            A_vec[j_int:j_int + n] = f
            A_vec *= 2.0 * t
            np.add(c, A_vec, out=total)

            qq = t * t
            total[2 * j_int] += qq
            total[2 * i_int] += qq
            total[i_int + j_int] -= 2.0 * qq
            mc = float(total.max())
            if np.isfinite(mc) and mc < Mc - 1e-15:
                candidates.append((mc, i_int, j_int, t))

        accepted_now = False

        if candidates:
            candidates.sort()

            seen_pairs = set()
            unique_candidates = []
            for est, x, y, t in candidates:
                key = (int(x), int(y))
                if key not in seen_pairs:
                    seen_pairs.add(key)
                    unique_candidates.append((est, int(x), int(y), float(t)))
                    if len(unique_candidates) >= 12:
                        break

            best_full = None
            for _, i_int, j_int, t in unique_candidates:
                if i_int == j_int or t <= 0.0:
                    continue
                ii = label_pos.get(i_int)
                jj = label_pos.get(j_int)
                if ii is not None and jj is not None:
                    np.subtract(shifts[jj], shifts[ii], out=A_vec)
                else:
                    A_vec.fill(0.0)
                    A_vec[i_int:i_int + n] = -f
                    A_vec[j_int:j_int + n] = f

                A_vec *= 2.0 * t
                np.add(c, A_vec, out=total)

                qq = t * t
                total[2 * j_int] += qq
                total[2 * i_int] += qq
                total[i_int + j_int] -= 2.0 * qq

                mc = float(total.max())
                if np.isfinite(mc) and mc < Mc - 1e-15:
                    if best_full is None or mc < best_full[0]:
                        best_full = (mc, i_int, j_int, t)

            if best_full is not None:
                mc, i_int, j_int, t = best_full
                ii = label_pos.get(i_int)
                jj = label_pos.get(j_int)
                if ii is not None and jj is not None:
                    np.subtract(shifts[jj], shifts[ii], out=A_vec)
                else:
                    A_vec.fill(0.0)
                    A_vec[i_int:i_int + n] = -f
                    A_vec[j_int:j_int + n] = f

                A_vec *= 2.0 * t
                np.add(c, A_vec, out=total)

                qq = t * t
                total[2 * j_int] += qq
                total[2 * i_int] += qq
                total[i_int + j_int] -= 2.0 * qq

                f[i_int] -= t
                if f[i_int] < 0.0 and f[i_int] > -1e-12:
                    f[i_int] = 0.0
                f[j_int] += t
                np.maximum(f, 0.0, out=f)

                ss = float(f.sum())
                if ss <= 1e-300 or abs(ss - 1.0) > 1e-11:
                    f = _normalize(f)
                    c = _autocorr(f)
                    mc = float(c.max())
                else:
                    c = total.copy()

                move_count += 1
                no_improve = 0
                accepted_now = True
                active_dirty = True

                cur_score = float(2.0 * n * mc)
                if np.isfinite(cur_score) and cur_score < best_score:
                    best_score = cur_score
                    best_f = f.copy()
                    if best_score < TARGET_SCORE:
                        print(f"[{tag}] move {move_count} improved below target: {best_score:.6f}")

        if not accepted_now:
            no_improve += 1
            escaped = False

            if no_improve >= 3 and time.monotonic() < deadline - 2.0:
                upd = _try_gradient_updates(f, c, Mc, g, deadline - 1.0)
                if upd is not None:
                    f, c, Mc = upd
                    f = _normalize(f)
                    c = _autocorr(f)
                    Mc = float(c.max())
                    no_improve = 0
                    active_dirty = True
                    escaped = True
                    cur_score = float(2.0 * n * Mc)
                    if np.isfinite(cur_score) and cur_score < best_score:
                        best_score = cur_score
                        best_f = f.copy()
                        if best_score < TARGET_SCORE:
                            print(f"[{tag}] gradient escaped below target: {best_score:.6f}")

            if not escaped and no_improve >= 12 and perturb_count < 30:
                if float(rng.random()) < 0.65:
                    f = _perturb_from_best(best_f, rng)
                else:
                    f = _perturb_from_best(f, rng)
                c = _autocorr(f)
                Mc = float(c.max())
                no_improve = 0
                perturb_count += 1
                active_dirty = True
                cur_score = float(2.0 * n * Mc)
                if np.isfinite(cur_score) and cur_score < best_score:
                    best_score = cur_score
                    best_f = f.copy()

            if no_improve >= 90 or (perturb_count >= 30 and no_improve >= 40):
                break

    bf = _normalize(best_f)
    bc = _autocorr(bf)
    bs = _score_from_c(bf, bc)
    return bf, bs, bc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative coefficient sequence minimizing evaluate_sequence.
    Returns a Python list of floats.
    """
    rng = np.random.default_rng(int(seed))
    t_start = time.monotonic()
    total_budget = max(120.0, float(budget_s))
    total_deadline = t_start + total_budget - 45.0

    # Load provided construction if available.
    try:
        raw = height_sequence_1  # noqa: F821
        f0 = np.asarray(raw, dtype=np.float64).copy()
    except Exception:
        f0 = None

    if f0 is None or f0.size == 0:
        f0 = np.ones(6121, dtype=np.float64)

    f0 = np.nan_to_num(f0, nan=0.0, posinf=1000.0, neginf=0.0)
    np.clip(f0, 0.0, 1000.0, out=f0)
    f0 = _normalize(f0)

    # Keep the working length in a reasonable range for direct evaluation.
    if f0.size < 512 or f0.size > 12000:
        target_n = 6121
        f0 = _interp_to_length(f0, target_n)
    n_full = int(f0.size)

    best_f = _normalize(f0)
    best_c = _autocorr(best_f)
    best_score = _score_from_c(best_f, best_c)
    print(f"initial length={best_f.size}, internal score={best_score:.6f}")

    now = time.monotonic()

    # -------------------- Coarse global exploration --------------------
    coarse_best = None
    if total_deadline - now > 1700.0:
        coarse_time = min(140.0, 0.06 * (total_deadline - now))
        n_c = 512
        seeds_c = []
        _add_seed(seeds_c, "coarse_given", f0, n_c)
        for i in range(4):
            _add_seed(seeds_c, f"coarse_fourier{i}", _fourier_seed(n_c, rng), n_c)
        for i in range(4):
            _add_seed(seeds_c, f"coarse_param{i}", _param_seed(n_c, rng), n_c)
        for i in range(3):
            _add_seed(seeds_c, f"coarse_random{i}", _random_smoothed_seed(n_c, rng), n_c)
        if not seeds_c:
            _add_seed(seeds_c, "coarse_uniform", np.ones(n_c), n_c)
        seeds_c.sort()

        top_c = seeds_c[:3]
        alloc = coarse_time / float(max(1, len(top_c)))
        for idx, item in enumerate(top_c):
            if total_deadline - time.monotonic() < 900.0:
                break
            if alloc <= 5.0:
                break
            cf, cs, cc = _coordinate_polish(item[3], None, alloc, rng, f"coarse{idx}")
            fin = _interp_to_length(cf, n_full)
            scin = _fast_score(fin)
            if np.isfinite(scin) and scin < best_score:
                best_f = _normalize(fin)
                best_score = scin
                best_c = _autocorr(best_f)
                print(f"coarse#{idx} lifted to n={n_full}, score={best_score:.6f}")
            coarse_best = _normalize(fin)

    now = time.monotonic()

    # Determine reserves for later stages.
    remaining = total_deadline - now
    final_reserve = min(260.0, 0.14 * remaining)
    if remaining > 1200.0:
        big_reserve = min(280.0, 0.11 * remaining)
    else:
        big_reserve = 0.0
    main_predict = total_deadline - final_reserve - big_reserve

    if main_predict < now + 100.0:
        main_predict = total_deadline - max(30.0, 0.10 * (total_deadline - now))

    # -------------------- Full-length main stage --------------------
    if main_predict > now + 80.0:
        seeds = []
        _add_seed(seeds, "best_current", best_f, n_full)
        _add_seed(seeds, "given", f0, n_full)
        if coarse_best is not None:
            _add_seed(seeds, "coarse_lift", coarse_best, n_full)

        for win in (3, 7, 15, 31, 101):
            _add_seed(seeds, f"smooth{win}", _smooth_with_window(f0, win), n_full)

        for gamma in (0.5, 0.7, 1.3, 1.7):
            base = f0 + 1e-12
            _add_seed(seeds, f"gamma{gamma}", np.power(base, gamma), n_full)

        _add_seed(seeds, "uniform", np.ones(n_full, dtype=np.float64), n_full)

        for i in range(6):
            _add_seed(seeds, f"fourier{i}", _fourier_seed(n_full, rng), n_full)
        for i in range(6):
            _add_seed(seeds, f"param{i}", _param_seed(n_full, rng), n_full)
        for i in range(3):
            _add_seed(seeds, f"random{i}", _random_smoothed_seed(n_full, rng), n_full)
        if not seeds:
            _add_seed(seeds, "fallback", np.ones(n_full, dtype=np.float64), n_full)

        seeds.sort()
        top = seeds[:4]
        weights = (0.60, 0.22, 0.12, 0.08)

        main_start = time.monotonic()
        main_budget = max(0.0, main_predict - main_start)

        for idx, item in enumerate(top):
            if idx >= len(weights):
                break
            if time.monotonic() >= main_predict - 12.0:
                break
            alloc = main_budget * float(weights[idx])
            remain = main_predict - time.monotonic() - 5.0
            alloc = min(alloc, remain)
            if alloc < 20.0:
                break

            print(
                f"full#{idx+1} seed={item[2]} initial={item[0]:.6f} "
                f"alloc={alloc:.1f}s current_best={best_score:.6f}"
            )
            fopt, sc, copt = _coordinate_polish(item[3], None, alloc, rng, f"full{idx}")
            if np.isfinite(sc) and sc < best_score:
                best_f = _normalize(fopt)
                best_score = float(sc)
                best_c = copt
                if best_score < TARGET_SCORE:
                    print(f"full#{idx+1} improved below target: {best_score:.6f}")

    now = time.monotonic()

    # -------------------- Optional higher-length refinement --------------------
    remaining_for_big = total_deadline - now - final_reserve
    if remaining_for_big > 300.0:
        big_budget = min(280.0, remaining_for_big - 20.0)
        if big_budget > 60.0:
            # Stay below about 10k to keep direct np.convolve evaluation fast.
            n_big = int(best_f.size)
            if n_big < 8192:
                target_big = max(8192, int(n_big * 1.25))
            else:
                target_big = int(n_big * 1.20)
            n_big = min(10000, max(n_big + 512, target_big))

            if n_big > best_f.size:
                print(f"big-length refinement n_big={n_big}, budget={big_budget:.1f}s")
                seed_big = _interp_to_length(best_f, n_big)
                fbig, sc_big, c_big = _coordinate_polish(seed_big, None, big_budget, rng, "big")
                if np.isfinite(sc_big) and sc_big < best_score:
                    best_f = _normalize(fbig)
                    best_score = float(sc_big)
                    best_c = c_big
                    if best_score < TARGET_SCORE:
                        print(f"big refinement below target: {best_score:.6f}")
                else:
                    print(f"big refinement did not improve: {sc_big:.6f} vs {best_score:.6f}")

    now = time.monotonic()

    # -------------------- Final dedicated polish --------------------
    final_time = max(0.0, total_deadline - now - 20.0)
    if final_time > 25.0:
        print(f"final coordinate polish budget={final_time:.1f}s")
        ffinal, sfinal, cfinal = _coordinate_polish(
            best_f, _autocorr(best_f), final_time, rng, "final"
        )
        if np.isfinite(sfinal) and sfinal < best_score:
            best_f = _normalize(ffinal)
            best_score = float(sfinal)
            best_c = cfinal

    # Safety normalization / sanitization.
    best_f = np.asarray(best_f, dtype=np.float64).copy()
    best_f = np.nan_to_num(best_f, nan=0.0, posinf=0.0, neginf=0.0)
    np.maximum(best_f, 0.0, out=best_f)
    np.clip(best_f, 0.0, 1000.0, out=best_f)
    s = float(best_f.sum())
    if s <= 1e-300:
        best_f = np.ones(max(1, int(best_f.size)), dtype=np.float64) / max(1, int(best_f.size))
        s = float(best_f.sum())
    best_f /= max(1e-300, s)

    best_c = _autocorr(best_f)
    best_score = _score_from_c(best_f, best_c)

    print(f"FINAL internal score = {best_score:.6f}")
    print(f"FINAL target = {TARGET_SCORE}")

    # Print official value if it is likely to be quick enough.
    if time.monotonic() < total_deadline - 3.0 and best_f.size <= 10000:
        try:
            official = evaluate_sequence([float(x) for x in best_f])  # noqa: F821
            if np.isfinite(official):
                print(f"FINAL official evaluate_sequence = {float(official):.6f}")
        except Exception:
            pass

    return [float(x) for x in best_f]
```
