Gradient-based multiplicative update optimizer for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Uses:
- Smooth max (softmax) gradient with adaptive temperature
- FFT-based convolution for O(n log n) scoring
- Adaptive step size with backtracking
- Multi-start with diverse initializations
- Coordinate descent refinement
- Spectral shaping for peak reduction

import numpy as np
import time

RNG = np.random.default_rng(42)
_CONV_CACHE = {}

def get_fft_len(n):
    if n in _CONV_CACHE:
        return _CONV_CACHE[n]
    need = 2 * n - 1
    L = 1
    while L < need:
        L <<= 1
    _CONV_CACHE[n] = L
    return L

def sanitize(a, normalize=True):
    a = np.asarray(a, dtype=float).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    if normalize:
        s = a.sum()
        if s < 1e-30:
            return None
        a = a / s
    return a

def score_fft(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = len(a)
    L = get_fft_len(n)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)
    c = c[:2*n-1]
    c = np.maximum(c, 0.0)
    return 2.0 * n * c.max(), c

def softmax_weights(c, beta):
    """Compute Boltzmann weights over convolution values."""
    c = np.maximum(c, 0.0)
    shift = c.max()
    e = np.exp(beta * (c - shift))
    s = e.sum()
    if s < 1e-30 or not np.isfinite(s):
        # Fallback: uniform or point mass
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        return w
    return e / s

"""
Gradient-based multiplicative update optimizer for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Uses smooth-max gradients with adaptive temperature, FFT-based scoring,
multi-start diversification, spectral shaping, and local refinement.
"""

import numpy as np
import time

_RNG = np.random.default_rng(777)
_CONV_LEN_CACHE = {}


def _fft_len(n):
    if n in _CONV_LEN_CACHE:
        return _CONV_LEN_CACHE[n]
    need = 2 * n - 1
    L = 1
    while L < need:
        L <<= 1
    _CONV_LEN_CACHE[n] = L
    return L


def _san(a, norm=True):
    if a is None:
        return None
    a = np.asarray(a, dtype=float).ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    if norm:
        s = a.sum()
        if s < 1e-30:
            return None
        a = a / s
    return a


def _score(a):
    n = len(a)
    L = _fft_len(n)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    sc = 2.0 * n * float(c.max())
    return sc, c


def _softmax_w(c, beta):
    m = float(c.max())
    if not np.isfinite(m):
        return np.full(len(c), 1.0 / len(c))
    e = np.exp(beta * (c - m))
    s = e.sum()
    if s < 1e-30:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
        return w
    return e / s


def _grad(a, p, L):
    n = len(a)
    Fp = np.fft.rfft(p, L)
    Fa = np.fft.rfft(a, L)
    g = 2.0 * np.fft.irfft(Fp * np.conj(Fa), L)[:n]
    return g


def _init_uniform(n):
    return np.ones(n) / n


def _init_raised_cos(n):
    t = np.arange(n)
    a = np.cos(np.pi * (t - n / 2.0 + 0.5) / n) ** 2 + 1e-6
    return a / a.sum()


def _init_gauss(n, sigma_frac=0.4):
    t = np.arange(n)
    c = n / 2.0
    sig = sigma_frac * n
    a = np.exp(-((t - c) ** 2) / (2 * sig * sig)) + 1e-6
    return a / a.sum()


def _init_beta_like(n, alpha=2.0, beta=2.0):
    t = (np.arange(n) + 0.5) / n
    a = t ** alpha * (1.0 - t) ** beta + 1e-6
    return a / a.sum()


def _init_multimodal(n, rng, k=4):
    a = np.zeros(n)
    t = np.arange(n)
    for _ in range(k):
        ctr = rng.uniform(0.1, 0.9) * n
        sig = rng.uniform(0.05, 0.3) * n
        wgt = rng.uniform(0.5, 2.0)
        a += wgt * np.exp(-((t - ctr) ** 2) / (2 * sig * sig))
    a += 1e-6
    return a / a.sum()


def _init_noisy_base(base, n, rng, sigma=0.1):
    if base is None:
        return _init_uniform(n)
    from_interp = np.interp(np.arange(n) + 0.5, np.arange(len(base)) + 0.5, base)
    a = from_interp * np.exp(rng.normal(0, sigma, n))
    a = np.maximum(a, 1e-8)
    return a / a.sum()


def _gradient_optimize(a0, time_budget, beta_start=30.0, beta_end=8000.0,
                       alpha_init=0.15, thresh_frac=0.95, verbose=False):
    a = _san(a0)
    if a is None:
        return None, float('inf')
    n = len(a)
    L = _fft_len(n)
    deadline = time.monotonic() + time_budget
    best = a.copy()
    best_sc, _ = _score(a)
    alpha = alpha_init
    beta = beta_start
    stall = 0
    it = 0
    last_improve = 0

    while time.monotonic() < deadline:
        sc, c = _score(a)
        if sc < best_sc - 1e-13:
            best_sc = sc
            best = a.copy()
            stall = 0
            last_improve = it
            if verbose and (it - last_improve >= 100):
                print(f"  grad n={n} it={it} best={best_sc:.8f}")

        maxc = float(c.max())
        target = maxc * thresh_frac

        if it % 3 == 0:
            p = _softmax_w(c, beta)
        else:
            t = target
            viol = np.maximum(c - t, 0.0)
            mv = float(viol.max())
            if mv > 1e-14 * maxc:
                w = viol / (mv + 1e-300)
                p = w ** 3 / w.sum()
            else:
                p = _softmax_w(c, beta)

        g = _grad(a, p, L)
        gm = float(g.max()) if g.size else 0.0
        if gm < 1e-15:
            a = a * np.exp(_RNG.normal(0, 0.05, n))
            a = _san(a)
            it += 1
            continue

        g = g / (gm + 1e-300)

        accepted = False
        for mult in [1.0, 0.5, 0.25, 0.125, 2.0]:
            step = alpha * mult
            a_try = a * np.exp(-step * g)
            a_try = np.maximum(a_try, 1e-18)
            a_try = a_try / a_try.sum()
            sc2, _ = _score(a_try)
            if sc2 < sc - 1e-13:
                a = a_try
                accepted = True
                break
            if mult == 0.125 and sc2 < best_sc * 1.01:
                a = a_try
                accepted = True
                break

        if accepted:
            alpha = min(2.0, alpha * 1.03)
        else:
            alpha = max(0.02, alpha * 0.75)
            stall += 1

            if stall > 30:
                sigma_noise = 0.05 + 0.005 * min(stall, 40)
                a_try = a * np.exp(_RNG.normal(0, sigma_noise, n))
                a_try = _san(a_try)
                if a_try is not None:
                    sc2, _ = _score(a_try)
                    if sc2 < best_sc * 1.05:
                        a = a_try
                stall = 0

            if stall > 150:
                a = best.copy()
                alpha = alpha_init
                beta = max(beta_start, beta * 0.4)
                stall = 0

        it += 1
        if it % 40 == 0:
            beta = min(beta_end, beta * 1.08)

    return best, best_sc


def _block_local_search(a0, time_budget, verbose=False):
    a = _san(a0)
    if a is None:
        return None, float('inf')
    n = len(a)
    L = _fft_len(n)
    best = a.copy()
    best_sc, _ = _score(a)
    stall = 0
    it = 0
    deadline = time.monotonic() + time_budget

    small = max(1, n // 4096)
    med = max(small + 1, n // 512)
    large = max(med + 1, min(n, n // 64))
    factors = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7, 2.5]

    while time.monotonic() < deadline:
        r = _RNG.random()
        if r < 0.25:
            w = max(1, min(n, _RNG.integers(1, max(2, small * 8))))
        elif r < 0.6:
            w = max(2, min(n, _RNG.integers(small, max(small + 2, med * 6))))
        else:
            w = max(2, min(n, _RNG.integers(med, max(med + 2, large * 5))))

        w = max(1, min(n, int(w)))
        start = int(_RNG.integers(0, max(1, n - w + 1)))

        best_factor = None
        best_sc_try = _score(a)[0]
        for f in factors:
            a_try = a.copy()
            a_try[start:start + w] *= f
            a_try = np.maximum(a_try, 1e-18)
            s = a_try.sum()
            if s < 1e-30:
                continue
            a_try = a_try / s
            sc_try, _ = _score(a_try)
            if sc_try < best_sc_try - 1e-13:
                best_sc_try = sc_try
                best_factor = a_try.copy()

        if best_factor is not None:
            a = best_factor
            stall = 0
            if best_sc_try < best_sc - 1e-13:
                best_sc = best_sc_try
                best = a.copy()
                if verbose and it % 200 == 0:
                    print(f"  local n={n} it={it} best={best_sc:.8f}")
        else:
            stall += 1
            if stall > 40:
                a_try = a * np.exp(_RNG.normal(0, 0.08, n))
                a_try = _san(a_try)
                if a_try is not None:
                    sc2, _ = _score(a_try)
                    if sc2 < best_sc * 1.03:
                        a = a_try
                    else:
                        a = best.copy()
                stall = 0
            if stall > 200:
                a = best.copy()
                stall = 0

        it += 1

    return best, best_sc


def _spectral_smooth(a, amount=0.1):
    a = _san(a)
    if a is None or a.size < 8:
        return a
    n = a.size
    L = _fft_len(n)
    F = np.fft.rfft(a, L)
    freqs = np.fft.rfftfreq(L)
    kernel = np.exp(-amount * (freqs * n) ** 2)
    F *= kernel
    a_sm = np.fft.irfft(F, L)[:n]
    a_sm = np.maximum(a_sm, 1e-10)
    return a_sm / a_sm.sum()


def _resample_to(a, target_n):
    src = _san(a)
    if src is None:
        return _init_uniform(target_n)
    n_src = len(src)
    if n_src == target_n:
        return src.copy()
    x_old = (np.arange(n_src) + 0.5) / n_src
    x_new = (np.arange(target_n) + 0.5) / target_n
    a_new = np.interp(x_new, x_old, src)
    a_new = np.maximum(a_new, 1e-12)
    return a_new / a_new.sum()
