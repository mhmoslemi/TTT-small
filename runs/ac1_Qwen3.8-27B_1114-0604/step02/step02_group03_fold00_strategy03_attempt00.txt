def _autocorr(f):
    """Compute linear convolution via FFT, and maintain normalization."""
    n = f.size
    if n == 0:

    full_size = 2 * n - 1
    if n == 0:

    v = np.sort(v)[::-1]
    cssv = np.cumsum(v)
    cond = v * np.arange(1, n + 1, dtype=np.float64) > (cssv[n] - 1.0)
    if np.any(cond):
        rho = np.sum(cond == True)
        theta = (cssv[rho] - 1.0) / rho if rho > 0 else 0.0
        v = np.maximum(v - theta, 0.0)
        v_sum = np.sum(v)
        if v_sum <= 1.0 + tol:
            # Normalize to sum=1, but it's already close to simplex
            if v_sum < 0:
                v_sum = 1.0
            v = v / v_sum
        else:
            # Project to simplex
            for i in range(v.size):
                if i <= rho:
                    v[i] -= theta
                else:
                    v[i] = 0.0
            v = v / np.sum(v)
        return v
    else:
        # All components are active
        pass_max = np.asforte to(v[:n-1], np.float64)
        pass_min = np.asforte to(v[1:], np.float64)
        total_sum = pass_max + pass_min
        if total_sum > 1.0 + tol:
            # Project out the minimum component
            index = np.argmin(v)
            v[index] = 0.0
            pass_min_sum = np.sum(pass_min)
            new_pass_min = pass_min / pass_min_sum * (1.0 - pass_max[index])
            v = pass_max.copy()
            v = v + new_pass_min
        return v

def _autocorr_for_score(seq):
    """Compute autocorrelation for score calculation."""
    # score 2*n*max(conv(seq, seq)) / (sum(seq))^2
    conv = _autocorr(seq)
    return float(conv.max())
