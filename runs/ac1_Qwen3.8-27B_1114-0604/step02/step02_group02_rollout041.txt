```
def _project_cap(c, T):
    c=np.asarray(c,float).copy()
    c=np.where(np.isfinite(c),c,0.0); c=np.maximum(c,0)
    s=c.sum(); L=c.size
    if s <= 1e-300:
        return np.ones(L)/L
    c=c/s
    avg=1.0/L
    if T <= avg:
        return np.ones(L)/L
    for _ in range(30):
        mask = c > T + 1e-15
        if not np.any(mask): break
        over_mask = mask
        excess = c[over_mask].sum() - T*np.count_nonzero(over_mask)
        if excess <= 0: break
        free = ~over_mask
        if not np.any(free):
            c = np.minimum(c,T); c = c/c.sum(); return c
        w = c[free]
        wsum=w.sum()
        if wsum > 1e-300:
            c[free] += excess * (w / wsum)
        else:
            c[free] += excess / float(np.count_nonzero(free))
        c[over_mask] = T
    c=np.maximum(c,0); csum=c.sum()
    if csum <=1e-300: return np.ones(L)/L
```

```
def _topk_grad(f, c, k):
   n=f.size; L=c.size
   if k <=0: return np.zeros(n)
   if k >= L:
      idx = np.arange(L)
   else:
      idx = np.argpartition(c, L-k)[-k:]
   grad=np.zeros(n)
   for idx_v in idx:
      idx_i = int(idx_v)
      lo = max(0, idx_i - n + 1)
      hi = min(n, idx_i + 1)
      j = np.arange(lo, hi)
      grad[j] += 2.0 * f[idx_i - j]
   grad /= float(k)
   return grad
```

```python
"""
Search for non-negative step-function coefficients minimizing

    score = 2 * n * max(np.convolve(a, a)) / (sum(a) ** 2)

The algorithm uses three complementary local/global mechanisms:

1. Diverse initializations:
   - the supplied construction if available,
   - ramps, powered/beta-like densities, skewed tents, exponentials,
   - low-frequency random smooth functions.

2. Spectral flattening:
   For the current squared convolution c = f * f, project c to a capped
   profile with a lower peak, take a phase-aligned frequency square root,
   inverse transform, project back to non-negative sequences, blend with the
   current state, and accept only if the peak decreases.

3. Entropy/subgradient mirror polish:
   Top-lag subgradients of the quadratic convolution objective are used in a
   softmax step, with simple line search and stochastic resets.

The full runtime is managed inside propose_candidate; the best valid sequence
found before the deadline is returned.
"""

import numpy as np
import time
import operator


def _fft_size(n):
    need = 2 * int(n) - 1
    if need <= 0:
        return 1
    return 1 << (need - 1).bit_length()


def _normalize(f):
    f = np.asarray(f, dtype=np.float64).ravel()
    f = np.where(np.isfinite(f), f, 0.0)
    f = np.maximum(f, 0.0)
    s = float(f.sum())
    n = f.size
    if n == 0:
        return np.ones(1, dtype=np.float64)
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return f / s


def _convolve_fft(f):
    f = np.asarray(f, dtype=np.float64).ravel()
    n = f.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    m = _fft_size(n)
    F = np.fft.rfft(f, m)
    c = np.fft.irfft(F * F, m)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    return c


def _convolve_from_F(F, n, m):
    c = np.fft.irfft(F * F, m)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    return c


def _blend_normalize(f, g, blend):
    g = np.asarray(g, dtype=np.float64).ravel()
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    s = float(g.sum())
    if s <= 1e-300:
        g = f
    else:
        g = g / s
    h = (1.0 - blend) * f + blend * g
    return _normalize(h)


def _random_perturb(f, rng, n, scale):
    if n <= 0:
        return f
    noise = (scale / np.sqrt(float(n))) * rng.standard_normal(n)
    return _normalize(f + noise)


def _project_cap(c, T):
    c = np.asarray(c, dtype=np.float64).copy()
    c = np.where(np.isfinite(c), c, 0.0)
    c = np.maximum(c, 0.0)
    s = float(c.sum())
    L = c.size
    if L == 0 or s <= 1e-300:
        return np.ones(max(L, 1), dtype=np.float64) / max(L, 1)

    c = c / s
    avg = 1.0 / float(L)
    if T <= avg:
        return np.ones(L, dtype=np.float64) / float(L)

    T = float(T)
    for _ in range(32):
        over = c > (T + 1e-15)
        if not np.any(over):
            break
        excess = float(c[over].sum()) - T * float(np.count_nonzero(over))
        if excess <= 0.0:
            break
        free = ~over
        if not np.any(free):
            c = np.minimum(c, T)
            s2 = c.sum()
            if s2 > 0.0:
                c = c / s2
            return c
        w = c[free]
        wsum = float(w.sum())
        if wsum > 1e-300:
            c[free] += excess * (w / wsum)
        else:
            c[free] += excess / float(np.count_nonzero(free))
        c[over] = T

    c = np.maximum(c, 0.0)
    s2 = float(c.sum())
    if s2 <= 1e-300:
        return np.ones(L, dtype=np.float64) / float(L)
    c = c / s2
    if c.max() > T:
        c = np.minimum(c, T)
        s3 = c.sum()
        if s3 > 0.0:
            c = c / s3
    return c


def _sqrt_aligned(C, F_ref, m):
    mag = np.sqrt(np.abs(C))
    ang = 0.5 * np.angle(C)

    if F_ref is not None and len(F_ref) == len(mag) and len(mag) > 0:
        ref_ph = np.angle(F_ref)
        diff = (ang - ref_ph + np.pi) % (2.0 * np.pi) - np.pi
        flip = np.abs(diff) > (0.5 * np.pi)
        ang = np.where(flip, ang + np.pi, ang)

    out = mag * np.exp(1j * ang)

    if len(out) > 0:
        out[0] = np.sqrt(max(float(np.real(C[0])), 0.0)) + 0j

    if m % 2 == 0 and len(out) > 0:
        out[-1] = float(np.real(out[-1])) + 0j

    return out


def _topk_grad(f, c, k):
    n = f.size
    L = c.size
    if n == 0 or L == 0 or k <= 0:
        return np.zeros(n, dtype=np.float64)

    if k >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, L - k)[-k:]

    grad = np.zeros(n, dtype=np.float64)
    for idx_v in idx:
        idx_i = int(idx_v)
        lo = max(0, idx_i - n + 1)
        hi = min(n, idx_i + 1)
        j = np.arange(lo, hi)
        grad[j] += 2.0 * f[idx_i - j]

    grad /= float(k)
    return grad


def _softmax_update(f, grad, eta):
    if f.size == 0:
        return f.copy()
    scale = float(np.max(np.abs(grad)))
    if scale < 1e-300:
        return f.copy()

    logf = np.log(f + 1e-300) - (eta / scale) * grad
    logf -= logf.max()
    g = np.exp(logf)
    s = float(g.sum())
    if s <= 1e-300:
        return f.copy()
    return g / s


def _make_starts(n, init_f, rng):
    starts = []

    if init_f is not None:
        starts.append(init_f.copy())
        if n > 11:

    for a in (0.5, 1.0, 2.0, 4.0):

        (1.0, 0.7),
        (1.4, 1.0),
        (1.0, 1.4),
        (1.7, 1.0),
        (1.0, 1.7),
    )

    xx = np.clip(x, 1e-6, 1.0 - 1e-6)
    for a, b in exponent_pairs:
        starts.append(xx ** (a - 1.0) * (1.0 - xx) ** (b - 1.0) + 1e-4)

    for a in (-4.0, -2.0, 0.0, 2.0, 4.0):
        for b in (-4.0, 0.0, 4.0):

    for p in (0.20, 0.35, 0.50, 0.65, 0.80):
        for w in (0.08, 0.15, 0.30, 0.50):
            starts.append(np.exp(-((x - p) / w) ** 2) + 1e-4)

    for _ in range(10):
        f = np.ones(n, dtype=np.float64)
        for m in range(1, 8):
            phase = rng.uniform(0.0, 2.0 * np.pi)
            amp = rng.uniform(0.0, 0.8 / float(m))
            f += amp * np.cos(2.0 * np.pi * m * x + phase)
        starts.append(f)

    for _ in range(5):
        starts.append(rng.exponential(1.0, n))

    out = []
    for f in starts:
        g = _normalize(f)
        if g.size == n and np.all(np.isfinite(g)) and float(g.sum()) > 1e-300:
            out.append(g)
    return out


def _sort_starts(scored):
    scored.sort(key=operator.itemgetter(0))
    return scored


def _optimize_spectral(f0, t_end, rng, n, floor_peak=None):
    f = _normalize(f0)
    m = _fft_size(n)
    F = np.fft.rfft(f, m)
    c = _convolve_from_F(F, n, m)
    cs = float(c.sum())
    if cs > 1e-300:
        c = c / cs
    peak = float(c.max())

    if not np.isfinite(peak) or peak <= 0.0:
        return f.copy(), 2.0 * float(n) * peak

    best_peak = peak
    best_f = f.copy()

    hard_floor = max(0.66 / float(n), 1.5 / float(2 * n - 1))
    if floor_peak is None:
        floor = peak * 0.9
    else:
        floor = float(floor_peak)
        if peak > 0.0:
            floor = min(floor, peak * 0.98)
    floor = max(hard_floor, floor)

    T = max(floor, peak * 0.95)
    stuck = 0
    last_improve = time.monotonic()
    iter_count = 0

    while time.monotonic() < t_end:
        if not np.isfinite(peak):
            break

        if peak < best_peak - 1e-15:
            best_peak = peak
            best_f = f.copy()
            last_improve = time.monotonic()

        if peak < floor and peak > 0.0:
            floor = max(hard_floor, peak * 0.97)

        if peak < T:
            T = max(floor, peak * 0.95)
        T = max(floor, T)

        c_proj = _project_cap(c, T)
        C = np.fft.rfft(c_proj, m)
        A = _sqrt_aligned(C, F, m)

        if not np.all(np.isfinite(A)):
            f = _random_perturb(best_f, rng, n, 0.03)
            F = np.fft.rfft(f, m)
            c = _convolve_from_F(F, n, m)
            peak = float(c.max())
            continue

        f_full = np.fft.irfft(A, m)
        f_next = f_full[:n]

        neg_frac = float(np.count_nonzero(f_next < -1e-12)) / float(n)
        out_mass = float(np.abs(f_full[n:]).sum())
        total_mass = float(np.abs(f_full).sum())
        out_frac = out_mass / total_mass if total_mass > 1e-300 else 1.0

        if neg_frac < 0.05 and out_frac < 0.10:
            blend = 0.70
        elif neg_frac < 0.15 and out_frac < 0.25:
            blend = 0.45
        elif neg_frac < 0.30 and out_frac < 0.50:
            blend = 0.25
        else:
            blend = 0.10

        f_new = _blend_normalize(f, f_next, blend)
        F_new = np.fft.rfft(f_new, m)
        c_new = _convolve_from_F(F_new, n, m)
        c_new_s = float(c_new.sum())
        if c_new_s > 1e-300:
            c_new = c_new / c_new_s
        p_new = float(c_new.max())

        if p_new < peak - 1e-15:
            f = f_new
            F = F_new
            c = c_new
            peak = p_new
            stuck = 0

            if p_new < best_peak:
                best_peak = p_new
                best_f = f.copy()
                last_improve = time.monotonic()

            if p_new < T:
                T = max(floor, min(p_new * 0.97, T * 0.995))
            else:
                T = max(floor, min(p_new * 0.97, T * 1.10))
        else:
            stuck += 1
            T = max(floor, min(peak * 0.99, T * 1.20))

            if stuck > 30 or (time.monotonic() - last_improve > 120.0 and rng.random() < 0.35):
                if rng.random() < 0.65:
                    f = _random_perturb(best_f, rng, n, 0.015)
                else:
                    f = _random_perturb(f, rng, n, 0.030)
                F = np.fft.rfft(f, m)
                c = _convolve_from_F(F, n, m)
                c_s2 = float(c.sum())
                if c_s2 > 1e-300:
                    c = c / c_s2
                peak = float(c.max())
                T = max(floor, peak * 0.85 if peak > 0.0 else peak)
                stuck = 0

        iter_count += 1
        if iter_count % 500 == 0:
            print(
                "    spectral iter {}: peak={:.8e} best={:.8e} T={:.8e}".format(
                    iter_count, peak, best_peak, T
                )
            )

    return best_f, 2.0 * float(n) * best_peak


def _optimize_mirror(f0, t_end, rng, n):
    f = _normalize(f0)
    c = _convolve_fft(f)
    c_s = float(c.sum())
    if c_s > 1e-300:
        c = c / c_s
    peak = float(c.max())

    if not np.isfinite(peak) or peak <= 0.0:
        return f.copy(), 2.0 * float(n) * peak

    best_peak = peak
    best_f = f.copy()
    eta = 0.25
    iter_count = 0
    stuck = 0
    last_improve = time.monotonic()

    while time.monotonic() < t_end:
        if not np.isfinite(peak):
            break

        c_s = float(c.sum())
        if c_s > 1e-300:
            c = c / c_s
        peak = float(c.max())

        if peak < best_peak - 1e-15:
            best_peak = peak
            best_f = f.copy()
            last_improve = time.monotonic()

        k = 4 + min(24, iter_count // 1000)
        grad = _topk_grad(f, c, k)
        scale = float(np.max(np.abs(grad)))
        if scale < 1e-20:
            f = _random_perturb(f, rng, n, 0.02)
            c = _convolve_fft(f)
            peak = float(c.max())
            iter_count += 1
            continue

        accepted = False
        for factor in (1.0, 0.5, 0.25):
            g = _softmax_update(f, grad, eta * factor)
            cg = _convolve_fft(g)
            cgs = float(cg.sum())
            if cgs > 1e-300:
                cg = cg / cgs
            pg = float(cg.max())

            if pg < peak - 1e-15:
                f = g
                c = cg
                peak = pg
                accepted = True
                eta = min(eta * 1.05, 3.0)
                break

        if accepted:
            stuck = 0
            if peak < best_peak:
                best_peak = peak
                best_f = f.copy()
                last_improve = time.monotonic()
        else:
            eta = max(0.03, eta * 0.85)
            stuck += 1

            if peak > best_peak * 1.03:
                f = _random_perturb(f, rng, n, 0.012)
                c = _convolve_fft(f)
                peak = float(c.max())
            elif rng.random() < 0.05:
                g = _softmax_update(f, grad, eta)
                f = g
                c = _convolve_fft(f)
                peak = float(c.max())

            if stuck > 40 or (time.monotonic() - last_improve > 120.0 and rng.random() < 0.35):
                if rng.random() < 0.55:
                    f = best_f.copy()
                f = _random_perturb(f, rng, n, 0.02)
                c = _convolve_fft(f)
                peak = float(c.max())
                eta = 0.25
                stuck = 0

        iter_count += 1
        if iter_count % 2000 == 0:
            print(
                "    mirror iter {}: peak={:.8e} best={:.8e}".format(
                    iter_count, peak, best_peak
                )
            )

    return best_f, 2.0 * float(n) * best_peak


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    budget_s = float(max(60.0, float(budget_s)))
    t0 = time.monotonic()
    deadline = t0 + budget_s - 12.0

    init_f = None
    try:
        init_seq = globals().get("height_sequence_1", None)
    except Exception:
        init_seq = None

    if init_seq is not None:
        try:
            arr = np.asarray(init_seq, dtype=np.float64).ravel()
            if arr.size > 0 and np.any(np.isfinite(arr)):
                arr = np.where(np.isfinite(arr), arr, 0.0)
                arr = np.maximum(arr, 0.0)
                n_init = arr.size

                if n_init > 10000:
                    target = 6121
                    x_old = np.linspace(0.0, 1.0, n_init)
                    x_new = np.linspace(0.0, 1.0, target)
                    arr = np.interp(x_new, x_old, arr)
                elif n_init < 512:
                    target = 6121
                    x_old = np.linspace(0.0, 1.0, n_init)
                    x_new = np.linspace(0.0, 1.0, target)
                    arr = np.interp(x_new, x_old, arr)

                init_f = _normalize(arr)
        except Exception:
            init_f = None

    if init_f is None:
        n = 6121
        init_f = np.ones(n, dtype=np.float64) / float(n)

    n = int(init_f.size)

    print("Starting search, length n =", n)
    start_c = _convolve_fft(init_f)
    start_peak = float(start_c.max())
    print("Initial construction peak: {:.8e}, score: {:.8f}".format(start_peak, 2.0 * n * start_peak))

    starts = _make_starts(n, init_f, rng)
    scored = []
    for f in starts:
        f = _normalize(f)
        c = _convolve_fft(f)
        c_s = float(c.sum())
        if c_s > 1e-300:
            c = c / c_s
        pk = float(c.max())
        if np.isfinite(pk) and pk > 0.0:
            scored.append((pk, f))

    if not scored:
        scored.append((start_peak, init_f))

    _sort_starts(scored)
    best_peak = float(scored[0][0])
    best_f = scored[0][1].copy()
    best_score = 2.0 * float(n) * best_peak

    print("Generated starts: {}".format(len(scored)))
    top_pk = []
    for pk, _ in scored[:5]:
        top_pk.append("{:.8e}".format(pk))
    print("Top start peaks: {}".format(",".join(top_pk)))
    print("Best initial score: {:.8f}".format(best_score))

    phase1_end = min(t0 + 0.45 * budget_s, deadline)
    phase2_end = min(t0 + 0.60 * budget_s, deadline)
    phase3_end = min(t0 + 0.80 * budget_s, deadline)
    phase4_end = min(t0 + 0.95 * budget_s, deadline)

    floorA = 0.745 / float(n)
    n_spec_starts = min(3, len(scored))
    phase1_start = time.monotonic()
    for i in range(n_spec_starts):
        planned = phase1_start + (phase1_end - phase1_start) * float(i + 1) / float(n_spec_starts)
        end = min(planned, phase1_end)
        if time.monotonic() >= max(end - 5.0, phase1_start):
            continue
        print("Spectral phase 1, start {} of {}, floor={:.8e}".format(i + 1, n_spec_starts, floorA))
        f_opt, score_opt = _optimize_spectral(scored[i][1], end, rng, n, floorA)
        if score_opt < best_score:
            best_score = score_opt
            best_f = f_opt.copy()
            print("  spectral improved to {:.8f}".format(best_score))

    if time.monotonic() < phase2_end - 5.0:
        print("Mirror polish, target deadline t={:.2f}".format(time.monotonic() + 0))
        f_opt, score_opt = _optimize_mirror(best_f, phase2_end, rng, n)
        if score_opt < best_score:
            best_score = score_opt
            best_f = f_opt.copy()
            print("  mirror improved to {:.8f}".format(best_score))

    floorB = 0.69 / float(n)
    if time.monotonic() < phase3_end - 5.0:
        print("Spectral phase 2, floor={:.8e}".format(floorB))
        f_opt, score_opt = _optimize_spectral(best_f, phase3_end, rng, n, floorB)
        if score_opt < best_score:
            best_score = score_opt
            best_f = f_opt.copy()
            print("  spectral phase 2 improved to {:.8f}".format(best_score))

    if time.monotonic() < phase4_end - 5.0:
        print("Final mirror polish")
        f_opt, score_opt = _optimize_mirror(best_f, phase4_end, rng, n)
        if score_opt < best_score:
            best_score = score_opt
            best_f = f_opt.copy()
            print("  final mirror improved to {:.8f}".format(best_score))

    if time.monotonic() < deadline - 8.0:
        floorC = max(0.67 / float(n), 0.995 * best_peak)
        print("Final spectral squeeze, floor={:.8e}".format(floorC))
        f_opt, score_opt = _optimize_spectral(best_f, deadline - 8.0, rng, n, floorC)
        if score_opt < best_score:
            best_score = score_opt
            best_f = f_opt.copy()
            print("  final spectral improved to {:.8f}".format(best_score))

    best_f = np.clip(best_f, 0.0, 1000.0)
    official = np.inf
    try:
        official = evaluate_sequence([float(x) for x in best_f])
        official = float(official)
    except Exception:
        official = np.inf

    if not np.isfinite(official):
        fallback = np.ones(n, dtype=np.float64)
        fb_off = np.inf
        try:
            fb_off = float(evaluate_sequence([float(x) for x in fallback]))
        except Exception:
            fb_off = np.inf

        if np.isfinite(fb_off):
            best_f = fallback
            official = fb_off
        else:
            official = 2.0

    if np.isfinite(official) and official < best_score:
        best_score = official

    print("Local best score: {:.8f}".format(best_score))
    print("Official score:   {:.8f}".format(official))
    print("Final returned length: {}".format(len(best_f)))

    return [float(x) for x in best_f]
```
