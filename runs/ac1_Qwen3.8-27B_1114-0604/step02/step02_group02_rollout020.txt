```python
import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def autocorr_fft(f):
    """Linear autocorrelation (self-convolution) of f via FFT."""
    n = len(f)
    size = 1 << (2 * n - 2).bit_length()
    F = rfft(f, size)
    result = irfft(F * F, size)
    return result[:2 * n - 1]

def solve_lp_direction(a, M, K_set):
    """
    Solve the LP: maximize sum(b) subject to conv(a,b)[k] <= M for k in K_set, b >= 0.
    Returns the optimal b.
    """
    n = len(a)
    # Objective: maximize sum(b) -> minimize -sum(b)
    c_obj = -np.ones(n)
    
    # Inequality constraints: conv(a, b)[k] <= M for k in K_set
    # conv(a, b)[k] = sum_j a[k-j] * b[j] (where 0 <= k-j < n)
    # So the constraint matrix A has A[k_idx, j] = a[k-j] if 0 <= k-j < n
    A_rows = []
    for k in K_set:
        row = np.zeros(n)
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        for j in range(lo, hi):
            row[j] = a[k - j]
        A_rows.append(row)
    
    if A_rows:
        A_ineq = np.array(A_rows)
        b_ineq = np.full(len(K_set), M)
    else:
        A_ineq = None
        b_ineq = None
    
    # Non-negativity: b >= 0, handled by bounds
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ineq, b_ub=b_ineq, bounds=bounds, method='highs')
    
    if result.status == 0:
        return result.x
    else:
        return None
```

"""
LP-based iterative minimization of 2*n*max(convolve(a,a))/(sum a)^2.
Uses a sequence of LP subproblems to find directions that reduce the
self-convolution peak, combined with line search and multiple strategies.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.fft import rfft, irfft
import time

def autocorr_fft(f):
    """Self-convolution of f via FFT."""
    n = len(f)
    size = 1 << (2*n - 2).bit_length()
    F = rfft(f, size)
    result = irfft(F * F, size)
    return result[:2*n-1]

def normalize(f):
    """Clip to non-negative and normalize to unit sum."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        return np.ones(len(f)) / len(f)
    return f / s

def compute_score(f):
    """Compute 2*n*max(conv)/sum^2."""
    n = len(f)
    c = autocorr_fft(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def solve_lp_direction(a, M, K_set):
    """
    Solve LP: max sum(b) s.t. conv(a,b)[k] <= M for k in K_set, b >= 0.
    Returns optimal b or None if infeasible.
    """
    n = len(a)
    c_obj = -np.ones(n)
    
    if len(K_set) == 0:
        return np.ones(n) / n
    
    # Build constraint matrix: A[k_idx, j] = a[k-j] if 0<=k-j<n
    A = np.zeros((len(K_set), n))
    for idx, k in enumerate(K_set):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        A[idx, lo:hi] = a[k - np.arange(lo, hi)][::-1]
    
    b_ub = np.full(len(K_set), M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 10})
    
    if res.status == 0:
        return res.x
    return None

def line_search(a, b, M, max_iter=20):
    """Find largest t in (0, 1] such that max(conv((1-t)a+tb, (1-t)a+tb)) <= M."""
    # Binary search
    lo, hi = 0.0, 1.0
    best_t = 0.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2
        a_t = (1 - mid) * a + mid * b
        c_t = autocorr_fft(a_t)
        if c_t.max() <= M * (1 + 1e-10):
            best_t = mid
            lo = mid
        else:
            hi = mid
    return best_t

    c = autocorr_fft(a)
    M = c.max()
    best_M = M
    best_a = a.copy()
    iter_count = 0
    stuck_count = 0
    
    while time.monotonic() < deadline:
        c = autocorr_fft(a)
        M = c.max()
        
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            stuck_count = 0
        else:
            stuck_count += 1
        
        # Choose K based on progress
        if stuck_count < 100:
            K = 50
        elif stuck_count < 500:
            K = 100
        elif stuck_count < 2000:
            K = 200
        else:
            K = 500
        
        K_set = np.argsort(c)[-min(K, 2*n-1)]
        
        # Solve LP
        b = solve_lp_direction(a, M, K_set)
        if b is None or b.sum() < 1e-10:
            # LP failed, try perturbation
            a = a + 0.001 / np.sqrt(n) * rng.standard_normal(n)
            a = normalize(a)
            stuck_count += 10
            continue
        
        # Normalize b to unit sum
        b = b / b.sum()
        
        # Line search
        if b.sum() > 1.000001 * a.sum():
            # b has more mass, so blending will change the shape
            t = line_search(a, b, M)
            if t > 1e-6:
                a_new = (1 - t) * a + t * b
                a = normalize(a_new)
            else:
                # Can't make progress with this direction
                stuck_count += 1
                if stuck_count > 1000:
                    # Perturb
                    a = a + 0.01 / np.sqrt(n) * rng.standard_normal(n)
                    a = normalize(a)
                    stuck_count = 0

            # b has similar or less mass, try anyway
            t = 0.1

            if c_new.max() < M:
                a = normalize(a_new)
