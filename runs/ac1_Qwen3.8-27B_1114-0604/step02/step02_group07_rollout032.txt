    Solve the LP: maximize sum(g) subject to g >= 0 and conv(a, g) <= M.
    Returns the optimal g.
    """

    # A_ub[k, i] = a[k-i] if 0 <= k-i < n
    # For fixed k: i ranges from max(0, k-n+1) to min(k, n-1)

    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):

            cols.append(i)
            vals.append(a[k - i])

"""
LP-based optimizer for minimizing 2n*max(conv(a,a))/(sum a)^2.
Uses iterative linear programming with lazy constraint generation as the main
optimization engine (fundamentally different from pure gradient descent),
combined with gradient-based polish for final refinement.

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv(a, b):
    """Linear convolution of a and b via FFT."""
    n = len(a) + len(b) - 1
    N = 1 << (n - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:n]


def compute_score(n, c, S):
    """Objective value given n, convolution array c, and sum S."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def solve_lp_active(p, M_p, active_lags):
    """Solve LP: max sum(g) s.t. conv(p,g)_k <= M_p for k in active_lags, g >= 0."""
    n = len(p)
    K = len(active_lags)
    if K == 0:
        return np.ones(n)

    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(active_lags):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        for i in range(i_start, i_end + 1):
            rows.append(idx)
            cols.append(i)
            vals.append(float(p[k - i]))

    if not rows:
        return np.ones(n)

    A = csr_matrix((vals, (rows, cols)), shape=(K, n))
    c_obj = -np.ones(n)
    b_ub = np.full(K, M_p)

    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')

    if res.success:
        return res.x
    return None


def lp_step(p, M_p, topK=60, max_rounds=10):
    """One LP step with lazy constraint generation. Returns g or None."""
    n = len(p)
    c_p = fast_self_conv(p)

    if topK < 2 * n - 1:
        active = sorted(np.argpartition(c_p, -topK)[-topK:].tolist())
    else:
        active = list(range(2 * n - 1))

    g = None
    for _ in range(max_rounds):
        g = solve_lp_active(p, M_p, active)
        if g is None:
            return None

        conv_pg = fast_conv(p, g)
        tol = M_p + 1e-9
        violated = np.where(conv_pg > tol)[0]

        if len(violated) == 0:
            return g

        active_set = set(active)
        new_lags = [int(k) for k in violated[:20] if int(k) not in active_set]
        if not new_lags:
            return g
        active.extend(new_lags)
        active = sorted(active)
        if len(active) > 400:
            active = sorted(np.argpartition(conv_pg, -400)[-400:].tolist())

    return g

    v = np.zeros(n)

    score = compute_score(n, c, S)

    stuck = 0

    while time.monotonic() < end_time:
        S = float(a.sum())
        if S < 1e-10:
            break

        score = compute_score(n, c, S)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 200:
                a = best_a * (1.0 + 0.03 * rng.standard_normal(n))

                m[:] = 0.0
                v[:] = 0.0
                t = 0
                lr = max(0.0005, lr * 0.7)
                stuck = 0
                continue

        k_act = min(topk, c.size)
        part = np.argpartition(c, -k_act)[-k_act:]
        vals = c[part].copy()
        order = np.argsort(vals)[::-1]
        part = part[order]
        vals = vals[order]

        top1 = float(vals[0])
        top2 = float(vals[1]) if k_act > 1 else 0.0
        ratio = top2 / top1 if top1 > 1e-15 else 1.0

        if ratio > 0.8:
            temp = max(0.05 * top1, 1e-15)
            wts = np.exp((vals[:min(k_act, 10)] - top1) / temp)

            active_idx = part[:min(k_act, 10)]
        else:

                continue
            grad_c[js:je + 1] += wi * 2.0 * a[js:je + 1][::-1]

        S2 = S * S
        S3 = S2 * S
        grad_obj = 2.0 * n * (grad_c / S2 - 2.0 * M / S3)

        t += 1
        t1 = 0.9 ** t
        t2 = 0.999 ** t
        m = 0.9 * m + 0.1 * grad_obj
        v = 0.999 * v + 0.001 * grad_obj * grad_obj
        a -= lr * (m / (1.0 - t1)) / (np.sqrt(v / (1.0 - t2)) + 1e-8)
        a = np.clip(a, 0.0, None)

        if verbose and t % 500 == 0:
            print(f"  [{label}] t={t} score={best_score:.6f} lr={lr:.6f}")

    return best_score, best_a


def init_guesses(n, rng):
    """Generate diverse initial guess sequences."""
    inits = []
    inits.append(np.ones(n))
    x = np.linspace(0, 1, n)
    inits.append(1.0 + 0.5 * np.cos(2.0 * np.pi * x))
    inits.append(1.0 + 0.3 * np.cos(4.0 * np.pi * x))
    inits.append(np.clip(0.3 + 0.7 * rng.random(n), 0.01, None))
    a_g = np.ones(n)
    a_g[1::2] = 0.4
    inits.append(a_g)
    inits.append(1.0 + 0.5 * (2.0 * x - 1.0))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum a)^2.
    Uses LP-based optimization with lazy constraints as main engine,
    combined with gradient-based polish.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = float(budget_s)
    deadline = start + total - 5.0

    n = 2048
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if parent.size > 0:
                n = int(parent.size)
    except Exception:
        pass
    n = min(max(256, n), 10000)
    print(f"n={n}, budget={budget_s:.0f}s, parent={'yes' if parent is not None else 'no'}")

    best_score = np.inf
    best_a = None

    # Phase 1: LP-based optimization on parent (55% of time)
    p1_end = start + 0.55 * (total - 5.0)

    if parent is not None and parent.size == n:
        a0 = np.clip(parent, 0.0, None)
        c0 = fast_self_conv(a0)
        s0 = compute_score(n, c0, float(a0.sum()))
        print(f"Parent score: {s0:.6f}")
        if np.isfinite(s0) and s0 < best_score:
            best_score = s0
            best_a = a0.copy()

        sc, aa = lp_optimize(a0, p1_end, rng, verbose=True, label="LP-parent")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 1 (LP parent): {best_score:.6f}")
    else:
        guesses = init_guesses(n, rng)
        per_time = (p1_end - start) / len(guesses)
        for i, a0 in enumerate(guesses):
            t_end = min(p1_end, start + per_time * (i + 1))
            if time.monotonic() >= p1_end - 1:
                break
            sc, aa = lp_optimize(a0, t_end, rng, verbose=False, label=f"LP-i{i}")
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = aa.copy()
            print(f"Phase 1 init {i}: {best_score:.6f}")

    # Phase 2: Gradient polish (15% of time)
    p2_end = start + 0.70 * (total - 5.0)
    if best_a is not None and time.monotonic() < p2_end:
        sc, aa = gradient_optimize(best_a, p2_end, rng, verbose=True, label="Grad1")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 2 (gradient): {best_score:.6f}")

    # Phase 3: Second LP refinement (15% of time)
    p3_end = start + 0.85 * (total - 5.0)
    if best_a is not None and time.monotonic() < p3_end:
        sc, aa = lp_optimize(best_a, p3_end, rng, verbose=True, label="LP-2")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 3 (LP 2nd): {best_score:.6f}")

    # Phase 4: Final gradient ultra-fine polish (remaining time)
    if best_a is not None and time.monotonic() < deadline - 2.0:
        sc, aa = gradient_optimize(best_a, deadline - 2.0, rng, verbose=True, label="Grad2")
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 4 (final grad): {best_score:.6f}")

    if best_a is None:
        best_a = np.ones(n)

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    verified = np.inf
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        pass
    print(f"FINAL: internal={best_score:.6f}, verified={verified:.6f}, n={len(out)}, "
          f"time={time.monotonic()-start:.1f}s")
    return out
