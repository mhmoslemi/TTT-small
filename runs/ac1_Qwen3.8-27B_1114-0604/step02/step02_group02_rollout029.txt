    n = len(f)

    # For numerical stability, subtract max

    c_scaled -= c_scaled.max()
    w = np.exp(c_scaled)

"""
Step function optimizer minimizing 2*n*max(conv(f,f))/(sum f)^2.
Uses FFT-based soft-max gradient with mirror descent, adaptive temperature,
diverse initializations, and simulated annealing for global search.
"""

import numpy as np
import time


def _fft_convolve(f):
    """Full linear self-convolution via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _compute_gradient(f, c, tau):
    """
    Compute soft-max gradient: grad[j] = 2 * sum_k w[k] * f[k-j]
    where w[k] = exp(c[k]/tau) / sum(exp(c/tau)).
    Uses FFT for O(n log n) computation.
    """
    n = len(f)
    # Soft-max weights (numerically stable)
    c_scaled = c / tau
    c_scaled -= c_scaled.max()
    w = np.exp(c_scaled)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # FFT-based convolution
    f_flip = f[::-1]
    size = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, size)
    FF = np.fft.rfft(f_flip, size)
    conv = np.fft.irfft(W * FF, size)
    
    # Extract: grad[j] = 2 * conv[n-1+j] for j = 0,...,n-1
    grad = 2.0 * conv[n - 1: 2 * n - 1]
    return grad


def _mirror_descent_step(f, grad, eta):
    """
    Mirror descent (exponentiated gradient) step on the probability simplex.
    f_new = exp(log(f) - eta * grad) / sum(exp(log(f) - eta * grad))
    """
    n = len(f)
    # Work in log space for numerical stability
    log_f = np.log(np.maximum(f, 1e-300))
    
    # Scale gradient for numerical stability
    grad_range = grad.max() - grad.min()
    if grad_range < 1e-15:
        return f
    
    log_f_new = log_f - (eta / grad_range) * (grad - grad.min())
    # Subtract max for numerical stability
    log_f_new -= log_f_new.max()
    f_new = np.exp(log_f_new)
    
    s = f_new.sum()
    if s < 1e-300:
        return f
    f_new /= s
    return f_new


def _optimize(f0, deadline, rng):
    """
    Optimize a single starting point using adaptive mirror descent
    with soft-max gradient and simulated annealing acceptance.
    """
    n = len(f0)
    f = f0.copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        return f0, np.inf
    f /= s
    
    c = _fft_convolve(f)
    peak = c.max()
    best_peak = peak
    best_f = f.copy()
    
    # Parameters
    tau = peak * 0.1  # Initial temperature for soft-max
    tau_min = peak * 1e-6
    eta = 0.5  # Step size
    T_sa = peak * 0.1  # Simulated annealing temperature
    T_sa_min = peak * 1e-10
    
    last_improve_time = time.monotonic()
    last_improve_peak = peak
    iter_count = 0
    
    while time.monotonic() < deadline:
        # Compute gradient
        grad = _compute_gradient(f, c, tau)
        
        # Mirror descent step
        f_new = _mirror_descent_step(f, grad, eta)
        
        # Evaluate
        c_new = _fft_convolve(f_new)
        new_peak = c_new.max()
        
        # Check improvement
        if new_peak < best_peak - 1e-15:
            best_peak = new_peak
            best_f = f_new.copy()
            last_improve_time = time.monotonic()
            last_improve_peak = best_peak
            eta = min(eta * 1.02, 2.0)  # Slightly increase step
        elif new_peak < peak:
            # Improvement but not new best
            f = f_new
            c = c_new
            peak = new_peak
            eta = min(eta * 1.01, 2.0)
        else:
            # No improvement - use SA or reject
            delta = new_peak - peak
            if T_sa > 1e-300:
                accept = np.log(max(rng.random(), 1e-300)) < -delta / T_sa
            else:
                accept = False
            
            if accept:
                f = f_new
                c = c_new
                peak = new_peak
            else:
                eta *= 0.9
                # Small random perturbation
                f = f + 0.001 / np.sqrt(n) * rng.standard_normal(n)
                f = np.maximum(f, 0)
                s = f.sum()
                if s > 1e-10:
                    f /= s
        
        # Adapt temperature
        stuck_time = time.monotonic() - last_improve_time
        if stuck_time > 20:
            # Reheat
            tau = peak * 0.05
            T_sa = peak * 0.05
            eta = max(eta, 0.5)
            # Larger perturbation
            mag = min(0.05, 0.01 + 0.005 * (stuck_time / 30.0))
            f = f + mag / np.sqrt(n) * rng.standard_normal(n)
            f = np.maximum(f, 0)
            s = f.sum()
            if s > 1e-10:
                f /= s
            last_improve_time = time.monotonic()
        else:
            tau = max(tau_min, tau * 0.999)
            T_sa = max(T_sa_min, T_sa * 0.998)
        
        iter_count += 1
        if iter_count % 10000 == 0:
            print(f"    iter {iter_count}: peak={peak:.10f}, best={best_peak:.10f}, "
                  f"tau={tau:.2e}, eta={eta:.3f}, T_sa={T_sa:.2e}")
    
    # Final check
    c_final = _fft_convolve(f)
    if c_final.max() < best_peak:
        best_peak = c_final.max()
        best_f = f.copy()
    
    return best_f, 2.0 * n * best_peak


def _polish(f, deadline, rng):
    """
    Coordinate descent polish: try moving mass between pairs of coordinates
    to reduce the peak convolution value.
    """
    n = len(f)
    f = f.copy()
    f = np.maximum(f, 0.0)
    f /= f.sum()
    
    c = _fft_convolve(f)
    best_peak = c.max()
    best_f = f.copy()
    
    passes = 0
    no_improve_count = 0
    max_no_improve = 200
    
    while time.monotonic() < deadline and no_improve_count < max_no_improve:
        c = _fft_convolve(f)
        cur_peak = c.max()
        
        if cur_peak < best_peak - 1e-15:
            best_peak = cur_peak
            best_f = f.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Find the peak lag(s)
        peak_lags = np.where(c > cur_peak * 0.999)[0]
        
        improved = False
        for k in peak_lags[:5]:  # Try top 5 peak lags
            # Contribution of coordinate j to c[k]: 2*f[k-j] (for valid k-j)
            lo = max(0, k - n + 1)
            hi = min(n, k + 1)
            contrib = np.zeros(n)
            contrib[lo:hi] = 2.0 * f[k - np.arange(lo, hi)]
            
            # Find high-contrib and low-contrib coordinates
            donor = int(np.argmax(contrib))
            # Try moving mass from donor to low-contrib positions
            low_positions = np.argsort(contrib)[:20]
            
            for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                for recv in low_positions:
                    if recv == donor:
                        continue
                    if f[donor] * frac < 1e-15:
                        continue
                    f_try = f.copy()
                    amt = f[donor] * frac
                    f_try[donor] -= amt
                    f_try[recv] += amt
                    s = f_try.sum()
                    if s < 1e-10:
                        continue
                    f_try /= s
                    c_try = _fft_convolve(f_try)
                    if c_try.max() < cur_peak - 1e-14:
                        f = f_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try random perturbations
            for _ in range(10):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                frac = rng.choice([0.01, 0.05, 0.1])
                amt = f[i] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[i] -= amt
                f_try[j] += amt
                if f_try[i] < -1e-15:
                    continue
                f_try[i] = max(0, f_try[i])
                s = f_try.sum()
                if s < 1e-10:
                    continue
                f_try /= s
                c_try = _fft_convolve(f_try)
                if c_try.max() < cur_peak - 1e-14:
                    f = f_try
                    improved = True
                    break
        
        passes += 1
        if passes % 100 == 0:
            print(f"    polish pass {passes}: peak={cur_peak:.10f}, best={best_peak:.10f}")
    
    return best_f, 2.0 * n * best_peak

            f_try = np.ones(n) + 0.5 * rng.standard_normal(n)
            f_try = np.maximum(f_try, 0)
        s = f_try.sum()
        if s > 1e-10:
