```python
import cvxpy as cp
import numpy as np

def solve_sdp(n, solver='CLARABEL'):
    X = cp.Variable((n+1, n+1), symmetric=True)
    t = cp.Variable()
    
    constraints = [X >> 0, cp.sum(X[:n, n]) == 1, X[n, n] == 1]
    for j in range(n):
        constraints.append(X[j, n] >= 0)
    
    for k in range(2*n - 1):
        # trace(X * P_k) <= t
        # P_k[j, l] = 1 if l = k - j (i.e., j + l = k) and j, l < n
        # In the augmented matrix, this involves X[:n, :n] only
        rows = []
        cols = []
        for j in range(n):
            l = k - j
            if 0 <= l < n:
                rows.append(j)
                cols.append(l)
        if rows:
            P_k = np.zeros((n, n))
            for j, l in zip(rows, cols):
                P_k[j, l] = 1.0
            # Augment to (n+1, n+1)
            P_aug = np.zeros((n+1, n+1))
            P_aug[:n, :n] = P_k
            constraints.append(cp.sum(X * P_aug) <= t)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=solver)
    
    u = X[:n, n].value
    u = np.clip(u, 0, None)
    u = u / u.sum()
    return u, t.value
```
