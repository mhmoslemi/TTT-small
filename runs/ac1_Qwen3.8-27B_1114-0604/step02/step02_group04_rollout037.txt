"""
Low-peak step-function optimization using Adam-optimized log-sum-exp smoothing
with Nesterov momentum, diverse multi-start, and coordinate-descent polishing.

import numpy as np
import time
import sys

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2*n-1]

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)

def _grad_conv(a, w):
    """Compute grad = 2 * cross_correlate(a, w) for w of length 2n-1."""
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]

def _project_simplex(v):
    """Project v onto the simplex {x: x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    # Algorithm from Duchi et al. 2008
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    """

    v = np.zeros(n)  # Second moment
    t = 0

def _coordinate_descent_polish(a, deadline):
    """
    Coordinate descent: optimize one element at a time using 1D bisection.
    """

    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size) / a.size
        s = 1.0
    a = a / s

    best_a = a.copy()
    best_score = _score(best_a)
    rng = np.random.RandomState(77)
    
    # Precompute FFT size
    N = _next_pow2(2 * n - 1)
    fa_cached = np.fft.rfft(a, N)
    
    # We'll do random coordinate descent
    while time.monotonic() < deadline:
        # Pick random coordinate
        m = rng.randint(n)
        
        # Current value
        f_m = a[m]
        
        # We want to find f_m in [0, 1 - sum_{j≠m} a_j] that minimizes max_k (f*f)_k
        # (f*f)_k = c_k + d_k * f_m + (1 if k==2m else 0) * f_m^2
        
        # Compute c_k and d_k
        # (f*f)_k = sum_j f_j * f_{k-j}
        # Terms not involving f_m: sum_{j≠m} f_j * f_{k-j} (with appropriate bounds)
        # This is expensive to compute directly. Instead, use FFT.
        
        # Alternative: just try a few candidate values and pick the best
        # Candidates: current value, 0, mean, and some perturbations
        candidates = [f_m, 0.0, a.mean(), 
                      f_m * 1.1, f_m * 0.9,
                      f_m + 0.001, f_m - 0.001]
        candidates = [max(0, min(c, 1.0 - (s - f_m))) for c in candidates]
        
        best_val = f_m
        best_obj = np.inf
        
        for val in set(candidates):  # deduplicate
            a[m] = val
            # Renormalize
            s_new = a.sum()
            if s_new < 0.01:
                a[m] = best_val
                continue
            a_temp = a / s_new
            c = _fft_convolve(a_temp)
            obj = np.max(c)
            if obj < best_obj:
                best_obj = obj
                best_val = val
            a[m] = f_m  # restore
        
        a[m] = best_val
        a = a / a.sum()
        
        # Periodic evaluation
        if rng.randint(10) == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score

"""
Low-peak step-function optimization using softmax-parameterized Adam with
log-sum-exp smoothing, Nesterov acceleration, multi-start diversity,
and active-set hard-max polishing. Key differences from prior approach:
(1) additive Adam in log-space (softmax parameterization) instead of
    multiplicative mirror descent, (2) much higher peak temperatures for
    sharper max-focusing, (3) Nesterov momentum phase, (4) structured
    and diverse initializations, (5) coordinate-descent final polish.
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    """Scale-invariant score matching evaluate_sequence."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _grad_cross(a, w):
    """
    Compute 2 * cross_correlate(a, w) where w has length >= n.
    Result has length n: result[m] = 2 * sum_k w[k] * a[k-m] (valid indices).
    """
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w[:N], N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]


def _lse_smoothed_grad(a, T):
    """
    Compute gradient of J_T(a) = (1/m) log(sum_k exp(T*(a*a)_k)) w.r.t. a.
    Returns (J_T, grad) where grad has length n.
    """
    c = _fft_convolve(a)
    m = c.size
    c_max = c.max()
    c_shift = c - c_max
    exp_Tc = np.exp(T * c_shift)
    S = exp_Tc.sum()
    if S < 1e-300:
        # Degenerate: put all weight on max
        w = np.zeros(m)
        w[np.argmax(c)] = 1.0
    else:
        w = exp_Tc / S
    J_T = c_max + np.log(S) / m
    # grad = (T/m) * sum_k w_k * 2*a_{k-m} = (T/m) * _grad_cross(a, w)
    raw_grad = _grad_cross(a, w)
    grad = (T / m) * raw_grad
    return J_T, grad


def _project_simplex(v):
    """Project v onto {x: x >= 0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v.copy()
    if np.all(v <= 0):
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rng = np.arange(1, n + 1)
    cond = u * rng > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _normalize(a):
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-15:
        return np.ones(a.size) / a.size
    return a / s


def _make_sine(n, amp, phase=0.5):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = 1.0 + amp * np.sin(theta)
    return _normalize(a)


def _make_cosine(n, amp, phase=0.0):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = 1.0 + amp * np.cos(theta)
    return _normalize(a)


def _make_multi_sine(n, modes):
    """modes: list of (amplitude, frequency, phase)."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for amp, freq, ph in modes:
        theta = 2.0 * np.pi * freq * (i + ph) / n
        a += amp * np.sin(theta)
    return _normalize(a)


def _make_linear(n, slope):
    i = np.arange(n, dtype=np.float64)
    t = 2.0 * i / max(1, n - 1) - 1.0
    a = 1.0 + slope * t
    return _normalize(a)


def _make_ramp_pair(n, ratio):
    """Ramp up on first half, ramp down on second, with a gap ratio."""
    half = n // 2
    a = np.zeros(n)
    a[:half] = np.linspace(0.01, 1.0, half)
    a[half:] = np.linspace(1.0, 0.01, n - half)
    a[:] = a ** (1.0 / ratio) if ratio > 0 else a
    return _normalize(a)


def _random_init(n, seed):
    rng = np.random.RandomState(seed)
    a = np.abs(rng.standard_normal(n))
    # Add some structure
    i = np.arange(n)
    a += 0.5 * (1 + np.sin(2 * np.pi * i / n))
    return _normalize(a)


def _optimize_adam_z(a_init, deadline, T_start=3.0, T_end=500.0,
                     lr_init=0.01, beta1=0.9, beta2=0.999,
                     eps=1e-8, patience=2000, eval_every=20):
    """
    Adam optimizer in softmax (z) parameterization.
    f = softmax(z), optimize over z unconstrained.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = _normalize(a)
    n = a.size

    # Initialize z from a: z = log(a) (clipped for safety)
    z = np.log(np.maximum(a, 1e-30))
    # Center z (gauge freedom)
    z -= z.mean()

    best_a = a.copy()
    best_score = _score(best_a)

    m_hat = np.zeros(n)
    v_hat = np.zeros(n)
    t = 0
    last_improve = t
    rng = np.random.RandomState(2024)

    total_iter_estimate = int((deadline - time.monotonic()) / 0.003) if deadline > time.monotonic() else 1000
    total_iter_estimate = max(1000, min(total_iter_estimate, 1000000))

    while time.monotonic() < deadline:
        t += 1
        # Temperature: exponential ramp
        frac = min(1.0, t / max(1, total_iter_estimate))
        T = T_start * (T_end / T_start) ** frac

        # Learning rate: decay
        lr = lr_init / (1.0 + 0.00005 * t)

        # Compute f = softmax(z)
        z_shift = z - z.max()
        exp_z = np.exp(z_shift)
        S = exp_z.sum()
        a = exp_z / S

        # Compute smoothed gradient w.r.t. a
        _, grad_a = _lse_smoothed_grad(a, T)

        # Convert to gradient w.r.t. z: dJ/dz_j = f_j * (g_j - sum_i f_i * g_i)
        fg = np.dot(a, grad_a)
        grad_z = a * (grad_a - fg)

        # Adam update
        m_hat = beta1 * m_hat + (1 - beta1) * grad_z
        v_hat = beta2 * v_hat + (1 - beta2) * grad_z ** 2
        m_hat_corr = m_hat / (1 - beta1 ** t)
        v_hat_corr = v_hat / (1 - beta2 ** t)

        z = z - lr * m_hat_corr / (np.sqrt(v_hat_corr) + eps)
        # Keep z bounded for numerical stability
        z = np.clip(z, -50.0, 50.0)
        z -= z.mean()

        # Evaluate periodically
        if t % eval_every == 0:
            # Recompute a from z
            z_shift2 = z - z.max()
            exp_z2 = np.exp(z_shift2)
            a = exp_z2 / exp_z2.sum()
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = t

            if t - last_improve > patience:
                # Restart: perturb z
                z = np.log(np.maximum(a, 1e-30))
                z -= z.mean()
                z += rng.standard_normal(n) * 0.1
                m_hat = np.zeros(n)
                v_hat = np.zeros(n)
                patience = max(1000, int(patience * 1.3))
                lr_init = max(0.001, lr_init * 0.85)

    return best_a, best_score


def _optimize_nesterov(a_init, deadline, T_start=20.0, T_end=800.0,
                       lr_init=0.1, momentum=0.8, patience=1500):
    """
    Nesterov accelerated gradient in simplex space with projection.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = _normalize(a)
    n = a.size

    best_a = a.copy()
    best_score = _score(best_a)

    v = np.zeros(n)
    t = 0
    last_improve = t
    rng = np.random.RandomState(31415)

    while time.monotonic() < deadline:
        t += 1
        T = T_start * (T_end / T_start) ** (min(1.0, t / 20000.0))
        lr = lr_init / (1.0 + 0.0005 * t)

        # Nesterov lookahead
        a_try = a + momentum * v
        a_try = _project_simplex(a_try)
        if a_try.sum() < 0.01:
            a_try = a.copy()

        _, grad = _lse_smoothed_grad(a_try, T)
        # Project to simplex tangent
        grad = grad - grad.mean()

        v = momentum * v - lr * grad
        a_new = a + v
        a_new = _project_simplex(a_new)

        if t % 15 == 0:
            sc = _score(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                last_improve = t
            if t - last_improve > patience:
                a = best_a.copy()
                v = np.zeros(n)
                noise = rng.standard_normal(n) * 0.003 * a.mean()
                a = _project_simplex(a + noise)
                patience = max(800, int(patience * 1.2))
                lr_init = max(0.01, lr_init * 0.8)

        a = a_new

    return best_a, best_score


def _optimize_hardmax(a_init, deadline, K=8, lr_init=0.3, patience=1000):
    """
    Hard-max top-K active set optimization in simplex space.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = _normalize(a)
    n = a.size

    best_a = a.copy()
    best_score = _score(best_a)

    t = 0
    last_improve = t
    rng = np.random.RandomState(55555)
    lr = lr_init

    while time.monotonic() < deadline:
        t += 1
        c = _fft_convolve(a)
        m_len = c.size

        if m_len <= K:
            top_idx = np.arange(m_len)
        else:
            top_idx = np.argpartition(c, -K)[-K:]

        # Soft weights within top-K
        top_vals = c[top_idx]
        top_shift = top_vals - top_vals.max()
        w_top = np.exp(2.0 * top_shift)
        w_top /= w_top.sum()

        # Build full weight vector
        w = np.zeros(m_len)
        for i, k in enumerate(top_idx):
            w[k] = w_top[i]

        # Gradient
        grad = _grad_cross(a, w)
        grad = grad - grad.mean()

        # Update with adaptive lr
        lr_curr = lr_init / (1.0 + 0.001 * t)
        a_new = a - lr_curr * grad
        a_new = _project_simplex(a_new)

        if t % 10 == 0:
            sc = _score(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                last_improve = t
            if t - last_improve > patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.002 * a.mean()
                a = _project_simplex(a + noise)
                patience = max(500, int(patience * 1.4))
                lr_init = max(0.05, lr_init * 0.75)

        a = a_new

    return best_a, best_score


def _coord_perturb(a_init, deadline, n_trials=5000):
    """
    Random single-coordinate perturbation search.
    Try changing one coordinate at a time and keep improvements.
    """
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = _normalize(a)
    n = a.size

    best_a = a.copy()
    best_score = _score(best_a)

    rng = np.random.RandomState(999)
    N = _next_pow2(2 * n - 1)

    # Precompute FFT of current a for fast updates
    # Actually, for small changes, we can update the FFT incrementally
    # But for simplicity, just recompute when needed

    it = 0
    last_improve = 0
    while time.monotonic() < deadline and it < n_trials:
        it += 1
        # Pick random coordinate
        m = rng.randint(n)
        old_val = a[m]

        # Try several perturbations
        candidates = [
            old_val * 1.05, old_val * 0.95,
            old_val + 0.002 * a.mean(), old_val - 0.002 * a.mean(),
            a.mean(), 0.0, old_val
        ]
        candidates = [max(0.0, c) for c in candidates]

        best_local = old_val
        best_local_score = np.inf

        # Only evaluate a subset for speed
        for val in candidates:
            if val == old_val:
                continue
            a[m] = val
            # Quick score check using FFT
            sc = _score(a)
            if sc < best_local_score:
                best_local_score = sc
                best_local = val
            a[m] = old_val

        a[m] = best_local
        a = _normalize(a)

        if it % 50 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it

    return best_a, best_score


def _generate_initializations(deadline):
    """Generate diverse initializations and return top candidates."""
    inits = []
    rng = np.random.RandomState(7)

    # Try to use previous best
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.clip(prev, 0, None)
        s = prev.sum()
        if s > 0.01:
            prev = prev / s
            sc = _score(prev)
            inits.append(('prev', prev, sc))
            print(f"  Previous: n={prev.size}, score={sc:.6f}", flush=True)
    except Exception:
        pass

    if time.monotonic() > deadline:
        return inits

    # Sinusoidal family
    for n in [512, 768, 1024, 1280, 1536, 2048]:
        if time.monotonic() > deadline:
            break
        for amp in [0.3, 0.5, 0.7, 0.9, 1.2]:
            for ph in [0.0, 0.25, 0.5]:
                a = _make_sine(n, amp, ph)
                sc = _score(a)
                inits.append((f'sine_{n}_{amp}_{ph}', a, sc))

        # Cosine
        for amp in [0.3, 0.6, 0.9]:
            a = _make_cosine(n, amp)
            sc = _score(a)
            inits.append((f'cos_{n}_{amp}', a, sc))

        # Linear
        for slope in [0.5, 1.0, 1.5, 2.0, 3.0]:
            a = _make_linear(n, slope)
            sc = _score(a)
            inits.append((f'lin_{n}_{slope}', a, sc))

        # Multi-sine (2 harmonics)
        for _ in range(3):
            amp1 = rng.uniform(0.3, 1.0)
            freq1 = rng.choice([1, 2, 3])
            ph1 = rng.uniform(0, 0.5)
            amp2 = rng.uniform(0.05, 0.4)
            freq2 = rng.choice([2, 3, 4, 5])
            ph2 = rng.uniform(0, 1.0)
            a = _make_multi_sine(n, [(amp1, freq1, ph1), (amp2, freq2, ph2)])
            sc = _score(a)
            inits.append((f'multisine_{n}', a, sc))

    # Random inits
    for n in [512, 1024, 2048]:
        if time.monotonic() > deadline:
            break
        for seed in range(5):
            a = _random_init(n, seed)
            sc = _score(a)
            inits.append((f'rand_{n}_{seed}', a, sc))

    # Sort and return top
    inits.sort(key=lambda x: x[2])
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for low-peak non-negative step function.
    Phases:
      1. Diverse initialization scan
      2. Adam in softmax space (bulk optimization)
      3. Nesterov momentum refinement
      4. Hard-max top-K active set polish
      5. Coordinate perturbation search
      6. Validation
    """
    t0 = time.monotonic()
    hard_deadline = t0 + float(budget_s) - 20.0

    np.random.seed(seed)
    print(f"[Start] Budget: {budget_s}s, deadline at t+{hard_deadline - t0:.0f}s", flush=True)

    # Phase 1: Initialization
    d1 = min(t0 + 45.0, hard_deadline - 100.0)
    print("[Phase 1] Generating initializations...", flush=True)
    inits = _generate_initializations(d1)
    if not inits:
        # Fallback
        a_fb = np.ones(1024) / 1024
        inits = [('uniform', a_fb, _score(a_fb))]
    print(f"  Best init: {inits[0][0]} = {inits[0][2]:.6f}", flush=True)

    best_a = inits[0][1].copy()
    best_score = inits[0][2]

    # Phase 2: Adam optimization from top-3 inits
    d2 = t0 + budget_s * 0.45
    remaining2 = max(0, d2 - time.monotonic())
    if remaining2 > 60:
        print(f"[Phase 2] Adam optimization ({remaining2:.0f}s)...", flush=True)
        n_starts = min(3, len(inits))
        time_per = remaining2 / n_starts - 5

        for i in range(n_starts):
            if time.monotonic() > d2 - 10:
                break
            name, a_init, sc_init = inits[i]
            d = time.monotonic() + time_per
            print(f"  Start {i+1}/{n_starts}: {name}={sc_init:.6f} ({time_per:.0f}s)", flush=True)

            a_opt, sc_opt = _optimize_adam_z(
                a_init, d,
                T_start=3.0, T_end=400.0,
                lr_init=0.01, beta1=0.9, beta2=0.999
            )
            print(f"    Result: {sc_opt:.6f}", flush=True)

            if sc_opt < best_score:
                best_score = sc_opt
                best_a = a_opt.copy()
    else:
        print("[Phase 2] Skipped (insufficient time)", flush=True)

    # Phase 3: Nesterov from best
    d3 = t0 + budget_s * 0.65
    remaining3 = max(0, d3 - time.monotonic())
    if remaining3 > 40:
        print(f"[Phase 3] Nesterov ({remaining3:.0f}s)...", flush=True)
        a_nest, sc_nest = _optimize_nesterov(
            best_a, d3,
            T_start=30.0, T_end=600.0,
            lr_init=0.08, momentum=0.8
        )
        print(f"  Nesterov: {sc_nest:.6f}", flush=True)
        if sc_nest < best_score:
            best_score = sc_nest
            best_a = a_nest.copy()

        # Also try Nesterov from a different init
        if len(inits) > 1 and remaining3 > 100:
            d3b = time.monotonic() + (d3 - time.monotonic()) * 0.5
            a_nest2, sc_nest2 = _optimize_nesterov(
                inits[1][1], d3b,
                T_start=20.0, T_end=400.0,
                lr_init=0.06, momentum=0.75
            )
            print(f"  Nesterov2: {sc_nest2:.6f}", flush=True)
            if sc_nest2 < best_score:
                best_score = sc_nest2
                best_a = a_nest2.copy()
    else:
        print("[Phase 3] Skipped", flush=True)

    # Phase 4: Adam again with higher T (focus on peak)
    d4 = t0 + budget_s * 0.80
    remaining4 = max(0, d4 - time.monotonic())
    if remaining4 > 40:
        print(f"[Phase 4] High-T Adam ({remaining4:.0f}s)...", flush=True)
        a_hi, sc_hi = _optimize_adam_z(
            best_a, d4,
            T_start=100.0, T_end=2000.0,
            lr_init=0.003, beta1=0.95, beta2=0.999
        )
        print(f"  High-T Adam: {sc_hi:.6f}", flush=True)
        if sc_hi < best_score:
            best_score = sc_hi
            best_a = a_hi.copy()
    else:
        print("[Phase 4] Skipped", flush=True)

    # Phase 5: Hard-max polish
    d5 = t0 + budget_s * 0.92
    remaining5 = max(0, d5 - time.monotonic())
    if remaining5 > 30:
        print(f"[Phase 5] Hard-max polish ({remaining5:.0f}s)...", flush=True)
        a_hm, sc_hm = _optimize_hardmax(
            best_a, d5, K=6, lr_init=0.15
        )
        print(f"  Hard-max: {sc_hm:.6f}", flush=True)
        if sc_hm < best_score:
            best_score = sc_hm
            best_a = a_hm.copy()
    else:
        print("[Phase 5] Skipped", flush=True)

    # Phase 6: Coordinate perturbation
    d6 = hard_deadline - 5
    remaining6 = max(0, d6 - time.monotonic())
    if remaining6 > 20:
        print(f"[Phase 6] Coordinate perturb ({remaining6:.0f}s)...", flush=True)
        a_cp, sc_cp = _coord_perturb(best_a, d6, n_trials=3000)
        print(f"  Coord perturb: {sc_cp:.6f}", flush=True)
        if sc_cp < best_score:
            best_score = sc_cp
            best_a = a_cp.copy()
    else:
        print("[Phase 6] Skipped", flush=True)

    # Phase 7: Final validation
    print("[Phase 7] Validating...", flush=True)
    arr_list = [float(max(0.0, min(1000.0, x))) for x in best_a]
    try:
        actual = float(evaluate_sequence(arr_list))
    except Exception:
        actual = best_score

    print(f"[Done] n={best_a.size}, internal={best_score:.8f}, actual={actual:.8f}", flush=True)
    return arr_list
