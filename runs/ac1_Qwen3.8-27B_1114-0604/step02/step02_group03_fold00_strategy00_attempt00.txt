def evaluate_sequence(seq):
    n = len(seq)
    # Defensive clipping to handle non-numbers, negatives, and too large/too small values
    clean_seq = [float(x) if isinstance(x, (int, float, bool)) else 0 for x in seq]
    clean_seq = [max(0, x) for x in clean_seq]
    total = sum(clean_seq)
    # If total is less than or equal to 10^-3, it's too small, return inf
    if total <= 1e-3:
        return np.inf
