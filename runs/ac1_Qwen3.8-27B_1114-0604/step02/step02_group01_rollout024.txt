def _multigrad(f, lags, weights):
    n = len(f)
    grad = np.zeros(n)
    for k, w in zip(lags, weights):
        js = max(0, k - n + 1)

        if js <= je:

def _mirror_step(f, grad, step):
    log_f = np.log(f + 1e-30) - step * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-30:
        return f.copy()
    return f_new / s

def _project(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return np.ones(len(f)) / len(f)
    return f / s

def _spectrum_flatten(f, strength=0.1):
    """Flatten the power spectrum to reduce autocorrelation peaks."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    P = np.abs(F) ** 2
    
    # Target: make P more uniform (excluding DC)
    P_target = P.copy()
    if len(P) > 1:
        P_ac = P[1:]
        mean_ac = P_ac.mean()
        if mean_ac > 1e-30:
            # Reduce peaks in spectrum, boost lows
            P_ac_new = P_ac + strength * (mean_ac - P_ac)
            P_ac_new = np.maximum(P_ac_new, 0)
            P_target[1:] = P_ac_new
    
    # Reconstruct: keep phase, adjust magnitude
    phases = np.angle(F)
    mags = np.sqrt(P_target)
    F_new = mags * np.exp(1j * phases)
    
    # Take real part of IFFT, keep non-negative
    f_new = np.fft.irfft(F_new, size)[:n]
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s < 1e-30:
        return f.copy()
    return f_new / s

def _adaptdescent(f0, n, deadline, rng, max_iters=50000):
    """Adaptive multi-peak mirror descent with backtracking."""
    f = _project(f0.copy())
    score = _score_fast(f)
    best_f = f.copy()
    best_score = score
    
    momentum = np.zeros(n)
    beta = 0.7
    step = 1.0
    K = 2
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Adaptive K
        K = min(2 + it // 5000, 8)
        
        c = _autocorr_fft(f)
        
        # Top K peaks
        if K < 2*n - 1:
            top_idx = np.argpartition(c, -K)[-K:]
        else:
            top_idx = np.arange(2*n-1)
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        top_vals = c[top_idx]
        
        # Water-filling weights: proportional to excess over median
        median_val = np.median(c)
        excess = np.maximum(top_vals - median_val, 0)
        if excess.sum() < 1e-30:
            weights = np.ones(K) / K
        else:
            weights = excess / excess.sum()
        
        grad = _multigrad(f, top_idx, weights)
        
        # Momentum
        momentum = beta * momentum + (1 - beta) * grad
        grad_eff = momentum
        
        # Mirror step
        f_new = _mirror_step(f, grad_eff, step)
        
        # Evaluate
        s_new = _score_fast(f_new)
        
        if s_new < score:
            f = f_new
            score = s_new
            if score < best_score:
                best_score = score
                best_f = f.copy()
            # Reduce step slightly on success
            step *= 0.99
        else:
            # Failed: reduce step
            step *= 0.7
            if step < 0.001:
                # Stagnation: try perturbation
                f = _perturb(f, rng)
                step = 0.5
                momentum *= 0.3
        
        # Periodic spectrum flattening
        if it % 10000 == 5000:
            f_fl = _spectrum_flatten(f, strength=0.05)
            s_fl = _score_fast(f_fl)
            if s_fl < score:
                f = f_fl
                score = s_fl
                if score < best_score:
                    best_score = score
                    best_f = f.copy()
    
    return best_f, best_score

def _perturb(f, rng):
    """Random perturbation for exploration."""
    n = len(f)

        alpha = rng.uniform(-2, 2)
        f_try = f * np.exp(alpha * (j/n - 0.5))

    elif mode == 3:

        amt = f[i1] * rng.uniform(0.1, 0.4)
        f_try[i1] -= amt
        f_try[i2] += amt

"""
Step function optimizer minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses reflection-antisymmetric initialization, adaptive multi-peak mirror
descent with backtracking, frequency-domain spectrum equalization,
coordinate polish, and diverse random restarts.

import numpy as np
import time


def _autocorr_fft(f):
    """Full linear convolution a*a via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _score_fast(f):
    """Fast score: 2*n*max(conv)/(sum)^2."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))


def _grad_at_lag(f, k):
    """Gradient of c[k]=sum_i f[i]*f[k-i] w.r.t. f."""
    n = len(f)
    grad = np.zeros(n)
    js = max(0, k - n + 1)
    je = min(n - 1, k)
    if js <= je:
        idx = k - np.arange(js, je + 1)
        grad[js:je + 1] = 2.0 * f[idx]
    return grad


def _multigrad(f, lags, weights):
    """Weighted sum of gradients at specified lags."""
    n = len(f)
    grad = np.zeros(n)
    for k, w in zip(lags, weights):
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js <= je:
            idx = k - np.arange(js, je + 1)
            grad[js:je + 1] += w * 2.0 * f[idx]
    return grad


def _mirror_step(f, grad, step):
    """Exponentiated gradient (mirror descent) step preserving non-negativity."""
    log_f = np.log(f + 1e-30) - step * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-30:
        return f.copy()
    return f_new / s


def _project(f):
    """Project onto non-negative, sum=1."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return np.ones(len(f)) / len(f)
    return f / s


def _spectrum_equalize(f, strength=0.05):
    """Flatten power spectrum to reduce autocorrelation peaks."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    P = np.abs(F) ** 2
    phases = np.angle(F)

    if len(P) > 1:
        P_ac = P[1:].copy()
        mean_ac = P_ac.mean()
        if mean_ac > 1e-30:
            # Compress the spectrum: reduce high-power bins, boost low-power bins
            P_ac = P_ac + strength * (mean_ac - P_ac)
            P_ac = np.maximum(P_ac, 0)

    mags = np.sqrt(np.concatenate([[P[0]], P_ac]))
    F_new = mags * np.exp(1j * phases)
    f_new = np.fft.irfft(F_new, size)[:n]
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s < 1e-30:
        return f.copy()
    return f_new / s


def _perturb(f, rng):
    """Apply a random structured perturbation."""
    n = len(f)
    mode = int(rng.integers(0, 7))
    j = np.arange(n, dtype=np.float64)
    f_try = f.copy()

    if mode == 0:
        freq = int(rng.integers(1, max(2, n // 4) + 1))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.1, 0.4)
        f_try = f * (1.0 + amp * np.sin(2 * np.pi * freq * j / n + phase))
    elif mode == 1:
        alpha = rng.uniform(-1.5, 1.5)
        f_try = f * np.exp(alpha * (j / n - 0.5))
    elif mode == 2:
        f_try = f + rng.normal(0, 0.08, size=n) * f.mean()
    elif mode == 3:
        c1, c2 = sorted(rng.uniform(0.1, 0.9, size=2))
        bw = rng.uniform(0.05, 0.2)
        boost = np.exp(-((j / n - c1) / bw) ** 2) + np.exp(-((j / n - c2) / bw) ** 2)
        f_try = f * (1.0 + 0.3 * boost)
    elif mode == 4:
        i1 = int(rng.integers(0, n // 2))
        i2 = int(rng.integers(n // 2, n))
        amt = f[i1] * rng.uniform(0.05, 0.3)
        f_try[i1] -= amt
        f_try[i2] += amt
    elif mode == 5:
        # Reflection-antisymmetric modulation (new!)
        m = int(rng.integers(1, max(2, n // 8))) * 2  # even m for antisymmetry
        alpha = rng.uniform(0.05, 0.3)
        r = np.sin(m * np.pi * j / max(n - 1, 1))
        f_try = f * (1.0 + alpha * r)
    else:
        # Multi-frequency
        f_try = f.copy()
        for _ in range(3):
            freq = int(rng.integers(1, max(2, n // 3) + 1))
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.05, 0.15)
            f_try = f_try * (1.0 + amp * np.sin(2 * np.pi * freq * j / n + phase))

    return _project(f_try)


def _adaptdescent(f0, n, deadline, rng, max_iters=50000, verbose=False):
    """
    Adaptive multi-peak mirror descent with:
    - Backtracking step size
    - Momentum
    - Periodic spectrum equalization
    - Stagnation detection with perturbation
    """
    f = _project(f0.copy())
    score = _score_fast(f)
    best_f = f.copy()
    best_score = score

    momentum = np.zeros(n)
    beta = 0.75
    step = 1.0
    stuck = 0
    patience = 15

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Adaptive K: start focused, broaden over time
        progress = it / max_iters
        K = min(2 + int(progress * 6), 8)

        c = _autocorr_fft(f)
        peak_val = c.max()

        # Top K peaks with water-filling weights
        total_lags = 2 * n - 1
        if K < total_lags:
            top_idx = np.argpartition(c, -K)[-K:]
        else:
            top_idx = np.arange(total_lags)
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        top_vals = c[top_idx]

        # Weights: emphasize peaks that exceed the median
        med = np.median(c)
        excess = np.maximum(top_vals - med, 0)
        if excess.sum() < 1e-30:
            weights = np.ones(len(top_idx)) / len(top_idx)
        else:
            weights = excess / excess.sum()

        grad = _multigrad(f, top_idx, weights)

        # Momentum update
        momentum = beta * momentum + (1.0 - beta) * grad

        # Mirror step
        f_new = _mirror_step(f, momentum, step)

        s_new = _score_fast(f_new)

        if s_new < score - 1e-12:
            f = f_new
            score = s_new
            stuck = 0
            if score < best_score:
                best_score = score
                best_f = f.copy()
                if verbose and it % 5000 == 0:
                    print(f"      iter {it}: {score:.7f}")
            # Slightly reduce step on success
            step = max(step * 0.995, 0.01)
        else:
            # Failed step: reduce
            step *= 0.6
            stuck += 1
            if step < 0.005 or stuck >= patience:
                # Stagnation: perturb and reset
                f = _perturb(f, rng)
                momentum *= 0.3
                step = 0.5
                stuck = 0

        # Periodic spectrum equalization
        if it > 0 and it % 5000 == 2500:
            f_eq = _spectrum_equalize(f, strength=0.03)
            s_eq = _score_fast(f_eq)
            if s_eq < score:
                f = f_eq
                score = s_eq
                if score < best_score:
                    best_score = score
                    best_f = f.copy()

    # Final check
    s_final = _score_fast(f)
    if s_final < best_score:
        best_score = s_final
        best_f = f.copy()

    return best_f, best_score


def _coordinate_polish(f, n, deadline, rng, max_iters=50000):
    """
    Coordinate descent: find best pair (donor, receiver) to transfer mass,
    using binary search for optimal amount.
    """
    f = f.copy()
    score = _score_fast(f)
    best_f = f.copy()
    best_score = score
    c = _autocorr_fft(f)
    peak = c.max()

    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Top 3 peaks
        top3 = np.argpartition(c, -3)[-3:]
        top3 = top3[np.argsort(c[top3])[::-1]]

        # Total contribution to top peaks from each position
        total_contrib = np.zeros(n)
        for k in top3:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx_k = k - np.arange(js, je + 1)
                total_contrib[js:je + 1] += f[js:je + 1] * f[idx_k]

        # Donor: highest contribution; Receiver: lowest
        donor = int(np.argmax(total_contrib))
        receiver = int(np.argmin(total_contrib))

        if donor == receiver or total_contrib[donor] < 1e-15:
            donor = int(rng.integers(0, n))
            receiver = int(rng.integers(0, n))
            if donor == receiver:
                no_improve += 1
                if no_improve > 50:
                    break
                continue

        # Binary search for best transfer amount
        best_amt = 0.0
        best_local = peak
        lo = 0.0
        hi = f[donor] * 0.8

        for _ in range(12):
            mid = (lo + hi) / 2.0
            f_try = f.copy()
            f_try[donor] -= mid
            f_try[receiver] += mid
            f_try = np.maximum(f_try, 0)
            s_t = f_try.sum()
            if s_t < 1e-10:
                hi = mid
                continue
            f_try /= s_t
            c_try = _autocorr_fft(f_try)
            if c_try.max() < best_local:
                best_local = c_try.max()
                best_amt = mid
                lo = mid
            else:
                hi = mid

        if best_amt > 1e-10:
            f[donor] -= best_amt
            f[receiver] += best_amt
            f = np.maximum(f, 0)
            f /= f.sum()
            c = _autocorr_fft(f)
            peak = c.max()
            score = _score_fast(f)
            no_improve = 0
            if score < best_score:
                best_score = score
                best_f = f.copy()
        else:
            no_improve += 1
            # Try a few random moves
            improved = False
            for _ in range(5):
                i = int(rng.integers(0, n))
                j2 = int(rng.integers(0, n))
                if i == j2:
                    continue
                amt = f[i] * 0.08
                if amt < 1e-12:
                    continue
                f_try = f.copy()
                f_try[i] -= amt
                f_try[j2] += amt
                f_try = _project(f_try)
                s_try = _score_fast(f_try)
                if s_try < score - 1e-12:
                    f = f_try
                    score = s_try
                    c = _autocorr_fft(f)
                    peak = c.max()
                    if score < best_score:
                        best_score = score
                        best_f = f.copy()
                    improved = True
                    no_improve = 0
                    break
            if not improved and no_improve > 100:
                break

    return best_f, best_score


def _fine_tune(f, n, deadline, rng, max_iters=30000):
    """Ultra-fine tuning with very small steps on the single peak."""
    f = f.copy()
    score = _score_fast(f)
    best_f = f.copy()
    best_score = score
    momentum = np.zeros(n)
    step = 0.02

    for it in range(max_iters):

            break

        c = _autocorr_fft(f)
        k_star = int(np.argmax(c))
        grad = _grad_at_lag(f, k_star)

        # Also include 2nd peak with smaller weight
        c_sorted_idx = np.argsort(c)[::-1]
        k2 = int(c_sorted_idx[1])
        w2 = c[k2] / max(c[k_star], 1e-30) * 0.3
        grad += w2 * _grad_at_lag(f, k2)

        momentum = 0.85 * momentum + 0.15 * grad
        f_new = _mirror_step(f, momentum, step)
        s_new = _score_fast(f_new)

        if s_new < score - 1e-13:
            f = f_new
            score = s_new
            step = max(step * 0.999, 0.001)
            if score < best_score:
                best_score = score
                best_f = f.copy()
        else:
            step *= 0.8
            if step < 0.0005:
                break

    s_final = _score_fast(f)
    if s_final < best_score:
        best_score = s_final
        best_f = f.copy()

        f = _project(f0)
        sc = _score_fast(f)
        start_results.append((sc, name, f))

    start_results.sort(key=lambda x: x[0])
    print("\nStarting point scores (top 10):")
    for sc, name, _ in start_results[:10]:
        print(f"  {name}: {sc:.7f}")

    if start_results[0][0] < best_score:
        best_score = start_results[0][0]
        best_f = start_results[0][2].copy()

    # Phase 1: Adaptive descent from top starts (60% time)
    print(f"\n=== Phase 1: Adaptive descent ===")
    phase1_end = time.monotonic() + (budget_s * 0.55)
    n_starts_to_try = 4
    time_per_start = (phase1_end - time.monotonic()) / n_starts_to_try

    for idx in range(n_starts_to_try):
        if time.monotonic() > phase1_end:
            break
        sc, name, f0 = start_results[idx]
        t_alloc = time_per_start * (1.0 - 0.12 * idx)
        d = time.monotonic() + t_alloc
        print(f"  '{name}' (init={sc:.6f}, budget={t_alloc:.0f}s)")
        f_opt, sc_opt = _adaptdescent(f0, n, d, rng, max_iters=40000, verbose=True)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"    -> NEW BEST: {best_score:.7f}")
        else:
            print(f"    -> Best: {sc_opt:.7f}")

    # Phase 2: Perturbed runs from best (20% time)
    print(f"\n=== Phase 2: Perturbed runs (score={best_score:.7f}) ===")
    phase2_end = min(deadline - 200, time.monotonic() + budget_s * 0.20)
    n_extra = 4
    t_each = (phase2_end - time.monotonic()) / n_extra
    for i in range(n_extra):
        if time.monotonic() > phase2_end or t_each < 30:
            break
        f_pert = _perturb(best_f, rng)
        d = time.monotonic() + t_each
        f_opt, sc_opt = _adaptdescent(f_pert, n, d, rng, max_iters=25000)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  extra{i}: NEW BEST {best_score:.7f}")
        else:
            print(f"  extra{i}: {sc_opt:.7f}")

    # Phase 3: Coordinate polish (12% time)
    print(f"\n=== Phase 3: Coordinate polish (score={best_score:.7f}) ===")
    phase3_end = min(deadline - 80, time.monotonic() + budget_s * 0.12)
    f_pol, sc_pol = _coordinate_polish(best_f, n, phase3_end, rng, max_iters=30000)
    if sc_pol < best_score:
        best_score = sc_pol
        best_f = f_pol.copy()
        print(f"  Polished to: {best_score:.7f}")
    else:
        print(f"  Polish result: {sc_pol:.7f}")

    # Phase 4: Fine tuning (7% time)
    print(f"\n=== Phase 4: Fine tuning (score={best_score:.7f}) ===")
    phase4_end = min(deadline - 30, time.monotonic() + budget_s * 0.07)
    f_fine, sc_fine = _fine_tune(best_f, n, phase4_end, rng, max_iters=25000)
    if sc_fine < best_score:
        best_score = sc_fine
        best_f = f_fine.copy()
        print(f"  Fine-tuned to: {best_score:.7f}")
    else:
        print(f"  Fine tune: {sc_fine:.7f}")

    # Phase 5: Final micro-polish with remaining time
    if time.monotonic() < deadline - 20:
        print(f"\n=== Phase 5: Final micro-polish ===")
        micro_end = deadline - 5.0
        f_micro, sc_micro = _coordinate_polish(best_f, n, micro_end, rng, max_iters=20000)
        if sc_micro < best_score:
            best_score = sc_micro
            best_f = f_micro.copy()
            print(f"  Micro-polished to: {best_score:.7f}")

    # Validate
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = evaluate_sequence([float(x) for x in best_f])
    print(f"\nOfficial score: {final_score:.7f}")
    if not np.isfinite(final_score) or final_score > best_score + 1e-9:
        best_score = final_score if np.isfinite(final_score) else _score_fast(best_f)

    print(f"\n=== FINAL SCORE: {best_score:.7f} ===")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")
    return [float(x) for x in best_f]
