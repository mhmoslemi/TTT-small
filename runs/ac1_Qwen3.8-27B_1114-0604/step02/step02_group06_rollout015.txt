def _sanitize(a, min_p=1e-15):
    a = np.asarray(a, dtype=float).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    a = a + min_p
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        return None
    return a / s

def _cross_corr(a, p, L2):
    """Compute corr[j] = sum_k a[k]*p[k+j] for j=0..n-1 via FFT."""
    Fa = np.fft.fft(a, L2)
    Fp = np.fft.fft(p, L2)
    corr = np.fft.ifft(np.conj(Fa) * Fp).real
    return corr

def _grad_objective(x, n, L, L2, beta):
    """Compute objective and gradient for log-sum-exp smoothed max."""
    x_shift = x - x.max()
    e = np.exp(x_shift)
    a = e / e.sum()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    Fp = np.fft.rfft(a, L)
    S = np.fft.irfft(Fp * Fp, L)
    S = S[:2 * n - 1]
    S = np.maximum(S, 0.0)

    # Log-sum-exp
    S_max = S.max()
    exp_beta_S = np.exp(beta * (S - S_max))
    log_sum_exp = S_max + np.log(exp_beta_S.sum()) / beta
    val = 2.0 * n * log_sum_exp

    # Softmax weights on S
    p = exp_beta_S / exp_beta_S.sum()

    # Cross-correlation: corr[j] = sum_k a[k] * p[k+j]
    corr = _cross_corr(a, p, L2)
    corr_n = corr[:n]

    # Mean of S under p
    S_bar = float(np.dot(p, S))

    # Gradient: 4n * a * (corr - S_bar)
    grad = 4.0 * n * a * (corr_n - S_bar)

    return val, grad

def _grad_optimize(a0, time_limit, beta_start=10.0, beta_end=20000.0, verbose=False):
    """L-BFGS-B optimization with increasing beta annealing."""
    n = a0.size
    L = _conv_len(n)
    L2 = 1
    while L2 < 3 * n - 2:
        L2 <<= 1

    a0 = _sanitize(a0)
    if a0 is None:
        return a0, float('inf')

    x0 = np.log(a0)
    best_score = float('inf')
    best_x = x0.copy()

    deadline = time.monotonic() + time_limit
    beta = beta_start
    step = 0

    while time.monotonic() < deadline and beta <= beta_end:
        remaining = deadline - time.monotonic()
        if remaining < 5.0:
            break

        # Determine max iterations based on remaining time
        maxiter = int(min(300, max(20, remaining * 50)))

        def obj_and_grad(x):
            return _grad_objective(x, n, L, L2, beta)

        try:
            res = minimize(obj_and_grad, x0, method='L-BFGS-B',
                          jac=True, options={'maxiter': maxiter, 'gtol': 1e-7,
                                             'ftol': 1e-9, 'maxls': 30})
            x0 = res.x
        except Exception:
            pass

        # Evaluate true score
        a_final = np.exp(x0 - x0.max())
        a_final = np.maximum(a_final, 1e-15)
        a_final = a_final / a_final.sum()
        sc, _ = _score(a_final)
        if sc < best_score:
            best_score = sc
            best_x = x0.copy()
            if verbose:
                print(f"  grad opt: beta={beta:.0f}, score={sc:.7f}, time_left={remaining:.1f}s")

        # Increase beta
        if beta < 100:
            beta *= 2.5
        elif beta < 1000:
            beta *= 2.0
        else:
            beta *= 1.8

        step += 1

    a_best = np.exp(best_x - best_x.max())
    a_best = np.maximum(a_best, 1e-15)
    a_best = a_best / a_best.sum()
    final_sc, _ = _score(a_best)
    if final_sc < best_score:
        best_score = final_sc
    return a_best, best_score

def _mmd_optimize(a0, time_limit, target=1.50, verbose=False):
    """Improved MMD multiplicative update optimization."""
    a = _sanitize(a0)
    if a is None:
        return a0, float('inf')
    n = a.size
    L = _conv_len(n)

    best = a.copy()
    best_sc, _ = _score(a)
    cur_sc = best_sc

    deadline = time.monotonic() + time_limit
    alpha = 0.1
    eps = 0.1
    beta_sm = 30.0
    stall = 0
    it = 0

    target_m = target / (2.0 * n)

    while time.monotonic() < deadline:
        Fp = np.fft.rfft(a, L)
        c = np.fft.irfft(Fp * Fp, L)
        c = c[:2 * n - 1]
        c = np.maximum(c, 0.0)
        maxc = c.max()
        cur_sc = 2.0 * n * maxc

        if cur_sc < best_sc:
            best_sc = cur_sc
            best = a.copy()
            if verbose and it % 500 == 0:
                print(f"  mmd: n={n}, it={it}, best={best_sc:.7f}")

        # Compute update direction
        if maxc > target_m * 1.001:
            # Threshold-based: penalize values above threshold
            thresh = (1.0 - eps) * maxc + eps * target_m
            viol = np.maximum(0.0, c - thresh)
            maxviol = viol.max()
            if maxviol > 1e-14 * maxc:
                w = viol / maxviol
                w = w * (0.3 + 0.7 * w)
                w = w * w
            else:
                # Softmax fallback
                d = n * c
                dm = d - d.max()
                w = np.exp(50.0 * dm)
                w = w / w.sum()
        else:
            # Below target: use softmax on full c
            d = n * c
            dm = d - d.max()
            w = np.exp(beta_sm * dm)
            w = w / w.sum()

        # Compute update: h_j = sum_m w_m * a_{m-j} (cross-correlation of w and a)
        L2 = 1
        while L2 < 3 * n - 2:
            L2 <<= 1
        Fw = np.fft.fft(w, L2)
        Fa = np.fft.fft(a, L2)
        h = np.fft.ifft(np.conj(Fa) * Fw).real[:n]
        h = np.maximum(h, 0.0)
        hm = h.max()

        if hm < 1e-300:
            # Fallback: random perturbation
            a = a * np.exp(_rng.normal(0, 0.05, n))
            a = _sanitize(a)
            if a is None:
                a = best.copy()
            continue
        h = h / hm

        # Try multiple step sizes
        best_trial = None
        best_trial_sc = cur_sc
        for mult in (0.3, 0.5, 1.0, 1.5, 2.0, 3.0):
            a_step = alpha * mult
            a_try = a * np.exp(-a_step * h)
            a_try = np.maximum(a_try, 1e-15)
            a_try = a_try / a_try.sum()
            sc_try, _ = _score(a_try)
            if sc_try < best_trial_sc - 1e-12:
                best_trial_sc = sc_try
                best_trial = a_try

        if best_trial is not None:
            a = best_trial
            cur_sc = best_trial_sc
            stall = 0
            alpha = min(1.0, alpha * 1.02)
        else:
            alpha = max(0.02, alpha * 0.8)
            stall += 1
            eps = min(1.0, eps * 1.01)

            if stall >= 15:
                std = min(0.3, 0.05 + 0.005 * stall)
                a_try = a * np.exp(_rng.normal(0, std, n))
                a_try = _sanitize(a_try)
                if a_try is not None:
                    sc_try, _ = _score(a_try)
                    if sc_try <= max(best_sc * 1.02, cur_sc * 1.05):
                        a = a_try
                        cur_sc = sc_try
                stall = 0

            if stall > 100:
                a = best.copy()
                cur_sc = best_sc
                alpha = 0.1
                eps = 0.1
                beta_sm = max(20.0, beta_sm * 0.5)
                stall = 0

        if it % 50 == 0:
            beta_sm = min(500.0, beta_sm * 1.1)

        it += 1

    return best, best_sc

def _local_search(a0, time_limit, verbose=False):
    """Local block scaling search."""
    a = _sanitize(a0)
    if a is None:
        return a0, float('inf')
    n = a.size
    L = _conv_len(n)

    best = a.copy()
    best_sc, _ = _score(a)
    cur_sc = best_sc

    deadline = time.monotonic() + time_limit
    stagnant = 0
    it = 0

    block_sizes = sorted(set([
        max(1, n // 4096),
        max(2, n // 1024),
        max(4, n // 256),
        max(8, n // 64),
        max(16, n // 16),
        max(32, n // 4),
        max(64, n // 2),
    ]))

    factors = (0.5, 0.7, 0.85, 1.2, 1.5, 2.0, 0.3, 3.0)

    while time.monotonic() < deadline:
        r = _rng.random()
        if r < 0.2:
            w = 1
        elif r < 0.5:
            w = block_sizes[int(_rng.integers(0, len(block_sizes) // 2 + 1))]
        elif r < 0.85:
            w = block_sizes[len(block_sizes) // 2 + int(_rng.integers(0, len(block_sizes) // 2))]
        else:
            w = block_sizes[-1]
        w = max(1, min(n, w))
        start = int(_rng.integers(0, n - w + 1))

        best_f = None
        best_f_sc = cur_sc

        for f in factors:
            trial = a.copy()
            trial[start:start + w] *= f
            trial = np.maximum(trial, 1e-15)
            trial = trial / trial.sum()
            sc, _ = _score(trial)
            if sc < best_f_sc - 1e-12:
                best_f_sc = sc
                best_f = trial

        if best_f is not None:
            a = best_f
            cur_sc = best_f_sc
            stagnant = 0
            if cur_sc < best_sc:
                best_sc = cur_sc
                best = a.copy()
                if verbose and it % 200 == 0:
                    print(f"  local: n={n}, it={it}, best={best_sc:.7f}")
        else:
            stagnant += 1
            if stagnant > 30:
                std = min(0.2, 0.03 + 0.003 * stagnant)
                a_try = a * np.exp(_rng.normal(0, std, n))
                a_try = _sanitize(a_try)
                if a_try is not None:
                    sc_try, _ = _score(a_try)
                    if sc_try <= max(best_sc * 1.01, cur_sc * 1.03):
                        a = a_try
                        cur_sc = sc_try
                stagnant = 0
            if stagnant > 200:
                a = best.copy()
                cur_sc = best_sc
                stagnant = 0

        it += 1

    return best, best_sc

def _resample(a, m):
    """Resample sequence to length m."""
    a = _sanitize(a, 1e-10)
    if a is None:
        return None
    n = a.size
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m) / m
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(m) + 0.5) / m
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0) + 1e-12
    return _sanitize(out)

def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    global _rng

    t_start = time.monotonic()
    budget = float(budget_s) if budget_s else 2400.0
    if not np.isfinite(budget) or budget < 60:
        budget = 2400.0
    total = max(60.0, budget - 30.0)
    deadline = t_start + total
    _init_rng(int(seed) % 1000000007 + 1)

    print(f"Starting optimization with budget={total:.0f}s, seed={seed}")

    # Try to get parent sequence
    parent = None
    try:
        parent_raw = height_sequence_1
        p = _sanitize(parent_raw)
        if p is not None:
            parent = p
            sc_parent, _ = _score(p)
            print(f"Parent sequence: n={p.size}, score={sc_parent:.7f}")
    except (NameError, Exception):
        pass

    best_seq = None
    best_score = float('inf')

    def update_best(a, sc):
        nonlocal best_seq, best_score
        if a is not None and np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_seq = a.copy()
            print(f"  New best: score={sc:.7f}, n={a.size}")

    if parent is not None:
        update_best(parent, best_score if best_score < float('inf') else _score(parent)[0])
        s, _ = _score(parent)
        update_best(parent, s)
    else:
        # Create initial sequence
        n0 = 2048
        a0 = np.ones(n0) / n0
        update_best(a0, _score(a0)[0])

    # Phase 1: Multi-scale MMD
    print("\n=== Phase 1: Multi-scale MMD ===")
    scales = [1024, 2048, 4096]
    if total > 1500:
        scales.append(8192)

    time_alloc_mmd = 0.55 * total
    per_scale = time_alloc_mmd / len(scales)

    for n_scale in scales:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(per_scale, rem - 20)
        if t_alloc < 15:
            continue

        # Resample current best to this scale
        if best_seq is not None and best_seq.size != n_scale:
            a_start = _resample(best_seq, n_scale)
        elif best_seq is not None:
            a_start = best_seq.copy()
        else:
            a_start = np.ones(n_scale) / n_scale

        if a_start is None:
            a_start = np.ones(n_scale) / n_scale

        # Run MMD
        print(f"  MMD at n={n_scale}, time={t_alloc:.0f}s")
        a_out, sc_out = _mmd_optimize(a_start, t_alloc, target=1.50)
        if a_out is not None:
            s, _ = _score(a_out)
            update_best(a_out, s)

    # Phase 2: Gradient-based refinement
    print("\n=== Phase 2: Gradient-based refinement (L-BFGS-B) ===")
    time_alloc_grad = 0.25 * total
    rem = deadline - time.monotonic()
    t_grad = min(time_alloc_grad, rem - 30)
    if t_grad > 20 and best_seq is not None:
        # Try gradient optimization on current best
        n_work = best_seq.size
        print(f"  Gradient optimization at n={n_work}, time={t_grad:.0f}s")

        # Multiple restarts
        n_restarts = 3 if t_grad > 60 else 1
        t_per_restart = t_grad / n_restarts

        for restart in range(n_restarts):
            rem = deadline - time.monotonic()
            if rem < 15:
                break
            t_this = min(t_per_restart, rem - 10)
            if t_this < 10:
                break

            if restart == 0:
                a_start = best_seq.copy()
            else:
                # Perturbed restart
                noise_level = 0.08 + 0.04 * restart
                a_start = best_seq * np.exp(_rng.normal(0, noise_level, best_seq.size))
                a_start = _sanitize(a_start)
                if a_start is None:
                    a_start = best_seq.copy()

            a_out, sc_out = _grad_optimize(a_start, t_this,
                                           beta_start=10.0, beta_end=20000.0,
                                           verbose=True)
            if a_out is not None:
                s, _ = _score(a_out)
                update_best(a_out, s)

    # Phase 3: Local search
    print("\n=== Phase 3: Local search ===")
    rem = deadline - time.monotonic()
    t_local = min(0.12 * total, rem - 30)
    if t_local > 15 and best_seq is not None:
        print(f"  Local search at n={best_seq.size}, time={t_local:.0f}s")
        a_out, sc_out = _local_search(best_seq, t_local, verbose=True)
        if a_out is not None:
            s, _ = _score(a_out)
            update_best(a_out, s)

    # Phase 4: Final gradient polish
    print("\n=== Phase 4: Final gradient polish ===")
    rem = deadline - time.monotonic()
    t_polish = min(60.0, rem - 20)
    if t_polish > 10 and best_seq is not None:
        print(f"  Final polish, time={t_polish:.0f}s")
        a_out, sc_out = _grad_optimize(best_seq, t_polish,
                                       beta_start=500.0, beta_end=50000.0,
                                       verbose=True)
        if a_out is not None:
            s, _ = _score(a_out)
            update_best(a_out, s)

    # Phase 5: Try a couple different sequence lengths
    print("\n=== Phase 5: Length exploration ===")
    rem = deadline - time.monotonic()
    if rem > 60 and best_seq is not None:
        for n_try in [2048, 4096, 6144]:
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            if n_try == best_seq.size:
                continue
            a_res = _resample(best_seq, n_try)
            if a_res is None:
                continue
            t_this = min(30.0, rem - 20)
            if t_this < 10:
                break
            a_out, sc_out = _grad_optimize(a_res, t_this,
                                           beta_start=100.0, beta_end=20000.0)
            if a_out is not None:
                s, _ = _score(a_out)
                update_best(a_out, s)

    # Final verification
    print(f"\n=== Final Result ===")
    print(f"Best score: {best_score:.7f}, n={best_seq.size}")

    final = best_seq.copy()
    final = np.clip(final, 0.0, 1000.0)
    s = final.sum()
    if s <= 0:
        final = np.ones(2048) / 2048
    else:
        final = final / s

    # Verify with evaluate_sequence if available and sequence is small enough
    if final.size <= 8192:
        try:
            eval_score = evaluate_sequence([float(x) for x in final])
            if np.isfinite(eval_score):
                print(f"Verified score via evaluate_sequence: {eval_score:.7f}")
        except Exception as e:
            print(f"Verification failed: {e}")

    # Check time
    elapsed = time.monotonic() - t_start
    print(f"Total time used: {elapsed:.1f}s / {total:.0f}s")

    return [float(x) for x in final]
