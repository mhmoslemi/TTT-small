    # For i in [lo, hi), k-i is in [0, n-1]
    grad[lo:hi] = 2.0 * p[k - np.arange(lo, hi)]
    return grad

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _eval_score(p):
    """Compute the score for a probability vector p (sum=1)."""
    n = len(p)
    c = _autocorr_fft(p)
    return 2.0 * n * c.max()

def _optimize_subgrad(p0, deadline, rng):
    """Subgradient descent with top-k gradient and line search."""
    n = len(p0)
    p = p0.copy()
    p = _project_simplex(p)
    
    best_score = _eval_score(p)
    best_p = p.copy()
    
    eta = 0.01  # step size
    T = 0.1     # temperature for SA
    k_top = 3   # number of top lags to use
    iter_count = 0
    last_improve = time.monotonic()
    
    while time.monotonic() < deadline:
        c = _autocorr_fft(p)
        cur_peak = c.max()
        cur_score = 2.0 * n * cur_peak
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            last_improve = time.monotonic()
        
        stuck = time.monotonic() - last_improve
        
        # Adaptive parameters
        if stuck > 20:
            # Reheat
            T = cur_peak * 0.1
            eta = 0.02
            # Perturb
            p = p + 0.01 * rng.standard_normal(n) / np.sqrt(n)
            p = _project_simplex(p)
            last_improve = time.monotonic()
        else:
            T *= 0.999
            eta *= 0.9995
        
        # Compute top-k gradient
        top_k = np.argsort(c)[-k_top:]
        grad = np.zeros(n)
        for idx in top_k:
            grad += _grad_lag(p, idx)
        grad /= len(top_k)
        
        # Mirror descent step (exponentiated gradient)
        log_p = np.log(p + 1e-300) - eta * grad
        log_p -= log_p.max()
        p_new = np.exp(log_p)
        p_new /= p_new.sum()
        
        # Check improvement
        c_new = _autocorr_fft(p_new)
        new_score = 2.0 * n * c_new.max()
        
        if new_score <= cur_score:
            p = p_new
            eta *= 1.01  # slightly increase step size
        else:
            # SA acceptance
            delta = (new_score - cur_score) / max(cur_score, 1e-10)
            if np.log(max(rng.uniform(), 1e-300)) < -delta / T:
                p = p_new
            else:
                eta *= 0.9
        
        iter_count += 1
        if iter_count % 5000 == 0:
            print(f"    Subgrad iter {iter_count}: score={cur_score:.8f}, best={best_score:.8f}, eta={eta:.5f}")
    
    return best_p, best_score

- The decrease is balanced (all q[ki] decrease by the same amount)
- The step is small enough to stay valid

where g_i = gradient of q[ki] = 2*p[ki - j] (as a vector in j).

We want to minimize t subject to g_i^T d <= t - q[ki] for all i.
This is a linear program! But with n=6121 variables, it might be slow.

The solution is d = -G^T (G G^T)^{-1} * 1.

This is efficient since m is small (3-10), so G G^T is at most 10×10.
