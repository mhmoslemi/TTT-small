on the simplex with softmax-smoothed gradient, adaptive temperature,
backtracking line search, and diverse restarts.

import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def proj_simplex(x):
    """Project x onto the simplex {y >= 0, sum(y) = 1}."""
    n = len(x)
    if n == 0:
        return x
    u = np.sort(x)[::-1]
    css = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    cond = u * rho_candidates > (css - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(rho_candidates[cond])
    theta = (css[rho - 1] - 1.0) / rho
    return np.maximum(x - theta, 0.0)


def smooth_max_and_grad(c, T, topk=50):
    """
    Compute softmax-smoothed max of c and the gradient contribution.
    Returns (smooth_max, weighted_indices, weights).
    """
    n_c = len(c)
    if T <= 0:
        idx = int(np.argmax(c))
        return float(c[idx]), [idx], [1.0]
    
    # Use top-k for efficiency
    k = min(topk, n_c)
    part = np.argpartition(c, -k)[-k:]
    max_val = float(c[part].max())
    exp_vals = np.exp((c[part] - max_val) / T)
    w = exp_vals / exp_vals.sum()
    
    # Add a correction for the parts we didn't include
    # (they should have negligible weight if T is reasonable)
    smooth_max = max_val + T * float(np.log(exp_vals.sum()))
    return smooth_max, part.tolist(), w.tolist()


def gradient_smoothed(a, c, T, topk=50):
    """
    Compute gradient of 2n * smooth_max(a*a) w.r.t. a (with sum(a)=1).
    Returns the gradient (not yet projected).
    """
    n = len(a)
    smooth_max, indices, weights = smooth_max_and_grad(c, T, topk)
    
    grad = np.zeros(n)
    for k, wk in zip(indices, weights):
        if wk < 1e-12:
            continue
        k = int(k)
        if k == 0:
            grad += wk * 2.0 * a
        else:
            # b_k = sum_{i=0}^{n-1-k} a_i * a_{i+k}
            # db_k/da_j = a_{j+k} for j=0..n-1-k, plus a_{j-k} for j=k..n-1
            if n > k:
                grad[:n-k] += wk * a[k:]
                grad[k:] += wk * a[:n-k]
    
    return 2.0 * n * grad, smooth_max


def objective_simplex(a, c):
    """Objective on the simplex: 2n * max(a*a)."""
    n = len(a)
    return 2.0 * n * float(c.max())


def backtracking_line_search(a, grad_dir, f_val, c, T, max_iter=20, topk=50):
    """
    Backtracking line search for projected gradient step.
    """
    n = len(a)
    lr = 1.0
    best_a = a.copy()
    best_f = f_val
    
    for _ in range(max_iter):
        # Project gradient onto tangent space
        grad_tan = grad_dir - np.sum(grad_dir) / n
        a_new = proj_simplex(a - lr * grad_tan)

            lr *= 0.5
            continue
        
        c_new = fast_self_conv(a_new)
        f_new = 2.0 * n * float(c_new.max())
        
        if f_new < f_val - 1e-12 * abs(f_val):
            return a_new, f_new, lr
        elif f_new < best_f:
            best_a = a_new.copy()
            best_f = f_new
        
        lr *= 0.5
        if lr < 1e-15:
            break
    
    return best_a, best_f, lr

        grad, _ = compute_grad(a, c, T, topk)

        # Project gradient onto tangent space of simplex (sum=1 constraint)
        grad_tan = grad - np.sum(grad) / n

        # Momentum
        if use_momentum:
            m = m_decay * m + (1.0 - m_decay) * grad_tan
            step_dir = m
        else:
            step_dir = grad_tan

        # Take step and project
        a_new = proj_simplex(a - lr * step_dir)

        # Update a and decay parameters
        a = a_new
        lr *= lr_decay
        T = max(T_min, T * 0.9995)
        topk = min(topk_max, int(topk * 1.001))

        it += 1
        if it % 1000 == 0:
            elapsed = time.monotonic() - (end_time - 300)  # rough
            print(f"    it={it}, obj={best_obj:.7f}, lr={lr:.6f}, T={T:.5f}, topk={topk}")

    return best_obj, best_a


def make_inits(n, rng):
    """Generate diverse initial conditions on the simplex."""
    inits = []
    x = np.linspace(0, 1, n, endpoint=False)

    # Uniform
    inits.append(np.ones(n) / n)

    # Cosine modulations at various frequencies and amplitudes
    for freq in [1, 2, 3, 4, 6]:
        for amp in [0.3, 0.5, 0.7, 0.9]:
            a = 1.0 + amp * np.cos(2.0 * np.pi * freq * x)
            inits.append(normalize_simplex(np.maximum(a, 0.001)))

    # Multi-frequency
    a = 1.0 + 0.4 * np.cos(2 * np.pi * x) + 0.3 * np.cos(4 * np.pi * x)
    inits.append(normalize_simplex(np.maximum(a, 0.001)))
    a = 1.0 + 0.3 * np.cos(2 * np.pi * 3 * x) + 0.2 * np.sin(2 * np.pi * 7 * x)
    inits.append(normalize_simplex(np.maximum(a, 0.001)))

    # Random
    for _ in range(2):
        inits.append(normalize_simplex(np.maximum(rng.random(n), 0.0)))

    # Gapped patterns
    a_g = np.ones(n)
    a_g[1::2] = 0.2
    inits.append(normalize_simplex(a_g))

    a_g2 = np.ones(n)
    a_g2[::3] = 0.05
    inits.append(normalize_simplex(a_g2))

    # Bell-shaped
    mid = n / 2.0
    dist = np.abs(np.arange(n) - mid)
    inits.append(normalize_simplex(np.exp(-dist / (n * 0.25))))

    # Two peaks
    p1, p2 = n // 4, 3 * n // 4
    a_two = np.exp(-np.abs(np.arange(n) - p1) / (n * 0.1)) + np.exp(-np.abs(np.arange(n) - p2) / (n * 0.1))
    inits.append(normalize_simplex(a_two))

    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses projected gradient on the positive simplex with softmax-smoothed gradient,
    momentum, adaptive temperature, and diverse restarts.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = float(budget_s)
    deadline = start + total - 12.0

    # Get parent if available
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if parent.size > 0:
                parent = np.clip(parent, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None and parent.size > 100:
        n = int(parent.size)
        p_conv = fast_self_conv(parent)
        p_sum = float(parent.sum())
        parent_score = 2.0 * n * float(p_conv.max()) / (p_sum * p_sum) if p_sum > 0 else np.inf
        print(f"Parent: n={n}, score={parent_score:.7f}")
    else:
        n = 8192
        parent_score = np.inf
        print(f"No parent, using n={n}")

    n = int(min(max(n, 1024), 16384))

    best_score = np.inf
    best_a = None

    # --- Phase 1: Quick exploration (15% time) ---
    p1_end = start + 0.12 * (total - 12.0)
    print(f"Phase 1: exploration until {time.strftime('%H:%M:%S', time.localtime(p1_end))}")

    candidates = []
    if parent is not None and parent.size == n:
        candidates.append(normalize_simplex(parent))
        # Slightly perturbed parents
        for sc in [0.03, 0.08, 0.15]:
            candidates.append(normalize_simplex(parent + sc * rng.standard_normal(n)))

    # Diverse inits (take a few)
    inits = make_inits(n, rng)
    candidates.extend(inits[:8])

    print(f"Phase 1: {len(candidates)} candidates, {p1_end - start:.0f}s total")
    per_time = (p1_end - start) / len(candidates)

    for i, a0 in enumerate(candidates):
        t_end = min(p1_end, start + per_time * (i + 1))
        if time.monotonic() >= p1_end:
            break
        sc, aa = optimize_simplex(a0, t_end, rng,
                                  lr_init=0.05, lr_decay=0.999,
                                  T_init=0.1, T_min=0.01,
                                  topk_init=20, topk_max=50,
                                  patience=80, use_momentum=True)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"  P1[{i:2d}]: obj={sc:.7f} (best={best_score:.7f})")

    # --- Phase 2: Main refinement (45% time) ---
    p2_end = start + 0.55 * (total - 12.0)
    print(f"Phase 2: main refinement until {time.strftime('%H:%M:%S', time.localtime(p2_end))}")
    if best_a is not None:
        sc, aa = optimize_simplex(best_a, p2_end, rng,
                                  lr_init=0.03, lr_decay=0.9995,
                                  T_init=0.05, T_min=0.005,
                                  topk_init=30, topk_max=100,
                                  patience=200, use_momentum=True)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 2: best={best_score:.7f}")

    # --- Phase 3: Fine polish (25% time) ---
    p3_end = start + 0.82 * (total - 12.0)
    print(f"Phase 3: fine polish until {time.strftime('%H:%M:%S', time.localtime(p3_end))}")
    if best_a is not None:
        sc, aa = optimize_simplex(best_a, p3_end, rng,
                                  lr_init=0.01, lr_decay=0.9998,
                                  T_init=0.02, T_min=0.002,
                                  topk_init=50, topk_max=200,
                                  patience=400, use_momentum=True)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 3: best={best_score:.7f}")

    # --- Phase 4: Ultra-fine final polish ---
    if best_a is not None and time.monotonic() < deadline - 5.0:
        print(f"Phase 4: ultra-fine polish")
        sc, aa = optimize_simplex(best_a, deadline - 5.0, rng,
                                  lr_init=0.003, lr_decay=0.9999,
                                  T_init=0.005, T_min=0.0005,
                                  topk_init=100, topk_max=500,
                                  patience=800, use_momentum=True)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 4: best={best_score:.7f}")

    # --- Final output ---
    if best_a is None:
        best_a = np.ones(n) / n

    # The objective is scale-invariant, so output at any positive scale
    # Use a moderate scale to avoid numerical issues
    out_a = best_a * 1.0  # already on simplex, values in [0, 1/n]
    # Scale up a bit for numerical safety in evaluation
    out_a = out_a / max(out_a.max(), 1e-15)  # scale so max = 1

    out = [float(max(0.0, min(1000.0, x))) for x in out_a]

    # Verify with the evaluator
    verified = np.inf
    try:
        verified = float(evaluate_sequence(out))
    except Exception as e:
        print(f"  Verification error: {e}")

    elapsed = time.monotonic() - start
    print(f"\nFINAL: internal_simplex_obj={best_score:.7f}, "
          f"verified={verified:.7f}, n={n}, time={elapsed:.1f}s")
    return out
