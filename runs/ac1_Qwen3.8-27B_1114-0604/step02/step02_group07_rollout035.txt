Multi-scale L-BFGS-B optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.

multiple restarts. Combines with adaptive-gradient fine-tuning for final polish.

import time

# Global state for scipy minimize callbacks
_G_T = 1.0
_G_n = 0

def fast_self_conv(a):
    """Full self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    N = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:M]

def compute_score(n, b, S):
    """Score = 2n * max(b) / S^2."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(b)) / (S * S)

def smooth_obj_grad(a, T, n):
    """
    Smoothed objective: f = 2n * T * logsumexp(b/T) / S^2 + 1e-10 * sum(a^2)
    and its gradient.
    """
    b = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-10:
        return np.inf, np.zeros_like(a)
    
    b_s = b / T
    bm = b_s.max()
    ex = np.exp(b_s - bm)
    lse = bm + np.log(ex.sum())
    L = T * lse
    w = ex / ex.sum()
    
    # dL/da via FFT
    a_rev = a[::-1].copy()
    N_fft = 1 << (3 * n - 3).bit_length()
    Ar = np.fft.rfft(a_rev, N_fft)
    W = np.fft.rfft(w, N_fft)
    dL_da = 2.0 * np.fft.irfft(Ar * W, N_fft)[n-1 : 2*n-1]
    
    f = 2.0 * n * L / (S * S) + 1e-10 * float(np.dot(a, a))
    grad = 2.0 * n * (dL_da * S - 2.0 * L * a) / (S * S * S) + 2e-10 * a
    
    return f, grad

    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]
    
    top1 = float(vals[0])
    top2 = float(vals[1]) if k_act > 1 else 0.0
    ratio = top2 / top1 if top1 > 1e-15 else 1.0
    
    exact_w = float(np.clip((0.9 - ratio) / 0.4, 0.0, 1.0))
    
    if exact_w > 0.99:
        wts = np.zeros(k_act)
        wts[0] = 1.0
        M = top1
    else:
        temp = max(temp_scale * top1, 1e-15)
        wts_sm = np.exp((vals - top1) / temp)
        wts_sm /= wts_sm.sum()
        if exact_w < 0.01:
            wts = wts_sm
        else:
            wts_ex = np.zeros(k_act)
            wts_ex[0] = 1.0
            wts = exact_w * wts_ex + (1.0 - exact_w) * wts_sm
            wts /= wts.sum()
        M = float(np.dot(vals, wts))
    
    grad_c = np.zeros(n)
    for i in range(k_act):
        wi = float(wts[i])
        if wi <= 1e-15:
            continue
        ki = int(idx[i])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        grad_c[js:je+1] += wi * 2.0 * a[js:je+1][::-1]
    
    grad = 2.0 * n * (grad_c / S2 - 2.0 * M * a / S3)
    return grad, M, S

def adam_update(a, g, m, v, t, lr):
    """In-place Adam step."""
    t1 = 0.9 ** t
    t2 = 0.999 ** t
    m[:] = 0.9 * m + 0.1 * g
    v[:] = 0.999 * v + 0.001 * g * g
    a -= lr * (m / (1.0 - t1)) / (np.sqrt(v / (1.0 - t2)) + 1e-8)

Multi-scale L-BFGS-B optimizer with logsumexp smoothing for minimizing
2n*max(conv(a,a))/(sum(a))^2. Combines adaptive temperature sweeps,
full-softmax gradients, diverse asymmetric initializations, and Adam

_G_n = 0

def fast_self_conv(a):
    """Full self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    N = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:M]

def compute_score(n, b, S):
    """Score = 2n * max(b) / S^2."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(b)) / (S * S)

def smooth_obj_grad(a, T, n):
    """
    Smoothed objective f = 2n*T*logsumexp(b/T)/S^2 + eps*||a||^2 and gradient.
    Uses FFT-based convolution for the gradient computation.
    """
    b = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-10:
        return np.inf, np.zeros_like(a)
    
    b_s = b / T
    bm = float(b_s.max())
    ex = np.exp(b_s - bm)
    ex_sum = float(ex.sum())
    lse = bm + np.log(ex_sum)
    L = T * lse
    w = ex / ex_sum
    
    # dL/da[i] = 2 * sum_j w[i+j] * a[j]
    # Computed via FFT: conv(a[::-1], w)[n-1 : 2n-1]
    a_rev = a[::-1].copy()
    N_fft = 1 << (3 * n - 3).bit_length()
    Ar = np.fft.rfft(a_rev, N_fft)
    W = np.fft.rfft(w, N_fft)
    dL_da = 2.0 * np.fft.irfft(Ar * W, N_fft)[n - 1 : 2 * n - 1]
    
    eps = 1e-10
    f = 2.0 * n * L / (S * S) + eps * float(np.dot(a, a))
    grad = 2.0 * n * (dL_da * S - 2.0 * L * a) / (S * S * S) + 2.0 * eps * a
    
    return f, grad
