Key insight: I'll use a **hybrid approach**:
1. **Phase 1**: Multi-scale multiplicative updates (like before) to get a good initial solution.
2. **Phase 2**: L-BFGS-B optimization in log-space for local polishing.
3. **Phase 3**: Additional local search (block scaling, pairwise swaps).

For the L-BFGS-B gradient, I'll compute it as follows:
- Work with `a` directly (not in log-space), using the soft-max smoothed objective.
- The gradient of the smooth objective `f_smooth = 2*n*(1/beta)*log(sum_k exp(beta*c_k))` with respect to `a_i` is:
  `df/da_i = 2*n * sum_k w_k * (a_{i+k} + a_{i-k})`
  where `w_k = exp(beta*c_k) / sum_j exp(beta*c_j)`
- To convert to log-space: `df/dx_i = (df/da_i) * a_i` (chain rule, since `a_i = exp(x_i)/S`, `da_i/dx_i = a_i * (1 - a_i)`, `da_i/dx_j = -a_i*a_j` for `i≠j`)
  Actually: `df/dx_i = sum_j (df/da_j) * (da_j/dx_i)`
  `= (df/da_i) * a_i * (1-a_i) + sum_{j≠i} (df/da_j) * (-a_i * a_j)`
  `= (df/da_i) * a_i - a_i * sum_j (df/da_j) * a_j`
  `= a_i * (df/da_i - sum_j df/da_j * a_j)`

So in log-space, the gradient is `a_i * (g_i - <g, a>)` where `g_i = df/da_i`.

This is the standard "softmax gradient" transformation. It makes sense: the gradient in log-space is the gradient in a-space projected to the tangent space of the simplex (via the softmax parameterization).

```python
def _log_objective_and_grad(x, n, L, beta):
    xm = x - x.max()
    ex = np.exp(xm)
    S = ex.sum()
    a = ex / S
    
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2*n-1]
    c = np.maximum(c, 0.0)
    
    cm = c.max()
    bc = beta * (c - cm)
    ebc = np.exp(bc)
    Z = ebc.sum()
    w = ebc / Z
    
    smooth_max = cm + np.log(Z) / beta
    obj = 2.0 * n * smooth_max
    
    # Compute h1 = cross-correlation of a with w (using the symmetry)
    v = np.zeros(L)
    v[0] = w[n-1]
    if n > 1:
        v[1:n] = w[n-2::-1]
        v[L-n+1:L] = w[2*n-2:n-1:-1]
    
    h1 = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(v, L))[:n]
    g = 2.0 * h1  # gradient of smooth_max w.r.t. a
    
    # Gradient in log-space
    m = np.dot(g, a)
    log_grad = 2.0 * n * a * (g - m)
    
    return obj, log_grad
```

```python
"""
L-BFGS-B based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2
with non-negative constraint, using log-parameterization and soft-max
smoothing of the max.
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize

_GLOBAL_RNG = np.random.default_rng(42)

def _next_pow2(x):
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_len(n):
    return _next_pow2(2 * n - 1)

def _sanitize(p, normalize=True):
    a = np.asarray(p, dtype=float).ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    a = a + 1e-15
    if normalize:
        s = a.sum()
        if s <= 0 or not np.isfinite(s):
            return None
        a = a / s
    return a

def _score_fft(a, L=None):
    n = len(a)
    if L is None:
        L = _conv_len(n)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2*n-1]
    c = np.maximum(c, 0.0)
    s = float(a.sum())
    score = float(2.0 * n * c.max() / (s*s)) if s > 0 else float('inf')
    return score, c

def _build_v(w, n, L):
    """Build the FFT kernel for computing h1[i] = sum_k w[n-1+k]*a[i+k]"""
    v = np.zeros(L)
    v[0] = w[n-1]
    if n > 1:
        v[1:n] = w[n-2::-1]
        v[L-n+1:L] = w[2*n-2:n-1:-1]
    return v

def _objective_and_grad(x, n, L, beta):
    """
    Soft-max smoothed objective and gradient in log-parameter space.
    """
    xm = x - np.max(x)
    ex = np.exp(xm)
    S = float(ex.sum())
    a = ex / S
    
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2*n-1]
    c = np.maximum(c, 0.0)
    
    cm = float(c.max())
    bc = beta * (c - cm)
    ebc = np.exp(bc)
    Z = float(ebc.sum())
    w = ebc / Z
    
    smooth_max = cm + np.log(Z) / beta
    obj = 2.0 * n * smooth_max
    
    v = _build_v(w, n, L)
    hv = np.fft.rfft(v, L)
    h1 = np.fft.irfft(F * hv)[:n]
    
    # Due to symmetry of w, g = 2*h1
    g = 2.0 * h1
    
    m = float(np.dot(g, a))
    grad = 2.0 * n * a * (g - m)
    
    return obj, grad

def _exact_score(a):
    """Exact score using direct computation"""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    c = np.convolve(a, a, mode='full')[:, :2*n-1]  # just in case
    # Actually np.convolve(a,a,'full') already has length 2n-1
    c = np.convolve(a, a, mode='full')
    return float(2.0 * n * c.max() / (s*s))

def _lbfgsb_optimize(a0, time_limit, beta_start=50.0, beta_end=5000.0, 
                     n_rounds=5, verbose=True):
    """
    Multi-round L-BFGS-B optimization in log-parameter space.
    """
    n = len(a0)
    L = _conv_len(n)
    x0 = np.log(a0)  # log-parameters
    
    deadline = time.monotonic() + time_limit
    best_a = a0.copy()
    best_score = _exact_score(a0)
    
    if verbose:
        print(f"  L-BFGS-B start: n={n}, score={best_score:.7f}")
    
    for round_idx in range(n_rounds):
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        
        # Beta for this round
        t = (round_idx + 1) / float(n_rounds)
        beta = beta_start * (beta_end / beta_start) ** t
        
        round_time = min(rem - 10, time_limit / (n_rounds * 1.5))
        
        def func(x):
            val, grad = _objective_and_grad(x, n, L, beta)
            return val, grad
        
        res = scipy_minimize(
            func, x0, method='L-BFGS-B',
            options={
                'maxiter': 500,
                'maxfun': 1000,
                'ftol': 1e-12,
                'gtol': 1e-8,
                'maxls': 50,
            }
        )
        
        x_new = res.x
        # Convert back to a
        xm = x_new - np.max(x_new)
        ex = np.exp(xm)
        a_new = ex / ex.sum()
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        score_new = _exact_score(a_new)
        
        if verbose:
            print(f"    Round {round_idx+1}: beta={beta:.0f}, "
                  f"lbfgs_score={res.fun:.7f}, exact_score={score_new:.7f}")
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            x0 = x_new.copy()
        else:
            # Keep the better x0
            x0 = x_new.copy()  # Still update x0 as it's in a good region
    
    return best_a, best_score

def _mmd_refine(a0, time_limit, target=1.500, verbose=False):

    """

    L = _conv_len(n)
    a = a0.copy()
    best_a = a.copy()
    best_score, _ = _score_fft(a, L)
    
    deadline = time.monotonic() + time_limit
    alpha = 0.1
    beta = 100.0
    stall = 0
    it = 0

        F = np.fft.rfft(a, L)
        c = np.fft.irfft(F * F, L)[:2*n-1]
        c = np.maximum(c, 0.0)
        score = float(2.0 * n * c.max())  # sum(a)=1
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        cm = float(c.max())
        
        # Soft-max weights
        bc = beta * (c - cm)

def _block_local_search(a0, time_limit, verbose=False):

def _initial_patterns(n, rng):
    """Generate diverse initial patterns."""
    t = (np.arange(n, dtype=float) + 0.5) / n
    patterns = []
    
    # Beta distribution
    for a, b in [(1,1), (0.5,0.5), (2,2), (0.5,2), (2,0.5), (0.3,0.3), (3,3)]:
        p = np.power(np.clip(t, 1e-8, 1-1e-8), a) * np.power(1-np.clip(t, 1e-8, 1-1e-8), b)
        p = p + 1e-10
        p = p / p.sum()
        patterns.append(p)
    
    # Gaussian bumps
    for K in [2, 3, 5]:
        p = np.zeros(n)
        for _ in range(K):
            center = rng.uniform(0.2, 0.8)
            sigma = rng.uniform(0.05, 0.3)
            p += np.exp(-((t - center)/sigma)**2)
        p = p + 1e-10
        p = p / p.sum()
        patterns.append(p)
    
    # Two-block
    for cut in [0.3, 0.5, 0.7]:
        for ratio in [0.5, 1.0, 2.0]:
            p = np.ones(n)
            idx = int(cut * n)
            p[idx:] = ratio
            p = p / p.sum()
            patterns.append(p)
    
    # Random smooth
    for _ in range(3):
        p = np.exp(rng.normal(0, 1, n))
        # Smooth
        from scipy.ndimage import gaussian_filter1d
        p = gaussian_filter1d(p, sigma=5)
        p = p / p.sum()
        patterns.append(p)
    
    return patterns
```
