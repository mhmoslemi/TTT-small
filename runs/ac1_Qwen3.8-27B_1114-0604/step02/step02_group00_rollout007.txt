import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def fft_cross_conv(a, b):
    """Cross-convolution a * b via FFT. Result has length len(a)+len(b)-1."""
    n = len(a) + len(b) - 1
    M_fft = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:n]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Numerical stability: clip to avoid overflow
    scaled = np.clip(scaled, -50, 50)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient
    rev_a = a[::-1]
    M2 = len(rev_a) + len(p) - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)
    g = 2.0 * conv_res[n-1:2*n-1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)

"""
LP-based best response with cutting plane, combined with L-BFGS LSE optimization.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy:
  - Warm up with L-BFGS on LSE-smoothed objective
  - Iteratively solve LP best-response (maximize sum(g) s.t. f*g <= M, g>=0)
  - Interpolate with adaptive step size
  - Alternate between LP and L-BFGS refinement
  - Final subgradient polish

import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fft_cross_conv(a, b):
    """Cross-convolution a*b via FFT. Result length = len(a)+len(b)-1."""
    n = len(a) + len(b) - 1
    M_fft = 1 << (n - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:n]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -100, 100)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = len(rev_a) + len(p) - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=3000, maxfun=60000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def build_constraint_sparse(f, k, n):
    """
    Build sparse row for constraint (f*g)[k] <= M.
    Coefficient of g[m] is f[k-m] for valid m.
    Returns (indices, values) for sparse row.
    """
    lo = max(0, k - n + 1)
    hi = min(k, n - 1)
    if hi < lo:
        return np.array([], dtype=int), np.array([], dtype=float)
    indices = np.arange(lo, hi + 1)
    values = f[k - indices]
    return indices, values


def lp_best_response(f, M, max_rounds=25, time_limit=60.0, verbose=False):
    """
    Solve: max sum(g) s.t. (f*g)[k] <= M for active lags, g >= 0.
    Cutting plane approach with CVXPY.
    Returns (g, success).
    """
    t_start = time.monotonic()
    n = len(f)
    g = cp.Variable(n, non_negative=True)
    
    # Initial active lags: top peaks of f*f
    c = fast_conv(f)
    top_peaks = np.argsort(c)[::-1][:min(15, 2 * n - 1)]
    active_lags = [int(k) for k in top_peaks]
    
    # Build initial constraints
    constraint_rows = []
    for k in active_lags:
        idx, vals = build_constraint_sparse(f, k, n)
        if len(idx) > 0 and np.max(np.abs(vals)) > 1e-15:
            constraint_rows.append((idx, vals))
    
    def build_and_solve(constraint_rows):
        cons = []
        for idx, vals in constraint_rows:
            # Build sparse constraint: vals[0]*g[idx[0]] + vals[1]*g[idx[1]] + ... <= M
            # Use cp表达
            expr = 0
            for i in range(len(idx)):
                expr += vals[i] * g[idx[i]]
            cons.append(expr <= M)
        prob = cp.Problem(cp.Maximize(cp.sum(g)), cons)
        # Try solvers in order
        for solver in [cp.MOSEK, cp.CLARABEL, cp.ECOS, cp.SCS]:
            try:
                prob.solve(solver=solver)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    return g.value, prob.status
            except Exception:
                continue
        return None, "failed"
    
    best_g = None
    best_sum = 0.0
    
    for rnd in range(max_rounds):
        if time.monotonic() - t_start > time_limit:
            break
        
        t_solve = time.monotonic()
        g_val, status = build_and_solve(constraint_rows)
        solve_time = time.monotonic() - t_solve
        
        if g_val is None:
            if verbose:
                print(f"    LP round {rnd}: solve failed ({status}), time={solve_time:.1f}s")
            break
        
        g_val = np.asarray(g_val, dtype=np.float64).flat
        g_val = np.clip(g_val, 0.0, 1000.0)
        g_sum = float(g_val.sum())
        
        if g_sum > best_sum and np.all(np.isfinite(g_val)):
            best_sum = g_sum
            best_g = g_val.copy()
        
        if verbose:
            print(f"    LP round {rnd}: sum(g)={g_sum:.2f}, solve_time={solve_time:.1f}s, constraints={len(constraint_rows)}")
        
        # Check feasibility: find most violated constraint
        fg = fft_cross_conv(f, g_val)
        violations = fg - M
        max_viol = float(violations.max())
        max_viol_idx = int(np.argmax(violations))
        
        if max_viol < 1e-6 * max(M, 1e-10):
            if verbose:
                print(f"    LP round {rnd}: FEASIBLE (max_viol={max_viol:.2e})")
            break
        
        # Add most violated constraint
        if max_viol_idx not in [int(x) for x in active_lags]:
            idx, vals = build_constraint_sparse(f, max_viol_idx, n)
            if len(idx) > 0 and np.max(np.abs(vals)) > 1e-15:
                constraint_rows.append((idx, vals))
                active_lags.append(max_viol_idx)
        
        # If too many constraints, stop
        if len(constraint_rows) > 100:
            if verbose:
                print(f"    LP round {rnd}: too many constraints ({len(constraint_rows)}), stopping")
            break
    
    return best_g, best_g is not None and float(best_g.sum()) > 0


def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements contributing most to peak convolution."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 40)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:n - top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)


def subgradient_polish(a, total_iters=500):
    """Projected subgradient descent on exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Subgradient of M
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            sg_M[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.3)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        step_decay = 0.0005 / (1.0 + 0.01 * it)
        a -= step_decay * grad / gn
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def multi_top_subgradient(a, total_iters=300, top_k=5):
    """Subgradient using top-k peaks combined."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.2
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Find top-k peaks
        c_sorted = np.sort(c)[::-1]
        threshold = c_sorted[min(top_k, len(c_sorted)) - 1]
        peak_lags = np.where(c >= threshold * 0.99)[0]
        # Combined subgradient
        sg = np.zeros(n)
        for k_star in peak_lags:
            k_star = int(k_star)
            j_start = max(0, k_star - n + 1)
            j_end = min(k_star, n - 1)
            if j_end >= j_start:
                sg[j_start:j_end + 1] += 2.0 * a[j_start:j_end + 1][::-1]
        sg /= max(len(peak_lags), 1)
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        # Adaptive step
        base_step = 0.01 / (1.0 + 0.005 * it)
        a -= base_step * grad / gn
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a

        if S_g < 1e-10:
            continue
        g_scaled = g * (S_curr / S_g)
        
        # Check ||g*g||_inf to determine safe step size
        gg = fast_conv(g_scaled)
        M_gg = float(gg.max())
        # Safe condition: M_gg <= M_curr for guaranteed decrease
        # For t: M_new <= M*(1-t^2) + t^2*M_gg
        # Want M_new < M: need t^2*(M_gg - M) < t^2*M => M_gg < M
        # If M_gg > M, use smaller t
        if M_gg <= M_curr:
            t_step = 0.3
        else:
            # M_new = M - t^2*(M - M_gg) = M + t^2*(M_gg - M)
            # For decrease we'd need M_gg < M, which isn't the case
            # Use very small t and check empirically
            t_step = 0.1
        
        # Try a few step sizes and pick the best
        best_t = t_step
        best_trial = a_cur.copy()
        for t_try in [t_step * 0.5, t_step, t_step * 1.5, t_step * 0.25]:
            a_trial = (1.0 - t_try) * a_cur + t_try * g_scaled
            a_trial = np.clip(a_trial, 0.0, 1000.0)
            sc_trial = compute_score(a_trial)
            if sc_trial < compute_score(a_cur):
                best_t = t_try
                best_trial = a_trial
        
        a_cur = best_trial
        sc_cur = compute_score(a_cur)
        report(sc_cur, a_cur, f"LP iter={n_lp_iters}, t={best_t:.2f}")
        
        if n_lp_iters % 5 == 0:
            print(f"  LP progress: iter={n_lp_iters}, score={best_score:.8f}, t={time.monotonic()-t_start:.0f}s")

    print(f"After Phase 2: score={best_score:.8f}, LP iters={n_lp_iters}")

    # ============================================================
    # Phase 3: L-BFGS refinement (200s)
    # ============================================================
    print(f"\n{'='*50}")
    print("Phase 3: L-BFGS Refinement")
    print(f"{'='*50}")
    phase3_deadline = t_start + 1260.0

    for beta in [500, 1000, 2000]:
        if time.monotonic() > phase3_deadline:
            break
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=5000, maxfun=100000)
        report(sc, a_opt, f"refine beta={beta}")

    # ============================================================
    # Phase 4: More LP iterations (800s)
    # ============================================================
    print(f"\n{'='*50}")
    print("Phase 4: LP Best Response (Round 2)")
    print(f"{'='*50}")
    phase4_deadline = t_start + 2060.0

    a_cur = best_a.copy()
    print(f"  Starting from score={best_score:.8f}")
    n_lp2 = 0
    while time.monotonic() < phase4_deadline:
        n_lp2 += 1
        c = fast_conv(a_cur)
        M_curr = float(c.max())
        S_curr = float(a_cur.sum())
        
        time_left = phase4_deadline - time.monotonic()
        lp_time = min(70.0, max(10.0, time_left * 0.12))
        
        g, success = lp_best_response(a_cur, M_curr, max_rounds=20, time_limit=lp_time, verbose=(n_lp2 <= 2))
        
        if not success or g is None:
            if n_lp2 < 3:
                g, success = lp_best_response(a_cur, M_curr * 1.0005, max_rounds=15, time_limit=lp_time, verbose=False)
            if not success or g is None:
                sc, a_opt, _ = run_lbfgs(a_cur, 1000, maxiter=3000, maxfun=60000)
                a_cur = a_opt
                report(sc, a_opt, f"phase4 fallback lbfgs iter={n_lp2}")
                continue
        
        S_g = float(g.sum())
        if S_g < 1e-10:
            continue
        g_scaled = g * (S_curr / S_g)
        
        # Adaptive step size
        best_t = 0.2
        best_trial = a_cur.copy()
        for t_try in [0.1, 0.15, 0.2, 0.3, 0.4]:
            a_trial = (1.0 - t_try) * a_cur + t_try * g_scaled
            a_trial = np.clip(a_trial, 0.0, 1000.0)
            sc_trial = compute_score(a_trial)
            if sc_trial < compute_score(a_cur):
                best_t = t_try
                best_trial = a_trial
        
        a_cur = best_trial
        sc_cur = compute_score(a_cur)
        report(sc_cur, a_cur, f"LP2 iter={n_lp2}, t={best_t:.2f}")
        
        if n_lp2 % 5 == 0:
            print(f"  LP2 progress: iter={n_lp2}, score={best_score:.8f}, t={time.monotonic()-t_start:.0f}s")

    print(f"After Phase 4: score={best_score:.8f}")

    # ============================================================
    # Phase 5: Final polish (remaining time)
    # ============================================================
    print(f"\n{'='*50}")
    print("Phase 5: Final Polish")
    print(f"{'='*50}")
    
    # Multi-peak subgradient
    if time.monotonic() < deadline - 30:
        sc, a_pol = multi_top_subgradient(best_a, total_iters=500, top_k=5)
        report(sc, a_pol, "multi-top subgrad")
    
    # Targeted perturbation + L-BFGS
    if time.monotonic() < deadline - 30:
        a_pert = targeted_perturb(best_a, rng, strength=0.04)
        sc, a_opt, _ = run_lbfgs(a_pert, 1000, maxiter=3000, maxfun=60000)
        report(sc, a_opt, "perturb+lbfgs")
    
    # Final L-BFGS with very high beta
    if time.monotonic() < deadline - 20:
        sc, a_opt, _ = run_lbfgs(best_a, 2000, maxiter=3000, maxfun=60000)
        report(sc, a_opt, "final lbfgs beta=2000")
    
    # Subgradient polish
    if time.monotonic() < deadline - 15:
        sc, a_pol = subgradient_polish(best_a, total_iters=300)
        report(sc, a_pol, "subgrad polish")

    # ============================================================
    # Final verification
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
