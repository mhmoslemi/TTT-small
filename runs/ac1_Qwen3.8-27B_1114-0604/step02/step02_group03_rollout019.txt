def _normalize(v):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v


def _fft_size(m):
    if m <= 2:
        return 2
    return 1 << (int(m) - 1).bit_length()


def _autocorr(f):
    n = f.size
    m = 2 * n - 1
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]


def _fast_score(f):
    f = _normalize(f)
    if f.size == 0:
        return np.inf
    c = _autocorr(f)
    if c.size == 0:
        return np.inf
    return float(2.0 * f.size * c.max())


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    out = v - theta
    np.maximum(out, 0.0, out)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    out /= s
    return out

def _transfer_max_fast(c, f, i, j, dv):
    """Compute max of conv after transfer f[i]-=dv, f[j]+=dv. Returns (max_val, delta_array)."""
    n = f.size
    L = c.size
    delta = np.zeros(L, dtype=np.float64)

    delta[j:j + n] += f
    delta *= (2.0 * dv)

    if 2 * j < L:
        delta[2 * j] += q
    if 2 * i < L:
        delta[2 * i] += q
    if i + j < L:
        delta[i + j] -= 2.0 * q
    new_c = c + delta
    return float(np.max(new_c)), delta


def _waterfilling_step(c, f, i, j, max_dv):
    """
    Find optimal dv in [0, max_dv] that minimizes max_k conv'_k.
    Uses water-filling: find where the active constraint's linear descent
    meets the next constraint.
    Returns (optimal_dv, new_max).
    """
    n = f.size
    L = c.size
    if max_dv <= 1e-15:
        return 0.0, float(c.max())

    # Compute linear coefficient L_k = f[k-j] - f[k-i] (with 0 for out-of-range)
    # conv'_k ≈ c[k] + dv * L_k (ignoring quadratic corrections at 3 positions)
    L_k = np.zeros(L, dtype=np.float64)
    # f[k-j] contribution: for k in [j, j+n-1], L_k[k] += f[k-j]
    L_k[j:j + n] += f
    # -f[k-i] contribution: for k in [i, i+n-1], L_k[k] -= f[k-i]
    L_k[i:i + n] -= f

    # Find current active constraint
    k_star = int(np.argmax(c))
    c_star = c[k_star]
    L_star = L_k[k_star]

    # If L_star >= 0, increasing dv doesn't decrease the active constraint
    # Try decreasing dv benefit is zero
    if L_star >= -1e-15:
        return 0.0, c_star

    # Find the constraint that catches up first
    # t_k = (c_k - c_star) / (L_star - L_k) for k where L_k > L_star (or c_k > c_star)
    # We need L_star - L_k < 0, i.e., L_k > L_star
    denom = L_star - L_k  # negative where L_k > L_star
    # Valid candidates: denom < 0 (i.e., L_k > L_star) AND c_k < c_star (below active)
    # OR: valid if the line c_k + dv*L_k crosses c_star + dv*L_star at positive dv
    # t_k = (c_k - c_star) / (L_star - L_k)
    # We need t_k > 0, which means (c_k - c_star) and (L_star - L_k) have same sign
    # Since L_star < 0 (we checked), L_star - L_k < 0 means L_k > L_star
    # And c_k - c_star < 0 means c_k < c_star

    valid = (denom < -1e-15) & (c < c_star + 1e-15)
    if np.any(valid):
        t_vals = np.where(valid, (c - c_star) / np.where(denom != 0, denom, 1e-300), np.inf)
        # Also ensure t > 0
        t_vals[~np.isfinite(t_vals)] = np.inf
        t_vals[t_vals < 0] = np.inf
        t_min = float(np.min(t_vals))
    else:
        t_min = np.inf

    # Also check: dv cannot exceed max_dv
    dv_opt = min(max_dv, t_min)

    # Clip to valid range
    if dv_opt <= 0:
        dv_opt = 0.0
    if dv_opt > max_dv:
        dv_opt = max_dv

    # Evaluate at dv_opt and a few nearby points to handle quadratic corrections
    best_mc = float(c.max())
    best_dv = 0.0

    candidates = [dv_opt]
    if dv_opt > 1e-10:
        candidates.extend([dv_opt * 0.8, dv_opt * 0.6, dv_opt * 1.1, dv_opt * 1.2])
    candidates = [max(0.0, min(x, max_dv)) for x in candidates]
    candidates = list(set([round(x, 15) for x in candidates]))

    for dv in candidates:
        if dv <= 1e-15:
            mc = float(c.max())
            if mc < best_mc:
                best_mc = mc
                best_dv = dv
            continue
        mc, _ = _transfer_max_fast(c, f, i, j, dv)
        if mc < best_mc:
            best_mc = mc
            best_dv = dv

    return best_dv, best_mc


def _coordinate_polish_wf(f0, c0, budget_s, rng):
    """
    Local search using water-filling optimal step sizes.
    """
    f = _normalize(f0)
    n = f.size
    if n < 2 or budget_s <= 0.0:
        c = _autocorr(f)
        return f, float(2.0 * n * c.max()), c

    L = 2 * n - 1
    try:
        c = np.asarray(c0, dtype=np.float64).copy()
    except Exception:
        c = _autocorr(f)
    if c.size != L or not np.all(np.isfinite(c)):
        c = _autocorr(f)

    maxc = float(np.max(c))
    best_f = f.copy()
    best_score = float(2.0 * n * maxc)

    deadline = time.monotonic() + float(budget_s)
    no_improve = 0
    move_count = 0

    # Adaptive D and R based on n
    if n > 10000:
        D = 8
        R = 8
        activeK = 80
    elif n > 3000:
        D = 12
        R = 12
        activeK = 120
    else:
        D = 16
        R = 16
        activeK = max(32, min(L, n // 5))

    activeK = int(min(L, activeK))

    while time.monotonic() < deadline and no_improve < 20:
        maxc = float(np.max(c))

        # Active lags
        if activeK >= L:
            active = np.arange(L)
        else:
            active = np.argpartition(c, L - activeK)[L - activeK:]

        p = np.zeros(L, dtype=np.float64)
        if active.size:
            p[active] = 1.0 / float(active.size)
        g = _weighted_gradient(p, f)

        donor_order = np.argsort(g)[::-1]
        recv_order = np.argsort(g)

        donors = []
        seen_d = set()
        for idx in donor_order:
            idx = int(idx)
            if f[idx] > 1e-11 and idx not in seen_d:
                donors.append(idx)
                seen_d.add(idx)
                if len(donors) >= D:
                    break

        receivers = []
        seen_r = set()
        for idx in recv_order:
            idx = int(idx)
            if idx not in seen_d and idx not in seen_r:
                receivers.append(idx)
                seen_r.add(idx)
                if len(receivers) >= R:
                    break

        if not donors or not receivers:
            no_improve += 1
            if no_improve >= 10:
                break
            continue

        best_mc = maxc
        best_move = None

        for i in donors:
            fi = float(f[i])
            if fi <= 1e-11:
                continue
            for j in receivers:
                dv, mc = _waterfilling_step(c, f, i, j, fi)
                if dv > 1e-12 and mc < best_mc:
                    best_mc = mc
                    best_move = (i, j, dv)

        # Random fallback
        if best_move is None or best_mc >= maxc - 1e-15:
            for _ in range(20):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j or f[i] <= 1e-11:
                    continue
                dv, mc = _waterfilling_step(c, f, i, j, float(f[i]))
                if dv > 1e-12 and mc < best_mc:
                    best_mc = mc
                    best_move = (i, j, dv)

        if best_move is None or best_mc >= maxc - 1e-15:
            no_improve += 1
            continue

        i, j, dv = best_move
        _, delta = _transfer_max_fast(c, f, i, j, dv)
        f[i] -= dv
        f[j] += dv
        c = c + delta
        move_count += 1
        no_improve = 0

        cur_score = float(2.0 * n * best_mc)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()

        # Periodic renormalization to avoid drift
        if move_count % 40 == 0:
            f = _normalize(f)
            c = _autocorr(f)
            maxc = float(c.max())
            cur_score = float(2.0 * n * maxc)
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()

    f = _normalize(f)
    c = _autocorr(f)
    mc = float(c.max()) if c.size else 0.0
    sc = float(2.0 * n * mc)
    if sc <= best_score:
        return f, sc, c
    bf = _normalize(best_f)
    bc = _autocorr(bf)
    bmc = float(bc.max()) if bc.size else 0.0
    return bf, float(2.0 * n * bmc), bc


def _perturb_from_best(best_f, rng):
    n = best_f.size
    r = float(rng.random())
    if r < 0.4:
        sigma = float(rng.uniform(0.03, 0.8))
        z = rng.standard_normal(n)
        f = best_f * np.exp(sigma * z)
    elif r < 0.7:
        alpha = float(rng.uniform(0.05, 0.4))
        noise = np.abs(rng.standard_normal(n))
        s = noise.sum()
        if s >= 1e-300:
            noise /= s
        f = (1.0 - alpha) * best_f + alpha * noise
    elif r < 0.85:
        # Anti-symmetric perturbation
        f = best_f.copy()
        shift = int(rng.integers(1, max(2, n // 10)))
        f[shift:] = best_f[:-shift] * float(rng.uniform(0.5, 1.5))
        f[:shift] = best_f[n - shift - 1:-shift + n - 1] * float(rng.uniform(0.5, 1.5)) if n - shift > 0 else best_f[:shift]
        f = np.abs(f)
    else:
        # Smooth then roughen
        win = max(3, n // 100)
        kernel = np.ones(win, dtype=np.float64) / win
        f = np.convolve(best_f, kernel, mode="same")
        sigma = float(rng.uniform(0.05, 0.4))
        f = f * np.exp(sigma * rng.standard_normal(n))
    return _normalize(f)


def _gradient_optimize_v2(f0, budget_s, rng, tag=""):
    """
    Annealed gradient optimizer with water-filling coordinate polish.
    """
    f0 = _normalize(f0)
    n = f0.size
    if n == 0:
        return f0, np.inf
    if budget_s <= 0.0:
        c = _autocorr(f0)
        return f0, float(2.0 * n * c.max())

    f = f0
    c = _autocorr(f)
    maxc = float(c.max())
    best_f = f.copy()
    best_score = float(2.0 * n * maxc)

    deadline = time.monotonic() + float(budget_s)
    sigma_rel = 0.06
    min_sigma_rel = 1e-12
    no_accept = 0
    it = 0
    last_coord = time.monotonic()

    if n < 2000:
        coord_budget = 0.3
        coord_interval = 0.35
    elif n < 10000:
        coord_budget = 0.5
        coord_interval = 0.5
    else:
        coord_budget = 0.7
        coord_interval = 0.6

    print(f"[{tag}] start score={best_score:.6f}")

    while time.monotonic() < deadline:
        it += 1
        if maxc <= 0.0:
            break

        # Softmax weights over lags
        T = maxc * sigma_rel + 1e-300
        z = (c - maxc) / T
        np.clip(z, -50.0, 0.0, out=z)
        p = np.exp(z)
        ps = float(p.sum())
        if ps < 1e-300:
            p.fill(1.0 / float(c.size))
        else:
            p /= ps

        g = _weighted_gradient(p, f)
        gm = g - float(g.mean())
        scale = float(gm.max() - gm.min())

        accepted = False
        if scale > 1e-300:
            best_mc = maxc
            best_ft = None
            best_ct = None

            if no_accept > 50:
                exps = (0.003, 0.01, 0.05, 0.2)
            else:
                exps = (0.01, 0.05, 0.3, 1.0, 3.0)

            for e in exps:
                arg = -e * (gm / scale)
                np.clip(arg, -40.0, 40.0, out=arg)
                ft = f * np.exp(arg)
                ft = _normalize(ft)
                ct = _autocorr(ft)
                mc = float(ct.max())
                if mc < best_mc:
                    best_mc = mc
                    best_ft = ft
                    best_ct = ct

                if e in (0.05, 0.3):
                    v = f - e * (gm / scale)
                    ft2 = _project_simplex(v)
                    ft2 = _normalize(ft2)
                    ct2 = _autocorr(ft2)
                    mc2 = float(ct2.max())
                    if mc2 < best_mc:
                        best_mc = mc2
                        best_ft = ft2
                        best_ct = ct2

            if best_ct is not None and best_mc < maxc - 1e-16:
                f = best_ft
                c = best_ct
                maxc = best_mc
                accepted = True

        if accepted:
            no_accept = 0
            cur_score = float(2.0 * n * maxc)
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()
                if best_score < TARGET_SCORE:
                    print(f"[{tag}] new best below target: {best_score:.6f}")
            sigma_rel = max(min_sigma_rel, sigma_rel * 0.992)
        else:
            no_accept += 1
            now = time.monotonic()
            if now - last_coord > coord_interval or no_accept > 15:
                remaining = deadline - now - 10.0
                cb = min(coord_budget, max(5.0, remaining * 0.05))
                fp, sc, cp = _coordinate_polish_wf(f, c, cb, rng)
                if sc < float(2.0 * n * maxc) - 1e-14:
                    f = _normalize(fp)
                    c = cp
                    maxc = float(c.max())
                    accepted = True
                    no_accept = 0
                    if sc < best_score:
                        best_score = sc
                        best_f = f.copy()
                        if best_score < TARGET_SCORE:
                            print(f"[{tag}] coord improved below target: {best_score:.6f}")
                else:
                    if no_accept > 40 or (it % 8000 == 0):
                        f = _perturb_from_best(best_f, rng)
                        c = _autocorr(f)
                        maxc = float(c.max())
                        sigma_rel = min(0.3, sigma_rel * 2.5 + 1e-4)
                        no_accept = 0
                last_coord = time.monotonic()
            else:
                sigma_rel = max(min_sigma_rel, sigma_rel * 0.997)

        if it % 30000 == 0:
            cur_score = float(2.0 * n * float(c.max()))
            print(f"[{tag}] iter={it} best={best_score:.6f} cur={cur_score:.6f} sigma={sigma_rel:.3e}")

    cur_score = float(2.0 * n * float(c.max()))
    if cur_score < best_score:
        best_score = cur_score
        best_f = f.copy()

    print(f"[{tag}] end best={best_score:.6f}")
    return _normalize(best_f), float(best_score)


def _downsample_to(f, K):
    f = _normalize(f)
    n = f.size
    if K <= 0:
        return np.empty(0, dtype=np.float64)
    if K >= n:
        return f
    x_old = np.linspace(0.0, n - 1, n, dtype=np.float64)
    x_new = np.linspace(0.0, K - 1, K, dtype=np.float64)
    return _normalize(np.interp(x_new, x_old, f))


def _upsample_to(f, n):
    return _interp_to_length(f, n)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative coefficient sequence minimizing evaluate_sequence.
    Uses water-filling coordinate descent + gradient annealing + anti-symmetric seeds.
    Returns a Python list of floats.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(120.0, float(budget_s)) - 20.0
    t_start = time.monotonic()

    # Load initial sequence
    try:
        f0 = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        f0 = np.ones(6121, dtype=np.float64)

    if f0.size == 0:
        f0 = np.ones(6121, dtype=np.float64)
    f0 = np.nan_to_num(f0, nan=0.0, posinf=1000.0, neginf=0.0)
    np.clip(f0, 0.0, 1000.0, out=f0)
    n = int(f0.size)
    if n < 3:
        f0 = np.ones(1024, dtype=np.float64)
        n = f0.size
    f0 = _normalize(f0)

    best_f = f0.copy()
    best_c = _autocorr(best_f)
    best_score = float(2.0 * n * best_c.max()) if best_c.size else np.inf
    print(f"Initial score: {best_score:.6f} (n={n})")

    # Build diverse seed pool
    seeds = []

    def try_add(name, fv):
        fv = _normalize(fv)
        try:
            sc = _fast_score(fv)
        except Exception:
            return
        if np.isfinite(sc):
            seeds.append((float(sc), len(seeds), name, fv))

    try_add("given", f0)
    try_add("reverse", f0[::-1])

    # Anti-symmetric seeds
    for i in range(4):
        try_add(f"antisym{i}", _anti_symmetric_seed(n, rng))

    # Ramp seeds
    try_add("ramp_up", _ramp_seed(n, rng, 1))
    try_add("ramp_down", _ramp_seed(n, rng, -1))

    # Fourier seeds
    for i in range(6):
        try_add(f"fourier{i}", _fourier_seed(n, rng))

    # Beta seeds
    for i in range(4):
        try_add(f"beta{i}", _beta_seed(n, rng))

    # Smoothed variants
    for win in (7, 21, 61, 201):
        try_add(f"smooth{win}", _smooth_with_window(f0, win))

    # Gamma transforms
    for g in (0.4, 0.6, 1.5, 2.5):
        base = f0 + 1e-12
        try_add(f"gamma{g}", np.power(base, g))

    # Uniform
    try_add("uniform", np.ones(n, dtype=np.float64))

    # Coarse upsampling
    if n > 512:
        for K in (128, 256, 512):
            coarse = _downsample_to(f0, K)
            try_add(f"coarse{K}", _upsample_to(coarse, n))

    seeds.sort()
    print(f"Seed pool: {len(seeds)} seeds, best={seeds[0][0]:.6f} ({seeds[0][2]})")

    # Coarse optimization phase
    now = time.monotonic()
    if total_deadline - now > 800.0 and n > 256:
        for K in (256, 512):
            if total_deadline - time.monotonic() < 80.0:
                break
            src = seeds[0][3]
            source = _downsample_to(src, K)
            cf, cs = _gradient_optimize_v2(source, 30.0, rng, f"coarse{K}")
            fin = _upsample_to(cf, n)
            try_add(f"coarse_lift_{K}", fin)
        seeds.sort()
        print(f"After coarse: best seed score={seeds[0][0]:.6f}")

    # Main optimization: run on top-3 seeds
    now = time.monotonic()
    remaining_main = total_deadline - now - 250.0  # reserve 250s for final polish
    if remaining_main < 60.0:
        remaining_main = max(0.0, total_deadline - now - 30.0)

    top = seeds[:3] if len(seeds) >= 3 else seeds
    weights = [0.55, 0.30, 0.15][:len(top)]

    for idx, item in enumerate(top):
        now = time.monotonic()
        remain = total_deadline - now - 250.0
        if remain < 30.0:
            break
        frac = weights[idx] / sum(weights[idx:])
        alloc = max(30.0, min(remain - 10.0, 0.92 * frac * remaining_main))
        sc, _id, name, fv = item
        print(f"\n=== Main optimize #{idx+1} seed={name} alloc={alloc:.1f}s best={best_score:.6f} ===")
        f_opt, sc_opt = _gradient_optimize_v2(fv, alloc, rng, f"main{name}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            best_c = _autocorr(best_f)
            print(f"  -> improved to {best_score:.6f}")

    # Final coordinate polish
    now = time.monotonic()
    final_time = max(0.0, total_deadline - now - 15.0)
    if final_time > 30.0:
        print(f"\n=== Final coordinate polish budget={final_time:.1f}s ===")
        bf, bs, bc = _coordinate_polish_wf(best_f, _autocorr(best_f), final_time, rng)
        if bs < best_score:
            best_score = bs
            best_f = bf.copy()
            best_c = bc
            print(f"  -> final polish: {best_score:.6f}")

    # Extra gradient pass if time remains
    now = time.monotonic()
    extra = max(0.0, total_deadline - now - 10.0)
    if extra > 60.0:
        grad_extra = min(120.0, 0.6 * extra)
        print(f"\n=== Extra gradient pass budget={grad_extra:.1f}s ===")
        f2, s2 = _gradient_optimize_v2(best_f, grad_extra, rng, "extra_grad")
        if s2 < best_score:
            best_score = s2
            best_f = f2.copy()
            best_c = _autocorr(best_f)

        now = time.monotonic()
        extra2 = max(0.0, total_deadline - now - 10.0)
        if extra2 > 20.0:
            bf, bs, bc = _coordinate_polish_wf(best_f, _autocorr(best_f), min(extra2, 100.0), rng)
            if bs < best_score:
                best_score = bs
                best_f = bf.copy()
                best_c = bc
                print(f"  -> final final: {best_score:.6f}")

    # Final safety
    best_f = _normalize(best_f)
    best_f = np.nan_to_num(best_f, nan=0.0, posinf=0.0, neginf=0.0)
    np.clip(best_f, 0.0, 1000.0, out=best_f)
    best_f = best_f / max(1e-300, float(best_f.sum()))
    best_c = _autocorr(best_f)
    best_score = float(2.0 * best_f.size * best_c.max())

    # Official evaluation
    try:
        out = [float(x) for x in best_f]
        official = evaluate_sequence(out)
        if np.isfinite(official) and official < best_score + 1e-12:
            best_score = float(official)
    except Exception:
        pass

    elapsed = time.monotonic() - t_start
    print(f"\nFINAL internal score = {best_score:.6f} (target={TARGET_SCORE})")
    print(f"Time elapsed: {elapsed:.1f}s")

    out = [float(x) for x in best_f]
    return out
