"""
Optimized search for non-negative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key improvements: exact subgradient with top-k peaks, soft-clipping frequency flattening,

import numpy as np
import time


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b, L=None):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 1500000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)


def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())


def _score_from_conv(p, c):
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    n = len(p)
    return float(2.0 * n * c.max() / (s * s))


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s


def _exact_subgrad_topk(p, k, L=None):
    """Compute subgradient using top-k peaks of the convolution."""
    n = len(p)
    c = _conv_full(p, p)
    cmax = c.max()
    if cmax < 1e-15:
        return np.zeros(n), cmax, c
    # Find top-k indices
    if k >= len(c):
        idxs = np.arange(len(c))
        weights = np.ones(len(idxs))
    else:
        # Use a threshold-based approach for robustness
        threshold = cmax * (1.0 - 0.05 * k)
        candidate_idxs = np.where(c >= threshold)[0]
        if len(candidate_idxs) == 0:
            candidate_idxs = np.array([np.argmax(c)])
        # Pick top k from candidates

            vals = c[candidate_idxs]
            top_local = np.argsort(vals)[-k:]
            candidate_idxs = candidate_idxs[top_local]
        idxs = candidate_idxs
        weights = c[idxs] / c[idxs].sum()
    # Compute subgradient: for each peak m, grad[i] += weight * p[m-i]
    grad = np.zeros(n)
    for m, w in zip(idxs, weights):
        m = int(m)
        i_lo = max(0, m - (n - 1))
        i_hi = min(n - 1, m)
        for i in range(i_lo, i_hi + 1):
            j = m - i
            if 0 <= j < n:
                grad[i] += w * p[j]
    return grad, cmax, c


def _multiplicative_update(p, grad, lr):
    """Apply multiplicative gradient update: p[i] *= exp(-lr * grad[i] / max_grad)."""
    max_grad = grad.max()
    if max_grad < 1e-15:
        return p.copy()
    p_new = p * np.exp(-lr * grad / max_grad)
    p_new = np.maximum(p_new, 1e-18 / len(p))
    s = p_new.sum()
    if s < 1e-10:
        return p.copy()
    return p_new / s


def _freq_flatten_power(p, alpha, L=None):
    """Flatten convolution via power transform on the convolution."""
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n)
    A = np.fft.rfft(p, L)
    C_freq = A * A
    C = np.fft.irfft(C_freq, L)[:2 * n - 1]
    cmax = C.max()
    if cmax < 1e-15:
        return p.copy()
    C_norm = C / cmax
    C_new = cmax * np.power(C_norm, 1.0 / alpha)
    C_new = np.maximum(C_new, 0.0)
    C_freq_new = np.fft.rfft(C_new, L)
    # Take square root in frequency domain

    a_new = np.fft.irfft(A_new, L)[:n]

        return None
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s


def _freq_flatten_clip(p, clip_frac, smooth, L=None):
    """Flatten convolution by soft-clipping the peaks."""
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n)
    A = np.fft.rfft(p, L)
    C_freq = A * A
    C = np.fft.irfft(C_freq, L)[:2 * n - 1]
    cmax = C.max()
    if cmax < 1e-15:
        return p.copy()
    # Soft clip: values above threshold are pulled down
    threshold = cmax * (1.0 - clip_frac)
    C_new = C.copy()
    mask = C > threshold
    if mask.any():
        excess = C[mask] - threshold
        C_new[mask] = threshold + excess * smooth
    # Renormalize to preserve sum
    s_orig = C.sum()
    s_new = C_new.sum()
    if s_new < 1e-10:
        return p.copy()
    C_new = C_new * (s_orig / s_new)
    C_new = np.maximum(C_new, 0.0)
    C_freq_new = np.fft.rfft(C_new, L)
    A_new = np.sqrt(C_freq_new.astype(complex) + 0j)
    a_new = np.fft.irfft(A_new, L)[:n]
    neg_frac = (a_new < 0).sum() / n
    if neg_frac > 0.35:
        return None
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:

            bs = int(rng.integers(max(8, n // 32), max(16, n // 4)))
        bs = min(bs, n)
        start = int(rng.integers(0, n - bs + 1))

        improved = False
        for f in factors:
            p_try = p.copy()
            p_try[start:start + bs] *= f
            p_try = _sanitize(p_try)
            if p_try is None:
                continue
            sc = _score(p_try)
            if sc < cur_score - 1e-13:
                cur_score = sc
                p = p_try
                improved = True
                break

        if improved:
            if cur_score < best_score - 1e-12:
                best = p.copy()
                best_score = cur_score
            stall = 0
        else:
            stall += 1
            if stall > 40:
                p = best.copy()
                cur_score = best_score
                stall = 0
        it += 1

    return best, best_score


def _optimize_main(p0, rng, time_budget, topk=3, verbose=False):
    """Main optimization: multiplicative gradient with exact subgradient + freq flattening."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)

    best = p.copy()
    best_score = _score(p)
    cur_score = best_score

    lr = max(0.5, 2.0 * n / 100.0)
    lr_base = lr
    deadline = time.monotonic() + time_budget
    stall = 0
    it = 0

    if verbose:
        print(f"  Main opt: n={n}, init_score={best_score:.6f}, budget={time_budget:.0f}s")

    while time.monotonic() < deadline:
        r = rng.random()

        if r < 0.60:
            # Gradient step with exact subgradient
            grad, cmax, c = _exact_subgrad_topk(p, topk)
            max_grad = grad.max()
            if max_grad < 1e-15:
                stall += 1
                continue

            # Line search: try multiple step sizes
            best_trial = None
            best_trial_sc = cur_score
            for lr_try in (lr * 0.25, lr * 0.5, lr, lr * 2.0, lr * 4.0):
                p_try = _multiplicative_update(p, grad, lr_try)
                sc_try = _score_from_conv(p_try, _conv_full(p_try, p_try))
                if sc_try < best_trial_sc - 1e-14:
                    best_trial = p_try
                    best_trial_sc = sc_try

            if best_trial is not None:
                p = best_trial
                cur_score = best_trial_sc
                stall = 0
                if cur_score < best_score - 1e-12:
                    best = p.copy()
                    best_score = cur_score
                    if verbose and it % 500 == 0:
                        print(f"    iter={it}, score={best_score:.6f}, lr={lr:.1f}")
            else:
                stall += 1
                if stall > 5:
                    lr = max(0.1, lr * 0.6)
                if stall > 30:
                    p = best.copy()
                    cur_score = best_score
                    lr = lr_base
                    stall = 10

        elif r < 0.75:
            # Frequency flattening (power)
            alpha_f = float(rng.uniform(1.5, 4.0))
            p_new = _freq_flatten_power(p, alpha_f)
            if p_new is not None:
                sc_new = _score(p_new)
                if sc_new < cur_score - 1e-13:
                    # Blend
                    t_blend = min(1.0, 0.3 + 0.1 * (cur_score - sc_new) * 100)
                    p_new_blend = (1 - t_blend) * p + t_blend * p_new
                    p_new_blend = _sanitize(p_new_blend)
                    if p_new_blend is not None:
                        sc_blend = _score(p_new_blend)
                        if sc_blend <= sc_new + 1e-12:
                            p = p_new_blend
                            cur_score = sc_blend
                        else:
                            p = p_new
                            cur_score = sc_new
                    else:
                        p = p_new
                        cur_score = sc_new
                    stall = 0
                    if cur_score < best_score - 1e-12:
                        best = p.copy()
                        best_score = cur_score
                        if verbose:
                            print(f"    FREQ_P iter={it}, alpha={alpha_f:.1f}, score={best_score:.6f}")
                    else:
                        stall += 1
                else:
                    stall += 1
            else:
                stall += 1

        elif r < 0.85:
            # Frequency flattening (clipping)
            clip_frac = float(rng.uniform(0.05, 0.3))
            smooth = float(rng.uniform(0.05, 0.4))
            p_new = _freq_flatten_clip(p, clip_frac, smooth)
            if p_new is not None:
                sc_new = _score(p_new)
                if sc_new < cur_score - 1e-13:
                    t_blend = min(1.0, 0.3 + 0.1 * (cur_score - sc_new) * 100)
                    p_new_blend = (1 - t_blend) * p + t_blend * p_new
                    p_new_blend = _sanitize(p_new_blend)
                    if p_new_blend is not None:
                        sc_blend = _score(p_new_blend)
                        if sc_blend <= sc_new + 1e-12:
                            p = p_new_blend
                            cur_score = sc_blend
                        else:
                            p = p_new
                            cur_score = sc_new
                    else:
                        p = p_new
                        cur_score = sc_new
                    stall = 0
                    if cur_score < best_score - 1e-12:
                        best = p.copy()
                        best_score = cur_score
                        if verbose:
                            print(f"    FREQ_C iter={it}, clip={clip_frac:.2f}, score={best_score:.6f}")
                    else:
                        stall += 1
                else:
                    stall += 1
            else:
                stall += 1

        elif r < 0.92:
            # Block perturbation
            bs = int(rng.choice([1, 2, 4, 8, 16, 32, 64, 128]))
            bs = min(bs, n)
            start = int(rng.integers(0, n - bs + 1))
            f = float(rng.choice([0.3, 0.5, 0.7, 1.4, 1.8, 2.5, 4.0]))
            p_try = p.copy()
            p_try[start:start + bs] *= f
            p_try = _sanitize(p_try)
            if p_try is not None:
                sc_try = _score(p_try)
                if sc_try < cur_score - 1e-13:
                    p = p_try
                    cur_score = sc_try
                    stall = 0
                    if sc_try < best_score - 1e-12:
                        best = p.copy()
                        best_score = sc_try
                else:
                    stall += 1
        else:
            # Random perturbation (annealing)
            if stall > 8:
                std = 0.01 + 0.005 * min(stall, 40)
                p_try = p * np.exp(rng.normal(0.0, std, n))
                p_try = _sanitize(p_try)
                if p_try is not None:
                    sc_try = _score(p_try)
                    # Accept with probability based on improvement or annealing
                    if sc_try < best_score * 1.05:
                        p = p_try
                        cur_score = sc_try
                        stall = 0
            else:
                stall += 1

        # Periodic learning rate adjustment
        if it % 50 == 0 and stall < 5:
            lr = min(lr_base * 3, lr * 1.1)
        if it % 50 == 0 and stall > 15:
            lr = max(0.1, lr * 0.7)

        # Pullback
        if cur_score > best_score * 1.1 and best_score < float('inf'):
            if rng.random() < 0.2:
                p = best.copy()
                cur_score = best_score
                stall = 0
                lr = lr_base

        it += 1

    return best, best_score


def _optimize_multiscale(p0, rng, time_budget):
    """Multi-resolution: downsample, optimize, upsample."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    best = p.copy()
    best_score = _score(p)

    deadline = time.monotonic() + time_budget

    # Try several downsampled resolutions
    resolutions = []
    for factor in (2, 4, 8, 16):
        n_small = n // factor
        if n_small >= 64:
            resolutions.append(n_small)
    resolutions.sort(reverse=True)

    for n_small in resolutions:
        if time.monotonic() > deadline:
            break
        # Downsample
        x_old = (np.arange(n) + 0.5) / n
        x_new = (np.arange(n_small) + 0.5) / n_small
        p_small = np.interp(x_new, x_old, best)
        p_small = np.maximum(p_small, 0)
        s = p_small.sum()
        if s < 1e-10:
            continue
        p_small = p_small / s

        # Optimize at small resolution
        t_alloc = max(15.0, min(120.0, (deadline - time.monotonic()) * 0.5))
        p_opt, sc_opt = _optimize_main(p_small, rng, t_alloc, topk=2, verbose=False)
        if p_opt is not None and sc_opt < float('inf'):
            # Upsample back
            x_old2 = (np.arange(n_small) + 0.5) / n_small
            x_new2 = (np.arange(n) + 0.5) / n
            p_up = np.interp(x_new2, x_old2, p_opt)
            p_up = np.maximum(p_up, 0)
            s2 = p_up.sum()
            if s2 < 1e-10:
                continue
            p_up = p_up / s2
            sc_up = _score(p_up)
            # Fine-tune at full resolution
            t_fine = max(20.0, min(150.0, (deadline - time.monotonic()) * 0.5))
            p_fine, sc_fine = _optimize_main(p_up, rng, t_fine, topk=3, verbose=False)
            if p_fine is not None and sc_fine < float('inf'):
                if sc_fine < best_score:
                    best = p_fine.copy()
                    best_score = sc_fine
                    if verbose:
                        print(f"  MultiScale: n_small={n_small}, score={best_score:.6f}")
                elif sc_fine < best_score * 1.02:
                    # Use as starting point for further optimization
                    pass

    return best, best_score


def _init_structured(n, rng):
    """Initialize with structured patterns that might have good autocorrelation."""
    r = rng.random()
    if r < 0.2:
        # Two-level random
        k = int(n * rng.uniform(0.3, 0.7))
        ratio = rng.uniform(1.5, 4.0)
        p = np.ones(n)
        p[k:] = ratio
        idx = rng.permutation(n)
        p = p[idx]
    elif r < 0.4:
        # Gaussian mixture
        K = int(rng.integers(2, 6))
        p = np.zeros(n)
        t = np.arange(n, dtype=float) / n
        for _ in range(K):
            c = rng.uniform(0, 1)
            s = rng.uniform(0.05, 0.25)
            w = rng.uniform(0.5, 2.0)
            x = np.minimum(np.abs(t - c), 1.0 - np.abs(t - c)) / s
            p += w * np.exp(-0.5 * x * x)
        p = np.maximum(p, 1e-10)
    elif r < 0.6:
        # Cosine combination
        t = np.arange(n, dtype=float) / n
        p = np.ones(n)
        for k in range(1, 12):
            amp = rng.uniform(-0.5, 0.5) / (k * 0.5)
            ph = rng.uniform(0, 2 * np.pi)
            p += amp * np.cos(2 * np.pi * k * t + ph)
        p = np.maximum(p, 0)
        s = p.sum()
        if s < 1e-10:
            p = np.ones(n)
    elif r < 0.8:
        # Block random
        K = int(rng.integers(4, 16))
        p = np.zeros(n)
        edges = np.sort(rng.uniform(0, 1, K - 1))
        blocks = np.concatenate([[0], edges, [1]])
        heights = rng.lognormal(0, 0.8, K)
        for i in range(K):
            s = int(round(blocks[i] * n))
            e = max(s + 1, int(round(blocks[i + 1] * n)))
            e = min(n, e)
            p[s:e] = heights[i]
        p = np.maximum(p, 1e-10)
    else:
        # Random with smoothing
        p = rng.uniform(0.5, 1.5, n)
        # Smooth with a window
        w = max(3, n // 200)
        kernel = np.ones(w) / w
        p = np.convolve(p, kernel, mode='same')
        p = np.maximum(p, 0)

    s = p.sum()
    if s < 1e-10:
        return np.ones(n, dtype=float) / n
    return p / s


def _resample(p_in, n_new, rng=None):
    p = _sanitize(p_in)
    if p is None:
        return None
    n_old = len(p)
    if n_old == n_new:
        result = p.copy()
    elif n_old == 1:
        result = np.ones(n_new, dtype=float)
    else:
        x_old = (np.arange(n_old) + 0.5) / n_old
        x_new = (np.arange(n_new) + 0.5) / n_new
        result = np.interp(x_new, x_old, p)
    result = np.maximum(result, 0)
    if rng is not None:
        result = result * np.exp(rng.normal(0, 0.02, n_new))
        result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s


def _try_upsample_improve(p_best, score_best, rng, time_budget, n_max=8192):
    """Try upsampled versions of the best sequence."""
    best = p_best.copy()
    best_score = score_best
    n = len(p_best)
    deadline = time.monotonic() + time_budget

    for n_new in (n * 2, n * 4):
        if n_new > n_max or time.monotonic() > deadline:
            break
        t_rem = deadline - time.monotonic()
        if t_rem < 60:
            break
        p_up = _resample(p_best, n_new, rng)
        if p_up is None:
            continue
        t_alloc = min(200.0, t_rem * 0.5)
        t_alloc = max(30.0, t_alloc)
        p_opt, sc_opt = _optimize_main(p_up, rng, t_alloc, topk=3, verbose=False)
        if p_opt is not None and sc_opt < best_score:
            best = p_opt.copy()
            best_score = sc_opt
            print(f"  Upsample to n={n_new}: score={best_score:.6f}")

    return best, best_score


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Search for best non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses exact subgradient optimization with frequency-domain flattening,
    multi-scale refinement, and adaptive exploration.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 2400.0

    start_time = time.monotonic()
    deadline = start_time + budget - 20

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng(iseed % (2**31 - 1))

    # Get parent sequence
    try:
        parent = _sanitize(np.asarray(height_sequence_1, dtype=float))
    except Exception:
        parent = None

    best_overall = None
    best_score_overall = float('inf')

    def _try_update(p, score):
        nonlocal best_overall, best_score_overall
        if p is not None and np.isfinite(score) and score < best_score_overall:
            best_overall = p.copy()
            best_score_overall = float(score)
            print(f"  *** NEW BEST: {score:.6f} (n={len(p)}) ***")

    # ========== PHASE 1: Quick exploration (12% of time) ==========
    print("=" * 60)
    print(f"Phase 1: Exploration (budget={budget:.0f}s)")
    t1_end = start_time + 0.12 * budget

    if parent is not None:
        sc = _score(parent)
        _try_update(parent, sc)
        print(f"  Parent score: {sc:.6f} (n={len(parent)})")

    # Try several initializations
    explore_lengths = [2048, 3072, 4096]
    if parent is not None:
        if len(parent) not in explore_lengths:
            explore_lengths.append(len(parent))
            explore_lengths.sort()

    for n in explore_lengths:
        if time.monotonic() > t1_end:
            break
        for init_fn_idx in range(4):
            sp = _init_structured(n, rng)
            sc = _score(sp)
            _try_update(sp, sc)
            if init_fn_idx == 0:
                print(f"  n={n}: best init score={sc:.6f}")

    if best_overall is None:
        best_overall = _init_structured(3072, rng)
        best_score_overall = _score(best_overall)

    print(f"  Phase 1 done. Best: {best_score_overall:.6f} (n={len(best_overall)})")

    # ========== PHASE 2: Main optimization (48% of time) ==========
    print("Phase 2: Main optimization")
    t2_end = start_time + 0.60 * budget

    n_main = len(best_overall)
    t_rem = t2_end - time.monotonic()

    # Run 1: Standard optimization
    if t_rem > 60:
        t_alloc = min(0.40 * (t2_end - start_time - 0.12 * budget), t_rem * 0.45, 400.0)
        t_alloc = max(30.0, t_alloc)
        p, sc = _optimize_main(best_overall, rng, t_alloc, topk=3, verbose=True)
        _try_update(p, sc)

    # Run 2: Higher topk (broader subgradient)
    t_rem = t2_end - time.monotonic()
    if t_rem > 60:
        t_alloc = min(250.0, t_rem * 0.4)
        t_alloc = max(30.0, t_alloc)
        p1 = _resample(best_overall, n_main, rng)
        if p1 is None:
            p1 = best_overall.copy()
        p, sc = _optimize_main(p1, rng, t_alloc, topk=5, verbose=True)
        _try_update(p, sc)

    # Run 3: Different topk
    t_rem = t2_end - time.monotonic()
    if t_rem > 60:
        t_alloc = min(200.0, t_rem * 0.4)
        t_alloc = max(30.0, t_alloc)
        p1 = _resample(best_overall, n_main, rng)
        if p1 is None:
            p1 = best_overall.copy()
        p, sc = _optimize_main(p1, rng, t_alloc, topk=1, verbose=True)
        _try_update(p, sc)

    # Run 4: Block refinement
    t_rem = t2_end - time.monotonic()
    if t_rem > 40:
        t_alloc = min(150.0, t_rem * 0.5)
        t_alloc = max(20.0, t_alloc)
        p, sc = _block_refine(best_overall, rng, t_alloc)
        _try_update(p, sc)

    print(f"  Phase 2 done. Best: {best_score_overall:.6f} (n={len(best_overall)})")

    # ========== PHASE 3: Multi-scale and upsample (15% of time) ==========
    print("Phase 3: Multi-scale and upsample")
    t3_end = start_time + 0.78 * budget

    t_rem = t3_end - time.monotonic()
    if t_rem > 60:
        # Multi-scale
        t_alloc = min(200.0, t_rem * 0.5)
        p, sc = _optimize_multiscale(best_overall, rng, t_alloc)
        _try_update(p, sc)

        # Try upsample
        t_rem = t3_end - time.monotonic()
        if t_rem > 60:
            t_alloc = min(150.0, t_rem * 0.6)
            p, sc = _try_upsample_improve(best_overall, best_score_overall, rng, t_alloc)
            _try_update(p, sc)

    print(f"  Phase 3 done. Best: {best_score_overall:.6f} (n={len(best_overall)})")

    # ========== PHASE 4: Diverse restarts (10% of time) ==========
    print("Phase 4: Diverse restarts")
    t4_end = start_time + 0.90 * budget

    t_rem = t4_end - time.monotonic()
    if t_rem > 60:
        n_best = len(best_overall)
        for restart in range(8):
            if time.monotonic() > t4_end - 15:
                break
            r = rng.random()
            if r < 0.3:
                sp = _init_structured(n_best, rng)
            elif r < 0.6:
                sp = _resample(best_overall, n_best, rng)
                if sp is None:
                    sp = best_overall.copy()
            else:
                sp = _resample(best_overall, n_best, rng)
                if sp is not None:
                    sp = sp * np.exp(rng.normal(0, 0.1, n_best))
                    sp = _sanitize(sp)
                if sp is None:
                    sp = best_overall.copy()

            t_alloc = min(50.0, (t4_end - time.monotonic()) / max(1, 8 - restart))
            t_alloc = max(15.0, t_alloc)
            p, sc = _optimize_main(sp, rng, t_alloc, topk=3, verbose=False)
            _try_update(p, sc)

    print(f"  Phase 4 done. Best: {best_score_overall:.6f}")

    # ========== PHASE 5: Final polishing (10% of time) ==========
    print("Phase 5: Final polishing")

    if best_overall is not None:
        # Fine gradient steps
        t_rem = deadline - time.monotonic()
        if t_rem > 30:
            t_alloc = min(0.35 * t_rem, 180.0)
            p, sc = _optimize_main(best_overall, rng, t_alloc, topk=5, verbose=True)
            _try_update(p, sc)

        # Block refine
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.25 * t_rem, 100.0)
            p, sc = _block_refine(best_overall, rng, t_alloc)
            _try_update(p, sc)

        # Final fine steps
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.35 * t_rem, 120.0)
            p, sc = _optimize_main(best_overall, rng, t_alloc, topk=2, verbose=True)
            _try_update(p, sc)

    # ========== Final output ==========
    if best_overall is None:
        best_overall = np.ones(3072, dtype=float) / 3072

    p_final = _sanitize(best_overall)
    if p_final is None:
        p_final = np.ones(3072, dtype=float) / 3072

    final_score = _score(p_final)
    try:
        eval_score = float(evaluate_sequence([float(x) for x in p_final]))
        if np.isfinite(eval_score):
            final_score = min(final_score, eval_score)
    except Exception:
        pass

    print(f"\n{'=' * 60}")
    print(f"FINAL: score={final_score:.6f} (n={len(p_final)})")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    out = np.where(np.isfinite(p_final), p_final, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = np.ones(3072, dtype=float) / 3072
    else:
        out = out / s

    return [float(x) for x in out]
