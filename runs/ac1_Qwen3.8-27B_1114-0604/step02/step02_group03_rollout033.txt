Step-function optimizer using central binomial coefficient seeds and
aggressive max-subgradient descent with coordinate polish.

Key insight: The linear self-convolution of truncating sqrt(1+x+x^2+...)
(given by central binomial coefficients / 4^k) produces a perfectly flat
first n lags, giving an initial score of ~1.54 (vs 2.0 for uniform).
We then optimize to flatten the remaining lags using exact/max-subgradient
descent and coordinate transfer polish.

def _normalize(v):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v


def _fft_size(m):
    if m <= 2:
        return 2
    return 1 << (int(m) - 1).bit_length()


def _autocorr(f):
    n = f.size
    m = 2 * n - 1
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]


def _fast_score(f):
    n = f.size
    c = _autocorr(f)
    if c.size == 0:
        return np.inf, c
    return float(2.0 * f.size * c.max()), c


def _central_binomial_seed(n):
    b = np.ones(n, dtype=np.float64)
    for k in range(1, n):
        b[k] = b[k - 1] * (2 * k - 1) / (2 * k)
    return b


def _foursier_seed(n, rng, alpha):
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    u = np.zeros(n, dtype=np.float64)
    for m in (1, 2, 3, 5, 8, 13, 21):
        amp = 0.5 / math.sqrt(m)
        u += float(rng.uniform(-1.0, 1.0)) * amp * np.cos(
            2.0 * np.pi * m * x + float(rng.uniform(0.0, 2.0 * np.pi))
        )
    f = np.exp(alpha * u)
    return _normalize(f)

def _gradient_at_lags(c, f, k_set):
    """Compute sum_{k in k_set} dc[k]/df as a vector of length n."""
    n = f.size
    grad = np.zeros(n, dtype=np.float64)
    size = _fft_size(2 * n)
    F = np.fft.rfft(f, size)
    freqs = np.fft.rfftfreq(size)
    for k in k_set:
        k = int(k)
        # dc[k]/df[j] = f[k-j] (if 0<=k-j<n) + f[j-k] (if 0<=j-k<n)
        # In circular FFT: this is f shifted by k plus f shifted by -k (circularly)
        # For linear, we need to be careful at boundaries, but for the argmax set
        # which is typically in the middle range, circular approx is good.
        phase = np.exp(-2j * np.pi * k * freqs)
        # Circular correlation: ifft(FFT(f) * conj(phase)) gives f shifted
        # Actually: circular cross-correlation of f with delta_k is f shifted by k
        # In FFT: IFFT(FFT(f) * exp(2*pi*i*k*m/N)) = f circularly shifted by k
        # But we want the LINEAR version. Let's use the property:
        # dc[k]/df = IFFT(FFT(f) * 2*real(FFT(delta_k))) for circular
        # FFT(delta_k) = exp(-2*pi*i*k*m/N) for m = 0,...,N-1
        # 2*real(exp(-2*pi*i*k*m/N)) = 2*cos(2*pi*k*m/N)
        cos_term = 2.0 * np.cos(2.0 * np.pi * k * freqs * size)
        shifted = np.fft.irfft(F * cos_term, size)
        grad += shifted[:n]
    grad /= float(len(k_set))
    return grad


def _gradient_fast(c, f, top_k=None):

def _coordinate_polish(f0, c0, budget_s, rng):

    # Compute gradient
    g = _gradient_at_lags(c, f, k_set)
    g = g - np.mean(g)  # center to preserve sum
    
    scale = float(np.max(np.abs(g)))
    if scale < 1e-300:
        return f, c, maxc, None
    
    best_mc = maxc
    best_f = None
    best_c = None
    
    for e in step_sizes:
        # Multiplicative (log-domain) step
        arg = -e * (g / scale)
        np.clip(arg, -30.0, 30.0, out=arg)
        ft = f * np.exp(arg)
        ft = _normalize(ft)
        ct = _autocorr(ft)
        mc = float(ct.max())
        if mc < best_mc:
            best_mc = mc
            best_f = ft
            best_c = ct
        
        # Additive (Euclidean) step
        vt = f - e * (g / scale)
        ft2 = np.maximum(vt, 0.0)
        s = ft2.sum()
        if s > 1e-300:
            ft2 /= s
        ct2 = _autocorr(ft2)
        mc2 = float(ct2.max())
        if mc2 < best_mc:
            best_mc = mc2
            best_f = ft2
            best_c = ct2
    
    if best_c is None:
        return f, c, maxc, None
    return f, c, maxc, (best_f, best_c, best_mc)


def _optimize(f0, budget_s, rng, tag=""):
    """Main optimization loop."""
    f = _normalize(f0)
    n = f.size
    if n == 0 or budget_s <= 0:
        c = _autocorr(f)
        return f, float(2.0 * n * c.max())
    
    c = _autocorr(f)
    maxc = float(c.max())
    best_f = f.copy()
    best_score = float(2.0 * n * maxc)
    
    deadline = time.monotonic() + float(budget_s)
    sigma = 0.1
    min_sigma = 1e-12
    no_accept = 0
    it = 0
    last_coord = time.monotonic()
    coord_interval = 0.5
    coord_budget = 0.3
    
    print(f"[{tag}] start: score={best_score:.6f}, n={n}")
    
    while time.monotonic() < deadline:
        it += 1
        now = time.monotonic()
        
        if now - last_coord > coord_interval or no_accept > 15:
            # Coordinate polish
            remaining = deadline - now
            if remaining > coord_budget + 10:
                fp, sc, cp = _coordinate_polish(f, c, coord_budget, rng)
                if sc < best_score:
                    best_score = sc
                    best_f = fp.copy()
                    if best_score < TARGET_SCORE:
                        print(f"[{tag}] polish below target: {best_score:.6f}")
                if sc < float(2.0 * n * maxc) - 1e-12:
                    f = _normalize(fp)
                    c = cp
                    maxc = float(c.max())
                    no_accept = 0
                last_coord = now
                continue
        
        # Gradient step
        step_sizes = (0.01, 0.05, 0.1, 0.3, 1.0)
        result = _gradient_step(c, f, step_sizes)
        best_mc_from_grad = result[3]
        
        accepted = False
        if best_mc_from_grad is not None:
            gf, gc, gmc = best_mc_from_grad
            if gmc < maxc - 1e-14:
                f = gf
                c = gc
                maxc = gmc
                accepted = True
                no_accept = 0
                cur_score = float(2.0 * n * maxc)
                if cur_score < best_score:
                    best_score = cur_score
                    best_f = f.copy()
                    if best_score < TARGET_SCORE:
                        print(f"[{tag}] gradient below target: {best_score:.6f}")
                sigma = max(min_sigma, sigma * 0.995)
            else:
                no_accept += 1
                sigma = max(min_sigma, sigma * 0.95)
                # Try perturbation after many failures
                if no_accept > 40:
                    # Perturb from best
                    alpha = float(rng.uniform(0.05, 0.3))
                    noise = rng.random(n)
                    s = noise.sum()
                    if s > 1e-300:
                        noise /= s
                    f = (1.0 - alpha) * _normalize(best_f) + alpha * noise
                    f = _normalize(f)
                    c = _autocorr(f)
                    maxc = float(c.max())
                    sigma = min(0.3, sigma * 3.0 + 0.01)
                    no_accept = 0
        else:
            no_accept += 1
            if no_accept > 20:
                f = _normalize(best_f * np.exp(float(rng.uniform(0.05, 0.3)) * rng.standard_normal(n)))
                c = _autocorr(f)
                maxc = float(c.max())
                sigma = min(0.3, sigma * 2.0 + 0.01)
                no_accept = 0
        
        if it % 5000 == 0:
            cur_score = float(2.0 * n * maxc)
            print(f"[{tag}] it={it} best={best_score:.6f} cur={cur_score:.6f} sigma={sigma:.4f}")
    
    # Final check
    c = _autocorr(f)
    sc = float(2.0 * n * c.max())
    if sc < best_score:
        best_score = sc
        best_f = f.copy()
    
    print(f"[{tag}] done: best={best_score:.6f}")
    return _normalize(best_f), float(best_score)


def _build_seeds(n, f_given, rng):
    """Build a pool of diverse seeds."""
    seeds = []
    
    # Central binomial (primary seed)
    cb = _central_binomial_seed(n)
    seeds.append(("cb", _normalize(cb)))
    seeds.append(("cb_rev", _normalize(cb[::-1])))
    
    # Blended central binomial
    for frac in (0.1, 0.25, 0.5, 0.75):
        seeds.append((f"cb_blend{frac}", _blended_seed(n, rng, frac)))
    
    # Given sequence
    if f_given is not None and f_given.size == n:
        fg = _normalize(f_given)
        seeds.append(("given", fg))
        seeds.append(("given_rev", _normalize(fg[::-1])))
    
    # Fourier seeds
    for i in range(8):
        alpha = float(rng.uniform(0.3, 1.5))
        seeds.append((f"fourier{i}", _foursier_seed(n, rng, alpha)))
    
    # Smoothed CB
    from numpy import convolve as npconv
    for win in (7, 15, 31):
        kernel = np.ones(win) / win
        sm = npconv(cb, kernel, mode='same')
        seeds.append((f"cb_smooth{win}", _normalize(sm)))
    
    # Evaluate and sort
    scored = []
    for name, f in seeds:
        sc, _ = _fast_score(f)
        if np.isfinite(sc):
            scored.append((sc, name, f))
    
    scored.sort(key=lambda x: x[0])
    return scored


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative coefficient sequence minimizing evaluate_sequence.
    Uses central binomial coefficient seeds and gradient/coordinate optimization.
    Returns a Python list of floats.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_deadline = t_start + max(100.0, float(budget_s)) - 30.0
    
    # Get initial sequence
    try:
        f_given = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        f_given = None
    
    if f_given is not None and f_given.size > 0:
        n = int(f_given.size)
    else:
        n = 6121
    
    if n < 3:
        n = 6121
        f_given = np.ones(n, dtype=np.float64)
    
    if f_given is not None:
        f_given = np.nan_to_num(f_given, nan=0.0, posinf=1000.0, neginf=0.0)
        np.clip(f_given, 0.0, 1000.0, out=f_given)
    
    print(f"n = {n}, budget = {budget_s}s")
    
    # Build seed pool
    seeds = _build_seeds(n, f_given, rng)
    print(f"Built {len(seeds)} seeds, best initial: {seeds[0][0]} score={seeds[0][1]:.6f}")
    
    # Track global best
    best_f = seeds[0][2].copy()
    best_score = seeds[0][1]
    
    # Allocate time to top seeds
    # Give most time to the best seed, some to others
    top_seeds = seeds[:min(6, len(seeds))]
    
    # Time allocation: first seed gets 50%, second 20%, third 12%, rest split
    if len(top_seeds) >= 3:
        weights = [0.45, 0.20, 0.12, 0.10, 0.07, 0.06][:len(top_seeds)]
    else:
        weights = [0.5, 0.3, 0.2][:len(top_seeds)]
    
    for idx, (sc, name, f) in enumerate(top_seeds):
        remaining = total_deadline - time.monotonic()
        if remaining < 30:
            break
        
        # Allocate time proportionally
        total_w = sum(weights[idx:])
        alloc = max(20.0, (remaining - 20.0) * weights[idx] / total_w)
        # Cap first seed
        if idx == 0:
            alloc = min(alloc, (remaining - 20.0) * 0.55)
        
        print(f"\n=== Optimizing seed '{name}' (initial={sc:.6f}), alloc={alloc:.1f}s, global_best={best_score:.6f} ===")
        
        f_opt, sc_opt = _optimize(f, alloc, rng, f"seed_{name}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  NEW GLOBAL BEST: {best_score:.6f}")
    
    # Final polish on the global best
    remaining = total_deadline - time.monotonic()
    if remaining > 30:
        print(f"\n=== Final polish, budget={remaining:.1f}s ===")
        bf, bs, bc = _coordinate_polish(best_f, _autocorr(best_f), min(remaining - 10, 120.0), rng)
        if bs < best_score:
            best_score = bs
            best_f = bf.copy()
            print(f"  FINAL POLISH BEST: {best_score:.6f}")
    
    # One more gradient pass if time allows
    remaining = total_deadline - time.monotonic()
    if remaining > 60:
        print(f"\n=== Final gradient pass, budget={remaining:.1f}s ===")
        f2, s2 = _optimize(best_f, min(remaining - 10, 100.0), rng, "final_grad")
        if s2 < best_score:
            best_score = s2
            best_f = f2.copy()
    
    # Final safety
    best_f = _normalize(best_f)
    best_f = np.nan_to_num(best_f, nan=0.0, posinf=0.0, neginf=0.0)
    np.clip(best_f, 0.0, 1000.0, out=best_f)
    
    # Verify with internal score
    c = _autocorr(best_f)
    final_score = float(2.0 * best_f.size * c.max())
    print(f"\nInternal final score: {final_score:.6f}")
    
    # Official evaluation
    try:
        out = [float(x) for x in best_f]
        official = evaluate_sequence(out)
        if np.isfinite(official):
            print(f"Official score: {official:.6f}")
            if official < final_score:
                final_score = float(official)
    except Exception:
        pass
    
    print(f"FINAL: {final_score:.6f} (target: {TARGET_SCORE})")
    return [float(x) for x in best_f]
