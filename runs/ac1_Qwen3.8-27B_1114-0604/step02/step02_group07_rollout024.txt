```python
import numpy as np
import time

def fast_self_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def softmax_grad(a, c, temp, n_active=1):
    """Compute gradient of soft-max objective."""
    n = len(a)
    S = float(a.sum())
    
    # Find top peaks
    k_act = min(n_active, c.size)
    part = np.argpartition(c, -k_act)[-k_act:]
    
    # Softmax weights
    vals = c[part]
    scaled = vals / temp
    scaled -= scaled.max()
    wts = np.exp(scaled)
    wts /= wts.sum()
    M_soft = float(np.dot(vals, wts))
    
    # Gradient of soft-max w.r.t. a
    grad_c = np.zeros(n)
    for i in range(k_act):
        wi = float(wts[i])
        if wi <= 1e-15:
            continue
        ki = int(part[i])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        grad_c[js:je+1] += wi * 2.0 * a[js:je+1][::-1]
    
    # Full gradient: d/d a [2n * M_soft / S^2]
    grad = (2.0 * n) * (grad_c / (S*S) - 2.0 * M_soft / (S**3))
    return grad, M_soft, S

def mirror_descent(a, grad, lr):
    """Multiplicative update for non-negative variables."""
    a = a * np.exp(-lr * grad)
    a = np.clip(a, 0, 1000)
    return a

def adam_step(a, g, m, v, t, lr):
    t1 = 0.9 ** t
    t2 = 0.999 ** t
    m[:] = 0.9 * m + 0.1 * g
    v[:] = 0.999 * v + 0.001 * g * g
    a -= lr * (m / (1.0 - t1)) / (np.sqrt(v / (1.0 - t2)) + 1e-8)
```

```python
def solve_lp_direction(f, M, K=50):
    """Solve reduced LP to find descent direction."""
    from scipy.optimize import linprog
    from scipy.sparse import csc_matrix, eye as speye
    
    n = len(f)
    c = fast_self_conv(f)
    
    # Select top K positions
    top_k = np.argpartition(c, -K)[-K:]
    
    # Build constraint matrix F (K x n)
    # F[k, j] = f[j] if 0 <= k-j < n
    rows, cols, data = [], [], []
    for idx, ki in enumerate(top_k):
        ki = int(ki)
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        for j in range(js, je + 1):
            rows.append(idx)
            cols.append(j)
            data.append(f[j])
    
    F = csc_matrix((data, (rows, cols)), shape=(K, n))
    
    # Primal: max 1^T g s.t. Fg <= M*1, g >= 0
    # In linprog form: min -1^T g s.t. Fg <= M*1, -g <= 0
    A_ub = csc_matrix.vstack([F, -speye(n, format='csr')])
    b_ub = np.concatenate([M * np.ones(K), np.zeros(n)])
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, method='highs')
    
    if res.status == 0:
        return res.x
    else:
        return None
```

```python
LP-guided optimization with Adam refinement for minimizing 2n*max(conv(a,a))/(sum a)^2.
Uses reduced linear programs to find descent directions that reduce the peak autocorrelation.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csc_matrix, eye as speye

def fast_self_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def solve_lp_direction(f, M, K, rng):
    """Solve reduced LP: max sum(g) s.t. (f*g)_k <= M for selected k, g >= 0.
    Returns g or None if LP fails."""
    n = len(f)
    c = fast_self_conv(f)
    
    # Select K positions with highest autocorrelation
    # Add some randomness for diversity
    top_K = np.argpartition(c, -K)[-K:]
    
    # Build constraint matrix F (K x n), sparse
    rows = []
    cols = []
    data = []
    for idx, ki in enumerate(top_K):
        ki = int(ki)
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        for j in range(js, je + 1):
            rows.append(idx)
            cols.append(j)
            data.append(f[j])
    
    if len(data) == 0:
        return None
    
    F = csc_matrix((data, (rows, cols)), shape=(K, n))
    
    # LP: max 1^T g s.t. Fg <= M*1, g >= 0
    c_obj = -np.ones(n)
    b_ub = M * np.ones(K)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=F, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            g = res.x
            g = np.clip(g, 0, 1000)
            if g.sum() < 1e-10:
                return None
            return g
    except Exception:
        pass
    return None

def softmax_gradient(a, c, temp, K):
    """Compute gradient of soft-max objective."""
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S
    
    k_act = min(K, c.size)
    part = np.argpartition(c, -k_act)[-k_act:]
    vals = c[part]
    
    # Softmax weights
    scaled = vals / temp
    scaled -= scaled.max()
    wts = np.exp(scaled)
    wts /= wts.sum()
    M_soft = float(np.dot(vals, wts))
    
    grad_c = np.zeros(n)
    for i in range(k_act):
        wi = float(wts[i])
        if wi <= 1e-15:
            continue
        ki = int(part[i])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        grad_c[js:je+1] += wi * 2.0 * a[js:je+1][::-1]
    
    grad = (2.0 * n) * (grad_c / S2 - 2.0 * M_soft / S3)
    return grad

def adam_update(a, g, m, v, t, lr):
    """Adam update in-place."""
    t1 = 0.9 ** t
    t2 = 0.999 ** t
    m[:] = 0.9 * m + 0.1 * g
    v[:] = 0.999 * v + 0.001 * g * g
    a -= lr * (m / (1.0 - t1)) / (np.sqrt(v / (1.0 - t2)) + 1e-8)

def optimize_lp_adam(a0, end_time, rng, init_lr=0.05, K_lp=30, K_soft=16):

    S = float(a.sum())

    m = np.zeros(n)

    t = 0
    lr = init_lr
    no_imp = 0
    patience = 200
    lp_count = 0
    lp_interval = 10  # Do LP every 10 Adam steps
    
    while time.monotonic() < end_time:
        c = fast_self_conv(a)
        S = float(a.sum())
        if S < 1e-10:
            a = np.clip(a0, 0.01, 1000)
            S = float(a.sum())

        score = compute_score(n, c, S)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if no_imp >= patience:
            # Restart with perturbation
            sc = 0.02 + 0.05 * rng.random()
            a = best_a * (1.0 + sc * rng.standard_normal(n))
            a = np.clip(a, 0, 1000)
            if a.sum() < 1e-3:
                a += 0.1
            m[:] = 0
            v[:] = 0
            t = 0
            lr = max(0.001, lr * 0.5)
            no_imp = 0
            patience = max(50, patience // 2)
            continue
        
        # Every lp_interval steps, do an LP step
        if t % lp_interval == 0 and t > 0:
            M = float(np.max(c))
            g = solve_lp_direction(a, M, K_lp, rng)
            if g is not None:

                a_new = (1 - t_interp) * a + t_interp * g
                a_new = np.clip(a_new, 0, 1000)
```

The LP is:
max `sum(g)` 
s.t. `(f*g)_k <= M` for k in S (selected set)
     `g >= 0`

The solution `g` maximizes the sum subject to the cross-correlation with `f` being bounded. This gives us a "direction" from `f` to `g` that is guaranteed to not increase the cross-correlation at the selected positions.

But the actual objective involves `a*a`, not `f*g`. So the LP direction is an approximation of the true descent direction.

A better formulation might be:
max `sum(g)` 
s.t. `((f+g)*(f+g))_k <= 2M` for k in S (doubled to account for the interpolation)
     `g >= 0`

But this is a QP, not an LP. Not as clean.

Alternatively, we can linearize:
`((f+εg)*(f+εg))_k ≈ (f*f)_k + 2ε(f*g)_k` for small ε.

Requiring this to be ≤ M: `(f*f)_k + 2ε(f*g)_k <= M`, i.e., `(f*g)_k <= (M - (f*f)_k)/(2ε)`.

For the peak position k*: `(f*f)_{k*} = M`, so the constraint becomes `(f*g)_{k*} <= 0`. Since `f >= 0` and `g >= 0`, this means `f*g = 0` at the peak, which is only possible if `f` or `g` is zero at the relevant positions. This is too restrictive.

So the original LP formulation (with the RHS being M instead of 0) is more practical. It allows some increase at the peak but bounds it.

Actually, let me reconsider. The original LP says `(f*g)_k <= M` for all selected k. This is the condition that ensures, for the interpolation `a_t = (1-t)f + tg`:
`(a_t * a_t)_k = (1-t)^2(f*f)_k + 2t(1-t)(f*g)_k + t^2(g*g)_k`

At the peak k*: `(1-t)^2 M + 2t(1-t)(f*g)_{k*} + t^2(g*g)_{k*} <= (1-t)^2 M + 2t(1-t) M + t^2(g*g)_{k*}`
`= M(1-t)^2 + 2t(1-t)M + t^2(g*g)_{k*}`
`= M(1-2t+t^2+2t-2t^2) + t^2(g*g)_{k*}`
`= M(1-t^2) + t^2(g*g)_{k*}`
`= M - t^2(M - (g*g)_{k*})`

```python
"""
LP-guided optimization combined with projected gradient on probability simplex
for minimizing 2n*max(conv(a,a))/(sum a)^2.

Key innovations over the previous gradient-only approach:
1. Reduced LP to find guaranteed descent directions reducing active peak correlations.
2. Projected gradient on probability simplex for clean non-negativity handling.
3. Adaptive switching between LP-guided and soft-softmax gradient steps.
4. Annealing schedule for peak selection (broad to narrow).
5. Coordinate descent polish phase for final improvements.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csc_matrix


def fast_self_conv(a):
    """Linear self-convolution via FFT, returns full convolution of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, c, S):
    """Objective value given n, convolution array c, and sum S."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-15:
        return np.ones(n) / n
    return w / s


def softmax_gradient_simplex(a, c, temp, K):
    """
    Compute gradient of soft-max of autocorrelation for a on the simplex (sum=1).
    Returns gradient vector of shape (n,).
    """
    n = len(a)
    k_act = min(K, c.size)
    part = np.argpartition(c, -k_act)[-k_act:]
    vals = c[part].copy()

    scaled = vals / max(temp, 1e-15)
    scaled -= scaled.max()
    wts = np.exp(scaled)
    wts /= wts.sum()
    M_soft = float(np.dot(vals, wts))

    grad_c = np.zeros(n)
    for i in range(k_act):
        wi = float(wts[i])
        if wi <= 1e-15:
            continue
        ki = int(part[i])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        grad_c[js:je + 1] += wi * 2.0 * a[js:je + 1][::-1]

    # On simplex (S=1): d/da [2n * M_soft] = 2n * grad_c
    # But we need to account for the simplex constraint via projection
    return 2.0 * n * grad_c


def exact_gradient_simplex(a, c):
    """Compute exact subgradient at the argmax position."""
    n = len(a)
    k_star = int(np.argmax(c))
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    grad = np.zeros(n)
    grad[js:je + 1] = 2.0 * a[js:je + 1][::-1]
    return 2.0 * n * grad


def solve_lp_direction(f, M, K):
    """
    Solve reduced LP: max sum(g) s.t. (f*g)_k <= M for top K positions, g >= 0.
    Returns g array or None.
    """
    n = len(f)
    c = fast_self_conv(f)

    # Select K positions with highest autocorrelation
    top_K = np.argpartition(c, -K)[-K:]

    # Build sparse constraint matrix F (K x n)
    # F[idx, j] = f[j] for valid (k, j) pairs where k = top_K[idx]
    rows = np.empty(K * n, dtype=np.int32)
    cols = np.empty(K * n, dtype=np.int32)
    data = np.empty(K * n, dtype=np.float64)
    nnz = 0

    for idx in range(K):
        ki = int(top_K[idx])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        for j in range(js, je + 1):
            if nnz >= len(rows):
                break
            rows[nnz] = idx
            cols[nnz] = j
            data[nnz] = f[j]
            nnz += 1

    if nnz == 0:
        return None

    F = csc_matrix((data[:nnz], (rows[:nnz], cols[:nnz])), shape=(K, n))

    # LP: max 1^T g s.t. Fg <= M*1, g >= 0
    # In linprog: min -1^T g
    c_obj = -np.ones(n)
    b_ub = M * np.ones(K)
    bounds = [(0, None)] * n

    try:
        res = linprog(c_obj, A_ub=F, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True})
        if res.status == 0 and res.x is not None:
            g = np.clip(res.x, 0, 1000.0)
            if g.sum() < 1e-10:
                return None
            return g
    except Exception:
        pass
    return None


def coordinate_descent_step(a, c, n, num_trials=200, rng=None):
    """
    Try coordinate perturbations on elements contributing to the peak.
    Returns improved a or original if no improvement.
    """
    if rng is None:
        rng = np.random.default_rng(0)

    S = float(a.sum())
    S2 = S * S
    cur_score = 2.0 * n * float(np.max(c)) / S2

    best_a = a.copy()
    best_score = cur_score

    # Try perturbing elements that contribute to top peaks
    top_positions = np.argpartition(c, -5)[-5:]
    candidate_indices = set()
    for ki in top_positions:
        ki = int(ki)
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        candidate_indices.update(range(js, je + 1))

    candidate_list = list(candidate_indices)
    if len(candidate_list) > num_trials:
        candidate_list = list(rng.choice(candidate_list, num_trials, replace=False))

    for j in candidate_list:
        for delta_frac in [0.01, -0.01, 0.05, -0.05, 0.1, -0.1]:
            a_test = a.copy()
            a_test[j] = max(0.0, a_test[j] * (1.0 + delta_frac))
            if a_test.sum() < 1e-10:
                continue
            c_test = fast_self_conv(a_test)
            S_test = float(a_test.sum())
            sc = 2.0 * n * float(np.max(c_test)) / (S_test * S_test)
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a_test.copy()

    return best_a, best_score


def optimize_combined(a0, end_time, rng, init_lr, lp_K, soft_K, lp_interval,
                      use_lp=True, use_coord=False):
    """
    Combined optimization: projected gradient + LP directions + optional coordinate descent.
    Works on the probability simplex (sum(a) = 1).
    """
    n = a0.size
    # Normalize to simplex
    S0 = float(a0.sum())
    if S0 < 1e-10:
        a = np.ones(n) / n
    else:
        a = a0 / S0

    c = fast_self_conv(a)
    score = compute_score(n, c, 1.0)
    best_score = score
    best_a = a.copy()

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = init_lr
    no_imp = 0
    patience = 300
    iter_count = 0

    while time.monotonic() < end_time:
        c = fast_self_conv(a)
        score = compute_score(n, c, 1.0)

        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp >= patience:
            # Restart with perturbation
            sc = 0.01 + 0.03 * rng.random()
            a_pert = best_a * (1.0 + sc * rng.standard_normal(n))
            a = project_simplex(a_pert)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            lr = max(1e-5, lr * 0.5)
            no_imp = 0
            patience = max(50, patience * 3 // 4)
            iter_count = 0
            continue

        # LP step every lp_interval iterations
        if use_lp and iter_count > 0 and iter_count % lp_interval == 0:
            M = float(np.max(c))
            g = solve_lp_direction(a, M, lp_K)
            if g is not None:
                g_sum = float(g.sum())
                if g_sum > 1e-10:
                    g_norm = g / g_sum  # normalize to simplex
                    # Interpolate on simplex
                    t_interp = min(0.4, 5.0 / np.sqrt(iter_count + 1))
                    a_mix = (1.0 - t_interp) * a + t_interp * g_norm
                    a = project_simplex(a_mix)
                    m[:] = 0.0
                    v[:] = 0.0
                    t = 0
                    iter_count += 1
                    continue

        # Coordinate descent periodically
        if use_coord and iter_count > 0 and iter_count % 500 == 0:
            a, _ = coordinate_descent_step(a, c, n, 100, rng)
            a = project_simplex(a)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            iter_count += 1
            continue

        # Gradient step
        # Adaptive: blend exact and softmax based on peak separation
        top1 = float(c.max())
        sorted_c = np.sort(c)[::-1]
        top2 = float(sorted_c[min(1, len(sorted_c)-1)])
        ratio = top2 / top1 if top1 > 1e-15 else 1.0

        if ratio > 0.95:
            # Multiple comparable peaks: use softmax
            temp = 0.05 * top1
            g = softmax_gradient_simplex(a, c, temp, soft_K)
        else:
            # Dominant peak: use exact subgradient
            g = exact_gradient_simplex(a, c)

        t += 1
        # Adam update
        t1 = 0.9 ** t
        t2 = 0.999 ** t
        m[:] = 0.9 * m + 0.1 * g
        v[:] = 0.999 * v + 0.001 * g * g
        step = lr * (m / (1.0 - t1)) / (np.sqrt(v / (1.0 - t2)) + 1e-8)
        a = a - step
        a = project_simplex(a)

        # Learning rate decay
        if t % 2000 == 0:
            lr *= 0.9

        iter_count += 1

    return best_score, best_a * float(best_a.sum()) if best_a.sum() > 0 else best_a


def init_candidates(n, rng):

            best_a = aa.copy()
    print(f"After Phase 4: {best_score:.6f}")

    # Phase 5: Ultra-fine final polish
    if best_a is not None and time.monotonic() < deadline - 3.0:
        sc, aa = optimize_combined(best_a, deadline - 3.0, rng,
                                    init_lr=0.0003, lp_K=40, soft_K=64,
                                    lp_interval=50, use_lp=True, use_coord=True)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"After Phase 5: {best_score:.6f}")

    # Final: convert and verify
    if best_a is None:
        best_a = np.ones(n) / n

    # Scale back to original scale (objective is scale-invariant, but ensure sum > 0.01)
    out_sum = float(best_a.sum())
    if out_sum < 0.01:
        best_a = best_a * (1.0 / out_sum)
        out_sum = 1.0

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]

    try:
        verified = float(evaluate_sequence(out))
        print(f"FINAL: internal={best_score:.6f}, verified={verified:.6f}, n={n}, "
              f"time={time.monotonic()-start:.1f}s")
    except Exception as e:
        print(f"FINAL: internal={best_score:.6f}, n={n}, time={time.monotonic()-start:.1f}s, err={e}")

    return out
```
