"""
Gradient-based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2
using softmax-approximated gradient, FFT, and adaptive temperature.
"""

import numpy as np
import time

def _fft_conv(a):
    """Compute full convolution of a with itself using FFT."""
    n = len(a)
    L = 1
    need = 2 * n - 1
    while L < need:
        L <<= 1
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)
    return c[:2*n-1]

def _score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    c = _fft_conv(a)
    return 2.0 * n * c.max() / (s * s)

def _softmax_weights(c, beta):
    """Compute softmax(beta * c) with numerical stability."""
    shifted = c - c.max()
    e = np.exp(beta * shifted)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        # Fallback: put all weight on the max
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
        return w
    return e / s

def _gradient(a, w, L_grad):
    """
    Compute gradient of sum(w*k * c_k) w.r.t. a.
    grad[i] = 2 * sum_k w[k] * a[k-i]
    Using FFT: 2 * IFFT(FFT(w) * conj(FFT(a)))[:n]
    """
    n = len(a)
    W = np.fft.fft(w, L_grad)
    A = np.fft.fft(a, L_grad)
    v = np.fft.ifft(W * np.conj(A))[:n]
    return 2.0 * np.real(v)

def _optimize_sequence(a_init, time_budget, beta_start=50.0, beta_end=5000.0, 
                        verbose=False):
    """Main gradient optimization loop."""
    a = np.maximum(a_init, 0.0).astype(float)
    s = a.sum()
    if s <= 0:
        return None
    a = a / s
    n = len(a)
    
    # FFT sizes
    L_conv = 1
    while L_conv < 2*n - 1:
        L_conv <<= 1
    L_grad = 1
    while L_grad < 3*n - 2:  # Need for cross-correlation
        L_grad <<= 1
    # Actually, for the cross-correlation of w (length 2n-1) and a (length n),
    # we need L >= (2n-1) + n - 1 = 3n - 2
    # But let's be safe and use 4n
    L_grad = 4 * n
    # Make it a power of 2
    while L_grad & (L_grad - 1):  # Not a power of 2
        L_grad >>= 1
    L_grad <<= 1
    # Actually let's just use the next power of 2 >= 3n-2
    L_grad = 1
    while L_grad < 3 * n:
        L_grad <<= 1
    
    best_a = a.copy()
    best_score = _score(a)
    beta = beta_start
    lr = 0.1
    stall = 0
    deadline = time.monotonic() + time_budget
    it = 0
    
    while time.monotonic() < deadline:
        # Compute convolution
        F = np.fft.rfft(a, L_conv)
        c = np.fft.irfft(F * F, L_conv)[:2*n-1]
        c = np.maximum(c, 0.0)  # Numerical safety
        
        cur_score = 2.0 * n * c.max()
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            stall = 0
            if verbose and it % 200 == 0:
                print(f"  it={it}, score={best_score:.7f}, beta={beta:.1f}")
        else:
            # Check if very close
            if abs(cur_score - best_score) < 1e-10:
                stall += 1
        
        # Compute softmax weights
        w = _softmax_weights(c, beta)
        
        # Compute gradient
        grad = _gradient(a, w, L_grad)
        
        # Line search: try multiple step sizes
        best_lr = lr
        best_a_try = None
        best_sc_try = cur_score

        lr_candidates = []

            if s_try <= 0:

            c_try = np.maximum(c_try, 0.0)

            a = best_a.copy()
            ns = 0.08 if stall < 100 else 0.15
            a = a * np.exp(_rng.normal(0, ns, n))
            a = np.maximum(a, 0)
            a = a / a.sum()
            stall = 0
            lr = 0.08
            beta = max(beta0, beta * 0.3)

        # Major reset
        if stall > 300:
            a = best_a.copy()
            a = a * _rng.lognormal(0, 0.4, n)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            stall = 0
            lr = 0.03
            beta = beta0

        it += 1

    return best_a, best_sc


def _local_polish(a_init, t_budget, verbose=False):
    """Block-based local search: scale random blocks and keep improvements."""
    a = np.maximum(a_init, 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        return None, float('inf')
    a = a / s
    n = len(a)
    L_conv = _next_pow2(2 * n - 1)

    sc, _ = _score_conv(a, L_conv)
    best_a = a.copy()
    best_sc = sc
    cur_sc = sc

    deadline = time.monotonic() + t_budget
    stall = 0
    it = 0
    last_print = -9999

    # Block sizes to try
    block_sizes = sorted(set([1, 2, 3, 4, 5, 8, 12, 16, 24, 32, 48, 64,
                               max(1, n // 256), max(1, n // 128),
                               max(1, n // 64), max(1, n // 32)]))
    factors = [0.3, 0.5, 0.7, 0.85, 1.0, 1.2, 1.5, 2.0, 3.0]

    while time.monotonic() < deadline:
        w = int(_rng.choice(block_sizes))
        w = min(w, n)
        start = int(_rng.integers(0, n - w + 1))

        best_block_sc = cur_sc
        best_block_a = None

        for f in factors:
            a_try = a.copy()
            a_try[start:start + w] *= f
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try, _ = _score_conv(a_try, L_conv)
            if sc_try < best_block_sc - 1e-13:
                best_block_sc = sc_try
                best_block_a = a_try

        if best_block_a is not None:
            a = best_block_a
            cur_sc = best_block_sc
            stall = 0
            if cur_sc < best_sc - 1e-12:
                best_sc = cur_sc
                best_a = a.copy()
                if verbose and (it - last_print) >= 500:
                    print(f"      polish it={it} best={best_sc:.8f}")
                    last_print = it
        else:
            stall += 1
            if stall > 50:
                a = best_a.copy()
                cur_sc = best_sc
                a = a * np.exp(_rng.normal(0, 0.05, n))
                a = np.maximum(a, 0)
                a = a / a.sum()
                stall = 0
            if stall > 400:
                a = best_a.copy()
                cur_sc = best_sc
                stall = 0

        it += 1

    return best_a, best_sc


def _make_init(n, method=0):
    """Create various initial sequences."""
    t = (np.arange(n, dtype=float) + 0.5) / n
    if method == 0:
        return np.ones(n) / n
    elif method == 1:
        # Beta-like distribution
        a = float(_rng.uniform(0.5, 3.0))
        b = float(_rng.uniform(0.5, 3.0))
        p = np.power(np.clip(t, 1e-8, 1 - 1e-8), a) * np.power(np.clip(1 - t, 1e-8, 1 - 1e-8), b)
        return p / p.sum()
    elif method == 2:
        # Gaussian bumps
        K = int(_rng.integers(2, 6))
        p = np.zeros(n)
        for _ in range(K):
            ctr = float(_rng.uniform(0.1, 0.9))
            sig = float(_rng.uniform(0.05, 0.3))
            p += np.exp(-0.5 * ((t - ctr) / sig) ** 2)
        return p / p.sum()
    elif method == 3:
        # Random smooth
        z = np.zeros(n)
        for k in range(1, 8):
            amp = float(_rng.normal(0, 1.0 / (k + 1)))
            ph = float(_rng.uniform(0, 2 * np.pi))
            z += amp * np.cos(2 * np.pi * k * t + ph)
        p = np.exp(z)
        return p / p.sum()
    elif method == 4:
        # Two-level
        cut = int(_rng.integers(n // 4, 3 * n // 4))
        ratio = float(_rng.uniform(0.2, 5.0))
        p = np.ones(n)
        p[cut:] = ratio
        return p / p.sum()
    else:
        # Linear ramp
        p = np.linspace(0.1, 2.0, n)
        return p / p.sum()


def _resample(a, m):
    """Resample sequence a to length m using interpolation."""
    n = len(a)
    if n == m:
        return a.copy()
    if n == 1:
        return np.full(m, a[0] / m)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(m) + 0.5) / m
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 1e-15)
    return out / out.sum()
