"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Combines adaptive LSE-smoothed L-BFGS, spectral-shaping initialization,
periodic perturb-and-restart, and exact subgradient polishing.
Key improvements over baseline: adaptive beta, spectral init, length adaptation,
multi-scale restart strategy, and improved time allocation.
"""

import numpy as np
from scipy.optimize import minimize
import time


# ---------------------------------------------------------------------------
# Core numerical helpers (all top-level, no closures)
# ---------------------------------------------------------------------------

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and its gradient.
    Returns (score, grad).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a: g_j = sum_m a[m] * (p[m+j] + p[j+m]) / 2
    # Actually g_j = sum_m a[m] * p[m+j] where p is over the convolution indices
    # conv(a_rev, p)[n-1:2n-1] gives sum_m a[m] * p[m + j - (n-1)]... 
    # Let me be more careful.
    # (a * p)_k = sum_j a[j] * p[k - j] for k = 0..2n-2
    # We want for each output index j (0..n-1): grad_j = sum_m a[m] * p[m + j]
    # This is the "cross-correlation" of a with p, which is conv(a_rev, p)[n-1 + j - ...]
    # Actually: sum_m a[m] * p[m+j] = (a_rev * p)_{n-1+j} where a_rev[j] = a[n-1-j]
    # So grad_L = conv(a_rev, p)[n-1 : 2n-1]
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    """Wrapper for scipy.minimize: returns (value, gradient)."""
    return lse_score_grad(a, beta)

def run_lbfgs(a0, beta, maxiter=3000, maxfun=60000):
    """Run scipy L-BFGS-B to minimize the LSE-smoothed score."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(

                 'maxls': 50}

    a_out = np.clip(res.x, 0.0, 1000.0)

def spectral_init(n, rng, target_flatness=0.3):
    """
    Create a non-negative sequence with a relatively flat magnitude spectrum.
    Uses iterative spectral projection (Griffin-Lim variant).
    """
    N = 1 << (2 * n - 2).bit_length()
    # Start with random non-negative signal
    a = np.abs(rng.standard_normal(n)) + 0.1
    a = a / a.sum() * np.sqrt(n)
    
    for _ in range(30):
        # Compute spectrum
        A = np.fft.rfft(a, N)
        mag = np.abs(A)
        # Compress dynamic range to flatten
        phase = np.angle(A)
        mag_target = mag ** target_flatness
        # Reconstruct
        A_new = mag_target * phase / (mag + 1e-30) * mag + (1 - target_flatness) * mag_target
        # Simpler: blend flat and original
        flat_mag = np.ones_like(mag)
        mag_blended = (1 - target_flatness) * mag + target_flatness * flat_mag * (mag.mean() + 1e-10)
        A_new = mag_blended * (phase / (np.abs(phase) + 1e-30))
        a_new = np.fft.irfft(A_new, N)[:n]
        # Project to non-negative
        a = np.maximum(a_new, 0.0)
        if a.sum() < 1e-10:
            a = np.abs(rng.standard_normal(n)) + 0.1
    a = a / a.sum() * np.sqrt(n)
    return a

        per = max(1, n // nb)
        vals = 0.1 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per + 1)[:n]
        inits.append(a.copy())
    
    # "Pulse-like" (sparse-ish)
    a = np.full(n, 0.05)
    n_spikes = max(3, n // 20)
    spike_pos = rng.choice(n, n_spikes, replace=False)
    a[spike_pos] = 1.0 + rng.random(n_spikes)
    inits.append(a)
    
    return inits


# ---------------------------------------------------------------------------
# Adaptive LSE optimization with restarts
# ---------------------------------------------------------------------------

def adaptive_lbfgs(a0, max_time_s=120.0):
    """
    Run L-BFGS with adaptive beta. Start with low beta, increase when
    progress stalls. Restart with perturbation if stuck.
    Returns (best_score, best_a).
    """
    t0 = time.monotonic()
    n = len(a0)
    a_cur = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    best_sc = compute_score(a_cur)
    best_a = a_cur.copy()
    
    beta_schedule = [15, 25, 40, 60, 100, 150, 250, 400, 600, 1000, 1500, 2000]
    max_iter_per_beta = max(500, 40000 // max(1, n // 1024))
    
    rng_local = np.random.default_rng(hash(tuple(a_cur[:10])) & 0xFFFFFFFF)
    
    for beta in beta_schedule:
        if time.monotonic() - t0 > max_time_s:
            break
        sc, a_opt, res = run_lbfgs(a_cur, beta, maxiter=max_iter_per_beta,
                                    maxfun=max_iter_per_beta * 3)
        a_cur = a_opt
        if sc < best_sc:
            best_sc = sc
            best_a = a_cur.copy()
        # Check progress: if L-BFGS converged immediately, try perturbing
        if res.nit < 5 and beta > 40:
            # Perturb and continue
            scale = 0.02 + 0.03 * (beta / 1000.0)
            a_cur = a_cur * (1.0 + scale * rng_local.standard_normal(n))
            a_cur = np.clip(a_cur, 0.0, 1000.0)
    
    return best_sc, best_a


# ---------------------------------------------------------------------------
# Exact subgradient polish (improved)
# ---------------------------------------------------------------------------

def exact_subgrad_polish(a, total_iters=2000):
    """
    Projected subgradient descent on the exact max-conv objective.
    Uses adaptive step size with Polyak + fixed component.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0  # Theoretical lower bound is 2n/(2n-1) > 1
    
    all_peaks = {}  # Track peak locations for averaged subgradient
    
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Find all near-peak lags (for smoother subgradient)
        threshold = M * 0.95
        peak_lags = np.where(c >= threshold)[0]
        
        # Averaged subgradient over near-peak lags
        sg = np.zeros(n)
        for k in peak_lags:
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                seg = a[j_start:j_end + 1]
                sg[j_start:j_end + 1] += 2.0 * seg[::-1]
        sg /= len(peak_lags)
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        
        # Polyak step + small fixed step
        polyak_step = (sc - sc_lb) / (gn * gn + 1e-30)
        polyak_step = min(polyak_step, 0.3)
        fixed_step = 0.005 / (gn + 1e-15)
        step = min(polyak_step + fixed_step, 0.5)
        
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
    
    return best_sc, best_a


# ---------------------------------------------------------------------------
# Targeted perturbation: reduce peak-contributing elements
# ---------------------------------------------------------------------------

def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements that contribute most to the peak convolution lag."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(5, n // 25)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)


# ---------------------------------------------------------------------------
# Length adaptation: try extending or trimming the best sequence
# ---------------------------------------------------------------------------

def try_length_adaptation(a_best, target_len, rng):
    """Adapt a sequence to a new length by interpolation or zero-padding."""
    n = len(a_best)
    if target_len == n:
        return a_best.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, target_len)
    a_new = np.interp(x_new, x_old, a_best)
    return a_new


# ---------------------------------------------------------------------------
# Main search function
# ---------------------------------------------------------------------------

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = float(budget_s)
    deadline = t_start + total_budget - 15.0

    # Load parent solution if available
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={time.monotonic()-t_start:.1f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = []
    if parent is not None:
        pl = parent.size
        lengths.append(pl)
        # Try nearby powers of 2
        for p2 in [4096, 8192]:
            if p2 not in lengths and abs(p2 - pl) < pl * 0.5:
                lengths.append(p2)
    for n_try in [2048, 4096, 8192]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))
    # Limit to at most 4 lengths
    lengths = lengths[:4]

    print(f"Lengths to try: {lengths}")
    print(f"Parent available: {parent is not None}, length: {parent.size if parent is not None else 'N/A'}")
    if parent is not None:
        print(f"Parent score: {compute_score(parent):.8f}")

    # ==================================================================
    # Phase 1: Multi-start adaptive LSE optimization (55% of time)
    # ==================================================================
    phase1_deadline = t_start + 0.55 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 1: Multi-start LSE optimization (deadline: {phase1_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"\n--- n = {n} ---")
        inits = diverse_init(n, rng, parent)
        
        # Limit number of inits based on time
        elapsed = time.monotonic() - t_start
        remaining_p1 = phase1_deadline - time.monotonic()
        max_inits = min(len(inits), max(4, int(remaining_p1 / 30)))
        
        # Sort: parent-derived first, then structured, then random
        parent_idx = 0
        if parent is not None:
            inits = [inits[0]] + inits[1:]
        
        for i, a0 in enumerate(inits[:max_inits]):
            if time.monotonic() > phase1_deadline:
                break
            t0 = time.monotonic()
            remaining = phase1_deadline - time.monotonic()
            alloc = min(remaining * 0.9 / max(1, max_inits - i), 180.0)
            sc, a_opt = adaptive_lbfgs(a0, max_time_s=alloc)
            report(sc, a_opt, f"n={n}, init={i}")
            print(f"    init={i}: score={sc:.8f} ({time.monotonic()-t0:.1f}s)")

    print(f"\nAfter Phase 1: best = {best_score:.8f} (n={len(best_a)})")

    # ==================================================================
    # Phase 2: Intensive refinement of best (25% of time)
    # ==================================================================
    phase2_deadline = t_start + 0.80 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 2: Intensive refinement (deadline: {phase2_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    # 2a: High-beta LSE from current best
    for beta in [800, 1500, 3000, 5000]:
        if time.monotonic() > phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 10:
            break
        maxit = min(5000, int(remaining * 3))
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxit, maxfun=maxit * 2)
        report(sc, a_opt, f"refine beta={beta}")

    # 2b: Length adaptation - try extending the best solution
    n_best = len(best_a)
    for target_n in [n_best * 2, n_best // 2, 8192, 4096]:
        if target_n == n_best or target_n < 512 or target_n > 16384:
            continue
        if time.monotonic() > phase2_deadline:
            break
        a_adapt = try_length_adaptation(best_a, target_n, rng)
        sc, a_opt = adaptive_lbfgs(a_adapt, max_time_s=min(60.0, phase2_deadline - time.monotonic()))
        report(sc, a_opt, f"adapt to n={target_n}")

    # 2c: Targeted perturbation + re-optimize
    for trial in range(5):
        if time.monotonic() > phase2_deadline:
            break
        a_pert = targeted_perturb(best_a, rng, strength=0.03 + 0.04 * rng.random())
        sc, a_opt = adaptive_lbfgs(a_pert, max_time_s=min(60.0, phase2_deadline - time.monotonic()))
        report(sc, a_opt, f"targeted perturb #{trial}")

    print(f"\nAfter Phase 2: best = {best_score:.8f} (n={len(best_a)})")

    # ==================================================================
    # Phase 3: Diversification (10% of time)
    # ==================================================================
    phase3_deadline = t_start + 0.90 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 3: Diversification (deadline: {phase3_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    n_best = len(best_a)
    for trial in range(10):
        if time.monotonic() > phase3_deadline:
            break
        strategy = trial % 5
        if strategy == 0:
            # Gaussian perturbation (medium scale)
            scale = 0.05 + 0.1 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Sinusoidal perturbation
            freq = 1.0 + 10.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.05 + 0.1 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 2:
            # Block permutation
            n_blocks = int(4 + 8 * rng.random())
            block_size = n_best // n_blocks
            a_pert = np.zeros(n_best)
            perm = rng.permutation(n_blocks)
            for i in range(n_blocks):
                start = i * block_size
                end = start + block_size if i < n_blocks - 1 else n_best
                a_pert[start:end] = best_a[perm[i] * block_size:(perm[i] * block_size + (end - start))]
        elif strategy == 3:
            # Spectal re-shaping
            a_pert = spectral_init(n_best, rng, target_flatness=0.2 + 0.6 * rng.random())
            # Blend with current best
            blend = 0.5 + 0.3 * rng.random()
            a_pert = blend * best_a + (1 - blend) * a_pert * (best_a.sum() / (a_pert.sum() + 1e-10))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Random restart at different length
            new_n = min(16384, max(512, int(n_best * (0.7 + 0.6 * rng.random()))))
            new_n = 1 << int(np.round(np.log2(new_n)))
            a_pert = spectral_init(new_n, rng, target_flatness=0.4)
        
        remaining = phase3_deadline - time.monotonic()
        if remaining < 5:
            break
        alloc = min(remaining * 0.8 / max(1, 10 - trial), 90.0)
        sc, a_opt = adaptive_lbfgs(a_pert, max_time_s=alloc)
        report(sc, a_opt, f"diversify #{trial} (strat={strategy})")

    print(f"\nAfter Phase 3: best = {best_score:.8f} (n={len(best_a)})")

    # ==================================================================
    # Phase 4: Final polish (10% of time)
    # ==================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Final Polish")
    print(f"{'='*60}")

    # 4a: Subgradient polish
    a_cur = best_a.copy()
    sc, a_sg = exact_subgrad_polish(a_cur, total_iters=3000)
    report(sc, a_sg, "subgrad polish")

    # 4b: LSE at very high beta from polished solution
    for beta in [3000, 5000, 8000]:
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        maxit = min(3000, int(remaining * 2))
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=maxit, maxfun=maxit * 2)
        report(sc, a_opt, f"final lbfgs beta={beta}")

    # 4c: One more subgradient pass
    if time.monotonic() < deadline - 30:
        sc, a_sg2 = exact_subgrad_polish(best_a, total_iters=2000)
        report(sc, a_sg2, "final subgrad")

    # 4d: Very high beta LSE
    if time.monotonic() < deadline - 10:
        sc, a_opt = run_lbfgs(best_a, 10000, maxiter=2000, maxfun=4000)
        report(sc, a_opt, "final lbfgs beta=10000")

    # ==================================================================
    # Final verification
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {total_budget}s")
    print(f"{'='*60}")
    return out
