    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _grad_via_fft(w, a):
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    if w.size >= N:
        fw = np.fft.rfft(w[:N], N)
    else:
        fw = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]


def _make_sine(n, amp, phase=0.5):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.sin(theta)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _make_multi_sine(n, amps, phases):
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64) / n
    for amp, ph in zip(amps, phases):
        theta = 2.0 * np.pi * (i + ph) / n
        a += amp * np.sin(theta) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _make_cosine(n, amp, phase=0.0):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.cos(theta)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _make_gaussian_envelope(n, width=0.3):
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - center) / (width * n)) ** 2)
    s = a.sum()
    if s > 0.01:
        a /= s
    return a


def _softmax_weights(c, T):
    c_min = float(np.min(c))
    c_max = float(np.max(c))
    rng = c_max - c_min
    if rng < 1e-15:
        return np.ones(c.size) / c.size
    c_norm = (c - c_min) / rng
    c_shift = c_norm - 1.0
    exp_c = np.exp(T * c_shift)
    s = exp_c.sum()
    if s < 1e-300:
        w = np.zeros(c.size)
        mask = c_norm >= 1.0 - 1e-10
        n_max = mask.sum()
        if n_max == 0:
            w[np.argmax(c)] = 1.0
        else:
            w[mask] = 1.0 / n_max
        return w
    return exp_c / s


def _optimize_mirror_descent(a_init, deadline, T_start=6.0, T_end=150.0, lr_scale=0.3, patience=8000):
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size, dtype=np.float64) / a.size
        s = 1.0
    a = a / s

    n = a.size
    lr = lr_scale * n

    best_a = a.copy()
    best_score = _score(best_a)

    it = 0
    last_improve = 0
    rng = np.random.RandomState(42)
    eval_interval = 4

    while time.monotonic() < deadline:
        it += 1

        t_frac = min(1.0, it / 150000.0)
        T = T_start + (T_end - T_start) * (t_frac ** 0.7)
        lr_cur = lr / (1.0 + 0.005 * it)

        c = _fft_convolve(a)
        if c.size == 0:
            break

        w = _softmax_weights(c, T)
        grad = _grad_via_fft(w, a)
        grad_proj = grad - np.mean(grad)

        update = -lr_cur * grad_proj
        update = np.clip(update, -8.0, 8.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        if it % eval_interval == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
                lr = min(0.6 * n, lr * 1.008)
            elif it - last_improve > patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.02
                a = a + noise * a.mean()
                a = np.clip(a, 0, 1000)
                a = a / a.sum()
                patience = max(3000, int(patience * 1.4))
                lr = max(0.1 * n, lr * 0.75)
                # Also try a larger perturbation every 3rd restart
                if (it - last_improve) > patience * 2:
                    a = best_a.copy()
                    mask = rng.random(n) < 0.3
                    a[mask] *= rng.uniform(0.3, 2.0, mask.sum())
                    a = np.clip(a, 0, 1000)
                    a = a / a.sum()
                    patience = 3000
                    lr = 0.2 * n

    return best_a, best_score


def _optimize_hard_max(a_init, deadline, K=10, lr_scale=0.4):
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size, dtype=np.float64) / a.size
        s = 1.0
    a = a / s

    n = a.size
    lr = lr_scale * n

    best_a = a.copy()
    best_score = _score(best_a)

    it = 0
    last_improve = 0
    patience = 4000
    rng = np.random.RandomState(77777)
    eval_interval = 3

    while time.monotonic() < deadline:
        it += 1
        lr_cur = lr / (1.0 + 0.008 * it)

        c = _fft_convolve(a)
        if c.size == 0:
            break

        if c.size <= K:
            top_idx = np.arange(c.size)
        else:
            top_idx = np.argpartition(c, -K)[-K:]

        top_vals = c[top_idx]
        T_local = 5.0
        top_shift = top_vals - top_vals.max()
        weights = np.exp(T_local * top_shift)
        weights /= weights.sum()

        w = np.zeros(c.size, dtype=np.float64)
        for i, k in enumerate(top_idx):
            w[k] = weights[i]

        grad = _grad_via_fft(w, a)
        grad_proj = grad - np.mean(grad)

        update = -lr_cur * grad_proj
        update = np.clip(update, -6.0, 6.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        if it % eval_interval == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
                lr = min(0.8 * n, lr * 1.004)
            elif it - last_improve > patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.01
                a = a + noise * a.mean()
                a = np.clip(a, 0, 1000)
                a = a / a.sum()
                patience = max(1500, int(patience * 1.5))
                lr = max(0.1 * n, lr * 0.7)

    return best_a, best_score


def _generate_initializations(deadline):
    candidates = []
    rng = np.random.RandomState(2024)

    # Try to use height_sequence_1 if available
    try:
        hs = np.asarray(height_sequence_1, dtype=np.float64)
        hs = np.clip(hs, 0, 1000)
        s = hs.sum()
        if s > 0.01:
            hs = hs / s
            sc = _score(hs)
            candidates.append((hs.copy(), sc, "height_seq_1"))
            print(f"  [init] height_sequence_1: n={hs.size}, score={sc:.8f}", flush=True)
    except Exception:
        pass

    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.monotonic() > deadline:
            break

        # Sinusoidal scan
        for amp in [0.5, 0.6, 0.65, 0.7, 0.72, 0.75, 0.78, 0.8, 0.82, 0.85, 0.9, 0.95, 1.0, 1.1, 1.2]:
            if time.monotonic() > deadline:
                break
            a = _make_sine(n, amp)
            sc = _score(a)
            candidates.append((a.copy(), sc, f"sine_n{n}_a{amp}"))

        # Cosine variants
        for amp in [0.6, 0.7, 0.8, 0.9, 1.0]:
            if time.monotonic() > deadline:
                break
            for ph in [0.0, 0.1, 0.25, 0.4]:
                a = _make_cosine(n, amp, ph)
                sc = _score(a)
                candidates.append((a.copy(), sc, f"cos_n{n}_a{amp}_p{ph}"))

        # Multi-mode
        for _ in range(8):
            if time.monotonic() > deadline:
                break
            a1 = rng.uniform(0.5, 1.0)
            a2 = rng.uniform(0.03, 0.25)
            p1 = rng.uniform(0.0, 0.5)
            p2 = rng.uniform(0.0, 1.0)
            a = _make_multi_sine(n, [a1, a2], [p1, p2])
            sc = _score(a)
            candidates.append((a.copy(), sc, f"multi_n{n}"))

        # 3-mode
        for _ in range(4):
            if time.monotonic() > deadline:
                break
            a1 = rng.uniform(0.5, 0.9)
            a2 = rng.uniform(0.05, 0.2)
            a3 = rng.uniform(0.02, 0.1)
            p1 = rng.uniform(0.0, 0.5)
            p2 = rng.uniform(0.0, 1.0)
            p3 = rng.uniform(0.0, 1.0)
            a = _make_multi_sine(n, [a1, a2, a3], [p1, p2, p3])
            sc = _score(a)
            candidates.append((a.copy(), sc, f"multi3_n{n}"))

    # Sort by score and return top unique candidates
    candidates.sort(key=lambda x: x[1])
    # Keep top 5, ensuring diversity (different n or very different scores)
    top = []
    seen_n = set()
    for a, sc, name in candidates:
        if len(top) >= 5:
            break
        # Allow same n but require score gap
        if a.size in seen_n and top and sc - top[-1][1] < 0.01:
            continue
        top.append((a, sc, name))
        seen_n.add(a.size)

    return top


def _hybrid_polish(a_init, deadline):
    """Alternates between soft and hard max phases for better convergence."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size, dtype=np.float64) / a.size
        s = 1.0
    a = a / s

    n = a.size
    best_a = a.copy()
    best_score = _score(best_a)

    it = 0
    last_improve = 0
    patience = 5000
    rng = np.random.RandomState(31415)
    eval_interval = 3
    lr_soft = 0.25 * n
    lr_hard = 0.35 * n

    # Phase cycling: soft for 20000 it, hard for 10000 it, repeat
    phase_duration_soft = 20000
    phase_duration_hard = 10000
    cycle = phase_duration_soft + phase_duration_hard

    while time.monotonic() < deadline:
        it += 1
        phase = (it % cycle)
        is_hard = phase >= phase_duration_soft

        if is_hard:
            T = 50.0
            lr_cur = lr_hard / (1.0 + 0.003 * it)
        else:
            t_frac = min(1.0, it / 200000.0)
            T = 6.0 + (100.0 - 6.0) * (t_frac ** 0.5)
            lr_cur = lr_soft / (1.0 + 0.003 * it)

        c = _fft_convolve(a)
        if c.size == 0:
            break

        w = _softmax_weights(c, T)
        grad = _grad_via_fft(w, a)
        grad_proj = grad - np.mean(grad)

        update = -lr_cur * grad_proj
        update = np.clip(update, -7.0, 7.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        if it % eval_interval == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.015
                a = a + noise * a.mean()
                a = np.clip(a, 0, 1000)
                a = a / a.sum()
                patience = max(2000, int(patience * 1.3))

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for low-peak nonnegative step function.
    Uses diverse initialization, mirror descent optimization,
    hybrid soft/hard polish, and hard-max refinement.
    """
    t0 = time.monotonic()
    hard_deadline = t0 + float(budget_s) - 15.0

    np.random.seed(seed)
    print(f"[Start] Budget: {budget_s}s", flush=True)

    # Phase 1: Generate and select initializations
    d1 = t0 + 90.0
    print("[Phase 1] Generating initializations...", flush=True)
    inits = _generate_initializations(d1)
    for a, sc, name in inits:
        print(f"  Candidate: {name}, n={a.size}, score={sc:.8f}", flush=True)

    if not inits:
        a0 = _make_sine(1024, 0.75)
        inits = [(a0, _score(a0), "default")]

    # Phase 2: Mirror descent on top candidates
    # Split time among top 3 candidates
    n_candidates = min(3, len(inits))
    phase2_deadline = hard_deadline - 400.0
    per_candidate_time = (phase2_deadline - time.monotonic()) / n_candidates

    results = []
    for idx in range(n_candidates):
        a_init, sc_init, name = inits[idx]
        cand_deadline = time.monotonic() + per_candidate_time
        print(f"[Phase 2.{idx+1}] Mirror descent on {name} ({per_candidate_time:.0f}s budget)...", flush=True)

        a_opt, sc_opt = _optimize_mirror_descent(a_init, cand_deadline)
        print(f"  Result: {sc_opt:.8f} (init was {sc_init:.8f})", flush=True)
        results.append((a_opt, sc_opt, f"md_{name}"))

        # Keep global best
        if results and results[-1][1] < min(r[1] for r in results[:-1]):
            pass

    # Phase 3: Hybrid polish on the best result
    results.sort(key=lambda x: x[1])
    best_a, best_s, best_name = results[0]
    print(f"[Phase 3] Hybrid polish on best ({best_name}, score={best_s:.8f})...", flush=True)

    d3 = hard_deadline - 150.0
    remaining3 = d3 - time.monotonic()
    if remaining3 > 60:
        a_hp, s_hp = _hybrid_polish(best_a, d3)
        print(f"  Hybrid polish result: {s_hp:.8f}", flush=True)
        if s_hp < best_s:
            best_a, best_s = a_hp, s_hp
        results.append((a_hp, s_hp, "hybrid"))
    else:
        print("[Phase 3] Skipped (insufficient time)", flush=True)

    # Phase 4: Hard-max polish
    d4 = hard_deadline - 30.0
    remaining4 = d4 - time.monotonic()
    if remaining4 > 60:
        print(f"[Phase 4] Hard-max polish ({remaining4:.0f}s budget, K=8)...", flush=True)
        a_hm, s_hm = _optimize_hard_max(best_a, d4, K=8)
        print(f"  Hard-max (K=8): {s_hm:.8f}", flush=True)
        if s_hm < best_s:
            best_a, best_s = a_hm, s_hm
        results.append((a_hm, s_hm, "hard8"))
    else:
        print("[Phase 4] Skipped (insufficient time)", flush=True)

    # Try hard-max with different K on the second-best candidate too
    d4b = hard_deadline - 15.0
    remaining4b = d4b - time.monotonic()
    if remaining4b > 40 and len(results) >= 2:
        second_a = results[1][0]
        if np.allclose(second_a, best_a):
            second_a = results[2][0] if len(results) >= 3 else results[1][0]
        print(f"[Phase 4b] Hard-max K=12 on second candidate ({remaining4b:.0f}s)...", flush=True)
        a_hm2, s_hm2 = _optimize_hard_max(second_a, d4b, K=12)
        print(f"  Hard-max (K=12): {s_hm2:.8f}", flush=True)
        if s_hm2 < best_s:
            best_a, best_s = a_hm2, s_hm2
        results.append((a_hm2, s_hm2, "hard12"))

    # Phase 5: Validate
    print("[Phase 5] Validating with evaluate_sequence...", flush=True)
    results.sort(key=lambda x: x[1])

    best_out = None
    best_actual = np.inf

    for name, arr, sc in results[:6]:
        if time.monotonic() > hard_deadline:
            break
        arr_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        try:
            actual = float(evaluate_sequence(arr_list))
        except Exception:
            actual = np.inf
        if np.isfinite(actual) and actual < best_actual:
            best_actual = actual
            best_out = arr_list
            print(f"  {name} (n={arr.size}): actual={actual:.8f}", flush=True)

    if best_out is None:
        best_out = [float(x) for x in best_a]
        try:
            best_actual = float(evaluate_sequence(best_out))
        except Exception:
            best_actual = best_s
        print(f"  Fallback: {best_actual:.8f}", flush=True)

    total_time = time.monotonic() - t0
    print(f"[Done] Time: {total_time:.1f}s, Best score: {best_actual:.8f}", flush=True)
    return best_out
