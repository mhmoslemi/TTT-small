"""
Step-function optimizer using smoothed-max gradient descent, exact coordinate 
polish with lookahead, and multi-scale restarts. Minimizes 2*n*max(conv(a,a))/(sum(a))^2.
Key improvements: adaptive-temperature softmax over active lags, Nesterov momentum,

import numpy as np
import time
import math


def _normalize(v):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v


def _fft_size(m):
    if m <= 2:
        return 2
    return 1 << (int(m) - 1).bit_length()


def _autocorr(f):
    n = f.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = 2 * n - 1
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]


def _fast_score(f):
    f = _normalize(f)
    if f.size == 0:
        return np.inf
    c = _autocorr(f)
    if c.size == 0:
        return np.inf
    return float(2.0 * f.size * c.max())


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    idx = int(np.nonzero(cond)[0][-1])
    theta = (cssv[idx] - 1.0) / float(idx + 1.0)
    out = np.maximum(v - theta, 0.0)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    out /= s
    return out


def _weighted_gradient(w, f):
    n = f.size
    L = w.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = L + n - 1
    size = _fft_size(m)
    W = np.fft.rfft(w, size)
    Fr = np.fft.rfft(f[::-1], size)
    conv = np.fft.irfft(W * Fr, size)
    return 2.0 * conv[n - 1:n + n - 1]


def _smoothmax_gradient(c, f, T):
    """Gradient of sum_k exp(c_k/T) w.r.t. f, times 1/T."""
    n = c.size
    maxc = c.max()
    z = (c - maxc) / T
    np.clip(z, -50.0, 0.0, out=z)
    w = np.exp(z)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.ones(n, dtype=np.float64) / n
    else:
        w /= wsum
    return _weighted_gradient(w, f)


def _interp_to_length(f, n):
    f = _normalize(f)
    m = f.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if m == n:
        return f
    if n == 1:
        return np.array([1.0])
    if m == 1:
        return _normalize(np.full(n, f[0]))
    x_old = np.linspace(0.0, m - 1, m)
    x_new = np.linspace(0.0, n - 1, n)
    return _normalize(np.interp(x_new, x_old, f))


def _fourier_seed(n, rng):
    x = np.linspace(0.0, 1.0, n)
    u = np.zeros(n)
    u += float(rng.uniform(-1.0, 1.0)) * (x - 0.5)
    for m_f in (1, 2, 3, 5, 8, 13):
        u += 0.35 * float(rng.uniform(-1.0, 1.0)) * np.cos(
            2.0 * np.pi * m_f * x + float(rng.uniform(0.0, 2.0 * np.pi)))
    f = np.exp(0.7 * u)
    return _normalize(f)


def _power_law_seed(n, rng):
    x = np.linspace(0.0, 1.0, n)
    alpha = float(rng.uniform(0.1, 2.0))
    beta = float(rng.uniform(0.1, 2.0))
    f = (x + 0.001) ** alpha * (1.0 - x + 0.001) ** beta
    return _normalize(f)


def _two_peak_seed(n, rng):
    """Sequence with mass concentrated at both ends."""
    width = max(1, int(n * float(rng.uniform(0.05, 0.25))))
    decay = float(rng.uniform(5.0, 30.0))
    f = np.zeros(n)
    for i in range(min(width, n)):
        f[i] = math.exp(-decay * i / width)
        f[n - 1 - i] = math.exp(-decay * i / width)
    return _normalize(f + 1e-10)


def _sawtooth_seed(n, rng):

    freq = float(rng.uniform(1.0, 4.0))
    amp = float(rng.uniform(0.3, 1.0))
    f = 1.0 + amp * np.sin(2.0 * np.pi * freq * x + float(rng.uniform(0, 2*np.pi)))
    f = np.maximum(f, 0.001)
    return _normalize(f)

    f = _normalize(f0)
    n = f.size
    if n < 2 or budget_s <= 0.0:
        c = _autocorr(f)
        return f, float(2.0 * n * c.max()), c

    L = 2 * n - 1

    except Exception:
        c = _autocorr(f)
    if c.size != L or not np.all(np.isfinite(c)):
        c = _autocorr(f)

    best_f = f.copy()
    best_score = float(2.0 * n * c.max())
    deadline = time.monotonic() + float(budget_s)
    delta = np.zeros(L, dtype=np.float64)
    total = np.zeros(L, dtype=np.float64)
    no_improve = 0

    if n > 15000:
        D = 5
        R = 5
        activeK = 80
    elif n > 5000:
        D = 8
        R = 8
        activeK = 150
    else:
        D = 12
        R = 12
        activeK = max(24, min(L, n // 10))

    activeK = int(min(L, max(20, activeK)))
    fracs = (0.01, 0.04, 0.10, 0.25, 0.50, 0.80)

    it = 0
    while time.monotonic() < deadline and no_improve < 15:
        it += 1
        maxc = float(c.max())

        if activeK >= L:
            active = np.arange(L)
        else:
            active = np.argpartition(c, L - activeK)[L - activeK:]
        p = np.zeros(L, dtype=np.float64)
        if active.size:
            p[active] = 1.0 / float(active.size)
        g = _weighted_gradient(p, f)

        donor_order = np.argsort(g)[::-1]
        recv_order = np.argsort(g)

        donors = []
        for idx in donor_order:
            idx = int(idx)
            if f[idx] > 1e-12:
                donors.append(idx)
                if len(donors) >= D:
                    break

        donor_set = set(donors)
        receivers = []
        for idx in recv_order:
            idx = int(idx)
            if idx not in donor_set:
                receivers.append(idx)
                if len(receivers) >= R:
                    break

        if not donors or not receivers:
            no_improve += 1
            if no_improve >= 6:
                break
            continue

        best_mc = maxc
        best_move = None

        for i in donors:
            fi = float(f[i])
            if fi <= 1e-12:
                continue
            ui = np.zeros(L, dtype=np.float64)
            ui[i:i + n] = -f
            for j in receivers:
                if i == j:
                    continue
                uj = np.zeros(L, dtype=np.float64)
                uj[j:j + n] = f
                for frac in fracs:
                    dv = fi * frac
                    if dv <= 0.0:
                        continue
                    delta.fill(0.0)
                    np.add(ui, uj, out=delta)
                    delta *= (2.0 * dv)
                    q = dv * dv
                    delta[2 * j] += q
                    delta[2 * i] += q
                    delta[i + j] -= 2.0 * q
                    np.add(c, delta, out=total)
                    mc = float(np.max(total))
                    if mc < best_mc - 1e-16:
                        best_mc = mc
                        best_move = (i, j, dv, delta.copy())

        # Random fallback
        if best_move is None or best_mc >= maxc - 1e-14:
            for _ in range(30):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j or f[i] <= 1e-12:
                    continue
                dv = float(f[i]) * float(rng.uniform(0.01, 0.4))
                delta.fill(0.0)
                delta[i:i + n] -= f
                delta[j:j + n] += f
                delta *= (2.0 * dv)
                q = dv * dv
                delta[2 * j] += q
                delta[2 * i] += q
                delta[i + j] -= 2.0 * q
                np.add(c, delta, out=total)
                mc = float(np.max(total))
                if mc < best_mc - 1e-16:
                    best_mc = mc
                    best_move = (i, j, dv, delta.copy())

        if best_move is None or best_mc >= maxc - 1e-14:
            no_improve += 1
            continue

        i, j, dv, d_saved = best_move
        f[i] -= dv
        f[j] += dv
        c = c + d_saved
        no_improve = 0
        cur_score = float(2.0 * n * best_mc)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()

        if it % 60 == 0:
            f = _normalize(f)
            c = _autocorr(f)
            cur_score = float(2.0 * n * c.max())
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()

    f = _normalize(best_f)
    c = _autocorr(f)
    mc = float(c.max()) if c.size else 0.0
    return f, float(2.0 * n * mc), c


def _perturb(best_f, rng, n):
    """Generate a perturbed version of the best sequence."""
    r = float(rng.random())
    if r < 0.4:
        sigma = float(rng.uniform(0.05, 0.8))
        z = rng.standard_normal(n)
        f = best_f * np.exp(sigma * z)
    elif r < 0.7:
        alpha = float(rng.uniform(0.1, 0.5))
        noise = rng.random(n)
        s = noise.sum()
        if s > 1e-300:
            noise /= s
        f = (1.0 - alpha) * best_f + alpha * noise
    else:
        # Swap-based perturbation
        f = best_f.copy()
        n_swaps = int(rng.integers(3, max(4, n // 100)))
        for _ in range(n_swaps):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                f[i], f[j] = f[j], f[i]
    return _normalize(f)


def _gradient_optimize(f0, budget_s, rng, tag=""):
    """Annealed smoothed-max gradient with momentum and periodic coordinate polish."""
    f0 = _normalize(f0)
    n = f0.size
    if n == 0:
        return f0, np.inf
    if budget_s <= 0.0:
        c = _autocorr(f0)
        return f0, float(2.0 * n * c.max())

    f = f0.copy()
    c = _autocorr(f)
    maxc = float(c.max())
    best_f = f.copy()
    best_score = float(2.0 * n * maxc)
    momentum = np.zeros(n, dtype=np.float64)

    deadline = time.monotonic() + float(budget_s)
    T = maxc * 0.05  # temperature for smoothed max
    min_T = maxc * 1e-8
    T_decay = 0.997
    beta_mom = 0.7  # momentum coefficient
    no_accept = 0
    it = 0

    if n < 3000:
        coord_interval = 0.35
        coord_budget = 0.30
    elif n < 10000:
        coord_interval = 0.50
        coord_budget = 0.40
    else:
        coord_interval = 0.70
        coord_budget = 0.50

    last_coord = time.monotonic()
    print(f"[{tag}] start score={best_score:.6f} n={n}")

    while time.monotonic() < deadline:
        it += 1
        if maxc <= 0.0:
            break

        # Smoothed-max gradient
        g = _smoothmax_gradient(c, f, T)
        g_centered = g - float(g.mean())
        g_range = float(g_centered.max() - g_centered.min())

        accepted = False
        if g_range > 1e-300:
            # Nesterov momentum update
            momentum = beta_mom * momentum + (1 - beta_mom) * g_centered
            grad_step = momentum / (g_range + 1e-300)

            best_mc = maxc
            best_ft = None
            best_ct = None

            if no_accept > 40:
                steps = (0.003, 0.01, 0.05, 0.2)
            else:
                steps = (0.01, 0.05, 0.2, 1.0, 3.0)

            for e in steps:
                # Log-domain (mirror) step
                arg = -e * grad_step
                np.clip(arg, -40.0, 40.0, out=arg)
                ft = f * np.exp(arg)
                ft = _normalize(ft)
                ct = _autocorr(ft)
                mc = float(ct.max())
                if mc < best_mc - 1e-16:
                    best_mc = mc
                    best_ft = ft
                    best_ct = ct

                # Euclidean projected step (for intermediate scales)
                if e in (0.05, 0.2):
                    v = f - e * grad_step
                    ft2 = _project_simplex(v)
                    ft2 = _normalize(ft2)
                    ct2 = _autocorr(ft2)
                    mc2 = float(ct2.max())
                    if mc2 < best_mc - 1e-16:
                        best_mc = mc2
                        best_ft = ft2
                        best_ct = ct2

            if best_ct is not None and best_mc < maxc - 1e-16:
                f = best_ft
                c = best_ct
                maxc = best_mc
                accepted = True

        if accepted:
            no_accept = 0
            cur_score = float(2.0 * n * maxc)
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()
                if best_score < 1.5027:
                    print(f"[{tag}] NEW BEST: {best_score:.6f}")
            T = max(min_T, T * T_decay)
        else:
            no_accept += 1
            now = time.monotonic()
            if now - last_coord > coord_interval or no_accept > 25:
                fp, sc, cp = _coordinate_polish(f, c, coord_budget, rng)
                if sc < best_score:
                    best_score = sc
                    best_f = fp.copy()
                    f = _normalize(fp)
                    c = cp
                    maxc = float(c.max())
                    accepted = True
                    no_accept = 0
                    momentum = np.zeros(n, dtype=np.float64)
                    T = maxc * 0.03
                else:
                    if no_accept > 60 or it % 8000 == 0:
                        f = _perturb(best_f, rng, n)
                        c = _autocorr(f)
                        maxc = float(c.max())
                        T = maxc * float(rng.uniform(0.05, 0.15))
                        momentum = np.zeros(n, dtype=np.float64)
                        no_accept = 0
                last_coord = time.monotonic()
            else:
                T = max(min_T, T * 0.999)

        if it % 5000 == 0:
            cur_score = float(2.0 * n * c.max())
            print(f"[{tag}] it={it} best={best_score:.6f} cur={cur_score:.6f} T={T:.3e} na={no_accept}")

    cur_score = float(2.0 * n * float(c.max()))
    if cur_score < best_score:
        best_score = cur_score
        best_f = f.copy()

    print(f"[{tag}] end best={best_score:.6f}")
    return _normalize(best_f), float(best_score)


def _build_seeds(f0, n, rng):
    """Build a diverse pool of starting sequences."""
    seeds = []

    def add_seed(name, f):
        f = _interp_to_length(f, n)
        try:
            sc = _fast_score(f)
        except Exception:
            return
        if np.isfinite(sc):
            seeds.append((float(sc), len(seeds), str(name), f))

    add_seed("given", f0)
    add_seed("reverse", f0[::-1])

    # Smoothed versions
    for win in (7, 15, 51, 121):
        if win < n:
            kernel = np.ones(win) / win
            sm = _normalize(np.convolve(f0, kernel, mode="same"))
            add_seed(f"smooth{win}", sm)

    # Power law
    for alpha in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0):
        x = np.linspace(0.0, 1.0, n)
        f = (x + 0.001) ** alpha
        add_seed(f"pow_{alpha}", _normalize(f))

    # Two-peak
    for i in range(4):
        add_seed(f"twopk{i}", _two_peak_seed(n, rng))

    # Fourier
    for i in range(6):
        add_seed(f"four{i}", _fourier_seed(n, rng))

    # Sawtooth
    for i in range(3):
        add_seed(f"saw{i}", _sawtooth_seed(n, rng))

    # Beta-like
    for i in range(4):
        add_seed(f"beta{i}", _power_law_seed(n, rng))

    # Coarse-to-fine
    if n > 500:
        for K in (128, 256, 512, 1024, 2048):
            if K < n:
                x_old = np.linspace(0.0, n - 1, n)
                x_new = np.linspace(0.0, K - 1, K)
                coarse = _normalize(np.interp(x_new, x_old, f0))
                fine = _interp_to_length(coarse, n)
                add_seed(f"coarse{K}", fine)

    if not seeds:
        f = np.ones(n, dtype=np.float64)
        seeds.append((2.0, 0, "uniform", _normalize(f)))

    seeds.sort()
    return seeds


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative coefficient sequence minimizing evaluate_sequence.
    Uses multi-start optimization with smoothed-max gradients, exact coordinate
    descent, and adaptive restarts.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(120.0, float(budget_s)) - 30.0

    # Get initial sequence
    try:
        f0 = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        f0 = np.ones(6121, dtype=np.float64)

    if f0.size == 0:
        f0 = np.ones(6121, dtype=np.float64)
    f0 = np.nan_to_num(f0, nan=0.0, posinf=1000.0, neginf=0.0)
    np.clip(f0, 0.0, 1000.0, out=f0)
    n = int(f0.size)
    if n < 3:
        f0 = np.ones(1024, dtype=np.float64)
        n = f0.size
    f0 = _normalize(f0)

    best_f = f0.copy()
    best_score = _fast_score(best_f)
    best_c = _autocorr(best_f)
    print(f"Initial: n={n}, score={best_score:.6f}")

    try:
        seed_pool = _build_seeds(f0, n, rng)
        print(f"Seed pool: {len(seed_pool)} candidates, best seed score={seed_pool[0][0]:.6f}")

        # Coarse optimization (low-dim, fast)
        now = time.monotonic()
        if total_deadline - now > 1200.0 and n > 512:
            for K in (512, 256):
                if total_deadline - now < 80.0:
                    break
                t_start = time.monotonic()
                x_old = np.linspace(0.0, n - 1, n)
                x_new = np.linspace(0.0, K - 1, K)
                coarse = _normalize(np.interp(x_new, x_old, f0))
                cf, cs = _gradient_optimize(coarse, 30.0, rng, f"coarse{K}")
                fin = _interp_to_length(cf, n)
                sc = _fast_score(fin)
                if sc < best_score:
                    best_score = sc
                    best_f = fin.copy()
                    best_c = _autocorr(best_f)
                # Add to pool
                seed_pool.append((sc, len(seed_pool), f"coarse_lift{K}", fin))
                elapsed = time.monotonic() - t_start
                print(f"  Coarse K={K}: score={sc:.6f} (took {elapsed:.1f}s)")
            seed_pool.sort()

        # Main optimization: try top diverse seeds
        # Select top seeds that are "diverse" (not all similar)
        top_seeds = []
        for item in seed_pool:
            if len(top_seeds) >= 5:
                break
            # Check diversity: correlation with already-selected seeds
            is_diverse = True
            for ts in top_seeds:
                corr = np.corrcoef(item[3], ts[3])[0, 1]
                if corr > 0.95:
                    is_diverse = False
                    break
            if is_diverse:
                top_seeds.append(item)

        if not top_seeds:
            top_seeds = seed_pool[:3]

        # Allocate time to each seed
        n_seeds = len(top_seeds)
        weights = []
        for i in range(n_seeds):
            w = 1.0 / (i + 1.5)
            weights.append(w)
        total_w = sum(weights)
        weights = [w / total_w for w in weights]

        # Reserve final coordinate polish
        reserve = min(300.0, 0.15 * float(budget_s))
        main_deadline = total_deadline - reserve

        t_cursor = time.monotonic()
        for idx, item in enumerate(top_seeds):
            sc, _sid, name, f_seed = item
            remaining = main_deadline - time.monotonic()
            if remaining < 40.0:
                break
            alloc = 0.85 * remaining * weights[idx] / sum(weights[idx:])
            alloc = max(30.0, alloc)
            print(f"\n=== Main #{idx+1}/{n_seeds}: seed={name} alloc={alloc:.0f}s cur_best={best_score:.6f} ===")
            f_opt, sc_opt = _gradient_optimize(f_seed, alloc, rng, f"main{idx}_{name}")
            if sc_opt < best_score:
                best_score = sc_opt
                best_f = f_opt.copy()
                best_c = _autocorr(best_f)
                print(f"  IMPROVED to {best_score:.6f}")

        # Final coordinate polish
        now = time.monotonic()
        final_time = max(0.0, total_deadline - now - 20.0)
        if final_time > 30.0:
            print(f"\n=== Final coordinate polish: {final_time:.0f}s ===")
            bf, bs, bc = _coordinate_polish(best_f, _autocorr(best_f), final_time, rng)
            if bs < best_score:
                best_score = bs
                best_f = bf.copy()
                best_c = bc
                print(f"  Coordinate polish improved to {best_score:.6f}")

        # Extra gradient pass if time remains
        now = time.monotonic()
        extra = max(0.0, total_deadline - now - 15.0)
        if extra > 120.0:
            print(f"\n=== Extra gradient pass: {min(extra, 200.0):.0f}s ===")
            # Perturb and optimize
            f_pert = _perturb(best_f, rng, best_f.size)
            f2, s2 = _gradient_optimize(f_pert, min(extra, 200.0), rng, "extra_grad")
            if s2 < best_score:
                best_score = s2
                best_f = f2.copy()
                best_c = _autocorr(best_f)
                print(f"  Extra pass improved to {best_score:.6f}")

            # One more coord polish
            remaining2 = max(0.0, total_deadline - time.monotonic() - 15.0)
            if remaining2 > 30.0:
                bf, bs, bc = _coordinate_polish(best_f, _autocorr(best_f), remaining2, rng)
                if bs < best_score:
                    best_score = bs
                    best_f = bf.copy()
                    best_c = bc

    except Exception as exc:
        print(f"Exception: {repr(exc)}")
        import traceback
        traceback.print_exc()

    # Final cleanup
    best_f = _normalize(best_f)
    best_f = np.nan_to_num(best_f, nan=0.0, posinf=0.0, neginf=0.0)
    np.clip(best_f, 0.0, 1000.0, out=best_f)
    best_f = best_f / max(1e-300, float(best_f.sum()))

    # Final score
    best_c = _autocorr(best_f)
    best_score = float(2.0 * best_f.size * best_c.max())

    # Try official evaluation
    try:
        out = [float(x) for x in best_f]
        official = evaluate_sequence(out)
        if np.isfinite(official) and official < best_score + 1e-12:
            best_score = float(official)
    except Exception:
        pass

    print(f"\nFINAL score = {best_score:.6f}")
    print(f"Target = 1.5027")

    out = [float(x) for x in best_f]
    return out
