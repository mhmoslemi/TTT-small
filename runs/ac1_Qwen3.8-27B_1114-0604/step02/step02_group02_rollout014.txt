    n = len(f)
    
    # Compute autocorrelation
    c = autocorr_fft(f)
    
    # Compute weights (softmax of c/T)
    c_scaled = c / T
    c_scaled -= c_scaled.max()

    # ifft(conj(fft(w, 2n)) * fft(f, 2n))

    # Substitute: let g = f reversed, then sum_k w[k]*f[k-j] = (w * g)[j] type thing
    # Simplest: just compute both and pick the right one
    cand1 = np.fft.irfft(W * np.conj(F), size)[:n]
    cand2 = np.fft.irfft(np.conj(W) * F, size)[:n]

    # Determine which is correct by checking at a small scale
    # h[0] should = sum_k w[k]*f[k] = dot(w[:n], f)
    h0_true = np.dot(w[:n], f)
    if abs(cand1[0] - h0_true) < abs(cand2[0] - h0_true):
        h = cand1
    else:
        h = cand2

    return (2.0 / T) * h


def _adam_step(z, grad_z, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step in log-space."""
    state['t'] += 1
    t = state['t']

    state['m'] = beta1 * state['m'] + (1 - beta1) * grad_z
    state['v'] = beta2 * state['v'] + (1 - beta2) * grad_z ** 2

    m_hat = state['m'] / (1 - beta1 ** t)
    v_hat = state['v'] / (1 - beta2 ** t)

    z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z_new


def _z_to_f(z, target_sum):
    """Convert log-space parameters to normalized f."""
    z_shifted = z - z.max()
    f = np.exp(z_shifted)
    s = f.sum()
    if s < 1e-300:
        return np.ones(len(z)) * target_sum / len(z)
    return f * (target_sum / s)


def _optimize_smooth(f_init, deadline, rng, n, target_sum):
    """
    Smooth optimization using log-sum-exp surrogate + Adam in log-space.
    """
    f = f_init.copy()
    f = np.maximum(f, 1e-30)
    f = f / f.sum() * target_sum

    z = np.log(f)
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}

    c = _convolve_fft(f)
    max_c = c.max()
    T = max_c  # Start very smooth

    local_best = max_c
    local_best_f = f.copy()
    local_best_score = 2.0 * n * local_best / (target_sum ** 2)

    lr = 0.05
    T_end = max_c * 1e-5
    total_iters = 0
    last_improve = time.monotonic()

    while time.monotonic() < deadline:
        c = _convolve_fft(f)
        cur_max = c.max()
        cur_score = 2.0 * n * cur_max / (target_sum ** 2)

        if cur_max < local_best - 1e-12:
            local_best = cur_max
            local_best_f = f.copy()
            local_best_score = 2.0 * n * local_best / (target_sum ** 2)
            last_improve = time.monotonic()

        # Temperature schedule: exponential decay
        stuck = time.monotonic() - last_improve
        if stuck < 30:
            T = max(T_end, T * 0.9997)
        else:
            # Reheat slightly when stuck
            T = max(T_end, T * 1.02)
            lr = max(0.001, lr * 0.95)

        # Compute gradient
        grad_f = _compute_smooth_grad(c, f, T)

        # Convert to z-space gradient (softmax parameterization)
        # f = softmax(z) * target_sum
        # df_j/dz_j = f_j - f_j * f_j / sum(f) = f_j(1 - f_j/target_sum)
        # df_j/dz_k = -f_j * f_k / target_sum (for j != k)
        # grad_z_j = sum_k grad_f_k * df_k/dz_j
        #           = grad_f_j * f_j(1 - f_j/S) - f_j * sum_k grad_f_k * f_k / S
        S = target_sum
        dot_gf = np.dot(grad_f, f)
        grad_z = grad_f * f - f * dot_gf / S

        # Adam step
        z = _adam_step(z, grad_z, state, lr)
        f = _z_to_f(z, target_sum)

        total_iters += 1

        if total_iters % 5000 == 0:
            print(f"    Smooth iter {total_iters}: max_c={cur_max:.10f}, "
                  f"score={cur_score:.8f}, T={T:.4e}, lr={lr:.6f}, "
                  f"best_score={local_best_score:.8f}")

    return local_best_f, local_best_score


def _optimize_subgrad(f_init, deadline, rng, n, target_sum):
    """
    Subgradient optimization with adaptive top-k and momentum.
    """
    f = f_init.copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n * target_sum
    else:
        f = f / s * target_sum

    c = _convolve_fft(f)
    local_best = c.max()
    local_best_f = f.copy()
    local_best_score = 2.0 * n * local_best / (target_sum ** 2)

    # Momentum-based approach in log-space
    z = np.log(np.maximum(f, 1e-300))
    momentum = np.zeros(n)
    beta_m = 0.9
    lr = 0.3
    last_improve = time.monotonic()
    iter_count = 0

    while time.monotonic() < deadline:
        c = _convolve_fft(f)
        cur_max = c.max()
        cur_score = 2.0 * n * cur_max / (target_sum ** 2)

        if cur_max < local_best - 1e-12:
            local_best = cur_max
            local_best_f = f.copy()
            local_best_score = 2.0 * n * local_best / (target_sum ** 2)
            last_improve = time.monotonic()

        stuck = time.monotonic() - last_improve

        # Adaptive top-k
        progress = min(1.0, iter_count / 10000.0)
        k = 3 + int(4 * progress)

        # Find top-k peaks
        top_indices = np.argsort(c)[-k:]

        # Compute subgradient: average of subgradients at top peaks
        grad_f = np.zeros(n)
        for idx in top_indices:
            # dc_idx/df_j = 2*f[idx-j] for 0<=idx-j<n
            lo = max(0, idx - n + 1)
            hi = min(n, idx + 1)
            # For j in [lo, hi-1], dc/df_j = 2*f[idx-j]
            js = np.arange(lo, hi)
            grad_f[js] += 2.0 * f[idx - js]
        grad_f /= k

        # Project to tangent space of {sum(f) = target_sum, f >= 0}
        # Use log-space gradient
        z_grad = grad_f * f - f * np.dot(grad_f, f) / target_sum

        # Momentum update
        momentum = beta_m * momentum + (1 - beta_m) * z_grad
        z = z - lr * momentum

        # Deal with stuck
        if stuck > 20:
            # Perturb
            mag = min(0.03, 0.005 + 0.002 * stuck / 30.0)
            z = z + mag * rng.standard_normal(n)
            lr = max(0.05, lr * 0.7)
            last_improve = time.monotonic()
        else:
            lr = max(0.01, lr * 0.9999)

        f = _z_to_f(z, target_sum)
        iter_count += 1

        if iter_count % 5000 == 0:
            print(f"    Subgrad iter {iter_count}: max_c={cur_max:.10f}, "
                  f"score={cur_score:.8f}, lr={lr:.4f}, best_score={local_best_score:.8f}")

    return local_best_f, local_best_score


def _polish_coordinate(f_init, deadline, rng, n, target_sum):
    """Coordinate descent polish: try moving mass from peak-contributing coords to others."""
    f = f_init.copy()
    f = np.maximum(f, 0.0)
    f = f / f.sum() * target_sum

    c = _convolve_fft(f)
    local_best = c.max()
    local_best_f = f.copy()
    local_best_score = 2.0 * n * local_best / (target_sum ** 2)

    passes = 0
    while time.monotonic() < deadline:
        c = _convolve_fft(f)
        cur_max = c.max()

        if cur_max < local_best - 1e-14:
            local_best = cur_max
            local_best_f = f.copy()
            local_best_score = 2.0 * n * local_best / (target_sum ** 2)

        k_star = int(np.argmax(c))

        # Compute contribution of each coordinate to c[k_star]
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        js = np.arange(lo, hi)
        contrib = np.zeros(n)
        contrib[js] = 2.0 * f[k_star - js]

        # Find high-contrib and low-contrib coordinates
        donor = int(np.argmax(contrib))
        low_candidates = np.argsort(contrib)[:max(5, n // 100)]

        improved = False
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
            for recv in low_candidates[:5]:
                if recv == donor:
                    continue
                amt = f[donor] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[donor] -= amt
                f_try[recv] += amt
                f_try = np.maximum(f_try, 0.0)
                if f_try.sum() < 1e-10:
                    continue
                f_try = f_try / f_try.sum() * target_sum
                c_try = _convolve_fft(f_try)
                if c_try.max() < cur_max - 1e-14:
                    f = f_try
                    improved = True
                    break
            if improved:
                break

        # Random pair moves
        if not improved:
            for _ in range(20):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                frac = rng.choice([0.01, 0.03, 0.08, 0.15, 0.3])
                amt = f[i] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[i] -= amt
                f_try[j] += amt
                if f_try[i] < 0:
                    continue
                f_try = f_try / f_try.sum() * target_sum
                c_try = _convolve_fft(f_try)
                if c_try.max() < cur_max - 1e-14:
                    f = f_try
                    break

        # Small global perturbation
        if not improved and passes % 30 == 0:
            f_try = f + 0.001 / np.sqrt(n) * rng.standard_normal(n)
            f_try = np.maximum(f_try, 0.0)
            if f_try.sum() < 1e-10:
                f_try = f.copy()
            else:
                f_try = f_try / f_try.sum() * target_sum
            c_try = _convolve_fft(f_try)
            if c_try.max() < cur_max - 1e-14:
                f = f_try

        passes += 1

    return local_best_f, local_best_score


def _try_symmetrize(f, target_sum, n):
    """Try to symmetrize f around different centers and check if it improves."""
    best_f = f.copy()
    c_best = _convolve_fft(f)
    best_peak = c_best.max()

    # Try symmetry around center of the peak
    c = _convolve_fft(f)
    k_star = int(np.argmax(c))
    center = k_star / 2.0

    # Symmetrize: f_sym[j] = (f[j] + f[round(2*center - j)]) / 2
    for center_offset in [-1, -0.5, 0, 0.5, 1]:
        c_offset = center + center_offset
        f_sym = np.zeros(n)
        for j in range(n):
            mirror = int(round(2 * c_offset - j))
            if 0 <= mirror < n:
                f_sym[j] = 0.5 * (f[j] + f[mirror])
            else:
                f_sym[j] = f[j]
        f_sym = np.maximum(f_sym, 0.0)
        if f_sym.sum() < 1e-10:
            continue
        f_sym = f_sym / f_sym.sum() * target_sum
        c_sym = _convolve_fft(f_sym)
        if c_sym.max() < best_peak:
            best_peak = c_sym.max()
            best_f = f_sym.copy()

    return best_f, best_peak


def _generate_starts(f_init, n, rng, target_sum):
    """Generate diverse starting points."""
    starts = [f_init.copy()]
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Gaussian
    for alpha in [0.5, 1.0, 2.0, 5.0, 10.0]:
        g = np.exp(-alpha * ((j - center) / center) ** 2)
        starts.append(g)

    # U-shaped
    for alpha in [1.0, 3.0, 10.0]:
        u = 1.0 + alpha * ((j - center) / center) ** 2
        starts.append(u)

    # Bimodal
    for pos in [0.2, 0.3, 0.4]:
        b = (np.exp(-((j - pos * n) / (0.05 * n)) ** 2) +
             np.exp(-((j - (1 - pos) * n) / (0.05 * n)) ** 2) + 0.001)
        starts.append(b)

    # Sine-based
    for freq in [1, 2, 3]:
        s1 = 1.0 + 0.5 * np.sin(freq * 2 * np.pi * j / n)
        s2 = 1.0 + 0.8 * np.abs(np.sin(freq * 2 * np.pi * j / n))
        starts.append(s1)
        starts.append(s2)

    # Ramp
    starts.append(j.astype(float) + 1.0)
    starts.append((n - j).astype(float) + 1.0)

    # Random
    for _ in range(15):
        r = rng.random()
        if r < 0.3:
            starts.append(rng.exponential(1.0, n))
        elif r < 0.5:
            starts.append(rng.dirichlet(np.ones(n) * 0.1))
        elif r < 0.7:
            starts.append(np.abs(rng.standard_normal(n)) + 0.01)
        elif r < 0.85:
            starts.append(np.ones(n) + 0.2 * rng.standard_normal(n))
        else:
            base = np.zeros(n)
            n_nz = n // 4
            idx = rng.choice(n, n_nz, replace=False)
            base[idx] = rng.exponential(1.0, n_nz)
            starts.append(base + 0.001)

    # Normalize all
    normalized = []
    for s in starts:
        s = np.maximum(s, 0.0)
        ss = s.sum()
        if ss < 1e-10:
            s = np.ones(n) / n
        else:
            s = s / ss * target_sum
        normalized.append(s)

    return normalized


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the non-negative sequence minimizing 2*n*max(conv(f,f))/(sum f)^2.
    Multi-phase: smooth Adam optimization -> subgradient -> coordinate polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0

    f_init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(f_init)
    target_sum = np.sqrt(2.0 * n)

    # Normalize initial
    s0 = f_init.sum()
    if s0 < 1e-10:
        f_init = np.ones(n) * target_sum / n
    else:
        f_init = f_init / s0 * target_sum

    c0 = _convolve_fft(f_init)
    init_score = 2.0 * n * c0.max() / (target_sum ** 2)
    print(f"n={n}, initial score={init_score:.6f}, peak={c0.max():.8f}")

    # Generate and score starts
    t0 = time.monotonic()
    starts = _generate_starts(f_init, n, rng, target_sum)
    scored = []
    for f_s in starts:
        c_s = _convolve_fft(f_s)
        scored.append((c_s.max(), f_s))
    scored.sort(key=lambda x: x[0])
    gen_time = time.monotonic() - t0
    print(f"Generated {len(starts)} starts in {gen_time:.1f}s")
    print(f"Top 5 peaks: {[f'{p:.8f}' for p, _ in scored[:5]]}")

    best_score = 2.0 * n * scored[0][0] / (target_sum ** 2)
    best_f = scored[0][1].copy()
    print(f"Best initial score: {best_score:.6f}")

    # Phase allocation
    phase1_end = time.monotonic() + budget_s * 0.40
    phase2_end = time.monotonic() + budget_s * 0.75
    phase3_end = deadline

    # Phase 1: Smooth optimization from top starts
    num_phase1_starts = 4
    for idx in range(num_phase1_starts):
        remaining = phase1_end - time.monotonic()
        if remaining < 30:
            break
        alloc = min(remaining * 0.25, 300.0)
        if alloc < 20:
            break

        peak0, f0 = scored[idx]
        print(f"\nPhase 1 start {idx+1}: peak={peak0:.8f}, budget={alloc:.0f}s")
        f_opt, score = _optimize_smooth(f0, time.monotonic() + alloc, rng, n, target_sum)

        if score < best_score:
            best_score = score
            best_f = f_opt.copy()
            print(f"  Improved to {best_score:.6f}")
        else:
            print(f"  Best from smooth: {score:.6f}")

    # Phase 2: Subgradient optimization from current best
    remaining = phase2_end - time.monotonic()
    if remaining > 30:
        print(f"\nPhase 2: Subgradient optimization, budget={remaining:.0f}s")
        f_opt, score = _optimize_subgrad(best_f, time.monotonic() + remaining - 5, rng, n, target_sum)
        if score < best_score:
            best_score = score
            best_f = f_opt.copy()
            print(f"  Improved to {best_score:.6f}")
        else:
            print(f"  Subgradient best: {score:.6f}")

    # Try symmetrization
    if time.monotonic() < phase3_end - 60:
        print("\nTrying symmetrization...")
        f_sym, sym_peak = _try_symmetrize(best_f, target_sum, n)
        sym_score = 2.0 * n * sym_peak / (target_sum ** 2)
        if sym_score < best_score:
            best_score = sym_score
            best_f = f_sym.copy()
            print(f"  Symmetrized to {best_score:.6f}")
        else:
            print(f"  Symmetric score: {sym_score:.6f} (no improvement)")

    # Phase 3: Coordinate descent polish
    remaining = phase3_end - time.monotonic()
    if remaining > 30:
        print(f"\nPhase 3: Coordinate polish, budget={remaining:.0f}s")
        f_pol, pol_score = _polish_coordinate(best_f, time.monotonic() + remaining - 5, rng, n, target_sum)
        if pol_score < best_score:
            best_score = pol_score
            best_f = f_pol.copy()
            print(f"  Polished to {best_score:.6f}")
        else:
            print(f"  Polish best: {pol_score:.6f}")

    # Final: one more smooth round if time
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\nFinal smooth round, budget={remaining:.0f}s")
        f_final, fin_score = _optimize_smooth(best_f, time.monotonic() + remaining - 5, rng, n, target_sum)
        if fin_score < best_score:
            best_score = fin_score
            best_f = f_final.copy()
            print(f"  Final: {best_score:.6f}")

    # Verify with official evaluator
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = evaluate_sequence(best_f.tolist())
    print(f"\nOfficial evaluation: {final_score:.6f}")
    if not np.isfinite(final_score):
        print("Invalid! Falling back to uniform.")
        best_f = np.ones(n) * target_sum / n
        final_score = evaluate_sequence(best_f.tolist())
        print(f"Uniform fallback: {final_score:.6f}")

    if final_score < best_score:
        best_score = final_score

    print(f"=== FINAL SCORE: {best_score:.6f} ===")
    return [float(x) for x in best_f]
