        p, sc = _optimize(best_overall, rng, t_alloc, beta_start=30, beta_end=2000, lr_start=0.12)
        _update(p, sc)

    # Run 2: Higher beta (sharper max focus)
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(200.0, t_rem * 0.4)
        t_alloc = max(15.0, t_alloc)
        p_init = _resample(best_overall, n_best, rng)
        if p_init is None:
            p_init = best_overall.copy()
        print(f"  Run 2: high-beta opt, t={t_alloc:.0f}s")
        p, sc = _optimize(p_init, rng, t_alloc, beta_start=100, beta_end=8000, lr_start=0.08)
        _update(p, sc)

    # Run 3: Frequency flattening + optimize
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        # Try several alpha values
        for alpha in [1.5, 2.0, 3.0, 4.0]:
            if time.monotonic() > t2_end - 10:
                break
            p_flat = _freq_flatten(best_overall, alpha)
            if p_flat is not None:
                sc = _score_norm(p_flat)
                _update(p_flat, sc)
                if sc < best_score_overall + 0.01:
                    p, sc2 = _optimize(p_flat, rng, min(30.0, max(10.0, t_rem * 0.1)),
                                       beta_start=50, beta_end=3000)
                    _update(p, sc2)

    # Run 4: Block refinement
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(120.0, t_rem * 0.5)
        t_alloc = max(10.0, t_alloc)
        print(f"  Run 4: block refine, t={t_alloc:.0f}s")
        p, sc = _block_refine(best_overall, rng, t_alloc)
        _update(p, sc)

    # Run 5: Try upsampled version
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        n_up = min(8192, n_best * 2)
        if n_up != n_best:
            t_alloc = min(100.0, t_rem * 0.4)
            t_alloc = max(10.0, t_alloc)
            p_up = _resample(best_overall, n_up, rng)
            if p_up is not None:
                print(f"  Run 5: upsample to n={n_up}, t={t_alloc:.0f}s")
                p, sc = _optimize(p_up, rng, t_alloc, beta_start=50, beta_end=3000)
                _update(p, sc)

    print(f"  Phase 2 done. Best: {best_score_overall:.8f} (n={len(best_overall)})")

    # ========== PHASE 3: Diverse restarts ==========
    print("Phase 3: Diverse restarts")
    t3_end = start_time + 0.78 * budget

    if best_overall is None:
        best_overall = np.ones(2048) / 2048
        best_score_overall = _score_norm(best_overall)

    n_best = len(best_overall)
    restart_inits = []

    # Various structured initializations
    restart_inits.append(_init_gaussian_mixture(n_best, rng))
    restart_inits.append(_init_gaussian_mixture(n_best, rng))
    restart_inits.append(_init_cosine(n_best, rng))
    restart_inits.append(_init_two_level(n_best, rng))

    # Noise-perturbed best
    for _ in range(2):
        noisy = best_overall * np.exp(rng.normal(0, 0.04, n_best))
        noisy = np.maximum(noisy, 1e-10)
        restart_inits.append(noisy / noisy.sum())

    # Resampled parent at best length
    if parent is not None and len(parent) != n_best:
        rp = _resample(parent, n_best, rng)
        if rp is not None:
            restart_inits.append(rp)

    t_rem = t3_end - time.monotonic()
    if len(restart_inits) > 0 and t_rem > 20:
        per_init = max(15.0, t_rem * 0.6 / len(restart_inits))
        for i, init in enumerate(restart_inits):
            if time.monotonic() > t3_end - 10:
                break
            t_alloc = min(per_init, t3_end - time.monotonic() - 5)
            if t_alloc < 5:
                break
            sc_init = _score_norm(init)
            if sc_init < best_score_overall * 1.1:
                p, sc = _optimize(init, rng, t_alloc, beta_start=50, beta_end=3000)
                _update(p, sc)

    print(f"  Phase 3 done. Best: {best_score_overall:.8f}")

    # ========== PHASE 4: Final polishing ==========
    print("Phase 4: Final polishing")

    if best_overall is not None:
        # High-beta polishing
        t_rem = deadline - time.monotonic()
        if t_rem > 30:
            t_alloc = min(0.35 * t_rem, 250.0)
            p, sc = _optimize(best_overall, rng, t_alloc, beta_start=200, beta_end=10000, lr_start=0.05)
            _update(p, sc)

        # Block refine
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.25 * t_rem, 100.0)
            p, sc = _block_refine(best_overall, rng, t_alloc)
            _update(p, sc)

        # One more gradient pass
        t_rem = deadline - time.monotonic()
        if t_rem > 15:
            t_alloc = min(0.3 * t_rem, 100.0)
            p, sc = _optimize(best_overall, rng, t_alloc, beta_start=500, beta_end=15000, lr_start=0.03)
            _update(p, sc)

    # ========== FINAL ==========
    print(f"{'=' * 60}")
    if best_overall is None:
        best_overall = np.ones(2048) / 2048

    p_final = _sanitize(best_overall)
    if p_final is None:
        p_final = np.ones(2048) / 2048

    final_score = _score_norm(p_final)
    print(f"  Final internal score: {final_score:.8f} (n={len(p_final)})")

    # Verify with official evaluator
    try:
        eval_score = evaluate_sequence([float(x) for x in p_final])
        if np.isfinite(eval_score):
            print(f"  Verified score via evaluator: {eval_score:.8f}")
    except Exception as e:
        print(f"  Evaluator error: {e}")

    print(f"  Total time: {time.monotonic() - start_time:.1f}s")

    out = np.where(np.isfinite(p_final), p_final, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = np.ones(2048) / 2048

    return [float(x) for x in out]
