import numpy as np
import time

def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def get_gradient_soft(a, c, topk, temp_frac):
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S
    
    k = min(topk, c.size)
    top_idx = np.argpartition(c, -k)[-k:]
    top_vals = c[top_idx].copy()
    
    v_max = float(top_vals[0])
    if v_max < 1e-30:
        return np.zeros(n), v_max
    
    t = max(temp_frac * v_max, 1e-15)
    w = np.exp((top_vals - v_max) / t)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros(k)
        w[0] = 1.0
    else:
        w /= w_sum
    
    M_wf = float(np.dot(top_vals, w))
    
    grad_c = np.zeros(n)
    for i in range(k):
        wi = float(w[i])
        if wi < 1e-15:
            continue
        ki = int(top_idx[i])
        lo = max(0, ki - n + 1)
        hi = min(ki, n - 1)
        if lo <= hi:
            grad_c[lo:hi+1] += 2.0 * wi * a[ki-hi:ki-lo+1][::-1]
    
    grad = 2.0 * n * (grad_c / S2 - 2.0 * M_wf / S3)
    return grad, M_wf

def optimize_phase(a, end_time, rng, lr_start, lr_end, topk_start, topk_end, 
                   temp_start, temp_end, patience, normalize=True):
    n = a.size
    a = a.astype(np.float64).copy()
    if normalize:
        a = np.clip(a, 0, 1000.0)
        if a.sum() > 0:
            a /= a.sum()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_score = np.inf
    best_a = a.copy()
    no_imp = 0
    total_steps = 10000  # for scheduling
    
    start_time = time.monotonic()
    # Estimate total time for scheduling
    elapsed_initial = time.monotonic() - start_time
    
    while time.monotonic() < end_time:
        # Schedule parameters
        # Use actual elapsed time to determine progress
        # For now, use step count
        t += 1
        progress = min(1.0, t / 50000.0)  # normalize over 50K steps
        
        lr = lr_start * (lr_end / lr_start) ** progress
        topk = int(topk_start + (topk_end - topk_start) * progress)
        topk = max(1, topk)
        temp = temp_start + (temp_end - temp_start) * progress
        
        c = fast_self_conv(a)
        S = float(a.sum())
        score = compute_score(n, c, S)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if no_imp >= patience:
            # Restart from best with perturbation
            pert = 0.05 + 0.1 * rng.random()
            a = best_a * (1.0 + pert * rng.standard_normal(n))
            a = np.clip(a, 0, 1000.0)
            if a.sum() < 1e-6:
                a = np.ones(n) / n
            else:
                a /= a.sum()
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            no_imp = 0
            continue
        
        grad, M = get_gradient_soft(a, c, topk, temp)
        
        # Adam step
        b1, b2 = 0.9, 0.999
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1**t)
        v_hat = v / (1 - b2**t)
        
        a -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        a = np.clip(a, 0, 1000.0)
        
        if normalize:
            if a.sum() > 1e-10:
                a /= a.sum()
    
    return best_score, best_a

Actually, let me think about the problem from a spectral perspective. We want to minimize max(c) where c = a*a (self-convolution) and sum(a) = 1, a >= 0.

In the frequency domain: C(f) = |A(f)|^2 (approximately, for the DFT of the zero-padded sequence). We want c to be flat, which means C should be concentrated at DC. But C[0] = (sum a)^2 = 1, and the sum of C is also 1 (by Parseval, up to scaling). So we want C[f] ≈ 0 for f ≠ 0.

This means we want A(f) ≈ 0 for f ≠ 0, i.e., a should be as close to a delta as possible in the frequency domain, which means a should be as spread out as possible in the time domain. But a spread-out a (like uniform) gives a triangular c, not a flat one!

The resolution is that the relationship between the time and frequency domains for FINITE sequences is more complex. The uniform sequence has A(f) = Dirichlet kernel, which is NOT zero at f ≠ 0. To make A(f) = 0 at f ≠ 0 (for all f in the DFT grid), we need a to be a delta, which gives c as a delta (peaky).

So there's a fundamental tension: we want A(f) small at f ≠ 0 (to make c flat), but this requires a to be concentrated (which makes c peaky). The optimum is somewhere in between.

This suggests that the optimal a has a frequency response that's "flat" in some sense, like a window function. Specifically, if A(f) = W(f) where W is a window (like Hann, Blackman, etc.), then c = IFFT(|W|^2). The peak of c is minimized when |W|^2 is as flat as possible.

Wait, but W(f) is the DFT of a, and a must be non-negative. So W(f) = sum_j a[j] exp(-2πifj/M) where a[j] >= 0. This means W(f) is a sum of complex exponentials with non-negative weights. The question is: can we choose the weights to make |W(f)|^2 flat?

This is related to the problem of finding a non-negative sequence with a flat power spectrum. Such sequences are known to exist (they're called "flat spectrum" sequences or "Welch bounded" sequences), but they're hard to find.

One approach: start with a window function in the frequency domain, take the IFFT, and clip to non-negative. Then iterate.
