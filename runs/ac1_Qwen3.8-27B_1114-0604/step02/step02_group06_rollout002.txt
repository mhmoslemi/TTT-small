"""
Optimized step function search using log-space L-BFGS with adaptive smooth-max,
combined with multiplicative field updates and local block search.
Minimizes 2n*max(conv(a,a))/sum(a)^2 for non-negative sequences.

import numpy as np
from scipy.optimize import minimize
import time

_GLOBAL_RNG = np.random.default_rng(7)


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _sanitize(p, min_val=1e-15, normalize=True):
    try:
        a = np.asarray(p, dtype=float).ravel()
    except Exception:
        return None
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if min_val > 0:
        a = a + min_val
    if normalize:
        s = float(a.sum())
        if s <= 0.0 or not np.isfinite(s):
            return None
        a = a / s
    return a


def _convolve_fft(a, L=None):
    n = int(a.size)
    if L is None:
        L = _next_pow2(2 * n - 1)
    Fp = np.fft.rfft(a, L)
    c = np.fft.irfft(Fp * Fp, L)
    c = c[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c, L


def _score(a, L=None):
    n = int(a.size)
    c, L = _convolve_fft(a, L)
    return float(2.0 * n * c.max()), c, L


def _smooth_obj_grad(x, epsilon, L, n):
    """
    x: log-space parameters, a = exp(x), normalized to sum=1.
    Returns (objective, gradient) of smooth approximation.
    """
    # For numerical stability, shift x
    x_shift = x - x.max()
    a = np.exp(x_shift)
    s = float(a.sum())
    if s <= 0 or not np.isfinite(s):
        return 1e10, np.zeros_like(x)
    a = a / s

    Fp = np.fft.rfft(a, L)
    c = np.fft.irfft(Fp * Fp, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)

    max_c = float(c.max())
    shifted = epsilon * (c - max_c)
    exp_shifted = np.exp(shifted)
    log_sum = float(max_c + np.log(np.sum(exp_shifted)) / epsilon)
    obj = 2.0 * n * log_sum

    # Softmax weights
    w = exp_shifted / float(exp_shifted.sum())

    # Gradient h[i] = sum_k w[k] * a[k-i] (cross-correlation)
    Fw = np.fft.rfft(w, L)
    h = np.fft.irfft(Fw * np.conjugate(Fp), L)[:n]
    h = np.maximum(h, 0.0)

    # dF/da_i = 4n * h_i
    # dF/dx_i = 4n * h_i * a_i (since a = softmax(x))
    grad = 4.0 * n * h * a

    return obj, grad

    n = int(a_init.size)
    L = _next_pow2(2 * n - 1)

    a_safe = a_safe / a_safe.sum()
    x0 = np.log(a_safe)

    best_a = a_init.copy()
    best_score = _score(a_init, L)[0]
    deadline = time.monotonic() + time_limit

    # Adaptive epsilon schedule
    eps = float(epsilon_start)
    stuck_counter = 0
    prev_score = best_score

    while time.monotonic() < deadline and eps < epsilon_end:
        t_sub = min(30.0, max(5.0, (deadline - time.monotonic()) * 0.3))

        def fun(x):
            return _smooth_obj_grad(x, eps, L, n)[0]

        def fun_and_grad(x):
            return _smooth_obj_grad(x, eps, L, n)

        try:
            res = minimize(fun_and_grad, x0, method='L-BFGS-B',
                           jac=True, options={'maxiter': 200, 'gtol': 1e-8,
                                              'maxls': 50, 'iprint': 0})
        except Exception:
            break

        x_new = res.x
        a_new = np.exp(x_new - x_new.max())
        s = float(a_new.sum())
        if s > 0 and np.isfinite(s):

            sc_new = _score(a_new, L)[0]
        else:

            best_score = sc_new
            best_a = a_new.copy()
            x0 = x_new.copy()
            stuck_counter = 0

                print(f"  L-BFGS eps={eps:.1f} score={best_score:.7f}")
        else:
            stuck_counter += 1
            x0 = x_new.copy()  # Still update to continue from here

        # Increase epsilon
        eps *= 1.5
        if stuck_counter > 3:

    parent = None
    try:
        p = _sanitize(height_sequence_1, 1e-15, True)
        if p is not None:
            parent = p
    except Exception:
        pass

    if parent is None:
        parent = np.ones(3072, dtype=float) / 3072.0

    n = int(parent.size)
    L = _next_pow2(2 * n - 1)

    best = parent.copy()
    best_score, _, _ = _score(parent, L)
    print(f"Initial (parent) score: {best_score:.7f}")

    # Build candidate list
    cands = [(best_score, n, best.copy())]

    def add_cand(a, sc):
        nonlocal best, best_score
        a = _sanitize(a, 1e-15, True)
        if a is None:
            return
        if sc is None:
            sc, _, _ = _score(a, L)
        if np.isfinite(sc):
            cands.append((sc, n, a.copy()))
            if sc < best_score:
                best_score = sc
                best = a.copy()
                print(f"  NEW BEST: {best_score:.7f}")

    # === Phase 1: L-BFGS from parent ===
    rem = float(deadline - time.monotonic())
    if rem > 120:
        t1 = min(rem * 0.25, 400.0)
        print(f"Phase 1: L-BFGS, time={t1:.0f}s")
        a1, sc1 = _lbfgs_phase(parent, t1, epsilon_start=20, epsilon_end=400, verbose=True)
        add_cand(a1, sc1)

    # === Phase 2: MMD from best ===
    rem = float(deadline - time.monotonic())
    if rem > 120:
        t2 = min(rem * 0.25, 350.0)
        print(f"Phase 2: MMD, time={t2:.0f}s")
        a2, sc2 = _mmd_phase(best, t2, target=1.500, verbose=True)
        add_cand(a2, sc2)

    # === Phase 3: L-BFGS from MMD result ===
    rem = float(deadline - time.monotonic())
    if rem > 120:
        t3 = min(rem * 0.2, 250.0)
        print(f"Phase 3: L-BFGS round 2, time={t3:.0f}s")
        a3, sc3 = _lbfgs_phase(best, t3, epsilon_start=50, epsilon_end=800, verbose=True)
        add_cand(a3, sc3)

    # === Phase 4: Local search ===
    rem = float(deadline - time.monotonic())
    if rem > 80:
        t4 = min(rem * 0.2, 200.0)
        print(f"Phase 4: Local search, time={t4:.0f}s")
        a4, sc4 = _local_search(best, t4, verbose=True)
        add_cand(a4, sc4)

    # === Phase 5: Restart from perturbed best ===
    for restart in range(3):
        rem = float(deadline - time.monotonic())
        if rem < 100:
            break
        sigma = 0.05 + 0.03 * restart
        perturbed = _noise_array(best, sigma)
        if perturbed is None:
            continue

        t_mmd = min(rem * 0.2, 150.0)
        t_lbfgs = min(rem * 0.15, 100.0)

        if t_mmd > 30:
            print(f"Phase 5.{restart}: MMD restart (sigma={sigma:.2f})")
            ar, scr = _mmd_phase(perturbed, t_mmd, target=1.500, verbose=False)
            add_cand(ar, scr)

        rem = float(deadline - time.monotonic())
        if rem > 60 and t_lbfgs > 20:
            ar2, scr2 = _lbfgs_phase(best, t_lbfgs, epsilon_start=100, epsilon_end=1000, verbose=False)
            add_cand(ar2, scr2)

    # === Phase 6: Final polish ===
    rem = float(deadline - time.monotonic())
    if rem > 60:
        t6 = min(rem - 20, 150.0)
        print(f"Phase 6: Final polish, time={t6:.0f}s")
        a6, sc6 = _lbfgs_phase(best, t6, epsilon_start=200, epsilon_end=2000, verbose=True)
        add_cand(a6, sc6)

    # === Phase 7: More local search ===
    rem = float(deadline - time.monotonic())
    if rem > 40:
        t7 = min(rem - 15, 100.0)
        print(f"Phase 7: Final local search, time={t7:.0f}s")
        a7, sc7 = _local_search(best, t7, verbose=True)
        add_cand(a7, sc7)

    # Final: select best
    cands.sort(key=lambda x: x[0])
    final_arr = cands[0][2].copy()
    final_score = float(cands[0][0])
    print(f"Final score from candidates: {final_score:.7f}")

    # Verify with evaluate_sequence if available and cheap
    if final_arr.size <= 4096 and float(deadline - time.monotonic()) > 10:
        lst = [float(x) for x in final_arr]
        try:
            verified = float(evaluate_sequence(lst))
            if np.isfinite(verified) and verified < final_score:
                final_score = verified
                print(f"Verified score: {verified:.7f}")
        except Exception:
            pass

    out = np.asarray(final_arr, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s <= 0 or not np.isfinite(s):
        out = np.ones(2048, dtype=float)
    else:
        out = out / s

    return [float(x) for x in out]
