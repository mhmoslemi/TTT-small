```python
"""
LP-guided optimization for non-negative sequences minimizing
    2*n*max(conv(a,a)) / sum(a)^2
Uses LP relaxation for global structure, L-BFGS for local refinement,
and multi-length exploration.
"""

import numpy as np
import time
import cvxpy as cp
from scipy.optimize import minimize
from scipy.fft import rfft, irfft, next_fast_len


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_fft(a, b, L=None):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 500000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = next_fast_len(nout)
    out = irfft(rfft(a, L) * rfft(b, L), L)
    return out[:nout].astype(float)


def _score_normalized(p):
    """Score assuming sum(p) = 1."""
    n = len(p)
    if n == 0:
        return float('inf')
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    c = _conv_fft(p, p)
    return float(2.0 * n * c.max())


def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    return _score_normalized(p)


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s


def _smoothed_max(c, beta):
    """Smoothed max of array c."""
    shift = c.max()
    e = np.exp(beta * (c - shift))
    return shift + np.log(e.sum()) / beta


def _smoothed_max_grad(c, beta, a):
    """
    Compute gradient of smoothed max of conv(a,a) w.r.t. a.
    Returns (value, gradient).
    """
    n = len(a)
    shift = c.max()
    e = np.exp(beta * (c - shift))
    w = e / e.sum()
    value = shift + np.log(e.sum()) / beta
    # gradient[j] = 2 * sum_k w[k] * a[k-j] (cross-correlation)
    # Using FFT
    L = next_fast_len(3 * n)
    W = rfft(w, L)
    A = rfft(a, L)
    R = irfft(W * np.conj(A), L)
    grad = 2.0 * R[:n]
    return value, grad


def _lp_update(a, M, solver='CLARABEL'):
    """
    Solve LP: maximize sum(b) subject to b >= 0, conv(a,b) <= M.
    Returns optimal b (normalized to sum 1) and optimal value.
    """
    n = len(a)
    b = cp.Variable(n, nonneg=True)
    
    # Build convolution constraint matrix
    # conv(a,b)_k = sum_j a[j] * b[k-j] for valid j
    # We need this for k = 0, ..., 2n-2
    
    # Using sparse matrix representation
    # Row k: (A x b)_k = sum_j a[j] * b[k-j]
    # For row k, the coefficient of b[j] is a[k-j] if 0 <= k-j <= n-1
    
    import scipy.sparse as sp
    
    rows = []
    cols = []
    vals = []
    
    for k in range(2 * n - 1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[idx])
    
    A_mat = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    constraints = [A_mat @ b <= M]
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
        if b.value is not None:
            b_val = np.asarray(b.value, dtype=float)
            b_val = np.maximum(b_val, 0)
            s = b_val.sum()
            if s > 1e-10:
                return b_val / s, float(s), float(prob.optimum)
    except Exception:
        pass
    
    return None, None, None

def _lp_iteration(a, M_target=None, solver='CLARABEL', verbose=False):

    One LP iteration: solve LP and return mixed sequence.

        c = _conv_fft(a, a)
        M_target = c.max()

def _resample(p_in, n_new, rng=None):
    p = _sanitize(p_in)
    if p is None:
        return None
    n_old = len(p)
    if n_old == n_new:
        result = p.copy()
    elif n_old == 1:
        result = np.ones(n_new, dtype=float)
    else:
        x_old = (np.arange(n_old) + 0.5) / n_old
        x_new = (np.arange(n_new) + 0.5) / n_new
        result = np.interp(x_new, x_old, p)
    result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s


def _multiply_update(a, h, alpha, floor=1e-15):
    """Multiplicative update: a_new = a * exp(-alpha * h / max(h)), then normalize."""
    max_h = h.max()
    if max_h < 1e-15:
        return a
    h_norm = h / max_h
    a_new = a * np.exp(-alpha * h_norm)
    a_new = np.maximum(a_new, floor)
    s = a_new.sum()
    if s < 1e-10:
        return a
    return a_new / s
```

```python
import scipy.sparse as sp

def _build_conv_matrix(a):
    n = len(a)
    # A is (2n-1) x n
    # A[k, j] = a[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):
            # a[k-j]
            if k < 2*n - 1:
                rows.append(k)
                cols.append(j)
                vals.append(a[k-j])
    A = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A
```

"""
LP-guided + Spectral Flattening + Gradient optimization for non-negative sequences
minimizing 2*n*max(conv(a,a))/sum(a)^2.

New approaches vs previous:
1. LP-based structural updates (SOTA-inspired but implemented with HiGHS)

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog, minimize
from scipy.fft import rfft, irfft, next_fast_len


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_fft(a, b, L=None):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 500000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = next_fast_len(nout)
    out = irfft(rfft(a, L) * rfft(b, L), L)
    return out[:nout].astype(float)


def _score_normalized(p):
    n = len(p)
    if n == 0:
        return float('inf')
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    c = _conv_fft(p, p)
    return float(2.0 * n * c.max())


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s


def _smoothed_max_grad(c, beta, a):
    n = len(a)
    shift = c.max()
    e = np.exp(beta * (c - shift))
    e_sum = e.sum()
    if e_sum < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = e / e_sum
    value = shift + np.log(e_sum) / beta
    L = next_fast_len(3 * n)
    W = rfft(w, L)
    A = rfft(a, L)
    R = irfft(W * np.conj(A), L)
    grad = 2.0 * R[:n]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return value, grad


def _build_conv_matrix(a):
    n = len(a)
    rows = np.repeat(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    # A[k,j] = a[k-j] for k in [j, j+n-1]
    # For column j, row k = j + offset, value = a[offset] for offset 0..n-1
    all_rows = []
    all_cols = []
    all_vals = []
    for j in range(n):
        offsets = np.arange(n)
        rows_j = j + offsets
        mask = rows_j < 2 * n - 1
        all_rows.extend(rows_j[mask].tolist())
        all_cols.extend([j] * int(mask.sum()))
        all_vals.extend(a[offsets[mask]].tolist())
    A = sp.csr_matrix((all_vals, (all_rows, all_cols)), shape=(2 * n - 1, n))
    return A


def _lp_solve(a, M, timeout_s=30.0):
    n = len(a)
    A_mat = _build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            b_val = np.maximum(res.x, 0)
            s = b_val.sum()
            if s > 1e-10:
                return b_val / s, float(s)
    except Exception:
        pass
    return None, None


def _spectral_flatten(a, iterations=10, n_ignore=0):
    n = len(a)
    L = next_fast_len(2 * n - 1)
    target = 1.0 / n
    for _ in range(iterations):
        A = rfft(a, L)
        C = A * A
        C_mag = np.abs(C)
        C_phase = C / np.maximum(C_mag, 1e-30)
        C_target = C_phase * target
        a_new = irfft(C_target, L)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        if np.allclose(a, a_new, atol=1e-10):
            break
        a = a_new
    return a


def _spectral_flatten_power(a, alpha, iterations=5):
    n = len(a)
    L = next_fast_len(2 * n - 1)
    for _ in range(iterations):
        A = rfft(a, L)
        C = A * A
        C_mag = np.abs(C)
        cmax = C_mag.max()
        if cmax < 1e-30:
            break
        C_flat = C_mag / cmax
        C_new_mag = C_flat ** (1.0 / alpha) * cmax
        C_phase = C / np.maximum(C_mag, 1e-30)
        C_target = C_phase * C_new_mag
        A_target = np.sqrt(C_target.astype(complex) + 0j)
        a_new = irfft(A_target, L)[:n]
        a_new = np.maximum(a_new, 0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        a = a_new
    return a


def _gradient_descent_loop(a, deadline, beta_start=30.0, beta_end=5000.0, verbose=False):
    p = _sanitize(a)
    if p is None:
        return None, float('inf')
    n = len(p)
    best = p.copy()
    best_score = _score_normalized(p)
    cur_score = best_score
    beta = beta_start
    alpha = 0.12
    stall = 0
    it = 0
    while time.monotonic() < deadline:
        c = _conv_fft(p, p)
        val, grad = _smoothed_max_grad(c, beta, p)
        max_h = grad.max()
        if max_h < 1e-15:
            break
        h_norm = grad / max_h
        improved = False
        for a_try in (alpha * 0.5, alpha, alpha * 1.5, alpha * 2.0):
            a_try = float(np.clip(a_try, 0.01, 3.0))
            p_try = p * np.exp(-a_try * h_norm)
            p_try = np.maximum(p_try, 1e-18)
            s_try = p_try.sum()
            if s_try < 1e-10:
                continue
            p_try = p_try / s_try
            sc_try = _score_normalized(p_try)
            if sc_try < cur_score - 1e-14:
                p = p_try
                cur_score = sc_try
                improved = True
                stall = 0
                break
        if not improved:
            stall += 1
            if stall > 8:
                alpha = max(0.01, alpha * 0.7)
            if stall > 40:
                p = best.copy()
                cur_score = best_score
                alpha = 0.12
                stall = 10
        if cur_score < best_score:
            best = p.copy()
            best_score = cur_score
        if it % 5 == 0:
            beta = min(beta_end, beta * 1.12)
        it += 1
    return best, best_score


def _lbgfs_optimize(p0, time_budget, beta_start=50.0, beta_end=1000.0, verbose=False):
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    best = p.copy()
    best_score = _score_normalized(p)
    deadline = time.monotonic() + time_budget
    beta = beta_start
    x = p.copy()
    it = 0
    while time.monotonic() < deadline:
        L = next_fast_len(3 * n)

        def obj_grad(xx):
            aa = np.maximum(xx, 1e-15)
            ss = aa.sum()
            if ss < 1e-10:
                return 1e10, np.zeros(n)
            aa = aa / ss
            cc = _conv_fft(aa, aa)
            shift = cc.max()
            ee = np.exp(beta * (cc - shift))
            ee_sum = ee.sum()
            if ee_sum < 1e-300:
                w = np.ones(2 * n - 1) / (2 * n - 1)
            else:
                w = ee / ee_sum
            val = shift + np.log(ee_sum) / beta
            W = rfft(w, L)
            A = rfft(aa, L)
            R = irfft(W * np.conj(A), L)
            grad_a = 2.0 * R[:n]
            grad_a = np.where(np.isfinite(grad_a), grad_a, 0.0)
            dot_ga = np.dot(grad_a, aa)
            grad_x = (grad_a - aa * dot_ga) / ss
            return val, grad_x

        try:
            res = minimize(
                obj_grad, x, method='L-BFGS-B',
                bounds=[(0, None)] * n,
                options={'maxiter': 30, 'ftol': 1e-14, 'gtol': 1e-8, 'maxls': 15}
            )
            if res.x is not None:
                x = np.maximum(res.x, 0)
                s = x.sum()
                if s > 1e-10:
                    x = x / s
                    sc = _score_normalized(x)
                    if sc < best_score:
                        best = x.copy()
                        best_score = sc
                        if verbose:
                            print(f"      L-BFGS iter={it}, beta={beta:.0f}, score={best_score:.6f}")
        except Exception:
            pass
        beta = min(beta * 1.5, beta_end)
        it += 1
        if beta >= beta_end:
            break
    return best, best_score


def _init_uniform(n):
    return np.ones(n, dtype=float) / n


def _init_two_level(n, rng):
    k = int(n * rng.uniform(0.3, 0.7))
    ratio = float(rng.uniform(1.5, 5.0))
    p = np.ones(n, dtype=float)
    p[k:] = ratio
    idx = rng.permutation(n)
    p = p[idx]
    p = p * (1.0 + 0.1 * rng.normal(0, 1, n))
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return _init_uniform(n)
    return p / s


def _init_gaussian(n, rng):
    K = int(rng.integers(2, 8))
    p = np.zeros(n, dtype=float)
    t = np.arange(n, dtype=float) / n
    for _ in range(K):
        c = rng.uniform(0.0, 1.0)
        s = rng.uniform(0.02, 0.3)
        w = rng.uniform(0.5, 2.0)
        x = (t - c) / s
        x = np.minimum(np.abs(x), 1.0 - np.abs(x))
        p += w * np.exp(-0.5 * x * x)
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _init_cosine(n, rng):
    t = np.arange(n, dtype=float) / n
    p = np.ones(n, dtype=float)
    for k in range(1, 12):
        amp = rng.uniform(-0.6, 0.6) / k
        ph = rng.uniform(0, 2 * np.pi)
        p += amp * np.cos(2 * np.pi * k * t + ph)
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return _init_uniform(n)
    return p / s


def _init_barker_like(n, rng):
    p = np.ones(n, dtype=float)
    block = int(max(2, n // int(rng.integers(20, 60))))
    for i in range(0, n - block + 1, block):
        if rng.random() < 0.5:
            p[i:i + block] *= rng.uniform(0.3, 0.7)
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _resample(p_in, n_new, rng=None):
    p = _sanitize(p_in)
    if p is None:
        return None
    n_old = len(p)
    if n_old == n_new:
        result = p.copy()
    elif n_old == 1:
        result = np.ones(n_new, dtype=float)
    else:
        x_old = (np.arange(n_old) + 0.5) / n_old
        x_new = (np.arange(n_new) + 0.5) / n_new
        result = np.interp(x_new, x_old, p)
    result = np.maximum(result, 0)
    if rng is not None:
        result = result * np.exp(rng.normal(0, 0.02, n_new))
        result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s

        best_score_overall = _score_normalized(best_overall)
    n_best = len(best_overall)
    # LP-guided on best
    t_alloc = min(t2_end - time.monotonic() - 60, 0.35 * budget)
    if t_alloc > 30:
        print(f"  LP-guided on n={n_best} (budget={t_alloc:.0f}s)")
        p, sc = _lp_guided_optimize(best_overall, time.monotonic() + t_alloc, verbose=True)
        _try_update(p, sc)
    # Spectral on best
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(120.0, t_rem * 0.5)
        print(f"  Spectral on n={n_best} (budget={t_alloc:.0f}s)")
        p, sc = _spectral_optimize(best_overall, time.monotonic() + t_alloc, verbose=True)
        _try_update(p, sc)
    # Try at different length
    t_rem = t2_end - time.monotonic()
    if t_rem > 60:
        for n_try in [2048, 4096]:
            if n_try == n_best or time.monotonic() > t2_end - 30:
                continue
            p_r = _resample(best_overall, n_try, rng)
            if p_r is None:
                p_r = _init_uniform(n_try)
            sc_r = _score_normalized(p_r)
            t_alloc_r = min(60.0, t2_end - time.monotonic() - 10)
            if t_alloc_r > 20:
                p, sc = _lp_guided_optimize(p_r, time.monotonic() + t_alloc_r, verbose=False)
                _try_update(p, sc)
    print(f"  Phase 2 done. Best: {best_score_overall:.6f} (n={len(best_overall)})")

    # Phase 3: Intensive gradient + L-BFGS (25%)
    print(f"\nPhase 3: Intensive optimization")
    t3_end = start_time + 0.80 * budget
    if best_overall is not None:
        # High-beta gradient
        t_rem = t3_end - time.monotonic()
        if t_rem > 30:
            t_alloc = min(100.0, t_rem * 0.4)
            p, sc = _gradient_descent_loop(best_overall, time.monotonic() + t_alloc,
                                           beta_start=100.0, beta_end=8000.0, verbose=True)
            _try_update(p, sc)
        # L-BFGS
        t_rem = t3_end - time.monotonic()
        if t_rem > 30:
            t_alloc = min(120.0, t_rem * 0.5)
            p, sc = _lbgfs_optimize(best_overall, t_alloc,
                                    beta_start=50.0, beta_end=2000.0, verbose=True)
            _try_update(p, sc)
        # More LP
        t_rem = t3_end - time.monotonic()
        if t_rem > 60 and len(best_overall) <= 4096:
            t_alloc = min(120.0, t_rem * 0.6)
            p, sc = _lp_guided_optimize(best_overall, time.monotonic() + t_alloc, verbose=True)
            _try_update(p, sc)
    print(f"  Phase 3 done. Best: {best_score_overall:.6f}")

    # Phase 4: Final polishing (last 20%)
    print(f"\nPhase 4: Final polishing")
    if best_overall is not None:
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(60.0, t_rem * 0.3)
            p, sc = _gradient_descent_loop(best_overall, time.monotonic() + t_alloc,
                                           beta_start=1000.0, beta_end=20000.0)
            _try_update(p, sc)
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(60.0, t_rem * 0.3)
            p, sc = _lbgfs_optimize(best_overall, t_alloc,
                                    beta_start=500.0, beta_end=5
