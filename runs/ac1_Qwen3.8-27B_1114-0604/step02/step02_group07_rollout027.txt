"""
Log-sum-exp gradient optimizer for minimizing 2n*max(conv(a,a))/(sum a)^2.
Uses smooth surrogate (log-sum-exp) for the max, with Adam optimizer,
adaptive temperature annealing, and periodic restarts.

import numpy as np
import time


def fast_self_conv(a):
    """Compute full self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, c, S):
    """Compute the objective value."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_gradient(a, tau):
    """
    Compute gradient of LSE(conv(a,a), tau) with respect to a.
    Returns (dM_da, p, M) where p = softmax(b/tau), M = LSE value.
    """
    n = len(a)
    b = fast_self_conv(a)  # length 2n-1
    
    # Compute softmax
    logits = b / tau
    logits -= logits.max()
    exp_logits = np.exp(logits)
    p = exp_logits / exp_logits.sum()  # length 2n-1
    M = tau * (float(np.log(exp_logits.sum())) + float(b.max()) / tau)
    # Actually M = tau * log(sum(exp(b/tau))) = tau * (log(sum(exp(b/tau - b_max/tau))) + b_max/tau)
    # = tau * log(exp_logits.sum()) + b_max
    M = tau * float(np.log(exp_logits.sum())) + float(b.max())
    
    # Compute dM/da using the derived formula:
    # dM/da[j] = T1[j] + T2[j]
    # T1[j] = conv(rev_a, p)[n-1+j]
    # T2[j] = conv(a, p)[j]
    
    rev_a = a[::-1]
    # Need FFT size at least (n-1) + (2n-2) + 1 = 3n-2 for the convolutions
    M_fft = 1 << (3 * n - 3).bit_length()
    A_fft = np.fft.rfft(a, M_fft)
    PA_fft = np.fft.rfft(p, M_fft)
    conv_a_p = np.fft.irfft(A_fft * PA_fft, M_fft)[:3*n-2]

    T1 = conv_ra_p[n-1:2*n-1].copy()

def full_objective_gradient(a, tau):
    """
    Compute gradient of f(a) = 2n * LSE(conv(a,a), tau) / (sum(a))^2.
    """
    n = len(a)
    S = float(a.sum())
    dM_da, p, M = lse_gradient(a, tau)
    
    # f = 2n * M / S^2
    # df/da[j] = 2n * (dM/da[j] * S - 2*M) / S^3
    grad = 2.0 * n * (dM_da * S - 2.0 * M) / (S * S * S)
    return grad


def adam_step(a, g, m, v, t, lr):
    """In-place Adam optimizer step."""
    t1 = 0.9 ** t
    t2 = 0.999 ** t
    m[:] = 0.9 * m + 0.1 * g
    v[:] = 0.999 * v + 0.001 * g * g
    a -= lr * (m / (1.0 - t1)) / (np.sqrt(v / (1.0 - t2)) + 1e-8)

"""
Log-sum-exp gradient optimizer for minimizing 2n*max(conv(a,a))/(sum a)^2.
Uses smooth LSE surrogate for max with Adam, temperature annealing,
FFT-based gradient computation, and periodic restarts for local minimum escape.
"""

import numpy as np
import time


def fast_self_conv(a):
    """Compute full self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, c, S):
    """Compute the objective value."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_grad_full(a, tau):
    """
    Compute gradient of LSE(conv(a,a), tau) w.r.t. a using FFT.
    Returns (dM_da, M).
    """
    n = len(a)
    b = fast_self_conv(a)  # length 2n-1

    # Softmax weights
    logits = b / tau
    logits -= logits.max()
    exp_logits = np.exp(logits)
    p = exp_logits / exp_logits.sum()  # length 2n-1
    M = float(b.max()) + tau * float(np.log(exp_logits.sum()))

    # dM/da[j] = T1[j] + T2[j]
    # T1[j] = conv(rev_a, p)[n-1+j]
    # T2[j] = conv(a, p)[j]
    # Both convolutions have length 3n-2

    rev_a = a[::-1]
    M_fft = 1 << (3 * n - 2).bit_length()
    A_fft = np.fft.rfft(a, M_fft)
    PA_fft = np.fft.rfft(p, M_fft)
    conv_a_p = np.fft.irfft(A_fft * PA_fft, M_fft)[:3 * n - 2]

    RA_fft = np.fft.rfft(rev_a, M_fft)
    conv_ra_p = np.fft.irfft(RA_fft * PA_fft, M_fft)[:3 * n - 2]

    T2 = conv_a_p[:n].copy()
    T1 = conv_ra_p[n - 1:2 * n - 1].copy()

    dM_da = T1 + T2
    return dM_da, M


def objective_grad(a, tau):
    """
    Compute gradient of f(a) = 2n * LSE(conv(a,a), tau) / (sum(a))^2.
    """
    n = len(a)
    S = float(a.sum())
    if S < 1e-10:
        return np.zeros(n)
    dM_da, M = lse_grad_full(a, tau)
    # f = 2n*M/S^2, df/da = 2n*(dM*S - 2*M)/S^3
    grad = 2.0 * n * (dM_da * S - 2.0 * M) / (S ** 3)
    return grad


def adam_update(a, g, m, v, t, lr):
    """In-place Adam step with bias correction."""
    t1 = 0.9 ** t
    t2 = 0.999 ** t
    m[:] = 0.9 * m + 0.1 * g
    v[:] = 0.999 * v + 0.001 * g * g
    step = (m / (1.0 - t1)) / (np.sqrt(v / (1.0 - t2)) + 1e-8)
    a -= lr * step

def exact_topk_grad(a, c, topk):
    """Compute gradient using exact top-k peaks (for very fine polish)."""
    n = len(a)
    S = float(a.sum())
    k_act = min(topk, c.size)
    part = np.argpartition(c, -k_act)[-k_act:]
    vals = c[part].copy()
    order = np.argsort(vals)[::-1]
    part = part[order]
    vals = vals[order]

    # Use softmax over top-k
    temp = max(0.05 * vals[0], 1e-15)
    wts = np.exp((vals - vals[0]) / temp)
    wts = wts / wts.sum()
    M = float(np.dot(vals, wts))

    grad_c = np.zeros(n)
    for i in range(k_act):
        wi = float(wts[i])
        if wi <= 1e-15:
            continue
        ki = int(part[i])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        grad_c[js:je + 1] += wi * 2.0 * a[js:je + 1][::-1]

    grad = 2.0 * n * (grad_c * S - 2.0 * M) / (S ** 3)
    return grad

        s0 = compute_score(n, c0, float(a0.sum()))
        print(f"Parent score: {s0:.6f}")
        if s0 < best_score and np.isfinite(s0):
            best_score = s0
            best_a = a0.copy()

        sc, aa = optimize_run(a0, p1_end, rng,
                              init_lr=0.03, init_tau=0.5, min_tau=0.05,
                              patience=150, tau_decay=0.9999)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 1: {best_score:.6f}, t={time.monotonic()-start:.1f}s")
    else:
        # Fallback: try structured initializations
        x = np.linspace(0, 1, n)
        inits = [
            np.ones(n),
            1.0 + 0.5 * np.cos(2.0 * np.pi * x),
            1.0 + 0.3 * np.cos(2.0 * np.pi * 3.0 * x),
            np.exp(-2.0 * x),
            np.clip(0.1 + 0.9 * rng.random(n), 0.01, 1000.0),
        ]
        per_time = (p1_end - start) / len(inits)
        for i, a0 in enumerate(inits):
            t_end = min(p1_end, start + per_time * (i + 1))
            if time.monotonic() >= p1_end:
                break
            sc, aa = optimize_run(a0, t_end, rng,
                                  init_lr=0.1, init_tau=1.0, min_tau=0.1,
                                  patience=50, tau_decay=0.9999)
            if sc < best_score:
                best_score = sc
                best_a = aa.copy()
            print(f"Phase 1 init {i}: {sc:.6f}")

    # Phase 2: Main LSE optimization with annealing (35% time)
    p2_end = start + 0.45 * (total - 5.0)
    if best_a is not None and time.monotonic() < p2_end:
        sc, aa = optimize_run(best_a, p2_end, rng,
                              init_lr=0.01, init_tau=0.3, min_tau=0.01,
                              patience=300, tau_decay=0.99995)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 2: {best_score:.6f}, t={time.monotonic()-start:.1f}s")

    # Phase 3: Finer LSE with smaller tau (25% time)
    p3_end = start + 0.70 * (total - 5.0)
    if best_a is not None and time.monotonic() < p3_end:
        sc, aa = optimize_run(best_a, p3_end, rng,
                              init_lr=0.004, init_tau=0.1, min_tau=0.005,
                              patience=500, tau_decay=0.99998)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 3: {best_score:.6f}, t={time.monotonic()-start:.1f}s")

    # Phase 4: Exact top-k gradient polish (20% time)
    p4_end = deadline - 2.0
    if best_a is not None and time.monotonic() < p4_end:
        sc, aa = optimize_exact(best_a, p4_end, rng,
                                init_lr=0.001, topk=32, patience=500)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 4: {best_score:.6f}, t={time.monotonic()-start:.1f}s")

    # Phase 5: Ultra-fine final polish
    if best_a is not None and time.monotonic() < deadline - 1.0:
        sc, aa = optimize_exact(best_a, deadline - 1.0, rng,
                                init_lr=5e-5, topk=64, patience=800)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"Phase 5: {best_score:.6f}, t={time.monotonic()-start:.1f}s")

    # Final output
    if best_a is None:
        best_a = np.ones(n) / n

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    verified = np.inf
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        pass
    print(f"FINAL: internal={best_score:.6f}, verified={verified:.6f}, n={n}, "
          f"time={time.monotonic()-start:.1f}s")
    return out
