```python
from scipy.optimize import linprog
from scipy.sparse import lil_matrix

def solve_lp(f, M, n, K=30):
    c_conv = np.convolve(f, f)
    top_idx = np.argpartition(c_conv, -K)[-K:]
    K_actual = len(top_idx)
    
    # Build sparse constraint matrix
    A = lil_matrix((K_actual, n))
    for i, k in enumerate(top_idx):
        m_min = max(0, k - n + 1)
        m_max = min(n - 1, k)
        for m in range(m_min, m_max + 1):
            A[i, m] = f[k - m]
    
    A = A.tocsr()
    
    res = linprog(-np.ones(n), A_ub=A, b_ub=np.full(K_actual, M), 
                  bounds=[(0, None)] * n, method='highs')
    
    if res.success:
        return res.x
    return None
```

If f has length n and I create g of length n/2 by g[i] = f[2i] + f[2i+1], then:
- sum(g) = sum(f) (approximately, depending on the downsampling)
- conv(g, g)[k] = sum_i g[i]*g[k-i] = sum_i (f[2i]+f[2i+1])*(f[2(k-i)]+f[2(k-i)+1])

This is not simply related to conv(f, f). The max of conv(g, g) could be larger or smaller than the max of conv(f, f).

The score for g is 2*(n/2)*max(conv(g,g))/(sum(g))^2 = n*max(conv(g,g))/(sum(g))^2.

Compared to f: 2*n*max(conv(f,f))/(sum(f))^2.

So the ratio is: [n*max(conv(g,g))/(sum(g))^2] / [2*n*max(conv(f,f))/(sum(f))^2] = max(conv(g,g)) / (2*max(conv(f,f))) * (sum(f)/sum(g))^2.

If sum(g) ≈ sum(f), the ratio is approximately max(conv(g,g)) / (2*max(conv(f,f))).

So the question is: is max(conv(g,g)) < 2*max(conv(f,f))? If the autocorrelation of f is "spread out" over 2n-1 lags, then averaging pairs might concentrate it, making max(conv(g,g)) larger. So downsampling probably doesn't help.

```python
"""
LP-based step function optimizer minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses iterative linear programming with adaptive constraint selection,
line search blending, and multi-phase perturbation strategy.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix, csr_matrix
import time


def _score(f):
    """Compute the objective score for a normalized sequence."""
    n = len(f)
    c = np.convolve(f, f)
    return 2.0 * n * c.max()


def _solve_lp(f, M, lags, n):
    """
    Solve the LP: max sum(b) s.t. b >= 0, (f*b)[k] <= M for k in lags.
    Uses scipy linprog with HiGHS and sparse constraints.
    """
    K = len(lags)
    if K == 0:
        return None

    # Build sparse constraint matrix A_ub of shape (K, n)
    # Row i corresponds to lag lags[i]:
    #   sum_m f[k-m] * b[m] <= M  for valid m
    A = lil_matrix((K, n))
    for i, k in enumerate(lags):
        m_min = max(0, k - n + 1)
        m_max = min(n - 1, k)
        # A[i, m] = f[k - m] for m in [m_min, m_max]
        for m in range(m_min, m_max + 1):
            A[i, m] = f[k - m]

    A = csr_matrix(A)

    res = linprog(
        -np.ones(n),
        A_ub=A,
        b_ub=np.full(K, M),
        bounds=[(0, None)] * n,
        method='highs',
        options={'presolve': True, 'time_limit': 30}
    )

    if res.success and res.x is not None:
        return np.maximum(res.x, 0.0)
    return None


def _solve_lp_reduced(f, M, lags, n, max_lags=50):
    """Solve LP with a reduced set of lags for speed."""
    if len(lags) > max_lags:
        c = np.convolve(f, f)
        # Keep the lags with highest c values
        lag_vals = c[lags]
        top_in_lags = np.argsort(lag_vals)[-max_lags:]
        lags = lags[top_in_lags]
    return _solve_lp(f, M, lags, n)


def _perturb(f, rng, mode=0):
    """Apply a perturbation to escape local optima."""
    n = len(f)
    j = np.arange(n, dtype=np.float64)
    f_try = f.copy()

    if mode == 0:
        # Exponential tilting
        alpha = rng.uniform(-2.0, 2.0)
        tilt = np.exp(alpha * (j / n - 0.5))
        f_try = f * tilt
    elif mode == 1:
        # Sinusoidal modulation
        freq = rng.integers(1, max(2, n // 3) + 1)
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.2, 0.8)
        f_try = f * (1.0 + amp * np.sin(2 * np.pi * freq * j / n + phase))
    elif mode == 2:
        # Bimodal injection
        c1, c2 = sorted(rng.uniform(0.1, 0.9, size=2))
        bw = rng.uniform(0.05, 0.2)
        boost = np.exp(-((j / n - c1) / bw) ** 2) + np.exp(-((j / n - c2) / bw) ** 2)
        f_try = f * (1.0 + 0.5 * boost)
    elif mode == 3:
        # Random noise
        f_try = f + rng.normal(0, 0.1 * f.mean(), size=n)
    elif mode == 4:
        # Power-law reshaping
        p = rng.uniform(0.3, 3.0)
        f_min = f.min() + 1e-10
        f_norm = (f - f.min()) / (f.max() - f.min() + 1e-10)
        f_try = f_norm ** p + 0.01
    else:
        # Mirror + noise
        f_try = f[::-1] + rng.normal(0, 0.05 * f.mean(), size=n)

    f_try = np.maximum(f_try, 0.0)
    s = f_try.sum()
    if s < 1e-10:
        return f.copy()
    return f_try / s


def _lp_optimize(f, n, deadline, rng, label=""):
    """
    Main LP-based optimization loop.
    Iteratively solves the LP to find descent directions.
    """
    f = f.copy()
    f = np.maximum(f, 0.0)
    f = f / f.sum()

    best_score = _score(f)
    best_f = f.copy()
    it = 0
    stall_count = 0
    max_stall = 15

    # Adaptive parameters
    threshold = 0.80  # fraction of M for constraint selection
    K_max = 300       # max number of LP constraints
    line_search_ts = [1.0, 0.75, 0.5, 0.3, 0.15, 0.05]

    while time.monotonic() < deadline:
        c = np.convolve(f, f)
        M = c.max()

        # Select lags: those with c[k] > threshold * M
        lags = np.where(c > threshold * M)[0]
        if len(lags) == 0:
            lags = np.array([np.argmax(c)])

        # Cap the number of lags
        if len(lags) > K_max:
            lag_vals = c[lags]
            top_idx = np.argsort(lag_vals)[-K_max:]
            lags = lags[top_idx]

        # Solve LP
        t0 = time.monotonic()
        b = _solve_lp(f, M, lags, n)
        t_lp = time.monotonic() - t0

        if b is None or b.sum() < 1e-15:
            # Try with fewer constraints
            b = _solve_lp_reduced(f, M, lags, n, max_lags=30)
            if b is None or b.sum() < 1e-15:
                break

        b = b / b.sum()

        # Line search for blending parameter
        best_t = 0.0
        best_local = best_score
        for t in line_search_ts:
            f_try = (1.0 - t) * f + t * b
            f_try = np.maximum(f_try, 0.0)
            s = f_try.sum()
            if s < 1e-10:
                continue
            f_try = f_try / s
            sc = _score(f_try)
            if sc < best_local:
                best_local = sc
                best_t = t

        if best_t == 0.0:
            stall_count += 1
            if stall_count >= max_stall:
                break
            # Try a direct replacement with the LP solution
            sc_b = _score(b)
            if sc_b < best_score:
                f = b.copy()
                best_score = sc_b
                best_f = f.copy()
                stall_count = 0
            continue

        # Accept the step
        f = (1.0 - best_t) * f + best_t * b
        f = np.maximum(f, 0.0)
        f = f / f.sum()

        score = _score(f)
        it += 1

        if score < best_score:
            best_score = score
            best_f = f.copy()
            stall_count = 0
            if it % 5 == 0 or best_score < 1.51:
                print(f"    {label} iter {it}: score={score:.7f} (K={len(lags)}, t={best_t:.2f}, lp_time={t_lp:.2f}s)")
        else:
            stall_count += 1
            if stall_count >= max_stall:
                break

        # Adapt threshold: if too many lags, increase threshold
        if len(lags) > K_max:
            threshold = min(0.95, threshold + 0.02)
        elif len(lags) < 10:
            threshold = max(0.5, threshold - 0.05)

    return best_f, best_score


def _gradient_polish(f, n, deadline, rng):
    """
    Gradient-based polish focusing on the top peaks.
    Uses mirror descent with adaptive step size.
    """
    f = f.copy()
    best_score = _score(f)
    best_f = f.copy()

    momentum = np.zeros(n)
    beta = 0.8
    K = 5
    it = 0

    while time.monotonic() < deadline:
        c = np.convolve(f, f)
        M = c.max()

        # Top K lags
        top_idx = np.argpartition(c, -K)[-K:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        top_vals = c[top_idx]

        # Weights (emphasize the top)
        gaps = np.maximum(top_vals[0] - top_vals, 1e-10)
        weights = 1.0 / gaps
        weights /= weights.sum()

        # Gradient
        grad = np.zeros(n)
        for k, w in zip(top_idx, weights):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                idx = k - np.arange(j_start, j_end + 1)
                grad[j_start:j_end + 1] += 2.0 * w * f[idx]

        # Mirror step
        momentum = grad + beta * momentum
        span = momentum.max() - momentum.min()
        if span < 1e-30:
            break
        eta = 0.5 / span
        log_f = np.log(f + 1e-300) - eta * momentum
        log_f -= log_f.max()
        f_new = np.exp(log_f)
        f_new = np.maximum(f_new, 0.0)
        f_new = f_new / f_new.sum()
        f = f_new

        it += 1
        if it % 100 == 0:
            score = _score(f)
            if score < best_score:
                best_score = score
                best_f = f.copy()

    score = _score(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()

    return best_f, best_score
```
