```python
import cvxpy as cp
import numpy as np

def lp_direction(f, M):
    """Find g that maximizes sum(g) subject to (f*g)_k <= M, g >= 0."""
    n = len(f)
    g = cp.Variable(n, non_negative=True)
    
    # Build constraints
    constraints = []
    for k in range(2*n - 1):
        # (f*g)_k = sum_i f_i * g_{k-i}
        # valid i: 0 <= i < n and 0 <= k-i < n
        # i ranges from max(0, k-n+1) to min(n-1, k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        # g indices: k-i, ranges from k-i_hi to k-i_lo
        g_lo = k - i_hi
        g_hi = k - i_lo
        # constraint: sum_{i=i_lo}^{i_hi} f[i] * g[k-i] <= M
        # = sum_{j=g_lo}^{g_hi} f[k-j] * g[j] <= M
        expr = sum(f[k-j] * g[j] for j in range(g_lo, g_hi + 1))
        constraints.append(expr <= M)
    
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    prob.solve()
    return g.value
```

def _normalize(v):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / v.size
    v /= s
    return v

def _fft_size(m):
    if m <= 2:
        return 2
    return 1 << (int(m) - 1).bit_length()

def _autocorr(f):
    n = f.size
    m = 2 * n - 1
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]

def _fast_score(f):
    f = _normalize(f)
    if f.size == 0:
        return np.inf
    c = _autocorr(f)
    if c.size == 0:
        return np.inf
    return float(2.0 * f.size * c.max())

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1)
    out = v - theta
    np.maximum(out, 0.0, out)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / n
    out /= s
    return out

def _lp_direction(f, c, K=50):
    """
    Solve LP: max sum(g) s.t. (f*g)_k <= max(c) for top-K active lags, g >= 0.
    Returns g (not normalized) or None if LP fails.
    """
    n = f.size
    M = float(c.max())
    if M <= 0:
        return f.copy()
    
    # Get top K lags
    if K >= c.size:
        top_lags = np.arange(c.size)
    else:
        top_lags = np.argpartition(c, -K)[-K:]
    
    # Build constraint matrix A_ub (K x n)
    # A_ub[k, j] = f[k-j] if 0 <= k-j < n
    A_ub = np.zeros((len(top_lags), n), dtype=np.float64)
    for idx, k in enumerate(top_lags):
        k = int(k)
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        for i in range(i_lo, i_hi + 1):
            A_ub[idx, k - i] = f[i]
    
    b_ub = np.full(len(top_lags), M)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, 
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0)
        if g.sum() > 1e-300:
            return g
    return None

def _coord_polish(f0, budget_s, rng):

    f = _normalize(f0)
    n = f.size
    if n < 2 or budget_s <= 0:
        c = _autocorr(f)
        return f, 2.0 * n * c.max(), c

    best_f = f.copy()
    best_score = 2.0 * n * maxc

    L = 2 * n - 1
    delta = np.zeros(L)
    total = np.zeros(L)
    no_improve = 0
    
    D = 8  # donors
    R = 8  # receivers
    
    while time.monotonic() < deadline:
        maxc = float(c.max())
        
        # Find top-lag weighted gradient
        activeK = min(L, max(30, n // 10))
        active = np.argpartition(c, L - activeK)[L - activeK:]
        p = np.zeros(L)
        p[active] = 1.0 / activeK
        
        # Compute weighted gradient
        m = L + n - 1
        size = _fft_size(m)
        P = np.fft.rfft(p, size)
        F_rev = np.fft.rfft(f[::-1], size)
        conv = np.fft.irfft(P * F_rev, size)
        g = 2.0 * conv[n-1:n+n-1]
        
        gm = g - g.mean()
        
        # Select donors (high g, high f) and receivers (low g)
        combined = g * (f + 1e-15)
        donor_candidates = np.argsort(combined)[::-1][:D]
        recv_candidates = np.argsort(g)[:R]
        
        donors = [int(i) for i in donor_candidates if f[i] > 1e-12]
        recv_set = set(int(i) for i in recv_candidates)
        receivers = [int(i) for i in recv_candidates if i not in set(donors)]
        
        if not donors or not receivers:
            no_improve += 1
            if no_improve >= 15:
                break
            continue
        
        best_mc = maxc
        best_move = None
        
        # Try deterministic transfers
        fracs = (0.01, 0.05, 0.1, 0.2, 0.5)
        for i in donors:
            fi = f[i]
            if fi <= 1e-12:
                continue
            for j in receivers:
                if i == j:
                    continue
                for frac in fracs:
                    dv = fi * frac
                    if dv <= 0:
                        continue
                    # Compute delta
                    delta.fill(0.0)
                    delta[i:i+n] -= f
                    delta[j:j+n] += f
                    delta *= 2.0 * dv
                    q = dv * dv
                    delta[2*j] += q
                    delta[2*i] += q
                    delta[i+j] -= 2.0 * q
                    total = c + delta
                    mc = float(total.max())
                    if mc < best_mc:

def _perturb(f, rng, n):
    """Generate a perturbed version of f."""
    f = _normalize(f)
    r = rng.random()
    if r < 0.4:
        # Multiplicative log-normal
        sigma = float(rng.uniform(0.05, 0.8))
        f = f * np.exp(sigma * rng.standard_normal(n))
    elif r < 0.7:
        # Mix with random
        alpha = float(rng.uniform(0.1, 0.5))
        noise = rng.random(n)
        noise /= noise.sum()
        f = (1 - alpha) * f + alpha * noise
    else:
        # Segment shuffle
        nseg = int(rng.integers(4, min(20, n // 100 + 1)))
        seg_len = n // nseg
        idx = np.zeros(n, dtype=int)
        for s in range(nseg):
            lo = s * seg_len
            hi = lo + seg_len if s < nseg - 1 else n
            perm = lo + rng.permutation(hi - lo)
            idx[lo:hi] = perm
        f = f[idx]
    return _normalize(f)

    f = np.exp(0.5 * u)
    return _normalize(f)

"""
Step-function optimizer using LP-based active set direction finding.
Combines active-set linear programming for strong descent directions,
FFT-based fast convolution, coordinate descent polish, and multi-scale
restarts to minimize 2*n*max(a*a) / (sum(a))^2.
"""

import numpy as np
import time
import math
from scipy.optimize import linprog

TARGET_SCORE = 1.5027


def _normalize(v):
    """Project to non-negative vector of sum 1."""
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / v.size
    v /= s
    return v


def _fft_size(m):
    """Smallest power of 2 >= m."""
    if m <= 2:
        return 2
    return 1 << (int(m) - 1).bit_length()


def _autocorr(f):
    """Full linear convolution f*f via FFT."""
    n = f.size
    m = 2 * n - 1
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]


def _fast_score(f):
    """Fast score for normalized non-negative vector."""
    f = _normalize(f)
    if f.size == 0:
        return np.inf
    c = _autocorr(f)
    if c.size == 0:
        return np.inf
    return float(2.0 * f.size * c.max())


def _project_simplex(v):
    """Euclidean projection onto x >= 0, sum x = 1."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1)
    out = v - theta
    np.maximum(out, 0.0, out)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / n
    out /= s
    return out


def _lp_direction(f, c, K=60):
    """
    Solve LP: max sum(g) s.t. (f*g)_k <= max(c) for top-K active lags, g >= 0.
    Returns normalized g or None.
    """
    n = f.size
    M = float(c.max())
    if M <= 1e-15:
        return _normalize(f)

    L = c.size
    if K >= L:
        top_lags = np.arange(L)
    else:
        top_lags = np.argpartition(c, -K)[-K:]

    ncons = len(top_lags)
    A_ub = np.zeros((ncons, n), dtype=np.float64)

    for idx, k in enumerate(top_lags):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_end >= j_start:
            A_ub[idx, j_start:j_end + 1] = f[k - j_end:k - j_start + 1][::-1]

    b_ub = np.full(ncons, M)
    c_obj = -np.ones(n)

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs')
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        gs = float(g.sum())
        if gs > 1e-15:
            return g / gs
    return None


def _softmax_gradient(f, c, temperature):
    """Compute softmax-weighted gradient for descent direction."""
    n = f.size
    L = c.size
    maxc = float(c.max())
    T = maxc * temperature + 1e-300
    z = (c - maxc) / T
    np.clip(z, -40.0, 0.0, out=z)
    p = np.exp(z)
    ps = float(p.sum())
    if ps < 1e-300:
        p.fill(1.0 / L)
    else:
        p /= ps

    m = L + n - 1
    size = _fft_size(m)
    P = np.fft.rfft(p, size)
    F_rev = np.fft.rfft(f[::-1], size)
    conv = np.fft.irfft(P * F_rev, size)
    g = 2.0 * conv[n - 1:n + n - 1]
    return g - g.mean()


def _coord_polish(f0, c0, budget_s, rng):
    """Exact coordinate descent by 2-point mass transfers."""
    f = _normalize(f0)
    n = f.size
    if n < 2 or budget_s <= 0:
        c = _autocorr(f)
        return f, 2.0 * n * float(c.max()), c

    try:
        c = np.asarray(c0, dtype=np.float64).copy()
    except Exception:
        c = _autocorr(f)
    L = 2 * n - 1
    if c.size != L or not np.all(np.isfinite(c)):
        c = _autocorr(f)

    maxc = float(c.max())
    best_f = f.copy()
    best_score = 2.0 * n * maxc

    deadline = time.monotonic() + budget_s
    delta = np.zeros(L)
    total = np.zeros(L)
    no_improve = 0
    move_count = 0

    D = 8
    R = 8
    activeK = min(L, max(30, n // 8))
    fracs = (0.01, 0.05, 0.1, 0.2, 0.5)

    while time.monotonic() < deadline and no_improve < 20:
        maxc = float(c.max())

        active = np.argpartition(c, L - activeK)[L - activeK:]
        p = np.zeros(L)
        p[active] = 1.0 / activeK

        m = L + n - 1
        size = _fft_size(m)
        P = np.fft.rfft(p, size)
        F_rev = np.fft.rfft(f[::-1], size)
        conv = np.fft.irfft(P * F_rev, size)
        g = 2.0 * conv[n - 1:n + n - 1]

        combined = g * (f + 1e-15)
        donor_order = np.argsort(combined)[::-1]
        recv_order = np.argsort(g)

        donors = []
        for idx in donor_order:
            idx = int(idx)
            if f[idx] > 1e-12:
                donors.append(idx)
                if len(donors) >= D:
                    break

        donor_set = set(donors)
        receivers = []
        for idx in recv_order:
            idx = int(idx)
            if idx not in donor_set:
                receivers.append(idx)
                if len(receivers) >= R:
                    break

        if not donors or not receivers:
            no_improve += 1
            if no_improve >= 10:
                break
            continue

        best_mc = maxc
        best_move = None

        for i in donors:
            fi = float(f[i])
            if fi <= 1e-12:
                continue
            for j in receivers:
                if i == j:
                    continue
                for frac in fracs:
                    dv = fi * frac
                    if dv <= 0:
                        continue
                    delta.fill(0.0)
                    delta[i:i + n] -= f
                    delta[j:j + n] += f
                    delta *= 2.0 * dv
                    q = dv * dv
                    delta[2 * j] += q
                    delta[2 * i] += q
                    delta[i + j] -= 2.0 * q
                    np.add(c, delta, out=total)
                    mc = float(total.max())
                    if mc < best_mc:
                        best_mc = mc
                        best_move = (i, j, dv, delta.copy())

        for _ in range(20):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j or f[i] <= 1e-12:
                continue
            frac = float(rng.uniform(0.01, 0.35))
            dv = f[i] * frac
            delta.fill(0.0)
            delta[i:i + n] -= f
            delta[j:j + n] += f
            delta *= 2.0 * dv
            q = dv * dv
            delta[2 * j] += q
            delta[2 * i]
