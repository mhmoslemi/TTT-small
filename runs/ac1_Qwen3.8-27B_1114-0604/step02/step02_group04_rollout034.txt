"""
Low-peak step-function optimization via L-BFGS with adaptive soft-max surrogate.

Algorithm:
1. Generate diverse initializations (sinusoidal, multi-frequency, Gaussian-smoothed random).
2. For each initialization, run L-BFGS on the soft-max surrogate objective with
   decreasing temperature (annealing schedule).
3. Final polish with top-K active-set coordinate descent.
4. Return the best sequence found.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy.fft import rfft, irfft


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear auto-convolution via FFT."""
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = rfft(a, N)
    c = irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    """Compute the evaluation score (scale-invariant)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    return 2.0 * a.size * float(np.max(c)) / (s * s)


def _soft_max_obj_and_grad(x, T):
    """
    Compute the soft-max objective and its gradient w.r.t. x.
    a = softmax(x), objective = T * log(sum_k exp(c_k/T)) where c = convolve(a,a).
    """
    n = len(x)
    # a = softmax(x)
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    a = exp_x / exp_x.sum()
    
    # c = convolve(a, a)
    N = _next_pow2(2 * n - 1)
    fa = rfft(a, N)
    c_full = irfft(fa * fa, N)
    c = c_full[:2 * n - 1]
    
    # Soft-max value
    c_max = c.max()
    log_sum_exp = c_max + np.log(np.sum(np.exp((c - c_max) / T)) / c.size)
    obj = T * log_sum_exp
    
    # Soft-max weights
    w = np.exp((c - c_max) / T)
    w = w / w.sum()
    
    # grad_a[u] = 2 * sum_v a[v] * w[v+u] for u=0,...,n-1
    # Using FFT cross-correlation
    N2 = _next_pow2(3 * n - 2)
    fa2 = rfft(a, N2)
    fw2 = rfft(w, N2)
    cross = irfft(np.conj(fa2) * fw2, N2)
    grad_a = 2.0 * cross[:n]
    
    # grad_x = a * (grad_a - dot(grad_a, a))
    dot = float(np.dot(grad_a, a))
    grad_x = a * (grad_a - dot)
    
    return obj, grad_x


def _make_sine_init(n, amp, freq=1, phase=0.0):
    """Create a sinusoidal initialization."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.sin(2.0 * np.pi * freq * (i + phase) / n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0:
        a = a / s
    return a


def _make_multi_sine_init(n, components):
    """Create a multi-frequency sinusoidal initialization.
    components: list of (amp, freq, phase)
    """
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64) / n
    for amp, freq, phase in components:
        a += amp * np.sin(2.0 * np.pi * freq * (i + phase) / n) / n
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def _make_gaussian_init(n, rng):
    """Create a random initialization smoothed with a Gaussian kernel."""
    a = rng.standard_normal(n)
    # Smooth with a simple moving average
    kernel_size = max(3, n // 64)
    kernel = np.ones(kernel_size) / kernel_size
    a = np.convolve(a, kernel, mode='same')
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n
    return a


def _optimize_l_bfgs(a_init, deadline, T_schedule=None, max_iter=500):
    """
    Run L-BFGS optimization with adaptive temperature.
    a_init: initial sequence (will be normalized to sum=1)
    Returns: (best_a, best_score)
    """
    n = a_init.size
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_score = _score(best_a)
    
    if T_schedule is None:
        # Default schedule: start smooth, end sharp
        c = _fft_convolve(a)
        c_max = c.max()
        T_schedule = [c_max / 5, c_max / 10, c_max / 20, c_max / 50, c_max / 100, c_max / 200]
    
    for T in T_schedule:
        if time.monotonic() > deadline:
            break
        
        # Convert a to x (log-softmax parameters)
        # a = softmax(x) → x = log(a) (up to a constant)
        # We'll use x = log(a + eps) as the parameterization
        eps = 1e-30
        x0 = np.log(a + eps)
        
        # Define objective and gradient for scipy
        def func(x):
            obj, _ = _soft_max_obj_and_grad(x, T)
            return obj
        
        def grad(x):
            _, g = _soft_max_obj_and_grad(x, T)
            return g
        
        # Run L-BFGS
        result = minimize(
            func, x0, jac=grad, method='L-BFGS-B',
            options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8, 'disp': False}
        )
        
        # Extract result
        x_opt = result.x
        a_opt = np.exp(x_opt - x_opt.max())
        a_opt = a_opt / a_opt.sum()
        a_opt = np.clip(a_opt, 0, 1000)
        s_opt = a_opt.sum()
        if s_opt > 0.01:
            a_opt = a_opt / s_opt
        
        sc = _score(a_opt)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a_opt.copy()
        
        # Update a for next T
        a = best_a.copy()
        time.sleep(0)  # yield
    
    return best_a, best_score


def _polish_active_set(a, deadline, K=8):
    """
    Active-set coordinate descent polish.
    Identify top-K peaks and reduce them via targeted adjustments.
    """
    n = a.size
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(best_a)
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        c = _fft_convolve(a)
        
        # Find top-K lags
        if c.size <= K:
            top_idx = np.arange(c.size)
        else:
            top_idx = np.argpartition(c, -K)[-K:]
        
        # For each top lag, find which positions of a contribute most
        # and try to reduce them
        improved = False
        for k in top_idx:
            # c[k] = sum_i a[i] * a[k-i] (with valid indices)
            # To reduce c[k], we want to decrease a[i] for i where a[k-i] is large
            # Simple approach: reduce a[i] proportionally to a[k-i] for valid i
            
            # Find valid range
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            
            if i_min > i_max:
                continue
            
            # Contribution of each a[i] to c[k] is a[k-i] (for i in valid range)
            contrib = np.zeros(n)
            for i in range(i_min, i_max + 1):
                contrib[i] = a[k - i]
            
            # Try reducing the top contributors
            top_contrib = np.argsort(contrib)[-K:]
            for i in top_contrib:
                if a[i] <= 0:
                    continue
                # Try a small reduction
                delta = a[i] * 0.01
                a_trial = a.copy()
                a_trial[i] -= delta
                if a_trial[i] < 0:
                    a_trial[i] = a[i] * 0.99
                    delta = a[i] - a_trial[i]
                # Renormalize
                a_trial = a_trial / a_trial.sum()
                sc = _score(a_trial)
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
        
        if not improved and it > 50:
            # Try a different approach: add to positions that don't contribute to peaks
            c = _fft_convolve(a)
            # Find positions with low total contribution to peaks
            total_contrib = np.zeros(n)
            for k in top_idx:
                i_min = max(0, k - (n - 1))
                i_max = min(n - 1, k)
                for i in range(i_min, i_max + 1):
                    total_contrib[i] += a[k - i]
            
            # Add to positions with lowest contribution
            low_contrib = np.argsort(total_contrib)[:K]
            for i in low_contrib:
                a_trial = a.copy()
                a_trial[i] += a.mean() * 0.01
                a_trial = a_trial / a_trial.sum()
                sc = _score(a_trial)
                if sc < best_score:
                    best_score = sc
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    improved = True
        
        if not improved:
            # Reset with small perturbation
            rng = np.random.RandomState(it)
            a = best_a.copy()
            noise = rng.standard_normal(n) * 0.001 * a.mean()
            a = a + noise
            a = np.clip(a, 0, 1000)
            a = a / a.sum()
    
    return best_a, best_score
