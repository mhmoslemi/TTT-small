"""
Low-peak step-function optimization using polynomial ramp initializations.

Key insight: Polynomial ramps a[i]=(i+1)^p/sum have naturally low center
convolution peaks (anti-palindromic structure), giving much better starting
points than sinusoidal sequences. Combined with mirror descent (multiplicative
weights) using FFT-accelerated gradients and active-set polishing.

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    """Compute evaluation score (scale-invariant)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _grad_conv(a, w):
    """Gradient of sum_j w_j*(a*a)[j] w.r.t. a using FFT cross-correlation.
    w has length 2n-1 (the convolution length)."""
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]


def _make_ramp(n, p):
    """Polynomial ramp: a[i] = (i+1)^p normalized to sum 1."""
    i = np.arange(n, dtype=np.float64) + 1.0
    a = i ** p
    a = np.clip(a, 0.0, 1e30)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def _make_ramp_modulated(n, p, amp, freq, phase=0.0):
    """Ramp modulated by sinusoid for additional variety."""
    i = np.arange(n, dtype=np.float64) + 1.0
    base = i ** p
    theta = 2.0 * np.pi * freq * (np.arange(n, dtype=np.float64) / n + phase)
    a = base * (1.0 + amp * np.sin(theta))
    a = np.clip(a, 0.0, 1e30)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = _make_ramp(n, p)
    return a


def _make_symmetric_ramp(n, p):
    """Symmetric concentration: high in middle, low at ends."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = (1.0 + p * (1.0 - np.abs(i - mid) / mid)) ** 2
    a = np.clip(a, 0.0, 1e30)
    s = a.sum()
    if s > 0.01:
        a /= s
    return a


def _softmax_weights(c, T):
    """Compute normalized softmax weights on convolution values."""
    c_max = float(np.max(c))
    c_shift = (c - c_max) / T
    e = np.exp(c_shift)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros(c.size)
        w[np.argmax(c)] = 1.0
        return w
    return e / s


def _optimize_mirror(a_init, deadline, T_start=None, T_end=None, lr0=None,
                     patience=8000, eval_every=5, seed=42):
    """Mirror descent (multiplicative weights) with soft-max target.
    Maintains sum(a)=1 by normalization after each step."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size) / a.size
    a = a / s

    n = a.size
    if T_start is None:
        T_start = 0.01
    if T_end is None:
        T_end = 0.001
    if lr0 is None:
        lr0 = 0.3 * n

    best_a = a.copy()
    best_score = 2.0 * n * float(np.max(_fft_convolve(a)))
    rng = np.random.RandomState(seed)

    it = 0
    last_improve = 0
    while time.monotonic() < deadline:
        it += 1

        # Annealing
        frac = min(1.0, it / 80000.0)
        T = T_start + (T_end - T_start) * frac
        lr = lr0 / (1.0 + 0.005 * it)

        # Convolution
        c = _fft_convolve(a)
        c_max = float(np.max(c))
        if c_max < 1e-15:
            break

        # Temperature in absolute units
        T_abs = T * c_max

        # Soft-max weights
        w = _softmax_weights(c, T_abs)

        # Gradient
        grad = _grad_conv(a, w)

        # Tangent projection (maintain sum in log-space)
        grad_proj = grad - np.mean(grad)

        # Multiplicative update
        update = -lr * grad_proj
        update = np.clip(update, -20.0, 20.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        # Evaluate periodically
        if it % eval_every == 0:
            c_check = _fft_convolve(a)
            sc = 2.0 * n * float(np.max(c_check))
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                # Restart with perturbation from best
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.02
                a = a + noise * a.mean()
                a = np.clip(a, 0.0, 1000.0)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                patience = max(3000, int(patience * 1.5))
                lr0 = max(0.1 * n, lr0 * 0.7)

    return best_a, best_score


def _optimize_active_set(a_init, deadline, K=8, lr0=None, seed=7):
    """Active-set refinement: focus gradient on top-K peaks only."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size) / a.size
    a = a / s

    n = a.size
    if lr0 is None:
        lr0 = 0.5 * n

    best_a = a.copy()
    best_score = 2.0 * n * float(np.max(_fft_convolve(a)))
    rng = np.random.RandomState(seed)

    it = 0
    last_improve = 0
    patience = 5000
    while time.monotonic() < deadline:
        it += 1
        lr = lr0 / (1.0 + 0.003 * it)

        c = _fft_convolve(a)
        c_max = float(np.max(c))
        if c_max < 1e-15:
            break

        # Top-K peaks
        if c.size <= K:
            top_idx = np.arange(c.size)
        else:
            top_idx = np.argpartition(c, -K)[-K:]

        # Soft weights within top-K
        top_vals = c[top_idx]
        T_local = 2.0 * c_max
        top_shift = (top_vals - top_vals.max()) / T_local
        weights = np.exp(top_shift)
        weights /= weights.sum()

        # Build full weight vector
        w = np.zeros(c.size, dtype=np.float64)
        for i, k in enumerate(top_idx):
            w[k] = weights[i]

        # Gradient
        grad = _grad_conv(a, w)
        grad_proj = grad - np.mean(grad)

        # Update
        update = -lr * grad_proj
        update = np.clip(update, -15.0, 15.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        if it % 3 == 0:
            sc = 2.0 * n * float(np.max(_fft_convolve(a)))
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.01
                a = a + noise * a.mean()
                a = np.clip(a, 0.0, 1000.0)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                patience = max(2000, int(patience * 1.4))
                lr0 = max(0.2 * n, lr0 * 0.75)

    return best_a, best_score


def _optimize_adam(a_init, deadline, lr0=0.01, seed=99):
    """Adam optimizer with soft-max objective for additional diversity."""
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    a = a / a.sum()
    n = a.size

    best_a = a.copy()
    best_score = 2.0 * n * float(np.max(_fft_convolve(a)))

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t = 0
    sigma_frac = 0.02  # sigma as fraction of max c

    while time.monotonic() < deadline:
        t += 1
        c = _fft_convolve(a)
        c_max = float(np.max(c))
        if c_max < 1e-15:
            break
        sigma = sigma_frac * c_max

        # Soft-max
        c_shift = (c - c_max) / sigma
        w = np.exp(c_shift)
        ws = w.sum()
        if ws > 1e-300:
            w /= ws
        else:
            w = np.zeros(c.size)
            w[np.argmax(c)] = 1.0

        grad = _grad_conv(a, w)

        # Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        lr = lr0 / (1 + 0.0005 * t)
        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s > 1e-10:
            a_new = a_new / s
            a = a_new

        if t % 10 == 0:
            sc = 2.0 * n * float(np.max(_fft_convolve(a)))
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score

                best_score = s_polish2
                best_a = a_polish2.copy()
                print(f"  Polished K=12: {s_polish2:.8f}", flush=True)

    # Phase 4: Adam optimization for diversity
    d4 = hard_deadline - 30.0
    remaining4 = d4 - time.monotonic()
    if remaining4 > 60:
        print(f"\n[Phase 4] Adam optimization (budget={remaining4:.0f}s)...", flush=True)
        a_adam, s_adam = _optimize_adam(best_a, d4, lr0=0.005, seed=300)
        if s_adam < best_score:
            best_score = s_adam
            best_a = a_adam.copy()
            print(f"  Adam: {s_adam:.8f}", flush=True)
        else:
            # Try Adam from best ramp init
            a_adam2, s_adam2 = _optimize_adam(results[0][1], d4, lr0=0.01, seed=301)
            if s_adam2 < best_score:
                best_score = s_adam2
                best_a = a_adam2.copy()
                print(f"  Adam from init: {s_adam2:.8f}", flush=True)

    # Phase 5: Final validation
    print(f"\n[Phase 5] Validating...", flush=True)
    arr_list = [float(max(0.0, min(1000.0, x))) for x in best_a]
    try:
        actual = float(evaluate_sequence(arr_list))
    except Exception as e:
        print(f"  Error in evaluation: {e}", flush=True)
        actual = np.inf

    # Compare with a few fallbacks
    fallback_candidates = [(actual, arr_list, "best")]
    for sc, a_cand, name in results[:2]:
        fb_list = [float(max(0.0, min(1000.0, x))) for x in a_cand]
        try:
            fb_actual = float(evaluate_sequence(fb_list))
        except Exception:
            fb_actual = np.inf
        fallback_candidates.append((fb_actual, fb_list, name))
        if np.isfinite(fb_actual):
            print(f"  Fallback {name}: {fb_actual:.8f}", flush=True)

    fallback_candidates.sort(key=lambda x: x[0])
    best_out = fallback_candidates[0][1]
    best_actual = fallback_candidates[0][0]
    print(f"\n[Done] Best score: {best_actual:.8f} ({fallback_candidates[0][2]})", flush=True)
    print(f"[Done] Time: {time.monotonic()-t0:.1f}s, Length: {len(best_out)}", flush=True)

    return best_out
