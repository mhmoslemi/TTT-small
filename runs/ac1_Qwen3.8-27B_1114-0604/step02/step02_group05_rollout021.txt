"""
Multi-strategy optimization for non-negative sequences minimizing
2*n*max(conv(a,a)) / sum(a)^2.

Key innovations over previous approach:
1. Multi-scale optimization (small to large n)
2. Exact subgradient with backtracking line search
3. Top-k averaging for robust subgradient
4. Combined with frequency-domain flattening
5. Adaptive restart strategy
"""

import numpy as np
import time

def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(a, b, L=None):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)

def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())

def _gradient_exact(p, top_k=5):
    """
    Compute the subgradient of max(c) using top-k averaging.
    Returns (score, gradient, max_c)
    """
    n = len(p)
    c = _conv_full(p, p)
    max_c = c.max()
    
    # Find top_k indices
    if top_k >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    
    # Weight by proximity to max (exponential)
    vals = c[top_idx]
    weights = np.exp(-(max_c - vals) * n / max_c)  # sharpness depends on scale
    weights = weights / weights.sum()
    
    # Compute gradient
    # dc[k]/da[m] = a[m+k]*1_{m+k<n} + a[m-k]*1_{m>=k} for k>0
    # dc[0]/da[m] = 2*a[m]
    grad = np.zeros(n)
    for w, k in zip(weights, top_idx):
        k = int(k)
        for m in range(n):
            if k == 0:
                grad[m] += w * 2.0 * p[m]
            else:
                if m + k < n:
                    grad[m] += w * p[m + k]
                if m >= k:
                    grad[m] += w * p[m - k]
    
    return max_c, grad, c

def _gradient_fft(p, top_k=10):
    """
    Compute subgradient using FFT for efficiency.
    """
    n = len(p)
    c = _conv_full(p, p)
    max_c = c.max()
    
    # Find top_k
    if top_k >= len(c):
        top_idx = np.arange(len(c))
    else:
        top_idx = np.argpartition(c, -top_k)[-top_k:]
    
    # Weight
    vals = c[top_idx]
    sharp = n / (max_c + 1e-10)
    weights = np.exp(-(max_c - vals) * sharp)
    s = weights.sum()
    if s < 1e-300:
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        weights / s
    
    # Build the "weighted autocorrelation vector"
    # grad[m] = sum_k w_k * (p[m+k]*1_{m+k<n} + p[m-k]*1_{m>=k})
    # This is a cross-correlation between p and the weighted impulse train
    # Let's build a vector v of length 2n-1: v[k] = w_k for k in top_idx
    # Then grad = cross_correlation(p, v) appropriately
    # Actually, grad[m] = sum_k w_k * p[m+k] + sum_k w_k * p[m-k]
    # = (p * v_reversed)[m] + (p * v)[m] in some sense
    
    # Simpler: just loop over top_k (which is small, like 10)
    grad = np.zeros(n)
    for wi, ki in zip(weights, top_idx):
        k = int(ki)
        if k == 0:
            grad += wi * 2.0 * p
        else:
            # p[m+k] for m = 0..n-1-k
            grad[:n-k] += wi * p[k:]
            # p[m-k] for m = k..n-1
            grad[k:] += wi * p[:n-k]
    
    return max_c, grad, c

def _line_search_subgrad(p, grad, max_c, n):
    """
    Backtracking line search for subgradient step.
    F(a) = 2n * max(conv(a,a)) / sum(a)^2
    """
    S = p.sum()
    F = 2.0 * n * max_c / (S * S)
    
    # Try several step sizes
    best_p = p.copy()
    best_F = F
    alpha = 1.0
    for _ in range(20):
        p_try = p - alpha * grad
        p_try = np.maximum(p_try, 0.0)
        s_try = p_try.sum()
        if s_try < 1e-10:
            alpha *= 0.5
            continue
        p_try = p_try / s_try
        c_try = _conv_full(p_try, p_try)
        F_try = 2.0 * n * c_try.max() / (p_try.sum() ** 2)
        # Wait, after normalization, sum = 1, so F_try = 2n * c_try.max()
        F_try = 2.0 * n * c_try.max()
        if F_try < best_F - 1e-12:
            best_p = p_try
            best_F = F_try
            break
        alpha *= 0.7
    
    return best_p, best_F

def _optimize_subgrad(p0, rng, time_budget, top_k=5):
    """Subgradient method with line search and projection."""
    p = np.asarray(p0, dtype=float).copy()
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return None, float('inf')
    p = p / s
    n = len(p)
    
    best = p.copy()
    best_score = _score(p)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0
    it = 0
    
    while time.monotonic() < deadline:
        max_c, grad, c = _gradient_fft(p, top_k)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Line search
        p_new, F_new = _line_search_subgrad(p, grad, max_c, n)
        
        if F_new < cur_score - 1e-12:
            p = p_new
            cur_score = F_new
            stall = 0
            if cur_score < best_score - 1e-12:
                best = p.copy()
                best_score = cur_score
        else:
            # Try smaller steps
            p_small = p - 0.01 * gnorm * grad * (cur_score / (2*n*max_c))  # rough Polyak step
            p_small = np.maximum(p_small, 0)
            s = p_small.sum()
            if s > 1e-10:
                p_small = p_small / s
                sc = _score(p_small)
                if sc < cur_score - 1e-12:
                    p = p_small
                    cur_score = sc
                    stall = 0
                    if cur_score < best_score - 1e-12:
                        best = p.copy()
                        best_score = cur_score
                else:
                    stall += 1
            else:
                stall += 1
        
        # Periodic restart if stuck
        if stall > 20:
            p = best.copy()
            cur_score = best_score
            stall = 0
            # Perturb
            p = p * np.exp(rng.normal(0, 0.01, n))
            p = np.maximum(p, 0)
            s = p.sum()
            if s > 1e-10:
                p = p / s
            else:
                p = best.copy()
                cur_score = best_score
        
        it += 1
    
    return best, best_score
