"""
Hybrid optimizer: Adam gradient descent + LP-based refinement for minimizing
2n*max(conv(a,a))/(sum(a))^2. Uses exact subgradient, top-k active set LP,
and log-parameterization for stability.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, c, S):
    """Objective value given n, convolution array c, and sum S."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def adam_step(a, g, m, v, t, lr):
    """In-place Adam optimizer step (exponential decay schedule)."""
    bc1 = 0.9 ** t
    bc2 = 0.999 ** t
    m[:] = 0.9 * m + 0.1 * g
    v[:] = 0.999 * v + 0.001 * g * g
    a -= lr * (m / (1.0 - bc1)) / (np.sqrt(v / (1.0 - bc2)) + 1e-8)


def exact_subgradient(a, c, n):
    """
    Exact subgradient of 2n*max(c)/S^2 w.r.t. a.
    Returns gradient array of length n.
    """
    S = float(a.sum())
    M = float(np.max(c))
    k_star = int(np.argmax(c))
    
    # d/d a_j of (a*a)(k*) = a_{k*-j} for valid j
    js = max(0, k_star - n + 1)
    je = min(k_star, n - 1)
    
    grad = np.zeros(n)
    if js <= je:
        # (a*a)(k*) derivative w.r.t. a_j = a_{k*-j} for j in [js, je]
        # a_{k*-j} for j in [js, je] is a[k*-je : k*-js+1] reversed
        vals = a[k_star - je : k_star - js + 1]
        grad[js:je + 1] = vals
    
    # Full gradient: 2n * (S * d(max)/da - 2M) / S^3
    # = 2n/S^3 * (S * d(max)/da - 2M * 1)
    grad = (2.0 * n / (S ** 3)) * (S * grad - 2.0 * M)
    return grad

    # The values a_{ki-j} for j in [js, je] are a[kstar-je ... kstar-js]

        if wi < 1e-15:

        js = max(0, ki - n + 1)

        if js > je:
            continue
        # a_{ki-j} for j in [js..je] = a[ki-je .. ki-js]
        seg = a[ki - je : ki - js + 1]

    grad = (2.0 * n / (S ** 3)) * (S * grad - 2.0 * M)

def k_star_idx(je, ki):
    return ki - je


def optimize_adam(a0, end_time, rng, lr, topk, patience, use_exact=True):
    """Adam optimization loop with given parameters."""
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = float(a.sum())
    if S < 1e-10:
        a = np.ones(n)
        S = n
    c = fast_self_conv(a)
    score = compute_score(n, c, S)
    best_score = score
    best_a = a.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_imp = 0
    
    while time.monotonic() < end_time:
        c = fast_self_conv(a)
        S = float(a.sum())
        if S < 1e-10:
            break
        score = compute_score(n, c, S)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if no_imp >= patience:
            # Restart with perturbation
            scale = 0.03 + 0.05 * rng.random()
            a = best_a * (1.0 + scale * rng.standard_normal(n))
            a = np.clip(a, 0.0, 1000.0)
            if a.sum() < 1e-3:
                a += 0.1
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            no_imp = 0
            continue
        
        if use_exact and topk <= 2:
            g = exact_subgradient(a, c, n)
        else:
            temp = 0.05 * float(np.max(c))
            g = softmax_gradient(a, c, n, topk, temp)
        
        t += 1
        adam_step(a, g, m, v, t, lr)
        a = np.clip(a, 0.0, 1000.0)
    
    return best_score, best_a


def lp_refine_step(f, topk=50, max_iter=1):
    """
    One LP refinement step: solve max sum(g) s.t. g>=0, (f*g)(k)<=M for top-k k.
    Returns the LP solution g (or None if LP fails).
    """
    n = len(f)
    c = fast_self_conv(f)
    M = float(np.max(c))
    if M < 1e-30:
        return None
    
    # Get top-k positions
    k_act = min(topk, c.size)
    top_pos = np.argpartition(c, -k_act)[-k_act:]
    
    # Build constraint matrix A (k_act x n)
    # A[i, j] = f[pos_i - j] if 0 <= pos_i - j <= n-1
    A = np.zeros((k_act, n))
    for i, pos in enumerate(top_pos):
        js = max(0, int(pos) - n + 1)
        je = min(int(pos), n - 1)
        if js <= je:
            A[i, js:je+1] = f[int(pos) - je : int(pos) - js + 1]
    
    b = M * np.ones(k_act)
    c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                      method='highs', options={'time_limit': 30, 'presolve': True})
        if res.status == 0:
            return res.x
        else:
            return None
    except Exception:
        return None


def lp_refine_phase(a0, end_time, rng, topk=80, step_size=None):
    """
    LP-based refinement: iteratively solve LP and do convex combination.
    """
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    S = float(a.sum())
    c = fast_self_conv(a)
    score = compute_score(n, c, S)
    best_score = score
    best_a = a.copy()
    
    if step_size is None:
        step_size = 0.15
    
    stall = 0
    stall_limit = 20
    
    while time.monotonic() < end_time:
        c = fast_self_conv(a)
        S = float(a.sum())
        if S < 1e-10:
            break
        score = compute_score(n, c, S)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        if stall >= stall_limit:
            # Try different topk or restart
            topk_try = topk + int(rng.integers(0, 40))
            g = lp_refine_step(best_a, topk=topk_try)
            if g is None or g.sum() < 1e-10:
                # Perturb and restart
                scale = 0.05
                a = best_a * (1.0 + scale * rng.standard_normal(n))
                a = np.clip(a, 0.0, 1000.0)
                stall = 0
                continue
            # Rescale g to match sum of best_a
            g = g * (S / g.sum())
            # Convex combination
            t = min(step_size, 0.3)
            a = (1.0 - t) * best_a + t * g
            a = np.clip(a, 0.0, 1000.0)
            continue
        
        # Solve LP
        g = lp_refine_step(a, topk=topk)
        if g is None or g.sum() < 1e-10:
            stall += 10
            continue
        
        # Rescale g
        g_sum = float(g.sum())
        if g_sum < 1e-10:
            continue
        g = g * (S / g_sum)
        
        # Line search for step size
        best_t = 0.0
        best_sc = score
        for t in [0.3, 0.15, 0.08, 0.04, 0.02]:
            trial = (1.0 - t) * a + t * g
            trial = np.clip(trial, 0.0, 1000.0)
            trial_S = float(trial.sum())
            if trial_S < 1e-10:
                continue
            trial_c = fast_self_conv(trial)
            trial_sc = compute_score(n, trial_c, trial_S)
            if np.isfinite(trial_sc) and trial_sc < best_sc:
                best_sc = trial_sc
                best_t = t
        
        if best_t > 0:
            a = (1.0 - best_t) * a + best_t * g
            a = np.clip(a, 0.0, 1000.0)
    
    return best_score, best_a


def multiplicative_descent(a0, end_time, rng, lr=0.5, patience=100):
    """Multiplicative (mirror descent) optimizer in log-space."""
    n = a0.size
    # Work in log-space with offset to handle zeros
    eps = 1e-8
    z = np.log(a0 + eps)
    a = np.clip(a0.astype(np.float64), eps, 1000.0).copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_a = a.copy()
    c = fast_self_conv(a)
    S = float(a.sum())
    best_score = compute_score(n, c, S)
    no_imp = 0
    
    while time.monotonic() < end_time:
        c = fast_self_conv(a)
        S = float(a.sum())
        score = compute_score(n, c, S)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if no_imp >= patience:
            z = np.log(best_a + eps)
            a = best_a * (1.0 + 0.05 * rng.standard_normal(n))
            a = np.clip(a, eps, 1000.0)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            no_imp = 0
            continue
        
        # Compute gradient in log-space
        # a = exp(z), da/dz = a (element-wise, ignoring normalization)
        # dF/dz_j = a_j * dF/da_j
        g_a = exact_subgradient(a, c, n)
        g_z = a * g_a
        
        t += 1
        adam_step(z, g_z, m, v, t, lr)
        a = np.exp(z)
        a = np.clip(a, eps, 1000.0)
    
    return best_score, best_a


def structured_init(n, rng):
    """Generate diverse initial sequences."""
    inits = []
    x = np.linspace(0, 1, n)
    
    # Uniform
    inits.append(np.ones(n))
    
    # Cosine windows
    inits.append(1.0 + 0.5 * np.cos(2.0 * np.pi * x))
    inits.append(1.0 + 0.7 * np.cos(4.0 * np.pi * x))
    inits.append(1.0 + 0.3 * np.cos(6.0 * np.pi * x))
    
    # Ramp
    inits.append(np.clip(1.0 + 0.5 * (2.0 * x - 1.0), 0.01, None))
    
    # Gapped
    a_g = np.ones(n)
    a_g[1::2] = 0.4
    inits.append(a_g)
    
    # Two-scale: high at even, low at odd (with modulation)
    a_ts = np.ones(n)
    a_ts[0::2] = 1.0 + 0.3 * np.cos(2.0 * np.pi * np.arange(0, n, 2) / max(1, n // 4))
    a_ts[1::2] = 0.5
    inits.append(a_ts)
    
    # Random
    inits.append(np.clip(0.5 + 0.5 * rng.random(n), 0.01, None))
    
    # Bimodal
    a_bm = np.zeros(n)
    mid = n // 2
    a_bm[:mid//2] = 1.0
    a_bm[mid - mid//4:mid + mid//4] = 1.0
    a_bm[3*mid//2:] = 0.5
    inits.append(np.clip(a_bm, 0.01, None))
    
    return inits
