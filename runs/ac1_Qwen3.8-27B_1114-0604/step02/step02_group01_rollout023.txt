"""
Step function optimizer using spectral water-filling and log-sum-exp
smoothed gradient descent with adaptive temperature annealing.
"""

import numpy as np
import time

def _autocorr_fft(f):
    """Full linear convolution a*a via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]

def _score_fast(f):
    """Fast score computation."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-30:
        return np.ones(n) / n
    # Sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()

def _soft_max_grad(c, tau):
    """Compute softmax weights and gradient of soft-max for the conv."""
    # For numerical stability
    c_shifted = c - c.max()

def _gradient_soft_max(f, c, weights):
    """
    Compute gradient of sum_k weights[k] * c[k] w.r.t. f.
    c[k] = sum_j f[j]*f[k-j]
    dc[k]/df[i] = 2*f[k-i] (when 0 <= k-i < n)
    """
    n = len(f)
    grad = np.zeros(n)
    # For each lag k, the contribution to grad[i] is weights[k] * 2 * f[k-i]
    # where 0 <= k-i < n, i.e., i is in [k-n+1, k] ∩ [0, n-1]
    # This is a convolution-like operation
    # grad = 2 * sum_k weights[k] * f[shifted by k]
    # = 2 * convolve(weights, f) (with appropriate indexing)
    
    # Actually: grad[i] = 2 * sum_k weights[k] * f[k-i] (when k-i valid)
    # Let m = k-i, so k = i+m, and m ranges over valid indices
    # grad[i] = 2 * sum_m weights[i+m] * f[m] (when 0 <= i+m < 2n-1)
    # This is a cross-correlation of weights and f!
    
    # cross_corr[i] = sum_m f[m] * weights[i+m]
    # We can compute this via FFT
    
    # Actually, let's just do it directly for the significant weights
    # For large n, most weights are negligible if tau is small
    significant = np.nonzero(weights > 1e-10)[0]
    for k in significant:

        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = k - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * weights[k] * f[idx]
    return grad

Step function optimizer using Adam-like gradient descent with soft-max
objectives, spectral corrections, targeted coordinate moves, and

optimization strategies with careful exploration-exploitation balance.

import numpy as np
import time


def _autocorr_fft(f):
    """Full linear convolution a*a via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _score_fast(f):
    """Fast score computation without calling evaluate_sequence."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if not np.all(np.isfinite(v)):
        return np.ones(n) / n
    s = v.sum()
    if s < 1e-30:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    ws = w.sum()
    if ws < 1e-30:
        return np.ones(n) / n
    return w / ws


def _gradient_softmax(f, c, weights):
    """
    Compute gradient of sum_k weights[k] * c[k] w.r.t. f using FFT.
    grad[i] = 2 * sum_k weights[k] * f[k-i] (where valid)
    This is a cross-correlation of weights and f.
    """
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    W = np.fft.rfft(weights, size)
    # cross-correlation: corr[i] = sum_m f[m] * weights[i+m]
    # = ifft(conj(F) * W)[i] ... let me think more carefully
    # Actually: sum_k weights[k] * f[k-i] for k-i in [0,n-1]
    # Let's use the direct approach via convolution
    # grad[i] = 2 * (weights * f_flipped)[2n-2-i] ... no
    # 
    # Simplest: use the fact that grad = 2 * ifft(conj(fft(f)) .* fft(weights_padded))
    # But the indexing is tricky. Let me just do it via FFT of the valid range.
    
    # Alternative: pad f to size, compute element-wise
    # grad[i] = 2 * sum_{k: 0<=k-i<n} weights[k] * f[k-i]
    # Let m = k-i, so k = i+m, m in [0,n-1], i+m in [0, 2n-2]
    # grad[i] = 2 * sum_{m=0}^{n-1} f[m] * weights[i+m] (where i+m < 2n-1)
    # This is the cross-correlation of f and weights, evaluated at lag i.
    
    # Cross-correlation via FFT:
    # corr[i] = sum_m f[m] * weights[i+m]
    # = ifft( conj(fft(f)) * fft(weights) )[i] ... not quite
    # Actually: sum_m f[m] * w[m+i] = (f * w_flipped)[n-1 + i] ... 
    # Let's just use the direct FFT formula:
    # Define g = f (length n), h = weights (length 2n-1)
    # We want: out[i] = sum_m g[m] * h[i+m] for i = 0..n-1
    # This is a "deconvolution-like" cross-correlation
    # 
    # Via FFT: out = ifft(fft(g, size) * fft(h, size)) ... no that's convolution
    # 
    # Let me just use the loop approach but vectorized:
    # For large n, only a few weights are significant (when tau is small)
    
    significant = np.nonzero(weights > weights.max() * 1e-8)[0]
    grad = np.zeros(n)
    for k in significant:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = k - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * weights[k] * f[idx]
    return grad


def _gradient_topk(f, c, top_k, K):
    """Compute gradient targeting only top K lags (faster for sharp max)."""
    n = len(f)
    top_idx = np.argpartition(c, -K)[-K:]
    top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
    top_vals = c[top_idx]
    # Inverse-gap weights
    gaps = np.maximum(top_vals[0] - top_vals, 1e-15)
    weights = 1.0 / gaps
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for k, w in zip(top_idx, weights):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = k - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * w * f[idx]
    return grad


def _adam_step(f, grad, m, v, t, lr, beta1, beta2, eps):
    """Adam optimizer step with projection onto simplex."""
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad * grad)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
    f_new = _project_simplex(f_new)
    return f_new, m, v


def _spectral_correction(f, n, deadline):
    """
    Apply a spectral correction: identify low-frequency components in the
    power spectrum that cause broad autocorrelation peaks, and dampen them.
    """
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    power = np.abs(F) ** 2
    
    # Target: make the power spectrum as flat as possible (excluding DC)
    # But we can only change the amplitude, not the phase (to keep f real)
    # Actually we can change both, but then f becomes complex.
    # Instead, compute the "ideal" flat spectrum and project back.
    
    # Simple approach: compute the current spectrum, identify peaks,
    # and apply a small dampening
    dc = power[0]
    non_dc = power[1:]
    mean_nc = non_dc.mean()
    
    # Find frequencies where power is significantly above mean
    threshold = mean_nc * 1.5
    excess = np.maximum(power[1:] - threshold, 0)
    
    # The contribution of frequency omega to c[k] is |F(omega)|^2 * cos(omega*k) / size
    # To reduce the max of c, we want to reduce |F(omega)| where it's too high
    # This is approximate - we'll just scale down the excess
    
    if excess.sum() < 1e-15:
        return f.copy()
    
    # Adjust: reduce the amplitude at high-power frequencies
    scale = np.ones(size // 2 + 1)
    scale[1:] = 1.0 - 0.05 * (excess / (excess.max() + 1e-15))
    
    F_new = F * scale
    f_new = np.fft.irfft(F_new, size)[:n]
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s < 1e-10:
        return f.copy()
    return f_new / s


def _coord_descent_polish(f, n, deadline, rng, max_passes=2000):
    """
    Coordinate descent: find the peak lag, identify high/low contribution
    positions, and transfer mass to reduce the peak.
    """
    f = f.copy()
    local_best = _score_fast(f)
    local_best_f = f.copy()
    
    c = _autocorr_fft(f)
    peak_val = c.max()
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        c = _autocorr_fft(f)
        peak_val = c.max()
        k_star = int(np.argmax(c))
        
        # Contribution of each position to c[k_star]
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        contrib = np.zeros(n)
        if j_start <= j_end:
            idx = k_star - np.arange(j_start, j_end + 1)
            contrib[j_start:j_end + 1] = f[j_start:j_end + 1] * f[idx]
        
        # Also consider total contribution across top 5 peaks
        top5 = np.argpartition(c, -5)[-5:]
        total_contrib = np.zeros(n)
        for k in top5:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx_k = k - np.arange(js, je + 1)
                total_contrib[js:je + 1] += f[js:je + 1] * f[idx_k]
        
        # Donor: position with highest total contribution to peaks
        # Receiver: position with lowest total contribution (but non-zero f)
        mask_nonzero = f > f.mean() * 0.1
        if not np.any(mask_nonzero):
            break
        
        donor_scores = total_contrib.copy()
        donor_scores[~mask_nonzero] = np.inf
        j_don = int(np.argmin(f * (donor_scores > 0) * donor_scores + (donor_scores == 0) * 0))
        
        # Better: among positions with high contrib, find the one to reduce
        high_contrib = np.nonzero(total_contrib > np.median(total_contrib))[0]
        if len(high_contrib) == 0:
            break
        j_don = int(high_contrib[np.argmax(f[high_contrib] * total_contrib[high_contrib])])
        
        # Receiver: position with low contribution and where adding mass won't create new peaks
        low_contrib = np.nonzero(total_contrib < np.median(total_contrib))[0]
        if len(low_contrib) == 0:
            low_contrib = np.nonzero(total_contrib == total_contrib.min())[0]
        
        # Try several receivers
        best_improvement = 0
        best_move = None
        candidates = low_contrib[rng.permutation(len(low_contrib))[:10]]
        
        for j_rec in candidates:
            if j_rec == j_don:
                continue
            for frac in [0.05, 0.1, 0.2, 0.3]:
                amt = f[j_don] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[j_don] -= amt
                f_try[j_rec] += amt
                f_try = np.maximum(f_try, 0)
                f_try /= f_try.sum()
                c_try = _autocorr_fft(f_try)
                if c_try.max() < peak_val - 1e-14:
                    improvement = peak_val - c_try.max()
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_move = (j_don, j_rec, amt)
        
        if best_move is not None:
            j_don, j_rec, amt = best_move
            f[j_don] -= amt
            f[j_rec] += amt
            f = np.maximum(f, 0)
            f /= f.sum()
            c = _autocorr_fft(f)
            peak_val = c.max()
            score = _score_fast(f)
            if score < local_best:
                local_best = score
                local_best_f = f.copy()
        else:
            # Try random small perturbations
            for _ in range(3):
                i = int(rng.integers(0, n))
                jx = int(rng.integers(0, n))
                if i == jx:
                    continue
                amt = f[i] * 0.05
                f_try = f.copy()
                f_try[i] -= amt
                f_try[jx] += amt
                f_try = np.maximum(f_try, 0)
                f_try /= f_try.sum()
                c_try = _autocorr_fft(f_try)
                if c_try.max() < peak_val - 1e-14:
                    f = f_try
                    c = c_try
                    peak_val = c.max()
                    score = _score_fast(f)
                    if score < local_best:
                        local_best = score
                        local_best_f = f.copy()
                    break
    
    return local_best_f, local_best


def _optimize_adam(f0, n, deadline, rng, label="", verbose=True):
    """
    Adam-based gradient descent with soft-max objective and
    periodic spectral corrections.
    """
    f = np.maximum(f0, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s

    local_best = _score_fast(f)
    local_best_f = f.copy()

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.01
    t = 0
    m = np.zeros(n)
    v = np.zeros(n)
    
    patience = 50
    stuck_count = 0
    min_improvement = 1e-9
    
    # Adaptive parameters
    K = 5  # top K peaks for gradient
    use_softmax = True
    tau_frac = 0.02  # tau = tau_frac * c_max
    
    total_budget_iters = 500000
    it = 0
    last_time = time.monotonic()
    
    while time.monotonic() < deadline:
        it += 1
        if it % 10000 == 0 and it < 100000:
            # Decay learning rate
            lr *= 0.8
        
        progress = min(1.0, it / total_budget_iters)
        
        # Compute convolution
        c = _autocorr_fft(f)
        c_max = c.max()
        
        # Choose gradient type based on progress
        if progress < 0.3 and use_softmax:
            # Early: use soft-max for smooth gradient
            tau = c_max * tau_frac
            c_shifted = c - c.max()
            exp_c = np.exp(c_shifted / tau)
            weights = exp_c / exp_c.sum()
            grad = _gradient_softmax(f, c, weights)
        else:
            # Later: focus on top peaks
            grad = _gradient_topk(f, c, None, K)
        
        # Adam step
        t += 1
        f, m, v = _adam_step(f, grad, m, v, t, lr, beta1, beta2, eps)
        
        # Periodic evaluation
        if it % 200 == 0:
            score = _score_fast(f)
            if score < local_best - min_improvement:
                local_best = score
                local_best_f = f.copy()
                stuck_count = 0
                if verbose and it % 10000 == 0:
                    print(f"    {label} iter {it}: score={score:.7f}")
            else:
                stuck_count += 1
            
            # If stuck, try interventions
            if stuck_count >= patience:
                # Option 1: increase K
                if K < 10:
                    K = min(10, K * 2)
                # Option 2: spectral correction
                elif it % 400 == 0:
                    f_spec = _spectral_correction(f, n, deadline)
                    sc_spec = _score_fast(f_spec)
                    if sc_spec < local_best:
                        f = f_spec
                        local_best = sc_spec
                        local_best_f = f.copy()
                # Option 3: perturbation
                else:
                    amp = 0.01 * (1 + 0.5 * (stuck_count // patience))
                    f = f + rng.normal(0, amp, size=n) * f.mean()
                    f = _project_simplex(f)
                    m *= 0.1
                    v *= 0.1
                stuck_count = 0
            
            # Adaptive K: reduce when close to convergence
            if local_best < 1.51 and K > 3:
                K = max(3, K - 1)
    
    # Final evaluation
    score = _score_fast(f)
    if score < local_best:
        local_best = score
        local_best_f = f.copy()
    
    return local_best_f, local_best


def _multi_start_optimize(f_init, n, deadline, rng):
    """
    Generate diverse starting points, evaluate them, and optimize
    the best ones.
    """
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    starts = {}
    starts['given'] = f_init.copy()
    starts['uniform'] = np.ones(n) / n
    starts['ramp_down'] = np.linspace(0.01, 1.0, n)
    starts['ramp_up'] = np.linspace(1.0, 0.01, n)
    starts['u_shape'] = 1.0 + 2.0 * ((j - center) / max(center, 1)) ** 2
    starts['bimodal'] = (np.exp(-((j / n - 0.2) / 0.08) ** 2) +
                         np.exp(-((j / n - 0.8) / 0.08) ** 2) + 0.001)
    starts['asym1'] = np.exp(-1.5 * j / n) + 0.01
    starts['asym2'] = np.exp(-1.5 * (n - 1 - j) / n) + 0.01
    starts['sawtooth'] = np.abs((j / n * 7) % 1 - 0.5) + 0.01
    starts['random_exp'] = rng.exponential(1.0, size=n)
    starts['random_gamma'] = rng.gamma(2.0, 1.0, size=n)
    starts['quad'] = 1.0 + 3.0 * (j / n - 0.3) ** 2
    
    # Evaluate all starts
    start_results = []
    for name, f0 in starts.items():
        f = np.maximum(f0, 0.0)
        s = f.sum()
        if s < 1e-10:
            f = np.ones(n) / n
        else:
            f = f / s
        sc = _score_fast(f)
        start_results.append((sc, name, f))
    
    start_results.sort(key=lambda x: x[0])
    
    print("Starting point scores (sorted):")
    for sc, name, _ in start_results[:6]:
        print(f"  {name}: {sc:.7f}")
    
    return start_results


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the non-negative sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    
    Strategy:
    1. Multiple diverse starting points
    2. Adam gradient descent with soft-max / top-K objectives
    3. Spectral corrections for flatness
    4. Coordinate descent polishing
    5. Final fine-tuning with very small steps
    6. Adaptive time allocation
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 20.0
    t_start = time.monotonic()
    
    # Load initial sequence
    f_init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(f_init)
    
    best_score = np.inf
    best_f = f_init.copy()
    s0 = best_f.sum()
    if s0 > 1e-10:
        best_f = best_f / s0
    else:
        best_f = np.ones(n) / n
    
    initial_score = _score_fast(best_f)
    print(f"n={n}, initial score={initial_score:.7f}")
    best_score = initial_score
    
    # Phase 0: Generate and evaluate starting points
    start_results = _multi_start_optimize(best_f, n, deadline, rng)
    
    if start_results[0][0] < best_score:
        best_score = start_results[0][0]
        best_f = start_results[0][2].copy()
        print(f"Best start: {best_score:.7f}")
    
    # Phase 1: Optimize from top starts
    print(f"\n=== Phase 1: Adam optimization ===")
    n_starts_to_try = 4
    time_per_start = (budget_s - 400) * 0.65 / n_starts_to_try
    
    for idx in range(n_starts_to_try):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        
        sc, name, f0 = start_results[idx]
        t_alloc = min(time_per_start, remaining - 300)
        if t_alloc < 30:
            break
        
        start_deadline = time.monotonic() + t_alloc
        print(f"  [{name}] initial={sc:.6f}, budget={t_alloc:.0f}s")
        
        f_opt, sc_opt = _optimize_adam(
            f0, n, start_deadline, rng,
            label=name, verbose=True
        )
        
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"    -> New best: {best_score:.7f}")
        else:
            print(f"    -> Best from this start: {sc_opt:.7f}")
    
    # Phase 2: Continue optimizing from the overall best
    print(f"\n=== Phase 2: Refine best (score={best_score:.7f}) ===")
    remaining = deadline - time.monotonic()
    time_phase2 = min(remaining * 0.25, 300.0)
    if time_phase2 > 30:
        d2 = time.monotonic() + time_phase2
        f_opt, sc_opt = _optimize_adam(
            best_f, n, d2, rng,
            label="refine", verbose=True
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"    Refined to: {best_score:.7f}")
        else:
            print(f"    Refine result: {sc_opt:.7f}")
    
    # Phase 3: Coordinate descent polish
    print(f"\n=== Phase 3: Coordinate polish ===")
    remaining = deadline - time.monotonic()
    time_phase3 = min(remaining * 0.3, 200.0)
    if time_phase3 > 20:
        d3 = time.monotonic() + time_phase3
        f_pol, sc_pol = _coord_descent_polish(best_f, n, d3, rng, max_passes=5000)
        if sc_pol < best_score:
            best_score = sc_pol
            best_f = f_pol.copy()
            print(f"    Polished to: {best_score:.7f}")
        else:
            print(f"    Polish result: {sc_pol:.7f}")
    
    # Phase 4: Final Adam fine-tune with very small steps
    print(f"\n=== Phase 4: Fine-tune ===")
    remaining = deadline - time.monotonic()
    time_phase4 = min(remaining * 0.4, 150.0)
    if time_phase4 > 10:
        d4 = time.monotonic() + time_phase4
        # Use a modified adam with smaller lr
        f = best_f.copy()
        m = np.zeros(n)
        v = np.zeros(n)
        t_adam = 0
        lr_small = 0.001
        local_best = _score_fast(f)
        local_best_f = f.copy()
        
        while time.monotonic() < d4:
            c = _autocorr_fft(f)
            c_max = c.max()
            K = 3
            grad = _gradient_topk(f, c, None, K)
            t_adam += 1
            f, m, v = _adam_step(f, grad, m, v, t_adam, lr_small, 0.9, 0.999, 1e-8)
            
            if t_adam % 500 == 0:
                score = _score_fast(f)
                if score < local_best:
                    local_best = score
                    local_best_f = f.copy()
        
        score = _score_fast(f)
        if score < local_best:
            local_best = score
            local_best_f = f.copy()
        
        if local_best < best_score:
            best_score = local_best
            best_f = local_best_f.copy()
            print(f"    Fine-tuned to: {best_score:.7f}")
        else:
            print(f"    Fine-tune result: {local_best:.7f}")
    
    # Phase 5: Final coordinate polish
    if time.monotonic() < deadline - 30:
        print(f"\n=== Phase 5: Final polish ===")
        d5 = deadline - 10.0
        f_pol2, sc_pol2 = _coord_descent_polish(best_f, n, d5, rng, max_passes=3000)
        if sc_pol2 < best_score:
            best_score = sc_pol2
            best_f = f_pol2.copy()
            print(f"    Final polish: {best_score:.7f}")
    
    # Validate with official evaluator
    best_f = np.clip(best_f, 0.0, 1000.0)
    # Ensure sum > 0.01
    if best_f.sum() < 0.01:
        best_f = best_f / best_f.sum() * 1.0
    
    final_score = evaluate_sequence([float(x) for x in best_f])
    print(f"\nOfficial evaluation score: {final_score:.7f}")
    
    if not np.isfinite(final_score) or final_score > best_score * 1.001:
        # Something went wrong, fallback
        print("WARNING: official score is worse, checking...")
        # Try with explicit float conversion
        test_list = [float(max(0.0, min(1000.0, x))) for x in best_f]
        final_score2 = evaluate_sequence(test_list)
        if np.isfinite(final_score2) and final_score2 < final_score:
            final_score = final_score2
            best_f = np.array(test_list)
    
    print(f"\n=== FINAL SCORE: {best_score:.7f} ===")
    print(f"Official: {final_score:.7f}")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")
    
    return [float(x) for x in best_f]
