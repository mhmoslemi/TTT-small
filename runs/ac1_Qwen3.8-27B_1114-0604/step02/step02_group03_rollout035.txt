"""
Step-function optimizer using momentum-based projected gradient, 
asymmetric exponential seeds, and enhanced coordinate polish.
Focuses on minimizing 2*n*max(autocorr) for non-negative sequences.
"""

import numpy as np
import time
import math

def _normalize(v):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v

def _fft_size(m):
    if m <= 2:
        return 2
    # Use next power of 2 for FFT efficiency
    return 1 << (int(m) - 1).bit_length()

def _autocorr(f):
    n = f.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = 2 * n - 1
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]

def _fast_score(f):
    f = _normalize(f)
    if f.size == 0:
        return np.inf
    c = _autocorr(f)
    if c.size == 0:
        return np.inf
    return float(2.0 * f.size * c.max())

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    out = v - theta
    np.maximum(out, 0.0, out)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    out /= s
    return out

def _corr_grad(f, k):
    """
    Gradient of (a*a)[k] w.r.t. a.
    (a*a)[k] = sum_i a[i] * a[k-i]
    d/d(a[j]) (a*a)[k] = a[k-j] + a[j-k] (with boundary checks)
    """
    n = f.size
    g = np.zeros(n, dtype=np.float64)
    # a[k-j] is valid when 0 <= k-j < n, i.e., k-n+1 <= j <= k
    lo1 = max(0, k - n + 1)
    hi1 = min(n - 1, k)
    if lo1 <= hi1:
        g[lo1:hi1+1] += f[k - np.arange(lo1, hi1+1)]
    # a[j-k] is valid when 0 <= j-k < n, i.e., k <= j < k+n
    lo2 = max(0, k)
    hi2 = min(n - 1, k + n - 1)
    if lo2 <= hi2:
        g[lo2:hi2+1] += f[np.arange(lo2, hi2+1) - k]
    return g

def _weighted_grad(f, c, K):
    """
    Weighted gradient using top-K active lags.
    Computes sum_{k in topK} w_k * grad((a*a)[k])
    where w_k = c[k] / sum_{j in topK} c[j]
    """
    n = f.size
    L = c.size
    if K >= L:
        idx = np.arange(L)
    else:
        idx = np.argpartition(c, L - K)[L - K:]
    w = c[idx] / max(1e-300, c[idx].sum())
    g = np.zeros(n, dtype=np.float64)
    for k_idx in range(len(idx)):
        g += w[k_idx] * _corr_grad(f, int(idx[k_idx]))
    return g

def _seed_exp_asym(n, alpha, rng):

    i = np.arange(n, dtype=np.float64)

def _seed_power_asym(n, gamma, rng):

"""
Step-function optimizer minimizing 2*n*max(autocorr)/(sum)^2 for non-negative sequences.
Uses FFT-based weighted gradients, momentum-augmented projected steps, and aggressive
coordinate polish with pre-allocated buffers. Employs diverse asymmetric seeds and
multi-start with perturbation to avoid local minima.

import numpy as np
import time
import math

# ---------- Core helpers ----------

def _normalize(v):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v

def _fft_size(m):
    if m <= 2:
        return 2
    return 1 << (int(m) - 1).bit_length()

def _autocorr(f):
    n = f.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = 2 * n - 1
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]

def _fast_score(f):
    f = _normalize(f)
    if f.size == 0:
        return np.inf
    c = _autocorr(f)
    if c.size == 0:
        return np.inf
    return float(2.0 * f.size * c.max())

def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    out = v - theta
    np.maximum(out, 0.0, out)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    out /= s
    return out

# ---------- Gradient computation via FFT ----------

def _weighted_grad_fft(f, c, K):
    """Compute sum_k w_k * grad_k where w_k = c[k]/sum(c[topK]) for top-K lags."""
    n = f.size
    L = c.size
    if K >= L:
        w = np.full(L, 1.0 / L, dtype=np.float64)
    else:
        idx = np.argpartition(c, L - K)[L - K:]
        w = np.zeros(L, dtype=np.float64)
        w[idx] = c[idx]
        s = float(w.sum())
        if s < 1e-300:
            w.fill(1.0 / L)
        else:
            w /= s

    size = _fft_size(n + L - 1)
    F = np.fft.rfft(f, size)
    W = np.fft.rfft(w, size)
    # cross-correlation: ifft(FFT(f) * conj(FFT(w)))
    xc = np.fft.irfft(F * np.conj(W), size)
    # convolution: ifft(FFT(f) * FFT(w))
    conv = np.fft.irfft(F * W, size)
    return xc[:n] + conv[:n]

def _transfer_apply(c, f, i, j, dv, delta_buf, total_buf):

        move_count += 1
        no_improve = 0
        cur_score = float(2.0 * n * best_mc)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()

        # Periodic recompute to avoid drift
        if move_count % 80 == 0:
            f = _normalize(f)
            c = _autocorr(f)
            maxc = float(c.max())
            cur_score = float(2.0 * n * maxc)
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()

    f = _normalize(best_f)
    c = _autocorr(f)
    mc = float(c.max()) if c.size else 0.0
    return f, float(2.0 * n * mc), c

# ---------- Gradient optimization ----------

def _gradient_optimize(f0, budget_s, rng, label=""):
    """Momentum-based projected gradient with periodic coordinate polish."""
    f = _normalize(f0)
    n = f.size
    if n == 0:
        return f0, np.inf
    if budget_s <= 0:
        c = _autocorr(f)
        return f, float(2.0 * n * c.max())

    c = _autocorr(f)
    maxc = float(c.max())
    best_f = f.copy()
    best_score = float(2.0 * n * maxc)
    deadline = time.monotonic() + float(budget_s)

    # Momentum state
    momentum = np.zeros(n, dtype=np.float64)
    beta = 0.7
    step_size = 0.5
    no_accept = 0
    it = 0
    last_polish = time.monotonic()

    # Polish interval (seconds between coordinate polish calls)
    if n < 3000:
        polish_interval = 0.5
        polish_budget = 0.3
    else:
        polish_interval = 0.8
        polish_budget = 0.5

    print(f"[{label}] start score={best_score:.6f}")

    while time.monotonic() < deadline:
        it += 1
        if maxc <= 1e-15:
            break

        # Choose K for active set
        K = min(60, 2 * n - 1)
        g = _weighted_grad_fft(f, c, K)
        g_mean = float(g.mean())
        g_c = g - g_mean
        scale = float(np.abs(g_c).max())
        if scale < 1e-30:
            break

        g_c /= scale

        # Try multiple step sizes with momentum
        accepted = False
        best_mc = maxc
        best_f_new = None
        best_c_new = None
        best_mom = None

        if no_accept > 40:
            steps = (0.01, 0.03, 0.1, 0.3, 1.0)
        else:
            steps = (0.05, 0.1, 0.3, 0.7, 1.5)

        for eps in steps:
            # Nesterov-like lookahead
            v_new = momentum * beta - eps * g_c
            f_trial = _project_simplex(f + 0.5 * v_new)
            c_trial = _autocorr(f_trial)
            mc = float(c_trial.max())
            if mc < best_mc:
                best_mc = mc
                best_f_new = f_trial
                best_c_new = c_trial
                best_mom = v_new.copy()

        if best_f_new is not None and best_mc < maxc - 1e-16:
            f = best_f_new
            c = best_c_new
            maxc = best_mc
            momentum = best_mom
            accepted = True
            no_accept = 0
            cur_score = float(2.0 * n * maxc)
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()
            step_size = min(5.0, step_size * 1.05)
        else:
            no_accept += 1
            step_size = max(0.01, step_size * 0.85)

            # Periodic coordinate polish
            now = time.monotonic()
            if now - last_polish > polish_interval or no_accept > 30:
                fp, sp, cp = _coordinate_polish(f, polish_budget, rng)
                if sp < float(2.0 * n * maxc) - 1e-12:
                    f = _normalize(fp)
                    c = cp
                    maxc = float(c.max())
                    accepted = True
                    no_accept = 0
                    momentum.zero()
                    if sp < best_score:
                        best_score = sp
                        best_f = f.copy()
                else:
                    if no_accept > 50:
                        # Perturb to escape
                        sigma = float(rng.uniform(0.05, 0.8))
                        f = _normalize(f * np.exp(sigma * rng.standard_normal(n)))
                        c = _autocorr(f)
                        maxc = float(c.max())
                        momentum.zero()
                        no_accept = 20
                last_polish = time.monotonic()

        if it % 5000 == 0:
            cur = float(2.0 * n * maxc)
            print(f"[{label}] it={it} best={best_score:.6f} cur={cur:.6f} step={step_size:.3f}")

    cur_score = float(2.0 * n * float(c.max()))
    if cur_score < best_score:
        best_score = cur_score
        best_f = f.copy()
    print(f"[{label}] end best={best_score:.6f}")
    return _normalize(best_f), float(best_score)

# ---------- Seed generation ----------

def _seed_exp_asym(n, alpha):
    i = np.arange(n, dtype=np.float64)
    f = np.exp(alpha * i / max(1, n - 1))
    return _normalize(f)

def _seed_power(n, gamma):
    i = np.arange(n, dtype=np.float64)
    x = i / max(1, n - 1) + 1e-8
    f = np.power(x, gamma)
    return _normalize(f)

def _seed_fourier(n, rng):
    x = np.arange(n, dtype=np.float64) / n
    u = np.zeros(n, dtype=np.float64)
    for m in (1, 2, 3, 5, 8, 13, 21):
        u += float(rng.uniform(0.2, 1.2)) * np.cos(
            2.0 * np.pi * m * x + float(rng.uniform(0, 2 * np.pi))
        )
    f = np.exp(0.9 * u)
    return _normalize(f)

def _seed_staircase(n, alpha):
    i = np.arange(n, dtype=np.float64)
    f = 1.0 + alpha * i / max(1, n - 1)
    f = np.maximum(f, 1e-8)
    return _normalize(f)

def _seed_bimodal(n, p1, p2, width):
    i = np.arange(n, dtype=np.float64)
    x = i / max(1.0, n - 1.0)
    f = np.exp(-((x - p1) ** 2) / (2 * width ** 2)) + np.exp(-((x - p2) ** 2) / (2 * width ** 2))
    return _normalize(f)

def _seed_laplacian(n, alpha):
    """Laplacian-shaped seed centered at position c."""
    i = np.arange(n, dtype=np.float64)
    ctr = 0.4 * n  # off-center for asymmetry
    f = np.exp(-alpha * np.abs(i - ctr) / n)
    return _normalize(f)

def _build_seeds(f0, n, rng):
    """Build a diverse seed pool."""
    seeds = []

    def add(name, f):
        f = _normalize(f)
        try:
            sc = _fast_score(f)
            if np.isfinite(sc):
                seeds.append((sc, name, f))
        except Exception:
            pass

    # Given sequence and variants
    add("given", f0)
    add("reverse", f0[::-1])

    # Asymmetric exponentials
    for alpha in (-3, -2, -1, -0.5, 0.5, 1, 2, 3, 5, 8):
        add(f"exp_a{alpha}", _seed_exp_asym(n, float(alpha)))

    # Power laws
    for gamma in (-2, -1, -0.5, 0.5, 1, 2, 3, 5):
        add(f"pow_g{gamma}", _seed_power(n, float(gamma)))

    # Staircases
    for alpha in (-2, -1, -0.5, 0.5, 1, 2, 3):
        add(f"stair_a{alpha}", _seed_staircase(n, float(alpha)))

    # Fourier seeds
    for i in range(10):
        add(f"fourier{i}", _seed_fourier(n, rng))

    # Bimodal
    for (p1, p2) in [(0.2, 0.8), (0.1, 0.7), (0.3, 0.9), (0.15, 0.85), (0.25, 0.75)]:
        for w in (0.05, 0.1, 0.15, 0.25):
            add(f"bi_{p1}_{p2}_{w}", _seed_bimodal(n, p1, p2, w))

    # Laplacian (off-center)
    for alpha in (0.5, 1, 2, 3, 5):
        add(f"lap_a{alpha}", _seed_laplacian(n, alpha))
        add(f"lap_a{alpha}_r", _seed_laplacian(n, alpha)[::-1])

    # Smoothed versions of the given sequence
    for win in (5, 11, 31, 101, 301):
        if win < n:
            kernel = np.ones(win, dtype=np.float64) / float(win)
            s = np.convolve(f0, kernel, mode='same')
            add(f"smooth{win}", s)

    if not seeds:
        seeds.append((2.0, "uniform", _normalize(np.ones(n))))

    seeds.sort(key=lambda x: x[0])
    return seeds

# ---------- Main entry point ----------

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative coefficient sequence minimizing evaluate_sequence.
    Returns a Python list of floats.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(60.0, float(budget_s)) - 20.0

    # Load initial sequence
    try:
        f0 = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        f0 = np.ones(6121, dtype=np.float64)

    if f0.size == 0:
        f0 = np.ones(6121, dtype=np.float64)
    f0 = np.nan_to_num(f0, nan=0.0, posinf=1000.0, neginf=0.0)
    np.clip(f0, 0.0, 1000.0, out=f0)
    n = int(f0.size)
    if n < 3:
        f0 = np.ones(1024, dtype=np.float64)
        n = f0.size
    f0 = _normalize(f0)

    print(f"n={n}, initial score={_fast_score(f0):.6f}")

    # Phase 1: Build seed pool (~2% of time)
    t0 = time.monotonic()
    seed_pool = _build_seeds(f0, n, rng)
    print(f"Seed pool: {len(seed_pool)} seeds, best={seed_pool[0][0]:.6f}, time={time.monotonic()-t0:.1f}s")

    # Phase 2: Optimize top seeds (~65% of time)
    best_f = seed_pool[0][2].copy()
    best_score = seed_pool[0][0]
    best_c = _autocorr(best_f)

    t0 = time.monotonic()
    main_budget = 0.65 * (total_deadline - time.monotonic())
    top_seeds = seed_pool[:min(5, len(seed_pool))]

    # Allocate time proportionally to seed quality (better seeds get more time)
    scores = [s[0] for s in top_seeds]
    min_sc = min(scores)
    # Inverse-quality weighting: better (lower) score gets more time
    weights = [1.0 / (s - min_sc + 0.01) for s in scores]
    wsum = sum(weights)
    weights = [w / wsum for w in weights]

    for idx, (sc, name, f) in enumerate(top_seeds):
        alloc = main_budget * weights[idx]
        alloc = max(10.0, alloc)
        if time.monotonic() > total_deadline - 60:
            break
        print(f"\n--- Optimizing seed '{name}' (score={sc:.6f}) alloc={alloc:.1f}s ---")
        f_opt, sc_opt = _gradient_optimize(f, alloc, rng, name)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            best_c = _autocorr(best_f)
            print(f"  NEW BEST: {best_score:.6f}")

    # Phase 3: Multiple perturbation + re-optimize cycles (~15% of time)
    t0 = time.monotonic()
    perturb_budget = 0.15 * (total_deadline - time.monotonic())
    n_perturb = max(2, int(perturb_budget / 60.0))
    print(f"\n=== Perturbation phase: {n_perturb} cycles, budget={perturb_budget:.1f}s ===")

    for pi in range(n_perturb):
        if time.monotonic() > total_deadline - 200:
            break
        # Perturb the best
        r = float(rng.random())
        if r < 0.4:
            sigma = float(rng.uniform(0.05, 1.5))
            fp = _normalize(best_f * np.exp(sigma * rng.standard_normal(n)))
        elif r < 0.7:
            alpha = float(rng.uniform(0.1, 0.5))
            noise = rng.random(n)
            noise /= max(1e-300, noise.sum())
            fp = _normalize((1 - alpha) * best_f + alpha * noise)
        else:
            # Blend with a random seed
            seed_f = seed_pool[int(rng.integers(0, len(seed_pool)))][2]
            alpha = float(rng.uniform(0.1, 0.4))
            fp = _normalize((1 - alpha) * best_f + alpha * seed_f)

        alloc = max(20.0, (total_deadline - time.monotonic() - 200.0) / max(1, n_perturb - pi))
        f_opt, sc_opt = _gradient_optimize(fp, alloc, rng, f"perturb{pi}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            best_c = _autocorr(best_f)
            print(f"  Perturb {pi}: NEW BEST {best_score:.6f}")
        else:
            print(f"  Perturb {pi}: {sc_opt:.6f} (best={best_score:.6f})")

    # Phase 4: Final long coordinate polish (~20% of time)
    now = time.monotonic()
    polish_time = max(30.0, total_deadline - now - 15.0)
    print(f"\n=== Final coordinate polish: budget={polish_time:.1f}s ===")
    bf, bs, bc = _coordinate_polish(best_f, polish_time, rng)
    if bs < best_score:
        best_score = bs
        best_f = bf.copy()
        best_c = bc
        print(f"  Polish improved: {best_score:.6f}")
    else:
        print(f"  Polish result: {bs:.6f} (best={best_score:.6f})")

    # Phase 5: One more gradient + polish if time remains
    now = time.monotonic()
    extra = max(0.0, total_deadline - now - 10.0)
    if extra > 30.0:
        print(f"\n=== Extra pass: budget={extra:.1f}s ===")
        f2, s2 = _gradient_optimize(best_f, extra * 0.4, rng, "extra")
        if s2 < best_score:
            best_score = s2
            best_f = f2.copy()
            best_c = _autocorr(best_f)

        extra2 = max(0.0, total_deadline - time.monotonic() - 10.0)
        if extra2 > 15.0:
            bf2, bs2, bc2 = _coordinate_polish(best_f, extra2, rng)
            if bs2 < best_score:
                best_score = bs2
                best_f = bf2.copy()
                best_c = bc2

    # Final evaluation
    best_f = _normalize(best_f)
    best_f = np.nan_to_num(best_f, nan=0.0, posinf=0.0, neginf=0.0)
    np.clip(best_f, 0.0, 1000.0, out=best_f)
    best_f = best_f / max(1e-300, float(best_f.sum()))
    best_c = _autocorr(best_f)
    best_score = float(2.0 * best_f.size * best_c.max())

    # Official evaluation
    try:
        out = [float(x) for x in best_f]
        official = evaluate_sequence(out)
        if np.isfinite(official) and official < best_score + 1e-12:
            best_score = float(official)
    except Exception:
        pass

    out = [float(x) for x in best_f]
    print(f"\nFINAL score = {best_score:.6f}")
    print(f"Length = {len(out)}")
    return out
