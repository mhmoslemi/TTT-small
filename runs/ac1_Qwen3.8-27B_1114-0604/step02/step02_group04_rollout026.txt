"""
Optimized step-function search using adaptive mirror descent with FFT gradients.
Key improvements over previous approach:
1. Warm start from previous best (height_sequence_1)
2. Aggressive temperature annealing (start sharp, broaden gradually)
3. Slower learning rate decay for sustained progress
4. More frequent restarts with diverse perturbations

import numpy as np
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]

def _score(a):
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)

def _grad_fft(a, w):
    """Gradient: 2 * cross-correlation of w and a, computed via FFT."""
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]

def _softmax_peaks(c, T):
    """Compute softmax weights on convolution peaks."""
    c_max = float(np.max(c))
    exp_c = np.exp(T * (c - c_max))
    s = exp_c.sum()
    if s < 1e-300:
        w = np.zeros(c.size)
        w[np.argmax(c)] = 1.0
        return w
    return exp_c / s

def _make_sine_init(n, amp, phase=0.5):

    s = a.sum()

    return a

def _make_random_init(n, rng):

def _make_block_init(n, n_blocks, rng):

    for i in range(n_blocks):

"""
Optimized step-function search using adaptive mirror descent with FFT gradients.
Key improvements:
1. Warm start from previous best (height_sequence_1)
2. Aggressive temperature annealing with slower LR decay
3. Multi-length exploration (n=1024, 2048)
4. Euclidean projection variant for alternative geometry
5. Hard-max polish with adaptive K
6. Diverse restart perturbations
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _grad_fft(a, w):
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]


def _softmax_peaks(c, T):
    c_max = float(np.max(c))
    exp_c = np.exp(T * (c - c_max))
    s = exp_c.sum()
    if s < 1e-300:
        w = np.zeros(c.size)
        w[np.argmax(c)] = 1.0
        return w
    return exp_c / s


def _make_sine_init(n, amp, phase=0.5):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.sin(2.0 * np.pi * (i + phase) / n)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    return a


def _make_multi_sine_init(n, n_modes, rng):
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for m in range(n_modes):
        amp = rng.uniform(0.1, 0.6)
        freq = rng.randint(1, max(2, n // 64))
        phase = rng.uniform(0, 1.0)
        a += amp * np.sin(2.0 * np.pi * freq * (i + phase) / n)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    return a


def _make_random_init(n, rng):
    a = rng.dirichlet(np.ones(n) * 0.5)
    return a


def _normalize(a):
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return a
    return a / s


def _optimize_mirror(a, deadline, T_start=2.0, T_end=50.0, lr_scale=0.35,
                     patience=2500, eval_every=8, seed=42):
    a = np.asarray(a, dtype=np.float64).copy()
    a = _normalize(a)
    n = a.size
    lr0 = lr_scale * n
    best_a = a.copy()

    last_improve = 0
    total_restarts = 0

        update = -lr * grad_proj
        update = np.clip(update, -6.0, 6.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)

                                     lr_scale=0.5, patience=2500, eval_every=8, seed=4)
        results.append((ar, sr, "euclidean"))
        print(f"  n={ar.size}, score={sr:.8f}", flush=True)

    # Phase 3: Hard-max polish (250s split into 2 runs)
    d3 = hard_deadline - 10.0
    best_overall = min(results, key=lambda x: x[1]) if results else (best_init, best_init_score, "init")

    t3a = (d3 - time.monotonic()) * 0.6
    d3a = time.monotonic() + t3a
    if t3a > 30:
        print(f"[Phase 3a] Hard-max polish K=3 ({t3a:.0f}s)...", flush=True)
        ar, sr = _hard_max_polish(best_overall[0], d3a, K=3, lr_scale=0.4, seed=5)
        results.append((ar, sr, "polish_K3"))
        print(f"  n={ar.size}, score={sr:.8f}", flush=True)

    t3b = d3 - time.monotonic()
    if t3b > 30:
        best_overall = min(results, key=lambda x: x[1])
        print(f"[Phase 3b] Hard-max polish K=1 ({t3b:.0f}s)...", flush=True)
        ar, sr = _hard_max_polish(best_overall[0], d3, K=1, lr_scale=0.25, seed=6)
        results.append((ar, sr, "polish_K1"))
        print(f"  n={ar.size}, score={sr:.8f}", flush=True)

    # Phase 4: Validate and return best
    print("[Phase 4] Validating...", flush=True)
    results.sort(key=lambda x: x[1])

    best_out = None
    best_actual = np.inf

    for ar, sr, name in results[:6]:
        if time.monotonic() > hard_deadline:
            break
        try:
            arr_list = [float(max(0.0, min(1000.0, x))) for x in ar]
            actual = float(evaluate_sequence(arr_list))
        except Exception:
            actual = np.inf
        if np.isfinite(actual) and actual < best_actual:
            best_actual = actual
            best_out = arr_list
            print(f"  {name} (n={ar.size}): actual={actual:.8f}", flush=True)
            if best_actual < 1.49:
                break

    if best_out is None:
        best_out = [float(max(0.0, min(1000.0, x))) for x in best_init]
        try:
            best_actual = float(evaluate_sequence(best_out))
        except Exception:
            best_actual = best_init_score
        print(f"  Fallback: {best_actual:.8f}", flush=True)

    total_time = time.monotonic() - t0
    print(f"[Done] Time: {total_time:.1f}s, Best score: {best_actual:.8f}", flush=True)
    return best_out
