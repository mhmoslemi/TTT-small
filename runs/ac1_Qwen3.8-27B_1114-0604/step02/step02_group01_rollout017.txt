def _autocorr_fft(f):
    """Full linear convolution f*f via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]

def _score(f):
    """Compute the score."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))

    Returns (lse_value, gradient).

    w = np.exp(tau * c_shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:

        w[np.argmax(c)] = 1.0
        w_sum = 1.0
    else:
        w /= w_sum

Wait, I need to double-check the FFT correlation formula.

The circular cross-correlation of a and b is:
`(a ⋆ b)[m] = sum_k a[k] * conj(b[(k-m) mod N])`

For real signals: `(a ⋆ b)[m] = sum_k a[k] * b[(k-m) mod N]`

If N >= len(a) + len(b) - 1, this equals the linear cross-correlation:
`(a ⋆ b)[m] = sum_k a[k] * b[k-m]` (where b is 0 outside [0, len(b)-1])

So with a = w (length 2n-1) and b = f (length n), padded to size N:
`(w ⋆ f)[m] = sum_k w[k] * f[k-m]`

And we want `g[i] = sum_k w[k] * f[k-i]`, so `g = (w ⋆ f)[0:n]`.

This is correct!

Hmm, but the issue is that Adam with projection onto the simplex might not work great because the projection (renormalization) can conflict with the Adam updates.

Let me think about a better approach. What about using **log-space** optimization?

Let `f[i] = exp(log_f[i]) / sum(exp(log_f))` (softmax parameterization). Then the constraint f ≥ 0, sum(f) = 1 is automatically satisfied.

The gradient w.r.t. log_f would be different. Let me derive it:

`f[i] = exp(z[i]) / Z` where `Z = sum_j exp(z[j])`.

`∂f[i]/∂z[k] = f[i] * (δ(i,k) - f[k])`

So `∂/∂z of max_k c[k]` where `c[k] = sum_i f[i] * f[k-i]`:

`∂c[m]/∂f[i] = 2 * f[m-i]` (as derived before)

`∂c[m]/∂z[k] = sum_i (∂c[m]/∂f[i]) * (∂f[i]/∂z[k])`
`= sum_i 2*f[m-i] * f[i]*(δ(i,k) - f[k])`
`= 2*f[m-k]*f[k] - f[k] * sum_i 2*f[m-i]*f[i]`
`= 2*f[m-k]*f[k] - 2*f[k]*c[m]`

So `∂(max c)/∂z[k] = 2*f[k]*(f[m*-k] - c[m*])` where m* is the argmax.

This is a nice form! The gradient in log-space is:
`grad_z[k] = 2 * f[k] * (f[m*-k] - peak)` (where valid)

Note that `f[m*-k]` is the "contribution" of position k to the peak lag, and we're subtracting the peak value. If `f[m*-k] > peak` (which means position k has high value and contributes a lot to the peak), the gradient is positive (we should increase z[k], hence increase f[k])... wait, that doesn't make sense. We want to DECREASE the peak.

Let me re-derive. We want to minimize `c[m*] = sum_i f[i] * f[m*-i]`.

`∂c[m*]/∂z[k] = 2*f[k] * (f[m*-k] - c[m*])`

If `f[m*-k] > c[m*]`, the gradient is positive, meaning increasing z[k] would increase c[m*]. So we should DECREASE z[k] (i.e., the update `z -= lr * grad` would decrease z[k]).

If `f[m*-k] < c[m*]`, the gradient is negative, meaning increasing z[k] would decrease c[m*]. So we should INCREASE z[k].

This makes sense: we want to reduce the concentration at positions that contribute heavily to the peak.

This log-space parameterization automatically handles the non-negativity and sum-to-1 constraints!

def _optimize_logspace(f0, n, deadline, rng, max_iters=100000):
    f = np.maximum(f0, 1e-300).copy()
    f = f / f.sum()
    z = np.log(f)  # log-space parameter
    
    best_f = f.copy()
    best_score = _score(f)
    
    # Adam in log-space
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.05
    t = 0
    
    # LSE temperature (in units of the convolution values)
    tau_base = 50.0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t += 1
        
        # Compute convolution
        c = _autocorr_fft(f)
        
        # LSE weights
        c_max = c.max()
        w = np.exp(tau_base * (c - c_max))
        w_sum = w.sum()
        w /= w_sum
        
        # Gradient in log-space:
        # grad_z[k] = sum_m w[m] * 2*f[k]*(f[m-k] - c[m])
        # = 2*f[k] * (sum_m w[m]*f[m-k] - sum_m w[m]*c[m])
        # = 2*f[k] * (g[k] - w_dot_c)
        # where g[k] = sum_m w[m]*f[m-k] (cross-correlation of w and f)
        # and w_dot_c = sum_m w[m]*c[m]
        
        w_dot_c = np.dot(w, c)
        
        # Compute g = cross-correlation of w and f
        # g[k] = sum_m w[m] * f[m-k]
        # Using FFT
        size = 1 << (3 * n - 2).bit_length()
        W = np.fft.rfft(w, size)
        F = np.fft.rfft(f, size)
        g = np.fft.irfft(W * np.conj(F), size)[:n]
        
        grad_z = 2.0 * f * (g - w_dot_c)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Convert back to f (softmax)
        z_shifted = z - z.max()
        f = np.exp(z_shifted)
        f_sum = f.sum()
        f /= f_sum
        
        # Periodic evaluation
        if it % 200 == 0:
            sc = _score(f)
            if sc < best_score:
                best_score = sc
                best_f = f.copy()
                print(f"    iter {it}: score = {sc:.7f}")
        
        # Adaptive learning rate
        lr = 0.05 / (1 + 0.005 * it)
    
    sc = _score(f)
    if sc < best_score:
        best_score = sc
        best_f = f.copy()
    
    return best_f, best_score

Actually wait - is the problem convex in f? Let me check.

f(p) = max_k (p*p)[k] where (p*p)[k] = sum_i p[i]*p[k-i].

Each (p*p)[k] is a quadratic form in p. Is it convex?

(p*p)[k] = sum_i p[i] * p[k-i]

The Hessian of this w.r.t. p: if k is even, the diagonal has 2*p[k/2] at position k/2, and off-diagonal has 1 at positions (i, k-i) and (k-i, i) for i ≠ k/2.

Essentially, the Hessian is 2 * P_k where P_k is the "permutation matrix" that maps p[i] -> p[k-i]. This matrix has eigenvalues +1 and -1 (it's a symmetric involution), so the Hessian has eigenvalues in {-2, 0, 2}.

So (p*p)[k] is NOT convex in general (the Hessian has negative eigenvalues). The max of non-convex functions is also not convex.

This means the problem is NOT convex, and we can get stuck in local minima. The log-space approach might still work well in practice, but we need multiple restarts.

- i=n/2: f[n/2] * f[n/2] (both high) - contributes a lot

That's worse!

What about a shape that alternates high and low? Like a sinusoidal modulation? Then the contributions to the central lag would be somewhat averaged out.

Actually, the key insight is: for the central lag n-1, we need f[i] * f[n-1-i] to be small for all i. This is minimized when f[i] and f[n-1-i] are "anti-correlated" (one is high when the other is low). But since f must be non-negative, the best we can do is make one of them zero.

For non-negative f, the minimum of sum_i f[i]*f[n-1-i] subject to sum f = 1 is achieved by putting all mass on one side (either all on the first half or all on the second half). In that case, f[i]*f[n-1-i] = 0 for all i, so the central lag is 0. But then other lags would be large.

So there's a trade-off: reducing the central lag increases other lags. The optimal solution balances all lags.

```python
"""
Step function optimizer using LSE-smoothed Adam in log-space.
Key innovations over previous approach:
1. Log-sum-exp smoothing of the max (all peaks contribute to gradient)
2. Adam optimizer in log-space (automatic simplex constraint)
3. FFT-based gradient computation (O(n log n) per step)
4. Temperature annealing for exploration-to-exploitation transition
5. Multiple diverse restarts with spectral-aware initializations
"""

import numpy as np
import time


def _autocorr_fft(f):
    """Full linear convolution f*f via FFT. Returns array of length 2n-1."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _score_fast(f):
    """Fast score computation."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))


def _income_crosstair(w, f, n):
    """
    Compute g[k] = sum_m w[m] * f[m-k] for k=0..n-1
    using FFT. This is the cross-correlation of w and f.
    """
    size = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, size)
    F = np.fft.rfft(f, size)
    corr = np.fft.irfft(W * np.conj(F), size)
    return corr[:n]


def _lse_score_and_grad(c, tau, f, n):
    """
    Compute LSE approximation of max(c) and its gradient w.r.t. f.
    Returns (lse_value, gradient).
    """
    c_max = c.max()
    shifted = tau * (c - c_max)
    # Numerical stability
    exp_shifted = np.exp(shifted)
    w_sum = exp_shifted.sum()
    if w_sum < 1e-300:
        # Fallback: hard max
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
        w_sum = 1.0
    else:
        w = exp_shifted / w_sum
    
    lse_val = c_max + np.log(w_sum) / tau
    
    # Gradient: grad[i] = 2 * sum_k w[k] * f[k-i] (where valid)
    # = 2 * cross_corr(w, f)[i] for i=0..n-1
    # where cross_corr(w, f)[i] = sum_k w[k] * f[k-i]
    
    g = _income_crosstair(w, f, n)
    grad = 2.0 * g
    
    return lse_val, grad


def _optimize_logspace_cuda(f0, n, deadline, rng, 
                              max_iters=500000, 
                              lr_init=0.05, lr_min=0.001,
                              tau_init=20.0, tau_max=200.0,
                              verbose=False):
    """
    Optimize in log-space using Adam with LSE-smoothed objective.
    """
    f = np.maximum(f0, 1e-300).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f = f / s
    
    z = np.log(f)  # log-space parameter
    z -= z.max()  # numerical stability
    
    best_f = f.copy()
    best_score = _score_fast(f)
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    
    # Schedule parameters
    t = 0
    no_improve = 0
    patience = 5000
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t += 1
        
        # Adaptive learning rate
        progress = it / max_iters
        lr = lr_init * (lr_min / lr_init) ** (progress ** 0.5)
        
        # Adaptive temperature
        tau = tau_init + (tau_max - tau_init) * progress ** 0.3
        
        # Compute convolution
        c = _autocorr_fft(f)
        
        # LSE gradient w.r.t. z (log-space)
        # grad_z[k] = 2*f[k] * (g[k] - w_dot_c)
        # where g[k] = sum_m w[m]*f[m-k], w_dot_c = sum_m w[m]*c[m]
        
        c_max = c.max()
        shifted = tau * (c - c_max)
        exp_shifted = np.exp(shifted)
        w_sum = exp_shifted.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w_int = np.argmax(c)
            w[w_int] = 1.0
            w_sum = 1.0
        else:
            w = exp_shifted / w_sum
        
        w_dot_c = np.dot(w, c)
        g = _income_crosstair(w, f, n)
        grad_z = 2.0 * f * (g - w_dot_c)
        
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * grad_z * grad_z
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Convert to f via softmax
        z_shifted = z - z.max()
        f_new = np.exp(z_shifted)
        f_sum = f_new.sum()
        if f_sum < 1e-300:
            # Reset
            z = np.zeros(n)
            f = np.ones(n) / n
        else:
            f = f_new / f_sum
        
        # Periodic evaluation
        if it % 500 == 0:
            sc = _score_fast(f)
            if sc < best_score - 1e-9:
                best_score = sc
                best_f = f.copy()
                no_improve = 0
                if verbose:
                    print(f"      iter {it}: score = {sc:.7f}")
            else:
                no_improve += 500
        
        # If stuck for too long, try a perturbation
        if no_improve >= patience:
            # Simulated annealing style: random kick in log-space
            kick_scale = lr * 2.0
            z = z + rng.normal(0, kick_scale, size=n)
            z_shifted = z - z.max()
            f = np.exp(z_shifted)
            f = f / f.sum()
            no_improve = 0
            # Reset Adam state partially
            m *= 0.5
            v *= 0.5
    
    # Final evaluation
    sc = _score_fast(f)
    if sc < best_score:
        best_score = sc
        best_f = f.copy()
    
    return best_f, best_score


def _polish_coordinate(f0, n, deadline, rng):
    """
    Coordinate descent polish: iteratively reduce the peak by
    transferring mass from high-contribution positions to low ones.
    """
    f = f0.copy()
    f = np.maximum(f, 0)
    f = f / f.sum()
    
    best_f = f.copy()
    best_score = _score_fast(f)
    
    c = _autocorr_fft(f)
    peak = c.max()
    
    max_passes = 2000
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        # Find top peak lag
        k_star = int(np.argmax(c))
        
        # Contribution of each position to c[k_star]
        # c[k_star] = sum_i f[i] * f[k_star - i] (where valid)
        # Position i contributes f[i] * f[k_star - i]
        
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        
        if j_start > j_end:
            break
        
        # For each valid i in [j_start, j_end], contribution is f[i] * f[k_star - i]
        idx = np.arange(j_start, j_end + 1)
        partner = k_star - idx
        contrib = f[idx] * f[partner]
        
        # Total contribution of each position to top 3 peaks
        top3 = np.argpartition(c, -3)[-3:]
        top3 = top3[np.argsort(c[top3])[::-1]]
        
        total_contrib = np.zeros(n)
        for k in top3:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx_k = np.arange(js, je + 1)
                par_k = k - idx_k
                total_contrib[idx_k] += f[idx_k] * f[par_k]
        
        # Best donor: highest total contribution
        j_don = int(np.argmax(total_contrib))
        # Best receiver: lowest total contribution (and not the donor)
        mask = np.ones(n, dtype=bool)
        mask[j_don] = False
        j_rec = int(np.argmax(np.where(mask, -total_contrib, -np.inf)))
        
        if total_contrib[j_don] < 1e-15:
            break
        
        # Try several transfer amounts
        best_amt = 0
        best_peak = peak
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
            amt = f[j_don] * frac
            if amt < 1e-15:
                break
            f_try = f.copy()
            f_try[j_don] -= amt
            f_try[j_rec] += amt
            f_try = np.maximum(f_try, 0)
            s_try = f_try.sum()

"""
Step function optimizer using LSE-smoothed Adam in log-space.
Key innovations:
1. Log-sum-exp smoothing of max (all lags contribute to gradient via softmax weights)
2. Adam optimizer in log-space (automatic non-negativity + normalization)
3. FFT-based O(n log n) gradient per step
4. Temperature annealing (explore -> exploit)
5. Multiple diverse restarts with spectral-aware initializations
6. Coordinate descent polish phase
"""

import numpy as np
import time


def _autocorr_fft(f):
    """Full linear convolution f*f via FFT. Returns array of length 2n-1."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _score_fast(f):
    """Fast score: 2*n*max(conv(f,f))/(sum(f))^2."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))


def _cross_corr_wf(w, f, n):
    """
    Compute g[k] = sum_m w[m] * f[m-k] for k=0..n-1 via FFT.
    w has length 2n-1, f has length n.
    """
    size = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, size)
    F = np.fft.rfft(f, size)
    corr = np.fft.irfft(W * np.conj(F), size)
    return corr[:n]


def _optimize_logspace(f0, n, deadline, rng,
                       max_iters=200000,
                       lr_init=0.05, lr_min=0.0005,
                       tau_init=30.0, tau_max=300.0,
                       verbose=False, label=""):
    """
    Optimize in log-space using Adam with LSE-smoothed objective.
    The LSE temperature anneals from tau_init to tau_max.
    """
    f = np.maximum(f0, 1e-300).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s

    z = np.log(f)
    z -= z.max()

    best_f = f.copy()
    best_score = _score_fast(f)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    no_improve = 0
    patience = 3000

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        t += 1

        progress = min(1.0, it / max_iters)
        lr = lr_init * (lr_min / lr_init) ** (progress ** 0.5)
        tau = tau_init + (tau_max - tau_init) * progress ** 0.3

        c = _autocorr_fft(f)
        c_max = c.max()

        # LSE softmax weights
        shifted = tau * (c - c_max)
        exp_vals = np.exp(shifted)
        w_sum = exp_vals.sum()
        if w_sum < 1e-300:
            w = np.zeros_like(c)
            w[np.argmax(c)] = 1.0
            w_sum = 1.0
        else:
            w = exp_vals / w_sum

        # Gradient in log-space: grad_z[k] = 2*f[k]*(g[k] - w_dot_c)
        w_dot_c = np.dot(w, c)
        g = _cross_corr_wf(w, f, n)
        grad_z = 2.0 * f * (g - w_dot_c)

        # Adam
        m = beta1 * m + (1.0 - beta1) * grad_z
        v = beta2 * v + (1.0 - beta2) * grad_z * grad_z
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Softmax to get f
        z_shifted = z - z.max()
        f_new = np.exp(z_shifted)
        f_sum = f_new.sum()
        if f_sum < 1e-300:
            z = rng.normal(0, 0.1, size=n)
            z -= z.max()
            f = np.exp(z)
            f = f / f.sum()
        else:
            f = f_new / f_sum

        # Periodic evaluation
        if it % 500 == 0:
            sc = _score_fast(f)
            if sc < best_score - 1e-9:
                best_score = sc
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 2000 == 0:
                    print(f"        [{label}] iter {it}: {sc:.7f}")
            else:
                no_improve += 500

        # Escape mechanism
        if no_improve >= patience:
            kick = lr * 3.0
            z = z + rng.normal(0, kick, size=n)
            z -= z.max()
            f = np.exp(z)
            f = f / f.sum()
            no_improve = 0
            m *= 0.3
            v *= 0.3

    sc = _score_fast(f)
    if sc < best_score:
        best_score = sc
        best_f = f.copy()

    return best_f, best_score


def _optimize_mirror(f0, n, deadline, rng,
                     max_iters=150000,
                     K=3, verbose=False, label=""):
    """
    Focused multi-peak mirror descent with momentum.
    Alternative to log-space Adam for diversity.
    """
    f = np.maximum(f0, 1e-300).copy()
    f = f / f.sum()

    best_f = f.copy()
    best_score = _score_fast(f)

    momentum = np.zeros(n)
    beta = 0.85
    step_base = 1.0
    step_min = 0.001
    no_improve = 0
    patience = 4000

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        progress = min(1.0, it / max_iters)
        step = step_base * (step_min / step_base) ** (progress ** 0.6)

        c = _autocorr_fft(f)

        # Top-K peaks
        top_idx = np.argpartition(c, -K)[-K:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        top_vals = c[top_idx]

        # Weights
        gaps = np.maximum(top_vals[0] - top_vals, 1e-10)
        weights = 1.0 / gaps
        weights /= weights.sum()

        # Gradient at top lags
        grad = np.zeros(n)
        for k, w in zip(top_idx, weights):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                idx = k - np.arange(j_start, j_end + 1)
                grad[j_start:j_end + 1] += 2.0 * w * f[idx]

        # Mirror step with momentum
        grad_eff = grad + beta * momentum
        momentum = grad + beta * momentum

        span = grad_eff.max() - grad_eff.min()
        if span < 1e-30:
            continue
        eta = step / span
        log_f = np.log(f + 1e-300) - eta * grad_eff
        log_f -= log_f.max()
        f_new = np.exp(log_f)
        s_new = f_new.sum()
        if s_new < 1e-300:
            continue
        f = f_new / s_new

        if it % 500 == 0:
            sc = _score_fast(f)
            if sc < best_score - 1e-9:
                best_score = sc
                best_f = f.copy()
                no_improve = 0
                if verbose and it % 2000 == 0:
                    print(f"        [{label}] iter {it}: {sc:.7f}")
            else:
                no_improve += 500

        if no_improve >= patience:
            # Perturb
            mode = int(rng.integers(0, 4))
            jj = np.arange(n, dtype=np.float64)
            if mode == 0:
                alpha = rng.uniform(-1.0, 1.0)
                f = f * np.exp(alpha * (jj / n - 0.5))
            elif mode == 1:
                freq = int(rng.integers(1, max(2, n // 5)))
                phase = rng.uniform(0, 2 * np.pi)
                amp = rng.uniform(0.1, 0.4)
                f = f * (1.0 + amp * np.sin(2 * np.pi * freq * jj / n + phase))
            elif mode == 2:
                f = f + rng.uniform(-0.02, 0.02, size=n) * f.mean()
            else:
                i1 = int(rng.integers(0, n // 2))
                i2 = int(rng.integers(n // 2, n))
                amt = f[i1] * 0.2
                f[i1] -= amt
                f[i2] += amt

            f = np.maximum(f, 1e-300)
            f = f / f.sum()
            momentum *= 0.3
            no_improve = 0

    sc = _score_fast(f)
    if sc < best_score:
        best_score = sc
        best_f = f.copy()

    return best_f, best_score


def _polish_coordinate(f0, n, deadline, rng):
    """
    Coordinate descent: reduce peak by transferring mass from
    high-contributing positions to low-contributing ones.
    """

            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx_k = np.arange(js, je + 1)
                total_contrib[idx_k] += f[idx_k] * f[k - idx_k]

        j_don = int(np.argmax(total_contrib))
        mask = np.ones(n, dtype=bool)
        mask[j_don] = False
        j_rec = int(np.argmax(np.where(mask, -total_contrib, -np.inf)))

        if total_contrib[j_don] < 1e-15:

        best_amt = 0
        best_peak = peak
        for frac in [0.05, 0.1, 0.2, 0.4]:
            amt = f[j_don] * frac

            if st < 1e-10:

            c_try = _autocorr_fft(f_try)
            if c_try.max() < best_peak - 1e-14:
                best_peak = c_try.max()
                best_amt = amt

        if best_amt > 0:
            f[j_don] -= best_amt
            f[j_rec] += best_amt
            f = np.maximum(f, 0)
            f = f / f.sum()
            c = _autocorr_fft(f)
            peak = c.max()
            stall_count = 0

            sc = _score_fast(f)
            if sc < best_score:
                best_score = sc
                best_f = f.copy()
        else:
            stall_count += 1
            # Try random moves when stuck
            if stall_count > 5:
                improved = False
                for _ in range(5):
                    i = int(rng.integers(0, n))
                    jx = int(rng.integers(0, n))
                    if i == jx:

                    f_try = f.copy()
                    amt = f_try[i] * 0.05
                    f_try[i] -= amt
                    f_try[jx] += amt

                    c_try = _autocorr_fft(f_try)

                        c = c_try
                        peak = c.max()
                        sc = _score_fast(f)
                        if sc < best_score:
                            best_score = sc
                            best_f = f.copy()
                        improved = True
                        break
                if not improved and stall_count > 100:
                    break

    return best_f, best_score

    # Phase 4: Coordinate polish
    print(f"\n=== Phase 4: Coordinate polish (score={best_score:.7f}) ===")
    polish_deadline = deadline - 30.0
    alloc_polish = min(150.0, max(30.0, polish_deadline - time.monotonic()))
    d = time.monotonic() + alloc_polish
    f_pol, sc_pol = _polish_coordinate(best_f, n, d, rng)
    if sc_pol < best_score:
        best_score = sc_pol
        best_f = f_pol.copy()
        print(f"  Polished to: {best_score:.7f}")
    else:
        print(f"  Polish: {sc_pol:.7f}")

    # Phase 5: Final micro-polish
    if time.monotonic() < deadline - 10:
        print(f"\n=== Phase 5: Micro-polish ===")
        f_micro, sc_micro = _polish_coordinate(best_f, n, deadline - 5.0, rng)
        if sc_micro < best_score:
            best_score = sc_micro
            best_f = f_micro.copy()
            print(f"  Micro-polished to: {best_score:.7f}")

    # Validate with official evaluator
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = evaluate_sequence([float(x) for x in best_f])
    print(f"\nOfficial score: {final_score:.7f}")

    if not np.isfinite(final_score) or final_score > best_score:
        best_score = final_score if np.isfinite(final_score) else _score_fast(f_norm)
        best_f = f_norm

    print(f"\n=== FINAL: {best_score:.7f} | Time: {time.monotonic()-t_start:.1f}s ===")
    return [float(x) for x in best_f]
