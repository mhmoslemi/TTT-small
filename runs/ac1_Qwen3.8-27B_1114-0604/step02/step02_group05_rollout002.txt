import time

def _next_pow2(x):

    L = 1
    while L < x:
        L <<= 1
    return L

def _conv(a, b):

    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 3000000:

    L = _next_pow2(nout)

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10 or not np.isfinite(s):

def _subgradient(a, gamma=0.0):
    """
    Compute a (sub)gradient of max_j c_j where c=conv(a,a).
    If gamma=0, use exact argmax (uniform over all argmaxes).
    If gamma>0, use soft weighting: w_j ∝ exp(gamma * c_j).
    Returns (gradient vector, score).

    n = len(a)
    c = _conv(a, a)
    score = float(2.0 * n * c.max())
    
    if gamma < 1e-8:
        # Exact: uniform over argmax
        cmax = c.max()
        mask = (c >= cmax - 1e-14)
        w = mask.astype(float) / mask.sum()
    else:
        # Soft: w_j ∝ exp(gamma * c_j)
        cmax = c.max()
        e = np.exp(gamma * (c - cmax))
        w = e / e.sum()
    
    # gradient at index k: sum_j w_j * 2*a_{j-k} (if 0<=j-k<n)

    # Normalized: 0 to 0.5

    A_new = A * shape

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10 or not np.isfinite(s):

    return a_new / s

def _spectral_flatten_autocorr(a, alpha=1.5, L=None):
    """
    In frequency domain, modify |A|^2 to flatten the autocorrelation.
    The autocorrelation is IDFT(|A|^2). We want it flat.
    So we want |A|^2 to be concentrated (narrow in frequency).
    Equivalently: take the current autocorrelation c, raise to power alpha<1 to flatten peaks,
    then find new A such that |A|^2 corresponds to the flattened c.
    """
    n = len(a)
    if L is None:
        L = _next_pow2(2 * n - 1)
    
    c = _conv(a, a)
    cmax = c.max()
    if cmax < 1e-15:
        return a.copy()
    
    # Flatten: c_new = cmax * (c/cmax)^(1/alpha) for alpha > 1 (flattens peaks)
    c_flat = cmax * np.power(c / cmax, 1.0 / alpha)
    c_flat = np.maximum(c_flat, 0.0)
    
    # Now find A_new such that |A_new|^2 = FFT(c_flat) (approximately)
    new_spec = np.fft.rfft(c_flat, L)
    # new_spec should be real and non-negative (since c_flat is an autocorrelation)
    # But numerically there might be small negative parts
    new_spec = np.maximum(new_spec, 0.0)
    
    A_new = np.sqrt(new_spec) + 0j
    # Phase: keep the original phase (or use zero phase)
    A_orig = np.fft.rfft(a, L)
    # Use the phase of the original
    phase = np.angle(A_orig)
    A_new = np.sqrt(np.abs(new_spec)) * np.exp(1j * phase)
    
    a_new = np.fft.irfft(A_new, L)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return a_new / s

def _block_move(a, rng, n_blocks=8):
    """Random block scaling move."""
    n = len(a)
    a_new = a.copy()
    n_blocks = min(n_blocks, n)
    edges = np.sort(rng.uniform(0, 1, n_blocks - 1))
    edges = np.concatenate([[0], edges, [1]])
    for i in range(n_blocks):
        s = int(edges[i] * n)
        e = max(s + 1, int(edges[i+1] * n))
        e = min(n, e)
        factor = rng.lognormal(0, 0.3)
        a_new[s:e] *= factor
    return _sanitize(a_new)

def _init_uniform(n):
    return np.ones(n, dtype=float) / n

def _init_Welch(n, rng):
    """Initialize with a Welch-window-like shape."""
    t = np.arange(n, dtype=float) / n
    # Raised cosine
    p = 0.5 * (1 + np.cos(2 * np.pi * t))
    p = p ** float(rng.uniform(1.0, 3.0))
    p = np.maximum(p, 1e-10)
    return p / p.sum()

def _init_two_gaussian(n, rng):
    """Two separated Gaussian bumps."""
    c1 = rng.uniform(0.1, 0.35)
    c2 = rng.uniform(0.65, 0.9)
    s1 = rng.uniform(0.05, 0.15)
    s2 = rng.uniform(0.05, 0.15)
    t = np.arange(n, dtype=float) / n
    p = np.exp(-0.5 * ((t - c1) / s1)**2) + np.exp(-0.5 * ((t - c2) / s2)**2)
    p = np.maximum(p, 1e-10)
    return p / p.sum()

def _init_cosine_series(n, rng):
    """Sum of cosines with random phases."""
    t = np.arange(n, dtype=float) / n
    p = np.ones(n) * 0.5
    for k in range(1, 15):
        amp = rng.uniform(-0.4, 0.4) / k
        ph = rng.uniform(0, 2 * np.pi)
        p += amp * np.cos(2 * np.pi * k * t + ph)
    p = np.maximum(p, 1e-10)
    return p / p.sum()

def _init_from_parent(parent, n, rng):
    """Resample parent to length n with perturbation."""
    n_old = len(parent)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n) + 0.5) / n
    p = np.interp(x_new, x_old, parent)
    p = p * np.exp(rng.normal(0, 0.03, n))
    p = np.maximum(p, 1e-10)
    s = p.sum()
    if s < 1e-10:
        return _init_uniform(n)
    return p / s
