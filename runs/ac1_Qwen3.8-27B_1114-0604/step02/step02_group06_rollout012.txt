def _grad_obj_func(x, beta, L, n):
    """
    Objective and gradient for softmax parameterization.
    x is the raw parameter, p = softmax(x).
    Returns (objective, gradient_wrt_x).
    """
    x_shift = x - x.max()
    e = np.exp(x_shift)
    p = e / e.sum()
    
    Fp = np.fft.rfft(p, L)
    c = np.fft.irfft(Fp * Fp, L)[:2*n-1]
    c = np.maximum(c, 0.0)
    
    # Softmax approximation of max
    c_shift = c - c.max()
    e_c = np.exp(beta * c_shift)
    S = e_c.sum()
    M = c.max() + np.log(S) / beta
    w = e_c / S
    
    # h[j] = sum_k w[k] * p[k-j] (linear conv, first n elements)
    # This is the gradient of M w.r.t. p (up to factor 1/beta)
    Fw = np.fft.rfft(w, L)
    h = np.fft.irfft(Fw * np.conjugate(Fp), L)[:n]
    h = np.maximum(h, 0.0)
    
    obj = 2.0 * n * M
    
    # Gradient w.r.t. x: dp[j]/dx[k] = p[j]*(delta_jk - p[k])
    # dM/dx[k] = (1/beta) * sum_j h[j] * p[j] * (delta_jk - p[k])
    #          = (1/beta) * (h[k]*p[k] - p[k]*sum_j h[j]*p[j])
    #          = (p[k] / beta) * (h[k] - dot(h, p))
    dot_hp = float(np.dot(h, p))
    grad = (p / beta) * (h - dot_hp)
    grad_obj = 2.0 * n * grad
    
    return obj, grad_obj
