"""
Step function optimizer using adaptive mirror descent with full active-set gradient,
momentum, backtracking line search, and periodic LP-based large steps.
Adds coordinate descent polish and spectral diversification.
"""
import numpy as np
import time

def _autocorr(f):
    """Full linear autocorrelation via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2*n-1]

def _grad_from_lags(f, lags):
    """Subgradient of max autocorrelation from given lags."""
    n = len(f)
    grad = np.zeros(n)
    for k in lags:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        idx = k - np.arange(lo, hi)
        grad[lo:hi] += 2.0 * f[idx]
    return grad / max(1, len(lags))

def _active_lags(c, top_k=20, frac=0.995):
    """Get active lags: top-k or all within frac of max."""
    max_c = c.max()
    mask = c >= max_c * frac
    active = np.where(mask)[0]
    if len(active) > top_k:
        # Take top-k by value
        all_lags = np.argsort(c)[::-1][:top_k]
        active = all_lags
    elif len(active) == 0:
        active = np.array([np.argmax(c)])
    return active

def _mirror_step(f, grad, eta):
    """Mirror descent step on the simplex (normalized)."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    return f_new / s

def _line_search(f, grad, eta, c, max_iter=10):
    """Backtracking line search for mirror descent."""
    cur_peak = c.max()
    for _ in range(max_iter):
        f_new = _mirror_step(f, grad, eta)
        c_new = _autocorr(f_new)
        new_peak = c_new.max()
        if new_peak < cur_peak:
            return f_new, c_new, new_peak
        eta *= 0.5
    return f, c, cur_peak

def _coordinate_polish(f, c, max_coords=30):
    """Optimize the most 'pressured' coordinates using 1D ternary search."""
    n = len(f)
    max_c = c.max()
    active = _active_lags(c, top_k=20, frac=0.999)
    
    # Compute pressure on each coordinate
    pressure = np.zeros(n)
    for k in active:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        pressure[lo:hi] += f[k - np.arange(lo, hi)]
    
    # Pick coordinates with highest pressure
    top_coords = np.argsort(pressure)[::-1][:max_coords]
    
    f_best = f.copy()
    best_peak = c.max()
    
    for i in top_coords:
        # 1D optimization: find x that minimizes max_k c_k(x)
        lo_x = 0.0
        hi_x = min(2.0 * f[i] + 0.01 / n, 1.0)  # reasonable upper bound
        
        # Ternary search
        for _ in range(40):
            m1 = lo_x + (hi_x - lo_x) / 3.0
            m2 = hi_x - (hi_x - lo_x) / 3.0
            
            f_try1 = f_best.copy()
            f_try1[i] = m1
            f_try1 /= f_try1.sum()
            p1 = _autocorr(f_try1).max()
            
            f_try2 = f_best.copy()
            f_try2[i] = m2
            f_try2 /= f_try2.sum()
            p2 = _autocorr(f_try2).max()
            
            if p1 < p2:
                hi_x = m2
            else:
                lo_x = m1
        
        x_opt = (lo_x + hi_x) / 2.0
        f_try = f_best.copy()
        f_try[i] = x_opt
        f_try /= f_try.sum()
        p_opt = _autocorr(f_try).max()
        
        if p_opt < best_peak - 1e-15:
            f_best = f_try
            best_peak = p_opt
    
    return f_best, best_peak

def _spectral_perturb(f, rng, strength):
    """Perturb in the frequency domain to explore different shapes."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    # Add random phase perturbation
    phase = rng.standard_normal(len(F)) * strength
    F_perturbed = F * np.exp(1j * phase)
    f_new = np.fft.irfft(F_perturbed, size)[:n]
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f
    return f_new / s

def _optimize(f0, deadline, rng, n):
    """Main optimization loop with mirror descent + momentum + occasional LP."""
    f = f0.copy()
    f = np.maximum(f, 0.0)
    f /= f.sum()
    
    c = _autocorr(f)
    best_peak = c.max()
    best_f = f.copy()
    
    eta = 1.0
    momentum = np.zeros(n)
    beta = 0.5  # momentum coefficient
    
    iter_count = 0
    last_improve = time.monotonic()

        cur_peak = c.max()

            best_f = f.copy()

        # Momentum update

def _autocorr(f):
    """Full linear autocorrelation via real FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _grad_from_lags(f, lags):
    """Compute averaged subgradient from given autocorrelation lags."""
    n = len(f)
    grad = np.zeros(n)
    for k in lags:
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        idx = k - np.arange(lo, hi)
        grad[lo:hi] += 2.0 * f[idx]
    return grad / max(1, len(lags))


def _active_lags(c, top_k=20, frac=0.997):
    """Identify active lags: those within frac of max, capped at top_k."""
    max_c = c.max()
    mask = c >= max_c * frac
    active = np.where(mask)[0]
    if len(active) > top_k:
        active = np.argsort(c)[::-1][:top_k]
    elif len(active) == 0:
        active = np.array([np.argmax(c)])
    return active


def _mirror_step(f, grad, eta):
    """One mirror descent (exponentiated gradient) step, preserving normalization."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    return f_new / s
