    """Compute the smoothed gradient of max-convolution.
    grad[i] = sum_k w[k] * (p[k-i] + p[i-k]) with bounds.

"""
Refined multiplicative-update search for non-negative sequences minimizing
2*n*max(conv(a,a)) / sum(a)^2.

Key innovations:
- Proper gradient computation via FFT (cross-correlation + convolution)
- Softmax-weighted peak reduction with adaptive temperature
- Geometric line search for step size
- Multi-phase optimization: coarse search, fine search, local refinement
- Symmetry-aware updates and diverse restarts
- Direct coordinate refinement for final polish
"""

import numpy as np
import time


def _fft_len(n):
    need = 2 * n - 1
    L = 1
    while L < need:
        L <<= 1
    return L


def _score_conv(p, L=None):
    n = len(p)
    if L is None:
        L = _fft_len(n)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return float(2.0 * n * c.max()), c


def _compute_grad(p, w, L):
    P = np.fft.rfft(p, L)
    W = np.fft.rfft(w, L)
    n = len(p)
    g1 = np.fft.irfft(W * np.conjugate(P), L)[:n]
    g2 = np.fft.irfft(W * P, L)[:n]
    return np.maximum(g1 + g2, 0.0)


def _make_weights(c, beta):
    cmax = c.max()
    if beta < 1e-8:
        return np.ones(len(c), dtype=float) / len(c)
    diff = cmax - c
    exp_vals = np.exp(-beta * diff)
    s = exp_vals.sum()
    if s <= 0 or not np.isfinite(s):
        w = np.zeros(len(c))
        max_idx = np.where(c >= cmax - 1e-12)[0]
        w[max_idx] = 1.0
        w /= w.sum()
        return w
    return exp_vals / s


def _do_step(p, L, beta, lr):
    n = len(p)
    score, c = _score_conv(p, L)
    w = _make_weights(c, beta)
    grad = _compute_grad(p, w, L)
    gmax = grad.max()
    if gmax < 1e-300:
        return p.copy(), score
    grad_norm = grad / gmax
    p_new = p * np.exp(-lr * grad_norm)
    p_new = np.maximum(p_new, 1e-300)
    p_new = p_new / p_new.sum()
    new_score, _ = _score_conv(p_new, L)
    return p_new, new_score


def _line_search(p, L, beta, lr_base):
    n = len(p)
    cur_score, _ = _score_conv(p, L)
    best_p = p
    best_score = cur_score
    for mult in (0.25, 0.5, 0.75, 1.0, 1.5, 2.0, 3.0):
        lr = lr_base * mult
        if lr < 1e-12:
            continue
        p_try, sc_try = _do_step(p, L, beta, lr)
        if sc_try < best_score - 1e-14:
            best_score = sc_try
            best_p = p_try
    return best_p, best_score


def _sanitize(p):
    p = np.asarray(p, dtype=float).ravel()
    p = np.where(np.isfinite(p), p, 0.0
                 )
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s <= 0:
        return None
    return p / s


def _resample_to(p, n):
    p = _sanitize(p)
    if p is None:
        return None
    n_old = len(p)
    if n_old == n:
        return p.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n) + 0.5) / n
    p_new = np.interp(x_new, x_old, p)
    return _sanitize(p_new)


def _init_uniform(n, rng, mode):
    t = np.arange(n, dtype=float) / n
    if mode == 0:
        p = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0.5, 2.0)
        b = rng.uniform(0.5, 2.0)
        tt = np.clip(t, 1e-8, 1 - 1e-8)
        p = tt ** a * (1 - tt) ** b
    elif mode == 2:
        z = 1.0
        for k in range(1, 8):
            amp = rng.normal(0, 1.0 / (k + 1))
            ph = rng.uniform(0, 2 * np.pi)
            z += amp * np.cos(2 * np.pi * k * t + ph)
        p = np.log1p(np.exp(np.clip(z, -30, 30)))
    elif mode == 3:
        K = int(rng.integers(2, 6))
        p = np.zeros(n)
        for _ in range(K):
            center = rng.uniform(0.1, 0.9)
            width = rng.uniform(0.05, 0.3)
            weight = rng.uniform(0.5, 2.0)
            p += weight * np.exp(-((t - center) / width) ** 2)
    elif mode == 4:
        cut = rng.uniform(0.3, 0.7)
        ratio = rng.uniform(0.2, 3.0)
        p = np.ones(n)
        p[int(n * cut):] = ratio
    else:
        p = 0.5 * (1 + np.cos(2 * np.pi * t))
        p += 0.3 * np.cos(rng.uniform(3, 10) * t + rng.uniform(0, 2 * np.pi))
    p = np.maximum(p, 1e-8)
    return _sanitize(p)

def _local_refine(p0, L, time_limit, rng=None, verbose=False):

        else:
            w = int(rng.integers(large_b, max(large_b + 1, min(n, large_b * 5))))

        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))
        current_best_score = cur_score

            if s <= 0:

            cand /= s

def _equalize_top(p0, L, rng, n_trials=200):
    """Try to equalize the top few convolution values by targeted adjustments."""
    p = _sanitize(p0)
    if p is None:
        return p0, float('inf')
    n = len(p)
    cur_score, c = _score_conv(p, L)
    best_p = p.copy()
    best_score = cur_score

    top_k = 5
    for _ in range(n_trials):
        top_idx = np.argsort(c)[-top_k:]
        top_val = c[max(top_idx)]
        # Find positions contributing most to the top peaks
        contrib = np.zeros(n)
        for k in top_idx:
            for j in range(max(0, k - n + 1), min(k + 1, n)):
                if 0 <= k - j < n:
                    contrib[j] += p[j] * p[k - j]
        # Decrease highest contributors slightly
        top_contrib_idx = np.argsort(contrib)[-10:]
        factor = rng.uniform(0.7, 0.95)
        cand = p.copy()
        cand[top_contrib_idx] *= factor
        cand = np.maximum(cand, 1e-12)
        cand /= cand.sum()
        sc, c_new = _score_conv(cand, L)
        if sc < best_score - 1e-14:
            best_score = sc
            best_p = cand.copy()
            c = c_new
            cur_score = sc
        else:
            c = _score_conv(p, L)[1]

    return best_p, best_score

        if p_fin is not None:

            print(f"  Final score: {sc_fin:.8f}")

    # Select best
    if not cands:
        arr = np.ones(2048, dtype=float) / 2048.0
        return [float(x) for x in arr]

    cands.sort(key=lambda x: x[0])
    best_score = cands[0][0]
    best_arr = cands[0][1].copy()

    # Verify with official evaluator for small sequences
    rem = deadline - time.monotonic()
    if len(best_arr) <= 8192 and rem > 5:
        lst = [float(x) for x in best_arr]
        try:
            official = float(evaluate_sequence(lst))
            if np.isfinite(official):
                print(f"Official score: {official:.8f}")
                if official < best_score:
                    best_score = official
        except Exception:
            pass

    # Also check a couple other candidates
    for item in cands[1:4]:
        rem = deadline - time.monotonic()
        if rem < 3:
            break
        if len(item[1]) <= 8192:
            lst = [float(x) for x in item[1]]
            try:
                official = float(evaluate_sequence(lst))
                if np.isfinite(official) and official < best_score:
                    best_score = official
                    best_arr = item[1].copy()
                    print(f"  Better candidate found: {official:.8f}")
            except Exception:
                pass

    # Final sanity check
    out = _sanitize(best_arr)
    if out is None:
        out = np.ones(2048, dtype=float) / 2048.0

    print(f"Final best score: {best_score:.8f}, length={len(out)}")
    print(f"Total time: {time.monotonic() - global_start:.1f}s")

    return [float(x) for x in out]
