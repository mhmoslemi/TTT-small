"""
Frequency-domain informed optimizer for minimizing 2n*max(conv(a,a))/(sum a)^2.
Uses soft-max smoothing with L-BFGS-B, progressive temperature reduction,
and diverse initializations including spectral designs.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_self_conv(a):
    """Full self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def objective_and_grad(a, T, n):
    """
    Compute the soft-max smoothed objective and its gradient.
    f(a) = 2n * M / S^2 where M = T*log(sum(exp(c/T))), S = sum(a)
    
    Returns (f, grad).
    """
    c = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-10:
        return np.inf, np.zeros_like(a)
    
    # Soft-max with numerical stability
    c_max = float(c.max())
    exp_c = np.exp((c - c_max) / T)
    Z = float(exp_c.sum())
    w = exp_c / Z
    M = c_max + T * np.log(Z)
    
    f = 2.0 * n * M / (S * S)
    
    # Gradient: df/da[j] = 4n/S^2 * g[j] - 4n*M/S^3
    # where g[j] = sum_k w[k] * a[k-j] (cross-correlation of w and a)
    N = 1 << (3 * n - 2 - 1).bit_length()
    Wf = np.fft.rfft(w, N)
    Af = np.fft.rfft(a, N)
    g = np.fft.irfft(Wf * np.conj(Af), N)[:n]
    
    grad = (4.0 * n / (S * S)) * g - (4.0 * n * M / (S * S * S))
    return f, grad


def run_lbfgsb(a0, T, n, maxiter=2000, tol=1e-10):
    """Run L-BFGS-B optimization with given temperature."""
    def fun(a):
        f, _ = objective_and_grad(a, T, n)
        return f
    
    def jac(a):
        _, g = objective_and_grad(a, T, n)
        return g
    
    result = minimize(fun, a0, jac=jac, method='L-BFGS-B',
                      bounds=[(0, 1000)] * n,
                      options={'maxiter': maxiter, 'ftol': tol, 'gtol': 1e-8,
                               'maxls': 50, 'maxfun': 5000})
    return result.x, result.fun
