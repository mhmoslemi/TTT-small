def _mw_dual(f, max_iters=50):
    """
    Solve: min sum(y) s.t. (f ⊛ y)_j >= 1 for j=0..n-1, y >= 0.
    Returns (y, dual_value) where dual_value = sum(y).
    """
    n = f.size
    y = np.ones(2*n-1, dtype=np.float64)
    eta = 0.5
    
    for it in range(max_iters):
        # Compute v_j = sum_i f_i * y_{i+j} for j=0..n-1
        # Using FFT cross-correlation
        N = 1 << (3*n).bit_length()  # >= 3n-1
        F = np.fft.rfft(f, N)
        Y = np.fft.rfft(y, N)
        corr = np.fft.irfft(F * np.conj(Y), N)
        # corr[k] = sum_i f[i] * y[i+k] (approximately, with wrap-around)
        # For k < n, this should be correct if N >= 2n-1 + n - 1 = 3n-2
        v = corr[:n]
        
        # Check feasibility
        min_v = v.min()
        if min_v >= 1.0 - 1e-8:
            break
        
        # Compute positive violations
        r = np.maximum(0.0, 1.0 - v)
        
        # Compute h_k = sum_i f_i * r_{k-i} (convolution)
        # h = conv(f, r)
        h = np.convolve(f, r, mode='full')  # length 2n-1
        
        # Update y
        h_norm = h.max()
        if h_norm < 1e-300:
            break
        y *= np.exp(eta * h / h_norm)
        
        # Prevent overflow
        y_max = y.max()
        if y_max > 1e100:
            y /= y_max
    
    return y, float(y.sum())

"""
Step-function optimizer combining soft-gradient descent, dual-LP perturbation via
multiplicative weights, and coordinate-pair polish. Exploits sequence asymmetry
to break the symmetric lower bound of 2. Multi-dimensional search with adaptive
time allocation.
"""

import numpy as np
import time


def _normalize(v):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v


def _fft_size(m):
    if m <= 2:
        return 2
    return 1 << (m - 1).bit_length()


def _autocorr(f):
    n = f.size
    m = 2 * n - 1
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]


def _fast_score(f):
    f = _normalize(f)
    if f.size == 0:
        return np.inf
    c = _autocorr(f)
    if c.size == 0:
        return np.inf
    return float(2.0 * f.size * c.max())


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    out = v - theta
    np.maximum(out, 0.0, out)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    out /= s
    return out


def _soft_grad(f, c, temperature):
    """Compute soft-max weighted gradient: g_m = 2 * sum_k p_k * f_{k-m}."""
    n = f.size
    L = c.size
    maxc = c.max()
    z = (c - maxc) / max(temperature, 1e-300)
    np.clip(z, -50.0, 0.0, out=z)
    p = np.exp(z)
    ps = float(p.sum())
    if ps < 1e-300:
        p.fill(1.0 / L)
    else:
        p /= ps
    # g_m = 2 * sum_k p_k * f_{k-m} = 2 * (p * f_rev)[m] where f_rev[i] = f[n-1-i]
    # Actually: sum_k p_k * f_{k-m} is cross-correlation.
    # Use FFT: convolve p (length 2n-1) with reversed f (length n)
    # (p * f_rev)[k] = sum_i p_i * f_rev[k-i] = sum_i p_i * f[n-1-(k-i)] = sum_i p_i * f[n-1-k+i]
    # Let j = n-1-k+i, i = j-(n-1-k) = j-k+n-1
    # Hmm, let me just use the direct formula with FFT.
    # g_m = 2 * sum_{k=0}^{2n-2} p_k * f_{k-m} (where f_x = 0 if x out of [0,n-1])
    # = 2 * sum_{k=m}^{m+2n-2} p_k * f_{k-m} (only valid range of f)
    # = 2 * sum_{i=0}^{n-1} p_{m+i} * f_i
    # This is the cross-correlation of p and f at lag m.
    # FFT method: IFFT(conj(FFT(f)) * FFT(p)) gives corr[m] = sum_i f_i * p_{i+m}
    size = _fft_size(3 * n)
    F = np.fft.rfft(f, size)
    P = np.fft.rfft(p, size)
    corr = np.fft.irfft(np.conj(F) * P, size)
    return 2.0 * corr[:n]


def _topk_gradient(f, c, K=20):
    """Gradient using top-K lags only."""
    n = f.size
    L = c.size
    K = min(K, L)
    top_idx = np.argpartition(c, L - K)[L - K:]
    top_c = c[top_idx]
    max_top = top_c.max()
    z = (top_c - max_top) / max(1e-300, max_top * 0.1)
    np.clip(z, -40.0, 0.0, out=z)
    w = np.exp(z)
    ws = float(w.sum())
    if ws < 1e-300:
        w.fill(1.0 / K)
    else:
        w /= ws
    p = np.zeros(L, dtype=np.float64)
    p[top_idx] = w
    size = _fft_size(3 * n)
    F = np.fft.rfft(f, size)
    P = np.fft.rfft(p, size)
    corr = np.fft.irfft(np.conj(F) * P, size)
    return 2.0 * corr[:n]


def _mw_dual_direction(f, max_iters=30):
    """
    Use multiplicative weights to approximately solve the dual covering LP:
      min sum(y) s.t. sum_i f_i * y_{i+j} >= 1 for j=0..n-1, y >= 0.
    Return a direction vector d of length n where d_j > 0 indicates
    that coordinate j should be increased (g_j > 0 in the primal).
    """
    n = f.size
    y = np.ones(2 * n - 1, dtype=np.float64) / (2 * n - 1)
    size = _fft_size(3 * n)
    F = np.fft.rfft(f, size)
    eta = 0.3

    for it in range(max_iters):
        Y = np.fft.rfft(y, size)
        corr = np.fft.irfft(np.conj(F) * Y, size)
        v = corr[:n]
        min_v = float(v.min())
        if min_v >= 0.999:
            break
        r = np.maximum(0.0, 1.0 - v)
        h = np.convolve(f, r, mode='full')
        hmax = float(h.max())
        if hmax < 1e-300:
            break
        y *= np.exp(eta * h / hmax)
        ymax = float(y.max())
        if ymax > 1e50:
            y /= ymax

    # The direction: where dual constraints are tight, primal g should be positive.
    # g_j proportional to 1/f_j where (f⊛y)_j is close to 1.
    Y = np.fft.rfft(y, size)
    corr = np.fft.irfft(np.conj(F) * Y, size)
    v = corr[:n]
    # Direction: inverse of f, weighted by tightness
    f_safe = np.maximum(f, 1e-15)
    tightness = np.exp(-3.0 * np.abs(v - 1.0))
    d = tightness / f_safe
    d = np.maximum(d, 0.0)
    ds = float(d.sum())
    if ds < 1e-300:
        d = np.ones(n, dtype=np.float64) / n
    else:
        d /= ds
    return d


def _coordinate_polish(f0, budget_s, rng, topK_lags=15):
    """
    Local search by exact single mass transfers with incremental convolution updates.
    """
    f = _normalize(f0)
    n = f.size
    if n < 2 or budget_s <= 0:
        c = _autocorr(f)
        return f, float(2.0 * n * c.max()), c

    L = 2 * n - 1
    c = _autocorr(f).copy()
    maxc = float(np.max(c))
    best_f = f.copy()
    best_score = float(2.0 * n * maxc)

    deadline = time.monotonic() + budget_s
    delta = np.zeros(L, dtype=np.float64)
    total = np.zeros(L, dtype=np.float64)
    no_improve = 0
    move_count = 0

    D = min(12, n // 4)
    R = min(12, n // 4)
    fracs = (0.02, 0.05, 0.1, 0.25, 0.5)

    while time.monotonic() < deadline and no_improve < 15:
        maxc = float(np.max(c))

        # Find active lags (top K by value)
        K = min(topK_lags, L)
        active = np.argpartition(c, L - K)[L - K:]
        p = np.zeros(L, dtype=np.float64)
        p[active] = 1.0 / K

        # Weighted gradient for donor/receiver selection
        size = _fft_size(3 * n)
        F = np.fft.rfft(f, size)
        P = np.fft.rfft(p, size)
        corr = np.fft.irfft(np.conj(F) * P, size)
        g = 2.0 * corr[:n]
        gm = g - g.mean()

        donor_order = np.argsort(gm)[::-1]
        recv_order = np.argsort(gm)

        donors = []
        for idx in donor_order:
            idx = int(idx)
            if f[idx] > 1e-14:
                donors.append(idx)
                if len(donors) >= D:
                    break

        donor_set = set(donors)
        receivers = []
        for idx in recv_order:
            idx = int(idx)
            if idx not in donor_set:
                receivers.append(idx)
                if len(receivers) >= R:
                    break

        if not donors or not receivers:
            no_improve += 1
            if no_improve >= 5:
                break
            continue

        # Precompute unit vectors for donors and receivers
        donor_units = []
        for i in donors:
            u = np.zeros(L, dtype=np.float64)
            u[i:i + n] = -f
            donor_units.append((i, u, float(f[i])))

        recv_units = []
        for j in receivers:
            u = np.zeros(L, dtype=np.float64)
            u[j:j + n] = f
            recv_units.append((j, u))

        best_mc = maxc
        best_move = None

        for i, ui, fi in donor_units:
            if fi <= 1e-14:
                continue
            for j, uj in recv_units:
                if i == j:
                    continue
                for frac in fracs:
                    dv = fi * frac
                    if dv <= 0:
                        continue
                    # Compute delta_c
                    delta.fill(0.0)
                    np.add(ui, uj, out=delta)
                    delta *= (2.0 * dv)
                    q = dv * dv
                    if 2 * j < L:
                        delta[2 * j] += q
                    if 2 * i < L:
                        delta[2 * i] += q
                    if i + j < L:
                        delta[i + j] -= 2.0 * q
                    np.add(c, delta, out=total)
                    mc = float(np.max(total))
                    if mc < best_mc:
                        best_mc = mc
                        best_move = (i, j, dv, delta.copy())

        # Random fallback
        if best_mc >= maxc - 1e-18:
            for _ in range(20):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j or f[i] <= 1e-14:
                    continue
                frac = float(rng.uniform(0.01, 0.3))
                dv = float(f[i]) * frac
                delta.fill(0.0)
                delta[i:i + n] -= f
                delta[j:j + n] += f
                delta *= (2.0 * dv)
                q = dv * dv
                if 2 * j < L:
                    delta[2 * j] += q
                if 2 * i < L:
                    delta[2 * i] += q
                if i + j < L:
                    delta[i + j] -= 2.0 * q
                np.add(c, delta, out=total)
                mc = float(np.max(total))
                if mc < best_mc:
                    best_mc = mc
                    best_move = (i, j, dv, delta.copy())

        if best_move is None or best_mc >= maxc - 1e-18:
            no_improve += 1
            continue

        i, j, dv, d_saved = best_move
        f[i] -= dv
        f[j] += dv
        c = c + d_saved
        move_count += 1
        no_improve = 0
        cur_score = float(2.0 * n * best_mc)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()

        if move_count % 80 == 0:
            f = _normalize(f)
            c = _autocorr(f)
            mc_now = float(c.max())
            sc_now = float(2.0 * n * mc_now)
            if sc_now < best_score:
                best_score = sc_now
                best_f = f.copy()

    f = _normalize(f)
    c = _autocorr(f)
    mc = float(c.max()) if c.size else 0.0
    sc = float(2.0 * n * mc)
    if sc <= best_score:
        return f, sc, c
    bf = _normalize(best_f)
    bc = _autocorr(bf)
    return bf, float(2.0 * n * bc.max()), bc


def _perturb(best_f, rng):
    """Generate a perturbation around the best state."""
    n = best_f.size
    r = float(rng.random())
    if r < 0.4:
        sigma = float(rng.uniform(0.05, 0.6))
        f = best_f * np.exp(sigma * rng.standard_normal(n))
    elif r < 0.7:
        alpha = float(rng.uniform(0.1, 0.5))
        noise = rng.random(n)
        s = noise.sum()
        if s > 1e-300:
            noise /= s
        f = (1 - alpha) * best_f + alpha * noise
    else:
        # Asymmetric shift
        shift = int(rng.integers(n // 10, n // 3))
        f = np.roll(best_f, shift)
        # Taper the wraparound
        taper = min(n // 5, 100)
        if taper > 1:
            for i in range(taper):
                w = i / taper
                f[i] *= w
                f[-1 - i] *= w
        f = np.maximum(f, 0)
    return _normalize(f)


def _gradient_optimize(f0, budget_s, rng, tag=""):
    """Main gradient optimization with dual-LP steps and coordinate polish."""
    f = _normalize(f0)
    n = f.size
    if n == 0:
        return f0, np.inf
    if budget_s <= 0:
        c = _autocorr(f)
        return f, float(2.0 * n * c.max())

    c = _autocorr(f)
    maxc = float(c.max())
    best_f = f.copy()
    best_score = float(2.0 * n * maxc)

    deadline = time.monotonic() + budget_s
    sigma = 0.1
    min_sigma = 1e-9
    no_accept = 0
    it = 0
    last_polish = time.monotonic()
    last_dual = time.monotonic()

    coord_interval = 0.5
    coord_budget = 0.4
    dual_interval = 2.0

    print(f"[{tag}] start n={n} score={best_score:.6f}")

    while time.monotonic() < deadline:
        it += 1
        if maxc <= 1e-15:
            break

        # Compute gradient using soft-max or top-K
        if it % 5000 < 2500:
            # Use top-K
            K = min(30, 2 * n - 1)
            g = _topk_gradient(f, c, K)
        else:
            # Use soft-max with adaptive temperature
            T = maxc * sigma + 1e-300
            g = _soft_grad(f, c, T)

        gm = g - g.mean()
        scale = float(gm.max() - gm.min())

        accepted = False
        if scale > 1e-300:
            best_mc = maxc
            best_ft = None
            best_ct = None

            if no_accept > 40:
                exps = (0.003, 0.01, 0.05, 0.2, 1.0)
            else:
                exps = (0.01, 0.05, 0.3, 1.5)

            for e in exps:
                # Log-domain step
                arg = -e * (gm / scale)
                np.clip(arg, -30.0, 30.0, out=arg)
                ft = f * np.exp(arg)
                ft = _normalize(ft)
                ct = _autocorr(ft)
                mc = float(ct.max())
                if mc < best_mc:
                    best_mc = mc
                    best_ft = ft
                    best_ct = ct

                # Euclidean step
                if e in (0.05, 0.3):
                    ft2 = _project_simplex(f - e * (gm / scale))
                    ct2 = _autocorr(ft2)
                    mc2 = float(ct2.max())
                    if mc2 < best_mc:
                        best_mc = mc2
                        best_ft = ft2
                        best_ct = ct2

            if best_ct is not None and best_mc < maxc - 1e-15:
                f = best_ft
                c = best_ct
                maxc = best_mc
                accepted = True

        if accepted:
            no_accept = 0
            cur_score = float(2.0 * n * maxc)
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()
            sigma = max(min_sigma, sigma * 0.995)
        else:
            no_accept += 1
            now = time.monotonic()

            # Periodic dual-LP direction
            if now - last_dual > dual_interval and no_accept > 10:
                d = _mw_dual_direction(f, max_iters=20)
                alpha = float(rng.uniform(0.05, 0.3))
                ft = (1 - alpha) * f + alpha * d
                ft = _normalize(ft)
                ct = _autocorr(ft)
                mc = float(ct.max())
                if mc < maxc:
                    f = ft
                    c = ct
                    maxc = mc
                    accepted = True
                    no_accept = 0
                    cur_score = float(2.0 * n * maxc)
                    if cur_score < best_score:
                        best_score = cur_score
                        best_f = f.copy()
                last_dual = time.monotonic()

            # Coordinate polish
            if now - last_polish > coord_interval or no_accept > 25:
                remaining = deadline - time.monotonic()
                if remaining > coord_budget + 10:
                    fp, sc, cp = _coordinate_polish(f, c, coord_budget, rng)
                    if sc < best_score:
                        best_score = sc
                        best_f = _normalize(fp)
                        f = _normalize(fp)
                        c = cp
                        maxc = float(c.max())
                    else:
                        f = fp
                        c = _autocorr(f)
                        maxc = float(c.max())
                    accepted = True
                    no_accept = 0
                last_polish = time.monotonic()

            # Perturb
            if no_accept > 50 or (it % 8000 == 0):
                f = _perturb(best_f, rng)
                c = _autocorr(f)
                maxc = float(c.max())
                sigma = min(0.3, sigma * 2.0 + 0.01)
                no_accept = 0

        if it % 5000 == 0:
            print(f"[{tag}] it={it} best={best_score:.6f} cur={2.0*n*maxc:.6f} sigma={sigma:.2e}")

    # Final check
    cur_score = float(2.0 * n * float(c.max()))
    if cur_score < best_score:
        best_score = cur_score
        best_f = f.copy()

    print(f"[{tag}] end best={best_score:.6f}")
    return _normalize(best_f), float(best_score)


def _interp_to_length(f, n):
    f = _normalize(f)
    m = f.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if m == n:
        return f
    if n == 1:
        return np.array([1.0])
    if m == 1:
        return np.ones(n, dtype=np.float64) / n
    x_old = np.linspace(0.0, m - 1, m)
    x_new = np.linspace(0.0, n - 1, n)
    return _normalize(np.interp(x_new, x_old, f))


def _asymmetric_seed(n, rng):
    """Create an asymmetric smooth seed."""
    x = np.linspace(0.0, 1.0, n)
    center = float(rng.uniform(0.2, 0.5))
    width = float(rng.uniform(0.15, 0.4))
    skew = float(rng.uniform(-1.0, 1.0))
    f = np.exp(-((x - center) / width) ** 2)
    f *= (1.0 + skew * (x - 0.5))
    f = np.maximum(f, 0.01)
    return _normalize(f)


def _two_bump_seed(n, rng):
    """Two-bump asymmetric seed."""
    x = np.linspace(0.0, 1.0, n)
    c1 = float(rng.uniform(0.1, 0.35))
    c2 = float(rng.uniform(0.55, 0.85))
    w1 = float(rng.uniform(0.08, 0.25))
    w2 = float(rng.uniform(0.08, 0.25))
    a1 = float(rng.uniform(0.3, 0.7))
    f = a1 * np.exp(-((x - c1) / w1) ** 2) + (1 - a1) * np.exp(-((x - c2) / w2) ** 2)
    return _normalize(f)


def _uniform_with_noise(n, rng):
    """Uniform with smooth noise."""
    f = np.ones(n, dtype=np.float64)
    for m in (1, 2, 3, 5):
        amp = float(rng.uniform(-0.3, 0.3))
        phase = float(rng.uniform(0, 2 * np.pi))
        f += amp * np.cos(2 * np.pi * m * np.arange(n) / n + phase)
    f = np.maximum(f, 0.0)
    return _normalize(f)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative coefficient sequence minimizing evaluate_sequence.
    Returns a Python list of floats.
    """
    rng = np.random.default_rng(seed)
    total_deadline = time.monotonic() + max(60.0, float(budget_s)) - 30.0

    # Load initial sequence
    try:
        f0 = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        f0 = np.ones(6121, dtype=np.float64)
    if f0.size == 0:
        f0 = np.ones(6121, dtype=np.float64)
    f0 = np.nan_to_num(f0, nan=0.0, posinf=1000.0, neginf=0.0)
    np.clip(f0, 0.0, 1000.0, out=f0)
    n0 = int(f0.size)
    f0 = _normalize(f0)

    best_f = f0.copy()
    best_c = _autocorr(best_f)
    best_score = float(2.0 * n0 * best_c.max()) if best_c.size else np.inf
    print(f"Initial score: {best_score:.6f} (n={n0})")

    # Try multiple dimensions
    # The score doesn't depend on n for resampled sequences, but more DOF helps
    dims = []
    for d in [2048, 4096, 6121, 8192]:
        if d not in dims:
            dims.append(d)

    # Allocate time: 40% to the primary dimension (n0 or 6121), rest to others
    # But first, do a quick exploration at each dimension
    quick_time = 3.0
    candidates = []

    for dim in dims:
        if time.monotonic() > total_deadline - 60:
            break
        # Quick evaluation at this dimension
        f_dim = _interp_to_length(f0, dim)
        sc = _fast_score(f_dim)
        candidates.append((sc, dim, f_dim))
        print(f"  dim={dim}: score={sc:.6f}")

    # Also try asymmetric seeds at the primary dimension
    primary_dim = n0 if n0 in dims else 6121
    for i in range(4):
        if time.monotonic() > total_deadline - 60:
            break
        f_seed = _asymmetric_seed(primary_dim, rng)
        sc = _fast_score(f_seed)
        candidates.append((sc, primary_dim, f_seed))
        print(f"  asymmetric seed {i}: score={sc:.6f}")

    for i in range(3):
        if time.monotonic() > total_deadline - 60:
            break
        f_seed = _two_bump_seed(primary_dim, rng)
        sc = _fast_score(f_seed)
        candidates.append((sc, primary_dim, f_seed))
        print(f"  two-bump seed {i}: score={sc:.6f}")

    candidates.sort(key=lambda x: x[0])
    print(f"\nTop candidates: {[(f'{c[0]:.4f}', c[1]) for c in candidates[:5]]}")

    # Phase 1: Optimize the top 3 candidates
    remaining = total_deadline - time.monotonic() - 120  # reserve 120s for final polish
    n_top = min(3, len(candidates))
    # Allocate time proportional to inverse score (better candidates get more time)
    inv_scores = [1.0 / max(c[0], 1e-10) for c in candidates[:n_top]]
    total_inv = sum(inv_scores)
    time_fracs = [iv / total_inv for iv in inv_scores]
    # Cap: first gets at most 50%, second at most 30%
    time_fracs[0] = min(time_fracs[0], 0.5)
    if n_top > 1:
        time_fracs[1] = min(time_fracs[1], 0.3)
    renormalize = 1.0 - sum(time_fracs[:min(n_top, 2)])
    if n_top > 2:
        time_fracs[2] = max(time_fracs[2], renormalize)

    for idx in range(n_top):
        if time.monotonic() > total_deadline - 100:
            break
        sc, dim, f_start = candidates[idx]
        alloc = max(30.0, remaining * time_fracs[idx])
        alloc = min(alloc, total_deadline - time.monotonic() - 80)
        if alloc < 20:
            break
        print(f"\n=== Optimizing candidate {idx} (dim={dim}, init_score={sc:.6f}, alloc={alloc:.0f}s) ===")
        f_opt, sc_opt = _gradient_optimize(f_start, alloc, rng, f"cand{idx}_d{dim}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  NEW BEST: {best_score:.6f}")

    # Phase 2: Final intensive polish at the best dimension
    best_n = int(best_f.size)
    remaining = total_deadline - time.monotonic() - 30
    if remaining > 40:
        print(f"\n=== Final polish (n={best_n}, budget={remaining:.0f}s) ===")
        # Interpolate to a "nice" dimension if needed
        target_n = best_n
        bp, bs, bc = _coordinate_polish(best_f, _autocorr(best_f), remaining * 0.6, rng, topK_lags=20)
        if bs < best_score:
            best_score = bs
            best_f = bp.copy()
            print(f"  Polish improved: {best_score:.6f}")

        # If time remains, do another gradient pass
        rem2 = total_deadline - time.monotonic() - 20
        if rem2 > 60:
            f2, s2 = _gradient_optimize(best_f, rem2 * 0.7, rng, "final_grad")
            if s2 < best_score:
                best_score = s2
                best_f = f2.copy()

            rem3 = total_deadline - time.monotonic() - 15
            if rem3 > 30:
                bp3, bs3, bc3 = _coordinate_polish(best_f, _autocorr(best_f), rem3, rng, topK_lags=25)
                if bs3 < best_score:
                    best_score = bs3
                    best_f = bp3.copy()

    # Final safety
    best_f = _normalize(best_f)
    best_f = np.nan_to_num(best_f, nan=0.0, posinf=0.0, neginf=0.0)
    np.clip(best_f, 0.0, 1000.0, out=best_f)
    s = float(best_f.sum())
    if s > 0:
        best_f /= s

    best_c = _autocorr(best_f)
    final_score = float(2.0 * best_f.size * best_c.max())
    print(f"\nFINAL internal score = {final_score:.6f}")

    # Try official evaluation
    try:
        out = [float(x) for x in best_f]
        official = evaluate_sequence(out)
        if np.isfinite(official) and official < final_score + 1e-12:
            final_score = float(official)
            print(f"Official score = {final_score:.6f}")
    except Exception:
        pass

    out = [float(x) for x in best_f]
    print(f"Returning sequence of length {len(out)}")
    return out
