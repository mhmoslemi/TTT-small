```python
"""
Step function optimizer using smooth max (log-sum-exp) with β-annealing,
Adam optimizer in log-space, and periodic coordinate descent polish.
Targets non-negative sequences to minimize 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time


def _conv_fft(a, b=None):
    """Linear convolution via FFT. If b is None, computes a*a."""
    if b is None:
        b = a
    n = len(a)
    m = len(b)
    size = 1 << (n + m - 1).bit_length()
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    return np.fft.irfft(A * B, size)[:n + m - 1]


def _score_fast(a):
    """Compute score = 2*n*max(conv(a,a))/(sum(a))^2"""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = _conv_fft(a)
    return float(2.0 * n * c.max() / (s * s))


def _softmax(x, beta=1.0):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(beta * x_shifted)
    return e / e.sum()


def _cross_corr_valid(g, a):
    """
    Compute h[j] = sum_{i=0}^{n-1} g[j+i] * a[i] for j=0,...,n-1
    where g has length 2n-1 and a has length n.
    Uses FFT for O(n log n) complexity.
    """
    n = len(a)
    # h[j] = (a * g_flipped)[2n-2-j]
    g_flipped = g[::-1]
    C = _conv_fft(a, g_flipped)
    # C has length n + (2n-1) - 1 = 3n-2
    # h[j] = C[2n-2-j] for j=0,...,n-1
    # So h = C[n-1 : 2n-1][::-1]
    return C[n-1:2*n-1][::-1]


def _optimize_smooth_max(a_init, deadline, rng, verbose=True):
    """
    Optimize using smooth max with β-annealing and Adam in log-space.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy()
    s = a.sum()
    a = a / s  # normalize to sum=1
    
    # Log-space initialization
    x = np.log(a)
    x_mean = x.mean()
    x = x - x.mean()  # center for numerical stability
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    
    best_score = _score_fast(a)
    best_a = a.copy()
    
    # β annealing schedule
    num_phases = 8
    beta_start = 3.0
    beta_end = 200.0
    total_steps = 2000
    steps_per_phase = total_steps // num_phases
    
    lr_start = 0.05
    lr_end = 0.001
    
    all_betas = np.geomspace(beta_start, beta_end, num_phases)
    all_lrs = np.geomspace(lr_start, lr_end, num_phases)
    
    for phase in range(num_phases):
        if time.monotonic() > deadline:
            break
        beta = all_betas[phase]
        lr = all_lrs[phase]
        
        if verbose:
            print(f"  Phase {phase}: beta={beta:.1f}, lr={lr:.4f}")
        
        for step in range(steps_per_phase):
            if time.monotonic() > deadline:
                break
            
            t += 1
            
            # Recover a from log-space
            x_norm = x - x.max()
            a = np.exp(x_norm)
            a = a / a.sum()
            
            # Compute convolution
            c = _conv_fft(a)
            
            # Smooth max and its gradient
            c_shifted = c - c.max()
            beta_c = beta * c_shifted
            exp_beta_c = np.exp(beta_c)
            sum_exp = exp_beta_c.sum()
            g = exp_beta_c / sum_exp  # softmax weights
            
            # Gradient: grad_a = 2 * cross_corr_valid(g, a)
            grad_a = 2.0 * _cross_corr_valid(g, a)
            
            # Transform to log-space gradient
            # da/dx_j = a_j (δ_kj - a_j) for softmax
            # grad_x_j = a_j * (grad_a_j - sum_k a_k * grad_a_k)
            grad_a_dot_a = np.dot(grad_a, a)
            grad_x = a * (grad_a - grad_a_dot_a)
            
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * grad_x ** 2
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
            
            # Periodic evaluation
            if step % 50 == 0 or step == steps_per_phase - 1:
                score = 2.0 * n * c.max()  # since sum(a) = 1
                if score < best_score - 1e-10:
                    best_score = score
                    best_a = a.copy()
            
            # Reset Adam state at phase boundaries for freshness
            if step == steps_per_phase - 1:
                m *= 0.5
                v *= 0.5
    
    # Final score
    a_final = np.exp(x - x.max())
    a_final = a_final / a_final.sum()
    final_score = _score_fast(a_final)
    if final_score < best_score:
        best_score = final_score
        best_a = a_final.copy()
    
    return best_a, best_score


def _optimize_active_set(a_init, deadline, rng, K=5, verbose=True):
    """
    Active set optimization: focus on the top-K peaks of the convolution.
    Uses gradient descent in log-space with Adam.
    """
    n = len(a_init)
    a = np.maximum(a_init, 1e-30).copy()
    a = a / a.sum()
    
    x = np.log(a) - np.log(a).max()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    
    best_score = _score_fast(a)
    best_a = a.copy()
    
    total_steps = 3000
    lr = 0.03
    
    for step in range(total_steps):
        if time.monotonic() > deadline:
            break
        t += 1
        
        # Decay learning rate
        lr_eff = lr * (1.0 - step / total_steps) ** 0.5 + 0.001
        
        # Recover a
        x_norm = x - x.max()
        a = np.exp(x_norm)
        a = a / a.sum()
        
        # Compute convolution
        c = _conv_fft(a)
        
        # Find top-K lags
        top_idx = np.argpartition(c, -K)[-K:]
        top_vals = c[top_idx]
        
        # Weights: proportional to how much each peak exceeds the mean
        excess = np.maximum(top_vals - c.mean(), 0)
        if excess.sum() < 1e-15:
            w = np.ones(K) / K
        else:
            w = excess / excess.sum()
        
        # Gradient at top lags
        # grad[j] = 2 * sum_k w[k] * a[k_idx - j] (for valid k_idx - j)
        grad_a = np.zeros(n)
        for ki, wi in zip(top_idx, w):
            # a[ki - j] is valid for j in [max(0, ki-n+1), min(n-1, ki)]
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            if j_start <= j_end:
                idx = ki - np.arange(j_start, j_end + 1)
                grad_a[j_start:j_end+1] += 2.0 * wi * a[idx]
        
        # Log-space gradient
        grad_a_dot_a = np.dot(grad_a, a)
        grad_x = a * (grad_a - grad_a_dot_a)
        
        # Adam
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * grad_x ** 2
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        
        if step % 100 == 0:
            score = 2.0 * n * c.max()
            if score < best_score - 1e-10:
                best_score = score
                best_a = a.copy()
    
    a_final = np.exp(x - x.max())
    a_final = a_final / a_final.sum()
    final_score = _score_fast(a_final)
    if final_score < best_score:
        best_score = final_score
        best_a = a_final.copy()
    
    return best_a, best_score


def _polish(a_init, deadline, rng, verbose=True):
    """
    Coordinate descent polish: try to reduce the max by adjusting individual elements.
    """
    n = len(a_init)
    a = np.maximum(a_init, 0.0).copy()
    a = a / a.sum()
    
    best_score = _score_fast(a)
    best_a = a.copy()
    c = _conv_fft(a)
    peak = c.max()
    peak_idx = c.argmax()
    
    n_attempts = 50000
    for attempt in range(n_attempts):
        if time.monotonic() > deadline:
            break
        
        # Focus on the peak lag
        # For lag k, conv[k] = sum_i a[i]*a[k-i]
        # To reduce conv[peak_idx], we want to reduce the product a[i]*a[peak_idx-i]
        # Strategy: find the pair (i, peak_idx-i) with the largest product and reduce one
        
        k = peak_idx
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        
        if j_start > j_end:
            break
            
        # Products at this lag
        products = np.maximum(a[j_start:j_end+1] * a[k - np.arange(j_start, j_end+1)], 0)
        
        if products.max() < 1e-20:
            break
        
        # Find the largest product pair
        local_max_idx = np.argmax(products)
        i_don = j_start + local_max_idx  # index in a to reduce
        i_part = k - i_don  # its partner
        
        if i_part < 0 or i_part >= n or i_part == i_don:
            # Can't reduce (partner is out of range or same element)
            # Try random perturbation
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            if i1 == i2:
                continue
            amt = a[i1] * 0.01
            a_try = a.copy()
            a_try[i1] -= amt
            a_try[i2] += amt
            a_try = np.maximum(a_try, 0)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try /= s
            c_try = _conv_fft(a_try)
            if c_try.max() < peak - 1e-12:
                a = a_try
                c = c_try
                peak = c.max()
                peak_idx = c.argmax()
                score = 2.0 * n * peak
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
            continue
        
        # Try reducing a[i_don] and increasing some other element
        # Find a good receiver: an element with low total contribution to peaks
        # Simple: random receiver
        for _ in range(3):
            i_rec = int(rng.integers(0, n))
            if i_rec == i_don:
                continue
            for frac in [0.05, 0.1, 0.2]:
                amt = a[i_don] * frac
                a_try = a.copy()
                a_try[i_don] -= amt
                a_try[i_rec] += amt
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try /= s
                c_try = _conv_fft(a_try)
                if c_try.max() < peak - 1e-12:
                    a = a_try
                    c = c_try
                    peak = c.max()
                    peak_idx = c.argmax()
                    score = 2.0 * n * peak
                    if score < best_score:
                        best_score = score
                        best_a = a.copy()
                    break
    
    return best_a, best_score


def _perturb(a, rng, strength=0.1):
    """Apply a random perturbation to a."""
    n = len(a)
    a_try = a.copy()
    
    mode = int(rng.integers(0, 4))
    if mode == 0:
        # Gaussian noise
        a_try += rng.normal(0, strength * a.mean(), size=n)
    elif mode == 1:
        # Sinusoidal modulation
        freq = int(rng.integers(1, n // 3 + 1))
        phase = rng.uniform(0, 2 * np.pi)
        j = np.arange(n)
        a_try *= (1 + strength * np.sin(2 * np.pi * freq * j / n + phase))
    elif mode == 2:
        # Local swap
        i1 = int(rng.integers(0, n // 2))
        i2 = int(rng.integers(n // 2, n))
        amt = a[i1] * strength
        a_try[i1] -= amt
        a_try[i2] += amt
    else:
        # Random rescale of a random block
        start = int(rng.integers(0, n - 10))
        length = int(rng.integers(5, min(50, n - start)))
        factor = rng.uniform(0.5, 1.5)
        a_try[start:start+length] *= factor
    
    a_try = np.maximum(a_try, 0)
    s = a_try.sum()
    if s < 1e-10:
        return a.copy()
    return a_try / s
