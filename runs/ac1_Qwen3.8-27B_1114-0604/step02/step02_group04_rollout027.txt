import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _gradient(a, w):
    """Compute gradient of weighted sum of convolution values.
    
    grad[j] = 2 * sum_{l=0}^{n-1} w[j+l] * a[l]
    
    Uses FFT cross-correlation.
    """
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N) if w.size <= N else np.fft.rfft(w[:N], N)
    # Cross-correlation: R[j] = sum_l w[j+l] * a[l]
    # = ifft(fft(w, N) * conj(fft(a, N)))[j]
    r = np.fft.irfft(fw * np.conj(fa), N)
    grad = 2.0 * r[:n]
    return grad


def _make_multi_sine(n, freqs, amps, phases):
    """Create multi-frequency sinusoidal sequence."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)

    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / n
    return a / s

    s = a.sum()
    if s < 1e-10:

    it = 0
    last_improve = 0

    while time.monotonic() < deadline:
        it += 1
        t_frac = min(1.0, it / max(1.0, patience * 3))

        c = _fft_convolve(a)
        if c.size == 0:
            break
        
        # Compute weights

            # Use a small epsilon to avoid numerical issues

        else:

            c_min = np.min(c)

"""
Low-peak step-function optimization via multi-frequency initialization,
adaptive mirror descent with exact subgradient polishing, and local refinement.
Uses FFT-based gradients, diverse restarts, and multiple sequence lengths.
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    """Compute the evaluation score (scale-invariant)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _gradient(a, w):
    """
    Compute gradient: grad[j] = 2 * sum_{l=0}^{n-1} w[j+l] * a[l]
    Using FFT cross-correlation.
    """
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]


def _make_sine(n, amp, phase=0.0):
    """Single-mode sinusoidal: a[i] = (1 + amp*cos(2*pi*(i+phase)/n))/n."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.cos(theta)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / n
    return a / s


def _make_multi_sine(n, freqs, amps, phases):
    """Multi-frequency: a[i] = (1 + sum_k A_k*cos(2*pi*f_k*i/n + phi_k))/n."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64) / n
    for f, A, ph in zip(freqs, amps, phases):
        a += (A / n) * np.cos(2.0 * np.pi * f * i / n + ph)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / n
    return a / s


def _make_random(n, rng, shape='exp'):
    """Generate a random nonnegative sequence."""
    if shape == 'exp':
        a = rng.exponential(1.0, size=n)
    elif shape == 'lognormal':
        a = np.exp(rng.standard_normal(n) * 0.5)
    else:
        a = rng.uniform(0.1, 1.0, size=n)
    s = a.sum()
    return a / s


def _optimize_soft(a, deadline, T_start=3.0, T_end=60.0, lr0=None,
                   patience=2500, rng_seed=42):
    """
    Soft-max mirror descent with adaptive parameters and restarts.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(a.size) / a.size
        s = 1.0
    a = a / s

    n = a.size
    if lr0 is None:
        lr0 = 0.18 * n

    best_a = a.copy()
    best_score = _score(best_a)

    it = 0
    last_improve = 0
    rng = np.random.RandomState(rng_seed)
    eval_count = 0

    while time.monotonic() < deadline:
        it += 1

        t_frac = min(1.0, it / (patience * 3.0 + 1))
        T = T_start + (T_end - T_start) * t_frac
        lr = lr0 / (1.0 + 0.003 * it)

        c = _fft_convolve(a)
        if c.size == 0:
            break

        # Soft-max weights
        c_max = float(np.max(c))
        c_min = float(np.min(c))
        rng_c = c_max - c_min
        if rng_c < 1e-15:
            w = np.ones(c.size) / c.size
        else:
            c_norm = (c - c_min) / rng_c
            c_shift = c_norm - 1.0  # max is 0
            exp_c = np.exp(T * c_shift)
            s_exp = float(exp_c.sum())
            if s_exp < 1e-300:
                w = np.zeros(c.size)
                w[np.argmax(c)] = 1.0
            else:
                w = exp_c / s_exp

        grad = _gradient(a, w)
        grad_proj = grad - np.mean(grad)

        update = -lr * grad_proj
        update = np.clip(update, -6.0, 6.0)
        a_new = a * np.exp(update)
        a_new = np.where(np.isfinite(a_new), a_new, 0.0)
        a_new = np.clip(a_new, 0.0, 1000.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a = best_a.copy()
            break
        a = a_new / s_new

        # Evaluate periodically
        eval_count += 1
        if eval_count % 4 == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                # Restart with perturbation
                a = best_a.copy()
                strategy = rng.randint(0, 3)
                if strategy == 0:
                    noise = rng.standard_normal(n) * 0.01 * a.mean()
                    a = a + noise
                elif strategy == 1:
                    # Perturb the top elements most
                    order = np.argsort(a)
                    top_k = max(1, n // 10)
                    a[order[-top_k:]] *= (1.0 + rng.standard_normal(top_k) * 0.02)
                    a[order[:top_k]] *= (1.0 + rng.standard_normal(top_k) * 0.02)
                else:
                    # Frequency-domain perturbation
                    fa = np.fft.rfft(a)
                    fa[1:len(fa)//2] *= (1.0 + rng.standard_normal(len(fa)-1).real * 0.01)
                    a = np.fft.irfft(fa, n)
                a = np.clip(a, 0, 1000)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                patience = max(800, int(patience * 1.15))
                lr0 = max(0.03 * n, lr0 * 0.85)

    return best_a, best_score

        a = np.ones(a.size) / a.size
        s = 1.0
    a = a / s

    n = a.size
    if lr0 is None:
        lr0 = 0.03 * n

    best_a = a.copy()
    best_score = _score(best_a)

    it = 0
    last_improve = 0
    rng = np.random.RandomState(rng_seed)
    consecutive_bad = 0

    while time.monotonic() < deadline:
        it += 1
        lr = lr0 / (1.0 + 0.001 * it)

        c = _fft_convolve(a)
        if c.size == 0:
            break

        max_c = float(np.max(c))

        # Exact argmax with small tolerance
        tol = max_c * 1e-12 + 1e-18
        mask = c >= max_c - tol
        w = np.zeros(c.size)
        w[mask] = 1.0
        w /= w.sum()

        grad = _gradient(a, w)
        grad_proj = grad - np.mean(grad)

        # Backtracking-like: try the step, if it doesn't help, use half
        update = -lr * grad_proj
        update = np.clip(update, -3.0, 3.0)
        a_new = a * np.exp(update)
        a_new = np.where(np.isfinite(a_new), a_new, 0.0)
        a_new = np.clip(a_new, 0.0, 1000.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a = best_a.copy()
            break
        a_candidate = a_new / s_new

        sc = _score(a_candidate)
        if np.isfinite(sc) and sc < best_score - 1e-14:
            a = a_candidate
            best_score = sc
            best_a = a.copy()
            last_improve = it
            consecutive_bad = 0
        else:
            # Try smaller step
            update = -lr * 0.3 * grad_proj
            update = np.clip(update, -1.5, 1.5)
            a_new = a * np.exp(update)
            a_new = np.where(np.isfinite(a_new), a_new, 0.0)
            a_new = np.clip(a_new, 0.0, 1000.0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_candidate = a_new / s_new
                sc2 = _score(a_candidate)
                if np.isfinite(sc2) and sc2 < best_score - 1e-14:
                    a = a_candidate
                    best_score = sc2
                    best_a = a.copy()
                    last_improve = it
                    consecutive_bad = 0
                else:
                    a = a_candidate  # accept anyway to move
                    consecutive_bad += 1
            else:
                consecutive_bad += 1

        if it - last_improve > patience:
            # Perturb and restart
            a = best_a.copy()
            scale = rng.uniform(0.002, 0.008)
            noise = rng.standard_normal(n) * scale * a.mean()
            a = a + noise
            a = np.clip(a, 0, 1000)
            s = a.sum()
            if s > 1e-10:
                a = a / s
            else:
                a = best_a.copy()
            patience = max(500, int(patience * 1.1))
            lr0 = max(0.01 * n, lr0 * 0.9)

    return best_a, best_score


def _top_k_optimize(a, deadline, K=6, lr0=None, patience=1500, rng_seed=77):
    """
    Top-K peak optimization: focus on the K largest convolution values.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(a.size) / a.size
        s = 1.0
    a = a / s

    n = a.size
    if lr0 is None:
        lr0 = 0.08 * n

    best_a = a.copy()
    best_score = _score(best_a)

    it = 0
    last_improve = 0
    rng = np.random.RandomState(rng_seed)

    while time.monotonic() < deadline:
        it += 1
        lr = lr0 / (1.0 + 0.002 * it)

        c = _fft_convolve(a)
        if c.size == 0:
            break

        # Top-K peaks with soft weighting
        if c.size <= K:
            top_idx = np.arange(c.size)
            top_vals = c
        else:
            top_idx = np.argpartition(c, -K)[-K:]
            top_vals = c[top_idx]

        # Soft weights within top-K
        T_local = 5.0
        shift = top_vals - top_vals.max()
        weights = np.exp(T_local * shift)
        sw = weights.sum()
        if sw < 1e-300:
            weights = np.ones(K) / K
        else:
            weights /= sw

        w = np.zeros(c.size)
        for i, k in enumerate(top_idx):
            w[k] = weights[i]

        grad = _gradient(a, w)
        grad_proj = grad - np.mean(grad)

        update = -lr * grad_proj
        update = np.clip(update, -4.0, 4.0)
        a_new = a * np.exp(update)
        a_new = np.where(np.isfinite(a_new), a_new, 0.0)
        a_new = np.clip(a_new, 0.0, 1000.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a = best_a.copy()
            break
        a = a_new / s_new

        if it % 3 == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.005 * a.mean()
                a = a + noise
                a = np.clip(a, 0, 1000)
                s = a.sum()
                if s > 1e-10:
                    a = a / s
                else:
                    a = best_a.copy()
                patience = max(600, int(patience * 1.2))
                lr0 = max(0.02 * n, lr0 * 0.85)

    return best_a, best_score
