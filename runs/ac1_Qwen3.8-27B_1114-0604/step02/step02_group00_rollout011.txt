```python
"""
LP-projection + LSE-LBFGS hybrid optimizer for minimal peak autocorrelation sequences.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy:
  - Phase 1: LP-based projection method (principled descent on the peak)
  - Phase 2: LSE-smoothed L-BFGS-B polish with continuation
  - Phase 3: Multi-length exploration and local search
"""

import numpy as np
from scipy.optimize import minimize, linprog
import time

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def lse_objective(a, beta):
    return lse_score_grad(a, beta)

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def build_conv_matrix(n, f):
    """Build the (2n-1) x n convolution constraint matrix A where A[k,i] = f[k-i] if valid."""
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A[k, j] = f[k - j]
    return A
```

```python
"""
LP-Projection + LSE-LBFGS hybrid optimizer for minimal peak autocorrelation.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Algorithm:
  Phase 1: LP-based projection method for principled peak reduction
  Phase 2: LSE-smoothed L-BFGS-B continuation for local polish
  Phase 3: Multi-start local search with targeted perturbations
"""

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import coo_matrix
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def precompute_conv_indices(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def solve_lp_projection(f, M, n, row_idx, col_idx, time_limit=30):
    vals = f[row_idx - col_idx]
    A = coo_matrix((vals, (row_idx, col_idx)), shape=(2 * n - 1, n)).tocsr()
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, 10.0)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                  options={'time_limit': time_limit, 'presolve': True})
    if res.success:
        g = res.x
        g_sum = float(g.sum())
        if g_sum > 1e-12:
            g = g / g_sum  # normalize to sum 1
            return g
    return None


def lp_iteration(f, n, row_idx, col_idx, t=0.4, time_limit=30):
    c = fast_conv(f)
    M = float(c.max())
    g = solve_lp_projection(f, M, n, row_idx, col_idx, time_limit)
    if g is None:
        return f, M
    f_new = (1.0 - t) * f + t * g
    f_new = np.clip(f_new, 0.0, 1000.0)
    f_new = f_new / f_new.sum()  # renormalize
    return f_new, M


def lp_phase(f_init, n, max_iters=200, time_limit_per_lp=30):
    row_idx, col_idx = precompute_conv_indices(n)
    f = f_init.astype(np.float64).copy()
    f = np.clip(f, 0.0, None)
    if f.sum() < 1e-12:
        f = np.ones(n) / n
    else:
        f = f / f.sum()
    
    best_f = f.copy()
    best_M = float(fast_conv(f).max())
    
    for it in range(max_iters):
        f, M = lp_iteration(f, n, row_idx, col_idx, t=0.4, time_limit=time_limit_per_lp)
        if M < best_M:
            best_M = M
            best_f = f.copy()
        if it % 10 == 0:
            print(f"    LP iter {it}: M = {M:.8f}, best_M = {best_M:.8f}")
    
    return best_f, best_M
```

Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.
"""

def fast_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(scaled_max + exp_scaled.sum()) if False else np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, gtol=1e-13):
    """Run L-BFGS-B on LSE objective."""
    n = len(a0)

def downsample(a, new_n):
    """Downsample sequence to new length using averaging."""
    n = len(a)
    if new_n >= n:
        # Upsample by interpolation
        x_old = np.linspace(0, 1, n)
        x_new = np.linspace(0, 1, new_n)
        return np.interp(x_new, x_old, a)
    # Downsample by averaging blocks
    block_size = n // new_n
    result = np.zeros(new_n)
    for i in range(new_n):
        start = i * block_size
        end = start + block_size if i < new_n - 1 else n
        result[i] = a[start:end].mean()
    return result
