def frequency_equalize(a, strength=0.1):
    """
    Apply frequency-domain equalization to flatten the power spectrum.
    This helps make the self-convolution flatter.
    """
    n = len(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    power = np.abs(A) ** 2
    
    # Target: make power more uniform
    # Reduce high-power bins, boost low-power bins
    p_mean = power.mean()
    p_max = power.max()
    
    if p_max < 1e-15:
        return a
    
    # Multiplicative correction in frequency domain
    # A_new = A * (1 - strength * (power - p_mean) / p_max)
    # This reduces peaks in the spectrum
    correction = 1.0 - strength * (power - p_mean) / (p_max + 1e-15)
    correction = np.maximum(correction, 0.0)  # Don't flip phase
    A_new = A * correction
    
    a_new = np.fft.irfft(A_new, M_fft)[:n]
    return np.maximum(a_new, 0.0)

def adam_step(a, g, m, v, t, lr):
    """Adam optimizer step (in-place)."""
    m[:] = 0.9 * m + 0.1 * g
    v[:] = 0.999 * v + 0.001 * g * g
    t1 = 0.9 ** t
    t2 = 0.999 ** t
    a -= lr * (m / (1.0 - t1)) / (np.sqrt(v / (1.0 - t2)) + 1e-8)


def generate_initializations(n, rng):

"""
Aggressive multi-strategy optimizer for minimizing 2n*max(conv(a,a))/(sum a)^2.
Uses soft-max gradient over all convolution peaks, frequency-domain equalization,
adaptive temperature scheduling, structural mutations, and multi-length exploration.
"""

import numpy as np
import time


def fast_self_conv(a):
    """Full self-convolution via FFT, length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, c, S):
    """Objective: 2n * max(conv) / S^2."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def softmax_gradient_full(a, c, temperature):
    """
    Compute gradient using soft-max weighting over ALL convolution peaks.
    Returns (gradient, weighted_max, sum).
    """
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S

    c_max = float(np.max(c))
    c_min = float(np.min(c))
    c_range = c_max - c_min
    if c_range < 1e-15:
        return np.zeros(n), c_max, S

    temp = temperature * c_range
    if temp < 1e-15:
        temp = 1e-15

    # Soft-max weights over all 2n-1 positions
    shifted = (c - c_max) / temp
    shifted = np.maximum(shifted, -50.0)  # prevent underflow
    wts = np.exp(shifted)
    wts_sum = wts.sum()
    if wts_sum < 1e-30:
        wts = np.zeros(c.shape)
        wts[int(np.argmax(c))] = 1.0
    else:
        wts = wts / wts_sum

    M = float(np.dot(c, wts))

    # Compute grad_M[j] = 2 * sum_m a[m] * wts[j+m]
    # This is 2 * cross_corr(a, wts)[j]
    # cross_corr(a, wts)[j] = sum_{m=0}^{n-1} a[m] * wts[m+j]
    # FFT size needs to be >= n + (2n-1) - 1 = 3n - 2
    L = 1 << (3 * n - 2 - 1).bit_length()
    a_padded = np.zeros(L)
    a_padded[:n] = a
    wts_padded = np.zeros(L)
    wts_padded[:2 * n - 1] = wts

    A_f = np.fft.rfft(a_padded)
    W_f = np.fft.rfft(wts_padded)
    cross = np.fft.irfft(np.conj(A_f) * W_f, L)
    grad_M = 2.0 * cross[:n]

    # Full gradient: d/d a[j] of (2n * M / S^2)
    grad = 2.0 * n * (grad_M / S2 - 2.0 * M / S3)
    return grad, M, S


def frequency_equalize(a, strength=0.1):
    """
    Flatten the power spectrum in frequency domain to promote flatter autocorrelation.
    Returns a new non-negative array.
    """
    n = len(a)
    M_fft = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    power = np.abs(A) ** 2
    p_mean = power.mean()
    p_max = power.max()

    if p_max < 1e-15:
        return a.copy()

    # Reduce high-power frequency components
    correction = 1.0 - strength * (power - p_mean) / (p_max + 1e-15)
    correction = np.maximum(correction, 0.0)
    A_new = A * correction

    a_new = np.fft.irfft(A_new, M_fft)[:n]
    return np.maximum(a_new, 0.0)


def adam_step(a, g, m, v, t, lr):
    """Adam optimizer step (in-place on a, m, v)."""
    m[:] = 0.9 * m + 0.1 * g
    v[:] = 0.999 * v + 0.001 * g * g
    bc1 = 1.0 - 0.9 ** t
    bc2 = 1.0 - 0.999 ** t
    a -= lr * (m / bc1) / (np.sqrt(v / bc2) + 1e-8)


def generate_initializations(n, rng):
    """Generate diverse initial guesses."""
    inits = []
    inits.append(np.ones(n))
    x = np.linspace(0, 1, n, endpoint=False)
    for freq in [1, 2, 3, 5, 8]:
        inits.append(np.maximum(1.0 + 0.5 * np.cos(2.0 * np.pi * freq * x), 0.01))
    inits.append(np.maximum(1.0 + 0.4 * np.cos(2.0 * np.pi * 7.3 * x) + 0.3 * np.sin(2.0 * np.pi * 3.1 * x), 0.01))
    inits.append(np.maximum(0.5 + 0.5 * rng.random(n), 0.01))
    a_g = np.ones(n)
    a_g[1::2] = 0.15
    inits.append(a_g.copy())
    a_g2 = np.ones(n)
    a_g2[1::3] = 0.1
    inits.append(a_g2.copy())
    # Exponential decay
    inits.append(np.maximum(np.exp(-np.linspace(0, 3, n)), 0.01))
    return inits


def restart_mutation(a_best, rng, strategy_idx):
    """Apply a structural mutation to escape local minima."""
    n = a_best.size
    a = a_best.copy()
    strat = strategy_idx % 5
    if strat == 0:
        # Gaussian noise
        sc = 0.05 + 0.15 * rng.random()
        a = a * (1.0 + sc * rng.standard_normal(n))
    elif strat == 1:
        # Frequency equalization + noise
        a = frequency_equalize(a, strength=0.4)
        a += 0.05 * rng.standard_normal(n)
    elif strat == 2:
        # Block swap
        block_size = max(16, n // 8)
        for _ in range(2):
            i = rng.integers(0, n - 2 * block_size)
            j = rng.integers(i + block_size, n - block_size)
            a[i:i+block_size], a[j:j+block_size] = a[j:j+block_size].copy(), a[i:i+block_size].copy()
    elif strat == 3:
        # Scale + shift
        a = a * (0.5 + 0.5 * rng.random())
        a += 0.05 * rng.standard_normal(n)
    else:
        # Random subsequence replacement
        block_size = max(16, n // 16)
        start = rng.integers(0, n - block_size)
        a[start:start+block_size] = rng.random(block_size) * a_best[start:start+block_size].mean()
    return np.clip(a, 0.0, 1000.0)


def optimize_core(a0, end_time, rng, init_lr, patience, freq_eq_interval, temp_start, temp_end):
    """
    Core optimization loop with soft-max gradient, Adam, frequency equalization,
    and adaptive restarts. Returns (best_score, best_array).
    """
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    # Normalize to sum = n for numerical stability
    s_init = a.sum()
    if s_init > 1e-10:
        a = a * (n / s_init)

    c = fast_self_conv(a)
    S = float(a.sum())
    score = compute_score(n, c, S)
    best_score = score
    best_a = a.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = init_lr
    no_imp = 0
    iter_count = 0
    cur_patience = patience
    mutation_count = 0

    while time.monotonic() < end_time:
        iter_count += 1
        S = float(a.sum())
        if S < 1e-10:
            a = best_a + 0.01 * rng.standard_normal(n)
            a = np.clip(a, 0.0, 1000.0)
            S = float(a.sum())

        c = fast_self_conv(a)
        score = compute_score(n, c, S)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        # Restart check
        if no_imp >= cur_patience:
            mutation_count += 1
            a = restart_mutation(best_a, rng, mutation_count)
            s_new = a.sum()
            if s_new > 1e-10:
                a = a * (n / s_new)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            lr = max(0.00005, lr * 0.4)
            no_imp = 0
            cur_patience = max(30, cur_patience * 2 // 3)
            continue

        # Adaptive temperature: start broad, narrow over time
        progress = min(1.0, float(no_imp) / float(cur_patience))
        global_progress = min(1.0, float(iter_count) / float(patience * 10))
        temperature = temp_end + (temp_start - temp_end) * (1.0 - global_progress)
        # Add some jitter
        temperature *= (0.8 + 0.4 * rng.random())

        g, M, S = softmax_gradient_full(a, c, temperature)
        t += 1
        adam_step(a, g, m, v, t, lr)
        a = np.clip(a, 0.0, 1000.0)

        # Periodic normalization to prevent drift
        if iter_count % 500 == 0:
            s_cur = a.sum()
            if s_cur > 1e-10 and (s_cur > 10 * n or s_cur < 0.1 * n):
                a = a * (n / s_cur)

        # Periodic frequency equalization
        if freq_eq_interval > 0 and iter_count % freq_eq_interval == 0:
            a_eq = frequency_equalize(a, strength=0.03)
            a = 0.95 * a + 0.05 * a_eq
            a = np.clip(a, 0.0, 1000.0)

    return best_score, best_a
