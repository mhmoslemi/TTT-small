import numpy as np
import time


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _fft_conv(a, b, L=None):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if L is None:
        L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout]


def _score(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    a = a / s
    n = len(a)
    L = _next_pow2(2 * n - 1)
    c = _fft_conv(a, a, L)
    return float(2.0 * n * c.max())


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s


def _gradient_and_score(p, beta, L=None):
    """Compute smoothed-max score and its gradient wrt p."""
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)
    Lg = _next_pow2(3 * n)
    c = _fft_conv(p, p, L)
    d = n * c
    exact_score = float(2.0 * d.max())
    if beta < 1e-8:
        w = np.ones(len(d)) / len(d)
    else:
        dm = d.max()
        e = np.exp(beta * (d - dm))
        sw = e.sum()
        if sw < 1e-300 or not np.isfinite(sw):
            w = np.ones(len(d)) / len(d)
        else:
            w = e / sw
    # grad[i] = 2n * sum_m w[i+m] * p[m] for m=0..n-1 (where i+m < 2n-1)
    Apad = np.zeros(Lg)
    Apad[:n] = p
    Wpad = np.zeros(Lg)
    Wpad[:len(w)] = w
    R = np.fft.irfft(np.fft.rfft(Apad) * np.conj(np.fft.rfft(Wpad)), Lg)[:n]
    grad = 2.0 * n * R
    grad = np.where(np.isfinite(grad), grad, 0.0)
    grad = np.maximum(grad, 0.0)
    return exact_score, grad


def _freq_flatten_A2(p, gamma, L=None):
    """
    Modify A^2 in frequency domain to flatten the convolution.
    gamma > 0 controls how much we compress the dynamic range of A^2.
    """
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)
    Apad = np.zeros(L)
    Apad[:n] = p
    A = np.fft.rfft(Apad)
    A2 = A * A
    A2_real = np.real(A2)
    # Work with the magnitude and phase separately
    mag = np.abs(A2)
    ph = np.angle(A2)
    # Compress dynamic range: mag_new = mag_max * (mag/mag_max)^gamma
    # gamma < 1 compresses (makes flatter), gamma > 1 expands
    mag_max = mag.max()
    if mag_max < 1e-15:
        return p.copy()
    mag_norm = mag / mag_max
    mag_new = mag_max * np.power(mag_norm, gamma)
    A2_new = mag_new * np.exp(1j * ph)
    # Take square root to get new A
    A_new = np.sqrt(A2_new.astype(complex))
    a_new = np.fft.irfft(A_new, L)[:n]
    neg_frac = (a_new < 0).sum() / n
    if neg_frac > 0.4:
        return None
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s


def _freq_flatten_C(p, alpha, L=None):
    """Flatten the convolution C directly in time domain, then recover a."""
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)
    c = _fft_conv(p, p, L)
    cmax = c.max()
    if cmax < 1e-15:
        return p.copy()
    c_norm = c / cmax
    c_new = cmax * np.power(c_norm, alpha)
    c_new = np.maximum(c_new, 0.0)
    C_freq = np.fft.rfft(c_new, L)
    A_new = np.sqrt(C_freq.astype(complex))
    a_new = np.fft.irfft(A_new, L)[:n]
    neg_frac = (a_new < 0).sum() / n
    if neg_frac > 0.35:
        return None
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s


def _peak_shave(p, n_peaks=5, strength=0.1, L=None):
    """Directly reduce the top peaks of the convolution."""
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)
    c = _fft_conv(p, p, L)
    # Find top peak positions
    top_idx = np.argsort(c)[::-1][:n_peaks]
    # For each peak at position k, the gradient direction is 2*a[k-i] for i in [0,n-1]
    # We want to reduce c[k], so we decrease p in the direction that reduces c[k]
    # d(c[k])/d(p[i]) = 2*p[k-i] (for valid k-i)
    # To reduce c[k], we want to decrease p[i] where p[k-i] is large
    # Simple approach: scale down p[i] proportionally to how much it contributes to peaks
    reduction = np.zeros(n)
    for k in top_idx:
        # p[i] contributes to c[k] via p[k-i]
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            reduction[i] += p[k - i] if (0 <= k - i < n) else 0.0
    reduction = np.maximum(reduction, 0.0)
    rmax = reduction.max()
    if rmax < 1e-15:
        return p.copy()
    reduction_norm = reduction / rmax
    p_new = p * (1.0 - strength * reduction_norm)
    p_new = np.maximum(p_new, 1e-15)
    s = p_new.sum()
    if s < 1e-10:
        return p.copy()
    return p_new / s


def _block_optimize(p, n_blocks=20, n_trials=3, L=None):
    """Try scaling random blocks to reduce the score."""
    n = len(p)
    best = p.copy()
    best_sc = _score(p)
    bs = max(2, n // n_blocks)
    for _ in range(n_trials):
        start = int(np.random.randint(0, n - bs + 1))
        for f in (0.5, 0.7, 1.0 / 0.7, 1.0 / 0.5, 0.3, 3.0):
            p_try = best.copy()
            p_try[start:start + bs] *= f
            p_try = _sanitize(p_try)
            if p_try is None:
                continue
            sc = _score(p_try)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best = p_try
    return best, best_sc


def _asymmetric_init(n, rng):
    """Initialize with mass concentrated asymmetrically (helps convolution vs autocorrelation)."""
    # Strategy: put more mass on one side, less on the other
    # This makes a and flip(a) different, reducing c[n-1]
    frac = float(rng.uniform(0.25, 0.45))
    n_left = int(n * frac)
    n_right = n - n_left
    p = np.zeros(n)
    # Left side: broad peak
    x = np.linspace(0, 1, n_left)
    p[:n_left] = np.exp(-0.5 * ((x - 0.3) / 0.25) ** 2)
    # Right side: broader, lower peak
    x = np.linspace(0, 1, n_right)
    p[n_left:] = 0.5 * np.exp(-0.5 * ((x - 0.7) / 0.3) ** 2)
    p = np.maximum(p, 1e-10)
    # Add some noise
    p = p * np.exp(rng.normal(0, 0.1, n))
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _bimodal_init(n, rng):
    """Bimodal initialization: two peaks at the ends."""
    width = int(n * rng.uniform(0.1, 0.3))
    p = np.zeros(n)
    p[:width] = 1.0
    p[-width:] = float(rng.uniform(0.5, 1.5))
    # Smooth the edges
    kernel = np.ones(5) / 5
    p = np.convolve(p, kernel, mode='same')
    p = np.maximum(p, 1e-10)
    p = p * np.exp(rng.normal(0, 0.05, n))
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _ramp_init(n, rng):
    """Monotonically decreasing initialization."""
    alpha = float(rng.uniform(1.0, 4.0))
    t = np.arange(n) / n
    p = np.exp(-alpha * t)
    p = p * np.exp(rng.normal(0, 0.05, n))
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _optimize_main(p0, rng, time_budget, beta_start=30.0, beta_end=5000.0, verbose=False):
    """Main hybrid optimizer."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    L = _next_pow2(2 * n - 1)
    best = p.copy()
    best_score = _score(p)
    cur_score = best_score
    beta = float(beta_start)
    alpha = 0.15
    deadline = time.monotonic() + time_budget
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        r = rng.random()

        if r < 0.50:
            # Gradient step
            sc, h = _gradient_and_score(p, beta, L)
            hmax = h.max()
            if hmax < 1e-15:
                stall += 1
                continue
            h_norm = h / hmax
            improved = False
            for a_try in (alpha * 0.5, alpha, alpha * 1.5, alpha * 2.0):
                a_try = float(np.clip(a_try, 0.005, 5.0))
                p_try = p * np.exp(-a_try * h_norm)
                p_try = np.maximum(p_try, 1e-18 / n)
                s_try = p_try.sum()
                if s_try < 1e-10:
                    continue
                p_try = p_try / s_try
                sc_try = _score(p_try)
                if sc_try < cur_score - 1e-12:
                    p = p_try
                    cur_score = sc_try
                    improved = True
                    break
            if improved:
                stall = 0
                if cur_score < best_score - 1e-12:
                    best = p.copy()
                    best_score = cur_score
                    if verbose and it % 100 == 0:
                        print(f"      iter={it}, score={best_score:.6f}, beta={beta:.0f}")
            else:
                stall += 1
                if stall > 5:
                    alpha = max(0.01, alpha * 0.8)
                if stall > 30:
                    p = best.copy()
                    cur_score = best_score
                    alpha = 0.15
                    stall = 5

        elif r < 0.70:
            # Frequency domain flattening (A^2 based)
            gamma = float(rng.uniform(0.3, 0.9))
            p_new = _freq_flatten_A2(p, gamma, L)
            if p_new is not None:
                sc_new = _score(p_new)
                if sc_new < cur_score - 1e-12:
                    p = p_new
                    cur_score = sc_new
                    stall = 0
                    if sc_new < best_score - 1e-12:
                        best = p.copy()
                        best_score = sc_new
                        if verbose:
                            print(f"      FREQ-A2 gamma={gamma:.2f}, score={best_score:.6f}")
                else:
                    stall += 1
            else:
                stall += 1

        elif r < 0.80:
            # Frequency domain flattening (C based)
            alpha_f = float(rng.uniform(0.4, 0.95))
            p_new = _freq_flatten_C(p, alpha_f, L)
            if p_new is not None:
                sc_new = _score(p_new)
                if sc_new < cur_score - 1e-12:
                    p = p_new
                    cur_score = sc_new
                    stall = 0
                    if sc_new < best_score - 1e-12:
                        best = p.copy()
                        best_score = sc_new
                else:
                    stall += 1
            else:
                stall += 1

        elif r < 0.90:
            # Peak shaving
            npk = int(rng.choice([3, 5, 10, 20]))
            strength = float(rng.uniform(0.02, 0.15))
            p_new = _peak_shave(p, npk, strength, L)
            sc_new = _score(p_new)
            if sc_new < cur_score - 1e-12:
                p = p_new
                cur_score = sc_new
                stall = 0
                if sc_new < best_score - 1e-12:
                    best = p.copy()
                    best_score = sc_new
            else:
                stall += 1

        elif r < 0.95:
            # Block scaling
            bs = int(rng.choice([4, 8, 16, 32, 64, 128, 256]))
            bs = min(bs, n)
            start = int(rng.integers(0, n - bs + 1))
            f = float(np.exp(rng.normal(0, 0.5)))
            p_try = p.copy()
            p_try[start:start + bs] *= f
            p_try = _sanitize(p_try)
            if p_try is not None:
                sc_try = _score(p_try)
                if sc_try < cur_score - 1e-12:
                    p = p_try
                    cur_score = sc_try
                    stall = 0
                    if sc_try < best_score - 1e-12:
                        best = p.copy()
                        best_score = sc_try
                else:
                    stall += 1
        else:
            # Random restart within neighborhood
            if stall > 10:
                std = 0.02 + 0.003 * min(stall, 40)
                p_try = p * np.exp(rng.normal(0, std, n))
                p_try = _sanitize(p_try)
                if p_try is not None:
                    sc_try = _score(p_try)
                    if sc_try < best_score * 1.05:
                        p = p_try
                        cur_score = sc_try
                        stall = 0
                    else:
                        stall += 1
            else:
                stall += 1

        # Update beta
        if it % 5 == 0:
            beta = min(beta_end, beta * 1.05)

        # Pullback
        if cur_score > best_score * 1.10 and best_score < float('inf'):
            if rng.random() < 0.3:
                p = best.copy()
                cur_score = best_score
                stall = 0

        it += 1

    return best, best_score


def _refine_blocks(p0, rng, time_budget):
    """Intensive block-level refinement."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    best = p.copy()
    best_score = _score(p)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0

    while time.monotonic() < deadline:
        # Variable block sizes
        r = rng.random()
        if r < 0.1:
            bs = int(rng.integers(1, 4))
        elif r < 0.4:
            bs = int(rng.integers(max(2, n // 512), max(4, n // 64)))
        elif r < 0.7:
            bs = int(rng.integers(max(4, n // 64), max(8, n // 16)))
        else:
            bs = int(rng.integers(max(8, n // 16), max(16, n // 4)))
        bs = min(bs, n)
        start = int(rng.integers(0, n - bs + 1))

        improved = False
        factors = (0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.5, 4.0)
        for f in factors:
            p_try = p.copy()
            p_try[start:start + bs] *= f
            p_try = _sanitize(p_try)
            if p_try is None:
                continue
            sc = _score(p_try)
            if sc < cur_score - 1e-12:
                cur_score = sc
                p = p_try
                improved = True
                break

        if improved:
            if cur_score < best_score - 1e-12:
                best = p.copy()
                best_score = cur_score
            stall = 0
        else:
            stall += 1
            if stall > 100:
                p = best.copy()
                cur_score = best_score
                stall = 0

    return best, best_score


def _resample(p_in, n_new, rng=None):
    p = _sanitize(p_in)
    if p is None:
        return None
    n_old = len(p)
    if n_old == n_new:
        result = p.copy()
    elif n_old == 1:
        result = np.ones(n_new, dtype=float)
    else:
        x_old = (np.arange(n_old) + 0.5) / n_old
        x_new = (np.arange(n_new) + 0.5) / n_new
        result = np.interp(x_new, x_old, p)
    result = np.maximum(result, 0)
    if rng is not None:
        result = result * np.exp(rng.normal(0, 0.02, n_new))
        result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Search for best non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Multi-phase: exploration, gradient+freq optimization, block refinement, polishing.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 2400.0

    start_time = time.monotonic()
    deadline = start_time + budget - 20

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng(iseed % (2**31 - 1))

    # Get parent
    try:
        parent = _sanitize(np.asarray(height_sequence_1, dtype=float))
    except Exception:
        parent = None

    best_overall = None
    best_score_overall = float('inf')

    def try_update(p, score):
        nonlocal best_overall, best_score_overall
        if p is not None and np.isfinite(score) and score < best_score_overall:
            best_overall = p.copy()
            best_score_overall = float(score)
            print(f"  *** NEW BEST: {score:.6f} (n={len(p)}) ***")

    # ========== PHASE 1: Exploration (10% time) ==========
    print("=" * 60)
    print(f"Phase 1: Exploration (budget={budget:.0f}s)")
    t1_end = start_time + 0.10 * budget

    if parent is not None:
        sc = _score(parent)
        try_update(parent, sc)
        print(f"  Parent: score={sc:.6f}, n={len(parent)}")

    # Try various initializations at several lengths
    lengths_to_try = [2048, 3072, 4096]
    if parent is not None:
        pn = len(parent)
        if pn not in lengths_to_try:
            lengths_to_try.append(pn)
            lengths_to_try.sort()

    for n in lengths_to_try:
        if time.monotonic() > t1_end:
            break
        inits = [
            _asymmetric_init(n, rng),
            _bimodal_init(n, rng),
            _ramp_init(n, rng),
            np.ones(n) / n,
        ]
        if parent is not None:
            sp = _resample(parent, n, rng)
            if sp is not None:
                inits.append(sp)

        for sp in inits:
            sc = _score(sp)
            try_update(sp, sc)

    print(f"  Phase 1 best: {best_score_overall:.6f}")

    # ========== PHASE 2: Main optimization (50% time) ==========
    print("Phase 2: Main optimization")
    t2_end = start_time + 0.60 * budget

    if best_overall is None:
        best_overall = np.ones(3072) / 3072
        best_score_overall = _score(best_overall)

    n_main = len(best_overall)

    # Run 1: Standard optimization on best
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(0.30 * (t2_end - start_time - 0.10 * budget), t_rem * 0.6, 400.0)
        t_alloc = max(20.0, t_alloc)
        print(f"  Run 1: n={n_main}, alloc={t_alloc:.0f}s")
        p, sc = _optimize_main(best_overall, rng, t_alloc, beta_start=30.0, beta_end=3000.0, verbose=True)
        try_update(p, sc)

    # Run 2: Higher beta
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(200.0, t_rem * 0.6)
        t_alloc = max(20.0, t_alloc)
        print(f"  Run 2: high beta, alloc={t_alloc:.0f}s")
        p1 = _resample(best_overall, n_main, rng)
        if p1 is None:
            p1 = best_overall.copy()
        p, sc = _optimize_main(p1, rng, t_alloc, beta_start=100.0, beta_end=8000.0, verbose=True)
        try_update(p, sc)

    # Run 3: Try a different length (upsample)
    t_rem = t2_end - time.monotonic()
    if t_rem > 60:
        n_up = min(4096, max(4096, n_main))
        if n_up != n_main:
            t_alloc = min(150.0, t_rem * 0.5)
            t_alloc = max(20.0, t_alloc)
            print(f"  Run 3: upsample to n={n_up}, alloc={t_alloc:.0f}s")
            p_up = _resample(best_overall, n_up, rng)
            if p_up is not None:
                p, sc = _optimize_main(p_up, rng, t_alloc, beta_start=50.0, beta_end=5000.0, verbose=True)
                try_update(p, sc)

    # Run 4: Block refinement
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(120.0, t_rem * 0.7)
        t_alloc = max(15.0, t_alloc)
        print(f"  Run 4: block refine, alloc={t_alloc:.0f}s")
        p, sc = _refine_blocks(best_overall, rng, t_alloc)
        try_update(p, sc)

    print(f"  Phase 2 best: {best_score_overall:.6f} (n={len(best_overall)})")

    # ========== PHASE 3: Diverse restarts (15% time) ==========
    print("Phase 3: Diverse restarts")
    t3_end = start_time + 0.78 * budget

    t_rem = t3_end - time.monotonic()
    if t_rem > 60:
        n_best = len(best_overall)
        restarts = [
            _asymmetric_init(n_best, rng),
            _bimodal_init(n_best, rng),
            _ramp_init(n_best, rng),
        ]
        # Also try resampling parent with more noise
        if parent is not None:
            sp = _resample(parent, n_best, rng)
            if sp is not None:
                sp = sp * np.exp(rng.normal(0, 0.1, n_best))
                sp = _sanitize(sp)
                if sp is not None:
                    restarts.append(sp)

        for i, sp in enumerate(restarts):
            if time.monotonic() > t3_end - 10:
                break
            t_alloc = min(80.0, (t3_end - time.monotonic()) * 0.8)
            t_alloc = max(15.0, t_alloc)
            print(f"  Restart {i}: alloc={t_alloc:.0f}s")
            p, sc = _optimize_main(sp, rng, t_alloc, beta_start=50.0, beta_end=4000.0, verbose=False)
            try_update(p, sc)

    print(f"  Phase 3 best: {best_score_overall:.6f}")

    # ========== PHASE 4: Final polishing (15% time) ==========
    print("Phase 4: Final polishing")

    if best_overall is not None:
        # High-beta gradient pass
        t_rem = deadline - time.monotonic()
        if t_rem > 30:
            t_alloc = min(0.35 * t_rem, 200.0)
            t_alloc = max(15.0, t_alloc)
            print(f"  Polish 1: high-beta gradient, alloc={t_alloc:.0f}s")
            p, sc = _optimize_main(best_overall, rng, t_alloc, beta_start=500.0, beta_end=15000.0, verbose=True)
            try_update(p, sc)

        # Block refine
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.30 * t_rem, 150.0)
            t_alloc = max(10.0, t_alloc)
            print(f"  Polish 2: block refine, alloc={t_alloc:.0f}s")
            p, sc = _refine_blocks(best_overall, rng, t_alloc)
            try_update(p, sc)

        # One more gradient pass with medium beta
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.30 * t_rem, 100.0)
            t_alloc = max(10.0, t_alloc)
            print(f"  Polish 3: medium-beta gradient, alloc={t_alloc:.0f}s")
            p, sc = _optimize_main(best_overall, rng, t_alloc, beta_start=200.0, beta_end=8000.0, verbose=True)
            try_update(p, sc)

    # ========== Final output ==========
    if best_overall is None:
        best_overall = np.ones(2048) / 2048

    p_final = _sanitize(best_overall)
    if p_final is None:
        p_final = np.ones(2048) / 2048

    final_score = _score(p_final)

    # Verify with official evaluator
    try:
        eval_score = float(evaluate_sequence([float(x) for x in p_final]))
        if np.isfinite(eval_score):
            final_score = min(final_score, eval_score)
    except Exception:
        pass

    print(f"\n{'=' * 60}")
    print(f"FINAL: score={final_score:.6f} (n={len(p_final)})")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    out = np.where(np.isfinite(p_final), p_final, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = np.ones(2048) / 2048
    else:
        out = out / s

    return [float(x) for x in out]
