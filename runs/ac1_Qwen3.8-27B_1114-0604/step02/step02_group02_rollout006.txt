def autocorr(f):
    """Fast autocorrelation via FFT."""
    n = len(f)
    size = 1 << (2*n - 1).bit_length()
    F = np.fft.rfft(f, size)
    result = np.fft.irfft(F * F, size)
    return result[:2*n-1]

def score(f):
    """Compute the objective value."""
    n = len(f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    c = autocorr(f)
    return 2.0 * n * c.max() / (s * s)

    """Fast subgradient descent with momentum on the peak autocorrelation."""
    f = f.copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-10:
        return f, np.inf
    f = f / s
    
    c = autocorr(f)
    best_peak = c.max()
    best_f = f.copy()
    
    momentum = np.zeros(n)
    lr = 0.5
    beta = 0.9
    
    for i in range(max_iter):

            break
        
        c = autocorr(f)
        peak = c.max()

            best_f = f.copy()

def autocorr_fft(f):

    n = len(f)

    return np.fft.irfft(F * F, size)[:2 * n - 1]

        for amp in [0.4, 0.7, 0.95]:
            pattern = 1.0 + amp * np.cos(2 * np.pi * j / period)
            starts.append(normalize(np.maximum(pattern, 0.0) + 1e-4))

    # Sine absolute value
    for freq in [1, 2, 3, 5, 8]:
        starts.append(normalize(1.0 + 0.5 * np.abs(np.sin(2 * np.pi * freq * j / n))))

    # Gaussian bells of various widths
    for alpha in [0.3, 1.0, 3.0, 8.0, 20.0]:
        starts.append(normalize(np.exp(-alpha * ((j - center) / max(center, 1)) ** 2) + 1e-5))

    # U-shaped
    for alpha in [0.5, 2.0, 8.0]:
        starts.append(normalize(1.0 + alpha * ((j - center) / max(center, 1)) ** 2))

    # Bimodal / multimodal
    for pos in [0.2, 0.25, 0.3, 0.35, 0.4]:
        starts.append(normalize(
            np.exp(-((j - pos * n) / (0.04 * n)) ** 2) +
            np.exp(-((j - (1 - pos) * n) / (0.04 * n)) ** 2) + 1e-4
        ))

    # Random structured
    for _ in range(max(0, count - len(starts))):
        r = rng.random()
        if r < 0.2:
            starts.append(normalize(rng.exponential(1.0, n)))
        elif r < 0.35:
            starts.append(normalize(rng.dirichlet(np.ones(n) * 0.05)))
        elif r < 0.5:
            starts.append(normalize(np.abs(rng.standard_normal(n)) + 0.05))
        elif r < 0.65:
            # Blocky random
            block = max(2, n // 200)
            n_blocks = n // block
            vals = rng.exponential(1.0, n_blocks)
            pattern = np.repeat(vals, block)[:n]
            starts.append(normalize(pattern + 0.01 * rng.standard_normal(n)))
        elif r < 0.8:
            # Sparse
            base = np.zeros(n)
            nnz = max(10, n // 5)
            idx = rng.choice(n, nnz, replace=False)
            base[idx] = rng.exponential(1.0, nnz)
            starts.append(normalize(base + 1e-6))
        else:
            # Random frequency pattern
            n_freqs = rng.integers(2, 8)
            freqs = rng.integers(1, n // 2, n_freqs)
            phases = rng.uniform(0, 2 * np.pi, n_freqs)
            amplitudes = rng.uniform(0.3, 1.0, n_freqs)
            pattern = np.ones(n)
            for k in range(n_freqs):
                pattern += amplitudes[k] * np.cos(2 * np.pi * freqs[k] * j / n + phases[k])
            starts.append(normalize(np.maximum(pattern, 0.0) + 1e-4))

    return starts[:count]


def differential_evolution(population, peaks, best_f, best_peak, deadline, rng, n,
                           pop_size=None, F=None, CR=None):
    """
    Differential Evolution with embedded local optimization and diversity maintenance.
    Returns (best_f, best_peak, population, peaks).
    """
    if pop_size is None:
        pop_size = len(population)
    if F is None:
        F = 0.7
    if CR is None:
        CR = 0.8

    stagnant = 0
    generation = 0
    last_improve_time = time.monotonic()
    total_lo_calls = 0

    while time.monotonic() < deadline:
        generation += 1
        new_pop = []
        new_peaks = []

        for i in range(pop_size):
            # Select three distinct parents
            a, b, c = rng.choice(pop_size, 3, replace=False)
            mutant = population[a] + F * (population[b] - population[c])
            mutant = normalize(mutant)

            # Binomial crossover
            cross_mask = rng.random(n) < CR
            cross_mask[int(rng.integers(n))] = True
            trial = np.where(cross_mask, mutant, population[i])
            trial = normalize(trial)

            # Evaluate
            trial_peak = peak_value(trial)

            # Selection
            if trial_peak < peaks[i]:
                new_pop.append(trial)
                new_peaks.append(trial_peak)
            else:
                new_pop.append(population[i])
                new_peaks.append(peaks[i])

        population = new_pop
        peaks = new_peaks

        # Track best
        gen_best_idx = int(np.argmin(peaks))
        gen_best_peak = peaks[gen_best_idx]

        if gen_best_peak < best_peak - 1e-12:
            best_peak = gen_best_peak
            best_f = population[gen_best_idx].copy()
            stagnant = 0
            last_improve_time = time.monotonic()
        else:
            stagnant += 1

        # Embedded local optimization when stagnant
        if stagnant >= 4 and time.monotonic() < deadline - 20:
            lo_time = min(15.0, (deadline - time.monotonic()) * 0.05)
            if lo_time > 3.0:
                f_lo, peak_lo = local_optimize(best_f, time.monotonic() + lo_time, rng, n)
                if peak_lo < best_peak:
                    best_peak = peak_lo
                    best_f = f_lo.copy()
                    stagnant = 0
                    last_improve_time = time.monotonic()
                # Re-inject improved solution
                population[gen_best_idx] = f_lo.copy()
                peaks[gen_best_idx] = peak_lo
                total_lo_calls += 1

        # Diversity injection when very stagnant
        if stagnant >= 20:
            n_new = max(2, pop_size // 5)
            for _ in range(n_new):
                idx = int(rng.integers(pop_size))
                population[idx] = normalize(population[idx] + 0.4 / np.sqrt(n) * rng.standard_normal(n))
                peaks[idx] = peak_value(population[idx])
            stagnant = 0
            last_improve_time = time.monotonic()

        # Adaptive F and CR
        if stagnant % 10 == 0:
            F = float(np.clip(F + 0.05 * rng.standard_normal(), 0.4, 1.0))
            CR = float(np.clip(CR + 0.05 * rng.standard_normal(), 0.5, 0.95))

        # Periodic logging
        if generation % 20 == 0:
            remaining = deadline - time.monotonic()
            print(f"    DE gen {generation}: peak={best_peak:.8f}, score={2*n*best_peak:.6f}, "
                  f"stag={stagnant}, F={F:.2f}, CR={CR:.2f}, time_left={remaining:.0f}s")

    return best_f, best_peak, population, peaks


def freq_domain_perturb(f, rng, n):
    """Perturb the phase of the FFT and project back."""
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    mag = np.abs(F)
    phase = np.angle(F)
    # Perturb phase
    phase += rng.standard_normal(len(phase)) * 0.3
    F_new = mag * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, size)[:n]
    return normalize(f_new)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Hybrid DE + Local Optimization minimizing 2*n*max(f*f)/(sum f)^2.
    Multi-phase: diverse init -> quick local opt -> DE with embedded LO -> polish.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 25.0

    n = len(height_sequence_1)
    f_known = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)

    print(f"Sequence length: n={n}")
    c0 = autocorr_fft(normalize(f_known))
    print(f"Known solution peak: {c0.max():.8f}, score: {2*n*c0.max():.6f}")

    # === Phase 1: Diverse initialization + quick local optimization ===
    print("\n=== Phase 1: Diverse starts + quick optimization ===")
    t_phase1 = time.monotonic()
    starts = generate_starts(n, rng, f_known=f_known, count=50)

    results = []
    quick_budget_each = 4.0  # seconds per start
    for i, f0 in enumerate(starts):
        remaining_for_phase1 = (start_time + budget_s * 0.15) - time.monotonic()
        if remaining_for_phase1 < 5:
            break
        alloc = min(quick_budget_each, remaining_for_phase1)
        f_opt, peak = local_optimize(f0, time.monotonic() + alloc, rng, n)
        results.append((peak, f_opt))

    results.sort(key=lambda x: x[0])
    best_peak = results[0][0]
    best_f = results[0][1].copy()
    print(f"Phase 1 complete in {time.monotonic()-t_phase1:.1f}s")
    print(f"  Best peak: {best_peak:.8f}, score: {2*n*best_peak:.6f}")
    print(f"  Top 5 peaks: {[f'{p:.6f}' for p, _ in results[:5]]}")

    # === Phase 2: Differential Evolution ===
    print(f"\n=== Phase 2: Differential Evolution ===")
    t_phase2 = time.monotonic()
    pop_size = 50

    # Build initial population from best results + diversity
    population = []
    peaks = []
    for i, (p, f) in enumerate(results[:min(pop_size, len(results))]):
        population.append(f.copy())
        peaks.append(p)
    # Fill remaining with random/freq-domain perturbations
    while len(population) < pop_size:
        if len(population) % 3 == 0:
            f_new = normalize(rng.standard_normal(n) + 1.0)
        elif len(population) % 3 == 1:
            f_new = freq_domain_perturb(best_f, rng, n)
        else:
            f_new = normalize(best_f + 0.3 / np.sqrt(n) * rng.standard_normal(n))
        population.append(f_new)
        peaks.append(peak_value(f_new))

    de_deadline = deadline - 250.0  # Reserve 250s for polish
    best_f, best_peak, population, peaks = differential_evolution(
        population, peaks, best_f, best_peak, de_deadline, rng, n,
        pop_size=pop_size, F=0.7, CR=0.8
    )

    print(f"Phase 2 complete in {time.monotonic()-t_phase2:.1f}s")
    print(f"  Best peak: {best_peak:.8f}, score: {2*n*best_peak:.6f}")

    # === Phase 3: Multi-start polish from top DE solutions ===
    print(f"\n=== Phase 3: Polish top solutions ===")
    t_phase3 = time.monotonic()
    peaks_arr = np.array(peaks)
    top_indices = np.argsort(peaks_arr)[:5]

    polish_deadline = deadline - 100.0
    time_per_polish = min(40.0, (polish_deadline - time.monotonic()) / max(1, len(top_indices)))

    for idx in top_indices:
        remaining = polish_deadline - time.monotonic()
        if remaining < 10:
            break
        f_to_polish = population[idx].copy()
        alloc = min(time_per_polish, remaining - 5)
        f_pol, pol_peak = coordinate_polish(f_to_polish, time.monotonic() + alloc, rng, n)
        if pol_peak < best_peak:
            best_peak = pol_peak
            best_f = f_pol.copy()
            print(f"  Polished solution {idx}: peak={pol_peak:.8f}, score={2*n*pol_peak:.6f}")

    print(f"Phase 3 best score: {2*n*best_peak:.6f}")

    # === Phase 4: Final deep optimization ===
    print(f"\n=== Phase 4: Final deep optimization ===")
    remaining = deadline - time.monotonic()
    if remaining > 30:
        alloc = min(remaining - 20, 120.0)
        f_fin, fin_peak = local_optimize(best_f, time.monotonic() + alloc, rng, n)
        if fin_peak < best_peak:
            best_peak = fin_peak
            best_f = f_fin.copy()
            print(f"  Deep opt: peak={fin_peak:.8f}, score={2*n*fin_peak:.6f}")

    # === Phase 5: Final coordinate polish ===
    remaining = deadline - time.monotonic()
    if remaining > 15:
        alloc = min(remaining - 5, 60.0)
        f_pol2, pol2_peak = coordinate_polish(best_f, time.monotonic() + alloc, rng, n)
        if pol2_peak < best_peak:
            best_peak = pol2_peak
            best_f = f_pol2.copy()
            print(f"  Final polish: peak={pol2_peak:.8f}, score={2*n*pol2_peak:.6f}")

    # === Final verification ===
    best_f = np.clip(best_f, 0.0, 1000.0)
    if best_f.sum() < 0.01:
        best_f = best_f * (1.0 / max(best_f.sum(), 1e-10))

    final_score = evaluate_sequence(best_f.tolist())
    print(f"\nOfficial evaluation: {final_score:.6f}")

    if not np.isfinite(final_score):
        print("Evaluation failed, falling back to known solution...")
        best_f = np.clip(f_known, 0.0, 1000.0)
        final_score = evaluate_sequence(best_f.tolist())
        print(f"Fallback score: {final_score:.6f}")

    # Return the better of our internal score and official
    if final_score < 2.0 * n * best_peak:
        print(f"Official score is better: {final_score:.6f}")

    print(f"=== FINAL: score={final_score:.6f}, internal={2*n*best_peak:.6f} ===")
    return [float(x) for x in best_f]
