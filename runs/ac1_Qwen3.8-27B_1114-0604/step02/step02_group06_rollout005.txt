"""
Optimized search for non-negative sequences minimizing 2*n*max(conv(a,a))/(sum a)^2.
Uses multiplicative weight updates with log-sum-exp smoothing, adaptive step sizes,
multi-scale refinement, and block-based local search.
"""

import numpy as np
import time

_GLOBAL_RNG = np.random.default_rng(42)

def _conv_len(n):
    need = 2 * n - 1
    L = 1
    while L < need:
        L <<= 1
    return L

def _score(a, L=None):
    """Compute the objective: 2*n*max(conv(a,a))/(sum a)^2. Assumes sum a = 1."""
    n = a.size
    if L is None:
        L = _conv_len(n)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)
    c = c[:2*n-1]
    c = np.maximum(c, 0.0)
    return 2.0 * n * c.max(), c

def _sanitize(a, normalize=True):
    a = np.asarray(a, dtype=float)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if normalize:
        s = a.sum()
        if s <= 0:
            return None
        a = a / s
    return a

def _mmd_update(a, time_limit, deadline, beta_init=50.0, beta_max=5000.0, verbose=False):

    n = a.size
    L = _conv_len(n)
    best_a = a.copy()
    best_score, _ = _score(a, L)
    cur_score = best_score
    
    beta = beta_init
    eta = 0.1  # initial step size
    stall = 0
    it = 0
    last_print = -99999
    
    while time.monotonic() < deadline and time.monotonic() < time_limit:
        F = np.fft.rfft(a, L)
        c = np.fft.irfft(F * F, L)
        c = c[:2*n-1]
        c = np.maximum(c, 0.0)
        cur_score = 2.0 * n * c.max()
        
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

                last_print = it

            stall = 0
            it += 1

def _block_refine(a, time_limit, deadline, verbose=False):
    """Block-based local search: scale groups of consecutive elements."""

                a = a / a.sum()
                cur_sc, _ = _score_and_conv(a, L)
                if cur_sc > best_sc * 1.005:
                    a = best_a.copy()
                    cur_sc = best_sc
                stagnant = 0
            if stagnant > 200:
                a = best_a.copy()
                cur_sc = best_sc
                stagnant = 0

        it += 1

    return best_a, best_sc


def _pairwise_block_search(a0, deadline, verbose=False):
    """Optimize pairs of adjacent blocks simultaneously."""
    n = a0.size
    L = _conv_len(n)
    a = a0.copy()
    best_a = a.copy()
    best_sc, _ = _score_and_conv(a, L)
    cur_sc = best_sc
    stagnant = 0
    it = 0
    last_print = -99999

    while time.monotonic() < deadline:
        # Choose two blocks
        bs1 = int(_GLOBAL_RNG.integers(4, max(5, n // 32)))
        bs2 = int(_GLOBAL_RNG.integers(4, max(5, n // 32)))
        total = bs1 + bs2
        if total >= n - 2:
            continue
        s1 = int(_GLOBAL_RNG.integers(0, n - total))
        s2 = s1 + bs1

        f1 = float(_GLOBAL_RNG.uniform(0.5, 2.0))
        f2 = float(_GLOBAL_RNG.uniform(0.5, 2.0))

        a_try = a.copy()
        a_try[s1:s1 + bs1] *= f1
        a_try[s2:s2 + bs2] *= f2
        a_try = np.maximum(a_try, 1e-15)
        s = a_try.sum()
        if s <= 0:
            continue
        a_try = a_try / s
        sc, _ = _score_and_conv(a_try, L)

        if sc < cur_sc - 1e-13:
            a = a_try
            cur_sc = sc
            stagnant = 0
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                if verbose and it - last_print > 500:
                    print(f"  Pair it={it} sc={best_sc:.8f}")
                    last_print = it
        else:
            stagnant += 1
            if stagnant > 50:
                std = 0.05
                a = a * np.exp(_GLOBAL_RNG.normal(0, std, n))
                a = np.maximum(a, 1e-15)
                a = a / a.sum()
                cur_sc, _ = _score_and_conv(a, L)
                if cur_sc > best_sc * 1.01:
                    a = best_a.copy()
                    cur_sc = best_sc
                stagnant = 0
            if stagnant > 300:
                a = best_a.copy()
                cur_sc = best_sc
                stagnant = 0
        it += 1

    return best_a, best_sc


def _coarse_to_fine(coarse_a, n_target, deadline, verbose=False):
    """Upsample and refine from a coarse solution."""
    n_coarse = coarse_a.size
    # Resample to target
    if n_coarse == n_target:
        a = coarse_a.copy()
    elif n_coarse == 1:
        a = np.ones(n_target, dtype=float)
    else:
        x_old = (np.arange(n_coarse) + 0.5) / float(n_coarse)
        x_new = (np.arange(n_target) + 0.5) / float(n_target)
        a = np.interp(x_new, x_old, coarse_a)
        a = np.maximum(a, 1e-10)
        a = a / a.sum()

    # Smooth slightly
    if n_target > 16:
        rad = max(1, n_target // 512)
        k = np.arange(-rad, rad + 1, dtype=float)
        kern = np.exp(-0.5 * (k / max(1.0, rad / 2.0)) ** 2)
        kern = kern / kern.sum()
        a = np.convolve(a, kern, mode='same')
        a = np.maximum(a, 1e-10)
        a = a / a.sum()

    # Add small noise for diversity
    a = a * np.exp(_GLOBAL_RNG.normal(0, 0.03, n_target))
    a = np.maximum(a, 1e-15)
    a = a / a.sum()

    return a


def propose_candidate(seed=777, budget_s=2400, **kwargs):
    global _GLOBAL_RNG

    start_time = time.monotonic()
    budget = min(float(budget_s), 2400.0)
    deadline = start_time + budget - 20  # 20s safety buffer

    _GLOBAL_RNG = np.random.default_rng(int(seed) % 1000000000 + 1)

    # Get parent
    parent = None
    try:
        p_raw = height_sequence_1
        p = np.asarray(p_raw, dtype=float)
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.clip(p, 0, 1000.0)
        s = p.sum()
        if s > 0 and p.size >= 2:
            parent = p / s
    except Exception:
        pass

    if parent is not None:
        n = parent.size
        L = _conv_len(n)
        parent_sc, _ = _score_and_conv(parent, L)
        print(f"Parent: n={n}, score={parent_sc:.8f}")
        best_a = parent.copy()
        best_sc = parent_sc
    else:
        n = 3072
        best_a = np.ones(n, dtype=float) / n
        best_sc = 2.0
        print(f"No parent, using uniform n={n}, score=2.0")

    # Phase 1: MMD refinement (45% of time)
    t1_end = start_time + (deadline - start_time) * 0.45
    print(f"\n=== Phase 1: MMD (t={t1_end - start_time:.0f}s) ===")
    p, sc = _mmd_refined(best_a.copy(), t1_end, beta_init=30, beta_max=5000,
                         eta_init=0.08, verbose=True)
    if sc < best_sc:
        best_sc = sc
        best_a = p
    print(f"Phase 1 done: best={best_sc:.8f} (elapsed={time.monotonic()-start_time:.0f}s)")

    # Phase 2: Block search (20% of time)
    t2_end = start_time + (deadline - start_time) * 0.65
    print(f"\n=== Phase 2: Block search (t={t2_end - t1_end:.0f}s) ===")
    p, sc = _block_search(best_a.copy(), t2_end, verbose=True)
    if sc < best_sc:
        best_sc = sc
        best_a = p
    print(f"Phase 2 done: best={best_sc:.8f} (elapsed={time.monotonic()-start_time:.0f}s)")

    # Phase 3: MMD with high beta (20% of time)
    t3_end = start_time + (deadline - start_time) * 0.85
    print(f"\n=== Phase 3: MMD high-beta (t={t3_end - t2_end:.0f}s) ===")
    p, sc = _mmd_refined(best_a.copy(), t3_end, beta_init=200, beta_max=20000,
                         eta_init=0.04, verbose=True)
    if sc < best_sc:
        best_sc = sc
        best_a = p
    print(f"Phase 3 done: best={best_sc:.8f} (elapsed={time.monotonic()-start_time:.0f}s)")

    # Phase 4: Fine block polish (15% of time)
    print(f"\n=== Phase 4: Fine polish (t={deadline - t3_end:.0f}s) ===")
    p, sc = _block_search(best_a.copy(), deadline, verbose=True)
    if sc < best_sc:
        best_sc = sc
        best_a = p
    print(f"Phase 4 done: best={best_sc:.8f} (elapsed={time.monotonic()-start_time:.0f}s)")

    # Final verification
    rem = deadline - time.monotonic()
    if rem > 5:
        try:
            lst = [float(x) for x in best_a]
            sc_eval = evaluate_sequence(lst)
            if np.isfinite(sc_eval):
                print(f"\nFinal evaluate_sequence: {sc_eval:.8f} (internal: {best_sc:.8f})")
                if sc_eval < best_sc:
                    best_sc = sc_eval
            else:
                print(f"\nFinal evaluate_sequence returned {sc_eval}")
        except Exception as e:
            print(f"Final eval error: {e}")

    result = np.asarray(best_a, dtype=float)
    result = np.clip(result, 0, 1000)
    s = result.sum()
    if s <= 0 or not np.isfinite(s):
        result = np.ones(n, dtype=float) / n
    else:
        result = result / s

    return [float(x) for x in result]
