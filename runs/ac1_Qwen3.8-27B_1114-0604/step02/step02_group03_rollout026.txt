"""
Optimizes non-negative step function sequences to minimize
score = 2*n*max(conv(a,a))/(sum(a))^2.

Uses LP-based direction finding (reduced active-set constraints),
subgradient descent with exact active lags, coordinate descent polish,
and multiple structured starting points with multi-resolution strategy.
"""

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog
from scipy.fft import rfft, irfft


def _normalize(v):
    """Project to non-negative float vector of sum 1."""
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v


def _fft_size(m):
    """Smallest power of two at least m."""
    if m <= 2:
        return 2
    return 1 << (int(m) - 1).bit_length()


def _autocorr(f):
    """Full linear convolution f*f via FFT."""
    n = f.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = 2 * n - 1
    size = _fft_size(m)
    F = rfft(f, size)
    full = irfft(F * F, size)
    return full[:m]


def _fast_score(f):
    """Score for normalized non-negative vector (sum=1)."""
    n = f.size
    c = _autocorr(f)
    if c.size == 0:
        return np.inf
    return float(2.0 * n * c.max())


def _project_simplex(v):
    """Euclidean projection onto x>=0, sum x=1."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    out = v - theta
    np.maximum(out, 0.0, out)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    out /= s
    return out


def _build_lp_sparse(f, lags):
    """Build sparse constraint matrix for LP. Row idx corresponds to lags[idx]."""
    n = f.size
    n_constraints = len(lags)
    rows = []
    cols = []
    vals = []
    for idx, k in enumerate(lags):
        k = int(k)
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        if i_min > i_max:
            continue
        coeffs = f[k - i_max : k - i_min + 1][::-1]
        c_start = len(cols)
        rows.extend([idx] * len(coeffs))
        cols.extend(range(i_min, i_max + 1))
        vals.extend(coeffs.tolist())
    A = sp.csr_matrix((vals, (rows, cols)), shape=(n_constraints, n))
    return A


def _lp_direction(f, M, n_top=300, rng=None):
    """
    Solve reduced LP: max sum(g) s.t. conv(f,g)[k] <= M for top n_top lags, g>=0.
    Returns normalized g (sum=1) or None.
    """
    n = f.size
    conv = _autocorr(f)
    
    # Get top lags (highest values of conv)
    if n_top >= len(conv):
        lags = np.arange(len(conv))
    else:
        lags = np.argpartition(conv, -n_top)[-n_top:]
    
    # Build and solve LP
    A = _build_lp_sparse(f, lags)
    b = np.full(len(lags), M)
    c_obj = -np.ones(n)
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    except Exception as e:
        return None
    
    if not result.success or result.x is None:
        return None
    
    g = np.maximum(0, result.x)
    s = float(g.sum())
    if s <= 1.0 + 1e-8:
        return None
    return g / s


def _subgradient_direction(f, M, n_active=50):
    """
    Compute subgradient direction from active lags.
    Returns a "push" vector: positive means reduce coordinate, negative means increase.
    """
    n = f.size
    conv = _autocorr(f)
    
    # Active lags: those where conv is close to max
    threshold = M - 1e-15
    active = np.where(conv >= threshold - M * 0.05)[0]
    if len(active) > n_active:
        active = np.argpartition(conv, -n_active)[-n_active:]
    
    # Subgradient: for each i, g[i] = sum_{k in active} (f[k-i] + f[i-k]) / |active|
    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        # f[k-i] for i in [max(0,k-n+1), min(k,n-1)]
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        if i_min <= i_max:
            grad[i_min:i_max+1] += f[k-i_max:k-i_min+1][::-1]
        # f[i-k] for i in [max(k,0), min(k+n-1, n-1)] = [k, min(k+n-1,n-1)] if k>=0
        # Actually f[i-k] is non-zero when 0<=i-k<n, i.e., k<=i<n+k
        j_min = max(0, k)
        j_max = min(n - 1, k + n - 1)
        if j_min <= j_max:
            grad[j_min:j_max+1] += f[0:j_max-j_min+1]
    
    grad /= max(1, len(active))
    return grad


def _line_search(f, direction, max_score, n_trials=8):
    """
    Line search: find best t in [0,1] such that f_new = normalize(f + t*direction)
    has score < max_score. Returns (best_t, best_f).
    """
    n = f.size
    best_t = 0.0
    best_f = f.copy()
    best_sc = max_score
    
    for t in np.linspace(0.05, 1.0, n_trials):
        f_new = _normalize(f + t * direction)
        sc = _fast_score(f_new)
        if sc < best_sc - 1e-14:
            best_sc = sc
            best_t = float(t)
            best_f = f_new.copy()
    
    # If no improvement tried full step, try it
    f_new = _normalize(f + direction)
    sc = _fast_score(f_new)
    if sc < best_sc - 1e-14:
        best_sc = sc
        best_t = 1.0
        best_f = f_new.copy()
    
    return best_t, best_f, best_sc

def _gen_gauss_seed(n, rng):
    """Gaussian-like seed."""
    x = np.linspace(0, 1, n)
    mu = 0.5 + 0.2 * rng.standard_normal()
    sig = 0.1 + 0.15 * rng.random()
    f = np.exp(-0.5 * ((x - mu) / sig) ** 2)
    f += 0.01
    return _normalize(f)


def _gen_cos_seed(n, freq, phase, rng):
    """Cosine-modulated seed."""
    x = np.linspace(0, 1, n)
    f = 1.0 + 0.5 * np.cos(2 * np.pi * freq * x + phase)
    f = np.maximum(f, 0.01)
    return _normalize(f)


def _gen_power_seed(n, alpha, rng):
    """Power-law seed."""
    x = np.linspace(0., 1., n)
    f = (x + 0.05) ** alpha * (1 - x + 0.05) ** (rng.uniform(0.5, 2.0))
    return _normalize(f)


def _gen_binary_smooth(n, rng):
    """Smoothed binary seed (random 0/1 pattern, then smoothed)."""
    b = rng.random(n) > 0.5
    b = b.astype(float)
    # Smooth with a window
    w = max(3, n // 500)
    kernel = np.ones(w) / w
    f = np.convolve(b, kernel, mode='same')
    f = np.maximum(f, 0.01)
    return _normalize(f)


def _gen_from_coarse(f_coarse, n):
    """Upsample a coarse sequence to length n."""
    m = f_coarse.size
    if m == n:
        return f_coarse
    x_old = np.linspace(0, m - 1, m)
    x_new = np.linspace(0, n - 1, n)
    return _normalize(np.interp(x_new, x_old, f_coarse))
