"""
Step function optimizer using LP-based direction finding combined with
mirror descent and coordinate polish. Explores multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import linprog
import time

def _autocorr_fft(f):
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    c = np.fft.irfft(F * F, size)[:2 * n - 1]
    return c

def _score_fast(f):
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return float('inf')
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-30:
        return np.ones(n) / n
    # Use the waterfall algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _lp_direction(a, K=100, tol=0.01):
    """
    Solve LP to find direction delta that reduces the autocorrelation peak.
    Minimize t s.t. conv(a, delta)[k] - t <= 0 for top-K lags,
                   sum(delta) = 0, delta >= -a
    """
    n = len(a)
    c = _autocorr_fft(a)
    M = c.max()
    
    # Top K lags (those closest to the peak)
    if K >= 2 * n - 1:
        top_idx = np.arange(2 * n - 1)
    else:
        top_idx = np.argpartition(c, -K)[-K:]
    
    # Build constraint matrix
    # Variables: [delta_0, ..., delta_{n-1}, t] (n+1 vars)
    # For each k in top_idx: sum_j a[j]*delta[k-j] - t <= 0
    rows = []
    for k in top_idx:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        row = np.zeros(n + 1)
        # conv(a, delta)[k] = sum_{j=j_start}^{j_end} a[j] * delta[k-j]
        # delta[k-j] for j in [j_start, j_end] = delta[k-j_end] to delta[k-j_start]
        # = delta[j_start] to delta[j_end] (since k-j_end = j_start when k = j_start + j_end... 
        # wait let me redo this)
        # k - j for j = j_start gives k - j_start
        # k - j for j = j_end gives k - j_end
        # Since j_start = max(0, k-n+1) and j_end = min(n-1, k):
        #   k - j_start = min(k, n-1) = j_end
        #   k - j_end = max(0, k-n+1) = j_start
        # So delta indices range from j_start to j_end
        for j in range(j_start, j_end + 1):
            d_idx = k - j
            row[d_idx] = a[j]
        row[n] = -1.0  # -t
        rows.append(row)
    
    if not rows:
        return None
    
    A_ub = np.array(rows)
    b_ub = np.zeros(len(rows))
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: delta_i >= -a[i], t free
    bounds = [(-a[i], None) for i in range(n)] + [(None, None)]
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n], result.x[n]
    return None

def _mirror_step(f, grad, step):
    """Exponentiated gradient step."""
    span = grad.max() - grad.min()
    if span < 1e-30:
        return f
    eta = step / span
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    return f_new / s

def _optimize_lp_mirror(f0, deadline, rng, K_lp=100, K_grad=10, lp_every=50):
    """
    Main optimization: alternate between LP direction finding and mirror descent.
    """
    f = np.maximum(f0, 0.0).copy()
    f /= f.sum()
    n = len(f)
    
    local_best = _score_fast(f)
    local_best_f = f.copy()
    
    momentum = np.zeros(n)
    beta = 0.8
    it = 0
    stuck = 0
    patience = 15
    
    while time.monotonic() < deadline:
        progress = min(1.0, it / 50000)
        step = 0.5 * (0.01) ** (progress ** 0.5)
        
        # Compute autocorrelation
        c = _autocorr_fft(f)
        peak = c.max()
        
        # Get active lags
        top_K = min(K_grad, 2 * n - 1)
        top_idx = np.argpartition(c, -top_K)[-top_K:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        top_vals = c[top_idx]
        
        # Subgradient: weighted average of conv(a, e_k) for active k
        gaps = np.maximum(top_vals[0] - top_vals, 1e-10)
        weights = 1.0 / gaps
        weights /= weights.sum()
        
        grad = np.zeros(n)
        for k, w in zip(top_idx, weights):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                idx = k - np.arange(j_start, j_end + 1)
                grad[j_start:j_end + 1] += 2.0 * w * f[idx]
        
        # LP direction (every lp_every iterations)
        if it % lp_every == 0:
            delta = _lp_direction(f, K=K_lp)
            if delta is not None:
                # Scale delta to be a valid direction
                delta = np.maximum(delta, -f)  # ensure f + delta >= 0
                delta_sum = delta.sum()
                delta = delta - delta_sum / n  # re-center (approximately)
                delta = np.maximum(delta, -f)
                # Line search
                best_t = 0
                best_peak = peak
                for t_try in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3]:
                    f_try = f + t_try * delta
                    f_try = np.maximum(f_try, 0)
                    f_try /= f_try.sum()
                    c_try = _autocorr_fft(f_try)
                    if c_try.max() < best_peak:
                        best_peak = c_try.max()
                        best_t = t_try
                if best_t > 0:
                    f = f + best_t * delta
                    f = np.maximum(f, 0)
                    f /= f.sum()
        
        # Mirror descent step
        grad_eff = grad + beta * momentum
        momentum = grad + beta * momentum
        f = _mirror_step(f, grad_eff, step)
        
        it += 1
        
        # Check improvement
        if it % 100 == 0:
            score = _score_fast(f)
            if score < local_best - 1e-8:
                local_best = score
                local_best_f = f.copy()
                stuck = 0
            else:
                stuck += 1
            
            if stuck >= patience:
                # Perturb
                mode = int(rng.integers(0, 4))
                j = np.arange(n, dtype=np.float64)
                if mode == 0:
                    alpha = rng.uniform(-0.5, 0.5)
                    f = f * np.exp(alpha * (j / n - 0.5))
                elif mode == 1:
                    freq = int(rng.integers(1, max(2, n // 5) + 1))
                    phase = rng.uniform(0, 2 * np.pi)
                    f = f * (1.0 + 0.3 * np.sin(2 * np.pi * freq * j / n + phase))
                elif mode == 2:
                    f = f + rng.uniform(-0.02, 0.02, n) * f.mean()
                else:
                    f = f * (1.0 + 0.2 * rng.uniform(0, 1, n) - 0.1)
                f = np.maximum(f, 0)
                f /= f.sum()
                momentum *= 0.3
                stuck = 0
    
    score = _score_fast(f)
    if score < local_best:
        local_best = score
        local_best_f = f.copy()
    
    return local_best_f, local_best

def _coordinate_polish(f, deadline, rng):
    """Coordinate descent polish."""
    f = f.copy()
    n = len(f)
    local_best = _score_fast(f)
    local_best_f = f.copy()
    c = _autocorr_fft(f)
    peak = c.max()
    
    iterations = 0
    while time.monotonic() < deadline and iterations < 10000:
        iterations += 1
        # Find the peak lag
        k_star = int(np.argmax(c))
        
        # Compute contribution of each position to c[k_star]
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        contrib = np.zeros(n)
        if j_start <= j_end:
            idx = k_star - np.arange(j_start, j_end + 1)
            contrib[j_start:j_end + 1] = f[np.arange(j_start, j_end + 1)] * f[idx]
        
        # Try moving mass from highest contributor to a random low contributor
        j_don = int(np.argmax(contrib))
        if contrib[j_don] < 1e-15:
            break
        
        # Find a good receiver
        # Positions that don't contribute to the peak lag
        mask = np.ones(n, dtype=bool)
        mask[j_start:j_end+1] = False
        candidates = np.where(mask)[0]
        if len(candidates) == 0:
            candidates = np.arange(n)
        
        j_rec = candidates[int(rng.integers(0, len(candidates)))]
        
        if j_don == j_rec:
            continue
        
        # Try several transfer amounts
        best_frac = 0
        best_p = peak
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
            amt = f[j_don] * frac
            f_try = f.copy()
            f_try[j_don] -= amt
            f_try[j_rec] += amt
            f_try = np.maximum(f_try, 0)
            f_try /= f_try.sum()
            c_try = _autocorr_fft(f_try)
            if c_try.max() < best_p:
                best_p = c_try.max()
                best_frac = frac
        
        if best_frac > 0:
            amt = f[j_don] * best_frac
            f[j_don] -= amt
            f[j_rec] += amt
            f = np.maximum(f, 0)
            f /= f.sum()
            c = _autocorr_fft(f)
            peak = c.max()
            score = _score_fast(f)
            if score < local_best:
                local_best = score
                local_best_f = f.copy()
        elif iterations > 200:
            # Stuck, try random moves
            for _ in range(10):
                i = int(rng.integers(0, n))
                jx = int(rng.integers(0, n))
                if i == jx:
                    continue
                f_try = f.copy()
                amt = f_try[i] * 0.1
                f_try[i] -= amt
                f_try[jx] += amt
                f_try = np.maximum(f_try, 0)
                f_try /= f_try.sum()
                c_try = _autocorr_fft(f_try)
                if c_try.max() < peak:
                    f = f_try
                    c = c_try
                    peak = c_try.max()
                    break
    
    return local_best_f, local_best

"""
Step function optimizer minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses LP-guided direction finding, spectral-shaping initialization,
mirror descent, multi-length search, and coordinate polish.
Key innovation: LP-based direction finding for precise peak reduction,
combined with fast mirror descent steps and diverse initializations.
"""

import numpy as np
from scipy.optimize import linprog
import time


def _autocorr_fft(f):
    """Full linear autocorrelation via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _score_fast(f):
    """Fast score: 2*n*max(conv(f,f))/(sum(f))^2."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return float('inf')
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} using waterfall algorithm."""
    n = len(v)
    v_sum = v.sum()
    if v_sum < 1e-30:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _mirror_step(f, grad, step):
    """Exponentiated gradient (mirror descent) step, preserves non-negativity."""
    span = grad.max() - grad.min()
    if span < 1e-30:
        return f
    eta = step / span
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    return f_new / s


def _lp_direction(a, K=100):
    """
    Solve LP to find direction delta minimizing linearized peak autocorrelation.
    min t s.t. 2*conv(a,delta)[k] - t <= -c[k] for top-K lags,
               sum(delta) = 0, delta >= -a
    Returns (delta, objective_value) or None.
    """
    n = len(a)
    c = _autocorr_fft(a)
    M = c.max()

    if K >= 2 * n - 1:
        top_idx = np.arange(2 * n - 1)
    else:
        top_idx = np.argpartition(c, -K)[-K:]

    n_cons = len(top_idx)
    A_ub = np.zeros((n_cons, n + 1))
    b_ub = np.zeros(n_cons)

    for i, k in enumerate(top_idx):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            b_ub[i] = 0.0
            continue
        # 2*conv(a,delta)[k] = 2*sum_j a[j]*delta[k-j]
        # For d = k-j: j = k-d, valid when 0<=k-d<=n-1, i.e., k-n+1<=d<=k
        # d range: [max(0, k-n+1), min(n-1, k)] = [j_start, j_end]
        # a[j] = a[k-d]
        d_range = np.arange(j_start, j_end + 1)
        a_vals = a[k - d_range]
        A_ub[i, j_start:j_end + 1] = 2.0 * a_vals
        A_ub[i, n] = -1.0
        b_ub[i] = -c[k]

    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    # Bounds
    bounds = [(-a[i], None) for i in range(n)] + [(None, None)]

    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs', options={'presolve': True})
        if result.success:
            delta = result.x[:n]
            return delta, result.x[n]
    except Exception:
        pass
    return None


def _spectral_init(n, rng, n_iter=50):
    """
    Generate a non-negative sequence with relatively flat autocorrelation
    using iterative spectral amplitude matching (Gerchberg-Saxton-like).
    """
    # Start with random phase, target flat amplitude
    size = 1 << (2 * n - 1).bit_length()
    target_amp = np.ones(size // 2 + 1)
    target_amp[0] = np.sqrt(n)  # DC component

    phase = rng.uniform(0, 2 * np.pi, size // 2 + 1)
    A = target_amp * np.exp(1j * phase)

    f = np.ones(n) / n
    for _ in range(n_iter):
        # Compute current spectrum
        F = np.fft.rfft(f, size)
        # Replace amplitudes with target
        A = target_amp * np.exp(1j * np.angle(F))
        # Inverse FFT and take first n
        f_full = np.fft.irfft(A, size)
        f = np.maximum(f_full[:n], 0)
        s = f.sum()
        if s < 1e-10:
            f = np.ones(n) / n
        else:
            f /= s
    return f


def _subgradient(f, top_idx, weights):
    """Compute weighted subgradient of max_k conv(f,f)[k]."""
    n = len(f)
    grad = np.zeros(n)
    for k, w in zip(top_idx, weights):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = k - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] += 2.0 * w * f[idx]
    return grad


def _optimize_combined(f0, deadline, rng, use_lp=True, lp_K=80, lp_every=40):
    """
    Combined optimizer: mirror descent + periodic LP direction finding.
    """
    f = np.maximum(f0, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(len(f))
    else:
        f /= s
    n = len(f)

    local_best = _score_fast(f)
    local_best_f = f.copy()

    momentum = np.zeros(n)
    beta = 0.82
    it = 0
    stuck = 0
    patience = 20
    last_improve_it = 0

    while time.monotonic() < deadline:
        progress = min(1.0, it / 80000)
        step = 0.6 * (0.005) ** (progress ** 0.5)

        c = _autocorr_fft(f)
        peak = c.max()

        # Find active lags
        K_g = min(12, 2 * n - 1)
        top_idx = np.argpartition(c, -K_g)[-K_g:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        top_vals = c[top_idx]
        gaps = np.maximum(top_vals[0] - top_vals, 1e-12)
        weights = 1.0 / gaps
        weights /= weights.sum()

        # LP direction (periodically)
        if use_lp and it % lp_every == 0 and it > 0:
            t_lp = time.monotonic()
            result = _lp_direction(f, K=lp_K)
            if result is not None:
                delta, obj_val = result
                # Line search on LP direction
                best_t = 0.0
                best_pk = peak
                for t_try in [0.02, 0.05, 0.1, 0.2, 0.4]:
                    f_try = f + t_try * delta
                    f_try = np.maximum(f_try, 0)
                    s_try = f_try.sum()
                    if s_try < 1e-10:
                        continue
                    f_try /= s_try
                    c_try = _autocorr_fft(f_try)
                    if c_try.max() < best_pk - 1e-12:
                        best_pk = c_try.max()
                        best_t = t_try
                if best_t > 0:
                    f = f + best_t * delta
                    f = np.maximum(f, 0)
                    f /= f.sum()
            # Don't do mirror step this iteration if LP was used
            it += 1
            if it % 200 == 0:
                sc = _score_fast(f)
                if sc < local_best - 1e-9:
                    local_best = sc
                    local_best_f = f.copy()
                    stuck = 0
                    last_improve_it = it
                else:
                    stuck += 1
                    if stuck >= patience:
                        _perturb(f, rng, n)
                        momentum *= 0.3
                        stuck = 0
            continue

        # Mirror descent step
        grad = _subgradient(f, top_idx, weights)
        grad_eff = grad + beta * momentum
        momentum = grad + beta * momentum
        f = _mirror_step(f, grad_eff, step)

        it += 1

        if it % 150 == 0:
            sc = _score_fast(f)
            if sc < local_best - 1e-9:
                local_best = sc
                local_best_f = f.copy()
                stuck = 0
                last_improve_it = it
            else:
                stuck += 1
                if stuck >= patience:
                    _perturb(f, rng, n)
                    momentum *= 0.3
                    stuck = 0

    sc = _score_fast(f)
    if sc < local_best:
        local_best = sc
        local_best_f = f.copy()
    return local_best_f, local_best


def _perturb(f, rng, n):
    """Apply a random structured perturbation to escape local minima."""
    j = np.arange(n, dtype=np.float64)
    mode = int(rng.integers(0, 6))
    if mode == 0:
        alpha = rng.uniform(-0.8, 0.8)
        f *= np.exp(alpha * (j / n - 0.5))
    elif mode == 1:
        freq = int(rng.integers(1, max(2, n // 4) + 1))
        phase = rng.uniform(0, 2 * np.pi)
        amp = rng.uniform(0.15, 0.5)
        f *= (1.0 + amp * np.sin(2 * np.pi * freq * j / n + phase))
    elif mode == 2:
        # Bimodal boost
        c1, c2 = sorted(rng.uniform(0.1, 0.9, 2))
        bw = rng.uniform(0.05, 0.2)
        f *= (1.0 + 0.4 * (np.exp(-((j / n - c1) / bw) ** 2) +
                           np.exp(-((j / n - c2) / bw) ** 2)))
    elif mode == 3:
        # Add noise
        f += rng.normal(0, 0.03 * f.mean(), n)
    elif mode == 4:
        # Multi-frequency
        for _ in range(2):
            freq = int(rng.integers(1, max(2, n // 3) + 1))
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.05, 0.2)
            f *= (1.0 + amp * np.sin(2 * np.pi * freq * j / n + phase))
    else:
        # Window
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.2, 0.5)
        w = 1.0 - ((j / n - center) / width) ** 2
        f *= np.maximum(w, 0.05)
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-10:
        f[:] = 1.0 / n
    else:
        f /= s


def _coordinate_polish(f, deadline, rng):
    """
    Coordinate descent: iteratively reduce the peak by transferring mass
    from high-contribution positions to low-contribution ones.
    """
    f = f.copy()
    n = len(f)
    local_best = _score_fast(f)
    local_best_f = f.copy()
    c = _autocorr_fft(f)
    peak = c.max()

    no_improve = 0
    while time.monotonic() < deadline and no_improve < 200:
        k_star = int(np.argmax(c))
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)

        # Contribution to peak lag
        contrib = np.zeros(n)
        if j_start <= j_end:
            idx = k_star - np.arange(j_start, j_end + 1)
            contrib[j_start:j_end + 1] = f[np.arange(j_start, j_end + 1)] * f[idx]

        # Also consider total contribution across top few lags
        top5 = np.argpartition(c, -5)[-5:]
        total_contrib = np.zeros(n)
        for k in top5:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                idx_k = k - np.arange(js, je + 1)
                total_contrib[js:je + 1] += f[np.arange(js, je + 1)] * f[idx_k]

        j_don = int(np.argmax(total_contrib))
        if total_contrib[j_don] < 1e-18:
            break

        # Positions not in any top-5 lag range (least risky to add mass)
        in_range = np.zeros(n, dtype=bool)
        for k in top5:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            in_range[js:je + 1] = True
        candidates = np.where(~in_range)[0]
        if len(candidates) == 0:
            candidates = np.where(in_range)[0]
        # Among candidates, pick one with lowest total contribution
        j_rec = candidates[np.argmin(total_contrib[candidates])]

        if j_don == j_rec:
            no_improve += 1
            # Random move
            j_rec = int(rng.integers(0, n))
            if j_rec == j_don:
                continue

        # Try transfer fractions
        improved = False
        for frac in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
            amt = f[j_don] * frac
            if amt < 1e-15:
                continue
            f_try = f.copy()
            f_try[j_don] -= amt
            f_try[j_rec] += amt
            f_try = np.maximum(f_try, 0)
            s_try = f_try.sum()
            if s_try < 1e-10:
                continue
            f_try /= s_try
            c_try = _autocorr_fft(f_try)
            if c_try.max() < peak - 1e-14:
                f = f_try
                c = c_try
                peak = c_try.max()
                sc = _score_fast(f)
                if sc < local_best:
                    local_best = sc
                    local_best_f = f.copy()
                improved = True
                break
        if not improved:
            no_improve += 1
        else:
            no_improve = 0

    return local_best_f, local_best


def _finetune(f, deadline, rng):
    """Ultra-fine mirror descent with small steps on the top 2-3 peaks."""
    f = f.copy()
    n = len(f)
    local_best = _score_fast(f)
    local_best_f = f.copy()

    momentum = np.zeros(n)
    beta = 0.9
    it = 0

    while time.monotonic() < deadline:
        progress = min(1.0, it / 30000)
        step = 0.05 * (0.001) ** (progress ** 0.7)

        c = _autocorr_fft(f)
        K = 3
        top_idx = np.argpartition(c, -K)[-K:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        top_vals = c[top_idx]
        gaps = np.maximum(top_vals[0] - top_vals, 1e-12)
        weights = 1.0 / gaps
        weights /= weights.sum()

        grad = _subgradient(f, top_idx, weights)
        grad_eff = grad + beta * momentum
        momentum = grad + beta * momentum
        f = _mirror_step(f, grad_eff, step)

        it += 1
        if it % 500 == 0:
            sc = _score_fast(f)
            if sc < local_best:
                local_best = sc
                local_best_f = f.copy()

    sc = _score_fast(f)
    if sc < local_best:

        local_best_f = f.copy()
    return local_best_f, local_best


def _generate_starts(n, rng, f_base=None):
    """Generate diverse starting points of length n."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    starts = {}

    starts['uniform'] = np.ones(n)
    starts['u_wide'] = 1.0 + 2.0 * ((j - center) / max(center, 0.5)) ** 2
    starts['u_narrow'] = 1.0 + 6.0 * ((j - center) / max(center, 0.5)) ** 2
    starts['bimodal'] = (np.exp(-((j / n - 0.3) / 0.12) ** 2) +
                         np.exp(-((j / n - 0.7) / 0.12) ** 2) + 0.01)
    starts['triple'] = (np.exp(-((j / n - 0.2) / 0.08) ** 2) +
                        np.exp(-((j / n - 0.5) / 0.08) ** 2) +
                        np.exp(-((j / n - 0.8) / 0.08) ** 2) + 0.01)
    starts['sine1'] = 1.0 + 0.4 * np.sin(2 * np.pi * 3.0 * j / n)
    starts['sine2'] = 1.0 + 0.6 * np.abs(np.sin(np.pi * 2.0 * j / n))
    starts['ramp'] = np.linspace(0.05, 1.0, n)
    starts['exp_l'] = np.exp(-1.5 * j / n) + 0.01
    starts['exp_r'] = np.exp(-1.5 * (n - 1 - j) / n) + 0.01
    starts['rand1'] = rng.exponential(1.0, n)
    starts['rand2'] = rng.dirichlet(np.ones(n) * 0.03)
    starts['spectral'] = _spectral_init(n, rng, n_iter=30)

    # Interpolate from base if provided
    if f_base is not None and len(f_base) != n:
        x_old = np.linspace(0, 1, len(f_base))
        x_new = np.linspace(0, 1, n)
        fb = f_base / f_base.sum()
        starts['interp'] = np.interp(x_new, x_old, fb)

    return starts

    if sc_fine < best_score:
        best_score = sc_fine
        best_f = f_fine.copy()
        print(f"  Fine-tuned: {best_score:.7f}")
    else:
        print(f"  Fine result: {sc_fine:.7f}")

    # Phase 5: Another LP polish round if time remains
    if time.monotonic() < deadline - 40:
        print(f"\n=== Phase 5: Final LP polish (best={best_score:.7f}) ===")
        d = deadline - 5
        f_lp, sc_lp = _optimize_combined(
            best_f, d, rng, use_lp=True, lp_K=150, lp_every=30)
        if sc_lp < best_score:
            best_score = sc_lp
            best_f = f_lp.copy()
            print(f"  LP polished: {best_score:.7f}")

    # Final coordinate micro-polish
    if time.monotonic() < deadline - 10:
        d = deadline - 2
        f_mp, sc_mp = _coordinate_polish(best_f, d, rng)
        if sc_mp < best_score:
            best_score = sc_mp
            best_f = f_mp.copy()
            print(f"  Micro-polish: {best_score:.7f}")

    # Validate
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = evaluate_sequence([float(x) for x in best_f])
    print(f"\nOfficial score: {final_score:.7f}")

    if not np.isfinite(final_score) or final_score > best_score * 1.01:
        # Fallback to base
        fb = np.clip(f_base, 0.0, 1000.0)
        fb_score = evaluate_sequence([float(x) for x in fb])
        if np.isfinite(fb_score):
            best_f = fb
            final_score = fb_score
            print(f"Fallback to base: {final_score:.7f}")

    print(f"\n=== FINAL: {final_score:.7f} (n={len(best_f)}, time={time.monotonic()-t_start:.0f}s) ===")
    return [float(x) for x in best_f]
