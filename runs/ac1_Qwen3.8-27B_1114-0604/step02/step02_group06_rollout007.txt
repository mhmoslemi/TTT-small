"""
LP-based and subgradient optimization for minimizing the maximum autocorrelation
peak of non-negative sequences. Uses cutting-plane LP direction finding combined
with multiplicative updates and local search in a multi-scale framework.
"""

import numpy as np
import time
from scipy.optimize import linprog
import scipy.sparse as sp

_GLOBAL_RNG = np.random.default_rng(42)
_CONV_CACHE = {}


def _conv_len(n):
    n = int(max(1, n))
    if n in _CONV_CACHE:
        return _CONV_CACHE[n]
    need = 2 * n - 1
    L = 1
    while L < need:
        L <<= 1
    _CONV_CACHE[n] = L
    return L


def _sanitize(p, normalize=True):
    if p is None:
        return None
    a = np.asarray(p, dtype=float).ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    a = np.maximum(a, 1e-16)
    if normalize:
        s = float(a.sum())
        if s <= 0.0 or not np.isfinite(s):
            return None
        a = a / s
    return a


def _score(p, L=None):
    """Returns (score, convolution array) for normalized p."""
    p = np.asarray(p, dtype=float)
    n = int(p.size)
    if n == 0:
        return float("inf"), np.array([])
    s = float(p.sum())
    if s <= 0:
        return float("inf"), np.array([])
    if L is None:
        L = _conv_len(n)
    Fp = np.fft.rfft(p, L)
    c = np.fft.irfft(Fp * Fp, L)
    c = c[:2 * n - 1]
    c = np.maximum(c, 0.0)
    max_c = float(c.max())
    sc = float(2.0 * n * max_c / (s * s))
    return sc, c


def _score_val(p):
    p = _sanitize(p)
    if p is None:
        return float("inf")
    if p.sum() < 0.01:
        return float("inf")
    sc, _ = _score(p)
    return sc


def _resample(p, m):
    m = int(max(1, m))
    a = _sanitize(p)
    if a is None:
        return None
    n = int(a.size)
    if n == m:
        return a.copy()
    if n == 1:
        return np.full(m, 1.0 / m)
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 1e-16)
    return _sanitize(out)


def _noise(p, sigma):
    a = _sanitize(p)
    if a is None:
        return None
    r = np.exp(sigma * _GLOBAL_RNG.standard_normal(a.size))
    b = a * r
    return _sanitize(b)


def _smooth(p, sigma):
    a = _sanitize(p)
    if a is None or a.size < 5 or sigma <= 0:
        return a
    rad = int(max(1, np.ceil(3.0 * sigma)))
    k = np.arange(-rad, rad + 1, dtype=float)
    kern = np.exp(-0.5 * (k / sigma) ** 2)
    kern = kern / kern.sum()
    out = np.convolve(a, kern, mode="same")
    return _sanitize(out)


def _lp_direction(p, M, max_iters=30, init_constraints=None):
    """
    Find a direction d >= 0 that maximizes sum(d) subject to (p*d)[k] <= M for all k.
    Uses cutting-plane method. Returns d or None.
    """
    n = int(p.size)
    if n < 4:
        return None

    # Build constraint rows on-the-fly
    # Constraint k: sum_j p[j] * d[k-j] <= M for k=0..2n-2
    # A[k,j] = p[j] if 0 <= k-j <= n-1, else 0

    def get_row(k):
        """Get row k of the constraint matrix as a sparse row."""
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo > j_hi:
            return sp.csr_matrix((1, n), dtype=float), M
        idx = np.arange(j_lo, j_hi + 1)
        vals = p[idx]
        row = sp.csr_matrix((vals, idx, (j_hi - j_lo + 1,)), shape=(1, n))
        return row, M

    # Determine initial constraint set
    if init_constraints is None:
        # Start with the top constraints (where p*p is largest)
        _, conv_p = _score(p)
        k_order = np.argsort(conv_p)[::-1]
        init_k = [int(k_order[i]) for i in range(min(30, len(k_order)))]
    else:
        init_k = list(init_constraints)

    active_k = set(init_k)
    d = None

    for iteration in range(max_iters):
        # Build sparse constraint matrix from active constraints
        rows = []
        rhs = []
        for k in active_k:
            row, m_val = get_row(k)
            rows.append(row)
            rhs.append(m_val)

        if not rows:
            return None

        A_con = sp.vstack(rows, format="csr")
        b_con = np.array(rhs, dtype=float)

        # Add d <= 1 upper bound to keep LP bounded
        A_ub = A_con
        b_ub = b_con

        # Full constraint: A_ub @ d <= b_ub, d >= 0, d <= 1
        # In scipy: c^T d, A_ub d <= b_ub, A_eq d = b_eq, bounds
        c = np.full(n, -1.0)  # minimize -sum(d)
        bounds = [(0.0, 1.0)] * n

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs",
                      options={"time_limit": 5.0, "random_seed": 42})

        if res.status != 0:
            return d  # Return last good direction or None

        d_new = res.x
        s_d = float(d_new.sum())
        if s_d < 1e-10:
            return d  # No feasible improvement

        # Check which constraints are violated
        # Compute (p*d_new) for all k
        L = _conv_len(n)
        Fp = np.fft.rfft(p, L)
        Fd = np.fft.rfft(d_new, L)
        conv_pd = np.fft.irfft(Fp * Fd, L)[:2 * n - 1]
        conv_pd = np.maximum(conv_pd, 0.0)

        # Find violated constraints (conv_pd[k] > M + tol)
        tol = 1e-8 * max(M, 1e-10)
        violations = np.where(conv_pd > M + tol)[0]

        if len(violations) == 0:
            d = d_new
            break

        # Add top violated constraints
        v_order = violations[np.argsort(conv_pd[violations])[::-1]]
        for vk in v_order[:10]:
            active_k.add(int(vk))

        d = d_new

    return d


def _subgradient_step(p, eta, n_max=5):
    """
    Multiplicative subgradient update. Returns updated p.
    """
    n = int(p.size)
    L = _conv_len(n)
    Fp = np.fft.rfft(p, L)
    c = np.fft.irfft(Fp * Fp, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)

    # Find top argmax indices
    k_max = int(np.argmax(c))
    # Subgradient: g[i] = 2 * (S_{k_max} p)[i]
    # (S_k p)[i] = p[k-i] if 0 <= k-i < n, else 0
    # Actually for linear conv: (p*p)[k] = sum_i p[i]*p[k-i]
    # d/dp[j] (p*p)[k] = p[k-j] + p[j-k] (if valid) ... but for our purposes:
    # The gradient of (p*p)[k] w.r.t. p[j] is p[k-j] if 0<=k-j<n, plus p[j-k] if 0<=j-k<n
    # For the linear convolution, it's simpler:
    # (p*p)[k] = sum_i p[i] p[k-i], so d/dp[j] = p[k-j] if 0<=k-j<n (from i=j term)
    # Wait, more carefully: (p*p)[k] = sum_i p[i]*p[k-i]
    # d/dp[j] (p*p)[k] = p[k-j] (from i=j) + p[j-k] (from k-i=j, i.e., i=k-j, but this is the same term if we're summing over all i)
    # Actually no. Let me be very careful.
    # (p*p)[k] = sum_{i: 0<=i<n, 0<=k-i<n} p[i] * p[k-i]
    # d/dp[j] of this = sum over terms where j appears:
    #   - When i=j: term is p[j]*p[k-j], derivative w.r.t. p[j] is p[k-j] (if 0<=k-j<n)
    #   - When k-i=j, i.e., i=k-j: term is p[k-j]*p[j], derivative w.r.t. p[j] is p[k-j] (if 0<=k-j<n)
    # So if j != k-j (i.e., j != k/2), the derivative is 2*p[k-j] (if 0<=k-j<n)
    # If j = k/2, the derivative is p[j] (single term)
    # But for our purposes, let's just use the "uncentered" gradient:
    # g[j] = p[(k-j) % n] ... no, that's circular.
    # For linear conv, the simplest is:
    # g[j] = p[k-j] if 0 <= k-j < n, else 0
    # (This is the gradient of the bilinear form when we only count each pair once,
    #  which is fine for subgradient purposes since it's still a valid subgradient direction)

    # Actually the correct gradient of f_j(p) = sum_i p[i]*p[k-i] w.r.t. p[j] is:
    # p[k-j] if 0 <= k-j < n, PLUS p[j-k] if 0 <= j-k < n
    # The second term is nonzero only if j >= k, in which case j-k < n means j < k+n which is always true.
    # So: g[j] = (p[k-j] if 0<=k-j<n else 0) + (p[j-k] if 0<=j-k<n else 0)

    g = np.zeros(n)
    # First part: p[k-j] for j = 0..k (when k-j in [0, n-1])
    if k >= 0:
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:
            g[j_lo:j_hi + 1] += p[k - np.arange(j_lo, j_hi + 1)]
    # Second part: p[j-k] for j = k..n-1 (when j-k in [0, n-1])
    if k < n:
        j_lo = k
        j_hi = min(n - 1, k + n - 1)
        if j_lo <= j_hi:
            g[j_lo:j_hi + 1] += p[np.arange(j_lo, j_hi + 1) - k]

    # Multiplicative update
    p_new = p * np.exp(-eta * g)
    p_new = np.maximum(p_new, 1e-16)
    p_new = p_new / p_new.sum()
    return p_new


def _multiplicative_optimize(p0, time_limit, target=1.50, verbose=False):
    """
    Main optimizer using multiplicative subgradient updates with adaptive step size.
    """
    p = _sanitize(p0)
    if p is None:
        return None, float("inf")

    time_limit = float(max(1.0, time_limit))
    n = int(p.size)
    L = _conv_len(n)
    deadline = time.monotonic() + time_limit

    sc, c = _score(p, L)
    best = p.copy()
    best_sc = sc
    cur_sc = sc

    eta = 0.5
    stall = 0
    it = 0
    last_print = -1000

    while time.monotonic() < deadline:
        sc, c = _score(p, L)
        cur_sc = sc

        if cur_sc < best_sc - 1e-12:
            best_sc = cur_sc
            best = p.copy()
            stall = 0
            if verbose and (it - last_print >= 100):
                print(f"  mmd n={n} it={it} sc={best_sc:.8f}")
                last_print = it
        else:
            stall += 1

        # Adaptive eta: decrease when stalling
        if stall > 20:
            eta *= 0.85
        elif stall == 0:
            eta = min(eta * 1.02, 2.0)

        # Subgradient step
        p = _subgradient_step(p, eta)

        # Occasionally add noise for exploration
        if stall > 50:
            noise_p = _noise(p, 0.03 + 0.001 * min(stall, 100))
            if noise_p is not None:
                sc2, _ = _score(noise_p, L)
                if sc2 < best_sc * 1.05:
                    p = noise_p
            stall = 0

        # Hard reset if stuck too long
        if stall > 200:
            p = best.copy()
            eta = 0.5
            stall = 0
            noise_p = _noise(p, 0.08)
            if noise_p is not None:
                p = noise_p

        it += 1

    return best, best_sc


def _lp_optimize(p0, time_limit, verbose=False):
    """
    LP-based optimizer using cutting planes for direction finding.
    """
    p = _sanitize(p0)
    if p is None:
        return None, float("inf")

    time_limit = float(max(1.0, time_limit))
    n = int(p.size)
    L = _conv_len(n)
    deadline = time.monotonic() + time_limit

    sc, c = _score(p, L)
    best = p.copy()
    best_sc = sc
    cur_sc = sc

    it = 0
    stall = 0
    last_print = -1000
    active_k = None

    while time.monotonic() < deadline:
        if time.monotonic() > deadline - 5:
            break

        sc, c = _score(p, L)
        cur_sc = sc
        M = float(c.max())
        n_conv = 2 * n - 1

        if cur_sc < best_sc - 1e-12:
            best_sc = cur_sc
            best = p.copy()
            stall = 0
            if verbose and (it - last_print >= 50):
                print(f"  lp n={n} it={it} sc={best_sc:.8f}")
                last_print = it
        else:
            stall += 1

        # Find LP direction
        t_lp = float(deadline - time.monotonic()) - 2.0
        if t_lp < 1.0:
            break

        d = _lp_direction(p, M, max_iters=15, init_constraints=active_k)

        if d is None or d.sum() < 1e-10:
            # Fallback to subgradient
            p = _subgradient_step(p, 0.3)
            it += 1
            continue

        d = d / d.sum()  # Normalize d to sum to 1

        # Line search: try several step sizes
        best_t = 0
        best_p_new = None
        best_sc_new = cur_sc
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7]:
            p_try = (1 - t) * p + t * d
            p_try = np.maximum(p_try, 1e-16)
            p_try = p_try / p_try.sum()
            sc_try, c_try = _score(p_try, L)
            if sc_try < best_sc_new:
                best_sc_new = sc_try
                best_t = t
                best_p_new = p_try

        if best_p_new is not None:
            p = best_p_new
            cur_sc = best_sc_new
            # Update active constraints
            active_k = list(np.argsort(c)[::-1][:20].astype(int))
        else:
            # No improvement, try subgradient
            p = _subgradient_step(p, 0.2)
            active_k = list(np.argsort(c)[::-1][:20].astype(int))

        it += 1

        # Reset if stuck
        if stall > 30:
            p = best.copy()
            stall = 0
            noise_p = _noise(p, 0.05)
            if noise_p is not None:
                p = noise_p

    return best, best_sc


def _local_search(p0, time_limit, verbose=False):
    """
    Local block/scaling search.
    """
    p = _sanitize(p0)
    if p is None:
        return None, float("inf")

    time_limit = float(max(1.0, time_limit))
    n = int(p.size)
    L = _conv_len(n)
    deadline = time.monotonic() + time_limit

    sc, _ = _score(p, L)
    best = p.copy()
    best_sc = sc
    cur_sc = sc

    stall = 0
    it = 0
    last_print = -1000

    # Block sizes to try
    block_sizes = []
    for frac in [0.001, 0.005, 0.01, 0.05, 0.1, 0.2]:
        bs = max(1, int(n * frac))
        if bs not in block_sizes:
            block_sizes.append(bs)
    block_sizes.sort()

    factors = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7, 2.5]

    while time.monotonic() < deadline:
        if time.monotonic() > deadline - 3:
            break

        # Choose random block
        bs = block_sizes[int(_GLOBAL_RNG.integers(0, len(block_sizes)))]
        start = int(_GLOBAL_RNG.integers(0, max(1, n - bs + 1)))
        start = min(start, n - bs)

        best_factor = 1.0
        best_sc_f = cur_sc
        for f in factors:
            p_try = p.copy()
            p_try[start:start + bs] *= f
            p_try = np.maximum(p_try, 1e-16)
            p_try = p_try / p_try.sum()
            sc_f, _ = _score(p_try, L)
            if sc_f < best_sc_f:
                best_sc_f = sc_f
                best_factor = f

        if best_factor != 1.0:
            p = p.copy()
            p[start:start + bs] *= best_factor
            p = np.maximum(p, 1e-16)
            p = p / p.sum()
            cur_sc = best_sc_f
            stall = 0
            if cur_sc < best_sc - 1e-12:
                best_sc = cur_sc
                best = p.copy()
                if verbose and (it - last_print >= 100):
                    print(f"  local n={n} it={it} sc={best_sc:.8f}")
                    last_print = it
        else:
            stall += 1
            if stall > 50:
                p = _noise(best, 0.05)
                cur_sc = _score_val(p)
                stall = 0
            if stall > 200:
                p = best.copy()
                cur_sc = best_sc
                stall = 0

        it += 1

    return best, best_sc


def _coordinate_descent(p0, time_limit, verbose=False):
    """
    Coordinate descent: optimize one or a few elements at a time.
    """
    p = _sanitize(p0)
    if p is None:
        return None, float("inf")

    time_limit = float(max(1.0, time_limit))
    n = int(p.size)
    L = _conv_len(n)
    deadline = time.monotonic() + time_limit

    sc, _ = _score(p, L)
    best = p.copy()
    best_sc = sc
    cur_sc = sc

    iterations = 0
    while time.monotonic() < deadline:
        if time.monotonic() > deadline - 2:
            break

        # Pick a random coordinate (or small group)
        group_size = int(_GLOBAL_RNG.integers(1, 4)) if n > 100 else 1
        idx = int(_GLOBAL_RNG.integers(0, n))
        group = list(range(max(0, idx), min(n, idx + group_size)))

        # Try scaling the group
        best_v = 1.0
        best_sc_v = cur_sc
        for v in [0.1, 0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0, 3.0]:
            p_try = p.copy()
            for gi in group:
                p_try[gi] = p[gi] * v
            p_try = np.maximum(p_try, 1e-16)
            p_try = p_try / p_try.sum()
            sc_v, _ = _score(p_try, L)
            if sc_v < best_sc_v:
                best_sc_v = sc_v
                best_v = v

        if best_v != 1.0:
            for gi in group:
                p[gi] *= best_v
            p = np.maximum(p, 1e-16)
            p = p / p.sum()
            cur_sc = best_sc_v
            if cur_sc < best_sc - 1e-12:
                best_sc = cur_sc
                best = p.copy()

        iterations += 1

    return best, best_sc


def _initial_sequences(n, rng, count=5):
    """Generate diverse initial sequences of length n."""
    seqs = []
    t = (np.arange(n) + 0.5) / n

    # Uniform
    seqs.append(np.ones(n) / n)

    # Beta-like
    for a, b in [(0.5, 0.5), (2.0, 0.5), (0.5, 2.0), (1.5, 1.5), (3.0, 0.3)]:
        tt = np.clip(t, 1e-10, 1 - 1e-10)
        p = tt ** a * (1 - tt) ** b
        p = p / p.sum()
        seqs.append(p)

    # Gaussian bumps
    for _ in range(3):
        K = int(rng.integers(1, 5))
        centers = np.sort(rng.uniform(0, 1, K))
        sigmas = rng.uniform(0.05, 0.3, K)
        weights = rng.uniform(0.5, 2.0, K)
        p = np.zeros(n)
        for i in range(K):
            p += weights[i] * np.exp(-((t - centers[i]) / sigmas[i]) ** 2)
        p = np.maximum(p, 1e-16)
        p = p / p.sum()
        seqs.append(p)

    # Block step
    for _ in range(2):
        K = int(rng.integers(2, 6))
        edges = np.sort(rng.uniform(0, 1, K - 1))
        blocks = np.concatenate([[0], edges, [1.0]])
        heights = rng.lognormal(0, 0.8, K)
        p = np.zeros(n)
        for i in range(K):
            s = int(round(blocks[i] * n))
            e = max(s + 1, int(round(blocks[i + 1] * n)))
            p[s:min(e, n)] = heights[i]
        p = np.maximum(p, 1e-16)
        p = p / p.sum()
        seqs.append(p)

    return seqs[:count]


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Multi-phase optimizer:
    Phase 1: Coarse-to-fine multiplicative optimization
    Phase 2: LP-based refinement
    Phase 3: Local search + coordinate descent polishing
    """
    global _GLOBAL_RNG

    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget < 60:
        budget = 60.0
    total = max(60.0, budget - 30.0)
    deadline = start_time + total

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    _GLOBAL_RNG = np.random.default_rng((iseed % 1000000007) + 7)

    # Try to get parent
    parent = None
    try:
        parent_raw = height_sequence_1
        p = _sanitize(parent_raw)
        if p is not None:
            parent = p
            sc = _score_val(p)
            print(f"Parent sequence: n={p.size}, score={sc:.8f}")
    except (NameError, Exception):
        pass

    # Candidate list: (score, n, array)
    candidates = []

    def add_cand(arr, sc=None):
        a = _sanitize(arr)
        if a is None:
            return
        if sc is None:
            sc = _score_val(a)
        if not np.isfinite(sc):
            return
        candidates.append((sc, int(a.size), a.copy()))
        candidates.sort(key=lambda x: x[0])
        if len(candidates) > 20:
            # Remove worst but keep diversity in size
            candidates = candidates[:15]

    if parent is not None:
        add_cand(parent)

    verbose = total > 300

    # ===== Phase 1: Coarse-to-fine multiplicative optimization =====
    if total > 1000:
        levels = [(512, 0.04), (1024, 0.06), (2048, 0.10), (4096, 0.18), (8192, 0.30), (16384, 0.32)]
    elif total > 600:
        levels = [(512, 0.05), (1024, 0.08), (2048, 0.15), (4096, 0.30), (8192, 0.42)]
    else:
        levels = [(512, 0.06), (2048, 0.20), (4096, 0.46), (8192, 0.28)]

    reserved = max(30.0, 0.03 * total)
    print(f"\n=== Phase 1: Coarse-to-fine ===")

    for nn, frac in levels:
        rem = float(deadline - time.monotonic())
        if rem <= reserved + 60:
            break
        talloc = frac * total
        talloc = min(talloc, max(0.0, rem - reserved - 30))
        if talloc < 15:
            continue

        print(f"\n  Level n={nn}, time={talloc:.1f}s")

        # Build starting points
        starts = []
        if parent is not None:
            sp = _resample(parent, nn)
            if sp is not None:
                starts.append(sp)
        # Add from candidates
        for cand in candidates[:3]:
            if cand[2].size != nn:
                sp = _resample(cand[2], nn)
                if sp is not None:
                    starts.append(sp)
        # Random inits
        inits = _initial_sequences(nn, _GLOBAL_RNG, 3)
        starts.extend(inits)

        # Remove duplicates (approximately)
        unique_starts = [starts[0]] if starts else []
        for s in starts[1:]:
            if not any(np.allclose(s, u, atol=1e-6) for u in unique_starts):
                unique_starts.append(s)

        # Optimize each start
        n_starts = min(len(unique_starts), 4)
        t_each = talloc / max(1, n_starts)

        for si in range(n_starts):
            rem = float(deadline - time.monotonic())
            if rem < 20:
                break
            t_this = min(t_each, rem - 15)
            if t_this < 10:
                break

            p_try = unique_starts[si]
            sc0 = _score_val(p_try)

            p_opt, sc_opt = _multiplicative_optimize(p_try, t_this, target=1.50, verbose=False)
            if p_opt is not None:
                add_cand(p_opt, sc_opt)
                if verbose:
                    print(f"    start {si}: n={nn}, sc={sc_opt:.8f}")

    # ===== Phase 2: LP-based refinement =====
    print(f"\n=== Phase 2: LP refinement ===")
    rem = float(deadline - time.monotonic())
    reserved2 = max(60.0, 0.08 * total)

    if candidates and rem > reserved2 + 30:
        # Pick best few candidates for LP refinement
        n_lp = min(3, len(candidates))
        t_each_lp = (rem - reserved2 - 20) / max(1, n_lp)

        for ci in range(n_lp):
            rem = float(deadline - time.monotonic())
            if rem < 30:
                break
            cand = candidates[ci]
            t_this = min(t_each_lp, rem - 20)
            if t_this < 20:
                break

            print(f"  LP on best[{ci}]: n={cand[1]}, sc={cand[0]:.8f}, t={t_this:.1f}s")
            p_opt, sc_opt = _lp_optimize(cand[2], t_this, verbose=False)
            if p_opt is not None:
                add_cand(p_opt, sc_opt)
                print(f"    LP result: sc={sc_opt:.8f}")

    # ===== Phase 3: Local search + coordinate descent =====
    print(f"\n=== Phase 3: Local polishing ===")
    rem = float(deadline - time.monotonic())
    reserved3 = max(30.0, 0.02 * total)

    if candidates and rem > reserved3 + 20:
        best_cand = candidates[0]
        t_local = min(rem - reserved3 - 10, max(20, 0.1 * total))

        print(f"  Local search: n={best_cand[1]}, sc={best_cand[0]:.8f}, t={t_local:.1f}s")
        p_opt, sc_opt = _local_search(best_cand[2], t_local, verbose=False)
        if p_opt is not None:
            add_cand(p_opt, sc_opt)
            print(f"    Local result: sc={sc_opt:.8f}")

    # ===== Phase 4: Coordinate descent =====
    rem = float(deadline - time.monotonic())
    if candidates and rem > 30:
        best_cand = candidates[0]
        t_cd = min(rem - 15, max(15, 0.05 * total))
        if t_cd > 10:
            print(f"\n=== Phase 4: Coordinate descent ===")
            print(f"  CD: n={best_cand[1]}, sc={best_cand[0]:.8f}, t={t_cd:.1f}s")
            p_opt, sc_opt = _coordinate_descent(best_cand[2], t_cd, verbose=False)
            if p_opt is not None:
                add_cand(p_opt, sc_opt)
                print(f"    CD result: sc={sc_opt:.8f}")

    # ===== Phase 5: Final short multiplicative pass =====
    rem = float(deadline - time.monotonic())
    if candidates and rem > 25:
        best_cand = candidates[0]
        t_final = min(rem - 10, 40.0)
        if t_final > 10:
            print(f"\n=== Phase 5: Final pass ===")
            p_opt, sc_opt = _multiplicative_optimize(best_cand[2], t_final, target=1.50, verbose=False)
            if p_opt is not None:
                add_cand(p_opt, sc_opt)
                print(f"    Final: sc={sc_opt:.8f}")

    # ===== Select best and verify =====
    candidates.sort(key=lambda x: x[0])
    final_arr = candidates[0][2].copy()
    final_score = float(candidates[0][0])

    # Try to verify with evaluate_sequence if available
    rem = float(deadline - time.monotonic())
    if final_arr.size <= 8192 and rem > 5:
        try:
            lst = [float(x) for x in final_arr]
            verified = float(evaluate_sequence(lst))
            if np.isfinite(verified):
                print(f"\nVerified score: {verified:.8f}")
                if verified < final_score:
                    final_score = verified
        except Exception as e:
            print(f"Verification failed: {e}")

    print(f"\nFinal: n={final_arr.size}, score={final_score:.8f}")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    # Ensure valid output
    out = np.asarray(final_arr, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s <= 0.01 or not np.isfinite(s):
        out = np.ones(2048, dtype=float)

    return [float(x) for x in out]
