from scipy.optimize import minimize, linprog
import time

def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_score_grad, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res

def solve_lp_direction(f, M):
    """
    Solve the LP: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0
    Returns the optimal b and the binding constraints.
    """
    n = len(f)
    # Build constraint matrix A: A[k,j] = f[k-j] if 0 <= k-j < n
    # A is (2n-1) x n
    # For memory efficiency, build it row by row
    m = 2 * n - 1
    A = np.zeros((m, n))
    for j in range(n):
        # f[k-j] for k = j, j+1, ..., j+n-1
        A[j:j+n, j] = f
    # LP: maximize c^T b subject to A b <= M (scalar), b >= 0
    # scipy linprog: minimize -c^T b subject to A b <= M, b >= 0
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(m, M),
                  bounds=[(0, None)] * n, method='highs')
    if res.success:
        b_opt = res.x
        # Find binding constraints
        residuals = M - A @ b_opt
        binding = np.where(residuals < 1e-8 * max(M, 1e-12))[0]
        return b_opt, binding
    return None, None

    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    """Wrapper for scipy.minimize."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed score."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def solve_lp_direction(f, M, top_k=300):
    """
    Solve reduced LP: max sum(b) s.t. (f*b)[k] <= M for top-k peak lags, b >= 0.
    Uses sparse constraint matrix.
    Returns optimal b or None on failure.
    """
    n = len(f)
    c = fast_conv(f)
    m_total = 2 * n - 1
    K = min(top_k, m_total)
    # Select top-K lags (highest convolution values)
    top_idx = np.argsort(c)[::-1][:K]
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for ki, k in enumerate(top_idx):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(ki)
            cols.append(j)
            vals.append(f[k - j])
    
    if not vals:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(K, n))
    b_ub = np.full(K, M)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                  bounds=[(0, None)] * n, method='highs',
                  options={'presolve': True, 'time_limit': 60})
    if res.success:
        return res.x
    return None


def normalize_a(a):
    """Normalize to unit sum."""
    S = a.sum()
    if S < 1e-12:
        return a
    return a / S


def build_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())
    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Cosine modulations
    x = np.arange(n, dtype=np.float64) / n
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(2.0 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x))
    inits.append(0.3 + 0.4 * np.cos(2.0 * np.pi * x) ** 2)
    # Quadratic
    inits.append(0.1 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(0.5 + 0.5 * (1.0 - (2.0 * x - 1.0) ** 2))
    # Random
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, 1000.0))
    # Block structure
    for nb in (4, 8, 16, 32):
        per = max(1, n // nb)
        vals = 0.2 + 0.6 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())
    # Gapped
    a = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0)
    a[1::2] *= 0.3
    inits.append(a.copy())
    return inits


def exact_subgrad_polish(a, total_iters=500):
    """Projected subgradient descent on exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Subgradient of M w.r.t. a
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        # Adaptive step
        step = min((sc - sc_lb) / (gn * gn + 1e-30), 0.5)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        # Small extra step with decay
        step_decay = 0.002 / (1.0 + 0.05 * it)
        a -= step_decay * grad / (gn + 1e-15)
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def targeted_perturb(a, rng, strength=0.05):
    """Perturb peak-contributing elements."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(3, n // 50)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:max(1, n - top_count)]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)


def load_parent():
    """Load parent solution from global variable."""
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                return np.clip(p, 0.0, 1000.0)
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Uses LP-guided descent, L-BFGS with LSE continuation, and local search.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    deadline = t_start + total_budget - 15.0

    parent = load_parent()
    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - t_start
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={elapsed:.0f}s)")
            return True
        return False

    if parent is not None:
        sc_parent = compute_score(parent)
        report(sc_parent, parent, "parent")
        print(f"Parent score: {sc_parent:.8f}, length: {len(parent)}")
    else:
        print("No parent available, starting from scratch.")

    # Determine lengths to explore
    n_parent = len(parent) if parent is not None else 4096
    lengths = sorted(set([min(max(n_parent, 1024), 16384), 4096, 8192]))
    lengths = lengths[:3]
    print(f"Lengths to try: {lengths}")

    # ==================================================================
    # Phase 1: LP-guided iterations at the best length (45% of time)
    # ==================================================================
    phase1_deadline = t_start + 0.45 * total_budget
    n_lp = lengths[0]  # Use parent length or first available
    
    print(f"\n{'='*60}")
    print(f"Phase 1: LP-guided descent, n={n_lp}")
    print(f"{'='*60}")
    
    # Initialize
    if parent is not None and len(parent) == n_lp:
        a_cur = parent.copy()
    else:
        a_cur = np.ones(n_lp) * 0.5
    a_cur = normalize_a(a_cur)

    lp_iters = 0
    while time.monotonic() < phase1_deadline:
        sc_cur = compute_score(a_cur)
        report(sc_cur, a_cur, f"LP iter {lp_iters}")
        
        c = fast_conv(a_cur)
        S = float(a_cur.sum())
        M = float(c.max())
        
        # Solve LP direction
        b_opt = solve_lp_direction(a_cur, M, top_k=min(400, 2 * n_lp - 1))
        if b_opt is not None:
            b_sum = float(b_opt.sum())
            if b_sum > 1e-10:
                # Normalize b to match current sum
                b_norm = b_opt * (S / b_sum)
                # Take a step: blend
                t_step = 0.3
                a_candidate = (1.0 - t_step) * a_cur + t_step * b_norm
                a_candidate = np.clip(a_candidate, 0.0, 1000.0)
                a_candidate = normalize_a(a_candidate)
                sc_cand = compute_score(a_candidate)
                if sc_cand < sc_cur:
                    a_cur = a_candidate
                    lp_iters += 1
                else:
                    # Try smaller step
                    t_step = 0.1
                    a_candidate = (1.0 - t_step) * a_cur + t_step * b_norm
                    a_candidate = np.clip(a_candidate, 0.0, 1000.0)
                    a_candidate = normalize_a(a_candidate)
                    sc_cand = compute_score(a_candidate)
                    if sc_cand < sc_cur:
                        a_cur = a_candidate
                        lp_iters += 1
                    else:
                        break
            lp_iters += 1
        else:
            break
        
        # Periodic L-BFGS polish
        if lp_iters % 3 == 0 and time.monotonic() < phase1_deadline - 30:
            sc, a_opt, _ = run_lbfgs(a_cur, beta=200, maxiter=3000, maxfun=50000)
            report(sc, a_opt, f"LP+LBFGS iter {lp_iters}")
            a_cur = a_opt

    print(f"\nPhase 1 complete: {lp_iters} LP iters, best={best_score:.8f}")

    # ==================================================================
    # Phase 2: L-BFGS continuation from multiple starts (25% of time)
    # ==================================================================
    phase2_deadline = t_start + 0.70 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 2: L-BFGS multi-start continuation")
    print(f"{'='*60}")

    beta_schedule = [50, 150, 400, 1000]
    
    for n in lengths:
        if time.monotonic() > phase2_deadline:
            break
        print(f"\n  --- n = {n} ---")
        inits = build_inits(n, rng, best_a if len(best_a) == n else None)
        # Limit initializations based on time
        max_inits = 6
        inits = inits[:max_inits]
        
        for i, a0 in enumerate(inits):
            if time.monotonic() > phase2_deadline:
                break
            a_cur = normalize_a(a0.copy())
            for bi, beta in enumerate(beta_schedule):
                if time.monotonic() > phase2_deadline:
                    break
                remaining = phase2_deadline - time.monotonic()
                if remaining < 15:
                    break
                maxiter = min(5000, int(remaining * 80))
                sc, a_opt, _ = run_lbfgs(a_cur, beta, maxiter=maxiter, maxfun=80000)
                a_cur = a_opt
                report(sc, a_opt, f"n={n}, init={i}, beta={beta}")

    print(f"\nPhase 2 complete: best={best_score:.8f}")

    # ==================================================================
    # Phase 3: Local search with diverse perturbations (20% of time)
    # ==================================================================
    phase3_deadline = t_start + 0.90 * total_budget
    print(f"\n{'='*60}")
    print(f"Phase 3: Local search")
    print(f"{'='*60}")

    n_best = len(best_a)
    trial = 0
    max_trials = 40
    while time.monotonic() < phase3_deadline and trial < max_trials:
        strategy = trial % 5
        if strategy == 0:
            scale = 0.02 + 0.08 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            a_pert = targeted_perturb(best_a, rng, strength=0.03 + 0.05 * rng.random())
        elif strategy == 2:
            freq = 1.0 + 8.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.02 + 0.06 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Sparse random perturbation
            a_pert = best_a.copy()
            n_perturb = max(10, n_best // 20)
            idx = rng.choice(n_best, size=n_perturb, replace=False)
            a_pert[idx] *= (1.0 + 0.1 * rng.standard_normal(n_perturb))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # LP direction from current best
            c = fast_conv(best_a)
            M = float(c.max())
            b_opt = solve_lp_direction(best_a, M, top_k=200)
            if b_opt is not None and b_opt.sum() > 1e-10:
                S = float(best_a.sum())
                b_norm = b_opt * (S / b_opt.sum())
                a_pert = 0.7 * best_a + 0.3 * b_norm
                a_pert = np.clip(a_pert, 0.0, 1000.0)
            else:
                a_pert = best_a * (1.0 + 0.05 * rng.standard_normal(n_best))
                a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Polish with L-BFGS
        a_pert = normalize_a(a_pert)
        sc, a_opt, _ = run_lbfgs(a_pert, beta=500, maxiter=4000, maxfun=80000)
        report(sc, a_opt, f"local #{trial} strat={strategy}")
        trial += 1

    print(f"\nPhase 3 complete: {trial} trials, best={best_score:.8f}")

    # ==================================================================
    # Phase 4: Final polish (remaining time)
    # ==================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Final polish")
    print(f"{'='*60}")

    # High-beta L-BFGS
    for beta in [1000, 2000, 5000]:
        if time.monotonic() > deadline - 10:
            break
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=8000, maxfun=150000)
        report(sc, a_opt, f"final lbfgs beta={beta}")

    # Subgradient polish
    if time.monotonic() < deadline - 10:
        sc, a_pol = exact_subgrad_polish(best_a, total_iters=800)
        report(sc, a_pol, "subgrad polish")
        # L-BFGS from polished
        if time.monotonic() < deadline - 10:
            sc, a_opt, _ = run_lbfgs(a_pol, beta=2000, maxiter=5000, maxfun=100000)
            report(sc, a_opt, "post-subgrad lbfgs")

    # ==================================================================
    # Final verification and return
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
