"""
Optimized search for non-negative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive mirror descent with exact max tracking, frequency-domain operations,
and coordinate descent for final polish.
"""

import numpy as np
import time

def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_fft(a, b):
    n = len(a) + len(b) - 1
    L = _next_pow2(n)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:n].copy()

def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    n = len(p)
    c = _conv_fft(p, p)
    return float(2.0 * n * c.max() / (s * s))

def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return None
    return p / s

def _gradient_smoothed(p, beta, L=None):
    """Compute smoothed gradient of n * max(conv(p,p)) with sum(p)=1 normalization."""
    n = len(p)
    if L is None:
        L = _next_pow2(3 * n)
    c = _conv_fft(p, p)
    d = n * c  # d[k] = n * conv(p,p)[k]
    # Softmax weights
    d_max = d.max()
    e = np.exp(beta * (d - d_max))
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(d)
        w[d.argmax()] = 1.0
    else:
        w = e / s
    # Gradient: h[j] = 2 * sum_{k} w[k] * p[k-j] for valid j
    # This is 2 * cross-correlation of w and p
    W = np.fft.rfft(w, L)
    P = np.fft.rfft(p, L)
    R = np.fft.irfft(W * np.conj(P), L)
    h = 2.0 * R[:n]
    h = np.maximum(h, 0.0)
    h = np.where(np.isfinite(h), h, 0.0)
    exact_score = float(2.0 * d.max())
    return exact_score, h, float(d.max())

def _gradient_exact_top_m(p, m, L=None):
    """Compute gradient using exact top-m positions of the convolution."""
    n = len(p)
    if L is None:
        L = _next_pow2(3 * n)
    c = _conv_fft(p, p)
    d = n * c
    # Find top m positions
    if m >= len(d):
        top_idx = np.arange(len(d))
    else:
        top_idx = np.argpartition(d, -m)[-m:]
    w = np.zeros_like(d)
    w[top_idx] = 1.0 / m
    # Gradient
    W = np.fft.rfft(w, L)
    P = np.fft.rfft(p, L)
    R = np.fft.irfft(W * np.conj(P), L)
    h = 2.0 * R[:n]
    h = np.maximum(h, 0.0)
    h = np.where(np.isfinite(h), h, 0.0)
    exact_score = float(2.0 * d.max())
    return exact_score, h, float(d.max())

    Mirror descent (multiplicative gradient) with adaptive temperature.
    """
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    L = _next_pow2(3 * n + 10)
    
    best = p.copy()
    best_score = _score(p)
    cur_score = best_score
    
    deadline = time.monotonic() + time_budget
    stall = 0
    it = 0
    eta = 0.3  # step size
    beta = 20.0  # initial temperature
    
    if verbose:
        print(f"  AMD: n={n}, init_score={best_score:.6f}, budget={time_budget:.0f}s")

        r = rng.random()

        else:

def _freq_flatten_step(p, alpha, L=None):
    """Frequency-domain convolution flattening."""
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(p, L)
    C_freq = A * A
    C = np.fft.irfft(C_freq, L)[:2 * n - 1]
    cmax = C.max()
    if cmax < 1e-15:
        return p.copy()
    C_norm = C / cmax
    C_new = cmax * np.power(C_norm, 1.0 / alpha)
    C_new = np.maximum(C_new, 0.0)
    C_freq_new = np.fft.rfft(C_new, L)
    A_new = np.sqrt(C_freq_new.astype(complex) + 0j)
    
    # Try to make A_new have a non-negative inverse
    a_new = np.fft.irfft(A_new, L)[:n]
    
    # Alternative: use the real part of sqrt with phase alignment
    # Or: minimize ||a_new||_+ while keeping conv close to C_new
    neg_frac = (a_new < 0).sum() / n
    if neg_frac > 0.25:
        # Try with different phase
        # Use the magnitude and zero phase
        A_mag = np.abs(A_new)
        a_new2 = np.fft.irfft(A_mag.astype(complex), L)[:n]
        if (a_new2 < 0).sum() / n < neg_frac:
            a_new = a_new2
    
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s

def _block_refine(p0, rng, time_budget):
    """Block-wise refinement."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    best = p.copy()
    best_score = _score(p)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0
    factors = (0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0)
    
    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.3:
            bs = int(rng.integers(1, 8))
        elif r < 0.7:
            bs = int(rng.integers(max(2, n // 128), max(4, n // 16)))
        else:
            bs = int(rng.integers(max(4, n // 16), max(8, n // 4)))
        bs = min(bs, n)
        start = int(rng.integers(0, n - bs + 1))
        
        improved = False
        for f in factors:
            p_try = p.copy()
            p_try[start:start + bs] *= f
            p_try = _sanitize(p_try)
            if p_try is None:
                continue
            sc = _score(p_try)
            if sc < cur_score - 1e-14:
                cur_score = sc
                p = p_try
                improved = True
                break
        
        if improved:
            if cur_score < best_score - 1e-13:
                best = p.copy()
                best_score = cur_score
            stall = 0
        else:
            stall += 1
            if stall > 100:
                p = best.copy()
                cur_score = best_score
                stall = 0
    
    return best, best_score

def _coord_descent(p0, rng, time_budget):
    """Coordinate descent on individual elements."""
    p = _sanitize(p0)
    if p is None:

    n = len(p)

def _resample(p_in, n_new, rng=None):
    p = _sanitize(p_in)
    if p is None:
        return None
    n_old = len(p)
    if n_old == n_new:
        result = p.copy()
    else:
        x_old = (np.arange(n_old) + 0.5) / n_old
        x_new = (np.arange(n_new) + 0.5) / n_new
        result = np.interp(x_new, x_old, p)
    result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s

def _init_uniform(n):
    return np.ones(n, dtype=float) / n

def _init_ramp(n, rng):
    """Linear ramp (increasing or decreasing)."""
    p = np.linspace(1, 3, n)
    if rng.random() < 0.5:
        p = p[::-1]
    return p / p.sum()

def _init_u_shape(n, rng):
    """U-shaped (high at edges, low in middle)."""
    t = np.arange(n, dtype=float) / n
    depth = rng.uniform(0.3, 0.8)
    p = 1.0 - depth * np.sin(np.pi * t)
    p = np.maximum(p, 0.1)
    return p / p.sum()

def _init_two_level(n, rng):
    k = int(n * rng.uniform(0.3, 0.7))
    ratio = float(rng.uniform(1.5, 4.0))
    p = np.ones(n, dtype=float)
    p[k:] = ratio
    idx = rng.permutation(n)
    p = p[idx]
    p = np.maximum(p * (1.0 + 0.1 * rng.normal(0, 1, n)), 0)
    s = p.sum()
    if s < 1e-10:
        return _init_uniform(n)
    return p / s

def _init_gaussian(n, rng):
    K = int(rng.integers(2, 6))
    p = np.zeros(n, dtype=float)
    t = np.arange(n, dtype=float) / n
    for _ in range(K):
        c = rng.uniform(0.05, 0.95)
        s = rng.uniform(0.05, 0.3)
        w = rng.uniform(0.5, 2.0)
        x = (t - c) / s
        x = np.minimum(np.abs(x), 1.0 - np.abs(x))  # wrap around
        p += w * np.exp(-0.5 * x * x)
    p = np.maximum(p, 1e-10)
    return p / p.sum()

            a_try = float(np.clip(a_try, 0.01, 4.0))
            p_try = p * np.exp(-a_try * h_norm)
            p_try = np.maximum(p_try, 1e-18 / n)
            s_try = p_try.sum()
            if s_try < 1e-10:
                continue
            p_try = p_try / s_try
            sc_try = _score(p_try)
            if sc_try < best_trial_sc - 1e-14:
                best_trial = p_try
                best_trial_sc = sc_try

        if best_trial is not None:
            p = best_trial
            cur_score = best_trial_sc
            stall = 0
            if cur_score < best_score - 1e-13:
                best = p.copy()
                best_score = cur_score
                if verbose and it % 200 == 0:
                    print(f"      iter={it}, score={best_score:.6f}, beta={beta:.0f}")
            beta = min(beta * 1.04, beta_end)
            eta = max(0.03, eta * 0.997)
        else:
            stall += 1
            if stall > 5:
                eta = max(0.02, eta * 0.7)
            if stall > 25:
                beta = max(beta_start, beta * 0.4)
                eta = 0.25
                stall = 5
            if stall > 60:
                p = best.copy()
                cur_score = best_score
                beta = max(beta_start, beta * 0.5)
                eta = 0.25
                stall = 0
        it += 1
    return best, best_score


def _lp_step(p, rng, max_time=10.0):
    """LP-based direction: find g maximizing sum(g) s.t. conv(p,g) <= M, g >= 0."""
    p = _sanitize(p)
    if p is None:
        return None
    n = len(p)
    c = _conv_fft(p, p)
    M = float(c.max())

    # Use reduced LP: only top m constraints
    m = min(n, 20)
    top_idx = np.argpartition(c, -m)[-m:]
    top_idx = top_idx[np.argsort(c[top_idx][::-1])[::-1]]

    # Build constraint rows for active set
    # A[k, j] = p[k-j] if 0 <= k-j < n
    rows = []
    cols = []
    vals = []
    b_ub = []
    for k in top_idx:
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        for j in range(j_min, j_max + 1):
            rows.append(len(b_ub))
            cols.append(j)
            vals.append(p[k - j])
        b_ub.append(M)

    if len(b_ub) == 0:
        return None

    from scipy.sparse import csr_matrix
    A_ub = csr_matrix((vals, (rows, cols)), shape=(len(b_ub), n))
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n

    t0 = time.monotonic()
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=np.array(b_ub), bounds=bounds,
                         method='highs', options={'time_limit': max_time, 'presolve': True})
    except Exception:
        return None

    if time.monotonic() - t0 > max_time * 1.5:
        return None

    if result.success and result.x is not None:
        g = np.maximum(result.x, 0.0)
        s = g.sum()
        if s < 1e-10:
            return None
        return g / s
    return None


def _freq_flatten(p, alpha, L=None):
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(p, L)
    C_freq = A * A
    C = np.fft.irfft(C_freq, L)[:2 * n - 1]
    cmax = C.max()
    if cmax < 1e-15:
        return p.copy()
    C_norm = C / cmax
    C_new = cmax * np.power(C_norm, 1.0 / alpha)
    C_new = np.maximum(C_new, 0.0)
    C_freq_new = np.fft.rfft(C_new, L)
    A_new = np.sqrt(C_freq_new.astype(complex) + 0j)
    a_new = np.fft.irfft(A_new, L)[:n]
    neg_frac = (a_new < 0).sum() / n
    if neg_frac > 0.25:
        A_mag = np.abs(A_new)
        a_new2 = np.fft.irfft(A_mag.astype(complex) + 0j, L)[:n]
        if (a_new2 < 0).sum() / n < neg_frac:
            a_new = a_new2
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s


def _block_refine(p0, rng, time_budget):
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    best = p.copy()
    best_score = _score(p)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0
    factors = (0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0)

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.3:
            bs = int(rng.integers(1, 8))
        elif r < 0.7:
            bs = int(rng.integers(max(2, n // 128), max(4, n // 16)))
        else:
            bs = int(rng.integers(max(4, n // 16), max(8, n // 4)))
        bs = min(bs, n)
        start = int(rng.integers(0, n - bs + 1))

        improved = False
        for f in factors:
            p_try = p.copy()
            p_try[start:start + bs] *= f
            p_try = _sanitize(p_try)
            if p_try is None:
                continue
            sc = _score(p_try)
            if sc < cur_score - 1e-14:
                cur_score = sc
                p = p_try
                improved = True
                break
        if improved:
            if cur_score < best_score - 1e-13:
                best = p.copy()
                best_score = cur_score
            stall = 0
        else:
            stall += 1
            if stall > 100:
                p = best.copy()
                cur_score = best_score
                stall = 0
    return best, best_score


def _coord_descent(p0, rng, time_budget):
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    best = p.copy()
    best_score = _score(p)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    factors = (0.7, 0.85, 1.15, 1.3)

    while time.monotonic() < deadline:
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            for f in factors:
                p_try = p.copy()
                p_try[j] *= f
                p_try = _sanitize(p_try)
                if p_try is None:
                    continue
                sc = _score(p_try)
                if sc < cur_score - 1e-14:
                    cur_score = sc
                    p = p_try
                    break
            if cur_score < best_score - 1e-13:
                best = p.copy()
                best_score = cur_score
    return best, best_score


def _resample(p_in, n_new, rng=None):
    p = _sanitize(p_in)
    if p is None:
        return None
    n_old = len(p)
    if n_old == n_new:
        result = p.copy()
    else:
        x_old = (np.arange(n_old) + 0.5) / n_old
        x_new = (np.arange(n_new) + 0.5) / n_new
        result = np.interp(x_new, x_old, p)
    result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s


def _init_uniform(n):
    return np.ones(n, dtype=float) / n


def _init_ramp(n, rng):
    p = np.linspace(1.0, 3.0, n)
    if rng.random() < 0.5:
        p = p[::-1]
    return p / p.sum()


def _init_u_shape(n, rng):
    t = np.arange(n, dtype=float) / n
    depth = rng.uniform(0.3, 0.7)
    p = 1.0 - depth * np.sin(np.pi * t)
    p = np.maximum(p, 0.1)
    return p / p.sum()


def _init_two_level(n, rng):
    k = int(n * rng.uniform(0.3, 0.7))
    ratio = float(rng.uniform(1.5, 4.0))
    p = np.ones(n, dtype=float)
    p[k:] = ratio
    idx = rng.permutation(n)
    p = p[idx]
    p = np.maximum(p * (1.0 + 0.1 * rng.normal(0, 1, n)), 0.0)
    s = p.sum()
    if s < 1e-10:
        return _init_uniform(n)
    return p / s


def _init_gaussian(n, rng):
    K = int(rng.integers(2, 6))
    p = np.zeros(n, dtype=float)
    t = np.arange(n, dtype=float) / n
    for _ in range(K):
        c = rng.uniform(0.05, 0.95)
        s = rng.uniform(0.05, 0.3)
        w = rng.uniform(0.5, 2.0)
        x = (t - c) / s
        x = np.minimum(np.abs(x), 1.0 - np.abs(x))
        p += w * np.exp(-0.5 * x * x)
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _hybrid_optimize(p0, rng, time_budget, beta_start=50.0, verbose=False):
    """Alternate between mirror descent and LP steps."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    best = p.copy()
    best_score = _score(p)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    last_lp = 0.0
    lp_count = 0

    if verbose:
        print(f"    Hybrid: n={n}, init={best_score:.6f}, budget={time_budget:.0f}s")

    # Inner mirror descent loop with periodic LP
    md_p = p.copy()
    md_score = cur_score
    md_deadline = deadline - 5.0
    beta = float(beta_start)
    eta = 0.25
    L = _next_pow2(3 * n + 10)
    stall = 0
    it = 0

    while time.monotonic() < md_deadline:
        # Check if time for LP
        if time.monotonic() - last_lp > 8.0 and lp_count < 30:
            g = _lp_step(md_p, rng, max_time=5.0)
            last_lp = time.monotonic()
            lp_count += 1
            if g is not None:
                # Blend: try different t values
                best_blend = None
                best_blend_sc = md_score
                for t in (0.1, 0.2, 0.3, 0.5, 0.7):
                    p_blend = (1.0 - t) * md_p + t * g
                    p_blend = _sanitize(p_blend)
                    if p_blend is None:
                        continue
                    sc_blend = _score(p_blend)
                    if sc_blend < best_blend_sc - 1e-14:
                        best_blend = p_blend
                        best_blend_sc = sc_blend
                if best_blend is not None:
                    md_p = best_blend
                    md_score = best_blend_sc
                    stall = 0
                    if md_score < best_score - 1e-13:
                        best = md_p.copy()
                        best_score = md_score
                        if verbose:
                            print(f"      LP[{lp_count}]: score={best_score:.6f}")
                    continue

        # Mirror descent step
        if beta < 300:
            sc, h = _grad_smoothed(md_p, beta, L)
        else:
            m = max(1, int((beta - 300) / 100) + 1)
            m = min(m, 40)
            sc, h = _grad_top_m(md_p, m, L)

        h_max = h.max()
        if h_max < 1e-15:
            stall += 1
            if stall > 50:
                break
            continue

        h_norm = h / h_max
        best_trial = None
        best_trial_sc = md_score
        for a_try in (eta * 0.5, eta, eta * 1.5, eta * 2.0):
            a_try = float(np.clip(a_try, 0.01, 4.0))
            p_try = md_p * np.exp(-a_try * h_norm)
            p_try = np.maximum(p_try, 1e-18 / n)
            s_try = p_try.sum()
            if s_try < 1e-10:
                continue
            p_try = p_try / s_try
            sc_try = _score(p_try)
            if sc_try < best_trial_sc - 1e-14:
                best_trial = p_try
                best_trial_sc = sc_try

        if best_trial is not None:
            md_p = best_trial
            md_score = best_trial_sc
            stall = 0
            if md_score < best_score - 1e-13:
                best = md_p.copy()
                best_score = md_score
                if verbose and it % 300 == 0:
                    print(f"      MD iter={it}: score={best_score:.6f}, beta={beta:.0f}")
            beta = min(beta * 1.03, 5000.0)
            eta = max(0.03, eta * 0.998)
        else:
            stall += 1
            if stall > 8:
                eta = max(0.02, eta * 0.7)
            if stall > 30:
                beta = max(beta_start, beta * 0.4)
                eta = 0.25
                stall = 5
            if stall > 80:
                md_p = best.copy()
                md_score = best_score
                beta = max(beta_start, 50.0)
                eta = 0.25
                stall = 0
        it += 1

    return best, best_score


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Search for best non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses hybrid mirror descent + LP direction finding + frequency flattening + coordinate descent.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 2400.0

    start_time = time.monotonic()
    deadline = start_time + budget - 25

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng(iseed % (2**31 - 1))

    try:
        parent = _sanitize(np.asarray(height_sequence_1, dtype=float))
    except Exception:
        parent = None

    best_overall = None
    best_score_overall = float('inf')

    def _try_update(p, score_val):
        nonlocal best_overall, best_score_overall
        if p is not None and np.isfinite(score_val) and score_val < best_score_overall:
            best_overall = p.copy()
            best_score_overall = float(score_val)
            print(f"  *** NEW BEST: {score_val:.6f} (n={len(p)}) ***")

    # Phase 1: Exploration (8%)
    print("=" * 60)
    print(f"Phase 1: Exploration (budget={budget:.0f}s)")
    t1_end = start_time + 0.08 * budget

    if parent is not None:
        sc = _score(parent)
        _try_update(parent, sc)
        print(f"  Parent: score={sc:.6f}, n={len(parent)}")

    lengths = [2048, 3072, 4096, 6144]
    if parent is not None:
        n_p = len(parent)
        if n_p not in lengths:
            lengths.append(n_p)
            lengths.sort()

    for n in lengths:
        if time.monotonic() > t1_end:
            break
        inits = [_init_uniform(n), _init_ramp(n, rng), _init_u_shape(n, rng),
                 _init_two_level(n, rng), _init_gaussian(n, rng)]
        if parent is not None:
            sp = _resample(parent, n, rng)
            if sp is not None:
                inits.append(sp)
        for sp in inits:
            sc = _score(sp)
            _try_update(sp, sc)

    print(f"  Phase 1 done. Best: {best_score_overall:.6f}")

    # Phase 2: Main hybrid optimization (52%)
    print("Phase 2: Main hybrid optimization")
    t2_end = start_time + 0.60 * budget

    if best_overall is None:
        best_overall = _init_uniform(3072)
        best_score_overall = _score(best_overall)

    n_main = len(best_overall)

    # Pass 1: Standard hybrid
    t_rem = t2_end - time.monotonic()
    if t_rem > 60:
        t_alloc = min(0.4 * (t2_end - start_time - 0.08 * budget), t_rem * 0.6)
        t_alloc = max(30.0, t_alloc)
        p, sc = _hybrid_optimize(best_overall, rng, t_alloc, beta_start=30.0, verbose=True)
        _try_update(p, sc)

    # Pass 2: Higher beta
    t_rem = t2_end - time.monotonic()
    if t_rem > 40:
        t_alloc = min(150.0, t_rem * 0.6)
        t_alloc = max(20.0, t_alloc)
        p_init = best_overall * np.exp(rng.normal(0, 0.03, n_main))
        p_init = _sanitize(p_init)
        if p_init is None:
            p_init = best_overall.copy()
        p, sc = _hybrid_optimize(p_init, rng, t_alloc, beta_start=100.0, verbose=True)
        _try_update(p, sc)

    # Pass 3: Try different length
    n_up = min(6144, n_main * 2)
    t_rem = t2_end - time.monotonic()
    if t_rem > 60 and n_up != n_main:
        t_alloc = min(120.0, t_rem * 0.6)
        t_alloc = max(20.0, t_alloc)
        p_up = _resample(best_overall, n_up, rng)
        if p_up is not None:
            p, sc = _hybrid_optimize(p_up, rng, t_alloc, beta_start=50.0, verbose=True)
            _try_update(p, sc)

    # Pass 4: Lower beta (broser search)
    t_rem = t2_end - time.monotonic()
    if t_rem > 40:
        t_alloc = min(100.0, t_rem * 0.7)
        t_alloc = max(15.0, t_alloc)
        p, sc = _mirror_descent(best_overall, rng, t_alloc, beta_start=15.0, beta_end=500.0, verbose=True)
        _try_update(p, sc)

    print(f"  Phase 2 done. Best: {best_score_overall:.6f} (n={len(best_overall)})")

    # Phase 3: Freq flattening + block refine (15%)
    print("Phase 3: Freq flattening + refinement")
    t3_end = start_time + 0.
