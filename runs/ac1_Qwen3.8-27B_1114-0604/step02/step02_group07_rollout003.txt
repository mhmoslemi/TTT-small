def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, c, S):
    """Objective value given n, convolution array c, and sum S."""
    if S < 1e-10:

    return 2.0 * n * float(np.max(c)) / (S * S)

def log_space_optimize(a0, end_time, rng, init_lr=0.1, topk=16, patience=200):

    t = 0

    no_imp = 0
    cur_pat = patience
    
    while time.monotonic() < end_time:

            v[:] = 0
            t = 0
            continue
        
        score = compute_score(n, c, S)

            best_score = score
            best_a = a.copy()

def multiplicative_update(a, c, topk=8, lr=0.01):
    """
    Multiplicative (mirror descent) update.
    a_i <- a_i * exp(-lr * grad_i / a_i) = a_i * exp(-lr * d(score)/d(log_a_i))
    """
    n = len(a)
    grad_log = log_space_gradient(a, c, topk)
    # Scale gradient
    a_new = a * np.exp(-lr * grad_log / (np.abs(grad_log).max() + 1e-15))
    return np.clip(a_new, 1e-10, 1000.0)

def peak_directed_descent(a0, end_time, rng, patience=100):
    """
    Focus on reducing the highest peaks in autocorrelation.
    At each step, identify the peak lag, and do a coordinate update
    that specifically reduces that peak.
    """

    best_score = np.inf
    best_a = a.copy()

            continue
        
        score = compute_score(n, c, S)

            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if no_imp >= cur_pat:
            a = np.clip(best_a * (1.0 + 0.1 * rng.standard_normal(n)), 1e-6, 1000.0)
            no_imp = 0
            cur_pat = max(20, cur_pat // 2)
            continue
        
        # Find the dominant peak
        peak_idx = int(np.argmax(c))
        peak_val = c[peak_idx]
        
        # The peak at lag k means: sum_j a[j]*a[k-j] is large
        # To reduce it, we want to decrease products a[j]*a[k-j]
        # Strategy: find the pair (j, k-j) with largest product and reduce it
        js = max(0, peak_idx - n + 1)
        je = min(peak_idx, n - 1)
        
        if js > je:
            continue
        
        products = a[js:je+1] * a[je:js-1:-1]
        max_prod_idx = int(np.argmax(products))
        i1 = js + max_prod_idx
        i2 = peak_idx - i1
        
        # Reduce the larger of a[i1], a[i2]
        if a[i1] >= a[i2]:
            a[i1] *= 0.95
        else:
            a[i2] *= 0.95
        
        a = np.clip(a, 1e-6, 1000.0)
    
    return best_score, best_a


def simulated_annealing(a0, end_time, rng, topk=16):

        c = fast_self_conv(a)
        S = float(a.sum())
        if S < 1e-10:

            continue
        
        score = compute_score(n, c, S)

            best_a = a.copy()

        grad = log_space_gradient(a, c, topk)
        grad_norm = np.linalg.norm(grad) + 1e-15

        scale = 0.01 / grad_norm

            temp *= 0.99
            total_steps += 1
            continue
        score_prop = compute_score(n, c_prop, S_prop)
        
        # Metropolis criterion
        if np.isfinite(score_prop):
            delta = score_prop - score  # negative if improvement
            if delta < 0 or (np.isfinite(score) and rng.random() < np.exp(-delta / max(temp, 1e-10))):
                a = proposal
                no_imp += 1
            else:
                no_imp += 1
        
        if no_imp >= patience:
            a = np.clip(best_a * (1.0 + 0.05 * rng.standard_normal(n)), 1e-6, 1000.0)
            no_imp = 0
            patience = max(50, patience // 2)
        
        temp *= 0.9999
        total_steps += 1
    
    return best_score, best_a

"""
Multi-strategy optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses log-space parameterization with Adam, multiplicative updates,
and multi-restart strategy with diverse initializations.
"""

import numpy as np
import time
import math


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, c, S):
    """Objective value."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def score_from_array(a):
    """Compute score from array."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-10:
        return np.inf
    c = fast_self_conv(a)
    return compute_score(n, c, S)


def gradient_log(a, c, topk=16, temp_frac=0.1):
    """
    Compute gradient of score w.r.t. log(a) = [log(a_0), ..., log(a_{n-1})].
    Uses soft-max approximation for the max in the objective.
    """
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S
    k_act = min(topk, c.size)
    
    # Get top-k peaks
    part = np.argpartition(c, -k_act)[-k_act:]
    vals = c[part].copy()
    order = np.argsort(vals)[::-1]
    part = part[order]
    vals = vals[order]
    
    top1 = float(vals[0])
    top2 = float(vals[1]) if k_act > 1 else 0.0
    
    if top1 > 1e-15:
        ratio = top2 / top1
    else:
        ratio = 1.0
    
    # Blend between exact (top-1 only) and soft (all top-k)
    exact_w = float(np.clip((0.95 - ratio) / 0.5, 0.0, 1.0))
    
    if exact_w > 0.99:
        wts = np.zeros(k_act)
        wts[0] = 1.0
        M = top1
    else:
        temp = max(temp_frac * top1, 1e-15)
        wts_smooth = np.exp((vals - top1) / temp)
        wts_smooth = wts_smooth / wts_smooth.sum()
        if exact_w < 0.01:
            wts = wts_smooth
        else:
            wts_exact = np.zeros(k_act)
            wts_exact[0] = 1.0
            wts = exact_w * wts_exact + (1.0 - exact_w) * wts_smooth
            wts = wts / wts.sum()
        M = float(np.dot(vals, wts))
    
    # Compute dM/da[i] for each i
    # dM/da[i] = sum_k w_k * 2 * a[k-i] (for valid k-i)
    grad_a = np.zeros(n)
    for i in range(k_act):
        wi = float(wts[i])
        if wi <= 1e-15:
            continue
        ki = int(part[i])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        # For i in [js, je]: d(c[ki])/da[i] = 2*a[ki-i]
        # a[ki-i] for i in [js,je] = a[je:js-1:-1] (reversed)
        grad_a[js:je + 1] += 2.0 * wi * a[je:js-1:-1]
    
    # df/da[i] = (2n/S^2) * dM/da[i] - (4nM/S^3) * a[i]
    grad_f_da = (2.0 * n / S2) * grad_a - (4.0 * n * M / S3) * a
    
    # df/d(log_a[i]) = a[i] * df/da[i]
    grad_log = a * grad_f_da
    
    return grad_log, M, S

def multiplicative_optimize(a0, end_time, rng, lr0=0.1, topk=16, patience=200):
    """
    Multiplicative weight update optimization.
    a_i <- a_i * exp(-lr * grad_f_da[i])
    where grad_f_da = df/da[i]
    """
    n = a0.size
    a = np.clip(a0, 1e-6, 1000.0).copy()
    best_score = np.inf
    best_a = a.copy()
    no_imp = 0
    cur_pat = patience
    lr = lr0
    
    while time.monotonic() < end_time:
        c = fast_self_conv(a)
        S = float(a.sum())
        if S < 1e-10:
            a = np.clip(a0, 1e-6, 1000.0)
            continue
        
        score = compute_score(n, c, S)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1
        
        if no_imp >= cur_pat:
            a = np.clip(best_a * (1.0 + 0.05 * rng.standard_normal(n)), 1e-6, 1000.0)
            no_imp = 0
            cur_pat = max(20, cur_pat // 2)
            lr = max(0.001, lr * 0.7)
            continue
        
        # Compute gradient
        grad_log, M, S = gradient_log(a, c, topk)
        grad_f_da = grad_log / a  # Convert back to df/da
        
        # Normalize for stable updates
        g_scale = np.abs(grad_f_da).max() + 1e-15
        
        a = a * np.exp(-lr * grad_f_da / g_scale)
        a = np.clip(a, 1e-8, 1000.0)
        
        lr *= 0.99995
    
    return best_score, best_a
