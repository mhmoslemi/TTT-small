Step-function optimizer using adaptive-temperature multiplicative gradient descent
on the log-sum-exp of the self-convolution, with FFT-based gradients,
asymmetric initializations, multi-length exploration, and exact coordinate polish.

def _normalize(v):
    """Project to non-negative float vector of sum 1."""
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s < 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    return v / s


def _fft_size(m):
    """Smallest power of two at least m."""
    if m <= 2:
        return 2
    return 1 << (int(m) - 1).bit_length()


def _autocorr(f):
    """Full linear convolution f*f via FFT."""
    n = f.size
    m = 2 * n - 1
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]


def _fast_score(f):
    """Score for a normalized non-negative vector (sum=1)."""
    n = f.size
    c = _autocorr(f)
    return float(2.0 * n * c.max())


def _soft_gradient(f, beta):
    """
    Compute the gradient of log-sum-exp(beta * (f*f)) w.r.t. f.
    Returns (gradient, convolution).
    """
    n = f.size
    c = _autocorr(f)
    max_c = float(c.max())
    z = beta * (c - max_c)
    np.clip(z, -60.0, 0.0, out=z)
    p = np.exp(z)
    ps = float(p.sum())
    if ps < 1e-300:
        p = np.ones(c.size, dtype=np.float64) / c.size
    else:
        p /= ps

    # g(i) = 2 * sum_k p(k) * f(k-i) [valid]
    # = 2 * (p cross-correlate f)(i)
    # Computed via: conv(f_reversed, p)[n-1 : 2n-1]
    f_rev = f[::-1]
    size = _fft_size(c.size + n)
    Fr = np.fft.rfft(f_rev, size)
    P = np.fft.rfft(p, size)
    conv = np.fft.irfft(Fr * P, size)[:c.size + n]
    g = 2.0 * conv[n - 1:2 * n - 1]
    return g, c


def _active_lag_gradient(f, c, K):
    """Gradient using only the top-K active lags (faster for large n)."""
    n = f.size
    L = c.size
    K = min(L, K)
    active = np.argpartition(c, L - K)[L - K:]
    p = np.zeros(L, dtype=np.float64)
    p[active] = 1.0 / K

    f_rev = f[::-1]
    size = _fft_size(L + n)
    Fr = np.fft.rfft(f_rev, size)
    P = np.fft.rfft(p, size)
    conv = np.fft.irfft(Fr * P, size)[:L + n]
    g = 2.0 * conv[n - 1:2 * n - 1]
    return g


def _multiplicative_step(f, g, eta):
    """
    Multiplicative update: f_new = f * exp(-eta * g_centered) / Z.
    Maintains non-negativity automatically.
    """
    n = f.size
    g_c = g - float(g.mean())
    g_std = float(np.std(g))
    if g_std < 1e-300:
        return f.copy()
    logits = -eta * g_c / g_std
    np.clip(logits, -40.0, 40.0, out=logits)
    f_new = f * np.exp(logits)
    return _normalize(f_new)


def _euclidean_step(f, g, eta):
    """
    Projected Euclidean gradient step on the simplex.
    """
    n = f.size
    g_c = g - float(g.mean())
    g_range = float(g.max() - g.min())
    if g_range < 1e-300:
        return f.copy()
    v = f - eta * g_c / g_range
    # Project to simplex {x >= 0, sum x = 1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    out = v - theta
    np.maximum(out, 0.0, out)
    s = out.sum()
    if s < 1e-300:
        return np.ones(n, dtype=np.float64) / n
    return out / s
