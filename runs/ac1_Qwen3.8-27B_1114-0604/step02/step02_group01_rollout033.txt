"""
Step function optimizer minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max gradient descent with log-parameterization (softmax parameterization
ensures non-negativity and sum-normalization automatically), Nesterov momentum,
adaptive learning rate, temperature annealing, followed by hard-gradient refinement
and coordinate descent polish.
"""

import numpy as np
import time


def _autocorr_fft(f):
    """Full linear convolution f*f via FFT."""
    n = len(f)
    N = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, N)
    return np.fft.irfft(F * F, N)[:2 * n - 1]


def _score_fast(f):
    """Fast score computation."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))


def _grad_soft_max(a, p):
    """
    Gradient of soft-max(c, tau) w.r.t. a.
    p[k] = exp(c[k]/tau) / sum(exp(c/tau)), length 2n-1.
    Returns grad_a of length n.
    """
    n = len(a)
    N_corr = 1 << (3 * n - 2).bit_length()
    P = np.fft.rfft(p, N_corr)
    Fr = np.fft.rfft(a[::-1], N_corr)
    pf_conv = np.fft.irfft(P * Fr, N_corr)
    return 2.0 * pf_conv[n - 1:2 * n - 1]


def _grad_hard(a, lag):
    """
    Gradient of R[lag] = conv(a,a)[lag] w.r.t. a.
    dR[lag]/da[m] = 2*a[lag-m] if 0 <= lag-m < n, else 0.
    """
    n = len(a)
    grad = np.zeros(n)
    m_start = max(0, lag - n + 1)
    m_end = min(n - 1, lag)
    if m_start <= m_end:
        idx = lag - np.arange(m_start, m_end + 1)
        grad[m_start:m_end + 1] = 2.0 * a[idx]
    return grad


def _optimize_smax(f_init, n, deadline, rng,
                   tau_start=None, tau_end=None,
                   lr_init=1.0, lr_end=0.1,
                   beta=0.9, max_iters=1000000,
                   eval_interval=300, label=""):
    """
    Soft-max gradient descent with log-parameterization.
    a = softmax(b), minimize soft_max(conv(a,a), tau).
    """
    if tau_start is None:
        tau_start = 1.0 / n
    if tau_end is None:
        tau_end = 0.05 / n

    b = np.log(np.maximum(f_init, 1e-300)).astype(np.float64)
    b -= b.max()

    momentum = np.zeros(n)
    best_score = np.inf
    best_b = b.copy()
    best_f = f_init.copy()
    n_evals = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        progress = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** progress
        lr = lr_init * (lr_end / lr_init) ** (progress ** 0.5)

        # Compute a from b (softmax)
        b_shift = b - b.max()
        a = np.exp(b_shift)
        a_sum = a.sum()
        if a_sum < 1e-300:
            break
        a /= a_sum

        # Compute autocorrelation
        c = _autocorr_fft(a)

        # Soft-max weights
        c_s = c / tau
        c_s -= c_s.max()
        exp_c = np.exp(c_s)
        sum_exp = exp_c.sum()
        if sum_exp < 1e-300:

        p = exp_c / sum_exp

        # Gradient w.r.t. a
        grad_a = _grad_soft_max(a, p)

        # Gradient w.r.t. b: a * (grad_a - mean(grad_a * a))
        mean_g = np.dot(grad_a, a)
        grad_b = a * (grad_a - mean_g)

                best_f = a.copy()

    a_best = np.exp(b_shift)

def _optimize_hard(f_init, n, deadline, rng,
                   K=2, lr_init=0.5, lr_end=0.01,
                   beta=0.8, max_iters=500000,
                   eval_interval=100, label=""):
    """
    Hard gradient descent: target top-K peak lags exactly.
    Uses log-parameterization for non-negativity.
    """
    b = np.log(np.maximum(f_init, 1e-300)).astype(np.float64)
    b -= b.max()
    momentum = np.zeros(n)
    best_score = np.inf
    best_b = b.copy()
    best_f = f_init.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        progress = min(1.0, it / max_iters)
        lr = lr_init * (lr_end / lr_init) ** (progress ** 0.7)

        # Compute a
        b_shift = b - b.max()
        a = np.exp(b_shift)
        a_sum = a.sum()
        a /= a_sum

        # Find top K lags
        c = _autocorr_fft(a)
        top_idx = np.argpartition(c, -K)[-K:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        top_vals = c[top_idx]

        # Weight peaks (emphasize the largest)
        gaps = np.maximum(top_vals[0] - top_vals, 1e-15)
        weights = 1.0 / gaps
        weights /= weights.sum()

        # Combined gradient
        grad_a = np.zeros(n)
        for k, w in zip(top_idx, weights):
            grad_a += w * _grad_hard(a, k)

        # Gradient w.r.t. b
        mean_g = np.dot(grad_a, a)
        grad_b = a * (grad_a - mean_g)

        # Momentum step
        momentum = beta * momentum + (1.0 - beta) * grad_b
        b = b - lr * momentum

        if np.any(~np.isfinite(b)):
            b = best_b + rng.normal(0, 0.005, n)
            momentum = np.zeros(n)
            continue
        b = np.clip(b, -80, 80)

        # Evaluate
        if it % eval_interval == 0:
            score = _score_fast(a)
            if score < best_score:
                best_score = score
                best_b = b.copy()
                best_f = a.copy()

    # Final
    b_shift = best_b - best_b.max()
    a_best = np.exp(b_shift)
    a_best /= a_best.sum()
    score = _score_fast(a_best)
    if score < best_score:
        best_score = score
        best_f = a_best.copy()

    return best_f, best_score

        j_rec_candidates = j_rec_candidates[j_rec_candidates != j_don]

        improved = False
        for j_rec in j_rec_candidates[:5]:
            if j_rec == j_don:
                continue
            for frac in [0.15, 0.3, 0.5, 0.7]:
                amt = f[j_don] * frac
                f_try = f.copy()
                f_try[j_don] -= amt
                f_try[j_rec] += amt
                f_try = np.maximum(f_try, 0)
                s = f_try.sum()
                if s < 1e-10:
                    continue
                f_try /= s
                c_try = _autocorr_fft(f_try)
                if c_try.max() < peak - 1e-12:
                    f = f_try
                    c = c_try
                    peak = c.max()
                    improved = True
                    score = _score_fast(f)
                    if score < best_score:
                        best_score = score
                        best_f = f.copy()
                    break
            if improved:
                break

        if not improved:
            # Try random pair swaps
            for _ in range(3):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                amt = f[i] * 0.1
                f_try = f.copy()
                f_try[i] -= amt
                f_try[j] += amt
                f_try = np.maximum(f_try, 0)
                f_try /= f_try.sum()
                c_try = _autocorr_fft(f_try)
                if c_try.max() < peak - 1e-12:
                    f = f_try
                    c = c_try
                    peak = c.max()
                    score = _score_fast(f)
                    if score < best_score:
                        best_score = score
                        best_f = f.copy()
                    break

    return best_f, best_score


def _finetune(f_init, n, deadline, rng, max_iters=200000):
    """
    Ultra-fine tuning: very small hard-gradient steps with K=1.
    """
    b = np.log(np.maximum(f_init, 1e-300)).astype(np.float64)
    b -= b.max()
    momentum = np.zeros(n)
    beta = 0.95
    best_score = np.inf
    best_b = b.copy()
    best_f = f_init.copy()

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        progress = min(1.0, it / max_iters)
        lr = 0.05 * (0.001 / 0.05) ** (progress ** 0.5)

        b_shift = b - b.max()
        a = np.exp(b_shift)
        a /= a.sum()

        c = _autocorr_fft(a)
        k_star = int(np.argmax(c))

        grad_a = _grad_hard(a, k_star)
        mean_g = np.dot(grad_a, a)
        grad_b = a * (grad_a - mean_g)

        momentum = beta * momentum + (1 - beta) * grad_b
        b = b - lr * momentum

        if np.any(~np.isfinite(b)):
            b = best_b.copy()
            momentum = np.zeros(n)
            continue

        if it % 500 == 0:
            score = _score_fast(a)
            if score < best_score:
                best_score = score
                best_b = b.copy()
                best_f = a.copy()

    b_shift = best_b - best_b.max()
    a_best = np.exp(b_shift)
    a_best /= a_best.sum()
    score = _score_fast(a_best)
    if score < best_score:
        best_score = score
        best_f = a_best.copy()

    return best_f, best_score


def _perturb(f, rng, strength=0.1):
    """Gaussian perturbation in log-space, maintaining non-neg and sum=1."""
    b = np.log(np.maximum(f, 1e-300))
    b += rng.normal(0, strength, len(b))
    a = np.exp(b - b.max())
    return a / a.sum()


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the non-negative sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.

    Strategy:
    1. Multi-start soft-max gradient descent with log-parameterization.
    2. Hard-gradient refinement targeting the exact peak lag.
    3. Coordinate descent polish.
    4. Ultra-fine final tuning.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0

    # Load initial sequence
    f_init_raw = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(f_init_raw)
    s0 = f_init_raw.sum()
    f_init = f_init_raw / s0 if s0 > 1e-10 else np.ones(n) / n

    best_score = _score_fast(f_init)
    best_f = f_init.copy()
    print(f"n={n}, initial score={best_score:.8f}")

    # Generate diverse initializations
    j = np.arange(n, dtype=np.float64)
    xn = j / n

    inits = {}
    inits['given'] = f_init.copy()

    # Exponential decays
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0, 6.0]:
        a = np.exp(-alpha * xn)
        inits[f'exp_{alpha}'] = a / a.sum()

    # Linear ramps
    for slope in [0.5, 1.0, 2.0, 3.0]:
        a = 1.0 + slope * xn
        inits[f'ramp_{slope}'] = a / a.sum()
        a2 = 1.0 + slope * (1.0 - xn)
        inits[f'rampR_{slope}'] = a2 / a2.sum()

    # Power laws
    for gamma in [0.3, 0.5, 0.7, 1.0, 1.5]:
        a = (xn + 0.01) ** (-gamma)
        inits[f'pow_{gamma}'] = a / a.sum()
        a2 = (1.0 - xn + 0.01) ** (-gamma)
        inits[f'powR_{gamma}'] = a2 / a2.sum()

    # Quadratic
    a = 1.0 + 2.0 * xn ** 2
    inits['quad'] = a / a.sum()

    # Sine modulated
    for freq in [1, 2, 3, 5]:
        a = 1.0 + 0.3 * np.sin(2 * np.pi * freq * xn)
        a = np.maximum(a, 0.01)
        inits[f'sine{freq}'] = a / a.sum()

    # Random
    for rs in range(3):
        a = rng.exponential(1.0, size=n)
        inits[f'rand{rs}'] = a / a.sum()

    # Bi-modal
    a = np.exp(-((xn - 0.3) / 0.15) ** 2) + np.exp(-((xn - 0.7) / 0.15) ** 2) + 0.01
    inits['bimodal'] = a / a.sum()

    # Evaluate all inits
    init_scores = []
    for name, a in inits.items():
        sc = _score_fast(a)
        init_scores.append((sc, name, a))

    init_scores.sort(key=lambda x: x[0])
    print("\nTop initializations:")
    for sc, name, _ in init_scores[:10]:
        print(f"  {name}: {sc:.8f}")

    # Update best
    if init_scores[0][0] < best_score:
        best_score = init_scores[0][0]
        best_f = init_scores[0][2].copy()

    # Phase 1: Soft-max optimization from top starts
    print(f"\n=== Phase 1: Soft-max optimization ===")
    time_p1_deadline = t_start + (budget_s - 300) * 0.55
    n_starts = min(6, len(init_scores))

    for idx in range(n_starts):
        if time.monotonic() > time_p1_deadline:
            break
        sc, name, a = init_scores[idx]
        t_alloc = (time_p1_deadline - time.monotonic()) / (n_starts - idx)
        t_alloc = min(t_alloc, 200.0)
        d = time.monotonic() + t_alloc

        print(f"  smax '{name}' (init={sc:.7f}, budget={t_alloc:.0f}s)", end=" ")
        f_opt, sc_opt = _optimize_smax(
            a, n, d, rng,
            tau_start=1.0 / n, tau_end=0.03 / n,
            lr_init=2.0, lr_end=0.1,
            beta=0.9, max_iters=500000,
            eval_interval=200, label=name
        )
        print(f"-> {sc_opt:.8f}")

        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"    ** New best: {best_score:.8f}")

    # Phase 2: Hard gradient from the best
    print(f"\n=== Phase 2: Hard gradient (score={best_score:.8f}) ===")
    time_p2_deadline = t_start + (budget_s - 200) * 0.75
    t_alloc2 = min(120.0, time_p2_deadline - time.monotonic())
    if t_alloc2 > 30:
        d2 = time.monotonic() + t_alloc2
        print(f"  hard K=2, budget={t_alloc2:.0f}s", end=" ")
        f_opt, sc_opt = _optimize_hard(
            best_f, n, d2, rng,
            K=2, lr_init=1.0, lr_end=0.02,
            beta=0.85, max_iters=300000,
            eval_interval=100
        )
        print(f"-> {sc_opt:.8f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"    ** New best: {best_score:.8f}")

    # Phase 2b: Hard gradient K=1
    time_p2b_deadline = t_start + (budget_s - 150) * 0.85
    t_alloc2b = min(100.0, time_p2b_deadline - time.monotonic())
    if t_alloc2b > 30:
        d2b = time.monotonic() + t_alloc2b
        print(f"  hard K=1, budget={t_alloc2b:.0f}s", end=" ")
        f_opt, sc_opt = _optimize_hard(
            best_f, n, d2b, rng,
            K=1, lr_init=0.5, lr_end=0.01,
            beta=0.9, max_iters=200000,
            eval_interval=50
        )
        print(f"-> {sc_opt:.8f}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"    ** New best: {best_score:.8f}")

    # Phase 2c: Additional soft-max runs from perturbed best
    time_p2c_deadline = t_start + (budget_s - 100) * 0.90
    n_extra = 3
    remaining2c = time_p2c_deadline - time.monotonic()
    if remaining2c > 60:
        t_each = remaining2c / n_extra
        for i in range(n_extra):
            if time.monotonic() > time_p2c_deadline:
                break
            f_pert = _perturb(best_f, rng, strength=0.05 * (i + 1))
            d = time.monotonic() + t_each
            f_opt, sc_opt = _optimize_smax(
                f_pert, n, d, rng,
                tau_start=0.5 / n, tau_end=0.02 / n,
                lr_init=1.5, lr_end=0.05,
                beta=0.92, max_iters=300000,
                eval_interval=100
            )
            print(f"  extra_smax {i}: {sc_opt:.8f}")
            if sc_opt < best_score:
                best_score = sc_opt
                best_f = f_opt.copy()
                print(f"    ** New best: {best_score:.8f}")

    # Phase 3: Coordinate polish
    print(f"\n=== Phase 3: Coordinate polish (score={best_score:.8f}) ===")
    time_p3_deadline = t_start + (budget_s - 60) * 0.95
    t_alloc3 = min(120.0, max(30.0, time_p3_deadline - time.monotonic()))
    d3 = time.monotonic() + t_alloc3
    f_pol, sc_pol = _coordinate_polish(best_f, n, d3, rng, max_passes=3000)
    print(f"  polish: {sc_pol:.8f}")
    if sc_pol < best_score:
        best_score = sc_pol
        best_f = f_pol.copy()
        print(f"    ** New best: {best_score:.8f}")

    # Phase 4: Fine-tune
    print(f"\n=== Phase 4: Fine-tune (score={best_score:.8f}) ===")
    t_alloc4 = min(120.0, max(20.0, deadline - time.monotonic() - 10))
    d4 = time.monotonic() + t_alloc4
    f_fine, sc_fine = _finetune(best_f, n, d4, rng, max_iters=1000000)
    print(f"  finetune: {sc_fine:.8f}")
    if sc_fine < best_score:
        best_score = sc_fine
        best_f = f_fine.copy()
        print(f"    ** New best: {best_score:.8f}")

    # Phase 5: Final micro-polish if time remains
    if time.monotonic() < deadline - 15:
        print(f"\n=== Phase 5: Final micro-polish ===")
        d5 = deadline - 5.0
        f_micro, sc_micro = _coordinate_polish(best_f, n, d5, rng, max_passes=5000)
        print(f"  micro-polish: {sc_micro:.8f}")
        if sc_micro < best_score:
            best_score = sc_micro
            best_f = f_micro.copy()
            print(f"    ** New best: {best_score:.8f}")

        # One more round of fine-tune if time left
        if time.monotonic() < deadline - 15:
            d5b = deadline - 3.0
            f_ft, sc_ft = _finetune(best_f, n, d5b, rng, max_iters=500000)
            if sc_ft < best_score:
                best_score = sc_ft
                best_f = f_ft.copy()
                print(f"  final finetune: {sc_ft:.8f} ** NEW BEST")

    # Validate with official evaluator
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = evaluate_sequence([float(x) for x in best_f])
    print(f"\nOfficial evaluation: {final_score:.8f}")

    if not np.isfinite(final_score) or final_score > best_score + 1e-10:
        print("Warning: official score differs, using max")
        best_score = min(best_score, final_score) if np.isfinite(final_score) else best_score

    print(f"\n=== FINAL SCORE: {best_score:.8f} ===")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")

    return [float(x) for x in best_f]
