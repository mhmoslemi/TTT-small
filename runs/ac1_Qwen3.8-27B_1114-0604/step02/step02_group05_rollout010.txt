            if p_try is None:
                continue
            sc = _score(p_try)
            if sc < cur_score - 1e-13:
                cur_score = sc
                p = p_try
                improved = True
                break

        if improved:
            if cur_score < best_score - 1e-12:
                best = p.copy()
                best_score = cur_score
            stall = 0
        else:
            stall += 1
            if stall > 80:
                p = best.copy()
                cur_score = best_score
                stall = 0

    return best, best_score


def _init_uniform(n):
    return np.ones(n, dtype=float) / n


def _init_gaussian_mixture(n, rng):
    K = int(rng.integers(2, 8))
    p = np.zeros(n, dtype=float)
    t = np.arange(n, dtype=float) / n
    for _ in range(K):
        c = rng.uniform(0.0, 1.0)
        s = rng.uniform(0.02, 0.3)
        w = rng.uniform(0.5, 2.0)
        x = (t - c) / s
        x = np.minimum(np.abs(x), 1.0 - np.abs(x))
        p += w * np.exp(-0.5 * x * x)
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _init_cosine(n, rng):
    t = np.arange(n, dtype=float) / n
    p = np.ones(n, dtype=float)
    for k in range(1, 12):
        amp = rng.uniform(-0.5, 0.5) / k
        ph = rng.uniform(0, 2 * np.pi)
        p += amp * np.cos(2 * np.pi * k * t + ph)
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return _init_uniform(n)
    return p / s


def _aggressive_freq_iterate(p0, rng, time_budget):
    """Iterative frequency-domain flattening with increasing alpha."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    L = _next_pow2(2 * n - 1)
    best = p.copy()
    best_score = _score(p)
    deadline = time.monotonic() + time_budget
    it = 0

    while time.monotonic() < deadline:
        alpha = float(rng.uniform(1.3, 6.0))
        p_new = _freq_flatten(p, alpha, L)
        if p_new is not None:
            sc = _score(p_new)
            if sc < best_score - 1e-13:
                p = p_new
                best = p.copy()
                best_score = sc
                stall_reset = True
            else:
                # Try blending
                for t_blend in [0.3, 0.5, 0.7]:
                    p_blend = (1 - t_blend) * p + t_blend * p_new
                    p_blend = _sanitize(p_blend)
                    if p_blend is not None:
                        sc_b = _score(p_blend)
                        if sc_b < best_score - 1e-13:
                            p = p_blend
                            best = p.copy()
                            best_score = sc_b
                            break
        it += 1

    return best, best_score


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing the evaluation score.
    Uses Adam in log-space with top-K peak gradient, frequency domain flattening,
    block refinement, and diverse restarts across multiple lengths.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 2400.0

    start_time = time.monotonic()
    deadline = start_time + budget - 30

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng(iseed % (2**31 - 1))

    # Get parent
    try:
        parent = _sanitize(np.asarray(height_sequence_1, dtype=float))
    except Exception:
        parent = None

    best_overall = None
    best_score_overall = float('inf')

    def _try_update(p, score):
        nonlocal best_overall, best_score_overall
        if p is not None and np.isfinite(score) and score < best_score_overall:
            best_overall = p.copy()
            best_score_overall = float(score)
            elapsed = time.monotonic() - start_time
            print(f"  *** NEW BEST: {score:.6f} (n={len(p)}) at t={elapsed:.0f}s ***")

    # ========== PHASE 1: Initialization and quick eval (5% time) ==========
    print("=" * 60)
    print(f"Phase 1: Initialization (budget={budget:.0f}s)")
    t1_end = start_time + 0.05 * budget

    if parent is not None:
        sc = _score(parent)
        _try_update(parent, sc)
        print(f"  Parent: score={sc:.6f} (n={len(parent)})")

    # Try a few initializations
    n_parent = len(parent) if parent is not None else 3072
    explore_lengths = [n_parent]
    if n_parent != 2048:
        explore_lengths.append(2048)
    if n_parent != 4096:
        explore_lengths.append(4096)

    for n in sorted(explore_lengths):
        if time.monotonic() > t1_end:
            break
        inits = [_init_uniform(n), _init_gaussian_mixture(n, rng), _init_cosine(n, rng)]
        if parent is not None:
            sp = _resample(parent, n, rng)
            if sp is not None:
                inits.insert(0, sp)
        for sp in inits:
            sc = _score(sp)
            _try_update(sp, sc)

    print(f"  Phase 1 done. Best: {best_score_overall:.6f} (n={len(best_overall)})")

    # ========== PHASE 2: Main Adam optimization (40% time) ==========
    print("Phase 2: Main Adam optimization")
    t2_end = start_time + 0.45 * budget

    n_main = len(best_overall) if best_overall is not None else n_parent
    if best_overall is None:
        best_overall = _init_uniform(n_parent)
        best_score_overall = _score(best_overall)

    # Run 1: Standard Adam
    t_rem = t2_end - time.monotonic()
    if t_rem > 60:
        t_alloc = min(250.0, 0.6 * t_rem)
        p, sc = _optimize_adam(best_overall, rng, t_alloc, K_start=3, K_end=40,
                               lr_start=0.06, lr_end=0.01, verbose=True)
        _try_update(p, sc)

    # Run 2: Higher K (broader peak targeting)
    t_rem = t2_end - time.monotonic()
    if t_rem > 60:
        t_alloc = min(150.0, 0.6 * t_rem)
        p1 = _resample(best_overall, len(best_overall), rng)
        if p1 is None:
            p1 = best_overall.copy()
        p, sc = _optimize_adam(p1, rng, t_alloc, K_start=10, K_end=80,
                               lr_start=0.04, lr_end=0.005, verbose=True)
        _try_update(p, sc)

    # Run 3: Aggressive frequency iteration
    t_rem = t2_end - time.monotonic()
    if t_rem > 60:
        t_alloc = min(100.0, 0.7 * t_rem)
        p, sc = _aggressive_freq_iterate(best_overall, rng, t_alloc)
        _try_update(p, sc)

    print(f"  Phase 2 done. Best: {best_score_overall:.6f} (n={len(best_overall)})")

    # ========== PHASE 3: Different lengths (20% time) ==========
    print("Phase 3: Different lengths")
    t3_end = start_time + 0.65 * budget

    n_best = len(best_overall)
    candidate_lengths = []
    for nl in [n_best // 2, n_best, n_best * 2, n_best * 4]:
        if 512 <= nl <= 16384 and nl != n_best:
            candidate_lengths.append(nl)

    for nl in candidate_lengths[:3]:
        t_rem = t3_end - time.monotonic()
        if t_rem < 40:
            break
        t_alloc = min(120.0, 0.7 * t_rem)
        p_init = _resample(best_overall, nl, rng)
        if p_init is None:
            p_init = _init_cosine(nl, rng)
        print(f"    Trying n={nl}, alloc={t_alloc:.0f}s")
        p, sc = _optimize_adam(p_init, rng, t_alloc, K_start=5, K_end=50,
                               lr_start=0.05, lr_end=0.01, verbose=False)
        _try_update(p, sc)

    print(f"  Phase 3 done. Best: {best_score_overall:.6f} (n={len(best_overall)})")

    # ========== PHASE 4: Intensive polishing (25% time) ==========
    print("Phase 4: Intensive polishing")
    t4_end = start_time + 0.90 * budget

    n_polish = len(best_overall)

    # Multiple Adam passes with different params
    for pass_idx in range(4):
        t_rem = t4_end - time.monotonic()
        if t_rem < 30:
            break
        t_alloc = min(180.0, 0.5 * t_rem)
        # Vary parameters each pass
        k_s = [3, 15, 8, 25][pass_idx % 4]
        k_e = [60, 100, 50, 80][pass_idx % 4]
        lr_s = [0.05, 0.03, 0.07, 0.04][pass_idx % 4]
        lr_e = [0.008, 0.005, 0.01, 0.006][pass_idx % 4]

        p_start = best_overall.copy()
        if pass_idx > 0:
            # Add small perturbation for diversity
            p_start = p_start * np.exp(rng.normal(0, 0.02, n_polish))
            p_start = _sanitize(p_start)
            if p_start is None:
                p_start = best_overall.copy()

        p, sc = _optimize_adam(p_start, rng, t_alloc, K_start=k_s, K_end=k_e,
                               lr_start=lr_s, lr_end=lr_e, verbose=(pass_idx == 0))
        _try_update(p, sc)

    print(f"  Phase 4 done. Best: {best_score_overall:.6f} (n={len(best_overall)})")

    # ========== PHASE 5: Final block refinement (5% time) ==========
    print("Phase 5: Final block refinement")

    if best_overall is not None:
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(80.0, 0.7 * t_rem)
            p, sc = _block_refine(best_overall, rng, t_alloc)
            _try_update(p, sc)

        # One final quick Adam pass
        t_rem = deadline - time.monotonic()
        if t_rem > 15:
            t_alloc = min(60.0, 0.8 * t_rem)
            p, sc = _optimize_adam(best_overall, rng, t_alloc, K_start=1, K_end=10,
                                   lr_start=0.02, lr_end=0.003, verbose=False)
            _try_update(p, sc)

    # ========== Final output ==========
    if best_overall is None:
        best_overall = _init_uniform(2048)

    p_final = _sanitize(best_overall)
    if p_final is None:
        p_final = _init_uniform(2048)

    final_score = _score(p_final)
    try:
        eval_score = float(evaluate_sequence([float(x) for x in p_final]))
        if np.isfinite(eval_score):
            final_score = min(final_score, eval_score)
    except Exception:
        pass

    print(f"\n{'=' * 60}")
    print(f"FINAL: score={final_score:.6f} (n={len(p_final)})")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    out = np.where(np.isfinite(p_final), p_final, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = _init_uniform(2048)
    else:
        out = out / s

    return [float(x) for x in out]
