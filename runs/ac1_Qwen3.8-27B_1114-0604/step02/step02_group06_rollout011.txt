7. `a_new = irfft(A_new, L)[:n]`
8. Project `a_new` onto {a >= 0, sum(a)=1}

   b. Find `m = argmax(c)`

   f. Accept if score improves

def _gs_flatten(a, L, gamma=0.7, n_steps=1):
    """GS-style spectral flattening."""
    n = a.size
    for _ in range(n_steps):
        F = np.fft.rfft(a, L)
        c = np.fft.irfft(F * F, L)[:2*n-1]
        c = np.maximum(c, 0)
        c_mean = c.mean()
        # Flatten: c_new = c^gamma * c_mean^(1-gamma)
        c_new = np.power(c + 1e-30, gamma) * np.power(c_mean + 1e-30, 1 - gamma)
        # Pad back
        c_pad = np.zeros(L)
        c_pad[:2*n-1] = c_new
        # New power spectrum
        P_new = np.fft.rfft(c_pad)
        P_new = np.maximum(P_new.real, 0)
        # Keep phase
        mag = np.abs(F)
        phase = np.where(mag > 1e-30, F / mag, 1.0 + 0j)
        A_new = np.sqrt(P_new) * phase
        a_new = np.fft.irfft(A_new, L)[:n]

        s = a_new.sum()
        if s > 0 and np.isfinite(s):
            a_new = a_new / s
        else:
            a_new = a.copy()
    return a_new
