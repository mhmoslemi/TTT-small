"""
Hybrid optimizer combining L-BFGS-B with smooth log-sum-exp surrogate
and gradient-based refinement for minimizing 2n*max(conv(a,a))/(sum a)^2.
"""

import numpy as np
import time
from scipy.optimize import minimize

def fast_self_conv(a):
    """Self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(n, c, S):
    """Objective value."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def smooth_obj_grad(a, t):
    """
    Smooth surrogate objective and its gradient.
    Uses log-sum-exp with temperature t to approximate max.
    """
    n = len(a)
    S = float(a.sum())
    if S < 1e-10:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    m = float(c.max())
    if m <= 0:
        return 0.0, np.zeros(n)
    
    shifted = t * (c - m)
    shifted = np.clip(shifted, -500, 0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    M_approx = m + np.log(sum_exp) / t
    weights = exp_shifted / sum_exp
    
    objective = 2.0 * n * M_approx / (S * S)
    
    # Compute cross-correlation: cross_corr[j] = sum_m a[m] * weights[j+m]
    a_rev = a[::-1]
    M_fft = 1 << (3 * n - 2).bit_length()
    A_rev_fft = np.fft.rfft(a_rev, M_fft)
    W_fft = np.fft.rfft(weights, M_fft)
    conv_full = np.fft.irfft(A_rev_fft * W_fft, M_fft)
    cross_corr = conv_full[n-1:n-1+n]
    
    grad = (4.0 * n / (S * S)) * cross_corr - (4.0 * n * M_approx / (S ** 3)) * a
    
    return objective, grad

def lfgs_phase(a0, end_time, n, rng, t_start, t_end, t_steps):
    """
    Run L-BFGS-B with annealing temperature.
    """
    bounds = [(0.0, 1000.0)] * n
    a = a0.copy()
    best_score = np.inf
    best_a = a.copy()
    
    # Compute initial score with exact max
    c0 = fast_self_conv(a)
    S0 = float(a.sum())
    score0 = compute_score(n, c0, S0)
    if score0 < best_score:
        best_score = score0
        best_a = a.copy()
    
    # Annealing schedule
    if t_steps > 1:
        temps = np.geomspace(t_start, t_end, t_steps)
    else:
        temps = [t_end]
    
    for t in temps:
        if time.monotonic() >= end_time:
            break
        
        def func_and_grad(x):
            val, g = smooth_obj_grad(x, t)
            return val, g
        
        result = minimize(
            func_and_grad, a, jac=True, method='L-BFGS-B',
            bounds=bounds, options={
                'maxiter': 300,
                'maxfun': 500,
                'ftol': 1e-10,
                'gtol': 1e-8,
                'iprint': -1
            }
        )
        
        a = np.clip(result.x, 0.0, 1000.0)
        
        # Check with exact objective
        c = fast_self_conv(a)
        S = float(a.sum())
        score = compute_score(n, c, S)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"  L-BFGS-B t={t:.0f}: score={score:.6f}")
        else:
            # Revert to best
            a = best_a.copy()
            print(f"  L-BFGS-B t={t:.0f}: no improvement, best={best_score:.6f}")
    
    return best_score, best_a

            a = best_a * (1.0 + sc * rng.standard_normal(n))

            m[:] = 0.0
            v[:] = 0.0
            t_count = 0
            lr = max(0.0001, lr * 0.5)
            no_imp = 0
            continue

        k_act = min(topk, c.size)
        part = np.argpartition(c, -k_act)[-k_act:]
        vals = c[part].copy()
        order = np.argsort(vals)[::-1]
        part = part[order]

            js = max(0, ki - n + 1)

                continue

            # So grad_c[j] += wi * 2 * a[ki-j] for j in [js, je]
            # ki-j ranges from ki-je to ki-js
            # We need a[ki-j] for j in [js, je]
            # ki-j ranges from ki-js to ki-je
            # a indices: ki-js to ki-je, but we need 0 <= index < n
            # Since js = max(0, ki-n+1) and je = min(ki, n-1)
            # ki - js <= ki - 0 = ki < 2n (but we need < n)
            # ki - je >= ki - (n-1) >= 0 (since ki >= n-1 when je = n-1)
            # Let me just compute it directly:
            a_slice = a[ki - je: ki - js + 1][::-1]  # a[ki-j] for j from js to je
            # Wait: when j=js, ki-j = ki-js. When j=je, ki-j = ki-je.
            # So a[ki-j] for j in [js, je] is a[ki-je], a[ki-je+1], ..., a[ki-js]
            # That's a[ki-je : ki-js+1]
            a_vals = a[ki - je: ki - js + 1]
            grad_c[js:je + 1] += wi * 2.0 * a_vals
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (grad_c / S2 - 2.0 * M / S3)

        v[:] = 0.999 * v + 0.001 * grad * grad
        a -= lr * (m / (1.0 - t1)) / (np.sqrt(v / (1.0 - t2)) + 1e-8)
        a = np.clip(a, 0.0, 1000.0)
    
    return best_score, best_a

import numpy as np
import time
from scipy.optimize import minimize


def fast_self_conv(a):
    """Full self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, c, S):
    """Objective value given n, convolution array c, and sum S."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def smooth_obj_grad(a, t):
    """
    Smooth surrogate objective using log-sum-exp with temperature t.
    Returns (objective_value, gradient_vector).
    """
    n = len(a)
    S = float(a.sum())
    if S < 1e-10:
        return 1e10, np.zeros(n)
    c = fast_self_conv(a)
    m = float(c.max())
    if m <= 1e-15:
        return 0.0, np.zeros(n)

    shifted = t * (c - m)
    shifted = np.clip(shifted, -500.0, 0.0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(exp_shifted.sum())
    M_approx = m + np.log(sum_exp) / t
    weights = exp_shifted / sum_exp

    objective = 2.0 * n * M_approx / (S * S)

    # Cross-correlation: cross[j] = sum_m a[m] * weights[j+m] for j=0..n-1
    # Computed via: cross[j] = conv(a_rev, weights)[j + n - 1]
    a_rev = a[::-1]
    M_fft = 1 << (3 * n - 2).bit_length()
    A_rev_fft = np.fft.rfft(a_rev, M_fft)
    W_fft = np.fft.rfft(weights, M_fft)
    conv_full = np.fft.irfft(A_rev_fft * W_fft, M_fft)
    cross = conv_full[n - 1:n - 1 + n]

    grad = (4.0 * n / (S * S)) * cross - (4.0 * n * M_approx / (S * S * S)) * a
    return objective, grad


def lfgs_anneal(a0, end_time, n, t_start, t_end, n_steps, maxiter_per_step):
    """
    Run L-BFGS-B with geometrically annealed temperature.
    Returns (best_exact_score, best_array).
    """
    bounds = [(0.0, 1000.0)] * n
    a = a0.astype(np.float64).copy()
    best_score = np.inf
    best_a = a.copy()

    # Initial exact score
    c0 = fast_self_conv(a)
    S0 = float(a.sum())
    s0 = compute_score(n, c0, S0)
    if np.isfinite(s0):
        best_score = s0
        best_a = a.copy()

    temps = np.geomspace(t_start, t_end, n_steps)

    for idx_t, t in enumerate(temps):
        if time.monotonic() >= end_time:
            break

        def func_wrap(x):
            val, g = smooth_obj_grad(x, t)
            return val, g

        try:
            result = minimize(
                func_wrap, a, jac=True, method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': maxiter_per_step, 'ftol': 1e-12,
                         'gtol': 1e-9, 'iprint': -1, 'maxfun': maxiter_per_step * 2}
            )
            a_new = np.clip(result.x, 0.0, 1000.0)
        except Exception as e:
            print(f"    L-BFGS-B failed at t={t:.1f}: {e}")
            a_new = a.copy()

        # Evaluate exact objective
        c_new = fast_self_conv(a_new)
        S_new = float(a_new.sum())
        score_new = compute_score(n, c_new, S_new)

        if np.isfinite(score_new) and score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new.copy()
            print(f"    t={t:.1f}: score={score_new:.7f} (improved)")
        else:
            a = best_a.copy()
            print(f"    t={t:.1f}: score={score_new:.7f} (no improvement, best={best_score:.7f})")

    return best_score, best_a
