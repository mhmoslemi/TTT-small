import time

def autocorr_fft(f):
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    result = np.fft.irfft(F * F, size)
    return result[:2 * n - 1]

def active_set_gradient(f, c, tol=0.0):
    """Compute averaged subgradient over active set."""
    n = len(f)
    max_c = c.max()
    if tol == 0.0:
        active = np.where(c >= max_c - 1e-10)[0]
    else:
        active = np.where(c >= max_c * (1 - tol))[0]
    
    grad = np.zeros(n)
    for k in active:
        lo = max(0, int(k) - n + 1)
        hi = min(n, int(k) + 1)
        grad[lo:hi] += f[int(k) - np.arange(lo, hi)]
    grad /= len(active)
    return grad, len(active)

def mirror_step(f, grad, eta):
    """Exponentiated gradient step."""
    log_f = np.log(np.maximum(f, 1e-300)) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    return f_new / s

def _autocorr_fft(f):

    F = np.fft.rfft(f, size)
    result = np.fft.irfft(F * F, size)
    return result[:2 * n - 1]

                    eta = min(eta * 1.3, 20.0)
                else:
                    eta = eta_try
                accepted = True
                break
        if not accepted:
            eta *= 0.5
        
        f = best_f_try
        iter_count += 1
        
        if verbose and iter_count % 10000 == 0:
            print(f"    MD iter {iter_count}: peak={cur_peak:.10f}, best={best_peak:.10f}, "
                  f"eta={eta:.3f}, n_active={n_active}, tol={tol:.4f}")
    
    c = _autocorr_fft(f)
    if c.max() < best_peak:
        best_peak = c.max()
        best_f = f.copy()
    
    return best_f, 2.0 * n * best_peak


def _kkt_polish(f0, deadline, rng, n, verbose=True):
    """
    KKT-based active set polishing. Solves the linear KKT system for the
    current active set and support, then iterates.
    """
    f = f0.copy()
    f = np.maximum(f, 0.0)
    f /= f.sum()
    
    c = _autocorr_fft(f)
    best_peak = c.max()
    best_f = f.copy()
    iteration = 0
    
    while time.monotonic() < deadline - 5:
        c = _autocorr_fft(f)
        cur_peak = c.max()
        
        if cur_peak < best_peak - 1e-14:
            best_peak = cur_peak
            best_f = f.copy()
        
        # Determine active set (all lags within 1% of max, or top 50)
        max_c = c.max()
        threshold = max_c * 0.999
        active_set = np.where(c >= threshold)[0]
        if len(active_set) > 100:
            # Keep only the top 100
            top_idx = np.argsort(c)[-100:]
            active_set = top_idx
        elif len(active_set) == 0:
            active_set = np.array([int(np.argmax(c))])
        
        # Determine support (indices where f > 1e-10)
        support = np.where(f > 1e-10)[0]
        if len(support) < 2:
            support = np.where(f > 0)[0]
        if len(support) < 2:
            break
        
        m = len(support)
        A = len(active_set)
        
        # Build the KKT matrix
        # M[j, i] = 1 if support[j] + support[i] is in active_set
        active_set_set = set(active_set.tolist())
        
        # Precompute for efficiency
        support_arr = support.astype(np.int64)
        # For each pair (j, i), check if support[j] + support[i] in active_set_set
        # This is an outer-sum check
        sums = support_arr[:, None] + support_arr[None, :]
        M = np.zeros((m, m))
        for idx in range(len(sums)):
            row = sums.flatten()
            mask = np.isin(row, active_set)
            M[idx // m, :] = mask[:m]
            # Actually this is getting complicated, let me do it differently
        
        # Simpler: build M row by row
        M = np.zeros((m, m))
        for j in range(m):
            sj = support[j]
            for i in range(m):
                si = support[i]
                if (sj + si) in active_set_set:
                    M[j, i] = 1.0
        
        # Full KKT system:
        # [M   -A*1] [f_S]   [0]
        # [1^T  0  ] [mu'] = [1]
        
        full_M = np.zeros((m + 1, m + 1))
        full_M[:m, :m] = M
        full_M[:m, m] = -float(A)
        full_M[m, :m] = 1.0
        full_M[m, m] = 0.0
        
        rhs = np.zeros(m + 1)
        rhs[m] = 1.0
        
        try:
            sol = np.linalg.solve(full_M, rhs)
        except np.linalg.LinAlgError:
            try:
                sol = np.linalg.lstsq(full_M, rhs, rcond=None)[0]
            except:
                break
        
        f_new = sol[:m]
        
        # Check validity: all non-negative
        if np.any(f_new < -1e-10):
            # Try with a slightly different active set (add more lags)
            threshold = max_c * 0.995
            active_set2 = np.where(c >= threshold)[0]
            if len(active_set2) > 100:
                active_set2 = np.argsort(c)[-100:]
            if len(active_set2) > len(active_set):
                # Retry with larger active set
                active_set_set2 = set(active_set2.tolist())
                A2 = len(active_set2)
                M2 = np.zeros((m, m))
                for j in range(m):
                    sj = support[j]
                    for i in range(m):
                        si = support[i]
                        if (sj + si) in active_set_set2:
                            M2[j, i] = 1.0
                full_M2 = np.zeros((m + 1, m + 1))
                full_M2[:m, :m] = M2
                full_M2[:m, m] = -float(A2)
                full_M2[m, :m] = 1.0
                try:
                    sol2 = np.linalg.solve(full_M2, rhs)
                except:
                    sol2 = np.linalg.lstsq(full_M2, rhs, rcond=None)[0]
                f_new = sol2[:m]
                if np.any(f_new < -1e-10):
                    # Fallback: blend with current
                    f_new = 0.5 * f_new + 0.5 * f[support]
            
            f_new = np.maximum(f_new, 0.0)
            if f_new.sum() < 1e-10:
                break
            f_new /= f_new.sum()
        
        f_new = np.maximum(f_new, 0.0)
        if f_new.sum() < 1e-10:
            break
        f_new /= f_new.sum()
        
        # Construct full solution
        f_try = np.zeros(n)
        f_try[support] = f_new
        f_try /= f_try.sum()
        
        # Evaluate
        c_try = _autocorr_fft(f_try)
        peak_try = c_try.max()
        
        if peak_try < cur_peak - 1e-14:
            f = f_try
        else:
            # Try a damped update
            for alpha in [0.5, 0.3, 0.1]:
                f_damped = (1 - alpha) * f + alpha * f_try
                f_damped = np.maximum(f_damped, 0.0)
                f_damped /= f_damped.sum()
                c_d = _autocorr_fft(f_damped)
                if c_d.max() < cur_peak - 1e-14:
                    f = f_damped
                    break
            else:
                # No improvement; try perturbation
                f = f + 0.001 / np.sqrt(n) * rng.standard_normal(n)
                f = np.maximum(f, 0.0)
                f /= f.sum()
        
        iteration += 1
        if verbose and iteration % 20 == 0:
            c_now = _autocorr_fft(f)
            print(f"    KKT iter {iteration}: peak={c_now.max():.10f}, best={best_peak:.10f}, "
                  f"|A|={A}, |S|={m}")
    
    c = _autocorr_fft(f)
    if c.max() < best_peak:
        best_peak = c.max()
        best_f = f.copy()
    
    return best_f, 2.0 * n * best_peak


def _coordinate_polish(f0, deadline, rng, n):
    """
    Fast coordinate-level polish: for the peak lag, try moving mass
    from high-contributing to low-contributing coordinates.
    """
    f = f0.copy()
    f = np.maximum(f, 0.0)
    f /= f.sum()
    
    c = _autocorr_fft(f)
    best_peak = c.max()
    best_f = f.copy()
    passes = 0
    
    while time.monotonic() < deadline:
        c = _autocorr_fft(f)
        cur_peak = c.max()
        
        if cur_peak < best_peak - 1e-14:
            best_peak = cur_peak
            best_f = f.copy()
        
        # Find peak lag
        k_star = int(np.argmax(c))
        
        # Contribution of each coordinate to c[k_star]
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        contrib = np.zeros(n)
        contrib[lo:hi] = 2.0 * f[k_star - np.arange(lo, hi)]
        
        # Top contributors and bottom contributors
        top_donors = np.argsort(contrib)[-10:]
        bottom_receivers = np.argsort(contrib)[:max(10, n // 40)]
        
        improved = False
        
        # Try moving from top donors to bottom receivers
        for donor in top_donors:
            if improved:
                break
            for recv in bottom_receivers:
                if recv == donor:
                    continue
                for frac in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    amt = f[int(donor)] * frac
                    if amt < 1e-15:
                        continue
                    f_try = f.copy()
                    f_try[int(donor)] -= amt
                    f_try[int(recv)] += amt
                    f_try /= f_try.sum()
                    c_try = _autocorr_fft(f_try)
                    if c_try.max() < cur_peak - 1e-14:
                        f = f_try
                        improved = True
                        break
        
        # Random perturbation every 100 passes
        if not improved and passes % 100 == 0:
            f_try = f + 0.002 / np.sqrt(n) * rng.standard_normal(n)
            f_try = np.maximum(f_try, 0.0)
            f_try /= f_try.sum()
            c_try = _autocorr_fft(f_try)
            if c_try.max() < cur_peak - 1e-14:
                f = f_try
        
        passes += 1
        if passes % 500 == 0:
            print(f"    Coord pass {passes}: peak={cur_peak:.10f}, best={best_peak:.10f}")
    
    return best_f, 2.0 * n * best_peak


def _spectral_smooth(f, n, alpha=0.5):
    """
    Smooth the spectrum of f to make |F|^2 flatter, then project back.
    This is a heuristic that can help escape local minima.
    """
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(f, size)
    mag = np.abs(F)
    phase = np.angle(F)
    
    # Smooth the magnitude spectrum
    # Blend towards a flat spectrum (keeping DC)
    target_mag = np.ones_like(mag)
    target_mag[0] = mag[0]  # Keep DC
    mag_smooth = alpha * mag + (1 - alpha) * target_mag * (mag.sum() / target_mag.sum())
    mag_smooth[0] = F[0].real  # Preserve DC exactly
    
    # Reconstruct
    F_new = mag_smooth * np.exp(1j * phase)
    f_new = np.fft.irfft(F_new, size)[:n]
    
    # Project to non-negative simplex
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f
    f_new /= s
    return f_new


def _generate_diverse_starts(n, rng, count=30):
    """Generate diverse starting points for multi-start optimization."""
    starts = []
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    # Gaussian bells
    for alpha in [0.2, 0.5, 1.0, 2.0, 4.0, 8.0]:
        starts.append(np.exp(-alpha * ((j - center) / center) ** 2))
    
    # U-shaped
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        starts.append(1.0 + alpha * ((j - center) / center) ** 2)
    
    # Bimodal
    for pos in [0.2, 0.25, 0.3, 0.35]:
        w = 0.05 * n
        starts.append(
            np.exp(-((j - pos * n) / w) ** 2) +
            np.exp(-((j - (1 - pos) * n) / w) ** 2) + 0.001
        )
    
    # Sine-based
    for freq in [1, 2, 3]:
        starts.append(1.0 + 0.5 * np.sin(2 * np.pi * freq * j / n))
        starts.append(1.0 + 0.8 * np.abs(np.cos(2 * np.pi * freq * j / n)))
    
    # Triangle
    starts.append(1.0 - np.abs(j - center) / center)
    starts.append(np.abs(j - center) / center + 0.1)
    
    # Random exponential
    for _ in range(count - len(starts)):
        r = rng.random()
        if r < 0.3:
            starts.append(rng.exponential(1.0, n) + 0.01)
        elif r < 0.5:
            starts.append(np.abs(rng.standard_normal(n)) + 0.1)
        elif r < 0.7:
            starts.append(np.ones(n) * (1.0 + 0.3 * rng.standard_normal(n)))
        elif r < 0.85:
            # Sparse
            base = np.zeros(n)
            nnz = n // 4
            idx = rng.choice(n, nnz, replace=False)
            base[idx] = rng.exponential(1.0, nnz)
            starts.append(base + 0.001)
        else:
            starts.append(rng.dirichlet(np.ones(n) * 0.1) * n)
    
    # Normalize all
    result = []
    for s in starts:
        s = np.maximum(s, 0.0)
        ssum = s.sum()
        if ssum > 1e-10:
            result.append(s / ssum)
        else:
            result.append(np.ones(n) / n)
    return result


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the non-negative sequence minimizing 2*n*max(f*f)/(sum f)^2.
    Uses adaptive mirror descent + KKT active set polishing + coordinate descent.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 10.0
    
    # Load initial sequence
    f_init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(f_init)
    s0 = f_init.sum()
    if s0 < 1e-10:
        f_init = np.ones(n)
    else:
        f_init = f_init / s0
    
    print(f"Sequence length: {n}")
    c0 = _autocorr_fft(f_init)
    init_score = 2.0 * n * c0.max()
    print(f"Initial peak: {c0.max():.10f}, initial score: {init_score:.6f}")
    
    best_score = init_score
    best_f = f_init.copy()
    
    # Generate diverse starts and score them
    t_start = time.monotonic()
    starts = _generate_diverse_starts(n, rng, count=40)
    # Add the initial sequence as a start
    starts.insert(0, f_init.copy())
    
    scored = []
    for f0 in starts:
        c = _autocorr_fft(f0)
        scored.append((c.max(), f0))
    scored.sort(key=lambda x: x[0])
    
    gen_time = time.monotonic() - t_start
    print(f"Generated {len(starts)} starts in {gen_time:.1f}s")
    print(f"Top 5 peaks: {[f'{p:.10f}' for p, _ in scored[:5]]}")
    print(f"Top 5 scores: {[f'{2*n*p:.6f}' for p, _ in scored[:5]]}")
    
    # Update best if a start is better
    if scored[0][0] < c0.max():
        best_score = 2.0 * n * scored[0][0]
        best_f = scored[0][1].copy()
        print(f"Better start found! Score: {best_score:.6f}")
    
    # Time allocation
    remaining_main = deadline - time.monotonic()
    
    # Phase 1: Mirror descent on top 4 starts (50% of remaining time)
    main_time = remaining_main * 0.45
    polish_time = remaining_main * 0.30
    polish2_time = remaining_main * 0.15
    final_time = remaining_main * 0.10
    
    print(f"\n=== Phase 1: Mirror Descent (budget {main_time:.0f}s) ===")
    num_starts = 4
    time_per_start = main_time / num_starts
    
    for idx in range(num_starts):
        remaining_all = deadline - time.monotonic()
        alloc = min(time_per_start, remaining_all - polish_time - polish2_time - final_time - 20)
        if alloc < 30:
            break
        
        peak0, f0 = scored[idx]
        print(f"\n  MD start {idx+1}: peak={peak0:.10f}, score={2*n*peak0:.6f}, budget={alloc:.0f}s")
        
        t_phase = time.monotonic()
        f_opt, score = _optimize_mirror(f0, t_phase + alloc, rng, n, verbose=True)
        
        if score < best_score:
            best_score = score
            best_f = f_opt.copy()
            print(f"  *** IMPROVED to {best_score:.6f} ***")
        else:
            print(f"  Best from this start: {score:.6f}")
    
    # Phase 2: KKT polishing (30% of time)
    remaining_all = deadline - time.monotonic()
    polish_budget = min(polish_time, remaining_all - polish2_time - final_time - 15)
    print(f"\n=== Phase 2: KKT Polish (budget {polish_budget:.0f}s) ===")
    print(f"  Starting score: {best_score:.6f}")
    
    t_phase = time.monotonic()
    f_kkt, kkt_score = _kkt_polish(best_f, t_phase + polish_budget, rng, n, verbose=True)
    
    if kkt_score < best_score:
        best_score = kkt_score
        best_f = f_kkt.copy()
        print(f"  *** KKT POLISHED to {best_score:.6f} ***")
    else:
        print(f"  KKT best: {kkt_score:.6f}")
    
    # Phase 3: Coordinate polish (15%)
    remaining_all = deadline - time.monotonic()
    coord_budget = min(polish2_time, remaining_all - final_time - 10)
    print(f"\n=== Phase 3: Coordinate Polish (budget {coord_budget:.0f}s) ===")
    
    t_phase = time.monotonic()
    f_coord, coord_score = _coordinate_polish(best_f, t_phase + coord_budget, rng, n)
    
    if coord_score < best_score:
        best_score = coord_score
        best_f = f_coord.copy()
        print(f"  *** COORD POLISHED to {best_score:.6f} ***")
    else:
        print(f"  Coord best: {coord_score:.6f}")
    
    # Phase 4: Final optimization round (10%)
    remaining_all = deadline - time.monotonic()
    if remaining_all > 30:
        print(f"\n=== Phase 4: Final MD round (budget {remaining_all:.0f}s) ===")
        t_phase = time.monotonic()
        f_fin, fin_score = _optimize_mirror(best_f, min(t_phase + remaining_all - 5, deadline), rng, n, verbose=True)
        
        if fin_score < best_score:
            best_score = fin_score
            best_f = f_fin.copy()
            print(f"  *** FINAL IMPROVED to {best_score:.6f} ***")
        else:
            print(f"  Final MD: {fin_score:.6f}")
    
    # Final KKT polish if time remains
    remaining_all = deadline - time.monotonic()
    if remaining_all > 30:
        print(f"\n=== Final KKT Polish ({remaining_all:.0f}s) ===")
        t_phase = time.monotonic()
        f_final, final_kkt = _kkt_polish(best_f, min(t_phase + remaining_all - 5, deadline), rng, n, verbose=False)
        if final_kkt < best_score:
            best_score = final_kkt
            best_f = f_final.copy()
            print(f"  *** FINAL KKT: {best_score:.6f} ***")
    
    # Verify with official evaluator
    best_f = np.clip(best_f, 0.0, 1000.0)
    official_score = evaluate_sequence(best_f.tolist())
    print(f"\nOfficial evaluation: {official_score:.6f}")
    
    if not np.isfinite(official_score):
        print("WARNING: Invalid result, falling back to initial sequence")
        best_f = f_init.copy()
        official_score = evaluate_sequence(best_f.tolist())
        print(f"Fallback score: {official_score:.6f}")
    
    if official_score < best_score:
        best_score = official_score
    
    print(f"\n{'='*50}")
    print(f"=== FINAL SCORE: {best_score:.6f} ===")
    print(f"{'='*50}")
    
    return [float(x) for x in best_f]
