def _autocorr_fft(f):
    """Linear autocorrelation (full convolution) of f with itself via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)

"""
Step function optimizer minimizing 2*n*max(convolve(f,f))/(sum f)^2.
Uses soft-max (log-sum-exp) smooth approximation with mirror descent,
adaptive temperature annealing, and coordinate descent polish.
Key improvement: soft-max gradient naturally weights ALL peaks proportionally,
avoiding the "top-k" bias and providing a smoother optimization landscape.
"""

import numpy as np
import time


def _autocorr_fft(f):
    """Full auto-convolution of f via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    result = np.fft.irfft(F * F, size)
    return result[:2 * n - 1]


def _soft_max_weights(c, T):
    """Compute softmax weights for the peaks of c."""
    c_s = c / T
    c_s -= c_s.max()
    exp_c = np.exp(c_s)
    return exp_c / exp_c.sum()


def _grad_smax(f, w):
    """
    Gradient of T*log(sum(exp(c[k]/T))) w.r.t. f,
    where c = convolve(f,f) and w = softmax(c/T).
    grad[i] = 2 * sum_k w[k] * f[k-i]
    Computed via FFT cross-correlation.
    """
    n = len(f)
    # L must be >= (2n-1) + n - 1 = 3n-2 to avoid circular aliasing
    L = 1 << (3 * n - 2).bit_length()
    
    w_pad = np.zeros(L)
    w_pad[:2*n-1] = w
    f_pad = np.zeros(L)
    f_pad[:n] = f
    
    W = np.fft.rfft(w_pad)
    F = np.fft.rfft(f_pad)
    R = np.fft.irfft(np.conj(W) * F, L)
    
    return 2.0 * R[:n]


def _mirror_step(f, grad, eta):
    """Exponentiated gradient step on the simplex."""
    # f_new proportional to f * exp(-eta * grad)
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy()
    return f_new / s


def _score(f, n):
    """Compute the objective value."""
    c = _autocorr_fft(f)
    return 2.0 * n * c.max(), c


def _optimize_smax_mirror(f0, deadline, rng, n):
    """
    Optimize using soft-max log-sum-exp mirror descent with adaptive temperature.
    """
    f = f0.copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    f /= s
    
    c = _autocorr_fft(f)
    peak = c.max()
    local_best = peak
    local_best_f = f.copy()
    
    # Soft-max temperature: start high, decrease over time
    T = peak * 0.1  # Initial temperature (10% of peak)
    T_min = peak * 1e-6
    T_decay = 0.9999  # Cooling rate
    
    eta = 0.5  # Step size
    last_improve = time.monotonic()
    iter_count = 0
    
    while time.monotonic() < deadline:
        c = _autocorr_fft(f)
        cur_peak = c.max()
        
        if cur_peak < local_best - 1e-14:
            local_best = cur_peak
            local_best_f = f.copy()
            last_improve = time.monotonic()
        
        stuck_time = time.monotonic() - last_improve
        
        # Adaptive perturbations when stuck
        if stuck_time > 10:
            mag = min(0.05, 0.005 + 0.003 * (stuck_time / 30.0))
            f = f + mag / np.sqrt(n) * rng.standard_normal(n)
            f = np.maximum(f, 0.0)
            f /= f.sum()
            last_improve = time.monotonic()
            T = max(T, peak * 0.1)  # Reheat
            iter_count += 1
            continue
        
        # Cooling
        T = max(T_min, T * T_decay)
        
        # Compute soft-max weights and gradient
        w = _soft_max_weights(c, T)
        grad = _grad_smax(f, w)
        
        # Normalize gradient for stable step size
        grad_norm = np.sqrt(np.sum(grad**2))
        if grad_norm > 1e-15:
            grad = grad / grad_norm
        
        # Mirror descent step
        f_new = _mirror_step(f, grad, eta)
        
        # Evaluate
        c_new = _autocorr_fft(f_new)
        new_peak = c_new.max()
        
        delta = new_peak - cur_peak
        if delta <= 0:
            f = f_new
            eta = min(eta * 1.01, 10.0)
        else:
            # SA acceptance
            if T > 1e-30 and np.log(max(rng.uniform(), 1e-300)) < -delta / (T * 0.1):
                f = f_new
            else:
                eta *= 0.9
        
        iter_count += 1
        if iter_count % 50000 == 0:
            print(f"    SMax iter {iter_count}: peak={cur_peak:.10f}, best={local_best:.10f}, "
                  f"T={T:.2e}, eta={eta:.4f}")
    
    c = _autocorr_fft(f)
    if c.max() < local_best:
        local_best = c.max()
        local_best_f = f.copy()
    
    return local_best_f, 2.0 * n * local_best


def _polish(f, deadline, rng, n):
    """
    Coordinate descent polish using pair moves and single-coordinate adjustments.
    """
    f = f.copy()
    f = np.maximum(f, 0.0)
    f /= f.sum()
    
    c = _autocorr_fft(f)
    local_best = c.max()
    local_best_f = f.copy()
    passes = 0
    
    while time.monotonic() < deadline:
        c = _autocorr_fft(f)
        cur_peak = c.max()
        
        if cur_peak < local_best - 1e-14:
            local_best = cur_peak
            local_best_f = f.copy()
        
        k_star = int(np.argmax(c))
        
        # Gradient direction at the peak
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        contrib = 2.0 * f[k_star - np.arange(lo, hi)]
        
        # Try to reduce the peak by moving mass from high-contrib to low-contrib
        # Expand the gradient to full n
        full_contrib = np.zeros(n)
        full_contrib[lo:hi] = contrib
        
        donor = int(np.argmax(full_contrib))
        
        # Find low-contribution receivers
        # Use all peaks for a better gradient
        top_peaks = np.argsort(c)[-min(20, 2*n-1):]
        full_grad = np.zeros(n)
        for k in top_peaks:
            lo_k = max(0, k - n + 1)
            hi_k = min(n, k + 1)
            full_grad[lo_k:hi_k] += f[k - np.arange(lo_k, hi_k)] / len(top_peaks)
        
        recv_candidates = np.argsort(full_grad)[:20]
        
        improved = False
        for frac in [0.01, 0.03, 0.08, 0.15, 0.3]:
            for recv in recv_candidates[:10]:
                if recv == donor:
                    continue
                amt = f[donor] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[donor] -= amt
                f_try[recv] += amt
                c_try = _autocorr_fft(f_try)
                if c_try.max() < cur_peak - 1e-15:
                    f = f_try
                    improved = True
                    break
            if improved:
                break
        
        # Random pair moves when stuck
        if not improved:
            for _ in range(20):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                frac = rng.uniform(0.01, 0.2)
                amt = f[i] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[i] -= amt
                f_try[j] += amt
                c_try = _autocorr_fft(f_try)
                if c_try.max() < cur_peak - 1e-15:
                    f = f_try
                    improved = True
                    break
        
        # Small global perturbation periodically
        if not improved and passes % 30 == 0:
            f_try = f + 0.001 / np.sqrt(n) * rng.standard_normal(n)
            f_try = np.maximum(f_try, 0.0)
            f_try /= f_try.sum()
            c_try = _autocorr_fft(f_try)
            if c_try.max() < cur_peak - 1e-15:
                f = f_try
        
        passes += 1
        
        if passes % 1000 == 0:
            print(f"    Polish pass {passes}: peak={cur_peak:.10f}, best={local_best:.10f}")
    
    c = _autocorr_fft(f)
    if c.max() < local_best:
        local_best = c.max()
        local_best_f = f.copy()
    
    return local_best_f, 2.0 * n * local_best


def _generate_starts(f_init, n, rng, count=30):
    """Generate diverse starting points."""
    starts = [f_init.copy()]
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    norm = center if center > 0 else 1.0
    
    # U-shaped (inverted Gaussian) - most promising for this problem
    for alpha in [0.3, 0.5, 1.0, 2.0, 3.0, 5.0, 10.0, 20.0]:
        starts.append(1.0 + alpha * ((j - center) / norm) ** 2)
    
    # Parabolic U
    for alpha in [1.0, 3.0, 10.0, 30.0]:
        starts.append(1.0 + alpha * ((2 * j / (n - 1) - 1)) ** 2)
    
    # Bimodal
    for pos in [0.15, 0.2, 0.25, 0.3]:
        sigma = 0.05 * n
        starts.append(
            np.exp(-((j - pos * n) / sigma) ** 2) +
            np.exp(-((j - (1 - pos) * n) / sigma) ** 2) + 0.001
        )
    
    # Sine-based
    for freq in [1, 2, 3, 4]:
        starts.append(1.0 + 0.5 * np.cos(2 * np.pi * freq * j / n))
        starts.append(1.0 + 0.8 * np.abs(np.cos(2 * np.pi * freq * j / n)))
    
    # Triangle U
    starts.append(0.5 + 0.5 * np.abs(2 * j / (n - 1) - 1))
    starts.append(0.3 + 0.7 * np.abs(2 * j / (n - 1) - 1) ** 3)
    
    # Random
    remaining = count - len(starts)
    for _ in range(max(0, remaining)):
        r = rng.random()
        if r < 0.3:
            starts.append(rng.exponential(1.0, n) + 0.01)
        elif r < 0.5:
            starts.append(np.abs(rng.standard_normal(n)) + 0.1)
        elif r < 0.7:
            starts.append(np.ones(n) + 0.2 * rng.standard_normal(n))
        else:
            starts.append(rng.dirichlet(np.ones(n) * 0.1) * n)
    
    # Clean up: ensure non-negative and normalized
    cleaned = []
    for f0 in starts:
        f0 = np.maximum(f0, 0.0)
        s = f0.sum()
        if s > 1e-10:
            cleaned.append(f0 / s)
        else:
            cleaned.append(np.ones(n) / n)
    
    return cleaned[:count]
