```python
def _mirror_descent(f, n, deadline, rng, beta_temp=5.0, verbose=False):
    """
    Mirror descent (exponential gradient) for minimizing max(conv(a,a)) 
    subject to a >= 0, sum(a) = 1.
    """
    f = f.copy()
    s = f.sum()
    f /= s
    
    best_score = _score(f)
    best_f = f.copy()
    
    step = 1.0
    patience = 0
    max_patience = 30
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        
        # Compute convolution
        c = _autocorr(f)
        c_max = c.max()
        
        # Softmax weights
        weights = np.exp(beta_temp * (c - c_max))
        weights /= weights.sum()
        
        # Gradient via FFT-based cross-correlation
        # grad[i] = 2 * sum_k weights[k] * f[k-i]
        size = 1 << (3 * n - 2).bit_length()
        W = np.fft.rfft(weights, size)
        F = np.fft.rfft(f, size)
        grad = np.fft.irfft(W * np.conj(F), size)[:n] * 2.0
        
        # Mirror step
        eta = step
        log_f = np.log(f + 1e-300) - eta * grad
        log_f -= log_f.max()
        f_new = np.exp(log_f)
        s_new = f_new.sum()
        if s_new < 1e-300:
            step *= 0.5
            continue
        f_new /= s_new
        
        # Evaluate
        c_new = _autocorr(f_new)
        peak_new = c_new.max()
        peak_old = c_max
        
        if peak_new < peak_old * (1 - 1e-12):
            f = f_new
            score = 2.0 * n * peak_new  # since sum = 1
            if score < best_score:
                best_score = score
                best_f = f.copy()
                patience = 0
            # Adaptive step
            improvement = (peak_old - peak_new) / peak_old
            if improvement > 0.01:
                step = min(step * 1.2, 5.0)
            else:
                step = step * 0.95
        else:
            step *= 0.5
            patience += 1
            if patience > max_patience:
                # Random perturbation
                mode = rng.integers(0, 4)
                j = np.arange(n, dtype=np.float64)
                if mode == 0:
                    alpha = rng.uniform(-1, 1)
                    tilt = np.exp(alpha * (j/n - 0.5))
                    f = f * tilt
                elif mode == 1:
                    freq = rng.integers(1, max(2, n//3) + 1)
                    phase = rng.uniform(0, 2*np.pi)
                    amp = rng.uniform(0.2, 0.6)
                    f = f * (1 + amp * np.sin(2*np.pi*freq*j/n + phase))
                elif mode == 2:
                    f = f + rng.uniform(-0.02, 0.02, n) * f.mean()
                else:
                    # Swap mass
                    i1 = rng.integers(0, n//2)
                    i2 = rng.integers(n//2, n)
                    amt = f[i1] * rng.uniform(0.1, 0.4)
                    f[i1] -= amt
                    f[i2] += amt
                
                f = np.maximum(f, 0)
                s = f.sum()
                if s < 1e-10:
                    f = np.ones(n)
                else:
                    f /= s
                step = 1.0
                patience = 0
    
    return best_f, best_score
```

```python
"""
Step function optimizer using adaptive mirror descent with FFT-based
gradient computation, softmax-weighted subgradients, and structured
perturbations. Includes LP-guided refinement and coordinate polish.
"""

import numpy as np
from scipy.optimize import linprog
import time

# --- Core functions ---

def _autocorr(f):
    """Full linear convolution f*f via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]

def _score(f):
    n = len(f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    c = _autocorr(f)
    return float(2.0 * n * c.max() / (s * s))

def _grad_softmax(f, c, temperature):
    """
    Compute gradient of sum_k w[k]*c[k] where w[k] = softmax(temperature * c[k]).
    Uses FFT-based cross-correlation for O(n log n) computation.
    """
    n = len(f)
    # Softmax weights
    c_max = c.max()
    w = np.exp(temperature * (c - c_max))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= w_sum
    
    # Gradient: grad[i] = 2 * sum_k w[k] * f[k-i] (for valid k-i)
    # This is 2 * cross_correlation(w, f)[i]
    size = 1 << (3 * n - 1).bit_length()
    W = np.fft.rfft(w, size)
    F = np.fft.rfft(f, size)
    # cross_correlation: ifft(fft(w) * conj(fft(f)))
    corr = np.fft.irfft(W * np.conj(F), size)[:n]
    return 2.0 * corr

def _mirror_step(f, grad, step):
    """One mirror descent step (exponential update)."""
    log_f = np.log(f + 1e-300) - step * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f, False
    f_new /= s
    return f_new, True

def _lp_direction(f, M, max_time=30.0):
    """
    Solve LP: max sum(b) s.t. conv(f,b) <= M, b >= 0.
    Returns the optimal b (normalized to sum=1) or None if infeasible/too slow.
    """
    n = len(f)
    t_start = time.monotonic()
    
    # Build constraint matrix (sparse)
    # A has shape (2n-1, n), A[k,j] = f[k-j] if 0 <= k-j < n
    # Use scipy sparse
    from scipy.sparse import lil_matrix
    A = lil_matrix((2*n-1, n), dtype=np.float64)
    for j in range(n):
        for k in range(j, j+n):
            if k < 2*n-1:
                A[k, j] = f[k-j]
    A = A.tocsr()
    
    # LP: max 1^T b s.t. A b <= M*1, b >= 0
    # linprog minimizes, so: min -1^T b s.t. A b <= M*1, b >= 0
    c_obj = -np.ones(n)
    A_ub = A
    b_ub = M * np.ones(2*n-1)
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
        if result.success:
            b = result.x
            s = b.sum()
            if s > 1e-10:
                b /= s
                return b
    except Exception:
        pass
    
    return None

def _polish(f, deadline):
    """Coordinate descent polish."""
    n = len(f)
    f = f.copy() / f.sum()
    best = _score(f)
    best_f = f.copy()
    
    c = _autocorr(f)
    peak = c.max()
    
    while time.monotonic() < deadline:
        # Find the peak lag
        k_star = int(np.argmax(c))
        
        # Find the best single-coordinate move
        best_improvement = 0
        best_move = None
        
        # Sample some candidate moves
        n_try = min(200, n // 3)
        rng_local = np.random.default_rng()
        indices = rng_local.integers(0, n, n_try)
        
        for i in indices:
            # Try moving mass from i to other positions
            # The contribution of f[i] to c[k_star] is f[k_star-i] (if valid)
            # We want to reduce this
            if 0 <= k_star - i < n:
                contrib = f[i] * f[k_star - i]
            else:
                contrib = 0
            
            if contrib < 1e-15:
                continue
            
            # Move a fraction of f[i] to a random other position
            j = rng_local.integers(0, n)
            if j == i:
                continue
            
            for frac in [0.05, 0.1, 0.2]:
                amt = f[i] * frac
                f_try = f.copy()
                f_try[i] -= amt
                f_try[j] += amt
                f_try = np.maximum(f_try, 0)
                s = f_try.sum()
                if s < 1e-10:
                    continue
                f_try /= s
                c_try = _autocorr(f_try)
                if c_try.max() < peak - best_improvement:
                    best_improvement = peak - c_try.max()
                    best_move = (f_try, c_try.max())
        
        if best_move is not None:
            f, peak = best_move
            c = _autocorr(f)
            peak = c.max()
            sc = 2.0 * n * peak
            if sc < best:
                best = sc
                best_f = f.copy()
        else:
            # No improvement found, try a random perturbation
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j:
                amt = f[i] * 0.05
                f[i] -= amt
                f[j] += amt
                f = np.maximum(f, 0)
                f /= f.sum()
                c = _autocorr(f)
                peak = c.max()
    
    return best_f, best
```
