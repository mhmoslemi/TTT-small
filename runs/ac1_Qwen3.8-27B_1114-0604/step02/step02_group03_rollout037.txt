def _normalize(v):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v


def _fft_size(m):
    if m <= 2:
        return 2
    return 1 << (int(m) - 1).bit_length()


def _autocorr(f):
    n = f.size
    m = 2 * n - 1
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]


def _fast_score(f):
    f = _normalize(f)
    if f.size == 0:
        return np.inf
    c = _autocorr(f)
    if c.size == 0:

    return float(2.0 * f.size * c.max())


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    out = v - theta
    np.maximum(out, 0.0, out)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    out /= s
    return out


def _weighted_gradient(p, f):
    n = f.size
    L = p.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = L + n - 1
    size = _fft_size(m)
    P = np.fft.rfft(p, size)
    F = np.fft.rfft(f[::-1], size)
    conv = np.fft.irfft(P * F, size)
    return 2.0 * conv[n - 1:n + n - 1]


def _interp_to_length(f, n):

    m = f.size

    x_old = np.linspace(0.0, m - 1, m, dtype=np.float64)
    x_new = np.linspace(0.0, n - 1, n, dtype=np.float64)
    return _normalize(np.interp(x_new, x_old, f))


def _smooth_with_window(f, win):

    out = np.convolve(f, kernel, mode="same")

def _two_component_seed(n, gamma_l, gamma_r, split_frac, rng):
    """Create a seed that is a mixture of two power-law shapes split at split_frac."""
    n = int(n)
    if n < 2:
        return np.ones(1, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    split = int(n * split_frac)
    a = np.zeros(n, dtype=np.float64)
    # Left component: power law decaying from left
    left_x = np.linspace(0.0, 1.0, split, dtype=np.float64)
    a[:split] = np.power(left_x + 0.02, -gamma_l)
    # Right component: power law decaying from right
    right_x = np.linspace(0.0, 1.0, n - split, dtype=np.float64)
    a[split:] = np.power(right_x + 0.02, -gamma_r)
    # Add small noise to break symmetry
    a *= np.exp(0.05 * rng.standard_normal(n))
    return _normalize(np.maximum(a, 0))


def _periodic_seed(n, period, duty, rng):
    """Create a periodic seed with given period and duty cycle."""
    n = int(n)
    a = np.zeros(n, dtype=np.float64)
    for i in range(n):
        phase = (i % period) / period
        if phase < duty:
            a[i] = 1.0
    # Smooth it
    win = max(3, period // 2)
    a = np.convolve(a, np.ones(win) / win, mode='same')
    # Add noise
    a *= np.exp(0.1 * rng.standard_normal(n))
    return _normalize(np.maximum(a, 0))


def _gaussian_mix_seed(n, rng):
    """Mixture of two Gaussians at different positions."""
    n = int(n)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)

    c2 = float(rng.uniform(0.6, 0.8))

    s1 = float(rng.uniform(0.05, 0.2))

    a = w1 * np.exp(-0.5 * ((x - c1) / s1) ** 2) + (1 - w1) * np.exp(-0.5 * ((x - c2) / s2) ** 2)
    a *= np.exp(0.05 * rng.standard_normal(n))
    return _normalize(np.maximum(a, 0))


def _fourier_seed(n, rng):
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    u = np.zeros(n, dtype=np.float64)
    u += float(rng.uniform(-1.0, 1.0)) * (x - 0.5)
    for m in (1, 2, 3, 5, 8, 13):
        u += 0.35 * float(rng.uniform(-1.0, 1.0)) * np.cos(
            2.0 * np.pi * m * x + float(rng.uniform(0.0, 2.0 * np.pi))

    u += 0.35 * float(rng.uniform(-1.0, 1.0)) * np.sin(
        2.0 * np.pi * x + float(rng.uniform(0.0, 2.0 * np.pi))
    )
    f = np.exp(0.6 * u)
    return _normalize(f)

    _try_add_seed(seeds, "given", f0, n)
    _try_add_seed(seeds, "reverse", f0[::-1], n)

    # Two-component seeds

        f = np.ones(max(1, n), dtype=np.float64)
        seeds.append((2.0, 0, "fallback", _normalize(f)))

def _transfer_max_eval(c, f, i, j, dv, delta, total):
    """Evaluate max(c') for transfer f[i]-=dv, f[j]+=dv."""
    n = f.size
    L = c.size
    delta.fill(0.0)
    delta[i:i + n] -= f
    delta[j:j + n] += f
    delta *= (2.0 * dv)
    q = dv * dv
    if 2 * j < L:
        delta[2 * j] += q
    if 2 * i < L:
        delta[2 * i] += q
    if i + j < L:
        delta[i + j] -= 2.0 * q
    np.add(c, delta, out=total)
    return float(np.max(total))


def _optimal_transfer(c, f, i, j, delta, total):
    """Find the dv in [0, f[i]] that minimizes max(c') using grid + refinement."""
    n = f.size
    L = c.size
    fi = float(f[i])
    if fi < 1e-15:
        return 0.0, float(c.max())
    # Coarse grid
    best_dv = 0.0
    best_mc = float(c.max())
    n_grid = 12
    for k in range(n_grid + 1):
        dv = fi * k / n_grid
        if dv < 1e-15:
            continue
        mc = _transfer_max_eval(c, f, i, j, dv, delta, total)
        if mc < best_mc:
            best_mc = mc
            best_dv = dv
    # Refine around best
    if best_dv > 0:
        lo = max(0.0, best_dv - fi / n_grid)
        hi = min(fi, best_dv + fi / n_grid)
        for _ in range(4):
            mid1 = lo + (hi - lo) / 3
            mid2 = hi - (hi - lo) / 3
            mc1 = _transfer_max_eval(c, f, i, j, mid1, delta, total)
            mc2 = _transfer_max_eval(c, f, i, j, mid2, delta, total)
            if mc1 < mc2:
                hi = mid2
            else:
                lo = mid1
            best_dv = (lo + hi) / 2
            mc = _transfer_max_eval(c, f, i, j, best_dv, delta, total)
            if mc < best_mc:
                best_mc = mc
    return best_dv, best_mc

    deadline = time.monotonic() + float(budget_s)
    delta = np.zeros(L, dtype=np.float64)
    total = np.zeros(L, dtype=np.float64)
    no_improve = 0

    # Find top peaks
    top_K = min(10, L)

    while time.monotonic() < deadline and no_improve < 20:
        maxc = float(np.max(c))
        # Get indices of top-K peaks
        top_idx = np.argpartition(c, L - top_K)[L - top_K:]
        # For each peak, find which coordinates contribute most
        # Peak at position k: c_k = sum_i a[i]*a[k-i]
        # Contribution of a[i] to c_k is a[k-i] (if valid)
        # To reduce c_k, we want to reduce a[i] where a[k-i] is large

        best_mc_local = maxc
        best_action = None

        for k in top_idx:
            k = int(k)
            # Find the i that maximizes a[i] * a[k-i]
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            if lo >= hi:
                continue
            # Contribution vector: for each i in [lo, hi], contrib_i = a[i] * a[k-i]
            contribs = np.zeros(n, dtype=np.float64)
            for i in range(lo, hi + 1):
                ki = k - i
                if 0 <= ki < n:
                    contribs[i] = f[i] * f[ki]
            # Find the largest contributor
            i_max = int(np.argmax(contribs))
            if contribs[i_max] < 1e-15:
                continue
            # Try transferring from i_max to a random receiver
            for _trial in range(3):
                j = int(rng.integers(0, n))
                if j == i_max or f[i_max] <= 1e-12:
                    continue
                dv, mc = _optimal_transfer(c, f, i_max, j, delta, total)
                if mc < best_mc_local - 1e-15 and dv > 1e-15:
                    best_mc_local = mc
                    best_action = (i_max, j, dv)
                    break

        # Also try gradient-guided transfers
        if best_action is None:
            p = np.zeros(L, dtype=np.float64)
            p[top_idx] = 1.0 / float(top_K)
            g = _weighted_gradient(p, f)
            donor_order = np.argsort(g)[::-1]
            recv_order = np.argsort(g)
            donors = [int(x) for x in donor_order[:6] if f[int(x)] > 1e-12]
            donor_set = set(donors)
            receivers = [int(x) for x in recv_order if int(x) not in donor_set][:6]
            for i in donors:
                for j in receivers:
                    if i == j:
                        continue
                    dv, mc = _optimal_transfer(c, f, i, j, delta, total)
                    if mc < best_mc_local - 1e-15 and dv > 1e-15:
                        best_mc_local = mc
                        best_action = (i, j, dv)
                        break
                if best_action is not None:
                    break

        if best_action is None:
            no_improve += 1
            continue

        i, j, dv = best_action
        f[i] -= dv
        f[j] += dv
        f = _normalize(f)
        c = _autocorr(f)
        maxc = float(c.max())
        no_improve = 0
        cur_score = float(2.0 * n * maxc)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()

    f = _normalize(f)
    c = _autocorr(f)
    if c.size:
        mc = float(c.max())
        sc = float(2.0 * n * mc)
        if sc <= best_score:
            return f, sc, c
    bf = _normalize(best_f)
    bc = _autocorr(bf)
    bmc = float(bc.max()) if bc.size else 0.0
    return bf, float(2.0 * n * bmc), bc


def _perturb(best_f, current_f, rng, temperature=1.0):
    best_f = _normalize(best_f)
    current_f = _normalize(current_f)
    n = best_f.size
    r = float(rng.random())
    if r < 0.4:
        sigma = float(rng.uniform(0.05, 0.8)) * temperature
        z = rng.standard_normal(n)
        f = best_f * np.exp(sigma * z)
    elif r < 0.65:
        alpha = float(rng.uniform(0.1, 0.4))
        noise = rng.random(n)
        s = noise.sum()
        if s >= 1e-300:
            noise /= s
        f = (1.0 - alpha) * best_f + alpha * noise
    elif r < 0.85:
        # Block swap perturbation
        block_size = max(2, n // 20)
        idx = int(rng.integers(0, n - block_size))
        f = best_f.copy()
        f[idx:idx + block_size] *= float(rng.uniform(0.5, 1.5))
    else:
        # Morphological perturbation
        f = _smooth_with_window(best_f, max(3, n // 100))
        sigma = float(rng.uniform(0.05, 0.5)) * temperature
        f = f * np.exp(sigma * rng.standard_normal(n))
    return _normalize(f)


def _gradient_optimize(f0, budget_s, rng, tag=""):
    """Annealed gradient optimizer with periodic polish and peak deflation."""
    f0 = _normalize(f0)
    n = f0.size
    if n == 0:
        return f0, np.inf
    if budget_s <= 0.0:
        c = _autocorr(f0)
        return f0, float(2.0 * n * c.max())

    f = f0.copy()
    c = _autocorr(f)
    maxc = float(c.max())
    best_f = f.copy()
    best_score = float(2.0 * n * maxc)

    deadline = time.monotonic() + float(budget_s)
    sigma_rel = 0.06
    min_sigma_rel = 1e-11
    no_accept = 0
    it = 0
    last_polish = time.monotonic()
    last_deflate = time.monotonic()

    if n < 2000:
        coord_budget = 0.2
        coord_interval = 0.5
        deflate_interval = 1.0
    elif n < 10000:
        coord_budget = 0.35
        coord_interval = 0.7
        deflate_interval = 1.5
    else:
        coord_budget = 0.5
        coord_interval = 0.9
        deflate_interval = 2.0

    print(f"[{tag}] start score={best_score:.6f}")

    while time.monotonic() < deadline:
        it += 1
        if maxc <= 0.0:
            break

        # Softmax weights - use sharper temperature for better resolution
        T = maxc * sigma_rel + 1e-300
        z = (c - maxc) / T
        np.clip(z, -50.0, 0.0, out=z)
        p = np.exp(z)
        ps = float(p.sum())
        if ps < 1e-300:
            p.fill(1.0 / float(c.size))
        else:
            p /= ps

        g = _weighted_gradient(p, f)
        gm = g - float(g.mean())
        scale = float(gm.max() - gm.min())

        accepted = False
        if scale > 1e-300:
            best_mc = maxc
            best_ft = None
            best_ct = None

            if no_accept > 40:
                exps = (0.003, 0.01, 0.05, 0.2, 1.0)
            else:
                exps = (0.01, 0.05, 0.2, 1.0, 3.0)

            for e in exps:
                arg = -e * (gm / scale)
                np.clip(arg, -40.0, 40.0, out=arg)
                ft = f * np.exp(arg)
                ft = _normalize(ft)
                ct = _autocorr(ft)
                mc = float(ct.max())
                if mc < best_mc:
                    best_mc = mc
                    best_ft = ft.copy()
                    best_ct = ct.copy()

                # Euclidean step
                if e in (0.05, 0.2, 1.0):
                    v = f - e * (gm / scale)
                    ft2 = _project_simplex(v)
                    ft2 = _normalize(ft2)
                    ct2 = _autocorr(ft2)
                    mc2 = float(ct2.max())
                    if mc2 < best_mc:
                        best_mc = mc2
                        best_ft = ft2.copy()
                        best_ct = ct2.copy()

            if best_ct is not None and best_mc < maxc - 1e-16:
                f = best_ft
                c = best_ct
                maxc = best_mc
                accepted = True

        if accepted:
            no_accept = 0
            cur_score = float(2.0 * n * maxc)
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()
                if best_score < TARGET_SCORE:
                    print(f"[{tag}] below target: {best_score:.6f}")
            sigma_rel = max(min_sigma_rel, sigma_rel * 0.995)
        else:
            no_accept += 1
            now = time.monotonic()

            # Coordinate polish
            if now - last_polish > coord_interval or no_accept > 25:
                fp, sc, cp = _coordinate_polish_binary(f, c, coord_budget, rng)
                if sc < float(2.0 * n * maxc) - 1e-12:
                    f = _normalize(fp)
                    c = cp
                    maxc = float(c.max())
                    accepted = True
                    no_accept = 0
                    if sc < best_score:
                        best_score = sc
                        best_f = f.copy()
                        if best_score < TARGET_SCORE:
                            print(f"[{tag}] coord below target: {best_score:.6f}")
                    last_polish = time.monotonic()
                else:
                    # Peak deflation
                    if now - last_deflate > deflate_interval:
                        fp, sc, cp = _peak_deflation(f, c, coord_budget, rng)
                        if sc < float(2.0 * n * maxc) - 1e-12:
                            f = _normalize(fp)
                            c = cp
                            maxc = float(c.max())
                            accepted = True
                            no_accept = 0
                            if sc < best_score:
                                best_score = sc
                                best_f = f.copy()
                        last_deflate = time.monotonic()

                    if no_accept > 60 or (it % 8000 == 0):
                        temperature = min(2.0, 1.0 + no_accept * 0.03)
                        f = _perturb(best_f, f, rng, temperature)
                        c = _autocorr(f)
                        maxc = float(c.max())
                        sigma_rel = min(0.3, sigma_rel * 2.5 + 1e-4)
                        no_accept = 0
                    last_polish = time.monotonic()
            if not accepted:
                sigma_rel = max(min_sigma_rel, sigma_rel * 0.998)

        if it % 15000 == 0:
            cur_score = float(2.0 * n * maxc)
            print(f"[{tag}] iter={it} best={best_score:.6f} cur={cur_score:.6f} sigma={sigma_rel:.3e} no_acc={no_accept}")

    cur_score = float(2.0 * n * float(c.max()))
    if cur_score < best_score:
        best_score = cur_score
        best_f = f.copy()

    print(f"[{tag}] end best={best_score:.6f}")
    return _normalize(best_f), float(best_score)


def _coarse_optimize(f_full, K, budget_s, rng, tag=""):
    """Optimize a coarse version and upsample back."""
    n = f_full.size
    x_old = np.linspace(0.0, n - 1, n, dtype=np.float64)
    x_new = np.linspace(0.0, K - 1, K, dtype=np.float64)
    f_coarse = _normalize(np.interp(x_new, x_old, f_full))
    f_opt, sc = _gradient_optimize(f_coarse, budget_s, rng, f"{tag}_coarse{K}")
    return _interp_to_length(f_opt, n), sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative coefficient sequence minimizing evaluate_sequence.
    Uses multi-phase: coarse optimization, gradient descent, binary-search polish, peak deflation.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_deadline = t_start + max(120.0, float(budget_s)) - 20.0

    # Load provided construction
    try:
        f0 = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        f0 = np.ones(6121, dtype=np.float64)

    if f0.size == 0:
        f0 = np.ones(6121, dtype=np.float64)
    f0 = np.nan_to_num(f0, nan=0.0, posinf=1000.0, neginf=0.0)
    np.clip(f0, 0.0, 1000.0, out=f0)
    n = int(f0.size)
    if n < 3:
        f0 = np.ones(1024, dtype=np.float64)
        n = f0.size
    f0 = _normalize(f0)

    best_f = f0.copy()
    best_c = _autocorr(best_f)
    best_score = float(2.0 * n * best_c.max()) if best_c.size else np.inf
    print(f"Initial score: {best_score:.6f}, n={n}")

    try:
        # === Phase 1: Coarse optimization ===
        remain = total_deadline - time.monotonic()
        if remain > 600 and n > 512:
            print(f"\n=== Phase 1: Coarse optimization (remain={remain:.0f}s) ===")
            for K in (256, 512):
                if total_deadline - time.monotonic() < 80:
                    break
                coarse_time = min(40.0, (total_deadline - time.monotonic()) * 0.1)
                if coarse_time < 10:
                    break
                f_coarse, sc_coarse = _coarse_optimize(f0, K, coarse_time, rng, "coarse")
                # Try different starting points for coarse
                f_extra = _fourier_seed(K, rng)
                f_extra2, sc_extra = _gradient_optimize(f_extra, coarse_time * 0.6, rng, f"coarse_f{K}")
                # Upsample both
                for cand, tag in [(f_coarse, f"coarse{K}"), (_interp_to_length(f_extra2, n), f"coarse_f{K}")]:
                    sc = _fast_score(cand)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_f = cand.copy()
                        best_c = _autocorr(best_f)
                        print(f"  Coarse improved: {best_score:.6f} ({tag})")

        # === Phase 2: Build seed pool and optimize top seeds ===
        remain = total_deadline - time.monotonic()
        if remain > 200:
            print(f"\n=== Phase 2: Seed pool optimization (remain={remain:.0f}s) ===")
            seed_pool = _build_seed_pool(best_f, n, rng)
            # Add current best
            seed_pool.append((best_score, 999, "current_best", best_f.copy()))
            seed_pool.sort(key=lambda x: x[0])

            # Optimize top 4 distinct seeds
            top_seeds = []
            seen_shapes = set()
            for sc, _, name, f in seed_pool:
                if len(top_seeds) >= 4:
                    break
                # Simple diversity check: use first 10 values as fingerprint
                fp = tuple(f[:min(10, f.size)].round(3))
                if fp not in seen_shapes:
                    seen_shapes.add(fp)
                    top_seeds.append((sc, name, f))

            # Allocate time
            n_seeds = len(top_seeds)
            seed_times = []
            for i in range(n_seeds):
                if i == 0:
                    seed_times.append(0.45 * remain)
                elif i == 1:
                    seed_times.append(0.25 * remain)
                elif i == 2:
                    seed_times.append(0.18 * remain)
                else:
                    seed_times.append(0.07 * remain)
            # Normalize to not exceed remain
            total_alloc = sum(seed_times)
            if total_alloc > remain * 0.75:
                seed_times = [t * remain * 0.75 / total_alloc for t in seed_times]

            for idx, (sc, name, f) in enumerate(top_seeds):
                if time.monotonic() >= total_deadline - 200:
                    break
                alloc = min(seed_times[idx], total_deadline - time.monotonic() - 100)
                if alloc < 15:
                    break
                print(f"  Optimizing seed '{name}' (score={sc:.6f}), alloc={alloc:.0f}s")
                f_opt, sc_opt = _gradient_optimize(f, alloc, rng, f"seed_{name}")
                if sc_opt < best_score:
                    best_score = sc_opt
                    best_f = f_opt.copy()
                    best_c = _autocorr(best_f)
                    print(f"  New best: {best_score:.6f}")

        # === Phase 3: Final intensive polish ===
        remain = total_deadline - time.monotonic()
        if remain > 60:
            print(f"\n=== Phase 3: Final polish (remain={remain:.0f}s) ===")
            # Coordinate polish
            polish_time = min(remain * 0.5, 180.0)
            if polish_time > 20:
                bf, bs, bc = _coordinate_polish_binary(best_f, best_c, polish_time, rng)
                if bs < best_score:
                    best_score = bs
                    best_f = bf.copy()
                    best_c = bc
                    print(f"  Polish improved: {best_score:.6f}")

            # Peak deflation
            remain = total_deadline - time.monotonic()
            if remain > 30:
                def_time = min(remain * 0.4, 120.0)
                if def_time > 15:
                    bf, bs, bc = _peak_deflation(best_f, best_c, def_time, rng)
                    if bs < best_score:
                        best_score = bs
                        best_f = bf.copy()
                        best_c = bc
                        print(f"  Deflation improved: {best_score:.6f}")

            # One more gradient pass
            remain = total_deadline - time.monotonic()
            if remain > 40:
                grad_time = min(remain * 0.6, 150.0)
                if grad_time > 20:
                    f2, s2 = _gradient_optimize(best_f, grad_time, rng, "final_grad")
                    if s2 < best_score:
                        best_score = s2
                        best_f = f2.copy()
                        best_c = _autocorr(best_f)

            # Final micro-polish
            remain = total_deadline - time.monotonic()
            if remain > 25:
                bf, bs, bc = _coordinate_polish_binary(best_f, best_c, min(remain - 10, 60.0), rng)
                if bs < best_score:
                    best_score = bs
                    best_f = bf.copy()
                    best_c = bc

    except Exception as exc:
        print("Exception during optimization:", repr(exc))
        import traceback
        traceback.print_exc()

    # Final safety
    best_f = _normalize(best_f)
    best_f = np.nan_to_num(best_f, nan=0.0, posinf=0.0, neginf=0.0)
    np.clip(best_f, 0.0, 1000.0, out=best_f)
    if best_f.sum() > 0:
        best_f = best_f / float(best_f.sum())
    best_c = _autocorr(best_f)
    best_score = float(2.0 * best_f.size * best_c.max())

    # Official evaluation
    try:
        out = [float(x) for x in best_f]
        official = evaluate_sequence(out)
        if np.isfinite(official) and official < best_score + 1e-12:
            best_score = float(official)
    except Exception:
        pass

    out = [float(x) for x in best_f]
    elapsed = time.monotonic() - t_start
    print(f"\nFINAL score = {best_score:.6f} (target={TARGET_SCORE}, elapsed={elapsed:.0f}s)")
    return out
