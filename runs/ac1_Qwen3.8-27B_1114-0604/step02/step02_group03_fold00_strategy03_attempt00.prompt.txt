def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def _normalize(v):
    """Project to non-negative float vector of sum 1."""
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v


def _fft_size(m):
    """Smallest power of two at least m and at least 2."""
    if m <= 2:
        return 2
    return 1 << (int(m) - 1).bit_length()


def _autocorr(f):
    """Full linear convolution f*f via FFT. f is assumed sum-normalized."""
    n = f.size
    m = 2 * n - 1
    if n == 0:
        return np.empty(0, dtype=np.float64)
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]


def _fast_score(f):
    """Fast score for a normalized non-negative vector."""
    f = _normalize(f)
    if f.size == 0:
        return np.inf
    c = _autocorr(f)
    if c.size == 0:
        return np.inf
    return float(2.0 * f.size * c.max())


def _project_simplex(v):
    """Euclidean projection onto x >= 0, sum x = 1."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1.0)
    out = v - theta
    np.maximum(out, 0.0, out)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    out /= s
    return out


def _weighted_gradient(p, f):
    """
    Gradient-like vector g_j = 2 * sum_k p_k f_{k-j}
    where p has length 2n-1 and f has length n.
    """
    n = f.size
    L = p.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = L + n - 1
    size = _fft_size(m)
    P = np.fft.rfft(p, size)
    F = np.fft.rfft(f[::-1], size)
    conv = np.fft.irfft(P * F, size)
    return 2.0 * conv[n - 1:n + n - 1]


def _interp_to_length(f, n):
    """Linearly resample f to length n and normalize."""
    f = _normalize(f)
    m = f.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if m == n:
        return f
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    if m == 1:
        return _normalize(np.full(n, f[0]))
    x_old = np.linspace(0.0, m - 1, m, dtype=np.float64)
    x_new = np.linspace(0.0, n - 1, n, dtype=np.float64)
    return _normalize(np.interp(x_new, x_old, f))


def _smooth_with_window(f, win):
    """Uniform moving-average smoothing."""
    f = _normalize(f)
    n = f.size
    if n <= 2:
        return f
    win = int(win)
    if win < 3:
        win = 3
    if win > n:
        win = max(3, n // 2)
    kernel = np.ones(win, dtype=np.float64) / float(win)
    out = np.convolve(f, kernel, mode="same")
    return _normalize(out)


def _beta_seed(n, rng):
    """Randomly asymmetric beta-shaped seed."""
    if n == 0:
        return np.empty(0, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    a = float(rng.uniform(0.2, 3.0))
    b = float(rng.uniform(0.2, 3.0))
    f = (x + 0.005) ** a * (1.0 - x + 0.005) ** b
    return _normalize(f)


def _fourier_seed(n, rng):
    """Smooth low-frequency perturbation of uniform."""
    if n == 0:
        return np.empty(0, dtype=np.float64)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    u = np.zeros(n, dtype=np.float64)
    u += float(rng.uniform(-1.0, 1.0)) * (x - 0.5)
    for m in (1, 2, 3, 5, 8):
        u += 0.40 * float(rng.uniform(-1.0, 1.0)) * np.cos(
            2.0 * np.pi * m * x + float(rng.uniform(0.0, 2.0 * np.pi))
        )
    u += 0.40 * float(rng.uniform(-1.0, 1.0)) * np.sin(
        2.0 * np.pi * x + float(rng.uniform(0.0, 2.0 * np.pi))
    )
    # Keep positive without extreme peaks.
    f = np.exp(0.65 * u)
    return _normalize(f)

def _transfer_max(c, f, i, j, dv, delta, total):
    """
    Return max_k c'_k for the two-coordinate transfer
        f[i] -= dv, f[j] += dv
    without modifying f/c. Pre-allocated delta,total are used.
    """
    n = f.size
    L = c.size
    delta.fill(0.0)
    delta[i:i + n] -= f
    delta[j:j + n] += f
    delta *= (2.0 * dv)
    q = dv * dv
    # i,j in [0,n-1] imply all these indices are valid in length 2n-1.
    delta[2 * j] += q
    delta[2 * i] += q
    delta[i + j] -= 2.0 * q
    np.add(c, delta, out=total)
    return float(np.max(total))

def _perturb(best_f, current_f, rng):
    """Perturb from the best state, either multiplicatively or by mixing."""
    best_f = _normalize(best_f)
    current_f = _normalize(current_f)
    n = best_f.size
    r = float(rng.random())
    if r < 0.55:
        sigma = float(rng.uniform(0.04, 1.0))
        z = rng.standard_normal(n)
        f = best_f * np.exp(sigma * z)
    elif r < 0.85:
        alpha = float(rng.uniform(0.05, 0.45))
        noise = rng.random(n)
        s = noise.sum()
        if s >= 1e-300:
            noise /= s
        f = (1.0 - alpha) * best_f + alpha * noise
    else:
        # Smooth best then roughen.
        f = _smooth_with_window(best_f, max(3, n // 200))
        sigma = float(rng.uniform(0.05, 0.5))
        f = f * np.exp(sigma * rng.standard_normal(n))
    return _normalize(f)


def _gradient_optimize(f0, budget_s, rng, tag=""):
    """
    Annealed gradient/line-search optimizer in log domain, with periodic
    exact coordinate polish and perturbations.
    """
    f0 = _normalize(f0)
    n = f0.size
    if n == 0:
        return f0, np.inf
    if budget_s <= 0.0:
        c = _autocorr(f0)
        return f0, float(2.0 * n * c.max())

    f = f0
    c = _autocorr(f)
    maxc = float(c.max())
    best_f = f.copy()
    best_score = float(2.0 * n * maxc)

    deadline = time.monotonic() + float(budget_s)
    sigma_rel = 0.08
    min_sigma_rel = 1e-10
    no_accept = 0
    it = 0
    last_coord = time.monotonic()

    if n < 2000:
        coord_budget = 0.25
        coord_interval = 0.40
    elif n < 10000:
        coord_budget = 0.45
        coord_interval = 0.60
    else:
        coord_budget = 0.70
        coord_interval = 0.80

    print(f"[{tag}] start score={best_score:.6f}")

    while time.monotonic() < deadline:
        it += 1
        if maxc <= 0.0:
            break

        # Softmax weights over lags, temperature relative to current peak.
        T = maxc * sigma_rel + 1e-300
        z = (c - maxc) / T
        np.clip(z, -45.0, 0.0, out=z)
        p = np.exp(z)
        ps = float(p.sum())
        if ps < 1e-300:
            p.fill(1.0 / float(c.size))
        else:
            p /= ps

        g = _weighted_gradient(p, f)
        gm = g - float(g.mean())
        scale = float(gm.max() - gm.min())

        accepted = False
        if scale > 1e-300:
            best_mc = maxc
            best_ft = None
            best_ct = None

            if no_accept > 30:
                exps = (0.005, 0.02, 0.1, 0.5)
            else:
                exps = (0.02, 0.1, 1.0, 5.0)

            for e in exps:
                # Log-domain mirror step.
                arg = -e * (gm / scale)
                np.clip(arg, -35.0, 35.0, out=arg)
                ft = f * np.exp(arg)
                ft = _normalize(ft)
                ct = _autocorr(ft)
                mc = float(ct.max())
                if mc < best_mc:
                    best_mc = mc
                    best_ft = ft
                    best_ct = ct

                # Projected Euclidean step for a couple of middle/large scales.
                if e in (0.1, 1.0):
                    v = f - e * (gm / scale)
                    ft2 = _project_simplex(v)
                    ft2 = _normalize(ft2)
                    ct2 = _autocorr(ft2)
                    mc2 = float(ct2.max())
                    if mc2 < best_mc:
                        best_mc = mc2
                        best_ft = ft2
                        best_ct = ct2

            if best_ct is not None and best_mc < maxc - 1e-18:
                f = best_ft
                c = best_ct
                maxc = best_mc
                accepted = True

        if accepted:
            no_accept = 0
            cur_score = float(2.0 * n * maxc)
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()
                if best_score < TARGET_SCORE:
                    print(f"[{tag}] new best below target: {best_score:.6f}")
            sigma_rel = max(min_sigma_rel, sigma_rel * 0.99)
        else:
            no_accept += 1
            now = time.monotonic()
            if now - last_coord > coord_interval or no_accept > 20:
                fp, sc, cp = _coordinate_polish(f, c, coord_budget, rng)
                if sc < float(2.0 * n * maxc) - 1e-12:
                    f = _normalize(fp)
                    c = cp
                    maxc = float(c.max())
                    accepted = True
                    no_accept = 0
                    if sc < best_score:
                        best_score = sc
                        best_f = f.copy()
                        if best_score < TARGET_SCORE:
                            print(f"[{tag}] coordinate improved below target: {best_score:.6f}")
                else:
                    if no_accept > 55 or (it % 6000 == 0):
                        f = _perturb(best_f, f, rng)
                        c = _autocorr(f)
                        maxc = float(c.max())
                        sigma_rel = min(0.25, sigma_rel * 2.0 + 1e-4)
                        no_accept = 0
                last_coord = time.monotonic()
            else:
                sigma_rel = max(min_sigma_rel, sigma_rel * 0.995)

        if it % 20000 == 0:
            cur_score = float(2.0 * n * maxc)
            print(f"[{tag}] iter={it} best={best_score:.6f} cur={cur_score:.6f} sigma={sigma_rel:.3e}")

    cur_score = float(2.0 * n * float(c.max()))
    if cur_score < best_score:
        best_score = cur_score
        best_f = f.copy()

    print(f"[{tag}] end best={best_score:.6f}")
    return _normalize(best_f), float(best_score)


def _downsample_to(f, K):
    """Downsample by linear interpolation to K grid points."""
    f = _normalize(f)
    n = f.size
    if K <= 0:
        return np.empty(0, dtype=np.float64)
    if K >= n:
        return f
    x_old = np.linspace(0.0, n - 1, n, dtype=np.float64)
    x_new = np.linspace(0.0, K - 1, K, dtype=np.float64)
    return _normalize(np.interp(x_new, x_old, f))


def _upsample_to(f, n):
    """Upsample to length n by linear interpolation."""
    return _interp_to_length(f, n)

    for i in range(12):

    for i in range(8):
        _try_add_seed(seeds, f"beta{i}", _beta_seed(n, rng), n)

    # Coarse-to-fine smoothings of the given construction.
    if n > 200:
        for K in (128, 256, 512, 1024):
            if K < n:
                coarse = _downsample_to(f0, K)
