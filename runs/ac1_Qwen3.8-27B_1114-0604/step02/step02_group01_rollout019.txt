"""
Step function optimizer: minimize 2*n*max(conv(a,a))/(sum(a))^2.
Uses L-BFGS-B on log-softmax parametrization with annealed smooth max,
multiple diverse starting points, and coordinate descent polish.
Key insight: the log-parametrization f=softmax(x) automatically handles
non-negativity and scale invariance, and the smooth max provides a
globally-informative gradient (all lags contribute).
"""

import numpy as np
from scipy.optimize import minimize
import time


def _autocorr_fft(f):
    """Full linear autocorrelation via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _cross_corr_weights(w, f, n):
    """Compute h[j] = sum_k w[k]*f[k-j] for j=0..n-1.
    w has length 2n-1, f has length n."""
    w_rev = w[::-1]
    size = 1 << (3 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    W = np.fft.rfft(w_rev, size)
    conv = np.fft.irfft(F * W, size)
    return conv[n - 1:2 * n - 1][::-1]


def _softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()


def _score_from_f(f):
    """Compute the exact score (scale-invariant)."""
    n = len(f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))


def _smooth_objective_and_grad(x, beta, n):
    """
    Objective and gradient in x-space where f = softmax(x).
    Uses smooth approximation of max via log-sum-exp.
    """
    f = _softmax(x)
    c = _autocorr_fft(f)
    c_max = c.max()
    # Smooth max
    exp_bc = np.exp(beta * (c - c_max))
    M_smooth = c_max + np.log(exp_bc.sum()) / beta
    score = 2.0 * n * M_smooth  # since sum(f)=1

    # Weights for smooth max gradient
    w = exp_bc / exp_bc.sum()

    # Gradient of M_smooth w.r.t. f: h[j] = sum_k w[k]*f[k-j]
    h = _cross_corr_weights(w, f, n)

    # d(score)/df[j] = 2*n*h[j] - 2*n*M_smooth (the constant drops in x-grad)
    g_f = 2.0 * n * h  # drop constant term (cancels in x-gradient)

    # d(score)/dx[k] = f[k] * (g_f[k] - sum_i f[i]*g_f[i])
    dot_fg = np.dot(f, g_f)
    grad_x = f * (g_f - dot_fg)

    return score, grad_x


def _smooth_objective_only(x, beta, n):
    f = _softmax(x)
    c = _autocorr_fft(f)
    c_max = c.max()
    M_smooth = c_max + np.log(np.exp(beta * (c - c_max)).sum()) / beta
    return 2.0 * n * M_smooth


def _lbfgs_optimize(x0, n, deadline, beta_schedule, verbose=False, label=""):
    """
    Run L-BFGS-B with annealed beta.
    beta_schedule: list of (beta, max_iter) tuples.
    """
    x = x0.copy()
    best_x = x0.copy()
    best_score = _score_from_f(_softmax(x0))

    for beta, max_iter in beta_schedule:
        if time.monotonic() > deadline:
            break

        # Define objective and gradient for this beta
        def obj(x_local):
            return _smooth_objective_only(x_local, beta, n)

        def grad(x_local):
            _, g = _smooth_objective_and_grad(x_local, beta, n)
            return g

        # Run L-BFGS-B
        t0 = time.monotonic()
        result = minimize(
            obj, x, method='L-BFGS-B', jac=grad,
            options={'maxiter': max_iter, 'gtol': 1e-8, 'ftol': 1e-10,
                     'maxls': 50}
        )
        x = result.x
        t_elapsed = time.monotonic() - t0

        f = _softmax(x)
        score = _score_from_f(f)
        if score < best_score:
            best_score = score
            best_x = x.copy()
            if verbose:
                print(f"    {label} beta={beta}: score={score:.7f} "
                      f"(iter={result.nit}, time={t_elapsed:.1f}s)")

    return best_x, best_score


def _coordinate_polish(f, n, deadline, rng, max_layers=5):
    """
    Coordinate descent polish: iteratively reduce the peak lag by
    transferring mass from high-contributing to low-contributing positions.
    """
    f = f.copy()
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-10:
        return f, _score_from_f(np.ones(n) / n)
    f /= s
    best_score = _score_from_f(f)
    best_f = f.copy()

    c = _autocorr_fft(f)
    peak = c.max()

    n_outer = 200
    for outer in range(n_outer):
        if time.monotonic() > deadline:
            break

        peak = c.max()
        # Find the argmax lag
        k_peak = int(np.argmax(c))

        # Contribution of each position to c[k_peak]
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        contrib = np.zeros(n)
        if js <= je:
            idx = k_peak - np.arange(js, je + 1)
            contrib[js:je + 1] = f[js:je + 1] * f[idx]

        # Also consider total contribution across top few lags
        top_k = np.argpartition(c, -3)[-3:]
        top_k = top_k[np.argsort(c[top_k])[::-1]]
        total_contrib = np.zeros(n)
        for k in top_k:
            js2 = max(0, k - n + 1)
            je2 = min(n - 1, k)
            if js2 <= je2:
                idx2 = k2 - np.arange(js2, je2 + 1) if 'k2' in dir() else k - np.arange(js2, je2 + 1)
                total_contrib[js2:je2 + 1] += f[js2:je2 + 1] * f[idx2]

        # Donor: highest total contribution, Receiver: lowest
        j_don = int(np.argmax(total_contrib))
        j_rec = int(np.argmin(total_contrib))

        if j_don == j_rec or total_contrib[j_don] < 1e-15:
            # Try random perturbations
            for _ in range(10):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                    f_try = f.copy()
                    amt = f_try[i] * frac
                    if amt <= 0:
                        continue
                    f_try[i] -= amt
                    f_try[j] += amt
                    f_try = np.maximum(f_try, 0)
                    fs = f_try.sum()
                    if fs < 1e-10:
                        continue
                    f_try /= fs
                    c_try = _autocorr_fft(f_try)
                    if c_try.max() < peak - 1e-12:
                        f = f_try
                        c = c_try
                        peak = c.max()
                        sc = _score_from_f(f)
                        if sc < best_score:
                            best_score = sc
                            best_f = f.copy()
                        break
        else:
            # Try several transfer fractions
            improved = False
            for frac in [0.1, 0.2, 0.3, 0.5, 0.7, 0.9]:
                amt = f[j_don] * frac
                if amt <= 0:
                    continue
                f_try = f.copy()
                f_try[j_don] -= amt
                f_try[j_rec] += amt
                f_try = np.maximum(f_try, 0)
                fs = f_try.sum()
                if fs < 1e-10:
                    continue
                f_try /= fs
                c_try = _autocorr_fft(f_try)
                if c_try.max() < peak - 1e-12:
                    f = f_try
                    c = c_try
                    peak = c.max()
                    improved = True
                    sc = _score_from_f(f)
                    if sc < best_score:
                        best_score = sc
                        best_f = f.copy()
                    break

        if not improved and outer > 100:
            # Random restart of coordinates
            for _ in range(20):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                f_try = f.copy()
                amt = f_try.mean() * 0.05
                f_try[i] -= amt
                f_try[j] += amt
                f_try = np.maximum(f_try, 0)
                f_try /= f_try.sum()
                c_try = _autocorr_fft(f_try)
                if c_try.max() < peak - 1e-12:
                    f = f_try
                    c = c_try
                    peak = c.max()
                    break

    return best_f, best_score


def _fine_mirror_descent(f, n, deadline, rng, n_iters=20000):
    """
    Fine-tuned mirror descent targeting the single peak lag.
    Smaller steps than the main optimizer.
    """
    f = f.copy()
    f = np.maximum(f, 0)
    f /= f.sum()
    best_score = _score_from_f(f)
    best_f = f.copy()

    step = 0.05
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        c = _autocorr_fft(f)
        k_peak = int(np.argmax(c))

        # Gradient at the peak lag
        js = max(0, k_peak - n + 1)
        je = min(n - 1, k_peak)
        grad = np.zeros(n)
        if js <= je:
            idx = k_peak - np.arange(js, je + 1)
            grad[js:je + 1] = 2.0 * f[idx]

        # Mirror step (exponentiated gradient)
        span = grad.max() - grad.min()
        if span < 1e-30:
            break
        eta = step / span
        log_f = np.log(f + 1e-300) - eta * grad
        log_f -= log_f.max()
        f_new = np.exp(log_f)
        f_new = np.maximum(f_new, 0)
        f_new /= f_new.sum()
        f = f_new

        # Anneal step size
        step = 0.05 * (0.001 / 0.05) ** (it / n_iters)

        if it % 500 == 0:
            sc = _score_from_f(f)
            if sc < best_score:
                best_score = sc
                best_f = f.copy()

            # Occasionally perturb if stuck
            if it % 2000 == 1000:
                perturb = rng.normal(0, 0.01, size=n)
                f = np.maximum(f + perturb * f.mean(), 0)
                f /= f.sum()

    sc = _score_from_f(f)
    if sc < best_score:
        best_score = sc
        best_f = f.copy()

    return best_f, best_score
