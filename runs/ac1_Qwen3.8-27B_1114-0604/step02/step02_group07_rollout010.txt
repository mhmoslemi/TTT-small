def fast_self_conv(a):
    """Compute a*a via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def grad_adaptive(a, c, topk, temp_scale=0.1):
    """Adaptive gradient blending exact and softmax."""
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S
    k_act = min(topk, c.size)
    
    part = np.argpartition(c, -k_act)[-k_act:]
    vals = c[part].copy()
    order = np.argsort(vals)[::-1]
    part = part[order]
    vals = vals[order]
    
    top1 = float(vals[0])
    top2 = float(vals[1]) if k_act > 1 else 0.0
    
    if top1 > 1e-15:
        ratio = top2 / top1
    else:
        ratio = 1.0
    
    exact_w = float(np.clip((0.9 - ratio) / 0.4, 0.0, 1.0))
    
    if exact_w > 0.99:
        wts = np.zeros(k_act)
        wts[0] = 1.0
        M = top1
    else:
        temp = max(temp_scale * top1, 1e-15)
        wts_smooth = np.exp((vals - top1) / temp)
        wts_smooth = wts_smooth / wts_smooth.sum()
        if exact_w < 0.01:
            wts = wts_smooth
        else:
            wts_exact = np.zeros(k_act)
            wts_exact[0] = 1.0
            wts = exact_w * wts_exact + (1.0 - exact_w) * wts_smooth
            wts = wts / wts.sum()
        M = float(np.dot(vals, wts))
    
    grad_c = np.zeros(n)
    for i in range(k_act):

    grad = (2.0 * n) * (grad_c / S2 - 2.0 * M / S3)
    return grad, M, S

    v[:] = 0.999 * v + 0.001 * g * g
    a -= lr * (m / (1.0 - t1)) / (np.sqrt(v / (1.0 - t2)) + 1e-8)

"""
Improved optimizer combining log-parameterized gradient descent, 
frequency-domain perturbations, and adaptive subgradient methods
to minimize 2n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, c, S):
    """Objective value given n, convolution array c, and sum S."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def gradient_submax(a, c, topk=1):
    """
    Compute subgradient using only the argmax peak.
    Returns the gradient vector.
    """
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S
    
    k = int(np.argmax(c))
    M = float(c[k])
    
    # c[k] = sum_{i} a[i] * a[k-i] where k-i is valid
    # d(c[k])/da[j] = a[k-j] (if k-j is in [0, n-1])
    js = max(0, k - n + 1)
    je = min(k, n - 1)
    
    grad = np.zeros(n)
    if js <= je:
        # d(c[k])/da[j] for j in [js, je] is a[k-j]
        # a[k-j] for j in [js, je] means indices k-je to k-js
        contrib = a[k - je : k - js + 1][::-1]
        grad[js:je+1] = contrib
    
    # Full gradient of score = 2n * M / S^2:
    # d(score)/da[j] = 2n * (dM/da[j] / S^2 - 2*M * da_sum/da[j] / S^3)
    # = 2n * (grad_c[j] / S^2 - 2*M / S^3)
    result = (2.0 * n) * (grad / S2 - 2.0 * M / S3)
    return result


def gradient_softmax(a, c, topk=16, temp_scale=0.05):
    """
    Compute gradient using softmax over top-k peaks.
    """
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S
    k_act = min(topk, c.size)
    
    part = np.argpartition(c, -k_act)[-k_act:]
    vals = c[part].copy()
    order = np.argsort(vals)[::-1]
    part = part[order]
    vals = vals[order]
    
    top1 = float(vals[0])
    temp = max(temp_scale * top1, 1e-15)
    
    wts = np.exp((vals - top1) / temp)
    wts = wts / wts.sum()
    M = float(np.dot(vals, wts))
    
    grad_c = np.zeros(n)
    for i in range(k_act):
        wi = float(wts[i])
        if wi <= 1e-15:
            continue
        ki = int(part[i])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        grad_c[js:je + 1] += wi * a[js:je + 1][::-1]
    
    result = (2.0 * n) * (grad_c / S2 - 2.0 * M / S3)
    return result


def gradient_adaptive(a, c, topk=16, temp_scale=0.05):
    """
    Adaptive gradient: blends exact subgradient with softmax based on peak separation.
    """
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S
    k_act = min(topk, c.size)
    
    part = np.argpartition(c, -k_act)[-k_act:]
    vals = c[part].copy()
    order = np.argsort(vals)[::-1]
    part = part[order]
    vals = vals[order]
    
    top1 = float(vals[0])
    top2 = float(vals[1]) if k_act > 1 else 0.0
    
    if top1 > 1e-15:
        ratio = top2 / top1
    else:
        ratio = 1.0
    
    # Weight for exact vs smooth
    exact_w = float(np.clip((0.92 - ratio) / 0.3, 0.0, 1.0))
    
    temp = max(temp_scale * top1, 1e-15)
    wts_smooth = np.exp((vals - top1) / temp)
    wts_smooth = wts_smooth / wts_smooth.sum()
    
    if exact_w > 0.99:
        wts = np.zeros(k_act)
        wts[0] = 1.0
        M = top1
    elif exact_w < 0.01:
        wts = wts_smooth
        M = float(np.dot(vals, wts))
    else:
        wts_exact = np.zeros(k_act)
        wts_exact[0] = 1.0
        wts = exact_w * wts_exact + (1.0 - exact_w) * wts_smooth
        wts = wts / wts.sum()
        M = float(np.dot(vals, wts))
    
    grad_c = np.zeros(n)
    for i in range(k_act):
        wi = float(wts[i])
        if wi <= 1e-15:
            continue
        ki = int(part[i])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        grad_c[js:je + 1] += wi * a[js:je + 1][::-1]
    
    result = (2.0 * n) * (grad_c / S2 - 2.0 * M / S3)
    return result


def adam_update(params, grads, m, v, t, lr):
    """Adam optimizer step (in-place on params)."""
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    bc1 = 1.0 - (beta1 ** t)
    bc2 = 1.0 - (beta2 ** t)
    m[:] = beta1 * m + (1.0 - beta1) * grads
    v[:] = beta2 * v + (1.0 - beta2) * grads * grads
    m_hat = m / bc1
    v_hat = v / bc2
    params -= lr * m_hat / (np.sqrt(v_hat) + eps)


def optimize_logspace(a0, end_time, rng, init_lr=0.05, topk=16, patience=400, temp_scale=0.05):

    # Center b so that the mean of a is around 1

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = init_lr
    no_imp = 0

            m[:] = 0.0
            v[:] = 0.0
            t = 0
            continue
        
        c = fast_self_conv(a)
        score = compute_score(n, c, S)
        
        if np.isfinite(score) and score < best_score:

            no_imp = 0

def symmetry_enforce(a, n):
    """Enforce approximate symmetry: a[i] = a[n-1-i]."""
    half = n // 2
    a[:half] = 0.5 * (a[:half] + a[half:n - n % 2][::-1] if n % 2 == 0 else a[half:n - 1][::-1])
    if n % 2 == 0:
        a[half:] = a[:half][::-1]

def freq_domain_perturb(a, rng, strength=0.05):

    # Add random magnitude and phase perturbation (skip DC)
    noise_mag = strength * rng.random(A.shape)
    noise_phase = 2.0 * np.pi * rng.random(A.shape)
    A_pert = A * (1.0 + noise_mag * np.exp(1j * noise_phase))
    A_pert[0] = A[0]  # Keep DC intact
    a_pert = np.fft.irfft(A_pert, M_fft)[:n]
    return np.clip(a_pert, 1e-10, 1000.0)


def optimize_logspace(a0, end_time, rng, init_lr, topk, patience, temp_scale, use_symmetry=False):
    """
    Optimize in log-space: a = exp(b), Adam on b.
    Ensures non-negativity automatically and gives multiplicative updates.
    """
    n = a0.size
    a_start = np.clip(a0, 1e-10, 1000.0)
    b = np.log(a_start).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = init_lr
    no_imp = 0
    cur_pat = patience
    step_count = 0

    a = np.exp(b)
    c = fast_self_conv(a)
    S = float(a.sum())
    score = compute_score(n, c, S)
    best_score = score
    best_a = a.copy()

    while time.monotonic() < end_time:
        a = np.exp(b)
        if use_symmetry:
            a = symmetry_enforce(a.copy(), n)
        S = float(a.sum())
        if S < 1e-10 or not np.isfinite(S):
            b = np.log(np.clip(a_start, 1e-10, 1000.0))
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            continue

        c = fast_self_conv(a)
        score = compute_score(n, c, S)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp >= cur_pat:
            # Restart with frequency-domain perturbation
            a_base = best_a.copy()
            a_pert = freq_domain_perturb(a_base, rng, strength=0.03 + 0.05 * rng.random())
            if a_pert.sum() < 1e-3:
                a_pert = a_base * (1.0 + 0.03 * rng.standard_normal(n))
                a_pert = np.clip(a_pert, 1e-10, 1000.0)
            b = np.log(a_pert)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            lr = max(0.001, lr * 0.6)
            no_imp = 0
            cur_pat = max(40, cur_pat // 2)
            continue

        # Compute gradient in a-space, convert to b-space via chain rule
        grad_a, M, S = gradient_adaptive(a, c, topk, temp_scale)
        grad_b = grad_a * a  # d(score)/db[j] = d(score)/da[j] * a[j]

        t += 1
        step_count += 1
        adam_update(b, grad_b, m, v, t, lr)

        # Keep b in reasonable range to avoid overflow
        b = np.clip(b, -25, 25)

        # Learning rate schedule: cosine-like decay
        decay = 0.5 * (1.0 + np.cos(np.pi * min(step_count / (patience * 4), 1.0)))
        lr = max(0.0005, init_lr * decay)

    return best_score, best_a


def optimize_timespace(a0, end_time, rng, init_lr, topk, patience, temp_scale, use_symmetry=False):
    """
    Optimize in time-domain with Adam and clipping.
    """
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = init_lr
    no_imp = 0
    cur_pat = patience
    step_count = 0

    c = fast_self_conv(a)
    S = float(a.sum())
    score = compute_score(n, c, S)
    best_score = score
    best_a = a.copy()

    while time.monotonic() < end_time:
        S = float(a.sum())
        if S < 1e-10:
            a = np.clip(a0, 0.01, 1000.0)
            S = float(a.sum())

        c = fast_self_conv(a)
        score = compute_score(n, c, S)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp >= cur_pat:
            # Restart with frequency-domain perturbation
            a_pert = freq_domain_perturb(best_a, rng, strength=0.02 + 0.04 * rng.random())
            if a_pert.sum() < 1e-3:
                a_pert = best_a * (1.0 + 0.02 * rng.standard_normal(n))
                a_pert = np.clip(a_pert, 0.0, 1000.0)
            a = a_pert
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            lr = max(0.001, lr * 0.5)
            no_imp = 0
            cur_pat = max(40, cur_pat // 2)
            continue

        grad, M, S = gradient_adaptive(a, c, topk, temp_scale)
        t += 1
        step_count += 1
        adam_update(a, grad, m, v, t, lr)
        a = np.clip(a, 0.0, 1000.0)

        if use_symmetry:
            a = symmetry_enforce(a, n)

        # Learning rate schedule
        decay = 0.5 * (1.0 + np.cos(np.pi * min(step_count / (patience * 4), 1.0)))
        lr = max(0.0005, init_lr * decay)

    return best_score, best_a


def spectral_flatten_step(a, target_ratio, n_peaks):
    """
    One step of spectral flattening: reduce the top n_peaks of |FFT(a)|^2
    while preserving the phase, aiming for flatter spectrum.
    """
    n = len(a)
    M_fft = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(a, M_fft)
    mag = np.abs(A)
    mag2 = mag ** 2

    # Exclude DC
    non_dc = np.ones(len(mag2), dtype=bool)
    non_dc[0] = False

    # Find target: the value at the n_peaks-th percentile
    vals = mag2[non_dc]
    target = np.percentile(vals, 100.0 * (1.0 - target_ratio))

    # Reduce peaks above target
    phase = A / (mag + 1e-30)
    for idx in np.where(mag2[non_dc] > target)[0]:
        # Blend toward target
        new_mag2 = 0.5 * (mag2[idx] + target)
        new_mag = np.sqrt(max(new_mag2, 0.0))
        A[idx] = phase[idx] * new_mag

    # Restore DC to maintain sum
    A[0] = float(a.sum())

    a_new = np.fft.irfft(A, M_fft)[:n]
    return np.clip(a_new, 0.0, 1000.0)


def optimize_spectral(a0, end_time, rng, init_strength=0.3, patience=100):
    """
    Spectral flattening optimization.
    Alternates between time-domain clipping and spectral peak reduction.
    """
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a.sum() < 1e-10:
        a = np.ones(n)

    c = fast_self_conv(a)
    S = float(a.sum())
    score = compute_score(n, c, S)
    best_score = score
    best_a = a.copy()
    no_imp = 0
    cur_pat = patience
    strength = init_strength

    while time.monotonic() < end_time:
        # Spectral flattening step
        a_flat = spectral_flatten_step(a, strength, 5)
        S_new = float(a_flat.sum())
        if S_new < 1e-10:
            a_flat = a.copy()
            S_new = float(a.sum())

        # Evaluate
        c_new = fast_self_conv(a_flat)
        score_new = compute_score(n, c_new, S_new)

        if np.isfinite(score_new) and score_new < best_score:
            best_score = score_new
            best_a = a_flat.copy()
            a = a_flat.copy()
            no_imp = 0
        else:
            no_imp += 1
            # If not improving, try a gradient step instead
            grad, M, S = gradient_adaptive(a, c, 8, 0.05)
            a -= 0.01 * grad / (np.sqrt(np.mean(grad * grad)) + 1e-10)
            a = np.clip(a, 0.0, 1000.0)

        if no_imp >= cur_pat:
            # Restart
            a = best_a * (1.0 + 0.05 * rng.standard_normal(n))
            a = np.clip(a, 0.0, 1000.0)
            if a.sum() < 1e-3:
                a = best_a.copy()
            c = fast_self_conv(a)
            no_imp = 0
            cur_pat = max(30, cur_pat // 2)
            strength = max(0.05, strength * 0.7)

        c = fast_self_conv(a)

    return best_score, best_a


def init_diverse(n, rng):
    """Generate diverse initial sequences for multi-start."""
    inits = []
    inits.append(np.ones(n))
    x = np.linspace(0, 1, n)
    inits.append(1.0 + 0.4 * np.cos(2.0 * np.pi * x))
    inits.append(1.0 + 0.6 * np.cos(4.0 * np.pi * x))
    inits.append(1.0 + 0.3 * np.cos(6.0 * np.pi * x + 0.5))
    # Gapped
    a_g = np.ones(n)
    a_g[1::3] = 0.4
    inits.append(a_g)
    # Random
    inits.append(np.clip(0.5 + 0.5 * rng.random(n), 0.01, 1000.0))
    # Triangular
    tri = 1.0 - 2.0 * np.abs(x - 0.5)
    inits.append(np.clip(tri, 0.01, 1000.0))
    # Log-periodic
    inits.append(1.0 + 0.3 * np.cos(2.0 * np.pi * np.log(x + 0.1)))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Multi-phase strategy:
      Phase 1 (25%): Log-space optimization from parent
      Phase 2 (20%): Time-domain refinement
      Phase 3 (20%): Spectral flattening exploration
      Phase 4 (25%): Fine log-space polish
      Phase 5 (10%): Final time-domain polish
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = float(budget_s)
    deadline = start + total - 5.0
    best_score = np.inf
    best_a = None

    # Determine n and get parent
    n = 6962
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if parent.size > 0:
                n = int(parent.size)
    except Exception:
        pass
    n = min(max(256, n), 10000)
    print(f"n={n}, budget={budget_s:.0f}s, parent={'yes' if parent is not None else 'no'}")

    # Initialize
    if parent is not None and parent.size == n:
        a_init = np.clip(parent, 0.0, 1000.0)
        s_init = compute_score(n, fast_self_conv(a_init), float(a_init.sum()))
        print(f"Parent score: {s_init:.6f}")
        if np.isfinite(s_init) and s_init < best_score:
            best_score = s_init
            best_a = a_init.copy()
    else:
        a_init = np.ones(n) / n
        best_score = 2.0
        best_a = a_init.copy()
        print(f"Default init score: {best_score:.6f}")

    # Phase 1: Log-space optimization from parent (25% of time)
    p1_end = start + 0.25 * (total - 5.0)
    if time.monotonic() < p1_end:
        sc, aa = optimize_logspace(best_a, p1_end, rng, init_lr=0.10, topk=12,
                                   patience=350, temp_scale=0.06, use_symmetry=False)
        print(f"Phase 1 (log-space, parent): {sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()

    # Phase 2: Time-domain refinement (20%)
    p2_end = start + 0.45 * (total - 5.0)
    if time.monotonic() < p2_end:
        sc, aa = optimize_timespace(best_a, p2_end, rng, init_lr=0.04, topk=20,
                                    patience=500, temp_scale=0.04, use_symmetry=False)
        print(f"Phase 2 (time-domain): {sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()

    # Phase 3: Spectral flattening exploration (15%)
    p3_end = start + 0.60 * (total - 5.0)
    if time.monotonic() < p3_end:
        sc, aa = optimize_spectral(best_a, p3_end, rng, init_strength=0.2, patience=80)
        print(f"Phase 3 (spectral): {sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()

    # Phase 4: Fine log-space polish (25%)
    p4_end = start + 0.85 * (total - 5.0)
    if time.monotonic() < p4_end:
        sc, aa = optimize_logspace(best_a, p4_end, rng, init_lr=0.008, topk=32,
                                   patience=700, temp_scale=0.03, use_symmetry=False)
        print(f"Phase 4 (fine log-space): {sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()

    # Phase 5: Final time-domain ultra-fine polish (10%)
    if time.monotonic() < deadline:
        sc, aa = optimize_timespace(best_a, deadline, rng, init_lr=0.002, topk=48,
                                    patience=1000, temp_scale=0.02, use_symmetry=False)
        print(f"Phase 5 (final polish): {sc:.6f}")
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()

    # Final output
    if best_a is None:
        best_a = np.ones(n) / n

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = np.inf
    print(f"FINAL: internal={best_score:.6f}, verified={verified:.6f}, n={n}, "
          f"time={time.monotonic()-start:.1f}s")
    return out
