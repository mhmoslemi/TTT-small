"""
Gradient-based multiplicative descent for minimizing 2*n*max(conv(a,a))/sum(a)^2
over non-negative sequences. Uses FFT-based convolution, annealed softmax
smoothing of the max operator, and adaptive learning rates. Combined with
block-scaling local search and diversification restarts.
"""

import numpy as np
import time

_GLOBAL_RNG = np.random.default_rng(42)

def _fft_conv_sq(a, L):
    """Compute full convolution of a with itself using FFT."""
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)
    return c[:2*len(a)-1]

def _score(a):
    """Compute the objective: 2*n*max(conv(a,a))/sum(a)^2"""
    n = len(a)
    c = _fft_conv_sq(a, _next_pow2(2*n-1))
    return 2.0 * n * float(c.max()) / float(a.sum())**2

def _next_pow2(x):
    x = int(x)
    L = 1
    while L < x:
        L <<= 1
    return L

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        return a
    return a / s

def _gradient_mmd(a, tau, L):
    """
    Compute the gradient of the smoothed objective and apply multiplicative update.
    Returns updated a and the current objective value.
    """
    n = len(a)
    c = _fft_conv_sq(a, L)
    maxc = float(c.max())
    
    # Smoothing weights
    shifted = c / tau
    shifted -= shifted.max()
    w = np.exp(shifted)
    w /= w.sum()
    
    # Gradient: g[j] = 2 * sum_k w[k] * a[k-j] (valid indices) = 2 * (w conv a)[j]
    # This is a cross-correlation / convolution
    Fw = np.fft.rfft(w, L)
    Fa = np.fft.rfft(a, L)
    g = np.fft.irfft(Fw * Fa, L)[:n]
    g = 2.0 * g
    
    return g, maxc

def _main_optimize(a0, time_budget, target=1.5027, verbose=True):
    """Main gradient-based optimization loop."""
    a = _normalize(a0.astype(float).copy())
    n = len(a)
    L = _next_pow2(2*n - 1)
    
    best = a.copy()
    best_sc = _score(a)
    cur_sc = best_sc
    
    deadline = time.monotonic() + time_budget
    lr = 0.5
    tau = n * 0.01  # initial smoothing
    
    stall = 0
    it = 0
    
    while time.monotonic() < deadline:
        g, maxc = _gradient_mmd(a, tau, L)
        
        # Multiplicative update
        a_new = a * np.exp(-lr * g / (g.max() + 1e-30))
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        sc = _score(a_new)
        
        if sc < cur_sc:
            # Accept, maybe increase lr slightly
            a = a_new
            cur_sc = sc
            lr = min(lr * 1.02, 5.0)
            if sc < best_sc:
                best_sc = sc
                best = a.copy()
                stall = 0
                if verbose and it % 5000 == 0:
                    print(f"  it={it} best={best_sc:.7f} cur={cur_sc:.7f} tau={tau:.4f} lr={lr:.3f}")
        else:
            # Reject, decrease lr
            lr = max(lr * 0.85, 0.01)
            stall += 1
        
        # Anneal tau
        if it % 100 == 0:
            tau = max(tau * 0.98, n * 0.001)
        
        # Diversification
        if stall > 200:
            a = best * np.exp(_GLOBAL_RNG.normal(0, 0.1, n))
            a = _normalize(a)

def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _fft_conv_sq(a, L):
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)
    return c[: 2 * len(a) - 1]


def _score(a, L=None):
    n = len(a)
    if L is None:
        L = _next_pow2(2 * n - 1)
    c = _fft_conv_sq(a, L)
    s = float(a.sum())
    if s < 1e-10:
        return float('inf')
    return 2.0 * n * float(c.max()) / (s * s)


def _normalize(a):
    a = np.maximum(a, 0.0)
    a = np.where(np.isfinite(a), a, 0.0)
    s = float(a.sum())
    if s <= 0 or not np.isfinite(s):
        return None
    return a / s


def _compute_gradient(a, w, L_grad):
    """Compute df_s/da = 2 * (w cross-correlate a) using FFT."""
    n = len(a)
    Fa = np.fft.rfft(a, L_grad)
    Fw = np.fft.rfft(w, L_grad)
    cross = np.fft.irfft(np.conjugate(Fa) * Fw, L_grad)[:n]
    return 2.0 * cross


def _gradient_mmd_step(a, tau, L_conv, L_grad, lr_list):
    """
    One gradient multiplicative descent step with line search.
    Returns (best_a, best_score, current_score).

    n = len(a)

    cur_sc = 2.0 * n * maxc  # assuming sum(a)=1

    shifted -= shifted.max()
    w = np.exp(shifted)
    w_sum = float(w.sum())
    if w_sum <= 0 or not np.isfinite(w_sum):
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum

    grad = _compute_gradient(a, w, L_grad)
    grad = np.maximum(grad, 0.0)
    grad_max = float(grad.max())

    best_sc = cur_sc

        sc = _score(a_try, L_conv)

def _spectral_init(n, rng):
    """Create a sequence with near-flat magnitude spectrum (circular)."""
    L = _next_pow2(2 * n - 1)
    # Target: flat magnitude for nonzero frequencies
    F = np.zeros(L // 2 + 1, dtype=complex)
    F[0] = 1.0  # DC = sum = 1
    n_hz = L // 2
    # Random phases for nonzero freqs
    phases = rng.uniform(0, 2 * np.pi, n_hz)
    mag = np.ones(n_hz) / np.sqrt(n_hz)
    F[1:] = mag * np.exp(1j * phases)
    a = np.fft.irfft(F, L)[:n]
    a = np.maximum(a, 0.0)
    a = _normalize(a)
    return a


def _resample(p, m):
    a = _normalize(p.astype(float))
    if a is None:
        return None
    n = len(a)
    if n == m:
        return a.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(m) + 0.5) / m
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 1e-12)
    return _normalize(out)

    def _add_cand(arr, sc=None):
        a = _normalize(arr.astype(float))
        if a is None:
            return
        if sc is None:
            sc = _score(a, L_conv)
        if not np.isfinite(sc):
            return
        candidates.append((sc, a.copy()))
        candidates.sort(key=lambda x: x[0])
        if len(candidates) > 12:
            del candidates[12:]

    if parent is not None:
        sc_p = _score(parent, L_conv)
        print(f"Parent score: {sc_p:.7f}")
        _add_cand(parent, sc_p)
    else:
        _add_cand(np.ones(N, dtype=float) / N)

    # Phase 1: Main optimization from parent (or best start)
    print(f"Phase 1: Main gradient optimization (target: {total * 0.40:.0f}s)")
    t1 = total * 0.40
    if len(candidates) > 0:
        p_best = candidates[0][1]
        p_opt, sc_opt = _main_optimize(p_best, t1, verbose=True)
        _add_cand(p_opt, sc_opt)
        print(f"  Phase 1 best: {candidates[0][0]:.7f}")

    # Phase 2: Coarse-to-fine from multiple starts
    print(f"Phase 2: Coarse-to-fine refinement")
    t2 = total * 0.15
    rem = deadline - time.monotonic()
    t2 = min(t2, max(0, rem - 60))

    coarse_n = N // 4
    starts = []
    if len(candidates) > 0:
        starts.append(_resample(candidates[0][1], coarse_n))
    starts.append(_spectral_init(coarse_n, _GLOBAL_RNG))
    starts.append(_resample(np.ones(coarse_n, dtype=float) / coarse_n, coarse_n))
    # Random block init
    a_rand = np.zeros(coarse_n)
    a_rand[:coarse_n // 2] = 1.0
    a_rand[coarse_n // 2:] = 0.3
    starts.append(_normalize(a_rand))

    valid_starts = [s for s in starts if s is not None]
    for idx, sp in enumerate(valid_starts[:3]):
        rem2 = deadline - time.monotonic()
        if rem2 < 30:
            break
        t_alloc = min(t2 / max(1, len(valid_starts[:3])), max(0, rem2 - 20))
        if t_alloc < 5:
            continue
        p_c, sc_c = _main_optimize(sp, t_alloc, verbose=False)
        # Upsample
        p_up = _resample(p_c, N)
        if p_up is not None:
            _add_cand(p_up)

    print(f"  Phase 2 best: {candidates[0][0]:.7f}")

    # Phase 3: Block scaling on best
    print(f"Phase 3: Block scaling local search")
    t3 = total * 0.20
    rem = deadline - time.monotonic()
    t3 = min(t3, max(0, rem - 60))
    if t3 > 10:
        p_block, sc_block = _block_search(candidates[0][1], t3, verbose=True)
        _add_cand(p_block, sc_block)
        print(f"  Phase 3 best: {candidates[0][0]:.7f}")

    # Phase 4: Diversification restarts
    print(f"Phase 4: Diversification restarts")
    t4 = total * 0.15
    rem = deadline - time.monotonic()
    t4 = min(t4, max(0, rem - 60))
    if t4 > 20:
        n_restarts = max(2, int(t4 / 80))
        t_per_restart = t4 / n_restarts
        for ri in range(n_restarts):
            rem4 = deadline - time.monotonic()
            if rem4 < 20:
                break
            # Create diverse starting point
            if ri % 2 == 0:
                base = candidates[0][1] * np.exp(_GLOBAL_RNG.normal(0, 0.15, N))
                base = _normalize(base)
            else:
                base = _spectral_init(N, _GLOBAL_RNG)
            if base is None:
                base = candidates[0][1].copy()
            p_div, sc_div = _main_optimize(base, t_per_restart, verbose=False)
            _add_cand(p_div, sc_div)
        print(f"  Phase 4 best: {candidates[0][0]:.7f}")

    # Phase 5: Final fine-tuning
    print(f"Phase 5: Final fine-tuning")
    rem = deadline - time.monotonic()
    t5 = min(total * 0.10, max(0, rem - 20))
    if t5 > 10:
        # Run gradient optimization with tighter parameters
        p_fin = candidates[0][1].copy()
        n = len(p_fin)
        Lc = _next_pow2(2 * n - 1)
        Lg = _next_pow2(3 * n - 1)
        best_fin = p_fin.copy()
        best_fin_sc = _score(p_fin, Lc)
        cur_fin = best_fin_sc
        lr = 0.2
        tau = n * 0.003
        dl5 = time.monotonic() + t5
        it5 = 0
        while time.monotonic() < dl5:
            c5 = _fft_conv_sq(p_fin, Lc)
            maxc5 = float(c5.max())
            sh = c5 / tau
            sh -= sh.max()
            w5 = np.exp(sh)
            ws = float(w5.sum())
            if ws > 0 and np.isfinite(ws):
                w5 /= ws
            else:
                w5 = np.zeros_like(c5)
                w5[int(np.argmax(c5))] = 1.0
            grad5 = _compute_gradient(p_fin, w5, Lg)
            grad5 = np.maximum(grad5, 0.0)
            gm = float(grad5.max())
            if gm > 1e-30:
                gn = grad5 / gm
                for lr_try in [lr * 0.5, lr, lr * 2.0]:
                    a_try = p_fin * np.exp(-lr_try * gn)
                    a_try = _normalize(a_try)
                    if a_try is None:
                        continue
                    sc_try = _score(a_try, Lc)
                    if sc_try < cur_fin - 1e-12:
                        cur_fin = sc_try
                        p_fin = a_try
                        if sc_try < best_fin_sc - 1e-12:
                            best_fin_sc = sc_try
                            best_fin = p_fin.copy()
                        break
                if verbose_f5 := True:
                    if it5 % 2000 == 0:
                        print(f"    fin it={it5} best={best_fin_sc:.7f}")
            it5 += 1
        _add_cand(best_fin, best_fin_sc)
        print(f"  Phase 5 best: {candidates[0][0]:.7f}")

    # Final block search on absolute best
    rem = deadline - time.monotonic()
    if rem > 30:
        t6 = min(rem - 15, 120.0)
        if t6 > 10:
            p6, sc6 = _block_search(candidates[0][1], t6, verbose=True)
            _add_cand(p6, sc6)
            print(f"  Final block best: {candidates[0][0]:.7f}")

    # Verify with evaluate_sequence on best candidates
    rem = deadline - time.monotonic()
    if rem > 5:
        for sc_est, arr in candidates[:4]:
            lst = [float(x) for x in arr]
            try:
                sc_eval = float(evaluate_sequence(lst))
                if np.isfinite(sc_eval):
                    print(f"  Evaluated candidate: score={sc_eval:.7f} (estimated={sc_est:.7f})")
                    if sc_eval < candidates[0][0]:
                        candidates[0] = (sc_eval, arr)
            except (Exception):
                pass

    # Return the best
    best_arr = candidates[0][1]
    out = np.asarray(best_arr, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s <= 0 or not np.isfinite(s):
        out = np.ones(N, dtype=float) / N
    else:
        out = out / s

    print(f"FINAL: best score = {candidates[0][0]:.7f}, length = {len(out)}")
    return [float(x) for x in out]
