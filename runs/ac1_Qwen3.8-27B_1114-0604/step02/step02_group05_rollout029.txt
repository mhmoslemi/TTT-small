def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(a, b, L=None):
    la, lb = len(a), len(b)
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)

"""
Optimized search for non-negative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses adaptive multiplicative gradient descent with line search, multi-peak tracking,
frequency-domain flattening, and block coordinate refinement.
Starts from parent sequence and explores neighborhoods via multiple strategies.

import numpy as np
import time


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_fft(a, b, L=None):
    la, lb = len(a), len(b)
    nout = la + lb - 1
    if nout <= 0:
        return np.array([], dtype=float)
    if la * lb <= 3000000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)


def _score_normalized(p):
    n = len(p)
    if n == 0:
        return float('inf')
    c = _conv_fft(p, p)
    if not np.isfinite(c).all():
        return float('inf')
    return float(2.0 * n * c.max())


def _score_full(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    pn = p / s
    return _score_normalized(pn)

def _gradient_topk(p, k_list, L=None):
    """Compute gradient of sum of top-k convolution peaks.
    Returns (score, gradient, peak_indices)."""
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)
    c = _conv_fft(p, p)
    score = float(2.0 * n * c.max())
    # Gradient of conv[k] w.r.t. a[j]: a[k-j] + a[j-(k-n+1)] type structure
    # For conv[k] = sum_i a[i]*a[k-i], d/d(a[j]) = a[k-j] (if valid) + a[j] (if k-j valid and j != k-j)
    # More precisely: d(conv[k])/d(a[j]) = a[k-j] (if 0<=k-j<n) + a[j] (if 0<=k-j<n and j != k-j)
    # Actually: conv[k] = sum_i a[i]*a[k-i], so d/d(a[j]) = a[k-j] if 0<=k-j<n
    #           plus a[j] if 0<=k-j<n (the second factor contributes when k-i=j, i.e., i=k-j)
    # So d(conv[k])/d(a[j]) = a[k-j] + a[j] if j != k-j, or 2*a[j] if j==k-j
    # Simplified: d(conv[k])/d(a[j]) = a[k-j] + a[j] (with a[x]=0 for x out of [0,n-1])
    # This is the cross-correlation of a with flip(a) evaluated at position j, shifted by k.
    
    # For efficiency, use FFT:
    # grad[j] = sum_k w[k] * (a[k-j] + a[j] * delta)
    # Actually the cleanest way: grad[j] = sum over peaks k of (a[k-j] + a[j]) where a[k-j]=0 if out of bounds
    # = (for each peak k): flip_correlate(a, k) + a
    
    # Let's use a different formulation. Define flip(a) = a[n-1::-1].
    # Then conv(a,a)[k] = <a, shift(flip_a, k-n+1)> (dot product with shifted reversed a)
    # The gradient of <a, shift(flip_a, s)> w.r.t. a is: shift(flip_a, s) + a (shifted appropriately)
    
    # Simpler: just compute directly for the top peaks.
    grad = np.zeros(n, dtype=float)
    for k in k_list:
        # conv[k] = sum_i a[i]*a[k-i] for valid i
        # d/d(a[j]) = a[k-j] if 0 <= k-j < n
        # Plus: if we write conv[k] = sum_i a[i]*a[k-i], the derivative w.r.t. a[j] collects:
        #   - terms where i=j: a[j]*a[k-j] -> derivative a[k-j]
        #   - terms where k-i=j, i.e., i=k-j: a[k-j]*a[j] -> derivative a[j]
        # If j == k-j (i.e., 2j==k), these are the same term, so derivative is 2*a[j]
        # Combined: d(conv[k])/d(a[j]) = a[k-j] (if valid) + a[j] (if k-j valid)
        # If k-j is out of [0,n-1], the first term is 0.
        # If j is such that k-j is out of range, the second term... 
        # Actually the second term a[j] appears when i=k-j is valid, i.e., 0<=k-j<n.
        # So both terms require 0<=k-j<n.
        # d(conv[k])/d(a[j]) = (a[k-j] if 0<=k-j<n else 0) + (a[j] if 0<=k-j<n else 0)
        #                     = (a[k-j] + a[j]) if 0<=k-j<n, else 0
        
        k_val = int(k)
        if 0 <= k_val < 2 * n - 1:

            j_lo = max(0, k_val - n + 1)
            j_hi = min(n - 1, k_val)
            if j_lo <= j_hi:

                idx = k_val - np.arange(j_lo, j_hi + 1)
                # idx ranges from k_val - j_lo to k_val - j_hi, both in [0, n-1]
                contrib = a_val_at(p, idx) + p[j_lo:j_hi + 1]
                grad[j_lo:j_hi + 1] += contrib
    grad = np.maximum(grad, 0.0)
    return score, grad


def a_val_at(p, idx):
    """Get p[idx] with out-of-bounds returning 0."""
    mask = (idx >= 0) & (idx < len(p))
    result = np.zeros(len(idx), dtype=float)
    result[mask] = p[idx[mask]]
    return result


def _optimize_gradient(p0, rng, time_budget, M=5, beta_schedule=None, verbose=False):
    """Adaptive multiplicative gradient descent focusing on top-M peaks."""
    p = np.array(p0, dtype=float)
    p = np.maximum(p, 1e-15)
    s = p.sum()
    p = p / s
    n = len(p)
    
    best_p = p.copy()
    best_score = _score_normalized(p)
    cur_score = best_score
    
    deadline = time.monotonic() + time_budget
    step_size = 0.1
    stall = 0
    it = 0
    
    while time.monotonic() < deadline:
        # Compute convolution and find top-M peaks
        c = _conv_fft(p, p)
        cur_score = float(2.0 * n * c.max())
        
        # Find top-M peak indices
        # Use a subset of peaks for efficiency
        if n <= 2000:
            topk_idx = np.argsort(c)[::-1][:M]
        else:
            # For large n, sample the top peaks
            c_sample = c[::max(1, n // 1000)]
            rough_top = np.argsort(c_sample)[::-1][:4*M // max(1, n // 1000)]
            # Refine around rough top
            candidates = []
            for ri in rough_top:
                lo = max(0, ri * max(1, n // 1000) - max(1, n // 1000))
                hi = min(len(c), (ri + 1) * max(1, n // 1000) + max(1, n // 1000))
                candidates.extend(range(lo, hi))
            candidates = list(set(candidates))
            if len(candidates) > 50:
                c_sub = np.array([c[i] for i in candidates])
                top_local = np.argsort(c_sub)[::-1][:M]
                topk_idx = np.array([candidates[i] for i in top_local])
            else:
                c_sub = np.array([c[i] for i in candidates])
                topk_idx = np.array([candidates[i] for i in np.argsort(c_sub)[::-1][:M]])
        
        # Compute gradient
        grad = np.zeros(n, dtype=float)
        for k in topk_idx:
            k_val = int(k)
            j_lo = max(0, k_val - n + 1)
            j_hi = min(n - 1, k_val)
            if j_lo <= j_hi:
                idx = k_val - np.arange(j_lo, j_hi + 1)
                valid = (idx >= 0) & (idx < n)
                contrib = np.zeros(j_hi - j_lo + 1, dtype=float)
                contrib[valid] = p[idx[valid]]
                contrib += p[j_lo:j_hi + 1]
                grad[j_lo:j_hi + 1] += contrib
        
        grad = np.maximum(grad, 1e-10)
        max_grad = grad.max()
        if max_grad < 1e-15:

            continue

        best_trial_score = cur_score
        for attempt in range(6):

            if a_try < 0.001:
                break

            if s_try < 1e-10:
                continue

                best_trial = p_try
                best_trial_score = sc_try
                step_size = min(10.0, step_size * 1.2)  # increase step size
            else:
                step_size = max(0.005, step_size * 0.7)  # decrease
        
        if best_trial is not None:
            p = best_trial
            cur_score = best_trial_score
            stall = 0
            if cur_score < best_score - 1e-12:

                best_score = cur_score

        else:

            if stall > 30:

                step_size = 0.1
                stall = 5
        
        it += 1
    
    return best_p, best_score


def _optimize_smoothed(p0, rng, time_budget, beta_start=20.0, beta_end=3000.0, verbose=False):

                w = e / s_e

        R = np.fft.irfft(W * np.conj(P), L)

        # Actually: IFFT(FFT(w) * conj(FFT(p))) gives circular cross-correlation
        # R[j] = sum_k w[(k+j) mod L] * p[k] ... this depends on convention
        # Let me use a different approach for clarity
        
        # Simpler: grad[j] = sum over k of w[k] * p[k-j] (circular with L padding)
        # Use FFT: IFFT(rfft(w, L) * conj(rfft(p, L))) gives circular xcorr
        # xcorr[j] = sum_k w[(j+k) mod L] * p[k]
        # We want: sum_k w[k] * p[k-j] = xcorr[(2n-1-j) mod L] or similar

        # Let m = j+k: = sum_m w[m] * p[(m-j) mod L]
        # So IFFT(FFT(w) * conj(FFT(p)))[j] = sum_m w[m] * p[m-j]
        # That's exactly g[j]!
        
        R = np.fft.irfft(W * np.conj(P), L)

            continue

        improved = False

            p_try = p * np.exp(-a_try * g_norm)

def _freq_flatten(p, alpha, L=None):
    """Frequency-domain convolution flattening."""
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(p, L)
    C_freq = A * A
    C = np.fft.irfft(C_freq, L)[:2 * n - 1]
    cmax = C.max()
    if cmax < 1e-15:
        return p.copy()
    C_norm = C / cmax
    C_new = cmax * np.power(C_norm, 1.0 / alpha)
    C_new = np.maximum(C_new, 0.0)
    C_freq_new = np.fft.rfft(C_new, L)
    # Take square root in frequency domain
    A_new = np.sqrt(C_freq_new.astype(complex) + 0j)
    a_new = np.fft.irfft(A_new, L)[:n]
    neg_frac = (a_new < -0.01).sum() / n
    if neg_frac > 0.2:
        return None
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s


def _block_refine(p0, rng, time_budget, verbose=False):
    """Block-wise multiplicative refinement."""
    p = np.array(p0, dtype=float)
    p = np.maximum(p, 1e-15)
    p = p / p.sum()
    n = len(p)
    best_p = p.copy()
    best_score = _score_normalized(p)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0
    it = 0
    factors = (0.5, 0.7, 0.85, 1.15, 1.3, 1.6, 2.0, 3.0)
    
    while time.monotonic() < deadline:
        # Choose block size and position
        r = rng.random()
        if r < 0.15:
            bs = int(rng.integers(1, 6))
        elif r < 0.5:
            bs = int(rng.integers(max(2, n // 300), max(3, n // 40)))
        else:
            bs = int(rng.integers(max(4, n // 50), max(8, n // 8)))
        bs = min(bs, n)
        start = int(rng.integers(0, n - bs + 1))
        
        improved = False
        for f in factors:
            p_try = p.copy()
            p_try[start:start + bs] *= f
            p_try = np.maximum(p_try, 1e-15)
            s_try = p_try.sum()
            if s_try < 1e-10:
                continue
            p_try = p_try / s_try
            c_try = _conv_fft(p_try, p_try)
            sc = float(2.0 * n * c_try.max())
            if sc < cur_score - 1e-12:

                    best_score = sc
                    if verbose and it % 100 == 0:
                        print(f"      block iter={it}, score={best_score:.6f}")
                break
        
        if not improved:
            stall += 1
            if stall > 40:
                p = best_p.copy()
                cur_score = best_score
                stall = 0
        
        it += 1
    
    return best_p, best_score


def _init_ramp(n, rng, ascending=True):
    """Linearly increasing or decreasing sequence."""
    if ascending:
        a = np.arange(1, n + 1, dtype=float)
    else:
        a = np.arange(n, 0, dtype=float)
    # Add some noise
    a = a * np.exp(rng.normal(0, 0.05, n))
    a = np.maximum(a, 0)
    return a / a.sum()


def _init_quadratic(n, rng, center=0.8):
    """Quadratic bump centered at 'center' fraction of the sequence."""
    t = np.arange(n, dtype=float) / n
    a = np.exp(-10 * (t - center) ** 2)
    a = a * np.exp(rng.normal(0, 0.03, n))
    a = np.maximum(a, 0)
    return a / a.sum()


def _init_two_level(n, rng):
    """Two-level random sequence."""
    k = int(n * rng.uniform(0.3, 0.7))
    ratio = rng.uniform(1.5, 4.0)
    a = np.ones(n, dtype=float)
    a[k:] = ratio
    idx = rng.permutation(n)
    a = a[idx]
    a = a * (1.0 + 0.1 * rng.normal(0, 1, n))
    a = np.maximum(a, 0)
    return a / a.sum()


def _init_parent_special(p, n, rng):
    """Resample parent to length n with perturbation."""
    old_n = len(p)
    if old_n == n:
        result = p.copy()
    else:
        x_old = (np.arange(old_n) + 0.5) / old_n
        x_new = (np.arange(n) + 0.5) / n
        result = np.interp(x_new, x_old, p)
    result = result * np.exp(rng.normal(0, 0.05, n))
    result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s

                sc = _score_normalized(a_par)
                try_update(a_par, sc)
    
    print(f"  Phase 1 done. Best: {best_score_overall:.6f}")
    
    # ========== PHASE 2: Main gradient optimization (40% time) ==========
    print("Phase 2: Main gradient optimization")
    t2_end = start_time + 0.50 * budget
    
    if best_overall is None:
        best_overall = _init_ramp(3072, rng)
        best_score_overall = _score_normalized(best_overall)
    
    n_main = len(best_overall)
    
    # Run 1: Smoothed gradient (moderate beta)
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(0.3 * (t2_end - start_time - 0.10 * budget), t_rem * 0.45, 300.0)
        t_alloc = max(20.0, t_alloc)
        print(f"  Run 1: smoothed gradient, n={n_main}, t={t_alloc:.0f}s")
        p, sc = _optimize_smoothed(best_overall, rng, t_alloc, beta_start=20.0, beta_end=2000.0, verbose=True)
        try_update(p, sc)
    
    # Run 2: Top-k gradient (sharp focus)
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(0.25 * (t2_end - start_time - 0.10 * budget), t_rem * 0.4, 200.0)
        t_alloc = max(15.0, t_alloc)
        print(f"  Run 2: top-k gradient, n={n_main}, t={t_alloc:.0f}s")
        p, sc = _optimize_gradient(best_overall, rng, t_alloc, M=5, verbose=True)
        try_update(p, sc)
    
    # Run 3: Start from slightly different perturbation
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(150.0, t_rem * 0.5)
        t_alloc = max(15.0, t_alloc)
        p_start = _init_parent_special(best_overall, n_main, rng)
        if p_start is None:
            p_start = best_overall * np.exp(rng.normal(0, 0.1, n_main))
            p_start = np.maximum(p_start, 0)
            p_start = p_start / p_start.sum()
        print(f"  Run 3: perturbed start, n={n_main}, t={t_alloc:.0f}s")
        p, sc = _optimize_smoothed(p_start, rng, t_alloc, beta_start=50.0, beta_end=3000.0, verbose=True)
        try_update(p, sc)
    
    # Run 4: Try a longer sequence
    t_rem = t2_end - time.monotonic()
    if t_rem > 60:
        n_long = min(8192, n_main * 2)
        if n_long != n_main:
            t_alloc = min(120.0, t_rem * 0.5)
            t_alloc = max(15.0, t_alloc)
            p_long = _init_parent_special(best_overall, n_long, rng)
            if p_long is not None:
                print(f"  Run 4: longer sequence, n={n_long}, t={t_alloc:.0f}s")
                p, sc = _optimize_smoothed(p_long, rng, t_alloc, beta_start=30.0, beta_end=2000.0, verbose=True)
                try_update(p, sc)
    
    print(f"  Phase 2 done. Best: {best_score_overall:.6f}")
    
    # ========== PHASE 3: Frequency domain + diverse restarts (20% time) ==========
    print("Phase 3: Frequency domain and diverse restarts")
    t3_end = start_time + 0.72 * budget
    
    t_rem = t3_end - time.monotonic()
    if t_rem > 60 and best_overall is not None:
        n_best = len(best_overall)
        L_fft = _next_pow2(2 * n_best - 1)
        
        # Try frequency flattening with various alphas
        for alpha_f in [1.3, 1.5, 2.0, 3.0, 5.0]:
            if time.monotonic() > t3_end - 10:
                break
            p_fl = _freq_flatten(best_overall, alpha_f, L_fft)
            if p_fl is not None:
                sc = _score_normalized(p_fl)
                try_update(p_fl, sc)
                # If improvement, continue optimizing from there
                if sc < best_score_overall + 0.01:
                    t_alloc = min(40.0, t3_end - time.monotonic() - 5)
                    if t_alloc > 10:
                        p, sc2 = _optimize_smoothed(p_fl, rng, t_alloc, beta_start=50.0, beta_end=2000.0, verbose=False)
                        try_update(p, sc2)
        
        # Diverse restarts
        restart_pbs = [
            _init_ramp(n_best, rng, ascending=True),
            _init_ramp(n_best, rng, ascending=False),
            _init_quadratic(n_best, rng, center=0.6),
            _init_two_level(n_best, rng),
        ]
        for sp in restart_pbs:
            if time.monotonic() > t3_end - 10:
                break
            t_alloc = min(40.0, t3_end - time.monotonic() - 5)
            if t_alloc > 10:
                p, sc = _optimize_smoothed(sp, rng, t_alloc, beta_start=30.0, beta_end=1500.0, verbose=False)
                try_update(p, sc)
    
    print(f"  Phase 3 done. Best: {best_score_overall:.6f}")
    
    # ========== PHASE 4: Block refinement (15% time) ==========
    print("Phase 4: Block refinement")
    t4_end = start_time + 0.88 * budget
    
    if best_overall is not None:
        t_rem = t4_end - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.6 * t_rem, 150.0)
            t_alloc = max(10.0, t_alloc)
            print(f"  Block refine: n={len(best_overall)}, t={t_alloc:.0f}s")
            p, sc = _block_refine(best_overall, rng, t_alloc, verbose=True)
            try_update(p, sc)
    
    print(f"  Phase 4 done. Best: {best_score_overall:.6f}")
    
    # ========== PHASE 5: Final polish (last 12%) ==========
    print("Phase 5: Final polish")
    
    if best_overall is not None:
        # High-beta gradient pass
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.4 * t_rem, 120.0)
            t_alloc = max(10.0, t_alloc)
            print(f"  Final gradient: t={t_alloc:.0f}s")
            p, sc = _optimize_smoothed(best_overall, rng, t_alloc, beta_start=200.0, beta_end=8000.0, verbose=True)
            try_update(p, sc)
        
        # Another block refine
        t_rem = deadline - time.monotonic()
        if t_rem > 15:
            t_alloc = min(0.3 * t_rem, 80.0)
            t_alloc = max(8.0, t_alloc)
            p, sc = _block_refine(best_overall, rng, t_alloc, verbose=False)
            try_update(p, sc)
        
        # One more gradient with very high beta
        t_rem = deadline - time.monotonic()
        if t_rem > 15:
            t_alloc = min(0.3 * t_rem, 60.0)
            t_alloc = max(5.0, t_alloc)
            p, sc = _optimize_smoothed(best_overall, rng, t_alloc, beta_start=500.0, beta_end=10000.0, verbose=True)
            try_update(p, sc)
    
    # ========== Final output ==========
    print(f"\n{'=' * 60}")
    if best_overall is None:
        best_overall = _init_ramp(3072, rng)
    
    p_final = _sanitize(best_overall)
    if p_final is None:
        p_final = np.ones(3072, dtype=float) / 3072
    
    final_score = _score_normalized(p_final)
    
    # Verify with official evaluator
    try:
        eval_score = float(evaluate_sequence([float(x) for x in p_final]))
        if np.isfinite(eval_score):
            final_score = min(final_score, eval_score)
            print(f"  Official eval score: {eval_score:.6f}")
    except Exception:
        pass
    
    print(f"FINAL: score={final_score:.6f} (n={len(p_final)})")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")
    
    out = np.where(np.isfinite(p_final), p_final, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = np.ones(3072, dtype=float) / 3072
    else:
        out = out / s
    
    return [float(x) for x in out]
