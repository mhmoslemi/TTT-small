"""
Step function optimizer using adaptive softmax-smoothed mirror descent with momentum,
backtracking line search, SA acceptance, and diverse restarts.
Key improvements over previous: O(n log n) gradient via FFT-based softmax-weighted
gradient, adaptive dual temperatures, and structured polish phase.
"""

import numpy as np
import time


def _get_fft_size(n):
    """Get FFT size for convolutions involving sequences of length n."""
    # Need size >= 3n-2 for convolve(w_rev, f) where w has length 2n-1
    return 1 << (3 * n - 2).bit_length()


def _convolve_self_fft(f, size):
    """Full self-convolution of f via FFT."""
    n = len(f)
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _softmax_grad(f, c, tau, size):
    """
    Compute the smoothed gradient of max(c) using softmax weighting.
    grad[j] = sum_k softmax(c/tau)[k] * (f[k-j] + f[j])
    = convolve(w_rev, f)[2n-2-j] + f[j]
    """
    n = len(f)
    # Softmax weights (numerically stable)
    c_shift = (c - c.max()) / tau
    w = np.exp(c_shift)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(c.argmax())] = 1.0
    else:
        w /= w_sum
    
    # Compute convolve(w_rev, f) via FFT
    w_rev = w[::-1].copy()
    W = np.fft.rfft(w_rev, size)
    F = np.fft.rfft(f, size)
    conv_wf = np.fft.irfft(W * F, size)
    
    # grad[j] = conv_wf[2n-2-j] + f[j]
    indices = 2 * n - 2 - np.arange(n)
    grad = conv_wf[indices] + f.copy()
    
    return grad


def _mirror_descent_step(f, grad, eta):
    """One mirror descent step on the simplex (exponentiated gradient)."""
    log_f = np.log(np.maximum(f, 1e-300))
    log_f = log_f - eta * grad
    # Subtract max for numerical stability
    log_f = log_f - log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy()
    return f_new / s


def _objective(f, size):
    """Compute 2n * max(convolve(f,f)) for a normalized sequence (sum=1)."""
    n = len(f)
    c = _convolve_self_fft(f, size)
    return 2.0 * n * c.max()


def _optimize_adaptive(f0, deadline, rng, n, size):
    """
    Adaptive mirror descent with softmax-smoothed gradient, momentum,
    backtracking line search, and SA acceptance.
    """
    f = f0.copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f /= s
    
    c = _convolve_self_fft(f, size)
    cur_peak = c.max()
    local_best = cur_peak
    local_best_f = f.copy()
    
    # Parameters
    eta = 0.5  # Learning rate
    tau = cur_peak * 0.1  # Softmax temperature
    tau_min = cur_peak * 1e-6
    beta = 0.85  # Momentum
    T_sa = cur_peak * 0.03  # SA temperature
    T_sa_floor = cur_peak * 1e-12
    grad_mom = np.zeros(n)
    
    last_improve = time.monotonic()
    iter_count = 0
    consecutive_worse = 0
    
    while time.monotonic() < deadline:
        # Recompute autocorrelation
        c = _convolve_self_fft(f, size)
        cur_peak = c.max()
        
        if cur_peak < local_best - 1e-15:
            local_best = cur_peak
            local_best_f = f.copy()
            last_improve = time.monotonic()
            consecutive_worse = 0
        
        stuck_time = time.monotonic() - last_improve
        
        # Adaptive parameters
        if stuck_time > 20:
            # Reheat
            tau = cur_peak * 0.3
            T_sa = cur_peak * 0.1
            eta = max(eta, 0.3)
            # Perturb
            mag = min(0.03, 0.005 + 0.002 * stuck_time / 60.0)
            f = f + mag / np.sqrt(n) * rng.standard_normal(n)
            f = np.maximum(f, 0.0)
            f /= f.sum()
            last_improve = time.monotonic()
            grad_mom = np.zeros(n)
        else:
            # Slow annealing
            tau = max(tau_min, tau * 0.999)
            T_sa = max(T_sa_floor, T_sa * 0.999)
        
        # Compute gradient
        grad = _softmax_grad(f, c, tau, size)
        
        # Momentum
        grad_mom = beta * grad_mom + (1.0 - beta) * grad
        
        # Backtracking line search
        best_f_try = f
        best_peak_try = cur_peak
        eta_try = eta
        for _ in range(5):
            f_try = _mirror_descent_step(f, grad_mom, eta_try)
            c_try = _convolve_self_fft(f_try, size)
            peak_try = c_try.max()
            if peak_try < cur_peak:
                best_f_try = f_try
                best_peak_try = peak_try
                eta_try = min(eta_try * 1.1, 5.0)  # Try larger step
            else:
                eta_try *= 0.5
                if eta_try < 1e-6:
                    break
        
        # SA acceptance
        delta = best_peak_try - cur_peak
        if delta <= 0:
            f = best_f_try
            eta = min(eta * 1.002, 3.0)
            consecutive_worse = 0
        elif T_sa > 1e-30:
            if np.log(max(rng.uniform(), 1e-300)) < -delta / T_sa:
                f = best_f_try
                consecutive_worse += 1
            else:
                eta *= 0.9
                consecutive_worse += 1
        else:
            eta *= 0.9
            consecutive_worse += 1
        
        iter_count += 1
        
        if iter_count % 5000 == 0:
            print(f"    iter {iter_count}: peak={cur_peak:.10f}, best={local_best:.10f}, "

    c = _convolve_self_fft(f, size)
    if c.max() < local_best:
        local_best = c.max()
        local_best_f = f.copy()
    
    return local_best_f, 2.0 * n * local_best


def _polish(f, deadline, rng, n, size):
    """Polish phase: coordinate descent with structured moves."""
    f = f.copy()
    f = np.maximum(f, 0.0)
    f /= f.sum()
    
    c = _convolve_self_fft(f, size)
    local_best = c.max()
    local_best_f = f.copy()
    passes = 0
    
    while time.monotonic() < deadline:
        c = _convolve_self_fft(f, size)
        cur_peak = c.max()
        
        if cur_peak < local_best - 1e-15:
            local_best = cur_peak
            local_best_f = f.copy()
        
        k_star = int(np.argmax(c))
        
        # Find high-contributing coordinates (donors)
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        contrib = np.zeros(n)
        idx = np.arange(lo, hi)
        contrib[lo:hi] = f[k_star - idx]
        # Actually, the contribution to c[k_star] from f[j] is f[k_star - j]
        # So contrib[j] = f[k_star - j] for j in [lo, hi)
        
        # Donors: high contrib values
        donor_candidates = np.argsort(contrib)[::-1][:20]
        donor_candidates = [d for d in donor_candidates if contrib[d] > 0.01 * contrib.max()]
        
        # Receivers: low contrib values
        receiver_candidates = np.argsort(contrib)[:20]
        
        improved = False
        for donor in donor_candidates[:10]:
            for frac in [0.02, 0.05, 0.1, 0.2, 0.3]:
                amt = f[donor] * frac
                if amt < 1e-15:
                    continue
                for recv in receiver_candidates[:10]:
                    if recv == donor:
                        continue
                    f_try = f.copy()
                    f_try[donor] -= amt
                    f_try[recv] += amt
                    f_try /= f_try.sum()
                    c_try = _convolve_self_fft(f_try, size)
                    if c_try.max() < cur_peak - 1e-14:
                        f = f_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        # Random moves if no structured improvement
        if not improved:
            for _ in range(20):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                frac = rng.choice([0.01, 0.03, 0.05, 0.1])
                amt = f[i] * frac
                if f[i] - amt < 0:
                    continue
                f_try = f.copy()
                f_try[i] -= amt
                f_try[j] += amt
                f_try /= f_try.sum()
                c_try = _convolve_self_fft(f_try, size)
                if c_try.max() < cur_peak - 1e-14:
                    f = f_try
                    improved = True
                    break
        
        # Small global perturbation
        if not improved and passes % 100 == 0:
            f_try = f + 0.001 / np.sqrt(n) * rng.standard_normal(n)
            f_try = np.maximum(f_try, 0.0)
            f_try /= f_try.sum()
            c_try = _convolve_self_fft(f_try, size)
            if c_try.max() < cur_peak - 1e-14:
                f = f_try
        
        passes += 1
    
    c = _convolve_self_fft(f, size)
    if c.max() < local_best:
        local_best = c.max()

import numpy as np
import time


def _get_fft_size(n):
    """Get FFT size sufficient for all convolutions needed."""
    # Self-conv needs >= 2n-1; gradient conv needs >= 3n-2
    return 1 << (3 * n - 2).bit_length()


def _convolve_self_fft(f, size):
    """Full self-convolution of f via FFT."""
    n = len(f)
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _convolve_self_fft_and_F(f, size):
    """Full self-convolution of f via FFT, also returns forward FFT of f."""
    n = len(f)
    F = np.fft.rfft(f, size)
    c = np.fft.irfft(F * F, size)[:2 * n - 1]
    return c, F


def _softmax_grad_fft(f, c, tau, F, size):
    """
    Compute smoothed gradient of max(c) using softmax weighting, reusing F=FFT(f).
    grad[j] = sum_k softmax(c/tau)[k] * f[k-j] + f[j]
    = convolve(w_rev, f)[2n-2-j] + f[j]

    c_shift = (c - c.max()) / tau
    w = np.exp(c_shift)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(c.argmax())] = 1.0
    else:
        w /= w_sum

    w_rev = w[::-1].copy()
    W = np.fft.rfft(w_rev, size)
    conv_wf = np.fft.irfft(W * F, size)

    indices = 2 * n - 2 - np.arange(n)
    grad = conv_wf[indices] + f.copy()
    return grad

    log_f = log_f - log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy()
    return f_new / s

        if iter_count % 10000 == 0:
            print(f"    iter {iter_count}: peak={cur_peak:.10f}, best={local_best:.10f}, "
                  f"eta={eta:.4f}, tau={tau:.2e}, stuck={stuck_time:.0f}s")

    c, _ = _convolve_self_fft_and_F(f, size)
    if c.max() < local_best:
        local_best = c.max()
        local_best_f = f.copy()

    return local_best_f, 2.0 * n * local_best


def _polish_sa(f, deadline, rng, n, size):
    """Polish with very small SA perturbations (local search)."""
    f = f.copy()
    f = np.maximum(f, 0.0)
    f /= f.sum()

    c, _ = _convolve_self_fft_and_F(f, size)
    cur_peak = c.max()
    local_best = cur_peak
    local_best_f = f.copy()

    T = local_best * 0.0005
    step_size = 0.003 / np.sqrt(n)
    passes = 0

    while time.monotonic() < deadline:
        # Try structured + random perturbations
        improved = False
        for _ in range(5):
            # Mix of global noise and targeted moves
            r = rng.random()
            if r < 0.4:
                f_try = f + step_size * rng.standard_normal(n)
            elif r < 0.7:
                # Move mass from high to low (based on current peak)
                c_cur, _ = _convolve_self_fft_and_F(f, size)
                k_star = int(c_cur.argmax())
                lo = max(0, k_star - n + 1)
                hi = min(n, k_star + 1)
                contrib = f[k_star - np.arange(lo, hi)]
                donor = lo + int(np.argmax(contrib))
                recv = int(rng.integers(0, n))
                if recv == donor:
                    continue
                amt = f[donor] * rng.uniform(0.01, 0.1)
                if f[donor] - amt < 0:
                    continue
                f_try = f.copy()
                f_try[donor] -= amt
                f_try[recv] += amt
            else:
                # Two-coordinate swap
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                amt = rng.uniform(0.005, 0.05) * max(f[i], f[j])
                if f[i] - amt < 0:
                    continue
                f_try = f.copy()
                f_try[i] -= amt
                f_try[j] += amt

            f_try = np.maximum(f_try, 0.0)
            s_try = f_try.sum()
            if s_try < 1e-10:
                continue
            f_try /= s_try

            c_try, _ = _convolve_self_fft_and_F(f_try, size)
            peak_try = c_try.max()

            if peak_try < cur_peak - 1e-15:
                f = f_try
                cur_peak = peak_try
                improved = True
                T *= 0.995
                step_size *= 0.998
                break
            elif T > 1e-30:
                if np.log(max(rng.uniform(), 1e-300)) < -(peak_try - cur_peak) / T:
                    f = f_try
                    cur_peak = peak_try
                    break

        if not improved:
            T = max(T * 0.999, local_best * 1e-14)
            step_size = max(step_size * 0.999, 1e-6 / np.sqrt(n))

        if cur_peak < local_best - 1e-15:
            local_best = cur_peak
            local_best_f = f.copy()

        passes += 1
        if passes % 5000 == 0:
            print(f"    polish pass {passes}: peak={cur_peak:.10f}, best={local_best:.10f}")

    return local_best_f, 2.0 * n * local_best


def _generate_starts(f_init, n, rng, count=30):
    """Generate diverse starting points."""
    starts = [f_init.copy()]
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    for alpha in [0.5, 1.0, 2.0, 5.0, 10.0]:
        starts.append(np.exp(-alpha * ((j - center) / center) ** 2))

    for alpha in [1.0, 3.0, 8.0]:
        starts.append(1.0 + alpha * ((j - center) / center) ** 2)

    for pos in [0.2, 0.25, 0.3, 0.35]:
        starts.append(
            np.exp(-((j - pos * n) / (0.04 * n)) ** 2) +
            np.exp(-((j - (1 - pos) * n) / (0.04 * n)) ** 2) + 0.001
        )

    for freq in [1, 2, 3, 4, 6, 8]:
        starts.append(1.0 + 0.5 * np.sin(2.0 * np.pi * freq * j / n))
        starts.append(1.0 + 0.6 * np.abs(np.sin(2.0 * np.pi * freq * j / n)))

    # Random
    for _ in range(max(0, count - len(starts))):
        r = rng.random()
        if r < 0.25:
            s = np.abs(rng.standard_normal(n)) + 0.1
        elif r < 0.5:
            s = rng.exponential(1.0, n)
        elif r < 0.7:
            s = np.ones(n) + 0.4 * rng.standard_normal(n)
            s = np.maximum(s, 0.01)
        elif r < 0.85:
            s = rng.dirichlet(np.ones(n) * 0.01)
        else:
            s = np.zeros(n)
            nnz = n // 3
            idx = rng.choice(n, nnz, replace=False)
            s[idx] = rng.exponential(1.0, nnz)
        starts.append(s)

    return starts


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(convolve(a,a))/(sum a)^2.
    Uses adaptive softmax-smoothed mirror descent with FFT-based gradient,
    momentum, backtracking, SA acceptance, diverse restarts, and SA polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0

    f_init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(f_init)
    s = f_init.sum()
    if s < 1e-10:
        f_init = np.ones(n)
    else:
        f_init = f_init / s

    size = _get_fft_size(n)
    print(f"n={n}, fft_size={size}")

    c0, _ = _convolve_self_fft_and_F(f_init, size)
    init_score = 2.0 * n * c0.max()
    print(f"Initial score: {init_score:.6f}")

    # Generate and score starts
    starts = _generate_starts(f_init, n, rng, count=35)
    scored = []
    for f0 in starts:
        f = np.maximum(f0, 0.0)
        s = f.sum()
        if s < 1e-10:
            f = np.ones(n) / n
        else:
            f = f / s
        c, _ = _convolve_self_fft_and_F(f, size)
        scored.append((c.max(), f))

    scored.sort(key=lambda x: x[0])
    print(f"Generated {len(starts)} starts in {time.monotonic()-t_start:.1f}s")
    print(f"Top 5 scores: {[f'{2*n*p:.6f}' for p, _ in scored[:5]]}")

    best_score = 2.0 * n * scored[0][0]
    best_f = scored[0][1].copy()
    print(f"Best initial: {best_score:.6f}")

    # Phase 1: Optimize top starts (65% budget)
    opt_budget = budget_s * 0.65
    polish_budget = budget_s * 0.22
    num_starts = 5
    time_per_start = opt_budget / num_starts

    for idx in range(num_starts):
        remaining = deadline - time.monotonic()
        if remaining < 90:
            break
        alloc = min(time_per_start, remaining - polish_budget - 30)
        if alloc < 30:
            break

        peak0, f0 = scored[idx]
        print(f"\nStart {idx+1} (score={2*n*peak0:.6f}, budget={alloc:.0f}s)")

        f_opt, score = _optimize_adaptive(f0, time.monotonic() + alloc, rng, n, size)

        if score < best_score:
            best_score = score
            best_f = f_opt.copy()
            print(f"  New best: {best_score:.6f}")
        else:
            print(f"  Best from start: {score:.6f}")

    # Phase 2: Polish (22% budget)
    print(f"\nPolishing (score={best_score:.6f})...")
    pol_deadline = min(time.monotonic() + polish_budget, deadline - 25)
    f_pol, pol_score = _polish_sa(best_f, pol_deadline, rng, n, size)
    if pol_score < best_score:
        best_score = pol_score
        best_f = f_pol.copy()
        print(f"  Polished: {best_score:.6f}")
    else:
        print(f"  Polish best: {pol_score:.6f}")

    # Phase 3: Final optimization if time remains
    remaining = deadline - time.monotonic()
    if remaining > 90:
        print(f"\nFinal optimization ({remaining:.0f}s)...")
        f_fin, fin_score = _optimize_adaptive(best_f, deadline - 10, rng, n, size)
        if fin_score < best_score:
            best_score = fin_score
            best_f = f_fin.copy()
            print(f"  Final: {best_score:.6f}")
        # Quick final polish
        remaining2 = deadline - time.monotonic()
        if remaining2 > 30:
            f_fin2, fin2_score = _polish_sa(best_f, min(time.monotonic() + remaining2 - 10, deadline - 5), rng, n, size)
            if fin2_score < best_score:
                best_score = fin2_score
                best_f = fin2_score and f_fin2.copy() or best_f
                if f_fin2 is not None:
                    best_f = f_fin2
                print(f"  Final polish: {best_score:.6f}")

    # Verify with official evaluator
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = evaluate_sequence(best_f.tolist())
    print(f"\nOfficial evaluation: {final_score:.6f}")

    if not np.isfinite(final_score):
        best_f = np.ones(n)
        final_score = evaluate_sequence(best_f.tolist())
        print(f"Fallback uniform: {final_score:.6f}")

    if np.isfinite(final_score) and final_score < best_score:
        best_score = final_score

    print(f"=== FINAL SCORE: {best_score:.6f} ===")
    return [float(x) for x in best_f]
