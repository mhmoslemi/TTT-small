import numpy as np
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]

def _score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)

def _grad_via_fft(w, a):
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    if w.size >= N:
        fw = np.fft.rfft(w, N)
    else:
        fw = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]

def _make_random_tone(n, num_tones, rng, max_total_amp=0.9):
    """Multi-tone with random frequencies."""
    freqs = []
    used = set()
    while len(freqs) < num_tones:
        f = rng.randint(1, n // 2 + 1)
        if f not in used:
            used.add(f)
            freqs.append(f)
    
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    
    amp_per = max_total_amp / num_tones
    phases = rng.uniform(0, 2 * np.pi, size=num_tones)
    
    for f, phi in zip(freqs, phases):
        a += amp_per * np.cos(2 * np.pi * f * i / n + phi)
    
    a = np.clip(a, 0.0, 1000.0)
    a /= a.sum()
    return a

def _optimize_mirror(a, deadline, T_start=5.0, T_end=50.0, momentum=0.9, 
                     lr0=None, patience=3000, seed=42):
    """
    Mirror descent with Nesterov momentum and adaptive learning rate.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size, dtype=np.float64) / a.size
        s = 1.0
    a = a / s
    
    n = a.size
    if lr0 is None:
        lr0 = 0.3 * np.sqrt(n)
    
    best_a = a.copy()
    best_score = _score(best_a)
    
    # Nesterov momentum
    v = np.zeros(n, dtype=np.float64)
    
    it = 0
    last_improve = 0
    rng = np.random.RandomState(seed)
    eval_interval = max(1, n // 256)  # Evaluate every n/256 steps
    
    # Backtracking parameters
    bt_factor = 0.5
    bt_max_tries = 10
    
    while time.monotonic() < deadline:
        it += 1
        
        # Annealing
        t_frac = min(1.0, it / 20000.0)
        T = T_start + (T_end - T_start) * t_frac ** 0.5
        lr = lr0 / (1.0 + 0.005 * it)
        
        # Compute convolution
        c = _fft_convolve(a)
        if c.size == 0:
            break
        
        # Softmax weights
        c_max = float(np.max(c))
        c_min = float(np.min(c))
        rng_c = c_max - c_min
        if rng_c < 1e-15:
            w = np.ones(c.size) / c.size
        else:
            c_norm = (c - c_min) / rng_c  # [0, 1]
            c_shift = c_norm * T - T  # [-T, 0]
            exp_c = np.exp(c_shift)
            w_sum = exp_c.sum()
            if w_sum < 1e-300:
                w = np.zeros(c.size)
                w[np.argmax(c)] = 1.0
            else:
                w = exp_c / w_sum
        
        # Gradient
        grad = _grad_via_fft(w, a)
        grad_proj = grad - np.mean(grad)
        
        # Nesterov: look-ahead step
        # v_new = momentum * v - lr * grad_proj
        # a_new = a * exp(v_new) / sum
        v = momentum * v - lr * grad_proj
        v = np.clip(v, -5.0, 5.0)
        
        a_new = a * np.exp(v)
        a_new = np.where(np.isfinite(a_new), a_new, 0.0)
        a_new = np.clip(a_new, 0.0, 1000.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
            s_new = s
        a_new = a_new / s_new
        
        # Simple acceptance: always accept (mirror descent doesn't need line search typically)
        # But check for improvement periodically
        a = a_new
        
        if it % eval_interval == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                # Restart with perturbation
                a = best_a.copy()
                noise_scale = 0.01 * a.mean()
                a = a + noise_scale * rng.standard_normal(n)
                a = np.clip(a, 0, 1000)
                s_r = a.sum()
                if s_r > 0.01:
                    a = a / s_r
                v = np.zeros(n, dtype=np.float64)
                patience = max(1000, int(patience * 1.5))
                lr0 = max(0.1 * np.sqrt(n), lr0 * 0.7)
    
    return best_a, best_score

def _optimize_hard_active(a, deadline, K=20, lr0=None, patience=2000, seed=137):
    """
    Hard active-set optimization: focus on top-K peaks with Nesterov momentum.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size, dtype=np.float64) / a.size
        s = 1.0
    a = a / s
    
    n = a.size
    if lr0 is None:
        lr0 = 0.2 * np.sqrt(n)
    
    best_a = a.copy()
    best_score = _score(best_a)
    
    v = np.zeros(n, dtype=np.float64)
    momentum = 0.85
    
    it = 0
    last_improve = 0
    rng = np.random.RandomState(seed)
    eval_interval = max(1, n // 256)
    
    while time.monotonic() < deadline:
        it += 1
        lr = lr0 / (1.0 + 0.003 * it)
        
        c = _fft_convolve(a)
        if c.size == 0:
            break
        
        # Top-K with soft weights
        if c.size <= K:
            top_idx = np.arange(c.size)
        else:
            top_idx = np.argpartition(c, -K)[-K:]
        
        top_vals = c[top_idx]
        T_local = max(1.0, 5.0 / (1.0 + it / 1000.0))  # Anneal from 5 to 1
        top_shift = top_vals - top_vals.max()
        weights = np.exp(T_local * top_shift)
        w_sum = weights.sum()
        if w_sum < 1e-300:
            weights = np.ones(K) / K
        else:
            weights /= w_sum
        
        w = np.zeros(c.size, dtype=np.float64)
        for i, k in enumerate(top_idx):
            w[k] = weights[i]
        
        grad = _grad_via_fft(w, a)
        grad_proj = grad - np.mean(grad)
        
        v = momentum * v - lr * grad_proj
        v = np.clip(v, -5.0, 5.0)
        
        a_new = a * np.exp(v)
        a_new = np.where(np.isfinite(a_new), a_new, 0.0)
        a_new = np.clip(a_new, 0.0, 1000.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
            s_new = s
        a_new = a_new / s_new
        a = a_new
        
        if it % eval_interval == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                a = best_a.copy()
                noise_scale = 0.005 * a.mean()
                a = a + noise_scale * rng.standard_normal(n)
                a = np.clip(a, 0, 1000)
                s_r = a.sum()
                if s_r > 0.01:
                    a = a / s_r
                v = np.zeros(n, dtype=np.float64)
                patience = max(500, int(patience * 1.5))
                lr0 = max(0.05 * np.sqrt(n), lr0 * 0.7)
    
    return best_a, best_score

def _ensemble_optimize(initial_candidates, deadline, num_parallel=3):
    """
    Run multiple optimization trajectories in sequence (since we have 1 CPU).
    Allocate time proportionally.
    """
    total_time = deadline - time.monotonic()
    time_per = total_time / len(initial_candidates)
    
    all_results = []
    
    for i, (a_init, source_name) in enumerate(initial_candidates):
        time_left = deadline - time.monotonic()
        if time_left < 30:
            break
        
        alloc = min(time_per, time_left - 30)
        if i == len(initial_candidates) - 1:
            alloc = time_left - 30
        
        if alloc < 10:
            break
        
        # Phase A: Soft optimization
        d_soft = time.monotonic() + alloc * 0.6
        a_soft, s_soft = _optimize_mirror(a_init, d_soft, seed=100 + i)
        
        # Phase B: Hard optimization from soft result
        d_hard = time.monotonic() + alloc * 0.4
        a_hard, s_hard = _optimize_hard_active(a_soft, d_hard, seed=200 + i)
        
        all_results.append((a_soft, s_soft, f"soft_{source_name}"))
        all_results.append((a_hard, s_hard, f"hard_{source_name}"))
        
        print(f"  [{source_name}] soft={s_soft:.8f}, hard={s_hard:.8f}", flush=True)
    
    return all_results

"""
Low-peak step-function optimization via multi-length mirror descent with
Nesterov momentum, adaptive learning rate, and multi-tone initialization.

Key improvements over previous:
1. Multi-tone initialization with well-spread DFT frequencies
2. Nesterov momentum for faster convergence
3. Adaptive learning rate with periodic restart
4. Multiple sequence lengths explored (1024 to 16384)
5. Temperature annealing from soft to hard constraints
6. Starts from height_sequence_1 if available
7. Active-set tracking for more aggressive focusing on critical peaks
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Full linear auto-convolution via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    """Compute scale-invariant evaluation score."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _grad_via_fft(w, a):
    """Gradient of sum_k w[k]*C[k] w.r.t. a, via FFT cross-correlation."""
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]


def _make_multi_tone(n, num_tones, rng, max_total_amp=0.85):
    """Multi-tone sinusoidal with well-spread frequencies (deterministic spacing)."""
    freqs = []
    used = set()
    for k in range(num_tones):
        f = int(round((k + 1) * n / (num_tones + 1)))
        if f < 1:
            f = 1
        if f >= n // 2:
            f = n // 2 - 1
        if f < 1:
            f = 1
        if f not in used:
            used.add(f)
            freqs.append(f)
    
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    
    amp_per = max_total_amp / max(1, len(freqs))
    phases = rng.uniform(0, 2 * np.pi, size=len(freqs))
    
    for f, phi in zip(freqs, phases):
        a += amp_per * np.cos(2.0 * np.pi * f * i / n + phi)
    
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    return a


def _make_random_tone(n, num_tones, rng, max_total_amp=0.80):
    """Multi-tone with random distinct frequencies."""
    freqs = []
    used = set()
    for _ in range(20 * num_tones):
        if len(freqs) >= num_tones:
            break
        f = rng.randint(1, n // 2 + 1)
        if f not in used:
            used.add(f)
            freqs.append(f)
    
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    
    amp_per = max_total_amp / max(1, len(freqs))
    phases = rng.uniform(0, 2 * np.pi, size=len(freqs))
    
    for f, phi in zip(freqs, phases):
        a += amp_per * np.cos(2.0 * np.pi * f * i / n + phi)
    
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    return a


def _make_chirp(n, rng, max_amp=0.7):
    """Linear chirp (quadratic phase) - good for spreading energy."""
    i = np.arange(n, dtype=np.float64)
    # Quadratic phase: frequency sweeps from low to high
    max_freq = n // 4
    theta = 2.0 * np.pi * max_freq * i * i / (2.0 * n * n)
    a = 1.0 + max_amp * np.cos(theta + rng.uniform(0, 2 * np.pi))
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    return a


def _optimize_mirror(a, deadline, T_start=5.0, T_end=60.0, momentum=0.9,
                     lr0=None, patience=4000, seed=42, topK_soft=0):
    """
    Mirror descent with Nesterov momentum.
    topK_soft: if > 0, blend between full softmax and top-K weighting.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size, dtype=np.float64) / a.size
        s = 1.0
    a = a / s
    
    n = a.size
    if lr0 is None:
        lr0 = 0.3 * np.sqrt(n)
    
    best_a = a.copy()
    best_score = _score(best_a)
    
    v = np.zeros(n, dtype=np.float64)
    
    it = 0
    last_improve = 0
    rng = np.random.RandomState(seed)
    eval_interval = max(3, n // 512)
    
    while time.monotonic() < deadline:
        it += 1
        
        # Annealing: temperature increases over time (softer weights = smoother landscape)
        # Actually we want to go from soft (high T) to hard (low T) for refinement
        t_frac = min(1.0, it / max(1, int((deadline - time.monotonic()) / 0.001)) if it > 1 else 0)
        T = T_start + (T_end - T_start) * (1.0 - t_frac) ** 0.7
        lr = lr0 / (1.0 + 0.003 * it)
        
        # Compute convolution
        c = _fft_convolve(a)
        if c.size == 0:
            break
        
        c_max = float(np.max(c))
        
        if topK_soft > 0 and c.size > topK_soft:
            # Blend: use topK for focus + softmax for smoothness
            top_idx = np.argpartition(c, -topK_soft)[-topK_soft:]
            top_vals = c[top_idx]
            T_local = max(1.0, T * 0.3)
            shift = top_vals - top_vals.max()
            w_top = np.exp(T_local * shift)
            w_top /= w_top.sum()
            
            # Full softmax with higher T (smoother)
            c_min = float(np.min(c))
            rng_c = c_max - c_min
            if rng_c < 1e-15:
                w_full = np.ones(c.size) / c.size
            else:
                c_norm = (c - c_min) / rng_c
                c_shift = c_norm * T - T
                exp_c = np.exp(c_shift)
                w_sum = exp_c.sum()
                w_full = exp_c / w_sum if w_sum > 1e-300 else np.ones(c.size) / c.size
            
            blend = 0.5 + 0.5 * t_frac  # More top-K focused as we progress
            w = (1.0 - blend) * w_full
            for i, k in enumerate(top_idx):
                w[k] += blend * w_top[i]
        else:
            # Pure softmax
            c_min = float(np.min(c))
            rng_c = c_max - c_min
            if rng_c < 1e-15:
                w = np.ones(c.size) / c.size
            else:
                c_norm = (c - c_min) / rng_c
                c_shift = c_norm * T - T
                exp_c = np.exp(c_shift)
                w_sum = exp_c.sum()
                w = exp_c / w_sum if w_sum > 1e-300 else np.ones(c.size) / c.size
        
        # Gradient
        grad = _grad_via_fft(w, a)
        grad_proj = grad - np.mean(grad)
        
        # Nesterov momentum update
        v = momentum * v - lr * grad_proj
        v = np.clip(v, -4.0, 4.0)
        
        a_new = a * np.exp(v)
        a_new = np.where(np.isfinite(a_new), a_new, 0.0)
        a_new = np.clip(a_new, 0.0, 1000.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            a_new = a.copy()
            s_new = s
        a_new = a_new / s_new
        a = a_new
        
        if it % eval_interval == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                # Restart with controlled perturbation
                a = best_a.copy()
                noise_scale = 0.008 * a.mean()
                a = a + noise_scale * rng.standard_normal(n)
                a = np.clip(a, 0, 1000)
                s_r = a.sum()
                if s_r > 0.01:
                    a = a / s_r
                v = np.zeros(n, dtype=np.float64)
                patience = max(2000, int(patience * 1.4))
                lr0 = max(0.08 * np.sqrt(n), lr0 * 0.75)
    
    return best_a, best_score

    """

    s = a.sum()

        a = np.ones(a.size, dtype=np.float64) / a.size

    last_improve = 0
    rng = np.random.RandomState(seed)
    eval_interval = max(3, n // 512)

        it += 1
        lr = lr0 / (1.0 + 0.002 * it)

        if c.size == 0:
            break
        
        # Adaptive K: start large, decrease for focus

        a_new = np.where(np.isfinite(a_new), a_new, 0.0)
        a_new = np.clip(a_new, 0.0, 1000.0)

        if s_new < 1e-10:

                noise_scale = 0.005 * a.mean()
                a = a + noise_scale * rng.standard_normal(n)
                a = np.clip(a, 0, 1000)
                s_r = a.sum()
                if s_r > 0.01:
                    a = a / s_r
                v = np.zeros(n, dtype=np.float64)
                patience = max(1500, int(patience * 1.4))
                lr0 = max(0.06 * np.sqrt(n), lr0 * 0.7)
    
    return best_a, best_score


def _coordinate_polish(a, deadline, num_rounds=1000, seed=555):
    """
    Coordinate descent polish: iteratively adjust individual elements
    to reduce the max convolution peak while maintaining non-negativity.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return a, _score(a)
    a = a / s
    
    n = a.size
    best_a = a.copy()
    best_score = _score(best_a)
    rng = np.random.RandomState(seed)
    
    it = 0
    while time.monotonic() < deadline and it < num_rounds:
        it += 1
        c = _fft_convolve(a)
        if c.size == 0:
            break
        peak_idx = int(np.argmax(c))
        peak_val = c[peak_idx]
        
        # Find which a[j] most increases C[peak_idx]
        # dC[peak_idx]/da[j] = a[peak_idx-j] + a[j-peak_idx] (for valid indices)
        # To DECREASE the peak, we want to decrease a[j] where derivative is positive
        # Or increase a[j] where derivative is negative (but a[j] >= 0)
        
        # Compute gradient of C[peak_idx] w.r.t. a
        w = np.zeros(c.size, dtype=np.float64)
        w[peak_idx] = 1.0
        grad = _grad_via_fft(w, a)  # 2 * crosscorr(w, a)[:n]
        
        # Find the element with largest positive gradient (most contributes to peak)
        if np.max(grad) > 0:
            j = int(np.argmax(grad))
            # Reduce a[j] slightly
            delta = 0.02 * a[j]
            a[j] = max(0, a[j] - delta)
            # Redistribute to a random element with small value
            j2 = int(np.argmin(a))
            a[j2] += delta
            a = a / a.sum()
        
        if it % 5 == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
    
    return best_a, best_score


def _run_trajectory(a_init, deadline, stage1_frac=0.55, topK=0):
    """Run a full optimization trajectory: soft phase then hard phase."""
    n = a_init.size
    
    # Stage 1: Soft mirror descent
    d1 = time.monotonic() + (deadline - time.monotonic()) * stage1_frac
    if time.monotonic() < d1:
        a1, s1 = _optimize_mirror(a_init, d1, T_start=4.0, T_end=80.0,
                                  momentum=0.9, topK_soft=topK, seed=hash(a_init.tobytes()) % 10000)
    else:
        a1, s1 = a_init, _score(a_init)
    
    # Stage 2: Hard active-set
    if time.monotonic() < deadline - 10:
        a2, s2 = _optimize_hard_active(a1, deadline - 10, K=max(8, n // 64), seed=hash(a_init.tobytes()) % 10000 + 5000)
    else:
        a2, s2 = a1, s1
    
    # Stage 3: Coordinate polish (brief)
    if time.monotonic() < deadline - 5:
        d_polish = min(deadline - 5, time.monotonic() + 30)
        a3, s3 = _coordinate_polish(a2, d_polish)
    else:
        a3, s3 = a2, s2
    
    # Return best of all stages
    candidates = [(a1, s1), (a2, s2), (a3, s3)]
    candidates.sort(key=lambda x: x[1])
    return candidates

        if np.isfinite(sc):
            scored_init.append((sc, a, name))
    scored_init.sort(key=lambda x: x[0])
    
    print(f"\n[Init Summary] {len(scored_init)} candidates, best={scored_init[0][0]:.6f}", flush=True)
    
    # Allocate time to trajectories
    # Give more time to shorter sequences (faster convergence) and the best inits
    # Strategy: top 4-6 candidates get optimized
    max_trajs = 5
    trajs_to_run = scored_init[:max_trajs]
    
    # Time allocation: proportional to sqrt(n) (longer = more needed but slower per iter)
    weights = [np.sqrt(a.size) for _, a, _ in trajs_to_run]
    total_w = sum(weights)
    total_time = deadline - time.monotonic() - 30  # Reserve 30s for validation
    
    trajectories = []
    
    for i, (sc_init, a_init, name) in enumerate(trajs_to_run):
        time_alloc = total_time * weights[i] / total_w
        # Cap: shorter sequences don't need too much time
        time_alloc = min(time_alloc, total_time * 0.5)
        # Minimum: at least 30s
        time_alloc = max(time_alloc, 30.0)
        
        d_end = time.monotonic() + time_alloc
        
        print(f"\n[Traj {i+1}/{len(trajs_to_run)}] {name} (n={a_init.size}, "
              f"init={sc_init:.6f}, alloc={time_alloc:.0f}s)", flush=True)
        
        if time.monotonic() >= d_end:
            trajectories.append((a_init, sc_init, name))
            continue
        
        # Choose topK for soft phase based on length
        topK = max(0, a_init.size // 128)
        
        result = _run_trajectory(a_init, d_end, stage1_frac=0.55, topK=topK)
        for a_r, s_r in result:
            trajectories.append((a_r, s_r, name))
        
        print(f"  Best: {result[0][1]:.8f}", flush=True)
    
    # Final: coordinate polish on the overall best
    if trajectories:
        trajectories.sort(key=lambda x: x[1])
        if trajectories[0][1] < np.inf and time.monotonic() < deadline - 10:
            best_a = trajectories[0][0]
            d_polish = deadline - 10
            polish_time = min(60.0, d_polish - time.monotonic())
            if polish_time > 10:
                print(f"\n[Polish] Best trajectory ({polish_time:.0f}s)...", flush=True)
                a_pol, s_pol = _coordinate_polish(best_a, time.monotonic() + polish_time)
                trajectories.append((a_pol, s_pol, "polished"))
                if s_pol < trajectories[0][1]:
                    trajectories.sort(key=lambda x: x[1])
                    print(f"  Polished: {s_pol:.8f}", flush=True)
    
    # Validation
    print("\n[Validation]", flush=True)
    
    # Deduplicate by keeping only the best score per length
    seen = {}
    for a, sc, name in trajectories:
        if not np.isfinite(sc):
            continue
        n = a.size
        if n not in seen or sc < seen[n][1]:
            seen[n] = (a, sc, name)
    
    final_candidates = sorted(seen.values(), key=lambda x: x[1])
    
    best_out = None
    best_actual = np.inf
    
    for a, sc, name in final_candidates[:10]:
        if time.monotonic() > deadline:
            break
        arr_list = [float(max(0.0, min(1000.0, x))) for x in a]
        try:
            actual = float(evaluate_sequence(arr_list))
        except Exception as e:
            actual = np.inf
            print(f"  {name}: eval error {e}", flush=True)
        
        if np.isfinite(actual):
            if actual < best_actual:
                best_actual = actual
                best_out = arr_list
            print(f"  {name} (n={a.size}): internal={sc:.8f}, actual={actual:.8f}", flush=True)
    
    if best_out is None:
        # Ultimate fallback
        a_fb = scored_init[0][1]
        best_out = [float(max(0.0, min(1000.0, x))) for x in a_fb]
        try:
            best_actual = float(evaluate_sequence(best_out))
        except:
            best_actual = scored_init[0][0]
        print(f"  Fallback: {best_actual:.8f}", flush=True)
    
    total_time = time.monotonic() - t0
    print(f"\n[Done] Time: {total_time:.1f}s, Best actual: {best_actual:.8f}", flush=True)
    return best_out
