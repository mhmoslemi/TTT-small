def _score_from_conv(a, c, metric='full', zero_weight=0.0):
    n = a.size
    s = float(np.sum(a))
    if s < 0.01 or not np.isfinite(s):
        return np.inf
    if c.size == 0:
        return np.inf
    full_score = 0.0
    off_score = 0.0
    need_full = (metric == 'full') or (zero_weight > 0.0)
    if need_full:
        fm = float(np.max(c))
        full_score = 2.0 * n * fm / (s * s)
    if metric != 'full':
        if n <= 1:
            om = 0.0
        else:
            left = c[:n-1]
            right = c[n:]
            lm = float(np.max(left)) if left.size else 0.0
            rm = float(np.max(right)) if right.size else 0.0
            om = lm if lm > rm else rm
        off_score = 2.0 * float(n) * om / (s * s)
    if metric == 'full':
        return float(full_score)
    return float(off_score + zero_weight * full_score)

def _population_score(A, metric='off', zero_weight=0.0):
    A = np.asarray(A, dtype=np.float64)
    if A.ndim == 1:
        A = A.reshape(1, -1)
    P, n = A.shape
    scores = np.full(P, np.inf, dtype=np.float64)
    S = A.sum(axis=1)
    valid = np.isfinite(S) & (S >= 0.01)
    if not np.any(valid): return scores
    Av = A[valid]
    Sv = S[valid]
    size = _fft_len(n)
    F = np.fft.rfft(Av, size=size, axis=1)
    C = np.fft.irfft(F * F, size=size, axis=1)[:, :2 * n - 1]
    np.maximum(C, 0.0, out=C)
    base = np.empty(Av.shape[0], dtype=np.float64)
    if metric == 'full':
        cmax = C.max(axis=1)
        base = 2.0 * n * cmax / (Sv * Sv)
    else:
        if n <= 1:
            cmax = np.zeros(Av.shape[0], dtype=np.float64)
        else:
            left = C[:, :n - 1]
            right = C[:, n:]
            if left.size == 0:
                lm = np.zeros(Av.shape[0], dtype=np.float64)
            else:
                lm = left.max(axis=1)
            if right.size == 0:
                rm = np.zeros(Av.shape[0], dtype=np.float64)
            else:
                rm = right.max(axis=1)
            cmax = np.maximum(lm, rm)
        off = 2.0 * n * cmax / (Sv * Sv)
        if zero_weight > 0.0:
            fullmax = C.max(axis=1)
            off = off + zero_weight * (2.0 * n * fullmax / (Sv * Sv))
        base = off
    scores[valid] = base
    return scores
```

def _perturb_seq(a, rng, mode=0):
    n=a.size
    a=np.asarray(a, dtype=np.float64)
    if mode == 0:
        b = a * (1.0 + 0.1 * rng.standard_normal(n))
    elif mode == 1:
        t = np.arange(n)/max(1,n)
        freq = int(rng.integers(1, max(2, n//8)+1))
        phase = float(rng.uniform(0, 2*np.pi))
        amp=float(rng.uniform(0.05,0.5))
        b = a * (1.0 + amp*np.sin(2*np.pi*freq*t+phase))
    elif mode == 2:
        b=a.copy()
        i=int(rng.integers(0,n)); j=int(rng.integers(0,n))
        if i != j:
           amt = b[i]*float(rng.uniform(0.05,0.4))
           b[i] = max(0.0, b[i]-amt)
           b[j] = b[j] + amt
    elif mode == 3:
        m=max(1, int(np.sqrt(n)))
        b=np.zeros(n)
        if n > m:
           idx = rng.choice(n, size=m, replace=False)
        else:
           idx = np.arange(n)
        b[idx]=1.0
    else:
        count=int(np.count_nonzero(a))
        scale = 0.05 / max(1, count)
        b = a + rng.normal(0.0, scale, n)
    return _normalize(b)
```

    M = float(b.max())

    best_actual_seq=best.copy()

    size2 = 1 << (3*n - 2).bit_length()
    start = time.monotonic()
    total = max(1.0, deadline - start)
    it=0; stuck=0; patience=20; step=0.5; evals=0
    while time.monotonic() < deadline:
        progress = min(1.0, (time.monotonic()-start)/total)
        if progress < 0.55:
            sp = progress / 0.55
            gamma = 1.0 + 60.0 * (sp**1.2)
            step = 0.55 * (1.0 - 0.75*sp)

```python
"""
Search for non-negative sequences minimizing
    2 * n * max(convolve(a, a)) / (sum(a))**2.

Key ideas:
- Normalize sum(a) = 1; scale invariance is used throughout.
- The convolution peak is updated with a smooth p-norm / log-sum-exp surrogate,
  then polished with exact top-K active-set gradients.
- The full gradient for weights w_k on b_k = (a*a)[k] is computed efficiently
  via an FFT convolution: grad_j = 2 * sum_k w_k a_{k-j}.
- Multiple initializations are used, including small-length solved shapes upsampled
  to the target length, random shapes, sinusoids, and the supplied sequence.
- The official evaluator is used only for final validation and limited checks,
  while a fast FFT-cone expression drives the local search.
"""

import numpy as np
import time


def _fft_len(need: int) -> int:
    m = 1
    need = max(1, int(need))
    while m < need:
        m <<= 1
    return m


def _get_base_sequence():
    base = globals().get("height_sequence_1", None)
    if base is None:
        return None
    try:
        arr = np.asarray(base, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    return arr


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 1e-12:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _convolve_fast(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    size = _fft_len(2 * n - 1)
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)
    c = c[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return c


def _score_fast(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 0.01:
        return np.inf
    c = _convolve_fast(a)
    if c.size == 0:
        return np.inf
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return float((2.0 * n * m) / (s * s))


def _call_evaluator(seq):
    try:
        py_seq = [float(x) for x in seq]
        val = evaluate_sequence(py_seq)
    except NameError:
        val = _score_fast(np.asarray(seq, dtype=np.float64))
    except Exception:
        val = np.inf
    try:
        return float(val)
    except Exception:
        return np.inf


def _mirror_step(a, grad, step):
    a = np.asarray(a, dtype=np.float64).ravel()
    grad = np.nan_to_num(np.asarray(grad, dtype=np.float64), nan=0.0, posinf=0.0, neginf=0.0)
    span = float(np.max(grad) - np.min(grad))
    if span < 1e-300:
        return a.copy()
    eta = step / span
    log_a = np.log(a + 1e-300) - eta * grad
    log_a -= float(np.max(log_a))
    na = np.exp(log_a)
    na = np.maximum(na, 0.0)
    s = float(np.sum(na))
    if s < 1e-300:
        return a.copy()
    return na / s


def _perturb_seq(a, rng, mode=0):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n == 0:
        return a.copy()
    t = np.arange(n, dtype=np.float64) / float(n)

    if mode == 0:
        b = a * (1.0 + 0.12 * rng.standard_normal(n))
    elif mode == 1:
        freq = int(rng.integers(1, max(2, n // 8 + 1)))
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        amp = float(rng.uniform(0.05, 0.55))
        b = a * (1.0 + amp * np.sin(2.0 * np.pi * freq * t + phase))
    elif mode == 2:
        b = a.copy()
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i != j:
            amt = float(b[i] * rng.uniform(0.05, 0.45))
            b[i] = max(0.0, b[i] - amt)
            b[j] = b[j] + amt
    elif mode == 3:
        m = int(max(1, min(n, 1.2 * np.sqrt(n))))
        b = np.zeros(n, dtype=np.float64)
        idx = rng.choice(n, size=m, replace=False)
        b[idx] = 1.0
    elif mode == 4:
        count = int(np.count_nonzero(a))
        scale = 0.05 / max(1, count)
        b = a + rng.normal(0.0, scale, n)
    else:
        alpha = float(rng.uniform(-1.5, 1.5))
        tilt = np.exp(alpha * (t - 0.5))
        b = a * tilt

    return _normalize(b)


def _gradient_from_weights(weights, a, size2):
    """
    Gradient of sum_k weights[k] * (a*a)[k].
    For j = 0..n-1:
        grad[j] = 2 * sum_l a[l] * weights[j + l]
    This is obtained as 2 * convolve(weights, reverse(a))[n-1 : 2n-1].
    """
    n = int(a.size)
    w = np.nan_to_num(np.asarray(weights, dtype=np.float64), nan=0.0, posinf=0.0, neginf=0.0)
    if w.size != 2 * n - 1:
        return np.zeros(n, dtype=np.float64)
    w = np.maximum(w, 0.0)
    if w.size == 0:
        return np.zeros(n, dtype=np.float64)
    fw = np.fft.rfft(w, size2)
    fa = np.fft.rfft(a[::-1].copy(), size2)
    conv = np.fft.irfft(fw * fa, size2)
    grad = 2.0 * conv[n - 1:2 * n - 1]
    grad = np.where(grad > 0.0, grad, 0.0)
    return np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)


def _gradient_top_conv(a, b, K):
    """
    Exact active-set gradient for the top K convolution coefficients.
    For b[k] = sum_i a[i] a[k-i], d b[k] / d a[j] = 2 a[k-j] if valid.
    """
    n = int(a.size)
    if n == 0 or b.size == 0:
        return np.zeros(n, dtype=np.float64)
    K = int(min(max(1, K), b.size))
    part = np.argpartition(b, -K)[-K:]
    vals = b[part]
    order = np.argsort(vals)[::-1]
    part = part[order]
    vals = vals[order]

    vmax = float(vals[0])
    if vmax <= 0.0:
        return np.zeros(n, dtype=np.float64)

    temp = max(1e-30, 0.05 * vmax)
    weights = np.exp((vals - vmax) / temp)
    ws = float(np.sum(weights))
    if ws <= 0.0 or not np.isfinite(ws):
        weights = np.ones(K, dtype=np.float64) / float(K)
    else:
        weights = weights / ws

    grad = np.zeros(n, dtype=np.float64)
    for k_idx, wt in zip(part, weights):
        k = int(k_idx)
        js = max(0, k - n + 1)
        je = min(n - 1, k)
        if js > je or wt <= 0.0:
            continue
        jrange = np.arange(js, je + 1)
        idx = k - jrange
        grad[js:je + 1] += 2.0 * wt * a[idx]
    return grad


def _upsample(a_small, n_target):
    a_small = _normalize(a_small)
    ns = int(a_small.size)
    nt = int(max(1, n_target))
    if ns == nt:
        return a_small.copy()
    xs = (np.arange(ns, dtype=np.float64) + 0.5) / float(ns)
    xt = (np.arange(nt, dtype=np.float64) + 0.5) / float(nt)
    dens_small = a_small * float(ns)
    dens = np.interp(xt, xs, dens_small, left=0.0, right=0.0)
    return _normalize(dens)


def _optimize_full(a0, n, deadline, rng, max_official_evals=80, official_interval=60, label=""):
    a = _normalize(a0)
    if a.size != n:
        a = _upsample(a, n)

    b = _convolve_fast(a)
    M = float(b.max()) if b.size else np.inf
    score = 0.0 if np.isfinite(M) else np.inf
    if a.size:
        s = float(np.sum(a))
        score = float((2.0 * n * M) / (s * s)) if s > 0 else np.inf

    best = a.copy()
    best_M = M
    best_score = score
    best_actual = _call_evaluator([float(x) for x in best])
    best_actual_seq = best.copy()

    if n <= 1:
        final_act = _call_evaluator([float(x) for x in best])
        if np.isfinite(final_act) and final_act < best_actual:
            best_actual = final_act
            best_actual_seq = best
        return best_actual_seq, best_actual

    size2 = _fft_len(3 * n - 2)
    start = time.monotonic()
    total = max(1.0, float(deadline - start))

    it = 0
    stuck = 0
    patience = 22
    evals = 0

    while time.monotonic() < deadline:
        elapsed = max(1e-9, time.monotonic() - start)
        progress = min(1.0, elapsed / total)

        if not np.isfinite(M) or M <= 0.0:
            a = _normalize(a + 1e-6 * rng.standard_normal(n))
            b = _convolve_fast(a)
            M = float(b.max())
            continue

        if progress < 0.62:
            sp = progress / 0.62
            p = 2.0 + 14.0 * (sp ** 1.4)
            step = 0.55 * (1.0 - 0.78 * sp) + 0.03
            u = b / M
            u = np.maximum(u, 0.0)
            weights = np.power(u, p - 1.0)
            wsum = float(np.sum(weights))
            if wsum <= 0.0 or not np.isfinite(wsum):
                weights = np.ones_like(b) / float(b.size)
            grad = _gradient_from_weights(weights, a, size2)
        else:
            tp = (progress - 0.62) / 0.38
            K = 32 + int(96 * tp)
            step = 0.16 * (1.0 - 0.85 * tp) + 0.015
            grad = _gradient_top_conv(a, b, K)

        improved = False
        cur_step = step
        for _attempt in range(3):
            trial = _mirror_step(a, grad, cur_step)
            tb = _convolve_fast(trial)
            tM = float(tb.max()) if tb.size else np.inf
            if np.isfinite(tM) and tM < M - 1e-15:
                a = trial
                b = tb
                M = tM
                improved = True
                break
            cur_step *= 0.5

        if improved:
            if M < best_M - 1e-12:
                best_M = M
                best = a.copy()
                stuck = 0
            else:
                stuck += 1
        else:
            stuck += 1
            if stuck >= patience:
                mode = int(rng.integers(0, 6))
                a = _perturb_seq(a, rng, mode)
                b = _convolve_fast(a)
                M = float(b.max())
                if not np.isfinite(M) or M <= 0.0:
                    a = _normalize(rng.exponential(1.0, size=n))
                    b = _convolve_fast(a)
                    M = float(b.max())
                stuck = 0
                if M < best_M:
                    best_M = M
                    best = a.copy()

        it += 1

        if evals < max_official_evals and (it % max(1, official_interval) == 0 or M < best_M):
            act = _call_evaluator([float(x) for x in a])
            evals += 1
            if np.isfinite(act) and act < best_actual:
                best_actual = act
                best_actual_seq = a.copy()

        if it % 5000 == 0:
            cur_score = _score_fast(a)
            print(f"    {label} it={it} score={cur_score:.8f} best={best_actual:.8f}")

    candidates = (best, a, best_actual_seq)
    for cand in candidates:
        if cand is None:
            continue
        act = _call_evaluator([float(x) for x in cand])
        if np.isfinite(act) and act < best_actual:
            best_actual = act
            best_actual_seq = cand.copy()

    return best_actual_seq, best_actual


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s) - 10.0)

    base_arr = _get_base_sequence()
    if base_arr is not None and base_arr.size > 0:
        raw_n = int(base_arr.size)
    else:
        raw_n = 6121

    max_n = 8192
    if raw_n > max_n:
        raw_n = max_n
    if raw_n < 16:
        raw_n = 64
    n = int(raw_n)

    if base_arr is not None:
        arr = base_arr[:n].astype(np.float64).copy()
        if arr.size < n:
            tmp = np.zeros(n, dtype=np.float64)
            tmp[:arr.size] = arr
            arr = tmp
        a_base = _normalize(arr)
    else:
        a_base = np.ones(n, dtype=np.float64) / float(n)

    # Quick measured evaluator cost (one call) and initial best.
    t_eval = time.monotonic()
    base_actual = _call_evaluator([float(x) for x in a_base])
    eval_time = max(1e-4, time.monotonic() - t_eval)

    best_seq = a_base.copy()
    best_actual = base_actual
    if not np.isfinite(best_actual):
        best_actual = _score_fast(a_base)

    print(f"n={n}, base score={best_actual:.8f}, eval_time~{eval_time:.4f}s")

    # Small-length warm starts.
    small_roots = []
    small_total_budget = min(280.0, max(0.0, deadline - time.monotonic() - 350.0) * 0.14)
    if small_total_budget > 80.0:
        for ns in (96, 192, 384):
            if time.monotonic() >= deadline - 350.0:
                break
            alloc = min(70.0, small_total_budget * 0.4)
            if alloc <= 5.0:
                break
            t0 = time.monotonic()
            small_start = _normalize(rng.exponential(1.0, size=ns))
            if base_arr is not None and base_arr.size >= ns:
                small_start_b = _normalize(base_arr[:ns])
                small_start = 0.5 * small_start + 0.5 * small_start_b
                small_start = _normalize(small_start)
            small_end = t0 + alloc
            small_opt, _ = _optimize_full(
                small_start,
                ns,
                small_end,
                rng,
                max_official_evals=max(10, int(10.0 / max(eval_time, 1e-4))),
                official_interval=12,
                label=f"small{ns}",
            )
            if small_opt is not None and small_opt.size == ns:
                small_roots.append(small_opt.copy())
                print(f"  small {ns}: fast={_score_fast(small_opt):.8f}")

    # Build diverse large starts.
    t = np.arange(n, dtype=np.float64) / float(n)
    start_seqs = []
    start_seqs.append(a_base)
    start_seqs.append(np.ones(n, dtype=np.float64) / float(n))
    start_seqs.append(_normalize(rng.exponential(1.0, size=n)))

    # Sinusoidal / tilting starts.
    for f, amp, phase in (
        (1, 0.55, 0.0),
        (2, 0.45, 0.9),
        (3, 0.40, 1.7),
        (5, 0.32, 2.5),
        (8, 0.25, 0.4),
    ):
        start_seqs.append(1.0 + amp * np.sin(2.0 * np.pi * f * t + phase))

    # Exponential tilt.
    alpha = float(rng.uniform(-1.2, 1.2))
    start_seqs.append(np.exp(alpha * (t - 0.5)))

    # Two Gaussian lumps.
    for c1, c2 in ((0.21, 0.79), (0.30, 0.70)):
        bw = 0.055
        start_seqs.append(
            np.exp(-((t - c1) / bw) ** 2)
            + np.exp(-((t - c2) / bw) ** 2)
            + 0.01
        )

    # Sparse 0/1 start.
    m_sp = int(max(1, min(n, 1.3 * np.sqrt(n))))
    a_sp = np.zeros(n, dtype=np.float64)
    idx_sp = rng.choice(n, size=m_sp, replace=False)
    a_sp[idx_sp] = 1.0
    start_seqs.append(a_sp)

    # Upsampled small solutions.
    for sr in small_roots:
        start_seqs.append(_upsample(sr, n))
        start_seqs.append(_perturb_seq(_upsample(sr, n), rng, 1))

    # Normalize and score starts with the fast objective.
    fast_scores = np.empty(len(start_seqs), dtype=np.float64)
    for i, seq in enumerate(start_seqs):
        seqn = _normalize(seq)
        start_seqs[i] = seqn
        sc = _score_fast(seqn)
        fast_scores[i] = sc if np.isfinite(sc) else np.inf

    order = np.argsort(fast_scores)
    top_idx = []
    for idx in order:
        if len(top_idx) >= 8:
            break
        top_idx.append(int(idx))

    # Officially validate the top candidates and reorder by official score when possible.
    official_start_scores = []
    for idx in top_idx:
        sc = _call_evaluator([float(x) for x in start_seqs[idx]])
        official_start_scores.append(sc)

    combined = []
    for i, idx in enumerate(top_idx):
        of = official_start_scores[i]
        ff = fast_scores[idx]
        base_of = of if np.isfinite(of) else 1e100
        base_ff = ff if np.isfinite(ff) else 1e100
        combined.append((base_of, min(base_ff, base_of - 1e-12 if base_of < 1e90 else base_ff), idx))
    compound_scores = [c[0] + 0.5 * c[1] for c in combined]
    order2 = np.argsort(compound_scores)
    run_idx = [combined[int(ii)][2] for ii in order2[:4]]

    # Update global best from validated starts.
    for idx in run_idx:
        seq = start_seqs[idx]
        act = _call_evaluator([float(x) for x in seq])
        if np.isfinite(act) and act < best_actual:
            best_actual = act
            best_seq = seq.copy()
        else:
            sc_fast = _score_fast(seq)
            if np.isfinite(sc_fast) and sc_fast < best_actual:
                best_actual = sc_fast
                best_seq = seq.copy()

    # Run main optimization from the best starts.
    official_interval = max(20, int(2.0 / max(eval_time, 1e-4)))
    max_official_evals = int(max(20, min(220, 18.0 / max(eval_time, 1e-4))))

    large_end = deadline - max(20.0, (deadline - time.monotonic()) * 0.06)
    avail = max(1.0, large_end - time.monotonic() - 20.0)
    weights = [0.46, 0.30, 0.18, 0.06]
    n_runs = min(len(run_idx), 4)
    wsum = 0.0
    for i in range(n_runs):
        wsum += weights[i]
    if wsum <= 0.0:
        n_runs = 0

    for i in range(n_runs):
        if time.monotonic() > large_end:
            break
        slice_time = avail * (weights[i] / wsum)
        run_deadline = min(large_end - 5.0, time.monotonic() + slice_time)
        start_seq = start_seqs[run_idx[i]].copy()
        if i == 0:
            # Slightly jitter the best start to avoid exact stationary symmetry.
            start_seq = _normalize(start_seq + 1e-5 * rng.standard_normal(n))
        else:
            # Keep later runs a bit more exploratory.
            start_seq = _perturb_seq(start_seq, rng, i % 5)

        print(f"  large run {i}: start fast={_score_fast(start_seq):.8f}, slice={slice_time:.1f}s")
        seq_opt, act_opt = _optimize_full(
            start_seq,
            n,
            run_deadline,
            rng,
            max_official_evals=max(10, max_official_evals // max(1, i + 1)),
            official_interval=official_interval,
            label=f"run{i}",
        )
        if np.isfinite(act_opt) and act_opt < best_actual:
            best_actual = act_opt
            best_seq = seq_opt.copy()
            print(f"  large run {i}: new best {best_actual:.8f}")
        else:
            print(f"  large run {i}: local best {act_opt:.8f}")

    # Final polish using the current best and lightly perturbed copies.
    if time.monotonic() < deadline - 45.0:
        for polish_i in range(3):
            if time.monotonic() > deadline - 15.0:
                break
            a_pert = best_seq.copy()
            if polish_i == 0:
                a_pert = _normalize(a_pert + 2e-5 * rng.standard_normal(n))
            else:
                a_pert = _perturb_seq(a_pert, rng, (polish_i * 2) % 5)
            small_deadline = deadline - max(10.0, 20.0 * polish_i)
            seq_opt, act_opt = _optimize_full(
                a_pert,
                n,
                small_deadline,
                rng,
                max_official_evals=max(5, min(60, int(8.0 / max(eval_time, 1e-4)))),
                official_interval=max(50, official_interval),
                label=f"polish{polish_i}",
            )
            if np.isfinite(act_opt) and act_opt < best_actual:
                best_actual = act_opt
                best_seq = seq_opt.copy()
                print(f"  polish {polish_i}: new best {best_actual:.8f}")
            else:
                print(f"  polish {polish_i}: local best {act_opt:.8f}")

    # Final scale: sum=10 is numerically comfortable and scale-invariant.
    final_seq = np.clip(np.asarray(best_seq, dtype=np.float64).ravel(), 0.0, 1000.0)
    s = float(np.sum(final_seq))
    if s > 1e-12 and np.isfinite(s):
        final_seq = final_seq * (10.0 / s)
    else:
        final_seq = np.ones(n, dtype=np.float64) / float(n)
        s = float(n)
    final_seq = np.clip(final_seq, 0.0, 1000.0)

    final_list = [float(x) for x in final_seq]
    final_act = _call_evaluator(final_list)

    # Also compare sum=1 representation in case of rounding sensitivity.
    sum1_seq = _normalize(final_seq)
    sum1_list = [float(x) for x in sum1_seq]
    sum1_act = _call_evaluator(sum1_list)

    if np.isfinite(final_act) and (not np.isfinite(sum1_act) or final_act <= sum1_act):
        out_list = final_list
        out_act = final_act
    else:
        out_list = sum1_list
        out_act = sum1_act

    if not np.isfinite(out_act):
        # Fallback to a guaranteed valid uniform sequence.
        out_list = [1.0]
        out_act = _call_evaluator(out_list)

    print(f"FINAL official score: {out_act:.8f}")
    print(f"Time used: {time.monotonic() - t_start:.2f}s")
    return out_list
```
