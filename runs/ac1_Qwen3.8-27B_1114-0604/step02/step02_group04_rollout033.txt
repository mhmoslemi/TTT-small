def _grad(a, w):

    n = a.size

    fw = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)

    a = np.ones(n) / n

        ph_m = phase + m * 0.3
        theta = 2.0 * np.pi * (i + ph_m) * (m + 1) / n
        a += amp_m * np.sin(theta) / n
    a = np.clip(a, 0.0, 1000.0)

    if s > 0.01:

"""
Low-peak step-function optimization via adaptive-temperature mirror descent
with coupled learning rate, multi-length search, and hard-active-set polishing.

Key improvements over previous approach:
- Temperature grows as optimization proceeds (smooth -> sharp focus)
- Learning rate coupled inversely to temperature (stable steps)
- Periodic hard top-3 active set steps interleaved with soft steps
- Explores n=1024 and n=2048 for more degrees of freedom
- Anti-palindromic multi-frequency initialization
- Final ultra-precise polish phase with K=2,3
"""

import numpy as np
import time


def _np2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _conv(a):
    """Full linear auto-convolution via FFT."""
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _np2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    return np.fft.irfft(fa * fa, N)[:2 * n - 1]


def _sc(a, c):
    """Score from precomputed convolution."""
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _sc_full(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    c = _conv(a)
    return _sc(a, c)


def _grad(a, w):
    """Gradient of sum_k w[k]*c[k] w.r.t. a, using FFT cross-correlation."""
    n = a.size
    N = _np2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w[:N], N) if w.size > N else np.fft.rfft(w, N)
    return 2.0 * np.fft.irfft(np.conj(fa) * fw, N)[:n]


def _make_anti_sine(n, amp1, ph1, amp2=0.0, ph2=0.0, amp3=0.0, ph3=0.0):
    """Anti-palindromic multi-frequency sinusoidal initialization."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64) / n
    # Mode 1 (fundamental, anti-palindromic with phase 0.5)
    a += (amp1 * np.sin(2.0 * np.pi * (i + ph1) / n)) / n
    # Mode 2 (optional)
    if amp2 > 0:
        a += (amp2 * np.sin(2.0 * np.pi * (i + ph2) / n)) / n
    # Mode 3 (optional, higher freq for texture)
    if amp3 > 0:
        a += (amp3 * np.sin(4.0 * np.pi * (i + ph3) / n)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a

                       hard_interval=200, hard_K=3):

    """
    a = np.asarray(a0, dtype=np.float64).copy()

    best_a = a.copy()

    it = 0
    last_imp = 0
    patience = 80000
    eval_every = 5

    while time.monotonic() < deadline:
        it += 1
        T = T0 * (1.0 + it / tau)

        c = _conv(a)
        if c.size == 0:
            break

        if it % hard_interval == 0:
            # Hard top-K step

            top_idx = np.argpartition(c, -K)[-K:]
            w = np.zeros(c.size, dtype=np.float64)

                    if nm == 0:

                    w = ec / es

        g = _grad(a, w)
        gp = g - g.mean()

    # Also try a second run on n=1024 with different params
    if time.monotonic() < hard_deadline - 250:
        t_end_1024b = min(hard_deadline - 250.0, t_end_1024 + 200)
        rem_b = t_end_1024b - time.monotonic()
        if rem_b > 60:
            print(f"[Phase 2c] Second n=1024 run ({rem_b:.0f}s, different T)...", flush=True)
            a_1024b, sc_1024b = _optimize_adaptive(init_1024, t_end_1024b, seed=3,
                                                   T0=15.0, tau=20000, lr_coeff=0.4,
                                                   hard_interval=100, hard_K=4)
            print(f"  n=1024b optimized: {sc_1024b:.8f}", flush=True)
            if sc_1024b < best_final_sc:
                best_final_sc = sc_1024b
                best_overall = a_1024b.copy()

    # --- Phase 3: Hard polish ---
    if best_overall is not None and time.monotonic() < hard_deadline - 15:
        polish_deadline = hard_deadline - 8.0
        rem_pol = polish_deadline - time.monotonic()
        if rem_pol > 30:
            print(f"[Phase 3] Hard polish ({rem_pol:.0f}s, K=2)...", flush=True)
            a_pol, sc_pol = _polish_hard(best_overall, polish_deadline, K=2,
                                         lr_coeff=0.04, seed=555)
            print(f"  Polish K=2: {sc_pol:.8f}", flush=True)
            if sc_pol < best_final_sc:
                best_final_sc = sc_pol
                best_overall = a_pol.copy()

    # Second polish with K=3 if time allows
    if best_overall is not None and time.monotonic() < hard_deadline - 8:
        rem2 = hard_deadline - 5.0 - time.monotonic()
        if rem2 > 20:
            a_pol2, sc_pol2 = _polish_hard(best_overall, hard_deadline - 5.0, K=3,
                                           lr_coeff=0.03, seed=666)
            print(f"  Polish K=3: {sc_pol2:.8f}", flush=True)
            if sc_pol2 < best_final_sc:
                best_final_sc = sc_pol2
                best_overall = a_pol2.copy()

    # --- Phase 4: Validate ---
    print("[Phase 4] Validating...", flush=True)
    if best_overall is None:
        best_overall = init_1024

    result = [float(max(0.0, min(1000.0, x))) for x in best_overall]
    try:
        actual = float(evaluate_sequence(result))
    except Exception:
        actual = best_final_sc

    print(f"[Done] n={len(result)}, internal={best_final_sc:.8f}, actual={actual:.8f}, "
          f"time={time.monotonic()-t0:.1f}s", flush=True)
    return result
