    return g, c_max

"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2.
Uses exact subgradient methods with multiplicative weight updates,
frequency-domain spectral shaping, and Frank-Wolfe diversification.
Starts from the parent sequence and refines using adaptive step sizes.
"""

import numpy as np
import time


def _next_pow2(x):
    L = 1
    x = max(1, int(x))
    while L < x:
        L <<= 1
    return L


def _conv_fft(a, L=None):
    """Linear convolution of a with itself using FFT."""
    n = len(a)
    if L is None:
        L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    return np.fft.irfft(fa * fa, L)[:2 * n - 1]


def _score_norm(a):
    """Score assuming sum(a) = 1: 2*n*max(conv(a,a))."""
    n = len(a)
    c = _conv_fft(a)
    return float(2.0 * n * c.max())


def _score_full(a):
    """Full score: 2*n*max(conv(a,a))/sum(a)^2."""
    s = float(a.sum())
    if s < 1e-15:
        return float('inf')
    return float(2.0 * len(a) * _conv_fft(a).max() / (s * s))


def _subgrad_at_k(a, k):
    """Subgradient of c[k] = conv(a,a)[k] w.r.t. a."""
    n = len(a)
    g = np.zeros(n)
    s = max(0, k - n + 1)
    e = min(n - 1, k)
    if s <= e:
        g[s:e + 1] = 2.0 * a[k - np.arange(s, e + 1)]
    return g


def _smooth_subgrad(a, L, top_k=3, sharpness=50.0):
    """Smoothed subgradient using top-k maximizing lags."""
    n = len(a)
    c = _conv_fft(a, L)
    cmax = c.max()
    if cmax < 1e-15:
        return np.zeros(n)
    # Get top-k indices
    if top_k >= len(c):
        k_vals = np.arange(len(c))
        c_vals = c
    else:
        k_vals = np.argsort(c)[::-1][:top_k]
        c_vals = c[k_vals]
    # Exponential weights
    w = np.exp(sharpness * (c_vals - cmax))
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(k_vals))
        w[0] = 1.0
    else:
        w = w / w_sum
    # Weighted sum of subgradients
    g = np.zeros(n)
    for i, k in enumerate(k_vals):
        g += w[i] * _subgrad_at_k(a, int(k))
    return g


def _mul_step(a, g, eta):
    """Multiplicative weight update with non-negativity projection."""
    gm = g.max()
    if gm < 1e-15:
        return a.copy()
    a_new = a * np.exp(-eta * g / gm)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _fw_step(a, g, t):
    """Frank-Wolfe step on the simplex."""
    i = int(np.argmin(g))
    gamma = 2.0 / (t + 2.0)
    a_new = a * (1.0 - gamma)
    a_new[i] += gamma
    return a_new


def _freq_shape(a, L, power=0.7):
    """
    Frequency-domain spectral shaping: modify the convolution spectrum
    to be more concentrated at DC, then recover a non-negative sequence.
    """
    n = len(a)
    A = np.fft.rfft(a, L)
    F2 = A * A
    c = np.fft.irfft(F2, L)[:2 * n - 1]
    cmax = c.max()
    if cmax < 1e-15:
        return None
    # Raise to power > 1 to concentrate, or < 1 to flatten
    # We want to flatten c, so raise to power < 1
    c_new = cmax * np.power(np.maximum(c, 0.0) / cmax, power)
    F2_new = np.fft.rfft(np.concatenate([c_new, np.zeros(L - len(c_new))]), L)
    # Square root in frequency domain
    A_new = np.sqrt(F2_new.astype(complex) + 0j)
    a_new = np.fft.irfft(A_new, L)[:n]
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return None
    return a_new / s


def _perturb(a, rng, scale=0.05):
    """Random multiplicative perturbation."""
    a_new = a * np.exp(rng.normal(0, scale, len(a)))
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _block_scale(a, rng, n_blocks=8):
    """Random block scaling perturbation."""
    n = len(a)
    a_new = a.copy()
    bs = n // n_blocks
    for i in range(n_blocks):
        s = i * bs
        e = s + bs if i < n_blocks - 1 else n
        f = float(rng.uniform(0.5, 1.5))
        a_new[s:e] *= f
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def _optimize_subgrad(a0, time_budget, L=None, verbose=False, top_k=3, sharpness=50.0):
    """
    Main optimizer: subgradient descent with multiplicative updates.
    """
    a = a0.copy()
    s = a.sum()
    a = a / s
    n = len(a)
    if L is None:
        L = _next_pow2(2 * n - 1)

    best = a.copy()
    best_sc = _score_norm(a)
    cur_sc = best_sc

    eta = 0.2
    deadline = time.monotonic() + time_budget
    stall = 0
    it = 0
    fw_counter = 0

    while time.monotonic() < deadline:
        # Compute smoothed subgradient
        g = _smooth_subgrad(a, L, top_k=top_k, sharpness=sharpness)
        
        if g.max() < 1e-15:
            stall += 1
            if stall > 100:
                break
            continue

        # Multiplicative step
        a_new = _mul_step(a, g, eta)
        sc = _score_norm(a_new)

        if sc < cur_sc - 1e-14:
            a = a_new
            cur_sc = sc
            stall = 0
            if cur_sc < best_sc - 1e-12:
                best = a.copy()
                best_sc = cur_sc
                if verbose and it % 2000 == 0:
                    print(f"      it={it}, sc={best_sc:.6f}, eta={eta:.4f}, sharp={sharpness:.0f}")
        else:
            stall += 1
            if stall % 30 == 0:
                eta = max(eta * 0.85, 0.01)
            if stall > 150:
                # Try diversification
                if fw_counter < 3:
                    # FW step
                    a_new = _fw_step(a, g, stall)
                    sc_new = _score_norm(a_new)
                    if sc_new < cur_sc:
                        a = a_new
                        cur_sc = sc_new
                        stall = 0
                    else:
                        fw_counter += 1
                else:
                    # Reset to best and perturb
                    a = _perturb(best, np.random.default_rng(it), 0.03)
                    cur_sc = _score_norm(a)
                    eta = 0.2
                    fw_counter = 0
                    stall = 0
                continue

        # Adapt eta (increase when making progress)
        if stall == 0 and it % 1000 == 0:
            eta = min(eta * 1.08, 3.0)
            # Also try increasing sharpness
            if sharpness < 500:
                sharpness = min(sharpness * 1.1, 500.0)

        it += 1

    return best, best_sc
