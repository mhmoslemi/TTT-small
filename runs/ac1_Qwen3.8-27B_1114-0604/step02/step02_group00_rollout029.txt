"""
Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Strategy: anti-palindromic structured initializations + LSE-continuation L-BFGS +
basis-reduced global exploration + exact subgradient polish + targeted local search.
Key insight: anti-palindromic perturbations of uniform reduce the central autocorrelation peak.

import numpy as np
import time
from scipy.optimize import minimize


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed max-conv score and gradient w.r.t. a."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_s = np.exp(scaled)
    L = c_max + np.log(exp_s.sum()) / beta
    p = exp_s / exp_s.sum()
    score = 2.0 * n * L / (S * S)
    # Gradient via cross-correlation
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective_wrapper(a, beta):
    """Scipy-compatible objective returning (value, gradient)."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, time_limit=None):
    """Run L-BFGS-B on LSE-smoothed objective. Returns (score, array, success)."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    t0 = time.monotonic()
    try:
        res = minimize(
            lse_objective_wrapper, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-16, 'gtol': 1e-14,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    except Exception:
        a_out = a0
        exact_sc = compute_score(a_out)
        return exact_sc, a_out, False
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res.success

def subgrad_polish(a, max_iters=3000):
    """Exact subgradient descent with Polyak step size."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 2.0 * n / (2.0 * n - 1.0)  # theoretical lower bound

    for it in range(max_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]

        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn2 = float(grad @ grad)
        if gn2 < 1e-30:
            break
        step = (sc - sc_lb) / gn2
        step = min(step, 0.5 / (1.0 + 0.005 * it))
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a


def targeted_perturb(a, rng, strength=0.05):
    """Reduce elements contributing most to the peak convolution lag."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    a_pert = a.copy()
    if j_end >= j_start:
        # Contributors: a[j] pairs with a[k_star - j]
        contrib = np.zeros(n)
        for j in range(j_start, j_end + 1):
            contrib[j] = a[j] * a[k_star - j]
        top_count = max(3, n // 100)
        top_idx = np.argsort(contrib)[::-1][:top_count]
        bottom_idx = np.argsort(contrib)[:-top_count]
        total_red = 0.0
        for j in top_idx:
            amt = a_pert[j] * strength
            a_pert[j] -= amt
            a_pert[j] = max(a_pert[j], 0.0)
            total_red += amt
        if len(bottom_idx) > 0 and total_red > 0:
            per = total_red / len(bottom_idx)
            a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)

                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for n_try in [4096, 6962, 8192, 12288]:
        if n_try not in lengths:
            lengths.append(n_try)
    lengths = sorted(set(lengths))
    lengths = lengths[:4]

    print(f"Lengths to try: {lengths}")
    print(f"Parent: {'yes (n=' + str(parent.size) + ')' if parent is not None else 'no'}")

    # ============================================================
    # Phase 1: Basis-reduced global search (fast)
    # ============================================================
    phase1_end = t_start + 0.15 * total_budget
    print(f"\n=== Phase 1: Basis Global Search (until t={phase1_end - t_start:.0f}s) ===")

    for n in lengths:
        if time.monotonic() > phase1_end:
            break
        print(f"  Basis search: n={n}")
        sc, a_basis = basis_global_search(n, rng, time_budget=30)
        report(sc, a_basis, f"basis n={n}")

    # ============================================================
    # Phase 2: Full LSE-continuation L-BFGS
    # ============================================================
    phase2_end = t_start + 0.60 * total_budget
    print(f"\n=== Phase 2: LSE Continuation (until t={phase2_end - t_start:.0f}s) ===")

    beta_schedule = [30, 100, 300, 800, 2000, 5000]

    for n in lengths:
        if time.monotonic() > phase2_end:
            break
        print(f"\n  n={n}")
        inits = build_inits(n, rng, parent)
        # Sort: put best candidates first (parent, ramp, etc.)
        # Limit inits based on time
        n_inits_avail = len(inits)
        time_left = phase2_end - time.monotonic()
        n_inits_use = min(n_inits_avail, max(3, int(time_left / 40)))

        for i in range(min(n_inits_use, 10)):
            if time.monotonic() > phase2_end:
                break
            t0 = time.monotonic()
            a_cur = inits[i].copy()
            for beta in beta_schedule:
                if time.monotonic() > phase2_end:
                    break
                # Scale iterations based on remaining time
                remaining = phase2_end - time.monotonic()
                est_time_per_call = 5.0 + 0.001 * n
                maxit = max(500, min(5000, int(remaining / est_time_per_call * 2)))
                sc, a_opt, _ = run_lbfgs(a_cur, beta, maxiter=maxit, maxfun=maxit * 20)
                a_cur = a_opt
                report(sc, a_opt, f"P2 n={n} i={i} b={beta}")

    print(f"\n  After Phase 2: best = {best_score:.8f}")

    # ============================================================
    # Phase 3: High-beta refinement on best solution
    # ============================================================
    phase3_end = t_start + 0.78 * total_budget
    print(f"\n=== Phase 3: High-Beta Refinement ===")

    n_best = len(best_a)
    for beta in [3000, 8000, 20000]:
        if time.monotonic() > phase3_end:
            break
        remaining = phase3_end - time.monotonic()
        maxit = max(1000, min(10000, int(remaining / 3)))
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=maxit, maxfun=maxit * 20)
        report(sc, a_opt, f"P3 beta={beta}")

    # ============================================================
    # Phase 4: Subgradient polish
    # ============================================================
    phase4_end = t_start + 0.88 * total_budget
    print(f"\n=== Phase 4: Subgradient Polish ===")

    if time.monotonic() < phase4_end:
        n_iters = max(500, int((phase4_end - time.monotonic()) * 100))
        sc, a_opt = subgrad_polish(best_a, max_iters=n_iters)
        report(sc, a_opt, "subgrad")

    # ============================================================
    # Phase 5: Local search
    # ============================================================
    print(f"\n=== Phase 5: Local Search ===")
    n_best = len(best_a)
    trial = 0
    while time.monotonic() < deadline - 10 and trial < 30:
        trial += 1
        strat = trial % 4
        if strat == 0:
            # Gaussian perturbation
            scale = 0.03 + 0.05 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
        elif strat == 1:
            # Targeted perturbation
            a_pert = targeted_perturb(best_a, rng, strength=0.03 + 0.04 * rng.random())
        elif strat == 2:
            # Anti-palindromic bump
            a_pert = best_a.copy()
            center = n_best // 2
            width = max(1, n_best // 15)
            for j in range(n_best):
                d = (j - center) / width
                sign = 1.0 if j < center else -1.0
                a_pert[j] *= (1.0 + 0.08 * sign * np.exp(-d * d))
        else:
            # Frequency-domain nudge: multiply DFT by flat-top window correction
            a_pert = best_a.copy()
            M_fft = 1 << (2 * n_best - 1 - 1).bit_length()
            A = np.fft.rfft(a_pert, M_fft)
            # Attenuate high frequencies slightly (smoothing)
            freqs = np.arange(len(A))
            att = 1.0 - 0.1 * (freqs / len(A)) ** 2
            A *= att
            a_pert = np.real(np.fft.irfft(A, M_fft))[:n_best]

        a_pert = np.clip(a_pert, 0.0, 1000.0)
        if a_pert.sum() < 1e-6:
            a_pert = best_a * 0.9

        sc, a_opt, _ = run_lbfgs(a_pert, 3000, maxiter=3000, maxfun=60000)
        report(sc, a_opt, f"P5 #{trial} s={strat}")

    # ============================================================
    # Final subgradient polish
    # ============================================================
    if time.monotonic() < deadline - 5:
        sc, a_opt = subgrad_polish(best_a, max_iters=2000)
        report(sc, a_opt, "final subgrad")

    # ============================================================
    # Output
    # ============================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    print(f"\n{'=' * 60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {time.monotonic() - t_start:.1f}s / {budget_s}s")
    print(f"{'=' * 60}")
    return out
