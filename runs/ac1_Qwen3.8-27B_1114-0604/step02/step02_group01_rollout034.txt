"""
Step function optimizer using log-sum-exp smoothing, LP direction finding,
and coordinate descent. Minimizes 2*n*max(conv(a,a))/(sum(a))^2.
"""

import numpy as np
import time
import scipy.sparse as sp
import cvxpy as cp


def _autocorr_fft(f):
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _score_fast(f):
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))


def _grad_logsumexp(f, tau):
    """Gradient of (1/tau)*log(sum exp(tau*(f*f)[k])) w.r.t. f."""
    n = len(f)
    c = _autocorr_fft(f)
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    p = exp_c / exp_c.sum()
    
    f_rev = f[::-1]
    # conv of p (len 2n-1) and f_rev (len n) -> len 3n-2
    size = 1 << (3 * n - 2).bit_length()
    P = np.fft.rfft(p, size)
    FR = np.fft.rfft(f_rev, size)
    conv = np.fft.irfft(P * FR, size)
    # grad[j] = 2 * conv[n-1+j]
    grad = 2.0 * conv[n - 1:2 * n - 1]
    return grad, c


def _project_simplex(f):
    """Project onto {x >= 0, sum(x) = 1}."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return np.ones(len(f)) / len(f)
    return f / s


def _optimize_logsumexp(f0, deadline, tau_start=50, tau_end=5000, max_iters=100000, verbose=False):
    """Optimize using log-sum-exp smoothed objective with annealed tau."""
    n = len(f0)
    f = np.maximum(f0, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f = f / s
    
    best_score = _score_fast(f)
    best_f = f.copy()
    
    # Momentum
    momentum = np.zeros(n)
    beta = 0.9
    
    stuck = 0
    patience = 20
    last_best = best_score
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        progress = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.7)
        
        grad, c = _grad_logsumexp(f, tau)
        
        # Tangent space projection
        grad_tan = grad - grad.mean()
        
        # Momentum
        momentum = beta * momentum + grad_tan
        grad_eff = momentum
        
        # Adaptive step size
        g_norm = np.linalg.norm(grad_eff)
        if g_norm < 1e-15:
            break
        lr = 0.5 / (tau * g_norm + 1e-10)
        lr = min(lr, 0.5)
        
        # Update
        f_new = f - lr * grad_eff
        f_new = _project_simplex(f_new)
        
        # Evaluate periodically
        if it % 50 == 0:
            score = _score_fast(f_new)
            if score < best_score - 1e-9:
                best_score = score
                best_f = f_new.copy()
                stuck = 0
                if verbose and it % 2000 == 0:
                    print(f"    LSE iter {it}: score={score:.7f}, tau={tau:.1f}")
            else:
                stuck += 1
            
            # If stuck, restart momentum and try larger step
            if stuck >= patience:
                momentum *= 0.3
                stuck = 0
        
        f = f_new
    
    # Final check
    score = _score_fast(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score


def _solve_lp_direction(f, M):
    """
    Solve LP: max sum(b) s.t. b >= 0, (f*b)[k] <= M for all k.
    Returns optimal b (or None if infeasible/too slow).
    """
    n = len(f)
    
    # Build sparse constraint matrix A (shape: (2n-1) x n)
    # A[k, j] = f[j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    data = []
    
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            data.extend(f[js].tolist())
    
    A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()
    
    # LP
    b_var = cp.Variable(n)
    objective = cp.Maximize(cp.sum(b_var))
    constraints = [b_var >= 0, A @ b_var <= M]
    prob = cp.Problem(objective, constraints)
    
    try:
        # Try solvers in order of preference
        for solver in [cp.MOSEK, cp.GUROBI, cp.SCS, cp.CLARABEL, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    b_val = b_var.value
                    if b_val is not None:
                        b_val = np.maximum(b_val, 0)
                        if b_val.sum() > 1e-10:
                            return b_val
                        return None
            except Exception:
                continue
        
        # Fallback to default
        prob.solve()
        if prob.status in ["optimal", "optimal_inaccurate"]:
            b_val = b_var.value
            if b_val is not None:
                b_val = np.maximum(b_val, 0)
                if b_val.sum() > 1e-10:
                    return b_val
    except Exception as e:
        print(f"    LP error: {e}")
    
    return None


def _lp_refine(f0, deadline, max_iters=20, verbose=True):
    """Refine using LP-based direction finding."""
    n = len(f0)
    f = np.maximum(f0, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f = f / s
    
    best_score = _score_fast(f)
    best_f = f.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _autocorr_fft(f)
        M = c.max()
        
        if verbose:
            print(f"    LP iter {it}: M={M:.8f}, score={best_score:.7f}")
        
        b = _solve_lp_direction(f, M)
        if b is None:
            if verbose:
                print("    LP failed, stopping.")
            break
        
        # Normalize b to have same sum as f
        s_b = b.sum()
        if s_b < 1e-10:
            break
        g = b * (s / s_b)  # g has same sum as f
        
        # Line search
        best_t = 0
        best_sc = best_score
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:

            f_try = np.maximum(f_try, 0)

            # Perturb and try again

def _subgradient_refine(f0, deadline, max_iters=50000, verbose=False):
    """Subgradient method on the exact max of autocorrelation."""
    n = len(f0)
    f = np.maximum(f0, 0.0).astype(np.float64).copy()
    s = f.sum()
    f = f / s
    
    best_score = _score_fast(f)

    beta = 0.8
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _autocorr_fft(f)
        peak_idx = np.argmax(c)
        
        # Subgradient of (f*f)[peak_idx] w.r.t. f
        # grad[j] = 2 * f[peak_idx - j] (for valid j)
        j_start = max(0, peak_idx - n + 1)
        j_end = min(n - 1, peak_idx)
        grad = np.zeros(n)
        if j_start <= j_end:
            idx = peak_idx - np.arange(j_start, j_end + 1)
            grad[j_start:j_end + 1] = 2.0 * f[idx]
        
        # Tangent space
        grad_tan = grad - grad.mean()
        
        # Momentum
        momentum = beta * momentum + grad_tan
        
        # Step size
        lr = 1.0 / (np.linalg.norm(momentum) + 1e-10)
        lr = min(lr, 0.1)
        
        f_new = f - lr * momentum
        f_new = _project_simplex(f_new)
        
        if it % 200 == 0:
            score = _score_fast(f_new)
            if score < best_score:
                best_score = score
                best_f = f_new.copy()
                if verbose and it % 5000 == 0:
                    print(f"    SG iter {it}: score={score:.7f}")

    score = _score_fast(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score


def _coordinate_polish(f0, deadline, rng, max_passes=200):
    """Coordinate descent: transfer mass from peak-contributing positions to others."""
    n = len(f0)
    f = np.maximum(f0, 0.0).astype(np.float64).copy()
    s = f.sum()
    f = f / s
    
    best_score = _score_fast(f)
    best_f = f.copy()
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        c = _autocorr_fft(f)
        peak_val = c.max()
        peak_idx = np.argmax(c)
        
        improved = False
        # Try several random coordinate moves
        for _ in range(20):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            
            # Try moving mass from i to j
            for frac in [0.05, 0.1, 0.2, 0.3]:
                amt = f[i] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[i] -= amt
                f_try[j] += amt
                f_try = np.maximum(f_try, 0)
                s_try = f_try.sum()
                if s_try < 1e-10:
                    continue
                f_try = f_try / s_try
                c_try = _autocorr_fft(f_try)
                new_peak = c_try.max()
                if new_peak < peak_val - 1e-12:
                    f = f_try
                    peak_val = new_peak
                    improved = True
                    break
        
        if improved:
            score = _score_fast(f)
            if score < best_score:
                best_score = score
                best_f = f.copy()
    
    return best_f, best_score

"""
Step function optimizer minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses Nesterov-accelerated log-sum-exp smoothing, LP-based direction finding,
subgradient refinement, and coordinate descent polish.
Key innovations over previous version:
- Principled log-sum-exp smoothing with proper gradient via FFT
- LP direction finding (from SOTA description) for exact descent directions
- Nesterov acceleration for faster convergence
- Better starting points (inverted sine, multi-modal structures)
"""

import numpy as np
import time
import scipy.sparse as sp


def _autocorr_fft(f):
    """Full linear autocorrelation a*a via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _score_fast(f):
    """Fast score = 2*n*max(conv(a,a))/(sum(a))^2."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))


def _grad_logsumexp(f, tau):
    """
    Gradient of (1/tau)*log(sum_k exp(tau*(f*f)[k])) w.r.t. f.
    Returns (gradient, autocorrelation).
    """
    n = len(f)
    c = _autocorr_fft(f)
    c_max = c.max()
    exp_c = np.exp(tau * (c - c_max))
    p = exp_c / exp_c.sum()

    f_rev = f[::-1]
    size = 1 << (3 * n - 2).bit_length()
    P = np.fft.rfft(p, size)
    FR = np.fft.rfft(f_rev, size)
    conv = np.fft.irfft(P * FR, size)
    grad = 2.0 * conv[n - 1: 2 * n - 1]
    return grad, c


def _project_simplex(f):
    """Project onto {x >= 0, sum(x) = 1}."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return np.ones(len(f)) / len(f)
    return f / s


def _nesterov_logsumexp(f0, deadline, tau_start=30.0, tau_end=8000.0,
                        max_iters=80000, verbose=False, label=""):
    """
    Nesterov-accelerated optimization of log-sum-exp smoothed objective.
    Minimizes (1/tau)*log(sum_k exp(tau*(f*f)[k])) with annealed tau.
    Subject to f >= 0, sum(f) = 1.
    """
    n = len(f0)
    f = np.maximum(f0, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f = f / s

    best_score = _score_fast(f)
    best_f = f.copy()

    # Nesterov state
    f_nest = f.copy()
    f_prev = f.copy()
    t_nest = 1.0

    momentum = np.zeros(n)
    beta = 0.85

    stuck = 0
    patience = 30
    last_best = best_score
    lr_base = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        progress = it / max_iters
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.6)

        # Nesterov extrapolation
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        theta = (t_nest - 1.0) / t_next
        f_nest = f + theta * (f - f_prev)
        f_nest = _project_simplex(f_nest)

        # Compute gradient at y
        grad, c = _grad_logsumexp(f_nest, tau)

        # Tangent space projection (sum constraint)
        grad_tan = grad - grad.mean()

        # Adaptive step size via backtracking
        g_norm = np.linalg.norm(grad_tan)
        if g_norm < 1e-15:
            f_prev = f.copy()
            f = f_nest
            t_nest = t_next
            continue

        # Step size: inversely proportional to gradient norm and tau
        lr = lr_base / (tau * g_norm + 1e-10)
        lr = min(lr, 0.5)

        # Update
        f_new = f_nest - lr * grad_tan
        f_new = _project_simplex(f_new)

        # Check for NaN
        if not np.all(np.isfinite(f_new)):
            f_new = f.copy()

        f_prev = f.copy()
        f = f_new
        t_nest = t_next

        # Periodic evaluation
        if it % 20 == 0:
            score = _score_fast(f)
            if score < best_score - 1e-10:
                best_score = score
                best_f = f.copy()
                stuck = 0
                if verbose and it % 2000 == 0:
                    print(f"    {label} iter {it}: score={score:.7f}, tau={tau:.0f}")
            else:
                stuck += 1

            if stuck >= patience:
                # Reset Nesterov momentum
                f_nest = f.copy()
                f_prev = f.copy()
                t_nest = 1.0
                stuck = 0
                # Slight random perturbation
                f = f * (1.0 + 0.02 * np.random.default_rng(it).standard_normal(n))
                f = _project_simplex(f)

        # Occasionally reduce lr if we're oscillating
        if it % 500 == 0 and it > 0:
            lr_base *= 0.95

    score = _score_fast(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()

    return best_f, best_score


def _solve_lp(f, M, timeout=120):
    """
    Solve LP: max sum(b) s.t. b >= 0, (f*b)[k] <= M for all k.
    Uses cvxpy. Returns optimal b or None.
    """
    import cvxpy as cp

    n = len(f)

    # Build sparse matrix A (shape (2n-1, n))
    # A[k,j] = f[j] if 0 <= k-j < n
    rows = np.concatenate([
        np.full(min(n, k + 1), k, dtype=np.int32)
        for k in range(n)
    ] + [
        np.full(2 * n - 1 - k, k, dtype=np.int32)
        for k in range(n, 2 * n - 1)
    ])

    cols = np.concatenate([
        np.arange(min(n, k + 1), dtype=np.int32)
        for k in range(n)
    ] + [
        np.arange(k - n + 1, n, dtype=np.int32)
        for k in range(n, 2 * n - 1)
    ])

    data = np.concatenate([
        f[:min(n, k + 1)]
        for k in range(n)
    ] + [
        f[k - n + 1:]
        for k in range(n, 2 * n - 1)
    ])

    A = sp.coo_matrix((data, (rows, cols)), shape=(2 * n - 1, n)).tocsr()

    b_var = cp.Variable(n)
    obj = cp.Maximize(cp.sum(b_var))
    cons = [b_var >= 0, A @ b_var <= M]
    prob = cp.Problem(obj, cons)

    t0 = time.monotonic()
    solvers_to_try = [cp.CLARABEL, cp.SCS, cp.ECOS, cp.GLPK, cp.CBC]
    for solver in solvers_to_try:
        try:
            prob.solve(solver=solver)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                bv = b_var.value
                if bv is not None:
                    bv = np.maximum(np.asarray(bv, dtype=np.float64), 0.0)
                    if bv.sum() > 1e-10:
                        return bv
        except Exception:
            continue
        if time.monotonic() - t0 > timeout:
            break

    return None


def _lp_refine(f0, deadline, max_iters=12, verbose=True):
    """
    LP-based direction finding refinement.
    At each step: solve LP to get direction g, line search, update.
    """
    n = len(f0)
    f = np.maximum(f0, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f = f / s

    best_score = _score_fast(f)
    best_f = f.copy()
    rng = np.random.default_rng(123)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _autocorr_fft(f)
        M = c.max()

        if verbose:
            print(f"    LP iter {it}: M={M:.8f}, current_score={best_score:.7f}")

        b = _solve_lp(f, M, timeout=100)
        if b is None:
            if verbose:
                print("    LP solve failed or timed out.")
            break

        s_b = b.sum()
        if s_b < 1e-10:
            break
        g = b / s_b  # normalize g to sum 1

        # Line search: try multiple step sizes
        best_t = -1
        best_sc = best_score
        for t in [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0]:
            f_try = (1.0 - t) * f + t * g
            f_try = np.maximum(f_try, 0.0)
            s_try = f_try.sum()
            if s_try < 1e-10:
                continue
            f_try = f_try / s_try
            sc = _score_fast(f_try)
            if sc < best_sc:
                best_sc = sc
                best_t = t

        if best_t >= 0:
            f = (1.0 - best_t) * f + best_t * g
            f = _project_simplex(f)
            if best_sc < best_score:
                best_score = best_sc
                best_f = f.copy()
                if verbose:
                    print(f"      t={best_t:.3f}, new best={best_score:.7f}")
        else:
            if verbose:
                print("    No improvement; perturbing.")
            f = f * (1.0 + 0.03 * rng.standard_normal(n))
            f = _project_simplex(f)

    return best_f, best_score


def _subgradient_refine(f0, deadline, max_iters=40000, verbose=False):
    """
    Subgradient method targeting the argmax lag of autocorrelation.
    Uses momentum and adaptive step size.
    """
    n = len(f0)
    f = np.maximum(f0, 0.0).astype(np.float64).copy()
    f = f / f.sum()

    best_score = _score_fast(f)
    best_f = f.copy()

    momentum = np.zeros(n)
    beta = 0.82
    lr = 0.1

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _autocorr_fft(f)
        peak_idx = int(np.argmax(c))

        # Subgradient at peak lag
        j_start = max(0, peak_idx - n + 1)
        j_end = min(n - 1, peak_idx)
        grad = np.zeros(n)
        if j_start <= j_end:
            idx = peak_idx - np.arange(j_start, j_end + 1)
            grad[j_start: j_end + 1] = 2.0 * f[idx]

        # Also add small gradient from 2nd peak for robustness
        c_sorted = np.sort(c)[::-1]
        if len(c_sorted) > 1 and c_sorted[1] > 0.9 * c_sorted[0]:
            # Find second peak (different from first)
            c_masked = c.copy()
            c_masked[peak_idx] = -1
            peak2 = int(np.argmax(c_masked))
            j2_start = max(0, peak2 - n + 1)
            j2_end = min(n - 1, peak2)
            if j2_start <= j2_end:
                idx2 = peak2 - np.arange(j2_start, j2_end + 1)
                grad[j2_start: j2_end + 1] += 0.5 * f[idx2]

        grad_tan = grad - grad.mean()
        momentum = beta * momentum + grad_tan

        g_norm = np.linalg.norm(momentum)
        if g_norm < 1e-15:
            break
        lr = 0.5 / (g_norm + 1e-10)
        lr = min(lr, 0.1)

        f_new = f - lr * momentum
        f_new = _project_simplex(f_new)

        if it % 100 == 0:
            score = _score_fast(f_new)
            if score < best_score - 1e-10:
                best_score = score
                best_f = f_new.copy()
                if verbose and it % 5000 == 0:
                    print(f"    SG iter {it}: {score:.7f}")

        f = f_new

    score = _score_fast(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()

    return best_f, best_score


def _coord_polish(f0, deadline, rng, max_passes=500):
    """
    Coordinate descent: try to reduce peak by transferring mass
    between pairs of coordinates.
    """
    n = len(f0)
    f = np.maximum(f0, 0.0).astype(np.float64).copy()
    f = f / f.sum()

    best_score = _score_fast(f)
    best_f = f.copy()

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        c = _autocorr_fft(f)
        peak_val = c.max()
        peak_idx = int(np.argmax(c))

        # Identify positions contributing most to the peak
        j_start = max(0, peak_idx - n + 1)
        j_end = min(n - 1, peak_idx)
        contrib = np.zeros(n)
        if j_start <= j_end:
            idx = peak_idx - np.arange(j_start, j_end + 1)
            contrib[j_start: j_end + 1] = f[j_start: j_end + 1] * f[idx]

        # Try: move mass from high-contrib to low-contrib positions
        if p % 3 == 0:
            # Structured: top contributor -> bottom contributor
            don = int(np.argmax(contrib))
            rec = int(np.argmin(contrib))
            if don == rec:
                continue
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                amt = f[don] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[don] -= amt
                f_try[rec] += amt
                f_try = np.maximum(f_try, 0)
                s_try = f_try.sum()
                if s_try < 1e-10:
                    continue
                f_try /= s_try
                c_try = _autocorr_fft(f_try)
                if c_try.max() < peak_val - 1e-12:
                    f = f_try
                    peak_val = c_try.max()
                    break
        else:
            # Random moves
            for _ in range(10):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                for frac in [0.05, 0.1, 0.2]:
                    amt = f[i] * frac
                    if amt < 1e-15:
                        continue
                    f_try = f.copy()
                    f_try[i] -= amt
                    f_try[j] += amt
                    f_try = np.maximum(f_try, 0)
                    f_try /= f_try.sum()
                    c_try = _autocorr_fft(f_try)
                    if c_try.max() < peak_val - 1e-12:
                        f = f_try
                        peak_val = c_try.max()
                        break

        score = _score_fast(f)
        if score < best_score:
            best_score = score
            best_f = f.copy()

    return best_f, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the non-negative sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.

    Strategy:
    1. Nesterov-accelerated log-sum-exp smoothing (O(n log n) per iter)
    2. LP-based direction finding for exact descent directions
    3. Subgradient refinement on the exact max
    4. Coordinate descent polish
    5. Final fine-tuning
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 12.0
    t_start = time.monotonic()

    # Load and normalize initial sequence
    f_init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(f_init)
    s0 = f_init.sum()
    if s0 < 1e-10:
        f_init = np.ones(n)
    else:
        f_init = f_init / s0

    best_score = _score_fast(f_init)
    best_f = f_init.copy()
    print(f"Initial: n={n}, score={best_score:.7f}")

    # Generate diverse starting points
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    xn = j / n  # normalized position [0, 1)

    starts = {}
    starts['given'] = f_init.copy()
    starts['uniform'] = np.ones(n)
    starts['u_wide'] = 1.0 + 1.2 * ((j - center) / max(center, 1)) ** 2
    starts['u_narrow'] = 1.0 + 5.0 * ((j - center) / max(center, 1)) ** 2
    starts['u_very'] = 1.0 + 12.0 * ((j - center) / max(center, 1)) ** 2
    starts['bimodal'] = (np.exp(-((xn - 0.25) / 0.12) ** 2) +
                         np.exp(-((xn - 0.75) / 0.12) ** 2) + 0.01)
    starts['bimodal_wide'] = (np.exp(-((xn - 0.2) / 0.2) ** 2) +
                              np.exp(-((xn - 0.8) / 0.2) ** 2) + 0.01)
    starts['triple'] = (np.exp(-((xn - 0.2) / 0.1) ** 2) +
                        np.exp(-((xn - 0.5) / 0.1) ** 2) +
                        np.exp(-((xn - 0.8) / 0.1) ** 2) + 0.01)
    starts['inv_sine'] = 1.0 / (0.3 + np.abs(np.sin(np.pi * xn)))
    starts['inv_sine2'] = 1.0 / (0.1 + np.abs(np.sin(np.pi * xn)))
    starts['sine_mod'] = 1.0 + 0.3 * np.sin(2.0 * np.pi * 7 * xn)
    starts['cosine'] = 1.0 + 0.4 * np.cos(2.0 * np.pi * xn)
    starts['quad'] = 1.0 + 4.0 * (xn - 0.5) ** 2
    starts['exp_lobe'] = np.exp(-4.0 * (xn - 0.5) ** 2) + 0.01
    starts['random1'] = rng.exponential(1.0, size=n)
    starts['dirichlet'] = rng.dirichlet(np.ones(n) * 0.05)
    starts['half_tri'] = np.maximum(0.01, 1.0 - np.abs(2.0 * xn - 1.0))
    starts['rect_inv'] = 1.0 / (1.0 + 5.0 * (np.abs(xn - 0.5) - 0.3) ** 2 + 0.1)

    # Evaluate all starts
    start_results = []
    for name, f0 in starts.items():
        f = np.maximum(f0, 0.0)
        s = f.sum()
        if s < 1e-10:
            f = np.ones(n)
        else:
            f = f / s
        sc = _score_fast(f)
        start_results.append((sc, name, f))

    start_results.sort(key=lambda x: x[0])
    print("\nTop starting points:")
    for sc, name, _ in start_results[:10]:
        print(f"  {name}: {sc:.7f}")

    if start_results[0][0] < best_score:
        best_score = start_results[0][0]
        best_f = start_results[0][2].copy()

    # === Phase 1: Nesterov log-sum-exp from top starts ===
    print(f"\n=== Phase 1: Nesterov LSE (best={best_score:.7f}) ===")
    time_p1 = min(700.0, (deadline - time.monotonic()) * 0.38)
    n_starts = 5
    t_each = time_p1 / n_starts

    for idx in range(n_starts):
        if time.monotonic() > deadline - 400:
            break
        sc, name, f0 = start_results[idx]
        d1 = time.monotonic() + t_each * (1.0 - 0.1 * idx)
        print(f"  LSE '{name}' (init={sc:.6f}, budget={d1 - time.monotonic():.0f}s)")
        f_opt, sc_opt = _nesterov_logsumexp(f0, d1, verbose=True, label=name)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"    -> NEW BEST: {best_score:.7f}")
        else:
            print(f"    -> best from this: {sc_opt:.7f}")

    # LSE from current best
    if time.monotonic() < deadline - 300:
        d1b = time.monotonic() + min(100.0, (deadline - time.monotonic()) * 0.15)
        f_opt, sc_opt = _nesterov_logsumexp(best_f, d1b, verbose=True, label="best")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  LSE from best -> {best_score:.7f}")

    # === Phase 2: LP refinement ===
    print(f"\n=== Phase 2: LP refinement (best={best_score:.7f}) ===")
    lp_budget = min(600.0, (deadline - time.monotonic()) * 0.25)
    if lp_budget > 90:
        d2 = time.monotonic() + lp_budget
        f_opt, sc_opt = _lp_refine(best_f, d2, max_iters=10, verbose=True)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  LP -> NEW BEST: {best_score:.7f}")
        else:
            print(f"  LP best: {sc_opt:.7f}")

    # === Phase 3: Subgradient refinement ===
    print(f"\n=== Phase 3: Subgradient (best={best_score:.7f}) ===")
    sg_budget = min(250.0, max(30.0, (deadline - time.monotonic()) * 0.15))
    if sg_budget > 30:
        d3 = time.monotonic() + sg_budget
        f_opt, sc_opt = _subgradient_refine(best_f, d3, max_iters=30000, verbose=True)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  SG -> NEW BEST: {best_score:.7f}")
        else:
            print(f"  SG best: {sc_opt:.7f}")

    # === Phase 4: Coordinate polish ===
    print(f"\n=== Phase 4: Coordinate polish (best={best_score:.7f}) ===")
    pol_budget = min(200.0, max(30.0, (deadline - time.monotonic()) * 0.2))
    d4 = time.monotonic() + pol_budget
    f_opt, sc_opt = _coord_polish(best_f, d4, rng)
    if sc_opt < best_score:
        best_score = sc_opt
        best_f = f_opt.copy()
        print(f"  Polish -> NEW BEST: {best_score:.7f}")

    # === Phase 5: Final fine-tuning ===
    if time.monotonic() < deadline - 20:
        print(f"\n=== Phase 5: Final LSE (best={best_score:.7f}) ===")
        d5 = deadline - 10.0
        f_opt, sc_opt = _nesterov_logsumexp(
            best_f, d5, tau_start=200.0, tau_end=50000.0,
            max_iters=40000, verbose=True, label="final"
        )
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  Final -> NEW BEST: {best_score:.7f}")

        # One more coordinate polish
        if time.monotonic() < deadline - 15:
            f_opt, sc_opt = _coord_polish(best_f, deadline - 12.0, rng, max_passes=100)
            if sc_opt < best_score:
                best_score = sc_opt
                best_f = f_opt.copy()
                print(f"  Final polish -> {best_score:.7f}")

    # Validate with official evaluator
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = evaluate_sequence([float(x) for x in best_f])
    print(f"\nOfficial evaluation: {final_score:.7f}")

    if not np.isfinite(final_score):
        print("  Warning: invalid official score, using computed score")
        final_score = best_score

    print(f"\n=== FINAL SCORE: {final_score:.7f} ===")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")

    return [float(x) for x in best_f]
