```python
from scipy.optimize import linprog

def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lp_direction(a, M, K=100):
    """Find LP direction to reduce max convolution."""
    n = len(a)
    c = fast_self_conv(a)
    topk_idx = np.argpartition(c, -K)[-K:]
    
    # Build constraint matrix T (K x n)

    T = np.zeros((len(topk_idx), n))
```

```python
"""
Hybrid LP-guided + adaptive gradient optimizer for minimizing 2n*max(conv(a,a))/(sum a)^2.
Uses LP-based direction finding (solving restricted LPs over top-K peak constraints)
interleaved with soft-max gradient descent and Adam. Phase-based approach with
diversified exploration and fine polishing.

import numpy as np
import time
from scipy.optimize import linprog


def fast_self_conv(a):
    """Linear self-convolution via real FFT, returns full convolution of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, c, S):
    """Objective: 2n * max(conv) / S^2."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def build_and_solve_lp(a, M, K=100):
    """
    Solve restricted LP: maximize sum(g) s.t. conv(a,g)_k <= M for top-K peaks, g >= 0.
    Returns the LP solution g or None on failure.
    """
    n = len(a)
    c = fast_self_conv(a)
    K = min(K, 2 * n - 1)
    topk_idx = np.argpartition(c, -K)[-K:].astype(int)

    # Build constraint matrix T (K x n)
    # Constraint k: sum_i a[k-i] * g_i <= M for valid i
    T = np.zeros((K, n))
    for idx, k in enumerate(topk_idx):
        i_start = max(0, k - n + 1)
        i_end = min(k, n - 1)
        if i_start <= i_end:
            # g_i appears with coefficient a[k-i] for i in [i_start, i_end]
            # a[k-i_end : k-i_start+1] reversed
            T[idx, i_start:i_end + 1] = a[k - i_end:k - i_start + 1][::-1]

    # LP: minimize -sum(g) s.t. T@g <= M, g >= 0
    c_obj = -np.ones(n)
    b_ub = np.full(K, M + 1e-12)
    try:
        res = linprog(c_obj, A_ub=T, b_ub=b_ub, method='highs',
                      options={'time_limit': 3.0, 'presolve': True})
        if res.success:
            return np.maximum(res.x, 0.0)
    except Exception:
        pass
    return None


def softmax_gradient(a, c, topk, temp_scale=0.1):
    """
    Compute soft-max gradient of 2n*max(c)/S^2 w.r.t. a.
    Returns (gradient, M_smooth).
    """
    n = len(a)
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S
    k_act = min(topk, c.size)

    part = np.argpartition(c, -k_act)[-k_act:]
    vals = c[part].copy()
    top1 = float(np.max(vals))

    if top1 < 1e-15:
        return np.zeros(n), 0.0

    temp = max(temp_scale * top1, 1e-15)
    wts = np.exp((vals - top1) / temp)
    wsum = wts.sum()
    if wsum < 1e-30:
        wts = np.zeros(k_act)
        wts[0] = 1.0
    else:
        wts = wts / wsum
    M_smooth = float(np.dot(vals, wts))

    grad = np.zeros(n)
    for i in range(k_act):
        wi = float(wts[i])
        if wi < 1e-12:
            continue
        ki = int(part[i])
        i_start = max(0, ki - n + 1)
        i_end = min(ki, n - 1)
        if i_start <= i_end:
            grad[i_start:i_end + 1] += wi * 2.0 * a[ki - i_end:ki - i_start + 1][::-1]

    grad = (2.0 * n) * (grad / S2 - 2.0 * M_smooth / S3)
    return grad, M_smooth


def adam_step(a, g, m, v, t, lr):
    """In-place Adam update."""
    m[:] = 0.9 * m + 0.1 * g
    v[:] = 0.999 * v + 0.001 * g * g
    b1 = 1.0 - 0.9 ** t
    b2 = 1.0 - 0.999 ** t
    a -= lr * (m / b1) / (np.sqrt(v / b2) + 1e-8)

def generate_inits_from_parent(parent, n, rng):
```
