def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv_grad(a):
    """
    For each k in conv, the gradient w.r.t. a[j] is 2*a[k-j] (if valid).
    Returns the full gradient matrix structure via a function that computes
    grad of c[k] for a given k.
    """
    return a


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_objective_and_grad(a, beta):
    """
    LSE-smoothed score and gradient.
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip to avoid overflow
    scaled = np.clip(scaled, -500, 500)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a[j]: sum_k p[k] * dc[k]/da[j] = sum_k p[k] * 2*a[k-j]
    # This is 2 * (cross-correlation of a and p)
    # dc[k]/da[j] = 2*a[k-j] for valid k-j
    # sum_k p[k] * 2*a[k-j] = 2 * (p * rev_a) cross-correlation at lag j
    # = 2 * ifft(fft(p) * conj(fft(rev_a))) at appropriate indices
    # More directly: conv of p with reversed a, extracted at right indices
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(P * RA, M2_fft)
    # conv_res[k] = sum_i p[i] * rev_a[k-i] = sum_i p[i] * a[n-1-(k-i)]
    # We want sum_k p[k] * a[k-j] for each j
    # = sum_k p[k] * a[k-j] = (p * a)[j] in correlation sense
    # Let's use: g[j] = sum_k p[k] * a[k-j] where k-j must be in [0,n-1]
    # This is the cross-correlation of p and a at lag j
    # = ifft(fft(p, 2n-1) * conj(fft(a, 2n-1))) but that gives autocorrelation style
    # Actually: g[j] = sum_k p[k] * a[k-j]
    # Let m = k-j, so k = m+j: g[j] = sum_m p[m+j] * a[m]
    # This is conv(p, a_rev)[n-1+j] where a_rev is reversed... 
    # Let me just compute it directly via FFT cross-correlation
    # g[j] = sum_k p[k] * a[k-j] = (p cross-correlate a)[j]
    # In FFT: ifft(fft(p, N) * conj(fft(a, N))) gives circular cross-correlation
    # For linear: use zero-padding
    N = 2 * n - 1
    N_fft = 1 << (N - 1).bit_length()
    P_fft = np.fft.rfft(p, N_fft)
    A_fft = np.fft.rfft(a, N_fft)
    cross = np.fft.irfft(P_fft * np.conj(A_fft), N_fft)
    # cross[j] = sum_k p[k] * a[(k-j) mod N] ... this is circular
    # For linear cross-correlation: cross[j] for j=0..n-1 is what we want
    # Actually cross[j] = sum_i p[i] * a[(i-j) mod N]
    # For i-j in [0, N-1] and both p[i], a[i-j] valid (i in [0,2n-2], i-j in [0,n-1])
    # cross[j] = sum_{i=j}^{j+2n-2} p[i] * a[i-j] = sum_{m=0}^{2n-2} p[m+j] * a[m]
    # But p has length 2n-1 (indices 0 to 2n-2), so p[m+j] is valid only for m+j <= 2n-2, i.e., m <= 2n-2-j
    # And a[m] is valid for m in [0, n-1]
    # So cross[j] = sum_{m=0}^{min(n-1, 2n-2-j)} p[m+j] * a[m]
    # This is exactly g[j] = sum_k p[k] * a[k-j] with k=m+j!
    g_L = 2.0 * cross[:n]
    # Full gradient of score = 2n * (g_L/S^2 - 2*L/S^3)
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g_L / S2 - 2.0 * L / S3)
    return score, grad


def power_sum_objective_and_grad(a, p_power):
    """
    Power-sum smoothed objective: (sum c[k]^p / (2n-1)) / (sum a)^2
    Times 2n for consistency with the score.
    Returns (score, gradient).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    # Use c^p (all c[k] >= 0 since a >= 0)
    c_p = c ** p_power
    c_p_sum = float(c_p.sum())
    score = 2.0 * n * c_p_sum / ((2 * n - 1) * S * S)
    # Gradient: d/da[j] of c[k]^p = p * c[k]^{p-1} * dc[k]/da[j] = p * c[k]^{p-1} * 2*a[k-j]
    # So d(score)/da[j] = 2n/((2n-1)*S^2) * sum_k p * c[k]^{p-1} * 2*a[k-j] - 2*score/S * ...
    # = 2n/((2n-1)*S^2) * 2p * sum_k c[k]^{p-1} * a[k-j] - 2 * (2n * c_p_sum/((2n-1)*S^2)) / S
    # = 4np/((2n-1)*S^2) * sum_k c[k]^{p-1} * a[k-j] - 4n*c_p_sum/((2n-1)*S^3)
    c_pm1 = c ** (p_power - 1)
    # sum_k c[k]^{p-1} * a[k-j] = cross-correlation of c_pm1 and a
    N = 2 * n - 1
    N_fft = 1 << (N - 1).bit_length()
    C_fft = np.fft.rfft(c_pm1, N_fft)
    A_fft = np.fft.rfft(a, N_fft)
    cross = np.fft.irfft(C_fft * np.conj(A_fft), N_fft)
    g = cross[:n]
    S2 = S * S
    S3 = S2 * S
    coeff = 4.0 * n * p_power / ((2 * n - 1) * S2)
    const = 4.0 * n * c_p_sum / ((2 * n - 1) * S3)
    grad = coeff * g - const
    return score, grad


def run_optimize(a0, obj_func, maxiter=5000, maxfun=100000, ftol=1e-15, gtol=1e-13):
    """Run L-BFGS-B with a given objective function returning (val, grad)."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        obj_func, a0,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': ftol, 'gtol': gtol,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def make_lse_func(beta):
    """Create a closure-free LSE objective function wrapper."""
    def _obj(a):
        return lse_objective_and_grad(a, beta)
    return _obj


def make_power_func(p_val):
    """Create a closure-free power-sum objective function wrapper."""
    def _obj(a):
        return power_sum_objective_and_grad(a, p_val)
    return _obj
