Let me also think about what makes this problem hard. The issue is that the optimum (if it exists in the interior) requires all convolution entries to be equal (or at least the top few to be close). This is a very constrained condition.

For a non-negative sequence with sum 1, the convolution entries sum to 1. If all 2n-1 entries were equal, each would be 1/(2n-1). But can we achieve this?

If `a` is uniform (all = 1/n), the convolution is a triangle: `conv[k] = (n-|k|+1)/n^2` for `k = -(n-1), ..., n-1`. The max is at k=0: `n/n^2 = 1/n`. The min is at k=±(n-1): `1/n^2`. So the ratio max/min = n, which is far from 1.

To make the convolution flatter, we need `a` to be more "random" or "spread out" in a non-trivial way.

One approach: use a sequence that's close to white noise but non-negative. For example, a mixture of uniform and random: `a = (1-α) * uniform + α * random`, then normalize.

For large n and α close to 1, this would give a nearly flat convolution. But for non-negative random sequences, the convolution might still have some structure.

Actually, let me think about this differently. If `a` is i.i.d. exponential (non-negative), then by the law of large numbers, for large n:
- `E[a_i] = 1/n` (after normalization)
- `E[conv[k]] = sum_{i} E[a_i * a_{k-i}] ≈ min(k+1, n, 2n-1-k) * (1/n)^2`

So the expected convolution is still triangular! The randomness doesn't help flatten it because the expected value of the product is the product of expectations (for independent variables).

To get a flat convolution, we need the products `a_i * a_{k-i}` to vary from their mean in a coordinated way. This is a non-trivial constraint.

"""
Step function optimizer using soft-max gradient descent with FFT acceleration,
spectral equalization, and adaptive restarts.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative a.
"""

import numpy as np
import time


def _conv_fft(a):
    n = len(a)
    N = 1 << (2*n - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2*n-1]


def _score_from_conv(c, n, sum_a):
    if sum_a < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (sum_a * sum_a)


def _score(a):
    n = len(a)
    s = a.sum()
    c = _conv_fft(a)
    return _score_from_conv(c, n, s), c


def _softmax_weights(c, tau):
    shifted = tau * c - tau * c.max()
    w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= s
    return w


def _grad_from_weights(w, a):
    """Compute grad[j] = 2 * sum_k w[k] * a[k-j] using FFT."""
    n = len(a)
    a_flip = a[::-1]
    N = 1 << (2*(2*n-1) - 1).bit_length()  # enough for linear conv of w (len 2n-1) and a_flip (len n)
    W = np.fft.rfft(w, N)
    Af = np.fft.rfft(a_flip, N)
    conv = np.fft.irfft(W * Af, N)
    return 2.0 * conv[n-1:n-1+n]

    #     momentum_new = beta * momentum + grad
    # Actually that makes grad_eff == momentum_new, which is redundant.
    # Let me use: effective_grad = grad + beta * momentum (LARS-style)
    # momentum = grad + beta * momentum
    pass
    eta_local = eta
    log_a = np.log(a + 1e-300) - eta_local * grad_eff
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    s = a_new.sum()
    if s < 1e-300:

    a_new /= s
    return a_new, momentum_new


def _spectral_equalize(a, n_iters=100, alpha=0.5):
    """
    FFT-based spectral equalization.
    Iteratively: FFT -> flatten magnitude -> IFFT -> clamp non-negative -> normalize.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        return a
    a /= s
    
    N = 1 << (2*n - 1).bit_length()
    
    for it in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        
        # Target: flat magnitude (all equal to mean)
        # Keep DC component
        mean_mag = np.mean(mag)
        target_mag = np.full(N, mean_mag)
        target_mag[0] = mag[0]  # preserve DC
        
        # Relaxation: blend current and target
        new_mag = alpha * mag + (1.0 - alpha) * target_mag
        # Actually, let me swap: alpha is how much of target to use
        new_mag = (1.0 - alpha) * mag + alpha * target_mag
        
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new)[:n].real
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        a = a_new
    
    return a


def _perturb(a, rng, mode=0):
    """Apply a perturbation to escape local minima."""
    n = len(a)
    j = np.arange(n, dtype=np.float64)
    a_try = a.copy()
    
    if mode == 0:
        # Exponential tilt
        alpha = rng.uniform(-0.5, 0.5)
        a_try *= np.exp(alpha * (j/n - 0.5))
    elif mode == 1:
        # Sinusoidal modulation
        freq = rng.integers(1, max(2, n//4) + 1)
        phase = rng.uniform(0, 2*np.pi)
        amp = rng.uniform(0.1, 0.4)
        a_try *= (1.0 + amp * np.sin(2*np.pi*freq*j/n + phase))
    elif mode == 2:
        # Random mask (zero out a random fraction)
        mask = (rng.random(n) > 0.1).astype(float)
        a_try *= mask
    elif mode == 3:
        # Swap structure
        a_try = a_try[rng.permutation(n)]
    elif mode == 4:
        # Add noise
        a_try += rng.normal(0, 0.01 * a.mean(), n)
    elif mode == 5:
        # Inverse (flip the sequence)
        a_try = a_try[::-1]
    
    a_try = np.maximum(a_try, 0.0)
    s = a_try.sum()
    if s < 1e-10:
        return a.copy()
    return a_try / s

    if s < 1e-10:

    else:
        a /= s

    beta = 0.9
    step = step_init

    patience = 30

        tau = tau_init * (tau_max / tau_init) ** (progress ** 0.7)
        
        # Compute convolution and weights
        c = _conv_fft(a)
        w = _softmax_weights(c, tau)
        
        # Compute gradient
        grad = _grad_from_weights(w, a)
        
        # Mirror step with momentum
        grad_eff = grad + beta * momentum
        momentum = grad + beta * momentum
        
        span = grad_eff.max() - grad_eff.min()
        if span < 1e-30:
            break

        log_a -= log_a.max()
        a_new = np.exp(log_a)

def _coord_descent(a0, deadline, top_k=3, n_trials_per_coord=5):
    """
    Coordinate descent: try moving mass from high-contribution to low-contribution positions.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    a /= a.sum()
    
    best_score = _score(a)[0]
    best_a = a.copy()
    
    rng = np.random.default_rng(12345)
    
    while time.monotonic() < deadline:
        c = _conv_fft(a)
        peak = c.max()
        
        # Top K lags
        top_idx = np.argpartition(c, -top_k)[-top_k:]
        
        # Compute total contribution per position
        contrib = np.zeros(n)
        for k in top_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                idx = k - np.arange(j_start, j_end + 1)
                contrib[j_start:j_end+1] += a[j_start:j_end+1] * a[idx]
        
        # Also consider global contribution (sum over all lags)
        # This is expensive, so just use top-k contribution
        
        # Try several random donor->receiver moves
        improved = False
        for trial in range(20):
            # Donor: random position with high contrib
            if contrib.max() > 0:
                # Weighted random selection from high-contrib positions
                weights = contrib ** 2
                weights /= weights.sum()
                j_don = rng.choice(n, p=weights)
            else:
                j_don = rng.integers(n)
            
            # Receiver: random position with low contrib
            low_weights = 1.0 / (contrib + 1e-10)
            low_weights /= low_weights.sum()
            j_rec = rng.choice(n, p=low_weights)
            
            if j_don == j_rec:
                continue
            
            # Try multiple transfer amounts
            best_move = None
            best_peak = peak
            for frac in [0.05, 0.1, 0.15, 0.2, 0.3]:
                amt = a[j_don] * frac
                if amt < 1e-15:
                    continue
                a_try = a.copy()
                a_try[j_don] -= amt
                a_try[j_rec] += amt
                if a_try[j_don] < -1e-15:
                    continue
                a_try = np.maximum(a_try, 0)
                a_try /= a_try.sum()
                c_try = _conv_fft(a_try)
                if c_try.max() < best_peak:
                    best_peak = c_try.max()
                    best_move = a_try
            
            if best_move is not None:
                a = best_move
                c = _conv_fft(a)
                peak = c.max()
                score = _score_from_conv(c, n, a.sum())
                if score < best_score:
                    best_score = score
                    best_a = a.copy()
                improved = True
                break  # restart the outer loop
    
        if not improved:
            # Try a structural perturbation
            a = _perturb(a, rng, mode=rng.integers(6))
    
    return best_a, best_score

"""
Step function optimizer using multi-phase strategy:
1. Spectral equalization (FFT magnitude flattening)
2. MSE-smooth gradient descent (mean-square error of convolution)
3. Soft-max annealed gradient descent (targets actual max objective)
4. Focused top-peak gradient descent
5. Coordinate descent polish
All gradients computed via FFT for O(n log n) per iteration.

import numpy as np
import time


def _conv_fft(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    N = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:2 * n - 1]


def _score(a):
    """Compute score and return (score, convolution)."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf, None
    c = _conv_fft(a)
    sc = 2.0 * n * c.max() / (s * s)
    return sc, c


def _softmax_weights(c, tau):
    """Compute softmax weights over convolution entries."""
    shifted = tau * c - tau * c.max()
    w = np.exp(shifted)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w /= s
    return w


def _grad_from_weights(w, a):
    """
    Compute grad[j] = 2 * sum_k w[k] * a[k-j] using FFT.
    Uses the identity: grad = 2 * (w * a_flipped)[n-1 : n-1+n]
    """
    n = len(a)
    a_flip = a[::-1]
    # Need N >= (2n-1) + n - 1 = 3n - 2 for linear convolution
    N = 1 << (3 * n).bit_length()
    W = np.fft.rfft(w, N)
    Af = np.fft.rfft(a_flip, N)
    conv = np.fft.irfft(W * Af, N)
    return 2.0 * conv[n - 1:n - 1 + n]


def _grad_mse(a, c=None):
    """
    Gradient of sum_k (conv_k - target)^2 where target = 1/len(conv).
    This equals 4 * cross_correlation(e, a) where e_k = conv_k - target.
    """
    n = len(a)
    if c is None:
        c = _conv_fft(a)
    target = 1.0 / len(c)
    e = c - target
    # grad[j] = 4 * sum_k e[k] * a[k-j]
    a_flip = a[::-1]
    N = 1 << (3 * n).bit_length()
    E = np.fft.rfft(e, N)
    Af = np.fft.rfft(a_flip, N)
    conv = np.fft.irfft(E * Af, N)
    return 4.0 * conv[n - 1:n - 1 + n]


def _mirror_step(a, grad, eta, momentum, beta):
    """Exponentiated gradient (mirror) descent step with momentum."""
    grad_eff = grad + momentum
    momentum_new = grad + beta * momentum
    log_a = np.log(a + 1e-300) - eta * grad_eff
    log_a -= log_a.max()
    a_new = np.exp(log_a)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy(), momentum
    a_new /= s
    return a_new, momentum_new


def _spectral_equalize(a, n_iters=200, alpha=0.3):
    """
    FFT-based spectral equalization.
    Iteratively flattens the power spectrum while maintaining non-negativity.
    """
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s

    N = 1 << (2 * n - 1).bit_length()

    for it in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)

        # Target flat magnitude, preserve DC
        mean_mag = np.mean(mag[1:]) if N > 1 else mag[0]
        target_mag = np.full(N, mean_mag)
        target_mag[0] = mag[0]  # Keep DC (sum)

        # Blend
        new_mag = (1.0 - alpha) * mag + alpha * target_mag
        A_new = new_mag * np.exp(1j * phase)
        a_new = np.fft.ifft(A_new)[:n].real
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new /= s
        a = a_new

    return a

def _coord_descent(a0, deadline, top_k=3, n_trials=30):
    """
    Coordinate descent: transfer mass from high-contribution to low-contribution positions.
    """
    n = len(a0)
    a = np.maximum(a0, 0.0).astype(np.float64)
    a /= a.sum()

    best_score, _ = _score(a)
    best_a = a.copy()

    rng = np.random.default_rng(99999)

    while time.monotonic() < deadline:
        c = _conv_fft(a)
        peak = c.max()

        top_idx = np.argpartition(c, -top_k)[-top_k:]

        # Compute contribution per position (to top k lags)
        contrib = np.zeros(n)
        for k in top_idx:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                idx = k - np.arange(j_start, j_end + 1)
                contrib[j_start:j_end + 1] += a[j_start:j_end + 1] * a[idx]

        # Also add total contribution (proxy: sum of all conv entries weighted)
        # For speed, just use top-k contribution

        improved = False
        for _ in range(n_trials):
            if contrib.max() < 1e-15:
                j_don = int(rng.integers(n))
            else:
                w_don = contrib ** 2
                w_don /= w_don.sum()
                j_don = int(rng.choice(n, p=w_don))

            w_rec = 1.0 / (contrib + 1e-10)
            w_rec /= w_rec.sum()
            j_rec = int(rng.choice(n, p=w_rec))

            if j_don == j_rec:
                continue

            best_move = None
            best_peak = peak
            for frac in [0.05, 0.1, 0.15, 0.2, 0.3, 0.5]:
                amt = a[j_don] * frac
                if amt < 1e-15:
                    continue
                a_try = a.copy()
                a_try[j_don] -= amt
                a_try[j_rec] += amt
                a_try = np.maximum(a_try, 0)
                s_try = a_try.sum()
                if s_try < 1e-10:
                    continue
                a_try /= s_try
                c_try = _conv_fft(a_try)
                if c_try.max() < best_peak:
                    best_peak = c_try.max()
                    best_move = a_try

            if best_move is not None:
                a = best_move
                c = _conv_fft(a)
                peak = c.max()
                sc = 2.0 * n * peak / (a.sum() ** 2)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                improved = True
                break

        if not improved:
            a = _perturb(a, rng, mode=int(rng.integers(7)))
            sc, _ = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    return best_a, best_score

    hard_deadline = t_start + budget_s - 30.0

    # Initialize from given sequence
    f_init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(f_init)
    s0 = f_init.sum()
    if s0 > 1e-10:
        best_a = f_init / s0
    else:
        best_a = np.ones(n) / n

    best_score, _ = _score(best_a)
    print(f"Initial: n={n}, score={best_score:.7f}")

    # --- Phase 1: Spectral Equalization ---
    print(f"\n=== Phase 1: Spectral Equalization ===")
    p1_budget = min(60.0, (budget_s - 60) * 0.03)
    d1 = time.monotonic() + p1_budget

    # Try from given and a few other starts
    candidates = [best_a.copy()]
    candidates.append(np.ones(n) / n)
    j = np.arange(n, dtype=np.float64)
    candidates.append(np.maximum(1.0 + 0.3 * np.sin(2 * np.pi * 7 * j / n), 0.01))
    candidates[-1] /= candidates[-1].sum()
    candidates.append(np.maximum(rng.standard_normal(n), 0))
    candidates[-1] /= candidates[-1].sum()

    for alpha in [0.1, 0.3, 0.5]:
        if time.monotonic() > d1:
            break
        for ci in range(min(3, len(candidates))):
            if time.monotonic() > d1:
                break
            a_eq = _spectral_equalize(candidates[ci], n_iters=150, alpha=alpha)
            sc, _ = _score(a_eq)
            if sc < best_score:
                best_score = sc
                best_a = a_eq.copy()
                print(f"  Spectral eq (alpha={alpha}, start={ci}): {sc:.7f}")

    print(f"  After Phase 1: best={best_score:.7f}")

    # --- Phase 2: MSE Gradient Descent ---
    print(f"\n=== Phase 2: MSE Gradient Descent ===")
    p2_budget = min(400.0, (budget_s - 300) * 0.2)
    d2 = time.monotonic() + p2_budget

    a_opt, sc_opt = _optimize_mse(best_a, d2, step_init=0.5,
                                   verbose=True, label="mse")
    if sc_opt < best_score:
        best_score = sc_opt
        best_a = a_opt.copy()
        print(f"  MSE result: {sc_opt:.7f}")
    else:
        print(f"  MSE best: {sc_opt:.7f}")

    # --- Phase 3: Soft-max Annealed Gradient Descent ---
    print(f"\n=== Phase 3: Soft-max Gradient Descent ===")
    p3_budget = min(800.0, (budget_s - 300) * 0.35)
    d3 = time.monotonic() + p3_budget

    a_opt, sc_opt = _optimize_softmax(best_a, d3,
                                       tau_init=20.0, tau_max=3000.0,
                                       step_init=1.0, step_min=0.005,
                                       verbose=True, label="softmax")
    if sc_opt < best_score:
        best_score = sc_opt
        best_a = a_opt.copy()
        print(f"  Softmax result: {sc_opt:.7f}")
    else:
        print(f"  Softmax best: {sc_opt:.7f}")

    # --- Phase 4: Focused Gradient Descent ---
    print(f"\n=== Phase 4: Focused Gradient Descent ===")
    p4_budget = min(500.0, (budget_s - 200) * 0.25)
    d4 = time.monotonic() + p4_budget

    for K in [3, 5]:
        if time.monotonic() > d4 - 60:
            break
        d4_sub = min(time.monotonic() + (d4 - time.monotonic()) / 2, d4 - 30)
        a_opt, sc_opt = _optimize_focused(best_a, d4_sub, K=K,
                                           verbose=True, label=f"K={K}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"  Focused K={K}: {sc_opt:.7f}")
        else:
            print(f"  Focused K={K}: {sc_opt:.7f}")

    # --- Phase 5: Restarts ---
    print(f"\n=== Phase 5: Restarts with Perturbations ===")
    p5_budget = min(300.0, max(60.0, (budget_s - 150) * 0.15))
    d5 = time.monotonic() + p5_budget

    for i in range(5):
        if time.monotonic() > d5 - 30:
            break
        a_pert = _perturb(best_a, rng, mode=i % 7)
        remaining = d5 - time.monotonic()
        if remaining < 30:
            break
        d5_sub = min(time.monotonic() + remaining / (5 - i) * 0.9, d5 - 10)

        if i % 2 == 0:
            a_opt, sc_opt = _optimize_softmax(a_pert, d5_sub,
                                               tau_init=50, tau_max=2000,
                                               verbose=False, label=f"r{i}")
        else:
            a_opt, sc_opt = _optimize_focused(a_pert, d5_sub, K=3,
                                               verbose=False, label=f"r{i}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"  Restart {i}: {sc_opt:.7f}")
        else:
            print(f"  Restart {i}: {sc_opt:.7f}")

    # --- Phase 6: Coordinate Descent Polish ---
    print(f"\n=== Phase 6: Coordinate Descent Polish ===")
    remaining = max(30.0, hard_deadline - time.monotonic() - 20.0)
    d6 = time.monotonic() + min(remaining, 300.0)

    a_opt, sc_opt = _coord_descent(best_a, d6, top_k=3, n_trials=30)
    if sc_opt < best_score:
        best_score = sc_opt
        best_a = a_opt.copy()
        print(f"  Coord descent: {sc_opt:.7f}")
    else:
        print(f"  Coord descent: {sc_opt:.7f}")

    # --- Phase 7: Final micro-polish if time remaining ---
    if time.monotonic() < hard_deadline - 40:
        print(f"\n=== Phase 7: Final Micro-Polish ===")
        d7 = hard_deadline - 10.0
        a_opt, sc_opt = _coord_descent(best_a, d7, top_k=5, n_trials=50)
        if sc_opt < best_score:
            best_score = sc_opt
            best_a = a_opt.copy()
            print(f"  Micro-polish: {sc_opt:.7f}")

    # Final validation with official evaluator
    best_a = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence([float(x) for x in best_a])
    print(f"\nOfficial evaluation: {final_score:.7f}")

    if not np.isfinite(final_score) or final_score > best_score + 1e-10:
        if np.isfinite(final_score):
            best_score = final_score
        else:
            best_score = _score(best_a)[0]

    print(f"\n=== FINAL SCORE: {best_score:.7f} ===")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")
    return [float(x) for x in best_a]
