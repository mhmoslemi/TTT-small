    if bs < 1:
        return None
    nb = n // bs
    eff_n = nb * bs
    main = p[:eff_n].reshape(nb, bs)
    tail = p[eff_n:] if eff_n < n else np.array([])
    perm = rng.permutation(nb)
    p_new = main[perm].reshape(-1)
    if len(tail) > 0:
        p_new = np.concatenate([p_new, tail])
    s = p_new.sum()
    if s < 1e-10:
        return None
    return p_new / s

def _alternating_arrange(p, rng):
    n = len(p)
    n_blocks = int(np.sqrt(n))
    bs = n // n_blocks
    nb = n // bs
    eff_n = nb * bs
    main = p[:eff_n].reshape(nb, bs)
    tail = p[eff_n:] if eff_n < n else np.array([])
    means = main.mean(axis=1)
    order = np.argsort(means)
    new_order = np.zeros(nb, dtype=int)
    for i in range(nb):
        if i % 2 == 0:
            new_order[i] = order[i // 2]
        else:
            new_order[i] = order[nb - 1 - (i + 1) // 2]
    p_new = main[new_order].reshape(-1)
    if len(tail) > 0:
        p_new = np.concatenate([p_new, tail])
    s = p_new.sum()
    if s < 1e-10:
        return None
    return p_new / s

def _optimize_core(p0, rng, time_budget, momentum_start=0.5, momentum_end=0.9,
                   K_start=15, K_end=80, lr_start=0.08, lr_decay=0.9995):
    p = np.asarray(p0, dtype=float).copy()
    s = p.sum()
    if s < 1e-10:
        return None, float('inf')
    p = p / s
    n = len(p)
    L = _next_pow2(2 * n - 1)

    best = p.copy()
    best_score = _score_norm1(p, L)
    cur_score = best_score

    q = np.log(np.maximum(p, 1e-15))
    v = np.zeros(n)
    lr = lr_start

    deadline = time.monotonic() + time_budget
    stall = 0
    it = 0
    n_iters_target = max(100, int(time_budget * 200))

    while time.monotonic() < deadline:
        progress = min(1.0, it / n_iters_target)
        momentum = momentum_start + (momentum_end - momentum_start) * progress
        K = int(K_start + (K_end - K_start) * progress)
        K = max(5, min(K, 150))
        thresh = 0.98 + 0.019 * progress

        r = rng.random()

        if r < 0.55:
            sc, grad = _topk_subgrad(p, K, thresh, L)
            if sc < best_score - 1e-10:
                best = p.copy()
                best_score = sc

            v = momentum * v - lr * grad
            q_new = q + v
            p_new = np.exp(q_new - q_new.max())
            s_new = p_new.sum()
            if s_new < 1e-10 or not np.isfinite(s_new):
                v *= 0.5
                stall += 1
                continue
            p_new = p_new / s_new
            p_new = np.maximum(p_new, 1e-15)
            p_new = p_new / p_new.sum()

            sc_new = _score_norm1(p_new, L)
            if sc_new < cur_score - 1e-13:
                q = np.log(np.maximum(p_new, 1e-15))
                p = p_new
                cur_score = sc_new
                stall = 0
                if sc_new < best_score - 1e-10:
                    best = p_new.copy()
                    best_score = sc_new
            else:
                stall += 1
                if stall % 5 == 0:
                    lr = max(0.001, lr * lr_decay)
                    v *= 0.5
                if stall > 60:
                    p = best.copy()
                    q = np.log(np.maximum(p, 1e-15))
                    v = np.zeros(n)
                    cur_score = best_score
                    lr = lr_start
                    stall = 10

        elif r < 0.75:
            alpha_f = float(rng.uniform(1.4, 5.0))
            if rng.random() < 0.6:
                p_new = _freq_flatten_phase(p, alpha_f, L)
            else:
                p_new = _freq_flatten_amp(p, alpha_f, L)
            if p_new is not None:
                sc_new = _score_norm1(p_new, L)
                if sc_new < cur_score - 1e-13:
                    p = p_new
                    q = np.log(np.maximum(p, 1e-15))
                    v = np.zeros(n)
                    cur_score = sc_new
                    stall = 0
                    if sc_new < best_score - 1e-10:
                        best = p.copy()
                        best_score = sc_new
                else:
                    stall += 1
            else:
                stall += 1

        elif r < 0.85:
            n_blocks = int(rng.choice([4, 8, 16, 32, 64, 128, 256]))
            n_blocks = min(n_blocks, n // 2)
            if rng.random() < 0.4:
                p_new = _alternating_arrange(p, rng)
            else:
                p_new = _block_permute(p, rng, n_blocks)
            if p_new is not None:
                sc_new = _score_norm1(p_new, L)
                if sc_new < best_score * 1.03:
                    p = p_new
                    q = np.log(np.maximum(p, 1e-15))
                    v = np.zeros(n)
                    cur_score = sc_new
                    stall = 0
                    if sc_new < best_score - 1e-10:
                        best = p.copy()
                        best_score = sc_new
                    # Quick subgradient refinement after permutation
                    for _ in range(5):
                        sc2, grad2 = _topk_subgrad(p, K, thresh, L)
                        v2 = -lr * grad2
                        p2 = p * np.exp(v2)
                        p2 = np.maximum(p2, 1e-15)
                        p2 = p2 / p2.sum()
                        sc2 = _score_norm1(p2, L)
                        if sc2 < cur_score - 1e-13:
                            p = p2
                            q = np.log(np.maximum(p, 1e-15))
                            cur_score = sc2
                            if sc2 < best_score - 1e-10:
                                best = p.copy()
                                best_score = sc2
                else:
                    stall += 1
            else:
                stall += 1

        else:
            if stall > 8:
                std = 0.03 + 0.008 * min(stall, 40)
                p_new = p * np.exp(rng.normal(0, std, n))
                p_new = np.maximum(p_new, 1e-15)
                p_new = p_new / p_new.sum()
                sc_new = _score_norm1(p_new, L)
                if sc_new < best_score * 1.08:
                    p = p_new
                    q = np.log(np.maximum(p, 1e-15))
                    v = np.zeros(n)
                    cur_score = sc_new
                    stall = 0
                    if sc_new < best_score - 1e-10:
                        best = p.copy()
                        best_score = sc_new
            else:
                stall += 1

        if cur_score > best_score * 1.15 and best_score < float('inf') and rng.random() < 0.15:
            p = best.copy()
            q = np.log(np.maximum(p, 1e-15))
            v = np.zeros(n)
            cur_score = best_score
            stall = 0

        it += 1

    return best, best_score

def _block_refine(p0, rng, time_budget):
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    L = _next_pow2(2 * n - 1)
    best = p.copy()
    best_score = _score_norm1(p, L)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0
    factors = (0.5, 0.7, 0.85, 1.0, 1.18, 1.4, 1.8, 2.5)

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.3:
            bs = int(rng.integers(1, 8))
        elif r < 0.7:
            bs = int(rng.integers(max(2, n // 256), max(3, n // 16)))
        else:
            bs = int(rng.integers(max(8, n // 16), max(16, n // 4)))
        bs = min(bs, n)
        start = int(rng.integers(0, n - bs + 1))

        improved = False
        for f in factors:
            p_try = p.copy()
            p_try[start:start + bs] *= f
            p_try = np.maximum(p_try, 1e-15)
            p_try = p_try / p_try.sum()
            sc = _score_norm1(p_try, L)
            if sc < cur_score - 1e-13:
                cur_score = sc
                p = p_try
                improved = True
                break

        if improved:
            if cur_score < best_score - 1e-12:
                best = p.copy()
                best_score = cur_score
            stall = 0
        else:
            stall += 1
            if stall > 40:
                p = best.copy()
                cur_score = best_score
                stall = 0

    return best, best_score

def _resample(p_in, n_new, rng=None):
    p = _sanitize(p_in)
    if p is None:
        return None
    n_old = len(p)
    if n_old == n_new:
        result = p.copy()
    elif n_old == 1:
        result = np.ones(n_new, dtype=float)
    else:
        x_old = (np.arange(n_old) + 0.5) / n_old
        x_new = (np.arange(n_new) + 0.5) / n_new
        result = np.interp(x_new, x_old, p)
    result = np.maximum(result, 1e-10)
    if rng is not None:
        result = result * np.exp(rng.normal(0, 0.02, n_new))
        result = np.maximum(result, 1e-10)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s

def _init_cosine_structured(n, rng):
    t = np.arange(n, dtype=float) / n
    p = np.ones(n, dtype=float)
    for k in range(1, 15):
        amp = rng.uniform(-0.5, 0.5) / (k ** 0.7)
        ph = rng.uniform(0, 2 * np.pi)
        p += amp * np.cos(2 * np.pi * k * t + ph)
    p = np.maximum(p, 0.01)
    return p / p.sum()

def _init_two_gaussian(n, rng):
    c1 = rng.uniform(0.2, 0.4)
    c2 = rng.uniform(0.6, 0.8)
    s1 = rng.uniform(0.05, 0.15)
    s2 = rng.uniform(0.05, 0.15)
    w1 = rng.uniform(0.5, 2.0)
    w2 = rng.uniform(0.5, 2.0)
    t = np.arange(n, dtype=float) / n
    x1 = np.minimum(np.abs(t - c1), 1 - np.abs(t - c1)) / s1
    x2 = np.minimum(np.abs(t - c2), 1 - np.abs(t - c2)) / s2
    p = w1 * np.exp(-0.5 * x1 * x1) + w2 * np.exp(-0.5 * x2 * x2) + 0.01
    return p / p.sum()

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 120:
        budget = 2400.0

    start_time = time.monotonic()
    deadline = start_time + budget - 20

    iseed = int(seed)
    rng = np.random.default_rng(iseed % (2**31 - 1))

    best_overall = None
    best_score_overall = float('inf')

    def _try_update(p, score):
        nonlocal best_overall, best_score_overall
        if p is not None and np.isfinite(score) and score < best_score_overall:
            best_overall = p.copy()
            best_score_overall = float(score)
            print(f"  *** NEW BEST: {score:.6f} (n={len(p)}) at t={time.monotonic()-start_time:.0f}s ***")

    # Load parent
    parent = None
    try:
        parent = _sanitize(np.asarray(height_sequence_1, dtype=float))
    except Exception:
        pass

    if parent is not None:
        sc = _score(parent)
        _try_update(parent, sc)
        print(f"Parent: score={sc:.6f}, n={len(parent)}")

    # Phase 1: Quick multi-length exploration (10% time)
    print("=" * 60)
    print("Phase 1: Multi-length exploration")
    t1_end = start_time + 0.10 * budget

    lengths_to_try = [2048, 3072, 4096]
    if parent is not None:
        pn = len(parent)
        for cand in [1024, 2048, 3072, 4096, 6144]:
            if cand not in lengths_to_try and abs(cand - pn) < pn * 0.5:
                lengths_to_try.append(cand)
        lengths_to_try.sort()

    candidates = []
    for n in lengths_to_try:
        if time.monotonic() > t1_end - 5:
            break
        inits = [
            np.ones(n) / n,
            _init_cosine_structured(n, rng),
            _init_two_gaussian(n, rng),
        ]
        if parent is not None:
            spr = _resample(parent, n, rng)
            if spr is not None:
                inits.append(spr)

        for sp in inits:
            sc = _score(sp)
            if sc < best_score_overall:
                _try_update(sp, sc)
            candidates.append((sp, sc))

        print(f"  n={n}: best so far={best_score_overall:.6f}")

    print(f"Phase 1 done. Best: {best_score_overall:.6f}")

    # Phase 2: Main optimization on top candidates (65% time)
    print("Phase 2: Main optimization")
    t2_end = start_time + 0.75 * budget

    # Sort candidates by score, take top 3
    candidates.sort(key=lambda x: x[1])
    top_candidates = candidates[:3]

    t_per_opt = (t2_end - time.monotonic()) / max(1, len(top_candidates))
    t_per_opt = max(30.0, min(t_per_opt, 400.0))

    for idx, (sp, sc) in enumerate(top_candidates):
        t_rem = t2_end - time.monotonic()
        if t_rem < 30:
            break
        t_alloc = min(t_per_opt, t_rem * 0.95)
        t_alloc = max(20.0, t_alloc)
        print(f"  Optimizing candidate {idx+1}/{len(top_candidates)}: n={len(sp)}, score={sc:.6f}, budget={t_alloc:.0f}s")
        p, sc = _optimize_core(sp, rng, t_alloc,
                               momentum_start=0.4, momentum_end=0.85,
                               K_start=15, K_end=80, lr_start=0.08)
        _try_update(p, sc)

    # Phase 3: Diverse restarts (10% time)
    print("Phase 3: Diverse restarts")
    t3_end = start_time + 0.88 * budget

    if best_overall is not None:
        n_best = len(best_overall)
        t_rem = t3_end - time.monotonic()
        if t_rem > 40:
            n_restarts = 3
            t_each = max(15.0, t_rem / n_restarts * 0.9)
            for ri in range(n_restarts):
                if time.monotonic() > t3_end - 5:
                    break
                if ri == 0:
                    sp = _resample(best_overall, n_best, rng)
                elif ri == 1:
                    sp = _init_cosine_structured(n_best, rng)
                else:
                    sp = _init_two_gaussian(n_best, rng)
                if sp is None:
                    continue
                p, sc = _optimize_core(sp, rng, t_each,
                                       momentum_start=0.5, momentum_end=0.9,
                                       K_start=20, K_end=60, lr_start=0.06)
                _try_update(p, sc)

    print(f"Phase 3 done. Best: {best_score_overall:.6f}")

    # Phase 4: Final polish (7% time)
    print("Phase 4: Final polish")
    if best_overall is not None:
        t_rem = deadline - time.monotonic()
        if t_rem > 30:
            # High-momentum gradient
            t_alloc = min(0.5 * t_rem, 180.0)
            t_alloc = max(20.0, t_alloc)
            print(f"  Polish 1: gradient (budget={t_alloc:.0f}s)")
            p, sc = _optimize_core(best_overall, rng, t_alloc,
                                   momentum_start=0.7, momentum_end=0.95,
                                   K_start=40, K_end=120, lr_start=0.03)
            _try_update(p, sc)

        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.3 * t_rem, 80.0)
            t_alloc = max(15.0, t_alloc)
            print(f"  Polish 2: block refine (budget={t_alloc:.0f}s)")
            p, sc = _block_refine(best_overall, rng, t_alloc)
            _try_update(p, sc)

        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.3 * t_rem, 60.0)
            t_alloc = max(10.0, t_alloc)
            print(f"  Polish 3: gradient (budget={t_alloc:.0f}s)")
            p, sc = _optimize_core(best_overall, rng, t_alloc,
                                   momentum_start=0.8, momentum_end=0.95,
                                   K_start=60, K_end=150, lr_start=0.02)
            _try_update(p, sc)

    # Final output
    if best_overall is None:
        best_overall = np.ones(2048) / 2048

    p_final = _sanitize(best_overall)
    if p_final is None:
        p_final = np.ones(2048) / 2048

    final_score = _score(p_final)
    try:
        eval_score = float(evaluate_sequence([float(x) for x in p_final]))
        if np.isfinite(eval_score):
            final_score = min(final_score, eval_score)
    except Exception:
        pass

    print(f"\n{'=' * 60}")
    print(f"FINAL: score={final_score:.6f} (n={len(p_final)})")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    out = np.where(np.isfinite(p_final), p_final, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = np.ones(2048) / 2048
    else:
        out = out / s

    return [float(x) for x in out]
