"""
Step function optimizer using L-BFGS with Lp-smoothed infinity norm,
Adam refinement, and coordinate descent polish.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
from scipy.signal import fftconvolve
from scipy.optimize import minimize
import time


def _next_pow2(x):
    return 1 << (int(x) - 1).bit_length()


def _compute_conv(a):
    """Full linear convolution a*a via FFT."""
    n = len(a)
    size = _next_pow2(2 * n - 1)
    F = np.fft.rfft(a, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _cross_corr(a, w):
    """Compute c[j] = sum_k a[k] * w[k+j] for j=0,...,n-1 using FFT.
    a has length n, w has length L (>= 2n-1).
    """
    n = len(a)
    L = len(w)
    N = _next_pow2(n + L - 1)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    # Circular cross-correlation: (a ⋆ w)[j] = sum_k a[k] * w[k+j mod N]
    corr = np.fft.irfft(np.conj(A) * W, N)
    return corr[:n]


def _score_smoothed(a, p=30):
    """Compute smoothed score: 2*n*(sum C[k]^p)^{1/p} / (sum a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    C = _compute_conv(a)
    T = np.sum(C ** p)
    if T < 1e-300:
        return 0.0
    return float(2.0 * n * T ** (1.0 / p) / (s * s))


def _score_and_grad(a, p=30):
    """Compute smoothed score and gradient w.r.t. a (unnormalized)."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10, np.zeros_like(a)
    C = _compute_conv(a)
    C_safe = np.maximum(C, 1e-300)
    T = np.sum(C_safe ** p)
    if T < 1e-300:
        return 0.0, np.zeros_like(a)
    score = float(2.0 * n * T ** (1.0 / p) / (s * s))
    # Gradient:
    # df/da[j] = (4*n/s^2) * T^{(1-p)/p} * c_raw[j] - (4*n/s^3) * T^{1/p}
    # where c_raw[j] = sum_k C[k]^{p-1} * a[k-j]
    w = C_safe ** (p - 1)
    c_raw = _cross_corr(a, w)
    T_factor = T ** ((1.0 - p) / p)
    T_factor2 = T ** (1.0 / p)
    grad = (4.0 * n / (s * s)) * T_factor * c_raw - (4.0 * n / (s ** 3)) * T_factor2
    return score, grad


def _score_exact(a):
    """Compute exact score: 2*n*max(conv(a,a))/(sum a)^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 1e10
    C = _compute_conv(a)
    return float(2.0 * n * np.max(C) / (s * s))


def _lbfgs_optimize(a0, deadline, p_schedule, max_iter=5000, verbose=False, label=""):
    """Run L-BFGS with a schedule of increasing p values."""
    a = np.maximum(a0.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(len(a))
    else:
        a = a / s

    best_score = _score_exact(a)
    best_a = a.copy()
    n = len(a)

    for idx, p in enumerate(p_schedule):
        if time.monotonic() > deadline:
            break
        remaining = deadline - time.monotonic()
        if remaining < 5:
            break

        # Set max iterations based on remaining time
        n_iter = min(max_iter, max(100, int(remaining * 200)))

        def obj(x):
            x = np.maximum(x, 0)
            sc, _ = _score_smoothed(x, p)
            return sc

        def grad(x):
            x = np.maximum(x, 0)
            _, g = _score_and_grad(x, p)
            return g

        try:
            res = minimize(
                obj, a, jac=grad, method='L-BFGS-B',
                bounds=[(0, None)] * n,
                options={'maxiter': n_iter, 'gtol': 1e-8, 'ftol': 1e-10,
                         'maxls': 50, 'maxfun': n_iter * 5}
            )
            a_new = np.maximum(res.x, 0)
            s_new = a_new.sum()
            if s_new > 1e-10:
                a_new = a_new / s_new
                sc_new = _score_exact(a_new)
                if sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    if verbose:
                        print(f"    {label} p={p}: {sc_new:.7f}")
                a = a_new
            else:
                if verbose:
                    print(f"    {label} p={p}: failed (sum too small)")
        except Exception as e:
            if verbose:
                print(f"    {label} p={p}: exception {e}")

    return best_a, best_score

        m = beta1 * m + (1 - beta1) * g

        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

    starts['sqrt_ramp'] = np.sqrt(np.arange(1, n + 1, dtype=np.float64))
    starts['exp_up'] = np.exp(3.0 * j / n)
    starts['exp_down'] = np.exp(-3.0 * j / n)
    starts['cos_bell'] = 0.5 + 0.5 * np.cos(2 * np.pi * j / n)
    starts['sine_abs'] = 1.0 + 0.5 * np.abs(np.sin(np.pi * j / n))
    starts['u_shape'] = 1.0 + 2.0 * ((j - center) / max(center, 1)) ** 2
    starts['bimodal'] = (np.exp(-((j / n - 0.3) / 0.15) ** 2) +
                         np.exp(-((j / n - 0.7) / 0.15) ** 2) + 0.01)
    starts['asym_gauss'] = np.exp(-((j / n - 0.4) / 0.2) ** 2) + 0.01
    starts['asym_gauss2'] = np.exp(-((j / n - 0.6) / 0.2) ** 2) + 0.01
    starts['log_ramp'] = np.log(np.arange(1, n + 1, dtype=np.float64) + 1)
    starts['quad'] = 1.0 + 3.0 * (j / n - 0.3) ** 2
    starts['cubic'] = 1.0 + 2.0 * (j / n - 0.5) ** 3 + 1.0
    starts['rand1'] = rng.exponential(1.0, size=n)
    starts['rand2'] = rng.dirichlet(np.ones(n) * 0.5) * n
    starts['zigzag'] = 1.0 + 0.3 * np.sin(2 * np.pi * 5 * j / n)
    starts['multi_sin'] = 1.0 + 0.2 * np.sin(2 * np.pi * 3 * j / n) + 0.1 * np.sin(2 * np.pi * 7 * j / n)

    # Evaluate and sort
    start_results = []
    for name, f0 in starts.items():
        f = np.maximum(f0, 0.0)
        s = f.sum()
        if s < 1e-10:
            f = np.ones(n)
        else:
            f = f / s
        sc = _score_exact(f)
        start_results.append((sc, name, f))

    start_results.sort(key=lambda x: x[0])

    print("\nTop starting points:")
    for sc, name, _ in start_results[:10]:
        print(f"  {name}: {sc:.7f}")

    if start_results[0][0] < best_score:
        best_score = start_results[0][0]
        best_a = start_results[0][2].copy()

    # === Phase 1: L-BFGS from top starting points ===
    print(f"\n=== Phase 1: L-BFGS (best={best_score:.7f}) ===")
    p_schedule = [4, 8, 16, 32, 64, 128]

    # Try top 8 starting points
    n_starts = min(8, len(start_results))
    time_per_start = (budget_s * 0.45) / n_starts

    for idx in range(n_starts):
        if time.monotonic() > deadline - 300:
            break
        sc_init, name, f0 = start_results[idx]
        t_alloc = min(time_per_start, deadline - time.monotonic() - 300)
        if t_alloc < 30:
            break

        print(f"  L-BFGS from '{name}' (init={sc_init:.6f}, budget={t_alloc:.0f}s)")
        f_out, sc_out = _lbfgs_optimize(f0, time.monotonic() + t_alloc,
                                         p_schedule, max_iter=3000,
                                         verbose=True, label=name)
        if sc_out < best_score:
            best_score = sc_out
            best_a = f_out.copy()
            print(f"    -> New best: {best_score:.7f}")

    # === Phase 2: Adam refinement ===
    print(f"\n=== Phase 2: Adam refinement (best={best_score:.7f}) ===")
    time_adam = min(300.0, (deadline - time.monotonic()) * 0.25)
    if time_adam > 60:
        # Run Adam from the current best
        f_out, sc_out = _adam_optimize(best_a, time.monotonic() + time_adam,
                                        p=50, lr_init=0.005,
                                        verbose=True, label="adam1")
        if sc_out < best_score:
            best_score = sc_out
            best_a = f_out.copy()
            print(f"    -> New best: {best_score:.7f}")

        # Also try from a perturbed version
        if time.monotonic() < deadline - 200:
            a_pert = best_a + rng.normal(0, 0.01, n) * best_a.mean()
            a_pert = np.maximum(a_pert, 0)
            a_pert = a_pert / a_pert.sum()
            t2 = min(150.0, deadline - time.monotonic() - 200)
            if t2 > 30:
                f_out2, sc_out2 = _adam_optimize(a_pert, time.monotonic() + t2,
                                                  p=50, lr_init=0.003,
                                                  verbose=True, label="adam2")
                if sc_out2 < best_score:
                    best_score = sc_out2
                    best_a = f_out2.copy()
                    print(f"    -> New best: {best_score:.7f}")

    # === Phase 3: Coordinate polish ===
    print(f"\n=== Phase 3: Coordinate polish (best={best_score:.7f}) ===")
    time_polish = min(180.0, (deadline - time.monotonic()) * 0.3)
    if time_polish > 30:
        f_out, sc_out = _coordinate_polish(best_a, time.monotonic() + time_polish,
                                           verbose=True, label="polish")
        if sc_out < best_score:
            best_score = sc_out
            best_a = f_out.copy()
            print(f"    -> New best: {best_score:.7f}")

    # === Phase 4: Fine L-BFGS with high p ===
    print(f"\n=== Phase 4: Fine L-BFGS (best={best_score:.7f}) ===")
    time_fine = min(max(60.0, deadline - time.monotonic() - 30), 200.0)
    if time_fine > 30:
        f_out, sc_out = _fine_tune(best_a, time.monotonic() + time_fine,
                                    verbose=True, label="fine")
        if sc_out < best_score:
            best_score = sc_out
            best_a = f_out.copy()
            print(f"    -> New best: {best_score:.7f}")

    # === Phase 5: Final micro-polish ===
    if time.monotonic() < deadline - 15:
        print(f"\n=== Phase 5: Final polish (best={best_score:.7f}) ===")
        t_micro = deadline - time.monotonic() - 5
        if t_micro > 20:
            f_out, sc_out = _coordinate_polish(best_a, time.monotonic() + t_micro,
                                               verbose=True, label="micro")
            if sc_out < best_score:
                best_score = sc_out
                best_a = f_out.copy()
                print(f"    -> New best: {best_score:.7f}")

    # Final validation
    best_a = np.clip(best_a, 0.0, 1000.0)
    final_score = evaluate_sequence([float(x) for x in best_a])
    print(f"\nOfficial score: {final_score:.7f}")

    if not np.isfinite(final_score) or final_score > best_score + 1e-6:
        # Fallback
        print(f"  Using internal score: {best_score:.7f}")

    print(f"\n=== FINAL: {best_score:.7f} (time: {time.monotonic()-t_start:.1f}s) ===")
    return [float(x) for x in best_a]
