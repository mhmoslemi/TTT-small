    n = len(f)
    f_rev = f[::-1]

    F = np.fft.rfft(f_rev, size)

    momentum = np.zeros(n)

    eta = 1.0  # step size

        grad = _gradient_fft(f, weights)

"""
Step function optimizer using FFT-based gradient descent with spectral equalization.
Key insight: minimize the peak-to-mean ratio of the linear convolution.
Uses mirror descent, diverse asymmetric initializations, and periodic spectral flattening.
"""

import numpy as np
import time

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

def _autocorr(f, size):
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * len(f) - 1]

def _conv_gradient(f, weights, size):
    n = len(f)
    f_rev = f[::-1]
    W = np.fft.rfft(weights, size)
    F = np.fft.rfft(f_rev, size)
    result = np.fft.irfft(W * F, size)
    return result[n-1:2*n-1]

def _score(f, size, n):
    c = _autocorr(f, size)
    s = f.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

def _normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return np.ones(len(f))
    return f / s

def _mirror_step(f, grad, eta):
    log_f = np.log(f + 1e-300) - eta * grad
    m = log_f.max()
    if not np.isfinite(m):
        return f.copy()
    log_f -= m
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-30:
        return f.copy()
    return f_new / s

def _spectral_equalize(f, alpha=0.5, iters=10):
    """Flatten the power spectrum of f."""
    n = len(f)
    size = _next_pow2(2 * n - 1)
    f_work = f.copy()
    
    for _ in range(iters):
        F = np.fft.rfft(f_work, size)
        mag = np.abs(F)
        target = np.sqrt(f_work.sum() ** 2 / (2 * n - 1) * size / size)
        # Target magnitude: want |F| to be roughly constant
        # sum |F|^2 = size * sum f^2 (by Parseval for rfft with padding)
        # If flat: |F|^2 = size * sum(f^2) / (size//2 + 1) ... this gets complicated
        # Simpler: target = geometric mean of current magnitudes
        positive_mag = mag[mag > 1e-10]
        if len(positive_mag) == 0:
            break
        target_mag = np.sqrt(np.sum(mag**2) / len(mag))  # RMS magnitude
        
        # Blend: new_mag = target * (mag/target)^alpha = mag^(1-alpha) * target^alpha
        new_mag = np.where(mag > 1e-10, 
                          target_mag * (mag / (target_mag + 1e-30)) ** alpha, 0)
        
        # Preserve phase
        new_F = new_mag * (F / (mag + 1e-30))
        f_new = np.fft.irfft(new_F, size)[:n]
        f_new = np.maximum(f_new, 0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_work = f_new / s
    
    return f_work

def _perturb(f, rng, mode=None):
    n = len(f)
    if mode is None:
        mode = int(rng.integers(0, 6))
    j = np.arange(n, dtype=np.float64)
    f_try = f.copy()
    
    if mode == 0:
        alpha = rng.uniform(-3, 3)
        f_try = f * np.exp(alpha * (j/n - 0.5))
    elif mode == 1:
        freq = int(rng.integers(1, max(2, n//5) + 1))
        phase = rng.uniform(0, 2*np.pi)
        amp = rng.uniform(0.2, 1.0)
        f_try = f * (1 + amp * np.sin(2*np.pi*freq*j/n + phase))
    elif mode == 2:
        # Blend with random
        t = rng.uniform(0.1, 0.5)
        fr = rng.exponential(1.0, size=n)
        fr = fr / fr.sum()
        f_try = (1-t)*f + t*fr
    elif mode == 3:
        # Random point modifications
        for _ in range(20):
            i = int(rng.integers(0, n))
            f_try[i] *= rng.uniform(0.3, 3.0)
    elif mode == 4:
        # Asymmetric window
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.2, 0.6)
        f_try = f * (1 + 0.5*np.exp(-((j/n - center)/width)**2))
    elif mode == 5:
        # Antisymmetric modulation (breaks symmetry)
        freq = int(rng.integers(1, max(2, n//10) + 1))
        f_try = f * (1 + 0.5 * np.sin(2*np.pi*freq*j/n))
    
    return _normalize(f_try)

def _optimize_grad(f0, deadline, rng, conv_size, grad_size, n, label=""):
    """Main optimization loop using gradient descent."""
    f = _normalize(f0.copy())
    best_score = _score(f, conv_size, n)
    best_f = f.copy()
    
    momentum = np.zeros(n)
    beta = 0.8
    eta = 1.0
    min_eta = 1e-5
    
    stuck = 0
    n_iter = 0
    
    while time.monotonic() < deadline:
        n_iter += 1
        c = _autocorr(f, conv_size)
        mean_c = 1.0 / (2 * n - 1)  # since sum(f) = 1
        
        # Weight excess peaks
        excess = np.maximum(c - mean_c, 0)
        total_excess = excess.sum()
        if total_excess < 1e-15:
            break
        weights = excess / total_excess
        
        # Gradient
        grad = _conv_gradient(f, weights, grad_size)
        
        # Momentum
        momentum = beta * momentum + (1 - beta) * grad
        
        # Adaptive step size
        span = momentum.max() - momentum.min()
        if span < 1e-30:
            eta *= 0.5
            if eta < min_eta:
                break
            continue
        
        eta_eff = eta / span
        
        # Mirror step
        f_new = _mirror_step(f, momentum, eta_eff)
        
        # Evaluate
        score = _score(f_new, conv_size, n)
        
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            f = f_new
            stuck = 0
            if stuck == 0 and n_iter % 1000 == 0:
                print(f"    {label} iter {n_iter}: {score:.7f}")
        else:
            stuck += 1
            eta *= 0.75
            if eta < min_eta:
                # Escalate: perturbation
                f = _perturb(f, rng)
                momentum *= 0.3
                eta = 0.5
                stuck = 0
                if best_score < np.inf:
                    # Re-anchor to best
                    t = rng.uniform(0.3, 0.7)
                    f = _normalize((1-t) * best_f + t * f)
        
        # Periodic spectral equalization
        if n_iter % 5000 == 0 and n_iter > 0:
            f_spec = _spectral_equalize(f, alpha=0.3, iters=5)
            score_spec = _score(f_spec, conv_size, n)
            if score_spec < best_score:
                best_score = score_spec
                best_f = f_spec.copy()
                f = f_spec
                momentum = np.zeros(n)
    
    return best_f, best_score

def _polish(f0, deadline, rng, conv_size, n):
    """Fine-tuning with smaller steps."""
    f = _normalize(f0.copy())
    best_score = _score(f, conv_size, n)
    best_f = f.copy()
    
    grad_size = _next_pow2(3 * n - 2)
    eta = 0.1
    n_iter = 0
    
    while time.monotonic() < deadline:
        n_iter += 1
        c = _autocorr(f, conv_size)
        mean_c = 1.0 / (2 * n - 1)
        
        # Target only the top peaks
        threshold = np.percentile(c, 99)
        excess = np.maximum(c - threshold, 0)
        if excess.sum() < 1e-15:
            break
        weights = excess / excess.sum()
        
        grad = _conv_gradient(f, weights, grad_size)
        
        span = grad.max() - grad.min()
        if span < 1e-30:
            break
        
        f_new = _mirror_step(f, grad, eta / span)
        score = _score(f_new, conv_size, n)
        
        if score < best_score:
            best_score = score
            best_f = f_new.copy()
            f = f_new
        else:
            eta *= 0.9
            if eta < 1e-6:
                break
    
    return best_f, best_score
