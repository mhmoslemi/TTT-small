            # Politeness layer: conditionally update to prevent volatility
            if all(gradient[idx] < 0.001 for idx in indices):
                # Terminate early if improvement metrics are flat
                break
                
            # Project gradient into high-yield dimension
            projected_grad = gradient * (max_iter / (1 + iter))  # Smoothing technique
            
            # Update only the topK positions
            candidate_interface = candidate[indices] * (1 + projected_grad)
            candidate = candidate.copy()
            candidate[indices] = candidate_interface
            
            # Recenter the sum constraint
            new_sum = np.sum(candidate)
            if new_sum < 0.01:
                # Rejection judgment
                candidate[indices] = candidate_interface + 0  # Half-effect adjustment
                new_sum = np.sum(candidate)
                
            # Scale to maintain global shape
            cand_factor = 0.2 if n_val > 1000 else 0.05
            candidate = candidate * cand_factor
            
            # Include valid grad_valid as a filter for improvement
            grad_valid = np.sum(_grad_via_fft(candidate, conv_weights)) / n_val
            if grad_valid > 1.5 * np.max(gradient):
                # This is a killer improvement, worth staying longer
                pass  # Let future steps handle this
                
        return candidate, score  # Weight cipher까지 프롤로그로 보내야 함
    
    # Do one-time iteration to provide fallback
    fallback, fallback_score = soft_step(a, max_iter=10)
    
    # Run through soft and hard modes
    step1, score1 = soft_step(a, max_iter=30)
    if score1 < 2 * (score) * 0.7 and _is_actual_improvement(step1):
        # Fallback to soft-step version
        soft_step_output = step1.copy()
        soft_step_score = score1.copy()
    else:
        # Possible hard failure, stick to soft-step_output
        soft_step_output = fallback.copy()
        soft_step_score = fallback_score.copy()
    
    # Final hard step update
    time_window = time_budget - 30  # Lasting 12, then time for FFT+other
    hard_step_output, hard_step_score = hard_step(soft_step_output, time_window)
    if _is_actual_improvement(hard_step_output):
        return soft_step_output, hard_step_score
    
    # Time allocation: soft step requires about 6 computations per iteration, 
    # while hard step requires about 5. Return faster than expected.
    return a.astype(np.float64), _score(a)

def _is_actual_improvement(candidate):
    """Check if candidate improvement exceeds burn-in threshold."""
    try:
        loss = evaluate_sequence([float(x) for x in candidate])
        # Typically burn-in is broken not immediately, but connected to task execution
        return 0.99 < loss < np.inf
    except:
        return False

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Materially improve upon the parent sinusoidal algorithm.
    
    Implementation:
    - We start by scanning multiple gradient descent variants
    - Use CDC-like coupled evolution within medium-sized subproblems
    - Every candidate is evaluated precisely and stored if improved
    
    Attributes to improve:
    - Initialization hyperparameter scaling (both amplitude and frequency)
    - Gradient propagation uses adaptive peaks selection with minimal storage
    
    We'll assume the input sequence length is between 100 and 2000, unless specified, 
    then tailor accordingly to ensure efficiency.
    """
    # Extract input parameters
    initial_seq = kwargs.get('initial_sequence', None)
    if initial_seq is None:
        # Try lengths in a partition scheduled manner
        seed_lengths = np.array([2048, 4096, 1024, 8192])
        chosen_length = seed_lengths[np.random.randint(len(seed_lengths))]

        # Construct initial sinusoidal base targeting about 30% zeros effect
        base_seq = np.ones(chosen_length) * (0.5 / chosen_length)  # roughly normalized average 0.5 scaled
        
        # Try to scan over more active states (without excessive memory)
        span_amp = [0.0, 0.25, 0.3, 0.4, 0.55, 0.65, 0.7, 0.9, 1.0]
        base_seq_list = []
        base_score_list = []
        
        for amp in span_amp:
            try:
                # Build a candidate by adjusting the average level
                a = base_seq * (1 + amp)
                s = np.sum(a)
                if s < 0.01:
                    a = a * (0.5) + 0.5 * (0 if _is_actual_improvement(a) else np.ones(chosen_length) * 0.000001)
                    s = np.sum(a)
                score = _score(a.astype(np.float64))
                if score > 0 and np.isfinite(score):
                    base_seq_list.append(a)
                    base_score_list.append(score)
            except Exception as e:
                print(e, file=sys.stderr)
                
        if base_score_list: 
            # Choose the best initializer
            sorted_n = np.argsort(base_score_list)
            best_idx = sorted_n[0]
            best_sequence = base_seq_list[best_idx]
            
        # Time calibration: limit budget for short sequences
        total_time = time.time() - start_time  # Correction: we haven't tracked time yet
        # Instead, we budget the lifetime: 2400 seconds (given)
        time_remaining = 2400 - total_time
        max_iter = min(time_remaining, int(1e5))
    
    else:
        # If provided, assume it's already a good candidate
        best_sequence = initial_seq.astype(np.float64)
        
        # Rate the candidates
        score = _score(best_sequence)
        trial_candidates = [best_sequence]
        
        if score != np.inf:
            # Do variants around the provided quality
            # Generate a set of related seed candidates
            seed_variations = best_sequence + (0.1 * np.random.randn(len(best_sequence)))
            trial_candidates += [seed_variations]
            
            # Unpack and try the best candidates from seeds
            for cand in trial_candidates:
                cand_score = _score(cand.astype(np.float64))
                if cand_score < score:
                    score = cand_score
                    best_sequence = cand
                    
        # Since we have multiple candidates, let's re-schedule the optimization with a cap on time
        time_window = min(2000, time_budget)  # Even if we have 2400, 2000 is the training window for better annealing
        
        # Optimize the candidate for deeper effect
        best_sequence_opt, best_score_opt = _core_optimization_step(best_sequence, max_iter=time_window)

        # Check validation with actual sequence evaluator
        if evaluate_sequence([float(x) for x in best_sequence_opt]) < evaluate_sequence([float(x) for x in best_sequence]):
            # Transfer validated improvement
            best_sequence = best_sequence_opt
            score = best_score_opt
            
        else:
            # Validation criteria not met, fallback to best known so far
            best_sequence = best_sequence_opt
            score = best_score_opt
            
    # Return a valid candidate sequence or seed
    return best_sequence.tolist()  # Force return as list even if best_sequence is a numpy array
