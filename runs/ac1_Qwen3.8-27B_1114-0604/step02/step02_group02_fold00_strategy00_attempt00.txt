def _autocorr_fft(f):
    """Linear autocorrelation of f with itself via FFT."""
    n = len(f)
    # Determine the smallest power of two >= (2*n - 1)
    size = 1 << (2*n - 1).bit_length()
    F = np.fft.rfft(f, size)
    result = np.fft.irfft(F * F, size)
    return result[:2*n - 1]

def _topk_grad(f, c, k):

    n_val = len(f)

            grad[idx] += np.max(c[lag-1:lag+1] / (n_val if lag == idx+1 else 1))
    grad /= max(1.0, k)
