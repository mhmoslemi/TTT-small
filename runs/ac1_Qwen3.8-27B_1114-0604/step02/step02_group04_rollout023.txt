    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    a = _normalize(a)
    n = a.size
    best_a = a.copy()
    best_score = _score(best_a)
    
    rng = np.random.RandomState(555)
    
    while time.monotonic() < deadline:
        c = _fft_convolve(a)
        c_max = c.max()
        c_mean = c.mean()
        
        # Find the peak location
        k_peak = c.argmax()
        
        # The peak c[k_peak] = sum_j a[j] * a[k_peak - j]
        # To reduce it, we should decrease the a[j] values that contribute most
        # The contribution of a[j] to c[k_peak] is a[j] * a[k_peak-j]
        # (with bounds: 0 <= j < n and 0 <= k_peak-j < n)
        
        j_min = max(0, k_peak - n + 1)
        j_max = min(n - 1, k_peak)
        
        if j_max < j_min:
            break
        
        # Compute contributions
        js = np.arange(j_min, j_max + 1)
        contribs = a[js] * a[k_peak - js]
        
        # Find the top contributors
        top_k = min(5, len(js))
        if top_k > 0:
            top_local = np.argsort(contribs)[-top_k:]
            top_js = js[top_local]
            
            # Try reducing the top contributor
            for j in top_js:
                for factor in [0.7, 0.5, 0.8, 0.3]:
                    a_test = a.copy()
                    reduction = a_test[j] * (1 - factor)
                    a_test[j] *= factor
                    # Distribute the reduction to other positions
                    # Add it to the position with the smallest value
                    other = np.argmin(a_test)
                    a_test[other] += reduction
                    a_test = _normalize(a_test)
                    sc = _score(a_test)
                    if sc < best_score:
                        best_score = sc
                        best_a = a_test.copy()
                        a = a_test.copy()
                        break
    
    return best_a, best_score

def _make_sine(n, amp, phase=0.5):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.sin(theta)) / n
    a = np.clip(a, 0.0, 1000.0)
    return _normalize(a)
