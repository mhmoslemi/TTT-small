Step function optimizer: multi-phase mirror descent with softmax lag selection,
symmetry projection, and adaptive step scheduling.
"""

import numpy as np
import time


def _autocorr_fft(f):
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _topk_grad(f, c, k):
    n = len(f)
    top_k = np.argsort(c)[-k:]
    grad = np.zeros(n)
    for idx in top_k:
        j_range = np.arange(n)
        m_j = idx - j_range
        valid = (m_j >= 0) & (m_j < n)
        grad[valid] += 2.0 * f[m_j[valid]]
    grad /= k
    return grad


def _mirror_update(f, grad, eta):
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    return f_new / s


def _generate_starts(f_init, n, rng):

    center = (n - 1) / 2.0

    for sigma_frac in [0.1, 0.15, 0.2, 0.3, 0.4, 0.5]:
        sigma = sigma_frac * n
        starts.append(np.exp(-0.5 * ((j - center) / sigma) ** 2))
    
    # Polynomial cap
    for p in [0.5, 1, 1.5, 2, 3, 5]:
        x = np.abs(j - center) / center
        starts.append(np.clip(1 - x ** 2, 0, None) ** p)
    
    # Sine modulation
    for freq in [1, 2, 3, 4, 6, 8]:
        for amp in [0.3, 0.5, 0.7, 0.9]:
            starts.append(1.0 + amp * np.sin(2 * np.pi * freq * j / n))

    for pos in [0.2, 0.25, 0.3, 0.35]:
        w = 0.05 * n
        starts.append(
            np.exp(-((j - pos * n) / w) ** 2) +
            np.exp(-((j - (1 - pos) * n) / w) ** 2) + 0.001
        )
    
    # U-shaped
    for alpha in [1, 2, 5, 10]:
        starts.append(1.0 + alpha * ((j - center) / center) ** 2)

            base[idx] += rng.exponential(1.0, n_members)
            starts.append(base)

"""
Step function optimizer using multi-phase mirror descent with adaptive lag selection,
symmetry projection, and gradient normalization by max-abs (not range) to avoid
instability near uniform solutions. Key improvements over previous: proper gradient
scaling, aggressive early focusing (k=1), periodic symmetry enforcement, and
better time allocation concentrated on fewer but deeper optimizations.
"""

import numpy as np
import time


def _autocorr_fft(f):
    """Linear autocorrelation of f with itself via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _topk_grad(f, c, k):
    """Compute average subgradient from top-k lag positions using exact formula.
    grad[j] = (1/k) * sum_{m in top_k} 2*f[m-j] for valid (m-j)."""
    n = len(f)
    top_k = np.argsort(c)[-k:]
    grad = np.zeros(n)
    for idx in top_k:
        j_range = np.arange(n)
        m_j = idx - j_range
        valid = (m_j >= 0) & (m_j < n)
        grad[valid] += 2.0 * f[m_j[valid]]
    grad /= k
    return grad


def _mirror_update(f, grad, eta):
    """Mirror descent (exponentiated gradient) step with max-abs gradient scaling."""
    g_max = np.max(np.abs(grad))
    if g_max < 1e-30:
        return f
    log_f = np.log(f + 1e-300) - (eta / g_max) * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    return f_new / s


def _generate_starts(f_init, n, rng):
    """Generate diverse starting points from various parametric families."""
    starts = [f_init.copy()]
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    # Gaussian bells
    for sigma_frac in [0.08, 0.12, 0.15, 0.2, 0.3, 0.5]:
        sigma = sigma_frac * n
        starts.append(np.exp(-0.5 * ((j - center) / sigma) ** 2))

    # Polynomial caps
    for p in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        x = np.abs(j - center) / center
        starts.append(np.clip(1.0 - x ** 2.0, 0, None) ** p)

    # Sine modulation
    for freq in [1, 2, 3, 4, 5, 7, 10]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            starts.append(1.0 + amp * np.sin(2.0 * np.pi * freq * j / n))

    # Bimodal
    for pos in [0.15, 0.2, 0.25, 0.3, 0.35]:
        w = 0.06 * n
        starts.append(
            np.exp(-((j - pos * n) / w) ** 2) +
            np.exp(-((j - (1.0 - pos) * n) / w) ** 2) + 0.001
        )

    # U-shaped (inverted)
    for alpha in [0.5, 1.0, 2.0, 5.0]:
        starts.append(1.0 + alpha * ((j - center) / center) ** 2)

    # Triangle
    starts.append(1.0 - np.abs(j - center) / (center + 1e-10))

    # Cosine taper (Hann-like)
    starts.append(0.5 * (1.0 - np.cos(2.0 * np.pi * j / n)))

    # Random diverse starts
    for _ in range(12):
        r = rng.random()
        if r < 0.25:
            starts.append(rng.exponential(1.0, n))
        elif r < 0.5:
            starts.append(np.abs(rng.standard_normal(n)) + 0.01)
        elif r < 0.7:
            base = np.ones(n) * 0.05
            n_mem = int(rng.integers(n // 5, n // 2))
            idx = rng.choice(n, n_mem, replace=False)
            base[idx] += rng.exponential(1.0, n_mem)
            starts.append(base)
        elif r < 0.85:
            starts.append(np.ones(n) + 0.3 * rng.standard_normal(n))
        else:
            # Power-law
            starts.append(1.0 / (1.0 + np.abs(j - center) / (n * 0.1)) ** 2)

    return starts


def _optimize(f0, deadline, rng, n):
    """Multi-phase mirror descent with adaptive k, symmetry projection, and SA acceptance."""
    f = f0.copy()
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f /= s

    c = _autocorr_fft(f)
    peak0 = c.max()
    best_peak = peak0
    best_f = f.copy()

    eta = 1.0
    T = peak0 * 0.08
    T_min = peak0 * 1e-12
    last_improve = time.monotonic()
    iter_count = 0
    sym_interval = 3000

    while time.monotonic() < deadline:
        c = _autocorr_fft(f)
        cur_peak = c.max()

        if cur_peak < best_peak - 1e-15:
            best_peak = cur_peak
            best_f = f.copy()
            last_improve = time.monotonic()

        stuck = time.monotonic() - last_improve
        progress = min(1.0, iter_count / 40000.0)

        # Phase-based k: start focused, grow broader
        if progress < 0.15:
            k = 1
        elif progress < 0.35:
            k = 2 + int(3 * (progress - 0.15) / 0.2)
        elif progress < 0.6:
            k = 5 + int(5 * (progress - 0.35) / 0.25)
        elif progress < 0.85:
            k = 10 + int(8 * (progress - 0.6) / 0.25)
        else:
            k = 18 + int(7 * min(1.0, (progress - 0.85) / 0.15))

        # Step size management
        if stuck > 45:
            eta = min(eta * 2.0, 8.0)
            T = peak0 * 0.15
            mag = min(0.04, 0.004 + 0.003 * stuck / 90.0)
            f = f + mag / np.sqrt(n) * rng.standard_normal(n)
            f = np.maximum(f, 0.0)
            s_f = f.sum()
            if s_f > 1e-10:
                f /= s_f
            last_improve = time.monotonic()
        elif stuck > 15:
            eta = min(eta * 1.3, 5.0)
            T = max(T, peak0 * 0.05)

        # Gentle eta decay
        eta *= max(0.9999, 1.0 - 0.8 / (1.0 + stuck * 2.0))

        # Compute gradient and take step
        grad = _topk_grad(f, c, k)
        f_new = _mirror_update(f, grad, eta)
        c_new = _autocorr_fft(f_new)
        new_peak = c_new.max()

        delta = new_peak - cur_peak
        if delta <= 0:
            f = f_new
            eta = min(eta * 1.02, 6.0)
        else:
            accept = False
            if T > T_min:
                log_u = np.log(max(rng.uniform(), 1e-300))
                if log_u < -delta / T:
                    accept = True
            if accept:
                f = f_new
            else:
                eta *= 0.92
            T = max(T_min, T * 0.997)

        # Periodic symmetry projection
        if iter_count > 0 and iter_count % sym_interval == 0:
            f_sym = 0.5 * (f + f[::-1])
            f_sym = np.maximum(f_sym, 0.0)
            s_sym = f_sym.sum()
            if s_sym > 1e-10:
                f_sym /= s_sym
                c_sym = _autocorr_fft(f_sym)
                if c_sym.max() < cur_peak * 1.001:
                    f = f_sym

        iter_count += 1

        if iter_count % 15000 == 0:
            print(f"    iter={iter_count}, peak={cur_peak:.10f}, best={best_peak:.10f}, "
                  f"eta={eta:.3f}, k={k}, stuck={stuck:.0f}s")

    c = _autocorr_fft(f)
    if c.max() < best_peak:
        best_peak = c.max()
        best_f = f.copy()

    return best_f, 2.0 * n * best_peak


def _polish(f0, deadline, rng, n):
    """Coordinate descent polish with structured pair moves."""
    f = f0.copy()
    f = np.maximum(f, 0.0)
    f /= f.sum()

    c = _autocorr_fft(f)
    best_peak = c.max()
    best_f = f.copy()
    passes = 0

    while time.monotonic() < deadline:
        c = _autocorr_fft(f)
        cur_peak = c.max()

        if cur_peak < best_peak - 1e-15:
            best_peak = cur_peak
            best_f = f.copy()

        # Gradient at top lags
        grad = _topk_grad(f, c, 3)

        # Try structured moves: high-grad donors to low-grad receivers
        high_coords = np.argsort(grad)[-15:]
        low_coords = np.argsort(grad)[:15]

        improved = False
        for frac in [0.003, 0.008, 0.02, 0.05, 0.1]:
            for d in high_coords[:8]:
                for rv in low_coords[:6]:
                    if d == rv:
                        continue
                    amt = f[d] * frac
                    if amt < 1e-16 or f[d] < 1e-12:
                        continue
                    f_try = f.copy()
                    f_try[d] -= amt
                    f_try[rv] += amt
                    f_try /= f_try.sum()
                    c_try = _autocorr_fft(f_try)
                    if c_try.max() < cur_peak - 1e-14:
                        f = f_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        # Random pair moves if no structured improvement
        if not improved:
            for _ in range(20):
                i = int(rng.integers(0, n))
                jj = int(rng.integers(0, n))
                if i == jj:
                    continue
                frac = rng.choice([0.005, 0.01, 0.03, 0.08])
                amt = f[i] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[i] -= amt
                f_try[jj] += amt
                if f_try[i] < 0:
                    continue
                f_try /= f_try.sum()
                c_try = _autocorr_fft(f_try)
                if c_try.max() < cur_peak - 1e-14:
                    f = f_try
                    improved = True
                    break

        # Small global perturbation
        if not improved and passes % 30 == 0:
            f_try = f + 0.001 / np.sqrt(n) * rng.standard_normal(n)
            f_try = np.maximum(f_try, 0.0)
            s_try = f_try.sum()
            if s_try > 1e-10:
                f_try /= s_try
                c_try = _autocorr_fft(f_try)
                if c_try.max() < cur_peak - 1e-14:
                    f = f_try

        passes += 1
        if passes % 1000 == 0:
            print(f"    polish pass={passes}, peak={cur_peak:.10f}, best={best_peak:.10f}")

    c = _autocorr_fft(f)
    if c.max() < best_peak:
        best_peak = c.max()
        best_f = f.copy()

    return best_f, 2.0 * n * best_peak
