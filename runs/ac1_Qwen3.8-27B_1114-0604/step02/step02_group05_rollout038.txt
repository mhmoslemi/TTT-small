```python
"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2
using frequency-domain convolution flattening combined with exact subgradient
descent and adaptive compression.
"""

import numpy as np
import time

def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(a, b, L=None):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)

def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())

def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s

def _freq_flatten_v2(p, alpha, L=None):
    """Frequency-domain convolution flattening with power compression."""
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(p, L)
    # C_freq = |A|^2
    C = np.fft.irfft(A * A, L)[:2 * n - 1]
    cmax = C.max()
    cmin = C.min()
    if cmax < 1e-15:
        return p.copy()
    # Power compression: C_new = cmax * (C/cmax)^alpha
    C_norm = C / cmax
    C_new = cmax * np.power(C_norm, alpha)
    C_new = np.maximum(C_new, 0.0)
    C_freq_new = np.fft.rfft(C_new, L)
    # Take square root (need to handle phase)
    A_new = np.sqrt(C_freq_new.astype(complex) + 0j)
    a_new = np.fft.irfft(A_new, L)[:n]
    neg_frac = (a_new < 0).sum() / n
    if neg_frac > 0.4:
        return None
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s

def _freq_flatten_rational(p, gamma, L=None):
    """Frequency-domain flattening with rational compression C/(1+gamma*C)."""
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(p, L)
    C = np.fft.irfft(A * A, L)[:2 * n - 1]
    cmax = C.max()
    if cmax < 1e-15:
        return p.copy()
    # Rational compression
    C_new = C / (1.0 + gamma * C)
    C_new = np.maximum(C_new, 0.0)
    C_freq_new = np.fft.rfft(C_new, L)
    A_new = np.sqrt(C_freq_new.astype(complex) + 0j)
    a_new = np.fft.irfft(A_new, L)[:n]
    neg_frac = (a_new < 0).sum() / n
    if neg_frac > 0.4:
        return None
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s

def _exact_subgradient_step(p, eta, L=None):
    """
    Exact subgradient step for minimizing max_k (a*a)[k].
    g[j] = mean over peaks k of a[k-j] (with zero-padding)
    Update: p_new = p * exp(-eta * g)
    """
    n = len(p)
    if L is None:
        L = _next_pow2(3 * n)
    # Compute full convolution
    c = _conv_full(p, p)
    cmax = c.max()
    if cmax < 1e-15:
        return p.copy(), float('inf')
    # Find peak locations (within tolerance)
    tol = 1e-6 * cmax
    peak_mask = c >= (cmax - tol)
    peak_indices = np.where(peak_mask)[0]
    
    # Compute subgradient: g[j] = (1/|K*|) * sum_{k in K*} a[max(0, k-j)] for valid indices
    # More efficiently: g[j] = (1/|K*|) * sum_{k in K*} p[k-j] where 0 <= k-j < n
    # This is a cross-correlation of p with the indicator of peak locations
    # g[j] = (1/|K*|) * sum_k I[k in K*] * p[k-j]
    # = (1/|K*|) * cross_correlation(I_peaks, p)[j]
    
    # Using FFT: cross_correlation(I, p) = IFFT(FFT(I) * conj(FFT(p)))
    I = np.zeros(2 * n - 1)
    I[peak_indices] = 1.0 / len(peak_indices)
    
    L_x = _next_pow2(3 * n)
    I_fft = np.fft.rfft(I, L_x)
    p_fft = np.fft.rfft(p, L_x)
    g = np.fft.irfft(I_fft * np.conj(p_fft), L_x)[:n]
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    
    # Normalize g to have max 1 for step size control
    gmax = g.max()
    if gmax < 1e-15:
        return p.copy(), float('inf')
    g_norm = g / gmax
    
    # Multiplicative update
    p_new = p * np.exp(-eta * g_norm)
    p_new = np.maximum(p_new, 1e-18 / n)
    s = p_new.sum()
    if s < 1e-10:
        return p.copy(), float('inf')
    p_new = p_new / s
    return p_new, _score(p_new)

def _soft_gradient_step(p, beta, eta, L=None):
    """
    Soft-max weighted gradient step.
    """
    n = len(p)
    if L is None:
        L = _next_pow2(3 * n)
    c = _conv_full(p, p)
    d = n * c  # This is proportional to the "contribution" of each convolution point
    # Soft weights
    shift = d.max()
    e = np.exp(beta * (d - shift))
    s_e = e.sum()
    if s_e < 1e-300 or not np.isfinite(s_e):
        w = np.ones(len(d)) / len(d)
    else:
        w = e / s_e
    
    # Gradient: h[j] = 2 * sum_m w[m] * p[m-j] (cross-correlation)
    L_x = _next_pow2(3 * n)
    W = np.fft.rfft(w, L_x)
    P = np.fft.rfft(p, L_x)
    h = 2.0 * np.fft.irfft(W * np.conj(P), L_x)[:n]
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.maximum(h, 0.0)
    hmax = h.max()
    if hmax < 1e-15:
        return p.copy(), float('inf')
    h_norm = h / hmax
    
    p_new = p * np.exp(-eta * h_norm)
    p_new = np.maximum(p_new, 1e-18 / n)
    s = p_new.sum()
    if s < 1e-10:
        return p.copy(), float('inf')
    p_new = p_new / s
    return p_new, _score(p_new)
```

```python
"""
Optimizes non-negative sequences to minimize 2*n*max(conv(a,a))/sum(a)^2
using a Fienup-style frequency-domain iterative algorithm with adaptive
compression, relaxation, and hybrid gradient refinement.

Key innovations over previous approach:
- Clean Fienup-style iteration preserving phase of the spectrum
- Adaptive compression exponent with both power and rational compressions
- Relaxation parameter for stable convergence
- Dual-branch complex square root selection
- Focused time allocation with progressive parameter tightening
"""

import numpy as np
import time


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b, L=None):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)


def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s


def _freq_flatten_step(p, beta, relaxation, L=None):
    """
    Fienup-style frequency-domain flattening step.
    beta in (0, 1): compression exponent. Lower = more aggressive flattening.
    relaxation in (0, 1]: step size / blending factor.
    """
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)

    c = _conv_full(p, p)
    cmax = c.max()
    if cmax < 1e-15:
        return p.copy()

    # Power compression: C_target = cmax * (c/cmax)^beta
    C_norm = c / cmax
    C_target = cmax * np.power(C_norm, beta)
    C_target = np.maximum(C_target, 0.0)

    # Frequency domain square root with branch selection
    C_target_freq = np.fft.rfft(C_target, L)
    A_new = np.sqrt(C_target_freq.astype(complex) + 0j)

    # Try both branches, pick the one with more non-negative values
    a_new1 = np.fft.irfft(A_new, L)[:n]
    a_new2 = -a_new1
    pos1 = np.sum(a_new1 >= 0)
    pos2 = np.sum(a_new2 >= 0)
    a_new = a_new1 if pos1 >= pos2 else a_new2

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return p.copy()
    a_new = a_new / s

    # Relaxation (blending)
    if relaxation < 1.0:
        p_new = (1.0 - relaxation) * p + relaxation * a_new
    else:
        p_new = a_new

    p_new = np.maximum(p_new, 0.0)
    s = p_new.sum()
    if s < 1e-10:
        return p.copy()
    return p_new / s


def _freq_rational_step(p, gamma, relaxation, L=None):
    """
    Frequency-domain flattening with rational compression C/(1+gamma*C).
    """
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)

    c = _conv_full(p, p)
    cmax = c.max()
    if cmax < 1e-15:
        return p.copy()

    # Rational compression
    C_target = c / (1.0 + gamma * c)
    C_target = np.maximum(C_target, 0.0)

    C_target_freq = np.fft.rfft(C_target, L)
    A_new = np.sqrt(C_target_freq.astype(complex) + 0j)

    a_new1 = np.fft.irfft(A_new, L)[:n]
    a_new2 = -a_new1
    pos1 = np.sum(a_new1 >= 0)
    pos2 = np.sum(a_new2 >= 0)
    a_new = a_new1 if pos1 >= pos2 else a_new2

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return p.copy()
    a_new = a_new / s

    if relaxation < 1.0:
        p_new = (1.0 - relaxation) * p + relaxation * a_new
    else:
        p_new = a_new

    p_new = np.maximum(p_new, 0.0)
    s = p_new.sum()
    if s < 1e-10:
        return p.copy()
    return p_new / s


def _gradient_step(p, eta, L=None):
    """
    Multiplicative gradient step using exact subgradient at max locations.
    """
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)

    c = _conv_full(p, p)
    cmax = c.max()
    if cmax < 1e-15:
        return p.copy(), float('inf')

    # Find peak locations
    tol = 1e-8 * cmax
    peak_mask = c >= (cmax - tol)
    n_peaks = np.sum(peak_mask)

    # Subgradient: g[j] = (1/n_peaks) * sum_{k: peak} p[k-j] (valid indices)
    # This is cross-correlation of p with indicator of peaks
    I = np.zeros(len(c))
    I[peak_mask] = 1.0 / n_peaks

    L_x = _next_pow2(2 * n + len(c))
    I_fft = np.fft.rfft(I, L_x)
    p_fft = np.fft.rfft(p, L_x)
    g = np.fft.irfft(I_fft * np.conj(p_fft), L_x)[:n]
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)

    gmax = g.max()
    if gmax < 1e-15:
        return p.copy(), float('inf')
    g_norm = g / gmax

    p_new = p * np.exp(-eta * g_norm)
    p_new = np.maximum(p_new, 1e-18 / n)
    s = p_new.sum()
    if s < 1e-10:
        return p.copy(), float('inf')
    p_new = p_new / s
    return p_new, _score(p_new)


def _soft_gradient_step(p, beta, eta, L=None):
    """
    Soft-max weighted gradient step for smoother optimization.
    """
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)

    c = _conv_full(p, p)
    shift = c.max()
    if shift < 1e-15:
        return p.copy(), float('inf')

    e = np.exp(-beta * (shift - c))
    s_e = e.sum()
    if s_e < 1e-300 or not np.isfinite(s_e):
        w = np.ones(len(c)) / len(c)
    else:
        w = e / s_e

    L_x = _next_pow2(2 * n + len(c))
    W = np.fft.rfft(w, L_x)
    P = np.fft.rfft(p, L_x)
    h = 2.0 * np.fft.irfft(W * np.conj(P), L_x)[:n]
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.maximum(h, 0.0)
    hmax = h.max()
    if hmax < 1e-15:
        return p.copy(), float('inf')
    h_norm = h / hmax

    p_new = p * np.exp(-eta * h_norm)
    p_new = np.maximum(p_new, 1e-18 / n)
    s = p_new.sum()
    if s < 1e-10:
        return p.copy(), float('inf')
    p_new = p_new / s
    return p_new, _score(p_new)


def _block_refine(p0, rng, time_budget):
    """Block-wise multiplicative refinement with exact scoring."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    best = p.copy()
    best_score = _score(p)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0
    factors = (0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0, 3.0)

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.15:
            bs = int(rng.integers(1, 8))
        elif r < 0.5:
            bs = int(rng.integers(max(2, n // 512), max(4, n // 64)))
        else:
            bs = int(rng.integers(max(4, n // 64), max(8, n // 16)))
        bs = min(bs, n)
        start = int(rng.integers(0, n - bs + 1))

        improved = False
        for f in factors:
            p_try = p.copy()
            p_try[start:start + bs] *= f
            p_try = _sanitize(p_try)
            if p_try is None:
                continue
            sc = _score(p_try)
            if sc < cur_score - 1e-13:
                cur_score = sc
                p = p_try
                improved = True
                break

        if improved:
            if cur_score < best_score - 1e-12:
                best = p.copy()
                best_score = cur_score
            stall = 0
        else:
            stall += 1
            if stall > 25:
                p = best.copy()
                cur_score = best_score
                stall = 0

    return best, best_score


def _init_two_level(n, rng):

def _init_gaussian_mixture(n, rng):
    K = int(rng.integers(2, 8))
    p = np.zeros(n, dtype=float)
    t = np.arange(n, dtype=float) / n
    for _ in range(K):
        c = rng.uniform(0.0, 1.0)
        s = rng.uniform(0.02, 0.3)
        w = rng.uniform(0.5, 2.0)
        x = (t - c) / s
        x = np.minimum(np.abs(x), 1.0 - np.abs(x))
        p += w * np.exp(-0.5 * x * x)
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _init_cosine(n, rng):

        ph = rng.uniform(0, 2 * np.pi)

def _resample(p_in, n_new, rng=None):
    p = _sanitize(p_in)
    if p is None:
        return None
    n_old = len(p)
    if n_old == n_new:
        result = p.copy()
    elif n_old == 1:
        result = np.ones(n_new, dtype=float)
    else:
        x_old = (np.arange(n_old) + 0.5) / n_old
        x_new = (np.arange(n_new) + 0.5) / n_new
        result = np.interp(x_new, x_old, p)
    result = np.maximum(result, 0)
    if rng is not None:
        result = result * np.exp(rng.normal(0, 0.02, n_new))
        result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s

        return None, float('inf')
    n = len(p)
    L = _next_pow2(2 * n - 1)

    best = p.copy()
    best_score = _score(p)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0
    it = 0

    if verbose:
        print(f"    Fienup opt: n={n}, init={best_score:.6f}, budget={time_budget:.0f}s")

    # Adaptive parameters
    beta = 0.5       # compression exponent (start moderate)
    relaxation = 0.5  # blending factor
    eta = 0.08       # gradient step size

    total_iters = 0
    while time.monotonic() < deadline:
        # Phase within the budget: early = explore, late = exploit
        frac_done = (time.monotonic() - (deadline - time_budget)) / time_budget
        # In early phase, use more aggressive flattening; late phase, gentler
        beta_range_early = (0.15, 0.6)
        beta_range_late = (0.4, 0.85)
        if frac_done < 0.3:
            beta_lo, beta_hi = beta_range_early
        elif frac_done < 0.7:
            beta_lo, beta_hi = (0.25, 0.75)
        else:
            beta_lo, beta_hi = beta_range_late

        r = rng.random()

        if r < 0.40:
            # Power compression step
            beta_try = float(rng.uniform(beta_lo, beta_hi))
            rel_try = float(rng.uniform(0.2, 0.7)) if frac_done < 0.5 else float(rng.uniform(0.1, 0.4))
            p_new = _freq_flatten_step(p, beta_try, rel_try, L)
            sc_new = _score(p_new)
            if sc_new < cur_score - 1e-14:
                p = p_new
                cur_score = sc_new
                stall = 0
                if cur_score < best_score - 1e-12:
                    best = p.copy()
                    best_score = cur_score
                    if verbose and total_iters % 200 == 0:
                        print(f"      [PWR] it={total_iters} beta={beta_try:.3f} rel={rel_try:.2f} score={best_score:.6f}")
            else:
                stall += 1

        elif r < 0.60:
            # Rational compression step
            c = _conv_full(p, p)
            cmax = c.max()
            gamma = float(rng.uniform(0.5 / cmax, 20.0 / cmax))
            rel_try = float(rng.uniform(0.2, 0.6))
            p_new = _freq_rational_step(p, gamma, rel_try, L)
            sc_new = _score(p_new)
            if sc_new < cur_score - 1e-14:
                p = p_new
                cur_score = sc_new
                stall = 0
                if cur_score < best_score - 1e-12:
                    best = p.copy()
                    best_score = cur_score
                    if verbose and total_iters % 200 == 0:
                        print(f"      [RAT] it={total_iters} gamma={gamma:.3f} score={best_score:.6f}")
            else:
                stall += 1

        elif r < 0.78:
            # Exact subgradient step
            best_trial = None
            best_trial_sc = cur_score
            for a_try in (eta * 0.5, eta, eta * 1.5, eta * 2.0):
                p_try, sc_try = _gradient_step(p, a_try, L)
                if sc_try < best_trial_sc - 1e-14:
                    best_trial = p_try
                    best_trial_sc = sc_try

            if best_trial is not None:
                p = best_trial
                cur_score = best_trial_sc
                stall = 0
                if cur_score < best_score - 1e-12:
                    best = p.copy()
                    best_score = cur_score
                    if verbose and total_iters % 200 == 0:
                        print(f"      [GRAD] it={total_iters} eta={eta:.4f} score={best_score:.6f}")
            else:
                stall += 1
                if stall > 8:
                    eta = max(0.01, eta * 0.85)

        elif r < 0.88:
            # Soft gradient step
            beta_soft = 30.0 + 200.0 * frac_done
            best_trial = None
            best_trial_sc = cur_score
            for a_try in (eta * 0.5, eta, eta * 1.5):
                p_try, sc_try = _soft_gradient_step(p, beta_soft, a_try, L)
                if sc_try < best_trial_sc - 1e-14:
                    best_trial = p_try
                    best_trial_sc = sc_try

            if best_trial is not None:
                p = best_trial
                cur_score = best_trial_sc
                stall = 0
                if cur_score < best_score - 1e-12:
                    best = p.copy()
                    best_score = cur_score
            else:
                stall += 1

        else:
            # Perturbation / block scaling (escape local minima)
            if stall > 5:
                rs = rng.random()
                if rs < 0.5:
                    # Block scaling
                    bs = int(rng.choice([2, 4, 8, 16, 32, 64, 128]))
                    bs = min(bs, n)
                    start = int(rng.integers(0, n - bs + 1))
                    f = float(rng.choice([0.3, 0.5, 0.7, 1.4, 1.8, 2.5, 4.0]))
                    p_try = p.copy()
                    p_try[start:start + bs] *= f
                    p_try = _sanitize(p_try)
                else:
                    # Global perturbation
                    std = 0.02 + 0.01 * min(stall, 20)
                    p_try = p * np.exp(rng.normal(0.0, std, n))
                    p_try = _sanitize(p_try)

                if p_try is not None:
                    sc_try = _score(p_try)
                    if sc_try < best_score * 1.05:
                        p = p_try
                        cur_score = sc_try
                        stall = 0
                        if sc_try < best_score - 1e-12:
                            best = p.copy()
                            best_score = sc_try

        # Adaptive step size increase when making progress
        if stall == 0 and it % 30 == 0:
            eta = min(0.3, eta * 1.03)

        # Pullback if diverging
        if cur_score > best_score * 1.08 and best_score < float('inf'):
            if rng.random() < 0.6:
                p = best.copy()
                cur_score = best_score
                stall = 0

        it += 1
        total_iters += 1

    if verbose:
        print(f"    Fienup done: {it} iters, best={best_score:.6f}")

    return best, best_score


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing the evaluation score.
    Uses Fienup-style frequency-domain iteration with adaptive compression,
    exact subgradient descent, and block refinement.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 2400.0

    start_time = time.monotonic()
    deadline = start_time + budget - 10

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng(iseed % (2**31 - 1))

    # Get parent
    try:
        parent = _sanitize(np.asarray(height_sequence_1, dtype=float))
    except Exception:
        parent = None

    best_overall = None
    best_score_overall = float('inf')

    def _try_update(p, score):
        nonlocal best_overall, best_score_overall
        if p is not None and np.isfinite(score) and score < best_score_overall:
            best_overall = p.copy()
            best_score_overall = float(score)
            print(f"  *** NEW BEST: {score:.6f} (n={len(p)}) ***")

    # Evaluate parent
    if parent is not None:
        sc = _score(parent)
        _try_update(parent, sc)
        print(f"  Parent: score={sc:.6f} (n={len(parent)})")
    else:
        parent = np.ones(3072, dtype=float) / 3072
        sc = _score(parent)
        _try_update(parent, sc)
        print(f"  No parent, using uniform n=3072: score={sc:.6f}")

    n_parent = len(parent)

    # ========== PHASE 1: Quick exploration (10%) ==========
    t1_end = start_time + 0.10 * budget
    print(f"\nPhase 1: Quick exploration")

    # Quick freq flattening on parent to get a better starting point
    L_p = _next_pow2(2 * n_parent - 1)
    p_quick = _freq_flatten_step(parent, 0.3, 0.6, L_p)
    sc_q = _score(p_quick)
    _try_update(p_quick, sc_q)

    p_quick2 = _freq_flatten_step(p_quick, 0.4, 0.5, L_p)
    sc_q2 = _score(p_quick2)
    _try_update(p_quick2, sc_q2)

    p_quick3 = _freq_flatten_step(p_quick2, 0.5, 0.4, L_p)
    sc_q3 = _score(p_quick3)
    _try_update(p_quick3, sc_q3)

    # Try other initializations
    inits = [
        _init_two_level(n_parent, rng),
        _init_gaussian_mixture(n_parent, rng),
        _init_cosine(n_parent, rng),
    ]
    for sp in inits:
        if time.monotonic() > t1_end:
            break
        sc = _score(sp)
        _try_update(sp, sc)

    print(f"  Phase 1 best: {best_score_overall:.6f}")

    # ========== PHASE 2: Main Fienup optimization (50%) ==========
    print(f"\nPhase 2: Main optimization")
    t2_end = start_time + 0.60 * budget

    if best_overall is None:
        best_overall = parent
        best_score_overall = _score(parent)

    # Run 1: Fienup on best
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(0.40 * t_rem, 400.0)
        t_alloc = max(20.0, t_alloc)
        print(f"  Run 1 (Fienup): t={t_alloc:.0f}s")
        p, sc = _optimize_fienup(best_overall, rng, t_alloc, verbose=True)
        _try_update(p, sc)

    # Run 2: Try upsampling to 2x
    t_rem = t2_end - time.monotonic()
    if t_rem > 60 and len(best_overall) * 2 <= 8192:
        n_up = len(best_overall) * 2
        t_alloc = min(0.30 * t_rem, 300.0)
        t_alloc = max(20.0, t_alloc)
        print(f"  Run 2 (Upsample to {n_up}): t={t_alloc:.0f}s")
        p_up = _resample(best_overall, n_up, rng)
        if p_up is not None:
            p, sc = _optimize_fienup(p_up, rng, t_alloc, verbose=True)
            _try_update(p, sc)

    # Run 3: Another Fienup pass (different trajectory)
    t_rem = t2_end - time.monotonic()
    if t_rem > 60:
        t_alloc = min(0.30 * t_rem, 300.0)
        t_alloc = max(20.0, t_alloc)
        print(f"  Run 3 (Fienup again): t={t_alloc:.0f}s")
        p2 = _resample(best_overall, len(best_overall), rng)
        if p2 is None:
            p2 = best_overall.copy()
        p, sc = _optimize_fienup(p2, rng, t_alloc, verbose=True)
        _try_update(p, sc)

    print(f"  Phase 2 best: {best_score_overall:.6f} (n={len(best_overall)})")

    # ========== PHASE 3: Block refinement (20%) ==========
    print(f"\nPhase 3: Block refinement")
    t3_end = start_time + 0.80 * budget

    t_rem = t3_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(0.7 * t_rem, 300.0)
        t_alloc = max(15.0, t_alloc)
        p, sc = _block_refine(best_overall, rng, t_alloc)
        _try_update(p, sc)

    print(f"  Phase 3 best: {best_score_overall:.6f}")

    # ========== PHASE 4: Final polishing (10%) ==========
    print(f"\nPhase 4: Final polishing")

    if best_overall is not None:
        # Final Fienup with gentler parameters
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.5 * t_rem, 200.0)
            p, sc = _optimize_fienup(best_overall, rng, t_alloc, verbose=True)
            _try_update(p, sc)

        # Final block refine
        t_rem = deadline - time.monotonic()
        if t_rem > 15:
            t_alloc = min(0.4 * t_rem, 100.0)
            p, sc = _block_refine(best_overall, rng, t_alloc)
            _try_update(p, sc)

    # ========== Final output ==========
    if best_overall is None:
        best_overall = parent

    p_final = _sanitize(best_overall)
    if p_final is None:
        p_final = np.ones(3072, dtype=float) / 3072

    final_score = _score(p_final)
    try:
        eval_score = float(evaluate_sequence([float(x) for x in p_final]))
        if np.isfinite(eval_score):
            final_score = min(final_score, eval_score)
    except Exception:
        pass

    print(f"\n{'=' * 60}")
    print(f"FINAL: score={final_score:.6f} (n={len(p_final)})")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    out = np.where(np.isfinite(p_final), p_final, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = np.ones(3072, dtype=float) / 3072
    else:
        out = out / s

    return [float(x) for x in out]
```
