```python
"""
Search for non-negative step function sequences minimizing

    2*n*max(autoconvolution) / (sum(a))^2

by combining:
  1. Fast structured initializations (Fourier/exponential/sinusoidal, resampling,
     low-dimensional Fourier L-BFGS tracks).
  2. Exponentiated-gradient / mirror-descent optimization with adaptive
     peak-softmax weights and robust lineless adaptive step.
  3. Log-domain Euclidean gradient refinement as an alternative trajectory.
  4. A final LP-direction polish inspired by the previous state of the art.
  5. Optional L-BFGS active-weight polishing on weighted peak sums.

All heavy convolution/gradient computations use FFT. The returned sequence is
non-negative and normalized to sum 1 (scale does not affect the objective).
"""

import numpy as np
import time

try:
    from scipy import optimize as _sp_opt
    from scipy.optimize import minimize as _sp_minimize
    _HAVE_SCIPY = True
except Exception:
    _HAVE_SCIPY = False

FLOOR = 1e-300


def _next_pow2(m):
    n = 1
    while n < int(m):
        n <<= 1
    return n


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([], dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    a = np.where(np.isfinite(a), a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-12:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    return a / s


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    c = c[:2 * n - 1]
    # The exact autoconvolution of a real sequence is symmetric.
    c = 0.5 * (c + c[::-1])
    c = np.maximum(c, 0.0)
    return c


def _score_fast(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    S = float(a.sum())
    if not np.isfinite(S) or S < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * a.size * mx / (S * S))


def _softmax_vec(x):
    x = np.asarray(x, dtype=np.float64)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, -700.0, 700.0)
    x = x - float(np.max(x))
    ex = np.exp(x)
    s = float(ex.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(x.size, 1.0 / float(max(1, x.size)), dtype=np.float64)
    return ex / s


def _softmax_x0(x, n):
    """Softmax of [0] + x, with x length n-1."""
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size != n - 1:
        tmp = np.zeros(n - 1, dtype=np.float64)
        m = min(x.size, n - 1)
        tmp[:m] = x[:m]
        x = tmp
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.clip(x, -700.0, 700.0)
    max_full = max(0.0, float(np.max(x))) if x.size else 0.0
    e0 = float(np.exp(-max_full))
    ej = np.exp(x - max_full)
    denom = e0 + float(ej.sum())
    if not np.isfinite(denom) or denom <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    a = np.empty(n, dtype=np.float64)
    a[0] = e0 / denom
    a[1:] = ej / denom
    return a


def _log_rats_from_x0(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, FLOOR)
    x0 = float(np.log(a[0]))
    x = np.log(a[1:]) - x0
    return np.clip(x, -50.0, 50.0)


def _grad_from_weights(a, w):
    """
    Gradient of F(a)=sum_k w_k c_k where c=autoconvolution(a).
    Returns dF/da of length n.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    m = 2 * n - 1
    if w.size < m:
        w = np.pad(w, (0, m - w.size), mode="constant")
    else:
        w = w[:m]
    wr = w[::-1]
    N = _next_pow2(n + wr.size - 1)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(wr, N)
    conv = np.fft.irfft(fa * fw, N)
    start = n - 1
    end = 2 * n - 1
    h = conv[start:end]
    if h.size < n:
        h = np.pad(h, (0, n - h.size), mode="constant")
    h = h[:n][::-1]
    return 2.0 * h


def _softmax_coeff_frac(c, tau):
    """Softmax with temperature tau*max(c)."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return np.array([], dtype=np.float64)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0.0:
        return np.full(c.size, 1.0 / c.size, dtype=np.float64)
    tau = float(max(1e-8, tau))
    z = (c - M) / (tau * M)
    z = np.maximum(z, -70.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros(c.size, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    return ez / s


def _softmax_beta(c, beta):
    """Softmax(beta*c), stable."""
    c = np.asarray(c, dtype=np.float64)
    if c.size == 0:
        return np.array([], dtype=np.float64)
    beta = float(max(1e-12, beta))
    z = beta * c
    z = z - float(np.max(z))
    z = np.maximum(z, -70.0)
    ez = np.exp(z)
    s = float(ez.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros(c.size, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    return ez / s


def _make_weights(c, broad_tau, sharp_tau, mix):
    c = np.asarray(c, dtype=np.float64)
    w_b = _softmax_coeff_frac(c, broad_tau)
    if mix >= 1.0:
        return _softmax_coeff_frac(c, sharp_tau)
    if mix <= 0.0:
        return w_b
    w_s = _softmax_coeff_frac(c, sharp_tau)
    w = (1.0 - mix) * w_b + mix * w_s
    s = float(w.sum())
    if not np.isfinite(s) or s <= 0.0:
        return w_b
    return w / s


def _eg_optimize(a, deadline, seed=0):
    """
    Exponentiated gradient / mirror descent using adaptive peak weights.
    """
    a = _normalize(a)
    n = a.size
    if n == 0:
        return a, np.inf
    t0 = time.monotonic()
    total = max(1e-6, deadline - t0)

    best = a.copy()
    best_sc = _score_fast(best)
    if not np.isfinite(best_sc):
        return best, best_sc
```

```python
Search for non-negative step function sequences minimizing

using:

def _next_pow2(m):
    n = 1
    while n < int(m):

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:

def _fft_convolve(a):

def _softmax_x0(x, n):
    """Softmax of [0] + x, with x length n-1."""
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size != n - 1:
        tmp = np.zeros(n - 1, dtype=np.float64)

def _grad_from_weights(a, w):
    """
    Gradient of F(a) = sum_k w_k c_k where c = autoconvolution(a).
    Returns dF/da of length n.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    m = 2 * n - 1
    if w.size < m:
        w = np.pad(w, (0, m - w.size), mode="constant")
    else:
        w = w[:m]
    wr = w[::-1]
    N = _next_pow2(n + wr.size - 1)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(wr, N)
    conv = np.fft.irfft(fa * fw, N)
    start = n - 1
    end = 2 * n - 1
    h = conv[start:end]
    if h.size < n:

def _fourier_basis(n, D, seed):

                args=(w,),
                method="L-BFGS-B",
                jac=_full_jac,
                bounds=bounds,
                options={
                    "maxfun": maxfun,
                    "maxiter": maxfun,
                    "ftol": 1e-13,
                    "gtol": 1e-9,
                    "disp": False,
                },
            )
            a_new = _softmax_x0(np.asarray(res.x, dtype=np.float64), n)
            a_new = _normalize(a_new)
            sc = _score_fast(a_new)
            if np.isfinite(sc) and sc < best_sc - 1e-12:
                best = a_new
                best_sc = sc
        except Exception:
            pass

    return best, best_sc


def _lp_direction(a, cap=5.0):
    """
    LP direction from the previous state of the art:
      maximize sum(y) s.t. conv(a,y) <= max(conv(a,a)), y>=0.
    Returns (y_normalized_or_None, sum_y_before_normalize).
    """
    if not _HAVE_SCIPY:
        return None, None
    a = _normalize(a)
    n = a.size
    if n == 0 or n > 1536:
        return None, None

    c = _fft_convolve(a)
    M = float(np.max(c))
    if not np.isfinite(M) or M <= 0.0:
        return None, None

    # Build dense A for n up to 1536; HIGHS handles it well.
    scale = float(n)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    ascale = a * scale
    for j in range(n):
        A[j:j + n, j] = ascale
    b = np.full(2 * n - 1, M * scale, dtype=np.float64)
    c_obj = np.ones(n, dtype=np.float64)
    bounds = [(0.0, float(cap))] * n

    kwargs = {
        "c": c_obj,
        "A_ub": A,
        "b_ub": b,
        "bounds": bounds,
        "options": {"presolve": True, "disp": False},
    }
    try:
        res = _sp_opt.linprog(method="highs", **kwargs)
        if res.status != 0 or res.x is None:
            return None, None
        y = np.asarray(res.x, dtype=np.float64)
        y = np.maximum(y, 0.0)
        y = np.where(np.isfinite(y), y, 0.0)
        sy = float(y.sum())
        if sy <= 1.0:
            return None, sy
        y = y / sy
        return y, sy
    except Exception:
        # Fallback if HiGHS unavailable in this SciPy build.
        try:
            res = _sp_opt.linprog(method="interior-point", **kwargs)
            if res.status != 0 or res.x is None:
                return None, None
            y = np.asarray(res.x, dtype=np.float64)
            y = np.maximum(y, 0.0)
            y = np.where(np.isfinite(y), y, 0.0)
            sy = float(y.sum())
            if sy <= 1.0:
                return None, sy
            y = y / sy
            return y, sy
        except Exception:
            return None, None


def _try_lp_blend(a, best_sc, deadline=None):
    a = _normalize(a)
    y, _ = _lp_direction(a, cap=5.0)
    if y is None:
        return a, best_sc
    best = a.copy()
    best_s = best_sc
    for t in (0.10, 0.25, 0.45, 0.65, 0.80, 0.92, 0.98):
        b = (1.0 - t) * a + t * y
        b = _normalize(b)
        sc = _score_fast(b)
        if np.isfinite(sc) and sc < best_s - 1e-12:
            best_s = sc
            best = b.copy()
        if deadline is not None and time.monotonic() > deadline:
            break
    return best, best_s


def _to_eval_list(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    a = np.where(np.isfinite(a), a, 0.0)
    return [float(x) for x in a]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t0 = time.monotonic()
    budget = float(max(120.0, float(budget_s)))
    hard_deadline = t0 + max(60.0, budget - 35.0)

    np.random.seed(seed)
    rng = np.random.RandomState(seed + 101)

    print("[init] budget=%.0fs hard_deadline in %.0fs" % (budget, hard_deadline - t0), flush=True)

    uid = 0
    raw = []

    # Previous best construction, if available.
    prev_raw = globals().get("height_sequence_1", None)
    prev_arr = None
    if prev_raw is not None:
        try:
            arr = np.asarray(prev_raw, dtype=np.float64).ravel()
            arr = np.clip(arr, 0.0, 1000.0)
            arr = np.where(np.isfinite(arr), arr, 0.0)
            if arr.size > 0 and arr.sum() > 0.01:
                prev_arr = _normalize(arr)
                sc = _score_fast(prev_arr)
                raw.append((sc, uid, "prev", prev_arr, prev_arr.size))
                uid += 1
                print("[init] previous sequence n=%d score=%.8f" % (prev_arr.size, sc), flush=True)
        except Exception:
            prev_arr = None

    lengths = [1024, 2048]
    if budget > 1500:
        lengths.append(4096)

    gen_deadline = t0 + min(180.0, 0.09 * budget)

    for n in lengths:
        if time.monotonic() > gen_deadline:
            break

        # Exponential sinusoids.
        for amp in (0.45, 0.75, 1.05, 1.45):
            a = _make_sine_base(n, amp, 1, 0.0, True)
            sc = _score_fast(a)
            if np.isfinite(sc):
                raw.append((sc, uid, "sine_exp", a, a.size))
                uid += 1

        # Bounded sine offsets.
        for amp in (0.45, 0.75):
            a = _make_sine_base(n, amp, 1, 0.0, False)
            sc = _score_fast(a)
            if np.isfinite(sc):
                raw.append((sc, uid, "sine_lin", a, a.size))
                uid += 1

        # Multi-mode exponentials.
        for _ in range(4):
            coeffs = [rng.uniform(0.35, 1.15), rng.uniform(-0.75, 0.75), rng.uniform(-0.45, 0.45)]
            modes = [1, 2, 4]
            phases = [rng.uniform(0.0, 1.0), rng.uniform(0.0, 1.0), rng.uniform(0.0, 1.0)]
            gamma = rng.uniform(0.6, 1.2)
            a = _make_multi_sine(n, coeffs, modes, phases, gamma)
            sc = _score_fast(a)
            if np.isfinite(sc):
                raw.append((sc, uid, "multi", a, a.size))
                uid += 1

        # Power-sine windows.
        for p in (0.65, 1.0, 1.8):
            a = _make_power_sine(n, p)
            sc = _score_fast(a)
            if np.isfinite(sc):
                raw.append((sc, uid, "powersine", a, a.size))
                uid += 1

        # Random low-frequency smooth.
        for _ in range(2):
            a = _random_smooth(n, rng, lowfreq=rng.randint(8, 28), gamma=rng.uniform(0.6, 1.4))
            sc = _score_fast(a)
            if np.isfinite(sc):
                raw.append((sc, uid, "randsmooth", a, a.size))
                uid += 1

        # Resample previous if possible.
        if prev_arr is not None and prev_arr.size != n:
            a = _resample_sequence(prev_arr, n)
            sc = _score_fast(a)
            if np.isfinite(sc):
                raw.append((sc, uid, "prev_resample", a, a.size))
                uid += 1

    # Low-dimensional Fourier L-BFGS tracks for better basins.
    for n in [1024]:
        if time.monotonic() < gen_deadline - 15.0:
            fd = min(gen_deadline - 3.0, time.monotonic() + 70.0)
            tracks = _run_fourier_tracks(n, fd, seed + 777, max_tries=3)
            tracks.sort()
            for sc, a in tracks[:2]:
                raw.append((sc, uid, "fourier", a, a.size))
                uid += 1

    if not raw:
        a = _make_sine_base(1024, 0.9, 1, 0.0, True)
        raw.append((_score_fast(a), uid, "fallback", a, a.size))
        uid += 1

    raw.sort()

    selected = []
    len_count = {}
    for item in raw:
        n = item[4]
        used = len_count.get(n, 0)
        if used >= 2:
            continue
        selected.append(item)
        len_count[n] = used + 1
        if len(selected) >= 5:
            break

    opt_start = time.monotonic()
    final_reserve = 300.0 if budget > 1500 else 180.0
    opt_end = hard_deadline - final_reserve
    if opt_end < opt_start + 80.0:
        opt_end = hard_deadline - 60.0

    opt_items = []
    shares = [0.42, 0.25, 0.18, 0.15, 0.10]

    for idx, item in enumerate(selected):
        if time.monotonic() > opt_end - 25.0:
            break
        sc0, id0, name0, a0, n0 = item
        frac = shares[idx] if idx < len(shares) else 0.10
        end = opt_start + frac * (opt_end - opt_start)
        if end <= time.monotonic() + 30.0:
            end = opt_end - 10.0
        if end <= time.monotonic() + 20.0:
            break

        best_here = a0.copy()
        best_sc_here = sc0
        print("[optimize] candidate %s n=%d init=%.8f block_deadline=%.0fs" %
              (name0, n0, best_sc_here, end - time.monotonic()), flush=True)

        # Phase 1: EG.
        try:
            eg_deadline = end - max(25.0, 0.35 * (end - time.monotonic()))
            if time.monotonic() < eg_deadline - 15.0:
                a1, s1 = _eg_optimize(best_here, eg_deadline, seed + 100 * idx)
                if s1 < best_sc_here:
                    best_here, best_sc_here = a1, s1
                else:
                    a1, s1 = best_here, best_sc_here
            else:
                a1, s1 = best_here, best_sc_here
        except Exception:
            a1, s1 = best_here, best_sc_here

        # Phase 2: log-GD alternative.
        try:
            if time.monotonic() < end - 40.0:
                a2, s2 = _log_gd_optimize(best_here, end - 35.0, seed + 500 + 100 * idx)
                if s2 < best_sc_here:
                    best_here, best_sc_here = a2, s2
                    a1, s1 = best_here, best_sc_here
        except Exception:
            pass

        # Phase 3: resample to double length if promising and within limit.
        try:
            if a1.size <= 2048 and time.monotonic() < end - 120.0:
                double_n = 2 * a1.size
                if double_n <= 4096:
                    a3 = _resample_sequence(best_here, double_n)
                    s3 = _score_fast(a3)
                    if np.isfinite(s3) and s3 < best_sc_here + 0.035:
                        a4, s4 = _eg_optimize(a3, end - 35.0, seed + 900 + 100 * idx)
                        if s4 < best_sc_here:
                            best_here, best_sc_here = a4, s4
        except Exception:
            pass

        opt_items.append((best_sc_here, uid, "opt_" + name0, best_here, best_here.size))
        uid += 1
        print("[optimize] done candidate %s best=%.8f" % (name0, best_sc_here), flush=True)

    all_candidates = []
    for item in raw[:4]:
        all_candidates.append(item)
    for item in opt_items:
        all_candidates.append(item)
    all_candidates.sort()

    if not all_candidates:
        a = _make_sine_base(1024, 0.9, 1, 0.0, True)
        return _to_eval_list(a)

    best_a = np.asarray(all_candidates[0][3], dtype=np.float64).copy()
    best_sc = float(all_candidates[0][0])
    print("[select] best pre-final n=%d fast=%.8f" % (best_a.size, best_sc), flush=True)

    # Final refinement: LP direction, EG, active L-BFGS, EG.
    remaining_final = max(0.0, hard_deadline - time.monotonic())

    try:
        if remaining_final > 220.0 and best_a.size <= 1536:
            lp_deadline = time.monotonic() + min(80.0, remaining_final * 0.25)
            a_lp, s_lp = _try_lp_blend(best_a, best_sc, deadline=lp_deadline)
            if s_lp < best_sc:
                best_a, best_sc = a_lp, s_lp
                all_candidates.append((best_sc, uid, "final_lp", best_a, best_a.size))
                uid += 1
                print("[final] LP blend improved to %.8f" % best_sc, flush=True)
    except Exception:
        pass

    try:
        seg_end = time.monotonic() + min(120.0, max(0.0, hard_deadline - time.monotonic()) * 0.45)
        if seg_end > time.monotonic() + 25.0:
            a_eg, s_eg = _eg_optimize(best_a, seg_end, seed + 2021)
            if s_eg < best_sc:
                best_a, best_sc = a_eg, s_eg
                all_candidates.append((best_sc, uid, "final_eg1", best_a, best_a.size))
                uid += 1
                print("[final] EG improved to %.8f" % best_sc, flush=True)
    except Exception:
        pass

    try:
        if _HAVE_SCIPY and best_a.size <= 2048 and time.monotonic() < hard_deadline - 90.0:
            lb_deadline = hard_deadline - 70.0
            a_lb, s_lb = _active_lbfgs_optimize(best_a, lb_deadline, seed + 3031)
            if s_lb < best_sc:
                best_a, best_sc = a_lb, s_lb
                all_candidates.append((best_sc, uid, "final_lbfgs", best_a, best_a.size))
                uid += 1
                print("[final] active L-BFGS improved to %.8f" % best_sc, flush=True)
    except Exception:
        pass

    try:
        last_deadline = hard_deadline - 15.0
        if time.monotonic() < last_deadline - 25.0:
            a_last, s_last = _eg_optimize(best_a, last_deadline, seed + 4041)
            if s_last < best_sc:
                best_a, best_sc = a_last, s_last
                all_candidates.append((best_sc, uid, "final_eg2", best_a, best_a.size))
                uid += 1
                print("[final] last EG improved to %.8f" % best_sc, flush=True)
    except Exception:
        pass

    all_candidates.sort()

    # Validate with the true evaluator when available.
    best_out = None
    best_actual = np.inf
    seen_ids = set()
    for item in all_candidates[:10]:
        if time.monotonic() > hard_deadline - 5.0:
            break
        key = (float(item[0]), int(item[1]))
        if key in seen_ids:
            continue
        seen_ids.add(key)
        arr = np.asarray(item[3], dtype=np.float64)
        lst = _to_eval_list(arr)
        try:
            val = float(evaluate_sequence(lst))
        except Exception:
            val = np.inf
        if np.isfinite(val) and val < best_actual:
            best_actual = val
            best_out = lst
            print("[validate] actual %.8f n=%d" % (val, arr.size), flush=True)
            if val < 1.48:
                break

    if best_out is None:
        best_out = _to_eval_list(best_a)
        try:
            best_actual = float(evaluate_sequence(best_out))
        except Exception:
            best_actual = best_sc
        print("[validate] fallback actual/best=%.8f" % best_actual, flush=True)

    print("[done] elapsed=%.1fs best_final=%.8f" % (time.monotonic() - t0, best_actual), flush=True)
    return best_out
```
