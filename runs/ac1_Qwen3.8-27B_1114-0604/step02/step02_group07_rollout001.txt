```python
from scipy.optimize import linprog

def solve_LP(f, M):
    """Find g that maximizes sum(g) subject to g >= 0 and (f*g)[k] <= M for all k."""
    n = len(f)
    # Decision variables: g_0, ..., g_{n-1}
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_j f[j] * g[k-j] (valid j)
    # For a given k, the coefficient of g[i] in (f*g)[k] is:
    #   f[k-i] if 0 <= k-i <= n-1, else 0
    # So (f*g)[k] = sum_i g[i] * f[k-i] (with f[x]=0 for x outside [0,n-1])
    
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                A_ub[k, i] = f[j]
    
    b_ub = np.full(2*n-1, M)
    
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    return result.x
```
