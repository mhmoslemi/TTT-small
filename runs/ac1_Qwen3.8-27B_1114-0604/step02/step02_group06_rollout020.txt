```python
"""
Soft-max mirror descent with Nesterov momentum for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Uses adaptive temperature for the soft-max surrogate, backtracking
line search, and multi-resolution search.
"""

import numpy as np
import time

RNG = np.random.default_rng(42)

def _conv_len(n):
    """Next power of 2 >= 2n-1."""
    need = 2 * n - 1
    L = 1
    while L < need:
        L <<= 1
    return L

def _score(p, L=None):
    """Compute 2*n*max(conv(p,p)) assuming sum(p)=1."""
    n = len(p)
    if L is None:
        L = _conv_len(n)
    F = np.fft.rfft(p, L)
    c = np.fft.irfft(F * F, L)[:2*n-1]
    c = np.maximum(c, 0.0)
    return float(2.0 * n * c.max()), c

def _sanitize(p, normalize=True):
    a = np.asarray(p, dtype=float).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if normalize:
        s = a.sum()
        if s <= 0:
            return None
        a = a / s
    return a

def _softmax_weights(c, t):
    """Softmax of c with temperature t (higher t = sharper peak)."""
    if t < 1e-8:
        return np.ones(len(c)) / len(c)
    shift = c.max()
    e = np.exp(t * (c - shift))
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        out = np.zeros(len(c))
        out[c.argmax()] = 1.0
        return out
    return e / s

def _autocorr_gradient(p, k_star, L=None):
    """
    Compute d conv(p,p)[k_star] / d p[i] for all i.
    k_star is the lag index in the full conv array (0 to 2n-2).
    """
    n = len(p)
    # The conv is symmetric around index n-1.
    # lag = k_star - (n-1)
    lag = k_star - (n - 1)
    abs_lag = abs(lag)
    
    grad = np.zeros(n)
    if abs_lag == 0:
        # d sum(p[i]^2) / d p[i] = 2*p[i]
        grad = 2.0 * p
    else:
        # d sum_j p[j]*p[j+abs_lag] / d p[i]
        # = p[i+abs_lag] (if i+abs_lag < n) + p[i-abs_lag] (if i >= abs_lag)
        if abs_lag < n:
            grad[:n-abs_lag] += p[abs_lag:]
            grad[abs_lag:] += p[:n-abs_lag]
    return grad

def _soft_gradient(p, c, t, L=None):
    """
    Compute gradient of the soft-max surrogate.
    f_t = (1/t) * log(sum exp(t*c[k]))
    grad f_t = sum_k w[k] * grad conv[k] where w = softmax(t*c)
    """
    n = len(p)
    w = _softmax_weights(c, t)
    
    # For efficiency, only use the top few terms
    # (the softmax weights decay quickly)
    n_eff = min(n, max(10, int(0.1 * len(c))))
    
    # Get top indices
    top_idx = np.argsort(c)[-n_eff:]
    w_top = w[top_idx]
    w_top = w_top / w_top.sum()
    
    grad = np.zeros(n)
    for idx in top_idx:
        g = _autocorr_gradient(p, idx)
        grad += w_top[int(np.where(top_idx == idx)[0][0])] * g
    
    return grad
```
