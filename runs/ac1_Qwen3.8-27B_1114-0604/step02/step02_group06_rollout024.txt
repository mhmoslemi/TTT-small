            s = _score_val(cand)

                best_cand_score = s
                best_cand = cand

        if best_cand is not None:
            p = best_cand
            if best_cand_score < best_score:
                best_score = best_cand_score
                best = p.copy()

        it += 1

    return best, best_score


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    global _rng
    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except (Exception):
        budget = 2400.0
    if not np.isfinite(budget) or budget < 60:
        budget = max(60.0, budget)
    total_budget = max(30.0, budget - 20.0)

    iseed = int(seed) if seed is not None else 42
    _rng = np.random.default_rng((iseed % 1000000007) + 7)

    # Try parent sequence
    parent = None
    try:
        parent_raw = height_sequence_1
        if parent_raw is not None and len(parent_raw) > 0:
            p = np.asarray(parent_raw, dtype=float).ravel()
            p = np.where(np.isfinite(p), p, 0.0)
            p = np.clip(p, 0, 1000)
            p = _normalize(p)
            if p is not None:
                parent = p
                print(f"Parent sequence loaded: n={len(p)}, score={_score_val(p):.7f}")
    except (NameError, Exception):
        parent = None

    # Determine working size
    work_n = 4096
    if total_budget > 2000:
        work_n = 8192

    # Phase 1: Initialization and coarse optimization (20% of time)
    t1 = 0.20 * total_budget
    candidates = []

    if parent is not None:
        # Resample parent to work_n
        p_parent = _normalize(parent)
        if p_parent is not None:
            if len(p_parent) != work_n:
                x_old = (np.arange(len(p_parent)) + 0.5) / len(p_parent)
                x_new = (np.arange(work_n) + 0.5) / work_n
                p_resampled = np.interp(x_new, x_old, p_parent)
                p_resampled = _normalize(np.maximum(p_resampled, 0))
            else:
                p_resampled = p_parent
            if p_resampled is not None:
                sc = _score_val(p_resampled)
                candidates.append((sc, p_resampled.copy()))
                print(f"Resampled parent: score={sc:.7f}")

    # Create diverse initial candidates
    n_init = 8 if total_budget > 1500 else 5
    for i in range(n_init):
        p_init = _make_initial(work_n, _rng)
        if p_init is not None:
            sc = _score_val(p_init)
            candidates.append((sc, p_init.copy()))

    candidates.sort(key=lambda x: x[0])
    print(f"Initial candidates: best={candidates[0][0]:.7f}")

    # Phase 2: Main gradient optimization on top candidates (50% of time)
    t2 = 0.50 * total_budget
    n_optimize = min(4, len(candidates))
    t_per_opt = t2 / n_optimize

    for i in range(n_optimize):
        rem = total_budget - (time.monotonic() - start_time) - 0.25 * total_budget
        if rem < 30:
            break
        t_alloc = min(t_per_opt, rem)
        if t_alloc < 15:
            continue
        p_start = candidates[i][1].copy()
        print(f"Optimizing candidate {i+1}/{n_optimize} (initial score={candidates[i][0]:.7f}), budget={t_alloc:.0f}s")
        p_opt, sc_opt = _gradient_descent(p_start, t_alloc, target=1.45, verbose=True)
        if p_opt is not None and np.isfinite(sc_opt):
            candidates.append((sc_opt, p_opt.copy()))
            print(f"  -> score={sc_opt:.7f}")

    # Also do a fresh start optimization
    rem = total_budget - (time.monotonic() - start_time) - 0.25 * total_budget
    if rem > 60:
        p_fresh = _make_initial(work_n, _rng)
        if p_fresh is not None:
            p_opt, sc_opt = _gradient_descent(p_fresh, rem, target=1.45, verbose=False)
            if p_opt is not None and np.isfinite(sc_opt):
                candidates.append((sc_opt, p_opt.copy()))
                print(f"Fresh start optimization: score={sc_opt:.7f}")

    candidates.sort(key=lambda x: x[0])
    print(f"After main optimization: best={candidates[0][0]:.7f}")

    # Phase 3: Local search on best candidate (15% of time)
    t3 = 0.15 * total_budget
    rem = total_budget - (time.monotonic() - start_time) - 0.10 * total_budget
    if rem > 30:
        t_alloc = min(t3, rem)
        p_ls = candidates[0][1].copy()
        print(f"Local search, budget={t_alloc:.0f}s")
        p_ls, sc_ls = _local_search(p_ls, t_alloc, verbose=True)
        if p_ls is not None and np.isfinite(sc_ls):
            candidates.append((sc_ls, p_ls.copy()))
            print(f"  -> score={sc_ls:.7f}")

    candidates.sort(key=lambda x: x[0])

    # Phase 4: Coordinate descent polish (10% of time)
    t4 = 0.10 * total_budget
    rem = total_budget - (time.monotonic() - start_time)
    if rem > 20:
        t_alloc = min(t4, rem - 5)
        if t_alloc > 10:
            p_cd = candidates[0][1].copy()
            print(f"Coordinate descent, budget={t_alloc:.0f}s")
            p_cd, sc_cd = _coordinate_descent(p_cd, t_alloc)
            if p_cd is not None and np.isfinite(sc_cd):
                candidates.append((sc_cd, p_cd.copy()))
                print(f"  -> score={sc_cd:.7f}")

    candidates.sort(key=lambda x: x[0])
    best_score = candidates[0][0]
    best_p = candidates[0][1].copy()
    print(f"Best found: score={best_score:.7f}, n={len(best_p)}")

    # Verify with the official evaluator if feasible
    if len(best_p) <= 8192:
        try:
            lst = [float(x) for x in best_p]
            sc_eval = float(evaluate_sequence(lst))
            if np.isfinite(sc_eval) and sc_eval < best_score:
                best_score = sc_eval
                print(f"Official evaluator confirms: {best_score:.7f}")
        except (Exception):
            pass

    # Final sanitization
    out = np.asarray(best_p, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = np.ones(2048, dtype=float)
    else:
        out = out / s

    return [float(x) for x in out]
