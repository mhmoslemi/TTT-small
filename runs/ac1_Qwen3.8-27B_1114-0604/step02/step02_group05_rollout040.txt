```python
"""
Search for non-negative sequences minimizing

    2*n*max(convolve(a,a)) / sum(a)^2

The solver builds on the known parent construction and combines:
  - FFT-based exact scoring and sub/soft-gradient descent in log-simplex space,
  - L-BFGS optimization of a smooth soft-max surrogate,
  - frequency-domain convolution flattening as a global escape move,
  - block-coordinate polishing,
  - multi-scale up-sampling / extended-length searches.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize as _scipy_minimize
    _HAVE_SCIPY = True
except Exception:
    _HAVE_SCIPY = False

_FLOOR = 1e-300


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _init_uniform(n):
    n = max(2, int(n))
    return np.ones(n, dtype=float) / float(n)


def _sanitize(p):
    if p is None:
        return None
    try:
        arr = np.asarray(p, dtype=float).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 1e-12:
        return None
    arr = arr / s
    return arr


def _score_fft(p):
    p = _sanitize(p)
    if p is None:
        return float("inf")
    n = len(p)
    if n <= 0:
        return float("inf")
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(p, L)
    c = np.fft.irfft(A * A, L)[:2 * n - 1]
    # Symmetrize for numerical stability; true convolution is symmetric.
    c = 0.5 * (c + c[::-1])
    mx = float(np.max(c))
    return float(2.0 * n * mx)


def _soft_weights(d, beta):
    n = d.size
    if n == 0:
        return np.array([], dtype=float)
    beta = float(beta)
    if beta < 1e-8:
        return np.ones(n, dtype=float) / n
    shift = float(np.max(d))
    e = np.exp(beta * (d - shift))
    s = float(e.sum())
    if not np.isfinite(s) or s < 1e-300:
        return np.ones(n, dtype=float) / n
    return e / s


def _gradient_h(p, beta, L):
    """
    Returns (exact_score, h) where h_i is the unnormalized contribution
    of coefficient i to the weighted convolution peaks.
    """
    p = _sanitize(p)
    if p is None:
        return float("inf"), None
    n = len(p)
    if L is None or L < 2 * n - 1:
        L = _next_pow2(2 * n - 1)

    A = np.fft.rfft(p, L)
    c = np.fft.irfft(A * A, L)[:2 * n - 1]
    c = 0.5 * (c + c[::-1])
    d = n * c
    score = 2.0 * float(np.max(d))

    w = _soft_weights(d, beta)
    W = np.fft.rfft(w, L)
    # r_i = sum_m a_m * w_{m+i} for i = 0..n-1 (valid lags), using L >= 2n-1.
    v = np.fft.irfft(W * np.conj(A), L)[:n]
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    return score, v


def _mult_step(p, h, step):
    """
    Log-simplex multiplicative update:
        q <- log(p) - step * normalized(h - mean(h))
        p <- softmax(q)
    """
    p = _sanitize(p)
    h = np.asarray(h, dtype=float)
    if p is None or h.size != p.size:
        return None

    step = float(step)
    if not np.isfinite(step) or step <= 0.0:
        return p.copy()
    step = float(np.clip(step, 1e-4, 2.0))

    pn = np.maximum(p, _FLOOR)
    logp = np.log(pn)
    hmean = float(np.mean(h))
    hrange = float(np.max(h) - np.min(h))
    if hrange < 1e-15:
        return p.copy()

    hn = (h - hmean) / hrange
    q = logp - step * hn
    qmax = float(np.max(q))
    if not np.isfinite(qmax):
        return None
    e = np.exp(q - qmax)
    s = float(e.sum())
    if not np.isfinite(s) or s < 1e-300:
        return None
    e /= s
    return e


def _freq_flatten(p, alpha, L=None):
    """
    Frequency-domain flattening: replace C by C^(1/alpha) in frequency domain
    and take a spectral square root. This is a heuristic global move.
    """
    p = _sanitize(p)
    if p is None:
        return None
    n = len(p)
    alpha = float(alpha)
    if not np.isfinite(alpha) or alpha <= 1.0 + 1e-12:
        alpha = 1.25 + 0.05 * float(n) / 1024.0
    alpha = float(np.clip(alpha, 1.1, 5.0))

    if L is None or L < 2 * n - 1:
        L = _next_pow2(2 * n - 1)

    A = np.fft.rfft(p, L)
    C = np.fft.irfft(A * A, L)[:2 * n - 1]
    C = np.maximum(C, 0.0)
    cm = float(C.max())
    if cm < 1e-30:
        return p.copy()

    Cn = C / cm
    Cn = np.power(Cn, 1.0 / alpha)
    Cn = np.maximum(Cn, 0.0)
    s = float(Cn.sum())
    if not np.isfinite(s) or s < 1e-300:
        return None
    Cn /= s

    Cf = np.fft.rfft(Cn, L).astype(complex) + 0j
    An = np.sqrt(Cf)
    an = np.fft.irfft(An, L)[:n]
    an = np.where(np.isfinite(an), an, 0.0)

    neg_frac = float(np.sum(an < -1e-10)) / float(n)
    if neg_frac > 0.45:
        return None

    an = np.maximum(an, 0.0)
    return _sanitize(an)


def _symmetrize(p):
    p = _sanitize(p)
    if p is None:
        return None
    q = 0.5 * (p + p[::-1])
    return _sanitize(q)


def _upsample(p, n_new, rng, sigma=0.0):
    p = _sanitize(p)
    if p is None:
        return None
    n_old = len(p)
    n_new = int(n_new)
    if n_new <= 0:
        return None

    if n_old == n_new:
        out = p.copy()
    elif n_old == 1:
        out = np.ones(n_new, dtype=float)
    else:
        x_old = (np.arange(n_old, dtype=float) + 0.5) / n_old
        x_new = (np.arange(n_new, dtype=float) + 0.5) / n_new
        out = np.interp(x_new, x_old, p)

    out = np.where(np.isfinite(out), out, 0.0)
    out = np.maximum(out, 0.0)

    if sigma > 0.0 and rng is not None:
        noise = np.exp(rng.normal(0.0, float(sigma), n_new))
        out = out * noise
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.maximum(out, 0.0)

    return _sanitize(out)


def _best_of_variants(variants):
    best = None
    best_sc = float("inf")
    for v in variants:
        if v is None:
            continue
        v = _sanitize(v)
        if v is None:
            continue
        sc = _score_fft(v)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = float(sc)
            best = v
    return best, best_sc

def _block_refine(p0, budget, rng):

        f = float(rng.choice(factors))
        p_try = p.copy()

            else:

                    cur_score = best_score

def _to_q_partial(p):
    p = _sanitize(p)
    if p is None:
        return None
    n = len(p)
    if n <= 1:
        return np.zeros(0, dtype=float)
    pn = np.maximum(p, _FLOOR)
    logp = np.log(pn)
    rel = logp - logp[-1]
    rel[-1] = 0.0
    return rel[:-1].astype(float, copy=True)


def _from_q(q):
    q = np.asarray(q, dtype=float).ravel()
    q = np.nan_to_num(q, nan=0.0, posinf=50.0, neginf=-50.0)
    n = len(q) + 1
    qfull = np.zeros(n, dtype=float)
    if len(q) > 0:
        qfull[:-1] = q
    m = float(np.max(qfull))
    a = np.exp(qfull - m)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-300:
        return None
    a /= s
    return _sanitize(a)


def _objective_q(q, n, beta, L):
    """
    Smooth soft-max surrogate for L-BFGS.
    q has length n-1; last log-amplitude is fixed to 0.
    """
    q = np.asarray(q, dtype=float).ravel()
    if q.shape[0] != n - 1:
        qq = np.zeros(max(0, n - 1), dtype=float)
        m = min(q.shape[0], max(0, n - 1))
        if m > 0:
            qq[:m] = q[:m]
        q = qq

    q = np.nan_to_num(q, nan=0.0, posinf=50.0, neginf=-50.0)

    n = int(n)
    if n <= 1:
        return 1e100, np.zeros(0, dtype=float)

    qfull = np.zeros(n, dtype=float)
    qfull[:-1] = q

    m = float(np.max(qfull))
    a = np.exp(qfull - m)
    s = float(a.sum())
    if not np.isfinite(s) or s < 1e-300:
        return 1e100, np.zeros(n - 1, dtype=float)
    a /= s

    L = int(L)
    if L < 2 * n - 1:
        L = _next_pow2(2 * n - 1)

    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:2 * n - 1]
    c = 0.5 * (c + c[::-1])
    d = n * c

    beta = max(1e-6, float(beta))
    dm = float(np.max(d))
    e = np.exp(beta * (d - dm))
    se = float(e.sum())
    if not np.isfinite(se) or se < 1e-300:
        e = np.zeros_like(d)
        idx = int(np.argmax(d))
        e[idx] = 1.0
        se = 1.0

    f = dm + np.log(se) / beta
    w = e / se

    W = np.fft.rfft(w, L)
    v = np.fft.irfft(W * np.conj(A), L)[:n]
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)

    ga = 2.0 * n * v
    avg = float(np.dot(a, ga))
    gfull = a * (ga - avg)
    grad = gfull[:-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(f), grad


def _lbfgs_restart(p0, beta, maxiter=8):
    p = _sanitize(p0)
    if p is None:
        return None, float("inf")

    n = len(p)
    sc0 = _score_fft(p)
    if n < 2 or not _HAVE_SCIPY:
        return p, sc0

    q = _to_q_partial(p)
    if q is None:
        return p, sc0

    L = _next_pow2(2 * n - 1)

    try:
        res = _scipy_minimize(
            _objective_q,
            q,
            method="L-BFGS-B",
            jac=True,
            args=(n, float(beta), int(L)),
            options={
                "maxiter": max(2, int(maxiter)),
                "maxfun": max(20, 50 * int(maxiter)),
                "ftol": 1e-12,
                "gtol": 1e-6,
                "maxcor": 10,
                "disp": False,
            }
        )
        q2 = res.x
    except Exception:
        return p, sc0

    p2 = _from_q(q2)
    if p2 is None:
        return p, sc0
    sc2 = _score_fft(p2)
    if np.isfinite(sc2) and sc2 < sc0 - 1e-15:
        return p2, float(sc2)
    return p, sc0


def _consider_candidate(p, score, best_p, best_score, best_by_len):
    if p is None:
        return best_p, best_score, best_by_len
    try:
        sc = float(score)
    except Exception:
        return best_p, best_score, best_by_len
    if not np.isfinite(sc):
        return best_p, best_score, best_by_len

    p_clean = _sanitize(p)
    if p_clean is None:
        return best_p, best_score, best_by_len

    if sc < best_score - 1e-13:
        best_p = p_clean.copy()
        best_score = float(sc)
        print(f"  *** NEW GLOBAL BEST: {best_score:.7f} (n={len(best_p)}) ***")

    L = len(p_clean)
    entry = best_by_len.get(L)
    if entry is None or sc < entry[1] - 1e-13:
        best_by_len[L] = (p_clean.copy(), float(sc))

    if len(best_p) not in best_by_len:
        best_by_len[len(best_p)] = (best_p.copy(), float(best_score))

    return best_p, best_score, best_by_len


def propose_candidate(seed=12345, budget_s=2400):
    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget < 180.0:
        budget = 2400.0
    deadline = start_time + max(120.0, budget - 30.0)

    try:
        seed_int = int(seed)
    except Exception:
        seed_int = 12345
    rng = np.random.default_rng(seed_int % (2**31 - 1))

    best_p = None
    best_score = float("inf")
    best_by_len = {}

    try:
        # -------- Parent / base sequence --------
        parent = None
        try:
            parent = _sanitize(height_sequence_1)
        except Exception:
            parent = None

        if parent is None or len(parent) < 16:
            parent = _init_uniform(3072)

        if len(parent) > 32768:
            parent_ds = _upsample(parent, 32768, rng, 0.0)
            if parent_ds is not None:
                parent = parent_ds

        if parent is None:
            parent = _init_uniform(3072)

        parent = parent.copy()
        parent_score = _score_fft(parent)
        best_p = parent.copy()
        best_score = parent_score
        best_by_len[len(best_p)] = (best_p.copy(), best_score)
        print(f"Parent score: {best_score:.7f} (n={len(best_p)})")

        # Symmetric parent can sometimes immediately improve.
        p_sym = _symmetrize(parent)
        if p_sym is not None:
            sc_sym = _score_fft(p_sym)
            best_p, best_score, best_by_len = _consider_candidate(
                p_sym, sc_sym, best_p, best_score, best_by_len
            )

        # -------- Phase 1: quick local polish of the base --------
        rem = deadline - time.monotonic()
        if rem > 90:
            b1 = min(180.0, rem*0.30)
            p1, sc1 = _fast_optimize(
                best_p.copy(),
                b1,
                beta_start=20.0,
                beta_end=800.0,
                step_start=0.15,
                use_freq=True,
                rng=rng,
                verbose=False
            )
            best_p, best_score, best_by_len = _consider_candidate(
                p1, sc1, best_p, best_score, best_by_len
            )

        if _HAVE_SCIPY:
            for beta_lb in (30.0, 120.0, 400.0):
                rem = deadline - time.monotonic()
                if rem < 30:
                    break
                p_lb, sc_lb = _lbfgs_restart(best_p.copy(), beta_lb, maxiter=6)
                best_p, best_score, best_by_len = _consider_candidate(
                    p_lb, sc_lb, best_p, best_score, best_by_len
                )

        rem = deadline - time.monotonic()
        if rem > 30:
            p1, sc1 = _fast_optimize(
                best_p.copy(),
                min(60.0, rem*0.25),
                beta_start=400.0,
                beta_end=2500.0,
                step_start=0.06,
                use_freq=False,
                rng=rng,
                verbose=False
            )
            best_p, best_score, best_by_len = _consider_candidate(
                p1, sc1, best_p, best_score, best_by_len
            )

        print(f"After base phase: best={best_score:.7f}, lengths={sorted(best_by_len.keys())[:12]}")

        # -------- Phase 2: multi-scale pipeline --------
        pipeline_deadline = start_time + 0.72 * budget

        target_lengths = []
        for L in (len(best_p), 4096, 8192, 16384, 32768):
            if 512 <= L <= 32768 and L not in target_lengths:
                target_lengths.append(L)
        target_lengths.sort()

        processed = {len(best_p)}

        for L in target_lengths:
            if L in processed:
                continue
            if time.monotonic() >= pipeline_deadline - 90:
                break

            rem_pipe = pipeline_deadline - time.monotonic()
            if rem_pipe < 90:
                break

            # Choose the best candidate with length <= L as source.
            src = None
            src_sc = float("inf")
            for l in best_by_len.keys():
                if l <= L:
                    _, sc = best_by_len[l]
                    if sc < src_sc:
                        src_sc = sc
                        src = best_by_len[l][0]
            if src is None:
                src = best_p

            base0 = _upsample(src, L, rng, 0.0)
            if base0 is None:
                base0 = _init_uniform(L)

            variants = [base0]
            v1 = _symmetrize(base0)
            if v1 is not None:
                variants.append(v1)

            base1 = _upsample(src, L, rng, 0.035)
            if base1 is not None:
                variants.append(base1)
                v2 = _symmetrize(base1)
                if v2 is not None:
                    variants.append(v2)

            base2 = _upsample(src, L, rng, 0.09)
            if base2 is not None:
                variants.append(base2)

            start_p, start_sc = _best_of_variants(variants)
            if start_p is not None:
                best_p, best_score, best_by_len = _consider_candidate(
                    start_p, start_sc, best_p, best_score, best_by_len
                )

            # Length-specific budgets.
            if L <= 4096:
                bfast = 120.0
                lb_betas = (30.0, 80.0, 200.0, 500.0)
                lb_iter = 8
                final_fast = 45.0
            elif L <= 8192:
                bfast = 180.0
                lb_betas = (50.0, 120.0, 300.0, 800.0)
                lb_iter = 7
                final_fast = 55.0
            elif L <= 16384:
                bfast = 250.0
                lb_betas = (80.0, 220.0, 600.0)
                lb_iter = 5
                final_fast = 65.0
            else:
                bfast = 320.0
                lb_betas = (120.0, 350.0, 900.0)
                lb_iter = 4
                final_fast = 75.0

            bfast = min(bfast, rem_pipe * 0.42)
            if bfast > 25 and start_p is not None:
                popt, scopt = _fast_optimize(
                    start_p,
                    bfast,
                    beta_start=20.0,
                    beta_end=1500.0,
                    step_start=0.15,
                    use_freq=True,
                    rng=rng,
                    verbose=False
                )
                best_p, best_score, best_by_len = _consider_candidate(
                    popt, scopt, best_p, best_score, best_by_len
                )

            cur_entry = best_by_len.get(L)
            if cur_entry is None:
                cur_entry = (start_p, start_sc)

            pcur = cur_entry[0]
            cur_sc = cur_entry[1]
            no_improve = 0

            if _HAVE_SCIPY:
                for beta_lb in lb_betas:
                    rem = pipeline_deadline - time.monotonic()
                    if rem < 40:
                        break
                    p_lb, sc_lb = _lbfgs_restart(pcur, beta_lb, maxiter=lb_iter)
                    if sc_lb < cur_sc - 1e-13:
                        pcur = p_lb
                        cur_sc = sc_lb
                        no_improve = 0
                    else:
                        no_improve += 1
                    best_p, best_score, best_by_len = _consider_candidate(
                        pcur, cur_sc, best_p, best_score, best_by_len
                    )
                    if no_improve >= 3:
                        break

            rem = pipeline_deadline - time.monotonic()
            if pcur is not None and rem > 35:
                bf = min(final_fast, rem * 0.35)
                pfin, scfin = _fast_optimize(
                    pcur,
                    bf,
                    beta_start=400.0,
                    beta_end=3000.0,
                    step_start=0.06,
                    use_freq=False,
                    rng=rng,
                    verbose=False
                )
                best_p, best_score, best_by_len = _consider_candidate(
                    pfin, scfin, best_p, best_score, best_by_len
                )

            processed.add(L)
            print(f"Length {L} done. best={best_score:.7f")

        # -------- Phase 3: final polishing of the global best --------
        if best_p is None:
            best_p = parent
            best_score = _score_fft(best_p)

        rem = deadline - time.monotonic()
        if rem > 40 and len(best_p) <= 8192:
            pbr, scbr = _block_refine(best_p, min(120.0, rem * 0.30), rng)
            best_p, best_score, best_by_len = _consider_candidate(
                pbr, scbr, best_p, best_score, best_by_len
            )

        rem = deadline - time.monotonic()
        if rem > 40 and len(best_p) > 8192:
            pbr, scbr = _block_refine(best_p, min(70.0, rem * 0.25), rng)
            best_p, best_score, best_by_len = _consider_candidate(
                pbr, scbr, best_p, best_score, best_by_len
            )

        if _HAVE_SCIPY:
            final_betas = (200.0, 700.0, 2000.0)
            maxit_final = 6 if len(best_p) <= 8192 else 4
            for beta_lb in final_betas:
                rem = deadline - time.monotonic()
                if rem < 35:
                    break
                p_lb, sc_lb = _lbfgs_restart(best_p.copy(), beta_lb, maxiter=maxit_final)
                best_p, best_score, best_by_len = _consider_candidate(
                    p_lb, sc_lb, best_p, best_score, best_by_len
                )

        rem = deadline - time.monotonic()
        if rem > 30:
            bf = min(220.0, rem * 0.45)
            pfin, scfin = _fast_optimize(
                best_p.copy(),
                bf,
                beta_start=500.0,
                beta_end=5000.0,
                step_start=0.05,
                use_freq=True,
                rng=rng,
                verbose=True
            )
            best_p, best_score, best_by_len = _consider_candidate(
                pfin, scfin, best_p, best_score, best_by_len
            )

        # Symmetrization at the very end is often harmless and can help.
        psym = _symmetrize(best_p)
        if psym is not None:
            scsym = _score_fft(psym)
            best_p, best_score, best_by_len = _consider_candidate(
                psym, scsym, best_p, best_score, best_by_len
            )

    except Exception:
        pass

    # -------- Final safe return --------
    if best_p is None:
        best_p = _init_uniform(3072)

    p_final = _sanitize(best_p)
    if p_final is None:
        p_final = _init_uniform(3072)

    p_final = np.where(np.isfinite(p_final), p_final, 0.0)
    p_final = np.clip(p_final, 0.0, 1000.0)
    s = float(p_final.sum())
    if not np.isfinite(s) or s < 1e-8:
        p_final = _init_uniform(3072)
    else:
        p_final = p_final / s

    try:
        print(f"FINAL: n={len(p_final)}, internal_score={_score_fft(p_final):.7f}")
        print(f"FINAL: elapsed={time.monotonic() - start_time:.1f}s")
    except Exception:
        pass

    return [float(x) for x in p_final]
```
