    a = rng.uniform(0.1, 1.0, n)
    return _normalize(a)


def _make_blocks(n, n_blocks, seed):
    """Block-pattern initialization."""
    rng = np.random.RandomState(seed)
    a = np.zeros(n)
    block_size = n // n_blocks
    for k in range(n_blocks):
        a[k * block_size:(k + 1) * block_size] = rng.uniform(0.3, 1.0)
    return _normalize(a)


def _build_lp_matrix(f):
    """Build LP constraint matrix A[k,j] = f[k-j] if 0<=k-j<n else 0."""
    n = len(f)
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f
    return A


def _lp_direction(f, M, upper_bound=50.0):
    """
    Solve LP: max sum(g) s.t. conv(f,g)[k] <= M for all k, g >= 0.
    Returns normalized g (sum=1) or None.
    """
    n = len(f)
    A = _build_lp_matrix(f)
    c = -np.ones(n)
    b = np.full(2 * n - 1, M)
    bounds = [(0.0, upper_bound)] * n

    try:
        result = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                         method='highs', options={'presolve': True,
                                                  'time_limit': 30})
    except Exception:
        return None

    if result.status != 0:
        return None

    g = result.x
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s


def _line_search_quad(C_aa, C_ag, C_gg, M_target):
    """
    Find largest t in [0,1] such that max((1-t)^2*C_aa + 2t(1-t)*C_ag + t^2*C_gg) <= M_target.
    Uses bisection.
    """
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        new_conv = (1 - mid) * (1 - mid) * C_aa + 2.0 * mid * (1 - mid) * C_ag + mid * mid * C_gg
        if float(np.max(new_conv)) <= M_target:
            lo = mid
        else:
            hi = mid
    return lo


def _lp_optimize(a_init, deadline, reduction=0.001, max_iters=20000):
    """
    LP-based optimization:
    1. Solve LP for direction g with current M
    2. Line search to find step size that reduces M
    3. Update and repeat
    """
    a = _normalize(a_init)
    n = a.size
    best_a = a.copy()
    best_score = _score(best_a)
    best_M = float(np.max(_fft_convolve(best_a)))

    it = 0
    stall_count = 0
    rng = np.random.RandomState(42)

    while time.monotonic() < deadline and it < max_iters:
        it += 1
        M = float(np.max(_fft_convolve(a)))

        # Solve LP
        g = _lp_direction(a, M)
        if g is None:
            break

        # Compute cross-convolutions for line search
        C_aa = _fft_convolve(a)
        C_ag = _fft_cross_convolve(a, g)
        C_gg = _fft_convolve(g)

        # Target: reduce M by a small factor
        M_target = M * (1.0 - reduction)

        # Line search
        t = _line_search_quad(C_aa, C_ag, C_gg, M_target)

        if t < 1e-8:
            # Try keeping M constant (reduction=0)
            M_target2 = M * 1.0001
            t = _line_search_quad(C_aa, C_ag, C_gg, M_target2)
            if t < 1e-8:
                stall_count += 1
                if stall_count > 50:
                    # Perturb and reset
                    a = best_a.copy()
                    noise = rng.standard_normal(n) * 0.01
                    a = a + noise * a.mean()
                    a = _normalize(a)
                    stall_count = 0
                    reduction = max(0.0001, reduction * 0.5)
                continue

        # Update
        a = (1.0 - t) * a + t * g
        a = _normalize(a)

        new_M = float(np.max(_fft_convolve(a)))
        sc = 2.0 * n * new_M  # Since sum(a) = 1 after normalization

        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            best_M = new_M
            stall_count = 0
            # Slowly decrease reduction for more aggressive progress
            if it % 200 == 0:
                reduction = max(0.0001, reduction * 0.8)
        else:
            stall_count += 1
            if stall_count > 100:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.005
                a = a + noise * a.mean()
                a = _normalize(a)
                stall_count = 0

        if it % 200 == 0:
            elapsed = time.monotonic() - (deadline - (deadline - time.monotonic()))
            print(f"    LP iter {it}: best_score={best_score:.8f}, M={M:.10f}, t={t:.4f}", flush=True)

    return best_a, best_score


def _softmax_weights(c, T):
    """Compute softmax weights on convolution values."""
    c_max = float(np.max(c))
    c_shift = c - c_max  # max is 0
    exp_c = np.exp(T * c_shift)
    s = exp_c.sum()
    if s < 1e-300:
        w = np.zeros(c.size)
        w[np.argmax(c)] = 1.0
        return w
    return exp_c / s


def _grad_via_fft(w, a):
    """
    Compute grad[j] = 2 * sum_k a[k] * w[k+j] via FFT cross-correlation.
    w has length 2n-1 (full convolution length).
    """
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w[:N], N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]


def _soft_mirror_descent(a_init, deadline, T_start=5.0, T_end=200.0,
                         lr0=None, momentum=0.7):
    """
    Soft-max mirror descent with Nesterov momentum.
    """
    a = _normalize(a_init)
    n = a.size
    if lr0 is None:
        lr0 = 0.3 * n

    best_a = a.copy()
    best_score = _score(best_a)

    it = 0
    last_improve = 0
    patience = 3000
    rng = np.random.RandomState(77)
    v = np.zeros(n)  # momentum accumulator

    while time.monotonic() < deadline:
        it += 1
        t_frac = min(1.0, it / 50000.0)
        T = T_start + (T_end - T_start) * (t_frac ** 0.5)
        lr = lr0 / (1.0 + 0.005 * it)

        c = _fft_convolve(a)
        w = _softmax_weights(c, T)
        grad = _grad_via_fft(w, a)
        grad_proj = grad - np.mean(grad)

        # Nesterov momentum
        v = momentum * v + (1.0 - momentum) * grad_proj

        # Mirror descent update
        update = -lr * v
        update = np.clip(update, -8.0, 8.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        a = _normalize(a)

        if it % 10 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                a = best_a.copy()
                v = np.zeros(n)
                noise = rng.standard_normal(n) * 0.008
                a = a + noise * a.mean()
                a = _normalize(a)
                patience = max(2000, int(patience * 1.3))
                lr0 = max(0.05 * n, lr0 * 0.85)

    return best_a, best_score


def _hard_max_polish(a_init, deadline, K_start=4, K_end=32):
    """Hard-max top-K active set refinement with adaptive K."""
    a = _normalize(a_init)
    n = a.size
    best_a = a.copy()
    best_score = _score(best_a)

    it = 0
    last_improve = 0
    patience = 2000
    lr0 = 0.4 * n
    rng = np.random.RandomState(123)
    v = np.zeros(n)
    momentum = 0.6

    while time.monotonic() < deadline:
        it += 1
        t_frac = min(1.0, it / 30000.0)
        K = K_start + int((K_end - K_start) * t_frac)
        lr = lr0 / (1.0 + 0.003 * it)

        c = _fft_convolve(a)

        # Top-K with softmax weighting
        if c.size <= K:
            top_idx = np.arange(c.size)
            weights = _softmax_weights(c, 5.0)
        else:
            top_idx = np.argpartition(c, -K)[-K:]
            top_vals = c[top_idx]
            w_top = _softmax_weights(top_vals, 5.0)
            weights = np.zeros(c.size)
            weights[top_idx] = w_top

        grad = _grad_via_fft(weights, a)
        grad_proj = grad - np.mean(grad)

        v = momentum * v + (1.0 - momentum) * grad_proj
        update = -lr * v
        update = np.clip(update, -6.0, 6.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        a = _normalize(a)

        if it % 5 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                a = best_a.copy()
                v = np.zeros(n)
                noise = rng.standard_normal(n) * 0.005
                a = a + noise * a.mean()
                a = _normalize(a)
                patience = max(1500, int(patience * 1.2))
                lr0 = max(0.05 * n, lr0 * 0.85)

    return best_a, best_score


def _generate_candidates(deadline, seed=42):
    """Generate diverse initial candidates and return top few by score."""
    rng = np.random.RandomState(seed)
    candidates = []

    # Try to use height_sequence_1 if available
    try:
        h1 = np.asarray(height_sequence_1, dtype=np.float64)
        if h1.size > 0:
            h1_norm = _normalize(h1)
            sc = _score(h1_norm)
            candidates.append((h1_norm, sc, "height_sequence_1"))
            print(f"  height_sequence_1: n={h1.size}, score={sc:.8f}", flush=True)
    except Exception:
        pass

    # Sinusoidal inits
    for n in [1024, 2048, 4096]:
        if time.monotonic() > deadline:
            break
        for amp in [0.6, 0.7, 0.8, 0.9, 1.0, 1.2]:
            if time.monotonic() > deadline:
                break
            a = _make_sine(n, amp)
            sc = _score(a)
            candidates.append((a, sc, f"sine_n{n}_a{amp}"))

    # Multi-sine inits
    for n in [1024, 2048]:
        if time.monotonic() > deadline:
            break
        for trial in range(5):
            modes = [(rng.uniform(0.5, 1.0), 1, rng.uniform(0, 0.5)),
                     (rng.uniform(0.1, 0.4), rng.randint(2, 5), rng.uniform(0, 1.0)),
                     (rng.uniform(0.05, 0.2), rng.randint(5, 10), rng.uniform(0, 1.0))]
            a = _make_multi_sine(n, modes)
            sc = _score(a)
            candidates.append((a, sc, f"multi_n{n}_t{trial}"))

    # Block inits
    for n in [1024, 2048]:
        if time.monotonic() > deadline:
            break
        for nb in [4, 8, 16]:
            a = _make_blocks(n, nb, seed + nb)
            sc = _score(a)
            candidates.append((a, sc, f"blocks_n{n}_{nb}"))

    # Random inits
    for n in [1024, 2048]:
        if time.monotonic() > deadline:
            break
        for s in range(3):
            a = _make_random(n, seed + s)
            sc = _score(a)
            candidates.append((a, sc, f"rand_n{n}_{s}"))

    # Sort by score
    candidates.sort(key=lambda x: x[1])
    print(f"  Generated {len(candidates)} candidates, best score: {candidates[0][1]:.8f}", flush=True)
    return candidates[:10]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search:
    1. Generate diverse candidates
    2. LP-based optimization on top candidates
    3. Soft-max mirror descent on alternative candidates
    4. Hard-max polish on best result
    5. Validate with evaluate_sequence
    """
    t0 = time.monotonic()
    hard_deadline = t0 + float(budget_s) - 15.0

    np.random.seed(seed)
    print(f"[Start] Budget: {budget_s}s", flush=True)

    # Phase 1: Generate candidates (60s max)
    d1 = t0 + 45.0
    print("[Phase 1] Generating candidates...", flush=True)
    cands = _generate_candidates(d1)
    if not cands:
        a = _make_sine(1024, 0.8)
        cands = [(a, _score(a), "fallback")]

    # Phase 2: LP optimization on top candidate (majority of time)
    best_overall = None
    best_overall_score = np.inf

    # Pick top 2-3 candidates for LP optimization
    top_cands = cands[:3]
    # Spread remaining time among them
    n_lp = len(top_cands)
    lp_time_each = max(200.0, (hard_deadline - t0 - 300.0) / n_lp)

    for idx, (cand_a, cand_sc, cand_name) in enumerate(top_cands):
        d2 = time.monotonic() + lp_time_each
        if time.monotonic() >= d2 or hard_deadline - time.monotonic() < 60:
            break
        print(f"[Phase 2.{idx}] LP optimize '{cand_name}' (n={cand_a.size}, "
              f"init_score={cand_sc:.8f}, budget={lp_time_each:.0f}s)...", flush=True)

        result_a, result_sc = _lp_optimize(cand_a, d2, reduction=0.002)
        print(f"  LP result: score={result_sc:.8f} (improved from {cand_sc:.8f})", flush=True)

        if result_sc < best_overall_score:
            best_overall_score = result_sc
            best_overall = result_a

    # Phase 3: Soft-max mirror descent on a different candidate
    d3 = hard_deadline - 200.0
    remaining3 = d3 - time.monotonic()
    if remaining3 > 60:
        # Use a candidate different from the LP best
        soft_cand = cands[3] if len(cands) > 3 else cands[0]
        if best_overall is not None and soft_cand[0].size != best_overall.size:
            print(f"[Phase 3] Soft-mirror on '{soft_cand[2]}' (n={soft_cand[0].size}, "
                  f"budget={remaining3:.0f}s)...", flush=True)
            sa, ss = _soft_mirror_descent(soft_cand[0], d3, T_start=5.0, T_end=150.0)
            print(f"  Soft-mirror result: score={ss:.8f}", flush=True)
            if ss < best_overall_score:
                best_overall_score = ss
                best_overall = sa
        elif best_overall is not None:
            # Run soft-mirror on the LP result for further improvement
            print(f"[Phase 3] Soft-mirror polish on LP best (n={best_overall.size}, "
                  f"budget={remaining3:.0f}s)...", flush=True)
            sa, ss = _soft_mirror_descent(best_overall, d3, T_start=20.0, T_end=100.0)
            print(f"  Soft-mirror result: score={ss:.8f}", flush=True)
            if ss < best_overall_score:
                best_overall_score = ss
                best_overall = sa

    # Phase 4: Hard-max polish
    d4 = hard_deadline - 10.0
    remaining4 = d4 - time.monotonic()
    if remaining4 > 30 and best_overall is not None:
        print(f"[Phase 4] Hard-max polish (n={best_overall.size}, "
              f"budget={remaining4:.0f}s)...", flush=True)
        ha, hs = _hard_max_polish(best_overall, d4, K_start=4, K_end=24)
        print(f"  Hard-max result: score={hs:.8f}", flush=True)
        if hs < best_overall_score:
            best_overall_score = hs
            best_overall = ha

    # Phase 5: Validate with evaluate_sequence
    print("[Phase 5] Validating...", flush=True)
    if best_overall is None:
        best_overall = cands[0][0]

    arr = np.clip(best_overall, 0.0, 1000.0)
    arr_list = [float(x) for x in arr]
    try:
        actual = float(evaluate_sequence(arr_list))
    except Exception as e:
        print(f"  Evaluation error: {e}", flush=True)
        actual = np.inf

    print(f"[Done] Best score: {actual:.8f}, n={arr.size}, "
          f"time={time.monotonic()-t0:.1f}s", flush=True)

    if not np.isfinite(actual):
        # Fallback
        arr_list = [float(x) for x in _make_sine(1024, 0.8)]

    return arr_list
