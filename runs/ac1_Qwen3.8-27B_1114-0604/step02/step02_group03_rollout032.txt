"""
Step-function optimizer for minimizing 2*n*max_k (a*a)_k / (sum a)^2.

Strategy:
- Start from height_sequence_1 and diverse seeds
- Use KKT-subgradient with adaptive temperature for gradient steps
- Heavy investment in coordinate polish with grid-search over transfer amounts
- Periodic structural perturbations to escape local minima
- Multi-phase time allocation: quick gradient -> long coordinate polish -> final polish

import numpy as np
import time
import math


def _normalize(v):
    """Project to non-negative float vector of sum 1."""
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v


def _fft_size(m):
    if m <= 2:
        return 2
    return 1 << (int(m) - 1).bit_length()


def _autocorr(f):
    """Full linear convolution f*f via FFT."""
    n = f.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = 2 * n - 1
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]


def _fast_score(f):
    f = _normalize(f)
    if f.size == 0:
        return np.inf
    c = _autocorr(f)
    if c.size == 0:
        return np.inf
    return float(2.0 * f.size * c.max())


def _project_simplex(v):
    """Euclidean projection onto x >= 0, sum x = 1."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    out = v - theta
    np.maximum(out, 0.0, out)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    out /= s
    return out


def _kkt_gradient(f, c, temp_frac=0.05):
    """
    Compute subgradient of max_k (a*a)_k using softmax over lags.
    Returns g where g_j = 2 * sum_k mu_k * a_{k-j}
    """
    n = f.size
    L = c.size
    maxc = float(c.max())
    minc = float(c.min())
    rng_c = maxc - minc
    if rng_c < 1e-300:
        return np.zeros(n, dtype=np.float64)
    
    T = max(rng_c * temp_frac, 1e-300)
    z = (c - maxc) / T
    np.clip(z, -50.0, 0.0, out=z)
    p = np.exp(z)
    ps = float(p.sum())
    if ps < 1e-300:
        p.fill(1.0 / L)
    else:
        p /= ps
    
    # g_j = 2 * sum_k p_k * a_{k-j}
    # a_{k-j} = f[k-j] if 0 <= k-j < n else 0
    # g = 2 * convolve(p, f_reversed)[n-1 : n-1+n]
    m = L + n - 1
    size = _fft_size(m)
    P = np.fft.rfft(p, size)
    F = np.fft.rfft(f[::-1], size)
    conv = np.fft.irfft(P * F, size)
    g = 2.0 * conv[n - 1:n + n - 1]
    return g


def _coord_move_max(c, f, i, j, dv):
    """
    Compute max of c' after transferring dv from position i to position j.
    c' = c + 2*dv*(f_shift_j - f_shift_i) + dv^2*(delta_2j - delta_2i)
    Returns the new max value without modifying c or f.
    """
    n = f.size
    L = c.size
    
    # f_shift_i[k] = f[k-i] if 0 <= k-i < n else 0
    # f_shift_j[k] = f[k-j] if 0 <= k-j < n else 0
    # We need: c + 2*dv*(f_shift_j - f_shift_i) + dv^2*(delta_2j - delta_2i)
    
    # Build f_shift_i and f_shift_j
    # f_shift_i[k] = f[k-i] for k in [i, i+n-1]
    # f_shift_j[k] = f[k-j] for k in [j, j+n-1]
    
    diff = np.zeros(L, dtype=np.float64)
    # f_shift_j has f in positions [j, j+n-1]
    # f_shift_i has f in positions [i, i+n-1]
    # The diff is f_shift_j - f_shift_i
    
    # For positions in [max(i,j), min(i+n, j+n)) they overlap
    # Simpler: just build both and subtract
    if i < L:
        end_i = min(i + n, L)
        diff[i:end_i] -= f[:end_i - i]
    if j < L:
        end_j = min(j + n, L)
        diff[j:end_j] += f[:end_j - j]
    
    # Quadratic term: dv^2 * (delta_{2j,k} - delta_{2i,k})
    # delta_{2j,k} = 1 if k == 2j and 0 <= 2j < L, else 0
    quad = np.zeros(L, dtype=np.float64)
    if 2 * j < L:
        quad[2 * j] = 1.0
    if 2 * i < L:
        quad[2 * i] = -1.0
    
    c_new = c + 2.0 * dv * diff + dv * dv * quad
    return float(np.max(c_new))


def _coord_polish(f, c, budget_s, rng):
    """
    Extended coordinate polish with grid search over transfer amounts.
    Multiple donor/receiver candidates, progressive refinement.
    """
    f = _normalize(f)
    c = _autocorr(f)
    n = f.size
    L = 2 * n - 1
    
    maxc = float(c.max())
    best_f = f.copy()
    best_maxc = maxc
    best_score = float(2.0 * n * maxc)
    
    deadline = time.monotonic() + float(budget_s)
    
    if n < 2:
        return f, best_score, c
    
    # Grid of fractions for transfer amount
    fracs_coarse = np.array([0.01, 0.03, 0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5, 0.7])
    fracs_fine = np.array([0.005, 0.015, 0.025, 0.04, 0.06, 0.09, 0.14, 0.2, 0.3, 0.45])
    
    D = min(30, n)  # number of donors to try
    R = min(30, n)  # number of receivers to try
    
    move_count = 0
    no_improve_rounds = 0

    while time.monotonic() < deadline and no_improve_rounds < 50:
        maxc = float(c.max())
        
        # Compute gradient for donor/receiver selection
        g = _kkt_gradient(f, c, temp_frac=0.08)
        
        # Select donors (high gradient) and receivers (low gradient)
        sorted_idx = np.argsort(g)
        recv_candidates = sorted_idx[:R]
        donor_candidates = sorted_idx[::-1][:D]
        
        # Filter: donors must have positive mass
        donors = [int(i) for i in donor_candidates if f[int(i)] > 1e-15]
        recv_set = set(int(i) for i in recv_candidates)

        if not donors or not receivers:
            no_improve_rounds += 1
            break
        
        best_improvement = maxc
        best_move = None
        
        # Try all (donor, receiver) pairs with grid search
        for i in donors:

            if fi <= 1e-15:
                continue
            for j in receivers:
                # Quick check: is there potential for improvement?
                # The linear term at the max lag should be negative
                max_k = int(np.argmax(c))
                linear_change = 2.0 * 0.01 * fi * (float(f[max(0, min(n-1, max_k - j))]) - float(f[max(0, min(n-1, max_k - i))]))

                    # (but this check is approximate; the optimal might reduce other lags)
                    # Only skip if it's clearly bad

                        continue
                
                # Grid search over transfer amounts
                for frac in fracs_coarse:
                    dv = fi * frac
                    if dv <= 0 or dv >= fi:
                        continue
                    new_max = _coord_move_max(c, f, i, j, dv)
                    if new_max < best_improvement:

        # Also try some random pairs

        # Apply the best move
        i, j, dv = best_move
        f[i] -= dv
        f[j] += dv
        c = _autocorr(f)  # Recompute to avoid drift

def _perturb_structural(f, rng):
    """
    Structural perturbation: pick a random block and reshape it.
    """
    f = _normalize(f)
    n = f.size
    
    r = float(rng.random())
    if r < 0.3:
        # Block uniformization: pick a block, make it more uniform
        block_len = int(rng.integers(max(50, n // 20), max(100, n // 3)))
        start = int(rng.integers(0, max(1, n - block_len)))
        block = f[start:start + block_len].copy()
        target = float(block.mean())
        f[start:start + block_len] = target * np.ones(block_len)
        f[start:start + block_len] *= 0.5 + 0.5 * np.exp(0.1 * rng.standard_normal(block_len))
    elif r < 0.5:
        # Mirror flip of a block
        block_len = int(rng.integers(max(50, n // 10), n // 2))
        start = int(rng.integers(0, max(1, n - block_len)))
        f[start:start + block_len] = f[start:start + block_len][::-1]
    elif r < 0.7:
        # Smooth + roughen
        win = max(3, n // 100)
        kernel = np.ones(win) / win
        f = _normalize(np.convolve(f, kernel, mode='same'))
        f = _normalize(f * np.exp(0.2 * rng.standard_normal(n)))
    elif r < 0.85:
        # Multiplicative noise
        sigma = float(rng.uniform(0.05, 0.4))
        f = _normalize(f * np.exp(sigma * rng.standard_normal(n)))
    else:
        # Add a bump
        center = int(rng.integers(0, n))
        width = int(rng.integers(max(10, n // 50), n // 5))
        x = np.arange(n)
        bump = np.exp(-0.5 * ((x - center) / width) ** 2)
        f = _normalize(f + 0.1 * bump)
    
    return _normalize(f)


def _build_seeds(f0, n, rng):
    """Build a pool of diverse seeds."""
    seeds = []
    
    def add_seed(name, f):

        if f.size != n:
            x_old = np.linspace(0, f.size - 1, f.size)
            x_new = np.linspace(0, n - 1, n)
            f = _normalize(np.interp(x_new, x_old, f))
        try:
            sc = _fast_score(f)
        except Exception:
            sc = np.inf
        if np.isfinite(sc):
            seeds.append((sc, name, f))
    
    add_seed("given", f0)
    add_seed("reversed", f0[::-1])
    add_seed("uniform", np.ones(n))
    
    # Smoothed versions
    for win in (7, 21, 51, 101, 201):
        if win < n:
            kernel = np.ones(win) / win
            add_seed(f"smooth{win}", np.convolve(f0, kernel, mode='same'))
    
    # Power transforms
    for gamma in (0.5, 0.8, 1.2, 1.5, 2.0):
        base = f0 + 1e-10
        add_seed(f"pow{gamma}", np.power(base, gamma))
    
    # Fourier seeds
    x = np.linspace(0, 1, n)
    for idx in range(8):
        u = np.zeros(n)
        u += float(rng.uniform(-1, 1)) * (x - 0.5)
        for m in (1, 2, 3, 5, 8, 13):
            u += 0。3 * float(rng.uniform(-1, 1)) * np.cos(2 * np.pi * m * x + float(rng.uniform(0, 2 * np.pi)))
        f = _normalize(np.exp(0.5 * u))
        add_seed(f"fourier{idx}", f)
    
    # Beta seeds

        a = float(rng.uniform(0.3, 3.0))

    # Bimodal seeds
    for idx in range(4):
