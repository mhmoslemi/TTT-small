def fast_self_conv(a):

    M = 2*n - 1
    N = 1 << (M-1).bit_length()

    return np.fft.irfft(A*A, N)[:M]

    if S < 1e-10:
        return np.inf
    c = fast_self_conv(a)
    return 2.0*n*c.max()/(S*S)

def grad_of_conv_at_peak(a, peak_idx, weights):
    """Compute weighted gradient of convolution peaks."""
    n = len(a)
    grad = np.zeros(n)
    for k, w in zip(peak_idx, weights):
        if w < 1e-15: continue
        # d(conv_k)/da_m = 2*a_{k-m} for 0<=k-m<n
        lo = max(0, k-n+1)
        hi = min(n-1, k)
        if lo > hi: continue
        grad[lo:hi+1] += w * 2.0 * a[k-hi:k-lo+1][::-1]
    return grad

def freq_flatten(a, alpha=0.5, iters=10):
    """Apply frequency-domain flattening."""
    n = len(a)
    M = 2*n-1
    N = 1 << (M-1).bit_length()
    for _ in range(iters):
        A = np.fft.rfft(a, N)
        mag2 = A * np.conj(A).real  # |A|^2
        # Target: flat spectrum
        target = (a.sum()**2) / N

        a = np.clip(a, 0, None)
        S = a.sum()
        if S > 1e-10:
            a = a / S
    return a

import time

def fast_self_conv(a):

    M = 2*n - 1
    N = 1 << (M-1).bit_length()
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A*A, N)[:M]

def obj_value(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fast_self_conv(a)
    return 2.0*n*c.max()/(S*S)

def freq_equalize(a, n_iter=5, alpha=0.3):
    """Frequency-domain equalization step."""
    n = len(a)
    M = 2*n-1
    N = 1 << (M-1).bit_length()

    S_orig = a_work.sum()

        s = a_work.sum()
        if s > 1e-10:
            a_work = a_work * (S_orig / s)
    
    return a_work

        w = wts[i]
        if w < 1e-15:
            continue
        ki = int(idx[i])
        lo = max(0, ki - n + 1)
        hi = min(n-1, ki)
        if lo > hi:
            continue
        grad_c[lo:hi+1] += w * 2.0 * a[ki-hi:ki-lo+1][::-1]
    
    max_c = float(c.max())
    g = 2.0*n*(grad_c/(S*S) - 2.0*max_c/(S*S*S))
    return g

def freq_flatten(a, n_iter=3, alpha=0.3):
    n = len(a)
    M = 2*n-1
    N = 1 << (M-1).bit_length()
    S = a.sum()
    a_work = a.copy()
    for _ in range(n_iter):
        A = np.fft.rfft(a_work, N)
        mag2 = A.real**2 + A.imag**2
        target = S*S / N
        ratio = np.sqrt(target / (mag2 + 1e-30))
        A_new = A * (ratio ** alpha)
        a_work = np.fft.irfft(A_new, N)[:n]
        a_work = np.clip(a_work, 0, None)
        s = a_work.sum()
        if s > 1e-10:
            a_work = a_work * S / s
    return a_work

def lp_step(f, M_val):
    """Solve LP: max sum(b) s.t. conv(b,f) <= M_val, b >= 0."""
    n = len(f)
    # Build constraint matrix: conv(b, f)_k = sum_j b_j * f_{k-j} <= M_val
    # This is a Toeplitz-like matrix
    # c_k = sum_j b_j * f_{k-j} for k = 0, ..., 2n-2
    # For efficiency, build the matrix explicitly (only for small n)
    
    # Actually, for large n, this matrix is too large to build explicitly.
    # Use the structure: c = conv(b, f), so c_k = sum_j b_j * f_{k-j}
    # The constraint matrix A_ub has A_ub[k, j] = f_{k-j} if 0 <= k-j < n, else 0
    
    # For n <= 2048, build explicitly
    if n > 2048:
        return None  # Too large
    
    A_ub = np.zeros((2*n-1, n))
    for j in range(n):
        lo = j  # k-j >= 0 => k >= j
        hi = j + n - 1  # k-j <= n-1 => k <= j+n-1
        A_ub[lo:hi+1, j] = f[::-1]  # f_{k-j} for k = j, j+1, ..., j+n-1
        # Wait, this is wrong. Let me redo.
        # A_ub[k, j] = f_{k-j} if 0 <= k-j < n
        # So for fixed j, A_ub[k, j] = f_{k-j} for k = j, j+1, ..., j+n-1
        # That is, A_ub[j:j+n, j] = f
    
    # Redo properly
    A_ub = np.zeros((2*n-1, n))
    for j in range(n):
        # k ranges from j to min(j+n-1, 2n-2)
        k_start = j
        k_end = min(j + n - 1, 2*n - 2)
        A_ub[k_start:k_end+1, j] = f[0:k_end-k_start+1]
    
    # c = -1 (minimize -sum(b))
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M_val)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x
