    return np.fft.irfft(F * F, size)[:2*n-1]

def _smooth_grad_fft(f, c, tau):

    grad[i] = 2 * sum_k w[k] * f[k-i] (for valid indices)
    where w = softmax(c/tau)

    c_shifted = c - c.max()
    w = np.exp(c_shifted / tau)
    w_sum = w.sum()
    if w_sum < 1e-300:

    else:
        w /= w_sum

    W = np.fft.rfft(w, N)
    result = np.fft.irfft(F_rev * W, N)
    
    grad = 2.0 * result[n-1 : 2*n-1]
    # Truncate to n elements (should already be n)
    grad = grad[:n]
    
    return grad
