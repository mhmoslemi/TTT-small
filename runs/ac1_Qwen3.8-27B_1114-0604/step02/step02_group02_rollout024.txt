Think of it this way: we have "resources" (the constraints $(f*g)_k \leq M$) and we want to allocate "mass" $g_j$ to maximize the total. Each unit of $g_j$ "uses up" $f_{k-j}$ of the resource at position $k$ (for each $k$ in the appropriate range).

This is like a bounded knapsack problem where each "item" $j$ has a different "cost vector" (determined by $f$).

The greedy approach (pick the $j$ that allows the most mass) is a reasonable heuristic.

Let me now write the full implementation. I'll combine:
1. Smart initialization ($1/\sqrt{i}$ family)
2. Mirror descent with soft gradient (fast)
3. Greedy Frank-Wolfe (medium speed, periodic)
4. SA for exploration
5. Coordinate descent polish

"""
Step function optimizer using greedy Frank-Wolfe + mirror descent + SA.
Key innovation: 1/sqrt(i) initialization and greedy LP for Frank-Wolfe direction.
"""

import numpy as np
import time

def _conv_fft(f):
    """Linear convolution of f with itself via FFT."""
    n = len(f)
    size = 1 << (2*n-1).bit_length()
    F = np.fft.rfft(f, size)
    result = np.fft.irfft(F * F, size)
    return result[:2*n-1]

def _soft_subgrad(f, c, K=10, beta=50.0):
    """Soft subgradient using top-K lags."""
    n = len(f)
    M = c.max()
    # Indices of top K lags
    # Use argpartition for efficiency
    if K >= len(c):
        top_idx = np.arange(len(c))
    else:
        part = np.argpartition(c, -K)[-K:]
        top_idx = part
    
    # Soft weights
    diffs = M - c[top_idx]
    weights = np.exp(-beta * diffs)
    wsum = weights.sum()
    if wsum < 1e-300:
        weights = np.ones(len(top_idx)) / len(top_idx)
    else:
        weights /= wsum
    
    grad = np.zeros(n)
    for w, idx in zip(weights, top_idx):
        lo = max(0, idx - n + 1)
        hi = min(n, idx + 1)
        grad[lo:hi] += w * f[idx - np.arange(lo, hi)]
    return grad

def _mirror_step(f, grad, eta):
    """Mirror descent (exponentiated gradient) step."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    return f_new / s

def _greedy_fw(f, M, max_iters=200):
    """Greedy Frank-Wolfe: find max-sum g s.t. f*g <= M, g >= 0."""
    n = len(f)
    c = np.zeros(2*n-1)  # current f*g (starts at 0)
    g = np.zeros(n)
    slack = M - c  # = M initially
    
    for _ in range(max_iters):
        # For each j, find max delta: min over k in [j, j+n-1] of slack[k] / f[k-j]
        # (only where f[k-j] > 0)
        
        best_j = -1
        best_delta = 0
        
        for j in range(n):
            # k ranges from j to j+n-1 (bounded by 2n-2)
            k_lo = j
            k_hi = min(2*n-2, j+n-1)
            # f[k-j] for k in [k_lo, k_hi] is f[0], f[1], ..., f[n-1] (or fewer)
            f_slice = f[:k_hi-k_lo+1]
            # slack[k] for k in [k_lo, k_hi]
            s_slice = slack[k_lo:k_hi+1]
            # delta_max = min(s_slice / f_slice) where f_slice > 0
            mask = f_slice > 1e-300
            if not mask.any():
                continue
            ratios = np.full(len(f_slice), np.inf)
            ratios[mask] = s_slice[mask] / f_slice[mask]
            d = ratios.min()
            if d > best_delta:
                best_delta = d
                best_j = j
        
        if best_j < 0 or best_delta <= 1e-15:
            break
        
        g[best_j] += best_delta
        # Update slack: slack[k] -= best_delta * f[k-best_j] for k in [best_j, best_j+n-1]
        k_lo = best_j
        k_hi = min(2*n-2, best_j+n-1)
        slack[k_lo:k_hi+1] -= best_delta * f[:k_hi-k_lo+1]
    
    return g, g.sum()

def _optimize(f0, deadline, rng, n):
    """Main optimization: mirror descent + periodic greedy FW + SA."""
    f = f0.copy()
    f = np.maximum(f, 0.0)
    f /= f.sum()
    
    c = _conv_fft(f)
    peak = c.max()
    best_peak = peak
    best_f = f.copy()
    
    eta = 0.5
    T = peak * 0.1
    last_improve = time.monotonic()
    iter_count = 0
    fw_count = 0
    
    while time.monotonic() < deadline:
        c = _conv_fft(f)
        cur_peak = c.max()
        
        if cur_peak < best_peak - 1e-14:
            best_peak = cur_peak
            best_f = f.copy()
            last_improve = time.monotonic()
        
        stuck = time.monotonic() - last_improve
        
        # Adaptive parameters
        if stuck > 30:
            T = peak * 0.3
            eta = max(eta, 0.3)
            # Perturb
            mag = min(0.03, 0.005 + 0.002 * stuck / 60)
            f = f + mag / np.sqrt(n) * rng.standard_normal(n)
            f = np.maximum(f, 0)
            f /= f.sum()
            last_improve = time.monotonic()
        else:
            T = max(peak * 1e-12, T * 0.999)
            eta *= 0.9998
        
        # Mirror descent step
        K = min(20, max(5, n // 200))
        beta = 30.0
        grad = _soft_subgrad(f, c, K, beta)
        grad_norm = grad.max() - grad.min()
        if grad_norm < 1e-18:
            grad_norm = 1.0
        f_new = _mirror_step(f, grad / grad_norm, eta)
        
        c_new = _conv_fft(f_new)
        new_peak = c_new.max()
        delta = new_peak - cur_peak
        
        if delta <= 0:
            f = f_new
            eta = min(eta * 1.003, 2.0)
        elif T > 1e-30 and np.log(max(rng.uniform(), 1e-300)) < -delta / T:
            f = f_new
        else:
            eta *= 0.9
        
        iter_count += 1
        
        # Periodic greedy Frank-Wolfe
        if iter_count % 500 == 0 and time.monotonic() < deadline - 30:
            t0 = time.monotonic()
            g, g_sum = _greedy_fw(f, cur_peak, max_iters=100)
            fw_time = time.monotonic() - t0
            if g_sum > 1e-10:
                g = g / g_sum  # normalize
                # Blend
                for t_blend in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    f_try = (1 - t_blend) * f + t_blend * g
                    f_try = np.maximum(f_try, 0)
                    f_try /= f_try.sum()
                    c_try = _conv_fft(f_try)
                    if c_try.max() < cur_peak:
                        f = f_try
                        cur_peak = c_try.max()
                        if cur_peak < best_peak:
                            best_peak = cur_peak
                            best_f = f.copy()
                            last_improve = time.monotonic()
                        break
            fw_count += 1
            if fw_count % 10 == 0:
                print(f"    iter={iter_count}, peak={cur_peak:.10f}, best={best_peak:.10f}, "
                      f"FW_time={fw_time:.1f}s, eta={eta:.3f}")
        
        if iter_count % 10000 == 0:
            print(f"    iter={iter_count}, peak={cur_peak:.10f}, best={best_peak:.10f}, "
                  f"eta={eta:.3f}, T={T:.2e}")
    
    c = _conv_fft(f)
    if c.max() < best_peak:
        best_peak = c.max()
        best_f = f.copy()
    
    return best_f, best_peak

def _polish(f, deadline, rng, n):
    """Coordinate descent polish."""
    f = f.copy()
    f = np.maximum(f, 0)
    f /= f.sum()
    
    c = _conv_fft(f)
    best_peak = c.max()
    best_f = f.copy()
    
    passes = 0
    while time.monotonic() < deadline:
        c = _conv_fft(f)
        cur_peak = c.max()
        
        if cur_peak < best_peak - 1e-14:
            best_peak = cur_peak
            best_f = f.copy()
        
        k_star = int(np.argmax(c))
        
        # Find the most contributing pair at k_star
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        contrib = f[k_star - np.arange(lo, hi)]
        # The contribution of element i (at position lo + offset) to c[k_star] is 2*f[k_star - (lo+offset)]
        # Actually, c[k_star] = sum_{i=lo}^{hi-1} f[i] * f[k_star - i]
        # The "gradient" with respect to f[i] is f[k_star - i]
        
        # Try moving mass from high-contrib to low-contrib positions
        donor = lo + int(np.argmax(contrib))
        
        # Candidates for receiving mass: positions that contribute little to the peak
        # and whose increase doesn't create new peaks
        low_contrib = np.argsort(contrib)[:20]
        recv_idx = [lo + x for x in low_contrib[:10] if lo + x != donor]
        
        improved = False
        for frac in [0.01, 0.03, 0.05, 0.1, 0.15, 0.25]:
            for recv in recv_idx:
                amt = f[donor] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[donor] -= amt
                f_try[recv] += amt
                f_try /= f_try.sum()
                c_try = _conv_fft(f_try)
                if c_try.max() < cur_peak - 1e-15:
                    f = f_try
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Random moves
            for _ in range(20):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                frac = rng.choice([0.01, 0.03, 0.08, 0.15])
                amt = f[i] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[i] -= amt
                f_try[j] += amt
                if f_try[i] < 0:
                    continue
                f_try /= f_try.sum()
                c_try = _conv_fft(f_try)
                if c_try.max() < cur_peak - 1e-15:
                    f = f_try
                    improved = True
                    break
        
        if not improved and passes % 20 == 0:
            # Small perturbation
            f_try = f + 0.001 / np.sqrt(n) * rng.standard_normal(n)
            f_try = np.maximum(f_try, 0)
            f_try /= f_try.sum()
            c_try = _conv_fft(f_try)
            if c_try.max() < cur_peak - 1e-15:
                f = f_try
        
        passes += 1
    
    c = _conv_fft(f)
    if c.max() < best_peak:
        best_peak = c.max()
        best_f = f.copy()
    
    return best_f, best_peak

def _make_starts(n, rng):
    """Generate diverse starting points."""
    starts = []
    j = np.arange(n, dtype=np.float64)
    
    # Primary: 1/sqrt(i+1)
    starts.append(1.0 / np.sqrt(j + 1))
    
    # Symmetric: 1/sqrt((i+1)(n-i))
    starts.append(1.0 / np.sqrt((j + 1) * (n - j)))
    
    # Various power laws
    for alpha in [0.3, 0.4, 0.45, 0.55, 0.6, 0.7]:
        starts.append(np.power(j + 1, -alpha))
    
    # Symmetric power laws
    for alpha in [0.3, 0.4, 0.5, 0.6]:
        w = 1.0 / np.sqrt(np.power(j + 1, alpha) * np.power(n - j, alpha))
        starts.append(w)
    
    # Linear taper
    starts.append(1.0 / (j/n + 0.1))
    starts.append(1.0 / (1 - j/n + 0.1))
    
    # Two-sided taper (stronger)
    starts.append(1.0 / (0.1 + 0.9*j/n) * 1.0 / (0.1 + 0.9*(1-j/n)))
    
    # Cosine taper
    starts.append(1.0 / (0.3 + 0.7 * np.cos(np.pi * j / n) ** 2))
    
    # Sine-based
    starts.append(1.0 / (1.0 + 2.0 * np.sin(np.pi * j / (2*n)) ** 2))
    
    # Gaussian (wide)
    center = (n-1) / 2.0
    for sigma in [0.3, 0.5, 0.7, 1.0]:
        starts.append(np.exp(-((j - center) / (sigma * n)) ** 2))
    
    # Uniform
    starts.append(np.ones(n))
    
    # Random
    for _ in range(10):
        starts.append(rng.exponential(1.0, n))
    
    # Random with structure
    for _ in range(5):
        base = 1.0 / np.sqrt(j + 1)
        noise = rng.standard_normal(n) * 0.1
        starts.append(np.maximum(base + noise, 0.01))
    
    return starts
