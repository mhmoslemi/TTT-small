        m_hi = min(n-1, k_star)
        if m_hi >= m_lo:
            sg[m_lo:m_hi+1] = 2.0 * a[k_star - np.arange(m_lo, m_hi+1)]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - 1.0) / (gn * gn + 1e-30)
        step = min(step, 0.1)
        a = a - step * grad
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def build_inits(n, rng, parent):
    inits = []
    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())
    inits.append(np.ones(n) * 0.5)
    x = np.arange(n, dtype=np.float64) / n
    # Various structured inits
    inits.append(0.5 + 0.3 * np.cos(2*np.pi*x))
    inits.append(0.5 + 0.3 * np.sin(2*np.pi*x))
    inits.append(0.3 + 0.4 * np.abs(np.sin(np.pi*x)))
    inits.append(np.linspace(0.01, 1.0, n))
    inits.append(np.linspace(1.0, 0.01, n))
    # Multi-cluster
    for m in [8, 16, 32]:
        a = np.zeros(n)
        positions = np.linspace(0, n-1, m, dtype=int)
        w = max(1, n // (m * 3))
        for pos in positions:
            lo = max(0, pos - w//2)
            hi = min(n, pos + w//2 + 1)
            a[lo:hi] += np.exp(-0.5*((np.arange(lo,hi) - pos)**2)/(w//2)**2)
        if a.sum() > 0:
            inits.append(a)
    # Random
    inits.append(np.clip(0.5 + 0.2*rng.standard_normal(n), 0.01, 10))
    inits.append(np.clip(0.5 + 0.5*rng.standard_normal(n), 0.01, 10))
    return inits
