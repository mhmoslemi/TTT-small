import numpy as np
import time
from scipy.optimize import minimize

def _fft_convolve(a, b):
    """Linear convolution via FFT."""
    n = len(a)
    m = len(b)
    L = 1
    while L < n + m - 1:
        L <<= 1
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    return np.fft.irfft(A * B, L)[:n+m-1]

def _objective_and_grad(x, beta, L_cache=None):
    """Compute objective and gradient for softmax parameterization."""
    # Compute p from x
    x_shifted = x - np.max(x)
    exp_x = np.exp(x_shifted)
    p = exp_x / exp_x.sum()
    n = len(p)
    
    # Compute convolution
    if L_cache is None:
        L = 1
        while L < 2*n - 1:
            L <<= 1
    else:
        L = L_cache
    Fp = np.fft.rfft(p, L)
    c = np.fft.irfft(Fp * Fp, L)[:2*n-1]
    c = np.maximum(c, 0.0)  # numerical errors
    
    # Soft-max weights
    beta_c = beta * c
    beta_c_shifted = beta_c - np.max(beta_c)
    exp_beta_c = np.exp(beta_c_shifted)
    w = exp_beta_c / exp_beta_c.sum()
    
    # Objective
    soft_max_val = (np.max(beta_c) - beta_c_shifted.max()) / beta + np.log(exp_beta_c.sum()) / beta
    # Actually: soft_max = (1/beta) * log(sum(exp(beta*c)))
    # = (1/beta) * (max(beta*c) + log(sum(exp(beta*c - max(beta*c)))))
    soft_max_val = (np.max(beta_c) + np.log(exp_beta_c.sum())) / beta
    obj = 2.0 * n * soft_max_val
    
    # Gradient
    g = np.correlate(w, p, mode='valid')  # length n
    wc = np.dot(w, c)
    grad = 4.0 * n * p * (g - wc)

    """Compute soft-max objective and gradient.
    
    x: parameter vector (length n)
    beta: sharpness parameter
    p: softmax(x) (length n)
    c: convolution of p with p (length 2n-1)
    L: FFT length used for c

    bc = beta * c
    bc_max = np.max(bc)
    exp_bc = np.exp(bc - bc_max)
    sum_exp = np.sum(exp_bc)
    w = exp_bc / sum_exp
    
    # Objective: 2*n * (1/beta) * (max(bc) + log(sum(exp(bc - max(bc)))))

    g = np.correlate(w, p, mode='valid')

    grad = 4.0 * n * p * (g - wc)
