def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b, L=None):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)


def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())


def _sanitize(p):

    return p / s

def _init_block_random(n, rng):

        result = np.ones(n_new, dtype=float)
    else:
        x_old = (np.arange(n_old) + 0.5) / n_old
        x_new = (np.arange(n_new) + 0.5) / n_new
        result = np.interp(x_new, x_old, p)
    result = np.maximum(result, 0)
    if rng is not None:
        result = result * np.exp(rng.normal(0, 0.02, n_new))
        result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Multi-phase search combining centered gradient descent, convolution square-root
    smoothing with adaptive parameters, and simulated annealing for robust
    optimization of non-negative sequences.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 2400.0

    start_time = time.monotonic()
    deadline = start_time + budget - 10

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng(iseed % (2**31 - 1))

    # Get parent
    try:
        parent = _sanitize(np.asarray(height_sequence_1, dtype=float))
    except Exception:
        parent = None

    best_overall = None
    best_score_overall = float('inf')

    def _try_update(p, score):
        nonlocal best_overall, best_score_overall
        if p is not None and np.isfinite(score) and score < best_score_overall:
            best_overall = p.copy()
            best_score_overall = float(score)
            print(f"  *** NEW BEST: {score:.6f} (n={len(p)}) ***")

    # Evaluate parent
    if parent is not None:
        sc = _score(parent)
        _try_update(parent, sc)
        print(f"  Parent score: {sc:.6f} (n={len(parent)})")

    if best_overall is None:
        best_overall = _init_uniform(3072)
        best_score_overall = _score(best_overall)
        print(f"  Uniform init score: {best_score_overall:.6f}")

    n_main = len(best_overall)
    print(f"  Main length: {n_main}")

    # ========== PHASE 1: Quick exploration (10% time) ==========
    print("=" * 60)
    print("Phase 1: Quick exploration")
    t1_end = start_time + 0.10 * budget

    explore_n = n_main
    # Try a few different initializations briefly
    inits = []
    if parent is not None:
        inits.append(parent.copy())
    inits.append(_init_cosine(explore_n, rng))
    inits.append(_init_two_level(explore_n, rng))
    inits.append(_init_block_random(explore_n, rng))

    for sp in inits:
        if time.monotonic() > t1_end:
            break
        sc = _score(sp)
        _try_update(sp, sc)
        # Brief optimization
        if time.monotonic() < t1_end - 5:
            t_alloc = min(15.0, t1_end - time.monotonic() - 2)
            if t_alloc > 3:
                p, sc2 = _optimize_main(sp, rng, t_alloc, verbose=False)
                _try_update(p, sc2)

    print(f"  Phase 1 done. Best: {best_score_overall:.6f}")

    # ========== PHASE 2: Main optimization (55% time) ==========
    print("Phase 2: Main optimization")
    t2_end = start_time + 0.65 * budget

    # Continuous optimization on the best
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        p, sc = _optimize_main(best_overall, rng, t_rem, verbose=True)
        _try_update(p, sc)

    # Additional pass with different parameters (perturbed start)
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        p_pert = _resample(best_overall, n_main, rng)
        if p_pert is not None:
            p, sc = _optimize_main(p_pert, rng, t_rem, verbose=False)
            _try_update(p, sc)

    print(f"  Phase 2 done. Best: {best_score_overall:.6f} (n={len(best_overall)})")

    # ========== PHASE 3: Targeted refinement (20% time) ==========
    print("Phase 3: Targeted refinement")
    t3_end = start_time + 0.85 * budget

    if best_overall is not None:
        # Apply convolution square root operations
        t_rem = t3_end - time.monotonic()
        if t_rem > 20:
            n = len(best_overall)
            conv_len = 2 * n - 1
            L = _next_pow2(conv_len + 10)
            p = best_overall.copy()
            for sp in [0.05, 0.1, 0.15, 0.2, 0.3, 0.2, 0.1]:
                if time.monotonic() > t3_end - 5:
                    break
                p_new = _conv_sqrt_smooth(p, sp, L)
                if p_new is not None:
                    sc = _score(p_new)
                    if sc < _score(p):
                        p = p_new
                        _try_update(p, sc)
            # Also try target-based
            for tf in [0.1, 0.15, 0.2, 0.25, 0.3]:
                if time.monotonic() > t3_end - 5:
                    break
                p_new = _conv_sqrt_target(p, tf, L)
                if p_new is not None:
                    sc = _score(p_new)
                    if sc < _score(p):
                        p = p_new
                        _try_update(p, sc)

        # Block refinement
        t_rem = t3_end - time.monotonic()
        if t_rem > 10:
            p_br, sc_br = _block_refine_local(best_overall, rng, 512, [0.5, 0.7, 0.9, 1.1, 1.3, 1.5])
            _try_update(p_br, sc_br)

    print(f"  Phase 3 done. Best: {best_score_overall:.6f}")

    # ========== PHASE 4: Final polishing (15% time) ==========
    print("Phase 4: Final polishing")

    if best_overall is not None:
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(t_rem * 0.6, 180.0)
            p, sc = _optimize_main(best_overall, rng, t_alloc, verbose=True)
            _try_update(p, sc)

        # One more round of sqrt smoothing
        t_rem = deadline - time.monotonic()
        if t_rem > 10 and best_overall is not None:
            n = len(best_overall)
            conv_len = 2 * n - 1
            L = _next_pow2(conv_len + 10)
            p = best_overall.copy()
            for sp in [0.03, 0.05, 0.08, 0.05, 0.03]:
                if time.monotonic() > deadline - 3:
                    break
                p_new = _conv_sqrt_smooth(p, sp, L)
                if p_new is not None:
                    sc = _score(p_new)
                    if sc < _score(p):
                        p = p_new
            _try_update(p, _score(p))

        # Final gradient pass
        t_rem = deadline - time.monotonic()
        if t_rem > 15 and best_overall is not None:
            t_alloc = min(t_rem * 0.7, 100.0)
            p, sc = _optimize_main(best_overall, rng, t_alloc, verbose=False)
            _try_update(p, sc)

    # ========== Final output ==========
    if best_overall is None:
        best_overall = _init_uniform(3072)

    p_final = _sanitize(best_overall)
    if p_final is None:
        p_final = _init_uniform(3072)

    final_score = _score(p_final)
    try:
        eval_score = float(evaluate_sequence([float(x) for x in p_final]))
        if np.isfinite(eval_score):
            final_score = min(final_score, eval_score)
    except Exception:
        pass

    print(f"\n{'=' * 60}")
    print(f"FINAL: score={final_score:.6f} (n={len(p_final)})")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    out = np.where(np.isfinite(p_final), p_final, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = _init_uniform(3072)
    else:
        out = out / s

    return [float(x) for x in out]
