"""
Step function optimizer using LP-guided mirror descent.
Key innovation: solve a focused LP at each iteration to determine optimal
lag weights (dual variables), then use these to guide a mirror descent step.
Combined with diverse restarts and coordinate descent polishing.
"""

import numpy as np
import cvxpy as cp
import time

def _autocorr_fft(f):
    """Full linear convolution a*a via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]

def _score_fast(f):
    """Fast score computation."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))

def _lp_direction(f, top_lags, M, solver='CLP'):
    """
    Solve the focused LP: max sum(g) s.t. conv(f, g)[k] <= M for k in top_lags, g >= 0.
    Returns (g, dual_vars) where g is the optimal solution and dual_vars are the dual values.
    """
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    
    constraints = []
    duals = {}
    
    for k in top_lags:
        # conv(f, g)[k] = sum_j f[j] * g[k-j] for valid j
        # = sum_j f[k-j] * g[j] for valid j (substituting)
        # Specifically: sum over j where 0 <= k-j < n, i.e., max(0, k-n+1) <= j <= min(n-1, k)
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start > j_end:
            continue
        # The coefficients for g[j] in this constraint are f[k-j]
        idx = np.arange(j_start, j_end + 1)
        coeffs = f[k - idx]
        # Build sparse row
        row = cp.sum(coeffs * g[idx])
        cons = row <= M
        constraints.append(cons)
        duals[k] = cons.dual_variable
    
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
    except:
        try:
            prob.solve(solver='CBC')
        except:
            prob.solve(solver='ECOS')
    
    if g.value is None:
        # Fallback: uniform
        return np.ones(n) / n, {k: 1.0/len(top_lags) for k in top_lags}
    
    g_val = np.maximum(g.value, 0)
    if g_val.sum() < 1e-10:
        g_val = np.ones(n) / n
    
    # Get dual values
    dual_vals = {}
    for k, d in duals.items():
        if d.value is not None:
            dual_vals[k] = float(d.value)
        else:
            dual_vals[k] = 0.0
    
    return g_val, dual_vals

def _gradient_from_duals(f, dual_vals):
    """
    Compute gradient using dual variables as lag weights.
    grad[i] = 2 * sum_k lambda_k * (f[k-i] + f[i-k]) for valid indices
    """
    n = len(f)
    grad = np.zeros(n)
    for k, lam in dual_vals.items():
        if lam < 1e-15:
            continue
        # Contribution to grad[i] from lag k:
        # f[k-i] for valid i (0 <= k-i < n → k-n+1 <= i <= k)
        # f[i-k] for valid i (0 <= i-k < n → k <= i <= k+n-1, but i < n so k <= i < n)
        i_start1 = max(0, k - n + 1)
        i_end1 = min(n - 1, k)
        if i_start1 <= i_end1:
            grad[i_start1:i_end1+1] += lam * f[k - np.arange(i_start1, i_end1+1)]
        
        i_start2 = max(0, k)
        i_end2 = min(n - 1, k + n - 1)
        if i_start2 <= i_end2:
            grad[i_start2:i_end2+1] += lam * f[np.arange(i_start2, i_end2+1) - k]
    
    return 2.0 * grad

def _mirror_step(f, grad, eta):
    """Exponentiated gradient step."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy()
    return f_new / s

def _polish_coordinate(f, deadline, rng, max_passes=200):
    """Coordinate descent polish."""
    f = f.copy()
    n = len(f)
    best_score = _score_fast(f)
    best_f = f.copy()
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        b = _autocorr_fft(f)
        peak = b.max()
        peak_lag = int(np.argmax(b))
        
        # Contribution of each position to the peak lag
        contrib = np.zeros(n)
        j_start = max(0, peak_lag - n + 1)
        j_end = min(n - 1, peak_lag)
        if j_start <= j_end:

            contrib[j_start:j_end+1] = f[np.arange(j_start, j_end+1)] * f[idx]

        top5 = np.argpartition(b, -5)[-5:]

            je = min(n - 1, k)

                idx_k = k - np.arange(js, je + 1)
                total_contrib[js:je+1] += f[np.arange(js, je+1)] * f[idx_k]
        
        j_rec = int(np.argmin(total_contrib))
        
        if j_don == j_rec:
            continue

        for frac in [0.05, 0.1, 0.2, 0.3]:

            f_try = f.copy()

            f_try[j_rec] += amt
            f_try = np.maximum(f_try, 0)
            s = f_try.sum()
            if s < 1e-10:

            f_try /= s
            sc = _score_fast(f_try)
            if sc < best_score - 1e-12:
                f = f_try
                best_score = sc
                best_f = f.copy()
                break

"""
Step function optimizer using LP-dual-guided mirror descent with diverse restarts.
Key innovation: solve the dual of a focused convex program to obtain optimal lag
weights, then use these to guide mirror descent steps on the log-simplex.
Combined with coordinate descent polishing and adaptive step sizing.
"""

import numpy as np
import cvxpy as cp
import time
from scipy.sparse import lil_matrix, csr_matrix


def _autocorr_fft(f):
    """Full linear convolution a*a via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _score_fast(f):
    """Fast score: 2*n*max(conv(f,f))/sum(f)^2."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))


def _build_dual_matrix(f, top_lags):
    """Build sparse n x K matrix: A_T[j, idx] = f[top_lags[idx]-j] if valid."""
    n = len(f)
    K = len(top_lags)
    A_T = lil_matrix((n, K))
    for idx, k in enumerate(top_lags):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            rows = np.arange(j_start, j_end + 1)
            cols = np.ones(j_end - j_start + 1, dtype=int) * idx
            vals = f[k - rows]
            A_T[rows, cols] = vals
    return A_T.tocsr()


def _solve_dual_lp(f, top_lags, M):
    """
    Solve dual: min M*sum(lam) s.t. A_T @ lam >= 1, lam >= 0.
    Returns optimal lambda vector (length K).
    """
    n = len(f)
    K = len(top_lags)
    A_T = _build_dual_matrix(f, top_lags)
    
    lam = cp.Variable(K, nonneg=True)
    constraints = [A_T @ lam >= np.ones(n)]
    objective = cp.Minimize(M * cp.sum(lam))
    prob = cp.Problem(objective, constraints)
    
    solvers_to_try = ['CLP', 'CBC', 'ECOS', 'SCS']
    lam_val = None
    for solver in solvers_to_try:
        try:
            prob.solve(solver=solver)
            if lam.value is not None:
                lam_val = np.maximum(lam.value, 0.0)
                break
        except Exception:
            continue
    
    if lam_val is None or not np.isfinite(lam_val).all():
        # Fallback: uniform weights
        lam_val = np.ones(K) / K
    
    # Normalize so that sum is meaningful
    total = lam_val.sum()
    if total < 1e-30:
        lam_val = np.ones(K) / K
    else:
        lam_val = lam_val / total
    
    return lam_val


def _gradient_from_weights(f, top_lags, weights):
    """
    Compute gradient of sum_k w_k * b[k] where b = conv(f,f).
    grad[i] = 2 * sum_k w_k * (f[k-i] + f[i-k]) for valid indices.
    """
    n = len(f)
    grad = np.zeros(n)
    for k, w in zip(top_lags, weights):
        if w < 1e-15:
            continue
        # Term 1: f[k-i] for valid i (0 <= k-i < n)
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        if i_start <= i_end:
            idx = np.arange(i_start, i_end + 1)
            grad[i_start:i_end + 1] += w * f[k - idx]
        # Term 2: f[i-k] for valid i (0 <= i-k < n, i.e., i >= k)
        i_start2 = max(0, k)
        i_end2 = min(n - 1, k + n - 1)
        if i_start2 <= i_end2:
            idx2 = np.arange(i_start2, i_end2 + 1)
            grad[i_start2:i_end2 + 1] += w * f[idx2 - k]
    
    return 2.0 * grad


def _mirror_step(f, grad, eta):
    """Exponentiated gradient (mirror) step on log-simplex."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy()
    return f_new / s


def _linear_mix(f, g, t):
    """Linear mixing: (1-t)*f + t*g, projected to non-negative simplex."""
    f_new = (1.0 - t) * f + t * g
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy()
    return f_new / s

            top_idx = np.sort(top_idx)

def _solve_primal_direction(f, top_lags, M):
    """
    Solve primal LP: max sum(g) s.t. conv(f,g)[k] <= M for top lags, g >= 0.
    Returns normalized g (sum=1).
    """
    n = len(f)
    g = cp.Variable(n, nonneg=True)
    constraints = []
    
    for k in top_lags:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        if i_start > i_end:
            continue
        idx = np.arange(i_start, i_end + 1)
        coeffs = f[k - idx]
        # sum_i f[k-i] * g[i] <= M
        constraint = coeffs @ g[idx] <= M
        constraints.append(constraint)
    
    if not constraints:
        return np.ones(n) / n
    
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    solvers_to_try = ['CLP', 'CBC', 'ECOS']
    for solver in solvers_to_try:
        try:
            prob.solve(solver=solver)
            if g.value is not None:
                g_val = np.maximum(g.value, 0.0)
                gs = g_val.sum()
                if gs > 1e-10:
                    return g_val / gs
                break
        except Exception:
            continue
    
    return np.ones(n) / n


def _optimize_gradient(f0, deadline, rng, max_iters=10000, K_lags=15,
                       verbose=False, label=""):
    """
    Pure gradient-based optimization (no LP) as backup/fast method.
    Uses top-K peak targeting with adaptive steps.
    """
    f = np.maximum(f0, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(len(f))
    else:
        f = f / s
    
    n = len(f)
    best_score = _score_fast(f)
    best_f = f.copy()
    
    eta = 0.5
    eta_min = 0.001
    momentum = np.zeros(n)
    beta = 0.8
    stuck = 0
    patience = 25
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        b = _autocorr_fft(f)
        peak = b.max()
        
        # Top K lags with weight = 1/gap
        K = min(K_lags, 2 * n - 1)
        top_idx = np.argpartition(b, -K)[-K:]
        top_idx = top_idx[np.argsort(b[top_idx])[::-1]]
        top_vals = b[top_idx]
        gaps = np.maximum(top_vals[0] - top_vals, 1e-10)
        weights = 1.0 / gaps
        weights /= weights.sum()
        
        grad = _gradient_from_weights(f, top_idx, weights)
        gmax = np.abs(grad).max()
        if gmax < 1e-30:
            break
        grad = grad / gmax
        
        # Momentum
        momentum = beta * momentum + (1 - beta) * grad
        grad_eff = momentum
        
        # Adaptive step
        progress = it / max_iters
        eta = 0.5 * (eta_min / 0.5) ** (progress ** 0.5)
        
        f_new = _mirror_step(f, grad_eff, eta)
        score_new = _score_fast(f_new)
        
        if score_new < best_score - 1e-12:
            f = f_new
            best_score = score_new
            best_f = f.copy()
            stuck = 0
            if verbose and it % 2000 == 0:
                print(f"    {label} grad it={it}: {best_score:.7f}")
        else:
            stuck += 1
            if stuck >= patience:
                # Perturb
                f = f + rng.normal(0, 0.03 * f.mean(), size=n)
                f = np.maximum(f, 0)
                f /= f.sum()
                momentum *= 0.3
                stuck = 0
    
    return best_f, best_score


def _polish_coordinate(f, deadline, rng, max_passes=500):
    """
    Coordinate descent polish: transfer mass from high-contribution
    positions to low-contribution positions targeting peak lags.
    """
    f = f.copy()
    n = len(f)
    best_score = _score_fast(f)
    best_f = f.copy()
    n_passes_no_improve = 0
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        b = _autocorr_fft(f)
        peak = b.max()
        
        # Focus on top 5 lags
        K_top = min(5, 2 * n - 1)
        top_lags = np.argpartition(b, -K_top)[-K_top:]
        
        # Compute total contribution of each position to top lags
        total_contrib = np.zeros(n)
        for k in top_lags:
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                idx = k - np.arange(j_start, j_end + 1)
                total_contrib[j_start:j_end + 1] += f[np.arange(j_start, j_end + 1)] * f[idx]
        
        improved = False
        # Try several donor-receiver pairs
        top_donors = np.argsort(total_contrib)[::-1][:10]
        top_receivers = np.argsort(total_contrib)[:10]
        
        for j_don in top_donors:
            for j_rec in top_receivers:
                if j_don == j_rec or total_contrib[j_don] < 1e-15:
                    continue
                for frac in [0.05, 0.1, 0.2]:
                    amt = f[j_don] * frac
                    f_try = f.copy()
                    f_try[j_don] -= amt
                    f_try[j_rec] += amt
                    f_try = np.maximum(f_try, 0)
                    s = f_try.sum()
                    if s < 1e-10:
                        continue
                    f_try /= s
                    sc = _score_fast(f_try)
                    if sc < best_score - 1e-13:
                        f = f_try
                        best_score = sc
                        best_f = f.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            n_passes_no_improve += 1
            if n_passes_no_improve > 50:
                # Add small random perturbation to escape
                i1 = int(rng.integers(0, n))
                i2 = int(rng.integers(0, n))
                if i1 != i2:
                    amt = f[i1] * 0.05
                    f[i1] -= amt
                    f[i2] += amt
                    f = np.maximum(f, 0)
                    f /= f.sum()
                n_passes_no_improve = 0
        else:
            n_passes_no_improve = 0
    
    return best_f, best_score


def _spectral_flatness_score(f):
    """Measure how flat the power spectrum is (lower = flatter)."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    power = np.abs(F) ** 2
    # Exclude DC and focus on first n bins
    power_mid = power[1:min(n, size // 2)]
    if len(power_mid) == 0:
        return 0.0
    mean_p = power_mid.mean()
    if mean_p < 1e-30:
        return 0.0
    return np.var(power_mid) / (mean_p * mean_p)

        sc = _score_fast(f)
        start_results.append((sc, name, f))
    
    start_results.sort(key=lambda x: x[0])
    print("\nStarting points (sorted by score):")
    for sc, name, _ in start_results[:6]:
        print(f"  {name}: {sc:.7f}")
    
    best_score = start_results[0][0]
    best_f = start_results[0][2].copy()
    
    # Phase 1: Quick gradient warm-up from top 3 starts
    print(f"\n=== Phase 1: Gradient warm-up ===")
    warmup_time = (budget_s - 400) * 0.15
    n_warmup = 3
    for idx in range(n_warmup):
        if time.monotonic() > deadline - 300:
            break
        sc, name, f0 = start_results[idx]
        t_alloc = warmup_time / n_warmup * (1.0 - 0.15 * idx)
        d = time.monotonic() + t_alloc
        print(f"  Warm-up '{name}' (score={sc:.6f})")
        
        f_opt, sc_opt = _optimize_gradient(f0, d, rng, max_iters=5000,
                                            K_lags=15, verbose=False, label=name)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"    -> New best: {best_score:.7f}")
        else:
            print(f"    -> {sc_opt:.7f}")
    
    # Phase 2: LP-dual-guided mirror descent (main optimization)
    print(f"\n=== Phase 2: LP-guided mirror descent (score={best_score:.7f}) ===")
    main_time = (budget_s - 400) * 0.55
    n_main = 3
    for idx in range(n_main):
        if time.monotonic() > deadline - 200:
            break
        
        if idx == 0:
            f0 = best_f.copy()
            name = "best"
        else:
            # Perturb best for diversity
            f0 = best_f.copy()
            mode = idx - 1
            if mode == 1:
                alpha = rng.uniform(-1.5, 1.5)
                f0 = f0 * np.exp(alpha * (j / n - 0.5))
            else:
                f0 = f0 + rng.normal(0, 0.05 * f0.mean(), size=n)
            f0 = np.maximum(f0, 0)
            f0 = f0 / f0.sum()
            name = f"pert{idx}"
        
        t_alloc = main_time / n_main * (1.0 - 0.1 * idx)
        d = time.monotonic() + t_alloc
        print(f"  Main run '{name}' (budget={t_alloc:.0f}s)")
        
        f_opt, sc_opt = _optimize_lp_mirror(f0, d, rng, max_iters=5000,
                                            K_lags=20, lp_interval=5,
                                            verbose=True, label=name)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"    -> New best: {best_score:.7f}")
        else:
            print(f"    -> Best from '{name}': {sc_opt:.7f}")
    
    # Phase 3: Extended LP-guided from current best
    print(f"\n=== Phase 3: Extended LP optimization (score={best_score:.7f}) ===")
    t3 = min(500.0, max(100.0, (deadline - time.monotonic()) * 0.35))
    if t3 > 60:
        d3 = time.monotonic() + t3
        f_opt, sc_opt = _optimize_lp_mirror(best_f, d3, rng, max_iters=8000,
                                            K_lags=30, lp_interval=3,
                                            verbose=True, label="extended")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"    -> New best: {best_score:.7f}")
        else:
            print(f"    -> {sc_opt:.7f}")
    
    # Phase 4: Coordinate descent polish
    print(f"\n=== Phase 4: Coordinate polish (score={best_score:.7f}) ===")
    t4 = min(150.0, max(30.0, (deadline - time.monotonic()) * 0.3))
    d4 = time.monotonic() + t4
    f_pol, sc_pol = _polish_coordinate(best_f, d4, rng, max_passes=500)
    if sc_pol < best_score:
        best_score = sc_pol
        best_f = f_pol.copy()
        print(f"    -> Polished to: {best_score:.7f}")
    else:
        print(f"    -> {sc_pol:.7f}")
    
    # Phase 5: Final micro-polish
    if time.monotonic() < deadline - 15:
        print(f"\n=== Phase 5: Final polish (score={best_score:.7f}) ===")
        d5 = deadline - 5.0
        f_fin, sc_fin = _polish_coordinate(best_f, d5, rng, max_passes=300)
        if sc_fin < best_score:
            best_score = sc_fin
            best_f = f_fin.copy()
            print(f"    -> Final: {best_score:.7f}")
        
        # One more gradient pass
        if time.monotonic() < deadline - 15:
            d5b = deadline - 5.0
            f_g, sc_g = _optimize_gradient(best_f, d5b, rng, max_iters=5000,
                                           K_lags=10, verbose=False, label="final_grad")
            if sc_g < best_score:
                best_score = sc_g
                best_f = f_g.copy()
                print(f"    -> Grad final: {best_score:.7f}")
    
    # Validate with official evaluator
    best_f = np.clip(best_f, 0.0, 1000.0)
    # Ensure sum is reasonable (not too small)
    if best_f.sum() < 1.0:
        best_f = best_f / best_f.sum() * np.sqrt(2 * n)
    
    final_score = evaluate_sequence([float(x) for x in best_f])
    print(f"\nOfficial evaluation: {final_score:.7f}")
    
    if not np.isfinite(final_score):
        # Fallback to initial
        print("WARNING: Invalid score, using initial!")
        final_score = _score_fast(f_init)
        best_f = f_init
    
    elapsed = time.monotonic() - t_start
    print(f"=== FINAL SCORE: {final_score:.7f} (time: {elapsed:.1f}s) ===")
    
    return [float(x) for x in best_f]
