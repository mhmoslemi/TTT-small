def _normalize(v):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v


def _fft_size(m):
    if m <= 2:
        return 2
    return 1 << (int(m) - 1).bit_length()


def _autocorr(f):
    n = f.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = 2 * n - 1
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]


def _fast_score(f):
    f = _normalize(f)
    if f.size == 0:
        return np.inf
    c = _autocorr(f)
    if c.size == 0:
        return np.inf
    return float(2.0 * f.size * c.max())


def _project_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    out = v - theta
    np.maximum(out, 0.0, out)
    s = float(out.sum())
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    out /= s
    return out


def _gradient_at_lag(k, f):
    """Compute d(c[k])/df = 2*f[k-i] (with bounds) for a single lag k."""
    n = f.size
    g = np.zeros(n, dtype=np.float64)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * f[k - idx]
    return g


def _weighted_gradient(p, f):
    n = f.size
    L = p.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = L + n - 1
    size = _fft_size(m)
    P = np.fft.rfft(p, size)
    F = np.fft.rfft(f[::-1], size)
    conv = np.fft.irfft(P * F, size)
    return 2.0 * conv[n - 1:n + n - 1]


def _transfer_max(c, f, i, j, dv, delta, total):
    n = f.size
    L = c.size
    delta.fill(0.0)
    delta[i:i + n] -= f
    delta[j:j + n] += f
    delta *= (2.0 * dv)
    q = dv * dv
    delta[2 * j] += q
    delta[2 * i] += q
    delta[i + j] -= 2.0 * q
    np.add(c, delta, out=total)
    return float(np.max(total))

            if best_ct is not None and best_mc < maxc - 1e-16:
                f = best_ft
                c = best_ct
                maxc = best_mc
                accepted = True

        if accepted:
            no_accept = 0
            cur_score = float(2.0 * n * maxc)
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()
                if best_score < TARGET_SCORE:
                    print(f"[{tag}] below target: {best_score:.8f}")
            sigma_rel = max(min_sigma_rel, sigma_rel * 0.995)
        else:
            no_accept += 1
            now = time.monotonic()
            if now - last_polish > polish_interval or no_accept > 15:
                # Alternate between coordinate polish and peak chase
                if no_accept % 2 == 0:
                    fp, sc, cp = _coordinate_polish(f, c, polish_budget, rng)
                else:
                    fp, sc, cp = _peak_chase(f, c, polish_budget, rng)
                if sc < float(2.0 * n * maxc) - 1e-14:
                    f = _normalize(fp)
                    c = cp
                    maxc = float(c.max())
                    accepted = True
                    no_accept = 0
                    if sc < best_score:
                        best_score = sc
                        best_f = f.copy()
                        if best_score < TARGET_SCORE:
                            print(f"[{tag}] polish below target: {best_score:.8f}")
                else:
                    if no_accept > 40 or (it % 4000 == 0):
                        strength = min(3.0, 1.0 + no_accept / 30.0)
                        f = _perturb(best_f, rng, strength)
                        c = _autocorr(f)
                        maxc = float(c.max())
                        sigma_rel = min(0.3, sigma_rel * 2.0 + 1e-3)
                        no_accept = 0
                last_polish = time.monotonic()
            else:
                sigma_rel = max(min_sigma_rel, sigma_rel * 0.998)

        if it % 15000 == 0:
            cur_score = float(2.0 * n * maxc)
            print(f"[{tag}] iter={it} best={best_score:.8f} cur={cur_score:.8f}")

    cur_score = float(2.0 * n * float(c.max()))
    if cur_score < best_score:
        best_score = cur_score
        best_f = f.copy()

    if it > 0:
        print(f"[{tag}] end best={best_score:.8f} iters={it}")
    return _normalize(best_f), float(best_score)


def _smooth_with_window(f, win):
    f = _normalize(f)
    n = f.size
    if n <= 2:
        return f
    win = int(win)
    if win < 3:
        win = 3
    if win > n:
        win = max(3, n // 2)
    kernel = np.ones(win, dtype=np.float64) / float(win)
    out = np.convolve(f, kernel, mode="same")
    return _normalize(out)


def _fourier_seed_v2(n, rng, n_frequencies=8):
    """Richer Fourier seed with more frequency components."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    u = np.zeros(n, dtype=np.float64)
    for m in range(1, n_frequencies + 1):
        amp = float(rng.uniform(0.1, 0.6)) / math.sqrt(m)
        phase = float(rng.uniform(0.0, 2.0 * np.pi))
        u += amp * np.cos(2.0 * np.pi * m * x + phase)
    f = np.exp(0.8 * u)
    return _normalize(f)


def _two_component_seed(n, rng):
    """Two exponential components on left and right halves."""
    mid = n // 2
    gl = float(rng.uniform(0.3, 4.0))
    gr = float(rng.uniform(0.3, 4.0))
    wl = float(rng.uniform(0.3, 1.5))
    f = np.zeros(n, dtype=np.float64)
    for i in range(mid):
        f[i] = (i / mid + 0.01) ** (-gl)
    for i in range(n - mid):
        f[mid + i] = (i / (n - mid) + 0.01) ** (-gr)
    f *= wl
    # Add a small uniform component
    f += 0.01
    return _normalize(f)


def _ripple_seed(n, rng):
    """Seed with a few concentrated peaks."""
    f = np.zeros(n, dtype=np.float64)
    n_peaks = int(rng.integers(2, 6))
    positions = rng.choice(n, size=n_peaks, replace=False)
    widths = rng.integers(max(3, n // 50), n // 10 + 1, size=n_peaks)
    heights = rng.uniform(0.5, 2.0, size=n_peaks)
    for pos, w, h in zip(positions, widths, heights):
        start = max(0, pos - w)
        end = min(n, pos + w + 1)
        x = np.arange(start, end)
        f[start:end] += h * np.exp(-((x - pos) ** 2) / (2 * (w / 3) ** 2))
    f += rng.uniform(0.001, 0.01, size=n)
    return _normalize(f)


def _interp_to_length(f, n_new):
    f = _normalize(f)
    m = f.size
    if n_new == 0:
        return np.empty(0, dtype=np.float64)
    if m == n_new:
        return f
    if n_new == 1:
        return np.array([1.0], dtype=np.float64)
    if m == 1:
        return _normalize(np.full(n_new, f[0]))
    x_old = np.linspace(0.0, m - 1, m, dtype=np.float64)
    x_new = np.linspace(0.0, n_new - 1, n_new, dtype=np.float64)
    return _normalize(np.interp(x_new, x_old, f))


def _downsample_to(f, K):
    f = _normalize(f)
    n = f.size
    if K <= 0:
        return np.empty(0, dtype=np.float64)
    if K >= n:
        return f
    x_old = np.linspace(0.0, n - 1, n, dtype=np.float64)
    x_new = np.linspace(0.0, K - 1, K, dtype=np.float64)
    return _normalize(np.interp(x_new, x_old, f))


def _build_seed_pool(f0, n, rng):
    """Create diverse pool of starting sequences."""
    seeds = []
    f0n = _interp_to_length(f0, n)

    def try_add(name, f):
        f = _interp_to_length(f, n)
        try:
            sc = _fast_score(f)
        except Exception:
            return
        if np.isfinite(sc):
            seeds.append((float(sc), len(seeds), name, f))

    try_add("given", f0n)
    try_add("reverse", f0n[::-1])
    for win in (5, 15, 50, 150):
        try_add(f"smooth{win}", _smooth_with_window(f0n, win))
    for g in (0.5, 0.8, 1.2, 1.5, 2.0):
        try_add(f"gamma{g}", np.power(f0n + 1e-12, g))
    try_add("uniform", np.ones(n, dtype=np.float64))

    for i in range(10):
        try_add(f"fourier{i}", _fourier_seed_v2(n, rng, n_frequencies=int(rng.integers(3, 12))))
    for i in range(8):
        try_add(f"two_comp{i}", _two_component_seed(n, rng))
    for i in range(6):
        try_add(f"ripple{i}", _ripple_seed(n, rng))

    # Coarse-fine seeds
    if n > 200:
        for K in (256, 512, 1024, 2048):
            if K < n:
                coarse = _downsample_to(f0n, K)
                # Perturb coarse
                pf = _perturb(coarse, rng, 0.3)
                up = _interp_to_length(pf, n)
                try_add(f"coarse{K}", up)

    if not seeds:
        f = np.ones(max(1, n), dtype=np.float64)
        seeds.append((2.0, 0, "fallback", _normalize(f)))

    seeds.sort(key=lambda x: x[0])
    return seeds


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative coefficient sequence minimizing evaluate_sequence.
    Uses multi-start with diverse seeds, gradient optimization, peak-chasing,
    and coordinate polish. Returns a Python list of floats.
    """
    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    total_deadline = start_time + max(120.0, float(budget_s)) - 30.0

    # Try to load the provided construction
    try:
        f0 = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        f0 = None

    if f0 is not None and f0.size >= 3:
        f0 = np.nan_to_num(f0, nan=0.0, posinf=1000.0, neginf=0.0)
        np.clip(f0, 0.0, 1000.0, out=f0)
        n = int(f0.size)
    else:
        n = 6121
        f0 = np.ones(n, dtype=np.float64)

    f0 = _normalize(f0)
    print(f"Starting: n={n}, initial score={_fast_score(f0):.8f}")

    best_f = f0.copy()
    best_score = _fast_score(f0)

    # Phase 1: Quick seed evaluation (2 min)
    seed_pool = _build_seed_pool(f0, n, rng)
    print(f"Seed pool: {len(seed_pool)} seeds, best seed score={seed_pool[0][0]:.8f}")

    # Update best from seed pool
    if seed_pool[0][0] < best_score:
        best_score = seed_pool[0][0]
        best_f = seed_pool[0][3].copy()

    # Phase 1b: Quick check with different lengths
    now = time.monotonic()
    remaining = total_deadline - now
    if remaining > 2000 and n > 1000:
        # Try a couple of different lengths quickly
        for trial_n in [n // 2, n * 3 // 2]:
            if trial_n < 100 or trial_n > 20000:
                continue
            if total_deadline - time.monotonic() < 100:
                break
            f_trial = _interp_to_length(f0, trial_n)
            sc_trial = _fast_score(f_trial)
            print(f"Trial n={trial_n}: score={sc_trial:.8f}")
            if sc_trial < best_score:
                # Quick optimization at this length
                f_opt, sc_opt = _gradient_optimize(f_trial, 60, rng, f"trial_n{trial_n}")
                if sc_opt < best_score:
                    best_score = sc_opt
                    best_f = f_opt.copy()
                    n = trial_n
                    print(f"  Improved! New best: {best_score:.8f} at n={n}")

    # Phase 2: Main optimization (majority of time)
    now = time.monotonic()
    remaining = total_deadline - now
    final_reserve = min(200.0, max(60.0, 0.12 * remaining))
    main_deadline = total_deadline - final_reserve

    # Select top seeds for optimization, prioritizing diversity
    top_seeds = []
    seen_scores = set()
    for item in seed_pool:
        sc = item[0]
        # Ensure diversity: skip if score is very close to an already-selected seed
        skip = False
        for tsc, _ in top_seeds:
            if abs(sc - tsc) < 0.005:
                skip = True
                break
        if not skip:
            top_seeds.append((sc, item[3]))
        if len(top_seeds) >= 5:
            break

    if not top_seeds:
        top_seeds = [(best_score, best_f.copy())]

    print(f"\nPhase 2: Optimizing {len(top_seeds)} top seeds")
    print(f"Main budget: {main_deadline - now:.1f}s")

    weights = [0.45, 0.25, 0.15, 0.10, 0.05][:len(top_seeds)]
    # Normalize weights
    wsum = sum(weights)
    weights = [w / wsum for w in weights]

    for idx, (sc, f_seed) in enumerate(top_seeds):
        if time.monotonic() >= main_deadline - 30:
            break
        alloc = (main_deadline - time.monotonic()) * (sum(weights[idx:]) / sum(weights[idx:]))
        alloc = max(30.0, min(alloc, 0.55 * (main_deadline - time.monotonic()) + 30))
        # Cap first seed's allocation
        if idx == 0:
            alloc = min(alloc, 0.55 * (main_deadline - now))

        print(f"\n=== Seed {idx+1}/{len(top_seeds)}: score={sc:.8f}, alloc={alloc:.1f}s ===")
        f_opt, sc_opt = _gradient_optimize(f_seed, alloc, rng, f"main{idx}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  New best: {best_score:.8f}")

    # Phase 3: Final intensive polish
    now = time.monotonic()
    final_time = max(0.0, total_deadline - now - 25)
    print(f"\nPhase 3: Final polish, budget={final_time:.1f}s")

    if final_time > 30:
        # Alternate between coordinate polish and peak chase
        t_split = final_time * 0.5
        bf, bs, bc = _coordinate_polish(best_f, _autocorr(best_f), t_split, rng)
        if bs < best_score:
            best_score = bs
            best_f = bf.copy()
            print(f"  After coord polish: {best_score:.8f}")

        remaining = max(0.0, final_time - t_split - 10)
        if remaining > 20:
            bf2, bs2, bc2 = _peak_chase(best_f, _autocorr(best_f), remaining, rng)
            if bs2 < best_score:
                best_score = bs2
                best_f = bf2.copy()
                print(f"  After peak chase: {best_score:.8f}")

    # Phase 4: If time remains, one more gradient pass from a perturbed best
    now = time.monotonic()
    extra = max(0.0, total_deadline - now - 20)
    if extra > 60:
        print(f"\nPhase 4: Extra gradient pass, budget={extra:.1f}s")
        f_pert = _perturb(best_f, rng, 0.3)
        f_opt, sc_opt = _gradient_optimize(f_pert, min(extra, 150), rng, "extra")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()

        # Final coordinate polish
        now = time.monotonic()
        remaining = max(0.0, total_deadline - now - 15)
        if remaining > 20:
            bf, bs, bc = _coordinate_polish(best_f, _autocorr(best_f), remaining, rng)
            if bs < best_score:
                best_score = bs
                best_f = bf.copy()

    # Final safety
    best_f = _normalize(best_f)
    best_f = np.nan_to_num(best_f, nan=0.0, posinf=0.0, neginf=0.0)
    np.clip(best_f, 0.0, 1000.0, out=best_f)
    best_f = best_f / max(1e-300, float(best_f.sum()))

    # Recompute
    c = _autocorr(best_f)
    best_score = float(2.0 * best_f.size * c.max()) if c.size else np.inf

    # Official evaluation
    try:
        out = [float(x) for x in best_f]
        official = evaluate_sequence(out)
        if np.isfinite(official) and official < best_score + 1e-12:
            best_score = float(official)
    except Exception:
        pass

    elapsed = time.monotonic() - start_time
    print(f"\nFINAL: score={best_score:.8f}, n={best_f.size}, time={elapsed:.1f}s")
    print(f"Target: {TARGET_SCORE}")

    return [float(x) for x in best_f]
