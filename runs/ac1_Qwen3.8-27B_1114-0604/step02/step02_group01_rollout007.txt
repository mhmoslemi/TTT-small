def _autocorr_fft(f):
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]

def _score_fast(f):
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))

def _convolve_fft(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(a, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _score(a):
    """Compute the objective function value."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = _convolve_fft(a)
    return 2.0 * n * c.max() / (s * s)


def _peak_shaving_step(a, c, K_peaks=5, n_moves=20, rng=None):
    """
    Try K_peaks-focused moves to reduce the maximum convolution value.
    Returns the best (a_new, score_new) found, or (a, score) if no improvement.
    """
    n = len(a)
    s = a.sum()
    current_max = c.max()
    current_score = 2.0 * n * current_max / (s * s)
    
    # Find top K peak lags
    top_k = np.argpartition(c, -K_peaks)[-K_peaks:]
    top_k = top_k[np.argsort(c[top_k])[::-1]]
    
    best_a = a.copy()
    best_score = current_score
    
    for k_star in top_k:
        # Gradient of c[k_star] w.r.t. a[j] is 2*a[k_star - j] (if valid)
        # Donor: position with largest gradient (most impact on reducing c[k_star])
        grad_don = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(n - 1, k_star)
        if j_start <= j_end:

            grad_don[j_arr] = 2.0 * a[k_star - j_arr]
        
        # Find top few donors
        donor_candidates = np.argsort(grad_don)[::-1][:10]

        # Impact of increasing a[m] on c[k] is 2*a[k-m]
        # Total impact = sum over top-K lags of 2*a[k-m]
        receiver_impact = np.zeros(n)
        for k in top_k:
            js = max(0, k - n + 1)
            je = min(n - 1, k)
            if js <= je:
                j_arr = np.arange(js, je + 1)
                receiver_impact[j_arr] += 2.0 * a[k - j_arr]
        
        # Best receivers: lowest total impact, and not donors
        recv_candidates = np.argsort(receiver_impact)[:10]
        
        # Try moves from top donors to best receivers
        for j_don in donor_candidates[:5]:
            if grad_don[j_don] < 1e-15:
                continue
            for j_rec in recv_candidates[:5]:
                if j_rec == j_don:

    return best_a, best_score


def _sa_step(a, c, temperature, rng, K_peaks=5):
    """
    Single simulated annealing step: propose a move, accept/reject based on temperature.
    """
    n = len(a)
    s = a.sum()
    current_max = c.max()
    current_score = 2.0 * n * current_max / (s * s)
    
    # Find the peak lag
    k_star = int(np.argmax(c))
    
    # Gradient at peak
    grad = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(n - 1, k_star)
    if j_start <= j_end:
        j_arr = np.arange(j_start, j_end + 1)
        grad[j_arr] = 2.0 * a[k_star - j_arr]
    
    # Select donor: weighted random from top gradient positions
    top_donors = np.argsort(grad)[::-1][:min(20, n)]
    donor_probs = grad[top_donors]
    donor_probs = donor_probs / (donor_probs.sum() + 1e-30)
    j_don = top_donors[rng.choice(len(top_donors), p=donor_probs)]
    
    # Select receiver: random position with low a value (prefer "empty" spots)
    # or a position that doesn't contribute to the peak
    non_contrib = np.diff(np.concatenate([[0], np.cumsum(grad == 0), [1]]))
    if non_contrib.sum() > 5:
        # Choose from non-contributing positions
        non_contrib_idx = np.where(grad == 0)[0]
        j_rec = non_contrib_idx[rng.integers(len(non_contrib_idx))]
    else:
        # Choose from low-impact positions
        low_impact = np.argsort(a)[:min(20, n)]
        j_rec = low_impact[rng.integers(len(low_impact))]
    
    if j_rec == j_don:
        j_rec = (j_rec + 1) % n
        if j_rec == j_don:
            j_rec = (j_rec + 1) % n
    
    # Move size: proportional to temperature and current value
    frac = rng.uniform(0.02, 0.3) * (1.0 + temperature)
    delta = a[j_don] * frac
    if delta < 1e-15:
        return a, c, current_score
    
    a_try = a.copy()
    a_try[j_don] = max(0, a_try[j_don] - delta)
    a_try[j_rec] += delta
    s_new = a_try.sum()
    if s_new < 1e-10:
        return a, c, current_score
    
    c_try = _convolve_fft(a_try)
    new_max = c_try.max()
    new_score = 2.0 * n * new_max / (s_new * s_new)
    
    # Acceptance criterion
    if new_score < current_score:
        return a_try, c_try, new_score
    else:
        # Metropolis criterion
        delta_e = new_score - current_score
        if delta_e > 0 and rng.random() < np.exp(-delta_e / max(temperature, 1e-10)):
            return a_try, c_try, new_score
        else:
            return a, c, current_score


def _spectral_diversify(a, rng, strength=0.3):
    """
    Diversify the sequence in the frequency domain while maintaining non-negativity.
    """
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(a, size)
    
    # Add random phase perturbations (preserves magnitude spectrum)
    # This changes the time-domain shape
    n_freq = len(F)
    for i in range(1, n_freq):  # skip DC
        if rng.random() < strength:
            F[i] *= np.exp(1j * rng.uniform(-0.5, 0.5))
    
    # Also add some random magnitude perturbation
    mag = np.abs(F)
    mag = mag * (1.0 + strength * 0.3 * rng.standard_normal(n_freq))
    mag = np.maximum(mag, 0)
    phase = np.angle(F)
    F_new = mag * np.exp(1j * phase)
    
    # Ensure DC is preserved (maintains the sum)
    F_new[0] = F[0]
    
    a_try = np.fft.irfft(F_new, size)[:n]
    
    # Project to non-negative
    a_try = np.maximum(a_try, 0)
    s = a_try.sum()
    if s < 1e-10:
        return a.copy()
    a_try /= s
    return a_try


def _multi_peak_gradient_descent(a, deadline, rng, n_iters=50000):
    """
    Multi-peak focused gradient descent using direct moves.
    More aggressive than SA: always takes the best improving move.
    """
    n = len(a)
    a = a.copy()
    c = _convolve_fft(a)
    current_score = _score(a)
    best_a = a.copy()
    best_score = current_score
    
    stuck = 0
    patience = 200
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        c = _convolve_fft(a)
        current_max = c.max()
        current_score = 2.0 * n * current_max / (a.sum() ** 2)
        
        if current_score < best_score:
            best_score = current_score
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
        
        if stuck > patience:
            # Diversify
            a = _spectral_diversify(a, rng, strength=0.5)
            stuck = 0
            continue
        
        # Find top 3 peaks
        top3 = np.argpartition(c, -3)[-3:]
        top3 = top3[np.argsort(c[top3])[::-1]]
        
        improved = False
        for k_star in top3:
            # Gradient at this peak
            grad = np.zeros(n)
            js = max(0, k_star - n + 1)
            je = min(n - 1, k_star)
            if js <= je:
                j_arr = np.arange(js, je + 1)
                grad[j_arr] = 2.0 * a[k_star - j_arr]
            
            if grad.max() < 1e-15:
                continue
            
            # Best donor
            j_don = int(np.argmax(grad))
            
            # Best receiver: minimize impact on all top peaks
            impact = np.zeros(n)
            for k in top3:
                js2 = max(0, k - n + 1)
                je2 = min(n - 1, k)
                if js2 <= je2:
                    j_arr2 = np.arange(js2, je2 + 1)
                    impact[j_arr2] += a[k - j_arr2]
            
            # Exclude donors
            impact[j_don] = np.inf
            j_rec = int(np.argmin(impact))
            
            if j_rec == j_don:
                continue
            
            # Try the move
            for frac in [0.05, 0.1, 0.2, 0.5]:
                delta = a[j_don] * frac
                if delta < 1e-15:
                    continue
                a_try = a.copy()
                a_try[j_don] -= delta
                a_try[j_rec] += delta
                if a_try[j_don] < -1e-15:
                    continue
                a_try = np.maximum(a_try, 0)
                c_try = _convolve_fft(a_try)
                if c_try.max() < current_max - 1e-12:
                    a = a_try
                    c = c_try
                    current_max = c.max()
                    improved = True
                    break
            
            if improved:
                break
        
        if not improved:
            stuck += 5
    
    return best_a, best_score
