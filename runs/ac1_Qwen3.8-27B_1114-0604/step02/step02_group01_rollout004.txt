    f /= f.sum()
    
    theta = np.log(f)
    theta -= theta.mean()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_score = _score_norm(f)
    best_f = f.copy()
    
    stuck = 0
    patience = 300
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        t += 1
        progress = min(1.0, it / max(1, n_iters))
        
        # Anneal
        tau = tau0 * (tau1 / tau0) ** (progress ** 0.4)
        lr = lr0 * (lr1 / lr0) ** (progress ** 0.6)
        
        # Compute gradient
        g, c = _grad_peaks(f, K=K, tau=tau)
        grad_theta = _grad_2theta(f, g)
        
        # Adam step
        theta, m, v = _adam_update(theta, grad_theta, m, v, t, lr)
        
        # Update f
        f = _softmax(theta)
        
        # Track best
        current_score = 2.0 * n * c.max()  # Use c from before the step
        # Actually, let's compute fresh every few iterations
        if it % 50 == 0 or it == n_iters - 1:
            c_new = _conv_fft(f)
            current_score = 2.0 * n * c_new.max()
            if current_score < best_score - 1e-10:
                best_score = current_score
                best_f = f.copy()
                stuck = 0
                if verbose and it % 5000 == 0:
                    print(f"      {label} iter {it}: {best_score:.7f}")
            else:
                stuck += 1
        
        # Stuck detection and restart
        if stuck >= patience:
            mode = rng.integers(0, 4)
            if mode == 0:
                theta += rng.normal(0, 0.05, size=n)
            elif mode == 1:
                # Perturb high-contribution positions
                peak_idx = np.argmax(c) if 'c' in dir() and c is not None else 0
                # Just do a moderate random kick
                theta += rng.normal(0, 0.1, size=n)
            elif mode == 2:
                # Partial reset: keep top 50% of theta (in magnitude)
                abs_theta = np.abs(theta - theta.mean())
                threshold = np.median(abs_theta)
                mask = abs_theta < threshold
                theta[mask] += rng.normal(0, 0.1, size=mask.sum())
            else:
                # Just reduce momentum
                m *= 0.3
                v *= 0.3
            
            m = np.zeros(n)
            v = np.zeros(n)
            stuck = 0
    
    # Final
    c = _conv_fft(best_f)
    best_score = float(2.0 * n * c.max())
    return best_f, best_score

def _coordinate_polish(f, deadline, rng, max_rounds=200):
    """
    Pairwise coordinate descent: find pairs (i,j) where moving mass from i to j
    reduces the peak convolution.
    """
    n = len(f)
    f = f.copy()
    best_score = _score_norm(f)
    best_f = f.copy()
    
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = _conv_fft(f)
        peak = c.max()
        peak_idx = np.argmax(c)
        
        # For the peak lag k*, find which positions contribute most
        k = peak_idx
        m_start = max(0, k - n + 1)
        m_end = min(n - 1, k)
        
        if m_start > m_end:
            break
        
        # Contribution of each m to C[k]
        idx = k - np.arange(m_start, m_end + 1)
        contrib = f[m_start:m_end+1] * f[idx]
        
        # Try: move mass from high-contrib position to low-contrib position
        # Sort by contribution
        order = np.argsort(contrib)
        
        # Try a few swaps
        improved = False
        for trial in range(min(5, len(order))):
            # Donor: high contribution
            j_don = m_start + order[-1 - trial]
            # Receiver: low contribution (or outside the peak's support)
            j_rec = m_start + order[trial]
            
            if j_don == j_rec:
                continue
            
            amount = f[j_don] * 0.1
            if amount < 1e-15:
                continue
            
            f_try = f.copy()
            f_try[j_don] -= amount
            f_try[j_rec] += amount
            f_try = np.maximum(f_try, 0)
            f_try /= f_try.sum()
            
            c_try = _conv_fft(f_try)
            if c_try.max() < peak - 1e-12:
                f = f_try
                peak = c_try.max()
                improved = True
                best_score = 2.0 * n * peak
                best_f = f.copy()
        
        if not improved:
            # Try random moves
            for _ in range(3):
                i1 = rng.integers(0, n)
                i2 = rng.integers(0, n)
                if i1 == i2:
                    continue
                amt = f[i1] * 0.05
                f_try = f.copy()
                f_try[i1] -= amt
                f_try[i2] += amt
                f_try = np.maximum(f_try, 0)
                f_try /= f_try.sum()
                c_try = _conv_fft(f_try)
                if c_try.max() < peak - 1e-12:
                    f = f_try
                    peak = c_try.max()
                    improved = True
                    best_score = 2.0 * peak
                    best_f = f.copy()
                    break
    
    return best_f, best_score

def _spectral_init(n, rng):
    """Generate a random non-negative sequence with approximately flat spectrum."""
    # Create random phases in frequency domain
    size = 1 << (2 * n - 1).bit_length()
    F = np.zeros(size // 2 + 1, dtype=np.complex128)
    
    # Set DC component
    F[0] = 1.0  # will be normalized later
    
    # Random phases for other frequencies
    for k in range(1, size // 2 + 1):
        angle = rng.uniform(0, 2 * np.pi)
        F[k] = np.exp(1j * angle)
    
    # Reconstruct
    f = np.fft.irfft(F, size)[:n]
    
    # Ensure non-negativity
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f /= s
    
    return f

def _known_good_init(n):
    """Generate a structured initialization with good properties."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    #combination of several structures
    components = [
        np.ones(n),
        np.exp(-((j - center) / (n * 0.3))**2),
        1.0 + 0.3 * np.cos(2 * np.pi * 3 * j / n),
        1.0 + 0.2 * np.cos(2 * np.pi * 7 * j / n),
        np.exp(-((j / n - 0.5)**2) / 0.05),
    ]
    
    # Weighted combination
    f = np.zeros(n)
    weights = [0.5, 0.3, 0.1, 0.05, 0.05]
    for comp, w in zip(components, weights):
        f += w * comp
    
    f = np.maximum(f, 0)
    f /= f.sum()
    return f
