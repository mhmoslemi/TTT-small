"""
Optimized search for non-negative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses multiplicative gradient descent with adaptive smoothed-max, frequency-domain
convolution flattening, and multi-length exploration. Builds on parent sequence
with targeted improvements: better gradient handling, adaptive temperature scheduling,

import numpy as np
import time


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_fft(a, b, L):
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:len(a) + len(b) - 1]


def _score(p):
    n = len(p)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    pn = p / s
    L = _next_pow2(2 * n - 1)
    c = _conv_fft(pn, pn, L)
    return float(2.0 * n * c.max())


def _score_raw(p):
    """Score for already-normalized sequence (sum=1)."""
    n = len(p)
    L = _next_pow2(2 * n - 1)
    c = _conv_fft(p, p, L)
    return float(2.0 * n * c.max())


def _gradient_smoothed(p, beta, L):
    """
    Returns (score, gradient, max_conv) for smoothed objective.
    p should be normalized (sum=1).
    gradient[j] = 4n * sum_m w[j+m] * p[m]
    where w = softmax(beta * n * c)
    """
    n = len(p)
    Lc = _next_pow2(2 * n - 1)
    c = _conv_fft(p, p, Lc)
    d = n * c
    d_max = d.max()
    score = float(2.0 * d_max)

    if beta < 1.0:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        shifted = d - d_max
        w = np.exp(beta * shifted)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(2 * n - 1)
            w[n - 1] = 1.0
        else:
            w = w / ws

    L2 = _next_pow2(3 * n - 2)
    W = np.fft.rfft(w, L2)
    P = np.fft.rfft(p, L2)
    R = np.fft.irfft(W * np.conj(P), L2)
    g = 4.0 * n * R[:n]
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.maximum(g, 0.0)
    return score, g, float(d_max)


def _freq_flatten(p, alpha, L=None):
    """
    Frequency-domain convolution flattening.
    Makes the convolution flatter by modifying the spectrum.
    alpha > 1: more flattening.
    """
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(p, L)
    C_freq = A * A
    C = np.fft.irfft(C_freq, L)[:2 * n - 1]
    cmax = C.max()
    if cmax < 1e-15:
        return None
    C_norm = C / cmax
    C_new = cmax * np.power(C_norm, 1.0 / alpha)
    C_new = np.maximum(C_new, 0.0)
    C_freq_new = np.fft.rfft(C_new, L)
    # Take square root in frequency domain
    A_new = np.sqrt(C_freq_new.astype(complex) + 0j)
    a_new = np.fft.irfft(A_new, L)[:n]
    neg_frac = (a_new < -1e-10).sum() / n
    if neg_frac > 0.25:
        return None
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s


def _freq_flatten_v2(p, alpha, L=None):
    """
    Alternative frequency flattening: modify |A(f)| directly.
    Target: make |A(f)| more uniform.
    """
    n = len(p)

        L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(p, L)
    mag = np.abs(A)
    ang = np.angle(A)
    if mag.max() < 1e-15:
        return None
    mag_norm = mag / mag.max()
    # Raise to power < 1 to flatten
    mag_new = mag.max() * np.power(mag_norm, 1.0 / alpha)
    A_new = mag_new * np.exp(1j * ang)
    a_new = np.fft.irfft(A_new, L)[:n]
    neg_frac = (a_new < -1e-10).sum() / n
    if neg_frac > 0.3:
        return None
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s


def _block_refine(p, rng, n_candidates=200):
    """
    Block-level multiplicative refinement. Try scaling random blocks
    and keep improvements.
    """
    n = len(p)
    best = p.copy()
    best_sc = _score_raw(p)
    cur = p.copy()
    cur_sc = best_sc
    factors = (0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 1.8, 2.5)

    for _ in range(n_candidates):
        r = rng.random()
        if r < 0.3:

        elif r < 0.7:

        else:

        bs = min(bs, n)
        start = int(rng.integers(0, n - bs + 1))

        improved = False
        for f in factors:
            p_try = cur.copy()
            p_try[start:start + bs] *= f

                best_sc = cur_sc
        else:

            if rng.random() < 0.1:
                cur = best.copy()
                cur_sc = best_sc

def _random_perturb(p, rng, magnitude=0.1):
    """Random multiplicative perturbation."""
    n = len(p)
    log_p = np.log(np.maximum(p, 1e-15))
    log_p += rng.normal(0, magnitude, n)
    new_p = np.exp(log_p)
    new_p = np.maximum(new_p, 0.0)
    s = new_p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return p.copy()
    return new_p / s

                    cur_score = sc_new
                    stall = 0
                    if sc_new < best_score - 1e-12:
                        best = p.copy()
                        best_score = sc_new
                else:
                    stall += 1
            else:
                stall += 1

        elif r < 0.92:
            # Block scaling
            bs = int(rng.choice([2, 4, 8, 16, 32, 64, 128, 256]))
            bs = min(bs, n)
            start = int(rng.integers(0, n - bs + 1))
            f = float(rng.choice([0.3, 0.5, 0.7, 1.4, 1.8, 2.5, 4.0]))
            p_try = p.copy()
            p_try[start:start + bs] *= f
            p_try = np.maximum(p_try, 0.0)
            s_try = p_try.sum()
            if s_try > 1e-10:
                p_try = p_try / s_try
                sc_try = _score_raw(p_try)
                if sc_try < cur_score - 1e-13:
                    p = p_try
                    cur_score = sc_try
                    stall = 0
                    if sc_try < best_score - 1e-12:
                        best = p.copy()
                        best_score = sc_try
                else:
                    stall += 1
            else:
                stall += 1

        else:
            # Random perturbation (especially when stalled)
            if stall > 3:
                mag = min(0.3, 0.05 + 0.01 * min(stall, 20))
                p_try = _random_perturb(p, rng, mag)
                sc_try = _score_raw(p_try)
                if sc_try < max(cur_score, best_score * 1.05):
                    p = p_try
                    cur_score = sc_try
                    stall = 0
            else:
                stall += 1

        # Update beta
        if it % 8 == 0:
            beta = min(beta_end, beta * 1.1)
        if it % 100 == 99:
            # Occasionally reset beta to allow global exploration
            if rng.random() < 0.1:
                beta = beta_start

        # Periodic pullback
        if cur_score > best_score * 1.1 and best_score < float('inf'):
            if rng.random() < 0.3:
                p = best.copy()
                cur_score = best_score
                stall = 0

        it += 1

    return best, best_score


def _optimize_with_blocks(p0, rng, time_budget):
    """Interleave gradient optimization with block refinement."""
    p = _optimize_main(p0, rng, time_budget * 0.6, beta_start=30, beta_end=3000)
    if p is None:
        return None, float('inf')
    sc = _score_raw(p)
    # Block refine
    p2, sc2 = _block_refine(p, rng, n_candidates=300)
    if sc2 < sc:
        return p2, sc2
    return p, sc


def _init_parent_resampled(parent, n, rng=None):
    p = np.asarray(parent, dtype=float)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return None
    p = p / s
    n_old = len(p)
    if n_old == n:
        result = p.copy()
    elif n_old == 1:
        result = np.ones(n, dtype=float)
    else:
        x_old = (np.arange(n_old) + 0.5) / n_old
        x_new = (np.arange(n) + 0.5) / n
        result = np.interp(x_new, x_old, p)
    result = np.maximum(result, 0.0)
    if rng is not None:
        result = result * np.exp(rng.normal(0, 0.02, n))
        result = np.maximum(result, 0.0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s


def _init_structured(n, rng, mode=0):
    """Various structured initializations."""
    t = np.arange(n, dtype=float) / n
    if mode == 0:
        # Two-level with smooth transition
        k = int(n * rng.uniform(0.3, 0.7))
        ratio = float(rng.uniform(1.5, 4.0))
        p = np.ones(n, dtype=float)
        p[k:] = ratio
        idx = rng.permutation(n)
        p = p[idx]
    elif mode == 1:
        # Sum of Gaussians
        K = int(rng.integers(2, 8))
        p = np.zeros(n, dtype=float)
        for _ in range(K):
            c = rng.uniform(0.0, 1.0)
            s = rng.uniform(0.02, 0.3)
            w = rng.uniform(0.5, 2.0)
            x = (t - c) / s
            x = np.minimum(np.abs(x), 1.0 - np.abs(x))
            p += w * np.exp(-0.5 * x * x)
    elif mode == 2:
        # Cosine modulation
        p = np.ones(n, dtype=float)
        for k in range(1, 12):
            amp = rng.uniform(-0.5, 0.5) / k
            ph = rng.uniform(0, 2 * np.pi)
            p += amp * np.cos(2 * np.pi * k * t + ph)
    elif mode == 3:
        # Block random
        K = int(rng.integers(4, 20))
        p = np.zeros(n, dtype=float)
        edges = np.sort(rng.uniform(0, 1, K - 1))
        blocks = np.concatenate([[0], edges, [1]])
        heights = rng.lognormal(0, 1, K)
        for i in range(K):
            s_idx = int(round(blocks[i] * n))
            e_idx = max(s_idx + 1, int(round(blocks[i + 1] * n)))
            e_idx = min(n, e_idx)
            p[s_idx:e_idx] = heights[i]
    else:
        # Edge-enhanced: higher at boundaries
        dist = np.minimum(t, 1 - t)
        p = 1.0 + rng.uniform(0.5, 2.0) * np.exp(-dist / rng.uniform(0.05, 0.2))
        p = p + 0.1 * rng.normal(0, 1, n)

    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return np.ones(n, dtype=float) / n
    return p / s


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Multi-phase: exploration -> gradient optimization -> spectral refinement -> polishing.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 2400.0

    start_time = time.monotonic()
    deadline = start_time + budget - 20

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng(iseed % (2**31 - 1))

    best_overall = None
    best_score_overall = float('inf')

    def _try_update(p, score):
        nonlocal best_overall, best_score_overall
        if p is not None and np.isfinite(score) and score < best_score_overall:
            best_overall = p.copy()
            best_score_overall = float(score)
            print(f"  *** NEW BEST: {score:.6f} (n={len(p)}) at t={time.monotonic()-start_time:.0f}s ***")

    # Get parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=float)
        parent = np.where(np.isfinite(parent), parent, 0.0)
        parent = np.maximum(parent, 0.0)
        if parent.sum() < 1e-10:
            parent = None
    except Exception:
        parent = None

    # Evaluate parent
    if parent is not None:
        sc = _score(parent)
        _try_update(parent, sc)
        print(f"Parent: score={sc:.6f}, n={len(parent)}")
    n_parent = len(parent) if parent is not None else 3072

    # ========== PHASE 1: Quick exploration (10% time) ==========
    print("=" * 60)
    print(f"Phase 1: Exploration (budget={budget:.0f}s)")
    t1_end = start_time + 0.10 * budget

    # Try several lengths and initializations
    lengths_to_try = [2048, 3072, 4096, 6144]
    if n_parent in (1536, 2048, 2560, 3072, 4096, 5120, 6144):
        if n_parent not in lengths_to_try:
            lengths_to_try.insert(0, n_parent)

    for n in lengths_to_try:
        if time.monotonic() > t1_end:
            break
        inits = []
        if parent is not None:
            sp = _init_parent_resampled(parent, n, rng)
            if sp is not None:
                inits.append(sp)
        for mode in range(5):
            inits.append(_init_structured(n, rng, mode))

        for sp in inits:
            sc = _score_raw(sp)
            _try_update(sp, sc)

    print(f"  Phase 1 done. Best: {best_score_overall:.6f}")

    if best_overall is None:
        best_overall = np.ones(3072, dtype=float) / 3072
        best_score_overall = _score_raw(best_overall)

    # ========== PHASE 2: Main gradient optimization (40% time) ==========
    print("Phase 2: Main optimization")
    t2_end = start_time + 0.50 * budget

    n_main = len(best_overall)
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(t_rem * 0.45, 400.0)
        t_alloc = max(30.0, t_alloc)
        print(f"  Run 1: n={n_main}, budget={t_alloc:.0f}s, low beta")
        p, sc = _optimize_main(best_overall, rng, t_alloc,
                               beta_start=15.0, beta_end=2000.0, verbose=True)
        _try_update(p, sc)

    # Higher beta (sharper focus)
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(250.0, t_rem * 0.6)
        t_alloc = max(20.0, t_alloc)
        p1 = _init_parent_resampled(best_overall, n_main, rng) if parent is not None else best_overall.copy()
        if p1 is None:
            p1 = best_overall.copy()
        print(f"  Run 2: n={n_main}, budget={t_alloc:.0f}s, high beta")
        p, sc = _optimize_main(p1, rng, t_alloc,
                               beta_start=100.0, beta_end=8000.0, verbose=True)
        _try_update(p, sc)

    # Try different length
    t_rem = t2_end - time.monotonic()
    if t_rem > 60:
        n_up = min(8192, n_main * 2)
        if n_up != n_main:
            t_alloc = min(150.0, t_rem * 0.5)
            t_alloc = max(20.0, t_alloc)
            p_up = _init_parent_resampled(best_overall, n_up, rng)
            if p_up is not None:
                print(f"  Run 3: upsample to n={n_up}, budget={t_alloc:.0f}s")
                p, sc = _optimize_main(p_up, rng, t_alloc,
                                       beta_start=50.0, beta_end=4000.0, verbose=True)
                _try_update(p, sc)

    print(f"  Phase 2 done. Best: {best_score_overall:.6f} (n={len(best_overall)})")

    # ========== PHASE 3: Diverse restarts + block refinement (20% time) ==========
    print("Phase 3: Diverse strategies")
    t3_end = start_time + 0.72 * budget

    # Block refinement
    t_rem = t3_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(120.0, t_rem * 0.4)
        p_br, sc_br = _block_refine(best_overall, rng, n_candidates=int(t_alloc * 5))
        _try_update(p_br, sc_br)
        # Continue optimizing from block-refined point
        if sc_br < best_score_overall * 1.01:
            t_rem = t3_end - time.monotonic()
            if t_rem > 20:
                t_alloc = min(80.0, t_rem)
                p, sc = _optimize_main(p_br, rng, t_alloc,
                                       beta_start=200.0, beta_end=6000.0, verbose=False)
                _try_update(p, sc)

    # Fresh restarts at best length
    n_best = len(best_overall)
    t_rem = t3_end - time.monotonic()
    if t_rem > 40:
        for mode in [0, 1, 2, 3, 4]:
            if time.monotonic() > t3_end - 10:
                break
            sp = _init_structured(n_best, rng, mode)
            # Small perturbation of best
            sp = sp * (1 + 0.1 * rng.normal(0, 1, n_best))
            sp = np.maximum(sp, 0)
            sp = sp / sp.sum()
            t_alloc = min(40.0, (t3_end - time.monotonic()) * 0.3)
            t_alloc = max(10.0, t_alloc)
            p, sc = _optimize_main(sp, rng, t_alloc,
                                   beta_start=50.0, beta_end=3000.0, verbose=False)
            _try_update(p, sc)

    print(f"  Phase 3 done. Best: {best_score_overall:.6f}")

    # ========== PHASE 4: Intensive polishing (30% time) ==========
    print("Phase 4: Polishing")
    t4_end = deadline

    if best_overall is not None:
        # High-beta gradient polishing
        t_rem = t4_end - time.monotonic()
        if t_rem > 30:
            t_alloc = min(0.3 * t_rem, 200.0)
            p, sc = _optimize_main(best_overall, rng, t_alloc,
                                   beta_start=500.0, beta_end=15000.0, verbose=True)
            _try_update(p, sc)

        # Block refine
        t_rem = t4_end - time.monotonic()
        if t_rem > 20:
            n_cand = int(max(100, t_rem * 8))
            p_br, sc_br = _block_refine(best_overall, rng, n_candidates=n_cand)
            _try_update(p_br, sc_br)

        # Another gradient pass with moderate beta
        t_rem = t4_end - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.3 * t_rem, 150.0)
            p, sc = _optimize_main(best_overall, rng, t_alloc,
                                   beta_start=100.0, beta_end=5000.0, verbose=True)
            _try_update(p, sc)

        # Final block refinement
        t_rem = t4_end - time.monotonic()
        if t_rem > 10:
            n_cand = int(max(50, t_rem * 10))
            p_br, sc_br = _block_refine(best_overall, rng, n_candidates=n_cand)
            _try_update(p_br, sc_br)

    # ========== Final validation ==========
    if best_overall is None:
        best_overall = np.ones(2048, dtype=float) / 2048

    p_final = np.asarray(best_overall, dtype=float)
    p_final = np.where(np.isfinite(p_final), p_final, 0.0)
    p_final = np.maximum(p_final, 0.0)
    s = p_final.sum()
    if s < 1e-10 or not np.isfinite(s):
        p_final = np.ones(2048, dtype=float) / 2048
        s = 1.0
    p_final = p_final / s

    final_score = _score_raw(p_final)

    # Verify with official evaluator
    try:
        eval_score = float(evaluate_sequence([float(x) for x in p_final]))
        if np.isfinite(eval_score):
            print(f"  Official eval score: {eval_score:.6f}")
            final_score = min(final_score, eval_score)
    except Exception:
        pass

    print(f"\n{'=' * 60}")
    print(f"FINAL: score={final_score:.6f} (n={len(p_final)})")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    out = np.where(np.isfinite(p_final), p_final, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = np.ones(2048, dtype=float) / 2048
    return [float(x) for x in out]
