from scipy.optimize import linprog
import time

def autocorr(f):
    n = len(f)
    size = 1 << (2*n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2*n-1]

def score(f):
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = autocorr(f)
    return 2.0 * n * c.max() / (s*s)

def solve_LP(a, M):
    """Solve: max sum(g) s.t. g >= 0, (a*g)[k] <= M for all k."""
    n = len(a)
    # (a*g)[k] = sum_j a[j] * g[k-j] for valid j, k-j
    # This is a matrix-vector product: A @ g <= M
    # A has shape (2n-1, n), A[k, j] = a[j] if 0 <= k-j < n, else 0
    
    # For efficiency, use the fact that A is a Toeplitz-like matrix
    # A[k, j] = a[j] * 1[0 <= k-j <= n-1] = a[j] * 1[j <= k <= j+n-1]
    
    # Build A as a sparse matrix or use the convolution structure
    # Actually, for n ~ 6000, A has ~ n*(n/2) = ~18M non-zeros, which is large.
    # Let's use scipy.sparse.
    
    from scipy.sparse import csr_matrix
    # Row k, column j: A[k,j] = a[j] if max(0, k-n+1) <= j <= min(n-1, k)
    # For each j, it appears in rows k = j, j+1, ..., j+n-1 (clamped to [0, 2n-2])
    
    rows = []
    cols = []
    vals = []
    for j in range(n):
        k_start = j
        k_end = min(2*n-2, j + n - 1)
        # Wait, (a*g)[k] = sum_j a[j] * g[k-j], so for a fixed j, 
        # g[j'] appears in (a*g)[k] where j' = k-j, i.e., k = j + j'
        # So a[j] * g[j'] contributes to row k = j + j'
        # For fixed j, j' ranges over [0, n-1], so k ranges over [j, j+n-1]
        k_start = j
        k_end = min(2*n-2, j + n - 1)
        rows.extend(range(k_start, k_end + 1))
        cols.extend([j] * (k_end - k_start + 1))
        vals.extend([a[j]] * (k_end - k_start + 1))
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Maximize sum(g) = -minimize -sum(g)
    c = -np.ones(n)
    # Constraints: A @ g <= M * ones(2n-1), g >= 0
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return res.x
    else:
        return np.zeros(n)
