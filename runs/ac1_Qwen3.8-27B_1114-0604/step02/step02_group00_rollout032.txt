def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    
    # Gradient computation
    # g[i] = conv(flip(a), p)[n-1+i] + conv(a, p)[i]
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()

"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and its gradient (corrected)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # dL/da_i = 2 * conv(flip(a), p)[n-1+i]
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()

    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def combined_score_grad(a, beta, lam):
    """Combined LSE + L2 objective: LSE_score + lam * L2_score."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    p = exp_scaled / exp_scaled.sum()
    c2_sum = float(np.dot(c, c))
    score = 2.0 * n * L / (S * S) + lam * c2_sum / (S ** 4)
    # Gradient of LSE part

def run_lbfgs(obj_func, a0, args=(), maxiter=3000, maxfun=50000, beta=None):
    """Generic L-BFGS-B wrapper."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        obj_func, a0, args=args,
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    return a_out, res

def build_inits(n, rng, parent=None):

    inits = []
    if parent is not None:
        p = np.asarray(parent, dtype=np.float64).ravel()
        if len(p) != n:

            x_old = np.linspace(0, 1, len(p))
            x_new = np.linspace(0, 1, n)
            p = np.interp(x_new, x_old, p)
        p = np.clip(p, 0.0, 1000.0)
        if p.sum() > 0.01:
            inits.append(p)
    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Structured modulations
    x = np.arange(n, dtype=np.float64) / n
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(2.0 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x))
    inits.append(0.3 + 0.4 * np.cos(2.0 * np.pi * x) ** 2)
    # Gaussian shape
    center = n / 2.0
    sigma = n / 4.0
    inits.append(np.exp(-0.5 * ((np.arange(n) - center) / sigma) ** 2))
    # Double-peak

    # Random
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.7 * rng.standard_normal(n), 0.01, 1000.0))
    # Sparse-like (every k-th element)
    for period in [3, 5]:
        a = np.zeros(n)
        a[::period] = 0.5 + 0.2 * rng.random(n // period + 1)[: (n + period - 1) // period]
        inits.append(a)
    # Block structure
    for nb in [4, 8, 16, 32]:
        per = max(1, n // nb)
        vals = 0.2 + 0.6 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]]
                              )
        inits.append(a[:n].copy())
    return inits


def peak_shave(a, rng, n_trials=5):
    """Try to reduce the peak by targeted perturbations."""
    n = len(a)
    best = a.copy()
    best_sc = compute_score(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    # Elements contributing to c[k_star]: a[j] where j in [max(0,k_star-n+1), min(k_star, n-1)]
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    for _ in range(n_trials):
        a_pert = a.copy()
        # Reduce top contributors
        contrib = np.zeros(n)
        contrib[j_start:j_end + 1] = a_pert[j_start:j_end + 1][::-1]
        top_k = max(2, n // 100)
        top_idx = np.argsort(contrib)[::-1][:top_k]
        bottom_mask = np.ones(n, dtype=bool)
        bottom_mask[top_idx] = False
        bottom_idx = np.where(bottom_mask)[0]
        strength = 0.02 + 0.08 * rng.random()
        total_red = 0.0
        for j in top_idx:
            red = a_pert[j] * strength
            a_pert[j] -= red
            a_pert[j] = max(a_pert[j], 0.0)
            total_red += red
        if len(bottom_idx) > 0 and total_red > 0:
            per = total_red / len(bottom_idx)
            a_pert[bottom_idx] += per
        a_pert = np.clip(a_pert, 0.0, 1000.0)
        sc = compute_score(a_pert)
        if sc < best_sc:
            best_sc = sc
            best = a_pert.copy()
            a = a_pert.copy()  # Continue from improved point
    return best_sc, best


def subgradient_polish(a, n_iters=200):
    """Polyak-style subgradient on exact max objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Subgradient of M w.r.t. a
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.3)
        a = np.clip(a - step * grad, 0.0, 1000.0)
    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Multi-phase: L2 flattening -> LSE peak (continuation) -> local search -> polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    effective_budget = max(30.0, float(budget_s)) - 15.0
    deadline = t_start + effective_budget

    # Load parent solution
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine lengths to try
    lengths = []
    if parent is not None:
        pl = len(parent)
        # Round to nearby power of 2 or use as-is
        lengths.append(pl)
    for n in [2048, 4096, 8192, 16384]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))
    # Limit to at most 4 lengths
    lengths = lengths[:4]

    print(f"Lengths: {lengths}, parent={'yes' if parent is not None else 'no'}")
    print(f"Budget: {budget_s}s")

    # =================================================================
    # Phase 1: L2 flattening (25% of time)
    # =================================================================
    ph1_deadline = t_start + 0.25 * effective_budget
    print(f"\n=== Phase 1: L2 Flattening (until t={ph1_deadline - t_start:.0f}s) ===")

    candidates = []  # Store (score, a, n) for later phases

    for n in lengths:
        if time.monotonic() > ph1_deadline:
            break
        print(f"  L2 phase: n={n}")
        inits = build_inits(n, rng, parent)
        # Limit inits for time management
        n_inits = min(len(inits), max(4, int(60.0 / (time.monotonic() - t_start + 1) * 0.25)))
        n_inits = min(n_inits, len(inits))
        n_inits = max(3, n_inits)

        for i in range(n_inits):
            if time.monotonic() > ph1_deadline:
                break
            a0 = inits[i]
            a_opt, _ = run_lbfgs(l2_wrapped, a0, maxiter=2000, maxfun=30000)
            sc = compute_score(a_opt)
            report(sc, a_opt, f"L2 n={n} init={i}")
            candidates.append((sc, a_opt.copy(), n))

    print(f"  After L2 phase: best={best_score:.8f}, candidates={len(candidates)}")

    # =================================================================
    # Phase 2: LSE peak optimization with continuation (45% of time)
    # =================================================================
    ph2_deadline = t_start + 0.70 * effective_budget
    print(f"\n=== Phase 2: LSE Peak (until t={ph2_deadline - t_start:.0f}s) ===")

    beta_schedule = [30, 100, 300, 1000, 3000]

    # Sort candidates by score, take top few as starting points
    candidates.sort(key=lambda x: x[0])
    top_candidates = candidates[:min(6, len(candidates))]

    # Also ensure we have entries at different lengths
    seen_lengths = set()
    final_starts = []
    for sc, a, n in top_candidates:
        if n not in seen_lengths:
            final_starts.append((sc, a.copy(), n))
            seen_lengths.add(n)
        if len(final_starts) >= 4:
            break
    # Fill remaining from top
    for sc, a, n in top_candidates:
        if (sc, a, n) not in final_starts and len(final_starts) < 6:
            final_starts.append((sc, a.copy(), n))

    # If no candidates, use the current best
    if not final_starts:
        for n in lengths[:2]:
            final_starts.append((best_score, best_a[:n].copy() if len(best_a) >= n else np.ones(n)*0.5, n))

    for sc0, a0, n in final_starts:
        if time.monotonic() > ph2_deadline:
            break
        print(f"  LSE: n={n}, start_score={sc0:.8f}")
        a_cur = a0.copy()
        for beta in beta_schedule:
            if time.monotonic() > ph2_deadline:
                break
            remaining = ph2_deadline - time.monotonic()
            if remaining < 10:
                break
            # Scale iterations based on remaining time
            maxit = min(5000, max(200, int(remaining * 2)))
            a_opt, _ = run_lbfgs(lse_wrapped, a_cur, args=(beta,),
                                  maxiter=maxit, maxfun=max(1000, maxit * 10))
            sc = compute_score(a_opt)
            report(sc, a_opt, f"LSE n={n} beta={beta}")
            a_cur = a_opt

    print(f"  After LSE phase: best={best_score:.8f}")

    # =================================================================
    # Phase 3: Combined LSE+L2 local refinement (15% of time)
    # =================================================================
    ph3_deadline = t_start + 0.85 * effective_budget
    print(f"\n=== Phase 3: Combined Refinement ===")

    n_best = len(best_a)
    # Try several perturbation + refinement cycles
    n_trials = 20
    for trial in range(n_trials):
        if time.monotonic() > ph3_deadline:
            break
        strategy = trial % 5
        if strategy == 0:
            # Gaussian perturbation
            scale = 0.02 + 0.06 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Peak shaving then LSE
            _, a_pert = peak_shave(best_a, rng, n_trials=3)
        elif strategy == 2:
            # Sinusoidal perturbation
            freq = 1.0 + 8.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.02 + 0.04 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Swap two blocks
            blen = max(10, n_best // 50)
            i = int(rng.integers(0, max(1, n_best - 2 * blen)))
            j = int(rng.integers(0, max(1, n_best - 2 * blen)))
            if abs(i - j) > blen:
                a_pert = best_a.copy()
                a_pert[i:i+blen], a_pert[j:j+blen] = a_pert[j:j+blen].copy(), a_pert[i:i+blen].copy()
            else:
                a_pert = best_a.copy()
                scale = 0.03
                a_pert = np.clip(best_a * (1.0 + scale * rng.standard_normal(n_best)), 0, 1000)
        else:
            # Random mask (zero out random fraction, redistribute)
            frac = 0.05 + 0.1 * rng.random()
            a_pert = best_a.copy()
            mask = rng.random(n_best) < frac
            total = a_pert[mask].sum()
            a_pert[mask] = 0
            kept = ~mask
            if kept.sum() > 0 and total > 0:
                a_pert[kept] += total / kept.sum() * (1 + 0.5 * rng.standard_normal(kept.sum()))
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Refine with combined objective
        remaining = ph3_deadline - time.monotonic()
        if remaining < 5:
            break
        maxit = min(2000, max(100, int(remaining * 1.5)))
        # First try combined, then pure LSE
        a_opt, _ = run_lbfgs(combined_wrapped, a_pert, args=(500, 0.5),
                              maxiter=maxit, maxfun=max(500, maxit * 5))
        sc = compute_score(a_opt)
        improved = report(sc, a_opt, f"comb #{trial} s={strategy}")
        if improved and time.monotonic() < ph3_deadline - 10:
            # Follow up with pure LSE
            a_opt2, _ = run_lbfgs(lse_wrapped, a_opt, args=(1000,),
                                   maxiter=min(2000, max(500, int((ph3_deadline - time.monotonic()) * 1.5))),
                                   maxfun=10000)
            sc2 = compute_score(a_opt2)
            report(sc2, a_opt2, f"comb+#LSE {trial}")

    print(f"  After refinement: best={best_score:.8f}")

    # =================================================================
    # Phase 4: Final polish (15% of time)
    # =================================================================
    print(f"\n=== Phase 4: Final Polish ===")

    # High-beta LSE
    for beta in [2000, 5000, 10000]:
        if time.monotonic() > deadline - 10:
            break
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        maxit = min(5000, max(500, int(remaining * 0.8)))
        a_opt, _ = run_lbfgs(lse_wrapped, best_a, args=(beta,),
                              maxiter=maxit, maxfun=max(2000, maxit * 8))
        sc = compute_score(a_opt)
        report(sc, a_opt, f"polish LSE beta={beta}")

    # Subgradient polish
    if time.monotonic() < deadline - 20:
        n_sg = max(100, min(500, int((deadline - time.monotonic()) * 5)))
        sc, a_sg = subgradient_polish(best_a, n_iters=n_sg)
        report(sc, a_sg, "subgrad")

        # One more LSE from polished
        if time.monotonic() < deadline - 15:
            a_opt, _ = run_lbfgs(lse_wrapped, a_sg, args=(3000,),
                                  maxiter=3000, maxfun=30000)
            sc = compute_score(a_opt)
            report(sc, a_opt, "post-subgrad LSE")

    # Final verification
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 256

    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
