```python
def lp_step(f, M=None):
    """
    Given f (normalized to sum=1), find g that maximizes sum(g)
    subject to g>=0 and (f*g)_k <= M for all k.
    Return g (unnormalized) and the LP value.
    """
    n = len(f)
    if M is None:
        M = np.max(np.convolve(f, f))
    
    # Build constraint matrix A (2n-1) x n, sparse
    # A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    rows = []
    cols = []
    vals = []
    for j in range(n):
        for k in range(j, j + n):  # k-j ranges from 0 to n-1
            if k < 2*n - 1:
                rows.append(k)
                cols.append(j)
                vals.append(f[k-j])
    
    from scipy.sparse import csr_matrix
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Maximize sum(g) subject to A@g <= M, g >= 0
    # scipy linprog minimizes, so negate objective
    c = -np.ones(n)
    A_ub = A
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        g = res.x
        return g, -res.fun
    return None, None
```

```python
minimizing 2*n*max(conv(a,a)) / sum(a)^2.

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

_GLOBAL_RNG = np.random.default_rng(42)

def _conv_len(n):
    need = 2 * n - 1
    L = 1
    while L < need:
        L <<= 1
    return L

def _score(p):
    """Compute score = 2*n*max(conv(p,p)) with p normalized to sum=1."""
    n = len(p)
    if n == 0:
        return float('inf')
    L = _conv_len(n)
    Fp = np.fft.rfft(p, L)
    c = np.fft.irfft(Fp * Fp, L)
    c = c[:2*n-1]
    c = np.maximum(c, 0.0)
    return float(2.0 * n * c.max()), c

def _sanitize(p, normalize=True):
    a = np.asarray(p, dtype=float).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if normalize:
        s = float(a.sum())
        if s <= 0:
            return None
        a = a / s
    return a

def _lp_direction(f, M=None, verbose=False):
    """
    Find LP direction: maximize sum(g) s.t. g>=0, f*g <= M.
    Returns normalized g (sum=1) and LP objective value.
    """
    n = len(f)
    if M is None:
        sc, c = _score(f)
        M = c.max()
    
    # Build sparse constraint matrix A (2n-1) x n
    # A[k,j] = f[k-j] if 0 <= k-j <= n-1, else 0
    rows = np.repeat(np.arange(n), n)  # for each column j, n rows
    # Actually, let me build it more carefully
    row_idx = []
    col_idx = []
    vals = []
    for j in range(n):
        # column j has non-zeros at rows k = j, j+1, ..., j+n-1 (clipped to [0, 2n-2])
        k_start = j
        k_end = min(j + n, 2*n - 1)
        for k in range(k_start, k_end):

            col_idx.append(j)
            vals.append(f[k-j])

def _resample(p, m):
    """Resample sequence p to length m."""
    n = len(p)
    if n == m:
        return p.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(m) + 0.5) / m
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 1e-12)
    out = out / out.sum()
    return out

def _init_cosine(n, rng):
    """Cosine-tapered initialization."""
    t = np.arange(n) / (n - 1)
    p = 0.5 * (1 + np.cos(2 * np.pi * t))
    p = np.maximum(p, 1e-10)
    return p / p.sum()

def _init_gaussian(n, rng):
    """Gaussian initialization."""
    t = np.linspace(-1, 1, n)
    sigma = rng.uniform(0.2, 0.6)
    p = np.exp(-t**2 / (2 * sigma**2))
    return p / p.sum()

def _init_two_block(n, rng):
    """Two-block initialization with random parameters."""
    cut = rng.uniform(0.3, 0.7)
    ratio = rng.uniform(0.2, 3.0)
    idx = int(cut * n)
    p = np.ones(n)
    p[idx:] = ratio
    return p / p.sum()

def _init_random_smooth(n, rng):
    """Random smooth initialization."""
    K = rng.integers(3, 10)
    knots = np.sort(rng.uniform(0, 1, K))
    heights = np.exp(rng.normal(0, 0.5, K))
    t = np.linspace(0, 1, n)
    p = np.interp(t, knots, heights)
    p = np.maximum(p, 1e-10)
    return p / p.sum()

def _init_from_spectrum(n, rng):
    """
    Initialization inspired by flat spectrum design.
    Generate a random phase, take inverse FFT, clip to non-negative.
    """
    # Want |A(omega)|^2 to be flat -> A(omega) has flat magnitude
    # So a = IFFT(exp(i*random_phase)) but this gives complex values
    # Instead: a = |IFFT(random)| gives a non-negative "envelope"
    freqs = np.fft.rfftfreq(n)
    # Flat magnitude with random phase
    phase = _GLOBAL_RNG.uniform(0, 2*np.pi, len(freqs))
    A = np.exp(1j * phase)
    a = np.fft.irfft(A * n, n)  # scale by n
    a = np.abs(a) + 1e-10
    return a / a.sum()
```

```python
def _build_conv_matrix_fast(f, n):
    """Build sparse convolution matrix more efficiently."""
    # Use the structure: A is a "convolution matrix"
    # Each column j is a shifted version of f
    # We can use scipy.sparse.diags or build it block by block

    # Create the lower triangle part and upper triangle part
    
    # For n=4096, let's just do it with lists (should be ~5 seconds)
    row_idx = np.zeros(n*n, dtype=np.int32)
    col_idx = np.zeros(n*n, dtype=np.int32)

        k_start = j
        k_end = min(j + n, 2*n - 1)
        length = k_end - k_start
        row_idx[idx:idx+length] = np.arange(k_start, k_end)
        col_idx[idx:idx+length] = j
        vals[idx:idx+length] = f[:length]
        idx += length

    row_idx = row_idx[:idx]
    col_idx = col_idx[:idx]
    vals = vals[:idx]
```

```python
"""

minimizing 2*n*max(conv(a,a)) / sum(a)^2.
```

```python
import numpy as np
import time
from scipy.sparse import csr_matrix, diags as sp_diags
from scipy.optimize import linprog

_GLOBAL_RNG = np.random.default_rng(999)
_CONV_LEN_CACHE = {}


def _conv_len(n):
    n = int(max(1, n))
    if n in _CONV_LEN_CACHE:
        return _CONV_LEN_CACHE[n]
    need = 2 * n - 1
    L = 1
    while L < need:
        L <<= 1
    _CONV_LEN_CACHE[n] = L
    return L


def _score_fft(p, L=None):
    """Compute (2*n*max(conv), convolution) using FFT."""
    n = int(p.size)
    if n == 0:
        return float('inf'), np.array([])
    if L is None:
        L = _conv_len(n)
    Fp = np.fft.rfft(p, L)
    c = np.fft.irfft(Fp * Fp, L)
    c = c[:2 * n - 1]
    c = np.maximum(c, 0.0)
    sc = float(2.0 * n * c.max())
    return sc, c


def _sanitize(p, normalize=True):
    a = np.asarray(p, dtype=float).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if normalize:
        s = float(a.sum())
        if s <= 0.0 or not np.isfinite(s):
            return None
        a = a / s
    return a


def _build_conv_matrix(f):
    """Build sparse (2n-1)x n convolution matrix A where (A@g)[k] = (f*g)[k]."""
    n = int(f.size)
    # Diagonal k (0-indexed sub-diagonal) has value f[k] in all n columns
    diags_data = [np.full(n, f[k]) for k in range(n)]
    offsets = list(range(n))
    A = sp_diags(diags_data, offsets, shape=(2 * n - 1, n), format='csr')
    return A


def _lp_direction(f, max_time=120.0, verbose=False):
    """
    Solve LP: maximize sum(g) s.t. A@g <= M, g >= 0, where M = max(f*f).
    Returns normalized g (sum=1) or None.
    """
    n = int(f.size)
    sc, c = _score_fft(f)
    M = float(c.max())
    if M <= 0:
        return None

    t0 = time.monotonic()
    A = _build_conv_matrix(f)
    t_build = time.monotonic() - t0

    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0.0, None)] * n

    t0 = time.monotonic()
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': float(max_time)})
    except Exception as e:
        if verbose:
            print(f"  LP error: {e}")
        return None
    t_lp = time.monotonic() - t0

    if verbose:
        print(f"  LP: n={n}, build={t_build:.2f}s, solve={t_lp:.2f}s, "
              f"status={res.status}, obj={-res.fun if res.fun is not None else 'N/A'}")

    if res.status in (0, 2) and res.x is not None:
        g = np.asarray(res.x, dtype=float)
        g = np.maximum(g, 0.0)
        s = float(g.sum())
        if s > 1e-12:
            return g / s
    return None

    n = int(f.size)

    best = f.copy()

    cur = f.copy()
    cur_sc = best_sc
    it = 0

    while it < max_iters:

        sc, c = _score_fft(cur, L)

        if best_sc <= 0:

        lp_time = 60.0 if n <= 4096 else 90.0
        if deadline is not None:

            lp_time = min(lp_time, max(5.0, rem * 0.4))

        g = _lp_direction(cur, max_time=lp_time, verbose=verbose)
        if g is None:
            if verbose:
                print(f"  LP failed at iter {it}, stopping LP phase")
            break

        # Line search for mixing parameter
        found = False
        for t_try in [t_mix, t_mix * 0.5, t_mix * 0.25, t_mix * 0.1, t_mix * 0.05]:

            h_sc, _ = _score_fft(h, L)

                t_mix = t_try

                best_try_sc = sc2
                best_try = p_try
                best_try_a = a

        if best_try is not None:
            cur = best_try
            cur_sc = best_try_sc
            alpha = best_try_a
            stall = 0
        else:
            stall += 1
            alpha = max(0.02, alpha * 0.75)

            if stall >= 15:
                # Perturbation restart
                std = 0.05 + min(0.2, 0.01 * stall)
                cur = best * np.exp(_GLOBAL_RNG.normal(0, std, n))
                cur = np.maximum(cur, 1e-15)
                cur = cur / cur.sum()
                cur_sc, _ = _score_fft(cur, L)
                alpha = 0.1
                stall = 0

            if stall > 150:
                cur = best.copy()
                cur_sc = best_sc
                alpha = 0.05
                beta = max(50.0, beta * 0.8)
                stall = 0

        if it % 50 == 0:
            beta = min(2000.0, beta * 1.05)

        it += 1

    return best, best_sc


def _resample_smooth(p, m, smooth_sigma=1.5):
    """Resample p to length m with Gaussian smoothing (improves score via spreading)."""
    n = int(p.size)
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, p)
    # Apply Gaussian smoothing
    if smooth_sigma > 0 and m > 10:
        rad = int(max(1, np.ceil(3.0 * smooth_sigma)))
        k = np.arange(-rad, rad + 1, dtype=float)
        kern = np.exp(-0.5 * (k / smooth_sigma) ** 2)
        kern = kern / kern.sum()
        out = np.convolve(out, kern, mode='same')
    out = np.maximum(out, 1e-12)
    out = out / out.sum()
    return out


def _resample_basic(p, m):
    """Basic resampling without extra smoothing."""
    n = int(p.size)
    if n == m:
        return p.copy()
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, p)
    out = np.maximum(out, 1e-12)
    out = out / out.sum()
    return out


def _init_variants(n, rng):
    """Generate diverse initializations."""
    t = (np.arange(n) + 0.5) / float(n)
    inits = []

    # 1. Cosine taper
    p = 0.5 * (1.0 + np.cos(2.0 * np.pi * t))
    p = np.maximum(p, 1e-10)
    inits.append(p / p.sum())

    # 2. Gaussian
    sigma = rng.uniform(0.15, 0.5)
    p = np.exp(-((t - 0.5) ** 2) / (2.0 * sigma * sigma))
    inits.append(p / p.sum())

    # 3. Two-block
    cut = rng.uniform(0.35, 0.65)
    ratio = rng.uniform(0.3, 2.5)
    idx = int(cut * n)
    p = np.ones(n)
    p[idx:] = ratio
    inits.append(p / p.sum())

    # 4. Random smooth (spline-like)
    K = rng.integers(4, 12)
    knots = np.sort(rng.uniform(0, 1, K))
    heights = np.exp(rng.normal(0, 0.8, K))
    p = np.interp(t, knots, heights)
    p = np.maximum(p, 1e-10)
    inits.append(p / p.sum())

    # 5. Spectral (flat magnitude, random phase)
    freqs = np.fft.rfftfreq(n)
    phase = rng.uniform(0, 2 * np.pi, len(freqs))
    A = np.exp(1j * phase)
    a = np.abs(np.fft.irfft(A * n, n)) + 1e-10
    inits.append(a / a.sum())

    # 6. Raised cosine squared
    p = (0.5 * (1.0 + np.cos(2.0 * np.pi * t))) ** 2
    inits.append(p / p.sum())

    # 7. Power law edges
    a = rng.uniform(0.1, 2.0)
    b = rng.uniform(0.1, 2.0)
    tt = np.clip(t, 1e-8, 1.0 - 1e-8)
    p = tt ** a * (1.0 - tt) ** b
    inits.append(p / p.sum())

    return inits


def _local_block_search(p, time_limit, verbose=False):
    """Local search by scaling random blocks."""
    n = int(p.size)
    L = _conv_len(n)
    best = p.copy()
    best_sc, _ = _score_fft(p, L)
    cur = p.copy()
    cur_sc = best_sc

    deadline = time.monotonic() + float(max(0.0, time_limit))
    stagnant = 0
    it = 0

    small = max(1, n // 4096)
    mid = max(small + 1, n // 512)
    large = max(mid + 1, min(n, n // 64))
    factors = (0.3, 0.5, 0.7, 0.9, 1.1, 1.3, 1.5, 2.0)

    while time.monotonic() < deadline:
        r = float(_GLOBAL_RNG.random())
        if r < 0.25:
            w = 1
        elif r < 0.50:
            w = int(_GLOBAL_RNG.integers(small, min(n, small * 8) + 1))
        elif r < 0.80:
            w = int(_GLOBAL_RNG.integers(mid, min(n, mid * 6) + 1))
        else:
            w = int(_GLOBAL_RNG.integers(large, min(n, large * 6) + 1))
        w = max(1, min(n, w))
        start = int(_GLOBAL_RNG.integers(0, n - w + 1))

        found = False
        for f in factors:
            cand = cur.copy()
            cand[start:start + w] *= float(f)
            cand = np.maximum(cand, 1e-15)
            s = float(cand.sum())
            if s <= 0 or not np.isfinite(s):
                continue
            cand = cand / s
            sc, _ = _score_fft(cand, L)
            if sc < cur_sc - 1e-12:
                cur = cand
                cur_sc = sc
                found = True
                if sc < best_sc:
                    best_sc = sc
                    best = cand.copy()
                    if verbose and it % 500 == 0:
                        print(f"    block it={it} best={best_sc:.7f}")
                break

        if not found:
            stagnant += 1
            if stagnant > 30:
                std = 0.04 + min(0.2, 0.005 * stagnant)
                cur = best * np.exp(_GLOBAL_RNG.normal(0, std, n))
                cur = np.maximum(cur, 1e-15)
                cur = cur / cur.sum()
                cur_sc, _ = _score_fft(cur, L)
                if cur_sc > best_sc * 1.5:
                    cur = best.copy()
                    cur_sc = best_sc
                stagnant = 0
            if stagnant > 200:
                cur = best.copy()
                cur_sc = best_sc
                stagnant = 0

        it += 1

    return best, best_sc


def propose_candidate(budget_s=2400, **kwargs):
    """
    Main search function: hybrid LP + multiplicative update optimization.
    Returns the best sequence found within the time budget.
    """
    global _GLOBAL_RNG
    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except (TypeError, ValueError):
        budget = 2400.0
    if not np.isfinite(budget) or budget < 30.0:
        budget = 30.0
    # Leave safety margin
    total_time = max(20.0, budget - 15.0)
    deadline = start_time + total_time

    _GLOBAL_RNG = np.random.default_rng(2024)
    cands = []  # list of (score, array)

    # Load parent if available
    parent = None
    try:
        parent_raw = height_sequence_1
        if parent_raw is not None:
            p = _sanitize(parent_raw)
            if p is not None:
                parent = p
                sc, _ = _score_fft(p)
                cands.append((sc, p.copy()))
                print(f"Parent: n={len(p)}, score={sc:.7f}")
    except (NameError, Exception):
        pass

    if not cands:
        p = np.ones(2048, dtype=float) / 2048.0
        cands.append((2.0, p))
        print("No parent, using uniform init")

    # ================================================================
    # PHASE 1: Small-scale exploration (n=1024 to 2048)
    # ================================================================
    t_rem = deadline - time.monotonic()
    if t_rem > 100:
        n_small = 1024
        t_p1 = min(60.0, t_rem * 0.12)
        print(f"\nPhase 1: Small scale (n={n_small}), time={t_p1:.0f}s")

        inits = _init_variants(n_small, _GLOBAL_RNG)
        if parent is not None:
            inits.append(_resample_smooth(parent, n_small, 1.0))

        for i, init_p in enumerate(inits):
            t_check = deadline - time.monotonic()
            if t_check < 30:
                break
            sc, _ = _score_fft(init_p)
            # Quick LP (2 iterations max)
            t_lp = min(10.0, max(2.0, (t_p1 * 0.6) / max(1, len(inits) - i)))
            if t_lp > 4:
                p_lp, sc_lp = _lp_iterations(init_p, max_iters=2, t_mix=0.4,
                                             deadline=time.monotonic() + t_lp)
            else:
                p_lp, sc_lp = init_p, sc
            # Short multiplicative
            t_mu = min(8.0, deadline - time.monotonic() - 20)
            if t_mu > 4:
                p_mu, sc_mu = _multiplicative_updates(p_lp, t_mu, target=1.500)
            else:
                p_mu, sc_mu = p_lp, sc_lp
            cands.append((sc_mu, p_mu))
            print(f"  init {i}: n={n_small}, score={sc_mu:.7f}")

    # ================================================================
    # PHASE 2: Medium scale with LP (n=2048, 4096)
    # ================================================================
    t_rem = deadline - time.monotonic()
    if t_rem > 150:
        cands.sort(key=lambda x: x[0])
        best_so_far = cands[0][1]

        for n_med in [2048, 4096]:
            t_rem = deadline - time.monotonic()
            if t_rem < 80:
                break
            t_alloc = min(t_rem * 0.25, 180.0)
            if t_alloc < 30:
                break
            t_deadline = time.monotonic() + t_alloc

            print(f"\nPhase 2: Medium scale (n={n_med}), time={t_alloc:.0f}s")

            # Resample with smoothing
            p_res = _resample_smooth(best_so_far, n_med, smooth_sigma=1.5)
            sc_init, _ = _score_fft(p_res)
            print(f"  Initial (resampled): score={sc_init:.7f}")
            cands.append((sc_init, p_res.copy()))

            # LP iterations
            max_lp = 6 if n_med <= 2048 else 4
            t_lp_alloc = t_alloc * 0.5
            p_lp, sc_lp = _lp_iterations(p_res, max_iters=max_lp, t_mix=0.3,
                                         deadline=time.monotonic() + t_lp_alloc,
                                         verbose=True)
            cands.append((sc_lp, p_lp))
            print(f"  After LP: score={sc_lp:.7f}")

            # Multiplicative refinement
            t_mu = t_deadline - time.monotonic() - 5
            if t_mu > 10:
                p_mu, sc_mu = _multiplicative_updates(p_lp, t_mu, target=1.500,
                                                      verbose=True)
                cands.append((sc_mu, p_mu))
                print(f"  After mult: score={sc_mu:.7f}")

            # Update best
            cands.sort(key=lambda x: x[0])
            best_so_far = cands[0][1]

    # ================================================================
    # PHASE 3: Large scale (n=8192, 16384)
    # ================================================================
    t_rem = deadline - time.monotonic()
    if t_rem > 200:
        cands.sort(key=lambda x: x[0])
        best_med = cands[0][1]

        # Choose target large scale
        n_large = 16384 if t_rem > 600 else 8192
        t_alloc = min(t_rem * 0.4, 400.0)
        t_deadline = time.monotonic() + t_alloc

        print(f"\nPhase 3: Large scale (n={n_large}), time={t_alloc:.0f}s")

        # Only do LP if n is manageable
        if n_large <= 8192:
            p_res = _resample_smooth(best_med, n_large, smooth_sigma=2.0)
            sc_init, _ = _score_fft(p_res)
            print(f"  Initial (resampled): score={sc_init:.7f}")
            cands.append((sc_init, p_res.copy()))

            # LP (fewer iterations, more expensive per iter)
            max_lp = 3
            t_lp_alloc = min(120.0, t_alloc * 0.3)
            t_lp_deadline = time.monotonic() + t_lp_alloc
            p_lp, sc_lp = _lp_iterations(p_res, max_iters=max_lp, t_mix=0.25,
                                         deadline=t_lp_deadline, verbose=True)
            cands.append((sc_lp, p_lp))
            print(f"  After LP: score={sc_lp:.7f}")
            p_cur = p_lp
        else:
            p_res = _resample_smooth(best_med, n_large, smooth_sigma=2.5)
            sc_init, _ = _score_fft(p_res)
            print(f"  Initial (resampled, no LP): score={sc_init:.7f}")
            cands.append((sc_init, p_res.copy()))
            p_cur = p_res

        # Multiplicative refinement at large scale
        t_mu = t_deadline - time.monotonic() - 10
        if t_mu > 20:
            p_mu, sc_mu = _multiplicative_updates(p_cur, t_mu, target=1.500,
                                                  verbose=True)
            cands.append((sc_mu, p_mu))
            print(f"  After mult: score={sc_mu:.7f}")

    # ================================================================
    # PHASE 4: Final refinement
    # ================================================================
    t_rem = deadline - time.monotonic()
    if t_rem > 40:
        cands.sort(key=lambda x: x[0])
        best_p = cands[0][1].copy()
        best_sc = cands[0][0]
        n_best = len(best_p)
        print(f"\nPhase 4: Final refinement, n={n_best}, score={best_sc:.7f}, "
              f"time remaining={t_rem:.0f}s")

        # Try LP on best (if feasible)
        if n_best <= 4096 and t_rem > 60:
            t_lp = min(40.0, t_rem * 0.3)
            p_lp, sc_lp = _lp_iterations(best_p, max_iters=3, t_mix=0.2,
                                         deadline=time.monotonic() + t_lp,
                                         verbose=True)
            if sc_lp < best_sc:
                best_p, best_sc = p_lp, sc_lp
                print(f"  After final LP: {best_sc:.7f}")
            cands.append((sc_lp, p_lp))

        # Multiplicative updates
        t_mu = deadline - time.monotonic() - 20
        if t_mu > 15:
            p_mu, sc_mu = _multiplicative_updates(best_p, t_mu, target=1.500,
                                                  verbose=True)
            if sc_mu < best_sc:
                best_p, best_sc = p_mu, sc_mu
            cands.append((sc_mu, p_mu))
            print(f"  After mult: {best_sc:.7f}")

        # Local block search
        t_rem = deadline - time.monotonic() - 10
        if t_rem > 20:
            p_blk, sc_blk = _local_block_search(best_p, t_rem, verbose=True)
            if sc_blk < best_sc:
                best_p, best_sc = p_blk, sc_blk
            cands.append((sc_blk, p_blk))
            print(f"  After block: {best_sc:.7f}")

        # One more multiplicative pass
        t_rem = deadline - time.monotonic() - 5
        if t_rem > 10:
            p_final, sc_final = _multiplicative_updates(best_p, t_rem, target=1.500)
            if sc_final < best_sc:
                best_p, best_sc = p_final, sc_final
            cands.append((sc_final, p_final))

    # ================================================================
    # Final selection and output
    # ================================================================
    cands.sort(key=lambda x: x[0])
    final_p = cands[0][1].copy()
    final_sc = cands[0][0]

    # Try a few candidates with the official evaluator for small sequences
    if len(final_p) <= 8192:
        for sc_c, p_c in cands[:5]:
            if sc_c > final_sc + 0.01:
                break
            try:
                lst = [float(x) for x in p_c]
                sc_off = float(evaluate_sequence(lst))
                if np.isfinite(sc_off) and sc_off < final_sc:
                    final_sc = sc_off
                    final_p = p_c.copy()
                    print(f"  Official evaluator found better: {sc_off:.7f}")
            except Exception:
                pass

    print(f"\n{'='*60}")
    print(f"FINAL: n={len(final_p)}, score={final_sc:.7f}")
    print(f"{'='*60}")

    # Sanitize output
    final_p = np.where(np.isfinite(final_p), final_p, 0.0)
    final_p = np.clip(final_p, 0.0, 1000.0)
    s = float(final_p.sum())
    if s <= 0 or not np.isfinite(s):
        final_p = np.ones(1024, dtype=float) / 1024.0
    else:
        final_p = final_p / s

    return [float(x) for x in final_p]
```
