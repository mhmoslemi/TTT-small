    c = _autocorr_fft(f)
    return 2.0 * n * c.max() / (s * s)

def _soft_gradient(f, c, tau, k_peaks=None):
    """Compute soft gradient using exponential weighting of lags."""
    n = len(f)
    # Weight lags by softmax
    max_c = c.max()
    weights = np.exp((c - max_c) / tau)
    weights /= weights.sum()
    
    # grad[j] = sum_k weights[k] * 2 * f[k-j] (for valid k-j)
    grad = np.zeros(n)

        lo = max(0, k - n + 1)

        idx = k - np.arange(lo, hi)

    n = len(f)

        hi = min(n, idx + 1)
        idx_arr = idx - np.arange(lo, hi)
        grad[lo:hi] += 2.0 * f[idx_arr]
    grad /= k

Step function optimizer using FFT-based soft gradient mirror descent with adaptive
temperature, diverse window-function initializations, Nesterov-style momentum,
and multi-phase coordinate descent polish. Key improvements over parent:
(1) O(n log n) soft gradient via FFT cross-correlation instead of O(n*k) top-k,
(2) adaptive temperature schedule for gradient focus,
(3) momentum in log-space for faster convergence,
(4) systematic coordinate polish with exact peak-lag sensitivity,
(5) broader and more structured initialization set.

import numpy as np
import time
from scipy.special import i0 as bessel_i0


def _conv_fft(f):
    """Full linear convolution f*f via real FFT."""
    n = len(f)
    size = 1 << (2 * n - 2).bit_length()
    F = np.fft.rfft(f, size)
    result = np.fft.irfft(F * F, size)
    return result[:2 * n - 1]


def _score_from_conv(f, c):
    """Compute 2*n*max(c)/(sum f)^2."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def _soft_grad_fft(f, c, tau):
    """
    Gradient of (1/tau)*sum_k exp(c[k]/tau) w.r.t. f.
    Computed as 2 * cross_correlation(w, f) via FFT.
    """
    n = len(f)
    max_c = c.max()
    w = np.exp((c - max_c) / tau)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= w_sum
    size = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, size)
    F = np.fft.rfft(f, size)
    cross = np.fft.irfft(W * np.conj(F), size)
    return 2.0 * cross[:n]


def _topk_grad_fft(f, c, k):
    """
    Average subgradient from top-k lags, computed via FFT for speed.
    Uses a hard (uniform) weight on the top-k lags.
    """
    n = len(f)
    k_eff = min(k, len(c))
    top_k = np.argpartition(c, -k_eff)[-k_eff:]
    w = np.zeros_like(c)
    w[top_k] = 1.0 / k_eff
    size = 1 << (3 * n - 2).bit_length()
    W = np.fft.rfft(w, size)
    F = np.fft.rfft(f, size)
    cross = np.fft.irfft(W * np.conj(F), size)
    return 2.0 * cross[:n]

    """
    Mirror descent (exponentiated gradient) step in log-space.
    Optionally uses momentum on the gradient.
    Returns new f and updated momentum.
    """
    if momentum_v is not None:
        grad_eff = beta * momentum_v + (1.0 - beta) * grad
        new_v = grad_eff
    else:
        grad_eff = grad
        new_v = None

    scale = grad_eff.max() - grad_eff.min()
    if scale < 1e-18:
        return f.copy(), new_v

    log_f = np.log(f + 1e-300) - (eta / scale) * grad_eff
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy(), new_v
    f_new /= s
    return f_new, new_v


def _optimize_mirror(f0, deadline, rng, use_momentum=True):
    """
    Mirror descent with soft gradient, adaptive temperature, momentum,
    and SA acceptance for escaping local minima.
    """
    n = len(f0)
    f = np.maximum(f0, 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f /= s

    c = _conv_fft(f)
    best_peak = c.max()
    best_f = f.copy()

    eta = 1.2
    beta = 0.3 if use_momentum else 0.0
    mom_v = np.zeros(n) if use_momentum else None
    tau = best_peak * 0.015
    tau_base = best_peak * 0.015
    last_improve = time.monotonic()
    stuck_count = 0
    iter_count = 0
    sa_temperature = best_peak * 0.03

    while time.monotonic() < deadline:
        c = _conv_fft(f)
        cur_peak = c.max()

        if cur_peak < best_peak - 1e-14:
            best_peak = cur_peak
            best_f = f.copy()
            last_improve = time.monotonic()
            sa_temperature *= 0.95
        else:
            stuck_count += 1

        stuck_time = time.monotonic() - last_improve

        # Adaptive tau: increase when stuck to broaden gradient
        if stuck_time > 20:
            tau = tau_base * (1.0 + 0.5 * min(stuck_time / 60.0, 3.0))
            # Perturb to escape
            mag = min(0.03, 0.005 + 0.005 * (stuck_time / 60.0))
            f = f + mag / np.sqrt(n) * rng.standard_normal(n)
            f = np.maximum(f, 0.0)
            f /= f.sum()
            last_improve = time.monotonic()
            mom_v = np.zeros(n) if use_momentum else None
        else:
            tau = tau_base * (1.0 + 0.1 * stuck_count / 5000.0)

        # Compute gradient (soft for smoothness, blend with top-k for focus)
        grad_soft = _soft_grad_fft(f, c, tau)
        # Also compute a focused gradient for blending
        k_focus = 3
        grad_focus = _topk_grad_fft(f, c, k_focus)
        # Blend: more focused early, more soft later
        blend = min(0.7, 0.3 + 0.1 * (iter_count / 10000.0))
        grad = blend * grad_soft + (1.0 - blend) * grad_focus

        # Mirror descent step with momentum
        f_new, mom_v = _mirror_step(f, grad, eta, mom_v, beta)
        c_new = _conv_fft(f_new)
        new_peak = c_new.max()

        delta = new_peak - cur_peak
        if delta < -1e-14:
            f = f_new
            eta = min(eta * 1.005, 4.0)
        elif delta < 1e-14:
            # No change, try SA
            if sa_temperature > best_peak * 1e-6:
                if np.log(max(rng.uniform(), 1e-300)) < -abs(delta) / sa_temperature:
                    f = f_new
            else:
                eta *= 0.9
        else:
            # Worsened
            eta *= 0.75
            if sa_temperature > best_peak * 1e-6:
                if np.log(max(rng.uniform(), 1e-300)) < -delta / sa_temperature:
                    f = f_new
                    eta *= 0.9
            # Decay SA temperature
            sa_temperature *= 0.999

        iter_count += 1

        if iter_count % 5000 == 0 and time.monotonic() < deadline:
            print(f"    iter {iter_count}: peak={cur_peak:.10f}, best={best_peak:.10f}, "
                  f"eta={eta:.3f}, tau={tau:.2e}, stuck={stuck_time:.0f}s")

    # Final check
    c = _conv_fft(f)
    if c.max() < best_peak:
        best_peak = c.max()
        best_f = f.copy()

    return best_f, 2.0 * n * best_peak


def _polish_coordinate(f, deadline, rng):
    """
    Coordinate descent polish: for the current peak lag, move mass from
    high-sensitivity positions to low-sensitivity positions.
    Also tries small global perturbations.
    """
    n = len(f)
    f = np.maximum(f, 0.0).copy()
    f /= f.sum()

    c = _conv_fft(f)
    best_peak = c.max()
    best_f = f.copy()
    passes = 0

    while time.monotonic() < deadline:
        c = _conv_fft(f)
        cur_peak = c.max()

        if cur_peak < best_peak - 1e-14:
            best_peak = cur_peak
            best_f = f.copy()

        k_star = int(np.argmax(c))

        # Sensitivity of b[k_star] to each f[j]: 2*f[k_star - j]
        lo = max(0, k_star - n + 1)
        hi = min(n, k_star + 1)
        idx = k_star - np.arange(lo, hi)
        sens = np.zeros(n)
        sens[lo:hi] = 2.0 * f[idx]

        # Find high-sensitivity donor and low-sensitivity receiver
        donor = int(np.argmax(sens))
        # Lowest sensitivity positions
        low_positions = np.argsort(sens)[:min(20, n // 10)]

        improved = False
        for recv in low_positions:
            if recv == donor:
                continue
            for frac in [0.01, 0.03, 0.05, 0.08, 0.12, 0.2, 0.3]:
                if f[donor] * frac < 1e-15:
                    break
                f_try = f.copy()
                amt = f[donor] * frac
                f_try[donor] -= amt
                f_try[recv] += amt
                s_try = f_try.sum()
                f_try /= s_try
                c_try = _conv_fft(f_try)
                if c_try.max() < cur_peak - 1e-14:
                    f = f_try
                    improved = True
                    break
            if improved:
                break

        # Also try: reduce the peak-contributing coordinate and spread to multiple low positions
        if not improved:
            top_sens = np.argsort(sens)[-5:]
            for d in top_sens:
                # Spread to 5 lowest positions
                targets = low_positions[:5]
                for frac in [0.02, 0.05, 0.1]:
                    f_try = f.copy()
                    amt = f[d] * frac
                    if amt < 1e-15:
                        continue
                    f_try[d] -= amt
                    for t in targets:
                        if t != d:
                            f_try[t] += amt / len(targets)
                    s_try = f_try.sum()
                    f_try /= s_try
                    c_try = _conv_fft(f_try)
                    if c_try.max() < cur_peak - 1e-14:
                        f = f_try
                        improved = True
                        break
                if improved:
                    break

        # Random pair moves when stuck
        if not improved:
            for _ in range(20):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                frac = rng.choice([0.01, 0.02, 0.05, 0.1])
                amt = f[i] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[i] -= amt
                f_try[j] += amt
                if f_try[i] < -1e-15:
                    continue
                f_try = np.maximum(f_try, 0.0)
                f_try /= f_try.sum()
                c_try = _conv_fft(f_try)
                if c_try.max() < cur_peak - 1e-14:
                    f = f_try
                    break

        # Periodic small perturbation
        if not improved and passes % 100 == 0:
            f_try = f + 0.001 / np.sqrt(n) * rng.standard_normal(n)
            f_try = np.maximum(f_try, 0.0)
            f_try /= f_try.sum()
            c_try = _conv_fft(f_try)
            if c_try.max() < cur_peak - 1e-14:
                f = f_try

        passes += 1

    c = _conv_fft(f)
    if c.max() < best_peak:
        best_peak = c.max()
        best_f = f.copy()

    return best_f, 2.0 * n * best_peak


def _generate_starts(n, rng, base_seq=None):
    """Generate diverse starting sequences."""
    starts = []
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    norm = n - 1

    # Use provided base sequence (normalized)
    if base_seq is not None:
        b = np.maximum(base_seq, 0.0)
        s = b.sum()
        if s > 0.01:
            starts.append(b / s)

    # Standard window functions
    for window_type in ['hann', 'hamming', 'blackman', 'bartlett', 'flattop']:
        if window_type == 'hann':
            w = 0.5 * (1 + np.cos(2 * np.pi * j / norm))
        elif window_type == 'hamming':
            w = 0.54 - 0.46 * np.cos(2 * np.pi * j / norm)
        elif window_type == 'blackman':
            w = 0.42 - 0.5 * np.cos(2 * np.pi * j / norm) + 0.08 * np.cos(4 * np.pi * j / norm)
        elif window_type == 'bartlett':
            w = 1.0 - np.abs(j - center) / center
        else:  # flattop
            a = [0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368]
            w = np.zeros(n)
            for k, ak in enumerate(a):
                if k == 0:
                    w += ak
                else:
                    w -= ak * np.cos(2 * np.pi * k * j / norm)
        w = np.maximum(w, 0.0)
        s = w.sum()
        if s > 0.01:
            starts.append(w / s)

    # Inverted windows (high at edges)
    for alpha in [0.5, 1.0, 2.0, 4.0]:
        w = 1.0 + alpha * (2.0 * j / norm - 1.0) ** 2
        s = w.sum()
        starts.append(w / s)

    # Gaussian centered
    for sigma in [0.15, 0.2, 0.25, 0.3, 0.4, 0.5]:
        w = np.exp(-((j - center) ** 2) / (2 * (sigma * n) ** 2))
        starts.append(w / w.sum())

    # Double-peaked (edges)
    for frac in [0.1, 0.15, 0.2]:
        w = np.exp(-((j / n - frac) ** 2) / (0.02)) + np.exp(-((j / n - (1 - frac)) ** 2) / (0.02)) + 0.01
        starts.append(w / w.sum())

    # Power-law taper: (1 - (2j/n - 1)^p)^q
    for p in [0.5, 1.0, 2.0, 4.0]:
        for q in [0.5, 1.0, 2.0]:
            base = np.abs(2.0 * j / n - 1.0)
            w = (1.0 - base ** p) ** q + 0.001
            starts.append(w / w.sum())

    # Sine-based
    for freq in [1, 2, 3, 5, 8]:
        for amp in [0.3, 0.6, 0.9]:
            w = 1.0 + amp * np.sin(freq * 2 * np.pi * j / n)
            w = np.maximum(w, 0.0)
            if w.sum() > 0.01:
                starts.append(w / w.sum())

    # Kaiser window
    for alpha in [2.0, 4.0, 6.0, 8.0]:
        x = 1.0 - (2.0 * j / norm - 1.0) ** 2
        x = np.maximum(x, 0.0)
        w = bessel_i0(alpha * np.sqrt(x))
        starts.append(w / w.sum())

    # Random starts
    for _ in range(20):
        r = rng.random()
        if r < 0.2:
            w = rng.exponential(1.0, n)
        elif r < 0.4:
            alpha = rng.uniform(0.1, 5.0)
            w = 1.0 + alpha * (2.0 * j / norm - 1.0) ** rng.uniform(0.5, 4.0)
        elif r < 0.6:
            # Two-scale: low-freq + high-freq
            f1 = rng.uniform(1, 3)
            f2 = rng.uniform(10, 50)
            w = 1.0 + 0.5 * np.sin(2 * np.pi * f1 * j / n) + 0.2 * np.sin(2 * np.pi * f2 * j / n + rng.uniform(0, 2 * np.pi))
        elif r < 0.8:
            # Sparse bumps
            n_bumps = int(rng.integers(2, 8))
            centers = rng.uniform(0, 1, n_bumps)
            widths = rng.uniform(0.02, 0.15, n_bumps)
            w = np.ones(n) * 0.01
            for c, wd in zip(centers, widths):
                w += np.exp(-((j / n - c) ** 2) / (2 * wd ** 2))
        else:
            # Combined: window * noise
            base_w = 0.5 * (1 + np.cos(2 * np.pi * j / norm))
            w = base_w * (1.0 + 0.3 * rng.standard_normal(n))
        w = np.maximum(w, 0.0)
        s = w.sum()
        if s > 0.01:
            starts.append(w / s)

    # Deduplicate (approximately)
    unique_starts = []
    for w in starts:
        is_dup = False
        for u in unique_starts:
            if np.allclose(w, u, atol=1e-8):
                is_dup = True
                break
        if not is_dup:
            unique_starts.append(w)

    return unique_starts


def _multi_lag_gradient_optimize(f0, deadline, rng):
    """
    Alternative optimizer: uses top-m lags (m = number of lags above 95% of peak)
    for a harder subgradient, with adaptive step size.
    """
    n = len(f0)
    f = np.maximum(f0, 0.0).copy()
    f /= f.sum()

    c = _conv_fft(f)
    best_peak = c.max()
    best_f = f.copy()

    eta = 1.0
    last_improve = time.monotonic()
    iter_count = 0

    while time.monotonic() < deadline:
        c = _conv_fft(f)
        cur_peak = c.max()

        if cur_peak < best_peak - 1e-14:
            best_peak = cur_peak
            best_f = f.copy()
            last_improve = time.monotonic()

        stuck_time = time.monotonic() - last_improve
        if stuck_time > 30:
            # Reheat: perturb and reset
            f = f + 0.02 / np.sqrt(n) * rng.standard_normal(n)
            f = np.maximum(f, 0.0)
            f /= f.sum()
            eta = 1.5
            last_improve = time.monotonic()

        # Determine active lags (above 90% of peak)
        threshold = cur_peak * 0.90
        active_mask = c >= threshold
        n_active = active_mask.sum()
        if n_active == 0:
            n_active = 1
        k_eff = max(1, min(int(n_active), 20))

        # Use top-k gradient
        grad = _topk_grad_fft(f, c, k_eff)

        # Mirror step
        f_new, _ = _mirror_step(f, grad, eta)
        c_new = _conv_fft(f_new)
        new_peak = c_new.max()

        if new_peak < cur_peak - 1e-14:
            f = f_new
            eta = min(eta * 1.02, 5.0)
        else:
            eta *= 0.8

        iter_count += 1
        if iter_count % 5000 == 0 and time.monotonic() < deadline:
            print(f"    multi-lag iter {iter_count}: peak={cur_peak:.10f}, best={best_peak:.10f}, k={k_eff}")

    c = _conv_fft(f)
    if c.max() < best_peak:
        best_peak = c.max()
        best_f = f.copy()

    return best_f, 2.0 * n * best_peak


def propose_candidate(height_sequence_1=None, budget_s=2400, **kwargs):
    """
    Search for the non-negative sequence minimizing 2*n*max(conv)/(sum)^2.
    Uses FFT-based soft gradient mirror descent with momentum, diverse starts,
    multi-phase optimization, and coordinate polish.
    """
    rng = np.random.default_rng(12345)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 20.0

    # Get base sequence
    if height_sequence_1 is not None:
        base = np.asarray(height_sequence_1, dtype=np.float64)
    else:
        base = np.ones(6121)

    n = len(base)
    print(f"Sequence length: n = {n}")

    # Generate and score starts
    print("Generating starts...")
    starts = _generate_starts(n, rng, base)
    print(f"Generated {len(starts)} diverse starts")

    scored = []
    for f0 in starts:
        f = np.maximum(f0, 0.0)
        c = _conv_fft(f)
        peak = c.max()
        scored.append((peak, f))

    scored.sort(key=lambda x: x[0])
    print(f"Top 10 start peaks: {[f'{p:.8f}' for p, _ in scored[:10]]}")
    print(f"Initial best score: {2*n*scored[0][0]:.6f}")

    best_score = 2.0 * n * scored[0][0]
    best_f = scored[0][1].copy()

    # Phase 1: Optimize top starts with soft gradient mirror descent
    # Allocate: 55% for main optimization, 20% for alternative optimizer,
    #           15% for polish, 10% buffer
    opt_budget = (deadline - time.monotonic()) * 0.55
    alt_budget = (deadline - time.monotonic()) * 0.20
    polish_budget = (deadline - time.monotonic()) * 0.15

    num_starts = 4
    time_per_start = opt_budget / num_starts

    for idx in range(num_starts):
        remaining = deadline - time.monotonic()
        if remaining < 120:
            break
        alloc = min(time_per_start, remaining - alt_budget - polish_budget - 60)
        if alloc < 30:
            break

        peak0, f0 = scored[idx]
        print(f"\nPhase 1: Optimizing start {idx+1} (peak={peak0:.8f}, score={2*n*peak0:.6f}), budget={alloc:.0f}s")
        f_opt, score = _optimize_mirror(f0, time.monotonic() + alloc, rng, use_momentum=True)

        if score < best_score:
            best_score = score
            best_f = f_opt.copy()
            print(f"  Improved to {best_score:.6f}")
        else:
            print(f"  Best from this start: {score:.6f}")

    # Phase 2: Alternative optimizer (multi-lag) from best found
    remaining = deadline - time.monotonic()
    if remaining > 100:
        alt_alloc = min(alt_budget, remaining - polish_budget - 30)
        if alt_alloc > 60:
            print(f"\nPhase 2: Multi-lag optimization, budget={alt_alloc:.0f}s")
            f_alt, alt_score = _multi_lag_gradient_optimize(best_f, time.monotonic() + alt_alloc, rng)
            if alt_score < best_score:
                best_score = alt_score
                best_f = f_alt.copy()
                print(f"  Improved to {best_score:.6f}")
            else:
                print(f"  Alt best: {alt_score:.6f}")

    # Phase 3: Polish
    remaining = deadline - time.monotonic()
    if remaining > 60:
        pol_alloc = min(polish_budget, remaining - 30)
        print(f"\nPhase 3: Coordinate polish, budget={pol_alloc:.0f}s")
        f_pol, pol_score = _polish_coordinate(best_f, time.monotonic() + pol_alloc, rng)
        if pol_score < best_score:
            best_score = pol_score
            best_f = f_pol.copy()
            print(f"  Polished to {best_score:.6f}")
        else:
            print(f"  Polish best: {pol_score:.6f}")

    # Phase 4: Final optimization from polished solution
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\nPhase 4: Final optimization, budget={remaining - 10:.0f}s")
        f_fin, fin_score = _optimize_mirror(best_f, deadline, rng, use_momentum=True)
        if fin_score < best_score:
            best_score = fin_score
            best_f = f_fin.copy()
            print(f"  Final: {best_score:.6f}")

    # Phase 5: Final quick polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        f_pol2, pol2_score = _polish_coordinate(best_f, time.monotonic() + min(remaining - 10, 120), rng)
        if pol2_score < best_score:
            best_score = pol2_score
            best_f = f_pol2.copy()
            print(f"  Final polish: {best_score:.6f}")

    # Ensure validity
    best_f = np.clip(best_f, 0.0, 1000.0)
    seq = [float(x) for x in best_f]

    # Verify with official evaluator
    final_score = evaluate_sequence(seq)
    print(f"\nOfficial evaluation: {final_score:.6f}")

    if not np.isfinite(final_score):
        # Fallback: uniform
        fallback = [1.0 / n] * n
        final_score = evaluate_sequence(fallback)
        print(f"Fallback uniform: {final_score:.6f}")
        return fallback

    return seq
