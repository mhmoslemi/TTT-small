Optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses multiplicative gradient descent with adaptive smoothed max, 
frequency-domain flattening, block refinement, and uniform blending.

import numpy as np

def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(a, b, L=None):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 4000000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)

def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())

def _sanitize(p):

def _soft_max_weights(d, beta):
    n = d.size

        return np.array([], dtype=float)
    d = np.where(np.isfinite(d), d, 0.0)
    if beta < 1e-8:

    e = np.exp(beta * (d - shift))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones(n, dtype=float) / n
    return e / s

def _gradient_step(p, beta, L):
    """Compute gradient info for multiplicative update."""
    n = len(p)
    c = _conv_full(p, p)
    d = n * c
    exact_score = float(2.0 * d.max())
    w = _soft_max_weights(d, beta)

    # dc[k]/da[j] = a[j-k] + a[j+k] (with bounds)
    # = 2 * sum_k w[k] * (p cross-correlated with delta at k)
    # More precisely: h[j] = 2 * sum_k w[k] * (p[j+k] if j+k<n else 0 + p[j-k] if j>=k else 0)
    # For k=0: contribution is 2*p[j]
    # For k>0: contribution is p[j+k] + p[j-k] (with bounds)
    # Using FFT: this is 2 * Ifft(Rfft(w) * conj(Rfft(p)))[:n]
    W = np.fft.rfft(w, L)
    P = np.fft.rfft(p, L)
    R = np.fft.irfft(W * np.conj(P), L)
    h = 2.0 * R[:n]
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.maximum(h, 1e-30)
    return exact_score, h

def _freq_flatten(p, alpha, L):
    """Flatten the convolution in frequency domain."""
    n = len(p)
    L = max(L, _next_pow2(2*n-1))
    A = np.fft.rfft(p, L)
    C_freq = A * A
    C = np.fft.irfft(C_freq, L)[:2*n-1]

    if cmax < 1e-15:

    C_norm = C / cmax

    # Take square root in frequency domain

def _resample(p_in, n_new, rng=None):
    p = _sanitize(p_in)
    if p is None:
        return None
    n_old = len(p)
    if n_old == n_new:
        return p.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    result = np.interp(x_new, x_old, p)
    if rng is not None:
        result = result * np.exp(rng.normal(0, 0.02, n_new))
        result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s

"""
Optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.
Uses alternating projections between flat-spectrum and non-negative constraints,
combined with multiplicative gradient descent and block refinement.
"""
import numpy as np
import time

def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(a, b, L=None):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 4000000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)

def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())

def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s

def _flat_spectrum_update(p, eta, L):
    """
    Project the power spectrum towards flat, then transform back.
    eta controls the strength of the projection (0 < eta <= 1).
    """
    n = len(p)
    L = max(L, _next_pow2(2*n-1))
    A = np.fft.rfft(p, L)
    P = np.abs(A)**2
    P_mean = P.sum() / len(P)
    if P_mean < 1e-30:
        return p.copy()
    # Project towards flat
    P_new = (1-eta) * P + eta * P_mean
    # Alternatively: P_new = P - eta*(P - P_mean) which is the same
    P_new = np.maximum(P_new, 0)
    # Take square root preserving phase
    A_new = np.sqrt(P_new) * np.exp(1j * np.angle(A))
    a_new = np.fft.irfft(A_new, L)[:n]
    return a_new

    best_score = _score(p)
    cur_score = best_score

            cur_score = sc_new

    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    L = _next_pow2(3*n + 10)
    
    best = p.copy()
    best_score = _score(p)
    cur_score = best_score

    eta = 0.5

    stall = 0
    it = 0

        c = _conv_full(p, p)
        d = n * c
        exact_score = float(2.0 * d.max())

        beta_local = beta
        shift = d.max()
        w = np.exp(beta_local * (d - shift))
