    c = np.fft.irfft(Fp * Fp, L)[:2*n-1]

def _subgradient(p, k_lag):

    k_lag is the index in the convolution array (0 to 2n-2).
    The actual lag is k_lag - (n-1).

            grad = 2.0 * p
        else:

            grad[lag:] += p[:n-lag]  # backward shift
    else:
        lag = -lag
        grad = np.zeros(n)
        grad[:n-lag] = p[lag:]
        grad[lag:] += p[:n-lag]
    return grad

    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

import numpy as np
import time

_MIN_P = 1e-15

def _conv_len(n):

    L = 1
    while L < need:

def _score_fft(p, L):
    n = p.size
    Fp = np.fft.rfft(p, L)
    c = np.fft.irfft(Fp * Fp, L)[:2*n-1]
    c = np.maximum(c, 0.0)

def _softmax_weights(d, beta):
    n = d.size
    if n == 0:
        return np.array([])
    if beta < 1e-8:
        return np.full(n, 1.0/n)
    shift = d.max()
    e = np.exp(beta * (d - shift))
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        w = np.zeros(n)
        w[int(np.argmax(d))] = 1.0
        return w
    return e / s

def _conv_len(n):

def _score_fft(p, L):
    n = p.size
    Fp = np.fft.rfft(p, L)
    c = np.fft.irfft(Fp * Fp, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)
    return float(2.0 * n * c.max()), c


def _softmax_weights(d, beta):
    n = d.size

        w = np.zeros(n, dtype=float)
        w[int(np.argmax(d))] = 1.0
        return w
    return e / s

    Fp = np.fft.rfft(p, L)
    h = np.fft.irfft(Fw * np.conjugate(Fp), L)[:n]
    return np.maximum(h, 0.0)

def _optimize_block(p0, time_limit, verbose=True):
    """Block coordinate descent with targeted blocks near peak influence."""
    p = np.asarray(p0, dtype=float).copy()
    n = p.size
    L = _conv_len(n)

    best = p.copy()
    best_sc, _ = _score_fft(p, L)

    deadline = time.monotonic() + time_limit
    stagnant = 0
    it = 0
    last_print = time.monotonic()

    # Get the current peak location to focus search
    _, c = _score_fft(p, L)
    peak_lag = int(np.argmax(c))

    # Block sizes
    sizes = [
        1, 2, 3, 4, 5, 8, 12, 16, 24, 32, 48, 64,
        n // 64, n // 32, n // 16, n // 8, n // 4
    ]
    sizes = [s for s in sizes if 1 <= s <= n]

    factors_small = [0.9, 0.95, 1.05, 1.1]
    factors_med = [0.8, 0.9, 1.1, 1.2]
    factors_large = [0.7, 0.85, 1.0, 1.15, 1.3]

    while time.monotonic() < deadline:
        r = _GLOBAL_RNG.random()

        # Choose block size
        if r < 0.25:
            w = int(_GLOBAL_RNG.integers(1, 10))
            factors = factors_small
        elif r < 0.55:
            w = int(_GLOBAL_RNG.integers(8, max(9, n // 64)))
            factors = factors_med
        elif r < 0.80:
            w = int(_GLOBAL_RNG.integers(max(9, n // 64), max(10, n // 8)))
            factors = factors_med
        else:
            w = int(_GLOBAL_RNG.integers(max(10, n // 8), n + 1))
            factors = factors_large

        w = max(1, min(n, w))

        # Choose start position: bias toward the "influence zone" of the peak
        # Elements near position (n-1-peak_lag) and (peak_lag) influence the peak most
        if r < 0.5:
            # Uniform
            start = int(_GLOBAL_RNG.integers(0, max(1, n - w + 1)))
        else:
            # Biased toward edges (where structure changes happen)
            if _GLOBAL_RNG.random() < 0.5:
                start = int(_GLOBAL_RNG.integers(0, max(1, n // 3)))
            else:
                start = int(_GLOBAL_RNG.integers(max(0, 2 * n // 3 - w), n - w + 1))
        start = max(0, min(n - w, start))

        best_trial = None
        best_trial_sc = best_sc

        for f in factors:
            v = p.copy()
            v[start:start + w] *= f
            v = np.maximum(v, _MIN_P)
            s = float(v.sum())
            if s <= 0:
                continue
            v /= s
            sc, _ = _score_fft(v, L)
            if sc < best_trial_sc:
                best_trial_sc = sc
                best_trial = v

        if best_trial is not None:
            p = best_trial
            if best_trial_sc < best_sc - 1e-12:
                best_sc = best_trial_sc
                best = p.copy()
                stagnant = 0
                if verbose and (time.monotonic() - last_print > 5.0):
                    last_print = time.monotonic()
                    print(f"    block it={it} best={best_sc:.7f}")
            else:
                stagnant += 1
        else:
            stagnant += 1

        if stagnant > 40:
            sigma = 0.04
            v = p * np.exp(_GLOBAL_RNG.normal(0, sigma, n))
            v = np.maximum(v, _MIN_P)
            s = float(v.sum())
            if s > 0:
                v /= s
                sc2, _ = _score_fft(v, L)
                if sc2 <= best_sc * 1.01:
                    p = v
                else:
                    p = best.copy()
            stagnant = 0

        if stagnant > 200:
            p = best.copy()
            stagnant = 0

        it += 1

    return best, best_sc


def _optimize_fine(p0, time_limit, verbose=False):
    """Fine coordinate-level search for final polishing."""
    p = np.asarray(p0, dtype=float).copy()
    n = p.size
    L = _conv_len(n)

    best = p.copy()
    best_sc, _ = _score_fft(p, L)

    deadline = time.monotonic() + time_limit
    stagnant = 0
    last_print = time.monotonic()

    factors = [0.95, 0.98, 1.02, 1.05]

    while time.monotonic() < deadline:
        # Pick 1-3 random coordinates
        n_changes = int(_GLOBAL_RNG.integers(1, 4))
        indices = _GLOBAL_RNG.integers(0, n, size=n_changes)

        improved = False
        for idx in indices:
            best_local = p
            best_local_sc = p_sc = float(_score_fft(p, L)[0])

            for f in factors:
                v = p.copy()
                v[idx] *= f
                v = np.maximum(v, _MIN_P)
                s = float(v.sum())
                if s <= 0:
                    continue
                v /= s
                sc2, _ = _score_fft(v, L)
                if sc2 < best_local_sc:
                    best_local_sc = sc2
                    best_local = v

            if best_local_sc < p_sc - 1e-12:
                p = best_local
                p_sc = best_local_sc
                improved = True

                if p_sc < best_sc - 1e-12:
                    best_sc = p_sc
                    best = p.copy()
                    stagnant = 0
                    if verbose and (time.monotonic() - last_print > 3.0):
                        last_print = time.monotonic()
                        print(f"    fine best={best_sc:.7f}")
        if not improved:
            stagnant += 1

        if stagnant > 80:
            sigma = 0.02
            v = p * np.exp(_GLOBAL_RNG.normal(0, sigma, n))
            v = np.maximum(v, _MIN_P)
            s = float(v.sum())
            if s > 0:
                v /= s
                sc2, _ = _score_fft(v, L)
                if sc2 <= best_sc * 1.005:
                    p = v
                else:
                    p = best.copy()
                    p_sc = best_sc
            stagnant = 0

        if stagnant > 400:
            p = best.copy()
            p_sc = best_sc
            stagnant = 0

    return best, best_sc


def _two_block_init(n):
    """Create a two-block initial sequence."""
    best = None
    best_sc = float('inf')
    for frac in [0.3, 0.35, 0.4, 0.45, 0.5, 0.55, 0.6]:
        for ratio in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
            k = int(round(frac * n))
            if k <= 0 or k >= n:
                continue
            p = np.ones(n)
            p[:k] = ratio
            p = np.maximum(p, 0)
            s = p.sum()
            if s <= 0:
                continue
            p /= s
            sc, _ = _score_fft(p, _conv_len(n))
            if sc < best_sc:
                best_sc = sc
                best = p.copy()
    return best


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    global _GLOBAL_RNG

    start_time = time.monotonic()
    _GLOBAL_RNG = np.random.default_rng(seed % 1000000007 + 1)

    # Get parent sequence
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=float)
        parent = np.where(np.isfinite(parent), parent, 0.0)
        parent = np.maximum(parent, 0.0)
        s = float(parent.sum())
        if s > 0.01:
            parent = parent / s
        else:
            parent = None
    except (NameError, Exception):
        parent = None

    if parent is None:
        n = 3072
        parent = _two_block_init(n)
        if parent is None:
            parent = np.ones(3072) / 3072

    n = parent.size
    L = _conv_len(n)
    initial_sc, _ = _score_fft(parent, L)
    print(f"Starting: n={n}, score={initial_sc:.7f}")

    total_time = min(float(budget_s) if budget_s else 2400.0, 2360.0)
    deadline = start_time + total_time

    best_seq = parent.copy()
    best_score = initial_sc

    # Phase 1: Main multiplicative optimization (45% of time)
    t1 = total_time * 0.45
    rem = deadline - time.monotonic()
    t1 = min(t1, max(0, rem - 60))
    if t1 > 20:
        print(f"\nPhase 1: Main optimization ({t1:.0f}s)")
        p, sc = _optimize_main(parent, t1, target=1.500, verbose=True)
        if sc < best_score:
            best_score = sc
            best_seq = p.copy()
        print(f"Phase 1 result: {sc:.7f} (best so far: {best_score:.7f})")

    # Phase 2: Block search (20% of time)
    rem = deadline - time.monotonic()
    t2 = min(total_time * 0.20, max(0, rem - 60))
    if t2 > 10:
        print(f"\nPhase 2: Block search ({t2:.0f}s)")
        p, sc = _optimize_block(best_seq, t2, verbose=True)
        if sc < best_score:
            best_score = sc
            best_seq = p.copy()
        print(f"Phase 2 result: {sc:.7f} (best so far: {best_score:.7f})")

    # Phase 3: Second multiplicative pass from perturbed start (15% of time)
    rem = deadline - time.monotonic()
    t3 = min(total_time * 0.15, max(0, rem - 40))
    if t3 > 15:
        # Create perturbed starting point
        v = best_seq * np.exp(_GLOBAL_RNG.normal(0, 0.05, n))
        v = np.maximum(v, _MIN_P)
        v /= v.sum()
        print(f"\nPhase 3: Refinement pass ({t3:.0f}s)")
        p, sc = _optimize_main(v, t3, target=1.495, verbose=True)
        if sc < best_score:
            best_score = sc
            best_seq = p.copy()
        print(f"Phase 3 result: {sc:.7f} (best so far: {best_score:.7f})")

    # Phase 4: Block search again (10% of time)
    rem = deadline - time.monotonic()
    t4 = min(total_time * 0.10, max(0, rem - 30))
    if t4 > 10:
        print(f"\nPhase 4: Block refinement ({t4:.0f}s)")
        p, sc = _optimize_block(best_seq, t4, verbose=True)
        if sc < best_score:
            best_score = sc
            best_seq = p.copy()
        print(f"Phase 4 result: {sc:.7f} (best so far: {best_score:.7f})")

    # Phase 5: Fine coordinate search (5% of time)
    rem = deadline - time.monotonic()
    t5 = min(total_time * 0.05, max(0, rem - 15))
    if t5 > 5:
        print(f"\nPhase 5: Fine-tuning ({t5:.0f}s)")
        p, sc = _optimize_fine(best_seq, t5, verbose=True)
        if sc < best_score:
            best_score = sc
            best_seq = p.copy()
        print(f"Phase 5 result: {sc:.7f} (best so far: {best_score:.7f})")

    # Final verification
    rem = deadline - time.monotonic()
    if rem > 3:
        final_list = [float(x) for x in best_seq]
        try:
            verified = float(evaluate_sequence(final_list))
            if np.isfinite(verified):
                print(f"\nVerified: {verified:.7f}")
                if verified < best_score:
                    best_score = verified
        except Exception as e:
            print(f"Verification issue: {e}")

    print(f"\nFinal answer: score={best_score:.7f}, n={best_seq.size}")
    return [float(x) for x in best_seq]
