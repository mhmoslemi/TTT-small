    dC[k]/da[i] = p[k-i] if 0<=k-i<n and k-i != i, else 2*p[i] if k==2i
    """
    n = len(p)
    # grad[i] = sum_{k=i}^{min(i+n-1, 2n-2)} w[k] * p[k-i] + w[2i]*p[i] (if 2i < 2n-1)
    #         = sum_{j=0}^{min(n-1, 2n-2-i)} w[i+j] * p[j] + w[2i]*p[i] (if 2i < 2n-1)
    # 
    # The main term is a "sliding dot product": for each i, dot(w[i:i+n], p)
    # This can be computed via FFT cross-correlation.
    
    L = _next_pow2(len(w) + n - 1)
    W = np.fft.rfft(w, L)
    P = np.fft.rfft(p, L)
    # Cross-correlation: (w ⋆ p)[i] = sum_j w[i+j] * p[j]
    # = IFFT(conj(W) * P)[i] ... let me just compute and verify
    R = np.fft.irfft(np.conj(W) * P, L)
    # R[i] should be sum_j w[i+j] * p[j] for i = 0, ..., n-1 (if no wrap-around)
    
    # Add the correction term for k = 2i
    correction = np.zeros(n)
    valid = 2 * np.arange(n) < len(w)
    correction[valid] = w[2 * np.arange(n)[valid]] * p[valid]
    
    grad = R[:n] + correction
    return grad

"""
Spectral + gradient optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Key innovations over previous approach:
1. Gerchberg-Saxton spectral method with adaptive multi-target spectra
2. Multi-scale optimization (coarse-to-fine)
3. Improved gradient computation with exact subgradient + softmax hybrid
4. Better time allocation and diversity in restarts
"""

import numpy as np
import time


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b, L=None):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 3000000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)


def _score(arr):
    p = np.maximum(arr, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):
        return None
    return p / s


def _gerchberg_saxton(p_init, target_pow, n_iter=200, verbose=False):
    """
    Gerchberg-Saxton / error reduction method.
    target_pow: desired |A(w)|^2 spectrum (1D array, length = L//2+1 for rfft)
    Returns non-negative normalized sequence or None.
    """
    n = len(p_init)
    L = _next_pow2(2 * n - 1)
    half = L // 2 + 1
    
    # Ensure target_pow has correct length
    if len(target_pow) < half:
        target_pow = np.pad(target_pow, (0, half - len(target_pow)))
    else:
        target_pow = target_pow[:half]
    
    target_mag = np.sqrt(np.maximum(target_pow, 0.0))
    a = p_init.copy()
    
    for it in range(n_iter):
        A = np.fft.rfft(a, L)
        mag = np.abs(A)
        # Avoid division by zero
        safe_mag = np.where(mag > 1e-15, mag, 1.0)
        phase = A / safe_mag
        phase = np.where(mag > 1e-15, phase, 1.0 + 0j)
        
        A_new = target_mag * phase
        a_new = np.fft.irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            return None
        a_new = a_new / s
        
        if it > 10 and np.max(np.abs(a_new - a)) < 1e-12:
            break
        a = a_new
    
    return a


def _gs_flat(p_init, n_iter=200, dc_weight=1.0):
    """GS with flat target (DC=1, rest uniform)."""
    n = len(p_init)
    L = _next_pow2(2 * n - 1)
    half = L // 2 + 1
    
    # Total power constraint: sum of target_pow = N * sum(a^2) for the DFT
    # We want the non-DC part to be as flat as possible
    # DC must be 1 (since sum(a)=1, A(0)=1, |A(0)|^2=1)
    # For flat: |A(k)|^2 = (L - 1) / (L - 1) = 1 for k != 0? 
    # Actually let's just set it to a constant and let GS figure out the rest.
    target = np.full(half, 1.0)
    target[0] = 1.0
    
    return _gerchberg_saxton(p_init, target, n_iter)


def _gs_ramped(p_init, ramp_alpha, n_iter=200):
    """GS with ramped target: higher near DC, lower away."""
    n = len(p_init)
    L = _next_pow2(2 * n - 1)
    half = L // 2 + 1
    k = np.arange(half, dtype=float)
    target = 1.0 + ramp_alpha * np.exp(-k / (half * 0.3))
    target[0] = 1.0
    
    return _gerchberg_saxton(p_init, target, n_iter)


def _compute_grad_smoothing(p, beta, L):
    """
    Compute gradient of smoothed objective:
    f(a) = sum_k w[k] * C[k] / (sum a)^2
    where w[k] = softmax(beta * C[k])
    Returns (score, grad) where grad is for the normalized problem.
    """
    n = len(p)
    C = _conv_full(p, p)
    cmax = C.max()
    score = 2.0 * n * cmax  # with sum(a)=1 normalization
    
    # Softmax weights
    if beta < 1e-6:
        # Exact subgradient: uniform over argmax set
        k_star = np.where(C >= cmax - 1e-10)[0]
        w = np.zeros_like(C)
        w[k_star] = 1.0 / len(k_star)
    else:
        shift = C.max()
        e = np.exp(beta * (C - shift))
        s = e.sum()
        if s < 1e-300:
            w = np.zeros_like(C)
            w[np.argmax(C)] = 1.0
        else:
            w = e / s
    
    # Gradient of sum_k w[k] * C[k] w.r.t. a[i]:
    # = sum_k w[k] * dC[k]/da[i]
    # dC[k]/da[i] = a[k-i] (if valid and k-i != i) + 2*a[i] (if k == 2i)
    # Main term: sum_k w[k] * a[k-i] for valid k-i
    # = sum_j a[j] * w[i+j] for j=0..n-1 (cross-correlation)
    
    # Compute via FFT: cross-correlation of w and a
    L2 = _next_pow2(len(w) + n - 1)
    W = np.fft.rfft(w, L2)
    A = np.fft.rfft(p, L2)
    # IFFT(conj(W) * A)[i] = sum_j a[j] * w[j+i] (circular, but L2 large enough)
    R = np.fft.irfft(np.conj(W) * A, L2)
    main_grad = R[:n]
    
    # Correction for the 2*a[i] term when k=2i
    correction = np.zeros(n)
    idx = 2 * np.arange(n)
    valid = idx < len(w)
    correction[valid] = w[idx[valid]] * p[valid]
    
    grad = main_grad + correction
    grad = np.where(np.isfinite(grad), grad, 0.0)
    
    return score, grad
