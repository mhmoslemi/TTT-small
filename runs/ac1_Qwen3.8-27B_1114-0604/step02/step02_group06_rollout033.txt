Let me now think about the implementation details:

For the multiplicative updates, the key step is:
- Compute c = conv(a, a) (using FFT).
- Compute the "gradient" g[i] = sum_{k: c[k] > threshold} c[k] * a[k-i] (or just the peak contribution).
- Update: a[i] *= exp(-eta * g[i]).
- Normalize: a = a / sum(a).

The threshold determines how "aggressive" the update is. A high threshold (close to max) means we only push down the peaks. A low threshold means we try to flatten everything.

For the LP approach:
- Given a and M = max(a*a):
  - LP: maximize sum(b) subject to (a*b)[k] <= M for all k, b >= 0.
  - The constraint (a*b)[k] = sum_i a[i] * b[k-i] <= M is linear in b.
  - Solve with scipy.optimize.linprog (method='highs' for speed).
  - Get b*, rescale to sum 1, and take a step toward it.

For L-BFGS:
- x is unconstrained (length n).
- a = exp(x - max(x)) / sum(exp(x - max(x))) (stable softmax).
- Objective: 2n * softmax_beta(conv(a,a)) where softmax_beta(c) = (1/beta) * log(sum(exp(beta * c))).
- Gradient computed via backprop through the softmax and convolution.
- Use scipy.optimize.minimize with method='L-BFGS-B'.

    return _next_pow2(2 * n - 1)

    a = np.asarray(p, dtype=float).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if normalize:
        s = float(a.sum())
        if s <= 0 or not np.isfinite(s):

        a = a / s
    return a

def _score(p, L=None):
    """Compute 2n*max(conv(p,p))/sum(p)^2 (scale-invariant)."""
    p = np.asarray(p, dtype=float)
    n = p.size
    if L is None:
        L = _conv_len(n)
    Fp = np.fft.rfft(p, L)
    c = np.fft.irfft(Fp * Fp, L)[:2*n-1]
    c = np.maximum(c, 0.0)
    s = p.sum()
    if s <= 0:
        return float('inf'), c
    return float(2.0 * n * c.max() / (s*s)), c

def _multiplicative_update(a, c, threshold_frac=0.95, eta=0.1, L=None):
    """One step of multiplicative update: reduce peaks in convolution."""
    n = a.size
    if L is None:
        L = _conv_len(n)

    threshold = threshold_frac * maxc
    excess = np.maximum(0.0, c - threshold)

        return a
    
    # Gradient: g[i] = sum_k excess[k] * a[k-i] (correlation of excess with a)
    Fe = np.fft.rfft(excess, L)
    g = np.fft.irfft(Fe * np.conjugate(np.fft.rfft(a, L)), L)[:n]

def _lp_direction(a, M=None, solver='highs'):
    """Solve LP to find optimal direction. Returns rescaled b*."""
    n = a.size
    if M is None:
        _, c = _score(a)
        M = c.max()
    
    # LP: max sum(b) s.t. (a*b)[k] <= M for k=0,...,2n-2, b >= 0
    # (a*b)[k] = sum_i a[i] * b[k-i] for valid i
    # This is a matrix-vector product: A @ b <= M * ones
    # where A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # But A is n*(2n-1) which is too large to form explicitly for n > a few hundred.
    # Instead, we use the fact that A is a banded Toeplitz-like matrix.
    
    # For large n, we'll skip exact LP and use an approximation.
    if n > 5000:
        return None
    
    # Build the constraint matrix A (size (2n-1) x n)
    # A[k, j] = a[j] if k >= j and k-j < n, i.e., j <= k < j+n
    # Actually: (a*b)[k] = sum_j a[j] * b[k-j] where 0<=j<n and 0<=k-j<n
    # = sum_j a[j] * b[k-j] for j in [max(0, k-n+1), min(n-1, k)]
    # So A[k, j] = a[j] if max(0, k-n+1) <= j <= min(n-1, k), else 0
    
    # This is a banded matrix. For scipy linprog, we can use sparse matrix.
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    c_obj = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_ub = np.full(2*n-1, M)
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method=solver)
    
    if res.success and res.x is not None:
        b = np.maximum(res.x, 0)
        s = b.sum()
        if s > 0:
            return b / s
    return None

    while time.monotonic() < deadline:
        sc, c = _score(a, L)
        cur_sc = sc

        if cur_sc < best_sc:
            best_sc = cur_sc
            best = a.copy()
            stall = 0
            if verbose and it % 500 == 0:
                print(f"      mul it={it} sc={best_sc:.7f}")
        else:
            stall = it - last_improve

        if stall > 400:
            a = best.copy()
            eta = max(0.03, eta * 0.85)
            tf = min(0.995, tf + 0.003)
            noise_count += 1
            if noise_count % 3 == 0:
                sg = np.random.default_rng(it + 1000)
                a = a * np.exp(sg.normal(0, 0.08, n))
                a = np.maximum(a, 1e-30)
                a = a / a.sum()
            stall = 0
            last_improve = it
            if noise_count > 20:
                break

        target_m = target / (2.0 * n)
        maxc = c.max()
        if maxc > target_m:
            tf_c = tf
        else:
            tf_c = max(0.5, tf * 0.7)

        best_a_t = None
        best_sc_t = cur_sc
        for mult in [0.4, 0.6, 0.8, 1.0, 1.4, 2.0]:
            e = eta * mult
            a_t = _mul_step(a, c, L, tf_c, e)
            if a_t is None or np.allclose(a_t, a):
                continue
            sc_t, _ = _score(a_t, L)
            if sc_t < best_sc_t:
                best_sc_t = sc_t
                best_a_t = a_t

        if best_a_t is not None:
            a = best_a_t
            cur_sc = best_sc_t
            last_improve = it
        else:
            pass

        it += 1

    return best, best_sc

def _lbfgs_obj_grad(x, n, L, Lc, beta):
    a = np.exp(x - x.max())
    s = a.sum()
    a = a / s
    Fa = np.fft.rfft(a, L)
    c = np.fft.irfft(Fa * Fa, L)[:2 * n - 1]
    c = np.maximum(c, 0.0)

    shift = c.max()
    w = np.exp(beta * (c - shift))
    ws = w.sum()
    if ws < 1e-30:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(c))] = 1.0
    else:
        w = w / ws

    C = float(np.dot(w, c))
    obj = 2.0 * n * (shift + np.log(ws) / beta)

    W = np.fft.rfft(w, Lc)
    A = np.fft.rfft(a, Lc)
    h = np.fft.irfft(W * np.conjugate(A), Lc)[:n]

    grad = 4.0 * n * a * (h - C)
    return obj, grad

def _lbfgs_optimize(a0, time_limit, beta_start=80.0, beta_end=800.0, verbose=False):
    a = _sanitize(a0)
    if a is None:
        return None, float('inf')
    n = a.size
    L = _clen(n)
    Lc = _clen_corr(n)
    deadline = time.monotonic() + time_limit

    x0 = np.log(np.maximum(a, 1e-30)).copy()

    sc, _ = _score(a, L)
    best = a.copy()
    best_sc = sc

    beta = beta_start
    phase = 0

    while time.monotonic() < deadline:
        ph_t = min(30.0, float(deadline - time.monotonic()))

        def obj_fun(x):
            o, _ = _lbfgs_obj_grad(x, n, L, Lc, beta)
            return o

        def grad_fun(x):
            _, g = _lbfgs_obj_grad(x, n, L, Lc, beta)
            return g

        res = minimize(obj_fun, x0, jac=grad_fun, method='L-BFGS-B',
                       options={'maxiter': 200, 'gtol': 1e-8, 'maxfun': 500})

        if res.success or res.fun < obj_fun(x0):
            xn = res.x
            a_new = np.exp(xn - xn.max())
            a_new = a_new / a_new.sum()
            sc_new, _ = _score(a_new, L)
            if sc_new < best_sc:
                best_sc = sc_new
                best = a_new.copy()
                x0 = xn.copy()
                if verbose:
                    print(f"      lbfgs beta={beta:.0f} sc={best_sc:.7f}")
            else:
                x0 = xn.copy()
        else:
            x0 = res.x if hasattr(res, 'x') and res.x is not None else x0

        # Increase beta
        beta = min(beta_end, beta * 1.5)
        phase += 1

        if beta >= beta_end and phase > 3:
            break

    return best, best_sc

def _resample(p, m):
    a = _sanitize(p)
    if a is None:
        return None
    n = a.size
    if n == m:
        return a.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(m) + 0.5) / m
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0) + 1e-12
    return _sanitize(out)

def _init_sequence(n, rng, mode=0):
    t = (np.arange(n, dtype=float) + 0.5) / n
    if mode == 0:
        p = np.ones(n)
    elif mode == 1:
        a = rng.uniform(0.3, 2.5)
        b = rng.uniform(0.3, 2.5)
        tt = np.clip(t, 1e-8, 1 - 1e-8)
        p = tt ** a * (1 - tt) ** b
    elif mode == 2:
        K = int(rng.integers(2, 6))
        p = np.zeros(n)
        centers = np.sort(rng.uniform(0.1, 0.9, K))
        sigmas = rng.uniform(0.05, 0.3, K)
        weights = rng.uniform(0.5, 2.0, K)
        for i in range(K):
            p += weights[i] * np.exp(-((t - centers[i]) / sigmas[i]) ** 2)
    elif mode == 3:
        p = 1.0 + 0.3 * np.sin(2 * np.pi * rng.uniform(1, 5) * t + rng.uniform(0, 6.28))
        p = p + 0.2 * np.sin(2 * np.pi * rng.uniform(3, 10) * t + rng.uniform(0, 6.28))
        p = np.abs(p) + 0.1
    elif mode == 4:
        K = int(rng.integers(3, 10))
        p = np.zeros(n)
        edges = np.sort(rng.uniform(0, 1, K - 1))
        blocks = np.concatenate([[0], edges, [1]])
        heights = rng.lognormal(0, 0.8, K)
        for i in range(K):
            s = int(blocks[i] * n)
            e = int(blocks[i + 1] * n)
            p[s:e] = heights[i]
    else:
        p = np.abs(rng.normal(0, 1, n)) + 0.1
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0, 1000) + 1e-10
    return _sanitize(p)

def _process_level(parent, cands, n, t_alloc, verbose=False):
    starts = []
    if parent is not None:
        sp = _resample(parent, n)
        if sp is not None:
            starts.append(sp)
    if cands:
        for item in sorted(cands, key=lambda x: x[0])[:2]:
            sp = _resample(item[2], n)
            if sp is not None:
                starts.append(sp)
    for mode in [1, 2, 3, 4]:
        if len(starts) >= 3:
            break
        sp = _init_sequence(n, _RNG, mode)
        if sp is not None:
            starts.append(sp)

    # Score and sort starts
    scored = []
    for sp in starts:
        sc, _ = _score(sp)
        if np.isfinite(sc):
            scored.append((sc, sp))
    scored.sort(key=lambda x: x[0])

    if not scored:
        return

    # Distribute time
    n_starts = min(len(scored), 3)
    t_each = t_alloc / n_starts

    for i in range(n_starts):
        rem = time_limit_remaining()
        if rem < 30:
            break
        t_each = min(t_each, rem - 20)
        if t_each < 10:
            continue

        sp = scored[i][1]
        # Coarse: multiplicative
        t_mul = t_each * 0.5
        p1, sc1 = _mul_optimize(sp, t_mul, target=1.50, verbose=verbose)

        # Fine: L-BFGS
        t_lbf = t_each * 0.5
        if p1 is not None:
            p2, sc2 = _lbfgs_optimize(p1, t_lbf, beta_start=60, beta_end=400, verbose=verbose)
            if p2 is not None:
                cands.append((sc2, n, p2.copy()))
            cands.append((sc1, n, p1.copy()))
        else:
            cands.append((sc1, n, sp.copy()))

    # Keep top candidates
    cands.sort(key=lambda x: x[0])
    if len(cands) > 12:
        del cands[12:]

def time_limit_remaining():
    return _DEADLINE[0] - time.monotonic()

_DEADLINE = [0.0]

def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    global _RNG, _DEADLINE

    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except (Exception):
        budget = 2400.0
    if not np.isfinite(budget) or budget < 60:
        budget = 60.0
    total = max(60.0, budget - 30.0)
    _DEADLINE = [start_time + total]

    try:
        iseed = int(seed)
    except (Exception):
        iseed = 12345
    _RNG = np.random.default_rng((iseed % 1000000007) + 1)

    print(f"Starting optimization, budget={total:.0f}s")

    cands = []

    # Try to use previous best
    try:
        parent_raw = height_sequence_1
        p = _sanitize(parent_raw)
        if p is not None:
            sc = _score(p)[0]
            cands.append((sc, p.size, p.copy()))
            print(f"  Loaded previous best: n={p.size}, score={sc:.7f}")
    except (NameError, Exception):
        pass

    if not cands:
        p = np.ones(1024, dtype=float)
        sc = _score(p)[0]
        cands.append((sc, 1024, p))
        print(f"  No previous best, starting uniform: score={sc:.7f}")

    verbose = total > 300

    # Multi-scale optimization
    if total > 2000:
        levels = [(512, 0.04), (1024, 0.06), (2048, 0.10), (4096, 0.22), (8192, 0.30)]
    elif total > 1000:
        levels = [(512, 0.05), (1024, 0.08), (2048, 0.12), (4096, 0.35), (8192, 0.25)]
    elif total > 500:
        levels = [(512, 0.06), (1024, 0.10), (2048, 0.15), (4096, 0.45)]
    else:
        levels = [(512, 0.10), (1024, 0.15), (2048, 0.20), (4096, 0.40)]

    reserved = max(120.0, 0.12 * total)
    parent = None

    for nn, frac in levels:
        rem = time_limit_remaining()
        if rem < reserved + 60:
            break
        t_alloc = frac * total
        t_alloc = min(t_alloc, rem - reserved - 30)
        if t_alloc < 20:
            continue

        print(f"  Level n={nn}, time={t_alloc:.0f}s")
        old_best = cands[0][0] if cands else float('inf')
        _process_level(parent, cands, nn, t_alloc, verbose)

        if cands:
            new_best = cands[0][0]
            if new_best < old_best:
                print(f"    Improved: {old_best:.7f} -> {new_best:.7f}")
            parent = cands[0][2]

    # Final fine optimization at the best level
    if not cands:
        arr = np.ones(2048, dtype=float)
        return [float(x) for x in arr]

    best_item = cands[0]
    best_arr = best_item[2].copy()
    best_sc = best_item[0]
    n_best = best_arr.size

    rem = time_limit_remaining()
    print(f"  Final optimization: n={n_best}, score={best_sc:.7f}, time={rem:.0f}s")

    # Phase 1: Extended multiplicative
    t1 = min(rem * 0.3, 600.0)
    if t1 > 30:
        p, sc = _mul_optimize(best_arr, t1, target=1.49, verbose=verbose)
        if p is not None and sc < best_sc:
            best_sc = sc
            best_arr = p
            print(f"    After mul: {best_sc:.7f}")

    rem = time_limit_remaining()
    # Phase 2: L-BFGS
    t2 = min(rem * 0.4, 600.0)
    if t2 > 30:
        p, sc = _lbfgs_optimize(best_arr, t2, beta_start=80, beta_end=1000, verbose=verbose)
        if p is not None and sc < best_sc:
            best_sc = sc
            best_arr = p
            print(f"    After lbfgs: {best_sc:.7f}")

    rem = time_limit_remaining()
    # Phase 3: More L-BFGS with higher beta
    t3 = min(rem * 0.3, 300.0)
    if t3 > 20:
        p, sc = _lbfgs_optimize(best_arr, t3, beta_start=200, beta_end=3000, verbose=verbose)
        if p is not None and sc < best_sc:
            best_sc = sc
            best_arr = p
            print(f"    After lbfgs2: {best_sc:.7f}")

    rem = time_limit_remaining()
    # Phase 4: Final multiplicative polish
    t4 = max(10.0, rem * 0.2)
    if t4 > 10:
        p, sc = _mul_optimize(best_arr, t4, target=1.49, verbose=verbose)
        if p is not None and sc < best_sc:
            best_sc = sc
            best_arr = p
            print(f"    After polish: {best_sc:.7f}")

    # Evaluate with official evaluator for small sequences
    if best_arr.size <= 4096 and time_limit_remaining() > 5:
        lst = [float(x) for x in best_arr]
        try:
            sc_official = float(evaluate_sequence(lst))
            if np.isfinite(sc_official):
                print(f"  Official score: {sc_official:.7f}")
        except (Exception):
            pass

    out = np.asarray(best_arr, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0:
        out = np.ones(2048, dtype=float)
    else:
        out = out / s

    print(f"  Final: n={out.size}, score={best_sc:.7f}")
    return [float(x) for x in out]
