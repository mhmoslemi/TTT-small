def solve_lp(a, M):
    """Solve: max sum(b) s.t. b >= 0, conv(a,b) <= M componentwise"""
    n = len(a)
    # conv(a,b)[k] = sum_j a[j]*b[j-k] for k = -(n-1)..(n-1)
    # For k >= 0: sum_{j=k}^{n-1} a[j-k]*b[j]
    # For k < 0: sum_{j=0}^{n-1-k} a[j-k]*b[j] = sum_{j=0}^{n-1+k} a[j+|k|]*b[j]

    # Row k (0-indexed, corresponding to lag k-(n-1)):
    #   For lag = 0: A[0] = [a[0], a[1], ..., a[n-1]] -> sum a[j]*b[j]
    #   For lag = 1: A[1] = [0, a[0], a[1], ..., a[n-2]] -> sum a[j-1]*b[j] for j>=1
    #   For lag = k (0<=k<=n-1): A[k] has a[j-k] at position j for j>=k
    #   For lag = -k (k>=1): A[n-1+k] has a[j+k] at position j for j<n-k
    
    # Actually, let's index differently.
    # conv(a,b)[k] for k = 0, 1, ..., 2n-2
    # conv(a,b)[k] = sum_j a[j]*b[k-j] where 0<=j<n, 0<=k-j<n
    # = sum_j a[j]*b[k-j] for j in [max(0, k-n+1), min(n-1, k)]
    
    n_constraints = 2*n - 1
    # A is n_constraints x n
    # A[k, j] = a[j] if 0 <= k-j < n, else 0
    # = a[j] if j <= k <= j+n-1, else 0
    # = a[j] if k-j in [0, n-1], else 0
    
    # So for column j, A[k, j] = a[j] for k in [j, j+n-1]
    # This is a banded structure!
    
    A = lil_matrix((n_constraints, n))
    for j in range(n):
        A[j:j+n, j] = a[j]
    
    A = csc_matrix(A)
    
    # Objective: max sum(b) -> min -sum(b)
    c_obj = -np.ones(n)
    
    # Bounds: b >= 0

    if res.status == 0:

    return None

def _next_pow2(x):

def _conv_full(a, b):

def _project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return np.array([])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

    return None


def _qp_toeplitz_mat(weights, n):
    """Build the n x n Toeplitz QP matrix from correlation weights."""
    first_col = np.zeros(n)
    first_col[0] = weights[0]
    if len(weights) > 1:
        first_col[1:] = weights[1:min(n, len(weights))] / 2.0
    return toeplitz(first_col)


def _solve_qp_osqp(Q, n):
    """Solve: min 0.5*a^T Q a s.t. sum(a)=1, a>=0 using OSQP."""
    if not HAS_OSQP:
        return None
    # A matrix: row 0 is ones (equality), rows 1..n are identity (a >= 0)
    A = np.zeros((n + 1, n))
    A[0, :] = 1.0
    A[1:, :] = np.eye(n)
    l = np.zeros(n + 1)
    l[0] = 1.0
    u = np.zeros(n + 1)
    u[0] = 1.0
    u[1:] = np.inf

    P = csc_matrix(Q)
    m = osqp.OSQP()
    m.setup(P, np.zeros(n), csc_matrix(A), l, u,
            eps_abs=1e-8, eps_rel=1e-8, max_iter=5000,
            polish=True, verbose=False)
    result = m.solve()
    if result.info.status_val == 1:  # solved
        x = np.maximum(result.x, 0)
        s = x.sum()
        if s > 1e-10:
            return x / s
    return None


def _solve_qp_projected_gradient(Q, n, max_iter=200, verbose=False):
    """Solve QP via projected gradient with backtracking."""
    a = np.ones(n) / n  # start at uniform
    # Estimate Lipschitz constant
    eig_max = np.linalg.eigvalsh(Q)[0] if n <= 5000 else 2.0 * np.trace(Q) / n
    L = max(2.0 * eig_max, 1.0)
    step = 1.0 / L

    for it in range(max_iter):
        grad = 2.0 * Q.dot(a)
        a_new = a - step * grad
        a_new = _project_simplex(a_new)
        # Check convergence
        if np.max(np.abs(a_new - a)) < 1e-9:
            a = a_new
            break
        a = a_new

    return a


def _qp_iteration(a, p_sharpen=1.0, use_osqp=True, max_iter=200):
    """
    One QP iteration: compute weights from conv(a,a), solve reweighted QP.
    Returns new sequence or None.
    """
    n = len(a)
    c = _conv_full(a, a)
    c = np.maximum(c, 0)
    c_max = c.max()
    if c_max < 1e-15:
        return a.copy()

    # Symmetrize: use first n values (lags 0 to n-1)
    w = c[:n].copy()
    # Normalize so w[0] = 1
    if w[0] > 0:
        w = w / w[0]
    # Sharpen
    if p_sharpen != 1.0:
        w = w ** p_sharpen
        w = w / w[0]  # renormalize peak to 1

    # Build Q matrix
    Q = _qp_toeplitz_mat(w, n)

    # Check PSD (eigenvalue check for small problems or trust it for large)
    if n <= 1000:
        eig_min = np.linalg.eigvalsh(Q)[-1]
        if eig_min < -0.01:
            Q += (-eig_min + 0.01) * np.eye(n)

    # Solve
    if use_osqp and HAS_OSQP:
        result = _solve_qp_osqp(Q, n)
    else:
        result = _solve_qp_projected_gradient(Q, n, max_iter=max_iter)

    if result is None:
        return None
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s


def _lp_optimize(a0, time_budget, verbose=False):
    """LP-based iterative optimization."""
    a = _sanitize(a0)
    if a is None:
        return None, float('inf')
    n = len(a)
    best = a.copy()
    best_score = _conv_score_normalized(a)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        c = _conv_full(a, a)
        M = float(c.max())
        if M < 1e-15:
            break

        b = _lp_direction(a, M)
        if b is None:
            stall += 1
            if stall > 20:
                break
            continue

        S = float(b.sum())
        if S < 1.0 + 1e-6:
            # LP says no improvement possible
            stall += 1
            if stall > 30:
                break
            continue

        g = b / S
        # Line search
        best_t = None
        best_sc = cur_score
        for t in (0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9):
            a_try = (1 - t) * a + t * g
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _conv_score_normalized(a_try)
            if sc_try < best_sc - 1e-12:
                best_sc = sc_try
                best_t = a_try

        if best_t is not None:
            a = best_t
            cur_score = best_sc
            stall = 0
            if cur_score < best_score - 1e-12:
                best = a.copy()
                best_score = cur_score
                if verbose and it % 20 == 0:
                    print(f"      LP it={it}, S={S:.4f}, score={best_score:.6f}")
        else:
            stall += 1
            if stall > 40:
                a = best.copy()
                cur_score = best_score
                stall = 10

        it += 1

    return best, best_score


def _qp_optimize(a0, time_budget, p_sharpen=2.0, verbose=False):
    """QP-based iterative optimization."""
    a = _sanitize(a0)
    if a is None:
        return None, float('inf')
    n = len(a)
    best = a.copy()
    best_score = _conv_score_normalized(a)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0
    it = 0
    use_osqp = HAS_OSQP

    while time.monotonic() < deadline:
        a_new = _qp_iteration(a, p_sharpen=p_sharpen, use_osqp=use_osqp,
                              max_iter=300)
        if a_new is None:
            stall += 1
            if stall > 20:
                break
            if stall % 10 == 0:
                p_sharpen = max(1.0, p_sharpen * 0.9)
            continue

        # Line search between a and a_new
        best_t = None
        best_sc = cur_score
        for t in (0.1, 0.3, 0.5, 0.7, 1.0):
            a_try = (1 - t) * a + t * a_new
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc_try = _conv_score_normalized(a_try)
            if sc_try < best_sc - 1e-12:
                best_sc = sc_try
                best_t = a_try

        if best_t is not None:
            a = best_t
            cur_score = best_sc
            stall = 0
            if cur_score < best_score - 1e-12:
                best = a.copy()
                best_score = cur_score
                if verbose and it % 50 == 0:
                    print(f"      QP it={it}, p={p_sharpen:.1f}, score={best_score:.6f}")
        else:
            stall += 1
            if stall > 50:
                p_sharpen = p_sharpen * 1.2
                if p_sharpen > 20.0:
                    p_sharpen = 1.5
            if stall > 100:
                a = best.copy()
                cur_score = best_score
                stall = 20

        it += 1

    return best, best_score


def _projected_gradient_optimize(a0, time_budget, verbose=False):
    """Projected gradient on the smoothed max of autocorrelation."""
    a = _sanitize(a0)
    if a is None:
        return None, float('inf')
    n = len(a)
    best = a.copy()
    best_score = _conv_score_normalized(a)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0
    beta = 50.0
    lr = 0.01
    it = 0

    while time.monotonic() < deadline:
        c = _conv_full(a, a)
        c = np.maximum(c, 0)
        c_max = c.max()
        if c_max < 1e-15:
            break
        c_norm = c / c_max

        # Softmax weights
        shift = c_norm.max()
        w = np.exp(beta * (c_norm - shift))
        s_w = w.sum()
        if s_w < 1e-300:
            w = np.ones(len(c)) / len(c)
        else:
            w = w / s_w

        # Gradient of sum_k w[k] * conv(a,a)[k] w.r.t. a[j]
        # d/d a[j] conv(a,a)[k] = a[j-k] + a[j+k] (with boundary handling)
        # = 2 * (a cross-correlated with itself at appropriate lags)
        # More precisely: grad[j] = sum_k w[k] * (a[j-k] + a[j+k]) where valid
        # This is: 2 * sum_k w[k] * (circulant cross-corr)
        # Using FFT: grad = 2 * ifft(rfft(w) * conj(rfft(a)))  [for circular]
        # For linear: need to be careful with boundaries

        L = _next_pow2(3 * n)
        W = np.fft.rfft(w, L)
        A = np.fft.rfft(a, L)
        R = np.fft.irfft(W * np.conj(A), L)
        grad = 2.0 * R[:n]

        # Normalize gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            stall += 1
            if stall > 50:
                break
            continue
        grad = grad / g_norm

        # Backtracking line search
        step = lr
        improved = False
        for _ in range(10):
            a_try = a - step * grad
            a_try = _project_simplex(a_try)
            sc_try = _conv_score_normalized(a_try)
            if sc_try < cur_score - 1e-13:
                a = a_try
                cur_score = sc_try
                improved = True
                break
            step *= 0.5

        if improved:
            stall = 0
            lr = min(lr * 1.1, 0.1)
            if cur_score < best_score - 1e-12:
                best = a.copy()
                best_score = cur_score
                if verbose and it % 100 == 0:
                    print(f"      PG it={it}, beta={beta:.0f}, lr={lr:.4f}, score={best_score:.6f}")
        else:
            stall += 1
            lr *= 0.7
            if stall > 30:
                beta = min(beta * 1.5, 5000.0)
            if stall > 80:
                a = best.copy()
                cur_score = best_score
                lr = 0.01
                stall = 10

        if it % 20 == 0:
            beta = min(beta * 1.05, 5000.0)
        it += 1

    return best, best_score


def _block_refine(a0, time_budget):
    """Block-level local search."""
    a = _sanitize(a0)
    if a is None:
        return None, float('inf')
    n = len(a)
    best = a.copy()
    best_score = _conv_score_normalized(a)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0
    rng = np.random.default_rng(42)
    factors = (0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 3.0)

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.15:
            bs = int(rng.integers(1, 4))
        elif r < 0.5:
            bs = int(rng.integers(max(2, n // 128), max(4, n // 16)))
        else:
            bs = int(rng.integers(max(4, n // 32), max(8, n // 4)))
        bs = min(bs, n)
        start = int(rng.integers(0, n - bs + 1))

        improved = False
        for f in factors:
            a_try = a.copy()
            a_try[start:start + bs] *= f
            a_try = np.maximum(a_try, 0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try = a_try / s_try
            sc = _conv_score_normalized(a_try)
            if sc < cur_score - 1e-12:
                cur_score = sc
                a = a_try
                improved = True
                break

        if improved:
            stall = 0
            if cur_score < best_score - 1e-12:
                best = a.copy()
                best_score = cur_score
        else:
            stall += 1
            if stall > 100:
                a = best.copy()
                cur_score = best_score
                stall = 0

    return best, best_score


def _spectral_flatten(a, alpha, L=None):
    """Frequency-domain spectral flattening."""
    n = len(a)
    if L is None:
        L = _next_pow2(2 * n)
    A = np.fft.rfft(a, L)
    C_freq = A * A
    C = np.fft.irfft(C_freq, L)[:2 * n - 1]
    cmax = C.max()
    if cmax < 1e-15:
        return a.copy()
    C_norm = C / cmax
    C_new = cmax * np.power(C_norm, 1.0 / alpha)
    C_new = np.maximum(C_new, 0)
    C_freq_new = np.fft.rfft(C_new, L)
    A_new = np.sqrt(C_freq_new.astype(complex) + 0j)
    a_new = np.fft.irfft(A_new, L)[:n]
    neg_frac = (a_new < 0).sum() / n
    if neg_frac > 0.4:
        return None
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s


def _init_uniform(n):
    return np.ones(n, dtype=float) / n


def _resample(p_in, n_new):
    p = _sanitize(p_in)
    if p is None:
        return None
    n_old = len(p)
    if n_old == n_new:
        return p.copy()
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    result = np.interp(x_new, x_old, p)
    result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s
