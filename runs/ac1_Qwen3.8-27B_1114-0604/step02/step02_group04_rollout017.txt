```python
def _fft_convolve(a):
    n = a.size
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
```

```python
def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]

    Compute subgradient of max(c) using soft-max weighting with temperature T.
    grad[m] = sum_k w[k] * a[k-m] (cross-correlation)

    n = a.size

    c_shift = c - c_max  # <= 0, max is 0
```

```python
from scipy.optimize import linprog

def _solve_lp_direction(a, M):
    """
    Solve: max sum(b) s.t. b >= 0, (a*b)[k] <= M for all k.
    """
    n = a.size
    # Objective: max sum(b) → min -sum(b)
    c_obj = -np.ones(n)
    
    # Constraints: (a*b)[k] <= M for k = 0, ..., 2n-2
    # (a*b)[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} a[i] * b[k-i]
    # For a given k, the coefficients of b[j] (j=0,...,n-1) are:
    #   coeff[j] = a[k-j] if 0 <= k-j < n, else 0
    
    # Build constraint matrix
    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_ub[k, j] = a[i]
    
    b_ub = np.full(2 * n - 1, M)
    
    # Bounds: b >= 0
    bounds = [(0, None)] * n
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if not result.success:
        return None
    return result.x
```

`max(a_new * a_new) <= (1-t)^2 * max(a*a) + 2t(1-t) * max(a*b) + t^2 * max(b*b)`
`= (1-t)^2 * M + 2t(1-t) * M + t^2 * max(b*b)`
`= M * ((1-t)^2 + 2t(1-t)) + t^2 * max(b*b)`
`= M * (1-t) * (1-t+2t) + t^2 * max(b*b)`
`= M * (1-t) * (1+t) + t^2 * max(b*b)`
`= M * (1-t^2) + t^2 * max(b*b)`
`= M - t^2 * M + t^2 * max(b*b)`

So `max(a_new * a_new) <= M + t^2 * (max(b*b) - M)`.

If `max(b*b) < M`, this is a decrease! And since `b` is optimized to maximize its sum (spreading it out), `max(b*b)` should be smaller than `M` (which is the peak of `a*a`).

But we also need the sum to be preserved. Since we rescale `b` to have the same sum as `a`, and the convolution is homogeneous of degree 2, the rescaled `b` satisfies:
`max(b_rescaled * b_rescaled) = max(b*b) * (sum(a)/sum(b))^2`

If `sum(b) > sum(a)` (which the LP tries to maximize), then `b_rescaled` has smaller values, and `max(b_rescaled * b_rescaled) < max(b*b)`.

So the overall effect should be a decrease in the peak, which is what we want.
