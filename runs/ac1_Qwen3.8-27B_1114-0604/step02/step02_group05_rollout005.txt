def _peak_targeted_update(p, rng, n_trials=20):
    """
    Directly target the convolution peaks.
    For the peak at position k*, reduce the contributing pairs.
    """

    b = _conv_fast(p)
    k_star = int(np.argmax(b))
    b_max = b[k_star]
    s = p.sum()
    
    current_score = 2.0 * n * b_max / (s * s)
    
    best_p = p.copy()
    best_score = current_score
    
    for _ in range(n_trials):
        p_try = p.copy()
        # Pick a random subset of indices that contribute to b_{k_star}
        valid = np.where((np.arange(n) <= k_star) & (k_star - np.arange(n) < n))[0]
        if len(valid) == 0:
            continue
        # Reduce a random few of these
        n_reduce = int(rng.integers(1, max(2, len(valid) // 4) + 1))
        idx = rng.choice(valid, size=min(n_reduce, len(valid)), replace=False)
        factor = float(rng.uniform(0.3, 0.8))
        p_try[idx] *= factor
        p_try = np.maximum(p_try, 0.0)
        sc = _score(p_try)
        if sc < best_score:
            best_score = sc
            best_p = p_try.copy()
    
    return best_p, best_score

def _freq_perturb(p, rng):
    """
    Perturb in frequency domain and project back.
    """
    n = len(p)
    L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(p, L)
    
    # Random cosine perturbation
    k1 = int(rng.integers(1, L // 2 + 1))
    k2 = int(rng.integers(1, L // 2 + 1))
    amp1 = float(rng.uniform(-0.3, 0.3))
    amp2 = float(rng.uniform(-0.3, 0.3))
    phase1 = float(rng.uniform(0, 2 * np.pi))
    phase2 = float(rng.uniform(0, 2 * np.pi))
    
    A_new = A.copy()
    # Multiply by (1 + amp*cos(k*omega + phase))
    omega = np.pi * np.arange(L // 2 + 1) / (L // 2)
    A_new *= (1.0 + amp1 * np.cos(k1 * omega + phase1) + amp2 * np.cos(k2 * omega + phase2))
    
    # Ensure Hermitian (for real signal)
    a_new = np.fft.irfft(A_new, L)[:n]
    a_new = np.maximum(a_new, 0.0)
    ss = a_new.sum()
    if ss < 1e-15:
        return None
    return a_new / ss

def _simulated_annealing(p0, rng, time_budget, T_start=0.01, T_end=0.0001):
    """

    p = p0.copy()
    n = len(p)
    best = p.copy()
    best_score = _score(p)

    it = 0
    n_iters = 0
    
    while time.monotonic() < deadline:
        # Temperature schedule
        frac = min(1.0, (time.monotonic() - (deadline - time_budget)) / time_budget)
        T = T_start * (T_end / T_start) ** frac
        
        # Choose a move
        r = rng.random()
        p_try = None
        
        if r < 0.3:
            # Local Gaussian perturbation
            window = int(rng.integers(max(1, n // 64), max(2, n // 8)))

        elif r < 0.5:
            # Peak-targeted reduction
            b = _conv_fast(p)

                p_try = p.copy()
                n_red = int(rng.integers(1, max(2, len(valid) // 8) + 1))
                idx = rng.choice(valid, size=min(n_red, len(valid)), replace=False)
                p_try[idx] *= float(rng.uniform(0.2, 0.7))
                p_try = np.maximum(p_try, 0)
                
        elif r < 0.7:
            # Frequency domain perturbation
            p_try = _freq_perturb(p, rng)
            
        elif r < 0.85:
            # Pair swap/blend
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                p_try = p.copy()
                t = float(rng.uniform(0.1, 0.9))
                p_try[i], p_try[j] = p[i] + t*(p[j]-p[i]), p[j] + t*(p[i]-p[j])
                p_try = np.maximum(p_try, 0)
                
        else:

            p_try = p * np.exp(rng.normal(0, 0.05, n))

        ss = p_try.sum()
        if ss < 1e-15:

        p_try = p_try / ss
        
        sc_try = _score(p_try)

        # Subgradient of max(b) w.r.t. p
        other = k_star - np.arange(n)
        mask = (other >= 0) & (other >= 0) & (other < n)
        s_grad = np.zeros(n)
        s_grad[mask] = 2.0 * p[other[mask]]
        
        # Normalize
        sg_max = s_grad.max()
        if sg_max < 1e-15:
            break
        s_grad_norm = s_grad / sg_max
        
        # Multiplicative update: reduce p where gradient is large
        # We want to reduce p_j where increasing it increases max(b)
        # The subgradient tells us which direction increases max(b)
        # So we decrease p in that direction: p *= exp(-alpha * s_grad_norm)
        
        best_trial = None
        best_trial_sc = cur_score
        for a_try in (alpha * 0.5, alpha, alpha * 1.5, alpha * 2):
            p_try = p * np.exp(-a_try * s_grad_norm)
            p_try = np.maximum(p_try, 1e-20)
            ss = p_try.sum()
            if ss < 1e-15:
                continue
            p_try = p_try / ss
            sc = _score(p_try)
            if sc < best_trial_sc:
                best_trial = p_try
                best_trial_sc = sc
        
        if best_trial is not None:
            p = best_trial
            cur_score = best_trial_sc
            stall = 0
            if cur_score < best_score:
                best = p.copy()
                best_score = cur_score
        else:
            stall += 1
            alpha *= 0.9
            if stall > 20:
                # Try a different peak
                b = _conv_fast(p)
                top_k = np.argsort(b)[-5:]
                k_try = int(rng.choice(top_k))
                other = k_try - np.arange(n)
                mask = (other >= 0) & (other < n)
                s_grad2 = np.zeros(n)
                s_grad2[mask] = 2.0 * p[other[mask]]
                if s_grad2.max() > 1e-15:
                    s_grad2_norm = s_grad2 / s_grad2.max()
                    p_try = p * np.exp(-0.1 * s_grad2_norm)
                    p_try = np.maximum(p_try, 1e-20)
                    p_try = p_try / p_try.sum()
                    sc = _score(p_try)
                    if sc < best_score:
                        p = p_try
                        cur_score = sc
                        best = p.copy()
                        best_score = sc
                    stall = 0
            if stall > 50:
                p = best.copy()
                cur_score = best_score
                alpha = 0.05
                stall = 0
    
    return best, best_score

def _top_k_penalty_method(p0, rng, time_budget, k=5):
    """
    Optimize sum of top-k convolution values (as a smooth proxy for max).
    """
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    
    best = p.copy()
    best_score = _score(p)
    cur_score = best_score
    alpha = 0.05
    deadline = time.monotonic() + time_budget
    stall = 0
    
    while time.monotonic() < deadline:
        b = _conv_fast(p)
        # Get top-k positions
        top_k_idx = np.argsort(b)[-k:]
        b_top = sorted(b[top_k_idx], reverse=True)
        
        # Gradient of sum of top-k w.r.t. p_j:
        # = sum_{k* in top_k} 2 * p_{k*-j} (if valid)
        s_grad = np.zeros(n)
        for ks in top_k_idx:
            other = ks - np.arange(n)
            mask = (other >= 0) & (other < n)
            s_grad[mask] += 2.0 * p[other[mask]]
        
        sg_max = s_grad.max()
        if sg_max < 1e-15:
            break
        s_grad_norm = s_grad / sg_max
        
        # Multiplicative update
        best_trial = None
        best_trial_sc = cur_score
        for a_try in (alpha * 0.5, alpha, alpha * 2):
            p_try = p * np.exp(-a_try * s_grad_norm)
            p_try = np.maximum(p_try, 1e-20)
            ss = p_try.sum()
            if ss < 1e-15:
                continue
            p_try = p_try / ss
            sc = _score(p_try)
            if sc < best_trial_sc:
                best_trial = p_try
                best_trial_sc = sc
        
        if best_trial is not None:
            p = best_trial
            cur_score = best_trial_sc
            stall = 0
            if cur_score < best_score:
                best = p.copy()
                best_score = cur_score
        else:
            stall += 1
            alpha *= 0.9
            if stall > 30:
                p = best.copy()
                cur_score = best_score
                alpha = 0.05
                stall = 0
    
    return best, best_score

def _init_from_parent(parent, n, rng):
    """Resample parent to length n with small perturbation."""
    if parent is None:
        return np.ones(n) / n
    p = np.asarray(parent, dtype=float)
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-15:
        return np.ones(n) / n
    
    # Interpolate
    x_old = (np.arange(len(p)) + 0.5) / len(p)
    x_new = (np.arange(n) + 0.5) / n
    p_new = np.interp(x_new, x_old, p / s)
    p_new = np.maximum(p_new, 0)
    p_new = p_new * np.exp(rng.normal(0, 0.02, n))
    p_new = np.maximum(p_new, 0)
    ss = p_new.sum()
    if ss < 1e-15:
        return np.ones(n) / n
    return p_new / ss

def _init_uniform(n):
    return np.ones(n) / n

def _init_two_level(n, rng):
    """Two-level sequence with random pattern."""
    ratio = float(rng.uniform(1.5, 4.0))
    frac = float(rng.uniform(0.2, 0.5))
    p = np.ones(n, dtype=float)
    n_high = int(n * frac)
    # Place high values in a spread pattern
    step = n // n_high
    p[::step][:n_high] = ratio  # This isn't quite right
    # Better:
    p = np.ones(n, dtype=float)
    indices = np.sort(rng.choice(n, size=n_high, replace=False))
    p[indices] = ratio
    p = p * np.exp(rng.normal(0, 0.05, n))
    p = np.maximum(p, 0)
    return p / p.sum()

def _init_cosine(n, rng):
    t = np.arange(n, dtype=float) / n
    p = 1.0 + 0.5 * np.cos(2 * np.pi * t * int(rng.integers(1, 5)))
    p = np.maximum(p, 0)
    return p / p.sum()

def _init_gaussian(n, rng):
    """Gaussian bump in the middle."""
    t = np.arange(n, dtype=float) / n
    center = float(rng.uniform(0.3, 0.7))
    width = float(rng.uniform(0.1, 0.4))
    p = np.exp(-0.5 * ((t - center) / width) ** 2)
    p = np.maximum(p, 1e-10)
    return p / p.sum()
