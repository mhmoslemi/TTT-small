    # d(conv[k])/da[i] = a[k-i] + a[k-i] (two terms) = 2*a[k-i] if 0 <= k-i <= n-1
    # (except when k=2i where it's 2*a[i]+da[i] but for gradient purposes it's 2*a[i])
    # So d(max(conv))/da[i] = sum_k w[k] * 2*a[k-i] (where 0 <= k-i <= n-1)
    
    # Using FFT: this is a cross-correlation of w with a
    N2 = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N2)
    fw = np.fft.rfft(conv, N2)  # w here is weight on conv positions
    # cross-correlation: grad[i] = sum_k w[k] * a[k-i]
    # This is (w reversed) * a in frequency domain? Let me think...
    
    # Actually, let me use the direct formula:
    # grad[i] = 2 * sum_{k: 0<=k-i<=n-1} w[k] * a[k-i]
    # Let j = k-i, so k = i+j:
    # grad[i] = 2 * sum_{j: 0<=j<=n-1, 0<=i+j<=2n-2} w[i+j] * a[j]
    # = 2 * sum_{j: 0<=j<=n-1} w[i+j] * a[j] (since i+j <= i+n-1 <= 2n-2 for i <= n-1)
    
    # This is the cross-correlation of w with a, evaluated at lag i.
    # In FFT: cross_corr[i] = IDFT(conj(FA) * FW)[i]... let me be careful.
    
    # Let's define: (w * a)[i] = sum_j w[i+j] * a[j]
    # This is the "anti-convolution" or cross-correlation.
    # In terms of standard convolution: if we define w_rev[k] = w[2n-2-k],
    # then (w_rev * a)[2n-2-i] = sum_j w_rev[2n-2-i-j] * a[j] = sum_j w[i+j] * a[j]
    
    # So grad[i] = 2 * (w_rev * a)[2n-2-i] where * is linear convolution.
    
    # Using FFT for linear convolution:
    N3 = _next_pow2(2 * n - 1)
    w_rev = conv[::-1]  # w_rev[k] = w[2n-2-k]
    fa = np.fft.rfft(a, N3)
    fr = np.fft.rfft(w_rev, N3)
    conv_result = np.fft.irfft(fa * fr, N3)
    # conv_result has length 2n-1
    # grad[i] = 2 * conv_result[2n-2-i]
    grad = 2.0 * conv_result[2*n-2 : 0 : -1]  # reverse: [2n-2, 2n-3, ..., 1, 0] -> first n elements
    
    # Wait, let me double-check the indexing.
    # w_rev * a = linear convolution of w_rev (length 2n-1) and a (length n)
    # Result has length (2n-1) + n - 1 = 3n-2
    # That's not what I want. Let me reconsider.
    
    # Actually, I think the cleanest way is:
    # grad[i] = 2 * sum_{j=0}^{n-1} a[j] * w[i+j]
    # where w[k] = 0 if k < 0 or k >= 2n-1
    
    # Define v[j] = w[j] (padded to length >= n, but w has length 2n-1)
    # Then grad[i] = 2 * sum_{j=0}^{n-1} a[j] * v[i+j]
    # This is the cross-correlation of a and v at lag i.
    
    # Using FFT: cross_corr[i] = IDFT(conj(FA) * FV)[i]
    # where FA = FFT(a padded to N), FV = FFT(v padded to N)
    # But we need N >= n + (2n-1) - 1 = 3n-2 for circular to equal linear cross-correlation.
    
    # Hmm, this is getting complicated. Let me just use the formula directly for small n,
    # or use a different FFT formulation.
    
    # Alternative: use the fact that grad[i] = 2 * (a * w_rev_padded)[i]
    # where w_rev_padded is w reversed and the convolution is circular with appropriate N.
    
    # Let me just use numpy for the cross-correlation directly (it's O(n^2) but n=1024 is fine):
    # Actually for n=1024, O(n^2) = 1M operations per gradient computation.
    # That's fast enough. Let me just do it directly.
    
    # For efficiency, let me use FFT properly.
    # The cross-correlation R[i] = sum_j a[j] * v[i+j]
    # = sum_j a[j] * v[j+i]
    # If I define a_rev[j] = a[n-1-j], then:
    # R[i] = sum_j a_rev[n-1-j] * v[i+j]
    # Let k = n-1-j, so j = n-1-k:
    # R[i] = sum_k a_rev[k] * v[i + n - 1 - k]
    # = (a_rev * v)[i + n - 1]  (linear convolution)
    
    # So grad[i] = 2 * (a_rev * v)[i + n - 1] where v = w (the weight vector on conv)
    
    N4 = _next_pow2((n) + (2*n-1) - 1)  # = _next_pow2(3n-2)
    a_rev = a[::-1]
    fa = np.fft.rfft(a_rev, N4)
    fv = np.fft.rfft(conv, N4)
    prod = np.fft.irfft(fa * fv, N4)
    # prod[k] = (a_rev * v)[k] for k = 0, ..., 3n-3
    # grad[i] = 2 * prod[i + n - 1] for i = 0, ..., n-1
    grad = 2.0 * prod[n-1 : 2*n-1]  # prod[n-1], prod[n], ..., prod[2n-2]
    
    return grad


def _make_multi_sine(n, modes, amps, phases):
    """Create multi-mode sinusoidal sequence."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64) / n
    for m, amp, ph in zip(modes, amps, phases):
        theta = 2.0 * np.pi * (m * i + ph) / n
        a += amp * np.sin(theta) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _optimize_mirror(a, deadline, T_start=20.0, T_end=80.0, lr0=None, 
                      lp_interval=0, lp_budget_max=30.0):
    """
    Mirror descent optimization with log-sum-exp smoothing.
    Optionally performs LP-based direction correction periodically.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size, dtype=np.float64) / a.size
        s = 1.0
    a = a / s

    n = a.size
    if lr0 is None:
        lr0 = 0.3 * n

    best_a = a.copy()
    best_score = _score(best_a)

    it = 0
    last_improve = 0
    patience = 3000
    eval_interval = 3
    rng = np.random.RandomState(42)
    lp_time_spent = 0.0

    while time.monotonic() < deadline:
        it += 1

        # Annealing schedule (nonlinear)
        t_frac = min(1.0, it / 200000.0)
        T = T_start + (T_end - T_start) * (t_frac ** 0.5)
        lr = lr0 / (1.0 + 0.005 * it)

        # Compute convolution
        c = _fft_convolve(a)
        if c.size == 0:
            break

        # Soft-max weights
        c_max = float(np.max(c))
        c_shift = (c - c_max) / max(c_max, 1e-15)  # normalize, max is 0
        # Use a scaled temperature for numerical stability
        T_eff = T * c_max  # This scales with the peak value
        # Actually, let's use a fixed scaling:
        # We want exp(T * (c[k] - c_max) / c_max) to be well-behaved
        # (c[k] - c_max)/c_max is in [-1, 0]
        # So we use T_scaled = T (which is in [T_start, T_end])
        c_norm = (c - c_max) / c_max  # in [-1, 0]
        exp_c = np.exp(T * c_norm)
        s_exp = exp_c.sum()
        if s_exp < 1e-300:
            w = np.zeros(c.size)
            w[np.argmax(c)] = 1.0
        else:
            w = exp_c / s_exp

        # Gradient via FFT
        grad = _grad_self_conv(a, w)

        # Project to zero-mean (preserve sum)
        grad_proj = grad - np.mean(grad)

        # Mirror descent update
        update = -lr * grad_proj / (np.std(grad_proj) + 1e-10)
        update = np.clip(update, -5.0, 5.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        # Periodic evaluation
        if it % eval_interval == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                # Try LP-based correction
                if lp_interval > 0 and it % lp_interval == 0 and lp_time_spent < lp_budget_max:
                    lp_start = time.monotonic()
                    g = _lp_direction(best_a)
                    lp_time_spent += time.monotonic() - lp_start
                    if g is not None and time.monotonic() < deadline:
                        # Rescale g to match sum(a)
                        g_sum = g.sum()
                        if g_sum > 0.01:
                            g = g * (s / g_sum)
                            # Try different blend factors
                            for t_blend in [0.1, 0.2, 0.3, 0.5]:
                                trial = (1 - t_blend) * best_a + t_blend * g
                                trial = np.clip(trial, 0, 1000)
                                ts = _score(trial)
                                if np.isfinite(ts) and ts < best_score:
                                    best_score = ts
                                    best_a = trial.copy()
                                    a = trial.copy()
                                    a = a / a.sum()
                                    break
                else:
                    # Standard restart
                    a = best_a.copy()
                    noise = rng.standard_normal(n) * 0.02
                    a = a + noise * a.mean()
                    a = np.clip(a, 0, 1000)
                    if a.sum() > 0.01:
                        a = a / a.sum()
                    else:
                        a = best_a.copy()
                    patience = max(2000, int(patience * 1.3))
                    lr0 = max(0.05 * n, lr0 * 0.85)

    return best_a, best_score


def _lp_direction(f, timeout=5.0):
    """
    Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0.
    Returns optimal g or None.
    """
    n = len(f)
    M = float(np.max(_fft_convolve(f)))
    if M <= 0:
        return None

    # Build constraint matrix A (2n-1 rows, n columns)
    # A[k, j] = f[k-j] if 0 <= k-j <= n-1, else 0
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    c = -np.ones(n)  # minimize -sum(g)
    b = np.full(2 * n - 1, M)

    try:
        result = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs',
                         options={'time_limit': timeout})
        if result.status == 0 and result.x is not None:
            return np.maximum(result.x, 0)
    except Exception:
        pass
    return None


def _coordinate_descent(a, deadline, max_rounds=50):
    """
    Coordinate descent with golden section search on each coordinate.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return a, np.inf
    a = a / s

    n = a.size
    best_a = a.copy()
    best_score = _score(best_a)
    conv = _fft_convolve(a)

    rng = np.random.RandomState(12345)
    coord_order = list(range(n))
    rng.shuffle(coord_order)

    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break

        improved = False
        for i in coord_order:
            if time.monotonic() > deadline:
                break

            old_val = a[i]
            if old_val < 1e-15:
                continue

            # Precompute sensitivity vector
            # conv_new[k] = conv[k] + 2*(x - old_val) * a[k-i] for valid k != 2i
            # conv_new[2i] = conv[2i] + 2*old_val*(x-old_val) + (x-old_val)^2
            
            # Define: for a given x, the max of conv_new is:
            # max_k (conv[k] + 2*(x-old_val)*a[k-i]) adjusted at k=2i
            
            # Sensitivity: sens[k] = 2*a[k-i] if 0 <= k-i <= n-1, else 0
            # (for k != 2i; at k=2i we add extra (x-old_val)^2)
            
            k_indices = np.arange(2 * n - 1)
            valid = (k_indices >= i) & (k_indices <= i + n - 1)
            sens = np.zeros(2 * n - 1)
            sens[valid] = 2.0 * a[k_indices[valid] - i]
            # At k=2i, the linear part is 2*a[i]*(x-old_val), and we add (x-old_val)^2
            # sens[2i] already = 2*a[i], so the linear part is correct.
            # We just need to add (x-old_val)^2 at position 2i.

            S_old = a.sum()
            i2 = 2 * i

            def score_at_x(x):
                delta = x - old_val
                S_new = S_old + delta
                if S_new < 0.01:
                    return np.inf
                conv_new_max = np.max(conv + 2 * delta * sens)
                if i2 < len(conv_new_max):
                    # Need to add delta^2 at position i2
                    val_at_i2 = conv[i2] + 2 * delta * sens[i2] + delta * delta
                    conv_new_max = max(conv_new_max, val_at_i2)
                if conv_new_max < 0:
                    conv_new_max = 0
                return 2.0 * n * conv_new_max / (S_new * S_new)

            # Golden section search on [0, upper]
            upper = max(4 * a.mean(), old_val * 3)
            lower = 0.0
            gr = (np.sqrt(5) - 1) / 2

            c1 = upper - gr * (upper - lower)
            c2 = lower + gr * (upper - lower)
            f1 = score_at_x(c1)
            f2 = score_at_x(c2)

            for _ in range(35):
                if f1 < f2:
                    upper = c2
                    c2 = c1
                    f2 = f1
                    c1 = upper - gr * (upper - lower)
                    f1 = score_at_x(c1)
                else:
                    lower = c1
                    c1 = c2
                    f1 = f2
                    c2 = lower + gr * (upper - lower)
                    f2 = score_at_x(c2)

            best_x = (lower + upper) / 2
            best_f = score_at_x(best_x)

            # Also try x=0 (setting the coordinate to zero)
            f_zero = score_at_x(0.0)
            if f_zero < best_f:
                best_x = 0.0
                best_f = f_zero

            if best_f < best_score - 1e-12:
                a[i] = best_x
                # Update convolution
                delta = best_x - old_val
                conv = conv + 2 * delta * sens
                if i2 < len(conv):
                    conv[i2] += delta * delta  # add the quadratic correction
                # Wait, this is wrong. When we update, the sensitivity changes for future iterations.
                # Let me just recompute conv periodically.
                best_a = a.copy()
                best_score = best_f
                improved = True
                old_val = best_x  # update for next use

        # Recompute convolution every round for accuracy
        conv = _fft_convolve(a)
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        if not improved:
            break

    return best_a, best_score


def _pairwise_swap(a, deadline, n_trials=100000):
    """
    Local search via pairwise mass transfer.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return a, np.inf
    a = a / s

    n = a.size
    best_a = a.copy()
    best_score = _score(best_a)
    rng = np.random.RandomState(777)

    trial = 0
    while time.monotonic() < deadline and trial < n_trials:
        trial += 1
        if trial % 1000 == 0:
            # Recompute score periodically (in case of drift)
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

        # Pick two random indices
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue

        # Try transferring some mass from i to j (or vice versa)
        ai = a[i]
        aj = a[j]
        if ai < 1e-15:
            continue

        # Try different transfer amounts
        for frac in [0.05, 0.1, 0.2, 0.5]:
            # Transfer from i to j
            delta = ai * frac
            a[i] = ai - delta
            a[j] = aj + delta
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                best_score = sc
                best_a = a.copy()
                ai = a[i]
                aj = a[j]
            else:
                # Revert
                a[i] = ai
                a[j] = aj

    return best_a, best_score


def _scan_initialization(deadline):
    """Scan various initialization families."""
    best_a = None
    best_score = np.inf
    rng = np.random.RandomState(777)

    lengths = [1024, 2048]

    for n in lengths:
        if time.monotonic() > deadline:
            break

        # Single-mode sinusoid
        for amp in [0.5, 0.7, 0.8, 0.9, 1.0, 1.2]:
            for phase in [0.0, 0.25, 0.5]:
                if time.monotonic() > deadline:
                    break
                a = _make_multi_sine(n, [1], [amp], [phase])
                sc = _score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()
                    print(f"  [init] n={n} mode=1 amp={amp} ph={phase}: {sc:.8f}", flush=True)

        # Two-mode
        for _ in range(10):
            if time.monotonic() > deadline:
                break
            modes = [1, rng.choice([2, 3])]
            amps = [rng.uniform(0.5, 1.0), rng.uniform(0.1, 0.4)]
            phases = [rng.uniform(0, 0.5), rng.uniform(0, 1.0)]
            a = _make_multi_sine(n, modes, amps, phases)
            sc = _score(a)
            if sc < best_score:
                best_score = sc
