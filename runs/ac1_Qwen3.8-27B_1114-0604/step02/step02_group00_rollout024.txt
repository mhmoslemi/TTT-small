        mag = np.abs(A)

        new_mag[0] = mag[0]  # Keep DC (sum) fixed
        if N % 2 == 0:

        a = np.maximum(a_new, 0)

        if S > 0:
            a = a / S * a0.sum()  # Preserve sum
            # Actually, score is scale-invariant, so we don't need to preserve sum

"""
Hybrid spectral-flattening + L-BFGS + chirp-init optimizer for minimum peak autocorrelation.
Uses iterative spectral flattening to generate diverse pre-optimized initializations,
then polishes with L-BFGS on LSE-smoothed objective. Includes CAZAC-inspired chirps.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def spectral_flatten(a0, N, n_iters=300, lam=0.5):
    """Iterative spectral flattening: alternate between flat-spectrum and non-negativity."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    for i in range(n_iters):
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc_mag = mag[1:]
        target = np.median(non_dc_mag)
        new_mag = mag.copy()
        new_mag[0] = mag[0]
        half = N // 2
        if N % 2 == 0:
            new_mag[half] = mag[half]
        for k in range(1, half + 1):
            if k == 0 or (N % 2 == 0 and k == half):
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (a0.sum() / S)
    return a


def spectral_flatten_anneal(a0, N, n_iters=400):
    """Spectral flattening with annealing: start strong, end weak."""
    a = a0.astype(np.float64).copy()
    n = len(a)
    for i in range(n_iters):
        progress = i / max(1, n_iters - 1)
        lam = 0.8 * (1.0 - progress) + 0.2
        A = np.fft.fft(a, N)
        mag = np.abs(A)
        phase = np.angle(A)
        non_dc_mag = mag[1:]
        target = np.median(non_dc_mag)
        new_mag = mag.copy()
        new_mag[0] = mag[0]
        half = N // 2
        if N % 2 == 0:
            new_mag[half] = mag[half]
        for k in range(1, half + 1):
            if k == 0 or (N % 2 == 0 and k == half):
                continue
            new_mag[k] = (1.0 - lam) * mag[k] + lam * target
            new_mag[N - k] = new_mag[k]
        new_A = new_mag * np.exp(1j * phase)
        a_new = np.random.default_rng(i).real(np.fft.ifft(new_A))[:n]
        a = np.maximum(a_new, 0.0)
        S = a.sum()
        if S > 1e-12:
            a = a * (a0.sum() / S)
    return a


def chirp_init(n, rng):
    """CAZAC-inspired chirp initializations (non-negative real parts)."""
    inits = []
    x = np.arange(n, dtype=np.float64)
    # Quadratic chirp: cos^2(pi * j^2 / (2n))
    inits.append(np.cos(np.pi * x * x / (2.0 * n)) ** 2 + 0.01)
    # Linear chirp variation
    freq = 1.0 + 3.0 * rng.random()
    inits.append(0.5 + 0.4 * np.cos(np.pi * freq * x * x / (2.0 * n)))
    # Cubic chirp
    inits.append(0.5 + 0.3 * np.cos(np.pi * x * x * x / (2.0 * n * n)))
    # Mixed chirp
    f1 = 1.0 + 4.0 * rng.random()
    f2 = 1.0 + 4.0 * rng.random()
    inits.append(0.5 + 0.3 * np.cos(np.pi * f1 * x * x / (2.0 * n)) + 0.2 * np.cos(np.pi * f2 * x / n))
    # Golay-like: sum of two shifted chirps
    a1 = np.cos(np.pi * x * x / (2.0 * n)) ** 2
    shift = int(n * rng.uniform(0.1, 0.4))
    a2 = np.roll(a1, shift)
    inits.append(0.5 * (a1 + a2) + 0.01)
    for s in inits:
        inits[-1] = np.clip(s, 0.01, 1000.0)
    return inits


def build_inits(n, rng, parent):
    """Generate a diverse set of initializations of length n."""
    inits = []
    if parent is not None and len(parent) == n:
        inits.append(parent.astype(np.float64).copy())
    inits.append(np.ones(n) * 0.5)
    x = np.arange(n, dtype=np.float64) / n
    for freq in [1, 2, 3, 5, 8]:
        inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * freq * x))
        inits.append(0.5 + 0.3 * np.sin(2.0 * np.pi * freq * x))
    inits.append(0.1 + 0.8 * (2.0 * x - 1.0) ** 2)
    inits.append(0.5 + 0.4 * (1.0 - (2.0 * x - 1.0) ** 2))
    # Chirp-based
    inits.extend(chirp_init(n, rng))
    # Random
    for scale in [0.1, 0.3, 0.5, 0.8]:
        inits.append(np.clip(0.5 + scale * rng.standard_normal(n), 0.01, 1000.0))
    # Block structure
    for nb in [4, 8, 16, 32]:
        per = max(1, n // nb)
        vals = 0.2 + 0.6 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())
    # Spectral flattening from simple inits
    N_fft = 1 << (2 * n - 1).bit_length()
    sf_uniform = spectral_flatten(np.ones(n) * 0.5, N_fft, n_iters=100, lam=0.5)
    inits.append(sf_uniform)
    sf_random = spectral_flatten(np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 10.0),
                                  N_fft, n_iters=100, lam=0.5)
    inits.append(sf_random)
    # Spectral flattening from chirp
    chirp = np.cos(np.pi * np.arange(n, dtype=np.float64) ** 2 / (2.0 * n)) ** 2 + 0.01
    sf_chirp = spectral_flatten(chirp, N_fft, n_iters=150, lam=0.4)
    inits.append(sf_chirp)
    return inits


def exact_subgrad_polish(a, total_iters=500):
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.3)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        a -= (0.001 / (1.0 + 0.01 * it)) * grad / (gn + 1e-30)
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def targeted_perturb(a, rng, strength=0.05):
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(3, n // 50)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:-top_count] if n > 2 * top_count else np.argsort(contrib)[:top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)

    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
