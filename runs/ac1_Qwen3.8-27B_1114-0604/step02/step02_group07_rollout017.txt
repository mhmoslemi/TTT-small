def fast_self_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1

    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

    """Compute soft-max objective and gradient."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return 1e10, np.zeros(n)
