    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score_fast(a):
    """Compute the evaluation score."""
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _grad_uploads(a, w):
    """Gradient of smooth max via FFT cross-correlation.
    Returns 2 * cross_correlation(a, w) of length n."""
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]


def _normalize(a):
    """Normalize to sum=1, clip to [0, 1000]."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones_like(a) / a.size
    return a / s


def _make_sine(n, amp, phase=0.0, freq=1):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * freq * (i + phase) / n
    a = 1.0 + amp * np.sin(theta)
    return _normalize(a)


def _make_multi_sine(n, components):
    """components: list of (amp, freq, phase)"""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for amp, freq, phase in components:
        theta = 2.0 * np.pi * freq * (i + phase) / n
        a += amp * np.sin(theta)
    return _normalize(a)


def _smooth_weights(c, T):
    """Compute softmax weights on convolution values."""
    c_max = np.max(c)
    exp_c = np.exp((c - c_max) / T)
    s = exp_c.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[c >= c_max - 1e-10] = 1.0
        w /= w.sum()
        return w
    return exp_c / s


def _top_k_weights(c, K, T_local=1.0):
    """Compute weights on top-K convolution peaks."""
    n = c.size
    if n <= K:
        return _smooth_weights(c, T_local)
    # Find top K indices
    top_idx = np.argpartition(c, -K)[-K:]
    w = np.zeros(n, dtype=np.float64)
    top_vals = c[top_idx]
    top_shift = top_vals - top_vals.max()
    weights = np.exp(top_shift / T_local)
    weights /= weights.sum()
    for i, k in enumerate(top_idx):
        w[k] = weights[i]
    return w

"""
Low-peak step-function optimization via momentum-based smoothed subgradient
descent with FFT acceleration, adaptive temperature, and active-set refinement.

Key improvements over prior approach:
- Nesterov-style momentum for faster convergence
- Gradient-norm-adaptive step size for stability
- Multi-phase: broad smooth optimization -> focused active-set polish
- Diverse initialization including provided sequence
- Robust time management to guarantee returning best solution
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear auto-convolution a*a via FFT, length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score_fast(a):
    """Compute the evaluation score (scale-invariant)."""
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _gradient(a, w):
    """
    Compute gradient of smooth max objective.
    dF/da[j] = 2 * sum_k w[k] * a[k-j] (where valid)
    = 2 * cross_correlation(a, w)[j]
    Using FFT: ifft(conj(fft(a)) * fft(w))
    """
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]

    """Normalize to sum=1, clip to [0, 1000]."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-12:
        return np.ones_like(a, dtype=np.float64) / a.size
    return a / s


def _make_sine(n, amp, phase=0.0, freq=1):
    """Sinusoidal initialization."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * freq * (i + phase) / n
    a = 1.0 + amp * np.sin(theta)
    return _normalize(a)


def _make_multi_freq(n, components):
    """Multi-frequency sinusoidal. components: list of (amp, freq, phase)."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for amp, freq, phase in components:
        theta = 2.0 * np.pi * freq * (i + phase) / n
        a += amp * np.sin(theta)
    return _normalize(a)


def _smooth_weights(c, T):
    """Softmax weights: w[k] = exp(c[k]/T) / sum(exp(c/T))."""
    c_max = float(np.max(c))
    exp_c = np.exp((c - c_max) / T)
    s = exp_c.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        mask = c >= c_max - 1e-10
        n_mask = mask.sum()
        if n_mask == 0:
            w[int(np.argmax(c))] = 1.0
        else:
            w[mask] = 1.0 / n_mask
        return w
    return exp_c / s


def _topk_weights(c, K, T_local=0.5):
    """Weights concentrated on top-K peaks with local softmax."""
    n = c.size
    if n <= K:
        return _smooth_weights(c, T_local)
    top_idx = np.argpartition(c, -K)[-K:]
    w = np.zeros(n, dtype=np.float64)
    top_vals = c[top_idx]
    top_shift = top_vals - top_vals.max()
    weights = np.exp(top_shift / T_local)
    weights /= weights.sum()
    for i, k in enumerate(top_idx):
        w[k] = weights[i]
    return w

    n = a.size

    cur_patience = patience

    while time.monotonic() < deadline:
        it += 1

        c = _fft_convolve(a)
        if c.size == 0:
            break

        w = _smooth_weights(c, T)
        g = _gradient(a, w)
        g = g - np.mean(g)

        # Momentum update
        velocity = beta * velocity + g

        v_norm = np.linalg.norm(velocity)
        if v_norm < 1e-15:
            break

        # Adaptive step: normalize velocity, scale by eta
        step = -eta * velocity / v_norm * np.sqrt(n)
        step = np.clip(step, -4.0, 4.0)

        a_new = a * np.exp(step)
        a_new = np.where(np.isfinite(a_new), a_new, 0.0)
        a = _normalize(a_new)

        if a.sum() < 1e-10:
            a = best_a.copy()
            velocity = np.zeros(n)
            continue

        if it % eval_every == 0:
            sc = _score_fast(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > cur_patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.015
                a = a * (1.0 + noise)
                a = _normalize(a)
                velocity = np.zeros(n)
                cur_patience = max(800, int(cur_patience * 0.75))
                T = min(T * 1.3, 50.0)
                if verbose:
                    print(f"        restart at iter {it}, T={T:.1f}, score={best_score:.8f}", flush=True)

        if verbose and it % 2000 == 0:
            print(f"        iter={it}, score={best_score:.8f}, T={T:.1f}", flush=True)

    return best_a, best_score


def _optimize_active(a_init, deadline, K=10, eta=0.12, beta=0.88,
                     eval_every=3, patience=2000):
    """Active-set optimization focusing on top-K peaks."""
    a = _normalize(a_init.astype(np.float64).copy())
    n = a.size

    best_a = a.copy()
    best_score = _score_fast(best_a)

    velocity = np.zeros(n)
    it = 0
    last_improve = it
    rng = np.random.RandomState(42)
    cur_patience = patience

    while time.monotonic() < deadline:
        it += 1

        c = _fft_convolve(a)
        if c.size == 0:
            break

        w = _topk_weights(c, K, T_local=0.4)
        g = _gradient(a, w)
        g = g - np.mean(g)

        velocity = beta * velocity + g
        v_norm = np.linalg.norm(velocity)
        if v_norm < 1e-15:
            break

        step = -eta * velocity / v_norm * np.sqrt(n)
        step = np.clip(step, -4.0, 4.0)

        a_new = a * np.exp(step)
        a_new = np.where(np.isfinite(a_new), a_new, 0.0)
        a = _normalize(a_new)

        if a.sum() < 1e-10:
            a = best_a.copy()
            velocity = np.zeros(n)
            continue

        if it % eval_every == 0:
            sc = _score_fast(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > cur_patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.008
                a = a * (1.0 + noise)
                a = _normalize(a)
                velocity = np.zeros(n)
                cur_patience = max(500, int(cur_patience * 0.7))

        if it % 2000 == 0:
            print(f"        [AS] iter={it}, score={best_score:.8f}", flush=True)

    return best_a, best_score


def _find_best_init(deadline):
    """Scan diverse initializations and return the best."""
    best_a = None
    best_score = np.inf
    rng = np.random.RandomState(123)

    # Use provided sequence if available
    if 'height_sequence_1' in globals() and height_sequence_1 is not None:
        try:
            a = np.array([float(max(0, min(1000, x))) for x in height_sequence_1])
            sc = _score_fast(a)
            if np.isfinite(sc):
                best_score = sc
                best_a = a.copy()
                print(f"    height_sequence_1: score={sc:.8f}", flush=True)
        except Exception:
            pass

    # Sinusoidal family
    for n in [1024, 2048]:
        if time.monotonic() > deadline:
            break
        for amp in [0.7, 0.8, 0.9, 1.0, 1.1, 1.2, 1.3]:
            for phase in [0.0, 0.25, 0.5, 0.75]:
                if time.monotonic() > deadline:
                    break
                a = _make_sine(n, amp, phase)
                sc = _score_fast(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

    # Multi-frequency
    for n in [1024, 2048]:
        for _ in range(5):
            if time.monotonic() > deadline:
                break
            comps = [
                (rng.uniform(0.6, 1.1), 1, rng.uniform(0, 1)),
                (rng.uniform(0.05, 0.25), rng.choice([2, 3, 4]), rng.uniform(0, 1)),
            ]
            a = _make_multi_freq(n, comps)
            sc = _score_fast(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()

    if best_a is None:
        best_a = _make_sine(1024, 0.85)
        best_score = _score_fast(best_a)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimization for low-peak nonnegative step function.
    
    Phase 1: Diverse initialization (30s)
    Phase 2: Momentum smooth optimization with multiple configs (bulk time)
    Phase 3: Active-set refinement
    Phase 4: Final polish pass
    Validation: Check with evaluate_sequence
    """
    t0 = time.monotonic()
    hard_deadline = t0 + float(budget_s) - 20.0

    np.random.seed(seed)
    print(f"[Start] Budget: {budget_s}s, deadline in {hard_deadline - t0:.0f}s", flush=True)

    # Phase 1: Initialization
    d1 = t0 + 40.0
    print("[Phase 1] Finding best initialization...", flush=True)
    init_a, init_score = _find_best_init(d1)
    print(f"  Best init: n={init_a.size}, score={init_score:.8f}", flush=True)

    best_a = init_a
    best_score = init_score

    # Phase 2: Main momentum optimization
    d2 = hard_deadline - 150.0
    remaining = d2 - time.monotonic()
    if remaining > 30:
        print(f"[Phase 2] Momentum optimization ({remaining:.0f}s budget)...", flush=True)

        # Multiple configurations to explore different landscapes
        configs = [
            (5.0, 0.15, 0.90, 2000),   # T, eta, beta, patience
            (8.0, 0.12, 0.88, 2500),
            (3.0, 0.18, 0.92, 1500),
            (12.0, 0.10, 0.85, 3000),
        ]

        time_per = remaining / len(configs)
        for ci, (T, eta, beta, pat) in enumerate(configs):
            # Allocate time: give each config its share, last gets remainder
            if ci < len(configs) - 1:
                alloc = t0 + 40.0 + (ci + 1) * time_per
            else:
                alloc = d2
            alloc = min(alloc, hard_deadline - 30.0)
            if time.monotonic() > alloc:
                break

            time_left = alloc - time.monotonic()
            print(f"  Config {ci+1}/{len(configs)}: T={T}, eta={eta}, beta={beta} ({time_left:.0f}s)", flush=True)

            a, s = _optimize_momentum(best_a, alloc, T=T, eta=eta, beta=beta,
                                      eval_every=4, patience=pat, verbose=False)
            if s < best_score:
                best_score = s
                best_a = a.copy()
                print(f"    -> Improved to {best_score:.8f}", flush=True)
            else:
                print(f"    -> No improvement (still {best_score:.8f})", flush=True)

        print(f"  Phase 2 best: {best_score:.8f}", flush=True)
    else:
        print("[Phase 2] Skipped (insufficient time)", flush=True)

    # Phase 3: Active-set refinement
    d3 = hard_deadline - 30.0
    remaining = d3 - time.monotonic()
    if remaining > 20:
        print(f"[Phase 3] Active-set refinement ({remaining:.0f}s)...", flush=True)
        for K in [6, 10, 16]:
            if time.monotonic() > d3:
                break
            alloc = d3
            a, s = _optimize_active(best_a, alloc, K=K)
            if s < best_score:
                best_score = s
                best_a = a.copy()
                print(f"  K={K}: Improved to {best_score:.8f}", flush=True)
    else:
        print("[Phase 3] Skipped", flush=True)

    # Phase 4: Final fine-tuning
    d4 = hard_deadline
    remaining = d4 - time.monotonic()
    if remaining > 20:
        print(f"[Phase 4] Final polish ({remaining:.0f}s)...", flush=True)
        a, s = _optimize_momentum(best_a, d4, T=2.0, eta=0.08, beta=0.90,
                                  eval_every=3, patience=1000, verbose=False)
        if s < best_score:
            best_score = s
            best_a = a.copy()
            print(f"  Final: {best_score:.8f}", flush=True)

    # Validation
    print("[Validation] Checking with evaluate_sequence...", flush=True)
    candidates = [best_a, init_a]
    # Also keep track of any intermediate results
    best_actual = np.inf
    best_out = None

    for arr in candidates:
        if time.monotonic() > t0 + float(budget_s) - 5.0:
            break
        arr_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        try:
            actual = float(evaluate_sequence(arr_list))
        except Exception:
            actual = np.inf
        if np.isfinite(actual) and actual < best_actual:
            best_actual = actual
            best_out = arr_list

    if best_out is None:
        best_out = [float(max(0.0, min(1000.0, x))) for x in init_a]
        best_actual = init_score

    total_time = time.monotonic() - t0
    print(f"[Done] Time: {total_time:.1f}s, Final score: {best_actual:.8f}", flush=True)
    return best_out
