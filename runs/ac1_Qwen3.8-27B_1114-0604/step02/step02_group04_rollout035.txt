Low-peak step-function optimization via adaptive mirror descent with decreasing temperature,
L-BFGS polishing, and diverse initialization.
"""

import numpy as np
import time
from scipy.optimize import minimize

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a, b=None):
    """Linear convolution of a and b (or a with itself if b is None) via FFT."""
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = b.size
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)
    return c[:n + m - 1]

def _score(a):
    """Compute the evaluation score."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)

def _grad_conv_obj(a, w):
    """
    Compute gradient of sum_k w[k]*c[k] w.r.t. a.
    g[j] = 2 * sum_m a[m] * w[j+m]
    Computed as 2 * (a * w_rev)[n-1:2n-1][::-1]
    """
    n = a.size
    w_rev = w[::-1]
    conv = _fft_convolve(a, w_rev)
    # conv has length 2n-1
    return 2.0 * conv[n-1:2*n-1][::-1]

def _make_sine(n, amp, phase=0.5):
    """Sinusoidal initialization."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.sin(theta)) / n
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0.01:
        a /= s
    return a

def _make_hann(n):
    """Hann window initialization."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    a = a / a.sum()
    return a

    """Gaussian bell initialization."""
    i = np.arange(n, dtype=np.float64)
    center = n / 2
    sigma = n / 4
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    a = a / a.sum()
    return a

    """Random initialization."""
    rng = np.random.RandomState(seed)
    a = rng.random(n)
    a = a / a.sum()
    return a

    """Uniform initialization."""
    return np.ones(n, dtype=np.float64) / n

    best_score = _score(best_a)
    it = 0

def _optimize_l_bfgs(a, deadline, max_iter=200):
    """
    L-BFGS optimization on the log-sum-exp objective.
    """
    a0 = np.asarray(a, dtype=np.float64).copy()
    a0 = np.clip(a0, 0, 1000)
    s0 = a0.sum()
    if s0 < 0.01:
        a0 = np.ones(a0.size, dtype=np.float64) / a0.size
        s0 = 1.0
    a0 = a0 / s0
    n = a0.size

    T = 5.0  # Temperature for log-sum-exp

    def objective(x):
        x = np.clip(x, 0, 1000)
        s = x.sum()
        if s < 0.01:
            return 1e10
        x = x / s
        c = _fft_convolve(x)
        # Log-sum-exp
        c_shift = c - c.max()
        lse = c.max() + T * np.log(np.sum(np.exp(c_shift / T)))
        return lse

    def gradient(x):
        x = np.clip(x, 0, 1000)
        s = x.sum()
        if s < 0.01:
            return np.ones_like(x)
        x = x / s
        c = _fft_convolve(x)
        c_shift = c - c.max()
        exp_c = np.exp(c_shift / T)
        w = exp_c / exp_c.sum()
        g = _grad_conv_obj(x, w)
        # For the log-sum-exp, the gradient w.r.t. x is:
        # d/dx log(sum_k exp(c[k]/T)) = (1/T) * sum_k (exp(c[k]/T)/sum_l exp(c[l]/T)) * dc[k]/dx
        # = (1/T) * sum_k w[k] * dc[k]/dx
        # And dc[k]/dx[j] gives 2 * sum_m x[m] * delta[k, j+m]... 
        # Actually the gradient I computed is for sum_k w[k]*c[k], so for log-sum-exp it's g/T
        g_lse = g / T
        # But we also need to account for the normalization x = x/s... 
        # Actually, since the objective is scale-invariant, we can just use the 
        # gradient of the unnormalized objective and project.
        # For simplicity, let's just use g directly (the constant factor doesn't matter for direction)
        return g_lse

    try:
        result = minimize(
            objective, a0,
            method='L-BFGS-B',
            jac=gradient,
            options={'maxiter': max_iter, 'maxfun': max_iter * 2, 
                     'ftol': 1e-10, 'gtol': 1e-8,
                     'maxls': 50}
        )
        a_opt = np.clip(result.x, 0, 1000)
        s_opt = a_opt.sum()
        if s_opt > 0.01:
            a_opt = a_opt / s_opt
        return a_opt, _score(a_opt)
    except Exception as e:
        print(f"  L-BFGS error: {e}", flush=True)
        return a0, _score(a0)

Low-peak step-function optimization via decreasing-temperature mirror descent,
L-BFGS polishing, and hard-max top-K refinement with diverse initialization.

Key improvements over previous approach:
1. DECREASING temperature (start broad, end focused) for better peak reduction

import numpy as np
import time
from scipy.optimize import minimize


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a, b=None):
    """Linear convolution of a and b (auto-convolution if b is None) via FFT."""
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = b.size
    N = _next_pow2(n + m - 1)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)
    return c[:n + m - 1]


def _score(a):
    """Compute the evaluation score (scale-invariant)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _grad_conv_obj(a, w):
    """
    Compute gradient of sum_k w[k]*c[k] w.r.t. a, where c = a*a (linear conv).
    g[j] = 2 * sum_{m=0}^{n-1} a[m] * w[j+m]
    Efficiently computed via FFT: g = 2 * (a * w_rev)[n-1:2n-1][::-1]
    """
    n = a.size
    w_rev = w[::-1]
    conv = _fft_convolve(a, w_rev)
    # conv has length 2n-1, we take indices n-1 through 2n-2 and reverse
    return 2.0 * conv[n - 1:2 * n - 1][::-1]


def _make_sine(n, amp, phase=0.5):
    """Sinusoidal (anti-palindromic) initialization."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.sin(theta)) / n
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0.01:
        a /= s
    return a


def _make_multi_sine(n, n_modes=3, seed=0):
    """Multi-frequency sinusoidal initialization."""
    rng = np.random.RandomState(seed)
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64) / n
    for k in range(1, n_modes + 1):
        amp = rng.uniform(0.1, 0.5) / k
        phase = rng.uniform(0, 1.0)
        theta = 2.0 * np.pi * k * (i + phase) / n
        a += amp * np.sin(theta) / n
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s > 0.01:
        a /= s
    return a


def _make_hann(n):
    """Hann window initialization."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / (n - 1)))
    s = a.sum()
    if s > 0.01:
        a /= s
    return a

    """Gaussian bell initialization."""
    i = np.arange(n, dtype=np.float64)
    center = n / 2.0
    sigma = n / 4.0
    a = np.exp(-0.5 * ((i - center) / sigma) ** 2)
    s = a.sum()
    if s > 0.01:

    return a

    rng = np.random.RandomState(seed)
    a = rng.random(n) + 0.01

        else:
            # Phase B: focused exploitation (T from T_start*0.2 to T_end)
            local_t = (t_frac - 0.5) / 0.5
            T = T_start * 0.2 + (T_end - T_start * 0.2) * local_t

        # Adaptive learning rate
        lr = lr0 / (1.0 + 0.0005 * it)

        # Compute convolution
        c = _fft_convolve(a)

        # Softmax weights (numerically stable)
        c_max = c.max()
        exp_c = np.exp((c - c_max) / T)
        w_sum = exp_c.sum()
        if w_sum < 1e-300:
            # Fallback: uniform weights on max
            w = np.zeros(c.size)
            w[np.argmax(c)] = 1.0
        else:
            w = exp_c / w_sum

        # Gradient via FFT
        g = _grad_conv_obj(a, w)

        # Project to simplex tangent space
        g_proj = g - g.mean()

        # Mirror descent update (multiplicative weights)
        update = -lr * g_proj
        update = np.clip(update, -5.0, 5.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        # Periodic score evaluation
        if it % score_interval == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
                recent_improvements.append(sc)
                # Slightly increase lr on improvement
                lr0 = min(0.3, lr0 * 1.02)
            elif it - last_improve > patience:
                # Perturbation-based restart
                a = best_a.copy()
                # Mix with random perturbation
                alpha = min(0.3, 0.05 + 0.02 * (patience // 8000))
                noise = rng.standard_normal(n)
                a = (1 - alpha) * a + alpha * (noise - noise.min()) / (n * 0.5)
                a = np.clip(a, 0, 1000)
                s = a.sum()
                if s > 0.01:
                    a /= s
                patience = max(5000, int(patience * 1.15))
                lr0 = max(0.02, lr0 * 0.7)
                # Increase aggressiveness
                T_start = max(10.0, T_start * 0.7)

    return best_a, best_score


def _optimize_l_bfgs(a, deadline, max_iter=500):
    """
    L-BFGS optimization on the smooth log-sum-exp surrogate.
    Uses second-order information for faster local convergence.
    """
    a0 = np.asarray(a, dtype=np.float64).copy()
    a0 = np.clip(a0, 0, 1000)
    s0 = a0.sum()
    if s0 < 0.01:
        a0 = np.ones(a0.size, dtype=np.float64) / a0.size
        s0 = 1.0
    a0 = a0 / s0
    n = a0.size

    # Start with moderate temperature, decrease during optimization
    T_values = [20.0, 10.0, 5.0, 2.0, 1.0]
    iters_per_T = max(50, max_iter // len(T_values))

    current_a = a0.copy()
    best_score = _score(current_a)
    best_a = current_a.copy()

    for T in T_values:
        if time.monotonic() > deadline:
            break

        def objective(x):
            x = np.clip(x, 0, 1000)
            s = x.sum()
            if s < 0.01:
                return 1e10
            x = x / s
            c = _fft_convolve(x)
            c_shift = c - c.max()
            lse = c.max() + T * np.log(np.sum(np.exp(c_shift / T)) + 1e-300)
            return lse

        def gradient(x):
            x = np.clip(x, 0, 1000)
            s = x.sum()
            if s < 0.01:
                return np.zeros_like(x)
            x = x / s
            c = _fft_convolve(x)
            c_shift = c - c.max()
            exp_c = np.exp(c_shift / T)
            w = exp_c / (exp_c.sum() + 1e-300)
            g = _grad_conv_obj(x, w) / T
            # Project to simplex tangent
            g = g - g.mean()
            return g

        try:
            result = minimize(
                objective, current_a,
                method='L-BFGS-B',
                jac=gradient,
                bounds=[(0, 1000)] * n,
                options={'maxiter': iters_per_T,
                         'maxfun': iters_per_T * 3,
                         'ftol': 1e-12,
                         'gtol': 1e-8,
                         'maxls': 30,
                         'iprint': -1}
            )
            current_a = np.clip(result.x, 0, 1000)
            s = current_a.sum()
            if s > 0.01:
                current_a = current_a / s
                sc = _score(current_a)
                if sc < best_score:
                    best_score = sc
                    best_a = current_a.copy()
        except Exception:
            pass

    return best_a, best_score


def _optimize_hard(a, deadline, K=8):
    """
    Hard-max top-K active set refinement for final polish.
    Focuses exclusively on reducing the K largest convolution peaks.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size, dtype=np.float64) / a.size
        s = 1.0
    a = a / s

    n = a.size
    best_a = a.copy()
    best_score = _score(best_a)
    it = 0
    last_improve = 0
    patience = 5000
    rng = np.random.RandomState(99)
    lr0 = 0.08

    while time.monotonic() < deadline:
        it += 1
        lr = lr0 / (1.0 + 0.003 * it)

        c = _fft_convolve(a)

        # Select top-K peaks with adaptive K
        effective_K = min(K + it // 10000, 2 * K)
        if c.size <= effective_K:
            top_idx = np.arange(c.size)
        else:
            top_idx = np.argpartition(c, -effective_K)[-effective_K:]

        # Soft weighting within top peaks
        top_vals = c[top_idx]
        T_local = 1.0 + 0.5 * (it / 10000.0)
        weights = np.exp(top_vals / T_local - top_vals.max() / T_local)
        weights /= (weights.sum() + 1e-300)

        w = np.zeros(c.size, dtype=np.float64)
        for i, k in enumerate(top_idx):
            w[k] = weights[i]

        g = _grad_conv_obj(a, w)
        g_proj = g - g.mean()

        a = a * np.exp(-lr * g_proj)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        if it % 10 == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                a = best_a.copy()
                alpha = 0.02 + 0.01 * (patience // 5000)
                noise = rng.standard_normal(n)
                a = (1 - alpha) * a + alpha * (noise - noise.min()) / (n * 0.5)
                a = np.clip(a, 0, 1000)
                a = a / (a.sum() + 1e-300)
                patience = max(3000, int(patience * 1.2))
                lr0 = max(0.01, lr0 * 0.75)

    return best_a, best_score


def _optimize_direct_peaks(a, deadline):
    """
    Direct peak-reduction: iteratively identify the single worst peak,
    compute which elements contribute most, and reduce them.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size, dtype=np.float64) / a.size
        s = 1.0
    a = a / s

    n = a.size
    best_a = a.copy()
    best_score = _score(best_a)
    it = 0
    last_improve = 0
    patience = 3000
    rng = np.random.RandomState(55)

    while time.monotonic() < deadline:
        it += 1
        c = _fft_convolve(a)
        peak_idx = int(np.argmax(c))

        # The peak at index peak_idx means:
        # c[peak_idx] = sum_i a[i] * a[peak_idx - i]
        # The elements contributing are a[i] and a[peak_idx - i] for valid i
        i_min = max(0, peak_idx - n + 1)
        i_max = min(n - 1, peak_idx)

        if i_max < i_min:
            break

        # Compute contribution of each pair
        contributions = np.zeros(n)
        for i in range(i_min, i_max + 1):
            j = peak_idx - i
            if 0 <= j < n:
                contributions[i] += a[j]
                if i != j:
                    contributions[j] += a[i]

        # Reduce high-contributing elements, increase low-contributing ones
        lr = 0.002 * max(1.0, peak_idx / n)
        contrib_proj = contributions - contributions.mean()
        a = a * np.exp(-lr * contrib_proj)
        a = np.clip(a, 0, 1000)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        if it % 100 == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.01 * a.mean()
                a = np.clip(a + noise, 0, 1000)
                a = a / a.sum()
                patience = int(patience * 1.3)

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for low-peak nonnegative step function.

    Phase 1: Diverse initialization (1 min)
    Phase 2: Decreasing-temperature mirror descent (22 min)
    Phase 3: L-BFGS second-order polishing (6 min)
    Phase 4: Hard-max top-K refinement (5 min)
    Phase 5: Validation and return (buffer)
    """
    t0 = time.monotonic()
    hard_deadline = t0 + float(budget_s) - 20.0

    np.random.seed(seed)
    print(f"[Start] Budget: {budget_s}s, effective deadline: {hard_deadline - t0:.0f}s", flush=True)

    # === Phase 1: Diverse initialization ===
    d1 = t0 + 60.0
    print("\n[Phase 1] Generating diverse initializations...", flush=True)

    n = 1024
    inits = []

    # Try to load previous best
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        prev = np.clip(prev, 0, 1000)
        s = prev.sum()
        if s > 0.01:
            prev = prev / s
            sc = _score(prev)
            inits.append(("prev_best", prev, sc))
            print(f"  prev_best: n={prev.size}, score={sc:.8f}", flush=True)
    except (NameError, Exception) as e:
        print(f"  (no previous best available: {e})", flush=True)

    # Generate diverse initializations
    init_configs = [
        ("sine_04", _make_sine(n, 0.4)),
        ("sine_06", _make_sine(n, 0.6)),
        ("sine_075", _make_sine(n, 0.75)),
        ("sine_085", _make_sine(n, 0.85)),
        ("sine_10", _make_sine(n, 1.0)),
        ("sine_12", _make_sine(n, 1.2, 0.3)),
        ("multi1", _make_multi_sine(n, 3, seed=1)),
        ("multi2", _make_multi_sine(n, 5, seed=2)),
        ("hann", _make_hann(n)),
        ("bell", _make_bell(n)),
        ("ramp", _make_ramp(n)),
        ("rand_0", _make_random(n, 0)),
        ("rand_1", _make_random(n, 1)),
        ("rand_2", _make_random(n, 42)),
        ("flat", _make_flat(n)),
    ]

    for name, a in init_configs:
        if time.monotonic() > d1:
            break
        sc = _score(a)
        inits.append((name, a, sc))

    inits.sort(key=lambda x: x[2])
    print(f"  Top 5 inits:", flush=True)
    for name, a, sc in inits[:5]:
        print(f"    {name}: {sc:.8f}", flush=True)

    # === Phase 2: Mirror descent with decreasing temperature ===
    d2_end = t0 + (budget_s - 200) * 0.65
    print(f"\n[Phase 2] Mirror descent (t={time.monotonic()-t0:.0f}s to t={d2_end-t0:.0f}s)...", flush=True)

    results = list(inits)  # Include inits as candidates

    # Optimize top 3 initializations
    top_k = min(3, len(inits))
    for idx in range(top_k):
        if time.monotonic() > d2_end:
            break
        name, a_init, sc_init = inits[idx]
        remaining = d2_end - time.monotonic()
        n_left = top_k - idx
        alloc = remaining / n_left

        if alloc < 60:
            print(f"  Skipping {name} (only {alloc:.0f}s left)", flush=True)
            break

        print(f"  Optimizing '{name}' (init_score={sc_init:.8f}, budget={alloc:.0f}s)...", flush=True)

        # Use different hyperparameters for each
        if idx == 0:
            params = dict(T_start=100.0, T_end=0.3, lr0=0.15)
        elif idx == 1:
            params = dict(T_start=60.0, T_end=0.5, lr0=0.1)
        else:
            params = dict(T_start=40.0, T_end=1.0, lr0=0.2)

        a_opt, sc_opt = _optimize_mirror(a_init, time.monotonic() + alloc, **params, name=name)
        print(f"    -> score={sc_opt:.8f} (improved: {sc_init - sc_opt:.8f})", flush=True)
        results.append((f"mirror_{name}", a_opt, sc_opt))

    # Sort results
    results.sort(key=lambda x: x[2])
    print(f"  Best after mirror: {results[0][0]} = {results[0][2]:.8f}", flush=True)

    # === Phase 3: L-BFGS polishing ===
    d3_end = d2_end + (budget_s - 200) * 0.25
    print(f"\n[Phase 3] L-BFGS polishing (t={time.monotonic()-t0:.0f}s to t={d3_end-t0:.0f}s)...", flush=True)

    # Polish the top 2 results
    for idx in range(min(2, len(results))):
        if time.monotonic() > d3_end:
            break
        name, a_cur, sc_cur = results[idx]
        remain = d3_end - time.monotonic()
        if remain < 30:
            break
        allocated = remain / (min(2, len(results)) - idx)
        print(f"  L-BFGS on '{name}' (score={sc_cur:.8f}, budget={allocated:.0f}s)...", flush=True)
        a_lb, sc_lb = _optimize_l_bfgs(a_cur, time.monotonic() + allocated, max_iter=300)
        print(f"    -> score={sc_lb:.8f}", flush=True)
        results.append((f"lbf_{name}", a_lb, sc_lb))

    results.sort(key=lambda x: x[2])
    print(f"  Best after L-BFGS: {results[0][0]} = {results[0][2]:.8f}", flush=True)

    d4_end = hard_deadline - 10.0

    if time.monotonic() < d4_end - 30:
        name, a_cur, sc_cur = results[0]
        remain = d4_end - time.monotonic() - 30
        allocated = min(remain, 300)

        results.append(("hard_best", a_h, sc_h))

    # Also try direct peak reduction on the best
    if time.monotonic() < d4_end - 10:
        remain = d4_end - time.monotonic() - 5
        if remain > 20:
            name, a_cur, sc_cur = results[0]
            print(f"  Direct peak reduction on '{name}' (budget={remain:.0f}s)...", flush=True)
            a_dp, sc_dp = _optimize_direct_peaks(a_cur, time.monotonic() + remain)
            results.append(("direct_peak", a_dp, sc_dp))

    # === Phase 5: Validation ===
    print(f"\n[Phase 5] Validation (t={time.monotonic()-t0:.0f}s)...", flush=True)

    results.sort(key=lambda x: x[2])

    best_out = None
    best_actual = np.inf

    for name, arr, sc in results[:8]:

            break
        arr_clipped = np.clip(arr, 0, 1000)
        if arr_clipped.sum() < 0.01:
            continue
        arr_list = [float(x) for x in arr_clipped]
        try:
            actual = float(evaluate_sequence(arr_list))
        except Exception:
            actual = np.inf
