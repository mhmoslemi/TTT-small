"""
Step function optimization via LSE-smoothed L-BFGS + LP-inspired multiplicative equalization.

Key ideas:
  1. LSE-smoothed max-convolution objective with continuation in beta.
  2. LP-dual-inspired multiplicative update: at optimum, conv(lambda, a) must be constant.
     We iteratively equalize this quantity using a damped multiplicative update.
  3. Multi-length exploration with diverse initializations.
  4. Final polish with high-beta L-BFGS and subgradient method.
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and its gradient.
    Returns (score, grad).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient of L w.r.t. a: grad_L[j] = 2 * sum_m p[m] * a[m+j] (cross-correlation)
    # = 2 * conv(a_rev, p)[n-1 : 2n-1]
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, deadline=None):
    """
    Run L-BFGS-B to minimize the LSE-smoothed score.
    Returns (exact_score, result_array).
    """
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n

    def obj_wrapper(x, b):
        return lse_score_grad(x, b)

    options = {'ftol': 1e-15, 'gtol': 1e-13, 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    if deadline is not None:
        remaining = deadline - time.monotonic()
        if remaining < 2.0:
            sc, _ = lse_score_grad(a0, beta)
            return compute_score(a0), a0
        # Estimate maxfun from remaining time
        options['maxfun'] = min(maxfun, max(100, int(remaining * 200)))
        options['maxiter'] = min(maxiter, max(100, int(remaining * 50)))

    try:
        res = minimize(
            obj_wrapper, a0, args=(beta,),
            method='L-BFGS-B', jac=True, bounds=bounds,
            options=options
        )
        a_out = np.clip(res.x, 0.0, 1000.0)
    except Exception:
        a_out = a0
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def build_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    if parent is not None:
        # Resample parent to length n
        p = parent.astype(np.float64)
        if len(p) != n:
            p = np.interp(np.linspace(0, len(p)-1, n), np.arange(len(p)), p)
        inits.append(np.clip(p, 0.01, 1000.0).copy())
    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Cosine modulations
    x = np.arange(n, dtype=np.float64) / n
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x))
    inits.append(0.5 + 0.2 * np.cos(6.0 * np.pi * x) * np.cos(2.0 * np.pi * x))
    inits.append(0.3 + 0.4 * np.sin(3.0 * np.pi * x) ** 2)
    # Quadratic
    inits.append(0.1 + 0.8 * (2.0 * x - 1.0) ** 2)
    # Random
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.4 * rng.standard_normal(n), 0.01, 1000.0))
    # Block structure
    for nb in (4, 8):
        per = max(1, n // nb)
        vals = 0.2 + 0.6 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())
    # Gapped
    a = np.clip(0.5 + 0.3 * rng.standard_normal(n), 0.01, 1000.0)
    a[1::2] = 0.1
    inits.append(a.copy())
    return inits


def multiplicative_equalize(a, n_iters=100, alpha=0.3, deadline=None):
    """
    LP-dual-inspired multiplicative update.
    At optimum, h[j] = 2*sum_k lambda[k]*a[k-j] should be constant.
    This function iteratively equalizes h using a damped multiplicative update.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    best_a = a.copy()
    best_sc = compute_score(a)

    # Precompute FFT size for cross-correlation
    M_fft = 1 << (3 * n - 2).bit_length()

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        M = float(c.max())
        if M < 1e-15:
            break
        # Active lags (those close to the max)
        threshold = M * 0.99
        lambda_v = (c >= threshold).astype(np.float64)
        if lambda_v.sum() == 0:
            lambda_v[int(np.argmax(c))] = 1.0

        # Compute h[j] = 2 * sum_{m=0}^{n-1} lambda[j+m] * a[m]
        # This is 2 * (lambda * a) cross-correlation at lags 0..n-1
        L = np.fft.rfft(lambda_v, M_fft)
        A = np.fft.rfft(a, M_fft)
        cross = np.fft.irfft(L * np.conj(A), M_fft)
        h = 2.0 * cross[:n]

        # Ensure h is positive (it should be since a >= 0 and lambda >= 0)
        h = np.maximum(h, 1e-15)
        h_mean = h.mean()

        # Damped multiplicative update: a[j] <- a[j] * (1 - alpha + alpha * h_mean/h[j])
        factor = 1.0 - alpha + alpha * (h_mean / h)
        a = a * factor
        a = np.clip(a, 0.0, 1000.0)

        # Renormalize to preserve sum
        S_new = a.sum()
        if S_new < 1e-12:
            a[:] = 0.5
            break
        a = a * (S / S_new)

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_sc, best_a


def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements that contribute most to the peak convolution lag."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(3, n // 50)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:-top_count] if n > 2 * top_count else np.argsort(contrib)[:top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)


def exact_subgrad_polish(a, total_iters=500, deadline=None):
    """Projected subgradient descent on the exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0

    for it in range(total_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            sg_M[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.5)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        step_decay = 1.0 / (1.0 + 0.01 * it)
        a -= step_decay * 0.001 * grad / gn
        a = np.clip(a, 0.0, 1000.0)

    return best_sc, best_a
