def _normalize(v):

    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())

def _fft_size(m):

def _autocorr(f):

    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]

def _transfer_max(c, f, i, j, dv, delta, total):

        if not donors or not receivers:
            no_improve += 1
            if no_improve >= 7:
                break
            continue

        best_mc = maxc
        best_move = None

        for i in donors:
            fi = float(f[i])
            if fi <= 1e-12:
                continue
            for j in receivers:
                if i == j:
                    continue
                # Binary search for best transfer amount
                lo_frac = 0.005
                hi_frac = min(0.9, fi / (fi + float(f[j]) + 1e-300))
                best_local_mc = best_mc
                best_local_frac = None
                # Try a few fractions
                cand_fracs = [lo_frac, 0.02, 0.05, 0.1, 0.2, 0.35, 0.5, hi_frac * 0.8]
                cand_fracs = [x for x in cand_fracs if 0 < x < 0.95]
                if not cand_fracs:
                    cand_fracs = [0.1]
                for frac in cand_fracs:
                    dv = fi * frac
                    if dv <= 1e-15:
                        continue
                    mc = _transfer_max(c, f, i, j, dv, delta, total)
                    if mc < best_local_mc:
                        best_local_mc = mc
                        best_local_frac = frac
                if best_local_frac is not None and best_local_mc < best_mc:
                    # Refine with binary search around best_local_frac
                    lo = max(0.001, best_local_frac * 0.5)
                    hi = min(0.95, best_local_frac * 1.5 + 0.01)
                    for _ in range(4):
                        mid = (lo + hi) / 2.0
                        dv = fi * mid
                        mc_mid = _transfer_max(c, f, i, j, dv, delta, total)
                        dv_lo = fi * lo
                        mc_lo = _transfer_max(c, f, i, j, dv_lo, delta, total)
                        if mc_mid < mc_lo:
                            hi = mid
                        else:
                            lo = mid
                    best_frac = (lo + hi) / 2.0
                    dv = fi * best_frac
                    mc_final = _transfer_max(c, f, i, j, dv, delta, total)
                    if mc_final < best_mc:
                        best_mc = mc_final
                        best_move = (i, j, dv)

        # Random fallback
        if best_move is None or best_mc >= maxc - 1e-18:
            for _ in range(40):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j or f[i] <= 1e-12:
                    continue
                frac = float(rng.uniform(0.01, 0.5))
                dv = float(f[i]) * frac
                mc = _transfer_max(c, f, i, j, dv, delta, total)
                if mc < best_mc:
                    best_mc = mc
                    best_move = (i, j, dv)

        if best_move is None or best_mc >= maxc - 1e-18:
            no_improve += 1
            continue

        i, j, dv = best_move
        # Apply transfer and update c
        f[i] -= dv
        f[j] += dv
        _compute_delta(c, f, i, j, dv, delta)  # This recomputes but with modified f
        # Actually we need the delta computed with the ORIGINAL f
        # Let me recompute properly
        # The delta was already computed in _transfer_max into 'total'
        # But we overwrote it. Let's just recompute c from scratch periodically.
        # For now, use the incrementally updated approach:
        # We stored the result in total during the last _transfer_max call
        # But we made more calls after that. Let's just recompute.
        
        # Simpler: just recompute c every few moves
        move_count += 1
        if move_count % 10 == 0:
            f = _normalize(f)
            c = _autocorr(f)
            maxc = float(np.max(c))
        else:
            # Recompute c from scratch (safe but slightly slower)
            c = _autocorr(f)
            maxc = float(np.max(c))
        
        no_improve = 0
        cur_score = float(2.0 * n * maxc)
        if cur_score < best_score:
            best_score = cur_score
            best_f = f.copy()

    f = _normalize(f)
    c = _autocorr(f)
    mc = float(c.max()) if c.size else 0.0
    sc = float(2.0 * n * mc)
    if sc <= best_score:
        return f, sc, c
    bf = _normalize(best_f)
    bc = _autocorr(bf)
    bmc = float(bc.max()) if bc.size else 0.0
    return bf, float(2.0 * n * bmc), bc


def _perturb(best_f, rng):
    n = best_f.size
    r = float(rng.random())
    if r < 0.4:
        sigma = float(rng.uniform(0.03, 0.8))
        z = rng.standard_normal(n)
        f = best_f * np.exp(sigma * z)
    elif r < 0.65:
        alpha = float(rng.uniform(0.05, 0.4))
        noise = rng.random(n)
        s = noise.sum()
        if s >= 1e-300:
            noise /= s
        f = (1.0 - alpha) * best_f + alpha * noise
    elif r < 0.8:
        win = max(3, n // 300)
        f = _smooth_with_window(best_f, win)
        sigma = float(rng.uniform(0.05, 0.5))
        f = f * np.exp(sigma * rng.standard_normal(n))
    elif r < 0.9:
        # Flip a random segment
        flen = min(n // 4, int(rng.integers(10, max(11, n // 8))))
        fstart = int(rng.integers(0, max(1, n - flen)))
        f = best_f.copy()
        f[fstart:fstart + flen] = best_f[fstart:fstart + flen][::-1]
        f = f * np.exp(0.1 * rng.standard_normal(n))
    else:
        # Symmetrize + noise
        f = _symmetrize(best_f)
        sigma = float(rng.uniform(0.02, 0.3))
        f = f * np.exp(sigma * rng.standard_normal(n))
    return _normalize(f)


def _gradient_optimize(f0, budget_s, rng, tag=""):
    f0 = _normalize(f0)
    n = f0.size
    if n == 0:
        return f0, np.inf
    if budget_s <= 0.0:
        c = _autocorr(f0)
        return f0, float(2.0 * n * c.max())

    f = f0.copy()
    c = _autocorr(f)
    maxc = float(c.max())
    best_f = f.copy()
    best_score = float(2.0 * n * maxc)

    deadline = time.monotonic() + float(budget_s)
    sigma_rel = 0.06
    min_sigma_rel = 5e-11
    no_accept = 0
    it = 0
    last_coord = time.monotonic()

    if n < 2000:
        coord_budget = 0.3
        coord_interval = 0.4
    elif n < 10000:
        coord_budget = 0.5
        coord_interval = 0.5
    else:
        coord_budget = 0.7
        coord_interval = 0.7

    print(f"[{tag}] start score={best_score:.6f} n={n}")
    t0 = time.monotonic()

    while time.monotonic() < deadline:
        it += 1
        if maxc <= 0.0:
            break

        # Adaptive temperature: look at the gap between top lags
        sorted_c = np.sort(c)[::-1]
        top1 = sorted_c[0]
        top5 = sorted_c[min(4, len(sorted_c)-1)]
        gap = top1 - top5
        # If gap is small, use lower temperature (more concentrated on peak)
        if gap < 1e-300:
            T = maxc * 0.05
        else:
            T = maxc * (0.02 + 0.08 * (gap / maxc))
        T = max(T, 1e-300)

        # Softmax weights
        z = (c - maxc) / T
        np.clip(z, -45.0, 0.0, out=z)
        p = np.exp(z)
        ps = float(p.sum())
        if ps < 1e-300:
            p = np.full(c.size, 1.0 / c.size)
        else:
            p /= ps

        g = _weighted_gradient_fft(p, f)
        gm = g - float(g.mean())
        scale = float(gm.max() - gm.min())

        accepted = False
        if scale > 1e-300:
            best_mc = maxc
            best_ft = None
            best_ct = None

            if no_accept > 40:
                exps = (0.003, 0.01, 0.05, 0.2, 0.5)
            elif no_accept > 15:
                exps = (0.01, 0.05, 0.2, 1.0, 3.0)
            else:
                exps = (0.02, 0.1, 0.5, 2.0, 5.0)

            for e in exps:
                # Log-domain mirror step
                arg = -e * (gm / scale)
                np.clip(arg, -30.0, 30.0, out=arg)
                ft = f * np.exp(arg)
                ft = _normalize(ft)
                ct = _autocorr(ft)
                mc = float(ct.max())
                if mc < best_mc:
                    best_mc = mc
                    best_ft = ft
                    best_ct = ct

                # Projected Euclidean step
                if e in (0.1, 0.5, 2.0):
                    v = f - e * (gm / scale)
                    ft2 = _project_simplex(v)
                    ft2 = _normalize(ft2)
                    ct2 = _autocorr(ft2)
                    mc2 = float(ct2.max())
                    if mc2 < best_mc:
                        best_mc = mc2
                        best_ft = ft2
                        best_ct = ct2

            if best_ct is not None and best_mc < maxc - 1e-16:
                f = best_ft
                c = best_ct
                maxc = best_mc
                accepted = True

        if accepted:
            no_accept = 0
            cur_score = float(2.0 * n * maxc)
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()
                if best_score < TARGET_SCORE:
                    print(f"[{tag}] new best below target: {best_score:.6f} at iter={it}")
            sigma_rel = max(min_sigma_rel, sigma_rel * 0.985)
        else:
            no_accept += 1
            now = time.monotonic()
            if now - last_coord > coord_interval or no_accept > 25:
                fp, sc, cp = _coordinate_polish(f, c, coord_budget, rng)
                if sc < float(2.0 * n * maxc) - 1e-12:
                    f = _normalize(fp)
                    c = cp
                    maxc = float(c.max())
                    accepted = True
                    no_accept = 0
                    if sc < best_score:
                        best_score = sc
                        best_f = f.copy()
                        if best_score < TARGET_SCORE:
                            print(f"[{tag}] coord improved below target: {best_score:.6f}")
                else:
                    if no_accept > 60 or (it % 8000 == 0):
                        f = _perturb(best_f, rng)
                        c = _autocorr(f)
                        maxc = float(c.max())
                        sigma_rel = min(0.2, sigma_rel * 2.5 + 1e-4)
                        no_accept = 0
                last_coord = time.monotonic()
            else:
                sigma_rel = max(min_sigma_rel, sigma_rel * 0.995)

        if it % 30000 == 0:
            cur_score = float(2.0 * n * maxc)
            elapsed = time.monotonic() - t0
            print(f"[{tag}] iter={it} best={best_score:.6f} cur={cur_score:.6f} elapsed={elapsed:.0f}s")

    cur_score = float(2.0 * n * float(c.max()))
    if cur_score < best_score:
        best_score = cur_score
        best_f = f.copy()

    # Try symmetrization
    sf = _symmetrize(best_f)
    sc = _fast_score(sf)
    if sc < best_score:
        best_score = sc
        best_f = sf

    elapsed = time.monotonic() - t0
    print(f"[{tag}] end best={best_score:.6f} (elapsed {elapsed:.0f}s)")
    return _normalize(best_f), float(best_score)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative coefficient sequence minimizing evaluate_sequence.
    Returns a Python list of floats.
    Uses hybrid gradient/coordinate descent with multi-scale and diverse restarts.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total_deadline = t_start + max(120.0, float(budget_s)) - 30.0

    # Load provided construction
    try:
        f0 = np.asarray(height_sequence_1, dtype=np.float64)
    except Exception:
        f0 = np.ones(6121, dtype=np.float64)

    if f0.size == 0:
        f0 = np.ones(6121, dtype=np.float64)
    f0 = np.nan_to_num(f0, nan=0.0, posinf=1000.0, neginf=0.0)
    np.clip(f0, 0.0, 1000.0, out=f0)
    n = int(f0.size)
    if n < 3:
        f0 = np.ones(6121, dtype=np.float64)
        n = f0.size
    f0 = _normalize(f0)

    best_f = f0.copy()
    best_c = _autocorr(best_f)
    best_score = float(2.0 * n * best_c.max()) if best_c.size else np.inf

    print(f"Initial score: {best_score:.6f} (n={n})")

    # Build seed pool
    seeds = []

    def _add_seed(name, f):
        f = _interp_to_length(f, n)
        try:
            sc = _fast_score(f)
        except Exception:
            return
        if np.isfinite(sc):
            seeds.append((float(sc), len(seeds), name, f))

    _add_seed("given", f0)
    _add_seed("reverse", f0[::-1])
    _add_seed("symmetric", _symmetrize(f0))
    _add_seed("uniform", np.ones(n, dtype=np.float64))

    for win in (5, 11, 31, 101, 301):
        _add_seed(f"smooth{win}", _smooth_with_window(f0, win))

    for gamma in (0.4, 0.7, 1.3, 1.8):
        base = f0 + 1e-12
        _add_seed(f"gamma{gamma}", np.power(base, gamma))

    for i in range(8):
        _add_seed(f"fourier{i}", _fourier_seed(n, rng))
    for i in range(6):
        _add_seed(f"gamma_seed{i}", _gamma_seed(n, rng))

    # Two-bump seeds with various parameters
    for gap in (0.1, 0.2, 0.3, 0.4):
        for width in (0.15, 0.3, 0.5):
            _add_seed(f"2bump_g{gap}_w{width}", _two_bump_seed(n, gap, width, rng))

    # Periodic seeds
    for ph in (3, 5, 7, 11):
        for amp in (0.3, 0.6, 1.0):
            _add_seed(f"per_{ph}_{amp}", _periodic_seed(n, ph, amp, rng))

    # Coarse-to-fine
    for K in (128, 256, 512, 1024):
        if K < n:
            coarse = _downsample_to(f0, K)
            _add_seed(f"coarse{K}", _interp_to_length(coarse, n))

    if not seeds:
        f = np.ones(max(1, n), dtype=np.float64)
        seeds.append((2.0, 0, "fallback", _normalize(f)))

    seeds.sort()
    print(f"Seed pool size: {len(seeds)}, best seed score: {seeds[0][0]:.6f}")

    # Phase 1: Coarse optimization (if time allows)
    now = time.monotonic()
    if total_deadline - now > 1800.0 and n > 512:
        coarse_time = 25.0
        for K in (256, 512):
            if total_deadline - time.monotonic() < coarse_time + 120.0:
                break
            source = _downsample_to(best_f, K)
            cf, cs = _gradient_optimize(source, coarse_time, rng, f"coarse{K}")
            fin = _interp_to_length(cf, n)
            _add_seed(f"coarse_lift_{K}", fin)
        seeds.sort()

    # Phase 2: Main optimization on top seeds
    now = time.monotonic()
    remaining = total_deadline - now
    final_reserve = min(400.0, max(80.0, 0.18 * remaining))
    main_deadline = total_deadline - final_reserve
    main_budget = max(1.0, main_deadline - time.monotonic())

    top = seeds[:5] if len(seeds) >= 5 else seeds
    weights = [0.45, 0.22, 0.15, 0.11, 0.07][:len(top)]

    print(f"\n=== Main optimization budget={main_budget:.0f}s ===")
    for idx, item in enumerate(top):
        sc, _id, name, f = item
        if time.monotonic() >= main_deadline - 20.0:
            break
        remain = main_deadline - time.monotonic() - 10.0
        if remain <= 30.0:
            break
        frac = weights[idx] / sum(weights[idx:]) if idx < len(weights) else 0.0
        alloc = max(20.0, min(remain - 5.0, 0.92 * main_budget * frac))
        if idx == 0:
            alloc = min(alloc, main_budget * 0.55)

        print(f"\n=== Main #{idx+1} seed={name} alloc={alloc:.0f}s best={best_score:.6f} ===")
        f_opt, sc_opt = _gradient_optimize(f, alloc, rng, f"main_{name}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            best_c = _autocorr(best_f)

    # Phase 3: Try a different length if time remains
    now = time.monotonic()
    extra_time = total_deadline - now - final_reserve
    if extra_time > 200.0:
        # Try a nearby length
        alt_n = n * 2 if n < 10000 else n // 2
        alt_n = max(1024, min(20000, alt_n))
        if alt_n != n:
            alloc_alt = min(180.0, extra_time * 0.5)
            print(f"\n=== Alt length n={alt_n} budget={alloc_alt:.0f}s ===")
            alt_f = _interp_to_length(best_f, alt_n)
            alt_f, alt_s = _gradient_optimize(alt_f, alloc_alt, rng, f"alt_{alt_n}")
            if alt_s < best_score:
                best_score = alt_s
                best_f = alt_f.copy()
                best_c = _autocorr(best_f)
                print(f"Alt length improved to {best_score:.6f}")

    # Phase 4: Final dedicated coordinate polish
    now = time.monotonic()
    final_time = max(0.0, total_deadline - now - 20.0)
    if final_time > 30.0:
        print(f"\n=== Final coordinate polish budget={final_time:.0f}s ===")
        bf, bs, bc = _coordinate_polish(best_f, _autocorr(best_f), final_time, rng)
        if bs < best_score:
            best_score = bs
            best_f = bf.copy()
            best_c = bc

    # Phase 5: One more gradient pass if time
    now = time.monotonic()
    extra = max(0.0, total_deadline - now - 15.0)
    if extra > 60.0:
        grad_extra = min(120.0, 0.5 * extra)
        print(f"\n=== Final gradient pass budget={grad_extra:.0f}s ===")
        f2, s2 = _gradient_optimize(best_f, grad_extra, rng, "final_grad")
        if s2 < best_score:
            best_score = s2
            best_f = f2.copy()

        # Tiny final coordinate polish
        extra2 = max(0.0, total_deadline - time.monotonic() - 10.0)
        if extra2 > 15.0:
            bf, bs, bc = _coordinate_polish(best_f, _autocorr(best_f), min(extra2, 60.0), rng)
            if bs < best_score:
                best_score = bs
                best_f = bf.copy()

    # Final symmetrization check
    sf = _symmetrize(best_f)
    sc = _fast_score(sf)
    if sc < best_score:
        best_score = sc
        best_f = sf
        print(f"Final symmetrization improved to {best_score:.6f}")

    # Safety: final evaluation
    best_f = _normalize(best_f)
    best_f = np.nan_to_num(best_f, nan=0.0, posinf=0.0, neginf=0.0)
    np.clip(best_f, 0.0, 1000.0, out=best_f)
    best_f = best_f / max(1e-300, float(best_f.sum()))
    best_c = _autocorr(best_f)
    best_score = float(2.0 * best_f.size * best_c.max())

    try:
        out = [float(x) for x in best_f]
        official = evaluate_sequence(out)
        if np.isfinite(official) and official < best_score + 1e-12:
            best_score = float(official)
    except Exception:
        pass

    out = [float(x) for x in best_f]
    print(f"\nFINAL internal score = {best_score:.6f}")
    print(f"FINAL target = {TARGET_SCORE}")
    return out
