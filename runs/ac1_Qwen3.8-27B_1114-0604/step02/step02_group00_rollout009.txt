                cols.append(j)

def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_cross_conv(f, g):

    n = len(f) + len(g) - 1

    F = np.fft.rfft(f, M_fft)
    G = np.fft.rfft(g, M_fft)
    return np.fft.irfft(F * G, M_fft)[:n]

"""
LP-alternating + L-BFGS + Multiplicative Weights hybrid optimizer.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Key strategies:
1. Focused on moderate lengths (1024-4096) where LP refinement is tractable
2. LP-based alternating method solves dual LP with top-K active constraints
3. Multiplicative weights (mirror descent) for non-negativity-preserving local search
4. L-BFGS with LSE continuation for smooth optimization
5. Diverse initializations including periodic, symmetric, and interpolated parent
"""

import numpy as np
from scipy.optimize import minimize
from scipy.sparse import csr_matrix
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    scaled = np.clip(scaled, -400, 400)
    exp_scaled = np.exp(scaled)
    exp_sum = exp_scaled.sum()
    log_sum = np.log(exp_sum + 1e-300)
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / (exp_sum + 1e-300)
    rev_a = a[::-1]
    M2 = 3 * n - 1
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)
    g = 2.0 * conv_res[n - 1:2 * n - 1]
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=3000, maxfun=60000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-14, 'gtol': 1e-12,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 40}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    sc = compute_score(a_out)
    return sc, a_out


def interpolate_sequence(a_src, n_dst):
    """Interpolate sequence from source length to destination length."""
    if len(a_src) == n_dst:
        return a_src.copy()
    x_src = np.linspace(0, 1, len(a_src))
    x_dst = np.linspace(0, 1, n_dst)
    return np.interp(x_dst, x_src, a_src)


def build_inits(n, rng, parent):

    inits = []
    # Interpolated parent
    if parent is not None and len(parent) >= 32:

        p = np.clip(p, 0.01, 1000.0)

    # Uniform
    inits.append(np.ones(n) * 0.5)
    # Periodic patterns
    for period in (4, 8, 12, 16, 24):
        if period < n:
            pat = 0.3 + 0.4 * rng.random(period)

    # Symmetric (palindromic) patterns
    half = np.clip(0.3 + 0.4 * rng.random(n // 2 + 1), 0.01, 1000.0)

    # Cosine modulation

    inits.append(0.3 + 0.4 * np.cos(2.0 * np.pi * x))

def lp_refine(f, K=200, n_iter=40, t_init=0.3, verbose=False):
    """
    LP-based alternating refinement.
    Solves: max sum(g) s.t. (f*g)[k] <= M for top-K lags, g >= 0
    Then updates f <- (1-t)*f + t*g/sum(g)
    """
    n = len(f)
    f = f.astype(np.float64).copy()
    S = f.sum()
    if S < 1e-12:
        f[:] = 1.0 / n
        S = 1.0
    f = f / S  # normalize to sum = 1
    M = float(np.max(fast_conv(f)))
    t = t_init
    best_M = M
    best_f = f.copy()

    for it in range(n_iter):
        # Find top-K lags in conv(f,f)
        conv_f = fast_conv(f)
        top_k_idx = np.argsort(conv_f)[::-1][:K]

        # Build constraint matrix A (K x n)
        # A[r, j] = f[k_r - j] if 0 <= k_r - j < n, else 0
        rows = []
        cols = []
        vals = []
        for r, k in enumerate(top_k_idx):
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_end >= j_start:
                js = np.arange(j_start, j_end + 1)
                rows.extend([r] * len(js))
                cols.extend(js.tolist())
                vals.extend(f[k - js].tolist())

        if len(vals) == 0:
            break

        A = csr_matrix((vals, (rows, cols)), shape=(K, n))

        # Solve LP: max sum(g) s.t. A@g <= M, g >= 0
        # Use scipy linprog (minimize -sum(g) s.t. A@g <= M, g >= 0)
        c_obj = -np.ones(n)
        try:
            from scipy.optimize import linprog
            res_lp = linprog(
                c_obj,
                A_ub=A,
                b_ub=np.full(K, M),
                bounds=[(0, None)] * n,
                method='highs'
            )
        except Exception:
            try:
                res_lp = linprog(
                    c_obj,
                    A_ub=A,
                    b_ub=np.full(K, M),
                    bounds=[(0, None)] * n,
                    method='interior-point'
                )
            except Exception:
                break

        if res_lp.status != 0 or res_lp.x is None:
            # Try with slightly relaxed M
            res_lp = None
            try:
                from scipy.optimize import linprog
                res_lp = linprog(
                    c_obj,
                    A_ub=A,
                    b_ub=np.full(K, M * 1.01),
                    bounds=[(0, None)] * n,
                    method='highs'
                )
            except Exception:
                break

        if res_lp is None or res_lp.x is None or res_lp.status != 0:
            break

        g = np.clip(res_lp.x, 0, None)
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Update
        f_new = (1 - t) * f + t * g
        f_new = np.clip(f_new, 0, None)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-12:
            break
        f_new = f_new / f_new_sum

        new_M = float(np.max(fast_conv(f_new)))
        if new_M < best_M:
            best_M = new_M
            best_f = f_new.copy()
            M = new_M
        f = f_new

        # Decay t
        t = max(0.02, t_init * (0.95 ** it))

        if verbose and it % 10 == 0:
            print(f"    LP iter {it}: M={M:.2e}, best_M={best_M:.2e}, t={t:.3f}")

    return best_f


def multiplicative_weights_refine(a, n_iter=200, eta_init=0.5, verbose=False):
    """
    Multiplicative weights (mirror descent) on the simplex.
    Naturally preserves non-negativity.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    S = a.sum()
    if S < 1e-12:
        a[:] = 1.0 / n
        return a
    a = a / S

    best_a = a.copy()
    best_sc = compute_score(a)
    eta = eta_init

    for it in range(n_iter):
        c = fast_conv(a)
        k_star = int(np.argmax(c))
        M = c[k_star]

        # Subgradient at max lag: sg[j] = 2*a[k_star - j] for valid k_star-j
        sg = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            sg[j_start:j_end + 1] = 2.0 * a[j_start:j_end + 1][::-1]

        # Also add a small uniform component to avoid collapse
        sg += 0.01

        # Multiplicative update
        a = a * np.exp(-eta * sg / a.sum())
        a = np.clip(a, 1e-15, None)
        a = a / a.sum()

        sc = compute_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Decay eta
        eta = eta_init * (0.99 ** it)

        if verbose and it % 50 == 0:
            print(f"    MW iter {it}: score={sc:.6f}, eta={eta:.4f}")

    return best_a

    n = len(a)

    best_sc = compute_score(a)
    sc_lb = 1.0

    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        a_norm = a / S
        M = float(np.max(fast_conv(a_norm)))
        k_star = int(np.argmax(fast_conv(a_norm)))
        sc = 2.0 * n * M  # score when sum=1

        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

        # Subgradient
        sg_M = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            seg = a_norm[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]

        # Projected gradient step (on simplex)
        step = 2.0 * (sc - sc_lb) / (np.sum(sg_M ** 2) + 1e-30)
        step = min(step, 0.1)
        a_norm = a_norm - step * sg_M
        a_norm = np.clip(a_norm, 0, None)
        s = a_norm.sum()
        if s > 1e-12:
            a_norm = a_norm / s
        a = a_norm * S  # maintain original scale

    return best_sc, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    overhead = 15.0
    deadline = t_start + max(30.0, float(budget_s)) - overhead

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 32:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f} (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    print(f"Parent available: {parent is not None}")
    if parent is not None:
        print(f"  Parent length: {len(parent)}, score: {compute_score(parent):.8f}")

    # Determine lengths to try
    lengths = [1024, 2048, 4096]
    if parent is not None:
        pn = len(parent)
        if pn not in lengths and pn <= 8192:
            lengths.append(pn)
    lengths = sorted(set(lengths))

    # ================================================================
    # Phase 1: L-BFGS with LSE continuation
    # ================================================================
    phase1_deadline = t_start + 0.35 * (float(budget_s) - overhead)
    print(f"\n=== Phase 1: L-BFGS (deadline at {phase1_deadline - t_start:.0f}s) ===")

    beta_schedule = [20, 50, 100, 300]

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"\n  n = {n}")
        inits = build_inits(n, rng, parent)
        # Limit initializations based on time
        time_per_init = 5.0  # rough estimate
        remaining = phase1_deadline - time.monotonic()
        max_inits = max(2, int(remaining / (time_per_init * len(beta_schedule))))
        inits = inits[:max_inits]
        print(f"    Trying {len(inits)} initializations")

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            a_cur = a0.copy() / (a0.sum() + 1e-12) * 0.5  # normalize
            for beta in beta_schedule:
                if time.monotonic() > phase1_deadline:
                    break
                sc, a_opt = run_lbfgs(a_cur, beta, maxiter=2000, maxfun=40000)
                a_cur = a_opt
                report(sc, a_opt, f"n={n}, init={i}, beta={beta}")

    print(f"\n  After Phase 1: best = {best_score:.8f}")

    # ================================================================
    # Phase 2: LP-based alternating refinement
    # ================================================================
    phase2_deadline = t_start + 0.65 * (float(budget_s) - overhead)
    print(f"\n=== Phase 2: LP Refinement (deadline at {phase2_deadline - t_start:.0f}s) ===")

    # Try LP refinement at various lengths
    lp_lengths = [l for l in lengths if l <= 4096]
    if not lp_lengths:
        lp_lengths = [2048]

    for n in lp_lengths:
        if time.monotonic() > phase2_deadline:
            break
        print(f"\n  LP refine at n={n}")

        # Get a good starting point for this length
        if len(best_a) == n:
            a_start = best_a.copy()
        elif parent is not None:
            a_start = interpolate_sequence(parent, n)
        else:
            a_start = np.ones(n) * 0.5
        a_start = np.clip(a_start, 0.01, 1000.0)

        # First run a quick L-BFGS to get a decent starting point
        if time.monotonic() < phase2_deadline - 30:
            sc, a_start = run_lbfgs(a_start, 100, maxiter=2000, maxfun=40000)
            report(sc, a_start, f"LP prep n={n}")

        # Run LP refinement with different K values
        for K in [100, 200, 400]:
            if time.monotonic() > phase2_deadline:
                break
            t0 = time.monotonic()
            a_lp = lp_refine(a_start, K=K, n_iter=40, t_init=0.3)
            sc = compute_score(a_lp)
            report(sc, a_lp, f"LP n={n}, K={K}")
            print(f"    LP K={K} took {time.monotonic()-t0:.1f}s, score={sc:.8f}")

    # ================================================================
    # Phase 3: Multiplicative weights refinement
    # ================================================================
    phase3_deadline = t_start + 0.80 * (float(budget_s) - overhead)
    print(f"\n=== Phase 3: Multiplicative Weights (deadline at {phase3_deadline - t_start:.0f}s) ===")

    for eta_init in [0.3, 0.5, 0.8]:
        if time.monotonic() > phase3_deadline:
            break
        a_mw = multiplicative_weights_refine(best_a, n_iter=200, eta_init=eta_init)
        sc = compute_score(a_mw)
        report(sc, a_mw, f"MW eta={eta_init}")

    # ================================================================
    # Phase 4: Subgradient polish + final L-BFGS
    # ================================================================
    phase4_deadline = deadline
    print(f"\n=== Phase 4: Final Polish (deadline at {phase4_deadline - t_start:.0f}s) ===")

    # Subgradient polish
    if time.monotonic() < phase4_deadline - 30:
        sc, a_pol = exact_subgrad_polish(best_a, total_iters=300)
        report(sc, a_pol, "subgrad polish")

    # Final L-BFGS rounds at high beta
    for beta in [300, 800, 1500]:
        if time.monotonic() > phase4_deadline - 10:
            break
        sc, a_opt = run_lbfgs(best_a, beta, maxiter=5000, maxfun=100000)
        report(sc, a_opt, f"final lbfgs beta={beta}")

    # One more LP round on the final best (if length allows)
    if len(best_a) <= 4096 and time.monotonic() < phase4_deadline - 20:
        a_lp2 = lp_refine(best_a, K=300, n_iter=30, t_init=0.2)
        sc = compute_score(a_lp2)
        report(sc, a_lp2, "final LP")
        if sc < best_score:
            # Quick L-BFGS polish
            if time.monotonic() < phase4_deadline - 5:
                sc2, a_t2 = run_lbfgs(a_lp2, 500, maxiter=3000, maxfun=60000)
                report(sc2, a_t2, "post-LP lbfgs")

    # ================================================================
    # Final verification and return
    # ================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
