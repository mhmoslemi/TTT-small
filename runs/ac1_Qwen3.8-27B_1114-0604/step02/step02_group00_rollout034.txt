```python
    return 2.0 * n * float(np.max(c)) / (S * S)

        # For k outside range: conv[k] = base_conv[k]
        conv = base_conv.copy()
        conv[k_lo:k_hi+1] = base_inside + t_opt * alpha_vals + t_opt * t_opt * beta_vals
```

- For k=2i (if 0≤2i<2n-1): delta = new_val^2 - old_val^2
- For k where 0≤i-k<n and i-k≠k: same as above (already covered)

Wait, let me think more carefully. conv[k] = sum_j a[j]*a[k-j].

If a[i] changes by δ = new_val - old_val:
- Terms where j=i: a[i]*a[k-i] changes by δ*a[k-i] (if 0≤k-i<n)
- Terms where k-j=i, i.e., j=k-i: a[k-i]*a[i] changes by a[k-i]*δ (if 0≤k-i<n and k-i≠i)

So total change: 
- If k-i is valid (0≤k-i<n):

  - If k=2i: δ*a[i] + a[i]*δ... wait, k-i = i, so both terms give δ*a[i]? No.

- For 0≤k-i<n and k≠2i: delta_conv[k] = 2*δ*a[k-i]
- For k=2i (and 0≤2i<2n-1): delta_conv[k] = new^2 - old^2

```python
def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad
```

```python
  - Phase 4: Final polish

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    """Wrapper for scipy.minimize."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent (top-level, no closure)."""
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)


def coordinate_descent_sweep(a, rng=None, n_coords=None):
    """
    One full sweep of coordinate descent with incremental conv updates.
    Returns (score, updated_a).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1
    
    if n_coords is None:
        n_coords = n
    
    if rng is not None:
        order = rng.permutation(n)
    else:
        order = np.arange(n)
    
    for idx in range(min(n_coords, n)):
        i = int(order[idx])
        a_i = a[i]
        
        # Affected lags: k in [i, i+n-1] ∩ [0, 2n-2]
        k_lo = i  # since i >= 0
        k_hi = min(M_conv - 1, i + n - 1)
        
        # Outside max
        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0
        
        inside_len = k_hi - k_lo + 1
        base_inside = conv[k_lo:k_hi + 1].copy()
        
        # Compute alpha and beta
        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        k_minus_i = ks - i  # ranges from 0 to inside_len-1
        a_vals = a[k_minus_i]
        alpha = 2.0 * a_vals.copy()
        is_2i = (ks == 2 * i)
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0
        
        S_rest = S - a_i
        
        # Upper bound for t
        t_upper = max(3.0 * a_i + 0.1, S_rest / n * 2.0)
        t_upper = min(t_upper, 1000.0)
        
        res = minimize_scalar(
            coord_1d_obj,
            args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
            bounds=(0.0, t_upper),
            method='bounded',
            options={'xatol': 1e-8}
        )
        t_opt = res.x
        
        # Update
        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef
    
    final_score = compute_score(a)
    return final_score, a


def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements contributing most to the peak."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(3, n // 50)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:n - top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)
```

```python
def golden_1d(f, a, b, tol=1e-7, max_iter=50):
    """Golden section search for minimum of f on [a, b]."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio ≈ 0.618
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd

            fd = f(d)
    return 0.5 * (a + b)
```

```python
"""
Hybrid coordinate descent + LSE optimizer for minimum peak autocorrelation.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy:
  - Phase 1: Many sweeps of exact coordinate descent (handles exact max, no smoothing)
  - Phase 2: LSE-based L-BFGS refinement for global moves
  - Phase 3: Coordinate descent polish after LSE
  - Phase 4: Perturbation + coordinate descent cycles to escape local minima
  - Phase 5: Final polish

Key insight: For coordinate i, the objective as a function of a[i] (with others fixed)
is a 1D problem that can be solved exactly (no smoothing needed). The convolution is
updated incrementally in O(n) per coordinate change.
"""

import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    """Wrapper for scipy.minimize."""
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000):
    """Run L-BFGS-B on LSE-smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-15, 'gtol': 1e-13,
                 'maxiter': maxiter, 'maxfun': maxfun,
                 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def coord_1d_obj(t, base_inside, alpha, beta_coef, outside_max, S_rest, n):
    """1D objective for coordinate descent (top-level, no closure)."""
    conv_in = base_inside + t * alpha + (t * t) * beta_coef
    im = float(np.max(conv_in))
    tot_max = im if im > outside_max else outside_max
    S_new = S_rest + t
    if S_new < 1e-12:
        return 1e10
    return 2.0 * n * tot_max / (S_new * S_new)


def coordinate_descent_sweep(a, rng=None, n_coords=None):
    """
    One full sweep of coordinate descent with incremental conv updates.
    Returns (score, updated_a).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    if n_coords is None:
        n_coords = n

    if rng is not None:
        order = rng.permutation(n)
    else:
        order = np.arange(n)

    for idx in range(min(n_coords, n)):
        i = int(order[idx])
        a_i = a[i]

        # Affected lags: k in [i, i+n-1] intersect [0, 2n-2]
        k_lo = i
        k_hi = min(M_conv - 1, i + n - 1)

        # Outside max (lags not affected by a[i])
        if k_lo > 0 and k_hi < M_conv - 1:
            outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
        elif k_lo > 0:
            outside_max = float(np.max(conv[:k_lo]))
        elif k_hi < M_conv - 1:
            outside_max = float(np.max(conv[k_hi + 1:]))
        else:
            outside_max = 0.0

        inside_len = k_hi - k_lo + 1
        # base_inside = convolution at affected lags with a[i] REMOVED
        base_inside = conv[k_lo:k_hi + 1].copy()
        
        # Subtract old contribution of a[i]
        ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
        a_vals = a[ks - i]  # a[k-i] for each affected k
        is_2i = (ks == 2 * i)
        old_contrib = 2.0 * a_i * a_vals
        old_contrib[is_2i] = a_i * a_i  # at k=2i, contribution is a_i^2 not 2*a_i^2
        base_inside -= old_contrib

        # Set up 1D problem: conv_new = base_inside + t*alpha + t^2*beta
        alpha = 2.0 * a_vals.copy()
        alpha[is_2i] = 0.0
        beta_coef = np.zeros(inside_len)
        beta_coef[is_2i] = 1.0

        S_rest = S - a_i

        # Upper bound for t
        t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
        t_upper = min(t_upper, 1000.0)

        res = minimize_scalar(
            coord_1d_obj,
            args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
            bounds=(0.0, t_upper),
            method='bounded',
            options={'xatol': 1e-7}
        )
        t_opt = float(res.x)

        # Update a and conv
        a[i] = t_opt
        S = S_rest + t_opt
        conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

    final_score = compute_score(a)
    return final_score, a


def targeted_perturb(a, rng, strength=0.05):
    """Perturb elements contributing most to the peak convolution."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    contrib = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(3, n // 50)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:n - top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] = max(a_pert[j] - reduce_amt, 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        a_pert[bottom_idx] += per
    return np.clip(a_pert, 0.0, 1000.0)


def hot_coordinate_sweep(a, n_sweeps=1, cooldown=5):
    """
    Coordinate descent that prioritizes coordinates contributing to the current peak.
    This focuses optimization effort where it matters most.
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    conv = fast_conv(a)
    S = float(a.sum())
    M_conv = 2 * n - 1

    for _ in range(n_sweeps):
        k_star = int(np.argmax(conv))
        # Compute contribution of each a[i] to conv[k_star]
        contrib = np.zeros(n)
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        if j_end >= j_start:
            contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
        # Sort by contribution (hottest first)
        order = np.argsort(contrib)[::-1]

        for i in order:
            i = int(i)
            a_i = a[i]

            k_lo = i
            k_hi = min(M_conv - 1, i + n - 1)

            if k_lo > 0 and k_hi < M_conv - 1:
                outside_max = max(float(np.max(conv[:k_lo])), float(np.max(conv[k_hi + 1:])))
            elif k_lo > 0:
                outside_max = float(np.max(conv[:k_lo]))
            elif k_hi < M_conv - 1:
                outside_max = float(np.max(conv[k_hi + 1:]))
            else:
                outside_max = 0.0

            inside_len = k_hi - k_lo + 1
            base_inside = conv[k_lo:k_hi + 1].copy()

            ks = np.arange(k_lo, k_hi + 1, dtype=np.int64)
            a_vals = a[ks - i]
            is_2i = (ks == 2 * i)
            old_contrib = 2.0 * a_i * a_vals
            old_contrib[is_2i] = a_i * a_i
            base_inside -= old_contrib

            alpha = 2.0 * a_vals.copy()
            alpha[is_2i] = 0.0
            beta_coef = np.zeros(inside_len)
            beta_coef[is_2i] = 1.0

            S_rest = S - a_i
            t_upper = max(3.0 * a_i + 0.01, S_rest / n * 3.0)
            t_upper = min(t_upper, 1000.0)

            res = minimize_scalar(
                coord_1d_obj,
                args=(base_inside, alpha, beta_coef, outside_max, S_rest, n),
                bounds=(0.0, t_upper),
                method='bounded',
                options={'xatol': 1e-7}
            )
            t_opt = float(res.x)
            a[i] = t_opt
            S = S_rest + t_opt
            conv[k_lo:k_hi + 1] = base_inside + t_opt * alpha + (t_opt * t_opt) * beta_coef

    final_score = compute_score(a)
    return final_score, a
```
