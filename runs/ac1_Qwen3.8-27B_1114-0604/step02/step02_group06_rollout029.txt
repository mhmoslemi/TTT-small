def _mmd_optimize(p0, time_limit, n_target=None, verbose=False):
    """
    MMD-style optimization with momentum and adaptive parameters.
    """
    a = _sanitize(p0)
    if a is None:
        return None, float('inf')
    
    n = len(a)
    L = _fft_len(n)
    
    deadline = time.monotonic() + time_limit
    best = a.copy()
    best_sc, _ = _objective_with_corr(a, L)
    cur_sc = best_sc
    
    # Momentum
    velocity = np.zeros(n)
    mu = 0.9  # momentum coefficient
    
    # Adaptive parameters
    beta = 50.0  # softmax temperature
    alpha = 0.1  # base step size
    eps = 0.15  # threshold relaxation
    
    it = 0
    stall = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        # Compute correlation
        F = np.fft.rfft(a, L)
        c = np.fft.irfft(F * F, L)
        c = c[:2*n-1]
        c = np.maximum(c, 0)
        max_c = c.max()
        cur_sc = float(2 * n * max_c)  # assuming sum(a) = 1
        
        # Update best
        if cur_sc < best_sc:
            best_sc = cur_sc
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute peak weights
        if max_c > 1e-15:
            # Softmax weights
            d = beta * c
            d_shift = d - d.max()
            w = np.exp(d_shift)
            w = w / w.sum()
        else:
            w = np.ones(2*n-1) / (2*n-1)
        
        # Compute update direction
        # h[j] ≈ gradient direction (which positions to reduce)
        W = np.fft.rfft(w, L)
        h = np.fft.irfft(W * np.conj(F), L)[:n]
        h = np.maximum(h, 0)
        
        if h.max() < 1e-15:
            # Random restart
            a = _sanitize(np.maximum(a, 0) * np.exp(_rng.normal(0, 0.3, n)))
            velocity = np.zeros(n)
            stall = 0
            it += 1
            continue
        
        h = h / h.max()
        
        # Nesterov momentum step
        # Lookahead
        a_lookahead = a - mu * velocity
        # Compute gradient at lookahead (approximation: just use current h)
        grad = h  # simplified: use current gradient direction
        
        velocity = mu * velocity - alpha * grad
        a_new = a + velocity
        
        # Project to simplex (non-negative, sum=1)
        a_new = np.maximum(a_new, 1e-15)
        a_new = a_new / a_new.sum()
        
        # Line search: check if this step improves
        sc_new, _ = _objective_with_corr(a_new, L)
        
        if sc_new < cur_sc:
            a = a_new
            cur_sc = sc_new
            stall = 0
        else:
            # Backtrack
            for step_factor in [0.5, 0.25, 0.1, 0.05]:

"""
Hybrid MMD + Block Coordinate Descent optimizer for minimizing
2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences.

Key improvements over prior approach:
- Nesterov momentum in log-space for faster convergence
- Adaptive softmax temperature with exponential warmup
- Two-phase optimization (global then local peak focusing)
- Diverse initializations (binary, Poisson, frequency-domain, symmetric)
- Block coordinate descent for final polish
- Multi-scale with larger final sizes (up to 32768)
- Backtracking line search with proper acceptance criteria

import numpy as np
import time

_GLOBAL_RNG = np.random.default_rng(777)
_CONV_CACHE = {}


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _fft_len(n):
    key = int(n)
    if key in _CONV_CACHE:
        return _CONV_CACHE[key]
    L = _next_pow2(2 * key - 1)
    _CONV_CACHE[key] = L
    return L


def _sanitize(a, normalize=True):
    if a is None:
        return None
    a = np.asarray(a, dtype=float).ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    if normalize:
        s = float(a.sum())
        if s <= 1e-15:
            return None
        a = a / s
    return a


def _score_and_corr(a, L=None):
    n = int(a.size)
    S = float(a.sum())
    if S < 1e-10:
        return float('inf'), None
    if L is None:
        L = _fft_len(n)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)
    c = c[:2 * n - 1]
    c = np.maximum(c, 0.0)
    max_c = float(c.max())
    sc = float(2.0 * n * max_c / (S * S))
    return sc, c


def _score(a, L=None):
    sc, _ = _score_and_corr(a, L)
    return sc


def _init_uniform(n):
    return np.ones(n, dtype=float) / n


def _init_binary(n, p=0.5):
    a = (_GLOBAL_RNG.random(n) < p).astype(float)
    return _sanitize(a)


def _init_poisson(n, lam=1.5):
    a = _GLOBAL_RNG.poisson(lam, n).astype(float)
    return _sanitize(a)


def _init_poisson_var(n):
    lam = float(_GLOBAL_RNG.uniform(0.5, 4.0))
    a = _GLOBAL_RNG.poisson(lam, n).astype(float)
    return _sanitize(a)


def _init_symmetric_binary(n):
    half = n // 2
    a_half = (_GLOBAL_RNG.random(half) < 0.5).astype(float)
    if n % 2 == 0:
        a = np.concatenate([a_half, a_half[::-1]])
    else:
        a = np.concatenate([a_half, [0.5], a_half[::-1]])
    return _sanitize(a)


def _init_smooth_gaussian(n):
    a = _GLOBAL_RNG.normal(0, 1, n)
    sigma = max(1.0, n * 0.03)
    rad = int(max(1, 3 * sigma))
    k = np.arange(-rad, rad + 1, dtype=float)
    kern = np.exp(-0.5 * (k / sigma) ** 2)
    kern = kern / kern.sum()
    a = np.convolve(a, kern, mode='same')
    a = np.abs(a)
    return _sanitize(a)


def _init_freq_random(n):
    L = _fft_len(n)
    F = _GLOBAL_RNG.normal(0, 1, L // 2 + 1) + 1j * _GLOBAL_RNG.normal(0, 1, L // 2 + 1)
    a = np.fft.irfft(F, L)[:n]
    a = np.abs(a) + 1e-10
    return _sanitize(a)


def _init_two_level(n):
    ratio = float(_GLOBAL_RNG.uniform(0.2, 0.7))
    r = float(_GLOBAL_RNG.uniform(0.05, 2.0))
    a = np.full(n, r)
    k = int(n * ratio)
    idx = _GLOBAL_RNG.choice(n, size=k, replace=False)
    a[idx] = 1.0
    return _sanitize(a)


def _init_multimode(n, rng):
    mode = int(rng.integers(0, 8))
    if mode == 0:
        return _init_binary(n)
    elif mode == 1:
        return _init_poisson_var(n)
    elif mode == 2:
        return _init_smooth_gaussian(n)
    elif mode == 3:
        return _init_freq_random(n)
    elif mode == 4:
        return _init_two_level(n)
    elif mode == 5:
        return _init_symmetric_binary(n)
    elif mode == 6:
        # Sum of shifted Gaussians
        K = int(rng.integers(2, 5))
        a = np.zeros(n)
        t = np.arange(n, dtype=float)
        for _ in range(K):
            c = float(rng.uniform(0, n))
            s = float(rng.uniform(n * 0.01, n * 0.2))
            w = float(rng.uniform(0.5, 2.0))

    else:

def _mmd_optimize(a0, time_limit, beta_start=30.0, beta_end=3000.0,
                  alpha_start=0.12, verbose=False, target=1.50):
    """
    Core MMD optimization with momentum in log-space.
    """
    a = _sanitize(a0)
    if a is None:
        return None, float('inf')

    n = int(a.size)
    L = _fft_len(n)
    deadline = time.monotonic() + float(max(0.0, time_limit))

    best = a.copy()
    best_sc, _ = _score_and_corr(a, L)
    cur_sc = best_sc

    # Log-space momentum
    log_a = np.log(a + 1e-300)
    velocity = np.zeros(n)
    mu = 0.7  # momentum

    beta = float(beta_start)
    alpha = float(alpha_start)
    threshold_mode = True

    it = 0
    stall = 0
    iprod = 0

    if time_limit <= 0:
        return best, best_sc

    while time.monotonic() < deadline:
        # Compute correlation
        F = np.fft.rfft(a, L)
        c = np.fft.irfft(F * F, L)
        c = c[:2 * n - 1]
        c = np.maximum(c, 0.0)
        max_c = float(c.max())
        cur_sc = float(2.0 * n * max_c)  # sum(a)=1 assumed

        if cur_sc < best_sc - 1e-12:
            best_sc = cur_sc
            best = a.copy()
            iprod = 0
            stall = 0
        else:
            iprod += 1

        # Compute weights on peaks
        mean_c = float(c.mean())
        target_m = target / (2.0 * n)

        if max_c > target_m * 1.001:
            # Above target: focus on peaks
            if threshold_mode and it % 3 == 0:
                # Threshold-based: only weight peaks above threshold
                thr = 0.7 * max_c + 0.3 * max(mean_c, target_m)
                viol = np.maximum(0.0, c - thr)
                sv = float(viol.sum())
                if sv > 1e-15:
                    w = viol / sv
                    # Sharpen
                    w = w ** 2
                    sw = float(w.sum())
                    if sw > 1e-15:
                        w = w / sw
                    else:
                        w = np.exp(beta * (c - max_c))
                        w = w / w.sum()
                else:
                    d = beta * (c - max_c)
                    w = np.exp(d)
                    w = w / w.sum()
            else:
                # Softmax
                d = beta * (c - max_c)
                w = np.exp(d)
                wsum = float(w.sum())
                if wsum < 1e-15 or not np.isfinite(wsum):
                    w = np.zeros(2 * n - 1)
                    w[int(np.argmax(c))] = 1.0
                else:
                    w = w / wsum
        else:
            # At or below target: try to flatten everything
            d = beta * c
            d = d - d.max()
            w = np.exp(d)
            wsum = float(w.sum())
            if wsum < 1e-15:
                w = np.ones(2 * n - 1) / (2 * n - 1)
            else:
                w = w / wsum

        # Compute update direction h
        W = np.fft.rfft(w, L)
        h = np.fft.irfft(W * np.conj(F), L)[:n]
        h = np.maximum(h, 0.0)
        hm = float(h.max())

        if hm < 1e-20:
            # Degenerate: random kick
            log_a = log_a + _GLOBAL_RNG.normal(0, 0.1, n)
            a = np.exp(log_a)
            a = np.maximum(a, 1e-15)
            a = a / a.sum()
            velocity *= 0.5
            stall += 1
            it += 1
            continue

        h = h / hm

        # Log-space multiplicative update with momentum
        # grad in log-space: h (positions to reduce)
        grad = h
        velocity = mu * velocity - alpha * grad
        new_log_a = log_a + velocity
        a_new = np.exp(new_log_a)
        a_new = np.maximum(a_new, 1e-15)
        s = float(a_new.sum())
        if s > 1e-15 and np.isfinite(s):
            a_new = a_new / s
        else:
            a_new = a.copy()

        # Evaluate
        sc_new, _ = _score_and_corr(a_new, L)

        accepted = False
        if np.isfinite(sc_new) and sc_new < cur_sc:
            a = a_new
            log_a = np.log(a + 1e-300)
            cur_sc = sc_new
            accepted = True
        else:
            # Backtracking
            for lam in [0.5, 0.25, 0.1, 0.05]:
                a_try = np.exp(log_a + lam * velocity)
                a_try = np.maximum(a_try, 1e-15)
                st = float(a_try.sum())
                if st > 1e-15:
                    a_try = a_try / st
                    sc_try, _ = _score_and_corr(a_try, L)
                    if np.isfinite(sc_try) and sc_try < cur_sc:
                        a = a_try
                        log_a = np.log(a + 1e-300)
                        cur_sc = sc_try
                        velocity = velocity * lam
                        accepted = True
                        break

            if not accepted:
                alpha = max(0.02, alpha * 0.85)
                velocity = velocity * 0.7
                stall += 1

                if stall >= 15:
                    # Perturbation
                    sigma = 0.05 + 0.01 * min(stall, 30)
                    pert = _GLOBAL_RNG.normal(0, sigma, n)
                    a_try = a * np.exp(pert)
                    a_try = np.maximum(a_try, 1e-15)
                    a_try = a_try / a_try.sum()
                    sc_try, _ = _score_and_corr(a_try, L)
                    if np.isfinite(sc_try) and sc_try < max(cur_sc * 1.03, best_sc * 1.02):
                        a = a_try
                        log_a = np.log(a + 1e-300)
                        cur_sc = sc_try
                    else:
                        a = best.copy()
                        log_a = np.log(a + 1e-300)
                        cur_sc = best_sc
                    velocity = np.zeros(n)
                    alpha = float(alpha_start)
                    stall = 0

                if stall >= 80:
                    # Reset momentum, reduce beta
                    velocity = np.zeros(n)
                    beta = max(float(beta_start) * 0.5, beta * 0.6)
                    alpha = float(alpha_start)
                    a = best.copy()
                    log_a = np.log(a + 1e-300)
                    cur_sc = best_sc
                    stall = 0

        # Adapt beta
        if it % 25 == 0:
            beta = min(float(beta_end), beta * 1.06)

        # Slowly increase alpha back if it was reduced
        if accepted and alpha < alpha_start:
            alpha = min(alpha_start, alpha * 1.02)

        it += 1
        if verbose and it % 300 == 0:
            print(f"    MMD it={it} cur={cur_sc:.6f} best={best_sc:.6f} beta={beta:.0f} alpha={alpha:.4f} stall={stall}")

    return best, best_sc


def _block_cd(a0, time_limit, verbose=False):
    """Block coordinate descent: scale contiguous blocks to reduce peak."""
    a = _sanitize(a0)

    cur_sc = best_sc

    deadline = time.monotonic() + float(max(0.0, time_limit))

    small = max(1, n // 2048)
    mid = max(small + 1, n // 256)
    large = max(mid + 1, min(n, n // 32))

    factors = (0.3, 0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5, 2.0, 3.0)
    stagnant = 0
    it = 0

    while time.monotonic() < deadline:

        else:
            w = int(_GLOBAL_RNG.integers(large, max(large + 1, min(n, large * 5))))

        w = max(1, min(n, w))
        start = int(_GLOBAL_RNG.integers(0, n - w + 1))

        best_block_sc = cur_sc
        best_block_a = None

        block_sum = float(a[start:start + w].sum())
        rest_sum = cur_sc  # not used, just for speed
        total = float(a.sum())

        for f in factors:
            new_block_sum = block_sum * f
            new_total = total - block_sum + new_block_sum
            if new_total <= 0:
                continue
            scale = new_total  # we'll normalize at the end
            a_try = a.copy()
            a_try[start:start + w] *= f
            a_try = a_try / a_try.sum()
            sc, _ = _score_and_corr(a_try, L)

def _refine_at_size(a0, n_target, time_limit, verbose=False):
    """Resample to target size and optimize."""
    a = _sanitize(a0)
    if a is None:
        return None, float('inf')

    n = int(a.size)
    if n == n_target:
        a_res = a.copy()
    elif n > n_target:
        # Downsample via interpolation
        x_old = (np.arange(n) + 0.5) / n
        x_new = (np.arange(n_target) + 0.5) / n_target
        a_res = np.interp(x_new, x_old, a)
        a_res = np.maximum(a_res, 1e-15)
        a_res = a_res / a_res.sum()
    else:
        # Upsample
        x_old = (np.arange(n) + 0.5) / n
        x_new = (np.arange(n_target) + 0.5) / n_target
        a_res = np.interp(x_new, x_old, a)
        a_res = np.maximum(a_res, 1e-15)
        a_res = a_res / a_res.sum()

    a_res = _sanitize(a_res)
    if a_res is None:
        return None, float('inf')

    # MMD optimization
    p, sc = _mmd_optimize(a_res, time_limit * 0.7, verbose=verbose)
    if p is not None:
        # Block CD for final polish
        p2, sc2 = _block_cd(p, time_limit * 0.3, verbose=verbose)
        if p2 is not None and sc2 < sc:
            return p2, sc2
    return p, sc if p is not None else float('inf')


def _build_diverse_starts(n, parent, rng, count=6):
    """Build diverse starting sequences of size n."""
    starts = []

    # From parent
    if parent is not None:
        sc = _score(parent)
        if np.isfinite(sc):
            starts.append((sc, parent.copy()))

    # Various random initializations
    init_funcs = [
        lambda: _init_binary(n),
        lambda: _init_poisson_var(n),
        lambda: _init_smooth_gaussian(n),
        lambda: _init_freq_random(n),
        lambda: _init_two_level(n),
        lambda: _init_symmetric_binary(n),
    ]

    for _ in range(count):
        fn = init_funcs[int(rng.integers(0, len(init_funcs)))]
        a = fn()
        if a is not None:
            sc = _score(a)
            if np.isfinite(sc):
                starts.append((sc, a))

    starts.sort(key=lambda x: x[0])
    return [s[1] for s in starts]

    except Exception as e:
        print(f"Failed to load parent: {e}")

    # Candidate pool: (score, size, array)
    cands = []

    def add_cand(arr, sc=None):
        a = _sanitize(arr)
        if a is None:
            return
        if sc is None:
            sc = _score(a)
        if not np.isfinite(sc):
            return
        cands.append((sc, int(a.size), a.copy()))
        cands.sort(key=lambda x: x[0])
        if len(cands) > 30:
            cands = cands[:30]

    nonlocal_cands = cands

    if parent is not None:
        add_cand(parent)

    if not cands:
        add_cand(np.ones(2048, dtype=float) / 2048.0)

    print(f"Initial best: {cands[0][0]:.6f} (n={cands[0][1]})")
    verbose = total_budget > 600

    # Decide sizes to try based on budget
    if total_budget > 2000:
        sizes = [1024, 2048, 4096, 8192, 16384, 32768]
        fracs = [0.04, 0.06, 0.10, 0.15, 0.25, 0.40]
    elif total_budget > 1200:
        sizes = [1024, 2048, 4096, 8192, 16384]
        fracs = [0.05, 0.08, 0.12, 0.20, 0.55]
    elif total_budget > 600:
        sizes = [1024, 2048, 4096, 8192]
        fracs = [0.06, 0.10, 0.20, 0.64]
    else:
        sizes = [1024, 2048, 4096]
        fracs = [0.10, 0.25, 0.65]

    reserved = max(60.0, 0.04 * total_budget)

    # Phase 1: Multi-scale search
    for i, (sz, frac) in enumerate(zip(sizes, fracs)):
        rem = deadline - time.monotonic()
        if rem < reserved + 30:
            break

        t_alloc = min(frac * total_budget, rem - reserved - 15)
        if t_alloc < 10:
            continue

        print(f"\n=== Size {sz}, time budget {t_alloc:.0f}s ===")

        # Build starts for this size
        best_parent = cands[0][2] if cands else None
        starts = _build_diverse_starts(sz, best_parent, _GLOBAL_RNG, count=4)

        # Allocate time among starts
        per_start = t_alloc / max(1, len(starts))
        for j, sp in enumerate(starts):
            rem = deadline - time.monotonic()
            if rem < reserved + 10:
                break
            t_s = min(per_start, rem - reserved - 5)
            if t_s < 5:
                continue

            # Optimize
            p, sc = _refine_at_size(sp, sz, t_s, verbose=(j == 0 and verbose))
            if p is not None and np.isfinite(sc):
                add_cand(p, sc)
                print(f"  Start {j}: score={sc:.6f}")

        # After processing this size, pick best and optionally refine
        if cands:
            best_item = cands[0]
            print(f"  Best so far: {best_item[0]:.6f} (n={best_item[1]})")

    # Phase 2: Final refinement on best candidate
    rem = deadline - time.monotonic()
    if rem > 60 and cands:
        best_item = cands[0]
        best_arr = best_item[2]
        best_n = best_item[1]
        best_sc = best_item[0]

        print(f"\n=== Final refinement: n={best_n}, score={best_sc:.6f} ===")

        # MMD with aggressive parameters
        t1 = min(rem * 0.5, 300.0)
        if t1 > 20:
            p, sc = _mmd_optimize(best_arr, t1,
                                   beta_start=100.0, beta_end=5000.0,
                                   alpha_start=0.05, verbose=verbose)
            if p is not None and sc < best_sc:
                add_cand(p, sc)
                print(f"  MMD refine: {sc:.6f}")

        # Block CD
        rem = deadline - time.monotonic()
        if rem > 30 and cands:
            t2 = min(rem * 0.4, 200.0)
            if t2 > 10:
                p, sc = _block_cd(cands[0][2], t2, verbose=verbose)
                if p is not None and sc < cands[0][0]:
                    add_cand(p, sc)
                    print(f"  BCD refine: {sc:.6f}")

        # Another round with different beta
        rem = deadline - time.monotonic()
        if rem > 40 and cands:
            t3 = min(rem * 0.3, 150.0)
            if t3 > 10:
                # Perturb and re-optimize
                noisy = cands[0][2] * np.exp(_GLOBAL_RNG.normal(0, 0.08, cands[0][2].size))
                noisy = _sanitize(noisy)
                if noisy is not None:
                    p, sc = _mmd_optimize(noisy, t3,
                                           beta_start=200.0, beta_end=8000.0,
                                           alpha_start=0.04, verbose=verbose)
                    if p is not None and np.isfinite(sc):
                        add_cand(p, sc)
                        print(f"  Perturbed MMD: {sc:.6f}")

    # Phase 3: Try a few more restarts from best if time remains
    rem = deadline - time.monotonic()
    if rem > 80 and cands:
        for restart in range(3):
            rem = deadline - time.monotonic()
            if rem < 40:
                break
            t_r = min(rem * 0.3, 100.0)
            if t_r < 10:
                break
            # Noisy restart
            noise_std = 0.05 + 0.05 * restart
            noisy = cands[0][2] * np.exp(_GLOBAL_RNG.normal(0, noise_std, cands[0][2].size))
            noisy = _sanitize(noisy)
            if noisy is not None:
                p, sc = _mmd_optimize(noisy, t_r,
                                       beta_start=50.0 + 50 * restart,
                                       beta_end=3000.0 + 1000 * restart,
                                       alpha_start=0.08,
                                       verbose=False)
                if p is not None and np.isfinite(sc):
                    add_cand(p, sc)
                    print(f"  Restart {restart}: {sc:.6f}")

    # Verify with official evaluator for small sequences
    rem = deadline - time.monotonic()
    if rem > 10 and cands:
        for item in cands[:5]:
            arr = item[2]
            if arr.size <= 8192:
                lst = [float(x) for x in arr]
                try:
                    sc = float(evaluate_sequence(lst))
                    if np.isfinite(sc) and sc < item[0]:
                        print(f"  Official eval: {sc:.6f} < {item[0]:.6f}, updating")
                        cands[0] = (sc, item[1], arr.copy())
                        cands.sort(key=lambda x: x[0])
                        break
                except Exception:
                    pass

    # Final output
    if not cands:
        out = np.ones(2048, dtype=float) / 2048.0
        print("FALLBACK: returning uniform")
        return [float(x) for x in out]

    final = cands[0]
    final_arr = final[2]
    final_score = final[0]
    print(f"\n=== FINAL: score={final_score:.6f}, n={final_arr.size} ===")

    out = np.asarray(final_arr, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s <= 0:
        out = np.ones(2048, dtype=float)
    else:
        out = out / s

    elapsed = time.monotonic() - t_start
    print(f"Total time: {elapsed:.1f}s / {budget}s")

    return [float(x) for x in out]
