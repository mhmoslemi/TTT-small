def fast_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient via cross-correlation
    rev_a = a[::-1]

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, use_topk=False, topk_k=5):
    """Run L-BFGS-B minimization."""

Hybrid optimizer for minimizing 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Combines L-BFGS-B with LSE-smoothed objective, LP-based direction finding (CVXPY),
coordinate descent, and multi-length exploration to escape local minima.

top-k objective variant, and simulated annealing with exact score.

import numpy as np
from scipy.optimize import minimize
import time
try:
    import cvxpy as cp
    HAVE_CVXPY = True

    HAVE_CVXPY = False


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_conv_cross(a, b):
    """Linear cross-convolution a*b via FFT. Returns array of length len(a)+len(b)-1."""
    na = len(a)
    nb = len(b)
    M = na + nb - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient. Returns (score, grad)."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def topk_score_grad(a, k):
    """Soft top-k convolution score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_sorted = np.sort(c)[::-1]
    if k < len(c_sorted):
        threshold = c_sorted[k - 1]
        gap = max(c_sorted[k - 1] - c_sorted[min(k + 2, len(c_sorted) - 1)], 1e-8)
        width = gap * 0.5 + 1e-8
        w = 1.0 / (1.0 + np.exp(-8.0 * (c - threshold) / width))
    else:
        w = np.ones(len(c))
    w = w / w.sum()
    L = float(np.dot(w, c))
    score = 2.0 * n * L / (S * S)
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(w, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

def run_lbfgs(a0, beta, maxiter=5000, maxfun=100000, use_topk=False, topk_k=5):
    """Run L-BFGS-B minimization with LSE or top-k objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    if use_topk:
        res = minimize(
            lambda x: topk_score_grad(x, topk_k), a0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    else:
        res = minimize(
            lambda x: lse_score_grad(x, beta), a0,
            method='L-BFGS-B', jac=True, bounds=bounds,
            options={'ftol': 1e-15, 'gtol': 1e-12,
                     'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
        )
    a_out = np.clip(res.x, 0.0, 1000.0)
    if a_out.sum() < 1e-6:
        a_out = a0.copy()
    exact_sc = compute_score(a_out)
    return exact_sc, a_out, res


def solve_lp_direction(a, max_wall_s=30.0):
    """
    Solve the LP: max sum(b) s.t. b>=0, (a*b)[k] <= M for all k.
    Returns the optimal b (rescaled to have same sum as a), or None if infeasible/slow.
    """
    if not HAVE_CVXPY:
        return None
    n = len(a)
    t0 = time.monotonic()
    if time.monotonic() - t0 > max_wall_s:
        return None
    M = float(np.max(fast_conv(a)))
    if M < 1e-12:
        return a.copy()
    # Build constraint matrix A_conv: (2n-1) x n where (A_conv @ b)[k] = (a*b)[k]
    # A_conv[k, j] = a[k-j] if 0<=k-j<n, else 0
    # This is a banded matrix. For efficiency, we'll use a sparse-like approach.
    # Actually, for n up to ~4000, let's just build it directly.
    if n > 5000:
        return None  # Too large for dense LP
    A_mat = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = min(j + n, 2 * n - 1)
        A_mat[k_start:k_end, j] = a[:k_end - k_start]
    b_var = cp.Variable(n, nonneg=True)
    constraints = [A_mat @ b_var <= M]
    objective = cp.Maximize(cp.sum(b_var))
    prob = cp.Problem(objective, constraints)
    try:
        t_check = time.monotonic()
        if time.monotonic() - t0 > max_wall_s * 0.3:
            prob.solve(solver=cp.PDLP, max_time=max_wall_s - (time.monotonic() - t0))
        else:
            prob.solve(solver=cp.PDLP, max_time=max_wall_s)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, max_time=max_wall_s)
        except Exception:
            return None
    if time.monotonic() - t0 > max_wall_s:
        return None
    if b_var.value is None:
        return None
    b_opt = np.clip(b_var.value, 0.0, 1000.0)
    if b_opt.sum() < 1e-10:
        return None
    # Rescale to match sum of a
    target_sum = float(a.sum())
    b_opt = b_opt * (target_sum / b_opt.sum())
    return b_opt


def lp_step(a, t=0.3):
    """
    Perform one LP-based direction step:
    1. Solve LP to find best b
    2. Update a <- (1-t)*a + t*b
    Returns (new_score, new_a) or (old_score, a) if LP fails.
    """
    old_sc = compute_score(a)
    b_opt = solve_lp_direction(a, max_wall_s=20.0)
    if b_opt is None:
        return old_sc, a
    # Try different step sizes and pick the best
    best_sc = old_sc
    best_a = a
    for t in [0.1, 0.2, 0.3, 0.5, 0.7]:
        a_new = (1.0 - t) * a + t * b_opt
        a_new = np.clip(a_new, 0.0, 1000.0)
        if a_new.sum() < 0.01:
            continue
        sc = compute_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best_a = a_new
    return best_sc, best_a


def build_inits(n, rng, parent):
    """Generate diverse initializations of length n."""
    inits = []
    x = np.arange(n, dtype=np.float64) / n

    if parent is not None:
        if len(parent) == n:
            inits.append(parent.astype(np.float64).copy())
        else:
            idx = np.linspace(0, len(parent) - 1, n).astype(int)
            inits.append(parent[idx].astype(np.float64).copy())

    # Structured initializations
    inits.append(np.ones(n) * 0.5)
    inits.append(0.5 + 0.3 * np.cos(2.0 * np.pi * x))
    inits.append(0.5 + 0.3 * np.sin(2.0 * np.pi * x))
    inits.append(0.5 + 0.4 * np.cos(4.0 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(np.pi * x) ** 2)
    inits.append(0.1 + 0.9 * (1.0 - (2.0 * x - 1.0) ** 4))
    inits.append(0.5 * (1.0 + np.cos(2.0 * np.pi * x)))
    # Sinc-like
    z = 2.0 * np.pi * x
    inits.append(np.clip(0.5 + 0.3 * np.sinc(3.0 * (x - 0.5)), 0.01, 1000.0))
    # Power law
    inits.append(np.clip(0.5 + 0.3 * x ** 0.5 * (1 - x) ** 0.5 * 4, 0.01, 1000.0))
    # Random
    inits.append(np.clip(0.5 + 0.2 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.01, 1000.0))
    inits.append(np.clip(0.3 + 0.4 * rng.random(n), 0.01, 1000.0))
    # Block structure
    for nb in (8, 16, 32):
        per = max(1, n // nb)
        vals = 0.2 + 0.8 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(a[:n].copy())
    # Alternating
    a_alt = np.ones(n)
    a_alt[0::2] = 1.0
    a_alt[1::2] = 0.3
    inits.append(a_alt * 0.5)
    return inits


def exact_subgrad_polish(a, total_iters=500):
    """Projected subgradient descent on exact max-conv objective."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    sc_lb = 1.0
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Subgradient of max(c) w.r.t. a
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.3)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        # Additional small step for stability
        a -= 0.001 * grad / (gn + 1e-15)
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a


def targeted_perturb(a, rng, strength):
    """Perturb elements that contribute most to the peak convolution lag."""
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    top_count = max(3, n // 50)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:max(top_count, n // 4)]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)


def block_coordinate_descent(a, n_blocks, max_wall_s, rng):
    """Optimize one block of coordinates at a time using L-BFGS-B."""
    n = len(a)
    a = a.astype(np.float64).copy()
    block_size = max(4, n // n_blocks)
    t_start = time.monotonic()
    for it in range(100):
        if time.monotonic() - t_start > max_wall_s:
            break
        for b in range(n_blocks):
            if time.monotonic() - t_start > max_wall_s:
                break
            start = b * block_size
            end = min(start + block_size, n)
            if end - start < 4:
                continue
            a0_block = a[start:end].copy()
            if a0_block.sum() < 1e-10:
                a0_block[:] = 0.1
            # Optimize just this block
            def block_obj(x):
                a_tmp = a.copy()
                a_tmp[start:end] = np.clip(x, 0, 1000)
                return compute_score(a_tmp)
            # Use a few random perturbations + simple hill climbing
            best_block = a0_block.copy()
            best_sc = block_obj(a0_block)
            for trial in range(5):
                scale = 0.1 + 0.2 * rng.random()
                perturbed = np.clip(a0_block * (1.0 + scale * rng.standard_normal(end - start)), 0, 1000)
                sc = block_obj(perturbed)
                if sc < best_sc:
                    best_sc = sc
                    best_block = perturbed
            a[start:end] = best_block
    return a


def simulated_annealing(a, n_iters, max_wall_s, rng):
    """Simulated annealing with exact score evaluation."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    cur_sc = best_sc
    t0 = time.monotonic()
    t_start = best_sc - 1.0 + 1e-3  # initial temperature in score units
    t_end = 1e-5
    for it in range(n_iters):
        if time.monotonic() - t0 > max_wall_s:
            break
        T = t_start * (t_end / t_start) ** (it / max(n_iters - 1, 1))
        n_pert = max(5, n // 80)
        idx = rng.integers(0, n, size=n_pert)
        a_new = a.copy()
        a_new[idx] *= (1.0 + 0.15 * rng.standard_normal(n_pert))
        a_new = np.clip(a_new, 0.0, 1000.0)
        if a_new.sum() < 0.01:
            a_new = a.copy()
            continue
        sc_new = compute_score(a_new)
        if sc_new < cur_sc or rng.random() < np.exp(-(sc_new - cur_sc) / max(T, 1e-12)):
            a = a_new
            cur_sc = sc_new
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
    return best_sc, best_a

            a_lp = a_pol
    else:
        print("  Skipping LP (no CVXPY or sequence too long)")
        # Alternative: use top-k objective for some iterations
        a_tk = best_a.copy()
        for it_tk in range(10):
            if time.monotonic() > phase2_deadline:
                break
            sc, a_opt, _ = run_lbfgs(a_tk, 200, maxiter=3000, maxfun=6000, use_topk=True, topk_k=5)
            report(sc, a_opt, f"topk {it_tk}")
            a_tk = a_opt

    print(f"\nAfter Phase 2: best = {best_score:.8f}")

    # =========================================================================
    # Phase 3: Local search with diverse perturbations (15% of time)
    # =========================================================================
    phase3_deadline = t_start + 0.90 * (total_budget - 15.0)
    print(f"\n{'='*60}")
    print(f"Phase 3: Local Search (deadline at {phase3_deadline - t_start:.0f}s)")
    print(f"{'='*60}")

    n_best = len(best_a)
    trial = 0
    while time.monotonic() < phase3_deadline and trial < 50:
        strategy = trial % 6
        if strategy == 0:
            scale = 0.02 + 0.08 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            a_pert = targeted_perturb(best_a, rng, strength=0.03 + 0.07 * rng.random())
        elif strategy == 2:
            freq = 1.0 + 8.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.02 + 0.06 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 3:
            # Block swap
            block_sz = max(10, n_best // 20)
            b1 = int(rng.integers(0, n_best - 2 * block_sz))
            b2 = int(rng.integers(0, n_best - 2 * block_sz))
            a_pert = best_a.copy()
            a_pert[b1:b1 + block_sz], a_pert[b2:b2 + block_sz] = \
                a_pert[b2:b2 + block_sz].copy(), a_pert[b1:b1 + block_sz].copy()
        elif strategy == 4:
            # Sparse perturbation
            n_sparse = max(10, n_best // 50)
            a_pert = best_a.copy()
            idx = rng.integers(0, n_best, size=n_sparse)
            a_pert[idx] *= (1.0 + 0.3 * rng.standard_normal(n_sparse))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Mix with a random structured sequence
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_struct = 0.5 + 0.3 * np.cos(2.0 * np.pi * rng.random() * 5 * x + rng.random() * 6.28)
            a_pert = (1 - 0.3) * best_a + 0.3 * a_struct * (best_a.sum() / a_struct.sum())
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        # Polish with L-BFGS-B
        sc, a_opt, _ = run_lbfgs(a_pert, 800, maxiter=3000, maxfun=5000)
        report(sc, a_opt, f"local #{trial} (strat={strategy})")
        trial += 1

    print(f"\nAfter Phase 3: best = {best_score:.8f}")

    # =========================================================================
    # Phase 4: Final refinement with high-beta LSE and subgradient (10% of time)
    # =========================================================================
    print(f"\n{'='*60}")
    print(f"Phase 4: Final Refinement")
    print(f"{'='*60}")

    # High-beta LSE polish
    for beta in [2000, 5000]:
        if time.monotonic() > deadline - 10.0:
            break
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=10000, maxfun=200000)
        report(sc, a_opt, f"final LBFGS beta={beta}")

    # Exact subgradient polish
    if time.monotonic() < deadline - 5.0:
        sc, a_pol = exact_subgrad_polish(best_a, total_iters=1000)
        report(sc, a_pol, "subgrad polish")

    # One more L-BFGS from the polished solution
    if time.monotonic() < deadline - 3.0:
        sc, a_opt, _ = run_lbfgs(best_a, 3000, maxiter=5000, maxfun=100000)
        report(sc, a_opt, "final final LBFGS")

    # =========================================================================
    # Final verification and return
    # =========================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
