def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _fft_conv(a, L=None):
    """Full convolution of a with itself via FFT."""
    n = len(a)
    if L is None:
        L = _next_pow2(2 * n - 1)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    return np.maximum(c, 0.0)


def _score_norm(a, L=None):
    """Objective value for sum-normalized sequence: 2*n*max(conv)."""
    n = len(a)
    c = _fft_conv(a, L)
    return 2.0 * n * c.max()


def _normalize(a):
    """Clip to [0,1000] and normalize to sum=1."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = a.sum()
    if s < 1e-15 or not np.isfinite(s):
        return None
    return a / s


def _compute_gradient(a, w, L):
    """
    Compute gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a.
    Uses FFT cross-correlation: grad[j] = 2 * sum_k w[k]*a[k-j]
    w has length >= 2n-1, a has length n.
    """
    n = len(a)
    # Cross-correlation: (w star a)[j] = sum_k w[k]*a[k-j]
    g = np.fft.irfft(np.fft.rfft(w, L) * np.conj(np.fft.rfft(a, L)), L)[:n]
    return 2.0 * g
