"""
Step-function coefficient optimizer using top-k subgradient descent
with adaptive step sizes, momentum, and coordinate polish.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
import time
import math

def _normalize(v):
    v = np.asarray(v, dtype=np.float64).copy()
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = v.sum()
    if s < 1e-300:
        return np.ones(v.shape[0]) / v.shape[0]
    v /= s
    return v

def _convolve(f):
    """Full linear convolution of f with itself via FFT."""
    n = f.size
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]

def _score_from_conv(f):
    n = f.size
    c = _convolve(f)
    return float(2.0 * n * c.max()), c

def _score(f):
    n = f.size
    c = _convolve(f)
    return float(2.0 * n * c.max())

def _project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1)
    out = np.maximum(v - theta, 0.0)
    return out

def _grad_topk(f, c, k=5):
    """Compute subgradient of max(c) using top-k lags."""
    n = f.size
    L = c.size
    # Find top-k lags
    if k >= L:
        lags = np.arange(L)
        weights = np.ones(L) / L
    else:
        top_idxs = np.argpartition(c, L - k)[L - k:]
        top_vals = c[top_idxs]
        # Use softmax-like weights (tempered)
        T = max(c.max() - top_vals.min(), 1e-30)
        weights = np.exp((top_vals - c.max()) / T)
        weights /= weights.sum()
        lags = top_idxs
    # Compute gradient
    g = np.zeros(n)
    for lag, w in zip(lags, weights):
        lag = int(lag)
        j_min = max(0, lag - n + 1)
        j_max = min(n - 1, lag)
        if j_min <= j_max:
            src_start = lag - j_max
            src_end = lag - j_min
            g[j_min:j_max+1] += 2.0 * w * f[src_start:src_end+1][::-1]
    return g

def _step_with_backtrack(f, g, c, lr_init=0.1, max_iter=10):
    """Take a gradient step with backtracking line search."""
    best_f = f.copy()
    best_c = c.copy()
    best_mc = c.max()
    lr = lr_init
    for i in range(max_iter):
        # Descent direction: reduce convolution at active lags
        f_new = f - lr * g
        f_new = _project_simplex(f_new)
        c_new = _convolve(f_new)
        mc_new = c_new.max()
        if mc_new < best_mc - 1e-15:
            best_mc = mc_new
            best_f = f_new
            best_c = c_new
            return best_f, best_c, best_mc
        lr *= 0.5
    return best_f, best_c, best_mc

def _coord_polish(f, c, budget_s, rng):
    """Coordinate descent: try to improve by transferring mass between pairs."""
    f = _normalize(f.copy())
    n = f.size
    c = _convolve(f)
    maxc = float(c.max())
    best_f = f.copy()
    best_c = c.copy()
    best_mc = maxc
    
    deadline = time.monotonic() + float(budget_s)
    no_improve = 0
    
    while time.monotonic() < deadline and no_improve < 30:
        # Find top lags
        top_k = 5
        L = c.size
        top_idxs = np.argpartition(c, L - top_k)[L - top_k:]
        
        # Compute gradient from top lags
        g = _grad_topk(f, c, top_k)
        
        # Find best donor (high g) and receiver (low g)
        # Exclude coordinates with near-zero mass
        mask = f > 1e-10
        g_masked = g.copy()
        g_masked[~mask] = -np.inf
        
        donor = int(np.argmax(g_masked))
        g_masked2 = g.copy()
        g_masked2[donor] = np.inf
        receiver = int(np.argmin(g_masked2))
        
        if f[donor] < 1e-12:
            no_improve += 1
            break
        
        # Try several transfer amounts
        best_dv = None
        best_mc_try = maxc
        for frac in (0.01, 0.05, 0.1, 0.2, 0.3, 0.5):
            dv = f[donor] * frac
            f_try = f.copy()
            f_try[donor] -= dv
            f_try[receiver] += dv
            f_try = _normalize(f_try)
            c_try = _convolve(f_try)
            mc_try = float(c_try.max())
            if mc_try < best_mc_try:
                best_mc_try = mc_try
                best_dv = (donor, receiver, dv)
        
        # Also try random pairs
        for _ in range(20):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j or f[i] < 1e-12:
                continue
            for frac in (0.05, 0.2, 0.5):
                dv = f[i] * frac
                f_try = f.copy()
                f_try[i] -= dv
                f_try[j] += dv
                f_try = _normalize(f_try)
                c_try = _convolve(f_try)
                mc_try = float(c_try.max())
                if mc_try < best_mc_try:
                    best_mc_try = mc_try
                    best_dv = (i, j, dv)
        
        if best_dv is not None and best_mc_try < maxc - 1e-14:
            i, j, dv = best_dv
            f[i] -= dv
            f[j] += dv
            f = _normalize(f)
            c = _convolve(f)
            maxc = float(c.max())
            no_improve = 0
            if maxc < best_mc:
                best_mc = maxc
                best_f = f.copy()
                best_c = c.copy()
        else:
            no_improve += 1
    
    return _normalize(best_f), best_c, float(best_mc)

Actually, I realize there's a potential issue with my approach. The subgradient method for non-smooth objectives can be slow to converge. The rate is O(1/T) for T iterations, which means we need a lot of iterations for high accuracy.

A better approach might be to use a "smoothing" technique: instead of the hard max, use a soft-max (LogSumExp) which is smooth and amenable to faster optimization.

score_smooth = 2n * (1/T) * log(sum_k exp(T*c[k]))

As T -> infinity, this approaches 2n * max_k c[k]. For finite T, it's a smooth upper bound on 2n * max_k c[k].

The gradient of the smooth score is:
d/dx_j [2n/T * log(sum_k exp(T*c[k]))] = 2n/T * sum_k exp(T*c[k]) * T*dc[k]/dx_j / sum_k exp(T*c[k])
= 2n * sum_k p_k * dc[k]/dx_j
where p_k = exp(T*c[k]) / sum_l exp(T*c[l])

This is exactly the softmax-weighted gradient that the previous program was using! So the previous program's approach is essentially a smoothed max optimization.

The difference is in the temperature T (or sigma_rel in the previous program). A lower T makes the function smoother (faster gradient computation, faster convergence to the smoothed optimum) but less accurate. A higher T makes it closer to the true max (more accurate) but harder to optimize.

The previous program starts with T related to sigma_rel=0.08 and decays it. This is a reasonable annealing schedule.

"""
Step-function coefficient optimizer.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 using:
  - LogSumExp smoothing with adaptive temperature
  - Top-k subgradient descent with momentum
  - Backtracking line search
  - Coordinate polish for local improvement
  - Multi-scale (coarse-to-fine) optimization
  - Diversification perturbations

import numpy as np

def _normalize(v):
    """Project to non-negative vector summing to 1."""
    global _global_best_f  # not needed, just here
    v = np.asarray(v, dtype=np.float64).copy()
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = v.sum()
    if s < 1e-300:
        n = v.size
        return np.ones(n) / n if n > 0 else np.array([1.0])
    v /= s
    return v

def _conv(f):
    """Full linear self-convolution via FFT."""
    n = f.size
    m = 2 * n - 1
    if m <= 1:
        return np.array([float(f.dot(f))])
    size = 1 << (m - 1).bit_length()
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]

def _score(f):
    """Compute 2*n*max(conv(f,f))."""
    n = f.size
    c = _conv(f)
    return float(2.0 * n * c.max())

def _project(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.nonzero(cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1)
    out = v - theta
    np.maximum(out, 0.0, out)
    return out

def _smooth_score(f, c, T):

    z = T * (c - c.max())
    np.clip(z, -50.0, 0.0, out=z)
    s = np.exp(z).sum()
    if s < 1e-300:

    return float(2.0 * f.size / T * math.log(s))

def _interpolate(f, n_new):

    if n_new == 1:
        return np.array([1.0])
    x_old = np.linspace(0.0, n_old - 1, n_old)
    x_new = np.linspace(0.0, n_new - 1, n_new)
    return _normalize(np.interp(x_new, x_old, f))

def _optimize_smooth(f0, budget_s, T_init, T_final, rng, tag=""):
    """
    Optimize using smoothed LogSumExp objective with annealing.
    """
    f = _normalize(f0.copy())
    n = f.size
    c = _conv(f)
    
    best_f = f.copy()
    best_mc = float(c.max())
    best_score = float(2.0 * n * best_mc)
    
    deadline = time.monotonic() + float(budget_s)
    T = float(T_init)
    
    # Adaptive parameters
    lr = 0.02
    momentum = np.zeros(n)
    beta = 0.5
    it = 0
    no_improve = 0
    
    total_time = float(budget_s)
    
    while time.monotonic() < deadline:
        it += 1
        remaining = deadline - time.monotonic()
        progress = 1.0 - max(0.0, remaining) / total_time
        
        # Anneal temperature
        T = T_init * (T_final / T_init) ** progress
        T = max(T, T_final)
        
        # Recompute convolution if needed (should be current)
        mc = float(c.max())
        
        # Compute gradient
        g = _smooth_grad(f, c, T)
        
        # Normalize gradient for stable steps
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-300:
            no_improve += 10
            continue
        g = g / g_norm
        
        # Momentum
        momentum = beta * momentum + (1.0 - beta) * g
        mom_norm = np.linalg.norm(momentum)
        if mom_norm < 1e-300:
            continue
        direction = momentum / mom_norm
        
        # Line search
        accepted = False
        for trial in range(8):
            step = lr * (0.5 ** trial)
            f_new = _project(f + step * (-direction))  # descent
            c_new = _conv(f_new)
            mc_new = float(c_new.max())
            if mc_new < mc - 1e-14:
                f = f_new
                c = c_new
                mc = mc_new
                accepted = True
                break
        
        if accepted:
            no_improve = 0
            beta = min(beta + 0.001, 0.85)
            lr = min(lr * 1.02, 0.3)
            cur_score = float(2.0 * n * mc)
            if cur_score < best_score:
                best_score = cur_score
                best_f = f.copy()
                if best_score < 1.51:
                    print(f"[{tag}] iter={it} T={T:.4f} score={best_score:.8f}")
        else:
            no_improve += 1
            lr *= 0.7
            if no_improve > 30:
                # Perturb
                alpha = float(rng.uniform(0.05, 0.2))
                noise = rng.random(n)
                noise /= noise.sum()
                f = _normalize((1 - alpha) * f + alpha * noise)
                c = _conv(f)
                mc = float(c.max())
                lr = 0.02
                no_improve = 0
                beta = 0.5
        
    return _normalize(best_f), float(2.0 * n * _conv(best_f).max())

def _coord_descent(f, c, budget_s, rng, tag=""):
    """Coordinate descent polish."""
    f = _normalize(f.copy())
    n = f.size
    c = _conv(f)
    maxc = float(c.max())
    best_f = f.copy()
    best_c = c.copy()
    best_mc = maxc
    
    deadline = time.monotonic() + float(budget_s)
    no_improve = 0
    
    print(f"[{tag}] coord descent start: maxc={maxc:.10f}")
    
    while time.monotonic() < deadline and no_improve < 50:
        # Get gradient from top lags
        L = c.size
        top_k = min(10, L)
        top_idxs = np.argpartition(c, L - top_k)[L - top_k:]
        
        # Compute top-k subgradient
        top_vals = c[top_idxs]
        T_local = max(top_vals.max() - top_vals.min(), 1e-10)
        weights = np.exp((top_vals - top_vals.max()) / T_local)
        weights /= weights.sum()
        g = np.zeros(n)
        for lag, w in zip(top_idxs, weights):
            lag = int(lag)
            j_min = max(0, lag - n + 1)
            j_max = min(n - 1, lag)
            if j_min <= j_max:
                src_start = lag - j_max
                src_end = lag - j_min
                g[j_min:j_max+1] += 2.0 * w * f[src_start:src_end+1][::-1]
        
        # Find best transfer
        best_mc_try = maxc
        best_move = None
        
        # Gradient-guided transfers
        donor_candidates = np.argsort(g)[::-1][:5]
        recv_candidates = np.argsort(g)[:5]
        
        for i in donor_candidates:
            i = int(i)
            if f[i] < 1e-12:
                continue
            for j in recv_candidates:
                j = int(j)
