Gradient-based optimization for minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses smooth max (log-sum-exp) with adaptive temperature, combined with
a tail penalty to flatten the convolution. Includes local coordinate
descent refinement and multi-scale search.
"""

import numpy as np
import time

def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_len(n):
    need = 2 * int(n) - 1
    return _next_pow2(need)

def _score(a):
    """Compute the objective value for a probability vector."""
    n = len(a)
    L = _conv_len(n)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2*n-1]
    c = np.maximum(c, 0.0)
    return 2.0 * n * c.max(), c

def _smooth_max_obj(c, T):
    """Compute T * log(sum(exp(c/T))) and the softmax weights."""
    c_shift = c / T - np.max(c) / T
    exp_c = np.exp(c_shift)
    log_sum = T * (np.log(exp_c.sum()) + np.max(c) / T)
    w = exp_c / exp_c.sum()
    return log_sum, w

def _cross_corr(x, y, N):
    """Compute linear cross-correlation: out[k] = sum_i x[i]*y[i+k]."""
    X = np.fft.rfft(x, N)
    Y = np.fft.rfft(y, N)
    return np.fft.irfft(np.conj(X) * Y, N)

def _gradient_smooth_max(a, c, w, N):
    """Compute gradient of smooth max objective."""
    # grad_j = 2 * sum_k w[k] * a[k-j] = 2 * (a cross-corr w)[j]
    # Wait, I need to be careful about the direction.
    # We derived: grad_j = 2 * sum_k w_k * a_{k-j}
    # This is 2 * (a ⋄ w)(-j) ... or is it (a ⋄ w)(j)?
    # 
    # Let me recheck: sum_k w_k * a_{k-j}
    # Let m = k-j: sum_m w_{m+j} * a_m = sum_m a_m * w_{m+j}
    # This is (a ⋄ w)(j) = cross-correlation of a and w at lag j.
    # 
    # Using FFT: IFFT(conj(FFT(a)) * FFT(w))[j] = sum_m a[m] * w[m+j] ✓
    corr = _cross_corr(a, w, N)
    return 2.0 * corr[:len(a)]

def _gradient_penalty(a, c, M, N):
    """Compute gradient of sum_k (c_k - M)_+^2."""
    v = np.maximum(0.0, c - M)
    if v.max() < 1e-15:
        return np.zeros(len(a))
    corr = _cross_corr(a, v, N)
    return 4.0 * corr[:len(a)]

    a = a_init.copy()
    a = np.maximum(a, 0)
    a = a / a.sum()
    
    n = len(a)
    L = _conv_len(n)
    N = _next_pow2(n + 2*n)  # for cross-correlation
    
    best_a = a.copy()
    best_score, best_c = _score(a)
    cur_score = best_score
    
    T = T_init
    lr = lr_init
    M_target = best_score / (2.0 * n) * 0.95  # start slightly below current max
    penalty_weight = 0.1
    
    deadline = time.monotonic() + time_limit
    it = 0
    stall = 0
    last_print = 0
    
    while time.monotonic() < deadline:
        # Compute convolution
        F = np.fft.rfft(a, L)
        c = np.fft.irfft(F * F, L)[:2*n-1]
        c = np.maximum(c, 0.0)
        cur_score = 2.0 * n * c.max()
        
        if cur_score < best_score - 1e-10:
            best_score = cur_score
            best_a = a.copy()
            stall = 0

                print(f"  it={it} score={best_score:.7f} T={T:.5f} lr={lr:.5f}")
                last_print = it
        else:
            stall += 1
        
        # Compute smooth max and gradient
        T_eff = max(1e-10, T)
        _, w = _smooth_max_obj(c, T_eff)
        grad_sm = _gradient_smooth_max(a, c, w, N)

        a_new = np.maximum(a_new, 1e-16)

Actually, `(a*a)[k] = sum_i a_i * a_{k-i}` is NOT convex in general. It's a quadratic form `a^T B_k a` where `B_k` is a matrix. For this to be convex, `B_k` must be positive semi-definite.

`B_k[j,l] = 1/2 * (delta_{j+l,k} + delta_{j+l,k})` ... hmm, let me think.

`(a*a)[k] = sum_i a_i * a_{k-i} = a^T B_k a` where `B_k` has entries `B_k[i, k-i] = 1` (and `B_k[k-i, i] = 1` to make it symmetric, with a factor of 1/2 to compensate).

Actually: `a^T B_k a = sum_{i,j} a_i B_k[i,j] a_j`. We want this to equal `sum_i a_i * a_{k-i}`.

So `B_k[i,j] = 1` if `i+j = k`, and `B_k[i,j] = 0` otherwise. But this makes `B_k` not symmetric in the usual sense... wait, `B_k[i,j] = 1` if `i+j=k` means `B_k[j,i] = 1` if `j+i=k`, which is the same. So `B_k` IS symmetric.

Is `B_k` positive semi-definite? `B_k` is the adjacency matrix of a "path graph" restricted to the anti-diagonal. For `k = 0`: `B_0 = [1, 0, ..., 0; 0, 0, ...; ...]` which is PSD (it's a rank-1 matrix with eigenvalue 1).

For `k = 1`: `B_1` has 1s at positions `(0,1)` and `(1,0)`. The eigenvalues are 1 and -1 (for the 2x2 block `[[0,1],[1,0]]`). So NOT PSD!

This means the constraint `(a*a)[k] <= M` is NOT convex in general. So we can't use standard convex optimization.

However, if we restrict to `k = 0`: `(a*a)[0] = a[0]^2 <= M` is convex (just a box constraint on `a[0]`).

"""
Gradient-based optimization for minimizing the step function objective.
Uses log-sum-exp smooth max with adaptive temperature, combined with
tail penalty and local coordinate descent.
"""

import numpy as np
import time

# Global RNG (will be reset in propose_candidate)
_rng = np.random.default_rng(12345)

def _npow2(x):
    """Next power of 2 >= x."""
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _fft_len(n):
    """FFT length for convolution of two length-n sequences."""
    return _npow2(2 * n - 1)

def _xcorr_len(n):
    """FFT length for cross-correlation of length-n and length-(2n-1) sequences."""
    return _npow2(3 * n - 2)

def _sanitize(a):
    """Ensure a is a valid non-negative array."""
    a = np.asarray(a, dtype=float).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones(len(a)) / len(a)
    return a / s

def _score_and_conv(a):
    """Compute objective score and convolution."""
    n = len(a)
    L = _fft_len(n)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2*n-1]
    c = np.maximum(c, 0.0)
    score = 2.0 * n * c.max()
    return score, c

def _cross_corr(a, b):
    """Linear cross-correlation: out[k] = sum_i a[i]*b[i+k]."""
    n = len(a)
    m = len(b)
    N = _npow2(n + m - 1)
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    return np.fft.irfft(np.conj(A) * B, N)

def _grad_smooth_max(a, c, T):
    """Gradient of T*log(sum(exp(c/T))) w.r.t. a."""
    c_max = c.max()
    exp_c = np.exp((c - c_max) / T)
    w = exp_c / exp_c.sum()
    corr = _cross_corr(a, w)
    return 2.0 * corr[:len(a)]

def _grad_penalized(a, c, M):
    """Gradient of sum_k (c_k - M)_+^2 w.r.t. a."""
    v = np.maximum(0.0, c - M)
    if v.max() < 1e-15:
        return np.zeros(len(a))
    corr = _cross_corr(a, v)
    return 4.0 * corr[:len(a)]

def _optimize(a_init, time_limit, T0=0.005, lr0=0.3, verbose=True):
    """Main gradient optimizer with line search and adaptive parameters."""
    a = _sanitize(a_init)
    n = len(a)
    
    best_a = a.copy()
    best_score, best_c = _score_and_conv(a)
    
    T = T0
    lr = lr0
    pen_weight = 0.5
    M_target = best_score / (2.0 * n) * 0.99  # slightly below current max
    
    deadline = time.monotonic() + time_limit
    it = 0
    stall = 0
    last_best_it = 0
    
    while time.monotonic() < deadline:
        # Compute current state
        _, c = _score_and_conv(a)
        cur_score = 2.0 * n * c.max()
        c_max = c.max()
        
        # Compute gradient
        grad_sm = _grad_smooth_max(a, c, T)
        grad_pen = _grad_penalized(a, c, M_target)
        grad = grad_sm + pen_weight * grad_pen
        
        # Center gradient
        grad = grad - grad.mean()
        
        # Line search: try several learning rates
        best_lr = lr
        best_a_new = a
        best_sc_new = cur_score
        
        for lr_try in [lr * 0.25, lr * 0.5, lr, lr * 2.0, lr * 4.0]:
            lr_try = min(lr_try, 2.0)
            a_try = a * np.exp(-lr_try * grad)
            a_try = np.maximum(a_try, 1e-15)
            s = a_try.sum()
            if s < 1e-10:
                continue
            a_try = a_try / s
            
            sc_try, _ = _score_and_conv(a_try)
            if sc_try < best_sc_new:
                best_sc_new = sc_try
                best_a_new = a_try
                best_lr = lr_try
        
        if best_sc_new < cur_score - 1e-12:
            a = best_a_new
            lr = best_lr
            stall = 0
            if best_sc_new < best_score:
                best_score = best_sc_new
                best_a = a.copy()
                last_best_it = it
                if verbose and it % 200 == 0:
                    print(f"    it={it} best={best_score:.7f} T={T:.6f} lr={lr:.4f}")
        else:
            stall += 1
            lr = max(0.01, lr * 0.8)

            T = max(1e-7, T * 0.97)
            M_target = max(c_max * 0.8, M_target * 0.995)

def _coord_descent(a_init, time_limit, verbose=True):
    """Coordinate descent with scaling."""
    a = _sanitize(a_init)
    n = len(a)
    best_a = a.copy()
    best_score, _ = _score_and_conv(a)
    
    deadline = time.monotonic() + time_limit
    factors = [0.5, 0.7, 0.85, 0.95, 1.05, 1.15, 1.3, 1.5, 2.0]
    stall = 0
    
    while time.monotonic() < deadline:
        improved = False
        # Random order of coordinates
        order = _rng.permutation(n)
        
        for i in order:
            if time.monotonic() > deadline:
                break
            
            for f in factors:
                a_try = a.copy()
                a_try[i] *= f
                a_try = _sanitize(a_try)
                sc, _ = _score_and_conv(a_try)
                if sc < best_score - 1e-12:
                    a = a_try
                    best_score = sc
                    best_a = a.copy()
                    improved = True
                    break
        
        if not improved:
            stall += 1
            if stall > 10:
                # Try block scaling
                block_len = _rng.integers(max(2, n//100), max(3, n//10))
                start = _rng.integers(0, n - block_len)
                for f in factors:
                    a_try = a.copy()
                    a_try[start:start+block_len] *= f
                    a_try = _sanitize(a_try)
                    sc, _ = _score_and_conv(a_try)
                    if sc < best_score - 1e-12:
                        a = a_try
                        best_score = sc
                        best_a = a.copy()
                        improved = True
                        break
                if not improved and stall > 50:
                    # Reset
                    a = best_a.copy()
                    stall = 0
        else:
            stall = 0
    
    return best_a, best_score
