"""
Hybrid optimization: LSE-smoothed L-BFGS + exact cyclic coordinate descent + mirror descent.
Minimizes 2n * max(self-convolution) / (sum)^2 over non-negative sequences.

Strategy:
  1. LSE continuation with L-BFGS-B (beta: 50 -> 20000)
  2. Cyclic coordinate descent on the exact max-conv objective
  3. High-beta LSE refinement
  4. Mirror descent (multiplicative updates) for final polish
"""

import numpy as np
from scipy.optimize import minimize
import time


def fast_conv(a):
    """Linear self-convolution via FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """LSE-smoothed score and gradient."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    # Clip for numerical stability
    scaled = np.clip(scaled, -700, 700)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: cross-correlation of a_rev and p
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad


def lse_objective(a, beta):
    return lse_score_grad(a, beta)


def run_lbfgs(a0, beta, maxiter=3000, maxfun=50000):
    """Run L-BFGS-B on LSE objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a0.sum() < 1e-6:
        a0[:] = 0.5
    bounds = [(0.0, 1000.0)] * n
    res = minimize(
        lse_objective, a0, args=(beta,),
        method='L-BFGS-B', jac=True, bounds=bounds,
        options={'ftol': 1e-16, 'gtol': 1e-14,
                 'maxiter': maxiter, 'maxfun': maxfun, 'maxls': 50}
    )
    a_out = np.clip(res.x, 0.0, 1000.0)
    exact_sc = compute_score(a_out)
    return exact_sc, a_out


def coord_descent_sweep(a, c, order, max_inner=15):
    """
    One sweep of cyclic coordinate descent.
    a: current sequence
    c: current convolution (length 2n-1)
    order: permutation of indices
    Returns updated (a, c, peak).
    """
    n = len(a)
    for i in order:
        x = a[i]
        if x < 1e-15:
            # Try setting to small positive
            # Compute contribution if we set a[i] = eps
            # conv change: for each k, delta = 2*a[k-i]*eps (approx)
            # Just skip for now unless it helps
            continue
        
        # Compute D and E vectors
        # D[k] = coefficient of x in conv[k]
        # E[k] = coefficient of x^2 in conv[k]
        k_mi = np.arange(2 * n - 1) - i
        valid = (k_mi >= 0) & (k_mi < n)
        
        D = np.zeros(2 * n - 1)
        D[valid] = a[k_mi[valid]]  # from j=i term
        valid2 = valid & (k_mi != i)  # from j=k-i term (different from i)
        D[valid2] += a[k_mi[valid2]]
        
        E = np.zeros(2 * n - 1)
        if 0 <= 2 * i <= 2 * n - 2:
            E[2 * i] = 1.0
        
        # C[k] = c[k] - D[k]*x - E[k]*x^2
        # f(x_new) = max_k (C[k] + D[k]*x_new + E[k]*x_new^2)
        
        # Find optimal x_new
        # Start with current peak
        k_star = int(np.argmax(c))
        
        # Suggested x from current peak
        if E[k_star] > 0:
            x_sugg = -D[k_star] / (2.0 * E[k_star])
        elif D[k_star] < 0:
            # Peak decreases with x, try increasing
            x_sugg = x * 1.5
        else:
            x_sugg = x * 0.7  # or 0
        
        x_sugg = max(0.0, min(1000.0, x_sugg))
        
        # Evaluate a few candidates
        candidates = [0.0, x, x_sugg]
        if x > 0:
            candidates.append(x * 0.5)
            if x_sugg > 0:
                candidates.append((x + x_sugg) * 0.5)
        
        C = c - D * x - E * x * x
        best_x = x
        best_f = float(np.max(C + D * x + E * x * x))  # = max(c)
        
        for xc in candidates:
            fc = float(np.max(C + D * xc + E * xc * xc))
            if fc < best_f - 1e-12:
                best_f = fc
                best_x = xc
        
        # Iterate to refine
        if best_x > 1e-10:
            for _ in range(max_inner):
                fvals = C + D * best_x + E * best_x * best_x
                k_new = int(np.argmax(fvals))
                if E[k_new] > 0:
                    new_x = -D[k_new] / (2.0 * E[k_new])
                elif D[k_new] < 0:
                    new_x = best_x * 1.05
                else:
                    new_x = best_x * 0.95
                new_x = max(0.0, min(1000.0, new_x))
                if abs(new_x - best_x) / (best_x + 1e-10) < 1e-8:
                    break
                best_x = new_x
                # Check improvement
                fc = float(np.max(C + D * best_x + E * best_x * best_x))
                if fc >= best_f + 1e-10:
                    break
                best_f = fc
        
        if abs(best_x - x) > 1e-12:
            # Update c incrementally
            c += D * (best_x - x) + E * (best_x * best_x - x * x)
            a[i] = best_x
    
    return a, c, float(np.max(c))


def coord_descent(a, n_sweeps=10, seed=0):
    """Run multiple sweeps of coordinate descent."""
    n = len(a)
    c = fast_conv(a)
    rng = np.random.default_rng(seed)
    
    for sweep in range(n_sweeps):
        order = rng.permutation(n)
        a, c, peak = coord_descent_sweep(a, c, order)
        S = float(a.sum())
        score = 2.0 * n * peak / (S * S)
        if sweep % 5 == 0:
            print(f"    Coord descent sweep {sweep}: score={score:.8f}")
    
    return a


def mirror_descent(a, n_iters=500, eta0=0.01):
    """
    Multiplicative (mirror) descent on the exact objective.
    Update: a[j] <- a[j] * exp(-eta * grad_j)
    Then re-normalize (optional, since objective is scale-invariant).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_sc = compute_score(a)
    best_a = a.copy()
    eta = eta0
    
    for it in range(n_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        # Subgradient of the exact objective
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg[j_start:j_end + 1] = 2.0 * seg[::-1]
        
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg / S2 - 2.0 * M / S3)
        
        # Multiplicative update
        a = a * np.exp(-eta * grad)
        a = np.clip(a, 1e-10, 1000.0)
        
        # Decay step size
        eta *= 0.995
        
        if it % 100 == 0:
            print(f"    Mirror descent iter {it}: score={sc:.8f}, eta={eta:.6f}")
    
    return best_a
