Step-function optimizer: minimizes 2*n*max(a*a) / (sum a)^2 
using log-domain gradient descent with hinge loss weighting,
exact coordinate descent, and spectral shaping kicks.
Multi-scale coarse-to-fine strategy.

import numpy as np
import time
import math

# --- Core utilities ---

def _normalize(v):
    v = np.asarray(v, dtype=np.float64).copy()
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = v.sum()
    if s < 1e-300:
        return np.ones(v.size, dtype=np.float64) / max(1, v.size)
    return v / s

def _fft_size(m):
    m = max(2, int(m))
    return 1 << (m - 1).bit_length()

def _autocorr(f):
    n = f.size
    size = _fft_size(2 * n - 1)
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]

def _score(f, c=None):
    if c is None:
        c = _autocorr(f)
    return 2.0 * f.size * float(c.max()) if c.size else np.inf

def _weighted_gradient(p, f):
    n = f.size
    L = p.size
    size = _fft_size(L + n - 1)
    P = np.fft.rfft(p, size)
    R = np.fft.rfft(f[::-1], size)
    return 2.0 * np.fft.irfft(P * R, size)[n - 1 : 2 * n - 1]

def _interp(f, n):
    f = _normalize(f)
    if f.size == n:
        return f
    if n == 1:
        return np.array([1.0])
    x0 = np.linspace(0, f.size - 1, f.size)
    x1 = np.linspace(0, n - 1, n)
    return _normalize(np.interp(x1, x0, f))

def _transfer_max(c, f, i, j, dv):
    n = f.size
    L = 2 * n - 1
    c_new = c.copy()
    lo = j; hi = min(L, j + n)
    if lo < hi:
        c_new[lo:hi] += 2.0 * dv * f[:hi - lo]
    lo = i; hi = min(L, i + n)
    if lo < hi:
        c_new[lo:hi] -= 2.0 * dv * f[:hi - lo]
    q = dv * dv
    if 2 * j < L: c_new[2 * j] += q
    if 2 * i < L: c_new[2 * i] += q
    if i + j < L: c_new[i + j] -= 2.0 * q
    return float(c_new.max())

    size = _fft_size(2 * n - 1)
    F = np.fft.rfft(f, size)

    # Target geometric mean of non-DC magnitudes
    nonzero = absF[1:]
    active = nonzero[nonzero > nonzero.max() * 1e-8]
    if active.size < 2:
        return f
    log_target = float(np.log(active).mean())
    target = np.exp(log_target)
    ratio = target / (absF + 1e-300)
    new_absF = absF * np.power(ratio, strength)
    new_absF[0] = absF[0]
    phase = np.exp(1j * np.angle(F))
    f_new = np.fft.irfft(new_absF * phase, size)[:n]
    return _normalize(np.maximum(f_new, 0.0))

                if mc < best_mc:
                    best_mc = mc
                    best_move = (i, j, dv)
        
        if best_move is None or best_mc >= maxc - 1e-15:
            no_imp += 1
            continue
        
        i, j, dv = best_move
        f[i] -= dv
        f[j] += dv
        f = _normalize(f)
        c = _autocorr(f)
        maxc = float(c.max())
        sc = 2.0 * n * maxc
        
        if sc < best_score - 1e-15:
            best_score = sc
            best_f = f.copy()
            best_c = c.copy()
            no_imp = 0
        else:
            no_imp += 1
    
    return best_f, best_score, best_c
