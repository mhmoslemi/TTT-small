and coordinate descent polish. Minimizes 2*n*max(convolve(a,a))/(sum(a))^2.

import time

def _normalize(v):
    """Project to non-negative float vector of sum 1."""
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v


def _fft_size(m):
    """Smallest power of two at least m."""
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()

    """Full linear convolution f*f via FFT. Returns array of length 2n-1."""
    n = f.size
    m = 2 * n - 1
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    full = np.fft.irfft(F * F, size)
    return full[:m]

    n = f.size
    c = _autocorr(f)

    """
    Gradient of (1/T)*log(sum_k exp(T*c[k])) w.r.t. a.
    a: length n, c: length 2n-1 (= convolve(a,a))
    """

def _sigmoid_u_shape(n, alpha, beta, rng=None):
    """
    Generate a U-shaped sequence using product of sigmoids.
    a[i] = sigmoid(alpha*(x-0.5)) * sigmoid(alpha*(0.5-x)) + beta
    where x = i/(n-1)
    This gives high values at edges, low in middle.
    """
    x = np.linspace(0.0, 1.0, n)
    # Product of two sigmoids creates a U-shape
    s1 = 1.0 / (1.0 + np.exp(-alpha * (x - 0.3)))
    s2 = 1.0 / (1.0 + np.exp(-alpha * (0.7 - x)))
    a = s1 * s2 + beta
    return _normalize(a)

def _normalize(v):
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return np.ones(1, dtype=np.float64)
    v[~np.isfinite(v)] = 0.0
    np.maximum(v, 0.0, out=v)
    s = float(v.sum())
    if s <= 1e-300:
        return np.ones(v.size, dtype=np.float64) / float(v.size)
    v /= s
    return v


def _fft_size(m):
    if m <= 1:
        return 1
    return 1 << (int(m) - 1).bit_length()


def _autocorr(f):
    n = f.size
    m = 2 * n - 1
    size = _fft_size(m)
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:m]
