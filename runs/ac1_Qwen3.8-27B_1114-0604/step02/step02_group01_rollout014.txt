"""
Step function optimizer using LP-based peak reduction and accelerated mirror descent.
Combines linear programming for large-step improvements with mirror descent for fine tuning.
"""

import numpy as np
import cvxpy as cp
import time


def _autocorr_fft(f):
    """Full linear convolution a*a via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _score_fast(f):
    """Fast score computation."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _autocorr_fft(f)
    return float(2.0 * n * c.max() / (s * s))


def _lp_direction(a, M, K, rng, max_time=5.0):
    """
    Solve LP: max sum(g) s.t. (a*g)[k_i] <= M for top-K lags, g >= 0.
    Returns g (or None if LP fails).
    """
    n = len(a)
    c = _autocorr_fft(a)
    top_idx = np.argpartition(c, -K)[-K:]
    
    # Build constraint matrix B (K x n)
    # B[i, j] = a[k_i - j] if 0 <= k_i - j < n
    B = np.zeros((K, n))
    for i, k in enumerate(top_idx):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            idx = k - np.arange(j_start, j_end + 1)
            B[i, j_start:j_end + 1] = a[idx]
    
    g_var = cp.Variable(n, non_neg=True)
    constraints = [B @ g_var <= M * np.ones(K)]
    try:
        prob = cp.Problem(cp.Maximize(cp.sum(g_var)), constraints)
        start = time.monotonic()
        prob.solve(solver=cp.SCS, verbose=False, max_iters=100000)
        if time.monotonic() - start > max_time:
            return None
        if g_var.value is None:
            return None
        g_val = np.maximum(g_var.value, 0.0)
        s = g_val.sum()
        if s < 1e-10:
            return None
        return g_val
    except Exception:
        return None


def _mirror_step(f, grad, step):
    """Exponentiated gradient step."""
    log_f = np.log(f + 1e-300) - step * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    return f_new / s


def _gradient_at_lag(f, k):
    """Gradient of c[k] w.r.t. f: grad[j] = 2*f[k-j] for valid k-j."""
    n = len(f)
    grad = np.zeros(n)
    j_start = max(0, k - n + 1)
    j_end = min(n - 1, k)
    if j_start <= j_end:
        idx = k - np.arange(j_start, j_end + 1)
        grad[j_start:j_end + 1] = 2.0 * f[idx]
    return grad


def _optimize_lp(a, deadline, rng, verbose=True):
    """Phase 1: LP-based iterations."""
    n = len(a)
    a = a / a.sum()
    best_f = a.copy()
    best_score = _score_fast(a)
    c = _autocorr_fft(a)
    M = c.max()
    
    t = 0.3  # step size
    K = 30  # number of top lags to constrain
    n_iters = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        c = _autocorr_fft(a)
        M = c.max()
        S = a.sum()
        
        g = _lp_direction(a, M, K, rng)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum <= S * (1 + 1e-6):
            # No significant improvement possible
            no_improve += 1
            if no_improve > 5:
                break
            # Try with more constraints
            K = min(K + 10, 100)
            continue
        
        # Update
        t = max(0.05, 0.3 * 0.95 ** n_iters)
        a_new = (1 - t) * a + t * g / g_sum
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()
        
        score_new = _score_fast(a_new)
        n_iters += 1
        
        if score_new < best_score - 1e-8:
            best_score = score_new
            best_f = a_new.copy()
            no_improve = 0
            if verbose and n_iters % 5 == 0:
                print(f"  LP iter {n_iters}: score={score_new:.7f}, t={t:.4f}, K={K}")
        else:
            no_improve += 1
            # Roll back
            a = a  # keep old a
            if no_improve > 10:
                t *= 0.7
                if t < 0.01:
                    break
        
        a = a_new if score_new < best_score + 1e-6 else a
        # Actually let's be more careful:
        a = a_new  # always accept (LP guarantees non-increase approximately)
    
    return best_f, best_score


def _optimize_mirror(a, deadline, rng, verbose=False):
    """Phase 2: Accelerated mirror descent with adaptive K."""
    n = len(a)
    a = a / a.sum()
    best_f = a.copy()
    best_score = _score_fast(a)
    
    momentum = np.zeros(n)
    beta = 0.7
    n_iters = 0
    K = 3
    no_improve = 0
    patience = 2000
    
    # step size schedule
    step0 = 1.0
    decay = 0.9998  # per iteration
    
    c = _autocorr_fft(a)
    
    while time.monotonic() < deadline:
        # Find top K peaks
        top_idx = np.argpartition(c, -K)[-K:]
        top_idx = top_idx[np.argsort(c[top_idx])[::-1]]
        top_vals = c[top_idx]
        
        # Weights: emphasize higher peaks
        max_val = top_vals[0]
        weights = (top_vals / max_val) ** 4  # sharp weighting
        weights = weights / weights.sum()
        
        # Compute combined gradient
        grad = np.zeros(n)
        for k, w in zip(top_idx, weights):
            grad += w * _gradient_at_lag(a, k)
        
        # Momentum
        momentum = beta * momentum + (1 - beta) * grad
        grad_eff = grad + 0.8 * momentum
        
        # Step size
        span = grad_eff.max() - grad_eff.min()
        if span < 1e-30:
            break
        step = step0 / (1 + 0.001 * n_iters)
        # Actually let's use a simpler schedule
        step = 1.0 / (0.5 + 0.1 * n_iters)
        
        a = _mirror_step(a, grad_eff, step)
        a = np.maximum(a, 0.0)
        a = a / a.sum()
        
        n_iters += 1
        
        # Periodic evaluation
        if n_iters % 1000 == 0:
            c = _autocorr_fft(a)
            score = 2.0 * c.max()  # sum=1
            if score < best_score - 1e-8:
                best_score = score
                best_f = a.copy()
                no_improve = 0
                if verbose:
                    print(f"  Mirror iter {n_iters}: score={score:.7f}, K={K}")
            else:
                no_improve += 1000
            
            # Adapt K
            if n_iters % 10000 == 0:
                K = min(K + 3, 30)
            
            # Periodic perturbation if stuck
            if no_improve >= patience:
                # Structured perturbation
                mode = int(rng.integers(0, 4))
                j = np.arange(n, dtype=np.float64)
                if mode == 0:
                    alpha = rng.uniform(-0.5, 0.5)
                    a = a * np.exp(alpha * (j / n - 0.5))
                elif mode == 1:
                    freq = int(rng.integers(1, max(2, n // 5) + 1))
                    phase = rng.uniform(0, 2 * np.pi)
                    a = a * (1.0 + 0.3 * np.sin(2 * np.pi * freq * j / n + phase))
                elif mode == 2:
                    # Random block swap
                    block = n // 10
                    i1 = int(rng.integers(0, n - 2 * block))
                    i2 = int(rng.integers(0, n - 2 * block))
                    if abs(i1 - i2) > block:
                        amt = a[i1:i1+block].sum() * 0.2
                        a[i1:i1+block] -= amt / block
                        a[i2:i2+block] += amt / block
                else:
                    a = a + rng.normal(0, 0.01 / np.sqrt(n), size=n)
                
                a = np.maximum(a, 0.0)
                a = a / a.sum()
                momentum *= 0.5
                no_improve = 0
        
        c = _autocorr_fft(a)
    
    # Final eval
    score = _score_fast(a)
    if score < best_score:
        best_score = score
        best_f = a.copy()
    
    return best_f, best_score


def _polish(a, deadline, rng):
    """Phase 3: Coordinate descent polish."""
    n = len(a)
    a = a / a.sum()
    best_f = a.copy()
    best_score = _score_fast(a)
    c = _autocorr_fft(a)
    
    n_passes = 0
    while time.monotonic() < deadline:
        # Find top 3 peaks
        top3 = np.argpartition(c, -3)[-3:]
        
        improved = False
        for k in top3:
            # Try moving mass from high-contribution to low-contribution positions
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start > j_end:
                continue
            
            idx = k - np.arange(j_start, j_end + 1)
            contrib = np.zeros(n)
            contrib[j_start:j_end+1] = a[j_start:j_end+1] * a[idx]
            
            # Find best swap
            j_don = int(np.argmax(contrib))
            # Try all positions with lowest total contribution
            total_c = np.zeros(n)
            for kk in top3:
                js = max(0, kk - n + 1)
                je = min(n - 1, kk)
                if js <= je:
                    ix = kk - np.arange(js, je + 1)
                    total_c[js:je+1] += a[js:je+1] * a[ix]
            
            j_rec = int(np.argmin(total_c))
            
            if j_don == j_rec:
                continue
            
            best_peak = c.max()
            for frac in [0.1, 0.2, 0.3, 0.5]:
                amt = a[j_don] * frac
                if amt < 1e-15:
                    continue
                a_try = a.copy()
                a_try[j_don] -= amt
                a_try[j_rec] += amt
                a_try = np.maximum(a_try, 0)
                s = a_try.sum()
                if s < 1e-10:
                    continue
                a_try /= s
                c_try = _autocorr_fft(a_try)
                if c_try.max() < best_peak - 1e-12:
                    a = a_try
                    c = c_try
                    best_peak = c.max()
                    score = _score_fast(a)
                    if score < best_score:
                        best_score = score
                        best_f = a.copy()
                    improved = True
                    break
        
        n_passes += 1
        if n_passes % 100 == 0:
            print(f"  Polish pass {n_passes}: score={best_score:.7f}")
            
        if not improved and n_passes > 50:
            # Try random moves
            i = int(rng.integers(0, n))
            jx = int(rng.integers(0, n))
            if i != jx:
                amt = min(a[i], a[jx]) * 0.1
                a_try = a.copy()
                a_try[i] -= amt
                a_try[jx] += amt
                a_try = np.maximum(a_try, 0)
                a_try /= a_try.sum()
                c_try = _autocorr_fft(a_try)
                if c_try.max() < c.max() - 1e-14:
                    a = a_try
                    c = c_try
                    improved = True
    
    return best_f, best_score

Hmm wait, I realize there might be issues with the LP approach. Let me reconsider.

The LP constraint is (a*g)[k] <= M for the top-K lags. But (a*g) is the convolution of a (fixed) and g (variable). The issue is that after updating a_new = (1-t)*a + t*g, the new self-convolution is:
(a_new * a_new)[k] = (1-t)^2*(a*a)[k] + 2t(1-t)*(a*g)[k] + t^2*(g*g)[k]

We control (a*a)[k] <= M and (a*g)[k] <= M, but NOT (g*g)[k]. For small t, the t^2 term is negligible, but for larger t, it might cause the peak to increase.

To be safe, I should use small t (e.g., t <= 0.1) or also constrain (g*g)[k] <= M/something.

Actually, the simpler approach is: after the update, check if the peak actually decreased. If not, use a smaller t.

Let me also reconsider the mirror descent approach. The key issue is the step size. If the step is too large, we might overshoot. If too small, we converge slowly.

For the mirror step: a_new[j] = a[j] * exp(-step * grad[j]) / Z.

The "Lipschitz constant" of the gradient is roughly L = max_j |grad[j]| = 2*max(a) (since grad[j] = 2*a[k-j]). With sum(a)=1 and n=6121, max(a) is at most 1 (but typically much less, around 0.001-0.01).

So L ~ 0.001-0.02. A safe step size is step = 1/L ~ 50-1000. But this might be too large in practice.

Actually, let me think about it differently. The mirror step changes a[j] by a factor of exp(-step * grad[j]). If grad[j] ~ 2*a[k-j] ~ 2/n ~ 0.0003, and step = 1, then the factor is exp(-0.0003) ≈ 0.9997. So each step changes a[j] by about 0.03%. That's very small.

To get meaningful changes, we need step * grad ~ O(1), so step ~ 1/grad ~ n/2 ~ 3000. Let me use step in the range [100, 10000].

**Alternative: Direct construction using known mathematics.**

The problem of minimizing the peak autocorrelation of a non-negative sequence is related to the "minimum energy" problem in signal processing. 

One key insight: if we think of the sequence as a probability distribution (sum=1, non-negative), then the autocorrelation c[k] is the "overlap" when the distribution is shifted by k. The peak at k=0 (or k=n-1 for the full overlap) is related to the "concentration" of the distribution.

For a distribution on {0, 1, ..., n-1}, the minimum possible maximum autocorrelation is related to the "anti-concentration" of the distribution.
