import numpy as np
import time

def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(a, b):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)

def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s

    c = _conv_full(p, p)

def _next_pow2(x):
    x = max(1, int(x))
    L = 1

    la, lb = len(a), len(b)
