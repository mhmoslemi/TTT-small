Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses multi-start MMD with adaptive top-K peaks, spectral flattening restarts,

import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b):
    la = len(a)
    lb = len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 1000000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)


def _conv_valid_sym(w, p):
    """
    Compute gradient-like vector: v[j] = sum_i w[i] * (p[j+i] + p[abs(j-i)] style)
    Specifically: v[j] = sum_i w[i]*p[j+i] (valid, j=0..len(p)-1)
                  + sum_i w[i]*p[abs(j-i)] (valid)
    This approximates 2 * gradient of sum w_k * c_k w.r.t. p_j.
    """
    n = len(p)
    nw = len(w)
    if n == 0:
        return np.array([], dtype=float)
    L = _next_pow2(max(nw, 2 * n))
    fw = np.fft.rfft(w, L)
    fp = np.fft.rfft(p, L)
    # First part: sum_i w[i] * p[j+i] -> valid correlation
    out1 = np.fft.irfft(fw * np.conjugate(fp), L)[:n]
    # Second part: sum_i w[i] * p[i-j] -> reversed correlation
    wr = w[::-1]
    fwr = np.fft.rfft(wr, L)
    out2 = np.fft.irfft(fwr * np.conjugate(fp), L)[:n]
    out = out1 + out2
    return out.astype(float, copy=False)


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return p / s


def _score(p):
    p = _sanitize(p)
    if p is None:
        return float('inf')
    n = p.size
    c = _conv_full(p, p)
    if c.size == 0:
        return float('inf')
    c = np.where(np.isfinite(c), c, 0.0)
    return float(2.0 * n * np.max(c))


def _score_and_max_lag(p):
    p = _sanitize(p)
    if p is None:
        return float('inf'), 0
    n = p.size
    c = _conv_full(p, p)
    if c.size == 0:
        return float('inf'), 0
    c = np.where(np.isfinite(c), c, 0.0)
    mx = float(np.max(c))
    idx = int(np.argmax(c))
    return float(2.0 * n * mx), idx


def _soft_weights_topk(d, beta, K=None):
    """Softmax weights, optionally zeroing out non-top-K elements."""
    d = np.asarray(d, dtype=float)
    n = d.size
    if n == 0:
        return np.array([], dtype=float)
    d = np.where(np.isfinite(d), d, 0.0)
    if K is not None and K < n:
        # Hard top-K: set non-top-K to -inf
        if K <= n:
            thresh = np.partition(d, n - K)[n - K] if K < n else d.min()
            d = np.where(d >= thresh - 1e-15, d, -1e10)
    if beta <= 1e-8:
        return np.full(n, 1.0 / n, dtype=float)
    shift = float(np.max(d))
    if not np.isfinite(shift):
        return np.full(n, 1.0 / n, dtype=float)
    e = np.exp(beta * (d - shift))
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n, dtype=float)
    return e / s


def _grad_metrics(p, beta, K=None):
    """Return (exact_score, smooth_obj, gradient_vector, max_grad)."""
    p = _sanitize(p)
    if p is None:
        return float('inf'), float('inf'), None, 0.0
    n = p.size
    c = _conv_full(p, p)
    if c.size == 0:
        return float('inf'), float('inf'), np.zeros(1), 0.0
    c = np.where(np.isfinite(c), c, 0.0)
    d = n * c
    exact = float(2.0 * max(0.0, np.max(d)))
    w = _soft_weights_topk(d, beta, K)
    grad = _conv_valid_sym(w, p)
    if grad.size == 0:
        grad = np.zeros(1)
    grad = np.where(np.isfinite(grad), grad, 0.0)
    grad = np.maximum(grad, 0.0)
    maxg = float(np.max(grad)) if grad.size else 0.0
    smooth = float(np.dot(w, d))
    return exact, smooth, grad, maxg


def _spectral_flatten(p, alpha=0.3):
    """
    spectral flattening step: soften peaks in the power spectrum.
    p -> IFFT(phase(F) * |F|_flattened) -> clip -> normalize
    """
    p = _sanitize(p)
    if p is None:
        return None
    n = p.size
    L = _next_pow2(2 * n)
    F = np.fft.rfft(p, L)
    mag = np.abs(F)
    phase = np.angle(F)
    # Flatten: mag_new = mag / (mag + alpha * mean_mag) ** 0.5 * (mag + alpha * mean_mag) ** 0.5
    # Simpler: mag_new = (mag^2 + alpha * mean(mag^2))^0.5 / norm
    mag2 = mag ** 2
    mean_mag2 = float(np.mean(mag2))
    if mean_mag2 <= 0:
        return p
    mag2_new = (mag2 + alpha * mean_mag2) / (1.0 + alpha)
    mag_new = np.sqrt(mag2_new)
    # Renormalize so that IFFT has sum = 1 (DC component controls the sum)
    F_new = mag_new * np.exp(1j * phase)
    p_new = np.fft.irfft(F_new, L)[:n]
    p_new = _sanitize(p_new)
    if p_new is None:
        return p
    return p_new


def _init_two_block(n, rng):
    """Generate two-block style initialization."""
    best_p = None
    best_s = float('inf')
    for frac in (0.3, 0.4, 0.5, 0.6, 0.7):
        for ratio in (0.5, 1.0, 2.0, 4.0, 8.0):
            p = np.ones(n, dtype=float)
            idx = int(round(frac * n))
            p[idx:] = ratio
            p = _sanitize(p)
            if p is None:
                continue
            s = _score(p)
            if s < best_s:
                best_s = s
                best_p = p.copy()
    return best_p


def _init_diverse(n, rng, mode):
    t = (np.arange(n, dtype=float) + 0.5) / float(n)
    if mode == 0:
        # Cosine series
        z = np.ones(n)
        for k in range(1, 15):
            amp = float(rng.normal(0, 1.0 / (k + 0.5)))
            ph = float(rng.uniform(0, 2 * np.pi))
            z += amp * np.cos(2 * np.pi * k * t + ph)
        p = np.log1p(np.exp(np.clip(z, -40, 40)))
    elif mode == 1:
        # Beta-like
        a = float(rng.uniform(0.1, 3.0))
        b = float(rng.uniform(0.1, 3.0))
        tt = np.clip(t, 1e-8, 1 - 1e-8)
        p = tt ** a * (1 - tt) ** b
    elif mode == 2:
        # Gaussian bumps
        K = int(rng.integers(1, 5))
        p = np.zeros(n)
        for _ in range(K):
            c = float(rng.uniform(0, 1))
            s = float(rng.uniform(0.03, 0.4))
            w = float(rng.uniform(0.5, 3.0))
            p += w * np.exp(-((t - c) / s) ** 2)
    elif mode == 3:
        # Spline
        K = int(rng.integers(4, 12))
        knots = np.sort(rng.uniform(0, 1, K))
        heights = rng.lognormal(0, 1.0, K)
        p = np.interp(t, knots, heights)
    else:
        # Logistic + gaussian
        c = float(rng.uniform(0.2, 0.8))
        k = float(rng.uniform(5, 30))
        p = 1.0 / (1.0 + np.exp(-np.clip(k * (t - c), -50, 50)))
        p *= np.exp(-((t - c) / 0.2) ** 2)
    p = _sanitize(p)
    return p


def _resample(arr, n):
    a = _sanitize(arr)
    if a is None:
        return None
    m = a.size
    if m == n:
        return a.copy()
    if m == 1:
        return np.ones(n, dtype=float) / n
    x_old = (np.arange(m) + 0.5) / m
    x_new = (np.arange(n) + 0.5) / n
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0) + 1e-15
    return _sanitize(out)


def _run_mmd(p0, rng, time_limit, beta_start=50.0, beta_end=3000.0,
             K_mode='auto', verbose=False):
    """
    MMD with optional hard top-K. K_mode: 'auto' ramps K from 1 to full.
    """
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    if time_limit <= 0:
        return p, _score(p)

    n = int(p.size)
    floor = 1e-20 / n
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_exact = _score(p)
    cur_exact, cur_smooth, _, _ = _grad_metrics(p, beta_start, None)

    beta = float(beta_start)
    alpha = 0.1
    deadline = time.monotonic() + time_limit
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        # Adaptive K: start with K=1 (hardest peak), increase over time
        if K_mode == 'auto':
            progress = min(1.0, it / max(1, int(time_limit * 3)))
            K = max(1, int(progress * n // 4))
            K = min(K, n)
        elif K_mode == 'top1':
            K = 1
        elif K_mode == 'top5':
            K = 5
        else:
            K = None

        cur_exact, cur_smooth, grad, maxg = _grad_metrics(p, beta, K)

        if not np.isfinite(maxg) or maxg <= 1e-18:
            p_try = _sanitize(p * np.exp(rng.normal(0, 0.05, n)))
            if p_try is not None:
                p_try = np.maximum(p_try, floor) / p_try.sum()
                e, _, _, _ = _grad_metrics(p_try, beta, K)
                if e <= best_exact * 1.1:
                    p = p_try
                    cur_exact = e
            stall += 1
        else:
            grad = grad / maxg
            # Try multiple step sizes
            tests = []
            for mult in (0.25, 0.5, 1.0, 2.0, 4.0):
                a = alpha * mult
                if 1e-5 <= a <= 30.0:
                    tests.append(a)
            if not tests:
                tests = [1e-4, 0.05]

            chosen = None
            ch_smooth = cur_smooth
            ch_exact = cur_exact

            for a in tests:
                dec = np.exp(-a * grad)
                p_try = p * dec
                p_try = np.maximum(p_try, floor)
                s = float(p_try.sum())
                if s <= 0 or not np.isfinite(s):
                    continue
                p_try = p_try / s
                e, s2, _, _ = _grad_metrics(p_try, beta, K)
                if np.isfinite(s2) and s2 < ch_smooth - 1e-12 * max(1, ch_smooth):
                    chosen = p_try
                    ch_smooth = s2
                    ch_exact = e

            if chosen is not None:
                p = chosen
                cur_smooth = ch_smooth
                cur_exact = ch_exact
                stall = 0
                if cur_exact < best_exact:
                    best = p.copy()
                    best_exact = cur_exact
                    if verbose and it % 500 == 0:
                        print(f"  MMD n={n} it={it} best={best_exact:.6f} beta={beta:.0f}")
                if cur_exact > best_exact * 1.15 and rng.random() < 0.3:
                    p = best.copy()
                    cur_exact, cur_smooth, _, _ = _grad_metrics(p, beta, K)
            else:
                stall += 1
                alpha = max(1e-5, alpha * 0.7)
                if stall >= 15:
                    std = min(0.3, 0.03 + 0.01 * stall)
                    p_try = _sanitize(p * np.exp(rng.normal(0, std, n)))
                    if p_try is not None:
                        p_try = np.maximum(p_try, floor) / p_try.sum()
                        e, _, _, _ = _grad_metrics(p_try, beta, K)
                        if e <= best_exact * 1.05:
                            p = p_try
                            cur_exact = e
                        else:
                            p = best.copy()
                            cur_exact, cur_smooth, _, _ = _grad_metrics(p, beta, K)
                    stall = 0
                if stall > 100:
                    p = best.copy()
                    cur_exact, cur_smooth, _, _ = _grad_metrics(p, beta, K)
                    alpha = 0.1
                    stall = 0

        it += 1
        if it % 15 == 0:
            beta = min(beta_end, beta * 1.2)

        # Periodic spectral flattening restart
        if it % 500 == 250 and rng.random() < 0.3:
            pf = _spectral_flatten(best, alpha=0.2)
            if pf is not None:
                ef = _score(pf)
                if ef < best_exact:
                    best = pf.copy()
                    best_exact = ef
                    p = pf.copy()
                    cur_exact = ef

    return best, best_exact


def _coord_descent(p0, rng, time_limit):
    """Block coordinate descent."""
    p = _sanitize(p0)
    if p is None or time_limit <= 0:
        return None, float('inf')
    n = p.size
    floor = 1e-20 / n
    p = np.maximum(p, floor) / p.sum()
    best = p.copy()
    best_s = _score(p)
    cur_s = best_s
    deadline = time.monotonic() + time_limit
    stagnant = 0
    factors = (0.4, 0.7, 0.9, 1.1, 1.3, 1.6, 2.5)

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.2:
            w = int(rng.integers(1, max(2, n // 300 + 1)))
        elif r < 0.6:
            w = int(rng.integers(max(2, n // 300), max(3, n // 50 + 1)))
        else:
            lo = max(3, n // 50)
            hi = max(lo + 1, n // 10 + 1)
            w = int(rng.integers(lo, hi + 1))
        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))

        base = p.copy()
        cur_best = base
        cur_score = cur_s
        for f in factors:
            cand = base.copy()
            cand[start:start + w] *= f
            cand = _sanitize(cand)
            if cand is None:
                continue
            cand = np.maximum(cand, floor) / cand.sum()
            sc = _score(cand)
            if sc < cur_score - 1e-13:
                cur_score = sc
                cur_best = cand
        if cur_best is not base:
            p = cur_best
            cur_s = cur_score
            if cur_s < best_s:
                best = p.copy()
                best_s = cur_s
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 80:
                p_try = _sanitize(p * np.exp(rng.normal(0, 0.04, n)))
                if p_try is not None:
                    p_try = np.maximum(p_try, floor) / p_try.sum()
                    sc = _score(p_try)
                    if sc <= best_s * 1.03:
                        p = p_try
                        cur_s = sc
                    else:
                        p = best.copy()
                        cur_s = best_s
                stagnant = 0
            if stagnant > 400:
                p = best.copy()
                cur_s = best_s
                stagnant = 0

    return best, best_s


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Main search function. Returns list of non-negative floats.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 120.0

    start_wall = time.monotonic()
    deadline = start_wall + budget - 20.0  # 20s safety margin

    try:
        iseed = int(seed)
    except Exception:
        iseed = 12345
    rng = np.random.default_rng((iseed % 1000003) + 7)

    # Load parent if available
    parent = None
    try:
        parent_raw = height_sequence_1
        parent = _sanitize(np.asarray(parent_raw, dtype=float))
    except (NameError, Exception):
        parent = None

    if parent is not None:
        ps = _score(parent)
        print(f"Parent score: {ps:.6f}, length: {parent.size}")

    # Choose primary length
    rem = deadline - time.monotonic()
    if rem > 2000:
        primary_n = 4096
    elif rem > 1500:
        primary_n = 3584
    else:
        primary_n = 3072

    print(f"Primary length: {primary_n}, time budget: {rem:.0f}s")

    # Build diverse starting points at primary_n
    starts = []
    scores = []

    if parent is not None:
        for target_n in [primary_n]:
            sp = _resample(parent, target_n)
            if sp is not None:
                sc = _score(sp)
                starts.append(sp)
                scores.append(sc)
        # Also try slight perturbations of parent
        sp = _resample(parent, primary_n)
        if sp is not None:
            sp2 = _sanitize(sp * np.exp(rng.normal(0, 0.03, primary_n)))
            if sp2 is not None:
                sc = _score(sp2)
                starts.append(sp2)
                scores.append(sc)

    # Two-block
    tb = _init_two_block(primary_n, rng)
    if tb is not None:
        sc = _score(tb)
        starts.append(tb)
        scores.append(sc)

    # Diverse random
    for mode in range(5):
        sp = _init_diverse(primary_n, rng, mode)
        if sp is not None:
            sc = _score(sp)
            starts.append(sp)
            scores.append(sc)

    # Sort by score and keep top 4
    order = np.argsort(scores)
    top_starts = [starts[i] for i in order[:4]]
    top_scores = [scores[i] for i in order[:4]]
    print(f"Starting scores: {[f'{s:.4f}' for s in top_scores]}")

    best_arr = top_starts[0] if top_starts else _init_diverse(primary_n, rng, 0)
    best_score = min(top_scores) if top_scores else float('inf')

    # Phase 1: Run MMD on top starts with time allocation
    rem = deadline - time.monotonic()
    fracs = [0.45, 0.25, 0.18, 0.12]
    for idx, sp in enumerate(top_starts):
        rem = deadline - time.monotonic()
        if rem < 100:
            break
        frac = fracs[idx] if idx < len(fracs) else 0.1
        talloc = min(rem - 100, rem * frac)
        if talloc < 30:
            talloc = max(10, rem - 30)

        print(f"MMD run {idx+1}/{len(top_starts)}: {talloc:.0f}s, start score={_score(sp):.4f}")
        p, sc = _run_mmd(sp, rng, talloc,
                         beta_start=50.0, beta_end=4000.0,
                         K_mode='auto', verbose=True)
        if p is not None and sc < best_score:
            best_arr = p.copy()
            best_score = sc
            print(f"  New best: {best_score:.6f}")

    # Phase 2: Coordinate descent polish
    rem = deadline - time.monotonic()
    if rem > 120:
        t1 = min(0.35 * rem, 500.0)
        print(f"Coord descent: {t1:.0f}s")
        p, sc = _coord_descent(best_arr, rng, t1)
        if p is not None and sc < best_score:
            best_arr = p.copy()
            best_score = sc
            print(f"  After coord: {best_score:.6f}")

    # Phase 3: Final MMD with high beta
    rem = deadline - time.monotonic()
    if rem > 100:
        t2 = min(0.3 * rem, 400.0)
        print(f"Final MMD: {t2:.0f}s")
        p, sc = _run_mmd(best_arr, rng, t2,
                         beta_start=200.0, beta_end=8000.0,
                         K_mode='top1', verbose=True)
        if p is not None and sc < best_score:
            best_arr = p.copy()
            best_score = sc
            print(f"  After final MMD: {best_score:.6f}")

    # Phase 4: Try a larger length if time permits
    rem = deadline - time.monotonic()
    if rem > 300 and best_arr is not None:
        big_n = min(6144, 2 * primary_n)
        if big_n > primary_n:
            big = _resample(best_arr, big_n)
            if big is not None:
                big_score = _score(big)
                print(f"Upscaled to {big_n}: score={big_score:.6f}")
                t3 = min(0.4 * rem, 600.0)
                pb, sb = _run_mmd(big, rng, t3,
                                  beta_start=100.0, beta_end=5000.0,
                                  K_mode='auto', verbose=True)
                if pb is not None and sb < best_score:
                    best_arr = pb.copy()
                    best_score = sb
                    print(f"  Big n={big_n} best: {best_score:.6f}")

                rem = deadline - time.monotonic()
                if rem > 60:
                    pl, sl = _coord_descent(best_arr, rng, min(60.0, 0.3 * rem))
                    if pl is not None and sl < best_score:
                        best_arr = pl.copy()
                        best_score = sl

    # Phase 5: Final coord descent
    rem = deadline - time.monotonic()
    if rem > 40:
        p, sc = _coord_descent(best_arr, rng, min(40.0, rem - 10))
        if p is not None and sc < best_score:
            best_arr = p.copy()
            best_score = sc

    # Final validation with evaluate_sequence
    final_cands = [best_arr]
    # Also include parent if it scored well
    if parent is not None:
        final_cands.append(parent)

    rem = deadline - time.monotonic()
    final_list = [float(x) for x in best_arr]
    for arr in final_cands:
        if rem < 5:
            break
        a = _sanitize(arr)
        if a is None:
            continue
        lst = [float(x) for x in a]
        try:
            sc = float(evaluate_sequence(lst))
        except Exception:
            sc = _score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            final_list = lst
        rem = deadline - time.monotonic()

    out = np.asarray(final_list, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s <= 0 or not np.isfinite(s):
        out = np.ones(primary_n, dtype=float)
    else:
        out = out / s

    print(f"Final: score={best_score:.6f}, length={out.size}")
    return [float(x) for x in out]
