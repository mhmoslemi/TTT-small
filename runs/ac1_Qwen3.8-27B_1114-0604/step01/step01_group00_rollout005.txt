```python
            med = np.median(c)
            weights = np.maximum(c[top_lags] - med, 0.0)
            if weights.sum() < 1e-30:
                weights = np.ones(K) / K
            else:
                weights = weights / weights.sum()
            
            grad = np.zeros(n)
            for w, m in zip(weights, top_lags):
                m = int(m)
                for j in range(n):
                    idx = m - j
                    if 0 <= idx < n:
                        grad[j] += 2.0 * w * f[idx]
        
        # Mirror descent step
        span = grad.max() - grad.min()
        if span < 1e-15:
            break
        eta = T / span
        log_f = np.log(f + 1e-30) - eta * grad
        log_f -= log_f.max()
        f_new = np.exp(log_f)
        f_new = _normalize(f_new)
        
        # Small entropy regularization
        f_new = 0.999 * f_new + 0.001 / n
        f_new = _normalize(f_new)
        
        f = f_new
        
        # Periodic evaluation
        if total_iters % eval_interval == 0:
            score = _eval(f)
            if score < best_score - 1e-8:
                best_score = score
                best_f = f.copy()
                stuck = 0
            else:
                stuck += 1
                if stuck > 10:
                    # Perturbation
                    noise = rng.standard_normal(n) * 0.01
                    f = _normalize(f + noise)
                    stuck = 0
    
    # Final evaluation
    score = _eval(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score


def _coord_descent_phase(f, deadline, rng, fine=False):
    """
    Pairwise coordinate descent.
    For each top contributor to the peak, try transferring to bottom contributors.
    Use grid search over transfer fractions.
    """
    n = len(f)
    f = _normalize(f.copy())
    best_f = f.copy()
    best_score = _eval(f)
    
    if fine:
        fracs = [0.01, 0.02, 0.03, 0.05, 0.07, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5]
        num_donors = 8
        num_receivers = 8
    else:
        fracs = [0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7]
        num_donors = 5
        num_receivers = 5
    
    iteration = 0
    no_improve_count = 0
    max_no_improve = 30 if not fine else 50
    
    while time.monotonic() < deadline:
        iteration += 1
        c = _full_conv(f)
        peak = c.max()
        m_star = int(np.argmax(c))
        
        # Compute contribution of each coordinate to the peak lag
        contrib = np.zeros(n)
        for k in range(n):
            idx = m_star - k
            if 0 <= idx < n:
                contrib[k] = f[k] * f[idx]
        
        # Donors: top contributors
        donor_indices = np.argsort(contrib)[::-1][:num_donors]
        # Receivers: bottom contributors (but with some mass)
        receiver_indices = np.argsort(contrib)[:num_receivers]
        # Filter out receivers with too little mass
        receiver_indices = [j for j in receiver_indices if f[j] > 1e-8]
        
        improved = False
        
        for k in donor_indices:
            if time.monotonic() > deadline:
                break
            for j in receiver_indices:
                if k == j or f[k] < 1e-10:
                    continue
                
                best_delta = 0
                best_val = peak
                
                for frac in fracs:
                    delta = f[k] * frac
                    f_try = f.copy()
                    f_try[k] -= delta
                    f_try[j] += delta
                    c_try = _full_conv(f_try)
                    val = c_try.max()
                    if val < best_val:
                        best_val = val
                        best_delta = delta
                
                if best_delta > 0 and best_val < peak - 1e-10:
                    f[k] -= best_delta
                    f[j] += best_delta
                    f = _normalize(f)
                    peak = best_val
                    improved = True
                    break
            
            if improved:
                break
        
        if improved:
            no_improve_count = 0
            score = _eval(f)
            if score < best_score:
                best_score = score
                best_f = f.copy()
        else:
            no_improve_count += 1
            if no_improve_count >= max_no_improve:
                # Try random moves
                for _ in range(20):
                    k = int(rng.integers(0, n))
                    j = int(rng.integers(0, n))
                    if k == j or f[k] < 1e-10:
                        continue
                    for frac in [0.05, 0.1, 0.2]:
                        delta = f[k] * frac
                        f_try = f.copy()
                        f_try[k] -= delta
                        f_try[j] += delta
                        c_try = _full_conv(f_try)
                        if c_try.max() < peak - 1e-10:
                            f = _normalize(f_try)
                            peak = c_try.max()
                            improved = True
                            break
                    if improved:
                        break
                
                if improved:
                    score = _eval(f)
                    if score < best_score:
                        best_score = score
                        best_f = f.copy()
                    no_improve_count = 0
                else:
                    # Perturb and reset
                    noise = rng.standard_normal(n) * 0.005
                    f = _normalize(f + noise)
                    no_improve_count = 0
    
    return best_f, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
```

Wait, I realize there's a potential issue with the mirror descent phase. The inner loop `for j in range(n)` inside the gradient computation is O(n) per lag, and we have K lags, so the gradient computation is O(n*K) per step. For n=6121 and K=10, that's ~60K operations per step, which is fine.

But the FFT is also O(n log n) ≈ 6121 * 13 ≈ 80K operations. So the FFT dominates.

Total operations per mirror descent step: ~140K (FFT + gradient). In Python, this might take ~1-5ms per step. Over 2400*0.4 = 960 seconds, we can do ~200K-1M steps. That should be enough.

For the coordinate descent phase: each "iteration" involves computing the convolution (FFT), then for each donor-receiver pair, trying several fractions (each requiring an FFT). With 5 donors * 5 receivers * 8 fractions = 200 FFTs per iteration, each taking ~1ms, that's 200ms per iteration. Over 960 seconds, that's ~4800 iterations. That might not be enough for convergence.

Let me optimize: instead of computing the full FFT for each candidate, I can compute the convolution more efficiently for small changes.

If we change f[k] by -δ and f[j] by +δ, the change in the convolution is:
Δ(conv)[m] = -δ*(f[m-k] + f[k-m]) + δ*(f[m-j] + f[j-m]) + cross terms

Wait, this isn't quite right because f[m-k] and f[m-j] are the OLD values (before the change). Let me be more precise.

Let f' = f + δ*(e_j - e_k) where e_i is the unit vector.
Then (f'*f')[m] = (f*f)[m] + δ*(e_j - e_k)*f)[m] + δ*(f*(e_j - e_k))[m] + δ^2*(e_j - e_k)*(e_j - e_k))[m]

Now:
(f * e_j)[m] = f[m-j] (if 0 <= m-j < n, else 0)
(e_j * f)[m] = f[m-j] (same, by commutativity)
(e_j * e_j)[m] = 1 if m = 2j, else 0
(e_k * e_k)[m] = 1 if m = 2k, else 0
(e_j * e_k)[m] = 1 if m = j+k, else 0
(e_k * e_j)[m] = 1 if m = j+k, else 0

So:
Δ(conv)[m] = -δ*(f[m-k] + f[k-m]) + δ*(f[m-j] + f[j-m]) + δ^2*(δ_{m,2j} - δ_{m,2k} - 2*δ_{m,j+k})

Wait, let me redo this:
(f'*f')[m] = (f + δe_j - δe_k) * (f + δe_j - δe_k)
= f*f + δ(f*e_j) - δ(f*e_k) + δ(e_j*f) - δ(e_k*f) + δ^2*(e_j*f) - δ^2*(e_k*f) - δ^2*(e_j*f) + δ^2*(e_j*e_j) - δ^2*(e_j*e_k) - δ^2*(e_k*e_j) + δ^2*(e_k*e_k)

Hmm, let me be more careful:
= f*f + δ*f*e_j + (-δ)*f*e_k + δ*e_j*f + δ^2*e_j*e_j + (-δ)*e_j*e_k + (-δ)*f*e_k + (-δ)*e_k*f + (-δ)^2*e_k*e_k...

This is getting confusing. Let me use the bilinear expansion:
(a+b)(c+d) = ac + ad + bc + bd

Let u = f, v = δ(e_j - e_k). Then f' = u + v.
(f'*f')[m] = (u+v)*(u+v)[m] = (u*u)[m] + (u*v)[m] + (v*u)[m] + (v*v)[m]

(u*v)[m] = sum_k u[k]*v[m-k] = sum_k f[k]*δ*(1[m-k=j] - 1[m-k=k])
= δ*(f[m-j] - f[m-k]) (where f[x] = 0 if x not in [0,n-1])

(v*u)[m] = same by commutativity = δ*(f[m-j] - f[m-k])

(v*v)[m] = δ^2 * sum (1[m-k=j] - 1[m-k=k]) * (1[m-2k... ])
= δ^2 * (1[m-2k... ])... 

Actually, v = δ*e_j - δ*e_k, so v*v = δ^2*(e_j - e_k)*(e_j - e_k) = δ^2*(e_j*e_j - e_j*e_k - e_k*e_j + e_k*e_k)
= δ^2*(e_{2j} - e_{j+k} - e_{j+k} + e_{2k})
= δ^2*(e_{2j} - 2*e_{j+k} + e_{2k})

So (v*v)[m] = δ^2*(1[m=2j] - 2*1[m=j+k] + 1[m=2k])

Therefore:
(f'*f')[m] = (f*f)[m] + 2δ*(f[m-j] - f[m-k]) + δ^2*(1[m=2j] - 2*1[m=j+k] + 1[m=2k])

This is a closed-form expression! We can compute the new convolution in O(n) (just add the delta term to the old convolution) instead of O(n log n) (FFT).

This is a huge speedup!Instead of doing an FFT for each candidate (k, j, δ), we just do O(n) operations.

Let me verify: for m = 2j (assuming 2j < 2n-1):
(f'*f')[2j] = (f*f)[2j] + 2δ*(f[2j-j] - f[2j-k]) + δ^2*(1 - 2*1[2j=j+k] + 1[2j=2k])
= (f*f)[2j] + 2δ*(f[j] - f[2j-k]) + δ^2*(1 - 2*1[j=k] + 1[2j=2k])

Since j ≠ k (we check this), 1[j=k] = 0. And 1[2j=2k] = 0 since j≠k and they're integers.
So: (f'*f')[2j] = (f*f)[2j] + 2δ*(f[j] - f[2j-k]) + δ^2

That makes sense: we're adding δ^2 to the "self-term" at position 2j.

Great, so the algorithm becomes:
1. Compute c = conv(f, f) once (O(n log n)).
2. For each candidate (k, j, δ):
   - Compute c_new = c + 2δ*(shifted_f_j - shifted_f_k) + δ^2*correction (O(n))
   - Check max(c_new)
3. Pick the best (k, j, δ) and update f and c.

Step 2 is O(n) per candidate. With 5*5*8 = 200 candidates, that's O(200n) = O(n) per iteration (with a constant factor of 200).

For n=6121, that's 200*6121 ≈ 1.2M operations per iteration. In Python, maybe 10-50ms per iteration. Over 960 seconds, that's ~20K-100K iterations. Much better!

But wait, I need to be careful: after updating f, the "shifted_f" arrays change. So I need to update them. But that's just changing two elements, so it's O(1).

Let me precompute the "shifted" arrays:
- For a fixed m, f[m-j] as a function of j is just f shifted.
- Actually, for the delta computation, I need:
  - For each m: f[m-j] and f[m-k]
  - These are just f shifted by j and k respectively.

So I can precompute:
- shift_j = np.roll(f, j) with appropriate zero-padding... actually it's not a simple roll because of the boundary conditions.

Let me define:
- g_j[m] = f[m-j] if 0 <= m-j < n, else 0. This is f "shifted right by j" with zeros.
- g_k[m] = f[m-k] if 0 <= m-k < n, else 0.

These are just f with j (or k) zeros prepended and the rest truncated.

So: g_j = np.concatenate([np.zeros(j), f])[:2*n-1]  (if j < n)
    or more precisely: g_j[m] = f[m-j] for m = j, j+1, ..., j+n-1, and 0 otherwise.
    The length is 2n-1 (the length of the full convolution).

Actually, since the convolution has length 2n-1, and g_j[m] = f[m-j] for m in [0, 2n-2]:
- g_j[m] = f[m-j] if 0 <= m-j < n, i.e., j <= m <= j+n-1
- otherwise 0.

So g_j is f "embedded" starting at position j in a zero array of length 2n-1.

I can precompute this for any j in O(n):
```python
def _shifted(f, j):
    n = len(f)
    result = np.zeros(2*n-1)
    result[j:j+n] = f
    return result
```

Then the delta computation is:
```python
c_new = c + 2*delta*(shifted_j - shifted_k) + delta^2*correction
```

where correction[m] = 1[m=2j] - 2*1[m=j+k] + 1[m=2k] (a vector with at most 3 non-zero entries).

This is O(n) per candidate!
