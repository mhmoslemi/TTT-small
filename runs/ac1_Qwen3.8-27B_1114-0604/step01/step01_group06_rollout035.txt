    N = int(n)
    NFFT = 1 << (3*N - 2).bit_length()
    TAU = float(tau)

def softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.array([], dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0:
        return np.full_like(z, 1.0/z.size)
    return e / s

def fft_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    L = 2*n - 1
    Nf = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, Nf)
    c = np.fft.irfft(fa * fa, Nf)[:L].real
    return np.maximum(c, 0.0)

def fast_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    s = float(np.sum(a))
    if s <= 1e-12:
        return np.inf
    c = fft_conv(a)
    m = float(np.max(c))
    return float(2 * len(a) * m / (s*s))

def obj_grad(z):
    n = N
    Nf = NFFT
    tau = TAU
    L = 2*n - 1
    
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n:
        return 1e12, np.zeros(n)
    z = np.clip(np.nan_to_num(z, nan=0.0, posinf=BOUND, neginf=-BOUND), -BOUND, BOUND)
    
    a = softmax(z)
    fa = np.fft.rfft(a, Nf)
    c = np.fft.irfft(fa * fa, Nf)[:L].real
    c = np.maximum(c, 0.0)
    
    m = float(np.max(c))
    diff = np.clip((c - m) / tau, -745.0, 0.0)
    e = np.exp(diff)
    se = float(np.sum(e))
    if se <= 0 or not np.isfinite(se):
        return 1e12, np.zeros(n)
    p = e / se
    
    fp = np.fft.rfft(p, Nf)
    h = np.fft.irfft(fp * np.conjugate(fa), Nf)[:n].real
    s_h = float(np.dot(h, a))
    g = a * (h - s_h)
    grad = 2.0 * g
    
    val = m + tau * np.log(se)
    scale = float(n)
    val *= scale
