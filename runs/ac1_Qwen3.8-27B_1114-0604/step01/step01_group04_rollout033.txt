```python
    n = a.size

    s = a.sum()
```

```python
    k_arr = np.arange(size)
    
    for p in range(n_passes):
        improved = False
        for i in range(n):
            S_old = S  # current sum
            x_old = a[i]
            S_rest = S_old - x_old  # sum of others

            ki = k_arr - i

            lo, hi = 0.0, x_max

            x2 = lo + gr*(hi-lo)
            
            def f_at(x):
                b_x = b0 + d*x + e*x*x
                return float(np.max(b_x)) / (S_rest + x)**2
            
            f1 = f_at(x1)
            f2 = f_at(x2)

                    x1 = lo + (1-gr)*(hi-lo)

                    x2 = lo + gr*(hi-lo)
```

```python
    return np.fft.irfft(fa * fa, fsize)[:size]

def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = full_conv(a)
    return 2.0 * n * float(np.max(b)) / (s * s)

def make_de(a, i, size):
    """Precompute d[k] and e[k] for coordinate i."""
    n = a.size
    k_arr = np.arange(size)
    ki = k_arr - i
    mask = (ki >= 0) & (ki < n)
    d = np.zeros(size)
    # d[k] = 2*a[k-i] for valid k-i, except when k=2i
    valid = mask & (k_arr != 2 * i)
    d[valid] = 2.0 * a[ki[valid]]
    e = np.zeros(size)
    if 0 <= 2 * i < size:
        e[2 * i] = 1.0
    return d, e

def optimize_one_coord(b, a, i, S, size, k_arr):
    """Optimize coordinate i. Returns (new_a_i, new_b, new_S)."""
    n = a.size
    x_old = a[i]
    S_rest = S - x_old
    
    ki = k_arr - i
    mask = (ki >= 0) & (ki < n)
    
    # d[k] = 2*a[k-i] for valid, k≠2i
    d = np.zeros(size)
    valid = mask & (k_arr != 2 * i)
    d[valid] = 2.0 * a[ki[valid]]
    
    # e[k] = 1 if k = 2i
    e = np.zeros(size)
    k_2i = 2 * i
    if 0 <= k_2i < size:
        e[k_2i] = 1.0
    
    # b0 = b - x_old*d - x_old^2*e
    x2_old = x_old * x_old
    b0 = b - x_old * d - x2_old * e
    
    # Golden section search
    x_max = 2.0 * S_rest + 1.0 if S_rest > 0 else 1.0
    
    gr = (np.sqrt(5) - 1) / 2
    lo, hi = 0.0, x_max
    
    def f_at(x):
        b_x = b0 + d * x + e * x * x
        denom = S_rest + x
        if denom < 1e-300:
            return np.inf
        return float(np.max(b_x)) / (denom * denom)
    
    x1 = lo + (1.0 - gr) * (hi - lo)
    x2 = lo + gr * (hi - lo)
    f1 = f_at(x1)
    f2 = f_at(x2)
    
    for _ in range(45):
        if f1 < f2:
            hi = x2
            x2, f2 = x1, f1
            x1 = lo + (1.0 - gr) * (hi - lo)
            f1 = f_at(x1)
        else:
            lo = x1
            x1, f1 = x2, f2
            x2 = lo + gr * (hi - lo)
            f2 = f_at(x2)
    
    x_new = max(0.0, (lo + hi) / 2.0)
    
    new_b = b0 + d * x_new + e * x_new * x_new
    new_S = S_rest + x_new
    
    return x_new, new_b, new_S

def coordinate_descent(a, time_budget, rng, cool_schedule=True):
    """
    Run coordinate descent with random ordering and simulated annealing.
    a is the initial sequence (non-negative).
    Returns (best_a, best_score).
    """
    n = a.size
    size = 2 * n - 1
    k_arr = np.arange(size)
    
    a = a.copy().astype(np.float64)
    S = a.sum()
    if S < 1e-10:
        a = np.ones(n) / n
        S = 1.0
    b = full_conv(a)
    
    best_sc = 2.0 * n * float(np.max(b)) / (S * S)
    best_a = a.copy()
    best_b = b.copy()
    best_S = S
    
    t0 = time.monotonic()
    pass_num = 0
    no_improve = 0
    temperature = 0.05  # annealing temperature (fraction of current score)
    
    while time.monotonic() - t0 < time_budget:
        # Random order
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_budget:
                break
            
            x_new, b_new, S_new = optimize_one_coord(b, a, i, S, size, k_arr)
            
            # Compute new score
            if S_new < 1e-10:
                continue
            new_sc = 2.0 * n * float(np.max(b_new)) / (S_new * S_new)
            cur_sc = 2.0 * n * float(np.max(b)) / (S * S)
            
            # Accept if better, or with annealing probability
            if new_sc <= cur_sc or rng.random() < np.exp(-(new_sc - cur_sc) / (temperature + 1e-10)):
                a[i] = x_new
                b = b_new
                S = S_new
            
            # Track best
            if new_sc < best_sc:
                best_sc = new_sc
                best_a = a.copy()
                best_b = b.copy()
                best_S = S
                no_improve = 0
            elif new_sc > best_sc * (1 + 1e-8):
                no_improve += 1
        
        pass_num += 1
        
        # Anneal
        if cool_schedule:
            temperature *= 0.95
        
        # Restart if stuck
        if no_improve > n * 5:
            a = best_a + 0.02 * best_S * rng.standard_normal(n)
            a = np.maximum(a, 0)
            S = a.sum()
            b = full_conv(a)
            no_improve = 0
            temperature = 0.05
    
    return best_a, best_sc
```

Hmm wait, there's an issue with my annealing acceptance criterion. The score difference `new_sc - cur_sc` could be very small (like 1e-6) while the temperature is 0.05. In that case, `exp(-1e-6/0.05)` ≈ 1, so we'd almost always accept. That's fine - it means we accept slightly worse moves easily early on.

But the temperature schedule might need adjustment. Let me think about what units to use. The score is around 1.5-2.0, so differences are typically 1e-3 to 1e-2. A temperature of 0.05 means we'd accept moves that increase the score by up to ~0.05*ln(1/p) where p is the acceptance probability. For p=0.5, that's 0.05*0.69 ≈ 0.035. That's a reasonable "annealing range."

Actually, I realize I should be more careful. The acceptance criterion should be:
- Accept if new_sc < cur_sc (improvement)
- Otherwise accept with probability exp(-(new_sc - cur_sc) / T)

where T is the temperature. As T → 0, we only accept improvements (greedy). As T is large, we accept almost everything (random walk).

Let me set T = 0.01 initially and cool to T = 0.001. This means initially we accept moves that increase score by up to ~0.01, and finally only accept moves that increase by ~0.
