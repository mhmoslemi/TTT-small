   Actually, let me also think about a completely different approach: **alternating minimization with a "potential" function**.

   Key insight: `max_k b[k] = min_M { M : b[k] <= M for all k }`. So:
   
   `min_a max_k b[k] = min_a min_M { M : b[k](a) <= M for all k, a >= 0, sum(a) = 1 }`

   This is a min-min problem, which is just a constrained optimization. But the constraint `b[k](a) <= M` is a quadratic constraint (since `b[k]` is quadratic in `a`). So this is a QCQP (quadratically constrained quadratic program), which is NP-hard in general.

   - But we can relax: use SOCP (second-order cone) relaxation, or use the fact that `b[k](a) = a^T C_k a` where `C_k` is a matrix, and use trust region methods.

   import numpy as np
   import time

   # --- Core functions ---

   def full_conv(a):
       """FFT-based full convolution of a with itself."""
       n = a.size
       size = 2 * n - 1
       fsize = 1 << (size - 1).bit_length()
       fa = np.fft.rfft(a, fsize)
       return np.fft.irfft(fa * fa, fsize)[:size]

   def get_score(a):
       """Compute the objective value."""
       n = a.size
       S = float(a.sum())
       if S < 1e-10:
           return np.inf
       b = full_conv(a)
       return 2.0 * n * float(np.max(b)) / (S * S)

   # --- Coordinate descent ---

   def optimize_one_coord(a, i, b, S, n):
       """
       Optimize a single coordinate a[i] to minimize the objective.
       Updates a[i] and b in place. Returns the new score.
       """
       ai = a[i]
       m_start = i
       m_end = min(i + n, 2 * n - 1)
       length_in_slice = m_end - m_start  # = min(n, 2n-1-i)
       
       # a[m-i] for m in [m_start, m_end) = a[0 : length_in_slice]
       a_slice = a[:length_in_slice].copy()
       b_slice = b[m_start:m_end].copy()
       
       # Precompute max of unchanged parts of b
       max_left = float(np.max(b[:m_start])) if m_start > 0 else 0.0
       max_right = float(np.max(b[m_end:])) if m_end < 2 * n - 1 else 0.0
       max_unchanged = max(max_left, max_right)
       
       # Define f(t) = objective when a[i] = t
       def f_of_t(t):
           if t < 0:
               return np.inf
           delta = 2.0 * (t - ai)
           if length_in_slice > 0:
               max_changed = float(np.max(b_slice + delta * a_slice))
           else:
               max_changed = 0.0
           max_b = max(max_changed, max_unchanged)
           S_new = S - ai + t
           if S_new < 1e-10:
               return np.inf
           return 2.0 * n * max_b / (S_new * S_new)
       
       # Quick candidate evaluation
       amax = float(a.max())
       candidates = [ai, 0.0]
       if ai > 0:
           candidates += [ai*0.3, ai*0.7, ai*1.3, ai*2.0]
       candidates += [max(0, ai - 0.05*amax), ai + 0.05*amax, max(0, ai - 0.02*amax), ai + 0.02*amax]
       
       best_t = ai
       best_f = f_of_t(ai)
       for t in candidates:
           if t < 0 or abs(t - ai) < 1e-15:
               continue
           f = f_of_t(t)
           if f < best_f:
               best_f = f
               best_t = t
       
       # Golden section search for refinement
       if best_t > 1e-10:
           # Determine bracket
           if best_t <= ai * 0.4:
               lo, hi = 0.0, ai + 0.1*amax
           elif best_t >= ai * 1.6:
               lo, hi = ai * 0.5, best_t + 0.1*amax
           else:
               span = max(0.05 * amax, 0.01 * best_t)
               lo, hi = max(0.0, best_t - span), best_t + span
           
           for _ in range(12):
               m1 = lo + (hi - lo) * 0.381966
               m2 = lo + (hi - lo) * 0.618034
               f1 = f_of_t(m1)
               f2 = f_of_t(m2)
               if f1 <= f2:
                   hi = m2
               else:
                   lo = m1
           
           t_star = (lo + hi) * 0.5
           f_star = f_of_t(t_star)
           if f_star < best_f:
               best_f = f_star
               best_t = t_star
       else:
           # best_at_zero: check if a tiny positive value helps
           for t_try in [1e-6, 1e-4, 0.001 * amax, 0.01 * amax]:
               f_try = f_of_t(t_try)
               if f_try < best_f:
                   best_f = f_try
                   best_t = t_try
       
       # Apply update
       old_ai = a[i]
       a[i] = best_t
       delta = best_t - old_ai
       if abs(delta) > 1e-14 and length_in_slice > 0:
           b[m_start:m_end] += 2.0 * delta * a_slice
       
       return best_f
