```python
def solve_lp(f, M):
    """Solve: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    # Objective: minimize -sum(g)
    c = -np.ones(n)
    # Constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_i f[i] * g[k-i] (with valid indices)
    # For a fixed k, the coefficient of g[j] is f[k-j] if 0 <= k-j < n, else 0
    A_ub = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A_ub[k, j] = f[idx]
    b_ub = np.full(2*n-1, M)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    return res.x
```

```python
"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based iterative
best-response method. At each step, solve an LP to find the direction g
that maximizes sum(g) subject to (f*g)[k] <= max(f*f) for all k.
This is a game-theoretic approach that converges to a local minimum.
"""

import numpy as np
from scipy.optimize import linprog
import time

def full_conv(a):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def normed_score(a):
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))

def project_nonneg_sum1(a):
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / a.size
    return a / s

def build_conv_matrix(f):
    """Build the (2n-1) x n matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n-1)
        for j in range(j_min, j_max+1):
            A[k, j] = f[k-j]
    return A

def solve_lp_step(f, M):
    """Solve: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k."""
    n = len(f)
    c = -np.ones(n)
    A = build_conv_matrix(f)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x

def lp_iterate(f, t0, max_iters=5000):
    """
    Run LP-based best-response iteration.
    f is normalized to sum to 1.
    Returns (best_f, best_score).
    """
    n = f.size
    f = project_nonneg_sum1(f.copy())
    best_f = f.copy()
    best_score = normed_score(f)
    
    t = 0.1  # mixing parameter
    iters = 0
    last_improve = 0
    stall_count = 0
    
    while time.monotonic() < t0 and iters < max_iters:
        b = full_conv(f)
        M = float(np.max(b))
        cur_score = 2.0 * n * M
        
        g = solve_lp_step(f, M)
        if g is None:
            # LP infeasible, reduce M (increase it actually)
            t *= 0.5
            continue
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        
        g = g / g_sum  # normalize to sum to 1
        
        # Update
        f_new = (1 - t) * f + t * g
        f_new = project_nonneg_sum1(f_new)
        
        new_score = normed_score(f_new)
        
        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
            last_improve = iters
            stall_count = 0
        elif iters - last_improve > 500:
            stall_count += 1
        
        f = f_new
        iters += 1
        
        # Adaptive learning rate
        if stall_count > 3:
            t *= 0.7
            stall_count = 0
        elif new_score < cur_score:
            t = min(t * 1.05, 0.3)
        
        # Periodic restart
        if iters - last_improve > 2000:
            # Try a perturbation of the best
            f = project_nonneg_sum1(best_f + 0.02 * np.random.standard_normal(n))
            t = 0.1
            last_improve = iters
            stall_count = 0
    
    return best_f, best_score
```

```python
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based best-response
iteration. At each step, solve an LP to find g maximizing sum(g) subject to
(f*g)[k] <= max(f*f) for all k. Update f via convex combination. Multiple
sequence lengths and diverse initializations are tried.

import numpy as np
from scipy.optimize import linprog
import time


def full_conv(a):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def project_nonneg_sum1(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / a.size
    return a / s


def build_conv_matrix(f):
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_step(f, M):
    n = len(f)
    c = -np.ones(n)
    A = build_conv_matrix(f)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x


def lp_iterate(f_init, time_limit, rng, verbose=False):
    n = f_init.size
    f = project_nonneg_sum1(f_init.astype(np.float64).copy())
    best_f = f.copy()
    best_score = normed_score(f)
    
    t = 0.1
    iters = 0
    last_improve = 0
    stall = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        b = full_conv(f)
        M = float(np.max(b))
        cur_score = 2.0 * n * M
        
        g = solve_lp_step(f, M)
        if g is None or g.sum() < 1e-15:
            t = max(t * 0.5, 0.005)
            iters += 1
            continue
        
        g = g / g.sum()
        
        f_new = (1.0 - t) * f + t * g
        f_new = project_nonneg_sum1(f_new)

            best_score = new_score
            best_f = f_new.copy()
            last_improve = iters
```

```python
"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based best-response
iteration combined with subgradient polishing. The LP at each step finds g
maximizing sum(g) subject to (f*g)[k] <= max(f*f) for all k, providing a
direction to reduce the autocorrelation peak. Multiple lengths and diverse
initializations are explored, with final subgradient polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def project_nonneg_sum1(a):
    """Project onto {x >= 0, sum(x) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-30:
        return np.ones_like(a) / a.size
    return a / s


def build_conv_matrix(f):
    """Build (2n-1) x n matrix A where A[k,j] = f[k-j] if 0<=k-j<n."""
    n = len(f)
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_step(f, M):
    """Solve LP: max sum(g) s.t. g>=0, (f*g)[k] <= M for all k."""
    n = len(f)
    c = -np.ones(n)
    A = build_conv_matrix(f)
    b = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x


def lp_iterate(f_init, time_limit, rng, verbose=False):
    """Run LP-based best-response iteration. f normalized to sum=1."""
    n = f_init.size
    f = project_nonneg_sum1(f_init.astype(np.float64).copy())
    best_f = f.copy()
    best_score = normed_score(f)

    t = 0.08
    iters = 0
    last_improve = 0
    stall = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_limit:
        b = full_conv(f)
        M = float(np.max(b))
        cur_score = 2.0 * n * M

        g = solve_lp_step(f, M)
        if g is None or g.sum() < 1e-15:
            t = max(t * 0.5, 0.003)
            iters += 1
            continue

        g = g / g.sum()

        f_new = (1.0 - t) * f + t * g
        f_new = project_nonneg_sum1(f_new)

        new_score = normed_score(f_new)

        if new_score < best_score:
            best_score = new_score
            best_f = f_new.copy()
            last_improve = iters
            stall = 0
            if verbose and (iters % 200 == 0 or best_score < 1.51):
                print(f"    [lp] iter={iters}, score={best_score:.6f}, t={t:.4f}")
        else:
            if iters - last_improve > 800:
                stall += 1

        f = f_new
        iters += 1

        if stall > 4:
            t *= 0.6
            stall = 0
        if new_score <= cur_score * 1.002:
            t = min(t * 1.03, 0.2)

        if iters - last_improve > 2500:
            f = project_nonneg_sum1(best_f + 0.01 * rng.standard_normal(n))
            t = 0.04 + 0.08 * rng.random()
            last_improve = iters
            stall = 0

    return best_f, best_score


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def init_u_shape(n):
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i).astype(np.float64) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_bimodal(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.1 + 0.15 * rng.random())
    c2 = n * (0.75 + 0.15 * rng.random())
    sigma = n * (0.03 + 0.05 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_trimodal(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.05 + 0.1 * rng.random())
    c2 = n * (0.5 + 0.1 * rng.random())
    c3 = n * (0.85 + 0.1 * rng.random())
    sigma = n * (0.03 + 0.06 * rng.random())
    a = (np.exp(-((i - c1) / sigma) ** 2) +
         np.exp(-((i - c2) / sigma) ** 2) +
         np.exp(-((i - c3) / sigma) ** 2))
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_cosine(n, rng):
    i = np.arange(n, dtype=np.float64)
    k = 1 + int(rng.integers(0, 4))
    alpha = 0.3 + 0.6 * rng.random()
    beta = 1.0 + 1.0 * rng.random()
    a = (1.0 + alpha * np.cos(2.0 * np.pi * k * i / n)) ** beta
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def init_sided(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / max(c, 1.0)) ** 2 + 0.05
    return a / a.sum()


def init_tanh_edge(n, rng):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    width = n * (0.2 + 0.3 * rng.random())
    a = np.tanh((np.abs(i - c) - width * 0.3) / (width * 0.2)) ** 2 + 0.05
    return a / a.sum()


def get_inits(n, rng, prev_a=None):
    inits = []
    if prev_a is not None and prev_a.size == n:
        inits.append(prev_a.copy())
    inits.append(init_bimodal(n, rng))
    inits.append(init_trimodal(n, rng))
    inits.append(init_u_shape(n))
    inits.append(init_cosine(n, rng))
    inits.append(init_random(n, rng))
    inits.append(init_sided(n))
    inits.append(init_tanh_edge(n, rng))
    return inits


def smooth_grad(a, b, tau):
    """Smoothed subgradient of max(conv) using softmax weights."""
    n = a.size
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(b.size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= s
    return 2.0 * np.correlate(w, a, mode='valid')


def polish_subgrad(a, time_budget, rng):
    """Local polish via smoothed subgradient descent with momentum."""
    n = a.size
    a = project_nonneg_sum1(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    eta = 0.04
    tau_frac = 0.025
    vel = np.zeros(n)
    beta = 0.5
    iters = 0
    last_improve = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))
        tau = tau_frac * float(np.max(b))
        if tau < 1e-18:
            tau = 1e-18

        g = smooth_grad(a, b, tau)
        vel = beta * vel + (1.0 - beta) * g
        a_new = project_nonneg_sum1(a - eta * vel)

        b_new = full_conv(a_new)
        new_sc = 2.0 * n * float(np.max(b_new))

        if new_sc <= cur_sc or rng.random() < 0.04:
            a = a_new
        else:
            a = project_nonneg_sum1(a - 0.4 * eta * vel)

        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = iters

        iters += 1
        if iters - last_improve > 3000:
            a = project_nonneg_sum1(best_a + 0.008 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.02 + 0.04 * rng.random()
            last_improve = iters

    return best_a, best_sc
```
