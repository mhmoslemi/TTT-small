import numpy as np
import time
from scipy.optimize import linprog

def fft_conv(a):
    """Compute a*a linear convolution via FFT."""
    n = len(a)
    size = 1
    while size < 2*n-1:
        size <<= 1
    F = np.fft.rfft(a, size)
    c = np.fft.irfft(F*F, size)[:2*n-1]
    return np.maximum(c, 0)

def fft_conv_ab(a, b):
    """Compute a*b linear convolution via FFT."""
    na, nb = len(a), len(b)
    size = 1
    while size < na+nb-1:
        size <<= 1
    Fa = np.fft.rfft(a, size)
    Fb = np.fft.rfft(b, size)
    c = np.fft.irfft(Fa*Fb, size)[:na+nb-1]
    return np.maximum(c, 0)

def smooth_max(b, beta):
    """Compute smooth max and softmax weights."""
    m = np.max(b)
    x = np.exp(beta*(b - m))
    Z = x.sum()
    w = x / Z
    G = m + np.log(Z)/beta
    return G, w

def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = len(v)
    if n == 0:
        return v
    lo, hi = np.min(v) - 1, np.max(v) + 1
    for _ in range(60):
        mid = (lo+hi)/2
        if np.sum(np.maximum(v - mid, 0)) > target:
            lo = mid
        else:
            hi = mid
    tau = (lo+hi)/2
    q = np.maximum(v - tau, 0)
    s = q.sum()
    if s > 0:
        q *= target/s
    return q

def compute_grad(a, w):
    """Compute gradient of smooth max using one-sided correlation.
    grad[i] ≈ 2 * sum_j a[j] * w[i+j]"""
    n = len(a)
    a_rev = a[::-1]
    size = 1
    need = (2*n-1) + n - 1
    while size < need:
        size <<= 1
    W = np.fft.rfft(w, size)
    AR = np.fft.rfft(a_rev, size)
    conv = np.fft.irfft(W*AR, size)[:need]
    # conv[i + n - 1] = sum_j w[i+n-1-j]*a_rev[j] = sum_j w[i+n-1-j]*a[n-1-j]
    # Let k = i+n-1-j: j = i+n-1-k, a[n-1-j] = a[n-1-i-n+1+k] = a[k-i]
    # So conv[i+n-1] = sum_k w[k]*a[k-i] = (w ⊛ a)[i]
    svec = conv[n-1:n-1+n]
    grad = 2.0 * np.maximum(svec, 0)
    return grad
