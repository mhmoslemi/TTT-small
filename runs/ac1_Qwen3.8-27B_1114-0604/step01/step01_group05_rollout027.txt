"""
Minimize 2*n*max(a*a)/(sum(a))^2 for non-negative sequences using 
softmax-parameterized L-BFGS optimization with smooth max.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def fft_convolution(a):
    """Linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max(b, beta):
    """Smooth max of b using log-sum-exp, returns (value, weights)."""
    m = float(np.max(b))
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return G, w


def softmax(z):
    """Stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    return e / float(np.sum(e))


def corr_pw(p, w):
    """s_i = sum_j p_j * w_{i+j} for i=0..n-1 (circular-ish correlation)."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n: m][::-1].copy()
    return s


def objective_and_grad(z, n, beta):
    """
    Objective and gradient for softmax-parameterized sequence.
    a = softmax(z), minimize 2n * smooth_max(a*a, beta).
    """
    a = softmax(z)
    b = fft_convolution(a)
    G, w = smooth_max(b, beta)
    
    if not np.isfinite(G):
        return float('inf'), np.zeros(n, dtype=np.float64)
    
    # Gradient: dG/da_i = 2 * s_i where s_i = corr_pw(a, w)_i
    s = corr_pw(a, w)
    g = 2.0 * s
    
    # dF/dz_i = 2n * a_i * (g_i - sum_j g_j * a_j)
    weighted_mean = float(np.dot(g, a))
    grad = 2.0 * n * a * (g - weighted_mean)
    
    obj = 2.0 * n * G
    return obj, grad


def run_lbfgs(z0, n, beta, maxiter=300, tol=1e-12):
    """Run L-BFGS-B on z with given beta."""
    result = minimize(
        objective_and_grad,
        z0,
        args=(n, beta),
        method='L-BFGS-B',
        jac=True,
        options={'maxiter': maxiter, 'ftol': tol, 'gtol': 1e-8, 'maxls': 50}
    )
    return np.asarray(result.x, dtype=np.float64)

def fft_convolution(a):
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0]*a[0]], dtype=np.float64)
    need = 2*n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F*F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    m = float(np.max(b))
    x = np.clip(beta*(b - m), -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return G, w


def softmax_fn(z):
    z = z - np.max(z)
    e = np.exp(z)
    return e / float(np.sum(e))


def corr_pw(p, w):
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P*WR, size)[:need]
    return C[m-n:m][::-1].copy()

"""
Minimize 2*n*max(a*a)/(sum(a))^2 using softmax-parameterized L-BFGS
with multi-beta smooth max, diverse restarts, and peak-shaving polish.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def fft_convolution(a):
    """Linear convolution a*a using FFT."""
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    """Smooth max via log-sum-exp. Returns (value, normalized_weights)."""
    m = float(np.max(b))
    x = np.clip(beta * (b - m), -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return G, w


def softmax_fn(z):
    """Stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(z) / float(z.size)
    return e / s


def corr_pw(p, w):
    """s_i = sum_j p_j * w_{i+j} for i=0..n-1."""
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n: m][::-1].copy()
    return s


def obj_gradient(z, n, beta):
    """Objective and gradient for softmax-parameterized sequence."""
    a = softmax_fn(z)
    b = fft_convolution(a)
    G, w = smooth_max_and_weights(b, beta)
    if not np.isfinite(G):
        return float('inf'), np.zeros(n, dtype=np.float64)
    s = corr_pw(a, w)
    g = 2.0 * s
    wm = float(np.dot(g, a))
    grad = 2.0 * n * a * (g - wm)
    obj = float(2.0 * n * G)
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(n, dtype=np.float64)
    return obj, grad


def fast_score(p):
    """Fast score: 2*n*max(a*a) for normalized p."""
    n = p.size
    S = float(np.sum(p))
    if S <= 0 or not np.isfinite(S):
        return float('inf')
    pn = p / S
    b = fft_convolution(pn)
    M = float(np.max(b))
    return 2.0 * n * M


def make_inits(n, rng, prev_p=None):
    """Generate diverse initializations in z-space."""
    inits = []
    inits.append(np.zeros(n, dtype=np.float64))

    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    for coeff in (-4.0, -3.0, -2.0, -1.0, 1.0, 2.0, 3.0, 4.0):
        inits.append(coeff * (x - 0.5))

    # Quartic
    inits.append(2.0 * (2.0 * x - 1.0) ** 2 - 1.0)
    inits.append(-2.0 * (2.0 * x - 1.0) ** 2 + 1.0)
    # Cubic
    inits.append(1.5 * (2.0 * x - 1.0) ** 3)

    # Periodic
    inits.append(np.sin(2.0 * np.pi * x))
    inits.append(np.cos(2.0 * np.pi * x))
    inits.append(0.5 * np.sin(4.0 * np.pi * x))
    inits.append(0.3 * np.cos(6.0 * np.pi * x))
    inits.append(0.4 * np.sin(2.0 * np.pi * x) + 0.3 * np.cos(4.0 * np.pi * x))

    # V-shape (peak at edges)
    inits.append(2.0 * np.abs(x - 0.5) - 0.5)
    # Inverted V (peak at center)
    inits.append(0.5 - 2.0 * np.abs(x - 0.5))

    # Random smooth (band-limited)
    for freq_scale in (1.0, 2.0, 4.0):
        zc = np.zeros(2 * n, dtype=np.complex128)
        mask = np.zeros(n + 1, dtype=bool)
        bw = max(2, int(freq_scale * n // 16))
        mask[:bw] = True
        mask[-bw:] = True
        zc[mask] = rng.normal(size=np.sum(mask)) + 1j * rng.normal(size=np.sum(mask))
        z = np.real(np.fft.ifft(zc))[:n]
        z = z - np.mean(z)
        inits.append(np.tanh(2.0 * z))

    # Pure random
    inits.append(rng.normal(0, 0.5, n))
    inits.append(rng.normal(0, 1.0, n))
    inits.append(0.8 * rng.normal(0, 1.0, n))

    # Previous best (if available and right length)
    if prev_p is not None and len(prev_p) == n:
        p = np.maximum(np.asarray(prev_p, dtype=np.float64), 1e-15)
        p = p / p.sum()
        z = np.log(p) - np.log(p).mean()
        inits.append(z)
        # Slightly perturbed version
        inits.append(z + 0.1 * rng.normal(0, 1, n))

    return inits


def peak_shave(a, n, steps=2000, rate=0.01):
    """
    Heuristic peak shaving: iteratively reduce the peak of a*a.
    Uses multiplicative updates on the elements contributing most to the peak.
    """
    a = a.copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = fast_score(a)

    for step in range(steps):
        b = fft_convolution(a)
        k_star = int(np.argmax(b))
        peak_val = float(b[k_star])

        # Find which pairs (i, k_star - i) contribute most
        # (a*a)_{k_star} = sum_i a_i * a_{k_star - i}
        valid_i = range(max(0, k_star - n + 1), min(n, k_star + 1))
        contribs = np.array([a[i] * a[k_star - i] for i in valid_i])
        total = contribs.sum()
        if total <= 0:
            break

        # Focus on top contributors
        top_frac = 0.3
        n_top = max(1, int(top_frac * len(valid_i)))
        top_idx = np.argsort(contribs)[::-1][:n_top]

        # Reduce a at those positions
        for ti in top_idx:
            i = valid_i[ti]
            j = k_star - i
            if i != j:
                reduction = rate * contribs[ti] / total
                a[i] = max(1e-15, a[i] * (1.0 - reduction))
                a[j] = max(1e-15, a[j] * (1.0 - reduction))
            else:
                a[i] = max(1e-15, a[i] * (1.0 - 0.5 * rate * contribs[ti] / total))

        # Renormalize
        a = a / a.sum()

        if step % 200 == 0:
            sc = fast_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    sc = fast_score(a)
    if sc < best_sc:
        best_a = a.copy()
        best_sc = sc

    return best_a, best_sc


def run_lbfgs(z0, n, beta, maxiter=200):
    """Run L-BFGS-B with given parameters."""
    try:
        result = minimize(
            obj_gradient,
            z0,
            args=(n, beta),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 30}
        )
        z_new = np.asarray(result.x, dtype=np.float64)
    except Exception:
        z_new = z0.copy()

    if not np.all(np.isfinite(z_new)):
        z_new = z0.copy()

    return z_new
