           return np.ones_like(z) / z.size
       return e / s
   
   def _conv(a):
       fa = np.fft.rfft(a, _NFFT)
       c = np.fft.irfft(fa * fa, _NFFT)[:2*_N-1]
       return np.maximum(c, 0.0)
   
   def _fastscore(a):
       s = a.sum()
       if s < 0.01:
           return np.inf
       c = _conv(a)
       return float(2 * _N * c.max() / (s*s))
   
   def _og(z):
       n = _N
       tau = _TAU
       z = np.asarray(z, dtype=np.float64)
       z = np.clip(z, -_BOUND, _BOUND)
       a = _sm(z)
       c = _conv(a)
       m = c.max()
       diff = np.clip((c - m) / tau, -745.0, 0.0)
       e = np.exp(diff)
       se = e.sum()
       if se <= 0 or not np.isfinite(se):
           return 1e12, np.zeros(n)
       p = e / se
       fa = np.fft.rfft(a, _NFFT)
       fp = np.fft.rfft(p, _NFFT)
       h = np.fft.irfft(fp * np.conjugate(fa), _NFFT)[:n]
       sh = np.dot(h, a)
       g = 2.0 * a * (h - sh)
       val = m + tau * np.log(se)
       # Scale for numerical comfort
       val *= n * n
       g *= n * n
       # Small penalty for gauge freedom
       val += 1e-10 * np.dot(z, z)
       g += 1e-10 * z
       if not np.isfinite(val):
           val = 1e12
           g = np.zeros(n)
       return float(val), g
   
   def _lbfgs(z0, maxfun=300, maxiter=100):
       z0 = np.asarray(z0, dtype=np.float64)
       z0 = z0 - z0.mean()
       z0 = np.clip(z0, -_BOUND, _BOUND)
       try:
           res = minimize(_og, z0, jac=True, method='L-BFGS-B', 
                         bounds=_BNDS,
                         options={'maxfun': maxfun, 'maxiter': maxiter,
                                  'ftol': 1e-14, 'pgtol': 1e-12, 'disp': False})
           return res
       except Exception:
           return None
