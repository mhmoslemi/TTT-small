def _log_space_optimize(a0, deadline):
    """
    Optimize in log-space using the smooth max approximation.
    a_i = softmax(z_i), minimize 2n * tau * logsumexp(conv/tau)

    def obj_grad(z):
        z = np.clip(z, -30, 30)
        z = z - np.max(z)
        e = np.exp(z)
        a = e / np.sum(e)
        
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)[:2*n-1].real
        c = np.maximum(c, 0)

        exp_diff = np.exp(diff)
        se = np.sum(exp_diff)

        # dF/da = 2 * NIFFT(p * conj(Fa))
        fp = np.fft.rfft(p, N)

        # dF/dz = dF/da * da/dz = dF/da * (a - a*a) ... for softmax

        dF_da = 2.0 * h
        dot = np.sum(dF_da * a)
        grad = a * (dF_da - dot)
        
        return F, grad
    
    # Use L-BFGS-B
    result = minimize(obj_grad, z, jac=True, method='L-BFGS-B',
