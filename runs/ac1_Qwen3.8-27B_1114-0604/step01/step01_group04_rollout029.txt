    """
    Solve min a^T W a s.t. a >= 0, sum(a) = 1
    using projected gradient descent.

    a = a_init.copy()
    for _ in range(n_inner):
        grad = 2.0 * (W @ a)
        a_new = project_simplex(a - eta * grad)
        # Backtracking: check if objective decreased
        obj_old = a @ (W @ a)
        obj_new = a_new @ (W @ a_new)
        if obj_new > obj_old:
            eta *= 0.5
            a_new = project_simplex(a - eta * grad)

            a = a_new

    W = build_W(w, n)

"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using iterative quadratic
approximation with softmax active-set identification, parametric family
initialization, adaptive temperature annealing, and multi-n search.
Key idea: at each step, form a weighted autocorrelation quadratic form
a^T W a (where W[i,j] = softmax_weight[i+j]) and solve it via projected
gradient, giving a "Newton-like" step toward peak equalization.
"""

import numpy as np
import time
import sys


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def score_norm(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def build_W(w, n):
    """Build n x n matrix where W[i,j] = w[i+j]. w has length 2n-1."""
    W = np.zeros((n, n))
    for i in range(n):
        W[i, :] = w[i:i + n]
    return W


def solve_qp_inner(W, a, n_steps, eta):
    """Solve min a^T W a s.t. a>=0, sum(a)=1 via projected gradient."""
    a = a.copy()
    for _ in range(n_steps):
        grad = 2.0 * (W @ a)
        a_new = project_simplex(a - eta * grad)
        # Simple backtracking
        obj_old = a @ (W @ a)
        obj_new = a_new @ (W @ a_new)
        if obj_new > obj_old * (1 + 1e-12):
            eta *= 0.5
            a_new = project_simplex(a - eta * grad)
        a = a_new
    return a


def qp_optimize(a, time_budget, rng, tau_start=5.0, tau_end=80.0,
                n_inner=8, eta_start=0.08, damping=0.6, restart_every=3000):
    """
    Main QP-based optimizer with adaptive temperature.
    Alternates: compute softmax weights -> form W -> solve QP -> update.
    """
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = score_norm(a)

    eta = eta_start
    t0 = time.monotonic()
    it = 0
    last_improve = 0
    eval_count = 0

    while time.monotonic() - t0 < time_budget:
        elapsed = time.monotonic() - t0
        frac = min(1.0, it / (restart_every * 3))

        # Adaptive temperature
        tau = tau_start * (tau_end / tau_start) ** frac

        c = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(c))

        # Softmax weights
        shifted = c - c.max()
        w = np.exp(tau * shifted)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(2 * n - 1)
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum

        # Build W and solve QP
        W = build_W(w, n)
        a_qp = solve_qp_inner(W, a, n_inner, eta)

        # Damped update
        a_new = (1.0 - damping) * a + damping * a_qp
        a_new = project_simplex(a_new)

        # Evaluate
        new_sc = score_norm(a_new)
        eval_count += 1

        if new_sc <= cur_sc * (1 + 1e-10) or (rng.random() < 0.08 and new_sc < cur_sc * 1.02):
            a = a_new
        else:

def smooth_grad_optimize(a, time_budget, rng, tau_base=0.03, n_scales=3):
    """Smoothed subgradient descent with momentum as a secondary method."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = score_norm(a)

    eta = 0.15
    beta = 0.6
    vel = np.zeros(n)
    t0 = time.monotonic()
    it = 0
    last_improve = 0

    while time.monotonic() - t0 < time_budget:
        c = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(c))

        # Multi-scale smoothed gradient
        g = np.zeros(n)
        for s in range(n_scales):
            tau = tau_base * (2.0 ** s) * float(np.max(c))
            if tau < 1e-18:
                tau = 1e-
            shifted = c / tau
            shifted -= shifted.max()
            w = np.exp(shifted)
            w_sum = w.sum()
            if w_sum < 1e-300:
                w = np.zeros(2 * n - 1)
                w[np.argmax(c)] = 1.0
            else:
                w /= w_sum
            # Gradient: 2 * (W @ a) where W[i,j] = w[i+j]
            g += 2.0 * (build_W(w, n) @ a)
        g /= n_scales

        # Nesterov-style momentum
        vel = beta * vel + (1.0 - beta) * g
        a_new = project_simplex(a - eta * vel)

        new_sc = score_norm(a_new)
        if new_sc <= cur_sc or rng.random() < 0.04:
            a = a_new
        else:
            a = project_simplex(a - 0.5 * eta * vel)

        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = it

        it += 1
        if it - last_improve > 3000:
            a = project_simplex(best_a + 0.03 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.1 + 0.1 * rng.random()
            last_improve = it

    return best_a, best_sc

                    eta_start=0.05 + 0.05 * rng.random(),
                    damping=0.5 + 0.3 * rng.random(),
                    restart_every=2000 + int(2000 * rng.random())
                )
            else:
                a_best, sc_best = smooth_grad_optimize(a_init, tb, rng)
                iters, evals = 0, 0

            global_iters += iters
            global_evals += evals

            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = a_best.copy()
                print(f"  [n={nv}, init={ii}] NEW BEST: {sc_best:.6f} (iters={iters})", flush=True)
            else:
                print(f"  [n={nv}, init={ii}] score={sc_best:.6f}", flush=True)

        print(f"[n={nv}] done. Overall best: {best_overall_sc:.6f}", flush=True)

    # Final polish phase
    if best_overall is not None:
        remain = deadline - time.monotonic() - 5.0
        if remain > 10.0:
            print(f"[polish] n={best_overall.size}, remaining={remain:.1f}s", flush=True)

            # QP-based micro-optimization
            pol_time1 = min(remain * 0.4, 120.0)
            a_p, sc_p, _, _ = qp_optimize(
                best_overall, pol_time1, rng,
                tau_start=30.0, tau_end=150.0,
                n_inner=10, eta_start=0.02, damping=0.3,
                restart_every=5000
            )
            if sc_p < best_overall_sc:
                best_overall_sc = sc_p
                best_overall = a_p.copy()
                print(f"[polish] QP improved to {sc_p:.6f}", flush=True)

            # Gradient polish
            if time.monotonic() < deadline - 10:
                pol_time2 = min(remain * 0.3, 60.0)
                a_p2, sc_p2 = smooth_grad_optimize(best_overall, pol_time2, rng)
                if sc_p2 < best_overall_sc:
                    best_overall_sc = sc_p2
                    best_overall = a_p2.copy()
                    print(f"[polish] grad improved to {sc_p2:.6f}", flush=True)

            # Random polish
            if time.monotonic() < deadline - 5:
                pol_time3 = min(remain * 0.2, 30.0)
                a_p3, sc_p3 = polish(best_overall, pol_time3, rng)
                if sc_p3 < best_overall_sc:
                    best_overall_sc = sc_p3
                    best_overall = a_p3.copy()
                    print(f"[polish] random improved to {sc_p3:.6f}", flush=True)

    # Fallback
    if best_overall is None:
        best_overall = np.ones(128) / 128
        best_overall_sc = 2.0
        print("[fallback] Using uniform", flush=True)

    # Final verification
    result = [float(max(0.0, min(1000.0, x))) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"[final] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}", flush=True)
    print(f"[stats] total_iters={global_iters}, total_evals={global_evals}", flush=True)

    return result
