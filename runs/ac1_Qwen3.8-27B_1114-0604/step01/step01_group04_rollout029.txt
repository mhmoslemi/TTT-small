"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using:
1. LP-based iterative method (SOTA-inspired)
2. Coordinate descent polish
3. Multiple sequence lengths and initializations
"""

import numpy as np
import time
from scipy.optimize import linprog

def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def conv_cross(f, g):
    """Computes conv(f, g) for arrays of same length n. Returns array of length 2n-1."""
    n = f.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    ff = np.fft.rfft(f, fsize)
    fg = np.fft.rfft(g, fsize)
    return np.fft.irfft(ff * fg, fsize)[:size]

def build_conv_matrix(f):
    """
    Build matrix A of shape (2n-1, n) such that A @ g = conv(f, g).
    A[k, m] = f[k-m] if 0 <= k-m <= n-1, else 0.
    """
    n = f.size
    A = np.zeros((2*n - 1, n))
    for k in range(2*n - 1):
        # A[k, m] = f[k-m] for valid m
        m_start = max(0, k - (n - 1))
        m_end = min(n - 1, k)
        for m in range(m_start, m_end + 1):
            A[k, m] = f[k - m]
    return A

def evaluate_normalized(a):
    """Compute score assuming a is already non-negative. Returns 2n*max(conv)/sum^2."""
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = full_conv(a)
    return 2.0 * n * float(np.max(b)) / (s * s)

def lp_iteration(f, M, max_iter=100):
    """
    Solve the LP: max sum(g) s.t. g>=0, conv(f,g) <= M.
    Returns the optimal g (unnormalized).
    """
    n = f.size
    A = build_conv_matrix(f)  # shape (2n-1, n)
    # Minimize -sum(g)
    c = -np.ones(n)
    # Constraints: A @ g <= M (vector of length 2n-1)
    b = np.full(2*n - 1, M)
    # Bounds: g >= 0
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    return None

"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using:
1. LP-based iterative optimization (SOTA-inspired: fix f, solve LP for g, blend)
2. Coordinate descent polish (exact 1D subproblem per coordinate)
3. Smoothed subgradient descent with multi-scale gradients
4. Multiple sequence lengths and diverse initializations
"""

import numpy as np
import time
from scipy.optimize import linprog


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    if n == 0:
        return np.array([0.0])
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def score_normalized(a):
    """Score assuming sum(a)=1, a>=0: returns 2n*max(conv(a,a))."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def build_conv_matrix(f):
    """
    Build matrix A of shape (2n-1, n) such that A @ g = conv(f, g).
    A[k, m] = f[k-m] if 0 <= k-m < n, else 0.
    """
    n = f.size
    A = np.zeros((2 * n - 1, n))
    for m in range(n):
        k_start = m
        k_end = min(2 * n - 2, m + n - 1)
        A[k_start:k_end + 1, m] = f[0:k_end - k_start + 1]
    return A


def lp_optimize(a, time_budget, rng, verbose=False):
    """
    LP-based iteration: fix f, solve LP max sum(g) s.t. g>=0, conv(f,g)<=M.
    Blend and repeat. Returns (best_a, best_score).
    """
    n = a.size
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a /= s

    best_a = a.copy()
    best_sc = score_normalized(a)
    t0 = time.monotonic()
    iters = 0
    last_improve = 0
    t = 0.5

    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        M = float(np.max(b))

        A = build_conv_matrix(a)
        c = -np.ones(n)
        b_ub = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        try:
            res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        except Exception:
            break

        if not res.success:
            break

        g = res.x
        g_sum = g.sum()
        if g_sum < 1e-10:
            break

        g_norm = g / g_sum

        # Try multiple blend values
        best_blend_sc = np.inf
        best_blend_a = a.copy()
        for tt in [t, t * 0.5, t * 0.75, 1.0, 0.1]:
            a_try = (1.0 - tt) * a + tt * g_norm
            a_try = np.maximum(a_try, 0.0)
            s_try = a_try.sum()
            if s_try < 1e-10:
                continue
            a_try /= s_try
            sc_try = score_normalized(a_try)
            if sc_try < best_blend_sc:
                best_blend_sc = sc_try
                best_blend_a = a_try.copy()

        if best_blend_sc < best_sc:
            best_sc = best_blend_sc
            best_a = best_blend_a.copy()
            last_improve = iters
            t = min(1.0, t * 1.05)
            a = best_blend_a.copy()
        else:
            # Accept if better than current a (even if not best overall)
            cur_sc = score_normalized(a)
            if best_blend_sc < cur_sc:
                a = best_blend_a.copy()
            t = max(0.1, t * 0.9)

        iters += 1

        if verbose and iters % 100 == 0:
            print(f"    LP iter {iters}: best={best_sc:.8f}, cur={score_normalized(a):.8f}, M={M:.8f}, t={t:.3f}")

        # Stall detection
        if iters - last_improve > 30:
            a = best_a + 0.005 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            t = 0.3 + 0.4 * rng.random()
            last_improve = iters
            if iters - last_improve > 100:
                break

    return best_a, best_sc


def coord_descent(a, time_budget, rng, verbose=False):
    """
    Coordinate descent polish. For each coordinate i, solve the 1D subproblem
    exactly (via grid + refinement) to minimize the score.
    """
    n = a.size
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return a, np.inf
    a /= s

    best_a = a.copy()
    best_sc = score_normalized(a)
    t0 = time.monotonic()
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_budget and no_improve_sweeps < 5:
        sweep_improved = False
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() - t0 > time_budget:
                break

            x_old = a[i]
            s_others = 1.0 - x_old

            # Compute conv with a[i]=0
            a[i] = 0.0
            conv0 = full_conv(a)
            a[i] = x_old

            # Precompute linear coefficients: c[k] = 2*a[k-i] for valid k, k != 2i
            c = np.zeros(2 * n - 1)
            m_start = max(0, i)
            m_end = min(2 * n - 2, i + n - 1)
            ks = np.arange(m_start, m_end + 1)
            ms = ks - i
            c[ks] = 2.0 * a[ms]
            # Handle the quadratic case at k = 2i
            quad_idx = 2 * i
            if quad_idx < 2 * n - 1:
                c[quad_idx] = 0.0  # no linear term, only quadratic

            has_quad = (quad_idx < 2 * n - 1)

            # Grid search for best x
            x_max = max(4.0 * x_old + 0.01, 4.0 / n)
            x_grid = np.linspace(0.0, x_max, 80)

            best_x = x_old
            best_val = np.inf

            for x in x_grid:
                Sw = s_others + x
                if Sw < 1e-10:
                    continue
                max_val = float(np.max(conv0 + c * x))
                if has_quad:
                    val_q = conv0[quad_idx] + x * x
                    if val_q > max_val:
                        max_val = val_q
                val = max_val / (Sw * Sw)
                if val < best_val:
                    best_val = val
                    best_x = x

            # Local refinement
            if best_x > 0 and best_x < x_max:
                x_fine = np.linspace(max(0, best_x - x_max / 200),
                                     min(x_max, best_x + x_max / 200), 30)
                for x in x_fine:
                    Sw = s_others + x
                    if Sw < 1e-10:
                        continue
                    max_val = float(np.max(conv0 + c * x))
                    if has_quad:
                        val_q = conv0[quad_idx] + x * x
                        if val_q > max_val:
                            max_val = val_q
                    val = max_val / (Sw * Sw)
                    if val < best_val:
                        best_val = val
                        best_x = x

            if abs(best_x - x_old) > 1e-12:
                a[i] = best_x
                s_new = a.sum()
                if s_new > 1e-10:
                    a /= s_new
                    sc = score_normalized(a)
                    if sc < best_sc - 1e-14:
                        best_sc = sc
                        best_a = a.copy()
                        sweep_improved = True

        if not sweep_improved:
            no_improve_sweeps += 1
            # Perturb if stuck
            if no_improve_sweeps >= 2:
                a = best_a + 0.003 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                no_improve_sweeps = 1
        else:
            no_improve_sweeps = 0

        if verbose:
            print(f"    CD sweep done: best={best_sc:.8f}, no_improve={no_improve_sweeps}")

    return best_a, best_sc


def smooth_grad(a, b, tau):
    """Smoothed subgradient at one temperature."""
    n = a.size
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        g = np.zeros(n)
        kmax = int(np.argmax(b))
        # d conv[kmax]/d a[i]
        for i in range(n):
            ki = kmax - i
            if 0 <= ki < n:
                if ki == i:
                    g[i] = 2.0 * a[i]
                else:
                    g[i] = 2.0 * a[ki]
        return g
    w /= s
    # g[i] = sum_k w[k] * d conv[k]/d a[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
    # This is 2 * conv(w, a) (cross-correlation)
    g = 2.0 * np.convolve(w, a, mode='valid')[:n]
    # Handle the diagonal correction (when k-i = i, the derivative is 2*a[i] not 2*a[k-i])
    # Actually for the smoothed version, the formula g[i] = 2*sum_k w[k]*a[k-i] is correct
    # because when k-i=i, a[k-i]=a[i], and the derivative is 2*a[i], which matches.
    # Wait: d conv[k]/d a[i] = 2*a[k-i] when k-i != i, and 2*a[i] when k-i=i.
    # But 2*a[k-i] = 2*a[i] when k-i=i. So the formula 2*sum_k w[k]*a[k-i] works for all cases!
    return g


def subgrad_optimize(a, time_budget, rng, verbose=False):
    """Smoothed subgradient descent with momentum and adaptive parameters."""
    n = a.size
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()

    best_a = a.copy()
    best_sc = score_normalized(a)

    eta = 0.15
    tau_frac = 0.03
    beta = 0.5
    vel = np.zeros(n)
    iters = 0
    last_improve = 0

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))
        tau = tau_frac * float(np.max(b))
        if tau < 1e-18:
            tau = 1e-18

        # Multi-scale gradient
        g = np.zeros(n)
        for s in range(3):
            g += smooth_grad(a, b, tau * (2.0 ** s))
        g /= 3.0

        vel = beta * vel + (1.0 - beta) * g
        a_new = a - eta * vel
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            break
        a_new /= s_new

        sc_new = score_normalized(a_new)

        if sc_new <= cur_sc or rng.random() < 0.05:
            a = a_new
        else:
            a = a - 0.3 * eta * vel
            a = np.maximum(a, 0.0)
            a /= a.sum()

        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a.copy()
            last_improve = iters

        iters += 1

        if iters - last_improve > 500:
            a = best_a + 0.02 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            vel = np.zeros(n)
            eta = 0.1 + 0.2 * rng.random()
            tau_frac = 0.02 + 0.04 * rng.random()
            beta = 0.3 + 0.4 * rng.random()
            last_improve = iters

        if iters % 5000 == 0:
            eta = 0.05 + 0.3 * rng.random()
            tau_frac = 0.01 + 0.05 * rng.random()

    return best_a, best_sc


def project_simplex(v):
    """Project onto probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.05
    return a / a.sum()


def init_two_cluster(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.15 + 0.15 * rng.random())
    c2 = n * (0.70 + 0.15 * rng.random())
    sigma = n * (0.04 + 0.08 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.02 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_u_shape(n):
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i).astype(np.float64) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_sided(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / c) ** 2 + 0.1
    return a / a.sum()


def init_noisy_uniform(n, rng):
    a = np.ones(n) + 0.8 * rng.standard_normal(n)
    a = np.abs(a) + 0.01
    return a / a.sum()


def init_sqrt_decay(n, rng):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a += 0.05 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_flat_target(n, rng):
    """Initialize with a pattern that tries to make conv flat."""
    a = np.ones(n) / n
    # Add structured noise to break symmetry
    a += 0.1 * rng.standard_normal(n)
    a = np.maximum(a, 0.0)
    return a / a.sum()


def get_inits(n, rng):

    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses LP-based iteration, coordinate descent, and subgradient methods
    across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    # Load previous best
    prev_a = None
    prev_sc = np.inf
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size >= 10 and np.all(np.isfinite(h)) and h.sum() > 0.01:
            prev_a = h / h.sum()
            prev_sc = score_normalized(prev_a)
            print(f"[init] Previous best: n={h.size}, score={prev_sc:.8f}")
    except Exception:
        pass

    best_overall = None
    best_overall_sc = np.inf

    if prev_a is not None and prev_sc < 2.0:
        best_overall = prev_a.copy()
        best_overall_sc = prev_sc

    # n values to try
    n_list = [64, 128, 256, 512, 1024]
    if prev_a is not None:
        np_ = prev_a.size
        if np_ not in n_list and 32 <= np_ <= 4096:
            n_list.append(np_)
        n_list = sorted(set(n_list))

    total_time = max(1.0, deadline - time.monotonic())
    print(f"[plan] n_list={n_list}, total_time={total_time:.1f}s")

    # Allocate time: more time for n values near the current best
    raw_w = []
    for nv in n_list:
        if prev_a is not None and nv == prev_a.size:
            raw_w.append(3.0)
        elif nv <= 256:
            raw_w.append(2.0)
        else:
            raw_w.append(1.0)
    total_w = sum(raw_w)
    weights = [w / total_w for w in raw_w]

    for idx_n, nv in enumerate(n_list):
        if time.monotonic() > deadline:
            break

        time_n = weights[idx_n] * total_time * 0.9
        print(f"[n={nv}] allocating {time_n:.1f}s")

        inits = get_inits(nv, rng)
        if prev_a is not None and prev_a.size == nv:
            inits.insert(0, prev_a.copy())

        # For each initialization: LP optimize + coord descent polish
        # Split time: 60% LP, 30% CD, 10% subgrad
        n_inits = len(inits)
        per_init_time = time_n / n_inits

        for ii, a_init in enumerate(inits):
            if time.monotonic() > deadline:
                break

            remain = deadline - time.monotonic() - 2.0
            if remain < 5.0:
                break

            tb = min(per_init_time, remain)

            # Phase 1: LP optimization
            lp_time = tb * 0.55
            if lp_time > 2.0:
                a_lp, sc_lp = lp_optimize(a_init, lp_time, rng)
            else:
                a_lp, sc_lp = a_init, score_normalized(a_init)

            # Phase 2: Coordinate descent polish
            cd_time = tb * 0.35
            if cd_time > 2.0:
                a_cd, sc_cd = coord_descent(a_lp, cd_time, rng)
            else:
                a_cd, sc_cd = a_lp, sc_lp

            # Phase 3: Short subgradient polish
            sg_time = tb * 0.1
            if sg_time > 2.0:
                a_sg, sc_sg = subgrad_optimize(a_cd, sg_time, rng)
            else:
                a_sg, sc_sg = a_cd, sc_cd

            # Pick best from all phases
            candidates = [(a_lp, sc_lp), (a_cd, sc_cd), (a_sg, sc_sg)]
            best_a, best_sc = min(candidates, key=lambda x: x[1])

            if best_sc < best_overall_sc:
                best_overall_sc = best_sc
                best_overall = best_a.copy()
                print(f"  [n={nv}, init={ii}] NEW BEST: score={best_sc:.8f}")
            else:
                print(f"  [n={nv}, init={ii}] score={best_sc:.8f} (best so far: {best_overall_sc:.8f})")

        print(f"[n={nv}] done. Overall best: {best_overall_sc:.8f}")

    # Final intensive polish on the best solution
    if best_overall is not None and time.monotonic() < deadline - 10.0:
        remain = deadline - time.monotonic() - 5.0
        if remain > 20.0:
            print(f"[final polish] remaining={remain:.1f}s, n={best_overall.size}, score={best_overall_sc:.8f}")

            # Extended coordinate descent
            cd_time = min(remain * 0.5, 600.0)
            a_p, sc_p = coord_descent(best_overall, cd_time, rng)
            if sc_p < best_overall_sc:
                best_overall_sc = sc_p
                best_overall = a_p.copy()
                print(f"  [final CD] improved to {sc_p:.8f}")

            # Extended LP
            if time.monotonic() < deadline - 10.0:
                lp_time = min((deadline - time.monotonic() - 5.0) * 0.4, 300.0)
                if lp_time > 5.0:
                    a_lp2, sc_lp2 = lp_optimize(best_overall, lp_time, rng)
                    if sc_lp2 < best_overall_sc:
                        best_overall_sc = sc_lp2
                        best_overall = a_lp2.copy()
                        print(f"  [final LP] improved to {sc_lp2:.8f}")

            # One more CD pass
            if time.monotonic() < deadline - 5.0:
                cd_time2 = min(remain * 0.2, 120.0)
                if cd_time2 > 3.0:
                    a_p2, sc_p2 = coord_descent(best_overall, cd_time2, rng)
                    if sc_p2 < best_overall_sc:
                        best_overall_sc = sc_p2
                        best_overall = a_p2.copy()
                        print(f"  [final CD2] improved to {sc_p2:.8f}")

    # Fallback
    if best_overall is None:
        best_overall = np.ones(128) / 128
        best_overall_sc = 2.0

    # Final verification
    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"[final] n={len(result)}, internal={best_overall_sc:.8f}, evaluator={final_sc:.8f}")

    return result
