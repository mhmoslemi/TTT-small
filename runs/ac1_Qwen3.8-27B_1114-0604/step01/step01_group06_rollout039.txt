def _fft_conv(a):
    """Compute linear convolution of a with itself using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    L = 2 * n - 1
    N = 1 << (L - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L].real
    return np.maximum(c, 0.0)


def _fast_score(a):
    """Compute the score efficiently using FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = len(a)
    if n == 0:
        return np.inf
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    c = _fft_conv(a)
    m = float(np.max(c))
    return float(2.0 * n * m / (s * s))


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - np.max(z)
    e = np.exp(z)
    s = float(np.sum(e))
    if s < 1e-300:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _build_conv_matrix(f):
    """Build sparse matrix A such that A @ g = np.convolve(f, g)."""
    n = len(f)
    A = lil_matrix((2 * n - 1, n), dtype=np.float64)
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            for i in range(i_lo, i_hi + 1):
                A[k, i] = f[k - i]
    return A.tocsr()


def _lp_step(f):
    """Solve the LP subproblem and return g (the Frank-Wolfe direction)."""
    n = len(f)
    b_conv = np.convolve(f, f)
    M = float(np.max(b_conv))
    if M <= 0 or not np.isfinite(M):
        return f.copy()
    M *= 1.0001
    A = _build_conv_matrix(f)
    c_obj = -np.ones(n)
    result = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M),
                     bounds=[(0, None)] * n, method='highs',
                     options={'presolve': True, 'time_limit': 180,
                              'dual_feasibility_tolerance': 1e-7,
                              'primal_feasibility_tolerance': 1e-7})
    if not result.success or result.x is None:
        return f.copy()
    g = np.maximum(result.x, 0.0)
    gs = float(np.sum(g))
    if gs < 1e-10:
        return f.copy()
    return g / gs


def _fw_line_search(f, g, f_score, deadline):
    """Find the best step size t for f_new = (1-t)*f + t*g."""
    best_f = f.copy()
    best_sc = f_score
    t_new = 0.0

        f_new = (1.0 - t) * f + t * g

        if sc < best_sc - 1e-12:
            best_sc = sc
            best_f = f_new.copy()
            t_new = t
        if time.monotonic() > deadline:
            break
    return best_f, best_sc, t_new


def _frank_wolfe(f_init, deadline):
    """Run the LP-based Frank-Wolfe method."""
    n = len(f_init)

    f /= np.sum(f)

    best_f = f.copy()
    best_score = f_score

    no_improve = 0
    it = 0

        g = _lp_step(f)
        f_new, sc_new, t_act = _fw_line_search(f, g, f_score, min(deadline, time.monotonic() + 200))
        it += 1

            best_score = sc_new
