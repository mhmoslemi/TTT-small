import numpy as np
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _score(a):
    """Fast score computation."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    return 2.0 * len(a) * c.max() / (s * s)

def _make_multi_harmonic(n, A1=0.0, A2=0.0, A3=0.0):
    """Create a multi-harmonic sinusoidal sequence."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / n
    val = 1.0 + A1 * np.sin(theta) + A2 * np.sin(2*theta) + A3 * np.sin(3*theta)
    val = np.clip(val, 0.0, None)
    a = val / val.sum()
    return a

def _gradient_step(a, K=10, eta=None):
    """One gradient step for minimizing max convolution."""
    n = len(a)
    c = _fft_convolve(a)
    
    # Find active set
    if len(c) <= K:
        active = np.arange(len(c))
    else:
        active = np.argpartition(c, -K)[-K:]
    
    # Compute subgradient
    g = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            idx = np.arange(j_min, j_max + 1)
            a_idx = k - idx
            g[idx] += 2.0 * a[a_idx]
    g /= len(active)
    
    # Project gradient onto the simplex tangent space
    g = g - g.mean()
    
    if eta is None:
        eta = 1.0 / (np.max(np.abs(g)) + 1e-10)
    
    # Step
    a_new = a - eta * g
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s > 0.01:
        a_new /= s
    else:
        a_new = a.copy()
    
    return a_new, c.max()

def _refine(a, budget_s=300.0, K=10):
    """Gradient refinement with adaptive learning rate."""
    start = time.monotonic()
    a = a.copy()
    best_a = a.copy()
    best_score = _score(a)
    
    eta = 1.0
    stagnant = 0
    
    for it in range(100000):
        if time.monotonic() - start > budget_s:
            break
        if it % 50 == 0 and it > 0:
            print(f"  iter {it}: score={best_score:.8f}, eta={eta:.4f}")
        
        a_new, _ = _gradient_step(a, K=K, eta=eta)
        s_new = _score(a_new)
        
        if s_new < best_score * (1 - 1e-9):
            a = a_new
            if s_new < best_score:
                best_score = s_new
                best_a = a.copy()
                stagnant = 0
            eta = min(eta * 1.05, 10.0)  # increase step
        else:
            stagnant += 1
            eta = max(eta * 0.8, 0.001)  # decrease step
            if stagnant > 200:
                break
    
    return best_a, best_score

"""
Low-peak nonnegative step-function search using multi-harmonic initialization
and gradient refinement with adaptive active-set management.
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_conv(a):
    """Linear convolution a*a via FFT."""
    n = a.size
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _score(a):
    """Compute the evaluation score."""
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_conv(a)
    mx = float(c.max())
    return float(2.0 * n * mx / (s * s))


def _make_harmonic(n, amps):
    """
    Create sequence: a_i = clip(1 + sum_m amp[m] * sin((m+1)*2*pi*(i+0.5)/n), 0) / sum
    """
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / n
    val = np.ones(n, dtype=np.float64)
    for m, amp in enumerate(amps):
        if abs(amp) > 1e-10:
            val += amp * np.sin((m + 1) * theta)
    val = np.clip(val, 0.0, None)
    s = val.sum()
    if s < 1e-10:
        return np.ones(n) / n
    return val / s


def _active_gradient(a, K=10):
    """
    Compute the subgradient of max_k (a*a)_k using the top-K active lags.
    Returns (gradient_array, max_value).
    """
    n = a.size
    c = _fft_conv(a)
    mx = float(c.max())
    
    if c.size <= K:
        active = np.arange(c.size)
    else:
        active = np.argpartition(c, -K)[-K:]
    
    g = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            idx = np.arange(j_min, j_max + 1)
            a_idx = k - idx
            g[idx] += 2.0 * a[a_idx]
    g /= float(len(active))
    return g, mx, c


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    if not np.any(cond):
        return np.ones(n) / n
    r = int(np.max(np.where(cond)[0])) + 1
    theta = (cssv[r - 1] - 1.0) / r
    return np.clip(v - theta, 0.0, None)


def _refine_gradient(a, budget_s=120.0, K=10, verbose=True):
    """
    Projected gradient refinement with adaptive step size and momentum.
    """
    start = time.monotonic()
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return a, np.inf
    a = a / s
    
    best_a = a.copy()
    best_score = _score(a)
    
    n = a.size
    lr = 1.0 / (np.max(a) * n)  # initial learning rate
    lr = min(lr, 1.0)
    momentum = 0.9
    v = np.zeros(n, dtype=np.float64)  # momentum buffer
    stagnant = 0
    it = 0
    
    while time.monotonic() - start < budget_s and it < 200000:
        it += 1
        g, mx, c = _active_gradient(a, K=K)
        
        # Project gradient onto simplex tangent (remove mean)
        g = g - g.mean()
        
        # Momentum update
        v = momentum * v - lr * g
        
        # Candidate step
        a_new = a + v
        a_new = np.clip(a_new, 0.0, None)
        sn = a_new.sum()
        if sn > 0.01:
            a_new = a_new / sn
        
        sc = _score(a_new)
        
        if np.isfinite(sc) and sc < best_score:
            a = a_new
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                stagnant = 0
            lr = min(lr * 1.02, 10.0 / (n * np.max(a) + 1e-10))
        else:
            stagnant += 1
            v = momentum * v  # keep momentum but don't take the step
            lr *= 0.9
            if stagnant > 500:
                # Try a random restart with small perturbation
                noise = np.random.randn(n) * 0.01
                a_new = a + noise
                a_new = np.clip(a_new, 0.0, None)
                a_new = a_new / a_new.sum()
                sc = _score(a_new)
                if sc < best_score:
                    a = a_new
                    stagnant = 0
                    lr = 1.0 / (n * np.max(a) + 1e-10)
                elif stagnant > 2000:
                    break
        
        if verbose and it % 2000 == 0:
            print(f"    refine iter {it}: score={best_score:.8f} lr={lr:.6f}")
    
    return best_a, best_score


def _scan_harmonics(deadline):
    """Scan multi-harmonic sinusoidal sequences."""
    best_a = None
    best_score = np.inf
    
    lengths = [4096, 2048, 8192]
    
    for n in lengths:
        if time.monotonic() > deadline:
            break
        
        # Scan A1

        best_sc_a1 = np.inf

            if sc < best_sc_a1:
                best_sc_a1 = sc
                best_a1 = a1
                if best_a is None or sc < best_score:
                    best_score = sc
                    best_a = a
        
        # Scan A2 given best A1
        best_a2 = 0.0
        best_sc = best_sc_a1
        for a2 in np.linspace(-0.4, 0.4, 33):
            if time.monotonic() > deadline:
                break
            a = _make_harmonic(n, [best_a1, a2])
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a2 = a2
                if sc < best_score:
                    best_score = sc
                    best_a = a
        
        # Scan A3 given best (A1, A2)
        best_a3 = 0.0
        best_sc2 = best_sc
        for a3 in np.linspace(-0.3, 0.3, 25):
            if time.monotonic() > deadline:
                break
            a = _make_harmonic(n, [best_a1, best_a2, a3])
            sc = _score(a)
            if sc < best_sc2:
                best_sc2 = sc
                best_a3 = a3
                if sc < best_score:
                    best_score = sc
                    best_a = a
        
        print(f"[scan] n={n} A1={best_a1:.4f} A2={best_a2:.4f} A3={best_a3:.4f} score={best_sc2:.8f}")
    
    if best_a is None:
        best_a = _make_harmonic(2048, [0.78])
        best_score = _score(best_a)

def propose_candidate(seed=42, budget_s=2400, **kwargs):
