Given f (length n), M = max(f*f) (the current peak), find g (length n) such that:
maximize: 1^T g
subject to: g >= 0
           (f * g)[k] <= M for k = 0, ..., 2n-2

Where (f*g)[k] = sum_{i=max(0,k-n+1)}^{min(k, n-1)} f[i] * g[k-i]

The constraint matrix A has rows corresponding to each lag k. For row k:
A[k, i] = f[k-i] if 0 <= k-i < n, else 0

So A is a banded matrix (specifically, a Toeplitz matrix if f is finite).

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _cross_convolve_fft(a, b):

    need = len(a) + len(b) - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)

    One LP refinement step. Given f and current peak M, find best response g.
    Uses active-set method for the LP.
    """
    n = len(f)
    
    # Build the cross-convolution constraint for a given g
    # We need (f*g)[k] for all k = 0, ..., 2n-2
    # (f*g)[k] = sum_i f[i]*g[k-i] where 0<=i<n and 0<=k-i<n

    conv_ff = _fft_convolve(f)
    active = np.argpartition(conv_ff, -K)[-K:]

    g = f.copy()

        A_ub = np.zeros((len(active), n))
        b_ub = np.full(len(active), M)

            # (f*g)[k] = sum_{i} f[i] * g[k-i] where 0 <= k-i < n
            # i ranges from max(0, k-(n-1)) to min(n-1, k)
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                j = k - i
                A_ub[idx, j] += f[i]

            return f

"""
Anti-palindromic low-peak sequence optimization.
Uses multi-harmonic anti-palindromic initialization followed by iterative
LP-based best-response refinement to minimize peak auto-convolution.
"""

import numpy as np
import time
from scipy.optimize import linprog


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _cross_convolve_fft(a, b):
    need = len(a) + len(b) - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)
    return c[:need]


def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.size == 0:
        return np.inf
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * arr.size * mx / (s * s))


def _make_antipal(n, amps):
    """Create anti-palindromic sequence: a[i] = (1 + sum_k amps[k]*sin(2*pi*k*(i+0.5)/n))/n."""
    i = np.arange(n, dtype=np.float64)
    f = np.zeros(n, dtype=np.float64)
    for idx, amp in enumerate(amps):
        k = idx + 1
        f += amp * np.sin(2.0 * np.pi * k * (i + 0.5) / n)
    a = (1.0 + f) / n
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        a = np.ones(n, dtype=np.float64) / n
    else:
        a = a / s
    return a


def _optimize_single_harmonic(n, deadline):
    best_score = np.inf
    best_amp = 0.7
    for amp in np.linspace(0.0, 1.0, 61):
        if time.monotonic() > deadline:
            break
        a = _make_antipal(n, [amp])
        sc = _internal_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_amp = float(amp)
    step = 0.01
    for _ in range(6):
        lo = max(0.0, best_amp - step)
        hi = min(1.0, best_amp + step)
        for amp in np.linspace(lo, hi, 21):
            if time.monotonic() > deadline:
                break
            a = _make_antipal(n, [amp])
            sc = _internal_score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_amp = float(amp)
        step *= 0.5
    best_a = _make_antipal(n, [best_amp])
    return best_amp, best_score, best_a


def _multi_harmonic_optimize(n, deadline):
    """Optimize up to 3 harmonic amplitudes via grid search."""
    amp1, sc1, a1 = _optimize_single_harmonic(n, deadline)
    best_score = sc1
    best_a = a1.copy()
    best_amps = [amp1]

    if time.monotonic() > deadline - 30:
        return best_a, best_score

    # Second harmonic
    best_amp2 = 0.0
    for amp2 in np.linspace(-0.6, 0.6, 25):
        if time.monotonic() > deadline:
            break
        a = _make_antipal(n, [amp1, amp2])
        sc = _internal_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            best_amps = [amp1, amp2]
            best_amp2 = float(amp2)

    # Refine second harmonic
    if len(best_amps) >= 2:
        for amp2 in np.linspace(max(-1, best_amp2 - 0.05), min(1, best_amp2 + 0.05), 21):
            if time.monotonic() > deadline:
                break
            a = _make_antipal(n, [amp1, amp2])
            sc = _internal_score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                best_amps = [amp1, amp2]

    # Third harmonic
    if len(best_amps) >= 2 and time.monotonic() < deadline - 30:
        for amp3 in np.linspace(-0.3, 0.3, 13):
            if time.monotonic() > deadline:
                break
            a = _make_antipal(n, best_amps + [amp3])
            sc = _internal_score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                best_amps = best_amps + [amp3]

    return best_a, best_score


def _build_conv_matrix(f, n):
    """Build matrix C such that C @ g = f*g (linear cross-convolution)."""
    C = np.zeros((2 * n - 1, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = f
    return C


def _solve_lp(f, M, n):
    """Solve LP: max sum(g) s.t. (f*g)[k] <= M for all k, g >= 0."""
    C = _build_conv_matrix(f, n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c_obj = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            return res.x
    except Exception:
        pass
    return f.copy()


def _lp_refine(f_init, budget_s=300.0, verbose=True):
    """Iterative LP-based best-response refinement."""
    t0 = time.monotonic()
    arr = np.asarray(f_init, dtype=np.float64).copy()
    n = len(arr)
    s = arr.sum()
    if s < 0.01:
        return arr, np.inf
    arr = arr / s

    best = arr.copy()
    best_score = _internal_score(best)
    if not np.isfinite(best_score):
        return best, best_score

    step_size = 0.4
    stagnation = 0
    it = 0
    lp_time_total = 0.0

    while time.monotonic() - t0 < budget_s:
        it += 1
        conv = _fft_convolve(arr)
        M = float(np.max(conv))
        if M <= 0 or not np.isfinite(M):
            break

        t_lp = time.monotonic()
        g = _solve_lp(arr, M, n)
        lp_time_total += time.monotonic() - t_lp

        sg = float(g.sum())
        if sg < 0.01:
            break
        g_norm = g / sg

        # Check similarity to current (avoid degenerate solutions)
        cos_sim = float(np.dot(arr, g_norm) / (np.sqrt(np.dot(arr, arr)) * np.sqrt(np.dot(g_norm, g_norm)) + 1e-15))
        if cos_sim < 0.3:
            step_size = max(0.05, step_size * 0.5)

        arr_new = (1.0 - step_size) * arr + step_size * g_norm
        arr_new = np.clip(arr_new, 0.0, None)
        ss = float(arr_new.sum())
        if ss < 0.01:
            step_size = max(0.05, step_size * 0.7)
            continue
        arr_new = arr_new / ss

        sc = _internal_score(arr_new)
        if np.isfinite(sc) and sc < best_score:
            best = arr_new.copy()
            best_score = sc
            arr = arr_new.copy()
            stagnation = 0
            step_size = min(0.6, step_size * 1.03)
            if verbose and it % 10 == 0:
                print(f"  LP it={it} score={sc:.8f} step={step_size:.4f} cos_sim={cos_sim:.4f} t={time.monotonic()-t0:.0f}s")
        else:
            stagnation += 1
            step_size = max(0.03, step_size * 0.85)
            if step_size < 0.03:
                break

        if stagnation > 30:
            break

        # Periodic check: if LP is too slow, reduce budget
        avg_lp_time = lp_time_total / it
        remaining = budget_s - (time.monotonic() - t0)
        if avg_lp_time * 5 > remaining:
            break

    return best, best_score


def _subgradient_refine(f_init, budget_s=60.0):
    """Light subgradient refinement on the simplex as a fast local optimizer."""
    t0 = time.monotonic()
    arr = np.asarray(f_init, dtype=np.float64).copy()
    n = len(arr)
    s = arr.sum()
    if s < 0.01:
        return arr, np.inf
    arr = arr / s
    best = arr.copy()
    best_score = _internal_score(best)
    if not np.isfinite(best_score):
        return best, best_score

    lr = 0.02
    K = 16
    it = 0
    last_improve = 0

    while time.monotonic() - t0 < budget_s:
        it += 1
        conv = _fft_convolve(arr)
        if conv.size == 0:
            break
        maxc = float(np.max(conv))
        if maxc <= 0:
            break

        if conv.size <= K:
            active = np.arange(conv.size)
        else:
            active = np.argpartition(conv, -K)[-K:]

        grad = np.zeros(n, dtype=np.float64)
        for kk in active:
            k = int(kk)
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                grad[i] += 2.0 * arr[k - i]
        grad /= float(len(active))
        grad -= np.mean(grad)

        if not np.any(grad):
            break

        exponent = -lr * grad
        exponent = np.clip(exponent, -1.5, 1.5)
        arr_new = arr * np.exp(exponent)
        arr_new = np.clip(arr_new, 0.0, None)
        ss = float(arr_new.sum())
        if ss < 0.01:
            break
        arr_new = arr_new / ss

        if it % 5 == 0:
            sc = _internal_score(arr_new)
            if np.isfinite(sc) and sc < best_score:
                best = arr_new.copy()
                best_score = sc
                arr = arr_new.copy()
                last_improve = it
                lr = min(0.1, lr * 1.05)
            else:
                lr = max(0.005, lr * 0.8)
                if it - last_improve > 200:
                    break

    return best, best_score


def _try_length(n, deadline, verbose=True):
    """Full pipeline for a given length n."""
    # Phase 1: Multi-harmonic initialization
    a, sc = _multi_harmonic_optimize(n, min(deadline - 800, time.monotonic() + 150))
    if verbose:
        print(f"  n={n} init score={sc:.8f}")

    # Phase 2: Subgradient warm-up (fast)
    if time.monotonic() < deadline - 700:
        a2, sc2 = _subgradient_refine(a, budget_s=40.0)
        if np.isfinite(sc2) and sc2 < sc:
            a = a2
            sc = sc2
            if verbose:
                print(f"  n={n} subgrad score={sc:.8f}")

    # Phase 3: LP refinement
    remaining = deadline - time.monotonic()
    if remaining > 100:
        a3, sc3 = _lp_refine(a, budget_s=min(remaining - 50, 1200.0), verbose=verbose)
        if np.isfinite(sc3) and sc3 < sc:
            a = a3
            sc = sc3
            if verbose:
                print(f"  n={n} LP score={sc:.8f}")

    return a, sc
