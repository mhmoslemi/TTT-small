    c = np.dot(grad_a, a)
    grad_z = a * (grad_a - c)
    return grad_z


def obj_and_grad(z):
    """Combined objective and gradient for scipy minimize."""
    tau = z.shape[0] * 0.02  # adaptive tau based on dimension
    val = lse_value(z, tau)
    grad = lse_grad(z, tau)
    return val, grad


def optimize_lbfgs(z_init, time_budget, rng):
    """Optimize using L-BFGS with adaptive temperature."""
    n = z_init.size
    best_z = z_init.copy()
    best_a = softmax_fn(z_init)
    best_score = compute_score(best_a)
    
    t0 = time.monotonic()
    tau = n * 0.05
    
    while time.monotonic() - t0 < time_budget:
        # Define objective with current tau
        def fun(z_tmp):
            return lse_value(z_tmp, tau)
        
        def jac(z_tmp):
            return lse_grad(z_tmp, tau)
        
        result = sp_minimize(
            fun, z_init, jac=jac, method='L-BFGS-B',
            options={'maxiter': 500, 'gtol': 1e-6, 'maxls': 50}
        )
        
        z_init = result.x
        a = softmax_fn(z_init)
        score = compute_score(a)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            best_z = z_init.copy()
        
        # Reduce temperature
        tau *= 0.7
        
        if time.monotonic() - t0 > time_budget * 0.8:
            break
    
    return best_a, best_score, best_z


def optimize_adam(z_init, time_budget, rng):
    """Adam optimizer with adaptive temperature for refinement."""
    n = z_init.size
    z = z_init.astype(np.float64).copy()
    m = np.zeros_like(z)
    v = np.zeros_like(z)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_z = z.copy()
    best_a = softmax_fn(z)
    best_score = compute_score(best_a)
    
    tau = n * 0.03
    lr = 0.05
    t0 = time.monotonic()
    it = 0
    last_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        val, grad = lse_value(z, tau), lse_grad(z, tau)
        
        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        a = softmax_fn(z)
        score = compute_score(a)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            last_improve = it
        
        # Adaptive schedules
        tau *= 0.995
        lr *= 0.999
        
        # Restart if stuck
        if it - last_improve > 5000:
            z = best_z + 0.1 * rng.standard_normal(n)
            m = np.zeros_like(z)
            v = np.zeros_like(z)
            lr = 0.03 + 0.05 * rng.random()
            tau = n * 0.03
            last_improve = it
    
    return best_a, best_score, best_z


def optimize_hybrid(a_init, n, time_budget, rng):
    """Hybrid optimization: L-BFGS then Adam refinement."""
    # Initialize z from a
    z = np.log(a_init + 1e-10)
    
    t0 = time.monotonic()
    # Phase 1: L-BFGS (coarse to fine)
    lbfgs_time = min(time_budget * 0.4, 60.0)
    if lbfgs_time > 5:
        a1, s1, z1 = optimize_lbfgs(z, lbfgs_time, rng)
    else:
        a1, s1, z1 = a_init.copy(), compute_score(a_init), z.copy()
    
    # Phase 2: Adam refinement
    remaining = time_budget - (time.monotonic() - t0) - 1.0
    if remaining > 5:
        a2, s2, z2 = optimize_adam(z1, remaining, rng)
        if s2 < s1:
            return a2, s2
    return a1, s1


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def init_asymmetric(n, rng):
    """Strongly asymmetric: heavy on one side."""
    i = np.arange(n, dtype=np.float64)
    center = n * (0.2 + 0.3 * rng.random())
    sigma = n * (0.1 + 0.15 * rng.random())
    a = np.exp(-((i - center) / sigma) ** 2)
    a += 0.05 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_two_lobe(n, rng):
    """Two lobes at asymmetric positions."""
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.1 + 0.2 * rng.random())
    c2 = n * (0.6 + 0.3 * rng.random())
    s1 = n * (0.05 + 0.1 * rng.random())
    s2 = n * (0.05 + 0.1 * rng.random())
    a = np.exp(-((i - c1) / s1) ** 2) + 0.8 * np.exp(-((i - c2) / s2) ** 2)
    a += 0.02 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_ramp(n, rng):
    """Linear ramp (strongly asymmetric)."""
    i = np.arange(n, dtype=np.float64)
    direction = rng.random()
    if direction < 0.5:
        a = i + 1.0
    else:
        a = (n - 1 - i) + 1.0
    a += 0.1 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_quadratic_asym(n, rng):
    """Quadratic, asymmetric peak."""
    i = np.arange(n, dtype=np.float64)
    c = n * (0.15 + 0.35 * rng.random())
    a = np.exp(-((i - c) / (n * 0.3)) ** 2)
    # Add a small tail on the other side
    other_c = n - c
    a += 0.3 * np.exp(-((i - other_c) / (n * 0.15)) ** 2)
    a += 0.05 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_sine_modulated(n, rng):
    """Sine-modulated envelope, asymmetric."""
    i = np.arange(n, dtype=np.float64)
    freq = 1.0 + 2.0 * rng.random()
    phase = 2.0 * np.pi * rng.random()
    envelope = np.exp(-((i - n * 0.3) / (n * 0.4)) ** 2)
    a = envelope * (1.0 + 0.5 * np.sin(2.0 * np.pi * freq * i / n + phase))
    a = np.abs(a) + 0.05
    return a / a.sum()


def init_flat_with_noise(n, rng):
    a = np.ones(n) + 0.5 * rng.standard_normal(n)
    a = np.abs(a) + 0.01
    return a / a.sum()


def get_inits(n, rng):
    """Generate diverse initializations for given n."""
    inits = []
    inits.append(init_asymmetric(n, rng))
    inits.append(init_two_lobe(n, rng))
    inits.append(init_ramp(n, rng))
    inits.append(init_quadratic_asym(n, rng))
    inits.append(init_sine_modulated(n, rng))
    inits.append(init_random(n, rng))
    inits.append(init_flat_with_noise(n, rng))
    return inits


def polish_direct(a, time_budget, rng):
    """Direct gradient polishing: minimize max(conv) using subgradient of max."""
    n = a.size
    a = a.copy()
    best_a = a.copy()
    best_score = compute_score(a)
    
    lr = 0.01
    t0 = time.monotonic()
    last_improve = 0
    it = 0
    
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        max_val = np.max(b)
        # Find all indices achieving (near) max
        threshold = max_val - 1e-10 * max_val
        hot = (b >= threshold).astype(np.float64)
        hot /= hot.sum()
        
        # Subgradient of max(conv) w.r.t. a
        grad = 2.0 * np.correlate(hot, a, mode='valid')
        
        # Projected gradient step on simplex
        a_new = a - lr * grad
        # Project to simplex {x >= 0, sum(x) = 1}
        a_new = project_simplex(a_new)
        
        # Check if this is a descent direction
        score_new = compute_score(a_new)
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new.copy()
            a = a_new.copy()
            last_improve = it
        else:
            # Try smaller step
            a_try = a - 0.3 * lr * grad
            a_try = project_simplex(a_try)
            score_try = compute_score(a_try)
            if score_try < best_score:
                best_score = score_try
                best_a = a_try.copy()
                a = a_try.copy()
                last_improve = it
        
        it += 1
        lr *= 0.999
        
        if it - last_improve > 3000:
            # Perturb and restart
            a = best_a + 0.01 * rng.standard_normal(n)
            a = project_simplex(a)
            lr = 0.005 + 0.02 * rng.random()
            last_improve = it
    
    return best_a, best_score


def project_simplex(v):
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses softmax parameterization with hybrid L-BFGS/Adam optimization,
    multi-restart, and multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    
    # Try to load previous best
    prev_a = None
    prev_score = np.inf
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size >= 10 and np.all(np.isfinite(h)) and h.sum() > 0.01:
            prev_a = h / h.sum()
            prev_score = compute_score(prev_a)
            print(f"[init] Previous best: n={h.size}, score={prev_score:.6f}")
    except Exception:
        pass
    
    best_overall = None
    best_overall_score = np.inf
    
    if prev_a is not None:
        best_overall = prev_a.copy()
        best_overall_score = prev_score
    
    # n values to try (prioritize n=128 since previous best was there)
    n_list = [128, 256, 512, 64]
    if prev_a is not None:
        np_ = prev_a.size
        if np_ not in n_list and 32 <= np_ <= 1024:
            n_list.insert(0, np_)
        n_list = sorted(set(n_list))
    
    # Time allocation: give more time to promising n values
    # Start with previous n, then others
    total_time = deadline - time.monotonic()
    print(f"[plan] n_list={n_list}, total_time={total_time:.1f}s")
    
    # Allocate time: 40% to first n (previous best), rest split
    if prev_a is not None:
        time_first = total_time * 0.35
        time_rest = total_time * 0.55
    else:
        time_first = total_time * 0.3
        time_rest = total_time * 0.6
    
    n_remaining = total_time * 0.1  # reserve for polish
    
    # Process n_list
    for idx_n, nv in enumerate(n_list):
        if time.monotonic() > deadline - n_remaining - 5:
            break
        
        if idx_n == 0:
            time_n = time_first
        else:
            time_n = time_rest / max(1, len(n_list) - 1)
        
        time_n = min(time_n, deadline - time.monotonic() - n_remaining - 5)
        if time_n < 10:
            break
        
        print(f"[n={nv}] allocating {time_n:.1f}s")
        
        inits = get_inits(nv, rng)
        
        # If we have previous best of this length, use it first
        if prev_a is not None and prev_a.size == nv:
            inits.insert(0, prev_a.copy())
        
        # Time per initialization
        n_inits = min(len(inits), max(1, int(time_n / 20) + 1))
        time_per_init = time_n / n_inits
        
        for ii in range(n_inits):
            if time.monotonic() > deadline - n_remaining - 5:
                break
            
            tb = min(time_per_init, deadline - time.monotonic() - n_remaining - 5)
            if tb < 5:
                break
            
            a_init = inits[ii] if ii < len(inits) else init_random(nv, rng)
            a_best, score_best = optimize_hybrid(a_init, nv, tb, rng)
            
            tag = "BEST" if score_best < best_overall_score else ""
            print(f"  [n={nv}, init={ii}] score={score_best:.6f} {tag}")
            
            if score_best < best_overall_score:
                best_overall_score = score_best
                best_overall = a_best.copy()
    
    # Final polish phase
    if best_overall is not None and time.monotonic() < deadline - 5:
        remain = deadline - time.monotonic() - 3.0
        if remain > 20:
            print(f"[polish] remaining={remain:.1f}s, n={best_overall.size}")
            
            # Direct subgradient polish
            pol_time = min(remain * 0.6, 120.0)
            a_p, sc_p = polish_direct(best_overall, pol_time, rng)
            if sc_p < best_overall_score:
                best_overall_score = sc_p
                best_overall = a_p.copy()
                print(f"[polish] direct improved to {sc_p:.6f}")
            
            # Adam polish
            if time.monotonic() < deadline - 5:
                remain2 = deadline - time.monotonic() - 3.0
                if remain2 > 10:
                    z = np.log(best_overall + 1e-10)
                    a_p2, sc_p2, _ = optimize_adam(z, min(remain2, 60.0), rng)
                    if sc_p2 < best_overall_score:
                        best_overall_score = sc_p2
                        best_overall = a_p2.copy()
                        print(f"[polish] adam improved to {sc_p2:.6f}")
    
    # Fallback
    if best_overall is None:
        best_overall = np.ones(128) / 128
        best_overall_score = 2.0
    
    # Ensure valid output
    result = [max(0.0, float(x)) for x in best_overall]
    if sum(result) < 0.01:
        result = [1.0 / 128] * 128
    
    final_sc = float(evaluate_sequence(result))
    print(f"[final] n={len(result)}, internal={best_overall_score:.6f}, evaluator={final_sc:.6f}")
    
    return result
