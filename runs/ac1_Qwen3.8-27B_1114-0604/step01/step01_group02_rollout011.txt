    A_sp = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # LP: max 1^T d s.t. A d <= M, d >= 0
    # linprog minimizes, so: min -1^T d s.t. A d <= M, d >= 0
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A_sp, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n,
                  method='highs', options={'time_limit': max_time})
    if res.success and res.x is not None:
        return res.x
    return None

def adam_step(a, grad, m, v, t, lr):
    """Perform an Adam update in-place."""
    b1, b2 = 0.9, 0.999
    eps = 1e-8
    m[:] = b1 * m + (1 - b1) * grad
    v[:] = b2 * v + (1 - b2) * grad * grad
    m_hat = m / (1 - b1**t)
    v_hat = v / (1 - b2**t)
    a[:] -= lr * m_hat / (np.sqrt(v_hat) + eps)
    a[:] = np.maximum(a, 0)
    a[:] = np.minimum(a, 1000)

"""
Step-function sequence optimizer using exact subgradient descent with Adam,
LP-based jolt directions, mass redistribution heuristics, and multi-length
exploration. Combines gradient-based local optimization with structured
global moves to escape local minima.
"""

import numpy as np
import time
import scipy.sparse as sp
from scipy.optimize import linprog


def fast_self_conv(a):
    """Fast linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    fft_len = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, fft_len)
    return np.fft.irfft(A * A, fft_len)[:M]


def compute_score(a):
    """Compute 2n * max(conv(a,a)) / (sum(a))^2."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fast_self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def exact_subgradient(a, c, rel_tol=1e-4):
    """Compute exact subgradient of max(c_k) w.r.t. a."""
    n = len(a)
    M = float(c.max())
    if M < 1e-15:
        return np.zeros(n), M
    threshold = M * (1.0 - rel_tol)
    active = np.where(c >= threshold)[0]
    grad = np.zeros(n)
    for k in active:
        ks = int(k)
        j_lo = max(0, ks - n + 1)
        j_hi = min(ks, n - 1)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += 2.0 * a[j_lo:j_hi + 1][::-1]
    grad /= len(active)
    return grad, M


def full_gradient(a, c, rel_tol=1e-4):
    """Full gradient of f(a) = 2n*max(c)/S^2."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-10:
        return np.zeros(n)
    grad_M, M = exact_subgradient(a, c, rel_tol)
    return 2.0 * n * (grad_M / (S * S) - 2.0 * M / (S * S * S))


def adam_update(a, g, m, v, t, lr):
    """Adam update in-place on a."""
    b1, b2, eps = 0.9, 0.999, 1e-8
    m[:] = b1 * m + (1.0 - b1) * g
    v[:] = b2 * v + (1.0 - b2) * g * g
    m_hat = m / (1.0 - b1 ** t)
    v_hat = v / (1.0 - b2 ** t)
    a[:] -= lr * m_hat / (np.sqrt(v_hat) + eps)
    np.maximum(a, 0.0, out=a)
    np.minimum(a, 1000.0, out=a)


def mass_redistribute(a, c, frac=0.05):
    """
    Simple jolt: move mass from positions contributing most to the peak
    to positions contributing least. Returns modified copy.
    """
    n = len(a)
    M_idx = int(np.argmax(c))
    # Contribution of a[j] to c[M_idx]: a[M_idx - j] for valid j
    j_lo = max(0, M_idx - n + 1)
    j_hi = min(M_idx, n - 1)
    contrib = np.zeros(n)
    contrib[j_lo:j_hi + 1] = a[j_lo:j_hi + 1][:] * a[j_hi - (j_hi - j_lo):::-1][:j_hi - j_lo + 1]
    # Actually just use the product with reversed slice
    contrib[j_lo:j_hi + 1] = a[j_lo:j_hi + 1] * a[j_lo:j_hi + 1][::-1]
    
    # Find top and bottom contributors
    if n < 10:
        return a.copy()
    k = max(1, n // 20)
    top_idx = np.argsort(contrib)[-k:]
    bot_idx = np.argsort(contrib)[:k]
    
    # Move mass from top to bottom
    move_amount = frac * a[top_idx].mean()
    a_new = a.copy()
    a_new[top_idx] -= move_amount
    a_new[bot_idx] += move_amount * len(top_idx) / len(bot_idx)
    np.maximum(a_new, 0.0, out=a_new)
    return a_new


def precompute_lp_indices(n):
    """Precompute row/col indices for the LP constraint matrix."""
    rows_parts = []
    cols_parts = []
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        length = j_hi - j_lo + 1
        rows_parts.append(np.full(length, k, dtype=np.int32))
        cols_parts.append(np.arange(j_lo, j_hi + 1, dtype=np.int32))
    return np.concatenate(rows_parts), np.concatenate(cols_parts)


def solve_lp_jolt(a, M, rows, cols, time_limit=20):
    """Solve the LP jolt problem."""
    n = len(a)
    vals = a[cols]
    A_sp = sp.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    try:
        res = linprog(c_obj, A_ub=A_sp, b_ub=b_ub,
                      bounds=[(0.0, None)] * n,
                      method='highs',
                      options={'time_limit': time_limit, 'presolve': True})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def optimize_subgradient(a0, end_time, rng, lr_init, topk_tol, patience=200):
    """Run subgradient + Adam optimization until end_time."""
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    c = fast_self_conv(a)
    S = a.sum()
    score = 2.0 * n * float(c.max()) / (S * S) if S > 1e-10 else np.inf
    best_score = score
    best_a = a.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr_init
    no_improve = 0
    local_patience = patience

    while time.time() < end_time:
        c = fast_self_conv(a)
        S = a.sum()
        if S < 1e-6:
            a[:] = 1.0
            c = fast_self_conv(a)
            S = a.sum()
        score = 2.0 * n * float(c.max()) / (S * S)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= local_patience:
            scale = 0.03 + 0.07 * rng.random()
            a = best_a * (1.0 + scale * rng.standard_normal(n))
            np.maximum(a, 0.0, out=a)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            lr = max(0.001, lr * 0.6)
            no_improve = 0
            local_patience = max(50, local_patience // 2)
            continue

        g = full_gradient(a, c, rel_tol=topk_tol)
        t += 1
        adam_update(a, g, m, v, t, lr)
        lr = lr * 0.9995  # slow decay

    return best_score, best_a


def optimize_with_lp(a0, end_time, rng, n, rows, cols, lr_init=0.05):
    """Optimize with periodic LP jolts."""
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    c = fast_self_conv(a)
    S = a.sum()
    score = 2.0 * n * float(c.max()) / (S * S) if S > 1e-10 else np.inf
    best_score = score
    best_a = a.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr_init
    iter_count = 0
    lp_interval = max(50, n // 20)  # solve LP every this many iterations
    last_lp_time = 0

    while time.time() < end_time:
        # Gradient step
        c = fast_self_conv(a)
        S = a.sum()
        if S < 1e-6:
            a[:] = 1.0
            continue
        score = 2.0 * n * float(c.max()) / (S * S)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()

        g = full_gradient(a, c, rel_tol=1e-4)
        t += 1
        adam_update(a, g, m, v, t, lr)
        iter_count += 1

        # Periodic LP jolt
        now = time.time()
        if iter_count % lp_interval == 0 and (now - last_lp_time) > 15 and (end_time - now) > 30:
            c_cur = fast_self_conv(a)
            M_cur = float(c_cur.max())
            S_cur = float(a.sum())
            d = solve_lp_jolt(a, M_cur, rows, cols, time_limit=15)
            last_lp_time = now
            if d is not None and d.sum() > 1e-10:
                d_scaled = d * (S_cur / d.sum())
                blend = 0.2 + 0.3 * rng.random()
                a = (1.0 - blend) * a + blend * d_scaled
                np.maximum(a, 0.0, out=a)
                np.minimum(a, 1000.0, out=a)
                m[:] = 0.0
                v[:] = 0.0
                t = 0
                lr = max(0.003, lr * 0.8)

        lr *= 0.999

    return best_score, best_a


def try_length(a_good, n_target, rng):
    """Resample a good sequence to a different length."""
    if n_target == len(a_good):
        return a_good.copy()
    x_old = np.linspace(0, 1, len(a_good))
    x_new = np.linspace(0, 1, n_target)
    a_new = np.interp(x_new, x_old, a_good)
    return np.clip(a_new, 0.0, 1000.0)
