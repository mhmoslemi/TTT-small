"""
Optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

The returned sequence is normalized to have L1 norm 1, so the score is
2*n*max(conv(p,p)).  The main search is a mirror-descent / exponentiated
gradient method with smoothed active-set weights for the convolution
coefficients.  Several diverse starting points are generated and optimized.
A final polishing phase uses a harder active set and occasional direction
searches based on minimizing cross-convolutions with the current sequence.
"""

import time
import numpy as np
from operator import itemgetter


def next_fft_len(L):
    L = int(L)
    if L <= 1:
        return 1
    return 1 << (L - 1).bit_length()


def normalize_positive(a, floor=0.0):
    y = np.asarray(a, dtype=np.float64).ravel().copy()
    if y.size == 0:
        return y
    y = np.nan_to_num(y, nan=0.0, posinf=1000.0, neginf=0.0)
    y = np.clip(y, 0.0, 1000.0)
    s = float(y.sum())
    if s <= 1e-12 or not np.isfinite(s):
        return np.ones(y.size, dtype=np.float64) / y.size
    y /= s
    if floor > 0.0:
        y = np.maximum(y, floor)
        y /= float(y.sum())
    return y


def convolve_p(p, n, size, P=None):
    if P is None:
        P = np.fft.rfft(p, size)
    full = np.fft.irfft(P * np.conjugate(P), size)
    L = 2 * int(n) - 1
    c = full[:L].copy()
    np.maximum(c, 0.0, out=c)
    return c, P


def fast_score_sequence(p, n, size=0):
    p = np.asarray(p, dtype=np.float64).ravel()
    if p.size == 0:
        return np.inf
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.inf
    if size <= 0:
        size = next_fft_len(2 * p.size - 1)
    p = p / s
    c, _ = convolve_p(p, p.size, size)
    m = float(c.max())
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * p.size * m)


def make_weights(c, alpha):
    L = len(c)
    m = float(c.max())
    if L == 0:
        return np.zeros(0, dtype=np.float64)
    if m <= 0.0 or not np.isfinite(m):
        w = np.zeros(L, dtype=np.float64)
        w[0] = 1.0
        return w
    rel = c / m
    np.clip(rel, 0.0, 1.0, out=rel)
    with np.errstate(over="ignore", under="ignore", invalid="ignore"):
        w = np.power(rel, float(alpha))
    w = np.where(np.isfinite(w), w, 0.0)
    w = np.asarray(w, dtype=np.float64)
    s = float(w.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros(L, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= s
    return w


def make_weights_active(c, alpha, active):
    if active is None:
        return make_weights(c, alpha)
    L = len(c)
    m = float(c.max())
    w = np.zeros(L, dtype=np.float64)
    if L == 0:
        return w
    if m <= 0.0 or not np.isfinite(m):
        w[0] = 1.0
        return w
    active = np.asarray(active, dtype=int)
    if active.size == 0:
        w[int(np.argmax(c))] = 1.0
        return w
    vals = c[active] / m
    np.clip(vals, 0.0, 1.0, out=vals)
    with np.errstate(over="ignore", under="ignore", invalid="ignore"):
        wv = np.power(vals, float(alpha))
    wv = np.where(np.isfinite(wv), wv, 0.0)
    w[active] = wv
    s = float(w.sum())
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros(L, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= s
    return w


def gradient_weights(w, P, size, n):
    W = np.fft.rfft(w, size)
    full = np.fft.irfft(W * np.conjugate(P), size)
    g = full[:n].copy()
    np.maximum(g, 0.0, out=g)
    return g


def symmetrize_sequence(p):
    if p.size == 0:
        return p.copy()
    y = 0.5 * (p + p[::-1])
    return normalize_positive(y, 1e-18)


def perturb_sequence(p, rng, scale, best=None, best_mix=0.0):
    if p.size == 0:
        return p.copy()
    y = p * (1.0 + float(scale) * rng.standard_normal(p.size))
    y = np.maximum(y, 0.0)
    if best is not None and best_mix > 0.0:
        y = (1.0 - float(best_mix)) * y + float(best_mix) * best
    y = np.asarray(y, dtype=np.float64)
    y += 1e-15
    return normalize_positive(y, 1e-18)


def cross_direction(p, P, n, size, rng, max_iter=12, time_limit=20.0, alpha=8.0, step=0.35):
    L = 2 * int(n) - 1
    q = p.copy()
    Q = np.fft.rfft(q, size)
    full = np.fft.irfft(P * Q, size)
    r = full[:L].copy()
    np.maximum(r, 0.0, out=r)
    m = float(r.max())
    best_q = q.copy()
    best_m = m

    end = time.monotonic() + max(0.0, float(time_limit))
    a = float(alpha)
    st = float(step)

    for _ in range(int(max_iter)):
        if time.monotonic() > end:
            break
        if m <= 0.0 or not np.isfinite(m):
            break

        w = make_weights(r, a)
        W = np.fft.rfft(w, size)
        full_h = np.fft.irfft(W * np.conjugate(P), size)
        h = full_h[:n].copy()
        np.maximum(h, 0.0, out=h)

        hmax = float(h.max())
        hmin = float(h.min())
        span = hmax - hmin
        updated = False
        if np.isfinite(span) and span > 1e-300:
            hmean = float(np.dot(q, h))
            shift = h - hmean
            scaled = -st * shift / span
            scaled = np.clip(scaled, -15.0, 15.0)
            q_new = q * np.exp(scaled)
            snew = float(q_new.sum())
            if snew > 0.0 and np.isfinite(snew):
                q_new /= snew
                q_new = np.maximum(q_new, 1e-18)
                q_new /= float(q_new.sum())
                q = q_new
                updated = True

        if not updated:
            q = perturb_sequence(q, rng, 0.02)

        Q = np.fft.rfft(q, size)
        full = np.fft.irfft(P * Q, size)
        r = full[:L].copy()
        np.maximum(r, 0.0, out=r)
        m = float(r.max())
        if np.isfinite(m) and m < best_m:
            best_m = m
            best_q = q.copy()

        a = min(60.0, a * 1.25)
        st = max(0.05, st * 0.95)

    return best_q


def line_search_pq(p0, q, P0, c0, m0, n, size):
    L = 2 * int(n) - 1
    Pq = np.fft.rfft(q, size)

    full_cq = np.fft.irfft(Pq * np.conjugate(Pq), size)
    cq = full_cq[:L].copy()
    np.maximum(cq, 0.0, out=cq)

    full_cross = np.fft.irfft(P0 * Pq, size)
    cross = full_cross[:L].copy()
    np.maximum(cross, 0.0, out=cross)

    best_t = 0.0
    best_m = float(m0)
    ts = (0.0, 0.01, 0.02, 0.03, 0.05, 0.08, 0.12, 0.15, 0.20,
          0.30, 0.40, 0.50, 0.65, 0.80, 1.00)
    for t in ts:
        if t <= 0.0:
            continue
        a = 1.0 - t
        ct = (a * a) * c0 + (2.0 * a * t) * cross + (t * t) * cq
        mt = float(ct.max())
        if np.isfinite(mt) and mt < best_m * (1.0 - 1e-12):
            best_m = mt
            best_t = t

    if best_t == 0.0:
        return p0.copy(), best_m

    pnew = (1.0 - best_t) * p0 + best_t * q
    pnew = normalize_positive(pnew, 1e-18)
    return pnew, best_m


def optimize_mirror(p0, n, rng, deadline,
                    label="",
                    alpha_start=1.0, alpha_end=25.0,
                    step_start=0.50, step_end=0.02,

    p = normalize_positive(p0, 1e-18)
    n = int(p.size)
    size = next_fft_len(2 * n - 1)
    L = 2 * n - 1

    P, c = convolve_p(p, n, size)
    m = float(c.max())
    best_p = p.copy()
    best_m = m
    best_score = 2.0 * n * best_m
    if not np.isfinite(best_score):
        p = np.ones(n, dtype=np.float64) / n
        P, c = convolve_p(p, n, size)
        m = float(c.max())

        best_m = m

    start_time = time.monotonic()
    alloc = max(1.0, float(deadline) - start_time)
    step_factor = 1.0
    no_improve = 0
    it = 0
    active = None
    use_top = int(top_k) > 0 and int(top_k) < L
    last_cross = -10 ** 9

    if cross_mode == "off":
        use_cross = False
        cross_gap = 10 ** 9
        cross_time = 0.0
    elif cross_mode == "polish":
        use_cross = True
        cross_gap = 900
        cross_time = 30.0
    else:
        use_cross = True
        cross_gap = 2400
        cross_time = 22.0

    print(f"[{label}] start best score = {best_score:.10f}")

    while True:
        now = time.monotonic()
        if now >= float(deadline):
            break

        progress = (now - start_time) / alloc

            progress = 1.0

        alpha = float(alpha_start) + (float(alpha_end) - float(alpha_start)) * progress

        g = gradient_weights(w, P, size, n)
        gmax = float(g.max())

        if np.isfinite(span) and span > 1e-300:
            gmean = float(np.dot(p, g))

            scaled = np.clip(scaled, -15.0, 15.0)

                p = p_new
            else:

            if m < best_m - 1e-15:
                best_m = m
                best_p = p.copy()
                best_score = 2.0 * n * best_m

            else:

                if no_improve > 25:

                    if no_improve > 120:

                            p = best_p.copy()
                        scale = 0.02 + min(0.08, 0.01 * (no_improve // 10))
                        mix = 0.05 if no_improve > 420 else 0.0
                        p = perturb_sequence(p, rng, scale, best=best_p, best_mix=mix)
                        P, c = convolve_p(p, n, size)

                        step_factor = 1.0
                        no_improve = 0

            if m > best_m * 2.0 and no_improve > 35:
                p = best_p.copy()
                P, c = convolve_p(p, n, size)

                step_factor = 1.0

                P = Ps
                c = cs
                m = ms
                if m < best_m:
                    best_m = m
                    best_p = p.copy()
                    best_score = 2.0 * n * best_m
                    no_improve = 0

        if (use_cross and now < float(deadline) - 90.0

    ps = symmetrize_sequence(best_p)
    Ps, cs = convolve_p(ps, n, size)
    ms = float(cs.max())
    if np.isfinite(ms) and ms < best_m:

    return best_p, float(2.0 * n * best_m)


def add_start(starts, name, arr, n, size):
    p = normalize_positive(arr, 0.0)
    if p.size != n:
        p = normalize_positive(p, 0.0)
        if p.size != n:
            return
    c, _ = convolve_p(p, n, size)
    m = float(c.max())
    if not np.isfinite(m):
        return
    starts.append((float(2.0 * n * m), name, p))


def build_start_list(init, n, rng, size):
    starts = []
    n = int(n)
    j = np.arange(n, dtype=np.float64)
    if n > 1:
        center = 0.5 * (n - 1)
        denom = center if center > 0.0 else 1.0
        x = (j - center) / denom
    else:
        center = 0.0
        x = np.zeros(n, dtype=np.float64)

    if init is not None:
        p = normalize_positive(init, 0.0)
        add_start(starts, "given", p, n, size)

        for gamma in (0.6, 0.8, 1.2, 1.6, 2.2):
            arr = np.power(p, float(gamma))
            add_start(starts, f"pow_{gamma}", arr, n, size)

        add_start(starts, "sym_given", 0.5 * (p + p[::-1]), n, size)

        add_start(starts, "noise_small", p * (1.0 + 0.05 * rng.standard_normal(n)), n, size)
        add_start(starts, "noise_big", p * (1.0 + 0.15 * rng.standard_normal(n)), n, size)
        add_start(starts, "blend_rand", 0.7 * p + 0.3 * rng.exponential(1.0, size=n), n, size)

        if n >= 64:
            k1 = max(3, n // 80)
            k1 = 2 * (k1 // 2) + 1
            kern1 = np.ones(k1, dtype=np.float64) / k1
            add_start(starts, "smooth_80", np.convolve(p, kern1, mode="same"), n, size)

            k2 = max(3, n // 20)
            k2 = 2 * (k2 // 2) + 1
            kern2 = np.ones(k2, dtype=np.float64) / k2
            add_start(starts, "smooth_20", np.convolve(p, kern2, mode="same"), n, size)

    add_start(starts, "uniform", np.ones(n, dtype=np.float64), n, size)

    add_start(starts, "u2", 1.0 + 2.0 * x * x, n, size)
    add_start(starts, "u8", 1.0 + 8.0 * x * x, n, size)
    add_start(starts, "u30", 1.0 + 30.0 * x * x, n, size)

    if n > 8:
        sig = max(1.0, 0.02 * n)
        bim = (np.exp(-((j - 0.25 * n) ** 2) / (2.0 * sig * sig)) +
               np.exp(-((j - 0.75 * n) ** 2) / (2.0 * sig * sig)) + 0.01)
        add_start(starts, "bimodal2575", bim, n, size)

        trim = (np.exp(-((j - 0.2 * n) ** 2) / (2.0 * sig * sig)) +
                np.exp(-((j - 0.5 * n) ** 2) / (2.0 * sig * sig)) +
                np.exp(-((j - 0.8 * n) ** 2) / (2.0 * sig * sig)) + 0.01)
        add_start(starts, "trimodal", trim, n, size)

    add_start(starts, "abs_sin", 1.0 + 0.8 * np.abs(np.sin(2.0 * np.pi * j / n)), n, size)
    add_start(starts, "sin4", 1.0 + 0.8 * np.sin(4.0 * np.pi * j / n), n, size)
    add_start(starts, "sin8", 1.0 + 0.6 * np.sin(8.0 * np.pi * j / n), n, size)

    add_start(starts, "rand_exp", rng.exponential(1.0, size=n), n, size)
    add_start(starts, "rand_gamma", rng.gamma(2.0, size=n), n, size)
    add_start(starts, "rand_absnorm", np.abs(rng.standard_normal(n)) + 0.1, n, size)

    add_start(starts, "comb2", (np.mod(j, 2.0) + 0.35), n, size)
    add_start(starts, "comb3", (np.mod(j, 3.0) + 0.35), n, size)

    return starts


def get_initial_sequence():
    obj = globals().get("height_sequence_1", None)
    if obj is None:
        return None
    arr = np.asarray(obj, dtype=np.float64).ravel()
    if arr.size == 0:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s <= 1e-12:
        return None
    return arr / s


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 2400.0
    budget = min(max(budget, 60.0), 2400.0)

    start_all = time.monotonic()
    deadline = start_all + budget - 25.0
    rng = np.random.default_rng(seed)

    init = get_initial_sequence()
    if init is None:
        n = 6121
        init = np.ones(n, dtype=np.float64)
    else:
        n = int(init.size)

    if n < 64:
        n = 1024
        init = np.ones(n, dtype=np.float64)

    if n > 20000:
        target = 8192
        idx = np.linspace(0.0, float(n - 1), target)
        init = np.interp(idx, np.arange(n, dtype=np.float64), init)
        n = int(target)

    init = normalize_positive(init, 0.0)
    size = next_fft_len(2 * n - 1)

    starts = build_start_list(init, n, rng, size)
    finite_starts = []
    for item in starts:
        if np.isfinite(item[0]):
            finite_starts.append(item)
    starts = finite_starts
    if not starts:
        starts.append((2.0, "uniform_safe", np.ones(n, dtype=np.float64) / n))

    starts.sort(key=itemgetter(0))

    best_p = starts[0][2].copy()
    best_score = float(starts[0][0])
    print(f"Best initial start score = {best_score:.10f}")

    final_alloc = max(90.0, min(420.0, budget * 0.14))
    main_budget = max(90.0, budget - final_alloc - 30.0)

    if len(starts) == 1:
        fracs = [1.0]
    elif len(starts) == 2:
        fracs = [0.70, 0.30]
    elif len(starts) == 3:
        fracs = [0.58, 0.24, 0.18]
    else:
        fracs = [0.52, 0.21, 0.15, 0.12]

    fracs = fracs[:len(starts)]
    sfx = float(sum(fracs))
    if sfx <= 0.0:
        fracs = [1.0 / len(fracs)] * len(fracs)
        sfx = 1.0
    fracs = [f / sfx for f in fracs]

    for idx, (sc, name, p0) in enumerate(starts[:min(4, len(starts))]):
        if time.monotonic() > deadline - 60.0:
            break
        alloc = main_budget * fracs[idx]
        sub_deadline = min(deadline - 30.0, time.monotonic() + alloc)
        if sub_deadline <= time.monotonic() + 5.0:
            continue

        if time.monotonic() > deadline - 90.0:
            after = 1.0
            alpha_end = 18.0
            step_start = 0.35
        else:
            after = 1.0
            alpha_end = 25.0
            step_start = 0.50

        p_opt, sc_opt = optimize_mirror(
            p0,
            n,
            rng,
            sub_deadline,
            label=f"start_{name}",
            alpha_start=after,
            alpha_end=alpha_end,
            step_start=step_start,
            step_end=0.02,
            top_k=0 if idx == 0 else 16,
            top_interval=60,
            cross_mode="default"
        )

        if np.isfinite(sc_opt) and sc_opt < best_score:
            best_score = float(sc_opt)
            best_p = p_opt
            print(f"New best after start {name}: {best_score:.10f}")

    remaining = deadline - time.monotonic()
    if remaining > 90.0:
        sub_deadline = min(deadline - 5.0, time.monotonic() + min(remaining - 10.0, 420.0))
        p_pol, sc_pol = optimize_mirror(
            best_p,
            n,
            rng,
            sub_deadline,
            label="final_polish",
            alpha_start=8.0,
            alpha_end=38.0,
            step_start=0.14,
            step_end=0.007,
            top_k=32,
            top_interval=40,
            cross_mode="polish"
        )
        if np.isfinite(sc_pol) and sc_pol < best_score:
            best_score = float(sc_pol)
            best_p = p_pol
            print(f"Polished best: {best_score:.10f}")

    # A few cheap deterministic post-processes.
    candidates = [(best_score, best_p)]
    candidates.append((None, symmetrize_sequence(best_p)))
    for gamma in (0.8, 0.9, 1.1, 1.25):
        candidates.append((None, normalize_positive(np.power(best_p, float(gamma)), 1e-18)))

    for _, cand in candidates:
        if cand is None or cand.size != n:
            continue
        c, _ = convolve_p(cand, n, size)
        m = float(c.max())
        if not np.isfinite(m):
            continue
        s = float(2.0 * n * m)
        if s < best_score:
            best_score = s
            best_p = cand

    best_p = np.asarray(best_p, dtype=np.float64).ravel()
    if best_p.size != n or not np.all(np.isfinite(best_p)):
        best_p = np.ones(n, dtype=np.float64) / n
    best_p = normalize_positive(best_p, 1e-18)

    final_list = [float(v) for v in best_p]

    direct_score = fast_score_sequence(best_p, n, size)
    if n <= 12000:
        try:
            eval_score = float(evaluate_sequence(final_list))
        except Exception:
            eval_score = direct_score
    else:
        eval_score = direct_score

    if not np.isfinite(direct_score):
        direct_score = eval_score
    if not np.isfinite(eval_score):
        eval_score = direct_score

    print(f"Final direct score: {direct_score:.10f}")
    print(f"Final evaluate_sequence score: {eval_score:.10f}")
    print(f"Sequence length: {n}")

    return final_list
