    if S < 1e-10:
        return np.zeros(n)
    smv, sw = compute_soft_max(c, temp)
    r = np.correlate(sw, x, mode='valid')
    d_sm = 2.0 * r
    return 2.0 * n * (d_sm * S - 2.0 * smv) / (S * S * S)


def init_uniform(n):
    return np.ones(n)


def init_cosine(n, freq, phase, alpha, rng):
    x = np.arange(n)
    a = 1.0 + alpha * np.cos(2.0 * np.pi * freq * x / n + phase)
    return np.clip(a, 0.01, 1000.0)


def init_multi_cosine(n, rng):
    a = np.ones(n)
    x = np.arange(n)
    for _ in range(rng.integers(2, 6)):
        freq = rng.integers(1, n // 4 + 1)
        phase = rng.uniform(0, 2 * np.pi)
        alpha = rng.uniform(0.2, 0.8)
        a += alpha * np.cos(2.0 * np.pi * freq * x / n + phase)

def init_ramp(n, rng):
    x = np.linspace(0, 1, n)
    alpha = rng.uniform(0.5, 2.0)
    a = 1.0 + alpha * (2.0 * x - 1.0)
    return np.clip(a, 0.01, 1000.0)


def init_blocks(n, rng):
    nb = rng.integers(4, 32)
    per = n // nb
    vals = rng.uniform(0.2, 1.5, nb)
    a = np.repeat(vals, per)
    if len(a) < n:
        a = np.concatenate([a, vals[:n - len(a)]])
    a = a[:n]
    return np.clip(a, 0.01, 1000.0)


def run_single_optimize(a0, n, max_time, rng, verbose=False):
    """Run temperature-annealed L-BFGS from initial point a0."""
    best_a = np.clip(a0.astype(np.float64), 0, 1000).copy()
    c = fast_self_conv(best_a)
    best_score = score_from_conv(n, c, best_a.sum())
    
    if verbose:
        print("  Initial score: %.6f" % best_score)
    
    start = time.monotonic()
    x_current = best_a.copy()
    
    # Temperature schedule: start smooth, get sharper
    n_steps = 4
    temps = []
    t0 = 1.0
    t_end = 0.005
    for i in range(n_steps):
        frac = i / max(1, n_steps - 1)
        temps.append(t0 * (t_end / t0) ** frac)
    
    for temp in temps:
        if time.monotonic() - start > max_time:
            break
        
        time_left = max_time - (time.monotonic() - start)
        # Scale max iterations by time remaining
        max_iter = max(50, min(300, int(100 * time_left / max_time * n_steps)))
        
        bounds = [(0, 1000)] * n
        func = partial(objective_func, n=n, temp=temp)
        jac = partial(gradient_func, n=n, temp=temp)
        
        try:
            res = minimize(func, x_current, jac=jac, method='L-BFGS-B',
                           bounds=bounds,
                           options={'maxiter': max_iter, 'maxfun': max_iter * 3,
                                    'ftol': 1e-12, 'gtol': 1e-10,
                                    'maxls': 50})
            x_candidate = np.maximum(res.x, 0)
        except Exception:
            break
        
        # Evaluate true score
        c = fast_self_conv(x_candidate)
        S = float(x_candidate.sum())
        sc = score_from_conv(n, c, S)
        
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = x_candidate.copy()
            if verbose:
                print("  temp=%.4f -> score=%.6f" % (temp, best_score))
        
        x_current = x_candidate
    
    return best_score, best_a


def subgradient_kick(a, n, n_steps=50, step_size=0.1, rng=None):
    """Apply subgradient steps to escape local minima."""
    a = a.copy()
    for _ in range(n_steps):
        c = fast_self_conv(a)
        S = float(a.sum())
        max_idx = np.argmax(c)
        M = c[max_idx]
        
        # Subgradient of max at a
        j_start = max(0, max_idx - n + 1)
        j_end = min(max_idx, n - 1)
        subgrad = np.zeros(n)
        subgrad[j_start:j_end+1] = 2.0 * a[j_start:j_end+1][::-1]
        
        # Direction: reduce the peak, increase the sum
        # We want to move in direction -subgrad / ||subgrad||
        # But also maintain sum
        sg_norm = np.linalg.norm(subgrad)
        if sg_norm < 1e-10:
            break
        
        # Projected gradient step
        a_new = a - step_size * subgrad / sg_norm
        a_new = np.maximum(a_new, 0)
        
        # Normalize to maintain sum (optional)
        a = a_new
    
    return a

"""
Optimized search for minimizing 2n*max(conv(a,a))/(sum a)^2.

Strategy: Temperature-annealed L-BFGS-B with soft-max smoothing,
combined with subgradient kicks for escaping local minima.
Multiple diverse initializations explored including parent-based.
"""

import numpy as np
import time
from scipy.optimize import minimize
from functools import partial


def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def score_from_conv(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def compute_soft_max(c, temp):
    c_clip = np.maximum(c, 1e-15)
    scaled = c_clip / temp
    sm_val = np.max(scaled)
    es = np.exp(scaled - sm_val)
    se = np.sum(es)
    sw = es / se
    smv = temp * (np.log(se) + sm_val)
    return smv, sw


def objective_func(x, n, temp):
    x = np.maximum(x, 0.0)
    c = fast_self_conv(x)
    S = float(x.sum())
    if S < 1e-10:
        return 1e10
    smv, _ = compute_soft_max(c, temp)
    return 2.0 * n * smv / (S * S)


def gradient_func(x, n, temp):
    x = np.maximum(x, 0.0)
    c = fast_self_conv(x)
    S = float(x.sum())
    if S < 1e-10:
        return np.zeros(n)
    smv, sw = compute_soft_max(c, temp)
    r = np.correlate(sw, x, mode='valid')
    d_sm = 2.0 * r
    return 2.0 * n * (d_sm * S - 2.0 * smv) / (S * S * S)

    return np.ones(n)
