def softmax(x):
    x = x - np.max(x)
    e = np.exp(x)
    return e / np.sum(e)


def log_soft_optimize(x0, n, deadline, beta_start=3000.0, beta_end=500000.0,
                      momentum=0.9, max_iter=1000000):
    """Nesterov-accelerated optimization in log-space.
    
    x are log-weights, p = softmax(x) is the probability vector.
    We minimize n * smooth_max(p*p) w.r.t. x.
    """
    x = np.asarray(x0, dtype=np.float64).copy()
    n = x.size
    
    # Initialize
    p = softmax(x)
    b = fft_convolution(p)
    G, w = smooth_max_grad(b, beta_start)
    best_p = p.copy()
    best_score = float(2*n*np.max(b))  # with sum(p)=1, score = 2n*max(b)
    best_G = float(n*G)
    
    v = np.zeros(n, dtype=np.float64)  # momentum velocity
    alpha = 1.0  # step size
    no_improve = 0
    
    total_iters = max_iter
    for it in range(total_iters):
        if time.monotonic() >= deadline:
            break
        
        # Adaptive beta: linearly interpolate from beta_start to beta_end
        frac = float(it) / float(total_iters)
        beta = beta_start + (beta_end - beta_start) * frac
        
        # Recompute with current beta
        b = fft_convolution(p)
        G, w = smooth_max_grad(b, beta)
        if not np.isfinite(G):
            beta = max(1000.0, beta * 0.5)
            b = fft_convolution(p)
            G, w = smooth_max_grad(b, beta)
            if not np.isfinite(G):
                break
        
        cur_val = float(n * G)
        cur_score = float(2*n*np.max(b))
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if it % 1000 == 0:
            print(f"  log-soft it={it} beta={beta:.0f} best_score={best_score:.8f} no_improve={no_improve}")
        
        if no_improve > 5000:
            # Try reducing momentum or resetting
            if no_improve > 10000:
                break
            v *= 0.5
        
        # Compute gradient in log-space
        gp = conv_grad(p, w)  # gradient w.r.t. p, shape (n,)
        # grad_x_j = p_j * (gp_j - sum_i gp_i * p_i)
        gp_dot_p = float(np.dot(gp, p))
        grad_x = p * (gp - gp_dot_p)
        
        # Nesterov momentum update
        v = momentum * v + grad_x
        v_norm = float(np.linalg.norm(v))
        if not np.isfinite(v_norm) or v_norm < 1e-15:
            break
        
        # Adaptive step size
        # Lipschitz constant ~ beta * 2 * max(p) * max(gp) roughly
        # Safe step: alpha = 1 / (beta * 4 * max_p * max_gp)
        max_p = float(np.max(p))
        max_gp = float(np.max(np.abs(gp)))
        if max_gp > 0 and max_p > 0:
            alpha = min(alpha, 0.5 / (beta * max_p * max_gp + 1e-12))
        
        # Try the step
        step_accepted = False
        for factor in (1.0, 0.5, 0.25, 0.1, 0.05, 0.01):
            trial_alpha = alpha * factor
            x_new = x - trial_alpha * v
            p_new = softmax(x_new)
            
            b_new = fft_convolution(p_new)
            G_new, _ = smooth_max_grad(b_new, beta)
            if not np.isfinite(G_new):
                continue
            
            if float(n * G_new) < cur_val - 1e-14:
                x = x_new
                p = p_new
                cur_val = float(n * G_new)
                cur_score = float(2*n*np.max(b_new))
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_p = p.copy()
                    no_improve = 0
                step_accepted = True
                break
        
        if step_accepted:
            alpha = min(alpha * 1.1, 10.0 / beta_start)
        else:
            alpha *= 0.5
            if alpha < 1e-10:
                # Reset
                alpha = 1.0 / beta
                v = grad_x.copy()
    
    return best_p, best_score


def block_to_p(n, B, c):
    """Convert block log-weights to full sequence (softmax over blocks, repeated)."""
    L = n // B
    z = np.zeros(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        m = min(B-1, c.size)
        z[:m] = c[:m]
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0 or not np.isfinite(s):
        wb = np.ones(B) / B
    else:
        wb /= s
    H = wb / L
    return np.repeat(H, L)


def block_obj_and_grad(c, n, B, beta):
    """Objective and gradient for block-constant sequences in log-space."""
    L = n // B
    z = np.zeros(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        m = min(B-1, c.size)
        z[:m] = c[:m]
    
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0 or not np.isfinite(s):
        return float('inf'), np.zeros(max(0, B-1))
    wb /= s
    H = wb / L
    p = np.repeat(H, L)
    
    b = fft_convolution(p)
    G, w = smooth_max_grad(b, beta)
    if not np.isfinite(G):
        return float('inf'), np.zeros(max(0, B-1))
    
    # Gradient w.r.t. p
    gp = conv_grad(p, w)
    
    # Now we need gradient w.r.t. block log-weights
    # p = repeat(H, L) where H = wb/L and wb = softmax(z[:-1], z[-1]=0)
    # dp_i/dz_j (for block j, i in block j) = H_j * (delta_jk - H_k * L) ... 
    # Actually H = wb/L, so dp_i = d(wb_{block(i)})/L for i in that block
    # d(wb_j)/dz_k = wb_j * (delta_jk - wb_k) for k < B-1 (since z[-1]=0 is fixed)
    # But we have B-1 free variables (z[0]..z[B-2]) with z[B-1]=0
    
    # Gradient w.r.t. z_k (k = 0..B-2):
    # = sum_i dp_i/dz_k * gp_i
    # = sum_{blocks j} (sum_{i in block j} gp_i) * dp_{block(j)}/dz_k
    # = sum_j GSum_j * d(wb_j/L)/dz_k
    # = (1/L) * sum_j GSum_j * wb_j * (delta_jk - wb_k)
    #   where for k < B-1: delta_jk is 1 if j==k
    #   and the constraint is that z has B-1 free params with z[-1]=0
    #   Actually we parameterize z[0]..z[B-2] freely, z[B-1]=0
    #   wb = softmax(z) where z has B components
    #   d(wb_j)/dz_k = wb_j*(delta_jk - wb_k) for all j,k
    
    # So grad_z_k = (1/L) * sum_j GSum_j * wb_j * (delta_jk - wb_k)
    #            = (1/L) * (GSum_k * wb_k - wb_k * sum_j GSum_j * wb_j)
    #            = (wb_k / L) * (GSum_k - sum_j GSum_j * wb_j)
    
    GSum = np.zeros(B, dtype=np.float64)
    for j in range(B):
        GSum[j] = float(np.sum(gp[j*L:(j+1)*L]))
    
    Wavg = float(np.dot(wb, GSum))
    grad_z = (wb / L) * (GSum - Wavg)
    
    grad = n * grad_z[:B-1]  # only B-1 free params
    grad = np.where(np.isfinite(grad), grad, 0.0)
    
    return float(n * G), grad


def run_block_opt(c0, n, B, beta, bounds, deadline):
    """Run L-BFGS-B on block representation."""
    if not HAVE_SCIPY:
        return c0
    
    c0 = np.asarray(c0, dtype=np.float64).reshape(-1)
    if c0.size != max(0, B-1):
        tmp = np.zeros(max(0, B-1))
        m = min(tmp.size, c0.size)
        tmp[:m] = c0[:m]
        c0 = tmp
    c0 = np.clip(c0, -8.0, 8.0)
    
    def obj_wrapper(c):
        val, grad = block_obj_and_grad(c, n, B, beta)
        if time.monotonic() >= deadline:
            val, grad = float('inf'), np.zeros_like(c)
        return val, grad
    
    try:
        res = minimize(obj_wrapper, c0, method='L-BFGS-B', jac=True,
                      bounds=bounds, options={'maxiter': 200, 'ftol': 1e-12,
                      'gtol': 1e-6, 'maxcor': 20, 'maxls': 50})
        c = np.asarray(res.x, dtype=np.float64)
    except Exception:
        c = c0
    
    if not np.all(np.isfinite(c)):
        c = c0
    return np.clip(c, -8.0, 8.0)

"""
Minimize 2*n*max(a*a)/(sum(a))^2 using multi-scale block optimization +
log-softmax Nesterov-accelerated refinement with adaptive smooth-max beta.
Starts from height_sequence_1 if available.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def fft_convolution(a):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def fast_score(p):
    """Compute 2*n*max(p*p)/(sum(p))^2 quickly."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float('inf')
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float('inf')
    b = fft_convolution(p)
    if b.size == 0:
        return float('inf')
    M = float(np.max(b))
    if not np.isfinite(M):
        return float('inf')
    return float(2.0 * n * M / (S * S))


def smooth_max_grad(b, beta):
    """Return (smooth_max_value, softmax_weights) for the convolution vector b."""
    b = np.asarray(b, dtype=np.float64)
    m = float(np.max(b))
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b, dtype=np.float64)
        w /= float(b.size)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w


def conv_grad_p(p, w):
    """Gradient of sum_k w_k*(p*p)_k w.r.t. p. Returns array of shape (n,)."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n:m][::-1].copy()
    return 2.0 * np.maximum(s, 0.0)


def softmax_fn(x):
    """Stable softmax."""
    x = np.asarray(x, dtype=np.float64).copy()
    x -= np.max(x)
    e = np.exp(x)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(x.shape, 1.0 / float(x.size), dtype=np.float64)
    return e / s


def normalize_simplex(p):
    """Clip nonnegative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size), dtype=np.float64)
    p *= 1.0 / s
    return p


def block_to_p(n, B, c):
    """Convert block log-weights c (length B-1) to full sequence."""
    L = n // B
    z = np.zeros(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        m = min(B - 1, c.size)
        z[:m] = c[:m]
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0.0 or not np.isfinite(s):
        wb = np.ones(B) / B
    else:
        wb /= s
    H = wb / L
    return np.repeat(H, L)


def block_obj_grad(c, n, B, beta):
    """Objective and gradient for block-constant log-weight parameterization."""
    L = n // B
    z = np.zeros(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        m = min(B - 1, c.size)
        z[:m] = c[:m]
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0.0 or not np.isfinite(s):
        return float('inf'), np.zeros(max(0, B - 1), dtype=np.float64)
    wb /= s
    H = wb / L
    p = np.repeat(H, L)

    b = fft_convolution(p)
    G, w = smooth_max_grad(b, beta)
    if not np.isfinite(G):
        return float('inf'), np.zeros(max(0, B - 1), dtype=np.float64)

    gp = conv_grad_p(p, w)

    GSum = np.zeros(B, dtype=np.float64)
    for j in range(B):
        GSum[j] = float(np.sum(gp[j * L:(j + 1) * L]))

    Wavg = float(np.dot(wb, GSum))
    grad_z = (wb / L) * (GSum - Wavg)

    grad = n * grad_z[:B - 1]
    grad = np.where(np.isfinite(grad), grad, 0.0)

    return float(n * G), grad.astype(np.float64)


def run_block_lbfgs(c0, n, B, beta, deadline):
    """Run L-BFGS-B on block log-weights."""
    if not HAVE_SCIPY:
        return c0

    c0 = np.asarray(c0, dtype=np.float64).reshape(-1)
    if c0.size != max(0, B - 1):
        tmp = np.zeros(max(0, B - 1), dtype=np.float64)
        m = min(tmp.size, c0.size)
        tmp[:m] = c0[:m]
        c0 = tmp
    c0 = np.clip(c0, -8.0, 8.0)

    bounds = [(-8.0, 8.0)] * max(0, B - 1)

    def wrapper(c):
        if time.monotonic() >= deadline:
            return float('inf'), np.zeros_like(c)
        return block_obj_grad(c, n, B, beta)

    try:
        res = minimize(wrapper, c0, method='L-BFGS-B', jac=True,
                       bounds=bounds,
                       options={'maxiter': 300, 'ftol': 1e-13, 'gtol': 1e-7,
                                'maxcor': 20, 'maxls': 50})
        c = np.asarray(res.x, dtype=np.float64)
    except Exception:
        c = c0

    if not np.all(np.isfinite(c)):
        c = c0
    return np.clip(c, -8.0, 8.0)


def nesterov_log_optimize(x0, n, deadline, beta_start=5000.0, beta_end=300000.0,
                          momentum=0.9, max_iter=5000000):
    """
    Nesterov-accelerated gradient descent in log-space (softmax parameterization).
    Minimizes n * smooth_max(softmax(x) * softmax(x)).
    """
    x = np.asarray(x0, dtype=np.float64).copy()
    p = softmax_fn(x)

    b = fft_convolution(p)
    best_score = float(2.0 * n * np.max(b))
    best_p = p.copy()

    v = np.zeros(n, dtype=np.float64)
    alpha = 1.0 / beta_start
    no_improve = 0
    it = 0

    while it < max_iter and time.monotonic() < deadline:
        it += 1
        frac = float(it) / float(max_iter)
        beta = beta_start + (beta_end - beta_start) * frac

        b = fft_convolution(p)
        G, w = smooth_max_grad(b, beta)
        if not np.isfinite(G):
            beta = max(1000.0, beta * 0.5)
            b = fft_convolution(p)
            G, w = smooth_max_grad(b, beta)
            if not np.isfinite(G):
                break

        cur_val = n * G
        cur_score = float(2.0 * n * np.max(b))
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 2000 == 0:
            print(f"    nest it={it} beta={beta:.0f} best={best_score:.8f} "
                  f"noimp={no_improve} alpha={alpha:.2e}")

        if no_improve > 8000:
            if no_improve > 15000:
                break
            v *= 0.3
            no_improve = 8000

        gp = conv_grad_p(p, w)
        gp_dot_p = float(np.dot(gp, p))
        grad_x = p * (gp - gp_dot_p)

        v = momentum * v + grad_x
        v_norm = float(np.linalg.norm(v))
        if not np.isfinite(v_norm) or v_norm < 1e-16:
            break

        accepted = False
        for factor in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.01):
            a = alpha * factor
            x_new = x - a * v
            p_new = softmax_fn(x_new)
            b_new = fft_convolution(p_new)
            G_new, _ = smooth_max_grad(b_new, beta)
            if not np.isfinite(G_new):
                continue
            if n * G_new < cur_val * (1.0 - 1e-12) - 1e-15:
                x = x_new
                p = p_new
                cur_val = n * G_new
                sc = float(2.0 * n * np.max(b_new))
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_p = p.copy()
                    no_improve = 0
                accepted = True
                break

        if accepted:
            alpha = min(alpha * 1.3, 5.0 / beta)
        else:
            alpha *= 0.4
            if alpha < 1e-12:
                alpha = 1.0 / beta
                v = grad_x.copy()

    return best_p, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):

    except Exception:
        seed = 42

        budget_s = 2400.0

    if budget_s <= 0.0:
        return [1.0]

    reserve = 100.0
    deadline = start_time + max(0.0, budget_s - reserve)

    N = 8192
    rng = np.random.default_rng(seed)

    best_p = np.full(N, 1.0 / float(N), dtype=np.float64)
    best_score = fast_score(best_p)
    print(f"Uniform N={N}: score={best_score:.8f}")

    # Try to load height_sequence_1
    try:
        h1 = height_sequence_1
        if isinstance(h1, list) and len(h1) > 0:
            p1 = np.array([float(x) for x in h1], dtype=np.float64)
            p1 = np.maximum(p1, 0.0)
            s1 = float(np.sum(p1))
            if s1 > 0.0 and np.isfinite(s1):
                p1 /= s1
                sc1 = fast_score(p1)
                print(f"height_sequence_1: len={len(h1)}, score={sc1:.8f}")
                if np.isfinite(sc1) and sc1 < best_score:
                    best_score = sc1
                    best_p = p1.copy()
    except NameError:
        pass
    except Exception as e:
        print(f"Could not load height_sequence_1: {e}")

    # === Phase 1: Block optimization (coarse to fine) ===
    if deadline - time.monotonic() > 120.0:
        block_dl = time.monotonic() + min(400.0, 0.22 * budget_s)
        print(f"Phase 1: block optimization, time budget={block_dl - time.monotonic():.0f}s")

        for B in (16, 32, 64, 128, 256, 512, 1024):
            if time.monotonic() >= block_dl:
                break
            if B >= N or N % B != 0:
                continue

            L = N // B
            # Initialize block weights from current best
            c0 = np.zeros(B - 1, dtype=np.float64)
            if B > 1:
                logs = np.zeros(B, dtype=np.float64)
                for j in range(B):
                    bx = best_p[j * L:(j + 1) * L]
                    nz = bx[bx > 1e-15]
                    if len(nz) > 0:
                        logs[j] = float(np.mean(np.log(nz)))
                logs -= np.max(logs)
                c0 = logs[:B - 1] - logs[B - 1]

            for beta in (5000.0, 50000.0, 300000.0):
                if time.monotonic() >= block_dl:
                    break
                sub_dl = min(block_dl, time.monotonic() + 60.0)
                c = run_block_lbfgs(c0, N, B, beta, sub_dl)
                p = block_to_p(N, B, c)
                sc = fast_score(p)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_p = p.copy()
                    print(f"  B={B} beta={beta:.0f}: score={sc:.8f}")
                c0 = c.copy()

        print(f"After Phase 1: score={best_score:.8f}")

    # === Phase 2: Full-length Nesterov in log-space ===

        print("Phase 2: full Nesterov log-space optimization")
        x0 = np.log(np.maximum(best_p, 1e-30) / float(np.sum(best_p)))

                best_p = p_a.copy()

            else:
                best_p = p_a

        # Sub-phase 2b: high beta (sharp max)

            t2 = deadline - 30.0

    # === Phase 3: Perturbation restarts ===

                                            beta_start=50000.0, beta_end=400000.0,

            if np.isfinite(scp) and scp < best_score:

                best_p = pp.copy()
                print(f"  Perturb {trial}: score={scp:.8f}")
            else:

    # === Final: normalize, clip, evaluate ===

    final_seq = [float(x) for x in final_p]
