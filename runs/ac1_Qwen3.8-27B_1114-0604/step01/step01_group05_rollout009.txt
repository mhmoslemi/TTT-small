This means `A(f) = e^{iθ(f)}` for some phase function, which implies `a` is a delta function (by the inverse DFT). Contradiction (since a delta function has `sum a = 1` but `max(a*a) = 1`).

So the minimum is strictly greater than `1/(2n-1)`.

For large n, the minimum should approach some constant. The question is what that constant is.

def fft_convolution(a):
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)

def smooth_max_and_gradient(b, beta):
    """Return (smooth_max, gradient_of_smooth_max w.r.t. b)."""
    m = float(np.max(b))
    if not np.isfinite(m) or m <= 0:
        w = np.ones_like(b) / b.size
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0 or not np.isfinite(Z):
        w = np.zeros_like(b)
        w[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w

def corr_vector(p, w):
    """Compute s_j = sum_k w_k * p_{k-j} (valid indices) for j=0..n-1."""
    n = p.size
    m = w.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n : m][::-1].copy()
    return s

    v = np.asarray(v, dtype=np.float64).copy()

    n = p.size

    G, w = smooth_max_and_gradient(b, beta)

    svec = np.maximum(svec, 0.0)

        if time.monotonic() >= deadline:
            break

            no_improve = 0

    # and from max(0, -j) = 0 (since j >= 0)

"""
Minimize 2*n*max(a*a)/(sum(a))^2 using FISTA (Nesterov-accelerated gradient)
with adaptive smoothing (beta annealing), backtracking line search, and
multiple starting points. Also uses power-sum objective as alternative.
Tries multiple sequence lengths and returns the best found.

import numpy as np
import time

def fft_conv(a):
    """Fast linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)

def softmax_max(b, beta):
    """Smooth max via softmax. Returns (value, weights)."""
    b = np.asarray(b, dtype=np.float64)
    m = float(np.max(b))
    if m <= 0.0 or not np.isfinite(m):
        return m, np.ones_like(b) / b.size
    x = np.clip(beta * (b - m), -745.0, 0.0)
    e = np.exp(x)
    Z = float(e.sum())
    if Z <= 0.0 or not np.isfinite(Z):
        w = np.zeros_like(b)
        w[int(np.argmax(b))] = 1.0
        return m, w
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return G, w

def power_sum_max(b, p):
    """Power-sum as smooth max approximation. Returns (value, gradient_weights)."""
    b = np.asarray(b, dtype=np.float64)
    bp = np.maximum(b, 0.0) ** p
    s = float(bp.sum())
    if s <= 0.0:
        return 0.0, np.ones_like(b) / b.size
    val = s ** (1.0 / p)
    # d/d b_i [ (sum b_j^p)^(1/p) ] = b_i^(p-1) / (sum b_j^p)^(1-1/p)
    w = bp / s
    return val, w

def grad_corr(p, w):
    """
    Compute gradient of sum_k w_k * (p*p)_k w.r.t. p.
    Result: 2 * [p correlated with reverse(w)]
    """
    n = p.size
    m = w.size
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n:m][::-1].copy()
    return 2.0 * np.maximum(s, 0.0)

    # Final averaged score
    avg_sc = fast_score(avg_p)
    if avg_sc < best_sc:
        best_sc = avg_sc
        best_p = avg_p.copy()

    return best_p, best_sc

def structured_starts(n, rng):
    """Generate diverse starting points."""
    starts = []

    # Uniform
    starts.append(np.full(n, 1.0 / n, dtype=np.float64))

    x = np.linspace(0.0, 1.0, n, dtype=np.float64)

    # Exponential tapers
    for a in (-3.0, -1.5, 1.5, 3.0, -5.0, 5.0):
        z = a * (x - 0.5)
        p = np.exp(z)
        p /= p.sum()
        starts.append(p)

    # Cosine raised
    for width in (0.4, 0.6, 0.8):
        p = np.ones(n, dtype=np.float64)
        edge = int(n * (1.0 - width) / 2.0)
        if edge > 0:
            t = np.linspace(0, np.pi / 2, edge)
            taper = np.cos(t) ** 2
            p[:edge] = taper
            p[-edge:] = taper[::-1]
        p = np.maximum(p, 1e-10)
        p /= p.sum()
        starts.append(p)

    # Bessel-like (J0 window)
    for param in (1.5, 2.5, 3.5):
        t = 1.0 - 2.0 * np.abs(x - 0.5)
        p = np.cos(param * np.pi * np.sqrt(np.maximum(1.0 - t * t, 0.0)))
        p = np.maximum(p, 1e-10)
        p /= p.sum()
        starts.append(p)

    # Two-peak
    for sep in (0.3, 0.5, 0.7):
        p = 0.5 * np.exp(-20.0 * (x - (1.0 - sep) / 2.0) ** 2) + \
            0.5 * np.exp(-20.0 * (x - (1.0 + sep) / 2.0) ** 2)
        p = np.maximum(p, 1e-10)
        p /= p.sum()
        starts.append(p)

    # Random smooth
    for _ in range(4):
        size = 1
        while size < n:
            size <<= 1
        zc = rng.normal(size=size // 2 + 1) + 1j * rng.normal(size=size // 2)
        z = np.real(np.fft.irfft(zc, size))[:n]
        z = z - z.mean()
        z = 2.0 * np.tanh(z)
        p = np.exp(z)
        p /= p.sum()
        starts.append(p)

    return starts

def resample_seq(old, new_n):
    """Resample a sequence to a new length using linear interpolation."""
    old_n = len(old)
    if old_n == new_n:
        return np.array(old, dtype=np.float64)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    new = np.interp(x_new, x_old, old)
    return np.maximum(new, 0.0)

def official_score(seq):
    """Call the authoritative evaluator."""
    try:
        val = evaluate_sequence(seq)
        val = float(val)
    except Exception:
        return float('inf')
    return val if np.isfinite(val) else float('inf')

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Uses FISTA with beta annealing, multiple starts,
    and multiple sequence lengths to minimize the evaluation function.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 10.0:
        return [1.0] * 1024

    start_time = time.monotonic()
    reserve = 60.0
    deadline = start_time + max(10.0, budget_s - reserve)

    rng = np.random.default_rng(seed)

    # Get initial sequence if available
    init_seq = None
    try:
        hs = height_sequence_1
        if isinstance(hs, list) and len(hs) >= 100:
            init_arr = np.array(hs, dtype=np.float64)
            init_arr = np.maximum(init_arr, 0.0)
            s = init_arr.sum()
            if s > 0 and np.isfinite(s):
                init_arr /= s
                init_seq = init_arr
                print(f"Loaded initial sequence, len={len(init_arr)}, score={fast_score(init_arr):.6f}")
    except Exception:
        pass

    # Determine lengths to try based on budget
    if budget_s >= 2000:
        lengths = [8192, 16384]
    elif budget_s >= 1000:
        lengths = [8192, 16384]
    else:
        lengths = [8192]

    best_overall = None
    best_overall_score = float('inf')

    for N in lengths:
        if time.monotonic() >= deadline:
            break
        n = N
        print(f"\n{'='*50}")
        print(f"  Optimizing n = {n}")
        print(f"{'='*50}")

        # Build starting points
        starts = structured_starts(n, rng)

        # Add initial sequence (resampled if needed)
        if init_seq is not None:
            if len(init_seq) == n:
                starts.append(init_seq.copy())
            else:
                resampled = resample_seq(init_seq, n)
                resampled = normalize(resampled)
                starts.append(resampled)

        # Score and sort starts
        scored = []
        for p in starts:
            sc = fast_score(p)
            if np.isfinite(sc):
                scored.append((sc, p))
        scored.sort(key=lambda x: x[0])

        print(f"  Top starts: {[f'{s:.6f}' for s, _ in scored[:5]]}")

        # Pick top 4 starts
        top_starts = [p for _, p in scored[:4]]

        # Time allocation
        time_since_start = time.monotonic() - start_time
        remaining_total = deadline - time.monotonic()
        if remaining_total <= 30:
            break

        # Allocate: more time to the best start, and try both modes
        fracs = [0.35, 0.25, 0.20, 0.20]

        for i, p0 in enumerate(top_starts):
            remaining = deadline - time.monotonic()
            if remaining < 30.0:
                break

            frac = fracs[i] if i < len(fracs) else 0.15
            alloc = remaining * frac
            sub_deadline = time.monotonic() + alloc

            print(f"  Start {i}: score={fast_score(p0):.8f}, alloc={alloc:.0f}s")

            # FISTA with softmax
            p_fit, sc_fit = fista_optimize(
                p0, sub_deadline, mode="softmax",
                beta_start=200.0, beta_end=10000000.0,
                verbose=(i == 0)
            )

            if np.isfinite(sc_fit) and sc_fit < best_overall_score:
                best_overall_score = sc_fit
                best_overall = p_fit.copy()
                print(f"    FISTA-softmax best: {sc_fit:.8f}")

            # If time remains, try power-sum mode from the best so far
            remaining2 = deadline - time.monotonic()
            if remaining2 > 60.0 and best_overall is not None and len(best_overall) == n:
                alloc2 = remaining2 * 0.4
                sub_deadline2 = time.monotonic() + alloc2
                p_fit2, sc_fit2 = fista_optimize(
                    best_overall, sub_deadline2, mode="power",
                    power_p=30.0, verbose=False
                )
                if np.isfinite(sc_fit2) and sc_fit2 < best_overall_score:
                    best_overall_score = sc_fit2
                    best_overall = p_fit2.copy()
                    print(f"    FISTA-power best: {sc_fit2:.8f}")

        # Subgradient averaging from the best (if time remains)
        remaining3 = deadline - time.monotonic()
        if remaining3 > 120.0 and best_overall is not None and len(best_overall) == n:
            alloc3 = remaining3 * 0.5
            sub_deadline3 = time.monotonic() + alloc3
            print(f"  Subgradient averaging: alloc={alloc3:.0f}s")
            p_sg, sc_sg = subgradient_avg(best_overall, sub_deadline3, max_iter=300000)
            if np.isfinite(sc_sg) and sc_sg < best_overall_score:
                best_overall_score = sc_sg
                best_overall = p_sg.copy()
                print(f"    Subgradient best: {sc_sg:.8f}")

    # Final cleanup
    if best_overall is None:
        best_overall = np.full(8192, 1.0 / 8192, dtype=np.float64)

    final_p = normalize(best_overall)
    final_p = np.clip(final_p, 0.0, 1000.0)
    final_seq = [float(x) for x in final_p]

    # Official evaluation
    off = official_score(final_seq)
    print(f"\n  Official score: {off:.8f}")

    if not np.isfinite(off) or off > 2.0 + 1e-9:
        # Fallback to uniform
        return [1.0] * len(final_seq)

    return final_seq
