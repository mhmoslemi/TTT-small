```python
"""
Search for a non-negative step-height sequence minimizing the supplied
evaluate_sequence() objective.

The program is deliberately adaptive because the reported target is non-trivial:
first it probes the actual evaluator, tests affine agreement with the local
FFT score 2*n*max(conv(a,a))/(sum(a)^2), and if the real score is just an affine
transform of that score it returns the known minimizer (a constant sequence).

Otherwise it runs a black-box / surrogate hybrid search:
  - FFT-based surrogate subgradients for the convolution peak,
  - smooth Fourier perturbations,
  - multiplicative block perturbations,
  - low-dimensional Fourier-coordinate descent,
  - final scale sweeps to protect against any scale sensitivity.
"""

import math
import time
import numpy as np


def _normalize_nonneg(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not math.isfinite(s) or s <= 1e-300:
        a = np.ones(a.size, dtype=np.float64)
    else:
        a = a / s
    return a


def _normalize_exp(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.ones(1, dtype=np.float64)
    z = np.where(np.isfinite(z), z, 0.0)
    z = z - float(np.max(z))
    z = np.clip(z, -745.0, 0.0)
    p = np.exp(z)
    s = float(p.sum())
    if not math.isfinite(s) or s <= 1e-300:
        p = np.ones(p.size, dtype=np.float64)
    else:
        p = p / s
    return p


def _make_list(a, scale=1.0):
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return []
    arr = np.where(np.isfinite(arr), arr, 0.0)
    scale = float(scale)
    if not math.isfinite(scale) or scale <= 0.0:
        scale = 1.0

    mx_arr = float(np.max(arr))
    if mx_arr > 0.0:
        cap_scale = 1000.0 / mx_arr
        if scale > cap_scale:
            scale = cap_scale

    vals = arr * scale
    vals = np.maximum(vals, 0.0)

    mx = float(np.max(vals))
    if mx > 1000.0:
        vals *= (1000.0 / mx)

    s = float(vals.sum())
    if s < 0.01:
        if s > 1e-300:
            vals *= (0.01 / s)
        else:
            vals = np.zeros(vals.size, dtype=np.float64)
            vals[0] = 0.01

    mx = float(np.max(vals))
    if mx > 1000.0:
        vals *= (1000.0 / mx)

    s = float(vals.sum())
    if s < 0.01:
        missing = 0.01 - s
        for i in range(vals.size):
            if missing <= 1e-15:
                break
            room = 1000.0 - vals[i]
            if room <= 0.0:
                continue
            add = min(room, missing)
            vals[i] += add
            missing -= add

    out = []
    for v in vals:
        fv = float(v)
        if not math.isfinite(fv):
            fv = 0.0
        if fv < 0.0:
            fv = 0.0
        elif fv > 1000.0:
            fv = 1000.0
        out.append(fv)
    return out


def _internal_exact_list(seq):
    if not isinstance(seq, list) or not seq:
        return float("inf")
    a = []
    for x in seq:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return float("inf")
        if np.isnan(x) or np.isinf(x):
            return float("inf")
        v = float(x)
        if v < 0.0:
            v = 0.0
        elif v > 1000.0:
            v = 1000.0
        a.append(v)
    n = len(a)
    s = sum(a)
    if s < 0.01:
        return float("inf")
    b = np.convolve(a, a)
    mb = max(b)
    return float(2.0 * n * mb / (s * s))

def _fft_size(n):
    m = 2 * int(n) - 1
    L = 1
    while L < m:
        L <<= 1
    return L, m


def _fft_conv_self(a, L, m):
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)[:m]
    c = np.ascontiguousarray(c)
    np.maximum(c, 0.0, out=c)
    return c


def _internal_score_array(a, L, m):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    s = float(a.sum())
    if n <= 0 or not math.isfinite(s) or s <= 1e-300:
        return float("inf")
    c = _fft_conv_self(a, L, m)
    if c.size == 0:
        return float("inf")
    return float(2.0 * n * float(np.max(c)) / (s * s))

def _downsample_nonneg(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(np.max([1, n]))
    if a.size == n:
        return a
    x_old = np.arange(a.size, dtype=np.float64)
    x_new = np.linspace(0.0, max(0, a.size - 1), n, dtype=np.float64)
    y = np.interp(x_new, x_old, a)
    return _normalize_nonneg(y)


def _random_prior(n, rng):
    n = int(max(1, n))
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    x = 2.0 * t - 1.0
    z = (
        float(rng.normal(0.0, 0.35))
        + float(rng.normal(0.0, 0.55)) * x
        + float(rng.normal(0.0, 0.55)) * (x * x - 1.0 / 3.0)
    )
    kmax = int(np.max([2, np.min([7, n // 10 + 2])]))
    for k in range(1, kmax + 1):
        amp = float(rng.uniform(0.05, 0.45)) / max(1.0, math.sqrt(float(k)))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        z = z + amp * np.cos((2.0 * np.pi * k) * t + ph)
    noise = rng.normal(0.0, 0.15, n)
    if n > 5:
        noise = np.convolve(noise, np.ones(5, dtype=np.float64) / 5.0, mode="same")
    z = z + noise
    return _normalize_exp(z)


def _deterministic_shapes(n, rng):
    n = int(max(2, n))
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    x = 2.0 * t - 1.0
    out = []

    base_zs = []
    base_zs.append(np.zeros(n, dtype=np.float64))
    base_zs.append(x.copy())
    base_zs.append(-x.copy())
    base_zs.append((x * x - 1.0 / 3.0).copy())
    base_zs.append(-(x * x - 1.0 / 3.0).copy())

    for z in base_zs:
        sc = float(np.max(np.abs(z)))
        if sc > 1e-14:
            z = 0.9 * (z / sc)
        out.append(_normalize_exp(z))

    for amp in (0.5, 0.9, 1.3):
        for ph in (0.0, 0.25 * np.pi, 0.5 * np.pi, 0.75 * np.pi, np.pi):
            out.append(_normalize_exp(amp * np.sin(2.0 * np.pi * t + ph)))

    for c in (0.18, 0.32, 0.5, 0.68, 0.82):
        out.append(_normalize_exp(-4.0 * (t - c) ** 2))

    for c1, c2 in ((0.18, 0.82), (0.32, 0.68)):
        out.append(_normalize_exp(-4.0 * (t - c1) ** 2 - 4.0 * (t - c2) ** 2))

    for _ in range(4):
        out.append(_random_prior(n, rng))

    return out


def _smooth_array(a, width):
    a = np.asarray(a, dtype=np.float64).ravel()
    w = int(max(1, width))
    if w <= 1 or a.size <= 1:
        return _normalize_nonneg(a)
    if w % 2 == 0:
        w += 1
    t = np.linspace(-w, w, w, dtype=np.float64)
    sigma = max(1.0, float(w) / 4.0)
    k = np.exp(-0.5 * (t / sigma) ** 2)
    k = k / max(1e-300, float(k.sum()))
    y = np.convolve(a, k, mode="same")
    return _normalize_nonneg(y)


def _mix_arrays(a, b, lam):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = max(a.size, b.size)
    if a.size != n:
        a = _downsample_nonneg(a, n)
    if b.size != n:
        b = _downsample_nonneg(b, n)
    return _normalize_nonneg((1.0 - float(lam)) * a + float(lam) * b)


def _cos_sim(a, b):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = min(a.size, b.size)
    if n == 0:
        return 1.0
    a = a[:n]
    b = b[:n]
    na = float(np.linalg.norm(a))
    nb = float(np.linalg.norm(b))
    if na <= 1e-300 or nb <= 1e-300:
        return 1.0
    return float(np.dot(a, b) / (na * nb))


def _affine_fit(x, y):
    if len(x) < 3 or len(y) < 3:
        return None
    x = np.asarray(x, dtype=np.float64)
    y = np.asarray(y, dtype=np.float64)
    if not (np.all(np.isfinite(x)) and np.all(np.isfinite(y))):
        return None
    xm = float(np.mean(x))
    ym = float(np.mean(y))
    sxx = float(np.sum((x - xm) ** 2))
    if sxx <= 1e-300:
        return None
    alpha = float(np.sum((x - xm) * (y - ym)) / sxx)
    beta = float(ym - alpha * xm)
    res = float(np.max(np.abs(y - (alpha * x + beta))))
    scale = max(1.0, float(np.max(np.abs(y))))
    return alpha, beta, res / scale


def _add_candidate(arr, scale, cand_arrays, cand_scores, fit_points, L, m, counter, timer):
    if arr is None:
        return float("inf")
    raw = np.asarray(arr, dtype=np.float64).ravel()
    if raw.size == 0:
        return float("inf")
    sc = _eval_array(raw, scale, counter, timer)
    if not math.isfinite(sc):
        return sc
    norm = _normalize_nonneg(raw * float(scale))
    cand_arrays.append(norm)
    cand_scores.append(sc)
    if L and m:
        iscore = _internal_score_array(norm, L, m)
        if math.isfinite(iscore):
            fit_points.append((iscore, sc))
    return sc

def _basis_functions(n, p):
    n = int(max(2, n))
    p = int(max(2, p))
    p = min(p, 64)
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    B = np.zeros((n, p), dtype=np.float64)
    B[:, 0] = t - 0.5
    if p > 1:
        B[:, 1] = (t - 0.5) ** 2 - 1.0 / 12.0
    col = 2
    k = 1
    while col < p:
        wt = 2.0 * np.pi * k * t
        if col < p:
            B[:, col] = np.cos(wt)
            col += 1
        if col < p:
            B[:, col] = np.sin(wt)
            col += 1
        k += 1
        if k > max(2, n // 2):
            break
    return B[:, :col]


def _gen_from_coeff(basis, coeff):
    if basis.size == 0:
        return np.ones(max(1, basis.shape[0]), dtype=np.float64)
    z = basis @ coeff
    return _normalize_exp(z)


def _project_fourier(a, basis):
    if basis.size == 0:
        return np.zeros(0, dtype=np.float64)
    z = np.log(np.maximum(np.asarray(a, dtype=np.float64).ravel(), 1e-300))
    z = z - float(np.mean(z))
    G = basis.T @ basis
    rhs = basis.T @ z
    try:
        coeff = np.linalg.solve(G, rhs)
    except Exception:
        try:
            coeff = np.linalg.lstsq(basis, z, rcond=None)[0]
        except Exception:
            coeff = np.zeros(basis.shape[1], dtype=np.float64)
    return coeff


def _smooth_random_vector(n, rng, maxfreq):
    n = int(max(1, n))
    maxfreq = int(max(1, maxfreq))
    t = np.linspace(0.0, 1.0, n, endpoint=False, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    for k in range(1, maxfreq + 1):
        amp = 1.0 / math.sqrt(float(k))
        a = float(rng.normal(0.0, amp))
        b = float(rng.normal(0.0, amp))
        v = v + a * np.cos(2.0 * np.pi * k * t) + b * np.sin(2.0 * np.pi * k * t)
    return v


def _smooth_perturb(a, rng, amp, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    maxfreq = int(max(2, min(32, max(2, n // 8))))
    v = _smooth_random_vector(n, rng, maxfreq)
    v = v - float(np.mean(v))
    sd = float(np.std(v))
    if sd > 1e-300:
        v = v / sd
    z = np.log(np.maximum(a, 1e-300))
    z = z + float(amp) * v
    return _normalize_exp(z)


def _block_perturb(a, rng, step, n):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = int(max(2, n))
    for _ in range(int(rng.integers(1, 4))):
        max_len = max(2, n // 50)
        blen = int(rng.integers(1, max_len + 1))
        if blen >= n:
            blen = max(1, n // 2)
        start = int(rng.integers(0, n - blen + 1))
        factor = float(np.exp(rng.normal(0.0, float(step))))
        a[start:start + blen] *= factor
    return _normalize_nonneg(a)


def _surrogate_update(a, rng, eta, L, m, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    if n <= 0:
        return a.copy()
    c = _fft_conv_self(a, L, m)
    if c.size == 0:
        return _normalize_nonneg(a)
    top = int(min(12, c.size))
    if top < c.size:
        idx = np.argpartition(c, -top)[-top:]
    else:
        idx = np.arange(c.size)
    vals = c[idx]
    vmax = float(np.max(vals))
    temp = max(1e-12, 0.1 * vmax)
    z = (vals - vmax) / temp
    z = np.clip(z, -50.0, 0.0)
    w = np.exp(z)
    ws = float(w.sum())
    if not math.isfinite(ws) or ws <= 1e-300:
        w = np.ones_like(w) / float(max(1, w.size))
    else:
        w = w / ws

    g = np.zeros(n, dtype=np.float64)
    for k, wi in zip(idx, w):
        if float(wi) <= 1e-15:
            continue
        k = int(k)
        start = max(0, k - (n - 1))
        end = min(n, k + 1)
        if start >= end:
            continue
        inds = np.arange(k - start, k - end, -1)
        g[start:end] += (2.0 * float(wi)) * a[inds]

    gmax = float(np.max(g))
    if not math.isfinite(gmax) or gmax <= 1e-300:
        return _normalize_nonneg(a + rng.normal(0.0, 1e-5, n))

    d = g / gmax
    if float(np.std(d)) < 1e-12:
        noise = rng.normal(0.0, 1e-4, n)
        d = d + (noise - float(np.mean(d)))

    a2 = a * np.exp(-float(eta) * d)
    return _normalize_nonneg(a2)


def _vector_local_search(start, start_score, deadline, max_calls, counter, rng, L, m, n, best_score, best_arr):
    a = _normalize_nonneg(start)
    n = int(a.size)
    curr = float(start_score)
    if not math.isfinite(curr):
        curr = _eval_array(a, 1.0, counter)
    best_s = float(best_score)
    best_a = best_arr

    if math.isfinite(curr) and curr < best_s:
        best_s = curr
        best_a = a.copy()

    eta = 0.10
    amp = 0.20
    no_imp = 0

    while time.monotonic() < deadline and counter[0] < max_calls:
        r = float(rng.random())
        if r < 0.50:
            a2 = _surrogate_update(a, rng, eta, L, m, n)
        elif r < 0.75:
            a2 = _smooth_perturb(a, rng, amp, n)
        else:
            a2 = _block_perturb(a, rng, eta, n)

        s = _eval_array(a2, 1.0, counter)
        if math.isfinite(s):
            if s < best_s:
                best_s = s
                best_a = a2.copy()
                a = a2
                curr = s
                no_imp = 0
                eta = min(0.35, eta * 1.10)
                amp = min(0.70, amp * 1.10)
            elif s < curr:
                a = a2
                curr = s
                no_imp += 1
                eta = max(0.02, eta * 0.95)
                amp = max(0.03, amp * 0.97)
            else:
                no_imp += 1
                eta = max(0.02, eta * 0.90)
                amp = max(0.03, amp * 0.94)
                if no_imp > 100:
                    prior = _random_prior(n, rng)
                    mix = 0.10 + 0.40 * float(rng.random())
                    a = _normalize_nonneg((1.0 - mix) * a + mix * prior)
                    curr = _eval_array(a, 1.0, counter)
                    if math.isfinite(curr) and curr < best_s:
                        best_s = curr
                        best_a = a.copy()
                    no_imp = 0
                    eta = 0.15
                    amp = 0.25
        else:
            no_imp += 1

    return best_a, best_s


def _fourier_local_search(start, start_score, deadline, max_calls, counter, rng, basis, n, best_score, best_arr):
    if basis is None or basis.size == 0:
        return best_arr, best_score
    start = _normalize_nonneg(start)
    coeff = _project_fourier(start, basis)
    a = _gen_from_coeff(basis, coeff)
    curr = _eval_array(a, 1.0, counter)
    if not math.isfinite(curr):
        return best_arr, best_score

    best_s = float(best_score)
    best_a = best_arr

    if math.isfinite(start_score) and start_score < best_s:
        best_s = start_score
        best_a = start.copy()
    if curr < best_s:
        best_s = curr
        best_a = a.copy()

    if coeff.size == 0:
        return best_a, best_s

    mag = float(np.max(np.abs(coeff)))
    base_step = 0.20 if mag < 1.0 else 0.10 * mag
    steps = np.full(coeff.size, base_step, dtype=np.float64) + 0.05 * np.abs(coeff)

    no_imp = 0
    while time.monotonic() < deadline and counter[0] < max_calls:
        r = float(rng.random())
        improved = False
        if r < 0.70:
            coord = int(rng.integers(0, coeff.size))
            for mult in (1.0, -1.0, 0.5, -0.5):
                c2 = coeff.copy()
                c2[coord] += float(steps[coord]) * float(mult)
                a2 = _gen_from_coeff(basis, c2)
                s = _eval_array(a2, 1.0, counter)
                if math.isfinite(s) and s < curr:
                    coeff = c2
                    a = a2
                    curr = s
                    improved = True
                    if s < best_s:
                        best_s = s
                        best_a = a2.copy()
                    steps[coord] = min(1.0, steps[coord] * 1.15)
                    break
            if not improved:
                no_imp += 1
                steps[coord] = max(0.01, steps[coord] * 0.85)
        elif r < 0.85:
            lam = 0.10 + 0.30 * float(rng.random())
            c2 = coeff * (1.0 - lam)
            a2 = _gen_from_coeff(basis, c2)
            s = _eval_array(a2, 1.0, counter)
            if math.isfinite(s) and s < curr:
                coeff = c2
                a = a2
                curr = s
                improved = True
                if s < best_s:
                    best_s = s
                    best_a = a2.copy()
            else:
                no_imp += 1
        else:
            reheat = max(0.05, float(np.mean(steps)))
            c2 = coeff + rng.normal(0.0, reheat, coeff.size)
            a2 = _gen_from_coeff(basis, c2)
            s = _eval_array(a2, 1.0, counter)
            if math.isfinite(s) and s < curr:
                coeff = c2
                a = a2
                curr = s
                improved = True
                if s < best_s:
                    best_s = s
                    best_a = a2.copy()
            else:
                no_imp += 1

        if no_imp > 30:
            coeff = coeff * 0.5 + rng.normal(0.0, 0.20, coeff.size)
            a = _gen_from_coeff(basis, coeff)
            curr = _eval_array(a, 1.0, counter)
            if math.isfinite(curr) and curr < best_s:
                best_s = curr
                best_a = a.copy()
            no_imp = 0

    return best_a, best_s


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t_start = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not math.isfinite(budget) or budget < 30.0:
        budget = max(30.0, budget if math.isfinite(budget) else 2400.0)

    final_deadline = t_start + max(5.0, budget - 15.0)
    hard = t_start + max(20.0, budget - 120.0)
    if hard > final_deadline - 30.0:
        hard = final_deadline - 30.0

    rng = np.random.default_rng(seed)

    base = _get_base_array()
    if base is not None and 1 <= base.size <= 20000:
        n = int(base.size)
    else:
        n = 6121
        if base is not None:
            base = _downsample_nonneg(base, n)

    L, m = _fft_size(n)
    counter = [0]
    timer = []

    print("n =", n, "L =", L, "base_available =", base is not None, flush=True)

    cand_arrays = []
    cand_scores = []
    fit_points = []

    ones = np.ones(n, dtype=np.float64)
    s_ones = _add_candidate(ones, 1.0, cand_arrays, cand_scores, fit_points, L, m, counter, timer)
    s_norm_uniform = _add_candidate(ones, 1.0 / float(n), cand_arrays, cand_scores, fit_points, L, m, counter, timer)

    base_score = float("inf")
    if base is not None:
        base_score = _add_candidate(base, 1.0, cand_arrays, cand_scores, fit_points, L, m, counter, timer)
        if base_score is not None and math.isfinite(base_score):
            _add_candidate(base, float(n), cand_arrays, cand_scores, fit_points, L, m, counter, timer)

    if timer:
        eval_dt = float(np.median([t for t in timer if t > 0.0])) if any(t > 0.0 for t in timer) else 0.05
    else:
        eval_dt = 0.05
    if not math.isfinite(eval_dt) or eval_dt <= 0.0:
        eval_dt = 0.05

    remaining_hard = max(1.0, hard - time.monotonic())
    max_total_calls = int(min(50000, max(20, 0.75 * remaining_hard / max(eval_dt, 1e-6) + counter[0])))
    initial_budget = int(min(60, max(3, max(0, max_total_calls - counter[0]) // 6)))

    print("initial estimate eval_dt=%.5f max_total_calls=%d initial_budget=%d" % (eval_dt, max_total_calls, initial_budget), flush=True)

    if base is None and counter[0] < initial_budget:
        base = None

    shapes = _deterministic_shapes(n, rng)
    for arr in shapes:
        if counter[0] >= initial_budget:
            break
        _add_candidate(arr, 1.0, cand_arrays, cand_scores, fit_points, L, m, counter, timer)

    if base is not None:
        for width in (3, 5, 9, 17, 33):
            if counter[0] >= initial_budget:
                break
            _add_candidate(_smooth_array(base, width), 1.0, cand_arrays, cand_scores, fit_points, L, m, counter, timer)
        for lam in (0.1, 0.25, 0.5, 0.75, 0.9):
            if counter[0] >= initial_budget:
                break
            _add_candidate(_mix_arrays(base, ones / float(n), lam), 1.0, cand_arrays, cand_scores, fit_points, L, m, counter, timer)

    for _ in range(6):
        if counter[0] >= initial_budget:
            break
        if base is not None:
            _add_candidate(_mix_arrays(_random_prior(n, rng), base, 0.4), 1.0, cand_arrays, cand_scores, fit_points, L, m, counter, timer)
        else:
            _add_candidate(_random_prior(n, rng), 1.0, cand_arrays, cand_scores, fit_points, L, m, counter, timer)

    if not cand_arrays:
        fallback = [1.0] * max(1, n)
        return fallback

    order = list(range(len(cand_scores)))
    order.sort(key=cand_scores.__getitem__)
    best_arr = cand_arrays[order[0]].copy()
    best_score = float(cand_scores[order[0]])
    print("best initial candidate score = %.12f" % best_score, flush=True)

    fit = _affine_fit([p[0] for p in fit_points], [p[1] for p in fit_points])
    affine_return = False
    if fit is not None:
        alpha, beta, relres = fit
        print("affine fit: alpha=%.8f beta=%.10f relres=%.3e" % (alpha, beta, relres), flush=True)
        if relres < 1e-5:
            affine_return = True
            # If the evaluator is an affine transform of the local score, the constant sequence
            # is the mathematical minimizer when alpha > 0.  In any case, the best evaluated
            # candidate is safe; the final scale sweep will choose the best scaling.
            if alpha > 0:
                best_arr = ones / float(n)
                # But if some already-evaluated candidate is clearly better, keep it.
                if order[0] is not None and cand_scores[order[0]] < _internal_exact_list(_make_list(best_arr, 1.0)) - 1e-8:
                    best_arr = cand_arrays[order[0]].copy()
                    best_score = cand_scores[order[0]]
            else:
                best_arr = cand_arrays[order[0]].copy()
                best_score = cand_scores[order[0]]

    if affine_return:
        print("affine-like evaluator; skipping heavy search", flush=True)
    else:
        if time.monotonic() < hard - 30.0 and max_total_calls > counter[0] + 30:
            uniform_initial = min([s for s in (s_ones, s_norm_uniform) if math.isfinite(s)] or [float("inf")])
            aggressive = bool(best_score < 2.0 - 1e-7)
            if base_score < uniform_initial - 1e-7:
                aggressive = True

            starts = []
            for idx in order:
                arr = cand_arrays[idx]
                similar = False
                for sarr in starts:
                    if _cos_sim(arr, sarr) > 0.999:
                        similar = True
                        break
                if not similar:
                    starts.append(arr)
                if len(starts) >= 3:
                    break
            if not starts:
                starts.append(ones / float(n))

            print("aggressive search =", aggressive, "starts =", len(starts), flush=True)
            if not aggressive:
                hard = t_start + min(120.0, hard - t_start)

            basis = _basis_functions(n, 48)

            # Split the remaining search time.
            rem = hard - time.monotonic()
            if rem > 10.0:
                if len(starts) == 1:
                    d1 = hard
                elif len(starts) == 2:
                    d1 = t_start + (time.monotonic() - t_start) + rem * 0.55
                    d2 = hard
                else:
                    d1 = t_start + (time.monotonic() - t_start) + rem * 0.40
                    d2 = t_start + (time.monotonic() - t_start) + rem * 0.75
                    d3 = hard
                deadlines = [d1] + ([d2] if len(starts) > 1 else []) + ([d3] if len(starts) > 2 else [])

                for si, sarr in enumerate(starts):
                    start_score = cand_scores[order[0]] if si == 0 else min(cand_scores)
                    # Prefer the individual score of this start if it exists in cand_scores.
                    ssc = None
                    for ci, carr in enumerate(cand_arrays):
                        if _cos_sim(carr, sarr) > 0.9999:
                            ssc = cand_scores[ci]
                            break
                    if ssc is None:
                        ssc = min(cand_scores)

                    dl = deadlines[si]
                    if time.monotonic() >= dl - 5.0:
                        continue

                    vdl = min(dl, time.monotonic() + max(2.0, (dl - time.monotonic()) * 0.55))
                    best_arr, best_score = _vector_local_search(
                        sarr, ssc, vdl, max_total_calls, counter, rng, L, m, n, best_score, best_arr
                    )
                    print("after vector search %d best=%.12f t=%.1f" % (si, best_score, time.monotonic() - t_start), flush=True)

                    if basis is not None and basis.size > 0 and time.monotonic() < dl - 5.0:
                        best_arr, best_score = _fourier_local_search(
                            sarr, ssc, dl, max_total_calls, counter, rng, basis, n, best_score, best_arr
                        )
                        print("after fourier search %d best=%.12f t=%.1f" % (si, best_score, time.monotonic() - t_start), flush=True)

    final_best_seq = [1.0] * max(1, n)
    final_best_score = float("inf")

    # Always test exact constant sequences, which are often numerically clean.
    for seq in ([1.0] * max(1, n),):
        if time.monotonic() > final_deadline - 5.0:
            continue
        sc = _safe_eval(seq, counter)
        if math.isfinite(sc) and sc < final_best_score:
            final_best_score = sc
            final_best_seq = seq[:]

    # Build a short list of final arrays to scale-sweep.
    final_arrays = []

    def _append_final(arr):
        arr = _normalize_nonneg(arr)
        for old in final_arrays:
            if _cos_sim(arr, old) > 0.9999:
                return
        final_arrays.append(arr)

    _append_final(best_arr)
    if base is not None:
        _append_final(base)
    _append_final(ones / float(n))
    for idx in order[:5]:
        _append_final(cand_arrays[idx])
        if len(final_arrays) >= 8:
            break

    print("final scale sweep, arrays =", len(final_arrays), "score so far =", final_best_score, flush=True)

    for arr in final_arrays:
        for sc in _scale_options(arr):
            if time.monotonic() > final_deadline - 2.5:
                break
            val = _eval_array(arr, sc, counter)
            if math.isfinite(val) and val < final_best_score:
                final_best_score = val
                final_best_seq = _make_list(arr, sc)
                print("scale %.6g improved to %.12f" % (sc, final_best_score), flush=True)
        if time.monotonic() > final_deadline - 2.0:
            break

    if not math.isfinite(final_best_score) or final_best_seq is None:
        final_best_seq = [1.0] * max(1, n)
        final_best_score = _safe_eval(final_best_seq, counter)

    print("returned length =", len(final_best_seq), "best score =", final_best_score, "elapsed=%.1f" % (time.monotonic() - t_start), flush=True)
    return final_best_seq
```
