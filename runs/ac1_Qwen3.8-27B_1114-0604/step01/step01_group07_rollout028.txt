def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)

    arr = np.asarray(a, dtype=np.float64)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    mx = float(np.max(c))
    return float(2.0 * arr.size * mx / (s * s))

    i = np.arange(n, dtype=np.float64)

    a = a / float(n)

    best_amps = None

def _mirror_refine(a, budget_s=60.0, K=16, initial_lr=None):
    """Mirror descent refinement on the non-negative simplex."""
    start = time.monotonic()
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.clip(arr, 0.0, None)
    s = float(arr.sum())
    if s < 0.01:
        return arr, np.inf
    arr = arr / s
    
    n = arr.size
    best = arr.copy()
    best_score = _score(best)
    
    if initial_lr is None:
        lr = 2.0
    else:
        lr = float(initial_lr)
    
    max_lr = 1.0 * float(n) / 10.0
    min_lr = 0.1
    
    it = 0
    last_improve = 0
    no_improve_count = 0
    
    while time.monotonic() - start < budget_s:
        it += 1
        
        c = _fft_convolve(arr)
        maxc = float(np.max(c))
        
        if c.size <= K:
            active = np.arange(c.size, dtype=np.int64)
            weights = np.ones(K) / K
        else:
            # Use softmax weighting over top-2K
            top_2k = 2 * K
            idx = np.argpartition(c, -top_2k)[-top_2k:]
            c_top = c[idx]
            # Softmax with temperature
            max_ct = np.max(c_top)
            exp_vals = np.exp((c_top - max_ct) / (0.1 * maxc + 1e-10))
            weights = exp_vals / np.sum(exp_vals)
            active = idx
        
        # Compute gradient
        grad = np.zeros(n, dtype=np.float64)
        for j, kk in enumerate(active):
            k = int(kk)
            w = float(weights[j])
            # grad[i] += 2 * w * a[k-i] for valid k-i
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            if i_max >= i_min:
                valid_i = np.arange(i_min, i_max + 1)
                valid_j = k - valid_i  # = k - i
                grad[valid_i] += 2.0 * w * arr[valid_j]
        
        # Project gradient to simplex tangent space (remove mean)
        g = grad - np.mean(grad)
        
        # Check if gradient is meaningful
        g_norm = np.sqrt(np.sum(g**2))
        if g_norm < 1e-12:
            break
        
        # Mirror descent step
        exponent = -lr * g / (g_norm / n)  # normalize gradient
        exponent = np.clip(exponent, -2.0, 2.0)
        arr_new = arr * np.exp(exponent)
        arr_new = np.clip(arr_new, 0.0, None)
        ts = float(arr_new.sum())
        if ts < 0.01:
            arr_new = arr.copy()
            lr = max(min_lr, lr * 0.7)
        else:
            arr_new = arr_new / ts
        
        # Evaluate
        sc = _score(arr_new)
        if sc < best_score:
            best = arr_new.copy()
            best_score = sc
            arr = arr_new
            last_improve = it
            lr = min(max_lr, lr * 1.02)
            no_improve_count = 0
        elif sc < best_score * 1.05:
            arr = arr_new
            lr = min(max_lr, lr * 1.01)
            no_improve_count += 1
        else:
            lr = max(min_lr, lr * 0.8)
            no_improve_count += 1
            if sc > best_score * 1.1:
                arr = best.copy()
        
        if it % 50 == 0:
            elapsed = time.monotonic() - start
            print(f"  [refine] it={it} lr={lr:.2f} score={best_score:.8f} elapsed={elapsed:.1f}s")
        
        if no_improve_count > 200:
            break
        if it - last_improve > 500 and it > 100:
            break
    
    return best, best_score

def _next_pow2(m):
    """Smallest power of two >= m."""
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

    # Multiple starting points
    starts = []
    starts.append(np.ones(num_freqs) * 0.8)
    if num_freqs >= 2:
        starts.append(np.concatenate([np.array([1.0]), np.full(num_freqs - 1, 0.3)]))
    if num_freqs >= 3:
        starts.append(np.linspace(0.9, 0.2, num_freqs))
    if num_freqs >= 2:
        starts.append(np.zeros(num_freqs))
        starts[-1][0] = 1.0
        starts[-1][1] = 0.5

    for x0 in starts:
        if time.monotonic() > deadline:
            break
        try:
            res = minimize(obj, x0, method='Nelder-Mead',
                          options={'maxiter': 500, 'xatol': 1e-5,
                                  'fatol': 1e-8, 'adaptive': True})
            if np.isfinite(res.fun) and res.fun < best_score:
                best_score = float(res.fun)
                best_amps = res.x.copy()
        except Exception:
            pass

    return best_amps, best_score


def _safe_resolve(x):
    """Sanitize a sequence for evaluation."""
    if x is None:
        return None
    arr = np.asarray(x, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return None
    return arr
