```python
def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)

    if c.size == 0:

    if not np.isfinite(mx) or mx < 0.0:

    return float(2.0 * arr.size * mx / (s * s))


def _simplex_project(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.arange(1, n + 1)
    mask = u * (rho_candidates - 1.0) > (cssv - 1.0)
    rho = np.where(mask)[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]  # largest valid rho
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # Ensure sum is exactly 1 (numerical fix)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.ones(n) / n
    return w


def _make_harmonic(n, amps, phases, use_cos=False):
    """Create a multi-harmonic sequence normalized to sum=1."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for m, (A, phi) in enumerate(zip(amps, phases), 1):
        if A == 0:
            continue
        theta = 2.0 * np.pi * m * (i + 0.5) / n + phi
        if use_cos:
            a += A * np.cos(theta)
        else:
            a += A * np.sin(theta)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-10:
        return np.ones(n) / n
    a /= s
    return a

    n = arr.size
```

Wait, I just realized I should also try a completely different approach: **direct optimization using scipy**.

scipy.optimize.minimize can minimize a function with constraints. The objective is max_k c_k(a), which is not smooth, but we can use a smooth approximation like the log-sum-exp:

f(a, t) = (1/t) * log(sum_k exp(t * c_k(a)))

As t → ∞, this approaches max_k c_k(a). For finite t, it's smooth and we can use gradient-based methods.

The gradient of f with respect to a is:
df/da = (1/t) * sum_k (t * exp(t*c_k(a)) / sum_j exp(t*c_j(a))) * dc_k/da
= sum_k (exp(t*c_k) / sum_j exp(t*c_j)) * dc_k/da

This is a softmax-weighted average of the gradients of the c_k's, which is actually a very nice subgradient!

Let me implement this:

```python
def _smooth_max_objective(a, t=100.0):
    """Compute (1/t) * log(sum exp(t * c_k)) and its gradient."""
    c = _fft_convolve(a)
    # Numerical stability: subtract max
    c_max = np.max(c)
    exp_t_c = np.exp(t * (c - c_max))
    log_sum = np.log(np.sum(exp_t_c)) / t + c_max
    # Gradient: softmax-weighted average of dc_k/da
    weights = exp_t_c / np.sum(exp_t_c)
    # But this is over all 2n-1 lags, which is a lot
    # For efficiency, only use the top K
    if len(c) > 64:
        top_idx = np.argpartition(c, -64)[-64:]
        # Renormalize weights over top K
        top_weights = exp_t_c[top_idx]
        top_weights /= np.sum(top_weights)
        active = top_idx
        w = top_weights
    else:
        active = np.arange(len(c))
        w = weights
    
    n = a.size
    grad = np.zeros(n, dtype=np.float64)
    for idx, k in enumerate(active):
        k = int(k)
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        if i_max >= i_min:

            grad[m_idx] += w[idx] * 2.0 * a[k - m_idx]
    grad -= np.mean(grad)  # tangent to simplex
    return log_sum, grad
```

Actually, I realize I need to be careful about the simplex projection. The standard Euclidean projection onto the simplex is:
1. Sort v in decreasing order: v_(1) >= v_(2) >= ... >= v_(n)
2. Find the largest rho such that v_(rho) + 1 - (1/rho) * sum_{i=1}^{rho} v_(i) > 0
3. theta = (1 - sum_{i=1}^{rho} v_(i)) / rho
4. w_i = max(v_i - theta, 0)

Let me make sure my implementation is correct.

```python
def _simplex_project(v):
    n = v.size
    u = np.sort(v)[::-1]  # decreasing
    cssv = np.cumsum(u)
    # rho = max{j in {1,...,n} : u_j - (cssv[j]-1)/j > 0}
    rho_candidates = np.arange(1, n + 1)
    mask = u > (cssv - 1.0) / rho_candidates
    # Find the largest j where mask is true
    valid = np.where(mask)[0]
    if len(valid) == 0:
        return np.ones(n) / n
    rho = valid[-1] + 1  # convert to 1-indexed
    theta = (cssv[rho - 1] - 1.0) / rho
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s > 1e-15:
        w /= s
    else:
        w = np.ones(n) / n
    return w
```

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1

    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    c = np.maximum(c, 0.0)  # remove tiny negative values from FFT
    return c[:need]
