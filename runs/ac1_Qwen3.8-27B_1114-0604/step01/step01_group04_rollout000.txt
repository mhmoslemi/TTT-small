"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based fixed-point
iterations, CMA-ES on parameterized cosine families, and subgradient polish.
Combines multiple optimization strategies across sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    v = v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix(f):
    """Build matrix A where (A @ g)_k = (f*g)_k for full convolution."""
    n = f.size
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp_for_g(f, M):
    """Solve LP: maximize sum(g) s.t. conv(f,g) <= M, g >= 0."""
    n = f.size
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            if g is not None and np.all(np.isfinite(g)):
                return g
    except Exception:
        pass
    return None


def lp_fixed_point(f, time_budget, rng, verbose=False):
    """
    LP-based fixed-point iteration:
    - Compute M = max(f*f)
    - Solve LP for g maximizing sum(g) s.t. conv(f,g) <= M, g>=0
    - Update f = (1-t)*f + t*g (rescaled to sum=1)
    """
    n = f.size
    f = project_simplex(f.astype(np.float64).copy())
    best_f = f.copy()
    best_sc = normed_score(f)

    t0 = time.monotonic()
    t = 0.3  # mixing parameter
    iter_count = 0
    last_improve = 0
    patience = 500

    while time.monotonic() - t0 < time_budget:
        b = full_conv(f)
        M = float(np.max(b))
        cur_sc = 2.0 * n * M

        # Solve LP
        g = solve_lp_for_g(f, M)
        if g is None:
            t *= 0.9
            continue

        g_sum = float(np.sum(g))
        if g_sum < 1e-10:
            t *= 0.9
            continue

        # Rescale g to sum=1
        g = g / g_sum

        # Mix
        f_new = (1.0 - t) * f + t * g
        f_new = project_simplex(f_new)

        new_sc = normed_score(f_new)
        f = f_new
        iter_count += 1

def param_cosine(n, params, K):
    """Generate sequence from cosine parameters: a_i = max(0, 1 + sum w_k cos(2*pi*k*i/n + phi_k))"""

        phi = params[2 * k + 1]

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:

    return a / s

def smooth_grad_single(a, b, tau):
    """Smoothed subgradient using softmax weights at one temperature."""
    n = a.size
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(b.size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= s
    g = 2.0 * np.correlate(w, a, mode='valid')
    return g


def subgrad_optimize(a, time_budget, rng):

    eta = 0.2
    tau_frac = 0.03
    beta = 0.5
    vel = np.zeros(n)
    iters = 0
    last_improve = 0

    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))
        tau = tau_frac * float(np.max(b))
        if tau < 1e-18:
            tau = 1e-18

        g = smooth_grad_single(a, b, tau)
        vel = beta * vel + (1.0 - beta) * g
        a_new = project_simplex(a - eta * vel)

        b_new = full_conv(a_new)
        new_sc = 2.0 * n * float(np.max(b_new))

            eta = 0.1 + 0.2 * rng.random()
            last_improve = iters

    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / c) ** 2 + 0.1
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.05
    return a / a.sum()


def init_noisy_uniform(n, rng):
    a = np.ones(n) + 0.8 * rng.standard_normal(n)
    a = np.abs(a) + 0.01
    return a / a.sum()


def init_cosine(n, rng):
    i = np.arange(n, dtype=np.float64)
    freq = 1 + int(rng.integers(0, 3))
    a = (1.0 + freq * np.cos(2.0 * np.pi * i / max(n - 1, 1))) ** 2
    a += 1e-6
    return a / a.sum()


def init_sqrt_decay(n, rng):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a += 0.05 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_v_shape(n):
    """High in middle, low at edges."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - 0.8 * ((i - c) / c) ** 2
    a = np.maximum(a, 0.01)
    return a / a.sum()


def get_inits(n, rng):
    inits = []
    inits.append(init_two_cluster(n, rng))
    inits.append(init_u_shape(n))
    inits.append(init_sided(n))
    inits.append(init_random(n, rng))
    inits.append(init_noisy_uniform(n, rng))
    inits.append(init_cosine(n, rng))
    inits.append(init_sqrt_decay(n, rng))
    inits.append(init_v_shape(n))
    return inits


def nlms_refine(a, time_budget, rng):
    """NLMS-style refinement focusing on reducing peak lags."""
    n = a.size
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    t0 = time.monotonic()
    alpha = 0.3
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        peak_idx = int(np.argmax(b))
        cur_sc = 2.0 * n * float(np.max(b))

        # Gradient of max b at peak_idx
        # b[peak_idx] = sum_i a_i * a[peak_idx - i] (with bounds)
        # d(b[peak_idx])/d(a_j) = a[peak_idx - j] + a[j - peak_idx] (valid indices)
        grad = np.zeros(n)
        for j in range(n):
            v1 = peak_idx - j
            v2 = j - peak_idx
            contrib = 0.0
            if 0 <= v1 < n:
                contrib += a[v1]
            if 0 <= v2 < n:
                contrib += a[v2]
            grad[j] = contrib

        # NLMS step
        denom = np.dot(grad, grad) + 1e-12
        step = alpha * grad / denom
        a_new = project_simplex(a - step)

        new_sc = normed_score(a_new)
        if new_sc < cur_sc:
            a = a_new
            if new_sc < best_sc:
                best_sc = new_sc
                best_a = a.copy()
        else:
            alpha *= 0.95

        if alpha < 0.01:
            alpha = 0.1 + 0.3 * rng.random()
            a = project_simplex(best_a + 0.02 * rng.standard_normal(n))

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Combines LP fixed-point, CMA-ES on cosine params, and subgradient methods.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    # Load previous best
    prev_a = None
    prev_sc = np.inf
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size >= 10 and np.all(np.isfinite(h)) and h.sum() > 0.01:
            prev_a = h / h.sum()
            prev_sc = 2.0 * h.size * float(np.max(full_conv(prev_a)))
            print(f"[init] Previous best: n={h.size}, score={prev_sc:.6f}")
    except Exception:
        pass

    best_overall = None
    best_overall_sc = np.inf

    if prev_a is not None and prev_sc < 2.0:
        best_overall = prev_a.copy()
        best_overall_sc = prev_sc

    # n values to try
    n_list = [64, 128, 256, 512]
    if prev_a is not None:
        np_ = prev_a.size
        if np_ not in n_list and np_ >= 32:
            n_list.append(np_)
        n_list = sorted(set(n_list))

    total_time = deadline - time.monotonic()
    print(f"[plan] n_list={n_list}, total_time={total_time:.1f}s")

    # Time allocation: more time for n=128 (where previous best is), some for others
    time_weights = {}
    for nv in n_list:
        if nv == 128:
            time_weights[nv] = 0.35
        elif nv == 64:
            time_weights[nv] = 0.15
        elif nv == 256:
            time_weights[nv] = 0.20
        elif nv == 512:
            time_weights[nv] = 0.15
        else:
            time_weights[nv] = 0.15
    total_w = sum(time_weights.values())
    for nv in time_weights:
        time_weights[nv] /= total_w

    for nv in n_list:
        if time.monotonic() > deadline:
            break

        time_n = time_weights.get(nv, 0.1) * total_time * 0.90
        t_start = time.monotonic()
        print(f"\n[n={nv}] Starting, budget={time_n:.1f}s")

        inits = get_inits(nv, rng)
        if prev_a is not None and prev_a.size == nv:
            inits.insert(0, prev_a.copy())

        # Phase 1: LP fixed-point (main optimizer)
        lp_time = time_n * 0.50
        # Distribute among a few initializations for LP
        n_inits_lp = min(3, len(inits))
        lp_per = lp_time / n_inits_lp

        for ii in range(n_inits_lp):
            if time.monotonic() > deadline:
                break
            tb = min(lp_per, deadline - time.monotonic() - 5.0)
            if tb < 5.0:
                break
            a_best, sc_best = lp_fixed_point(inits[ii], tb, rng)
            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = a_best.copy()
                print(f"  [n={nv}] LP init={ii}: score={sc_best:.6f} (BEST)")
            else:
                print(f"  [n={nv}] LP init={ii}: score={sc_best:.6f}")

        # Phase 2: CMA-ES on parameterized family
        cma_time = min(time_n * 0.25, deadline - time.monotonic() - 5.0)
        if cma_time > 10.0:
            K = min(8, max(4, nv // 32))
            a_cma, sc_cma, _ = cma_es_parametric(nv, K, cma_time, rng)
            if sc_cma < best_overall_sc:
                best_overall_sc = sc_cma
                best_overall = a_cma.copy()
                print(f"  [n={nv}] CMA-ES: score={sc_cma:.6f} (BEST)")
            else:
                print(f"  [n={nv}] CMA-ES: score={sc_cma:.6f}")

        # Phase 3: Subgradient + NLMS polish on best for this n
        if best_overall is not None and best_overall.size == nv:
            pol_time = min(time_n * 0.15, deadline - time.monotonic() - 5.0)
            if pol_time > 5.0:
                a_p, sc_p = subgrad_optimize(best_overall, pol_time, rng)
                if sc_p < best_overall_sc:
                    best_overall_sc = sc_p
                    best_overall = a_p.copy()
                    print(f"  [n={nv}] subgrad polish: score={sc_p:.6f} (BEST)")

            pol_time2 = min(time_n * 0.10, deadline - time.monotonic() - 5.0)
            if pol_time2 > 5.0:
                a_p2, sc_p2 = nlms_refine(best_overall, pol_time2, rng)
                if sc_p2 < best_overall_sc:
                    best_overall_sc = sc_p2
                    best_overall = a_p2.copy()
                    print(f"  [n={nv}] NLMS polish: score={sc_p2:.6f} (BEST)")

        # Phase 4: More LP iterations on the best found for this n
        lp2_time = max(0.0, time_n - (time.monotonic() - t_start) - 5.0)
        if lp2_time > 10.0 and best_overall is not None and best_overall.size == nv:
            a_lp2, sc_lp2 = lp_fixed_point(best_overall.copy(), lp2_time, rng)
            if sc_lp2 < best_overall_sc:
                best_overall_sc = sc_lp2
                best_overall = a_lp2.copy()
                print(f"  [n={nv}] LP round 2: score={sc_lp2:.6f} (BEST)")

        print(f"[n={nv}] Done. Overall best: {best_overall_sc:.6f}")

    # Final polish on overall best
    if best_overall is not None and time.monotonic() < deadline - 10.0:
        remain = deadline - time.monotonic() - 5.0
        print(f"\n[final polish] remaining={remain:.1f}s, n={best_overall.size}")

        # Multiple rounds of LP + subgrad
        n_rounds = 3
        per_round = remain / n_rounds
        for r in range(n_rounds):
            if time.monotonic() > deadline:
                break
            tb = min(per_round * 0.5, deadline - time.monotonic() - 3.0)
            if tb > 3.0:
                a_r, sc_r = lp_fixed_point(best_overall.copy(), tb, rng)
                if sc_r < best_overall_sc:
                    best_overall_sc = sc_r
                    best_overall = a_r.copy()
                    print(f"  [polish r{r}] LP: score={sc_r:.6f}")

            tb2 = min(per_round * 0.4, deadline - time.monotonic() - 3.0)
            if tb2 > 3.0:
                a_r2, sc_r2 = subgrad_optimize(best_overall.copy(), tb2, rng)
                if sc_r2 < best_overall_sc:
                    best_overall_sc = sc_r2
                    best_overall = a_r2.copy()
                    print(f"  [polish r{r}] subgrad: score={sc_r2:.6f}")

    # Fallback
    if best_overall is None:
        best_overall = np.ones(128) / 128
        best_overall_sc = 2.0

    # Final verification
    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"\n[final] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")

    return result
