```python
"""
Optimizer for minimizing
    2*n*max(conv(a,a)) / (sum(a)^2)
over non-negative sequences a, using:
  * FFT-based convolution and exact analytic gradients of smoothed max
  * candidate generation from parent, low-discrepancy shapes, spectral
    square-root flattening, and capped-convolution Newton iterations
  * Adam-style projected gradient methods with adaptive top-k peaks,
    restarts, mixing, and final polish
"""

import numpy as np
import time


def next_pow2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    if not np.all(np.isfinite(a)):
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    a /= s
    return a


def resample_array(arr, n):
    arr = np.asarray(arr, dtype=np.float64).ravel()
    n = int(n)
    if arr.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    if arr.size == n:
        return arr.copy()
    x_old = np.linspace(0.0, 1.0, arr.size, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n, endpoint=False)
    return np.interp(x_new, x_old, arr)


def fast_convolve(a, L):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = int(a.size)
    M = 2 * n - 1
    if L is None:
        L = next_pow2(M)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:M]
    c = np.maximum(c, 0.0)
    return c


def score_of_array(a, n, L):
    a = normalize(a)
    c = fast_convolve(a, L)
    s = float(a.sum())
    if s <= 1e-15:
        return np.inf
    return 2.0 * float(n) * float(np.max(c)) / (s * s)


def smooth_noise(n, rng, band):
    N = next_pow2(n)
    x = rng.standard_normal(n)
    X = np.fft.rfft(x, N)
    f = np.fft.rfftfreq(N)
    X *= 1.0 / (1.0 + (float(band) * f) ** 2)
    noise = np.fft.irfft(X, N)[:n]
    return noise - noise.mean()


def topk_weights(c, K, beta=0.05):
    c = np.asarray(c, dtype=np.float64).ravel()
    M = int(c.size)
    K = int(max(1, min(K, M)))

    if K == 1:
        return np.array([int(np.argmax(c))], dtype=np.int64), np.array([1.0])

    if K >= M:
        idx = np.arange(M, dtype=np.int64)
    else:
        idx = np.argpartition(c, -K)[-K:]

    vals = c[idx].astype(np.float64)
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]

    top = float(vals[0])
    if top <= 1e-300:
        return idx, np.ones(K, dtype=np.float64) / float(K)

    last = float(vals[-1])
    spread = max(1e-300, top - last)
    if spread <= 1e-300:
        w = np.zeros(K, dtype=np.float64)
        w[0] = 1.0
        return idx, w

    tau = max(1e-300, float(beta) * spread)
    w = np.exp((vals - top) / tau)
    w = np.where(w > 0.0, w, 0.0)
    sw = float(w.sum())
    if not np.isfinite(sw) or sw <= 1e-300:
        w = np.zeros(K, dtype=np.float64)
        w[0] = 1.0
    else:
        w = w / sw

    w[w < 1e-10] = 0.0
    sw = float(w.sum())
    if sw > 1e-300:
        w = w / sw
    else:
        w = np.zeros(K, dtype=np.float64)
        w[0] = 1.0

    return idx, w


def gradient_direct(a, idx, wts, n):
    g = np.zeros(n, dtype=np.float64)
    for t in range(idx.size):
        wi = float(wts[t])
        if wi <= 1e-12:
            continue
        ki = int(idx[t])
        st = ki - n + 1
        if st < 0:
            st = 0
        en = ki
        if en >= n:
            en = n - 1
        if st <= en:
            g[st:en + 1] += (2.0 * wi) * a[st:en + 1][::-1]
    return g


def gradient_fft(a, idx, wts, n, Lcorr):
    M = 2 * n - 1
    w = np.zeros(M, dtype=np.float64)
    w[idx] = wts
    W = np.fft.rfft(w, Lcorr)
    A = np.fft.rfft(a, Lcorr)
    corr = np.fft.irfft(W * np.conj(A), Lcorr)[:n]
    return 2.0 * corr


def newton_square(a, G, L, iterations=1, alpha=0.5, reg=1e-8):
    a = normalize(a)
    if iterations <= 0:
        return a
    for _ in range(int(iterations)):
        A = np.fft.rfft(a, L)
        denom = A * np.conj(A)
        maxd = float(np.max(denom)) if denom.size else 1.0
        if maxd <= 1e-300:
            break
        B = G * np.conj(A) / (denom + float(reg) * maxd)
        Anew = float(alpha) * A + (1.0 - float(alpha)) * B
        an = np.fft.irfft(Anew, L)[:a.size]
        an = np.real(an)
        a = normalize(an)
    return a


def score_key(item):
    return item[1]


def add_candidate(cands, a, n, L):
    a = normalize(a)
    sc = score_of_array(a, n, L)
    cands.append((a, sc))


def add_capped_newton_initial(cands, a_start, n, L, max_iters=18, alpha=0.45):
    n = int(n)
    a = normalize(a_start)
    best_a = a.copy()
    best_sc = score_of_array(a, n, L)
    c = fast_convolve(a, L)
    M = 2 * n - 1
    avg = 1.0 / float(M)
    cap = max(avg * 1.01, float(np.max(c)) * 0.93)
    stagnant = 0

    for _ in range(int(max_iters)):
        target = np.minimum(c, cap)
        ts = float(target.sum())
        if ts <= 1e-300:
            target = np.ones(M, dtype=np.float64) / float(M)
        else:
            target = target / ts

        G = np.fft.rfft(target, L)
        cand = newton_square(a, G, L, iterations=2, alpha=alpha, reg=1e-9)
        sc = score_of_array(cand, n, L)

        if np.isfinite(sc) and sc < best_sc - 1e-13:
            best_sc = sc
            best_a = cand.copy()
            a = cand.copy()
            stagnant = 0
        else:
            stagnant += 1
            blend = 0.70 * best_a + 0.30 * cand
            blend = normalize(blend)
            sb = score_of_array(blend, n, L)
            if np.isfinite(sb) and sb < best_sc - 1e-13:
                best_sc = sb
                best_a = blend.copy()
                a = blend.copy()
                stagnant = 0

        if stagnant >= 8:
            break

        cap = max(avg * 1.002, cap * 0.97)
        c = fast_convolve(a, L)

    add_candidate(cands, best_a, n, L)


def make_inits(n, rng, L, parent, Gflat):
    cands = []
    n = int(n)

    if parent is not None:
        add_candidate(cands, resample_array(parent, n), n, L)

    add_candidate(cands, np.ones(n, dtype=np.float64), n, L)

    x = np.linspace(0.0, 1.0, n, endpoint=False)

    for freq in (1, 2, 3):
        for amp in (0.3, 0.6, 1.0, 1.5, 2.0):
            for ph in (0.0, 0.25, 0.5, 0.75):
                v = 1.0 + float(amp) * np.cos(2.0 * np.pi * float(freq) * x + 2.0 * np.pi * ph)
                add_candidate(cands, np.clip(v, 1e-8, None), n, L)

    for amp in (0.5, 1.0, 2.0):
        add_candidate(cands, 1.0 + float(amp) * (2.0 * x - 1.0), n, L)
        add_candidate(cands, 1.0 - float(amp) * (2.0 * x - 1.0), n, L)
        add_candidate(cands, 1.0 + float(amp) * ((2.0 * x - 1.0) ** 2), n, L)
        add_candidate(cands, 1.0 + float(amp) * ((2.0 * x - 1.0) ** 4), n, L)

    for sig in (0.05, 0.10, 0.20, 0.35):
        for mu in (0.0, 0.25, 0.5, 0.75, 1.0):
            v = np.exp(-((x - mu) ** 2) / (2.0 * sig * sig))
            add_candidate(cands, v, n, L)

    for band in (1.0, 3.0, 10.0, 30.0):
        v = 1.0 + 0.5 * smooth_noise(n, rng, band)
        add_candidate(cands, v, n, L)

    for band in (2.0, 8.0):
        v = np.clip(1.0 + smooth_noise(n, rng, band), 1e-8, None)
        add_candidate(cands, v, n, L)

    cands.sort(key=score_key)

    # Keep a manageable number of diverse candidates.
    kept = []
    for a, sc in cands:
        if len(kept) >= 24:
            break
        if any(np.linalg.norm(a - b) < 0.05 for b, _ in kept):
            continue
        kept.append((a, sc))

    cands = kept[:16]

    # Spectral square-root flattening from very good plain initial images.
    for a, sc in cands[:6]:
        cand = newton_square(a, Gflat, L, iterations=14, alpha=0.5, reg=1e-10)
        add_candidate(cands, cand, n, L)

    cands.sort(key=score_key)

    # Capped convolution Newton from the best two candidates.
    for a, sc in cands[:3]:
        add_capped_newton_initial(cands, a, n, L, max_iters=16, alpha=0.45)

    cands.sort(key=score_key)
    return cands


def perturb_best(best, rng, n, L):
    best = np.asarray(best, dtype=np.float64).ravel()
    n = int(n)
    r = rng.random()

    if r < 0.40:
        a = best + 0.015 + 0.035 * rng.random(n)
    elif r < 0.70:
        alpha = 0.04 + 0.10 * rng.random()
        band = 1.5 + 12.0 * rng.random()
        noise = smooth_noise(n, rng, band)
        a = best * (1.0 + alpha * noise)
        a += 0.004 * np.ones(n, dtype=np.float64)
    else:
        blocks = max(1, n // 64)
        per = max(1, (n + blocks - 1) // blocks)
        mult = 0.45 + rng.random(blocks)
        scale = np.repeat(mult, per)
        if scale.size > n:
            scale = scale[:n]
        elif scale.size < n:
            scale = np.concatenate([scale, np.ones(n - scale.size)])
        a = best * scale + 0.004 * np.ones(n, dtype=np.float64)

    return normalize(a)


def mix_two(a, b, theta):
    a = np.asarray(a, dtype=np.float64).ravel()
    b = np.asarray(b, dtype=np.float64).ravel()
    n = a.size
    b = resample_array(b, n)
    mix = float(theta) * a + (1.0 - float(theta)) * b
    mix += 0.001 * np.ones(n, dtype=np.float64)
    return normalize(mix)


def optimize_internal(a0, end_time, rng, n, L, Lcorr,
                       K_start=64, K_end=8, lr0=0.05, patience=1800):
    a = normalize(a0)
    n = int(n)

    c = fast_convolve(a, L)
    s = float(a.sum())
    score = 2.0 * float(n) * float(np.max(c)) / (s * s)

    best_a = a.copy()
    best_score = score

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    b1_t = 1.0
    b2_t = 1.0
    t = 0
    lr = float(lr0)
    no_improve = 0

    t0 = time.monotonic()
    total = max(1e-6, float(end_time) - t0)

    while time.monotonic() < end_time:
        c = fast_convolve(a, L)
        s = float(a.sum())
        score = 2.0 * float(n) * float(np.max(c)) / (s * s)

        improved = False
        if np.isfinite(score) and score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            improved = True

        if improved:
            lr = min(0.12, lr * 1.01)
        else:
            no_improve += 1
            if no_improve >= patience:
                a = perturb_best(best_a, rng, n, L)
                m[:] = 0.0
                v[:] = 0.0
                b1_t = 1.0
                b2_t = 1.0
                t = 0
                lr = max(0.005, lr * 0.70)
                patience = max(200, int(patience * 0.70))
                no_improve = 0
                continue

            if no_improve > 120 and no_improve % 50 == 0:
                lr = max(0.005, lr * 0.995)

        frac = min(1.0, max(0.0, (time.monotonic() - t0) / total))
        K = int(max(K_end, K_start + float(K_end - K_start) * frac))
        if no_improve > max(100, patience // 4):
            K = max(6, K // 2)

        # If the maximum is clearly unique, attack it exactly.
        if K > 1 and c.size >= 3:
            part2 = np.argpartition(c, -2)[-2:]
            vals2 = np.sort(c[part2])[::-1]
            top2 = float(vals2[0])
            second = float(vals2[1])
            if top2 > 1e-300 and top2 > second * 1.007:
                if top2 - second > 1e-14 * max(1.0, top2):
                    K = 1

        idx, wts = topk_weights(c, K, beta=0.05)
        eff = int(np.count_nonzero(wts > 1e-10))
        if eff > 80:
            g = gradient_fft(a, idx, wts, n, Lcorr)
        else:
            g = gradient_direct(a, idx, wts, n)

        g = g - float(np.mean(g))
        if not np.all(np.isfinite(g)):
            a = perturb_best(best_a, rng, n, L)
            m[:] = 0.0
            v[:] = 0.0
            b1_t = 1.0
            b2_t = 1.0
            t = 0
            lr = max(0.005, lr * 0.70)
            no_improve = 0
            continue

        rms = float(np.sqrt(np.mean(g * g)))
        if rms <= 1e-30:
            a = perturb_best(best_a, rng, n, L)
            m[:] = 0.0
            v[:] = 0.0
            b1_t = 1.0
            b2_t = 1.0
            t = 0
            lr = max(0.005, lr * 0.70)
            no_improve = 0
            continue

        g = g / rms

        t += 1
        b1_t *= 0.90
        b2_t *= 0.99
        m = 0.90 * m + 0.10 * g
        v = 0.99 * v + 0.01 * g * g
        mh = m / (1.0 - b1_t)
        vh = v / (1.0 - b2_t)
        step = mh / (np.sqrt(vh) + 1e-8)

        a = a - lr * step
        a = np.clip(a, 0.0, None)
        sa = float(a.sum())
        if sa <= 1e-15:
            a = np.ones(n, dtype=np.float64) / float(n)
        else:
            a = a / sa

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    try:
        rng = np.random.default_rng(seed)
        start = time.monotonic()
        budget = max(60.0, float(budget_s))
        deadline = start + budget - 40.0

        parent = None
        if 'height_sequence_1' in globals():
            try:
                parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            except Exception:
                parent = None

        if parent is not None and parent.size > 0:
            n = int(parent.size)
        else:
            n = 8192

        # Keep final direct evaluator cost reasonable.
        n = int(max(1024, min(n, 16384)))

        if parent is not None:
            parent = resample_array(parent, n)

        L = next_pow2(2 * n - 1)
        Lcorr = next_pow2(3 * n - 2)
        M = 2 * n - 1

        print("optimizer n=%d L=%d Lcorr=%d budget=%.0fs" % (n, L, Lcorr, budget))

        Gflat = np.fft.rfft(np.ones(M, dtype=np.float64) / float(M), L)

        cands = make_inits(n, rng, L, parent, Gflat)
        if not cands:
            cands = [ (np.ones(n, dtype=np.float64) / float(n), score_of_array(np.ones(n), n, L)) ]

        cands.sort(key=score_key)

        selected = []
        for a, sc in cands:
            if len(selected) >= 8:
                break
            if not any(np.linalg.norm(a - b) < 0.04 for b, _ in selected):
                selected.append((a, sc))

        if not selected:
            selected = cands[:1]

        best_a, best_score = selected[0]
        print("best initial internal score = %.10g" % best_score)

        # Phase A: short local refinements of several diverse starts.
        phase_end1 = min(deadline - 150.0, start + 0.15 * (deadline - start))
        if phase_end1 > start + 20.0:
            total1 = max(0.0, phase_end1 - start)
            per1 = max(12.0, total1 / 4.0)
            for a0, _ in selected[:4]:
                t_end = min(phase_end1, time.monotonic() + per1)
                a1, s1 = optimize_internal(a0, t_end, rng, n, L, Lcorr,
                                            K_start=80, K_end=16, lr0=0.06, patience=700)
                if np.isfinite(s1) and s1 < best_score:
                    best_score = s1
                    best_a = a1.copy()
                    print("phase A improvement -> %.10g" % best_score)

        # Phase B: main refinement of the global best.
        phase_end2 = min(deadline - 90.0, start + 0.70 * (deadline - start))
        if phase_end2 > time.monotonic() + 30.0:
            a2, s2 = optimize_internal(best_a, phase_end2, rng, n, L, Lcorr,
                                       K_start=56, K_end=8, lr0=0.05, patience=2600)
            if np.isfinite(s2) and s2 < best_score:
                best_score = s2
                best_a = a2.copy()
            print("phase B best internal score = %.10g" % best_score)

        # Phase C: mix with a strong alternative, or perturb, then refine.
        phase_end3 = min(deadline - 50.0, start + 0.87 * (deadline - start))
        if phase_end3 > time.monotonic() + 30.0:
            if len(selected) >= 2 and selected[1][1] < best_score + 0.04:
                theta = 0.55 + 0.35 * rng.random()
                start_mix = mix_two(best_a, selected[1][0], theta)
            else:
                start_mix = perturb_best(best_a, rng, n, L)
            a3, s3 = optimize_internal(start_mix, phase_end3, rng, n, L, Lcorr,
                                       K_start=40, K_end=6, lr0=0.035, patience=1400)
            if np.isfinite(s3) and s3 < best_score:
                best_score = s3
                best_a = a3.copy()
            print("phase C best internal score = %.10g" % best_score)

        # Phase D: final polish with very small active sets.
        phase_end4 = max(time.monotonic(), deadline - 5.0)
        if phase_end4 > time.monotonic() + 10.0:
            a4, s4 = optimize_internal(best_a, phase_end4, rng, n, L, Lcorr,
                                       K_start=24, K_end=3, lr0=0.018, patience=900)
            if np.isfinite(s4) and s4 < best_score:
                best_score = s4
                best_a = a4.copy()

        if best_a is None or best_a.size == 0:
            best_a = np.ones(n, dtype=np.float64) / float(n)

        best_a = normalize(best_a)
        out = [float(x) for x in best_a]

        try:
            verified = float(evaluate_sequence(out))
        except Exception:
            verified = np.inf

        print("final internal=%.10g verified=%.10g len=%d elapsed=%.1fs"
              % (best_score, verified, len(out), time.monotonic() - start))

        if not np.isfinite(verified):
            out = [float(1.0) for _ in range(1024)]
            try:
                verified = float(evaluate_sequence(out))
            except Exception:
                verified = np.inf
            print("fallback uniform verified=%.10g" % verified)

        return out

    except Exception as exc:
        try:
            n = 1024
            out = [float(1.0) for _ in range(n)]
            print("top-level error, returning uniform length %d: %s" % (n, exc))
            return out
        except Exception:
            return [1.0]
```
