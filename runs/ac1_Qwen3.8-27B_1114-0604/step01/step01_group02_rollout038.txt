```
def project_simplex(v, s=1.0):
    n=v.size
    if n==0: return v
    u=np.sort(v)[::-1]
    cssv=np.cumsum(u)
    rho=np.nonzero(u * np.arange(1,n+1) > (cssv - s))[0]
    if len(rho)==0: return np.zeros(n)
    rho=rho[-1]
    theta=(cssv[rho]-s)/(rho+1.0)
    w=np.maximum(v-theta,0)
    return w
```

```
    mask=maxv-1

    full_poly=(1<<m) | p
    sq_mono=[]
```

```
    mstate = np.zeros(n, dtype=np.float64)

    step = 0
    no_improve = 0

    max_restarts = 30
    while time.monotonic() < end:

        progress = min(1.0, elapsed / total_est)
        a = normalize_inplace(a, target_sum)
        c = fast_self_conv(a)
        score = score_from_arrays(a,c)
        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve >= max(80, patience):
            if restarts >= max_restarts:
                break
            a = perturb_vector(best_a, rng, 0.08 + 0.12*float(rng.random()))
            a = normalize_inplace(a, target_sum)
            mstate.fill(0.0); vstate.fill(0.0)
            step=0; no_improve=0; restarts += 1
            continue
        use_mn = bool(minnorm and (step % max(1,int(minnorm_every)) == 0))
        if use_mn:
            mncur = int(minnorm_k * (1.0 + 0.5*progress))
            if mncur < 4: mncur = 4
            idx, vals = topk_idx_vals(c, mncur)
            G = compute_minnorm_direction(a, idx, c, mncur, 30)

            idx, vals = topk_idx_vals(c, ksoft)
            if vals.size > 1:
```

```python
"""
Search for non-negative sequences minimizing
    2*n*max(conv(a,a)) / (sum(a)^2).

Strategy:
  * Use many structured and random initialization families:
      - Zadoff-Chu-like cosine chirps,
      - smooth dense shapes,
      - m-sequences (via trace of primitive-polynomial fields),
      - Legendre quadratic-residue sequences,
      - random sinusoidal and Gaussian shapes.
  * Score them with a fast FFT self-convolution.
  * Run a projected nonsmooth optimizer with Adam on the simplex
    (sum fixed to n, non-negativity enforced by clipping+renormalization).

  * The nonsmooth max is handled by:
      - soft top-k surrogate gradients in coarse phases,
      - active-set min-norm subgradient directions in refinement phases.
  * Periodic randomized restarts from the best incumbent prevent many local traps.
  * Longer lengths are tried (up to about 16384) because larger sequences tend
    to give more freedom in the peak-averaging trade-off.
"""

import numpy as np
import time


# -----------------------------------------------------------------------------
# Basic helpers
# -----------------------------------------------------------------------------

def next_power_of_two(x):
    if x <= 1:
        return 1
    return 1 << (int(x) - 1).bit_length()


def fast_self_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    m = 2 * n - 1
    L = next_power_of_two(m)
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:m]
    return c


def score_from_arrays(a, c):
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    if c.size == 0:
        return float("inf")
    mb = float(np.max(c))
    if not np.isfinite(mb):
        return float("inf")
    return float(2.0 * a.size * mb / (s * s))


def compute_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n < 2:
        return float("inf")
    s = float(np.sum(a))
    if s < 1e-12:
        return float("inf")
    c = fast_self_conv(a)
    return score_from_arrays(a, c)


def normalize_inplace(a, target_sum):
    a = np.asarray(a, dtype=np.float64)
    if a.size == 0:
        return a
    np.nan_to_num(a, copy=False, nan=0.0, posinf=0.0, neginf=0.0)
    np.clip(a, 0.0, 1000.0, out=a)
    s = float(np.sum(a))
    if s <= 1e-12:
        a.fill(max(1e-9, float(target_sum) / max(1, a.size)))
        return a
    a *= float(target_sum) / s
    if a.size > 0 and float(np.max(a)) > 990.0:
        np.clip(a, 0.0, 990.0, out=a)
        s = float(np.sum(a))
        if s <= 1e-12:
            a.fill(max(1e-9, float(target_sum) / max(1, a.size)))
        else:
            a *= float(target_sum) / s
    return a


def topk_idx_vals(c, k):
    M = int(c.size)
    if M == 0:
        return np.array([], dtype=np.int64), np.array([], dtype=np.float64)
    k = min(max(1, int(k)), M)
    if k >= M:
        idx = np.argsort(c)[::-1]
        return idx, c[idx]
    idx = np.argpartition(c, -k)[-k:]
    vals = c[idx]
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]
    return idx, vals


def softmax_weights(vals, temp):
    vals = np.asarray(vals, dtype=np.float64).ravel()
    k = vals.size
    if k == 0:
        return np.zeros(0, dtype=np.float64)
    if k == 1:
        return np.ones(1, dtype=np.float64)
    m = float(np.max(vals))
    temp = max(1e-30, float(temp))
    e = np.exp((vals - m) / temp)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(k, dtype=np.float64) / k
    w = e / s
    if not np.all(np.isfinite(w)):
        return np.ones(k, dtype=np.float64) / k
    return w


def project_simplex(v, s=1.0):
    w = np.asarray(v, dtype=np.float64).copy()
    n = w.size
    if n == 0:
        return w
    if not np.all(np.isfinite(w)):
        w = np.zeros(n, dtype=np.float64)
        w[0] = 1.0
        return w
    u = np.sort(w)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - s))[0]
    if rho.size == 0:
        return np.ones(n, dtype=np.float64) / n
    rho = int(rho[-1])
    theta = (cssv[rho] - s) / (rho + 1.0)
    w = np.maximum(w - theta, 0.0)
    ss = float(np.sum(w))
    if ss > 0.0:
        w /= ss
    else:
        w = np.ones(n, dtype=np.float64) / n
    return w


def accumulate_weighted_gradient(a, idx, w):
    n = a.size
    G = np.zeros(n, dtype=np.float64)
    L = min(idx.size, w.size)
    for i in range(L):
        wi = float(w[i])
        if wi <= 0.0:
            continue
        k = int(idx[i])
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if js > je:
            continue
        G[js:je + 1] += 2.0 * wi * a[js:je + 1][::-1]
    if n > 0:
        G -= float(np.mean(G))
    return G


def compute_minnorm_direction(a, idx, c, minnorm_k, w_iters=30):
    n = a.size
    K = min(int(minnorm_k), int(idx.size))
    if K <= 0:
        return np.zeros(n, dtype=np.float64)
    if K == 1:
        return accumulate_weighted_gradient(a, idx[:1], np.ones(1, dtype=np.float64))

    Gmat = np.zeros((K, n), dtype=np.float64)
    for r in range(K):
        k = int(idx[r])
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        if js <= je:
            Gmat[r, js:je + 1] = 2.0 * a[js:je + 1][::-1]
        Gmat[r] -= float(np.mean(Gmat[r]))

    vals = c[idx][:K]
    gap = float(vals[0] - vals[-1]) if K > 1 else 0.0
    maxval = float(vals[0]) if vals.size else 1.0
    temp = max(1e-12, 0.02 * maxval + 0.1 * gap)
    w = softmax_weights(vals, temp)

    Q = np.dot(Gmat, Gmat.T)
    if not np.all(np.isfinite(Q)):
        Q = np.eye(K, dtype=np.float64)

    maxdiag = float(np.max(np.diag(Q))) if K > 0 else 0.0
    if maxdiag > 1e-15:
        step = 1.0 / (maxdiag + 1e-12)
        for _ in range(max(0, int(w_iters))):
            grad = Q.dot(w)
            w = w - step * grad
            w = project_simplex(w, 1.0)
    else:
        w = np.ones(K, dtype=np.float64) / K

    G = Gmat.dot(w)
    if not np.all(np.isfinite(G)):
        G = np.zeros(n, dtype=np.float64)
    G -= float(np.mean(G))
    return G


def adam_update(a, g, m, v, step, lr, b1=0.9, b2=0.999, eps=1e-8):
    m *= b1
    m += (1.0 - b1) * g
    v *= b2
    v += (1.0 - b2) * (g * g)
    b1_pow = b1 ** step
    b2_pow = b2 ** step
    mhat = m / (1.0 - b1_pow)
    vhat = v / (1.0 - b2_pow)
    a -= lr * mhat / (np.sqrt(vhat) + eps)


def perturb_vector(best, rng, scale):
    a = np.array(best, dtype=np.float64, copy=True)
    n = a.size
    if n == 0:
        return a
    scale = float(scale)
    a += scale * rng.standard_normal(n)
    if n >= 16:
        cnt = max(1, n // 20)
        idx = rng.choice(n, size=cnt, replace=False)
        a[idx] += 0.5 * scale * rng.standard_normal(cnt)
        zero = rng.random(n) < 0.03
        a[zero] = 0.0
    return a

def make_chirp(n, q, phase=0.0):
    i = np.arange(n, dtype=np.float64)
    ph = 2.0 * np.pi * float(q) * (i * i) / float(n) + float(phase)
    return 0.5 * (1.0 + np.cos(ph))


def make_chirp_linear(n, q, phase=0.0):
    i = np.arange(n, dtype=np.float64)
    ph = 2.0 * np.pi * float(q) * i * (i + 1.0) / (2.0 * float(n)) + float(phase)
    return 0.5 * (1.0 + np.cos(ph))


def make_random_sinus(n, rng, ncomp=3):
    a = np.full(n, 0.5, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    for _ in range(int(ncomp)):
        f = int(rng.integers(1, max(2, n // 3 + 1)))
        amp = float(rng.random()) * 0.25
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        a += amp * np.cos(2.0 * np.pi * float(f) * i / float(n) + ph)
    return np.clip(a, 0.0, 1.0)


def add_candidate(cands, a, tag, max_total=300):
    if len(cands) >= int(max_total):
        return
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size < 64 or a.size > 16384:
        return
    a = np.nan_to_num(a, nan=0.0, posinf=1e3, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s <= 1e-8:
        return
    sc = compute_score(a)
    if np.isfinite(sc):
        cands.append((float(sc), np.array(a, dtype=np.float64, copy=True), str(tag)))


def candidate_key(cand):
    return cand[0]


def add_sequence_variants(cands, base, tag_prefix, rng, max_variants=24):
    N = base.size
    if N < 2:
        return
    shifts = [0, N // 4, N // 2, (N + 3) // 4, (3 * N) // 8]
    if N > 10:
        shifts.extend([int(x) for x in rng.integers(0, N, size=min(12, N))])
    count = 0
    for sh in shifts:
        if count >= int(max_variants):
            break
        sh = int(sh) % N
        for inv in (False, True):
            if count >= int(max_variants):
                break
            for rev in (False, True):
                if count >= int(max_variants):
                    break
                a = base.copy()
                if sh:
                    a = np.roll(a, sh)
                if inv:
                    a = 1.0 - a
                if rev:
                    a = a[::-1]
                add_candidate(cands, a, tag_prefix + "_" + str(count))
                count += 1

def is_small_prime(n):
    n = int(n)
    if n < 2:
        return False
    if n % 2 == 0:
        return n == 2
    r = 3
    while r * r <= n:
        if n % r == 0:
            return False
        r += 2
    return True


def is_primitive_x(m, p):
    N = (1 << m) - 1
    mask = (1 << m) - 1
    val = 1
    top_bit = 1 << (m - 1)
    for i in range(N):
        if val & top_bit:
            val = ((val << 1) & mask) ^ int(p)
        else:
            val = (val << 1) & mask
        if i + 1 < N and val == 1:
            return False
    return val == 1


def find_primitive_poly(m, rng, n_random=96):
    known = {
        4: 3,
        5: 5,
        6: 3,
        7: 3,
        8: 29,
        9: 29,
        10: 15,
        11: 5,
        12: 83,
        13: 29,
        14: 45,
    }
    extra = {
        8: [27, 43, 87, 95],
        9: [29, 62, 102, 130],
        10: [15, 139, 141, 385],
        11: [5, 83, 99, 251],
        12: [83, 115, 341, 571],
        13: [29, 79, 185, 191],
        14: [45, 169, 225, 265],
    }
    candidates = []
    if m in known:
        candidates.append(int(known[m]))
    if m in extra:
        candidates.extend([int(x) for x in extra[m]])

    max_p = 1 << (m - 1)
    for _ in range(int(n_random)):
        if max_p <= 3:
            break
        p = int(rng.integers(3, max_p)) | 1
        candidates.append(p)

    seen = set()
    for p in candidates:
        if p in seen:
            continue
        seen.add(int(p))
        p = int(p)
        if p < 1 or p >= max_p:
            continue
        if (p & 1) == 0:
            continue
        if is_primitive_x(m, p):
            return p
    return None


def msequence_trace(m, p):
    N = (1 << m) - 1
    maxv = 1 << m
    mask = maxv - 1
    full_poly = (1 << m) | int(p)

    sq_mono = []
    for e in range(m):
        r = 2 * e
        val = 1 << r
        while val >> m:
            h = val.bit_length() - 1
            val ^= full_poly << (h - m)
        sq_mono.append(val & mask)

    sq_tab = np.zeros(maxv, dtype=np.uint32)
    for v in range(1, maxv):
        lsb = v & -v
        e = lsb.bit_length() - 1
        sq_tab[v] = sq_tab[v ^ lsb] ^ sq_mono[e]

    tr_tab = np.zeros(maxv, dtype=np.uint8)
    for v in range(maxv):
        t = 0
        x = v
        for _ in range(m):
            t ^= x & 1
            x = int(sq_tab[x])
        tr_tab[v] = t

    seq = np.empty(N, dtype=np.float64)
    val = 1
    top_bit = 1 << (m - 1)
    for i in range(N):
        seq[i] = float(tr_tab[val])
        if val & top_bit:
            val = ((val << 1) & mask) ^ int(p)
        else:
            val = (val << 1) & mask
    return seq


def legendre_seq(p):
    p = int(p)
    a = np.zeros(p, dtype=np.float64)
    e = (p - 1) // 2
    for x in range(1, p):
        if pow(x, e, p) == 1:
            a[x] = 1.0
    return a


def build_structured_candidates(cands, rng, budget):
    large = budget > 1200.0
    if large:
        lenses = [1024, 2048, 4096, 8192, 16384]
    elif budget > 600.0:
        lenses = [2048, 4096, 8192, 16384]
    else:
        lenses = [1024, 2048, 4096]

    for n in lenses:
        # Chirps.
        for q in (1, 2, 3, 5):
            for ph in (0.0, 1.5707963267948966):
                add_candidate(cands, make_chirp(n, q, ph), "chirp_%d_%d" % (n, q))
                add_candidate(cands, make_chirp_linear(n, q, ph), "chirplinear_%d_%d" % (n, q))

        # Random smooth / sinusoidal.
        for r in range(2):
            a = np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.0, 1.0)
            add_candidate(cands, a, "randnormal_%d_%d" % (n, r))
        for r in range(3):
            a = make_random_sinus(n, rng, 3)
            add_candidate(cands, a, "randsin_%d_%d" % (n, r))

        # Deterministic low-complexity shapes.
        x = np.linspace(0.0, 1.0, n)
        for f in (1, 2, 3):
            add_candidate(cands, np.clip(0.5 + 0.5 * np.cos(2.0 * np.pi * f * x), 0.0, 1.0),
                          "cos_%d_%d" % (n, f))
        for p in (0.5, 1.0, 2.0):
            add_candidate(cands, np.abs(np.sin(np.pi * x)) ** p, "smooth_%d_p%d" % (n, int(10 * p)))
        for sig in (0.03, 0.08, 0.15, 0.30):
            a = np.exp(-((x - 0.5) ** 2) / (2.0 * sig * sig))
            add_candidate(cands, a, "gauss_mean_%d_s%d" % (n, int(100 * sig)))
            a = np.exp(-((x - 0.25) ** 2) / (2.0 * sig * sig)) + \
                np.exp(-((x - 0.75) ** 2) / (2.0 * sig * sig))
            add_candidate(cands, a, "gauss_two_%d_s%d" % (n, int(100 * sig)))

    # m-sequences.
    if budget > 400.0:
        if large:
            mvals = (10, 11, 12, 13, 14)
        else:
            mvals = (10, 11, 12, 13)
        for m in mvals:
            p = find_primitive_poly(m, rng, 96)
            if p is None:
                continue
            base = msequence_trace(m, p)
            add_sequence_variants(cands, base, "mseq_%d" % m, rng, max_variants=20)

    # Legendre / Paley sequences.
    if budget > 800.0:
        for p in (4099, 8191, 16381):
            if not is_small_prime(p):
                continue
            base = legendre_seq(p)
            add_sequence_variants(cands, base, "legendre_%d" % p, rng, max_variants=16)

def optimize_sequence(a0, end_time, rng,
                      lr_start=0.35,
                      topk_soft=20,
                      temp_frac=0.08,
                      minnorm=False,
                      minnorm_k=16,
                      minnorm_every=1,
                      patience=200,
                      tag=""):
    a = np.array(a0, dtype=np.float64, copy=True).ravel()
    n = a.size
    if n == 0:
        return float("inf"), a

    target_sum = float(n)
    a = normalize_inplace(a, target_sum)

    start = time.monotonic()
    end = float(end_time)
    if end <= start:
        c = fast_self_conv(a)
        return score_from_arrays(a, c), a

    c = fast_self_conv(a)
    best_score = score_from_arrays(a, c)
    best_a = a.copy()

    mstate = np.zeros(n, dtype=np.float64)
    vstate = np.zeros(n, dtype=np.float64)

    step = 0
    no_improve = 0
    restarts = 0
    max_restarts = 35
    total_est = max(1e-9, end - start)

    while time.monotonic() < end:
        now = time.monotonic()
        elapsed = max(0.0, now - start)
        progress = min(1.0, elapsed / total_est)

        a = normalize_inplace(a, target_sum)
        c = fast_self_conv(a)
        score = score_from_arrays(a, c)

        if np.isfinite(score) and score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= max(80, int(patience)):
            if restarts >= max_restarts:
                break
            a = perturb_vector(best_a, rng, 0.08 + 0.12 * float(rng.random()))
            a = normalize_inplace(a, target_sum)
            mstate.fill(0.0)
            vstate.fill(0.0)
            step = 0
            no_improve = 0
            restarts += 1
            continue

        use_mn = bool(minnorm and (step % max(1, int(minnorm_every)) == 0))

        if use_mn:
            mncur = int(int(minnorm_k) * (1.0 + 0.5 * progress))
            k_cap = 32 if n > 12000 else 48
            mncur = max(4, min(mncur, k_cap))
            idx, vals = topk_idx_vals(c, mncur)
            G = compute_minnorm_direction(a, idx, c, mncur, w_iters=30)
        else:
            ksoft = int(int(topk_soft) * (1.0 + 1.5 * progress))
            k_soft_cap = 64 if n > 12000 else 96
            ksoft = max(4, min(ksoft, k_soft_cap))
            idx, vals = topk_idx_vals(c, ksoft)

            if vals.size > 1:
                gap = float(vals[0] - vals[-1])
            else:
                gap = 0.0
            maxval = float(vals[0]) if vals.size else 1.0
            temp = max(1e-12, float(temp_frac) * maxval + 0.1 * gap)
            w = softmax_weights(vals, temp)
            G = accumulate_weighted_gradient(a, idx, w)

        if G.size == 0:
            break
        gnorm = float(np.sqrt(np.mean(G * G)))
        if (not np.isfinite(gnorm)) or gnorm < 1e-14:
            if restarts >= max_restarts:
                break
            a = perturb_vector(best_a, rng, 0.12)
            a = normalize_inplace(a, target_sum)
            mstate.fill(0.0)
            vstate.fill(0.0)
            step = 0
            no_improve = 0
            restarts += 1
            continue

        step += 1
        lr = float(lr_start) * max(0.05, 1.0 - 0.85 * progress)
        if restarts > 0:
            lr *= max(0.35, 1.0 - 0.1 * restarts)
        adam_update(a, G, mstate, vstate, step, lr)
        a = normalize_inplace(a, target_sum)

    a = normalize_inplace(a, target_sum)
    c = fast_self_conv(a)
    score = score_from_arrays(a, c)
    if np.isfinite(score) and score < best_score:
        best_score = score
        best_a = a.copy()

    return best_score, best_a

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = max(120.0, float(budget_s))
    deadline = start + max(60.0, total - 20.0)

    cands = []
    parent_raw = None

    try:
        raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if raw.size > 0:
            parent_raw = np.array(raw, dtype=np.float64, copy=True)
            add_candidate(cands, parent_raw, "parent")
    except Exception:
        parent_raw = None

    build_structured_candidates(cands, rng, total)

    if not cands:
        a = np.ones(2048, dtype=np.float64)
        add_candidate(cands, a, "default_uniform")

    if not cands:
        # Extremely defensive fallback.
        out = [1.0] * 256
        return out

    cands.sort(key=candidate_key)

    max_probe = 7 if total > 1200.0 else (5 if total > 600.0 else 3)
    selected = []
    selected_tags = []

    # Include one best candidate from several tag families for diversity.
    fams = ("parent", "mseq", "legendre", "chirp", "chirplinear", "cos", "rand", "smooth", "gauss")
    for fam in fams:
        if len(selected) >= max_probe:
            break
        for cand in cands:
            if fam in cand[2]:
                selected.append(cand)
                selected_tags.append(cand[2])
                break

    if parent_raw is not None:
        for cand in selected:
            if cand[2] == "parent":
                break
        else:
            for cand in cands:
                if cand[2] == "parent":
                    selected.append(cand)
                    selected_tags.append(cand[2])
                    break

    for cand in cands:
        if len(selected) >= max_probe:
            break
        if cand[2] not in selected_tags:
            selected.append(cand)
            selected_tags.append(cand[2])

    if not selected:
        selected = [cands[0]]

    probe_total = min(800.0, max(150.0, 0.30 * total))
    available = max(0.0, deadline - time.monotonic())
    if probe_total > available:
        probe_total = available

    if len(selected) > 1 and probe_total > 0.0:
        base_score = min(c[0] for c in selected)
        weights = []
        for cand in selected:
            gap = max(0.0, cand[0] - base_score)
            weights.append(1.0 / (1.0 + 20.0 * gap))
        sumw = float(sum(weights))
        if sumw <= 0.0 or not np.isfinite(sumw):
            weights = [1.0] * len(selected)
            sumw = float(len(selected))
        times = [probe_total * w / sumw for w in weights]
        for i in range(len(times)):
            if times[i] < 10.0:
                times[i] = 10.0
        sumt = float(sum(times))
        if sumt > probe_total and sumt > 0.0:
            times = [t * probe_total / sumt for t in times]
    else:
        times = [probe_total]

    best_score = float("inf")
    best_a = None
    probe_results = []

    for i, cand in enumerate(selected):
        if time.monotonic() >= deadline:
            break
        sc0, a0, tag = cand
        t_dur = max(5.0, float(times[i]))
        if time.monotonic() + t_dur > deadline:
            t_dur = max(0.0, deadline - time.monotonic())
        if t_dur < 5.0:
            break

        end_probe = time.monotonic() + t_dur
        sc, aa = optimize_sequence(
            a0,
            end_probe,
            rng,
            lr_start=0.40,
            topk_soft=18,
            temp_frac=0.08,
            minnorm=False,
            patience=120,
            tag=tag
        )
        if np.isfinite(sc):
            probe_results.append((float(sc), np.array(aa, dtype=np.float64, copy=True), str(tag)))
            if sc < best_score:
                best_score = sc
                best_a = aa
        print("probe", tag, sc, "best_so_far", best_score)

    if best_a is None:
        sc, a, tag = selected[0]
        best_a = np.array(a, dtype=np.float64, copy=True)
        best_score = compute_score(best_a)

    probe_results.sort(key=candidate_key)

    # Allow a close second branch to be chased briefly.
    remaining = max(0.0, deadline - time.monotonic())
    if len(probe_results) > 1 and remaining > 400.0:
        pr0 = probe_results[0]
        if abs(pr0[0] - best_score) < 1e-9:
            pr1 = probe_results[1]
            if pr1[0] <= best_score * 1.005 + 1e-9:
                t2 = min(0.20 * remaining, 360.0)
                if time.monotonic() + t2 < deadline:
                    sc, aa = optimize_sequence(
                        pr1[1],
                        time.monotonic() + t2,
                        rng,
                        lr_start=0.30,
                        topk_soft=24,
                        temp_frac=0.06,
                        minnorm=True,
                        minnorm_k=16,
                        minnorm_every=2,
                        patience=200,
                        tag=pr1[2] + "_branch2"
                    )
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_a = np.array(aa, dtype=np.float64, copy=True)

    # Main refinement with active-set min-norm directions.
    remaining = max(0.0, deadline - time.monotonic())
    if remaining > 30.0:
        phase1_end = time.monotonic() + 0.55 * remaining
        if phase1_end > deadline:
            phase1_end = deadline
        sc, aa = optimize_sequence(
            best_a,
            phase1_end,
            rng,
            lr_start=0.15,
            topk_soft=32,
            temp_frac=0.05,
            minnorm=True,
            minnorm_k=24,
            minnorm_every=2,
            patience=350,
            tag="main_refine"
        )
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = np.array(aa, dtype=np.float64, copy=True)
        print("main_refine", sc, "best_so_far", best_score)

    # Final polish with more active constraints.
    remaining = max(0.0, deadline - time.monotonic())
    if remaining > 20.0:
        sc, aa = optimize_sequence(
            best_a,
            deadline,
            rng,
            lr_start=0.035,
            topk_soft=72,
            temp_frac=0.03,
            minnorm=True,
            minnorm_k=48,
            minnorm_every=1,
            patience=250,
            tag="polish"
        )
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = np.array(aa, dtype=np.float64, copy=True)
        print("polish", sc, "best_so_far", best_score)

    if best_a is None:
        best_a = np.ones(2048, dtype=np.float64)

    best_a = np.asarray(best_a, dtype=np.float64).ravel().copy()
    best_a = normalize_inplace(best_a, float(best_a.size))
    out = [float(x) for x in best_a]

    verified = float("nan")
    if best_a.size <= 4096:
        try:
            verified = float(evaluate_sequence(out))
        except Exception:
            verified = float("nan")
    else:
        try:
            verified = score_from_arrays(best_a, fast_self_conv(best_a))
        except Exception:
            verified = float("nan")

    # If something unexpected happened, fall back to a known parent when possible.
    if (not np.isfinite(verified)) or verified > best_score * 1.2 + 1e-6:
        if parent_raw is not None and parent_raw.size <= 8192:
            p = np.asarray(parent_raw, dtype=np.float64).ravel().copy()
            p = normalize_inplace(p, float(p.size))
            pout = [float(x) for x in p]
            try:
                if p.size <= 4096:
                    pver = float(evaluate_sequence(pout))
                else:
                    pver = score_from_arrays(p, fast_self_conv(p))
            except Exception:
                pver = score_from_arrays(p, fast_self_conv(p))
            if np.isfinite(pver) and (not np.isfinite(verified) or pver < verified):
                out = pout
                verified = float(pver)

    print("final internal best", best_score, "verified", verified, "len", len(out))
    return out
```
