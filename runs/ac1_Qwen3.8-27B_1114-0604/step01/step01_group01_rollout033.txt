Actually, `|A|^2` has N values (one per frequency bin), but they're related to the n time-domain values. So there's a lot of freedom.

The key constraint is: `a >= 0`. In the frequency domain, this means `a = IDFT(A) >= 0`, which is a linear constraint on the real and imaginary parts of A.

So the problem is:
minimize max_j IDFT(|A|^2)[j]
subject to: IDFT(A)[0:n] >= 0, IDFT(A)[n:N] = 0, sum(IDFT(A)[0:n]) = 1

This is non-convex (the objective involves |A|^2 which is quadratic in the real and imaginary parts of A).

The Gerchberg-Saxton approach would be:
1. Start with some valid A (from a valid a >= 0)
2. Modify |A| to make IDFT(|A|^2) flatter
3. Keep the phase of A fixed
4. Take IDFT to get new a
5. Project a to be non-negative and zero outside [0, n)
6. Take DFT to get new A
7. Repeat

The challenge is step 2: how to modify |A| to flatten the autocorrelation?

One approach: compute R = IDFT(|A|^2), identify the large values of R, and try to reduce them. Since R = IDFT(|A|^2), reducing R[j] for large j means reducing the "contrast" of |A|^2.

A simple approach: `|A_new|^2 = |A|^2 - α * (R - mean(R)) * |A|^2 / max|R|`... this is getting complicated.

Actually, a simpler and more effective approach might be:

**Subgradient method in the time domain (which is what the previous program does):**

The subgradient of `max_j R[j]` with respect to `a` is `2 * circ(R_argmax) * a` where `circ(R_argmax)` is the circulant matrix formed by the indicator of the argmax positions.

The previous program uses a smoothed version: `h = convolve(weights, a)` where weights are the softmax of n*R.

1. Better step size selection (Armijo backtracking instead of fixed multipliers)
2. Nesterov momentum
3. Better handling of the non-negativity constraint (log-barrier instead of clipping)
4. More aggressive exploration of different starting points and lengths

using L-BFGS with softmax parameterization, frequency-domain flattening,

import numpy as np

from scipy.fft import rfft, irfft

def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full_a(a):
    """Full linear convolution of a with itself using FFT."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=float)
    nout = 2 * n - 1
    L = _next_pow2(nout)
    fa = rfft(a, L)
    out = irfft(fa * fa, L)
    return out[:nout].astype(float, copy=False)

def _sanitize(p):
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return p / s

def _exact_score(p):
    p = _sanitize(p)
    if p is None:
        return float('inf')
    n = p.size
    c = _conv_full_a(p)
    if c.size == 0:
        return float('inf')
    c = np.where(np.isfinite(c), c, 0.0)
    return float(2.0 * n * np.max(c))

def _smooth_max(x, beta):
    """Log-sum-exp approximation to max."""
    if x.size == 0:
        return 0.0
    shift = np.max(x)
    e = np.exp(beta * (x - shift))
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return shift
    return shift + np.log(s) / beta

def _score_smooth(z, n, beta):
    """Score as function of log-parameters z (softmax parameterization)."""
    z_shift = z - np.max(z)
    a = np.exp(z_shift)
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        return 1e10
    a = a / s
    c = _conv_full_a(a)
    d = n * c
    return float(2.0 * _smooth_max(d, beta))

    z_shift = z - np.max(z)
    a = np.exp(z_shift)

        return 1e10, np.zeros(n)
    a = a / s
    
    c = _conv_full_a(a)

    # a = softmax(z), so da_i/dz_j = a_i*(delta_ij - a_j)

def _lbsb_optimize(p0, rng, time_limit, n):
    """L-BFGS-B optimization with softmax parameterization."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    
    n = p.size
    z0 = np.log(p + 1e-30)
    z0 = z0 - z0.mean()  # center for numerical stability
    
    beta = 100.0
    
    def obj_grad(z):
        val, grad = _score_grad(z, n, beta)
        if not np.isfinite(val):
            return val, np.zeros(n)
        if not np.all(np.isfinite(grad)):
            grad = np.where(np.isfinite(grad), grad, 0.0)
        return val, grad
    
    best_z = z0.copy()
    best_score = _exact_score(p)
    
    deadline = time.monotonic() + time_limit
    current_beta = 50.0
    
    # Progressively increase beta
    while time.monotonic() < deadline:
        def current_obj_grad(z):
            val, grad = _score_grad(z, int(n), float(current_beta))
            if not np.isfinite(val):
                return val, np.zeros(int(n))
            if not np.all(np.isfinite(grad)):
                grad = np.where(np.isfinite(grad), grad, 0.0)
            return val, grad
        
        result = minimize(
            current_obj_grad, z0, method='L-BFGS-B',
            jac=True, options={'maxiter': 50, 'ftol': 1e-12, 'gtol': 1e-8}
        )
        
        z0 = result.x
        # Convert back to a
        z_shift = z0 - np.max(z0)
        a = np.exp(z_shift)
        a = a / a.sum()
        a = _sanitize(a)
        
        if a is not None:
            sc = _exact_score(a)
            if sc < best_score:
                best_score = sc
                best_z = z0.copy()
                print(f"L-BFGS n={n} beta={current_beta:.0f} score={best_score:.6f}")
        
        current_beta = min(3000.0, current_beta * 2.0)
        if time.monotonic() > deadline - 5:
            break
    
    # Final evaluation
    z_shift = best_z - np.max(best_z)
    a = np.exp(z_shift)
    a = a / a.sum()
    a = _sanitize(a)
    if a is not None:
        sc = _exact_score(a)
        if sc < best_score:
            best_score = sc
    
    return a, best_score

def _gs_flatten(p0, n_iter=50, alpha=0.3):
    """Gerchberg-Saxton style spectral flattening."""
    p = _sanitize(p0)
    if p is None:
        return None
    
    n = p.size
    best = p.copy()
    best_score = _exact_score(p)
    
    for it in range(n_iter):
        N = _next_pow2(2 * n)
        A = fft(p, N)  # complex FFT
        power = np.abs(A)**2
        
        # Flatten: reduce dynamic range of power spectrum
        power_mean = power.mean()
        power_std = power.std()
        if power_std < 1e-10:
            break
        
        # Compress: new_power = (power + alpha * power_mean) / (1 + alpha)
        # This pulls extreme values toward the mean
        new_power = (power + alpha * power_mean) / (1.0 + alpha)
        
        # Reconstruct with same phase
        new_mag = np.sqrt(new_power)
        phase = A / np.abs(A + 1e-30)
        A_new = new_mag * phase
        
        a_new = ifft(A_new).real[:n]
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-10:
            break
        a_new = a_new / s
        a_new = _sanitize(a_new)
        
        if a_new is not None:
            sc = _exact_score(a_new)
            if sc < best_score:
                best_score = sc
                best = a_new.copy()
        
        p = a_new
    
    return best, best_score

def _initial_sequences(n, rng):
    """Generate diverse initial sequences."""
    sequences = []
    t = (np.arange(n) + 0.5) / n
    
    # Two-block (asymmetric)

        p = np.ones(n)
        idx = int(frac * n)
        p[idx:] = 0.3
        sequences.append(_sanitize(p))
    
    # Gaussian bumps

        return np.array([], dtype=float)
    nout = 2 * n - 1
    L = _next_pow2(nout)
    fa = rfft(a, L)
    out = irfft(fa * fa, L)
    return out[:nout].astype(float, copy=False)


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return p / s


def _exact_score(p):
    p = _sanitize(p)
    if p is None:
        return float('inf')
    n = p.size
    c = _conv_full_a(p)
    if c.size == 0:
        return float('inf')
    c = np.where(np.isfinite(c), c, 0.0)
    return float(2.0 * n * np.max(c))


def _smooth_max_val(x, beta):
    if x.size == 0:
        return 0.0, np.array([])
    shift = float(np.max(x))
    e = np.exp(beta * (x - shift))
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        return shift, np.full(x.size, 1.0 / x.size)
    w = e / s
    val = shift + np.log(s) / beta
    return float(val), w


def _score_and_grad_z(z, n, beta, reg=1e-10):
    """Score (smoothed) and gradient w.r.t. z (softmax parameterization)."""
    z_shift = z - np.max(z)
    a = np.exp(z_shift)
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        return 1e10, np.zeros(n)
    a = a / s

    c = _conv_full_a(a)
    d = n * c

    smooth_val, w = _smooth_max_val(d, beta)
    if not np.isfinite(smooth_val):
        return 1e10, np.zeros(n)

    score = 2.0 * smooth_val

    L = _next_pow2(3 * n)
    fa = rfft(a, L)
    fw = rfft(w, L)
    u = irfft(np.conjugate(fa) * fw, L)[:n].astype(float)
    v = irfft(fa * np.conjugate(fw), L)[:n].astype(float)

    grad_a = 2.0 * n * (u + v)
    grad_a = np.where(np.isfinite(grad_a), grad_a, 0.0)

    dot = float(np.sum(a * grad_a))
    grad_z = a * (grad_a - dot)

    # Small regularization to break flat direction
    grad_z += 2.0 * reg * z

    if not np.all(np.isfinite(grad_z)):
        grad_z = np.where(np.isfinite(grad_z), grad_z, 0.0)

    return score, grad_z


def _mmd_descent(p0, rng, time_limit, beta_start=50.0, beta_end=2000.0):
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')

    n = p.size
    floor = 1e-18 / n
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_exact = _exact_score(p)
    cur_exact = best_exact

    c = _conv_full_a(p)
    d = n * c
    smooth_cur, _ = _smooth_max_val(d, beta_start)

    beta = float(beta_start)
    deadline = time.monotonic() + time_limit
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        c = _conv_full_a(p)
        d = n * c
        smooth_cur, w = _smooth_max_val(d, beta)

        if not np.isfinite(smooth_cur):
            p = best.copy()
            cur_exact = best_exact
            stall += 5
            if stall > 20:
                break
            continue

        L = _next_pow2(3 * n)
        fa = rfft(p, L)
        fw = rfft(w, L)
        u = irfft(np.conjugate(fa) * fw, L)[:n].astype(float)
        v = irfft(fa * np.conjugate(fw), L)[:n].astype(float)
        grad = 2.0 * n * (u + v)
        grad = np.maximum(grad, 0.0)
        grad = np.where(np.isfinite(grad), grad, 0.0)

        max_grad = float(np.max(grad))
        if max_grad < 1e-15:
            p = best.copy()
            cur_exact = best_exact
            stall += 10
            if stall > 30:
                break
        else:
            grad = grad / max_grad

            best_trial = None
            best_trial_smooth = smooth_cur
            best_trial_exact = cur_exact

            a_candidates = [0.1, 0.2, 0.5, 1.0, 2.0, 3.0, 5.0, 8.0]
            for a_mult in a_candidates:
                trial = p * np.exp(-a_mult * grad)
                trial = np.maximum(trial, floor)
                ts = trial.sum()
                if ts <= 0 or not np.isfinite(ts):
                    continue
                trial = trial / ts

                c_t = _conv_full_a(trial)
                d_t = n * c_t
                st, _ = _smooth_max_val(d_t, beta)
                if not np.isfinite(st):
                    continue
                if st < best_trial_smooth - 1e-12:
                    best_trial = trial
                    best_trial_smooth = st
                    best_trial_exact = _exact_score(trial)

            if best_trial is not None:
                p = best_trial
                cur_exact = best_trial_exact
                smooth_cur = best_trial_smooth
                stall = 0
                if cur_exact < best_exact:
                    best = p.copy()
                    best_exact = cur_exact
            else:
                stall += 1
                if stall > 10:
                    noise = np.exp(rng.normal(0, 0.06, n))
                    p_try = _sanitize(p * noise)
                    if p_try is not None:
                        p_try = np.maximum(p_try, floor)
                        p_try = p_try / p_try.sum()
                        e_try = _exact_score(p_try)
                        if e_try <= best_exact * 1.1:
                            p = p_try
                            cur_exact = e_try
                            c = _conv_full_a(p)
                            smooth_cur, _ = _smooth_max_val(n * c, beta)
                        else:
                            p = best.copy()
                            cur_exact = best_exact
                            c = _conv_full_a(p)
                            smooth_cur, _ = _smooth_max_val(n * c, beta)
                    else:
                        p = best.copy()
                        cur_exact = best_exact
                    stall = 0
                if stall > 50:
                    p = best.copy()
                    cur_exact = best_exact
                    beta = float(beta_start)
                    c = _conv_full_a(p)
                    smooth_cur, _ = _smooth_max_val(n * c, beta)
                    stall = 0

        it += 1
        if it % 25 == 0:
            beta = min(float(beta_end), beta * 1.12)

    return best, best_exact


def _lbsb_optimize(p0, rng, time_limit, n):
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')

    z0 = np.log(p + 1e-30)
    z0 = z0 - z0.mean()

    best_score = _exact_score(p)
    best_a = p.copy()
    current_beta = 30.0
    deadline = time.monotonic() + time_limit

    while time.monotonic() < deadline - 5:
        def obj_grad(z):
            val, grad = _score_and_grad_z(z, n, current_beta)
            if not np.isfinite(val):
                return val, np.zeros(n)
            return val, grad

        try:
            result = minimize(
                obj_grad, z0, method='L-BFGS-B',
                jac=True, options={'maxiter': 80, 'ftol': 1e-14, 'gtol': 1e-7,
                                   'maxls': 20}
            )
            z0 = result.x
        except Exception:
            pass

        z_shift = z0 - np.max(z0)
        a = np.exp(z_shift)
        s = a.sum()
        if s > 0 and np.isfinite(s):
            a = a / s
            a = _sanitize(a)
            if a is not None:
                sc = _exact_score(a)
                if sc < best_score:
                    best_score = sc
                    best_a = a.copy()

        current_beta = min(4000.0, current_beta * 2.5)

        if time.monotonic() > deadline - 10:
            break

    return best_a, best_score
