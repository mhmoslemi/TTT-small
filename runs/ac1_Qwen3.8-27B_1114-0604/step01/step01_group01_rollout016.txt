import numpy as np
from scipy.optimize import linprog
from scipy import sparse
import time

def convolve_a_g(a, g):
    """Compute a * g (full convolution) efficiently."""
    return np.convolve(a, g)

def solve_lp(a, T, tol=1e-8):
    """
    Solve: max sum(g) s.t. (a*g)_k <= T for all k, g >= 0.
    Returns optimal g or None if infeasible.
    """
    n = len(a)
    # Objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    
    # Constraints: A @ g <= T * ones(2n-1)
    # A[k, j] = a[k-j] if 0 <= k-j < n, else 0
    # This is a Toeplitz-like matrix
    
    # Build sparse matrix
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[idx])
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    b = T * np.ones(2*n - 1)
    
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    return None

The real lower bound comes from the fact that $\sum_k (a*a)_k = 1$ and there are $2n-1$ terms, but the max must be at least the average: $\max \geq 1/(2n-1)$. This gives $S \geq 2n/(2n-1) \approx 1$.

But we also know $(a*a)_0 = a_0^2$ and the max is at least this. For non-negative $a$ with $\sum a = 1$, $a_0^2$ can be very small (if $a_0$ is small).

The key constraint is: for non-negative $a$, the autocorrelation $(a*a)_k$ is always non-negative, and the "mass" of the autocorrelation is concentrated around $k = n-1$ (the middle). This is because:

$\sum_k (a*a)_k = \sum_k \sum_i a_i a_{k-i} = \sum_i a_i \sum_j a_j = 1$

And the "peak" at $k = n-1$ has value $\sum_i a_i a_{n-1-i}$, which for non-negative $a$ is the "overlap" of $a$ with its reversal.

By Cauchy-Schwarz: $\sum_i a_i a_{n-1-i} \leq \sqrt{\sum a_i^2} \cdot \sqrt{\sum a_{n-1-i}^2} = \sum a_i^2$

The average is $1/(2n-1)$, so the max is at least $1/(2n-1)$. For $n=3072$, that's about $1/6143 \approx 0.000163$, giving $S \geq 2 \cdot 3072 \cdot 0.000163 \approx 1.0003$.

"""
LP-based iterative descent for minimizing 2n*max(conv(a,a))/sum(a)^2.

Algorithm:
1. Start from good initial points (previous best, two-block, random smooth)
2. Iterate: solve LP max sum(g) s.t. (a*g)_k <= T, g>=0; update a toward g
3. Polish with local coordinate search
4. Return best sequence found

The LP at each step finds the "best" direction to move that keeps the
cross-correlation bounded by the current peak, giving a descent direction
for the peak of the autocorrelation.
"""

import numpy as np
from scipy.optimize import linprog
from scipy import sparse
import time

# Global cache for LP structures
_LP_CACHE = {}

def _lp_structure(n):
    """Build the sparsity pattern for the LP constraint matrix of size n."""
    if n in _LP_CACHE:
        return _LP_CACHE[n]
    m = 2 * n - 1
    rows_parts = []
    cols_parts = []
    for j in range(n):
        length = min(n, m - j)
        rows_parts.append(np.arange(j, j + length, dtype=np.int64))
        cols_parts.append(np.full(length, j, dtype=np.int64))
    rows = np.concatenate(rows_parts)
    cols = np.concatenate(cols_parts)
    _LP_CACHE[n] = (rows, cols)
    return rows, cols

def _score(a):
    """Compute 2n * max(a*a) assuming sum(a)=1."""
    n = len(a)
    if n == 0:
        return float('inf')
    c = np.convolve(a, a)
    return 2.0 * n * float(np.max(c))

def _normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.asarray(a, dtype=float)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones(len(a)) / len(a)
    return a / s

def _solve_lp(a):
    """
    Solve: maximize sum(g) subject to (a*g)_k <= T for all k, g >= 0.
    Returns normalized g or None.
    """
    n = len(a)
    rows, cols = _lp_structure(n)
    m = 2 * n - 1
    
    c_aa = np.convolve(a, a)
    T = float(np.max(c_aa))
    if T <= 1e-15:
        return None
    
    # Build constraint matrix: A[k,j] = a[k-j] if 0<=k-j<n else 0
    idx = rows - cols
    vals = a[idx]
    
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(m, n))
    b_ub = np.full(m, T, dtype=float)
    c_obj = -np.ones(n, dtype=float)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub,
                      bounds=[(0.0, None)] * n, method='highs')
    except Exception:
        return None
    
    if res.success and res.x is not None:
        g = np.maximum(res.x, 0.0)
        s = g.sum()
        if s > 1e-10:
            return g / s
    return None

def _lp_descent(a0, time_budget, rng, step_sizes=None, verbose=False):
    """
    LP-based descent: iterate (solve LP, update a).
    Returns (best_a, best_score).
    """
    if step_sizes is None:
        step_sizes = [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]
    
    a = _normalize(a0)
    n = len(a)
    deadline = time.monotonic() + time_budget
    
    best_a = a.copy()
    best_sc = _score(a)
    stagnant = 0
    it = 0
    
    while time.monotonic() < deadline - 2.0:
        t0 = time.monotonic()
        g = _solve_lp(a)
        if g is None:
            stagnant += 1
            if stagnant > 10:
                break
            a = _normalize(a * np.exp(rng.normal(0, 0.05, n)))
            continue
        
        # Line search
        best_new_sc = best_sc
        best_new_a = None
        best_t = None
        
        for t in step_sizes:
            a_new = (1.0 - t) * a + t * g
            a_new = _normalize(a_new)
            sc = _score(a_new)
            if sc < best_new_sc - 1e-12:
                best_new_sc = sc
                best_new_a = a_new
                best_t = t
        
        if best_new_a is not None:
            a = best_new_a
            if best_new_sc < best_sc:
                best_sc = best_new_sc
                best_a = a.copy()
                if verbose and it % 50 == 0:
                    print(f"  iter {it}: score = {best_sc:.6f} (t={best_t})")
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 15:
                # Perturb to escape
                std = min(0.3, 0.05 * stagnant)
                a = _normalize(a * np.exp(rng.normal(0, std, n)))
                stagnant = 0
        
        it += 1
        elapsed = time.monotonic() - t0
        if verbose and it % 100 == 0:
            print(f"  iter {it}: time/iter={elapsed:.3f}s, score={best_sc:.6f}")
    
    return best_a, best_sc

def _local_search(a0, time_budget, rng, verbose=False):
    """
    Local coordinate/block search: multiply random contiguous blocks by factors.
    """
    a = _normalize(a0)
    n = len(a)
    deadline = time.monotonic() + time_budget
    best_a = a.copy()
    best_sc = _score(a)
    it = 0

               0.9, 1.0/0.9, 0.95, 1.0/0.95]
    
    while time.monotonic() < deadline - 2.0:
        # Choose block size
        r = rng.random()
        if r < 0.3:
            w = 1
        elif r < 0.6:
            w = int(rng.integers(1, max(2, n // 100)))
        elif r < 0.9:
            w = int(rng.integers(max(2, n // 100), max(3, n // 20)))
        else:
            w = int(rng.integers(max(3, n // 20), min(n, n // 5)))
        
        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))
        
        best_f = None
        best_new_sc = best_sc
        best_new_a = None
        
        for f in factors:
            a_new = a.copy()
            a_new[start:start+w] *= f
            a_new = _normalize(a_new)
            sc = _score(a_new)
            if sc < best_new_sc - 1e-12:

                best_new_a = a_new

            a = best_new_a
            best_sc = best_new_sc
            best_a = a.copy()
        it += 1
    
    return best_a, best_sc

def _two_block(n, ratio=1.5, position=0.5):
    """Create a two-block sequence."""
    a = np.ones(n, dtype=float)
    split = int(position * n)
    if 0 < split < n:
        a[split:] = ratio
    return _normalize(a)

def _random_smooth(n, rng):
    """Create a random smooth positive sequence."""
    # Sum of random Gaussian bumps
    K = int(rng.integers(2, 6))
    a = np.zeros(n)
    for _ in range(K):
        c = rng.uniform(0, n)
        s = rng.uniform(n * 0.05, n * 0.4)
        w = rng.uniform(0.5, 2.0)
        x = np.arange(n)
        a += w * np.exp(-((x - c) / s) ** 2)
    return _normalize(a)
