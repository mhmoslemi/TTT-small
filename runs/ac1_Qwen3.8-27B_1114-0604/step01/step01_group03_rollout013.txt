    np.maximum(grad, 0.0, out=grad)  # gradient should be non-negative for non-negative a,w

def _topk_gradient_fft(c, a, n, K, T):
    """
    Compute gradient focusing on top-K autocorrelation lags.
    Uses FFT for efficiency.
    """
    m_total = len(c)  # 2n-1
    
    # Compute weights (softmax over top K)
    if K >= m_total:
        vmax = c.max()
        z = c / T - vmax / T
        z = np.clip(z, -500.0, 0.0)
        w = np.exp(z)
        ws = w.sum()
        if ws < 1e-300:
            w = np.ones(m_total) / m_total
        else:
            w /= ws
    else:
        idx = np.argpartition(c, -K)[-K:]
        c_top = c[idx]
        vmax = c_top.max()
        z = c_top / T - vmax / T
        z = np.clip(z, -500.0, 0.0)
        w_top = np.exp(z)
        ws = w_top.sum()
        if ws < 1e-300:
            w_top = np.ones(K) / K
        else:
            w_top /= ws
        w = np.zeros(m_total)
        w[idx] = w_top
    
    # Compute gradient using FFT
    N = 1 << (n + m_total - 1).bit_length()
    Aa = np.fft.rfft(a, N)
    Aw = np.fft.rfft(w, N)
    
    # g2[j] = (w * a)[j]
    conv_wa = np.fft.irfft(Aw * Aa, N)
    g2 = conv_wa[:n]
    
    # g1[j] = (w * flip(a))[n-1+j]

def _normalize(a):
    """Normalize to simplex (non-negative, sum=1)."""
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300:
        return np.ones_like(a) / len(a)
    return a / s


def _fft_self_conv(a, L, m):
    """Compute full self-convolution via FFT."""
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _score(c, n):
    """Compute objective value."""
    return float(2.0 * n * np.max(c))


def _compute_weights(c, K, T):
    """
    Compute gradient weights.
    Uses top-K softmax with a uniform background.
    """
    m = len(c)
    
    if K >= m:
        # Full softmax
        vmax = float(c.max())
        z = c / T - vmax / T
        z = np.clip(z, -700.0, 0.0)
        w = np.exp(z)
        ws = float(w.sum())
        if ws < 1e-300:
            return np.ones(m) / m
        return w / ws
    
    # Top-K with softmax + uniform background
    thresh = np.partition(c, m - K)[m - K]  # K-th largest value
    top_idx = np.where(c >= thresh)[0]
    # Handle ties: if more than K are equal to thresh, take first K
    if len(top_idx) > K:
        top_idx = np.where(c > thresh)[0]
        # If still not enough, add from equals
        if len(top_idx) < K:
            eq_idx = np.where(c == thresh)[0]
            need = K - len(top_idx)
            top_idx = np.concatenate([top_idx, eq_idx[:need]])
    
    K_actual = len(top_idx)
    
    c_top = c[top_idx]
    vmax = float(c_top.max())
    z = c_top / T - vmax / T
    z = np.clip(z, -700.0, 0.0)
    w_top = np.exp(z)
    ws_top = float(w_top.sum())
    
    if ws_top < 1e-300:
        w_top = np.ones(K_actual) / K_actual
    else:
        w_top = w_top / ws_top
    
    # Background: uniform over all
    w = np.full(m, 1.0 / m)
    # Blend: top-K get boosted weight
    alpha = 0.9  # 90% weight on top-K, 10% uniform background
    w[top_idx] = alpha * w_top + (1.0 - alpha) * (1.0 / m)
    # Renormalize
    w /= w.sum()
    return w


def _grad_from_weights(w, a, n, L_grad):
    """
    Compute gradient of weighted sum of autocorrelation values.
    grad[j] = sum_k w[k] * (a[k-j]*1[valid] + a[j-k]*1[valid])
    Using FFT:
      grad = (w * flip(a))[n-1 : 2n-1] + (w * a)[0 : n]
    """
    m = len(w)  # 2n-1
    
    # Need FFT size >= n + m - 1 = 3n - 2
    # L_grad should be a power of 2 >= 3n-2
    Aw = np.fft.rfft(w, L_grad)
    Aa = np.fft.rfft(a, L_grad)
    
    # (w * a)[j] for j = 0, ..., n-1
    conv_wa = np.fft.irfft(Aw * Aa, L_grad)
    g2 = conv_wa[:n]
    
    # (w * flip(a))[n-1+j] for j = 0, ..., n-1
    Afa = np.fft.rfft(a[::-1], L_grad)
    conv_wfa = np.fft.irfft(Aw * Afa, L_grad)
    g1 = conv_wfa[n-1:2*n-1]
    
    grad = g1 + g2
    np.maximum(grad, 0.0, out=grad)
    return grad
