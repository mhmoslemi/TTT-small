import time
import scipy.sparse as sp
import scipy.optimize

def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def score(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * max(c) / (S * S)

def normalize(a):
    S = a.sum()
    if S < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / S

def proj_simplex(v, s=1.0):
    """Project v onto {a: sum(a) = s, a >= 0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - s))[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def build_H(active, n):
    """Build H = (1/|A|) * sum_{k in A} Q_k as a sparse matrix."""
    row = []
    col = []
    val = []
    w = 1.0 / len(active)
    for k in active:
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            m = k - j
            if m == j:
                row.append(j)
                col.append(j)
                val.append(w)
            else:
                row.append(j)
                col.append(m)
                val.append(0.5 * w)
                row.append(m)
                col.append(j)
                val.append(0.5 * w)
    H = sp.csr_matrix((val, (row, col)), shape=(n, n))
    return H

def solve_qp_projected(H, n, max_iter=500, tol=1e-10):
    """Solve min a^T H a s.t. sum(a)=1, a>=0 using projected gradient."""
    a = np.ones(n) / n  # start at center of simplex
    
    # Estimate step size
    # Use power iteration to estimate largest eigenvalue
    v = np.ones(n) / np.sqrt(n)
    for _ in range(50):
        v_new = np.array(H @ v)
        norm = np.linalg.norm(v_new)
        if norm < 1e-15:
            break
        v = v_new / norm
    lambda_max = v @ np.array(H @ v)
    if lambda_max < 1e-15:
        lambda_max = 1.0
    step = 1.0 / (2.0 * lambda_max)
    
    for it in range(max_iter):
        grad = 2.0 * np.array(H @ a)
        a_new = proj_simplex(a - step * grad, 1.0)
        if np.linalg.norm(a_new - a) < tol:
            a = a_new
            break
        a = a_new
    
    return a

def qp_optimize(a, n, time_limit, epsilon=0.05, verbose=True):

    best_score = np.inf
    best_a = a.copy()
    start = time.monotonic()

            break
        
        c = fast_self_conv(a)
        M = c.max()
        S = a.sum()
        cur_score = 2.0 * n * M / (S * S)
        
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()

        active = np.where(c >= threshold)[0].tolist()

        a_new = solve_qp_projected(H, n)
        
        # Line search: find best t in (0, 1] such that a(t) = (1-t)*a + t*a_new
        best_t = 1.0
        best_s = cur_score
        for t in [0.5, 0.75, 1.0, 0.25, 1.0]:
            a_t = (1 - t) * a + t * a_new
            a_t = np.maximum(a_t, 0)
            a_t = normalize(a_t)
            c_t = fast_self_conv(a_t)
            s_t = 2.0 * n * c_t.max() / (a_t.sum()**2)
            if s_t < best_s:
                best_s = s_t
                best_t = t
        
        a = (1 - best_t) * a + best_t * a_new
        a = np.maximum(a, 0)
        a = normalize(a)
        
        if verbose and iteration % 100 == 0:
            print(f"  iter {it}, score = {cur_score:.6f}, active = {len(active)}")

def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def norm_simplex(v, s=1.0):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = np.arange(1, n + 1) * u > (cssv - s)
    if not np.any(cond):
        rho = n - 1
    else:
        rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - s) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_H_sparse(active, n):
    row, col, val = [], [], []
    w = 1.0 / len(active)
    for k in active:
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        for j in range(js, je + 1):
            m = k - j
            if m == j:
                row.append(j); col.append(j); val.append(w)
            else:
                row.append(j); col.append(m); val.append(0.5 * w)
                row.append(m); col.append(j); val.append(0.5 * w)
    H = sp.csr_matrix((val, (row, col)), shape=(n, n))
    return H


def solve_qp(H, n, step=None, max_iter=2000):
    """Solve min a^T H a s.t. sum(a)=1, a>=0 via projected gradient."""
    a = np.ones(n) / n
    if step is None:
        # estimate Lipschitz constant
        v = np.random.default_rng(0).random(n)
        v /= np.linalg.norm(v)
        for _ in range(100):

            nv = np.linalg.norm(Hv)
            if nv < 1e-15:
                break
            v = Hv / nv

        a_new = norm_simplex(a - step * grad, 1.0)
        if np.linalg.norm(a_new - a) < 1e-12:
            break
        a = a_new
    return a

    s = a.sum()
    if s < 1e-15:
        return np.full_like(a, target / len(a))
    return a * (target / s)


def proj_simplex(v, s=1.0):
    """Euclidean projection onto {x: sum(x)=s, x>=0}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - s)
    if not np.any(cond):
        rho_val = n
    else:
        rho_val = np.where(cond)[0][-1] + 1
    theta = (cssv[rho_val - 1] - s) / rho_val
    return np.maximum(v - theta, 0.0)


def build_H_sparse(active, n):
    """Build H = (1/|A|) * sum Q_k as sparse CSR."""
    row, col, val = [], [], []
    w = 1.0 / len(active)
    for k in active:
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        for j in range(js, je + 1):
            m = k - j
            if m == j:
                row.append(j)
                col.append(j)
                val.append(w)
            else:
                row.append(j)
                col.append(m)
                val.append(0.5 * w)
                row.append(m)
                col.append(j)
                val.append(0.5 * w)
    H = sp.csr_matrix((val, (row, col)), shape=(n, n))
    return H


def build_H_weighted(active, weights, n):
    """Build H = sum w_k * Q_k with given weights (should sum to 1)."""
    row, col, val = [], [], []
    for i, k in enumerate(active):
        wi = weights[i]
        js = max(0, k - n + 1)
        je = min(k, n - 1)
        for j in range(js, je + 1):
            m = k - j
            if m == j:
                row.append(j)
                col.append(j)
                val.append(wi)
            else:
                row.append(j)
                col.append(m)
                val.append(0.5 * wi)
                row.append(m)
                col.append(j)
                val.append(0.5 * wi)
    H = sp.csr_matrix((val, (row, col)), shape=(n, n))
    return H


def solve_qp_direct(H, n):
    """Solve min a^T H a s.t. sum(a)=1, a>=0.
    First try closed-form (interior), fall back to active set."""
    H_csc = H.tocsc()
    # Try interior solution: a = H^{-1} * 1 / (1^T H^{-1} * 1)
    ones = np.ones(n)
    # Use CG for speed (H is PSD)
    try:
        H_reg = H_csc + sp.eye(n) * 1e-12
        x, info = spla.cg(H_reg, ones, tol=1e-10, maxiter=500)
        s = x.sum()
        if s < 1e-15:
            raise ValueError("zero sum")
        a = x / s
        if np.all(a >= -1e-10):
            a = np.clip(a, 0, None)
            a = a / a.sum()
            return a
    except Exception:
        pass
    # Active set fallback: iteratively fix negative components
    a = np.ones(n) / n
    fixed = np.zeros(n, dtype=bool)
    for _ in range(n):
        free = ~fixed
        nf = free.sum()
        if nf == 0:
            break
        if nf == 1:
            a[free] = 1.0
            break
        idx = np.where(free)[0]
        H_sub = H[np.ix_(idx, idx)].tolil()
        ones_sub = np.ones(nf)
        H_sub_reg = H_sub + sp.eye(nf) * 1e-12
        try:
            x_sub, info = spla.cg(H_sub_reg.tocsr(), ones_sub, tol=1e-10, maxiter=200)
        except Exception:
            x_sub = np.ones(nf) / nf
        s_sub = x_sub.sum()
        if s_sub < 1e-15:
            x_sub = np.ones(nf) / nf
            s_sub = 1.0
        x_sub = x_sub / s_sub
        if np.all(x_sub >= -1e-10):
            a[free] = np.clip(x_sub, 0, None)
            a[free] = a[free] / a[free].sum() if a[free].sum() > 0 else 1.0 / nf
            break
        else:
            min_local = np.argmin(x_sub)
            fixed[idx[min_local]] = True
            a[idx[min_local]] = 0.0
    a = np.clip(a, 0, None)
    a = a / a.sum() if a.sum() > 1e-15 else np.ones(n) / n
    return a


def solve_qp_projected(H, n, max_iter=3000):
    """Solve min a^T H a s.t. sum(a)=1, a>=0 via projected gradient."""
    a = np.ones(n) / n
    # Estimate step size
    v = np.ones(n) / np.sqrt(n)
    lam_est = 0.0
    for _ in range(30):
        Hv = np.asarray(H @ v).ravel()
        nv = np.linalg.norm(Hv)
        if nv < 1e-15:
            break
        lam_est = float(v @ Hv)
        v = Hv / nv
    step = 1.0 / max(2.0 * lam_est, 1e-8)
    step = min(step, 1.0)
    
    prev_obj = np.inf
    for it in range(max_iter):
        grad = 2.0 * np.asarray(H @ a).ravel()
        a_new = proj_simplex(a - step * grad, 1.0)
        # Adaptive step: increase if objective decreased a lot
        obj = float(a_new @ np.asarray(H @ a_new))
        if obj < prev_obj * 0.99:
            step = min(step * 1.2, 10.0 / max(2.0 * lam_est, 1e-8))
        elif obj > prev_obj * 1.01:
            step *= 0.7
        prev_obj = obj
        if np.linalg.norm(a_new - a) < 1e-13:
            a = a_new
            break
        a = a_new
    return a


def compute_score_normalized(a, n):
    """Score = 2n * max(conv) when sum(a) = 1."""
    c = fast_self_conv(a)
    return 2.0 * n * float(c.max()), c


def qp_refine(a, n, duration, eps_rel=0.05, weight_power=2.0, verbose=True):
    """Main QP-based refinement loop."""
    a = np.maximum(a.astype(np.float64), 0)
    a = normalize_sum(a, 1.0)
    best = a.copy()
    best_sc, _ = compute_score_normalized(a, n)
    start = time.monotonic()
    no_improve = 0
    it = 0
    
    while time.monotonic() - start < duration:
        if it % 500 == 0 and verbose:
            print(f"    QP it={it}, score={best_sc:.6f}, no_improve={no_improve}")
        
        c = fast_self_conv(a)
        M = float(c.max())
        cur_sc = 2.0 * n * M
        
        if cur_sc < best_sc - 1e-12:
            best_sc = cur_sc
            best = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 300:
            # Restart with perturbation
            scale = 0.03 + 0.05 * np.random.random()
            a = best * (1.0 + scale * np.random.standard_normal(n))
            a = np.maximum(a, 0)
            a = normalize_sum(a, 1.0)
            no_improve = 0
            it += 1
            continue
        
        # Active set with relative threshold
        thresh = M * (1.0 - eps_rel)
        active_idx = np.where(c >= thresh)[0]
        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(c))])
        
        # Adaptive weights: power of (c[k] - min_active) + 1
        c_active = c[active_idx]
        c_min = c_active.min()
        raw_w = (c_active - c_min + 1e-10) ** weight_power + 1.0
        weights = raw_w / raw_w.sum()
        
        # Build H with weights
        H = build_H_weighted(active_idx.tolist(), weights, n)
        
        # Solve QP
        a_qp = solve_qp_direct(H, n)
        if not np.all(np.isfinite(a_qp)):
            a_qp = solve_qp_projected(H, n, max_iter=500)
        
        # Line search
        best_t, best_s = 0.0, cur_sc
        for t in (1.0, 0.7, 0.5, 0.3, 0.1, 1.0):
            a_t = (1.0 - t) * a + t * a_qp
            a_t = np.maximum(a_t, 0)
            a_t = normalize_sum(a_t, 1.0)
            c_t = fast_self_conv(a_t)
            s_t = 2.0 * n * float(c_t.max())
            if s_t < best_s:
                best_s = s_t
                best_t = t
        
        if best_t == 0.0:
            # No improvement possible with this active set; perturb
            a = best * (1.0 + 0.01 * np.random.standard_normal(n))
            a = np.maximum(a, 0)
            a = normalize_sum(a, 1.0)
        else:
            a = (1.0 - best_t) * a + best_t * a_qp
            a = np.maximum(a, 0)
            a = normalize_sum(a, 1.0)
        
        it += 1
    
    return best_sc, best


def logsumexp_smooth_grad(a, n, beta, topk=200):
    """Gradient of 2n * LSE(beta * c) / beta, where c = conv(a,a)."""
    c = fast_self_conv(a)
    if topk < len(c):
        idx = np.argpartition(c, -topk)[-topk:]
    else:
        idx = np.arange(len(c))
    c_top = c[idx]
    max_c = c_top.max()
    shifted = beta * (c_top - max_c)
    # Clamp for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_c = np.exp(shifted)
    w = exp_c / exp_c.sum()
    
    # grad[j] = 2n * sum_k w[k] * 2 * a[k-j] / beta ... 
    # Actually the gradient of (2n/beta)*LSE(beta*c) = 2n * sum w_k * grad(c_k)
    # = 2n * sum w_k * 2*a[k-j] = 4n * sum w_k * a[k-j]
    grad = np.zeros(n)
    for i in range(len(idx)):
        wi = float(w[i])
        if wi < 1e-14:
            continue
        ki = int(idx[i])
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        m_start = ki - je
        m_end = ki - js
        seg = a[m_start:m_end + 1][::-1]
        grad[js:je + 1] += wi * seg
    
    grad *= 4.0 * n
    return grad


def grad_polish(a, n, duration, beta_init=50.0, beta_end=2000.0, lr=0.05, verbose=True):
    """Adam-based gradient polish on log-sum-exp smoothed objective."""
    a = np.maximum(a.astype(np.float64), 0)
    a = normalize_sum(a, 1.0)
    best = a.copy()
    best_sc, _ = compute_score_normalized(a, n)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    start = time.monotonic()
    total_iters_est = duration * 50  # rough estimate
    total_hours = max(duration / 3600.0, 0.01)
    
    while time.monotonic() - start < duration:
        c = fast_self_conv(a)
        cur_sc = 2.0 * n * float(c.max())
        if cur_sc < best_sc:
            best_sc = cur_sc
            best = a.copy()
        
        # Animate beta
        frac = min((time.monotonic() - start) / duration, 1.0)
        beta = beta_init * (beta_end / beta_init) ** frac
        
        g = logsumexp_smooth_grad(a, n, beta, topk=200)
        t += 1
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        
        # Adaptive LR
        lr_t = lr * (0.9995 ** t)
        a = a - lr_t * mh / (np.sqrt(vh) + eps)
        a = np.maximum(a, 0)
        a = normalize_sum(a, 1.0)
        
        if verbose and t % 1000 == 0:
            print(f"    Grad it={t}, score={cur_sc:.6f}, beta={beta:.1f}, lr={lr_t:.6f}")
    
    return best_sc, best


def build_inits(n, rng):
    inits = []
    x = np.linspace(-1, 1, n)
    
    # Parent
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size > 0:
            if p.size == n:
                inits.insert(0, ("parent", np.maximum(p, 0)))
            else:
                idx = np.linspace(0, p.size - 1, n).astype(int)
                inits.insert(0, ("parent_rs", np.maximum(p[idx], 0)))
    except Exception:
        pass
    
    inits.append(("uniform", np.ones(n)))
    
    # Inverted polynomials (high at edges)
    for p in (2, 3, 4, 6, 8):
        for alpha in (0.5, 1.0, 2.0, 3.0):
            a = 1.0 + alpha * np.abs(x) ** p
            inits.append((f"invp{p}_a{alpha}", a))
    
    # Decreasing from center (U-shape)
    for sigma in (0.3, 0.5, 0.7, 1.0):
        a = 1.0 / (1.0 + sigma * (x ** 2))
        inits.append((f"sec_{sigma}", a))
    
    # Cosine variants
    for freq in (1, 2, 3):
        for alpha in (0.5, 0.8):
            a = 1.0 + alpha * np.cos(np.pi * freq * x)
            inits.append((f"cos{freq}_{alpha}", np.maximum(a, 0.01)))
    
    # Edge-peaked
    for w in (0.1, 0.2, 0.3):
        a = np.ones(n)
        edge = int(w * n)
        a[:edge] = 2.0
        a[-edge:] = 2.0
        inits.append((f"edge{w}", a))
    
    # Random
    inits.append(("rand", np.maximum(0.2 + 0.8 * rng.random(n), 0.01)))
    
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 10.0
    total = budget_s - 10.0
    
    # Choose n
    n = 2048
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size >= 512:
            n = min(int(p.size), 8192)
    except Exception:
        pass
    n = max(512, min(n, 8192))
    print(f"=== n={n}, budget={budget_s}s ===")
    
    # Phase 1: Score initializations (30s)
    t1_end = start + min(30.0, 0.1 * total)
    inits = build_inits(n, rng)
    best_sc = np.inf
    best_a = None
    scored = []
    
    for tag, a0 in inits:
        if time.monotonic() > t1_end:
            break
        a0 = np.maximum(a0.astype(np.float64), 0)
        if a0.sum() < 1e-15:
            continue
        a0 = normalize_sum(a0, 1.0)
        sc, c = compute_score_normalized(a0, n)
        scored.append((sc, tag))
        if sc < best_sc:
            best_sc = sc
            best_a = a0.copy()
    
    scored.sort()
    print(f"Phase 1: best={best_sc:.6f}, top5={[f'{s:.4f}({t})' for s,t in scored[:5]]}")
    
    # Phase 2: QP refine from best (50% of remaining time)
    t2_end = start + 0.55 * total
    t2_time = max(t2_end - time.monotonic(), 10.0)
    if best_a is not None:
        print(f"Phase 2: QP refine, {t2_time:.0f}s")
        sc, a = qp_refine(best_a, n, t2_time, eps_rel=0.05, weight_power=2.0)
        if sc < best_sc:
            best_sc, best_a = sc, a
        print(f"  After QP: {best_sc:.6f}")
    
    # Phase 2b: QP from second-best init (15%)
    t2b_end = start + 0.75 * total
    t2b_time = max(t2b_end - time.monotonic(), 5.0)
    if len(scored) > 1 and t2b_time > 5:
        # Find second best that's different from current best
        for sc2, tag2 in scored[1:]:
            if tag2 != scored[0][1]:
                a2 = None
                for t3, a3 in inits:
                    if t3 == tag2:
                        a2 = np.maximum(a3.astype(np.float64), 0)
                        if a2.sum() > 0:
                            a2 = normalize_sum(a2, 1.0)
                        break
                if a2 is not None:
                    sc, a = qp_refine(a2, n, t2b_time, eps_rel=0.03, weight_power=3.0)
                    if sc < best_sc:
                        best_sc, best_a = sc, a
                        print(f"  Phase 2b improved: {best_sc:.6f}")
                break
    
    # Phase 3: QP with tighter epsilon on best (10%)
    t3_end = start + 0.88 * total
    t3_time = max(t3_end - time.monotonic(), 5.0)
    if best_a is not None and t3_time > 5:
        print(f"Phase 3: QP tight, {t3_time:.0f}s")
        sc, a = qp_refine(best_a, n, t3_time, eps_rel=0.01, weight_power=4.0)
        if sc < best_sc:
            best_sc, best_a = sc, a
        print(f"  After tight QP: {best_sc:.6f}")
    
    # Phase 4: Gradient polish (remaining)
    t4_time = max(deadline - time.monotonic(), 3.0)
    if best_a is not None and t4_time > 3:
        print(f"Phase 4: Grad polish, {t4_time:.0f}s")
        sc, a = grad_polish(best_a, n, t4_time, beta_init=50.0, beta_end=2000.0, lr=0.03)
        if sc < best_sc:
            best_sc, best_a = sc, a
        print(f"  After grad: {best_sc:.6f}")
    
    # Final verification
    if best_a is None:
        best_a = np.ones(n) / n
    
    out = [float(x) for x in best_a]
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - start
    print(f"=== FINAL: internal={best_sc:.6f}, verified={verified:.6f}, n={n}, time={elapsed:.1f}s ===")
    
    return out
