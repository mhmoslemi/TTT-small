```python
"""
Dense log-sum-exp gradient search for non-negative coefficients minimizing
2*n*max(conv(a,a)) / (sum a)^2.

Uses FFT convolutions, Adam-style adaptive steps on the simplex tangent,
annealed smooth-max weights, multiplicative restarts, a fixed-a linear-fit
operator, multi-resolution upsampling, and many structured/random starts.
"""
import time
import numpy as np
from operator import itemgetter


def p2(x):
    x = int(x)
    if x <= 1:
        return 1
    return 1 << (x - 1).bit_length()


def rescale_to_sum(a, target):
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    n = a.size
    if n == 0:
        return a
    a = np.clip(a, 0.0, 1000.0)
    a += 1e-12
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-12:
        return np.full(n, max(1.0, target) / n, dtype=np.float64)
    a *= target / s
    a = np.clip(a, 0.0, 1000.0)
    s2 = float(a.sum())
    if s2 > 1e-12 and abs(s2 - target) > 1e-6 * max(1.0, target):
        a *= target / s2
        a = np.clip(a, 0.0, 1000.0)
    return a


def upsample(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    m = a.size
    if m == 0:
        return np.zeros(n, dtype=np.float64)
    if m == 1:
        return np.full(n, a[0], dtype=np.float64)
    if m == n:
        return a
    x0 = (np.arange(m) + 0.5) / float(m)
    x1 = (np.arange(n) + 0.5) / float(n)
    v = np.interp(x1, x0, a, left=a[0], right=a[-1])
    return np.clip(v, 0.0, 1000.0)


def self_conv(a):
    n = a.size
    m = 2 * n - 1
    N = p2(m)
    A = np.fft.rfft(a, N)
    return np.fft.irfft(A * A, N)[:m]


def score_sequence(a):
    n = a.size
    if n <= 0:
        return np.inf
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = self_conv(a)
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0:
        return np.inf
    return float(2.0 * n * mx / (S * S))


def smooth_weights(c, beta):
    cmax = float(np.max(c))
    if cmax <= 0.0:
        return np.full(c.size, 1.0 / c.size, dtype=np.float64), cmax
    T = max(float(beta) * cmax, 1e-30)
    z = (c - cmax) / T
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        p = np.zeros(c.size, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
        return p, cmax
    return e / s, cmax


def adam_step(a, g, m, v, t, lr):
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8
    t1 = b1 ** t
    t2 = b2 ** t
    m[:] = b1 * m + (1.0 - b1) * g
    v[:] = b2 * v + (1.0 - b2) * g * g
    mh = m / (1.0 - t1)
    vh = v / (1.0 - t2)
    a -= lr * mh / (np.sqrt(vh) + eps)


def linear_fit(a, steps, beta, stop_time):
    """
    Approximate argmin_b max_k (a*b)_k by mirror-descent on the simplex.
    Used as a restart operator to escape local minima of self-convolution.
    """
    n = a.size
    target = float(n)
    A = np.fft.rfft(a, p2(2 * n - 1))
    N = p2(2 * n - 1)
    mlen = 2 * n - 1
    b = rescale_to_sum(a, target)
    for _ in range(steps):
        if stop_time is not None and time.monotonic() >= stop_time:
            break
        B = np.fft.rfft(b, N)
        q = np.fft.irfft(A * B, N)[:mlen]
        p, _ = smooth_weights(q, beta)
        P = np.fft.rfft(p, N)
        h = np.fft.irfft(A * P, N)[:n].real
        g = h - h.mean()
        scale = float(np.max(np.abs(g)))
        if scale <= 1e-14:
            break
        b = b * np.exp(-0.5 * g / scale)
        b = rescale_to_sum(b, target)
    return b


def optimize_smooth(a0, until, rng, lr0, beta0, patience):
    n = a0.size
    target = float(n)
    a = rescale_to_sum(a0, target)
    N = p2(2 * n - 1)
    mlen = 2 * n - 1
    mom = np.zeros(n, dtype=np.float64)
    var = np.zeros(n, dtype=np.float64)
    t = 0
    lr = lr0
    beta = beta0
    no_improve = 0
    pat = patience

    best_s = score_sequence(a)
    best_a = a.copy()
    if not np.isfinite(best_s):
        best_a = np.ones(n, dtype=np.float64)
        best_s = score_sequence(best_a)

    while time.monotonic() < until:
        if no_improve > pat:
            if rng.random() < 0.35:
                a = linear_fit(best_a, 8, min(0.8, beta * 2.0 + 0.1), until)
            else:
                scale = 0.05 + 0.30 * rng.random()
                a = best_a * np.exp(scale * rng.standard_normal(n))
                a = a + 1e-7
            a = rescale_to_sum(a, target)
            mom.fill(0.0)
            var.fill(0.0)
            t = 0
            lr = max(0.01, lr * 0.55)
            beta = min(1.8, beta * 2.5 + 0.05)
            no_improve = 0
            pat = max(80, int(pat * 0.7))
            continue

        A = np.fft.rfft(a, N)
        c = np.fft.irfft(A * A, N)[:mlen]
        S = float(a.sum())
        cmax = float(np.max(c))
        if S <= 1e-12 or not np.isfinite(cmax):
            break
        sc = 2.0 * n * cmax / (S * S)
        if np.isfinite(sc) and sc < best_s - 1e-12:
            best_s = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        beta = max(0.03, beta * 0.9995)
        p, _ = smooth_weights(c, beta)
        P = np.fft.rfft(p, N)
        h = np.fft.irfft(A * P, N)[:n].real
        g = h - h.mean()
        t += 1
        adam_step(a, g, mom, var, t, lr)
        a = rescale_to_sum(a, target)

    return best_s, best_a


def make_shapes(n, rng, parent):
    sh = []
    x = (np.arange(n) + 0.5) / float(n)
    y = 1.0 - x
    th = 2.0 * np.pi * x
    c1 = np.cos(th)
    s1 = np.sin(th)
    s2 = np.sin(2.0 * th)
    c2 = np.cos(2.0 * th)

    if parent is not None and parent.size > 0:
        pp = parent if parent.size == n else upsample(parent, n)
        sh.append(('par', pp))
        pr = parent[::-1]
        pr = pr if pr.size == n else upsample(pr, n)
        sh.append(('prr', pr))
        pb = 0.5 * (parent + parent[::-1])
        pb = pb if pb.size == n else upsample(pb, n)
        sh.append(('par', pb))

    sh.append(('uni', np.ones(n, dtype=np.float64)))

    beta_pairs = (
        (0.5, 0.2), (0.2, 0.5), (0.8, 0.2), (0.2, 0.8),
        (0.1, 0.4), (0.4, 0.1), (1.0, 0.2), (0.2, 1.0),
        (0.5, 0.5), (0.3, 0.6), (0.6, 0.3), (0.1, 0.1),
    )
    for p, q in beta_pairs:
        a = np.power(x, p) * np.power(y, q)
        if a.sum() > 0.0:
            sh.append(('bet', a))

    for g in (0.2, 0.6, 0.9, 1.1):
        W = 1.0 + g * c1
        for b1 in (-1.0, -0.5, 0.5, 1.0):
            for b3 in (-0.4, 0.0, 0.4):
                a = W * (1.0 + b1 * s1 + b3 * s2)
                a = np.clip(a, 0.0, 1000.0)
                if a.sum() > 0.05 * n:
                    sh.append(('prd', a))

    for a1 in (-0.8, -0.4, 0.4, 0.8):
        for a2 in (-0.6, 0.0, 0.6):
            for a3 in (-0.4, 0.4):
                a = 1.0 + a1 * s1 + a2 * s2 + a3 * c2
                a = np.clip(a, 0.0, 1000.0)
                if a.sum() > 0.05 * n:
                    sh.append(('add', a))

    for r in range(12):
        a = np.ones(n, dtype=np.float64)
        for k in range(1, 7):
            amp = rng.uniform(0.08, 0.55) / (1.0 + 0.35 * k)
            ph = rng.uniform(0.0, 2.0 * np.pi)
            a += amp * np.sin(k * th + ph) + 0.5 * amp * np.cos(k * th + ph)
        a = np.clip(a, 0.0, 1000.0)
        sh.append(('rnd', a))

    a = np.ones(n, dtype=np.float64)
    mask = (x > 0.18) & (x < 0.30)
    a[mask] = 0.25
    mask = (x > 0.62) & (x < 0.80)
    a[mask] = 0.25
    sh.append(('blk', a))

    a = np.abs(s1)
    sh.append(('abs', a))

    a = (1.0 + 0.9 * s1) * (1.0 + 0.45 * s2)
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() > 0.05 * n:
        sh.append(('mix', a))

    return sh


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = max(1.0, float(budget_s))
    reserve = 12.0
    time_viewer = max(1.0, total - reserve)
    deadline = start + total - max(3.0, reserve * 0.3)

    parent_arr = None
    try:
        if 'height_sequence_1' in globals():
            cand = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if cand.size > 0:
                cand = np.where(np.isfinite(cand), cand, 0.0)
                parent_arr = rescale_to_sum(np.clip(cand, 0.0, 1000.0), float(cand.size))
    except Exception:
        parent_arr = None

    best_s = np.inf
    best_arr = None
    if parent_arr is not None:
        s = score_sequence(parent_arr)
        if np.isfinite(s):
            best_s = s
            best_arr = parent_arr.copy()
            print('initial parent score=%.6f len=%d' % (best_s, parent_arr.size))

    lengths = [2048, 4096, 8192]
    if parent_arr is not None and parent_arr.size >= 8192:
        lengths = [8192]
    elif parent_arr is not None and parent_arr.size >= 4096:
        lengths = [4096, 8192]
    elif parent_arr is not None and parent_arr.size >= 2048:
        lengths = [2048, 4096, 8192]

    if len(lengths) == 3:
        fr = (0.38, 0.30)
    elif len(lengths) == 2:
        fr = (0.55,)
    else:
        fr = ()

    prev_top = []

    for idx, n in enumerate(lengths):
        if idx == len(lengths) - 1:
            stage_end = deadline
        else:
            stage_end = start + fr[idx] * time_viewer
        if time.monotonic() >= stage_end:
            break

        if n <= 2048:
            topk, lr0, beta0, pat0 = 8, 0.10, 1.00, 300
        elif n <= 4096:
            topk, lr0, beta0, pat0 = 5, 0.07, 0.55, 450
        else:
            topk, lr0, beta0, pat0 = 4, 0.05, 0.35, 650

        shapes = make_shapes(n, rng, parent_arr)
        if best_arr is not None:
            shapes.append(('gb', upsample(best_arr, n)))
        for nm, sc, ar in prev_top:
            if ar.size == n:
                shapes.append(('tp', ar))
            else:
                shapes.append(('tp', upsample(ar, n)))

        scored = []
        for nm, ar in shapes:
            if time.monotonic() >= stage_end:
                break
            ar = rescale_to_sum(ar, float(n))
            sc = score_sequence(ar)
            if np.isfinite(sc):
                scored.append((sc, nm, ar))

        if not scored:
            uni = np.ones(n, dtype=np.float64)
            scored = [(score_sequence(uni), 'fallback', uni)]

        scored.sort(key=itemgetter(0))
        res_top = []

        for j in range(min(topk, len(scored))):
            now = time.monotonic()
            if now >= stage_end:
                break
            remain = stage_end - now
            left = min(topk, len(scored)) - j
            te = remain / max(1, left)
            sc0, nm, ar0 = scored[j]
            opt_end = min(stage_end, now + max(3.0, te))
            sc, ar2 = optimize_smooth(ar0, opt_end, rng, lr0, beta0, pat0)
            if np.isfinite(sc):
                res_top.append((sc, nm, ar2))
                if sc < best_s:
                    best_s = sc
                    best_arr = ar2.copy()
                    print('stage n=%d %s -> %.6f' % (n, nm, best_s))

        res_top.sort(key=itemgetter(0))
        prev_top = res_top[:3]
        print('stage done n=%d best=%.6f len=%d' % (n, best_s, best_arr.size))

    if best_arr is not None and time.monotonic() < deadline - 2.0:
        n = best_arr.size
        lr0 = 0.025 if n > 4096 else 0.045
        beta0 = 0.22 if n > 4096 else 0.35
        sc, ar = optimize_smooth(best_arr, deadline, rng, lr0, beta0, 800)
        if np.isfinite(sc) and sc < best_s:
            best_s = sc
            best_arr = ar.copy()
        print('final polish best=%.6f len=%d' % (best_s, best_arr.size))

    if best_arr is None or best_arr.size == 0:
        best_arr = np.ones(2048, dtype=np.float64)

    out = [float(x) for x in best_arr]

    if time.monotonic() < start + total - 2.0:
        try:
            ve = float(evaluate_sequence(out))
            print('internal=%.6f verified=%.6f len=%d time=%.1f' %
                  (best_s, ve, len(out), time.monotonic() - start))
            if not np.isfinite(ve):
                if parent_arr is not None and parent_arr.size > 0:
                    out = [float(x) for x in parent_arr]
                else:
                    out = [1.0] * 2048
            else:
                best_s = ve
        except Exception as e:
            print('verification failed:', e)
    else:
        print('skipped verification due to time, internal=%.6f len=%d' % (best_s, len(out)))

    return out
```
