So the gradient is well-defined. The question is whether I can compute it efficiently.

For the FFT approach:
- p has length 2n-1
- a has length n
- We need: for each j, sum_k p[k] * dc[k]/da[j]
  = sum_k p[k] * (a[k-j] * [0<=k-j<n] + a[j] * [0<=k-j<n and k!=2j])

The first part: sum_{k} p[k] * a[k-j] * [0<=k-j<n] = sum_{m=0}^{n-1} p[j+m] * a[m]

This is a cross-correlation that can be computed via FFT in O(n log n).

The second part: a[j] * sum_{k: 0<=k-j<n, k!=2j} p[k] = a[j] * (sum_{m=0}^{n-1} p[j+m] - p[2j] * [2j < 2n-1])

This is O(n) with prefix sums.

Total: O(n log n) per gradient evaluation. Good.

```python
"""
Optimizes non-negative sequence to minimize 2n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp (LSE) surrogate with geometric temperature annealing,
FFT-accelerated gradients, Adam optimizer with adaptive learning rate,
structured restarts, and coordinate descent polishing.
Explores multiple sequence lengths and initialization strategies.
"""

import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def lse_value(c, T):
    """Compute LSE_T(c) = T * log(sum(exp(c/T))) with numerical stability."""
    c_max = float(np.max(c))
    if c_max <= 0:
        return 0.0
    logits = (c - c_max) / T
    logits = np.clip(logits, -60.0, 0.0)
    exp_logits = np.exp(logits)
    log_sum = c_max / T + np.log(float(np.sum(exp_logits)))
    return T * log_sum


def lse_weights(c, T):
    """Compute softmax weights p[k] = exp(c[k]/T) / sum(exp(c[j]/T))."""
    c_max = float(np.max(c))
    logits = (c - c_max) / T
    logits = np.clip(logits, -60.0, 0.0)
    exp_logits = np.exp(logits)
    return exp_logits / float(np.sum(exp_logits))


def grad_lse_fft(a, c, T):
    """
    Compute gradient of 2n * LSE_T(c) / S^2 w.r.t. a.
    Uses FFT for the cross-correlation part.
    Returns (grad, LSE_value).
    """
    n = len(a)
    S = float(a.sum())
    M = 2 * n - 1

    p = lse_weights(c, T)
    L = lse_value(c, T)

    # Part 1: grad_base[j] = sum_{m=0}^{n-1} a[m] * p[j+m]
    # This equals (a_rev * p)[n-1+j] where a_rev = a[::-1]
    a_rev = a[::-1].copy()
    M_fft = 1 << (M - 1).bit_length()
    A_rev = np.fft.rfft(a_rev, M_fft)
    P = np.fft.rfft(p, M_fft)
    cross = np.fft.irfft(A_rev * P, M_fft)[:M]
    grad_base = cross[n - 1:n - 1 + n].copy()

    # Part 2: a[j] * (sum_{m=0}^{n-1} p[j+m] - p[2j] * [2j < M])
    p_cum = np.zeros(M + 1)
    p_cum[1:] = np.cumsum(p)

    sum_p = np.zeros(n)
    for j in range(n):
        hi = j + n - 1
        s = p_cum[hi + 1] - p_cum[j]
        if 2 * j < M:
            s -= p[2 * j]
        sum_p[j] = s

    grad_L = grad_base + a * sum_p
    grad = (2.0 * n / (S * S)) * grad_L - (4.0 * n * L / (S ** 3))
    return grad, L


def topk_grad(a, c, k):
    """
    Gradient using top-k peaks with softmax weights.
    Returns (grad, weighted_max, S).
    """
    n = len(a)
    S = float(a.sum())
    k_act = min(k, len(c))
    if k_act <= 0:
        return np.zeros(n), 0.0, S

    idx = np.argpartition(c, -k_act)[-k_act:]
    vals = c[idx]
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]

    top = float(vals[0])
    if top > 1e-12:
        temp = 0.08 * top
        wts = np.exp((vals - top) / temp)
        wts = wts / float(wts.sum())
    else:
        wts = np.ones(k_act) / k_act

    grad_c = np.zeros(n)
    for i in range(idx.size):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-14:
            continue
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start > j_end:
            continue
        seg = a[j_start:j_end + 1][::-1]
        grad_c[j_start:j_end + 1] += wi * 2.0 * seg

    M_w = float(np.dot(vals, wts))
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_c / S2 - 2.0 * M_w / S3)
    return grad, M_w, S


def adam_step(a, g, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    """In-place Adam update."""
    bc1 = b1 ** t
    bc2 = b2 ** t
    m[:] = b1 * m + (1.0 - b1) * g
    v[:] = b2 * v + (1.0 - b2) * g * g
    mh = m / (1.0 - bc1)
    vh = v / (1.0 - bc2)
    a -= lr * mh / (np.sqrt(vh) + eps)


def clamp_nonneg(a, lo=0.0, hi=1000.0):
    """Clamp to valid range ensuring positive sum."""
    a = np.clip(a, lo, hi)
    if a.sum() < 1e-3:
        a = a + 1e-3 / a.size
    return a


def score(a):
    """Compute the objective value."""
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-10:
        return np.inf
    return float(2.0 * n * np.max(c) / (S * S))


def init_parent(n, rng):
    """Try to use parent sequence, resized if needed."""
    try:
        if "height_sequence_1" in globals() and height_sequence_1 is not None:
            arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if arr.size > 0:
                if arr.size == n:
                    return np.clip(arr, 0.0, 1000.0).copy()
                else:
                    from numpy import interp
                    new_x = np.linspace(0, 1, n)
                    old_x = np.linspace(0, 1, arr.size)
                    resampled = interp(new_x, old_x, arr)
                    return np.clip(resampled, 0.0, 1000.0).copy()
    except Exception:
        pass
    return np.ones(n)


def init_smooth_bump(n, rng):
    """Raised cosine window with random phase."""
    x = np.linspace(0.0, 1.0, n)
    phase = rng.uniform(-1.0, 1.0)
    a = 0.3 + 0.7 * np.cos(np.pi * (x - 0.5 - phase * 0.3)) ** 2
    return np.clip(a, 0.01, 1000.0)


def init_tapered(n, rng):
    """Tapered edges, flat middle."""
    x = np.linspace(0.0, 1.0, n)
    edge = 0.15 + rng.uniform(0.0, 0.1)
    a = np.ones(n)
    mask_left = x < edge
    mask_right = x > 1.0 - edge
    a[mask_left] = np.cos(np.pi * x[mask_left] / (2.0 * edge)) ** 2
    a[mask_right] = np.cos(np.pi * (x[mask_right] - 1.0 + edge) / (2.0 * edge)) ** 2
    a = 0.3 + 0.7 * a
    return np.clip(a, 0.01, 1000.0)


def init_random_lowfreq(n, rng):
    """Sum of a few random low-frequency sinusoids."""
    x = np.linspace(0.0, 1.0, n)
    a = np.ones(n)
    for _ in range(rng.integers(2, 5)):
        freq = rng.integers(1, 4)
        amp = rng.uniform(0.1, 0.4)
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.sin(2 * np.pi * freq * x + phase)
    return np.clip(a, 0.05, 1000.0)


def structured_perturb(a, rng, scale=0.1):
    """Apply a structured perturbation to escape local minima."""
    n = len(a)
    strategy = rng.integers(0, 5)
    if strategy == 0:
        # Add low-freq sinusoid
        x = np.linspace(0, 1, n)
        freq = rng.integers(1, 4)
        phase = rng.uniform(0, 2 * np.pi)
        a = a + scale * np.sin(2 * np.pi * freq * x + phase)
    elif strategy == 1:
        # Multiply by smooth window
        x = np.linspace(0, 1, n)
        center = rng.uniform(0.3, 0.7)
        width = rng.uniform(0.1, 0.3)
        window = 1.0 + scale * np.exp(-((x - center) / width) ** 2)
        a = a * window
    elif strategy == 2:
        # Gaussian noise
        a = a * (1.0 + scale * rng.standard_normal(n))
    elif strategy == 3:
        # Block flip (invert a random contiguous segment)
        start = rng.integers(0, n // 2)
        length = rng.integers(n // 10, n // 3)
        end = min(start + length, n)
        segment = a[start:end].copy()
        a[start:end] = a[start:end].max() + a[start:end].min() - segment
    else:
        # Polynomial perturbation
        x = np.linspace(0, 1, n)
        coeffs = scale * rng.standard_normal(4)
        poly = coeffs[0] + coeffs[1] * x + coeffs[2] * x ** 2 + coeffs[3] * x ** 3
        a = a + poly
    return np.clip(a, 0.0, 1000.0)


def run_lse_anneal(a0, T_start, T_end, n_steps, deadline, rng, init_lr=0.1):
    """
    Run Adam with LSE surrogate, annealing T from T_start to T_end over n_steps.
    Returns (best_score, best_a).
    """
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    lr = init_lr

    best_score = score(a)
    best_a = a.copy()
    no_improve = 0
    patience = 200

    for step in range(n_steps):
        if time.monotonic() > deadline:
            break

        # Geometric temperature schedule
        frac = step / max(1, n_steps - 1)
        T = T_start * (T_end / T_start) ** frac

        c = fast_self_conv(a)
        S = float(a.sum())
        if S < 1e-10:
            a += 1.0 / n
            continue

        # Compute gradient
        if T < 0.01 * float(np.max(c)):
            # Very low T: use top-k for efficiency and accuracy
            g, M_w, S_val = topk_grad(a, c, k=20)
        else:
            g, L = grad_lse_fft(a, c, T)

        t_adam += 1
        adam_step(a, g, m, v, t_adam, lr)
        a = clamp_nonneg(a)

        # Track best
        sc = score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Adaptive learning rate
        if no_improve >= patience:
            # Structured restart
            a = structured_perturb(best_a, rng, scale=0.05 + 0.1 * rng.random())
            m[:] = 0.0
            v[:] = 0.0
            t_adam = 0
            lr = max(0.01, lr * 0.75)
            no_improve = 0
            patience = max(100, patience // 2)

        # Gradually reduce LR
        if step % 100 == 0 and step > 0:
            lr = max(0.005, lr * 0.9)

    return best_score, best_a


def coordinate_descent_polish(a, max_rounds=100, deadline=None):
    """
    Coordinate descent: optimize one element at a time via golden section search.
    """
    n = len(a)
    a = a.copy()
    best_score = score(a)
    best_a = a.copy()
    rng = np.random.default_rng(12345)

    for rnd in range(max_rounds):
        if deadline is not None and time.monotonic() > deadline:
            break

        improved = False
        # Shuffle coordinate order
        order = rng.permutation(n)

        for j in order:
            if deadline is not None and time.monotonic() > deadline:
                break

            old_val = a[j]
            a[j] = 0.0
            c_without = fast_self_conv(a)
            S_without = float(a.sum())

            # f(v) = 2n * max(c_without + v*(a_other_contrib) + v^2 at appropriate spots) / (S_without + v)^2
            # This is complex; use a simpler approach: try a few candidate values
            candidates = [old_val,
                          old_val * 0.8, old_val * 1.2,
                          old_val * 0.5, old_val * 1.5,
                          max(0.01, old_val + 0.1),
                          max(0.01, old_val - 0.1)]

            best_local = old_val
            best_local_score = np.inf

            for cand in candidates:
                cand = float(np.clip(cand, 0.0, 1000.0))
                a[j] = cand
                sc = score(a)
                if sc < best_local_score:
                    best_local_score = sc
                    best_local = cand

            a[j] = best_local

            sc = score(a)
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2n*max(conv(a,a))/(sum(a))^2.
    Uses LSE annealing + Adam + structured restarts + coordinate descent polish.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = max(1.0, float(budget_s))
    deadline = start + total - 10.0

    # Try multiple lengths; weight towards 2048 (proven effective)
    candidate_ns = [2048]
    # Add a couple alternatives
    candidate_ns.append(1024)
    candidate_ns.append(4096)

    best_overall_score = np.inf
    best_overall_a = None

    # Time allocation: 70% to primary n=2048, 15% to n=1024, 15% to n=4096
    time_alloc = [(2048, 0.70), (1024, 0.15), (4096, 0.15)]

    for n, frac in time_alloc:
        t_avail = start + frac * (total - 10.0)
        if time.monotonic() >= t_avail:
            continue

        print("Working on n=%d, time budget=%.0fs" % (n, t_avail - time.monotonic()))

        # --- Initialization ---
        inits = []
        inits.append(("parent", init_parent(n, rng)))
        inits.append(("bump", init_smooth_bump(n, rng)))
        inits.append(("tapered", init_tapered(n, rng)))
        inits.append(("lowfreq", init_random_lowfreq(n, rng)))
        inits.append(("uniform", np.ones(n)))
        inits.append(("rand", np.clip(0.5 + 0.5 * rng.random(n), 0.05, 1000.0)))

        best_n_score = np.inf
        best_n_a = None

        # Phase A: Quick exploration from each init (use ~30% of time for this n)
        phase_a_end = time.monotonic() + 0.30 * (t_avail - time.monotonic())
        time_per_init = max(5.0, (phase_a_end - time.monotonic()) / len(inits))

        for tag, a0 in inits:
            if time.monotonic() > phase_a_end:
                break
            t_end = min(phase_a_end, time.monotonic() + time_per_init)
            # Short LSE run with high temperature
            T_start = 1.0 * float(np.max(fast_self_conv(a0)) + 1e-6)
            T_end = 0.1 * T_start
            sc, aa = run_lse_anneal(a0, T_start, T_end, n_steps=300,
                                    deadline=t_end, rng=rng, init_lr=0.15)
            if sc < best_n_score:
                best_n_score = sc
                best_n_a = aa.copy()
                print("  [%s n=%d] score=%.6f" % (tag, n, best_n_score))
            if sc < best_overall_score:
                best_overall_score = sc
                best_overall_a = aa.copy()

        # Phase B: Extended annealing from best (use ~50% of time)
        phase_b_end = time.monotonic() + 0.50 * (t_avail - time.monotonic())
        if best_n_a is not None and time.monotonic() < phase_b_end:
            c_best = fast_self_conv(best_n_a)
            peak = float(np.max(c_best))
            T_start = 0.5 * peak
            T_end = 0.01 * peak
            sc, aa = run_lse_anneal(best_n_a, T_start, T_end, n_steps=5000,
                                    deadline=phase_b_end, rng=rng, init_lr=0.08)
            if sc < best_n_score:
                best_n_score = sc
                best_n_a = aa.copy()
                print("  [anneal n=%d] score=%.6f" % (n, best_n_score))
            if sc < best_overall_score:
                best_overall_score = sc
                best_overall_a = aa.copy()

        # Phase C: Top-k refinement with low temperature (use remaining ~20%)
        if best_n_a is not None and time.monotonic() < t_avail:
            # Use top-k gradient with small k for precision
            a = best_n_a.copy()
            n_len = len(a)
            m = np.zeros(n_len)
            v = np.zeros(n_len)
            t_adam = 0
            lr = 0.02
            sc = score(a)
            best_local = sc
            best_local_a = a.copy()
            no_imp = 0

            while time.monotonic() < t_avail:
                c = fast_self_conv(a)
                g, Mw, S = topk_grad(a, c, k=8)
                t_adam += 1
                adam_step(a, g, m, v, t_adam, lr)
                a = clamp_nonneg(a)

                sc = score(a)
                if sc < best_local - 1e-9:
                    best_local = sc
                    best_local_a = a.copy()
                    no_imp = 0
                    if best_local < best_overall_score:
                        best_overall_score = best_local
                        best_overall_a = best_local_a.copy()
                else:
                    no_imp += 1
                    if no_imp > 300:
                        a = structured_perturb(best_local_a, rng, scale=0.03)
                        m[:] = 0.0
                        v[:] = 0.0
                        t_adam = 0
                        lr = max(0.005, lr * 0.8)
                        no_imp = 0

                if t_adam % 500 == 0:
                    lr = max(0.003, lr * 0.95)

            if best_local < best_n_score:
                best_n_score = best_local
                best_n_a = best_local_a.copy()
                print("  [topk_refine n=%d] score=%.6f" % (n, best_n_score))

    # Final coordinate descent polish on the global best
    if best_overall_a is not None and time.monotonic() < deadline:
        print("Applying coordinate descent polish...")
        polish_deadline = deadline
        sc, aa = coordinate_descent_polish(best_overall_a, max_rounds=20,
                                           deadline=polish_deadline)
        if sc < best_overall_score:
            best_overall_score = sc
            best_overall_a = aa.copy()
            print("  After polish: %.6f" % best_overall_score)

    # Safety fallback
    if best_overall_a is None:
        best_overall_a = np.ones(2048)

    out = [float(x) for x in best_overall_a]

    # Verify with the authoritative evaluator
    verified = float(evaluate_sequence(out))
    print("Final: internal=%.6f, verified=%.6f, n=%d, time=%.1fs" %
          (best_overall_score, verified, len(out), time.monotonic() - start))

    return out
```
