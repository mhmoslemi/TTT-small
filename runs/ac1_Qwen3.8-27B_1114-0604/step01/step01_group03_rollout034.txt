    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)
    c = c[:2*len(a)-1]
    return np.maximum(c, 0.0)

    """Compute 2*n*max(autocorr) / (sum(a))^2 using precomputed FFT length."""
    n = len(a)

"""
Search for non-negative step heights minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Uses a combination of:
- Exact LP direction finding (scipy HiGHS)
- Spectral amplitude compression
- Subgradient (anti-reversal) steps
- Adaptive line search and random restarts
"""

import numpy as np
import time
import math
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def _norm1(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / len(a)
    return a / s


def _autocorr_fft(a, L):
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:2*len(a)-1]
    np.maximum(c, 0.0, out=c)
    return c


def _score_fast(a, L):
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return np.inf
    c = _autocorr_fft(a, L)
    return 2.0 * n * c.max() / (s * s)


def _build_conv_matrix(a, n):
    """Build sparse (2n-1) x n matrix T[k,j] = a[k-j] if 0<=k-j<n else 0."""
    # Vectorized construction
    # For column j: non-zeros at rows j, j+1, ..., j+n-1 with values a[0], a[1], ..., a[n-1]
    j_idx = np.repeat(np.arange(n, dtype=np.int32), n)
    i_idx = np.tile(np.arange(n, dtype=np.int32), n)
    rows = j_idx + i_idx  # row = column + offset
    cols = j_idx
    vals = np.tile(a, n)
    T = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return T


def _lp_direction(a, n, T):
    """Solve LP: max sum(b) s.t. T@b <= M, b >= 0. Returns normalized b or None."""
    L = 1 << (2*n - 1).bit_length()
    c_auto = _autocorr_fft(a, L)
    M = float(c_auto.max())
    if M < 1e-20:
        return None
    
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    
    try:
        res = linprog(c_obj, A_ub=T, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 10})
        if res.success and res.x is not None:
            b = np.maximum(res.x, 0.0)
            s = b.sum()
            if s > 1e-10:
                return b / s
    except Exception as e:
        pass
    return None


def _spectral_compress(a, alpha, L):
    """Compress amplitude spectrum and project back to non-negative."""
    n = len(a)
    f = np.fft.rfft(a, L)
    amp = np.abs(f).astype(np.float64)
    phase = np.exp(1j * np.angle(f))
    
    # Compress non-DC amplitudes
    if len(amp) > 1:
        amp[1:] = np.power(np.maximum(amp[1:], 1e-30), alpha)
    
    f_new = amp * phase
    a_new = np.fft.irfft(f_new, L)[:n]
    return _norm1(a_new)


def _peak_grad(a, k_star, n):
    """Gradient of (a*a)_{k_star} w.r.t. a."""
    g = np.zeros(n)
    j_min = max(0, k_star - n + 1)
    j_max = min(n - 1, k_star)
    if j_min <= j_max:
        # g[j] = a[k_star - j] for j in [j_min, j_max]
        idx = np.arange(j_min, j_max + 1)
        g[idx] = a[k_star - idx]
    return g


def _multi_peak_grad(a, L, top_k=5):
    """Combined gradient from top-k peaks."""
    n = len(a)
    c = _autocorr_fft(a, L)
    # Get top-k peak locations
    # We want the k values with highest c[k]
    # Use partial sort for efficiency
    if top_k >= len(c):
        peaks = np.argsort(c)[::-1]
    else:
        # Get indices of top-k values
        part = np.argpartition(c, -top_k)[-top_k:]
        peaks = part[np.argsort(c[part])[::-1]]
    
    g = np.zeros(n)
    for k in peaks:
        g += _peak_grad(a, int(k), n)
    g /= len(peaks)
    return g


def _line_search(a, direction, L, n, base_score, steps):
    """Try mixing (1-t)*a + t*direction, return best."""
    best_a = a.copy()
    best_sc = base_score
    for t in steps:
        h = (1.0 - t) * a + t * direction
        h = _norm1(h)
        sc = _score_fast(h, L)
        if sc < best_sc:
            best_sc = sc
            best_a = h.copy()
    return best_a, best_sc
