from scipy.optimize import linprog
from scipy.fft import rfft, irfft

def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(a, b):
    la, lb = len(a), len(b)
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)

def _sanitize(p):
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 0 or not np.isfinite(s):
        return None
    return p / s

def _score(p):

    if p is None:

    m = p.size
    c = _conv_full(p, p)
    c = np.where(np.isfinite(c), c, 0.0)

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 over non-negative sequences a.
Uses LP-based alternating direction (max-sum LP for best direction),
combined with mirror descent and local block search for polishing.
"""

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog

def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(a, b):
    la = len(a)
    lb = len(b)
    nout = la + lb - 1
    if nout == 0:
        return np.array([], dtype=float)
    if la * lb <= 2000000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)

def _sanitize(p):
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return p / s

def _score(p):
    p = _sanitize(p)
    if p is None:
        return float('inf')
    m = p.size
    c = _conv_full(p, p)
    if c.size == 0:
        return float('inf')
    c = np.where(np.isfinite(c), c, 0.0)
    return float(2.0 * m * np.max(c))

def _build_lp_data(n, f):
    """Build data vector for the LP constraint matrix (n^2 entries)."""
    # Each column k has entries f[0], f[1], ..., f[n-1]
    return np.tile(f, n)

def _build_lp_pattern(n):
    """Build row/col indices for the LP constraint matrix."""
    total = n * n
    rows = np.empty(total, dtype=np.int32)
    cols = np.empty(total, dtype=np.int32)
    idx = 0
    for k in range(n):
        j_end = k + n - 1  # always valid since k <= n-1, so k+n-1 <= 2n-2
        length = n  # always n
        rows[idx:idx+length] = np.arange(k, j_end + 1, dtype=np.int32)
        cols[idx:idx+length] = k
        idx += length
    return rows, cols

def _lp_solve(f, M, rows, cols):
    """Solve max sum(g) s.t. conv(f,g) <= M, g >= 0. Returns normalized g."""
    n = len(f)
    data = np.tile(f, n)
    A = csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    c_obj = -np.ones(n)
    bounds = [(0, 10.0)] * n  # upper bound 10 to prevent unboundedness
    b_ub = np.full(2*n-1, M)
    
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 60, 'presolve': True})
    except Exception:
        return None
    
    if res.status != 0:
        return None
    
    g = res.x
    g = np.where(np.isfinite(g), g, 0.0)
    g = np.clip(g, 0.0, None)
    s = float(g.sum())
    if s <= 1e-15 or not np.isfinite(s):
        return None
    return g / s

def _lp_optimize(f0, rng, deadline, verbose=True):
    """LP-based alternating optimization."""
    f = _sanitize(f0)
    if f is None:
        return None, float('inf')
    n = f.size
    
    rows, cols = _build_lp_pattern(n)
    
    best = f.copy()
    best_score = _score(f)
    cur_score = best_score
    stall = 0
    it = 0
    
    while time.monotonic() < deadline:
        # Compute M = max(conv(f,f))
        c = _conv_full(f, f)
        M = float(np.max(c))
        
        # Solve LP
        g = _lp_solve(f, M, rows, cols)
        if g is None:
            stall += 10
            if stall > 30:
                break
            # Perturb and retry
            f_try = _sanitize(f * (1 + 0.01 * rng.normal(0, 1, n)))
            if f_try is not None:
                f = f_try
            continue
        
        # Line search: try several step sizes
        best_t = None
        best_sc = cur_score
        for t in (1.0, 0.7, 0.5, 0.3, 0.1):
            f_try = (1 - t) * f + t * g
            f_try = _sanitize(f_try)
            if f_try is None:
                continue
            sc = _score(f_try)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_t = t
        
        if best_t is not None:
            f = (1 - best_t) * f + best_t * g
            f = _sanitize(f)
            cur_score = best_sc
            stall = 0
            
            if cur_score < best_score:
                best = f.copy()
                best_score = cur_score
                if verbose and it % 10 == 0:
                    print(f"LP iter {it}: score = {best_score:.6f}, t = {best_t}")
        else:
            stall += 1
            if stall % 20 == 0 and verbose:
                print(f"LP iter {it}: stall = {stall}, score = {cur_score:.6f}")
            if stall > 50:
                # Perturb
                f_try = _sanitize(f * np.exp(rng.normal(0, 0.05, n)))
                if f_try is not None:
                    sc_try = _score(f_try)
                    if sc_try < best_score * 1.1:
                        f = f_try
                        cur_score = sc_try
                    else:
                        f = best.copy()
                        cur_score = best_score
                stall = 0
            if stall > 200:
                break
        
        it += 1
    
    return best, best_score
