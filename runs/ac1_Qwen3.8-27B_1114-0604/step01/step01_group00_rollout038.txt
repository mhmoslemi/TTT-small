"""
Minimizes 2*n*max(f*f)/(sum f)^2 for non-negative step function f.
Uses mirror descent with soft-max gradient, frequency-domain smoothing,
adaptive step sizes, and multi-phase optimization.
"""

import numpy as np
import time


def _autocorr(f):
    """Full linear autocorrelation (self-convolution) via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    ac = np.fft.irfft(F * F.conjugate(), size)
    return ac[:2*n - 1]


def _soft_gradient(ac, f, T, top_k=None):
    """Compute gradient using soft-max weighting of autocorrelation peaks."""
    n = len(f)
    ac_shifted = ac - ac.max()
    if T < 1e-300:
        # Hard peak
        p = np.zeros(len(ac))
        p[int(np.argmax(ac))] = 1.0
    else:
        p = np.exp(ac_shifted / T)
        s = p.sum()
        if s < 1e-300:
            p = np.zeros(len(ac))
            p[int(np.argmax(ac))] = 1.0
        else:
            p /= s
    
    if top_k is not None and top_k < len(ac):
        # Keep only top-k
        idx = np.argsort(ac)[::-1][:top_k]
        mask = np.zeros(len(ac))
        mask[idx] = 1.0
        p = p * mask
        s = p.sum()
        if s < 1e-300:
            p = np.zeros(len(ac))
            p[int(np.argmax(ac))] = 1.0
        else:
            p /= s
    
    # grad[j] = 2 * sum_k p[k] * f[k-j] for valid j
    # This is a correlation: grad = 2 * correlate(p, f, 'valid')
    # p has length 2n-1, f has length n, 'valid' gives length (2n-1) - n + 1 = n
    # But we need to be careful about the indexing
    # correlate(p, f, 'valid')[j] = sum_i p[i+j] * f[i] for i=0..n-1
    # That gives: sum_i p[i+j] * f[i] = sum_i p[(i+j)] * f[i]
    # We want: sum_k p[k] * f[k-j] = sum_k p[k] * f[k-j]
    # Let m = k-j: sum_m p[m+j] * f[m] = correlate(p, f, 'valid')[j] ✓
    
    from scipy.signal import correlate
    grad = 2.0 * correlate(p, f, mode='valid')
    return grad, p


def _mirror_step(f, grad, eta):
    """Mirror descent (exponentiated gradient) step on simplex."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy()
    f_new /= s
    return f_new


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _score(f):
    """Compute 2n * max(f*f) / (sum f)^2."""
    n = len(f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    c = _autocorr(f)
    return 2.0 * n * c.max() / (s * s)


def _freq_smooth(f, alpha):
    """Apply Gaussian low-pass filter in frequency domain."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    freqs = np.fft.rfftfreq(size)
    max_freq = freqs.max()
    if max_freq > 0:
        F *= np.exp(-alpha * (freqs / max_freq) ** 2)
    f_new = np.fft.irfft(F, size)[:n]
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f.copy()
    return f_new / s

    n = len(f)
    f = np.maximum(f, 0.0).copy()
    f /= f.sum()
    
    best_f = f.copy()
    best_score = _score(f)
    
    ac = _autocorr(f)
    peak0 = ac.max()
    
    T_start = peak0
    T_end = peak0 * 1e-7
    eta_start = 2.0
    eta_end = 0.01
    
    n_total = 50000
    stuck = 0
    last_improve = best_score
    
    for it in range(n_total):
        if time.monotonic() > deadline:
            break
        
        progress = min(it / n_total, 1.0)
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress
        
        ac = _autocorr(f)
        grad, p = _soft_gradient(ac, f, T)
        
        f = _mirror_step(f, grad, eta)
        
        # Evaluate every 200 iterations
        if it % 200 == 0 or it == n_total - 1:
            sc = _score(f)
            if sc < best_score - 1e-8:

                break

"""
Minimizes 2*n*max(f*f)/(sum f)^2 for non-negative step function f.
Uses multi-phase optimization: mirror descent with soft-max gradient,
projected gradient with Nesterov momentum, frequency-domain smoothing,
and local peak-targeting polish. Multiple diverse starting points.

import numpy as np
from scipy.signal import correlate
import time


def _autocorr(f):
    """Full linear autocorrelation (self-convolution) via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    ac = np.fft.irfft(F * F.conjugate(), size)
    return ac[:2 * n - 1]


def _soft_gradient(ac, f, T, top_k=None):
    """Compute gradient using soft-max weighting of autocorrelation peaks."""
    n = len(f)
    ac_shifted = ac - ac.max()
    if T < 1e-300:
        p = np.zeros(len(ac))
        p[int(np.argmax(ac))] = 1.0
    else:
        p = np.exp(ac_shifted / T)
        s = p.sum()
        if s < 1e-300:
            p = np.zeros(len(ac))
            p[int(np.argmax(ac))] = 1.0
        else:
            p /= s

    if top_k is not None and top_k < len(ac):
        idx = np.argsort(ac)[::-1][:top_k]
        mask = np.zeros(len(ac))
        mask[idx] = 1.0
        p = p * mask
        s = p.sum()
        if s < 1e-300:
            p = np.zeros(len(ac))
            p[int(np.argmax(ac))] = 1.0
        else:
            p /= s

    grad = 2.0 * correlate(p, f, mode='valid')
    return grad, p


def _mirror_step(f, grad, eta):
    """Mirror descent (exponentiated gradient) step on simplex."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy()
    f_new /= s
    return f_new


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _score(f):
    """Compute 2n * max(f*f) / (sum f)^2."""
    n = len(f)
    s = f.sum()
    if s < 1e-10:
        return np.inf
    c = _autocorr(f)
    return 2.0 * n * c.max() / (s * s)


def _freq_smooth(f, alpha):
    """Apply Gaussian low-pass filter in frequency domain."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    freqs = np.fft.rfftfreq(size)
    max_freq = freqs[-1]
    if max_freq > 0:
        F *= np.exp(-alpha * (freqs / max_freq) ** 2)
    f_new = np.fft.irfft(F, size)[:n]
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f.copy()
    return f_new / s


def _perturb_structured(f, rng, n):

    j = np.arange(n, dtype=np.float64)
    freq_choice = rng.integers(1, max(2, n // 10))
    phase = rng.uniform(0, 2 * np.pi)
    mag = 0.02 / n
    f_try = f + mag * np.sin(2 * np.pi * freq_choice * j / n + phase)
    f_try = np.maximum(f_try, 0.0)
    s = f_try.sum()
    if s < 1e-10:
        return f
    return f_try / s

    else:
        f = f / s

    best_f = f.copy()
    best_score = _score(f)

    ac = _autocorr(f)
    peak0 = ac.max()
    if peak0 < 1e-300:
        return best_f, best_score

    T_start = peak0
    T_end = peak0 * 1e-8
    eta_start = 3.0
    eta_end = 0.005

    n_total = 80000
    stuck = 0
    eval_interval = 200

    for it in range(n_total):
        if time.monotonic() > deadline:
            break

        progress = min(it / n_total, 1.0)
        T = T_start * (T_end / T_start) ** (progress ** 0.8)
        eta = eta_start * (eta_end / eta_start) ** (progress ** 0.6)

        ac = _autocorr(f)

        # Use top-k for sharper targeting in later stages
        top_k = None
        if progress > 0.5:
            top_k = max(3, int(50 * (1.0 - progress)))

        grad, p = _soft_gradient(ac, f, T, top_k=top_k)

        f = _mirror_step(f, grad, eta)

        if it % eval_interval == 0 or it == n_total - 1:
            sc = _score(f)
            if sc < best_score - 1e-9:
                best_score = sc
                best_f = f.copy()
                stuck = 0
            else:
                stuck += 1

            if stuck > 6:
                f = _perturb_structured(f, rng, n)
                stuck = 0

        # Frequency smoothing periodically in later stages
        if progress > 0.4 and it % 10000 == 9999:
            f = _freq_smooth(f, 0.03)

    sc = _score(f)
    if sc < best_score:
        best_score = sc
        best_f = f.copy()

    return best_f, best_score


def _optimize_projected(f, deadline, rng, label=""):
    """Projected gradient descent with Nesterov momentum."""
    n = len(f)
    f = np.maximum(f, 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f = f / s

    best_f = f.copy()
    best_score = _score(f)

    ac = _autocorr(f)
    peak0 = ac.max()
    if peak0 < 1e-300:
        return best_f, best_score

    T_start = peak0
    T_end = peak0 * 1e-8
    eta_start = 1.0
    eta_end = 0.005

    n_total = 80000
    stuck = 0
    momentum = 0.0
    f_prev = f.copy()

    for it in range(n_total):
        if time.monotonic() > deadline:
            break

        progress = min(it / n_total, 1.0)
        T = T_start * (T_end / T_start) ** (progress ** 0.8)
        eta = eta_start * (eta_end / eta_start) ** (progress ** 0.6)

        # Nesterov lookahead
        f_y = f + momentum * (f - f_prev)
        f_y = np.maximum(f_y, 0.0)
        sy = f_y.sum()
        if sy < 1e-10:
            f_y = f.copy()
        else:
            f_y = f_y / sy

        ac = _autocorr(f_y)

        top_k = None
        if progress > 0.5:
            top_k = max(3, int(50 * (1.0 - progress)))

        grad, p = _soft_gradient(ac, f_y, T, top_k=top_k)

        f_trial = f_y - eta * grad
        f_new = _project_simplex(f_trial)

        # Update momentum (Nesterov schedule)
        momentum = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0
        momentum = min(momentum, 0.95)
        f_prev = f_y
        f = f_new

        if it % 300 == 0 or it == n_total - 1:
            sc = _score(f)
            if sc < best_score - 1e-9:
                best_score = sc
                best_f = f.copy()
                stuck = 0
            else:
                stuck += 1

            if stuck > 5:
                f = _perturb_structured(f, rng, n)
                f = _project_simplex(f)
                stuck = 0
                momentum = 0.0
                f_prev = f.copy()

    sc = _score(f)
    if sc < best_score:
        best_score = sc
        best_f = f.copy()

    return best_f, best_score


def _polish(f, deadline, rng, label=""):
    """Local peak-targeting polish with mass transfers."""
    n = len(f)
    f = np.maximum(f, 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f = f / s

    best_f = f.copy()
    best_score = _score(f)

    ac = _autocorr(f)
    peak_val = ac.max()

    n_iters = 20000
    no_improve = 0

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        ac = _autocorr(f)
        k_star = int(np.argmax(ac))
        peak_val = ac[k_star]

        # Compute contribution to peak
        contrib = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                contrib[j] = f[j] * f[idx]

        # Also consider second peak
        ac_sorted = np.sort(ac)[::-1]
        second_peak = ac_sorted[1] if len(ac_sorted) > 1 else 0.0

        improved = False
        top_idx = np.argsort(contrib)[::-1]
        bot_idx = np.argsort(contrib)

        # Try transfers from top contributors to bottom
        for j_don in top_idx[:8]:
            if improved:
                break
            for j_rcv in bot_idx[:8]:
                if j_don == j_rcv:
                    continue
                for frac in [0.03, 0.08, 0.15, 0.25]:
                    amt = f[j_don] * frac
                    if amt < 1e-15:
                        continue
                    f_try = f.copy()
                    f_try[j_don] -= amt
                    f_try[j_rcv] += amt
                    f_try = np.maximum(f_try, 0.0)
                    f_try /= f_try.sum()
                    ac_try = _autocorr(f_try)
                    if ac_try.max() < peak_val - 1e-12:
                        f = f_try
                        peak_val = ac_try.max()
                        improved = True
                        break

        if improved:
            no_improve = 0
            sc = _score(f)
            if sc < best_score:
                best_score = sc
                best_f = f.copy()
        else:
            no_improve += 1
            if no_improve > 50:
                # Try random pair transfer
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j:
                    f_try = f.copy()
                    amt = f_try[i] * 0.1
                    f_try[i] -= amt
                    f_try[j] += amt
                    f_try = np.maximum(f_try, 0.0)
                    f_try /= f_try.sum()
                    ac_try = _autocorr(f_try)
                    if ac_try.max() < peak_val:
                        f = f_try
                        peak_val = ac_try.max()
                        no_improve = 0

                if no_improve > 200:
                    # Frequency smooth to escape
                    f = _freq_smooth(f, 0.05)
                    no_improve = 0

    sc = _score(f)
    if sc < best_score:
        best_score = sc
        best_f = f.copy()

    return best_f, best_score


def _optimize_combined(f, deadline, rng, label=""):
    """Alternating mirror and projected gradient with shared state."""
    n = len(f)
    f = np.maximum(f, 0.0).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f = f / s

    best_f = f.copy()
    best_score = _score(f)

    ac = _autocorr(f)
    peak0 = ac.max()
    if peak0 < 1e-300:
        return best_f, best_score

    T = peak0
    T_end = peak0 * 1e-8

    block_size = 2000
    n_blocks = 100
    stuck = 0

    for block in range(n_blocks):
        if time.monotonic() > deadline:
            break

        progress = min(block / n_blocks, 1.0)
        T = peak0 * (T_end / peak0) ** (progress ** 0.7)

        # Alternate between mirror and projected for diversity
        if block % 2 == 0:
            eta = 2.0 * (0.01) ** (progress ** 0.6)
            for it in range(block_size):
                if time.monotonic() > deadline:
                    break
                ac = _autocorr(f)
                top_k = None
                if progress > 0.5:
                    top_k = max(3, int(30 * (1.0 - progress)))
                grad, p = _soft_gradient(ac, f, T, top_k=top_k)
                f = _mirror_step(f, grad, eta)
        else:
            eta = 0.5 * (0.01) ** (progress ** 0.6)
            momentum = 0.3
            f_prev = f.copy()
            for it in range(block_size):
                if time.monotonic() > deadline:
                    break
                f_y = f + momentum * (f - f_prev)
                f_y = np.maximum(f_y, 0.0)
                sy = f_y.sum()
                if sy > 1e-10:
                    f_y = f_y / sy
                ac = _autocorr(f_y)
                top_k = None
                if progress > 0.5:
                    top_k = max(3, int(30 * (1.0 - progress)))
                grad, p = _soft_gradient(ac, f_y, T, top_k=top_k)
                f_new = _project_simplex(f_y - eta * grad)
                f_prev = f_y
                f = f_new

        sc = _score(f)
        if sc < best_score - 1e-9:
            best_score = sc
            best_f = f.copy()
            stuck = 0
            if block % 5 == 0:
                print(f"    {label} block {block}: {best_score:.6f}")
        else:
            stuck += 1

        if stuck > 3:
            f = _perturb_structured(f, rng, n)
            stuck = 0

        if progress > 0.5 and block % 5 == 4:
            f = _freq_smooth(f, 0.04)

    sc = _score(f)
    if sc < best_score:
        best_score = sc
        best_f = f.copy()

    return best_f, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(f*f)/(sum f)^2.

    Multi-phase approach with diverse starts:
    1. Generate diverse starting points (given, structured, random)
    2. Mirror descent with annealing (exploration)
    3. Combined mirror/projected optimization (refinement)
    4. Frequency smoothing + local polish
    5. Final optimization rounds
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    f_init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(f_init)
    print(f"Sequence length: {n}")

    # Normalize
    f_start = f_init.copy()
    s = f_start.sum()
    if s < 1e-10:
        f_start = np.ones(n)
    else:
        f_start = f_start / s

    best_f = f_start.copy()
    best_score = _score(f_start)
    print(f"Initial score: {best_score:.6f}")

    # Generate diverse starting points
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    starts = {}
    starts['given'] = f_start.copy()
    starts['uniform'] = np.ones(n)
    starts['sine_low'] = 1.0 + 0.4 * np.sin(2.0 * np.pi * j / n)
    starts['sine_med'] = 1.0 + 0.4 * np.sin(4.0 * np.pi * j / n)
    starts['sine_high'] = 1.0 + 0.4 * np.sin(8.0 * np.pi * j / n)
    starts['cos2'] = 1.0 + 0.5 * np.cos(4.0 * np.pi * j / n)
    starts['cos4'] = 1.0 + 0.5 * np.cos(8.0 * np.pi * j / n)
    starts['u_shape'] = 1.0 + 2.0 * ((2.0 * j / (n - 1) - 1.0)) ** 2
    starts['inv_u'] = 1.0 - 0.5 * ((2.0 * j / (n - 1) - 1.0)) ** 2
    starts['bimodal'] = (np.exp(-((j - 0.25 * (n - 1)) / (0.1 * n)) ** 2) +
                         np.exp(-((j - 0.75 * (n - 1)) / (0.1 * n)) ** 2) + 0.01)
    starts['sinc'] = np.abs(np.sinc(j / 30.0)) + 0.01
    starts['rand1'] = rng.exponential(1.0, n)
    starts['rand2'] = rng.dirichlet(np.ones(n) * 0.1)

    # Normalize all and evaluate
    start_scores = {}
    for name in starts:
        f0 = np.maximum(starts[name], 0.0)
        s0 = f0.sum()
        if s0 > 1e-10:
            f0 = f0 / s0
        else:
            f0 = np.ones(n) / n
        starts[name] = f0
        start_scores[name] = _score(f0)

    print("Starting points:")
    for name, sc in sorted(start_scores.items(), key=lambda x: x[1]):
        print(f"  {name}: {sc:.6f}")

    # Pick best start
    best_name = min(start_scores, key=start_scores.get)
    if start_scores[best_name] < best_score:
        best_score = start_scores[best_name]
        best_f = starts[best_name].copy()
    print(f"Best start: '{best_name}' with score {best_score:.6f}")

    # Sort starts by quality, use top ones for optimization
    sorted_starts = sorted(start_scores.items(), key=lambda x: x[1])

    # Phase 1: Mirror descent from best start (40% time)
    phase1_budget = (budget_s - 60) * 0.35
    phase1_deadline = time.monotonic() + phase1_budget
    print(f"\nPhase 1: Mirror descent (budget {phase1_budget:.0f}s)")
    f1, s1 = _optimize_mirror(best_f.copy(), phase1_deadline, rng, "P1")
    if s1 < best_score:
        best_score = s1
        best_f = f1.copy()
        print(f"  Phase 1 result: {best_score:.6f}")

    # Phase 2: Combined optimization from best (25% time)
    phase2_budget = (budget_s - 60) * 0.25
    phase2_deadline = time.monotonic() + phase2_budget
    print(f"\nPhase 2: Combined optimization (budget {phase2_budget:.0f}s)")
    f2, s2 = _optimize_combined(best_f.copy(), phase2_deadline, rng, "P2")
    if s2 < best_score:
        best_score = s2
        best_f = f2.copy()
        print(f"  Phase 2 result: {best_score:.6f}")

    # Phase 3: Mirror descent from second-best start (15% time)
    if len(sorted_starts) > 1:
        second_name = sorted_starts[1][0]
        second_score = sorted_starts[1][1]
        if second_score < best_score + 0.05:
            phase3_budget = (budget_s - 60) * 0.15
            phase3_deadline = time.monotonic() + phase3_budget
            print(f"\nPhase 3: Mirror from '{second_name}' (budget {phase3_budget:.0f}s)")
            f3, s3 = _optimize_mirror(starts[second_name].copy(), phase3_deadline, rng, "P3")
            if s3 < best_score:
                best_score = s3
                best_f = f3.copy()
                print(f"  Phase 3 result: {best_score:.6f}")

    # Phase 4: Polish (15% time)
    phase4_budget = (budget_s - 60) * 0.15
    phase4_deadline = time.monotonic() + phase4_budget
    print(f"\nPhase 4: Polish (budget {phase4_budget:.0f}s)")
    f4, s4 = _polish(best_f.copy(), phase4_deadline, rng)
    if s4 < best_score:
        best_score = s4
        best_f = f4.copy()
        print(f"  Phase 4 result: {best_score:.6f}")

    # Phase 5: Final optimization rounds (10% time)
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\nPhase 5: Final optimization (budget {remaining:.0f}s)")
        f5, s5 = _optimize_mirror(best_f.copy(), min(time.monotonic() + remaining * 0.6, deadline), rng, "P5a")
        if s5 < best_score:
            best_score = s5
            best_f = f5.copy()
            print(f"  P5a result: {best_score:.6f}")

        remaining = deadline - time.monotonic()
        if remaining > 10:
            f6, s6 = _polish(best_f.copy(), min(time.monotonic() + remaining * 0.8, deadline), rng)
            if s6 < best_score:
                best_score = s6
                best_f = f6.copy()
                print(f"  P5b result: {best_score:.6f}")

    # Ensure valid output
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = _score(best_f)
    if final_score < best_score or not np.isfinite(best_score):
        best_score = final_score

    elapsed = time.monotonic() - t_start
    print(f"\n=== FINAL SCORE: {best_score:.6f} (elapsed: {elapsed:.1f}s) ===")
    return [float(x) for x in best_f]
