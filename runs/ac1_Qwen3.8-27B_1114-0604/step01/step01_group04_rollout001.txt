def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))

    # This is 2 * (w * a) at specific positions
    # w has length 2n-1, a has length n
    # For position j in [0, n-1]: g_j = 2 * sum_k w_k * a_{k-j}
    # Let m = k-j, so k = m+j: g_j = 2 * sum_m w_{m+j} * a_m
    # So g = 2 * (a * w_shifted) where w_shifted[j] = sum_m a_m * w_{m+j}
    # This is the cross-correlation of a with w, evaluated at lags 0..n-1
    # Actually it's: g_j = 2 * sum_{m=0}^{n-1} a_m * w_{m+j}
    # which is the valid cross-correlation of w with a (w is longer)
    # corr[j] = sum_m a_m * w_{m+j} for j = 0, 1, ..., (2n-1)-n = n-1
    # Use FFT-based approach
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fw = np.fft.rfft(w, fsize)
    # We need sum_m a_m * w_{m+j} = correlation
    # In FFT: ifft(conj(FA) * FW) gives correlation
    # Actually: ifft(FA * conj(FW)) gives one cross-correlation
    # Let's use: corr[j] = sum_m a[m] * w[m+j]
    # This is the "long" cross-correlation. We can compute via:
    # pad a to size, compute ifft(conj(FFT(a_padded)) * FFT(w_padded))
    # But simpler: use np.correlate for w and a
    # corr = np.correlate(w, a, mode='valid') gives corr[j] = sum_m w[m+j] * a[m]
    corr = np.correlate(w, a, mode='valid')
    return 2.0 * corr


def mirror_descent_step(a, g, eta):
    """Multiplicative weight update: a_i <- a_i * exp(-eta * g_i), then normalize."""
    # Scale gradient to have reasonable magnitude
    g_max = np.max(np.abs(g))
    if g_max < 1e-30:
        return a.copy()
    g_scaled = g / g_max
    log_update = -eta * g_scaled
    # Clip to avoid underflow/overflow
    log_update = np.clip(log_update, -50.0, 50.0)
    a_new = a * np.exp(log_update)
    s = a_new.sum()
    if s < 1e-300:
        return a.copy()
    return a_new / s


def peak_surgery(a, b, amount=0.01):
    """
    Directly reduce mass at positions contributing to the convolution peak.
    Move mass from peak-contributing pairs to low-contribution positions.
    """
    n = a.size
    k_star = int(np.argmax(b))
    # Gradient at peak: g_j = 2 * a_{k*-j} for valid j
    # Elements with high a_{k*-j} contribute most to the peak
    # We want to reduce a_j where a_{k*-j} is large (and j != k*-j)
    # And add mass where a_{k*-j} is small for all important k
    
    # Compute "contribution score" for each element
    # c_j = a_{k*-j} (how much a_j contributes to peak at k*)
    k_star = int(np.clip(k_star, 0, 2 * n - 2))
    c = np.zeros(n)
    valid_j = np.arange(max(0, k_star - n + 1), min(k_star + 1, n))
    for j in valid_j:
        c[j] = a[k_star - j]
    
    # Source: elements with high c_j (contribute to peak)
    # Sink: elements with low c_j (don't contribute to peak)
    source_mask = c > np.mean(c)
    sink_mask = ~source_mask
    
    if not source_mask.any() or not sink_mask.any():
        return a.copy()
    
    # Transfer mass
    source_mass = a[source_mask]
    sink_mass = a[sink_mask]
    
    # Reduce sources, increase sinks proportionally
    total_source = source_mass.sum()
    total_sink = sink_mass.sum()
    
    if total_source < 1e-15 or total_sink < 1e-15:
        return a.copy()
    
    transfer = min(amount * total_source, 0.5 * total_source)
    
    a_new = a.copy()
    # Reduce sources proportionally
    a_new[source_mask] -= transfer * source_mass / total_source
    # Add to sinks proportionally (weighted by their "safety")
    # Safety = 1 - c_j (lower c means safer to add)
    safety = np.maximum(1.0 - c[sink_mask] / (np.max(c) + 1e-15), 0.0)
    safety_sum = safety.sum()
    if safety_sum < 1e-15:
        safety = np.ones(sink_mask.sum())
        safety_sum = safety.sum()
    a_new[sink_mask] += transfer * safety / safety_sum
    
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-15:
        return a.copy()
    return a_new / s


def mirror_descent_optimize(a_init, time_budget, rng, use_surgery=True):
    """
    Mirror descent (multiplicative weights) with adaptive temperature and
    optional peak surgery. Returns (best_a, best_score).
    """
    n = a_init.size
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()
    
    best_a = a.copy()
    best_sc = normed_score(a)
    
    eta = 0.5
    tau_frac = 0.03
    decay_rate = 0.9995
    last_improve = 0
    iters = 0
    stall_count = 0
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))
        
        # Multi-scale gradient
        tau = tau_frac * float(np.max(b))
        if tau < 1e-18:
            tau = 1e-18
        
        # Use 2-3 temperature scales
        n_scales = 3
        g = np.zeros(n)
        for s in range(n_scales):
            tau_s = tau * (2.0 ** s)
            w = softmax_weights(b, tau_s)
            g += smoothed_subgrad(a, b, tau_s)
        g /= n_scales
        
        # Mirror descent step
        a_new = mirror_descent_step(a, g, eta)
        
        # Optional peak surgery
        if use_surgery and iters % 5 == 0:
            a_new = peak_surgery(a_new, full_conv(a_new), amount=0.005)
        
        b_new = full_conv(a_new)
        new_sc = 2.0 * n * float(np.max(b_new))
        
        # Accept with simulated annealing
        accept = (new_sc <= cur_sc) or (rng.random() < 0.02 * np.exp(-iters / 5000.0))
        if accept:
            a = a_new
        else:
            # Try smaller step
            a_half = mirror_descent_step(a, g, eta * 0.3)
            a = a_half
        
        if new_sc < best_sc - 1e-10:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = iters
            stall_count = 0
        else:
            stall_count += 1
        
        iters += 1
        
        # Adaptive: decay step size
        eta *= decay_rate
        
        # Restart if stalled
        if iters - last_improve > 5000:
            # Perturb from best
            noise_scale = 0.03 + 0.05 * rng.random()
            a = best_a + noise_scale * rng.standard_normal(n)
            a = np.maximum(a, 1e-10)
            a = a / a.sum()
            eta = 0.3 + 0.4 * rng.random()
            tau_frac = 0.02 + 0.04 * rng.random()
            last_improve = iters
            stall_count = 0
        
        # Reheat periodically
        if iters % 10000 == 0:
            eta = max(eta, 0.2 + 0.3 * rng.random())
            tau_frac = 0.02 + 0.05 * rng.random()
    
    return best_a, best_sc


def gradient_descent_optimize(a_init, time_budget, rng):
    """
    Standard projected gradient descent as a secondary method.
    Different from mirror descent - uses additive updates with simplex projection.
    """
    n = a_init.size
    a = a_init.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_sc = normed_score(a)
    
    eta = 0.2
    tau_frac = 0.04
    beta = 0.7
    vel = np.zeros(n)
    last_improve = 0
    iters = 0
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))
        
        tau = tau_frac * float(np.max(b))
        if tau < 1e-18:
            tau = 1e-18
        
        # Multi-scale
        g = np.zeros(n)
        for s in range(3):
            tau_s = tau * (2.0 ** s)
            w = softmax_weights(b, tau_s)
            g += smoothed_subgrad(a, b, tau_s)
        g /= 3.0
        
        # Nesterov momentum
        vel = beta * vel + (1.0 - beta) * g
        a_new = a - eta * vel
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-15:
            a_new = a.copy()
        else:
            a_new = a_new / s
        
        b_new = full_conv(a_new)
        new_sc = 2.0 * n * float(np.max(b_new))
        
        if new_sc <= cur_sc or rng.random() < 0.03:
            a = a_new
        else:
            a = a - 0.5 * eta * vel
            a = np.maximum(a, 0.0)
            a = a / a.sum()
        
        if new_sc < best_sc - 1e-10:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = iters
        
        iters += 1
        eta *= 0.9997
        
        if iters - last_improve > 4000:
            a = best_a + 0.04 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            vel = np.zeros(n)
            eta = 0.1 + 0.2 * rng.random()
            last_improve = iters
    
    return best_a, best_sc


def init_u_shape(n, alpha=2.0):
    """U-shaped initialization: high at edges, low in middle."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    t = (i - c) / c
    a = 1.0 + alpha * t ** 2
    return a / a.sum()


def init_v_shape(n, alpha=2.0):
    """V-shaped: low at edges, high in middle."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    t = (i - c) / c
    a = 1.0 - alpha * t ** 2
    a = np.maximum(a, 0.01)
    return a / a.sum()


def init_two_peak(n, rng):
    """Two Gaussian peaks."""
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.1 + 0.2 * rng.random())
    c2 = n * (0.7 + 0.2 * rng.random())
    sigma = n * (0.05 + 0.1 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.01
    return a / a.sum()


def init_fringed(n, rng):
    """U-shape with oscillations (fringes)."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    t = (i - c) / c
    freq = 2.0 + 4.0 * rng.random()
    a = (1.0 + 2.0 * t ** 2) * (1.0 + 0.3 * np.cos(2.0 * np.pi * freq * t))
    a = np.maximum(a, 0.01)
    return a / a.sum()


def init_resampled(prev_a, n, rng):
    """Resample a previous sequence to a new length n."""
    # Use interpolation with some noise
    old_n = prev_a.size
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, prev_a)
    a += 0.02 * np.abs(rng.standard_normal(n))
    a = np.maximum(a, 0.0)
    return a / a.sum()


def fine_polish(a, time_budget, rng):
    """Very fine local optimization with tiny steps."""
    n = a.size
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    
    best_a = a.copy()
    best_sc = normed_score(a)
    
    eta = 0.05
    tau_frac = 0.01
    last_improve = 0
    iters = 0
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))
        
        tau = tau_frac * float(np.max(b))
        if tau < 1e-18:
            tau = 1e-18
        w = softmax_weights(b, tau)
        g = smoothed_subgrad(a, b, tau)
        
        # Try mirror descent
        a_md = mirror_descent_step(a, g, eta)
        sc_md = normed_score(a_md)
        
        # Try gradient descent
        a_gd = np.maximum(a - eta * 0.5 * g, 0.0)
        a_gd = a_gd / a_gd.sum()
        sc_gd = normed_score(a_gd)
        
        # Try peak surgery
        a_ps = peak_surgery(a, b, amount=0.003)
        sc_ps = normed_score(a_ps)
        
        # Pick best
        candidates = [(sc_md, a_md), (sc_gd, a_gd), (sc_ps, a_ps)]
        candidates.sort(key=lambda x: x[0])
        best_sc_cand, best_a_cand = candidates[0]
        
        if best_sc_cand < best_sc - 1e-12:
            best_sc = best_sc_cand
            best_a = best_a_cand.copy()
            a = best_a_cand.copy()
            last_improve = iters
        else:
            # Small random perturbation
            if rng.random() < 0.1:
                a = a + 0.005 * rng.standard_normal(n)
                a = np.maximum(a, 0.0)
                a = a / a.sum()
        
        iters += 1
        eta *= 0.9999
        
        if iters - last_improve > 2000:
            break
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses mirror descent (multiplicative weights), peak surgery, and multi-method
    optimization across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)
    
    # Load previous best
    prev_a = None
    prev_sc = np.inf
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size >= 10 and np.all(np.isfinite(h)) and h.sum() > 0.01:
            prev_a = h / h.sum()
            prev_sc = 2.0 * h.size * float(np.max(full_conv(prev_a)))
            print(f"[init] Previous best: n={h.size}, score={prev_sc:.6f}")
    except Exception:
        pass
    
    best_overall = None
    best_overall_sc = np.inf
    
    if prev_a is not None and prev_sc < 2.0:
        best_overall = prev_a.copy()
        best_overall_sc = prev_sc
    
    # n values to try - focus on sizes near previous best and some larger
    n_list = [128, 256, 512, 1024, 2048, 4096]
    if prev_a is not None:
        np_ = prev_a.size
        if np_ not in n_list and 32 <= np_ <= 8192:
            n_list.append(np_)
        n_list = sorted(set(n_list))
    
    total_time = max(1.0, deadline - time.monotonic())
    print(f"[plan] n_list={n_list}, total_time={total_time:.1f}s")
    
    # Allocate time: more time for sizes near previous best
    if prev_a is not None:
        np_ = prev_a.size
        raw_w = [1.0 / (0.3 + abs(nv - np_) / 512.0 + nv / 2048.0) for nv in n_list]
    else:
        raw_w = [1.0 / (0.3 + nv / 512.0) for nv in n_list]
    total_w = sum(raw_w)
    weights = [w / total_w for w in raw_w]
    
    for idx_n, nv in enumerate(n_list):
        if time.monotonic() > deadline:
            break
        
        time_n = weights[idx_n] * total_time * 0.80
        print(f"[n={nv}] allocated time: {time_n:.1f}s")
        
        # Generate initializations
        inits = []
        if prev_a is not None:
            inits.append(init_resampled(prev_a, nv, rng))
        inits.append(init_u_shape(nv, alpha=1.5))
        inits.append(init_u_shape(nv, alpha=3.0))
        inits.append(init_v_shape(nv, alpha=1.0))
        inits.append(init_two_peak(nv, rng))
        inits.append(init_fringed(nv, rng))
        # Random
        a_rand = np.abs(rng.standard_normal(nv)) + 0.1
        inits.append(a_rand / a_rand.sum())
        
        n_inits = len(inits)
        time_per_init = time_n / n_inits
        
        for ii, a_init in enumerate(inits):
            if time.monotonic() > deadline:
                break
            
            tb = min(time_per_init, deadline - time.monotonic() - 2.0)
            if tb < 5.0:
                break
            
            # Alternate between mirror descent and gradient descent
            if ii % 2 == 0:
                a_best, sc_best = mirror_descent_optimize(a_init, tb, rng, use_surgery=True)
            else:
                a_best, sc_best = gradient_descent_optimize(a_init, tb, rng)
            
            tag = "MIRROR" if ii % 2 == 0 else "GRAD"
            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = a_best.copy()
                print(f"  [{tag}] n={nv}, init={ii}, score={sc_best:.6f} *** NEW BEST ***")
            else:
                print(f"  [{tag}] n={nv}, init={ii}, score={sc_best:.6f}")
    
    # Also try mirror descent on the previous best directly (if length matches or resample)
    if prev_a is not None and time.monotonic() < deadline - 60.0:
        tb = min(120.0, deadline - time.monotonic() - 30.0)
        if tb > 10.0:
            print(f"[prev] Optimizing previous best (n={prev_a.size}) for {tb:.1f}s")
            a_b, sc_b = mirror_descent_optimize(prev_a, tb, rng, use_surgery=True)
            if sc_b < best_overall_sc:
                best_overall_sc = sc_b
                best_overall = a_b.copy()
                print(f"[prev] Improved to {sc_b:.6f} *** NEW BEST ***")
            else:
                print(f"[prev] Best so far: {sc_b:.6f}")
    
    # Final polish phase
    if best_overall is not None and time.monotonic() < deadline - 20.0:
        remain = deadline - time.monotonic() - 5.0
        if remain > 15.0:
            print(f"[polish] remaining={remain:.1f}s, n={best_overall.size}")
            
            # Fine polish
            pol_time = min(remain * 0.7, 180.0)
            a_p, sc_p = fine_polish(best_overall, pol_time, rng)
            if sc_p < best_overall_sc:
                best_overall_sc = sc_p
                best_overall = a_p.copy()
                print(f"[polish] Improved to {sc_p:.6f} *** NEW BEST ***")
            else:
                print(f"[polish] No improvement: {sc_p:.6f}")
            
            # One more round of very short optimizations from the polished best
            if time.monotonic() < deadline - 15.0:
                remain2 = deadline - time.monotonic() - 5.0
                if remain2 > 10.0:
                    a_p2, sc_p2 = mirror_descent_optimize(best_overall, remain2 * 0.5, rng, use_surgery=True)
                    if sc_p2 < best_overall_sc:
                        best_overall_sc = sc_p2
                        best_overall = a_p2.copy()
                        print(f"[polish2] Improved to {sc_p2:.6f} *** NEW BEST ***")
    
    # Fallback
    if best_overall is None:
        best_overall = init_u_shape(256, alpha=2.0)
        best_overall_sc = normed_score(best_overall)
    
    # Final answer: use normalized values (sum = 1 is fine, evaluator handles it)
    result = [float(max(0.0, x)) for x in best_overall]
    
    # Verify with evaluator
    final_sc = float(evaluate_sequence(result))
    print(f"[final] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")
    
    return result
