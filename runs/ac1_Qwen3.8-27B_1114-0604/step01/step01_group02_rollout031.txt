def compute_pressure(a, n, m, alpha, mu):
    L = 2*n -1
    A = np.fft.rfft(a, m)
    A2 = A*A
    c = np.fft.irfft(A2, m)[:L]
    cmax = float(c.max())
    if not np.isfinite(cmax) or cmax <= 0:
       return np.inf, np.zeros(n), cmax
    cmean = 1.0/float(L)  # since sum 1
    range_c = max(cmax - cmean, 0.02*cmax, 1e-300)
    T = max(alpha * range_c, 1e-300)
    if T <= 1e-300:
       p = np.zeros(L); p[int(np.argmax(c))]=1.0
    else:
       u = (c - cmax)/T
       # clip positive to 0 to avoid exp>1 due round
       np.minimum(u, 0.0, out=u)
       e = np.exp(u)
       s = float(e.sum())
       if not np.isfinite(s) or s <= 0:
           p = np.zeros(L); p[int(np.argmax(c))]=1.0
       else:
           p = e / s
    P = np.fft.rfft(p, m)
    AR = np.fft.rfft(a[::-1], m)
    W = P + mu * A2
    r = np.fft.irfft(W * AR, m)[n-1:2*n-1]
    return 2.0*n*cmax, r, cmax
```

```python
"""
Optimizer for non-negative step functions minimizing

    2 * n * max_k convolve(a, a)[k] / (sum(a))^2

The program searches over several dimensions using a hybrid strategy:

1. Interpolation/multi-start from the previously known construction, when
   available.
2. Fast iterative Fourier transform (IFT) examples produced by imposing a
   flat convolution spectrum.
3. Long smoothed-gradient runs in log-space on the simplex.  The gradient is
   composed from a clipped log-sum-exp of the convolution peaks plus an
   L2 flattening term, which avoids the simple uniform stationary point and
   helps flatten secondary peaks.
4. Final polishing on the best incumbent.

All helpers are top-level and there are no nested functions or lambdas.
"""

import numpy as np
import time


def _next_pow2_atleast(k):
    """Smallest power of two >= k, with k treated as a positive integer."""
    k = int(k)
    if k <= 1:
        return 1
    return 1 << (k - 1).bit_length()


def _fast_score(a, fft_len=None):
    """Scale-invariant exact score from a real array, using FFT convolution."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.inf
    if fft_len is None:
        fft_len = _next_pow2_atleast(2 * n - 1)
    A = np.fft.rfft(a, fft_len)
    c = np.fft.irfft(A * A, fft_len)[:2 * n - 1]
    cmax = float(c.max())
    if not np.isfinite(cmax):
        return np.inf
    return float(2.0 * n * cmax / (s * s))


def _compute_pressure(a, n, fft_len, alpha, mu):
    """
    Return (score, pressure) for a sum-normalized sequence.

    pressure_i is proportional to the gradient of a surrogate:
        smooth max of convolution + mu * sum(convolution^2)
    It is computed as a weighted self-correlation.
    """
    L = 2 * n - 1
    A = np.fft.rfft(a, fft_len)
    A2 = A * A
    c = np.fft.irfft(A2, fft_len)[:L]

    cmax = float(c.max())
    if not np.isfinite(cmax) or cmax <= 0.0:
        return np.inf, np.zeros(n, dtype=np.float64)

    cmean = float(c.mean())
    range_c = max(cmax - cmean, 0.02 * cmax, 1e-300)
    T = max(alpha * range_c, 1e-300)

    if T <= 1e-300:
        p = np.zeros(L, dtype=np.float64)
        p[int(np.argmax(c))] = 1.0
    else:
        u = (c - cmax) / T
        np.minimum(u, 0.0, out=u)
        e = np.exp(u)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0.0:
            p = np.zeros(L, dtype=np.float64)
            p[int(np.argmax(c))] = 1.0
        else:
            p = e / se

    P = np.fft.rfft(p, fft_len)
    AR = np.fft.rfft(a[::-1], fft_len)

    # W is the rFFT of p + mu*c.  Its convolution with reversed a gives the
    # score derivative up to a global constant.
    W = P + mu * A2
    r = np.fft.irfft(W * AR, fft_len)[n - 1:2 * n - 1]
    r = np.ascontiguousarray(r.real, dtype=np.float64)

    # For sum(a) = 1 the exact score is 2 n max(c).
    return float(2.0 * n * cmax), r


def _z_to_a(z):
    """Stable softmax: map unnormalized log heights to a probability vector."""
    z = np.asarray(z, dtype=np.float64).ravel()
    n = z.size
    if n == 0:
        return np.array([], dtype=np.float64)
    if not np.all(np.isfinite(z)):
        return np.ones(n, dtype=np.float64) / float(n)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def _a_to_z(a):
    """Map a positive normalized sequence to centered log coordinates."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 1e-300)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.zeros(n, dtype=np.float64)
    a = a / s
    z = np.log(a)
    z -= float(np.mean(z))
    return z


def _prepare_a(arr, n):
    """
    Convert a candidate array to a positive sum-one array of length n,
    using linear interpolation if the length changes.
    """
    n = int(n)
    arr = np.asarray(arr, dtype=np.float64).ravel()
    if arr.size == 0 or n <= 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)

    if arr.size == n:
        vals = arr.copy()
    elif arr.size == 1:
        vals = np.full(n, float(arr[0]), dtype=np.float64)
    else:
        x_old = np.linspace(0.0, 1.0, arr.size, dtype=np.float64)
        x_new = np.linspace(0.0, 1.0, n, dtype=np.float64)
        vals = np.interp(x_new, x_old, arr)

    vals = np.maximum(vals, 1e-300)
    s = float(vals.sum())
    if not np.isfinite(s) or s <= 0.0:
        vals = np.ones(n, dtype=np.float64)
    else:
        vals = vals / s
    return vals


def _edge_shape_a(n, mode, rng):

    n = int(n)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)

    if mode == 0:
        a = 1.0 + 0.8 * np.cos(2.0 * np.pi * x) + 0.4 * np.abs(2.0 * x - 1.0)
    elif mode == 1:

    elif mode == 2:

            1.0
            + 1.5 * np.exp(-((x - 0.10) ** 2) / (2.0 * 0.04 ** 2))
            + 1.5 * np.exp(-((x - 0.90) ** 2) / (2.0 * 0.04 ** 2))
            + 0.4 * np.cos(4.0 * np.pi * x)

    else:

        per = max(1, n // nb)
        if rng is None:
            block = 0.4 + 0.8 * np.random.random(nb)
        else:
            block = 0.4 + 0.8 * rng.random(nb)
        a = np.repeat(block, per)
        if a.size < n:
            a = np.concatenate([a, block[:n - a.size]])
        a = a[:n]
        a = a + 0.4 * np.abs(2.0 * x - 1.0)

    if rng is not None:
        a = a * np.exp(0.25 * rng.standard_normal(n))

    a = np.maximum(a, 1e-300)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a = np.ones(n, dtype=np.float64)
    else:
        a = a / s
    return a


def _adam_update(z, g, adam_m, adam_v, t, lr):
    """In-place Adam step for z, with gradient g."""
    b1 = 0.9
    b2 = 0.999
    eps = 1e-8

    t1 = b1 ** float(t)
    t2 = b2 ** float(t)

    adam_m += (1.0 - b1) * (g - adam_m)
    adam_v += (1.0 - b2) * (g * g - adam_v)

    mh = adam_m / (1.0 - t1)
    vh = adam_v / (1.0 - t2)
    z -= lr * mh / (np.sqrt(vh) + eps)


def run_gradient(n, z0, seg_start, seg_end, rng, tag,
                 lr0=0.05, temp0=0.25, temp1=0.02,
                 mu0=0.6, mu1=0.05, patience=1000):
    """
    Run time-bounded smoothed Adam in log-space on the simplex.
    """
    n = int(n)
    seg_start = float(seg_start)
    seg_end = float(seg_end)
    if seg_end <= time.monotonic():
        a = _z_to_a(z0)
        return _fast_score(a), a, z0

    fft_len = _next_pow2_atleast(3 * n - 2)

    z = np.asarray(z0, dtype=np.float64).ravel().copy()
    a = _z_to_a(z)

    best_score = _fast_score(a, fft_len)
    best_a = a.copy()
    best_z = z.copy()

    adam_m = np.zeros(n, dtype=np.float64)
    adam_v = np.zeros(n, dtype=np.float64)
    t = 0
    no_improve = 0
    restart_count = 0
    patience_cur = int(max(80, patience))

    while time.monotonic() < seg_end:
        now = time.monotonic()
        frac = (now - seg_start) / max(1e-12, seg_end - seg_start)
        if frac < 0.0:
            frac = 0.0
        elif frac > 1.0:
            frac = 1.0

        alpha = temp0 + (temp1 - temp0) * frac
        mu = mu0 + (mu1 - mu0) * frac
        lr = lr0 * (1.0 - 0.7 * frac)
        if lr < 0.005:
            lr = 0.005

        z -= float(np.mean(z))
        np.clip(z, -12.0, 12.0, out=z)
        a = _z_to_a(z)

        score, r = _compute_pressure(a, n, fft_len, alpha, mu)

        if np.isfinite(score) and score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience_cur:
            noise_scale = min(2.0, 0.3 + 0.3 * float(restart_count) + 0.5 * rng.random())
            z = best_z + noise_scale * rng.standard_normal(n)
            adam_m.fill(0.0)
            adam_v.fill(0.0)
            t = 0
            no_improve = 0
            restart_count += 1
            patience_cur = max(80, int(patience_cur * 0.85))
            continue

        mean_r = float(np.mean(r))
        g = r - mean_r

        if not np.all(np.isfinite(g)):
            z = best_z.copy()
            adam_m.fill(0.0)
            adam_v.fill(0.0)
            t = 0
            no_improve = patience_cur
            continue

        if float(np.max(np.abs(g))) < 1e-15:
            no_improve = patience_cur
            continue

        t += 1
        _adam_update(z, g, adam_m, adam_v, t, lr)

    return best_score, best_a, best_z


def optimize_length(n, phase_end, rng, current_a, tag):
    """
    Multi-start screen followed by a long refinement at a fixed length.
    """
    n = int(n)
    phase_start = time.monotonic()
    phase_end = float(phase_end)

    if phase_end - phase_start < 20.0:
        a = _prepare_a(current_a, n) if current_a is not None else _edge_shape_a(n, 0, rng)
        if a is None:
            a = _edge_shape_a(n, 0, rng)
        return _fast_score(a), a

    seed_frac = 0.20 if n >= 4096 else 0.25
    seed_end = phase_start + seed_frac * (phase_end - phase_start)

    seeds = []
    if current_a is not None:
        pa = _prepare_a(current_a, n)
        if pa is not None:
            seeds.append(pa)
    seeds.append(_edge_shape_a(n, 0, rng))
    seeds.append(_edge_shape_a(n, 1, rng))
    seeds.append(_edge_shape_a(n, 2, rng))

    best_score = np.inf
    best_a = None

    for idx, a in enumerate(seeds):
        now = time.monotonic()
        if now >= seed_end - 2.0:
            break
        remaining_seeds = max(1, len(seeds) - idx)
        chunk_end = min(seed_end, now + max(5.0, (seed_end - now) / float(remaining_seeds)))
        z = _a_to_z(a)
        sc, aa, _zz = run_gradient(
            n, z, now, chunk_end, rng, tag + "_seed%d" % idx,
            lr0=0.10, temp0=0.70, temp1=0.20,
            mu0=1.2, mu1=0.5, patience=250
        )
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = aa

    if best_a is None:
        best_a = seeds[0]
        best_score = _fast_score(best_a)

    now = time.monotonic()
    if now < phase_end - 20.0:
        z = _a_to_z(best_a)
        sc, aa, _zz = run_gradient(
            n, z, now, phase_end, rng, tag + "_refine",
            lr0=0.05, temp0=0.25, temp1=0.02,
            mu0=0.6, mu1=0.05, patience=1000
        )
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = aa

    return best_score, best_a


def run_ift(n, a0, seg_end, rng, check_every=100):
    """
    Short IFT phase imposing a flat convolution spectrum.
    Returns a candidate useful as an initial seed for gradient polishing.
    """
    n = int(n)
    seg_end = float(seg_end)
    if time.monotonic() >= seg_end - 5.0:
        return np.inf, None

    L = 2 * n - 1
    M = _next_pow2_atleast(L)

    target_c = np.ones(L, dtype=np.float64) / float(L)
    D = np.fft.rfft(target_c, M)
    target_mag = np.sqrt(np.abs(D))
    target_mag[0] = 1.0
    target_mag += 1e-9 * float(target_mag.max() + 1.0)

    a = _prepare_a(a0, n)
    if a is None:
        a = _edge_shape_a(n, 0, rng)

    s = float(a.sum())
    if s <= 0.0:
        a = np.ones(n, dtype=np.float64) / float(n)
    else:
        a = a / s

    best_score = _fast_score(a, M)
    best_a = a.copy()

    it = 0
    damping = 0.55

    while time.monotonic() < seg_end:
        A = np.fft.rfft(a, M)
        amp = np.abs(A) + 1e-12
        phase = A / amp
        A_new = phase * target_mag
        A_new[0] = target_mag[0] + 0.0j

        a_new = np.fft.irfft(A_new, M)[:n]
        a_new = np.maximum(a_new, 0.0)

        if rng is not None and rng.random() < 0.01:
            a_new = a_new + 1e-6 * rng.standard_normal(n)
            a_new = np.maximum(a_new, 0.0)

        s = float(a_new.sum())
        if not np.isfinite(s) or s <= 0.0:
            break
        a_new = a_new / s

        a = (1.0 - damping) * a + damping * a_new
        a = np.maximum(a, 0.0)
        s = float(a.sum())
        if not np.isfinite(s) or s <= 0.0:
            break
        a = a / s

        it += 1
        if it % check_every == 0:
            sc = _fast_score(a, M)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

    sc = _fast_score(a, M)
    if np.isfinite(sc) and sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence within the wall-clock budget.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = max(60.0, float(budget_s))
    deadline = start + total - 15.0

    best_score = np.inf
    best_a = None
    seed_a = None
    parent_out = None

    # Load any previous construction if available.
    try:
        parent_raw = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        parent_raw = parent_raw[np.isfinite(parent_raw)]
        parent_raw = np.maximum(parent_raw, 0.0)
        if parent_raw.size > 0:
            parent_out = [float(x) for x in parent_raw]
            opt_parent_len = min(parent_raw.size, 16384)
            pa = _prepare_a(parent_raw, opt_parent_len)
            if pa is not None:
                ps = _fast_score(pa)
                if np.isfinite(ps):
                    best_score = ps
                    best_a = pa
                    seed_a = pa
    except Exception:
        parent_out = None
        best_a = None

    if best_a is None:
        base_n = 2048
        best_a = _edge_shape_a(base_n, 0, rng)
        best_score = _fast_score(best_a)
        seed_a = best_a

    print("initial best = %.7f, len = %d" % (best_score, best_a.size))

    # Small incumbent polish if it is short and there is ample time.
    if best_a.size <= 2048 and (deadline - time.monotonic()) > 2100.0:
        low_end = time.monotonic() + 250.0
        sc, aa = optimize_length(int(best_a.size), low_end, rng, seed_a, "low")
        if np.isfinite(sc) and aa is not None:
            if sc < best_score:
                best_score = sc
                best_a = aa
                seed_a = aa
            elif sc < best_score + 0.05:
                seed_a = aa
        print("low end best = %.7f" % best_score)

    # Medium length: 4096.
    if time.monotonic() < deadline:
        mid_time = min(550.0, max(120.0, (deadline - time.monotonic()) * 0.25))
        mid_end = time.monotonic() + mid_time
        sc, aa = optimize_length(4096, mid_end, rng, seed_a if seed_a is not None else best_a, "mid")
        if np.isfinite(sc) and aa is not None:
            if sc < best_score:
                best_score = sc
                best_a = aa
                seed_a = aa
            elif sc < best_score + 0.03:
                seed_a = aa
        print("mid end best = %.7f" % best_score)

    # High length: 8192, with optional IFT seed generation.
    if time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem > 1600.0:
            ilt_time = min(70.0, rem * 0.04)
            if ilt_time > 30.0:
                a_high = _prepare_a(seed_a if seed_a is not None else best_a, 8192)
                if a_high is None:
                    a_high = _edge_shape_a(8192, 0, rng)
                sc_ilt, a_ilt = run_ift(8192, a_high, time.monotonic() + ilt_time, rng)
                if np.isfinite(sc_ilt) and a_ilt is not None:
                    if sc_ilt < best_score:
                        best_score = sc_ilt
                        best_a = a_ilt
                        seed_a = a_ilt
                    elif sc_ilt < best_score + 0.02:
                        seed_a = a_ilt
                    print("ift seed best = %.7f" % best_score)

        rem = deadline - time.monotonic()
        if rem > 1500.0:
            reserve = 650.0
        else:
            reserve = 300.0
        high_time = min(1300.0, max(200.0, rem - reserve))
        high_end = time.monotonic() + high_time
        sc, aa = optimize_length(8192, high_end, rng, seed_a if seed_a is not None else best_a, "high")
        if np.isfinite(sc) and aa is not None:
            if sc < best_score:
                best_score = sc
                best_a = aa
                seed_a = aa
            elif sc < best_score + 0.02:
                seed_a = aa
        print("high end best = %.7f" % best_score)

    # Optional exploration at 16384 if we are close to the target already.
    if time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem > 450.0 and best_score < 1.55:
            ultra_time = min(300.0, rem - 150.0)
            if ultra_time > 80.0:
                n_ultra = 16384
                a_ultra = _prepare_a(seed_a if seed_a is not None else best_a, n_ultra)
                if a_ultra is None:
                    a_ultra = _edge_shape_a(n_ultra, 0, rng)
                z_ultra = _a_to_z(a_ultra)
                ultra_end = time.monotonic() + ultra_time
                sc, aa = run_gradient(
                    n_ultra, z_ultra, time.monotonic(), ultra_end, rng, "ultra",
                    lr0=0.045, temp0=0.22, temp1=0.02,
                    mu0=0.55, mu1=0.06, patience=900
                )
                if np.isfinite(sc) and aa is not None and sc < best_score:
                    best_score = sc
                    best_a = aa
                    seed_a = aa
                print("ultra end best = %.7f" % best_score)

    # Final polish on the best incumbent.
    if time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem > 60.0:
            n_final = int(best_a.size)
            n_final = min(max(512, n_final), 16384)
            a_final = _prepare_a(best_a, n_final)
            if a_final is None:
                a_final = _edge_shape_a(n_final, 0, rng)
            z_final = _a_to_z(a_final)
            sc, aa = run_gradient(
                n_final, z_final, time.monotonic(), deadline, rng, "polish",
                lr0=0.02, temp0=0.12, temp1=0.01,
                mu0=0.25, mu1=0.03, patience=2000
            )
            if np.isfinite(sc) and aa is not None and sc < best_score:
                best_score = sc
                best_a = aa

    if best_a is None or best_a.size == 0:
        best_a = _edge_shape_a(2048, 0, rng)

    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.where(np.isfinite(best_a), best_a, 0.0)
    best_a = np.maximum(best_a, 0.0)
    s = float(best_a.sum())
    if not np.isfinite(s) or s <= 0.0:
        best_a = np.ones(2048, dtype=np.float64)
    else:
        best_a = best_a / s

    out = [float(x) for x in best_a]
    try:
        val = float(evaluate_sequence(out))
    except Exception:
        val = np.inf

    # Fallback to the previous construction if it is short enough and better.
    if parent_out is not None and len(parent_out) <= 10000:
        try:
            p = np.asarray(parent_out, dtype=np.float64).ravel()
            p = p[np.isfinite(p)]
            p = np.maximum(p, 0.0)
            if p.size > 0:
                pout = [float(x) for x in p]
                pval = float(evaluate_sequence(pout))
                if np.isfinite(pval) and (not np.isfinite(val) or pval < val):
                    out = pout
                    val = pval
        except Exception:
            pass

    if not np.isfinite(val):
        out = [1.0] * 2048
        try:
            val = float(evaluate_sequence(out))
        except Exception:
            val = np.inf

    print("final best = %.7f, verified = %.7f, len = %d, elapsed = %.1f s"
          % (best_score, val, len(out), time.monotonic() - start))
    return out
