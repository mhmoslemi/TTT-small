# --- Convolution helpers ---
def fast_conv(a):
    n = len(a)
    M = 2*n - 1
    M_fft = 1 << (M-1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A*A, M_fft)[:M]

def score(a):

    c = fast_conv(a)
    S = a.sum()
    if S < 0.01:
        return np.inf
    return 2.0*n*max(c)/(S*S)

    if top_k is not None and top_k < len(c):

        weights = exps / exps.sum()

        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size > 50:
            parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    if parent is not None:
        n = min(len(parent), 8192)
        a_init = parent[:n].copy()
    else:
        n = 2048
        a_init = np.ones(n)

    # Normalize to sum = sqrt(2n)
    S0 = a_init.sum()
    if S0 < 0.01:
        a_init = np.ones(n)
        S0 = n
    a_init = a_init * (np.sqrt(2.0 * n) / S0)

    c = fast_conv(a_init)
    best_score = compute_score(n, c, a_init.sum())
    best_a = a_init.copy()
    print(f"n={n}, initial score={best_score:.6f}")

    best_patience = 0
    max_patience = 200

    def check_and_update(a, label):
        nonlocal best_score, best_a
        c_new = fast_conv(a)
        S_new = a.sum()
        if S_new < 0.01:
            return
        sc = compute_score(n, c_new, S_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] {best_score:.6f} (t={time.monotonic()-start:.1f}s)")

    # Phase structure
    phase1_end = start + 0.50 * (total - 5.0)
    phase2_end = start + 0.80 * (total - 5.0)

    # === Phase 1: Gradient optimization with diverse settings ===
    a = best_a.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    no_improve = 0

    config_list = [
        # (topk, tau, lr, label)
        (8, 0.05, 0.5, "explore_stiff"),
        (16, 0.1, 0.4, "explore_med"),
        (32, 0.2, 0.3, "explore_soft"),
        (48, 0.4, 0.2, "refine_soft"),
    ]

    seg_duration = (phase1_end - start) / len(config_list)
    for ci, (topk, tau, lr_init, label) in enumerate(config_list):
        seg_start = time.monotonic()
        seg_end = min(phase1_end, start + (ci + 1) * seg_duration)
        lr = lr_init
        no_improve = 0
        t_adam = 0
        # Reset momentum for fresh start on new config
        if ci > 0:
            m[:] = 0.0
            v[:] = 0.0
            a = best_a.copy()

        while time.monotonic() < seg_end:
            c = fast_conv(a)
            S = a.sum()
            sc = compute_score(n, c, S)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
                if int(best_score * 10000) % 100 == 0:
                    print(f"  [{label}] {best_score:.6f} (t={time.monotonic()-start:.1f}s)")
            else:
                no_improve += 1

            # Adaptive learning rate decay
            lr = lr_init * (0.995 ** no_improve)
            if lr < 0.005:
                lr = 0.005

            # Restart if stuck
            if no_improve >= max_patience:
                scale = 0.08 + 0.12 * rng.random()
                a = best_a * (1.0 + scale * rng.standard_normal(n))
                clamp_nonneg(a)
                if a.sum() < 0.01:
                    a = best_a.copy()
                m[:] = 0.0
                v[:] = 0.0
                t_adam = 0
                no_improve = 0
                max_patience = max(50, max_patience // 2)
                continue

            # Compute gradient
            idx, wts = get_topk_weights(c, topk, tau)
            if idx.size == 0:
                break
            grad_c = gradient_topk(a, idx, wts)
            M_w = float(np.dot(c[idx], wts))
            S2 = S * S
            S3 = S2 * S
            grad = (2.0 * n) * (grad_c / S2 - 2.0 * M_w / S3)

            t_adam += 1
            adam_step(a, grad, m, v, t_adam, lr)
            clamp_nonneg(a)

            # Occasionally normalize (helps with scale invariance)
            if t_adam % 50 == 0:
                S = a.sum()
                if S > 0.01:
                    a *= (np.sqrt(2.0 * n) / S)

    print(f"Phase 1 done: {best_score:.6f} (t={time.monotonic()-start:.1f}s)")

    # === Phase 2: LP refinement + gradient ===
    a = best_a.copy()
    m[:] = 0.0
    v[:] = 0.0
    t_adam = 0
    no_improve = 0
    lp_count = 0
    last_lp_time = 0.0

    # Determine if LP is feasible for this n
    use_lp = n <= 4096

    while time.monotonic() < phase2_end:
        # LP step every ~40 seconds
        do_lp = (use_lp and time.monotonic() - last_lp_time > 40.0
                 and no_improve > 30)

        if do_lp:
            c_curr = fast_conv(a)
            S_curr = a.sum()
            M_curr = float(np.max(c_curr))
            g = solve_lp(a, M_curr, time_limit=25.0)
            last_lp_time = time.monotonic()
            if g is not None:
                lp_count += 1
                # Rescale g to match sum
                g_sum = g.sum()
                if g_sum > 1e-10:
                    g = g * (S_curr / g_sum)
                # Mix
                for t_mix in [0.5, 0.3, 0.15, 0.05]:
                    a_trial = (1.0 - t_mix) * best_a + t_mix * g
                    clamp_nonneg(a_trial)
                    check_and_update(a_trial, f"lp_{lp_count}_t{t_mix}")
                a = best_a.copy()
                # Also try pure LP direction scaled
                a_trial = g * (np.sqrt(2.0 * n) / g_sum) if g_sum > 1e-10 else best_a.copy()
                check_and_update(a_trial, f"lp_{lp_count}_pure")
                m[:] = 0.0
                v[:] = 0.0
                t_adam = 0

        # Gradient step
        c = fast_conv(a)
        S = a.sum()
        sc = compute_score(n, c, S)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 150:
            scale = 0.05 + 0.1 * rng.random()
            a = best_a * (1.0 + scale * rng.standard_normal(n))
            clamp_nonneg(a)
            m[:] = 0.0
            v[:] = 0.0
            t_adam = 0
            no_improve = 0
            continue

        lr = 0.15 * (0.997 ** min(no_improve, 200))
        lr = max(lr, 0.005)
        topk = min(64, 16 + no_improve // 10)
        tau = 0.15 + 0.01 * min(no_improve, 100)

        idx, wts = get_topk_weights(c, topk, tau)
        if idx.size == 0:
            break
        grad_c = gradient_topk(a, idx, wts)
        M_w = float(np.dot(c[idx], wts))
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (grad_c / S2 - 2.0 * M_w / S3)

        t_adam += 1
        adam_step(a, grad, m, v, t_adam, lr)
        clamp_nonneg(a)

        if t_adam % 100 == 0:
            S = a.sum()
            if S > 0.01:
                a *= (np.sqrt(2.0 * n) / S)

    print(f"Phase 2 done: {best_score:.6f}, LP calls={lp_count} (t={time.monotonic()-start:.1f}s)")

    # === Phase 3: Final polish ===
    a = best_a.copy()
    m[:] = 0.0
    v[:] = 0.0
    t_adam = 0
    no_improve = 0

    while time.monotonic() < deadline:
        c = fast_conv(a)
        S = a.sum()
        sc = compute_score(n, c, S)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= 100:
            # Small perturbation
            a = best_a * (1.0 + 0.03 * rng.standard_normal(n))
            clamp_nonneg(a)
            m[:] = 0.0
            v[:] = 0.0
            t_adam = 0
            no_improve = 0
            continue

        lr = max(0.003, 0.02 * (0.999 ** no_improve))
        topk = min(128, 32 + no_improve // 5)
        tau = 0.3 + 0.005 * min(no_improve, 200)

        idx, wts = get_topk_weights(c, topk, tau)
        if idx.size == 0:
            break
        grad_c = gradient_topk(a, idx, wts)
        M_w = float(np.dot(c[idx], wts))
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (grad_c / S2 - 2.0 * M_w / S3)

        t_adam += 1
        adam_step(a, grad, m, v, t_adam, lr)
        clamp_nonneg(a)

    # Verify with evaluator
    out = [float(x) for x in best_a]
    verified = float(evaluate_sequence(out))
    if not np.isfinite(verified):
        # Fallback
        out = [float(x) for x in np.ones(n) * 0.1]
        verified = float(evaluate_sequence(out))

    print(f"FINAL: internal={best_score:.6f}, verified={verified:.6f}, n={n}, time={time.monotonic()-start:.1f}s")
    return out
