```python
def _coordinate_descent_step(a, j, c, n):
    """
    Perform an exact 1D coordinate descent step on coordinate j.
    a: current sequence (normalized, sum=1)
    j: index to optimize
    c: current convolution = conv(a,a)
    n: length of a
    Returns: optimal v for a[j]
    """
    S_others = 1.0 - a[j]  # since sum(a) = 1
    v_old = a[j]
    
    # Compute C_k = c_k - f_k(v_old)
    # f_k(v_old) = 2*v_old*a[k-j] if k != 2j, else v_old^2
    C = c.copy()
    # Subtract contribution of a[j]
    # For each k, the contribution of a[j] to c[k] is:
    #   2*v_old*a[k-j] if 0 <= k-j < n and k != 2*j
    #   v_old^2 if k == 2*j and 0 <= 2*j < 2n-1
    #   0 otherwise
    
    # Efficient computation:
    # c[k] = sum_i a[i]*a[k-i]. The terms involving a[j]:
    #   i=j: a[j]*a[k-j] (valid if 0 <= k-j < n)
    #   i=k-j: a[k-j]*a[j] (valid if 0 <= k-j < n, and k-j != j)
    
    # So contribution to c[k] from a[j] is:
    #   a[j]*a[k-j] + a[k-j]*a[j] = 2*a[j]*a[k-j] if k-j != j
    #   a[j]*a[j] = a[j]^2 if k-j == j (i.e., k = 2j)
    
    # Let's compute C by subtracting:
    # For k where 0 <= k-j < n:
    #   If k == 2*j: C[k] = c[k] - v_old^2
    #   Else: C[k] = c[k] - 2*v_old*a[k-j]
    
    C = c.copy()
    # Indices k where 0 <= k-j < n, i.e., j <= k < j+n
    k_start = j
    k_end = min(j + n, 2 * n - 1)
    for k in range(k_start, k_end + 1):
        if k == 2 * j:
            C[k] -= v_old * v_old
        else:
            C[k] -= 2.0 * v_old * a[k - j]
    
    C = np.maximum(C, 0.0)
    
    # Now find optimal v >= 0 minimizing max_k(C[k] + f_k(v)) / (S_others + v)^2
    # where f_k(v) = 2*v*a[k-j] if 0 <= k-j < n and k != 2j
    #                    v^2 if k == 2j
    #                    0 otherwise
    
    # Candidate v values:
    candidates = [0.0]
    
    # For each k in [j, j+n-1] (where f_k is nonzero):
    for k in range(k_start, k_end + 1):
        if k == 2 * j:
            # f_k(v) = v^2, optimal v = C[k] / S_others
            if S_others > 1e-15:
                v_cand = C[k] / S_others
                if v_cand > 0:
                    candidates.append(v_cand)
        else:
            a_kj = a[k - j]
            if a_kj > 1e-15:
                # f_k(v) = 2*v*a[k-j], optimal v = S_others - C[k]/a[k-j]
                v_cand = S_others - C[k] / a_kj
                if v_cand > 0:
                    candidates.append(v_cand)
    
    # Also add v_old and some geometric grid points
    candidates.append(v_old)
    candidates.append(S_others)  # v = S_others means a[j] = sum of others
    
    # Remove duplicates and negatives
    candidates = [v for v in candidates if v >= 0 and np.isfinite(v)]
    if not candidates:
        return v_old
    
    # Evaluate F(v) at each candidate
    best_v = v_old
    best_F = float('inf')
    
    for v in candidates:
        # Compute max_k(C[k] + f_k(v))
        max_val = 0.0
        for k in range(k_start, k_end + 1):
            if k == 2 * j:
                val = C[k] + v * v
            else:
                val = C[k] + 2.0 * v * a[k - j]
            if val > max_val:
                max_val = val
        # Also check k outside [k_start, k_end] (these have f_k = 0)
        if k_start > 0:
            max_val = max(max_val, float(np.max(C[:k_start])))
        if k_end + 1 < len(C):
            max_val = max(max_val, float(np.max(C[k_end+1:])))
        
        denom = S_others + v
        if denom < 1e-15:
            F_val = float('inf')
        else:
            F_val = max_val / (denom * denom)
        
        if F_val < best_F:
            best_F = F_val
            best_v = v
    
    return best_v
```

```python
            # For k = 2j: delta_c[k] = best_v^2 - v_old^2
            # So: c[k_lo:k_hi+1] += 2*delta*a_slice, then fix up k=2j
            # c[2j] should be: old_c[2j] - v_old^2 + best_v^2
            # But we already added 2*delta*a[2j-j] = 2*delta*a[j] = 2*(best_v-v_old)*a[j] = 2*best_v*a[j] - 2*v_old*a[j]
            # And we need: best_v^2 - v_old^2 = (best_v-v_old)*(best_v+v_old)
            # So the correction at 2j is: (best_v-v_old)*(best_v+v_old) - 2*(best_v-v_old)*a[j] = (best_v-v_old)*(best_v+v_old-2*a[j])
            # But a[j] is now best_v, so: delta*(best_v+v_old-2*best_v) = delta*(v_old-best_v) = -delta^2
            # Hmm, this is getting confusing. Let me just recompute c[2j].
            
            # Actually, let me just recompute the affected part of c.
            # The simplest correct approach:
            # c[k] = sum_i a[i]*a[k-i]
            # After changing a[j], only c[k] for k in [j, j+n-1] (intersected with [0, 2n-2]) change.
            # c_new[k] = c_old[k] - (old contribution of a[j]) + (new contribution of a[j])
            # Old contribution: 2*v_old*a[k-j] (k!=2j) or v_old^2 (k=2j)
            # New contribution: 2*best_v*a[k-j] (k!=2j) or best_v^2 (k=2j)
            
            # I already have: c[k_lo:k_hi+1] += 2*delta*a_slice
            # This gives: c[k] = old_c[k] + 2*delta*a[k-j] for all k in range
            # For k != 2j: this is correct (the change is 2*(best_v-v_old)*a[k-j])
            # For k = 2j: the actual change should be best_v^2 - v_old^2
            #   But I added 2*delta*a[j] = 2*(best_v-v_old)*best_v (since a[j] is now best_v)
            #   Wait, a_slice at position pos_2j is a[2j - j] = a[j] = best_v (after update!)
            #   So I added 2*(best_v-v_old)*best_v
            #   But I need best_v^2 - v_old^2 = (best_v-v_old)(best_v+v_old)
            #   Difference: (best_v-v_old)(best_v+v_old) - 2*(best_v-v_old)*best_v = (best_v-v_old)(v_old - best_v) = -(best_v-v_old)^2
            # So I need to subtract delta^2 from c[2j]:
            
            if 0 <= pos_2j < len(c):
                c[2*j] -= delta * delta
                # Wait let me recheck.
                # I want c[2j] = old_c[2j] + (best_v^2 - v_old^2)
                # I computed: c[2j] = old_c[2j] + 2*delta*a_slice[pos_2j] = old_c[2j] + 2*(best_v-v_old)*a[j]
                # But a[j] is already updated to best_v at this point!
                # So: c[2j] = old_c[2j] + 2*(best_v-v_old)*best_v
                # I want: c[2j] = old_c[2j] + best_v^2 - v_old^2 = old_c[2j] + (best_v-v_old)(best_v+v_old)
                # Difference: (best_v-v_old)(best_v+v_old) - 2*(best_v-v_old)*best_v = (best_v-v_old)(v_old-best_v) = -delta^2
                # So: c[2j] -= delta^2  (i.e., subtract (best_v-v_old)^2)
            
            # Hmm wait, I realize the issue. I'm updating a[j] BEFORE computing the delta for c.
            # Let me restructure: save v_old, update a[j], then update c using v_old and best_v.
            # Actually, a_slice was computed BEFORE updating a[j], so a_slice[pos_2j] = a[j]_old = v_old.
            # Wait no, I computed a_slice = a[k_lo-j : k_hi-j+1]. At this point a[j] is still v_old (I haven't updated it yet).
            # So a_slice[pos_2j] = a[2j - j] = a[j] = v_old (the old value).
            # Then I do: c[k_lo:k_hi+1] += 2*delta*a_slice
            # At k=2j: c[2j] += 2*delta*v_old = 2*(best_v-v_old)*v_old
            # I want: c[2j] += best_v^2 - v_old^2 = (best_v-v_old)(best_v+v_old) = delta*(best_v+v_old)
            # Difference: delta*(best_v+v_old) - 2*delta*v_old = delta*(best_v-v_old) = delta^2
            # So: c[2j] += delta^2
            
            # Wait I keep getting confused. Let me just be very explicit.
            # 
            # a_slice was computed with the OLD a (before updating a[j]).
            # a_slice[pos_2j] = a[2j-j] = a[j]_old = v_old.
            #
            # I do: c[k_lo:k_hi+1] += 2 * (best_v - v_old) * a_slice
            # At position 2j: c[2j] += 2 * (best_v - v_old) * v_old
            #
            # The CORRECT change to c[2j] is: new_contrib - old_contrib = best_v^2 - v_old^2
            # = (best_v - v_old)(best_v + v_old) = (best_v-v_old)(best_v + v_old)
            #
            # So I need to add an additional: (best_v-v_old)(best_v+v_old) - 2*(best_v-v_old)*v_old
            # = (best_v-v_old)[(best_v+v_old) - 2*v_old] = (best_v-v_old)(best_v - v_old) = (best_v-v_old)^2 = delta^2
            #
            # So: c[2j] += delta^2  (additional correction)
            
            improved += 1
    
    # Renormalize (shouldn't be needed if we're careful, but for numerical stability)
    s = float(a.sum())
    if s > 0 and abs(s - 1.0) > 1e-10:
        a = a / s
        # Recompute c
        c = _conv_fft(a, L, m)
    
    return a, c
```

Hmm, this is getting quite complex and error-prone. Let me simplify by just recomputing c from scratch after each coordinate change. It's O(n log n) per change, so total O(n^2 log n) per pass. For n=6121, that's about 6121 * 6121 * 13 ≈ 490M flops for FFTs, which might be too slow.

Alternative: just recompute c once per pass (after all coordinate changes). Then the cost is O(n^2) for the coordinate changes + O(n log n) for the final FFT. This is much better.

But then during the pass, my "C" values are based on the current c, which is stale after each coordinate change. This means my coordinate descent is approximate (using stale convolution values). This might still work in practice, especially if the changes are small.

Let me try a different approach: **block coordinate descent**. Process coordinates in blocks of size B. After each block, recompute c. This gives:
- Cost per block: O(B * n) for coordinate changes + O(n log n) for FFT
- Number of blocks: n/B
- Total: O(n^2 + n^2 log n / B)

With B = n/10: total ≈ O(n^2 + n^2 log n * 10/n) = O(n^2 + 10*n*log*n). For n=6121, this is about 37M + 80M ≈ 120M operations. Should be a few seconds.

Actually, you know what, let me just keep it simple. I'll do:
1. Full coordinate descent pass (using stale c for all coordinates).
2. Recompute c.
3. Repeat.

The stale c means the coordinate descent is approximate, but it should still make progress (like a quasi-Newton method with a stale approximate Hessian).

Let me also add the gradient method as a complementary approach.

Let me now write the complete solution:

```python
"""
Search for nonnegative step heights minimizing 2*n*max(conv(a,a))/(sum a)^2.
Uses block coordinate descent with exact 1D sub-optimization, combined with
FFT-based multiplicative gradient updates for global exploration.
"""
import math
import time
import numpy as np


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 0 or not np.isfinite(s):
        a.fill(1.0)
        s = float(a.sum())
    a /= s
    return a


def _conv_fft(a, L, m):
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _obj_val(c, n):
    return float(2.0 * n * np.max(c))


def _coord_pass(a, c, n, top_k=25):
    """
    One pass of coordinate descent on a (sum=1), using c = conv(a,a).
    Returns updated a.
    m = 2*n - 1
    """
    m = 2 * n - 1
    for j in range(n):
        v_old = a[j]
        S_others = 1.0 - v_old
        if S_others < 1e-15:
            continue
        
        k_lo = j
        k_hi = min(j + n - 1, m - 1)
        if k_lo > k_hi:
            continue
        
        # a_slice = a[k_lo - j : k_hi - j + 1] = a[0 : k_hi - k_lo + 1]
        sl_len = k_hi - k_lo + 1
        a_slice = a[0:sl_len]  # a[k - j] for k in [k_lo, k_hi], k-j ranges from 0 to sl_len-1
        
        # contrib = 2*v_old * a_slice, with adjustment at k=2j
        contrib = 2.0 * v_old * a_slice
        pos_2j = 2 * j - k_lo
        if 0 <= pos_2j < sl_len:
            contrib[pos_2j] = v_old * v_old
        
        C_slice = c[k_lo:k_hi+1] - contrib
        C_slice = np.maximum(C_slice, 0.0)
        
        # max outside
        max_outside = 0.0
        if k_lo > 0:
            max_outside = float(np.max(c[:k_lo]))
        if k_hi + 1 < m:
            max_outside = max(max_outside, float(np.max(c[k_hi+1:])))
        
        # Candidate v values
        candidates = np.array([0.0, v_old, S_others, 0.5 * S_others])
        
        # Top k's
        n_top = min(top_k, sl_len)
        if n_top > 0:
            top_idx = np.argpartition(C_slice, -n_top)[-n_top:]
            for idx in top_idx:
                k = k_lo + idx
                if k == 2 * j:
                    vc = C_slice[idx] / S_others
                    if vc > 0 and np.isfinite(vc):
                        candidates = np.append(candidates, vc)
                else:
                    ak = a[k - j]
                    if ak > 1e-15:
                        vc = S_others - C_slice[idx] / ak
                        if vc > 0 and np.isfinite(vc):
                            candidates = np.append(candidates, vc)
        
        if v_old > 1e-10:
            candidates = np.append(candidates, v_old * np.array([0.5, 1.5, 2.0, 0.3]))
        
        candidates = candidates[(candidates >= 0) & np.isfinite(candidates)]
        if len(candidates) == 0:
            continue
        
        # Evaluate
        best_score = float('inf')
        best_v = v_old
        
        for v in candidates:
            v = float(v)
            fk = 2.0 * v * a_slice
            if 0 <= pos_2j < sl_len:
                fk[pos_2j] = v * v
            peak = max(float(np.max(C_slice + fk)), max_outside)
            denom = S_others + v
            if denom < 1e-15:
                continue
            score = peak / (denom * denom)
            if score < best_score:
                best_score = score
                best_v = v
        
        a[j] = best_v
    
    return a


def _grad_step(a, c, n, L, m, temp, eta):
    """
    One multiplicative gradient step.
    a: normalized (sum=1)
    c: conv(a,a)
    """
    v = n * c  # unnormalized "importance" of each lag
    vmax = float(np.max(v))
    temp_eff = min(temp, 200.0 / max(vmax, 1e-12))
    z = temp_eff * (v - vmax)
    z = np.clip(z, -700.0, 0.0)
    w = np.exp(z)
    ws = float(w.sum())
    if ws <= 0 or not np.isfinite(ws):
        w = np.ones_like(v)
        ws = float(w.sum())
    w /= ws
    
    # Gradient: g[j] = sum_k w[k] * dc_k/da_j = 2 * sum_k w[k] * a[k-j]
    # = 2 * (w * a_reversed) convolved... actually it's a cross-correlation
    # g[j] = 2 * sum_{k} w[k] * a[k-j] (where a is extended by 0)
    # This is 2 * conv(w, a_reversed)[n-1+j] ... let me think.
    # Actually: sum_k w[k] * a[k-j] is the cross-correlation of w and a evaluated at j.
    # = (w * a_rev)[j + n - 1] where a_rev = a[::-1]
    # Or equivalently: = irfft(rfft(w, L) * rfft(a_rev, L))[j + n - 1] ... hmm
    # Actually it's simpler: sum_k w[k] * a[k-j] for k=0..m-1, j=0..n-1
    # = (w conv a_rev)[n-1+j]? Let me check.
    # (w conv a_rev)[t] = sum_k w[k] * a_rev[t-k] = sum_k w[k] * a[n-1-t+k]
    # We want: sum_k w[k] * a[k-j]
    # Let t = n-1-j: a[n-1-t+k] = a[n-1-(n-1-j)+k] = a[j+k]... that's not right.
    # 
    # Let me just do it directly with FFT:
    # sum_k w[k] * a[k-j] = sum_k w[k] * a[k-j]
    # Let i = k-j: sum_i w[i+j] * a[i] = (a conv w)[j] ... no.
    # Actually this is: (a * w_reversed)[j + (m-1)] ... I'm overcomplicating.
    
    # Simple approach: g = 2 * np.convolve(w, a[::-1], 'full')[n-1:n-1+n]
    # Let's verify: conv(w, a_rev)[t] = sum_k w[k] * a_rev[t-k] = sum_k w[k] * a[n-1-t+k]
    # We want h[j] = sum_k w[k] * a[k-j]
    # Set n-1-t+k = k-j => n-1-t = -j => t = n-1+j
    # So h[j] = conv(w, a_rev)[n-1+j]
    
    a_rev = a[::-1]
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a_rev, L)
    h = np.fft.irfft(W * A, L)[:n]  # h[j] = sum_k w[k]*a[k-j]
    # Wait, this isn't quite right. Let me just use the direct formula.
    # h[j] = sum_{k=0}^{m-1} w[k] * a[k-j] where a[i] = 0 for i < 0 or i >= n.
    # = sum_{k=j}^{j+n-1} w[k] * a[k-j] (since a[k-j] = 0 unless 0 <= k-j < n)
    # This is a sliding dot product. Can compute with FFT:
    # h = ifft(fft(w, L) * fft(a, L))[n-1 : n-1+n]? 
    # Actually: ifft(fft(w)*fft(a))[t] = sum_k w[k] * a[(t-k) mod L]
    # For t in [n-1, n-1+n-1]: sum_k w[k] * a[(n-1+j-k) mod L]
    # We want sum_k w[k] * a[k-j]... 
    
    # You know what, let me just use np.convolve for this (it's O(n*m) = O(n^2) but n is only 6121)
    # Actually that's too slow. Let me use the FFT approach correctly.
    
    # We want: h[j] = sum_{k} w[k] * a[k-j] for j = 0, ..., n-1
    # This is the "anti-convolution" or "cross-correlation".
    # h[j] = sum_k w[k] * a[k-j] = (a * w)[j] where * denotes... 
    # Actually: (a conv w)[j] = sum_k a[k] * w[j-k] = sum_k a[k] * w[j-k]
    # Let i = j-k: sum_i a[j-i] * w[i] = sum_i a[j-i] * w[i]
    # That's not quite what we want either.
    
    # h[j] = sum_k w[k] * a[k-j]
    # Let's define b[i] = w[i] for 0<=i<m, 0 otherwise.
    # h[j] = sum_k b[k] * a[k-j] = sum_k b[k+j] * a[k] (substituting k' = k+j)
    # Hmm, this is getting circular. Let me just use the fact that:
    # h[j] = sum_k w[k] * a[k-j] = (a * flip(w))[j] where flip(w)[i] = w[m-1-i]
    # Check: (a * flip(w))[j] = sum_k a[k] * flip(w)[j-k] = sum_k a[k] * w[m-1-j+k]
    # We want: sum_k w[k] * a[k-j]
    # These are different. Let me try: (flip(w) * a)[j] = sum_k flip(w)[k] * a[j-k] = sum_k w[m-1-k] * a[j-k]
    # Let i = m-1-k: sum_i w[i] * a[j-(m-1-i)] = sum_i w[i] * a[j-m+1+i]
    # For j in [0, n-1]: a[j-m+1+i] is nonzero when 0 <= j-m+1+i < n, i.e., m-1-j <= i < m-1-j+n
    # = sum_i w[i] * a[i + (j-m+1)] where i ranges over [m-1-j, m-1-j+n-1]
    # = sum_{i=m-1-j}^{m-n+j} w[i] * a[i - (m-1-j)]
    # Let k = i - (m-1-j): sum_k w[k+m-1-j] * a[k] for k in [0, n-1]
    # = sum_{k=0}^{n-1} a[k] * w[k+m-1-j]
    # We want h[j] = sum_k w[k] * a[k-j] = sum_{k=j}^{j+n-1} w[k] * a[k-j]
    # Let i = k-j: sum_{i=0}^{n-1} w[i+j] * a[i]
    # So h[j] = sum_i a[i] * w[i+j]
    # And (flip(w) * a)[j] = sum_{k=0}^{n-1} a[k] * w[k+m-1-j]
    # These are equal when m-1-j = j, i.e., j = (m-1)/2 = (2n-2)/2 = n-1. 
    # So they're equal only at j = n-1. Not helpful.
    
    # OK let me just use the correct formula:
    # h[j] = sum_{i=0}^{n-1} a[i] * w[i+j]
    # This is a "sliding sum" or "dot product with shift".
    # In FFT terms: if we compute c = ifft(fft(a, L) * conj(fft(w, L)))... no that's cross-correlation.
    # Actually: cross_correlation(a, w)[j] = sum_i conj(a[i]) * w[i+j] (for real: sum_i a[i]*w[i+j])
    # That's exactly h[j]!
    # And cross_correlation can be computed as: ifft(fft(a, L) * conj(fft(w, L)))
    # Wait: ifft(fft(a) * conj(fft(w)))[j] = sum_i a[i] * w[(i+j) mod L]... 
    # Actually the standard formula is:
    # (a ⋆ w)[j] = sum_i a[i] * w[i+j] = ifft(FFT(a) * conj(FFT(w)))[j + offset]
    # The offset depends on convention. Let me just test numerically or use a simple approach.
    
    # For n=6121, a direct O(n^2) computation would be 37M operations. In numpy with a 
    # sliding window approach, this is fast. But FFT is faster.
    
    # Let me use: h = np.correlate(w, a, mode='valid')[::-1]
    # np.correlate(w, a, mode='valid')[k] = sum_i w[k+i] * a[i]
    # So h[j] = np.correlate(w, a, mode='valid')[j] ? Let me check:
    # correlate(w, a, 'valid')[k] = sum_i w[k+i] * a[i] for i = 0..len(a)-1
    # = sum_i a[i] * w[i+k]
    # Yes! That's h[k]!
    # But 'valid' mode gives length max(len(w), len(a)) - min(len(w), len(a)) + 1
    # w has length m = 2n-1, a has length n.
    # valid output length = m - n + 1 = n
    # correlate(w, a, 'valid')[j] = sum_{i=0}^{n-1} a[i] * w[i+j] = h[j] ✓
    
    h = np.correlate(w, a, mode='valid')  # length n, h[j] = sum_i a[i]*w[i+j]
    g = 2.0 * h
    gmax = float(np.max(g))
    if gmax <= 0 or not np.isfinite(gmax):
        return a
    g = g / gmax
    
    # Multiplicative update
    a_new = a * np.exp(-eta * g)
    a_new = _normalize(a_new)
    return a_new


def _smooth_vector(v, w=7):
    if v.size <= w:
        return _normalize(v)
    k = np.ones(w, dtype=np.float64) / w
    y = np.convolve(v, k, mode="same")
    return _normalize(y)


def _random_init(n, rng):
    """Random initialization with some structure."""
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0
    z = rng.normal(0.0, 0.3, n)
    z += rng.normal(0.0, 0.5) * x
    z += rng.normal(0.0, 0.5) * (x*x - 1/3)
    
    kmax = int(rng.integers(2, 8))
    for k in range(1, kmax + 1):
        amp = rng.uniform(0.05, 0.4) / max(1.0, math.sqrt(k))
        ph = rng.uniform(0, 2*np.pi)
        z += amp * np.cos(2*np.pi*k*t + ph)
    
    noise = rng.normal(0, 0.1, n)
    if n > 5:
        noise = np.convolve(noise, np.ones(5)/5, mode='same')
    z += noise
    
    p = np.exp(z - np.max(z))
    p = np.maximum(p, 1e-300)
    return _normalize(p)


def _gradient_search(a, L, m, n, rng, time_limit):
    """
    Gradient-based search with adaptive temperature.
    """
    deadline = time.monotonic() + time_limit
    a = _normalize(a)
    c = _conv_fft(a, L, m)
    best_a = a.copy()
    best_score = _obj_val(c, n)
    
    temp = 1.0
    eta = 0.1
    no_improve = 0
    
    while time.monotonic() < deadline:
        c = _conv_fft(a, L, m)
        score = _obj_val(c, n)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            # Restart with perturbation
            a = best_a.copy()
            temp = max(1.0, temp * 0.5)
            eta = 0.1
            no_improve = 0
            # Add random perturbation
            noise = rng.normal(0, 0.1, n)
            a = _normalize(a + noise * a)
            continue
        
        a_new = _grad_step(a, c, n, L, m, temp, eta)
        
        c_new = _conv_fft(a_new, L, m)
        score_new = _obj_val(c_new, n)
        
        if score_new < score:
            a = a_new
            eta = min(0.5, eta * 1.1)
        else:
            # Try smaller step
            a_try = _normalize((1-0.5*eta) * a + 0.5*eta * a_new)  
            # Hmm, that's not quite right. Let me just reduce eta.
            eta = max(0.02, eta * 0.8)
        
        if no_improve % 50 == 0 and no_improve > 20:
            temp = min(300.0, temp * 1.3)
    
    return best_a, best_score


def _coord_search(a, L, m, n, time_limit, top_k=25):
    """
    Coordinate descent search.
    """
    deadline = time.monotonic() + time_limit
    a = _normalize(a)
    c = _conv_fft(a, L, m)
    best_a = a.copy()
```
