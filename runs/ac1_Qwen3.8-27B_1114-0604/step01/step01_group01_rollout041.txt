def _softmax_weights(d, beta):
   d=np.asarray(d,float)
   n=d.size
   if n==0: return np.array([])
   if beta <= 1e-8: return np.full(n,1.0/n)
   shift=float(np.max(d))
   if not np.isfinite(shift): return np.full(n,1.0/n)
   e=np.exp(float(beta)*(d-shift))
   s=float(np.sum(e))
   if s <=0 or not np.isfinite(s):
       w=np.zeros(n)
       w[int(np.argmax(d))]=1.0
       return w
   w=e/s
   return w

"""
Multi-scale multiplicative/threshold search for non-negative sequences
minimizing

    2*n*max(conv(a,a)) / sum(a)**2.

The search keeps the sequence normalized (sum = 1), so the objective is
2*n*max(conv(p,p)).  The optimization routine combines:
  - threshold-penalized multiplicative updates against a moving target
    coefficient level,
  - softmax smoothed-max updates,
  - random perturbations / resets to escape local basins,
  - exact convolution score with FFT,
  - coarse-to-fine resampling,
  - small exact block-scaling local search.
"""

import numpy as np
import time

_GLOBAL_RNG = np.random.default_rng(12345)
_CONV_LEN_CACHE = {}
_MIN_P = 1e-16


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_len(n):
    n = int(max(1, n))
    if n in _CONV_LEN_CACHE:
        return _CONV_LEN_CACHE[n]
    need = 2 * n - 1
    L = 1
    while L < need:
        L <<= 1
    _CONV_LEN_CACHE[n] = L
    return L


def _sanitize(p, min_p=0.0, normalize=True):
    if p is None:
        return None
    try:
        a = np.asarray(p, dtype=float).ravel()
    except Exception:
        return None
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if min_p > 0.0:
        a = a + float(min_p)
    if normalize:
        s = float(a.sum())
        if s <= 0.0 or not np.isfinite(s):
            return None
        a = a / s
    return a


def _score_c_ones(p, L=None):
    p = np.asarray(p, dtype=float)
    n = int(p.size)
    if n == 0:
        return float("inf"), np.array([], dtype=float)
    if L is None:
        L = _conv_len(n)
    Fp = np.fft.rfft(p, L)
    c = np.fft.irfft(Fp * Fp, L)
    c = c[:2 * n - 1]
    c = np.maximum(c, 0.0)
    sc = float(2.0 * n * c.max()) if c.size else float("inf")
    return sc, c


def _exact_score_general(p):
    a = _sanitize(p, 0.0, True)
    if a is None:
        return float("inf")
    sc, _ = _score_c_ones(a)
    return sc


def _score_key(item):
    return item[0]


def _add_candidate(cands, p, score=None, keep=24):
    a = _sanitize(p, 0.0, True)
    if a is None:
        return
    if score is None or not np.isfinite(score):
        score = _exact_score_general(a)
    if not np.isfinite(score):
        return
    cands.append((float(score), int(a.size), a.copy()))
    if len(cands) > keep:
        cands.sort(key=_score_key)
        del cands[keep:]


def _noise_array(p, sigma):
    a = _sanitize(p, 0.0, True)
    if a is None or sigma <= 0.0:
        return a
    r = np.exp(float(sigma) * _GLOBAL_RNG.normal(0.0, 1.0, a.size))
    b = a * r
    return _sanitize(b, _MIN_P, True)


def _smooth_array(p, sigma):
    a = _sanitize(p, 0.0, True)
    if a is None or sigma <= 0.0 or a.size < 5:
        return a
    sigma = float(sigma)
    rad = int(max(1, np.ceil(3.0 * sigma)))
    k = np.arange(-rad, rad + 1, dtype=float)
    kern = np.exp(-0.5 * (k / sigma) ** 2)
    s = float(kern.sum())
    if s <= 0.0 or not np.isfinite(s):
        return a
    kern = kern / s
    out = np.convolve(a, kern, mode="same")
    return _sanitize(out, _MIN_P, True)


def _resample(p, m):
    m = int(max(1, m))
    a = _sanitize(p, 0.0, True)
    if a is None:
        return None
    n = int(a.size)
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m, dtype=float)
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0) + 1e-12
    return _sanitize(out, _MIN_P, True)


def _initial_sequence(n, rng, mode=None):
    if n <= 0:
        return None
    n = int(n)
    if mode is None:
        mode = int(rng.integers(0, 6))
    t = (np.arange(n, dtype=float) + 0.5) / float(n)
    p = np.ones(n, dtype=float)

    if mode == 0:
        z = 1.0
        for k in range(1, 13):
            amp = float(rng.normal(0.0, 1.0 / (k + 1)))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            z = z + amp * np.cos(2.0 * np.pi * k * t + ph)
        p = np.log1p(np.exp(np.clip(z, -50.0, 50.0)))

    elif mode == 1:
        a = float(rng.uniform(0.05, 3.0))
        b = float(rng.uniform(0.05, 3.0))
        tt = np.clip(t, 1e-8, 1.0 - 1e-8)
        p = np.power(tt, a) * np.power(1.0 - tt, b)

    elif mode == 2:
        K = int(rng.integers(1, 4))
        p = np.zeros(n, dtype=float)
        centers = np.sort(rng.uniform(0.0, 1.0, K))
        sigmas = rng.uniform(0.02, 0.35, K)
        weights = rng.uniform(0.5, 2.0, K)
        for ii in range(K):
            s = max(1e-6, float(sigmas[ii]))
            p = p + float(weights[ii]) * np.exp(-((t - float(centers[ii])) / s) ** 2)

    elif mode == 3:
        K = int(rng.integers(3, 9))
        knots = np.sort(rng.uniform(0.0, 1.0, K))
        heights = rng.lognormal(0.0, 1.0, K)
        p = np.interp(t, knots, heights)

    elif mode == 4:
        K = int(rng.integers(3, 16))
        edges = np.sort(rng.uniform(0.0, 1.0, K - 1))
        blocks = [0.0]
        blocks.extend([float(x) for x in edges])
        blocks.append(1.0)
        heights = rng.lognormal(0.0, 1.0, K)
        p = np.zeros(n, dtype=float)
        for ii in range(K):
            s = max(0, int(round(blocks[ii] * n)))
            e = max(s + 1, int(round(blocks[ii + 1] * n)))
            e = min(n, e)
            p[s:e] = float(heights[ii])

    else:
        c = float(rng.uniform(0.2, 0.8))
        k = float(rng.uniform(5.0, 30.0))
        p = 1.0 / (1.0 + np.exp(-np.clip(k * (t - c), -50.0, 50.0)))
        p = p * np.exp(-((t - c) / 0.25) ** 2)

    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0) + 1e-12
    return _sanitize(p, _MIN_P, True)


def _two_block_best(n):
    n = int(max(1, n))
    if n < 8:
        return None
    best = None
    best_sc = float("inf")
    cuts = (0.25, 0.35, 0.45, 0.55, 0.65, 0.75)
    ratios = (0.1, 0.3, 0.6, 1.0, 2.0, 4.0)
    for cut in cuts:
        idx = int(round(cut * n))
        if idx <= 0 or idx >= n:
            continue
        for rat in ratios:
            p = np.ones(n, dtype=float)
            p[idx:] = float(rat)
            p = _sanitize(p, 0.0, True)
            if p is None:
                continue
            sc, _ = _score_c_ones(p)
            if sc < best_sc:
                best_sc = sc
                best = p.copy()
    return best


def _softmax_weights(d, beta):
    d = np.asarray(d, dtype=float)
    n = int(d.size)
    if n == 0:
        return np.array([], dtype=float)
    if beta <= 1e-8:
        return np.full(n, 1.0 / n, dtype=float)
    shift = float(np.max(d))
    if not np.isfinite(shift):
        return np.full(n, 1.0 / n, dtype=float)
    e = np.exp(float(beta) * (d - shift))
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros(n, dtype=float)
        w[int(np.argmax(d))] = 1.0
        return w
    return e / s


def _fracs_for(k):
    k = int(k)
    if k <= 0:
        return []
    if k == 1:
        return [1.0]
    if k == 2:
        return [0.62, 0.38]
    return [0.5, 0.3, 0.2][:k]


def _combined_mmd(
    p0,
    time_limit,
    target_score=1.500,
    beta_start=50.0,
    beta_end=1000.0,
    alpha_start=0.08,
    eps_start=0.10,
    verbose=False,
):
    p = _sanitize(p0, _MIN_P, True)
    if p is None:
        return None, float("inf")

    time_limit = float(max(0.0, time_limit))
    n = int(p.size)
    L = _conv_len(n)

    if time_limit <= 0.0:
        sc, _ = _score_c_ones(p, L)
        return p, sc

    best = p.copy()
    best_sc, _ = _score_c_ones(p, L)
    cur_sc = best_sc

    beta = float(beta_start)
    alpha = float(alpha_start)
    eps = float(eps_start)
    if not np.isfinite(eps) or eps <= 0.0 or eps > 1.0:
        eps = 0.10

    target_m = max(1e-12 / float(n), float(target_score) / (2.0 * float(n)))
    deadline = time.monotonic() + time_limit
    stall = 0
    it = 0
    last_print = -100000

    mults = (0.5, 1.0, 2.0, 4.0) if n >= 8192 else (0.25, 0.5, 1.0, 2.0, 4.0)

    while time.monotonic() < deadline:
        Fp = np.fft.rfft(p, L)
        c = np.fft.irfft(Fp * Fp, L)
        c = c[:2 * n - 1]
        c = np.maximum(c, 0.0)
        maxc = float(c.max())
        cur_sc = float(2.0 * n * maxc)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best = p.copy()
            if verbose and (it - last_print >= 250):
                print(f"mmd n={n} it={it} best={best_sc:.7f}")
                last_print = it

        d = n * c
        use_threshold = (it % 20) < 16
        key_type = 1
        w = None
        t = 0.0
        viol = None

        if maxc > target_m * 1.000001:
            if use_threshold:
                t = (1.0 - eps) * maxc + eps * target_m
                if t >= maxc:
                    t = max(target_m, maxc * (1.0 - 1e-5))
                viol = np.maximum(0.0, c - t)
                maxviol = float(viol.max())
                if maxviol > 1e-14 * maxc:
                    w = viol / (maxviol + 1e-300)
                    w = w * (0.25 + 0.75 * w)
                    w = w * w
                    key_type = 0
                else:
                    w = _softmax_weights(d, 80.0)
                    key_type = 1
            else:
                w = _softmax_weights(d, beta)
                key_type = 1
        else:
            if use_threshold:
                t = max(target_m * 0.25, maxc * (1.0 - 0.01 - 0.05 * eps))
                viol = np.maximum(0.0, c - t)
                maxviol = float(viol.max())
                if maxviol > 1e-14 * maxc:
                    w = viol / (maxviol + 1e-300)
                    w = w * (0.25 + 0.75 * w)
                    w = w * w
                    key_type = 0
                else:
                    w = _softmax_weights(d, beta)
                    key_type = 1
            else:
                w = _softmax_weights(d, beta)
                key_type = 1

        if w is None:
            w = _softmax_weights(d, beta)
            key_type = 1

        Fw = np.fft.rfft(w, L)
        h = np.fft.irfft(Fw * np.conjugate(Fp), L)[:n]
        h = np.maximum(h, 0.0)
        hm = float(h.max())

        if hm <= 1e-300:
            std = 0.05
            p_try = p * np.exp(_GLOBAL_RNG.normal(0.0, std, n))
            p_try = np.maximum(p_try, _MIN_P)
            s = float(p_try.sum())
            if s > 0.0 and np.isfinite(s):
                p = p_try / s
            else:
                p = best.copy()
                cur_sc = best_sc
            stall = 0
            it += 1
            continue

        h = h / hm

        if key_type == 0:
            cur_key = float(np.dot(viol, viol))
        else:
            cur_key = float(np.dot(w, d))

        tests = []
        for mult in mults:
            a = alpha * float(mult)
            if a < 0.01:
                a = 0.01
            if a > 1.5:
                a = 1.5
            if not tests or abs(a - tests[-1]) > 1e-8:
                tests.append(float(a))

        best_exact_p = None
        best_exact_sc = cur_sc
        best_exact_a = alpha

        best_key_p = None
        best_key_sc = cur_sc
        best_key_val = float("inf")
        best_key_a = alpha

        for a in tests:
            p_try = p * np.exp(-a * h)
            p_try = np.maximum(p_try, _MIN_P)
            s = float(p_try.sum())
            if s <= 0.0 or not np.isfinite(s):
                continue
            p_try = p_try / s

            sc2, c2 = _score_c_ones(p_try, L)
            if not np.isfinite(sc2):
                continue

            if sc2 < best_exact_sc - 1e-12:
                best_exact_sc = sc2
                best_exact_p = p_try
                best_exact_a = float(a)

            if key_type == 0:
                viol2 = np.maximum(0.0, c2 - t)
                key2 = float(np.dot(viol2, viol2)) + 1e-9 * sc2
            else:
                key2 = float(np.dot(w, n * c2)) + 1e-9 * sc2

            if (
                np.isfinite(key2)
                and key2 < best_key_val
                and key2 < cur_key - 1e-14
                and sc2 <= cur_sc * 1.10
                and sc2 <= best_sc * 1.08
            ):
                best_key_val = key2
                best_key_p = p_try
                best_key_sc = sc2
                best_key_a = float(a)

        chosen_p = None
        chosen_sc = cur_sc
        chosen_a = alpha

        if best_exact_p is not None:
            chosen_p = best_exact_p
            chosen_sc = best_exact_sc
            chosen_a = best_exact_a
        elif best_key_p is not None:
            chosen_p = best_key_p
            chosen_sc = best_key_sc
            chosen_a = best_key_a

        if chosen_p is not None:
            p = chosen_p
            cur_sc = chosen_sc
            alpha = chosen_a
            stall = 0
            if key_type == 0 and maxc > target_m:
                eps = min(1.0, eps * 1.004 + 1e-5)
        else:
            alpha = max(0.01, alpha * 0.7)
            stall += 1
            eps = min(1.0, eps * 1.005)

            if stall >= 12:
                std = 0.05 if stall < 50 else min(0.35, 0.05 + 0.003 * stall)
                p_try = p * np.exp(_GLOBAL_RNG.normal(0.0, std, n))
                p_try = np.maximum(p_try, _MIN_P)
                s = float(p_try.sum())
                if s > 0.0 and np.isfinite(s):
                    p_try = p_try / s
                    sc2, _ = _score_c_ones(p_try, L)
                    if np.isfinite(sc2) and sc2 <= max(best_sc * 1.03, cur_sc * 1.10):
                        p = p_try
                        cur_sc = sc2
                    else:
                        p = best.copy()
                        cur_sc = best_sc
                else:
                    p = best.copy()
                    cur_sc = best_sc
                stall = 0

            if stall > 120:
                p = best.copy()
                cur_sc = best_sc
                alpha = float(alpha_start)
                beta = max(float(beta_start), beta * 0.5)
                if cur_sc > float(target_score):
                    eps = max(float(eps_start), min(1.0, eps * 0.5))
                stall = 0

        it += 1
        if it % 30 == 0:
            beta = min(float(beta_end), beta * 1.08)

    return best, best_sc


def _local_block_search(p0, time_limit, verbose=False):
    p = _sanitize(p0, _MIN_P, True)
    if p is None:
        return None, float("inf")

    time_limit = float(max(0.0, time_limit))
    if time_limit <= 0.0:
        sc, _ = _score_c_ones(p)
        return p, sc

    n = int(p.size)
    L = _conv_len(n)
    best = p.copy()
    best_sc, _ = _score_c_ones(p, L)
    cur_sc = best_sc

    deadline = time.monotonic() + time_limit
    stagnant = 0
    it = 0
    last_print = -100000

    small = max(1, n // 2048)
    mid = max(small + 1, n // 256)
    large = max(mid + 1, min(n, n // 32))
    factors = (0.5, 0.7, 0.85, 1.2, 1.5, 2.0)

    while time.monotonic() < deadline:
        r = float(_GLOBAL_RNG.random())
        if r < 0.30:
            w = 1
        elif r < 0.55:
            lo = small
            hi = min(n + 1, max(lo + 1, small * 4 + 1))
            w = int(_GLOBAL_RNG.integers(lo, hi))
        elif r < 0.85:
            lo = mid
            hi = min(n + 1, max(lo + 1, int(mid * 4) + 1))
            w = int(_GLOBAL_RNG.integers(lo, hi))
        else:
            lo = large
            hi = min(n + 1, max(lo + 1, int(large * 4) + 1))
            w = int(_GLOBAL_RNG.integers(lo, hi))

        w = max(1, min(n, w))
        start = int(_GLOBAL_RNG.integers(0, n - w + 1))

        current_best = None
        current_score = cur_sc

        for f in factors:
            cand = p.copy()
            cand[start:start + w] = cand[start:start + w] * float(f)
            cand = np.maximum(cand, _MIN_P)
            s = float(cand.sum())
            if s <= 0.0 or not np.isfinite(s):
                continue
            cand = cand / s
            sc, _ = _score_c_ones(cand, L)
            if sc < current_score - 1e-12:
                current_score = sc
                current_best = cand

        if current_best is not None:
            p = current_best
            cur_sc = current_score
            stagnant = 0
            if cur_sc < best_sc:
                best_sc = cur_sc
                best = p.copy()
                if verbose and (it - last_print >= 200):
                    print(f"local n={n} it={it} best={best_sc:.7f}")
                    last_print = it
        else:
            stagnant += 1
            if stagnant > 25:
                std = 0.05 if stagnant < 80 else 0.12
                p_try = p * np.exp(_GLOBAL_RNG.normal(0.0, std, n))
                p_try = np.maximum(p_try, _MIN_P)
                s = float(p_try.sum())
                if s > 0.0 and np.isfinite(s):
                    p_try = p_try / s
                    sc2, _ = _score_c_ones(p_try, L)
                    if np.isfinite(sc2) and sc2 <= max(best_sc * 1.02, cur_sc * 1.05):
                        p = p_try
                        cur_sc = sc2
                    else:
                        p = best.copy()
                        cur_sc = best_sc
                else:
                    p = best.copy()
                    cur_sc = best_sc
                stagnant = 0

            if stagnant > 180:
                p = best.copy()
                cur_sc = best_sc
                stagnant = 0

        it += 1

    return best, best_sc


def _build_starts(parent, cands, n, limit=3):
    n = int(max(1, n))
    starts = []

    if parent is not None:
        sp = _resample(parent, n)
        if sp is not None:
            sp = _smooth_array(sp, 0.75)
            sp = _noise_array(sp, 0.02)
            if sp is not None:
                starts.append(sp)

    if cands:
        sorted_cands = sorted(cands, key=_score_key)
        added = 0
        limit2 = min(2, limit)
        for item in sorted_cands:
            if added >= limit2:
                break
            arr = item[2]
            if arr is None:
                continue
            sp = _resample(arr, n)
            if sp is not None:
                sp = _smooth_array(sp, 0.5)
                sp = _noise_array(sp, 0.03)
                if sp is not None:
                    starts.append(sp)
                    added += 1

    if n <= 4096:
        tb = _two_block_best(n)
        if tb is not None:
            tb = _smooth_array(tb, 1.0)
            if tb is not None:
                starts.append(tb)

    for mode in (1, 2, 4, 0):
        if len(starts) >= max(3, limit + 1):
            break
        rp = _initial_sequence(n, _GLOBAL_RNG, mode)
        if rp is not None:
            starts.append(rp)

    scored = []
    for sp in starts:
        if sp is None:
            continue
        sc = _exact_score_general(sp)
        if np.isfinite(sc):
            scored.append((sc, sp))

    if not scored:
        sp = _initial_sequence(n, _GLOBAL_RNG, 1)
        if sp is not None:
            sc = _exact_score_general(sp)
            scored.append((sc, sp))

    if not scored:
        return []

    scored.sort(key=_score_key)
    limit = max(1, min(limit, len(scored)))
    out = []
    for i in range(limit):
        out.append(scored[i][1])
    return out


def _process_level(parent, cands, n, t_full, deadline, verbose=False):
    n = int(max(1, n))
    t_full = float(max(0.0, t_full))
    if t_full < 20.0:
        return

    starts = _build_starts(parent, cands, n, 3 if n <= 2048 else 2)
    if not starts:
        return

    fracs = _fracs_for(len(starts))
    for idx, sp in enumerate(starts):
        rem = float(deadline - time.monotonic())
        if rem < 40.0:
            break
        talloc = t_full * fracs[idx]
        talloc = min(talloc, max(0.0, rem - 30.0))
        if talloc < 15.0:
            talloc = min(10.0, max(0.0, rem - 10.0))
        if talloc < 5.0:
            continue

        sc0 = _exact_score_general(sp)
        eps0 = 0.15 if sc0 < 1.55 else 0.05

        p, sc = _combined_mmd(
            sp,
            talloc,
            target_score=1.500,
            beta_start=50.0,
            beta_end=800.0,
            alpha_start=0.08,
            eps_start=eps0,
            verbose=verbose,
        )
        if p is not None and np.isfinite(sc):
            _add_candidate(cands, p, sc, 24)


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    global _GLOBAL_RNG

    start_time = time.monotonic()
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget < 60.0:
        budget = 60.0
    total = max(60.0, budget - 30.0)
    deadline = start_time + total

    try:
        iseed = int(seed)
    except Exception:
        iseed = 12345
    _GLOBAL_RNG = np.random.default_rng((iseed % 1000000007) + 1)

    cands = []
    parent = None

    try:
        parent_raw = height_sequence_1
    except NameError:
        parent_raw = None
    except Exception:
        parent_raw = None

    if parent_raw is not None:
        p = _sanitize(parent_raw, 0.0, True)
        if p is not None:
            sc = _exact_score_general(p)
            _add_candidate(cands, p, sc, 24)
            parent = p

    if not cands:
        p = np.ones(2048, dtype=float)
        p = p / p.sum()
        _add_candidate(cands, p, None, 24)

    verbose = total > 300.0

    if total > 1800.0:
        levels = [
            (512, 0.03),
            (1024, 0.06),
            (2048, 0.09),
            (4096, 0.16),
            (8192, 0.26),
            (16384, 0.36),
        ]
    elif total > 1000.0:
        levels = [
            (512, 0.03),
            (1024, 0.06),
            (2048, 0.11),
            (4096, 0.22),
            (8192, 0.55),
        ]
    elif total > 500.0:
        levels = [
            (512, 0.04),
            (1024, 0.08),
            (2048, 0.20),
            (4096, 0.60),
        ]
    else:
        levels = [
            (512, 0.05),
            (2048, 0.25),
            (4096, 0.70),
        ]

    reserved = max(30.0, 0.02 * total)
    for nn, frac in levels:
        rem = float(deadline - time.monotonic())
        if rem <= reserved + 40.0:
            break
        talloc = frac * total
        talloc = min(talloc, max(0.0, rem - reserved - 20.0))
        if talloc < 10.0:
            continue
        _process_level(parent, cands, int(nn), talloc, deadline, verbose)

    if not cands:
        arr = np.ones(2048, dtype=float) / 2048.0
        return [float(x) for x in arr]

    rem = float(deadline - time.monotonic())
    if rem > 60.0:
        best_item = min(cands, key=_score_key)
        best_arr = best_item[2]
        base = best_arr.size
        t1 = min(rem - 30.0, 240.0 if base > 8192 else 180.0)
        if t1 > 20.0:
            p, sc = _combined_mmd(
                best_arr,
                t1,
                target_score=1.500,
                beta_start=100.0,
                beta_end=2000.0,
                alpha_start=0.05,
                eps_start=0.3 if best_item[0] < 1.55 else 0.1,
                verbose=verbose,
            )
            if p is not None:
                _add_candidate(cands, p, sc, 24)

    rem = float(deadline - time.monotonic())
    if rem > 40.0:
        best_item = min(cands, key=_score_key)
        bl_time = min(rem - 15.0, 150.0 if best_item[2].size > 8192 else 120.0)
        if bl_time > 15.0:
            p, sc = _local_block_search(best_item[2], bl_time, verbose=verbose)
            if p is not None:
                _add_candidate(cands, p, sc, 24)

    rem = float(deadline - time.monotonic())
    if rem > 60.0:
        best_item = min(cands, key=_score_key)
        noisy = _noise_array(best_item[2], 0.12)
        if noisy is not None:
            trestart = min(rem - 40.0, 100.0)
            if trestart > 20.0:
                p, sc = _combined_mmd(
                    noisy,
                    trestart,
                    target_score=1.500,
                    beta_start=80.0,
                    beta_end=2000.0,
                    alpha_start=0.06,
                    eps_start=0.3,
                    verbose=verbose,
                )
                if p is not None:
                    _add_candidate(cands, p, sc, 24)

    rem = float(deadline - time.monotonic())
    if rem > 30.0:
        best_item = min(cands, key=_score_key)
        t2 = min(rem - 10.0, 80.0)
        if t2 > 10.0:
            p, sc = _combined_mmd(
                best_item[2],
                t2,
                target_score=1.500,
                beta_start=200.0,
                beta_end=3000.0,
                alpha_start=0.03,
                eps_start=0.5,
                verbose=verbose,
            )
            if p is not None:
                _add_candidate(cands, p, sc, 24)

    cands.sort(key=_score_key)
    final_arr = cands[0][2].copy()
    final_score = float(cands[0][0])
    selected = [float(x) for x in final_arr]

    # Use the official evaluator only when it is reasonably cheap.
    if final_arr.size <= 4096 and float(deadline - time.monotonic()) > 10.0:
        evaluated = 0
        for item in cands[:4]:
            if evaluated >= 4:
                break
            arr = _sanitize(item[2], 0.0, True)
            if arr is None:
                continue
            lst = [float(x) for x in arr]
            try:
                sc = float(evaluate_sequence(lst))
            except Exception:
                sc = _exact_score_general(lst)
            if np.isfinite(sc) and sc < final_score:
                final_score = sc
                selected = lst
            evaluated += 1

    out = np.asarray(selected, dtype=float)
    if out.size == 0:
        out = np.asarray(final_arr, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s <= 0.0 or not np.isfinite(s):
        out = np.ones(2048, dtype=float)
    else:
        out = out / s

    return [float(x) for x in out]
