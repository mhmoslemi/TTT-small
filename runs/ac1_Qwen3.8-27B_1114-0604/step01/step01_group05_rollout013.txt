def fft_convolution(a):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    """Smooth max via softmax."""
    m = float(np.max(b))
    if m <= 0.0:
        w = np.ones_like(b, dtype=np.float64)
        w /= float(b.size)
        return m, w
    
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    
    if Z <= 0.0 or not np.isfinite(Z):
        w = np.zeros_like(b, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
        return m, w
    
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w

"""
Minimize 2*n*max(a*a)/(sum(a))^2 for nonnegative sequences.

Uses multi-start search with:
1. Nesterov accelerated projected gradient on the simplex
2. Adaptive smooth max (annealing beta from low to high)
3. Simulated annealing to escape local minima
4. Multiple sequence lengths and diverse starting points
5. Final coordinate-descent polish

import numpy as np
import time

try:
    from scipy.optimize import minimize
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def fft_convolution(a):
    """Linear convolution a*a via FFT, clipped to non-negative."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    """Smooth max via log-sum-exp with normalized weights."""
    m = float(np.max(b))
    if m <= 0.0:
        w = np.ones_like(b, dtype=np.float64)
        w /= float(b.size)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        w = np.zeros_like(b, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
        return m, w
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w


def fast_score(p):
    """Compute 2*n*max(p*p) / (sum(p))^2 using FFT."""
    n = p.size
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float("inf")
    b = fft_convolution(p)
    M = float(np.max(b))
    return float(2.0 * n * M / (S * S))


def correlation_pw(p, w):
    """Compute s_i = sum_j p_j * w_{i+j} for i=0..n-1 via FFT."""
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n : m][::-1].copy()
    return np.maximum(s, 0.0)


def project_simplex(v, target=1.0):
    """Project vector onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    d = v.size
    if d == 0:
        return v
    if target <= 0.0:
        return np.zeros_like(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(d, target / d, dtype=np.float64)
    
    # Sort-based simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, d + 1) > (cssv - target))[0][-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    q = np.maximum(v - theta, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(d, target / d, dtype=np.float64)
    q *= target / s
    return q


def normalize_simplex(p):
    """Clip to nonnegative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / p.size, dtype=np.float64)
    p *= 1.0 / s
    return p


def nesterov_refine(p0, deadline, rng, max_iter=500000):
    """
    Nesterov accelerated projected gradient with adaptive smooth max.
    """
    p = normalize_simplex(p0)
    n = p.size
    if n == 0:
        return p, float("inf")

    # Break symmetry
    mean_p = float(np.mean(p))
    if n > 1 and float(np.max(np.abs(p - mean_p))) < 1e-14:
        p = normalize_simplex(p * (1.0 + 1e-4 * np.sin(0.1 * np.arange(n))))

    best_p = p.copy()
    best_score = fast_score(p)

    # Momentum
    x_prev = p.copy()
    beta_mom = 0.0
    step = 1.0
    no_improve = 0

    # Adaptive beta for smooth max
    beta_sm = 500.0
    beta_target = 500000.0
    beta_growth = 1.08

    it = 0
    anneal_period = 200
    anneal_temp = 0.05

    while it < max_iter and time.monotonic() < deadline:
        it += 1

        # Nesterov lookahead
        if it == 1:
            y = p.copy()
        else:
            beta_new = (1.0 + np.sqrt(1.0 + 4.0 * beta_mom * beta_mom)) / 2.0
            t_k = (1.0 + beta_mom) / beta_new
            beta_mom = beta_new - 1.0
            y = p + t_k * (p - x_prev)
            y = np.maximum(y, 0.0)
            sy = float(np.sum(y))
            if sy > 0:
                y /= sy
            else:
                y = p.copy()

        # Compute gradient at y
        b = fft_convolution(y)
        G, w = smooth_max_and_weights(b, beta_sm)
        if not np.isfinite(G):
            break

        # Track best with exact max
        m_exact = float(np.max(b))
        cur_score = float(2.0 * n * m_exact / (float(np.sum(y)) ** 2))
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_p = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 5000:
            # Annealing: random perturbation
            if anneal_temp > 0.001:
                perturb = rng.normal(0, anneal_temp, n)
                p_new = normalize_simplex(p + perturb)
                sc_new = fast_score(p_new)
                if sc_new < best_score * 1.5:  # accept if not too bad
                    p = p_new
                    x_prev = p_new
                    anneal_temp *= 0.9
                    no_improve = 0
                    continue
            else:
                break

        if it % 500 == 0:
            print(f"  nesterov it={it} best={best_score:.8f} beta_sm={beta_sm:.0f} step={step:.3e}")

        # Compute gradient
        svec = correlation_pw(y, w)
        g = 2.0 * svec
        grad = g - float(np.mean(g))

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-15:
            break

        # Line search with backtracking
        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625):
            alpha = step * mult
            q = project_simplex(y - alpha * grad, 1.0)
            qb = fft_convolution(q)
            qm = float(np.max(qb))
            qG, _ = smooth_max_and_weights(qb, beta_sm)
            if not np.isfinite(qG):
                continue
            if float(n) * qG < float(n) * G * (1.0 - 1e-7) - 1e-12:
                p = q
                accepted = True
                break

        if not accepted:
            step *= 0.5
            if step < 1e-8:
                break
        else:
            if step < 5.0:
                step = min(5.0, step * 1.15)

        x_prev = p.copy()

        # Anneal beta_sm
        if beta_sm < beta_target:
            beta_sm = min(beta_target, beta_sm * beta_growth)

    return best_p, best_score


def coordinate_descent_polish(p0, deadline, rng, max_sweeps=50):
    """
    Coordinate descent: for each coordinate, try to find a better value.
    Uses a simple 1D search per coordinate.
    """
    p = normalize_simplex(p0)
    n = p.size
    best_score = fast_score(p)

    for sweep in range(max_sweeps):
        if time.monotonic() >= deadline:
            break
        improved = False
        # Randomize coordinate order
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() >= deadline:
                break
            v = p[i]
            if v < 1e-12:
                continue

            # Try a few candidate values (multiplicative perturbation)
            best_local = p[i]
            for factor in (0.5, 0.7, 0.85, 1.15, 1.3, 1.5):
                t = v * factor
                p_trial = p.copy()
                p_trial[i] = t
                p_trial = normalize_simplex(p_trial)
                sc = fast_score(p_trial)
                if sc < best_score - 1e-12:
                    best_score = sc
                    p = p_trial
                    best_local = t
                    improved = True
                    break

        if not improved and sweep > 5:
            break
        if sweep % 10 == 0:
            print(f"  coord desc sweep={sweep} score={best_score:.8f}")

    return p, best_score


def block_to_p(n, B, c):
    """Convert block log-weights to full sequence."""
    L = n // B
    z = np.zeros(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        m = min(B - 1, c.size)
        z[:m] = c[:m]
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0 or not np.isfinite(s):
        wb = np.ones(B) / B
    else:
        wb /= s
    H = wb / L
    return np.repeat(H, L)


def block_objective(c, n, B, beta):
    """Objective and gradient for block-constant sequences."""
    L = n // B
    z = np.zeros(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        m = min(B - 1, c.size)
        z[:m] = c[:m]
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0 or not np.isfinite(s):
        wb = np.ones(B) / B
    else:
        wb /= s
    H = wb / L
    p = np.repeat(H, L)

    b = fft_convolution(p)
    G, w = smooth_max_and_weights(b, beta)
    if not np.isfinite(G):
        return float("inf"), np.zeros(B - 1, dtype=np.float64)

    svec = correlation_pw(p, w)
    g = 2.0 * svec
    Gsum = g.reshape(B, L).sum(axis=1)
    Wavg = float(np.dot(wb, Gsum))
    gradz = (wb / L) * (Gsum - Wavg)
    grad = n * gradz[:-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(n * G), grad.astype(np.float64)


def run_block_optimize(c0, n, B, beta, maxiter=200):
    """Run L-BFGS-B on block representation."""
    if not HAVE_SCIPY:
        return c0
    m = B - 1
    if m <= 0:
        return c0
    c0 = np.asarray(c0, dtype=np.float64).reshape(-1)
    c0 = np.clip(c0, -8.0, 8.0)
    bounds = [(-8.0, 8.0)] * m
    options = {"maxiter": maxiter, "ftol": 1e-12, "gtol": 1e-6, "maxls": 30}
    try:
        res = minimize(block_objective, c0, args=(n, B, beta),
                       method="L-BFGS-B", jac=True, bounds=bounds, options=options)
        c = np.asarray(res.x, dtype=np.float64)
    except Exception:
        c = c0
    if not np.all(np.isfinite(c)):
        c = c0
    return np.clip(c, -8.0, 8.0)


def generate_starts(n, rng):
    """Generate diverse starting sequences."""
    starts = []
    # Uniform
    starts.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Linear tapers
    x = np.linspace(0, 1, n)
    for a in (-3, -2, -1, 1, 2, 3):
        p = np.exp(a * (x - 0.5))
        p = normalize_simplex(p)
        starts.append(p)

    # Cosine / raised cosine
    for amp in (0.5, 1.0, 2.0):
        p = 1.0 + amp * np.cos(2 * np.pi * x)
        p = np.maximum(p, 1e-6)
        p = normalize_simplex(p)
        starts.append(p)

    # Triangle
    p = 1.0 - 2.0 * np.abs(x - 0.5)
    p = np.maximum(p, 1e-6)
    p = normalize_simplex(p)
    starts.append(p)

    # Two-hump
    for center in (0.3, 0.35, 0.4, 0.6, 0.65, 0.7):
        p = np.exp(-50.0 * (x - center) ** 2) + np.exp(-50.0 * (x - (1 - center)) ** 2) + 0.1
        p = normalize_simplex(p)
        starts.append(p)

    # Random smooth
    for _ in range(3):
        size = 1
        while size < n * 2:
            size <<= 1
        zc = rng.normal(size=size) + 1j * rng.normal(size=size)
        z = np.real(np.fft.ifft(zc))[:n]
        z = z - z.mean()
        z = np.tanh(2.0 * z)
        p = np.exp(z)
        p = normalize_simplex(p)
        starts.append(p)

    return starts

        # If we have a warm start, also try running nesterov on it directly
        if warm_start is not None and warm_start.size == n and time.monotonic() < n_deadline - 20:
            ws = normalize_simplex(warm_start)
            p_fit, sc_fit = nesterov_refine(ws, n_deadline - 20, rng)
            if sc_fit < best_score:
                best_score = sc_fit
                best_p = p_fit.copy()
                print(f"  New best after warm nesterov: {best_score:.8f}")

    if best_p is None:
        # Fallback
        return [1.0] * 1024

    # Final: try a few more Nesterov rounds on the overall best
    if time.monotonic() < deadline - 30 and best_p is not None:
        n_best = best_p.size
        print(f"\n=== Final polish on n={n_best}, current best={best_score:.8f} ===")
        p_final, sc_final = nesterov_refine(best_p, deadline - 30, np.random.default_rng(seed + 999))
        if sc_final < best_score:
            best_score = sc_final
            best_p = p_final.copy()
            print(f"  Final best: {best_score:.8f}")

    # Convert to list and verify
    final_p = normalize_simplex(best_p)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_seq = [float(x) for x in final_p]

    # Check official score
    off = official_score(final_seq)
    print(f"\nOfficial score: {off:.8f}")

    if not np.isfinite(off) or off > 2.0:
        # Fallback to uniform
        return [1.0] * 1024

    return final_seq
