    while time.monotonic() - start < budget_s and it < max_iters:
        it += 1

        # Compute convolution
        c = _fft_convolve(arr)
        if c.size == 0:
            break
        maxc = float(np.max(c))
        if not np.isfinite(maxc):
            break

        # Find top-K indices
        k_eff = min(K, c.size)
        if k_eff <= 1:
            active = np.array([np.argmax(c)], dtype=np.int64)
        else:
            active = np.argpartition(c, -k_eff)[-k_eff:]

        # Compute subgradient (average over top-K)
        grad = np.zeros(n, dtype=np.float64)
        for kk in active:
            k = int(kk)
            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            if i_max >= i_min:
                idx = np.arange(i_min, i_max + 1, dtype=np.int64)
                # dc[k]/da[i] = 2*a[k-i] for i in valid range
                # But we need to be careful: k-i must be in [0, n-1]
                partners = k - idx
                grad[idx] += 2.0 * arr[partners]

        grad /= float(len(active))

        # Project to tangent space of simplex (sum = 1)
        # Subtract mean to maintain normalization approximately
        g = grad - np.mean(grad)

        # Momentum
        vel = momentum * vel + (1 - momentum) * g

        # Mirror descent update
        exponent = -lr * vel
        exponent = np.clip(exponent, -2.0, 2.0)
        arr_new = arr * np.exp(exponent)
        arr_new = np.clip(arr_new, 0.0, 1000.0)
        s_new = arr_new.sum()
        if s_new < 0.01 or not np.isfinite(s_new):
            lr = max(min_lr, lr * 0.7)
            continue
        arr_new = arr_new / s_new

        # Evaluate periodically
        if it % 10 == 0:
            sc = _internal_score(arr_new)
            if np.isfinite(sc) and sc < best_score:
                best = arr_new.copy()
                best_score = sc
                last_improve = it
                lr = min(max_lr, lr * 1.03)
            else:
                if sc > best_score * 1.01:
                    arr_new = best.copy()
                    lr = max(min_lr, lr * 0.8)
                    vel *= 0.5
                else:
                    lr = min(max_lr, lr * 1.01)
                if it - last_improve > patience:
                    # Try a smaller lr
                    lr = max(min_lr, lr * 0.5)
                    vel = np.zeros(n, dtype=np.float64)
                    if it - last_improve > patience * 3:
                        break

        arr = arr_new

    return best, best_score

"""
Low-peak nonnegative step-function search using multi-frequency sinusoidal
initialization with aggressive mirror-descent refinement and soft-max variants.

Key improvements over previous approach:
1. Multi-frequency anti-palindromic sinusoidal initialization
2. More aggressive mirror descent with momentum, adaptive step size, and patience restarts
3. Soft-max smoothed gradient as alternative to hard top-K
4. Multiple restarts with different parameters to escape local minima
5. Extended refinement with increasing K for smoother late-stage optimization
"""

import numpy as np
import time


def _next_pow2(m):
    """Smallest power of two >= m."""
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    """Fast internal evaluation matching evaluate_sequence."""
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))


def _make_sine_n(n, A, mode=1):
    """Single-frequency anti-palindromic sinusoidal, sum-normalized."""
    i = np.arange(n, dtype=np.float64)
    val = 1.0 + float(A) * np.sin(2.0 * np.pi * float(mode) * (i + 0.5) / float(n))
    val = np.clip(val, 0.0, None)
    a = val / float(n)
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / float(n)
    return a / s


def _make_multi_sine(n, amps):
    """Multi-frequency anti-palindromic sinusoidal, sum-normalized."""
    i = np.arange(n, dtype=np.float64)
    val = np.ones(n, dtype=np.float64)
    for k, A in enumerate(amps, start=1):
        val += float(A) * np.sin(2.0 * np.pi * float(k) * (i + 0.5) / float(n))
    val = np.clip(val, 0.0, None)
    a = val / float(n)
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / float(n)
    return a / s


def _compute_topk_grad(c, arr, n, K):
    """Compute averaged top-K subgradient of max(convolution)."""
    k_eff = min(K, c.size)
    if k_eff <= 1:
        active = np.array([np.argmax(c)], dtype=np.int64)
    else:
        active = np.argpartition(c, -k_eff)[-k_eff:]

    grad = np.zeros(n, dtype=np.float64)
    for kk in active:
        k = int(kk)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        if i_max >= i_min:
            idx = np.arange(i_min, i_max + 1, dtype=np.int64)
            partners = k - idx
            grad[idx] += 2.0 * arr[partners]
    grad /= float(len(active))
    return grad


def _compute_softmax_grad(c, arr, n, temperature):
    """Compute soft-max smoothed gradient."""
    # Use softmax weights on the convolution values
    c_max = np.max(c)
    exp_c = np.exp(temperature * (c - c_max))
    weights = exp_c / np.sum(exp_c)

    grad = np.zeros(n, dtype=np.float64)
    # Only process lags with significant weight for efficiency
    threshold = 1.0 / (n * 4)
    significant = np.where(weights > threshold)[0]
    if len(significant) == 0:
        significant = np.array([np.argmax(weights)], dtype=np.int64)

    for k in significant:
        k = int(k)
        w = weights[k]
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        if i_max >= i_min:
            idx = np.arange(i_min, i_max + 1, dtype=np.int64)
            partners = k - idx
            grad[idx] += 2.0 * w * arr[partners]

    return grad


def _mirror_descent(a, budget_s, K=16, lr_init=None, momentum=0.5, use_softmax=False, softmax_temp=None):
    """
    Mirror descent optimization on the simplex with top-K or soft-max subgradient.
    """
    start = time.monotonic()
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return arr, _internal_score(arr)
    arr = arr / s

    n = arr.size
    best = arr.copy()
    best_score = _internal_score(best)
    if not np.isfinite(best_score):
        return best, best_score

    if lr_init is None:
        lr_init = max(2.0, 0.08 * float(n))

    lr = lr_init
    max_lr = 0.4 * float(n)
    min_lr = 0.5

    it = 0
    max_iters = 50000
    last_improve = 0
    patience = 400
    vel = np.zeros(n, dtype=np.float64)
    eval_period = 8

    while time.monotonic() - start < budget_s and it < max_iters:
        it += 1

        # Compute convolution
        c = _fft_convolve(arr)
        if c.size == 0:
            break
        maxc = float(np.max(c))
        if not np.isfinite(maxc):
            break

        # Compute gradient
        if use_softmax:
            temp = softmax_temp if softmax_temp is not None else (n * 0.5)
            grad = _compute_softmax_grad(c, arr, n, temp)
        else:
            grad = _compute_topk_grad(c, arr, n, K)

        # Project to tangent space of simplex
        g = grad - np.mean(grad)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break

        # Normalize gradient for stable step size
        g = g / g_norm

        # Momentum
        vel = momentum * vel + (1.0 - momentum) * g

        # Mirror descent update
        exponent = -lr * vel
        exponent = np.clip(exponent, -2.5, 2.5)
        arr_new = arr * np.exp(exponent)
        arr_new = np.clip(arr_new, 0.0, 1000.0)
        s_new = arr_new.sum()
        if s_new < 0.01 or not np.isfinite(s_new):
            lr = max(min_lr, lr * 0.7)
            continue
        arr_new = arr_new / s_new

        # Evaluate periodically
        if it % eval_period == 0:
            sc = _internal_score(arr_new)
            if np.isfinite(sc) and sc < best_score:
                best = arr_new.copy()
                best_score = sc
                last_improve = it
                lr = min(max_lr, lr * 1.02)
            else:
                if sc > best_score * 1.005:
                    arr_new = best.copy()
                    lr = max(min_lr, lr * 0.85)
                    vel *= 0.3
                else:
                    lr = min(max_lr, lr * 1.005)
                if it - last_improve > patience:
                    # Restart with reduced lr and zero momentum
                    arr_new = best.copy()
                    lr = max(min_lr, lr * 0.5)
                    vel = np.zeros(n, dtype=np.float64)
                    last_improve = it
                    patience = int(patience * 1.3)
                    if patience > 3000:
                        break

        arr = arr_new

    # Final evaluation
    final_sc = _internal_score(arr)
    if np.isfinite(final_sc) and final_sc < best_score:
        best = arr.copy()
        best_score = final_sc

    return best, best_score


def _try_multi_sine_optimization(n, deadline_abs):
    """Optimize multi-frequency amplitudes for a given length."""
    best_score = np.inf
    best_a = None

    # Coarse grid over primary amplitude
    for A1 in [0.7, 0.75, 0.8, 0.85, 0.9, 0.95, 1.0]:
        if time.monotonic() > deadline_abs:
            break
        # Single frequency
        a = _make_sine_n(n, A1)
        sc = _internal_score(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Try adding secondary frequency
        for A2 in [-0.15, -0.05, 0.05, 0.1, 0.15]:
            if time.monotonic() > deadline_abs:
                break
            a2 = _make_multi_sine(n, [A1, A2])
            sc2 = _internal_score(a2)
            if sc2 < best_score:
                best_score = sc2
                best_a = a2.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a low evaluation score and return the best valid sequence found.
    """
    t0 = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    hard_deadline = t0 + total_budget - 40.0

    best_out = None
    best_score = np.inf

    print(f"[search] Starting with budget {total_budget:.0f}s, hard deadline in {hard_deadline - t0:.0f}s")

    # Phase 1: Generate diverse initial candidates
    init_deadline = t0 + min(90.0, total_budget * 0.04)
    candidates = []

    # Single-frequency candidates
    for n in [2048, 4096, 8192, 1024, 16384]:
        if time.monotonic() > init_deadline:
            break
        for A in [0.7, 0.78, 0.85, 0.9, 0.95, 1.0]:
            a = _make_sine_n(n, A)
            sc = _internal_score(a)
            candidates.append((sc, a))
            if sc < best_score:
                best_score = sc
                best_out = a.copy()

    # Multi-frequency candidates
    for n in [2048, 4096]:
        if time.monotonic() > init_deadline:
            break
        amp_sets = [
            [0.78, 0.1], [0.78, -0.1], [0.85, 0.15], [0.85, -0.1],
            [0.9, 0.1], [0.9, -0.15], [0.95, 0.05], [0.95, -0.1],
            [0.78, 0.2, 0.05], [0.85, 0.15, -0.05], [0.9, 0.1, 0.1],
            [0.8, 0.2, 0.1, -0.05],
        ]
        for amps in amp_sets:
            if time.monotonic() > init_deadline:
                break
            a = _make_multi_sine(n, amps)
            sc = _internal_score(a)
            candidates.append((sc, a))
            if sc < best_score:
                best_score = sc
                best_out = a.copy()

    # Include previous solution if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.ndim == 1 and 0 < prev.size < 100000:
            prev = np.clip(prev, 0, 1000)
            s = prev.sum()
            if s > 0.01:
                prev = prev / s
                sc = _internal_score(prev)
                candidates.append((sc, prev.copy()))
                if sc < best_score:
                    best_score = sc
                    best_out = prev.copy()
                print(f"[init] Previous solution: n={prev.size} score={sc:.6f}")
    except Exception:
        pass

    # Sort candidates by score
    candidates.sort(key=lambda x: x[0])
    print(f"[init] Best initial score: {best_score:.6f}, {len(candidates)} candidates")

    # Phase 2: Aggressive refinement of top candidates
    refine_deadline = hard_deadline - 80.0

    # Refine the best candidate with the most time
    if best_out is not None and time.monotonic() < refine_deadline:
        remaining = refine_deadline - time.monotonic()
        main_budget = min(remaining * 0.55, 900.0)
        main_budget = max(main_budget, 60.0)
        print(f"[refine] Main refinement: budget={main_budget:.0f}s, initial_score={best_score:.6f}")

        refined, ref_score = _mirror_descent(best_out, main_budget, K=16, momentum=0.5)
        if np.isfinite(ref_score) and ref_score < best_score:
            best_score = ref_score
            best_out = refined
            print(f"[refine] Main improved: score={ref_score:.6f}")
        else:
            print(f"[refine] Main no improvement: {ref_score:.6f} vs {best_score:.6f}")

    # Try soft-max variant on best result
    if best_out is not None and time.monotonic() < refine_deadline:
        remaining = refine_deadline - time.monotonic()
        sm_budget = min(remaining * 0.25, 300.0)
        sm_budget = max(sm_budget, 30.0)
        print(f"[refine] Soft-max refinement: budget={sm_budget:.0f}s")

        refined_sm, sm_score = _mirror_descent(best_out, sm_budget, K=32, momentum=0.4, use_softmax=True, softmax_temp=None)
        if np.isfinite(sm_score) and sm_score < best_score:
            best_score = sm_score
            best_out = refined_sm
            print(f"[refine] Soft-max improved: score={sm_score:.6f}")
        else:
            print(f"[refine] Soft-max no improvement: {sm_score:.6f}")

    # Try second-best initial candidate
    if len(candidates) > 1 and time.monotonic() < refine_deadline:
        remaining = refine_deadline - time.monotonic()
        if remaining > 60:
            second_a = candidates[1][1]
            # Skip if it's the same as best
            if not np.allclose(second_a, best_out):
                b2 = min(remaining * 0.2, 150.0)
                print(f"[refine] Second candidate: n={second_a.size} init_score={candidates[1][0]:.6f} budget={b2:.0f}s")
                r2, s2 = _mirror_descent(second_a, b2, K=16, momentum=0.5)
                if np.isfinite(s2) and s2 < best_score:
                    best_score = s2
                    best_out = r2
                    print(f"[refine] Second improved: score={s2:.6f}")

    # Extended refinement with larger K for smoother optimization
    if best_out is not None and time.monotonic() < refine_deadline:
        remaining = refine_deadline - time.monotonic()
        if remaining > 40:
            ext_budget = min(remaining * 0.6, 200.0)
            print(f"[refine] Extended K=32: budget={ext_budget:.0f}s")
            re, se = _mirror_descent(best_out, ext_budget, K=32, momentum=0.3, lr_init=max(2.0, 0.05 * best_out.size))
            if np.isfinite(se) and se < best_score:
                best_score = se
                best_out = re
                print(f"[refine] Extended improved: score={se:.6f}")

    # Final short refinement pass
    if best_out is not None and time.monotonic() < refine_deadline:
        remaining = refine_deadline - time.monotonic()
        if remaining > 20:
            final_budget = min(remaining * 0.8, 60.0)
            rf, sf = _mirror_descent(best_out, final_budget, K=8, momentum=0.6, lr_init=max(1.0, 0.03 * best_out.size))
            if np.isfinite(sf) and sf < best_score:
                best_score = sf
                best_out = rf
                print(f"[refine] Final pass improved: score={sf:.6f}")

    # Phase 3: Validation
    if best_out is None:
        best_out = _make_sine_n(2048, 0.78)
        best_score = _internal_score(best_out)
        print(f"[fallback] Using default sinusoidal, score={best_score:.6f}")

    # Sanitize final output
    final = np.asarray(best_out, dtype=np.float64)
    final = np.where(np.isfinite(final), final, 0.0)
    final = np.clip(final, 0.0, 1000.0)
    s_final = final.sum()
    if s_final < 0.01:
        final = np.ones(2048) / 2048.0
    else:
        final = final / s_final

    final_score = _internal_score(final)
    print(f"[done] Final: n={final.size} internal_score={final_score:.6f} time={time.monotonic()-t0:.1f}s")

    # Validate with actual function if available
    try:
        actual = float(evaluate_sequence(final.tolist()))
        print(f"[done] Actual evaluate_sequence score: {actual:.6f}")
        if not np.isfinite(actual):
            safe = _make_sine_n(2048, 0.78)
            return [float(x) for x in safe]
    except NameError:
        pass
    except Exception as e:
        print(f"[done] Validation error: {e}")

    return [float(x) for x in final]
