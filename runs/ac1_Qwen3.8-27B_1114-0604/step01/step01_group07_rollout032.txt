def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)

    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))

    return float(2.0 * arr.size * mx / (s * s))

    try:
        h1 = np.asarray(height_sequence_1, dtype=np.float64)
        if h1.size > 0 and np.all(np.isfinite(h1)):
            h1 = np.clip(h1, 0.0, None)
            s = h1.sum()
            if s > 0.01:
                h1 = h1 / s
                sc1 = _internal_score(h1)
                initializations.append((sc1, h1.size, h1))
                print(f"[init] height_sequence_1 n={h1.size} score={sc1:.8f}")
    except (NameError, Exception):
        pass

    if not initializations:
        initializations.append((2.0, 2048, _make_sinusoidal(2048, 0.8)))

    initializations.sort(key=lambda x: x[0])

    # Phase 2: Mirror descent optimization
    optimal_deadline = hard_deadline - 20.0
    remaining = optimal_deadline - time.monotonic()

    n_starts = min(3, len(initializations))
    if n_starts == 1:
        fractions = [1.0]
    elif n_starts == 2:
        fractions = [0.75, 0.25]
    else:
        fractions = [0.6, 0.25, 0.15]

    best_overall = None
    best_overall_score = np.inf

    for idx in range(n_starts):
        if time.monotonic() >= optimal_deadline:
            break

        sc, n, a = initializations[idx]
        alloc = remaining * fractions[idx]
        if alloc < 15.0:
            alloc = 15.0
        # Cap so we don't exceed deadline
        avail = optimal_deadline - time.monotonic()
        alloc = min(alloc, avail)

        print(f"[optimize] Point {idx+1}: n={n}, init_score={sc:.8f}, budget={alloc:.1f}s")

        opt_a, opt_sc = _mirror_descent(a, alloc, K=40, seed=seed + idx * 7)

        if np.isfinite(opt_sc) and opt_sc < best_overall_score:
            best_overall_score = opt_sc
            best_overall = opt_a
            print(f"[optimize] Point {idx+1} result: score={opt_sc:.8f} n={opt_a.size}")

        # Also compare with the initial
        if np.isfinite(sc) and sc < best_overall_score:
            best_overall_score = sc
            best_overall = a

    # Phase 3: Final polish - try a short additional optimization on the best
    if best_overall is not None and time.monotonic() < hard_deadline - 15.0:
        polish_budget = min(120.0, hard_deadline - time.monotonic() - 10.0)
        if polish_budget > 10.0:
            print(f"[polish] budget={polish_budget:.1f}s")
            pol_a, pol_sc = _mirror_descent(best_overall, polish_budget, K=50, seed=seed + 99)
            if np.isfinite(pol_sc) and pol_sc < best_overall_score:
                best_overall_score = pol_sc
                best_overall = pol_a
                print(f"[polish] improved to score={pol_sc:.8f}")
            else:
                # Try scalar polish
                pol_a, pol_sc = _polish_with_scalar_mult(best_overall, k_trials=30)
                if np.isfinite(pol_sc) and pol_sc < best_overall_score:
                    best_overall_score = pol_sc
                    best_overall = pol_a
                    print(f"[scalar_polish] improved to score={pol_sc:.8f}")

    # Phase 4: Final validation
    if best_overall is not None:
        arr = np.asarray(best_overall, dtype=np.float64)
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        s = float(arr.sum())
        if not np.isfinite(s) or s < 0.01:
            arr = _make_sinusoidal(2048, 0.8)
            s = arr.sum()
        arr = arr / s

        try:
            result = [float(x) for x in arr]
            real_score = float(evaluate_sequence(result))
            print(f"[FINAL] n={arr.size} actual_score={real_score:.8f}")
            return result
        except Exception as e:
            print(f"[FINAL] error: {e}, falling back")

    # Guaranteed fallback
    fallback = _make_sinusoidal(2048, 0.8)
    return [float(x) for x in fallback]
