Uses a "best response" LP to find globally optimal directions for escape

local optimization. Multi-scale approach: optimize at smaller n first,
then upscale and continue.

import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(a, b):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)

def _sanitize(p):
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return p / s

    if p is None:
        return float('inf')
    m = p.size
    c = _conv_full(p, p)
    if c.size == 0:
        return float('inf')
    c = np.where(np.isfinite(c), c, 0.0)
    return float(2.0 * m * np.max(c))

def _resample(arr, m):
    if m <= 0:
        return None
    a = _sanitize(arr)
    if a is None:
        return None
    n = a.size
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m, dtype=float) / m
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0)
    out = out + 1e-12
    return _sanitize(out)

def _lp_best_response(f, M):
    """
    Solve: max sum(b) s.t. (f*b)_k <= M for all k, b >= 0.
    Returns the solution b (not normalized).
    """
    n = len(f)
    # Construct the convolution matrix C: C[k,j] = f[k-j] if valid
    C = np.zeros((2*n-1, n))
    for j in range(n):
        C[j:j+n, j] = f
    
    # LP: max 1^T b s.t. C*b <= M*1, b >= 0
    # Convert to min -1^T b s.t. C*b <= M*1, b >= 0
    c_obj = -np.ones(n)
    A_ub = C
    b_ub = M * np.ones(2*n-1)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    
    if res.success:
        return np.maximum(res.x, 0.0)
    else:
        return None
