    a = np.asarray(a, dtype=np.float64)

Uses smooth-max projected gradient descent with FFT-based gradients,

periodic restarts to escape local minima. Explores the full sequence
directly (not block-structured) for maximum flexibility.
"""

import numpy as np

def fft_convolve(a):
    """Full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] ** 2], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max(b, beta):
    """Smooth max and softmax weights."""
    m = float(np.max(b))
    if m <= 0:
        return m, np.full(b.shape, 1.0 / b.size, dtype=np.float64)
    x = beta * (b - m)
    x = np.clip(x, -700.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z < 1e-300:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
        return m, w
    w = e / Z
    G = m + np.log(Z) / beta
    return G, w


def cross_corr_pw(w, p):
    """C[i] = sum_k p[k]*w[i+k] for i=0..n-1 via FFT."""
    n = p.size
    L = w.size
    need = L + n - 1
    size = 1
    while size < need:
        size <<= 1
    W = np.fft.rfft(w, size)
    PR = np.fft.rfft(p[::-1], size)
    conv_result = np.fft.irfft(W * PR, size)
    return np.ascontiguousarray(conv_result[n - 1:2 * n - 1], dtype=np.float64)


def smooth_obj_grad(p, beta):
    """Return (smooth_objective, gradient) for minimizing 2n*max(b) with sum(p)=1."""
    n = p.size
    b = fft_convolve(p)
    G, w = smooth_max(b, beta)
    C = cross_corr_pw(w, p)
    grad = 4.0 * n * C
    # Project gradient onto tangent space of simplex (remove mean)
    grad = grad - np.mean(grad)
    obj = 2.0 * n * G
    return obj, grad, G, w, b


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if target <= 0:
        return np.zeros_like(v)
    if not np.all(np.isfinite(v)):
        return np.full(n, target / n, dtype=np.float64)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0 or not np.isfinite(s):
        return np.full(n, target / n, dtype=np.float64)
    return q * (target / s)


def normalize_mp(p):
    """Normalize p to sum=1, non-negative."""
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0:
        return np.full(p.shape, 1.0 / p.size, dtype=np.float64)
    return p / s


def actual_score(p):
    """True objective value."""
    n = p.size
    S = float(np.sum(p
    b = fft_convolve(p)
    M = float(np.max(b))
    if S < 1e-15:
        return np.inf

def make_geometric(n, ratio, rng):
    """Geometric decay sequence."""
    if ratio >= 1.0:
        return np.full(n, 1.0 / n)
    a = ratio ** np.arange(n, dtype=np.float64)
    # Add small noise to break exact structure
    a *= (1.0 + 0.01 * rng.standard_normal(n))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_exponential(n, rate, rng):
    """Exponential decay."""
    a = np.exp(-rate * np.arange(n, dtype=np.float64) / n)
    a *= (1.0 + 0.01 * rng.standard_normal(n))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_power_law(n, alpha, rng):
    """Power law decay."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / (idx ** alpha)
    a *= (1.0 + 0.01 * rng.standard_normal(n))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_bimodal(n, center_frac, width, rng):
    """Two gaussian bumps."""
    x = np.arange(n, dtype=np.float64) / n
    c = center_frac
    a = np.exp(-0.5 * ((x - c) / width) ** 2) + np.exp(-0.5 * ((x - (1 - c)) / width) ** 2)
    a *= (1.0 + 0.01 * rng.standard_normal(n))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_tapered(n, power, rng):
    """Tapered (sinusoidal) window."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.sin(np.pi * x) ** power
    a *= (1.0 + 0.01 * rng.standard_normal(n))
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def make_random_structured(n, rng):
    """Random low-pass filtered sequence."""
    a = rng.standard_normal(n)
    # Low-pass filter in frequency domain
    size = 1
    while size < n:
        size <<= 1
    A = np.fft.rfft(a, size)
    # Keep low frequencies
    cutoff = n // 8
    A[cutoff:] *= 0.1
    a = np.fft.irfft(A, size)[:n]
    a = np.abs(a) + 0.1
    return a / a.sum()


def generate_starts(n, rng, existing_seq=None):
    """Generate diverse initial sequences."""
    starts = []
    # Uniform
    starts.append(np.full(n, 1.0 / n))
    # Geometric sequences
    for r in (0.99, 0.995, 0.999, 0.98, 0.95):
        starts.append(make_geometric(n, r, rng))
    # Exponential
    for rate in (1.0, 2.0, 5.0, 10.0, 20.0):
        starts.append(make_exponential(n, rate, rng))
    # Power law
    for alpha in (0.5, 1.0, 2.0, 3.0):
        starts.append(make_power_law(n, alpha, rng))
    # Tapered
    for pw in (0.5, 1.0, 2.0, 4.0):
        starts.append(make_tapered(n, pw, rng))
    # Bimodal
    for cf in (0.25, 0.35, 0.45):
        for w in (0.05, 0.1, 0.2):
            starts.append(make_bimodal(n, cf, w, rng))
    # Random structured
    for _ in range(4):
        starts.append(make_random_structured(n, rng))
    # Existing sequence if available
    if existing_seq is not None and len(existing_seq) > 0:
        es = np.asarray(existing_seq, dtype=np.float64)
        if es.shape[0] == n:
            starts.append(normalize_mp(es))
        elif es.shape[0] > n:
            # Downsample
            idx = np.linspace(0, es.shape[0] - 1, n).astype(int)
            starts.append(normalize_mp(es[idx]))
        else:
            # Upsample by interpolation
            from numpy import interp
            new_idx = np.linspace(0, es.shape[0] - 1, n)
            old_idx = np.arange(es.shape[0])
            starts.append(normalize_mp(interp(new_idx, old_idx, es)))
    return starts


def quick_refine(p0, deadline, beta_start=500.0, beta_end=200000.0, max_iter=5000):
    """Quick projected gradient refinement with increasing beta."""
    p = normalize_mp(p0)
    n = p.size
    best_p = p.copy()
    best_score = actual_score(p)
    step = 0.5
    it = 0
    no_improve = 0

    while it < max_iter and time.monotonic() < deadline:
        it += 1
        # Interpolate beta
        frac = min(1.0, it / max(1, max_iter * 0.8))
        beta = beta_start * (beta_end / beta_start) ** frac

        obj, grad, G, w, b = smooth_obj_grad(p, beta)
        cur_score = 2.0 * n * float(np.max(b))

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            break

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        # Backtracking line search
        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.01):
            alpha = step * mult
            q = project_simplex(p - alpha * grad / gnorm, 1.0)
            qb = fft_convolve(q)
            qm = float(np.max(qb))
            if 2.0 * n * qm < obj - 1e-12 * max(1.0, abs(obj)):
                p = q
                accepted = True
                break

        if accepted:
            step = min(step * 1.3, 5.0)
        else:
            step *= 0.4
            if step < 1e-8:
                break

    return best_p, best_score

def coordinate_descent(p, deadline, max_sweeps=50):

            # Try a few values for p[i]
            old_pi = p[i]

                p[i] = old_pi

def flatten_heuristic(p, deadline, n_steps=100):
    """Iteratively identify the peak autocorrelation and reduce it by adjusting p."""
    p = normalize_mp(p).copy()
    n = p.size
    best_p = p.copy()
    best_score = actual_score(p)

    for step in range(n_steps):
        if time.monotonic() >= deadline:
            break

        b = fft_convolve(p)
        M = float(np.max(b))
        peak_idx = int(np.argmax(b))
        S = float(np.sum(p))
        cur_score = 2.0 * n * M / (S * S)

        if cur_score < best_score:
            best_score = cur_score
            best_p = p.copy()

        # Find which p elements contribute most to b[peak_idx]
        # b[peak_idx] = sum_m p[m] * p[peak_idx - m] (valid indices)
        # The pair (m, peak_idx - m) contributes p[m] * p[peak_idx - m]
        # To reduce b[peak_idx], we should reduce large p values that pair up

        # Identify contributing pairs
        contrib = np.zeros(n)
        for m in range(n):
            k = peak_idx - m
            if 0 <= k < n:
                contrib[m] += p[m] * p[k]

        # Reduce the elements with highest contribution
        top_mask = contrib > 0.5 * np.max(contrib)
        n_top = max(1, int(np.sum(top_mask) * 0.1))
        top_indices = np.argsort(contrib)[-n_top:]

        # Reduce top contributors slightly, redistribute to low contributors
        old_sum = float(np.sum(p[top_indices]))
        reduce_factor = 0.9
        p[top_indices] *= reduce_factor
        redistribute = old_sum * (1 - reduce_factor)

        # Add to elements with lowest contribution
        bottom_indices = np.argsort(contrib)[:n_top]
        if len(bottom_indices) > 0:
            per_elem = redistribute / len(bottom_indices)
            p[bottom_indices] += per_elem

        p = normalize_mp(p)

    return best_p, best_score
