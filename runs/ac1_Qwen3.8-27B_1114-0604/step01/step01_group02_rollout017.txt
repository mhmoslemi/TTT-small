def gradient_at_peaks(a, c, k, temperature):
    """Compute gradient using top-k peaks with exponential weights."""
    n = len(a)
    S = a.sum()
    
    # Get top k peaks
    if k >= len(c):
        k = len(c)
    
    # Use log-sum-exp style for numerical stability
    top_vals = np.sort(c)[::-1][:k]
    top_vals = np.clip(top_vals, 0, None)
    max_val = top_vals[0]
    
    if max_val < 1e-15:
        return np.zeros(n), max_val, S
    
    # Exponential weights
    wts = np.exp((top_vals - max_val) / (temperature * max_val))
    wts = wts / wts.sum()
    
    # Get indices of top k
    # Use argpartition for efficiency
    part = np.argpartition(c, -k)[-k:]

    sort_idx = np.argsort(c[part])[::-1]
    top_idx = part[sort_idx]

        wi = float(wts[i])

            continue
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start > j_end:
            continue

        # We need a[ki - j] for j from j_start to j_end
        # ki - j goes from ki - j_start to ki - j_end (decreasing)
        grad[j_start:j_end+1] += wi * 2.0 * a[ki - np.arange(j_start, j_end+1)]

Multi-strategy optimizer to minimize f(a) = 2n*max(conv(a,a))/(sum a)^2.
Uses: (1) FFT-based convolution, (2) adaptive-temperature top-k softmax
gradient with Adam, (3) L-BFGS-B polish, (4) peak-targeted perturbation,
(5) multi-length exploration, (6) simulated-annealing restarts.

import numpy as np
import time
import scipy.optimize


def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def score_from_conv(n, c, S):
    if S < 1e-10:
        return np.inf
    return float(2.0 * n * np.max(c) / (S * S))


def get_topk(c, k):
    k_act = min(k, len(c))
    part = np.argpartition(c, -k_act)[-k_act:]
    order = np.argsort(c[part])[::-1]
    idx = part[order]
    vals = c[idx]
    return idx, vals


def compute_grad_softmax(a, c, k, temp):
    n = len(a)
    S = float(a.sum())
    idx, vals = get_topk(c, k)
    if vals[0] < 1e-15:
        return np.zeros(n)
    wts = np.exp((vals - vals[0]) / (temp * vals[0]))
    wts = wts / wts.sum()
    grad_c = np.zeros(n)
    for i in range(len(idx)):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-15:
            continue
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        jj = np.arange(js, je + 1)
        grad_c[jj] += wi * 2.0 * a[ki - jj]
    M = float(np.dot(vals, wts))
    S2 = S * S
    S3 = S2 * S
    return 2.0 * n * (grad_c / S2 - 2.0 * M / S3)


def compute_grad_subgrad(a, c):
    n = len(a)
    S = float(a.sum())
    ki = int(np.argmax(c))
    M = float(c[ki])
    grad_c = np.zeros(n)
    js = max(0, ki - n + 1)
    je = min(ki, n - 1)
    if js <= je:
        jj = np.arange(js, je + 1)
        grad_c[jj] = 2.0 * a[ki - jj]
    S2 = S * S
    S3 = S2 * S
    return 2.0 * n * (grad_c / S2 - 2.0 * M / S3)


def lse_objective(a, k, temp):
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    if S < 1e-10:
        return np.inf

    m = top[0]
    if m < 1e-15:
        return 0.0
    lse = m + temp * np.log(np.sum(np.exp((top - m) / temp)))
    return lse


def lse_gradient(a, k, temp):

    c = fast_self_conv(a)
    S = float(a.sum())
    idx, vals = get_topk(c, k)
    if vals[0] < 1e-15:

    return 2.0 * n * (grad_c / S2 - 2.0 * lse_val / S3)


def adam_step(a, g, m_buf, v_buf, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    t1 = b1 ** t
    t2 = b2 ** t
    m_buf[:] = b1 * m_buf + (1.0 - b1) * g
    v_buf[:] = b2 * v_buf + (1.0 - b2) * g * g
    mh = m_buf / (1.0 - t1)
    vh = v_buf / (1.0 - t2)
    a -= lr * mh / (np.sqrt(vh) + eps)


def clamp_nonneg(a):
    np.clip(a, 0.0, 1000.0, out=a)
    if a.sum() < 1e-4:
        a += 1e-4 / len(a)
    return a


def optimize_adam(a0, time_limit, rng, lr_init=0.1, topk_init=8, temp_init=0.5):
    n = a0.size
    a = a0.astype(np.float64).copy()
    clamp_nonneg(a)
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    t = 0
    lr = lr_init
    topk = topk_init
    temp = temp_init
    best_score = np.inf
    best_a = a.copy()
    no_improve = 0
    max_patience = 300
    t_start = time.monotonic()

    while time.monotonic() - t_start < time_limit:
        c = fast_self_conv(a)
        S = a.sum()
        score = score_from_conv(n, c, S)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            # Adapt: increase topk and decrease temp as we improve
            topk = min(32, topk_init + int(no_improve // 20))
        else:
            no_improve += 1

        if no_improve > max_patience:
            # Restart with perturbation
            scale = 0.1 + 0.15 * rng.random()
            a = best_a * (1.0 + scale * rng.standard_normal(n))
            clamp_nonneg(a)
            m_buf[:] = 0.0
            v_buf[:] = 0.0
            t = 0
            lr = max(0.01, lr * 0.8)
            no_improve = 0
            max_patience = max(50, max_patience - 20)
            temp = min(1.0, temp * 1.5)
            continue

        # Compute gradient (mix of softmax and subgradient)
        g_sm = compute_grad_softmax(a, c, topk, temp)
        g_sg = compute_grad_subgrad(a, c)
        mix = min(0.3, 0.1 + no_improve * 0.002)
        g = (1.0 - mix) * g_sm + mix * g_sg

        t += 1
        adam_step(a, g, m_buf, v_buf, t, lr)
        clamp_nonneg(a)

        # Adaptive temperature: cool over time
        if t % 200 == 0:
            temp = max(0.05, temp * 0.97)
            lr = max(0.005, lr * 0.995)

    return best_score, best_a


def optimize_lbfgs(a0, time_limit, k=16, temp=0.3):
    n = a0.size
    a = a0.astype(np.float64).copy()
    clamp_nonneg(a)

    def obj(x):
        x = np.clip(x, 0, 1000)
        c = fast_self_conv(x)
        S = x.sum()
        if S < 1e-10:
            return 1e10
        top = np.sort(c)[::-1][:k]
        top = np.clip(top, 0, None)
        m = top[0]
        if m < 1e-15:
            return 0.0
        lse = m + temp * np.log(np.sum(np.exp((top - m) / temp)))
        return float(2.0 * n * lse / (S * S))

    def grad(x):
        x = np.clip(x, 0, 1000)
        return lse_gradient(x, k, temp)

    t_start = time.monotonic()
    best_a = a0.copy()
    best_obj = obj(a0)

    try:
        result = scipy.optimize.minimize(
            obj, a0, method='L-BFGS-B',
            jac=grad,
            bounds=[(0, 1000)] * n,
            options={'maxiter': 5000, 'maxfun': 50000,
                    'ftol': 1e-10, 'gtol': 1e-8}
        )
        a_new = np.clip(result.x, 0, 1000)
        obj_new = obj(a_new)
        if obj_new < best_obj:
            best_a = a_new
            best_obj = obj_new
    except Exception:
        pass

    return best_obj, best_a


def peak_reduction(a, n_peaks=5, n_iter=50, rng=None):
    """Directly reduce the top peaks by adjusting contributing variables."""
    n = len(a)
    a = a.copy()
    for _ in range(n_iter):
        c = fast_self_conv(a)
        idx, vals = get_topk(c, n_peaks)
        for ki in idx[:min(3, len(idx))]:
            ki = int(ki)
            js = max(0, ki - n + 1)
            je = min(ki, n - 1)
            if js > je:
                continue
            # Identify which a[j] contribute most to this peak
            contributions = a[js:je+1] * a[ki - np.arange(js, je+1)]
            top_contrib = np.argsort(contributions)[::-1][:3]
            for tc in top_contrib:
                j = js + int(tc)
                if a[j] > 1e-6:
                    # Reduce a[j] slightly and compensate elsewhere
                    reduce = 0.02 * a[j]
                    a[j] -= reduce
                    # Add to a random position that doesn't increase peaks
                    rpos = rng.randint(n) if rng is not None else (j + n // 2) % n
                    a[rpos] += reduce
        a = np.clip(a, 0, 1000)
        if a.sum() < 1e-4:
            a += 1e-4 / n
    return a


def diversify_init(n, rng, prev=None):
    """Generate diverse initializations."""
    inits = []
    if prev is not None:
        if prev.size == n:
            inits.append(prev.copy())
        else:
            inits.append(np.interp(np.linspace(0, 1, n),
                                   np.linspace(0, 1, prev.size), prev))
    # Uniform
    inits.append(np.ones(n) / n)
    # Ramp
    inits.append(np.linspace(0.3, 1.0, n))
    inits.append(np.linspace(1.0, 0.3, n))
    # Two bumps
    a2 = np.zeros(n)
    a2[:n//5] = 1.0
    a2[4*n//5:] = 1.0
    inits.append(a2)
    # Four bumps
    a4 = np.zeros(n)
    a4[:n//8] = 1.0
    a4[n//4:3*n//8] = 0.5
    a4[5*n//8:7*n//8] = 0.5
    a4[7*n//8:] = 1.0
    inits.append(a4)
    # Cosine window
    x = np.linspace(0, 1, n)
    inits.append(0.5 + 0.5 * np.cos(2 * np.pi * x))
    inits.append(0.3 + 0.7 * np.cos(2 * np.pi * 2 * x))
    # Random
    inits.append(np.clip(0.5 + 0.5 * rng.standard_normal(n), 0.05, None))
    # Sparse (25% active)
    mask = (rng.random(n) < 0.25).astype(float)
    if mask.sum() > 0:
        inits.append(mask / mask.sum() * n)
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):

    best_score = np.inf

        if np.isfinite(score) and score < best_score:

    n_candidates = [2048, 4096, 8192]

        n_candidates.insert(0, prev.size)

    # Phase 1: Explore with Adam on multiple inits and n values
    phase1_time = 0.55 * (total_budget - 15)
    phase1_end = t_start + phase1_time
    print("Phase 1: Explore (until t=%.1fs)" % phase1_time)

    for n in n_candidates:
        if time.monotonic() > phase1_end:
            break
        inits = diversify_init(n, rng, prev)
        time_per_init = max(3.0, (phase1_end - time.monotonic()) / len(inits))
        print("  n=%d, %d initializations, %.1fs each" % (n, len(inits), time_per_init))
        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_end:
                break

            avail = min(time_per_init, phase1_end - time.monotonic())

    # Phase 2: Refine best with Adam (more peaks, lower temp) + peak reduction
    phase2_end = t_start + 0.78 * (total_budget - 15.0)

    if best_a is not None and time.monotonic() < phase2_end:

        for trial in range(3):

    print("FINAL: internal=%.6f, verified=%.6f, n=%d, time=%.1fs" %

    return out
