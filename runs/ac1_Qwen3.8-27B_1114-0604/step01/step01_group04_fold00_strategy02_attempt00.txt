def project_simplex(v):
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u[:-1]) + u[-1]
    idx = np.arange(1, n+1) - 1
    mask = u*idx > cssv[idx] - 1.0
    if ~mask.any():
        return np.ones(n)/n
    rho = np.max(np.nonzero(~mask)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
