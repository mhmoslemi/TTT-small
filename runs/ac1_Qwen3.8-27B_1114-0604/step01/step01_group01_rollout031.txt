```python
"""
Multi-resolution search for non-negative sequences minimizing
2*n*max(conv(a,a))/sum(a)^2.
"""

import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _fft_conv_full(a, b):
    """Full convolution of a and b using FFT when beneficial."""
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.convolve(a, b)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa * fb, L)[:nout]


def _cross_corr_weights(w, a):
    """
    Compute r_i = sum_{j=0}^{n-1} a[j] * w[i+j] for i = 0,...,n-1.
    w has length 2n-1.
    """
    n = len(a)
    # r = conv(a[::-1], w, 'full')[n-1 : 2n-1]
    total_len = n + len(w) - 1
    if n * len(w) <= 2000000:
        conv_res = np.convolve(a[::-1], w)
    else:
        L = _next_pow2(total_len)
        fa = np.fft.rfft(a[::-1], L)
        fw = np.fft.rfft(w, L)
        conv_res = np.fft.irfft(fa * fw, L)
    return conv_res[n-1 : n-1+n]


def _exact_score(a):
    """Compute the exact objective value (scale-invariant)."""
    a = np.asarray(a, dtype=float)
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    a = a / s
    n = len(a)
    c = _fft_conv_full(a, a)
    return float(2 * n * c.max())


def _smooth_score_and_weights(a, p):
    """
    Compute smooth max (power-sum) of conv(a,a) and the corresponding weights.
    Returns (score, weights) where weights sum to 1 and are concentrated on large conv values.
    """
    n = len(a)
    c = _fft_conv_full(a, a)
    c = np.maximum(c, 1e-15)  # avoid log(0) or 0^(p-1)
    
    if p == 1:
        # Just the sum (not a good max approximation, but useful)
        w = np.ones_like(c) / len(c)
        return float(2 * n * c.max()), w
    
    c_p = c ** p
    S = c_p.sum()
    if S < 1e-300:
        w = np.ones_like(c) / len(c)
        return float(2 * n * c.max()), w
    w = c_p / S  # These are the "weights" w_k ∝ c_k^p
    smooth_val = S ** (1.0 / p)  # This approximates max(c)
    score = float(2 * n * smooth_val)
    return score, w


def _gradient_direction(w, a):
    """
    Compute the gradient direction for the multiplicative update.
    Returns r_i = sum_k w_k * a_{k-i} (cross-correlation of w and a).
    """
    return _cross_corr_weights(w, a)


def _mmd_optimize(a0, time_limit, p_start=5.0, p_end=80.0, alpha0=0.5, rng=None):
    """
    Multiplicative Mirror Descent with power-sum smooth max.
    """
    if rng is None:
        rng = np.random.default_rng()
    
    a = np.asarray(a0, dtype=float).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        return a, float('inf')
    a = a / s
    n = len(a)
    
    deadline = time.monotonic() + time_limit
    best = a.copy()
    best_score = _exact_score(a)
    cur_score = best_score
    
    alpha = alpha0
    p = p_start
    stall = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        # Update p (smoothness parameter) for each iteration
        fraction = min(1.0, it / max(1, time_limit * 10))  # rough estimate
        p = p_start + (p_end - p_start) * fraction
        
        # Compute smooth score and weights
        smooth_s, w = _smooth_score_and_weights(a, p)
        
        # If p is large enough, also check exact
        if it % 20 == 0 or it == 1:
            exact_s = _exact_score(a)
            if exact_s < best_score:
                best_score = exact_s
                best = a.copy()
            cur_score = exact_s
        else:
            cur_score = smooth_s
        
        # Compute gradient direction
        r = _gradient_direction(w, a)
        r = np.maximum(r, 0)  # should be non-negative for non-negative a, w
        
        if r.max() < 1e-15:
            # Degenerate: perturb
            a = a * np.exp(rng.normal(0, 0.1, n))
            a = np.maximum(a, 0)
```

```python
"""
Multi-resolution MMD search for non-negative sequences minimizing
2*n*max(conv(a,a))/sum(a)^2.

Key improvements over previous approach:
- Multi-resolution (small→large) to find good basins before high-res refinement
- Power-sum smooth max (tunable, more flexible than pure softmax)
- Structured initializations: smoothed random binary, inverse-FFT, two-block
- Aggressive local polishing with adaptive block sizes
- Multiple parallel tracks (parent-based and exploratory)

import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _fft_conv_full(a, b):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 3000000:
        return np.convolve(a, b)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa * fb, L)[:nout]


def _cross_corr_valid(w, a):
    """r_i = sum_j a[j]*w[i+j] for i=0..n-1, w has length 2n-1."""
    n = len(a)
    total = n + len(w) - 1
    if n * len(w) <= 4000000:
        c = np.convolve(a[::-1], w)
    else:
        L = _next_pow2(total)
        fa = np.fft.rfft(a[::-1], L)
        fw = np.fft.rfft(w, L)
        c = np.fft.irfft(fa * fw, L)
    return c[n - 1: n - 1 + n]


def _sanitize(a):
    a = np.asarray(a, dtype=float)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-15:
        return None
    return a / s


def _exact_score(a):
    a = _sanitize(a)
    if a is None:
        return float('inf')
    n = len(a)
    c = _fft_conv_full(a, a)
    if c.size == 0:
        return float('inf')
    return float(2.0 * n * np.max(c))


def _smooth_and_weights(a, p):
    """Return (smooth_score, weights) using power-sum of order p."""
    n = len(a)
    c = _fft_conv_full(a, a)
    c = np.maximum(c, 1e-20)
    if p <= 1.0:
        w = np.ones_like(c) / c.size
        return float(2.0 * n * c.max()), w
    cp = c ** p
    S = cp.sum()
    if S < 1e-300:
        w = np.ones_like(c) / c.size
        return float(2.0 * n * c.max()), w
    w = cp / S
    smooth = S ** (1.0 / p)
    return float(2.0 * n * smooth), w


def _mmd(a0, tlimit, p_start=8.0, p_end=120.0, alpha0=1.0, rng=None):

def _upsample(a, new_n):
    n = len(a)
    if n == new_n:
        return a.copy()
    if n == 1:
        return np.ones(new_n, dtype=float)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n

    out = np.maximum(out, 0.0)
    s = out.sum()
    if s < 1e-15:
        return np.ones(new_n, dtype=float)
    return out / s


def _init_smoothed_binary(n, rng, sigma=3.0):
    """Random binary sequence smoothed with Gaussian."""
    a = rng.integers(0, 2, n).astype(float)
    rad = int(3 * sigma)
    k = np.arange(-rad, rad + 1, dtype=float)
    kern = np.exp(-0.5 * (k / sigma) ** 2)
    kern /= kern.sum()
    a = np.convolve(a, kern, mode='same')
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-15:
        return np.ones(n, dtype=float)
    return a / s


def _init_inv_fft(n, rng, n_iter=50):
    """Iteratively project: random phases → IFFT → clip non-negative → normalize."""
    a = None
    for _ in range(n_iter):
        phases = rng.uniform(0, 2 * np.pi, n)
        # Enforce conjugate symmetry for real result
        spec = np.exp(1j * phases)
        spec[0] = 1.0
        for k in range(1, (n // 2) + 1):
            spec[n - k] = np.conj(spec[k])
        a = np.real(np.fft.ifft(spec))
        a = np.clip(a, 0, None)
        s = a.sum()
        if s < 1e-10:
            continue
        a = a / s
    if a is None or a.sum() < 1e-10:
        return _init_smoothed_binary(n, rng)
    return a


def _init_two_block(n, rng):

    split = int(rng.integers(0.2 * n, 0.8 * n))
    ratio = float(rng.uniform(0.3, 3.0))

    # Smooth edges

    kern = np.exp(-0.5 * (k / (rad / 2.0)) ** 2)

    a = np.convolve(a, kern, mode='same')
    a = np.maximum(a, 1e-10)
    s = a.sum()
    if s < 1e-15:

def _init_bump(n, rng):
    """Sum of Gaussian bumps."""
    a = np.zeros(n, dtype=float)
    t = np.linspace(0, 1, n)
    K = int(rng.integers(2, 6))

    s = a.sum()

    return a / s

            if rem > 30:
                a2, sc2 = _local_blocks(a, min(20.0, rem * 0.3), rng=rng)
                _update_best(a2, sc2)

    # ============================================================
    # PHASE 5: Final verification and return
    # ============================================================
    if best_seq is not None:
        final_sc = _exact_score(best_seq)
        if final_sc < best_sc:
            best_sc = final_sc
        print(f"Final score: {best_sc:.6f}, n={len(best_seq)}")
        # Verify with official evaluator
        try:
            lst = [float(x) for x in best_seq]
            off_sc = float(evaluate_sequence(lst))
            if np.isfinite(off_sc) and off_sc < best_sc:
                best_sc = off_sc
                print(f"Official evaluator score: {off_sc:.6f}")
        except Exception:
            pass
        out = np.asarray(best_seq, dtype=float)
        out = np.where(np.isfinite(out), out, 0.0)
        out = np.clip(out, 0.0, 1000.0)
        return [float(x) for x in out]
    else:
        # Fallback
        n = 2048
        a = np.ones(n, dtype=float) / n
        return [float(x) for x in a]
```
