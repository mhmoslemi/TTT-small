```python
import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import gammaln

"""
Algorithm:
1. Generate analytical seeds (binomial coefficient sequence, cosine windows, etc.)
2. Subgradient descent with momentum and adaptive active set
3. L-BFGS polish in log-space
4. Return best sequence found
"""

def _fft_conv_linear(a, N):
    """Compute linear convolution a*a using FFT."""
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2*len(a)-1]

def _fast_score(a):
    """Compute 2n*max(conv(a,a))/(sum a)^2."""
    n = len(a)
    s = np.sum(a)
    if s < 1e-12:
        return np.inf
    N = 1 << (2*n-1).bit_length()
    c = _fft_conv_linear(a, N)
    return 2.0 * n * np.max(c) / (s*s)

def _binomial_sequence(n):
    """Generate a[k] = C(2k,k)/4^k normalized to sum=1."""
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i-1] * (2*i-1.0) / (2*i)
    a /= a.sum()
    return a

def _cosine_window(n):
    """Generate a Hamming-like window."""
    k = np.arange(n, dtype=np.float64)
    a = 0.54 + 0.46 * np.cos(2.0*np.pi*k/(n-1))
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a

def _gaussian_window(n, width=0.3):
    k = np.arange(n, dtype=np.float64)
    center = (n-1)/2.0
    a = np.exp(-0.5*((k-center)/(n*width))**2)
    a /= a.sum()
    return a

def _tapered_window(n, p=2.0):
    k = np.arange(n, dtype=np.float64)
    x = 2.0*k/(n-1) - 1.0
    a = np.maximum(1.0 - np.abs(x)**p, 1e-10)
    a /= a.sum()
    return a

def _subgradient_step(a, N, K=50, eta=1.0, momentum=0.9, velocity=None):
    """
    One step of subgradient descent.
    Returns (new_a, velocity, score, max_c).
    """
    n = len(a)
    c = _fft_conv_linear(a, N)
    m = np.max(c)
    score = 2.0 * n * m / (np.sum(a)**2)
    
    # Find active set: top-K lags
    # Use a threshold instead of fixed K for adaptivity
    threshold = m - 0.01 * m  # lags within 1% of max
    active = c >= threshold
    # Also ensure at least K and at most 5*K are active
    if np.sum(active) < K:
        # Take top-K
        idx = np.argsort(c)[-K:]
        active = np.zeros(len(c), dtype=bool)
        active[idx] = True
    elif np.sum(active) > 5*K:
        idx = np.argsort(c)[-5*K:]
        active = np.zeros(len(c), dtype=bool)
        active[idx] = True
    
    # Subgradient: g[i] = (2/|S|) * sum_{k in S} a[k-i]
    # This is (2/|S|) * (a * active_set)[i] truncated to length n
    indicator = np.zeros(N)
    indicator[:len(active)] = active.astype(np.float64)
    fi = np.fft.rfft(indicator, N)
    fa = np.fft.rfft(a, N)
    g_full = np.fft.irfft(fa * fi, N)[:n]
    g = 2.0 * g_full / np.sum(active)
    
    # Momentum
    if velocity is None:
        velocity = np.zeros(n)
    velocity = momentum * velocity + (1 - momentum) * g
    
    # Update
    a_new = a - eta * velocity
    
    # Project: non-negative and normalized
    a_new = np.maximum(a_new, 0.0)
    s = np.sum(a_new)
    if s < 1e-12:
        a_new = a  # revert
    else:
        a_new = a_new / s
    
    return a_new, velocity, score, m

def _subgradient_optimize(a0, deadline, N, n, eta0=1.0, K=50, momentum=0.8):
    """
    Run subgradient descent until deadline.
    """
    a = a0.copy()
    velocity = np.zeros(n)
    best_a = a0.copy()
    best_score = _fast_score(a0)
    
    eta = eta0
    t = 0
    no_improve = 0
    rng = np.random.default_rng(12345)
    
    while time.monotonic() < deadline:
        a, velocity, score, m = _subgradient_step(a, N, K=K, eta=eta, momentum=momentum, velocity=velocity)
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive step size
        if no_improve > 500:
            eta *= 0.9  # decrease step size
            no_improve = 0
        elif score < best_score:
            eta = min(eta * 1.05, 100.0)  # increase slightly
        
        # Random perturbation every 2000 iterations
        if t % 2000 == 1999 and t > 0:
            a += 0.01 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a /= a.sum()
            velocity = np.zeros(n)
        
        t += 1
        if t % 50000 == 0:
            print(f"  subgrad t={t} score={score:.6f} best={best_score:.6f} eta={eta:.4f}")
    
    return best_a, best_score

def _logspace_l_bfgs(a0, deadline, n, N, tau=1e-4):
    """
    L-BFGS polish in log-space with smooth max.
    """
    # Convert to log-space
    z0 = np.log(np.maximum(a0, 1e-300))
    z0 = z0 - z0.mean()
    z0 = np.clip(z0, -30, 30)
    
    # Objective function
    def obj_grad(z):
        z = np.clip(z, -30, 30)
        z_shifted = z - z.max()
        e = np.exp(z_shifted)
        a = e / e.sum()
        
        # Smooth max: tau * log(sum exp(c/tau))
        c = _fft_conv_linear(a, N)
        diff = (c - np.max(c)) / tau
        diff = np.clip(diff, -700, 0)
        es = np.exp(diff)
        se = es.sum()
        if se < 1e-300:
            lse = np.max(c)
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
        else:
            lse = np.max(c) + tau * np.log(se)
            p = es / se
        
        # Gradient
        fa = np.fft.rfft(a, N)
        fp = np.fft.rfft(p, N)
        h = np.fft.irfft(fp * np.conjugate(fa), N)[:n]
        s_h = np.dot(h, a)
        g = a * (h - s_h)
        
        # d/dz of softmax: a * (e_i - mean)
        # chain rule: dF/dz_i = sum_j dF/da_j * da_j/dz_i

"""
Algorithm: Minimize 2n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

Key insight: The optimal sequence approximates a[k] ∝ C(2k,k)/4^k (truncated
Taylor coefficients of (1-x)^(-1/2)), whose infinite convolution is perfectly
flat. We refine this analytical seed using:
1. Subgradient descent with momentum and adaptive active set
2. Coordinate descent polish
3. L-BFGS in log-space with smooth max
"""

import numpy as np
import time
from scipy.optimize import minimize


def _fft_conv(a, N):
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    n = len(a)
    return c[:2 * n - 1]


def _score_fast(a, N):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    c = _fft_conv(a, N)
    m = float(np.max(c))
    return 2.0 * n * m / (s * s)


def _binomial_seq(n):
    a = np.empty(n, dtype=np.float64)
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * (2.0 * i - 1.0) / (2.0 * i)
    a /= a.sum()
    return a


def _win_cos(n):
    k = np.arange(n, dtype=np.float64)
    a = 0.54 + 0.46 * np.cos(2.0 * np.pi * k / (n - 1))
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    return a


def _win_gauss(n, w=0.25):
    k = np.arange(n, dtype=np.float64)
    ctr = (n - 1) / 2.0
    a = np.exp(-0.5 * ((k - ctr) / (n * w)) ** 2)
    a /= a.sum()
    return a


def _win_taper(n, p=2.0):
    k = np.arange(n, dtype=np.float64)
    x = 2.0 * k / (n - 1) - 1.0
    a = np.maximum(1.0 - np.abs(x) ** p, 1e-10)
    a /= a.sum()
    return a


def _subgrad_step(a, N, K, eta, momentum, vel):
    n = len(a)
    c = _fft_conv(a, N)
    m = float(np.max(c))
    s = float(np.sum(a))
    score = 2.0 * n * m / (s * s)

    thr = m - max(0.02 * m, 1e-15)
    active = c >= thr
    nact = int(np.sum(active))
    if nact < K:
        idx = np.argsort(c)[-K:]
        active = np.zeros(len(c), dtype=bool)
        active[idx] = True
        nact = K
    elif nact > 8 * K:
        idx = np.argsort(c)[-8 * K:]
        active = np.zeros(len(c), dtype=bool)
        active[idx] = True
        nact = 8 * K

    ind = np.zeros(N, dtype=np.float64)
    ind[:len(active)] = active.astype(np.float64)
    fi = np.fft.rfft(ind, N)
    fa = np.fft.rfft(a, N)
    g = 2.0 * np.fft.irfft(fa * fi, N)[:n] / nact

    vel = momentum * vel + (1.0 - momentum) * g
    a_new = a - eta * vel
    a_new = np.maximum(a_new, 0.0)
    sn = float(np.sum(a_new))
    if sn < 1e-12:
        a_new = a.copy()
    else:
        a_new /= sn
    return a_new, vel, score, m


def _subgrad_opt(a0, deadline, N, n, eta0=5.0, K=50, momentum=0.85, seed=42):
    a = a0.copy()
    vel = np.zeros(n, dtype=np.float64)
    best = a0.copy()
    best_sc = _score_fast(a0, N)
    eta = eta0
    t = 0
    stagn = 0
    rng = np.random.default_rng(seed)
    K_cur = K
    mom_cur = momentum

    while time.monotonic() < deadline:
        a, vel, sc, m = _subgrad_step(a, N, K_cur, eta, mom_cur, vel)

        if sc < best_sc - 1e-14:
            best_sc = sc
            best = a.copy()
            stagn = 0
        else:
            stagn += 1

        if stagn > 800:
            if K_cur < 200:
                K_cur = min(K_cur + 25, 200)
            eta *= 0.7
            if mom_cur < 0.95:
                mom_cur = min(mom_cur + 0.02, 0.95)
            stagn = 0
        elif sc < best_sc:
            eta = min(eta * 1.02, 200.0)

        if t % 3000 == 2999:
            a += 0.005 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            sn = float(np.sum(a))
            if sn > 1e-12:
                a /= sn
            vel = np.zeros(n, dtype=np.float64)

        t += 1
        if t % 80000 == 0:
            print(f"    subgrad t={t} sc={sc:.6f} best={best_sc:.6f} eta={eta:.2f} K={K_cur}")

    return best, best_sc


def _coord_descent(a0, deadline, N, n, block=64, max_sweeps=10000):
    a = a0.copy()
    best = a0.copy()
    best_sc = _score_fast(a0, N)
    rng = np.random.default_rng(777)
    sweep = 0

    while time.monotonic() < deadline and sweep < max_sweeps:
        idx_blocks = np.array_split(np.arange(n), (n + block - 1) // block)
        rng.shuffle(idx_blocks)
        improved = False

        for blk in idx_blocks:
            if time.monotonic() > deadline:
                break
            old_block = a[blk].copy()
            sn_rest = float(np.sum(a)) - float(np.sum(old_block))

            for mult in (0.5, 0.7, 0.85, 1.2, 1.5, 2.0, 0.3):
                trial_block = old_block * mult
                trial_a = a.copy()
                trial_a[blk] = trial_block
                tsum = sn_rest + float(np.sum(trial_block))
                if tsum < 1e-12:
                    continue
                tnorm = copy_normalize(trial_a, tsum)
                sc = _score_fast(tnorm, N)
                if sc < best_sc - 1e-15:
                    best_sc = sc
                    best = tnorm.copy()
                    a = tnorm.copy()
                    old_block = tnorm[blk].copy()
                    sn_rest = float(np.sum(a)) - float(np.sum(old_block))
                    improved = True
                    break

        if not improved and sweep > 200:
            break
        sweep += 1

        if sweep % 100 == 0:
            print(f"    coord sweep={sweep} best={best_sc:.6f}")

    return best


def copy_normalize(a, s):
    out = a.copy()
    out /= s
    return out


def _lbfgs_polish(a0, deadline, N, n, tau=1e-4):
    z0 = np.log(np.maximum(a0, 1e-300))
    z0 = z0 - z0.mean()
    z0 = np.clip(z0, -25.0, 25.0)

    def obj_grad(z):
        z = np.clip(z, -25.0, 25.0)
        zs = z - z.max()
        e = np.exp(zs)
        s = float(np.sum(e))
        if s < 1e-300:
            return 1e12, np.zeros(n)
        a = e / s

        c = _fft_conv(a, N)
        cm = float(np.max(c))
        d = (c - cm) / tau
        d = np.clip(d, -700.0, 0.0)
        ed = np.exp(d)
        se = float(np.sum(ed))
        if se < 1e-300:
            lse = cm
            p = np.zeros_like(c)
            p[np.argmax(c)] = 1.0
        else:
            lse = cm + tau * np.log(se)
            p = ed / se

        fa = np.fft.rfft(a, N)
        fp = np.fft.rfft(p, N)
        h = np.fft.irfft(fp * fa, N)[:n]
        wh = float(np.dot(h, a))
        grad_a = 2.0 * (h - a * wh)
        gaa = float(np.dot(grad_a, a))
        grad_z = a * (grad_a - gaa)

        sc = float(n) * float(n)
        val = lse * sc + 1e-10 * 0.5 * float(np.dot(z, z))
        grad_z = grad_z * sc + 1e-10 * z

        if not np.isfinite(val):
            return 1e12, np.zeros(n)
        return val, grad_z

    best_a = a0.copy()
    try:
        res = minimize(
            obj_grad, z0, jac=True, method='L-BFGS-B',
            bounds=[(-25.0, 25.0)] * n,
            options={'maxfun': 3000, 'maxiter': 600, 'ftol': 1e-14, 'pgtol': 1e-12, 'disp': False}
        )
        if hasattr(res, 'x') and res.x is not None:
            zb = np.clip(res.x, -25.0, 25.0)
            ab = np.exp(zb - zb.max())
            ab /= ab.sum()
            sc_b = _score_fast(ab, N)
            if np.isfinite(sc_b) and sc_b < _score_fast(a0, N):
                best_a = ab
    except Exception:
        pass
    return best_a
