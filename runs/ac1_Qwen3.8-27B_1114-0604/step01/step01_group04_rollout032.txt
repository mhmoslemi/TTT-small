def project_simplex(v):
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    # We want to project onto {x >= 0, sum(x) = 1}
    # This is equivalent to finding theta such that sum(max(v - theta, 0)) = 1
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

1. LP-based direction finding for coarse optimization

3. Incremental convolution updates for O(n) per coordinate step

import numpy as np

def full_conv(a):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = len(a)
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def project_simplex(v):
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def update_b_inplace(b, a, i, old_val, new_val):
    """Update b=conv(a,a) in-place after a[i] changes from old_val to new_val."""

    delta_sq = new_val * new_val - old_val * old_val
    two_delta = 2.0 * delta
    
    k_min = i
    k_max = min(i + n - 1, L - 1)

    two_i = 2 * i

            # C[k] = b[k] - 2*old_val*a[k-i] for k in [i, i+n-1], k != 2i
            # For k not in range: just b[k]
            
            # Let me compute this efficiently.
            
            # max_out = max of b[k] for k not in [k_min, k_max]
            if k_min > 0:
                max_out_left = float(np.max(b[:k_min]))
            else:
                max_out_left = 0.0
            if k_max < L - 1:
                max_out_right = float(np.max(b[k_max+1:]))
            else:
                max_out_right = 0.0
            max_out = max(max_out_left, max_out_right, 0.0)
            
            # max_C_in_range = max of C[k] for k in [k_min, k_max], k != 2i, a[k-i] > 0
            # C[k] = b[k] - 2*old_val*a[k-i]
            # For a[k-i] > 0: C[k] = b[k] - 2*old_val*a[k-i] < b[k]
            # For a[k-i] = 0: C[k] = b[k] (and A[k] = 0, so no constraint from this k)
            
            # The max of C[k] over k in range with a[k-i] > 0:
            # C[k] = b[k] - 2*old_val*a[k-i]
            # We want max of this.
            
            if k_min <= k_max:
                a_slice = a[0:k_max+1-k_min]  # a[k-i] for k in [k_min, k_max]
                b_slice = b[k_min:k_max+1]
                # Exclude k = 2i
                if k_min <= two_i <= k_max:
                    # Handle the 2i position separately
                    idx_2i = two_i - k_min
                    # Max over non-2i positions where a_slice > 0
                    a_masked = a_slice.copy()
                    b_masked = b_slice.copy()
                    a_masked[idx_2i] = 0.0  # exclude 2i
                    C_slice = b_masked - 2.0 * old_val * a_masked
                    pos = a_masked > 1e-15
                    if np.any(pos):
                        max_C = float(np.max(C_slice[pos]))
                    else:
                        max_C = 0.0
                else:
                    C_slice = b_slice - 2.0 * old_val * a_slice
                    pos = a_slice > 1e-15
                    if np.any(pos):
                        max_C = float(np.max(C_slice[pos]))
                    else:
                        max_C = 0.0
            else:
                max_C = 0.0
            
            t_star = max(c_quad, max_C, max_out)
            
            # Now find x*:
            # x_upper = min(sqrt(t_star - c_quad), min_{k in range, a[k-i]>0, k!=2i} (t_star - C[k]) / (2*a[k-i]))
            
            if t_star < c_quad - 1e-15:
                new_val = old_val  # shouldn't happen
            else:
                x_upper = np.sqrt(max(0.0, t_star - c_quad))
                
                # Find min of (t_star - C[k]) / (2*a[k-i]) over valid k
                if k_min <= k_max:
                    a_slice = a[0:k_max+1-k_min]
                    b_slice = b[k_min:k_max+1]
                    if k_min <= two_i <= k_max:
                        idx_2i = two_i - k_min
                        a_m = a_slice.copy()
                        a_m[idx_2i] = 0.0
                        pos = a_m > 1e-15
                        if np.any(pos):
                            C_vals = b_slice[pos] - 2.0 * old_val * a_m[pos]
                            A_vals = 2.0 * a_m[pos]
                            bounds = (t_star - C_vals) / A_vals
                            x_upper = min(x_upper, float(np.min(bounds)))
                    else:
                        pos = a_slice > 1e-15
                        if np.any(pos):
                            C_vals = b_slice[pos] - 2.0 * old_val * a_slice[pos]
                            A_vals = 2.0 * a_slice[pos]
                            bounds = (t_star - C_vals) / A_vals
                            x_upper = min(x_upper, float(np.min(bounds)))
                
                new_val = max(0.0, x_upper)
            
            if abs(new_val - old_val) > 1e-15:
                update_b_inplace(b, a, i, old_val, new_val)
                a[i] = new_val
                new_max = float(np.max(b))
                if new_max < best_b:
                    best_b = new_max
            
            it += 1
    
    return best_b
