def project_simplex(v):

"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using Adam optimizer with
smoothed-max subgradient, multi-scale gradients, coordinate descent polish,
and multiple diverse initializations across several sequence lengths.
FFT-based convolution for O(n log n) per iteration.

import numpy as np
import time


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v.copy()
    v = v.astype(np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    # Ensure exact sum=1
    s = result.sum()
    if s > 0:
        result /= s
    return result


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def smooth_grad(a, b, tau):
    """
    Smoothed subgradient of max_k b_k using softmax weights.
    Returns gradient of sum_k w_k * b_k where w_k = softmax(b_k / tau).
    """
    n = a.size
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(b.size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= s
    # g_i = 2 * sum_{j=0}^{n-1} a_j * w_{i+j}
    # This is 2 * correlate(w, a, 'valid')
    g = 2.0 * np.correlate(w, a, mode='valid')
    return g


def multi_scale_grad(a, b, tau_base, n_scales):
    """Average subgradients at multiple temperature scales."""
    g_total = np.zeros(a.size)
    for s in range(n_scales):
        tau = tau_base * (2.0 ** s)
        g_total += smooth_grad(a, b, tau)
    return g_total / n_scales


def adam_optimize(a, time_budget, rng, lr_init=0.01):
    """
    Adam-based optimization of the smoothed max subgradient.
    a should be normalized to sum=1.
    Returns (best_a, best_score).
    """
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    lr = lr_init
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0

    tau_frac = 0.04
    n_scales = 3
    patience = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))

        tau = tau_frac * float(np.max(b))
        if tau < 1e-20:
            tau = 1e-20

        if n_scales > 1:
            g = multi_scale_grad(a, b, tau, n_scales)
        else:
            g = smooth_grad(a, b, tau)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        b_new = full_conv(a_new)
        new_sc = 2.0 * n * float(np.max(b_new))

        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        a = a_new

        # Adaptive: reduce learning rate if stalling
        if patience > 3000:
            lr *= 0.8
            if patience > 8000:
                # Restart
                a = project_simplex(best_a + 0.05 * rng.standard_normal(n))
                m = np.zeros(n)
                v = np.zeros(n)
                lr = lr_init * (0.5 + 0.5 * rng.random())
                tau_frac = 0.03 + 0.04 * rng.random()
                n_scales = 2 + int(rng.integers(0, 3))
                patience = 0

        # Periodic reheat
        if t % 10000 == 0:
            lr = lr_init * (0.5 + rng.random())
            tau_frac = 0.02 + 0.05 * rng.random()

    return best_a, best_sc


def coord_descent(a, time_budget):
    """
    Coordinate descent polish. For each coordinate, try a range of values
    and pick the best. Scales other coordinates to maintain sum=1.
    """
    n = a.size
    a = a.copy()
    best_a = a.copy()
    best_sc = normed_score(a)

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_budget:
                break

            others_sum = 1.0 - a[i]
            if others_sum < 1e-10:
                continue

            best_local_sc = best_sc
            best_local_a = None

            # Generate candidate values
            cands = [0.0, a[i] * 0.5, a[i] * 0.7, a[i] * 1.3, a[i] * 1.5,
                     a[i] + 0.005, a[i] - 0.005, a[i] + 0.01, a[i] - 0.01,
                     a[i] + 0.02, a[i] - 0.02,
                     1.0 / n * 0.5, 1.0 / n * 2.0]

            for t_val in cands:
                if t_val < 0 or t_val > 0.99:
                    continue
                a_trial = a.copy()
                a_trial[i] = t_val
                others_mask = np.ones(n, dtype=bool)
                others_mask[i] = False
                os = a_trial[others_mask].sum()
                if os < 1e-10:
                    continue
                a_trial[others_mask] *= (1.0 - t_val) / os
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s > 0:
                    a_trial /= s

                sc = normed_score(a_trial)
                if sc < best_local_sc - 1e-12:
                    best_local_sc = sc
                    best_local_a = a_trial.copy()

            if best_local_a is not None:
                a = best_local_a
                if best_local_sc < best_sc:
                    best_sc = best_local_sc
                    best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_sc


def init_uniform(n):
    return np.ones(n) / n


def init_u_shape(n):
    """High at edges, low in middle."""
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i + 1, n - i).astype(np.float64)
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_quadratic(n):
    """Quadratic: maximum at edges."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / c) ** 2 + 0.1
    return a / a.sum()


def init_gaussian(n, rng):
    """Gaussian centered."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    sigma = n * (0.2 + 0.3 * rng.random())
    a = np.exp(-((i - c) / sigma) ** 2) + 0.01
    return a / a.sum()


def init_two_cluster(n, rng):
    """Two Gaussian clusters."""
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.2 + 0.2 * rng.random())
    c2 = n * (0.6 + 0.2 * rng.random())
    sigma = n * (0.05 + 0.1 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2) + 0.01
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def init_fejer_sqrt(n):
    """Square root of Fejér kernel: triangular shape."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / (c + 0.5)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def init_cosine(n, rng):
    i = np.arange(n, dtype=np.float64)
    freq = 1.0 + rng.integers(0, 4)
    a = (1.0 + freq * np.cos(2.0 * np.pi * i / max(n - 1, 1))) ** 2
    a += 1e-4
    return a / a.sum()


def get_inits(n, rng):

        init_two_cluster(n, rng),
        init_random(n, rng),
        init_fejer_sqrt(n),
        init_cosine(n, rng),

    return inits

"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using Adam optimizer with
smoothed-max subgradient, multi-scale gradients, multiplicative updates,
coordinate descent polish, and multiple diverse initializations across
several sequence lengths. FFT-based convolution for O(n log n) per iteration.
"""

import numpy as np
import time


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v.copy()
    v = v.astype(np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    result = np.maximum(v - theta, 0.0)
    s = result.sum()
    if s > 0:
        result /= s
    return result


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def smooth_grad(a, b, tau):
    """
    Smoothed subgradient of max_k b_k using softmax weights.
    g_i = 2 * sum_{j=0}^{n-1} a_j * w_{i+j}
    """
    n = a.size
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(b.size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= s
    g = 2.0 * np.correlate(w, a, mode='valid')
    return g


def multi_scale_grad(a, b, tau_base, n_scales):
    """Average subgradients at multiple temperature scales."""
    g_total = np.zeros(a.size)
    for s in range(n_scales):
        tau = tau_base * (2.0 ** s)
        g_total += smooth_grad(a, b, tau)
    return g_total / n_scales


def adam_optimize(a, time_budget, rng, lr_init=0.008):
    """Adam-based optimization with smoothed max subgradient."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    lr = lr_init
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    tau_frac = 0.04
    n_scales = 3
    patience = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))

        tau = tau_frac * float(np.max(b))
        if tau < 1e-20:
            tau = 1e-20

        if n_scales > 1:
            g = multi_scale_grad(a, b, tau, n_scales)
        else:
            g = smooth_grad(a, b, tau)

        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = project_simplex(a_new)

        b_new = full_conv(a_new)
        new_sc = 2.0 * n * float(np.max(b_new))

        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_new.copy()
            patience = 0
        else:
            patience += 1

        a = a_new

        if patience > 3000:
            lr *= 0.85
            if patience > 8000:
                a = project_simplex(best_a + 0.05 * rng.standard_normal(n))
                m = np.zeros(n)
                v = np.zeros(n)
                lr = lr_init * (0.5 + 0.5 * rng.random())
                tau_frac = 0.03 + 0.04 * rng.random()
                n_scales = 2 + int(rng.integers(0, 3))
                patience = 0

        if t % 8000 == 0:
            lr = lr_init * (0.5 + rng.random())
            tau_frac = 0.02 + 0.05 * rng.random()

    return best_a, best_sc


def multiplicative_optimize(a, time_budget, rng):
    """Multiplicative weight update: reduce coordinates contributing to peak."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    eta = 0.15
    tau_frac = 0.05
    patience = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))

        tau = tau_frac * float(np.max(b))
        if tau < 1e-20:
            tau = 1e-20

        # Compute contribution vector
        shift = b / tau
        shift -= shift.max()
        w = np.exp(shift)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(b.size)
            w[int(np.argmax(b))] = 1.0
        else:
            w /= ws

        contrib = np.correlate(w, a, mode='valid')
        max_c = contrib.max()
        if max_c > 1e-15:
            update = 1.0 - eta * contrib / max_c
            update = np.maximum(update, 0.1)
            a = a * update
            a = np.maximum(a, 1e-12)
            a /= a.sum()

        b_new = full_conv(a)
        new_sc = 2.0 * n * float(np.max(b_new))

        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
            if patience > 2000:
                a = best_a.copy()
                eta *= 0.9
                patience = 0
            if patience > 8000:
                a = project_simplex(best_a + 0.04 * rng.standard_normal(n))
                eta = 0.1 + 0.1 * rng.random()
                tau_frac = 0.03 + 0.04 * rng.random()
                patience = 0

        if patience % 5000 == 0 and patience > 0:
            eta = 0.05 + 0.15 * rng.random()
            tau_frac = 0.03 + 0.05 * rng.random()

    return best_a, best_sc


def coord_descent(a, time_budget):
    """Coordinate descent polish with candidate values."""
    n = a.size
    a = a.copy()
    best_a = a.copy()
    best_sc = normed_score(a)

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_budget:
                break

            target_sum_others = 1.0 - a[i]
            if target_sum_others < 1e-10:
                continue

            best_local_sc = best_sc
            best_local_a = None

            cands = [0.0,
                     a[i] * 0.5, a[i] * 0.7, a[i] * 1.3, a[i] * 1.5,
                     a[i] + 0.005, a[i] - 0.005,
                     a[i] + 0.01, a[i] - 0.01,
                     a[i] + 0.02, a[i] - 0.02,
                     0.5 / n, 2.0 / n]

            for t_val in cands:
                if t_val < -1e-10 or t_val > 0.999:
                    continue
                t_val = max(0.0, t_val)
                a_trial = a.copy()
                a_trial[i] = t_val
                others_mask = np.ones(n, dtype=bool)
                others_mask[i] = False
                os = a_trial[others_mask].sum()
                if os < 1e-10:
                    continue
                a_trial[others_mask] *= (1.0 - t_val) / os
                a_trial = np.maximum(a_trial, 0.0)
                s = a_trial.sum()
                if s > 0:
                    a_trial /= s

                sc = normed_score(a_trial)
                if sc < best_local_sc - 1e-12:
                    best_local_sc = sc
                    best_local_a = a_trial.copy()

            if best_local_a is not None:
                a = best_local_a
                if best_local_sc < best_sc:
                    best_sc = best_local_sc
                    best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_a, best_sc


def init_uniform(n):
    return np.ones(n) / n


def init_u_shape(n):
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i + 1, n - i).astype(np.float64)
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_quadratic(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / c) ** 2 + 0.1
    return a / a.sum()


def init_gaussian(n, rng):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    sigma = n * (0.2 + 0.3 * rng.random())
    a = np.exp(-((i - c) / sigma) ** 2) + 0.01
    return a / a.sum()


def init_two_cluster(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.2 + 0.2 * rng.random())
    c2 = n * (0.6 + 0.2 * rng.random())
    sigma = n * (0.05 + 0.1 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2) + 0.01
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def init_fejer_sqrt(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 - np.abs(i - c) / (c + 0.5)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def init_cosine(n, rng):
    i = np.arange(n, dtype=np.float64)
    freq = 1 + int(rng.integers(0, 4))
    a = (1.0 + freq * np.cos(2.0 * np.pi * i / max(n - 1, 1))) ** 2
    a += 1e-4
    return a / a.sum()


def init_power_edge(n, rng):
    """a_i proportional to (i+1)^(-alpha) * ((n-i)/n)^(beta) for edge-heavy."""
    alpha = rng.random() * 2.0
    beta = rng.random() * 2.0
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** (-alpha) + (n + 1 - i) ** (-alpha)
    a += 0.01
    return a / a.sum()


def get_inits(n, rng):
    inits = [
        init_uniform(n),
        init_u_shape(n),
        init_quadratic(n),
        init_gaussian(n, rng),
        init_two_cluster(n, rng),
        init_random(n, rng),
        init_fejer_sqrt(n),
        init_cosine(n, rng),
        init_power_edge(n, rng),
    ]
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses Adam optimizer, multiplicative updates, coordinate descent polish,
    multiple sequence lengths, and diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_overall = None
    best_overall_sc = np.inf

    # Try to load previous best
    prev_a = None
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size >= 10 and np.all(np.isfinite(h)) and h.sum() > 0.01:
            prev_a = h / h.sum()
            prev_sc = normed_score(prev_a)
            print(f"[init] Previous best: n={h.size}, score={prev_sc:.6f}")
    except Exception:
        pass

    if prev_a is not None and prev_sc < 2.0:
        best_overall = prev_a.copy()
        best_overall_sc = prev_sc

    # Sequence lengths to try
    n_list = [128, 256, 512]
    if prev_a is not None:
        np_ = prev_a.size
        if np_ not in n_list and 64 <= np_ <= 1024:
            n_list.append(np_)
        n_list = sorted(set(n_list))

    total_time = max(1.0, deadline - time.monotonic())
    print(f"[plan] n_list={n_list}, total_time={total_time:.1f}s")

    # Time allocation: give more time to smaller n (proven) but also explore larger
    raw_w = []
    for nv in n_list:
        if nv == 128:
            raw_w.append(1.5)
        elif nv == 256:
            raw_w.append(1.0)
        elif nv == 512:
            raw_w.append(0.7)
        else:
            raw_w.append(0.5)
    total_w = sum(raw_w)
    weights = [w / total_w for w in raw_w]

    for idx_n, nv in enumerate(n_list):
        if time.monotonic() > deadline:
            break

        time_n = weights[idx_n] * total_time * 0.82
        inits = get_inits(nv, rng)

        if prev_a is not None and prev_a.size == nv:
            inits.insert(0, prev_a.copy())

        n_inits = len(inits)
        inits_per = max(4.0, time_n / n_inits)

        for ii, a_init in enumerate(inits):
            if time.monotonic() > deadline:
                break

            tb = min(inits_per, max(4.0, deadline - time.monotonic() - 5.0))
            if tb < 4.0:
                break

            # Phase 1: Adam optimization (60% of time)
            adam_time = tb * 0.6
            a_best, sc_best = adam_optimize(a_init, adam_time, rng, lr_init=0.008)

            # Phase 2: Multiplicative optimization (20% of time)
            if time.monotonic() < deadline - 5.0:
                mult_time = tb * 0.2
                a_m, sc_m = multiplicative_optimize(a_best, mult_time, rng)
                if sc_m < sc_best:
                    a_best = a_m
                    sc_best = sc_m

            # Phase 3: Coordinate descent polish (20% of time)
            if time.monotonic() < deadline - 5.0:
                cd_time = min(tb * 0.2, max(2.0, deadline - time.monotonic() - 5.0))
                a_cd, sc_cd = coord_descent(a_best, cd_time)
                if sc_cd < sc_best:
                    a_best = a_cd
                    sc_best = sc_cd

            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = a_best.copy()
                print(f"[opt] n={nv}, init={ii}, score={sc_best:.6f} (NEW BEST)")
            else:
                print(f"[opt] n={nv}, init={ii}, score={sc_best:.6f}")

        print(f"[opt] n={nv} done. Overall best: {best_overall_sc:.6f}")

    # Final intensive polish
    if best_overall is not None and time.monotonic() < deadline - 10.0:
        remain = deadline - time.monotonic() - 5.0
        if remain > 30.0:
            print(f"[polish] remaining={remain:.1f}s, n={best_overall.size}, score={best_overall_sc:.6f}")

            # Adam with small lr
            pol1_time = remain * 0.35
            a_p, sc_p = adam_optimize(best_overall, pol1_time, rng, lr_init=0.002)
            if sc_p < best_overall_sc:
                best_overall_sc = sc_p
                best_overall = a_p.copy()
                print(f"[polish] Adam small-lr improved to {sc_p:.6f}")

            # Multiplicative
            if time.monotonic() < deadline - 10.0:
                pol2_time = remain * 0.25
                a_p2, sc_p2 = multiplicative_optimize(best_overall, pol2_time, rng)
                if sc_p2 < best_overall_sc:
                    best_overall_sc = sc_p2
                    best_overall = a_p2.copy()
                    print(f"[polish] Multiplicative improved to {sc_p2:.6f}")

            # Coordinate descent (most of remaining time)
            if time.monotonic() < deadline - 5.0:
                pol3_time = max(5.0, deadline - time.monotonic() - 5.0)
                a_p3, sc_p3 = coord_descent(best_overall, pol3_time)
                if sc_p3 < best_overall_sc:
                    best_overall_sc = sc_p3
                    best_overall = a_p3.copy()
                    print(f"[polish] CoordDescent improved to {sc_p3:.6f}")

            # One more Adam pass after coord descent
            if time.monotonic() < deadline - 5.0:
                pol4_time = max(5.0, (deadline - time.monotonic() - 5.0) * 0.5)
                a_p4, sc_p4 = adam_optimize(best_overall, pol4_time, rng, lr_init=0.003)
                if sc_p4 < best_overall_sc:
                    best_overall_sc = sc_p4
                    best_overall = a_p4.copy()
                    print(f"[polish] Final Adam improved to {sc_p4:.6f}")

                # Final coord descent
                if time.monotonic() < deadline - 3.0:
                    pol5_time = max(2.0, deadline - time.monotonic() - 3.0)
                    a_p5, sc_p5 = coord_descent(best_overall, pol5_time)
                    if sc_p5 < best_overall_sc:
                        best_overall_sc = sc_p5
                        best_overall = a_p5.copy()
                        print(f"[polish] Final CD improved to {sc_p5:.6f}")

    # Fallback
    if best_overall is None:
        best_overall = np.ones(128) / 128
        best_overall_sc = 2.0

    # Final verification
    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"[final] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")

    return result
