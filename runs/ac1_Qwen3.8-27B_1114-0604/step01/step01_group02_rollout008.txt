import numpy as np

def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    if M_fft < 64:
        M_fft = 64
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score_from_conv(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def get_topk(c, k):
    k = min(k, len(c))
    idx = np.argpartition(c, -k)[-k:]
    vals = c[idx]
    order = np.argsort(vals)[::-1]
    return idx[order], vals[order]

def compute_gradient(a, c, idx, wts, n, S, Mw):
    """Gradient of 2n * Mw / S^2 where Mw = sum_i wts[i] * c[idx[i]]."""
    g = np.zeros(n)
    for i in range(idx.size):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-15:
            continue
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js <= je:
            g[js:je+1] += wi * 2.0 * a[js:je+1][::-1]
    return 2.0 * n * (g / (S*S) - 2.0 * Mw * a / (S**3))

def adam_update(a, g, m, v, t, lr):
    b1, b2, eps = 0.9, 0.999, 1e-8
    m[:] = b1 * m + (1.0 - b1) * g
    v[:] = b2 * v + (1.0 - b2) * g * g
    mh = m / (1.0 - b1**t)
    vh = v / (1.0 - b2**t)
    a -= lr * mh / (np.sqrt(vh) + eps)

def backtracking_line_search(a, g, n, f0, max_steps=30, c1=1e-4):
    """Backtracking line search: find alpha such that f(a - alpha*g) <= f0 - c1*alpha*||g||^2."""
    if not np.isfinite(f0):
        return a, f0
    gnorm = np.dot(g, g)
    if gnorm < 1e-30:
        return a, f0
    alpha = 1.0

        S_try = a_try.sum()
        f_try = compute_score_from_conv(n, c_try, S_try)
        if np.isfinite(f_try) and f_try <= f0 - c1 * alpha * gnorm:
            return a_try, f_try
        alpha *= 0.5
    return a, f0

    return a

Optimize non-negative step function to minimize 2n*max(conv(a,a))/(sum a)^2.

Strategy:
  1. Start from parent sequence (if available) plus diverse inits (chirps, ramps, cosines).
  2. Adam optimizer with progressive temperature annealing on soft-max peak surrogate.
  3. Backtracking line search on the TRUE objective (not just the surrogate).
  4. Periodic restarts with geometrically larger perturbations to escape local minima.
  5. Coordinate descent polish (try candidate values per element).
  6. Try multiple sequence lengths (1024, 2048, 4096) and keep the best.
"""

import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT, returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    if M_fft < 64:
        M_fft = 64
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def score_from_conv(n, c, S):
    """Compute objective 2n*max(c)/S^2."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def get_topk(c, k):
    """Return (indices, values) of top-k largest entries of c."""
    k = min(k, len(c))
    idx = np.argpartition(c, -k)[-k:]
    vals = c[idx]
    order = np.argsort(vals)[::-1]
    return idx[order], vals[order]


def gradient_surrogate(a, c, idx, wts, n, S, Mw):
    """
    Gradient of 2n * Mw / S^2 where Mw = sum_i wts[i] * c[idx[i]].
    Uses the identity: d conv(k)/d a[j] = 2*a[k-j] (for valid k-j).
    """
    g = np.zeros(n)
    for i in range(idx.size):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-15:
            continue
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js <= je:
            g[js:je + 1] += wi * 2.0 * a[js:je + 1][::-1]
    S2 = S * S
    S3 = S2 * S
    return 2.0 * n * (g / S2 - 2.0 * Mw * a / S3)


def adam_update(a, g, m, v, t, lr):
    """In-place Adam update."""
    b1, b2, eps = 0.9, 0.999, 1e-8
    m[:] = b1 * m + (1.0 - b1) * g
    v[:] = b2 * v + (1.0 - b2) * g * g
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)
    a -= lr * mh / (np.sqrt(vh) + eps)


def backtracking(a, g, n, f0, max_steps=25, c1=1e-4):
    """
    Backtracking line search on the true objective.
    Returns (a_new, f_new). If no step accepted, returns original.
    """
    if not np.isfinite(f0):
        return a, f0
    gnorm2 = float(np.dot(g, g))
    if gnorm2 < 1e-30:
        return a, f0
    alpha = 1.0
    for _ in range(max_steps):
        a_try = np.clip(a - alpha * g, 0.0, 1000.0)
        S_try = float(a_try.sum())
        if S_try < 1e-3:
            alpha *= 0.5
            continue
        c_try = fast_self_conv(a_try)
        f_try = score_from_conv(n, c_try, S_try)
        if np.isfinite(f_try) and f_try <= f0 - c1 * alpha * gnorm2:
            return a_try, f_try
        alpha *= 0.5
    return a, f0


def coordinate_polish(a, n, rounds=1):
    """
    Simple coordinate descent: for each element try a few candidate values.
    Only modify elements where improvement is found.
    """
    for _ in range(rounds):
        c_full = fast_self_conv(a)
        S_full = float(a.sum())
        f_full = score_from_conv(n, c_full, S_full)
        if not np.isfinite(f_full):
            break
        current_best = f_full
        changed = False
        for j in range(n):
            old = a[j]
            if old < 1e-12:
                continue
            # Try several candidates
            candidates = [0.0, old * 0.4, old * 0.7, old * 1.4, old * 2.0]
            best_f = current_best
            best_val = old
            for cand in candidates:
                if cand == old:
                    continue
                a[j] = cand
                S_tmp = float(a.sum())
                if S_tmp < 1e-3:
                    a[j] = old
                    continue
                c_tmp = fast_self_conv(a)
                f_tmp = score_from_conv(n, c_tmp, S_tmp)
                if np.isfinite(f_tmp) and f_tmp < best_f - 1e-10:
                    best_f = f_tmp
                    best_val = cand
            if best_val != old:
                a[j] = best_val
                current_best = best_f
                changed = True
            else:
                a[j] = old
        if not changed:
            break
    return a


def build_inits(n, rng, parent=None):
    """Build a diverse set of initializations for length n."""
    inits = []
    x = np.linspace(0.0, 1.0, n)

    if parent is not None and parent.size == n:
        inits.append(("parent", np.clip(parent, 0.01, 1000.0).copy()))

    inits.append(("uniform", np.ones(n)))
    inits.append(("ramp_up", 0.3 + 0.7 * x))
    inits.append(("ramp_down", 0.3 + 0.7 * (1.0 - x)))
    inits.append(("ramp_sym", 0.3 + 0.7 * np.abs(2.0 * x - 1.0)))

    for bw in [0.5, 1.0, 2.0, 4.0]:
        inits.append((f"chirp_{bw}", np.abs(1.0 + bw * (x - 0.5)) + 0.05))

    for freq in [1, 2, 3, 5]:
        inits.append((f"cos_{freq}", np.abs(np.cos(np.pi * freq * x)) + 0.05))

    for nb in [4, 8, 16]:
        per = n // nb
        vals = 0.3 + 0.7 * rng.random(nb)
        blk = np.repeat(vals, per)
        if len(blk) < n:
            blk = np.concatenate([blk, vals[:n - len(blk)]])
        inits.append((f"block_{nb}", blk[:n]))

    inits.append(("rand", np.clip(0.3 + 0.7 * rng.random(n), 0.01, 1000.0)))
    inits.append(("sparse_2", np.where(rng.random(n) < 0.3, rng.random(n) * 2 + 0.1, 0.01)))

    return inits


def optimize_one(a0, n, t_end, rng, lr_init=0.4, topk_init=3, topk_end=25):
    """
    Run Adam with progressive temperature annealing and backtracking on true objective.
    Returns best (score, array) found.
    """
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a.sum() < 1e-3:
        a += 0.1
    m = np.zeros(n)
    v = np.zeros(n)
    t_adam = 0
    lr = lr_init
    best_local = np.inf
    best_local_a = a.copy()
    no_improve = 0
    patience = 200
    t_start = time.monotonic()

    while time.monotonic() < t_end:
        c = fast_self_conv(a)
        S = float(a.sum())
        if S < 1e-3:
            a += 0.01
            continue
        score = score_from_conv(n, c, S)

        if np.isfinite(score) and score < best_local:
            best_local = score
            best_local_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            # Restart with perturbation
            scale = 0.05 + 0.25 * rng.random()
            a = best_local_a * (1.0 + scale * rng.standard_normal(n))
            a = np.clip(a, 0.0, 1000.0)
            if a.sum() < 1e-3:
                a += 0.1
            m[:] = 0.0
            v[:] = 0.0
            t_adam = 0
            lr = max(0.01, lr * 0.6)
            no_improve = 0
            patience = max(50, patience // 2)
            continue

        # Progressive temperature: start broad, narrow over time
        elapsed_frac = min(1.0, (time.monotonic() - t_start) / max(1e-6, t_end - t_start))
        temp_frac = 0.4 * (1.0 - elapsed_frac) + 0.005 * elapsed_frac
        topk = int(topk_init + (topk_end - topk_init) * elapsed_frac)
        topk = max(1, topk)

        idx, vals = get_topk(c, topk)
        top_val = float(vals[0])
        temp_abs = temp_frac * top_val
        if temp_abs < 1e-12:
            temp_abs = 1e-12
        wts = np.exp((vals - vals[0]) / temp_abs)
        wts_sum = wts.sum()
        if wts_sum < 1e-30:
            wts = np.ones(topk) / topk
        else:
            wts /= wts_sum
        Mw = float(np.dot(vals, wts))

        g = gradient_surrogate(a, c, idx, wts, n, S, Mw)
        t_adam += 1
        adam_update(a, g, m, v, t_adam, lr)
        a = np.clip(a, 0.0, 1000.0)

        # Periodic line search on true objective (every 20 steps to save time)
        if t_adam % 20 == 0:
            c_check = fast_self_conv(a)
            S_check = float(a.sum())
            f_check = score_from_conv(n, c_check, S_check)
            if np.isfinite(f_check) and f_check < best_local:
                best_local = f_check
                best_local_a = a.copy()
                no_improve = 0

        # Learning rate schedule: slow decay
        if t_adam % 300 == 0:
            lr = max(0.005, lr * 0.97)

    return best_local, best_local_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = max(1.0, float(budget_s))
    deadline = start + total - 15.0

    best_score = np.inf
    best_a = None

    def track(a, label=""):
        nonlocal best_score, best_a
        n = a.size
        c = fast_self_conv(a)
        S = float(a.sum())
        sc = score_from_conv(n, c, S)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()
            elapsed = time.monotonic() - start
            print(f"[{elapsed:6.1f}s] {label}: {sc:.6f}")
        return sc

    # Load parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if 200 <= p.size <= 16384:
                parent = np.clip(p, 0.0, 1000.0)
                ps = track(parent, "parent")
                print(f"  Parent score: {ps:.6f}")
    except Exception as e:
        print(f"  No parent: {e}")

    # Choose lengths to try
    lengths = []
    if parent is not None:
        lengths.append(parent.size)
    for L in [1024, 2048, 4096, 8192]:
        if L not in lengths:
            lengths.append(L)
    # Sort by likely quality: parent first, then smaller lengths
    lengths = list(dict.fromkeys(lengths))

    # Time allocation: give more time to likely-better lengths
    # If parent exists, spend 40% on its length, split rest among others
    if parent is not None and parent.size in lengths:
        idx_p = lengths.index(parent.size)
        time_fracs = [0.15] * len(lengths)
        time_fracs[idx_p] = 1.0 - 0.15 * (len(lengths) - 1)
    else:
        n_L = len(lengths)
        time_fracs = [1.0 / n_L] * n_L

    available = deadline - start

    for Li, n in enumerate(lengths):
        t0 = time.monotonic()
        t_end = t0 + time_fracs[Li] * available
        t_end = min(t_end, deadline)
        if t0 >= deadline:
            break

        print(f"\n=== n={n} (budget {t_end - t0:.1f}s) ===")

        # Build and sort initializations
        inits = build_inits(n, rng, parent)
        scored = []
        for name, a0 in inits:
            a0 = np.clip(a0.astype(np.float64), 0.01, 1000.0)
            c0 = fast_self_conv(a0)
            s0 = score_from_conv(n, c0, float(a0.sum()))
            if np.isfinite(s0):
                scored.append((s0, name, a0))
        scored.sort(key=lambda x: x[0])

        # Allocate optimization time: top gets 50%, next 30%, rest 20%
        n_top = min(len(scored), 4)
        if n_top == 1:
            fracs = [1.0]
        elif n_top == 2:
            fracs = [0.55, 0.45]
        elif n_top == 3:
            fracs = [0.45, 0.30, 0.25]
        else:
            fracs = [0.40, 0.25, 0.20, 0.15]

        opt_time = t_end - time.monotonic()

        for rank in range(n_top):
            if time.monotonic() >= t_end:
                break
            s0, name, a0 = scored[rank]
            alloc = fracs[rank] * opt_time
            t_alloc_end = time.monotonic() + alloc

            print(f"  [{name}] init={s0:.6f} time={alloc:.1f}s")
            sc, aa = optimize_one(a0, n, t_alloc_end, rng,
                                  lr_init=0.4, topk_init=3, topk_end=25)
            track(aa, f"n={n}/{name}")

            # If this is the best so far, do coordinate polish
            if aa is not None and best_a is not None and best_a.size == n and \
               time.monotonic() < deadline - 60:
                polished = coordinate_polish(aa.copy(), n, rounds=1)
                track(polished, f"n={n}/{name}/pol")

    # Final polish on global best
    if best_a is not None and time.monotonic() < deadline - 10:
        n_best = best_a.size
        print(f"\nFinal coordinate polish on best (n={n_best})...")
        polished = coordinate_polish(best_a.copy(), n_best, rounds=2)
        track(polished, "final_polish")

        # One more refinement round with very small lr
        if time.monotonic() < deadline - 5:
            sc, aa = optimize_one(best_a, n_best, time.monotonic() + min(30, deadline - time.monotonic() - 5),
                                  rng, lr_init=0.01, topk_init=5, topk_end=15)
            track(aa, "micro_refine")
            polished2 = coordinate_polish(aa.copy(), n_best, rounds=1)
            track(polished2, "micro_polish")

    # Fallback
    if best_a is None:
        best_a = np.ones(2048)

    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    verified = float(evaluate_sequence(out))
    print(f"\n=== DONE: internal={best_score:.6f} verified={verified:.6f} "
          f"len={len(out)} time={time.monotonic()-start:.1f}s ===")
    return out
