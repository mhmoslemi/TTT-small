"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using exact coordinate descent
with adaptive restarts, multiple initializations, and multi-scale exploration.
Each coordinate update solves a 1D convex subproblem exactly via golden section search.
"""

import numpy as np
import time
import sys

def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def normed_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(b.max())

def actual_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = full_conv(a)
    return 2.0 * n * float(b.max()) / (s * s)

def golden_section(f, a, b, xatol=1e-10, maxiter=200):
    """Golden section search for unimodal function f on [a, b]. Returns (x, f(x))."""
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(maxiter):
        if b - a < xatol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    x = (a + b) / 2.0
    return x, f(x)

def coord_descent_step(a, conv, k):
    """
    Perform one exact coordinate descent step for coordinate k.
    Returns new value of a[k] and updated conv.
    """
    n = a.size
    ak = a[k]
    
    # Compute d[l] = linear coefficient of a[k] in conv[l]
    # and identify the quadratic position
    d = np.zeros(2 * n - 1)
    quad_l = -1
    l_min = max(0, k)
    l_max = min(2 * n - 2, k + n - 1)
    for l in range(l_min, l_max + 1):
        idx = l - k
        if 0 <= idx <= n - 1:
            if l == 2 * k:
                quad_l = l
            else:
                d[l] = 2.0 * a[idx]
    
    # Compute base
    base = conv - ak * d
    if quad_l >= 0:
        base[quad_l] = conv[quad_l] - ak * ak
    
    # Minimize f(t) = max(base[l] + t*d[l] for l != quad_l, base[quad_l] + t^2) over t >= 0
    # Note: d[quad_l] = 0, so base[l] + t*d[l] = base[l] for l = quad_l if we don't special-case
    # But we want base[quad_l] + t^2, not base[quad_l] + t*0
    
    def f(t):
        vals = base + t * d
        if quad_l >= 0:
            vals[quad_l] = base[quad_l] + t * t
        return vals.max()
    
    # Search range: [0, max(ak * 3, 3.0 / n)]
    # Actually, since sum(a) = 1 and a >= 0, a[k] <= 1. Use [0, 1] for safety.
    # But for efficiency, use a tighter bound.
    upper = max(ak * 5.0, 4.0 / n, 0.01)
    upper = min(upper, 1.0)
    
    new_ak, _ = golden_section(f, 0.0, upper, xatol=1e-12, maxiter=100)
    
    # Update conv
    conv_new = conv.copy()
    delta = new_ak - ak
    conv_new += delta * d
    if quad_l >= 0:
        conv_new[quad_l] = base[quad_l] + new_ak * new_ak
    
    return new_ak, conv_new

def coord_descent(a, time_budget, rng, verbose=False):
    """
    Run coordinate descent until time budget is exhausted.
    a is assumed to be non-negative and normalized to sum 1.
    Returns (best_a, best_score).
    """
    n = a.size
    a = np.maximum(a.astype(np.float64), 0.0).copy()
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
        s = 1.0
    a = a / s
    
    best_a = a.copy()
    best_sc = normed_score(a)
    conv = full_conv(a)
    
    t0 = time.monotonic()
    sweep = 0
    stall = 0
    
    while time.monotonic() - t0 < time_budget:
        # Random coordinate ordering
        order = rng.permutation(n)
        
        for k in order:
            new_ak, conv = coord_descent_step(a, conv, k)
            a[k] = new_ak
        
        # Renormalize
        s = a.sum()
        if s > 1e-10:
            a = a / s
        conv = full_conv(a)
        
        cur_sc = normed_score(a)
        if cur_sc < best_sc - 1e-12:
            best_sc = cur_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        sweep += 1
        
        if verbose and (sweep % 5 == 0):
            elapsed = time.monotonic() - t0
            print(f"    CD sweep={sweep}, score={cur_sc:.6f}, best={best_sc:.6f}, t={elapsed:.1f}s")

        if stall > 10:
            a = best_a + 0.03 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            conv = full_conv(a)
            stall = 0
        elif stall > 20:
            a = best_a + 0.1 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            conv = full_conv(a)
            stall = 0
        elif stall > 40:

            a = a / a.sum()
            conv = full_conv(a)
            stall = 0
            best_a = a.copy()  # Reset best to current (forced exploration)

def init_uniform(n):

def init_u_shape(n):
    """High at edges, low in middle."""
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()

def init_inverted_u(n):
    """High in middle, low at edges."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = np.exp(-((i - c) / (n / 4.0)) ** 2) + 0.01
    return a / a.sum()
