import numpy as np
import time

def fft_conv(a):
    """Linear convolution a * a using FFT."""
    n = a.size
    if n == 0:
        return np.empty(0)
    if n == 1:
        return np.array([a[0]**2])
    need = 2*n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F*F, size)[:need]
    return np.maximum(b, 0)

def smooth_max_obj(b, beta):
    """Smooth max of b returns (smooth_max_value, weights)."""
    m = np.max(b)
    x = beta * (b - m)
    x = np.clip(x, -700, 0)
    e = np.exp(x)
    Z = np.sum(e)
    if Z <= 0:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1
        return m, w
    w = e / Z
    G = m + np.log(Z) / beta
    return G, w

def p_norm_obj(b, p):
    """p-norm of b and its 'weights' (proportional to b^(p-1))."""
    b_max = np.max(b)
    if b_max <= 0:
        return 0, np.ones_like(b) / len(b)
    # Scale to avoid overflow
    bn = b / b_max
    vals = bn**p
    Z = np.sum(vals)
    if Z <= 0 or not np.isfinite(Z):
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1
        return b_max, w
    w = vals / Z
    G = b_max * Z**(1.0/p)
    return G, w

def corr_vec(p, w):
    """Compute s_i = sum_j p_j * w_{i+j} for i=0..n-1."""
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    # s_i = C[m-1-i] for i=0..n-1
    s = C[m-n:m][::-1].copy()
    return np.maximum(s, 0)

def project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    n = v.size
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    if target <= 0:
        return np.zeros(n)
    lo = np.min(v) - 1
    hi = np.max(v) + 1

        s = np.sum(np.maximum(v-mid, 0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5*(lo+hi)
    q = np.maximum(v-tau, 0)
    s = np.sum(q)
    if s <= 0 or not np.isfinite(s):
        return np.full(n, target/n)
    return q * (target/s)

    p = np.where(np.isfinite(p), p, 0)
    s = np.sum(p)

    return p / s

Minimize 2n*max(a*a)/(sum(a))^2 for non-negative sequences.

Strategy:
1. Start from height_sequence_1 (previous best) for guaranteed head start.
2. Multi-phase adaptive gradient projection with momentum:
   - Coarse phase (low beta) for broad optimization
   - Sharp phase (high beta) for peak reduction
   - Peak-shaving phase (extreme beta) for final polish
3. Spectral diversification: modify sequence in frequency domain to escape local minima.
4. p-norm objective as alternative landscape.
5. Adaptive step-size with backtracking line search.
6. Periodic restarts with structured perturbations when stagnated.

import numpy as np
import time


def _fft_conv(a):
    """Linear convolution a*a via FFT."""
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def _smooth_max(b, beta):
    """Smooth max: returns (value, weights)."""
    m = float(np.max(b))
    if m < 1e-300:
        w = np.ones_like(b) / len(b)
        return m, w
    x = np.clip(beta * (b - m), -700.0, 0.0)
    e = np.exp(x)
    Z = float(e.sum())
    if Z < 1e-300:
        w = np.zeros_like(b)
        w[int(np.argmax(b))] = 1.0
        return m, w
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return G, w


def _pnorm_obj(b, p):
    """p-norm objective: returns (value, normalized b^p weights)."""
    bm = float(np.max(b))
    if bm < 1e-300:
        w = np.ones_like(b) / len(b)
        return 0.0, w
    bn = b / bm
    vals = np.power(bn, p)
    Z = float(vals.sum())
    if Z < 1e-300:
        w = np.zeros_like(b)
        w[int(np.argmax(b))] = 1.0
        return bm, w
    w = vals / Z
    G = bm * Z ** (1.0 / p)
    return G, w


def _corr(p, w):
    """s_i = sum_j p_j * w_{i+j} for i=0..n-1 (via FFT)."""
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    Wr = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * Wr, size)[:need]
    s = C[m - n:m][::-1].copy()
    return np.maximum(s, 0.0)


def _proj_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target} via bisection."""
    n = v.size
    if n == 0:
        return v
    v = np.nan_to_num(v, nan=0.0, posinf=1e6, neginf=-1e6)
    lo = float(v.min()) - 1.0
    hi = float(v.max()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if float(np.sum(np.maximum(v - mid, 0.0))) > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(q.sum())
    if s < 1e-300:
        return np.full(n, target / n, dtype=np.float64)
    return (q * (target / s)).astype(np.float64)


def _norm1(p):
    """Normalize to sum=1, non-negative."""
    p = np.maximum(np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0), 0.0)
    s = float(p.sum())
    if s < 1e-300:

    return (p / s).astype(np.float64)


def _score(p):

    best_p = p.copy()
    best_s = _score(p)

    step = 1.0
    stagnate = 0
    it = 0

def _spectral_perturb(p, n, strength, rng):
    """Perturb in frequency domain: modify magnitude/phase, project back to non-negative simplex."""
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(p, size)
    # Random phase perturbation
    phase = rng.uniform(-strength, strength, F.size)
    F_perturbed = F * np.exp(1j * phase)
    # Small magnitude perturbation
    mag_perturb = rng.uniform(1.0 - strength * 0.5, 1.0 + strength * 0.5, F.size)
    F_perturbed = F_perturbed * mag_perturb
    q = np.fft.irfft(F_perturbed, size)[:n]
    return _norm1(q)


def _structured_perturb(p, n, rng, type_id=0):
    """Apply a structured perturbation to escape local minima."""
    if type_id == 0:
        # Gaussian noise
        return _norm1(p + 0.01 * rng.standard_normal(n))
    elif type_id == 1:
        # Block-wise shift
        block = max(1, n // 16)
        q = p.copy()
        for _ in range(4):
            b0 = rng.integers(0, n - block)
            eps = 0.02 * rng.uniform(-1, 1)
            q[b0:b0 + block] += eps
        return _norm1(q)
    elif type_id == 2:
        # Mirror flip of a segment
        seg_len = n // 4
        s0 = rng.integers(0, n - seg_len)
        q = p.copy()
        q[s0:s0 + seg_len] = p[s0:s0 + seg_len][::-1]
        return _norm1(q)
    else:
        # Taper modification
        x = np.linspace(0, 1, n)
        amp = rng.uniform(0.05, 0.2)
        q = p * (1.0 + amp * np.sin(2 * np.pi * x * rng.uniform(1, 5)))
        return _norm1(q)


def _diverse_starts(n, rng, base_p, count=6):
    """Generate diverse starting points from a base sequence."""
    starts = [base_p.copy()]

    # Spectral perturbations
    for s in (0.1, 0.3, 0.5):
        starts.append(_spectral_perturb(base_p, n, s, rng))

    # Structured perturbations
    for t in range(3):
        starts.append(_structured_perturb(base_p, n, rng, type_id=t))

    # Fresh structured starts
    x = np.linspace(0, 1, n)

        q = np.exp(a * (x - 0.5))
        starts.append(_norm1(q))

    q = np.maximum(q, 0.01)
    starts.append(_norm1(q))

    # Concentrated center
    q = np.exp(-50.0 * (x - 0.5) ** 2)
    starts.append(_norm1(q))

    return starts[:count + 4]

    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0  # Reserve 30s for final evaluation

    # Get initial sequence
    n = 8192
    p0 = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p0 = np.array(height_sequence_1, dtype=np.float64).flatten()
            if p0.size == n:
                p0 = _norm1(p0)
            elif p0.size > 0:
                # Resample to our target length
                p0 = _norm1(p0)
                # Interpolate to length n
                from scipy.interpolate import interp1d
                old_x = np.linspace(0, 1, p0.size)
                new_x = np.linspace(0, 1, n)
                f = interp1d(old_x, p0, kind='cubic', fill_value='extrapolate')
                p0 = _norm1(np.maximum(f(new_x), 0))
            else:
                p0 = None
    except Exception:
        p0 = None

    if p0 is None:
        p0 = np.full(n, 1.0 / n, dtype=np.float64)
        # Break symmetry
        x = np.linspace(0, 2 * np.pi, n, dtype=np.float64)
        p0 = _norm1(p0 * (1.0 + 0.1 * np.sin(x)))

    best_p = p0.copy()
    best_score = _score(p0)
    print(f"Starting: n={n}, initial score={best_score:.8f}")

    if not np.isfinite(best_score):
        best_p = np.full(n, 1.0 / n, dtype=np.float64)
        best_score = _score(best_p)
        print(f"Fallback to uniform: score={best_score:.8f}")

    # Phase 1: Coarse optimization (beta=1000) - 15% time
    phase_deadline = time.monotonic() + 0.15 * (budget_s - 30)
    print(f"\nPhase 1: Coarse (beta=1000)")
    p1, s1 = _grad_refine(best_p, 1000.0, min(phase_deadline, deadline), n, mu=0.8)
    if s1 < best_score:
        best_score = s1
        best_p = p1
        print(f"  Phase 1 best: {best_score:.8f}")

    # Phase 2: Medium optimization (beta=10000) - 20% time
    phase_deadline = time.monotonic() + 0.20 * (budget_s - 30)
    print(f"\nPhase 2: Medium (beta=10000)")
    p2, s2 = _grad_refine(best_p, 10000.0, min(phase_deadline, deadline), n, mu=0.85)
    if s2 < best_score:
        best_score = s2
        best_p = p2
        print(f"  Phase 2 best: {best_score:.8f}")

    # Phase 3: Sharp optimization (beta=100000) - 25% time
    phase_deadline = time.monotonic() + 0.25 * (budget_s - 30)
    print(f"\nPhase 3: Sharp (beta=100000)")
    p3, s3 = _grad_refine(best_p, 100000.0, min(phase_deadline, deadline), n, mu=0.9)
    if s3 < best_score:
        best_score = s3
        best_p = p3
        print(f"  Phase 3 best: {best_score:.8f}")

    # Phase 4: Diversification + re-refinement - 25% time
    print(f"\nPhase 4: Diversification")
    phase_deadline = min(time.monotonic() + 0.25 * (budget_s - 30), deadline)
    starts = _diverse_starts(n, rng, best_p, count=4)
    alloc = (phase_deadline - time.monotonic()) / len(starts)
    for i, s_p in enumerate(starts):
        sub_dl = time.monotonic() + alloc
        if sub_dl > deadline:
            sub_dl = deadline
        if time.monotonic() >= deadline - 10:
            break
        s_cur = _score(s_p)
        print(f"  Start {i}: initial={s_cur:.8f}")
        pi, si = _grad_refine(s_p, 50000.0, sub_dl, n, mu=0.85, max_stagnate=1000)
        if si < best_score:
            best_score = si
            best_p = pi
            print(f"  Start {i} improved: {best_score:.8f}")

    # Phase 5: Peak shaving (beta=1000000) - 15% time
    phase_deadline = min(time.monotonic() + 0.15 * (budget_s - 30), deadline)
    print(f"\nPhase 5: Peak shaving (beta=1000000)")
    p5, s5 = _grad_refine(best_p, 1000000.0, phase_deadline, n, mu=0.9, max_stagnate=5000)
    if s5 < best_score:
        best_score = s5
        best_p = p5
        print(f"  Phase 5 best: {best_score:.8f}")

    # Phase 6: p-norm refinement (alternative landscape) - remaining time
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\nPhase 6: p-norm refinement (p=80)")
        p6, s6 = _grad_refine(best_p, 100000.0, deadline, n, mu=0.9,
                              use_pnorm=True, p_power=80, max_stagnate=3000)
        if s6 < best_score:
            best_score = s6
            best_p = p6
            print(f"  Phase 6 best: {best_score:.8f}")

    # Final: one more pass with moderate beta from best
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"\nFinal polish (beta=200000)")
        pf, sf = _grad_refine(best_p, 200000.0, deadline, n, mu=0.9, max_stagnate=2000)
        if sf < best_score:
            best_score = sf
            best_p = pf
            print(f"  Final polish best: {best_score:.8f}")

    # Cleanup and return
    best_p = _norm1(best_p)
    result = [float(x) for x in best_p]

    # Official verification
    try:
        off = float(evaluate_sequence(result))
    except Exception:
        off = float('inf')

    if not np.isfinite(off) or off > 2.0 + 1e-6:
        print("WARNING: official score invalid, returning fallback")
        return [1.0] * n

    print(f"\nFinal result: local_score={best_score:.8f}, official={off:.8f}")
    return result
