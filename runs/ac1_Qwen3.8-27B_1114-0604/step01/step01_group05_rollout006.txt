def fft_convolution(a):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    b = np.ascontiguousarray(b, dtype=np.float64)
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    """Smooth max of b via log-sum-exp and normalized softmax weights."""
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float("inf"), np.empty(0, dtype=np.float64)
    m = float(np.max(b))
    if not np.isfinite(m):
        return float("inf"), np.zeros_like(b, dtype=np.float64)
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b, dtype=np.float64)
        w /= float(b.size)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w


def fast_score(p):
    """Fast local objective: 2*n*max(p*p)/(sum(p))^2."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float("inf")
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float("inf")
    b = fft_convolution(p)
    if b.size == 0:
        return float("inf")
    M = float(np.max(b))
    if not np.isfinite(M):
        return float("inf")
    return float(2.0 * n * M / (S * S))


def corr_pw(p, w):
    """Return s_i = sum_j p_j * w_{i+j} for i=0..n-1 (cross-correlation)."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    if n == 1:
        return np.array([float(np.dot(p, w[:1]))], dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n:m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)


def project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target} via bisection on threshold."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    if target <= 0.0:
        return np.zeros_like(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(v.shape, target / float(v.size), dtype=np.float64)
    low = float(np.min(v)) - 1.0
    high = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (low + high)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        q = np.full(v.shape, target / float(v.size), dtype=np.float64)
    else:
        q *= target / s
    return q


def normalize_simplex(p):
    """Clip to nonnegative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size), dtype=np.float64)
    p *= 1.0 / s
    return p


def block_to_p(n, B, c):
    """Convert relative block log-weights c (length B-1) to full sequence."""
    L = n // B
    z = np.empty(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        tmp = np.zeros(B - 1, dtype=np.float64)
        m = min(tmp.size, c.size)
        tmp[:m] = c[:m]
        c = tmp
        z[:-1] = c
    z[-1] = 0.0
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0.0 or not np.isfinite(s):
        wb = np.ones(B, dtype=np.float64) / float(B)
    else:
        wb /= s
    H = wb / float(L)
    return np.repeat(H, L)


def block_objective(c, n, B, beta):
    """Objective and gradient for block-constant sequences."""
    L = n // B
    z = np.empty(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        tmp = np.zeros(B - 1, dtype=np.float64)
        m = min(tmp.size, c.size)
        tmp[:m] = c[:m]
        c = tmp
        z[:-1] = c
    z[-1] = 0.0
    if not np.all(np.isfinite(z)):
        return float("inf"), np.zeros(B - 1, dtype=np.float64)
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0.0 or not np.isfinite(s):
        wb = np.ones(B, dtype=np.float64) / float(B)
    else:
        wb /= s
    H = wb / float(L)
    p = np.repeat(H, L)
    b = fft_convolution(p)
    G, w = smooth_max_and_weights(b, beta)
    if not np.isfinite(G):
        return float("inf"), np.zeros(B - 1, dtype=np.float64)
    svec = corr_pw(p, w)

    grad = n * gradz[:-1]

def lp_peak_reduction(f, M, solver=cp.GLOP):
    """
    Solve LP: max sum(g) s.t. (f*g)_j <= M for all j, g >= 0.
    Returns the optimal g (normalized to sum 1) or None on failure.
    """
    n = f.size
    if n < 4:
        return None
    try:
        g_var = cp.Variable(n, non_neg=True)
        # Build constraint: A @ g <= M * ones(2n-1)
        # A[j, k] = f[j-k] if 0 <= j-k <= n-1 else 0
        # Use sparse construction
        rows = []
        cols = []
        vals = []
        for j in range(2 * n - 1):
            k_min = max(0, j - (n - 1))
            k_max = min(j, n - 1)
            for k in range(k_min, k_max + 1):
                rows.append(j)
                cols.append(k)
                vals.append(f[j - k])
        from scipy.sparse import csr_matrix
        A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
        rhs = M * np.ones(2 * n - 1)
        constraints = [A @ g_var <= rhs]
        objective = cp.Maximize(cp.sum(g_var))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=solver)
        if g_var.value is not None and np.all(np.isfinite(g_var.value)):
            g = np.asarray(g_var.value, dtype=np.float64)
            g = np.maximum(g, 0.0)
            s = float(np.sum(g))
            if s > 0:
                g /= s
                return g
    except Exception:
        pass
    return None


def lp_phase(f, deadline, max_iters=40, n_lp=4096):
    """
    LP-based peak reduction phase.
    If f is longer than n_lp, work with the first n_lp elements or downsample.
    Returns improved sequence (same length as input).
    """
    if not HAVE_CVPY:
        return f
    n = f.size
    if n <= n_lp:
        work = f.copy()
        use_lp_len = n
    else:
        # Downsample by averaging blocks
        L = n // n_lp
        if L < 2:
            return f
        work = f[:n_lp * L].reshape(n_lp, L).mean(axis=1)
        use_lp_len = n_lp
    work = normalize_simplex(work)
    best_work = work.copy()
    best_score = fast_score(work)
    t_mix = 0.3
    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break
        b = fft_convolution(work)
        M = float(np.max(b))
        if not np.isfinite(M) or M <= 0:
            break
        g = lp_peak_reduction(work, M)
        if g is None:
            break
        # Mix and check
        accepted = False
        for t in (t_mix, 0.2, 0.1, 0.05, 0.02):
            candidate = (1 - t) * work + t * g
            candidate = normalize_simplex(candidate)
            cb = fft_convolution(candidate)
            cM = float(np.max(cb))
            if cM < M * 0.999:
                work = candidate
                cur_score = fast_score(work)
                if cur_score < best_score:
                    best_score = cur_score
                    best_work = work.copy()
                accepted = True
                break
        if accepted:
            if t < t_mix:
                t_mix = max(0.01, t * 0.7)
        else:
            t_mix *= 0.5
            if t_mix < 0.005:
                break
        if it % 5 == 0:
            print(f"    LP iter {it+1}: M={M:.8e} score={best_score:.6f} t={t_mix:.4f}")
    # Upsample back if needed
    if n > use_lp_len:
        L = n // use_lp_len
        result = np.repeat(best_work, L)[:n]
    else:
        result = best_work
    return normalize_simplex(result)

            restart_count += 1
            noise_scale = 0.02 * np.std(p)
            p = normalize_simplex(p + noise_scale * rng.normal(0, 1, n))
            no_improve = 0
            step = 1.0
            beta = max(500.0, beta * 0.5)
            restart_threshold = max(500, restart_threshold // 2)
            continue

        if no_improve > restart_threshold * 3:
            break

        # Increase beta adaptively
        if it % 200 == 0:
            beta = min(beta_max, beta * beta_growth)

        svec = corr_pw(p, w)
        svec = np.maximum(svec, 0.0)
        g = 2.0 * svec
        grad = g - float(np.mean(g))

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-14:
            break

        accepted = False
        base = step
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005):
            alpha = base * mult
            q = project_simplex(p - alpha * grad, 1.0)
            qb = fft_convolution(q)
            G2, _ = smooth_max_and_weights(qb, beta)
            if not np.isfinite(G2):
                continue
            if float(n) * G2 < cur_smooth - 1e-12:
                p = q
                cur_smooth = float(n) * G2
                cur_score = float(2.0 * n * float(np.max(qb)))
                accepted = True
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_p = p.copy()
                    no_improve = 0
                break
        if not accepted:
            step *= 0.5
            if step < 1e-8:
                break
        else:
            if step < 3.0:
                step = min(3.0, step * 1.3)

    return best_p, best_score


def structured_starts(n, rng, max_count=10):
    """A variety of analytic and random starting profiles."""
    starts = []
    p = np.full(n, 1.0 / float(n), dtype=np.float64)
    starts.append(p)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    for a in (-3.0, -2.0, -1.0, 1.0, 2.0, 3.0):
        z = a * (x - 0.5)
        pp = np.exp(z)
        pp /= pp.sum()
        starts.append(pp)
    # Triangle-like
    tri = 1.0 - 2.0 * np.abs(x - 0.5)
    tri = np.maximum(tri, 0.01)
    tri /= tri.sum()
    starts.append(tri)
    # Sine-modulated
    for freq in (2, 4, 8):
        z = np.sin(freq * np.pi * x)
        pp = np.exp(0.5 * z)
        pp /= pp.sum()
        starts.append(pp)
    # Random smooth
    size = 1
    while size < n:
        size <<= 1
    for _ in range(3):
        zc = rng.normal(size=size) + 1j * rng.normal(size=size)
        z = np.real(np.fft.ifft(zc))[:n]
        z = z - float(np.mean(z))
        z = 2.0 * np.tanh(z)
        pp = np.exp(z)
        pp /= pp.sum()
        starts.append(pp)
    return starts[:max_count]


def choose_best_starts(starts, count):
    """Return up to count starts with the lowest fast_score."""
    scored = []
    for p in starts:
        sc = fast_score(p)
        if np.isfinite(sc):
            scored.append((sc, p))
    scored.sort(key=lambda x: x[0])
    return [s[1] for s in scored[:count]]


def official_score(seq):
    """Call the authoritative evaluator safely."""
    try:
        val = evaluate_sequence(seq)
        val = float(val)
    except Exception:
        return float("inf")
    if np.isfinite(val):
        return val
    return float("inf")


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search function. Returns a list of nonnegative floats minimizing
    2*n*max(a*a)/(sum(a))^2.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 0.0:
        return [1.0]

    start_time = time.monotonic()
    reserve = 60.0
    deadline = start_time + max(10.0, budget_s - reserve)
    N = 8192
    rng = np.random.default_rng(seed)

    # Get initial best from height_sequence_1 if available
    best_p = np.full(N, 1.0 / float(N), dtype=np.float64)
    try:
        hs = height_sequence_1
        if hs is not None and len(hs) > 0:
            arr = np.array(hs, dtype=np.float64)
            arr = np.maximum(arr, 0.0)
            s = float(np.sum(arr))
            if s > 0 and np.isfinite(s):
                arr /= s
                if len(arr) == N:
                    best_p = arr
                elif len(arr) > 0:
                    # Resample to length N
                    from numpy import interp
                    old_x = np.linspace(0, 1, len(arr))
                    new_x = np.linspace(0, 1, N)
                    best_p = interp(new_x, old_x, arr)
                    best_p = normalize_simplex(best_p)
    except Exception:
        pass

    best_score = fast_score(best_p)
    print(f"Initial: N={N} score={best_score:.8f}")
    print(f"Budget: {budget_s:.0f}s, deadline in {deadline - start_time:.0f}s")

    # ---- Phase 1: Block L-BFGS ----
    t0 = time.monotonic()
    phase1_budget = min(250.0, 0.25 * budget_s)
    phase1_deadline = start_time + phase1_budget
    print(f"Phase 1: Block L-BFGS (budget {phase1_budget:.0f}s)")
    p_param, sc_param = param_block_search(N, phase1_deadline, rng, best_p, max_B=512)
    if np.isfinite(sc_param) and sc_param < best_score:
        best_score = sc_param
        best_p = p_param.copy()
        print(f"  After Phase 1: score={best_score:.8f}")
    t1 = time.monotonic()
    print(f"Phase 1 took {t1 - t0:.1f}s")

    # ---- Phase 2: LP Peak Reduction ----
    t0 = time.monotonic()
    phase2_budget = min(700.0, 0.30 * budget_s)
    phase2_deadline = start_time + phase2_budget
    print(f"Phase 2: LP Peak Reduction (budget {phase2_budget:.0f}s)")
    if HAVE_CVPY and time.monotonic() < phase2_deadline:
        lp_result = lp_phase(best_p, phase2_deadline, max_iters=30, n_lp=4096)
        lp_score = fast_score(lp_result)
        if np.isfinite(lp_score) and lp_score < best_score:
            best_score = lp_score
            best_p = lp_result.copy()
            print(f"  After Phase 2: score={best_score:.8f}")
        else:
            print(f"  LP did not improve (lp_score={lp_score:.6f})")
    else:
        print("  Skipping (no cvxpy or no time)")
    t1 = time.monotonic()
    print(f"Phase 2 took {t1 - t0:.1f}s")

    # ---- Phase 3: Full Gradient Refinement ----
    t0 = time.monotonic()
    print(f"Phase 3: Full Gradient Refinement")
    # Use best_p as primary start, plus a few structured starts
    starts = [best_p]
    if time.monotonic() < deadline - 30:
        for sp in structured_starts(N, rng, max_count=6):
            starts.append(sp)
    top_starts = choose_best_starts(starts, 3)

    # Allocate remaining time among starts
    remaining = deadline - time.monotonic() - 30.0
    if remaining > 60:
        fracs = (0.60, 0.25, 0.15)
        for i, p0 in enumerate(top_starts):
            rem = deadline - time.monotonic() - 30.0
            if rem < 30:
                break
            frac = fracs[i] if i < len(fracs) else 0.1
            alloc = rem * frac
            sub_deadline = time.monotonic() + alloc
            print(f"  Start {i}: allocating {alloc:.0f}s")
            p_fit, sc_fit = full_refine(p0, sub_deadline, rng)
            if np.isfinite(sc_fit) and sc_fit < best_score:
                best_score = sc_fit
                best_p = p_fit.copy()
                print(f"    -> improved to {best_score:.8f}")
    t1 = time.monotonic()
    print(f"Phase 3 took {t1 - t0:.1f}s")

    # ---- Final cleanup ----
    final_p = normalize_simplex(best_p)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_s = float(np.sum(final_p))
    if final_s > 0.0 and np.isfinite(final_s):
        final_p *= 1.0 / final_s
    final_seq = [float(x) for x in final_p]

    off = official_score(final_seq)
    print(f"Official score: {off:.8f}")

    if not np.isfinite(off) or off > 2.0 + 1e-9:
        return [1.0] * N

    return final_seq
