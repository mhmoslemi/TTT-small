    """Full linear cross-convolution a*b via FFT."""
    n = max(a.size, b.size)
    size = a.size + b.size - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
    return np.fft.irfft(fa * fb, fsize)[:size]

    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = full_conv(a)
    return 2.0 * n * float(np.max(b)) / (s * s)

def conv_matrix(a):
    """Build the (2n-1) x n matrix A such that (a*b)[k] = sum_m A[k,m]*b[m]."""
    n = a.size
    A = np.zeros((2*n-1, n))
    for m in range(n):
        A[m:m+n, m] = a
    return A

def lp_solve(a, M):

    A = conv_matrix(a)
    c = -np.ones(n)

                     bounds=[(0, None)] * n, method='highs')

        return result.x
    return None

- If k = 2i: f_k(a[i]) = a[i]^2 + 2*a[i]*sum_{j: k-j=i, j≠i} a[j] + const = a[i]^2 + c1*a[i] + c0
  Actually, for k = 2i: (a*a)[2i] includes the term a[i]*a[i] = a[i]^2, plus terms a[j]*a[2i-j] for j≠i.
  
Let me be more careful:
(a*a)[k] = sum_j a[j] * a[k-j]

For a specific k, the terms involving a[i] are:
- When j = i: a[i] * a[k-i] (if 0 <= k-i < n)
- When k-j = i, i.e., j = k-i: a[k-i] * a[i] (if 0 <= k-i < n)

If i ≠ k-i (i.e., k ≠ 2i), these are two separate terms, each contributing a[i] * a[k-i].
If i = k-i (i.e., k = 2i), it's a single term a[i]^2.

So:
- For k = 2i: (a*a)[k] = a[i]^2 + [terms not involving a[i]]
- For k ≠ 2i but 0 <= k-i < n: (a*a)[k] = 2*a[i]*a[k-i] + [terms not involving a[i]]
- For k-i not in [0, n-1]: (a*a)[k] = [terms not involving a[i]] (no a[i] dependence)

So the score as a function of a[i] is:
score(a[i]) = 2n * max_k { 
    a[i]^2 + c0_k (if k=2i),
    2*a[i]*a[k-i] + c0_k (if k≠2i, k-i in range),
    c0_k (otherwise)
} / (s + a[i])^2

where c0_k are constants (terms not involving a[i]).

This is a ratio of max of (potentially) quadratics to a quadratic. It's a 1D optimization problem.

For the numerator, the max of quadratics (in a[i]) is a convex piecewise-quadratic function (since each piece is convex or linear). The denominator is a convex quadratic. The ratio of a convex function to a convex function is not necessarily convex, but it should be unimodal in practice.

I can solve this 1D problem by:
1. Grid search over a[i] in [0, some_upper_bound]
2. Refine with golden section or ternary search

The upper bound for a[i] is at most 1 (since sum = 1 and all are non-negative).

"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using an iterative LP-based
direction-finding method (solving max sum(b) s.t. a*b <= M, b >= 0 at each step),
combined with line search, multiple sequence lengths, diverse initializations,
and coordinate descent polishing.
"""

import numpy as np
from scipy.optimize import linprog
import time


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1. Returns inf if sum too small."""
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = full_conv(a)
    return 2.0 * n * float(np.max(b)) / (s * s)


def build_conv_mat(a):
    """Build (2n-1) x n matrix A such that (a*b)[k] = sum_m A[k,m]*b[m]."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for m in range(n):
        A[m:m + n, m] = a
    return A


def lp_solve(a, M):
    """Solve: max sum(b) s.t. ConvMat(a)*b <= M, b >= 0. Returns b* or None."""
    n = a.size
    A = build_conv_mat(a)
    c = -np.ones(n)
    result = linprog(c, A_ub=A, b_ub=np.full(2 * n - 1, M),
                     bounds=[(0, None)] * n, method='highs',
                     options={'presolve': True})
    if result.success:
        return result.x
    return None


def lp_optimize(a, time_budget, rng, verbose=True):
    """
    Run the iterative LP-based optimization.
    a is a non-negative sequence (will be normalized internally).
    Returns (best_a_normalized, best_normed_score).
    """
    n = a.size
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s

    aa = full_conv(a)
    M = float(np.max(aa))
    best_a = a.copy()
    best_M = M
    best_sc = 2.0 * n * M  # since sum=1

    t = 1.0
    stall = 0
    iters = 0
    consecutive_fail = 0

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        # Solve LP
        b = lp_solve(a, M)
        if b is None or b.sum() < 1e-15:
            consecutive_fail += 1
            if consecutive_fail > 20:
                break
            # Try perturbing
            a = best_a + 0.01 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            ss = a.sum()
            if ss < 1e-10:
                break
            a = a / ss
            M = float(np.max(full_conv(a)))
            continue

        consecutive_fail = 0
        S = b.sum()
        g = b / S  # normalize so sum(g) = 1

        # Try the update
        a_new = (1.0 - t) * a + t * g
        a_new = np.maximum(a_new, 0)
        s_new = a_new.sum()
        if s_new < 1e-10:
            t *= 0.5
            iters += 1

                stall = 0

                M = best_M
                t = 1.0
                # Perturb to escape

                if ss > 1e-10:

    while time.monotonic() - t0 < time_budget:
        improved = False
        for i in range(n):
            if time.monotonic() - t0 > time_budget:
                break
            old = a[i]
            s_others = 1.0 - old  # since sum=1

            # Generate candidate values for a[i]
            if old < 1e-8:
                candidates = [0.0, 0.001, 0.005, 0.01, 0.02, 0.05]
            else:
                scale = old
                candidates = np.array([
                    0.0,
                    old * 0.1, old * 0.3, old * 0.5, old * 0.7,
                    old,
                    old * 1.3, old * 1.5, old * 2.0, old * 3.0,
                    old + 0.01, old + 0.05,
                    s_others * 0.5, s_others * 0.9, s_others * 1.5
                ])
                candidates = np.unique(np.clip(candidates, 0, 1.0))

            best_x = old
            best_local = np.inf
            best_aa = aa

            for x in candidates:
                if abs(x - old) < 1e-12:
                    continue
                delta = x - old
                # Update convolution: (a_new*a_new)[k] = aa[k] + 2*delta*a_old[k-i] + delta^2*[k=2i]
                new_aa = aa.copy()
                # Add 2*delta*a[j] to positions k = i+j for j in range(n)
                for j in range(n):
                    k = i + j
                    if 0 <= k < 2 * n - 1:
                        new_aa[k] += 2.0 * delta * a[j]  # a[j] is current (old for j=i)
                # Fix: when j=i, we added 2*delta*a[i] = 2*delta*old, but we need 2*delta*old + delta^2
                # The formula gives: aa[2i] + 2*delta*a[i] + delta^2 = aa[2i] + 2*delta*old + delta^2
                # But we added 2*delta*a[i] where a[i] = old (not yet updated), so that's correct.
                # Then add delta^2:
                new_aa[2 * i] += delta * delta

                new_max = float(np.max(new_aa))
                new_s = s_others + x
                if new_s < 1e-10:
                    continue
                new_sc = 2.0 * n * new_max / (new_s * new_s)

                if new_sc < best_local:
                    best_local = new_sc
                    best_x = x
                    best_aa = new_aa

            if best_x != old:
                a[i] = best_x
                aa = best_aa
                # Renormalize
                s_new = a.sum()
                if s_new > 1e-10:
                    a = a / s_new
                    aa = full_conv(a)  # recompute after normalization
                    cur_sc = 2.0 * n * float(np.max(aa))
                    if cur_sc < best_sc:
                        best_sc = cur_sc
                        best_a = a.copy()
                improved = True

        sweep += 1
        if not improved:
            # Try random perturbation
            a = best_a + 0.005 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            aa = full_conv(a)
            if not improved and sweep > 5:
                break

    return best_a, best_sc


def init_from_u_shape(n, rng):
    """U-shaped initialization: high at edges, low in middle."""
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i) + 1.0
    a = 1.0 / np.sqrt(m)
    a += 0.05 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_two_cluster(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.1 + 0.2 * rng.random())
    c2 = n * (0.6 + 0.3 * rng.random())
    sigma = n * (0.05 + 0.1 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.02 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_anti_palindrome(n, rng):
    """Initialize with strong non-palindromic structure."""
    a = np.ones(n)
    # Make first half larger, second half smaller
    a[:n // 2] *= (1.5 + rng.random())
    a[n // 2:] *= (0.3 + 0.4 * rng.random())
    a += 0.05 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_ramp(n, rng):
    i = np.arange(n, dtype=np.float64)
    direction = rng.random()
    if direction < 0.5:
        a = i + 1.0
    else:
        a = n - i
    a += 0.1 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def get_inits(n, rng):
    """Return list of diverse initializations."""
    inits = []
    inits.append(init_from_u_shape(n, rng))
    inits.append(init_two_cluster(n, rng))
    inits.append(init_anti_palindrome(n, rng))
    inits.append(init_ramp(n, rng))
    inits.append(init_random(n, rng))
    # Noisy versions
    base = init_from_u_shape(n, rng)
    inits.append((base + 0.2 * rng.standard_normal(n)).clip(0) + 0.01)
    inits[-1] = inits[-1] / inits[-1].sum()
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses iterative LP-based direction finding as the primary method,
    with coordinate descent polishing and multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_overall = None
    best_overall_sc = np.inf

    # Try to load previous best
    prev_a = None
    prev_sc = np.inf
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size >= 10 and np.all(np.isfinite(h)) and h.sum() > 0.01:
            prev_a = h / h.sum()
            prev_sc = normed_score(prev_a)
            print(f"[init] Previous best: n={h.size}, score={prev_sc:.6f}")
            if prev_sc < best_overall_sc:
                best_overall_sc = prev_sc
                best_overall = prev_a.copy()
    except Exception:
        pass

    # Sequence lengths to try
    n_list = [128, 256, 512]
    if prev_a is not None:
        np_ = prev_a.size
        if np_ not in n_list and 64 <= np_ <= 2048:
            n_list.append(np_)
        n_list = sorted(set(n_list))

    total_time = deadline - time.monotonic()
    print(f"[plan] n_list={n_list}, total_time={total_time:.1f}s")

    # Time allocation: give more time to smaller n (faster LPs) and to n that matches prev
    raw_w = []
    for nv in n_list:
        w = 1.0 / (nv ** 0.7)
        if prev_a is not None and nv == prev_a.size:
            w *= 1.5
        raw_w.append(w)
    total_w = sum(raw_w)
    weights = [w / total_w for w in raw_w]

    for idx_n, nv in enumerate(n_list):
        if time.monotonic() > deadline:
            break

        time_n = weights[idx_n] * total_time * 0.88
        inits = get_inits(nv, rng)

        # If we have previous best of this length, use it first
        if prev_a is not None and prev_a.size == nv:
            inits.insert(0, prev_a.copy())

        n_inits = len(inits)
        time_per_init = time_n / n_inits
        print(f"[n={nv}] {n_inits} inits, {time_per_init:.1f}s each")

        # For first few inits, use LP method (slower but more powerful)
        # For remaining, use shorter time
        for ii, a_init in enumerate(inits):
            if time.monotonic() > deadline:
                break

            tb = min(time_per_init, deadline - time.monotonic() - 5.0)
            if tb < 5.0:
                break

            # Run LP optimization
            a_best, sc_best = lp_optimize(a_init, tb * 0.7, rng, verbose=(ii == 0))

            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = a_best.copy()
                print(f"[n={nv}, init={ii}] LP score={sc_best:.6f} (NEW BEST)")
            else:
                print(f"[n={nv}, init={ii}] LP score={sc_best:.6f}")

            # Polish with coordinate descent if time remains
            if time.monotonic() < deadline - 10.0 and sc_best < best_overall_sc + 0.1:
                pol_time = min(tb * 0.3, deadline - time.monotonic() - 5.0)
                if pol_time > 5.0:
                    a_pol, sc_pol = coord_descent(a_best, pol_time, rng)
                    if sc_pol < best_overall_sc:
                        best_overall_sc = sc_pol
                        best_overall = a_pol.copy()
                        print(f"[n={nv}, init={ii}] polish score={sc_pol:.6f} (NEW BEST)")

        print(f"[n={nv}] done. Overall best: {best_overall_sc:.6f}")

    # Final polish phase on the best solution
    if best_overall is not None and time.monotonic() < deadline - 10.0:
        remain = deadline - time.monotonic() - 5.0
        if remain > 20.0:
            print(f"[final polish] remaining={remain:.1f}s, n={best_overall.size}, score={best_overall_sc:.6f}")

            # LP polish
            lp_time = min(remain * 0.5, 300.0)
            a_lp, sc_lp = lp_optimize(best_overall, lp_time, rng, verbose=True)
            if sc_lp < best_overall_sc:
                best_overall_sc = sc_lp
                best_overall = a_lp.copy()
                print(f"[polish LP] score={sc_lp:.6f}")

            # Coordinate descent polish
            if time.monotonic() < deadline - 10.0:
                cd_time = min(remain * 0.3, 120.0)
                a_cd, sc_cd = coord_descent(best_overall, cd_time, rng)
                if sc_cd < best_overall_sc:
                    best_overall_sc = sc_cd
                    best_overall = a_cd.copy()
                    print(f"[polish CD] score={sc_cd:.6f}")

            # One more LP round
            if time.monotonic() < deadline - 10.0:
                lp_time2 = min(remain * 0.15, 60.0)
                a_lp2, sc_lp2 = lp_optimize(best_overall, lp_time2, rng, verbose=True)
                if sc_lp2 < best_overall_sc:
                    best_overall_sc = sc_lp2
                    best_overall = a_lp2.copy()
                    print(f"[polish LP2] score={sc_lp2:.6f}")

    # Fallback
    if best_overall is None:
        best_overall = np.ones(128) / 128
        best_overall_sc = 2.0

    # Final verification
    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"[FINAL] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")

    return result
