import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve_self(a):
    """Linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _fft_convolve_pair(a, b):
    """Linear convolution a*b using FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    need = a.size + b.size - 1
    if need <= 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)
    return c[:need]

        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * arr.size * mx / (s * s))


def _make_sine_init(n, harmonics):
    """
    Create multi-harmonic anti-palindromic initialization.
    harmonics: list of (mode, amplitude) tuples.
    a_i = (1 + sum_m A_m * sin(2*pi*m*(i+0.5)/n)) / n, clipped to >= 0, then normalized.
    """
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for mode, amp in harmonics:
        theta = 2.0 * np.pi * float(mode) * (i + 0.5) / float(n)
        a = a + float(amp) * np.sin(theta)
    a = a / float(n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 0.01:
        return np.ones(n, dtype=np.float64) / float(n)
    return a / s


def _compute_grad_fft(arr, h):
    """
    Compute subgradient using FFT.
    h: indicator/scaled array of active lags (length 2n-1).
    Returns grad (length n).
    """
    n = arr.size
    # Reverse h
    h_rev = h[::-1]
    # Linear convolution h_rev * arr
    conv = _fft_convolve_pair(h_rev, arr)
    # grad[j] = conv[2n-2+j] for j = 0..n-1
    start = 2 * n - 2
    if start + n > conv.size:
        # Pad if necessary (shouldn't happen with correct sizes)
        conv = np.pad(conv, (0, start + n - conv.size))
    grad = conv[start:start + n].copy()
    return grad


def _subgradient_refine(a, budget_s=200.0, verbose=True):
    """
    FFT-accelerated subgradient descent on the simplex.
    Minimizes max(autocorrelation) subject to sum(a)=1, a>=0.
    """
    start_time = time.monotonic()
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        arr = np.ones_like(arr) / arr.size
        s = 1.0
    arr = arr / s
    n = arr.size

    best = arr.copy()
    best_score = _score(best)
    if verbose:
        print(f"    [refine] n={n}, initial_score={best_score:.8f}")

    # Parameters
    lr = 2.0
    max_lr = 20.0
    min_lr = 0.1
    active_frac = 0.92  # include lags within 92% of max
    min_active = 10
    max_active = max(50, n // 50)
    patience = 80
    last_improve = 0
    it = 0
    eval_period = 5

    while time.monotonic() - start_time < budget_s:
        it += 1
        elapsed = time.monotonic() - start_time
        if elapsed > budget_s:
            break

        # Compute convolution
        c = _fft_convolve_self(arr)
        cmax = float(np.max(c))
        if not np.isfinite(cmax):
            break

        # Determine active set
        threshold = active_frac * cmax
        active = np.where(c >= threshold)[0]
        if len(active) < min_active:
            # Fall back to top-K
            K = min(min_active, c.size)
            active = np.argpartition(c, -K)[-K:]
        if len(active) > max_active:
            # Thin out: keep only top max_active
            K = max_active
            active = np.argpartition(c, -K)[-K:]

        # Build h array and compute gradient via FFT
        h = np.zeros(2 * n - 1, dtype=np.float64)
        h[active] = 2.0 / float(len(active))
        grad = _compute_grad_fft(arr, h)

        # Project to simplex tangent (remove mean)
        g = grad - np.mean(grad)
        gmax = float(np.max(np.abs(g)))
        if gmax < 1e-14:
            break

        # Normalize for consistent step size
        g = g / gmax

        # Try a few learning rates (backtracking)
        best_trial = None
        best_trial_score = np.inf
        for lr_try in [lr, lr * 0.5, lr * 0.25, lr * 2.0]:
            steps = -lr_try * g
            steps = np.clip(steps, -2.0, 2.0)
            trial = arr * np.exp(steps)
            trial = np.clip(trial, 0.0, None)
            ts = float(trial.sum())
            if ts < 0.01:
                continue
            trial = trial / ts
            sc = _score(trial)
            if np.isfinite(sc) and sc < best_trial_score:
                best_trial_score = sc
                best_trial = trial

        if best_trial is not None and best_trial_score < best_score:
            best = best_trial.copy()
            best_score = best_trial_score
            arr = best_trial
            last_improve = it
            lr = min(max_lr, lr * 1.08)
            if verbose and it % 50 == 1:
                remaining = budget_s - (time.monotonic() - start_time)
                print(f"    [refine] iter={it} score={best_score:.8f} lr={lr:.3f} active={len(active)} remaining={remaining:.0f}s")
        else:
            # No improvement; reduce lr
            lr = max(min_lr, lr * 0.75)
            arr = best.copy()
            if it - last_improve > patience:
                # Try perturbation to escape local minimum
                if time.monotonic() - start_time < budget_s * 0.8:
                    noise_scale = 0.05 / n
                    arr = best + noise_scale * (np.random.randn(n) - 0.5)
                    arr = np.clip(arr, 0.0, None)
                    arr = arr / arr.sum()
                    lr = max(min_lr, lr * 0.5)
                    last_improve = it - patience // 2  # give it some room
                else:
                    break

    return best, best_score


def _random_anti_palindromic_init(n, seed=None):
    """Generate a random anti-palindromic non-negative sequence."""
    if seed is not None:
        rng = np.random.RandomState(seed)
    else:
        rng = np.random.RandomState()

    # Generate first half randomly
    half = n // 2
    a_first = rng.uniform(0.1, 1.0, half)
    # Make full sequence anti-palindromic: a[i] + a[n-1-i] = constant
    # Choose constant so mean is 1/n
    c = 2.0 / float(n)
    a = np.zeros(n)
    a[:half] = a_first
    a[half:n] = c - a_first[::-1]
    # Clip and normalize
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / float(n)
    return a / s


def _random_init(n, seed=None):
    """Generate a random non-negative initialization."""
    if seed is not None:
        rng = np.random.RandomState(seed)
    else:
        rng = np.random.RandomState()
    a = rng.exponential(1.0, n)
    a = a / a.sum()
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a low evaluation score and return the best valid sequence found.
    
    Strategy:
    1. Initialize with anti-palindromic multi-harmonic sinusoids
    2. Refine using FFT-accelerated subgradient descent
    3. Try multiple lengths and initializations
    4. Return the best candidate validated by evaluate_sequence
    """
    t0 = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    hard_deadline = t0 + total_budget - 30.0  # leave time for validation

    best_out = None
    best_actual_score = np.inf

    # Define configurations to try (length, initial harmonics)
    configs = [
        (4096, [(1, 0.78)]),
        (2048, [(1, 0.78)]),
        (4096, [(1, 0.85), (2, 0.3)]),
        (2048, [(1, 0.85), (2, 0.3)]),
        (8192, [(1, 0.78)]),
        (4096, [(1, 0.9), (2, 0.2), (3, 0.15)]),
    ]

    # Allocate time per config based on remaining budget
    n_configs = len(configs)
    
    for cfg_idx, (n, harmonics) in enumerate(configs):
        remaining = hard_deadline - time.monotonic()
        if remaining < 30:
            break

        # Allocate time: more time for earlier configs, less for later
        if cfg_idx == 0:
            refine_budget = min(300.0, remaining * 0.35)
        elif cfg_idx == 1:
            refine_budget = min(200.0, remaining * 0.25)
        else:
            refine_budget = min(120.0, remaining * 0.15)

        refine_budget = max(10.0, refine_budget)
        if remaining - refine_budget < 20:
            refine_budget = remaining - 30
        if refine_budget < 5:
            break

        print(f"[config {cfg_idx}] n={n}, harmonics={harmonics}, refine_budget={refine_budget:.0f}s")

        # Create initialization
        a_init = _make_sine_init(n, harmonics)
        init_score = _score(a_init)
        print(f"    init_score={init_score:.8f}")

        # Refine
        a_ref, ref_score = _subgradient_refine(a_init, refine_budget, verbose=True)
        print(f"    refined_score={ref_score:.8f} (improvement: {init_score - ref_score:.6f})")

        # Validate with actual evaluate_sequence
        if np.isfinite(ref_score):
            try:
                real_score = float(evaluate_sequence(a_ref.tolist()))
            except Exception as e:
                print(f"    validation error: {e}")
                real_score = np.inf

            if np.isfinite(real_score) and real_score < best_actual_score:
                best_actual_score = real_score
                best_out = a_ref.copy()
                print(f"    NEW BEST: actual_score={real_score:.8f}")

        # Also validate the initial (in case refinement made it worse)
        if init_score < ref_score:
            try:
                real_score = float(evaluate_sequence(a_init.tolist()))
            except Exception:
                real_score = np.inf
            if np.isfinite(real_score) and real_score < best_actual_score:
                best_actual_score = real_score
                best_out = a_init.copy()
                print(f"    NEW BEST (init): actual_score={real_score:.8f}")

        elapsed = time.monotonic() - t0
        print(f"    [config {cfg_idx}] done in {elapsed:.1f}s, best so far: {best_actual_score:.8f}")

    # If we have remaining time, try a random initialization with refinement
    remaining = hard_deadline - time.monotonic()
    if remaining > 60:
        print(f"[random init] trying with {remaining:.0f}s remaining")
        n_rand = 4096
        rng = np.random.RandomState(seed)
        a_rand = _random_anti_palindromic_init(n_rand, seed=seed + 100)
        rand_score = _score(a_rand)
        print(f"    random init score={rand_score:.8f}")

        if rand_score < 1.8:  # only refine if it's promising
            refine_budget = min(remaining - 30, 120.0)
            a_ref, ref_score = _subgradient_refine(a_rand, refine_budget, verbose=True)
            try:
                real_score = float(evaluate_sequence(a_ref.tolist()))
            except Exception:
                real_score = np.inf
            if np.isfinite(real_score) and real_score < best_actual_score:
                best_actual_score = real_score
                best_out = a_ref.copy()
                print(f"    NEW BEST (random): actual_score={real_score:.8f}")

    # Final fallback
    if best_out is None:
        print("[fallback] using default sinusoid")
        best_out = _make_sine_init(2048, [(1, 0.78)])
        try:
            best_actual_score = float(evaluate_sequence(best_out.tolist()))
        except Exception:
            best_out = np.ones(256, dtype=np.float64) / 256.0
            best_actual_score = 2.0

    print(f"\n[FINAL] best_actual_score={best_actual_score:.8f}, n={best_out.size}")
    print(f"[FINAL] total time={time.monotonic() - t0:.1f}s")

    return [float(x) for x in best_out]
