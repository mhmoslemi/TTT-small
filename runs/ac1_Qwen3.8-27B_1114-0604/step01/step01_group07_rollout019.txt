Low-peak nonnegative step-function optimization via gradient-based active-set method.

Uses FFT-based convolution evaluation, soft/hard active-set gradient computation,
multiplicative (mirror descent) updates with adaptive step size, and multi-start
random restarts. Explores single and multi-sinusoidal initializations.

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _score(a):
    arr = np.asarray(a, dtype=np.float64)
    s = float(arr.sum())
    if s < 1e-10:
        return np.inf
    c = _fft_convolve(arr)
    mx = float(np.max(c))
    return 2.0 * arr.size * mx / (s * s)


def _init_sine(n, modes):
    """modes: list of (frequency, amplitude) pairs."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for (freq, amp) in modes:
        a += amp * np.sin(2.0 * np.pi * freq * (i + 0.5) / n)
    a = np.clip(a, 0.0, 1000.0)
    a /= a.sum()
    return a


def _compute_gradient(a, weights, active_indices):
    """Compute gradient g_j = 2 * sum_k weights[k] * a[k-j]."""
    n = a.size
    g = np.zeros(n, dtype=np.float64)
    for k_idx in active_indices:

        w_val = float(weights[k_int])

            continue

            g[i_min:i_max + 1] += 2.0 * w_val * seg[::-1]
    return g
