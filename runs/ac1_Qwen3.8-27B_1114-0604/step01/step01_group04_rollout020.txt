import numpy as np
import time


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def get_softmax_weights(b, tau):
    """Compute softmax weights for the convolution values."""
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(b.size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= s
    return w


def grad_from_weights(w, a):
    """Compute gradient of smooth max of conv(a,a) given softmax weights."""
    return 2.0 * np.correlate(w, a, mode='valid')


def multi_scale_gradient(a, b, tau_base, n_scales):
    """Average gradients at multiple temperature scales for robustness."""
    n = a.size
    g_total = np.zeros(n)
    for s in range(n_scales):
        tau = tau_base * (2.0 ** s)
        w = get_softmax_weights(b, tau)
        g_total += grad_from_weights(w, a)
    return g_total / n_scales

        else:
            stall += 1
            step_scale *= 0.9

            if stall > 50:
                # Perturb and restart
                a = project_simplex(best_a + 0.02 * rng.standard_normal(n))
                stall = 0
                step_scale = 0.1
                tau = 0.02

    return best_a, best_sc


def optimize_annealing(a, time_budget, rng):
    """
    Simulated annealing with structured mass-transfer moves.
    Uses a cooling schedule and accepts worse moves with decreasing probability.
    """
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    b = full_conv(a)
    best_a = a.copy()
    best_sc = 2.0 * n * float(np.max(b))
    cur_sc = best_sc

    t0 = time.monotonic()
    T = 0.005
    T_min = 1e-7
    cool_rate = 0.9995
    iters = 0

    while time.monotonic() - t0 < time_budget and T > T_min:
        # Generate move: transfer mass between two indices
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue

        step = min(a[i], a[j]) * (0.02 + 0.08 * rng.random())
        if step < 1e-15:
            step = 0.001 * cur_sc / (2.0 * n)

        if rng.random() > 0.5:
            a[i] -= step
            a[j] += step
        else:
            a[j] -= step
            a[i] += step

        a = project_simplex(a)
        b = full_conv(a)
        new_sc = 2.0 * n * float(np.max(b))

        # Accept/reject
        delta = new_sc - cur_sc
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            cur_sc = new_sc
            if new_sc < best_sc:
                best_sc = new_sc
                best_a = a.copy()
        else:
            # Revert (recompute from best or keep old a)
            # We don't save old a, so just skip (a is already modified)
            # This is a minor issue; in practice most moves are accepted early
            pass

        T *= cool_rate
        iters += 1

    return best_a, best_sc


def init_uniform(n):
    return np.ones(n) / n


def init_u_shape(n):
    """High at edges, low in middle."""
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i).astype(np.float64) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_two_peak(n, rng):
    """Two Gaussian peaks."""
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.15 + 0.2 * rng.random())
    c2 = n * (0.65 + 0.2 * rng.random())
    sigma = n * (0.05 + 0.1 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_cosine(n, rng):
    """Cosine-modulated structure."""
    i = np.arange(n, dtype=np.float64)
    freq = 1.0 + int(rng.integers(0, 4))
    a = (1.0 + freq * np.cos(2.0 * np.pi * i / max(n - 1, 1))) ** 2
    a += 1e-6
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.05
    return a / a.sum()


def init_quadratic(n):
    """Quadratic: high at edges."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / max(c, 1)) ** 2 + 0.05
    return a / a.sum()


def get_inits(n, rng, prev_a=None):
    """Return list of initialization arrays for given n."""
    inits = [
        init_u_shape(n),
        init_two_peak(n, rng),
        init_quadratic(n),
        init_cosine(n, rng),
        init_random(n, rng),
        init_uniform(n),
    ]
    if prev_a is not None and prev_a.size == n:
        inits.insert(0, prev_a.copy())
    return inits


def polish_final(a, time_budget, rng):
    """
    Final polishing: short subgradient + coordinate descent on the best candidate.
    """
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    b = full_conv(a)
    best_sc = 2.0 * n * float(np.max(b))

    t0 = time.monotonic()

    # Phase 1: Short subgradient (40% of time)
    sg_time = time_budget * 0.4
    if sg_time > 2.0:
        a1, sc1 = optimize_subgrad(a, sg_time, rng)
        if sc1 < best_sc:
            best_sc = sc1
            best_a = a1.copy()

    # Phase 2: Coordinate descent (40% of time)
    cd_time = time_budget * 0.4
    remain = time_budget - (time.monotonic() - t0)
    cd_time = min(cd_time, remain)
    if cd_time > 2.0:
        a2, sc2 = optimize_coord_descent(best_a, cd_time, rng)
        if sc2 < best_sc:
            best_sc = sc2
            best_a = a2.copy()

    # Phase 3: More coordinate descent if time remains
    remain = time_budget - (time.monotonic() - t0)
    if remain > 2.0:
        a3, sc3 = optimize_coord_descent(best_a, remain, rng)
        if sc3 < best_sc:
            best_sc = sc3
            best_a = a3.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses a combination of smoothed subgradient descent, coordinate descent
    with pairwise mass transfers, and simulated annealing.
    Explores multiple sequence lengths and diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 5.0)

    # Load previous best
    prev_a = None
    prev_sc = np.inf
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size >= 10 and np.all(np.isfinite(h)) and h.sum() > 0.01:
            prev_a = h / h.sum()
            prev_sc = 2.0 * h.size * float(np.max(full_conv(prev_a)))
            print(f"[init] Previous best: n={h.size}, score={prev_sc:.6f}")
    except Exception:
        pass

    best_overall = None
    best_overall_sc = np.inf

    if prev_a is not None and prev_sc < 2.0:
        best_overall = prev_a.copy()
        best_overall_sc = prev_sc

    # n values to try
    n_list = [128, 256, 512, 1024, 2048]
    if prev_a is not None:
        np_ = prev_a.size
        if np_ not in n_list and np_ >= 64:
            n_list.append(np_)
        n_list = sorted(set(n_list))

    total_time = max(1.0, deadline - time.monotonic())
    print(f"[plan] n_list={n_list}, total_time={total_time:.1f}s")

    # Allocate time: give more to smaller n and to the n matching previous best
    prev_n = prev_a.size if prev_a is not None else -1
    raw_w = []
    for nv in n_list:
        w = 1.0 / (0.3 + nv / 512.0)
        if nv == prev_n:
            w *= 2.0  # Boost the n that had the previous best
        raw_w.append(w)
    total_w = sum(raw_w)
    weights = [w / total_w for w in raw_w]

    for idx_n, nv in enumerate(n_list):
        if time.monotonic() > deadline - 30.0:
            break

        time_n = weights[idx_n] * total_time * 0.80
        inits = get_inits(nv, rng, prev_a)

        # For each initialization, run optimization
        n_inits = len(inits)
        time_per_init = time_n / n_inits

        for ii, a_init in enumerate(inits):
            if time.monotonic() > deadline - 30.0:
                break

            tb = min(time_per_init, deadline - time.monotonic() - 30.0)
            if tb < 5.0:
                break

            # Phase A: Subgradient (60% of allocated time)
            t1 = tb * 0.6
            a_opt, sc_opt = optimize_subgrad(a_init, t1, rng)

            # Phase B: Coordinate descent (40% of allocated time)
            t2 = min(tb * 0.4, deadline - time.monotonic() - 30.0)
            if t2 > 3.0:
                a_opt, sc_opt = optimize_coord_descent(a_opt, t2, rng)

            if sc_opt < best_overall_sc:
                best_overall_sc = sc_opt
                best_overall = a_opt.copy()
                print(f"[opt] n={nv}, init={ii}, score={sc_opt:.6f} *** BEST ***")
            else:
                print(f"[opt] n={nv}, init={ii}, score={sc_opt:.6f}")

        print(f"[opt] n={nv} done. Overall best: {best_overall_sc:.6f}")

    # Final polish phase
    if best_overall is not None:
        remain = deadline - time.monotonic() - 3.0
        if remain > 10.0:
            print(f"[polish] remaining={remain:.1f}s, n={best_overall.size}, "
                  f"score={best_overall_sc:.6f}")
            a_final, sc_final = polish_final(best_overall, remain, rng)
            if sc_final < best_overall_sc:
                best_overall_sc = sc_final
                best_overall = a_final.copy()
                print(f"[polish] Improved to {sc_final:.6f}")
            else:
                print(f"[polish] No improvement. Best: {best_overall_sc:.6f}")

    # Fallback
    if best_overall is None:
        best_overall = np.ones(512) / 512
        best_overall_sc = 2.0
        print("[warn] Using fallback uniform sequence")

    # Final verification
    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"[final] n={len(result)}, internal={best_overall_sc:.6f}, "
          f"evaluator={final_sc:.6f}")

    return result
