    Ar = np.fft.rfft(a_rev, M_fft)
    conv_res = np.fft.irfft(W * Ar, M_fft)[:conv_len]
    grad_M = 2.0 * conv_res[n - 1:2 * n - 1]

    grad = (2.0 * n / S2) * grad_M - (4.0 * n * M_sm / (S * S2))
    return grad, M_sm, S


def adam_update(a, g, m, v, t, lr):
    b1, b2, eps = 0.9, 0.999, 1e-8
    m[:] = b1 * m + (1 - b1) * g
    v[:] = b2 * v + (1 - b2) * g * g
    m_hat = m / (1.0 - b1 ** t)
    v_hat = v / (1.0 - b2 ** t)
    a -= lr * m_hat / (np.sqrt(v_hat) + eps)

"""
Minimize f(a) = 2n * max(conv(a,a)) / (sum a)^2 using Adam with FFT-based
smooth-max gradient (O(n log n) per step), adaptive temperature, diverse
initializations, and periodic restarts. Key improvements:
  - Full softmax-weighted gradient via FFT cross-correlation (no loop over peaks)
  - Larger n (4096-8192) for more degrees of freedom
  - Adaptive temperature cooling schedule
  - Multiple diverse initializations + parent-based init
  - Periodic L1 normalization to prevent scale drift
"""

import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = a.size
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_objective(a):
    n = a.size
    c = fast_self_conv(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf, c
    max_c = np.max(c)
    val = float(2.0 * n * max_c / (S * S))
    return val, c


def soft_max_gradient(a, c, T):
    """Gradient of f = 2n * M_smooth / S^2 using log-sum-exp smooth max.
    Uses FFT cross-correlation for O(n log n) gradient."""
    n = a.size
    S = a.sum()
    S2 = S * S

    # Softmax weights over convolution peaks
    Tc = T * (c - c.max())
    exp_vals = np.exp(Tc)
    w = exp_vals / exp_vals.sum()
    M_sm = float(c.max() + np.log(exp_vals.sum()) / T)

    # grad_M[j] = 2 * sum_{l=0}^{n-1} w[j+l] * a[l]
    # = 2 * conv(w, a[::-1])[j + n - 1]
    a_rev = a[::-1]
    conv_len = (2 * n - 1) + n - 1
    M_fft = 1 << (conv_len - 1).bit_length()
    W = np.fft.rfft(w, M_fft)
    Ar = np.fft.rfft(a_rev, M_fft)
    conv_res = np.fft.irfft(W * Ar, M_fft)[:conv_len]
    grad_M = 2.0 * conv_res[n - 1:2 * n - 1]

    # f = 2n * M_sm / S^2
    # df/da = 2n/S^2 * grad_M - 2n * 2 * M_sm / S^3
    grad = (2.0 * n / S2) * grad_M - (4.0 * n * M_sm) / (S * S2)
    return grad, M_sm, S

    m[:] = b1 * m + (1.0 - b1) * g

def local_peak_reduction(a, max_iters=500):
    """Greedy local search: reduce contribution to top peak."""
    n = a.size
    a = a.copy()
    c = fast_self_conv(a)
    S = a.sum()
    score = float(2.0 * n * c.max() / (S * S))
    best_score = score
    best_a = a.copy()

    for _ in range(max_iters):
        k_peak = int(np.argmax(c))
        # Elements contributing to c[k_peak]: a[j]*a[k_peak-j]
        j_start = max(0, k_peak - n + 1)
        j_end = min(k_peak, n - 1)
        if j_start > j_end:
            break

        # Find the element with largest contribution
        contributions = a[j_start:j_end + 1] * a[j_start:j_end + 1][::-1]
        j_local = int(np.argmax(contributions))
        j_peak = j_start + j_local

        # Reduce a[j_peak] slightly, redistribute to others
        delta = 0.01 * a[j_peak]
        if delta < 1e-10:
            break
        a[j_peak] -= delta
        # Add to a random element far from the peak
        j_other = int(np.random.randint(0, n))
        if abs(j_other - j_peak) < n // 4:
            j_other = (j_other + n // 2) % n
        a[j_other] += delta
        a = clamp_a(a)

        c = fast_self_conv(a)
        S = a.sum()
        new_score = float(2.0 * n * c.max() / (S * S))
        if new_score < best_score:
            best_score = new_score
            best_a = a.copy()
        else:
            a = best_a.copy()
            if delta < 1e-12:
                break

    return best_score, best_a

            if sc < best_score:

                best_a = aa.copy()
            print(f"Phase 4 (local): best = {best_score:.6f}")

    if best_a is None:
        best_a = np.ones(n)

    out = [float(x) for x in best_a]
    verified = float(evaluate_sequence(out))
    print(f"FINAL: internal={best_score:.6f}, verified={verified:.6f}, n={n}, "
          f"time={time.monotonic()-start:.1f}s")
    # Return the verified one if it's actually better
    if verified < best_score:
        print("Verified is better!")
    return out
