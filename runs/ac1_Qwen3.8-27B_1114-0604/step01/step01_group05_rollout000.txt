"""
Minimize 2*n*max(a*a)/(sum(a))^2 for nonnegative sequences.

Algorithm:
1. Start from previous best (height_sequence_1) if available, else uniform.
2. Block-constant L-BFGS for rapid exploration of the landscape.
3. Adaptive smooth-max momentum gradient descent on the full simplex with:
   - Increasing smooth-max parameter (coarse-to-fine)
   - Nesterov momentum
   - Adaptive step size with restarts
4. Local perturbation search for final polish.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def fft_convolution(a):
    """Linear convolution a*a via FFT with roundoff cleanup."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    """Smooth max and softmax weights for gradient computation."""
    b = np.asarray(b, dtype=np.float64)
    m = float(np.max(b))
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b, dtype=np.float64)
        w /= float(b.size)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w


def fast_score(p):
    """Compute 2*n*max(p*p) with sum(p)=1."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float("inf")
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float("inf")
    b = fft_convolution(p)
    M = float(np.max(b))
    return float(2.0 * n * M / (S * S))


def corr_pw(p, w):
    """s_i = sum_j p_j * w_{i+j} for gradient of smooth max of p*p."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    if n == 1:
        return np.array([float(np.dot(p, w[:1]))], dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n : m][::-1].copy()
    return s


def project_simplex(v, target=1.0):
    """Project vector onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    if target <= 0.0:
        return np.zeros_like(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(v.shape, target / float(v.size), dtype=np.float64)
    low = float(np.min(v)) - 1.0
    high = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (low + high)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        q = np.full(v.shape, target / float(v.size), dtype=np.float64)
    else:
        q *= target / s
    return q


def normalize_simplex(p):
    """Clip to nonnegative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size), dtype=np.float64)
    p *= 1.0 / s
    return p


def block_to_p(n, B, c):
    """Convert block log-weights to full sequence."""
    L = n // B
    z = np.empty(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        tmp = np.zeros(B - 1, dtype=np.float64)
        m = min(tmp.size, c.size)
        tmp[:m] = c[:m]
        c = tmp
        z[:-1] = c
    z[-1] = 0.0
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0.0 or not np.isfinite(s):
        wb = np.ones(B, dtype=np.float64) / float(B)
    else:
        wb /= s
    H = wb / float(L)
    return np.repeat(H, L)


def block_objective(c, n, B, beta):
    """Objective and gradient for block-constant sequences."""
    L = n // B
    z = np.empty(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        tmp = np.zeros(B - 1, dtype=np.float64)
        m = min(tmp.size, c.size)
        tmp[:m] = c[:m]
        c = tmp
        z[:-1] = c
    z[-1] = 0.0
    if not np.all(np.isfinite(z)):
        return float("inf"), np.zeros(B - 1, dtype=np.float64)
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0.0 or not np.isfinite(s):
        wb = np.ones(B, dtype=np.float64) / float(B)
    else:
        wb /= s
    H = wb / float(L)
    p = np.repeat(H, L)
    b = fft_convolution(p)
    G, w = smooth_max_and_weights(b, beta)
    if not np.isfinite(G):
        return float("inf"), np.zeros(B - 1, dtype=np.float64)
    svec = corr_pw(p, w)
    svec = np.maximum(svec, 0.0)
    g = 2.0 * svec
    Gsum = g.reshape(B, L).sum(axis=1)
    Wavg = float(np.dot(wb, Gsum))
    gradz = (wb / float(L)) * (Gsum - Wavg)
    grad = n * gradz[:-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(n * G), grad.astype(np.float64)


def run_block_minimize(c0, n, B, beta, bounds, maxiter=120):
    """Run L-BFGS-B on block representation."""
    if not HAVE_SCIPY:
        return np.asarray(c0, dtype=np.float64)
    c0 = np.asarray(c0, dtype=np.float64).reshape(-1)
    m = max(0, B - 1)
    if c0.size != m:
        tmp = np.zeros(m, dtype=np.float64)
        k = min(tmp.size, c0.size)
        tmp[:k] = c0[:k]
        c0 = tmp
    c0 = np.clip(c0, -8.0, 8.0)
    options = {"maxiter": maxiter, "ftol": 1e-12, "gtol": 1e-6, "maxls": 50, "maxcor": 15}
    try:
        res = minimize(block_objective, c0, args=(n, B, beta), method="L-BFGS-B",
                       jac=True, bounds=bounds, options=options)
        c = np.asarray(res.x, dtype=np.float64)
    except Exception:
        c = c0
    if not np.all(np.isfinite(c)):
        c = c0
    return np.clip(c, -8.0, 8.0)


def param_block_search(n, deadline, rng, max_B=256):
    """Block optimization with multiple resolutions."""
    best_p = np.full(n, 1.0 / float(n), dtype=np.float64)
    best_score = fast_score(best_p)
    if not HAVE_SCIPY or n < 8:
        return best_p, best_score
    candidates = []
    for B in (4, 8, 16, 32, 64, 128, 256):
        if B <= max_B and B < n and (n % B) == 0:
            candidates.append(B)
    for B in candidates:
        if time.monotonic() >= deadline:
            break
        bounds = [(-8.0, 8.0)] * (B - 1)
        m = B - 1
        best_c = np.zeros(m, dtype=np.float64)
        best_B_score = float("inf")
        x = np.linspace(0.0, 1.0, m, dtype=np.float64)
        inits = [
            np.zeros(m, dtype=np.float64),
            2.0 * (x - 0.5),
            -2.0 * (x - 0.5),
            3.0 * (x - 0.5),
            -3.0 * (x - 0.5),
            np.cos(2.0 * np.pi * x),
            np.sin(2.0 * np.pi * x),
            np.cos(4.0 * np.pi * x),
            0.8 * rng.normal(0.0, 1.0, m),
            1.5 * rng.normal(0.0, 1.0, m),
        ]
        for beta in (5000.0, 50000.0):
            for c0 in inits:
                if time.monotonic() >= deadline:
                    break
                c = run_block_minimize(c0, n, B, beta, bounds)
                p = block_to_p(n, B, c)
                sc = fast_score(p)
                if np.isfinite(sc) and sc < best_B_score:
                    best_B_score = sc
                    best_c = c.copy()
        if np.isfinite(best_B_score) and best_B_score < best_score:
            best_score = best_B_score
            best_p = block_to_p(n, B, best_c).copy()
            print(f"block B={B} score={best_score:.8f}")
    return best_p, best_score


def momentum_refine(p0, deadline, rng, max_total_it=5000000):
    """
    Adaptive smooth-max momentum gradient descent on the simplex.
    Uses Nesterov momentum and adaptive step size with restarts.
    """
    p = normalize_simplex(p0)
    n = p.size
    if n == 0:
        return p, float("inf")

    # Break symmetry
    mean_p = float(np.mean(p))
    if n > 1 and float(np.max(np.abs(p - mean_p))) < 1e-13:
        t = 0.123 * np.arange(n, dtype=np.float64)
        p = normalize_simplex(p * (1.0 + 1e-4 * np.sin(t)))

    best_p = p.copy()
    best_score = fast_score(p)
    v = np.zeros(n, dtype=np.float64)  # momentum
    momentum = 0.85
    step = 2.0
    beta = 2000.0  # start smooth
    beta_target = 200000.0
    no_improve = 0
    restart_interval = 3000
    it = 0
    last_print = 0

    while it < max_total_it and time.monotonic() < deadline:
        it += 1

        # Gradually increase beta (coarse to fine)
        progress = min(1.0, float(it) / float(max_total_it))
        beta = 2000.0 + (beta_target - 2000.0) * (progress ** 0.5)

        # Nesterov lookahead
        p_look = project_simplex(p - momentum * step * v, 1.0)

        b = fft_convolution(p_look)
        G, w = smooth_max_and_weights(b, beta)
        if not np.isfinite(G):
            break

        svec = corr_pw(p_look, w)
        svec = np.maximum(svec, 0.0)
        g = 2.0 * svec
        grad = g - float(np.mean(g))

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-15:
            break

        # Update momentum
        v = momentum * v + (1.0 - momentum) * grad
        v *= 1.0 / (1.0 + (1.0 - momentum) * 0)  # normalize-ish

        # Tentative step
        p_new = project_simplex(p - step * v, 1.0)

        # Evaluate
        b_new = fft_convolution(p_new)
        m_new = float(np.max(b_new))
        G_new, _ = smooth_max_and_weights(b_new, beta)
        score_new = float(2.0 * n * m_new)

        if score_new < best_score - 1e-15:
            best_score = score_new
            best_p = p_new.copy()
            no_improve = 0
        else:
            no_improve += 1

        p = p_new

        # Adaptive step size
        if no_improve > 1000:
            step *= 0.5
            no_improve = 0
        elif no_improve == 0 and step < 5.0:
            step = min(5.0, step * 1.1)

        # Restart with noise if stuck too long
        if no_improve > restart_interval:
            noise = 0.05 * rng.standard_normal(n)
            p = normalize_simplex(p + noise)
            v = np.zeros(n, dtype=np.float64)
            beta = max(2000.0, beta * 0.1)  # go back to smoother
            step = 2.0
            no_improve = 0
            sc_after_restart = fast_score(p)
            if sc_after_restart < best_score:
                best_score = sc_after_restart
                best_p = p.copy()
            print(f"  restart at it={it}, beta={beta:.0f}, score={best_score:.8f}")

        if it - last_print > 5000:
            print(f"  momentum it={it} beta={beta:.0f} step={step:.3f} best={best_score:.8f} no_imp={no_improve}")
            last_print = it

    return best_p, best_score


def local_search(p0, deadline, rng, n_trials=50000):
    """Local perturbation search: try random small perturbations."""
    p = normalize_simplex(p0)
    n = p.size
    best_p = p.copy()
    best_score = fast_score(p)
    it = 0
    pert_scale = 0.02

    while it < n_trials and time.monotonic() < deadline:
        # Random sparse perturbation
        k = max(1, n // 100)
        idx = rng.choice(n, size=k, replace=False)
        pert = pert_scale * rng.standard_normal(k)
        q = p.copy()
        q[idx] += pert
        q = normalize_simplex(q)
        sc = fast_score(q)
        if sc < best_score - 1e-12:
            best_score = sc
            best_p = q.copy()
            p = q.copy()
            pert_scale = min(0.05, pert_scale * 1.1)
        else:
            pert_scale = max(0.001, pert_scale * 0.999)
        it += 1
        if it % 10000 == 0:
            print(f"  local it={it} best={best_score:.8f} pert={pert_scale:.4f}")

    return best_p, best_score


def official_score(seq):
    """Call the authoritative evaluator."""
    try:
        val = evaluate_sequence(seq)
        val = float(val)
    except Exception:
        return float("inf")
    if np.isfinite(val):
        return val
    return float("inf")


def get_seed_sequence(n):
    """Try to get height_sequence_1 and resize to n."""
    try:
        hs = height_sequence_1
        if isinstance(hs, list) and len(hs) > 0:
            arr = np.array(hs, dtype=np.float64)
            arr = np.maximum(arr, 0.0)
            s = float(np.sum(arr))
            if s > 0.01:
                arr /= s
                # Resize to n using interpolation
                if len(arr) != n:
                    old_n = len(arr)
                    new_indices = np.linspace(0.0, old_n - 1, n)
                    lo = np.floor(new_indices).astype(int)
                    hi = np.minimum(lo + 1, old_n - 1)
                    frac = new_indices - lo
                    resized = arr[lo] * (1.0 - frac) + arr[hi] * frac
                    resized = np.maximum(resized, 0.0)
                    rs = float(np.sum(resized))
                    if rs > 0:
                        resized /= rs
                        return resized
                return arr
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search function that returns a list of nonnegative floats minimizing
    2*n*max(convolve(a,a))/(sum(a))^2.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 0.0:
        return [1.0]

    reserve = 60.0
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - reserve)

    rng = np.random.default_rng(seed)

    # Choose length
    N = 8192

    print(f"Starting search: N={N}, budget={budget_s:.0f}s")

    # Get initial point
    seed_p = get_seed_sequence(N)
    if seed_p is not None:
        best_p = normalize_simplex(seed_p)
        print(f"Using seed sequence, initial score={fast_score(best_p):.8f}")
    else:
        best_p = np.full(N, 1.0 / float(N), dtype=np.float64)
        print(f"No seed available, using uniform, score={fast_score(best_p):.8f}")

    best_score = fast_score(best_p)

    # Phase 1: Block optimization for exploration
    if time.monotonic() < deadline - 120.0:
        if budget_s >= 2000.0:
            max_B = 256
            phase1_time = 300.0
        elif budget_s >= 1000.0:
            max_B = 128
            phase1_time = 150.0
        else:
            max_B = 64
            phase1_time = 60.0

        p1_deadline = min(deadline, time.monotonic() + phase1_time)
        print(f"Phase 1: Block optimization (max_B={max_B}, time={phase1_time:.0f}s)")
        p_param, sc_param = param_block_search(N, p1_deadline, rng, max_B=max_B)
        if np.isfinite(sc_param) and sc_param < best_score:
            best_score = sc_param
            best_p = p_param.copy()
            print(f"After block: score={best_score:.8f}")

    # Phase 2: Momentum gradient refinement
    if time.monotonic() < deadline - 60.0:
        print(f"Phase 2: Momentum refinement (remaining={deadline - time.monotonic():.0f}s)")
        p_mom, sc_mom = momentum_refine(best_p, deadline - 60.0, rng)
        if np.isfinite(sc_mom) and sc_mom < best_score:
            best_score = sc_mom
            best_p = p_mom.copy()
            print(f"After momentum: score={best_score:.8f}")

    # Phase 3: Local search polish
    if time.monotonic() < deadline - 30.0:
        print(f"Phase 3: Local search (remaining={deadline - time.monotonic():.0f}s)")
        p_ls, sc_ls = local_search(best_p, deadline - 30.0, rng)
        if np.isfinite(sc_ls) and sc_ls < best_score:
            best_score = sc_ls
            best_p = p_ls.copy()
            print(f"After local: score={best_score:.8f}")

    # Final: one more short momentum pass from the best
    if time.monotonic() < deadline - 30.0:
        remaining = deadline - time.monotonic() - 30.0
        if remaining > 30.0:
            print(f"Phase 4: Final momentum pass ({remaining:.0f}s)")
            p_final, sc_final = momentum_refine(best_p, time.monotonic() + remaining, rng,
                                                max_total_it=2000000)
            if np.isfinite(sc_final) and sc_final < best_score:
                best_score = sc_final
                best_p = p_final.copy()
                print(f"After final momentum: score={best_score:.8f}")

    # Prepare final output
    final_p = normalize_simplex(best_p)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_s = float(np.sum(final_p))
    if final_s > 0.0 and np.isfinite(final_s):
        final_p *= 1.0 / final_s

    final_seq = [float(x) for x in final_p]
    print(f"Final local score: {best_score:.8f}")

    # Verify with official evaluator
    off = official_score(final_seq)
    print(f"Official score: {off:.8f}")

    if not np.isfinite(off) or off > 2.0 + 1e-9:
        print("Fallback to uniform")
        return [1.0] * N

    return final_seq
