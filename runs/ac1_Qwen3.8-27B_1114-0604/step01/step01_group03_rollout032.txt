"""
Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses momentum-enhanced multiplicative gradient descent with adaptive
temperature, exact subgradient refinement, and vectorized coordinate descent polish.
"""

import math
import time
import numpy as np

def _get_parent():
    global height_sequence_1
    try:
        return np.asarray(height_sequence_1, dtype=np.float64).ravel()
    except (NameError, Exception):
        return None

def _normalize(a):
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300 or not np.isfinite(s):
        a[:] = 1.0
        s = float(a.sum())
    a /= s
    return a

def _conv_self(a, L, m):
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c

def _score(c, n):
    return 2.0 * n * float(c.max())

def _grad_from_weights(w, a, L, m, n):
    W = np.fft.rfft(w, L)
    Ar = np.fft.rfft(a[::-1], L)
    g = np.fft.irfft(W * Ar, L)[(n - 1):m]
    np.maximum(g, 0.0, out=g)
    return g

def _soft_weights(c, m, temp):
    vmax = float(c.max())
    z = temp * (c - vmax)
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-300 or not np.isfinite(s):
        return np.ones(m) / m
    return e / s

def _hard_weights(c, m):
    cmax = float(c.max())
    tol = 1e-10 * max(cmax, 1e-15)
    w = np.zeros(m)
    w[c >= cmax - tol] = 1.0
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.ones(m) / m
    return w

def _random_init(n, rng):
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0
    z = rng.normal(0, 0.3, n)
    z += 0.5 * rng.normal(0, 1) * x
    z += 0.3 * rng.normal(0, 1) * (x * x - 1.0 / 3.0)
    for k in range(1, 6):
        amp = rng.uniform(0.05, 0.3) / math.sqrt(k)
        ph = rng.uniform(0, 2 * np.pi)
        z += amp * np.cos(2 * np.pi * k * t + ph)
    z += rng.normal(0, 0.1, n)
    p = np.exp(z - z.max())
    return _normalize(np.maximum(p, 1e-300))

def _optimize_phase(a, L, m, n, time_limit, rng, temp_start, temp_end,
                    mom_start, mom_end, eta_init=0.12, label=""):
    deadline = time.monotonic() + time_limit
    t_start = time.monotonic()
    a = _normalize(a.copy())
    c = _conv_self(a, L, m)
    cur = _score(c, n)
    best_a = a.copy()
    best = cur
    v = np.zeros(n)
    eta = eta_init
    no_imp = 0
    it = 0

    while time.monotonic() < deadline - 1.0:
        it += 1
        elapsed = time.monotonic() - t_start
        frac = min(1.0, elapsed / max(1.0, time_limit))
        temp = temp_start * (temp_end / temp_start) ** frac
        mom = mom_start + (mom_end - mom_start) * frac

        if temp > 0.05:
            w = _soft_weights(c, m, temp)
        else:
            w = _hard_weights(c, m)

        g = _grad_from_weights(w, a, L, m, n)
        gm = float(g.max())
        if gm < 1e-300:
            break
        g /= gm

        v = mom * v + (1.0 - mom) * g

        accepted = False
        for step in (eta, 0.5 * eta, 0.25 * eta, 0.125 * eta):
            if step < 1e-10:
                continue
            a2 = a * np.exp(-step * v)
            a2 = _normalize(a2)
            c2 = _conv_self(a2, L, m)
            s2 = _score(c2, n)
            if s2 < cur - 1e-14:
                a, c, cur = a2, c2, s2
                accepted = True
                break

        if accepted:
            if cur < best - 1e-14:
                best = cur
                best_a = a.copy()
            no_imp = 0
            eta = min(0.5, eta * 1.02)
        else:
            no_imp += 1
            eta = max(0.003, eta * 0.88)
            if no_imp > 100:
                a = best_a.copy()
                a = _normalize(a + rng.normal(0, 0.04, n))
                c = _conv_self(a, L, m)
                cur = _score(c, n)
                v[:] = 0
                eta = eta_init * 0.4
                no_imp = 0

        if it % 800 == 0:
            print(f"  {label} it={it} best={best:.6f} eta={eta:.4f} temp={temp:.4f} mom={mom:.2f}")

    return best_a, best

def _coord_descent(a, L, m, n, time_limit, rng, label="CD"):
    deadline = time.monotonic() + time_limit
    a = _normalize(a.copy())
    c = _conv_self(a, L, m)
    best = _score(c, n)
    best_a = a.copy()

    factors = np.array([0.6, 0.75, 0.85, 0.92, 1.0, 1.08, 1.15, 1.3, 1.5])
    passes = 0

    while time.monotonic() < deadline - 3.0:
        passes += 1
        improved = 0
        idx = rng.permutation(n)

        for i in idx:
            if time.monotonic() > deadline - 3.0:
                break

            ai = a[i]
            if ai < 1e-14:
                candidates = np.array([1e-8, 1e-6, 1e-4, 1e-2])
            else:
                candidates = np.maximum(ai * factors, 0.0)

            k_lo = i
            k_hi = min(i + n, m)
            if k_hi <= k_lo:
                continue

            a_slice = a[0:k_hi - k_lo]
            c_aff = c[k_lo:k_hi]

            if k_lo > 0:
                left_max = float(c[:k_lo].max())
            else:
                left_max = 0.0
            if k_hi < m:
                right_max = float(c[k_hi:].max())
            else:
                right_max = 0.0
            unaff_max = max(left_max, right_max)

            deltas = candidates - ai
            dc = 2.0 * deltas[:, None] * a_slice[None, :]
            c_test = c_aff[None, :] + dc
            max_aff = c_test.max(axis=1)
            max_tot = np.maximum(max_aff, unaff_max)

            cur_max = float(c.max())
            best_j = int(max_tot.argmin())
            if max_tot[best_j] < cur_max - 1e-14:
                a[i] = candidates[best_j]
                delta = candidates[best_j] - ai
                c[k_lo:k_hi] += 2.0 * delta * a_slice
                improved += 1

        a = _normalize(a)
        c = _conv_self(a, L, m)
        sc = _score(c, n)
        if sc < best:
            best = sc
            best_a = a.copy()

        if passes % 10 == 0:
            print(f"  {label} pass={passes} score={best:.6f} improved={improved}/{n}")

        if improved == 0 and passes > 5:
            break

    return best_a, best

def _exact_score(seq):
    try:
        val = evaluate_sequence(seq)
    except NameError:
        if not isinstance(seq, list) or not seq:
            return float('inf')
        a = [max(0.0, min(1000.0, float(x))) for x in seq]
        nn = len(a)
        s = sum(a)
        if s < 0.01:
            return float('inf')
        b = np.convolve(a, a)
        return float(2.0 * nn * max(b) / (s * s))
    except Exception:
        return float('inf')
    if val is None:
        return float('inf')
    return float(val)

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    try:
        budget = float(budget_s)
    except (TypeError, ValueError):
        budget = 2400.0
    if not np.isfinite(budget) or budget < 10.0:
        budget = 2400.0

    start_time = time.monotonic()
    hard_deadline = start_time + budget - 5.0

    rng = np.random.default_rng(seed)
    n = 6121
    L = 1 << (2 * n - 2).bit_length()
    m = 2 * n - 1

    parent = _get_parent()
    if parent is not None and parent.size == n:
        a = _normalize(parent.copy())
        print(f"Using parent sequence, n={n}")
    else:
        a = _random_init(n, rng)
        print(f"No valid parent, using random init, n={n}")

    c = _conv_self(a, L, m)
    init_score = _score(c, n)
    print(f"Initial score: {init_score:.6f}")

    best_a = a.copy()
    best_score = init_score

    # Phase 1: Broad search
    t1 = min(250.0, 0.15 * budget)
    if time.monotonic() < hard_deadline - t1 - 100:
        print(f"Phase 1: broad search ({t1:.0f}s)")
        ta, sa = _optimize_phase(best_a, L, m, n, t1, rng,
                                  temp_start=2.0, temp_end=0.1,
                                  mom_start=0.0, mom_end=0.5,
                                  eta_init=0.15, label="P1")
        if sa < best_score:
            best_score = sa
            best_a = ta.copy()
        print(f"  After Phase 1: {best_score:.6f}")

    # Phase 2: Focused search
    t2 = min(400.0, 0.25 * budget)
    if time.monotonic() < hard_deadline - t2 - 80:
        print(f"Phase 2: focused search ({t2:.0f}s)")
        ta, sa = _optimize_phase(best_a, L, m, n, t2, rng,
                                  temp_start=0.1, temp_end=0.01,
                                  mom_start=0.3, mom_end=0.7,
                                  eta_init=0.08, label="P2")
        if sa < best_score:
            best_score = sa
            best_a = ta.copy()
        print(f"  After Phase 2: {best_score:.6f}")

    # Phase 3: Exact subgradient refinement
    t3 = min(300.0, 0.2 * budget)
    if time.monotonic() < hard_deadline - t3 - 60:
        print(f"Phase 3: exact subgradient ({t3:.0f}s)")
        ta, sa = _optimize_phase(best_a, L, m, n, t3, rng,
                                  temp_start=0.01, temp_end=0.001,
                                  mom_start=0.5, mom_end=0.8,
                                  eta_init=0.04, label="P3")
        if sa < best_score:
            best_score = sa
            best_a = ta.copy()
        print(f"  After Phase 3: {best_score:.6f}")

    # Phase 4: Coordinate descent
    t4 = min(500.0, 0.3 * budget)
    if time.monotonic() < hard_deadline - t4 - 30:
        print(f"Phase 4: coordinate descent ({t4:.0f}s)")
        ta, sa = _coord_descent(best_a, L, m, n, t4, rng, label="P4")
        if sa < best_score:
            best_score = sa
            best_a = ta.copy()
        print(f"  After Phase 4: {best_score:.6f}")

    # Phase 5: Fine coordinate descent
    t5 = min(200.0, 0.12 * budget)
    if time.monotonic() < hard_deadline - t5 - 20:
        print(f"Phase 5: fine CD ({t5:.0f}s)")
        ta, sa = _coord_descent_fine(best_a, L, m, n, t5, rng, label="P5")
        if sa < best_score:
            best_score = sa
            best_a = ta.copy()
        print(f"  After Phase 5: {best_score:.6f}")

    # Final short optimization burst if time remains
    remaining = hard_deadline - time.monotonic()
    if remaining > 50:
        t6 = min(remaining - 15, 100.0)
        if t6 > 20:
            print(f"Final polish ({t6:.0f}s)")
            ta, sa = _optimize_phase(best_a, L, m, n, t6, rng,
                                      temp_start=0.005, temp_end=0.001,
                                      mom_start=0.6, mom_end=0.85,
                                      eta_init=0.02, label="PF")
            if sa < best_score:
                best_score = sa
                best_a = ta.copy()
            print(f"  After final: {best_score:.6f}")

    # Evaluate with exact function
    seq = [float(x) for x in best_a]
    exact_val = _exact_score(seq)
    print(f"Exact evaluation: {exact_val:.6f}")

    if not np.isfinite(exact_val):
        seq = [1.0 / n] * n
        exact_val = _exact_score(seq)
        print(f"Fallback exact: {exact_val:.6f}")

    elapsed = time.monotonic() - start_time
    print(f"Total time: {elapsed:.1f}s, final score: {exact_val:.6f}")
    return seq
