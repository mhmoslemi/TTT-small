So the sequence elements can be any non-negative values up to 1000. The score is scale-invariant (if we multiply all elements by c, both max_b and sum_a^2 get multiplied by c^2, so the score is unchanged).

This means we can work with any normalization. I'll use sum(a) = 1 for the optimization.

So the score is approximately 1 + 2n * excess. The question is: can we make 2n * excess smaller with larger n?

If the sequence is "optimized" for a given n, the excess likely scales as 1/n (since we have n degrees of freedom to work with). In that case, 2n * excess would be O(1), and the score would be 1 + O(1), which is consistent with what we see.

But if the excess scales better than 1/n (e.g., as 1/n^{3/2}), then larger n would help.

In practice, for the "flat autocorrelation" problem, the best known constructions give excess = O(1/sqrt(n)) for the peak off-diagonal value (from the Welch bound or similar). Let me also add one more technique: **coordinate descent on the active set**. At the optimum, the active set S (indices where c_k = max) should be as large as possible (for a given max value). Here's a concrete algorithm for growing the active set:
1. Compute c = a * a, M = max(c), S = {k : c_k >= M - δ} for small δ
2. For each k in S, the constraint (a*a)[k] = M is "active"
3. The subgradient of max(c) at this point is the average of the gradients over S
4. Take a step in the negative subgradient direction (in log-space)
5. After the step, recompute c and update S

This is exactly what the active-set polish does. The key improvement is in the step size selection.

import numpy as np
import time
from scipy.optimize import minimize

"""
Multi-phase optimizer for minimum peak self-convolution sequences.

Strategy:
1. Coarse-to-fine Adam optimization in log-space with adaptive tau schedule
2. L-BFGS refinement on smooth surrogate  
3. Active-set gradient descent with Nesterov momentum and adaptive step sizes
4. Multiple n values explored; best result kept

Key improvements over baseline:
- Adaptive tau schedule (geometric from large to tiny)
- Nesterov momentum in polishing phase
- Aggressive step-size exploration with periodic restarts
- Multiplicative weight "pre-conditioning" step
- Multiple sequence lengths tried
"""

_G_N = 0
_G_NFFT = 0
_G_TAU = 1e-6
_G_L = 0


def _init_state(n, tau=1e-6):
    global _G_N, _G_NFFT, _G_TAU, _G_L
    _G_N = int(n)
    _G_L = 2 * _G_N - 1
    _G_NFFT = 1 << (3 * _G_N - 2).bit_length()
    _G_TAU = float(tau)


def _vec_softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.array([], dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.full_like(z, 1.0 / z.size, dtype=np.float64)
    return e / s


def _fft_selfconv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    N = 1 << (3 * n - 2).bit_length()
    L = 2 * n - 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L].real
    return np.maximum(c, 0.0)


def _fast_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-15:
        return np.inf
    c = _fft_selfconv(a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return float(2 * a.size * m / (s * s))


def _smooth_obj_grad(z, tau):
    n = _G_N
    N = _G_NFFT
    L = _G_L

    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n:
        return 1e12, np.zeros(max(1, n), dtype=np.float64)

    if not np.all(np.isfinite(z)):
        z = np.nan_to_num(z, nan=0.0, posinf=25.0, neginf=-25.0)
    z = np.clip(z, -25.0, 25.0)

    a = _vec_softmax(z)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L].real
    c = np.maximum(c, 0.0)

    m = float(np.max(c))
    if not np.isfinite(m):
        return 1e12, np.zeros(n, dtype=np.float64)

    diff = (c - m) / tau
    diff = np.clip(diff, -700.0, 0.0)
    e = np.exp(diff)
    se = float(np.sum(e))
    if se <= 0.0 or not np.isfinite(se):
        return 1e12, np.zeros(n, dtype=np.float64)

    p = e / se
    fp = np.fft.rfft(p, N)
    h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real

    s_h = float(np.dot(h, a))
    g = a * (h - s_h)
    grad = 2.0 * g

    val = m + tau * np.log(se)
    return float(val), grad


def _score_sort_key(item):
    return item[0]


def _adam_run(z0, tau_start, tau_end, n_iters, lr_start, lr_end, deadline):
    n = _G_N
    z = z0.copy()
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)

    best_z = z.copy()
    best_sc = _fast_score(_vec_softmax(z))

    for t in range(1, n_iters + 1):
        if time.monotonic() > deadline:
            break

        frac = (t - 1) / max(1, n_iters - 1)
        tau = tau_start * (tau_end / tau_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac

        val, grad = _smooth_obj_grad(z, tau)
        if not np.isfinite(val):
            break

        m_buf = beta1 * m_buf + (1.0 - beta1) * grad
        v_buf = beta2 * v_buf + (1.0 - beta2) * grad * grad
        m_hat = m_buf / (1.0 - beta1 ** t)
        v_hat = v_buf / (1.0 - beta2 ** t)

        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z = np.clip(z, -25.0, 25.0)

        a = _vec_softmax(z)
        sc = _fast_score(a)
        if sc < best_sc:
            best_sc = sc
            best_z = z.copy()

        if t % 500 == 0:
            print(f"    Adam t={t}/{n_iters} tau={tau:.2e} sc={sc:.9f} best={best_sc:.9f}")

    return best_z, best_sc


def _lbfgs_run(z0, tau, maxfun, deadline):
    global _G_TAU
    _G_TAU = tau
    n = _G_N

    def objfun(z):
        return _smooth_obj_grad(z, tau)

    z0 = np.clip(z0 - np.mean(z0), -25.0, 25.0)
    bounds = [(-25.0, 25.0)] * n
    opts = {
        "maxfun": int(maxfun),
        "maxiter": min(300, int(maxfun)),
        "ftol": 1e-14,
        "pgtol": 1e-12,
        "disp": False,
    }
    try:
        res = minimize(objfun, z0, jac=True, method="L-BFGS-B", bounds=bounds, options=opts)
        if time.monotonic() > deadline:
            return z0
        return np.asarray(res.x, dtype=np.float64).ravel()
    except Exception:
        return z0


def _mul_weight_precondition(a0, n_iters=200, eta=0.01, deadline=None):
    """Multiplicative weight update to roughly equalize convolution peaks."""
    n = _G_N
    N = _G_NFFT
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-30)
    s = float(np.sum(a))
    if s <= 0:
        return a0
    a = a / s

    best_a = a.copy()
    best_m = float(np.max(_fft_selfconv(a)))

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        c = _fft_selfconv(a)
        M = float(np.max(c))

        if M < best_m - 1e-18:
            best_m = M
            best_a = a.copy()

        # Compute "pressure" on each element
        active = c >= M * 0.99
        p = active.astype(np.float64)
        ps = float(np.sum(p))
        if ps > 0:
            p = p / ps

        fa = np.fft.rfft(a, N)
        fp = np.fft.rfft(p, N)
        h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real
        s_h = float(np.dot(h, a))
        contrib = h - s_h

        # Multiplicative update: reduce high-contribution elements
        a = a * np.exp(-eta * contrib / (M + 1e-30))
        a = np.maximum(a, 1e-30)
        a = a / float(np.sum(a))

    return best_a


def _active_polish(a0, deadline, momentum=0.85):
    """Active-set gradient descent with Nesterov momentum."""
    n = _G_N
    N = _G_NFFT
    L = _G_L

    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-15:
        return a0
    a = a / s

    b = np.log(np.maximum(a, 1e-300))
    b = b - np.mean(b)
    b = np.clip(b, -25.0, 25.0)

    best_a = a.copy()
    best_sc = _fast_score(a)
    best_m = best_sc / (2.0 * n)

    v = np.zeros(n)
    rng = np.random.default_rng(777)
    no_improve = 0
    it = 0
    last_escalation = 0

    while time.monotonic() < deadline:
        c = _fft_selfconv(a)
        F = float(np.max(c))

        if F < best_m - 1e-18:
            best_m = F
            best_sc = 2.0 * n * F
            best_a = a.copy()

        # Subgradient via active set
        tol = max(1e-12, F * 1e-8)
        active = c >= F - tol
        p = active.astype(np.float64)
        ps = float(np.sum(p))
        if ps == 0:
            idx = int(np.argmax(c))
            p[idx] = 1.0
            ps = 1.0
        p = p / ps

        fa = np.fft.rfft(a, N)
        fp = np.fft.rfft(p, N)
        h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real
        s_h = float(np.dot(h, a))
        g = a * (h - s_h)

        # Nesterov momentum
        v = momentum * v - g
        direction = v

        dn = float(np.dot(direction, direction))
        if dn < 1e-30:
            no_improve += 1
            if no_improve > 200:
                break
            continue

        # Adaptive step: backtracking with escalation
        accepted = False
        # Try a range of step sizes, larger ones first if we've been stuck
        base_steps = (1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 0.1, 0.3, 1.0, 3.0, 10.0)
        if no_improve > 50:
            base_steps = (1e-3, 3e-3, 1e-2, 0.05, 0.1, 0.3, 1.0, 3.0, 10.0, 30.0, 100.0)

        for eta in base_steps:
            b_t = b - eta * direction
            b_t = np.clip(b_t, -25.0, 25.0)
            a_t = _vec_softmax(b_t)
            F_t = float(np.max(_fft_selfconv(a_t)))
            if F_t < F - 1e-16:
                b = b_t
                a = a_t
                accepted = True
                break

        if not accepted:
            # Random escape in log-space
            escaped = False
            scales = (0.002, 0.01, 0.05, 0.2, 0.5, 1.0, 3.0)
            if no_improve > 100:
                scales = (0.05, 0.2, 0.5, 1.0, 3.0, 5.0, 10.0)
            for scale in scales:
                b_t = np.clip(b + scale * rng.standard_normal(n), -25.0, 25.0)
                a_t = _vec_softmax(b_t)
                F_t = float(np.max(_fft_selfconv(a_t)))
                if F_t < best_m - 1e-16:
                    b = b_t
                    a = a_t
                    v = np.zeros(n)
                    escaped = True
                    break

            if escaped:
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 400:
                    break
        else:
            no_improve = 0

        it += 1
        if it % 200 == 0:
            cur_sc = 2.0 * n * F
            print(f"    Polish it={it} cur={cur_sc:.9f} best={best_sc:.9f} "
                  f"noimp={no_improve} t_left={max(0.0, deadline - time.monotonic()):.0f}s")

    return best_a


def _gen_starts(n, rng):
    """Generate diverse starting points. Returns list of (score, z)."""
    starts = []
    u = np.linspace(0.0, 1.0, n, endpoint=True)
    x = 2.0 * u - 1.0

    def _add_f(f):
        f = np.asarray(f, dtype=np.float64).ravel()
        if f.size != n:
            return
        f = np.maximum(f, 1e-10)
        z = np.log(f)
        z = z - np.mean(z)
        z = np.clip(z, -20.0, 20.0)
        a = _vec_softmax(z)
        sc = _fast_score(a)
        if np.isfinite(sc):
            starts.append((float(sc), z.copy()))

    # Provided sequence
    try:
        g = globals().get("height_sequence_1")
        if g is not None:
            g = np.asarray(g, dtype=np.float64).ravel()
            if g.size > 1:
                g = np.clip(g, 1e-10, 1000.0)
                if np.all(np.isfinite(g)) and float(np.sum(g)) > 0.0:
                    x_old = np.linspace(0.0, 1.0, g.size, endpoint=True)
                    z_old = np.log(np.maximum(g, 1e-10))
                    z_old = z_old - np.mean(z_old)
                    z = np.interp(np.linspace(0.0, 1.0, n, endpoint=True), x_old, z_old)
                    _add_f(_vec_softmax(z))
    except Exception:
        pass

    # Two-level blocks with various ratios
    half = n // 2
    for ratio in (1.2, 1.5, 2.0, 3.0, 5.0, 8.0, 15.0):
        f = np.empty(n)
        f[:half] = 1.0
        f[half:] = float(ratio)
        _add_f(f)

    # Three-level blocks
    third = n // 3
    for r1, r2 in ((1.0, 3.0), (1.0, 5.0), (2.0, 5.0), (0.5, 4.0)):
        f = np.empty(n)
        f[:third] = r1
        f[third:2 * third] = 1.0
        f[2 * third:] = r2
        _add_f(f)

    # Linear profiles
    for A in (0.2, 0.5, 1.0, 1.5, 2.5, 4.0):
        f = 0.5 + A * (x + 1.0)
        _add_f(f)

    # Smooth step functions
    for A in (0.5, 1.0, 2.0, 4.0):
        for w in (0.01, 0.03, 0.08, 0.15, 0.3):
            sig = 1.0 / (1.0 + np.exp(-(x / max(1e-3, w))))
            f = 0.5 + A * sig
            _add_f(f)

    # Cosine/harmonic profiles
    for A in (0.2, 0.5, 1.0, 2.0):
        for m in (1, 2, 3, 5):
            f = 1.0 + A * np.cos(2.0 * np.pi * m * u)
            _add_f(f)
            f = 1.0 + A * np.sin(2.0 * np.pi * m * u)
            _add_f(f)

    # Quadratic
    for A in (0.3, 0.8, 1.5, 3.0):
        f = 1.0 + A * (x * x - 1.0 / 3.0)
        _add_f(f)

    # "Endpoint boosted": larger at ends
    for boost in (2.0, 4.0, 8.0):
        f = np.ones(n)
        f[0] = boost
        f[-1] = boost
        _add_f(f)

    # Random smooth (Fourier series)
    for _ in range(6):
        f = np.ones(n)
        for m in range(1, 8):
            f += 0.2 * rng.standard_normal() * np.sin(2.0 * np.pi * m * u + rng.uniform(0, 2 * np.pi))
            f += 0.2 * rng.standard_normal() * np.cos(2.0 * np.pi * m * u)
        _add_f(f)

    # Random two-level with smoothing
    for _ in range(4):
        lo, hi = sorted(rng.uniform(0.3, 5.0, size=2))
        f = np.where(rng.random(n) < 0.5, lo, hi)
        kernel = np.ones(min(11, n // 4 + 1)) / min(11, n // 4 + 1)
        f = np.convolve(f, kernel, mode='same')
        _add_f(f)

    starts.sort(key=_score_sort_key)
    return starts

            t_polish = max(time.monotonic() + 10, t_phase_polish - 10 - ci * 30)
            print(f"  Polish cand[{ci}] pre={sc:.9f}, time={t_polish - time.monotonic():.0f}s")

            a_pol = _active_polish(a_in, deadline=t_polish, momentum=0.85)
            sc_pol = _fast_score(a_pol)

            if sc_pol < best_overall_score:
                best_overall_score = sc_pol
                best_overall_n = n
                best_overall_a = a_pol.copy()
                print(f"    POLISH NEW BEST: {sc_pol:.9f}")

    # Final validation
    if best_overall_a is None:
        n = 8192
        best_overall_a = np.ones(n) / n
        best_overall_n = n

    best_overall_a = np.asarray(best_overall_a, dtype=np.float64).ravel()
    best_overall_a = np.nan_to_num(best_overall_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_overall_a = np.clip(best_overall_a, 0.0, 1000.0)
    s = float(np.sum(best_overall_a))
    if s < 0.01 or not np.isfinite(s):
        best_overall_a = np.ones(best_overall_n) / best_overall_n

    try:
        exact = evaluate_sequence(best_overall_a.tolist())
        print(f"\n{'='*60}")
        print(f"FINAL: n={best_overall_n}, official_score={exact:.9f}")
        print(f"{'='*60}")
    except Exception as e:
        print(f"Validation error: {e}")

    print(f"Total elapsed: {time.monotonic() - t_start:.1f}s")
    return [float(v) for v in best_overall_a.tolist()]
