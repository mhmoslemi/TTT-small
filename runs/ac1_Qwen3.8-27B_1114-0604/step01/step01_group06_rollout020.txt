"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 using:
1. Power-law family optimization (1D golden section)
2. L-BFGS-B in softmax log-space from diverse starts
3. Exact active-set gradient polishing
Uses FFT for fast convolution evaluation.
"""
import numpy as np
import time
from scipy.optimize import minimize

_OPT_N = 0
_OPT_NFFT = 0
_OPT_TAU = 1e-6
_OPT_BOUND = 30.0
_OPT_BOUNDS = []


def _set_opt_state(n, tau=1e-6):
    global _OPT_N, _OPT_NFFT, _OPT_TAU, _OPT_BOUND, _OPT_BOUNDS
    _OPT_N = int(n)
    _OPT_NFFT = 1 << (3 * _OPT_N - 2).bit_length()
    _OPT_TAU = float(tau)
    _OPT_BOUND = 30.0
    _OPT_BOUNDS = [(-_OPT_BOUND, _OPT_BOUND)] * _OPT_N


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.array([], dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.full_like(z, 1.0 / z.size, dtype=np.float64)
    return e / s


def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    N = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1].real
    return np.maximum(c, 0.0)


def _fast_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    c = _fft_conv(a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return float(2 * a.size * m / (s * s))


def _obj_grad(z):
    n = _OPT_N
    N = _OPT_NFFT
    tau = _OPT_TAU
    L = 2 * n - 1

    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n:
        return 1e12, np.zeros(max(1, z.size), dtype=np.float64)

    if not np.all(np.isfinite(z)):
        z = np.nan_to_num(z, nan=0.0, posinf=_OPT_BOUND, neginf=-_OPT_BOUND)
    z = np.clip(z, -_OPT_BOUND, _OPT_BOUND)

    a = _softmax(z)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L].real
    c = np.maximum(c, 0.0)

    m = float(np.max(c))
    if not np.isfinite(m):
        return 1e12, np.zeros(n, dtype=np.float64)

    diff = (c - m) / tau
    diff = np.clip(diff, -745.0, 0.0)
    e = np.exp(diff)
    se = float(np.sum(e))
    if se <= 0.0 or not np.isfinite(se):
        return 1e12, np.zeros(n, dtype=np.float64)

    p = e / se
    fp = np.fft.rfft(p, N)
    h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real

    s_h = float(np.dot(h, a))
    g = a * (h - s_h)
    grad = 2.0 * g

    val = m + tau * np.log(se)
    scale = float(n)
    val = val * scale
    grad = grad * scale

    pen = 1e-10
    val += 0.5 * pen * float(np.dot(z, z))
    grad = grad + pen * z

    if not np.isfinite(val):
        val = 1e12
        grad = np.zeros(n, dtype=np.float64)

    return float(val), grad


def _power_law_family_score(params):
    """Evaluate score for a = (t + eps)^p normalized to sum=1."""
    p = params[0]
    eps = params[1]
    n = _OPT_N
    t = np.arange(n, dtype=np.float64) / (n - 1)  # t in [0,1]
    a = (t + eps) ** p
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 0:
        return np.inf
    a = a / s
    return _fast_score(a)


def _optimize_power_law(deadline):
    """Optimize the power-law family (t+eps)^p over p and eps."""
    best_score = np.inf
    best_params = None

    # Coarse grid
    for p in np.arange(0.1, 4.1, 0.1):
        for eps in [0.01, 0.05, 0.1, 0.2, 0.5]:
            sc = _power_law_family_score([p, eps])
            if sc < best_score:
                best_score = sc
                best_params = [p, eps]

    if best_params is None:
        return None

    # Golden section refinement
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a_p, b_p = best_params[0] - 0.5, best_params[0] + 0.5
    a_e, b_e = max(0.001, best_params[1] - 0.15), best_params[1] + 0.15

    for _ in range(30):
        if time.monotonic() > deadline:
            break
        m1 = a_p + gr * (b_p - a_p)
        m2 = b_p - gr * (b_p - a_p)
        f1 = _power_law_family_score([m1, best_params[1]])
        f2 = _power_law_family_score([m2, best_params[1]])
        if f1 < f2:
            b_p = m2
        else:
            a_p = m1

    p_opt = (a_p + b_p) / 2.0

    a_p, b_p = best_params[0] - 0.3, best_params[0] + 0.3
    a_e, b_e = max(0.001, best_params[1] - 0.1), best_params[1] + 0.1

    for _ in range(20):
        if time.monotonic() > deadline:
            break
        m1 = a_e + gr * (b_e - a_e)
        m2 = b_e - gr * (b_e - a_e)
        f1 = _power_law_family_score([p_opt, m1])
        f2 = _power_law_family_score([p_opt, m2])
        if f1 < f2:
            b_e = m2
        else:
            a_e = m1

    eps_opt = (a_e + b_e) / 2.0
    final_sc = _power_law_family_score([p_opt, eps_opt])

    if final_sc < best_score:
        best_score = final_sc
        best_params = [p_opt, eps_opt]

    # Generate the actual sequence
    n = _OPT_N
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = (t + best_params[1]) ** best_params[0]
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s > 0:
        a = a / s
    else:
        a = np.ones(n) / n

    print(f"Power-law family: p={best_params[0]:.4f}, eps={best_params[1]:.4f}, score={best_score:.9f}")
    return a


def _make_variational_starts(n):
    """Generate diverse starting points for L-BFGS."""
    starts = []
    t = np.arange(n, dtype=np.float64) / (n - 1)
    x = 2.0 * t - 1.0  # x in [-1, 1]

    # Add various structured profiles
    profiles = []

    # Power laws
    for p in [0.8, 1.0, 1.2, 1.5, 2.0, 2.5, 3.0]:
        for eps in [0.0, 0.05, 0.1]:
            f = (t + eps) ** p
            profiles.append(f)

    # Cosine windows
    for width in [0.5, 0.7, 0.9, 1.0, 1.3, 1.6]:
        f = np.maximum(np.cos(np.pi * x / width), 0.0)
        profiles.append(f)

    # Beta-like distributions
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        for beta in [0.5, 1.0, 1.5, 2.0, 3.0]:
            f = t ** (alpha - 1) * (1 - t) ** (beta - 1)
            f = np.where(t > 0.001, f, 0.0)
            f = np.where(t < 0.999, f, 0.0)
            profiles.append(f)

    # Polynomial profiles
    for c2 in [-0.3, -0.15, 0.0, 0.15, 0.3]:
        for c4 in [-0.1, 0.0, 0.1]:
            f = 1.0 + c2 * x * x + c4 * x ** 4
            f = np.maximum(f, 0.01)
            profiles.append(f)

    # Sigmoid-like
    for steep in [2, 3, 5, 8, 12]:
        f = 1.0 / (1.0 + np.exp(-steep * x))
        profiles.append(f)

    # Exponential decay from edges
    for rate in [1, 2, 3, 5, 8]:
        f = np.exp(-rate * t)
        profiles.append(f)
        f = np.exp(-rate * (1 - t))
        profiles.append(f)

    # Convert to z-space
    for f in profiles:
        f = np.asarray(f, dtype=np.float64)
        if f.size != n:
            continue
        f = np.maximum(f, 1e-10)
        s = float(np.sum(f))
        if s <= 0:
            continue
        f = f / s
        z = np.log(np.maximum(f, 1e-300))
        z = z - np.mean(z)
        z = np.clip(z, -_OPT_BOUND + 1.0, _OPT_BOUND - 1.0)
        a = _softmax(z)
        sc = _fast_score(a)
        if np.isfinite(sc):
            starts.append((sc, z))

    starts.sort(key=lambda x: x[0])
    return starts[:25]


def _resample_sequence(g, n):
    """Resample a sequence to length n using log-space interpolation."""
    g = np.asarray(g, dtype=np.float64).ravel()
    if g.size < 2:
        return None
    g = np.clip(g, 1e-300, 1e6)
    z_old = np.log(g)
    z_old = z_old - np.mean(z_old)
    x_old = np.linspace(0.0, 1.0, g.size)
    x_new = np.linspace(0.0, 1.0, n)
    z = np.interp(x_new, x_old, z_old)
    z = z - np.mean(z)
    z = np.clip(z, -_OPT_BOUND + 1.0, _OPT_BOUND - 1.0)
    return z


def _polish(a0, deadline):
    """Exact active-set gradient polishing in log-space."""
    n = _OPT_N
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    if a.size != n:
        return a0

    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-12:
        return a0
    a = a / s

    b = np.log(np.maximum(a, 1e-300))
    b = np.clip(b - np.mean(b), -_OPT_BOUND, _OPT_BOUND)

    best = a.copy()
    best_score = _fast_score(a)
    best_maxc = float(np.max(_fft_conv(best)))

    t0 = time.monotonic()
    no_improve = 0
    it = 0
    tau = _OPT_TAU

    while time.monotonic() < deadline:
        c = _fft_conv(a)
        F = float(np.max(c))

        if F < best_maxc - 1e-18:
            best = a.copy()
            best_score = 2.0 * n * F
            best_maxc = F

        # Compute smooth max gradient
        diff = (c - F) / tau
        diff = np.clip(diff, -745.0, 0.0)
        e = np.exp(diff)
        se = float(np.sum(e))
        if se <= 0.0 or not np.isfinite(se):
            break
        p = e / se

        N = _OPT_NFFT
        fa = np.fft.rfft(a, N)
        fp = np.fft.rfft(p, N)
        h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real
        s_h = float(np.dot(h, a))
        g = a * (h - s_h)
        g = g * float(n)
        g = np.clip(g, -100.0, 100.0)

        accepted = False
        for eta in [5e-5, 1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 1e-1, 3e-1, 1.0, 3.0, 10.0]:
            b_t = np.clip(b - eta * g, -_OPT_BOUND, _OPT_BOUND)
            a_t = _softmax(b_t)
            F_t = float(np.max(_fft_conv(a_t)))
            if F_t < F - 1e-15:
                b = b_t
                a = a_t
                accepted = True
                break

        if not accepted:
            # Try random perturbations of decreasing scale
            rng = np.random.default_rng(it * 7919 + 12345)
            escaped = False
            for scale in [0.003, 0.01, 0.03, 0.1, 0.3, 1.0]:
                b_t = np.clip(b + scale * rng.standard_normal(n), -_OPT_BOUND, _OPT_BOUND)
                a_t = _softmax(b_t)
                F_t = float(np.max(_fft_conv(a_t)))
                if F_t < best_maxc - 1e-15:
                    b = b_t
                    a = a_t
                    escaped = True
                    break
            if escaped:
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 300:
                    break
        else:
            no_improve = 0

        it += 1
        if it % 100 == 0:
            cur = 2.0 * n * F
            print(f"  polish it={it} cur={cur:.9f} best={best_score:.9f} "
                  f"t_left={max(0, deadline - time.monotonic()):.0f}s")

    return best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for optimal non-negative coefficient sequence minimizing
    2*n*max(convolve(a,a))/(sum(a))^2 using multi-stage optimization.
    """
    budget_s = max(120.0, float(budget_s))
    deadline = time.monotonic() + budget_s - 10.0
    n = 8192
    _set_opt_state(n, tau=1e-6)

    rng = np.random.default_rng(int(seed))
    t0 = time.monotonic()
    print(f"Starting optimization: n={n}, budget={budget_s:.0f}s")

    best_a = None
    best_score = np.inf

    # Stage 1: Load previous best and resample
    starts = []
    try:
        g = globals().get("height_sequence_1")
        if g is not None:
            g = np.asarray(g, dtype=np.float64)
            if g.size > 10:
                z = _resample_sequence(g, n)
                if z is not None:
                    a = _softmax(z)
                    sc = _fast_score(a)
                    if np.isfinite(sc):
                        starts.append((sc, z))
                        best_a = a.copy()
                        best_score = sc
                        print(f"Resampled previous best: score={sc:.9f}")
    except Exception as e:
        print(f"Could not load previous sequence: {e}")

    # Stage 2: Power-law family optimization
    try:
        if time.monotonic() < deadline - 300:
            pl_a = _optimize_power_law(deadline - 300)
            if pl_a is not None:
                sc = _fast_score(pl_a)
                if np.isfinite(sc):
                    if sc < best_score:
                        best_score = sc
                        best_a = pl_a.copy()
                    z = _resample_sequence(pl_a * np.sum(pl_a), n)
                    if z is not None:
                        starts.append((sc, z))
    except Exception as e:
        print(f"Power-law optimization failed: {e}")

    # Stage 3: Generate variational starts
    if time.monotonic() < deadline - 200:
        var_starts = _make_variational_starts(n)
        for sc, z in var_starts:
            starts.append((sc, z))
            if sc < best_score:
                best_score = sc
                a = _softmax(z)
                best_a = a.copy()

    # Sort starts by score
    starts.sort(key=lambda x: x[0])
    print(f"Total starts: {len(starts)}, best start score: {starts[0][0]:.9f}")

    # Stage 4: L-BFGS-B optimization from best starts
    candidates = []
    if best_a is not None:
        candidates.append((best_score, best_a.copy()))

    for idx, (start_score, z0) in enumerate(starts[:10]):
        if time.monotonic() > deadline - 360:
            break

        if idx < 3:
            maxfun = 800
            maxiter = 250
        elif idx < 6:
            maxfun = 500
            maxiter = 150
        else:
            maxfun = 300
            maxiter = 100

        z0 = np.asarray(z0, dtype=np.float64).ravel()
        if z0.size != n:
            continue
        z0 = np.clip(z0 - np.mean(z0), -_OPT_BOUND, _OPT_BOUND)

        opts = {
            "maxfun": int(maxfun),
            "maxiter": int(maxiter),
            "ftol": 1e-13,
            "pgtol": 1e-11,
            "disp": False,
        }
        try:
            res = minimize(_obj_grad, z0, jac=True, method="L-BFGS-B",
                           bounds=_OPT_BOUNDS, options=opts)
            if res is not None and hasattr(res, "x"):
                z_new = np.asarray(res.x, dtype=np.float64).ravel()
                if z_new.size == n:
                    a = _softmax(np.clip(z_new, -_OPT_BOUND, _OPT_BOUND))
                    sc = _fast_score(a)
                    if np.isfinite(sc):
                        if sc < best_score:
                            best_score = sc
                            best_a = a.copy()
                            print(f"  L-BFGS {idx}: score={sc:.9f} "
                                  f"t={time.monotonic() - t0:.0f}s")
                        candidates.append((sc, a.copy()))
        except Exception as e:
            print(f"  L-BFGS {idx} failed: {e}")

    # Keep top candidates
    candidates.sort(key=lambda x: x[0])
    # Deduplicate
    unique = []
    seen = set()
    for sc, a in candidates:
        key = tuple(np.round(a[::64], 8).tolist())
        if key not in seen:
            seen.add(key)
            unique.append((sc, a))
    candidates = unique[:5]

    # Stage 5: Polish best candidates
    if candidates:
        for ci, (sc, a) in enumerate(candidates):
            time_left = deadline - time.monotonic()
            if time_left < 30:
                break

            if ci == 0:
                pol_deadline = time.monotonic() + min(time_left - 15, 1200)
            else:
                pol_deadline = time.monotonic() + min(time_left / (ci + 1), 400)

            try:
                a2 = _polish(a, pol_deadline)
                sc2 = _fast_score(a2)
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_a = np.asarray(a2, dtype=np.float64).copy()
                    print(f"  Polished cand {ci}: score={sc2:.9f} "
                          f"t={time.monotonic() - t0:.0f}s")
            except Exception as e:
                print(f"  Polish {ci} failed: {e}")

    # Final validation
    if best_a is None:
        best_a = np.ones(n, dtype=np.float64) / n

    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = float(np.sum(best_a))
    if s < 0.01 or not np.isfinite(s):
        best_a = np.ones(n, dtype=np.float64) / n
    else:
        best_a = best_a / s

    final_sc = _fast_score(best_a)
    print(f"Final internal score: {final_sc:.9f}")

    try:
        exact = evaluate_sequence(best_a.tolist())
        if np.isfinite(exact):
            print(f"Official evaluator score: {float(exact):.9f}")
    except Exception:
        pass

    return [float(v) for v in best_a.tolist()]
