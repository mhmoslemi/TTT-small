    """Project v onto probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if n == 0:

    vs = v[order]
    cssv = np.cumsum(vs)
    rho = np.arange(1, n + 1)

```python
"""
Multi-phase optimizer for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Phases:
  1. L2-flatness pre-training (minimize ||conv||_2^2 with simplex constraint)
  2. Smooth max with temperature annealing (log-sum-exp surrogate)
  3. Sharp max refinement (adaptive top-k peaks)
  4. Final polish with very sharp top-k

Uses projected Adam (simplex constraint), FFT-based gradients,
periodic mutation restarts, and diverse initializations.
"""

import numpy as np
import time


# ---------------------------------------------------------------------------
# Core FFT helpers
# ---------------------------------------------------------------------------

def _fft_convolve(a, b):
    """Linear convolution of 1-D arrays via real FFT."""
    n, m = len(a), len(b)
    M = n + m - 1
    N = 1 << (M - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b, N), N)[:M]


def _self_conv(a):
    """Linear self-convolution of a (length n) -> length 2n-1."""
    return _fft_convolve(a, a)


def _cross_corr_p_a(p, a):
    """
    Compute g[i] = sum_{l=0}^{n-1} p[i+l] * a[l] for i=0..n-1.
    p has length M=2n-1, a has length n.
    Uses FFT cross-correlation.
    """
    n = len(a)
    M = len(p)
    N = 1 << (n + M - 1 - 1).bit_length()
    A = np.fft.rfft(a, N)
    P = np.fft.rfft(p, N)
    cc = np.fft.irfft(np.conj(A) * P, N)
    # g[i] = cc[(N-i) % N]
    # g[0] = cc[0], g[1] = cc[N-1], ..., g[n-1] = cc[N-n+1]
    if n == 1:
        return cc[:1].copy()
    idx = np.empty(n, dtype=np.intp)
    idx[0] = 0
    idx[1:] = np.arange(N - 1, N - n, -1)
    return cc[idx]


# ---------------------------------------------------------------------------
# Simplex projection
# ---------------------------------------------------------------------------

def _simplex_project(v):
    """Project vector v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    order = np.argsort(v)[::-1]
    vs = v[order]
    cssv = np.cumsum(vs)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = vs > (cssv - 1.0) / rho
    indices = np.where(cond)[0]
    if len(indices) == 0:
        return np.ones(n) / n
    k = indices[-1] + 1
    tau = (cssv[indices[-1]] - 1.0) / k
    result = np.zeros(n)
    result[order] = np.maximum(0.0, vs - tau)
    return result


# ---------------------------------------------------------------------------
# Adam optimizer
# ---------------------------------------------------------------------------

def _adam_step(a, g, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    """One Adam step, returns updated a (not in-place for cleanliness)."""
    t1 = b1 ** t
    t2 = b2 ** t
    m[:] = b1 * m + (1.0 - b1) * g
    v[:] = b2 * v + (1.0 - b2) * g * g
    mh = m / (1.0 - t1)
    vh = v / (1.0 - t2)
    return a - lr * mh / (np.sqrt(vh) + eps)


# ---------------------------------------------------------------------------
# Objective
# ---------------------------------------------------------------------------

def _obj_value(n, a):
    """Compute 2*n*max(conv)/(sum)^2 for normalized a (sum=1)."""
    c = _self_conv(a)
    return 2.0 * n * float(c.max())

# Initializations
# ---------------------------------------------------------------------------

def _make_inits(n, rng):
    """Generate diverse initializations, all on the simplex."""
    inits = []
    inits.append(np.ones(n) / n)
    # Ramps
    for alpha in (0.3, 0.7, 1.2, 2.0):
        x = np.linspace(0, 1, n)
        a = np.clip(1.0 + alpha * (2.0 * x - 1.0), 0.01, None)
        inits.append(a / a.sum())
    # Quadratic
    for alpha in (0.5, 1.5, 3.0):
        x = np.linspace(0, 1, n)
        a = np.clip(1.0 + alpha * (2.0 * x - 1.0) ** 2, 0.01, None)
        inits.append(a / a.sum())
    # Cosine
    for freq, alpha in ((1, 0.5), (2, 0.5), (3, 0.4), (5, 0.3), (8, 0.2)):
        x = np.linspace(0, 1, n)
        a = np.clip(1.0 + alpha * np.cos(2.0 * np.pi * freq * x), 0.01, None)
        inits.append(a / a.sum())
    # Gaussian bumps
    for center, width in ((0.5, 0.15), (0.3, 0.1), (0.7, 0.1), (0.5, 0.3)):
        x = np.linspace(0, 1, n)
        a = np.exp(-((x - center) ** 2) / (2.0 * width ** 2))
        a = np.clip(a, 1e-6, None)
        inits.append(a / a.sum())
    # Chirp-like
    for freq in (2, 5, 10):
        x = np.linspace(0, 1, n)
        a = np.clip(1.0 + 0.4 * np.sin(2.0 * np.pi * freq * x ** 2), 0.01, None)
        inits.append(a / a.sum())
    # Random
    for _ in range(5):
        a = rng.random(n) + 0.05
        inits.append(a / a.sum())
    # Sparse-ish (gapped)
    for gap in (3, 5):
        a = np.zeros(n)
        a[::gap] = 1.0
        a += 0.01
        inits.append(a / a.sum())
    # Two-sided (high at edges)
    x = np.linspace(0, 1, n)
    a = np.clip(1.0 - 1.5 * np.sin(np.pi * x), 0.01, None)
    inits.append(a / a.sum())
    return inits

def _phase_l2(a0, end_time, lr=0.3, patience=300, rng=None):
    """Minimize ||conv(a,a)||_2^2 (equiv to flattening) on simplex."""
    n = len(a0)
    a = a0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_L = np.inf
    best_a = a.copy()
    no_improve = 0

    while time.monotonic() < end_time:
        c = _self_conv(a)
        c2_sum = float((c * c).sum())
        S = float(a.sum())
        L = c2_sum / (S ** 4)

        if L < best_L:
            best_L = L
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            break

        # Gradient of ||c||_2^2 / S^4
        # d||c||_2^2/da[i] = 4 * sum_l c[i+l]*a[l]
        corr = _cross_corr_p_a(c, a)  # corr[i] = sum_l c[i+l]*a[l]
        g = 4.0 * corr / (S ** 4) - 4.0 * c2_sum / (S ** 5)
        t += 1
        a = _adam_step(a, g, m, v, t, lr)
        a = _simplex_project(a)

    return best_a

def _phase_smooth_max(a0, end_time, T_start=3.0, T_end=2000.0, lr=0.05,
                      patience=400, rng=None):
    """Minimize log-sum-exp smoothed max of convolution."""
    n = len(a0)
    a = a0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_score = np.inf
    best_a = a.copy()
    no_improve = 0
    start_t = time.monotonic()
    total = max(1e-6, end_time - start_t)

    while time.monotonic() < end_time:
        elapsed = time.monotonic() - start_t
        progress = min(1.0, elapsed / total)
        # Exponential temperature ramp
        T = T_start * (T_end / T_start) ** progress

        c = _self_conv(a)
        score = 2.0 * n * float(c.max())

        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            if rng is None:
                rng = np.random.default_rng(t)
            scale = 0.03 + 0.07 * rng.random()
            a = best_a + scale * rng.standard_normal(n)
            a = _simplex_project(a)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            no_improve = 0
            patience = max(60, patience // 2)
            continue

        # Gradient of (1/T)log(sum exp(Tc))
        tc = T * c
        tc_max = tc.max()
        exp_tc = np.exp(tc - tc_max)
        p = exp_tc / exp_tc.sum()

        corr = _cross_corr_p_a(p, a)
        g = 2.0 * corr  # This is the gradient of the smooth max (times 2 from conv)

        t += 1
        a = _adam_step(a, g, m, v, t, lr)
        a = _simplex_project(a)

    return best_a, best_score
