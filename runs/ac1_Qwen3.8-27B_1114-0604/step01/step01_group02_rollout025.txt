    # (grad)_j = 2 * a_{k-j} for 0 <= k-j <= n-1
    j_start = max(0, k - n + 1)
    j_end = min(k, n - 1)
    if j_start <= j_end:
        # For j in [j_start, j_end], k-j ranges from k-j_start to k-j_end
        # a_{k-j} for j in [j_start, j_end] is a[k-j_start : k-j_end+1] reversed
        grad[j_start:j_end+1] = 2.0 * a[k-j_end:k-j_start+1][::-1]
    return grad

def objective_and_grad(a, k_peak):
    """Compute f and its gradient, assuming peak at k_peak."""
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return 1e10, np.zeros_like(a)
    c = fast_self_conv(a)
    M = c[k_peak]
    f = 2.0 * n * M / (S * S)
    gc = grad_conv_at_k(a, k_peak)
    grad = (2.0 * n / (S * S)) * (gc - (2.0 * M / S) * a)
    return f, grad

def line_search(a, grad, f0, k_peak, max_iter=50):
    """Backtracking line search."""
    n = len(a)
    step = 1.0
    # Scale step to be reasonable
    grad_norm = np.linalg.norm(grad)
    if grad_norm < 1e-15:
        return a, f0, 0
    step = 0.1 * np.linalg.norm(a) / grad_norm
    
    for it in range(max_iter):
        a_trial = np.clip(a - step * grad, 0.0, 1000.0)
        S = np.sum(a_trial)
        if S < 1e-10:
            step *= 0.5
            continue
        c = fast_self_conv(a_trial)
        f_trial = 2.0 * n * np.max(c) / (S * S)
        if f_trial < f0 * (1 - 1e-4 * step):  # sufficient decrease
            return a_trial, f_trial, it + 1
        step *= 0.5
    return a_trial, f_trial, max_iter

def peak_shaving_optimize(a0, time_limit, rng):
    """Iteratively reduce the dominant peak using gradient + line search."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    # Normalize sum to n (homogeneous, so this doesn't change objective)
    S = a.sum()
    if S > 1e-10:
        a = a * (n / S)  # Now sum = n
    
    c = fast_self_conv(a)
    f = 2.0 * n * np.max(c) / (a.sum() ** 2)
    best_a = a.copy()
    best_f = f
    start = time.monotonic()
    stall_count = 0
    
    while time.monotonic() - start < time_limit:
        c = fast_self_conv(a)
        f = 2.0 * n * np.max(c) / (a.sum() ** 2)
        if f < best_f:
            best_f = f
            best_a = a.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        if stall_count > 200:
            # Restart with perturbation
            a = best_a * (1.0 + 0.05 * rng.standard_normal(n))
            a = np.clip(a, 0.0, 1000.0)
            S = a.sum()
            if S > 1e-10:
                a = a * (n / S)
            stall_count = 0
            continue
        
        k_peak = int(np.argmax(c))
        f_curr, grad = objective_and_grad(a, k_peak)
        a, f_new, _ = line_search(a, grad, f_curr, k_peak)
    
    return best_f, best_a

"""
Optimized step-function search using multi-strategy gradient methods.
Combines log-sum-exp smoothed L-BFGS, adaptive peak-shaving, and Fourier-mode
restarts to minimize 2n*max(conv(a,a))/(sum(a))^2 for non-negative a.
"""

import numpy as np
import time
from scipy.optimize import minimize


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def lse_gradients(a, beta):
    """Compute log-sum-exp smoothed objective and gradient.
    
    phi(a) = (2n/S^2) * (1/beta) * log(sum_k exp(beta * c_k))
    """
    n = len(a)
    S = np.sum(a)
    if S < 1e-10:
        return 1e10, np.zeros_like(a)
    c = fast_self_conv(a)
    bc = beta * c
    bc_max = np.max(bc)
    logZ = bc_max + np.log(np.sum(np.exp(bc - bc_max)))
    p = np.exp(bc - bc_max) / np.sum(np.exp(bc - bc_max))  # softmax weights
    
    # Gradient of logZ w.r.t. a: sum_k p_k * grad(c_k)
    # (grad c_k)_j = 2*a_{k-j} for valid k-j
    # sum_k p_k * 2*a_{k-j} = 2 * (flip(y) convolve a)_j where y_k = p_k
    # Actually: sum_k p_k * a_{k-j} = cross-correlation
    # Let's compute it as: for each j, sum over k where 0<=k-j<=n-1 of p_k * a_{k-j}
    # This equals (p_reversed * a) at appropriate position, computed via FFT
    M_total = 2 * n - 1
    M_fft = 1 << (M_total - 1).bit_length()
    P = np.zeros(M_fft)
    P[:M_total] = p
    A = np.fft.rfft(a, M_fft)
    # sum_k p_k * a_{k-j}: this is the cross-correlation of p and a
    # cross_corr[j] = sum_k p[k] * a[k-j] = IFFT(FFT(p) * conj(FFT(a)))
    cross = np.fft.irfft(np.fft.rfft(P, M_fft) * np.conj(A), M_fft)
    # cross[j] = sum_k P_k * a_{(k-j) mod M_fft}
    # We need the "linear" version: sum_k p_k * a_{k-j} for 0<=k-j<=n-1
    # For j in 0..n-1, the valid k are max(0,j) to min(2n-2, j+n-1)
    # cross[j] should give us this directly (with zero-padding handling)
    grad_logZ = 2.0 * beta * cross[:n]
    
    # Full gradient of phi
    phi = (2.0 * n / (S * S)) * (logZ / beta)
    grad = (2.0 * n / (S * S)) * (grad_logZ / beta - (2.0 * logZ / (beta * S)) * a)
    return phi, grad


def topk_gradient(a, c, k_top, temp_frac):
    """Gradient using top-k peaks with exponential weighting."""
    n = len(a)
    S = np.sum(a)
    k_act = min(k_top, len(c))
    if k_act == 1:
        idx = np.array([int(np.argmax(c))])
        wts = np.array([1.0])
    else:
        idx = np.argpartition(c, -k_act)[-k_act:]
        vals = c[idx]
        order = np.argsort(vals)[::-1]
        idx = idx[order]
        vals = vals[order]
        top = float(vals[0])
        if top > 1e-12:
            temp = temp_frac * top
            wts = np.exp((vals - top) / temp)
            wts = wts / wts.sum()
        else:
            wts = np.ones(k_act) / k_act
    
    grad_c = np.zeros(n)
    M_w = 0.0
    for i in range(idx.size):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi <= 1e-15:
            continue
        M_w += wi * c[ki]
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start <= j_end:
            grad_c[j_start:j_end + 1] += wi * 2.0 * a[j_start:j_end + 1][::-1]
    
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_c / S2 - 2.0 * M_w / S3)
    return grad, M_w


def adam_step(a, grad, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    m[:] = b1 * m + (1.0 - b1) * grad
    v[:] = b2 * v + (1.0 - b2) * grad * grad
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)
    a -= lr * mh / (np.sqrt(vh) + eps)


def fourier_perturb(a, rng, n_modes=8, amplitude=0.1):
    """Add Fourier-mode perturbations to escape local minima."""
    n = len(a)
    perturb = np.zeros(n)
    for m in range(1, n_modes + 1):
        freq = m * n // n_modes
        phase = rng.random() * 2 * np.pi
        amp = amplitude / np.sqrt(m)
        if m == 1:
            perturb += amp * (rng.random() - 0.5)  # mean shift
        perturb += amp * np.cos(2 * np.pi * freq * np.arange(n) / n + phase)
    return a + perturb


def optimize_adam(a0, end_time, rng, lr_init=0.1, topk=8, temp_frac=0.1, patience=300):
    """Adam optimization with top-k gradient and restarts."""
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    if a.sum() < 1e-6:
        a = np.ones(n)
    
    c = fast_self_conv(a)
    S = a.sum()
    score = 2.0 * n * np.max(c) / (S * S)
    best_score = score
    best_a = a.copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr_init
    no_improve = 0
    pat = patience
    
    while time.monotonic() < end_time:
        c = fast_self_conv(a)
        S = a.sum()
        if S < 1e-6:
            a = best_a.copy()
            c = fast_self_conv(a)
            S = a.sum()
        score = 2.0 * n * np.max(c) / (S * S)
        
        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= pat:
            # Fourier-mode restart
            a = fourier_perturb(best_a, rng, n_modes=4 + int(rng.random() * 8),
                                amplitude=0.05 + 0.1 * rng.random())
            a = np.clip(a, 0.0, 1000.0)
            if a.sum() < 1e-6:
                a = best_a.copy()
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            lr = max(0.01, lr * 0.6)
            no_improve = 0
            pat = max(60, pat // 2)
            continue
        
        grad, _ = topk_gradient(a, c, topk, temp_frac)
        t += 1
        adam_step(a, grad, m, v, t, lr)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 1e-6:
            a = np.clip(a + 0.1 * best_a, 0.0, 1000.0)
    
    return best_score, best_a


def optimize_lse_lbfgs(a0, end_time, beta=5.0):
    """L-BFGS optimization using log-sum-exp smoothed objective."""
    n = len(a0)
    a0 = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    
    def obj_and_grad(x):
        if time.monotonic() > end_time:
            raise RuntimeError("timeout")
        f, g = lse_gradients(x, beta)
        # Penalize negative values
        neg = np.maximum(0, -x)
        f += 1000.0 * np.sum(neg ** 2)
        g += 2000.0 * neg
        return f, g
    
    try:
        res = minimize(obj_and_grad, a0, method='L-BFGS-B',
                       options={'maxiter': 5000, 'ftol': 1e-12, 'gtol': 1e-8,
                                'maxls': 50, 'disp': False})
        if hasattr(res, 'x') and res.x is not None:
            a_opt = np.clip(res.x, 0.0, 1000.0)
            c = fast_self_conv(a_opt)
            S = a_opt.sum()
            if S > 1e-6:
                score = 2.0 * n * np.max(c) / (S * S)
                return score, a_opt
    except (RuntimeError, Exception):
        pass
    return np.inf, a0
