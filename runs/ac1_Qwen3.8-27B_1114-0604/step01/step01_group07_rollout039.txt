    """Compute subgradient of max(lin_k) using top K lags."""
    n = len(a)
    if K < len(c):
        top_idx = np.argpartition(c, -K)[-K:]
    else:
        top_idx = np.arange(len(c))
    
    grad = np.zeros(n, dtype=np.float64)
    for k in top_idx:
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_hi >= j_lo:
            start = k - j_hi
            end = k - j_lo
            grad[j_lo:j_hi+1] += 2.0 * a[start:end+1]
    grad /= float(len(top_idx))
    return grad

    Projective gradient descent to minimize max(linear_autocorrelation).
    Works on the simplex {a >= 0, sum(a) = 1}.
    """

    a = a.astype(np.float64).copy()
    a /= a.sum()
    
    best_a = a.copy()
    best_score = _score_internal(a)
    best_conv = np.max(_fft_convolve(a))

    no_improve = 0
    it = 0
    
    if verbose:

        mx = float(np.max(c))

            no_improve = 0

        else:
            no_improve += 1
        
        # Compute subgradient
        grad = _subgradient_topk(a, c, K)
        grad -= np.mean(grad)
        
        # Multiplicative update
        a = a * np.exp(-lr * grad)
        a = np.clip(a, 1e-15, None)
        a /= a.sum()
        
        # Adapt learning rate
        if no_improve == 0:
            lr = min(lr * 1.005, 10.0)
        elif no_improve > 50:
            lr = max(lr * 0.95, 0.001)
            if no_improve > 500:
                if verbose:

                break
    
    if verbose:
        print(f"  [opt] done n={n} best_score={best_score:.8f} iterations={it}")
    
    return best_a, best_score
