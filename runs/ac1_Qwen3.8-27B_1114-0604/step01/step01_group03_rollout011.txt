"""
Search for non-negative step heights minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses projected gradient descent with Adam optimizer on the simplex,
log-sum-exp smoothing with geometric temperature annealing,
multiple restarts, and coordinate descent refinement.

import time

def _next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} using Euler's algorithm."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _compute_gradient(a, w, L):
    """Compute g[i] = sum_j a[j] * w[i+j] via FFT."""
    n = len(a)
    a_rev = a[::-1]
    F_ar = np.fft.rfft(a_rev, L)
    F_w = np.fft.rfft(w, L)
    g_full = np.fft.irfft(F_ar * F_w, L)
    g = g_full[n - 1:2 * n - 1]
    return g


def _compute_conv(a, L):
    """Compute conv(a, a) via FFT."""
    F_a = np.fft.rfft(a, L)
    c = np.fft.irfft(F_a * F_a, L)
    return c


def _exact_score(a):
    """Compute exact score for array a (assumes sum(a)=1)."""
    n = len(a)
    c = np.convolve(a, a)
    return float(2.0 * n * np.max(c))


def _adam_update(a, g, m_buf, v_buf, adam_t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam update with projection onto simplex."""
    m_buf[:] = beta1 * m_buf + (1 - beta1) * g
    v_buf[:] = beta2 * v_buf + (1 - beta2) * g * g
    m_hat = m_buf / (1 - beta1 ** adam_t)
    v_hat = v_buf / (1 - beta2 ** adam_t)
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a_new, m_buf, v_buf


def _run_adam_phase(a, L, T_start, T_end, lr_start, lr_end, n_iters, n):
    """Run Adam optimization with temperature annealing for n_iters steps."""
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    adam_t = 0
    a = a.copy()

    for it in range(n_iters):
        frac = it / max(1, n_iters - 1)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac

        # Compute convolution
        c = _compute_conv(a, L)[:2 * n - 1]
        c = np.maximum(c, 0.0)

        # Compute weights
        if T > 1e-8:
            z = c / T
            z_max = np.max(z)
            w = np.exp(z - z_max)
            w_sum = np.sum(w)
            if w_sum > 0:
                w /= w_sum
            else:
                w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        else:
            c_max = np.max(c)
            active = c >= (c_max - 1e-12)
            wc = active.astype(np.float64)
            ws = np.sum(wc)
            if ws > 0:
                w = wc / ws
            else:
                w = np.full(2 * n - 1, 1.0 / (2 * n - 1))

        # Compute gradient
        g = _compute_gradient(a, w, L)
        g = np.maximum(g, 0.0)

        # Adam update
        adam_t += 1
        a, m_buf, v_buf = _adam_update(a, g, m_buf, v_buf, adam_t, lr)

    return a
