def two_peak_broaden(seq, spread, magnitude):
    """Modify the sequence to have broader peaks."""
    n_val = len(seq)
    centers = sorted(set(np.array([n//4, 3*n//4])))
    for c in centers:
        if seq[c] > 0:
            # Gaussian broadening
            form_fac = 2.0 * spread / n_val
            for i in range(n_val):
                seq[i] += magnitude * (1.0 - 0.5 * (1.0 - form_fac))
    return seq
