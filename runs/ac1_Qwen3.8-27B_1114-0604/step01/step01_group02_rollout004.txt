```python
"""
Multi-n gradient optimizer with adaptive top-k for minimizing
2*n*max(self_conv(a)) / (sum(a))^2.

Key improvements over previous:
- Work in normalized space (sum=1) for numerical stability
- Adaptive top-k that grows over time for accurate max tracking
- Multiple n values explored in parallel (time-multiplexed)
- Cosine annealing learning rate
- Better initializations (raised cosine, sine-squared)
- Smart restarts with increasing perturbation
"""

import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_obj(a):
    """Compute objective value (scale-invariant)."""
    n = a.size
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def normalize_sum(a, target=1.0):
    """Normalize to have sum = target."""
    S = a.sum()
    if S < 1e-15:
        return a * target / a.size  # uniform
    return a * target / S


def topk_softmax_weights(c, k, temperature_scale=0.05):
    """Get top-k indices and softmax weights."""
    k_act = min(k, len(c))
    if k_act == 1:
        idx = np.array([int(np.argmax(c))])
        return idx, np.array([1.0])
    # Get top k
    part = np.argpartition(c, -k_act)[-k_act:]
    vals = c[part]
    order = np.argsort(vals)[::-1]
    idx = part[order]
    vals = vals[order]
    top = float(vals[0])
    if top < 1e-15:
        wts = np.ones(k_act) / k_act
    else:
        temp = temperature_scale * top
        wts = np.exp((vals - top) / temp)
        wts = wts / wts.sum()
    return idx, wts


def compute_gradient(a, c, topk, temp_scale=0.05):
    """
    Compute gradient of 2*n*weighted_peak / S^2.
    Weighted peak is a smooth approximation to max(conv).
    """
    n = a.size
    S = a.sum()
    idx, wts = topk_softmax_weights(c, topk, temp_scale)
    S2 = S * S
    
    # Gradient of weighted sum of conv peaks
    grad_conv = np.zeros(n)
    for i in range(idx.size):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-14:
            continue
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start > j_end:
            continue
        seg = a[j_start:j_end+1]
        grad_conv[j_start:j_end+1] += wi * 2.0 * seg[::-1]
    
    M_w = float(np.dot(c[idx], wts))
    
    # d(2n*M_w/S^2)/da = 2n*(grad_conv/S^2 - 2*M_w*S_grad/S^3)
    # where S_grad = ones (dS/da[j] = 1)
    grad = (2.0 * n) * (grad_conv / S2 - 2.0 * M_w / (S2 * S))
    return grad


def adam_step(a, grad, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    """Adam update in place."""
    bc1 = b1 ** t
    bc2 = b2 ** t
    m[:] = b1 * m + (1 - b1) * grad
    v[:] = b2 * v + (1 - b2) * grad * grad
    m_hat = m / (1 - bc1)
    v_hat = v / (1 - bc2)
    a -= lr * m_hat / (np.sqrt(v_hat) + eps)


def cosine_lr(base_lr, t, total_steps):
    """Cosine annealing."""
    return base_lr * 0.5 * (1 + np.cos(np.pi * min(t, total_steps) / max(total_steps, 1)))


def clamp_nonneg(a):
    """Clip to [0, 1000] and ensure positive sum."""
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 1e-6:
        a += 1e-6 / a.size
    return a

def init_uniform(n):
    return np.ones(n) * (1.0 / n)


def init_sine_sq(n):
    x = (np.arange(n) + 0.5) / n
    a = np.sin(np.pi * x) ** 2
    return a / a.sum()


def init_cosine_window(n):
    x = (np.arange(n) + 0.5) / n
    a = 0.5 * (1 + np.cos(2 * np.pi * (x - 0.5)))
    # This is a Hann window centered
    a = np.append(a, 0)  # to make it length n+1 then trim
    a = a[:n]
    return a / a.sum()


def init_raised_cosine(n):
    x = (np.arange(n) + 0.5) / n
    # Raised cosine: high in middle, low at edges
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    return a / a.sum()


def init_bell(n):
    x = (np.arange(n) + 0.5) / n
    a = np.exp(-4 * (x - 0.5) ** 2)
    return a / a.sum()


def init_zigzag(n):
    x = (np.arange(n) + 0.5) / n
    a = 0.5 + 0.3 * np.sin(4 * np.pi * x)
    a = np.clip(a, 0.01, None)
    return a / a.sum()


def init_blocks_equal(n, nb):
    per = n // nb
    a = np.ones(n)
    for i in range(nb):
        start = i * per
        end = min(n, start + per + (1 if i < n % nb else 0))
        a[start:end] = 1.0 + 0.5 * (i % 3)
    return a / a.sum()


def init_alt(n, contrast=0.5):
    a = np.ones(n)
    a[::2] = 1.0 + contrast
    a[1::2] = 1.0 - contrast
    a = np.clip(a, 0.01, None)
    return a / a.sum()
```

```python
"""
Multi-n gradient optimizer with adaptive top-k for minimizing
2*n*max(self_conv(a)) / (sum(a))^2.

Improvements over prior approach:
- Work in normalized space (sum=1) for numerical stability
- Adaptive top-k with quadratic growth for accurate max tracking
- Multiple n values explored (1024, 2048, 4096, 8192)
- Cosine annealing learning rate with exponential floor
- Diverse initializations (sine-squared, raised cosine, bell, zigzag)
- Smart restarts with diminishing perturbation
- Phase structure: explore -> refine top-2 -> polish
"""

import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_obj(a):
    """Compute objective value (scale-invariant)."""
    n = a.size
    S = a.sum()
    if S < 1e-12:
        return np.inf
    c = fast_self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S)


def normalize_sum(a, target=1.0):
    """Normalize to have sum = target."""
    S = a.sum()
    if S < 1e-15:
        return np.ones_like(a) * target / a.size
    return a * (target / S)


def topk_weights(c, k, temp_scale=0.05):
    """Get top-k indices and softmax weights."""
    k_act = min(k, len(c))
    if k_act == 1:
        idx = np.array([int(np.argmax(c))])
        return idx, np.array([1.0])
    part = np.argpartition(c, -k_act)[-k_act:]
    vals = c[part]
    order = np.argsort(vals)[::-1]
    idx = part[order]
    vals = vals[order]
    top = float(vals[0])
    if top < 1e-15:
        wts = np.ones(k_act) / k_act
    else:
        temp = temp_scale * top
        wts = np.exp((vals - top) / temp)
        wts = wts / wts.sum()
    return idx, wts


def compute_gradient(a, c, topk, temp_scale=0.05):
    """Gradient of 2n * weighted_peak / S^2."""
    n = a.size
    S = a.sum()
    idx, wts = topk_weights(c, topk, temp_scale)
    S2 = S * S

    grad_conv = np.zeros(n)
    for i in range(idx.size):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-14:
            continue
        i_start = max(0, ki - n + 1)
        i_end = min(ki, n - 1)
        if i_start > i_end:
            continue
        seg = a[i_start:i_end + 1]
        grad_conv[i_start:i_end + 1] += wi * 2.0 * seg[::-1]

    M_w = float(np.dot(c[idx], wts))
    grad = (2.0 * n) * (grad_conv / S2 - 2.0 * M_w / (S2 * S))
    return grad


def adam_step(a, grad, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    """Adam update in place."""
    bc1 = b1 ** t
    bc2 = b2 ** t
    m[:] = b1 * m + (1 - b1) * grad
    v[:] = b2 * v + (1 - b2) * grad * grad
    m_hat = m / (1 - bc1)
    v_hat = v / (1 - bc2)
    a -= lr * m_hat / (np.sqrt(v_hat) + eps)


def cosine_lr(base_lr, t, total_steps):
    """Cosine annealing with floor."""
    if total_steps <= 0:
        return base_lr
    return base_lr * 0.5 * (1 + np.cos(np.pi * min(t, total_steps) / total_steps))


def clamp_nonneg(a):
    """Clip to [0, 1000] and ensure positive sum."""
    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 1e-6:
        a = a + 1e-6 / a.size
    return a


# --- Initializations (all return sum-normalized arrays) ---

def init_uniform(n):
    return np.ones(n) / n


def init_sine_sq(n):
    x = (np.arange(n) + 0.5) / n
    a = np.sin(np.pi * x) ** 2
    return a / a.sum()


def init_raised_cosine(n):
    x = (np.arange(n) + 0.5) / n
    a = 0.5 * (1 + np.cos(2 * np.pi * x))
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def init_bell(n):
    x = (np.arange(n) + 0.5) / n
    a = np.exp(-3.0 * (x - 0.5) ** 2)
    return a / a.sum()


def init_zigzag(n):
    x = (np.arange(n) + 0.5) / n
    a = 0.5 + 0.35 * np.sin(6.0 * np.pi * x)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def init_alt(n, contrast=0.4):
    a = np.ones(n)
    a[::2] = 1.0 + contrast
    a[1::2] = 1.0 - contrast
    a = np.maximum(a, 0.01)
    return a / a.sum()


def init_blocks(n, nb):
    per = max(1, n // nb)
    a = np.ones(n)
    for i in range(nb):
        s = i * per
        e = min(n, s + per + (1 if i < n % nb else 0))
        a[s:e] = 1.0 + 0.4 * (i % 3)
    return a / a.sum()


def init_parent(n):
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size == n:
                p = np.clip(p, 0, 1000)
                if p.sum() > 1e-6:
                    return p / p.sum()
    except Exception:
        pass
    return None


def build_inits(n, rng):
    inits = []
    p = init_parent(n)
    if p is not None:
        inits.append(("parent", p))
    inits.append(("uniform", init_uniform(n)))
    inits.append(("sine_sq", init_sine_sq(n)))
    inits.append(("raised_cos", init_raised_cosine(n)))
    inits.append(("bell", init_bell(n)))
    inits.append(("zigzag", init_zigzag(n)))
    inits.append(("alt", init_alt(n)))
    inits.append(("blocks4", init_blocks(n, 4)))
    inits.append(("blocks8", init_blocks(n, 8)))
    for i in range(4):
        a = np.abs(rng.standard_normal(n)) + 0.1
        inits.append(("rand%d" % i, a / a.sum()))
    # Additional structured inits
    x = (np.arange(n) + 0.5) / n
    a = 0.3 + 0.7 * np.cos(4 * np.pi * x) ** 2
    inits.append(("cos4sq", np.maximum(a, 0.01) / np.maximum(a, 0.01).sum()))
    a = 0.2 + 0.8 * np.sin(2 * np.pi * x) ** 4
    inits.append(("sin4", np.maximum(a, 0.01) / np.maximum(a, 0.01).sum()))
    return inits


def run_adam_optimize(a0, time_end, rng, base_lr=0.4, topk_start=3, topk_end=200,
                      temp_start=0.12, temp_end=0.02, patience=300,
                      total_est_steps=3000, tag=""):
    """Core Adam optimization loop with adaptive top-k and cosine LR."""
    n = a0.size
    a = normalize_sum(np.clip(a0.astype(np.float64), 0, 1000).copy(), 1.0)

    c = fast_self_conv(a)
    score = 2.0 * n * float(c.max())
    best_score = score
    best_a = a.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_improve = 0
    cur_patience = patience

    while time.monotonic() < time_end:
        progress = min(1.0, t / max(total_est_steps, 1))
        # Quadratic growth of topk
        topk = max(topk_start, int(topk_start + (topk_end - topk_start) * progress * progress))
        temp_scale = temp_start + (temp_end - temp_start) * progress

        c = fast_self_conv(a)
        S = a.sum()
        score = 2.0 * n * float(c.max()) / (S * S)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= cur_patience:
            scale = 0.03 + 0.06 * rng.random()
            a = best_a * (1.0 + scale * rng.standard_normal(n))
            a = clamp_nonneg(a)
            a = normalize_sum(a, 1.0)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            no_improve = 0
            cur_patience = max(60, int(cur_patience * 0.7))
            continue

        c = fast_self_conv(a)
        g = compute_gradient(a, c, topk, temp_scale)
        t += 1
        lr = max(0.0005, cosine_lr(base_lr, t, total_est_steps))
        adam_step(a, g, m, v, t, lr)
        a = clamp_nonneg(a)
        a = normalize_sum(a, 1.0)

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(self_conv)/(sum)^2.
    Multi-n exploration with adaptive-topk gradient descent.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = max(1.0, float(budget_s))
    deadline = start + total - 10.0

    # Determine n values to try
    n_candidates = [1024, 2048, 4096, 8192]

    # If parent available with different length, try it too
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size > 0 and p.size not in n_candidates:
                n_candidates.insert(0, int(p.size))
    except Exception:
        pass

    n_candidates = sorted(set(n_candidates))

    best_score = np.inf
    best_a_final = None
    best_n = None

    # ============================================================
    # Phase 1: Quick exploration across n values (45% of time)
    # ============================================================
    phase1_end = start + 0.45 * (total - 10.0)
    print("Phase 1: Exploring n=%s, budget=%.0fs" % (n_candidates, phase1_end - start))

    results_per_n = {}

    for nv in n_candidates:
        t_rem = phase1_end - time.monotonic()
        if t_rem < 3.0:
            break
        print("  Starting n=%d, remaining phase1 time=%.0fs" % (nv, t_rem))

        inits = build_inits(nv, rng)
        best_local = np.inf
        best_local_a = None
        n_inits = len(inits)
        time_per_init = max(2.0, t_rem / n_inits)

        for tag, a0 in inits:
            t_rem2 = phase1_end - time.monotonic()
            if t_rem2 < 2.0:
                break
            t_alloc = min(time_per_init, t_rem2)

            sc, aa = run_adam_optimize(
                a0, time.monotonic() + t_alloc, rng,
                base_lr=0.4, topk_start=3, topk_end=150,
                temp_start=0.12, temp_end=0.03,
                patience=200, total_est_steps=2000, tag=tag
            )
            if sc < best_local:
                best_local = sc
                best_local_a = aa
                if sc < best_score:
                    best_score = sc
                    best_a_final = aa.copy()
                    best_n = nv
                    print("    [%s, n=%d] new global best: %.6f" % (tag, nv, best_score))

        results_per_n[nv] = (best_local, best_local_a)
        print("  n=%d phase1 best: %.6f" % (nv, best_local))

    # ============================================================
    # Phase 2: Refine top-2 n values (35% of time)
    # ============================================================
    phase2_end = start + 0.80 * (total - 10.0)
    if time.monotonic() < phase2_end and len(results_per_n) > 0:
        sorted_ns = sorted(results_per_n.keys(),
                          key=lambda x: results_per_n[x][0] if np.isfinite(results_per_n[x][0]) else 100)
        top_ns = sorted_ns[:2]
        print("Phase 2: Refining n=%s" % top_ns)

        time_per_refine = (phase2_end - time.monotonic()) / max(len(top_ns), 1)

        for nv in top_ns:
            t_rem = phase2_end - time.monotonic()
            if t_rem < 5.0:
                break
            t_alloc = min(time_per_refine, t_rem)

            _, a_local = results_per_n[nv]
            if a_local is None:
                a_local = init_uniform(nv)

            # Run with refined parameters
            sc, aa = run_adam_optimize(
                a_local, time.monotonic() + t_alloc, rng,
                base_lr=0.15, topk_start=10, topk_end=300,
                temp_start=0.06, temp_end=0.015,
                patience=500, total_est_steps=8000, tag="refine"
            )
            if sc < best_score:
                best_score = sc
                best_a_final = aa.copy()
                best_n = nv
                print("  Phase2 n=%d: %.6f" % (nv, best_score))

    # ============================================================
    # Phase 3: Final polish (18% of time)
    # ============================================================
    if best_a_final is not None and time.monotonic() < deadline:
        n = best_a_final.size
        print("Phase 3: Final polish, n=%d, current best=%.6f" % (n, best_score))

        # Multiple short runs with increasing topk for ultra-fine tracking
        subsplit = 3
        sub_time = (deadline - time.monotonic()) / subsplit

        for si in range(subsplit):
            t_sub_end = time.monotonic() + sub_time
            topk_s = [100, 300, 500][si]
            topk_e = [300, 500, 800][si]
            temp_s = [0.04, 0.02, 0.01][si]
            temp_e = [0.02, 0.01, 0.005][si]
            lr_s = [0.05, 0.02, 0.008][si]

            sc, aa = run_adam_optimize(
                best_a_final, t_sub_end, rng,
                base_lr=lr_s, topk_start=topk_s, topk_end=topk_e,
                temp_start=temp_s, temp_end=temp_e,
                patience=600, total_est_steps=10000, tag="polish%d" % si
            )
            if sc < best_score:
                best_score = sc
                best_a_final = aa.copy()
                print("  Polish step %d: %.6f" % (si, best_score))

    # ============================================================
    # Final output
    # ============================================================
    if best_a_final is None:
        best_a_final = np.ones(1024)

    out_arr = np.clip(best_a_final, 0.0, 1000.0)
    if out_arr.sum() < 0.01:
        out_arr = np.ones(1024)

    # Scale to a comfortable range (sum around 1 is fine, but let's use sum=1)
    out = [float(x) for x in out_arr]
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - start
    print("FINAL: internal=%.6f, verified=%.6f, n=%d, time=%.1fs" %
          (best_score, verified, len(out), elapsed))
    return out
```
