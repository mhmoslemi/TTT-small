"""
Multi-harmonic anti-palindromic step-function optimization.
Uses soft-max gradient descent with mirror descent on the probability simplex.
"""

import numpy as np
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    n = arr.size
    s = float(arr.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    mx = float(np.max(c))
    return float(2.0 * n * mx / (s * s))

def _make_harmonic_candidate(n, amps):
    """Create anti-palindromic multi-harmonic candidate.
    amps: list of amplitudes for odd harmonics 1, 3, 5, ...
    """
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    a = np.ones(n, dtype=np.float64)
    for k_idx, A_k in enumerate(amps):
        mode = 2 * k_idx + 1  # odd harmonics: 1, 3, 5, ...
        if A_k != 0:
            a += A_k * np.sin(mode * theta)
    a /= float(n)

    s = float(a.sum())

    return a / s

def _softmax_gradient(a, tau):
    """Compute the soft-max gradient of the score with respect to a."""
    n = a.size
    c = _fft_convolve(a)
    
    # Soft-max weights
    # Use shifted values to avoid overflow
    c_max = np.max(c)
    exp_c = np.exp(tau * (c - c_max))
    w = exp_c / np.sum(exp_c)
    
    # Gradient: g_i = 2 * sum_k w[k] * a[k-i]
    # Using FFT cross-correlation
    N = _next_pow2(3 * n)
    W = np.fft.rfft(w, N)
    A = np.fft.rfft(a, N)
    cross = np.fft.irfft(np.conj(W) * A, N)
    g = 2.0 * cross[:n]
    
    return g, float(np.max(c))

def _local_optimize(a, budget_s, tau=None):
    """Mirror descent with soft-max gradient on the simplex."""
    start = time.monotonic()
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())

    while time.monotonic() - start < budget_s:
        g, maxc = _softmax_gradient(arr, tau)
        
        # Project to tangent space of simplex: subtract mean
        g = g - np.mean(g)
        
        # Mirror descent step
        exponent = -eta * g
        exponent = np.clip(exponent, -2.0, 2.0)
        new_arr = arr * np.exp(exponent)
        new_arr = np.clip(new_arr, 0.0, 1000.0)
        s = float(new_arr.sum())
        if s < 0.01:
            arr = best.copy()
            eta *= 0.5
            continue
        new_arr = new_arr / s
        
        new_score = _internal_score(new_arr)

            best = new_arr.copy()
            best_score = new_score
            arr = new_arr

Multi-harmonic anti-palindromic step-function optimization using

import numpy as np

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    n = arr.size
    if n == 0:
        return np.inf
    s = float(arr.sum())
