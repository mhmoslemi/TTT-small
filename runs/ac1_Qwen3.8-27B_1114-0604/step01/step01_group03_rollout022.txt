def _normalize(a):
    """Normalize to simplex: sum=1, all non-negative."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / len(a)
    return a / s

def _fft_len(n):
    """Smallest power of 2 >= 2n-1."""
    m = 2 * n - 1
    L = 1
    while L < m:
        L <<= 1
    return L

def _convolve_full(a, L):
    """Full linear convolution of a with itself via FFT."""
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:2*len(a)-1]
    return np.maximum(c, 0.0)

def _compute_gradient(a, c, T, L, n):
    """Compute the log-sum-exp smoothed gradient."""
    m = 2 * n - 1
    # Softmax for weights
    vc = c  # c = convolve(a,a), already non-negative
    Tc = T * vc
    Tc_max = Tc.max()
    w = np.exp(Tc - Tc_max)
    sw = w.sum()
    if sw < 1e-300:
        w = np.ones(m) / m
    else:
        w /= sw
    
    # Gradient: grad[j] = 2 * sum_k w[k] * a[k-j] (valid indices)
    # = 2 * (w * a_rev)[n-1+j]
    a_rev = a[::-1]
    W = np.fft.rfft(w, L)
    Ar = np.fft.rfft(a_rev, L)
    C = np.fft.irfft(W * Ar, L)
    grad = 2.0 * C[n-1:n-1+n]
    np.maximum(grad, 0.0, out=grad)  # gradient should be non-negative
    return grad

def _mirror_descent(a, n, L, time_limit, T_start=50.0, T_end=1.0, 
                    eta_start=0.5, beta=0.9, perturb_every=500, 
                    perturb_frac=0.05, rng=None):
    """Mirror descent with log-sum-exp gradient and momentum."""
    if rng is None:
        rng = np.random.default_rng()
    
    m = 2 * n - 1
    a = _normalize(a.copy())
    v = np.zeros(n)  # momentum
    
    c = _convolve_full(a, L)
    current_obj = c.max()
    best_a = a.copy()
    best_obj = current_obj
    
    eta = eta_start
    T = T_start
    deadline = time.monotonic() + time_limit
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        it += 1
        
        # Compute gradient
        Tc = T * c
        Tc_max = Tc.max()
        w = np.exp(Tc - Tc_max)
        sw = w.sum()
        if sw < 1e-300:
            w = np.ones(m) / m
        else:
            w /= sw
        
        a_rev = a[::-1]
        W = np.fft.rfft(w, L)
        Ar = np.fft.rfft(a_rev, L)
        C = np.fft.irfft(W * Ar, L)
        grad = 2.0 * C[n-1:n-1+n]
        np.maximum(grad, 0.0, out=grad)
        
        # Momentum update
        v = beta * v + (1.0 - beta) * grad
        
        # Multiplicative update
        # Scale gradient so that the update is reasonable
        gnorm = grad.mean()
        if gnorm < 1e-300:
            gnorm = 1.0
        step = eta / gnorm
        
        a_new = a * np.exp(-step * v)
        a_new = _normalize(a_new)
        
        # Evaluate
        c_new = _convolve_full(a_new, L)
        obj_new = c_new.max()
        
        if obj_new < current_obj:
            a = a_new
            c = c_new
            current_obj = obj_new
            eta = min(2.0, eta * 1.05)
            if obj_new < best_obj - 1e-12:
                best_obj = obj_new
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            eta = max(0.01, eta * 0.7)
            # Revert
            a_new = a
            c_new = c
            no_improve += 1
        
        # Temperature annealing
        frac = min(1.0, it / max(1, int(time_limit * 200)))
        T = T_start * (T_end / T_start) ** frac
        
        # Perturbation
        if it % perturb_every == 0 and no_improve > perturb_every // 2:
            # Mix with random direction
            t = perturb_frac * (0.5 + 0.5 * rng.random())
            random_a = _normalize(rng.dirichlet(np.ones(n) * 0.5))
            a = _normalize((1 - t) * a + t * random_a)
            c = _convolve_full(a, L)
            current_obj = c.max()
            v *= 0.5  # Damp momentum after perturbation
            no_improve = 0
        
        # Check time
        if it % 1000 == 0:
            elapsed = time.monotonic() - (deadline - time_limit)
            print(f"  Iter {it}, obj={2*n*current_obj:.6f}, T={T:.2f}, eta={eta:.4f}")
    
    return best_a, 2.0 * n * best_obj

    except ImportError:
        return a, 2.0 * n * _convolve_full(a, L).max()

        Tc = T_small * c
        vmax = Tc.max()
        return float(vmax / T_small + math.log(np.exp(Tc - vmax).sum()) / T_small)

        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-10:
            return np.zeros_like(x)
        x = x / s
        c = _convolve_full(x, L)
        m = 2 * n - 1
        Tc = T_small * c
        Tc_max = Tc.max()
        w = np.exp(Tc - Tc_max)
        w /= w.sum()
        a_rev = x[::-1]
        W = np.fft.rfft(w, L)
        Ar = np.fft.rfft(a_rev, L)
        C = np.fft.irfft(W * Ar, L)

def _random_wave(n, rng):
    """Generate a random wave-like sequence."""
    t = np.linspace(0, 1, n)
    z = np.zeros(n)
    n_harmonics = rng.integers(2, 8)
    for _ in range(n_harmonics):
        k = rng.integers(1, 10)
        amp = rng.uniform(0.3, 1.0) / math.sqrt(k)
        phase = rng.uniform(0, 2 * np.pi)
        z += amp * np.cos(2 * np.pi * k * t + phase)
    z += rng.normal(0, 0.3, n)
    z = 2.0 * t - 1.0  # Add a linear component
    z += rng.normal(0, 0.2, n) * (1 - t)
    p = np.exp(z - z.max())
    p = np.maximum(p, 1e-300)
    return _normalize(p)

def _random_blocks(n, rng, n_blocks=(5, 20)):

        end = min(n, start + width)
        a[start:end] += amp
    if a.max() < 1e-10:
        a = np.ones(n) / n
    return _normalize(a)

def _normalize(a):
    """Project onto probability simplex: non-negative, sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / len(a)
    return a / s


def _fft_len(n):
    """Smallest power of 2 >= 2n-1."""
    m = 2 * n - 1
    L = 1
    while L < m:
        L <<= 1
    return L


def _conv_self(a, L):
    """Full linear self-convolution via FFT, non-negative."""
    A = np.fft.rfft(a, L)
    c = np.fft.irfft(A * A, L)[:2 * len(a) - 1]
    return np.maximum(c, 0.0)


def _lsx_grad(a, c, T, L, n):
    """Compute log-sum-exp smoothed gradient of max(convolve(a,a)).

    Tc_max = Tc.max()
    w = np.exp(Tc - Tc_max)
    sw = w.sum()
    if sw < 1e-300:

        lse = c.max() / 1.0

                z = rng.normal(0, 0.5, n)
                for k in range(1, 4):
                    z += rng.uniform(0.2, 0.8) * np.cos(2 * np.pi * k * t + rng.uniform(0, 2 * np.pi))
                z += rng.uniform(-0.5, 0.5) * (2 * t - 1)
                z = np.exp(z - z.max())
                z = np.maximum(z, 1e-300)
                z /= z.sum()
            else:
                # Sparse perturbation
                z = np.ones(n) / n
                n_spikes = rng.integers(3, 10)
                for _ in range(n_spikes):
                    pos = rng.integers(0, n)
                    width = rng.integers(1, n // 10 + 1)
                    z[pos:pos + width] += rng.uniform(0.5, 2.0)
                z = _normalize(z)
            
            a = _normalize((1 - t_p) * best_a + t_p * z)
            c = _conv_self(a, L)
            current_obj = c.max()
            v *= 0.3
            no_improve = 0

        # Stagnation reset: if stuck for a long time, hard restart towards random
        if no_improve > perturb_every * 3:
            t_reset = rng.uniform(0.1, 0.3)
            a_rand = _normalize(rng.dirichlet(np.full(n, 0.3)))
            a = _normalize((1 - t_reset) * best_a + t_reset * a_rand)
            c = _conv_self(a, L)
            current_obj = c.max()
            v = np.zeros(n)
            eta = eta_start
            no_improve = 0

        if verbose and it % 2000 == 0:
            elapsed = time.monotonic() - t0
            print(f"    MD iter {it}, elapsed {elapsed:.0f}s, "
                  f"obj={2*n*current_obj:.6f}, best={2*n*best_obj:.6f}, "
                  f"T={T:.1f}, eta={eta:.4f}")

    return best_a, 2.0 * n * best_obj


def _coord_descent(a, n, L, time_limit, rng, verbose=False):
    """Coordinate descent: optimize one coordinate at a time."""
    m = 2 * n - 1
    a = _normalize(a.copy())
    c = _conv_self(a, L)
    current_obj = c.max()
    best_a = a.copy()
    best_obj = current_obj

    deadline = time.monotonic() + time_limit
    t0 = time.monotonic()
    pass_count = 0
    order = np.arange(n)

    while time.monotonic() < deadline:
        pass_count += 1
        rng.shuffle(order)

        for j in order:
            if time.monotonic() > deadline:
                break

            old_x = a[j]
            if old_x < 1e-15:
                # Try a small positive value
                trial_vals = [old_x + 1e-6, 2e-6, 5e-6, 0.01 / n]
            else:
                # Try several multipliers
                trial_vals = [0.0, old_x * 0.25, old_x * 0.5, old_x * 0.75,
                              old_x, old_x * 1.33, old_x * 1.75, old_x * 2.5]

            best_trial_x = old_x
            best_trial_obj = current_obj

            # Compute slopes: for lag k != 2j, c_new[k] = c[k] + 2*(x-old_x)*a[k-j]
            # For lag k = 2j: c_new[2j] = c[2j] - old_x^2 + x^2
            k_j = (n - 1 - j)  # offset for "a[k-j]" -> reversed index
            # Actually: for lag k in [0, 2n-2], the contribution to c[k] from a[j]
            # is: 2*x*a[k-j] if 0<=k-j<n and k!=2j, or x^2 if k=2j.
            # The "slope" at lag k is 2*a[k-j] (for k != 2j).
            # Let's precompute the slopes.
            # slope[k] = 2*a[k-j] if 0 <= k-j < n and k != 2*j, else 0
            # quad_flag at k=2*j
            
            two_j = 2 * j
            # slope vector: slope[k] for k in 0..2n-2
            # a[k-j] is valid when 0 <= k-j < n, i.e., j <= k <= j+n-1
            slopes = np.zeros(m)
            lo = j
            hi = min(j + n - 1, m - 1)
            if lo <= hi:
                # a[k-j] for k in [lo, hi] is a[0], a[1], ..., a[hi-lo]
                slopes[lo:hi + 1] = 2.0 * a[0:hi - lo + 1]
            # Zero out the quad position
            if 0 <= two_j < m:
                slopes[two_j] = 0.0

            for x in trial_vals:
                if x < 0 or x > 1.0:
                    continue
                dx = x - old_x
                if abs(dx) < 1e-15:
                    trial_obj = current_obj
                else:
                    # c_new[k] = c[k] + dx * slopes[k] + (x^2 - old_x^2) if k == 2j
                    c_new = c + dx * slopes
                    if 0 <= two_j < m:
                        c_new[two_j] += x * x - old_x * old_x
                    trial_obj = c_new.max()
                
                if trial_obj < best_trial_obj - 1e-15:
                    best_trial_obj = trial_obj
                    best_trial_x = x

            if best_trial_x != old_x:
                a[j] = best_trial_x
                # Renormalize
                a = _normalize(a)
                c = _conv_self(a, L)
                current_obj = c.max()
                if current_obj < best_obj - 1e-14:
                    best_obj = current_obj
                    best_a = a.copy()

        if verbose and pass_count % 20 == 0:
            elapsed = time.monotonic() - t0
            print(f"    CD pass {pass_count}, elapsed {elapsed:.0f}s, "
                  f"obj={2*n*current_obj:.6f}, best={2*n*best_obj:.6f}")

    return best_a, 2.0 * n * best_obj


def _scipy_polish(a, n, L, time_limit, T_small=15.0):
    """Polish using scipy L-BFGS-B on smooth log-sum-exp objective."""
    try:
        from scipy.optimize import minimize
    except ImportError:
        return a, 2.0 * n * _conv_self(a, L).max()

    a0 = _normalize(a.copy())
    m = 2 * n - 1

    def objective(x):
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-30:
            return 1e10
        x = x / s
        c = _conv_self(x, L)
        Tc = T_small * c
        vmax = Tc.max()
        return float(vmax / T_small + math.log(np.exp(Tc - vmax).sum()) / T_small)

    def gradient(x):
        x = np.maximum(x, 0.0)
        s = x.sum()
        if s < 1e-30:
            return np.zeros_like(x)
        x = x / s
        c = _conv_self(x, L)
        Tc = T_small * c
        Tc_max = Tc.max()
        w = np.exp(Tc - Tc_max)
        w /= w.sum()
        a_rev = x[::-1]
        W = np.fft.rfft(w, L)
        Ar = np.fft.rfft(a_rev, L)
        C = np.fft.irfft(W * Ar, L)
        grad = 2.0 * C[n - 1:n - 1 + n]
        np.maximum(grad, 0.0, out=grad)
        return grad

    try:
        result = minimize(objective, a0, method='L-BFGS-B',
                          jac=gradient,
                          options={'maxiter': 3000, 'maxfun': 6000,
                                   'maxls': 40, 'gtol': 1e-12})
        a_pol = _normalize(result.x)
        obj = 2.0 * n * _conv_self(a_pol, L).max()
        return a_pol, obj
    except Exception as e:
        print(f"    scipy error: {e}")
        return a, 2.0 * n * _conv_self(a, L).max()


def _random_wave(n, rng):
    """Generate random wave-like probability sequence."""
    t = np.linspace(0, 1, n)
    z = np.zeros(n)
    n_h = rng.integers(2, 7)
    for _ in range(n_h):
        k = rng.integers(1, 12)
        amp = rng.uniform(0.3, 1.0) / math.sqrt(k)
        ph = rng.uniform(0, 2 * np.pi)
        z += amp * np.cos(2 * np.pi * k * t + ph)
    z += rng.uniform(-0.8, 0.8) * (2 * t - 1)
    z += rng.uniform(-0.5, 0.5) * (4 * t * t - 2)
    z += rng.normal(0, 0.2, n)
    p = np.exp(z - z.max())
    p = np.maximum(p, 1e-300)
    return _normalize(p)


def _random_blocks(n, rng):
    """Generate block-structured probability sequence."""
    a = np.zeros(n)
    nb = rng.integers(5, 25)
    for _ in range(nb):
        start = rng.integers(0, n)
        width = rng.integers(1, n // 8)
        amp = rng.uniform(0.1, 1.5)
        end = min(n, start + width)
        a[start:end] += amp
    if a.max() < 1e-10:
        a = np.ones(n) / n
    return _normalize(a)


def _initial_candidates(n, rng, n_extra=16):
    """Generate diverse initial candidates."""
    cands = []

    # Parent
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if parent.size == n and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0.0)
            if parent.sum() > 1e-10:
                cands.append(_normalize(parent))
    except (NameError, Exception):
        pass

    # Uniform
    cands.append(np.ones(n) / n)

    # Triangular (peaked in middle)
    t = np.linspace(0, 1, n)
    cands.append(_normalize(1.0 - np.abs(2 * t - 1)))

    # Two bumps
    cands.append(_normalize(np.exp(-10 * (t - 0.3) ** 2) + np.exp(-10 * (t - 0.7) ** 2)))

    # Random waves
    for _ in range(n_extra // 2):
        cands.append(_random_wave(n, rng))

    # Random blocks
    for _ in range(n_extra // 2):
        cands.append(_random_blocks(n, rng))

    return cands


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function. Returns best sequence found."""
    global L_global  # For potential reuse
    
    start_time = time.monotonic()
    hard_deadline = start_time + max(60.0, budget_s - 3.0)

    try:
        n = 6121
    except Exception:
        n = 6121

    L = _fft_len(n)
    m = 2 * n - 1
    L_global = L

    rng = np.random.default_rng(seed)

    print(f"n={n}, L={L}, m={m}, budget={budget_s}s")
    print(f"Theoretical lower bound: {2*n/(2*n-1):.6f}")

    # === Phase 1: Initialization ===
    print("Phase 1: Initialization")
    t0 = time.monotonic()
    candidates = _initial_candidates(n, rng, n_extra=16)

    best_a = None
    best_obj = float('inf')
    cand_list = []

    for i, a in enumerate(candidates):
        c = _conv_self(a, L)
        obj = 2.0 * n * c.max()
        cand_list.append((a, obj))
        if obj < best_obj:
            best_obj = obj
            best_a = a.copy()

    cand_list.sort(key=lambda x: x[1])
    print(f"  Best initial: {best_obj:.6f}, {len(candidates)} candidates, "
          f"time={time.monotonic()-t0:.1f}s")

    # === Phase 2: Mirror Descent ===
    print("Phase 2: Mirror Descent")
    t0 = time.monotonic()

    remaining = hard_deadline - time.monotonic()
    # Allocate: 60% to MD, 25% to CD, 12% to scipy, 3% buffer
    md_time = 0.58 * remaining
    cd_time = 0.25 * remaining
    sp_time = 0.10 * remaining

    # Multiple MD runs with different params
    md_configs = [
        # (T_start, T_end, eta_start, beta, perturb_every, perturb_frac)
        (50, 1, 0.5, 0.90, 400, 0.08),
        (100, 0.5, 0.3, 0.95, 300, 0.10),
        (30, 1, 0.7, 0.85, 500, 0.06),
        (80, 1, 0.4, 0.92, 350, 0.12),
        (200, 0.5, 0.2, 0.95, 250, 0.15),
        (40, 1, 0.6, 0.88, 450, 0.07),
    ]

    n_runs = len(md_configs)
    time_per_run = md_time / n_runs

    all_results = [(best_a.copy(), best_obj)]

    for run in range(n_runs):
        if time.monotonic() > hard_deadline - (cd_time + sp_time + 30):
            break

        t_avail = hard_deadline - time.monotonic() - (cd_time + sp_time + 20)
        t_run = min(time_per_run, t_avail)
        if t_run < 10:
            break

        # Starting point
        if run == 0:
            a_start = best_a.copy()
        elif run == 1:
            # Second best candidate
            if len(cand_list) > 1:
                a_start = cand_list[1][0].copy()
            else:
                a_start = best_a.copy()
        else:
            # Perturb best
            t_p = rng.uniform(0.03, 0.20)
            a_rand = _random_wave(n, rng) if rng.random() < 0.5 else _random_blocks(n, rng)
            a_start = _normalize((1 - t_p) * best_a + t_p * a_rand)

        cfg = md_configs[run]
        print(f"  MD run {run}: T={cfg[0]}->{cfg[1]}, eta={cfg[2]}, "
              f"beta={cfg[3]}, time={t_run:.0f}s")

        a_best, obj = _mirror_descent(a_start, n, L, t_run,
                                      T_start=cfg[0], T_end=cfg[1],
                                      eta_start=cfg[2], beta=cfg[3],
                                      perturb_every=cfg[4],
                                      perturb_frac=cfg[5],
                                      rng=rng,
                                      verbose=True)
        all_results.append((a_best, obj))
        if obj < best_obj:
            best_obj = obj
            best_a = a_best.copy()
            print(f"    *** NEW BEST: {obj:.6f} ***")
        else:
            print(f"    Best so far: {obj:.6f} (global: {best_obj:.6f})")

    print(f"Phase 2 done. Global best: {best_obj:.6f}, "
          f"time={time.monotonic()-t0:.0f}s")

    # === Phase 3: Coordinate Descent ===
    print("Phase 3: Coordinate Descent")
    t0 = time.monotonic()

    remaining = hard_deadline - time.monotonic()
    cd_actual = min(cd_time, 0.7 * remaining)
    sp_actual = min(sp_time, 0.25 * remaining)

    all_results.sort(key=lambda x: x[1])

    # Run CD on top candidates
    n_cd = min(3, len(all_results))
    cd_time_each = cd_actual / max(1, n_cd)

    for i in range(n_cd):
        if time.monotonic() > hard_deadline - (sp_actual + 20):
            break
        t_cd = min(cd_time_each, hard_deadline - time.monotonic() - (sp_actual + 15))
        if t_cd < 5:
            break

        a_c = all_results[i][0]
        print(f"  CD on candidate {i} (obj={all_results[i][1]:.6f}), time={t_cd:.0f}s")
        a_cd, obj_cd = _coord_descent(a_c, n, L, t_cd, rng, verbose=True)
        all_results.append((a_cd, obj_cd))
        if obj_cd < best_obj:
            best_obj = obj_cd
            best_a = a_cd.copy()
            print(f"    *** NEW BEST: {obj_cd:.6f} ***")
        else:
            print(f"    Result: {obj_cd:.6f}")

    print(f"Phase 3 done. Global best: {best_obj:.6f}, "
          f"time={time.monotonic()-t0:.0f}s")

    # === Phase 4: Scipy Polish ===
    print("Phase 4: Scipy Polish")
    t0 = time.monotonic()

    all_results.sort(key=lambda x: x[1])
    n_sp = min(3, len(all_results))
    sp_each = sp_actual / max(1, n_sp)

    for i in range(n_sp):
        if time.monotonic() > hard_deadline - 15:
            break
        t_sp = min(sp_each, hard_deadline - time.monotonic() - 10)
        if t_sp < 3:
            break

        a_c = all_results[i][0]
        print(f"  Scipy on candidate {i} (obj={all_results[i][1]:.6f}), time={t_sp:.0f}s")
        a_sp, obj_sp = _scipy_polish(a_c, n, L, t_sp, T_small=20.0)
        all_results.append((a_sp, obj_sp))
        if obj_sp < best_obj:
            best_obj = obj_sp
            best_a = a_sp.copy()
            print(f"    *** NEW BEST: {obj_sp:.6f} ***")
        else:
            print(f"    Result: {obj_sp:.6f}")

    print(f"Phase 4 done. Global best: {best_obj:.6f}, "
          f"time={time.monotonic()-t0:.0f}s")

    # === Final: Exact evaluation ===
    print("Final evaluation")
    all_results.sort(key=lambda x: x[1])

    best_seq = None
    best_exact = float('inf')
    seen = set()

    for a_r, obj_r in all_results[:10]:
        if time.monotonic() > hard_deadline - 3:
            break
        key = tuple(np.round(a_r, 10).tolist())
        if key in seen:
            continue
        seen.add(key)
        seq = [float(x) for x in a_r]
        try:
            val = evaluate_sequence(seq)
        except Exception:
            val = obj_r
        if np.isfinite(val) and val < best_exact:
            best_exact = val
            best_seq = seq
        if len(seen) >= 5:
            break

    if best_seq is None:
        best_seq = [float(x) for x in best_a]
        best_exact = best_obj

    print(f"Final exact value: {best_exact:.6f}")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    # Ensure valid output
    result = [float(max(0.0, min(1000.0, x))) for x in best_seq]
    if sum(result) < 0.01:
        result = [1.0] * n

    return result
