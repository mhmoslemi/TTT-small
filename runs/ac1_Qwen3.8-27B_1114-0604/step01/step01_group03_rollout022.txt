"""
Optimization of non-negative step function heights to minimize
2*n*max(conv(a,a)) / (sum(a))^2.
Uses active-set subgradient descent with FFT convolution,
adaptive step size, periodic restarts, and multi-length search.

import numpy as np

def _fft_conv_self(a, L):
    """Compute full self-convolution via FFT, zeroing negative values."""
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)
    np.maximum(c, 0.0, out=c)
    return c


def _active_set(c, top_k=30, threshold_frac=0.92):
    """Find indices of near-maximum convolution coefficients."""
    cmax = float(np.max(c))
    if cmax <= 0:
        return np.array([c.size // 2], dtype=np.int64)
    mask = c >= threshold_frac * cmax
    idx = np.where(mask)[0]
    if len(idx) > top_k:
        vals = c[idx]
        top = np.argsort(vals)[-top_k:]
        idx = idx[top]
    return idx


def _subgrad(a, active_idx, n):
    """Compute subgradient of max_k c_k at the active set."""
    g = np.zeros(n)
    for k in active_idx:
        lo = max(0, int(k) - n + 1)
        hi = min(n - 1, int(k))
        if lo > hi:
            continue
        length = hi - lo + 1
        g[lo:hi + 1] += 2.0 * a[k - hi:k - lo + 1][::-1]
    return g


def _score_from_conv(c, n):
    return float(2.0 * n * np.max(c[:2 * n - 1]))

def _soft_gradient_optimize(a_init, n, time_budget, rng, temp_start=5.0, temp_end=200.0):
    """Soft-gradient optimization as a polishing step."""
    L = 1 << (2 * n - 2).bit_length()
    m = 2 * n - 1
    a = np.asarray(a_init, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / a.sum()  # normalize to sum=1

    best_a = a.copy()
    best_score = float(np.inf)

    deadline = time.monotonic() + time_budget
    temp = temp_start
    eta = 0.3

    while time.monotonic() < deadline:
        c = _fft_conv_self(a, L)
        c = c[:m]
        cur_score = _score_from_conv(c, n)

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_a = a.copy()

        # Soft max gradient
        v = c  # already the convolution
        vmax = float(np.max(v))
        z = temp * (v - vmax)
        z = np.clip(z, -50.0, 0.0)
        e = np.exp(z)
        s = float(e.sum())
        if s <= 0 or not np.isfinite(s):
            e = np.ones_like(v)
            s = float(e.sum())
        w = e / s

        # Gradient: g[j] = 2 * sum_k w[k] * a[k-j]
        # = 2 * cross_correlation(w, a) at lag j
        # Use FFT
        L2 = 1 << (n + m - 2).bit_length()
        Wf = np.fft.rfft(w, L2)
        Af = np.fft.rfft(a, L2)
        # cross corr: ifft(conj(fft(a)) * fft(w)) gives sum a[l]*w[l+j]
        corr = np.fft.irfft(np.conj(Af) * Wf, L2)
        g = 2.0 * corr[:n]
        g = np.maximum(g, 0.0)

        gmax = float(np.max(g))
        if gmax <= 1e-30:
            break
        g_norm = g / gmax

        # Update
        a_try = a * np.exp(-eta * g_norm)
        a_try = np.maximum(a_try, 1e-15)
        a_try = a_try / a_try.sum()

        c_try = _fft_conv_self(a_try, L)[:m]
        sc_try = _score_from_conv(c_try, n)

            a = a_try

    t_remaining = time_budget - (time_budget - t_alloc)
    if t_remaining > 5:
        a_pol, sc_pol = _soft_gradient_optimize(a_opt, n, t_remaining, rng)
        if sc_pol < sc:
            a_opt = a_pol
            sc = sc_pol

    return a_opt, sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main entry point: search for the best sequence."""
    start = time.monotonic()
    try:
        total_budget = float(budget_s)
    except Exception:
        total_budget = 2400.0
    if not np.isfinite(total_budget) or total_budget < 10:
        total_budget = 2400.0

    hard_deadline = start + max(10.0, total_budget) - 10.0
    rng = np.random.default_rng(seed)

    # Get parent sequence
    parent = None
    n_parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        n_parent = len(parent)
        if not np.all(np.isfinite(parent)) or n_parent < 10:
            parent = None
            n_parent = None
    except NameError:
        pass
    except Exception:
        pass

    best_overall_a = None
    best_overall_n = None
    best_score = float(np.inf)

    # Phase 1: Optimize parent if available
    if parent is not None and n_parent is not None:
        t_alloc = min(total_budget * 0.35, 600.0)
        print(f"Phase 1: Optimizing parent, n={n_parent}, budget={t_alloc:.0f}s")
        a_opt, sc = _optimize_fixed_n(parent, n_parent, t_alloc, rng)
        print(f"  Parent result: score={sc:.8f}")
        if sc < best_score:
            best_score = sc
            best_overall_a = a_opt
            best_overall_n = n_parent

    # Phase 2: Try other sequence lengths
    remaining = hard_deadline - time.monotonic()
    if remaining > 60:
        n_list = []
        if n_parent is not None:
            n_list.append(n_parent)
        n_list.extend([3072, 4096, 5120, 6144, 8192, 10240, 12288, 16384, 20480])
        # Remove duplicates, keep order
        seen = set()
        n_list_unique = []
        for x in n_list:
            if x not in seen and x >= 128:
                seen.add(x)
                n_list_unique.append(x)
        n_list = n_list_unique

        # Sort: prioritize n values near parent, then by size
        if n_parent:
            n_list.sort(key=lambda x: (abs(x - n_parent), x))
        else:
            n_list.sort()

        for n in n_list:
            if time.monotonic() > hard_deadline - 30:
                break
            if n == n_parent:
                continue  # Already tried

            # Estimate if this n is feasible
            L = 1 << (2 * n - 2).bit_length()
            # Each iteration ~ L*log(L) for FFT + K*n for subgrad
            est_time_per_iter = L * math.log2(L) * 1e-8 + 30 * n * 1e-8
            est_max_iters = max(100, int((hard_deadline - time.monotonic() - 30) / max(est_time_per_iter, 1e-6)))
            if est_max_iters < 200:
                print(f"  Skipping n={n}: estimated {est_max_iters} iters (too few)")
                continue

            # Allocate time for this n
            t_alloc = min(remaining * 0.2, max(30.0, 180.0 * math.sqrt(6000.0 / n)))
            t_alloc = min(t_alloc, (hard_deadline - time.monotonic() - 30))
            if t_alloc < 15:
                continue

            print(f"Phase 2: Trying n={n}, budget={t_alloc:.0f}s")
            a_opt, sc = _try_length(n, t_alloc, rng)
            print(f"  n={n}: score={sc:.8f}")
            if sc < best_score:
                best_score = sc
                best_overall_a = a_opt
                best_overall_n = n

    # Phase 3: Final polish on the best found
    remaining = hard_deadline - time.monotonic()
    if best_overall_a is not None and remaining > 20:
        print(f"Phase 3: Polishing best (n={best_overall_n}), budget={remaining:.0f}s")
        t_polish = min(remaining - 10, 120.0)
        a_pol, sc_pol = _soft_gradient_optimize(best_overall_a, best_overall_n, t_polish, rng)
        if sc_pol < best_score:
            best_score = sc_pol
            best_overall_a = a_pol
            print(f"  Polish result: score={sc_pol:.8f}")

        # One more round of subgradient
        remaining2 = hard_deadline - time.monotonic()
        if remaining2 > 20:
            a_fin, sc_fin = _optimize_fixed_n(best_overall_a, best_overall_n, min(remaining2 - 10, 90.0), rng)
            if sc_fin < best_score:
                best_score = sc_fin
                best_overall_a = a_fin
                print(f"  Final subgrad result: score={sc_fin:.8f}")

    # Construct final answer
    if best_overall_a is not None:
        # Scale to have sum=1 for cleanliness (scale doesn't matter for objective)
        seq = [float(max(0.0, min(1000.0, x))) for x in best_overall_a]
        # Ensure sum >= 0.01
        s = sum(seq)
        if s < 0.01:
            seq = [1.0] * best_overall_n
    else:
        # Fallback
        n_fb = n_parent if n_parent else 2048
        seq = [1.0] * n_fb

    # Final exact evaluation
    try:
        val = evaluate_sequence(seq)
        print(f"FINAL exact score: {val:.8f} (n={len(seq)})")
    except Exception as e:
        print(f"Error in final eval: {e}")
        val = float('inf')

    return seq
