def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w

            z += rng.standard_normal() * np.sin(2 * np.pi * m * u + rng.standard_normal())
        z *= 0.5
        starts.append(z)

    # Bimodal
    for A in [2.0, 4.0, 6.0]:
        starts.append(A * (np.cos(2 * np.pi * x) + 1))

    return starts


def _fista_gradient(func, b0, max_iter, time_limit, lr_init=0.1, verbose_every=500):
    """FISTA for smooth minimization. func(b) -> (val, grad)."""
    n = b0.size
    b = b0.copy()
    y = b0.copy()
    t = 1.0
    lr = lr_init
    best_val = float('inf')
    best_b = b0.copy()

    start = time.monotonic()

    for it in range(max_iter):
        if time.monotonic() > time_limit:
            break

        val_y, grad_y = func(y)
        if not np.isfinite(val_y):
            break

        # Backtracking line search
        lr_bt = lr
        for _ in range(30):
            b_new = y - lr_bt * grad_y
            val_new, _ = func(b_new)
            if np.isfinite(val_new) and val_new <= val_y - 0.5 * lr_bt * np.dot(grad_y, grad_y):
                break
            lr_bt *= 0.5
        else:
            lr_bt = lr * 0.5

        # Nesterov extrapolation
        alpha = (1.0 + math.sqrt(1.0 + 4.0 * t * t)) / 2.0
        theta = (t - 1.0) / alpha
        y = b_new + theta * (b_new - b)
        b = b_new
        t = alpha
        lr = lr_bt

        # Track best
        if val_new < best_val:
            best_val = val_new
            best_b = b.copy()

        if verbose_every and it % verbose_every == 0:
            print(f"    FISTA it={it} val={val_y:.8f} best={best_val:.8f} lr={lr:.6f} t={time.monotonic()-start:.1f}s")

    return best_b, best_val


def _coordinate_descent(a, time_limit, n_trials=10000):
    """Try to improve a by adjusting individual elements."""
    n = a.size
    a = a.copy()
    best_score = _score(a)
    best_a = a.copy()
    rng = np.random.default_rng(12345)
    start = time.monotonic()

    while time.monotonic() < time_limit:
        # Pick a random index
        i = rng.integers(0, n)
        # Try several values for a[i]
        for delta in [1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 0.1, 0.3]:
            for sign in [1, -1]:
                a_new = a.copy()
                a_new[i] = max(0, a[i] + sign * delta * a[i])
                s = a_new.sum()
                if s < 1e-10:
                    continue
                a_new /= s
                sc = _score(a_new)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = a_new.copy()
                    a = a_new
                    break
        if (n_trials > 0) and rng.integers(0, n) == 0:
            break

    return best_a, best_score

Phase 1: Flatten auto-convolution via FISTA on variance objective.
Phase 2: Reduce peak via L-BFGS-B on annealed log-sum-exp.
Phase 3: Polish via coordinate descent.
Log-space softmax parameterization ensures non-negativity.

_N = 8192
_N_FFT = 32768


def _conv(a):
    fa = np.fft.rfft(a, _N_FFT)
    c = np.fft.irfft(fa * fa, _N_FFT)
    return c[:2 * a.size - 1]


def _softmax(z):
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s <= 0.0 or not np.isfinite(s):
        return np.full(z.shape, 1.0 / z.size)
    return e / s


def _score(a):
    n = a.size
    c = _conv(a)
    return float(2.0 * n * np.max(c))


def _cross_corr(p, a):
    fp = np.fft.rfft(p, _N_FFT)
    fa = np.fft.rfft(a, _N_FFT)
    h = np.fft.irfft(fp * np.conjugate(fa), _N_FFT)
    return h[:a.size]

    g_a = 4.0 * h
    g_b = a * (g_a - np.dot(a, g_a))
