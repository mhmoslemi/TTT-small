"""

import numpy as np

def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b):
    la = len(a)
    lb = len(b)
    nout = la + lb - 1
    if la * lb <= 1500000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)


def _corr_valid(w, p):
    """Return v[j] = sum_i p[i] * w[j+i] for j=0..len(p)-1 (valid cross-correlation)."""
    lenw = len(w)
    lenp = len(p)
    if lenp == 0:
        return np.array([], dtype=float)
    L = _next_pow2(max(lenw, 2 * lenp))
    fw = np.fft.rfft(w, L)
    fp = np.fft.rfft(p, L)
    out = np.fft.irfft(fw * np.conjugate(fp), L)
    return out[:lenp].astype(float, copy=False)


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return p / s

        return float('inf')
    m = p.size
    c = _conv_full(p, p)

def _soft_weights(d, beta):
    d = np.asarray(d, dtype=float)
    n = d.size
    if n == 0:
        return np.array([], dtype=float)
    d = np.where(np.isfinite(d), d, 0.0)
    if beta <= 1e-8:
        return np.full(n, 1.0 / n, dtype=float)
    shift = float(np.max(d))
    e = np.exp(beta * (d - shift))
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n, dtype=float)
    return e / s


def _gradient_info(p, beta):

    p = _sanitize(p)

    exact = float(2.0 * np.max(d)) if d.size else float('inf')
    w = _soft_weights(d, beta)
    h = _corr_valid(w, p)
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.maximum(h, 0.0)
    maxh = float(np.max(h)) if h.size else 0.0
    smooth = float(np.dot(w, d))
    return exact, smooth, h, maxh

def _run_lag_focus(p0, rng, time_limit, n_lags=5):
    """Optimize by directly targeting the top convolving lags."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')

    m = p.size
    floor = 1e-20 / m
    p = np.maximum(p, floor) / p.sum()

    best = p.copy()
    best_score = _score(p)
    deadline = time.monotonic() + time_limit
    step = 0.05
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        c = _conv_full(p, p)
        c = np.where(np.isfinite(c), c, 0.0)
        d = c[:m]  # positive lags
        cur_max = m * np.max(d)

        # Find top lags
        n_active = min(n_lags, m)
        top_idx = np.argsort(d)[-n_active:]
        top_mask = np.zeros(m, dtype=float)
        top_mask[top_idx] = 1.0

        # Gradient of sum of top lag values
        # d/dp_i (p*p)[k] = p[i-k] + p[i+k] (for valid i)
        grad = np.zeros(m, dtype=float)
        for k in top_idx:
            if k == 0:
                grad += 2.0 * p
            else:
                # p[i] contributes to (p*p)[k] via p[i]*p[i+k] and p[i+k]*p[i]
                # Actually (p*p)[k] = sum_i p[i]*p[i+k]
                # d/dp[j] = p[j-k] (if j>=k) + p[j+k] (if j+k < m)
                if k < m:
                    grad[:m-k] += p[k:]
                    grad[k:] += p[:m-k]

        # Project gradient: minimize grad^T p subject to sum p = 1, p >= 0
        # Use multiplicative update: p <- p * exp(-step * grad / max_grad)
        max_grad = np.max(grad)
        if max_grad > 1e-15:
            grad_norm = grad / max_grad
            p_try = p * np.exp(-step * grad_norm)
            p_try = np.maximum(p_try, floor) / p_try.sum()
            sc = _score(p_try)
            if np.isfinite(sc) and sc < best_score:
                p = p_try
                best = p.copy()
                best_score = sc
                step = min(step * 1.05, 0.5)
                stall = 0
            elif np.isfinite(sc) and sc < cur_max:
                p = p_try
                step *= 0.9
                stall += 1
            else:
                step *= 0.7
                stall += 1
        else:
            stall += 1

        if stall >= 20:
            p_try = _sanitize(p * np.exp(rng.normal(0, 0.08, m)))
            if p_try is not None:
                p_try = np.maximum(p_try, floor) / p_try.sum()
                sc = _score(p_try)
                if np.isfinite(sc) and sc <= best_score * 1.05:
                    p = p_try
                else:
                    p = best.copy()
            step = 0.05
            stall = 0

        if stall > 150:
            p = best.copy()
            step = 0.05
            stall = 0

        it += 1
        if it % 50 == 0 and verbose:
            print(f"  lag_focus: score={best_score:.6f}, iter={it}")

    return best, best_score

        start = int(rng.integers(0, n - w + 1))

        current_best = None
        current_score = cur_score

        for f in factors:
            cand = p.copy()
            cand[start:start+w] *= f
            cand = _sanitize(cand)
            if cand is None:
                continue
            cand = np.maximum(cand, floor) / cand.sum()
            sc = _score(cand)
            if sc < current_score - 1e-13:
                current_score = sc
                current_best = cand

        if current_best is not None:
            p = current_best
            cur_score = current_score
            if cur_score < best_score:
                best = p.copy()
                best_score = cur_score
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 80:
                p_try = _sanitize(p * np.exp(rng.normal(0, 0.06, n)))
                if p_try is not None:
                    p_try = np.maximum(p_try, floor) / p_try.sum()
                    sc = _score(p_try)
                    if np.isfinite(sc) and sc <= best_score * 1.05:
                        p = p_try
                        cur_score = sc
                    else:
                        p = best.copy()
                        cur_score = best_score
                stagnant = 0

        if stagnant > 400:
            p = best.copy()
            cur_score = best_score
            stagnant = 0

    return best, best_score


def _resample(arr, m):
    if m <= 0:
        return None
    a = _sanitize(arr)
    if a is None:
        return None
    n = a.size
    if n == m:
        return a.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(m) + 0.5) / m
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0) + 1e-12
    return _sanitize(out)


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses smoothed-max descent, lag-focused optimization, block search,
    and multi-scale refinement.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 60.0

    start_time = time.monotonic()
    deadline = start_time + budget - 15.0

    try:
        iseed = int(seed)
    except Exception:
        iseed = 12345
    rng = np.random.default_rng((iseed % 10**9) + 7)

    # Collect candidates: (score, array)
    candidates = []

    def add_cand(arr, sc=None):
        a = _sanitize(arr)
        if a is None:
            return
        if sc is None or not np.isfinite(sc):
            sc = _score(a)
        if not np.isfinite(sc):
            return
        candidates.append((sc, a.copy()))
        candidates.sort(key=lambda x: x[0])
        if len(candidates) > 20:
            candidates = candidates[:20]
        print(f"  candidate: score={sc:.6f}, len={a.size}")

    # Try to load previous best
    parent = None
    try:
        parent_raw = height_sequence_1
        parent = _sanitize(np.asarray(parent_raw, dtype=float))
        if parent is not None:
            sc = _score(parent)
            print(f"Parent score: {sc:.6f}, len={parent.size}")
            add_cand(parent, sc)
    except NameError:
        pass
    except Exception:
        pass

    # === Phase 1: Generate diverse initial candidates at multiple sizes ===
    target_n = 3072
    small_n = 768

    # Generate initial candidates at small size
    small_starts = []
    small_starts.append(_init_beta_density(small_n, rng))
    small_starts.append(_init_beta_density(small_n, rng))
    small_starts.append(_init_gaussians(small_n, rng, K=2))
    small_starts.append(_init_gaussians(small_n, rng, K=4))
    small_starts.append(_init_two_block(small_n, rng))
    small_starts.append(_init_two_block(small_n, rng))

    if parent is not None:
        sp = _resample(parent, small_n)
        if sp is not None:
            small_starts.append(sp)
            sp2 = _resample(parent, target_n)
            if sp2 is not None:
                add_cand(_smooth_vec(sp2, 1.0))

    rem = deadline - time.monotonic()
    print(f"Phase 1: small-n search, time remaining={rem:.1f}s")

    for sp in small_starts:
        rem = deadline - time.monotonic()
        if rem < 30:
            break
        t_alloc = min(25.0, max(8.0, (rem - 20) / max(1, len(small_starts))))
        p, sc = _run_descent(sp, rng, t_alloc, beta_start=30, beta_end=800)
        if p is not None:
            add_cand(p, sc)

    # Upsample best small candidates to target_n
    for sc, arr in candidates[:4]:
        if arr.size != target_n:
            big = _resample(arr, target_n)
            if big is not None:
                add_cand(big)
                # Also try smoothed versions
                add_cand(_smooth_vec(big, 1.0))
                add_cand(_smooth_vec(big, 0.5))

    # Generate fresh candidates at target_n
    print("Phase 1b: generating target-n candidates")
    for _ in range(3):
        p = _init_beta_density(target_n, rng)
        add_cand(p)
    for _ in range(2):
        p = _init_gaussians(target_n, rng, K=rng.integers(2, 5))
        add_cand(p)
    for _ in range(2):
        p = _init_two_block(target_n, rng)
        add_cand(_smooth_vec(p, 1.0))

    # === Phase 2: Main optimization on best candidates ===
    rem = deadline - time.monotonic()
    print(f"Phase 2: main optimization, time remaining={rem:.1f}s")

    # Select top 3-4 candidates for main optimization
    top_cands = [(sc, arr) for sc, arr in candidates if arr.size == target_n][:4]
    if not top_cands:
        top_cands = [(candidates[0][0], _resample(candidates[0][1], target_n))]
        if top_cands[0][1] is None:
            top_cands = [(_score(_init_beta_density(target_n, rng)), _init_beta_density(target_n, rng))]

    fracs = [0.35, 0.25, 0.20, 0.12]

    for idx, (sc0, p0) in enumerate(top_cands):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        frac = fracs[idx] if idx < len(fracs) else 0.1
        t_alloc = max(30, min(600, rem * frac))

        print(f"  Optimizing candidate {idx+1} (initial score={sc0:.6f}), budget={t_alloc:.0f}s")

        # Step 2a: Smoothed descent
        p, sc = _run_descent(p0, rng, t_alloc * 0.6, beta_start=40, beta_end=1500, verbose=False)
        if p is not None:
            add_cand(p, sc)

        rem = deadline - time.monotonic()
        if rem < 60:
            continue

        # Step 2b: Lag-focused optimization
        t_lag = min(t_alloc * 0.25, rem * 0.4)
        if t_lag > 20:
            p, sc = _run_lag_focus(p if p is not None else p0, rng, t_lag, n_lags=4)
            if p is not None:
                add_cand(p, sc)

        rem = deadline - time.monotonic()
        if rem < 60:
            continue

        # Step 2c: Descent again with higher beta
        t2 = min(t_alloc * 0.3, rem * 0.4)
        if t2 > 20:
            p, sc = _run_descent(p if p is not None else p0, rng, t2,
                                  beta_start=200, beta_end=3000, verbose=False)
            if p is not None:
                add_cand(p, sc)

    # === Phase 3: Local refinement ===
    rem = deadline - time.monotonic()
    print(f"Phase 3: local refinement, time remaining={rem:.1f}s")

    if rem > 60 and candidates:
        best_sc, best_arr = candidates[0]
        t_block = min(300, rem * 0.5)
        p, sc = _local_block_search(best_arr, rng, t_block)
        if p is not None:
            add_cand(p, sc)

        rem = deadline - time.monotonic()
        if rem > 60 and candidates:
            best_sc, best_arr = candidates[0]
            t_descent = min(200, rem * 0.5)
            p, sc = _run_descent(best_arr, rng, t_descent,
                                  beta_start=300, beta_end=4000, verbose=False)
            if p is not None:
                add_cand(p, sc)

        rem = deadline - time.monotonic()
        if rem > 40 and candidates:
            best_sc, best_arr = candidates[0]
            t_lag = min(150, rem * 0.5)
            p, sc = _run_lag_focus(best_arr, rng, t_lag, n_lags=3)
            if p is not None:
                add_cand(p, sc)

        rem = deadline - time.monotonic()
        if rem > 40 and candidates:
            best_sc, best_arr = candidates[0]
            t_block = min(200, rem * 0.5)
            p, sc = _local_block_search(best_arr, rng, t_block)
            if p is not None:
                add_cand(p, sc)

    # === Phase 4: Try upscaling to larger n ===
    rem = deadline - time.monotonic()
    print(f"Phase 4: upscaling, time remaining={rem:.1f}s")

    if rem > 180 and candidates:
        best_sc, best_arr = candidates[0]
        big_n = min(6144, 2 * best_arr.size)
        big = _resample(best_arr, big_n)
        if big is not None:
            t_big = min(120, rem * 0.4)
            p, sc = _run_descent(big, rng, t_big, beta_start=80, beta_end=2500, verbose=False)
            if p is not None:
                add_cand(p, sc)

            rem = deadline - time.monotonic()
            if rem > 60:
                p, sc = _local_block_search(p, rng, min(80, rem * 0.4))
                if p is not None:
                    add_cand(p, sc)

    # === Final: select best and verify ===
    if not candidates:
        p = _init_beta_density(target_n, rng)
        candidates = [(_score(p), p)]

    best_sc, best_arr = candidates[0]
    print(f"\nBest score: {best_sc:.6f}, length={best_arr.size}")

    # Final verification with evaluate_sequence
    out = np.asarray(best_arr, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s <= 0:
        out = np.ones(1)
    out = out / s

    # Check a few candidates with the official evaluator
    final_list = [float(x) for x in out]
    best_eval = float('inf')
    best_final = final_list

    for sc, arr in candidates[:5]:
        rem = deadline - time.monotonic()
        if rem < 3:
            break
        a = _sanitize(arr)
        if a is None:
            continue
        lst = [float(x) for x in a]
        try:
            ev = float(evaluate_sequence(lst))
        except Exception:
            ev = float('inf')
        if np.isfinite(ev) and ev < best_eval:
            best_eval = ev
            best_final = lst
            print(f"  eval candidate: {ev:.6f}")

    # Also evaluate our normalized best
    try:
        ev = float(evaluate_sequence(final_list))
    except Exception:
        ev = float('inf')
    if np.isfinite(ev) and ev < best_eval:
        best_eval = ev
        best_final = final_list

    print(f"Final best: {best_eval:.6f}")
    return best_final
