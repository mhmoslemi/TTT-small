"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS on the probability simplex
with a log-sum-exp smoothed max, multiple diverse restarts, and block-scaling polish.
"""

import numpy as np
from scipy.optimize import minimize
import time

def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(a, b):
    nout = len(a) + len(b) - 1
    if len(a) * len(b) <= 2000000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)

def _exact_score(a):
    a = np.asarray(a, dtype=float)
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    a_norm = a / s
    n = len(a_norm)
    c = _conv_full(a_norm, a_norm)
    return float(2.0 * n * np.max(c))

def _smoothed_obj(x, n, beta, L, fw_cache=None):
    """Returns (objective_value, gradient) for softmax parametrization."""
    # Compute a from x
    x_shifted = x - np.max(x)
    ex = np.exp(x_shifted)
    a = ex / ex.sum()
    
    # Compute convolution
    c = _conv_full(a, a)
    
    # Smoothed max
    max_c = float(np.max(c))
    e = np.exp(beta * (c - max_c))
    se = float(np.sum(e))
    w = e / se
    smooth_max = max_c + np.log(se) / beta
    obj = 2.0 * n * smooth_max
    
    # Gradient
    # cross_corr[i] = sum_{k=0}^{n-1} w[i+k] * a[k]
    # = (w * a_rev)[n-1+i]
    a_rev = a[::-1]
    fwr = np.fft.rfft(w, L)
    far = np.fft.rfft(a_rev, L)
    cross = np.fft.irfft(fwr * far, L)
    cross_corr = cross[n-1:2*n-1]
    
    grad_a = 2.0 * cross_corr - w[0::2] * a
    
    # Convert to x-gradient via softmax
    lam = float(np.dot(a, grad_a))
    grad_x = 2.0 * n * a * (grad_a - lam)
    
    return obj, grad_x
