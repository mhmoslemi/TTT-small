def _cross_corr(w, a):
    """
    Compute g[j] = sum_k w[k] * a[k-j] for j = 0, ..., n-1
    where w has length M and a has length n.
    Returns array of length n.
    """
    M = len(w)
    n = len(a)

    while L < M + n:
        L *= 2
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
