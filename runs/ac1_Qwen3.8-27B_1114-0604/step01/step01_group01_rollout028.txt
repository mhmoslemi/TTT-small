```python
def _compute_grad(p, w):
    """Compute gradient of sum_k w[k] * (p*p)[k] w.r.t. p.
    p has length L, w has length 2L-1.
    Returns grad of length L.
    """
    L = len(p)
    LFFT = _next_pow2(len(w) + L - 1)  # >= 3L-2
    Fp = np.fft.rfft(p, LFFT)
    Fw = np.fft.rfft(w, LFFT)
    # circular cross-correlation: ifft(conj(Fp) * Fw)
    corr = np.fft.irfft(np.conjugate(Fp) * Fw, LFFT)
    # Take first L values
    return 2.0 * corr[:L]
```

2. Compute weights w = topk_weights(b, k) (top-k with uniform weight, or softmax)
3. Compute grad = _compute_grad(p, w)
4. Multiplicative update: p_new = p * exp(-alpha * grad / max(grad))
5. Project onto simplex (clip at 0, normalize)

The key difference from the previous approach: instead of a smooth softmax over all values, I'll use top-k weights. This gives a much sharper gradient that focuses only on the peaks.

"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses top-k peak-focused multiplicative descent with adaptive step sizes,
direct peak shaving, coordinate refinement, and annealing restarts.
"""

import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 4000000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)


def _compute_gradient(p, w):
    """Compute gradient of sum_k w[k]*(p*p)_lin[k] w.r.t. p[j].
    p has length L, w has length 2L-1. Returns grad of length L.
    grad[j] = 2 * sum_m w[m+j]*p[m] (valid cross-correlation).
    """
    L = len(p)
    Lw = len(w)
    LFFT = _next_pow2(Lw + L - 1)
    Fp = np.fft.rfft(p, LFFT)
    Fw = np.fft.rfft(w, LFFT)
    corr = np.fft.irfft(np.conjugate(Fp) * Fw, LFFT)
    return 2.0 * corr[:L]


def _sanitize(p):
    """Clip to [0, 1000], normalize to sum 1."""
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 0 or not np.isfinite(s):
        return None
    return p / s


def _score(p):
    """Compute 2*n*max(conv(p,p)) for normalized p (sum=1)."""
    p = _sanitize(p)
    if p is None:
        return float('inf')
    n = len(p)
    c = _conv_full(p, p)
    if c.size == 0:
        return float('inf')
    return float(2.0 * n * np.max(c))


def _topk_weights(b, k, epsilon=0.01):
    """Top-k weights with small uniform background for stability."""
    n = len(b)
    if n == 0:
        return np.array([])
    b = np.clip(b, 0, None)
    k = max(1, min(k, n))
    threshold = np.partition(b, n - k)[n - k]
    w = np.zeros(n)
    w[b >= threshold] += 1.0 / k
    w[:] += epsilon / n
    s = w.sum()
    if s > 0:
        w /= s
    return w


def _softmax_weights(b, beta):
    """Softmax weights over convolution values."""
    n = len(b)
    if n == 0:
        return np.array([])
    b = np.clip(b, 0, None)
    shift = np.max(b)
    e = np.exp(beta * (b - shift))
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return e / s


def _two_block(n, frac, ratio):
    """Create two-block sequence."""
    p = np.ones(n, dtype=float)
    idx = int(round(frac * n))
    if 0 < idx < n:
        p[idx:] = ratio
    return _sanitize(p)


def _initial_diverse(n, rng, max_starts=8):
    """Generate diverse initial sequences."""
    starts = []
    # Two-block variants
    for frac in (0.3, 0.5, 0.7):
        for ratio in (0.5, 1.0, 2.0):
            starts.append(_two_block(n, frac, ratio))
    # Gaussian bumps
    for _ in range(3):
        K = int(rng.integers(2, 6))
        t = np.linspace(0, 1, n)
        p = np.zeros(n)
        for _ in range(K):
            c = rng.uniform(0.1, 0.9)
            s = rng.uniform(0.05, 0.3)
            p += np.exp(-((t - c) / s) ** 2)
        p = _sanitize(p)
        if p is not None:
            starts.append(p)
    # Linear ramp
    t = np.linspace(0, 1, n)
    p = _sanitize(t + 0.01)
    if p is not None:
        starts.append(p)
    p = _sanitize(1 - t + 0.01)
    if p is not None:
        starts.append(p)
    # Symmetric U-shape
    p = _sanitize((t - 0.5) ** 2 + 0.01)
    if p is not None:
        starts.append(p)
    return [s for s in starts if s is not None][:max_starts]


def _resample(arr, m):
    """Resample normalized vector to length m."""
    a = _sanitize(arr)
    if a is None or m <= 0:
        return None
    n = a.size
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m, dtype=float)
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0)
    return _sanitize(out)


def _mmd_optimize(p0, rng, time_limit, topk_frac=0.05, beta_max=500.0,
                  k_min=5, alpha_init=0.5, verbose=False):
    """Top-k focused multiplicative descent on the simplex."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    floor = 1e-16 / n
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_score = _score(p)
    deadline = time.monotonic() + time_limit
    alpha = alpha_init
    stall = 0
    it = 0
    k_top = max(k_min, int(n * topk_frac))
    beta = 0.0  # 0 means use top-k only

    while time.monotonic() < deadline:
        c = _conv_full(p, p)
        if c.size == 0:
            break
        b = np.maximum(c, 0.0)
        cur_score = float(2.0 * n * np.max(b))

        if cur_score < best_score:
            best = p.copy()
            best_score = cur_score
            stall = 0
            if verbose and it % 200 == 0:
                print(f"  MMD iter={it} score={best_score:.6f}")
        else:
            stall += 1

        # Compute weights: blend top-k and softmax for robustness
        if beta > 0:
            w_topk = _topk_weights(b, k_top, epsilon=0.0)
            w_sm = _softmax_weights(b, beta)
            w = 0.5 * w_topk + 0.5 * w_sm
        else:
            w = _topk_weights(b, k_top, epsilon=0.05)

        # Compute gradient
        grad = _compute_gradient(p, w)
        grad = np.maximum(grad, 0.0)
        max_grad = np.max(grad)
        if max_grad <= 1e-15:
            # Perturb
            p = _sanitize(p * np.exp(rng.normal(0, 0.1, n)))
            p = np.maximum(p, floor)
            p = p / p.sum()
            continue

        grad = grad / max_grad

        # Try multiple step sizes
        best_try = None
        best_try_score = cur_score
        for mult in (0.25, 0.5, 1.0, 2.0, 4.0, 8.0):
            a = alpha * mult
            if a < 1e-6 or a > 50:
                continue
            dec = np.exp(-a * grad)
            p_try = p * dec
            p_try = np.maximum(p_try, floor)
            s = p_try.sum()
            if s <= 0:
                continue
            p_try = p_try / s
            sc = _score(p_try)
            if sc < best_try_score - 1e-12:
                best_try_score = sc
                best_try = p_try
                alpha = a

        if best_try is not None:
            p = best_try
            stall = 0
        else:
            alpha = max(1e-4, alpha * 0.5)
            if stall > 30:
                # Annealing: random perturbation
                std = min(0.5, 0.05 + 0.01 * stall)
                p_try = _sanitize(p * np.exp(rng.normal(0, std, n)))
                if p_try is not None:
                    p_try = np.maximum(p_try, floor)
                    p_try = p_try / p_try.sum()
                    sc = _score(p_try)
                    if sc < best_score * 1.1:
                        p = p_try
                else:
                    p = best.copy()
                stall = 0

            if stall > 100:
                p = best.copy()
                alpha = alpha_init
                stall = 0

        it += 1
        if it % 30 == 0:
            k_top = max(k_min, int(n * max(0.01, topk_frac * (1 - 0.5 * it / max(1, time_limit)))))
            beta = min(beta_max, beta + 1.0)

    return best, best_score


def _peak_shave(p, rng, time_limit, num_shaves=50):
    """Directly reduce peaks by shaving contributing elements."""
    p = _sanitize(p)
    if p is None:
        return None, float('inf')
    n = len(p)
    floor = 1e-16 / n
    p = np.maximum(p, floor)
    p = p / p.sum()
    best = p.copy()
    best_score = _score(p)
    deadline = time.monotonic() + time_limit
    it = 0

    while time.monotonic() < deadline and it < num_shaves:
        c = _conv_full(p, p)
        if c.size == 0:
            break
        b = np.maximum(c, 0.0)
        cur_score = float(2.0 * n * np.max(b))

        if cur_score < best_score:
            best = p.copy()
            best_score = cur_score

        # Find top peak lag
        k_peak = int(np.argmax(b))
        # Find elements contributing most to this peak
        i_min = max(0, k_peak - n + 1)
        i_max = min(n - 1, k_peak)
        if i_max < i_min:
            it += 1
            continue

        # Products a[i] * a[k_peak - i] for valid i
        idx = np.arange(i_min, i_max + 1)
        products = p[idx] * p[k_peak - idx]
        # Reduce the elements with largest products
        top_idx = idx[np.argsort(products)[-max(1, n // 50):]]
        # Shrink them
        factor = rng.uniform(0.5, 0.9)
        p_try = p.copy()
        p_try[top_idx] *= factor
        # Compensate by boosting low elements
        low_mask = p_try < np.median(p_try)
        boost = (p[top_idx].sum() * (1 - factor)) / max(1, low_mask.sum())
        p_try[low_mask] += boost
        p_try = np.maximum(p_try, floor)
        p_try = _sanitize(p_try)
        if p_try is not None:
            sc = _score(p_try)
            if sc < cur_score:
                p = p_try

        it += 1

    return best, best_score


def _coordinate_descent(p, rng, time_limit, num_iter=2000):
    """Coordinate-wise multiplicative adjustments."""
    p = _sanitize(p)
    if p is None:
        return None, float('inf')
    n = len(p)
    floor = 1e-16 / n
    p = np.maximum(p, floor)
    p = p / p.sum()
    best = p.copy()
    best_score = _score(p)
    deadline = time.monotonic() + time_limit
    cur_score = best_score
    stagnant = 0
    factors = (0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5)

    while time.monotonic() < deadline and stagnant < num_iter:
        # Pick random coordinate or small block
        r = rng.random()
        if r < 0.7:
            idx = int(rng.integers(0, n))
            w = 1
            start = idx
        elif r < 0.9:
            w = int(rng.integers(2, max(3, min(20, n // 100))))
            start = int(rng.integers(0, n - w + 1))
        else:
            w = int(rng.integers(max(3, n // 100), max(4, n // 20)))
            start = int(rng.integers(0, n - w + 1))

        best_local = p.copy()
        best_local_score = cur_score
        for f in factors:
            p_try = p.copy()
            p_try[start:start + w] *= f
            p_try = np.maximum(p_try, floor)
            p_try = p_try / p_try.sum()
            sc = _score(p_try)
            if sc < best_local_score - 1e-12:
                best_local_score = sc
                best_local = p_try

        if best_local_score < cur_score:
            p = best_local
            cur_score = best_local_score
            if cur_score < best_score:
                best = p.copy()
                best_score = cur_score
            stagnant = 0
        else:
            stagnant += 1

    return best, best_score
