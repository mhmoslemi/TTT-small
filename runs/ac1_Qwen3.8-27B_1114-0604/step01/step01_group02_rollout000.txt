def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def score_from_conv(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def logsumexp_weights(c, beta):
    """Compute softmax-like weights for the convolution peaks."""
    c_max = np.max(c)
    if c_max < 1e-15:
        return np.ones(len(c)) / len(c), 0.0
    shifted = beta * (c - c_max)
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 0)
    weights = np.exp(shifted)
    total = weights.sum()
    if total < 1e-30:
        weights = np.zeros(len(c))
        weights[np.argmax(c)] = 1.0
        return weights, c_max
    weights = weights / total
    # The log-sum-exp value (scaled)
    lse = c_max + np.log(total) / beta
    return weights, lse

def compute_grad_smooth(a, c, weights, n, S):

def clamp_nonneg(a):
    return np.clip(a, 0.0, 1000.0)

def adam_step(a, g, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    bc1 = b1 ** t
    bc2 = b2 ** t
    m[:] = b1 * m + (1 - b1) * g
    v[:] = b2 * v + (1 - b2) * g * g
    m_hat = m / (1 - bc1)
    v_hat = v / (1 - bc2)
    a -= lr * m_hat / (np.sqrt(v_hat) + eps)

def optimize_continuation(a0, budget, rng, n):
    """Optimize using continuation method with adaptive smoothing."""
    best_score = np.inf
    best_a = a0.copy()
    
    a = clamp_nonneg(a0.copy())
    S = a.sum()
    if S < 1e-3:
        a = np.ones(n)
        S = n
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    # Continuation: start with large top-k (smooth), decrease to small (sharp)
    # We'll use a schedule of top-k values
    n_total = int(budget / 0.003)  # estimate total iterations
    n_phases = 8
    iters_per_phase = max(50, n_total // n_phases)
    
    topk_schedule = [max(5, n // (i + 1)) for i in range(n_phases)]
    topk_schedule = [min(tk, 2*n-1) for tk in topk_schedule]
    # Make it decreasing
    topk_schedule = [max(3, int(x)) for x in topk_schedule]
    
    phase_time = budget / n_phases
    
    for phase in range(n_phases):
        phase_end = time.monotonic() + phase_time
        topk = topk_schedule[phase]
        lr = 0.1 / (phase + 1)
        iters = 0
        no_improve = 0
        
        while time.monotonic() < phase_end:
            c = fast_self_conv(a)
            S = a.sum()
            if S < 1e-10:
                break
            score = score_from_conv(n, c, S)
            
            if score < best_score:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > 50 * (phase + 1):
                # Restart from best with perturbation
                scale = 0.03 * (phase + 1)
                a = best_a * (1 + scale * rng.standard_normal(n))
                a = clamp_nonneg(a)
                if a.sum() < 1e-3:
                    a = best_a.copy()
                m[:] = 0
                v[:] = 0
                t = 0
                no_improve = 0
                continue
            
            # Compute weighted gradient
            idx = np.argpartition(c, -topk)[-topk:]
            c_topk = c[idx]
            # Softmax weights on top-k
            if c_topk[0] > 1e-12:
                temp = 0.1 * c_topk[0]
                w_topk = np.exp((c_topk - c_topk[0]) / temp)
                w_topk = w_topk / w_topk.sum()
            else:
                w_topk = np.ones(topk) / topk
            
            # Build full weight vector
            w = np.zeros(2*n-1)
            w[idx] = w_topk
            
            grad, M_w = compute_grad_smooth(a, c, w, n, S)
            
            t += 1
            adam_step(a, grad, m, v, t, lr)
            a = clamp_nonneg(a)
            iters += 1
    
    return best_score, best_a
