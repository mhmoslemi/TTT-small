    c = fast_self_conv(a)
    S = a.sum()
    c_max = c.max()
    exp_bc = np.exp(beta * (c - c_max))
    w = exp_bc / exp_bc.sum()
    lse = c_max + np.log(exp_bc.sum()) / beta
    
    # gradient of lse w.r.t. a
    grad_a = np.zeros(n)
    for j in range(n):
        k_start = j
        k_end = min(j + n - 1, 2*n - 2)
        grad_a[j] = 2.0 * np.dot(w[k_start:k_end+1], a[0:k_end-k_start+1])
    
    # gradient of 2n * lse / S^2 w.r.t. a
    S2 = S * S
    grad_f_a = 2.0 * n * (grad_a / S2 - 2.0 * lse / (S2 * S))
    
    # chain rule: d/dx = d/da * a
    grad_f_x = grad_f_a * a
    
    return smooth_objective(x, beta, n), grad_f_x

    M = 2 * n - 1

    return np.fft.irfft(A * A, M_fft)[:M]

    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())

    """
    Compute grad[j] = 2 * sum_{j'=0}^{n-1} w[j+j'] * a[j'] for j = 0..n-1.
    Uses FFT-based cross-correlation.

    x = np.linspace(0.0, 1.0, n)
    inits.append("cos", 0.5 + 0.5 * np.cos(np.pi * x))
    inits.append("gauss", 0.3 + 0.7 * np.exp(-6.0 * (x - 0.5) ** 2))
    inits.append("tri", np.maximum(1.0 - 2.0 * np.abs(x - 0.5), 0.01))
    inits.append("rand", 0.2 + 0.6 * rng.random(n))
    inits.append("blocks", np.repeat(0.3 + 0.5 * rng.random(n // 8 + 1), 8)[:n])
    return inits


def optimize_seq(a0, deadline, rng, beta_start=0.3, beta_end=80.0,
                 lr_start=0.15, lr_end=0.002, patience=300, verbose_tag=""):
    n = len(a0)
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    c = fast_self_conv(a)
    S = float(a.sum())
    score = 2.0 * n * float(c.max()) / (S * S) if S > 1e-10 else np.inf
    best_score = score
    best_a = a.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_imp = 0
    max_pat = patience
    n_restarts = 0

    while time.monotonic() < deadline:
        if no_imp >= max_pat:
            n_restarts += 1
            mode = rng.random()
            if mode < 0.4:
                sc = 0.02 + 0.08 * rng.random()
                a = best_a * (1.0 + sc * rng.standard_normal(n))
            elif mode < 0.7:
                sc = 0.05 + 0.10 * rng.random()
                a = np.maximum(best_a * np.exp(sc * rng.standard_normal(n)), 0.0)
            else:
                a = best_a.copy()
                k = int(n * (0.1 + 0.3 * rng.random()))
                idx = rng.choice(n, size=k, replace=False)
                a[idx] = np.maximum(a[idx] * (1.0 + 0.3 * rng.standard_normal(k)), 0.0)
            a = np.minimum(a, 1000.0)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            no_imp = 0
            max_pat = max(100, max_pat * 8 // 10)
            continue

        frac = min(1.0, t / 3000.0)
        beta = beta_start + (beta_end - beta_start) * (frac ** 0.7)
        lr = lr_start * (lr_end / lr_start) ** frac

        c = fast_self_conv(a)
        S = float(a.sum())
        if S < 1e-10:
            break
        grad_a, sm_val = smooth_max_and_grad(a, c, beta)
        S2 = S * S
        grad = 2.0 * n * (grad_a / S2 - 2.0 * sm_val * 2.0 / (S2 * S))

        t += 1
        b1, b2, eps = 0.9, 0.999, 1e-8
        m[:] = b1 * m + (1.0 - b1) * grad
        v[:] = b2 * v + (1.0 - b2) * grad * grad
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        a -= lr * mh / (np.sqrt(vh) + eps)
        a = np.maximum(a, 0.0)

        c = fast_self_conv(a)
        S = float(a.sum())
        if S < 1e-10:
            score = np.inf
        else:
            score = 2.0 * n * float(c.max()) / (S * S)

        if score < best_score:
            best_score = score
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total = max(1.0, float(budget_s))
    deadline = t_start + total - 6.0

    best_score = np.inf
    best_a = None

    parent_arr = None
    try:
        pa = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if pa.size > 0:
            parent_arr = np.maximum(pa, 0.0)
            s, _, _ = compute_obj(parent_arr)
            print(f"Parent: len={pa.size}, score={s:.6f}")
    except Exception:
        pass

    # Determine which lengths to try and time allocation
    # Strategy: spend most time on 2-3 lengths, refine best at end
    candidates = []

    if parent_arr is not None:
        n_p = len(parent_arr)
        n_p = min(max(n_p, 1024), 16384)
        # Round to power of 2 for FFT efficiency
        n_p = 1 << (n_p - 1).bit_length()
        candidates.append(("parent", n_p, parent_arr[:n_p].copy()))

    # Always try these lengths
    candidates.append(("n2048", 2048, None))
    candidates.append(("n4096", 4096, None))
    candidates.append(("n8192", 8192, None))

    # Allocate time: 60% to exploration, 40% to final polish
    explore_end = t_start + 0.60 * (total - 6.0)
    polish_start = explore_end
    polish_deadline = deadline

    # Phase 1: Explore multiple lengths with diverse inits
    # Time per length proportional to expected value
    n_lengths = len(candidates)
    time_per_length = (explore_end - t_start) / n_lengths

    for label, n, parent_a in candidates:
        if time.monotonic() > explore_end:
            break
        sub_dl = min(explore_end, time.monotonic() + time_per_length)
        print(f"\n=== Exploring {label} (n={n}), time budget {time_per_length:.0f}s ===")

        # Build initializations
        inits = []
        if parent_a is not None and len(parent_a) == n:
            inits.append(("parent", parent_a.copy()))
        else:
            inits.extend(build_inits(n, rng))

        time_each = (sub_dl - time.monotonic()) / max(1, len(inits))
        print(f"  {len(inits)} initializations, ~{time_each:.1f}s each")

        for i, (tag, a0) in enumerate(inits):
            if time.monotonic() > sub_dl:
                break
            sub_sub_dl = min(sub_dl, time.monotonic() + time_each)
            sc, aa = optimize_seq(a0, sub_sub_dl, rng, verbose_tag=f"{label}/{tag}")
            s_eval, _, _ = compute_obj(aa)
            if s_eval < best_score:
                best_score = s_eval
                best_a = aa.copy()
                print(f"  [{label}/{tag}] NEW BEST: {s_eval:.6f} "
                      f"(restarts={n_restarts if 'n_restarts' in dir() else '?'})")
            else:
                print(f"  [{label}/{tag}] score={s_eval:.6f}")

    print(f"\nAfter exploration: best = {best_score:.6f}")

    # Phase 2: Polish the global best
    if best_a is not None and time.monotonic() < polish_deadline:
        n_best = len(best_a)
        print(f"\n=== Polishing n={n_best} ===")
        sc, aa = optimize_seq(best_a, polish_deadline, rng,
                              beta_start=5.0, beta_end=200.0,
                              lr_start=0.005, lr_end=0.00005, patience=600)
        s_eval, _, _ = compute_obj(aa)
        if s_eval < best_score:
            best_score = s_eval
            best_a = aa.copy()
            print(f"  Polish improved: {s_eval:.6f}")
        else:
            # Re-evaluate the original best
            s_orig, _, _ = compute_obj(best_a)
            best_score = min(best_score, s_orig)
            print(f"  No improvement from polish: {s_orig:.6f}")

    # Safety fallback
    if best_a is None:
        best_a = np.ones(2048) * 0.5

    out = [float(x) for x in best_a]
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - t_start
    print(f"\nFINAL: internal={best_score:.6f}, verified={verified:.6f}, "
          f"len={len(out)}, time={elapsed:.1f}s")
    return out
