def softmax_objective(z,n,beta):
    z=np.asarray(z,float).reshape(-1)
    if z.size != n:
       tmp=np.zeros(n); tmp[:min(n,z.size)]=z[:min(n,z.size)]; z=tmp

       return BIG, np.zeros(n)

    b=fft_convolution(a)

    g=np.where(np.isfinite(g),g,0)

    grad=np.where(np.isfinite(grad),grad,0)

def add_candidate(pool,a,score,max_size=16):
   if a is None or a.size==0: return
   if not np.isfinite(score): return
   for i in range(len(pool)):
       ex,sv=pool[i]
       if ex.size==a.size:
           diff=float(np.max(np.abs(ex-a)))

"""
Multi-resolution nonnegative sequence search for
min  2*n*max(a*a)/(sum(a))^2.

The search parameterizes a as a softmax of a full log-vector, uses FFT-based
convolution/gradient, and optimizes a smooth max of the convolution with
L-BFGS-B.  It warms up through several lengths, uses the previous construction
when available, and combines several deterministic, random, and perturbed
starting profiles.  A simple projected log-gradient fallback is included if
SciPy is unavailable.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize as scipy_minimize
    HAVE_SCIPY = True
except Exception:
    scipy_minimize = None
    HAVE_SCIPY = False

ZMAX = 22.0
BIG = 1e100
MAX_BETA = 3000000.0


def fft_convolution(a):
    """Linear convolution a*a using FFT, clipped to nonnegative."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)

    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1

    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    b = np.ascontiguousarray(b, dtype=np.float64)
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    """
    Smooth maximum of b and normalized softmax weights.
    Returns (G, w), where G = max(b) + beta^-1 log sum exp(beta(b-max(b))).
    """
    b = np.asarray(b, dtype=np.float64).reshape(-1)
    if b.size == 0:
        return float("inf"), np.empty(0, dtype=np.float64)

    m = float(np.max(b))
    if not np.isfinite(m):
        return float("inf"), np.zeros(b.size, dtype=np.float64)

    if m <= 0.0 or beta <= 0.0:
        w = np.full(b.size, 1.0 / float(b.size), dtype=np.float64)
        return m, w

    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))

    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros(b.size, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0

    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w


def corr_pw(p, w):
    """
    Return s_i = sum_j p_j w_{i+j}, for i=0..n-1.
    This is the one-sided cross-correlation needed for d(a*a)/da.
    """
    p = np.asarray(p, dtype=np.float64).reshape(-1)
    w = np.asarray(w, dtype=np.float64).reshape(-1)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    if n == 1:
        return np.array([float(p[0] * w[0])], dtype=np.float64)

    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1

    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]

    s = C[m - n:m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)


def normalize_simplex(a):
    """Clip to nonnegative and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64).reshape(-1).copy()
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(a.size, 1.0 / float(a.size), dtype=np.float64)
    return a / s


def softmax_a(z):
    """Stable full softmax."""
    z = np.asarray(z, dtype=np.float64).reshape(-1)
    n = z.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if not np.all(np.isfinite(z)):
        z = np.zeros(n, dtype=np.float64)

    m = float(np.max(z))
    e = np.exp(z - m)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return e / s


def a_to_z(a, n):
    """Convert a positive sequence to log-softmax parameters."""
    a = normalize_simplex(a)
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if a.size != n:
        a = np.resize(a, n)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 1e-300)
    z = np.log(a)
    z -= float(np.mean(z))
    z = np.where(np.isfinite(z), z, 0.0)
    return np.clip(z, -ZMAX, ZMAX).astype(np.float64)


def fast_score(a):
    """Fast scale-invariant objective using FFT convolution."""
    a = np.asarray(a, dtype=np.float64).reshape(-1)
    n = a.size
    if n == 0:
        return float("inf")

    s = float(np.sum(a))
    if s <= 0.0 or not np.isfinite(s):
        return float("inf")

    b = fft_convolution(a)
    if b.size == 0:
        return float("inf")

    m = float(np.max(b))
    if not np.isfinite(m):
        return float("inf")

    return float(2.0 * n * m / (s * s))


def softmax_objective(z, n, beta):
    """
    Objective and gradient in log-softmax coordinates.
    Objective is n * smooth_max(a*a), which is half the target scale.
    """
    z = np.asarray(z, dtype=np.float64).reshape(-1)
    if z.size != n:
        tmp = np.zeros(n, dtype=np.float64)
        m = min(n, z.size)
        tmp[:m] = z[:m]
        z = tmp

    if n == 0:
        return BIG, np.empty(0, dtype=np.float64)

    if not np.all(np.isfinite(z)):
        return BIG, np.zeros(n, dtype=np.float64)

    zc = np.clip(z, -ZMAX, ZMAX)
    a = softmax_a(zc)

    b = fft_convolution(a)
    G, w = smooth_max_and_weights(b, beta)

    if not np.isfinite(G):
        return BIG, np.zeros(n, dtype=np.float64)

    h = corr_pw(a, w)
    if h.size != n:
        h = np.resize(h, n)

    g = 2.0 * h
    g = np.where(np.isfinite(g), g, 0.0)

    mean_g = float(np.dot(a, g))
    grad = a * (g - mean_g)

    val = float(n * G)
    if not np.isfinite(val):
        val = BIG
        grad = np.zeros(n, dtype=np.float64)

    grad = np.where(np.isfinite(grad), grad, 0.0)
    return val, grad.astype(np.float64)


def lbfgs_run(z0, n, beta, bounds, maxfun, maxiter, maxcor):
    """Run L-BFGS-B robustly."""
    if n == 0:
        return np.empty(0, dtype=np.float64)

    z0 = np.asarray(z0, dtype=np.float64).reshape(-1)
    if z0.size != n:
        tmp = np.zeros(n, dtype=np.float64)
        m = min(n, z0.size)
        tmp[:m] = z0[:m]
        z0 = tmp
    z0 = np.clip(z0, -ZMAX, ZMAX)

    if not HAVE_SCIPY:
        return z0

    options = {
        "maxiter": int(maxiter),
        "ftol": 1e-11,
        "gtol": 1e-6,
        "maxls": 30,
        "maxcor": int(maxcor),
        "maxfun": int(maxfun),
    }

    try:
        res = scipy_minimize(
            softmax_objective,
            z0,
            args=(n, beta),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options=options,
        )
        z = np.asarray(res.x, dtype=np.float64).reshape(-1)
    except Exception:
        z = z0.copy()

    if z.size != n or not np.all(np.isfinite(z)):
        z = z0.copy()

    return np.clip(z, -ZMAX, ZMAX).astype(np.float64)


def simple_refine(z0, n, beta, deadline):
    """Fallback log-gradient refinement if SciPy is unavailable."""
    if n == 0:
        return np.empty(0, dtype=np.float64)

    z = np.asarray(z0, dtype=np.float64).reshape(-1)
    if z.size != n:
        z = np.zeros(n, dtype=np.float64)
    z = np.clip(z, -ZMAX, ZMAX)

    best_z = z.copy()
    a = softmax_a(z)
    best_sc = fast_score(a)

    current_base = 1.0
    no_improve = 0

    for _ in range(1000000):
        if time.monotonic() >= deadline:
            break

        val, grad = softmax_objective(z, n, beta)
        if not np.isfinite(val):
            break

        gmax = 0.0
        if grad.size:
            gmax = float(np.max(np.abs(grad)))
        if gmax < 1e-14:
            break

        if current_base <= 0.0:
            current_base = min(1e6, 0.5 / gmax)

        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001):
            alpha = current_base * mult
            zc = np.clip(z - alpha * grad, -ZMAX, ZMAX)
            sc = fast_score(softmax_a(zc))
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_z = zc.copy()
                z = zc
                current_base = min(1e6, current_base * 1.2)
                accepted = True
                no_improve = 0
                break

        if not accepted:
            current_base *= 0.5
            no_improve += 1
            if current_base < 1e-8 or no_improve > 300:
                break

    return best_z

    return 0.14


def base_maxfun(n):
    """Base function-call budget for L-BFGS at a given length."""
    if n <= 256:
        return 8000
    if n <= 512:
        return 5000
    if n <= 1024:
        return 3000
    if n <= 2048:
        return 2000
    if n <= 4096:
        return 1400
    if n <= 8192:
        return 1000
    if n <= 16384:
        return 800
    return 700


def maxfun_for_budget(n, time_left):
    """Choose maxfun for one L-BFGS call, respecting remaining time."""
    if time_left <= 0.0:
        return 60
    t = estimate_eval_time(n)
    allowed = int(max(0.0, time_left) / max(t, 1e-6) * 0.55)
    mf = min(base_maxfun(n), max(60, allowed))
    return int(mf)


def beta_schedule(n, score):
    """Coarse-to-fine smooth-max betas."""
    if not np.isfinite(score) or score <= 1e-12:
        score = 2.0

    base = 2.0 * float(n) / float(score)
    mults = (0.2, 1.0, 4.0, 16.0, 64.0, 256.0)

    out = []
    for m in mults:
        b = base * m
        if b < 50.0:
            b = 50.0
        if b > MAX_BETA:
            b = MAX_BETA
        if not out or b > out[-1] * 1.25:
            out.append(float(b))
        else:
            if b > out[-1]:
                out[-1] = float(b)
    return out


def lbfgs_refine_seq(a0, n, deadline, ref_score):
    """
    Refine one normalized sequence at length n until deadline.
    Uses a continuation over increasing beta.
    """
    a0 = normalize_simplex(a0)
    if a0.size == 0:
        return a0, float("inf")

    best_a = a0.copy()
    best_sc = fast_score(a0)

    z = a_to_z(a0, n)
    z_best = z.copy()
    z_cont = z.copy()
    z_cont_score = best_sc

    score_base = ref_score if np.isfinite(ref_score) else best_sc
    if not np.isfinite(score_base) or score_base <= 1e-12:
        score_base = best_sc
    score_for_beta = min(float(score_base), float(best_sc))

    betas = beta_schedule(n, score_for_beta)
    if not betas:
        betas = [1e5]

    if not HAVE_SCIPY:
        last_beta = betas[-1]
        z_new = simple_refine(z, n, last_beta, deadline)
        a_new = softmax_a(z_new)
        sc_new = fast_score(a_new)
        if sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_a = a_new.copy()
        return best_a, best_sc

    bounds = [(-ZMAX, ZMAX)] * n

    for i, beta in enumerate(betas):
        now = time.monotonic()
        if now >= deadline:
            break

        time_left = max(0.0, deadline - now)
        if time_left < 3.0:
            break

        maxfun = maxfun_for_budget(n, time_left)
        maxiter = max(20, maxfun // 10)

        if z_cont_score < best_sc * 1.02 + 1e-7:
            z_start = z_cont
        else:
            z_start = z_best

        z_new = lbfgs_run(z_start, n, beta, bounds, maxfun, maxiter, 12)
        a_new = softmax_a(z_new)
        sc_new = fast_score(a_new)

        z_cont = z_new
        z_cont_score = sc_new

        if sc_new < best_sc - 1e-12:
            best_sc = sc_new
            best_a = a_new.copy()
            z_best = z_new.copy()
            print(f"  refine n={n} beta={beta:.3e} score={best_sc:.8f}")

    return best_a, best_sc


def upsample_to_n(a, n):
    """Linearly resample a normalized sequence to length n."""
    a = normalize_simplex(a)
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    m = a.size
    if n == 1:
        return np.array([1.0], dtype=np.float64)
    if m == n:
        return a.astype(np.float64)
    if m == 1:
        return np.full(n, 1.0 / float(n), dtype=np.float64)

    x_old = np.linspace(0.0, 1.0, m, endpoint=False, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, n, endpoint=False, dtype=np.float64)
    b = np.interp(x_new, x_old, a)
    b = np.maximum(b, 0.0)
    return normalize_simplex(b)


def upsample_block(a, n):
    """Block-repeat resampling to keep the previous discrete shape."""
    a = normalize_simplex(a)
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    m = a.size
    if m == n:
        return a.astype(np.float64)
    if m > n:
        return upsample_to_n(a, n)
    if m == 1:
        return np.full(n, 1.0 / float(n), dtype=np.float64)

    counts = np.floor(n / float(m)).astype(np.int64)
    rem = n - int(counts.sum())
    if rem > 0:
        counts[:rem] += 1

    b = np.repeat(a, counts)
    if b.size > n:
        b = b[:n]
    elif b.size < n:
        b = np.concatenate([b, np.full(n - b.size, b[-1], dtype=np.float64)])

    return normalize_simplex(b)


def smooth_noise(n, scale, rng):
    """Real low-pass random field with standard deviation scale."""
    if n <= 0:
        return np.empty(0, dtype=np.float64)

    L = 1
    while L < 2 * n:
        L <<= 1

    half = L // 2 + 1
    k = np.arange(half, dtype=np.float64)
    decay = (k + 1.0) ** (-1.0)

    zr = decay * (rng.normal(size=half) + 1j * rng.normal(size=half))
    z = np.fft.irfft(zr, n=L)[:n]

    z = np.asarray(z, dtype=np.float64)
    z -= float(np.mean(z))
    sd = float(np.std(z))
    if sd > 1e-12:
        z = z * (scale / sd)
    else:
        z = np.zeros(n, dtype=np.float64)

    return z


def perturb_sequence(a, n, scale, rng):
    """Smooth random perturbation in log-space."""
    a = upsample_to_n(a, n)
    z = a_to_z(a, n)
    dz = smooth_noise(n, scale, rng)
    z = np.clip(z + dz, -ZMAX, ZMAX)
    return softmax_a(z)


def mix_sequences(a, b, alpha, n):
    """Linear normalized mixture of two sequences resampled to n."""
    a = upsample_to_n(a, n)
    b = upsample_to_n(b, n)
    c = alpha * a + (1.0 - alpha) * b
    return normalize_simplex(c)


def analytic_profiles(n, rng):
    """Diverse nonnegative starting profiles."""
    out = []
    if n <= 0:
        return out

    x = np.linspace(0.0, 1.0, n, endpoint=False, dtype=np.float64)

    out.append(np.ones(n, dtype=np.float64))

    for coeff in (0.2, 0.45, 0.7):
        a = np.maximum(1.0 + coeff * np.cos(2.0 * np.pi * x), 0.02)
        out.append(a)
        a = np.maximum(1.0 + coeff * np.cos(4.0 * np.pi * x + 0.7), 0.02)
        out.append(a)

    for coeff in (0.2, 0.5, 0.8):
        a = np.maximum(1.0 + coeff * np.cos(np.pi * x), 0.02)
        out.append(a)

    for p in (0.4, 0.8, 1.5, 2.5):
        s = np.sin(np.pi * x)
        s = np.maximum(s, 1e-3)
        out.append(s ** p)

    out.append(np.maximum(1.0 + 0.6 * (x - 0.5), 0.02))

    for sc in (0.3, 0.6, 1.0):
        z = smooth_noise(n, sc, rng)
        out.append(softmax_a(z))

    res = []
    for a in out:
        res.append(normalize_simplex(a))
    return res


def add_unique(ordered, a, max_len):
    """Add a sequence to a list if not already present and list not full."""
    if a is None or a.size == 0 or max_len <= 0:
        return False
    if len(ordered) >= max_len:
        return False

    for b in ordered:
        if b.size == a.size:
            d = float(np.max(np.abs(a - b)))
            if d < 1e-8:
                return False

    ordered.append(a)
    return True


def select_top(scored, count):
    """Return up to count arrays with smallest scores from (score, array) list."""
    if count <= 0 or not scored:
        return []

    pool = [item for item in scored]
    out = []
    for _ in range(count):
        if not pool:
            break
        bi = 0
        bs = pool[0][0]
        for i in range(1, len(pool)):
            si = pool[i][0]
            if si < bs:
                bi = i
                bs = si
        out.append(pool[bi][1])
        del pool[bi]
    return out


def add_candidate(pool, a, score, max_size=16):
    """Maintain a small pool of good/diverse candidate sequences."""
    if a is None or a.size == 0:
        return
    if not np.isfinite(score):
        return

    for i in range(len(pool)):
        ex, sv = pool[i]
        if ex.size == a.size:
            d = float(np.max(np.abs(ex - a)))
            if d < 1e-8:
                if score < sv:
                    pool[i] = (a.copy(), float(score))
                return

    pool.append((a.copy(), float(score)))
    if len(pool) > max_size:
        worst = 0
        worst_score = -1.0
        for i in range(len(pool)):
            sv = pool[i][1]
            if sv > worst_score:
                worst = i
                worst_score = sv
        del pool[worst]


def make_start_list(a_prev, a_best, n, rng, max_count, pool=None):
    """
    Build an ordered list of starts.
    Earlier entries are trained with larger time budgets.
    """
    if max_count <= 0 or n <= 0:
        return []

    ordered = []

    for a_in in (a_best, a_prev):
        if a_in is not None:
            au = upsample_to_n(a_in, n)
            add_unique(ordered, au, max_count)

    if a_best is not None and n >= a_best.size and a_best.size > 0 and (n % a_best.size) == 0:
        au = upsample_block(a_best, n)
        add_unique(ordered, au, max_count)

    if pool:
        pcopy = [(sv, av) for av, sv in pool]
        top = select_top(pcopy, 2)
        for av in top:
            au = upsample_to_n(av, n)
            add_unique(ordered, au, max_count)

    raw = analytic_profiles(n, rng)
    scored = []
    for av in raw:
        sc = fast_score(av)
        if np.isfinite(sc):
            scored.append((sc, av))

    rem = max(0, max_count - len(ordered))
    if rem > 0:
        top = select_top(scored, rem)
        for av in top:
            add_unique(ordered, av, max_count)

    if a_best is not None:
        base = upsample_to_n(a_best, n)
        for sc in (0.2, 0.5, 0.8, 1.2):
            if len(ordered) >= max_count:
                break
            p = perturb_sequence(base, n, sc, rng)
            add_unique(ordered, p, max_count)

    if pool and len(ordered) < max_count:
        pcopy = [(sv, av) for av, sv in pool]
        top = select_top(pcopy, 2)
        if len(top) >= 2:
            mix = mix_sequences(top[0], top[1], 0.5, n)
            add_unique(ordered, mix, max_count)

    if len(ordered) < max_count:
        for sc, av in scored:
            if len(ordered) >= max_count:
                break
            add_unique(ordered, av, max_count)

    if not ordered:
        ordered.append(np.full(n, 1.0 / float(n), dtype=np.float64))

    return ordered[:max_count]


def run_search_starts(n, deadline, rng, a_best, best_score, a_prev, pool, max_starts):
    """Run several L-BFGS starts at a fixed length."""
    now = time.monotonic()
    if deadline <= now or n <= 0:
        return a_best, best_score

    starts = make_start_list(a_prev, a_best, n, rng, max_starts, pool)
    if not starts:
        return a_best, best_score

    total = deadline - now
    for idx, start in enumerate(starts):
        now = time.monotonic()
        if now >= deadline:
            break

        remaining = max(0.0, deadline - now)
        if idx == 0:
            alloc = 0.55 * remaining
        elif idx == 1:
            alloc = 0.30 * remaining
        elif idx == 2:
            alloc = 0.15 * remaining
        else:
            alloc = 0.05 * remaining

        alloc = max(2.0, min(alloc, remaining * 0.5))
        sub_deadline = min(deadline, now + alloc)

        if sub_deadline - time.monotonic() < 0.75:
            break

        a_new, sc_new = lbfgs_refine_seq(start, n, sub_deadline, best_score)
        add_candidate(pool, a_new, sc_new, 16)

        if sc_new < best_score - 1e-12:
            best_score = sc_new
            a_best = a_new.copy()

    return a_best, best_score


def get_existing_sequence():
    """Safely read height_sequence_1 if present."""
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None

    if not isinstance(seq, (list, tuple, np.ndarray)):
        return None

    try:
        a = np.asarray(seq, dtype=np.float64).reshape(-1)
    except Exception:
        return None

    if a.size == 0:
        return None

    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 0.0 or not np.isfinite(s):
        return None

    return a / s


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the minimizing nonnegative coefficient sequence.
    Returns a Python list of floats.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42

    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0

    if budget <= 0.0:
        return [1.0, 1.0, 1.0, 1.0]

    start_time = time.monotonic()

    if budget > 1200.0:
        reserve = 120.0
    elif budget > 400.0:
        reserve = 60.0
    elif budget > 100.0:
        reserve = 20.0
    else:
        reserve = 0.0

    deadline = start_time + max(0.0, budget - reserve)

    rng = np.random.default_rng(seed)

    a_prev = get_existing_sequence()
    a_best = None
    best_score = float("inf")

    if a_prev is not None:
        a_best = a_prev.copy()
        best_score = fast_score(a_best)
        print(f"existing sequence N={a_best.size} score={best_score:.8f}")

    if a_best is not None and not np.isfinite(best_score):
        a_best = None
        best_score = float("inf")

    if budget >= 1800.0:
        final_N = 32768
    elif budget >= 800.0:
        final_N = 16384
    elif budget >= 300.0:
        final_N = 8192
    elif budget >= 120.0:
        final_N = 4096
    else:
        final_N = 2048

    if a_prev is not None and a_prev.size > 0 and a_prev.size <= 16384 and budget < 1200.0:
        if a_prev.size > final_N:
            final_N = int(a_prev.size)

    pool = []
    if a_best is not None:
        add_candidate(pool, a_best, best_score, 16)

    # Mildly downscale stage budgets for short runs.
    scale = 1.0
    if budget < 1200.0:
        scale = 0.6
    if budget < 400.0:
        scale = 0.3

    if final_N > 8192:
        base_plan = [(256, 10), (512, 12), (1024, 20), (2048, 30), (4096, 40), (8192, 70)]
    elif final_N > 4096:
        base_plan = [(256, 10), (512, 12), (1024, 20), (2048, 30), (4096, 40)]
    elif final_N > 1024:
        base_plan = [(256, 8), (512, 10), (1024, 20)]
    else:
        base_plan = []

    for N, tsec in base_plan:
        if N >= final_N:
            continue
        tsec = float(tsec) * scale
        now = time.monotonic()
        if deadline - now <= tsec + 120.0:
            break

        stage_deadline = min(deadline, now + tsec)
        if N <= 1024:
            max_starts = 4
        elif N <= 4096:
            max_starts = 3
        else:
            max_starts = 2

        a_best, best_score = run_search_starts(
            N, stage_deadline, rng, a_best, best_score, a_prev, pool, max_starts
        )
        if a_best is not None:
            add_candidate(pool, a_best, best_score, 16)

    if deadline - time.monotonic() > 5.0:
        if budget >= 1500.0:
            max_starts_final = 5
        elif budget >= 600.0:
            max_starts_final = 4
        else:
            max_starts_final = 3

        a_best, best_score = run_search_starts(
            final_N, deadline, rng, a_best, best_score, a_prev, pool, max_starts_final
        )

    if a_best is None or a_best.size == 0:
        a_best = np.full(final_N, 1.0 / float(final_N), dtype=np.float64)
        best_score = fast_score(a_best)

    final_seq = np.asarray(a_best, dtype=np.float64).reshape(-1)
    final_seq = np.where(np.isfinite(final_seq), final_seq, 0.0)
    final_seq = np.maximum(final_seq, 0.0)

    s = float(np.sum(final_seq))
    if s <= 0.0 or not np.isfinite(s):
        size = max(1, final_seq.size)
        final_seq = np.full(size, 1.0 / float(size), dtype=np.float64)
    else:
        final_seq = final_seq / s

    final_score = fast_score(final_seq)
    print(f"final N={final_seq.size} fft_score={final_score:.8f} elapsed={time.monotonic()-start_time:.1f}s")

    return [float(x) for x in final_seq]
