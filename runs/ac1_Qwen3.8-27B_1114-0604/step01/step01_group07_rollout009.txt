dF/da[j] = 2*n * dc[k*]/da[j] / (sum(a))^2 - 2*F(a) * 1/sum(a)
= 2*n * 2*a[k*-j] * 1_{j in S_{k*}} / (sum(a))^2 - 2*F(a)/sum(a)

For the purpose of gradient descent (or mirror descent), we want to decrease F. So we want to decrease a[j] where the gradient is positive and increase where it's negative.

The first term is 4*n*a[k*-j]/(sum(a))^2 for j in S_{k*}, which is positive (since a >= 0).
The second term is -2*F(a)/sum(a) which is negative.

So we're pulling down the coefficients that contribute to the peak lag, and pushing up all others. This makes sense!

For the mirror descent approach:

- Find k* = argmax c
- Compute gradient: g[j] = 2*a[k*-j] (for j in S_{k*}), 0 otherwise
- Subtract the mean (to maintain sum constraint in mirror descent on the simplex): g[j] -= mean(g)
- Update: a[j] *= exp(-lr * g[j])
- Renormalize

import numpy as np

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())

    if c.size == 0:
        return np.inf

    if not np.isfinite(mx) or mx < 0.0:

    return float(2.0 * arr.size * mx / (s * s))

def _soft_max_gradient_descent(a, budget_s=60.0, beta_init=None):
    """Optimize using soft-max gradient descent on the simplex."""
    start = time.monotonic()
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())

    n = arr.size

import numpy as np
import time
import math

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    n = arr.size
    if n == 0:
        return np.inf
    s = float(arr.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    mx = float(np.max(c))
    return float(2.0 * n * mx / (s * s))

def _make_anti_palindromic(n, amps):
    """Create anti-palindromic sinusoidal sequence."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)

        a += A * np.sin(2.0 * np.pi * m * (i + 0.5) / n)
    a /= n
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / n
    return a / s

def _gradient_step(arr, c, beta, n):
    """Compute the soft-max weighted gradient."""

    exp_c = np.exp(c_scaled - c_max)

    # This is 2 * (w cross-correlated with a)[j]
    # = 2 * IFFT(conj(FFT(w)) * FFT(a))... 
    # Actually: (w ⋆ a)[j] = sum_k w[k] a[k-j]
    # In FFT: this is the circular cross-correlation
    # For linear, we need to pad
    
    N = _next_pow2(4 * n)  # enough for linear correlation
    W = np.fft.rfft(w, N)

    corr = np.fft.irfft(np.conj(W) * A, N)
    # corr[j] = sum_k w[(k-j) mod N] * a[k] (circular)
    # For the linear version with zero-padding, this should be correct
    g = 2.0 * corr[:n]
    return g, w
