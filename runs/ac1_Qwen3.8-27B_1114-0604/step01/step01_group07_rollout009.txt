Low-peak nonnegative step-function search using multi-frequency sinusoidal
parameterization followed by aggressive active-set mirror descent refinement.
"""

import numpy as np
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))

def _make_poly_sine(n, amps, phases):

    n = int(n)

    s = float(base.sum())
    if s < 0.01:
        base = np.ones(n, dtype=np.float64) / n
    else:
        base = base / s
    return base

def _mirror_descent(a, budget_s, initial_lr=None, K=16, patience=300):
    """Active-set mirror descent on the simplex to minimize max autocorrelation."""
    start = time.monotonic()
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        return arr, np.inf
    arr = arr / s
    
    best = arr.copy()
    best_score = _internal_score(best)
    n = arr.size
    
    if initial_lr is None:
        lr = 0.1 * n
    else:
        lr = initial_lr
    max_lr = 0.5 * n
    min_lr = 0.5
    
    it = 0
    last_improve = 0
    check_interval = 30
    
    while time.monotonic() - start < budget_s:
        it += 1
        
        c = _fft_convolve(arr)
        if c.size == 0:
            break
        
        maxc = float(np.max(c))
        if not np.isfinite(maxc):
            break
        
        # Find active set: top K lags
        if c.size <= K:
            active = np.arange(c.size)
        else:
            active = np.argpartition(c, -K)[-K:]
        
        # Compute subgradient
        grad = np.zeros(n, dtype=np.float64)
        for k in active:
            k = int(k)
            # c_k = sum_i a_i * a_{k-i} for valid i
            # d(c_k)/d(a_i) = a_{k-i} + a_{i-k} (for valid indices)
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            for i in range(i_min, i_max + 1):
                j1 = k - i
                j2 = i - k
                g = 0.0
                if 0 <= j1 < n:
                    g += arr[j1]
                if 0 <= j2 < n:
                    g += arr[j2]
                grad[i] += g
        
        grad /= float(len(active))
        
        # Project gradient: remove mean for simplex constraint
        g = grad - np.mean(grad)
        if np.max(np.abs(g)) < 1e-12:
            break
        
        # Multiplicative update
        exponent = -lr * g
        exponent = np.clip(exponent, -2.0, 2.0)
        arr = arr * np.exp(exponent)
        arr = np.clip(arr, 0.0, 1000.0)
        ts = float(arr.sum())
        if ts < 0.01:
            arr = best.copy()
            lr = max(min_lr, lr * 0.5)
            continue
        arr = arr / ts
        
        # Periodic check
        if it % check_interval == 0:
            sc = _internal_score(arr)
            if np.isfinite(sc) and sc < best_score:
                best = arr.copy()
                best_score = sc
                last_improve = it
                lr = min(max_lr, lr * 1.08)
            else:
                if sc > best_score * 1.01:
                    arr = best.copy()
                lr = max(min_lr, lr * 0.85)
            
            if it - last_improve > patience:
                # Restart with smaller lr
                arr = best.copy()
                lr = max(min_lr, lr * 0.5)
                patience = max(100, patience // 2)
                if lr < min_lr:
                    break
    
    return best, best_score

"""
Low-peak nonnegative step-function search using multi-harmonic sinusoidal
parameterization followed by aggressive active-set mirror descent with
vectorized subgradient computation on the simplex.
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))

            if s < 0.01:
                continue
            shape = shape / s
            sc = _internal_score(shape)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_arr = shape.copy()

    return best_arr, best_score


def _mirror_descent(a, budget_s, K=20, initial_lr=None):
    """Aggressive active-set mirror descent on the simplex."""
    start = time.monotonic()
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        return arr, np.inf
    arr = arr / s

    best = arr.copy()
    best_score = _internal_score(best)
    n = arr.size

    if initial_lr is None:
        lr = 0.3 * n
    else:
        lr = initial_lr
    max_lr = 0.6 * n
    min_lr = 0.5

    it = 0
    last_improve = 0
    check_interval = 20
    patience = 400

    # Precompute index arrays for speed
    while time.monotonic() - start < budget_s:
        it += 1

        c = _fft_convolve(arr)
        if c.size == 0:
            break
        maxc = float(np.max(c))
        if not np.isfinite(maxc):
            break

        # Find active set: top K lags
        if c.size <= K:
            active = np.arange(c.size, dtype=np.int64)
        else:
            active = np.argpartition(c, -K)[-K:]

        # Compute subgradient (vectorized per lag)
        grad = _compute_subgradient(arr, active)
        grad /= float(len(active))

        # Remove mean for simplex constraint
        g = grad - np.mean(grad)
        gmax = np.max(np.abs(g))
        if gmax < 1e-14:
            break

        # Normalize gradient for stable updates
        g = g / (gmax + 1e-30)

        # Multiplicative update with clipped exponent
        exponent = -lr * 0.01 * g
        exponent = np.clip(exponent, -1.5, 1.5)
        arr_new = arr * np.exp(exponent)
        arr_new = np.clip(arr_new, 0.0, 1000.0)
        ts = float(arr_new.sum())
        if not np.isfinite(ts) or ts < 0.01:
            arr = best.copy()
            lr = max(min_lr, lr * 0.5)
            continue
        arr = arr_new / ts

        # Periodic score check
        if it % check_interval == 0:
            sc = _internal_score(arr)
            if np.isfinite(sc) and sc < best_score:
                best = arr.copy()
                best_score = sc
                last_improve = it
                lr = min(max_lr, lr * 1.05)
            else:
                if sc > best_score * 1.005:
                    arr = best.copy()
                lr = max(min_lr, lr * 0.92)

            if it - last_improve > patience:
                arr = best.copy()
                lr = max(min_lr, lr * 0.4)
                patience = max(150, patience // 2)
                if lr <= min_lr * 1.1:
                    break

    return best, best_score

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t0 = time.monotonic()
    hard_deadline = t0 + max(60.0, float(budget_s)) - 30.0

    # Phase 1: Find best sinusoidal starting point across lengths
    lengths = [2048, 4096, 1024]
    best_a = None
    best_score = np.inf

    scan_deadline = t0 + 90.0
    for n in lengths:
        if time.monotonic() > scan_deadline:
            break
        dl = min(scan_deadline, time.monotonic() + 30.0)
        amp, sc = _optimize_sine_1d(n, dl)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = _make_poly_sine(n, [amp], [0.0])
            print(f"[scan] n={n} amp={amp:.6f} score={sc:.8f}")

    if best_a is None:
        n = 2048
        best_a = _make_poly_sine(n, [0.78], [0.0])
        best_score = _internal_score(best_a)
        print(f"[init] n=2048 amp=0.78 score={best_score:.8f}")

    # Try multi-harmonic improvement on the best
    n_best = best_a.size
    if time.monotonic() < hard_deadline - 60:
        mh_deadline = min(hard_deadline - 60, time.monotonic() + 40)
        mh_a, mh_score = _try_multi_harmonic(n_best, 0.78, mh_deadline)
        if mh_score < best_score:
            best_a = mh_a
            best_score = mh_score
            print(f"[multi-harm] improved to {best_score:.8f}")

    print(f"[phase1] done, best={best_score:.8f}, t={time.monotonic()-t0:.1f}s")

    # Also check global parent
    parent_score = np.inf
    parent_a = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64)
            p = np.clip(p, 0.0, 1000.0)
            ps = float(p.sum())
            if ps > 0.01:
                p = p / ps
                parent_score = _internal_score(p)
                parent_a = p
                if parent_score < best_score:
                    print(f"[parent] better start: {parent_score:.8f}")
                    best_a = p.copy()
                    best_score = parent_score
    except Exception:
        pass

    # Phase 2: First mirror descent pass (aggressive)
    remaining = hard_deadline - time.monotonic()
    if remaining > 60:
        budget1 = min(remaining - 60, 800.0)
        ra, rs = _mirror_descent(best_a, budget1, K=20, initial_lr=0.3 * n_best)
        if rs < best_score:
            best_a = ra
            best_score = rs
            print(f"[md1] improved to {best_score:.8f}, t={time.monotonic()-t0:.1f}s")
        else:
            print(f"[md1] no improvement, t={time.monotonic()-t0:.1f}s")

    # Phase 3: Second mirror descent pass (different schedule, from best)
    remaining = hard_deadline - time.monotonic()
    if remaining > 60:
        budget2 = min(remaining - 60, 1000.0)
        ra, rs = _mirror_descent_v2(best_a, budget2, K=24, initial_lr=0.15 * n_best)
        if rs < best_score:
            best_a = ra
            best_score = rs
            print(f"[md2] improved to {best_score:.8f}, t={time.monotonic()-t0:.1f}s")
        else:
            print(f"[md2] no improvement, t={time.monotonic()-t0:.1f}s")

    # Phase 4: If we have time, try from a different starting point
    remaining = hard_deadline - time.monotonic()
    if remaining > 120 and parent_a is not None and parent_score < 1.55:
        # Try refining the parent separately
        budget3 = min(remaining - 30, 400.0)
        if budget3 > 30:
            ra, rs = _mirror_descent(parent_a, budget3, K=20, initial_lr=0.25 * parent_a.size)
            if rs < best_score:
                best_a = ra
                best_score = rs
                print(f"[md3] improved to {best_score:.8f}, t={time.monotonic()-t0:.1f}s")

    # Phase 5: Final fine-tuning if time remains
    remaining = hard_deadline - time.monotonic()
    if remaining > 30:
        budget4 = remaining - 10
        if budget4 > 10:
            ra, rs = _mirror_descent(best_a, budget4, K=16, initial_lr=0.05 * n_best)
            if rs < best_score:
                best_a = ra
                best_score = rs
                print(f"[md4] improved to {best_score:.8f}, t={time.monotonic()-t0:.1f}s")

    # Final validation
    arr = np.asarray(best_a, dtype=np.float64)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        arr = np.ones(256, dtype=np.float64) / 256.0
    else:
        arr = arr / s

    print(f"[final] internal_score={_internal_score(arr):.8f}, n={arr.size}")

    try:
        real_score = float(evaluate_sequence(arr.tolist()))
        print(f"[final] evaluate_sequence={real_score:.8f}")
    except Exception as e:
        print(f"[final] error: {e}")

    return [float(x) for x in arr]
