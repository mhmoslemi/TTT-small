        if s < 1e-10:
            continue
        f0 = f0 / s
        sc = _score(f0)
        sc_official = _eval(f0)
        print(f"  {name}: internal={sc:.6f}, official={sc_official:.6f}")
        if sc_official < best_score:
            best_score = sc_official
            best_f = f0.copy()

    # Phase 2: Randomized hill climbing from best start
    print("\nPhase 2: Hill climbing exploration...")
    if time.monotonic() < deadline - 60:
        hc_deadline = time.monotonic() + min(60.0, deadline - time.monotonic())
        f_scaled = best_f.copy()
        s = f_scaled.sum()
        if s > 1e-10:
            f_scaled = f_scaled / s

        f_hc, sc_hc = _hill_climb(f_scaled, n, 50000, rng, hc_deadline)
        sc_hc_official = _eval(f_hc)
        print(f"  Hill climb result: {sc_hc_official:.6f}")
        if sc_hc_official < best_score:
            best_score = sc_hc_official
            best_f = f_hc.copy()

    # Phase 3: Try from the provided sequence
    if time.monotonic() < deadline - 30:
        print("\nPhase 3: Optimizing from provided sequence...")
        opt_deadline = time.monotonic() + min(30.0, deadline - time.monotonic())
        f_opt, sc_opt = _hill_climb(f_init_norm, n, 20000, rng, opt_deadline)
        sc_opt_official = _eval(f_opt)
        print(f"  Optimization from initial: {sc_opt_official:.6f}")
        if sc_opt_official < best_score:
            best_score = sc_opt_official
            best_f = f_opt.copy()

    # Phase 4: Mirror descent with gradient from autocorrelation
    if time.monotonic() < deadline - 20:
        print("\nPhase 4: Gradient-based optimization...")
        f = best_f.copy()
        s = f.sum()
        if s > 1e-10:
            f = f / s

        mirror_deadline = time.monotonic() + min(20.0, deadline - time.monotonic())
        lr = 0.1
        for it in range(20000):
            if time.monotonic() > mirror_deadline:
                break
            c = _autocorr_fft(f)
            peak_idx = np.argmax(c)
            # Gradient of max(c) w.r.t. f: if peak at lag k, grad = 2*f shifted by k
            # (for the peak position)
            grad = np.zeros(n)
            k = peak_idx
            for j in range(n):
                if k - j >= 0 and k - j < n:
                    grad[j] += 2.0 * f[k - j]
                if j - k >= 0 and j - k < n:
                    grad[j] += 2.0 * f[j - k]

            # Normalize gradient
            g_norm = np.linalg.norm(grad)
            if g_norm > 1e-15:
                grad = grad / g_norm

            f = f - lr * grad
            f = np.maximum(f, 0.0)
            s = f.sum()
            if s < 1e-10:
                f = np.ones(n) / n
            else:
                f = f / s

            if it % 5000 == 0:
                sc = _score(f)
                if sc < best_score:
                    sc_off = _eval(f)
                    print(f"    Mirror iter {it}: {sc_off:.6f}")
                    if sc_off < best_score:
                        best_score = sc_off
                        best_f = f.copy()

            # Anneal learning rate
            lr *= 0.9995

    # Final verification
    final_score = _eval(np.clip(best_f, 0.0, 1000.0).tolist())
    print(f"\n=== FINAL SCORE: {final_score:.6f} ===")

    # Return best sequence
    best_f = np.clip(best_f, 0.0, 1000.0)
    s_final = best_f.sum()
    if s_final < 0.01:
        best_f = np.ones(n)

    return [float(x) for x in best_f]
