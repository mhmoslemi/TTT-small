def fft_convolution(a):
    """Linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    """Smooth max via log-sum-exp and softmax weights."""
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float("inf"), np.empty(0, dtype=np.float64)
    m = float(np.max(b))
    if not np.isfinite(m):
        return float("inf"), np.zeros_like(b, dtype=np.float64)
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b, dtype=np.float64)
        w /= float(b.size)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -700.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w


def fast_score(p):
    """Fast score using FFT."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float("inf")
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):

    M = float(np.max(b))
    if not np.isfinite(M):

    return float(2.0 * n * M / (S * S))


def corr_pw(p, w):
    """s_i = sum_j p_j * w_{i+j} via FFT."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    if n == 1:
        return np.array([float(np.dot(p, w[:1]))], dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n: m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)


def project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    if target <= 0.0:
        return np.zeros_like(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(v.shape, target / float(v.size), dtype=np.float64)
    low = float(np.min(v)) - 1.0
    high = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (low + high)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        q = np.full(v.shape, target / float(v.size), dtype=np.float64)
    else:
        q *= target / s
    return q


def normalize_simplex(p):
    """Clip and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size), dtype=np.float64)
    p *= 1.0 / s
    return p


def nesterov_stage(p, deadline, beta, momentum=0.9, max_iter=500000,
                   anneal_period=3000, anneal_scale=0.005, rng=None):
    """
    Nesterov momentum with projected gradient and simulated annealing.
    Returns (best_p, best_score, iterations_run).
    """
    if rng is None:
        rng = np.random.default_rng(0)

    p = normalize_simplex(p)
    n = p.size
    v = np.zeros(n, dtype=np.float64)
    best_p = p.copy()
    b = fft_convolution(p)
    best_score = float(2.0 * n * np.max(b))
    step = 1.0 / max(1.0, beta * 0.01)
    it = 0
    no_improve = 0

    while it < max_iter and time.monotonic() < deadline:
        it += 1

        # Nesterov lookahead
        p_look = p + momentum * v
        p_look = project_simplex(p_look, 1.0)

        # Gradient at lookahead point
        b = fft_convolution(p_look)
        G, w = smooth_max_and_weights(b, beta)
        if not np.isfinite(G):
            break

        svec = corr_pw(p_look, w)
        svec = np.maximum(svec, 0.0)
        g = 2.0 * svec
        grad = g - float(np.mean(g))

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-16:
            break

        # Momentum update
        v = momentum * v - step * grad
        v = np.clip(v, -1.0, 1.0)  # Prevent divergence

        p_next = project_simplex(p + v, 1.0)

        # Evaluate
        b_next = fft_convolution(p_next)
        M_next = float(np.max(b_next))
        score_next = float(2.0 * n * M_next)

        if not np.isfinite(score_next):
            p = p_next
            v = 0.5 * v
            continue

        # Accept if better (or with annealing probability)
        if score_next < best_score:
            best_score = score_next
            best_p = p_next.copy()
            no_improve = 0
            p = p_next
        elif no_improve > 0 and it % anneal_period == 0:
            # Annealing: accept with probability based on "temperature"
            progress = it / max(1, max_iter)
            temp = anneal_scale * (1.0 - progress)
            if rng.random() < math.exp(-min(50.0, (score_next - best_score) / max(1e-12, temp))):
                p = p_next
                best_p = p_next.copy()
                best_score = score_next
                no_improve = 0
            else:
                p = p_next
        else:
            p = p_next
            no_improve += 1

        # Adapt step size
        if it % 500 == 0:
            if no_improve > 2000:
                step *= 0.7
            elif no_improve < 200:
                step = min(step * 1.05, 2.0)
            if step < 1e-8:
                break

        if it % 2000 == 0:
            print(f"  nest stage beta={beta:.0f} it={it} score={best_score:.8f} step={step:.3e} no_imp={no_improve}")

        if no_improve > 15000:
            break

    return best_p, best_score, it


def lp_update(f, M, solver=None):
    """
    LP-based direction: max sum(g) s.t. (f*g)[k] <= M, g >= 0.
    Returns g or None if LP fails.
    """
    n = f.size
    if n > 4096:
        return None  # Too large for LP

    try:
        g = cp.Variable(n, nonneg=True)
        constraints = []

        # Build constraints using the banded structure
        # (f*g)[k] = sum_j f[k-j]*g[j] for valid j
        for k in range(2 * n - 1):
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            if j_min > j_max:
                continue
            # f[k-j] for j in [j_min, j_max]
            idx = np.arange(j_min, j_max + 1)
            f_vals = f[k - idx]
            # dot(f_vals, g[idx]) <= M
            constraints.append(f_vals @ g[idx] <= M)

        prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

        if solver is not None:
            prob.solve(solver=solver)
        else:
            # Try CLARABEL first, fall back to others
            try:
                prob.solve(solver=cp.CLARABEL)
            except Exception:
                try:
                    prob.solve(solver=cp.ECOS)
                except Exception:
                    prob.solve(solver=cp.SCS, max_iter=5000)

        if g.value is not None and np.all(np.isfinite(g.value)):
            g_val = np.maximum(g.value, 0.0)
            s = float(np.sum(g_val))
            if s > 1e-10:
                return g_val / s
    except Exception:
        pass
    return None


def lp_iteration(f, t=0.3):
    """One LP-based iteration: find direction, blend, return new f."""
    n = f.size
    b = fft_convolution(f)
    M = float(np.max(b))
    if not np.isfinite(M) or M <= 0:
        return f

    g = lp_update(f, M)
    if g is None:
        return f

    g = normalize_simplex(g)
    new_f = (1.0 - t) * f + t * g
    new_f = normalize_simplex(new_f)
    return new_f


def generate_diverse_starts(n, rng, base=None, count=6):
    """Generate diverse starting sequences."""
    starts = []

    if base is not None:
        starts.append(normalize_simplex(base.copy()))

    # Uniform
    starts.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Linear ramps
    x = np.linspace(0, 1, n, dtype=np.float64)
    for a in (-2.0, 2.0, -3.0, 3.0):
        p = np.exp(a * (x - 0.5))
        p = np.maximum(p, 1e-10)
        starts.append(normalize_simplex(p))

    # Two-level
    p = np.ones(n, dtype=np.float64)
    half = n // 2
    p[:half] = 1.5
    p[half:] = 0.7
    starts.append(normalize_simplex(p))

    # Triangular
    p = np.abs(1.0 - 2.0 * x)
    p = np.maximum(p, 1e-10)
    starts.append(normalize_simplex(p))

    # Gaussian bump
    p = np.exp(-5.0 * (x - 0.5) ** 2)
    p = np.maximum(p, 1e-10)
    starts.append(normalize_simplex(p))

    # Random smooth
    for _ in range(2):
        z = rng.normal(size=min(16, n))
        # Upsample via interpolation
        if n > 16:
            z_full = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, 16), z)
        else:
            z_full = z[:n]
        p = np.exp(1.0 * z_full)
        p = np.maximum(p, 1e-10)
        starts.append(normalize_simplex(p))

    return starts[:count]


def official_score(seq):
    """Call the authoritative evaluator."""
    try:
        val = evaluate_sequence(seq)
        val = float(val)
    except Exception:
        return float("inf")
    return val if np.isfinite(val) else float("inf")


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-stage optimization:
    1. Load previous best, evaluate
    2. LP-based iterations for diverse candidates
    3. Multi-stage Nesterov momentum with annealing
    4. Return best sequence found
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 10.0:
        return [1.0]

    start_time = time.monotonic()
    reserve = 60.0
    deadline = start_time + max(10.0, budget_s - reserve)

    rng = np.random.default_rng(seed)

    # Choose length
    if budget_s < 300.0:
        N = 4096
    elif budget_s < 800.0:
        N = 6144
    else:
        N = 8192

    print(f"propose_candidate: N={N}, budget={budget_s:.0f}s")

    # Load previous best if available
    base = None
    try:
        if 'height_sequence_1' in globals():
            hs1 = np.asarray(height_sequence_1, dtype=np.float64)
            if hs1.size > 0 and np.all(np.isfinite(hs1)) and np.all(hs1 >= 0):
                base = normalize_simplex(hs1)
                print(f"Loaded height_sequence_1 of length {base.size}")
    except Exception:
        pass

    # If base has different length, interpolate or resize
    if base is not None and base.size != N:
        print(f"Resizing base from {base.size} to {N}")
        x_old = np.linspace(0, 1, base.size)
        x_new = np.linspace(0, 1, N)
        base = np.interp(x_new, x_old, base)
        base = normalize_simplex(base)

    # Generate diverse starts
    starts = generate_diverse_starts(N, rng, base=base, count=8)

    # Evaluate all starts
    scored_starts = []
    for i, s in enumerate(starts):
        sc = fast_score(s)
        if np.isfinite(sc):
            scored_starts.append((sc, s))
            print(f"  start {i}: score={sc:.8f}")

    scored_starts.sort(key=lambda x: x[0])
    best_score = scored_starts[0][0] if scored_starts else 2.0
    best_p = scored_starts[0][1].copy() if scored_starts else np.full(N, 1.0/N)

    # ---- Phase 1: LP-based iterations (if available and time permits) ----
    lp_deadline = min(deadline, start_time + max(30.0, 0.08 * budget_s))
    if HAVE_CVXPY and time.monotonic() < lp_deadline:
        print("Phase 1: LP-based iterations")
        for start_i, (sc, s) in enumerate(scored_starts[:3]):
            if time.monotonic() >= lp_deadline:
                break
            try:
                for iter_i in range(5):
                    if time.monotonic() >= lp_deadline:
                        break
                    s_new = lp_iteration(s, t=0.3)
                    sc_new = fast_score(s_new)
                    if np.isfinite(sc_new) and sc_new < sc:
                        s = s_new
                        sc = sc_new
                        print(f"    LP start {start_i} iter {iter_i}: {sc:.8f}")
                    else:
                        break
                if sc < best_score:
                    best_score = sc
                    best_p = s.copy()
                    print(f"    LP improved best to {best_score:.8f}")
            except Exception as e:
                print(f"    LP error: {e}")
                break

    # ---- Phase 2: Multi-stage Nesterov optimization ----
    print(f"Phase 2: Nesterov optimization (best so far: {best_score:.8f})")

    # Select top starts for Nesterov
    top_starts = [best_p]
    for sc, s in scored_starts:
        if len(top_starts) >= 4:
            break
        if s is not best_p and sc < best_score * 1.05:
            top_starts.append(s.copy())

    remaining = deadline - time.monotonic()
    if remaining < 30.0:
        print("Not enough time for Nesterov, returning best")
    else:
        # Allocate time across starts
        n_starts = len(top_starts)
        time_per_start = remaining / max(1, n_starts) * 0.9  # leave 10% buffer

        # Beta schedule: multiple stages
        betas = [
            (5000.0, 0.30),    # smooth, 30% of time for this start
            (50000.0, 0.40),   # medium, 40%
            (500000.0, 0.30),  # sharp, 30%
        ]

        for si, p0 in enumerate(top_starts):
            if time.monotonic() >= deadline - 10.0:
                break

            alloc = time_per_start
            print(f"  Start {si}: allocating {alloc:.0f}s")

            # Sub-stages with different beta
            current_p = normalize_simplex(p0.copy())
            current_score = fast_score(current_p)

            for beta_frac, time_frac in betas:
                sub_deadline = time.monotonic() + alloc * time_frac
                if time.monotonic() >= sub_deadline:
                    break

                momentum = 0.9 if beta_frac < 50000 else 0.85
                anneal_scale = 0.01 if beta_frac < 50000 else 0.003
                anneal_period = 2000 if beta_frac < 50000 else 5000

                p_new, sc_new, iters = nesterov_stage(
                    current_p, sub_deadline, beta_frac,
                    momentum=momentum,
                    anneal_period=anneal_period,
                    anneal_scale=anneal_scale,
                    rng=rng
                )

                if np.isfinite(sc_new) and sc_new < current_score:
                    current_score = sc_new
                    current_p = p_new.copy()

                if np.isfinite(sc_new) and sc_new < best_score:
                    best_score = sc_new
                    best_p = p_new.copy()
                    print(f"  ** New best: {best_score:.8f} (start {si}, beta={beta_frac:.0f})")

            # After finishing all sub-stages for this start,
            # if this start's result is better, update
            if np.isfinite(current_score) and current_score < best_score:
                best_score = current_score
                best_p = current_p.copy()

    # ---- Phase 3: Final polish with very sharp beta ----
    if time.monotonic() < deadline - 15.0 and np.isfinite(best_score) and best_score < 2.0:
        print(f"Phase 3: Final polish (score={best_score:.8f})")
        polish_deadline = min(deadline - 5.0, time.monotonic() + max(10.0, 0.05 * budget_s))
        p_final, sc_final, _ = nesterov_stage(
            best_p, polish_deadline, beta=2000000.0,
            momentum=0.99, anneal_period=10000,
            anneal_scale=0.001, rng=rng
        )
        if np.isfinite(sc_final) and sc_final < best_score:
            best_score = sc_final
            best_p = p_final.copy()
            print(f"  Polish improved to {best_score:.8f}")

    # ---- Final cleanup ----
    final_p = normalize_simplex(best_p)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_seq = [float(x) for x in final_p]

    # Verify with official evaluator
    off = official_score(final_seq)
    print(f"Official score: {off:.10f}")

    # Fallback
    if not np.isfinite(off) or off > 2.0 + 1e-9:
        print("Falling back to uniform")
        return [1.0] * N

    total_time = time.monotonic() - start_time
    print(f"Done in {total_time:.1f}s, final score={off:.10f}")
    return final_seq
