    p = p.copy()
    best_p = p.copy()
    best_score = evaluate_sequence(p.tolist())
    
    for t in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = np.convolve(p, p)
        max_c = c.max()
        
        # Find active set
        if top_k >= len(c):
            K = set(range(len(c)))
        else:
            K = set(np.argsort(c)[::-1][:top_k].tolist())
        
        # Compute subgradient
        h = np.zeros(len(c))
        for k in K:
            h[k] = 1.0
        g = 2.0 * np.convolve(h, p, 'valid')
        
        # Mirror descent step
        lr = lr_schedule(t)
        log_p = np.log(p + 1e-300) - lr * g
        log_p -= log_p.max()
        p_new = np.exp(log_p)
        p_new /= p_new.sum()
        p_new = np.maximum(p_new, 0)
        p_new /= p_new.sum()
        
        p = p_new
        
        # Evaluate periodically
        if t % 100 == 0 or t == max_iters - 1:
            score = evaluate_sequence(p.tolist())
            if score < best_score:
                best_score = score
                best_p = p.copy()
    
    return best_p, best_score

Now, the question is: what learning rate schedule and top_k to use?

For mirror descent on the simplex, a typical learning rate is lr = O(1/sqrt(t)) or lr = O(1/t). Since the function is not convex, we might need to be more careful.

Let me also think about the "top_k" parameter. If we use top_k=1, we get the steepest descent direction (targeting only the single peak). If we use a larger top_k, we get a smoother gradient that targets multiple peaks, which might help with convergence but might be less aggressive.

I think a good strategy is:
- Start with a moderate top_k (e.g., n//5) to get a coarse optimization.
- Gradually reduce top_k to 1 for fine-tuning.
- Use a learning rate that starts moderate and decreases.

Let me also add:
- Multiple restarts from different points
- Periodic "flattening" perturbations
- A final polish phase

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(f,f))/(sum f)^2.
Uses subgradient mirror descent with exact subgradient computation,
adaptive top-K peak targeting, and multiple restarts.
"""

import numpy as np
import time


def _compute_subgradient(p, top_k):
    """Compute exact subgradient of max of top_k convolution values."""
    n = len(p)
    c = np.convolve(p, p)
    if top_k >= len(c):
        idxs = np.arange(len(c))
    else:
        idxs = np.argsort(c)[::-1][:top_k]
    
    h = np.zeros(len(c))
    h[idxs] = 1.0
    g = 2.0 * np.convolve(h, p, 'valid')
    return g, c, idxs


def _mirror_step(p, g, lr):
    """Mirror descent step on the probability simplex."""
    log_p = np.log(p + 1e-300)
    log_p -= lr * g
    log_p -= log_p.max()
    p_new = np.exp(log_p)
    s = p_new.sum()
    if s < 1e-300:
        return p
    p_new /= s
    return p_new


def _eval_normalized(p):
    """Evaluate the normalized sequence."""
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 0.01:
        return np.inf
    return evaluate_sequence(p.tolist())


def _optimize_from(p0, deadline, rng, label=""):
    """Optimize from starting point p0 using subgradient mirror descent."""
    n = len(p0)
    p = np.maximum(p0, 0).astype(np.float64).copy()
    s = p.sum()
    if s < 1e-10:
        p = np.ones(n) / n
    else:
        p = p / s
    
    best_p = p.copy()
    best_score = _eval_normalized(p)
    
    max_iters = 50000
    # Adaptive top_k: start large, decrease over time
    top_k_initial = max(1, n // 3)
    top_k_final = max(1, n // 50)
    
    # Learning rate schedule
    lr_initial = 1.0
    lr_final = 0.01
    
    stuck_count = 0
    improvement_threshold = 1e-7
    
    for t in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        progress = t / max_iters
        
        # Anneal top_k
        top_k = max(top_k_final, int(top_k_initial * (1 - progress) + top_k_final * progress))
        # Anneal learning rate
        lr = lr_initial * (lr_final / lr_initial) ** (progress ** 0.5)
        
        g, c, idxs = _compute_subgradient(p, top_k)
        
        # Mirror descent step
        p = _mirror_step(p, g, lr)
        
        # Evaluate periodically
        if t % 200 == 0 or t == max_iters - 1:
            score = _eval_normalized(p)
            if score < best_score - improvement_threshold:
                best_score = score
                best_p = p.copy()
                stuck_count = 0
                if t % 2000 == 0:
                    print(f"    {label} iter {t}: {score:.6f} (top_k={top_k}, lr={lr:.4f})")
            else:
                stuck_count += 1
            
            if stuck_count > 5:
                # Perturbation
                perturbation = 0.01 * rng.standard_normal(n)
                p = p + perturbation
                p = np.maximum(p, 0)
                s = p.sum()
                if s > 1e-10:
                    p = p / s
                stuck_count = 0
    
    score = _eval_normalized(p)
    if score < best_score:
        best_score = score
        best_p = p.copy()
    
    return best_p, best_score


def _polish(p0, deadline, rng):
    """Coordinate descent polish: try to reduce peaks by moving mass."""
    n = len(p0)
    p = np.maximum(p0, 0).astype(np.float64).copy()
    s = p.sum()
    if s < 1e-10:
        p = np.ones(n) / n
    else:
        p = p / s
    
    best_p = p.copy()
    best_score = _eval_normalized(p)
    
    # Focus on top 1 peak
    max_rounds = 10000
    
    for t in range(max_rounds):
        if time.monotonic() > deadline:
            break
        
        c = np.convolve(p, p)
        k_peak = int(np.argmax(c))
        peak_val = c[k_peak]
        
        # Subgradient targeting only the peak
        h = np.zeros(len(c))
        h[k_peak] = 1.0
        g = 2.0 * np.convolve(h, p, 'valid')
        
        # Try different step sizes
        best_local = None
        for lr in [0.0001, 0.001, 0.005, 0.01, 0.05]:
            p_try = _mirror_step(p, g, lr)
            score = _eval_normalized(p_try)
            if score < best_score - 1e-9:
                best_local = (p_try, score)
                break
        
        if best_local is not None:
            p, best_score = best_local
            best_p = p.copy()
        else:
            # Try random perturbations
            for _ in range(5):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                p_try = p.copy()
                amt = p_try[i] * 0.05
                p_try[i] -= amt
                p_try[j] += amt
                if p_try[i] < 0:
                    continue
                s = p_try.sum()
                if s > 1e-10:
                    p_try = p_try / s
                score = _eval_normalized(p_try)
                if score < best_score - 1e-9:
                    p = p_try
                    best_score = score
                    best_p = p.copy()
                    break
    
    return best_p, best_score

"""
Optimizes non-negative sequence to minimize 2*n*max(conv(f,f))/(sum f)^2.
Uses exact subgradient mirror descent with multi-scale top-K peak targeting,
adaptive learning rate, diverse restarts, and targeted peak-reduction polish.
"""

import numpy as np
import time


def _normalize(p):
    """Normalize to probability vector (non-negative, sum=1)."""
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return np.ones(len(p)) / len(p)
    return p / s


def _eval(p):
    """Evaluate the sequence using the provided evaluator."""
    return evaluate_sequence(p.tolist())


def _convolve_p(p):
    """Compute full linear convolution of p with itself."""
    return np.convolve(p, p)


def _subgradient(p, top_k=1):
    """
    Compute exact subgradient of max of top_k convolution values.
    Returns (gradient, max_convolution_value).
    """
    n = len(p)
    c = np.convolve(p, p)
    max_val = c.max()
    if top_k >= len(c):
        idxs = np.arange(len(c))
    else:
        idxs = np.argsort(c)[::-1][:top_k]
    h = np.zeros(len(c))
    h[idxs] = 1.0
    g = 2.0 * np.convolve(h, p, 'valid')
    return g, max_val


def _mirror_step(p, g, lr):
    """Mirror descent (exponentiated gradient) step on the simplex."""
    log_p = np.log(p + 1e-300) - lr * g
    log_p -= log_p.max()
    new = np.exp(log_p)
    s = new.sum()
    if s < 1e-300:
        return p
    new /= s
    return np.maximum(new, 0.0)


def _optimize_loop(p0, deadline, rng, label=""):
    """
    Multi-scale subgradient mirror descent with adaptive restarts.
    Phases: coarse (top-K large) -> medium -> fine -> peak targeting.
    """
    n = len(p0)
    p = _normalize(p0).astype(np.float64).copy()
    best_p = p.copy()
    best_score = _eval(p)
    best_peak = np.convolve(p, p).max()

    total_iters = 100000
    stuck = 0
    threshold = 1e-9

    for t in range(total_iters):
        if time.monotonic() > deadline:
            break

        progress = t / total_iters

        # Multi-scale top-K schedule with smooth transitions
        if progress < 0.15:
            top_k = max(10, n // 5)
            lr = 1.0 * (0.4 ** (progress / 0.15))
        elif progress < 0.35:
            top_k = max(5, n // 15)
            lr = 0.4 * (0.3 ** ((progress - 0.15) / 0.20))
        elif progress < 0.60:
            top_k = max(3, n // 50)
            lr = 0.12 * (0.3 ** ((progress - 0.35) / 0.25))
        elif progress < 0.80:
            top_k = max(2, n // 200)
            lr = 0.04 * (0.5 ** ((progress - 0.60) / 0.20))
        else:
            top_k = 1
            lr = 0.02 * (0.5 ** ((progress - 0.80) / 0.20))

        # Compute exact subgradient
        g, peak = _subgradient(p, top_k)

        # Mirror descent step
        p = _mirror_step(p, g, lr)

        # Evaluate periodically (every 300 iters or at end)
        if t % 300 == 0 or t == total_iters - 1:
            score = _eval(p)
            if score < best_score - threshold:
                best_score = score
                best_p = p.copy()
                best_peak = peak
                stuck = 0
                if t % 3000 == 0:
                    print(f"    {label} t={t} topk={top_k} lr={lr:.4f} score={score:.6f} peak_ratio={peak*2*n:.6f}")
            else:
                stuck += 1

            # Restart with perturbation if stuck
            if stuck > 15:
                intensity = 0.003 + 0.007 * rng.random()
                p = p + intensity * rng.standard_normal(n)
                p = _normalize(p)
                stuck = 0

    # Final evaluation
    score = _eval(p)
    if score < best_score:
        best_score = score
        best_p = p.copy()

    return best_p, best_score


def _peak_reduction(p0, deadline, rng):
    """
    Targeted peak reduction: identify the peak lag and move mass
    from high-contributing positions to low-contributing ones.
    """
    n = len(p0)
    p = _normalize(p0).astype(np.float64).copy()
    best_p = p.copy()
    best_score = _eval(p)

    for t in range(200000):
        if time.monotonic() > deadline:
            break

        c = np.convolve(p, p)
        k_peak = int(np.argmax(c))
        peak_val = c[k_peak]

        # Contribution of each position i to the peak:
        # (p*p)[k_peak] = sum_i p[i] * p[k_peak - i]
        # The "effective" contribution of p[i] is p[i] * p[k_peak - i]
        # (plus p[k_peak-i] * p[i] if k_peak-i != i)
        contrib = np.zeros(n)
        for i in range(n):
            j = k_peak - i
            if 0 <= j < n:
                contrib[i] = p[i] * p[j]
                if i != j:
                    contrib[i] += p[i] * p[j]  # double count for the symmetric term
        # Actually the gradient at position i for the peak is 2*p[k_peak-i]
        # So the "cost" of having mass at position i is proportional to p[k_peak-i]

        # Strategy: move mass from positions where p[k_peak-i] is large (high cost)
        # to positions where p[k_peak-i] is small (low cost)
        cost = np.zeros(n)
        for i in range(n):
            j = k_peak - i
            if 0 <= j < n:
                cost[i] = p[j]
            else:
                cost[i] = 0.0

        # Find best donor (high cost, has mass) and best receiver (low cost)
        # Only consider positions that currently have non-trivial mass
        mass_mask = p > 1e-10
        if not mass_mask.any():
            break

        # Donor: highest cost among positions with mass
        donor_costs = np.where(mass_mask, cost, -1)
        i_donate = int(np.argmax(donor_costs))
        # Receiver: lowest cost (preferably with low mass to avoid creating new peaks)
        receiver_costs = cost.copy()
        i_receive = int(np.argmin(receiver_costs))

        if i_donate == i_receive:
            # Try second best receiver
            receiver_costs[i_donate] = np.inf
            i_receive = int(np.argmin(receiver_costs))

        improved = False
        for frac in [0.002, 0.005, 0.01, 0.02, 0.05, 0.1]:
            amt = p[i_donate] * frac
            if amt <= 0:
                continue
            p_try = p.copy()
            p_try[i_donate] -= amt
            p_try[i_receive] += amt
            if p_try[i_donate] < -1e-15:
                continue
            p_try = _normalize(p_try)
            c_try = np.convolve(p_try, p_try)
            if c_try.max() < peak_val * (1 - 1e-10):
                p = p_try
                best_p = p.copy()
                best_score = _eval(p)
                improved = True
                break

        if not improved:
            # Try a few random transfers
            did_improve = False
            for _ in range(8):
                i = int(rng.integers(0, n))
                jx = int(rng.integers(0, n))
                if i == jx:
                    continue
                frac = 0.005 + 0.05 * rng.random()
                amt = p[i] * frac
                if amt <= 0:
                    continue
                p_try = p.copy()
                p_try[i] -= amt
                p_try[jx] += amt
                if p_try[i] < -1e-15:
                    continue
                p_try = _normalize(p_try)
                c_try = np.convolve(p_try, p_try)
                if c_try.max() < peak_val * (1 - 1e-10):
                    p = p_try
                    best_p = p.copy()
                    best_score = _eval(p)
                    improved = True
                    did_improve = True
                    break

            # If nothing worked, add noise to escape
            if not did_improve and t % 5000 == 0 and t > 0:
                p = p + 0.003 * rng.standard_normal(n)
                p = _normalize(p)

    return best_p, best_score


def _multi_restart_optimize(p0, deadline, rng, label=""):
    """
    Run optimization with multiple restarts from perturbations of the current best.
    """
    best_p, best_score = _optimize_loop(p0, deadline, rng, label=label)

    remaining = deadline - time.monotonic()
    if remaining > 60:
        # Try a few restarts from perturbations of the best
        for restart in range(3):
            if time.monotonic() > deadline - 30:
                break
            perturbed = best_p + 0.01 * rng.standard_normal(len(best_p))
            perturbed = _normalize(perturbed)
            sub_deadline = min(time.monotonic() + remaining * 0.3, deadline - 30)
            p_r, sc_r = _optimize_loop(perturbed, sub_deadline, rng, label=f"{label}_r{restart}")
            if sc_r < best_score:
                best_score = sc_r
                best_p = p_r.copy()
            remaining = deadline - time.monotonic()

    return best_p, best_score
