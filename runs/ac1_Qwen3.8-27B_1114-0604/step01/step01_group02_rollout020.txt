```python
"""
Hybrid optimizer: adaptive gradient descent + multi-scale multiplicative coordinate descent
for minimizing 2n*max_k(conv(a,a)[k]) / (sum(a))^2 over non-negative sequences.

Key improvements over previous approach:
  1. Multi-scale multiplicative coordinate descent (coarse-to-fine candidates)
  2. Efficient peak tracking (only evaluate max over top-K convolution peaks)
  3. Incremental convolution update (O(n) per coordinate, not O(n log n))
  4. Adaptive restarts based on stagnation detection
  5. Gradient descent warmup followed by long coordinate descent refinement
  6. Final polish with golden-section 1D optimization
"""

import numpy as np
import time


def _fft_full_conv(a, b=None):
    """Full linear convolution via real FFT."""
    if b is None:
        b = a
    M = len(a) + len(b) - 1
    Mf = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, Mf)
    B = np.fft.rfft(b, Mf)
    return np.fft.irfft(A * B, Mf)[:M]


def _self_conv(a):
    """Full self-convolution of a (length n) -> length 2n-1."""
    return _fft_full_conv(a, a)


def _topk_indices(c, K):
    """Return indices of the K largest values in c, sorted descending."""
    K = min(K, len(c))
    if K <= 0:
        return np.array([], dtype=int)
    idx = np.argpartition(c, -K)[-K:]
    vals = c[idx]
    order = np.argsort(vals)[::-1]
    return idx[order]


def _eval_at_peaks(c, a, i, x, peaks):
    """
    Compute max of (c with a[i] replaced by x) at the given peak indices.
    Uses the formula: c_new[k] = c[k] + (x-old)*2*a[k-i] (if 2i!=k) or x^2-old^2 (if 2i==k).
    """
    old = a[i]
    if x == old:
        return c[peaks].max()
    n = len(a)
    ki = peaks - i
    valid = (ki >= 0) & (ki < n)
    # Work on a copy of c at peak positions
    c_sub = c[peaks].astype(np.float64)
    # Linear contribution
    lm = valid & (peaks != 2 * i)
    if lm.any():
        c_sub[lm] += (x - old) * 2.0 * a[ki[lm]]
    # Quadratic contribution at k = 2*i
    qm = (peaks == 2 * i)
    if qm.any():
        c_sub[qm] += x * x - old * old
    return c_sub.max()


def _update_conv_inplace(c, a, i, new_val):
    """Update convolution c in-place after setting a[i] = new_val. Returns delta_S."""
    old = a[i]
    diff = new_val - old
    if abs(diff) < 1e-15:
        return 0.0
    n = len(a)
    # For k in [0, 2n-2]: c[k] gets contribution from a[i]*a[k-i] and a[k-i]*a[i]
    # delta c[k] = diff * a[k-i] + diff * a[k-i] = 2*diff*a[k-i]  (if k-i != i, i.e., k != 2i)
    # delta c[2i] = new_val^2 - old^2 = diff * (new_val + old)
    k_range = np.arange(2 * n - 1)
    ki = k_range - i
    valid = (ki >= 0) & (ki < n)
    linear = valid & (k_range != 2 * i)
    if linear.any():
        c[linear] += diff * 2.0 * a[ki[linear]]
    if 0 <= 2 * i < 2 * n - 1:
        c[2 * i] += new_val * new_val - old * old
    a[i] = new_val
    return diff


def _objective(n, c_max, S):
    """Compute 2n*c_max/S^2."""
    if S < 1e-12:
        return np.inf
    return 2.0 * n * c_max / (S * S)


def _cd_pass(a, c, peaks, multipliers, S):
    """
    One pass of multiplicative coordinate descent.
    For each coordinate (random order), try a[ i ] * mult for each mult,
    and a[i]=0, pick the best, update c incrementally.
    Returns updated S.
    """
    n = len(a)
    order = np.random.permutation(n)
    for i in order:
        old = a[i]
        # Current objective (at peaks only, as proxy)
        cur_max = c[peaks].max()
        cur_S2 = S * S
        cur_obj = cur_max / cur_S2

        best_obj = cur_obj
        best_x = old

        # Try candidates
        cands = [0.0]
        for m in multipliers:
            cands.append(old * m)
        # Also try a large value
        cands.append(old + max(a[np.random.choice(n, min(20, n), replace=False)]))

        for x in cands:
            if x < 0:
                x = 0.0
            S_new = S + x - old
            if S_new < 1e-12:
                continue
            new_max = _eval_at_peaks(c, a, i, x, peaks)
            new_obj = new_max / (S_new * S_new)
            if new_obj < best_obj:
                best_obj = new_obj
                best_x = x

        if best_x != old:
            _update_conv_inplace(c, a, i, best_x)
            S += best_x - old
    return S


def _cd_pass_fine(a, c, peaks, S, n_fracs=8):
    """
    Fine coordinate descent: for each coordinate, do a golden-section search
    over a local range around the current value.
    """
    n = len(a)
    order = np.random.permutation(n)
    gr = (1.0 - np.sqrt(5.0)) / 2.0  # golden ratio conjugate ~0.382
    for i in order:
        old = a[i]
        # Define search range
        scale = max(old * 0.5, 0.01 * a.mean(), 1e-4)
        lo = max(0.0, old - scale)
        hi = old + scale

        def f_of_x(x):
            S_new = S + x - old
            if S_new < 1e-12:
                return np.inf
            return _eval_at_peaks(c, a, i, x, peaks) / (S_new * S_new)

        # Golden section
        x1 = hi - gr * (hi - lo)
        x2 = lo + gr * (hi - lo)
        f1 = f_of_x(x1)
        f2 = f_of_x(x2)
        for _ in range(n_fracs):
            if f1 < f2:
                hi = x2
                x2 = x1
                f2 = f1
                x1 = hi - gr * (hi - lo)
                f1 = f_of_x(x1)
            else:
                lo = x1
                x1 = x2
                f1 = f2
                x2 = lo + gr * (hi - lo)
                f2 = f_of_x(x2)

        best_x = (lo + hi) * 0.5
        best_f = f_of_x(best_x)
        cur_f = f_of_x(old)
        if best_f < cur_f:
            _update_conv_inplace(c, a, i, best_x)
            S += best_x - old
    return S


def _grad_descent_phase(a, deadline, rng, n, topk_init=8):
    """
    Gradient descent with soft-max peak tracking (similar to previous program
    but with adaptive temperature and more aggressive restarts).
    Returns (best_score, best_a).
    """
    c = _self_conv(a)
    S = a.sum()
    c_max = c.max()
    best_score = _objective(n, c_max, S)
    best_a = a.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = 0.2
    no_improve = 0
    patience = 200

    while time.monotonic() < deadline:
        c = _self_conv(a)
        S = a.sum()
        c_max = c.max()
        sc = _objective(n, c_max, S)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            # Restart from best with perturbation
            scale = 0.05 + 0.15 * rng.random()
            a = best_a * (1.0 + scale * rng.standard_normal(n))
            a = np.clip(a, 0.0, 1000.0)
            if a.sum() < 1e-3:
                a += 0.1
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            lr = max(0.01, lr * 0.6)
            no_improve = 0
            patience = max(50, patience // 2)
            continue

        # Compute gradient using soft-max over top peaks
        K = min(topk_init, len(c))
        peaks = _topk_indices(c, K)
        peak_vals = c[peaks]
        top = peak_vals[0]
        temp = 0.1 * top
        if temp < 1e-12:
            temp = 1e-12
        wts = np.exp((peak_vals - top) / temp)
        wts = wts / wts.sum()
        M_w = float(peak_vals @ wts)

        # Gradient of weighted peak surrogate
        grad = np.zeros(n)
        for ii in range(peaks.size):
            ki = int(peaks[ii])
            wi = float(wts[ii])
            if wi < 1e-14:
                continue
            j_lo = max(0, ki - n + 1)
            j_hi = min(ki, n - 1)
            if j_lo > j_hi:
                continue
            # d(conv[k])/da[j] = 2*a[k-j] for j in range
            seg = a[j_lo:j_hi+1]
            rev_seg = a[ki - j_hi : ki - j_lo + 1][::-1]
            grad[j_lo:j_hi+1] += wi * 2.0 * rev_seg

        S2 = S * S
        S3 = S2 * S
        # d(2n*M_w/S^2)/da = 2n*(grad_c/S^2 - 2*M_w/S^3)
        grad = 2.0 * n * (grad / S2 - 2.0 * M_w / S3)

        # Adam step
        t += 1
        b1, b2, eps = 0.9, 0.999, 1e-8
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1 ** t)
        vh = v / (1 - b2 ** t)
        a = a - lr * mh / (np.sqrt(vh) + eps)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 1e-3:
            a += 0.1

        # Decay learning rate
        lr *= 0.9995
    return best_score, best_a

        best_score = _objective(n, _self_conv(best_a).max(), best_a.sum())

    # === Phase 1: Extended gradient descent from best (20% of time) ===
    phase1_end = t_start + 0.20 * total
    if time.monotonic() < phase1_end:
        sc, aa = _grad_descent_phase(best_a.copy(), phase1_end, rng, n, topk_init=12)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print(f"[Phase1 grad] score = {best_score:.6f}")

    # === Phase 2: Coarse coordinate descent (35% of time) ===
    phase2_end = t_start + 0.55 * total
    a = best_a.copy()
    c = _self_conv(a)
    S = a.sum()
    K_coarse = 300
    peaks = _topk_indices(c, K_coarse)
    coarse_mults = [0.0, 0.4, 0.6, 0.8, 1.2, 1.5, 2.0]

    no_improve = 0
    patience_cd = 80
    pass_count = 0

    while time.monotonic() < phase2_end:
        # periodic recompute of c and peaks
        if pass_count % 5 == 0:
            c = _self_conv(a)
            S = a.sum()
            sc = _objective(n, c.max(), S)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            peaks = _topk_indices(c, K_coarse)

            if no_improve >= patience_cd:
                # Restart from best
                a = best_a * (1.0 + 0.08 * rng.standard_normal(n))
                a = np.clip(a, 0.0, 100.0)
                if a.sum() < 1e-3:
                    a += 0.5
                c = _self_conv(a)
                S = a.sum()
                no_improve = 0
                patience_cd = max(30, patience_cd // 2)
                pass_count = 0
                continue

        S = _cd_pass(a, c, peaks, coarse_mults, S)
        pass_count += 1

        if pass_count % 200 == 0:
            c_chk = _self_conv(a)
            sc_chk = _objective(n, c_chk.max(), a.sum())
            print(f"[Phase2 CD coarse] pass={pass_count}, score={sc_chk:.6f}, best={best_score:.6f}")

    # === Phase 3: Medium coordinate descent (20% of time) ===
    phase3_end = t_start + 0.75 * total
    a = best_a.copy()
    c = _self_conv(a)
    S = a.sum()
    K_med = 500
    peaks = _topk_indices(c, K_med)
    med_mults = [0.7, 0.85, 0.95, 1.05, 1.15, 1.3]
    no_improve = 0
    patience_cd = 60

    while time.monotonic() < phase3_end:
        if pass_count % 5 == 0:
            c = _self_conv(a)
            S = a.sum()
            sc = _objective(n, c.max(), S)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            peaks = _topk_indices(c, K_med)

            if no_improve >= patience_cd:
                a = best_a * (1.0 + 0.05 * rng.standard_normal(n))
                a = np.clip(a, 0.0, 100.0)
                if a.sum() < 1e-3:
                    a += 0.5
                c = _self_conv(a)
                S = a.sum()
                no_improve = 0
                continue

        S = _cd_pass(a, c, peaks, med_mults, S)
        pass_count += 1

        if pass_count % 500 == 0:
            c_chk = _self_conv(a)
            sc_chk = _objective(n, c_chk.max(), a.sum())
            print(f"[Phase3 CD med] pass={pass_count}, score={sc_chk:.6f}, best={best_score:.6f}")

    # === Phase 4: Fine coordinate descent with golden section (15% of time) ===
    phase4_end = t_start + 0.90 * total
    a = best_a.copy()
    c = _self_conv(a)
    S = a.sum()
    K_fine = 800
    peaks = _topk_indices(c, K_fine)
    no_improve = 0
    patience_fine = 40

    while time.monotonic() < phase4_end:
        c = _self_conv(a)
        S = a.sum()
        sc = _objective(n, c.max(), S)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        peaks = _topk_indices(c, K_fine)

        if no_improve >= patience_fine:
            a = best_a * (1.0 + 0.03 * rng.standard_normal(n))
            a = np.clip(a, 0.0, 100.0)
            no_improve = 0
            continue

        S = _cd_pass_fine(a, c, peaks, S, n_fracs=12)

    # === Phase 5: Final gradient polish (10% of time) ===
    if time.monotonic() < deadline:
        sc, aa = _grad_descent_phase(best_a.copy(), deadline, rng, n, topk_init=20)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()

    # Final verification
    c_final = _self_conv(best_a)
    S_final = best_a.sum()
    final_score = _objective(n, c_final.max(), S_final)
    print(f"\n[Final] internal score = {final_score:.6f}")
    print(f"[Final] best tracked score = {best_score:.6f}")
    print(f"[Final] n = {n}, time = {time.monotonic()-t_start:.1f}s")

    out = [float(x) for x in best_a]
    verified = float(evaluate_sequence(out))
    print(f"[Final] verified score = {verified:.6f}")

    # If verified is worse than internal (shouldn't happen), fall back to uniform
    if not np.isfinite(verified) or verified > 2.0:
        print("WARNING: verification failed, using uniform fallback")
        out = [1.0] * n
        verified = float(evaluate_sequence(out))
        print(f"[Fallback] verified = {verified:.6f}")

    return out
```
