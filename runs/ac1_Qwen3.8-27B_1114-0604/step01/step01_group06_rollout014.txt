"""
Multi-phase optimization for minimizing 2n*max(conv(a,a))/(sum(a))^2
over non-negative sequences a.

Strategy:
1. Parameterize a = softmax(z) to enforce non-negativity and sum=1
2. Use smooth approximation (log-sum-exp) of max(autocorrelation) with adaptive temperature
3. L-BFGS-B in log-space with FFT-based gradient
4. Active-set gradient polish for final refinement
5. Multi-start with diverse initializations including previous best
6. Coarse-to-fine for better basin finding

_G_N = 0
_G_NFFT = 0
_G_TAU = 0.01


def _init(n):
    global _G_N, _G_NFFT, _G_TAU
    _G_N = n

    _G_TAU = 0.01


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - z.max()
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(z) / z.size
    return e / s

    n = _G_N
    fa = np.fft.rfft(a, _G_NFFT)

    return np.maximum(c, 0.0)

    n = _G_N
    c = _conv(a)
    return 2.0 * n * float(c.max())

    global _G_TAU
    n = _G_N

    a = _softmax(z0)
    c = _conv(a)
    m0 = float(c.max())
    score0 = 2.0 * n * m0
    print(f"  {label} initial score: {score0:.7f}")

    best_z = z0.copy()
    best_score = score0
    best_m = m0

    # Temperature annealing schedule
    tau_fracs = [0.05, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001]

    for phase, tf in enumerate(tau_fracs):
        if time.monotonic() > deadline - 30:
            break

        _G_TAU = tf * best_m
        remaining = max(10.0, deadline - time.monotonic() - 30)

        # Allocate more iterations to earlier phases (smoother landscape)
        if phase < 2:
            maxiter = int(min(300, remaining * 1.5))
            maxfun = int(min(900, remaining * 3))
        elif phase < 5:
            maxiter = int(min(200, remaining * 1.0))
            maxfun = int(min(600, remaining * 2))
        else:
            maxiter = int(min(100, remaining * 0.5))
            maxfun = int(min(300, remaining * 1))

        maxiter = max(20, maxiter)
        maxfun = max(40, maxfun)

        z_new = _lbf(best_z, maxiter=maxiter, maxfun=maxfun)
        a_new = _softmax(z_new)
        c_new = _conv(a_new)
        m_new = float(c_new.max())
        score_new = 2.0 * n * m_new

        improved = score_new < best_score - 1e-12
        if improved:
            best_score = score_new
            best_z = z_new.copy()
            best_m = m_new

        print(f"    Phase {phase} (tau_frac={tf:.4f}): score={score_new:.7f} "
              f"best={best_score:.7f} {'*' if improved else ''}")

    return _softmax(best_z), best_score


def _polish_active(a0, deadline):
    """Active-set gradient descent polish in log-space."""
    n = _G_N
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        return a0
    a /= s

    z = np.log(np.maximum(a, 1e-300))
    z = z - z.mean()

    best_a = a.copy()
    best_score = _score(a)

    rng = np.random.default_rng(777)
    stall = 0
    max_stall = 300
    it = 0

    # Adaptive active set fraction
    active_frac = 0.999

    while time.monotonic() < deadline - 3:
        c = _conv(a)
        m = float(c.max())
        score = 2.0 * n * m

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            active_frac = max(0.995, active_frac * 0.999)

        # Active set
        threshold = m * active_frac
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])
        p = np.zeros(2 * n - 1)
        p[active] = 1.0 / len(active)

        # Gradient
        q = p[::-1]
        fq = np.fft.rfft(q, _G_NFFT)
        fa = np.fft.rfft(a, _G_NFFT)
        conv_full = np.fft.irfft(fq * fa, _G_NFFT)
        g = conv_full[n - 1:2 * n - 1].copy()[::-1]

        sg = float(a @ g)
        grad_a = 2.0 * (g - sg)
        grad_z = a * grad_a

        # Line search
        accepted = False
        # Normalize gradient for step size selection
        gnorm = np.linalg.norm(grad_z)
        if gnorm < 1e-15:
            stall += 1
            if stall > max_stall:
                break
            # Try random escape
            z = z + 0.1 * rng.standard_normal(n)
            a = _softmax(z)
            continue

        # Backtracking line search
        c_m = m
        step_scale = 1.0
        for attempt in range(8):
            eta = step_scale * (0.5 ** attempt)
            z_t = np.clip(z - eta * grad_z, -50, 50)
            a_t = _softmax(z_t)
            c_t = _conv(a_t)
            if c_t.max() < c_m - 1e-14:
                z = z_t
                a = a_t
                accepted = True
                break
            step_scale = max(step_scale * 0.5, 1e-12)

        if not accepted:
            # Random perturbation escape
            escaped = False
            for scale in [0.005, 0.02, 0.05, 0.1, 0.3, 0.5, 1.0]:
                z_t = np.clip(z + scale * rng.standard_normal(n), -50, 50)
                a_t = _softmax(z_t)
                sc_t = _score(a_t)
                if sc_t < best_score - 1e-10:
                    z = z_t
                    a = a_t
                    escaped = True
                    break
            if not escaped:
                stall += 1
                # Gradually shrink active set for more focused search
                if stall % 50 == 0:
                    active_frac = max(0.9999, active_frac + 0.0001)
                if stall > max_stall:
                    break
            else:
                stall = 0
        else:
            stall = 0

        it += 1
        if it % 100 == 0:
            print(f"    polish it={it} score={score:.7f} best={best_score:.7f} "
                  f"left={max(0, deadline - time.monotonic()):.0f}s")

    return best_a


def _coarse_to_fine(n_fine, deadline):
    """Optimize at coarse resolution, then upsample to fine."""
    n_coarse = max(256, n_fine // 16)
    _init(n_coarse)
    rng = np.random.default_rng(999)

    # Quick optimization at coarse level
    z0 = np.zeros(n_coarse)
    _G_TAU = 0.05 / n_coarse
    z1 = _lbf(z0, maxiter=100, maxfun=200)

    _G_TAU = 0.01 / n_coarse
    z2 = _lbf(z1, maxiter=100, maxfun=200)

    _G_TAU = 0.001 / n_coarse
    z3 = _lbf(z2, maxiter=100, maxfun=200)

    a_coarse = _softmax(z3)
    sc_coarse = _score(a_coarse)
    print(f"  Coarse (n={n_coarse}) score: {sc_coarse:.7f}")

    # Upsample to fine
    _init(n_fine)
    u_fine = np.linspace(0, 1, n_fine)
    u_coarse = np.linspace(0, 1, n_coarse)
    a_fine = np.interp(u_fine, u_coarse, a_coarse)
    a_fine = np.maximum(a_fine, 1e-10)
    a_fine /= a_fine.sum()

    z_fine = np.log(a_fine) - np.log(a_fine).mean()
    z_fine = np.clip(z_fine, -50, 50)
    return z_fine


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    budget_s = max(60.0, float(budget_s))
    deadline = time.monotonic() + budget_s - 5.0
    t0 = time.monotonic()

    n = 8192
    _init(n)
    rng = np.random.default_rng(int(seed))

    print(f"n={n}, budget={budget_s:.0f}s, NFFT={_G_NFFT}")

    # Generate starts
    starts = _make_starts(rng, n)
    print(f"Generated {len(starts)} starts")

    # Also add coarse-to-fine start
    if time.monotonic() < deadline - 500:
        try:
            z_cf = _coarse_to_fine(n, deadline)
            starts.append(z_cf)
        except Exception as e:
            print(f"  Coarse-to-fine failed: {e}")

    # Sort by initial score
    starts_scored = []
    for z in starts:
        a = _softmax(z)
        s = _score(a)
        starts_scored.append((s, z))
    starts_scored.sort(key=lambda t: t[0])

    print(f"Best start score: {starts_scored[0][0]:.7f}")

    best_a = None
    best_score = np.inf

    # Optimize top starts
    num_starts_to_try = min(len(starts_scored), 8)
    for idx in range(num_starts_to_try):
        s0, z0 = starts_scored[idx]
        remaining = deadline - time.monotonic()
        if remaining < 120:
            break

        # Allocate time proportionally: earlier starts get more time
        time_alloc = min(remaining - 60, max(60, 1800.0 / (idx + 1) * 0.5))
        local_deadline = time.monotonic() + time_alloc

        label = f"Start[{idx}]"
        try:
            a_cand, sc_cand = _optimize_start(z0, local_deadline, label=label)
            if sc_cand < best_score:
                best_score = sc_cand
                best_a = a_cand.copy()
                print(f"  --> New best: {best_score:.7f} at {time.monotonic()-t0:.0f}s")
        except Exception as e:
            print(f"  Error on start {idx}: {e}")

    if best_a is None:
        # Fallback: use the best start directly
        a = _softmax(starts_scored[0][1])
        best_a = a
        best_score = _score(a)

    print(f"\nPre-polish best: {best_score:.7f} at {time.monotonic()-t0:.0f}s")

    # Polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        print(f"Polishing with {remaining:.0f}s remaining...")
        best_a = _polish_active(best_a, deadline)
        final_score = _score(best_a)
        print(f"Post-polish score: {final_score:.7f}")

        if final_score < best_score:
            best_score = final_score

    # Try a second polish round if time remains
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"Second polish round with {remaining:.0f}s...")
        best_a = _polish_active(best_a, deadline)
        final_score = _score(best_a)
        print(f"Second polish score: {final_score:.7f}")
        if final_score < best_score:
            best_score = final_score

    # Final validation
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = float(best_a.sum())
    if s < 0.01 or not np.isfinite(s):
        best_a = np.ones(n) / n
    else:
        best_a /= s

    print(f"\nFinal internal score: {_score(best_a):.7f}")
    print(f"Total time: {time.monotonic()-t0:.0f}s")

    # Validate with official evaluator
    try:
        seq = [float(v) for v in best_a.tolist()]
        official = evaluate_sequence(seq)
        print(f"Official evaluator score: {official:.7f}")
    except Exception as e:
        print(f"Evaluation error: {e}")

    return [float(v) for v in best_a.tolist()]
