def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return v
    if not np.all(np.isfinite(v)):
        return np.full(v.shape, target / float(v.size))
    
    n = v.size
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0 or not np.isfinite(s):
        q = np.full(n, target / n)
    else:
        q *= target / s
    return q


def normalize_simplex(p):
    """Ensure p >= 0 and sum(p) = 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / p.size)
    return p / s


def momentum_gradient_descent(p0, deadline, beta_start=5000, beta_end=500000, max_iter=500000):
    """
    Nesterov momentum projected gradient descent on the simplex.
    Adaptive beta for smooth max.
    """
    p = normalize_simplex(p0)
    n = p.size
    if n < 2:
        return p, fast_score(p)
    
    best_p = p.copy()
    best_score = fast_score(p)
    
    # Momentum
    v = np.zeros(n)
    mu = 0.9  # momentum coefficient
    
    # Step size
    step = 1.0 / n  # initial step
    
    it = 0
    no_improve = 0
    
    total_iters = max_iter
    beta_current = beta_start
    
    while it < total_iters and time.monotonic() < deadline:
        it += 1
        
        # Adaptive beta: increase over iterations
        progress = min(1.0, it / (total_iters * 0.8))
        beta_current = beta_start * (beta_end / beta_start) ** progress
        
        # Nesterov lookahead
        p_look = project_simplex(p + mu * v, 1.0)
        
        # Compute gradient at lookahead point
        G, grad = compute_gradient(p_look, beta_current)
        if not np.isfinite(G):
            break
        
        # True score at current point
        cur_score = fast_score(p)
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if it % 500 == 0:
            print(f"  MGD it={it} score={best_score:.8f} beta={beta_current:.0f} step={step:.2e}")
        
        if no_improve > 3000:
            # Try a smaller step
            step *= 0.1
            if step < 1e-10:
                break
            no_improve = 0
        
        # Gradient step with backtracking
        grad = grad - np.mean(grad)  # project gradient to tangent space of simplex
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break
        
        # Update momentum
        v = mu * v - step * grad
        
        # Take step
        p_new = project_simplex(p + v, 1.0)
        
        # Check if it improved (using smooth objective for fast check)
        b_new = fft_convolution(p_new)
        G_new, _ = smooth_max(b_new, beta_current)
        b_old = fft_convolution(p)
        G_old, _ = smooth_max(b_old, beta_current)
        
        if np.isfinite(G_new) and G_new <= G_old + 1e-12:
            p = p_new
        else:
            # Backtrack
            v = mu * v - (step * 0.1) * grad
            p = project_simplex(p + v, 1.0)
            step *= 0.5
        
        # Occasionally increase step
        if it % 100 == 0 and step < 1.0:
            step = min(1.0, step * 1.3)
    
    return best_p, best_score

def fft_conv(a):
    """Full linear convolution a*a via FFT, nonneg-clipped."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def fast_score(p):
    """Compute 2*n*max(conv(p,p))/(sum(p))^2."""
    n = p.size
    if n == 0:
        return float('inf')
    S = float(p.sum())
    if S < 1e-15:
        return float('inf')
    b = fft_conv(p)
    M = float(b.max())
    return 2.0 * n * M / (S * S)


def smooth_max_vals(b, beta):
    """Return (smooth_max_value, softmax_weights, true_max)."""
    m = float(b.max())
    if m <= 0.0 or not np.isfinite(m):

        e[int(b.argmax())] = 1.0
        Z = 1.0
    w = e / Z
    G = m + np.log(Z) / beta
    return G, w, m


def compute_grad(p, beta):
    """
    Compute gradient of smooth_max(conv(p,p)) w.r.t. p.
    Returns (smooth_max_value, gradient).
    """
    n = p.size
    b = fft_conv(p)
    G, w, M = smooth_max_vals(b, beta)
    if not np.isfinite(G):

    # Use FFT cross-correlation: grad[i] = 2 * C[n-1+i] where C = ifft(fft(p_rev) * fft(w))
    p_rev = p[::-1]

    final_sc = fast_score(p)

def generate_starts(n, rng):
    """Generate a diverse set of starting sequences."""
    starts = []
    x = np.linspace(0, 1, n)

    # Uniform
    starts.append(np.full(n, 1.0 / n))

    # Exponential ramps (various rates)
    for rate in (0.5, 1.0, 2.0, 3.0, 5.0, 8.0, 12.0):
        a = np.exp(rate * x)
        starts.append(a / a.sum())
        a = np.exp(rate * (1 - x))  # reverse
        starts.append(a / a.sum())

    # Polynomial ramps
    for deg in (2, 3, 4):
        a = x ** deg
        starts.append(a / a.sum())
        a = (1 - x) ** deg
        starts.append(a / a.sum())

    # Bell-shaped (concentrated in middle)
    for width in (0.1, 0.2, 0.3, 0.5):
        a = np.exp(-((x - 0.5) ** 2) / (2 * width * width))
        starts.append(a / a.sum())

    # Sinusoidal modulations
    for freq in (1, 2, 3, 5):
        a = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x)
        a = np.maximum(a, 0.01)
        starts.append(a / a.sum())

    # Step functions (two levels)
    for split in (0.3, 0.4, 0.5, 0.6, 0.7):
        for ratio in (0.1, 0.3, 0.5, 2.0, 5.0, 10.0):
            a = np.ones(n)
            a[:int(split * n)] = ratio
            a = np.maximum(a, 0.001)
            starts.append(a / a.sum())

    # Random starts
    for scale in (0.5, 1.0, 2.0):
        a = np.abs(rng.normal(0, scale, n)) + 0.01
        starts.append(a / a.sum())

    # "Good" structure: high at one end, low at other, with smooth transition
    for sharpness in (2, 4, 8, 16):
        a = 1.0 / (1.0 + np.exp(-sharpness * (x - 0.5)))
        starts.append(a / a.sum())

    return starts

        # Give more time to larger lengths
        weight = n / float(sum(l for l in lengths if l >= n) + 1)
        alloc = remaining * (weight / (sum(l / float(sum(ll for ll in lengths if ll >= l) + 1) for l in lengths)))
        alloc = max(60.0, min(remaining * 0.8, alloc))
        n_deadline = time.monotonic() + alloc

        print(f"\n  === Length n={n}, time alloc={alloc:.0f}s ===")
        t0 = time.monotonic()

        # Generate starts
        starts = generate_starts(n, rng)

        # Add height_sequence_1 if compatible length
        if hs1 is not None and len(hs1) == n:
            starts.append(hs1.copy())
            print(f"  Added height_sequence_1 as start")

        # Also add interpolated version of hs1 if different length
        if hs1 is not None and len(hs1) != n:
            hs1_interp = np.interp(np.linspace(0, 1, n), np.linspace(0, 1, len(hs1)), hs1)
            hs1_interp = normalize(hs1_interp)
            starts.append(hs1_interp)
            print(f"  Added interpolated height_sequence_1")

        # Score all starts and pick top ones
        scored = []
        for s in starts:
            s = normalize(s)
            sc = fast_score(s)
            if np.isfinite(sc):
                scored.append((sc, s))
        scored.sort(key=lambda x: x[0])

        print(f"  Generated {len(starts)} starts, best start score={scored[0][0]:.8f}")

        # Run gradient descent from top starts
        top_k = min(5, len(scored))
        time_per_start = (n_deadline - time.monotonic()) / max(1, top_k)

        for rank in range(top_k):
            if time.monotonic() >= n_deadline - 30:
                break

            p0 = scored[rank][1]
            start_sc = scored[rank][0]
            print(f"  Start {rank}: score={start_sc:.8f}, allocating {time_per_start:.0f}s")

            sub_deadline = time.monotonic() + max(30.0, time_per_start)

            # Phase 1: Coarse optimization (lower beta)
            p1, sc1 = gradient_descent(
                p0, sub_deadline,
                beta_start=1000.0, beta_end=100000.0,
                max_iter=1000000, mu=0.8
            )

            if np.isfinite(sc1) and sc1 < best_overall_score:
                best_overall_score = sc1
                best_overall = p1.copy()
                print(f"  Phase1 {rank}: score={sc1:.8f}")

            if time.monotonic() >= n_deadline - 30:
                break

            # Phase 2: Fine optimization (higher beta)
            remaining_fine = n_deadline - time.monotonic()
            if remaining_fine > 20:
                p2, sc2 = gradient_descent(
                    p1, time.monotonic() + remaining_fine,
                    beta_start=50000.0, beta_end=5000000.0,
                    max_iter=1000000, mu=0.6
                )
                if np.isfinite(sc2) and sc2 < best_overall_score:
                    best_overall_score = sc2
                    best_overall = p2.copy()
                    print(f"  Phase2 {rank}: score={sc2:.8f}")

        # Also run a dedicated optimization from the best start with more time
        if time.monotonic() < n_deadline - 10:
            best_start = scored[0][1]
            p3, sc3 = gradient_descent(
                best_start, n_deadline,
                beta_start=5000.0, beta_end=5000000.0,
                max_iter=2000000, mu=0.7
            )
            if np.isfinite(sc3) and sc3 < best_overall_score:
                best_overall_score = sc3
                best_overall = p3.copy()
                print(f"  Dedicated n={n}: score={sc3:.8f}")

        elapsed = time.monotonic() - t0
        print(f"  Length {n} done in {elapsed:.0f}s, best so far: {best_overall_score:.8f}")

    # Final: if we have a best, verify with official_score
    if best_overall is not None:
        final_p = normalize(best_overall)
        final_seq = [float(x) for x in final_p]
        off = official_score(final_seq)
        print(f"\nFinal official score: {off:.8f}")

        if np.isfinite(off) and off <= 2.0:
            return final_seq

    # Fallback: uniform sequence
    n_fb = min(4096, max(64, len(lengths[-1]) if lengths else 256))
    print(f"Falling back to uniform n={n_fb}")
    return [1.0] * n_fb
