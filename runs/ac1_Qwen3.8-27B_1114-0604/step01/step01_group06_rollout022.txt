        f = np.ones(n, dtype=np.float64)
        for m in range(1, 9):
            f += 0.12 * rng.standard_normal() * np.sin(2.0 * np.pi * m * u)
            f += 0.12 * rng.standard_normal() * np.cos(2.0 * np.pi * m * u)
        if f.min() > 0.001:
            add_from_f(f)

    # Uniform
    add_from_f(np.ones(n, dtype=np.float64))

    # Two-level block sequences
    half = n // 2
    for ratio in (1.5, 2.0, 3.0, 5.0):
        f = np.empty(n, dtype=np.float64)
        f[:half] = 1.0
        f[half:] = ratio
        add_from_f(f)

    starts.sort(key=_score_key)
    return starts[:min(12, len(starts))]


def _adam_optimize(z0, deadline, tau_init=0.1, tau_min=1e-7, lr_init=0.05,
                   perturb_every=400, perturb_scale=0.3, seed=0):
    n = _G_N
    rng = np.random.default_rng(seed)
    z = np.array(z0, dtype=np.float64, copy=True)
    z = np.clip(z - z.mean(), -_G_BOUND, _G_BOUND)

    m_arr = np.zeros(n, dtype=np.float64)
    v_arr = np.zeros(n, dtype=np.float64)
    t = 0

    best_z = z.copy()
    best_a = _softmax(z)
    best_score = _score(best_a)

    total_iters = 8000
    step = 0

    while time.monotonic() < deadline and step < total_iters:
        progress = step / max(1, total_iters - 1)
        # Log-linear tau annealing
        tau = tau_init * (tau_min / tau_init) ** progress
        # Cosine learning rate schedule
        lr = lr_init * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr = max(lr, lr_init * 0.01)

        val, grad = _obj_grad_smooth(z, tau)
        if not np.isfinite(val):
            break

        t += 1
        z = _adam_update(z, grad, m_arr, v_arr, t, lr)

        # Periodic random perturbation to escape local minima
        if step % perturb_every == perturb_every - 1:
            a_test = _softmax(z + perturb_scale * rng.standard_normal(n))
            sc_test = _score(a_test)
            if sc_test < best_score:
                z = np.clip(z + perturb_scale * rng.standard_normal(n) - np.mean(
                    z + perturb_scale * rng.standard_normal(n)), -_G_BOUND, _G_BOUND)
                # Re-derive from the perturbed direction
                pass  # Keep current z, the test was just exploratory

        a = _softmax(z)
        sc = _score(a)
        if sc < best_score - 1e-12:
            best_score = sc
            best_z = z.copy()
            best_a = a.copy()

        step += 1
        if step % 1000 == 0:
            print(f"  adam phase: step={step} tau={tau:.2e} score={sc:.9f} "
                  f"best={best_score:.9f} time_left={max(0, deadline-time.monotonic()):.0f}s")

    return best_a, best_score, best_z


def _polish_active(a0, deadline, max_iter=5000, seed=12345):
    n = _G_N
    rng = np.random.default_rng(seed)
    a = np.array(a0, dtype=np.float64, copy=True)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-12:
        return a0, _score(a0)
    a /= s

    z = np.log(np.maximum(a, 1e-300))
    z = z - z.mean()
    z = np.clip(z, -_G_BOUND, _G_BOUND)

    m_arr = np.zeros(n, dtype=np.float64)
    v_arr = np.zeros(n, dtype=np.float64)
    t = 0

    best_z = z.copy()
    best_score = _score(_softmax(z))
    best_a = _softmax(z).copy()

    lr = 0.02
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        val, grad = _obj_grad_active(z, frac=0.93)
        if not np.isfinite(val):
            break

        t += 1
        z = _adam_update(z, grad, m_arr, v_arr, t, lr)

        a = _softmax(z)
        sc = _score(a)

        if sc < best_score - 1e-12:
            best_score = sc
            best_z = z.copy()
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            # Try escape
            z_test = z + 0.2 * rng.standard_normal(n)
            z_test = np.clip(z_test - z_test.mean(), -_G_BOUND, _G_BOUND)
            a_test = _softmax(z_test)
            sc_test = _score(a_test)
            if sc_test < best_score:
                z = z_test
                no_improve = 0
            else:
                lr *= 0.8
                if lr < 1e-5:
                    break

        # Gradually reduce lr
        if it % 500 == 499:
            lr *= 0.7

        if it % 500 == 0:
            print(f"  polish: it={it} score={sc:.9f} best={best_score:.9f} "
                  f"lr={lr:.2e} time_left={max(0, deadline-time.monotonic()):.0f}s")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative coefficient sequence using Adam with
    tau-annealing, followed by active-set polishing.
    """
    budget_s = max(60.0, float(budget_s))
    t_start = time.monotonic()
    deadline = t_start + budget_s - 5.0

    n = 8192
    _init_state(n)

    rng = np.random.default_rng(int(seed))
    print(f"n={n}, budget={budget_s:.0f}s, NFFT={_G_NFFT}")

    # Generate starting points
    starts = _make_starts(n, rng)
    if not starts:
        starts = [(2.0, np.zeros(n, dtype=np.float64))]

    best_a = _softmax(starts[0][1])
    best_score = starts[0][0]
    best_z = starts[0][1].copy()
    print(f"Initial best score: {best_score:.9f}")

    # Phase 1: Adam with smooth objective (large tau)
    # Process top starts
    phase1_deadline = t_start + 0.35 * budget_s
    candidates = []

    for idx in range(min(6, len(starts))):
        if time.monotonic() > phase1_deadline:
            break
        sc0, z0 = starts[idx]
        time_left = phase1_deadline - time.monotonic()
        if time_left < 30.0:
            break

        print(f"Phase 1: start {idx} (init_score={sc0:.6f}), time_left={time_left:.0f}s")
        a1, sc1, z1 = _adam_optimize(
            z0, deadline=phase1_deadline,
            tau_init=0.1, tau_min=1e-5,
            lr_init=0.08,
            perturb_every=300, perturb_scale=0.4,
            seed=seed + idx * 100
        )
        if sc1 < best_score:
            best_score = sc1
            best_a = a1.copy()
            best_z = z1.copy()
            print(f"  Improved! score={sc1:.9f}")
        candidates.append((sc1, a1.copy(), z1.copy()))

    # Phase 2: Adam with shorter tau (refinement)
    candidates.sort(key=_score_key)
    phase2_deadline = t_start + 0.7 * budget_s

    for idx in range(min(4, len(candidates))):
        if time.monotonic() > phase2_deadline:
            break
        sc0, a0, z0 = candidates[idx]
        time_left = phase2_deadline - time.monotonic()
        if time_left < 30.0:
            break

        print(f"Phase 2: candidate {idx} (score={sc0:.9f}), time_left={time_left:.0f}s")
        a2, sc2, z2 = _adam_optimize(
            z0, deadline=phase2_deadline,
            tau_init=0.01, tau_min=1e-7,
            lr_init=0.03,
            perturb_every=500, perturb_scale=0.2,
            seed=seed + 1000 + idx * 100
        )
        if sc2 < best_score:
            best_score = sc2
            best_a = a2.copy()
            best_z = z2.copy()
            print(f"  Improved! score={sc2:.9f}")
        candidates.append((sc2, a2.copy(), z2.copy()))

    # Phase 3: Active-set polishing
    candidates.sort(key=_score_key)
    for idx in range(min(3, len(candidates))):
        if time.monotonic() > deadline:
            break
        sc0, a0, z0 = candidates[idx]
        time_left = deadline - time.monotonic()
        if time_left < 20.0:
            break

        print(f"Phase 3: polishing candidate {idx} (score={sc0:.9f}), time_left={time_left:.0f}s")
        a3, sc3 = _polish_active(a0, deadline=deadline, seed=seed + 2000 + idx * 100)
        if sc3 < best_score:
            best_score = sc3
            best_a = a3.copy()
            print(f"  Improved! score={sc3:.9f}")

    print(f"Final best score: {best_score:.9f}")

    # Validate with official evaluator
    try:
        best_a = np.asarray(best_a, dtype=np.float64).ravel()
        best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
        best_a = np.clip(best_a, 0.0, 1000.0)
        s = float(best_a.sum())
        if s < 0.01 or not np.isfinite(s):
            best_a = np.ones(n, dtype=np.float64) / n
        else:
            best_a = best_a / s
        exact = evaluate_sequence(best_a.tolist())
        if np.isfinite(exact):
            print(f"Official evaluator score: {float(exact):.9f}")
    except Exception as e:
        print(f"Validation error: {e}")

    # Final normalization
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = float(best_a.sum())
    if s < 0.01 or not np.isfinite(s):
        best_a = np.ones(n, dtype=np.float64) / n
    else:
        best_a = best_a / s

    return [float(v) for v in best_a.tolist()]
