    return 2.0 * len(a) * c.max() / (s*s)

def smooth_grad(z, tau):
    """Compute value and gradient of smooth approximation."""
    n = _N
    N = _NFFT
    L = 2*n - 1
    
    a = softmax(z)
    c = fft_conv(a)
    m = c.max()
    
    diff = (c - m) / tau
    diff = np.clip(diff, -745.0, 0.0)
    e = np.exp(diff)
    se = e.sum()
    p = e / se
    
    # Gradient of F w.r.t. a: h[j] = 2 * sum_k p[k] * a[k-j] (valid k-j)
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real
    
    # Gradient w.r.t. z (softmax reparameterization)
    s_h = np.dot(h, a)
    g = a * (h - s_h)
    
    val = m + tau * np.log(se)
    return val, g

def adam_optimize(z, tau, steps, lr, deadline):
    """Adam optimizer in log-space."""
    n = _N
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    best_z = z.copy()
    best_val = score(softmax(z))
    
    for t in range(1, steps+1):
        if time.monotonic() > deadline:
            break
        
        val, g = smooth_grad(z, tau)
        
        m = beta1*m + (1-beta1)*g
        v = beta2*v + (1-beta2)*g*g
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z_new = np.clip(z_new, -50.0, 50.0)
        
        # Check true score periodically
        if t % 100 == 0:
            a_new = softmax(z_new)
            true_score = score(a_new)
            if true_score < best_val - 1e-10:
                best_val = true_score
                best_z = z_new.copy()
        
        z = z_new
    
    # Final check
    a = softmax(z)
    true_score = score(a)
    if true_score < best_val:
        best_val = true_score
        best_z = z.copy()
    
    return best_z, best_val

def exact_line_search(z, max_steps=200):
    """Exact line search using the true (non-smooth) objective."""
    n = _N
    tau = 1e-6
    lr = 0.1

        diff = np.clip(diff, -745.0, 0.0)
        e = np.exp(diff)
        se = e.sum()
        p = e / se
        
        N = _NFFT
        fa = np.fft.rfft(a, N)
        fp = np.fft.rfft(p, N)
        h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real
        
        s_h = np.dot(h, a)
        g = a * (h - s_h)
        
        # Try multiple step sizes
        improved = False
        for step in lr * np.array([1.0, 0.5, 0.1, 0.01, 0.001, 0.0001]):
            z_new = np.clip(z - step * g, -50.0, 50.0)
            a_new = softmax(z_new)
            sc = score(a_new)
            if sc < best_score - 1e-12:

                best_score = sc
                best_z = z_new.copy()
                improved = True
                break
        
        if not improved:
            break
    
    return best_z, best_score

**Alternative approach: Spectral method**

The autocorrelation c is related to the power spectrum S by:
c[k] = inverse DFT of S

We want to minimize max c[k]. In the frequency domain, this is related to making the spectrum as "flat" as possible (white noise has flat spectrum and delta autocorrelation).

But we have the constraint that a ≥ 0, which translates to a constraint on S (it must be the power spectrum of a non-negative sequence). This is a hard constraint.

However, if we ignore the non-negativity constraint and just minimize max c[k] over all sequences (including negative), we can use the fact that:
max c[k] ≥ c[0] = sum a[i]²

And by the uncertainty principle, if a is spread out in the time domain, it's concentrated in the frequency domain, and vice versa.

For non-negative sequences, the best we can do is probably related to the "Golomb ruler" or "Barker sequence" type constructions, but I don't think there's a closed form.

**Alternative approach: Water-filling**

Think of it this way: we have 2n-1 "bins" (the convolution lags) and we want to fill them as evenly as possible. The constraint is that the "filling pattern" must correspond to a valid autocorrelation of a non-negative sequence.

This is similar to the "optimal design" problem in statistics. The idea is to iteratively:
1. Start with a candidate sequence
2. Compute the autocorrelation
3. Identify the "over-filled" bins (where c[k] > average)
4. Adjust the sequence to reduce the over-filled bins

This is essentially what the gradient-based approach does, but the "water-filling" interpretation might suggest a different algorithm.

Δc[k] = 2*a_new[j]*a[k-j] - 2*a_old[j]*a[k-j] + a_new[j]² - a_old[j]² (if k-j=j, i.e., k=2j)

If we change a[j] from a_old to a_new:
Δc[k] = a_new[k-j]*a[k-j] + a[j]*a_new[k-j] - a[j]*a[k-j] - a[j]*a[k-j]... 

No wait. c[k] = sum_i a[i]*a[k-i].
The terms involving a[j] are:
- When i=j: term is a[j]*a[k-j]
- When k-i=j, i.e., i=k-j: term is a[k-j]*a[j]

If j ≠ k-j (i.e., k ≠ 2j): both terms are present, total contribution = 2*a[j]*a[k-j]
If j = k-j (i.e., k = 2j): only one term, a[j]²

So Δc[k] when changing a[j] from old to new:
- If k ≠ 2j: Δc[k] = 2*(new-old)*a[k-j]
- If k = 2j: Δc[k] = new² - old²

This is O(n) per coordinate update (since k ranges over n to 2n-1 values), so a full coordinate sweep is O(n²). For n=8192, that's about 67 million operations per sweep, which might be too slow for many sweeps.
