def fft_convolution(a):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).copy()

        return v

    v = np.maximum(v, 0.0)
    s = float(np.sum(v))
    if s <= 0 or not np.isfinite(s):
        return np.full(v.shape, target / float(v.size), dtype=np.float64)
    v *= target / s
    # Bisection for exact projection
    low = float(np.min(v)) - 1.0
    high = float(np.max(v)) + 1.0
    for _ in range(50):
        mid = 0.5 * (low + high)
        s2 = float(np.sum(np.maximum(v - mid, 0.0)))
        if s2 > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0.0)
    s3 = float(np.sum(q))
    if s3 <= 0 or not np.isfinite(s3):

    return q

        bounds = [(-p[m], None) for m in range(n)] + [(None, None)]

        try:
            res = linprog(c, A_ub=A_ub, A_eq=A_eq, b_ub=b_ub, b_eq=b_eq,
                          bounds=bounds, method='highs',
                          options={'presolve': True, 'time_limit': 30})
        except Exception as e:
            print(f"  LP error: {e}")
            break

        if res.status != 0:
            if it < 20:
                print(f"  LP status={res.status} msg={res.message}")
            break

        d = res.x[:n]
        if not np.all(np.isfinite(d)):
            break

        # Line search
        d_norm = float(np.linalg.norm(d))
        if d_norm < 1e-15:
            break

        d = d / d_norm  # normalize direction

        found_better = False
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01):
            alpha = step_scale * mult
            q = project_simplex(p + alpha * d, 1.0)
            qb = fft_convolution(q)
            m2 = float(np.max(qb))
            score2 = 2.0 * n * m2
            if score2 < cur_score - 1e-12:
                p = q
                found_better = True
                if score2 < best_score:
                    best_score = score2
                    best_p = p.copy()
                    no_improve = 0
                break

        if not found_better:
            step_scale *= 0.5
            if step_scale < 1e-6:
                break
        else:
            step_scale = min(5.0, step_scale * 1.2)

    return best_p, best_score

def spectral_flatten(a0, n_iters=200, alpha=0.5):
    """
    GRASP-style spectral flattening.
    Iteratively:
    1. Compute power spectrum
    2. Flatten it (geometric interpolation with flat target)
    3. Convert back to time domain
    4. Project to non-negative, sum=1
    """
    p = normalize_simplex(a0)
    n = p.size
    best_p = p.copy()
    best_score = fast_score(p)

    # Zero-pad for linear convolution
    pad_n = 2 * n
    size = 1
    while size < 4 * n:
        size <<= 1

    for it in range(n_iters):
        A = np.fft.rfft(p, size)
        P = np.abs(A)**2
        phase = np.angle(A)

        # Flatten: P_new = P^t * P_mean^(1-t)
        P_mean = float(np.mean(P))
        P_safe = np.maximum(P, 1e-30)
        P_new = np.power(P_safe, alpha) * np.power(P_mean, 1.0 - alpha)

        # Preserve DC: P_new[0] should relate to (sum(p))^2 = 1
        P_new[0] = 1.0  # sum(p) = 1 => |A(0)|^2 = 1

        A_new = np.sqrt(P_new) * np.exp(1j * phase)
        a_new = np.fft.irfft(A_new, size)[:n]

        # Project
        p = project_simplex(a_new, 1.0)
        sc = fast_score(p)
        if sc < best_score:
            best_score = sc
            best_p = p.copy()

    return best_p, best_score


def official_score(seq):
    """Call the authoritative evaluator."""
    try:
        val = evaluate_sequence(seq)
        val = float(val)
    except Exception:
        return float('inf')
    if np.isfinite(val):
        return val
    return float('inf')


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search for minimum 2*n*max(conv(a,a))/(sum(a))^2.
    
    Phase 1: Spectral flattening (GRASP) - quick exploration
    Phase 2: Smooth max gradient descent - refinement
    Phase 3: LP-based peak reduction - precise peak targeting
    Phase 4: Final smooth max polish
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 10:
        return [1.0]

    start_time = time.monotonic()
    reserve = 120.0 if budget_s > 500 else 30.0
    deadline = start_time + max(0.0, budget_s - reserve)

    N = 8192
    rng = np.random.default_rng(seed)

    # Get initial point
    try:
        if 'height_sequence_1' in dir() or 'height_sequence_1' in globals():
            hs = height_sequence_1
            if isinstance(hs, list) and len(hs) > 0:
                a0 = np.array(hs, dtype=np.float64)
                a0 = np.maximum(a0, 0)
                if a0.sum() > 0:
                    a0 /= a0.sum()
                else:
                    a0 = np.ones(N) / N
                # Resize to N
                if a0.size != N:
                    new_a = np.zeros(N)
                    m = min(a0.size, N)
                    new_a[:m] = a0[:m]
                    s = new_a.sum()
                    if s > 0:
                        new_a /= s
                    else:
                        new_a = np.ones(N) / N
                    a0 = new_a
                print(f"Loaded height_sequence_1, score={fast_score(a0):.8f}")
            else:
                a0 = np.ones(N) / N
        else:
            a0 = np.ones(N) / N
    except Exception as e:
        print(f"Failed to load initial: {e}")
        a0 = np.ones(N) / N

    best_p = a0.copy()
    best_score = fast_score(best_p)
    print(f"Initial score: {best_score:.8f}")

    # Phase 1: Spectral flattening (quick, ~60s)
    t1_deadline = min(deadline, start_time + 60.0)
    if t1_deadline - time.monotonic() > 5:
        print("Phase 1: Spectral flattening...")
        for alpha in (0.3, 0.5, 0.7):
            if time.monotonic() >= t1_deadline - 5:
                break
            p_new, sc_new = spectral_flatten(best_p, n_iters=50, alpha=alpha)
            if sc_new < best_score:
                best_score = sc_new
                best_p = p_new.copy()
                print(f"  spectral alpha={alpha}: score={best_score:.8f}")

    # Phase 2: Smooth max gradient descent (~600s)
    t2_deadline = min(deadline, time.monotonic() + 600.0)
    if t2_deadline - time.monotonic() > 30:
        print("Phase 2: Smooth max gradient descent...")
        p_new, sc_new = run_smooth_max_descent(best_p, t2_deadline,
                                                beta_start=5000, beta_end=100000,
                                                max_iter=30000)
        if sc_new < best_score:
            best_score = sc_new
            best_p = p_new.copy()
            print(f"After smooth max: score={best_score:.8f}")

    # Phase 3: LP-based peak reduction (most of remaining time)
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"Phase 3: LP peak reduction (remaining={remaining:.0f}s)...")
        # Use adaptive K
        for K in (20, 40, 60):
            sub_deadline = time.monotonic() + max(30, remaining * 0.25)
            p_new, sc_new = lp_peak_reduction_timed(best_p, sub_deadline, K=K)
            if sc_new < best_score:
                best_score = sc_new
                best_p = p_new.copy()
                print(f"  LP K={K}: score={best_score:.8f}")
            remaining = deadline - time.monotonic()
            if remaining < 30:
                break

    # Phase 4: Final smooth max polish
    remaining = deadline - time.monotonic()
    if remaining > 20:
        print(f"Phase 4: Final polish (remaining={remaining:.0f}s)...")
        p_new, sc_new = run_smooth_max_descent(best_p, time.monotonic() + remaining * 0.8,
                                                beta_start=50000, beta_end=300000,
                                                max_iter=10000)
        if sc_new < best_score:
            best_score = sc_new
            best_p = p_new.copy()
            print(f"Final polish: score={best_score:.8f}")

    # Final cleanup
    final_p = normalize_simplex(best_p)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_seq = [float(x) for x in final_p]

    # Official check
    off = official_score(final_seq)
    if not np.isfinite(off) or off > 2.0 + 1e-9:
        # Fallback to uniform
        off = official_score([1.0] * N)
        if np.isfinite(off):
            return [1.0] * N
        return [1.0]

    print(f"Final official score: {off:.8f}")
    return final_seq
