import numpy as np
import time

    HAVE_SCIPY = True

def fft_convolution(a):
    """Linear convolution a*a using FFT. Returns non-negative array of length 2n-1."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)

    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1

    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    b = np.ascontiguousarray(b, dtype=np.float64)
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    """Compute smooth max of b and softmax weights for gradient."""
    b = np.asarray(b, dtype=np.float64)
    m = b.size
    if m == 0:
        return 0.0, np.empty(0, dtype=np.float64)

    mx = float(np.max(b))
    if not np.isfinite(mx):
        return float("inf"), np.zeros(m, dtype=np.float64)

    if mx <= 0.0 or beta <= 0.0:
        w = np.ones(m, dtype=np.float64) / float(m)
        return mx, w

    x = beta * (b - mx)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))

    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros(m, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0

    w = e / Z
    G = mx + float(np.log(Z)) / beta
    return float(G), w


def corr_pw(p, w):
    """
    Compute s[i] = sum_k w[k] * p[i] * p[k-i] (valid indices).
    This is the gradient factor for d(max b)/d(p[i]).
    Uses FFT: s = ifft(FFT(p) * FFT(rev(w))) restricted to valid range.
    """
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    if n == 1:
        return np.array([float(np.dot(p, w[:1]))], dtype=np.float64)

    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1

    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]

    # s[i] = C[m-1-i] for i = 0..n-1
    s = C[m - n:m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    if target <= 0.0:
        return np.zeros_like(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(v.shape, target / float(v.size), dtype=np.float64)

    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0

    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid

    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        q = np.full(v.shape, target / float(v.size), dtype=np.float64)
    else:
        q *= target / s
    return q


def normalize_simplex(p):
    """Clip to nonnegative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size), dtype=np.float64)
    p *= 1.0 / s
    return p


def fast_score(p):
    """Compute 2*n*max(convolve(p)) / (sum(p))^2."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float("inf")
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float("inf")
    b = fft_convolution(p)
    if b.size == 0:
        return float("inf")
    M = float(np.max(b))
    if not np.isfinite(M):
        return float("inf")
    return float(2.0 * n * M / (S * S))

def nesterov_projected_gradient(p0, deadline, rng, beta_start=5000.0, beta_end=500000.0,
                                max_iter=1000000, verbose=True):
    """
    Nesterov accelerated projected gradient with continuation on beta
    and Barzilai-Borwein step size adaptation.
    """
    p = normalize_simplex(p0).copy()
    n = p.size
    if n == 0:
        return p, float("inf")

    # Break symmetry if starting uniform
    if n > 1 and float(np.max(np.abs(p - p.mean()))) < 1e-14:
        t = 0.3 * np.arange(n, dtype=np.float64)
        p = normalize_simplex(p * (1.0 + 5e-5 * np.sin(t) + 3e-5 * np.cos(2.0 * t)))

    best_p = p.copy()
    best_score = fast_score(p)

    # Continuation: geometric schedule from beta_start to beta_end
    n_cont = 6  # number of continuation stages
    betas = np.geomspace(beta_start, beta_end, n_cont) if beta_end > beta_start else [beta_end]

    # Nesterov state
    y = p.copy()
    old_p = p.copy()
    momentum = 0.0
    target_momentum = 0.8

    step = 2.0
    prev_grad = None
    prev_step_dir = None
    prev_step_val = None

    it = 0
    no_improve = 0

    for stage, beta in enumerate(betas):
        stage_time_limit = max(30.0, 0.8 * (deadline - time.monotonic()) / max(1, n_cont - stage))
        stage_deadline = time.monotonic() + stage_time_limit

        # Reset momentum at start of each continuation stage
        y = p.copy()
        old_p = p.copy()
        momenta_val = 0.0
        step = 2.0
        prev_grad = None
        prev_step_dir = None
        prev_step_val = None
        no_improve = 0

        while it < max_iter and time.monotonic() < min(deadline, stage_deadline):
            it += 1

            # Compute gradient at y
            b = fft_convolution(y)
            G, w = smooth_max_and_weights(b, beta)
            if not np.isfinite(G):
                break

            svec = corr_pw(y, w)
            svec = np.maximum(svec, 0.0)
            grad = 2.0 * svec - float(np.mean(2.0 * svec))
            grad = np.where(np.isfinite(grad), grad, 0.0)

            # Nesterov momentum update
            t_k = (1.0 + np.sqrt(1.0 + 4.0 * momenta_val * momenta_val)) / 2.0
            momenta_val = (1.0 + t_k) / (1.0 + t_k + 1.0)  # simplified momentum
            # Standard Nesterov: momentum = (1 + momentum) / (1 + momentum + 1)
            # Use polyak momentum instead for stability:
            momenta_val = min(0.9, momenta_val * 1.01 + 0.001)

            # Determine step size (Barzilai-Borwein if possible)
            if prev_grad is not None and prev_step_val is not None and prev_step_val > 1e-15:
                delta = p - old_p
                gamma = prev_grad - grad
                s_gamma = float(np.dot(delta, gamma))
                g_delta = float(np.dot(grad, delta))
                if s_gamma > 1e-15:
                    bb_step = g_delta / s_gamma
                    if 0.1 < bb_step < 10.0:
                        step = 0.5 * (step + bb_step)
            step = max(0.01, min(5.0, step))

            # Nesterov step
            y_new = p + momenta_val * (p - old_p)
            old_p = p.copy()

            # Projected gradient step from y
            q = project_simplex(y_new - step * grad, 1.0)

            # Check improvement (using smooth objective for line search)
            qb = fft_convolution(q)
            G2, _ = smooth_max_and_weights(qb, beta)
            if not np.isfinite(G2):
                q = p.copy()
                G2 = G

            cur_smooth = float(n) * G
            new_smooth = float(n) * G2

            if new_smooth > cur_smooth + 1e-12:
                # Backtracking
                for factor in (0.5, 0.25, 0.1, 0.05, 0.01):
                    q = project_simplex(y_new - step * factor * grad, 1.0)
                    qb = fft_convolution(q)
                    G2, _ = smooth_max_and_weights(qb, beta)
                    if np.isfinite(G2) and float(n) * G2 <= cur_smooth + 1e-12:
                        break
                else:
                    p = old_p.copy()
                    y = old_p.copy()
                    momenta_val = 0.0
                    continue

            p = q
            y = p.copy()

            # Track best using exact (non-smooth) score
            m_exact = float(np.max(qb))
            cur_score = float(2.0 * n * m_exact) / 1.0  # sum is 1

            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_p = p.copy()
                no_improve = 0
            else:
                no_improve += 1

            if verbose and it % 500 == 0:
                print(f"  NPG stage={stage} it={it} beta={beta:.0f} score={best_score:.8f} step={step:.4f} mom={momenta_val:.3f}")

# ---------- Block L-BFGS ----------

def block_to_p(n, B, c):
    """Convert log-block-weights to full sequence (sum=1)."""
    L = n // B
    z = np.zeros(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        tmp = np.zeros(B - 1, dtype=np.float64)
        m = min(tmp.size, c.size)
        tmp[:m] = c[:m]
        c = tmp
        z[:-1] = c
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0.0 or not np.isfinite(s):
        wb = np.ones(B) / B
    else:
        wb /= s
    H = wb / L
    return np.repeat(H, L)


def block_objective(c, n, B, beta):
    """Objective and gradient for block-constant sequences."""
    L = n // B
    z = np.zeros(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        tmp = np.zeros(B - 1, dtype=np.float64)
        m = min(tmp.size, c.size)
        tmp[:m] = c[:m]
        c = tmp
        z[:-1] = c

    if not np.all(np.isfinite(z)):
        return float("inf"), np.zeros(B - 1, dtype=np.float64)

    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0.0 or not np.isfinite(s):
        wb = np.ones(B) / B
    else:
        wb /= s

    H = wb / L
    p = np.repeat(H, L)

    b = fft_convolution(p)
    G, w = smooth_max_and_weights(b, beta)
    if not np.isfinite(G):
        return float("inf"), np.zeros(B - 1, dtype=np.float64)

    svec = corr_pw(p, w)
    svec = np.maximum(svec, 0.0)
    g = 2.0 * svec

    Gsum = g.reshape(B, L).sum(axis=1)
    Wavg = float(np.dot(wb, Gsum))
    gradz = (wb / L) * (Gsum - Wavg)

    grad = n * gradz[:-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(n * G), grad.astype(np.float64)


def run_block_minimize(c0, n, B, beta, bounds, maxiter=200):
    """Run L-BFGS-B on block representation."""
    if not HAVE_SCIPY:
        return np.asarray(c0, dtype=np.float64)

    c0 = np.asarray(c0, dtype=np.float64).reshape(-1)
    if c0.size != max(0, B - 1):
        tmp = np.zeros(max(0, B - 1), dtype=np.float64)
        m = min(tmp.size, c0.size)
        tmp[:m] = c0[:m]
        c0 = tmp
    c0 = np.clip(c0, -8.0, 8.0)

    options = {"maxiter": maxiter, "ftol": 1e-12, "gtol": 1e-6, "maxls": 30, "maxcor": 15}

    try:
        res = minimize(block_objective, c0, args=(n, B, beta),
                       method="L-BFGS-B", jac=True, bounds=bounds, options=options)
        c = np.asarray(res.x, dtype=np.float64)
    except Exception:
        c = c0

    if not np.all(np.isfinite(c)):
        c = c0
    return np.clip(c, -8.0, 8.0)


def extract_block_weights(p, B):
    """Extract log-block-weights from a full sequence."""
    n = p.size
    L = n // B
    if L == 0:
        return np.zeros(max(0, B - 1), dtype=np.float64)
    blocks = p[:B * L].reshape(B, L).mean(axis=1)
    blocks = np.maximum(blocks, 1e-30)
    z = np.log(blocks) - np.log(blocks[-1])
    return z[:-1].copy()


def block_search(p_init, n, deadline, rng, max_B=512):
    """Run block optimization at increasing resolutions."""
    best_p = p_init.copy()
    best_score = fast_score(p_init)

    candidates = []
    for B in (16, 32, 64, 128, 256, 512):
        if B <= max_B and B < n and (n % B) == 0:
            candidates.append(B)

    for B in candidates:
        if time.monotonic() >= deadline:
            break

        bounds = [(-8.0, 8.0)] * (B - 1)

        # Initialize from current best
        c0 = extract_block_weights(best_p, B)

        # Also try a few random initializations
        inits = [c0]
        for _ in range(3):
            inits.append(np.clip(rng.normal(0, 1.5, B - 1), -8, 8))

        best_c = c0.copy()
        best_B_score = best_score

        for c_init in inits:
            if time.monotonic() >= deadline:
                break

            # Two-stage: coarse then fine
            for beta in (20000.0, 200000.0, 500000.0):
                c = run_block_minimize(c_init, n, B, beta, bounds, maxiter=150)
                p = block_to_p(n, B, c)
                sc = fast_score(p)
                if np.isfinite(sc) and sc < best_B_score:
                    best_B_score = sc
                    best_c = c.copy()
                    c_init = c.copy()

        p = block_to_p(n, B, best_c)
        sc = fast_score(p)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = p.copy()
            print(f"  block B={B} score={best_score:.8f}")

    return best_p, best_score

def local_refine(p, deadline, rng, max_passes=200):

    # Precompute which positions contribute to each b[k]
    # b[k] = sum_j p[j]*p[k-j], so changing p[i] affects b[k] for k in [i, i+n-1]
    # with coefficient p[k-i] (or p[i] on the other side)

    walk_length = min(n // 4, 512)  # max distance for pair transfer
    no_improve = 0
    pass_count = 0

    while pass_count < max_passes and time.monotonic() < deadline:
        pass_count += 1
        improved_this_pass = False
        no_improve += 1

        for _ in range(min(n, 2048)):
            if time.monotonic() >= deadline:
                break

            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue

            # Try transferring epsilon from i to j (or vice versa)
            if p[i] < 1e-10:
                continue

            # Compute current b locally
            cur_max_b = max_b

            # Try: move fraction f from i to j
            for f in (0.1, 0.05, 0.02, -0.1, -0.05, -0.02):
                transfer = f * p[i]
                new_pi = p[i] - transfer
                new_pj = p[j] + transfer
                if new_pi < -1e-15 or new_pj < -1e-15:
                    continue

                p[i] = max(0, new_pi)
                p[j] = max(0, new_pj)

                # Quick check: only recompute affected b positions
                # Full recompute is O(n log n), but for small local changes
                # we can approximate. For safety, do full recompute every few tries.
                # Actually, let's just do a fast check on the max.
                # The affected b[k] are those where i or j contributes.
                # b[k] changes by: -p[j]*p[k-i] + p[k-i]*... this is complex.
                # Just do a full FFT recompute (fast enough for n=8192)

                qb = fft_convolution(p)
                new_max = float(np.max(qb))

                if new_max < cur_max_b - 1e-14:
                    # Accept
                    b = qb
                    max_b = new_max
                    if new_max * 2 * n < best_score - 1e-10:
                        best_score = 2.0 * n * new_max
                    improved_this_pass = True
                    no_improve = 0
                    break
                else:
                    # Reject
                    p[i] = new_pi if new_pi >= 0 else 0  # already set
                    # Undo
                    p[i] = p[i]  # stays at new value, need to undo
                    # Actually let me redo this more carefully

            # Undo if not accepted
            # We need to restore p[i] and p[j]
            # Let me restructure this

        if improved_this_pass:
            p = normalize_simplex(p)
            b = fft_convolution(p)
            max_b = float(np.max(b))
            best_score = 2.0 * n * max_b
        elif no_improve > 30:
            break

    return p, best_score


def local_refine_v2(p, deadline, rng, max_iter=50000):
    """
    Better local refinement using gradient-guided coordinate descent.
    At each step, identify the coordinate with steepest descent direction
    and transfer mass to the coordinate with steepest ascent.
    """
    p = normalize_simplex(p.copy())
    n = p.size
    best_score = fast_score(p)
    best_p = p.copy()

    it = 0
    while it < max_iter and time.monotonic() < deadline:
        it += 1

        b = fft_convolution(p)
        M = float(np.max(b))
        cur_score = 2.0 * n * M

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()

        # Use high beta for near-exact gradient
        _, w = smooth_max_and_weights(b, 100000.0)
        svec = corr_pw(p, w)
        svec = np.maximum(svec, 0.0)
        grad = 2.0 * svec - float(np.mean(2.0 * svec))

        # Find best donor (positive grad, has mass) and recipient (negative grad)
        donor_candidates = np.where((grad > 0) & (p > 1e-8))[0]
        recip_candidates = np.where(grad < 0)[0]

        if len(donor_candidates) == 0 or len(recip_candidates) == 0:
            break

        # Pick the extreme ones
        donor = donor_candidates[np.argmax(grad[donor_candidates])]
        recip = recip_candidates[np.argmin(grad[recip_candidates])]

        # Try transferring mass from donor to recip
        best_gain = 0
        best_eps = 0

        for frac in (0.05, 0.1, 0.2, 0.3, 0.5):
            eps = frac * p[donor]
            new_p = p.copy()
            new_p[donor] -= eps
            new_p[recip] += eps
            new_p = np.maximum(new_p, 0)
            new_p /= new_p.sum()

            nb = fft_convolution(new_p)
            nM = float(np.max(nb))
            gain = M - nM
            if gain > best_gain:
                best_gain = gain
                best_eps = eps

        if best_gain > 1e-14:
            p[donor] -= best_eps
            p[recip] += best_eps
            p = np.maximum(p, 0)
            p /= p.sum()
        else:
            if it > 50:
                break

        if it % 200 == 0:
            print(f"  local it={it} score={best_score:.8f}")

    return best_p, best_score


# ---------- Structured Initializations ----------

def generate_starts(n, rng, prev_p=None, count=6):
    """Generate diverse starting profiles."""
    starts = []

    # Previous best
    if prev_p is not None:
        starts.append(normalize_simplex(prev_p.copy()))

    # Uniform (breaks symmetry slightly)
    p = np.ones(n, dtype=np.float64) / n
    t = np.arange(n, dtype=np.float64)
    p *= 1.0 + 1e-4 * np.sin(0.1 * t)
    starts.append(normalize_simplex(p))

    # Cosine modulation (U-shape)
    x = np.linspace(0, n, n + 1)[:-1]
    for r in (0.3, 0.5, 0.7):
        p = 1.0 + r * np.cos(2.0 * np.pi * x / n)
        p = np.maximum(p, 0.01)
        starts.append(normalize_simplex(p))

    # Linear tilt
    x = np.linspace(0, 1, n)
    for a in (-2.0, 2.0):
        p = np.exp(a * (x - 0.5))
        starts.append(normalize_simplex(p))

    # Random smooth profile
    for _ in range(2):
        n_basis = 8
        coeffs = rng.normal(0, 1, n_basis)
        t = np.linspace(0, 2 * np.pi, n)
        p = np.ones(n, dtype=np.float64)
        for k in range(n_basis):
            p += coeffs[k] * np.cos((k + 1) * t)
        p = np.maximum(p, 0.01)
        starts.append(normalize_simplex(p))

    return starts[:count]


# ---------- Main Search ----------

def official_score(seq):
    """Call the authoritative evaluator."""
    try:
        val = evaluate_sequence(seq)
        val = float(val)
    except Exception:
        return float("inf")
    return val if np.isfinite(val) else float("inf")


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Returns a list of non-negative floats.
    Multi-phase optimization: Nesterov PG -> Block L-BFGS -> Local refine.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 0:
        return [1.0]

    reserve = 60.0
    start_time = time.monotonic()
    deadline = start_time + max(10.0, budget_s - reserve)

    N = 8192
    rng = np.random.default_rng(seed)

    # Get starting point
    prev_p = None
    try:
        hs = height_sequence_1
        if isinstance(hs, list) and len(hs) > 0:
            prev_p = np.array(hs, dtype=np.float64)
            if prev_p.size != N:
                # Resize via interpolation
                if prev_p.size >= 2:
                    new_x = np.linspace(0, 1, N)
                    old_x = np.linspace(0, 1, prev_p.size)
                    prev_p = np.interp(new_x, old_x, prev_p)
            prev_p = normalize_simplex(prev_p)
            print(f"Using height_sequence_1, len={len(hs)}, score={fast_score(prev_p):.8f}")
    except Exception:
        prev_p = None

    # Generate diverse starts
    all_starts = generate_starts(N, rng, prev_p, count=10)
    scored_starts = []
    for p in all_starts:
        sc = fast_score(p)
        if np.isfinite(sc):
            scored_starts.append((sc, p))
    scored_starts.sort(key=lambda x: x[0])

    print(f"Best initial score: {scored_starts[0][0]:.8f}")
    print(f"Starting optimization with budget {budget_s:.0f}s")

    best_p = scored_starts[0][1].copy()
    best_score = scored_starts[0][0]

    # Phase 1: Nesterov Projected Gradient with continuation (50% of time)
    phase1_deadline = start_time + 0.45 * budget_s
    if time.monotonic() < phase1_deadline:
        print("Phase 1: Nesterov PG (continuation)")
        for trial in range(3):
            if time.monotonic() >= phase1_deadline:
                break
            p_start = scored_starts[min(trial, len(scored_starts) - 1)][1]
            p_fit, sc_fit = nesterov_projected_gradient(
                p_start, min(deadline, phase1_deadline + 60), rng,
                beta_start=3000.0, beta_end=800000.0,
                max_iter=500000, verbose=True
            )
            if np.isfinite(sc_fit) and sc_fit < best_score:
                best_score = sc_fit
                best_p = p_fit.copy()
                print(f"Phase 1 trial {trial}: score={best_score:.8f}")

    # Phase 2: Block L-BFGS (30% of time)
    phase2_deadline = start_time + 0.78 * budget_s
    if time.monotonic() < phase2_deadline:
        print("Phase 2: Block L-BFGS")
        max_B = 512 if budget_s > 1500 else 256
        p_fit, sc_fit = block_search(best_p, N, phase2_deadline, rng, max_B=max_B)
        if np.isfinite(sc_fit) and sc_fit < best_score:
            best_score = sc_fit
            best_p = p_fit.copy()
            print(f"Phase 2: score={best_score:.8f}")

    # Phase 3: Local refinement (20% of time)
    phase3_deadline = deadline
    if time.monotonic() < phase3_deadline:
        print("Phase 3: Local refinement")
        # First do a quick Nesterov pass with very high beta
        remaining = phase3_deadline - time.monotonic()
        p_fit, sc_fit = nesterov_projected_gradient(
            best_p, min(phase3_deadline, time.monotonic() + remaining * 0.6), rng,
            beta_start=200000.0, beta_end=1000000.0,
            max_iter=200000, verbose=True
        )
        if np.isfinite(sc_fit) and sc_fit < best_score:
            best_score = sc_fit
            best_p = p_fit.copy()
            print(f"Phase 3 NPG: score={best_score:.8f}")

        # Then local coordinate transfers
        if time.monotonic() < phase3_deadline:
            p_fit, sc_fit = local_refine_v2(best_p, phase3_deadline, rng, max_iter=20000)
            if np.isfinite(sc_fit) and sc_fit < best_score:
                best_score = sc_fit
                best_p = p_fit.copy()
                print(f"Phase 3 Local: score={best_score:.8f}")

    # Final: one more Nesterov pass if time allows
    if time.monotonic() < deadline - 5:
        remaining = deadline - time.monotonic() - 5
        if remaining > 30:
            p_fit, sc_fit = nesterov_projected_gradient(
                best_p, time.monotonic() + remaining, rng,
                beta_start=500000.0, beta_end=2000000.0,
                max_iter=100000, verbose=True
            )
            if np.isfinite(sc_fit) and sc_fit < best_score:
                best_score = sc_fit
                best_p = p_fit.copy()
                print(f"Final NPG: score={best_score:.8f}")

    # Normalize and prepare output
    final_p = normalize_simplex(best_p)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)

    final_seq = [float(x) for x in final_p]

    # Verify with official evaluator
    off = official_score(final_seq)
    print(f"Official score: {off:.8f}")

    if not np.isfinite(off) or off > 2.0 + 1e-6:
        # Fallback: uniform
        return [1.0] * N

    return final_seq
