Strategy:
1. Start from the best known solution (height_sequence_1) or generate new candidates
2. Use multiplicative smoothed-max descent (improved version)

4. Use parametric model optimization (Gaussian bumps) for global exploration
5. Combine results and return the best

import numpy as np

def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(a, b):
    la = len(a)
    lb = len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 1000000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)

def _sanitize_vec(p):
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return p / s

def _exact_score(p):
    p = _sanitize_vec(p)
    if p is None:
        return float("inf")
    m = p.size
    c = _conv_full(p, p)
    if c.size == 0:
        return float("inf")
    c = np.where(np.isfinite(c), c, 0.0)
    return float(2.0 * m * np.max(c))

def _compute_subgradient(p, beta):
    """
    Compute the smoothed subgradient of max(conv(p,p)) * n.
    Returns array of length n (same as p).
    """
    m = p.size
    c = _conv_full(p, p)
    d = m * c  # scale by n
    
    # Softmax weights
    shift = float(np.max(d))
    e = np.exp(beta * (d - shift))
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.ones(len(d), dtype=float) / len(d)
    else:
        w = e / s
    
    # Compute subgradient using FFT cross-correlation
    L = _next_pow2(3 * m)
    fp = np.fft.rfft(p, L)
    fw = np.fft.rfft(w, L)
    
    # first[k] = sum_i p[i] * w[k+i] = ifft(conj(fp) * fw)[k]
    first = np.fft.irfft(np.conjugate(fp) * fw, L)[:m]
    
    # second[k] = sum_i p[i] * w[k-i] (with w[j]=0 for j<0 or j>=2m-1)
    # Use w_rev = w[::-1]
    fw_rev = np.fft.rfft(w[::-1], L)
    second_full = np.fft.irfft(np.conjugate(fp) * fw_rev, L)
    # second[k] = second_full[2*m-2-k]
    second = second_full[2*m-2 : m-1 : -1]
    
    subgrad = first + second
    subgrad = np.where(np.isfinite(subgrad), subgrad, 0.0)
    return np.maximum(subgrad, 0.0)

def _pairwise_move_search(p0, rng, time_limit):

    cur_score = 2.0 * n * float(np.max(c))
    
    stagnant = 0
    
    while time.monotonic() < deadline:
        # Find the peak positions
        c = _conv_full(p, p)
        max_c = float(np.max(c))
        cur_score = 2.0 * n * max_c
        
        # Find positions that contribute most to the peak
        peak_pos = int(np.argmax(c))
        # Contributors: pairs (i, peak_pos - i) where 0 <= i < n and 0 <= peak_pos-i < n
        valid_i = []
        for i in range(n):
            j = peak_pos - i
            if 0 <= j < n:
                valid_i.append(i)
        
        if not valid_i:
            break
        
        valid_i = np.array(valid_i, dtype=int)
        contributions = p[valid_i] * p[peak_pos - valid_i]
        
        # Sort by contribution (largest first)
        order = np.argsort(-contributions)
        top_i = valid_i[order[:min(50, len(order))]]

        # a[k - j] is small for j near the peak

        peak_partners = np.zeros(n)
        for j in range(n):
            kp = peak_pos - j
            if 0 <= kp < n:
                peak_partners[j] = p[kp]
        
        # Sort by partner value (ascending) - we want to move TO positions with low partner
        low_order = np.argsort(peak_partners)
        target_j = low_order[:min(50, n)]

            for j in target_j[:20]:

                    continue

                    p_try[i] -= delta
                    p_try[j] += delta
                    p_try = np.maximum(p_try, floor)
                    p_try = p_try / p_try.sum()
                    
                    sc = _exact_score(p_try)
                    if sc < cur_score - 1e-10:
                        p = p_try
                        cur_score = sc
                        if cur_score < best_score:
                            best = p.copy()
                            best_score = cur_score
                        found_improvement = True
                        break
                if found_improvement:
                    break
            if found_improvement:
                break
        
        if not found_improvement:

                    else:

def _gaussian_bump_model(n, K, rng):
    """Generate a sequence as a sum of K Gaussian bumps."""
    t = np.arange(n, dtype=float) / n
    p = np.zeros(n)
    centers = np.sort(rng.uniform(0.05, 0.95, K))
    sigmas = rng.uniform(0.02, 0.2, K)
    weights = rng.uniform(0.5, 2.0, K)
    
    for k in range(K):
        p += weights[k] * np.exp(-((t - centers[k]) / sigmas[k]) ** 2)
    
    return _sanitize_vec(p)

        t = np.arange(n, dtype=float) / n
        p = np.zeros(n)
        for k in range(K):
            c_k = params[3*k]
            s_k = max(0.005, params[3*k+1])
            w_k = max(0.0, params[3*k+2])
            p += w_k * np.exp(-((t - c_k) / s_k) ** 2)
        
        p = np.where(np.isfinite(p), p, 0.0)
        p = np.clip(p, 0, 1000)
        s = float(p.sum())
        if s <= 1e-10:
            return 1e10
        p = p / s
        return _exact_score(p)
    
    # Initial random parameters
    best_params = None
    best_score = float("inf")
    
    for trial in range(min(5, max(1, int(time_limit / 5)))):
        params0 = []
        centers = np.sort(rng.uniform(0.1, 0.9, K))
        for k in range(K):
            params0.extend([centers[k], float(rng.uniform(0.03, 0.15)), float(rng.uniform(0.5, 2.0))])
        
        result = minimize(
            _score_from_params,
            np.array(params0),
            method='L-BFGS-B',
            options={'maxiter': 50, 'ftol': 1e-8},
        )
        
        sc = _score_from_params(result.x)
        if sc < best_score:
            best_score = sc
            best_params = result.x.copy()
    
    if best_params is None:
        return None, float("inf")
    
    # Reconstruct
    t = np.arange(n, dtype=float) / n
    p = np.zeros(n)
    for k in range(K):
        c_k = best_params[3*k]
        s_k = max(0.005, best_params[3*k+1])
        w_k = max(0.0, best_params[3*k+2])
        p += w_k * np.exp(-((t - c_k) / s_k) ** 2)
    
    p = _sanitize_vec(p)
    if p is None:
        return None, float("inf")
    
    return p, _exact_score(p)

def _block_search(p0, rng, time_limit):
    """Block scaling local search (from previous code, kept as a secondary method)."""
    p = _sanitize_vec(p0)
    if p is None or time_limit <= 0:
        return None, float("inf")
    
    n = int(p.size)
    if n <= 1:
        return p, _exact_score(p)
    
    floor = 1e-16 / float(n)
    p = np.maximum(p, floor)
    p = p / p.sum()
    
    best = p.copy()
    best_score = _exact_score(p)
    cur_score = best_score
    deadline = time.monotonic() + float(time_limit)
    stagnant = 0
    
    factors = (0.3, 0.5, 0.7, 1.0, 1.3, 1.7, 2.5)
    
    while time.monotonic() < deadline:
        r = float(rng.random())
        if r < 0.3:
            w = int(rng.integers(1, max(2, n // 100 + 1)))
        elif r < 0.7:
            w = int(rng.integers(max(2, n // 100), max(3, n // 20 + 1)))
        else:
            low = max(3, n // 40)
            high = max(low + 1, min(n, n // 5 + 1))
            w = int(rng.integers(low, high + 1))
        
        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))
        
        base = p.copy()
        current_best = base
        current_score = cur_score
        
        for f in factors:
            cand = base.copy()
            cand[start:start + w] *= float(f)
            cand = _sanitize_vec(cand)
            if cand is None:
                continue
            cand = np.maximum(cand, floor)
            cand = cand / cand.sum()
            sc = _exact_score(cand)
            if sc < current_score - 1e-12:
                current_score = sc
                current_best = cand
        
        if current_best is not base:
            p = current_best.copy()
            cur_score = current_score
            if cur_score < best_score:
                best = p.copy()
                best_score = cur_score
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 40:
                p_try = _sanitize_vec(p * np.exp(rng.normal(0.0, 0.05, n)))
                if p_try is not None:
                    p_try = np.maximum(p_try, floor)
                    p_try = p_try / p_try.sum()
                    sc = _exact_score(p_try)
                    if sc <= best_score * 1.02:
                        p = p_try
                        cur_score = sc
                    else:
                        p = best.copy()
                        cur_score = best_score
                stagnant = 0
    
    return best, best_score

def _resample(arr, m):
    """Resample a normalized vector to length m by linear interpolation."""
    if m <= 0:
        return None
    a = _sanitize_vec(arr)
    if a is None:
        return None
    n = a.size
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m, dtype=float) / m
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0)
    out = out + 1e-12
    return _sanitize_vec(out)
