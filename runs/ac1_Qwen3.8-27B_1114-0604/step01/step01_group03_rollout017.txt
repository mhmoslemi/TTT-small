    f = rfft(a, L)

    np.maximum(c, 0.0, out=c)

"""
Minimize the self-convolution peak objective for non-negative sequences.
Combines active-set gradient descent, LP-direction polishing, and 
multi-start randomization for global optimization.
"""

import math

import numpy as np

# Use numpy's FFT for speed (scipy.fft is equivalent but let's stick with one)

def _normalize(a):
    """Project to the simplex {a >= 0, sum(a) = 1}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        raise ValueError("Cannot normalize zero vector")
    return a / s

def _self_conv(a, L, m):
    """Compute self-convolution of a (length n) using FFT, returning length m=2n-1."""
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    # Remove tiny negative values from FFT roundoff
    np.maximum(c, 0.0, out=c)
    return c

def _objective(c, n):
    """Compute 2*n*max(c) assuming input is on the simplex."""
    return 2.0 * n * float(np.max(c))

def _active_gradient(a, active_idx, n):
    """
    Compute subgradient of max(conv(a,a)) using the active set.
    active_idx: array of indices where conv is (near) maximum.
    """
    g = np.zeros(n, dtype=np.float64)
    for k in active_idx:
        k = int(k)
        lo = max(0, k - n + 1)
        hi = min(n - 1, k)
        if lo <= hi:
            g[lo:hi+1] += 2.0 * a[k-hi:k-lo+1][::-1]
    return g / float(len(active_idx))

def _cosine_init(n, freq, phase, power, rng):
    """Create a cosine-based initialization."""
    t = np.arange(n, dtype=np.float64) / n
    x = (1.0 + np.cos(2.0 * np.pi * freq * t + phase)) ** power
    x = np.maximum(x, 1e-10)
    # Add small noise to break symmetry
    x *= (1.0 + 0.01 * rng.standard_normal(n))
    return _normalize(x)

def _random_smooth_init(n, rng, width=20):
    """Create a random smooth initialization."""
    z = rng.standard_normal(n)
    k = np.ones(width, dtype=np.float64) / width
    z = np.convolve(z, k, mode='same')
    x = np.exp(z - z.max())
    return _normalize(x)

def _binary_smooth_init(n, rng, width=5):
    """Create a binary-like initialization with smoothing."""
    x = (rng.random(n) > 0.5).astype(np.float64)
    k = np.ones(width, dtype=np.float64) / width
    x = np.convolve(x, k, mode='same')
    return _normalize(x)

def _lp_direction(a, L, m, n, rng, inner_iters=30):
    """
    Compute an LP-like direction using multiplicative weights on the dual.
    The dual is: maximize sum(y) subject to conv(a, y) <= 1, y >= 0.
    We solve approximately with multiplicative weights.
    """
    c = _self_conv(a, L, m)
    c_max = float(np.max(c))
    if c_max < 1e-15:
        return _normalize(a)
    
    # Work with normalized c: c/c_max, so constraint is conv(a, y) <= c_max
    # Dual: max sum(y) s.t. conv(a, y) <= c_max, y >= 0
    # We use multiplicative weights on the dual variables (y over lags 0..2n-2)
    # But that's 2n-1 variables, which is a lot. Instead, we use the primal-dual.
    
    # Simpler approach: use the "importance sampling" method from the SOTA.
    # Let y be a distribution over lags. The constraint is E_y[conv(a, y)] <= c_max.
    # We want to find g (the direction) that maximizes sum(g) subject to conv(a,g) <= c_max.
    # By linearity, g = c_max * y where y is the optimal dual variable... 
    # Actually, the LP is:
    #   max sum(g)
    #   s.t. conv(a, g)[k] <= c_max for all k = 0, ..., 2n-2
    #        g >= 0
    # The dual is:
    #   min c_max * sum(y)
    #   s.t. sum_k y[k] * conv(a, g_component)[k] >= 1 for all directions...
    # This is getting complicated. Let me use the SOTA's approach.
    
    # The SOTA solves: max sum(b) s.t. conv(a, b) <= M, b >= 0
    # via multiplicative weights on the "constraint" side.
    
    # Let's use a simplified version:
    # x = current guess for b (the direction)
    # y = distribution over constraints (lags)
    
    x = np.ones(n, dtype=np.float64) / n
    y = np.ones(m, dtype=np.float64) / m
    avg_x = np.zeros(n, dtype=np.float64)
    
    F = np.fft.rfft(a, L)
    a_reversed = a[::-1]
    F_rev = np.fft.rfft(a_reversed, L)
    
    for it in range(inner_iters):
        # Update y based on current x: put more weight on violated constraints
        X = np.fft.rfft(x, L)
        u = np.fft.irfft(F * X, L)[:m]  # conv(a, x)
        np.maximum(u, 0.0, out=u)
        
        # y ~ exp(eta * u / c_max) (we put more weight on large u)
        eta_y = 5.0 + 20.0 * it / inner_iters
        v = u / c_max
        y = np.exp(eta_y * (v - v.max()))
        y /= y.sum()
        
        avg_x += x
        
        # Update x based on current y
        Y = np.fft.rfft(y, L)
        s = np.fft.irfft(Y * F_rev, L)[(n-1):m]  # This gives the "gradient" for x
        np.maximum(s, 0.0, out=s)
        
        # x ~ exp(eta * s) (we want large x where s is large)
        eta_x = 5.0 + 20.0 * it / inner_iters
        x = np.exp(eta_x * (s - s.max()))
        x = np.maximum(x, 1e-300)
        x /= x.sum()
    
    x = avg_x / inner_iters
    return _normalize(x)

def _line_search(a, direction, L, m, n, best_score):
