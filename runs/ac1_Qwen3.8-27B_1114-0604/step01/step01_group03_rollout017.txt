def _init_structured(n, rng):

    t = np.linspace(0, 1, n)
    # Mix of a bump function and noise
    center = 0.5
    width = rng.uniform(0.15, 0.4)
    base = np.exp(-0.5 * ((t - center) / width) ** 2)
    noise = rng.normal(0, 0.3, n)
    # Smooth noise
    k = max(3, n // 50)
    kernel = np.ones(k) / k
    noise = np.convolve(noise, kernel, mode='same')

    a = np.exp(z - z.max())
    return _normalize(a)

"""
Search for non-negative step heights minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses accelerated multiplicative weight descent with:
  - U-shaped initialization based on asymptotic edge analysis
  - FFT-based soft-max gradient computation
  - Adaptive stepping and temperature annealing
  - Multi-scale search over sequence lengths
  - Final exact-max polishing
"""
import math
import time
import numpy as np


def _normalize(a):
    """Normalize to non-negative with sum 1."""
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def _convolution_fft(a, L):
    """Linear self-convolution via FFT."""
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)
    c = np.maximum(c, 0.0)
    return c


def _grad_fft(a, w, L):
    """
    Compute g[i] = sum_k w[k] * a[k-i] for i in [0, n-1].
    w has length m = 2n-1 (the convolution array).
    Uses the reversal trick: g[i] = conv(a, w_rev)[2n-2-i]
    """
    n = len(a)
    w_rev = w[::-1]
    A = np.fft.rfft(a, L)
    Wr = np.fft.rfft(w_rev, L)
    conv = np.fft.irfft(A * Wr, L)
    # g[i] = conv[2n-2-i] for i in [0, n-1]
    # So g = conv[n-1, n, ..., 2n-2] reversed = conv[n-1:2n-1][::-1]
    g = conv[n-1:2*n-1][::-1]
    g = np.maximum(g, 0.0)
    return g


def _init_u_shape(n):
    """U-shaped initialization: high at edges, low in middle."""
    k = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((k + 0.5) * (n - k + 0.5))
    return _normalize(a)


def _init_gaussian(n, rng):
    """Random Gaussian bump initialization."""
    center = 0.2 + 0.6 * rng.random()
    width = 0.05 + 0.3 * rng.random()
    k = np.arange(n, dtype=np.float64) / n
    a = np.exp(-0.5 * ((k - center) / width) ** 2)
    noise = rng.normal(0, 0.2, n)
    ker = np.ones(max(3, n // 100), dtype=np.float64)
    ker /= ker.sum()
    noise = np.convolve(noise, ker, mode='same')
    a *= (1.0 + 0.3 * noise)
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _init_binary_like(n, rng):
    """Initialize from a random subset (like a Sidon set approximation)."""
    a = np.zeros(n)
    num_ones = max(10, int(n * rng.uniform(0.02, 0.1)))
    positions = rng.choice(n, size=num_ones, replace=False)
    for p in positions:
        a[p] = 1.0
    # Add small background
    a += 0.01
    return _normalize(a)


def _mwd_optimize(a, L, n, time_limit, rng, temp_start=5.0, temp_end=0.2, verbose=False):
    """
    Multiplicative Weight Descent to minimize max(conv(a,a)) s.t. a>=0, sum(a)=1.
    Returns (best_a, best_max_c).
    """
    t0 = time.monotonic()
    deadline = t0 + time_limit
    a = _normalize(a.copy())
    m = 2 * n - 1

    # Initial evaluation
    c = _convolution_fft(a, L)[:m]
    best_score = float(c.max())
    best_a = a.copy()

    eta = 0.3
    temp = temp_start
    no_improve = 0
    it = 0
    patience = 8000

    while time.monotonic() < deadline:
        c = _convolution_fft(a, L)[:m]
        cur = float(c.max())

        if cur < best_score - 1e-10:
            best_score = cur
            best_a
