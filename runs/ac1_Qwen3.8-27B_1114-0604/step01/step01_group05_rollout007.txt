Minimize 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences.

Uses a repulsion-based multiplicative update algorithm combined with

def fft_conv(a):
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0]*a[0]], dtype=np.float64)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F*F, size)[:2*n-1]
    return np.maximum(b, 0.0)

def score_fast(a):
    n = a.size
    S = np.sum(a)
    if S < 1e-15:
        return float('inf')
    b = fft_conv(a)
    return 2.0 * n * np.max(b) / (S*S)

def normalize(a):
    a = np.maximum(a, 0.0)
    S = np.sum(a)
    if S < 1e-15:
        return np.ones_like(a) / a.size
    return a / S

def cross_corr(a, w):
    """Compute T[i] = sum_j a[j] * w[i+j] for i=0..n-1."""
    n = a.size
    m = w.size
    size = 1
    while size < n + m - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(A * WR, size)[:n + m - 1]
    # T[i] = C[m-1-i] for i=0..n-1
    T = C[m-n : m][::-1].copy()
    return T

    T = np.maximum(T, 1e-30)
    
    a_new = a / (T ** gamma)

    return a_new, m

"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences.

Uses a repulsion-based multiplicative update (multiplicative weight / mirror descent
on the simplex) as the primary optimizer, combined with smooth-max gradient
refinement for final polishing. Multi-start from diverse initializations including

import numpy as np
import time


def fft_conv(a):
    """Compute linear convolution a*a via FFT, clipped to non-negative."""
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:2 * n - 1]
    return np.maximum(b, 0.0)


def score_fast(a):
    """Fast score: 2*n*max(b)/S^2."""
    n = a.size
    S = np.sum(a)
    if S < 1e-15:
        return float('inf')
    b = fft_conv(a)
    return 2.0 * n * float(np.max(b)) / (S * S)


def normalize(a):
    """Clip to non-negative and normalize to sum 1."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    S = float(np.sum(a))
    if S < 1e-15:
        return np.full(a.shape, 1.0 / a.size, dtype=np.float64)
    return a / S


def cross_corr(a, w):
    """Compute T[i] = sum_j a[j] * w[i+j] for i=0..n-1 using FFT."""
    n = a.size
    m = w.size
    size = 1
    while size < n + m - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(A * WR, size)[:n + m - 1]
    T = C[m - n:m][::-1].copy()
    return np.maximum(T, 0.0)


def softmax_weights(b, beta):
    """Compute normalized softmax weights emphasizing the peak(s) of b."""
    m = float(np.max(b))
    if m < 1e-300:
        return np.full(b.shape, 1.0 / b.size, dtype=np.float64), m
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z < 1e-300:
        w = np.zeros_like(b)
        w[int(np.argmax(b))] = 1.0
        return w, m
    return e / Z, m

def gradient_refine(a0, deadline, rng, max_iters=500000):
    """Smooth-max projected gradient descent on the probability simplex."""
    a = normalize(a0)
    n = a.size
    best_a = a.copy()

    step = 1.0

    return best_a, best_score


def make_diverse_starts(n, rng):
    """Generate diverse asymmetric starting sequences."""
    starts = []
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)

    # Asymmetric exponentials (the key: asymmetry is needed for score < 2)
    for alpha in (0.5, 1.0, 2.0, 3.0, 5.0, 8.0):
        starts.append(normalize(np.exp(alpha * x)))
        starts.append(normalize(np.exp(-alpha * x)))

    # Skewed Gaussians
    for center in (0.2, 0.3, 0.4, 0.6, 0.7, 0.8):
        for width in (0.1, 0.2, 0.3):
            starts.append(normalize(np.exp(-((x - center) / width) ** 2)))

    # Power law
    for p in (0.3, 0.5, 0.7, 1.5, 2.0, 3.0):
        starts.append(normalize((x + 0.01) ** p))
        starts.append(normalize((1.0 - x + 0.01) ** p))

    # Product of linear and exponential
    for a in (1.0, 2.0, 3.0):
        starts.append(normalize(np.exp(a * x) * (1.0 - x + 0.01)))
        starts.append(normalize(np.exp(-a * x) * (x + 0.01)))

    # Sine-modulated
    for k in (1, 2, 3, 5):
        for amp in (0.3, 0.6, 1.0):
            starts.append(normalize(np.exp(amp * np.sin(2 * np.pi * k * x + 0.5))))

    # Random
    for _ in range(4):
        starts.append(normalize(rng.random(n) ** 2))

    return starts
