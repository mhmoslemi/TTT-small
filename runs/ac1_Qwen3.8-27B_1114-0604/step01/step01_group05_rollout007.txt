"""
Minimizes 2*n*max(a*a)/(sum(a))^2 using multiplicative weights optimization
with beta annealing, diverse parametric initializations, and periodic restarts.
Uses FFT-based convolution for fast evaluation during optimization.
"""

import numpy as np
import time

def fft_convolution(a):
    n = a.size
    if n == 0:
        return np.empty(0)
    if n == 1:
        return np.array([a[0]*a[0]])
    need = 2*n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)

def fast_score(p):
    n = p.size
    S = float(np.sum(p))
    if S <= 0:
        return float('inf')
    b = fft_convolution(p)
    M = float(np.max(b))
    return 2.0 * n * M / (S * S)

def smooth_max_and_weights(b, beta):
    m = float(np.max(b))
    if m <= 0:
        return m, np.ones_like(b) / b.size
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return G, w

def corr_pw(p, w):
    n = p.size
    m = w.size
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m-n:m][::-1].copy()
    return s

def project_simplex(v, target=1.0):
    n = v.size
    if n == 0:
        return v
    low = float(np.min(v)) - 1.0
    high = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (low + high)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0:
        return np.full(n, target / n)
    return q * (target / s)

def normalize(p):
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0:
        return np.full(p.size, 1.0 / p.size)
    return p / s

def mu_step(log_p, pg, eta):
    """Multiplicative weights step in log-space.
    log_p_new = log_p - eta * pg, then normalize via softmax.
    """
    lp_new = log_p - eta * pg
    lp_new = lp_new - np.max(lp_new)
    e = np.exp(lp_new)
    Z = e.sum()
    if Z <= 0:
        return log_p.copy()
    p_new = e / Z
    p_new = np.maximum(p_new, 1e-300)
    p_new = p_new / p_new.sum()
    return p_new

def parametric_starts(n, rng):
    """Generate diverse parametric starting points."""
    starts = []
    x = np.linspace(0, 1, n)
    
    # Uniform
    starts.append(np.ones(n) / n)

        p = np.maximum(p, 0)
        p = p / p.sum()
        starts.append(p)

        p = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        p = p / p.sum()
        starts.append(p)

    for alpha in [2, 5, 10, 20]:
        p = np.exp(-alpha * (x - 0.5) ** 2)
        p = p / p.sum()

            starts.append(p)

        p = p / p.sum()

"""
Minimizes 2*n*max(a*a)/(sum(a))^2 using natural-gradient multiplicative weights
optimization with beta annealing, diverse parametric initializations, and
periodic restarts. Uses FFT-based convolution for fast evaluation.
Key improvements over prior approach:
- Natural gradient in softmax parameterization (better conditioned)
- More diverse parametric starting family
- Adaptive step size with both multiplicative weights and projected gradient
- Aggressive beta annealing schedule

import numpy as np
import time


def fft_convolution(a):
    """Linear convolution a*a via FFT, clipped for numerical safety."""
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def fast_score(p):
    """Fast score using FFT convolution. Assumes sum(p) != 0."""
    n = p.size
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float('inf')
    b = fft_convolution(p)
    M = float(np.max(b))
    return 2.0 * n * M / (S * S)


def smooth_max_and_weights(b, beta):
    """Smooth max approximation and softmax weights."""
    b = np.asarray(b, dtype=np.float64)
    m = float(np.max(b))
    if m <= 0.0 or beta <= 0:
        return m, np.ones_like(b) / b.size
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0 or not np.isfinite(Z):
        idx = int(np.argmax(b))
        w = np.zeros_like(b)
        w[idx] = 1.0
        return m, w
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return G, w


def corr_pw(p, w):
    """s_i = sum_j p_j * w_{i+j} for i=0..n-1 (via FFT)."""
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(0, dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n:m][::-1].copy()
    return s


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        return np.full(n, target / n, dtype=np.float64)
    low = float(np.min(v)) - 2.0
    high = float(np.max(v)) + 2.0
    for _ in range(60):
        mid = 0.5 * (low + high)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0 or not np.isfinite(s):
        return np.full(n, target / n, dtype=np.float64)
    return q * (target / s)


def normalize(p):
    """Clip to non-negative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0:
        return np.full(p.size, 1.0 / p.size, dtype=np.float64)
    return p / s


def natural_gradient_mu_step(p, g, eta):
    """
    Multiplicative weights step using natural gradient in log-space.
    p_new = softmax(log(p) - eta * p * (g - g^T p))
    """
    gp = float(np.dot(g, p))
    nat_grad = p * (g - gp)
    # Clip to prevent overflow
    nat_grad = np.clip(nat_grad, -50.0 / max(eta, 1e-10), 50.0 / max(eta, 1e-10))
    log_p = np.log(np.maximum(p, 1e-300))
    log_p_new = log_p - eta * nat_grad
    log_p_new -= float(np.max(log_p_new))
    e = np.exp(log_p_new)
    Z = float(np.sum(e))
    if Z <= 0 or not np.isfinite(Z):
        return p.copy()
    p_new = e / Z
    p_new = np.maximum(p_new, 1e-300)
    p_new = p_new / float(np.sum(p_new))
    return p_new


def projected_gradient_step(p, g, eta):
    """Standard projected gradient step on the simplex."""
    gp = float(np.dot(g, p))
    pg = g - gp
    q = project_simplex(p - eta * pg, 1.0)
    return q


def parametric_starts(n, rng):
    """Generate diverse parametric starting points."""
    starts = []
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)

    # Uniform (score = 2)
    starts.append(np.ones(n, dtype=np.float64) / n)

    # Cosine to various powers (smooth bump)
    for power in [0.3, 0.5, 1.0, 1.5, 2.0, 3.0, 5.0]:
        base = np.cos(np.pi * (x - 0.5))
        base = np.maximum(base, 0)
        p = base ** power
        s = p.sum()
        if s > 0:
            starts.append(p / s)

    # Gaussian with various widths
    for sigma in [0.05, 0.1, 0.15, 0.2, 0.25, 0.3, 0.4, 0.5]:
        p = np.exp(-0.5 * ((x - 0.5) / sigma) ** 2)
        p = p / p.sum()
        starts.append(p)

    # Power-law decay from center
    for alpha in [0.5, 1.0, 1.5, 2.0, 3.0]:
        p = (1.0 - 2.0 * np.abs(x - 0.5)) ** alpha
        p = np.maximum(p, 0)
        s = p.sum()
        if s > 0:
            starts.append(p / s)

    # Two-bump
    for gap in [0.1, 0.15, 0.2, 0.25]:
        for width in [0.03, 0.05, 0.08]:
            c1 = 0.5 - gap
            c2 = 0.5 + gap
            p = np.exp(-0.5 * ((x - c1) / width) ** 2) + np.exp(-0.5 * ((x - c2) / width) ** 2)
            p = p / p.sum()
            starts.append(p)

    # Sine perturbations of uniform
    for k in [1, 2, 3, 5, 8]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            p = 1.0 + amp * np.sin(2 * np.pi * k * x)
            p = np.maximum(p, 1e-8)
            p = p / p.sum()
            starts.append(p)

    # Multiple sine (Fourier series)
    for _ in range(4):
        coeffs = rng.normal(size=10) * 0.3
        p = np.ones(n)
        for j, c in enumerate(coeffs):
            if j > 0:
                p += c * np.cos(2 * np.pi * j * x)
        p = np.maximum(p, 1e-6)
        p = p / p.sum()
        starts.append(p)

    # Random spectral (band-limited noise)
    size = 1
    while size < n:
        size <<= 1
    for bandwidth in [0.1, 0.2, 0.3, 0.5]:
        zc = np.zeros(size // 2 + 1, dtype=complex)

            z = z - z.mean()
            std = z.std()
            if std > 1e-10:
                z = 1.5 * np.tanh(z / std)
                p = np.exp(z)
                p = p / p.sum()
                starts.append(p)

    return starts


def optimize_natural_gw(p0, deadline, beta_start, beta_end, eta_init, rng):
    """
    Natural gradient multiplicative weights optimizer with beta annealing.
    Alternates between MW steps and projected gradient steps.
    """
    n = p0.size
    p = normalize(p0.copy())
    if n == 0:
        return p, float('inf')

    best_p = p.copy()
    best_score = 2.0 * n * float(np.max(fft_convolution(p)))
    eta = eta_init
    total_iters = 0
    no_improve = 0
    cur_score = best_score

    while time.monotonic() < deadline:
        total_iters += 1

        # Beta annealing: exponential schedule
        progress = min(1.0, total_iters / 8000.0)
        beta = beta_start * (beta_end / beta_start) ** progress

        b = fft_convolution(p)
        G, w = smooth_max_and_weights(b, beta)
        if not np.isfinite(G):
            break

        g = 2.0 * corr_pw(p, w)
        g = np.asarray(g, dtype=np.float64)
        if not np.all(np.isfinite(g)):
            break

        cur_score = 2.0 * n * float(np.max(b))

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 300:
            # Try a perturbation to escape
            if total_iters % 100 == 0:
                perturb = 0.01 * rng.normal(size=n)
                p = normalize(p + perturb)
                no_improve = 0
            elif no_improve > 500:
                break

        if total_iters % 1000 == 0:
            print(f"    it={total_iters} score={best_score:.8f} eta={eta:.2e} beta={beta:.0f}")

        # Try natural gradient MW step with backtracking
        accepted = False
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01]:
            step_eta = eta * mult
            if step_eta < 1e-12:
                break
            p_cand = natural_gradient_mu_step(p, g, step_eta)
            b_cand = fft_convolution(p_cand)
            sc_cand = 2.0 * n * float(np.max(b_cand))
            if sc_cand < cur_score - 1e-10:
                p = p_cand
                b = b_cand
                cur_score = sc_cand
                accepted = True
                break

        # If MW step failed, try projected gradient
        if not accepted:
            for mult in [1.0, 0.5, 0.1, 0.01]:
                step_eta = eta * mult
                if step_eta < 1e-12:
                    break
                p_cand = projected_gradient_step(p, g, step_eta)
                b_cand = fft_convolution(p_cand)
                sc_cand = 2.0 * n * float(np.max(b_cand))
                if sc_cand < cur_score - 1e-10:
                    p = p_cand
                    b = b_cand
                    cur_score = sc_cand
                    accepted = True
                    break

        # Adaptive step size
        if accepted:
            if eta < eta_init * 0.5:
                eta = min(eta_init, eta * 1.5)
        else:
            eta *= 0.5
            if eta < 1e-10:
                # Last resort: random nudge
                p = normalize(p + 0.001 * rng.normal(size=n))
                eta = eta_init * 0.1

    return best_p, best_score


def coordinate_refinement(p, deadline, beta=50000.0):
    """
    Local refinement: try small perturbations to individual coordinates
    or small groups, accepting if it reduces the score.
    """
    n = p.size
    best_p = p.copy()
    best_score = 2.0 * n * float(np.max(fft_convolution(p)))

    rng_local = np.random.default_rng(12345)

    it = 0
    while time.monotonic() < deadline:
        it += 1

        # Pick a random group of coordinates to perturb

        idx = rng_local.choice(n, size=group_size, replace=False)

        # Try increasing and decreasing these coordinates
        for direction in [1, -1]:
            for magnitude in [0.01, 0.005, 0.001]:

                sc_cand = 2.0 * n * float(np.max(b_cand))
                if sc_cand < best_score - 1e-12:
                    best_score = sc_cand
                    best_p = p_cand.copy()
                    p = p_cand.copy()
                    break
        else:
            continue
        break

        if it % 500 == 0:
            print(f"    coord refin it={it} score={best_score:.8f}")

    return best_p, best_score

    for p in starts:
        p = normalize(p)
        sc = 2.0 * N * float(np.max(fft_convolution(p)))
        if np.isfinite(sc):
            scored.append((sc, p))
    scored.sort(key=lambda x: x[0])

    print("  Top 5 starts:")
    for i, (sc, _) in enumerate(scored[:5]):
        print(f"    {i}: {sc:.8f}")

    best_score = scored[0][0] if scored else float('inf')
    best_p = scored[0][1].copy() if scored else np.ones(N) / N

    # Incorporate provided construction if available
    try:
        h = height_sequence_1
        if isinstance(h, list) and len(h) > 0:
            hp = np.array(h, dtype=np.float64)
            hp = normalize(hp)
            hp_b = fft_convolution(hp)
            hp_score = 2.0 * len(hp) * float(np.max(hp_b))
            print(f"  height_sequence_1 (len={len(hp)}): score={hp_score:.8f}")
            if hp_score < best_score:
                # Resize to N if needed
                if len(hp) == N:
                    best_score = hp_score
                    best_p = hp.copy()
                else:
                    # Interpolate to length N
                    old_x = np.linspace(0, 1, len(hp))
                    new_x = np.linspace(0, 1, N)
                    hp_interp = np.interp(new_x, old_x, hp)
                    hp_interp = normalize(hp_interp)
                    hp_interp_b = fft_convolution(hp_interp)
                    hp_interp_score = 2.0 * N * float(np.max(hp_interp_b))
                    print(f"  Interpolated to N={N}: score={hp_interp_score:.8f}")
                    if hp_interp_score < best_score:
                        best_score = hp_interp_score
                        best_p = hp_interp.copy()
            elif np.isfinite(hp_score):
                scored.append((hp_score, hp))
                scored.sort(key=lambda x: x[0])
    except Exception:
        pass

    # Main optimization: run natural gradient from top starts
    top_k = 5
    elapsed = time.monotonic() - start_time
    remaining_time = deadline - time.monotonic()
    time_per_start = max(20.0, remaining_time * 0.7 / top_k)

    print(f"\nMain optimization phase: {top_k} starts, ~{time_per_start:.0f}s each")

    for i in range(min(top_k, len(scored))):
        remaining = deadline - time.monotonic()
        if remaining < 30.0:
            break

        sc0, p0 = scored[i]
        alloc = min(time_per_start, remaining - 20.0)
        if alloc < 15.0:
            break

        sub_deadline = time.monotonic() + alloc
        print(f"\n  Start {i} (score={sc0:.8f}):")

        p_fit, sc_fit = optimize_natural_gw(
            p0, sub_deadline,
            beta_start=50.0,
            beta_end=200000.0,
            eta_init=0.05,
            rng=rng
        )

        if sc_fit < best_score:
            best_score = sc_fit
            best_p = p_fit.copy()
            print(f"  ** NEW GLOBAL BEST: {best_score:.8f}")
        else:
            print(f"  Best from start {i}: {sc_fit:.8f}")

    # Secondary round: optimize from the current best with different hyperparams
    remaining = deadline - time.monotonic()
    if remaining > 40.0:
        print(f"\nSecondary optimization round ({remaining:.0f}s remaining):")
        for eta_init in [0.02, 0.1, 0.01]:
            remaining = deadline - time.monotonic()
            if remaining < 20.0:
                break
            alloc = min(remaining * 0.35, 120.0)
            sub_deadline = time.monotonic() + alloc

            # Slight perturbation of best
            p_pert = normalize(best_p + 0.005 * rng.normal(size=N))

            p_fit, sc_fit = optimize_natural_gw(
                p_pert, sub_deadline,
                beta_start=1000.0,
                beta_end=300000.0,
                eta_init=eta_init,
                rng=rng
            )

            if sc_fit < best_score:
                best_score = sc_fit
                best_p = p_fit.copy()
                print(f"  ** polished (eta={eta_init}): {best_score:.8f}")
            else:
                print(f"  polish eta={eta_init}: {sc_fit:.8f} (no improvement)")

    # Final: coordinate-level micro-refinement
    remaining = deadline - time.monotonic()
    if remaining > 15.0:
        print(f"\nCoordinate refinement ({remaining:.0f}s):")
        sub_deadline = time.monotonic() + max(10.0, remaining - 10.0)
        p_fit, sc_fit = coordinate_refinement(best_p, sub_deadline, beta=100000.0)
        if sc_fit < best_score:
            best_score = sc_fit
            best_p = p_fit.copy()
            print(f"  coord refined: {best_score:.8f}")

    # Final cleanup
    final_p = normalize(best_p)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_seq = [float(x) for x in final_p]

    # Verify with official evaluator
    try:
        off = evaluate_sequence(final_seq)
        off = float(off)
    except Exception:
        off = float('inf')

    print(f"\nFinal official score: {off}")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    # Fallback
    if not np.isfinite(off) or off > 2.0 + 1e-9:
        return [1.0] * N

    return final_seq
