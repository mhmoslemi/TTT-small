import numpy as np

def _autocorr_fft(f):
    """Full linear convolution f*f via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _gradient_softmax(c, f, p):
    """Gradient of smooth max: grad[j] = 2 * sum_k p[j+k] * f[k]."""
    n = len(f)
    # p has length 2n-1, f has length n
    # grad[j] = 2 * sum_{k=0}^{n-1} p[j+k] * f[k]
    # This is a cross-correlation
    grad = np.zeros(n)
    for j in range(n):
        grad[j] = 2.0 * np.dot(p[j:j+n], f)
    return grad


def _gradient_softmax_fast(c, f, p):
    """Fast gradient via FFT: grad[j] = 2 * sum_{k=0}^{n-1} p[j+k] * f[k]."""
    n = len(f)
    # We need the cross-correlation of p (length 2n-1) with f (length n)
    # Specifically: (p cross f)[j] = sum_k p[j+k] * f[k] for k=0..n-1
    # This is equivalent to convolution of reversed f with p
    f_rev = f[::-1]
    size = 1 << (3 * n).bit_length()
    F_rev = np.fft.rfft(f_rev, size)
    P = np.fft.rfft(p, size)
    conv = np.fft.irfft(F_rev * P, size)
    # conv[m] = sum_i f_rev[i] * p[m-i] = sum_i f[n-1-i] * p[m-i]
    # Let j = n-1-i: conv[m] = sum_j f[j] * p[m-n+1+j]
    # We want grad[j] = sum_k p[j+k] * f[k], which is sum_k f[k] * p[j+k]
    # conv[m] with m = j + n - 1: sum_k f[k] * p[j+k]
    # So grad[j] = conv[j + n - 1]
    grad = conv[n - 1:n - 1 + n]
    return 2.0 * grad


def _soft_max_gradient(f, T):
    """Compute smooth gradient of max(conv(f)) with respect to f."""
    n = len(f)
    c = _autocorr_fft(f)
    cs = c / T
    cs -= cs.max()
    exp_cs = np.exp(cs)
    p = exp_cs / exp_cs.sum()
    grad = _gradient_softmax_fast(c, f, p)
    return p, grad, c


def _mirror_step(f, grad, step):
    """Exponentiated gradient step on the simplex."""
    span = grad.max() - grad.min()
    if span < 1e-15:
        return f
    eta = step / span
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    return f_new / s

    else:
        f = f / s

    local_best = _eval(f)
    local_best_f = f.copy()

    # Divide into blocks
    n_blocks = min(20, n // 10)
    if n_blocks < 2:
        n_blocks = 2
    block_size = n // n_blocks

    # Randomly select pairs of blocks to optimize
    n_iters = 10000
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        # Pick two blocks
        b1 = int(rng.integers(0, n_blocks))
        b2 = int(rng.integers(0, n_blocks))
        if b1 == b2:
            continue

        start1 = b1 * block_size
        end1 = min(start1 + block_size, n)
        start2 = b2 * block_size
        end2 = min(start2 + block_size, n)

        # Try redistributing mass between these blocks
        total_mass = f[start1:end1].sum() + f[start2:end2].sum()
        if total_mass < 1e-15:
            continue

        # Try different splits
        for alpha in [0.3, 0.5, 0.7]:
            f_try = f.copy()
            target1 = total_mass * alpha
            target2 = total_mass * (1 - alpha)

            block1 = f_try[start1:end1].copy()
            block2 = f_try[start2:end2].copy()

            s1 = block1.sum()
            s2 = block2.sum()

            if s1 > 1e-15:
                block1 = block1 / s1 * target1
            else:
                block1 = np.ones(end1 - start1) * target1 / (end1 - start1)

            if s2 > 1e-15:
                block2 = block2 / s2 * target2
            else:
                block2 = np.ones(end2 - start2) * target2 / (end2 - start2)

            f_try[start1:end1] = block1
            f_try[start2:end2] = block2

            c_try = _autocorr_fft(f_try)
            if c_try.max() < local_best * 0 + _autocorr_fft(local_best_f).max() - 1e-14:
                f = f_try
                local_best = _eval(f)
                local_best_f = f.copy()
                break

    return local_best_f, local_best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the non-negative sequence minimizing 2*n*max(f*f)/(sum f)^2.
    Uses multiple optimization strategies: softmax mirror descent, subgradient
    targeting, coordinate descent, and block optimization.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 30.0

    f_init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(f_init)
    print(f"Initial sequence length: {n}")

    # Normalize
    s0 = f_init.sum()
    if s0 > 1e-10:
        f_init = f_init / s0
    else:
        f_init = np.ones(n) / n

    best_score = _eval(f_init)
    best_f = f_init.copy()
    print(f"Initial score: {best_score:.6f}")

    # Check uniform
    f_uni = np.ones(n) / n
    score_uni = _eval(f_uni)
    print(f"Uniform score: {score_uni:.6f}")

    # Generate diverse starting points
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    starts = {}
    starts['given'] = f_init.copy()
    starts['uniform'] = np.ones(n)
    starts['triang'] = 1.0 - np.abs(j - center) / (center + 1)
    starts['triang2'] = (1.0 - np.abs(j - center) / (center + 1)) ** 2
    starts['exp_left'] = np.exp(-0.5 * j / n)
    starts['exp_right'] = np.exp(-0.5 * (n - 1 - j) / n)
    starts['exp_both'] = np.exp(-1.0 * j / n) + np.exp(-1.0 * (n - 1 - j) / n)
    starts['ramp_up'] = j / n + 0.1
    starts['ramp_down'] = (n - 1 - j) / n + 0.1
    starts['u_shape'] = 1.0 + 5.0 * ((j - center) / center) ** 2
    starts['w_shape'] = 1.0 + 3.0 * (np.cos(4 * np.pi * j / n) ** 2)
    starts['sine'] = 1.0 + 0.5 * np.sin(2 * np.pi * j / n)
    starts['sine2'] = 1.0 + 0.3 * np.sin(6 * np.pi * j / n) + 0.2 * np.sin(10 * np.pi * j / n)
    starts['random_exp'] = rng.exponential(1.0, size=n)
    starts['random_dir'] = rng.dirichlet(np.ones(n) * 0.1)
    starts['asym_exp'] = np.exp(-2.0 * j / (0.3 * n)) + 0.01

    # Evaluate all starts
    start_results = []
    for name, f0 in starts.items():
        f = np.maximum(f0, 0.0)
        s = f.sum()
        if s < 1e-10:
            f = np.ones(n) / n
        else:
            f = f / s
        sc = _eval(f)
        start_results.append((sc, name, f))

    start_results.sort(key=lambda x: x[0])
    print("\nStarting point scores:")
    for sc, name, _ in start_results:
        print(f"  {name}: {sc:.6f}")

    for sc, name, f0 in start_results:
        if sc < best_score:
            best_score = sc
            best_f = f0.copy()

    # Phase 1: Mirror descent from top starts
    print("\n=== Phase 1: Mirror Descent ===")
    top_starts = start_results[:6]
    time_phase1 = min(900.0, (budget_s - 30) * 0.35)
    time_per_start = time_phase1 / len(top_starts)

    for idx, (sc, name, f0) in enumerate(top_starts):
        if time.monotonic() > deadline - 300:
            break
        t_alloc = time_per_start
        start_deadline = time.monotonic() + t_alloc
        print(f"\n  Mirror descent from '{name}' (score={sc:.6f}, time={t_alloc:.0f}s)")

        T_init = _autocorr_fft(f0).max() * 0.3
        f_opt, sc_opt = _optimize_mirror(f0, n, start_deadline, rng, T_init=T_init,
                                         n_iters=30000, label=name)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"    New best: {best_score:.6f}")

    # Phase 2: Subgradient from best
    print(f"\n=== Phase 2: Subgradient (current best={best_score:.6f}) ===")
    if time.monotonic() < deadline - 300:
        t_alloc = min(400.0, deadline - time.monotonic() - 300)
        start_deadline = time.monotonic() + t_alloc
        print(f"  Time budget: {t_alloc:.0f}s")
        f_opt, sc_opt = _optimize_subgradient(best_f, n, start_deadline, rng, label="subgrad")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"    New best: {best_score:.6f}")
        else:
            print(f"    No improvement: {sc_opt:.6f}")

    # Phase 3: Another round of mirror descent from current best with different T
    print(f"\n=== Phase 3: Mirror Descent Round 2 (current best={best_score:.6f}) ===")
    if time.monotonic() < deadline - 200:
        t_alloc = min(400.0, deadline - time.monotonic() - 200)
        start_deadline = time.monotonic() + t_alloc
        print(f"  Time budget: {t_alloc:.0f}s")

        # Try with a colder start
        T_cold = _autocorr_fft(best_f).max() * 0.1
        f_opt, sc_opt = _optimize_mirror(best_f, n, start_deadline, rng, T_init=T_cold,
                                         n_iters=40000, label="cold")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"    New best: {best_score:.6f}")

        # Try from a perturbed version
        if time.monotonic() < deadline - 100:
            f_pert = best_f + 0.01 / np.sqrt(n) * rng.standard_normal(n)
            f_pert = np.maximum(f_pert, 0.0)
            f_pert /= f_pert.sum()
            t_alloc2 = min(200.0, deadline - time.monotonic() - 100)
            f_opt, sc_opt = _optimize_mirror(f_pert, n, time.monotonic() + t_alloc2, rng,
                                             T_init=_autocorr_fft(f_pert).max() * 0.3,
                                             n_iters=20000, label="perturbed")
            if sc_opt < best_score:
                best_score = sc_opt
                best_f = f_opt.copy()
                print(f"    New best (perturbed): {best_score:.6f}")

    # Phase 4: Coordinate descent polish
    print(f"\n=== Phase 4: Coordinate Descent Polish (current best={best_score:.6f}) ===")
    polish_deadline = min(time.monotonic() + 300.0, deadline)
    f_pol, sc_pol = _coordinate_polish(best_f, n, polish_deadline, rng)
    if sc_pol < best_score:
        best_score = sc_pol
        best_f = f_pol.copy()
        print(f"  Polished to: {best_score:.6f}")

    # Phase 5: Final short optimization
    print(f"\n=== Phase 5: Final Optimization (current best={best_score:.6f}) ===")
    if time.monotonic() < deadline:
        t_final = min(150.0, deadline - time.monotonic())
        if t_final > 30:
            f_opt, sc_opt = _optimize_mirror(best_f, n, time.monotonic() + t_final, rng,
                                             T_init=_autocorr_fft(best_f).max() * 0.05,
                                             n_iters=20000, label="final")
            if sc_opt < best_score:
                best_score = sc_opt
                best_f = f_opt.copy()
                print(f"  Final: {best_score:.6f}")

            f_pol2, sc_pol2 = _coordinate_polish(best_f, n, min(time.monotonic() + 60.0, deadline), rng)
            if sc_pol2 < best_score:
                best_score = sc_pol2
                best_f = f_pol2.copy()
                print(f"  Final polish: {best_score:.6f}")

    # Ensure valid output
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = _eval(best_f)
    if not np.isfinite(best_score) or final_score < best_score:
        best_score = final_score

    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f}")
    print(f"Sequence length: {n}")
    print(f"Sum: {best_f.sum():.6f}")
    print(f"Max: {best_f.max():.6f}")
    print(f"{'='*50}")

    return [float(x) for x in best_f]
