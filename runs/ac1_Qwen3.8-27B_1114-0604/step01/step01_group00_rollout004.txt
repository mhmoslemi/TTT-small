def project_simplex(v):
    """Project v onto {x >= 0, sum x = 1} in O(n)."""
    n = len(v)
    v_sum = v.sum()
    if v_sum < 1e-15:
        return np.ones(n) / n

    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1.0)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def soft_gradient(c, f, T, top_k=0):

    Compute gradient direction: 2 * sum_k p_k * S_k(f).

            break

        progress = it / n_iters

        # Piecewise temperature and step schedule
        if progress < 0.2:
            T = peak0 * (0.5 - 2.0 * progress)
            step = 3.0
            top_k = 0
        elif progress < 0.5:
            T = peak0 * (0.1 - 0.3 * (progress - 0.2) / 0.3)
            step = 1.5
            top_k = 20
        elif progress < 0.8:
            T = peak0 * (0.01 - 0.06 * (progress - 0.5) / 0.3)
            step = 0.5
            top_k = 5
        else:
            T = peak0 * max(1e-7, 0.005 * (1.0 - progress) / 0.2)
            step = 0.1
            top_k = 3

        c = autocodev(f)
        grad = compute_gradient(c, f, T, top_k=top_k)

        # Mirror descent step
        span = grad.max() - grad.min()
        if span < 1e-15:
            span = 1e-15
        eta = step / span
        log_f = np.log(f + 1e-300) - eta * grad
        log_f -= log_f.max()
        f_new = np.exp(log_f)
        s_new = f_new.sum()
        if s_new < 1e-300:
            f = project_simplex(f + 0.001 * rng.standard_normal(n))
        else:
            f = f_new / s_new

        # Periodic evaluation and stuck detection
        if it % 2000 == 0:
            score = evaluate_sequence(f.tolist())
            if score < local_best - 1e-8:
                local_best = score
                local_best_f = f.copy()
                stuck = 0
            else:
                stuck += 1

            if stuck > 5:
                # Perturbation
                mag = 0.01 / np.sqrt(n)
                f = f + mag * rng.standard_normal(n)
                f = project_simplex(f)
                stuck = 0

    final_score = evaluate_sequence(f.tolist())
    if final_score < local_best:
        local_best = final_score
        local_best_f = f.copy()

    return local_best_f, local_best


def optimize_projected(f_init, n, deadline, rng, label=""):
    """
    Projected gradient descent with backtracking line search.
    """
    f = np.maximum(f_init, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f = f / s

    c = autocodev(f)
    peak0 = c.max()
    local_best = evaluate_sequence(f.tolist())
    local_best_f = f.copy()

    n_iters = 100000
    stuck = 0

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        progress = it / n_iters
        if progress < 0.3:
            T = peak0 * 0.3
            top_k = 0
        elif progress < 0.6:
            T = peak0 * 0.1
            top_k = 10
        else:
            T = peak0 * 0.01
            top_k = 3

        c = autocodev(f)
        current_peak = c.max()
        grad = compute_gradient(c, f, T, top_k=top_k)

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break

        # Backtracking
        eta = 5.0 / (gnorm * np.sqrt(n))
        eta = min(eta, 5.0)
        improved = False
        for bt in range(20):
            f_trial = project_simplex(f - eta * grad)
            c_trial = autocodev(f_trial)
            if c_trial.max() < current_peak * (1.0 - 1e-12):
                f = f_trial
                improved = True
                break
            eta *= 0.5

        if not improved:
            mag = 0.003 / np.sqrt(n)
            f = project_simplex(f + mag * rng.standard_normal(n))
            stuck += 1
        else:
            stuck = 0

        if stuck > 8:
            f = project_simplex(f + 0.01 / np.sqrt(n) * rng.standard_normal(n))
            stuck = 0

        if it % 3000 == 0:
            score = evaluate_sequence(f.tolist())
            if score < local_best:
                local_best = score
                local_best_f = f.copy()
                stuck = 0

    final_score = evaluate_sequence(f.tolist())
    if final_score < local_best:
        local_best = final_score
        local_best_f = f.copy()

    return local_best_f, local_best


def polish_peak(f_init, n, deadline, rng):
    """
    Greedy peak reduction: identify peak lags, find donor/receiver pairs,
    and transfer mass to reduce the peak.
    """
    f = f.copy()
    c = autocodev(f)
    local_best = evaluate_sequence(f.tolist())
    local_best_f = f.copy()
    peak = c.max()

    n_passes = 2000
    no_improve = 0

    for p in range(n_passes):
        if time.monotonic() > deadline:
            break

        c = autocodev(f)
        peak = c.max()

        # Find top peak lags
        n_peaks = min(5, 2 * n - 1)
        top_lags = np.argsort(c)[::-1][:n_peaks]

        improved = False

        for k_star in top_lags:
            k_star = int(k_star)

            # Compute contribution of each position to this peak
            contrib = np.zeros(n)
            valid = np.arange(n)
            idx = k_star - valid
            mask = (idx >= 0) & (idx < n)
            contrib[mask] = f[mask] * f[idx[mask]]

            # Donor: highest contribution
            j_don = int(np.argmax(contrib))
            if contrib[j_don] < 1e-15:
                continue

            # Receiver: lowest contribution, also check it doesn't create new peaks
            # Use a weighted criterion: low contribution at this lag AND low values
            rec_score = contrib + 0.1 * f
            j_rec = int(np.argmin(rec_score))

            if j_don == j_rec:
                # Second choice
                rec_sorted = np.argsort(rec_score)
                for candidate in rec_sorted[:10]:
                    if candidate != j_don:
                        j_rec = int(candidate)
                        break
                else:
                    continue

            # Try multiple transfer amounts
            best_frac = None
            best_peak = peak
            for frac in [0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3]:
                amt = f[j_don] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[j_don] -= amt
                f_try[j_rec] += amt
                f_try = project_simplex(f_try)
                c_try = autocodev(f_try)
                if c_try.max() < best_peak:
                    best_peak = c_try.max()
                    best_frac = frac

            if best_frac is not None and best_peak < peak - 1e-14:
                amt = f[j_don] * best_frac
                f[j_don] -= amt
                f[j_rec] += amt
                f = project_simplex(f)
                peak = best_peak
                improved = True

        # Random pair moves for diversity
        for _ in range(30):
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            for frac in [0.01, 0.02, 0.05]:
                amt = f[i] * frac
                if amt < 1e-15:
                    break
                f_try = f.copy()
                f_try[i] -= amt
                f_try[j] += amt
                f_try = project_simplex(f_try)
                c_try = autocodev(f_try)
                if c_try.max() < peak - 1e-14:
                    f = f_try
                    peak = c_try.max()
                    improved = True
                    break

        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                # Restart with perturbation from best
                f = local_best_f + 0.005 / np.sqrt(n) * rng.standard_normal(n)
                f = project_simplex(f)
                no_improve = 0

        if p % 200 == 0:
            score = evaluate_sequence(f.tolist())
            if score < local_best:
                local_best = score
                local_best_f = f.copy()

    final_score = evaluate_sequence(f.tolist())
    if final_score < local_best:
        local_best = final_score
        local_best_f = f.copy()

    return local_best_f, local_best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: multi-phase optimization combining mirror descent,
    projected gradient, and greedy peak reduction.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 10.0

    f_init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(f_init)

    s0 = f_init.sum()
    if s0 < 1e-10:
        f_init = np.ones(n)
    else:
        f_init = f_init / s0

    print(f"n = {n}, initial sum = {s0:.4f}")
    init_score = evaluate_sequence(f_init.tolist())
    print(f"Initial score: {init_score:.6f}")

    # Generate diverse starting points
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    xc = (j - center) / center  # normalized to [-1, 1]

    starts = {}
    starts['given'] = f_init.copy()
    starts['uniform'] = np.ones(n)

    # U-shapes
    for alpha in [0.3, 1.0, 2.0, 5.0, 10.0, 20.0]:
        starts[f'u{alpha}'] = 1.0 + alpha * xc ** 2

    # Edge-weighted
    starts['edge1'] = np.abs(xc) + 0.05
    starts['edge2'] = xc ** 2 + 0.01
    starts['edge3'] = xc ** 4 + 0.005

    # Bimodal / multimodal
    starts['bimodal'] = (np.exp(-((j - 0.15 * n)) ** 2 / (0.04 * n ** 2)) +
                         np.exp(-((j - 0.85 * n)) ** 2 / (0.04 * n ** 2)) + 0.01)
    starts['trimodal'] = (np.exp(-((j - 0.1 * n)) ** 2 / (0.03 * n ** 2)) +
                          np.exp(-((j - 0.5 * n)) ** 2 / (0.03 * n ** 2)) +
                          np.exp(-((j - 0.9 * n)) ** 2 / (0.03 * n ** 2)) + 0.005)

    # Cosine-based (non-negative by construction)
    starts['cos1'] = 0.5 + 0.5 * np.cos(np.pi * xc)
    starts['cos2'] = 0.3 + 0.7 * np.cos(np.pi * xc) ** 2
    starts['cos3'] = 1.0 + 0.5 * np.cos(2 * np.pi * xc)

    # Random
    starts['rand_exp'] = rng.exponential(1.0, size=n)
    starts['rand_dir1'] = rng.dirichlet(np.ones(n) * 0.05)
    starts['rand_dir2'] = rng.dirichlet(np.ones(n) * 0.3)
    starts['rand_gauss'] = np.abs(rng.standard_normal(n)) + 0.01

    # Evaluate all starts
    start_results = []
    for name, f0 in starts.items():
        f = np.maximum(f0, 0.0)
        s = f.sum()
        if s < 1e-10:
            f = np.ones(n)
        else:
            f = f / s
        sc = evaluate_sequence(f.tolist())
        start_results.append((sc, name, f))

    start_results.sort(key=lambda x: x[0])

    print("\nTop starting points:")
    for sc, name, _ in start_results[:10]:
        print(f"  {name}: {sc:.6f}")

    best_score = start_results[0][0]
    best_f = start_results[0][2].copy()

    if best_score < init_score:
        print(f"Best start improves over given: {init_score:.6f} -> {best_score:.6f}")

    # Phase 1: Mirror descent from top starts
    t_elapsed = time.monotonic() - t_start
    t_remaining = budget_s - 10.0 - t_elapsed
    t_mirror = t_remaining * 0.45
    t_projected = t_remaining * 0.2
    t_polish = t_remaining * 0.35

    print(f"\nTime allocation: mirror={t_mirror:.0f}s, projected={t_projected:.0f}s, polish={t_polish:.0f}s")

    # Run mirror descent from top 3 starts
    mirror_starts = start_results[:3]
    for idx, (sc, name, f0) in enumerate(mirror_starts):
        if time.monotonic() > t_start + t_mirror + t_projected + 30:
            break
        t_alloc = t_mirror * [0.5, 0.3, 0.2][idx]
        d = min(time.monotonic() + t_alloc, t_start + t_mirror + t_projected + 30)
        print(f"\n[Mirror {idx}] start='{name}' score={sc:.6f} budget={t_alloc:.0f}s")
        f_opt, sc_opt = optimize_mirror(f0, n, d, rng, label=name)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  New best: {best_score:.6f}")
        else:
            print(f"  Best from this: {sc_opt:.6f} (not better)")

    # Phase 2: Projected gradient from best
    if time.monotonic() < deadline - t_polish:
        d = min(time.monotonic() + t_projected, deadline - t_polish)
        print(f"\n[Projected] from best score={best_score:.6f} budget={t_projected:.0f}s")
        f_opt, sc_opt = optimize_projected(best_f, n, d, rng, label="proj")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  New best: {best_score:.6f}")

    # Phase 3: Polish
    if time.monotonic() < deadline - 10:
        d = deadline
        print(f"\n[Polish] score={best_score:.6f} budget={deadline - time.monotonic():.0f}s")
        f_pol, sc_pol = polish_peak(best_f, n, d, rng)
        if sc_pol < best_score:
            best_score = sc_pol
            best_f = f_pol.copy()
            print(f"  Polished to: {best_score:.6f}")

        # If we have more time, do another mirror+polish round
        if time.monotonic() < deadline - 30:
            t_left = deadline - time.monotonic() - 10
            if t_left > 60:
                t_m = t_left * 0.5
                t_p = t_left * 0.5
                d_m = time.monotonic() + t_m
                print(f"\n[Round 2] Mirror budget={t_m:.0f}s")
                f_opt2, sc_opt2 = optimize_mirror(best_f, n, d_m, rng, label="r2m")
                if sc_opt2 < best_score:
                    best_score = sc_opt2
                    best_f = f_opt2.copy()
                    print(f"  New best: {best_score:.6f}")

                if time.monotonic() < deadline - 10:
                    print(f"\n[Round 2 Polish]")
                    f_pol2, sc_pol2 = polish_peak(best_f, n, deadline, rng)
                    if sc_pol2 < best_score:
                        best_score = sc_pol2
                        best_f = f_pol2.copy()
                        print(f"  Polished to: {best_score:.6f}")

    # Verify and return
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = evaluate_sequence(best_f.tolist())
    if final_score < best_score or not np.isfinite(best_score):
        best_score = final_score

    elapsed = time.monotonic() - t_start
    print(f"\n=== FINAL: score={best_score:.6f}, time={elapsed:.0f}s ===")
    return [float(x) for x in best_f]
