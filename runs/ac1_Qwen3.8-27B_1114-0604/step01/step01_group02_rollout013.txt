LP-based alternating optimization to minimize 2n*max(conv(a,a))/sum(a)^2.
Uses the Sinkhorn-like LP iteration: at each step, solve max sum(g) s.t. a*g<=M, g>=0,
then blend the result back into a. Multiple n values and initializations are tried.
"""

import numpy as np
from scipy import sparse
from scipy.optimize import linprog
import time

def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def score_from_conv(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def build_conv_matrix(a):
    """Build the sparse convolution matrix A where (A @ g) = a * g (linear conv)."""
    n = len(a)
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        l_start = max(0, k - n + 1)
        l_end = min(k, n - 1)
        for l in range(l_start, l_end + 1):
            rows.append(k)
            cols.append(l)
            vals.append(a[k - l])
    A = sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A

def solve_lp(a, M):
    """Solve: max sum(g) s.t. A@g <= M, g >= 0."""
    n = len(a)
    A = build_conv_matrix(a)
    c = -np.ones(n)
    b = M * np.ones(2*n-1)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.status != 0:
        return None
    return res.x

def lp_iterate(a, max_iters, time_limit, t_start=0.5, t_end=0.1, M_shrink=0.999, rng=None):
    """Run LP-based iterations. Returns (best_score, best_a)."""

    a = a.astype(np.float64).copy()
    best_score = np.inf
    best_a = a.copy()
    S = a.sum()
    
    for it in range(max_iters):
        c = fast_self_conv(a)
        M = float(np.max(c))
        score = score_from_conv(n, c, S)
        if score < best_score:
            best_score = score
            best_a = a.copy()
        
        # Solve LP
        # Use a slightly shrunk M to push harder
        M_lp = M * M_shrink
        g = solve_lp(a, M_lp)
        if g is None:
            break
        g_sum = float(g.sum())
        if g_sum < 1e-10:
            break
        g = g * (S / g_sum)  # rescale to match sum
        
        # Blend
        t = t_start * (t_end / t_start) ** (it / max(1, max_iters - 1))
        a_new = (1 - t) * a + t * g
        a_new = np.clip(a_new, 0, 1000)
        if a_new.sum() < 1e-3:
            a_new = a
        a = a_new
    
    # Final check
    c = fast_self_conv(a)
    score = score_from_conv(n, c, a.sum())
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_score, best_a

"""
LP-alternating optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Primary method: Sinkhorn-like LP iteration (solve max sum(g) s.t. a*g<=M, g>=0).
Secondary: gradient-based polishing with smooth max approximation.

import numpy as np
from scipy import sparse
from scipy.optimize import linprog
import time

# --- Convolution helpers ---

def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def score_from_conv(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def build_conv_structure(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        c_start = max(0, k - n + 1)
        c_end = min(k, n - 1)
        rows.extend([k] * (c_end - c_start + 1))
        cols.extend(range(c_start, c_end + 1))
    return np.array(rows), np.array(cols)

def make_conv_matrix(a, structure):
    rows, cols = structure
    n = len(a)
    vals = a[rows - cols]
    return sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp(a, M, structure):
    n = len(a)
    A = make_conv_matrix(a, structure)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.status != 0:
        return None
    return res.x

def lp_optimize(a0, max_iters, deadline, structure, rng, M_shrink=0.999, 
                t_init=0.4, t_final=0.05, patience=50):
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = a.sum()
    best_score = np.inf
    best_a = a.copy()
    no_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = fast_self_conv(a)
        M = float(np.max(c))
        score = score_from_conv(n, c, S)
        
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > patience:
            break
        
        # Solve LP
        M_lp = M * M_shrink
        g = solve_lp(a, M_lp, structure)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        
        g = g * (S / g_sum)
        
        # Blend with decreasing step
        frac = it / max(it + 1, max_iters - 1)
        t = t_init * (t_final / t_init) ** frac
        a = (1 - t) * a + t * g
        a = np.clip(a, 0.0, 1000.0)
    
    # Final evaluation
    c = fast_self_conv(a)
    score = score_from_conv(n, c, a.sum())
    if score < best_score:
        best_score = score
        best_a = a.copy()
    
    return best_score, best_a

# --- Gradient-based polishing (from previous approach, improved) ---

def topk_softmax(c, k, temp_frac=0.1):
    k = min(k, len(c))
    idx = np.argpartition(c, -k)[-k:]
    vals = c[idx]
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]
    top = vals[0]
    if top > 1e-12:
        temp = temp_frac * top
        wts = np.exp((vals - top) / temp)
        wts /= wts.sum()
    else:
        wts = np.ones(k) / k
    return idx, vals, wts

def grad_smooth_max(a, k, temp_frac=0.1):
    n = len(a)
    c = fast_self_conv(a)
    S = a.sum()
    idx, vals, wts = topk_softmax(c, k, temp_frac)
    M = float(np.dot(vals, wts))
    grad_c = np.zeros(n)
    for i in range(idx.size):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-15:
            continue
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        grad_c[j_start:j_end+1] += wi * 2.0 * a[j_start:j_end+1][::-1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_c / S2 - 2.0 * M / S3)
    return grad, M, S

def gradient_polish(a0, deadline, rng, init_lr=0.1, k=8, patience=200):
    n = len(a0)
    a = a0.astype(np.float64).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = init_lr
    best_score = np.inf
    best_a = a.copy()
    no_improve = 0
    
    while time.monotonic() < deadline:
        grad, M, S = grad_smooth_max(a, k)
        t += 1
        # Adam
        b1, b2, eps = 0.9, 0.999, 1e-8
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        a = a - lr * mh / (np.sqrt(vh) + eps)
        a = np.clip(a, 0.0, 1000.0)
        
        if a.sum() < 1e-3:
            a = best_a.copy()
            continue
        
        c = fast_self_conv(a)
        score = score_from_conv(n, c, a.sum())
        
        if score < best_score - 1e-10:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= patience:
            # Restart
            a = best_a * (1 + 0.05 * rng.standard_normal(n))
            a = np.clip(a, 0, 1000)
            m[:] = 0
            v[:] = 0
            t = 0
            lr *= 0.5
            no_improve = 0
            patience = max(50, patience // 2)
            if lr < 1e-4:

"""
LP-alternating optimizer for minimizing 2n*max(conv(a,a))/sum(a)^2.
Primary method: Sinkhorn-like LP iteration solving max sum(g) s.t. a*g<=M*(1-eps), g>=0.
Secondary: gradient-based polishing with smooth max approximation.
Multiple sequence lengths (1024, 2048, 4096) and diverse initializations are explored.
"""

import numpy as np
from scipy import sparse
from scipy.optimize import linprog
import time

def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def score_from_conv(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def build_conv_structure(n):
    rows = []
    cols = []
    for k in range(2 * n - 1):
        c_start = max(0, k - n + 1)
        c_end = min(k, n - 1)
        rows.extend([k] * (c_end - c_start + 1))
        cols.extend(range(c_start, c_end + 1))
    return np.array(rows), np.array(cols)

def make_conv_matrix(a, structure):
    rows, cols = structure
    n = len(a)
    vals = a[rows - cols]
    return sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))

def solve_lp(a, M, structure):
    n = len(a)
    A = make_conv_matrix(a, structure)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.status != 0:
        return None
    return res.x

def lp_optimize(a0, deadline, structure, rng, M_shrink=0.999,
                t_init=0.4, t_final=0.03, patience=80):
    n = len(a0)
    a = a0.astype(np.float64).copy()
    S = float(a.sum())
    best_score = np.inf
    best_a = a.copy()
    no_improve = 0
    it = 0
    while time.monotonic() < deadline:
        c = fast_self_conv(a)
        M = float(np.max(c))
        score = score_from_conv(n, c, S)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > patience:
            M_shrink = max(0.99, M_shrink * 0.995)
            no_improve = 0
            if M_shrink < 0.99:
                break
        M_lp = M * M_shrink
        g = solve_lp(a, M_lp, structure)
        if g is None:
            break
        g_sum = float(g.sum())
        if g_sum < 1e-10:
            break
        g = g * (S / g_sum)
        frac = min(1.0, it / 200.0)
        t = t_init * (t_final / t_init) ** frac
        a = (1.0 - t) * a + t * g
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 1e-3:
            a = best_a.copy()
        it += 1
    c = fast_self_conv(a)
    score = score_from_conv(n, c, float(a.sum()))
    if score < best_score:
        best_score = score
        best_a = a.copy()
    return best_score, best_a

def grad_smooth_max(a, k, temp_frac=0.08):
    n = len(a)
    c = fast_self_conv(a)
    S = float(a.sum())
    k = min(k, len(c))
    idx = np.argpartition(c, -k)[-k:]
    vals = c[idx]
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]
    top = float(vals[0])
    if top > 1e-12:
        temp = temp_frac * top
        wts = np.exp((vals - top) / temp)
        wts /= wts.sum()
    else:
        wts = np.ones(k) / k
    M = float(np.dot(vals, wts))
    grad_c = np.zeros(n)
    for i in range(idx.size):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-15:
            continue
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        grad_c[j_start:j_end + 1] += wi * 2.0 * a[j_start:j_end + 1][::-1]
    S2 = S * S
    S3 = S2 * S
    grad = 2.0 * n * (grad_c / S2 - 2.0 * M / S3)
    return grad, M, S

def gradient_polish(a0, deadline, rng, init_lr=0.08, k=10, patience=300):
    n = len(a0)
    a = a0.astype(np.float64).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = init_lr
    best_score = np.inf
    best_a = a.copy()
    no_improve = 0
    while time.monotonic() < deadline:
        grad, M, S = grad_smooth_max(a, k)
        t += 1
        b1, b2, eps = 0.9, 0.999, 1e-8
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        a = a - lr * mh / (np.sqrt(vh) + eps)
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 1e-3:
            a = best_a.copy()
            continue
        c = fast_self_conv(a)

            best_a = a.copy()
            no_improve = 0

            m[:] = 0.0
            v[:] = 0.0
            t = 0
            lr *= 0.6
            no_improve = 0
            patience = max(60, patience // 2)
            if lr < 1e-5:
                break
    return best_score, best_a
