def _subgradient_refine(a, budget_s):
    """Subgradient descent on the simplex with adaptive step size."""
    start = time.monotonic()
    arr = np.asarray(a, dtype=np.float64).copy()
    n = arr.size
    s = float(np.sum(arr))
    if s < 0.01:
        return arr, np.inf
    arr /= s
    
    best = arr.copy()
    best_score = _score(best)
    
    # Parameters
    K = min(16, 2 * n - 1)  # number of active lags
    lr = float(n) * 0.02    # initial learning rate
    min_lr = float(n) * 0.001
    max_lr = float(n) * 0.5
    last_improve = 0
    
    it = 0
    while time.monotonic() - start < budget_s:
        it += 1
        
        # Compute convolution
        c = _fft_convolve(arr)
        if c.size == 0:
            break
        
        max_c = float(np.max(c))
        if not np.isfinite(max_c):
            break
        
        # Find active lags (top K by value)
        if c.size <= K:
            active = np.arange(c.size)
        else:
            active = np.argpartition(c, -K)[-K:]
        
        # Compute subgradient: for each active lag k, grad[j] += 2*(a[k-j] + a[j-k]) / 2
        # Actually: d(b_k)/d(a_j) = a[k-j] + a[j-k] (where out-of-range = 0)
        grad = np.zeros(n, dtype=np.float64)
        for k in active:
            k = int(k)
            # For lag k: b_k = sum_i a[i]*a[k-i]
            # d(b_k)/d(a_j) = a[k-j] (if 0<=k-j<n) + a[j-k] (if 0<=j-k<n)
            # i.e., a[k-j] when k-j in [0,n-1] plus a[j-k] when j-k in [0,n-1]
            idx1 = k - np.arange(n)  # k-j for j=0..n-1
            mask1 = (idx1 >= 0) & (idx1 < n)
            if np.any(mask1):
                grad[mask1] += arr[idx1[mask1]]
            idx2 = np.arange(n) - k  # j-k for j=0..n-1
            mask2 = (idx2 >= 0) & (idx2 < n)
            if np.any(mask2):
                grad[mask2] += arr[idx2[mask2]]
        
        # Average over active lags (subgradient of max)
        grad /= float(len(active))
        
        # Project to tangent space of simplex (remove mean)
        grad = grad - np.mean(grad)
        
        # Check if gradient is zero
        if np.max(np.abs(grad)) < 1e-12:
            break
        
        # Mirror descent step
        exponent = -lr * grad
        exponent = np.clip(exponent, -2.0, 2.0)
        arr_new = arr * np.exp(exponent)
        arr_new = np.clip(arr_new, 0.0, None)
        s_new = float(np.sum(arr_new))
        if s_new < 0.01 or not np.isfinite(s_new):
            lr = max(min_lr, lr * 0.5)
            continue
        arr_new /= s_new
        
        sc = _score(arr_new)
        
        if sc < best_score:
            best = arr_new.copy()
            best_score = sc
            arr = arr_new
            last_improve = it
            lr = min(max_lr, lr * 1.05)
        else:
            # Step was bad, reduce learning rate
            if sc > best_score * 1.01:
                lr = max(min_lr, lr * 0.8)
            arr = arr_new  # still move to new point (stochastic-like)
            # Actually, let's be more conservative:
            # Don't update arr if the step was too bad
            if sc > best_score * 1.05:
                arr = best.copy()  # revert
        
        if it - last_improve > 200:
            break
    
    return best, best_score

    a = np.ones(n, dtype=np.float64)
    for m_idx, amp in enumerate(amps, 1):
        if abs(amp) < 1e-12:
            continue
        theta = 2.0 * np.pi * m_idx * (i + 0.5) / float(n)
        a += amp * np.sin(theta)
    a /= float(n)
    a = np.clip(a, 0.0, None)
    s = float(np.sum(a))
    if s < 0.01:
        return np.ones(n, dtype=np.float64) / float(n)
    a /= s
    return a


def _optimize_single(n, deadline):
    """Optimize single-harmonic amplitude via adaptive grid search."""
    best_score = np.inf
    best_amp = 0.5
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    sin_vals = np.sin(theta)

    # Coarse scan
    for amp in np.linspace(0.0, 1.0, 41):
        if time.monotonic() > deadline:
            break
        a = (1.0 + amp * sin_vals) / float(n)
        a = np.clip(a, 0.0, None)
        s = float(np.sum(a))
        if s < 0.01:
            continue
        a /= s
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_amp = float(amp)

    # Fine scan: 5 levels of refinement
    step = 0.025
    for _ in range(6):
        lo = max(0.0, best_amp - step)
        hi = min(1.0, best_amp + step)
        if hi - lo < 1e-8:
            break
        for amp in np.linspace(lo, hi, 21):
            if time.monotonic() > deadline:
                break
            a = (1.0 + amp * sin_vals) / float(n)
            a = np.clip(a, 0.0, None)
            s = float(np.sum(a))
            if s < 0.01:
                continue
            a /= s
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_amp = float(amp)
        step *= 0.5

    return best_amp, best_score


def _optimize_two_harm(n, deadline):
    """Optimize 2-harmonic amplitudes via grid search."""
    best_score = np.inf
    best_amps = np.array([0.7, 0.2])
    i = np.arange(n, dtype=np.float64)
    sin1 = np.sin(2.0 * np.pi * (i + 0.5) / float(n))
    sin2 = np.sin(2.0 * np.pi * 2.0 * (i + 0.5) / float(n))

    # Coarse grid
    for a1 in np.linspace(0.3, 0.95, 14):
        if time.monotonic() > deadline:
            break
        for a2 in np.linspace(-0.3, 0.35, 14):
            if time.monotonic() > deadline:
                break
            a = (1.0 + a1 * sin1 + a2 * sin2) / float(n)
            a = np.clip(a, 0.0, None)
            s = float(np.sum(a))
            if s < 0.01:
                continue
            a /= s
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_amps = np.array([a1, a2])

    # Fine grid around best
    for _ in range(3):
        lo1 = max(0.0, best_amps[0] - 0.05)
        hi1 = min(1.0, best_amps[0] + 0.05)
        lo2 = max(-0.5, best_amps[1] - 0.05)
        hi2 = min(0.5, best_amps[1] + 0.05)
        for a1 in np.linspace(lo1, hi1, 11):
            if time.monotonic() > deadline:
                break
            for a2 in np.linspace(lo2, hi2, 11):
                a = (1.0 + a1 * sin1 + a2 * sin2) / float(n)
                a = np.clip(a, 0.0, None)
                s = float(np.sum(a))
                if s < 0.01:
                    continue
                a /= s
                sc = _score(a)
                if sc < best_score:
                    best_score = sc
                    best_amps = np.array([a1, a2])

    return best_amps, best_score

def _subgradient_refine(a, budget_s, verbose=True):
    """
    Mirror-descent subgradient refinement on the simplex.
    Minimizes max_k (a*a)[k] subject to a >= 0, sum(a) = 1.
    Uses top-K active lags for subgradient estimation.
    """
    start = time.monotonic()
    arr = np.asarray(a, dtype=np.float64).copy()
    n = arr.size
    s = float(np.sum(arr))
    if s < 0.01:
        return arr, _score(arr)
    arr /= s

    best = arr.copy()
    best_score = _score(best)
    if verbose:
        print(f"    [refine] start: n={n} score={best_score:.8f}")

    # Adaptive parameters
    K = min(32, 2 * n - 1)
    lr = float(n) * 0.01
    min_lr = float(n) * 0.0005
    max_lr = float(n) * 0.2
    stall_count = 0
    last_improve = 0
    it = 0

    # Pre-compute index arrays for gradient computation
    idx_all = np.arange(n, dtype=np.int64)

    while time.monotonic() - start < budget_s:
        it += 1

        # Compute convolution
        c = _fft_convolve(arr)
        if c.size == 0:
            break
        max_c = float(np.max(c))
        if not np.isfinite(max_c):
            break

        # Select active lags: top K by value, with a small threshold
        if c.size <= K:
            active = np.arange(c.size, dtype=np.int64)
        else:
            active = np.argpartition(c, -K)[-K:]

        # For better subgradient, weight active lags by their values
        active_vals = c[active]
        # Use only lags within 20% of max for a robust subgradient
        threshold = max_c * 0.80
        active_mask = active_vals >= threshold
        if np.sum(active_mask) < 2:
            # Fallback: use top 5
            top_k = min(5, c.size)
            active = np.argpartition(c, -top_k)[-top_k:]
        else:
            active = active[active_mask]

        # Compute subgradient (average gradient of active lag b_k's)
        grad = np.zeros(n, dtype=np.float64)
        for k in active:
            k = int(k)
            # d(b_k)/d(a_j) = a[k-j] (if valid) + a[j-k] (if valid)
            # Vectorized:
            cand1 = k - idx_all  # k - j for each j
            m1 = (cand1 >= 0) & (cand1 < n)
            if np.any(m1):
                grad[m1] += arr[cand1[m1]]
            cand2 = idx_all - k  # j - k for each j
            m2 = (cand2 >= 0) & (cand2 < n)
            if np.any(m2):
                grad[m2] += arr[cand2[m2]]

        grad /= float(len(active))

        # Project to tangent space of simplex
        grad = grad - np.mean(grad)

        grad_norm = float(np.max(np.abs(grad)))
        if grad_norm < 1e-14:
            break

        # Mirror descent step with line search
        improved = False
        for lr_try in [lr, lr * 0.6, lr * 0.36, lr * 0.216]:
            exponent = -lr_try * grad
            exponent = np.clip(exponent, -3.0, 3.0)
            arr_new = arr * np.exp(exponent)
            arr_new = np.clip(arr_new, 0.0, None)
            s_new = float(np.sum(arr_new))
            if s_new < 0.01 or not np.isfinite(s_new):
                continue
            arr_new /= s_new
            sc = _score(arr_new)
            if sc < best_score - 1e-10:
                best = arr_new.copy()
                best_score = sc
                arr = arr_new
                last_improve = it
                improved = True
                lr = min(max_lr, lr_try * 1.1)
                stall_count = 0
                break
            elif sc < best_score:
                # Slight improvement, accept but don't increase lr
                arr = arr_new
                break
        else:
            # No improvement with any step size
            stall_count += 1
            lr = max(min_lr, lr * 0.75)

        # Check for stall
        if it - last_improve > 300:
            if verbose:
                print(f"    [refine] stalled at it={it} score={best_score:.8f}")
            break

        # Periodic reporting
        if verbose and it % 500 == 0:
            elapsed = time.monotonic() - start
            print(f"    [refine] it={it} score={best_score:.8f} lr={lr:.2f} elapsed={elapsed:.1f}s")

    if verbose:
        elapsed = time.monotonic() - start
        print(f"    [refine] done: it={it} best_score={best_score:.8f} elapsed={elapsed:.1f}s")

    return best, best_score


def _refine_with_restarts(a, budget_s, n_restarts=3):
    """Try refinement with different initial perturbations."""
    best = np.asarray(a, dtype=np.float64).copy()
    best_score = _score(best)

    total_budget = float(budget_s)
    per_restart = total_budget / float(n_restarts)

    for r in range(n_restarts):
        remaining = total_budget - r * per_restart
        if remaining < 10.0:
            break

        if r == 0:
            start_arr = best.copy()
        else:
            # Perturb: add small random noise and renormalize
            rng = np.random.default_rng(42 + r)
            noise = rng.normal(0, 0.01 * best / max(1.0, best.size / 100.0), best.size)
            start_arr = best + noise
            start_arr = np.clip(start_arr, 0.0, None)
            s = float(np.sum(start_arr))
            if s < 0.01:
                start_arr = best.copy()
            else:
                start_arr /= s

        refined, sc = _subgradient_refine(start_arr, min(per_restart, remaining), verbose=(r == 0))
        if sc < best_score:
            best_score = sc
            best = refined
            print(f"  [restart {r}] improved to {best_score:.8f}")

    return best, best_score


def _try_hann_shape(n, deadline):
    """Try a Hann-window-based shape as alternative initialization."""
    best_score = np.inf
    best_arr = None
    i = np.arange(n, dtype=np.float64)

    # Han-like: a[i] proportional to (1 + cos(2*pi*i/n))^alpha for various alpha
    for alpha in [0.5, 1.0, 1.5, 2.0, 2.5, 3.0]:
        if time.monotonic() > deadline:
            break
        a = (1.0 + np.cos(2.0 * np.pi * i / float(n))) ** alpha
        a = np.clip(a, 0.0, None)
        s = float(np.sum(a))
        if s < 0.01:
            continue
        a /= s
        sc = _score(a)
        if sc < best_score:
            best_score = sc
            best_arr = a.copy()

    return best_arr, best_score
