Wait, one more important consideration: the simplex projection. For a vector v, the projection onto {x >= 0, sum(x) = 1} is:
- Find tau such that sum(max(v[i] - tau, 0)) = 1
- x[i] = max(v[i] - tau, 0)

The standard O(n log n) algorithm:
1. Sort v in decreasing order: v_(1) >= v_(2) >= ... >= v_(n)
2. Find the largest rho such that v_(rho) - (sum_{i=1}^{rho} v_(i) - 1)/rho > 0
3. tau = (sum_{i=1}^{rho} v_(i) - 1)/rho
4. x[i] = max(v[i] - tau, 0)

This is O(n log n) due to sorting. For n = 8192, this is fast.

Actually, there's a simpler O(n) algorithm using "Dahn's algorithm" (a selection algorithm), but O(n log n) is fine.

def fast_conv(a):
    """Compute full self-convolution using FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def hankel_matvec(p, a):
    """Compute Hp * a where H[j,l] = p[j+l]."""
    n = len(a)
    q = p[::-1]  # length 2n-1
    # Convolution of q (length 2n-1) and a (length n) has length 3n-2
    L = len(q) + n - 1  # = 3n-2
    M_fft = 1 << (L - 1).bit_length()
    Q = np.fft.rfft(q, M_fft)
    A = np.fft.rfft(a, M_fft)
    full_conv = np.fft.irfft(Q * A, M_fft)[:L]
    # (Hp*a)[j] = full_conv[2n-2-j] for j=0..n-1
    return full_conv[n-1:2n-1][::-1]

def simplex_project(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = len(v)
    # Scale v so that the target sum is 1 for the algorithm
    # Actually, let's handle it directly
    # We need sum(max(v[i] - tau, 0)) = target_sum
    # Sort v in decreasing order
    u = np.sort(v)[::-1]

    cssv = np.cumsum(u)

    rho = np.max(np.where(cond > 0)[0]) + 1

        # Check score

        if s < best_score:
            best_score = s
            best_a = a.copy()
    
    return best_a, best_score

- sum(a) = 1

This is equivalent to: `2 * (Hp*a) - lambda * 1 >= 0` (component-wise), with equality where a > 0.

"""
IRLS (Iteratively Reweighted Least Squares) optimizer for minimizing
2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences a.

The key idea: alternate between (1) solving a weighted quadratic
minimize sum_k w[k]*conv(a,a)[k] s.t. a>=0, sum(a)=1 (a QP with Hankel
structure, solved via FISTA with FFT-based matvec), and (2) updating
weights w[k] proportional to current autocorrelation peaks. This directly
targets the max peak rather than using a smooth approximation.

Multiple sequence lengths are explored, and the best result is polished
with aggressive peak-focusing variants.
"""

import numpy as np
import time


# ============================================================================
# Core computational primitives
# ============================================================================

def fast_self_conv(a):
    """Full self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def hankel_matvec(p, a):
    """
    Compute H_p * a where H_p[j,l] = p[j+l] for j,l in 0..n-1.
    Uses FFT: (H_p*a)[j] = conv(reversed_p, a)[2n-2-j].
    """
    n = len(a)
    q = p[::-1]  # length 2n-1
    L = len(q) + n - 1  # = 3n-2
    M_fft = 1 << (L - 1).bit_length()
    Q = np.fft.rfft(q, M_fft)
    A = np.fft.rfft(a, M_fft)
    full = np.fft.irfft(Q * A, M_fft)[:L]
    return full[n - 1:2 * n - 1][::-1]


def simplex_project(v, target=1.0):
    """
    Project v onto the simplex {x >= 0, sum(x) = target}.
    O(n log n) via sorting.
    """
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_arr = np.arange(1, n + 1, dtype=np.float64)
    cond = u - (cssv - target) / rho_arr
    valid = np.where(cond > 0)[0]
    if len(valid) == 0:
        return np.full(n, target / n)
    rho = valid[-1] + 1
    theta = (cssv[rho - 1] - target) / rho
    return np.maximum(v - theta, 0.0)


def score_from_conv(n, c, S):
    """Objective = 2*n*max_c / S^2."""
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(c.max()) / (S * S)


def compute_score(a):
    """Full score computation."""
    n = len(a)
    c = fast_self_conv(a)
    return score_from_conv(n, c, a.sum())


# ============================================================================
# FISTA solver for the inner QP
# ============================================================================

def solve_quadratic_hankel(p, a_init, n, max_iter=300, verbose=False):
    """
    Solve: min a^T (H_p) a  s.t.  a >= 0, sum(a) = 1
    using FISTA (accelerated projected gradient).
    
    H_p is the Hankel matrix with entries p[j+l].
    Gradient = 2 * H_p * a, computed via FFT.
    """
    a = np.clip(a_init, 0.0, None).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    # Upper bound on Lipschitz constant: L = 2 * lambda_max(H_p)
    # lambda_max(H_p) <= ||H_p||_inf <= max_j sum_l p[j+l] <= max_k (n * p[k])
    # So L <= 2 * n * max(p)
    L = 2.0 * float(n) * float(np.max(p)) * 1.5  # extra safety margin
    lr = 1.0 / L

    # FISTA
    t = 1.0
    a_prev = a.copy()

    for it in range(max_iter):
        grad = 2.0 * hankel_matvec(p, a)
        a_new = a - lr * grad
        a_new = simplex_project(a_new, 1.0)

        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        a_y = a_new + ((t - 1.0) / t_new) * (a_new - a_prev)
        a_y = simplex_project(a_y, 1.0)

        a_prev = a_new
        a = a_y
        t = t_new

        if verbose and it % 50 == 0:
            obj = float(a @ hankel_matvec(p, a))
            print(f"    FISTA iter {it}: obj={obj:.10f}")

    return a


# ============================================================================
# IRLS outer loop
# ============================================================================

def irls_optimize(a_init, n, time_budget, alpha=1.0, inner_iters=200,
                  restart_every=30, verbose=True):
    """
    IRLS optimization loop.
    
    Parameters:
        a_init: initial sequence (will be normalized to sum=1)
        n: sequence length
        time_budget: seconds to run
        alpha: weight exponent (1 = exact min-max, >1 = peak-focused)
        inner_iters: FISTA iterations per outer step
        restart_every: restart after this many non-improving outer steps
        verbose: print progress
    """
    start = time.monotonic()
    deadline = start + time_budget

    a = np.clip(a_init, 0.0, None).astype(np.float64)
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s

    best_a = a.copy()
    best_score = compute_score(a)

    # Initial weights: uniform
    p = np.ones(2 * n - 1) / (2 * n - 1)

    no_improve = 0
    outer = 0
    rng = np.random.default_rng(12345 + outer)

    while time.monotonic() < deadline:
        # Solve inner QP
        a = solve_quadratic_hankel(p, best_a, n, max_iter=inner_iters)

        # Evaluate
        c = fast_self_conv(a)
        sc = 2.0 * n * float(c.max())  # sum=1, so score = 2n*max_c

        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update weights
        c_pos = np.maximum(c, 1e-15)
        w = c_pos ** alpha
        p = w / w.sum()

        # Restart logic
        if no_improve >= restart_every:
            # Perturb the best solution
            scale = 0.05 + 0.1 * rng.random()
            pert = best_a * (1.0 + scale * rng.standard_normal(n))
            pert = np.clip(pert, 0.0, None)
            ps = pert.sum()
            if ps > 1e-12:
                pert = pert / ps
            else:
                pert = np.ones(n) / n
            best_a = pert
            best_score = compute_score(pert)
            no_improve = 0
            # Reset weights to something less peaked
            c = fast_self_conv(best_a)
            c_pos = np.maximum(c, 1e-12)
            p = c_pos / c_pos.sum()

        outer += 1
        if verbose and outer % 20 == 0:
            elapsed = time.monotonic() - start
            print(f"  [{n}] outer={outer}, score={best_score:.6f}, "
                  f"time={elapsed:.0f}s/{time_budget:.0f}s")

    return best_a, best_score


# ============================================================================
# Gradient-based polish (soft-max approach, complementary to IRLS)
# ============================================================================

def gradient_polish(a_init, n, time_budget, topk=8, lr_init=0.05, 
                    verbose=True):
    """
    Gradient descent with top-k soft-max approximation of the max.
    Used as a polish step after IRLS.
    """
    start = time.monotonic()
    deadline = start + time_budget

    a = np.clip(a_init, 0.0, None).astype(np.float64)
    a /= a.sum()

    best_a = a.copy()
    best_score = compute_score(a)

    rng = np.random.default_rng(999)
    lr = lr_init
    no_improve = 0
    it = 0

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8

    while time.monotonic() < deadline:
        c = fast_self_conv(a)
        S = a.sum()
        sc = score_from_conv(n, c, S)

        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            # Perturb and restart
            scale = 0.05 * (1 + rng.random())
            a = best_a * (1.0 + scale * rng.standard_normal(n))
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            m[:] = 0.0
            v[:] = 0.0
            lr = max(0.005, lr * 0.7)
            no_improve = 0
            it = 0
            continue

        # Compute gradient using top-k soft-max
        k_act = min(topk, len(c))
        if k_act == 1:
            idx = np.array([np.argmax(c)])
            wts = np.array([1.0])
        else:
            idx = np.argpartition(c, -k_act)[-k_act:]
            order = np.argsort(c[idx])[::-1]
            idx = idx[order]
            top_val = c[idx[0]]
            temp = 0.1 * top_val
            wts = np.exp((c[idx] - top_val) / temp)
            wts /= wts.sum()

        # Gradient of weighted sum of peaks
        grad_c = np.zeros(n)
        for i in range(idx.size):
            ki = int(idx[i])
            wi = float(wts[i])
            if wi < 1e-15:
                continue
            j_lo = max(0, ki - n + 1)
            j_hi = min(ki, n - 1)
            if j_lo <= j_hi:
                # d conv(ki)/d a[j] = 2 * a[ki-j]
                grad_c[j_lo:j_hi+1] += wi * 2.0 * a[j_lo:j_hi+1][::-1]

        M_w = float(np.dot(c[idx], wts))
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (grad_c / S2 - 2.0 * M_w / S3)

        # Adam step
        it += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        mc = m / (1 - b1**it)
        vc = v / (1 - b2**it)
        a = a - lr * mc / (np.sqrt(vc) + eps)
        a = np.clip(a, 0.0, None)
        if a.sum() < 1e-12:
            a = best_a.copy()
        else:
            a = a / a.sum()

        if verbose and it % 200 == 0:
            print(f"    grad iter {it}, score={best_score:.6f}, "
                  f"lr={lr:.4f}")

    return best_a, best_score

def make_initializations(n, rng, parent=None):
    """Generate diverse initializations for sequence of length n."""
    inits = {}

    # Parent (if available and correct length)
    if parent is not None:
        p = np.clip(np.asarray(parent, dtype=np.float64), 0, None)
        if len(p) == n and p.sum() > 1e-12:
            inits['parent'] = p / p.sum()

    # Uniform
    inits['uniform'] = np.ones(n) / n

    # Random
    r = rng.random(n)
    inits['random'] = r / r.sum()

    # Cosine-modulated
    x = np.linspace(0, 1, n)
    for freq, amp in [(1, 0.3), (2, 0.4), (4, 0.5), (8, 0.6)]:
        c = 1.0 + amp * np.cos(2 * np.pi * freq * x)
        c = np.maximum(c, 0.01)
        inits[f'cos_{freq}'] = c / c.sum()

    # Ramp
    r = 1.0 + 0.8 * (2 * x - 1)
    r = np.maximum(r, 0.01)
    inits['ramp'] = r / r.sum()

    # Blocks
    for nb in [4, 8, 16]:
        per = n // nb
        bv = 0.3 + 0.7 * rng.random(nb)
        b = np.repeat(bv, per)
        if len(b) < n:
            b = np.concatenate([b, bv[:n - len(b)]])
        b = b[:n]
        inits[f'blocks_{nb}'] = b / b.sum()

    return inits
