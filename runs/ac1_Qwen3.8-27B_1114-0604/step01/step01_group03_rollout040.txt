def _gn_direction(a, c, L, m, n, temp, reg):
    C = c.max()
    # weights
    if temp <= 0: uniform
    else:
      z = c - C
      # To avoid under/overflow, clip z/temp.
      w = np.exp(np.clip(z/temp, -50, 0))
      w /= sum
    # weighted mean target t

"""
Search for non-negative step heights minimizing
    2*n*max(conv(a, a)) / (sum(a)**2).

The implementation uses a large, wrap-safe FFT length for accurate linear
convolutions.  It combines:
  - multi-start initialization from resampled known bases and smooth random
    Fourier-like priors,
  - Gauss-Newton / iterative-reshaping directions for weighted active-set
    flattening of the self-convolution,
  - fallback gradient directions,
  - a multiplicative-weights minimax-like candidate generator inspired by the
    previous state of the art,
  - light local coordinate-insensitive pair polishing,
  - final validation through evaluate_sequence when available.
"""

import math
import time
import numpy as np


# Cycles of weighted active-set directions for the Gauss-Newton-like search.
GN_MODES = [
    ('gn', 0.10),
    ('gn', 0.05),
    ('grad', 0.03),
    ('gn', 0.03),
    ('gn_soft', 0.20, 0.30),
    ('gn', 0.02),
    ('gn', 0.01),
    ('grad', 0.02),
    ('gn_soft', 0.10, 0.20),
    ('gn', 0.012),
    ('gn', 0.006),
    ('gn_soft', 0.05, 0.15),
    ('grad', 0.01),
    ('gn', 0.004),
    ('gn_soft', 0.03, 0.10),
    ('gn', 0.003),
    ('gn', 0.002),
    ('uniform',),

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not math.isfinite(s) or s <= 0.0:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    a = a / s
    return a


def _search_fft_len(n):
    # Enough for both a*a (length 2n-1) and transposed convolution slices
    # (requires avoiding wrap in v * reverse(a), length 3n-2).
    need = max(1, 3 * int(n) - 2)
    return 1 << (need - 1).bit_length()


def _fft_square(a, L, m):
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    c = np.ascontiguousarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c


def _score_c(c, n):
    if c.size == 0:
        return float("inf")
    mc = float(np.max(c))
    if not math.isfinite(mc) or mc < 0.0:
        mc = 0.0
    return float(2.0 * float(n) * mc)


def _score_a(a, L, m, n):
    c = _fft_square(a, L, m)
    return _score_c(c, n)


def _score_array(arr):
    a = np.asarray(arr, dtype=np.float64).ravel()
    n = int(a.size)
    if n <= 0:
        return float("inf")
    a = _normalize(a)
    L = _search_fft_len(n)
    m = 2 * n - 1
    return _score_a(a, L, m, n)


def _internal_exact(seq):
    if not isinstance(seq, (list, tuple)) or not seq:
        return float("inf")
    a = []
    for x in seq:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return float("inf")
        fx = float(x)
        if math.isnan(fx) or math.isinf(fx):
            return float("inf")
        a.append(max(0.0, min(1000.0, fx)))
    n = len(a)
    s = float(sum(a))
    if s < 0.01:
        return float("inf")
    b = np.convolve(a, a)
    mb = float(np.max(b))
    return float(2.0 * n * mb / (s * s))


def _exact_score(seq):
    try:
        val = evaluate_sequence(seq)
    except NameError:
        val = _internal_exact(seq)
    except Exception:
        val = float("inf")
    try:
        val = float(val)
    except Exception:
        val = float("inf")
    if not math.isfinite(val):
        return float("inf")
    return val


def _get_base():
    try:
        base = height_sequence_1
    except NameError:
        return None
    except Exception:
        return None
    try:
        arr = np.asarray(base, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    s = float(arr.sum())
    if not math.isfinite(s) or s <= 0.0:
        return None
    return arr / s


def _resample(arr, n):
    arr = np.asarray(arr, dtype=np.float64).ravel()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    n = int(n)
    if n <= 1:
        if arr.size == 0:
            return np.ones(n)
        return _normalize(arr[:n] if arr.size >= n else np.resize(arr, n))
    if arr.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    if arr.size == 1:
        return _normalize(np.full(n, float(arr[0]), dtype=np.float64))
    x_old = (np.arange(arr.size, dtype=np.float64) + 0.5) / float(arr.size)
    x_new = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    y = np.interp(x_new, x_old, arr)
    return _normalize(y)


def _random_prior(n, rng):
    n = int(n)
    if n <= 1:
        return _normalize(np.ones(n))
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    edge = np.sin(np.pi * t)
    edge = np.clip(edge, 1e-12, 1.0)

    z = np.zeros(n, dtype=np.float64)
    kmax = int(rng.integers(3, 11))
    amp_scale = float(rng.uniform(0.30, 0.95))
    for k in range(1, kmax + 1):
        amp = amp_scale / (math.sqrt(float(k)) + 0.5)
        amp *= float(rng.uniform(0.20, 1.0))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        z += amp * np.cos(2.0 * np.pi * float(k) * t + ph)
        ph2 = float(rng.uniform(0.0, 2.0 * np.pi))
        z += 0.7 * amp * np.sin(2.0 * np.pi * float(k) * t + ph2)

    ep_choices = (0.0, 0.25, 0.6, 1.2, 2.0, 3.0)
    ep = float(ep_choices[int(rng.integers(len(ep_choices)))])
    if ep > 0.0:
        z += ep * np.log(edge)

    nz = rng.normal(0.0, 0.08, n)
    if n > 5:
        nz = np.convolve(nz, np.ones(5, dtype=np.float64) / 5.0, mode="same")
    z += nz

    p = np.exp(z - float(np.max(z)))
    p = np.maximum(p, 1e-300)
    return _normalize(p)


def _shape_candidates(n, rng, count=7):
    out = []
    n = int(n)
    if n <= 1:
        return [np.ones(n, dtype=np.float64)]

    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    edge = np.sin(np.pi * t)
    edge = np.clip(edge, 1e-12, 1.0)

    out.append(_normalize(np.ones(n, dtype=np.float64)))

    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        out.append(_normalize(np.power(edge, p)))

    for k in (1, 2, 3, 4, 6, 8, 12, 16, 24):
        amp = float(rng.uniform(0.5, 2.5))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        z = np.log(edge) + amp * np.cos(2.0 * np.pi * float(k) * t + ph)
        out.append(_normalize(np.exp(z - float(np.max(z)))))

        amp2 = float(rng.uniform(0.3, 1.8))
        ph2 = float(rng.uniform(0.0, 2.0 * np.pi))
        z = np.log(edge) - amp2 * np.cos(2.0 * np.pi * float(k) * t + ph2)
        out.append(_normalize(np.exp(z - float(np.max(z)))))

    for _ in range(int(count)):
        out.append(_random_prior(n, rng))

    return out


def _initial_candidates(n, base_arr, global_arr, rng):
    n = int(n)
    L = _search_fft_len(n)
    m = 2 * n - 1

    cands = []
    if base_arr is not None and len(np.asarray(base_arr)) > 0:
        cands.append(_resample(base_arr, n))
    if global_arr is not None and len(np.asarray(global_arr)) > 0:
        cands.append(_resample(global_arr, n))
    cands.append(np.ones(n, dtype=np.float64) / float(n))
    cands.extend(_shape_candidates(n, rng))

    best_a = _normalize(np.ones(n, dtype=np.float64) / float(n))
    best_s = _score_a(best_a, L, m, n)

    for a in cands:
        try:
            a = _normalize(a)
            s = _score_a(a, L, m, n)
        except Exception:
            continue
        if math.isfinite(s) and s < best_s:
            best_s = s
            best_a = a
    return best_a, best_s


def _add_candidate(store_scores, store_arrays, arr, score, max_len=32):
    if arr is None:
        return
    try:
        score = float(score)
    except Exception:
        return
    if not math.isfinite(score):
        return
    a = np.asarray(arr, dtype=np.float64).ravel().copy()
    if a.size == 0:
        return

    if len(store_scores) < int(max_len):
        store_scores.append(score)
        store_arrays.append(a)
    else:
        worst_idx = 0
        worst_val = store_scores[0]
        for i in range(len(store_scores)):
            if store_scores[i] > worst_val:
                worst_val = store_scores[i]
                worst_idx = i
        if score < worst_val - 1e-14:
            store_scores[worst_idx] = score
            store_arrays[worst_idx] = a


def _kyt_vector(v, a_rev_fft, L, m, n):
    V = np.fft.rfft(v, L)
    out = np.fft.irfft(V * a_rev_fft, L)[n - 1:m]
    out = np.where(np.isfinite(out), out, 0.0)
    return out


def _cg_matvec(h, W, Aa, Aarev, L, m, n, reg):
    H = np.fft.rfft(h, L)
    y = np.fft.irfft(Aa * H, L)[:m]
    wy = W * y
    Z = np.fft.rfft(wy, L)
    z = np.fft.irfft(Z * Aarev, L)[n - 1:m]
    if reg != 0.0:
        z = z + reg * h
    z = np.where(np.isfinite(z), z, 0.0)
    return z


def _cg_solve_direction(a, r, W, Aa, Aarev, L, m, n, reg_factor=1e-8, max_iter=10, tol=1e-5):
    Wr = W * r
    V = np.fft.rfft(Wr, L)
    b = np.fft.irfft(V * Aarev, L)[n - 1:m]
    b = np.where(np.isfinite(b), b, 0.0)
    bnorm = float(np.linalg.norm(b))
    if bnorm <= 1e-300:
        return None

    h0 = np.ones(n, dtype=np.float64)
    if n > 1:
        h0[n // 2:] -= 1.0
    h0n = float(np.linalg.norm(h0))
    if h0n > 1e-300:
        h0 /= h0n
    z0 = _cg_matvec(h0, W, Aa, Aarev, L, m, n, 0.0)
    anorm = float(np.linalg.norm(z0))
    if not math.isfinite(anorm) or anorm <= 1e-300:
        anorm = 1.0
    reg = max(1e-300, float(reg_factor) * anorm)

    x = np.zeros(n, dtype=np.float64)
    res = b.copy()
    p = res.copy()
    rdot = float(np.dot(res, res))
    if rdot <= 1e-300:
        return None

    for _ in range(int(max_iter)):
        Ap = _cg_matvec(p, W, Aa, Aarev, L, m, n, reg)
        pap = float(np.dot(p, Ap))
        if not math.isfinite(pap) or abs(pap) <= 1e-300:
            break
        if pap < 0.0:
            pap = abs(pap)
        alpha = rdot / pap
        x += alpha * p
        res -= alpha * Ap
        rnew = float(np.dot(res, res))
        if not math.isfinite(rnew) or rnew <= 1e-300:
            break
        if math.sqrt(rnew) <= float(tol) * bnorm:
            break
        p = res + (rnew / rdot) * p
        rdot = rnew

    x = np.where(np.isfinite(x), x, 0.0)
    if float(np.linalg.norm(x)) > 1e-300:
        return x
    return None


def _weights_from_mode(c, mode):
    m = int(c.size)
    if m <= 0:
        return np.ones(1), 0.0

    kind = mode[0]
    if kind == 'uniform':
        w = np.ones(m, dtype=np.float64) / float(m)
        t = float(np.dot(w, c))
        return w, t

    cmax = float(np.max(c))
    denom = max(cmax, 1e-300)

    if kind == 'gn' or kind == 'grad':
        frac = float(mode[1])
        k = int(round(frac * float(m)))
        k = max(24, k)
        if k >= m:
            w = np.ones(m, dtype=np.float64) / float(m)
        else:
            idx = np.argpartition(c, -k)[-k:]
            w = np.zeros(m, dtype=np.float64)
            w[idx] = 1.0 / float(k)
        t = float(np.dot(w, c))
        return w, t

    if kind == 'gn_soft':
        beta = float(mode[1]) if len(mode) > 1 else 0.2
        alpha = float(mode[2]) if len(mode) > 2 else 0.2
        alpha = min(1.0, max(0.0, alpha))
        if beta <= 0.0:
            s = np.ones(m, dtype=np.float64) / float(m)
        else:
            zz = (c - cmax) / (beta * denom)
            zz = np.clip(zz, -80.0, 0.0)
            s = np.exp(zz)
            ss = float(s.sum())
            if not math.isfinite(ss) or ss <= 0.0:
                s = np.ones(m, dtype=np.float64) / float(m)
            else:
                s = s / ss
        w = (1.0 - alpha) * s + (alpha / float(m))
        ws = float(w.sum())
        if not math.isfinite(ws) or ws <= 0.0:
            w = np.ones(m, dtype=np.float64) / float(m)
        else:
            w = w / ws
        t = float(np.dot(w, c))
        return w, t

    # fallback
    w = np.ones(m, dtype=np.float64) / float(m)
    t = float(np.dot(w, c))
    return w, t


def _direction_z(a, c, mode, L, m, n, reg_factor=1e-8, max_iter=10):
    W, tval = _weights_from_mode(c, mode)
    a_rev_fft = np.fft.rfft(a[::-1], L)

    kind = mode[0]
    if kind == 'gn' or kind == 'gn_soft':
        r = W * (tval - c)
        r = np.where(np.isfinite(r), r, 0.0)
        if float(np.linalg.norm(r)) > 1e-300:
            Aa = np.fft.rfft(a, L)
            extra = 0
            if len(mode) > 1 and float(mode[1]) < 0.01:
                extra = 4
            h = _cg_solve_direction(
                a, r, W, Aa, a_rev_fft, L, m, n,
                reg_factor=reg_factor,
                max_iter=max_iter + extra,
                tol=1e-5
            )
            if h is not None:
                h = np.where(np.isfinite(h), h, 0.0)
                if float(np.linalg.norm(h)) > 1e-300:
                    z = np.zeros(n, dtype=np.float64)
                    mask = a > 1e-20
                    z[mask] = h[mask] / a[mask]
                    z = np.where(np.isfinite(z), z, 0.0)
                    z -= float(z.mean())
                    zmax = float(np.max(np.abs(z)))
                    if zmax > 1e-300:
                        z = z / zmax
                        return z

    # Gradient fallback / pure gradient mode.
    g = _kyt_vector(W, a_rev_fft, L, m, n)
    z = -g
    z = np.where(np.isfinite(z), z, 0.0)
    z -= float(z.mean())
    zmax = float(np.max(np.abs(z)))
    if zmax <= 1e-300:
        return None
    return z / zmax


def _gn_search(a, L, m, n, rng, timeLimit, reg_factor=1e-8, max_iter=10):
    t0 = time.monotonic()
    if timeLimit is None or timeLimit <= 0.0:
        a = _normalize(a)
        return a, _score_a(a, L, m, n)

    a = _normalize(a)
    best_a = a.copy()
    best_score = _score_a(a, L, m, n)

    deadline = t0 + float(timeLimit)
    modes = GN_MODES
    mode_shift = 0
    eta_scale = 0.55
    no_improve = 0
    it = 0
    it_since_best = 0

    while time.monotonic() < deadline - 0.25:
        c = _fft_square(a, L, m)
        current_score = _score_c(c, n)
        if not math.isfinite(current_score):
            a = best_a.copy()
            current_score = best_score
            c = _fft_square(a, L, m)

        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
            it_since_best = 0
        else:
            it_since_best += 1
            if it_since_best > 600:
                a = best_a.copy()
                current_score = best_score
                c = _fft_square(a, L, m)
                no_improve = 0
                it_since_best = 0
                it += 1
                continue

        mode = modes[(it + mode_shift) % len(modes)]
        z = _direction_z(a, c, mode, L, m, n, reg_factor=reg_factor, max_iter=max_iter)

        if z is None:
            prior = _random_prior(n, rng)
            mix = 0.08
            a = _normalize((1.0 - mix) * a + mix * prior)
            no_improve += 1
            it += 1
            continue

        etas = []
        for fac in (1.0, 0.55, 0.25, 0.10, 0.04):
            e = eta_scale * fac
            if e not in etas:
                etas.append(e)
        neg = -0.05 * eta_scale
        if neg not in etas:
            etas.append(neg)

        best_step = a
        best_step_score = current_score
        best_eta = 0.0

        for e in etas:
            if abs(e) < 1e-6:
                continue
            a2 = a * np.exp(np.clip(e * z, -50.0, 50.0))
            a2 = _normalize(a2)
            sc2 = _score_a(a2, L, m, n)
            if math.isfinite(sc2) and sc2 < best_step_score - 1e-13:
                best_step_score = sc2
                best_step = a2
                best_eta = e

        if best_eta != 0.0 and best_step_score < current_score - 1e-13:
            a = best_step
            no_improve = 0
            if best_eta > 0.0:
                if best_eta >= 0.95 * eta_scale:
                    eta_scale = min(3.0, eta_scale * 1.20)
                elif best_eta < 0.30 * eta_scale:
                    eta_scale = max(0.05, eta_scale * 0.80)
            else:
                eta_scale = max(0.05, eta_scale * 0.85)
        else:
            no_improve += 1
            if no_improve > 18 and no_improve % 4 == 0:
                mode_shift = int(rng.integers(len(modes)))
            if no_improve > 45:
                prior = _random_prior(n, rng)
                mix = min(0.65, 0.06 + 0.012 * no_improve)
                a = _normalize((1.0 - mix) * a + mix * prior)
                no_improve = 0
                it_since_best = min(it_since_best, 120)
            elif no_improve > 140:
                a = best_a.copy()
                no_improve = 0

        it += 1
        if it % 750 == 0:
            print(
                f"  gn n={n} it={it} current={current_score:.9f} best={best_score:.9f}",
                flush=True,
            )

    c = _fft_square(a, L, m)
    fs = _score_c(c, n)
    if math.isfinite(fs) and fs < best_score:
        best_score = fs
        best_a = a.copy()
    return best_a, best_score


def _mws_direction(f, prior, L, m, n, inner=18, eta_start=2.0, eta_end=40.0):
    f = _normalize(f)
    prior = _normalize(prior)
    inner = max(1, int(inner))

    F = np.fft.rfft(f, L)
    Frev = np.fft.rfft(f[::-1], L)

    x = prior.copy()
    y = np.full(m, 1.0 / float(m), dtype=np.float64)
    avg = np.zeros(n, dtype=np.float64)
    count = 0
    logprior = np.log(prior + 1e-300)

    for it in range(inner):
        frac = 0.0 if inner <= 1 else float(it) / float(inner - 1)
        eta = float(eta_start) + (float(eta_end) - float(eta_start)) * frac

        X = np.fft.rfft(x, L)
        u = np.fft.irfft(F * X, L)[:m]
        np.maximum(u, 0.0, out=u)
        vu = float(n) * u
        vmax = float(np.max(vu))
        if not math.isfinite(vmax) or vmax <= 0.0:
            y.fill(1.0 / float(m))
        else:
            z = eta * (vu - vmax)
            z = np.clip(z, -50.0, 0.0)
            e = np.exp(z)
            ss = float(e.sum())
            if math.isfinite(ss) and ss > 0.0:
                y = e / ss
            else:
                y.fill(1.0 / float(m))

        avg += x
        count += 1

        Y = np.fft.rfft(y, L)
        s = np.fft.irfft(Y * Frev, L)[n - 1:m]
        np.maximum(s, 0.0, out=s)
        ps = float(n) * s
        pmax = float(np.max(ps))
        if not math.isfinite(pmax) or pmax <= 0.0:
            x = prior.copy()
        else:
            zx = -eta * (ps - pmax) + 0.8 * logprior
            zx -= float(np.max(zx))
            zx = np.clip(zx, -50.0, 0.0)
            e = np.exp(zx)
            ss = float(e.sum())
            if math.isfinite(ss) and ss > 0.0:
                x = e / ss
            else:
                x = prior.copy()

    if count <= 0:
        return _normalize(prior)
    return _normalize(0.5 * (avg / float(count)) + 0.5 * x)


def _mws_search(a, L, m, n, rng, timeLimit, inner=16):
    t0 = time.monotonic()
    if timeLimit is None or timeLimit <= 0.0:
        a = _normalize(a)
        return a, _score_a(a, L, m, n)

    a = _normalize(a)
    best_a = a.copy()
    best_score = _score_a(a, L, m, n)

    if timeLimit > 200.0:
        inner = max(inner, 20)
    else:
        inner = max(10, int(inner))

    deadline = t0 + float(timeLimit)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline - 2.0:
        prior = _random_prior(n, rng)
        x = _mws_direction(best_a, prior, L, m, n, inner=inner, eta_start=2.0, eta_end=45.0)
        if x is None:
            x = prior

        cand = best_a
        cand_score = best_score
        for lam in (0.06, 0.15, 0.30, 0.50, 0.75, 1.0):
            mix = _normalize((1.0 - lam) * best_a + lam * x)
            sc = _score_a(mix, L, m, n)
            if math.isfinite(sc) and sc < cand_score - 1e-13:
                cand_score = sc
                cand = mix

        if cand_score < best_score - 1e-12:
            best_a = cand
            best_score = cand_score
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 25:
                for _try in range(2):
                    pr = _random_prior(n, rng)
                    sc = _score_a(pr, L, m, n)
                    if math.isfinite(sc) and sc < best_score - 1e-12:
                        best_a = pr
                        best_score = sc
                        break
                no_improve = 0

        it += 1
        if it % 150 == 0:
            print(
                f"  mws n={n} it={it} best={best_score:.9f}",
                flush=True,
            )

    c = _fft_square(a, L, m)
    fs = _score_c(c, n)
    if math.isfinite(fs) and fs < best_score:
        best_score = fs
        best_a = a.copy()
    return best_a, best_score

    if timeLimit is None or timeLimit <= 0.0:

    a = _normalize(a)
    best_a = a.copy()
    best_score = _score_a(a, L, m, n)

    no_improve = 0

    if n < 2:
        return best_a, best_score

    while time.monotonic() < deadline - 0.2:
        if rng.random() < 0.62:
            i = int(rng.integers(0, n - 1))
            j = i + 1
        else:
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                j = (j + 1) % n

        old_i = a[i]
        old_j = a[j]

        if rng.random() < 0.22:
            f = 0.25 + 1.45 * float(rng.random())
            ai = old_i * f
            aj = old_j + (old_i - ai)
            if not (math.isfinite(ai) and math.isfinite(aj)) or aj < 0.0 or ai < 0.0:
                ai = old_i + old_j
                aj = 0.0
        else:
            lam = 0.08 + 0.85 * float(rng.random())
            ai = (1.0 - lam) * old_i + lam * old_j
            aj = (1.0 - lam) * old_j + lam * old_i

        if math.isfinite(ai) and math.isfinite(aj) and ai >= 0.0 and aj >= 0.0:
            a[i] = ai
            a[j] = aj
            sc = _score_a(a, L, m, n)
            if math.isfinite(sc) and sc < best_score - 1e-13:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                a[i] = old_i
                a[j] = old_j
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > 120:
            a = best_a.copy()
            no_improve = 0

    c = _fft_square(a, L, m)
    fs = _score_c(c, n)
    if math.isfinite(fs) and fs < best_score:
        best_score = fs
        best_a = a.copy()
    return best_a, best_score


def _run_length(n, base_arr, global_arr, rng, timeLimit, store_scores, store_arrays):
    start = time.monotonic()
    if timeLimit is None or timeLimit <= 2.0:
        a, s = _initial_candidates(n, base_arr, global_arr, rng)
        _add_candidate(store_scores, store_arrays, a, s)
        return a, s

    n = int(n)
    L = _search_fft_len(n)
    m = 2 * n - 1

    a0, s0 = _initial_candidates(n, base_arr, global_arr, rng)
    _add_candidate(store_scores, store_arrays, a0, s0)
    best_a = a0.copy()
    best_s = s0
    print(
        f"time={time.monotonic()-start:7.2f}s length={n:6d} init={s0:.9f}",
        flush=True,
    )

    if timeLimit < 60.0:
        a1, s1 = _gn_search(a0, L, m, n, rng, max(1.0, timeLimit - 1.0),
                            reg_factor=1e-8, max_iter=8)
        if s1 < best_s:
            best_a, best_s = a1, s1
        _add_candidate(store_scores, store_arrays, best_a, best_s)
        return best_a, best_s

    rem = float(timeLimit) - (time.monotonic() - start)
    t1 = min(0.46 * float(timeLimit), max(5.0, rem - 5.0))
    if t1 > 5.0:
        a1, s1 = _gn_search(a0, L, m, n, rng, t1, reg_factor=1e-8, max_iter=9)
        _add_candidate(store_scores, store_arrays, a1, s1)
        if s1 < best_s:
            best_a, best_s = a1, s1

    rem = float(timeLimit) - (time.monotonic() - start)
    t2 = min(0.26 * float(timeLimit), max(0.0, rem - 25.0))
    if t2 > 6.0:
        a2, s2 = _mws_search(best_a, L, m, n, rng, t2, inner=18)
        _add_candidate(store_scores, store_arrays, a2, s2)
        if s2 < best_s:
            best_a, best_s = a2, s2

    rem = float(timeLimit) - (time.monotonic() - start)
    t3 = min(0.34 * float(timeLimit), max(0.0, rem))
    if t3 > 6.0:
        a3, s3 = _gn_search(best_a, L, m, n, rng, t3, reg_factor=5e-9, max_iter=13)
        _add_candidate(store_scores, store_arrays, a3, s3)
        if s3 < best_s:
            best_a, best_s = a3, s3

    rem = float(timeLimit) - (time.monotonic() - start)
    tc = min(45.0, max(0.0, rem - 1.0))
    if tc > 2.0:
        a4, s4 = _coordinate_polish(best_a, L, m, n, rng, tc)
        _add_candidate(store_scores, store_arrays, a4, s4)
        if s4 < best_s:
            best_a, best_s = a4, s4

    print(
        f"time={time.monotonic()-start:7.2f}s length={n:6d} best={best_s:.9f}",
        flush=True,
    )
    return best_a, best_s


def _make_final_list(arr, fallback_n=6121):
    try:
        a = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        a = np.zeros(0, dtype=np.float64)
    if a.size == 0:
        n = int(fallback_n)
        return [1.0 / float(n) for _ in range(n)]

    out = []
    for x in a:
        fx = float(x)
        if not math.isfinite(fx) or fx < 0.0:
            fx = 0.0
        if fx > 1000.0:
            fx = 1000.0
        out.append(fx)

    s = float(sum(out))
    if not math.isfinite(s) or s <= 0.0:
        n = int(fallback_n)
        return [1.0 / float(n) for _ in range(n)]
    if s < 0.01:
        scale = 1.0 / s
        out = [float(x * scale) for x in out]
        s = float(sum(out))
        if s <= 0.0:
            n = int(fallback_n)
            return [1.0 / float(n) for _ in range(n)]
    return out


def propose_candidate(seed=42, budget_s=2400):
    start = time.monotonic()
    try:
        total = float(budget_s)
    except Exception:
        total = 2400.0
    if not math.isfinite(total) or total <= 0.0:
        total = 2400.0
    total = max(10.0, total)

    rng = np.random.default_rng(seed)
    base_arr = _get_base()

    store_scores = []
    store_arrays = []

    global_best_arr = None
    global_best_score = float("inf")

    if base_arr is not None:
        bs = _score_array(base_arr)
        _add_candidate(store_scores, store_arrays, base_arr, bs)
        global_best_arr = base_arr
        global_best_score = bs
        print(f"base length={base_arr.size} internal={bs:.9f}", flush=True)

    # Choose search lengths.  The base length is usually the safest final
    # length; larger lengths can reduce discretization error but cost more.
    if base_arr is not None and base_arr.size > 0:
        n0 = int(base_arr.size)
    else:
        n0 = 6121

    if n0 < 512:
        n0 = 6121
    if n0 > 15000:
        n0 = 12288

    lengths = [n0, 10240]
    if total >= 1500.0:
        lengths.append(12288)

    uniq = []
    for x in lengths:
        x = int(x)
        if 256 <= x <= 16000 and x not in uniq:
            uniq.append(x)
    if not uniq:
        uniq = [6121]

    k = len(uniq)
    if k == 1:
        fractions = [1.0]
    elif k == 2:
        fractions = [0.58, 0.42]
    else:
        fractions = [0.46, 0.31, 0.23]

    final_reserve = 30.0 if total > 300.0 else max(5.0, 0.10 * total)
    search_total = max(10.0, total - final_reserve)
    search_deadline = start + search_total

    for idx in range(len(uniq)):
        now = time.monotonic()
        if now > search_deadline - 10.0:
            break
        n = uniq[idx]
        cap = fractions[idx] * search_total
        rem = search_deadline - now
        tlim = min(cap, rem)
        if tlim <= 5.0:
            continue
        print(f"--- block length={n} allocated={tlim:.2f}s ---", flush=True)

        a_best, s_best = _run_length(
            n, base_arr, global_best_arr, rng, tlim, store_scores, store_arrays
        )
        if s_best < global_best_score:
            global_best_score = s_best
            global_best_arr = a_best

    # Use any remaining search time to polish the current global best.
    if global_best_arr is not None:
        n = int(np.asarray(global_best_arr).size)
        L = _search_fft_len(n)
        m = 2 * n - 1
        remaining = max(0.0, total - max(20.0, total * 0.03) - (time.monotonic() - start))
        if remaining > 30.0 and n <= 15000:
            a_pol, s_pol = _gn_search(
                global_best_arr, L, m, n, rng, remaining * 0.8,
                reg_factor=3e-9, max_iter=14
            )
            _add_candidate(store_scores, store_arrays, a_pol, s_pol)
            if s_pol < global_best_score:
                global_best_score = s_pol
                global_best_arr = a_pol

            rem2 = max(0.0, total - max(15.0, total * 0.02) - (time.monotonic() - start))
            if rem2 > 8.0:
                a_co, s_co = _coordinate_polish(global_best_arr, L, m, n, rng, min(45.0, rem2))
                _add_candidate(store_scores, store_arrays, a_co, s_co)
                if s_co < global_best_score:
                    global_best_score = s_co
                    global_best_arr = a_co

    if global_best_arr is None and store_arrays:
        # choose best internal from store
        bidx = 0
        for i in range(len(store_arrays)):
            if store_scores[i] < store_scores[bidx]:
                bidx = i
        global_best_arr = store_arrays[bidx]
        global_best_score = store_scores[bidx]

    if global_best_arr is None:
        global_best_arr = np.ones(6121, dtype=np.float64) / 6121.0

    # Exact final selection.  Evaluate only a few candidates and prefer
    # reasonably small lengths if scores are very close, to keep the final
    # direct convolution exact cheap enough.
    best_seq = _make_final_list(global_best_arr)
    best_exact = _exact_score(best_seq)
    if not math.isfinite(best_exact):
        best_exact = float("inf")

    if len(store_arrays) > 0:
        top_count = min(3, len(store_arrays))
        top_indices = []
        for _ in range(top_count):
            bi = -1
            bv = float("inf")
            for i in range(len(store_arrays)):
                if i in top_indices:
                    continue
                ni = int(store_arrays[i].size)
                penalty = 0.0006 * max(0.0, (float(ni) - 10240.0) / 10240.0)
                val = store_scores[i] + penalty
                if val < bv:
                    bv = val
                    bi = i
            if bi < 0:
                break
            top_indices.append(bi)

        hard_exact = start + total - 4.0
        for i in top_indices:
            if time.monotonic() > hard_exact - 2.0:
                break
            arr = store_arrays[i]
            seq = _make_final_list(arr)
            # If a very long sequence is selected and time is tight, skip exact
            # and rely on FFT score except for the overall best.
            ni = int(len(seq))
            if ni > 13000 and time.monotonic() > hard_exact - 6.0:
                continue
            val = _exact_score(seq)
            if math.isfinite(val) and val < best_exact:
                best_exact = val
                best_seq = seq

    # Final safety guard: never let the program overrun the hard limit here.
    if time.monotonic() > start + total - 1.0:
        return best_seq

    # One more sanity check on the returned sequence.
    if best_seq is None or len(best_seq) == 0:
        best_seq = _make_final_list(global_best_arr)
    for x in best_seq:
        if not isinstance(x, float):
            best_seq = [float(y) for y in best_seq]
            break

    return best_seq
