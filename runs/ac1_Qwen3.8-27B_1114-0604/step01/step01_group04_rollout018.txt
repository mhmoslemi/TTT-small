import numpy as np

def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def conv_w_a(w, a):
    """Compute conv(w, a) for w of length 2n-1 and a of length n, return first n elements."""
    n = a.size
    fsize = 1 << (w.size + n - 2).bit_length()  # length of full conv is (2n-1) + n - 1 = 3n-2
    fw = np.fft.rfft(w, fsize)
    fa = np.fft.rfft(a, fsize)
    result = np.fft.irfft(fw * fa, fsize)
    return result[:n]

def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    w_2i = w[2 * np.arange(n)]  # w[0], w[2], w[4], ..., w[2(n-1)]
    grad = 2.0 * conv_wa - w_2i * a
    grad *= lp_val

def score_from_conv(a, c):
    """Compute score given a and its convolution c."""
    n = a.size
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * c.max() / (s * s)

    # s[i] = conv_result[i + n - 1] for i = 0, ..., n-1
    return conv_result[n-1 : 2*n-1]

"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using L_p-smoothed
subgradient descent with FFT-based O(n log n) gradients, multiple
sequence lengths (up to 8192), adaptive p-scheduling, momentum,
and coordinate descent polish. Key innovation: exact L_p gradient
computed via a single FFT convolution per iteration.
"""

import numpy as np
import time


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_gradient(c, a, p):
    """
    Compute gradient of (sum_k c_k^p)^(1/p) w.r.t. a.
    c = conv(a,a) has length 2n-1, a has length n.
    Uses FFT for the key convolution step.
    """
    n = a.size
    # Weights: w_k = c_k^(p-1) / sum(c_k^p)
    c_clipped = np.maximum(c, 1e-300)
    cp = c_clipped ** p
    sum_cp = cp.sum()
    w = cp / sum_cp  # length 2n-1, sums to 1
    
    # s[i] = sum_{j=0}^{n-1} w[i+j] * a[j]
    # = conv(w, a_rev)[i + n - 1]
    a_rev = a[::-1]
    fsize = 1 << (3 * n - 3).bit_length()
    fw = np.fft.rfft(w, fsize)
    fa = np.fft.rfft(a_rev, fsize)
    conv_result = np.fft.irfft(fw * fa, fsize)
    s = conv_result[n - 1:2 * n - 1]  # length n
    
    # grad[i] = 2*s[i] - w[2*i]*a[i]
    w_2i = w[2 * np.arange(n)]
    grad = 2.0 * s - w_2i * a
    
    # Scale by L_p value: (sum c^p)^(1/p)
    lp_val = sum_cp ** (1.0 / p)
    grad *= lp_val
    
    return grad


def score_from_conv(a, c):
    """Compute 2*n*max(c)/sum(a)^2."""
    n = a.size
    s = a.sum()
    if s < 1e-15:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (s * s)


def lp_subgrad_optimize(a0, time_budget, rng, p=30, eta=0.3, beta=0.7):
    """
    L_p smoothed subgradient descent with momentum.
    a0 is normalized to sum=1.
    """
    n = a0.size
    a = project_simplex(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = score_from_conv(a, full_conv(a))
    
    vel = np.zeros(n)
    iters = 0
    last_improve = 0
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_budget:
        c = full_conv(a)
        cur_sc = score_from_conv(a, c)
        
        grad = lp_gradient(c, a, p)
        
        # Normalize gradient for stable step
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            grad = grad / gnorm
        
        vel = beta * vel + (1.0 - beta) * grad
        a_new = project_simplex(a - eta * vel)
        c_new = full_conv(a_new)
        new_sc = score_from_conv(a_new, c_new)
        
        if new_sc <= cur_sc:
            a = a_new
        elif rng.random() < 0.15:
            a = a_new
        else:
            a = project_simplex(a - 0.5 * eta * vel)
        
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = iters
        
        iters += 1
        
        # Restart if stalled
        if iters - last_improve > 2000:
            a = project_simplex(best_a + 0.005 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.1 + 0.4 * rng.random()
            p = int(15 + 30 * rng.random())
            last_improve = iters
        
        # Slowly decrease p (sharpen towards max)
        if iters % 1000 == 0 and p > 8:
            p = max(5, p - 1)
    
    return best_a, best_sc


def coordinate_descent_polish(a0, time_budget, rng):
    """
    Coordinate descent: randomly pick coordinates, try multiple values,
    keep the best.
    """
    n = a0.size
    a = project_simplex(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = score_from_conv(a, full_conv(a))
    
    t0 = time.monotonic()
    nCoord = 2 * max(n, 32)
    
    while time.monotonic() - t0 < time_budget:
        i = rng.integers(0, n)
        old_val = a[i]
        if old_val < 1e-15:
            continue
        
        multipliers = [0.3, 0.5, 0.7, 1.3, 1.5, 0.1, 2.0, 0.0]
        rng.shuffle(multipliers)
        
        for m in multipliers:
            a[i] = old_val * m
            if a[i] < 0:
                a[i] = 0
            a[i] = min(a[i], 10.0)
            # Renormalize
            a = project_simplex(a)
            sc = score_from_conv(a, full_conv(a))
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
        
        # Reset to best
        a = best_a.copy()
    
    return best_a, best_sc


def init_u_shape(n):
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i).astype(np.float64) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_sided(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / c) ** 2 + 0.1
    return a / a.sum()


def init_two_gauss(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.2 + 0.1 * rng.random())
    c2 = n * (0.7 + 0.1 * rng.random())
    sig = n * (0.05 + 0.05 * rng.random())
    a = np.exp(-((i - c1) / sig) ** 2) + np.exp(-((i - c2) / sig) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_flat_with_noise(n, rng):
    a = np.ones(n) + 0.5 * rng.standard_normal(n)
    a = np.abs(a) + 0.01
    return a / a.sum()


def init_decay(n, rng):
    i = np.arange(n, dtype=np.float64)
    alpha = 0.5 + rng.random()
    a = np.exp(-alpha * i / n) + np.exp(-alpha * (n - 1 - i) / n)
    a += 0.01
    return a / a.sum()


def init_cosine_struct(n, rng):
    i = np.arange(n, dtype=np.float64)
    f1 = 1 + int(rng.integers(0, 3))
    f2 = 2 + int(rng.integers(0, 4))
    a = (1.0 + np.cos(2 * np.pi * f1 * i / n)) * (1.0 + np.cos(2 * np.pi * f2 * i / n))
    a = np.abs(a) + 0.01
    return a / a.sum()


def get_inits(n, rng):
    inits = []
    inits.append(init_u_shape(n))
    inits.append(init_sided(n))
    inits.append(init_two_gauss(n, rng))
    inits.append(init_flat_with_noise(n, rng))

    inits.append(init_cosine_struct(n, rng))
    # Random

    inits.append(a / a.sum())
    return inits
