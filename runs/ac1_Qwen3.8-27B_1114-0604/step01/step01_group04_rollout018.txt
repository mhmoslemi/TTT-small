def lp_solve_direction(p, M):
    """
    Solve LP: max sum(g) s.t. conv(p,g)_k <= M for all k, g >= 0.
    Returns optimal g or None on failure.
    """
    n = p.size
    A = build_lp_matrix(p)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def lp_refine(p, time_budget, rng, max_iters=500000):
    """
    LP-based refinement: at each step, solve the LP to get optimal direction,
    interpolate towards it, and update M.
    """
    n = p.size
    p = p.astype(np.float64).copy()
    b = full_conv(p)
    M = float(np.max(b))
    best_p = p.copy()
    best_sc = 2.0 * n * M

    t0 = time.monotonic()
    t_step = 0.5
    it = 0
    last_improve = 0

    while time.monotonic() - t0 < time_budget and it < max_iters:
        g = lp_solve_direction(p, M)
        if g is None or g.sum() < 1e-15:
            t_step = max(0.1, t_step * 0.8)
            it += 1
            continue

        g_norm = g / g.sum()
        p_new = (1.0 - t_step) * p + t_step * g_norm
        p_new = np.maximum(p_new, 0.0)
        s = p_new.sum()
        if s < 1e-15:
            break
        p_new /= s

        b_new = full_conv(p_new)
        M_new = float(np.max(b_new))
        sc_new = 2.0 * n * M_new

        it += 1

        if sc_new < best_sc:
            best_sc = sc_new
            best_p = p_new.copy()
            p = p_new
            M = M_new
            last_improve = it
            # Slightly increase step if we're making good progress
            if it - last_improve <= 1:
                t_step = min(0.8, t_step * 1.05)
        else:
            t_step *= 0.8
            if t_step < 0.05:
                # Perturb and restart
                p = best_p + 0.02 * rng.standard_normal(n)
                p = project_simplex(p)
                b = full_conv(p)
                M = float(np.max(b))
                t_step = 0.3
                last_improve = it

        if it - last_improve > 2000:
            p = project_simplex(best_p + 0.01 * rng.standard_normal(n))
            b = full_conv(p)
            M = float(np.max(b))
            t_step = 0.3 + 0.3 * rng.random()
            last_improve = it

    return best_p, best_sc


def init_uniform(n):
    return np.ones(n) / n


def init_cosine(n, freq=1):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + freq * np.cos(2.0 * np.pi * i / n)
    a = np.maximum(a, 0.0) + 1e-4
    return a / a.sum()


def init_two_bump(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.2 + 0.1 * rng.random())
    c2 = n * (0.7 + 0.1 * rng.random())
    sigma = n * (0.05 + 0.05 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.05 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def init_u_shape(n):
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_arc_sine(n):
    """Arc-sine-like distribution: high at edges."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = 1.0 / np.sqrt(i * (n + 1 - i))
    return a / a.sum()


def init_multi_cos(n, rng):
    """Sum of multiple cosines with random frequencies."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for _ in range(3):
        f = rng.integers(1, max(2, n // 4))
        phase = rng.random() * 2.0 * np.pi
        a += 0.5 * np.cos(2.0 * np.pi * f * i / n + phase)
    a = np.maximum(a, 0.0) + 0.01
    return a / a.sum()


def get_initializations(n, rng, prev_p=None):
    """Return list of initializations for sequence length n."""
    inits = [
        init_uniform(n),
        init_cosine(n, 1),
        init_cosine(n, 2),
        init_two_bump(n, rng),
        init_random(n, rng),
        init_u_shape(n),
        init_arc_sine(n),
        init_multi_cos(n, rng),
    ]
    if prev_p is not None and prev_p.size == n:
        inits.insert(0, prev_p.copy())
    return inits


def polish_local(p, time_budget, rng):
    """Short Adam polish around a good solution."""
    n = p.size
    return adam_logspace(n, p, time_budget, rng, lr=0.3, max_tau=0.02, min_tau=0.002)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses natural-gradient Adam in log-space + LP refinement.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    # Try to load previous best
    prev_p = None
    prev_sc = np.inf
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size >= 10 and np.all(np.isfinite(h)) and h.sum() > 0.01:
            prev_p = h / h.sum()
            prev_sc = 2.0 * h.size * float(np.max(full_conv(prev_p)))
            print(f"[init] Previous best: n={h.size}, score={prev_sc:.6f}")
    except Exception:
        pass

    best_overall = None
    best_overall_sc = np.inf

    if prev_p is not None and prev_sc < 2.0:
        best_overall = prev_p.copy()
        best_overall_sc = prev_sc

    # Sequence lengths to try
    n_list = [128, 256, 512]
    if prev_p is not None:
        np_ = prev_p.size
        if np_ not in n_list and 64 <= np_ <= 1024:
            n_list.append(np_)
    n_list = sorted(set(n_list))

    total_time = max(1.0, deadline - time.monotonic())
    # Allocate time: smaller n gets more (faster per iter, more inits to try)
    raw_w = [1.0 / (0.3 + nv / 256.0) for nv in n_list]
    total_w = sum(raw_w)
    weights = [w / total_w for w in raw_w]

    print(f"[plan] n_list={n_list}, total_time={total_time:.1f}s")

    # Phase 1: Adam optimization
    adam_time_frac = 0.55
    adam_budget = total_time * adam_time_frac

    for idx_n, nv in enumerate(n_list):
        if time.monotonic() > deadline:
            break

        time_n = min(weights[idx_n] * adam_budget, deadline - time.monotonic() - 30.0)
        if time_n < 5.0:
            break

        inits = get_initializations(nv, rng, prev_p)
        inits_per = max(3.0, time_n / len(inits))

        print(f"[adam] n={nv}, {len(inits)} inits, {time_n:.1f}s budget")

        for ii, a_init in enumerate(inits):
            if time.monotonic() > deadline - 30.0:
                break
            tb = min(inits_per, deadline - time.monotonic() - 30.0)
            if tb < 2.0:
                break

            a_best, sc_best = adam_logspace(nv, a_init, tb, rng)

            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = a_best.copy()
                print(f"[adam] n={nv}, init={ii}, score={sc_best:.6f} (BEST)")
            else:
                print(f"[adam] n={nv}, init={ii}, score={sc_best:.6f}")

    print(f"[adam] Phase 1 done. Best: {best_overall_sc:.6f}")

    # Phase 2: LP refinement
    lp_time_frac = 0.35
    lp_budget = total_time * lp_time_frac

    if best_overall is not None:
        remain = min(lp_budget, deadline - time.monotonic() - 20.0)
        if remain > 5.0:
            print(f"[lp] Refining n={best_overall.size}, budget={remain:.1f}s")
            a_lp, sc_lp = lp_refine(best_overall, remain, rng)
            if sc_lp < best_overall_sc:
                best_overall_sc = sc_lp
                best_overall = a_lp.copy()
                print(f"[lp] Improved to {sc_lp:.6f}")
            else:
                print(f"[lp] No improvement (best={best_overall_sc:.6f})")

    # Phase 3: Final polish with Adam
    polish_budget = min(total_time * 0.1, deadline - time.monotonic() - 10.0)
    if best_overall is not None and polish_budget > 3.0:
        print(f"[polish] budget={polish_budget:.1f}s, n={best_overall.size}")
        a_pol, sc_pol = polish_local(best_overall, polish_budget, rng)
        if sc_pol < best_overall_sc:
            best_overall_sc = sc_pol
            best_overall = a_pol.copy()
            print(f"[polish] Improved to {sc_pol:.6f}")

        # One more LP round if time allows
        if time.monotonic() < deadline - 15.0:
            remain2 = min(30.0, deadline - time.monotonic() - 10.0)
            if remain2 > 3.0:
                a_lp2, sc_lp2 = lp_refine(best_overall, remain2, rng)
                if sc_lp2 < best_overall_sc:
                    best_overall_sc = sc_lp2
                    best_overall = a_lp2.copy()
                    print(f"[polish-lp] Improved to {sc_lp2:.6f}")

    # Fallback
    if best_overall is None:
        best_overall = np.ones(128) / 128
        best_overall_sc = 2.0

    # Final verification
    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"[final] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")

    return result
