import numpy as np

"""
Algorithm: Adam-based gradient optimization in log-space with adaptive smoothing.
Minimizes 2n * max(a*a) / (sum(a))^2 for non-negative sequences a.
Uses softmax parameterization, FFT-based convolution, and adaptive learning rate.

def _opt_state(n):
    """Set up optimization state for sequence of length n."""
    NFFT = 1 << (3 * n - 2).bit_length()
    return n, NFFT

def _softmax(z, bound=30.0):
    z = np.clip(z, -bound, bound)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(z) / z.size
    return e / s

def _fft_conv(a, NFFT):
    n = a.size
    fa = np.fft.rfft(a, NFFT)
    c = np.fft.irfft(fa * fa, NFFT)[:2*n-1].real
    return np.maximum(c, 0.0)

def _score_normalized(a):
    """Score assuming sum(a) = 1. Returns 2n * max(a*a)."""
    n = a.size
    c = _fft_conv(a, 1 << (3*n-2).bit_length())
    return 2.0 * n * float(c.max())

def _obj_grad(z, tau, NFFT, n):
    """Compute smooth objective and gradient w.r.t. z."""
    a = _softmax(z)
    c = _fft_conv(a, NFFT)
    
    m = c.max()
    diff = (c - m) / tau
    diff = np.clip(diff, -745.0, 0.0)
    e = np.exp(diff)
    se = e.sum()
    if se <= 0 or not np.isfinite(se):
        return 1e10, np.zeros(n)
    
    p = e / se
    fa = np.fft.rfft(a, NFFT)
    fp = np.fft.rfft(p, NFFT)
    h = np.fft.irfft(fp * np.conjugate(fa), NFFT)[:n].real
    
    s_h = float(h @ a)
    g = a * (h - s_h)
    grad = 2.0 * g

"""
