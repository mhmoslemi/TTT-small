def _lp_direction(f, M, tol=1e-6):
    """Find direction g >= 0 with (f*g)[k] <= M for all k, maximizing sum(g).
    f has sum 1, M is the current peak of f*f.
    Returns g normalized to sum 1, or None if infeasible."""
    n = len(f)
    # Variables: g[0], ..., g[n-1]
    # Objective: maximize sum(g) => minimize -sum(g)
    c_obj = -np.ones(n)
    
    # Constraints: (f*g)[k] <= M for k = 0, ..., 2n-2
    # (f*g)[k] = sum_j f[j] * g[k-j] (for valid j, k-j)
    # This is a linear function of g: sum_j f[j] * g[k-j]
    # For a fixed k: (f*g)[k] = sum_{j: 0<=j<n, 0<=k-j<n} f[j] * g[k-j]
    # Let i = k-j, so j = k-i: sum_{i: 0<=i<n, 0<=k-i<n} f[k-i] * g[i]
    # = sum_{i=max(0,k-n+1)}^{min(n-1,k)} f[k-i] * g[i]
    
    # Build constraint matrix A (shape: (2n-1, n))
    # A[k, i] = f[k-i] if 0 <= k-i < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_min = max(0, k-n+1)
        i_max = min(n-1, k)
        for i in range(i_min, i_max+1):
            A[k, i] = f[k-i]
    
    b = np.full(2*n-1, M)
    
    # Solve: minimize c_obj @ g subject to A @ g <= b, g >= 0
    res = linprog(c_obj, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.status != 0:
        return None
    
    g = res.x
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s

"""
Step function optimizer minimizing 2*n*max(conv(f,f))/(sum f)^2.
Uses LP-based direction finding (natural gradient for the peak constraint),
Fista acceleration, soft-max multi-peak targeting, simulated annealing,
and multiple diverse initializations.
"""

import numpy as np
from scipy.signal import correlate, fftconvolve
from scipy.optimize import linprog
import time


def _convolve_full(f):
    """Full linear convolution using FFT."""
    return fftconvolve(f, f, mode='full')


def _soft_gradient(f, c, T):
    """Soft-max gradient: weighted sum of peak gradients."""
    n = len(f)
    cs = c / T
    cs -= cs.max()
    p = np.exp(cs)
    s = p.sum()
    if s < 1e-300:
        k_star = int(np.argmax(c))
        grad = np.zeros(n)
        for i in range(n):
            idx = k_star - i
            if 0 <= idx < n:
                grad[i] = 2.0 * f[idx]
        return grad
    p /= s
    return 2.0 * correlate(p, f, mode='valid')


def _project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    s = v.sum()
    if s < 1e-300:
        return np.ones(n) / n
    v = v / s
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.max(np.nonzero(cond)[0]))
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _eval(f):
    """Evaluate the sequence."""
    return evaluate_sequence(f.tolist())


def _lp_direction(f, M, K=20, max_time=5.0):
    """Solve reduced LP to find direction g maximizing sum(g)
    subject to g >= 0 and (f*g)[k] <= M for top K peaks of f*f."""
    n = len(f)
    c = _convolve_full(f)
    # Get top K peak indices
    top_k = np.argpartition(c, -K)[-K:]
    
    # Build constraint matrix A (K x n): A[idx, i] = f[k - i] if valid
    A = np.zeros((K, n))
    for idx, k in enumerate(top_k):
        k = int(k)
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            A[idx, i_min:i_max+1] = f[k - i_max:k - i_min + 1][::-1]
    
    b = np.full(K, M)
    
    start_time = time.monotonic()
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': max_time})
    except Exception:
        return None
    
    if res.status != 0 or time.monotonic() - start_time > max_time + 1:
        return None
    
    g = res.x
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s


def _mirror_descent_step(f, grad, eta):
    """Exponentiated gradient (mirror descent) step on the simplex."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    return f_new / s


def _optimize_lp_based(f, n, deadline, rng, verbose=False, label=""):
    """Main LP-based optimization loop with SA and line search."""
    f = f.copy()
    best_f = f.copy()
    best_score = _eval(f)
    
    c = _convolve_full(f)
    M = c.max()
    
    it = 0
    stuck_count = 0
    T_sa = 0.5  # SA temperature for accepting worse LP solutions
    
    while time.monotonic() < deadline:
        it += 1
        
        # Solve LP
        g = _lp_direction(f, M, K=20, max_time=3.0)
        
        if g is None:
            # Fallback: use gradient step
            c = _convolve_full(f)
            M = c.max()
            T = M * 0.1
            grad = _soft_gradient(f, c, T)
            eta = 0.5 / (np.linalg.norm(grad) + 1e-10)
            f = _mirror_descent_step(f, grad, eta)
            c = _convolve_full(f)
            M = c.max()
            continue
        
        # Line search: try different blend fractions
        best_t = 0.0
        best_peak = M
        for t in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            f_try = (1 - t) * f + t * g
            f_try = _project_simplex(f_try)
            c_try = _convolve_full(f_try)
            peak_try = c_try.max()
            if peak_try < best_peak:
                best_peak = peak_try
                best_t = t
        
        if best_t == 0.0:
            # No improvement; try SA
            t = 0.3
            f_try = _project_simplex((1 - t) * f + t * g)
            c_try = _convolve_full(f_try)
            peak_try = c_try.max()
            if rng.random() < np.exp(-(peak_try - M) / (T_sa + 1e-10)):
                f = f_try
                M = peak_try
            else:
                # Perturb
                f = f + 0.01 * rng.standard_normal(n)
                f = np.maximum(f, 0)
                f = f / f.sum()
                c = _convolve_full(f)
                M = c.max()
            stuck_count += 1
        else:
            f = _project_simplex((1 - best_t) * f + best_t * g)
            c = _convolve_full(f)
            M = c.max()
            stuck_count = 0
        
        # Periodic evaluation
        if it % 50 == 0:
            score = _eval(f)
            if score < best_score:
                best_score = score
                best_f = f.copy()
                if verbose:
                    print(f"  {label} iter {it}: {score:.6f}")
            # Decay SA temperature
            T_sa *= 0.95
        
        # If stuck for too long, do a bigger perturbation
        if stuck_count > 10:
            f = f + 0.05 * rng.standard_normal(n)
            f = np.maximum(f, 0)
            f = f / f.sum()
            c = _convolve_full(f)
            M = c.max()
            stuck_count = 0
            T_sa = 0.5  # Reset SA temperature
    
    score = _eval(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score


def _gradient_anneal(f, n, deadline, rng, verbose=False, label=""):
    """Mirror descent with soft-max gradient and temperature annealing."""
    f = f.copy()
    best_f = f.copy()
    best_score = _eval(f)
    
    c = _convolve_full(f)
    peak0 = c.max()
    T = peak0 * 0.1
    T_min = peak0 * 1e-8
    
    n_iters = 50000
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        progress = it / n_iters
        T = peak0 * 0.1 * (T_min / (peak0 * 0.1)) ** (progress ** 0.5)
        step = 0.8 * (0.02 / 0.8) ** (progress ** 0.7)
        
        c = _convolve_full(f)
        grad = _soft_gradient(f, c, T)
        
        eta = step / (np.std(grad) + 1e-10)
        f_new = _mirror_descent_step(f, grad, eta)
        
        # Accept if better or with SA probability
        c_new = _convolve_full(f_new)
        if c_new.max() <= c.max() or (it > n_iters * 0.5 and rng.random() < 0.1):
            f = f_new
        
        if it % 5000 == 0:
            score = _eval(f)
            if score < best_score:
                best_score = score
                best_f = f.copy()
                if verbose:
                    print(f"  {label} iter {it}: {score:.6f}")
    
    score = _eval(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score


def _polish_coordinate(f, n, deadline, rng):
    """Coordinate descent polish: move mass between pairs of coordinates."""
    f = f.copy()
    best_f = f.copy()
    best_score = _eval(f)
    c = _convolve_full(f)
    peak = c.max()
    
    while time.monotonic() < deadline:
        # Find the peak lag
        k_star = int(np.argmax(c))
        
        # Compute contribution of each pair to the peak
        # c[k_star] = sum_i f[i] * f[k_star - i]
        # Try moving mass from high-contribution to low-contribution positions
        
        improved = False
        for _ in range(200):
            if time.monotonic() > deadline:
                break
            
            # Find positions that contribute most to the peak
            contrib = np.zeros(n)
            for i in range(n):
                j = k_star - i
                if 0 <= j < n:
                    contrib[i] = f[i] * f[j]
            
            # Random pair swap
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            
            # Try small transfer
            for frac in [0.01, 0.05, 0.1]:
                amt = f[i] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[i] -= amt
                if f_try[i] < 0:
                    continue
                f_try[j] += amt
                f_try = _project_simplex(f_try)
                
                c_try = _convolve_full(f_try)
                if c_try.max() < peak - 1e-12:
                    f = f_try
                    peak = c_try.max()
                    c = c_try
                    improved = True
                    break
        
        if not improved:
            # Try targeted move: reduce the highest contributor
            i_donate = int(np.argmax(contrib))
            j_receive = int(np.argmin(contrib))
            if i_donate == j_receive:
                break
            for frac in [0.01, 0.05, 0.1, 0.2]:
                amt = f[i_donate] * frac
                f_try = f.copy()
                f_try[i_donate] -= amt
                if f_try[i_donate] < 0:
                    continue
                f_try[j_receive] += amt
                f_try = _project_simplex(f_try)
                c_try = _convolve_full(f_try)
                if c_try.max() < peak - 1e-12:
                    f = f_try
                    peak = c_try.max()
                    c = c_try
                    break
            else:
                # No improvement, perturb
                f = f + 0.005 * rng.standard_normal(n)
                f = np.maximum(f, 0)
                f = _project_simplex(f)
                c = _convolve_full(f)
                peak = c.max()
    
    score = _eval(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score

"""
Step function optimizer minimizing 2*n*max(conv(f,f))/(sum f)^2.
Uses LP-based natural gradient direction finding with line search,
soft-max multi-peak targeting, simulated annealing, Fista-style acceleration,
and coordinate descent polish. Combines multiple optimization paradigms
for robust global search.
"""

import numpy as np
from scipy.signal import correlate,.fftconvolve
from scipy.optimize import linprog
import time


def _convolve_full(f):
    """Full linear convolution using FFT."""
    return fftconvolve(f, f, mode='full')


def _soft_gradient(f, c, T):
    """Soft-max gradient: weighted sum of peak gradient vectors."""
    n = len(f)
    cs = c / T
    cs -= cs.max()
    p = np.exp(cs)
    s = p.sum()
    if s < 1e-300:
        k_star = int(np.argmax(c))
        grad = np.zeros(n)
        for i in range(n):
            idx = k_star - i
            if 0 <= idx < n:
                grad[i] = 2.0 * f[idx]
        return grad
    p /= s
    return 2.0 * correlate(p, f, mode='valid')


def _project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    s = v.sum()
    if s < 1e-300:
        return np.ones(n) / n
    v = v / s
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.max(np.nonzero(cond)[0]))
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    out = np.maximum(v - theta, 0.0)
    out_sum = out.sum()
    if out_sum < 1e-300:
        return np.ones(n) / n
    return out / out_sum


def _eval(f):
    """Evaluate the sequence using the provided evaluate_sequence."""
    return evaluate_sequence(f.tolist())


def _lp_direction(f, M, K=50, max_time=5.0):
    """Solve reduced LP: max sum(g) s.t. g>=0, (f*g)[k]<=M for top K peaks.
    Returns g normalized to sum 1, or None."""
    n = len(f)
    c = _convolve_full(f)
    top_k = np.argpartition(c, -K)[-K:]
    
    A = np.zeros((K, n))
    for idx, k in enumerate(top_k):
        k = int(k)
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_min <= i_max:
            A[idx, i_min:i_max+1] = f[k - i_max:k - i_min + 1][::-1]
    
    b = np.full(K, M * 1.0001)
    
    start_time = time.monotonic()
    try:
        res = linprog(-np.ones(n), A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': max_time})
    except Exception:
        return None
    
    if res.status != 0:
        return None
    
    g = res.x
    s = g.sum()
    if s < 1e-10:
        return None
    return g / s


def _mirror_step(f, grad, eta):
    """Exponentiated gradient (mirror descent) step on simplex."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy()
    return f_new / s


def _line_search_blend(f, g, M, n):
    """Line search along blend (1-t)*f + t*g to minimize peak."""
    best_t = 0.0
    best_peak = M
    best_f = f.copy()
    for t in [0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 0.9, 1.0]:
        f_try = _project_simplex((1 - t) * f + t * g)
        c_try = _convolve_full(f_try)
        peak_try = c_try.max()
        if peak_try < best_peak:
            best_peak = peak_try
            best_t = t
            best_f = f_try
    return best_f, best_peak, best_t


def _optimize_lp(f, n, deadline, rng, verbose=False, label=""):
    """Main LP-based optimization with line search and SA."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = _eval(f)
    
    c = _convolve_full(f)
    M = c.max()
    
    it = 0
    stuck = 0
    T_sa = 1.0
    no_lp_count = 0
    
    while time.monotonic() < deadline:
        it += 1
        elapsed_frac = 0.0
        total_budget = deadline - (deadline - deadline)  # placeholder
        
        # Try LP direction
        g = _lp_direction(f, M, K=50, max_time=3.0)
        
        if g is not None:
            no_lp_count = 0
            f_try, peak_try, t_used = _line_search_blend(f, g, M, n)
            
            if peak_try < M * (1 - 1e-10):
                f = f_try
                M = peak_try
                stuck = 0
            else:
                # SA: accept worse with probability
                if rng.random() < np.exp(-(peak_try - M) / (T_sa * M + 1e-10)):
                    f = f_try
                    M = peak_try
                else:
                    stuck += 1

            # Fallback: gradient step
            c = _convolve_full(f)

            T = M * 0.05

            c = _convolve_full(f)

            stuck = 0

    return best_f, best_score


def _gradient_anneal(f, n, deadline, rng, verbose=False, label=""):
    """Mirror descent with soft-max gradient, Fista momentum, and SA."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = _eval(f)
    
    c = _convolve_full(f)
    peak0 = c.max()
    T_init = peak0 * 0.1
    T_min = peak0 * 1e-10
    
    n_iters = 200000
    t_prev = f.copy()
    t_seq = 1.0
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        progress = it / n_iters
        T = T_init * (T_min / T_init) ** (progress ** 0.5)
        step = 0.6 * (0.01 / 0.6) ** (progress ** 0.6)
        
        c = _convolve_full(f)
        grad = _soft_gradient(f, c, T)
        
        eta = step / (np.std(grad) + 1e-10)
        f_new = _mirror_step(f, grad, eta)
        
        # Fista momentum
        t_curr = (1.0 + np.sqrt(1.0 + 4.0 * t_seq * t_seq)) / 2.0
        beta = (t_seq - 1.0) / t_curr
        y = f_new + beta * (f_new - t_prev)
        y = np.maximum(y, 0.0)
        y = _project_simplex(y)
        
        # Accept if better
        c_y = _convolve_full(y)
        c_new = _convolve_full(f_new)
        if c_y.max() < c.max():
            f = y
        elif c_new.max() < c.max():
            f = f_new
        elif it > n_iters * 0.6 and rng.random() < 0.05:
            f = f_new  # SA exploration
        
        t_prev = f.copy()
        t_seq = t_curr
        
        if it % 10000 == 0:
            score = _eval(f)
            if score < best_score:
                best_score = score
                best_f = f.copy()
                if verbose:
                    print(f"    {label} it{it}: {score:.6f}")
    
    score = _eval(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score


def _polish(f, n, deadline, rng):
    """Coordinate descent polish with targeted pair moves."""
    f = f.copy().astype(np.float64)
    best_f = f.copy()
    best_score = _eval(f)
    c = _convolve_full(f)
    peak = c.max()
    
    passes = 0
    while time.monotonic() < deadline:
        passes += 1
        improved = False
        k_star = int(np.argmax(c))
        
        # Compute per-coordinate contribution to peak
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = 2.0 * f[i] * f[j]
        
        # Try targeted moves: reduce top contributors, increase low ones
        top_idx = np.argsort(contrib)[::-1]
        bot_idx = np.argsort(contrib)[:n // 2]
        
        for _ in range(500):
            if time.monotonic() > deadline:
                break
            i = int(top_idx[rng.integers(0, min(50, n))])
            j = int(bot_idx[rng.integers(0, len(bot_idx))])
            if i == j:
                continue
            for frac in [0.01, 0.05, 0.1, 0.2]:
                amt = f[i] * frac
                if amt < 1e-15:
                    break
                f_try = f.copy()
                f_try[i] -= amt
                f_try[j] += amt
                if f_try[i] < -1e-15:
                    continue
                f_try = _project_simplex(f_try)
                c_try = _convolve_full(f_try)
                if c_try.max() < peak * (1 - 1e-10):
                    f = f_try
                    peak = c_try.max()
                    c = c_try
                    improved = True
                    break
        
        # Also try random swaps
        for _ in range(200):
            if time.monotonic() > deadline:
                break
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i == j:
                continue
            amt = 0.02 * max(f[i], f[j])
            for sign in [1, -1]:
                f_try = f.copy()
                f_try[i] -= sign * amt
                f_try[j] += sign * amt
                if f_try[i] < -1e-15 or f_try[j] < -1e-15:
                    continue
                f_try = _project_simplex(f_try)
                c_try = _convolve_full(f_try)
                if c_try.max() < peak * (1 - 1e-10):
                    f = f_try
                    peak = c_try.max()
                    c = c_try
                    improved = True
                    break
        
        if not improved:
            # Perturb to escape
            f = f + 0.003 * rng.standard_normal(n)
            f = np.maximum(f, 0.0)
            f = _project_simplex(f)
            c = _convolve_full(f)
            peak = c.max()
        
        if passes % 10 == 0:
            score = _eval(f)
            if score < best_score:
                best_score = score
                best_f = f.copy()
    
    score = _eval(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score
