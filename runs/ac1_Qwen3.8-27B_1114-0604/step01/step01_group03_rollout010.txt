"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.
Uses FFT-based soft-max gradient descent with Nesterov momentum,
adaptive temperature scheduling, diverse initializations, and a
multiplicative-weights saddle-point method. Includes a final local
exploit phase on the active (near-maximal) lags.
"""

import numpy as np
import time
import math


def _norm(a):
    """Normalize array to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a), dtype=np.float64) / len(a)
    return a / s


def _next_pow2(m):
    return 1 << (m - 1).bit_length() if m > 1 else 2


def _self_conv(a, L):
    """Linear self-convolution via FFT."""
    n = len(a)
    m = 2 * n - 1
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:m]
    return np.maximum(c, 0.0, out=c)


def _score_from_conv(c, n):
    """Compute objective from convolution array."""
    return 2.0 * n * float(np.max(c))


def _full_score(a, L):
    """Compute objective from sequence."""
    n = len(a)
    c = _self_conv(a, L)
    s = a.sum()
    if s < 1e-300:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (s * s)


def _soft_max_and_grad(a, L, temp):
    """Compute soft-max objective, weights, and gradient."""
    n = len(a)
    m = 2 * n - 1
    c = _self_conv(a, L)
    v = n * c
    vmax = float(np.max(v))
    z = (v - vmax) / temp
    z = np.clip(z, -100.0, 0.0)
    e = np.exp(z)
    se = float(e.sum())
    if se < 1e-300:
        e = np.ones(m)
        se = float(m)
    w = e / se
    J = vmax + temp * math.log(se)
    
    # Gradient: dJ/da_i = 2n * sum_k w[k] * a[k-i]
    # = 2n * (cross-correlation of w and a at lag i)
    # Using the trick: (w * a_rev)[n-1+i] = sum_j w[j] * a[j-i]
    a_rev = a[::-1]
    W = np.fft.rfft(w, L)
    Frev = np.fft.rfft(a_rev, L)
    g = np.fft.irfft(W * Frev, L)[(n-1):m]
    np.maximum(g, 0.0, out=g)
    g *= 2.0 * n
    return J, w, g


def _init_uniform(n):
    return np.ones(n, dtype=np.float64) / n


def _init_ushape(n, alpha):
    """U-shaped: high at ends, low in middle."""
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0
    a = np.exp(alpha * x * x)
    return a / a.sum()


def _init_vshape(n, alpha):
    """V-shaped: low at ends, high in middle."""
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0
    a = 1.0 + alpha * np.abs(x)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _init_poly(n, coeffs):
    """Polynomial shape: a_j ∝ p(j/(n-1)) where p is polynomial."""
    t = np.linspace(0.0, 1.0, n)
    a = np.zeros(n)
    for power, c in coeffs:
        a += c * (t ** power)
    a = np.maximum(a, 1e-10)
    return a / a.sum()


def _init_spectral(n, rng, k):
    """Initialize using a random spectral (Fourier) structure."""
    t = np.linspace(0.0, 1.0, n)
    z = np.zeros(n)
    for _ in range(k):
        freq = rng.uniform(0.5, 5.0)
        amp = rng.uniform(0.3, 1.5)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos(2.0 * np.pi * freq * t + ph)
    a = np.exp(z)
    return a / a.sum()


def _init_random_smooth(n, rng, sigma=0.5):
    """Random smooth initialization."""
    z = rng.normal(0.0, sigma, n)
    # Smooth with a box filter
    if n > 5:
        k = np.ones(5) / 5.0
        z = np.convolve(z, k, mode='same')
    a = np.exp(z)
    return a / a.sum()

def _soft_max_descent(a, L, time_limit, rng, temp_init=2.0, temp_final=50.0):
    """
    Soft-max gradient descent with momentum.
    Minimizes the soft approximation to max_k (a*a)_k.
    """
    n = len(a)
    a = _norm(a.copy())
    current_conv = _self_conv(a, L)
    current_score = 2.0 * n * float(np.max(current_conv))
    
    best_a = a.copy()
    best_score = current_score
    
    temp = temp_init
    momentum = 0.7  # Nesterov momentum coefficient
    v = np.zeros(n, dtype=np.float64)  # velocity in log-space
    eta = 0.5  # step size
    no_improve = 0
    
    deadline = time.monotonic() + time_limit
    total_iters = 0
    n_iters_target = max(500, int(time_limit * 50))  # rough estimate
    
    while time.monotonic() < deadline:
        c = _self_conv(a, L)
        sc = 2.0 * n * float(np.max(c))
        
        if sc < best_score - 1e-15:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Compute soft max and gradient
        v_conv = n * c
        vmax = float(np.max(v_conv))
        temp_eff = max(temp, 1e-6)
        z = (v_conv - vmax) / temp_eff
        z = np.clip(z, -100.0, 0.0)
        e = np.exp(z)
        se = float(e.sum())
        if se < 1e-300:
            break
        w = e / se
        J = vmax + temp_eff * math.log(se)
        
        # Gradient
        a_rev = a[::-1]
        W = np.fft.rfft(w, L)
        Frev = np.fft.rfft(a_rev, L)
        g = np.fft.irfft(W * Frev, L)[(n-1):(2*n-1)]
        np.maximum(g, 0.0, out=g)
        g *= 2.0 * n
        
        # Normalize gradient for step size control
        gnorm = float(np.max(g))
        if gnorm < 1e-300:
            break
        d = g / gnorm
        
        # Nesterov momentum
        v = momentum * v + (1.0 - momentum) * d
        
        # Line search
        accepted = False
        for step in [eta, 0.5 * eta, 0.25 * eta, 0.125 * eta]:
            a_new = a * np.exp(-step * v)
            a_new = _norm(a_new)
            c_new = _self_conv(a_new, L)
            sc_new = 2.0 * n * float(np.max(c_new))
            
            # Also check soft max objective
            v_new = n * c_new
            vmax_new = float(np.max(v_new))
            z_new = (v_new - vmax_new) / temp_eff
            z_new = np.clip(z_new, -100.0, 0.0)
            e_new = np.exp(z_new)
            se_new = float(e_new.sum())
            J_new = vmax_new + temp_eff * math.log(se_new) if se_new > 0 else np.inf
            
            if sc_new < sc - 1e-15 or J_new < J - 1e-10:
                a = a_new
                accepted = True
                break
        
        if accepted:
            no_improve = 0
            eta = min(2.0, eta * 1.05)
        else:
            eta = max(0.05, eta * 0.7)
            no_improve += 1
        
        # Temperature schedule: gradually increase (broader exploration)
        frac = min(1.0, total_iters / max(1, n_iters_target))
        temp = temp_init + (temp_final - temp_init) * frac
        # But also cool down if no improvement for a while
        if no_improve > 200:
            temp = max(temp_init, temp * 0.5)
            v *= 0.5  # dampen momentum
        
        # Random restart if very stuck
        if no_improve > 500:
            prior = _init_random_smooth(n, rng, sigma=1.0)
            mix = 0.3
            a = _norm((1.0 - mix) * best_a + mix * prior)
            v = np.zeros(n, dtype=np.float64)
            eta = 0.5
            no_improve = 0
            temp = temp_init
        
        total_iters += 1
    
    return best_a, best_score


def _mws_direction(f, L, n, prior, inner_iters=20, eta_start=2.0, eta_end=50.0):
    """
    Multiplicative weights saddle point method.
    Solves the minimax problem approximately.
    """
    m = 2 * n - 1
    f = _norm(f)
    prior = _norm(prior)
    
    F = np.fft.rfft(f, L)
    Frev = np.fft.rfft(f[::-1], L)
    logprior = np.log(np.maximum(prior, 1e-300))
    
    x = prior.copy()
    y = np.full(m, 1.0 / m)
    avg = np.zeros(n)
    
    for it in range(inner_iters):
        frac = it / max(1, inner_iters - 1)
        eta = eta_start + (eta_end - eta_start) * frac
        
        # Compute u = f * x (convolution)
        X = np.fft.rfft(x, L)
        u = np.fft.irfft(F * X, L)[:m]
        np.maximum(u, 0.0, out=u)
        
        # Update y (player 2: chooses which lag to focus on)
        vu = n * u
        vma = float(np.max(vu))
        z = eta * (vu - vma)
        z = np.clip(z, -50.0, 0.0)
        y = np.exp(z)
        ys = float(y.sum())
        if ys < 1e-300:
            y.fill(1.0 / m)
        else:
            y /= ys
        
        avg += x
        
        # Update x (player 1: chooses the sequence)
        Y = np.fft.rfft(y, L)
        s = np.fft.irfft(Y * Frev, L)[(n-1):m]
        np.maximum(s, 0.0, out=s)
        
        ps = n * s
        sma = float(np.max(ps))
        zx = -eta * (ps - sma) + 0.5 * logprior
        zx -= float(np.max(zx))
        zx = np.clip(zx, -50.0, 0.0)
        x = np.exp(zx)
        xs = float(x.sum())
        if xs < 1e-300:
            x = prior.copy()
        else:
            x /= xs
    
    return _norm(0.5 * avg / inner_iters + 0.5 * x)


def _mws_search(a, L, n, rng, time_limit):
    """Main MWS search loop."""
    deadline = time.monotonic() + time_limit
    best_a = _norm(a)
    best_score = _full_score(best_a, L)
    
    inner = 20
    if time_limit > 200:
        inner = 30
    
    while time.monotonic() < deadline:
        prior = _init_random_smooth(n, rng, sigma=0.8)
        x = _mws_direction(best_a, L, n, prior, inner, 2.0, 50.0)
        
        # Line search: mix best_a with x
        for lam in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            h = _norm((1.0 - lam) * best_a + lam * x)
            sc = _full_score(h, L)
            if sc < best_score:
                best_score = sc
                best_a = h
                break
    
    return best_a, best_score


def _local_exploit(a, L, n, time_limit, top_k=5):

    best_score = 2.0 * n * float(np.max(c))

    eta = 0.3

            best_score = sc

        if no_improve > 200:
            break
        
        # Focus on top-k lags
        m = 2 * n - 1
        v = n * c
        top_idx = np.argsort(v)[-top_k:]
        # Weight: emphasize the maximum
