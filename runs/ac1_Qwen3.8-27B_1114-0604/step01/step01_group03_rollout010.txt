def _normalize(a, eps=1e-300):
    """Normalize to probability vector (non-negative, sum=1)."""
    a = np.asarray(a, dtype=np.float64).copy()
    a[~np.isfinite(a)] = 0.0
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a.fill(1.0)
        s = float(a.sum())
    a /= s
    return a


def _fft_self_conv(a, L):
    """Compute a*a (linear convolution) via FFT, return first 2n-1 elements."""
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)
    np.maximum(c, 0.0, out=c)
    return c


def _score_from_conv(c, n):
    """Compute score from convolution array."""
    return 2.0 * n * float(np.max(c))


def _cross_correlate(y, a, L2):
    """
    Compute g_j = sum_m a_m * y_{j+m} for j = 0..n-1.
    Uses the identity g_j = (a * y^R)_{(2n-2-j)} where y^R is y reversed.
    """

    m = len(y)  # should be 2n-1
    yR = y[::-1]
    fa = np.fft.rfft(a, L2)
    fyR = np.fft.rfft(yR, L2)
    conv = np.fft.irfft(fa * fyR, L2)[:n + m - 1]

def _golden_section_minimize(func, a=0.0, b=1.0, tol=1e-10, max_iter=80):
    """Golden section search for minimization of a univariate function."""
    gr = (math.sqrt(5.0) - 1.0) / 2.0  # golden ratio ~ 0.618
    
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = func(c)
    fd = func(d)
    
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = func(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = func(d)
    
    x_mid = (a + b) / 2.0
    f_mid = func(x_mid)
    
    # Return the best of the evaluated points
    candidates = [(c, fc), (d, fd), (x_mid, f_mid), (a, func(a)), (b, func(b))]
    best = min(candidates, key=lambda x: x[1])
    return best[0], best[1]

        c = _normalize(c)

    t_best, f_best = _golden_section_minimize(objective, 0.0, 1.0, tol=1e-12, max_iter=60)
    return (1.0 - t_best) * a + t_best * b, f_best, t_best


def _random_prior(n, rng):
    """Generate a structured random probability vector."""
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0
    z = np.zeros(n)
    
    # Polynomial components
    z += rng.normal(0.0, 0.3)
    z += rng.normal(0.0, 0.4) * x
    z += rng.normal(0.0, 0.3) * (x*x - 1.0/3.0)
    
    # Sinusoidal components
    for k in range(1, 8):
        amp = rng.uniform(0.05, 0.4) / math.sqrt(k)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos(2.0 * np.pi * k * t + ph)
    
    # Smooth noise
    noise = rng.normal(0.0, 0.08, n)
    if n > 9:
        kernel = np.ones(9) / 9.0
        noise = np.convolve(noise, kernel, mode='same')
    z += noise
    
    p = np.exp(z - np.max(z))
    p = np.maximum(p, 1e-300)
    return _normalize(p)
