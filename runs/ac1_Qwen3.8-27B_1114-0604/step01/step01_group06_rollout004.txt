```python
def project_simplex(y):
    """Project y onto the probability simplex {a >= 0, sum(a) = 1}."""
    n = len(y)
    u = np.sort(y)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(u * np.arange(1, n+1) - (cssv - 1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(y - theta, 0)
```

```python
import numpy as np
import time

# Global state
_N = 0
_NFFT = 0

def _init(n):
    global _N, _NFFT
    _N = n
    _NFFT = 1 << (3 * n - 2).bit_length()

def _conv(a):
    """Compute linear convolution a * a using FFT."""
    fa = np.fft.rfft(a, _NFFT)
    c = np.fft.irfft(fa * fa, _NFFT)[:2*_N-1]
    return np.maximum(c, 0.0)

def _score(a):
    """Compute the score (assuming sum(a) = 1)."""
    c = _conv(a)
    return 2.0 * _N * float(np.max(c))

def _project_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1)
    cond = u * idx - (cssv - 1) > 0
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _soft_grad(a, tau=1e-4):
    """Compute the smooth gradient of max_k (a*a)[k] using soft-max."""
    c = _conv(a)
    m = float(np.max(c))
    diff = np.clip((c - m) / tau, -700.0, 0.0)
    e = np.exp(diff)
    se = float(np.sum(e))
    if se <= 0 or not np.isfinite(se):
        return np.zeros(_N), m
    p = e / se
    
    # Gradient: grad[i] = sum_k p[k] * a[k-i] (with a[x]=0 outside [0,n-1])
    # This is the cross-correlation of p and a.
    fp = np.fft.rfft(p, _NFFT)
    fa = np.fft.rfft(a, _NFFT)
    grad = np.fft.irfft(np.conjugate(fa) * fp, _NFFT)[:_N]
    return grad, m

def _fista(a0, deadline, tau=1e-4, L=2.0, print_interval=1000):
    """FISTA optimization on the simplex."""
    a = a0.copy()
    z = a0.copy()
    t = 1.0
    best_a = a0.copy()
    best_sc = _score(a0)
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        # FISTA extrapolation
        if t > 1:
            y = z + ((t - 1.0) / (t + 1.0)) * (z - a)
        else:
            y = z.copy()
        
        # Ensure y is on or near the simplex
        y = np.maximum(y, 0)
        sy = float(np.sum(y))
        if sy > 0:
            y = y / sy
        else:
            y = np.ones(_N) / _N
        
        # Compute gradient at y
        grad, m = _soft_grad(y, tau)
        
        # Gradient step
        step = 1.0 / L
        a_new = y - step * grad
        
        # Project onto simplex
        a_new = _project_simplex(a_new)
        
        # Store
        a = a_new
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        
        # Check
        sc = _score(a)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        z = a
        t = t_new
        it += 1
        
        if it % print_interval == 0:
            print(f"FISTA: it={it}, score={_score(a):.10f}, best={best_sc:.10f}, "
                  f"tau={tau:.1e}, L={L:.1f}, no_improve={no_improve}")
            if no_improve > 5000:
                break
    
    return best_a, best_sc
```
