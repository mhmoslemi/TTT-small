    """Full linear autocorrelation via FFT."""
    n = len(f)
    size = 1 << (2 * n - 2).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]

def _soft_gradient(c, f, T):
    """Compute soft-max gradient over top peaks."""
    n = len(f)
    cs = c / T
    cs -= cs.max()
    p = np.exp(cs)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros(len(c))
        p[n - 1] = 1.0
    else:
        p /= s
    grad = 2.0 * correlate(p, f, mode='valid')
    return grad, p

def _top_k_gradient(c, f, K):
    """Gradient using only top K peaks (hard-max variant)."""
    n = len(f)
    # Find top K indices
    top_k = np.argsort(c)[::-1][:K]

    grad = 2.0 * correlate(p, f, mode='valid')

def _mirror_step(f, grad, step):
    """One exponentiated-gradient (mirror descent) step."""
    span = grad.max() - grad.min()
    if span < 1e-15:
        return f
    eta = step / span

    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()

    f_new /= s
    return f_new
