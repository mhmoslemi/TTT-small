    """
    Generate a perturbation that specifically attacks the current peak.
    Moves mass away from positions contributing to the peak lag.
    """
    peak_idx = int(np.argmax(c))
    peak_val = c[peak_idx]
    
    # Compute contribution of each pair to the peak
    contrib = np.zeros(n)
    lo = max(0, peak_idx - (n - 1))
    hi = min(n - 1, peak_idx)
    for j in range(lo, hi + 1):
        idx = peak_idx - j
        contrib[j] = f[j] * f[idx]

    f_new = f.copy()
    moved = 0.0

        # Find a good receiver (doesn't contribute to peak)

            contributors.add(k)

        if non_contrib_receivers:

                portion = take / len(picks)
                f_new[j] -= portion

        contrib = np.zeros(n)
        lo = max(0, peak_idx - (n - 1))
        hi = min(n - 1, peak_idx)
        for j in range(lo, hi + 1):
            idx = peak_idx - j
            contrib[j] = f[j] * f[idx]
        
        # Find top contributors (to reduce) and non-contributors (to increase)
        contributors = list(range(lo, hi + 1))
        non_contributors = [k for k in range(n) if k not in set(contributors)]
        
        if not non_contributors:
            non_contributors = list(range(n))
        
        # Sort contributors by contribution (descending)
        contributors_sorted = sorted(contributors, key=lambda x: -contrib[x])
        
        # Try transfers
        improved = False
        tried = 0
        max_tries = 5
        
        for j in contributors_sorted[:max_tries]:
            if tried >= max_tries:
                break
            for k in non_contributors[:3]:
                if j == k:
                    continue
                for frac in [0.05, 0.1, 0.2, 0.3]:
                    tried += 1
                    f_try = f.copy()
                    amt = f_try[j] * frac
                    if amt < 1e-15:
                        continue
                    f_try[j] -= amt
                    f_try[k] += amt
                    c_try = _autocorr_fft(f_try)
                    if c_try.max() < peak_val - 1e-12:
                        f = f_try
                        peak_val = c_try.max()
                        c = c_try
                        improved = True
                        score = _eval(f)
                        if score < local_best:
                            local_best = score
                            local_best_f = f.copy()
                        break
                if improved:
                    break
            if improved:
                break
        
        # Also try random coordinate moves
        for _ in range(20):
            i = int(rng.integers(0, n))
            jx = int(rng.integers(0, n))
            if i == jx:
                continue
            f_try = f.copy()
            amt = f_try[i] * 0.05
            f_try[i] -= amt
            f_try[jx] += amt
            c_try = _autocorr_fft(f_try)
            if c_try.max() < peak_val - 1e-12:
                f = f_try
                peak_val = c_try.max()
                c = c_try
                score = _eval(f)
                if score < local_best:
                    local_best = score
                    local_best_f = f.copy()
    
    score = _eval(f)
    if score < local_best:
        local_best = score
        local_best_f = f.copy()
    
    return local_best_f, local_best


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the non-negative sequence minimizing 2*n*max(conv(f,f))/(sum f)^2.
    
    Strategy:
    1. Multiple diverse starting points.
    2. Phase 1: Simulated annealing with peak-attacking perturbations.
    3. Phase 2: Gradient-based mirror descent (exact + softmax hybrid).
    4. Phase 3: Greedy coordinate descent polish.
    5. Iterative refinement cycles.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0
    
    # Get initial sequence
    f_init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(f_init)
    
    # Generate diverse starting points
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    
    starts = {}
    starts['given'] = f_init.copy()
    starts['uniform'] = np.ones(n)
    starts['u_shape'] = 1.0 + 3.0 * ((j - center) / max(center, 1)) ** 2
    starts['bimodal'] = (np.exp(-(j - 0.3 * n) ** 2 / (0.05 * n * n)) +
                         np.exp(-(j - 0.7 * n) ** 2 / (0.05 * n * n)) + 0.01)
    starts['cosine_taper'] = 0.5 + 0.5 * np.cos(np.pi * (j - center) / center)
    starts['random_exp'] = rng.exponential(1.0, size=n)
    starts['random_uniform'] = rng.uniform(0.1, 1.0, size=n)
    
    # Normalize and evaluate starts
    start_results = []
    for name, f0 in starts.items():
        f = _normalize(f0)
        sc = _eval(f)
        start_results.append((sc, name, f))
    
    start_results.sort(key=lambda x: x[0])
    
    print("Starting point scores:")
    for sc, name, _ in start_results:
        print(f"  {name}: {sc:.6f}")
    
    best_score = start_results[0][0]
    best_f = start_results[0][2].copy()
    
    # === Phase 1: Annealing search on top starts ===
    # Give more time to better starts
    phase1_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.45
    top_starts = start_results[:4]
    
    time_per_start = (phase1_deadline - time.monotonic()) / len(top_starts)
    
    for idx, (sc, name, f0) in enumerate(top_starts):
        if time.monotonic() > phase1_deadline:
            break
        
        t_alloc = time_per_start * (1.0 - 0.15 * idx)
        s_deadline = time.monotonic() + t_alloc
        print(f"\nPhase 1: Annealing from '{name}' (budget {t_alloc:.0f}s)")
        
        f_opt, sc_opt = _phase_anneal_search(f0, n, s_deadline, rng, label=name)
        
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  New best: {best_score:.6f}")
    
    # === Phase 2: Gradient optimization on best solution ===
    phase2_deadline = time.monotonic() + (deadline - time.monotonic()) * 0.35
    print(f"\nPhase 2: Gradient optimization (deadline in {phase2_deadline - time.monotonic():.0f}s)")
    
    f_opt, sc_opt = _phase_gradient_optimize(best_f, n, phase2_deadline, rng, label="main_grad")
    if sc_opt < best_score:
        best_score = sc_opt
        best_f = f_opt.copy()
        print(f"  New best: {best_score:.6f}")
    
    # === Phase 3: Polish ===
    polish_deadline = min(time.monotonic() + 300.0, deadline)
    print(f"\nPhase 3: Coordinate polish (deadline in {polish_deadline - time.monotonic():.0f}s)")
    
    f_opt, sc_opt = _phase_polish(best_f, n, polish_deadline, rng, label="polish")
    if sc_opt < best_score:
        best_score = sc_opt
        best_f = f_opt.copy()
        print(f"  New best: {best_score:.6f}")
    
    # === Phase 4: Iterate refinement if time remains ===
    if time.monotonic() < deadline - 60:
        remaining = deadline - time.monotonic()
        print(f"\nPhase 4: Iterative refinement ({remaining:.0f}s remaining)")
        
        for cycle in range(5):
            if time.monotonic() > deadline - 30:
                break
            
            # Alternate between annealing and gradient
            t_alloc = (deadline - time.monotonic()) * 0.3
            c_deadline = time.monotonic() + t_alloc
            
            if cycle % 2 == 0:
                # Perturb best and do short anneal
                f_perturbed = _random_perturb(best_f, n, rng, 0.005)
                f_opt, sc_opt = _phase_anneal_search(f_perturbed, n, c_deadline, rng, label=f"refine_a{cycle}")
            else:
                f_opt, sc_opt = _phase_gradient_optimize(best_f, n, c_deadline, rng, label=f"refine_g{cycle}")
            
            if sc_opt < best_score:
                best_score = sc_opt
                best_f = f_opt.copy()
                print(f"  Cycle {cycle}: new best {best_score:.6f}")
            
            # Quick polish
            if time.monotonic() < deadline - 20:
                p_deadline = min(time.monotonic() + 60.0, deadline - 10)
                f_opt, sc_opt = _phase_polish(best_f, n, p_deadline, rng, label=f"polish{cycle}")
                if sc_opt < best_score:
                    best_score = sc_opt
                    best_f = f_opt.copy()
                    print(f"  Cycle {cycle} polish: {best_score:.6f}")
    
    # Final evaluation
    best_f = np.clip(best_f, 0.0, 1000.0)
    # Ensure sum is reasonable (not too small)
    S = best_f.sum()
    if S < 1.0:
        best_f = best_f * (10.0 / max(S, 1e-10))
    
    final_score = _eval(best_f)
    if final_score < best_score or not np.isfinite(best_score):
        best_score = final_score
    
    print(f"\n=== FINAL SCORE: {best_score:.6f} ===")
    return [float(x) for x in best_f]
