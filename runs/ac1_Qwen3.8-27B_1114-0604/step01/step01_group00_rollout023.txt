```python
def build_A(g, n):

    # A[k, j] = g[k-j] if 0 <= k-j < n else 0
    from scipy.sparse import csr_matrix
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            idx = k - j
            if 0 <= idx < n:
                rows.append(k)
                cols.append(j)
                vals.append(g[idx])
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    return A
```

```python
"""
Step function optimizer minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
Uses multi-peak mirror descent with exact subgradients, adaptive restarts,
diverse U-shaped initializations, and targeted coordinate descent polish.
Key improvements over prior approach: exact peak-targeting (not soft-max),
multi-peak gradient for stability, backtracking step size, and population
diversity via U-shape parameter sweep.
"""

import numpy as np
import time


def _autocorr(f):
    """Full linear self-convolution via FFT."""
    n = len(f)
    size = 1 << (2 * n - 2).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _eval_seq(f):
    """Evaluate the objective for a numpy array."""
    return evaluate_sequence(f.tolist())


def _peak_partner(f, k_star):
    """
    Compute partner array for peak lag k_star.
    partner[j] = f[k_star - j] if 0 <= k_star - j < n, else 0.
    """
    n = len(f)
    partner = np.zeros(n)
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    if j_start <= j_end:
        j_idx = np.arange(j_start, j_end + 1)
        f_idx = k_star - j_idx
        partner[j_idx] = f[f_idx]
    return partner


def _multi_peak_grad(f, c, top_k=3):
    """Compute gradient as weighted sum of partner arrays for top_k peaks."""
    n = len(f)
    sorted_idx = np.argsort(c)[::-1][:top_k]
    vals = c[sorted_idx]
    max_val = vals[0]
    # Weight: exponential decay from max
    wts = np.exp(4.0 * (vals - max_val) / max(max_val, 1e-10))
    wts /= wts.sum()
    grad = np.zeros(n)
    for i, pos in enumerate(sorted_idx):
        grad += wts[i] * _peak_partner(f, int(pos))
    return grad


def _mirror_step(f, grad, eta):
    """Exponentiated gradient (mirror descent) step on the simplex."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    return f_new / s


def _normalize(f):
    """Ensure f >= 0 and sum(f) = 1."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s


def _optimize(f_init, n, deadline, rng, top_k=3, verbose=True):
    """
    Peak-chasing mirror descent with adaptive step size and restarts.
    """
    f = _normalize(f_init.astype(np.float64).copy())
    local_best = _eval_seq(f)
    local_best_f = f.copy()

    max_iters = 80000
    stuck = 0
    eval_every = 200
    best_eta = 1.0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        progress = it / max_iters
        # Step size: large early, small late
        eta = best_eta * (3.0 / 0.05) ** ((1.0 - progress) ** 0.8) * (0.05 / 3.0)
        eta = max(eta, 0.005)

        c = _autocorr(f)
        if top_k <= 1 or progress > 0.8:
            k_star = int(np.argmax(c))
            grad = _peak_partner(f, k_star)
        else:
            grad = _multi_peak_grad(f, c, top_k)

        f_new = _mirror_step(f, grad, eta)
        f_new = _normalize(f_new)

        # Backtracking: accept if peak decreased
        c_new = _autocorr(f_new)
        if c_new.max() <= c.max() * (1.0 + 1e-9):
            f = f_new
            best_eta = min(eta * 1.3, 5.0)
        else:
            # Reject, reduce eta
            best_eta = max(eta * 0.5, 0.001)

        if it % eval_every == 0 or it == max_iters - 1:
            score = _eval_seq(f)
            if score < local_best - 1e-9:
                local_best = score
                local_best_f = f.copy()
                stuck = 0
                if verbose and it % (eval_every * 5) == 0:
                    print(f"    iter {it}: {score:.6f}")
            else:
                stuck += 1

            if stuck >= 4:
                # Restart strategy
                if stuck == 4:
                    # Small perturbation
                    f = f + 0.003 * rng.standard_normal(n)
                elif stuck == 5:
                    # Blend with random U-shape
                    j_arr = np.arange(n, dtype=np.float64)
                    center = (n - 1) / 2.0
                    x = (j_arr - center) / center
                    alpha = rng.uniform(1.0, 12.0)
                    u = 1.0 + alpha * x ** 2
                    f = 0.6 * f + 0.4 * _normalize(u)
                else:
                    # Larger restart
                    j_arr = np.arange(n, dtype=np.float64)
                    center = (n - 1) / 2.0
                    x = (j_arr - center) / center
                    alpha = rng.uniform(0.5, 20.0)
                    u = np.maximum(1.0 + alpha * x ** 2 + 0.05 * rng.standard_normal(n), 0.0)
                    f = 0.4 * f + 0.6 * _normalize(u)
                f = _normalize(f)
                best_eta = 1.0
                stuck = 0

    score = _eval_seq(f)
    if score < local_best:
        local_best = score
        local_best_f = f.copy()

    return local_best_f, local_best


def _polish(f_init, n, deadline, rng):
    """
    Coordinate descent polish: targeted mass transfers to reduce peaks.
    """
    f = _normalize(f_init.astype(np.float64).copy())
    local_best = _eval_seq(f)
    local_best_f = f.copy()
    c = _autocorr(f)
    peak = c.max()

    max_passes = 2000
    no_improve = 0

    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        c = _autocorr(f)
        peak = c.max()
        k_star = int(np.argmax(c))
        partner = _peak_partner(f, k_star)
        contrib = f * partner

        order = np.argsort(contrib)
        moved = False

        # Try targeted transfers from high to low contribution
        n_pairs = min(8, n // 200 + 1)
        for _ in range(n_pairs):
            hi = int(order[rng.integers(max(1, n // 2), n)])
            lo = int(order[rng.integers(0, max(1, n // 2))])
            if hi == lo:
                continue
            for frac in [0.02, 0.05, 0.1, 0.2]:
                f_try = f.copy()
                amt = f_try[hi] * frac
                if amt < 1e-15:
                    continue
                f_try[hi] -= amt
                f_try[lo] += amt
                f_try = _normalize(f_try)
                c_try = _autocorr(f_try)
                if c_try.max() < peak * (1.0 - 1e-7):
                    f = f_try
                    peak = c_try.max()
                    local_best = _eval_seq(f)
                    local_best_f = f.copy()
                    moved = True
                    break
            if moved:
                break

        if not moved:
            no_improve += 1
            if no_improve % 30 == 0:
                # Random exploration
                for _ in range(40):
                    i = int(rng.integers(0, n))
                    j = int(rng.integers(0, n))
                    if i == j:
                        continue
                    f_try = f.copy()
                    amt = f_try[i] * 0.02
                    f_try[i] -= amt
                    f_try[j] += amt
                    if f_try[i] < 0:
                        continue
                    f_try = _normalize(f_try)
                    c_try = _autocorr(f_try)
                    if c_try.max() < peak * (1.0 - 1e-7):
                        f = f_try
                        peak = c_try.max()
                        local_best = _eval_seq(f)
                        local_best_f = f.copy()
                        moved = True
                        break
                if moved:
                    no_improve = 0

        if moved:
            no_improve = 0

    return local_best_f, local_best
```
