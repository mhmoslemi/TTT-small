`(f*f)[k] = (f_0*f_0)[k] + 2*f[i]*(f_0*f)[k]_i + f[i]^2*e_i*e_i[k]`

Where `(f_0*f)[k]_i = f_0[k-i]` (if valid) and `(e_i*e_i)[k] = 1` if k = 2i.

So `(f*f)[k] = c_k + 2*f[i]*f_0[k-i] + f[i]^2*delta_{k,2i}`

This is a quadratic in f[i] for each k, and we want to minimize the max over k. This is a 1D optimization problem!

For a given f_0, we want to find f[i] that minimizes:
`max_k (c_k + 2*f[i]*d_k + f[i]^2*delta_{k,2i})`

where d_k = f_0[k-i] (or 0 if k-i is out of range) and c_k = (f_0*f_0)[k].

This is the minimax of a set of quadratic functions in 1D, which can be solved by bisection or by finding the point where the maximum is minimized (the "bottleneck" point).

This block coordinate descent might converge slowly but is guaranteed to not increase the objective at each step.

def compute_convolution(f):
    """Compute linear convolution f*f using FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2*n-1]

def compute_gradient_at_peak(f, k_star, n):
    """Compute exact subgradient of (f*f)[k_star] w.r.t. f.
    d(f*f)[k]/df[i] = 2*f[k-i] if 0 <= k-i < n, else 0."""
    grad = np.zeros(n)
    # k_star - i must be in [0, n-1]
    # i ranges from max(0, k_star-(n-1)) to min(n-1, k_star)
    i_min = max(0, k_star - (n - 1))
    i_max = min(n - 1, k_star)
    if i_min <= i_max:
        indices = np.arange(i_min, i_max + 1)
        grad[indices] = 2.0 * f[k_star - indices]
    return grad

    else:
        f_new /= s
    return f_new

    n = len(f)

    """Subgradient at lag k: grad[i] = 2*f[k-i] for valid i."""
    grad = np.zeros(n)
    lo = max(0, k - n + 1)

    """Exponentiated gradient step (mirror descent on simplex)."""
    lf = np.log(f + 1e-30) - eta * grad

    fn = np.exp(lf)

Step function optimizer minimizing 2*n*max(f*f)/(sum f)^2.
Uses exponentiated gradient (mirror descent) with exact subgradient at peak lags,
top-k weighted gradient, adaptive step size, annealed restarts, and targeted polish.

import numpy as np
from scipy.signal import correlate
import time


def _conv(f):
    """Linear convolution f*f via FFT. Returns array of length 2n-1."""
    n = len(f)
    size = 1 << (2 * n - 2).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _grad_at_lag(f, k, n):
    """Exact subgradient of (f*f)[k] w.r.t. f. grad[i] = 2*f[k-i] for valid i."""
    grad = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        grad[idx] = 2.0 * f[k - idx]
    return grad


def _topk_gradient(f, conv, n, k_count=5, T_ratio=0.05):
    """Weighted gradient from top-k peak lags with softmax weighting."""
    conv_half = conv[:n]
    T = T_ratio * conv_half.max()
    if T < 1e-15:
        T = 1e-15
    # Find top k lags
    top_idx = np.argsort(conv_half)[::-1][:k_count]
    vals = conv_half[top_idx]
    w = np.exp((vals - vals[0]) / T)
    w /= w.sum()
    grad = np.zeros(n)
    for wi, ki in zip(w, top_idx):
        grad += wi * _grad_at_lag(f, ki, n)
    return grad

    """Exponentiated gradient (mirror descent) step on the simplex."""
    lf = np.log(f + 1e-30) - eta * grad

    if s < 1e-30:

    return fn / s
