Search for non-negative step heights minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses subgradient descent with exact max tracking, log-sum-exp smoothing,
coordinate refinement, and adaptive step sizes. Combines multiple search
strategies with diversity maintenance.

import math
import time
import numpy as np

def _normalize_simplex(a, eps=1e-300):
    """Project onto the probability simplex (non-negative, sum to 1)."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a.fill(1.0)
        s = float(a.sum())
    a /= s
    a = np.maximum(a, eps)
    a /= float(a.sum())
    return a

def _conv_self(a, L, m):
    """Compute self-convolution using FFT."""
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    c = np.ascontiguousarray(c)
    np.maximum(c, 0.0, out=c)
    return c

    """Compute 2*n*max(c)."""
    return float(2.0 * n * np.max(c))

    z = temp * (v - vmax)
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return float(vmax)
    return float(temp * vmax + math.log(s))

def _softmax_weights(v, temp):
    """Compute softmax weights for subgradient averaging."""
    vmax = float(np.max(v))
    z = temp * (v - vmax)
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        n = len(v)
        return np.full(n, 1.0/n)
    return e / s

def _subgradient_direction(a, c, n, temp, L):
    """
    Compute the subgradient direction: weighted average of flipped+shifted a
    for the hot positions of the convolution.
    """
    v = n * c
    w = _softmax_weights(v, temp)
    # For each position k with weight w[k], the gradient contribution is a[k-i]
    # We need to compute: sum_k w[k] * a[k-i] for each i
    # This is: w * a (convolution-like, but with w indexed by k and a by k-i)
    # result[i] = sum_k w[k] * a[k-i] = (w * a_reversed) (correlation)
    # Actually: result[i] = sum_k w[k] * a[k-i]
    # Let j = k-i, so k = j+i: result[i] = sum_j w[j+i] * a[j]
    # This is the cross-correlation of w and a, shifted.
    
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    # We want: for each i, sum_k w[k] * a[k-i]
    # This is the cross-correlation: (w ⋆ a)[i] = sum_k w[k] * a[k-i]
    # In FFT: IFFT(conj(W) * A) gives cross-correlation... 
    # Actually let me think more carefully.
    # If w and a are both length m and n respectively (m = 2n-1, n = n),
    # and we want result[i] = sum_k w[k]*a[k-i] for i = 0,...,n-1
    # where k ranges over valid indices (0 <= k < m, 0 <= k-i < n)
    # 
    # Let's define a_padded of length L with a in positions 0..n-1
    # and w_padded of length L with w in positions 0..m-1
    # Then the circular cross-correlation at shift i is:
    # sum_k w_padded[k] * a_padded[(k-i) mod L]
    # For i in 0..n-1 and our setup, this should give us what we want.
    # 
    # Cross-correlation in FFT: IFFT(conj(FFT(w)) * FFT(a))
    # But we need to be careful about the shift.
    
    # Actually, let's use a different approach:
    # result[i] = sum_k w[k] * a[k-i]
    # = sum_j w[j+i] * a[j]  (substituting j = k-i)
    # This is the cross-correlation of w and a at lag i.
    # cross_corr[i] = sum_j w[j+i] * a[j]
    
    # In FFT terms: cross_corr = IFFT(conj(FFT(w)) * FFT(a))
    # But this gives circular cross-correlation with a specific convention.
    
    # Let me just use the direct FFT correlation:
    # If C = IFFT(conj(W) * A), then C[i] = sum_k w[k] * a[(k-i) mod L]
    # For our case with zero-padding, this should work for i = 0, ..., n-1.
    
    C = np.fft.irfft(np.conj(W) * A, L)
    result = C[:n].copy()
    np.maximum(result, 0.0, out=result)
    return result

def _subgradient_direction2(a, c, n, temp, L):
    """
    Alternative: compute direction as sum_k w[k] * a_shifted[k]
    where a_shifted[k][i] = a[k-i] (the signal shifted by k).
    
    This is equivalent to the correlation of w with a.
    """
    v = n * c
    w = _softmax_weights(v, temp)
    
    # We want: d[i] = sum_k w[k] * a[k-i] for i = 0, ..., n-1
    # (where a[k-i] = 0 if k-i is out of [0, n-1])
    
    # Method: FFT correlation
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    # corr[i] = sum_k w[k] * conj(a[(k-i) mod L])
    # Since a is real: corr[i] = sum_k w[k] * a[(k-i) mod L]
    corr = np.fft.irfft(np.conj(W) * A, L)
    d = corr[:n].copy()
    np.maximum(d, 0.0, out=d)
    return d

def _line_search(a, d, L, m, n, best_score, max_iter=20):
    """
    Line search: find step size that minimizes the score along a - t*d.
    Returns (best_a, best_score)
    """
    best_a = a.copy()
    cur_score = _score(a, _conv_self(a, L, m), n)
    
    # Try decreasing step sizes
    t = 0.5
    for _ in range(max_iter):
        a_new = a - t * d
        a_new = _normalize_simplex(a_new)
        c_new = _conv_self(a_new, L, m)
        sc = _score(a_new, c_new, n)
        if sc < cur_score:
            best_a = a_new.copy()
            cur_score = sc
            t = min(t * 1.5, 2.0)
        else:
            t *= 0.6
        if t < 1e-8:
            break
    
    return best_a, cur_score

def _coordinate_refine(a, L, m, n, rng, n_trials=200):
    """
    Try individual coordinate changes and keep improvements.
    For each trial, pick a random coordinate and try to optimize it.
    """
    a = a.copy()
    c = _conv_self(a, L, m)
    score = _score(a, c, n)
    s = float(a.sum())
    
    for trial in range(n_trials):
        i = int(rng.integers(n))
        # Try a few values for a[i]
        base = a[i]
        best_val = base
        best_sc = score
        
        for factor in (0.5, 0.75, 1.25, 1.5, 0.25, 2.0, 0.1):
            new_val = base * factor
            if new_val < 0:
                continue
            a[i] = new_val
            # Renormalize
            a_norm = _normalize_simplex(a)
            c_new = _conv_self(a_norm, L, m)
            sc = _score(a_norm, c_new, n)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_val = new_val
                a[:] = a_norm
                c = c_new
                score = sc
        
        a[i] = best_val
    
    return a, score

def _random_prior(n, rng):
    """Generate a random prior distribution."""
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0
    z = rng.normal(0.0, 0.4)
    z += rng.normal(0.0, 0.6) * x
    z += rng.normal(0.0, 0.4) * (x*x - 1.0/3.0)
    
    kmax = int(rng.integers(2, 6))
    for k in range(1, kmax+1):
        amp = rng.uniform(0.05, 0.4) / max(1.0, math.sqrt(float(k)))
        ph = rng.uniform(0.0, 2.0*np.pi)
        z += amp * np.cos(2.0*np.pi*k*t + ph)
    
    noise = rng.normal(0.0, 0.2, n)
    if n > 5:
        noise = np.convolve(noise, np.ones(min(5, n), dtype=np.float64)/min(5, n), mode="same")
    z += noise
    
    p = np.exp(z - float(np.max(z)))
    p = np.maximum(p, 1e-300)
    return _normalize_simplex(p)

def _internal_exact(seq):
    """Internal exact evaluation."""
    if not isinstance(seq, list) or not seq:
        return float(np.inf)
    for x in seq:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return float(np.inf)
        if np.isnan(x) or np.isinf(x):
            return float(np.inf)
    a = [0.0 if float(x) < 0.0 else min(1000.0, float(x)) for x in seq]
    n = len(a)
    s = sum(a)
    if s < 0.01:
        return float(np.inf)
    b = np.convolve(a, a)
    mb = max(b)
    return float(2.0 * n * mb / (s * s))

def _subgradient_search(a, L, m, n, rng, time_limit, temp_init=1.0, temp_end=200.0):
    """
    Main subgradient-based search loop.
    """
    start = time.monotonic()
    deadline = start + float(time_limit)
    
    a = _normalize_simplex(a)
    c = _conv_self(a, L, m)
    best_score = _score(a, c, n)
    best_a = a.copy()
    
    temp = float(temp_init)
    step_size = 0.1
    
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        c = _conv_self(a, L, m)
        cur_score = _score(a, c, n)
        
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Periodic restart if stuck
        if no_improve > 200:
            prior = _random_prior(n, rng)
            mix = 0.2 + 0.3 * float(rng.random())
            a = _normalize_simplex((1.0 - mix) * best_a + mix * prior)
            c = _conv_self(a, L, m)
            cur_score = _score(a, c, n)
            if cur_score < best_score:
                best_score = cur_score
                best_a = a.copy()
            no_improve = 0
            temp = float(temp_init)
            step_size = 0.1
            continue
        
        # Compute subgradient direction
        temp_eff = min(temp, 100.0 / max(float(np.max(c)), 1e-12))
        if temp_eff < 0.01:
            temp_eff = 0.01
        
        d = _subgradient_direction2(a, c, n, temp_eff, L)
        dmax = float(np.max(d))
        if dmax < 1e-15:
            # Break symmetry
            a = _random_prior(n, rng)
            continue
        d /= dmax
        
        # Try step
        for t in (step_size, 0.5*step_size, 0.25*step_size):
            a_new = a - t * d
            a_new = _normalize_simplex(a_new)
            c_new = _conv_self(a_new, L, m)
            sc = _score(a_new, c_new, n)
            if sc < cur_score:
                a = a_new
                step_size = min(0.5, step_size * 1.1)
                break
            elif sc < best_score:
                a = a_new
                step_size = min(0.5, step_size * 1.05)
                break
        else:
            step_size = max(0.01, step_size * 0.7)
        
        # Periodic coordinate refinement
        if it % 100 == 50:
            a, sc = _coordinate_refine(a, L, m, n, rng, n_trials=100)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
        
        # Cool down temperature
        if it % 30 == 0:
            temp = min(temp_end, temp * 1.15)
        
        it += 1
    
    return best_a, best_score

def _mws_search(a, L, m, n, rng, time_limit):
    """
    Multiplicative weight subroutine search - finds a direction that
    reduces the max convolution.
    """
    start = time.monotonic()
    deadline = start + float(time_limit)
    
    a = _normalize_simplex(a)
    FA = np.fft.rfft(a, L)
    FA_rev = np.fft.rfft(a[::-1], L)
    
    best_score = _score(a, _conv_self(a, L, m), n)
    best_a = a.copy()
    
    inner = 20
    eta_s = 2.0
    eta_e = 30.0
    
    while time.monotonic() < deadline:
        prior = _random_prior(n, rng)
        
        x = prior.copy()
        y = np.full(m, 1.0/float(m))
        avg = np.zeros(n)
        
        for it in range(inner):
            frac = float(it) / max(1, inner - 1)
            eta = eta_s + (eta_e - eta_s) * frac
            
            # Update y (dual variable for convolution constraints)
            X = np.fft.rfft(x, L)
            u = np.fft.irfft(FA * X, L)[:m]
            np.maximum(u, 0.0, out=u)
            
            vu = n * u
            vma = float(np.max(vu))
            z = eta * (vu - vma)
            z = np.clip(z, -50.0, 0.0)
            y = np.exp(z)
            ys = float(y.sum())
            if ys > 0 and np.isfinite(ys):
                y /= ys
            else:
                y.fill(1.0/float(m))
            
            avg += x
            
            # Update x (primal variable)
            Y = np.fft.rfft(y, L)
            s = np.fft.irfft(Y * FA_rev, L)[(n-1):m]
            np.maximum(s, 0.0, out=s)
            
            ps = n * s
            sma = float(np.max(ps))
            logprior = np.log(np.maximum(prior, 1e-300))
            zx = -eta * (ps - sma) + 0.5 * logprior
            zx -= float(np.max(zx))
            zx = np.clip(zx, -50.0, 0.0)
            x = np.exp(zx)
            xs = float(x.sum())
            if xs > 0 and np.isfinite(xs):
                x /= xs
            else:
                x = prior.copy()
        
        x = _normalize_simplex(0.5 * avg / inner + 0.5 * x)
        
        # Line search along best_a -> x
        for lam in (0.1, 0.2, 0.3, 0.5, 0.7, 1.0):
            cand = _normalize_simplex((1.0-lam)*best_a + lam*x)
            sc = _score(cand, _conv_self(cand, L, m), n)
            if sc < best_score:
                best_score = sc
                best_a = cand
        
    return best_a, best_score

"""
Search for non-negative step heights minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses a combination of:
- Subgradient descent with exact max tracking and Nesterov momentum
- Log-sum-exp smoothed gradient for differentiable approximation
- Adaptive step sizes with backtracking
- Coordinate-wise 1D refinement (golden section)
- Simulated annealing for global exploration
- Monotone-shape initialization (exploiting rearrangement inequality)
- Multiple restarts with diverse initial conditions
- FFT-based convolution for efficiency

import math

import numpy as np


def _normalize_simplex(a, eps=1e-300):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a.fill(1.0)
        s = float(a.sum())
    a /= s
    a = np.maximum(a, eps)
    a /= float(a.sum())
    return a


def _conv_self(a, L, m):
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    c = np.ascontiguousarray(c)
    np.maximum(c, 0.0, out=c)
    return c


def _score_from_conv(c, n):
    return float(2.0 * n * np.max(c))


def _lse_obj(c, n, temp):
    v = n * c
    vmax = float(np.max(v))
    z = temp * (v - vmax)
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return float(vmax)
    return float(temp * vmax + math.log(s))


def _softmax_weights(v, temp):
    vmax = float(np.max(v))
    z = temp * (v - vmax)
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        n = len(v)
        return np.full(n, 1.0 / n)
    return e / s


def _gradient_direction(a, c, n, temp, L):
    """Compute subgradient direction: weighted correlation of softmax weights with a."""
    v = n * c
    w = _softmax_weights(v, temp)
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    corr = np.fft.irfft(np.conj(W) * A, L)
    d = corr[:n].copy()
    np.maximum(d, 0.0, out=d)
    return d


def _exact_subgradient(a, c, n, L):
    """Compute exact subgradient at the argmax position."""
    k = int(np.argmax(c))
    # Gradient of (conv)_k wrt a[i] = 2*a[k-i] (if valid)
    d = np.zeros(n)
    for i in range(n):
        j = k - i
        if 0 <= j < n:
            d[i] = 2.0 * a[j]
    return d


def _smoothed_subgradient(a, c, n, temp, L):
    """Combined exact + smoothed subgradient."""
    d_exact = _exact_subgradient(a, c, n, L)
    d_smooth = _gradient_direction(a, c, n, temp, L)
    d = 0.5 * d_exact + 0.5 * d_smooth
    dmax = float(np.max(d))
    if dmax > 0:
        d /= dmax
    return d

    for k in range(1, kmax + 1):
        amp = rng.uniform(0.05, 0.35) / max(1.0, math.sqrt(float(k)))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos(2.0 * np.pi * k * t + ph)
    noise = rng.normal(0.0, 0.15, n)
    if n > 10:
        win = min(7, n)
        noise = np.convolve(noise, np.ones(win) / win, mode="same")
    z += noise
    p = np.exp(z - float(np.max(z)))
    p = np.maximum(p, 1e-300)
    return _normalize_simplex(p)


def _shape_candidates(n, rng):
    """Generate diverse shape candidates."""
    out = []
    t = np.linspace(0.0, 1.0, n)
    
    # Uniform
    out.append(np.ones(n))
    
    # Linear decreasing/increasing
    out.append(1.0 - t)
    out.append(t)
    
    # Quadratic
    out.append(1.0 - t * t)
    out.append(t * t)
    out.append(1.0 - (2*t - 1)**2)
    
    # Power law
    for p in (0.5, 1.0, 2.0, 3.0, 5.0):
        out.append(t**p)
        out.append((1-t)**p)
    
    # Exponential
    for rate in (1.0, 2.0, 3.0, 5.0):
        out.append(np.exp(-rate * t))
        out.append(np.exp(-rate * (1-t)))
    
    # Sine/cosine
    for k in (1, 2, 3, 5):
        for ph in (0, np.pi/2, np.pi):
            out.append(1.0 + 0.5 * np.sin(2*np.pi*k*t + ph))
    
    # Gaussian-like
    for sigma in (0.15, 0.25, 0.35, 0.5):
        out.append(np.exp(-0.5 * ((t - 0.5) / sigma)**2))
    
    # Scaled and normalized
    result = []
    for arr in out:
        arr = np.asarray(arr, dtype=np.float64)
        arr = np.maximum(arr, 0.0)
        s = arr.sum()
        if s > 0 and np.isfinite(s):
            result.append(_normalize_simplex(arr))
    
    # Add random priors
    for _ in range(4):
        result.append(_random_prior(n, rng))
    
    return result


def _subgradient_momentum_search(a, L, m, n, rng, time_limit, temp_start=1.0, temp_end=200.0, momentum=0.9):
    """Subgradient descent with Nesterov momentum and adaptive step size."""
    start = time.monotonic()
    deadline = start + float(time_limit)
    
    a = _normalize_simplex(a)
    c = _conv_self(a, L, m)
    best_score = _score_from_conv(c, n)
    best_a = a.copy()
    
    vel = np.zeros(n)
    temp = float(temp_start)
    step = 0.3
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        c = _conv_self(a, L, m)
        cur_score = _score_from_conv(c, n)
        
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Restart if stuck
        if no_improve > 150:
            prior = _random_prior(n, rng)
            mix = 0.15 + 0.4 * float(rng.random())
            a = _normalize_simplex((1.0 - mix) * best_a + mix * prior)
            vel = np.zeros(n)
            temp = float(temp_start)
            step = 0.3
            no_improve = 0
            c = _conv_self(a, L, m)
            cur_score = _score_from_conv(c, n)
            continue
        
        # Compute direction
        temp_eff = min(temp, 50.0 / max(float(np.max(c)), 1e-12))
        if temp_eff < 0.05:
            temp_eff = 0.05
        
        d = _smoothed_subgradient(a, c, n, temp_eff, L)
        
        # Nesterov update
        y = a + momentum * vel
        y_new = y - step * d
        y_new = _normalize_simplex(y_new)
        vel = y_new - y
        a = y_new
        
        cur_score = _score_from_conv(_conv_self(a, L, m), n)
        
        # Adaptive step
        if cur_score < cur_score - 1e-15:
            pass
        else:
            # Check if we made progress vs best
            if cur_score > best_score * 1.02:
                step = max(0.02, step * 0.7)
            else:
                step = min(1.0, step * 1.05)
        
        # Periodic coordinate refinement
        if it % 200 == 100:
            a, sc = _coord_refine_fast(a, L, m, n, rng, 80)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
        
        if it % 50 == 0:
            temp = min(temp_end, temp * 1.12)
        
        it += 1
    
    return best_a, best_score


def _coord_refine_fast(a, L, m, n, rng, n_trials):
    """Fast coordinate refinement: try random coordinates with golden section."""
    a = a.copy()
    c = _conv_self(a, L, m)
    score = _score_from_conv(c, n)
    
    for _ in range(n_trials):
        i = int(rng.integers(n))
        base = a[i]
        if base < 1e-15:
            continue
        
        # Try several multipliers
        best_val = base
        best_sc = score
        
        for factor in (0.3, 0.5, 0.7, 1.3, 1.6, 2.0, 0.1):
            a[i] = base * factor
            a_norm = _normalize_simplex(a)
            c_new = _conv_self(a_norm, L, m)
            sc = _score_from_conv(c_new, n)
            if sc < best_sc - 1e-15:
                best_sc = sc
                best_val = base * factor
                a[:] = a_norm
                c = c_new
                score = sc
        
        a[i] = best_val
    
    return a, score


def _annealing_search(a, L, m, n, rng, time_limit):
    """Simulated annealing with structured perturbations."""
    start = time.monotonic()
    deadline = start + float(time_limit)
    
    a = _normalize_simplex(a)
    c = _conv_self(a, L, m)
    best_score = _score_from_conv(c, n)
    best_a = a.copy()
    
    T = 0.05
    T_min = 0.001
    cooldown = 0.999
    
    while time.monotonic() < deadline:
        c = _conv_self(a, L, m)
        cur_score = _score_from_conv(c, n)
        
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
        
        # Generate perturbation
        perturb_type = rng.random()
        a_new = a.copy()
        
        if perturb_type < 0.3:
            # Smooth perturbation
            noise = rng.normal(0, 0.05, n)
            if n > 10:
                noise = np.convolve(noise, np.ones(5)/5, mode="same")
            a_new = a + noise
        elif perturb_type < 0.5:
            # Block perturbation
            block_size = int(rng.integers(max(2, n//50), max(3, n//10)))
            start_idx = int(rng.integers(0, max(1, n - block_size)))
            a_new[start_idx:start_idx+block_size] *= rng.uniform(0.5, 1.8)
        elif perturb_type < 0.7:
            # Swap-based perturbation
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            delta = a[i] * rng.uniform(-0.3, 0.3)
            a_new[i] -= delta
            a_new[j] += delta
        elif perturb_type < 0.9:
            # Blend with random prior
            prior = _random_prior(n, rng)
            a_new = (1 - rng.uniform(0.05, 0.2)) * a + rng.uniform(0.05, 0.2) * prior
        else:
            # Multiply by random smooth function
            t = np.linspace(0, 1, n)
            k = int(rng.integers(1, 4))
            ph = rng.uniform(0, 2*np.pi)
            mod = 1.0 + 0.2 * np.sin(2*np.pi*k*t + ph)
            a_new = a * mod
        
        a_new = _normalize_simplex(a_new)
        c_new = _conv_self(a_new, L, m)
        new_score = _score_from_conv(c_new, n)
        
        delta = new_score - cur_score
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-10)):
            a = a_new
        
        T = max(T_min, T * cooldown)
    
    return best_a, best_score


def _mws_search(a, L, m, n, rng, time_limit):
    """Multiplicative weight subroutine for finding good directions."""
    start = time.monotonic()
    deadline = start + float(time_limit)
    
    a = _normalize_simplex(a)
    FA = np.fft.rfft(a, L)
    FA_rev = np.fft.rfft(a[::-1], L)
    
    best_score = _score_from_conv(_conv_self(a, L, m), n)
    best_a = a.copy()
    
    inner = 20
    eta_s = 2.0
    eta_e = 35.0
    
    while time.monotonic() < deadline:
        prior = _random_prior(n, rng)
        logprior = np.log(np.maximum(prior, 1e-300))
        
        x = prior.copy()
        y = np.full(m, 1.0 / m)
        avg = np.zeros(n)
        
        for it in range(inner):
            frac = float(it) / max(1, inner - 1)
            eta = eta_s + (eta_e - eta_s) * frac
            
            X = np.fft.rfft(x, L)
            u = np.fft.irfft(FA * X, L)[:m]
            np.maximum(u, 0.0, out=u)
            
            vu = n * u
            vma = float(np.max(vu))
            z = eta * (vu - vma)
            z = np.clip(z, -50.0, 0.0)
            y = np.exp(z)
            ys = float(y.sum())
            if ys > 0 and np.isfinite(ys):
                y /= ys
            else:
                y.fill(1.0 / m)
            
            avg += x
            
            Y = np.fft.rfft(y, L)
            s_vec = np.fft.irfft(Y * FA_rev, L)[(n-1):m]
            np.maximum(s_vec, 0.0, out=s_vec)
            
            ps = n * s_vec
            sma = float(np.max(ps))
            zx = -eta * (ps - sma) + 0.6 * logprior
            zx -= float(np.max(zx))
            zx = np.clip(zx, -50.0, 0.0)
            x = np.exp(zx)
            xs = float(x.sum())
            if xs > 0 and np.isfinite(xs):
                x /= xs
            else:
                x = prior.copy()
        
        x = _normalize_simplex(0.5 * avg / inner + 0.5 * x)
        
        for lam in (0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0):
            cand = _normalize_simplex((1.0 - lam) * best_a + lam * x)
            sc = _score_from_conv(_conv_self(cand, L, m), n)
            if sc < best_score:
                best_score = sc
                best_a = cand.copy()
    
    return best_a, best_score


def _internal_exact(seq):
    if not isinstance(seq, list) or not seq:
        return float(np.inf)
    for x in seq:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return float(np.inf)
        if np.isnan(x) or np.isinf(x):
            return float(np.inf)
    a = [0.0 if float(x) < 0.0 else min(1000.0, float(x)) for x in seq]
    nn = len(a)
    s = sum(a)
    if s < 0.01:
        return float(np.inf)
    b = np.convolve(a, a)
    mb = max(b)
    return float(2.0 * nn * mb / (s * s))


def _exact_score(seq):
    try:
        val = evaluate_sequence(seq)
    except NameError:
        val = _internal_exact(seq)
    except Exception:
        val = float(np.inf)
    if val is None:
        return float(np.inf)
    return float(val)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search: combines subgradient, annealing, MWS, and coordinate refinement."""
    try:
        budget = float(budget_s)
    except (ValueError, TypeError):
        budget = 2400.0
    if not np.isfinite(budget) or budget < 10.0:
        budget = 2400.0
    
    start_time = time.monotonic()
    hard_deadline = start_time + budget - 5.0
    search_deadline = start_time + budget - 80.0
    
    rng = np.random.default_rng(seed)
    
    # Determine n from parent
    n = 6121
    parent_arr = None
    try:
        parent = height_sequence_1
        parent_arr = np.asarray(parent, dtype=np.float64).ravel()
        if parent_arr.size > 10 and np.all(np.isfinite(parent_arr)):
            n = parent_arr.size
    except NameError:
        pass
    except Exception:
        pass
    
    L = 1 << (2 * n - 1).bit_length()
    m = 2 * n - 1
    print(f"n={n}, L={L}, budget={budget:.0f}s")
    
    # Generate initial candidates
    inits = []
    if parent_arr is not None and parent_arr.size == n:
        inits.append(_normalize_simplex(np.maximum(parent_arr, 0)))
    
    shapes = _shape_candidates(n, rng)
    inits.extend(shapes)
    
    # Smoother version of parent
    if parent_arr is not None and parent_arr.size == n:
        k = max(3, n // 300)
        sm = np.convolve(parent_arr, np.ones(k) / k, mode="same")
        inits.append(_normalize_simplex(np.maximum(sm, 0)))
    
    # Find best initial
    best_a = None
    best_score = float('inf')
    for a in inits:
        c = _conv_self(a, L, m)
        sc = _score_from_conv(c, n)
        if sc < best_score:
            best_score = sc
            best_a = a
            print(f"  init: {sc:.8f}")
    
    print(f"Best initial: {best_score:.8f}")
    
    candidate_pool = [(best_a.copy(), best_score)]
    
    # Phase 1: Subgradient with momentum (main workhorse)
    t0 = time.monotonic()
    lim = min(500.0, 0.22 * budget)
    tmp, sc = _subgradient_momentum_search(best_a, L, m, n, rng, lim, 1.0, 150.0)
    if sc < best_score:
        best_score = sc
        best_a = tmp
    candidate_pool.append((best_a.copy(), best_score))
    print(f"Phase 1 ({time.monotonic()-t0:.0f}s): {best_score:.8f}")
    
    # Phase 2: MWS
    t0 = time.monotonic()
    remaining = search_deadline - time.monotonic()
    if remaining > 50:
        lim = min(300.0, 0.12 * budget)
        tmp, sc = _mws_search(best_a, L, m, n, rng, lim)
        if sc < best_score:
            best_score = sc
            best_a = tmp
        candidate_pool.append((best_a.copy(), best_score))
        print(f"Phase 2 ({time.monotonic()-t0:.0f}s): {best_score:.8f}")
    
    # Phase 3: Annealing for exploration
    t0 = time.monotonic()
    remaining = search_deadline - time.monotonic()
    if remaining > 50:
        lim = min(250.0, 0.10 * budget)
        tmp, sc = _annealing_search(best_a, L, m, n, rng, lim)
        if sc < best_score:
            best_score = sc
            best_a = tmp
        candidate_pool.append((best_a.copy(), best_score))
        print(f"Phase 3 ({time.monotonic()-t0:.0f}s): {best_score:.8f}")
    
    # Phase 4: More subgradient from best
    t0 = time.monotonic()
    remaining = search_deadline - time.monotonic()
    if remaining > 50:
        lim = min(400.0, 0.18 * budget)
        tmp, sc = _subgradient_momentum_search(best_a, L, m, n, rng, lim, 5.0, 300.0, 0.95)
        if sc < best_score:
            best_score = sc
            best_a = tmp
        candidate_pool.append((best_a.copy(), best_score))
        print(f"Phase 4 ({time.monotonic()-t0:.0f}s): {best_score:.8f}")
    
    # Phase 5: Aggressive coordinate refinement
    t0 = time.monotonic()
    remaining = search_deadline - time.monotonic()
    if remaining > 30:
        a_ref = best_a.copy()
        sc_ref = _score_from_conv(_conv_self(a_ref, L, m), n)
        n_rounds = max(1, int(remaining / 15))
        for rnd in range(n_rounds):
            if time.monotonic() > search_deadline:
                break
            a_ref, sc_ref = _coord_refine_fast(a_ref, L, m, n, rng, 300)
            if sc_ref < best_score:
                best_score = sc_ref
                best_a = a_ref.copy()
                candidate_pool.append((best_a.copy(), best_score))
            if (rnd + 1) % 3 == 0:
                print(f"  coord round {rnd+1}: {best_score:.8f}")
    print(f"Phase 5 ({time.monotonic()-t0:.0f}s): {best_score:.8f}")
    
    # Phase 6: Final MWS polish
    t0 = time.monotonic()
    remaining = search_deadline - time.monotonic()
    if remaining > 30:
        lim = min(150.0, remaining - 20)
        tmp, sc = _mws_search(best_a, L, m, n, rng, lim)
        if sc < best_score:
            best_score = sc
            best_a = tmp
        candidate_pool.append((best_a.copy(), best_score))
    print(f"Phase 6 ({time.monotonic()-t0:.0f}s): {best_score:.8f}")
    
    # Phase 7: Final subgradient polish with low temperature
    t0 = time.monotonic()
    remaining = search_deadline - time.monotonic()
    if remaining > 20:
        lim = min(100.0, remaining - 15)
        tmp, sc = _subgradient_momentum_search(best_a, L, m, n, rng, lim, 0.5, 20.0, 0.5)
        if sc < best_score:
            best_score = sc
            best_a = tmp
        candidate_pool.append((best_a.copy(), best_score))
    print(f"Phase 7 ({time.monotonic()-t0:.0f}s): {best_score:.8f}")
    
    # Evaluate candidates with exact function
    # Sort by internal score
    candidate_pool.sort(key=lambda x: x[1])
    
    # Deduplicate
    unique_cands = []
    seen = set()
    for arr, sc in candidate_pool:
        key = tuple(np.round(arr, 12))
        if key not in seen:
            seen.add(key)
            unique_cands.append((arr, sc))
    
    # Also add parent
    if parent_arr is not None and parent_arr.size == n:
        unique_cands.append((*[_normalize_simplex(np.maximum(parent_arr, 0))], _score_from_conv(_conv_self(_normalize_simplex(np.maximum(parent_arr, 0)), L, m), n)))
    
    best_seq = None
    best_exact = float('inf')
    
    for arr, _ in unique_cands[:20]:
        if time.monotonic() > hard_deadline - 10:
            break
        seq = [float(x) for x in arr]
        val = _exact_score(seq)
        if np.isfinite(val) and val < best_exact:
            best_exact = val
            best_seq = seq
            print(f"  exact: {val:.8f}")
    
    if best_seq is None:
        best_seq = [float(x) for x in best_a]
        best_exact = _exact_score(best_seq)
    
    # Final micro-perturbations
    if best_seq is not None and time.monotonic() < hard_deadline - 10:
        mean_val = sum(best_seq) / len(best_seq)
        for trial in range(30):
            if time.monotonic() > hard_deadline - 5:
                break
            seq_pert = [max(0.0, x + rng.normal(0, 0.002 * mean_val)) for x in best_seq]
            val_pert = _exact_score(seq_pert)
            if np.isfinite(val_pert) and val_pert < best_exact:
                best_exact = val_pert
                best_seq = seq_pert
                print(f"  micro-perturb improved: {best_exact:.8f}")
    
    print(f"FINAL: {best_exact:.8f}")
    return best_seq
