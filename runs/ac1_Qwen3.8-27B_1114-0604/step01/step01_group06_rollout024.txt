"""
Algorithm: Multi-scale log-space optimization with active-set polishing.

1. Generate diverse structured starting points (smooth steps, sinusoidal,
   resampled provided sequence, etc.) for a chosen length n.
2. Optimize each start with L-BFGS-B in log-space (softmax parameterization)
   using a smooth max approximation (log-sum-exp) of the autocorrelation peak.
3. Polish the best candidates with an active-set gradient method that
   targets the lag(s) where the autocorrelation is near its maximum.
4. Periodically inject random perturbations to escape local minima.
5. Try multiple values of n and keep the overall best sequence.

The score to minimize is: 2 * n * max(conv(a, a)) / (sum(a))^2
which is homogeneous, so we optimize with sum(a) = 1 and minimize 2*n*max_c.
"""

import numpy as np
import time
from scipy.optimize import minimize

# Global state for the optimizer
_G = {}

def _init(n, tau=1e-5):
    """Initialize global optimizer state for a given sequence length."""
    _G['n'] = n
    _G['N'] = 1 << (3 * n - 2).bit_length()
    _G['tau'] = tau
    _G['bound'] = 25.0
    _G['bounds'] = [(-_G['bound'], _G['bound'])] * n

def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.full(z.shape, 1.0 / z.size)
    return e / s

def _fft_conv(a):
    """Compute linear convolution a * a using FFT."""
    N = _G['N']
    n = _G['n']
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1].real
    return np.maximum(c, 0.0)

def _fast_score(a):
    """Compute score = 2*n*max_c / (sum_a)^2 quickly."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    c = _fft_conv(a)
    m = c.max()
    return 2.0 * _G['n'] * m / (s * s)

def _obj_grad(z):
    """Objective and gradient for L-BFGS. Uses smooth max (log-sum-exp)."""
    n = _G['n']
    N = _G['N']
    tau = _G['tau']
    
    z = np.clip(z, -_G['bound'], _G['bound'])
    a = _softmax(z)
    
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1].real
    c = np.maximum(c, 0.0)
    
    m = c.max()
    diff = (c - m) / tau
    diff = np.clip(diff, -700.0, 0.0)
    e = np.exp(diff)
    se = e.sum()
    if se < 1e-300:
        return 1e10, np.zeros(n)
    
    p = e / se
    
    # Gradient: dF/dz where F = tau * log(sum exp(c/tau))
    # dF/da = 2 * (p * a) roughly (cross-correlation)
    fp = np.fft.rfft(p, N)
    h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real
    
    # dF/da[i] ≈ 2 * h[i] (the factor 2 comes from a appearing twice)
    # dF/dz[i] = dF/da[i] * a[i] * (1 - a[i]) = dF/da[i] * a[i] - (sum_j dF/da[j]*a[j]) * a[i]
    # For softmax: dF/dz[i] = a[i] * (dF/da[i] - sum_j dF/da[j]*a[j])
    s_h = np.dot(h, a)
    grad_a = 2.0 * h
    grad_z = a * (grad_a - np.dot(grad_a, a))
    
    val = m + tau * np.log(se)
    
    # Scale for numerical stability
    scale = float(n)
    val *= scale
    grad_z *= scale
    
    return float(val), grad_z

def _run_lbfgs(z0, maxiter=200, maxfun=600):
    """Run L-BFGS-B optimization from initial point z0."""
    try:
        res = minimize(
            _obj_grad, z0, jac=True, method='L-BFGS-B',
            bounds=_G['bounds'],
            options={'maxiter': maxiter, 'maxfun': maxfun, 'ftol': 1e-14, 'pgtol': 1e-12, 'disp': False}
        )
        return res
    except Exception:
        return None

def _make_starts(n, rng):
    """Generate diverse starting points."""
    starts = []
    u = np.linspace(0, 1, n)
    x = 2*u - 1  # in [-1, 1]
    
    def add_from_f(f):
        f = np.maximum(f, 1e-10)
        f = f / f.sum()
        z = np.log(f)
        z = z - z.mean()
        z = np.clip(z, -_G['bound']+1, _G['bound']-1)
        a = _softmax(z)
        sc = _fast_score(a)
        if np.isfinite(sc):
            starts.append((sc, z.copy()))
    
    # 1. Use provided height_sequence_1 if available
    try:
        g = globals().get('height_sequence_1')
        if g is not None:
            g = np.asarray(g, dtype=np.float64)
            if g.size > 10:
                g = np.clip(g, 1e-10, 1000)
                if g.sum() > 0:
                    # Resample to length n
                    x_old = np.linspace(0, 1, g.size)
                    log_g = np.log(np.maximum(g, 1e-10))
                    log_g = log_g - log_g.mean()
                    z = np.interp(np.linspace(0, 1, n), x_old, log_g)
                    z = np.clip(z, -_G['bound']+1, _G['bound']-1)
                    a = _softmax(z)
                    sc = _fast_score(a)
                    if np.isfinite(sc):
                        starts.append((sc, z.copy()))
    except Exception:
        pass
    
    # 2. Smooth step functions
    for alpha in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for w in [0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            f = alpha + (1-alpha) * (1/(1+np.exp(-(x/w))))
            add_from_f(f)
    
    # 3. Gaussian bumps
    for center in [0.3, 0.5, 0.7]:
        for sigma in [0.1, 0.2, 0.3, 0.4]:
            f = 0.3 + 0.7 * np.exp(-((u-center)**2)/(2*sigma**2))
            add_from_f(f)
    
    # 4. Sinusoidal
    for m in [1, 2, 3, 4]:
        for amp in [0.2, 0.4, 0.6, 0.8]:
            f = 1.0 + amp * np.sin(2*np.pi*m*u + np.pi/2)
            add_from_f(f)
            f = 1.0 + amp * np.cos(2*np.pi*m*u)
            add_from_f(f)
    
    # 5. Linear ramp
    for slope in [0.5, 1.0, 2.0, 3.0, 5.0]:
        f = 1.0 + slope * x
        add_from_f(f)
    
    # 6. "Dip" shapes (high at edges, low in middle)
    for depth in [0.3, 0.5, 0.7]:
        for width in [0.1, 0.2, 0.3]:
            f = 1.0 - depth * np.exp(-((u-0.5)**2)/(2*width**2))
            add_from_f(f)
    
    # 7. Exponential decay from each end
    for rate in [1, 2, 3, 5, 8]:
        f = np.exp(-rate * np.minimum(u, 1-u))
        add_from_f(f)

    for _ in range(5):

        for m in range(1, 5):

            f += 0.3 * rng.standard_normal() * np.cos(2*np.pi*m*u)

def _polish_active_set(a, deadline):
    """
    Active-set polishing: identify peak lags and do targeted gradient steps.
    """
    n = _G['n']
    N = _G['N']
    a = a.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-12:
        return a
    a /= s
    
    z = np.log(np.maximum(a, 1e-30))
    z = z - z.mean()
    z = np.clip(z, -_G['bound'], _G['bound'])
    
    best_a = a.copy()
    best_score = _fast_score(a)
    rng = np.random.default_rng(12345)
    no_improve = 0
    
    # Use a sequence of decreasing tau for adaptivity
    tau_sched = [1e-3, 5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]
    tau_idx = 0
    tau = tau_sched[0]
    
    while time.monotonic() < deadline:
        c = _fft_conv(a)
        F = c.max()
        m = F
        
        if F * _G['n'] * 2 < best_score * (s*s) / (s*s):  # i.e., 2*n*F < best_score
            best_a = a.copy()
            best_score = 2.0 * n * F
        
        # Compute soft-max weights
        diff = (c - m) / tau
        diff = np.clip(diff, -700.0, 0.0)
        e = np.exp(diff)
        se = e.sum()
        if se < 1e-300:
            break
        p = e / se
        
        # Gradient
        fa = np.fft.rfft(a, N)
        fp = np.fft.rfft(p, N)
        h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real
        s_h = np.dot(h, a)
        g = 2.0 * a * (h - s_h)
        
        # Line search in log-space
        accepted = False
        for eta in [1e-5, 3e-5, 1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 0.1, 0.3, 1.0]:
            z_t = np.clip(z - eta * g, -_G['bound'], _G['bound'])
            a_t = _softmax(z_t)
            F_t = _fft_conv(a_t).max()
            if F_t < F * (1 - 1e-10):
                z = z_t
                a = a_t
                accepted = True
                no_improve = 0
                break
        
        if not accepted:
            no_improve += 1
            # Try random escape
            escaped = False
            for scale in [0.01, 0.05, 0.1, 0.3, 0.5]:
                z_t = np.clip(z + scale * rng.standard_normal(n), -_G['bound'], _G['bound'])
                a_t = _softmax(z_t)
                F_t = _fft_conv(a_t).max()
                if F_t < F * 1.05:  # allow slight increase, just need to escape
                    z = z_t
                    a = a_t
                    escaped = True
                    break
            if not escaped:
                # Decrease tau (make the max approximation sharper)
                if tau_idx < len(tau_sched) - 1:
                    tau_idx += 1
                    tau = tau_sched[tau_idx]
                else:
                    no_improve += 10
                if no_improve > 500:
                    break
            else:
                no_improve = 0
        else:
            # Decrease tau occasionally when making progress
            if no_improve == 0 and tau_idx < len(tau_sched) - 1:
                # Only decrease tau every few successful steps
                pass  # Keep tau constant for a while
    
    return best_a
