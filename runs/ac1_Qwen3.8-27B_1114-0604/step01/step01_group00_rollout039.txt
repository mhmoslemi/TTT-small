def _lp_direction(f, M):
    """Solve LP to find best response direction."""
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    
    # Constraint: (f*b)[k] <= M for all k
    # (f*b)[k] = sum_j f[j] * b[k-j]
    # This is a linear constraint in b (since f is fixed)
    
    # Build the constraint matrix
    # A[k, j] = f[j] if 0 <= k-j < n, else 0
    # This is a Toeplitz matrix
    
    # For efficiency, use the structure
    constraints = []
    for k in range(2 * n - 1):
        # (f*b)[k] = sum_j f[j] * b[k-j] where 0 <= k-j < n
        # = sum_j f[j] * b[k-j] where j is such that 0 <= k-j < n
        # = sum_{j=max(0,k-n+1)}^{min(k,n-1)} f[j] * b[k-j]
        # Let i = k-j, so i ranges from max(0, k-n+1) to min(k, n-1)
        # = sum_{i=max(0,k-n+1)}^{min(k,n-1)} f[k-i] * b[i]
        # 
        # So the constraint is: sum_{i} f[k-i] * b[i] <= M
        # where i = k-j = k - max(0,k-n+1) ... k - min(k,n-1)
        # = min(k, n-1) ... max(0, k-n+1)
        
        i_max = min(k, n - 1)
        i_min = max(0, k - n + 1)
        coeffs = f[k - i_min : k - i_max + 1]  # f[k-i] for i = i_min to i_max
        constraints.append(coeffs @ b[i_min:i_max+1] <= M)
    
    objective = cp.Maximize(cp.sum(b))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.GLOP)  # or another solver
    
    if b.value is None:
        return f.copy()
    return np.maximum(b.value, 0)

The problem is to minimize max_k (f*f)[k] / (sum f)^2. We showed that the theoretical minimum is 2n/(2n-1) ≈ 1 (perfectly flat convolution). We're at 1.515767, so we're about 51.6% of the way from the theoretical minimum (1) to the uniform baseline (2).

The question is: what structure allows us to get closer to 1?

For a "flat" convolution, we need (f*f)[k] ≈ constant for all k. This means:
- The "energy" of f is spread evenly across all convolution lags
- This is related to the "ideal" or "flat-top" properties in spectral analysis

One approach: think of f as a probability distribution, and (f*f) as the distribution of the sum of two independent copies. We want this sum distribution to be as uniform as possible over {0, ..., 2n-2}.

For a uniform distribution on {0, ..., n-1}, the sum distribution is triangular (peaked at the center). To make it flatter, we need a distribution whose "sum" is more spread out.

"""
Step function optimizer minimizing 2*n*max(convolve(f,f))/(sum f)^2.
Uses projected gradient descent with adaptive softmax temperature,
multiple diverse starting points (Hann window, Gaussian, etc.),
backtracking line search, and a low-temperature polish phase.
"""

import numpy as np
from scipy.signal import correlate
import time


def _convolve_fft(f):
    """Full linear convolution f*f via FFT."""
    n = len(f)
    size = 1 << (2 * n - 2).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _project_simplex(v):
    """Project v onto {x >= 0, sum x = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    total = v.sum()
    if total < 1e-300:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1)
    cond = u * idx > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = int(np.max(np.nonzero(cond)[0]))
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _eval_seq(f):
    """Evaluate the objective."""
    return evaluate_sequence(f.tolist())


def _gradient(f, T):
    """Compute gradient and convolution."""
    c = _convolve_fft(f)
    cs = c / T
    cs -= cs.max()
    p = np.exp(cs)
    ps = p.sum()
    if ps < 1e-300:
        p = np.zeros(len(c))
        p[len(c) // 2] = 1.0
    else:
        p /= ps
    grad = correlate(p, f, mode='valid')
    return grad, p, c


def _optimize_pg(f, n, deadline, rng, label=""):
    """Projected gradient descent with adaptive temperature and backtracking."""
    f = np.maximum(f, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f /= s

    best_score = _eval_seq(f)
    best_f = f.copy()

    c0 = _convolve_fft(f)
    peak0 = c0.max()
    best_peak = peak0

    T = (peak0 - c0[n - 1]) * 0.03
    if T < 1e-10:
        T = peak0 * 0.01
    T_min = peak0 * 1e-15
    lr = 2.0
    lr_min = 1e-7
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        c = _convolve_fft(f)
        peak = c.max()

        # Softmax gradient
        cs = c / T
        cs -= cs.max()
        p = np.exp(cs)
        ps = p.sum()
        if ps < 1e-300:
            p = np.zeros(len(c))
            p[n - 1] = 1.0
        else:
            p /= ps

        grad = correlate(p, f, mode='valid')

        # Adaptive temperature via effective number of peaks
        N_eff = 1.0 / max((p ** 2).sum(), 1e-30)
        if N_eff < 4:
            T = min(T * 1.8, peak0 * 0.5)
        elif N_eff > 25:
            T = max(T * 0.6, T_min)

        # Backtracking line search
        improved = False
        lr_try = lr
        for _ in range(12):
            f_try = _project_simplex(f - lr_try * grad)
            c_try = _convolve_fft(f_try)
            if c_try.max() < best_peak - 1e-14:
                f = f_try
                best_peak = c_try.max()
                lr = lr_try
                improved = True
                break
            lr_try *= 0.5

        if improved:
            no_improve = 0
            if it % 3000 == 0:
                sc = _eval_seq(f)
                if sc < best_score:
                    best_score = sc
                    best_f = f.copy()
                    print(f"    {label} it {it}: {best_score:.6f}")
        else:
            no_improve += 1
            if no_improve > 100:
                # Multiple perturbation strategies
                strategy = it % 3
                if strategy == 0:
                    sigma = f.std() * 0.03
                    f = _project_simplex(f + sigma * rng.standard_normal(n))
                elif strategy == 1:
                    # Block swap perturbation
                    blen = max(1, n // 10)
                    i = int(rng.integers(0, n - blen))
                    j = int(rng.integers(0, n - blen))
                    f_try = f.copy()
                    f_try[i:i+blen], f_try[j:j+blen] = f_try[j:j+blen].copy(), f_try[i:i+blen].copy()
                    f_try = _project_simplex(f_try)
                    if _convolve_fft(f_try).max() < best_peak * 1.05:
                        f = f_try
                else:
                    # Blend with Hann window
                    j_arr = np.arange(n, dtype=np.float64)
                    hann = 0.5 * (1 + np.cos(2 * np.pi * j_arr / (n - 1)))
                    hann /= hann.sum()
                    for t in [0.1, 0.3, 0.5]:
                        f_try = _project_simplex((1 - t) * f + t * hann)
                        if _convolve_fft(f_try).max() < best_peak * 1.05:
                            f = f_try
                            break
                best_peak = _convolve_fft(f).max()
                lr = max(2.0, lr * 5)
                T = max(best_peak * 0.02, T_min)
                no_improve = 0

    sc = _eval_seq(f)
    if sc < best_score:
        best_score = sc
        best_f = f.copy()
    return best_f, best_score


def _polish_low_T(f, n, deadline, rng):
    """Polish with very low temperature focusing on the top peak."""
    f = f.copy().astype(np.float64)
    f = np.maximum(f, 0)
    f /= f.sum()

    best_score = _eval_seq(f)
    best_f = f.copy()
    best_peak = _convolve_fft(f).max()

    lr = 0.5
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        c = _convolve_fft(f)
        peak = c.max()
        k* = int(np.argmax(c))

        # Soft weights around the peak
        p = np.zeros(len(c))
        width = 1.5
        for dk in range(-int(3 * width), int(3 * width) + 1):
            kk = k* + dk
            if 0 <= kk < len(c):
                p[kk] = np.exp(-0.5 * (dk / width) ** 2)
        ps = p.sum()
        if ps < 1e-300:
            p[k*] = 1.0
        else:
            p /= ps

        grad = correlate(p, f, mode='valid')

        improved = False
        lr_try = lr
        for _ in range(10):
            f_try = _project_simplex(f - lr_try * grad)
            c_try = _convolve_fft(f_try)
            if c_try.max() < best_peak - 1e-13:
                f = f_try
                best_peak = c_try.max()
                lr = lr_try
                improved = True
                break
            lr_try *= 0.5

        if not improved:
            no_improve += 1
            if no_improve > 30:
                # Try targeted exchanges: move mass from peak contributors to low-impact positions
                c = _convolve_fft(f)
                k_star = int(np.argmax(c))
                # Find top contributor
                contrib = np.zeros(n)
                for jj in range(n):
                    j2 = k_star - jj
                    if 0 <= j2 < n:
                        contrib[jj] = f[jj] * f[j2]
                top_idx = int(np.argmax(contrib))
                # Find low-impact receiver
                sensitivity = np.zeros(n)
                for jj in range(n):
                    # Max impact of adding mass at position jj
                    lo = max(0, -jj)
                    hi = min(n - 1, len(c) - 1 - jj)
                    sensitivity[jj] = c[lo:hi+1].max() if hi >= lo else 0
                # Prefer positions with low f and low sensitivity
                score_rec = sensitivity + f * c.max()
                recv_candidates = np.argsort(score_rec)[:20]
                for recv in recv_candidates:
                    if recv == top_idx:
                        continue
                    for frac in [0.02, 0.05, 0.1]:
                        amt = f[top_idx] * frac
                        if amt < 1e-15:
                            break
                        f_try = f.copy()
                        f_try[top_idx] -= amt
                        f_try[recv] += amt
                        f_try /= f_try.sum()
                        c_try = _convolve_fft(f_try)
                        if c_try.max() < best_peak:
                            f = f_try
                            best_peak = c_try.max()
                            improved = True
                            break
                    if improved:
                        break
                lr = max(lr * 1.5, 0.5)
                no_improve = 0

        if it % 5000 == 0:
            sc = _eval_seq(f)
            if sc < best_score:
                best_score = sc
                best_f = f.copy()
                print(f"    Polish it {it}: {best_score:.6f}")

    sc = _eval_seq(f)
    if sc < best_score:
        best_score = sc
        best_f = f.copy()
    return best_f, best_score


def _lp_direction(f, M, n):
    """
    Solve (approximately) the LP: max sum(b) s.t. (f*b)[k] <= M, b >= 0.
    Uses multiplicative weight updates as a fast approximation.
    Returns the direction b normalized.
    """
    b = np.ones(n) / n
    # Dual variables y_k for each constraint
    y = np.ones(2 * n - 1) / (2 * n - 1)
    eta = 1.0

    for _ in range(50):
        # Compute constraint violations: A^T @ y where A[k,j] = f[k-j] if valid
        # violation[j] = sum_k y[k] * A[k,j] = sum_k y[k] * f[k-j] (if 0<=k-j<n)
        # This is correlate(y, f, 'valid') but with y indexed as the longer signal
        # violation[j] = sum_k y[k] * f[k-j] for valid k,j
        # = sum_i y[i+j] * f[i] for i=0..n-1  (letting i = k-j, so k = i+j)
        # = correlate(y, f, 'valid')[j]
        
        # y has length 2n-1, f has length n
        # correlate(y, f, 'valid')[j] = sum_i y[j+i] * f[i] for j = 0..(2n-1-n) = n-1
        viol = correlate(y, f, mode='valid')
        
        # Update b: b_j proportional to exp(-eta * viol_j)
        exp_arg = -eta * viol
        exp_arg -= exp_arg.max()
        b_new = np.exp(exp_arg)
        b_sum = b_new.sum()
        if b_sum < 1e-300:
            b_new = np.ones(n) / n
        else:
            b_new /= b_sum
        b = b_new
        
        # Update y: increase weights on violated constraints
        c_fb = _convolve_fft(f * 0 + 1)  # not quite right, need f*b
        # Actually compute f*b:
        # (f*b)[k] = sum_j f[j]*b[k-j]
        fb = _convolve_fft(f * b)  # WRONG - this is element-wise product then FFT
        # Need actual convolution of f and b
        size = 1 << (2 * n - 2).bit_length()
        F = np.fft.rfft(f, size)
        B = np.fft.rfft(b, size)
        fb = np.fft.irfft(F * B, size)[:2 * n - 1]
        
        # Update y: y_k proportional to y_k * exp(eta * max(0, fb[k] - M) / M)
        violation_amt = np.maximum(0, (fb - M) / max(M, 1e-10))
        exp_arg = eta * violation_amt
        exp_arg -= exp_arg.max() / max(exp_arg.max(), 1)
        y_new = y * np.exp(exp_arg)
        y_new /= y_new.sum()
        y = y_new
        eta *= 1.1

    return b
