```
             a += room * (deficit/total)
```

```
def peak_weights(c, tol, maxK, minK, temp_frac):
    L=c.size
    M=float(c.max())
    if M <= 0.0 or L==0:
       mid=L//2
       return np.array([mid], dtype=np.int64), np.array([1.0])
    minK = max(1, min(int(minK), L))
    maxK = max(minK, min(int(maxK), L))
    top_idx = np.argpartition(c, -maxK)[-maxK:]
    top_idx = np.sort(top_idx)
    threshold = M * (1.0 - float(tol))
    active = top_idx[c[top_idx] >= threshold]
    if active.size < minK:
        extra = np.argpartition(c, -minK)[-minK:]
        if active.size:
            active = np.unique(np.concatenate((active, extra)))
        else:
            active = extra
    # If active still large? <=maxK or minK union could exceed maxK if minK>maxK? We ensured maxK>=minK. Unique of active (subset top maxK) plus extra (minK <=maxK) could exceed maxK if active and extra disjoint? But extra subset top minK, which is subset of top maxK, so union <= top maxK? Since active subset top maxK, extra subset top minK subset top maxK, union size <=maxK. Good.
    vals = c[active]
    temp = max(1e-12, float(temp_frac) * M)
    w = np.exp((vals - vals.max()) / temp)
    wsum = w.sum()
    if wsum <= 0: w = np.ones(active.size)/active.size
    else: w = w / wsum
```

```
def grad_fft_weights(a, active, w):
    n=a.size
    L=2*n-1
    wf=np.zeros(L, dtype=np.float64)
    wf[active]=w
    wrev=wf[::-1]
    m_len = n + L - 1  # 3n-2
    if m_len <= 1:

    A=np.fft.rfft(a, N)
    B=np.fft.rfft(wrev, N)
    conv=np.fft.irfft(A*B, N)
    d=conv[2*n-2::-1][:n]
```

```python
"""
Multi-resolution smooth-gradient search for low self-convolution sequences.

Approach:
  * Work in the scale-invariant normalization sum(a) = n (mean 1).
  * Fast exact FFT evaluation of a * a and FFT-based gradients for a
    weighted sum of active convolution peaks.
  * Broad-to-narrow active peak sets, Adam with restarts, low-frequency
    perturbations, and occasional linear-programming directions inspired
    by the previous state-of-the-art projection idea.
  * Several initialized families (one-sided power laws, two-sided power
    laws, trigonometric/random smooth shapes, sparse seeded shapes, parent
    sequence) are ranked and refined in phases, optionally at a longer
    length before final polishing.
"""

import numpy as np
import time

try:
    from scipy.optimize import linprog
    from scipy.sparse import csc_matrix
    _HAS_SCIPY = True
except Exception:
    _HAS_SCIPY = False


def fast_full_conv(a):
    """Linear convolution a * a via FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = 2 * n - 1
    N = 1 << (m - 1).bit_length()
    A = np.fft.rfft(a, N)
    c = np.fft.irfft(A * A, N)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def objective_value(a):
    """Objective used by the public evaluator, computed with FFT."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.inf
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    c = fast_full_conv(a)
    if c.size == 0:
        return np.inf
    M = float(c.max())
    return float(2.0 * n * M / (S * S))


def project_sum(a, target):
    """Project to nonnegative, capped at 1000, and force sum = target."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    n = a.size
    if n == 0:
        return a
    target = float(target)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-12:
        val = target / float(n)
        if val > 1000.0:
            val = 1000.0
        return np.full(n, val, dtype=np.float64)

    if abs(s - target) <= 1e-9 * max(1.0, target):
        return a

    if s < target:
        scale = target / s
        if float(a.max()) * scale <= 1000.0:
            a = a * scale
        else:
            deficit = target - s
            room = 1000.0 - a
            total = float(room.sum())
            if total > 0.0:
                a = a + room * (deficit / total)
                a = np.clip(a, 0.0, 1000.0)
    else:
        a = a * (target / s)
        a = np.clip(a, 0.0, 1000.0)

    s2 = float(a.sum())
    if s2 <= 1e-12:
        return np.full(n, min(1000.0, target / float(n)), dtype=np.float64)

    if s2 < target:
        deficit = target - s2
        room = 1000.0 - a
        total = float(room.sum())
        if total > 0.0:
            a = a + room * (deficit / total)
            a = np.clip(a, 0.0, 1000.0)
    elif s2 > target:
        a = a * (target / s2)
        a = np.clip(a, 0.0, 1000.0)

    return a


def resize_sequence(a, new_n, rng=None):
    """Resample a sequence to another length while preserving shape."""
    a = np.asarray(a, dtype=np.float64).ravel()
    old_n = a.size
    if new_n <= 0:
        return np.empty(0, dtype=np.float64)
    if old_n == 0:
        return np.ones(new_n, dtype=np.float64)
    if old_n == 1:
        b = np.full(new_n, float(a[0]), dtype=np.float64)
        return project_sum(b, new_n)
    if old_n == new_n:
        return project_sum(a, new_n)

    x_old = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    b = np.interp(x_new, x_old, a)

    if rng is not None:
        noise = rng.standard_normal(new_n) * 1e-4
        b = b * np.exp(noise)

    return project_sum(b, new_n)


def score_key(item):
    return item[0]


def get_parent_sequence():
    """Fetch the globally supplied previous construction, if present."""
    d = globals()
    p = d.get("height_sequence_1", None)
    if p is None:
        return None
    try:
        arr = np.asarray(p, dtype=np.float64).ravel()
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = arr[np.isfinite(arr)]
    if arr.size == 0:
        return None
    return arr


def build_initials(n, rng, parent=None):
    """Diverse initial mean-1 sequences for length n."""
    n = int(n)
    X = np.arange(n, dtype=np.float64)
    if n > 1:
        y = X / float(n - 1)
    else:
        y = np.zeros(n, dtype=np.float64)

    out = []

    if parent is not None:
        p = np.asarray(parent, dtype=np.float64).ravel()
        if p.size == n:
            pp = p
        else:
            pp = resize_sequence(p, n, rng)
        out.append(project_sum(pp, n))

    out.append(project_sum(np.ones(n, dtype=np.float64), n))

    # One-sided power laws (very important: inverse-square-root family).
    for p in (0.0, 0.1, 0.2, 0.3, 0.4, 0.45, 0.5):
        for off in (1.0, 5.0):
            if p == 0.0:
                a = np.ones(n, dtype=np.float64)
            else:
                a = np.power(X + off, -p)
            out.append(project_sum(a, n))

    # Two-sided power laws.
    for p in (0.0, 0.2, 0.35, 0.5):
        for q in (0.0, 0.1, 0.2, 0.3):
            if p == 0.0 and q == 0.0:
                continue
            a = np.ones(n, dtype=np.float64)
            if p > 0.0:
                a = a * np.power(X + 1.0, -p)
            if q > 0.0:
                a = a * np.power((n - X) + 1.0, -q)
            out.append(project_sum(a, n))

    # Exponential shapes.
    for al in (0.5, 1.0, 2.0, 5.0):
        out.append(project_sum(np.exp(-al * y), n))
    for al in (1.0, 2.0, 4.0):
        out.append(project_sum(np.exp(-al * (np.abs(y - 0.5) * 2.0)), n))

    # Low-frequency trigonometric modulation.
    for f in (1, 2, 3, 5, 9):
        ph = rng.random() * 6.28318530718
        for amp in (0.5, 0.8):
            a = 1.0 + amp * np.cos(2.0 * np.pi * f * y + ph)
            out.append(project_sum(np.maximum(a, 1e-3), n))

    # Random smooth Fourier shapes.
    for _ in range(4):
        a = np.ones(n, dtype=np.float64)
        for f, amp in ((1, rng.uniform(0.5, 1.0)),
                       (3, rng.uniform(0.2, 0.7)),
                       (7, rng.uniform(0.1, 0.4))):
            a = a + amp * np.cos(2.0 * np.pi * f * y + rng.uniform(0.0, 6.28318530718))
        out.append(project_sum(np.maximum(a, 1e-3), n))

    # Seeded sparse shapes plus a small background.
    for mult in (0.8, 1.2, 1.7, 2.5):
        count = int(max(8, mult * np.sqrt(float(n))))
        count = min(n, count)
        idx = rng.choice(n, size=count, replace=False)
        a = np.zeros(n, dtype=np.float64)
        a[idx] = 1.0
        a = a + 0.05
        out.append(project_sum(a, n))

    return out


def peak_weights(c, tol, maxK, minK, temp_frac):
    """
    Choose active convolution peaks and softmax weights.

    The active set is restricted to top maxK peaks, but at least minK peaks
    are always included. The tolerance gives a wider window.
    """
    L = c.size
    M = float(c.max()) if L else 0.0
    if L == 0 or M <= 1e-15:
        mid = L // 2
        return np.array([mid], dtype=np.int64), np.array([1.0], dtype=np.float64)

    minK = int(max(1, min(minK, L)))
    maxK = int(max(minK, min(maxK, L)))

    top_idx = np.argpartition(c, -maxK)[-maxK:]
    top_idx = np.sort(top_idx)

    threshold = M * (1.0 - float(tol))
    active = top_idx[c[top_idx] >= threshold]

    if active.size < minK:
        extra = np.argpartition(c, -minK)[-minK:]
        if active.size:
            active = np.unique(np.concatenate((active, extra)))
        else:
            active = extra

    vals = c[active]
    temp = max(1e-12, float(temp_frac) * M)
    w = np.exp((vals - float(vals.max())) / temp)
    wsum = float(w.sum())
    if wsum <= 0.0 or not np.isfinite(wsum):
        w = np.ones(active.size, dtype=np.float64) / float(active.size)
    else:
        w = w / wsum

    return active, w


def grad_fft_weights(a, active, w):
    """
    Gradient of sum_k w_k (a*a)_k with respect to a.

    Uses the identity:
        d_i = sum_j a_j w_{i+j} = (a * w_rev)_{2n-2-i}
    where w_rev is the reversed full weight vector.
    """
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)

    L = 2 * n - 1
    wf = np.zeros(L, dtype=np.float64)
    wf[active] = w
    wrev = wf[::-1]

    conv_len = n + L - 1
    if conv_len <= 1:
        return np.zeros(n, dtype=np.float64)

    N = 1 << (conv_len - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(wrev, N)
    conv = np.fft.irfft(A * B, N)

    d = conv[2 * n - 2 :: -1][:n]
    return 2.0 * d


def perturb(a, rng, scale):
    """Low-frequency multiplicative perturbation for restarts."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return a

    noise = rng.standard_normal(n)
    if n > 16:
        A = np.fft.rfft(noise)
        cutoff = max(2, int(0.12 * len(A)))
        A[cutoff:] = 0.0
        noise = np.fft.irfft(A, n)

    # A slight asymmetric bias helps leave perfectly symmetric saddles.
    lin = np.linspace(-0.3, 0.3, n, dtype=np.float64)
    a2 = a * np.exp(scale * (noise + 0.5 * lin))
    return project_sum(a2, n)


def lp_direction(a, max_seconds=6):
    """
    Optional LP direction:
        maximize sum(b)  s.t.  (a*b)_k <= M, b >= 0.

    This is the linear direction idea from the previous SOTA.  It is used
    only rarely and only for moderate lengths, because the constraint matrix
    is dense in a banded/hankel sense.
    """
    if not _HAS_SCIPY:
        return None

    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n < 16 or n > 2048:
        return None

    aa = a.copy()
    orient = False
    if aa[0] < aa[-1]:
        aa = aa[::-1].copy()
        orient = True

    aa = project_sum(aa, n)
    c = fast_full_conv(aa)
    M = float(c.max()) if c.size else 0.0
    if M <= 1e-15:
        return None

    try:
        cols = np.repeat(np.arange(n, dtype=np.int32), n)
        offsets = np.tile(np.arange(n, dtype=np.int32), n)
        rows = cols + offsets
        data = np.tile(aa, n)

        A = csc_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        B = max(1000.0, 20.0 * M / max(1e-12, float(aa[0])))
        B = min(B, 1.0e6)

        bounds = np.empty((n, 2), dtype=np.float64)
        bounds[:, 0] = 0.0
        bounds[:, 1] = float(B)

        res = linprog(
            -np.ones(n, dtype=np.float64),
            A_ub=A,
            b_ub=np.full(2 * n - 1, M, dtype=np.float64),
            bounds=bounds,
            method="highs",
            options={"time_limit": float(max_seconds), "presolve": True},
        )

        if res is None or not getattr(res, "success", False):
            return None

        b = np.asarray(res.x, dtype=np.float64).ravel()
        b = np.nan_to_num(b, nan=0.0, posinf=0.0, neginf=0.0)
        if b.size != n or float(b.sum()) <= 1e-12:
            return None

        if orient:
            b = b[::-1].copy()

        b = project_sum(b, n)
        if float(b.sum()) <= 1e-12:
            return None
        return b
    except Exception:
        return None


def run_local(a0, end_time, rng,
              init_lr=0.1,
              active_tol=0.2,
              temp_frac=0.1,
              maxK=None,
              minK=64,
              patience=150,
              use_lp=False,
              lp_seconds=6):
    """
    Local optimizer on the sum=n manifold.

    Adam on the projected gradient of a smooth weighted-peak surrogate.
    Restarts with low-frequency perturbations and optionally LP directions.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    if n == 0:
        return np.inf, a0

    if maxK is None:
        maxK = int(min(4096, max(128, n // 4)))

    a = project_sum(a0.copy(), n)
    c = fast_full_conv(a)
    S = float(a.sum())
    score = float(2.0 * n * float(c.max()) / (S * S))

    best_score = score
    best_a = a.copy()

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    no_improve = 0
    cur_lr = float(init_lr)
    cur_tol = float(active_tol)
    cur_temp = float(temp_frac)
    cur_maxK = int(maxK)
    cur_minK = int(minK)

    restarts = 0
    last_lp = 0.0
    lp_calls = 0

    while time.monotonic() < end_time:
        if not np.isfinite(score):
            break

        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            improved = False

            if (use_lp and _HAS_SCIPY and n <= 2048 and lp_calls < 3 and
                    (time.monotonic() - last_lp) > max(15.0, 0.1 * patience)):
                b_dir = lp_direction(best_a, lp_seconds)
                last_lp = time.monotonic()
                lp_calls += 1
                if b_dir is not None:
                    for tb in (0.5, 0.3, 0.2, 0.1, 0.05, 0.02):
                        cand = project_sum((1.0 - tb) * best_a + tb * b_dir, n)
                        sc = objective_value(cand)
                        if np.isfinite(sc) and sc < best_score:
                            best_score = sc
                            best_a = cand.copy()
                            a = cand
                            c = fast_full_conv(a)
                            S = float(a.sum())
                            score = sc
                            no_improve = 0
                            improved = True
                            print("  LP direction improvement:", best_score, flush=True)
                            break

            if not improved:
                scale = 0.08 + 0.18 * rng.random()
                a = perturb(best_a, rng, scale)
                m.fill(0.0)
                v.fill(0.0)
                t = 0
                no_improve = 0
                cur_lr = max(0.005, cur_lr * 0.8)

                if rng.random() < 0.7:
                    cur_tol = max(0.03, min(0.5, cur_tol * 0.75))
                    cur_temp = max(0.02, min(0.4, cur_temp * 0.75))
                else:
                    cur_tol = min(0.5, cur_tol * 1.4)
                    cur_temp = min(0.4, cur_temp * 1.4)

                cur_maxK = int(min(10000, max(64, cur_maxK * (1.2 if rng.random() < 0.6 else 0.8))))
                patience = max(60, patience // 2)
                restarts += 1

                if patience == 60 and restarts > 10:
                    cur_tol = 0.3
                    cur_temp = 0.25
                    patience = 150

        M = float(c.max()) if c.size else 0.0
        if M <= 1e-15:
            break

        idx, w = peak_weights(c, cur_tol, cur_maxK, cur_minK, cur_temp)
        g = grad_fft_weights(a, idx, w)
        g = g - g.mean()

        gnorm = float(np.linalg.norm(g))
        if not np.isfinite(gnorm) or gnorm <= 1e-13:
            a = perturb(best_a, rng, 0.05)
            c = fast_full_conv(a)
            S = float(a.sum())
            score = float(2.0 * n * float(c.max()) / (S * S))
            m.fill(0.0)
            v.fill(0.0)
            t = 0
            no_improve = min(no_improve + 1, patience // 2)
            continue

        t += 1
        b1 = 0.9
        b2 = 0.999

        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * (g * g)

        bias1 = b1 ** t
        bias2 = b2 ** t

        mh = m / (1.0 - bias1)
        vh = v / (1.0 - bias2)
        step = mh / (np.sqrt(vh) + 1e-8)

        mx = float(np.max(np.abs(step)))
        if mx > 0.5:
            step = step * (0.5 / mx)

        a_new = a - cur_lr * step
        a = project_sum(a_new, n)

        c = fast_full_conv(a)
        S = float(a.sum())
        score = float(2.0 * n * float(c.max()) / (S * S))

    if np.isfinite(score) and score < best_score:
        best_score = score
        best_a = a.copy()

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best nonnegative height sequence within the wall-clock budget.
    Returns a Python list of floats.
    """
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()
    total = max(60.0, float(budget_s))
    deadline = t0 + total - 10.0

    parent = get_parent_sequence()

    if parent is not None and parent.size >= 512:
        base_n = int(parent.size)
    else:
        base_n = 2048
    base_n = int(min(max(base_n, 512), 8192))

    if parent is not None:
        base_a = resize_sequence(parent, base_n, rng)
    else:
        base_a = np.ones(base_n, dtype=np.float64)
    base_a = project_sum(base_a, base_n)

    best_score = objective_value(base_a)
    best_a = base_a.copy()
    if not np.isfinite(best_score):
        best_a = np.ones(base_n, dtype=np.float64)
        best_score = objective_value(best_a)
    print("parent/base n=%d score=%.9f" % (base_n, best_score), flush=True)

    # ------------------------------------------------------------
    # Build and rank many initializations at the base length.
    # ------------------------------------------------------------
    inits = build_initials(base_n, rng, parent)
    cand = []
    for a in inits:
        sc = objective_value(a)
        if np.isfinite(sc):
            cand.append((sc, a))
    cand.append((best_score, best_a))
    cand.sort(key=score_key)

    # ------------------------------------------------------------
    # Phase 1: short exploration from the best-ranked starts.
    # ------------------------------------------------------------
    if total < 600:
        phase1_end = t0 + 0.12 * total
        num_quick = min(3, len(cand))
    elif total < 1200:
        phase1_end = t0 + 0.15 * total
        num_quick = min(5, len(cand))
    else:
        phase1_end = t0 + 0.20 * total
        num_quick = min(8, len(cand))

    phase1_end = min(phase1_end, deadline - 60.0)
    if num_quick > 0 and phase1_end > time.monotonic() + 15.0:
        phase_start1 = time.monotonic()
        slot1 = (phase1_end - phase_start1) / float(num_quick)
        for i in range(num_quick):
            if phase_start1 + slot1 < time.monotonic() + 5.0:
                break
            end = phase_start1 + (i + 1) * slot1
            a0 = cand[i][1]
            sc, a2 = run_local(
                a0, end, rng,
                init_lr=0.25,
                active_tol=0.30,
                temp_frac=0.20,
                maxK=int(min(2048, max(128, base_n // 4))),
                minK=64,
                patience=120,
                use_lp=False,
            )
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a2.copy()
                print("  quick %d -> %.9f" % (i, best_score), flush=True)
            cand.append((sc, a2))

    cand.sort(key=score_key)
    print("after phase1 best=%.9f" % best_score, flush=True)

    # ------------------------------------------------------------
    # Phase 2: longer refinement from a few diverse candidates.
    # ------------------------------------------------------------
    phase2_end = min(t0 + 0.55 * total, deadline - 90.0)
    if time.monotonic() < phase2_end - 30.0 and len(cand) > 0:
        sel = [0]
        if len(cand) > 2:
            sel.append(2)
        if total > 1200 and len(cand) > 4:
            sel.append(4)

        sel = [i for i in sel if 0 <= i < len(cand)]
        phase2_cands = [cand[i][1] for i in sel]

        phase_start2 = time.monotonic()
        if phase2_cands:
            slot2 = (phase2_end - phase_start2) / float(len(phase2_cands))
            for i, a0 in enumerate(phase2_cands):
                if phase_start2 + slot2 < time.monotonic() + 10.0:
                    break
                end = phase_start2 + (i + 1) * slot2
                sc, a2 = run_local(
                    a0, end, rng,
                    init_lr=0.10,
                    active_tol=0.15,
                    temp_frac=0.10,
                    maxK=int(min(4096, max(256, base_n // 2))),
                    minK=128,
                    patience=300,
                    use_lp=(base_n <= 2048 and total > 900),
                    lp_seconds=8,
                )
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a2.copy()
                    print("  phase2 %d -> %.9f" % (i, best_score), flush=True)
                cand.append((sc, a2))

    cand.sort(key=score_key)
    print("after phase2 best=%.9f" % best_score, flush=True)

    # ------------------------------------------------------------
    # Phase 3: try a longer sequence if there is time.
    # ------------------------------------------------------------
    if base_n < 8192 and time.monotonic() < deadline - 180.0:
        large_n = min(8192, base_n * 2)
        if large_n < 2048:
            large_n = 2048
        phase3_end = min(t0 + 0.85 * total, deadline - 100.0)

        if phase3_end > time.monotonic() + 60.0:
            a_large = resize_sequence(best_a, large_n, rng)
            rem = phase3_end - time.monotonic()

            end_best = time.monotonic() + 0.75 * rem
            sc, aL = run_local(
                a_large, end_best, rng,
                init_lr=0.10,
                active_tol=0.15,
                temp_frac=0.10,
                maxK=int(min(4096, max(256, large_n // 2))),
                minK=128,
                patience=250,
                use_lp=False,
            )
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = aL.copy()
                print("  large best n=%d -> %.9f" % (large_n, best_score), flush=True)

            if len(cand) > 1 and phase3_end > time.monotonic() + 60.0:
                a_other = resize_sequence(cand[1][1], large_n, rng)
                sc2, aL2 = run_local(
                    a_other, phase3_end, rng,
                    init_lr=0.10,
                    active_tol=0.20,
                    temp_frac=0.15,
                    maxK=int(min(4096, max(256, large_n // 2))),
                    minK=128,
                    patience=200,
                    use_lp=False,
                )
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_a = aL2.copy()
                    print("  large alt n=%d -> %.9f" % (large_n, best_score), flush=True)

    print("after phase3 best=%.9f len=%d" % (best_score, best_a.size), flush=True)

    # ------------------------------------------------------------
    # Final polish.
    # ------------------------------------------------------------
    final_end = deadline - 2.0
    if time.monotonic() < final_end - 30.0:
        nbest = int(best_a.size)
        sc, aF = run_local(
            best_a, final_end, rng,
            init_lr=0.04,
            active_tol=0.08,
            temp_frac=0.05,
            maxK=int(min(8192, max(512, nbest // 2))),
            minK=256,
            patience=800,
            use_lp=(nbest <= 2048 and total > 1200),
            lp_seconds=5,
        )
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = aF.copy()

    out = project_sum(np.asarray(best_a, dtype=np.float64).ravel(), best_a.size)
    if out.size == 0:
        out = np.ones(base_n, dtype=np.float64)

    # Ensure the evaluator's positivity sum guard.
    if float(out.sum()) < 0.01:
        out = out + (0.01 - float(out.sum())) / float(max(1, out.size))
    out = np.clip(out, 0.0, 1000.0)

    out_list = [float(x) for x in out]

    try:
        verified = float(evaluate_sequence(out_list))
    except Exception:
        verified = np.nan

    print("final internal=%.9f verified=%s len=%d elapsed=%.1fs" %
          (best_score, verified, len(out_list), time.monotonic() - t0), flush=True)

    if not np.isfinite(verified):
        fallback = [1.0] * int(base_n)
        try:
            verified2 = float(evaluate_sequence(fallback))
        except Exception:
            verified2 = np.nan
        print("fallback verified=%s" % verified2, flush=True)
        return fallback

    return out_list
```
