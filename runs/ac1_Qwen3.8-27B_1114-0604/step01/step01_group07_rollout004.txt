import numpy as np
import time
from scipy.optimize import linprog

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    mx = float(np.max(c))
    return float(2.0 * arr.size * mx / (s * s))

def _solve_lp_direction(f, M):
    """
    Solve: max sum(b) s.t. b >= 0, (f*b)(k) <= M for all k.
    Returns the optimal b.
    """
    n = len(f)
    # Constraint matrix A: shape (2n-1, n)
    # A[k, i] = f[i] if 0 <= k-i < n, else 0
    # i.e., A[k, i] = f[i] if max(0, k-n+1) <= i <= min(n-1, k)
    
    # For efficiency, use a sparse matrix
    from scipy.sparse import csr_matrix
    
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        for i in range(i_min, i_max + 1):
            rows.append(k)
            cols.append(i)
            vals.append(f[i])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # linprog minimizes c^T x subject to A_ub x <= b_ub, bounds
    c = -np.ones(n)  # minimize -sum(x) = maximize sum(x)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x
    else:
        return None

Low-peak nonnegative step-function search using LP-based refinement and
subgradient descent on the simplex.

import numpy as np
import time
from scipy.sparse import csr_matrix
from scipy.optimize import linprog


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    mx = float(np.max(c))
    return float(2.0 * arr.size * mx / (s * s))


def _make_sine_candidate(n, amp, mode, phase=0.5):

    a = (1.0 + float(amp) * np.sin(float(mode) * theta)) / float(n)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        a = np.ones(n) / n
    else:

    return a

def _lp_direction(f, M):
    """
    Solve LP: max sum(b) s.t. b >= 0, (f*b)(k) <= M for all k.
    Returns optimal b or None.
    """
    n = len(f)
    n_constr = 2 * n - 1
    
    # Build constraint matrix A (n_constr x n)
    # A[k, j] = f[j] if max(0,k-n+1) <= j <= min(n-1,k), else 0
    rows = np.empty(n * n, dtype=np.int32)  # upper bound on nnz
    cols = np.empty(n * n, dtype=np.int32)
    vals = np.empty(n * n, dtype=np.float64)
    cnt = 0
    for k in range(n_constr):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        length = j_max - j_min + 1
        rows[cnt:cnt+length] = k
        cols[cnt:cnt+length] = np.arange(j_min, j_max + 1)
        vals[cnt:cnt+length] = f[j_min:j_max+1]
        cnt += length
    
    A = csr_matrix((vals[:cnt], (rows[:cnt], cols[:cnt])), shape=(n_constr, n))
    
    c_obj = -np.ones(n)
    b_ub = np.full(n_constr, M)
    bounds = [(0, None)] * n
    
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'time_limit': 10})
    
    if res.status == 0:
        return res.x
    return None


def _lp_refine(a, budget_s):
    """
    LP-based refinement: at each step, solve LP to find best direction,
    then take a line-search step.
    """
    start = time.monotonic()
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    arr = arr / s
    n = len(arr)
    
    best = arr.copy()
    best_score = _internal_score(best)
    
    it = 0
    no_improve = 0
    
    while time.monotonic() - start < budget_s:
        c = _fft_convolve(arr)
        M = float(np.max(c))
        if not np.isfinite(M) or M <= 0:
            break
        
        # Solve LP for direction
        if time.monotonic() - start > budget_s:
            break
        t_lp_start = time.monotonic()
        g0 = _lp_direction(arr, M)
        t_lp = time.monotonic() - t_lp_start
        
        if g0 is None:
            break
        
        s_g = float(g0.sum())
        if s_g < 1e-10:
            break
        g = g0 / s_g  # normalize to sum 1
        
        # Line search: try different step sizes
        best_t = 0.0
        best_sc = best_score
        
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01]:
            candidate = (1 - t) * arr + t * g
            candidate = np.clip(candidate, 0.0, 1000.0)
            sc = _internal_score(candidate)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_t = t
        
        if best_t > 0:
            arr = (1 - best_t) * arr + best_t * g
            arr = np.clip(arr, 0.0, 1000.0)
            arr = arr / arr.sum()
            
            if best_sc < best_score:
                best = arr.copy()
                best_score = best_sc
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        it += 1
        if it % 20 == 0:
            print(f"  [lp] it={it} score={best_score:.8f} last_lp={t_lp:.2f}s")
        
        if no_improve > 20:
            break
    
    print(f"  [lp] done: it={it} best_score={best_score:.8f} time={time.monotonic()-start:.1f}s")
    return best, best_score


def _grad_refine(a, budget_s):
    """
    Subgradient descent on the simplex.
    """
    start = time.monotonic()
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.clip(arr, 0.0, 1000.0)
    arr = arr / arr.sum()
    n = len(arr)
    
    best = arr.copy()
    best_score = _internal_score(best)
    
    lr = 0.01
    momentum = 0.9
    momentum_buf = np.zeros(n)
    no_improve = 0
    it = 0
    
    while time.monotonic() - start < budget_s:
        c = _fft_convolve(arr)
        M = float(np.max(c))
        
        # Find active set (top K)
        K = min(16, 2 * n - 1)
        active = np.argpartition(c, -K)[-K:]
        
        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            j_min = max(0, k - n + 1)
            j_max = min(n - 1, k)
            partners = k - np.arange(j_min, j_max + 1)
            grad[j_min:j_max+1] += 2.0 * arr[partners]
        grad /= float(K)
        
        # Project to tangent plane of simplex
        grad = grad - np.mean(grad)
        
        # Momentum
        momentum_buf = momentum * momentum_buf + (1 - momentum) * grad
        
        # Step
        arr_new = arr - lr * momentum_buf
        arr_new = np.clip(arr_new, 0.0, 1000.0)
        s_new = arr_new.sum()
        if s_new < 0.01:
            arr_new = arr.copy()
        else:
            arr_new = arr_new / s_new
        
        sc = _internal_score(arr_new)
        if np.isfinite(sc) and sc < best_score:
            best = arr_new.copy()
            best_score = sc
            no_improve = 0
            lr = min(0.05, lr * 1.02)
        else:
            arr_new = arr  # revert
            no_improve += 1
            if sc > best_score * 1.5:
                lr = max(0.001, lr * 0.8)
        
        arr = arr_new
        it += 1
        
        if it % 100 == 0:
            elapsed = time.monotonic() - start
            print(f"  [grad] it={it} score={best_score:.8f} lr={lr:.5f} time={elapsed:.1f}s")
        
        if no_improve > 200:
            break
    
    print(f"  [grad] done: it={it} best_score={best_score:.8f} time={time.monotonic()-start:.1f}s")
    return best, best_score
