```python
import numpy as np
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _grad_smoothed(a, b, m):
    """Compute gradient of log-sum-exp smoothed max of b.
    Returns g where g_i = 2 * sum_k p_k * a_{k-i} (valid indices).
    """
    n = len(a)
    # p_k proportional to exp(m * b_k)
    mb = m * b
    mb = mb - np.max(mb)  # for numerical stability
    p = np.exp(mb)
    p = p / p.sum()
    
    # g_i = 2 * sum_j a_j * p_{i+j} for j=0..n-1, i=0..n-1
    # = 2 * cross_correlate(a, p) at lag i
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    cc = np.fft.irfft(fa * np.conj(fp), N)
    g = 2.0 * cc[:n]
    return g

def _mirror_descent_step(a, g, lr):
    """Multiplicative weight update on the simplex."""
    a_new = a * np.exp(-lr * g)
    a_new = np.where(np.isfinite(a_new), a_new, 0.0)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 1e-300:
        return a
    return a_new / s

def _internal_score(a):
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = _fft_convolve(a)
    return 2.0 * len(a) * b.max() / (s * s)

    best_score = float(evaluate_sequence(a.tolist()))
    print(f"Initial: score={best_score:.8f}")
    
    # Multi-temperature mirror descent
    temperatures = [5, 10, 20, 50, 100, 200, 500, 1000, 2000]
    
    for m_val in temperatures:
        if time.monotonic() > hard_deadline:
            break
        
        # Learning rate schedule
        lr = 0.5 * n / m_val  # heuristic
        best_lr = lr
        
        current_a = best_a.copy()
        iter_count = 0
        plateaus = 0
        last_best_score = best_score
        no_improve_iters = 0
        
        max_iters = 50000
        check_interval = 100
        
        while iter_count < max_iters:
            if time.monotonic() > hard_deadline:
                break
            iter_count += 1
            
            b = _fft_convolve(current_a)
            g = _grad_smoothed(current_a, b, m_val)
            
            # Normalize gradient to have unit norm (or some scale)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-15:
                break
            
            # Mirror descent step
            current_a = _mirror_descent_step(current_a, g, lr)
            
            # Periodic check
            if iter_count % check_interval == 0:
                sc = _internal_score(current_a)
                if sc < last_best_score:
                    last_best_score = sc
                    no_improve_iters = 0
                    # Slightly increase lr
                    lr = min(lr * 1.01, 2.0 * n)
                else:
                    no_improve_iters += 1
                    if no_improve_iters >= 20:
                        # Reduce lr
                        lr = max(lr * 0.7, 0.01 * n)
                        no_improve_iters = 0
                        if lr <= 0.01 * n:
                            break
            
        # After inner loop, check if we improved
        sc = _internal_score(current_a)
        if sc < best_score:
            # Validate with actual evaluator
            real_sc = float(evaluate_sequence(current_a.tolist()))
            if real_sc < best_score:
                best_score = real_sc
                best_a = current_a.copy()
                print(f"m={m_val}: improved to {best_score:.8f} (iter={iter_count})")
    
    # Final validation
    final_score = float(evaluate_sequence(best_a.tolist()))
    print(f"Final: score={final_score:.8f}")
    
    return [float(x) for x in best_a]
```
