def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    rho = np.where(cond)[0]
    if len(rho) == 0:
        return np.ones(n, dtype=np.float64) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _score(a):
    """Compute 2*n*max_conv / sum^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    mx = c.max()
    return 2.0 * n * mx / (s * s)

def _score_with_conv(a, c):
    """Score given precomputed convolution."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    mx = c.max()
    return 2.0 * n * mx / (s * s)

def _make_sine(n, mode, amp):
    """Create sinusoidal anti-palindromic sequence."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    a = 1.0 + float(amp) * np.sin(float(mode) * theta)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    return a

def _optimize_amplitude(n, mode, deadline):
    """Find best amplitude for given n and mode."""
    best_sc = np.inf
    best_amp = 0.7
    amps = np.linspace(0.0, 1.5, 31)
    for amp in amps:
        if time.monotonic() > deadline:
            break
        a = _make_sine(n, mode, amp)
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_amp = amp
    # Refine
    for _ in range(4):
        lo = max(0.0, best_amp - 1.5/30.0)
        hi = min(1.5, best_amp + 1.5/30.0)
        amps = np.linspace(lo, hi, 11)
        for amp in amps:
            if time.monotonic() > deadline:
                break
            a = _make_sine(n, mode, amp)
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_amp = amp
    return _make_sine(n, mode, best_amp), best_sc

def _variance_grad_desc(a, max_iters=5000, deadline=np.inf):
    """Minimize variance of self-convolution via projected gradient."""
    n = len(a)
    a = a.copy()
    lr = 0.05
    best_score = _score(a)
    best_a = a.copy()
    
    N = _next_pow2(3 * n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        # Gradient of sum(c^2) = 4 * (c ⋆ a)[j] for j in [0, n-1]
        C = np.fft.rfft(c, N)
        A = np.fft.rfft(a, N)
        corr = np.fft.irfft(C * np.conj(A), N)
        grad = 4.0 * corr[:n]
        
        # Projected gradient step
        a_new = _project_simplex(a - lr * grad)
        
        # Line search (simple): evaluate and accept/reject
        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            lr = min(lr * 1.05, 0.5)
        else:
            lr = max(lr * 0.8, 0.001)
            a = a_new  # still move, but with smaller effective step next time
    
    return best_a, best_score

def _active_set_subgrad(a, K=8, max_iters=5000, deadline=np.inf):
    """Minimize max of self-convolution via active-set subgradient."""
    n = len(a)
    a = a.copy()
    lr = 1.0 / np.sqrt(n)
    best_score = _score(a)
    best_a = a.copy()
    last_improve = 0
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        mx = c.max()
        
        # Find active set (top K lags)
        if c.size <= K:
            active = np.arange(c.size)
        else:
            active = np.argpartition(c, -K)[-K:]
        
        # Compute subgradient
        grad = np.zeros(n, dtype=np.float64)
        for k in active:
            k = int(k)
            # grad[j] += 2 * a[k-j] if 0 <= k-j <= n-1
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, k)
            if j_max >= j_min:
                # grad[j] += 2 * a[k-j] for j in [j_min, j_max]
                # a[k-j] for j in [j_min, j_max] is a[k-j_min] down to a[k-j_max]
                # = a[min(n-1, k-n+1) ... wait]
                # a[k-j] for j = j_min: a[k - j_min] = a[min(n-1, k)] (since j_min = max(0, k-n+1))
                # a[k-j] for j = j_max: a[k - j_max] = a[max(0, k-n+1)] = a[0] (if j_max = k) or a[n-1] (if j_max = n-1)
                # So a[k-j] for j = j_min..j_max is a reversed slice of a
                idx = k - np.arange(j_min, j_max + 1)
                grad[j_min:j_max+1] += 2.0 * a[idx]
        
        grad /= float(len(active))
        
        # Step
        a = _project_simplex(a - lr * grad)
        
        # Adaptive lr
        if (it + 1) % 50 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
                lr = min(lr * 1.02, 1.0)
            else:
                lr = max(lr * 0.95, 0.001)
        
        # Decaying lr
        lr = lr * (1.0 - 1.0 / (it + 100.0))
        
        if it - last_improve > 500 and it > 100:
            break
    
    return best_a, best_score

def _soft_max_grad_desc(a, T_init=0.1, T_final=0.01, max_iters=3000, deadline=np.inf):
    """Minimize soft-max of self-convolution (smooth approximation of max)."""
    n = len(a)
    a = a.copy()
    c = _fft_convolve(a)
    base = c.max()
    T = T_init * base
    lr = 0.01
    best_score = _score(a)
    best_a = a.copy()
    last_improve = 0
    
    N_corr = _next_pow2(3 * n)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        mx = c.max()
        T = max(T_final * mx, 1e-10)
        
        # Soft-max weights
        shifted = c - mx  # for numerical stability
        weights = np.exp(shifted / T)
        wsum = weights.sum()
        weights /= wsum
        
        # Gradient: sum_k weights[k] * 2*a[k-j] (if valid)
        # = 2 * (weights ⋆ a)[j]
        W = np.fft.rfft(weights, N_corr)
        A = np.fft.rfft(a, N_corr)
        corr = np.fft.irfft(W * np.conj(A), N_corr)
        grad = 2.0 * corr[:n]
        
        a_new = _project_simplex(a - lr * grad)
        
        sc = _score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            last_improve = it
            lr = min(lr * 1.01, 0.1)
        else:
            a = a_new
            lr = max(lr * 0.97, 0.001)
        
        if it - last_improve > 300 and it > 50:
            break
    
    return best_a, best_score

The cross-correlation (as defined in signal processing) is:
`(x ⋆ y)[n] = sum_{k} x[k] * conj(y[k-n])`

For real sequences: `(x ⋆ y)[n] = sum_{k} x[k] * y[k-n]`

The DFT of the circular cross-correlation:
`DFT(x ⋆ y)[m] = DFT(x)[m] * conj(DFT(y)[m])`

This is a standard result. So:
`x ⋆ y = IDFT(DFT(x) * conj(DFT(y)))`

In numpy:

"""
Low-peak nonnegative step-function search using projected gradient methods.

Algorithm:
1. Initialize with anti-palindromic sinusoidal shapes (multiple modes/lengths).
2. Smooth phase: minimize variance of self-convolution via projected gradient (FFT-based).
3. Sharp phase: soft-max gradient descent for smooth max approximation.
4. Active-set subgradient refinement.
5. Random perturbation local search.
6. Validate with evaluate_sequence.
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u - (cssv - 1.0) / rho > 0
    idx = np.where(cond)[0]
    if len(idx) == 0:
        return np.ones(n, dtype=np.float64) / n
    rho_val = idx[-1]
    theta = (cssv[rho_val] - 1.0) / (rho_val + 1.0)
    return np.maximum(v - theta, 0.0)


def _score(a):
    """Compute 2*n*max_conv / sum^2."""
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    mx = float(c.max())
    return 2.0 * n * mx / (s * s)


def _make_sine(n, mode, amp, taper=0.0):
    """Create sinusoidal anti-palindromic sequence with optional symmetric taper."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    a = 1.0 + float(amp) * np.sin(float(mode) * theta)
    if taper > 0.0:
        # Symmetric (palindromic) taper: product with anti-palindromic stays anti-palindromic
        t = 0.5 * (1.0 + np.cos(np.pi * (i - (n - 1) / 2.0) / ((n - 1) / 2.0 + 1e-10)))
        a = a * (1.0 - taper + taper * t)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def _optimize_amplitude(n, mode, deadline, taper=0.0):
    """Find best amplitude for given n, mode, taper."""
    best_sc = np.inf
    best_amp = 0.7
    amps = np.linspace(0.0, 1.5, 31)
    for amp in amps:
        if time.monotonic() > deadline:
            break
        a = _make_sine(n, mode, amp, taper)
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_amp = amp
    # Refine
    step = 1.5 / 30.0
    for _ in range(5):
        lo = max(0.0, best_amp - step)
        hi = min(1.5, best_amp + step)
        amps = np.linspace(lo, hi, 11)
        for amp in amps:
            if time.monotonic() > deadline:
                break
            a = _make_sine(n, mode, amp, taper)
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_amp = amp
        step *= 0.4
    return _make_sine(n, mode, best_amp, taper), best_sc


def _variance_grad_desc(a, max_iters=20000, deadline=np.inf):
    """Minimize variance of self-convolution via projected gradient (FFT-based)."""
    n = len(a)
    a = a.astype(np.float64).copy()
    lr = 0.02
    best_score = _score(a)
    best_a = a.copy()
    last_improve = 0

    N = _next_pow2(3 * n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)
        # Gradient of sum(c^2): grad[j] = 4 * (c cross_corr a)[j]
        C = np.fft.rfft(c, N)
        A = np.fft.rfft(a, N)
        corr = np.fft.irfft(C * np.conj(A), N)
        grad = 4.0 * corr[:n]

        a = _project_simplex(a - lr * grad)

        if (it + 1) % 100 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
                lr = min(lr * 1.03, 0.3)
            else:
                lr = max(lr * 0.9, 0.001)

        if it - last_improve > 2000 and it > 200:
            break

    # Final check
    sc = _score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def _soft_max_grad_desc(a, max_iters=15000, deadline=np.inf):
    """Minimize soft-max of self-convolution (smooth max approximation)."""
    n = len(a)
    a = a.astype(np.float64).copy()
    c = _fft_convolve(a)
    base = max(float(c.max()), 1e-10)
    T_ratio_init = 0.15
    T_ratio_final = 0.005
    lr = 0.005
    best_score = _score(a)
    best_a = a.copy()
    last_improve = 0

    N_corr = _next_pow2(3 * n)

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        # Anneal temperature
        frac = it / max(iters_left, 1) if (iters_left := max_iters) > 0 else 1.0
        T_ratio = T_ratio_init + (T_ratio_final - T_ratio_init) * min(frac, 1.0)

        c = _fft_convolve(a)
        mx = max(float(c.max()), 1e-15)
        T = T_ratio * mx

        # Soft-max weights (numerically stable)
        shifted = c - mx
        weights = np.exp(shifted / T)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.zeros_like(c)
            weights[np.argmax(c)] = 1.0
        else:
            weights /= wsum

        # Gradient: sum_k weights[k] * 2*a[k-j] = 2 * (weights cross_corr a)[j]
        W = np.fft.rfft(weights, N_corr)
        A = np.fft.rfft(a, N_corr)
        corr = np.fft.irfft(W * np.conj(A), N_corr)
        grad = 2.0 * corr[:n]

        a = _project_simplex(a - lr * grad)

        if (it + 1) % 200 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
                lr = min(lr * 1.02, 0.05)
            else:
                lr = max(lr * 0.95, 0.0005)

        if it - last_improve > 2000 and it > 200:
            break

    sc = _score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def _active_set_subgrad(a, K=10, max_iters=15000, deadline=np.inf):
    """Minimize max of self-convolution via active-set subgradient method."""
    n = len(a)
    a = a.astype(np.float64).copy()
    lr = 1.0 / max(np.sqrt(n), 1.0)
    best_score = _score(a)
    best_a = a.copy()
    last_improve = 0

    for it in range(max_iters):
        if time.monotonic() > deadline:
            break

        c = _fft_convolve(a)

        # Active set: top K lags
        if c.size <= K:
            active = np.arange(c.size)
        else:
            active = np.argpartition(c, -K)[-K:]

        # Subgradient: average of gradients at active lags
        # grad[j] = (1/|K|) * sum_{k in active} 2*a[k-j] (if 0<=k-j<=n-1)
        grad = np.zeros(n, dtype=np.float64)
        for k in active:
            k = int(k)
            j_min = max(0, k - (n - 1))
            j_max = min(n - 1, k)
            if j_max >= j_min:
                idx = k - np.arange(j_min, j_max + 1)
                grad[j_min:j_max + 1] += 2.0 * a[idx]
        grad /= float(len(active))

        # Step with projection
        a = _project_simplex(a - lr * grad)

        # Evaluate periodically
        if (it + 1) % 100 == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
                lr = min(lr * 1.01, 2.0)
            else:
                lr = max(lr * 0.97, 0.005)

        # Decay
        lr *= (1.0 - 0.5 / (it + 50.0))

        if it - last_improve > 2000 and it > 200:
            break

    sc = _score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def _perturbation_search(a, max_iters=50000, deadline=np.inf, seed=42):
    """Random perturbation local search."""
    n = len(a)
    a = a.astype(np.float64).copy()
    best_score = _score(a)
    best_a = a.copy()
    rng = np.random.RandomState(seed)
    scale = 0.005
    count = 0

    while time.monotonic() < deadline and count < max_iters:
        # Generate perturbation
        p = rng.randn(n)
        p -= p.mean()
        p_std = p.std()
        if p_std < 1e-10:
            continue
        p = p / p_std

        a_new = a + scale * p
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 0.01:
            scale *= 0.9
            continue
        a_new = a_new / s

        sc = _score(a_new)
        count += 1

        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()
            a = a_new
            scale = min(scale * 1.05, 0.05)
        else:
            scale = max(scale * 0.995, 0.001)

        if count % 10000 == 0:
            print(f"  [perturb] iter={count} score={best_score:.8f} scale={scale:.5f}")

    return best_a, best_score


def _multi_harmonic_init(n, deadline):
    """Try multi-harmonic shapes."""
    best_a = None
    best_sc = np.inf
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    s1 = np.sin(theta)
    s2 = np.sin(2.0 * theta)
    s3 = np.sin(3.0 * theta)

    for a1 in [0.5, 0.7, 0.8]:
        for a2 in [0.0, 0.1, 0.2, -0.1]:
            if time.monotonic() > deadline:
                break
            a = 1.0 + a1 * s1 + a2 * s2
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 0.01:
                continue
            a = a / s
            sc = _score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a low evaluation score and return the best valid sequence found.

    Strategy:
    1. Sinusoidal initialization (multiple modes, lengths, tapers)
    2. Variance-based gradient descent (smooth phase)
    3. Soft-max gradient descent (transitional)
    4. Active-set subgradient (sharp phase)
    5. Random perturbation local search
    6. Final polish
    """
    t0 = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    hard_deadline = t0 + total_budget - 30.0

    best_a = None
    best_score = np.inf

    # Phase 1: Initialization
    init_deadline = t0 + min(120.0, 0.1 * total_budget)
    print(f"[Phase 1] Initializing (deadline at {init_deadline - t0:.0f}s)")

    for n in [4096, 2048, 8192]:
        if time.monotonic() > init_deadline:
            break
        for mode in [1, 2, 3]:
            if time.monotonic() > init_deadline:
                break
            for taper in [0.0, 0.3]:
                if time.monotonic() > init_deadline:
                    break
                a, sc = _optimize_amplitude(n, mode, init_deadline, taper)
                if sc < best_score:
                    best_score = sc
                    best_a = a
                    print(f"  n={n} mode={mode} taper={taper} score={sc:.8f}")

    # Multi-harmonic
    if time.monotonic() < init_deadline:
        a, sc = _multi_harmonic_init(4096, init_deadline)
        if a is not None and sc < best_score:
            best_score = sc
            best_a = a
            print(f"  multi-harmonic n=4096 score={sc:.8f}")

    if best_a is None:
        best_a = _make_sine(2048, 1, 0.7)
        best_score = _score(best_a)
        print(f"  Fallback sinusoidal score={best_score:.8f}")

    # Phase 2: Variance gradient descent
    remaining = hard_deadline - time.monotonic()
    if remaining > 180:
        p2_deadline = t0 + 0.30 * total_budget
        print(f"[Phase 2] Variance gradient descent (deadline at {p2_deadline - t0:.0f}s)")
        a, sc = _variance_grad_desc(best_a, max_iters=30000, deadline=p2_deadline)
        if sc < best_score:
            best_score = sc
            best_a = a
            print(f"  Improved to {sc:.8f}")
        else:
            print(f"  No improvement (best={best_score:.8f})")

    # Phase 3: Soft-max gradient descent
    remaining = hard_deadline - time.monotonic()
    if remaining > 120:
        p3_deadline = t0 + 0.50 * total_budget
        print(f"[Phase 3] Soft-max gradient descent (deadline at {p3_deadline - t0:.0f}s)")
        a, sc = _soft_max_grad_desc(best_a, max_iters=20000, deadline=p3_deadline)
        if sc < best_score:
            best_score = sc
            best_a = a
            print(f"  Improved to {sc:.8f}")
        else:
            print(f"  No improvement (best={best_score:.8f})")

    # Phase 4: Active-set subgradient
    remaining = hard_deadline - time.monotonic()
    if remaining > 90:
        p4_deadline = t0 + 0.70 * total_budget
        print(f"[Phase 4] Active-set subgradient (deadline at {p4_deadline - t0:.0f}s)")
        a, sc = _active_set_subgrad(best_a, K=10, max_iters=20000, deadline=p4_deadline)
        if sc < best_score:
            best_score = sc
            best_a = a
            print(f"  Improved to {sc:.8f}")
        else:
            print(f"  No improvement (best={best_score:.8f})")

    # Phase 5: Random perturbation
    remaining = hard_deadline - time.monotonic()
    if remaining > 60:

        a, sc = _perturbation_search(best_a, max_iters=100000, deadline=p5_deadline, seed=seed)

        else:
            print(f"  No improvement (best={best_score:.8f})")

    # Phase 6: Final polish with active-set

    if remaining > 20:
        print(f"[Phase 6] Final polish")
        a, sc = _active_set_subgrad(best_a, K=5, max_iters=10000, deadline=hard_deadline)

            print(f"  Improved to {sc:.8f}")

    # Sanitize and validate
    print(f"\n[Validate] Best internal score: {best_score:.8f}, n={len(best_a)}")

    best_a = np.asarray(best_a, dtype=np.float64)
    best_a = np.where(np.isfinite(best_a), best_a, 0.0)

    # Try a few candidates for final validation

        if np.isfinite(real) and real < best_actual:

        if best_actual < 1.50:
            break

    if best_out is None:
        best_out = _make_sine(2048, 1, 0.7)
        print("[Fallback] Using basic sinusoidal")

    print(f"[Done] Final score: {best_actual:.8f}")
    return [float(x) for x in best_out]
