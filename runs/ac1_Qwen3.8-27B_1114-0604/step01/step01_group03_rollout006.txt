def _normalize(a):
    """Normalize to simplex (sum=1, all non-negative)."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0:
        a.fill(1.0)
        s = float(a.size)
    a /= s
    return a


def _score(a, L, m, n):
    """Compute 2*n*max(conv(a,a)) using FFT."""
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    np.maximum(c, 0.0, out=c)
    return float(2.0 * n * np.max(c))

    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    np.maximum(c, 0.0, out=c)

def _mws_round(a, L, m, n, rng, n_inner=20, tau=5.0):
    """
    One round of multiplicative weights for the dual variable.
    Returns a new direction to mix with a.
    """
    f = _normalize(a)
    F = np.fft.rfft(f, L)
    Frev = np.fft.rfft(f[::-1], L)
    
    x = _random_init(n, rng)
    y = np.ones(m) / m
    
    for it in range(n_inner):
        # Update y based on current x
        X = np.fft.rfft(x, L)
        u = np.fft.irfft(F * X, L)[:m]
        np.maximum(u, 0.0, out=u)
        
        vmax = float(np.max(u))
        if vmax > 0:
            z = tau * (u - vmax)
            z = np.clip(z, -50.0, 0.0)
            y = np.exp(z)
            ys = float(y.sum())
            if ys > 0 and np.isfinite(ys):
                y /= ys
            else:
                y = np.ones(m) / m
        
        # Update x based on current y
        Y = np.fft.rfft(y, L)
        s = np.fft.irfft(Y * Frev, L)[(n-1):m]
        np.maximum(s, 0.0, out=s)
        
        vmax_s = float(np.max(s))
        if vmax_s > 0:
            zx = -tau * (s - vmax_s)
            zx -= float(np.max(zx))
            zx = np.clip(zx, -50.0, 0.0)
            x = np.exp(zx)
            xs = float(x.sum())
            if xs > 0 and np.isfinite(xs):
                x /= xs
            else:
                x = _random_init(n, rng)
        else:
            x = _random_init(n, rng)
    
    return _normalize(x)

            return float(np.inf)

            if isinstance(x, bool) or not isinstance(x, (int, float)):

                return float(np.inf)
        a = [max(0.0, min(1000.0, float(x))) for x in seq]
        nn = len(a)

        return float(2.0 * nn * mb / (s * s))

    return float(val)

            temp = temp_start
            eta = eta_start
            no_improve = 0
            cycle_count += 1
            continue

        # Softmax weights on convolution
        vmax = float(np.max(c))
        if vmax <= 1e-300:
            break
        temp_eff = temp / vmax
        temp_eff = min(temp_eff, 80.0)
        temp_eff = max(temp_eff, 0.01)

        z = temp_eff * (c - vmax)
        z = np.clip(z, -50.0, 0.0)
        w = np.exp(z)
        wsum = float(w.sum())
        if wsum <= 0 or not np.isfinite(wsum):
            w = np.ones(m) / m
        else:
            w /= wsum

        # Gradient
        g = _grad_soft(a, w, L, m, n)
        gmax = float(np.max(g))
        if gmax <= 1e-300 or not np.isfinite(gmax):
            break
        g = g / gmax

        # Multiplicative update with backtracking
        accepted = False
        for step_mult in (1.0, 0.7, 0.5, 0.35, 0.25, 0.15):
            step = eta * step_mult
            if step < 1e-6:
                break
            a_new = a * np.exp(-step * g)
            a_new = _normalize(a_new)
            c_new = _conv_self(a_new, L, m)
            sc_new = float(2.0 * n * np.max(c_new))

            if sc_new < current_score - 1e-15:
                a = a_new
                current_score = sc_new
                accepted = True
                break

        if accepted:
            eta = min(0.4, eta * 1.08)
        else:
            eta = max(0.01, eta * 0.8)

        it += 1

        # Temperature schedule: cyclic annealing
        if it % 40 == 0:
            phase = (it // 40) % 6
            if phase < 3:
                temp = min(temp_end, temp * 1.3)
            else:
                temp = max(temp_start, temp * 0.6)

    return best_a, best_score


def _mws_direction(f, L, m, n, rng, n_inner=20, tau=5.0):
    """
    Compute a direction using multiplicative weights for the zero-sum game.
    """
    f = _normalize(f)
    F = np.fft.rfft(f, L)
    Frev = np.fft.rfft(f[::-1], L)

    x = _random_init(n, rng)

    for it in range(n_inner):
        frac = float(it) / max(1, n_inner - 1)
        eta_t = tau * (0.5 + frac)

        # Update dual from primal
        X = np.fft.rfft(x, L)
        u = np.fft.irfft(F * X, L)[:m]
        np.maximum(u, 0.0, out=u)

        vu = n * u
        vma = float(np.max(vu))
        if vma > 0:
            z = eta_t * (vu - vma)
            z = np.clip(z, -40.0, 0.0)
            y = np.exp(z)
            ys = float(y.sum())
            if ys > 0 and np.isfinite(ys):
                y /= ys
            else:
                y = np.ones(m) / m
        else:
            y = np.ones(m) / m

        # Update primal from dual
        Y = np.fft.rfft(y, L)
        s = np.fft.irfft(Y * Frev, L)[(n - 1):m]
        np.maximum(s, 0.0, out=s)

        ps = n * s
        sma = float(np.max(ps))
        if sma > 0:
            zx = -eta_t * 0.7 * (ps - sma)
            zx -= float(np.max(zx))
            zx = np.clip(zx, -40.0, 0.0)
            x = np.exp(zx)
            xs = float(x.sum())
            if xs > 0 and np.isfinite(xs):
                x /= xs
            else:
                x = _random_init(n, rng)

    return _normalize(x)


def _mws_step(a, L, m, n, rng, n_inner=15, tau=5.0):
    """One MWS step: find direction and line search for best mix."""
    x = _mws_direction(a, L, m, n, rng, n_inner, tau)

    best_sc = _score_fast(a, L, m, n)
    best_a = a.copy()

    for lam in (1.0, 0.8, 0.65, 0.5, 0.4, 0.3, 0.2, 0.1):
        cand = _normalize((1.0 - lam) * a + lam * x)
        sc = _score_fast(cand, L, m, n)
        if sc < best_sc:
            best_sc = sc
            best_a = cand

    return best_a, best_sc


def _exact_eval(seq):
    """Evaluate using evaluate_sequence if available."""
    try:
        val = evaluate_sequence(seq)
    except NameError:
        if not isinstance(seq, list) or not seq:
            return float(np.inf)
        for x in seq:
            if isinstance(x, bool) or not isinstance(x, (int, float)):
                return float(np.inf)
            if np.isnan(x) or np.isinf(x):
                return float(np.inf)
        a = [max(0.0, min(1000.0, float(x))) for x in seq]
        nn = len(a)
        s = sum(a)
        if s < 0.01:
            return float(np.inf)
        b = np.convolve(a, a)
        mb = max(b)
        return float(2.0 * nn * mb / (s * s))
    except Exception:
        return float(np.inf)
    if val is None:
        return float(np.inf)
    return float(val)


def _polish(a, L, m, n, rng, time_limit):
    """Fine polishing with very low temperature."""
    deadline = time.monotonic() + time_limit
    best_a = _normalize(a)
    best_score = _score_fast(best_a, L, m, n)
    eta = 0.02
    it = 0

    while time.monotonic() < deadline:
        c = _conv_self(best_a, L, m)
        current_score = float(2.0 * n * np.max(c))

        if current_score < best_score:
            best_score = current_score

        # Very focused softmax (low temperature)
        vmax = float(np.max(c))
        if vmax <= 1e-300:
            break
        temp_eff = 1.0 / vmax
        temp_eff = min(temp_eff, 5.0)

        z = temp_eff * (c - vmax)
        z = np.clip(z, -40.0, 0.0)
        w = np.exp(z)
        wsum = float(w.sum())
        if wsum <= 0:
            w = np.ones(m) / m
        else:
            w /= wsum

        g = _grad_soft(best_a, w, L, m, n)
        gmax = float(np.max(g))
        if gmax <= 1e-300:
            break
        g = g / gmax

        a_new = best_a * np.exp(-eta * g)
        a_new = _normalize(a_new)
        c_new = _conv_self(a_new, L, m)
        sc_new = float(2.0 * n * np.max(c_new))

        if sc_new < best_score:
            best_a = a_new
            best_score = sc_new
            eta = min(0.1, eta * 1.02)
        else:
            eta = max(0.005, eta * 0.9)

        it += 1
        if it % 200 == 0:
            # Tiny random perturbation
            p = rng.normal(0, 0.001, n)
            best_a = _normalize(best_a * (1.0 + p))

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search: explores multiple sequence lengths, uses adaptive
    multiplicative gradient with temperature cycling and MWS game steps,
    then polishes the best result.
    """
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget < 20:
        budget = 2400.0

    t_start = time.monotonic()
    hard_deadline = t_start + budget - 8.0

    rng = np.random.default_rng(seed)

    # Load previous best if available
    prev_arr = None
    prev_n = None
    try:
        hs = height_sequence_1
        if hs is not None:
            arr = np.asarray(hs, dtype=np.float64).ravel()
            if arr.size >= 100 and np.all(np.isfinite(arr)):
                prev_arr = arr.copy()
                prev_n = arr.size
    except (NameError, Exception):
        pass

    # --- Phase 1: Multi-n exploration ---
    print("Phase 1: Exploring sequence lengths...")

    n_list = []
    if prev_n is not None:
        n_list.append(prev_n)
    for nv in [4000, 5000, 6121, 7000, 8000, 9000, 10000, 12000, 14000]:
        if nv not in n_list:
            n_list.append(nv)
    n_list = sorted(set(n_list))

    # Limit to at most 7 values for time management
    if len(n_list) > 7:
        n_list = [n_list[0], n_list[1], n_list[3], n_list[4],
                  n_list[5], n_list[6], n_list[-1]]
        n_list = sorted(set(n_list))

    phase1_time = min(500.0, budget * 0.2)
    per_n_time = phase1_time / len(n_list)

    results = {}
    for n in n_list:
        if time.monotonic() > hard_deadline - 600:
            break

        L = 1 << (2 * n - 1).bit_length()
        m = 2 * n - 1

        # Build initial candidates
        inits = []
        if prev_arr is not None and prev_n == n:
            inits.append(_normalize(prev_arr))
        inits.append(_random_init(n, rng))
        inits.append(_tapered_init(n, rng, 1.5))
        inits.append(_tapered_init(n, rng, 3.0))
        inits.append(_normalize(np.ones(n, dtype=np.float64) *
                                (1.0 + 0.3 * np.sin(np.pi * np.linspace(0, 1, n)))))

        best_a = None
        best_sc = float(np.inf)
        for a0 in inits:
            sc = _score_fast(a0, L, m, n)
            if sc < best_sc:
                best_sc = sc
                best_a = a0.copy()

        # Quick optimization
        tl = min(per_n_time - 2.0, hard_deadline - time.monotonic() - 600)
        if tl > 3.0:
            a_opt, sc_opt = _optimize_adaptive(
                best_a, L, m, n, rng, tl,
                temp_start=5.0, temp_end=200.0,
                eta_start=0.08, verbose=False)
            if sc_opt < best_sc:
                best_sc = sc_opt
                best_a = a_opt

        results[n] = {'score': best_sc, 'a': best_a, 'L': L, 'm': m}
        print(f"  n={n}: score={best_sc:.6f}")

    # --- Phase 2: Focused optimization on top candidates ---
    print("\nPhase 2: Focused optimization...")

    sorted_res = sorted(results.items(), key=lambda x: x[1]['score'])
    top_ns = [nv for nv, _ in sorted_res[:3]]

    best_overall = float(np.inf)
    best_a_overall = None
    best_n = None
    best_L = None
    best_m = None

    for nv in top_ns:
        info = results[nv]
        if info['score'] < best_overall:
            best_overall = info['score']
            best_a_overall = info['a']
            best_n = nv
            best_L = info['L']
            best_m = info['m']

    n = best_n
    L = best_L
    m = best_m
    a = best_a_overall.copy()

    phase2_deadline = t_start + budget - 120.0
    method = 0
    stale = 0
    log_count = 0

    while time.monotonic() < phase2_deadline:
        rem = phase2_deadline - time.monotonic()
        if rem < 15:
            break

        if method % 4 == 0:
            # Adaptive gradient (medium temperature)
            lim = min(150.0, max(10.0, 0.30 * rem))
            ts = 2.0 + 3.0 * (method // 4) % 4
            tmp, sc = _optimize_adaptive(a, L, m, n, rng, lim,
                                         temp_start=ts, temp_end=400.0,
                                         eta_start=0.06)
        elif method % 4 == 1:
            # MWS game step (multiple sub-steps)
            lim = min(120.0, max(10.0, 0.25 * rem))
            deadline_mws = time.monotonic() + lim
            while time.monotonic() < deadline_mws:
                tau = 3.0 + 3.0 * rng.random()
                tmp, sc = _mws_step(a, L, m, n, rng,
                                    n_inner=12 + int(8 * rng.random()),
                                    tau=tau)
                if sc < best_overall:
                    best_overall = sc
                    a = tmp
                else:
                    a = _normalize(0.7 * a + 0.3 * tmp)
                if time.monotonic() > deadline_mws:
                    break
            sc = _score_fast(a, L, m, n)
            tmp = a
        elif method % 4 == 2:
            # Low temperature gradient (exploitation)
            lim = min(100.0, max(10.0, 0.20 * rem))
            tmp, sc = _optimize_adaptive(a, L, m, n, rng, lim,
                                         temp_start=0.5, temp_end=30.0,
                                         eta_start=0.03)
        else:
            # High temperature exploration
            lim = min(80.0, max(10.0, 0.15 * rem))
            tmp, sc = _optimize_adaptive(a, L, m, n, rng, lim,
                                         temp_start=50.0, temp_end=500.0,
                                         eta_start=0.15)

        if sc < best_overall - 1e-12:
            best_overall = sc
            best_a_overall = tmp.copy()
            a = tmp
            stale = 0
            log_count += 1
            if log_count % 5 == 1:
                print(f"  [{method}] n={n} score={sc:.6f} (improved)")
        else:
            stale += 1
            a = tmp
            if stale >= 5:
                # Aggressive restart
                r = _random_init(n, rng)
                a = _normalize(0.6 * best_a_overall + 0.4 * r)
                stale = 3

        method += 1

    # --- Phase 3: Polish ---
    print(f"\nPhase 3: Polishing (current best={best_overall:.6f})...")

    rem = hard_deadline - time.monotonic()
    if rem > 20:
        polish_time = min(rem - 10, 180.0)
        a_pol, sc_pol = _polish(best_a_overall, L, m, n, rng, polish_time)
        if sc_pol < best_overall:
            best_overall = sc_pol
            best_a_overall = a_pol
            print(f"  After polish: {sc_pol:.6f}")

    # Try one more round of MWS on the polished result
    rem = hard_deadline - time.monotonic()
    if rem > 30:
        for _ in range(5):
            if time.monotonic() > hard_deadline - 15:
                break
            tmp, sc = _mws_step(best_a_overall, L, m, n, rng,
                                n_inner=20, tau=2.0)
            if sc < best_overall:
                best_overall = sc
                best_a_overall = tmp
                print(f"  MWS polish: {sc:.6f}")

    # --- Phase 4: Exact evaluation ---
    print(f"\nPhase 4: Exact evaluation...")

    candidates = [best_a_overall]
    if prev_arr is not None:
        candidates.append(_normalize(prev_arr))

    # Also evaluate a few variants
    for pert in (0.01, 0.005):
        r = rng.normal(0, pert, n)
        candidates.append(_normalize(best_a_overall * (1.0 + r)))

    best_exact = float(np.inf)
    best_seq = None

    seen_keys = set()
    for arr in candidates:
        if time.monotonic() > hard_deadline:
            break
        # Dedup by sampling
        step = max(1, len(arr) // 200)
        key = tuple(np.round(arr[::step], 6))
        if key in seen_keys:
            continue
        seen_keys.add(key)

        seq = [float(x) for x in arr]
        val = _exact_eval(seq)
        print(f"  Exact: {val:.6f}")
        if np.isfinite(val) and val < best_exact:
            best_exact = val
            best_seq = seq

    if best_seq is None:
        best_seq = [float(x) for x in best_a_overall]

    elapsed = time.monotonic() - t_start
    print(f"\nDone in {elapsed:.1f}s. Best score: {best_exact:.6f}, n={len(best_seq)}")
    return best_seq
