```python
"""
Minimize 2*n*max(a*a)/(sum(a))^2 using multi-stage top-k projected gradient
with Nesterov momentum. Different from previous approach which used smooth-max
(LSE) gradient; this uses exact top-k subset gradients with a k-annealing
schedule, giving sharper descent directions in later stages.
"""

import numpy as np
import time

def fft_convolution(a):
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0]*a[0]], dtype=np.float64)
    need = 2*n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F*F, size)[:need]
    return np.maximum(b, 0.0)

def project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    if not np.all(np.isfinite(v)):
        return np.full(v.shape, target/float(v.size), dtype=np.float64)
    low = float(np.min(v)) - 1.0
    high = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5*(low+high)
        s = float(np.sum(np.maximum(v-mid, 0.0)))
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5*(low+high)
    q = np.maximum(v-tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        q = np.full(v.shape, target/float(v.size), dtype=np.float64)
    else:
        q *= target/s
    return q

def normalize_simplex(p):
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0/float(p.size), dtype=np.float64)
    p *= 1.0/s
    return p

def compute_gradient_topk(a, b, k):
    """Compute (1/k)*sum of top-k gradients of b = a*a."""
    n = a.size
    b_sorted_idx = np.argpartition(-b, min(k, b.size-1))[:k]
    # For each top-k lag, compute the gradient contribution
    # grad[m] for lag k_lag is: a[k_lag - m] + a[m - k_lag] (with a[j]=0 outside [0,n-1])
    # We need to sum over the top-k lags
    grad = np.zeros(n, dtype=np.float64)
    for lag in b_sorted_idx:
        lag = int(lag)
        # a[lag - m] is non-zero when 0 <= lag-m <= n-1, i.e., lag-(n-1) <= m <= lag
        # a[m - lag] is non-zero when 0 <= m-lag <= n-1, i.e., lag <= m <= lag+(n-1)
        # Combine: m ranges from max(0, lag-(n-1)) to min(n-1, lag+(n-1))
        lo1 = max(0, lag - (n-1))
        hi1 = min(n-1, lag)
        if lo1 <= hi1:
            # a[lag - m] for m in [lo1, hi1] -> indices lag-hi1 to lag-lo1 in a
            idx_m = np.arange(lo1, hi1+1)
            idx_a = lag - idx_m
            grad[lo1:hi1+1] += a[idx_a]
        
        lo2 = max(0, lag)
        hi2 = min(n-1, lag + (n-1))
        if lo2 <= hi2:
            # a[m - lag] for m in [lo2, hi2] -> indices m-lag in a
            idx_m = np.arange(lo2, hi2+1)
            idx_a = idx_m - lag
            grad[lo2:hi2+1] += a[idx_a]
    
    grad /= float(k)
    return grad
```

def fft_convolution(a):
    """Linear convolution a*a via FFT, clipped at 0."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    m = v.size
    if m == 0:
        return v
    if target <= 0:
        return np.zeros(m, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(m, target / m, dtype=np.float64)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0 or not np.isfinite(s):
        q = np.full(m, target / m, dtype=np.float64)
    else:
        q *= target / s
    return q


def normalize_simplex(p):
    """Clip to nonnegative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0 or not np.isfinite(s):
        return np.full(p.size, 1.0 / p.size, dtype=np.float64)
    return p / s

def make_u_shape(n, alpha, rng=None):
    """Create a U-shaped sequence: high at edges, low in middle."""

    if rng is not None:

    p = np.exp(z)

    z = -beta * (x - 0.5) ** 2 + gamma * (x - 0.5) ** 4
    if rng is not None:

def make_random_smooth(n, rng, smoothness=5):
    """Random sequence with spatial smoothness."""

```python
Key differences from previous approach:
- Uses exact top-k subset gradients (piecewise-linear) instead of smooth-max (LSE)
- k-annealing schedule: start with large k (smooth surrogate), end with k=1 (exact max)
- Nesterov-style momentum for faster convergence
- Diverse U-shaped and bimodal starting points
- Perturbation-based restarts to escape local minima
"""

import numpy as np
import time


def fft_convolution(a):
    """Linear convolution a*a via FFT, clipped to nonnegative."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target} using bisection."""
    v = np.asarray(v, dtype=np.float64)
    m = v.size
    if m == 0:
        return v
    if target <= 0:
        return np.zeros(m, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(m, target / m, dtype=np.float64)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0 or not np.isfinite(s):
        q = np.full(m, target / m, dtype=np.float64)
    else:
        q *= target / s
    return q


def normalize_simplex(p):
    """Clip to nonnegative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0 or not np.isfinite(s):
        return np.full(p.size, 1.0 / p.size, dtype=np.float64)
    return p / s


def topk_gradient_fast(a, b, k, fft_size):
    """
    Compute (2/k) * sum of subgradients for the top-k lags of b = a*a.
    
    The subgradient of (a*a)[L] w.r.t. a[m] is 2*a[L-m] if 0<=L-m<=n-1 else 0.
    The sum over top-k lags is computed via FFT cross-correlation.
    """
    n = a.size
    b_size = b.size  # 2n-1
    k = min(k, b_size)
    if k <= 0:
        return np.zeros(n, dtype=np.float64)

    # Indicator vector of top-k lags
    if k >= b_size:
        v = np.ones(b_size, dtype=np.float64)
    else:
        top_idx = np.argpartition(-b, k - 1)[:k]
        v = np.zeros(b_size, dtype=np.float64)
        v[top_idx] = 1.0

    # Reverse v for convolution formulation
    v_r = v[::-1].copy()

    # FFT linear convolution of a (length n) and v_r (length 2n-1)
    need = n + b_size - 1  # = 3n-2
    size = 1
    while size < need:
        size <<= 1

    A = np.fft.rfft(a, size)
    VR = np.fft.rfft(v_r, size)
    conv = np.fft.irfft(A * VR, size)[:need]

    # c[m] = conv[2n-2-m] for m=0..n-1
    c = conv[n - 1:2 * n - 1][::-1].copy()

    grad = (2.0 / k) * c
    return grad


def fast_score(p):
    """Compute 2*n*max(p*p) assuming sum(p)=1."""
    b = fft_convolution(p)
    M = float(np.max(b))
    return 2.0 * p.size * M


def make_u_shape(n, alpha, rng=None):
    """U-shaped sequence: high at edges, low in middle."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    z = -alpha * (x - 0.5) ** 2
    if rng is not None:
        z = z + 0.05 * rng.standard_normal(n)
    p = np.exp(z)
    return normalize_simplex(p)


def make_bimodal(n, beta, gamma, rng=None):
    """Bimodal: two humps away from center."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    z = -beta * (x - 0.5) ** 2 + gamma * (x - 0.5) ** 4
    if rng is not None:
        z = z + 0.05 * rng.standard_normal(n)
    p = np.exp(z)
    return normalize_simplex(p)


def make_random_smooth(n, rng, smoothness=5.0):
    """Random sequence with low-frequency dominance."""
    frequencies = np.fft.rfftfreq(n)
    mask = np.exp(-smoothness * frequencies * n * 0.1)
    nr = n // 2 + 1
    zc = (rng.standard_normal(nr) + 1j * rng.standard_normal(nr)) * mask
    z = np.real(np.fft.irfft(zc, n))
    z = 1.5 * np.tanh(z)
    p = np.exp(z)
    return normalize_simplex(p)


def make_step_trap(n, rng):
    """Step function with trapezoidal taper at edges."""
    p = np.ones(n, dtype=np.float64)
    taper = n // 20
    if taper > 0:
        for i in range(taper):
            frac = (i + 1) / (taper + 1)
            p[i] = frac
            p[n - 1 - i] = frac
    # Make left/right asymmetric
    p[:n // 3] *= 1.3
    p[2 * n // 3:] *= 1.2
    if rng is not None:
        p = p * (1.0 + 0.02 * rng.standard_normal(n))
    return normalize_simplex(np.maximum(p, 0.0))


def generate_starts(n, rng, count=10):
    """Generate diverse starting points."""
    starts = []
    # U-shapes with increasing concentration
    for alpha in (3.0, 8.0, 15.0, 30.0, 60.0):
        starts.append(make_u_shape(n, alpha, rng))
    # Bimodal
    starts.append(make_bimodal(n, 5.0, 10.0, rng))
    starts.append(make_bimodal(n, 3.0, 15.0, rng))
    # Random smooth
    starts.append(make_random_smooth(n, rng, smoothness=3.0))
    starts.append(make_random_smooth(n, rng, smoothness=10.0))
    # Step/trap
    starts.append(make_step_trap(n, rng))
    return starts[:count]


def optimize_from_start(a0, deadline, k_schedule, beta_mom=0.85, step_init=0.01):
    """
    Multi-stage projected gradient with momentum.
    k_schedule: list of k values (annealing from large to small).
    Returns (best_a, best_score).
    """
    n = a0.size
    a = normalize_simplex(a0).copy()
    momentum = np.zeros(n, dtype=np.float64)
    step = step_init
    best_a = a.copy()
    best_score = fast_score(a)

    no_improve = 0
    max_no_improve = 3000
    total_iters = 0

    for k in k_schedule:
        k = max(1, int(k))
        n_k = min(k, 2 * n - 1)
        it = 0

        while time.monotonic() < deadline:
            it += 1
            total_iters += 1

            b = fft_convolution(a)
            M = float(np.max(b))
            score = 2.0 * n * M

            if score < best_score - 1e-13:
                best_score = score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > max_no_improve:
                # Perturb to escape local minimum
                perturb_size = 0.005 * step
                a = normalize_simplex(a + perturb_size * np.random.standard_normal(n))
                momentum *= 0.1
                no_improve = 0
                max_no_improve = max(500, max_no_improve // 2)
                continue

            # Compute top-k gradient
            grad = topk_gradient_fast(a, b, n_k, None)
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-15:
                break

            # Momentum update
            momentum = beta_mom * momentum - step * grad
            z = a + momentum
            a_new = project_simplex(z, 1.0)

            # Evaluate new point
            b_new = fft_convolution(a_new)
            M_new = float(np.max(b_new))

            if M_new < M - 1e-15:
                a = a_new
                # Gradually increase step
                if step < 0.05:
                    step = min(step * 1.08, 0.05)
            else:
                # Rejected: reduce step
                step *= 0.4
                if step < 1e-9:
                    break

            if it % 10000 == 0:
                elapsed = time.monotonic() - (deadline - (deadline - time.monotonic()))
                print(f"    k={k}, iter={it}, score={best_score:.8f}, step={step:.4f}")

        if it > 0:
            print(f"  Phase k={k}: {it} iters, best_score={best_score:.8f}")

    return best_a, best_score


def optimize_smooth_max(a0, deadline, beta=50000.0, beta_mom=0.9, step_init=0.005):
    """
    Smooth-max (LSE) projected gradient as alternative/complement.
    Used for final polishing after top-k phases.
    """
    n = a0.size
    a = normalize_simplex(a0).copy()
    momentum = np.zeros(n, dtype=np.float64)
    step = step_init
    best_a = a.copy()
    best_score = fast_score(a)
    no_improve = 0

    # FFT size for correlation
    need = 2 * n - 1
    fft_size_corr = 1
    while fft_size_corr < need:
        fft_size_corr <<= 1

    while time.monotonic() < deadline:
        b = fft_convolution(a)
        M = float(np.max(b))
        score = 2.0 * n * M

        if score < best_score - 1e-13:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 5000:
            break

        # Smooth max gradient: w = softmax(beta*(b - M))
        x = beta * (b - M)
        x = np.clip(x, -700.0, 0.0)
        e = np.exp(x)
        Z = float(np.sum(e))
        if Z <= 0 or not np.isfinite(Z):
            e = np.zeros_like(b)
            e[np.argmax(b)] = 1.0
            Z = 1.0
        w = e / Z

        # Gradient via FFT: s[m] = sum_j a[j] * w[m-j] (cross-correlation)
        # Same trick as topk_gradient but with continuous weights w
        w_r = w[::-1].copy()
        need2 = n + b.size - 1  # 3n-2
        size2 = 1
        while size2 < need2:
            size2 <<= 1
        A = np.fft.rfft(a, size2)
        WR = np.fft.rfft(w_r, size2)
        conv = np.fft.irfft(A * WR, size2)[:need2]
        c = conv[n - 1:2 * n - 1][::-1].copy()
        grad = 2.0 * c

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        momentum = beta_mom * momentum - step * grad
        z = a + momentum
        a_new = project_simplex(z, 1.0)

        b_new = fft_convolution(a_new)
        M_new = float(np.max(b_new))

        if M_new < M - 1e-15:
            a = a_new
            if step < 0.03:
                step = min(step * 1.05, 0.03)
        else:
            step *= 0.4
            if step < 1e-10:
                break

        if no_improve % 2000 == 0 and no_improve > 0:
            a = normalize_simplex(a + 0.003 * step * np.random.standard_normal(n))
            momentum *= 0.5
            no_improve = 0

    return best_a, best_score


def official_score(seq):
    """Call the authoritative evaluator with exception safety."""
    try:
        val = evaluate_sequence(seq)
        val = float(val)
    except Exception:
        return float('inf')
    return val if np.isfinite(val) else float('inf')


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best nonnegative sequence minimizing 2*n*max(a*a)/(sum(a))^2.
    
    Strategy:
    1. Generate diverse starting points (U-shapes, bimodal, random smooth).
    2. For each top start, run multi-stage top-k gradient with momentum.
    3. Polish the best with smooth-max gradient.
    4. Return the best sequence found.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 1.0:
        return [1.0]

    start_time = time.monotonic()
    reserve = 30.0
    deadline = start_time + max(5.0, budget_s - reserve)

    # Choose sequence length
    if budget_s < 150:
        N = 2048
    elif budget_s < 500:
        N = 4096
    else:
        N = 8192

    rng = np.random.default_rng(seed)
    np.random.seed(seed)

    print(f"=== propose_candidate: N={N}, budget={budget_s:.0f}s ===")

    # Collect starting points
    starts = []

    # Try to use previous best
    try:
        hs = height_sequence_1
        if isinstance(hs, (list, np.ndarray)) and len(hs) == N:
            p = normalize_simplex(np.array(hs, dtype=np.float64))
            starts.append((fast_score(p), p))
            print(f"  Loaded height_sequence_1: score={starts[0][0]:.8f}")
    except Exception:
        pass

    # Generate new starts
    gen_starts = generate_starts(N, rng, count=10)
    for i, p in enumerate(gen_starts):
        sc = fast_score(p)
        starts.append((sc, p))
        print(f"  Gen start {i}: score={sc:.8f}")

    # Sort by score
    starts.sort(key=lambda x: x[0])

    # Select top starts for optimization
    sel_count = min(5, len(starts))
    selected = [starts[i][1] for i in range(sel_count)]

    # Time allocation
    total_avail = deadline - start_time
    # 70% for top-k optimization, 20% for smooth-max polish, 10% buffer
    topk_budget = total_avail * 0.70
    polish_budget = total_avail * 0.20

    # Allocate topk budget among starts
    fracs = [0.35, 0.25, 0.18, 0.12, 0.10]
    best_a = selected[0].copy()
    best_score = starts[0][0]

    for idx in range(sel_count):
        remaining = deadline - time.monotonic() - polish_budget - 10.0
        if remaining < 5.0:
            break
        alloc = min(remaining, topk_budget * fracs[idx])
        sub_deadline = time.monotonic() + alloc

        # k-annealing schedule
        k_schedule = [
            max(1, N // 4),
            max(1, N // 8),
            max(1, N // 16),
            max(1, N // 32),
            max(1, N // 64),
            max(1, N // 128),
            1,
        ]

        print(f"\nOptimizing start {idx} (initial score={fast_score(selected[idx]):.6f}), "
              f"time={alloc:.1f}s")

        a_fit, sc_fit = optimize_from_start(
            selected[idx], sub_deadline, k_schedule,
            beta_mom=0.85, step_init=0.01
        )

        if sc_fit < best_score:
            best_score = sc_fit
            best_a = a_fit.copy()
            print(f"  New best: {best_score:.8f}")

    # Phase 2: Smooth-max polish on the best solution
    remaining = deadline - time.monotonic() - 5.0
    if remaining > 10.0:
        print(f"\nPolishing with smooth-max, time={remaining:.1f}s")
        a_pol, sc_pol = optimize_smooth_max(
            best_a, time.monotonic() + remaining,
            beta=100000.0, beta_mom=0.9, step_init=0.005
        )
        if sc_pol < best_score:
            best_score = sc_pol
            best_a = a_pol.copy()
            print(f"  After polish: {best_score:.8f}")

    # Phase 3: One more top-k refinement from the polished solution
    remaining = deadline - time.monotonic() - 5.0
    if remaining > 10.0:
        k_schedule_fine = [
            max(1, N // 64),
            max(1, N // 128),
            max(1, N // 256),
            1,
        ]
        print(f"\nFinal top-k refinement, time={remaining:.1f}s")
        a_fin, sc_fin = optimize_from_start(
            best_a, time.monotonic() + remaining,
            k_schedule_fine, beta_mom=0.9, step_init=0.005
        )
        if sc_fin < best_score:
            best_score = sc_fin
            best_a = a_fin.copy()
            print(f"  After final refinement: {best_score:.8f}")

    # Final cleanup
    final_p = normalize_simplex(best_a)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_seq = [float(x) for x in final_p]

    # Verify with official evaluator
    off = official_score(final_seq)
    print(f"\nOfficial score: {off:.8f}")

    if not np.isfinite(off) or off > 2.0:
        # Fallback to uniform
        return [1.0 / N] * N

    elapsed = time.monotonic() - start_time
    print(f"Total time: {elapsed:.1f}s")
    return final_seq
```
