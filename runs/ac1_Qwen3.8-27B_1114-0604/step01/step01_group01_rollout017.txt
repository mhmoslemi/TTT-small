"""

import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 3000000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)


def _conv_correlate_front(w, p):
    """v[j] = sum_i p[i] * w[j+i] for j=0..len(p)-1 (front correlation)."""
    lenw, lenp = len(w), len(p)
    if lenp == 0:
        return np.array([], dtype=float)
    L = _next_pow2(max(lenw, 2 * lenp) + 1)
    fw = np.fft.rfft(w, L)
    fp = np.fft.rfft(p, L)
    out = np.fft.irfft(fw * np.conjugate(fp), L)
    return out[:lenp].astype(float, copy=False)


def _sanitize(p):

    if p.ndim != 1 or p.size == 0:
        return None

        return None
    return p / s

def _soft_weights(d, beta):
    d = np.asarray(d, dtype=float)
    n = d.size
    if n == 0:
        return np.array([], dtype=float)
    d = np.where(np.isfinite(d), d, 0.0)

        return np.full(n, 1.0 / n)
    shift = float(np.max(d))

    e = np.exp(beta * (d - shift))
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return e / s


def _gradient_info(p, beta):

    h = np.maximum(h, 0.0)
    maxh = float(np.max(h)) if h.size else 0.0
    smooth = float(np.dot(w, d))
    return exact, smooth, h, maxh


def _init_periodic(m, rng, npeaks, phase=0.0, width=1.0, asym=1.0):
    """Create a periodic bump structure."""
    t = np.arange(m, dtype=float)
    p = np.zeros(m)
    period = max(1, m // npeaks)
    for k in range(npeaks):
        center = k * period + int(phase * period)
        center = center % m
        for di in range(-3 * width * period, 3 * width * period):
            idx = (center + di) % m
            p[idx] += np.exp(-0.5 * (di / (width * period)) ** 2)
    p = np.maximum(p, 1e-15)
    # Apply asymmetry
    ramp = np.linspace(1.0, asym, m)
    p = p * ramp
    p = _sanitize(p)

def _init_fourier(m, rng, ncomp=8):

    t = np.arange(m, dtype=float) / m

                        p = p_try
                    else:
                        p = best.copy()
                stag = 0

    return best, best_score


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """Main search: returns best non-negative sequence found."""
    budget = float(budget_s) if budget_s else 2400.0
    if not np.isfinite(budget) or budget < 60:
        budget = 2400.0

    start_t = time.monotonic()
    deadline = start_t + budget - 30.0
    rng = np.random.default_rng(int(seed) % (2**31 - 1) + 7)

    best_arr = None
    best_score = float("inf")

    # Try parent
    try:
        par = _sanitize(np.asarray(height_sequence_1, dtype=float))
    except (NameError, Exception):
        par = None

    if par is not None:
        sc = _score(par)
        best_arr = par.copy()
        best_score = sc
        print(f"Parent: n={par.size}, score={sc:.6f}")

    # Multi-resolution search
    # Phase 1: Small n for quick basin finding
    small_n = 256
    small_candidates = []

    # Diverse initializations at small scale
    inits_small = []
    inits_small.append(_init_ramp(small_n, rng, 1.0))
    inits_small.append(_init_ramp(small_n, rng, 0.5))
    inits_small.append(_init_ramp(small_n, rng, 2.0))
    for npk in (3, 5, 7, 10, 15):
        inits_small.append(_init_periodic(small_n, rng, npk, phase=rng.random(),
                                          width=rng.uniform(0.3, 1.0),
                                          asym=rng.uniform(0.5, 2.0)))
    inits_small.append(_init_fourier(small_n, rng, 6))
    inits_small.append(_init_two_block(small_n, 2.0, 0.5))
    inits_small.append(_init_two_block(small_n, 0.5, 0.5))
    if par is not None:
        inits_small.append(_resample(par, small_n))

    # Score and sort
    scored = []
    for init in inits_small:
        if init is not None:
            sc = _score(init)
            scored.append((sc, init))
    scored.sort(key=lambda x: x[0])
    print(f"Small-n best init score: {scored[0][0]:.6f}")

    # Run mirror descent on top small-n candidates
    rem = deadline - time.monotonic()
    top_k = min(4, len(scored))
    for i in range(top_k):
        if rem < 15:
            break
        t_alloc = min(20.0, max(5.0, rem / (top_k - i + 1) - 5))
        p, sc = _mirror_descent(scored[i][1], rng, t_alloc,
                                beta_start=15, beta_max=800,
                                alpha_init=0.12)
        if p is not None and sc < best_score:
            best_arr = p.copy()
            best_score = sc
        small_candidates.append((sc, p))
        rem = deadline - time.monotonic()

    small_candidates.sort(key=lambda x: x[0])

    # Phase 2: Medium n
    med_n = 1024
    med_inits = []
    for sc, arr in small_candidates[:3]:
        if arr is not None:
            med_inits.append(_resample(arr, med_n))
    med_inits.append(_init_ramp(med_n, rng, 1.5))
    med_inits.append(_init_ramp(med_n, rng, 0.7))
    for npk in (5, 8, 12, 16):
        med_inits.append(_init_periodic(med_n, rng, npk, phase=rng.random(),
                                        width=rng.uniform(0.3, 0.8),
                                        asym=rng.uniform(0.5, 1.5)))
    med_inits.append(_init_fourier(med_n, rng, 10))
    med_inits.append(_init_two_block(med_n, 1.8, 0.45))
    med_inits.append(_init_two_block(med_n, 0.6, 0.55))
    if par is not None:
        med_inits.append(_resample(par, med_n))

    med_scored = []
    for init in med_inits:
        if init is not None:
            sc = _score(init)
            med_scored.append((sc, init))
    med_scored.sort(key=lambda x: x[0])

    rem = deadline - time.monotonic()
    top_m = min(4, len(med_scored))
    for i in range(top_m):
        if rem < 40:
            break
        t_alloc = min(200.0, max(30.0, rem / (top_m - i + 1) - 10))
        p, sc = _mirror_descent(med_scored[i][1], rng, t_alloc,
                                beta_start=30, beta_max=3000,
                                alpha_init=0.12, verbose=True)
        if p is not None and sc < best_score:
            best_arr = p.copy()
            best_score = sc
            print(f"Med-n update: score={sc:.6f}, time={time.monotonic()-start_t:.1f}s")
        rem = deadline - time.monotonic()

    # Phase 3: Final n
    final_n = 3072
    final_inits = []
    # Resample best medium candidate
    if med_scored:
        bsc, barr = min(med_scored, key=lambda x: x[0])
        final_inits.append(_resample(barr, final_n))
    # Also resample a couple more for diversity
    for i in range(1, min(3, len(med_scored))):
        final_inits.append(_resample(med_scored[i][1], final_n))
    # Fresh inits at full scale
    final_inits.append(_init_ramp(final_n, rng, 1.3))
    final_inits.append(_init_ramp(final_n, rng, 0.8))
    for npk in (6, 10, 14):
        final_inits.append(_init_periodic(final_n, rng, npk, phase=rng.random(),
                                          width=rng.uniform(0.3, 0.7),
                                          asym=rng.uniform(0.6, 1.4)))
    final_inits.append(_init_fourier(final_n, rng, 12))
    final_inits.append(_init_two_block(final_n, 1.7, 0.48))
    if par is not None:
        final_inits.append(_resample(par, final_n))
        # Perturbed parent
        pp = _resample(par, final_n)
        if pp is not None:
            pp = _sanitize(pp * np.exp(rng.normal(0, 0.05, final_n)))
            final_inits.append(pp)

    final_scored = []
    for init in final_inits:
        if init is not None:
            sc = _score(init)
            final_scored.append((sc, init))
    final_scored.sort(key=lambda x: x[0])
    print(f"Final-n best init: score={final_scored[0][0]:.6f}")

    rem = deadline - time.monotonic()
    top_f = min(3, len(final_scored))
    fracs = [0.5, 0.3, 0.2]
    for i in range(top_f):
        if rem < 60:
            break
        t_alloc = min(rem * fracs[i], rem - 60)
        t_alloc = max(30.0, t_alloc)
        p, sc = _mirror_descent(final_scored[i][1], rng, t_alloc,
                                beta_start=50, beta_max=8000,
                                alpha_init=0.1, verbose=True)
        if p is not None and sc < best_score:
            best_arr = p.copy()
            best_score = sc
            print(f"Final-n update: score={sc:.6f}, time={time.monotonic()-start_t:.1f}s")
        rem = deadline - time.monotonic()

    # Phase 4: Polish with block descent
    if best_arr is not None and time.monotonic() < deadline - 80:
        t_polish = min(200.0, deadline - time.monotonic() - 30)
        p, sc = _block_descent(best_arr, rng, t_polish)
        if p is not None and sc < best_score:
            best_arr = p.copy()
            best_score = sc
            print(f"Block polish: score={sc:.6f}")

    # Phase 5: Final mirror descent refinement
    if best_arr is not None and time.monotonic() < deadline - 30:
        t_ref = min(120.0, deadline - time.monotonic() - 10)
        p, sc = _mirror_descent(best_arr, rng, t_ref,
                                beta_start=500, beta_max=15000,
                                alpha_init=0.05)
        if p is not None and sc < best_score:
            best_arr = p.copy()
            best_score = sc
            print(f"Final refine: score={sc:.6f}")

    # Fallback
    if best_arr is None:
        best_arr = np.ones(256, dtype=float) / 256
        best_score = _score(best_arr)

    best_arr = _sanitize(best_arr)
    if best_arr is None:
        best_arr = np.ones(256, dtype=float) / 256

    # Verify with evaluate_sequence if available
    out_list = [float(x) for x in best_arr]
    try:
        eval_sc = evaluate_sequence(out_list)
        if np.isfinite(eval_sc):
            print(f"evaluate_sequence score: {eval_sc:.6f}")
    except Exception:
        pass

    out = np.asarray(best_arr, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s > 0 and np.isfinite(s):
        out = out / s
    return [float(x) for x in out]
