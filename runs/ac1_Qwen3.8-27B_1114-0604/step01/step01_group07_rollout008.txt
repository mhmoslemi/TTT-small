"""
Low-peak nonnegative step-function search using mirror descent with soft active set.
Combines parametric initialization (sinusoidal families) with iterative gradient-based
refinement to minimize 2*n*max autocorrelation / (sum)^2.
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a, b=None):
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)
    need = a.size + b.size - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)
    return c[:need]


def _score_internal(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.inf
    s = float(a.sum())
    if s < 0.01 or not np.isfinite(s):
        return np.inf
    c = _fft_convolve(a)
    mx = float(c.max())
    if not np.isfinite(mx):
        return np.inf
    return 2.0 * n * mx / (s * s)


def _make_sine(n, amp, mode=1, phase=0.0):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * float(mode) * (i + 0.5) / float(n) + float(phase)
    a = 1.0 + float(amp) * np.sin(theta)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    a = a / s
    return a


def _make_two_harmonic(n, amp1, amp2, phase2=0.0):
    i = np.arange(n, dtype=np.float64)
    theta1 = 2.0 * np.pi * (i + 0.5) / float(n)
    theta2 = 2.0 * np.pi * 2.0 * (i + 0.5) / float(n) + float(phase2)
    a = 1.0 + float(amp1) * np.sin(theta1) + float(amp2) * np.sin(theta2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    a = a / s
    return a


def _make_three_harmonic(n, amp1, amp2, amp3, phase2=0.0, phase3=0.0):
    i = np.arange(n, dtype=np.float64)
    theta1 = 2.0 * np.pi * (i + 0.5) / float(n)
    theta2 = 2.0 * np.pi * 2.0 * (i + 0.5) / float(n) + float(phase2)
    theta3 = 2.0 * np.pi * 3.0 * (i + 0.5) / float(n) + float(phase3)
    a = 1.0 + float(amp1) * np.sin(theta1) + float(amp2) * np.sin(theta2) + float(amp3) * np.sin(theta3)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    a = a / s
    return a


def _gradient_from_weights(w, a):
    n = a.size
    rev_a = a[::-1]
    conv = _fft_convolve(w, rev_a)
    grad = 2.0 * conv[:n][::-1]
    return grad


def _optimize_mirror(a, budget_s, verbose=False):
    start = time.monotonic()
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return a, np.inf
    a = a / s
    n = a.size

    c = _fft_convolve(a)
    c_max = float(c.max())
    tau = max(c_max * 0.08, 1e-15)
    min_tau = c_max * 0.005
    eta = 5.0
    max_eta = 200.0
    min_eta = 0.05

    best = a.copy()
    best_score = _score_internal(best)

    it = 0
    max_iters = 200000
    last_improve = 0
    stall_count = 0
    tau_sched_factor = 0.96

    while time.monotonic() - start < budget_s and it < max_iters:
        it += 1

        if it % 2000 == 0 and verbose:
            print(f"  iter {it}, score={best_score:.8f}, tau={tau:.2e}, eta={eta:.2f}")

        c = _fft_convolve(a)
        c_max = float(c.max())
        if not np.isfinite(c_max):
            break

        scaled = (c - c_max) / tau
        scaled = np.clip(scaled, -60.0, 60.0)
        weights = np.exp(scaled)
        w_sum = weights.sum()
        if w_sum < 1e-30:
            break
        weights = weights / w_sum

        grad = _gradient_from_weights(weights, a)

        improved = False
        for factor in [1.0, 0.5, 0.25, 0.125]:
            try_eta = eta * factor
            if try_eta < min_eta:
                break
            exponent = -try_eta * grad
            exponent = np.clip(exponent, -6.0, 6.0)
            a_new = a * np.exp(exponent)
            a_new = np.maximum(a_new, 0.0)
            s_new = float(a_new.sum())
            if s_new < 0.01:
                continue
            a_new = a_new / s_new
            sc = _score_internal(a_new)
            if np.isfinite(sc) and sc < best_score - 1e-14:
                best = a_new.copy()
                best_score = sc
                a = a_new.copy()
                last_improve = it
                stall_count = 0
                improved = True
                tau = max(tau * tau_sched_factor, min_tau)
                eta = min(try_eta * 1.03, max_eta)
                break

        if not improved:
            eta = max(eta * 0.6, min_eta)
            stall_count += 1
            if stall_count > 100:
                tau = min(tau * 2.5, c_max * 0.5)
                eta = 5.0
                stall_count = 0
            if it - last_improve > 5000:
                break

    return best, best_score


def _optimize_mirror_v2(a, budget_s, verbose=False):
    """Mirror descent with adaptive active-set size (top-K hard active set)."""
    start = time.monotonic()
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        return a, np.inf
    a = a / s
    n = a.size

    eta = 3.0
    K = max(8, n // 64)
    best = a.copy()
    best_score = _score_internal(best)

    it = 0
    max_iters = 200000
    last_improve = 0

    while time.monotonic() - start < budget_s and it < max_iters:
        it += 1

        if it % 2000 == 0 and verbose:
            print(f"  v2 iter {it}, score={best_score:.8f}, eta={eta:.2f}, K={K}")

        c = _fft_convolve(a)
        if c.size <= K:
            active_idx = np.arange(c.size)
        else:
            active_idx = np.argpartition(c, -K)[-K:]

        # Compute gradient from top-K lags only
        grad = np.zeros(n, dtype=np.float64)
        for k in active_idx:
            k = int(k)
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            if j_max >= j_min:
                idx = np.arange(j_min, j_max + 1)
                grad[idx] += 2.0 * a[k - idx]

        grad /= float(len(active_idx))

        # Project gradient to preserve sum (remove mean)
        grad = grad - grad.mean()

        improved = False
        for factor in [1.0, 0.5, 0.25]:
            try_eta = eta * factor
            exponent = -try_eta * grad
            exponent = np.clip(exponent, -5.0, 5.0)
            a_new = a * np.exp(exponent)
            a_new = np.maximum(a_new, 0.0)
            s_new = float(a_new.sum())
            if s_new < 0.01:
                continue
            a_new = a_new / s_new
            sc = _score_internal(a_new)
            if np.isfinite(sc) and sc < best_score - 1e-14:
                best = a_new.copy()
                best_score = sc
                a = a_new.copy()
                last_improve = it
                improved = True
                eta = min(try_eta * 1.02, 100.0)
                # Gradually increase K
                if it % 5000 == 0 and K < n // 4:
                    K = min(K + n // 64, n // 4)
                break

        if not improved:
            eta = max(eta * 0.5, 0.01)
            if it - last_improve > 3000:
                break

    return best, best_score


def _perturb_sequence(a, strength=0.05, seed=None):
    if seed is not None:
        rng = np.random.RandomState(seed)
    else:
        rng = np.random.RandomState()
    noise = rng.randn(a.size) * strength
    a_new = a * (1.0 + noise)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s < 0.01:
        return a.copy()
    return a_new / s


def _safe_validate(arr):
    arr = np.asarray(arr, dtype=np.float64).ravel()
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return None
    return arr
