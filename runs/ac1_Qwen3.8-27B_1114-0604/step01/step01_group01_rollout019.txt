a_new = max(0, a_new)
a_new = a_new / sum(a_new)

This is simple but effective near the optimum.

"""
Search for non-negative sequences minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses multiplicative mirror descent with adaptive smoothed-max, spectral and
structured initializations, active-set local refinement, and multi-scale search
across sequence lengths.
"""

import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _convolve_full(a, b):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 4000000:
        return np.convolve(a, b)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)


def _normalize(a):
    a = np.asarray(a, dtype=float).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        return None
    return a / s


def _score(a):
    a = _normalize(a)
    if a is None:
        return float('inf')
    n = len(a)
    c = _convolve_full(a, a)
    if c.size == 0:
        return float('inf')
    return float(2.0 * n * np.max(c))


def _soft_weights(d, beta):
    d = np.asarray(d, dtype=float)
    n = d.size
    if n == 0:
        return np.array([], dtype=float)
    if beta < 1e-6:
        return np.full(n, 1.0 / n)
    ds = d - d.max()
    e = np.exp(beta * ds)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return e / s


def _smooth_obj_and_grad(a, beta):
    """
    Compute exact score, smoothed objective value, and gradient of smoothed obj.
    Works with normalized a (sum=1).
    """
    a = _normalize(a)
    if a is None:
        return float('inf'), float('inf'), None
    n = len(a)
    c = _convolve_full(a, a)
    if c.size == 0:
        return float('inf'), float('inf'), None
    c_max = c.max()
    exact = 2.0 * n * c_max

    d = n * c
    w = _soft_weights(d, beta)
    smooth_val = float(w @ d)

    # grad[i] = 2 * sum_j a[j] * w[j+i] for j=0..n-1
    # = 2 * (a * flip(w))[2n-2-i] for i=0..n-1
    flip_w = w[::-1]
    conv_aw = _convolve_full(a, flip_w)
    # indices needed: 2n-2, 2n-3, ..., n-1
    seg = conv_aw[n - 1:2 * n - 1]
    grad = 2.0 * seg[::-1]
    grad = np.maximum(grad, 0.0)

    return exact, smooth_val, grad


def _init_uniform(n):
    return np.ones(n) / n


def _init_two_block(n, frac, ratio):
    a = np.ones(n)
    idx = int(round(frac * n))
    if 0 < idx < n:
        a[idx:] = ratio
    return _normalize(a)


def _init_u_shape(n, alpha):
    t = np.linspace(0, 1, n)
    a = (t - 0.5) ** 2 * alpha + (1 - alpha)
    return _normalize(a)


def _init_gaussian_bumps(n, rng, K):
    a = np.zeros(n)
    t = np.linspace(0, 1, n)
    for _ in range(K):
        c = rng.uniform(0.1, 0.9)
        s = rng.uniform(0.02, 0.3)
        w = rng.uniform(0.5, 2.0)
        a += w * np.exp(-((t - c) / s) ** 2)
    return _normalize(a + 1e-10)


def _init_spectral(n, rng, n_phases=None):
    """Generate a sequence with approximately flat power spectrum."""
    if n_phases is None:
        n_phases = min(30, n // 4)
    best_a = None
    best_sc = float('inf')
    for _ in range(n_phases):
        # Random phases with conjugate symmetry
        phases = rng.uniform(0, 2 * np.pi, n)
        phases[0] = 0
        if n % 2 == 0:
            phases[n // 2] = 0
        if n > 1:
            for k in range(1, n // 2 + 1):
                if k < n - k and k < n:
                    phases[n - k] = -phases[k]
        mag = np.ones(n) / np.sqrt(n)
        A = mag * np.exp(1j * phases)
        a = np.real(np.fft.ifft(A))[:n]
        a = np.maximum(a, 0.0) + 1e-10
        a = a / a.sum()
        sc = _score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    return best_a


def _init_symmetric_two_block(n, ratio):
    a = np.ones(n)
    half = n // 2
    a[:half] = ratio
    a[-half:] = ratio
    return _normalize(a)


def _run_mirror_descent(p0, rng, time_limit, beta_start=20.0, beta_end=3000.0, verbose=False):
    """
    Multiplicative mirror descent on the simplex with adaptive smoothed max.
    Returns (best_sequence, best_score).
    """
    a = _normalize(p0)
    if a is None:
        return None, float('inf')
    n = len(a)
    floor = 1e-18 / n
    a = np.maximum(a, floor)
    a = a / a.sum()

    best = a.copy()
    best_score = _score(a)
    cur_exact, cur_smooth = _smooth_obj_and_grad(a, beta_start)
    if not np.isfinite(cur_smooth):
        return best, best_score

    beta = float(beta_start)
    alpha = 0.3
    deadline = time.monotonic() + time_limit
    stall = 0
    it = 0
    last_report = time.monotonic()

    while time.monotonic() < deadline:
        cur_exact, cur_smooth, grad = _smooth_obj_and_grad(a, beta)
        if grad is None or not np.isfinite(cur_smooth):
            break

        if cur_exact < best_score:
            best = a.copy()
            best_score = cur_exact
            stall = 0
            if verbose and time.monotonic() - last_report > 30:
                last_report = time.monotonic()
                print(f"  MD: n={n} it={it} score={best_score:.6f} beta={beta:.0f}")
        else:
            stall += 1

        if cur_smooth <= 0 or not np.isfinite(cur_smooth):
            break

        # Backtracking line search
        max_grad = grad.max()
        if max_grad < 1e-15:
            stall += 10
        else:
            grad_norm = grad / max_grad
            found = False
            for mult in [0.3, 0.7, 1.0, 1.5, 2.5, 4.0]:
                eta = alpha * mult
                if eta < 1e-6:
                    continue
                dec = np.exp(-eta * grad_norm)
                a_try = a * dec
                a_try = np.maximum(a_try, floor)
                s = a_try.sum()
                if s <= 0:
                    continue
                a_try = a_try / s
                _, smooth_try, _ = _smooth_obj_and_grad(a_try, beta)
                if np.isfinite(smooth_try) and smooth_try < cur_smooth * (1 - 1e-10):
                    a = a_try
                    cur_smooth = smooth_try
                    cur_exact = _score(a)
                    found = True
                    stall = 0
                    break
            if not found:
                alpha = max(1e-5, alpha * 0.7)
                stall += 1

        it += 1
        if it % 15 == 0:
            beta = min(beta_end, beta * 1.12)

        if stall > 50:
            # Perturb and restart
            a = best.copy()
            noise = np.exp(rng.normal(0, 0.05, n))
            a = a * noise
            a = np.maximum(a, floor)
            a = a / a.sum()
            alpha = 0.3
            beta = max(beta_start, beta * 0.5)
            cur_exact, cur_smooth = _smooth_obj_and_grad(a, beta)
            stall = 0

        if stall > 200:
            a = best.copy()
            cur_exact, cur_smooth = _smooth_obj_and_grad(a, beta)
            alpha = 0.3
            beta = beta_start
            stall = 0

    return best, best_score


def _run_pbfgs_descent(p0, time_limit, max_iter=200):
    """
    Projected subgradient descent with adaptive step size.
    Uses the exact subgradient at the max positions.
    """
    a = _normalize(p0)
    if a is None:
        return None, float('inf')
    n = len(a)
    best = a.copy()
    best_score = _score(a)
    deadline = time.monotonic() + time_limit
    eta = 0.5 / n
    patience = 0

    while time.monotonic() < deadline and patience < 300:
        c = _convolve_full(a, a)
        c_max = c.max()
        # Find active set (within 5% of max)
        threshold = c_max * 0.95
        active = np.where(c >= threshold)[0]
        w_active = 1.0 / len(active)

        # Compute subgradient
        grad = np.zeros(n)
        for k in active:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    grad[i] += w_active * 2.0 * a[j]

        # Projected gradient step
        a_new = a - eta * grad
        # Check if any negative
        if a_new.min() < -1e-4:
            eta *= 0.5
            a_new = a - eta * grad
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s <= 0:
            break
        a_new = a_new / s
        sc = _score(a_new)

        if sc < best_score * (1 - 1e-10):
            best = a_new.copy()
            best_score = sc
            a = a_new
            patience = 0
            eta = min(eta * 1.05, 2.0 / n)
        else:
            patience += 1
            eta *= 0.85

    return best, best_score


def _run_coordinate_search(p0, rng, time_limit, block_sizes=None):
    """
    Coordinate/block descent: try scaling random blocks of the sequence.
    """
    a = _normalize(p0)
    if a is None:
        return None, float('inf')
    n = len(a)
    best = a.copy()
    best_score = _score(a)
    cur_score = best_score
    deadline = time.monotonic() + time_limit
    stagnant = 0

    if block_sizes is None:
        block_sizes = [1, 2, 4, 8, 16, 32, 64, max(64, n // 32)]
    factors = [0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.6, 2.0]

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.2:
            w = 1
        elif r < 0.5:
            w = int(rng.integers(2, max(3, n // 64)))
        elif r < 0.8:
            w = int(rng.integers(max(2, n // 64), max(3, n // 16)))
        else:
            w = int(rng.integers(max(3, n // 16), max(4, n // 4)))
        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))

        base_score = cur_score
        base = a.copy()
        improved = False

        for f in factors:
            cand = base.copy()
            cand[start:start + w] *= f
            cand = _normalize(cand)
            if cand is None:
                continue
            sc = _score(cand)
            if sc < base_score * (1 - 1e-12):
                base = cand
                base_score = sc
                improved = True

        if improved:
            a = base
            cur_score = base_score
            if cur_score < best_score:
                best = a.copy()
                best_score = cur_score
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 100:
                # Random perturbation
                a = best.copy()
                noise = np.exp(rng.normal(0, 0.03, n))
                a = a * noise
                a = _normalize(a)
                if a is not None:
                    a = a * np.exp(rng.normal(0, 0.05, n))
                    a = _normalize(a)
                cur_score = _score(a) if a is not None else best_score
                stagnant = 0
            if stagnant > 500:
                a = best.copy()
                cur_score = best_score
                stagnant = 0

    return best, best_score


def _run_peak_equalization(p0, time_limit, max_iter=500):
    """
    Targeted peak equalization: iteratively reduce a-values that contribute
    most to the peak convolution values.
    """
    a = _normalize(p0)
    if a is None:
        return None, float('inf')
    n = len(a)
    best = a.copy()
    best_score = _score(a)
    deadline = time.monotonic() + time_limit

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break
        c = _convolve_full(a, a)
        c_max = c.max()
        if not np.isfinite(c_max):
            break

        # Find peaks (within 2% of max)
        threshold = c_max * 0.98
        peak_positions = np.where(c >= threshold)[0]

        if len(peak_positions) == 0:
            break

        # For each peak, find which a[i] contributes most
        reduce_mask = np.zeros(n)
        for k in peak_positions:
            for i in range(n):
                j = k - i
                if 0 <= j < n:
                    reduce_mask[i] += a[j]

        # Reduce a proportionally to its contribution to peaks
        if reduce_mask.max() > 0:
            scale = 1.0 - 0.02 * reduce_mask / reduce_mask.max()
            a_new = a * np.maximum(scale, 0.5)
            a_new = _normalize(a_new)
            if a_new is not None:
                sc = _score(a_new)
                if sc < best_score:
                    best = a_new.copy()
                    best_score = sc
                    a = a_new
                else:
                    # Try less aggressive reduction
                    scale2 = 1.0 - 0.005 * reduce_mask / reduce_mask.max()
                    a_new2 = a * np.maximum(scale2, 0.9)
                    a_new2 = _normalize(a_new2)
                    if a_new2 is not None:
                        sc2 = _score(a_new2)
                        if sc2 < best_score:
                            best = a_new2.copy()
                            best_score = sc2
                            a = a_new2

    return best, best_score


def _resample(a, new_n):
    """Resample a normalized vector to new length."""
    a = _normalize(a)
    if a is None:
        return None
    n = len(a)
    if n == new_n:
        return a.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(new_n) + 0.5) / new_n
    out = np.interp(x_new, x_old, a)
    return _normalize(out)
