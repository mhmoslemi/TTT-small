import numpy as np

def fft_convolution(a):
    n = len(a)
    need = 2*n - 1
    size = 1
    while size < need:
        size *= 2
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F*F, size)[:need]
    return np.maximum(b, 0.0)

def smooth_max(b, beta):
    m = np.max(b)
    if m <= 0:
        return m, np.ones(len(b))/len(b)
    x = beta * (b - m)
    x = np.clip(x, -700, 0)
    e = np.exp(x)
    Z = np.sum(e)
    w = e / Z
    G = m + np.log(Z) / beta
    return G, w

def corr_pw(p, w):
    """s_i = sum_j p[j] * w[i+j] for i=0..n-1"""
    n = len(p)
    m = len(w)
    need = n + m - 1
    size = 1
    while size < need:
        size *= 2
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m-n:m][::-1].copy()
    return np.maximum(s, 0.0)

def project_simplex(v, target=1.0):
    if v.size == 0:
        return v
    if target <= 0:
        return np.zeros_like(v)
    v = np.where(np.isfinite(v), v, 0.0)
    low = np.min(v) - 1
    high = np.max(v) + 1
    for _ in range(60):
        mid = 0.5 * (low + high)
        s = np.sum(np.maximum(v - mid, 0))
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0)
    s = np.sum(q)
    if s <= 0:
        return np.full(v.shape, target / v.size)
    return q * (target / s)

def fft_convolution(a):
    """Linear convolution a*a via FFT, clipped to [0, inf)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0]*a[0]])
    need = 2*n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F*F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    """Smooth maximum and softmax weights."""
    b = np.asarray(b, dtype=np.float64)
    m = float(np.max(b))
    if m <= 0.0 or beta <= 0:
        w = np.ones_like(b) / b.size
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -700.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z < 1e-300:
        w = np.zeros_like(b)
        w[int(np.argmax(b))] = 1.0
        return m, w
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w


def grad_convolution(p, w):
    """
    Gradient of smooth_max(conv(p,p)) w.r.t. p.
    Returns g_i = 2 * sum_j p[j] * w[i+j] (with appropriate indexing).
    """
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0:
        return np.empty(0)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    # s_i = C[m-1-i + 0]... let me be careful
    # C[k] = sum_i P[i] _ WR[k-i] where WR = rfft(w[::-1])
    # Actually: C = ifft(rfft(p) * rfft(w[::-1])) = p * w[::-1] (convolution)
    # So C[k] = sum_i p[i] * w_rev[k-i] = sum_i p[i] * w[m-1-(k-i)] = sum_i p[i] * w[m-1-k+i]
    # We want s_i = sum_j p[j] * w[i+j]
    # Let k = m-1-i: C[m-1-i] = sum_j p[j] * w[m-1-(m-1-i)+j] = sum_j p[j] * w[i+j]
    # So s_i = C[m-1-i] for i = 0, ..., n-1
    s = C[m-n:m][::-1].copy()
    return 2.0 * np.maximum(s, 0.0)


def project_simplex(v, target=1.0):
    """Euclidean projection onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    nv = v.size
    if nv == 0:
        return v
    if target <= 0:
        return np.zeros(nv)
    v = np.where(np.isfinite(v), v, 0.0)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5*(lo+hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5*(lo+hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0 or not np.isfinite(s):
        return np.full(nv, target/nv)
    return q * (target/s)

"""

for minimizing 2*n*max(conv(a,a))/(sum(a))^2 over non-negative sequences.

import numpy as np
import time

try:
    from scipy.optimize import minimize
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def fft_convolution(a):
    """Linear convolution a*a via FFT."""
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_w(b, beta):
    """Smooth max and softmax weights."""
    m = float(np.max(b))
    if m <= 0.0 or beta <= 0:
        w = np.ones(b.size, dtype=np.float64) / b.size
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -700.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z < 1e-300 or not np.isfinite(Z):
        w = np.zeros(b.size, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
        return m, w
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return G, w


def grad_corr(p, w):
    """g_i = 2 * sum_j p[j] * w[i+j], the gradient of smooth max conv."""
    n = p.size
    m = w.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([2.0 * p[0] * w[0]], dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n:m][::-1].copy()
    g = 2.0 * np.maximum(s, 0.0)
    return g


def project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    nv = v.size
    if nv == 0:
        return v
    if target <= 0:
        return np.zeros(nv, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(nv, target / nv, dtype=np.float64)
    return q * (target / s)


def normalize_to_one(p):
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0:
        return np.full(p.shape, 1.0 / p.size, dtype=np.float64)
    return p / s


def compute_score(p):
    """Score = 2*n*max(conv(p,p))/(sum(p))^2."""
    n = p.size
    s = float(np.sum(p))
    if s < 1e-15:
        return float('inf')
    b = fft_convolution(p)
    M = float(np.max(b))
    return 2.0 * n * M / (s * s)


def nesterov_optimize(p0, deadline, beta_schedule, max_total_iters=5000000):
    """
    Multi-stage Nesterov accelerated projected gradient.
    beta_schedule: list of (beta, max_iters_for_stage)
    Returns (best_p_normalized, best_score).
    """
    p = normalize_to_one(p0)
    n = p.size
    best_p = p.copy()
    best_score = compute_score(p)
    total_it = 0

    for beta, stage_max in beta_schedule:
        if time.monotonic() >= deadline or total_it >= max_total_iters:
            break

        y = p.copy()
        p_prev = p.copy()
        mu = 0.9
        step = 0.3
        no_improve = 0
        stage_it = 0
        val_prev = float('inf')

        while stage_it < stage_max and time.monotonic() < deadline:
            total_it += 1
            stage_it += 1

            # Evaluate at y (Nesterov look-ahead point)
            b = fft_convolution(y)
            G, w = smooth_max_w(b, beta)
            if not np.isfinite(G):
                y = p.copy()
                p_prev = p.copy()
                continue

            val = n * G
            m = float(np.max(b))
            score = 2.0 * n * m

            if score < best_score - 1e-14:
                best_score = score
                best_p = p.copy()
                no_improve = 0
            else:
                no_improve += 1

            if stage_it % 10000 == 0:
                print(f"    beta={beta:.0f} it={stage_it} val={val:.8f} score={score:.8f} best={best_score:.8f} step={step:.5f}")

            if no_improve > 50000:
                break

            # Compute gradient at y
            g = grad_corr(y, w)
            grad = g - float(np.mean(g))

            gn = float(np.linalg.norm(grad))
            if gn < 1e-16:
                break

            # Normalize step by gradient magnitude for scale invariance
            alpha = step / max(gn, 1e-10)

            # Nesterov update
            p_new = project_simplex(y - alpha * grad)
            y_new = p_new + mu * (p_new - p)
            y_new = project_simplex(y_new)

            # Quick check: evaluate smooth value at y_new
            b2 = fft_convolution(y_new)
            G2, _ = smooth_max_w(b2, beta)
            val_new = n * G2 if np.isfinite(G2) else float('inf')

            if val_new < val_prev * (1.0 - 1e-14) or val_new < val:
                p = p_new
                y = y_new
                p_prev = p
                val_prev = val_new
                if step < 15.0:
                    step = min(15.0, step * 1.02)
            else:
                step *= 0.5
                if step < 1e-6:
                    # Reset momentum and restart
                    y = p.copy()
                    p_prev = p.copy()
                    mu = 0.0
                    step = 0.3
                    val_prev = n * G
                    if stage_it % 5000 == 0:
                        print(f"      restart at beta={beta:.0f} it={stage_it}")

    return best_p, best_score


def subgradient_polish(p, deadline, max_rounds=500):
    """
    Polish using subgradient of the true max (not smooth approximation).
    At each step, identify active lags and take a step to reduce them.
    """
    p = normalize_to_one(p)
    n = p.size
    best_p = p.copy()
    best_score = compute_score(p)

    for rnd in range(max_rounds):
        if time.monotonic() >= deadline:
            break

        b = fft_convolution(p)
        M = float(np.max(b))
        score = 2.0 * n * M

        if score < best_score - 1e-14:
            best_score = score
            best_p = p.copy()

        # Find lags within 2% of max (active constraints)
        threshold = 0.98 * M
        active = np.where(b >= threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(b)])

        # Subgradient: average gradient over active lags
        # grad_k[i] = 2*a[k-i] if 0<=k-i<n, (plus a[i] if k=2i)
        # Simplified: just use the smooth max weights on active lags
        w_active = np.zeros(2 * n - 1, dtype=np.float64)
        w_active[active] = 1.0 / len(active)

        g = grad_corr(p, w_active)
        grad = g - float(np.mean(g))

        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break

        # Adaptive step
        alpha = 0.1 / gn
        p_trial = project_simplex(p - alpha * grad)
        b_trial = fft_convolution(p_trial)
        M_trial = float(np.max(b_trial))
        score_trial = 2.0 * n * M_trial

        if score_trial < score - 1e-15 * M:
            p = p_trial
        else:
            # Try smaller step
            alpha *= 0.1
            p_trial = project_simplex(p - alpha * grad)
            b_trial = fft_convolution(p_trial)
            M_trial = float(np.max(b_trial))
            score_trial = 2.0 * n * M_trial
            if score_trial < score - 1e-15 * M:
                p = p_trial
            else:
                if rnd > 0:
                    break

    return best_p, best_score


def make_starts(n, rng):
    """Generate diverse starting sequences of length n, all normalized to sum 1."""
    starts = []
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)

    # Uniform
    starts.append(np.ones(n, dtype=np.float64) / n)

    # Cosine bell (max at center)
    c = np.cos(np.pi * (x - 0.5)) ** 2
    c = np.maximum(c, 1e-8)
    starts.append(c / c.sum())

    # Exponential decay
    for alpha in (1.5, 3.0):
        e = np.exp(-alpha * x)
        starts.append(e / e.sum())

    # Edge-weighted (high at boundaries)
    edge = 1.0 + 0.8 * np.cos(np.pi * x) ** 2
    edge = np.maximum(edge, 0.05)
    starts.append(edge / edge.sum())

    # Sine power
    for pw in (1.0, 2.0, 3.0):
        s = np.sin(np.pi * x) ** pw
        s = np.maximum(s, 0.01)
        starts.append(s / s.sum())

    # Asymmetric smooth
    a = np.exp(-2.0 * x) + 0.3 * np.exp(-0.5 * x)
    starts.append(a / a.sum())

    # Smooth random (low frequency)
    freqs = [1, 2, 3]
    phases = rng.uniform(0, 2 * np.pi, size=3)
    sig = np.zeros(n, dtype=np.float64)
    for f, ph in zip(freqs, phases):
        sig += np.cos(2 * np.pi * f * x + ph)
    sig = np.tanh(2.0 * sig) + 1.0
    sig = np.maximum(sig, 0.01)
    starts.append(sig / sig.sum())

    # Polynomial
    poly = 1.0 - 0.7 * (2 * x - 1) ** 2
    poly = np.maximum(poly, 0.05)
    starts.append(poly / poly.sum())

    return starts


def block_optimize(n, deadline, rng):
    """
    Block-constant parameterized optimization with L-BFGS-B.
    Returns best block-constant sequence.
    """
    if not HAVE_SCIPY:
        return None, float('inf')

    best_score = float('inf')
    best_p = np.ones(n) / n

    for B in (16, 32, 64, 128, 256):
        if time.monotonic() >= deadline:
            break
        if B >= n or n % B != 0:
            continue

        L = n // B

        def obj_and_grad(c):
            c = np.asarray(c, dtype=np.float64)
            z = np.empty(B, dtype=np.float64)
            z[:-1] = c
            z[-1] = 0.0
            z = z - np.max(z)
            wb = np.exp(z)
            s_wb = float(np.sum(wb))
            if s_wb <= 0 or not np.isfinite(s_wb):
                return 1e10, np.zeros(B - 1)
            wb /= s_wb
            p = np.repeat(wb / L, L)

            b = fft_convolution(p)
            G, w = smooth_max_w(b, 500.0)
            if not np.isfinite(G):
                return 1e10, np.zeros(B - 1)

            gv = grad_corr(p, w)
            Gsum = gv.reshape(B, L).sum(axis=1)
            Wavg = float(np.dot(wb, Gsum))
            gradz = (wb / L) * (Gsum - Wavg)
            grad = n * gradz[:-1]
            grad = np.where(np.isfinite(grad), grad, 0.0)
            return n * G, grad

        # Try a few initializations
        m = B - 1
        inits = [np.zeros(m, dtype=np.float64)]
        xt = np.linspace(0, 1, m)
        inits.append(2.0 * (xt - 0.5))
        inits.append(-2.0 * (xt - 0.5))
        inits.append(np.cos(2 * np.pi * xt))
        inits.append(1.0 * rng.normal(0, 1, m))

        bounds = [(-5.0, 5.0)] * m
        best_B_score = float('inf')
        best_c = np.zeros(m)

        for c0 in inits:
            if time.monotonic() >= deadline:
                break
            try:
                res = minimize(obj_and_grad, c0, method='L-BFGS-B', jac=True,
                               bounds=bounds, options={'maxiter': 60, 'ftol': 1e-10, 'gtol': 1e-5})
                c = np.clip(np.asarray(res.x, dtype=np.float64), -5, 5)
                sc = obj_and_grad(c)[0] / n  # normalize
                if sc < best_B_score:
                    best_B_score = sc
                    best_c = c.copy()
            except Exception:
                pass

        if np.isfinite(best_B_score):
            z = np.empty(B, dtype=np.float64)
            z[:-1] = best_c
            z[-1] = 0.0
            z = z - np.max(z)
            wb = np.exp(z)
            wb /= wb.sum()
            p = np.repeat(wb / L, L)
            sc = compute_score(p)
            if sc < best_score:
                best_score = sc
                best_p = p.copy()
                print(f"  block B={B}: score={sc:.8f}")

    return best_p, best_score
