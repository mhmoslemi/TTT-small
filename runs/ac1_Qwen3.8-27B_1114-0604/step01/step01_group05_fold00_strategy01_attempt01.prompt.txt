def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

Minimize 2*n*max(a*a)/(sum(a))^2 for nonnegative sequences.

The objective is scale invariant, so we normalize sum(a)=1 and minimize

import numpy as np
import time

try:
    from scipy.optimize import minimize
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False

LOW_BETA = 20000.0
HIGH_BETA = 200000.0
FULL_BETA = 1000000.0


def fft_convolution(a):
    """Linear convolution a*a, clipped to remove tiny FFT negative roundoff."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)

    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1

    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    b = np.ascontiguousarray(b, dtype=np.float64)
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    """Smooth max of b and normalized softmax weights."""
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float("inf"), np.empty(0, dtype=np.float64)

    m = float(np.max(b))
    if not np.isfinite(m):
        return float("inf"), np.zeros_like(b, dtype=np.float64)

    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b, dtype=np.float64)
        w /= float(b.size)
        return m, w

    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))

    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0

    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w


def fast_score(p):
    """Fast local objective using FFT."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float("inf")

    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float("inf")

    b = fft_convolution(p)
    if b.size == 0:
        return float("inf")

    M = float(np.max(b))
    if not np.isfinite(M):
        return float("inf")

    return float(2.0 * n * M / (S * S))


def corr_pw(p, w):
    """
    Return s_i = sum_j p_j w_{i+j}, for i=0..n-1.
    This is used for the derivative of max(p*p) with softmax weights w.
    """
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    if n == 1:
        return np.array([float(np.dot(p, w[:1]))], dtype=np.float64)

    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1

    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]

    # s_i = C[m-1-i], i=0..n-1
    s = C[m - n : m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)


def project_simplex(v, target=1.0):
    """Project a vector onto {x >= 0, sum(x)=target}."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    if target <= 0.0:
        return np.zeros_like(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(v.shape, target / float(v.size), dtype=np.float64)

    low = float(np.min(v)) - 1.0
    high = float(np.max(v)) + 1.0

    for _ in range(50):
        mid = 0.5 * (low + high)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            low = mid
        else:
            high = mid

    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))

    if s <= 0.0 or not np.isfinite(s):
        q = np.full(v.shape, target / float(v.size), dtype=np.float64)
    else:
        q *= target / s

    return q


def normalize_simplex(p):
    """Clip to nonnegative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size), dtype=np.float64)
    p *= 1.0 / s
    return p


def block_to_p(n, B, c):
    """Convert relative block log-weights c (length B-1) to full sequence."""
    L = n // B
    z = np.empty(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        tmp = np.zeros(B - 1, dtype=np.float64)
        m = min(tmp.size, c.size)
        tmp[:m] = c[:m]
        c = tmp
        z[:-1] = c
    z[-1] = 0.0

    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0.0 or not np.isfinite(s):
        wb = np.ones(B, dtype=np.float64) / float(B)
    else:
        wb /= s

    H = wb / float(L)
    return np.repeat(H, L)


def block_objective(c, n, B, beta):
    """
    Objective and gradient for block-constant sequences.
    The returned objective is n * smooth_max(p*p), so the scale is O(1).
    """
    L = n // B
    z = np.empty(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        tmp = np.zeros(B - 1, dtype=np.float64)
        m = min(tmp.size, c.size)
        tmp[:m] = c[:m]
        c = tmp
        z[:-1] = c
    z[-1] = 0.0

    if not np.all(np.isfinite(z)):
        return float("inf"), np.zeros(B - 1, dtype=np.float64)

    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0.0 or not np.isfinite(s):
        wb = np.ones(B, dtype=np.float64) / float(B)
    else:
        wb /= s

    H = wb / float(L)
    p = np.repeat(H, L)

    b = fft_convolution(p)
    G, w = smooth_max_and_weights(b, beta)
    if not np.isfinite(G):
        return float("inf"), np.zeros(B - 1, dtype=np.float64)

    svec = corr_pw(p, w)
    svec = np.maximum(svec, 0.0)
    g = 2.0 * svec

    Gsum = g.reshape(B, L).sum(axis=1)
    Wavg = float(np.dot(wb, Gsum))
    gradz = (wb / float(L)) * (Gsum - Wavg)

    grad = n * gradz[:-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(n * G), grad.astype(np.float64)


def run_block_minimize(c0, n, B, beta, bounds):
    """Run L-BFGS-B on the block representation, robustly."""
    if not HAVE_SCIPY:
        return np.asarray(c0, dtype=np.float64)

    c0 = np.asarray(c0, dtype=np.float64).reshape(-1)
    if c0.size != max(0, B - 1):
        tmp = np.zeros(max(0, B - 1), dtype=np.float64)
        m = min(tmp.size, c0.size)
        tmp[:m] = c0[:m]
        c0 = tmp

    c0 = np.clip(c0, -6.0, 6.0)

    options = {
        "maxiter": 90,
        "ftol": 1e-10,
        "gtol": 1e-5,
        "maxls": 40,
        "maxcor": 10,
    }

    try:
        res = minimize(
            block_objective,
            c0,
            args=(n, B, beta),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options=options,
        )
        c = np.asarray(res.x, dtype=np.float64)
    except Exception:
        c = c0

    if not np.all(np.isfinite(c)):
        c = c0

    c = np.clip(c, -6.0, 6.0)
    return c


def block_initial_list(B, rng):
    """Several simple initial block weight directions."""
    inits = []
    if B <= 1:
        return [np.empty(0, dtype=np.float64)]

    m = B - 1
    inits.append(np.zeros(m, dtype=np.float64))

    x = np.linspace(0.0, 1.0, m, dtype=np.float64)
    for a in (-3.0, 3.0):
        inits.append(a * (x - 0.5))

    c = np.zeros(m, dtype=np.float64)
    half = m // 2
    if half > 0:
        c[:half] = 2.0
    if m - half > 0:
        c[half:] = -2.0
    inits.append(c)

    t = np.linspace(0.0, 2.0 * np.pi, m, dtype=np.float64)
    inits.append(np.cos(t))
    inits.append(np.sin(t))

    inits.append(0.8 * rng.normal(0.0, 1.0, m))
    inits.append(1.5 * rng.normal(0.0, 1.0, m))

    return inits


def param_block_search(n, deadline, rng, max_B=128):
    """Optimize piecewise-constant log-weights at increasing resolution."""
    best_p = np.full(n, 1.0 / float(n), dtype=np.float64)
    best_score = fast_score(best_p)

    if not HAVE_SCIPY or n < 8:
        return best_p, best_score

    candidates = []
    for B in (8, 16, 32, 64, 128, 256):
        if B <= max_B and B < n and (n % B) == 0:
            candidates.append(B)

    for B in candidates:
        if time.monotonic() >= deadline:
            break

        bounds = [(-6.0, 6.0)] * (B - 1)
        best_c = np.zeros(B - 1, dtype=np.float64)
        best_B_score = float("inf")

        inits = block_initial_list(B, rng)

        # Coarse smooth-max stage.
        for c0 in inits:
            if time.monotonic() >= deadline:
                break
            c = run_block_minimize(c0, n, B, LOW_BETA, bounds)
            p = block_to_p(n, B, c)
            sc = fast_score(p)
            if np.isfinite(sc) and sc < best_B_score:
                best_B_score = sc
                best_c = c.copy()

        # Sharper stage from the best coarse solution.
        if time.monotonic() < deadline and np.isfinite(best_B_score):
            c = run_block_minimize(best_c, n, B, HIGH_BETA, bounds)
            p = block_to_p(n, B, c)
            sc = fast_score(p)
            if np.isfinite(sc) and sc < best_B_score:
                best_B_score = sc
                best_c = c.copy()
            p = block_to_p(n, B, best_c)
        else:
            p = block_to_p(n, B, best_c)

        if np.isfinite(best_B_score) and best_B_score < best_score:
            best_score = best_B_score
            best_p = p.copy()
            print(f"block B={B} best_score={best_score:.8f}")

    return best_p, best_score


def structured_starts(n, rng, max_count=8):
    """A few analytic/random starting profiles for full refinement."""
    starts = []

    p = np.full(n, 1.0 / float(n), dtype=np.float64)
    starts.append(p)

    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    for a in (-2.5, -1.5, 1.5, 2.5):
        z = a * (x - 0.5)
        p = np.exp(z)
        p /= p.sum()
        starts.append(p)

    for eps in (0.35, 0.65):
        p = np.ones(n, dtype=np.float64)
        if n >= 2:
            p[: n // 2] = 1.0 + eps
            p[n // 2 :] = max(1e-3, 1.0 - eps)
        p = np.maximum(p, 1e-8)
        p /= p.sum()
        starts.append(p)

    size = 1
    while size < n:
        size <<= 1

    for _ in range(2):
        zc = rng.normal(size=size) + 1j * rng.normal(size=size)
        z = np.real(np.fft.ifft(zc))[:n]
        z = z - float(np.mean(z))
        z = 1.5 * np.tanh(z)
        p = np.exp(z)
        p /= p.sum()
        starts.append(p)

    return starts[:max_count]


def choose_best_starts(starts, count):
    """Return up to count starts with the lowest fast_score."""
    scored = []
    for p in starts:
        sc = fast_score(p)
        if np.isfinite(sc):
            scored.append([sc, p])

    top = []
    for _ in range(count):
        if not scored:
            break
        bi = 0
        for i in range(1, len(scored)):
            if scored[i][0] < scored[bi][0]:
                bi = i
        top.append(scored[bi][1])
        del scored[bi]

    return top


def full_refine(p0, deadline, rng, beta=FULL_BETA, max_iter=1000000):
    """Full-length projected-gradient refinement on the simplex."""
    p = normalize_simplex(p0)
    n = p.size
    if n == 0:
        return p, float("inf")

    # Break perfect symmetry if starting at uniform.
    mean_p = float(np.mean(p))
    if n > 1 and float(np.max(np.abs(p - mean_p))) < 1e-13:
        t = 0.123 * np.arange(n, dtype=np.float64)
        p = normalize_simplex(p * (1.0 + 1e-4 * np.sin(t)))

    best_p = p.copy()
    best_score = fast_score(p)

    step = 1.0
    no_improve = 0
    it = 0

    while it < max_iter and time.monotonic() < deadline:
        it += 1

        b = fft_convolution(p)
        G, w = smooth_max_and_weights(b, beta)
        if not np.isfinite(G):
            break

        cur_smooth = float(n) * G
        m = float(np.max(b))
        cur_score = float(2.0 * n * m)

        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 250 == 0:
            print(f"full refine it={it} best={best_score:.8f} step={step:.3e}")

        if no_improve > 2500:
            break

        svec = corr_pw(p, w)
        svec = np.maximum(svec, 0.0)
        g = 2.0 * svec
        grad = g - float(np.mean(g))

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-14:
            break

        accepted = False
        base = step

        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001):
            alpha = base * mult
            q = project_simplex(p - alpha * grad, 1.0)

            qb = fft_convolution(q)
            m2 = float(np.max(qb))
            G2, _ = smooth_max_and_weights(qb, beta)
            if not np.isfinite(G2):
                continue

            if float(n) * G2 < cur_smooth - 1e-13:
                p = q
                cur_smooth = float(n) * G2
                cur_score = float(2.0 * n * m2)
                accepted = True

                if cur_score < best_score - 1e-15:
                    best_score = cur_score
                    best_p = p.copy()
                    no_improve = 0

                break

        if not accepted:
            step *= 0.5
            if step < 1e-7:
                break
        else:
            if step < 2.0:
                step = min(2.0, step * 1.25)

    return best_p, best_score


def official_score(seq):
    """Call the authoritative evaluator, with exception safety."""
    try:
        val = evaluate_sequence(seq)
        val = float(val)
    except Exception:
        return float("inf")
    if np.isfinite(val):
        return val
    return float("inf")


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search function.

    Returns a list of nonnegative floats.  The search length is chosen as a
    power-of-two-ish size (8192 under normal budget) to balance attack surface
    and final O(n^2) evaluator cost.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42

    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 0.0:
        return [1.0]

    reserve = 90.0 if budget_s > 900.0 else 30.0
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - reserve)

    if budget_s < 120.0:
        N = 2048
    elif budget_s < 600.0:
        N = 4096
    else:
        N = 8192

    rng = np.random.default_rng(seed)

    best_p = np.full(N, 1.0 / float(N), dtype=np.float64)
    best_score = fast_score(best_p)
    print(f"initial N={N} score={best_score:.8f}")

    # ---- Phase 1: parametric block optimization ----
    if deadline - time.monotonic() > 20.0:
        if budget_s >= 2000.0:
            max_B = 256
        elif budget_s >= 1000.0:
            max_B = 128
        else:
            max_B = 64

        param_deadline = min(deadline, time.monotonic() + max(60.0, 0.40 * budget_s))
        p_param, sc_param = param_block_search(N, param_deadline, rng, max_B=max_B)

        if np.isfinite(sc_param) and sc_param < best_score:
            best_score = sc_param
            best_p = p_param.copy()
            print(f"after param block search score={best_score:.8f}")

    # ---- Phase 2: full refinement from several good starts ----
    candidate_starts = [best_p]

    if time.monotonic() < deadline - 60.0:
        for p0 in structured_starts(N, rng, max_count=8):
            if time.monotonic() >= deadline - 40.0:
                break
            candidate_starts.append(p0)

    top_starts = choose_best_starts(candidate_starts, 3)
    fracs = (0.55, 0.30, 0.15)

    for i, p0 in enumerate(top_starts):
        remaining = deadline - time.monotonic() - 30.0
        if remaining <= 20.0:
            break

        frac = fracs[i] if i < len(fracs) else 0.1
        alloc = remaining * frac
        sub_deadline = time.monotonic() + alloc

        p_fit, sc_fit = full_refine(p0, sub_deadline, rng, beta=FULL_BETA)
        if np.isfinite(sc_fit) and sc_fit < best_score:
            best_score = sc_fit
            best_p = p_fit.copy()
            print(f"full refine start {i} score={best_score:.8f}")

    # ---- Final cleanup and authoritative check ----
    final_p = normalize_simplex(best_p)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_s = float(np.sum(final_p))
    if final_s > 0.0 and np.isfinite(final_s):
        final_p *= 1.0 / final_s

    final_seq = [float(x) for x in final_p]

    # The uniform sequence is a valid score-2 fallback.
    off = official_score(final_seq)
    if not np.isfinite(off) or off > 2.0 + 1e-9:
        return [1.0] * N

    return final_seq
