```python
def optimize_coordinate(a, i, n, n_iter=20):
    """Optimize coordinate i of a (normalized to sum 1) while keeping others proportional."""
    old_v = a[i]
    
    def f(v):
        """Evaluate objective with a[i] = v, others scaled to maintain sum=1."""
        if v < 0 or v > 1:
            return np.inf
        if abs(v - old_v) < 1e-15:
            # No change
            b = full_conv(a)
            return 2.0 * n * float(np.max(b))
        c = (1.0 - v) / (1.0 - old_v) if old_v < 1.0 else 0.0
        a_new = a.copy()
        a_new[i] = v
        mask = np.ones(n, dtype=bool)
        mask[i] = False
        a_new[mask] *= c
        b = full_conv(a_new)
        return 2.0 * n * float(np.max(b))
    
    # Golden section search on [0, 1]
    phi = (np.sqrt(5) - 1) / 2  # golden ratio conjugate ≈ 0.618
    lo = 0.0
    hi = 1.0
    x1 = hi - phi * (hi - lo)
    x2 = lo + phi * (hi - lo)
    f1 = f(x1)
    f2 = f(x2)
    
    for _ in range(n_iter):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - phi * (hi - lo)
            f1 = f(x1)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + phi * (hi - lo)
            f2 = f(x2)
    
    best_v = (lo + hi) / 2
    best_f = f(best_v)
    
    # Update a
    if old_v < 1.0:
        c = (1.0 - best_v) / (1.0 - old_v)
        mask = np.ones(n, dtype=bool)
        mask[i] = False
        a[mask] *= c
    a[i] = best_v
    
    return a, best_f

    """Solve the LP to find best direction b."""
    # Variables: b_0, ..., b_{n-1} >= 0
    # Objective: maximize sum(b) => minimize -sum(b)
    c = -np.ones(n)

    A_ub = np.zeros((2*n-1, n))

    bounds = [(0, None)] * n

    if res.success:
        return res.x
    else:
        return np.ones(n) * M / (2*n)  # fallback

```python
"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding
with adaptive step size, combined with coordinate descent polishing.
The LP finds the best non-negative direction b that keeps conv(f,b) bounded,
effectively "filling in" valleys of the autocorrelation.
"""

import numpy as np
from scipy.optimize import linprog
import time


def full_conv(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    n = a.size
    m = b.size
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
    return np.fft.irfft(fa * fb, fsize)[:size]


def build_conv_matrix(f):
    """Build matrix A such that A @ b = conv(f, b)."""
    n = f.size
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        A[k, j_lo:j_hi + 1] = f[j_lo:j_hi + 1]
    return A


def score_normalized(f):
    """Compute score assuming f is normalized to sum=1."""
    n = f.size
    b = full_conv(f)
    return 2.0 * n * float(np.max(b))


def lp_find_direction(f, M):
    """Solve LP to find best direction b."""
    n = f.size
    c = -np.ones(n)
    A_ub = build_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'dual_feasibility_tolerance': 1e-8})
    if res.success:
        return np.maximum(res.x, 0)
    return None

    f = np.maximum(f, 0)
    if f.sum() < 1e-15:
        f = np.ones(n) / n
    f = f / f.sum()  # normalize to sum=1
    
    best_f = f.copy()
    best_M = float(np.max(full_conv(f)))  # score = 2n * best_M
    
    t = 0.1  # step size

    last_improve = 0
    stall_count = 0

    while time.monotonic() - t0 < time_budget:
        b_conv = full_conv(f)
        M = float(np.max(b_conv))
        
        if M < best_M:
            best_M = M
            best_f = f.copy()
            last_improve = n_iter
        
        # Find direction
        b_dir = lp_find_direction(f, M)
        if b_dir is None:

            b_dir /= b_dir.sum()

        b_dir_sum = b_dir.sum()
        if b_dir_sum < 1e-15:
            b_dir = np.ones(n) / n
        else:

        f_new = np.maximum(f_new, 0)
        f_new_sum = f_new.sum()
        if f_new_sum < 1e-15:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / f_new_sum

def init_u_shape(n):
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i).astype(np.float64) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_two_cluster(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.1 + 0.2 * rng.random())
    c2 = n * (0.7 + 0.2 * rng.random())
    sigma = n * (0.05 + 0.1 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_cosine(n, rng):
    i = np.arange(n, dtype=np.float64)
    freq = 1.0 + rng.integers(0, 4)
    a = (1.0 + 0.5 * np.cos(2.0 * np.pi * freq * i / max(n - 1, 1))) ** 2
    a += 1e-6
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.01
    return a / a.sum()


def init_sided(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / c) ** 2 + 0.05
    return a / a.sum()


def coordinate_polish(f, time_budget, rng):
    """Coordinate descent polish."""
    n = f.size
    f = f.copy()
    best_f = f.copy()
    best_M = float(np.max(full_conv(f)))
    
    t0 = time.monotonic()
    sweep = 0
    while time.monotonic() - t0 < time_budget:
        improved = False
        # Randomize coordinate order
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_budget:
                break
            old_v = f[i]
            s_other = f.sum() - old_v
            if s_other < 1e-15:
                continue
            
            # Try several candidate values
            candidates = [old_v * 0.5, old_v * 0.8, old_v, old_v * 1.2, old_v * 1.5, 0.0]
            # Also try values based on "equalizing" the convolution
            candidates.extend([s_other * 0.01, s_other * 0.05, s_other * 0.1])
            
            best_v = old_v
            best_fval = np.inf
            
            for v in candidates:
                if v < 0 or v > 10:
                    continue
                f[i] = v
                total = f.sum()
                if total < 1e-15:
                    f[i] = old_v
                    continue
                # Score is scale-invariant, so no need to normalize
                M_v = float(np.max(full_conv(f)))
                fval = 2.0 * n * M_v / (total * total)
                if fval < best_fval:
                    best_fval = fval
                    best_v = v
            
            f[i] = best_v
            f = f / f.sum()
            
            if best_v != old_v:
                improved = True
        
        # Check overall improvement
        cur_M = float(np.max(full_conv(f)))
        if cur_M < best_M:
            best_M = cur_M
            best_f = f.copy()
        
        sweep += 1
        if not improved and sweep > 3:
            break
    
    best_score = 2.0 * n * best_M
    return best_f, best_score

The key insight is that this can be approached via an LP-based direction-finding method: at each step, find the "most spread out" non-negative vector `b` whose convolution with the current `f` stays below the peak, then move toward `b`. This is more principled than subgradient descent and should converge to better local minima.

```python
"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding
(Safonov-Pinkserov style), adaptive step size, coordinate descent polishing,
multiple sequence lengths, and diverse initializations.
The LP at each step finds the best non-negative direction b that keeps
conv(f,b) bounded by the current peak, effectively flattening the autocorrelation.
"""

import numpy as np
from scipy.optimize import linprog
import time


def full_conv(a, b=None):
    """Full linear convolution via FFT."""
    if b is None:
        b = a
    n = a.size
    m = b.size
    size = n + m - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
    return np.fft.irfft(fa * fb, fsize)[:size]


def build_conv_matrix(f):
    """Build matrix A such that A @ b = conv(f, b). A has shape (2n-1, n)."""
    n = f.size
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_hi >= j_lo:
            A[k, j_lo:j_hi + 1] = f[j_lo:j_hi + 1]
    return A


def score_normalized(f):
    """Compute score assuming f is normalized to sum=1."""
    n = f.size
    b = full_conv(f)
    return 2.0 * n * float(np.max(b))


def lp_find_direction(f, M):
    """Solve LP: max sum(b) s.t. conv(f,b)_k <= M, b >= 0."""
    n = f.size
    c = -np.ones(n)
    A_ub = build_conv_matrix(f)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None


def lp_optimize(f_init, time_budget, rng, verbose=True):
    """
    LP-based optimization of the autocorrelation peak.
    f_init: initial vector (normalized to sum=1).
    Returns (best_f, best_score).
    """
    n = f_init.size
    f = f_init.copy().astype(np.float64)
    f = np.maximum(f, 0)
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_f = f.copy()
    best_M = float(np.max(full_conv(f)))

    t = 0.15
    n_iter = 0
    last_improve = 0
    consecutive_worse = 0

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        b_conv = full_conv(f)
        M = float(np.max(b_conv))

        if M < best_M - 1e-15:
            best_M = M
            best_f = f.copy()
            last_improve = n_iter
            consecutive_worse = 0

        # Find LP direction
        b_dir = lp_find_direction(f, M)
        if b_dir is None:
            b_dir = np.ones(n) / n
        else:
            b_sum = b_dir.sum()
            if b_sum < 1e-15:
                b_dir = np.ones(n) / n
            else:
                b_dir = b_dir / b_sum

        # Safe step size: t < M / max(conv(b,b))
        b_conv_dir = full_conv(b_dir)
        max_b2 = float(np.max(b_conv_dir))
        if max_b2 > 1e-15:
            t_safe = 0.85 * M / max_b2
        else:
            t_safe = 1.0
        t_use = min(t, t_safe, 0.4)
        if t_use < 0.005:
            t_use = 0.005

        # Line search: try t_use, t_use/2, t_use/4
        best_new_f = None
        best_new_M = M + 1e-10
        for frac in [1.0, 0.5, 0.25]:
            tt = t_use * frac
            f_test = (1 - tt) * f + tt * b_dir
            f_test = np.maximum(f_test, 0)
            f_sum = f_test.sum()
            if f_sum < 1e-15:
                continue
            f_test = f_test / f_sum
            new_M = float(np.max(full_conv(f_test)))
            if new_M < best_new_M:
                best_new_M = new_M
                best_new_f = f_test

        if best_new_f is not None and best_new_M <= M + 1e-12:
            f = best_new_f
            consecutive_worse = 0
            # Increase step if improving
            if best_new_M < M - 1e-8:
                t = min(t * 1.03, 0.3)
        else:
            consecutive_worse += 1
            t = max(t * 0.9, 0.02)
            if consecutive_worse > 30:
                # Perturb and restart
                f = np.maximum(best_f + 0.03 * rng.standard_normal(n), 0)
                f = f / f.sum()
                t = 0.05 + 0.1 * rng.random()
                consecutive_worse = 0
                last_improve = n_iter

        n_iter += 1

        # Periodic large restart if stalled
        if n_iter - last_improve > 300:
            f = np.maximum(best_f + 0.1 * rng.standard_normal(n), 0)
            f = f / f.sum()
            t = 0.03 + 0.07 * rng.random()
            last_improve = n_iter
            consecutive_worse = 0
            if n_iter > 5000 and (n_iter - last_improve) > 100:
                break

    best_score = 2.0 * n * best_M
    if verbose:
        print(f"    [lp] n={n}, iters={n_iter}, score={best_score:.6f}")
    return best_f, best_score


def init_u_shape(n):
    """U-shaped: high at edges, low center."""
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i).astype(np.float64) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_two_cluster(n, rng):
    """Two Gaussian clusters."""
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.1 + 0.25 * rng.random())
    c2 = n * (0.65 + 0.25 * rng.random())
    sigma = n * (0.04 + 0.12 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_three_cluster(n, rng):
    """Three Gaussian clusters."""
    i = np.arange(n, dtype=np.float64)
    cs = sorted(rng.uniform(0.1, 0.9, size=3) * n)
    sigma = n * (0.04 + 0.08 * rng.random())
    a = np.zeros(n)
    for c in cs:
        a += np.exp(-((i - c) / sigma) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_sided(n):
    """Quadratic: high at edges."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / c) ** 2 + 0.05
    return a / a.sum()


def init_cosine(n, rng):
    """Cosine-modulated."""
    i = np.arange(n, dtype=np.float64)
    freq = 1.0 + rng.integers(0, 4)
    a = (1.0 + 0.4 * np.cos(2.0 * np.pi * freq * i / max(n - 1, 1))) ** 2
    a += 1e-6
    return a / a.sum()


def init_random_smooth(n, rng):
    """Smooth random initialization via low-pass filtering."""
    a = rng.random(n)
    # Simple moving average smoothing
    kernel_size = max(3, n // 32)
    kernel = np.ones(kernel_size) / kernel_size
    a = np.convolve(a, kernel, mode='same')
    a += 0.01
    return a / a.sum()


def init_fourier_based(n, rng):
    """Design via frequency domain: flat-ish power spectrum."""
    # Want |FFT(a)|^2 to be as flat as possible => a has low autocorrelation peak
    # Start with random phase, take IFFT, make non-negative
    for _ in range(20):
        phases = rng.uniform(0, 2 * np.pi, n)
        magnitudes = np.random.random(n) ** 0.5  # mild bias
        X = magnitudes * np.exp(1j * phases)
        # Make real: X[k] = conj(X[n-k])
        X_real = np.zeros(n, dtype=np.complex128)
        X_real[0] = magnitudes[0]
        for k in range(1, (n // 2) + 1):
            if k < n:
                X_real[k] = magnitudes[k % n] * np.exp(1j * phases[k % n])
            if n - k < n and n - k != k:
                X_real[n - k] = np.conj(X_real[k])
        a = np.real(np.fft.ifft(X_real))
        a = np.maximum(a, 0)
        if a.sum() > 0.1:
            return a / a.sum()
    return np.ones(n) / n


def get_inits(n, rng, prev_a=None):
    """Generate diverse initializations for given n."""
    inits = []
    if prev_a is not None and prev_a.size == n:
        inits.append(prev_a.copy())
        # Perturbed version
        p = prev_a + 0.1 * rng.standard_normal(n)
        p = np.maximum(p, 0)
        inits.append(p / p.sum())
    inits.append(init_u_shape(n))
    inits.append(init_two_cluster(n, rng))
    inits.append(init_three_cluster(n, rng))
    inits.append(init_sided(n))
    inits.append(init_cosine(n, rng))
    inits.append(init_random_smooth(n, rng))
    inits.append(init_fourier_based(n, rng))
    return inits


def coordinate_polish(f, time_budget, rng):
    """Coordinate descent: try multiple values per coordinate."""
    n = f.size
    f = f.copy()
    f = np.maximum(f, 0)
    f = f / f.sum()
    best_f = f.copy()
    best_score = score_normalized(f)

    t0 = time.monotonic()
    sweep = 0
    no_improve_sweeps = 0

    while time.monotonic() - t0 < time_budget:
        start_sweep_score = best_score
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() - t0 > time_budget:
                break
            old_v = f[i]
            s_other = f.sum() - old_v
            if s_other < 1e-12:
                continue

            # Generate candidates around current value
            candidates = set()
            candidates.add(0.0)
            candidates.add(old_v)
            candidates.add(old_v * 0.3)
            candidates.add(old_v * 0.6)
            candidates.add(old_v * 1.4)
            candidates.add(old_v * 2.0)
            candidates.add(s_other * 0.01)
            candidates.add(s_other * 0.03)
            candidates.add(s_other / n)
            # Golden-ratio-based candidates
            candidates.add(old_v * 0.618)
            candidates.add(old_v * 1.618)

            best_v = old_v
            best_local_score = np.inf

            for v in candidates:
                if v < 0 or v > s_other * 2 + 1:
                    continue
                f[i] = v
                total = f.sum()
                if total < 1e-12:
                    f[i] = old_v
                    continue
                M_v = float(np.max(full_conv(f)))
                sc = 2.0 * n * M_v / (total * total)
                if sc < best_local_score:
                    best_local_score = sc
                    best_v = v

            f[i] = best_v
            f_sum = f.sum()
            if f_sum > 1e-12:
                f = f / f_sum

        cur_score = score_normalized(f)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_f = f.copy()
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps > 5:
                break

        sweep += 1

    print(f"    [cd] n={n}, sweeps={sweep}, score={best_score:.6f}")
    return best_f, best_score


def subgrad_polish(f, time_budget, rng):
    """Smoothed subgradient descent polish."""
    n = f.size
    f = f.copy()
    f = np.maximum(f, 0)
    f = f / f.sum()
    best_f = f.copy()
    best_sc = score_normalized(f)

    eta = 0.05
    tau_frac = 0.03
    vel = np.zeros(n)
    beta = 0.5
    last_improve = 0
    iters = 0

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        b = full_conv(f)
        M = float(np.max(b))
        cur_sc = 2.0 * n * M

        tau = tau_frac * M
        if tau < 1e-18:
            tau = 1e-18

        # Softmax subgradient
        shift = b / tau
        shift -= shift.max()
        w = np.exp(shift)
        s = w.sum()
        if s < 1e-300:
            w = np.zeros(b.size)
            w[int(np.argmax(b))] = 1.0
        else:
            w /= s
        g = 2.0 * np.correlate(w, f, mode='valid')

        vel = beta * vel + (1.0 - beta) * g
        f_new = np.maximum(f - eta * vel, 0)
        f_new = f_new / f_new.sum()

        new_sc = score_normalized(f_new)
        if new_sc <= cur_sc or rng.random() < 0.03:
            f = f_new
        else:
            f = np.maximum(f - 0.5 * eta * vel, 0)
            f = f / f.sum()

        if new_sc < best_sc - 1e-10:
            best_sc = new_sc
            best_f = f.copy()
            last_improve = iters

        iters += 1
        if iters - last_improve > 2000:
            f = np.maximum(best_f + 0.02 * rng.standard_normal(n), 0)
            f = f / f.sum()
            vel = np.zeros(n)
            eta = 0.03 + 0.05 * rng.random()
            last_improve = iters

    return best_f, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Primary method: LP-based direction finding with adaptive step size.
    Secondary: coordinate descent and subgradient polishing.
    Tries multiple sequence lengths and diverse initializations.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)

    best_overall = None
    best_overall_sc = np.inf
    all_results = []

    # Try to load previous best
    prev_a = None
    prev_sc = np.inf
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size >= 10 and np.all(np.isfinite(h)) and h.sum() > 0.01:
            prev_a = h / h.sum()
            prev_sc = 2.0 * h.size * float(np.max(full_conv(prev_a)))
            print(f"[init] Previous best: n={h.size}, score={prev_sc:.6f}")
    except Exception:
        pass

    if prev_a is not None and prev_sc < 2.0:
        best_overall = prev_a.copy()
        best_overall_sc = prev_sc
        all_results.append((prev_sc, prev_a.copy()))

    # Sequence lengths to try
    n_list = [64, 128, 256, 512]
    if prev_a is not None:
        np_ = prev_a.size
        if np_ not in n_list and 32 <= np_ <= 2048:
            n_list.append(np_)
        n_list = sorted(set(n_list))

    # Add larger n if time permits
    if time.monotonic() < deadline - 600:
        if 1024 not in n_list:
            n_list.append(1024)
        n_list = sorted(n_list)

    print(f"[plan] n_list={n_list}, deadline in {deadline - time.monotonic():.0f}s")

    # Time allocation: weight inversely proportional to n (smaller n = faster per iter)
    # But give more total time to larger n (better potential score)
    raw_w = [np.sqrt(nv) for nv in n_list]
    total_w = sum(raw_w)
    # Use 80% for main optimization
    main_time = (deadline - time.monotonic()) * 0.75

    for idx_n, nv in enumerate(n_list):
        if time.monotonic() > deadline - 30:
            break

        time_n = main_time * raw_w[idx_n] / total_w
        time_n = min(time_n, deadline - time.monotonic() - 30)
        if time_n < 10:
            continue

        print(f"\n{'='*50}")
        print(f"[n={nv}] budget={time_n:.0f}s")
        print(f"{'='*50}")

        inits = get_inits(nv, rng, prev_a)
        n_inits = len(inits)
        time_per_init = max(3.0, time_n / n_inits)

        for ii, a_init in enumerate(inits):
            if time.monotonic() > deadline - 30:
                break
            tb = min(time_per_init, deadline - time.monotonic() - 30)
            if tb < 3.0:
                break

            label = f"init{ii}" if ii < 2 else f"init{ii}"
            f_best, sc_best = lp_optimize(a_init, tb, rng, verbose=False)
            all_results.append((sc_best, f_best.copy()))

            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = f_best.copy()
                print(f"  [{label}] n={nv} score={sc_best:.6f} *** NEW BEST ***")
            else:
                print(f"  [{label}] n={nv} score={sc_best:.6f}")

        # Sort results for this n and keep track
        print(f"[n={nv}] done. Best so far: {best_overall_sc:.6f}")

    # Sort all results and keep top few for polishing
    all_results.sort(key=lambda x: x[0])
    print(f"\n[summary] Top 5 results:")
    for sc, f in all_results[:5]:
        print(f"  n={f.size}, score={sc:.6f}")

    # Polish phase: use remaining time to polish the best solutions
    if time.monotonic() < deadline - 15:
        remain = deadline - time.monotonic() - 10
        print(f"\n[polish] remaining={remain:.0f}s")

        # Polish top 3 distinct solutions
        top_solutions = []
        seen_ns = set()
        for sc, f in all_results[:10]:
            if f.size not in seen_ns:
                top_solutions.append((sc, f))
                seen_ns.add(f.size)
            if len(top_solutions) >= 3:
                break

        polish_time_each = remain / max(len(top_solutions), 1) * 0.7

        for sc, f in top_solutions:
            if time.monotonic() > deadline - 10:
                break
            tb = min(polish_time_each, deadline - time.monotonic() - 10)
            if tb < 5:
                continue

            print(f"\n  Polishing n={f.size}, current_score={sc:.6f}, time={tb:.0f}s")

            # LP polish (most effective)
            lp_time = tb * 0.5
            if lp_time > 5:
                f_p, sc_p = lp_optimize(f, lp_time, rng, verbose=False)
                if sc_p < best_overall_sc:
                    best_overall_sc = sc_p
                    best_overall = f_p.copy()
                    print(f"    LP polish: {sc_p:.6f} *** NEW BEST ***")
                f = f_p
                sc = sc_p

            # Subgradient polish
            if time.monotonic() < deadline - 10:
                sg_time = min(tb * 0.25, 30.0)
                if sg_time > 3:
                    f_sg, sc_sg = subgrad_polish(f, sg_time, rng)
                    if sc_sg < best_overall_sc:
                        best_overall_sc = sc_sg
                        best_overall = f_sg.copy()
                        print(f"    SG polish: {sc_sg:.6f} *** NEW BEST ***")
                    f = f_sg
                    sc = sc_sg

            # Coordinate descent polish
            if time.monotonic() < deadline - 10:
                cd_time = min(tb * 0.2, 30.0)
                if cd_time > 3:
                    f_cd, sc_cd = coordinate_polish(f, cd_time, rng)
                    if sc_cd < best_overall_sc:
                        best_overall_sc = sc_cd
                        best_overall = f_cd.copy()
                        print(f"    CD polish: {sc_cd:.6f} *** NEW BEST ***")

    # Final: one more LP polish on the absolute best if time remains
    if best_overall is not None and time.monotonic() < deadline - 10:
        remain = deadline - time.monotonic() - 5
        if remain > 20:
            print(f"\n[final polish] time={remain:.0f}s")
            f_final, sc_final = lp_optimize(best_overall, remain, rng, verbose=False)
            if sc_final < best_overall_sc:
                best_overall_sc = sc_final
                best_overall = f_final.copy()
                print(f"  Final polish: {sc_final:.6f} *** NEW BEST ***")

    # Fallback
    if best_overall is None:
        best_overall = np.ones(128) / 128
        best_overall_sc = 2.0

    # Format result
    result = [float(max(0.0, x)) for x in best_overall]
    final_sc = evaluate_sequence(result)
    print(f"\n{'='*50}")
    print(f"[FINAL] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")
    print(f"{'='*50}")

    return result
