Multi-phase optimizer for minimum peak self-convolution of non-negative sequences.
Combines subgradient descent with momentum, spectral shaping, and coordinate descent.

from scipy.signal import fftconvolve
import time

    """Compute full linear convolution of f with itself."""
    return fftconvolve(f, f)

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1} using the standard算法."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _eval(f):
    """Evaluate the objective function."""
    return evaluate_sequence(f.tolist())

def _subgradient(f, k_star, n):
    """Compute subgradient at lag k_star."""
    # grad[m] = 2 * f[k_star - m] for valid m
    grad = np.zeros(n)
    # f[k_star - m] is valid when 0 <= k_star - m < n, i.e., k_star - n + 1 <= m <= k_star
    m_min = max(0, k_star - n + 1)
    m_max = min(n - 1, k_star)
    if m_min <= m_max:
        grad[m_min:m_max+1] = 2.0 * f[k_star:m_min-1:-1]  # f[k_star], f[k_star-1], ...
    return grad

def _subgradient_topK(f, top_k_lags, n):
    """Compute averaged subgradient over top K lags."""
    grad = np.zeros(n)
    for k in top_k_lags:
        m_min = max(0, k - n + 1)
        m_max = min(n - 1, k)
        if m_min <= m_max:
            grad[m_min:m_max+1] += 2.0 * f[k:m_min-1:-1]
    grad /= len(top_k_lags)
    return grad

    N = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, N)
    power = np.fft.rfft(f, N)
    mag = np.abs(power)
    phase = np.angle(power)

    # New magnitude: mag^alpha (alpha < 1 compresses)
    max_mag = mag.max()
    if max_mag < 1e-30:
        return f.copy()
    mag_norm = mag / max_mag
    mag_new = max_mag * (mag_norm ** alpha)

    f_new = np.fft.irfft(F_new, N)[:n]
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-15:
        return f.copy()
    return f_new / s
