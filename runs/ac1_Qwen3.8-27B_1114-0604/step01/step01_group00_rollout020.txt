Uses FFT-based mirror descent with soft-peak gradient, multiple shape-based

import numpy as np
from scipy.signal import correlate
import time


def _fft_convolve(p):
    """Full linear convolution via FFT."""
    n = len(p)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(p, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _soft_peak_gradient(c, p, T, top_k=30):
    """
    Compute soft-max weighted gradient for the peak of the convolution.
    Returns gradient vector g where g[i] = sum_k w_k * p[k-i].
    """
    n = len(p)
    m = len(c)  # 2n-1
    # Find top_k lags
    if m > top_k:
        # Use partial sort for efficiency
        idx = np.argpartition(c, -top_k)[-top_k:]
        top_vals = c[idx]
    else:
        idx = np.arange(m)
        top_vals = c.copy()
    
    # Softmax weights over top lags
    cs = top_vals / max(T, 1e-15)
    cs -= cs.max()
    w = np.exp(cs)
    w /= w.sum()
    
    # Compute gradient: g[i] = sum_k w_k * p[k-i]
    # For each top lag k, the contribution to p[i] is w_k * p[k-i]
    # We can compute this as a weighted sum of shifted p vectors
    g = np.zeros(n)
    for kk, ww in zip(idx, w):
        k_int = int(kk)
        # p[k-i] is valid when 0 <= k-i < n, i.e., max(0, k-n+1) <= i <= min(k, n-1)
        i_start = max(0, k_int - n + 1)
        i_end = min(k_int, n - 1)
        if i_start <= i_end:
            # p[k_i - i] for i in [i_start, i_end]
            vals = p[k_int - i_end : k_int - i_start + 1][::-1]
            g[i_start:i_end+1] += ww * vals
    
    return g


def _hard_peak_gradient(c, p):
    """Gradient using only the single peak lag."""
    n = len(p)
    k_star = int(np.argmax(c))
    g = np.zeros(n)
    i_start = max(0, k_star - n + 1)
    i_end = min(k_star, n - 1)
    if i_start <= i_end:
        vals = p[k_star - i_end : k_star - i_start + 1][::-1]
        g[i_start:i_end+1] = vals
    return g


def _mirror_step(log_p, grad, eta):
    """Exponentiated gradient step in log-space."""
    log_p = log_p - eta * grad
    log_p -= log_p.max()
    p = np.exp(log_p)
    s = p.sum()
    if s < 1e-300:
        return log_p, np.ones(len(p)) / len(p)
    p /= s
    log_p = np.log(p + 1e-300)
    return log_p, p

def _coordinate_polish(p, n, deadline, rng):
    """
    Coordinate-descent polish: iteratively move mass from high-contributing
    positions to low-contributing positions.
    """
    p = p.copy().astype(np.float64)
    p = np.maximum(p, 0)
    p /= p.sum()
    
    local_best = _eval(p)
    local_best_p = p.copy()
    c = _fft_convolve(p)
    peak = c.max()
    
    n_passes = 0
    max_passes = 50000
    consecutive_no_improve = 0
    
    while n_passes < max_passes:
        if time.monotonic() > deadline:
            break
        n_passes += 1
        
        # Recompute convolution every few passes
        if n_passes % 5 == 0:
            c = _fft_convolve(p)
            peak = c.max()
        
        # Find top peak lags
        top_k = 5
        if len(c) > top_k:
            idx = np.argpartition(c, -top_k)[-top_k:]
        else:
            idx = np.arange(len(c))
        
        # Compute combined contribution for each position
        contrib = np.zeros(n)
        for k in idx:
            k_int = int(k)
            i_start = max(0, k_int - n + 1)
            i_end = min(k_int, n - 1)
            if i_start <= i_end:
                vals = p[k_int - i_end : k_int - i_start + 1][::-1]
                contrib[i_start:i_end+1] += vals
        
        # Find best donate/receive pair
        # Donate from positions with high contribution
        # Receive at positions with low contribution
        top_donors = np.argsort(contrib)[-10:]
        top_receivers = np.argsort(contrib)[:10]
        
        improved = False
        for d in top_donors:
            for r in top_receivers:
                if d == r:
                    continue
                # Try transferring a small fraction
                for frac in [0.02, 0.05, 0.1]:
                    amt = p[d] * frac
                    if amt < 1e-15:
                        continue
                    p_new = p.copy()
                    p_new[d] -= amt
                    p_new[r] += amt
                    if p_new[d] < 0:
                        continue
                    p_new /= p_new.sum()
                    c_new = _fft_convolve(p_new)
                    new_peak = c_new.max()
                    if new_peak < peak - 1e-14:
                        p = p_new
                        peak = new_peak
                        c = c_new
                        local_best = _eval(p)
                        local_best_p = p.copy()
                        improved = True
                        consecutive_no_improve = 0
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            consecutive_no_improve += 1
            if consecutive_no_improve >= 20:
                # Random perturbation to escape
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i != j:
                    amt = p[i] * 0.1
                    p[i] -= amt
                    p[j] += amt
                    p = np.maximum(p, 0)
                    p /= p.sum()
                else:
                    break
                consecutive_no_improve = 0
            elif consecutive_no_improve >= 100:
                break
    
    return local_best_p, local_best


def _asymmetric_hann(n, shift=0.0, width=1.0, skew=0.0):
    """Generate an asymmetric Hann-like window."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0 + shift
    x = (j - center) / (width * n / 2.0)
    # Base Hann
    base = np.sin(np.pi * np.clip((j + 0.5 - shift) / n, 0, 1)) ** 2
    # Add skew
    if skew != 0.0:
        base = base * (1.0 + skew * (2.0 * j / (n - 1) - 1.0))
    base = np.maximum(base, 0)
    s = base.sum()
    if s < 1e-15:
        base = np.ones(n) / n
    else:
        base /= s
    return base

    starts['random_exponential'] = r
    
    # Evaluate all starts
    start_results = []
    for name, f0 in starts.items():
        if f0 is None:
            continue
        f0 = np.maximum(f0, 0)
        s = f0.sum()
        if s < 1e-10:
            f0 = np.ones(n) / n
        else:
            f0 = f0 / s
        sc = _eval(f0)
        start_results.append((sc, name, f0.copy()))
    
    start_results.sort(key=lambda x: x[0])
    print("Starting point scores:")
    for sc, name, _ in start_results:
        print(f"  {name}: {sc:.6f}")
    
    best_score = start_results[0][0]
    best_p = start_results[0][2].copy()
    
    # Phase 1: Optimize from top starts with mirror descent
    # Allocate time proportionally to quality
    n_starts_to_optimize = min(6, len(start_results))
    remaining_time = deadline - time.monotonic()
    time_per_start = remaining_time * 0.55 / n_starts_to_optimize
    
    print(f"\nPhase 1: Mirror descent optimization ({n_starts_to_optimize} starts)")
    
    for idx in range(n_starts_to_optimize):
        if time.monotonic() > deadline - 60:
            break
        
        sc, name, f0 = start_results[idx]
        t_alloc = time_per_start * (1.0 - 0.1 * idx)
        start_deadline = time.monotonic() + t_alloc
        
        print(f"\n  Optimizing '{name}' (initial={sc:.6f}, budget={t_alloc:.0f}s)")
        
        p_opt, sc_opt = _optimize_mirror(f0, n, start_deadline, rng)
        
        if sc_opt < best_score:
            best_score = sc_opt
            best_p = p_opt.copy()
            print(f"    New best: {best_score:.6f}")
        else:
            print(f"    Best from this start: {sc_opt:.6f} (not better than {best_score:.6f})")
    
    # Phase 2: Coordinate descent polish
    print(f"\nPhase 2: Coordinate descent polish (current best={best_score:.6f})")
    
    polish_time = min(300.0, max(60.0, (deadline - time.monotonic()) * 0.4))
    polish_deadline = time.monotonic() + polish_time
    
    p_polished, sc_polished = _coordinate_polish(best_p, n, polish_deadline, rng)
    if sc_polished < best_score:
        best_score = sc_polished
        best_p = p_polished.copy()
        print(f"  Polished to: {best_score:.6f}")
    else:
        print(f"  Polish result: {sc_polished:.6f} (no improvement)")
    
    # Phase 3: Final optimization round from polished solution
    remaining = deadline - time.monotonic()
    if remaining > 60:
        print(f"\nPhase 3: Final optimization round (budget={remaining:.0f}s)")
        final_deadline = time.monotonic() + min(remaining - 10, 300.0)
        p_final, sc_final = _optimize_mirror(best_p, n, final_deadline, rng)
        if sc_final < best_score:
            best_score = sc_final
            best_p = p_final.copy()
            print(f"  Final optimization: {best_score:.6f}")
        else:
            print(f"  Final optimization: {sc_final:.6f} (no improvement)")
        
        # Another polish
        if time.monotonic() < deadline - 30:
            p_p2, sc_p2 = _coordinate_polish(best_p, n, min(time.monotonic() + 60, deadline), rng)
            if sc_p2 < best_score:
                best_score = sc_p2
                best_p = p_p2.copy()
                print(f"  Final polish: {best_score:.6f}")
    
    # Final validation
    best_p = np.clip(best_p, 0.0, 1000.0)
    final_score = _eval(best_p)
    if final_score < best_score or not np.isfinite(best_score):
        best_score = final_score
    
    print(f"\n=== FINAL SCORE: {best_score:.6f} ===")
    return [float(x) for x in best_p]
