Uses the active-set LP direction plus fast mirror descent refinement.

def _fft_convolve(a):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = 1
    while N < need:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _score(a):
    """Compute the evaluation score (normalized to sum=1)."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    mx = float(c.max())
    return float(2.0 * n * mx / (s * s))

    """Create normalized sinusoidal candidate."""

    """
    Solve LP: max sum(g) s.t. g>=0, (f*g)_k <= M for active lags k.
    Returns normalized g (sum=1) or None.
    """
    n = len(f)
    c_ff = _fft_convolve(f)
    
    # Select active lags (those near the peak)
    threshold = 0.85 * M
    active = np.where(c_ff >= threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c_ff)])
    if len(active) > max_constraints:
        # Keep the top max_constraints
        top_idx = np.argsort(c_ff[active])[-max_constraints:]
        active = active[top_idx]
    
    active = np.asarray(active, dtype=np.int64)
    n_c = len(active)
    
    # Build sparse constraint matrix
    rows = []
    cols = []
    vals = []
    for row_idx, k in enumerate(active):
        k = int(k)
        i_min = max(0, k - n + 1)
        i_max = min(k, n - 1)
        # (f*g)_k = sum_{i=i_min}^{i_max} f[k-i] * g[i]
        fi_vals = f[k - i_max : k - i_min + 1][::-1]
        nonzero = fi_vals > 1e-15
        if np.any(nonzero):
            nnz_idx = np.where(nonzero)[0]
            rows.extend([row_idx] * len(nnz_idx))
            cols.extend((i_min + nnz_idx).tolist())
            vals.extend(fi_vals[nonzero].tolist())
    
    if len(rows) == 0:
        return np.ones(n) / n
    
    A_ub = csr_matrix((vals, (rows, cols)), shape=(n_c, n))
    b_ub = np.full(n_c, M)
    c_obj = -np.ones(n)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 30})
    except Exception:
        return None
    
    if res.status != 0:
        return None
    
    g = np.asarray(res.x, dtype=np.float64)
    g = np.clip(g, 0, 1000)
    gs = g.sum()
    if gs < 0.01:
        return None
    g = g / gs
    return g
