"""
Low-peak nonnegative step-function search using multi-harmonic initialization
and improved active-set gradient refinement with momentum and restarts.

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))


def _make_multi_harmonic(n, amplitudes, phases):
    """Create a multi-harmonic anti-palindromic sequence."""
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for m, (A, ph) in enumerate(zip(amplitudes, phases), start=1):
        if abs(A) > 1e-12:
            a += A * np.sin(2.0 * np.pi * m * (i + 0.5) / n + ph)
    a = a / n
    a = np.where(np.isfinite(a), a, 0.0)

    if not np.isfinite(s) or s < 0.01:

def _refine_with_momentum(a, budget_s=60.0, K=16, momentum=0.9):
    """
    Active-set gradient refinement with momentum.
    """
    start = time.monotonic()
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return arr, np.inf
    arr = arr / s
    
    best = arr.copy()
    best_score = _internal_score(best)
    if not np.isfinite(best_score):
        return best, best_score
    
    n = arr.size
    lr = 0.1 * float(n)
    max_lr = 0.5 * float(n)
    min_lr = 0.5
    max_iters = 10000
    it = 0
    last_improve = 0
    vel = np.zeros(n, dtype=np.float64)  # momentum
    
    while time.monotonic() - start < budget_s and it < max_iters:
        it += 1
        
        c = _fft_convolve(arr)
        if c.size == 0:
            break
        maxc = float(np.max(c))
        if not np.isfinite(maxc):
            break
        
        # Get top-K active lags
        if c.size <= K:
            active = np.arange(c.size, dtype=np.int64)
        else:
            active = np.argpartition(c, -K)[-K:]
        
        # Compute gradient
        grad = np.zeros(n, dtype=np.float64)
        cnt = 0
        for kk in active:
            k = int(kk)
            cnt += 1
            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            if i_max >= i_min:
                idx = np.arange(i_min, i_max + 1, dtype=np.int64)
                grad[idx] += 2.0 * arr[k - idx]
        
        if cnt <= 0:
            break
        grad /= float(cnt)
        
        # Project to tangent space (zero mean for simplex constraint)
        g = grad - np.mean(grad)
        
        # Momentum update
        vel = momentum * vel + g
        exponent = -lr * vel
        exponent = np.clip(exponent, -2.0, 2.0)
        arr_new = arr * np.exp(exponent)
        arr_new = np.where(np.isfinite(arr_new), arr_new, 0.0)
        arr_new = np.clip(arr_new, 0.0, 1000.0)
        ts = float(arr_new.sum())
        if not np.isfinite(ts) or ts < 0.01:
            arr_new = best.copy()
            vel *= 0.5
        else:
            arr_new = arr_new / ts
            arr = arr_new
        
        # Periodic evaluation
        if it % 5 == 0:
            sc = _internal_score(arr)
            if np.isfinite(sc) and sc < best_score:
                best = arr.copy()
                best_score = sc
                last_improve = it
                lr = min(max_lr, lr * 1.03)
            else:
                if sc > best_score * 1.05:
                    arr = best.copy()
                    vel *= 0.3
                    lr = max(min_lr, lr * 0.8)
                else:
                    lr = min(max_lr, lr * 1.01)
                if it - last_improve > 500:
                    # Stuck - break
                    break
    
    return best, best_score


def _refine_coordinate(a, budget_s=30.0):

    Coordinate-style refinement: at each step, find the pair of indices
    that most reduces the score by redistributing mass.
    """
    start = time.monotonic()
    arr = np.asarray(a, dtype=np.float64).copy()

    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return arr, np.inf
    arr = arr / s
    
    best = arr.copy()
    best_score = _internal_score(best)
    if not np.isfinite(best_score):
        return best, best_score
    
    n = arr.size
    
    # Precompute the convolution
    c = _fft_convolve(arr)
    maxc = float(np.max(c))
    
    # Find the lag that achieves the max
    k_max = int(np.argmax(c))
    
    # The gradient w.r.t. a[i] at the critical lag is:
    # d/dc_k (a*a) = a[k-i] for each i where both i and k-i are valid
    # To reduce c_k, we want to decrease a[i]*a[k-i] for the pairs that contribute most
    
    # Strategy: for the critical lag k_max, find pairs (i, k_max-i) with large product
    # and transfer mass from one to the other (or to a position that doesn't affect k_max)
    
    it = 0
    max_iters = 5000
    while time.monotonic() - start < budget_s and it < max_iters:
        it += 1
        
        c = _fft_convolve(arr)
        maxc = float(np.max(c))
        if not np.isfinite(maxc):
            break
        
        k_max = int(np.argmax(c))
        
        # Find the "hot" positions for this lag
        i_min = max(0, k_max - (n - 1))
        i_max = min(k_max, n - 1)
        if i_max < i_min:
            break

def _refine_activated(a, budget_s=60.0):
    """
    More sophisticated refinement: use multiple active lags with weighting.
    """
    start = time.monotonic()
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return arr, np.inf
    arr = arr / s
    
    best = arr.copy()
    best_score = _internal_score(best)
    if not np.isfinite(best_score):
        return best, best_score
    
    n = arr.size
    lr = 0.5 * float(n)
    max_lr = 2.0 * float(n)
    min_lr = 1.0
    max_iters = 20000
    it = 0
    last_improve = 0
    K = 32 if n <= 4096 else 64
    
    while time.monotonic() - start < budget_s and it < max_iters:
        it += 1
        
        c = _fft_convolve(arr)
        if c.size == 0:
            break
        maxc = float(np.max(c))
        if not np.isfinite(maxc):
            break
        
        # Weighted active set: weight by (c_k - mean_c)^+ 
        c_mean = float(np.mean(c))
        weights = np.clip(c - c_mean, 0, None)
        
        # Get top K by weight
        if c.size <= K:
            active = np.arange(c.size, dtype=np.int64)
            w_active = weights
        else:
            top_idx = np.argpartition(weights, -K)[-K:]
            active = top_idx
            w_active = weights[top_idx]
        
        # Normalize weights
        w_sum = float(np.sum(w_active))
        if w_sum < 1e-10:
            w_active = np.ones_like(w_active) / K
        else:
            w_active = w_active / w_sum
        
        # Compute weighted gradient
        grad = np.zeros(n, dtype=np.float64)
        for kk, ww in zip(active, w_active):
            k = int(kk)
            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            if i_max >= i_min:
                idx = np.arange(i_min, i_max + 1, dtype=np.int64)
                grad[idx] += float(ww) * 2.0 * arr[k - idx]
        
        # Project to tangent space
        g = grad - np.mean(grad)
        
        # Adaptive step
        g_norm = float(np.sqrt(np.sum(g**2)))
        if g_norm < 1e-15:
            break
        
        exponent = -lr * g / g_norm
        exponent = np.clip(exponent, -3.0, 3.0)
        arr_new = arr * np.exp(exponent)
        arr_new = np.where(np.isfinite(arr_new), arr_new, 0.0)
        arr_new = np.clip(arr_new, 0.0, 1000.0)
        ts = float(arr_new.sum())
        if not np.isfinite(ts) or ts < 0.01:
            arr_new = best.copy()
            lr = max(min_lr, lr * 0.5)
        else:
            arr_new = arr_new / ts
            arr = arr_new

        if it % 3 == 0:
            sc = _internal_score(arr)
            if np.isfinite(sc) and sc < best_score:
                best = arr.copy()
                best_score = sc
                last_improve = it
                lr = min(max_lr, lr * 1.05)
            else:

                    arr = best.copy()
                    lr = max(min_lr, lr * 0.7)
                if it - last_improve > 1000:
                    break

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

        configs.append((n, [1, 3, 5], [0.8, 0.1, 0.05], [0.0, 0.0, 0.0]))
        configs.append((n, [1, 3, 5], [0.7, 0.2, 0.05], [0.0, 0.0, 0.0]))

    for n, freqs, amps, phases in configs:
        if time.monotonic() > deadline:
            break
        a = _make_harmonic(n, freqs, amps, phases)
        sc = _internal_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = a.copy()

    if best_a is None:
        best_a = _make_harmonic(4096, [1], [0.78], [0.0])
        best_score = _internal_score(best_a)

    return best_a, best_score


def _compute_weighted_gradient(arr, c, K):
    """Compute weighted gradient from top-K active lags."""
    n = arr.size
    c_shifted = c - np.mean(c)
    weights = np.clip(c_shifted, 0.0, None)

    if c.size <= K:
        active = np.arange(c.size, dtype=np.int64)
        w_active = weights.copy()
    else:
        top_idx = np.argpartition(weights, -K)[-K:]
        active = top_idx
        w_active = weights[top_idx]

    w_sum = float(np.sum(w_active))
    if w_sum < 1e-15:
        w_active = np.ones(K if c.size > K else c.size, dtype=np.float64) / max(K, c.size)
    else:
        w_active = w_active / w_sum

    grad = np.zeros(n, dtype=np.float64)
    for kk, ww in zip(active, w_active):
        k = int(kk)
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        if i_max >= i_min:
            idx = np.arange(i_min, i_max + 1, dtype=np.int64)
            grad[idx] += float(ww) * 2.0 * arr[k - idx]

    return grad


def _refine_primary(a, budget_s=120.0):
    """
    Main refinement loop: weighted active-set mirror descent with adaptive lr.
    """
    start = time.monotonic()
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return arr, np.inf
    arr = arr / s

    best = arr.copy()
    best_score = _internal_score(best)
    if not np.isfinite(best_score):
        return best, best_score

    n = arr.size
    K = max(16, min(64, n // 32))
    lr = 2.0 * float(n)
    max_lr = 10.0 * float(n)
    min_lr = 1.0
    max_iters = 200000
    it = 0
    last_improve = 0
    eval_interval = 3

    while time.monotonic() - start < budget_s and it < max_iters:
        it += 1

        c = _fft_convolve(arr)
        if c.size == 0:
            break
        maxc = float(np.max(c))
        if not np.isfinite(maxc):
            break

        grad = _compute_weighted_gradient(arr, c, K)
        g = grad - np.mean(grad)

        g_max = float(np.max(np.abs(g)))
        if g_max < 1e-15:
            break

        # Scale step so that lr * g_max is in a good range
        exponent = -lr * g
        exponent = np.clip(exponent, -4.0, 4.0)
        arr_new = arr * np.exp(exponent)
        arr_new = np.where(np.isfinite(arr_new), arr_new, 0.0)
        arr_new = np.clip(arr_new, 0.0, 1000.0)
        ts = float(arr_new.sum())
        if not np.isfinite(ts) or ts < 0.01:
            arr_new = best.copy()
            lr = max(min_lr, lr * 0.5)
        else:
            arr_new = arr_new / ts
            arr = arr_new

        if it % eval_interval == 0:
            sc = _internal_score(arr)
            if np.isfinite(sc) and sc < best_score:
                best = arr.copy()
                best_score = sc
                last_improve = it
                lr = min(max_lr, lr * 1.02)
            else:
                if it - last_improve > 50:
                    arr = best.copy()
                    lr = max(min_lr, lr * 0.85)
                if it - last_improve > 500:
                    break

    return best, best_score


def _refine_momentum(a, budget_s=60.0):
    """Refinement with momentum term."""
    start = time.monotonic()
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return arr, np.inf
    arr = arr / s

    best = arr.copy()
    best_score = _internal_score(best)
    if not np.isfinite(best_score):
        return best, best_score

    n = arr.size
    K = 32
    lr = 1.0 * float(n)
    max_lr = 5.0 * float(n)
    min_lr = 0.5
    momentum = 0.8
    vel = np.zeros(n, dtype=np.float64)
    max_iters = 100000
    it = 0
    last_improve = 0

    while time.monotonic() - start < budget_s and it < max_iters:
        it += 1

        c = _fft_convolve(arr)
        if c.size == 0:
            break
        maxc = float(np.max(c))
        if not np.isfinite(maxc):
            break

        grad = _compute_weighted_gradient(arr, c, K)
        g = grad - np.mean(grad)

        vel = momentum * vel + g
        exponent = -lr * vel
        exponent = np.clip(exponent, -4.0, 4.0)
        arr_new = arr * np.exp(exponent)
        arr_new = np.where(np.isfinite(arr_new), arr_new, 0.0)
        arr_new = np.clip(arr_new, 0.0, 1000.0)
        ts = float(arr_new.sum())
        if not np.isfinite(ts) or ts < 0.01:
            arr_new = best.copy()
            vel *= 0.5
            lr = max(min_lr, lr * 0.7)
        else:
            arr_new = arr_new / ts
            arr = arr_new

        if it % 5 == 0:
            sc = _internal_score(arr)
            if np.isfinite(sc) and sc < best_score:
                best = arr.copy()
                best_score = sc
                last_improve = it
                lr = min(max_lr, lr * 1.03)
            else:
                if it - last_improve > 80:
                    arr = best.copy()
                    vel *= 0.5
                    lr = max(min_lr, lr * 0.8)
                if it - last_improve > 600:
                    break

    return best, best_score


def _try_perturbation(best, budget_s, nConfigs=4):
    """Try perturbing the best solution and re-refining."""
    start = time.monotonic()
    n = best.size
    results = []

    for trial in range(nConfigs):
        remaining = budget_s - (time.monotonic() - start)
        if remaining < 15:
            break

        # Different perturbation strategies
        if trial == 0:
            # Small Gaussian
            pert = 0.02 * np.random.randn(n) * np.mean(best)
        elif trial == 1:
            # Larger Gaussian
            pert = 0.05 * np.random.randn(n) * np.mean(best)
        elif trial == 2:
            # Anti-palindromic perturbation
            base_pert = np.random.randn(n) * 0.03 * np.mean(best)
            pert = base_pert - base_pert[::-1]  # anti-palindromic
        else:
            # Frequency domain perturbation
            fa = np.fft.rfft(best)
            freq_pert = np.random.randn(len(fa)) * 0.02
            freq_pert[0] = 0
            pert = np.fft.irfft(fa + freq_pert, n) - best

        perturbed = best + pert
        perturbed = np.clip(perturbed, 0.0, None)
        ps = float(perturbed.sum())
        if ps < 0.01:
            continue
        perturbed = perturbed / ps

        sub_budget = min(25.0, remaining * 0.6)
        result, sc = _refine_primary(perturbed, sub_budget)
        if np.isfinite(sc):
            results.append((sc, result))

    if not results:
        return best, _internal_score(best)

    results.sort(key=lambda x: x[0])
    best_sc, best_arr = results[0]
    cur_sc = _internal_score(best)
    if best_sc < cur_sc:
        return best_arr, best_sc
    return best, cur_sc


def _safe_resolve_sequence(x):
    if x is None:
        return None
    arr = np.asarray(x, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return None
    arr = arr / s
    return arr


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a low evaluation score using multi-harmonic initialization
    and multi-phase active-set gradient refinement.
    """
    t0 = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    hard_deadline = t0 + total_budget - 30.0

    np.random.seed(seed)

    # Phase 0: Get starting point
    start_a = None
    start_score = np.inf
    try:
        hs = height_sequence_1
        if hs is not None and len(hs) > 0:
            start_a = np.array(hs, dtype=np.float64)
            start_score = _internal_score(start_a)
            print(f"[init] height_sequence_1: n={len(hs)}, score={start_score:.8f}")
    except Exception:
        pass

    # Phase 1: Scan for good initializations
    scan_deadline = t0 + min(200.0, total_budget * 0.1)
    scan_a, scan_score = _scan_init(scan_deadline)
    print(f"[scan] best initial: n={scan_a.size}, score={scan_score:.8f}")

    # Pick the best starting point
    if start_a is not None and start_score < scan_score:
        current = start_a.copy()
        current_score = start_score
        print(f"[init] Using previous best as start: {current_score:.8f}")
    else:
        current = scan_a.copy()
        current_score = scan_score
        print(f"[init] Using scan best as start: {current_score:.8f}")

    best_overall = current.copy()
    best_overall_score = current_score

    # Phase 2: Primary refinement
    refine1_deadline = t0 + min(total_budget * 0.4, 600.0)
    remain1 = refine1_deadline - time.monotonic()
    if remain1 > 20:
        print(f"\n[refine1] Primary refinement, budget={remain1:.0f}s")
        result, sc = _refine_primary(current, remain1)
        if np.isfinite(sc) and sc < best_overall_score:
            best_overall = result
            best_overall_score = sc
            print(f"[refine1] score={sc:.8f}")

    # Phase 3: Momentum refinement
    refine2_deadline = t0 + min(total_budget * 0.65, 1200.0)
    remain2 = refine2_deadline - time.monotonic()
    if remain2 > 20:
        print(f"\n[refine2] Momentum refinement, budget={remain2:.0f}s")
        result, sc = _refine_momentum(best_overall, remain2)
        if np.isfinite(sc) and sc < best_overall_score:
            best_overall = result
            best_overall_score = sc
            print(f"[refine2] score={sc:.8f}")

    # Phase 4: Perturbation + re-refine
    refine3_deadline = t0 + min(total_budget * 0.85, 2000.0)
    remain3 = refine3_deadline - time.monotonic()
    if remain3 > 30:
        print(f"\n[refine3] Perturbation restarts, budget={remain3:.0f}s")
        result, sc = _try_perturbation(best_overall, remain3, nConfigs=4)
        if np.isfinite(sc) and sc < best_overall_score:
            best_overall = result
            best_overall_score = sc
            print(f"[refine3] score={sc:.8f}")

    # Phase 5: Final polish
    remain5 = hard_deadline - time.monotonic()
    if remain5 > 15:
        print(f"\n[final] Final polish, budget={remain5:.0f}s")
        result, sc = _refine_primary(best_overall, min(remain5, 300.0))
        if np.isfinite(sc) and sc < best_overall_score:
            best_overall = result
            best_overall_score = sc
            print(f"[final] score={sc:.8f}")

    # Output
    arr_out = _safe_resolve_sequence(best_overall)
    if arr_out is None:
        arr_out = np.ones(2048, dtype=np.float64) / 2048.0

    print(f"\n[done] Best internal score: {best_overall_score:.8f}, n={arr_out.size}")
    print(f"[done] Total time: {time.monotonic() - t0:.2f}s")

    # Validate with actual evaluator
    try:
        real_score = float(evaluate_sequence(arr_out.tolist()))
        print(f"[validate] Actual score: {real_score:.8f}")
        if not np.isfinite(real_score) or real_score > 2.0:
            arr_out = np.ones(2048, dtype=np.float64) / 2048.0
            print("[validate] Fallback to constant")
    except Exception as e:
        print(f"[validate] Error: {e}, using fallback")
        arr_out = np.ones(2048, dtype=np.float64) / 2048.0

    return [float(x) for x in arr_out]
