```python
"""
Minimizer for 2*n*max(convolve(a,a))/(sum a)^2 using subgradient descent
with simulated annealing, soft-max peak weighting, and local search polish.
"""

import numpy as np
from scipy.signal import correlate
import time


def _fast_score(f):
    """Compute score without security checks (fast path)."""
    c = np.convolve(f, f)
    max_b = c.max()
    sum_a = f.sum()
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * len(f) * max_b / (sum_a * sum_a)


def _simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    # Use the standard water-filling algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    proj = np.maximum(v - theta, 0.0)
    # Ensure sum is exactly 1
    s = proj.sum()
    if s > 1e-15:
        proj /= s
    return proj


def _soft_grad(f, c, T):
    """Compute softmax-weighted subgradient of max(conv)."""
    n = len(f)
    cs = c / max(T, 1e-30)
    cs -= cs.max()
    w = np.exp(cs)
    w /= w.sum()
    grad = 2.0 * correlate(w, f, mode='valid')
    return grad


def _peaked_grad(f, c):
    """Compute subgradient at the exact peak (or near-peak lags)."""
    n = len(f)
    peak = c.max()
    # Find all lags within 1% of peak
    active = np.where(c >= peak * 0.99)[0]
    if len(active) == 0:
        active = np.array([c.argmax()])
    # Average subgradient over active lags
    grad = np.zeros(n)
    for k in active:
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        grad[j_start:j_end+1] += 2.0 * f[k - j_start:k - j_end][::-1]
    grad /= len(active)
    return grad


def _local_search(f, n, deadline, rng, max_iters=5000):
    """Local search: try to reduce the peak by moving mass."""
    best_f = f.copy()
    best_score = _fast_score(f)
    c = np.convolve(f, f)
    peak = c.max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Find peak lag
        k_star = int(np.argmax(c))
        
        # Compute contribution of each pair to the peak
        # c[k_star] = sum_i f[i] * f[k_star - i]
        # We want to reduce this by moving mass from high-contribution positions
        
        # Find the "donor" position: the j that maximizes f[j] * f[k_star - j]
        j_min = max(0, k_star - (n - 1))
        j_max = min(n - 1, k_star)
        
        if j_max < j_min:
            break
        
        contributions = f[j_min:j_max+1] * f[k_star - j_min:k_star - j_max][::-1]
        j_don_rel = int(np.argmax(contributions))
        j_don = j_min + j_don_rel
        
        # Find a "receiver" position with low contribution
        # We want to add mass to a position that doesn't increase the peak
        # Try positions far from the peak's active region
        all_contribs = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                all_contribs[j] = f[j] * f[idx]
        
        # Receiver: position with lowest contribution, giving it mass won't increase peak much
        # But we also want it to not increase OTHER peaks
        # Simple heuristic: pick the position with the smallest f value
        sorted_idx = np.argsort(f)
        
        improved = False
        for j_rec in sorted_idx[:5]:  # Try 5 lowest-value positions
            if j_rec == j_don:
                continue
            
            for frac in [0.05, 0.1, 0.2, 0.5]:
                amt = f[j_don] * frac
                f_try = f.copy()
                f_try[j_don] -= amt
                f_try[j_rec] += amt
                if f_try[j_don] < 0:
                    continue
                f_try = np.maximum(f_try, 0)
                c_try = np.convolve(f_try, f_try)
                if c_try.max() < peak - 1e-12:
                    f = f_try
                    c = c_try
                    peak = c.max()
                    score = _fast_score(f)
                    if score < best_score:
                        best_score = score
                        best_f = f.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            # Try random moves
            for _ in range(5):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                for frac in [0.05, 0.1, 0.3]:
                    amt = f[i] * frac
                    if amt <= 0:
                        continue
                    f_try = f.copy()
                    f_try[i] -= amt
                    f_try[j] += amt
                    if f_try[i] < 0:
                        break
                    c_try = np.convolve(f_try, f_try)
                    if c_try.max() < peak - 1e-12:
```

Hmm, I realize the step size handling is a bit off. Let me think more carefully.

The subgradient method for minimizing f(x) = g(x) (where g is the max of convex-ish functions) uses:
x_{k+1} = project(x_k - eta_k * ∇g(x_k))

The step size eta_k should decrease over time. A common choice is eta_k = c / sqrt(k) for convergence, but in practice, a geometric decay works well.

The key issue is that the gradient magnitude varies. The "natural" step size depends on the scale of the gradient. Since we're working on the simplex (sum f = 1), the components of f are O(1/n), and the gradient components are O(1/n) as well (since grad[j] = 2*f[k-j] and f values are O(1/n)).

So the gradient norm is roughly O(sqrt(n) * 1/n) = O(1/sqrt(n)). A step size of O(sqrt(n)) would give an O(1) movement, which is too large. A step size of O(1) gives an O(1/sqrt(n)) movement, which is reasonable.

Let me just use a fixed step size that I tune empirically, with occasional restarts.

Actually, I think the cleanest approach is:

1. Normalize so sum(f) = 1
2. At each step, compute the soft gradient
3. Take a step: f_new = f - eta * grad
4. Project to simplex (non-negative, sum = 1)
5. Accept/reject based on score change

The step size eta should be related to the "curvature" of the problem. Since we're minimizing the max of quadratic functions, the effective curvature is roughly O(1) (the second derivative of a[k] with respect to f is O(1)). So eta should be O(1) for reasonable progress.

Let me set eta = 1.0 initially and decrease it over time.

Actually, I just realized there might be an issue with my gradient formula. Let me double-check.

We have h_k(f) = sum_{i: valid} f[i] * f[k-i]

∂h_k/∂f[j] = f[k-j] (from the i=j term) + f[k-j] (from the i=k-j term, if k-j ≠ j)
= 2*f[k-j] if 0 <= k-j < n and j ≠ k/2
= 2*f[k-j] if j = k/2 (since f[j]*f[j] has derivative 2*f[j])

Wait, let me redo this. h_k = sum_i f[i] * f[k-i] where i ranges over valid indices.

∂h_k/∂f[j] = sum_i [δ(i,j) * f[k-i] + f[i] * δ(k-i, j)]
= f[k-j] + sum_i f[i] * δ(i, k-j)
= f[k-j] + f[k-j] (if 0 <= k-j < n)
= 2 * f[k-j]

Yes, so the gradient is always 2*f[k-j] for valid k-j. My formula is correct.

Now, for the soft gradient:
grad[j] = sum_k w[k] * ∂h_k/∂f[j] = sum_k w[k] * 2*f[k-j] (for 0 <= k-j < n)
= 2 * sum_{k: 0<=k-j<n} w[k] * f[k-j]
= 2 * sum_{m: 0<=m<n} w[j+m] * f[m]
= 2 * correlate(w, f, 'valid')[j]

```python
with softmax-weighted gradient, simulated annealing, adaptive step size,

∇h_k[j] = 2*f[k-j] for valid indices, enabling O(n log n) per iteration.

def _fast_score(f):
    """Compute score without security checks (fast path)."""
    c = np.convolve(f, f)
    max_b = c.max()
    sum_a = f.sum()
    if sum_a < 1e-10:
        return np.inf
    return 2.0 * len(f) * max_b / (sum_a * sum_a)


def _simplex_project(v):
    """Project v onto {x >= 0, sum(x) = 1} via water-filling."""
    n = len(v)
    v_sum = v.sum()
    if v_sum < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    idx = np.max(np.nonzero(cond)[0])
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    proj = np.maximum(v - theta, 0.0)
    s = proj.sum()
    if s > 1e-15:
        proj /= s
    else:
        proj = np.ones(n) / n
    return proj


def _soft_gradient(f, c, T):
    """Compute softmax-weighted subgradient: 2*correlate(softmax(c/T), f, 'valid')."""
    n = len(f)
    cs = c / max(T, 1e-30)
    cs -= cs.max()
    w = np.exp(cs)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(len(c))
        w[n - 1] = 1.0
    else:
        w /= s
    grad = 2.0 * correlate(w, f, mode='valid')
    return grad


def _peak_lags(c, frac=0.95):
    """Return indices of lags that are within frac of the maximum."""
    peak = c.max()

def _exact_subgradient(f, c, frac=0.95):
    """Average subgradient over near-peak lags."""
    n = len(f)
    active = _peak_lags(c, frac)
    grad = np.zeros(n)
    for k in active:
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        if j_end >= j_start:
            grad[j_start:j_end + 1] += 2.0 * f[k - j_start:k - j_end + 1][::-1]
    grad /= len(active)
    return grad
```
