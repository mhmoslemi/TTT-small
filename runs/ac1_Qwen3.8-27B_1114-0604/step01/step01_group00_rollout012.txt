        return f.copy()
    return f_new / s

"""
Step function optimizer: multi-phase approach combining mirror descent,
multi-temperature soft-max gradients, peak-aware local search, and fine-tuning.
"""

import numpy as np
from scipy.signal import correlate
import time

def _convolve_fft(f):
    """Full linear convolution f*f via FFT. Returns array of length 2n-1."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]

def _eval_seq(f):
    """Evaluate sequence using the provided evaluate_sequence function."""
    return evaluate_sequence(f.tolist())

def _multi_temp_gradient(c, f, temps):
    """
    Compute gradient of soft-approximation to max(c) using multiple temperatures.
    Returns gradient array of length n.
    """
    n = len(f)
    grad = np.zeros(n)
    for T in temps:
        cs = c / T
        cs = cs - cs.max()
        p = np.exp(cs)
        s = p.sum()
        if s < 1e-300:
            p = np.zeros(len(c))
            p[n - 1] = 1.0
        else:
            p /= s
        grad += correlate(p, f, mode='valid')
    return 2.0 * grad

def _mirror_step(f, grad, step):
    """Mirror descent step: f_new = softmax(log(f) - step * grad)."""
    log_f = np.log(f + 1e-30)
    log_f -= step * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy()
    return f_new / s

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum x = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _peak_contrib(c, f, k_star):
    """Compute contribution of each position to c[k_star]."""
    n = len(f)
    contrib = np.zeros(n)
    # c[k_star] = sum_j f[j] * f[k_star - j]
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            contrib[j] = f[j] * f[idx]
        # Also: if k_star - j != j, then f[k_star-j]*f[j] is counted once in the sum
        # But the formula c[k] = sum_i f[i]*f[k-i] counts each pair once
        # So contrib[j] should be f[j]*f[k_star-j] (which is what we have)
    return contrib

def _optimize_phase1(f, n, deadline, rng, label=""):
    """
    Phase 1: Mirror descent with multi-temperature gradient and adaptive step.
    """
    f = np.maximum(f, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f /= s
    
    best_score = _eval_seq(f)
    best_f = f.copy()
    
    c = _convolve_fft(f)
    peak0 = c.max()
    
    stuck = 0
    it = 0
    max_iters = 300000
    eval_interval = 1500
    
    while it < max_iters:
        if time.monotonic() > deadline:
            break
        it += 1
        progress = min(1.0, it / max_iters)
        
        # Multi-temperature: use 3 temperatures
        # T1: captures global structure (high T)
        # T2: intermediate
        # T3: focuses on peak (low T)
        T_scale = peak0 * max(1e-13, (1.0 - progress) ** 2)
        T1 = T_scale * 10.0
        T2 = T_scale * 3.0
        T3 = T_scale * 1.0
        temps = [T1, T2, T3]
        
        # Step size: adaptive
        step = 0.6 * (1.0 - progress) ** 0.6 + 0.005
        
        c = _convolve_fft(f)
        grad = _multi_temp_gradient(c, f, temps)
        f = _mirror_step(f, grad, step)
        
        # Evaluate periodically
        if it % eval_interval == 0:
            score = _eval_seq(f)
            if score < best_score - 1e-8:
                best_score = score
                best_f = f.copy()
                stuck = 0
            else:
                stuck += 1

                # Peak-aware perturbation
                c = _convolve_fft(f)
                peak_k = int(np.argmax(c))
                contrib = _peak_contrib(c, f, peak_k)
                
                # Move mass from high-contrib to low-contrib
                top = np.argsort(contrib)[::-1][:max(5, n // 50)]
                bottom = np.argsort(contrib)[:max(5, n // 50)]
                
                f_pert = f.copy()

        best_f = f.copy()
