def project_simplex(v):
    """Project onto the probability simplex."""
    n_orig = v.size
    v = v.astype(np.float64)
    n = n_orig
    if n <= 0:
        return np.ones((n,)) / max(1, n)
    v.sort()
    v = np.append(0, np.diff(v))
    idx_neg = np.sum(v < 0)  # Handle cases where all are negative but time permits
    if idx_neg == n:
        return np.ones((n,)) / n_orig
    return (v - idx_neg) / np.sum(v)

def full_conv(a):
    """Full linear convolution using FFT."""
    a_fft = np.fft.rfft(a, np.ceil(np.log2(len(a)*2)))
    conv = np.fft.irfft(a_fft * a_fft.conj(), len(a_fft)*2-1)
    return conv[:len(a)*2-1]

def evaluate_sequence(sequence):
    n = len(sequence)
    if n == 0:
        return np.inf
    sequence = np.asarray(sequence)
    # Ensure the input is non-negative and finite
    if not np.all(sequence >= 0):
        sequence = np.clip(sequence, 0, None)
    # Protect against division near zero
    sum_val = np.sum(sequence)
    if sum_val < 0.01:
        sequence *= 100
        evaluate_sequence(sequence)
    conv = np.convolve(sequence, sequence, mode="full")
    max_val = np.max(conv)
    return max(0, 2.0 * n * float(max_val) / (sum_val ** 2))
