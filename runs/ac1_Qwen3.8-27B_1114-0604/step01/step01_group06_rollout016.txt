_N = 8192
_NFFT = 0
_BOUNDS = 30.0

def _set_state(n):
    global _N, _NFFT
    _N = n
    _NFFT = 1 << (3*n - 2).bit_length()

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = e.sum()
    if s < 1e-300:
        return np.ones(_N) / _N
    return e / s

def _fft_conv(a):
    N = _NFFT
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*_N-1].real
    return np.maximum(c, 0.0)

def _score(a):
    s = a.sum()
    if s < 1e-12:
        return np.inf
    c = _fft_conv(a)
    return 2.0 * _N * c.max() / (s*s)

def _smooth_max_grad(c, tau):
    m = c.max()
    diff = np.clip((c - m) / tau, -700, 0)
    e = np.exp(diff)
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return m, np.full_like(c, 1.0/c.size)
    p = e / s
    F = m + tau * np.log(s)
    return F, p

def _compute_grad_a(a, p):
    """Compute dF/da where F = sum p_k * c_k, c = conv(a,a)."""
    N = _NFFT
    a_pad = np.zeros(N)
    a_pad[:_N] = a
    p_pad = np.zeros(N)
    p_pad[:2*_N-1] = p
    fa = np.fft.rfft(a_pad)
    fp = np.fft.rfft(p_pad)
    grad_a = 2.0 * np.fft.irfft(np.conj(fp) * fa)[:_N]
    return grad_a

def _adam_optimize(z0, deadline, tau_start=0.05, tau_end=1e-5, lr=0.01, warmup=200):
    n = _N
    N = _NFFT
    z = z0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    
    a = _softmax(z)
    best_a = a.copy()
    best_F = _fft_conv(a).max()
    
    total_time = deadline - time.monotonic()
    step_time_est = 0.005  # ~5ms per step for n=8192
    n_steps_est = total_time / step_time_est
    
    log_tau_start = np.log(tau_start)
    log_tau_end = np.log(tau_end)
    
    patience = 0
    max_patience = 1000
    
    rng = np.random.default_rng(12345)
    
    while time.monotonic() < deadline:
        t += 1
        progress = min(1.0, t / max(100, n_steps_est))
        
        # Temperature schedule
        log_tau = log_tau_start + progress * (log_tau_end - log_tau_start)
        tau = np.exp(log_tau)
        
        # Learning rate schedule
        if t <= warmup:
            lr_t = lr * t / warmup
        else:
            # Cosine decay
            decay = 0.5 * (1 + np.cos(np.pi * (t - warmup) / max(1, n_steps_est - warmup)))
            lr_t = lr * decay
        
        # Compute
        a = _softmax(z)
        c = _fft_conv(a)
        F, p = _smooth_max_grad(c, tau)
        grad_a = _compute_grad_a(a, p)
        grad_z = a * (grad_a - np.dot(grad_a, a))
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_z
        v = beta2 * v + (1 - beta2) * grad_z * grad_z
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        z = np.clip(z, -_BOUNDS, _BOUNDS)
        z = z - np.mean(z)
        
        # Check exact score periodically
        if t % 100 == 0:
            F_exact = c.max()  # we already have c
            if F_exact < best_F - 1e-15:
                best_F = F_exact
                best_a = a.copy()
                patience = 0
            else:
                patience += 100
        
        if patience >= max_patience:
            # Random kick
            kick = 0.05 * rng.standard_normal(n)
            z = z + kick
            z = np.clip(z, -_BOUNDS, _BOUNDS)
            z = z - np.mean(z)
            patience = 0
            
        if t % 500 == 0:
            score = 2.0 * n * best_F
            print(f"  t={t} tau={tau:.2e} F={F:.8f} best_F={best_F:.10f} score={score:.9f} time_left={deadline-time.monotonic():.0f}s")
    
    return best_a, 2.0 * n * best_F

Actually, you know what, let me also try a completely different approach: **randomized coordinate descent / Frank-Wolfe type methods**.

The Frank-Wolfe (conditional gradient) method works as follows:
1. At current point a, compute the gradient g = dF/da
2. Find the vertex v of the feasible set that minimizes g^T v (i.e., the "linear minimization oracle")
3. Move: a_new = (1-t)*a + t*v for some step size t

For our problem, the feasible set is {a : a_i >= 0, sum(a_i) = 1} (the probability simplex). The linear minimization oracle is: v = e_{argmin_i g_i} (the unit vector at the coordinate with the smallest gradient component).

This is very simple! Each step is:
1. Compute gradient g (O(n log n) with FFT)
2. Find argmin of g (O(n))
3. Update a = (1-t)*a + t*e_j (O(n))

The step size t can be chosen as t = 2/(k+2) (standard Frank-Wolfe) or by line search.

The advantage of Frank-Wolfe is that it's very simple, doesn't require a parameterization, and naturally handles the non-negativity constraint. The disadvantage is that it can be slow to converge (sublinear rate).

However, for our problem, the "smooth max" objective is not convex (since the convolution is quadratic), so Frank-Wolfe doesn't have the same convergence guarantees. But it might still work as a heuristic.

Let me also think about whether the objective is "almost" convex. The function max_k c[k] is the max of quadratic functions, which is convex. But c[k] = sum_i a[i]*a[k-i] is NOT convex in a (it's a quadratic form that's not positive semi-definite in general).

For k=0: c[0] = sum a[i]^2, which IS convex.
For k>0: c[k] = sum a[i]*a[i+k] (approximately), which is a bilinear form, NOT convex.

So the overall objective is non-convex, and we can get stuck in local minima. This is why multiple starts are important.
