```python
def _error_reduction(f, n, max_iters, deadline):
    """Gerchberg-Saxton error reduction for flat convolution."""
    f = np.maximum(f, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s
    
    size = 1 << (2 * n - 1).bit_length()
    target_mag = np.sqrt(size / (2 * n - 1))
    
    best_f = f.copy()
    best_peak = _autocorr_fft(f).max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Forward transform
        F = np.fft.rfft(f, size)
        
        # Replace magnitude with target
        phase = np.angle(F)
        F_new = target_mag * np.exp(1j * phase)
        
        # Inverse transform
        f_new = np.fft.irfft(F_new, size)[:n]
        
        # Enforce non-negativity
        f_new = np.maximum(f_new, 0.0)
        
        # Normalize
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new = f_new / s
        f = f_new
        
        # Track best
        if it % 100 == 0:
            conv = _autocorr_fft(f)
            peak = conv.max()
            if peak < best_peak:
                best_peak = peak
                best_f = f.copy()
    
    conv = _autocorr_fft(f)
    if conv.max() < best_peak:
        best_peak = conv.max()
        best_f = f.copy()
    
    return best_f, best_peak
```

```python
"""
Step function optimizer using Gerchberg-Saxton error reduction combined
with gradient-based peak reduction and coordinate polish.
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
import time
from scipy.signal import correlate


def _conv_fft(f):
    """Full linear convolution f*f via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1], size


def _conv_fft_size(f, size):
    """Full linear convolution f*f via FFT with pre-computed size."""
    n = len(f)
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _eval(f):
    """Evaluate the objective function."""
    return evaluate_sequence(f.tolist())


def _error_reduce(f, n, size, max_iters, deadline, relax=1.0, rng=None):
    """
    Gerchberg-Saxton error reduction for flat convolution.
    Returns (best_f, best_peak).
    """
    f = np.maximum(f, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s
    
    target_mag = np.sqrt(size / (2 * n - 1))
    best_f = f.copy()
    conv, _ = _conv_fft(f)
    best_peak = conv.max()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        # Forward transform
        F = np.fft.rfft(f, size)
        
        # Replace magnitude with target (with relaxation)
        current_mag = np.abs(F)
        # Blend: new_mag = (1-relax)*current_mag + relax*target_mag
        # Actually, standard GS just replaces magnitude entirely
        phase = np.angle(F)
        F_new = target_mag * np.exp(1j * phase)
        
        # Inverse transform
        f_new = np.fft.irfft(F_new, size)[:n]
        
        # Enforce non-negativity
        f_new = np.maximum(f_new, 0.0)
        
        # Normalize
        s = f_new.sum()
        if s < 1e-10:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / s
        f = f_new
        
        # Track best every 200 iterations
        if it % 200 == 0:
            conv = _conv_fft_size(f, size)
            peak = conv.max()
            if peak < best_peak:
                best_peak = peak
                best_f = f.copy()
    
    conv = _conv_fft_size(f, size)
    if conv.max() < best_peak:
        best_peak = conv.max()
        best_f = f.copy()
    
    return best_f, best_peak


def _gradient_peak_reduce(f, n, size, deadline, max_iters=50000):
    """
    Gradient-based peak reduction using soft-max gradient.
    Minimizes max(conv) via mirror descent.
    """
    f = np.maximum(f, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s
    
    best_f = f.copy()
    conv = _conv_fft_size(f, size)
    best_peak = conv.max()
    T = best_peak
    T_min = best_peak * 1e-10
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        progress = it / max_iters
        T = best_peak * (T_min / best_peak) ** progress
        step = 1.0 * (0.01 / 1.0) ** (progress ** 0.8)
        
        # Compute soft gradient
        conv = _conv_fft_size(f, size)
        cs = conv / T
        cs -= cs.max()
        p = np.exp(cs)
        ps = p.sum()
        if ps < 1e-300:
            p = np.zeros(2 * n - 1)
            p[n - 1] = 1.0
        else:
            p = p / ps
        
        # Gradient: 2 * correlate(p, f, 'valid')
        grad = 2.0 * correlate(p, f, mode='valid')
        
        # Mirror descent (exponentiated gradient)
        span = grad.max() - grad.min()
        if span < 1e-15:
            break
        eta = step / span
        log_f = np.log(f + 1e-300) - eta * grad
        log_f -= log_f.max()
        f_new = np.exp(log_f)
        f_new = f_new / f_new.sum()
        f = f_new
        
        # Track best
        if it % 1000 == 0:
            conv = _conv_fft_size(f, size)
            peak = conv.max()
            if peak < best_peak - 1e-8:
                best_peak = peak
                best_f = f.copy()
    
    conv = _conv_fft_size(f, size)
    if conv.max() < best_peak:
        best_peak = conv.max()
        best_f = f.copy()
    
    return best_f, best_peak


def _polish(f, n, size, deadline, rng):
    """
    Coordinate-descent polish: try moving mass from high-contribution
    positions to low-contribution positions.
    """
    f = f.copy()
    conv = _conv_fft_size(f, size)
    best_peak = conv.max()
    best_f = f.copy()
    
    n_passes = 2000
    for p in range(n_passes):
        if time.monotonic() > deadline:
            break
        
        conv = _conv_fft_size(f, size)
        peak = conv.max()
        k_star = int(np.argmax(conv))
        
        # Contribution of each position j to conv[k_star]
        # conv[k_star] = sum_j f[j] * f[k_star - j]
        # d(conv[k_star])/d(f[j]) = f[k_star - j] + f[k_star - j] (if j != k_star-j)
        # Actually d(conv[k])/d(f[j]) = f[k-j] for the term where first index is j
        # plus f[k-j] for the term where second index is j
        # So d(conv[k])/d(f[j]) = f[k-j] (if 0 <= k-j < n) + f[k-j] (same thing)
        # Wait, conv[k] = sum_j f[j]*f[k-j]. d/d(f[j]) = f[k-j] (if 0<=k-j<n)
        # But also if j appears as the second index: conv[k] includes f[k-j']*f[j] where j'=k-j
        # So d(conv[k])/d(f[j]) = f[k-j] + f[k-j] = 2*f[k-j] if k != 2j
        # and f[k-j] if k = 2j (i.e., j = k/2)
        
        # Simpler: just compute the contribution directly
        contrib = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                if j == idx:
                    contrib[j] = f[j] * f[j]  # derivative is f[j]
                else:
                    contrib[j] = 2.0 * f[j] * f[idx]  # derivative is 2*f[idx]... 
                    # Actually the derivative of conv[k_star] w.r.t. f[j] is:
                    # f[k_star - j] (from the term f[j]*f[k_star-j])
                    # + f[k_star - j] (from the term f[k_star-j]*f[j] when j' = k_star-j)
                    # = 2*f[k_star-j] if j != k_star-j
                    # = f[j] if j == k_star-j
                    # So contribution for "how much f[j] contributes to the peak" is f[j]*f[k_star-j]
                    # but for gradient descent, the gradient is 2*f[k_star-j] (or f[j] if j==idx)
                pass
        
        # Actually, let me just use the gradient directly
        # grad[j] = d(max_conv)/d(f[j]) ≈ 2*f[k_star - j] (if 0 <= k_star-j < n)
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                if j == idx:
                    grad[j] = f[idx]
                else:
                    grad[j] = 2.0 * f[idx]
        
        # Decrease f where grad is large, increase where grad is small
        # Find top and bottom gradient positions
        grad_sorted = np.argsort(grad)
        j_donate = int(np.argmax(grad))  # highest gradient -> decrease
        j_receive = int(grad_sorted[0])  # lowest gradient -> increase
        
        if j_donate == j_receive or f[j_donate] < 1e-15:
            continue
        
        # Try different transfer amounts
        for frac in [0.01, 0.05, 0.1, 0.2, 0.5]:
            amt = f[j_donate] * frac
            if amt > f[j_donate]:
                continue
            f_try = f.copy()
            f_try[j_donate] -= amt
            f_try[j_receive] += amt
            f_try = np.maximum(f_try, 0)
            s = f_try.sum()
            if s < 1e-10:
                continue
            f_try /= s
            conv_try = _conv_fft_size(f_try, size)
            if conv_try.max() < peak - 1e-12:
                f = f_try
                best_peak = conv_try.max()
                best_f = f_try.copy()
                break
        
        # Random perturbation occasionally
        if p % 50 == 0:
            f_try = f + 0.001 * rng.standard_normal(n)
            f_try = np.maximum(f_try, 0)
            f_try /= f_try.sum()
            conv_try = _conv_fft_size(f_try, size)
            if conv_try.max() < best_peak:
                f = f_try
                best_f = f_try.copy()
    
    return best_f, best_peak
```

```python
"""
Step function optimizer using spectral error reduction (Gerchberg-Saxton variant)
with adaptive peak-shaving target, combined with gradient-based refinement and
coordinate polish. Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
"""

import numpy as np
import time
from scipy.signal import correlate


def _conv_fft(f, size):
    """Full linear convolution f*f via FFT with given FFT size."""
    n = len(f)
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _eval_local(f):
    """Local evaluation (same as evaluate_sequence but without the overhead of list conversion)."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    conv = _conv_fft(f, size)
    max_b = conv.max()
    sum_a = f.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a ** 2))


def _spectral_reduce(f, n, size, deadline, max_iters=200000, rng=None):
    """
    Spectral error reduction with adaptive peak-shaving.
    Iteratively enforces a target convolution shape in the frequency domain
    and non-negativity in the spatial domain.
    """
    f = np.maximum(f, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s
    
    conv = _conv_fft(f, size)
    best_peak = conv.max()
    best_f = f.copy()
    
    # Annealing: threshold goes from current peak to ideal flat value
    ideal = 1.0 / (2 * n - 1)
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        progress = min(1.0, it / max_iters)
        # Anneal threshold from current peak to ideal
        threshold = best_peak + (ideal - best_peak) * (progress ** 0.5)
        
        conv = _conv_fft(f, size)
        
        # Create target convolution: cap the peak
        target_conv = conv.copy()
        target_conv[target_conv > threshold] = threshold
        # Ensure target is non-negative (should be, since conv >= 0 for non-neg f)
        target_conv = np.maximum(target_conv, 0)
        
        # Compute target magnitude in frequency domain
        # FFT(target_conv_padded) should = |F|^2, so |F| = sqrt(|FFT(target_conv_padded)|)
        target_padded = np.zeros(size)
        target_padded[:2*n-1] = target_conv
        target_fft = np.fft.rfft(target_padded)
        target_mag = np.sqrt(np.abs(target_fft))
        
        # Get current phase
        F = np.fft.rfft(f, size)
        phase = np.angle(F)
        
        # New spectrum: target magnitude, current phase
        F_new = target_mag * np.exp(1j * phase)
        
        # Inverse transform
        f_new = np.fft.irfft(F_new, size)[:n]
        
        # Enforce non-negativity
        f_new = np.maximum(f_new, 0.0)
        
        # Normalize
        s = f_new.sum()
        if s < 1e-10:
            f_new = np.ones(n) / n
        else:
            f_new = f_new / s
        f = f_new
        
        # Track best
        if it % 500 == 0:
            conv_check = _conv_fft(f, size)
            peak = conv_check.max()
            if peak < best_peak:
                best_peak = peak
                best_f = f.copy()
    
    conv_check = _conv_fft(f, size)
    if conv_check.max() < best_peak:
        best_peak = conv_check.max()
        best_f = f.copy()
    
    return best_f, best_peak


def _gs_reduce(f, n, size, deadline, max_iters=200000):

    best_f = f.copy()
    
    for it in range(max_iters):
        if time.monotonic() > deadline:

        F = np.fft.rfft(f, size)
        phase = np.angle(F)
        F_new = target_mag * np.exp(1j * phase)
        f_new = np.fft.irfft(F_new, size)[:n]
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:

        f = f_new

            conv_check = _conv_fft(f, size)

            if peak < best_peak:

        best_peak = conv_check.max()
        best_f = f.copy()

def _gradient_refine(f, n, size, deadline, max_iters=30000):
    """
    Gradient-based refinement using soft-max gradient and mirror descent.
    """
    f = np.maximum(f, 0.0).astype(np.float64)
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s
    
    conv = _conv_fft(f, size)
    best_peak = conv.max()
    best_f = f.copy()
    T = best_peak
    T_min = best_peak * 1e-12
    
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        
        progress = min(1.0, it / max_iters)
        T = best_peak * (T_min / best_peak) ** (progress ** 0.5)
        step = 0.5 * (0.01 / 0.5) ** progress
        
        conv = _conv_fft(f, size)
        
        # Soft-max gradient
        cs = conv / T
        cs -= cs.max()
        p = np.exp(cs)
        ps = p.sum()
        if ps < 1e-300:
            p = np.zeros(2 * n - 1)
            p[n - 1] = 1.0
        else:
            p = p / ps
        
        grad = 2.0 * correlate(p, f, mode='valid')
        
        # Mirror descent step
        span = grad.max() - grad.min()
        if span < 1e-15:
            break
        eta = step / span
        log_f = np.log(f + 1e-300) - eta * grad
        log_f -= log_f.max()
        f_new = np.exp(log_f)
        f_new = f_new / f_new.sum()
        f = f_new
        
        if it % 2000 == 0:
            conv_check = _conv_fft(f, size)
            peak = conv_check.max()
            if peak < best_peak - 1e-9:
```

```python
"""
Step function optimizer using spectral error reduction (Gerchberg-Saxton variant)
with adaptive peak-shaving target, combined with gradient-based refinement and
coordinate polish. Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.

import numpy as np

def _conv_fft(f, size):
    """Full linear convolution f*f via FFT with given FFT size."""
    n = len(f)
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _eval_local(f):
    """Compute objective value locally (avoids list conversion overhead)."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    conv = _conv_fft(f, size)
    max_b = conv.max()
    sum_a = f.sum()
    if sum_a < 0.01:
        return np.inf
    return float(2 * n * max_b / (sum_a ** 2))


def _normalize(f):
    """Normalize to sum=1, non-negative."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-15:
        return np.ones(len(f)) / len(f)
    return f / s

    Anneals a threshold from current peak toward ideal flat value.
    Uses random phase perturbations to escape local minima.

    best_peak = conv.max()
    best_f = f.copy()

    improvement_threshold = 1e-10

            break

    Gradient-based refinement using soft-max gradient and mirror descent.

            break

    starts['sine'] = 1.0 + 0.7 * np.sin(2.0 * np.pi * j / n) ** 2
    starts['cos2'] = 1.0 + 0.5 * np.cos(4.0 * np.pi * j / n)
    starts['random1'] = rng.exponential(1.0, size=n)
    starts['random2'] = rng.dirichlet(np.ones(n) * 0.05)
    starts['random3'] = np.abs(rng.standard_normal(n)) + 0.01
    starts['power'] = (j / (n - 1)) ** 2 + ((n - 1 - j) / (n - 1)) ** 2 + 0.001
    
    # Normalize all starts
    for name in starts:
        starts[name] = _normalize(starts[name])
    
    # Evaluate and rank all starts
    print("\nStarting point scores:")
    start_scores = []
    for name, f0 in starts.items():
        conv = _conv_fft(f0, size)
        peak = conv.max()
        score = 2.0 * n * peak
        start_scores.append((score, name, f0))
        print(f"  {name}: {score:.6f}")
    
    start_scores.sort(key=lambda x: x[0])
    update_best(start_scores[0][2])
    print(f"\nBest start: {start_scores[0][1]} with score {start_scores[0][0]:.6f}")
    
    # === Phase 1: Adaptive Spectral Error Reduction (top 4 starts) ===
    print(f"\n{'='*50}")
    print("Phase 1: Adaptive Spectral Error Reduction")
    print(f"{'='*50}")
    
    top_starts = start_scores[:4]
    phase1_budget = min(budget_s * 0.35, time.monotonic() + budget_s * 0.35 - start_time)
    time_per = phase1_budget / len(top_starts)
    
    for idx, (sc, name, f0) in enumerate(top_starts):
        if time.monotonic() > deadline - 120:
            break
        t_alloc = time_per * (1.0 - 0.15 * idx)
        d = min(time.monotonic() + t_alloc, deadline - 90)
        
        print(f"\n  [{name}] budget={t_alloc:.0f}s, initial={sc:.6f}")
        f_opt, peak = _spectral_reduce_adaptive(f0, n, size, d, max_iters=1000000, rng=rng)
        score = 2.0 * n * peak
        print(f"  [{name}] result: peak={peak:.12f}, score={score:.6f}")
        update_best(f_opt)
    
    # === Phase 2: Gerchberg-Saxton with flat target ===
    print(f"\n{'='*50}")
    print("Phase 2: Gerchberg-Saxton (flat target)")
    print(f"{'='*50}")
    
    if time.monotonic() < deadline - 90:
        gs_budget = min(budget_s * 0.15, deadline - 90 - time.monotonic())
        d = time.monotonic() + gs_budget
        print(f"  Budget: {gs_budget:.0f}s, starting from best (score={best_score:.6f})")
        f_gs, peak = _gs_flat(best_f.copy(), n, size, d, max_iters=1000000, rng=rng)
        score = 2.0 * n * peak
        print(f"  GS result: peak={peak:.12f}, score={score:.6f}")
        update_best(f_gs)
        
        # Also try GS from the second-best start
        if time.monotonic() < deadline - 60 and len(start_scores) > 1:
            d2 = min(time.monotonic() + 30, deadline - 60)
            f_gs2, peak2 = _gs_flat(start_scores[1][2], n, size, d2, max_iters=200000, rng=rng)
            score2 = 2.0 * n * peak2
            print(f"  GS from 2nd start: score={score2:.6f}")
            update_best(f_gs2)
    
    # === Phase 3: Gradient Refinement ===
    print(f"\n{'='*50}")
    print("Phase 3: Gradient Refinement (Mirror Descent)")
    print(f"{'='*50}")
    
    if time.monotonic() < deadline - 30:
        grad_budget = min(budget_s * 0.2, deadline - 30 - time.monotonic())
        d = time.monotonic() + grad_budget
        print(f"  Budget: {grad_budget:.0f}s")
        f_grad, peak = _gradient_refine(best_f.copy(), n, size, d, max_iters=200000)
        score = 2.0 * n * peak
        print(f"  Gradient result: peak={peak:.12f}, score={score:.6f}")
        update_best(f_grad)
    
    # === Phase 4: Polish ===
    print(f"\n{'='*50}")
    print("Phase 4: Coordinate Polish")
    print(f"{'='*50}")
    
    if time.monotonic() < deadline - 5:
        polish_budget = min(budget_s * 0.15, deadline - 5 - time.monotonic())
        d = time.monotonic() + polish_budget
        print(f"  Budget: {polish_budget:.0f}s")
        f_pol, peak = _polish(best_f.copy(), n, size, d, rng)
        score = 2.0 * n * peak
        print(f"  Polish result: peak={peak:.12f}, score={score:.6f}")
        update_best(f_pol)
    
    # === Final: One more short spectral pass from the best ===
    if time.monotonic() < deadline - 3:
        print(f"\nFinal spectral pass...")
        d_final = deadline
        f_final, peak_final = _spectral_reduce_adaptive(best_f, n, size, d_final, max_iters=200000, rng=rng)
        score_final = 2.0 * n * peak_final
        print(f"  Final spectral: score={score_final:.6f}")
        update_best(f_final)
    
    # Final verification with actual evaluator
    result = [float(max(0.0, x)) for x in best_f]
    final_score = evaluate_sequence(result)
    print(f"\n{'='*50}")
    print(f"FINAL SCORE (evaluator): {final_score:.6f}")
    print(f"{'='*50}")
    
    return result
```
