Minimizes 2*n*max(f*f)/(sum f)^2 for non-negative sequences.

def _autocorr(f):
    """Compute linear autocorrelation (full convolution with itself)."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.fft(f, size)
    b = np.fft.ifft(F * np.conj(F), size).real[:2*n-1]
    return b

def _soap_step(f, alpha, rng):
    """One SOAP-style frequency-domain peak shaving step."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.fft(f, size)
    B = np.abs(F)**2
    b = np.fft.ifft(B, size).real[:2*n-1]
    
    # Find peak lag
    k_star = int(np.argmax(b))
    
    # Compute the cosine fingerprint
    m = np.arange(size)
    cos_term = np.cos(2 * np.pi * k_star * m / size)
    
    # Max alpha such that B_new >= 0
    # B[m] - alpha*cos_term[m] >= 0 for all m
    # Where cos_term > 0: alpha <= B[m] / cos_term[m]
    # Where cos_term <= 0: no constraint (subtracting negative = adding)
    pos_mask = cos_term > 1e-10
    if np.any(pos_mask):
        max_alpha = np.min(B[pos_mask] / cos_term[pos_mask])
    else:
        max_alpha = alpha
    
    alpha_use = min(alpha, max_alpha * 0.99)
    if alpha_use < 1e-15:
        return f.copy()
    
    B_new = B - alpha_use * cos_term
    B_new = np.maximum(B_new, 0)
    
    # Reconstruct F with modified amplitude, same phase
    phase = np.angle(F)
    F_new = np.sqrt(B_new) * np.exp(1j * phase)
    
    f_new = np.fft.ifft(F_new, size).real[:n]
    f_new = np.maximum(f_new, 0)
    s = f_new.sum()
    if s < 1e-10:
        return f.copy()
    f_new /= s
    return f_new

def _subgrad_mirror_step(f, eta):
    """Exact subgradient mirror descent step."""
    n = len(f)
    b = _autocorr(f)
    k_star = int(np.argmax(b))
    
    # Subgradient of b_{k_star} w.r.t. f
    # db_k/db_f_j = f_{j+k} + f_{j-k} (with zero padding)
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        idx1 = j + k_star
        idx2 = j - k_star
        if 0 <= idx1 < n:
            val += f[idx1]
        if 0 <= idx2 < n:
            val += f[idx2]
        g[j] = 2.0 * val
    
    # Mirror descent: f_new = f * exp(-eta * g), normalize
    log_f = np.log(np.maximum(f, 1e-300)) - eta * g
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    f_new /= f_new.sum()
    return f_new, b, k_star

    f = f.copy()

    local_best_f = f.copy()

        low = int(np.argmin(g * f))  # lowest contribution

            f_try[low] += amt

                peak = b_try.max()
                b = b_try
