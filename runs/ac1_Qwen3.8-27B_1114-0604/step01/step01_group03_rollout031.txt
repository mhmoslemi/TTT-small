```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 for non-negative sequences.
Uses FFT-based hybrid gradient (log-sum-exp + exact peak subgradient),
spectral domain flattening, multi-temperature annealing, adaptive
restarts, and scipy L-BFGS-B polish for final refinement.
"""

import math
import time
import numpy as np


def _normalize(a):
    """Normalize to sum=1, clip to non-negative."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300:
        a[:] = 1.0
        s = float(a.sum())
    a /= s
    return a


def _fft_len(n):
    """Next power of 2 >= 2n-1."""
    return 1 << (2 * n - 2).bit_length()


def _conv_self(a, L, m):
    """FFT-based self-convolution."""
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _score(c, n):
    """Compute 2*n*max(c)."""
    return float(2.0 * n * np.max(c))


def _logsumexp_weights(v, temp):
    """Compute LSE value and normalized weights."""
    vmax = float(np.max(v))
    z = (v - vmax) / temp
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-300:
        return vmax + 1e-30, np.full_like(v, 1.0 / len(v))
    w = e / s
    return vmax + temp * math.log(s), w


def _gradient_from_weights(a, w, L, m, n):
    """
    Compute gradient of sum_k w[k]*c[k] w.r.t. a.
    Uses: grad[j] = 2 * sum_k w[k] * a[k-j] (for valid k-j)
    Computed via FFT: 2 * ifft(FFT(w) * FFT(a_reversed))[n-1:m]
    """
    W = np.fft.rfft(w, L)
    Frev = np.fft.rfft(a[::-1], L)
    g = np.fft.irfft(W * Frev, L)[(n - 1):m]
    g = 2.0 * g
    np.maximum(g, 0.0, out=g)
    return g


def _gradient_peak(a, k_peak, n):
    """Exact subgradient of c[k_peak] w.r.t. a."""
    grad = np.zeros(n, dtype=np.float64)
    lo = max(0, k_peak - n + 1)
    hi = min(n - 1, k_peak)
    if lo <= hi:
        indices = np.arange(lo, hi + 1)
        grad[indices] = 2.0 * a[k_peak - indices]
    return grad


def _spectral_flatten(a, L, m, n, n_iter, strength):
    """Iteratively flatten |FFT(a)| while maintaining a >= 0."""
    a = _normalize(a)
    F_target = float(np.mean(np.abs(np.fft.rfft(a, L))))
    for _ in range(n_iter):
        A = np.fft.rfft(a, L)
        mag = np.abs(A)
        # Blend towards flat
        mag_new = (1.0 - strength) * mag + strength * F_target
        # Preserve DC
        mag_new[0] = mag[0]
        # Reconstruct
        phase = A / np.where(mag > 1e-30, mag, 1.0)
        A_new = mag_new * phase
        a_new = np.fft.irfft(A_new, L)[:n]
        a_new = np.maximum(a_new, 0.0)
        a = _normalize(a_new)
    return a


def _smooth_step(a, sigma, n):
    """Gaussian smoothing of the sequence."""
    if sigma < 0.5:
        return a
    x = np.arange(n) - n // 2
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    smoothed = np.convolve(a, kernel, mode='same')
    return _normalize(smoothed)


def _random_prior(n, rng):
    """Generate a random structured prior."""
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0
    z = rng.normal(0.0, 0.4)
    z += rng.normal(0.0, 0.5) * x
    z += rng.normal(0.0, 0.5) * (x * x - 1.0 / 3.0)
    kmax = int(rng.integers(2, 10))
    for k in range(1, kmax + 1):
        amp = rng.uniform(0.05, 0.4) / max(1.0, math.sqrt(float(k)))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos(2.0 * np.pi * k * t + ph)
    noise = rng.normal(0.0, 0.12, n)
    if n > 10:
        noise = np.convolve(noise, np.ones(7, dtype=np.float64) / 7.0, mode="same")
    z += noise
    p = np.exp(z - float(np.max(z)))
    p = np.maximum(p, 1e-300)
    return _normalize(p)


def _structured_priors(n, rng):
    """Generate structured initial candidates."""
    out = []
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0

    # Uniform
    out.append(_normalize(np.ones(n)))

    # Cosine / Hann
    out.append(_normalize(0.5 * (1 - np.cos(2 * np.pi * t))))

    # Raised cosine variants
    for beta in (0.3, 0.5, 0.7):
        rc = np.where(np.abs(x) < beta,
                      0.5 * (1 + np.cos(np.pi * x / beta)), 0.0)
        out.append(_normalize(rc))

    # Power-law shapes
    for p in (0.5, 1.0, 2.0, 3.0, 5.0):
        out.append(_normalize(np.abs(x) ** p + 0.1))

    # Multi-hump
    for n_h in (2, 3, 4, 5):
        humps = np.zeros(n)
        for i in range(n_h):
            center = (i + 0.5) / n_h
            humps += np.exp(-20.0 * (t - center) ** 2)
        out.append(_normalize(humps + 0.01))

    # Flat-top window
    a0, a1, a2, a3, a4 = 0.21557895, 0.41663158, 0.277263158, 0.083578947, 0.006947368
    ft = (a0 - a1 * np.cos(2 * np.pi * t)
          + a2 * np.cos(4 * np.pi * t)
          - a3 * np.cos(6 * np.pi * t)
          + a4 * np.cos(8 * np.pi * t))
    out.append(_normalize(np.maximum(ft, 0.0)))

    # Exponential decay (asymmetric)
    for rate in (1.0, 2.0, 3.0):
        out.append(_normalize(np.exp(-rate * t)))
        out.append(_normalize(np.exp(-rate * (1 - t))))

    # Random structured
    for _ in range(5):
        out.append(_random_prior(n, rng))

    return out


def _line_search(a, grad, L, m, n, current_score, max_factor=4.0):
    """Backtracking line search with multiplicative updates."""
    gmax = float(np.max(np.abs(grad)))
    if gmax < 1e-30:
        return a, current_score
    grad_normed = grad / gmax
    eta = 0.3
    best_a = a
    best_sc = current_score
    for factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
        step = eta * factor
        a2 = a * np.exp(-step * grad_normed)
        a2 = _normalize(a2)
        c2 = _conv_self(a2, L, m)
        sc2 = _score(c2, n)
        if sc2 < best_sc:
            best_sc = sc2
            best_a = a2
        elif sc2 < current_score:
            return a2, sc2
    return best_a, best_sc


def _main_optimize(a, L, m, n, rng, time_limit):
    """Main optimization loop."""
    deadline = time.monotonic() + float(time_limit)
    a = _normalize(a)
    c = _conv_self(a, L, m)
    best_a = a.copy()
    best_score = _score(c, n)

    eta = 0.25
    temp = 0.5
    no_improve = 0
    it = 0
    print(f"  [optimize] start: score={best_score:.6f}, n={n}")

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 2.0:
            break

        c = _conv_self(a, L, m)
        score = _score(c, n)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
            if it % 200 == 0:
                print(f"  [optimize] it={it}, improved: score={best_score:.6f}")
        else:
            no_improve += 1

        # Reset if stuck for too long
        if no_improve > 800:
            a = best_a.copy()
            temp = max(0.3, temp * 0.4)
            eta = 0.25
            no_improve = 0
            # Add small perturbation
            a += rng.normal(0.0, 0.002, n)
            a = _normalize(a)
            continue

        # Choose strategy
        phase = it % 20

        if phase < 10:
            # LSE gradient step
            v = n * c
            lse_val, w = _logsumexp_weights(v, temp)
            grad = _gradient_from_weights(a, w, L, m, n)
            a, score = _line_search(a, grad, L, m, n, score)
            eta = min(0.9, eta * 1.05)

        elif phase < 15:
            # Exact peak subgradient
            k_peak = int(np.argmax(c))
            grad = _gradient_peak(a, k_peak, n)
            a, score = _line_search(a, grad, L, m, n, score)
            eta = min(0.9, eta * 1.05)

        elif phase < 18:
            # Hybrid: 50/50 blend of LSE and peak gradient
            v = n * c
            lse_val, w = _logsumexp_weights(v, temp)
            grad_lse = _gradient_from_weights(a, w, L, m, n)
            k_peak = int(np.argmax(c))
            grad_peak = _gradient_peak(a, k_peak, n)
            # Normalize both to same scale
            ml = float(np.max(grad_lse))
            mp = float(np.max(grad_peak))
            if ml > 1e-30 and mp > 1e-30:
                grad_hybrid = 0.5 * (grad_lse / ml) + 0.5 * (grad_peak / mp)
            else:
                grad_hybrid = grad_lse
            a, score = _line_search(a, grad_hybrid, L, m, n, score)

        elif phase < 19:
            # Spectral flattening move
            strength = 0.05 + 0.15 * (temp / 500.0)
            a = _spectral_flatten(a, L, m, n, n_iter=3, strength=strength)

        else:
            # Small random perturbation
            a += rng.normal(0.0, 0.003, n)
            a = _normalize(a)

        it += 1

        # Anneal temperature
        if it % 150 == 0:
            temp = min(500.0, temp * 1.4)
        if it % 600 == 0:
            # Periodic structured restart
            prior = _random_prior(n, rng)
            mix = 0.05 + 0.15 * float(rng.random())
            a = _normalize((1.0 - mix) * best_a + mix * prior)
            temp = max(0.5, temp * 0.6)
            eta = 0.25

    return best_a, best_score
