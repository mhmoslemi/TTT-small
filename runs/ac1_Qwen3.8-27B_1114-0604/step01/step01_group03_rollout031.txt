import math

def _next_pow2(m):
    return 1 << (m - 1).bit_length()


def _normalize_simplex(a):
    """Normalize to non-negative simplex."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 0:
        return np.ones(len(a)) / len(a)
    a /= s
    return a


def _self_convolve(b, L):
    """Compute self-convolution of b using FFT. Returns array of length 2*len(b)-1."""
    F = np.fft.rfft(b, L)
    c = np.fft.irfft(F * F, L)
    return c[:2 * len(b) - 1]


def _cross_corr(w, b, L):
    """Compute cross-correlation: result[i] = sum_j w[i+j]*b[j] for i in range(len(b))."""
    W = np.fft.rfft(w, L)
    B = np.fft.rfft(b, L)
    r = np.fft.irfft(W * np.conj(B), L)
    return r[:len(b)]


def _logsumexp(v, T):
    """Compute T * logsumexp(v/T) stably."""
    vmax = float(np.max(v))
    z = (v - vmax) / T
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        return vmax, np.full_like(v, 1.0 / len(v))
    w = e / s
    val = T * (vmax / T + math.log(s))
    return val, w


def _score(b, L):
    """Compute 2n * max(conv(b,b)) for b on simplex."""
    c = _self_convolve(b, L)
    return 2.0 * len(b) * float(np.max(c))

    # grad[i] = 2 * sum_j w[i+j] * b[j] = 2 * cross_corr(w, b)[i]
    cc = _cross_corr(w, b, L3)
    return 2.0 * cc

"""
Search for nonnegative step heights minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses FFT-based gradient descent on the log-sum-exp approximation of the max-convolution,
with adaptive temperature annealing, multiplicative weight updates, structured perturbations,
and a dual-alternation polish phase.
"""

import math
import time
import numpy as np


def _next_pow2(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _normalize_simplex(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 0:
        return np.ones(len(a)) / len(a)
    a /= s
    return a


def _self_convolve(b, L):
    F = np.fft.rfft(b, L)
    c = np.fft.irfft(F * F, L)
    return c[:2 * len(b) - 1]


def _cross_corr(w, b, L):
    """Compute cross[i] = sum_j w[i+j]*b[j] for i in range(len(b))."""
    W = np.fft.rfft(w, L)
    B = np.fft.rfft(b, L)
    r = np.fft.irfft(W * np.conj(B), L)
    return r[:len(b)]


def _score(b, L):
    c = _self_convolve(b, L)
    return 2.0 * len(b) * float(np.max(c))


def _compute_grad(b, conv_b, T, L3):
    """Compute gradient of T*logsumexp(conv_b/T) w.r.t. b."""
    vmax = float(np.max(conv_b))
    z = (conv_b - vmax) / T
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.full(len(conv_b), 1.0 / len(conv_b))
    else:
        w = e / s
    cc = _cross_corr(w, b, L3)
    return 2.0 * cc


def _initial_candidates(n, rng):
    out = []
    # Parent sequence
    try:
        base = height_sequence_1
        arr = np.asarray(base, dtype=np.float64).ravel()
        if arr.size == n and np.all(np.isfinite(arr)) and np.any(arr > 0):
            out.append(_normalize_simplex(arr))
    except (NameError, Exception):
        pass

    # Uniform
    out.append(np.ones(n) / n)

    # Sine-based shapes
    t = np.linspace(0, 1, n)
    x = 2.0 * t - 1.0
    for freq in [1, 2, 3, 5, 7]:
        for phase in [0.0, 0.25, 0.5, 0.75]:
            z = 0.6 * np.cos(2 * np.pi * freq * t + 2 * np.pi * phase)
            p = np.exp(z)
            out.append(_normalize_simplex(p))

    # Quadratic
    for c in [-0.5, 0.0, 0.5, -1.0]:
        z = c * x * x
        p = np.exp(0.5 * z)
        out.append(_normalize_simplex(p))

    # Sine bump (zero at endpoints)
    bump = np.sin(np.pi * t)
    for alpha in [0.5, 1.0, 2.0, 3.0]:
        p = bump ** alpha
        out.append(_normalize_simplex(p))

    # Two-peak
    for c1, c2 in [(0.3, 0.7), (0.2, 0.8), (0.35, 0.65)]:
        z = -2.0 * (t - c1) ** 2 - 2.0 * (t - c2) ** 2
        p = np.exp(z)
        out.append(_normalize_simplex(p))

    # Random smooth
    for _ in range(8):
        z = rng.normal(0, 1, n)
        k = max(3, n // 200)
        kernel = np.ones(k) / k
        z = np.convolve(z, kernel, mode='same')
        p = np.exp(0.4 * z)
        out.append(_normalize_simplex(p))

    # Random Fourier
    for _ in range(5):
        coeffs = rng.normal(0, 1, min(n // 2, 50))
        phases = rng.uniform(0, 2 * np.pi, len(coeffs))
        z = np.zeros(n)
        for k_idx in range(len(coeffs)):
            z += coeffs[k_idx] * np.cos(2 * np.pi * (k_idx + 1) * t + phases[k_idx])
        p = np.exp(0.3 * z)
        out.append(_normalize_simplex(p))

    return out


def _anneal_descent(b, L2, L3, n, rng, time_limit,
                    T_start, T_end, eta_start, label=""):
    deadline = time.monotonic() + time_limit
    b = _normalize_simplex(b.copy())
    conv_b = _self_convolve(b, L2)
    current_score = 2.0 * n * float(np.max(conv_b))
    best_b = b.copy()
    best_score = current_score

    T = T_start
    eta = eta_start
    no_improve = 0
    it = 0
    n_iters_expected = 10000

    while time.monotonic() < deadline:
        frac = min(1.0, it / float(n_iters_expected))
        T = T_start * (T_end / T_start) ** frac
        T = max(T, 1e-7)

        conv_b = _self_convolve(b, L2)
        current_score = 2.0 * n * float(np.max(conv_b))

        if current_score < best_score - 1e-13:
            best_score = current_score
            best_b = b.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Compute gradient
        vmax = float(np.max(conv_b))
        z = (conv_b - vmax) / T
        z = np.clip(z, -50.0, 0.0)
        e = np.exp(z)
        s = float(e.sum())
        if s <= 0 or not np.isfinite(s):
            w = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        else:
            w = e / s
        cc = _cross_corr(w, b, L3)
        grad = 2.0 * cc
        grad = np.maximum(grad, 1e-30)

        # Line search with multiplicative update
        accepted = False
        for step in [eta, 0.5 * eta, 0.25 * eta, 0.1 * eta, 0.05 * eta]:
            if step < 1e-6:
                continue
            Log_b = np.log(b) - step * grad
            Log_b -= np.max(Log_b)
            b_try = np.exp(Log_b)
            b_try = _normalize_simplex(b_try)
            conv_try = _self_convolve(b_try, L2)
            sc_try = 2.0 * n * float(np.max(conv_try))
            if sc_try < current_score - 1e-14:
                b = b_try
                current_score = sc_try
                accepted = True
                break

        if accepted:
            eta = min(5.0, eta * 1.08)
        else:
            eta = max(0.005, eta * 0.7)

        # Anti-stalling
        if no_improve > 80:
            strength = min(0.25, 0.03 * (no_improve / 40.0))
            noise = rng.normal(0, 1, n)
            k = max(3, n // 300)
            kernel = np.ones(k) / k
            noise = np.convolve(noise, kernel, mode='same')
            b = _normalize_simplex(b + strength * noise * np.std(b))
            conv_b = _self_convolve(b, L2)
            current_score = 2.0 * n * float(np.max(conv_b))
            if current_score < best_score:
                best_score = current_score
                best_b = b.copy()
            no_improve = 0
            eta = eta_start
            T = max(T_start * 0.5, T)

        # Reset to best if very stuck
        if no_improve > 300:
            b = best_b.copy()
            current_score = best_score
            no_improve = 0
            T = T_start
            eta = eta_start

        it += 1
        if it % 500 == 0:
            print(f"  {label} it={it} score={best_score:.6f} T={T:.6f} eta={eta:.4f}")

    return best_b, best_score


def _dual_polish(b, L2, L3, n, time_limit):
    """Alternating primal-dual method for final polish."""
    deadline = time.monotonic() + time_limit
    b = _normalize_simplex(b.copy())
    best_b = b.copy()
    best_score = _score(b, L2)

    T = 5e-5
    eta_inner = 0.3

    while time.monotonic() < deadline:
        conv_b = _self_convolve(b, L2)
        current_score = 2.0 * n * float(np.max(conv_b))
        if current_score < best_score - 1e-14:
            best_score = current_score
            best_b = b.copy()

        # Dual: compute weights
        vmax = float(np.max(conv_b))
        z = (conv_b - vmax) / T
        z = np.clip(z, -50.0, 0.0)
        e = np.exp(z)
        s = float(e.sum())
        if s <= 0:
            break
        w = e / s

        # Primal update: several inner steps
        for _ in range(5):
            cc = _cross_corr(w, b, L3)
            grad = 2.0 * np.maximum(cc, 1e-30)
            Log_b = np.log(b) - eta_inner * grad
            Log_b -= np.max(Log_b)
            b_new = np.exp(Log_b)
            b_new = _normalize_simplex(b_new)
            sc = _score(b_new, L2)
            if sc <= current_score:
                b = b_new
                current_score = sc
                conv_b = _self_convolve(b, L2)
                # Update dual
                vmax = float(np.max(conv_b))
                z = (conv_b - vmax) / T
                z = np.clip(z, -50.0, 0.0)
                e = np.exp(z)
                s = float(e.sum())
                if s > 0:
                    w = e / s

        T = max(1e-7, T * 0.995)

    return best_b, best_score


def _try_other_n(n_parent, rng, time_limit, L2_parent, L3_parent):
    """Try a different sequence length."""
    n = n_parent
    candidates = [int(n * 0.5), int(n * 0.75), int(n * 1.25), int(n * 1.5)]
    candidates = [max(100, c) for c in candidates]

    best_b = None
    best_score = float('inf')

    for n_try in candidates:
        if time_limit <= 5:
            break
        t_per = min(time_limit / (len(candidates) + 1), 100.0)
        n_try = max(100, n_try)
        L2_t = _next_pow2(2 * n_try)
        L3_t = _next_pow2(3 * n_try)

        # Interpolate parent to new length
        try:
            base = height_sequence_1
            arr = np.asarray(base, dtype=np.float64).ravel()
            if arr.size == n_parent and np.any(arr > 0):
                x_old = np.linspace(0, 1, n_parent)
                x_new = np.linspace(0, 1, n_try)
                interp = np.interp(x_new, x_old, arr)
                b_init = _normalize_simplex(interp)
            else:
                b_init = np.ones(n_try) / n_try
        except (NameError, Exception):
            b_init = np.ones(n_try) / n_try

        b_try, sc = _anneal_descent(b_init, L2_t, L3_t, n_try, rng, t_per,
                                     T_start=0.03, T_end=0.0005, eta_start=0.4,
                                     label=f"n={n_try}")
        if sc < best_score:
            best_score = sc
            best_b = b_try
        time_limit -= (time.monotonic() - (time.monotonic() - time_limit))  # approximate
        time_limit = max(0, time_limit - t_per)

    if best_b is not None:
        # Map back to parent length if needed
        if len(best_b) != n_parent:
            x_new = np.linspace(0, 1, len(best_b))
            x_old = np.linspace(0, 1, n_parent)
            mapped = np.interp(x_old, x_new, best_b)
            best_b = _normalize_simplex(mapped)
            best_score = _score(best_b, L2_parent)

    return best_b, best_score
