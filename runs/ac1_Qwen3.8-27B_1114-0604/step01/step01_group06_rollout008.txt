_OPT_N = 0
_OPT_NFFT = 0
_OPT_TAU = 1e-6
_OPT_BOUND = 30.0
_OPT_BOUNDS = []

def _set_opt_state(n, tau=1e-6):
    global _OPT_N, _OPT_NFFT, _OPT_TAU, _OPT_BOUND, _OPT_BOUNDS
    _OPT_N = int(n)
    _OPT_NFFT = 1 << (2 * _OPT_N).bit_length()  # >= 2n for linear conv
    _OPT_TAU = float(tau)
    _OPT_BOUND = 30.0
    _OPT_BOUNDS = [(-_OPT_BOUND, _OPT_BOUND)] * _OPT_N

def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.array([], dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.full_like(z, 1.0 / z.size, dtype=np.float64)
    return e / s

def _fft_conv(a):
    n = a.size
    N = _OPT_NFFT
    L = 2 * n - 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L].real
    return np.maximum(c, 0.0)

def _fast_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    c = _fft_conv(a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return float(2 * a.size * m / (s * s))

def _obj_grad(z, tau=None):
    n = _OPT_N
    N = _OPT_NFFT
    if tau is None:
        tau = _OPT_TAU
    L = 2 * n - 1
    
    z = np.asarray(z, dtype=np.float64).ravel()
    z = np.clip(z, -_OPT_BOUND, _OPT_BOUND)
    
    a = _softmax(z)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L].real
    c = np.maximum(c, 0.0)
    
    m = float(np.max(c))
    
    diff = (c - m) / tau
    diff = np.clip(diff, -745.0, 0.0)
    e = np.exp(diff)
    se = float(np.sum(e))
    if se <= 0.0 or not np.isfinite(se):
        return 1e12, np.zeros(n, dtype=np.float64)
    
    p = e / se
    
    # Compute gradient exactly
    fp = np.fft.rfft(p, N)
    # t1[i] = sum_k p[k]*a[i-k] (linear conv)
    t1 = np.fft.irfft(fp * fa, N)[:n].real
    # t2[i] = sum_k p[k]*a[i+k] (reverse conv)
    p_rev = p[::-1]
    fp_rev = np.fft.rfft(p_rev, N)
    t2 = np.fft.irfft(fp_rev * fa, N)[n-1:2*n-1].real
    
    g_raw = t1 + t2
    
    s_g = float(np.dot(a, g_raw))
    grad = a * (g_raw - s_g)
    
    val = m + tau * np.log(se)
    scale = float(n)
    val *= scale

    pen = 1e-10

    grad = grad + pen * z
    
    if not np.isfinite(val):
        val = 1e12
        grad = np.zeros(n, dtype=np.float64)
    
    return float(val), grad
