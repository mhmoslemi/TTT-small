def _softmax(z):

    s = float(np.sum(e))

    return e / s


def _fft_conv(a):

    N = _OPT_NFFT
    L = 2 * n - 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L].real
    return np.maximum(c, 0.0)


def _fast_score(a):

def _obj_grad(z):

    """
    Compute gradient direction in log-space using top-k peaks of convolution.
    Returns the gradient vector g such that moving z -= eta*g decreases the max.
    """

    N = _OPT_NFFT

    if top_k == 1:
        k_max = int(np.argmax(c))
        h = np.zeros(n, dtype=np.float64)
        for i in range(n):
            j = k_max - i
            if 0 <= j < n:
                h[i] = a[j]
    else:
        # Weighted distribution over top-k peaks
        sorted_idx = np.argsort(c)[::-1][:top_k]
        weights = c[sorted_idx]
        w_sum = float(np.sum(weights))
        if w_sum <= 0:
            w_sum = 1.0
        weights = weights / w_sum

        h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real

            starts.append(f.copy())

    return starts


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimization:
    1. Load previous best or create structured starts
    2. L-BFGS with tau annealing (0.1 -> 0.01 -> 0.001 -> 1e-4 -> 1e-5 -> 1e-6)
    3. Aggressive gradient polish with multi-peak gradients
    4. Final L-BFGS restart and polish
    """
    budget_s = max(60.0, float(budget_s))
    t0 = time.monotonic()
    deadline = t0 + budget_s - 10.0

    n = 8192
    _set_opt_state(n, tau=0.1)

    rng = np.random.default_rng(int(seed))

    # === Phase 0: Get initial candidates ===
    z = None
    a0 = None

    # Try height_sequence_1
    try:
        g = globals().get("height_sequence_1")
        if g is not None:
            g = np.asarray(g, dtype=np.float64).ravel()
            if g.size == n:
                g = np.maximum(g, 1e-15)
                g = g / np.sum(g)
                z = _z_from_a(g)
                a0 = g.copy()
                print(f"Loaded height_sequence_1, n={n}, score={_fast_score(g):.9f}")
            elif g.size > 1:
                x_old = np.linspace(0, 1, g.size)
                x_new = np.linspace(0, 1, n)
                f = np.maximum(g, 1e-15)
                z_old = np.log(f) - np.mean(np.log(f))
                z_resampled = np.interp(x_new, x_old, z_old)
                z = np.clip(z_resampled - np.mean(z_resampled), -_OPT_BOUND, _OPT_BOUND)
                a0 = _softmax(z)
                print(f"Resampled height_sequence_1: {g.size} -> {n}, score={_fast_score(a0):.9f}")
    except Exception as e:
        print(f"Could not load height_sequence_1: {e}")

    if z is None:
        # Use best structured start
        starts = _make_structured_starts(n, rng)
        best_start_score = np.inf
        for s in starts:
            sc = _fast_score(s)
            if sc < best_start_score:
                best_start_score = sc
                a0 = s.copy()
        z = _z_from_a(a0)
        print(f"Using structured start, score={best_start_score:.9f}")

    best_a = a0.copy()
    best_score = _fast_score(a0)
    print(f"Initial best score = {best_score:.9f}")

    # === Phase A: L-BFGS with tau annealing ===
    tau_schedule = [0.1, 0.03, 0.01, 0.003, 0.001, 3e-4, 1e-4, 3e-5, 1e-5]
    for idx, tau in enumerate(tau_schedule):
        time_left = deadline - time.monotonic()
        if time_left < 120:
            break

        # Allocate time: earlier phases get more time
        time_alloc = min(time_left - 60, max(60, 400 - idx * 30))
        maxfun = int(min(1500, max(200, time_alloc * 3)))
        maxiter = int(min(500, max(80, time_alloc)))

        print(f"\n[L-BFGS] tau={tau}, maxfun={maxfun}, time_left={time_left:.0f}s")
        t_phase = time.monotonic()
        res = _run_lbfgs(z, tau=tau, maxfun=maxfun, maxiter=maxiter)
        phase_time = time.monotonic() - t_phase

        if res is not None and hasattr(res, 'x'):
            z_new = np.asarray(res.x, dtype=np.float64).ravel()
            a_new = _softmax(np.clip(z_new, -_OPT_BOUND, _OPT_BOUND))
            sc = _fast_score(a_new)
            if np.isfinite(sc):
                print(f"  tau={tau}: score={sc:.9f} (took {phase_time:.1f}s)")
                if sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = z_new

    # === Phase B: Aggressive gradient polish ===
    time_left = deadline - time.monotonic()
    print(f"\n[Polish] Starting aggressive polish, best={best_score:.9f}, time_left={time_left:.0f}s")
    if time_left > 30:
        polish_deadline = min(deadline, time.monotonic() + time_left - 20)
        a_polished = _polish_aggressive(best_a, polish_deadline)
        sc_polished = _fast_score(a_polished)
        if np.isfinite(sc_polished) and sc_polished < best_score:
            best_score = sc_polished
            best_a = a_polished.copy()
            print(f"  After polish: score={best_score:.9f}")
        else:
            print(f"  Polish didn't improve: {sc_polished:.9f} vs {best_score:.9f}")

    # === Phase C: More L-BFGS iterations from the best point ===
    time_left = deadline - time.monotonic()
    if time_left > 60:
        z = _z_from_a(best_a)
        if z is not None:
            # Fine-tune with very small tau
            tau_final = 1e-6
            maxfun = int(min(1000, max(200, (time_left - 30) * 2)))
            print(f"\n[Final L-BFGS] tau={tau_final}, maxfun={maxfun}")
            res = _run_lbfgs(z, tau_final, maxfun=maxfun, maxiter=300)
            if res is not None and hasattr(res, 'x'):
                z_new = np.asarray(res.x, dtype=np.float64).ravel()
                a_new = _softmax(np.clip(z_new, -_OPT_BOUND, _OPT_BOUND))
                sc = _fast_score(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_a = a_new.copy()
                    z = z_new
                    print(f"  Final L-BFGS: score={sc:.9f}")

            # One more polish round
            time_left = deadline - time.monotonic()
            if time_left > 30:
                a_p = _polish_aggressive(best_a, min(deadline, time.monotonic() + time_left - 10))
                sc_p = _fast_score(a_p)
                if np.isfinite(sc_p) and sc_p < best_score:
                    best_score = sc_p
                    best_a = a_p.copy()
                    print(f"  Final polish: score={sc_p:.9f}")

    # === Phase D: Try a different n if time permits ===
    time_left = deadline - time.monotonic()
    if time_left > 180:
        n2 = 16384
        print(f"\n[Try n={n2}] time_left={time_left:.0f}s")
        try:
            _set_opt_state(n2, tau=1e-4)
            global _OPT_N, _OPT_NFFT, _OPT_TAU
            _OPT_N = n2
            _OPT_NFFT = 1 << (3 * n2 - 2).bit_length()

            # Resample best_a to n2
            x_old = np.linspace(0, 1, n)
            x_new = np.linspace(0, 1, n2)
            z_old = np.log(np.maximum(best_a, 1e-15))
            z_old = z_old - np.mean(z_old)
            z2 = np.interp(x_new, x_old, z_old)
            z2 = np.clip(z2 - np.mean(z2), -_OPT_BOUND, _OPT_BOUND)
            a2 = _softmax(z2)
            sc2 = _fast_score(a2)
            print(f"  Resampled to n={n2}: score={sc2:.9f}")

            # Quick L-BFGS
            _OPT_TAU = 1e-4
            time_alloc2 = time_left - 60
            maxfun2 = int(min(600, max(100, time_alloc2 * 2)))
            res2 = _run_lbfgs(z2, tau=1e-4, maxfun=maxfun2, maxiter=200)
            if res2 is not None and hasattr(res2, 'x'):
                z2_new = np.asarray(res2.x, dtype=np.float64).ravel()
                a2_new = _softmax(np.clip(z2_new, -_OPT_BOUND, _OPT_BOUND))
                sc2_new = _fast_score(a2_new)
                print(f"  L-BFGS n={n2}: score={sc2_new:.9f}")
                if sc2_new < best_score:
                    _set_opt_state(n, tau=1e-4)  # restore
                    # Convert back to n-length for consistency? No, return n2-length
                    best_score = sc2_new
                    best_a = a2_new.copy()
                    n = n2
                    print(f"  New best with n={n2}!")
            # Polish
            time_left2 = deadline - time.monotonic()
            if time_left2 > 30:
                a2_p = _polish_aggressive(a2_new if res2 is not None else a2, min(deadline, time.monotonic() + time_left2 - 10))
                sc2_p = _fast_score(a2_p)
                print(f"  Polish n={n2}: score={sc2_p:.9f}")
                if sc2_p < best_score:
                    best_score = sc2_p
                    best_a = a2_p.copy()
                    n = n2
        except Exception as e:
            print(f"  n={n2} attempt failed: {e}")
        # Restore state
        _set_opt_state(8192, tau=1e-6)
        _OPT_N = n  # use actual n
        _OPT_NFFT = 1 << (3 * n - 2).bit_length()

    # === Final validation ===
    print(f"\n=== FINAL RESULTS ===")
    print(f"Best score (internal) = {best_score:.9f}")
    print(f"Sequence length = {n}")
    print(f"Total time = {time.monotonic() - t0:.1f}s")

    try:
        best_a = np.asarray(best_a, dtype=np.float64).ravel()
        best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
        best_a = np.clip(best_a, 0.0, 1000.0)
        s = float(np.sum(best_a))
        if s < 0.01 or not np.isfinite(s):
            best_a = np.ones(n, dtype=np.float64) / n
        else:
            best_a = best_a / s
        exact = evaluate_sequence(best_a.tolist())
        print(f"Official evaluator score = {float(exact):.9f}")
    except Exception as e:
        print(f"Evaluation error: {e}")

    # Defensive final normalization
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = float(np.sum(best_a))
    if s < 0.01 or not np.isfinite(s):
        best_a = np.ones(n, dtype=np.float64) / n
    else:
        best_a = best_a / s

    return [float(v) for v in best_a.tolist()]
