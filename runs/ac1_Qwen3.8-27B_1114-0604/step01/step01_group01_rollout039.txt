```
def _random_block_search(base, score, rng, deadline, verbose=False):
    p=_sanitize(base)
    if p is None: return None, inf
    n=p.size
    best=p.copy(); best_score=score
    if not np.isfinite(best_score): best_score=_evaluate_arr(best)
    cur=p.copy(); cur_score=best_score
    max_width=max(1, n//4)
    max_pow=max(1, int(np.log2(max_width))+1)
    stale=0; no_best=0
    sig=0.03
    while time.monotonic() < deadline:
       if n < 2: break
       r=rng.random()
       cand=None
       if r < 0.7:
          w = 1 << int(rng.integers(0, max_pow))
          if w >= n: w=1
          s=int(rng.integers(0, n-w+1))
          f=_random_factor(rng)
          cand=cur.copy(); cand[s:s+w]*=f
       elif r < 0.9:
          w1 = 1 << int(rng.integers(0, max_pow)); w2=1<<...
          if w1>=n: w1=1; if w2>=n: w2=1
          s1=int(rng.integers(0,n-w1+1)); s2=int(rng.integers(0,n-w2+1))
          f=_random_factor(rng)
          cand=cur.copy(); cand[s1:s1+w1]*=f; cand[s2:s2+w2]*=(1.0/f)
       else:
          cand=cur * np.exp(rng.normal(0.0, sig, n))
       cand=_sanitize(cand)
       if cand is None: continue
       sc=_evaluate_arr(cand)
       if _better(sc, cur_score):
          cur=cand; cur_score=sc; stale=0
          if _better(cur_score, best_score):
             best=cur.copy(); best_score=cur_score; no_best=0
       else:
          stale += 1; no_best += 1
       if stale > 200:
          # perturb or reset
          cand=_sanitize(cur * np.exp(rng.normal(0.0, sig, n)))
          if cand is not None:
             sc=_evaluate_arr(cand)
             if (not np.isfinite(best_score)) or sc <= best_score * 1.05 + 1e-9:
                cur=cand; cur_score=sc
             else:
                cur=best.copy(); cur_score=best_score
          sig = min(0.3, sig*1.2)
          stale=0
```

```python
"""
Black-box search for a non-negative coefficient sequence minimizing the
provided evaluation function.  The code starts from the available previous
construction if present, adds diverse probes, then refines the best candidate
with:
  * multiplicative block coordinate descent,
  * random/two-block multiplicative local search,
  * an active-set linear-programming update inspired by the previous
    state of the art,
  * short multistart restarts and optional upsampling.

All helper functions are top-level and use no closures or lambdas.
"""

import numpy as np
import time

try:
    from scipy import optimize as _scopt
except Exception:
    _scopt = None


def _cand_key(item):
    return item[0]


def _better(new_score, old_score):
    if not np.isfinite(new_score):
        return False
    if not np.isfinite(old_score):
        return True
    tol = 1e-10 * max(1.0, abs(old_score))
    return new_score < old_score - tol


def _sanitize(arr, normalize=True):
    if arr is None:
        return None
    try:
        a = np.asarray(arr, dtype=float)
    except Exception:
        return None
    if a.ndim != 1 or a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    if normalize:
        s = float(a.sum())
        if not np.isfinite(s) or s <= 0.0:
            return np.full(a.size, 1.0 / float(a.size), dtype=float)
        a = a / s
    return a


def _to_list(a):
    a = np.asarray(a, dtype=float)
    return a.tolist()


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _full_conv(a, b):
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    la = a.size
    lb = b.size
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 3500000:
        out = np.convolve(a, b)
    else:
        L = _next_pow2(nout)
        out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
        out = out[:nout]
    out = np.asarray(out, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    return out


def _internal_score(a):
    a = _sanitize(a)
    if a is None:
        return float("inf")
    n = int(a.size)
    c = _full_conv(a, a)
    if c.size == 0:
        return float("inf")
    return float(2.0 * n * float(np.max(c)))


def _evaluate_arr(a):
    a = _sanitize(a)
    if a is None:
        return float("inf")
    lst = a.astype(float).tolist()
    try:
        val = float(evaluate_sequence(lst))
    except Exception:
        val = _internal_score(lst)
    if not np.isfinite(val):
        return float("inf")
    return val


def _resample(a, m):
    a = _sanitize(a)
    m = int(max(1, m))
    if a is None:
        return None
    n = int(a.size)
    if n == m:
        return a.copy()
    if n == 1:
        return np.full(m, 1.0 / float(m), dtype=float)
    x_old = (np.arange(n, dtype=float) + 0.5) / float(n)
    x_new = (np.arange(m, dtype=float) + 0.5) / float(m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0)
    out = out + 1e-15
    return _sanitize(out)


def _smooth_array(a, sigma):
    a = _sanitize(a)
    if a is None:
        return None
    sigma = float(sigma)
    if sigma <= 0.0 or a.size < 5:
        return a
    rad = int(max(1, np.ceil(3.0 * sigma)))
    k = np.arange(-rad, rad + 1, dtype=float)
    w = np.exp(-0.5 * (k / sigma) ** 2)
    w = w / w.sum()
    out = np.convolve(a, w, mode="same")
    return _sanitize(out)


def _initial_sequence(m, rng, mode):
    m = int(max(2, m))
    t = (np.arange(m, dtype=float) + 0.5) / float(m)
    p = np.ones(m, dtype=float)

    if mode == 0:
        z = np.ones(m, dtype=float)
        for k in range(1, 13):
            amp = float(rng.normal(0.0, 1.0 / (k + 1)))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            z = z + amp * np.cos(2.0 * np.pi * k * t + ph)
        p = np.log1p(np.exp(np.clip(z, -50.0, 50.0)))

    elif mode == 1:
        aa = float(rng.uniform(0.05, 3.0))
        bb = float(rng.uniform(0.05, 3.0))
        tt = np.clip(t, 1e-8, 1.0 - 1e-8)
        p = np.power(tt, aa) * np.power(1.0 - tt, bb)

    elif mode == 2:
        K = int(rng.integers(1, 4))
        p = np.zeros(m, dtype=float)
        centers = np.sort(rng.uniform(0.0, 1.0, K))
        sigmas = rng.uniform(0.02, 0.35, K)
        weights = rng.uniform(0.5, 2.0, K)
        for ii in range(K):
            c = float(centers[ii])
            s = max(1e-6, float(sigmas[ii]))
            w = float(weights[ii])
            p = p + w * np.exp(-((t - c) / s) ** 2)

    elif mode == 3:
        K = int(rng.integers(3, 9))
        knots = np.sort(rng.uniform(0.0, 1.0, K))
        heights = rng.lognormal(0.0, 1.0, K)
        p = np.interp(t, knots, heights)

    elif mode == 4:
        K = int(rng.integers(3, 16))
        edges = np.sort(rng.uniform(0.0, 1.0, K - 1))
        blocks = [0.0]
        blocks.extend(list(edges))
        blocks.append(1.0)
        heights = rng.lognormal(0.0, 1.0, K)
        p = np.zeros(m, dtype=float)
        for ii in range(K):
            s = max(0, int(round(blocks[ii] * m)))
            e = max(s + 1, int(round(blocks[ii + 1] * m)))
            e = min(m, e)
            p[s:e] = float(heights[ii])

    else:
        c = float(rng.uniform(0.2, 0.8))
        kk = float(rng.uniform(5.0, 30.0))
        p = 1.0 / (1.0 + np.exp(-kk * (t - c)))
        p = p * np.exp(-((t - c) / 0.25) ** 2)

    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    p = p + 1e-12
    return _sanitize(p)


def _add_candidate(cands, arr, keep=12):
    a = _sanitize(arr)
    if a is None:
        return
    sc = _evaluate_arr(a)
    cands.append((sc, a.copy()))
    cands.sort(key=_cand_key)
    if len(cands) > keep:
        del cands[keep:]


def _detect_mode(score, arr):
    a = _sanitize(arr)
    if a is None or a.size < 2:
        return "full"
    n = int(a.size)
    c = _full_conv(a, a)
    if c.size != 2 * n - 1:
        return "full"
    mid = n - 1
    M_full = float(c[mid]) if c.size > mid else 0.0
    side_idx = np.concatenate((np.arange(0, mid), np.arange(mid + 1, c.size)))
    M_side = float(np.max(c[side_idx])) if side_idx.size else 0.0
    if not np.isfinite(score):
        return "side"
    scoreM = float(score) / (2.0 * float(n))
    if M_full > 0.0 and np.isfinite(scoreM) and scoreM < 0.98 * M_full:
        return "side"
    return "full"


def _random_factor(rng):
    r = float(rng.random())
    if r < 0.30:
        return 0.8
    if r < 0.60:
        return 1.25
    if r < 0.75:
        return 0.65
    if r < 0.90:
        return 1.5
    return 1.15 if float(rng.random()) < 0.5 else 0.9


def _block_descent(base, score, rng, deadline,
                   B_list=(4, 8, 16, 32, 64, 128),
                   small_factors=(0.8, 1.2),
                   wide_up=(1.5, 2.0),
                   wide_down=(0.65, 0.5)):
    p = _sanitize(base)
    if p is None:
        return None, float("inf")

    n = int(p.size)
    best = p.copy()
    best_score = float(score) if np.isfinite(score) else _evaluate_arr(best)
    cur = p.copy()
    cur_score = best_score

    for B in B_list:
        if n < 2:
            break
        B = min(int(B), n)
        if B < 2:
            B = 2

        bounds = [0] * (B + 1)
        for i in range(B + 1):
            bounds[i] = int(round(i * n / float(B)))
        bounds[0] = 0
        bounds[B] = n
        for i in range(1, B):
            if bounds[i] <= bounds[i - 1]:
                bounds[i] = bounds[i - 1] + 1
        if bounds[B - 1] >= n:
            bounds[B - 1] = n - 1

        while time.monotonic() < deadline:
            improved = False
            order = np.arange(B)
            rng.shuffle(order)

            for bi in order:
                if time.monotonic() >= deadline:
                    break
                s = int(bounds[bi])
                e = int(bounds[bi + 1])
                if e <= s:
                    continue

                best_cand = None
                best_sc = cur_score
                best_f = 1.0

                for f in small_factors:
                    cand = cur.copy()
                    cand[s:e] *= float(f)
                    cand = np.maximum(cand, 0.0)
                    ss = float(cand.sum())
                    if ss <= 0.0 or not np.isfinite(ss):
                        continue
                    cand = cand / ss
                    sc = _evaluate_arr(cand)
                    if _better(sc, best_sc):
                        best_sc = sc
                        best_cand = cand
                        best_f = float(f)

                if best_cand is not None:
                    wide = wide_up if best_f > 1.0 else wide_down
                    for f in wide:
                        cand = best_cand.copy()
                        cand[s:e] *= (float(f) / best_f)
                        cand = np.maximum(cand, 0.0)
                        ss = float(cand.sum())
                        if ss <= 0.0 or not np.isfinite(ss):
                            continue
                        cand = cand / ss
                        sc = _evaluate_arr(cand)
                        if _better(sc, best_sc):
                            best_sc = sc
                            best_cand = cand
                            best_f = float(f)

                    cur = best_cand
                    cur_score = best_sc
                    improved = True
                    if _better(cur_score, best_score):
                        best = cur.copy()
                        best_score = cur_score

            if time.monotonic() >= deadline:
                break

            if not improved:
                cand = _sanitize(cur * np.exp(rng.normal(0.0, 0.03, n)))
                if cand is not None:
                    sc = _evaluate_arr(cand)
                    if (not np.isfinite(best_score)) or sc <= best_score * 1.05 + 1e-9:
                        cur = cand
                        cur_score = sc
                    else:
                        cur = best.copy()
                        cur_score = best_score
                break

    return best, best_score


def _random_block_search(base, score, rng, deadline):
    p = _sanitize(base)
    if p is None:
        return None, float("inf")

    n = int(p.size)
    best = p.copy()
    best_score = float(score) if np.isfinite(score) else _evaluate_arr(best)
    cur = p.copy()
    cur_score = best_score

    if n < 2:
        return best, best_score

    max_width = max(1, n // 4)
    max_pow = max(1, int(np.log2(max_width)) + 1)
    stale = 0
    no_best = 0
    sig = 0.03

    while time.monotonic() < deadline:
        r = float(rng.random())
        cand = None

        if r < 0.70:
            w = 1 << int(rng.integers(0, max_pow))
            if w >= n:
                w = 1
            s = int(rng.integers(0, n - w + 1))
            f = _random_factor(rng)
            cand = cur.copy()
            cand[s:s + w] *= f

        elif r < 0.90:
            w1 = 1 << int(rng.integers(0, max_pow))
            w2 = 1 << int(rng.integers(0, max_pow))
            if w1 >= n:
                w1 = 1
            if w2 >= n:
                w2 = 1
            s1 = int(rng.integers(0, n - w1 + 1))
            s2 = int(rng.integers(0, n - w2 + 1))
            f = _random_factor(rng)
            cand = cur.copy()
            cand[s1:s1 + w1] *= f
            cand[s2:s2 + w2] *= (1.0 / f)

        else:
            cand = cur * np.exp(rng.normal(0.0, sig, n))

        cand = _sanitize(cand)
        if cand is None:
            continue

        sc = _evaluate_arr(cand)
        if _better(sc, cur_score):
            cur = cand
            cur_score = sc
            stale = 0
            if _better(cur_score, best_score):
                best = cur.copy()
                best_score = cur_score
                no_best = 0
        else:
            stale += 1
            no_best += 1

        if stale > 220:
            cand = _sanitize(cur * np.exp(rng.normal(0.0, sig, n)))
            if cand is not None:
                sc = _evaluate_arr(cand)
                if (not np.isfinite(best_score)) or sc <= best_score * 1.05 + 1e-9:
                    cur = cand
                    cur_score = sc
                else:
                    cur = best.copy()
                    cur_score = best_score
            sig = min(0.30, sig * 1.2)
            stale = 0

        if no_best > 4000:
            break

    return best, best_score


def _solve_lp(f, indices, M, time_limit):
    if _scopt is None:
        return None
    f = _sanitize(f, normalize=False)
    if f is None:
        return None
    n = int(f.size)
    K = int(len(indices))
    if K <= 0 or n <= 0:
        return None

    time_limit = int(max(5, min(3600, time_limit)))

    try:
        A = np.zeros((K, n), dtype=np.float64)
        for r, k in enumerate(indices):
            k = int(k)
            j0 = max(0, k - (n - 1))
            j1 = min(n - 1, k)
            if j1 >= j0:
                f_start = k - j0
                f_end = f_start + (j1 - j0 + 1)
                A[r, j0:j1 + 1] = f[f_start:f_end]

        if not np.isfinite(M) or M <= 0.0:
            M = 1e-12
        M = float(M) * (1.0 + 1e-9) + 1e-15

        A_append = np.ones((1, n), dtype=np.float64)
        A_all = np.vstack((A, A_append))
        b = np.empty(K + 1, dtype=np.float64)
        b[:K] = M
        b[K] = 20.0

        c = -np.ones(n, dtype=np.float64)
        bounds = [(0.0, None)] * n

        sol = _scopt.linprog(
            c,
            A_ub=A_all,
            b_ub=b,
            bounds=bounds,
            method="highs",
            options={"time_limit": time_limit, "disp": False},
        )
        if getattr(sol, "status", 1) == 0 and getattr(sol, "x", None) is not None:
            g = np.asarray(sol.x, dtype=float)
            g = np.where(np.isfinite(g), g, 0.0)
            g = np.clip(g, 0.0, 1e100)
            gs = float(g.sum())
            if gs <= 0.0 or not np.isfinite(gs):
                return None
            return g / gs
    except Exception:
        return None

    return None


def _lp_refine(base, score, deadline, mode="side"):
    f = _sanitize(base)
    if f is None:
        return None, float("inf")

    n = int(f.size)
    if n < 2 or _scopt is None or n > 20000:
        return f.copy(), float(score)

    f = np.maximum(f, 1e-15 / float(n))
    f = f / f.sum()

    best = f.copy()
    best_score = _evaluate_arr(best)
    cur = best.copy()
    cur_score = best_score

    start = time.monotonic()
    lp_budget = min(max(0.0, float(deadline) - start) - 90.0, 780.0)
    if lp_budget < 60.0:
        return best, best_score

    it = 0

    while time.monotonic() < deadline and (time.monotonic() - start) < lp_budget:
        it += 1
        c = _full_conv(cur, cur)
        if c.size != 2 * n - 1:
            break
        mid = n - 1

        use_full = (mode == "full")
        if use_full:
            all_idx = np.arange(c.size)
            all_vals = c
        else:
            all_idx = np.concatenate((np.arange(0, mid), np.arange(mid + 1, c.size)))
            if all_idx.size == 0:
                break
            all_vals = c[all_idx]

        maxv = float(np.max(all_vals)) if all_vals.size else 0.0
        if not np.isfinite(maxv) or maxv <= 1e-30:
            break

        max_entries = 3000000
        K_cap = min(1024, max(96, max_entries // max(1, n)))
        K = min(K_cap, int(all_idx.size))
        if K <= 0:
            break

        order = np.argsort(all_vals)
        sel_idx = all_idx[order[-K:]]
        M = max(float(np.max(c[sel_idx])), float(cur_score) / (2.0 * float(n)))

        rem = float(deadline) - time.monotonic() - 60.0
        if rem < 45.0:
            break
        solve_time = min(180.0, max(20.0, rem * 0.6))

        g = _solve_lp(cur, sel_idx, M, solve_time)
        if g is None:
            M2 = M * 1.15
            g = _solve_lp(cur, sel_idx, M2, min(90.0, solve_time * 0.5))
        if g is None:
            break

        best_cand = None
        best_sc = cur_score

        for t in (1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05):
            cand = (1.0 - t) * cur + t * g
            cand = _sanitize(cand)
            if cand is None:
                continue
            sc = _evaluate_arr(cand)
            if _better(sc, best_sc):
                best_cand = cand
                best_sc = sc

        if best_cand is None:
            break

        cur = best_cand
        cur_score = best_sc
        if _better(cur_score, best_score):
            best = cur.copy()
            best_score = cur_score
            print("LP phase improve:", best_score)

    return best, best_score


def _multistart_refine(best, best_score, rng, deadline):
    if best is None:
        return None, float("inf")
    b = _sanitize(best)
    if b is None:
        return None, float("inf")
    n = int(b.size)
    if n < 2:
        return b, float(best_score)

    for sig in (0.03, 0.08, 0.15, 0.30):
        rem = float(deadline) - time.monotonic() - 30.0
        if rem < 70.0:
            break
        cand = _sanitize(b * np.exp(rng.normal(0.0, sig, n)))
        if cand is None:
            continue
        sc = _evaluate_arr(cand)
        sub = min(65.0, rem * 0.45)
        p, s = _block_descent(
            cand,
            sc,
            rng,
            time.monotonic() + max(10.0, sub),
            B_list=(8, 16, 32, 64),
            small_factors=(0.8, 1.2),
            wide_up=(1.5, 2.0),
            wide_down=(0.65, 0.5),
        )
        if p is not None and _better(s, best_score):
            best = p
            best_score = s

    return best, best_score


def _try_larger_length(best, best_score, rng, deadline):
    if best is None:
        return None, float("inf")
    b = _sanitize(best)
    if b is None:
        return None, float("inf")
    n = int(b.size)

    for n2 in (min(2 * n, 8192), min(4 * n, 16384)):
        if n2 <= n:
            continue
        rem = float(deadline) - time.monotonic() - 60.0
        if rem < 140.0:
            break
        arr = _resample(b, n2)
        if arr is None:
            continue
        sc = _evaluate_arr(arr)
        sub = min(120.0, rem * 0.55)
        p, s = _block_descent(
            arr,
            sc,
            rng,
            time.monotonic() + max(20.0, sub),
            B_list=(8, 16, 32, 64),
            small_factors=(0.8, 1.2),
            wide_up=(1.5, 2.0),
            wide_down=(0.65, 0.5),
        )
        if p is not None and _better(s, best_score):
            best = p
            best_score = s

    return best, best_score


def propose_candidate(seed=12345, budget_s=2400.0, **kwargs):
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget < 120.0:
        budget = 2400.0

    start = time.monotonic()
    deadline = start + max(120.0, budget - 60.0)

    try:
        iseed = int(seed)
    except Exception:
        iseed = 12345
    rng = np.random.default_rng((iseed % 1000000007) + 1)

    cands = []

    parent_arr = None
    try:
        raw_parent = height_sequence_1
        parent_arr = _sanitize(raw_parent)
    except Exception:
        parent_arr = None

    if parent_arr is not None:
        if parent_arr.size > 20000:
            _add_candidate(cands, _resample(parent_arr, 16384), keep=16)
        else:
            _add_candidate(cands, parent_arr, keep=16)

    if parent_arr is not None and parent_arr.size >= 16:
        n0 = int(parent_arr.size)
        if n0 > 12288:
            n0 = 12288
    else:
        n0 = 3072

    n0 = int(min(max(256, n0), 12288))

    _add_candidate(cands, np.ones(n0, dtype=float), keep=16)

    _add_candidate(cands, _initial_sequence(n0, rng, 1), keep=16)
    _add_candidate(cands, _initial_sequence(n0, rng, 2), keep=16)
    _add_candidate(cands, _initial_sequence(n0, rng, 4), keep=16)
    _add_candidate(cands, _initial_sequence(n0, rng, 0), keep=16)

    spike1 = np.zeros(n0, dtype=float)
    spike1[0] = 1.0
    _add_candidate(cands, spike1, keep=16)

    spike2 = np.zeros(n0, dtype=float)
    spike2[n0 // 2] = 1.0
    _add_candidate(cands, spike2, keep=16)

    half = np.ones(n0, dtype=float)
    half[n0 // 2:] = 2.0
    _add_candidate(cands, half, keep=16)

    probe_lengths = set()
    for m in (n0, 3072, n0 // 2, n0 * 2):
        if 256 <= m <= 16384:
            probe_lengths.add(int(m))

    for m in sorted(probe_lengths):
        if time.monotonic() > deadline - 90.0:
            break
        if parent_arr is not None:
            _add_candidate(cands, _resample(parent_arr, m), keep=16)
        _add_candidate(cands, np.ones(m, dtype=float), keep=16)

    if not cands:
        cands.append((float("inf"), np.ones(n0, dtype=float)))

    cands.sort(key=_cand_key)
    best_arr = cands[0][1].copy()
    best_score = cands[0][0]

    if not np.isfinite(best_score):
        best_arr = np.ones(n0, dtype=float)
        best_score = _evaluate_arr(best_arr)

    print("initial best:", best_score, "length:", int(best_arr.size))

    rem = float(deadline) - time.monotonic()
    if rem < 80.0:
        return _to_list(_sanitize(best_arr))

    t1 = min(0.28 * rem, 720.0)
    best_arr, best_score = _block_descent(
        best_arr,
        best_score,
        rng,
        time.monotonic() + max(30.0, t1),
        B_list=(4, 8, 16, 32, 64),
        small_factors=(0.8, 1.2),
        wide_up=(1.5, 2.0),
        wide_down=(0.65, 0.5),
    )
    print("after block descent:", best_score, "length:", int(best_arr.size))

    rem = float(deadline) - time.monotonic()
    if rem > 360.0:
        mode = "side" if best_score < 1.95 else _detect_mode(best_score, best_arr)
        t2 = min(0.36 * rem, 820.0)
        best_arr, best_score = _lp_refine(
            best_arr,
            best_score,
            time.monotonic() + max(60.0, t2),
            mode=mode,
        )
        print("after LP phase:", best_score, "length:", int(best_arr.size))

    rem = float(deadline) - time.monotonic()
    if rem > 120.0:
        t3 = min(0.22 * rem, 520.0)
        best_arr, best_score = _block_descent(
            best_arr,
            best_score,
            rng,
            time.monotonic() + max(30.0, t3),
            B_list=(16, 32, 64, 128),
            small_factors=(0.85, 1.15),
            wide_up=(1.3, 1.6),
            wide_down=(0.75, 0.6),
        )
        print("after second block descent:", best_score, "length:", int(best_arr.size))

    rem = float(deadline) - time.monotonic()
    if rem > 160.0:
        t4 = min(0.18 * rem, 360.0)
        best_arr, best_score = _random_block_search(
            best_arr,
            best_score,
            rng,
            time.monotonic() + max(20.0, t4),
        )
        print("after random block search:", best_score, "length:", int(best_arr.size))

    rem = float(deadline) - time.monotonic()
    if rem > 220.0:
        best_arr, best_score = _multistart_refine(best_arr, best_score, rng, deadline)
        print("after multistart:", best_score, "length:", int(best_arr.size))

    rem = float(deadline) - time.monotonic()
    if rem > 220.0 and best_arr is not None and best_arr.size < 8192:
        best_arr, best_score = _try_larger_length(best_arr, best_score, rng, deadline)
        print("after length try:", best_score, "length:", int(best_arr.size))

    rem = float(deadline) - time.monotonic()
    if rem > 90.0:
        best_arr, best_score = _block_descent(
            best_arr,
            best_score,
            rng,
            min(deadline - 10.0, time.monotonic() + min(160.0, rem * 0.6)),
            B_list=(64, 128, 256),
            small_factors=(0.85, 1.15),
            wide_up=(1.25, 1.5),
            wide_down=(0.8, 0.65),
        )
        print("after final block polish:", best_score, "length:", int(best_arr.size))

    final_candidates = [best_arr]
    for item in cands[:8]:
        final_candidates.append(item[1])
    if parent_arr is not None:
        final_candidates.append(parent_arr)
    final_candidates.append(np.ones(n0, dtype=float))

    selected = best_arr
    selected_score = best_score

    for arr in final_candidates:
        if time.monotonic() >= deadline - 5.0:
            break
        a = _sanitize(arr)
        if a is None:
            continue
        sc = _evaluate_arr(a)
        if _better(sc, selected_score):
            selected = a
            selected_score = sc

    out = _sanitize(selected)
    if out is None:
        out = np.ones(max(32, n0), dtype=float)

    out = np.asarray(out, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s <= 0.0 or not np.isfinite(s):
        out = np.ones(max(32, n0), dtype=float)
    else:
        out = out / s

    print("final chosen score:", selected_score, "final length:", int(out.size))
    return [float(x) for x in out]
```
