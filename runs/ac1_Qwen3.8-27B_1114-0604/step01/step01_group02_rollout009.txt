from scipy.optimize import linprog
import numpy as np

def lp_step(f, n):
    """Solve: min t s.t. (f*g)[k] <= t for all k, sum(g) = 1, g >= 0."""
    # Variables: [g_0, g_1, ..., g_{n-1}, t]
    # Total: n+1 variables
    # Objective: minimize t -> c = [0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequalities: (f*g)[k] - t <= 0 for k = 0, ..., 2n-2
    # (f*g)[k] = sum_j f[j] g[k-j] (with g[m] = 0 for m < 0 or m >= n)
    # So (f*g)[k] - t <= 0
    A_ineq = np.zeros((2*n-1, n+1))
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(k, n-1)+1):
            A_ineq[k, j] = f[j]
        A_ineq[k, n] = -1.0
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    # Bounds: g_j >= 0, t >= 0 (or t >= min possible)
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ineq, b_ub=np.zeros(2*n-1), A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')

Uses alternating LP steps (finding best cross-convolution direction)
combined with gradient-based refinement and periodic restarts.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time


def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_conv_matrix(a, n):
    """Build sparse matrix A such that A @ g = conv(a, g)."""
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[j])
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp(a, n, M_val):
    """Solve: max sum(g) s.t. conv(a,g)[k] <= M_val for all k, g >= 0."""
    A_mat = cross_conv_matrix(a, n)
    c = -np.ones(n)  # minimize -sum(g)
    res = linprog(
        c,
        A_ub=A_mat,
        b_ub=np.full(2 * n - 1, M_val),
        bounds=[(0, None)] * n,
        method='highs'
    )
    if res.status == 0:
        return res.x
    return None


def score_seq(a, n):
    c = fast_self_conv(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def topk_grad(a, c, k=8):
    """Gradient of smooth approximation to max of convolution."""
    n = len(a)
    S = a.sum()
    k_act = min(k, len(c))
    idx = np.argpartition(c, -k_act)[-k_act:]
    vals = c[idx]
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]
    top = vals[0]
    temp = 0.1 * top
    wts = np.exp((vals - top) / temp)
    wts = wts / wts.sum()
    
    grad_c = np.zeros(n)
    for i in range(idx.size):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-15:
            continue
        j_s = max(0, ki - n + 1)
        j_e = min(ki, n - 1)
        grad_c[j_s:j_e+1] += wi * 2.0 * a[j_s:j_e+1][::-1]
    
    M_w = float(np.dot(vals, wts))
    S2 = S * S
    grad = (2.0 * n) * (grad_c / S2 - 2.0 * M_w / (S2 * S))
    return grad

    return np.fft.irfft(A * A, M_fft)[:M]


def precompute_conv_structure(n):
    """Precompute (row, col) indices for the convolution matrix A where A@g = conv(a,g)."""
    rows = []
    cols = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
    return np.array(rows, dtype=np.int32), np.array(cols, dtype=np.int32)


def build_conv_matrix(a, rows, cols, n):
    """Build sparse matrix A such that A @ g = conv(a, g)."""
    vals = a[cols]
    return csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))


def solve_lp_minpeak(a, rows, cols, n):
    """
    Solve: min t s.t. conv(a,g)[k] - t <= 0 for all k, sum(g)=1, g>=0.
    Returns (g, t) or (None, None) on failure.
    """
    A_mat = build_conv_matrix(a, rows, cols, n)
    # Variables: [g_0, ..., g_{n-1}, t]
    # Objective: min t
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Inequalities: A_mat @ g - t <= 0
    # Build A_ub: shape (2n-1, n+1)
    A_ub_data = []
    A_ub_rows = []
    A_ub_cols = []
    for k in range(2 * n - 1):
        # Get non-zero entries in row k of A_mat
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A_ub_data.append(a[j])
            A_ub_rows.append(k)
            A_ub_cols.append(j)
        # -t term
        A_ub_data.append(-1.0)
        A_ub_rows.append(k)
        A_ub_cols.append(n)
    
    A_ub = csr_matrix((A_ub_data, (A_ub_rows, A_ub_cols)), shape=(2 * n - 1, n + 1))
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(g) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x[:n], float(res.x[n])
    return None, None


def score_seq(a, n):
    """Compute the objective value."""
    c = fast_self_conv(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def topk_gradient(a, c, k=8, temp_scale=0.1):
    """Smooth gradient using top-k weighted peaks."""
    n = len(a)
    S = a.sum()
    k_act = min(k, len(c))
    if k_act <= 0:
        return np.zeros(n)
    
    idx = np.argpartition(c, -k_act)[-k_act:]
    vals = c[idx]
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]
    top = float(vals[0])
    
    if top > 1e-12:
        temp = temp_scale * top
        wts = np.exp((vals - top) / temp)
        wts = wts / wts.sum()
    else:
        wts = np.ones(k_act) / k_act
    
    grad_c = np.zeros(n)
    for i in range(idx.size):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-15:
            continue
        j_s = max(0, ki - n + 1)
        j_e = min(ki, n - 1)
        if j_s <= j_e:
            grad_c[j_s:j_e + 1] += wi * 2.0 * a[j_s:j_e + 1][::-1]
    
    M_w = float(np.dot(vals, wts))
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_c / S2 - 2.0 * M_w / S3)
    return grad


def adam_step(a, grad, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    """Adam optimizer step (in-place on a, m, v)."""
    t1 = b1 ** t
    t2 = b2 ** t
    m[:] = b1 * m + (1.0 - b1) * grad
    v[:] = b2 * v + (1.0 - b2) * grad * grad
    mh = m / (1.0 - t1)
    vh = v / (1.0 - t2)
    a -= lr * mh / (np.sqrt(vh) + eps)


def normalize_sum(a):
    """Normalize to sum 1, ensuring non-negativity."""
    a = np.clip(a, 0.0, None)
    S = a.sum()
    if S < 1e-10:
        a[:] = 1.0 / a.size
    else:
        a /= S
    return a


def init_from_parent(parent, n):
    """Create initial sequence from parent."""
    if parent is None:
        return np.ones(n)
    if parent.size == n:
        a = parent.copy()
    elif parent.size > n:
        a = parent[:n].copy()
    else:
        # Interpolate to length n
        a = np.interp(np.linspace(0, parent.size - 1, n),
                      np.arange(parent.size), parent)
    a = np.clip(a, 0.0, None)
    return a


def init_cosine_window(n):
    """Raised cosine window initialization."""
    x = np.linspace(0, 1, n)
    a = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    a = np.maximum(a, 0.01)
    return a


def init_bell(n):
    """Gaussian-like bell shape."""
    x = np.linspace(0, 1, n)
    a = np.exp(-4.0 * (x - 0.5) ** 2)
    return a


def init_bimodal(n):
    """Two-peaked initialization."""
    x = np.linspace(0, 1, n)
    a = np.exp(-16.0 * (x - 0.3) ** 2) + np.exp(-16.0 * (x - 0.7) ** 2)
    return a


def init_symmetric_random(n, rng):
    """Random symmetric initialization."""
    half = n // 2
    a_half = 0.3 + 0.7 * rng.random(half)
    a = np.zeros(n)
    a[:half] = a_half
    a[half:] = a_half[::-1] if n % 2 == 0 else np.concatenate([a_half[::-1], [a_half[-1]]])
    return a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    
    Strategy:
    1. LP-based iteration: minimize cross-convolution peak to find search direction.
    2. Gradient-based refinement with smooth max approximation (Adam).
    3. Multiple initializations and periodic restarts.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    deadline = start + budget_s - 12.0

    # Determine n from parent
    n_default = 2048
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size > 100:
            n_default = p.size
            parent = p.copy()
    except Exception:
        pass

    n = int(min(max(512, n_default), 4096))
    print("Using n = %d, parent available: %s" % (n, parent is not None))

    # Precompute convolution matrix structure
    rows, cols = precompute_conv_structure(n)
    print("Conv matrix structure: %d non-zeros" % len(rows))

    # Build list of initializations
    inits = []
    if parent is not None:
        inits.append(("parent", init_from_parent(parent, n)))
    inits.append(("uniform", np.ones(n)))
    inits.append(("cosine", init_cosine_window(n)))
    inits.append(("bell", init_bell(n)))
    inits.append(("bimodal", init_bimodal(n)))
    inits.append(("symrand", init_symmetric_random(n, rng)))
    # A few more diverse inits
    for _ in range(3):
        a = np.clip(0.2 + rng.random(n), 0.01, None)
        inits.append(("rand_%d" % _, a))

    best_score = np.inf
    best_a = None

    # Phase 1: Try each initialization with a short LP+gradient run
    phase1_end = start + 0.35 * (budget_s - 12.0)
    time_per_init = max(2.0, (phase1_end - start) / len(inits))
    print("Phase 1: %d initializations, ~%.1fs each" % (len(inits), time_per_init))

    for tag, a0 in inits:
        if time.monotonic() > phase1_end:
            break
        a = normalize_sum(a0.copy())
        sc = score_seq(a, n)
        print("  [%s] initial score = %.6f" % (tag, sc))

        # Short optimization: mix of LP and gradient
        t_end = min(phase1_end, time.monotonic() + time_per_init)
        m = np.zeros(n)
        v = np.zeros(n)
        t_count = 0
        local_best = sc
        local_best_a = a.copy()
        no_imp = 0

        while time.monotonic() < t_end:
            # Alternate between LP and gradient
            if rng.random() < 0.6:
                # LP step
                g, t_val = solve_lp_minpeak(a, rows, cols, n)
                if g is not None:
                    g = np.clip(g, 0, None)
                    g = normalize_sum(g)
                    # Small step towards g
                    tau = 0.3 + 0.3 * rng.random()
                    a = (1 - tau) * a + tau * g
                    a = normalize_sum(a)
                else:
                    # Fallback to gradient
                    c = fast_self_conv(a)
                    grad = topk_gradient(a, c, k=8)
                    t_count += 1
                    adam_step(a, grad, m, v, t_count, 0.1)
                    a = normalize_sum(a)
            else:
                # Gradient step
                c = fast_self_conv(a)
                grad = topk_gradient(a, c, k=8, temp_scale=0.08)
                t_count += 1
                adam_step(a, grad, m, v, t_count, 0.05)
                a = normalize_sum(a)

            sc = score_seq(a, n)
            if sc < local_best - 1e-9:
                local_best = sc
                local_best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1
                if no_imp > 50:
                    break

        if local_best < best_score:
            best_score = local_best
            best_a = local_best_a.copy()
            print("  [%s] best so far = %.6f" % (tag, best_score))

    print("Phase 1 complete. Best = %.6f" % best_score)

    # Phase 2: Extended LP iteration on best
    phase2_end = start + 0.75 * (budget_s - 12.0)
    a = best_a.copy()
    no_imp = 0
    lp_fail_count = 0

    print("Phase 2: Extended LP iteration, budget until t=%.0f" % (phase2_end - start))

    while time.monotonic() < phase2_end:
        # LP step
        g, t_val = solve_lp_minpeak(a, rows, cols, n)

        if g is not None and g.sum() > 1e-6:
            g = np.clip(g, 0, None)
            g = normalize_sum(g)
            lp_fail_count = 0

            # Line search
            best_tau = 0.5
            best_local = np.inf
            for tau in [0.2, 0.3, 0.5, 0.7, 1.0]:
                cand = (1 - tau) * a + tau * g
                cand = normalize_sum(cand)
                sc = score_seq(cand, n)
                if sc < best_local:
                    best_local = sc
                    best_tau = tau

            a = (1 - best_tau) * a + best_tau * g
            a = normalize_sum(a)
        else:
            lp_fail_count += 1
            # Gradient fallback
            c = fast_self_conv(a)
            grad = topk_gradient(a, c, k=12, temp_scale=0.06)
            step = 0.01 / (np.max(np.abs(grad)) + 1e-10)
            a = a - step * grad
            a = normalize_sum(a)
            if lp_fail_count > 20:
                # Restart
                a = best_a * (1 + 0.02 * rng.standard_normal(n))
                a = normalize_sum(a)
                lp_fail_count = 0

        sc = score_seq(a, n)
        if sc < best_score - 1e-9:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
        else:
            no_imp += 1

        if no_imp > 40:
            # Perturb and restart
            a = best_a * (1 + 0.03 * rng.standard_normal(n))
            a = np.clip(a, 0, None)
            a = normalize_sum(a)
            no_imp = 0

        # Progress
        if int(time.monotonic() - start) % 30 < 1:
            print("  t=%.0f, best=%.6f, no_imp=%d" % (time.monotonic() - start, best_score, no_imp))

    print("Phase 2 complete. Best = %.6f" % best_score)

    # Phase 3: Fine gradient polish with Adam
    print("Phase 3: Gradient polish")
    a = best_a.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t_count = 0
    no_imp = 0
    lr = 0.02

    while time.monotonic() < deadline:
        c = fast_self_conv(a)
        grad = topk_gradient(a, c, k=24, temp_scale=0.04)
        t_count += 1
        adam_step(a, grad, m, v, t_count, lr)
        a = np.clip(a, 0, 1000)
        a = normalize_sum(a)

        sc = score_seq(a, n)
        if sc < best_score - 1e-9:
            best_score = sc
            best_a = a.copy()
            no_imp = 0
            lr = min(lr * 1.02, 0.1)
        else:
            no_imp += 1
            if no_imp > 100:
                lr *= 0.5
                no_imp = 0
                if lr < 0.001:
                    break

    print("Phase 3 complete. Best = %.6f" % best_score)

    # Final: one more LP pass if time allows
    if time.monotonic() < deadline - 5:
        a = best_a.copy()
        no_imp = 0
        while time.monotonic() < deadline - 3 and no_imp < 30:
            g, t_val = solve_lp_minpeak(a, rows, cols, n)
            if g is not None:
                g = normalize_sum(np.clip(g, 0, None))
                a = 0.7 * a + 0.3 * g
                a = normalize_sum(a)
            sc = score_seq(a, n)
            if sc < best_score - 1e-9:
                best_score = sc
                best_a = a.copy()
                no_imp = 0
            else:
                no_imp += 1

    # Final output
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    verified = evaluate_sequence(out)
    print("FINAL: internal=%.6f, verified=%.6f, n=%d, time=%.1fs" %
          (best_score, verified, len(out), time.monotonic() - start))
    return out
