"""
Minimize f(a) = 2n*max(conv(a,a))/(sum a)^2 for non-negative a.

Strategy:
- FFT-based self-convolution for O(n log n) evaluation
- Log-sum-exp (soft-max) smoothing of the max with temperature annealing
- Adam optimizer with adaptive learning rate and backtracking
- Diverse initializations across multiple sequence lengths
- Periodic restarts from best incumbent with Gaussian perturbation

import numpy as np
import time


def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

    if s < 1e-300:
        w = np.zeros(len(c))
        w[int(np.argmax(c))] = 1.0
        return w
    return ez / s
