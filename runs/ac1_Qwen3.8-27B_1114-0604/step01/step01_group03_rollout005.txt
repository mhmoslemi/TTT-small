        for kk in range(1, kmax + 1):
            amp = rng.uniform(0.1, 0.5) / math.sqrt(kk)
            ph = rng.uniform(0, 2 * np.pi)
            z += amp * np.cos(2 * np.pi * kk * t + ph)
        a = np.exp(z - z.max())
        cands.append(_normalize(a))

    return cands


def _optimize(a, n, L, m, rng, time_limit):
    """Main optimization loop with adaptive temperature and step size."""
    deadline = time.monotonic() + time_limit
    a = _normalize(a.copy())

    c = _conv_self(a, L, m)
    best_a = a.copy()
    best_max = float(c.max())

    temp = 1.0
    step = 0.08
    topk_frac = 1.0
    no_improve = 0
    it = 0
    phase = 0  # 0=explore, 1=exploit

    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 0.05:
            break

        # Phase transition: switch to exploit after 40% of budget used
        if phase == 0 and (time.monotonic() - (deadline - time_limit)) > 0.4 * time_limit:
            phase = 1
            temp = max(0.3, temp * 0.4)
            step = 0.04
            topk_frac = 0.5

        c = _conv_self(a, L, m)
        cur_max = float(c.max())

        if cur_max < best_max - 1e-15:
            best_max = cur_max
            best_a = a.copy()
            no_improve = 0

        # Stuck detection
        if no_improve > 200:
            # Aggressive restart from best
            strength = 0.15 + 0.25 * rng.random()
            noise = rng.normal(0, strength, n)
            a = _normalize(best_a * np.exp(noise))
            temp = 0.5 if phase == 0 else 0.2
            step = 0.05
            topk_frac = 1.0 if phase == 0 else 0.5
            no_improve = 0
            it = 0
            continue

        # Compute gradient
        g, J = _grad_lse(c, a, n, L, m, temp, topk_frac)
        gmax = float(g.max())
        if gmax < 1e-30:
            no_improve += 1
            it += 1
            continue
        g = g / gmax

        # Line search
        accepted = False
        for s in (step, 0.5 * step, 0.25 * step, 0.1 * step):
            a_new = a * np.exp(-s * g)
            a_new = _normalize(a_new)
            c_new = _conv_self(a_new, L, m)
            sc = float(c_new.max())
            if sc < cur_max:
                a = a_new
                accepted = True
                break

        if accepted:
            step = min(0.4, step * 1.04)
            no_improve = 0
        else:
            step = max(0.003, step * 0.85)
            no_improve += 1

        # Temperature schedule
        if it % 20 == 0:
            if phase == 0:
                temp = min(30.0, temp * 1.06)
            else:
                temp = max(0.05, temp * 0.98)

        # In exploit phase, gradually focus more
        if phase == 1:
            topk_frac = max(0.1, topk_frac * 0.999)

        it += 1

    return best_a, best_max


def _polish_exact(a, n, L, m, rng, time_limit):
    """Polish using exact max (zero temperature) with careful local search."""
    deadline = time.monotonic() + time_limit
    a = _normalize(a.copy())
    c = _conv_self(a, L, m)
    best_a = a.copy()
    best_max = float(c.max())

    step = 0.02
    no_improve = 0

    while time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        if remaining < 0.05:
            break

        c = _conv_self(a, L, m)
        cur_max = float(c.max())

        if cur_max < best_max - 1e-15:
            best_max = cur_max
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 100:
            # Small perturbation
            noise = rng.normal(0, 0.05, n)
            a = _normalize(best_a * np.exp(noise))
            step = 0.02
            no_improve = 0
            continue

        # Use exact top-k focus
        topk = max(5, n // 100)
        idx = np.argpartition(c, -topk)[-topk:]
        # Build weight vector concentrated on top entries
        top_vals = c[idx]
        tmax = top_vals.max()
        # Soft focus on top entries
        w = np.zeros(m)
        for i in idx:
            w[i] = 1.0
        # Weight by proximity to max
        for i in idx:
            w[i] = np.exp(-5.0 * (tmax - c[i]) / max(tmax, 1e-30))
        wsum = w.sum()
        if wsum > 0:
            w /= wsum
        else:
            w[idx] = 1.0 / topk

        # Compute gradient
        a_rev = a[::-1]
        F_rev = np.fft.rfft(a_rev, L)
        W = np.fft.rfft(w, L)
        g = np.fft.irfft(F_rev * W, L)[(n - 1):(n - 1) + n]
        np.maximum(g, 0.0, out=g)
        gmax = float(g.max())
        if gmax < 1e-30:
            no_improve += 1
            continue
        g = g / gmax

        accepted = False
        for s in (step, 0.5 * step, 0.25 * step):
            a_new = a * np.exp(-s * g)
            a_new = _normalize(a_new)
            c_new = _conv_self(a_new, L, m)
            if float(c_new.max()) < cur_max:
                a = a_new
                accepted = True
                break

        if accepted:
            step = min(0.1, step * 1.02)
        else:
            step = max(0.002, step * 0.8)

    return best_a, best_max


def _evaluate_exact(seq):
    """Evaluate using the provided evaluate_sequence function."""
    try:
        val = evaluate_sequence(seq)
        return float(val)
    except Exception:
        # Fallback internal evaluation
        a = [max(0.0, min(1000.0, float(x))) for x in seq]
        n = len(a)
        s = sum(a)
        if s < 0.01:
            return float('inf')
        b = np.convolve(a, a)
        return float(2.0 * n * max(b) / (s * s))


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function. Returns best sequence found within budget."""
    start = time.monotonic()
    hard_deadline = start + budget_s - 10.0

    rng = np.random.default_rng(seed)

    # Load previous best
    prev_seq = None
    prev_n = None
    try:
        h = height_sequence_1
        if h is not None:
            arr = np.asarray(h, dtype=np.float64).ravel()
            if arr.size > 100 and np.all(np.isfinite(arr)) and np.all(arr >= 0):
                prev_seq = arr
                prev_n = arr.size
                print(f"Loaded previous best with n={prev_n}")
    except (NameError, Exception):
        pass

    # Determine primary n
    if prev_n is not None:
        n_primary = prev_n
    else:
        n_primary = 6121

    # Quick n exploration (first 60 seconds)
    # Try a few n values with quick optimization to find the best one
    n_candidates = [n_primary]
    for d in [-1000, -500, 500, 1000, 1500]:
        nc = n_primary + d
        if 2000 <= nc <= 15000 and nc != n_primary:
            n_candidates.append(nc)

    # Remove duplicates, keep order
    seen = set()
    n_candidates = [x for x in n_candidates if x not in seen and not seen.add(x)]

    print(f"Exploring n values: {n_candidates}")

    quick_budget = min(60.0, (hard_deadline - start) * 0.03)
    per_n_quick = quick_budget / max(1, len(n_candidates))

    quick_results = {}
    for nc in n_candidates:
        if time.monotonic() > hard_deadline - 1800:
            break
        m = 2 * nc - 1
        L = 1 << (m - 1).bit_length()
        init = None
        if prev_seq is not None and prev_n == nc:
            init = _normalize(prev_seq)
        else:
            init = _normalize(rng.uniform(0.5, 1.5, nc))
        c = _conv_self(init, L, m)
        base_score = 2.0 * nc * float(c.max())
        quick_results[nc] = (init, base_score)
        print(f"  n={nc}: initial score={base_score:.6f}")

    if not quick_results:
        nc = n_primary
        m = 2 * nc - 1
        L = 1 << (m - 1).bit_length()
        init = _normalize(rng.uniform(0.5, 1.5, nc))
        quick_results[nc] = (init, 2.0 * nc * float(c.max()))

    # Pick best n from quick exploration
    best_n = min(quick_results.keys(), key=lambda x: quick_results[x][1])
    print(f"Best n from quick exploration: {best_n} (score={quick_results[best_n][1]:.6f})")

    # Main optimization on best n
    n = best_n
    m = 2 * n - 1
    L = 1 << (m - 1).bit_length()

    # Generate multiple initial candidates and pick the best after a short trial
    inits = _initializations(n, rng, prev_seq if prev_n == n else None)
    print(f"Trying {len(inits)} initial candidates")

    best_a = None
    best_score = float('inf')
    trial_budget = min(30.0, (hard_deadline - time.monotonic()) * 0.02)
    per_trial = trial_budget / max(1, len(inits))

    for i, init_a in enumerate(inits):
        if time.monotonic() > hard_deadline - 1700:
            break
        t = min(per_trial, 10.0)
        if t < 1.0:
            break
        ra, rm = _optimize(init_a, n, L, m, rng, t)
        sc = 2.0 * n * rm
        if sc < best_score:
            best_score = sc
            best_a = ra
            print(f"  Init {i}: score={sc:.6f}")

    if best_a is None:
        best_a = _normalize(rng.uniform(0.5, 1.5, n))
        best_score = 2.0 * n * float(_conv_self(best_a, L, m).max())

    print(f"Best after initial trials: score={best_score:.6f}")

    # Phase 1: Main optimization (exploit the best basin)
    remaining_main = hard_deadline - 300.0 - time.monotonic()
    if remaining_main > 30:
        print(f"Phase 1: Main optimization with {remaining_main:.0f}s")
        pa, pm = _optimize(best_a, n, L, m, rng, remaining_main)
        ps = 2.0 * n * pm
        if ps < best_score:
            best_score = ps
            best_a = pa
        print(f"  After main: score={ps:.6f}")

    # Phase 2: Polish with exact max
    remaining_polish = hard_deadline - 30.0 - time.monotonic()
    if remaining_polish > 20:
        print(f"Phase 2: Polish with {remaining_polish:.0f}s")
        ra, rm = _polish_exact(best_a, n, L, m, rng, remaining_polish)
        rs = 2.0 * n * rm
        if rs < best_score:
            best_score = rs
            best_a = ra
        print(f"  After polish: score={rs:.6f}")

    # Phase 3: Try a different n if time remains
    remaining = hard_deadline - 15.0 - time.monotonic()
    if remaining > 120:
        # Try the second-best n
        sorted_ns = sorted(quick_results.keys(), key=lambda x: quick_results[x][1])
        for alt_n in sorted_ns:
            if alt_n == n or remaining < 60:
                continue
            alt_m = 2 * alt_n - 1
            alt_L = 1 << (alt_m - 1).bit_length()
            alt_init = quick_results[alt_n][0]
            print(f"Trying alternative n={alt_n} with {remaining:.0f}s")
            ra2, rm2 = _optimize(alt_init, alt_n, alt_L, alt_m, rng, min(remaining - 10, 120))
            sc2 = 2.0 * alt_n * rm2
            if sc2 < best_score:
                best_score = sc2
                best_a = ra2
                n = alt_n
                print(f"  Alt n={alt_n}: score={sc2:.6f} (NEW BEST)")
            remaining = hard_deadline - 15.0 - time.monotonic()
            break  # Only try one alternative

    # Final exact evaluation
    seq = [float(x) for x in best_a]
    final_score = _evaluate_exact(seq)
    print(f"Final exact score: {final_score:.6f} (n={len(seq)})")

    # If the FFT-based score was optimistic, try to fix
    if final_score > best_score + 0.01 and time.monotonic() < hard_deadline - 5:
        print("  FFT score was optimistic, attempting correction...")
        # Use exact convolution for a few more iterations
        for _ in range(50):
            if time.monotonic() > hard_deadline - 3:
                break
            a = np.array(seq)
            c = np.convolve(a, a)
            cur_max = float(c.max())
            topk = max(3, len(a) // 500)
            idx = np.argpartition(c, -topk)[-topk:]
            w = np.zeros(2 * n - 1)
            for i in idx:
                w[i] = 1.0
            a_rev = a[::-1]
            F_rev = np.fft.rfft(a_rev, L)
            W = np.fft.rfft(w, L)
            g = np.fft.irfft(F_rev * W, L)[(n - 1):(n - 1) + n]
            np.maximum(g, 0.0, out=g)
            gmax = g.max()
            if gmax < 1e-30:
                break
            g /= gmax
            a_new = _normalize(a * np.exp(-0.02 * g))
            seq_new = [float(x) for x in a_new]
            sc_new = _evaluate_exact(seq_new)
            if sc_new < final_score:
                final_score = sc_new
                seq = seq_new

    print(f"FINAL: score={final_score:.6f}")
    return seq
