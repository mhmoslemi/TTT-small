    """Approximate the LP direction using projected gradient."""
    n = len(f)
    g = np.ones(n) / n  # start uniform
    
    for it in range(n_iter):
        # Compute f * g
        conv = fft_convolution_cross(f, g)  # linear convolution of f and g
        
        # Penalty: sum of violations
        violations = np.maximum(conv - M, 0)
        
        # Gradient of penalty w.r.t. g:
        # d/dg_j sum_k max(0, (f*g)_k - M) = sum_{k: conv_k > M} f_{k-j} (with bounds)
        # This is a cross-correlation of f with the violation indicator
        indicator = (conv > M).astype(float)  # or use violations / max for smoothness

"""
Minimize 2*n*max(f*f)/(sum(f))^2 for non-negative sequences.

Uses an LP-direction-based optimization approach:
1. At each step, approximate the LP direction (maximize sum(g) s.t. f*g <= M, g >= 0)
   using projected gradient in the "dual" space.
2. Take a convex combination step toward the LP direction.
3. Polish with direct gradient descent on the smooth max objective.

This avoids the limitations of block parameterization and can work directly
in the full n-dimensional space.
"""

import numpy as np
import time

def fft_conv(a, b=None):
    """Linear convolution of a with b (or with itself if b is None)."""
    if b is None:
        b = a
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    na = a.size
    nb = b.size
    need = na + nb - 1
    if need <= 0:
        return np.empty(0)
    size = 1
    while size < need:
        size <<= 1
    FA = np.fft.rfft(a, size)
    FB = np.fft.rfft(b, size)
    result = np.fft.irfft(FA * FB, size)[:need]
    return np.maximum(result, 0.0)


def cross_corr_a_b(a, b):
    """
    Compute c_j = sum_i a[i] * b[j+i] for j=0..len(a)-1.
    where b has length >= len(a)-1 + len(b) ... 
    Actually: a has length n, b has length m.
    c_j = sum_{i=0}^{n-1} a[i] * b[j+i] for j=0..m-n (valid range)
    
    In our use case: a = f (length n), b = viol (length 2n-1).
    c_j = sum_{i=0}^{n-1} f[i] * viol[j+i] for j=0..n-1.
    
    Computed as: convolve(f_rev, viol)[n-1 : n-1+n]
    """
    n = a.size
    m = b.size
    if n == 0 or m == 0:
        return np.empty(0)
    f_rev = a[::-1]
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    FR = np.fft.rfft(f_rev, size)
    BR = np.fft.rfft(b, size)
    conv = np.fft.irfft(FR * BR, size)
    # c_j = conv[n-1+j] for j=0..n-1
    c = conv[n-1 : n-1+n]
    return c


def fast_score(p):
    """Compute 2*n*max(p*p)/(sum(p))^2."""
    n = p.size
    S = float(np.sum(p))
    if S <= 0 or not np.isfinite(S):
        return float('inf')
    b = fft_conv(p)
    M = float(np.max(b))
    return float(2.0 * n * M / (S * S))


def lp_direction(f, M, n_iter=3000, verbose=False):
    """
    Approximate the LP direction: find g that approximately maximizes sum(g)
    subject to f*g <= M (componentwise) and g >= 0.
    
    Uses projected gradient on the penalty formulation:
    minimize 0.5*sum(max(0, (f*g)_k - M)^2) - sum(g)  s.t. g >= 0
    
    Returns g (non-negative, not necessarily normalized).
    """
    n = f.size
    g = np.ones(n) / n  # uniform start
    
    # Precompute FFT of reversed f (for cross-correlation)
    size_xc = 1
    while size_xc < 3 * n - 2:
        size_xc <<= 1
    f_rev = f[::-1]
    FR = np.fft.rfft(f_rev, size_xc)
    
    # For computing f*g
    size_conv = 1
    while size_conv < 2 * n - 1:
        size_conv <<= 1
    FF = np.fft.rfft(f, size_conv)
    
    lr = 2.0 / (float(np.max(f)) * float(np.max(f)) * n / M + 1e-10) if M > 0 else 1.0
    # Actually, the Lipschitz constant of the gradient is ||T_f||^2 = max eigenvalue of T_f^T T_f
    # T_f^T T_f has eigenvalues at most (sum f)^2 = 1 (for normalized f)
    # So the gradient of 0.5*||T_f @ g - M*1_+||^2 has Lipschitz constant at most 1
    # (where 1_+ means we only keep positive parts)
    # So lr = 1 is safe, but we can use a larger value with backtracking
    
    lr = 1.0
    best_g = g.copy()
    best_val = float('inf')
    
    for it in range(n_iter):
        # Compute f * g
        GG = np.fft.rfft(g, size_conv)
        conv = np.fft.irfft(FF * GG, size_conv)[:2*n-1]
        
        # Violations
        viol = np.maximum(conv - M, 0.0)
        
        # For tracking
        val = 0.5 * float(np.sum(viol * viol)) - float(np.sum(g))
        if val < best_val:
            best_val = val
            best_g = g.copy()
        
        if np.max(viol) < 1e-18:
            # All constraints satisfied
            break
        
        # Gradient: d/dg [0.5*sum(viol^2) - sum(g)]
        # = T_f^T @ viol - 1
        # T_f^T @ viol = cross-correlate(f, viol)
        # c_j = sum_i f[i] * viol[j+i]
        VR = np.fft.rfft(viol, size_xc)
        c = np.fft.irfft(FR * VR, size_xc)[n-1 : 2*n-1]
        
        grad = c - 1.0
        
        # Projected gradient step
        g_new = np.maximum(g - lr * grad, 0.0)
        
        # Check if step was good (Armijo condition)
        GG_new = np.fft.rfft(g_new, size_conv)
        conv_new = np.fft.irfft(FF * GG_new, size_conv)[:2*n-1]
        viol_new = np.maximum(conv_new - M, 0.0)
        val_new = 0.5 * float(np.sum(viol_new * viol_new)) - float(np.sum(g_new))
        
        if val_new < val - 1e-10:
            g = g_new
        else:
            # Backtrack
            lr *= 0.5
            if lr < 1e-6:
                break
            continue
    
    # If all constraints are satisfied at the end, we can scale up g
    # (since the LP is homogeneous in the "slack" - well, not quite, but we can try)
    # Actually, let's just return best_g
    return best_g


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if target <= 0:
        return np.zeros_like(v)
    
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = float(np.sum(v))
    if s <= 0:
        return np.full(n, target / n)
    return v * (target / s)


def smooth_max_beta(b, beta):
    """Compute smooth max and weights."""
    m = float(np.max(b))
    if m <= 0:
        return m, np.ones_like(b) / b.size
    x = beta * (b - m)
    x = np.clip(x, -700, 0)
    e = np.exp(x)
    Z = float(np.sum(e))
    w = e / Z
    G = m + np.log(Z) / beta
    return G, w


def grad_smooth_max(f, b, w):
    """
    Gradient of smooth_max(f*f) w.r.t. f.
    = sum_k w_k * 2*f_{k-.*} where the sum is over k
    = 2 * cross_correlate(f, w)
    """
    n = f.size
    c = cross_corr_a_b(f, w[:2*n-1])  # w has length 2n-1
    return 2.0 * c

                break

            step *= 0.5
            if step < 1e-8:
                break

"""
Minimize 2*n*max(f*f)/(sum(f))^2 for non-negative sequences.

Uses an LP-direction-based optimization: at each step, approximate the LP
(maximize sum(g) s.t. f*g <= M, g >= 0) via projected gradient, then take
a convex step toward g. Followed by smooth-max gradient polishing.
Works in the full n-dimensional space (no block parameterization).
"""

import numpy as np
import time


def fft_conv(a, b=None):
    """Linear convolution a*b (or a*a if b is None)."""
    if b is None:
        b = a
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    need = a.size + b.size - 1
    if need <= 0:
        return np.empty(0, dtype=np.float64)
    size = 1
    while size < need:
        size <<= 1
    FA = np.fft.rfft(a, size)
    FB = np.fft.rfft(b, size)
    result = np.fft.irfft(FA * FB, size)[:need]
    return np.maximum(result, 0.0)


def cross_corr(f, w):
    """
    Compute c_j = sum_i f[i] * w[j+i] for j=0..len(f)-1.
    f has length n, w has length >= n (typically 2n-1).
    Uses FFT: c = convolve(f[::-1], w)[n-1 : 2n-1]
    """
    n = f.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    m = w.size
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    f_rev = f[::-1]
    FR = np.fft.rfft(f_rev, size)
    WR = np.fft.rfft(w, size)
    conv = np.fft.irfft(FR * WR, size)
    c = conv[n - 1: n - 1 + n]
    return c


def fast_score(p):
    """Compute 2*n*max(p*p)/(sum(p))^2."""
    n = p.size
    S = float(np.sum(p))
    if S <= 0 or not np.isfinite(S):
        return float('inf')
    b = fft_conv(p)
    M = float(np.max(b))
    return float(2.0 * n * M / (S * S))


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target} via simple rescale."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if target <= 0:
        return np.zeros_like(v)
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = float(np.sum(v))
    if s <= 0 or not np.isfinite(s):
        return np.full(n, target / n, dtype=np.float64)
    return v * (target / s)


def smooth_max_beta(b, beta):
    """Compute smooth max value and softmax weights."""
    m = float(np.max(b))
    if m <= 0:
        return m, np.ones_like(b, dtype=np.float64) / b.size
    x = beta * (b - m)
    x = np.clip(x, -700.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0 or not np.isfinite(Z):
        idx = int(np.argmax(b))
        e = np.zeros_like(b, dtype=np.float64)
        e[idx] = 1.0
        Z = 1.0
    w = e / Z
    G = m + np.log(Z) / beta
    return float(G), w


def lp_direction(f, M, n_iter=4000, warm_start=None):
    """
    Approximate the LP direction: find g maximizing sum(g)
    subject to f*g <= M (componentwise), g >= 0.
    
    Uses projected gradient on: min 0.5*sum(max(0,f*g - M)^2) - sum(g) s.t. g>=0
    Returns g (non-negative).
    """
    n = f.size
    if warm_start is not None:
        g = np.maximum(warm_start, 0.0).copy()
        sg = float(np.sum(g))
        if sg > 0:
            g /= sg
        else:
            g = np.ones(n) / n
    else:
        g = np.ones(n) / n

    # Precompute FFTs
    size_conv = 1
    while size_conv < 2 * n - 1:
        size_conv <<= 1
    size_xc = 1
    while size_xc < 3 * n - 2:
        size_xc <<= 1

    FF = np.fft.rfft(f, size_conv)
    f_rev = f[::-1]
    FR = np.fft.rfft(f_rev, size_xc)

    lr = 1.0
    best_g = g.copy()
    best_val = float('inf')
    stall = 0

    for it in range(n_iter):
        # Compute f * g
        GG = np.fft.rfft(g, size_conv)
        conv = np.fft.irfft(FF * GG, size_conv)[:2 * n - 1]

        # Violations
        viol = np.maximum(conv - M, 0.0)

        # Objective value
        val = 0.5 * float(np.dot(viol, viol)) - float(np.sum(g))
        if val < best_val - 1e-15:
            best_val = val
            best_g = g.copy()
            stall = 0
        else:
            stall += 1

        if stall > 500:
            break
        if np.max(viol) < 1e-16:
            break

        # Gradient: T_f^T @ viol - 1 = cross_corr(f, viol) - 1
        VR = np.fft.rfft(viol, size_xc)
        c = np.fft.irfft(FR * VR, size_xc)[n - 1: 2 * n - 1]
        grad = c - 1.0

        # Projected gradient step
        g_new = np.maximum(g - lr * grad, 0.0)

        # Check improvement (simple)
        if float(np.sum(g_new)) <= 0:
            lr *= 0.5
            if lr < 1e-8:
                break
            continue

        # Accept or backtrack
        GG2 = np.fft.rfft(g_new, size_conv)
        conv2 = np.fft.irfft(FF * GG2, size_conv)[:2 * n - 1]
        viol2 = np.maximum(conv2 - M, 0.0)
        val2 = 0.5 * float(np.dot(viol2, viol2)) - float(np.sum(g_new))

        if val2 < val - 1e-12:
            g = g_new
        else:
            lr *= 0.5
            if lr < 1e-8:
                g = g_new  # accept anyway to make progress
            continue

    return best_g


def gradient_refine(p, deadline, beta_init=3000, beta_end=300000, max_iter=50000):
    """
    Projected gradient descent on smooth max of f*f.
    p normalized to sum=1.
    """
    n = p.size
    p = project_simplex(p, 1.0).copy()
    best_p = p.copy()
    best_score = fast_score(p)

    size_conv = 1
    while size_conv < 2 * n - 1:
        size_conv <<= 1
    size_xc = 1
    while size_xc < 3 * n - 2:
        size_xc <<= 1

    step = 1.0
    no_improve = 0
    it = 0

    while it < max_iter and time.monotonic() < deadline:
        it += 1

        frac = min(1.0, it / max_iter)
        beta = beta_init * (beta_end / beta_init) ** frac

        b = fft_conv(p)
        G, w = smooth_max_beta(b, beta)
        if not np.isfinite(G):
            break

        cur_score = fast_score(p)
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 8000:
            break
        if it % 2000 == 0:
            print(f"    grad it={it} best={best_score:.8f} beta={beta:.0f} step={step:.4f}")

        # Gradient: 2 * cross_corr(p, w) projected to sum-0
        f_rev = p[::-1]
        FR = np.fft.rfft(f_rev, size_xc)
        WR = np.fft.rfft(w, size_xc)
        c = np.fft.irfft(FR * WR, size_xc)[n - 1: 2 * n - 1]
        grad = 2.0 * c
        grad -= np.mean(grad)

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-16:
            break

        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02):
            alpha = step * mult
            q = project_simplex(p - alpha * grad, 1.0)
            qb = fft_conv(q)
            qG, _ = smooth_max_beta(qb, beta)
            if qG < G - 1e-15:
                p = q
                accepted = True
                break

        if not accepted:
            step *= 0.5
            if step < 1e-9:
                break
        else:
            step = min(step * 1.15, 5.0)

    return best_p, best_score


def main_lp_loop(f, deadline, verbose=True, max_steps=60):
    """
    Main optimization loop: iteratively find LP direction and take convex steps.
    """
    n = f.size
    f = project_simplex(f, 1.0)
    best_p = f.copy()
    best_score = fast_score(f)
    prev_g = None

    if verbose:
        print(f"  LP loop start: score={best_score:.8f}")

    no_improve = 0
    for it in range(max_steps):
        if time.monotonic() >= deadline:
            break
        t0 = time.monotonic()

        b = fft_conv(f)
        M = float(np.max(b))

        # Determine LP iterations based on remaining time
        remaining = deadline - time.monotonic()
        if remaining < 10:
            break
        n_lp_iter = min(4000, max(1000, int(remaining * 200)))

        g = lp_direction(f, M, n_iter=n_lp_iter, warm_start=prev_g)

        sg = float(np.sum(g))
        if sg <= 0 or not np.isfinite(sg):
            break
        g = g / sg
        prev_g = g.copy()

        # Try step sizes
        accepted = False
        for alpha in (0.7, 0.5, 0.3, 0.2, 0.1, 0.05):
            f_new = (1.0 - alpha) * f + alpha * g
            f_new = np.maximum(f_new, 0.0)
            s = float(np.sum(f_new))
            if s <= 0:
                continue
            f_new = f_new / s

            sc = fast_score(f_new)
            if sc < best_score - 1e-13:
                f = f_new
                best_score = sc
                best_p = f.copy()
                prev_g = g.copy()
                accepted = True
                no_improve = 0
                if verbose:
                    print(f"  LP step {it+1}: score={sc:.8f} alpha={alpha} time={time.monotonic()-t0:.1f}s")
                break

        if not accepted:
            no_improve += 1
            if verbose:
                print(f"  LP step {it+1}: no improvement, time={time.monotonic()-t0:.1f}s")
            if no_improve >= 8:
                # Try a perturbation to escape
                rng = np.random.default_rng(it * 12345 + 7)
                f = project_simplex(f + 0.01 * rng.standard_normal(n), 1.0)
                no_improve = 0

    return best_p, best_score

    Directly target the convolution peaks with gradient steps.
    """

    best_p = p.copy()
    best_score = fast_score(p)

        M = float(np.max(b))

        if wsum <= 0:

            wsum = 1.0
