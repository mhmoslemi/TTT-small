import numpy as np
import time

def fft_convolution(a):
    n = a.size
    if n == 0:
        return np.empty(0)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)

def smooth_max(b, beta):
    m = np.max(b)
    x = beta * (b - m)
    x = np.clip(x, -745, 0)
    e = np.exp(x)
    Z = np.sum(e)
    G = m + np.log(Z) / beta
    w = e / Z
    return G, w

def simplex_project(v, target=1.0):
    if v.size == 0:
        return v
    low = np.min(v) - 1
    high = np.max(v) + 1
    for _ in range(60):
        mid = 0.5 * (low + high)
        s = np.sum(np.maximum(v - mid, 0))
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0)
    s = np.sum(q)
    if s <= 0:
        return np.full(v.shape, target / v.size)
    return q * (target / s)

def fast_score(p):

    if S <= 0:
        return float('inf')
    b = fft_convolution(p)
    M = np.max(b)

"""
Minimize 2*n*max(a*a)/(sum(a))^2 using Adam optimization with smooth max,
progressive beta annealing, peak shaving, symmetric-restricted optimization,
and local perturbation search. Combines multiple strategies to escape local
minima and converge to a near-optimal solution.
"""

import numpy as np
import time

# --- Core utility functions ---

def _fft_conv(a):
    """Fast linear autocorrelation a*a via FFT."""
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    b = np.maximum(b, 0.0)
    return b


def _smooth_max(b, beta):
    """Smooth maximum via log-sum-exp. Returns (value, weights)."""
    m = float(np.max(b))
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b, dtype=np.float64) / float(b.size)
        return m, w
    x = beta * (b - m)
    np.minimum(x, 0.0, out=x)
    x = np.maximum(x, -745.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z < 1e-300:
        w = np.zeros_like(b, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
        return m, w
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return G, w


def _project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    n = v.size
    if n == 0:
        return v
    v = np.array(v, dtype=np.float64, copy=True)
    if not np.all(np.isfinite(v)):
        v[:] = target / n
        return v
    lo = float(np.min(v)) - 2.0
    hi = float(np.max(v)) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s < 1e-300:
        return np.full(n, target / n, dtype=np.float64)
    q *= target / s
    return q


def _score(p):
    """Score = 2*n*max(conv(p,p))/(sum(p))^2."""
    S = float(np.sum(p))
    if S < 1e-300:
        return float('inf')
    b = _fft_conv(p)
    M = float(np.max(b))
    return 2.0 * p.size * M / (S * S)


def _corr_pw(p, w):
    """Compute s_i = sum_j p_j * w_{i+j} for i=0..n-1."""
    n = p.size
    m = w.size
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[n - 1:2 * n - 1][::-1].copy()
    return s


def _grad_obj(p, beta):
    """
    Gradient of n * smooth_max(conv(p,p), beta) w.r.t. p.
    Returns (objective, gradient).
    """
    n = p.size
    b = _fft_conv(p)
    G, w = _smooth_max(b, beta)
    s = _corr_pw(p, w)
    s = np.maximum(s, 0.0)
    grad = 2.0 * s
    obj = float(n) * G
    return obj, grad

def _adam_optimize(p0, deadline, beta_start=500.0, beta_end=500000.0,
                   lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8,
                   n_phases=5, min_phase_time=10.0):
    """
    Adam optimization with progressive beta annealing on the simplex.
    """
    n = p0.size
    p = np.array(p0, dtype=np.float64, copy=True)
    p = _project_simplex(p, 1.0)

    m_v = np.zeros(n, dtype=np.float64)
    v_v = np.zeros(n, dtype=np.float64)
    t = 0

    total_time = deadline - time.monotonic()
    if total_time < 1.0:
        return p, _score(p)

    phase_time = total_time / n_phases

    best_p = p.copy()
    best_score = _score(p)

    for phase in range(n_phases):
        phase_deadline = time.monotonic() + phase_time
        frac = (phase + 1) / n_phases
        beta = beta_start * (beta_end / beta_start) ** frac

        # Reset Adam state at phase boundaries for fresh start
        if phase > 0:
            m_v[:] = 0.0
            v_v[:] = 0.0
            t = 0

        while time.monotonic() < phase_deadline:
            t += 1
            obj, grad = _grad_obj(p, beta)
            if not np.isfinite(obj):
                break

            # Center the gradient (since we're on the simplex)
            grad = grad - float(np.mean(grad))

            # Adam update
            m_v = beta1 * m_v + (1 - beta1) * grad
            v_v = beta2 * v_v + (1 - beta2) * grad * grad
            m_hat = m_v / (1 - beta1 ** t)
            v_hat = v_v / (1 - beta2 ** t)

            step = p - lr * m_hat / (np.sqrt(v_hat) + eps)
            p = _project_simplex(step, 1.0)

                sc = _score(p)

                    best_score = sc
                    best_p = p.copy()

    if sc < best_score:
        best_score = sc
        best_p = p.copy()

    return best_p, best_score

def _symmetric_project(half, target=1.0):

    - half >= 0
    - 2*sum(half) + center = target (for odd case)

    So sum = 2*sum(half). We want 2*sum(half) = target.
    """

    h = np.maximum(h, 0.0)
    s = float(np.sum(h))
    target_half = target / 2.0
    if s < 1e-300:
        return np.full(n, target_half / n, dtype=np.float64)
    # Need sum(h) = target_half
    # Simple: just scale
    h *= target_half / s
    return h


def _half_to_full(half):

    n = half.size
    return np.concatenate([half, half[::-1]])

    n = p.size
    return p[:n // 2].copy()

    h_grad = grad_full[:n // 2] + grad_full[n // 2:][::-1]

            break

        alpha = lr / gnorm
        q = _project_simplex(p - alpha * grad, 1.0)
        sc = _score(q)

        if sc < best_score:
            p = q
            best_score = sc
            best_p = q.copy()
            lr = min(0.1, lr * 1.05)
        else:
            lr *= 0.8
            if lr < 1e-7:
                break

    return best_p, best_score


# --- Multi-scale refinement ---

def _multiscale_refine(p_coarse, n_fine, deadline):
    """
    Upscale coarse solution and refine at fine scale.
    p_coarse has length n_coarse, we create p_fine of length n_fine.
    """
    n_c = p_coarse.size
    n_f = n_fine
    if n_f < n_c:
        return p_coarse[:n_f].astype(np.float64)

    # Upscale by repeating/interpolating
    ratio = n_f / n_c
    idx = np.linspace(0, n_c - 1, n_f)
    i0 = np.floor(idx).astype(int)
    i1 = np.minimum(i0 + 1, n_c - 1)
    frac = idx - i0
    p_fine = p_coarse[i0] * (1 - frac) + p_coarse[i1] * frac
    p_fine = _project_simplex(p_fine, 1.0)
    return p_fine


# --- Main search function ---

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing
    2*n*max(conv(a,a))/(sum(a))^2.

    Strategy:
    1. Warm start from known best or structured initializations
    2. Adam optimization with progressive beta annealing
    3. Symmetric-restricted Adam
    4. Peak shaving
    5. Targeted lag optimization
    6. Local perturbation search
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 0:
        return [1.0]

    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    deadline = start_time + budget_s - 60.0  # reserve 60s for finalization

    # Choose sequence length
    if budget_s < 300:
        N = 4096
    else:
        N = 8192

    print(f"Starting search: N={N}, budget={budget_s:.0f}s")

    # --- Get initial candidates ---
    candidates = []

    # Try to use height_sequence_1 if available
    try:
        hs1 = height_sequence_1
        if isinstance(hs1, list) and len(hs1) > 0:
            hs1_arr = np.array(hs1, dtype=np.float64)
            hs1_arr = np.maximum(hs1_arr, 0.0)
            s = float(np.sum(hs1_arr))
            if s > 0.01:
                # Resize to N if needed
                if len(hs1_arr) != N:
                    idx = np.linspace(0, len(hs1_arr) - 1, N)
                    i0 = np.floor(idx).astype(int)
                    i1 = np.minimum(i0 + 1, len(hs1_arr) - 1)
                    frac = idx - i0
                    hs1_resized = hs1_arr[i0] * (1 - frac) + hs1_arr[i1] * frac
                else:
                    hs1_resized = hs1_arr.copy()
                hs1_resized = _project_simplex(hs1_resized, 1.0)
                candidates.append(hs1_resized)
                print(f"Loaded height_sequence_1, score={_score(hs1_resized):.8f}")
    except Exception:
        pass

    # Uniform
    candidates.append(np.full(N, 1.0 / N, dtype=np.float64))

    # Structured starts
    x = np.linspace(0, 1, N, dtype=np.float64)
    for a in (-2.0, -1.0, 1.0, 2.0, -3.0, 3.0):
        z = a * (x - 0.5)
        p = np.exp(z)
        p = _project_simplex(p, 1.0)
        candidates.append(p)

    # Gaussian-like
    center = N // 2
    for sigma_frac in (0.1, 0.2, 0.3, 0.5):
        sigma = sigma_frac * N
        z = -((np.arange(N) - center) / sigma) ** 2
        p = np.exp(z)
        p = _project_simplex(p, 1.0)
        candidates.append(p)

    # Triangular
    t = np.abs(np.arange(N) - N / 2) / (N / 2)
    p = np.maximum(0, 1 - t)
    p = _project_simplex(p + 1e-10, 1.0)
    candidates.append(p)

    # Random smooth (band-limited noise)
    for _ in range(3):
        zc = rng.normal(size=N // 2 + 1) + 1j * rng.normal(size=N // 2 + 1)
        z = np.real(np.fft.irfft(zc, N))
        z = (z - z.mean()) / (np.std(z) + 1e-10)
        p = np.exp(0.5 * z)
        p = _project_simplex(p, 1.0)
        candidates.append(p)

    # Score all candidates
    scored = []
    for c in candidates:
        sc = _score(c)
        if np.isfinite(sc):
            scored.append((sc, c))
    scored.sort(key=lambda x: x[0])

    print(f"Best initial candidate score: {scored[0][0]:.8f}")
    best_p = scored[0][1].copy()
    best_score = scored[0][0]

    # --- Phase 1: Adam from best initial ---
    if time.monotonic() < deadline - 120:
        t0 = time.monotonic()
        alloc = min(300.0, (deadline - time.monotonic()) * 0.3)
        sub_dl = time.monotonic() + alloc
        p1, sc1 = _adam_optimize(best_p, sub_dl,
                                 beta_start=500.0, beta_end=200000.0,
                                 lr=0.03, n_phases=5)
        if sc1 < best_score:
            best_score = sc1
            best_p = p1.copy()
        print(f"After Adam phase 1: score={best_score:.8f}, time={time.monotonic()-t0:.1f}s")

    # --- Phase 2: Symmetric Adam ---
    if time.monotonic() < deadline - 120:
        t0 = time.monotonic()
        # Make a symmetric version of best_p
        half0 = best_p[:N // 2].copy()
        half0 = np.maximum(half0, 0.0)
        s = float(np.sum(half0))
        if s > 0:
            half0 *= 0.5 / s

        alloc = min(200.0, (deadline - time.monotonic()) * 0.25)
        sub_dl = time.monotonic() + alloc
        p_sym, sc_sym = _symmetric_adam(half0, sub_dl,
                                        beta_start=500.0, beta_end=200000.0,
                                        lr=0.015, n_phases=4)
        if sc_sym < best_score:
            best_score = sc_sym
            best_p = p_sym.copy()
            print(f"Symmetric improved! score={best_score:.8f}")
        print(f"After symmetric phase: score={best_score:.8f}, time={time.monotonic()-t0:.1f}s")

    # --- Phase 3: Peak shaving ---
    if time.monotonic() < deadline - 60:
        t0 = time.monotonic()
        alloc = min(150.0, (deadline - time.monotonic()) * 0.2)
        sub_dl = time.monotonic() + alloc
        p3, sc3 = _peak_shave(best_p, sub_dl, max_iter=3000)
        if sc3 < best_score:
            best_score = sc3
            best_p = p3.copy()
        print(f"After peak shaving: score={best_score:.8f}, time={time.monotonic()-t0:.1f}s")

    # --- Phase 4: Targeted lag optimization ---
    if time.monotonic() < deadline - 40:
        t0 = time.monotonic()
        alloc = min(150.0, (deadline - time.monotonic()) * 0.2)
        sub_dl = time.monotonic() + alloc
        p4, sc4 = _targeted_lag_opt(best_p, sub_dl, beta=500000.0, max_iter=3000)
        if sc4 < best_score:
            best_score = sc4
            best_p = p4.copy()
        print(f"After targeted lag opt: score={best_score:.8f}, time={time.monotonic()-t0:.1f}s")

    # --- Phase 5: Second Adam pass from best ---
    if time.monotonic() < deadline - 40:
        t0 = time.monotonic()
        alloc = min(200.0, (deadline - time.monotonic()) * 0.3)
        sub_dl = time.monotonic() + alloc
        p5, sc5 = _adam_optimize(best_p, sub_dl,
                                 beta_start=10000.0, beta_end=800000.0,
                                 lr=0.01, n_phases=4)
        if sc5 < best_score:
            best_score = sc5
            best_p = p5.copy()
        print(f"After Adam phase 2: score={best_score:.8f}, time={time.monotonic()-t0:.1f}s")

    # --- Phase 6: Local search ---
    if time.monotonic() < deadline - 20:
        t0 = time.monotonic()
        alloc = min(120.0, (deadline - time.monotonic()) * 0.4)
        sub_dl = time.monotonic() + alloc
        p6, sc6 = _local_search(best_p, sub_dl, n_trials=5000, rng=rng)
        if sc6 < best_score:
            best_score = sc6
            best_p = p6.copy()
        print(f"After local search: score={best_score:.8f}, time={time.monotonic()-t0:.1f}s")

    # --- Phase 7: Final targeted polish ---
    if time.monotonic() < deadline - 10:
        t0 = time.monotonic()
        alloc = min(100.0, (deadline - time.monotonic()) * 0.5)
        sub_dl = time.monotonic() + alloc
        p7, sc7 = _targeted_lag_opt(best_p, sub_dl, beta=1000000.0, max_iter=5000)
        if sc7 < best_score:
            best_score = sc7
            best_p = p7.copy()
        print(f"After final polish: score={best_score:.8f}, time={time.monotonic()-t0:.1f}s")

    # --- Finalize ---
    final_p = best_p.copy()
    final_p = np.maximum(final_p, 0.0)
    final_p = _project_simplex(final_p, 1.0)

    # Verify with official evaluator
    final_seq = [float(x) for x in final_p]
    try:
        official = evaluate_sequence(final_seq)
        official = float(official)
    except Exception:
        official = float('inf')

    if np.isfinite(official) and official < 2.0:
        print(f"Final official score: {official:.8f}")
        return final_seq
    else:
        # Fallback: use the best we found (clipped)
        print(f"Official score invalid ({official}), using clipped version")
        fallback = [max(0.0, float(x)) for x in final_p]
        s = sum(fallback)
        if s > 0.01:
            fallback = [x / s * np.sqrt(2 * N) for x in fallback]
            return fallback
        return [1.0] * N
