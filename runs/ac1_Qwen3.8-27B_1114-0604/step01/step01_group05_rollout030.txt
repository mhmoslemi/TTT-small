"""
Minimize 2*n*max(a*a)/(sum(a))^2 for nonnegative sequences using:
1. Block-constant L-BFGS initialization (coarse-to-fine)
2. Full-length Adam optimization in log-space with adaptive smooth-max beta
3. High-beta projected-gradient polishing
4. Active-set peak-reduction local search

import numpy as np
import time

try:
    from scipy.optimize import minimize
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def fft_convolution(a):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max(b, beta):
    """Smooth max and softmax weights."""
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float('inf'), np.empty(0, dtype=np.float64)
    m = float(np.max(b))
    if not np.isfinite(m):
        return float('inf'), np.zeros_like(b, dtype=np.float64)
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b, dtype=np.float64)
        w /= float(b.size)
        return m, w
    x = np.clip(beta * (b - m), -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w


def softmax(z):
    """Stable softmax."""
    z = np.asarray(z, dtype=np.float64)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        n = z.size
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return e / s


def corr_pw(p, w):
    """s_i = sum_j p_j * w_{i+j} via FFT."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    if n == 1:
        return np.array([float(np.dot(p, w[:1]))], dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n:m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    if target <= 0.0:
        return np.zeros_like(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(v.shape, target / float(v.size), dtype=np.float64)
    low = float(np.min(v)) - 1.0
    high = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (low + high)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        q = np.full(v.shape, target / float(v.size), dtype=np.float64)
    else:
        q *= target / s
    return q


def normalize_simplex(p):
    """Clip nonnegative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size), dtype=np.float64)
    return p * (1.0 / s)


def fast_score(p):
    """Compute 2*n*max(p*p) for normalized p (sum=1)."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float('inf')
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float('inf')
    b = fft_convolution(p)
    M = float(np.max(b))
    if not np.isfinite(M):
        return float('inf')
    return float(2.0 * n * M / (S * S))


def grad_log_space(p, w, beta):
    """
    Compute gradient of smooth_max(p*p) w.r.t. log-parameters z,
    where p = softmax(z).
    
    Returns grad_z = 2 * p * (s - mean_s) where s = corr_pw(p, w).
    """
    s = corr_pw(p, w)
    s = np.maximum(s, 0.0)
    mean_s = float(np.dot(p, s))
    grad_z = 2.0 * p * (s - mean_s)
    return grad_z


def block_to_p(n, B, c):
    """Convert block log-weights to full sequence."""
    L = n // B
    z = np.zeros(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        m = min(B - 1, c.size)
        z[:m] = c[:m]
    z = z - float(np.max(z))
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0.0 or not np.isfinite(s):
        wb = np.ones(B, dtype=np.float64) / float(B)
    else:
        wb /= s
    H = wb / float(L)
    return np.repeat(H, L)


def block_objective(c, n, B, beta):
    """Objective and gradient for block-constant sequences."""
    L = n // B
    z = np.zeros(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        m = min(B - 1, c.size)
        z[:m] = c[:m]
    if not np.all(np.isfinite(z)):
        return float('inf'), np.zeros(max(0, B-1), dtype=np.float64)
    z = z - float(np.max(z))
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0.0 or not np.isfinite(s):
        wb = np.ones(B, dtype=np.float64) / float(B)
    else:
        wb /= s
    H = wb / float(L)
    p = np.repeat(H, L)

    b = fft_convolution(p)
    G, w = smooth_max(b, beta)
    if not np.isfinite(G):
        return float('inf'), np.zeros(max(0, B-1), dtype=np.float64)

    svec = corr_pw(p, w)
    svec = np.maximum(svec, 0.0)
    g = 2.0 * svec
    Gsum = g.reshape(B, L).sum(axis=1)
    Hsum = H.reshape(B, L).sum(axis=1)  # = wb
    Wavg = float(np.dot(wb, Gsum))
    gradz = wb * (Gsum - Wavg) / float(L)
    grad = n * gradz[:-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(n * G), grad.astype(np.float64)


def run_block_minimize(c0, n, B, beta, bounds):
    """Run L-BFGS-B on block representation."""
    if not HAVE_SCIPY:
        return np.asarray(c0, dtype=np.float64)
    c0 = np.asarray(c0, dtype=np.float64).reshape(-1)
    if c0.size != max(0, B - 1):
        tmp = np.zeros(max(0, B - 1), dtype=np.float64)
        m = min(tmp.size, c0.size)
        tmp[:m] = c0[:m]
        c0 = tmp
    c0 = np.clip(c0, -6.0, 6.0)
    options = {"maxiter": 100, "ftol": 1e-12, "gtol": 1e-6, "maxls": 40, "maxcor": 10}
    try:
        res = minimize(block_objective, c0, args=(n, B, beta),
                       method="L-BFGS-B", jac=True, bounds=bounds, options=options)
        c = np.asarray(res.x, dtype=np.float64)
    except Exception:
        c = c0
    if not np.all(np.isfinite(c)):
        c = c0
    return np.clip(c, -6.0, 6.0)


def adam_optimize(z0, n, deadline, lr, beta_start, beta_end, beta_schedule_iters,
                  restart_iters=2000, verbose=True):
    """
    Adam optimization in log-space with adaptive beta.
    
    z0: initial log-parameters
    n: sequence length
    deadline: time.monotonic deadline
    lr: learning rate
    beta_start, beta_end: smooth max beta range
    beta_schedule_iters: over how many iterations to ramp beta
    restart_iters: restart Adam state after this many non-improving iterations
    """
    p = softmax(z0)
    best_p = p.copy()
    best_score = fast_score(p)
    
    m_state = np.zeros(n, dtype=np.float64)
    v_state = np.zeros(n, dtype=np.float64)
    t = 0
    a1, a2, eps = 0.85, 0.999, 1e-8
    
    no_improve = 0
    it = 0
    patience = restart_iters
    
    while time.monotonic() < deadline:
        it += 1
        
        # Adaptive beta
        frac = min(1.0, it / float(beta_schedule_iters))
        beta = beta_start + (beta_end - beta_start) * frac
        
        # Compute gradient
        b = fft_convolution(p)
        G, w = smooth_max(b, beta)
        if not np.isfinite(G):
            break
        
        grad_z = grad_log_space(p, w, beta)
        grad_norm = float(np.linalg.norm(grad_z))
        
        # Track best
        cur_score = 2.0 * n * float(np.max(b))
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and it % 5000 == 0:
            print(f"  adam it={it} beta={beta:.0f} score={cur_score:.8f} best={best_score:.8f} norm={grad_norm:.4e}")
        
        if no_improve > patience:
            # Restart Adam momenta
            m_state *= 0.0
            v_state *= 0.0
            t = 0
            no_improve = 0
            lr *= 0.7
            if lr < 1e-6:
                break
            patience = max(1000, patience // 2)
            continue
        
        # Adam step
        t += 1
        m_state = a1 * m_state + (1 - a1) * grad_z
        v_state = a2 * v_state + (1 - a2) * grad_z * grad_z
        m_hat = m_state / (1.0 - a1 ** t)
        v_hat = v_state / (1.0 - a2 ** t)
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Keep z in reasonable range
        z = np.clip(z, -20.0, 20.0)
        
        # Update p
        p = softmax(z)
        
    return best_p, best_score


def peak_reduction_search(p0, n, deadline, verbose=True):

    Active-set peak reduction: iteratively reduce the dominant peaks
    by adjusting contributing coordinates.
    """
    p = normalize_simplex(p0)
    best_p = p.copy()
    best_score = fast_score(p)
    
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        it += 1
        b = fft_convolution(p)
        M = float(np.max(b))
        cur_score = 2.0 * n * M
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if verbose and it % 100 == 0:
            print(f"  peak it={it} score={cur_score:.8f} best={best_score:.8f}")
        
        if no_improve > 500:
            break
        
        # Find the peak position
        k_peak = int(np.argmax(b))
        
        # Find top contributing pairs

"""
Minimize 2*n*max(a*a)/(sum(a))^2 for nonnegative sequences using:
1. Block-constant L-BFGS coarse initialization
2. Full-length Adam optimization in log-space with adaptive smooth-max beta
3. Nesterov accelerated gradient (NAG) polishing
4. Peak-reduction local search on the active set
Combines multiple starting points including the parent solution for robustness.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def fft_convolution(a):
    """Linear convolution a*a via FFT, clipped to non-negative."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max(b, beta):
    """Compute smooth max and softmax weights."""
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float('inf'), np.empty(0, dtype=np.float64)
    m = float(np.max(b))
    if not np.isfinite(m):
        return float('inf'), np.zeros_like(b, dtype=np.float64)
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b, dtype=np.float64)
        w /= float(b.size)
        return m, w
    x = np.clip(beta * (b - m), -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w


def softmax_z(z):
    """Stable softmax."""
    z = np.asarray(z, dtype=np.float64)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        n = z.size
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    return e / s


def seq_to_z(p):
    """Convert a non-negative sequence to log-parameters (with small epsilon floor)."""
    p = np.asarray(p, dtype=np.float64)
    p = np.maximum(p, 1e-30)
    return np.log(p)


def corr_pw(p, w):
    """s_i = sum_j p_j * w_{i+j} via FFT cross-correlation."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    if n == 1:
        return np.array([float(np.dot(p, w[:1]))], dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n:m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target} via bisection."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    if target <= 0.0:
        return np.zeros_like(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(v.shape, target / float(v.size), dtype=np.float64)
    low = float(np.min(v)) - 1.0
    high = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (low + high)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        q = np.full(v.shape, target / float(v.size), dtype=np.float64)
    else:
        q *= target / s
    return q


def normalize_simplex(p):
    """Clip to non-negative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size), dtype=np.float64)
    return p * (1.0 / s)


def fast_score(p):
    """Compute 2*n*max(p*p)/(sum(p))^2."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float('inf')
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float('inf')
    b = fft_convolution(p)
    M = float(np.max(b))
    if not np.isfinite(M):
        return float('inf')
    return float(2.0 * n * M / (S * S))


def grad_in_log_space(p, w):
    """
    Gradient of smooth_max(p*p) w.r.t. z where p = softmax(z).
    grad_z = 2 * p * (s - mean_s), where s = corr_pw(p, w)
    """
    s = corr_pw(p, w)
    s = np.maximum(s, 0.0)
    mean_s = float(np.dot(p, s))
    grad_z = 2.0 * p * (s - mean_s)
    return grad_z


def grad_simplex(p, w):
    """
    Gradient of smooth_max(p*p) w.r.t. p on the simplex.
    grad_p = 2 * s, projected to be orthogonal to the simplex constraint.
    """
    s = corr_pw(p, w)
    s = np.maximum(s, 0.0)
    grad_p = 2.0 * s - float(np.mean(2.0 * s))
    return grad_p


def block_to_p(n, B, c):
    """Convert block log-weights to full sequence."""
    L = n // B
    z = np.zeros(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        m = min(B - 1, c.size)
        z[:m] = c[:m]
    z = z - float(np.max(z))
    wb = np.exp(z)
    ss = float(np.sum(wb))
    if ss <= 0.0 or not np.isfinite(ss):
        wb = np.ones(B, dtype=np.float64) / float(B)
    else:
        wb /= ss
    H = wb / float(L)
    return np.repeat(H, L)


def block_objective(c, n, B, beta):
    """Objective and gradient for block-constant sequences."""
    L = n // B
    z = np.zeros(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        m = min(B - 1, c.size)
        z[:m] = c[:m]
    if not np.all(np.isfinite(z)):
        return float('inf'), np.zeros(max(0, B - 1), dtype=np.float64)
    z = z - float(np.max(z))
    wb = np.exp(z)
    ss = float(np.sum(wb))
    if ss <= 0.0 or not np.isfinite(ss):
        wb = np.ones(B, dtype=np.float64) / float(B)
    else:
        wb /= ss
    H = wb / float(L)
    p = np.repeat(H, L)

    b = fft_convolution(p)
    G, w = smooth_max(b, beta)
    if not np.isfinite(G):
        return float('inf'), np.zeros(max(0, B - 1), dtype=np.float64)

    svec = corr_pw(p, w)
    svec = np.maximum(svec, 0.0)
    g = 2.0 * svec
    Gsum = g.reshape(B, L).sum(axis=1)
    Wavg = float(np.dot(wb, Gsum))
    gradz = wb * (Gsum - Wavg) / float(L)
    grad = n * gradz[:-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(n * G), grad.astype(np.float64)


def run_block_minimize(c0, n, B, beta, bounds):
    """Run L-BFGS-B on the block representation."""
    if not HAVE_SCIPY:
        return np.asarray(c0, dtype=np.float64)
    c0 = np.asarray(c0, dtype=np.float64).reshape(-1)
    if c0.size != max(0, B - 1):
        tmp = np.zeros(max(0, B - 1), dtype=np.float64)
        m = min(tmp.size, c0.size)
        tmp[:m] = c0[:m]
        c0 = tmp
    c0 = np.clip(c0, -6.0, 6.0)
    options = {"maxiter": 80, "ftol": 1e-12, "gtol": 1e-6, "maxls": 30, "maxcor": 10}
    try:
        res = minimize(block_objective, c0, args=(n, B, beta),
                       method="L-BFGS-B", jac=True, bounds=bounds, options=options)
        c = np.asarray(res.x, dtype=np.float64)
    except Exception:
        c = c0
    if not np.all(np.isfinite(c)):
        c = c0
    return np.clip(c, -6.0, 6.0)

    inits = []

    m = B - 1
    inits.append(np.zeros(m, dtype=np.float64))
    x = np.linspace(0.0, 1.0, m, dtype=np.float64)

        inits.append(a * (x - 0.5))
    c = np.zeros(m, dtype=np.float64)
    half = m // 2
    if half > 0:
        c[:half] = 2.0
    if m - half > 0:
        c[half:] = -2.0
    inits.append(c)
    t = np.linspace(0.0, 2.0 * np.pi, m, dtype=np.float64)
    inits.append(np.cos(t))
    inits.append(np.sin(t))
    inits.append(0.8 * rng.normal(0.0, 1.0, m))

    # Add a "ramp" that emphasizes one side
    inits.append(1.0 * x)
    inits.append(-1.0 * x)

def generate_starts(n, rng, parent_seq=None):

    starts = []

        starts.append(p)

    # Linear ramps
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    for a in (-4.0, -2.0, 2.0, 4.0, 6.0):
        p = np.exp(a * x)
        starts.append(normalize_simplex(p))
    for a in (-4.0, -2.0, 2.0, 4.0):
        p = np.exp(a * (x - 0.5))
        starts.append(normalize_simplex(p))

    # Quadratic / parabolic
    for a in (-3.0, 3.0):
        p = np.exp(a * (2.0 * x - 1.0) ** 2)
        starts.append(normalize_simplex(p))

    # Sine / cosine profiles
    for k in (1, 2, 3):
        p = np.exp(0.5 * np.sin(2.0 * np.pi * k * x))
        starts.append(normalize_simplex(p))
        p = np.exp(0.5 * np.cos(2.0 * np.pi * k * x))
        starts.append(normalize_simplex(p))

    # Random smooth profiles (via random Fourier)
    for _ in range(4):
        zc = rng.normal(size=n // 2 + 1) + 1j * rng.normal(size=n // 2 + 1)
        # Zero out high frequencies for smoothness
        cutoff = max(2, n // 8)
        zc[cutoff:] = 0.0
        z = np.real(np.fft.irfft(zc, n))
        z = z - float(np.mean(z))
        p = np.exp(1.0 * z)
        starts.append(normalize_simplex(p))

    # Asymmetric: high on one side
    for eps in (0.3, 0.5, 0.7):
        p = np.ones(n, dtype=np.float64)
        p[:n // 3] = 1.0 + eps
        p[2 * n // 3:] = max(0.01, 1.0 - eps)
        starts.append(normalize_simplex(p))

    return starts


def adam_full_optimize(z0, n, deadline, lr_init, beta_init, beta_final,
                       beta_ramp_iters, patience=5000, lr_decay=0.7,
                       verbose=True, tag=""):
    """
    Adam optimization in log-space with adaptive beta.
    Returns (best_p, best_score).
    """
    p = softmax_z(z0)
    best_p = p.copy()
    best_score = fast_score(p)

    m_state = np.zeros(n, dtype=np.float64)
    v_state = np.zeros(n, dtype=np.float64)
    t = 0
    a1, a2, eps = 0.85, 0.999, 1e-8
    lr = lr_init

    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        # Adaptive beta schedule (log-linear ramp)
        frac = min(1.0, float(it) / float(beta_ramp_iters))
        beta = beta_init * (beta_final / beta_init) ** frac

        b = fft_convolution(p)
        G, w = smooth_max(b, beta)
        if not np.isfinite(G):
            break

        grad_z = grad_in_log_space(p, w)
        gn = float(np.linalg.norm(grad_z))
        if not np.isfinite(gn) or gn < 1e-15:
            break

        cur_score = 2.0 * n * float(np.max(b))
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 2000 == 0:
            print(f"  {tag} adam it={it} beta={beta:.0f} cur={cur_score:.8f} best={best_score:.8f} gn={gn:.3e}")

        if no_improve > patience:
            # Restart
            m_state *= 0.0
            v_state *= 0.0
            t = 0
            no_improve = 0
            lr *= lr_decay
            if lr < 1e-7:
                break
            # Small perturbation to escape
            pert = 0.1 * np.linalg.norm(z0) * np.random.standard_normal(n)
            z0 = z0 + pert / max(1.0, np.linalg.norm(pert) * np.max(np.abs(z0)) + 1e-10)
            z0 = np.clip(z0, -20.0, 20.0)
            p = softmax_z(z0)
            continue

        # Adam update
        t += 1
        m_state = a1 * m_state + (1.0 - a1) * grad_z
        v_state = a2 * v_state + (1.0 - a2) * grad_z * grad_z
        m_hat = m_state / (1.0 - a1 ** t)
        v_hat = v_state / (1.0 - a2 ** t)
        z0 = z0 - lr * m_hat / (np.sqrt(v_hat) + eps)
        z0 = np.clip(z0, -20.0, 20.0)
        p = softmax_z(z0)

    return best_p, best_score


def nag_polish(p0, n, deadline, beta=100000.0, verbose=True, tag=""):
    """
    Nesterov Accelerated Gradient on the simplex for polishing.
    Minimizes smooth_max(p*p) with sum(p)=1, p>=0.
    """
    p = normalize_simplex(p0)
    best_p = p.copy()
    best_score = fast_score(p)

    # Nesterov parameters
    t_nest = 1.0
    alpha = 1.0
    no_improve = 0
    it = 0
    y = p.copy()  # momentum variable

    while time.monotonic() < deadline:
        it += 1
        b = fft_convolution(y)
        G, w = smooth_max(b, beta)
        if not np.isfinite(G):
            break

        grad_y = grad_simplex(y, w)
        gn = float(np.linalg.norm(grad_y))
        if not np.isfinite(gn) or gn < 1e-14:
            break

        # Step size
        step = alpha / (it + 1.0)
        # Adaptive: if gradient is large, use smaller step
        if gn > 1.0:
            step = min(step, 1.0 / (gn + 1e-8))

        p_new = project_simplex(y - step * grad_y, 1.0)

        # Nesterov momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest)) / 2.0
        momentum = (t_nest - 1.0) / (t_new + 1.0)
        y = p_new + momentum * (p_new - p)
        y = project_simplex(y, 1.0)
        t_nest = t_new
        p = p_new

        cur_score = 2.0 * n * float(np.max(fft_convolution(p)))
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 1000 == 0:
            print(f"  {tag} NAG it={it} cur={cur_score:.8f} best={best_score:.8f}")

        if no_improve > 3000:
            break

        # Decay alpha slowly
        if it % 5000 == 0:
            alpha *= 0.8

    return best_p, best_score


def projected_gradient_polish(p0, n, deadline, beta, verbose=True, tag=""):
    """Projected gradient with line search, similar to parent but with improvements."""
    p = normalize_simplex(p0)
    best_p = p.copy()
    best_score = fast_score(p)

    step = 0.5
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1
        b = fft_convolution(p)
        G, w = smooth_max(b, beta)
        if not np.isfinite(G):
            break

        grad_p = grad_simplex(p, w)
        gn = float(np.linalg.norm(grad_p))
        if not np.isfinite(gn) or gn < 1e-14:
            break

        cur_score = 2.0 * n * float(np.max(b))
        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and it % 500 == 0:
            print(f"  {tag} PG it={it} beta={beta:.0f} cur={cur_score:.8f} best={best_score:.8f}")

        if no_improve > 5000:
            break

        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.01):
            alpha = step * mult
            q = project_simplex(p - alpha * grad_p, 1.0)
            qb = fft_convolution(q)
            m2 = float(np.max(qb))
            cur2 = 2.0 * n * m2
            if cur2 < cur_score - 1e-12:
                p = q
                cur_score = cur2
                accepted = True
                break

        if accepted:
            step = min(step * 1.3, 5.0)
        else:
            step *= 0.5
            if step < 1e-8:
                break

    return best_p, best_score


def official_score(seq):
    """Call the authoritative evaluator safely."""
    try:
        val = evaluate_sequence(seq)
        val = float(val)
    except Exception:
        return float('inf')
    return val if np.isfinite(val) else float('inf')


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Returns a list of non-negative floats.
    Uses Adam in log-space + NAG polishing + block initialization.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 0.0:
        return [1.0]

    reserve = 60.0
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - reserve)

    rng = np.random.default_rng(seed)

    # Choose sequence length
    if budget_s < 100.0:
        N = 1024
    elif budget_s < 400.0:
        N = 2048
    elif budget_s < 800.0:
        N = 4096
    else:
        N = 8192

    print(f"Starting optimization: N={N}, budget={budget_s:.0f}s")

    # Get parent solution if available
    parent = None
    try:
        hs = height_sequence_1
        if isinstance(hs, list) and len(hs) > 0:
            parent = hs
    except NameError:
        pass
    except Exception:
        pass

    # ---- Phase 1: Block initialization (up to 60s) ----
    print("Phase 1: Block initialization")
    p_best = np.full(N, 1.0 / float(N), dtype=np.float64)
    score_best = fast_score(p_best)

    block_deadline = min(deadline, start_time + min(60.0, 0.15 * budget_s))
    if budget_s >= 300.0 and time.monotonic() < block_deadline:
        p_bl, sc_bl = param_block_search(N, block_deadline, rng, max_B=128)
        if np.isfinite(sc_bl) and sc_bl < score_best:
            score_best = sc_bl
            p_best = p_bl.copy()
            print(f"  Block init score: {score_best:.8f}")

    # ---- Phase 2: Generate diverse starts and pick best few ----
    print("Phase 2: Generating diverse starts")
    starts = generate_starts(N, rng, parent)

    # Score all starts
    scored_starts = []
    for p_s in starts:
        sc = fast_score(p_s)
        if np.isfinite(sc):
            scored_starts.append((sc, p_s.copy()))

    # Add block result
    if np.isfinite(score_best):
        scored_starts.append((score_best, p_best.copy()))

    # Sort by score
    scored_starts.sort(key=lambda x: x[0])
    print(f"  Generated {len(scored_starts)} starts, best initial: {scored_starts[0][0]:.8f}")

    # Select top starts for Adam runs
    n_adam_runs = 4 if budget_s >= 1500 else 3
    top_starts = [p_s for _, p_s in scored_starts[:min(n_adam_runs, len(scored_starts))]]

    # ---- Phase 3: Adam optimization (bulk of time) ----
    print("Phase 3: Adam optimization")
    adam_deadline = deadline - 200.0  # Reserve 200s for polishing
    time_per_adam = max(30.0, (adam_deadline - time.monotonic()) / max(1, len(top_starts)))

    for i, p_s in enumerate(top_starts):
        if time.monotonic() >= adam_deadline:
            break
        remaining = adam_deadline - time.monotonic()
        if remaining < 30.0:
            break

        alloc = min(time_per_adam, remaining)
        sub_dl = time.monotonic() + alloc

        # Convert to log-space
        p_s = np.maximum(p_s, 1e-20)
        z0 = np.log(p_s / float(np.sum(p_s)))

        # Different hyperparameters per run for diversity
        lr_val = [0.05, 0.03, 0.02, 0.01][i % 4]
        beta_s = [1000.0, 500.0, 2000.0, 800.0][i % 4]
        beta_e = [200000.0, 100000.0, 300000.0, 50000.0][i % 4]

        p_fit, sc_fit = adam_full_optimize(
            z0, N, sub_dl,
            lr_init=lr_val,
            beta_init=beta_s,
            beta_final=beta_e,
            beta_ramp_iters=5000,
            patience=3000 + 1000 * i,
            lr_decay=0.6,
            verbose=True,
            tag=f"Adam[{i}]"
        )

        if np.isfinite(sc_fit) and sc_fit < score_best:
            score_best = sc_fit
            p_best = p_fit.copy()
            print(f"  Adam[{i}] improved to {score_best:.8f}")

    # ---- Phase 4: NAG + PG polishing (last ~200s) ----
    print("Phase 4: Polishing")
    polish_start = time.monotonic()
    polish_alloc = max(30.0, min(200.0, deadline - polish_start))

    # NAG with high beta
    nag_dl = polish_start + polish_alloc * 0.6
    p_nag, sc_nag = nag_polish(p_best, N, nag_dl, beta=200000.0, verbose=True, tag="NAG")
    if np.isfinite(sc_nag) and sc_nag < score_best:
        score_best = sc_nag
        p_best = p_nag.copy()
        print(f"  NAG improved to {score_best:.8f}")

    # PG with very high beta for final polish
    pg_dl = deadline - 10.0
    if time.monotonic() < pg_dl - 5.0:
        p_pg, sc_pg = projected_gradient_polish(p_best, N, pg_dl, beta=500000.0,
                                                 verbose=True, tag="PG")
        if np.isfinite(sc_pg) and sc_pg < score_best:
            score_best = sc_pg
            p_best = p_pg.copy()
            print(f"  PG improved to {score_best:.8f}")

    # ---- Final: Convert and evaluate ----
    print("Final evaluation")
    final_p = normalize_simplex(p_best)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)

    # Scale: the evaluator clips to [0, 1000] and the score is scale-invariant
    # Keep sum around 1 for safety
    final_s = float(np.sum(final_p))
    if final_s > 0:
        final_p /= final_s

    final_seq = [float(x) for x in final_p]
    off = official_score(final_seq)
    print(f"  Official score: {off:.8f}")

    # Fallback
    if not np.isfinite(off) or off > 2.0:
        print("  Fallback to uniform")
        return [1.0] * N

    return final_seq
