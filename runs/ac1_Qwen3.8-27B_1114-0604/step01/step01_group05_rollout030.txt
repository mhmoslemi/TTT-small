def fft_conv(a):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0]*a[0]], dtype=np.float64)
    size = 1
    while size < 2*n - 1:
        size <<= 1
    F = np.fft.rfft(a, size)
    c = np.fft.irfft(F * F, size)[:2*n-1]
    return np.maximum(c, 0.0)


def smooth_max(b, beta):
    """Smooth max value and softmax weights."""
    m = float(np.max(b))

        # In FFT: IFFT(conj(FFT(padded_p)) * FFT(padded_viol))

"""
Minimize 2*n*max(a*a)/(sum(a))^2 using multi-scale smooth-max gradient

import numpy as np
import time


def fft_conv(a):
    """Linear convolution a*a via FFT, clipped to non-negative."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    F = np.fft.rfft(a, size)
    c = np.fft.irfft(F * F, size)[:2 * n - 1]
    return np.maximum(c, 0.0)


def fft_conv_ab(a, b):
    """Linear convolution a*b via FFT."""
    n = a.size
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b, size)
    c = np.fft.irfft(A * B, size)[:2 * n - 1]
    return np.maximum(c, 0.0)


def smooth_max(b, beta):
    """Smooth max value G and softmax weights w."""
    m = float(np.max(b))
    if m <= 0.0 or beta <= 0.0:
        w = np.full(b.shape, 1.0 / b.size, dtype=np.float64)
        return m, w
    x = np.clip(beta * (b - m), -700.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return G, w


def corr_pw(p, w):
    """s_i = sum_j p_j * w[i+j] for i=0..n-1 (cross-correlation via FFT)."""
    n = p.size
    size = 1
    while size < 2 * n - 1:
        size <<= 1
    Pr = np.fft.rfft(p[::-1], size)
    W = np.fft.rfft(w, size)
    c = np.fft.irfft(Pr * W, size)
    return c[n - 1:2 * n - 1].copy()


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target} via bisection."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = float(np.sum(v))
    if s <= target:
        if s <= 0:
            return np.full(n, target / n, dtype=np.float64)
        return v * (target / s)
    lo = 0.0
    hi = float(np.max(v))
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0:
        return np.full(n, target / n, dtype=np.float64)
    return q * (target / s)


def fast_score(p):

    c = fft_conv(p)
    M = float(np.max(c))
    return float(2.0 * n * M / (S * S))

    Minimizes n * smooth_max(p*p, beta) projected onto the simplex.
    Returns (best_p, best_true_score).
    """

    n = p.size
    v = np.zeros(n, dtype=np.float64)
    step = step0
    best_p = p.copy()
    best_score = fast_score(p)
    no_improve = 0

    for it in range(max_iters):
        if time.monotonic() >= deadline:
            break

        c = fft_conv(p)
        G, w = smooth_max(c, beta)
        if not np.isfinite(G):
            break

        s = corr_pw(p, w)

        grad_centered = grad - float(np.mean(grad))

        gnorm = float(np.linalg.norm(grad_centered))

        if no_improve > 200:
            break

    return best_p, best_score
