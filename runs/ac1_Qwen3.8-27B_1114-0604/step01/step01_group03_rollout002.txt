import math

import numpy as np


def _normalize(a, eps=1e-300):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = float(a.sum())
    if s < eps or not np.isfinite(s):
        a = np.ones_like(a)
        s = float(a.sum())
    a = a / s
    return a


def _conv_self(a, L, m):
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _score(c, n):
    return float(2.0 * n * np.max(c))


def _cross_corr_uv(u, v, L, n_u, n_v):
    """Linear cross-correlation: result[i] = sum_k u[k]*v[k-i-n_v+1] appropriately.
    We want: result[i] = sum_k u[k]*v[k-i] for i=0..n_v-1 (valid range)."""
    v_rev = v[::-1]
    U = np.fft.rfft(u, L)
    VR = np.fft.rfft(v_rev, L)
    r = np.fft.irfft(U * VR, L)
    # conv(u, v_rev)[i] = sum_j u[j]*v_rev[i-j] = sum_j u[j]*v[n_v-1-(i-j)]
    # We want sum_k u[k]*v[k-i] = sum_k u[k]*v_rev[n_v-1-(k-i)]
    # = conv(u, v_rev)[n_v-1-i + n_v-1]... let me just index properly.
    # conv(u, v_rev)[t] = sum_j u[j] * v_rev[t-j]
    # We want sum_k u[k] * v[k-i] = sum_k u[k] * v_rev[(n_v-1)-(k-i)]
    # = sum_k u[k] * v_rev[n_v-1-k+i]
    # Let t = n_v-1+i, then we need sum_k u[k]*v_rev[t-k] = conv(u,v_rev)[t]
    # So result[i] = conv(u, v_rev)[n_v-1+i]
    start = n_v - 1
    end = start + n_v
    return r[start:end]


def _gradient_softmax(c, a, L, n, temp):
    """Gradient of sum_k p_k * c[k] where p = softmax(c/temp), w.r.t. a.
    = 2 * cross_correlation(p, a)
    Then the full objective gradient is 2*n * this (chained through max approx).
    We just return 2 * cross_corr(p, a) as the "direction to decrease"."""
    c_shifted = c - c.max()
    z = np.clip(c_shifted / temp, -500.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-300:
        p = np.ones_like(c) / len(c)
    else:
        p = e / s
    gc = _cross_corr_uv(p, a, L, len(p), n)
    return 2.0 * gc


def _gradient_topk(c, a, L, n, K):
    """Gradient using top-K lags: 2 * cross_correlation(w, a) where w[k]=c[k] for top-K."""
    m = len(c)

    ws = float(w.sum())

                    # Big restart
                    mix = rng.uniform(0.3, 0.7)
                    prior = _random_prior(n, rng, style=int(rng.integers(0, 5)))
                    a = _normalize((1 - mix) * best_a + mix * prior)
                    T = max(0.01, c.max() * 0.1)
                    K = max(10, (2 * n - 1) // 10)
                    lr = 0.1
                    no_improve = 0
                else:
                    # Moderate perturbation
                    noise = rng.normal(0, 0.1, n)
                    a = _normalize(a + noise)
                    T *= 1.5
                    lr = 0.05
                    no_improve = patience // 2

        # Adapt temperature
        if total_iters % 50 == 0:
            cmax = float(c.max())
            T = max(0.005, T * 1.1)
            K = max(5, int(K * 0.95))

        total_iters += 1

    return best_a, best_score


def _optimize_dual(a, L, m, n, rng, time_limit):
    """Dual-tracking optimization: maintain a distribution q over lags."""
    deadline = time.monotonic() + time_limit
    a = _normalize(a)
    m_total = 2 * n - 1

    q = np.full(m_total, 1.0 / m_total)  # Dual variable
    Tq = 1.0
    lr = 0.05
    score = None
    best_a = a.copy()
    best_score = float(np.inf)
    no_improve = 0

    while time.monotonic() < deadline:
        if time.monotonic() > deadline - 0.1:
            break

        c = _conv_self(a, L, m_total)
        score = _score(c, n)

        if score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Update dual q to track peaks
        Tq_eff = max(0.01, c.max() * 0.03)
        z = np.clip((c - c.max()) / Tq_eff, -50.0, 0.0)
        e = np.exp(z)
        s = float(e.sum())
        if s > 0:
            q = e / s

        # Compute gradient using q
        grad = _cross_corr_uv(q, a, L, m_total, n)
        grad = np.maximum(grad, 0.0)
        gmax = float(grad.max())
        if gmax < 1e-30:
            break
        grad = grad / gmax

        # Line search
        a_new, new_score = _backtrack_line_search(a, grad, c, score, L, m_total, n, lr, max_tries=6)
        if new_score < score - 1e-15:
            a = a_new
            score = new_score
            lr = min(lr * 1.05, 1.0)
        else:
            lr = max(lr * 0.8, 0.005)

        if no_improve > 150:
            prior = _random_prior(n, rng, style=int(rng.integers(0, 5)))
            mix = rng.uniform(0.2, 0.5)
            a = _normalize((1 - mix) * best_a + mix * prior)
            no_improve = 0
            lr = 0.05

    return best_a, best_score


def _polish(a, L, m, n, time_limit):
    """Fine-tuning with very small steps and increasing K."""
    deadline = time.monotonic() + time_limit
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score(_conv_self(a, L, m), n)

    T = 0.5
    lr = 0.01
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        if time.monotonic() > deadline - 0.05:
            break

        c = _conv_self(a, L, m)
        score = _score(c, n)
        if score < best_score - 1e-15:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Use very peaked weights (small T means focus on the actual max)
        T_eff = max(0.001, c.max() * 0.005)
        grad = _gradient_topk(c, a, L, n, K=5)
        grad = np.maximum(grad, 0.0)
        gmax = float(grad.max())
        if gmax < 1e-30:
            break
        grad = grad / gmax

        a_new, new_score = _backtrack_line_search(a, grad, c, score, L, m, n, lr, max_tries=6)
        if new_score < score - 1e-16:
            a = a_new
            lr = min(lr * 1.02, 0.1)
        else:
            lr = max(lr * 0.85, 0.001)

        if no_improve > 100:
            # Tiny perturbation
            idx = np.random.randint(0, n, size=20)
            a[idx] *= (1 + np.random.uniform(-0.05, 0.05, 20))
            a = _normalize(a)
            no_improve = 0

        it += 1

    return best_a, best_score


def _exact_score(seq):
    try:
        val = evaluate_sequence(seq)
    except NameError:
        pass
    except Exception:
        return float(np.inf)
    if val is None:
        return float(np.inf)
    return float(val)


def _internal_exact(seq):
    if not isinstance(seq, list) or not seq:
        return float(np.inf)
    for x in seq:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return float(np.inf)
        if np.isnan(x) or np.isinf(x):
            return float(np.inf)
    a = [max(0.0, min(1000.0, float(x))) for x in seq]
    n = len(a)
    s = sum(a)
    if s < 0.01:
        return float(np.inf)
    b = np.convolve(a, a)
    mb = float(np.max(b))
    return float(2.0 * n * mb / (s * s))


def _evaluate_exact(seq):
    try:
        val = evaluate_sequence(seq)
    except NameError:
        val = _internal_exact(seq)
    except Exception:
        val = _internal_exact(seq)
    if val is None:
        return _internal_exact(seq)
    return float(val)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    try:
        budget = float(budget_s)
    except (ValueError, TypeError):
        budget = 2400.0
    if not np.isfinite(budget) or budget < 10:
        budget = 2400.0

    start_time = time.monotonic()
    hard_deadline = start_time + budget - 5.0

    rng = np.random.default_rng(seed)
    n = 6121
    L = 1 << (2 * n - 1).bit_length()
    m = 2 * n - 1

    # Get initial candidates
    candidates = []
    scores = []

    # Try to load previous best
    try:
        base = height_sequence_1
        arr = np.asarray(base, dtype=np.float64).ravel()
        if arr.size == n and np.all(np.isfinite(arr)) and np.all(arr >= 0):
            a0 = _normalize(arr)
            c0 = _conv_self(a0, L, m)
            s0 = _score(c0, n)
            candidates.append(a0)
            scores.append(s0)
            print(f"Loaded previous best: score={s0:.6f}")
    except (NameError, Exception) as e:
        print(f"No previous best available: {e}")

    # Generate shape priors
    priors = _shape_priors(n, rng)
    for p in priors:
        c = _conv_self(p, L, m)
        s = _score(c, n)
        candidates.append(p)
        scores.append(s)

    # Sort by score and pick best few as starting points
    order = np.argsort(scores)
    print(f"Initial best score: {scores[order[0]]:.6f}")

    best_a = candidates[order[0]].copy()
    best_score = scores[order[0]]

    # Phase 1: Multi-start optimization (20% of time)
    t1 = 0.20 * budget
    best_start = start_time + t1
    top_starts = [candidates[order[i]] for i in range(min(4, len(order)))]

    for si, sa in enumerate(top_starts):
        if time.monotonic() > best_start - 2:
            break
        lim = (best_start - time.monotonic()) / max(1, len(top_starts) - si)
        if lim < 2:
            break
        out_a, out_score = _optimize_soft(sa.copy(), L, m, n, rng, lim, mode='adaptive')
        if out_score < best_score:
            best_score = out_score
            best_a = out_a
            print(f"Phase1 start {si}: score={out_score:.6f}")

    # Phase 2: Main optimization with dual tracking (30%)
    t2 = 0.30 * budget
    start2 = time.monotonic()
    out_a, out_score = _optimize_soft(best_a.copy(), L, m, n, rng, t2, mode='adaptive')
    if out_score < best_score:
        best_score = out_score
        best_a = out_a
    print(f"Phase2 (soft): score={best_score:.6f}")

    out_a, out_score = _optimize_dual(best_a.copy(), L, m, n, rng, 0.15 * budget)
    if out_score < best_score:
        best_score = out_score
        best_a = out_a
    print(f"Phase2 (dual): score={best_score:.6f}")

    # Phase 3: Extended soft optimization from multiple angles (25%)
    t3 = 0.25 * budget
    start3 = time.monotonic()
    while time.monotonic() < start3 + t3:
        rem = start3 + t3 - time.monotonic()
        if rem < 5:
            break
        # Alternate between soft and top-K modes
        mode = 'soft' if int((time.monotonic() - start3) % 10) < 5 else 'adaptive'
        lim = min(60.0, rem * 0.6)
        out_a, out_score = _optimize_soft(best_a.copy(), L, m, n, rng, lim, mode=mode)
        if out_score < best_score - 1e-13:
            best_score = out_score
            best_a = out_a
            print(f"Phase3 iter: score={out_score:.6f}")
        else:
            # Try a perturbation
            if rng.random() < 0.3:
                prior = _random_prior(n, rng, style=int(rng.integers(0, 5)))
                mix = rng.uniform(0.1, 0.4)
                probe = _normalize((1 - mix) * best_a + mix * prior)
                out_a2, out_score2 = _optimize_soft(probe, L, m, n, rng, min(30, rem * 0.3))
                if out_score2 < best_score:
                    best_score = out_score2
                    best_a = out_a2
                    print(f"Phase3 probe: score={out_score2:.6f}")

    # Phase 4: Polish (15%)
    t4 = 0.15 * budget
    rem4 = hard_deadline - time.monotonic()
    lim4 = min(t4, max(5, rem4))
    out_a, out_score = _polish(best_a.copy(), L, m, n, lim4)
    if out_score < best_score:
        best_score = out_score
        best_a = out_a
    print(f"Phase4 (polish): score={best_score:.6f}")

    # Phase 5: Final micro-polish and exact evaluation
    rem5 = hard_deadline - time.monotonic()
    if rem5 > 10:
        out_a, out_score = _polish(best_a.copy(), L, m, n, min(30, rem5 - 5))
        if out_score < best_score:
            best_score = out_score
            best_a = out_a

    # Exact evaluation
    seq = [float(x) for x in best_a]
    exact = _evaluate_exact(seq)
    print(f"Final exact score: {exact:.6f}")

    # Try a few more exact evaluations with slight perturbations
    if time.monotonic() < hard_deadline - 3:
        for _ in range(5):
            if time.monotonic() > hard_deadline - 2:
                break
            pert = best_a.copy()
            idx = rng.integers(0, n, size=10)
            pert[idx] *= (1 + rng.uniform(-0.02, 0.02, 10))
            pert = _normalize(pert)
            seq2 = [float(x) for x in pert]
            exact2 = _evaluate_exact(seq2)
            if exact2 < exact:
                exact = exact2
                best_a = pert
                seq = seq2
                print(f"  Perturbation improved: {exact:.6f}")

    return [float(x) for x in best_a]
