    a_new = _normalize(a_new)
    return a_new, True


def _harsh_delete(a, fraction, rng):
    n = len(a)
    mask = rng.random(n) > fraction
    a_new = a * mask
    if a_new.sum() < 1e-10:
        return _normalize(a)
    return _normalize(a_new)


def _smooth_perturb(a, rng, sigma=0.3):
    noise = rng.normal(0.0, sigma, len(a))
    noise = np.convolve(noise, np.ones(11) / 11.0, mode="same")
    a_new = a * (1.0 + noise)
    return _normalize(a_new)


def _optimize_phase(a, n, L_conv, L_grad, time_limit, rng, temp0, temp1,
                    step0, phase_name=""):
    deadline = time.monotonic() + time_limit
    a = _normalize(a).copy()
    c = _fft_convolve(a, L_conv)
    best_a = a.copy()
    best_score = _score_from_conv(c, n)
    
    temp = temp0
    step = step0
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 0.5:
            break
        
        c = _fft_convolve(a, L_conv)
        cur_score = _score_from_conv(c, n)
        
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 100:
            r = rng.random()
            if r < 0.4:
                a = _harsh_delete(best_a, rng.uniform(0.2, 0.5), rng)
            elif r < 0.7:
                prior = _random_prior(n, rng)
                mix = rng.uniform(0.2, 0.5)
                a = _normalize((1 - mix) * best_a + mix * prior)
            else:
                a = _smooth_perturb(best_a, rng, rng.uniform(0.2, 0.5))
            no_improve = 0
            step = step0 * 0.5
            temp = temp0
        
        a_new, ok = _subgradient_step(a, c, n, L_conv, L_grad, temp, step, 'soft')
        if not ok:
            break
        
        c_new = _fft_convolve(a_new, L_conv)
        sc_new = _score_from_conv(c_new, n)
        
        if sc_new < cur_score - 1e-15:
            a = a_new
            step = min(2.0, step * 1.1)
        else:
            step = max(0.01, step * 0.7)
            c2 = _fft_convolve(0.5 * (a + a_new) / _normalize(0.5 * (a + a_new)), L_conv)
            sc2 = _score_from_conv(c2, n)
            if sc2 < cur_score - 1e-15:
                a = _normalize(0.5 * (a + a_new))
                step = max(0.01, step * 0.8)
        
        it += 1
        if it % 20 == 0:
            frac = min(1.0, it / max(1, int(time_limit * 5)))
            temp = temp0 + (temp1 - temp0) * frac
    
    return best_a, best_score


def _polish(a, n, L_conv, L_grad, time_limit, rng):
    deadline = time.monotonic() + time_limit
    a = _normalize(a).copy()
    c = _fft_convolve(a, L_conv)
    best_a = a.copy()
    best_score = _score_from_conv(c, n)
    step = 0.1
    
    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 0.2:
            break
        
        c = _fft_convolve(a, L_conv)
        cur_score = _score_from_conv(c, n)
        
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
        
        for k in [10, 20, 50, 100]:
            weights = _hard_max_weights(c, k)
            grad = _compute_grad(a, weights, L_grad)
            grad = np.maximum(grad, 0.0)
            gmax = grad.max()
            if gmax < 1e-300:
                continue
            grad_norm = grad / gmax
            for s in [step, step * 0.5, step * 0.2]:
                a_new = a * np.exp(-s * grad_norm)
                a_new = _normalize(a_new)
                c_new = _fft_convolve(a_new, L_conv)
                sc_new = _score_from_conv(c_new, n)
                if sc_new < cur_score - 1e-15:
                    a = a_new
                    cur_score = sc_new
                    break
        
        step = max(0.001, step * 0.98)
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    try:
        budget = float(budget_s)
    except (ValueError, TypeError):
        budget = 2400.0
    if not np.isfinite(budget) or budget < 10.0:
        budget = 2400.0
    
    start_time = time.monotonic()
    deadline = start_time + budget - 15.0
    
    rng = np.random.default_rng(seed)
    
    try:
        parent = np.array(height_sequence_1, dtype=np.float64).ravel()
    except (NameError, Exception):
        parent = None
    
    if parent is not None and len(parent) > 100:
        n = len(parent)
    else:
        n = 6121
    
    L_conv = 1 << (2 * n - 1).bit_length()
    L_grad = 1 << (3 * n - 2).bit_length()
    
    # Build initial candidates
    candidates = []
    if parent is not None and len(parent) == n:
        p = np.maximum(parent, 0.0)
        if p.sum() > 0.01:
            candidates.append(_normalize(p))
    
    inits = _structured_inits(n, rng)
    candidates.extend(inits)
    
    # Score all candidates
    best_a = None
    best_score = float('inf')
    
    for a in candidates:
        c = _fft_convolve(a, L_conv)
        sc = _score_from_conv(c, n)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
    
    print(f"Initial best score: {best_score:.6f} (n={n}, time={time.monotonic()-start_time:.1f}s)")
    
    # Phase 1: Exploration with multiple diverse starts
    phase1_time = min(0.30 * budget, time.monotonic() + 0.30 * budget - deadline - 60)
    phase1_time = max(30.0, min(0.30 * budget, deadline - time.monotonic() - 60))
    phase1_deadline = time.monotonic() + phase1_time
    
    explore_results = []
    while time.monotonic() < phase1_deadline:
        rem = phase1_deadline - time.monotonic()
        if rem < 15.0:
            break
        
        if rng.random() < 0.6 and best_a is not None:
            a_start = best_a.copy()
            if rng.random() < 0.5:
                a_start = _harsh_delete(a_start, rng.uniform(0.15, 0.4), rng)
            else:
                a_start = _smooth_perturb(a_start, rng, rng.uniform(0.3, 0.7))
        else:
            a_start = _random_prior(n, rng)
        
        t_lim = min(60.0, rem - 5.0)
        if t_lim < 5.0:
            break
        
        temp0 = rng.uniform(0.5, 2.0)
        temp1 = rng.uniform(3.0, 15.0)
        step0 = rng.uniform(0.3, 0.8)
        
        a_res, sc_res = _optimize_phase(a_start, n, L_conv, L_grad, t_lim, rng,
                                         temp0, temp1, step0, "explore")
        explore_results.append((a_res, sc_res))
        
        if sc_res < best_score - 1e-12:
            best_score = sc_res
            best_a = a_res.copy()
            print(f"  Explore improvement: {best_score:.6f} (t={time.monotonic()-start_time:.0f}s)")
    
    print(f"After exploration: {best_score:.6f} (time={time.monotonic()-start_time:.1f}s)")
    
    # Phase 2: Refinement on best candidate
    phase2_time = min(0.45 * budget, deadline - time.monotonic() - 40)
    phase2_time = max(30.0, phase2_time)
    phase2_deadline = time.monotonic() + phase2_time
    
    while time.monotonic() < phase2_deadline:
        rem = phase2_deadline - time.monotonic()
        if rem < 20.0:
            break
        
        t_chunk = min(120.0, rem - 10.0)
        if t_chunk < 10.0:
            break
        
        temp0 = 0.3 + 0.5 * rng.random()
        temp1 = 2.0 + 3.0 * rng.random()
        step0 = 0.2 + 0.3 * rng.random()
        
        a_res, sc_res = _optimize_phase(best_a, n, L_conv, L_grad, t_chunk, rng,
                                         temp0, temp1, step0, "refine")
        if sc_res < best_score - 1e-12:
            best_score = sc_res
            best_a = a_res.copy()
            print(f"  Refine improvement: {best_score:.6f} (t={time.monotonic()-start_time:.0f}s)")
    
    print(f"After refinement: {best_score:.6f} (time={time.monotonic()-start_time:.1f}s)")
    
    # Phase 3: Additional exploration from explore results
    phase3_deadline = time.monotonic() + max(20.0, min(0.10 * budget, deadline - time.monotonic() - 40))
    
    explore_results.sort(key=lambda x: x[1])
    for a_exp, sc_exp in explore_results[:5]:
        if time.monotonic() > phase3_deadline:
            break
        rem = phase3_deadline - time.monotonic()
        if rem < 15.0:
            break
        a_res, sc_res = _optimize_phase(a_exp, n, L_conv, L_grad, min(40.0, rem - 5), rng,
                                         0.2, 2.0, 0.15, "explore2")
        if sc_res < best_score - 1e-12:
            best_score = sc_res
            best_a = a_res.copy()
            print(f"  Phase3 improvement: {best_score:.6f}")
    
    # Phase 4: Polishing with hard max
    polish_time = max(10.0, min(0.15 * budget, deadline - time.monotonic() - 10))
    a_final, sc_final = _polish(best_a, n, L_conv, L_grad, polish_time, rng)
    if sc_final < best_score - 1e-12:
        best_score = sc_final
        best_a = a_final.copy()
        print(f"  Polish improvement: {best_score:.6f}")
    
    print(f"Final internal score: {best_score:.6f} (time={time.monotonic()-start_time:.1f}s)")
    
    # Final evaluation with exact method
    rem = deadline - time.monotonic()
    a_final = _normalize(best_a)
    seq = [float(x) for x in a_final]
    
    # Verify with evaluate_sequence
    try:
        val = evaluate_sequence(seq)
        print(f"Exact evaluation: {val:.6f}")
    except NameError:
        # Fallback: compute manually
        arr = np.array(seq)
        conv = np.convolve(arr, arr)
        s = arr.sum()
        val = 2.0 * n * conv.max() / (s * s)
        print(f"Manual evaluation: {val:.6f}")
    except Exception as e:
        print(f"Eval error: {e}")
        val = float('inf')
    
    # If evaluation failed, try a safe fallback
    if not np.isfinite(val):
        seq = [1.0 / n] * n
    
    return seq
