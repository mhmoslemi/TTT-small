- Constraints: (f*g)[k] <= M for k = 0, ..., 2n-2 (2n-1 constraints)
- Objective: maximize sum(g)

"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.

Strategy:
1. Warm-start from height_sequence_1 if available.
2. Aggressive projected-gradient optimization with top-k active-set smooth max.
3. Adaptive step sizes with backtracking line search.
4. Periodic random restarts to escape local minima.
5. Multi-scale: try n=8192 and n=16384, pick best.
6. Final polish with very sharp max approximation.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def fft_convolution(a):
    """Linear convolution a*a via FFT, clipped to non-negative."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def active_set_smooth_max(b, beta, k=1):
    """
    Smooth max using top-k active set.
    Returns (smooth_max_value, weight_vector_on_b).
    """
    n = b.size
    if n == 0:
        return float('inf'), np.empty(0, dtype=np.float64)
    if k >= n:
        k = n
    m = float(np.max(b))
    if not np.isfinite(m) or m <= 0:
        w = np.ones(n) / n
        return m, w
    if k == 1:
        idx = int(np.argmax(b))
        return m, np.eye(1)[0, 0] if False else _make_sparse_w(n, idx, 1.0)

    top_idx = np.argpartition(b, -k)[-k:]
    top_vals = b[top_idx]
    tm = float(np.max(top_vals))
    x = beta * (top_vals - tm)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0 or not np.isfinite(Z):
        e = np.zeros(k)
        e[int(np.argmax(top_vals))] = 1.0
        Z = 1.0
    w_topk = e / Z
    G = tm + float(np.log(Z)) / beta
    w = np.zeros(n, dtype=np.float64)
    w[top_idx] = w_topk
    return float(G), w


def _make_sparse_w(n, idx, val):
    w = np.zeros(n, dtype=np.float64)
    w[idx] = val
    return w


def fast_score(p):
    """Score = 2*n*max(conv)/sum^2, with sum=1 normalization assumed."""
    n = p.size
    if n == 0:
        return float('inf')
    S = float(np.sum(p))
    if S <= 0 or not np.isfinite(S):
        return float('inf')
    b = fft_convolution(p)
    M = float(np.max(b))
    if not np.isfinite(M):
        return float('inf')
    return float(2.0 * n * M / (S * S))


def compute_grad(p, w):
    """
    Gradient of smooth_max(conv(p,p)) w.r.t. p.
    grad_i = 2 * sum_j w_j * p_{i+j} (cross-correlation of p with w).
    """
    n = p.size
    m = w.size
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    # s_i = sum_j w_j * p_{i+j} for i=0..n-1
    s = C[m - n:m][::-1].copy()
    return 2.0 * np.maximum(s, 0.0)


def project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    nv = v.size
    if nv == 0:
        return v
    if target <= 0:
        return np.zeros_like(v)
    if not np.all(np.isfinite(v)):
        return np.full(nv, target / nv, dtype=np.float64)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0 or not np.isfinite(s):
        return np.full(nv, target / nv, dtype=np.float64)
    q *= target / s
    return q


def normalize_simplex(p):
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / p.size, dtype=np.float64)
    return p / s


def refine_sequence(p0, deadline, rng, beta=200000.0, k_active=5, max_iter=500000):
    """
    Aggressive projected-gradient refinement with top-k active set.
    Returns (best_p, best_score).
    """
    p = normalize_simplex(p0)
    n = p.size
    if n == 0:
        return p, float('inf')

    best_p = p.copy()
    best_score = fast_score(p)
    print(f"  refine start: score={best_score:.8f}, n={n}")

    step = 0.5
    no_improve = 0
    it = 0
    restart_interval = 3000
    restart_count = 0

    while it < max_iter and time.monotonic() < deadline:
        it += 1

        b = fft_convolution(p)
        G, w = active_set_smooth_max(b, beta, k_active)
        if not np.isfinite(G):
            break

        cur_smooth = n * G
        m_true = float(np.max(b))
        cur_score = 2.0 * n * m_true

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 1000 == 0:
            print(f"  refine it={it} best={best_score:.8f} step={step:.3e} k={k_active} restarts={restart_count}")

        if no_improve > 5000:
            # Try increasing k_active for a different perspective
            if k_active < 20:
                k_active = min(20, k_active * 2)
                no_improve = 0
                print(f"  -> increased k_active to {k_active}")
            else:
                # Random restart
                restart_count += 1
                noise = 0.01 * rng.standard_normal(n)
                p = normalize_simplex(p + noise)
                no_improve = 0
                step = 0.5
                if restart_count > 20:
                    break
            continue

        grad = compute_grad(p, w)
        grad = grad - float(np.mean(grad))
        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-15:
            break

        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005):
            alpha = step * mult
            q = project_simplex(p - alpha * grad, 1.0)
            qb = fft_convolution(q)
            G2, _ = active_set_smooth_max(qb, beta, k_active)
            if not np.isfinite(G2):
                continue
            if n * G2 < cur_smooth - 1e-13:
                p = q
                cur_smooth = n * G2
                accepted = True
                m2 = float(np.max(qb))
                cur_score = 2.0 * n * m2
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_p = p.copy()
                    no_improve = 0
                break

        if not accepted:
            step *= 0.7
            if step < 1e-8:
                break
        else:
            step = min(2.0, step * 1.3)

    return best_p, best_score


def gaussian_start(n, sigma_frac, rng):
    """Gaussian-shaped sequence."""
    center = (n - 1) / 2.0
    sigma = sigma_frac * n
    x = np.arange(n, dtype=np.float64)
    p = np.exp(-0.5 * ((x - center) / sigma) ** 2)
    # Add small random perturbation
    p += 0.01 * rng.standard_normal(n)
    p = np.maximum(p, 0.0)
    return p / p.sum()


def structured_start(n, rng):
    """Various structured starting profiles."""
    starts = []
    # Uniform
    starts.append(np.ones(n) / n)
    # Gaussian widths
    for sf in (0.15, 0.25, 0.35, 0.5):
        starts.append(gaussian_start(n, sf, rng))
    # Two-peak
    p = np.ones(n) / n
    p[:n//4] *= 1.5
    p[3*n//4:] *= 1.5
    p[n//4:3*n//4] *= 0.7
    starts.append(p / p.sum())
    # Cosine taper
    x = np.linspace(0, np.pi, n)
    p = np.sin(x) ** 2 + 0.1
    starts.append(p / p.sum())
    # Random low-freq
    size = 1
    while size < n:
        size <<= 1
    zc = (rng.normal(size=size//2+1) + 1j*rng.normal(size=size//2+1))
    z = np.real(np.fft.irfft(zc, size))[:n]
    z = 0.5 * np.tanh(z)
    p = np.exp(z)
    starts.append(p / p.sum())
    return starts
