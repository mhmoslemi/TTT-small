def mws_direction(f, L, m, n, prior, inner=20, eta_start=2.0, eta_end=50.0):
    f = normalize(f)
    prior = normalize(prior)
    F = np.fft.rfft(f, L)
    Frev = np.fft.rfft(f[::-1], L)
    x = prior.copy()
    y = np.full(m, 1.0/float(m))
    avg = np.zeros_like(x)
    count = 0
    logprior = np.log(np.maximum(prior, 1e-300))
    
    for it in range(inner):
        frac = 0.0 if inner <= 1 else float(it)/float(inner-1)
        eta = eta_start + (eta_end - eta_start) * frac
        
        X = np.fft.rfft(x, L)
        u = np.fft.irfft(F * X, L)[:m]
        np.maximum(u, 0.0, out=u)
        vu = n * u
        vma = float(np.max(vu))
        z = eta * (vu - vma)
        z = np.clip(z, -50.0, 0.0)
        y = np.exp(z)
        ys = float(y.sum())
        if not np.isfinite(ys) or ys <= 0.0:
            y.fill(1.0/float(m))
        else:
            y /= ys
        
        avg += x
        count += 1
        
        Y = np.fft.rfft(y, L)
        s = np.fft.irfft(Y * Frev, L)[(n-1):m]
        np.maximum(s, 0.0, out=s)
        ps = n * s
        sma = float(np.max(ps))
        zx = -eta * (ps - sma) + 0.8 * logprior
        zx -= float(np.max(zx))
        zx = np.clip(zx, -50.0, 0.0)
        x = np.exp(zx)
        xs = float(x.sum())
        if not np.isfinite(xs) or xs <= 0.0:
            x = prior.copy()
        else:
            x /= xs
    
    if count == 0:
        return x
    return normalize(0.5 * (avg/count) + 0.5 * x)

"""
Search for non-negative step heights minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Combines refined multiplicative updates with L-BFGS soft-objective polishing,
active-set focused gradients, and multi-phase temperature annealing.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize


def _normalize(a, eps=1e-300):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a.fill(1.0)
        s = float(a.sum())
    a /= s
    a = np.maximum(a, eps)
    a /= float(a.sum())
    return a


def _conv_self(a, L, m):
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _score_from_conv(c, n):
    return 2.0 * n * float(np.max(c))


def _score(a, L, m, n):
    return _score_from_conv(_conv_self(a, L, m), n)


def _gradient_conv(a, L, m, n, w):
    W = np.fft.rfft(w, L)
    Frev = np.fft.rfft(a[::-1], L)
    g = np.fft.irfft(W * Frev, L)[(n - 1):m]
    np.maximum(g, 0.0, out=g)
    g *= 2.0
    return g


def _random_prior(n, rng):
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0
    z = np.zeros(n)
    z += rng.normal(0.0, 0.35)
    z += rng.normal(0.0, 0.55) * x
    z += rng.normal(0.0, 0.55) * (x * x - 1.0 / 3.0)
    kmax = int(rng.integers(2, 8))
    for k in range(1, kmax + 1):
        amp = rng.uniform(0.05, 0.45) / max(1.0, math.sqrt(float(k)))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos((2.0 * np.pi * k) * t + ph)
    noise = rng.normal(0.0, 0.15, n)
    if n > 5:
        noise = np.convolve(noise, np.ones(5, dtype=np.float64) / 5.0, mode="same")
    z += noise
    p = np.exp(z - float(np.max(z)))
    p = np.maximum(p, 1e-300)
    return _normalize(p)

    if time_limit is None or time_limit <= 0:
        return _normalize(a), _score(_normalize(a), L, m, n)

    deadline = time.monotonic() + float(time_limit)
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score(a, L, m, n)
    temp = float(temp0)
    eta = 0.08
    no_accept = 0
    it = 0

    while time.monotonic() < deadline:
        c = _conv_self(a, L, m)
        current_score = _score_from_conv(c, n)

        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            no_accept = 0

        if no_accept > 80:

            mix = 0.05 + 0.25 * float(rng.random())
            a = _normalize((1.0 - mix) * best_a + mix * prior)
            no_accept = 0
            temp = float(temp0)
            eta = 0.08
            continue

        v = n * c
        w = v - float(np.max(v))
        w *= temp
        w = np.clip(w, -700.0, 0.0)
        e = np.exp(w)
        es = float(e.sum())
        if not np.isfinite(es) or es <= 0:
            e = np.ones_like(v)
            es = float(e.sum())
        w = e / es

        g = _gradient_conv(a, L, m, n, w)

        d = g / gmax

                accepted = True

            eta = min(0.35, eta * 1.04)

            no_accept += 1
            eta = max(0.01, eta * 0.82)

def _active_set_search(a, L, m, n, time_limit):

    if time_limit is None or time_limit <= 0:
        return _normalize(a), _score(_normalize(a), L, m, n)

    deadline = time.monotonic() + float(time_limit)
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score(a, L, m, n)
    eta = 0.04
    it = 0

    while time.monotonic() < deadline:
        c = _conv_self(a, L, m)
        current_score = _score_from_conv(c, n)

        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()

        K = min(m, 3 + it // 50)
        top_idx = np.argpartition(c, -K)[-K:]
        top_vals = c[top_idx]
        shift = top_vals - float(np.max(top_vals))
        shift *= 20.0
        shift = np.clip(shift, -50.0, 0.0)
        w = np.zeros(m)
        w[top_idx] = np.exp(shift)
        ws = float(w.sum())
        if ws <= 0 or not np.isfinite(ws):
            w = np.ones(m) / m
        else:
            w /= ws

        g = _gradient_conv(a, L, m, n, w)
        gmax = float(g.max())
        if not np.isfinite(gmax) or gmax <= 0:
            break
        d = g / gmax

        accepted = False
        for step in (eta, 0.5 * eta, 0.2 * eta):
            if step < 1e-10:
                continue
            a2 = a * np.exp(-step * d)
            a2 = _normalize(a2)

            eta = min(0.15, eta * 1.02)
        else:
            eta = max(0.005, eta * 0.85)

def _lbfgs_refine(a, L, m, n, T, maxiter=300):
    """L-BFGS on soft log-sum-exp objective in log-parameterization."""
    a0 = _normalize(a)
    b0 = np.log(np.maximum(a0, 1e-15))
    b0 -= float(np.max(b0))

    def obj_and_grad(b):
        scale = float(np.max(b))
        logits = b - scale
        logits = np.clip(logits, -700.0, 50.0)
        exp_logits = np.exp(logits)
        exp_sum = float(exp_logits.sum())
        if exp_sum <= 0 or not np.isfinite(exp_sum):
            exp_logits = np.ones_like(b) / n
        a_val = exp_logits / exp_sum

        c = _conv_self(a_val, L, m)
        v = T * c
        vmax = float(np.max(v))
        z = v - vmax
        z = np.clip(z, -700.0, 0.0)
        e = np.exp(z)
        s = float(e.sum())
        if s <= 0 or not np.isfinite(s):
            return 2.0 * n * float(np.max(c)), np.zeros_like(b)

        obj = (2.0 * n / T) * (vmax + math.log(s))
        w = e / s

        g_a = _gradient_conv(a_val, L, m, n, w)
        dot = float(np.dot(a_val, g_a))
        g_b = a_val * (g_a - dot)

        return obj, g_b

    result = minimize(
        obj_and_grad, b0, method='L-BFGS-B',
        options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10, 'maxls': 50}
    )
    b_fin = result.x
    scale = float(np.max(b_fin))
    a_fin = np.exp(b_fin - scale)
    a_fin = np.maximum(a_fin, 0.0)
    a_fin = _normalize(a_fin)
    return a_fin, _score(a_fin, L, m, n)


def _mws_direction(f, L, m, n, prior, inner=20, eta_start=3.0, eta_end=60.0):
    f = _normalize(f)
    prior = _normalize(prior)
    F = np.fft.rfft(f, L)
    Frev = np.fft.rfft(f[::-1], L)
    x = prior.copy()
    y = np.full(m, 1.0 / float(m))
    avg = np.zeros_like(x)
    logprior = np.log(np.maximum(prior, 1e-300))

    for it in range(inner):
        frac = 0.0 if inner <= 1 else float(it) / (inner - 1)
        eta = eta_start + (eta_end - eta_start) * frac

        X = np.fft.rfft(x, L)
        u = np.fft.irfft(F * X, L)[:m]
        np.maximum(u, 0.0, out=u)
        vu = n * u
        vma = float(np.max(vu))
        z = eta * (vu - vma)
        z = np.clip(z, -50.0, 0.0)
        y = np.exp(z)
        ys = float(y.sum())
        if not np.isfinite(ys) or ys <= 0:
            y.fill(1.0 / float(m))
        else:
            y /= ys

        avg += x

        Y = np.fft.rfft(y, L)
        s = np.fft.irfft(Y * Frev, L)[(n - 1):m]
        np.maximum(s, 0.0, out=s)
        ps = n * s
        sma = float(np.max(ps))
        zx = -eta * (ps - sma) + 0.7 * logprior
        zx -= float(np.max(zx))
        zx = np.clip(zx, -50.0, 0.0)
        x = np.exp(zx)
        xs = float(x.sum())
        if not np.isfinite(xs) or xs <= 0:
            x = prior.copy()
        else:
            x /= xs

    return _normalize(0.5 * (avg / inner) + 0.5 * x)


def _line_search(a, b, L, m, n, base_score):
    best_a = a
    best_score = base_score
    for lam in (0.02, 0.05, 0.10, 0.15, 0.25, 0.40, 0.60, 1.0):
        h = _normalize((1.0 - lam) * a + lam * b)
        sc = _score(h, L, m, n)
        if sc < best_score:
            best_score = sc
            best_a = h
    return best_a, best_score


def _internal_exact(seq):
    if not isinstance(seq, list) or not seq:
        return float('inf')
    for x in seq:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return float('inf')
        if np.isnan(x) or np.isinf(x):
            return float('inf')
    a = [0.0 if float(x) < 0.0 else min(1000.0, float(x)) for x in seq]
    n = len(a)
    s = sum(a)
    if s < 0.01:
        return float('inf')
    b = np.convolve(a, a)
    mb = max(b)
    return float(2.0 * n * mb / (s * s))


def _exact_score(seq):
    try:
        val = evaluate_sequence(seq)
    except NameError:
        val = _internal_exact(seq)
    except Exception:
        return float('inf')
    if val is None:
        return float('inf')
    return float(val)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search function. Returns best sequence found within budget."""
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 1.0:
        budget = max(10.0, budget if np.isfinite(budget) else 2400.0)

    start = time.monotonic()
    hard_deadline = start + budget - 15.0

    rng = np.random.default_rng(seed)
    n = 6121
    L = 1 << (2 * n - 2).bit_length()
    m = 2 * n - 1

    # Get initial candidate
    a_init = None
    try:
        base = height_sequence_1
        arr = np.asarray(base, dtype=np.float64).ravel()
        if arr.size == n and np.all(np.isfinite(arr)):
            a_init = _normalize(arr)
    except (NameError, Exception):
        pass

    if a_init is None:
        a_init = _random_prior(n, rng)

    best_score = _score(a_init, L, m, n)
    best_a = a_init.copy()
    print(f"Initial score: {best_score:.8f}")

    # Phase 1: Multiplicative search (main workhorse)
    rem = hard_deadline - time.monotonic()
    if rem > 100:
        t1 = min(800.0, 0.40 * rem)
        print(f"Phase 1: multiplicative search for {t1:.0f}s")
        a1, s1 = _multiplicative_search(a_init, L, m, n, rng, t1, temp0=0.5, temp_max=200.0)
        if s1 < best_score:
            best_score = s1
            best_a = a1
        print(f"  After Phase 1: {s1:.8f}")

    # Phase 2: MWS-based direction finding + line search
    rem = hard_deadline - time.monotonic()
    if rem > 200:
        t2 = min(400.0, 0.25 * rem)
        deadline2 = time.monotonic() + t2
        print(f"Phase 2: MWS search for {t2:.0f}s")
        while time.monotonic() < deadline2:
            prior = _random_prior(n, rng)
            x = _mws_direction(best_a, L, m, n, prior, inner=25, eta_start=3.0, eta_end=80.0)
            cand, sc = _line_search(best_a, x, L, m, n, best_score)
            if sc < best_score - 1e-12:
                best_score = sc
                best_a = cand
                print(f"  MWS improvement: {sc:.8f}")

    # Phase 3: Active set refinement
    rem = hard_deadline - time.monotonic()
    if rem > 100:
        t3 = min(200.0, 0.20 * rem)
        print(f"Phase 3: active-set for {t3:.0f}s")
        a3, s3 = _active_set_search(best_a, L, m, n, t3)
        if s3 < best_score:
            best_score = s3
            best_a = a3
        print(f"  After Phase 3: {s3:.8f}")

    # Phase 4: L-BFGS refinement with increasing T
    rem = hard_deadline - time.monotonic()
    if rem > 60:
        t4 = min(150.0, 0.30 * rem)
        deadline4 = time.monotonic() + t4
        print(f"Phase 4: L-BFGS refinement for {t4:.0f}s")
        for T in [50.0, 100.0, 200.0, 400.0]:
            if time.monotonic() > deadline4:
                break
            a4, s4 = _lbfgs_refine(best_a, L, m, n, T=T, maxiter=200)
            if s4 < best_score:
                best_score = s4
                best_a = a4
                print(f"  L-BFGS T={T}: {s4:.8f}")

    # Phase 5: Final multiplicative polish
    rem = hard_deadline - time.monotonic()
    if rem > 50:
        t5 = min(120.0, 0.50 * rem)
        print(f"Phase 5: final multiplicative for {t5:.0f}s")
        a5, s5 = _multiplicative_search(best_a, L, m, n, rng, t5, temp0=20.0, temp_max=400.0)
        if s5 < best_score:
            best_score = s5
            best_a = a5
        print(f"  After Phase 5: {s5:.8f}")

    # Phase 6: Final active set
    rem = hard_deadline - time.monotonic()
    if rem > 30:
        t6 = min(60.0, 0.60 * rem)
        a6, s6 = _active_set_search(best_a, L, m, n, t6)
        if s6 < best_score:
            best_score = s6
            best_a = a6
        print(f"  After Phase 6: {s6:.8f}")

    # Final exact evaluation
    seq = [float(x) for x in best_a]
    exact_val = _exact_score(seq)
    print(f"Internal best: {best_score:.8f}, Exact: {exact_val:.8f}")

    if not np.isfinite(exact_val):
        seq = [1.0 / float(n)] * n
        exact_val = _exact_score(seq)
        print(f"Fallback exact: {exact_val:.8f}")

    return seq
