def _next_pow2(m):
    """Smallest power of two >= m."""
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return float(2.0 * arr.size * mx / (s * s))

        if np.all(np.abs(g) < 1e-15):
            break
        
        # Normalize gradient for stable updates
        g_max = np.max(np.abs(g))
        if g_max < 1e-15:
            break
        g = g / g_max
        
        # Multiplicative update
        exponent = -lr * g
        exponent = np.clip(exponent, -0.5, 0.5)  # Conservative clipping
        arr = arr * np.exp(exponent)
        arr = np.clip(arr, 1e-30, None)  # Avoid underflow
        s = float(arr.sum())
        arr = arr / s
        
        # Periodic evaluation
        if it % eval_period == 0:
            sc = _score(arr)
            if sc < best_score:
                best = arr.copy()
                best_score = sc
                last_improve = it
                best_lr_for_best = lr
                lr = min(lr_max, lr * 1.03)
                if verbose and it % (eval_period * 10) == 0:
                    print(f"  [opt] it={it} score={sc:.8f} lr={lr:.2f}")
            else:
                # No improvement - reduce lr or reset
                if it - last_improve > eval_period * 5:
                    lr = max(lr_min, lr * 0.7)
                    if sc > best_score * 1.5:
                        # Bad move - revert
                        arr = best.copy()
                else:
                    lr = max(lr_min, lr * 0.95)
        
        # Periodic annealing
        if it % anneal_period == 0 and lr > lr_min:
            noise = rng.normal(0, anneal_scale, size=n)
            arr = arr * np.exp(np.clip(noise, -0.3, 0.3))
            arr = np.clip(arr, 1e-30, None)
            arr = arr / float(arr.sum())
            anneal_scale *= 0.9  # Decay annealing
        
        # Convergence check
        if it - last_improve > eval_period * 20:
            if verbose:
                print(f"  [opt] Stagnation at it={it}, best_score={best_score:.8f}")
            # Try a more aggressive anneal
            noise = rng.normal(0, 0.05, size=n)
            arr = arr * np.exp(np.clip(noise, -0.5, 0.5))
            arr = np.clip(arr, 1e-30, None)
            arr = arr / float(arr.sum())
            last_improve = it  # Reset counter
            anneal_period = min(anneal_period * 2, 5000)
            if anneal_period > 3000:
                break
    
    return best, best_score


def _try_initialization(n, time_budget, candidates):
    """Try multiple initializations andreturnthe best."""
    best_a = None
    best_score = np.inf
    
    for name, init in candidates:
        if time.monotonic() - time_budget[0] > time_budget[1]:
            break
        sc = _score(init)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = init.copy()
    
    return best_a, best_score
