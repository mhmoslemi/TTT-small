def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

    if n == 0:
        return np.array([], dtype=np.float64)

    c = np.fft.irfft(fa * fa, N)

        return np.inf

    Compute the subgradient of max convex function at the given active lags.
    active_lags: array of indices in the full convolution (0 to 2n-2)

"""
Low-peak nonnegative step-function optimization using adaptive
multiplicative weight descent with multiple active set sizes and
cosine-annealed learning rate. Leverages the previous best solution
as a warm start and uses random restarts to escape local minima.
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a, N=None):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    if N is None:
        N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _score_from_conv(c, n, s=1.0):
    if c.size == 0:
        return np.inf
    mx = float(np.max(c[:n]))  # by symmetry, max of first n is max of all
    if not np.isfinite(mx) or mx < 0:
        mx = 0.0
    return float(2.0 * n * mx / (s * s))


def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    return _score_from_conv(c, arr.size, s)


def _compute_grad(a, active_indices, N=None):
    """
    Compute the subgradient of the max-convolution at the active indices.
    active_indices: indices in the full convolution (0 to 2n-2)
    """
    n = a.size
    grad = np.zeros(n, dtype=np.float64)
    K = active_indices.size
    for k in active_indices:
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        if j_max < j_min:
            continue
        j_range = np.arange(j_min, j_max + 1)
        grad[j_range] += 2.0 * a[k - j_range]
    grad /= float(K)
    return grad


def _optimize_single_K(a, K, max_time_s, N=None):
    """
    Run optimization with a fixed number of active lags K.
    Returns (best_a, best_score).
    """
    a = np.asarray(a, dtype=np.float64).copy()
    n = a.size
    if N is None:
        N = _next_pow2(2 * n - 1)
    
    s = float(a.sum())
    a = a / s
    
    best_a = a.copy()
    best_score = _internal_score(a)
    
    start_time = time.monotonic()
    deadline = start_time + max_time_s
    
    # Learning rate schedule: cosine annealing
    lr_init = n / 20.0
    lr_min = n / 50000.0
    total_plan_iters = int(max_time_s / (n * 2e-6))  # estimated iterations
    
    lr = lr_init
    it = 0
    last_improve_it = 0
    stall = 0
    
    while time.monotonic() < deadline:
        it += 1
        
        # Compute convolution
        fa = np.fft.rfft(a, N)
        c = np.fft.irfft(fa * fa, N)
        c = c[:2*n-1]
        
        max_c = float(np.max(c[:n]))  # by symmetry
        cur_score = 2.0 * n * max_c
        
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            last_improve_it = it
            stall = 0
        
        # Cosine annealing
        cos_factor = 0.5 * (1.0 + np.cos(np.pi * min(it, total_plan_iters) / max(1, total_plan_iters)))
        lr = lr_min + (lr_init - lr_min) * cos_factor
        
        # Find active lags
        if K >= n:
            active = np.arange(n, dtype=np.int64)
        else:
            active = np.argpartition(c[:n], -K)[-K:]
        
        # Compute gradient
        grad = _compute_grad(a, active, N)
        
        # Project to tangent space of simplex (remove mean)
        grad = grad - np.mean(grad)
        
        # Update
        exponent = -lr * grad
        exponent = np.clip(exponent, -1.5, 1.5)
        a = a * np.exp(exponent)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        new_s = float(a.sum())
        if not np.isfinite(new_s) or new_s < 0.01:
            a = best_a.copy()
            break
        a = a / new_s
        
        # Stall detection
        if it - last_improve_it > 5000:
            break
    
    return best_a, best_score
