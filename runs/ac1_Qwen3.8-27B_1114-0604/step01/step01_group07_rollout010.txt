def _fft_convolve(a, b=None):
    """Linear convolution a*b (or a*a if b is None) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)
    n, m = a.size, b.size
    if n == 0 or m == 0:
        return np.array([], dtype=np.float64)
    need = n + m - 1
    N = 1
    while N < need:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)
    return c[:need]


def _score(a):
    """Fast internal score matching evaluate_sequence."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    mx = c.max()
    return 2.0 * a.size * mx / (s * s)


def _soft_max_weights(c, T):
    """Compute softmax weights for the convolution peaks."""
    c_shifted = c - c.max()
    # Numerical stability: clip to avoid overflow
    c_shifted = np.clip(c_shifted, -1000.0, 0.0)
    exp_c = np.exp(T * c_shifted)
    w = exp_c / exp_c.sum()
    return w


def _gradient_from_weights(w, a):
    """
    Compute gradient of soft-max objective w.r.t. a.
    grad[j] = 2 * sum_l w[l+j] * a[l] for l in [0, n-1]
    Computed via FFT cross-correlation.
    """
    n = a.size
    N = 1
    while N < 2 * n:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    # Cross-correlation: sum_l a[l] * w[l+j]
    # This is ifft(conj(fa) * fw) for circular correlation
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    # xcorr[j] = sum_l a[l] * w[(l+j) mod N]
    # For our range (j < n, l < n), l+j < 2n <= N, so no wraparound
    grad = 2.0 * xcorr[:n]
    return grad


def _mirror_descent_step(a, grad, lr):
    """Mirror descent (multiplicative weights) step on the simplex."""
    # Project gradient to simplex tangent
    grad_proj = grad - grad.mean()
    # Multiplicative update
    update = -lr * grad_proj
    update = np.clip(update, -50.0, 50.0)  # Numerical stability
    a_new = a * np.exp(update)
    a_new = np.where(np.isfinite(a_new), a_new, 0.0)
    a_new = np.clip(a_new, 0.0, 1000.0)
    s = a_new.sum()
    if s < 1e-10:
        return a.copy()
    a_new = a_new / s
    return a_new


def _make_sine(n, amp, mode=1, phase=0.5):
    """Create anti-palindromic sinusoidal sequence."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.sin(float(mode) * theta)) / n
    a = np.clip(a, 0.0, 1000.0)
    a = a / a.sum()
    return a


def _make_multi_sine(n, amps, modes, phases):
    """Create multi-sinusoidal sequence."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64) / n
    for amp, mode, ph in zip(amps, modes, phases):
        theta = 2.0 * np.pi * (i + ph) / n
        a += amp * np.sin(float(mode) * theta) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a = a / s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _soft_max_gradient_descent(a, deadline, T_initial=50.0, T_final=500.0,
                                lr_initial=0.5, n_iterations=None):
    """
    Main optimization loop using soft-max gradient descent.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(best_a)
    
    n = a.size
    total_time = deadline - time.monotonic()
    
    if n_iterations is None:
        n_iterations = max(100, int(total_time / 0.05))  # Estimate: ~50ms per iter for n=2048
    
    it = 0
    last_improve = 0
    patience = 2000
    
    while time.monotonic() < deadline and it < n_iterations:
        it += 1
        
        # Anneal temperature
        progress = min(1.0, it / (0.5 * n_iterations))
        T = T_initial + (T_final - T_initial) * progress
        # Learning rate decay
        lr = lr_initial / (1.0 + 0.1 * it)
        
        # Compute convolution
        c = _fft_convolve(a)
        if c.size == 0:
            break
        
        # Compute soft-max weights
        w = _soft_max_weights(c, T)
        
        # Compute gradient
        grad = _gradient_from_weights(w, a)
        
        # Update
        a_new = _mirror_descent_step(a, grad, lr)
        
        # Evaluate periodically
        if it % 10 == 0:
            sc = _score(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                last_improve = it
                # Slightly increase lr on improvement
                lr_initial = min(2.0, lr_initial * 1.02)
            elif it - last_improve > patience:
                # Reset to best and reduce lr
                a = best_a.copy()
                lr_initial = max(0.05, lr_initial * 0.8)
                patience = max(500, patience // 2)
        
        a = a_new
    
    return best_a, best_score


def _refine_with_hard_max(a, deadline, max_iters=5000):
    """Refine using hard-max top-K gradient (similar to previous approach but improved)."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(best_a)
    
    n = a.size
    K = min(16, n)
    lr = 2.0
    it = 0
    last_improve = 0
    
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        
        c = _fft_convolve(a)
        if c.size == 0:
            break
        
        # Get top-K peaks
        if c.size <= K:
            active = np.arange(c.size)
            weights = np.ones(K) / K
        else:
            top_k = np.argpartition(c, -K)[-K:]
            top_c = c[top_k]
            # Soft weights within top-K
            T_local = 10.0
            top_c_shift = top_c - top_c.max()
            weights = np.exp(T_local * top_c_shift)
            weights = weights / weights.sum()
            active = top_k
        
        # Compute gradient from weighted top-K
        # grad[j] = 2 * sum_k w[k_idx] * a[k - j] for each active peak k
        # Use FFT: grad = 2 * cross_corr of a with weighted-delta
        w_vec = np.zeros(c.size, dtype=np.float64)
        for idx, k in enumerate(active):
            w_vec[k] = weights[idx]
        
        grad = _gradient_from_weights(w_vec, a)
        
        a_new = _mirror_descent_step(a, grad, lr)
        
        if it % 20 == 0:
            sc = _score(a_new)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                last_improve = it
                lr = min(10.0, lr * 1.01)
            else:
                if it - last_improve > 500:
                    a = best_a.copy()
                    lr = max(0.5, lr * 0.85)
        
        a = a_new
    
    return best_a, best_score

"""
Low-peak step-function optimization via soft-max mirror descent.

Algorithm:
1. Initialize with anti-palindromic sinusoidal sequence (tuned amplitude).
2. Optimize using mirror descent (multiplicative weights) with a log-sum-exp
   surrogate for the max of the self-convolution.
3. Gradient computed via FFT cross-correlation in O(n log n).
4. Final polish with hard-max top-K active set refinement.
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear auto-convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    """Compute the evaluation score (scale-invariant)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _grad_via_fft(w, a):
    """
    Compute grad[j] = 2 * sum_{l=0}^{n-1} a[l] * w[l+j]
    using FFT cross-correlation.
    w has length 2n-1 (the convolution length).
    """
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    # Cross-correlation: xcorr[j] = sum_l a[l] * w[l+j]
    # Using: ifft(conj(fft(a)) * fft(w)) gives circular cross-correlation
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]


def _make_sine(n, amp, phase=0.5):
    """Anti-palindromic sinusoidal: a[i] = (1 + amp*sin(2*pi*(i+phase)/n)) / n."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.sin(theta)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _softmax_weights(c, T):
    """Compute softmax(T*c) weights (numerically stable)."""
    c_max = np.max(c)
    c_shift = c - c_max
    c_shift = np.clip(c_shift, -50.0, 0.0)
    exp_c = np.exp(T * c_shift)
    s = exp_c.sum()
    if s < 1e-300:
        # Fallback: uniform over the max element
        n_max = np.sum(c == c_max)
        w = np.zeros(c.size)
        w[c == c_max] = 1.0 / n_max
        return w
    return exp_c / s


def _optimize_soft(a, deadline, T_start=50.0, T_end=500.0, lr0=1.0):
    """
    Soft-max mirror descent optimization.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(best_a)
    
    n = a.size
    it = 0
    last_improve = 0
    patience = 3000
    eval_interval = max(1, n // 2048)  # Evaluate every ~2048 steps adjusted
    
    # Estimate how many iterations we can do
    # Each iteration: 1 FFT convolve (O(n log n)) + 1 FFT cross-corr (O(n log n))
    # For n=2048: ~1ms per iteration, so ~2.4M iterations in 2400s (theoretical)
    # In practice, Python overhead limits us to maybe 1000-5000 per second
    
    while time.monotonic() < deadline:
        it += 1
        
        # Annealing
        t_frac = min(1.0, it / 50000.0)
        T = T_start + (T_end - T_start) * t_frac
        lr = lr0 / (1.0 + 0.05 * it)
        
        # Compute convolution
        c = _fft_convolve(a)
        if c.size == 0:
            break
        
        # Soft-max weights
        w = _softmax_weights(c, T)
        
        # Gradient
        grad = _grad_via_fft(w, a)
        
        # Project to simplex tangent
        grad_proj = grad - np.mean(grad)
        
        # Mirror descent update
        update = -lr * grad_proj
        update = np.clip(update, -20.0, 20.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s
        
        # Periodic evaluation
        if it % eval_interval == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
            elif it - last_improve > patience:
                # Restart from best
                a = best_a.copy()
                patience = max(500, patience * 2)
                lr0 = max(0.1, lr0 * 0.9)
    
    return best_a, best_score


def _optimize_hard(a, deadline, K=8, lr0=2.0):
    """
    Hard-max top-K active set refinement.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score(best_a)
    
    n = a.size
    it = 0
    last_improve = 0
    patience = 3000
    eval_interval = 5
    
    while time.monotonic() < deadline:
        it += 1
        lr = lr0 / (1.0 + 0.02 * it)
        
        c = _fft_convolve(a)
        if c.size == 0:
            break
        
        # Top-K peaks
        if c.size <= K:
            top_idx = np.arange(c.size)
        else:
            top_idx = np.argpartition(c, -K)[-K:]
        
        # Soft weights within top-K (for smoothness)
        top_vals = c[top_idx]
        T_local = 5.0
        top_shift = top_vals - top_vals.max()
        weights = np.exp(T_local * top_shift)
        weights /= weights.sum()
        
        # Build weight vector
        w = np.zeros(c.size, dtype=np.float64)
        for i, k in enumerate(top_idx):
            w[k] = weights[i]
        
        # Gradient
        grad = _grad_via_fft(w, a)
        
        # Update
        grad_proj = grad - np.mean(grad)
        update = -lr * grad_proj
        update = np.clip(update, -20.0, 20.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:

"""
Low-peak step-function optimization via soft-max mirror descent with FFT gradients.

Algorithm:
1. Initialize with anti-palindromic sinusoidal sequence (scanned over amplitude/length).
2. Optimize using mirror descent (multiplicative weights) with a log-sum-exp surrogate
   for the max of the self-convolution. Gradient computed via FFT cross-correlation in O(n log n).
3. Final polish with hard-max top-K active set refinement.
4. Adaptive learning rate and temperature annealing with restarts.
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear auto-convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    N = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * n - 1]


def _score(a):
    """Compute the evaluation score (scale-invariant, matches evaluate_sequence)."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return 2.0 * a.size * mx / (s * s)


def _grad_via_fft(w, a):
    """
    Compute grad[j] = 2 * sum_{l=0}^{n-1} a[l] * w[l+j]
    using FFT cross-correlation. w has length >= n (typically 2n-1).
    """
    n = a.size
    N = _next_pow2(2 * n)
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w[:N], N) if w.size > N else np.fft.rfft(w, N)
    xcorr = np.fft.irfft(np.conj(fa) * fw, N)
    return 2.0 * xcorr[:n]


def _make_sine(n, amp, phase=0.5):
    """Anti-palindromic sinusoidal: a[i] = (1 + amp*sin(2*pi*(i+phase)/n)) / n, normalized."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / n
    a = (1.0 + amp * np.sin(theta)) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _make_multi_sine(n, amps, phases):
    """Multi-mode sinusoidal (all mode 1 with different phases for variety)."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64) / n
    for amp, ph in zip(amps, phases):
        theta = 2.0 * np.pi * (i + ph) / n
        a += amp * np.sin(theta) / n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n, dtype=np.float64) / n
    return a


def _softmax_weights_normalized(c, T):
    """
    Compute softmax weights on normalized convolution peaks.
    Normalizes c to [0,1] range then applies softmax with temperature T.
    """
    c_min = float(np.min(c))
    c_max = float(np.max(c))
    rng = c_max - c_min
    if rng < 1e-15:
        return np.ones(c.size) / c.size
    c_norm = (c - c_min) / rng  # in [0, 1]
    c_shift = c_norm - 1.0  # in [-1, 0], max is 0
    exp_c = np.exp(T * c_shift)
    s = exp_c.sum()
    if s < 1e-300:
        w = np.zeros(c.size)
        mask = c_norm >= 1.0 - 1e-10
        n_max = mask.sum()
        if n_max == 0:
            w[np.argmax(c)] = 1.0
        else:
            w[mask] = 1.0 / n_max
        return w
    return exp_c / s


def _optimize_soft(a, deadline, T_start=8.0, T_end=80.0, lr0=None):
    """
    Soft-max mirror descent optimization with adaptive lr and temperature annealing.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size, dtype=np.float64) / a.size
        s = 1.0
    a = a / s

    n = a.size
    if lr0 is None:
        lr0 = 0.25 * n

    best_a = a.copy()
    best_score = _score(best_a)

    it = 0
    last_improve = 0
    patience = 5000
    eval_interval = 5
    rng = np.random.RandomState(12345)

    while time.monotonic() < deadline:
        it += 1

        # Annealing schedule
        t_frac = min(1.0, it / 100000.0)
        T = T_start + (T_end - T_start) * t_frac
        lr = lr0 / (1.0 + 0.01 * it)

        # Compute convolution
        c = _fft_convolve(a)
        if c.size == 0:
            break

        # Soft-max weights (normalized)
        w = _softmax_weights_normalized(c, T)

        # Gradient via FFT
        grad = _grad_via_fft(w, a)

        # Project to simplex tangent
        grad_proj = grad - np.mean(grad)

        # Mirror descent update
        update = -lr * grad_proj
        update = np.clip(update, -10.0, 10.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        # Periodic evaluation

            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()
                last_improve = it
                lr0 = min(0.5 * n, lr0 * 1.01)
            elif it - last_improve > patience:
                # Restart with perturbation
                a = best_a.copy()
                noise = rng.standard_normal(n) * 0.01
                a = a + noise * a.mean()
                a = np.clip(a, 0, 1000)
                a = a / a.sum()
                patience = max(2000, int(patience * 1.3))
                lr0 = max(0.05 * n, lr0 * 0.8)

def _optimize_hard(a, deadline, K=12, lr0=None):

    Hard-max top-K active set refinement for final polish.
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size, dtype=np.float64) / a.size
        s = 1.0
    a = a / s

    n = a.size
    if lr0 is None:
        lr0 = 0.5 * n

    best_a = a.copy()

    it = 0
    last_improve = 0
    patience = 3000
    eval_interval = 3
    rng = np.random.RandomState(99999)

    while time.monotonic() < deadline:
        it += 1
        lr = lr0 / (1.0 + 0.005 * it)

        c = _fft_convolve(a)
        if c.size == 0:
            break

        # Top-K peaks
        if c.size <= K:
            top_idx = np.arange(c.size)
        else:

        top_vals = c[top_idx]
        T_local = 3.0
        top_shift = top_vals - top_vals.max()
        weights = np.exp(T_local * top_shift)
        weights /= weights.sum()

        # Build full weight vector
        w = np.zeros(c.size, dtype=np.float64)
        for i, k in enumerate(top_idx):
            w[k] = weights[i]

        # Gradient via FFT
        grad = _grad_via_fft(w, a)

        # Update
        grad_proj = grad - np.mean(grad)
        update = -lr * grad_proj
        update = np.clip(update, -10.0, 10.0)
        a = a * np.exp(update)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a.copy()
            break
        a = a / s

        if it % eval_interval == 0:
            sc = _score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a.copy()

                noise = rng.standard_normal(n) * 0.005
                a = a + noise * a.mean()
                a = np.clip(a, 0, 1000)
                a = a / a.sum()
                patience = max(1000, int(patience * 1.4))

        arr_list = [float(max(0.0, min(1000.0, x))) for x in arr]
        try:
            actual = float(evaluate_sequence(arr_list))
        except Exception:
            actual = np.inf
        if np.isfinite(actual) and actual < best_actual:
            best_actual = actual
            best_out = arr_list
            print(f"  {name} (n={arr.size}): actual={actual:.8f}", flush=True)
            if best_actual < 1.48:
                break

    if best_out is None:
        best_out = [float(x) for x in init_a]
        try:
            best_actual = float(evaluate_sequence(best_out))
        except Exception:
            best_actual = init_score
        print(f"  Fallback: {best_actual:.8f}", flush=True)

    total_time = time.monotonic() - t0
    print(f"[Done] Time: {total_time:.1f}s, Best score: {best_actual:.8f}", flush=True)
    return best_out
