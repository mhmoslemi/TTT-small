import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))

def _make_sine_candidate(n, amp, mode):
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)

    a = (1.0 + float(amp) * delta) / float(n)
    a = np.where(np.isfinite(a), a, 0.0)

LP-based iterative refinement for low-peak nonnegative step functions.
Uses the active-set LP formulation from the paper, combined with sinusoidal
initialization and adaptive step sizes.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import diags, csr_matrix


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))


def _make_sine_candidate(n, amp, mode):
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    delta = np.sin(float(mode) * theta)
    a = (1.0 + float(amp) * delta) / float(n)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        a = np.ones(n, dtype=np.float64) / float(n)
        return a
    a = a / s
    return a


def _build_conv_matrix(f, n):
    """Build the (2n-1) x n convolution constraint matrix for (f*b)."""
    # A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # Diagonal d (offset d) has value f[d] in all n positions
    diagonals = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = list(range(n))
    A = csr_matrix(diags(diagonals, offsets, shape=(2*n-1, n), format='csr'))
    return A


def _solve_lp(f, M, n):
    """Solve: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0."""
    c = -np.ones(n, dtype=np.float64)
    A = _build_conv_matrix(f, n)
    b_ub = np.full(2*n-1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 60})
    
    if res.status != 0 or res.x is None:
        return None
    
    b = np.asarray(res.x, dtype=np.float64)
    b = np.clip(b, 0.0, None)
    return b


def _lp_refine(f_init, budget_s=1500.0):
    """Iterative LP-based refinement."""
    t0 = time.monotonic()
    arr = np.asarray(f_init, dtype=np.float64).copy()
    arr = np.clip(arr, 0.0, None)
    s = float(arr.sum())
    if s < 0.01:
        return arr, np.inf
    arr = arr / s
    
    n = arr.size
    best = arr.copy()
    best_score = _internal_score(best)
    print(f"[lp] start: n={n}, score={best_score:.10f}")
    
    t_step = 0.15
    patience = 0
    max_patience = 50
    it = 0
    stalling = 0
    
    while time.monotonic() - t0 < budget_s:
        it += 1
        f = arr.copy()
        conv = _fft_convolve(f)
        M = float(np.max(conv))
        
        if not np.isfinite(M) or M <= 0:
            break
        
        b = _solve_lp(f, M, n)
        if b is None:
            print(f"[lp] LP failed at iter {it}")
            break
        
        b_sum = float(np.sum(b))
        if b_sum < 1e-12:
            break
        
        g = b / b_sum
        g = np.clip(g, 0.0, None)
        g_s = float(np.sum(g))
        if g_s < 0.01:
            break
        g = g / g_s
        
        # Try the update
        new_f = (1.0 - t_step) * f + t_step * g
        new_f = np.clip(new_f, 0.0, None)
        new_s = float(np.sum(new_f))
        if new_s < 0.01:
            t_step = max(0.02, t_step * 0.5)
            continue
        new_f = new_f / new_s
        
        new_score = _internal_score(new_f)
        
        if np.isfinite(new_score) and new_score < best_score - 1e-10:
            prev_best = best_score
            best = new_f.copy()
            best_score = new_score
            arr = new_f
            patience = 0
            stalling = 0
            if it <= 5 or it % 20 == 0 or best_score < 1.52:
                print(f"[lp] iter {it}: score {prev_best:.10f} -> {best_score:.10f} (t={t_step:.3f})")
        else:
            patience += 1
            stalling += 1
            if stalling > 10:
                t_step = max(0.02, t_step * 0.6)
                stalling = 0
            # Revert
            arr = f
        
        if patience > max_patience:
            print(f"[lp] converged/stalled at iter {it}, best={best_score:.10f}")
            break
    
    elapsed = time.monotonic() - t0
    print(f"[lp] done: iter={it}, best_score={best_score:.10f}, time={elapsed:.1f}s")
    return best, best_score


def _optimize_sine_amplitude(n, mode, max_amps=49):
    """Find best amplitude for sinusoidal family."""
    best_score = np.inf
    best_amp = 0.78
    base = 1.0 / n
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    delta = np.sin(float(mode) * theta)
    
    amps = np.linspace(0.0, 1.5, max_amps)
    for amp in amps:
        a = base * (1.0 + amp * delta)
        a = np.clip(a, 0.0, None)
        sc = _internal_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_amp = amp
    
    # Refine
    step = 1.5 / (max_amps - 1)
    for _ in range(4):
        low = max(0.0, best_amp - step)
        high = min(1.5, best_amp + step)
        grid = np.linspace(low, high, 25)
        for amp in grid:
            a = base * (1.0 + amp * delta)
            a = np.clip(a, 0.0, None)
            sc = _internal_score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_amp = amp
        step *= 0.4
    
    return best_amp, best_score


def _multiplicative_refine(arr, budget_s=60.0):
    """Multiplicative update refinement (from previous program)."""
    start = time.monotonic()
    arr = np.asarray(arr, dtype=np.float64).copy()
    arr = np.clip(arr, 0.0, None)
    s = float(arr.sum())
    if s < 0.01:
        return arr, np.inf
    arr = arr / s
    
    best = arr.copy()
    best_score = _internal_score(best)
    n = arr.size
    lr = max(1.0, 0.05 * float(n))
    max_lr = 0.25 * float(n)
    period = 10
    max_iters = 3000
    it = 0
    last_improve = 0
    K = 8 if n <= 4096 else 12
    
    while time.monotonic() - start < budget_s and it < max_iters:
        it += 1
        c = _fft_convolve(arr)
        if c.size == 0:
            break
        maxc = float(np.max(c))
        if not np.isfinite(maxc):
            break
        
        if c.size <= K:
            active = np.arange(c.size)
        else:
            active = np.argpartition(c, -K)[-K:]
        
        grad = np.zeros(n, dtype=np.float64)
        cnt = 0
        for kk in active:
            k = int(kk)
            cnt += 1
            i_min = max(0, k - n + 1)
            i_max = min(k, n - 1)
            if i_max >= i_min:
                idx = np.arange(i_min, i_max + 1)
                grad[idx] += 2.0 * arr[k - idx]
        
        if cnt <= 0:
            break
        grad /= float(cnt)
        
        g = grad - np.mean(grad)
        if not np.any(g):
            break
        
        exponent = -lr * g
        exponent = np.clip(exponent, -1.0, 1.0)
        arr = arr * np.exp(exponent)
        arr = np.clip(arr, 0.0, None)
        ts = float(arr.sum())
        if not np.isfinite(ts) or ts < 0.01:
            arr = best.copy()
            break
        arr = arr / ts
        
        if it % period == 0:
            sc = _internal_score(arr)
            if np.isfinite(sc) and sc < best_score:
                best = arr.copy()
                best_score = sc
                last_improve = it
                lr = min(max_lr, lr * 1.05)
            else:
                if sc > best_score * 1.02:
                    arr = best.copy()
                    lr = max(1.0, lr * 0.75)
                if it - last_improve > 200:
                    break
    
    return best, best_score

    arr = arr / s
    return arr

"""
LP-based iterative refinement for low-peak nonnegative step functions.
Combines sinusoidal initialization with the active-set LP orientation from
the paper (max sum(b) s.t. b>=0, (f*b)<=M), plus multiplicative polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import diags, csr_matrix


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))


def _make_sine_candidate(n, amp, mode):
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    delta = np.sin(float(mode) * theta)
    a = (1.0 + float(amp) * delta) / float(n)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        a = np.ones(n, dtype=np.float64) / float(n)
        return a
    a = a / s
    return a


def _build_conv_matrix(f, n):
    """Build (2n-1) x n matrix A where (A @ b)[k] = (f*b)[k]."""
    diagonals = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = list(range(n))
    A = csr_matrix(diags(diagonals, offsets, shape=(2 * n - 1, n), format='csr'))
    return A


def _solve_lp(f, M, n, time_limit=30):
    """Solve: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0."""
    c = -np.ones(n, dtype=np.float64)
    A = _build_conv_matrix(f, n)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': time_limit})
    if res.status != 0 or res.x is None:
        return None
    b = np.asarray(res.x, dtype=np.float64)
    b = np.clip(b, 0.0, None)
    return b


def _lp_refine(f_init, budget_s=1200.0):
    """Core LP-based iterative refinement."""
    t0 = time.monotonic()
    arr = np.asarray(f_init, dtype=np.float64).copy()
    arr = np.clip(arr, 0.0, None)
    s = float(arr.sum())
    if s < 0.01:
        return arr, np.inf
    arr = arr / s

    n = arr.size
    best = arr.copy()
    best_score = _internal_score(best)
    print(f"[lp] start: n={n}, score={best_score:.10f}")

    t_step = 0.12
    patience = 0
    max_patience = 60
    it = 0
    fail_count = 0

    while time.monotonic() - t0 < budget_s:
        it += 1
        f = arr.copy()
        conv = _fft_convolve(f)
        M = float(np.max(conv))
        if not np.isfinite(M) or M <= 0:
            break

        b = _solve_lp(f, M, n, time_limit=25)
        if b is None:
            fail_count += 1
            if fail_count > 5:
                print(f"[lp] too many LP failures, stopping")
                break
            continue

        b_sum = float(np.sum(b))
        if b_sum < 1e-12:
            break

        g = b / b_sum
        g = np.clip(g, 0.0, None)
        g_s = float(np.sum(g))
        if g_s < 0.01:
            break
        g = g / g_s

        # Fixed point detection
        diff = float(np.sum(np.abs(g - f)))
        if diff < 1e-8:
            print(f"[lp] fixed point at iter {it}")
            break

        new_f = (1.0 - t_step) * f + t_step * g
        new_f = np.clip(new_f, 0.0, None)
        new_s = float(np.sum(new_f))
        if new_s < 0.01:
            t_step = max(0.02, t_step * 0.5)
            continue
        new_f = new_f / new_s

        new_score = _internal_score(new_f)

        if np.isfinite(new_score) and new_score < best_score - 1e-11:
            prev_best = best_score
            best = new_f.copy()
            best_score = new_score
            arr = new_f
            patience = 0
            fail_count = 0
            # Gradually increase step size when improving
            if patience == 0 and it > 5:
                t_step = min(0.35, t_step * 1.03)
            if it <= 5 or it % 25 == 0 or best_score < 1.52:
                print(f"[lp] iter {it}: {prev_best:.10f} -> {best_score:.10f} (t={t_step:.3f})")
        else:
            patience += 1
            t_step = max(0.02, t_step * 0.75)
            arr = f  # revert

        if patience > max_patience:
            # Try a larger jump
            patience = 0
            t_step = min(0.4, 0.5 * t_step + 0.1)
            # If still stuck after another round, stop
            if patience == 0 and t_step > 0.35:
                print(f"[lp] stalled at iter {it}, best={best_score:.10f}")
                break

    elapsed = time.monotonic() - t0
    print(f"[lp] done: iter={it}, best={best_score:.10f}, time={elapsed:.1f}s")
    return best, best_score


def _optimize_sine_amplitude(n, mode, deadline_s=10.0):
    """Find best amplitude for sinusoidal family at given n and mode."""
    best_score = np.inf
    best_amp = 0.78
    base = 1.0 / float(n)
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    delta = np.sin(float(mode) * theta)

    t_start = time.monotonic()
    amps = np.linspace(0.0, 1.5, 41)
    for amp in amps:
        if time.monotonic() - t_start > deadline_s:
            break
        a = base * (1.0 + float(amp) * delta)
        a = np.clip(a, 0.0, None)
        sc = _internal_score(a)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_amp = float(amp)

    step = 1.5 / 40.0
    for _ in range(4):
        if time.monotonic() - t_start > deadline_s:
            break
        low = max(0.0, best_amp - step)
        high = min(1.5, best_amp + step)
        grid = np.linspace(low, high, 21)
        for amp in grid:
            a = base * (1.0 + float(amp) * delta)
            a = np.clip(a, 0.0, None)
            sc = _internal_score(a)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_amp = float(amp)
        step *= 0.4

    return best_amp, best_score


def _multiplicative_refine(arr, budget_s=60.0):
    """Multiplicative update refinement on active lags."""
    start = time.monotonic()
    arr = np.asarray(arr, dtype=np.float64).copy()
    arr = np.clip(arr, 0.0, None)
    s = float(arr.sum())
    if s < 0.01:
        return arr, np.inf
    arr = arr / s

    best = arr.copy()
    best_score = _internal_score(best)
    n = arr.size
    lr = max(1.0, 0.04 * float(n))
    max_lr = 0.2 * float(n)
    period = 10
    max_iters = 5000
    it = 0
    last_improve = 0
    K = 10 if n <= 4096 else 14

    while time.monotonic() - start < budget_s and it < max_iters:
        it += 1
        c = _fft_convolve(arr)
        if c.size == 0:
            break
        if not np.isfinite(np.max(c)):
            break

        if c.size <= K:
            active = np.arange(c.size)
        else:
            active = np.argpartition(c, -K)[-K:]

        grad = np.zeros(n, dtype=np.float64)
        cnt = 0
        for kk in active:
            k = int(kk)
            cnt += 1
            i_min = max(0, k - n + 1)
            i_max = min(k, n - 1)
            if i_max >= i_min:
                idx = np.arange(i_min, i_max + 1)
                grad[idx] += 2.0 * arr[k - idx]

        if cnt <= 0:
            break
        grad /= float(cnt)
        g = grad - np.mean(grad)
        if not np.any(g):
            break

        exponent = -lr * g
        exponent = np.clip(exponent, -0.8, 0.8)
        arr = arr * np.exp(exponent)
        arr = np.clip(arr, 0.0, None)
        ts = float(arr.sum())
        if not np.isfinite(ts) or ts < 0.01:
            arr = best.copy()
            break
        arr = arr / ts

        if it % period == 0:
            sc = _internal_score(arr)
            if np.isfinite(sc) and sc < best_score:
                best = arr.copy()
                best_score = sc
                last_improve = it
                lr = min(max_lr, lr * 1.04)
            else:
                if sc > best_score * 1.01:
                    arr = best.copy()
                    lr = max(1.0, lr * 0.8)
                if it - last_improve > 300:
                    break

    return best, best_score


def _safe_resolve(x):
    arr = np.asarray(x, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return None
    arr = arr / s
    return arr
