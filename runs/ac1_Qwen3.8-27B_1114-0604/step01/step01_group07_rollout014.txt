def _next_pow2(m):
    p = 1
    while p < m:
        p <<= 1
    return p

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _score(a):

    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    mx = c.max()
    return 2.0 * a.size * mx / (s * s)

def _optimize(a, deadline, K=16):
    """Active-set multiplicative weight optimization."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0, 1000)
    s = a.sum()
    a = a / s  # Normalize to sum=1
    
    best = a.copy()
    best_score = _score(best)
    n = a.size
    
    lr = 0.5
    for it in range(50000):
        if time.monotonic() > deadline:
            break
        
        c = _fft_convolve(a)
        max_c = c.max()

        if len(c) <= K:
            active = np.arange(len(c))
        else:

        # Compute gradient
        grad = np.zeros(n)
        for k in active:
            k = int(k)
            i_min = max(0, k - n + 1)
            i_max = min(k, n - 1)
            for i in range(i_min, i_max + 1):

        grad /= len(active)
        
        # Center the gradient (project onto sum-preserving subspace)
        grad = grad - grad.mean()

        print(f"Validation error: {e}")
        real_score = np.inf

    # If validation failed, fallback to uniform
    if not np.isfinite(real_score) or real_score > 2.01:
        fallback = [1.0 / 1024.0] * 1024
        try:
            fb_score = evaluate_sequence(fallback)
            print(f"Fallback uniform score: {fb_score:.8f}")
            return fallback
        except Exception:
            return [1.0 / 256.0] * 256

    return [float(max(0.0, min(1000.0, x))) for x in best_a]
