from scipy.optimize import linprog


def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def compute_score(a):
    """Compute 2*n*max(a*a)/sum(a)^2."""
    n = a.size
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    b = full_conv(a)
    return 2.0 * n * float(np.max(b)) / (s * s)


def build_conv_matrix(f):
    """Build matrix A where (A@g)[k] = convolve(f,g)[k]."""
    n = f.size
    k_idx = np.arange(2 * n - 1)[:, None].astype(np.int64)
    j_idx = np.arange(n)[None, :].astype(np.int64)
    d = k_idx - j_idx
    A = np.zeros((2 * n - 1, n))
    valid = (d >= 0) & (d < n)
    A[valid] = f[d[valid]]
    return A


def solve_lp(f, M):
    """Solve LP: max sum(g) s.t. g>=0, convolve(f,g) <= M."""
    n = f.size
    A = build_conv_matrix(f)
    c = -np.ones(n)

    return res.x

"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based iteration
(the state-of-the-art approach) with multiple sequence lengths, diverse
initializations, coarse-to-fine upsampling, gradient polish, and local search.
The LP step solves: max sum(g) s.t. g>=0, convolve(f,g) <= M * 1.
"""

import numpy as np
import time
from scipy.optimize import linprog


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def compute_score(a):
    """Compute 2*n*max(a*a)/sum(a)^2."""
    n = a.size
    s = float(np.sum(a))
    if s < 1e-15:
        return np.inf
    b = full_conv(a)
    return 2.0 * n * float(np.max(b)) / (s * s)


def compute_score_normalized(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def build_conv_matrix(f):
    """Build matrix A where (A@g)[k] = convolve(f,g)[k]. A is (2n-1) x n."""
    n = f.size
    A = np.zeros((2 * n - 1, n))
    for j in range(n):
        A[j:j + n, j] = f
    return A


def solve_lp(f, M, time_limit=10.0):
    """Solve LP: max sum(g) s.t. g>=0, convolve(f,g) <= M elementwise."""
    n = f.size
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.status != 0:
            return None
        g = res.x
        g = np.maximum(g, 0)
        return g
    except Exception:
        return None


def lp_iterate(f0, time_budget, rng, t_initial=0.3, t_min=0.03, verbose=True):
    """
    Run LP-based iteration to minimize peak autocorrelation.
    f0 should be normalized to sum=1, non-negative.
    Returns (best_f, best_score) where best_score = 2*n*max(conv(f,f)).
    """
    n = f0.size
    f = np.maximum(f0, 0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_f = f.copy()
    best_score = compute_score_normalized(f)

    t = t_initial
    no_improve = 0
    t_start = time.monotonic()
    it = 0

    while time.monotonic() - t_start < time_budget:
        b = full_conv(f)
        M = float(np.max(b))
        cur_score = 2.0 * n * M

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1

        if verbose and (it % 5 == 0 or it < 3):
            print(f"    LP iter {it}: score={cur_score:.8f}, best={best_score:.8f}, t={t:.4f}, M={M:.6f}")

        # Solve LP
        remaining = time_budget - (time.monotonic() - t_start)
        if remaining < 1.0:
            break
        lp_time = min(15.0, remaining * 0.3)
        g = solve_lp(f, M, time_limit=lp_time)
        if g is None or g.sum() < 1e-15:
            if verbose:
                print(f"    LP failed or trivial at iter {it}")
            break

        # Normalize g
        g = g / g.sum()

        # Blend
        f_new = (1.0 - t) * f + t * g

        # Evaluate new score
        b_new = full_conv(f_new)
        new_score = 2.0 * n * float(np.max(b_new))

        if new_score <= cur_score + 1e-12:
            f = f_new
            # Slightly increase step
            t = min(t * 1.02, 0.6)
        else:
            # Reduce step
            t = max(t * 0.75, t_min)
            # Still accept with smaller step
            f = (1.0 - t * 0.5) * f + t * 0.5 * g

        if no_improve > 30:
            # Perturb and restart
            if verbose:
                print(f"    Perturbing at iter {it} (no improve for {no_improve})")
            f = best_f + 0.02 * rng.standard_normal(n)
            f = np.maximum(f, 0)
            fs = f.sum()
            if fs < 1e-15:
                f = np.ones(n) / n
            else:
                f = f / fs
            t = t_initial
            no_improve = 0

        it += 1

    if verbose:
        print(f"    LP done: {it} iters, best_score={best_score:.8f}")
    return best_f, best_score


def gradient_polish(f, time_budget, rng):
    """Polish using smoothed subgradient descent with momentum."""
    n = f.size
    f = np.maximum(f, 0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_f = f.copy()
    best_score = compute_score_normalized(f)

    eta = 0.15
    tau_frac = 0.04
    beta = 0.5
    vel = np.zeros(n)
    t_start = time.monotonic()
    it = 0
    last_improve = 0

    while time.monotonic() - t_start < time_budget:
        b = full_conv(f)
        M = float(np.max(b))
        cur_score = 2.0 * n * M

        # Smoothed subgradient
        tau = tau_frac * M
        if tau < 1e-18:
            tau = 1e-18
        shift = b / tau
        shift -= shift.max()
        w = np.exp(shift)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(b.size)
            w[int(np.argmax(b))] = 1.0
        else:
            w /= w_sum

        # Gradient: g = 2 * correlate(w, f, 'valid')
        grad = 2.0 * np.correlate(w, f, mode='valid')

        vel = beta * vel + (1.0 - beta) * grad

        # Normalize velocity for stable step
        vnorm = np.linalg.norm(vel)
        if vnorm > 1e-18:
            vel = vel / vnorm

        f_new = f - eta * vel
        f_new = np.maximum(f_new, 0)
        s_new = f_new.sum()
        if s_new < 1e-15:
            f_new = f
        else:
            f_new = f_new / s_new

        b_new = full_conv(f_new)
        new_score = 2.0 * n * float(np.max(b_new))

        if new_score < best_score - 1e-10:
            best_score = new_score
            best_f = f_new.copy()
            last_improve = it
            f = f_new
        elif new_score <= cur_score or rng.random() < 0.05:
            f = f_new
        else:
            f = (1 - 0.5 * eta) * f + 0.5 * eta * f_new  # damped

        it += 1
        if it - last_improve > 3000:
            f = best_f + 0.01 * rng.standard_normal(n)
            f = np.maximum(f, 0)
            fs = f.sum()
            if fs > 1e-15:
                f = f / fs
            vel = np.zeros(n)
            eta = 0.05 + 0.05 * rng.random()
            tau_frac = 0.02 + 0.03 * rng.random()
            last_improve = it

    return best_f, best_score


def local_polish(f, time_budget, rng):
    """Local polish via random perturbations with adaptive scale."""
    n = f.size
    f = np.maximum(f, 0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-15:
        f = np.ones(n) / n
    else:
        f = f / s

    best_f = f.copy()
    best_score = compute_score_normalized(f)

    t_start = time.monotonic()
    scale = 0.03
    no_improve_rounds = 0

    while time.monotonic() - t_start < time_budget and scale > 1e-8:
        improved = False
        n_trials = min(20000, max(100, int(50000 * (scale / 0.03))))
        for _ in range(n_trials):
            if time.monotonic() - t_start > time_budget:
                break
            cand = f + scale * rng.standard_normal(n)
            cand = np.maximum(cand, 0)
            cs = cand.sum()
            if cs < 1e-15:
                continue
            cand = cand / cs
            sc = compute_score_normalized(cand)
            if sc < best_score:
                best_score = sc
                best_f = cand.copy()
                f = cand.copy()
                improved = True
        if not improved:
            no_improve_rounds += 1
            if no_improve_rounds > 8:
                break
            scale *= 0.7
        else:
            no_improve_rounds = 0
            scale *= 0.92

    return best_f, best_score


def init_uniform(n):
    return np.ones(n) / n


def init_u_shape(n):
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_sided(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / c) ** 2 + 0.1
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.05
    return a / a.sum()


def init_two_cluster(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.1 + 0.2 * rng.random())
    c2 = n * (0.6 + 0.2 * rng.random())
    sigma = n * (0.03 + 0.07 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_sqrt_decay(n, rng):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a += 0.03 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_cosine(n, rng):
    i = np.arange(n, dtype=np.float64)
    freq = 1.0 + int(rng.integers(0, 3))
    phase = rng.random() * 2 * np.pi
    a = (1.0 + freq * np.cos(2.0 * np.pi * i / max(n - 1, 1) + phase)) ** 2
    a = np.maximum(a, 0.01)
    return a / a.sum()


def upscale(f, factor):
    """Upscale by factor using zero insertion."""
    n = f.size
    n_new = n * factor
    out = np.zeros(n_new)
    out[::factor] = f
    return out / out.sum()


def upscale_replicate(f, factor):
    """Upscale by factor using replication."""
    n = f.size
    n_new = n * factor
    out = np.zeros(n_new)
    for i in range(n):
        out[i * factor:(i + 1) * factor] = f[i] / factor
    return out / out.sum()

    n_values_to_try = []
    if prev_a is not None:
        n_values_to_try.append(prev_a.size)
    n_values_to_try.extend([128, 256, 512, 1024])
    n_values_to_try = sorted(set(n_values_to_try))

    # Time allocation: prioritize the n that has the current best, and smaller n (faster)
    total_avail = deadline - time.monotonic()
    print(f"[plan] n_values={n_values_to_try}, total_time={total_avail:.1f}s")

    # Allocate time proportional to 1/sqrt(n) but boost current best n
    raw_w = []
    for nv in n_values_to_try:
        w = 1.0 / np.sqrt(nv)
        if best_overall is not None and best_overall.size == nv:
            w *= 2.0  # boost current best
        raw_w.append(w)
    total_w = sum(raw_w)
    # Reserve 15% for polish
    time_frac_opt = 0.85
    time_frac_polish = 0.15
    weights = [w / total_w * time_frac_opt * total_avail for w in raw_w]

    global_best_lp = None
    global_best_lp_sc = np.inf

    for ni, nv in enumerate(n_values_to_try):
        if time.monotonic() > deadline:
            break

        time_n = weights[ni]
        if time_n < 10.0:
            time_n = min(10.0, max(5.0, (deadline - time.monotonic()) * 0.1))

        print(f"\n[phase] n={nv}, time_budget={time_n:.1f}s")

        # Determine number of starts
        # LP for n=128: ~10-50ms per iter. n=512: ~100-500ms. n=1024: ~500-2000ms.
        est_lp_time = 0.01 * (nv / 128)
        n_starts = max(2, min(8, int(time_n / (est_lp_time * 50))))
        time_per_start = time_n / n_starts

        # Initializations
        all_inits = [
            init_u_shape(nv),
            init_sided(nv),
            init_random(nv, rng),
            init_two_cluster(nv, rng),
            init_sqrt_decay(nv, rng),
            init_cosine(nv, rng),
            init_uniform(nv),
        ]

        # If we have previous best at this n, use it first
        if prev_a is not None and prev_a.size == nv:
            all_inits.insert(0, prev_a.copy())

        # If we have a good LP result at smaller n, upsample
        if global_best_lp is not None and nv > global_best_lp.size:
            factor = nv // global_best_lp.size
            if factor >= 2:
                up1 = upscale(global_best_lp, factor)
                up2 = upscale_replicate(global_best_lp, factor)
                all_inits.insert(0, up1)
                all_inits.insert(1, up2)

        n_starts = min(n_starts, len(all_inits))

        for si in range(n_starts):
            if time.monotonic() > deadline:
                break

            a_init = all_inits[si].copy()
            tb = min(time_per_start, deadline - time.monotonic() - 2.0)
            if tb < 5.0:
                break

            # Vary LP parameters
            t_init = 0.2 + 0.2 * rng.random()
            t_min = 0.02 + 0.03 * rng.random()

            a_best, sc_best = lp_iterate(a_init, tb, rng,
                                         t_initial=t_init, t_min=t_min,
                                         verbose=(si == 0 and ni == 0))

            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = a_best.copy()
                global_best_lp = a_best.copy()
                global_best_lp_sc = sc_best
                print(f"  [LP] n={nv}, start={si}: {sc_best:.8f} *** NEW BEST ***")
            else:
                print(f"  [LP] n={nv}, start={si}: {sc_best:.8f}")

            # Track best LP result for upsampling
            if sc_best < global_best_lp_sc:
                global_best_lp = a_best.copy()
                global_best_lp_sc = sc_best

    # Phase 2: Gradient polish on best
    if best_overall is not None and time.monotonic() < deadline - 15:
        remain = deadline - time.monotonic() - 5.0
        print(f"\n[polish] remaining={remain:.1f}s, n={best_overall.size}, score={best_overall_sc:.8f}")

        # Gradient polish
        gp_time = min(remain * 0.5, 400.0)
        if gp_time > 10:
            a_p, sc_p = gradient_polish(best_overall, gp_time, rng)
            if sc_p < best_overall_sc:
                best_overall_sc = sc_p
                best_overall = a_p.copy()
                print(f"  [polish] gradient: {sc_p:.8f} *** NEW BEST ***")
            else:
                print(f"  [polish] gradient: {sc_p:.8f}")

    # Phase 3: Local random polish
    if best_overall is not None and time.monotonic() < deadline - 10:
        remain = deadline - time.monotonic() - 5.0
        if remain > 10:
            rp_time = min(remain * 0.6, 200.0)
            a_r, sc_r = local_polish(best_overall, rp_time, rng)
            if sc_r < best_overall_sc:
                best_overall_sc = sc_r
                best_overall = a_r.copy()
                print(f"  [polish] random: {sc_r:.8f} *** NEW BEST ***")
            else:
                print(f"  [polish] random: {sc_r:.8f}")

    # Phase 4: Final short LP polish
    if best_overall is not None and time.monotonic() < deadline - 15:
        remain = deadline - time.monotonic() - 5.0
        if remain > 15:
            print(f"\n[final LP] remaining={remain:.1f}s")
            a_fl, sc_fl = lp_iterate(best_overall, min(remain * 0.8, 120.0), rng,
                                     t_initial=0.1, t_min=0.01, verbose=False)
            if sc_fl < best_overall_sc:
                best_overall_sc = sc_fl
                best_overall = a_fl.copy()
                print(f"  [final LP] {sc_fl:.8f} *** NEW BEST ***")

    # Fallback
    if best_overall is None:
        best_overall = np.ones(128) / 128
        best_overall_sc = 2.0
        print("[fallback] Using uniform")

    # Final verification
    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"\n[final] n={len(result)}, internal={best_overall_sc:.8f}, evaluator={final_sc:.8f}")
    print(f"[timing] total={time.monotonic() - t_global:.1f}s")

    return result
