"""
Search for non-negative step heights minimizing
    2 * n * max(conv(a, a)) / (sum(a) ** 2).

Strategy:
- Use the supplied previous best sequence when available.
- Combine a low-dimensional smooth parameterization (polynomial + log-endpoint
  + Fourier + Gaussian bump bases) with analytic-gradient L-BFGS runs and a
  bounded differential evolution global pass.
- Refine strong candidates with an FFT-based multiplicative soft-max local
  search.  The local search periodically uses a primal-dual/multiplicative-
  weights "minimax direction" escape and smooth line-search mixins.
- Finish with exact evaluations of the most promising candidates.
"""

import math
import time

import numpy as np

    from scipy.optimize import minimize as _scipy_minimize

    _HAS_SCIPY = False


def _safe_budget(value, default=2400.0):
    try:
        value = float(value)
    except Exception:
        return float(default)
    if not np.isfinite(value) or value <= 10.0:
        return float(default)
    return float(value)

        except Exception:
            pass
    return 6121


def _get_parent(n):
    base = globals().get("height_sequence_1", None)
    if base is None:
        return None
    try:
        arr = np.asarray(base, dtype=np.float64).ravel()
        if arr.size != n:
            return None

    except Exception:

def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a.fill(1.0)
        s = float(a.sum())
    a /= s
    a = np.maximum(a, 1e-300)
    s2 = float(a.sum())
    if not np.isfinite(s2) or s2 <= 0.0:
        return np.full_like(a, 1.0 / float(a.size))
    return a / s2


def _softmax_v(v):
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return np.array([1.0], dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    vmax = float(np.max(v))
    if not np.isfinite(vmax):
        vmax = 0.0
        v = np.where(np.isfinite(v), v, 0.0)
    z = v - vmax
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        e = np.ones_like(v)
        s = float(v.size)
    a = e / s
    a = np.maximum(a, 1e-300)
    a /= float(a.sum())
    return a


def _fft_len(n):
    m = 2 * int(n) - 1
    L = 1
    while L < m:
        L <<= 1
    return L


def _conv_self(a, L, m):
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    c = np.ascontiguousarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c


def _grad_weights(weights, a, L, n):
    # Gradient of sum_k weights[k] * (a*a)[k] with respect to a.
    W = np.fft.rfft(weights, L)
    Ar = np.fft.rfft(a[::-1], L)
    g = np.fft.irfft(W * Ar, L)[n - 1:n - 1 + n]
    g = np.ascontiguousarray(g, dtype=np.float64)
    np.maximum(g, 0.0, out=g)
    g *= 2.0
    return g


def _smooth_score_b(b, temp):
    b = np.asarray(b, dtype=np.float64)
    if temp is None or not np.isfinite(temp) or temp <= 0.0:
        temp = 1e-12
    maxb = float(np.max(b))
    if not np.isfinite(maxb):
        return maxb, None, maxb
    z = (b - maxb) / float(temp)
    z = np.clip(z, -700.0, 0.0)
    w = np.exp(z)
    s = float(w.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.zeros_like(b)
        idx = int(np.argmax(b))
        w[idx] = 1.0
        s = 1.0
        J = maxb
    else:
        w /= s
        J = maxb + float(temp) * math.log(s)
        if not np.isfinite(J):
            J = maxb
    return J, w, maxb


def _hard_score(a, n, L, m):
    b = _conv_self(a, L, m)
    score = 2.0 * float(n) * float(np.max(b))
    return score, b


def _build_basis(n):
    n = int(n)
    if n <= 1:
        return np.ones((max(n, 1), 1), dtype=np.float64)
    idx = np.arange(n, dtype=np.float64)
    u = (idx + 0.5) / float(n)
    x = 2.0 * u - 1.0
    cols = []

    for d in range(1, 6):
        cols.append(x ** d)

    eps = 1e-6
    cols.append(np.log(u + eps))
    cols.append(np.log(1.0 - u + eps))

    for k in range(1, 6):
        ang = 2.0 * np.pi * float(k) * u
        cols.append(np.sin(ang))
        cols.append(np.cos(ang))

    for c in (0.20, 0.40, 0.60, 0.80):
        cols.append(np.exp(-((u - c) / 0.12) ** 2))

    B = np.column_stack(cols).astype(np.float64)
    for j in range(B.shape[1]):
        col = B[:, j]
        r = float(np.sqrt(np.mean(col * col)))
        if np.isfinite(r) and r > 1e-12:
            B[:, j] = col / r
    B = np.where(np.isfinite(B), B, 0.0)
    return B


def _theta_to_a(theta, basis):
    theta = np.asarray(theta, dtype=np.float64).ravel()
    theta = np.where(np.isfinite(theta), theta, 0.0)
    v = basis.dot(theta)
    return _softmax_v(v)


def _a_to_theta(a, basis):
    a = _normalize(a)
    v = np.log(np.maximum(a, 1e-200))
    v = v - float(np.mean(v))
    try:
        theta, _, _, _ = np.linalg.lstsq(basis, v, rcond=1e-8)
    except Exception:
        theta = np.zeros(basis.shape[1], dtype=np.float64)
    theta = np.where(np.isfinite(theta), theta, 0.0)
    return np.clip(theta, -10.0, 10.0)


def _beta_shape(n, p, q, eps=1e-6):
    idx = np.arange(n, dtype=np.float64)
    u = (idx + 0.5) / float(n)
    a = np.power(u + eps, float(p)) * np.power(1.0 - u + eps, float(q))
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    return _normalize(a)


def _param_scalar(theta, n, L, m, basis, temp, ridge):
    try:
        theta = np.asarray(theta, dtype=np.float64).ravel()
        if not np.all(np.isfinite(theta)):
            return 1e100
        a = _theta_to_a(theta, basis)
        b = _conv_self(a, L, m)
        J, _, _ = _smooth_score_b(b, temp)
        score = 2.0 * float(n) * J
        if ridge:
            score += float(ridge) * float(np.dot(theta, theta))
        if not np.isfinite(score):
            return 1e100
        return float(score)
    except Exception:
        return 1e100


def _param_grad(theta, n, L, m, basis, temp, ridge):
    theta = np.asarray(theta, dtype=np.float64).ravel()
    zeros = np.zeros(theta.shape, dtype=np.float64)
    try:
        if not np.all(np.isfinite(theta)):
            return 1e100, zeros
        a = _theta_to_a(theta, basis)
        b = _conv_self(a, L, m)
        J, w, _ = _smooth_score_b(b, temp)
        if w is None:
            return 1e100, zeros
        score = 2.0 * float(n) * J
        gJ = _grad_weights(w, a, L, int(a.size))
        grad_a = 2.0 * float(n) * gJ
        dot = float(np.dot(a, grad_a))
        grad_v = grad_a - a * dot
        gth = basis.T.dot(grad_v)
        if ridge:
            score += float(ridge) * float(np.dot(theta, theta))
            gth += 2.0 * float(ridge) * theta
        gth = np.where(np.isfinite(gth), gth, 0.0)
        gth = np.clip(gth, -1e8, 1e8)
        if not np.isfinite(score):
            score = 1e100
        return float(score), gth
    except Exception:
        return 1e100, zeros


def _dedup_theta(thetas, tol=0.5):
    out = []
    for th in thetas:
        try:
            th = np.asarray(th, dtype=np.float64).ravel()
            if not np.all(np.isfinite(th)):
                continue
            keep = True
            for old in out:
                if float(np.linalg.norm(th - old)) <= tol:
                    keep = False
                    break
            if keep:
                out.append(th)
        except Exception:
            continue
    return out


def _random_prior_basis(basis, rng, scale=1.0):
    theta = rng.normal(0.0, float(scale), int(basis.shape[1]))
    return _theta_to_a(theta, basis)


def _line_search_mix(base, direction, L, m, n):
    base = _normalize(base)
    direction = _normalize(direction)
    best_a = base.copy()
    best_score, _ = _hard_score(base, n, L, m)
    for lam in (0.05, 0.10, 0.20, 0.35, 0.55, 0.80):
        h = (1.0 - lam) * base + lam * direction
        h = _normalize(h)
        sc, _ = _hard_score(h, n, L, m)
        if sc < best_score - 1e-13:
            best_score = sc
            best_a = h
    return best_a, best_score


def _mws_direction(f, L, m, n, prior, inner=20, eta_start=2.0, eta_end=40.0):
    f = _normalize(f)
    prior = _normalize(prior)
    inner = max(1, int(inner))
    F = np.fft.rfft(f, L)
    Frev = np.fft.rfft(f[::-1], L)

    x = prior.copy()
    y = np.full(m, 1.0 / float(m), dtype=np.float64)
    avg = np.zeros_like(x)
    count = 0
    logp = np.log(np.maximum(prior, 1e-300))

    for it in range(inner):
        frac = 0.0 if inner <= 1 else float(it) / float(inner - 1)
        eta = float(eta_start) + (float(eta_end) - float(eta_start)) * frac
        temp = 1.0 / max(eta, 1e-12)

        X = np.fft.rfft(x, L)
        u = np.fft.irfft(F * X, L)[:m]
        u = np.ascontiguousarray(u, dtype=np.float64)
        np.maximum(u, 0.0, out=u)

        _, y, _ = _smooth_score_b(float(n) * u, temp)
        if y is None:
            y = np.full(m, 1.0 / float(m), dtype=np.float64)
        y = np.maximum(y, 0.0)
        sy = float(y.sum())
        if not np.isfinite(sy) or sy <= 0.0:
            y = np.full(m, 1.0 / float(m), dtype=np.float64)
        else:
            y /= sy

        avg += x
        count += 1

        Y = np.fft.rfft(y, L)
        s = np.fft.irfft(Y * Frev, L)[n - 1:n - 1 + n]
        s = np.ascontiguousarray(s, dtype=np.float64)
        np.maximum(s, 0.0, out=s)

        ps = float(n) * s
        psmax = float(np.max(ps))
        z = -eta * (ps - psmax) + 0.8 * logp
        zmax = float(np.max(z)) if np.all(np.isfinite(z)) else 0.0
        z = z - zmax
        z = np.clip(z, -80.0, 0.0)
        e = np.exp(z)
        se = float(e.sum())
        if not np.isfinite(se) or se <= 0.0:
            x = prior.copy()
        else:
            x = e / se
            x = _normalize(x)

    if count <= 0:
        return prior
    mix = 0.5 * (avg / float(count)) + 0.5 * x
    return _normalize(mix)


def _local_search(a, L, m, n, rng, basis, time_limit):
    deadline = time.monotonic() + max(0.0, float(time_limit))
    a = _normalize(a)
    b = _conv_self(a, L, m)
    best_a = a.copy()
    best_hard = 2.0 * float(n) * float(np.max(b))

    temp_factor = 0.12
    min_factor = 0.001
    max_factor = 0.30
    eta = 0.30
    no_accept = 0
    stale = 0
    it = 0

    while time.monotonic() < deadline:
        rem = deadline - time.monotonic()
        if rem < 1.0:
            break

        maxb = float(np.max(b))
        current_hard = 2.0 * float(n) * maxb
        if current_hard < best_hard - 1e-12:
            best_hard = current_hard
            best_a = a.copy()
            stale = 0
            if it % 100 == 0:
                print(f"  local: {best_hard:.8f}", flush=True)
        else:
            stale += 1

        if stale > 700:
            a = best_a.copy()
            b = _conv_self(a, L, m)
            maxb = float(np.max(b))
            current_hard = 2.0 * float(n) * maxb
            temp_factor = max(min_factor, min(0.12, temp_factor * 0.5))
            eta = 0.30
            no_accept = 0
            stale = 350

        if no_accept > 35:
            if basis is not None:
                prior = _random_prior_basis(basis, rng, 1.0)
            else:
                prior = _normalize(rng.normal(0.0, 1.0, n))
            mix = 0.10 + 0.30 * float(rng.random())
            a = _normalize((1.0 - mix) * best_a + mix * prior)
            b = _conv_self(a, L, m)
            maxb = float(np.max(b))
            current_hard = 2.0 * float(n) * maxb
            temp_factor = min(max_factor, max(0.05, 0.05 + 0.15 * float(rng.random())))
            eta = 0.30
            no_accept = 0
            stale = min(stale, 120)

        if time.monotonic() > deadline - 1.0:
            break

        if stale > 250 and basis is not None and it % 10 == 0:
            prior = _random_prior_basis(basis, rng, 1.0)
            g = _mws_direction(best_a, L, m, n, prior, 20)
            h, hscore = _line_search_mix(best_a, g, L, m, n)
            if hscore < best_hard - 1e-12:
                best_a = h
                best_hard = hscore
                a = h.copy()
                b = _conv_self(a, L, m)
                no_accept = 0
                stale = 0
                temp_factor = max(min_factor, temp_factor * 0.8)
            else:
                mix = 0.05 + 0.15 * float(rng.random())
                a = _normalize((1.0 - mix) * best_a + mix * g)
                b = _conv_self(a, L, m)

        temp = max(1e-18, maxb * max(min_factor, temp_factor))
        J, w, _ = _smooth_score_b(b, temp)
        if w is None:
            a = best_a.copy()
            b = _conv_self(a, L, m)
            continue

        g = _grad_weights(w, a, L, int(a.size))
        gmax = float(np.max(g))
        if not np.isfinite(gmax) or gmax <= 1e-300:
            no_accept += 5
            if basis is not None:
                prior = _random_prior_basis(basis, rng, 0.7)
            else:
                prior = _normalize(rng.normal(0.0, 1.0, n))
            a = _normalize(0.9 * a + 0.1 * prior)
            b = _conv_self(a, L, m)
            it += 1
            continue

        d = g / gmax
        accepted = False
        chosen_step = None
        for step in (eta, 0.5 * eta, 0.12 * eta, 1.6 * eta):
            if step <= 1e-10:
                continue
            a2 = a * np.exp(-step * d)
            a2 = _normalize(a2)
            b2 = _conv_self(a2, L, m)
            J2, _, _ = _smooth_score_b(b2, temp)
            hard2 = 2.0 * float(n) * float(np.max(b2))
            if J2 < J - 1e-12 or hard2 < current_hard - 1e-12:
                a = a2
                b = b2
                accepted = True
                chosen_step = step
                break

        if accepted:
            no_accept = 0
            if chosen_step == eta:
                eta = min(1.10, eta * 1.08)
            elif chosen_step == 1.6 * eta:
                eta = min(1.10, eta * 1.15)
            else:
                eta = max(0.03, eta * 0.90)
            temp_factor = max(min_factor, temp_factor * 0.985)
        else:
            no_accept += 1
            eta = max(0.02, eta * 0.80)

        it += 1

    return best_a, best_hard


def _add_candidate(arrays, scores, a, score, maxlen=128):
    if a is None:
        return
    try:
        score = float(score)
    except Exception:
        return
    if not np.isfinite(score):
        return
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    if a.sum() <= 0.0:
        return

    if len(arrays) < maxlen:
        arrays.append(a.copy())
        scores.append(score)
        return

    worst = 0
    worst_val = scores[0]
    for i in range(1, len(scores)):
        if scores[i] > worst_val:
            worst_val = scores[i]
            worst = i
    if score < worst_val:
        arrays[worst] = a.copy()
        scores[worst] = score


def _sort_indices(scores):
    order = list(range(len(scores)))
    order.sort(key=scores.__getitem__)
    return order


def _internal_exact(seq):
    try:
        a = np.asarray(seq, dtype=np.float64).ravel()
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.maximum(a, 0.0)
        a = np.minimum(a, 1000.0)
        n = int(a.size)
        if n <= 0:
            return float(np.inf)
        s = float(a.sum())
        if s < 0.01:
            return float(np.inf)
        b = np.convolve(a, a)
        mb = float(np.max(b))
        return float(2.0 * n * mb / (s * s))
    except Exception:
        return float(np.inf)

def _make_valid_sequence(a, n):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size != n:
        a = np.ones(max(1, n), dtype=np.float64) / float(max(1, n))
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a = np.ones(max(1, n), dtype=np.float64)
    return [float(x) for x in a]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    budget = _safe_budget(budget_s, 2400.0)
    start_time = time.monotonic()
    hard_deadline = start_time + max(15.0, budget - 3.0)

    rng = np.random.default_rng(int(seed) % (2**32))
    n = _choose_n()
    L = _fft_len(n)
    m = 2 * n - 1

    parent = _get_parent(n)

    cand_arrays = []
    cand_scores = []

    best_exact = float(np.inf)
    best_seq = None

    if parent is not None:
        parent = _normalize(parent)
        p_score, _ = _hard_score(parent, n, L, m)
        _add_candidate(cand_arrays, cand_scores, parent, p_score)
        seqp = [float(x) for x in parent]
        pe = _exact_score(seqp)
        if np.isfinite(pe):
            best_exact = pe
            best_seq = seqp
            print(f"parent exact: {best_exact:.8f}", flush=True)
    else:
        uniform = _normalize(np.ones(n, dtype=np.float64))
        u_score, _ = _hard_score(uniform, n, L, m)
        _add_candidate(cand_arrays, cand_scores, uniform, u_score)

    fallback = _make_valid_sequence(
        cand_arrays[0] if cand_arrays else np.ones(n, dtype=np.float64), n
    )

    # Low-dimensional parameterized global/local search.
    basis = _build_basis(n)
    dim = int(basis.shape[1])
    if dim <= 0:
        basis = np.ones((n, 1), dtype=np.float64)
        dim = 1

    bounds = [(-8.0, 8.0) for _ in range(dim)]
    ridge = 1e-7
    temp_scale = 1.0 / (2.0 * float(n))

    param_deadline = start_time + min(720.0, 0.30 * budget)
    if budget <= 240.0:
        param_deadline = start_time + max(30.0, 0.15 * budget)

    theta_starts = []

    if _HAS_SCIPY and time.monotonic() < param_deadline - 120.0:
        de_temp = 0.15 * temp_scale
        try:
            de_res = _scipy_differential_evolution(
                _param_scalar,
                bounds,
                args=(n, L, m, basis, de_temp, ridge),
                maxiter=35,
                maxfev=900,
                popsize=12,
                tol=1e-5,
                seed=int(seed) % (2**32),
                polish=False,
                update_interval=5,
                disp=False,
            )
            if de_res is not None and hasattr(de_res, "x"):
                th_de = np.asarray(de_res.x, dtype=np.float64).ravel()
                th_de = np.clip(th_de, -8.0, 8.0)
                theta_starts.append(th_de)
                a_de = _theta_to_a(th_de, basis)
                sc_de, _ = _hard_score(a_de, n, L, m)
                _add_candidate(cand_arrays, cand_scores, a_de, sc_de)
                if np.isfinite(sc_de):
                    print(f"DE best soft-objective candidate hard: {sc_de:.8f}", flush=True)
        except Exception:
            pass

    theta_starts.append(np.zeros(dim, dtype=np.float64))

    if parent is not None:
        theta_starts.append(_a_to_theta(parent, basis))

    beta_pq = (
        (1.0, 1.0),
        (2.0, 1.0),
        (1.0, 2.0),
        (3.0, 1.0),
        (1.0, 3.0),
        (2.0, 2.0),
        (4.0, 1.0),
        (1.0, 4.0),
    )
    for p, q in beta_pq:
        a_beta = _beta_shape(n, p, q)
        theta_starts.append(_a_to_theta(a_beta, basis))
        sc_beta, _ = _hard_score(a_beta, n, L, m)
        _add_candidate(cand_arrays, cand_scores, a_beta, sc_beta)

    for _ in range(4):
        theta_starts.append(np.clip(rng.normal(0.0, 1.0, dim), -8.0, 8.0))

    theta_starts = _dedup_theta(theta_starts, tol=0.45)
    if len(theta_starts) > 10:
        theta_starts = theta_starts[:10]

    if _HAS_SCIPY:
        param_temps = (0.08 * temp_scale, 0.03 * temp_scale, 0.012 * temp_scale)
        for idx, th0 in enumerate(theta_starts):
            now = time.monotonic()
            if now > param_deadline - 25.0:
                break

            th0 = np.clip(th0, -8.0, 8.0)
            a0 = _theta_to_a(th0, basis)
            sc0, _ = _hard_score(a0, n, L, m)
            _add_candidate(cand_arrays, cand_scores, a0, sc0)

            th_cur = th0.copy()
            for temp in param_temps:
                now = time.monotonic()
                if now > param_deadline - 8.0:
                    break
                try:
                    res = _scipy_minimize(
                        _param_grad,
                        th_cur,
                        jac=True,
                        args=(n, L, m, basis, temp, ridge),
                        method="L-BFGS-B",
                        bounds=bounds,
                        options={
                            "maxiter": 120,
                            "maxfun": 140,
                            "ftol": 1e-12,
                            "gtol": 1e-8,
                            "maxls": 20,
                        },
                    )
                    if res is not None and hasattr(res, "x"):
                        th1 = np.asarray(res.x, dtype=np.float64).ravel()
                        th1 = np.where(np.isfinite(th1), th1, th_cur)
                        th1 = np.clip(th1, -8.0, 8.0)
                        a1 = _theta_to_a(th1, basis)
                        sc1, _ = _hard_score(a1, n, L, m)
                        _add_candidate(cand_arrays, cand_scores, a1, sc1)
                        th_cur = th1
                except Exception:
                    pass

    # Some cheap non-linear transformations of promising candidates.
    if cand_arrays:
        order0 = _sort_indices(cand_scores)
        for idx in order0[:5]:
            a = cand_arrays[idx]
            sc, _ = _hard_score(a, n, L, m)
            for transform in (0, 1):
                try:
                    if transform == 0:
                        a2 = _normalize(np.sqrt(np.maximum(a, 1e-300)))
                    else:
                        a2 = _normalize(0.75 + 0.25 * (1.0 - a / (float(np.max(a)) + 1e-300)))
                    sc2, _ = _hard_score(a2, n, L, m)
                    _add_candidate(cand_arrays, cand_scores, a2, sc2)
                except Exception:
                    pass

    # Raw multi-start local refinement.
    raw_deadline = hard_deadline - 45.0
    if cand_arrays:
        order = _sort_indices(cand_scores)
        top = [order[i] for i in range(min(6, len(order)))]
    else:
        top = []

    for rank, idx in enumerate(top):
        now = time.monotonic()
        rem = raw_deadline - now
        if rem < 35.0:
            break
        remaining_count = max(1, len(top) - rank)
        base_slice = rem / float(remaining_count)
        limit = min(520.0, base_slice + (120.0 if rank == 0 else 40.0))
        limit = max(35.0, limit)
        print(f"raw refine candidate {rank}: {cand_scores[idx]:.8f} for {limit:.1f}s", flush=True)
        ref_a, ref_score = _local_search(
            cand_arrays[idx], L, m, n, rng, basis, limit
        )
        _add_candidate(cand_arrays, cand_scores, ref_a, ref_score)

    # Final exact checks.
    final_arrays = []
    for a in cand_arrays:
        a = np.asarray(a, dtype=np.float64).ravel()
        if np.all(np.isfinite(a)):
            final_arrays.append(a)
    if best_seq is not None:
        final_arrays.append(np.asarray(best_seq, dtype=np.float64).ravel())

    unique = []
    for a in final_arrays:
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size != n or not np.all(np.isfinite(a)):
            continue
        dup = False
        for b in unique:
            if float(np.linalg.norm(a - b)) <= 1e-6:
                dup = True
                break
        if not dup:
            unique.append(a)

    if unique:
        uidx = list(range(len(unique)))
        uidx.sort(key=lambda: None)  # placeholder to avoid lambda? see below
    # The above sort is not useful; instead score them.
    scored_final = []
    for a in unique:
        sc, _ = _hard_score(a, n, L, m)
        if np.isfinite(sc):
            scored_final.append((sc, a))
    scored_final.sort(key=lambda x: x[0])

    exact_budget = min(75.0, max(10.0, (hard_deadline - start_time) * 0.25))
    exact_start = time.monotonic()
    checked = 0
    for sc, a in scored_final[:36]:
        now = time.monotonic()
        if now > hard_deadline - 7.0 or checked >= 30 or now - exact_start > exact_budget:
            break
        seq = _make_valid_sequence(a, n)
        val = _exact_score(seq)
        checked += 1
        if np.isfinite(val):
            if val < best_exact:
                best_exact = val
                best_seq = seq
                print(f"exact improved: {best_exact:.8f} (candidate internal {sc:.8f})", flush=True)

    if best_seq is None or not np.isfinite(best_exact):
        best_seq = fallback
        best_exact = float("inf")

    # Safety: ensure a valid finite sequence.
    final_seq = _make_valid_sequence(best_seq, n) if best_seq is not None else fallback
    if len(final_seq) != n or any((not np.isfinite(float(x))) or float(x) < 0.0 for x in final_seq):
        final_seq = fallback

    print(f"final reported exact/last: {best_exact:.8f}", flush=True)
    return final_seq
