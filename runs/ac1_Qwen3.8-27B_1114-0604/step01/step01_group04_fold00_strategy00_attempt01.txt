def evaluate_sequence(sequence):
    """Evaluates a sequence of coefficients with enhanced security checks."""
    if not isinstance(sequence, list):
        print("Invalid sequence type")
        return float('inf')

    if not sequence:
        return float('inf')

    # Convert to numpy but with floats and within range [0, 1000]
    for x in sequence:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return float('inf')
        if np.isnan(x) or np.isinf(x):
            return float('inf')
    seq = np.array(sequence, dtype=np.float64)
    normalized = [float(x) for x in seq]
    # Apply max to 0 and min to 1000
    normalized = [max(0.0, x) for x in normalized]
    normalized = [min(1000.0, x) for x in normalized]

    # Return best candidate found along with evaluation metric
    n = len(normalized)
    b_seq = np.convolve(normalized, normalized)  # Mode is 'full'
    max_b = max(b_seq)
    sum_a = sum(normalized)
    
    # In our normalization, sum_a is at the level of 1000 multiplied by n should be fine
    if sum_a < 0.01:
        return float('inf')
        
    return float(2.0 * n * max_b / sum_a**2)
