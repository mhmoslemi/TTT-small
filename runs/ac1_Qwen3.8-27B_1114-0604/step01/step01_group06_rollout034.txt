Let me re-examine. The issue is that the convolution `(f*g)[j] = sum_k f[k]*g[j-k]` requires both f[k] and g[j-k] to be in range. When f and g have different lengths, the valid range of k is restricted.

The second term: `sum_k p[k]*a[j-k]` where k ranges over [0, 2n-2] (length of p = 2n-1) and j-k ranges over [0, n-1] (length of a = n).

The first term: `sum_k p[k]*a[k-j]` where k ranges over [0, 2n-2] and k-j ranges over [0, n-1].

These are different because the constraints on k are different:
- Second: 0 <= k <= 2n-2 AND 0 <= j-k <= n-1, i.e., j-n+1 <= k <= j
- First: 0 <= k <= 2n-2 AND 0 <= k-j <= n-1, i.e., j <= k <= j+n-1

So they're cross-correlations at different lags. Specifically:
- Second term = (p convolved with a)[j] = linear convolution
- First term = (p cross-correlated with a)[j] = cross-correlation

For the first term: `sum_k p[k]*a[k-j]`. Let me substitute l = k-j, so k = l+j:
= `sum_l p[l+j]*a[l]` where l ranges over [max(0,-j), min(2n-2-j, n-1)]
= cross-correlation of p and a at lag j (shift p by j and dot with a)

- The element at index j (0-based) gives `sum_k p[k] * a[k-j]` for the valid range.

The circular cross-correlation of x and h (both length N) is:
`(x ⋆ h)[n] = sum_k x[k] * conj(h[k-n]) = sum_k x[k+n] * conj(h[k])`

In the frequency domain: `ifft(X * conj(H))`.

So `ifft(fft(p,N) * conj(fft(a,N)))[j] = sum_k p[k] * a[k-j]` (circular, with a padded to length N).

With zero-padding to N >= 3n-2, this gives the LINEAR cross-correlation for j = 0, 1, ..., 3n-3.

And `sum_k p[k]*a[k-j]` for a specific j is exactly what we want for the first term!

- First term: `c1 = ifft(fft(p,N) * conj(fft(a,N)))[:n]` gives `c1[j] = sum_k p[k]*a[k-j]` for j=0,...,n-1 ✓

And `g = c1 + c2`, then `grad_z = a * (g - dot(g, a))`.

```python
import numpy as np
import time

# Global state
_G_N = 0
_G_NFFT = 0
_G_TAU = 1e-4
_G_TAU_MIN = 1e-7


def _init_state(n, tau=1e-4):
    global _G_N, _G_NFFT, _G_TAU, _G_TAU_MIN
    _G_N = int(n)
    _G_NFFT = 1 << (3 * _G_N - 2).bit_length()
    _G_TAU = float(tau)
    _G_TAU_MIN = 1e-7


def _fft_autocorr(a, N=None):
    """Compute linear autocorrelation (convolution with itself) using FFT."""
    if N is None:
        N = _G_NFFT
    n = len(a)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1].real
    return np.maximum(c, 0.0)


def _score(a):
    """Compute the objective: 2n * max(conv) / (sum a)^2."""
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    c = _fft_autocorr(a)
    m = float(np.max(c))
    return 2.0 * n * m / (s * s)


def _softmax(z):
    """Stable softmax."""
    z = np.asarray(z, dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = float(np.sum(e))
    if s < 1e-300:
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _logsumexp_max(c, tau):
    """Compute tau * logsumexp(c/tau), a smooth approximation to max(c)."""
    c = np.asarray(c, dtype=np.float64)
    m = float(np.max(c))
    if not np.isfinite(m):
        return 1e12, np.zeros_like(c)
    x = (c - m) / tau
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    se = float(np.sum(e))
    if se < 1e-300:
        val = m
        p = np.zeros_like(c)
        p[np.argmax(c)] = 1.0
    else:
        val = m + tau * np.log(se)
        p = e / se
    return val, p


def _grad_z(a, p, N=None):
    """Gradient of the soft-max objective w.r.t. log-space parameters."""
    if N is None:
        N = _G_NFFT
    n = _G_N
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    # g[j] = sum_k p[k]*(a[k-j] + a[j-k])
    c1 = np.fft.irfft(fp * np.conjugate(fa), N)[:n]
    c2 = np.fft.irfft(fp * fa, N)[:n]
    g = c1 + c2
    S = float(np.dot(g, a))
    grad = a * (g - S)
    return grad


def _adam_step(state, grad, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step. Returns updated z and state."""
    z, m, v, t = state
    t = t + 1
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * (grad * grad)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    z_new = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return (z_new, m, v, t)


def _cosine_lr(t, T, lr0, lr_min=0):
    """Cosine annealing learning rate schedule."""
    return lr_min + 0.5 * (lr0 - lr_min) * (1 + np.cos(np.pi * t / T))


def _generate_starts(n, rng, n_starts=20):
    """Generate diverse starting points in log-space."""
    starts = []
    u = np.linspace(0, 1, n, endpoint=False)
    
    # Try to use the provided sequence as a start
    try:
        g = globals().get('height_sequence_1')
        if g is not None:
            g = np.asarray(g, dtype=np.float64)
            if g.size > 1 and np.all(np.isfinite(g)) and np.sum(g) > 0:
                # Resample to length n
                g_clipped = np.clip(g, 1e-10, 1000.0)
                x_old = np.linspace(0, 1, g_clipped.size, endpoint=False)
                x_new = np.linspace(0, 1, n, endpoint=False)
                g_resampled = np.interp(x_new, x_old, np.log(g_clipped + 1e-20))
                z = g_resampled - np.mean(g_resampled)
                z = np.clip(z, -20, 20)
                a = _softmax(z)
                sc = _score(a)
                starts.append((sc, z))
                print(f"  Provided sequence: score={sc:.6f}")
    except Exception:
        pass
    
    # Cosine-based starts
    for freq in [1, 2, 3, 5, 8, 13]:
        for amp in [0.1, 0.3, 0.5, 0.8]:
            f = 1.0 + amp * np.cos(2 * np.pi * freq * u)
            f = np.maximum(f, 1e-6)
            z = np.log(f) - np.mean(np.log(f))
            z = np.clip(z, -20, 20)
            a = _softmax(z)
            sc = _score(a)
            starts.append((sc, z))
    
    # Linear ramp
    for slope in [0.5, 1.0, 2.0, 3.0]:
        f = 1.0 + slope * u
        z = np.log(f) - np.mean(np.log(f))
        z = np.clip(z, -20, 20)
        a = _softmax(z)
        sc = _score(a)
        starts.append((sc, z))
    
    # Random smooth (low-frequency Fourier)
    for _ in range(8):
        f = np.ones(n)
        for m in range(1, 10):
            c = rng.standard_normal() * 0.15 / m
            f += c * np.sin(2 * np.pi * m * u + rng.standard_normal())
        f = np.maximum(f, 1e-6)
        z = np.log(f) - np.mean(np.log(f))
        z = np.clip(z, -20, 20)
        a = _softmax(z)
        sc = _score(a)
        starts.append((sc, z))
    
    # Two-level block
    for split in [0.3, 0.4, 0.5, 0.6]:
        for ratio in [1.5, 2.0, 3.0, 5.0]:
            f = np.ones(n)
            k = int(n * split)
            f[:k] = 1.0
            f[k:] = ratio
            z = np.log(f) - np.mean(np.log(f))
            z = np.clip(z, -20, 20)
            a = _softmax(z)
            sc = _score(a)
            starts.append((sc, z))
    
    starts.sort(key=lambda x: x[0])
    return starts[:n_starts]


def _optimize_adam(z0, n_iters=5000, lr0=0.05, deadline=None):
    """Run Adam optimization from z0."""
    n = _G_N
    N = _G_NFFT
    
    z = z0.copy()
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0
    
    # Adaptive temperature
    tau = _G_TAU
    
    best_score = np.inf
    best_z = z0.copy()
    
    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Anneal temperature
        frac = it / max(1, n_iters)
        tau = max(_G_TAU_MIN, _G_TAU * (1 - frac) + _G_TAU_MIN * frac)
        
        a = _softmax(z)
        c = _fft_autocorr(a, N)
        val, p = _logsumexp_max(c, tau)
        
        current_score = 2.0 * n * val  # (sum a = 1, so this is proportional)
        # Actually, score = 2n * max_c / (sum_a)^2. With sum_a = 1, it's 2n * max_c.
        # But val is the smooth approximation to max_c.
        
        if current_score < best_score:
            best_score = current_score
            best_z = z.copy()
        
        if it % 100 == 0:
            exact_score = _score(a)
            if exact_score < best_score:
                best_score = exact_score
                best_z = z.copy()
            print(f"    Adam it={it}, score={exact_score:.6f}, best={best_score:.6f}, tau={tau:.2e}")
        
        # Compute gradient
        grad = _grad_z(a, p, N)
        
        # Learning rate schedule
        lr = _cosine_lr(t, n_iters, lr0, lr_min=lr0 * 0.01)
        
        # Adam step
        z, m, v, t = _adam_step((z, m, v, t), grad, lr)
        z = np.clip(z, -30, 30)
    
    return best_z, best_score

            if float(np.max(c_rand)) < F:
```
