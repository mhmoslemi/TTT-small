```python
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def full_conv(a):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def solve_lp(a, M, n):
    """Solve: maximize sum(g) s.t. (a*g)_k <= M for all k, g >= 0"""
    # Build constraint matrix C where C[k, j] = a[k-j] if valid
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])
    
    C = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Maximize sum(g) = minimize -sum(g)
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=C, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    
    if res.status != 0:
        return None
    return res.x

def lp_optimize(a, time_budget, rng):
    """Alternating LP optimization."""
    n = a.size
    a = a / a.sum()
    best_a = a.copy()
    best_sc = 2.0 * n * float(np.max(full_conv(a)))
    
    t0 = time.monotonic()
    iters = 0
    last_improve = 0
    t_blend = 1.0  # Full LP update initially
    
    while time.monotonic() - t0 < time_budget:
        conv_a = full_conv(a)
        M = float(np.max(conv_a))
        cur_sc = 2.0 * n * M
        
        g = solve_lp(a, M, n)
        if g is None:
            break
        
        sum_g = float(np.sum(g))
        gap = sum_g - 1.0
        
        if gap < 1e-6:
            # At or near local optimum
            break
        
        # Update: blend with LP solution
        g_norm = g / sum_g
        t_eff = min(t_blend, gap)  # Limit step by gap
        a_new = (1 - t_eff) * a + t_eff * g_norm
        a_new = np.maximum(a_new, 0)
        a_new = a_new / a_new.sum()
        
        new_conv = full_conv(a_new)
        new_sc = 2.0 * n * float(np.max(new_conv))
        
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_new.copy()
            last_improve = iters
        elif iters - last_improve > 50:
            # Stuck, try different approach
            a = best_a.copy()
            t_blend = max(0.1, t_blend * 0.5)
        
        a = a_new
        iters += 1
        
        # Adapt blend factor
        if iters % 20 == 0:
            t_blend = min(1.0, t_blend * 1.1)
    
    return best_a, best_sc
```

"""
Minimizes the evaluation function 2*n*max(convolve(a,a))/sum(a)^2 using:
1. Alternating LP optimization (fix one variable, optimize the other via LP)
2. Mirror descent / exponential weights for fast iterations
3. Local polish with random perturbations
4. Multiple sequence lengths and diverse initializations
Uses FFT-based convolution for O(n log n) per evaluation.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

# --- Core functions ---

def _full_conv(a):
    """Full linear convolution of a with itself via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def _internal_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2 for non-negative a."""
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = _full_conv(a)
    return 2.0 * n * float(np.max(b)) / (s * s)

def _score_norm1(a):
    """Score assuming sum(a) = 1. Returns 2*n*max(conv(a,a))."""
    n = a.size
    b = _full_conv(a)
    return 2.0 * n * float(np.max(b))

# --- LP-based optimization ---

def _build_conv_matrix(a, n):
    """Sparse matrix C: C[k,j] = a[k-j] if 0<=k-j<n, else 0."""
    nnz_per_row = []
    for k in range(2*n-1):
        jmin = max(0, k-n+1)
        jmax = min(k, n-1)
        nnz_per_row.append(jmax - jmin + 1)
    total_nnz = sum(nnz_per_row)
    
    rows = np.empty(total_nnz, dtype=np.int32)
    cols = np.empty(total_nnz, dtype=np.int32)
    vals = np.empty(total_nnz, dtype=np.float64)
    idx = 0
    for k in range(2*n-1):
        jmin = max(0, k-n+1)
        jmax = min(k, n-1)
        for j in range(jmin, jmax+1):
            rows[idx] = k
            cols[idx] = j
            vals[idx] = a[k-j]
            idx += 1
    return csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))

def _solve_lp(a, M):
    """Solve max 1^T g s.t. C(a)*g <= M*1, g >= 0. Returns g or None."""
    n = a.size
    C = _build_conv_matrix(a, n)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None) for _ in range(n)]
    try:
        res = linprog(c_obj, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
        if res.status == 0:
            return res.x
        elif res.status == 3:  # infeasible? shouldn't happen
            return None
        else:
            return None
    except Exception:
        return None

def _mirror_descent(a, budget, rng, verbose=False):

        shift = b / tau
        shift -= shift.max()
        w = np.exp(shift)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(b.size)
            w[int(np.argmax(b))] = 1.0
        else:
            w = w / ws
        
        grad = 2.0 * np.correlate(w, a, mode='valid')
        
        # Mirror descent update
        log_a = np.log(np.maximum(a, 1e-300)) - eta * grad
        log_a -= log_a.max()
        a_new = np.exp(log_a)
        s_new = a_new.sum()
        if s_new > 1e-300:
            a_new = a_new / s_new
        else:
            a_new = a

            eta = 0.1 + 0.4 * rng.random()
            tau = 0.01 + 0.04 * rng.random()
            last_improve = iters
