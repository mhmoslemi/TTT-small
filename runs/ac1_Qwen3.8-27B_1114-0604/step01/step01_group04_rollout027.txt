"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using cyclic coordinate descent
with vectorized 1D sub-optimization, multiple structured initializations,
diversified restarts, and local polishing.
"""

import numpy as np
import time


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = a.size
    b = full_conv(a)
    s = float(np.sum(a))
    if s < 1e-10:
        return np.inf
    return float(2.0 * n * float(np.max(b)) / (s * s))


def init_exp_ramp(n, alpha, rng=None):
    """Exponential ramp: a[j] = exp(alpha * j / n), normalized."""
    j = np.arange(n, dtype=np.float64)
    a = np.exp(alpha * j / n)
    if rng is not None:
        a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_power_ramp(n, p, rng=None):
    """Power ramp: a[j] = ((j+1)/n)^p, normalized."""
    j = np.arange(n, dtype=np.float64)
    a = ((j + 1) / n) ** p
    if rng is not None:
        a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_sigmoid(n, alpha, center=0.5, rng=None):
    j = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(-alpha * (j / max(n-1, 1) - center)))
    if rng is not None:
        a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_two_bump(n, c1, c2, sigma, rng=None):
    j = np.arange(n, dtype=np.float64)
    a = np.exp(-((j - c1) / sigma) ** 2) + np.exp(-((j - c2) / sigma) ** 2)
    if rng is not None:
        a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_random_asym(n, rng):
    """Random asymmetric initialization."""
    a = np.abs(rng.standard_normal(n))
    # Add asymmetry: multiply by a ramp
    j = np.arange(n, dtype=np.float64)
    ramp = 0.5 + 0.5 * j / n
    a *= ramp
    a += 0.1
    return a / a.sum()


def get_initializations(n, rng):
    """Generate a diverse set of initializations for length n."""
    inits = []
    # Exponential ramps with various alphas (positive = increasing, negative = decreasing)
    for alpha in [0.5, 1.0, 2.0, 3.0, 5.0, -0.5, -1.0, -2.0]:
        inits.append(init_exp_ramp(n, alpha, rng))
    # Power ramps
    for p in [0.5, 1.0, 1.5, 2.0, 3.0]:
        inits.append(init_power_ramp(n, p, rng))
    # Sigmoids
    for alpha in [2, 4, 8]:
        inits.append(init_sigmoid(n, alpha, 0.3, rng))
        inits.append(init_sigmoid(n, alpha, 0.7, rng))
    # Two bumps (asymmetric)
    inits.append(init_two_bump(n, n*0.2, n*0.7, n*0.1, rng))
    inits.append(init_two_bump(n, n*0.15, n*0.6, n*0.15, rng))
    # Random asymmetric
    for _ in range(3):
        inits.append(init_random_asym(n, rng))
    return inits


def optimize_coordinate(b, a, i, S_total):
    """
    Optimize coordinate i of a, given the current full convolution b.
    Returns the optimal value for a[i].
    """
    n = a.size
    x0 = a[i]
    S = S_total - x0  # sum of all other coordinates
    
    # Determine affected lags: k in [i, n-1+i]
    lo = i
    hi = n - 1 + i
    
    # Compute unaffected max
    mask = np.ones(2 * n - 1, dtype=bool)
    mask[lo:hi+1] = False
    if mask.any():
        C_max = float(np.max(b[mask]))
    else:
        C_max = 0.0
    
    # For affected lags, precompute B_k and c_k
    # For k in [lo, hi], k ≠ 2i: b[k] = B_k + 2*a[i]*a[k-i], so B_k = b[k] - 2*x0*a[k-i], c_k = 2*a[k-i]
    # For k = 2i: b[k] = B_k + a[i]^2, so B_k = b[k] - x0^2, and the x-dependent term is x^2
    
    B_arr = np.zeros(hi - lo + 1)
    c_arr = np.zeros(hi - lo + 1)
    has_quad = (2 * i >= lo) and (2 * i <= hi)
    quad_idx = -1
    B_quad = 0.0
    
    for idx, k in enumerate(range(lo, hi + 1)):
        other = k - i
        if other < 0 or other >= n:
            B_arr[idx] = b[k]
            c_arr[idx] = 0.0
        elif k == 2 * i:
            B_quad = b[k] - x0 * x0
            quad_idx = idx
            B_arr[idx] = 0.0
            c_arr[idx] = 0.0
        else:
            B_arr[idx] = b[k] - 2.0 * x0 * a[other]
            c_arr[idx] = 2.0 * a[other]

        if x < -1e-15:
            return np.inf
        x = max(0.0, x)
        new_sum = S + x
        if new_sum < 1e-10:
            return np.inf
        # Compute max of affected lags
        vals = B_arr + c_arr * x  # linear parts
        if has_quad:
            vals[quad_idx] = B_quad + x * x
        max_affected = float(np.max(vals))
        max_conv = max(C_max, max_affected)
        return 2.0 * n * max_conv / (new_sum * new_sum)
    
    # Golden section search for minimum of f(x) over x in [0, x_max]

    lo_x = x_scan[max(0, idx_min - 1)]
    hi_x = x_scan[min(n_scan - 1, idx_min + 1)]

    gmr = (np.sqrt(5.0) - 1.0) / 2.0
    a_g = lo_x + (1.0 - gmr) * (hi_x - lo_x)
    b_g = lo_x + gmr * (hi_x - lo_x)
    fa = f(a_g)
    fb = f(b_g)
    
    for _ in range(40):
        if hi_x - lo_x < 1e-10:

            hi_x = b_g
            b_g = a_g
            fb = fa

            fa = fb

            fb = f(b_g)

    f_opt = f(x_opt)
    f_orig = f(x0)
    
    if f_opt < f_orig:

    return x0

    a: input array (will be normalized to sum 1).
    time_budget: seconds to run.
    Returns (best_a, best_score).
    """

    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a /= s
    
    best_a = a.copy()
    best_sc = compute_score(a)

    cycle = 0
    last_improve = 0
    stall = 0
    
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        S_total = float(np.sum(a))
        
        improved_this_cycle = False
        for i in range(n):
            x_new = optimize_coordinate(b, a, i, S_total)
            if x_new != a[i]:
                a[i] = x_new
                S_total += x_new - a[i]  # wait, this is wrong, a[i] was already updated
                # Actually: a[i] was x_old, now it's x_new
                # S_total was sum of all = old_S_total
                # New S_total = old_S_total - x_old + x_new
                # But I already set a[i] = x_new, so S_total needs update
                # Let me just recompute
                improved_this_cycle = True
        
        # Recompute S_total after the cycle
        S_total = float(np.sum(a))
        
        # Update best
        sc = 2.0 * n * float(np.max(full_conv(a))) / (S_total * S_total)

                a = np.maximum(a + perturbation, 0)
                s = a.sum()

                    a /= s
                else:
                    a = best_a.copy()
                stall = 0
                last_improve = cycle
