def _pow2(m):
    return 1 << max(0, (m - 1).bit_length())

def _proj_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return v / s

def _conv_fft(a, L):
    """Full self-convolution via FFT."""
    f = np.fft.rfft(a, L)
    return np.fft.irfft(f * f, L)[:2 * len(a) - 1]

def _score_internal(a, L):
    """Score = 2n * max(conv) / sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 1e30
    c = _conv_fft(a, L)
    return float(2.0 * n * c.max() / (s * s))

def _softmax_weights(c, tau):
    """Compute softmax(c/tau) weights."""
    cm = c.max()
    z = (c - cm) / tau
    z = np.clip(z, -100.0, 0.0)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
        return w
    return e / s

def _grad_from_weights(a, w, L2):
    """
    Compute grad[j] = 2 * sum_i a[i] * w[i+j] for j=0..n-1.
    This is the gradient of the soft-max objective.
    """
    n = len(a)
    Fa = np.fft.rfft(a, L2)
    Fw = np.fft.rfft(w, L2)
    cross = np.fft.irfft(Fa * np.conj(Fw), L2)
    return 2.0 * cross[:n]

def _greedy_refine(a, L, deadline, rng, verbose=True):

    Move mass from high-contributing coordinates to low-contributing ones.
    """

    a = _proj_simplex(a)
    best_a = a.copy()
    c = _conv_fft(a, L)

    while time.monotonic() < deadline:

        max_c = c.max()

            best_mc = max_c
            best_a = a.copy()

        k_star = c.argmax()
        
        # Gradient at k_star: 2 * a[k_star - j] for valid j
        # We want to reduce c[k_star], so decrease a[j] where a[k_star-j] is large
        # and increase a[j] where a[k_star-j] is small
        
        # Compute the contribution vector: contrib[j] = a[k_star - j] (if valid)
        contrib = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                contrib[j] = a[idx]
        
        # Find top and bottom contributors
        top_j = np.argmax(contrib)
        bottom_j = np.argmin(contrib)
        
        if top_j == bottom_j:
            continue
        
        # Move a small amount from top_j to bottom_j

        if delta < 1e-15:
            continue
        
        a_new = a.copy()
        a_new[top_j] -= delta
        a_new[bottom_j] += delta
        a_new = _proj_simplex(a_new)
        
        c_new = _conv_fft(a_new, L)
        mc_new = c_new.max()
        
        if mc_new < max_c:
            a = a_new
        else:
            # Try a different pair
            top_candidates = np.argsort(contrib)[::-1][:10]
            bottom_candidates = np.argsort(contrib)[:10]
            improved = False
            for ti in top_candidates:
                for bi in bottom_candidates:
                    if ti == bi:
                        continue
                    d2 = min(a[ti], 0.005 * max_c / (n * contrib[ti] + 1e-30))
                    if d2 < 1e-15:
                        continue
                    a_try = a.copy()
                    a_try[ti] -= d2
                    a_try[bi] += d2
                    a_try = _proj_simplex(a_try)
                    c_try = _conv_fft(a_try, L)
                    if c_try.max() < max_c:
                        a = a_try
                        improved = True
                        break
                if improved:
                    break
        
        it += 1
        if verbose and it % 100 == 0:
            print(f"    greedy it={it}, max_c={best_mc:.6f}")
    
    return best_a, best_mc

- Periodic restarts to escape local minima

import numpy as np
import time
import math


def _pow2(m):
    """Smallest power of 2 >= m."""
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _proj_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return v / s


def _conv_fft(a, L):
    """Full self-convolution via FFT, returns array of length 2*len(a)-1."""
    f = np.fft.rfft(a, L)
    return np.fft.irfft(f * f, L)[:2 * len(a) - 1]


def _softmax_w(c, tau):
    """Softmax weights of c / tau."""
    cm = c.max()
    z = np.clip((c - cm) / tau, -80.0, 0.0)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        w = np.zeros_like(c)
        w[c.argmax()] = 1.0
        return w
    return e / s


def _grad_soft(a, w, L2):
    """
    grad[j] = 2 * sum_i a[i]*w[i+j], computed via cross-correlation FFT.
    """
    n = len(a)
    Fa = np.fft.rfft(a, L2)
    Fw = np.fft.rfft(w, L2)
    cross = np.fft.irfft(Fa * np.conj(Fw), L2)
    return 2.0 * cross[:n]


def _score(a, L):
    """Compute 2n * max(conv) / sum^2."""
    n = len(a)
    s = a.sum()
    if s < 1e-10:
        return 1e30
    c = _conv_fft(a, L)
    return float(2.0 * n * c.max() / (s * s))


def _init_parent(n):
    """Try to load from height_sequence_1 global."""
    try:
        base = height_sequence_1
        arr = np.asarray(base, dtype=np.float64).ravel()
        if arr.size == n:
            arr = np.maximum(arr, 0.0)
            s = arr.sum()
            if s > 1e-10:
                return arr / s
        elif arr.size > n:
            arr = arr[:n]
            arr = np.maximum(arr, 0.0)
            s = arr.sum()
            if s > 1e-10:
                return arr / s
    except (NameError, Exception):
        pass
    return None


def _init_structured(n, rng):
    """Generate a structured random initialization."""
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0
    # Polynomial basis + random harmonics
    z = (
        rng.uniform(-0.5, 0.5) * x +
        rng.uniform(-0.3, 0.3) * (x * x - 1.0 / 3.0) +
        rng.normal(0.0, 0.2, n)
    )
    for k in range(rng.integers(1, 5)):
        amp = rng.uniform(0.1, 0.5)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.sin(2.0 * np.pi * k * t + ph)
    z += rng.normal(0.0, 0.1, n)
    p = np.exp(z - z.max())
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _init_bimodal(n, rng):
    """Bimodal (two-peaked) initialization."""
    t = np.linspace(0.0, 1.0, n)
    c1 = rng.uniform(0.15, 0.35)
    c2 = rng.uniform(0.65, 0.85)
    sig = rng.uniform(0.08, 0.2)
    z = np.exp(-((t - c1) / sig) ** 2) + np.exp(-((t - c2) / sig) ** 2)
    z += 0.02 * rng.random(n)
    return z / z.sum()


def _optimize_softmax(a, rng, deadline, L, L2, verbose=True):
    """
    Main soft-max gradient optimization.
    Minimizes max(conv(a,a)) s.t. a>=0, sum(a)=1 via multiplicative updates.
    """
    n = len(a)
    a = _proj_simplex(a)
    c = _conv_fft(a, L)
    best_mc = c.max()
    best_a = a.copy()

    tau = 0.15 * best_mc
    tau_min = 0.002 * best_mc
    eta = 0.08
    no_improve = 0
    it = 0
    last_mb_time = time.monotonic()

    if verbose:
        print(f"    init max_c={best_mc:.8f}, score={2.0*n*best_mc:.6f}")

    while time.monotonic() < deadline:
        c = _conv_fft(a, L)
        mc = c.max()

        if mc < best_mc - 1e-15:
            best_mc = mc
            best_a = a.copy()
            no_improve = 0
            last_mb_time = time.monotonic()
        else:
            no_improve += 1

        # Decrease temperature
        tau = max(tau_min, tau * 0.9985)

        # Compute gradient via soft-max weights
        w = _softmax_w(c, tau)
        grad = _grad_soft(a, w, L2)
        gscale = grad / (mc + 1e-30)

        # Multiplicative update with line search
        accepted = False
        for factor in (1.0, 0.5, 0.25):
            step = eta * factor
            if step < 1e-8:
                continue
            a_try = a * np.exp(-step * gscale)
            a_try = _proj_simplex(a_try)
            c_try = _conv_fft(a_try, L)
            if c_try.max() < mc:
                a = a_try
                accepted = True
                if factor < 1.0:
                    eta = max(0.005, eta * 0.85)
                break

        if not accepted:
            eta = max(0.003, eta * 0.7)

        # Restart if stuck
        if no_improve > 400:
            mix = 0.05 + 0.20 * rng.random()
            pert = _init_structured(n, rng)
            a = _proj_simplex((1.0 - mix) * best_a + mix * pert)
            tau = max(tau, 0.1 * _conv_fft(a, L).max())
            eta = 0.08
            no_improve = 0
            last_mb_time = time.monotonic()

        it += 1
        if verbose and it % 10000 == 0:
            el = time.monotonic() - last_mb_time
            print(f"    it={it}, max_c={best_mc:.8f}, tau={tau:.4e}, eta={eta:.4f}, "
                  f"iter/s~{it/(time.monotonic()-deadline+time.monotonic()-last_mb_time+el+1):.0f}")

    return best_a, best_mc
