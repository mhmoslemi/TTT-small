Search for non-negative step heights minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max gradient descent with adaptive step sizes, multi-start exploration,
and periodic restarts.
"""

import math
import time
import numpy as np


def _normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = a.sum()
    if s <= 1e-300:
        return np.full_like(a, 1.0 / len(a))
    return a / s


def _fft_params(n):
    """Compute FFT length and convolution length."""
    m = 2 * n - 1
    L = 1 << (m - 1).bit_length()
    if L < m:
        L <<= 1
    return L, m


def _conv_self(a, L, m):
    """Compute linear self-convolution via FFT."""
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _score(c, n):
    """Compute the objective (with sum(a)=1)."""
    return 2.0 * n * float(np.max(c))


def _gradient(c, a, w, L, n, m):
    """Compute gradient of weighted convolution sum via FFT."""
    W = np.fft.rfft(w, L)
    Ar = np.fft.rfft(a[::-1], L)
    g = np.fft.irfft(W * Ar, L)[(n-1):m]
    np.maximum(g, 0.0, out=g)
    return g


def _soft_weights(c, n, beta):
    """Compute softmax weights for the convolution."""
    v = n * c
    vmax = float(v.max())
    z = beta * (v - vmax)
    z = np.clip(z, -50.0, 0.0)
    w = np.exp(z)
    s = float(w.sum())
    if s <= 1e-300:
        w[:] = 1.0 / len(w)
    else:
        w /= s
    return w


def _random_init(n, rng):

    t = np.linspace(0, 1, n)
    x = 2 * t - 1

    for k in range(1, 4):
        amp = rng.uniform(0.1, 0.5) / math.sqrt(k)

        z += amp * np.cos(2 * np.pi * k * t + ph)

    z += rng.normal(0, 0.2, n)

    if n > 10:
        k_size = 5
        z = np.convolve(z, np.ones(k_size) / k_size, mode='same')
    
    p = np.exp(z - z.max())
    p = np.maximum(p, 1e-300)
    return _normalize(p)

    v_state = np.zeros(n)

    stall = 0

            stall = 0

        v_hat = v_state / (1 - 0.999**t)

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-10)

        a = a * np.exp(-step)
        a = _normalize(a)

            lr = min(0.6, lr * 1.03)
        elif stall > 100:
            lr = max(0.02, lr * 0.97)

def _spectral_nudge(a, L, m, n, alpha=0.1):

    # Nudge: reduce the magnitude of A where |A| is large
    mag = np.abs(A)
    if mag.max() > 1e-15:
        # Compress dynamic range
        mag_new = mag ** (1 - alpha) * (mag.max() ** alpha)  # compress
        # Or: mag_new = 1 / (1 + mag.mag.max() / mag)
        # Use a softer compression
        mag_norm = mag / mag.max()
        mag_new = mag / (1 + alpha * mag_norm)
        phase = A / (mag + 1e-300)
        A_new = mag_new * phase
        a_new = np.fft.irfft(A_new, L)[:n]
        a_new = _normalize(a_new)
        return a_new
    return a.copy()

"""
Search for non-negative step heights minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max gradient descent with Adam-adaptive multiplicative updates,
multi-start exploration, spectral nudging, and phased time allocation.
Improves over parent by: adaptive per-coordinate steps, better annealing,
diverse initialization, and spectral-domain refinement.

import math
import time
import numpy as np


def _normalize(a):
    """Normalize array to non-negative, sum=1."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-300:
        return np.full(a.shape, 1.0 / a.size)
    return a / s


def _fft_params(n):
    """Compute FFT length L and convolution output length m."""
    m = 2 * n - 1
    L = 1 << (m - 1).bit_length()
    if L < m:
        L <<= 1
    return L, m


def _conv_self(a, L, m):
    """Compute linear self-convolution via FFT."""
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _score(c, n):
    """Objective with sum(a)=1: 2*n*max(conv)."""
    return 2.0 * n * float(np.max(c))


def _gradient(c, a, w, L, n, m):
    """Gradient of sum_k w[k]*c[k] wrt a, via FFT."""
    W = np.fft.rfft(w, L)
    Ar = np.fft.rfft(a[::-1], L)
    g = np.fft.irfft(W * Ar, L)[(n - 1):m]
    np.maximum(g, 0.0, out=g)
    return g


def _soft_weights(c, n, beta):
    """Softmax weights over convolution (scaled by n)."""
    v = n * c
    vmax = float(v.max())
    z = beta * (v - vmax)
    np.clip(z, -50.0, 0.0, out=z)
    w = np.exp(z)
    s = float(w.sum())
    if s <= 1e-300:
        w[:] = 1.0 / w.size
    else:
        w /= s
    return w

        amp = rng.uniform(0.1, 0.4) / math.sqrt(k)
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos(2.0 * np.pi * k * t + ph)
    z += rng.normal(0.0, 0.15, n)
    if n > 10:
        z = np.convolve(z, np.ones(5) / 5.0, mode='same')
    p = np.exp(z - z.max())
    p = np.maximum(p, 1e-300)
    return _normalize(p)

"""
Search for non-negative step heights minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses soft-max gradient descent with Adam-adaptive multiplicative updates,
multi-start exploration, spectral nudging, phased time allocation,

def _normalize(a):
    """Normalize array to non-negative, sum=1."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 1e-300:
        return np.full(a.shape, 1.0 / a.size)
    return a / s


def _fft_params(n):
    """Compute FFT length L and convolution output length m."""
    m = 2 * n - 1
    L = 1 << (m - 1).bit_length()
    if L < m:
        L <<= 1
    return L, m


def _conv_self(a, L, m):
    """Compute linear self-convolution via FFT."""
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _score(c, n):
    """Objective with sum(a)=1: 2*n*max(conv)."""
    return 2.0 * n * float(np.max(c))


def _gradient(c, a, w, L, n, m):
    """Gradient of sum_k w[k]*c[k] wrt a, via FFT."""
    W = np.fft.rfft(w, L)
    Ar = np.fft.rfft(a[::-1], L)
    g = np.fft.irfft(W * Ar, L)[(n - 1):m]
    np.maximum(g, 0.0, out=g)
    return g


def _soft_weights(c, n, beta):
    """Softmax weights over convolution (scaled by n)."""
    v = n * c
    vmax = float(v.max())
    z = beta * (v - vmax)
    np.clip(z, -50.0, 0.0, out=z)
    w = np.exp(z)
    s = float(w.sum())
    if s <= 1e-300:
        w[:] = 1.0 / w.size
    else:
        w /= s
    return w


def _random_init(n, rng):
