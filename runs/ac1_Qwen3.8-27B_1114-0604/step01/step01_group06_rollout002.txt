    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:3*n-2].real
    return np.maximum(c, 0.0)

    s = float(np.sum(a))
    if s <= 1e-12:
        return np.inf
    c = _fft_conv(a)
    m = float(np.max(c))
    return 2.0 * n * m / (s * s)

import numpy as np
import time
from scipy.optimize import minimize

"""
Algorithm: Multi-scale adaptive log-sum-exp gradient optimization with
active-set polishing for minimizing 2*n*max(a*a)/(sum(a))^2.
Uses softmax parameterization, geometric tau annealing, L-BFGS at multiple
scales, then aggressive exact active-set gradient polishing with momentum.
"""

_G_N = 0
_G_NFFT = 0
_G_TAU = 1e-6
_G_BOUND = 30.0


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - np.max(z)
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(z.shape, 1.0 / z.size, dtype=np.float64)
    return e / s


def _fft_conv(a, N=None):
    n = len(a)
    if N is None:
        N = 1 << (3 * n - 2).bit_length()
    L = 3 * n - 2
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L].real
    return np.maximum(c, 0.0)


def _fast_score(a):
    n = len(a)
    s = float(np.sum(a))
    if s <= 1e-12 or not np.isfinite(s):
        return np.inf
    c = _fft_conv(a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return 2.0 * n * m / (s * s)


def _obj_grad(z, tau):
    n = _G_N
    N = _G_NFFT
    L = 2 * n - 1
    z = np.clip(np.asarray(z, dtype=np.float64).ravel(), -_G_BOUND, _G_BOUND)
    a = _softmax(z)
    c = _fft_conv(a, N)
    m = float(np.max(c))
    if not np.isfinite(m):
        return 1e12, np.zeros(n, dtype=np.float64)
    diff = (c - m) / tau
    diff = np.clip(diff, -745.0, 0.0)
    e = np.exp(diff)
    se = float(np.sum(e))
    if se <= 0.0 or not np.isfinite(se):
        return 1e12, np.zeros(n, dtype=np.float64)
    p = e / se
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real
    s_h = float(np.dot(h, a))
    g = a * (h - s_h)
    grad = 2.0 * g
    val = m + tau * np.log(se)
    scale = float(n)
    val *= scale
    grad *= scale
    val += 1e-12 * 0.5 * float(np.dot(z, z))
    grad += 1e-12 * z
    if not np.isfinite(val):
        val = 1e12
        grad = np.zeros(n, dtype=np.float64)
    return float(val), grad


def _z_from_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 1e-300)
    z = np.log(a)
    z = z - float(np.mean(z))
    return np.clip(z, -_G_BOUND, _G_BOUND)


def _run_l_bfgs(z0, tau, maxfun, maxiter):
    z0 = np.asarray(z0, dtype=np.float64).ravel()
    z0 = np.clip(z0 - np.mean(z0), -_G_BOUND, _G_BOUND)
    bounds = [(-_G_BOUND, _G_BOUND)] * _G_N

    def fun(z):
        return _obj_grad(z, tau)

    try:
        res = minimize(
            fun, z0, jac=True, method="L-BFGS-B",
            bounds=bounds,
            options={"maxfun": maxfun, "maxiter": maxiter,
                     "ftol": 1e-14, "pgtol": 1e-12, "disp": False}
        )
        return res
    except Exception:
        return None


def _polish_active_set(a0, deadline, ramp_iters=0):

    n = _G_N
    N = _G_NFFT
    a = np.maximum(np.asarray(a0, dtype=np.float64).ravel(), 0.0).copy()

    rng = np.random.default_rng(123456789)
    momentum = np.zeros(n, dtype=np.float64)
    beta = 0.9

    it = 0
    no_improve = 0

    while time.monotonic() < deadline:

                a = np.maximum(a, 0.0)
                a /= np.sum(a)

        active_float = np.zeros(2 * n - 1, dtype=np.float64)
        active_float[active] = 1.0
        fa = np.fft.rfft(a, N)
        fp = np.fft.rfft(active_float, N)
        h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real
        num_active = float(len(active))

        # db = a * (grad_a - mean(a * grad_a))
        s_ag = float(np.dot(a, grad))

        direction = -momentum

        # Line search
        accepted = False

        if gnorm < 1e-15:

            F_new = float(np.max(_fft_conv(a_new, N)))
            if F_new < F - 1e-15:

                a = a_new
                accepted = True

            for s in [1e-6, 3e-7, 1e-7, 3e-8, 1e-8]:

                a_new = _softmax(b_new)
                F_new = float(np.max(_fft_conv(a_new, N)))
                if F_new < F - 1e-15:
                    b = b_new
                    a = a_new
                    accepted = True
                    break

        if not accepted:
            # Small random perturbation in log-space
            pert = 0.002 * rng.standard_normal(n)
            b = np.clip(b + pert, -_G_BOUND, _G_BOUND)
            a = _softmax(b)
            a = np.maximum(a, 0.0)
            a /= np.sum(a)

        it += 1
        if it % 200 == 0:
            cur = 2.0 * n * F
            print(f"  polish it={it} cur={cur:.9f} best={best_score:.9f} "

        z = _z_from_a(a)
        for ti in range(len(tau_schedule)):
            if time.monotonic() > deadline - 300:
                break
            tau = tau_schedule[ti]
            _G_TAU = tau
            maxfun = iters_per_tau[ti]
            res = _run_l_bfgs(z, tau, maxfun, maxfun)
            if res is not None and hasattr(res, "x"):
                z_new = np.asarray(res.x, dtype=np.float64).ravel()
                z_new = np.clip(z_new, -_G_BOUND, _G_BOUND)
                a_new = _softmax(z_new)
                a_new = np.maximum(a_new, 0.0)
                s = float(np.sum(a_new))
                if s > 1e-12:
                    a_new /= s
                    sc_new = _fast_score(a_new)
                    if np.isfinite(sc_new) and sc_new < best_score:
                        best_score = sc_new
                        best_a = a_new.copy()
                        print(f"  L-BFGS c={ci} tau={tau:.1e}: score={sc_new:.9f} "
                              f"t={time.monotonic()-t0:.0f}s")
                    a = a_new
                    z = _z_from_a(a)

    # Phase 2: Active-set polishing
    if time.monotonic() < deadline - 60:
        print(f"Phase 2: Active-set polishing, current best={best_score:.9f}")
        pol_deadline = deadline - 30
        a_polished = _polish_active_set(best_a, pol_deadline)
        sc_polished = _fast_score(a_polished)
        if np.isfinite(sc_polished) and sc_polished < best_score:
            best_score = sc_polished
            best_a = a_polished.copy()
            print(f"  After active-set polish: score={sc_polished:.9f}")

    # Phase 3: Try SMA polish on the result
    if time.monotonic() < deadline - 30:
        pol_deadline2 = deadline - 10
        a_sma = _polish_sma(best_a, pol_deadline2)
        sc_sma = _fast_score(a_sma)
        if np.isfinite(sc_sma) and sc_sma < best_score:
            best_score = sc_sma
            best_a = a_sma.copy()
            print(f"  After SMA polish: score={sc_sma:.9f}")

    # Phase 4: Quick active-set polish on other top candidates
    if time.monotonic() < deadline - 20:
        for ci in range(1, min(4, len(scored))):
            if time.monotonic() > deadline - 15:
                break
            quick_deadline = time.monotonic() + min(60.0, deadline - time.monotonic() - 15)
            a_q = _polish_active_set(scored[ci][1], quick_deadline)
            sc_q = _fast_score(a_q)
            if np.isfinite(sc_q) and sc_q < best_score:
                best_score = sc_q
                best_a = a_q.copy()
                print(f"  Quick polish c={ci}: score={sc_q:.9f}")

    # Validate with official evaluator
    try:
        best_a = np.asarray(best_a, dtype=np.float64).ravel()
        best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
        best_a = np.clip(best_a, 0.0, 1000.0)
        s = float(np.sum(best_a))
        if s < 0.01 or not np.isfinite(s):
            best_a = np.ones(n, dtype=np.float64) / n
        else:
            best_a = best_a / s
        exact = evaluate_sequence(best_a.tolist())
        if np.isfinite(exact):
            print(f"Official evaluator: score={float(exact):.9f}")
    except Exception as e:
        print(f"Validation warning: {e}")

    # Final normalization
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = float(np.sum(best_a))
    if s < 0.01 or not np.isfinite(s):
        best_a = np.ones(n, dtype=np.float64) / n
    else:
        best_a = best_a / s

    print(f"Final score (fast): {_fast_score(best_a):.9f}, n={n}, "
          f"total time={time.monotonic()-t0:.0f}s")
    return [float(v) for v in best_a.tolist()]
