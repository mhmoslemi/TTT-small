    F = np.fft.rfft(a, L)
    return np.fft.irfft(F * F, L)[:2*len(a)-1]

    n = len(a)
    c = _fft_conv(a, L)
    M = float(c.max())
    S = float(a.sum())

def _grad_at_peak(a, c, L):
    """Exact gradient of max(conv) at the argmax position."""
    n = len(a)
    k = int(np.argmax(c))
    # d(conv[k])/da[i] = a[k-i] + a[i] if 0<=k-i<n, else 0
    # Valid i: max(0, k-n+1) <= i <= min(n-1, k)
    i_lo = max(0, k - n + 1)
    i_hi = min(n - 1, k)
    grad = np.zeros(n)
    i = np.arange(i_lo, i_hi + 1)
    ki = k - i
    grad[i] = a[ki] + a[i]

    n = len(a)
    # grad[i] = sum_k w[k] * (a[k-i] + a[i]) * 1[0<=k-i<n]
    # = sum_{k=i}^{i+n-1} w[k]*a[k-i] + a[i] * sum_{k=i}^{i+n-1} w[k]

    Lw = np.fft.rfft(w, fftL)

def _peak_grad(a, c, topk=1):

- Proper step size scaling (normalized gradient)
- Coordinate-wise refinement for final polish
- Better temperature/step management

import time
import math
import numpy as np


def _norm1(a):
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s


def _fft_conv(a, L):
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)
    m = 2 * len(a) - 1
    return c[:m]


def _full_obj(a):
    """Full objective for a sequence (not necessarily normalized)."""
    n = len(a)
    S = float(a.sum())
    if S < 1e-300:
        return float('inf')
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    c = _fft_conv(a, L)
    M = float(c.max())
    return 2.0 * n * M / (S * S)


def _peak_grad(a, c, topk=1):
    """Exact gradient of max (or top-k weighted) of convolution."""
    n = len(a)
    if topk <= 1:
        k = int(np.argmax(c))
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        idx = np.arange(i_lo, i_hi + 1)
        return a[k - idx] + a[idx]
    else:
        # Top-k with softmax weighting
        kvals = np.argpartition(c, -topk)[-topk:]
        cvals = c[kvals]
        cmax = cvals.max()
        T = 50.0 / (max(cmax - cvals.min(), 1e-30))
        z = T * (cvals - cmax)
        z = np.clip(z, -700, 0)
        e = np.exp(z)
        se = e.sum()
        if se < 1e-300:
            w = np.ones(topk) / topk
        else:
            w = e / se
        grad = np.zeros(n)
        for w_k, k in zip(w, kvals):
            k = int(k)
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += w_k * (a[k - idx] + a[idx])
        return grad


def _gradient_step(a, c, eta, topk=1):
    """One multiplicative gradient step."""
    grad = _peak_grad(a, c, topk)
    gmax = float(grad.max())
    if gmax <= 0:
        return a
    grad_norm = grad / gmax
    a_new = a * np.exp(-eta * grad_norm)
    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()
    if s <= 0:
        return a
    return a_new / s


def _grad_search(a, L, n, eta, time_limit, topk=1, verbose=False):
    """Iterative gradient descent with adaptive step size."""
    a = _norm1(a.copy())
    c = _fft_conv(a, L)
    obj = 2.0 * n * float(c.max())
    best_a = a.copy()
    best_obj = obj
    deadline = time.monotonic() + time_limit
    it = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        if no_improve > 500:
            break
        
        a_new = _gradient_step(a, c, eta, topk)
        c_new = _fft_conv(a_new, L)
        obj_new = 2.0 * n * float(c_new.max())
        
        if obj_new < obj - 1e-15:
            a = a_new
            c = c_new
            obj = obj_new
            no_improve = 0
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
        else:
            eta *= 0.7
            no_improve += 1
        
        it += 1
        if verbose and it % 10000 == 0:
            print(f"  grad_search iter {it}, obj={best_obj:.8f}, eta={eta:.2f}")
    
    return best_a, best_obj
