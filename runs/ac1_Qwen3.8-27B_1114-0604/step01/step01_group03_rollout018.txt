        return np.ones(a.size, dtype=np.float64) / a.size
    return a / s

def _score_normalized(a, n):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    conv = np.convolve(a, a)
    return 2.0 * n * float(np.max(conv))

def _score_full(a, n):
    """Compute 2*n*max(conv(a,a))/(sum(a))^2."""
    s = float(a.sum())
    if s < 1e-30:
        return float('inf')
    conv = np.convolve(a, a)
    return 2.0 * n * float(np.max(conv)) / (s * s)

def _build_lp_matrix(f, active_idx, n):
    """Build K x n constraint matrix where A[i,j] = f[k_i - j] if valid."""
    K = len(active_idx)
    A = np.zeros((K, n), dtype=np.float64)
    for i in range(K):
        k = active_idx[i]
        j_min = max(0, int(k) - n + 1)
        j_max = min(n - 1, int(k))
        if j_min <= j_max:
            js = np.arange(j_min, j_max + 1)
            A[i, js] = f[k - js]
    return A

def _solve_lp_cutting_plane(f, M, n, max_cuts=30, initial_K=300, timeout=30.0):
    """Solve max sum(b) s.t. (f*b)[k] <= M, b >= 0 using cutting plane."""
    conv_ff = np.convolve(f, f)
    active_idx = np.sort(np.argsort(conv_ff)[-initial_K:])
    b_prev = np.ones(n) / n
    start_time = time.monotonic()

    for cut_iter in range(max_cuts):
        if time.monotonic() - start_time > timeout:
            if b_prev is not None and np.max(b_prev) > 0:
                return b_prev
            return None

        A = _build_lp_matrix(f, active_idx, n)
        c_instr = -np.ones(n)
        b_ub = np.full(len(active_idx), M, dtype=np.float64)

        try:
            res = linprog(c_instr, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n, method='highs')
        except Exception:
            if b_prev is not None and np.max(b_prev) > 0:
                return b_prev
            return None

        if res.status != 0:
            if b_prev is not None and np.max(b_prev) > 0:
                return b_prev
            return None

        b_star = np.maximum(res.x, 0.0)
        if b_star.sum() < 1e-15:
            return b_prev if b_prev is not None and np.max(b_prev) > 0 else None

        conv_fb = np.convolve(f, b_star)
        max_violation = float(np.max(conv_fb)) - M

        if max_violation < 1e-9 * max(M, 1e-15):
            return b_star

        violated_k = int(np.argmax(conv_fb))
        if violated_k not in active_idx:
            active_idx = np.append(active_idx, violated_k)
        else:
            top_viol = np.argsort(conv_fb)[::-1][:10]
            added = False
            for k in top_viol:
                if k not in active_idx:
                    active_idx = np.append(active_idx, k)
                    added = True
                    break
            if not added:
                return b_star

        b_prev = b_star

    return b_prev

def _line_search(f, g, n, n_grid=80):
    """Find t in [0,1] minimizing max(((1-t)f+tg)*((1-t)f+tg))."""
    best_t = 0.0
    best_val = float('inf')
    ts = np.linspace(0.0, 1.0, n_grid)
    for t in ts:
        h = (1.0 - t) * f + t * g
        h = np.maximum(h, 0.0)
        s = h.sum()
        if s < 1e-15:
            continue
        h = h / s
        val = float(np.max(np.convolve(h, h)))
        if val < best_val:
            best_val = val
            best_t = t

    # Refine around best
    if 0 < best_t < 1:
        lo = max(0.0, best_t - 1.0 / n_grid)
        hi = min(1.0, best_t + 1.0 / n_grid)
        ts2 = np.linspace(lo, hi, 40)
        for t in ts2:
            h = (1.0 - t) * f + t * g
            h = np.maximum(h, 0.0)
            s = h.sum()
            if s < 1e-15:
                continue
            h = h / s
            val = float(np.max(np.convolve(h, h)))
            if val < best_val:
                best_val = val
                best_t = t

    return best_t, best_val

def _smooth_gradient_step(f, n, temp, lr, rng):

    conv = np.convolve(f, f)

    z = np.clip(z, -500.0, 0.0)

        w = np.ones_like(conv) / len(conv)
    else:
        w /= sw

    # (conv)[k] = sum_i a[i] * a[k-i]

        idx1 = np.arange(0, n) + j  # k-j = 0..n-1 => k = j..j+n-1
        # a[k-j] for k where 0<=k-j<n => k in [j, j+n-1], clipped to [0, 2n-2]

    # grad[j] = sum_k w[k] * (a[k-j] + a[j-k]) for valid indices

    # Actually: grad[j] = sum_k w[k] * a[k-j] * 1[0<=k-j<n] + sum_k w[k] * a[j-k] * 1[0<=j-k<n]

    # c1 = cross_correlation(w, f): c1[j] = sum_k w[k] * f[k-j]
    # In numpy: np.convolve(w, f[::-1]) gives sum_k w[k] * f_rev[k'] ...

    # c1[j] = sum_k w[k] f[k-j] = sum_m w[m+j] f[m]
    # Define W(m) = w[m + j] (shifted w), then c1[j] = sum_m W(m) f[m] = dot product
    # This is the cross-correlation: c1 = cross_corr(w, f)
    # In terms of convolution: cross_corr(w, f)[j] = sum_m w[m] f[m+j]
    # Wait, I need c1[j] = sum_m w[m+j] f[m] = sum_m f[m] w[m+j]
    # Let n = m+j => m = n-j: c1[j] = sum_n f[n-j] w[n] = (f_rev * w)[j]
    # where f_rev[i] = f[-i]... 
    # OK let me just use the direct approach with FFT:

    W = np.fft.rfft(w, L)
    F = np.fft.rfft(f, L)

    # c1[j] = sum_k w[k] f[k-j] = ifft(fft(w) * fft(f))[j + shift]
    # Actually: (w * f)[m] = sum_k w[k] f[m-k]
    # c1[j] = sum_k w[k] f[k-j] = sum_k w[k] f[-(j-k)] 
    # Let me define g = f reversed: g[i] = f[n-1-i]
    # c1[j] = sum_k w[k] f[k-j]
    # = sum_k w[k] g[n-1-k+j]
    # Hmm this is getting complicated. Let me just do it directly for n=6121.
    # For each j, c1[j] = sum over k where 0<=k-j<n of w[k]*f[k-j]
    # = sum over m where 0<=m<n of w[m+j]*f[m] (where m=k-j)
    # So c1[j] = dot(w[j:j+n], f) for the valid part

    # Let me use FFT properly:
    # c1[j] = sum_m f[m] * w[m+j] for m=0..n-1 (with w out of range = 0)
    # = sum_m f[m] * w[m+j]

    # This is the cross-correlation of f and w: c1[j] = (f star w)[j]
    # In numpy: np.correlate(f, w, mode='full') gives:
    # result[m] = sum_k f[k] w[k-m+n-1] ... not quite
    # Let me just use: c1 = np.convolve(w, f, mode='full')[n-1:n-1+2*n-1]... 
    # No. Let me think again.
    # np.convolve(a, b, 'full')[m] = sum_k a[k] b[m-k]
    # I want c1[j] = sum_m f[m] w[m+j]
    # = sum_m f[m] w[m+j]

    # = np.convolve(f_flipped, w, 'full')[j + n - 1] where f_flipped[i] = f[n-1-i]
    # Actually: np.convolve(f, w, 'full')[j + n - 1] = sum_k f[k] w[j+n-1-k]

    # Let me just compute it as: c1 = np.correlate(w, f, mode='full')

    # Simple O(n^2) computation:
    grad = np.zeros(n, dtype=np.float64)
    # For each j: grad[j] = sum_k w[k] * (f[k-j] if 0<=k-j<n else 0 + f[j-k] if 0<=j-k<n else 0)
    # = sum_k w[k] * f[k-j] * 1[0<=k-j<n] + sum_k w[k] * f[j-k] * 1[0<=j-k<n]
    # First term: for fixed j, k ranges from j to j+n-1 (clipped to [0, 2n-2])
    #   sum_{k=j}^{min(2n-2, j+n-1)} w[k] * f[k-j]
    # Second term: for fixed j, k ranges from 0 to min(j, n-1)
    #   sum_{k=0}^{min(j, n-1)} w[k] * f[j-k]
    
    # Vectorized:
    # First term: grad1[j] = sum_{m=0}^{min(n-1, 2n-2-j)} w[j+m] * f[m]
    # Second term: grad2[j] = sum_{m=j}^{min(j, n-1)} w[m] * f[j-m] = sum_{m=0}^{min(j, n-1)} w[j-m] * f[m]
    # Wait, second term: sum_{k=0}^{min(j,n-1)} w[k] * f[j-k], let m=j-k, k=j-m:
    # = sum_{m=max(0,j-n+1)}^{j} w[j-m] * f[m] (m ranges 0 to j, but f[m] needs m<n and w[j-m] needs j-m < 2n-1)
    
    # Let me just use a different approach: the gradient is the cross-correlation
    # grad = cross_corr(w, f) + cross_corr(rev(w), rev(f)) ... 
    
    # Actually, the simplest correct approach:
    # (a*a)[k] = sum_i a[i] a[k-i]
    # d(a*a)[k] / da[j] = a[k-j] + a[j-k] (if indices valid)
    # So grad[j] = sum_k w[k] * (a[k-j] * 1[0<=k-j<n] + a[j-k] * 1[0<=j-k<n])
    
    # Let me compute using FFT:
    # Term 1: t1[j] = sum_k w[k] * a[k-j] * 1[0<=k-j<n]
    #   = sum_{m=0}^{n-1} a[m] * w[m+j] * 1[m+j < 2n-1]
    #   = sum_{m=0}^{n-1} a[m] * w[m+j]
    # This is the cross-correlation of a and w: t1 = np.correlate(w, a, 'full')[n-1:]
    # np.correlate(w, a, 'full')[k] = sum_m w[m] a[m + k - (len(a)-1)]
    # Hmm, let me check: np.correlate(in1, in2, mode='full')[k] = sum_m in1[m+k] * conj(in2[m])

    # = np.correlate(w, a, 'full')[j] (with n playing role of m)

    # t1[k] = sum_n W[n] * conj(A[n... ] ... 
    # Actually ifft(W * conj(A))[k] = sum_n w[n+k] * a[n] (circular)
    # For linear: we need enough zero padding (L >= 2n-1 + n - 1 = 3n-2)

    # Direct computation (O(n^2) but n=6121 so ~37M ops, should be <100ms in numpy)
    # t1[j] = dot(f, w[j:j+n]) for j=0..n-1
    # This is a sliding dot product, which can be computed as a convolution!
    # t1[j] = sum_{m=0}^{n-1} f[m] * w[m+j]
    # = (f_reversed * w)[n-1+j] where f_reversed[i] = f[n-1-i]
    # np.convolve(f_reversed, w, 'full')[n-1+j] = sum_k f_reversed[k] w[n-1+j-k]
    # = sum_k f[n-1-k] w[n-1+j-k]
    # Let m = n-1-k: = sum_m f[m] w[j+m] ✓
    
    f_rev = f[::-1]
    conv_fw = np.convolve(f_rev, w, 'full')
    t1 = conv_fw[n - 1: 2 * n - 1]  # t1[j] for j=0..n-1
    
    # t2[j] = sum_k w[k] * f[j-k] * 1[0<=j-k<n, 0<=k<2n-1]
    # = sum_{m=0}^{min(j,n-1)} w[j-m] * f[m] (letting m=j-k)
    # = dot(w[j-n+1:j+1], f[0:n]) ... similar sliding dot
    # t2[j] = sum_{m=0}^{n-1} w[j-m] * f[m] (with w out of range = 0)
    # = (w_reversed * f)[j + 2n-2 - 1] ... 
    # Let me use: t2[j] = sum_m f[m] * w[j-m]
    # = (f_reversed * w)[2n-2-1-(j)] ... 
    # Simpler: t2[j] = (w_conv_f)[j] where w_conv_f[m] = sum_k w[k] f[m-k] = (w*f)[m]
    # t2[j] = sum_m f[m] w[j-m] = (w * f)[j] ... YES!
    # (w * f)[j] = sum_k w[k] f[j-k] = t2[j] ✓
    # But w has length 2n-1 and f has length n, so (w*f) has length 3n-2
    conv_wf = np.convolve(w, f, 'full')
    t2 = conv_wf[0:n]  # t2[j] for j=0..n-1
    
    grad = t1 + t2
    # Normalize gradient
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-30:
        return f
    grad /= gnorm
    
    # Multiplicative update (keep non-negativity)
    new_f = f * np.exp(-lr * grad)
    new_f = _normalize(new_f)
    return new_f

def _random_prior(n, rng):
    """Generate a random non-negative vector."""
    t = np.linspace(0.0, 1.0, n)
    z = rng.normal(0, 1, n)
    # Add some structure
    for k in range(1, 6):
        amp = rng.uniform(0.1, 0.5) / k
        ph = rng.uniform(0, 2 * np.pi)
        z += amp * np.cos(2 * np.pi * k * t + ph)
    z += rng.normal(0, 0.3, n)
    p = np.exp(z - z.max())
    p = np.maximum(p, 1e-30)
    return p / p.sum()

def _try_different_n(f, n_new, n_old):
    """Resample a sequence from length n_old to n_new."""
    if n_old == n_new:
        return f.copy()
    x_old = np.linspace(0, 1, n_old)
    x_new = np.linspace(0, 1, n_new)
    return np.interp(x_new, x_old, f)

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative step heights minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    
    Strategy:
    1. Start from parent sequence (or random init).
    2. LP-based iterations with cutting plane for direction finding.
    3. Smoother gradient polishing.
    4. Periodic randomization to escape local optima.
    """
    start_time = time.monotonic()
    hard_deadline = start_time + budget_s - 10.0  # leave 10s buffer
    rng = np.random.default_rng(seed)
    
    # Get initial sequence
    n = 6121
    try:
        f_raw = np.array(height_sequence_1, dtype=np.float64).ravel()
        if len(f_raw) != n:
            n = len(f_raw)
        f = _normalize(f_raw)
    except (NameError, Exception):
        f = _random_prior(n, rng)
    
    print(f"Initial n={n}, score={_score_full(f, n):.8f}")
    best_f = f.copy()
    best_score = _score_full(f, n)
    
    def _unroll_and_evaluate(arr):
        """Evaluate using the external evaluator."""
        seq = [float(x) for x in arr]
        try:
            val = evaluate_sequence(seq)
        except NameError:
            val = _score_full(arr, len(arr))
        except Exception:
            val = float('inf')
        return float(val)
    
    best_exact = best_score
    
    # Phase 1: LP-based iteration (60% of time)
    phase1_end = start_time + 0.60 * (budget_s - 30)
    it_count = 0
    stale_count = 0
    last_improve_time = start_time
    
    while time.monotonic() < phase1_end:
        it_count += 1
        
        # Compute current M
        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))
        
        # Solve LP for direction
        t0 = time.monotonic()
        b_star = _solve_lp_cutting_plane(f, M, n, max_cuts=40, initial_K=400, timeout=60.0)
        lp_time = time.monotonic() - t0
        
        if b_star is None or np.sum(b_star) < 1e-15:
            stale_count += 5
            if stale_count > 10:
                break
            # Try a different direction
            g = _random_prior(n, rng)
        else:
            g = _normalize(b_star)
        
        # Line search
        t_opt, val_opt = _line_search(f, g, n, n_grid=60)
        
        if val_opt < float(np.max(np.convolve(f, f))) - 1e-15:
            new_f = (1.0 - t_opt) * f + t_opt * g
            new_f = np.maximum(new_f, 0.0)
            new_f = _normalize(new_f)
            
            new_score = _score_full(new_f, n)
            if new_score < best_score:
                best_score = new_score
                best_f = new_f.copy()
                stale_count = 0
                last_improve_time = time.monotonic()
            
            f = new_f
        else:
            stale_count += 1
            if stale_count % 5 == 0:
                # Randomize
                mix = rng.uniform(0.1, 0.4)
                rand = _random_prior(n, rng)
                f = _normalize((1.0 - mix) * f + mix * rand)
                stale_count = 0
        
        if it_count % 10 == 0:
            elapsed = time.monotonic() - start_time
            print(f"  LP iter {it_count}: score={best_score:.8f}, "
                  f"cur={_score_full(f,n):.8f}, tp={lp_time:.1f}s, "
                  f"elapsed={elapsed:.0f}s")
    
    print(f"Phase 1 done: best={best_score:.8f} after {it_count} iters")
    
    # Phase 2: Smoothed gradient polishing (25% of time)
    phase2_end = min(start_time + 0.85 * (budget_s - 30), hard_deadline)
    f = best_f.copy()
    temp = 0.5
    lr = 0.3
    smooth_iters = 0
    smooth_stale = 0
    
    while time.monotonic() < phase2_end:
        # Compute smoothed gradient step
        conv = np.convolve(f, f)
        vmax = float(np.max(conv))
        z = conv / temp - vmax / temp
        z = np.clip(z, -500.0, 0.0)
        w = np.exp(z)
        sw = float(w.sum())
        if sw < 1e-30:
            w = np.ones_like(conv) / len(conv)
        else:
            w = w / sw
        
        # Compute gradient via convolution tricks
        f_rev = f[::-1]
        conv_fw = np.convolve(f_rev, w, 'full')
        t1 = conv_fw[n - 1: 2 * n - 1]
        conv_wf = np.convolve(w, f, 'full')
        t2 = conv_wf[0:n]
        grad = t1 + t2
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-30:
            break
        grad = grad / gnorm
        
        # Try multiple step sizes
        improved = False
        for step in [lr, 0.5 * lr, 0.25 * lr, 2.0 * lr]:
            if step <= 0:
                continue
            new_f = f * np.exp(-step * grad)
            new_f = _normalize(new_f)
            new_score = _score_full(new_f, n)
            if new_score < _score_full(f, n) - 1e-14:
                f = new_f
                improved = True
                break
        
        if improved:
            smooth_stale = 0
            lr = min(1.0, lr * 1.1)
            if _score_full(f, n) < best_score:
                best_score = _score_full(f, n)
                best_f = f.copy()
                last_improve_time = time.monotonic()
        else:
            smooth_stale += 1
            lr = max(0.01, lr * 0.8)
            temp = min(50.0, temp * 1.2)
            if smooth_stale > 30:
                # Reheat
                mix = rng.uniform(0.05, 0.2)
                f = _normalize((1.0 - mix) * f + mix * _random_prior(n, rng))
                temp = 0.5
                lr = 0.3
                smooth_stale = 0
        
        smooth_iters += 1
        if smooth_iters % 50 == 0:
            print(f"  Smooth iter {smooth_iters}: best={best_score:.8f}, "
                  f"cur={_score_full(f,n):.8f}, temp={temp:.2f}, lr={lr:.3f}")
    
    print(f"Phase 2 done: best={best_score:.8f} after {smooth_iters} smooth iters")
    
    # Phase 3: Final LP polishing (10% of time)
    phase3_end = min(hard_deadline - 20.0, start_time + 0.95 * (budget_s - 30))
    f = best_f.copy()
    lp_final_iters = 0
    
    while time.monotonic() < phase3_end:
        conv_ff = np.convolve(f, f)
        M = float(np.max(conv_ff))
        
        b_star = _solve_lp_cutting_plane(f, M, n, max_cuts=60, initial_K=600, timeout=45.0)
        
        if b_star is None or np.sum(b_star) < 1e-15:
            break
        
        g = _normalize(b_star)
        t_opt, val_opt = _line_search(f, g, n, n_grid=80)
        
        if val_opt < float(np.max(conv_ff)) - 1e-15:
            new_f = _normalize(np.maximum((1.0 - t_opt) * f + t_opt * g, 0.0))
            new_score = _score_full(new_f, n)
            if new_score < best_score:
                best_score = new_score
                best_f = new_f.copy()
            f = new_f
        lp_final_iters += 1
        
        if lp_final_iters % 5 == 0:
            print(f"  Final LP iter {lp_final_iters}: best={best_score:.8f}")
    
    print(f"Phase 3 done: best={best_score:.8f} after {lp_final_iters} iters")
    
    # Final: evaluate with exact function
    final_seq = [float(x) for x in best_f]
    final_val = _unroll_and_evaluate(final_seq)
    print(f"Final exact evaluation: {final_val:.8f}")
    
    # Also try with the un-normalized parent (in case scaling helps)
    try:
        parent_seq = [float(x) for x in height_sequence_1]
        parent_val = _unroll_and_evaluate(parent_seq)
        if parent_val < final_val:
            print(f"Parent is better: {parent_val:.8f}")
            final_val = parent_val
            final_seq = parent_seq
    except (NameError, Exception):
        pass
    
    # Try a few random perturbations of the best
    for _ in range(20):
        if time.monotonic() > hard_deadline:
            break
        noise = rng.normal(0, 0.01 * np.max(best_f), n)
        perturbed = _normalize(best_f + noise)
        seq = [float(x) for x in perturbed]
        val = _unroll_and_evaluate(seq)
        if val < final_val:
            final_val = val
            final_seq = seq
            print(f"  Perturbation improved: {final_val:.8f}")
    
    print(f"FINAL: {final_val:.8f}")
    return final_seq
