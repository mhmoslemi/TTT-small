def _fft_conv(a, N=None):
    """Fast FFT-based linear convolution of a with itself."""
    n = len(a)
    if N is None:
        N = 1 << (3*n - 2).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1].real

def _score(a):

    s = np.sum(a)

    c = _fft_conv(a)
    return 2.0 * n * np.max(c) / (s * s)

    try:

        da = J.T @ sol
    except np.linalg.LinAlgError:

    a_new = a + t * da
    a_new = np.maximum(a_new, 0.0)  # project onto non-negative
    s = np.sum(a_new)
    if s > 0:

    return a_new

"""
Hybrid Active-Set Gauss-Newton + Logsumexp Gradient Optimization
for minimizing peak autocorrelation of non-negative sequences.

Key improvements over previous approach:
1. Active-set Gauss-Newton: directly equalizes peak convolution lags
   (faster convergence than smooth max approximation near the optimum).
2. Adaptive active set management with ill-conditioning guard.
3. Multi-length search (n=8192 and n=16384).
4. Diverse parametric initialization with exponential and Fourier families.
5. Combined active-set + gradient polishing in the final phase.
"""

import numpy as np
import time
from scipy.optimize import minimize

# === Global state for FFT ===
_G_N = 0
_G_NFFT = 0
_G_TAU = 1e-4


def _set_global(n, tau=1e-4):
    global _G_N, _G_NFFT, _G_TAU
    _G_N = int(n)
    _G_NFFT = 1 << (3 * _G_N - 2).bit_length()
    _G_TAU = float(tau)


def _fft_conv(a):
    n = len(a)
    N = _G_NFFT
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1].real
    return np.maximum(c, 0.0)


def _fast_score(a):
    n = len(a)
    s = float(np.sum(a))
    if s < 1e-12 or not np.isfinite(s):
        return np.inf
    c = _fft_conv(a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return 2.0 * n * m / (s * s)


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - np.max(z)
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0 or not np.isfinite(s):
        return np.full_like(z, 1.0 / z.size)
    return e / s


# === Active Set Gauss-Newton ===

def _active_set_step(a, max_active=60):

    Compute one Gauss-Newton step for the active set of peak lags.
    Returns (step_direction, current_max_c, active_set_size).

    # Row idx (lag k): J[idx, j] = a[k-j] for valid j (one-sided derivative)
    # As shown, J[idx, lo:hi] = a[lo:hi][::-1] where lo=max(0,k-n+1), hi=min(n,k+1)

        da = J.T @ sol
    except:

def _logsumexp_obj_grad(z):

    n = _G_N
    N = _G_NFFT
    tau = _G_TAU
    L = 2 * n - 1

    c = np.fft.irfft(fa * fa, N)[:L].real
    c = np.maximum(c, 0.0)

    if not np.isfinite(m):
        return 1e12, np.zeros(n)

    s_h = float(np.dot(h, a))
    g = a * (h - s_h)
    grad = 2.0 * g

    val = m + tau * np.log(se)
    scale = float(n) * float(n)
    val *= scale
    grad *= scale

    # Tiny regularization
    pen = 1e-12
    val += 0.5 * pen * float(np.dot(z, z))
    grad += pen * z

    if not np.isfinite(val):
        val = 1e12
        grad = np.zeros(n)

    return float(val), grad


def _gradient_polish(a, deadline, label="grad"):
    """
    Polish using logsumexp gradient with adaptive tau and line search.
    Returns (best_a, best_score).
    """
    n = len(a)
    log_a = np.log(np.maximum(a, 1e-300))
    log_a = log_a - log_a.mean()
    log_a = np.clip(log_a, -25.0, 25.0)

    best_a = a.copy()
    best_score = _fast_score(a)
    rng = np.random.default_rng(777)
    stall = 0
    it = 0

    # Adaptive tau: start large, decrease
    taus = [1e-2, 5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]
    tau_idx = 0

    while time.monotonic() < deadline:
        if tau_idx < len(taus):
            _set_global(n, taus[tau_idx])

        # Gradient step
        _, grad = _logsumexp_obj_grad(log_a)

        if not np.all(np.isfinite(grad)):
            stall += 1
            if stall > 100:
                break
            it += 1
            continue

        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            tau_idx += 1
            stall += 1
            if stall > 50:
                break
            it += 1
            continue

        grad = grad / gnorm

        # Line search
        a_current = _softmax(log_a)
        F_current = float(np.max(_fft_conv(a_current)))

        accepted = False
        for eta in [0.01, 0.005, 0.002, 0.001, 0.0005, 0.0001,
                    0.02, 0.05, 0.1, 0.2, 0.5]:
            log_a_new = np.clip(log_a - eta * grad, -30.0, 30.0)
            a_new = _softmax(log_a_new)
            F_new = float(np.max(_fft_conv(a_new)))
            s_new = float(np.sum(a_new))
            if s_new < 1e-10:
                continue
            a_new = a_new / s_new
            sc = 2.0 * n * F_new
            if sc < best_score - 1e-12:
                log_a = log_a_new
                a_current = a_new
                F_current = F_new
                best_score = sc
                best_a = a_new.copy()
                accepted = True
                break

        if accepted:
            stall = 0
        else:
            stall += 1
            # Random perturbation
            if stall % 20 == 0:
                for sc in [0.005, 0.02, 0.08, 0.2]:
                    log_a_try = log_a + sc * rng.standard_normal(n)
                    log_a_try = np.clip(log_a_try, -30.0, 30.0)
                    a_try = _softmax(log_a_try)
                    sc_try = _fast_score(a_try)
                    if sc_try < best_score:
                        log_a = log_a_try
                        a_current = a_try
                        best_score = sc_try
                        best_a = a_try.copy()
                        stall = 0
                        break
            # Decrease tau periodically
            if stall > 50:
                tau_idx = min(tau_idx + 1, len(taus) - 1)
            if stall > 200:
                break

        it += 1
        if it % 200 == 0:
            print(f"  [{label}] it={it} score={best_score:.10f} "
                  f"tau={taus[tau_idx]:.1e} stall={stall} "
                  f"time_left={max(0, deadline - time.monotonic()):.0f}s")

    return best_a, best_score


# === Parametric Initialization ===

def _make_starts(n, rng, deadline):
    """Create diverse initial sequences and return them sorted by score."""
    starts = []
    u = np.linspace(0.0, 1.0, n)
    x = 2.0 * u - 1.0

    # Use provided sequence if available
    try:
        g = globals().get("height_sequence_1")
        if g is not None:
            g = np.asarray(g, dtype=np.float64)
            if len(g) > 10 and np.all(np.isfinite(g)) and np.sum(np.maximum(g, 0)) > 0:
                g = np.clip(g, 1e-300, 1000.0)
                z_old = np.log(g)
                z_old -= z_old.mean()
                z = np.interp(u, np.linspace(0, 1, len(g)), z_old)
                a = _softmax(z)
                a /= a.sum()
                starts.append(a.copy())
    except:
        pass

    # Exponential family: exp(alpha*x + beta*x^2 + gamma*x^3)
    for alpha in [-3, -2, -1, -0.5, 0, 0.5, 1, 2]:
        for beta in [-1.5, -0.5, 0, 0.5, 1.5]:
            for gamma in [0, -0.5, 0.5]:
                log_a = alpha * x + beta * x**2 + gamma * x**3
                a = _softmax(log_a)
                a /= a.sum()
                starts.append(a)

    # Fourier modes
    for n_m in [1, 2, 3, 4]:
        for trial in range(2):
            log_a = np.zeros(n)
            for k in range(1, n_m + 1):
                c = rng.uniform(0.5, 1.5)
                ph = rng.uniform(0, 2 * np.pi)
                log_a += c * np.cos(2 * np.pi * k * u + ph)
            a = _softmax(log_a)
            a /= a.sum()
            starts.append(a)

    # Polynomial (Chebyshev-based)
    for deg in [2, 4, 6, 8]:
        for A in [0.3, 0.6, 0.9, 1.2, 1.5]:
            Tk = np.polynomial.chebyshev.chebval(x, [0] * (deg - 1) + [1.0])
            a = np.maximum(1.0 + A * Tk, 1e-8)
            a = a * a  # square for stronger shape
            a /= a.sum()
            if np.all(np.isfinite(a)) and a.sum() > 0.01:
                starts.append(a)

    # Two-level blocks
    for split in [0.3, 0.35, 0.4, 0.45, 0.5, 0.55, 0.6]:
        for ratio in [0.3, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 5.0]:
            a = np.ones(n)
            idx = int(n * split)
            a[idx:] *= ratio
            a /= a.sum()
            starts.append(a)

    # Smooth transitions (sigmoid)
    for split in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for width in [0.01, 0.03, 0.05, 0.1, 0.2]:
            for ratio in [1.5, 2.0, 3.0, 5.0]:
                a = 1.0 + (ratio - 1.0) / (1.0 + np.exp(-(u - split) / width))
                a /= a.sum()
                starts.append(a)

    # "U-shape" (low in middle, high at edges)
    for A in [0.5, 1.0, 2.0, 3.0, 5.0]:
        a = 1.0 + A * x**2
        a = np.maximum(a, 0.1)
        a /= a.sum()
        starts.append(a)

    # "W-shape"
    for A in [0.5, 1.0, 2.0]:
        a = 1.0 + A * np.cos(4 * np.pi * u)
        a = np.maximum(a, 0.05)
        a /= a.sum()
        starts.append(a)

    # Random smooth (low-frequency noise)
    for _ in range(8):
        log_a = np.zeros(n)
        for k in range(1, 8):
            c = 0.3 * rng.standard_normal()
            ph = rng.uniform(0, 2 * np.pi)
            log_a += c * np.cos(2 * np.pi * k * u + ph)
        a = _softmax(log_a)
        a /= a.sum()
        starts.append(a)

    # Score and sort
    scored = []
    for a in starts:
        if np.all(np.isfinite(a)) and np.sum(a) > 0.01:
            sc = _fast_score(a)
            if np.isfinite(sc):
                scored.append((sc, a.copy()))

    scored.sort(key=lambda x: x[0])
    return scored[:25]


# === L-BFGS from a start ===

def _run_lbfgs(a0, maxfun=800, maxiter=200):
    """Run L-BFGS in log-space starting from a0."""
    n = _G_N
    log_a = np.log(np.maximum(a0, 1e-300))
    log_a = log_a - log_a.mean()
    log_a = np.clip(log_a, -25.0, 25.0)

    bounds = [(-25.0, 25.0)] * n
    opts = {
        "maxfun": int(maxfun),
        "maxiter": int(maxiter),
        "ftol": 1e-14,
        "pgtol": 1e-10,
        "disp": False,
    }
    try:
        res = minimize(_logsumexp_obj_grad, log_a, jac=True,
                       method="L-BFGS-B", bounds=bounds, options=opts)
        if res.success or hasattr(res, 'x'):
            z = np.asarray(res.x, dtype=np.float64).ravel()
            a = _softmax(np.clip(z, -30, 30))
            a /= a.sum()
            return a
    except:
        pass
    return None


# === Main optimization for one length ===

def _optimize_length(n, deadline, rng):
    """Full optimization pipeline for a given sequence length n."""
    global _G_N, _G_NFFT, _G_TAU
    _set_global(n, tau=1e-4)

    print(f"\n=== Optimizing n={n}, deadline in {deadline - time.monotonic():.0f}s ===")
    t0 = time.monotonic()

    # Phase 1: Get diverse starts
    starts = _make_starts(n, rng, deadline)
    if not starts:
        a = np.ones(n) / n
        print(f"  Fallback uniform, score={_fast_score(a):.6f}")
        return a, _fast_score(a)

    best_score = starts[0][0]
    best_a = starts[0][1].copy()
    print(f"  Best initial: score={best_score:.10f}")

    # Phase 2: L-BFGS on best starts (quick, gets close to a local min)
    lbfgs_deadline = time.monotonic() + min(300, (deadline - time.monotonic()) * 0.25)
    candidates = []
    for idx, (sc, a0) in enumerate(starts[:8]):
        if time.monotonic() > lbfgs_deadline:
            break
        _set_global(n, tau=1e-3)  # larger tau for L-BFGS stability
        a_new = _run_lbfgs(a0, maxfun=600, maxiter=150)
        if a_new is not None:
            sc_new = _fast_score(a_new)
            if np.isfinite(sc_new):
                candidates.append((sc_new, a_new.copy()))
                if sc_new < best_score:
                    best_score = sc_new
                    best_a = a_new.copy()
                    print(f"  L-BFGS start {idx}: score={sc_new:.10f} "
                          f"elapsed={time.monotonic()-t0:.0f}s")

    # Sort candidates
    candidates.sort(key=lambda x: x[0])
    # Keep top candidates
    candidates = sorted(set((round(s, 10), a) for s, a in candidates),
                        key=lambda x: x[0])[:5]
    if not candidates:
        candidates = [(best_score, best_a.copy())]

    # Phase 3: Active-set optimization on best candidates
    as_deadline = time.monotonic() + min(600, (deadline - time.monotonic()) * 0.4)
    for ci, (sc, a0) in enumerate(candidates[:3]):
        if time.monotonic() > as_deadline:
            break
        if ci == 0:
            sub_dl = min(as_deadline, time.monotonic() + 400)
        else:
            sub_dl = min(as_deadline, time.monotonic() + 150)
        a_res, sc_res = _active_set_optimize(a0, sub_dl, max_iter=8000,
                                             label=f"AS{ci}")
        if sc_res < best_score:
            best_score = sc_res
            best_a = a_res.copy()
            print(f"  Phase3 AS cand {ci}: score={sc_res:.10f} "
                  f"elapsed={time.monotonic()-t0:.0f}s")

    # Phase 4: Gradient polish on overall best
    grad_deadline = time.monotonic() + min(600, (deadline - time.monotonic()) * 0.5)
    _set_global(n, tau=1e-4)
    if time.monotonic() < grad_deadline:
        a_res, sc_res = _gradient_polish(best_a, grad_deadline, label="GP")
        if sc_res < best_score:
            best_score = sc_res
            best_a = a_res.copy()
            print(f"  Phase4 GradPolish: score={sc_res:.10f} "
                  f"elapsed={time.monotonic()-t0:.0f}s")

    # Phase 5: Final active-set polish
    if time.monotonic() < deadline - 30:
        final_dl = deadline - 30
        a_res, sc_res = _active_set_optimize(best_a, final_dl, max_iter=5000,
                                             label="ASfin")
        if sc_res < best_score:
            best_score = sc_res
            best_a = a_res.copy()
            print(f"  Phase5 ASfinal: score={sc_res:.10f} "
                  f"elapsed={time.monotonic()-t0:.0f}s")

    print(f"  Final n={n}: score={best_score:.10f} "
          f"total_time={time.monotonic()-t0:.0f}s")
    return best_a, best_score


# === Main entry point ===

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing the peak autocorrelation score.
    """
    budget_s = max(120.0, float(budget_s))
    deadline = time.monotonic() + budget_s - 5.0
    rng = np.random.default_rng(int(seed))

    print(f"Starting optimization with budget={budget_s:.0f}s")

    best_a = None
    best_score = np.inf

    # Try n=8192 first (well-optimized, reliable)
    t1_deadline = time.monotonic() + min(1000, (deadline - time.monotonic()) * 0.45)
    try:
        a1, s1 = _optimize_length(8192, t1_deadline, rng)
        if s1 < best_score:
            best_score = s1
            best_a = a1
    except Exception as e:
        print(f"  n=8192 failed: {e}")

    # Try n=16384 if time remains
    time_left = deadline - time.monotonic()
    if time_left > 400:
        t2_deadline = time.monotonic() + min(time_left - 100, 900)
        try:
            a2, s2 = _optimize_length(16384, t2_deadline, rng)
            if s2 < best_score:
                best_score = s2
                best_a = a2
        except Exception as e:
            print(f"  n=16384 failed: {e}")

    # If we have time left, do one more polish on the best
    time_left = deadline - time.monotonic()
    if best_a is not None and time_left > 100:
        n = len(best_a)
        _set_global(n, tau=1e-5)
        polish_dl = deadline - 10
        try:
            a_res, s_res = _active_set_optimize(best_a, polish_dl, max_iter=20000,
                                                label="FINAL")
            if s_res < best_score:
                best_score = s_res
                best_a = a_res
        except:
            pass

    # Final validation and normalization
    if best_a is None:
        best_a = np.ones(8192, dtype=np.float64) / 8192

    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = float(np.sum(best_a))
    if s < 0.01 or not np.isfinite(s):
        best_a = np.ones(8192, dtype=np.float64) / 8192
        s = 1.0
    else:
        best_a = best_a / s

    # Validate with official evaluator
    try:
        exact = evaluate_sequence(best_a.tolist())
        print(f"\nOfficial evaluator score: {float(exact):.10f}")
    except:
        pass

    print(f"Returning sequence of length {len(best_a)} with internal score {best_score:.10f}")
    return [float(v) for v in best_a.tolist()]
