def fft_convolution(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)

def smooth_max_and_weights(b, beta):
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float('inf'), np.empty(0, dtype=np.float64)
    m = float(np.max(b))
    if not np.isfinite(m):
        return float('inf'), np.zeros_like(b, dtype=np.float64)
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b, dtype=np.float64)
        w /= float(b.size)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):

        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta

def fast_score(p):

def project_simplex(v, target=1.0):

    if v.size == 0:
        return v
    if target <= 0.0:

    if not np.all(np.isfinite(v)):

    high = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (low + high)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        q = np.full(v.shape, target / float(v.size), dtype=np.float64)
    else:
        q *= target / s
    return q

def normalize_simplex(p):
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size), dtype=np.float64)
    p *= 1.0 / s
    return p

def compute_gradient(p, w):
    """Compute grad_i = 2 * sum_j p[j] * w[i+j] for i = 0..n-1."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size  # 2n-1
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n: m][::-1].copy()
    return 2.0 * s

def build_conv_matrix(n, f):
    """Build the (2n-1) x n matrix A where A[j,i] = f[j-i] if 0<=j-i<n."""
    A = np.zeros((2*n-1, n), dtype=np.float64)
    for j in range(2*n-1):
        for i in range(n):
            k = j - i
            if 0 <= k < n:
                A[j, i] = f[k]
    return A

def lp_water_filling(f, M, max_iter=50):
    """
    Solve: max sum(g) s.t. A@g <= M*1, g >= 0
    where A is the convolution matrix for f.
    Returns the optimal g (normalized to sum 1) and max sum.
    """
    n = len(f)
    m = 2 * n - 1
    
    # Build sparse-like constraint matrix
    # A has a banded structure: A[j,i] = f[j-i] if 0 <= j-i < n
    # We'll build it densely for small n
    A = np.zeros((m, n), dtype=np.float64)
    for j in range(m):
        lo = max(0, j - (n - 1))
        hi = min(j, n - 1) + 1
        for i in range(lo, hi):
            A[j, i] = f[j - i]
    
    c = -np.ones(n, dtype=np.float64)  # minimize -sum(g)
    A_ub = A
    b_ub = np.full(m, M, dtype=np.float64)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            g = res.x
            s = float(np.sum(g))
            if s > 0:
                return g / s, s
    except Exception:
        pass
    
    return None, 0.0

    no_improve = 0

        return float('inf'), np.empty(0, dtype=np.float64)
    m = float(np.max(b))
    if not np.isfinite(m):
        return float('inf'), np.zeros_like(b, dtype=np.float64)
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b, dtype=np.float64)
        w /= float(b.size)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w

def fast_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float('inf')
    b = fft_convolution(p)
    M = float(np.max(b))
    if not np.isfinite(M):
        return float('inf')
    return float(2.0 * n * M / (S * S))

def project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    if target <= 0.0:
        return np.zeros_like(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(v.shape, target / float(v.size), dtype=np.float64)
    low = float(np.min(v)) - 1.0
    high = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (low + high)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        q = np.full(v.shape, target / float(v.size), dtype=np.float64)
    else:
        q *= target / s
    return q

def normalize_simplex(p):
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        if p.size == 0:
            return p
        return np.full(p.shape, 1.0 / float(p.size), dtype=np.float64)
    p *= 1.0 / s
    return p

def compute_gradient(p, w):
    """Compute grad_i = 2 * sum_j p[j] * w[i+j] for i = 0..n-1."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n: m][::-1].copy()
    return 2.0 * s

def build_conv_matrix_sparse(n, f):
    """Build convolution matrix as list of (row, col, val) triples for each constraint."""
    # A[j,i] = f[j-i] if 0 <= j-i < n
    # For each row j, non-zero entries are at columns i where 0 <= j-i < n
    # i.e., i in [max(0, j-n+1), min(j, n-1)]
    rows = []
    cols = []
    vals = []
    for j in range(2 * n - 1):
        lo = max(0, j - (n - 1))
        hi = min(j, n - 1)
        for i in range(lo, hi + 1):
            rows.append(j)
            cols.append(i)
            vals.append(f[j - i])
    return rows, cols, vals

def lp_solve_water_fill(f, M):
    """
    Solve: max sum(g) s.t. A@g <= M*1, g >= 0
    where A[j,i] = f[j-i] if 0<=j-i<n.
    """
    n = len(f)
    m = 2 * n - 1
    
    rows, cols, vals = build_conv_matrix_sparse(n, f)
    
    from scipy.sparse import coo_matrix
    A_sparse = coo_matrix((vals, (rows, cols)), shape=(m, n))
    
    c = -np.ones(n, dtype=np.float64)
    b_ub = np.full(m, M, dtype=np.float64)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c, A_ub=A_sparse, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.success:
            g = res.x
            s = float(np.sum(g))
            if s > 0 and np.isfinite(s):
                return g / s, s
    except Exception:
        pass
    return None, 0.0

def lp_iteration(f_init, n_lp, max_iters=25, time_limit=120.0):
    """Iterate the LP water-filling approach at length n_lp."""
    f = normalize_simplex(f_init[:n_lp].copy()) if len(f_init) >= n_lp else None
    if f is None:
        f = np.interp(np.linspace(0, 1, n_lp), np.linspace(0, 1, len(f_init)), f_init)
        f = normalize_simplex(f)
    
    start_t = time.monotonic()
    best_f = f.copy()
    best_score = fast_score(f)
    
    for it in range(max_iters):
        if time.monotonic() - start_t > time_limit:
            break
        
        b = fft_convolution(f)
        M = float(np.max(b))
        
        g, sum_g = lp_solve_water_fill(f, M)
        if g is None or sum_g <= 0:
            break
        
        # Blend with mixing parameter
        t = 0.4
        f_new = (1.0 - t) * f + t * g
        f_new = normalize_simplex(f_new)
        
        sc = fast_score(f_new)
        if sc < best_score:
            best_score = sc
            best_f = f_new.copy()
        
        f = f_new
        print(f"    LP iter {it+1}: score={best_score:.8f} (sum_g={sum_g:.6f})")
    
    return best_f, best_score

def fista_optimize(p0, budget_s, beta_start=10.0, beta_end=200000.0, beta_ramp=150, verbose=True):
    """
    FISTA (Nesterov accelerated projected gradient) on the simplex.
    Minimizes n * smooth_max(conv(p,p)) with sum(p)=1, p>=0.
    """
    p = normalize_simplex(p0)
    n = p.size
    
    best_p = p.copy()
    best_score = fast_score(p)
    
    start_time = time.monotonic()
    step = 0.5
    y = p.copy()
    t_mom = 1.0
    it = 0
    no_improve = 0
    f_curr = float(n * smooth_max_and_weights(fft_convolution(p), 100.0)[0])
    
    while time.monotonic() - start_time < budget_s:
        it += 1
        
        if it < beta_ramp:
            frac = it / beta_ramp
            beta = beta_start * (beta_end / beta_start) ** frac
        else:
            beta = beta_end
        
        # Compute gradient at y (momentum point)
        b = fft_convolution(y)
        G, w = smooth_max_and_weights(b, beta)
        if not np.isfinite(G):
            break
        
        grad_raw = compute_gradient(y, w)
        grad = grad_raw - float(np.mean(grad_raw))
        
        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-16:
            break
        
        # FISTA update
        x_new = project_simplex(y - step * grad, 1.0)
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom))
        y_new = x_new + ((t_mom - 1.0) / t_new) * (x_new - p)
        
        # Evaluate
        b_new = fft_convolution(x_new)
        M_new = float(np.max(b_new))
        cur_score = float(2.0 * n * M_new)
        
        if np.isfinite(cur_score) and cur_score < best_score - 1e-13:
            best_score = cur_score
            best_p = x_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive step
        if no_improve > 200:
            step *= 0.5
            if step < 1e-9:
                break
            no_improve = 0
        elif no_improve > 100:
            step *= 0.7
        elif no_improve == 0 and step < 3.0:
            step = min(3.0, step * 1.1)
        
        p = x_new
        y = y_new
        t_mom = t_new
        
        if verbose and (it % 300 == 0 or (it < 20 and it % 5 == 0)):
            elapsed = time.monotonic() - start_time
            print(f"    FISTA it={it} t={elapsed:.1f}s beta={beta:.0f} step={step:.3f} best={best_score:.8f}")
    
    return best_p, best_score

def upscale_sequence(p_small, n_large):
    """Upscale via cubic interpolation."""
    n_small = len(p_small)
    if n_small == n_large:
        return p_small.copy()
    x_small = np.linspace(0, 1, n_small)
    x_large = np.linspace(0, 1, n_large)
    p_large = np.interp(x_large, x_small, p_small)
    return normalize_simplex(p_large)

def get_starting_point(N, rng):
    """Get initial sequence, preferring height_sequence_1."""
    try:
        hs = np.array(height_sequence_1, dtype=np.float64)
        hs = np.maximum(hs, 0.0)
        s = float(np.sum(hs))
        if s > 0 and np.isfinite(s):
            if len(hs) == N:
                return hs / s
            else:
                return upscale_sequence(hs / s, N)
    except Exception:
        pass
    return np.full(N, 1.0 / N, dtype=np.float64)

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best sequence minimizing 2*n*max(a*a)/(sum(a))^2.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0
    
    if budget_s <= 10:
        return [1.0]
    
    start_time = time.monotonic()
    deadline = start_time + budget_s - 30.0
    rng = np.random.default_rng(seed)
    
    N = 8192
    print(f"=== Starting: N={N}, budget={budget_s:.0f}s ===")
    
    # Get starting point
    p_start = get_starting_point(N, rng)
    score_start = fast_score(p_start)
    print(f"Starting point score: {score_start:.8f}")
    
    # Also prepare alternative starts
    p_uniform = np.full(N, 1.0 / N, dtype=np.float64)
    x = np.linspace(0, 1, N)
    p_cos = 1.0 + 0.7 * np.cos(np.pi * x)
    p_cos = np.maximum(p_cos, 0.01)
    p_cos = normalize_simplex(p_cos)
    
    best_p = p_start.copy()
    best_score = score_start
    
    # Phase 1: LP water-filling at small n (if budget allows)
    if budget_s > 200:
        lp_budget = min(90.0, 0.12 * budget_s)
        n_lp = 512
        print(f"\n--- Phase 1: LP at n={n_lp}, budget={lp_budget:.0f}s ---")
        
        f_lp = normalize_simplex(p_start[:n_lp])
        f_lp_res, lp_score = lp_iteration(f_lp, n_lp, max_iters=20, time_limit=lp_budget)
        print(f"LP result (n={n_lp}): score={lp_score:.8f}")
        
        # Upscale and evaluate
        f_up = upscale_sequence(f_lp_res, N)
        up_score = fast_score(f_up)
        print(f"Upscaled to N={N}: score={up_score:.8f}")
        
        if up_score < best_score:
            best_score = up_score
            best_p = f_up.copy()
    
    # Phase 2: Main FISTA optimization
    remaining = deadline - time.monotonic()
    print(f"\n--- Phase 2: FISTA, remaining={remaining:.0f}s ---")
    
    if remaining > 30:
        # Allocate time: 70% to best start, 30% divided among others
        main_budget = remaining * 0.65
        other_budget = remaining * 0.35
        
        # Primary: FISTA on best_p
        p_fit, sc_fit = fista_optimize(best_p, main_budget,
                                        beta_start=5.0, beta_end=300000.0,
                                        beta_ramp=200, verbose=True)
        if sc_fit < best_score:
            best_score = sc_fit
            best_p = p_fit.copy()
            print(f"After main FISTA: {best_score:.8f}")
        
        # Secondary: FISTA on a different start (if time remains)
        remaining2 = deadline - time.monotonic()
        if remaining2 > 30:
            p_cos2 = normalize_simplex(p_start * (1.0 + 0.1 * np.sin(2 * np.pi * 3 * x)))
            p_fit2, sc_fit2 = fista_optimize(p_cos2, min(other_budget, remaining2 - 10),
                                              beta_start=5.0, beta_end=200000.0,
                                              beta_ramp=100, verbose=True)
            if sc_fit2 < best_score:
                best_score = sc_fit2
                best_p = p_fit2.copy()
                print(f"After secondary FISTA: {best_score:.8f}")
    
    # Phase 3: Final sharpening (very high beta, few iterations)
    remaining3 = deadline - time.monotonic()
    if remaining3 > 15:
        print(f"\n--- Phase 3: Sharpening, remaining={remaining3:.0f}s ---")
        p_sharp, sc_sharp = fista_optimize(best_p, remaining3,
                                            beta_start=100000.0, beta_end=2000000.0,
                                            beta_ramp=30, verbose=True)
        if sc_sharp < best_score:
            best_score = sc_sharp
            best_p = p_sharp.copy()
            print(f"After sharpening: {best_score:.8f}")
    
    # Final processing
    final_p = normalize_simplex(best_p)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_seq = [float(x) for x in final_p]
    
    print(f"\nFinal fast score: {best_score:.8f}")
    
    try:
        off = float(evaluate_sequence(final_seq))
        print(f"Official score: {off:.8f}")
    except Exception as e:
        print(f"Eval error: {e}")
        off = float('inf')
    
    if not np.isfinite(off) or off > 2.0:
        # Fallback
        return [1.0] * N
    
    return final_seq
