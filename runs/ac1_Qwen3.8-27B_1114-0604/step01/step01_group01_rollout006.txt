```python
"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using LP-based direction finding and local refinement.
"""

import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(a, b):
    la = len(a)
    lb = len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)

def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = p.sum()
    if s <= 0 or not np.isfinite(s):
        return None
    return p / s

def _score(a):
    a = _sanitize(a)
    if a is None:
        return float('inf')
    n = len(a)
    c = _conv_full(a, a)
    return 2.0 * n * np.max(c)

def _solve_lp(f, M):
    """
    Solve: max sum(g) s.t. f*g <= M (element-wise), g >= 0
    Returns the optimal g (normalized) and the optimal objective value.
    """
    n = len(f)
    
    # Build constraint matrix A such that A @ g <= M_vec
    # A[k, j] = f[j] if 0 <= k-j <= n-1, else 0
    # This is the convolution matrix
    
    # We can build this as a sparse matrix
    rows = []
    cols = []
    vals = []
    
    for k in range(2*n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(f[j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    
    # Objective: maximize sum(g) = minimize -sum(g)
    c_obj = -np.ones(n)
    
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, M), 
                  bounds=[(0, None)]*n, method='highs')
    
    if res.status != 0:
        return None, res.fun
    
    g = res.x
    s = g.sum()
    if s <= 0:
        return None, 0
    
    return g / s, s
```

$P_k$ is a "permutation-like" matrix. Specifically, it's the adjacency matrix of a bipartite graph (or a "anti-identity" matrix shifted). The eigenvalues of $P_k$ are $\pm 1$ (and possibly 0). So it's NOT PSD in general.

For example, for $n=3$, $k=1$:
$P_1 = \begin{pmatrix} 0 & 1 & 0 \\ 1 & 0 & 1 \\ 0 & 1 & 0 \end{pmatrix}$

The eigenvalues are $\sqrt{2}, 0, -\sqrt{2}$. So not PSD.

This means the constraint $a^T P_k a \leq t$ is NOT convex. The problem is non-convex.
