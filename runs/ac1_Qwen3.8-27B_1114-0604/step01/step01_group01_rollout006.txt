"""

import numpy as np

def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b):
    la = len(a)
    lb = len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)


def _conv_grad(a, w):
    """
    Compute gradient of sum_k w[k]*c[k] where c=convolve(a,a).
    grad[i] = 2 * sum_{m=0}^{n-1} a[m] * w[m+i]
    Uses FFT cross-correlation.
    """
    n = len(a)
    L = _next_pow2(3 * n)
    fa = np.fft.rfft(a, L)
    fw = np.fft.rfft(w, L)
    corr = np.fft.irfft(np.conjugate(fa) * fw, L)
    return 2.0 * corr[:n].astype(float, copy=False)


def _sanitize(p):
    """Clip to non-negative finite, normalize to sum 1."""
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return p / s


def _exact_score(p):
    """Score for normalized p: 2*n*max(convolve(p,p))."""
    p = _sanitize(p)
    if p is None:
        return float("inf")
    m = p.size
    c = _conv_full(p, p)
    if c.size == 0:
        return float("inf")
    c = np.where(np.isfinite(c), c, 0.0)
    return float(2.0 * m * np.max(c))


def _soft_weights(d, beta):
    """Softmax weights on array d."""
    n = d.size
    if n == 0:
        return np.array([], dtype=float)
    d = np.where(np.isfinite(d), d, 0.0)
    if beta <= 1e-8:
        return np.full(n, 1.0 / n, dtype=float)
    shift = float(np.max(d))
    if not np.isfinite(shift):
        return np.full(n, 1.0 / n, dtype=float)
    e = np.exp(beta * (d - shift))
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n, dtype=float)
    return e / s


def _mmd_step(p, beta, dataseta, datasetb):
    """
    Compute gradient info for MMD step.
    Returns (exact_score, smooth_value, gradient_vector, max_grad_component)
    """
    m = p.size
    c = _conv_full(p, p)
    c = np.where(np.isfinite(c), c, 0.0)
    d = m * c  # d[k] = m * c[k], so max(d) = m * max(c) = smooth target
    exact = float(2.0 * np.max(d)) if d.size else float("inf")
    w = _soft_weights(d, beta)
    grad = _conv_grad(p, w)
    grad = np.where(np.isfinite(grad), grad, 0.0)
    smooth = float(np.dot(w, d))
    return exact, smooth, grad, w

def _resample(arr, m):
    """Linear interpolation resample to length m."""
    if m <= 0:
        return None
    a = _sanitize(arr)
    if a is None:
        return None
    n = a.size
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m, dtype=float)
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0)
    out = out + 1e-12
    return _sanitize(out)

        s = m
    return p / s


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses improved MMD + block refinement + multi-scale approach.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 60.0:
        budget = 120.0

    start_time = time.monotonic()
    deadline = start_time + max(60.0, budget - 20.0)

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng((iseed % 1000000007) + 7)

    # Get parent
    parent_arr = None
    try:
        parent_raw = height_sequence_1
        parent_arr = _sanitize(np.asarray(parent_raw, dtype=float))
    except (NameError, Exception):
        pass

    best_arr = None
    best_score = float("inf")

    def _update_best(arr, sc):
        global_best_arr = arr
        global_best_score = sc
        return global_best_arr, global_best_score

    # Phase 1: Quick evaluation of starting points (up to 60s)
    print(f"Phase 1: Generating starting points...")
    print(f"  Parent available: {parent_arr is not None}")

    candidates = []  # list of (score, arr)

    if parent_arr is not None:
        sc = _exact_score(parent_arr)
        candidates.append((sc, parent_arr.copy()))
        print(f"  Parent score: {sc:.8f}, length: {len(parent_arr)}")

    # Try different lengths
    lengths_to_try = [3072, 4096, 6144]
    if parent_arr is not None:
        # Always try parent length first
        pl = len(parent_arr)
        if pl not in lengths_to_try:
            lengths_to_try.insert(0, pl)

    # Generate quick starts at the best parent length
    n_main = parent_arr.size if parent_arr is not None else 3072

    # Two-block best
    tb = _two_block_best(n_main)
    if tb is not None:
        sc = _exact_score(tb)
        candidates.append((sc, tb))
        print(f"  Two-block score: {sc:.8f}")

    # Random initializations
    for mode in range(6):
        p = _gen_initial(n_main, rng, mode)
        if p is not None:
            sc = _exact_score(p)
            candidates.append((sc, p))

    candidates.sort(key=lambda x: x[0])
    print(f"  Best starting candidate: {candidates[0][0]:.8f}")

    # Phase 2: Short MMD on top candidates (up to 200s)
    print(f"Phase 2: Short MMD runs...")
    t2 = deadline - time.monotonic()
    if t2 > 60:
        top_k = min(4, len(candidates))
        for idx in range(top_k):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            talloc = min(40.0, max(15.0, rem - 20.0))
            p, sc = _run_mmd_v2(candidates[idx][1], rng, talloc,
                                beta_start=40.0, beta_end=400.0,
                                verbose=False, label=f"phase2_{idx}")
            if p is not None and np.isfinite(sc) and sc < best_score:
                best_arr = p.copy()
                best_score = sc
            print(f"  Phase 2 cand {idx}: score={sc:.8f} (best={best_score:.8f})")

    if best_arr is None and candidates:
        best_arr = candidates[0][1].copy()
        best_score = candidates[0][0]

    # Phase 3: Try larger lengths by upscaling the best
    print(f"Phase 3: Multi-scale exploration...")
    t3 = deadline - time.monotonic()
    if t3 > 300 and best_arr is not None:
        # Upscale to a larger length and optimize
        for big_n in [4096, 6144]:
            if big_n <= best_arr.size:
                continue
            rem = deadline - time.monotonic()
            if rem < 120:
                break
            big = _resample(best_arr, big_n)
            if big is None:
                continue
            # Add noise
            big_noisy = _sanitize(big * np.exp(rng.normal(0.0, 0.05, big_n)))
            if big_noisy is None:
                big_noisy = big
            talloc = min(80.0, max(20.0, rem - 40.0))
            p, sc = _run_mmd_v2(big_noisy, rng, talloc,
                                beta_start=60.0, beta_end=600.0,
                                verbose=True, label=f"big{big_n}")
            if p is not None and np.isfinite(sc) and sc < best_score:
                best_arr = p.copy()
                best_score = sc
                print(f"  Upscaled to {big_n}: score={sc:.8f}")
            elif p is not None and np.isfinite(sc):
                print(f"  Upscaled to {big_n}: score={sc:.8f} (not better)")

    # Phase 4: Main optimization with the best candidate
    print(f"Phase 4: Main MMD optimization...")
    t4 = deadline - time.monotonic()
    if t4 > 100 and best_arr is not None:
        # Long MMD run with slower beta increase
        talloc = min(0.5 * t4, 600.0)
        p, sc = _run_mmd_v2(best_arr, rng, talloc,
                            beta_start=80.0, beta_end=1500.0,
                            verbose=True, label="main_mmd")
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_arr = p.copy()
            best_score = sc
        print(f"  Main MMD done: best={best_score:.8f}")

    # Phase 5: Block refinement
    t5 = deadline - time.monotonic()
    if t5 > 80 and best_arr is not None:
        talloc = min(0.4 * t5, 400.0)
        p, sc = _block_refine(best_arr, rng, talloc,
                              verbose=True, label="block")
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_arr = p.copy()
            best_score = sc
        print(f"  Block refine done: best={best_score:.8f}")

    # Phase 6: Second MMD pass with higher beta
    t6 = deadline - time.monotonic()
    if t6 > 80 and best_arr is not None:
        talloc = min(0.4 * t6, 300.0)
        p, sc = _run_mmd_v2(best_arr, rng, talloc,
                            beta_start=150.0, beta_end=3000.0,
                            verbose=True, label="mmd_high")
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_arr = p.copy()
            best_score = sc
        print(f"  High-beta MMD done: best={best_score:.8f}")

    # Phase 7: Final block refinement + MMD
    t7 = deadline - time.monotonic()
    if t7 > 60 and best_arr is not None:
        talloc = min(0.5 * t7, 200.0)
        p, sc = _block_refine(best_arr, rng, talloc,
                              verbose=True, label="block2")
        if p is not None and np.isfinite(sc) and sc < best_score:
            best_arr = p.copy()
            best_score = sc

        rem = deadline - time.monotonic()
        if rem > 40:
            talloc2 = min(0.5 * rem, 100.0)
            p, sc = _run_mmd_v2(best_arr, rng, talloc2,
                                beta_start=300.0, beta_end=5000.0,
                                verbose=True, label="final_mmd")
            if p is not None and np.isfinite(sc) and sc < best_score:
                best_arr = p.copy()
                best_score = sc

    print(f"  Final block+MMD done: best={best_score:.8f}")

    # Phase 8: Try a fresh start at a different length and see if it's better
    t8 = deadline - time.monotonic()
    if t8 > 200:
        print(f"Phase 8: Fresh exploration...")
        fresh_n = 4096
        best_fresh = None
        best_fresh_score = float("inf")

        for mode in range(4):
            rem = deadline - time.monotonic()
            if rem < 50:
                break
            p0 = _gen_initial(fresh_n, rng, mode)
            talloc = min(40.0, max(15.0, rem - 20.0))
            p, sc = _run_mmd_v2(p0, rng, talloc,
                                beta_start=50.0, beta_end=500.0,
                                verbose=False, label=f"fresh{mode}")
            if p is not None and np.isfinite(sc) and sc < best_fresh_score:
                best_fresh = p.copy()
                best_fresh_score = sc

        if best_fresh is not None and best_fresh_score < best_score:
            # Refine the fresh solution
            rem = deadline - time.monotonic()
            if rem > 60:
                talloc = min(0.5 * rem, 150.0)
                p, sc = _run_mmd_v2(best_fresh, rng, talloc,
                                    beta_start=100.0, beta_end=2000.0,
                                    verbose=True, label="fresh_refine")
                if p is not None and np.isfinite(sc) and sc < best_score:
                    best_arr = p.copy()
                    best_score = sc

            rem = deadline - time.monotonic()
            if rem > 40:
                talloc = min(0.4 * rem, 80.0)
                p, sc = _block_refine(best_arr, rng, talloc,
                                      verbose=False, label="fresh_block")
                if p is not None and np.isfinite(sc) and sc < best_score:
                    best_arr = p.copy()
                    best_score = sc

        print(f"  Fresh exploration done: best={best_score:.8f}")

    # Final: verify with evaluate_sequence
    if best_arr is None:
        best_arr = np.ones(n_main, dtype=float) / n_main

    best_arr = _sanitize(best_arr)
    if best_arr is None:
        best_arr = np.ones(n_main, dtype=float) / n_main

    out = best_arr
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s <= 0 or not np.isfinite(s):
        out = np.ones(n_main, dtype=float)
    else:
        out = out / s

    # Final validation with evaluator
    lst = [float(x) for x in out]
    try:
        final_sc = evaluate_sequence(lst)
        print(f"  Final evaluate_sequence score: {final_sc:.8f}")
    except Exception as e:
        print(f"  Evaluation error: {e}")
        final_sc = _exact_score(out)

    elapsed = time.monotonic() - start_time
    print(f"  Total time: {elapsed:.1f}s")
    print(f"  Final sequence length: {len(lst)}")
    print(f"  Final score: {final_sc:.8f}")

    return lst
