```python
from scipy.optimize import linprog

def solve_lp_direction(a, M):
    """Find direction b that maximizes sum(b) subject to (a*b)(k) <= M, b >= 0."""
    n = len(a)
    # (a*b)(k) = sum_i a_i * b_{k-i} (with bounds)
    # This is a linear function of b.
    # The coefficient of b_j in (a*b)(k) is a_{k-j} (if 0 <= k-j < n, else 0)
    
    # Objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    
    # Constraints: (a*b)(k) <= M for k = 0, ..., 2n-2
    # A[i, j] = a_{i-j} if 0 <= i-j < n, else 0
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(max(0, k-(n-1)), min(n-1, k)+1):
            A[k, j] = a[k-j]
    
    b_vec = np.full(2*n-1, M)
    
    res = linprog(c, A_ub=A, b_ub=b_vec, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None
```

```python
"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding
combined with smoothed subgradient descent and coordinate descent polish.
The LP approach finds a direction b that maximizes sum(b) subject to (a*b)(k) <= M,
which guarantees improvement when mixed with the current point.

import numpy as np
import time
from scipy.optimize import linprog


def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_conv_matrix_row(a, k):
    """Build the k-th row of the cross-correlation matrix.
    Row k has A[k,j] = a[k-j] if 0 <= k-j < n, else 0.
    """
    n = a.size
    row = np.zeros(n)
    j_min = max(0, k - (n - 1))
    j_max = min(n - 1, k)
    if j_min <= j_max:
        row[j_min:j_max+1] = a[k - j_max:k - j_min + 1][::-1]
    return row


def solve_lp_direction(a, M):
    """Solve LP: max sum(b) s.t. (a*b)(k) <= M for all k, b >= 0."""
    n = a.size
    c = -np.ones(n)
    
    # Build constraint matrix A where A[k,j] = a[k-j] if valid
    # (a*b)(k) = sum_j A[k,j] * b_j
    num_constraints = 2 * n - 1
    A = np.zeros((num_constraints, n))
    for k in range(num_constraints):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        if j_min <= j_max:
            # A[k, j] = a[k-j] for j in [j_min, j_max]
            A[k, j_min:j_max+1] = a[k - j_max : k - j_min + 1][::-1]
    
    b_ub = np.full(num_constraints, M)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    return None


def lp_optimize(a, time_budget, rng, verbose=True):
    """LP-based optimization."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    t0 = time.monotonic()
    iters = 0
    last_improve = 0
    t_mix = 0.3
    
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        M = float(np.max(b))
        cur_sc = 2.0 * n * M
        
        # Solve LP
        b_dir = solve_lp_direction(a, M)
        if b_dir is None:
            # Fallback: use subgradient step
            break
        
        sum_b = float(np.sum(b_dir))
        if sum_b < 1e-10:
            break
            
        # Rescale b_dir to have same sum as a (which is 1)
        b_rescaled = b_dir / sum_b
        
        # Mix
        a_new = (1.0 - t_mix) * a + t_mix * b_rescaled
        a_new = project_simplex(a_new)
        
        new_sc = normed_score(a_new)
        
        if new_sc < cur_sc:
            a = a_new
            if new_sc < best_sc:
                best_sc = new_sc
                best_a = a.copy()
                last_improve = iters
                if verbose and iters % 50 == 0:
                    print(f"    [lp] iter={iters}, score={new_sc:.6f}")
        else:
            # Try smaller mix
            t_mix = max(0.01, t_mix * 0.5)
        
        iters += 1
        
        # Reset mix parameter periodically
        if iters % 20 == 0:
            t_mix = min(0.5, t_mix * 1.5 + 0.05)
        
        # Check for stall
        if iters - last_improve > 500:
            # Try a perturbation
            a = project_simplex(best_a + 0.01 * rng.standard_normal(n))
            last_improve = iters
            t_mix = 0.3
    
    return best_a, best_sc

def subgrad_polish(a, time_budget, rng):
    """Polish using smoothed subgradient descent."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    eta = 0.1
    tau = 0.01
    vel = np.zeros(n)
    beta = 0.5
    iters = 0
    last_improve = 0
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))
        
        g = smooth_subgrad_step(a, b, tau)
        vel = beta * vel + (1.0 - beta) * g
        a_new = project_simplex(a - eta * vel)
        
        b_new = full_conv(a_new)
        new_sc = 2.0 * n * float(np.max(b_new))
        
        if new_sc <= cur_sc or rng.random() < 0.05:
            a = a_new
        else:
            a = project_simplex(a - 0.5 * eta * vel)
        
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = iters
        
        iters += 1
        if iters - last_improve > 500:
            a = project_simplex(best_a + 0.01 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.05 + 0.1 * rng.random()
            tau = 0.005 + 0.02 * rng.random()
            last_improve = iters
        
        if iters % 1000 == 0:
            eta *= 0.7
            tau *= 0.8
    
    return best_a, best_sc


def coord_descent_polish(a, time_budget, rng):
    """Coordinate descent: try adjusting pairs of coordinates."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    t0 = time.monotonic()

        improved = False
        # Randomly select a few coordinate pairs to adjust

        for _ in range(num_trials):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = scale * (2.0 * rng.random() - 1.0)
            a[i] -= delta
            a[j] += delta
            if a[i] < 0 or a[j] < 0:
                # Project back
                a = project_simplex(a)
            else:
                # Check if sum is preserved (it is by construction)
                pass
            
            sc = normed_score(a)

            else:
                # Revert
                a[i] += delta
                a[j] -= delta

def init_gaussian_mixture(n, rng):
    """Initialize with a mixture of Gaussian peaks."""
    i = np.arange(n, dtype=np.float64)
    n_peaks = 2 + int(rng.integers(0, 2))
    a = np.zeros(n)
    for _ in range(n_peaks):
        c = rng.uniform(0, n)
        sigma = n * rng.uniform(0.05, 0.2)
        a += np.exp(-((i - c) / sigma) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    a = np.maximum(a, 0)
    return a / a.sum()


def init_edges(n, rng):
    """High at edges, low in middle."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    width = n * rng.uniform(0.1, 0.3)
    a = np.exp(-((i - c) / width) ** 2)
    a = 1.0 / (1.0 + a)  # Invert: high at edges
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_bimodal(n, rng):
    """Two well-separated peaks."""
    i = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.1, 0.3)
    c2 = n * rng.uniform(0.7, 0.9)
    sigma = n * rng.uniform(0.05, 0.15)
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_random_smooth(n, rng):
    """Random smooth function via Fourier series."""
    freqs = rng.integers(1, 5, size=5)
    phases = rng.uniform(0, 2*np.pi, size=5)
    amps = rng.uniform(0.1, 1.0, size=5)
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for f, p, amp in zip(freqs, phases, amps):
        a += amp * np.cos(2*np.pi*f*i/n + p)
    a = np.abs(a) + 0.05
    return a / a.sum()


def get_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    inits.append(init_gaussian_mixture(n, rng))
    inits.append(init_edges(n, rng))
    inits.append(init_bimodal(n, rng))
    inits.append(init_random_smooth(n, rng))
    inits.append(np.ones(n) / n)  # Uniform
    return inits
```

```python
"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding
combined with smoothed subgradient descent and coordinate descent polish.
The LP approach (from SOTA) finds a direction b maximizing sum(b) subject to
(a*b)(k) <= M, guaranteeing improvement when mixed with the current point.
Multi-scale subgradient polishing and coordinate descent refine the result.
"""

import numpy as np
import time
from scipy.optimize import linprog


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_constraint_matrix(a):
    """Build matrix A where A[k,j] = a[k-j] if 0<=k-j<n, else 0.
    Shape: (2n-1, n)."""
    n = a.size
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_min = max(0, k - (n - 1))
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max + 1] = a[k - j_max:k - j_min + 1][::-1]
    return A


def solve_lp_direction(a, M, max_coord_frac=0.3):
    """Solve LP: max sum(b) s.t. (a*b)(k) <= M for all k, b >= 0, b_j <= max_coord_frac*M/a_j_min."""
    n = a.size
    c = -np.ones(n)
    A = build_constraint_matrix(a)
    b_ub = np.full(2 * n - 1, M)
    
    # Add upper bound constraints to prevent spiky solutions
    a_min_pos = np.where(a > 1e-15, a, 1e-15)
    coord_bound = np.minimum(max_coord_frac * M / a_min_pos, M * 10)
    bounds = [(0.0, coord_bound[j]) for j in range(n)]
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success and res.x is not None:
        return res.x
    return None


def lp_optimize(a, time_budget, rng, verbose=False):
    """LP-based optimization with line search on mixing parameter."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    t0 = time.monotonic()
    iters = 0
    last_improve = 0
    t_mix = 0.3
    consecutive_fail = 0
    
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        M = float(np.max(b))
        cur_sc = 2.0 * n * M
        
        if M < 1e-18:
            break
        
        b_dir = solve_lp_direction(a, M)
        if b_dir is None:
            consecutive_fail += 1
            if consecutive_fail > 5:
                break
            a = project_simplex(a + 0.01 * rng.standard_normal(n))
            continue
        
        sum_b = float(np.sum(b_dir))
        if sum_b < 1e-10:
            break
        
        b_rescaled = b_dir / sum_b
        
        # Line search: try several mixing ratios
        best_new_a = None
        best_new_sc = cur_sc
        for t_candidate in [t_mix, t_mix * 0.5, t_mix * 0.7, t_mix * 1.3]:
            t_candidate = np.clip(t_candidate, 0.05, 0.8)
            a_candidate = (1.0 - t_candidate) * a + t_candidate * b_rescaled
            a_candidate = project_simplex(a_candidate)
            sc_candidate = normed_score(a_candidate)
            if sc_candidate < best_new_sc:
                best_new_sc = sc_candidate
                best_new_a = a_candidate.copy()
        
        if best_new_a is not None and best_new_sc < cur_sc:
            a = best_new_a
            consecutive_fail = 0
            if best_new_sc < best_sc:
                best_sc = best_new_sc
                best_a = a.copy()
                last_improve = iters
                if verbose and (iters % 20 == 0 or best_sc < 1.51):
                    print(f"      [lp] iter={iters}, score={best_sc:.6f}")
            # Adjust t_mix based on which candidate was best
            if best_new_sc < cur_sc:
                t_mix = min(0.6, t_mix * 1.1)
        else:
            consecutive_fail += 1
            t_mix = max(0.05, t_mix * 0.6)
        
        iters += 1
        
        # Stall detection
        if iters - last_improve > 200:
            a = project_simplex(best_a + 0.02 * rng.standard_normal(n))
            last_improve = iters
            t_mix = 0.3
            consecutive_fail = 0
    
    return best_a, best_sc


def smooth_subgrad_step(a, b, tau):
    """Compute smoothed subgradient of max_k (a*a)(k)."""
    n = a.size
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(b.size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= s
    
    g = np.zeros(n)
    for j in range(n):
        k_min = j
        k_max = min(2 * n - 1, j + n)
        if k_min < k_max:
            g[j] = 2.0 * np.dot(w[k_min:k_max], a[0:k_max - k_min])
    return g


def multi_scale_subgrad(a, b, tau_base, n_scales):
    """Average subgradients at multiple temperature scales."""
    g_total = np.zeros(a.size)
    for s in range(n_scales):
        tau = tau_base * (2.0 ** s)
        g_total += smooth_subgrad_step(a, b, tau)
    return g_total / n_scales


def subgrad_optimize(a, time_budget, rng, verbose=False):
    """Smoothed subgradient descent with momentum."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    eta = 0.2
    tau_frac = 0.03
    n_scales = 3
    beta = 0.5
    vel = np.zeros(n)
    iters = 0
    last_improve = 0
    stall = 0
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))
        
        tau = tau_frac * float(np.max(b))
        if tau < 1e-18:
            tau = 1e-18
        
        if n_scales > 1:
            g = multi_scale_subgrad(a, b, tau, n_scales)
        else:
            g = smooth_subgrad_step(a, b, tau)
        
        vel = beta * vel + (1.0 - beta) * g
        a_new = project_simplex(a - eta * vel)
        
        b_new = full_conv(a_new)
        new_sc = 2.0 * n * float(np.max(b_new))
        
        if new_sc <= cur_sc or rng.random() < 0.03:
            a = a_new
        else:
            a = project_simplex(a - 0.4 * eta * vel)
        
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = iters
            stall = 0
            if verbose and iters % 50 == 0:
                print(f"      [sg] iter={iters}, score={best_sc:.6f}")
        elif iters - last_improve > 1000:
            stall += 1
        
        iters += 1
        
        if stall > 0 and stall % 3 == 0:
            eta *= 0.7
        
        if iters - last_improve > 4000:
            a = project_simplex(best_a + 0.03 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.1 + 0.2 * rng.random()
            tau_frac = 0.02 + 0.03 * rng.random()
            last_improve = iters
            stall = 0
    
    return best_a, best_sc


def coord_polish(a, time_budget, rng):
    """Coordinate descent: adjust pairs of coordinates to reduce score."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    t0 = time.monotonic()
    scale = 0.03
    
    while time.monotonic() - t0 < time_budget and scale > 1e-7:
        improved = False
        num_trials = min(n, 100)
        
        for _ in range(num_trials):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i == j:
                continue
            delta = scale * (2.0 * rng.random() - 1.0)
            a[i] -= delta
            a[j] += delta
            
            if a[i] < 0 or a[j] < 0:
                a = project_simplex(a)
            
            sc = normed_score(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                improved = True
            else:
                # Revert
                if not (a[i] < 0 or a[j] < 0):
                    a[i] += delta
                    a[j] -= delta
                else:
                    a = project_simplex(a)
        
        if improved:
            a = best_a.copy()
        else:
            scale *= 0.6
    
    return best_a, best_sc


def init_gaussian_mixture(n, rng):
    """Initialize with a mixture of Gaussian peaks."""
    i = np.arange(n, dtype=np.float64)
    n_peaks = 2 + int(rng.integers(0, 2))
    a = np.zeros(n)
    for _ in range(n_peaks):
        c = rng.uniform(0, n)
        sigma = n * rng.uniform(0.05, 0.2)
        a += np.exp(-((i - c) / sigma) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    a = np.maximum(a, 0)
    return a / a.sum()


def init_edges(n, rng):
    """High at edges, low in middle."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    width = n * rng.uniform(0.1, 0.3)
    a = np.exp(-((i - c) / width) ** 2)
    a = 1.0 / (1.0 + a * 5.0)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_bimodal_separated(n, rng):
    """Two well-separated peaks, not summing to n-1."""
    i = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.1, 0.35)
    c2 = n * rng.uniform(0.55, 0.85)
    sigma = n * rng.uniform(0.05, 0.15)
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_random_smooth(n, rng):
    """Random smooth function via Fourier series."""
    freqs = rng.integers(1, 8, size=6)
    phases = rng.uniform(0, 2 * np.pi, size=6)
    amps = rng.uniform(0.1, 1.0, size=6)
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for f, p, amp in zip(freqs, phases, amps):
        a += amp * np.cos(2 * np.pi * f * i / n + p)
    a = np.abs(a) + 0.05
    return a / a.sum()


def init_u_shape(n, rng):
    """U-shape: high at both edges."""
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i).astype(np.float64) + 1.0
    exponent = rng.uniform(0.5, 2.0)
    a = 1.0 / (m ** exponent)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()

    inits.append(init_gaussian_mixture(n, rng))
    inits.append(init_edges(n, rng))
    inits.append(init_bimodal_separated(n, rng))
    inits.append(init_random_smooth(n, rng))
    inits.append(init_u_shape(n, rng))
    inits.append(np.ones(n) / n)
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses LP-based direction finding (SOTA), smoothed subgradient descent,
    and coordinate descent polish across multiple sequence lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 20.0)
    
    best_overall = None
    best_overall_sc = np.inf
    
    # Load previous best
    prev_a = None
    prev_sc = np.inf
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size >= 10 and np.all(np.isfinite(h)) and h.sum() > 0.01:
            prev_a = h / h.sum()
            prev_sc = 2.0 * h.size * float(np.max(full_conv(prev_a)))
            print(f"[init] Previous best: n={h.size}, score={prev_sc:.6f}")
            best_overall = prev_a.copy()
            best_overall_sc = prev_sc
    except Exception as e:
        print(f"[init] No previous: {e}")
    
    # n values to try
    n_list = [128, 256, 512, 1024, 2048]
    if prev_a is not None:
        np_ = prev_a.size
        if np_ not in n_list and 64 <= np_ <= 4096:
            n_list.append(np_)
            n_list.sort()
    
    total_time = max(1.0, deadline - time.monotonic())
    print(f"[plan] n_list={n_list}, total_time={total_time:.1f}s")
    
    # Time allocation weights
    raw_w = []
    for nv in n_list:
        w = 1.0 / (0.5 + nv / 512.0)
        if prev_a is not None and prev_a.size == nv:
            w *= 2.5
        raw_w.append(w)
    total_w = sum(raw_w)
    weights = [w / total_w for w in raw_w]
    
    for idx_n, nv in enumerate(n_list):
        if time.monotonic() > deadline:
            break
        
        time_n = weights[idx_n] * total_time * 0.88
        remaining = deadline - time.monotonic()
        time_n = min(time_n, remaining)
        
        if time_n < 15.0:
            continue
        
        print(f"\n[n={nv}] allocated: {time_n:.1f}s")
        
        inits = get_inits(nv, rng)
        if prev_a is not None and prev_a.size == nv:
            inits.insert(0, prev_a.copy())
        
        time_per_init = time_n / len(inits)
        
        for ii, a_init in enumerate(inits):
            if time.monotonic() > deadline:
                break
            
            # Allocate phase times
            remain = deadline - time.monotonic() - 3.0
            if remain < 10.0:
                break
            
            tb_lp = min(time_per_init * 0.5, remain * 0.5)
            tb_sg = min(time_per_init * 0.3, remain * 0.3)
            tb_cd = min(time_per_init * 0.2, remain * 0.2)
            
            if tb_lp < 3.0 and tb_sg < 3.0:
                continue
            
            # Phase 1: LP optimization
            a_cur, sc_cur = a_init.copy(), normed_score(a_init)
            if tb_lp >= 3.0:
                a_lp, sc_lp = lp_optimize(a_init, tb_lp, rng, verbose=(ii == 0))
                if sc_lp < sc_cur:
                    a_cur, sc_cur = a_lp, sc_lp
            
            # Phase 2: Subgradient polish
            if time.monotonic() < deadline - 10.0:
                tb_sg_actual = min(tb_sg, deadline - time.monotonic() - 5.0)
                if tb_sg_actual > 3.0:
                    a_sg, sc_sg = subgrad_optimize(a_cur, tb_sg_actual, rng, verbose=(ii == 0))
                    if sc_sg < sc_cur:
                        a_cur, sc_cur = a_sg, sc_sg
            
            # Phase 3: Coordinate descent
            if time.monotonic() < deadline - 5.0:
                tb_cd_actual = min(tb_cd, deadline - time.monotonic() - 2.0)
                if tb_cd_actual > 2.0:
                    a_cd, sc_cd = coord_polish(a_cur, tb_cd_actual, rng)
                    if sc_cd < sc_cur:
                        a_cur, sc_cur = a_cd, sc_cd
            
            if sc_cur < best_overall_sc:
                best_overall_sc = sc_cur
                best_overall = a_cur.copy()
                print(f"  [init {ii}] *** NEW BEST *** n={nv}, score={sc_cur:.6f}")
            else:
                print(f"  [init {ii}] n={nv}, score={sc_cur:.6f}")
        
        print(f"[n={nv}] done. Overall best: {best_overall_sc:.6f}")
    
    # Final intensive polish on best found
    if best_overall is not None and time.monotonic() < deadline - 30.0:
        remain = deadline - time.monotonic() - 10.0
        n_best = best_overall.size
        print(f"\n[final_polish] n={n_best}, remaining={remain:.1f}s")
        
        # Multiple rounds of LP + SG
        for round_idx in range(3):
            if time.monotonic() > deadline - 15.0:
                break
            phase_time = min(remain / (3 - round_idx), 120.0)
            
            # LP phase
            tb = min(phase_time * 0.5, deadline - time.monotonic() - 10.0)
            if tb > 5.0:
                a_p, sc_p = lp_optimize(best_overall, tb, rng, verbose=True)
                if sc_p < best_overall_sc:
                    best_overall_sc = sc_p
                    best_overall = a_p.copy()
                    print(f"  [polish_r{round_idx}] LP: {sc_p:.6f}")
            
            # SG phase
            tb = min(phase_time * 0.35, deadline - time.monotonic() - 10.0)
            if tb > 5.0:
                a_p2, sc_p2 = subgrad_optimize(best_overall, tb, rng, verbose=True)
                if sc_p2 < best_overall_sc:
                    best_overall_sc = sc_p2
                    best_overall = a_p2.copy()
                    print(f"  [polish_r{round_idx}] SG: {sc_p2:.6f}")
            
            # CD phase
            tb = min(phase_time * 0.15, deadline - time.monotonic() - 5.0)
            if tb > 3.0:
                a_p3, sc_p3 = coord_polish(best_overall, tb, rng)
                if sc_p3 < best_overall_sc:
                    best_overall_sc = sc_p3
                    best_overall = a_p3.copy()
                    print(f"  [polish_r{round_idx}] CD: {sc_p3:.6f}")
    
    # Fallback
    if best_overall is None:
        best_overall = np.ones(128) / 128
        best_overall_sc = 2.0
    
    # Final verification
    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"\n[FINAL] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")
    
    return result
```
