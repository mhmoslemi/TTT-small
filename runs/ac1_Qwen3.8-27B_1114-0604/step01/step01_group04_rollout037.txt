    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size

    while time.monotonic() - t0 < time_budget:
        # Choose move type
        r = rng.random()
        if r < 0.4:
            # Gaussian perturbation
            cand = a + scale * rng.standard_normal(n)
            cand = project_simplex(cand)
        elif r < 0.7:
            # Pair transfer: move mass from i to j
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            delta = scale * a[i] * rng.random()
            cand = a.copy()
            cand[i] -= delta
            cand[j] += delta
            if cand[i] < 0:
                continue
            cand = project_simplex(cand)
        else:
            # Block swap: exchange two blocks
            block_len = max(1, n // (4 + rng.integers(8)))
            i = rng.integers(0, n - block_len)
            j = rng.integers(0, n - block_len)
            if abs(i - j) < block_len:
                continue
            cand = a.copy()
            cand[i:i + block_len], cand[j:j + block_len] = cand[j:j + block_len].copy(), cand[i:i + block_len].copy()
            cand = project_simplex(cand)

        sc = score_normalized(cand)
        if sc < best_score:
            best_score = sc
            best_a = cand.copy()
            a = cand.copy()
            scale = max(0.001, scale * 1.0)
        else:
            # Accept with small probability (simulated annealing)
            if rng.random() < 0.05:
                a = cand
            scale = max(0.001, scale * 0.98)

        iters += 1

    return best_a, best_score


def init_complementary(n, rng):
    """Initialize with complementary structure: a[i] high where a[n-1-i] is low."""
    t = rng.random(n)
    # Make pairs complementary
    a = np.zeros(n)
    for i in range(n):
        j = n - 1 - i
        a[i] = t[i] * (1.0 - t[j]) + 0.01
        if i == j:
            a[i] = 0.3 + 0.3 * rng.random()
    a = np.abs(a) + 0.01
    return a / a.sum()


def init_two_bump(n, rng):
    """Two Gaussian bumps at random positions."""
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.2 + 0.15 * rng.random())
    c2 = n * (0.65 + 0.15 * rng.random())
    sig = n * (0.05 + 0.1 * rng.random())
    a = np.exp(-((i - c1) / sig) ** 2) + np.exp(-((i - c2) / sig) ** 2)
    a += 0.02 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_three_bump(n, rng):
    """Three Gaussian bumps."""
    i = np.arange(n, dtype=np.float64)
    centers = [n * p for p in sorted(rng.random(3) * 0.8 + 0.1)]
    sig = n * (0.04 + 0.06 * rng.random())
    a = np.zeros(n)
    for c in centers:
        a += np.exp(-((i - c) / sig) ** 2)
    a += 0.02 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_u_shape(n, rng):
    """U-shaped: high at both edges."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / (c + 1)) ** 2 + 0.1
    a += 0.02 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_cosine_mod(n, rng):
    """Cosine-modulated amplitude."""
    i = np.arange(n, dtype=np.float64)
    freq = 1.0 + rng.integers(0, 4)
    phase = rng.random() * np.pi
    a = (1.0 + np.cos(2.0 * np.pi * freq * i / n + phase)) ** 2 + 0.05
    return a / a.sum()


def init_random_smooth(n, rng):
    """Random smooth sequence (low-pass filtered noise)."""
    a = np.abs(rng.standard_normal(n)) + 0.1
    # Smooth with a small kernel
    k = max(3, n // 32)
    kernel = np.exp(-0.5 * (np.arange(2 * k + 1) - k) ** 2 / (k * 0.5))
    kernel /= kernel.sum()
    a = np.convolve(a, kernel, mode='same')
    a = np.abs(a) + 0.01
    return a / a.sum()


def init_fourier_flat(n, rng):
    """Initialize from random phases in frequency domain (flat spectrum target)."""
    # Generate random complex spectrum with random phases
    freqs = np.fft.rfftfreq(n)
    # Suppress DC
    magnitude = np.ones(n // 2 + 1)
    magnitude[0] = 0
    phase = rng.random(n // 2 + 1) * 2 * np.pi
    spectrum = magnitude * np.exp(1j * phase)
    a = np.fft.irfft(spectrum * n)
    a = np.abs(a) + 0.05
    return a / a.sum()


def get_inits(n, rng, prev_a=None):
    """Generate diverse initializations for given n."""
    inits = []
    if prev_a is not None and prev_a.size == n:
        inits.append(prev_a.copy())
    inits.append(init_complementary(n, rng))
    inits.append(init_two_bump(n, rng))
    inits.append(init_three_bump(n, rng))
    inits.append(init_u_shape(n, rng))
    inits.append(init_cosine_mod(n, rng))
    inits.append(init_random_smooth(n, rng))
    inits.append(init_fourier_flat(n, rng))
    # Pure random
    a = np.abs(rng.standard_normal(n)) + 0.05
    inits.append(a / a.sum())
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses smooth gradient + coordinate descent + local search.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    # Try to load previous best
    prev_a = None
    prev_sc = np.inf
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size >= 8 and np.all(np.isfinite(h)) and h.sum() > 0.01:
            prev_a = h / h.sum()
            prev_sc = 2.0 * h.size * float(np.max(full_conv(prev_a)))
            print(f"[init] Previous best: n={h.size}, score={prev_sc:.6f}")
    except Exception:
        pass

    best_overall = None
    best_overall_sc = np.inf

    if prev_a is not None and prev_sc < 2.0:
        best_overall = prev_a.copy()
        best_overall_sc = prev_sc

    # n values to try
    n_list = [96, 128, 160, 192, 256]
    if prev_a is not None:
        np_ = prev_a.size
        if np_ not in n_list and np_ >= 32:
            n_list.append(np_)
        n_list = sorted(set(n_list))

    total_time = max(1.0, deadline - time.monotonic())
    print(f"[plan] n_list={n_list}, total_time={total_time:.1f}s")

    # Allocate time: give more to n=128 and nearby
    weights = []
    for nv in n_list:
        w = 1.0 / (1.0 + 0.1 * abs(nv - 128) / 32.0)
        weights.append(w)
    total_w = sum(weights)
    weights = [w / total_w for w in weights]

    for idx_n, nv in enumerate(n_list):
        if time.monotonic() > deadline:
            break

        time_n = weights[idx_n] * total_time * 0.9
        inits = get_inits(nv, rng, prev_a)

        # Phase 1: Smooth optimization for each init
        inits_per_smooth = max(2.0, time_n * 0.4 / len(inits))
        best_for_n = None
        best_sc_n = np.inf

        for ii, a_init in enumerate(inits):
            if time.monotonic() > deadline:
                break
            tb = min(inits_per_smooth, deadline - time.monotonic() - 2.0)
            if tb < 1.0:
                break

            a_best, sc_best = smooth_optimize(a_init, tb, rng)

            if sc_best < best_sc_n:
                best_sc_n = sc_best
                best_for_n = a_best.copy()

            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = a_best.copy()
                print(f"[smooth] n={nv}, init={ii}, score={sc_best:.6f} (BEST)")

        print(f"[smooth] n={nv} done. Best for n: {best_sc_n:.6f}")

        # Phase 2: Coordinate descent on best for this n
        if best_for_n is not None and time.monotonic() < deadline - 5.0:
            time_cd = min(time_n * 0.4, deadline - time.monotonic() - 3.0)
            if time_cd > 3.0:
                a_cd, sc_cd = coord_descent(best_for_n, time_cd, rng, use_exact=True)
                if sc_cd < best_sc_n:
                    best_sc_n = sc_cd
                    best_for_n = a_cd.copy()
                if sc_cd < best_overall_sc:
                    best_overall_sc = sc_cd
                    best_overall = a_cd.copy()
                    print(f"[coord] n={nv}, score={sc_cd:.6f} (BEST)")
                else:
                    print(f"[coord] n={nv}, score={sc_cd:.6f}")

        # Phase 3: Local search
        if best_for_n is not None and time.monotonic() < deadline - 3.0:
            time_ls = min(time_n * 0.2, deadline - time.monotonic() - 2.0)
            if time_ls > 2.0:
                a_ls, sc_ls = local_search(best_for_n, time_ls, rng)
                if sc_ls < best_overall_sc:
                    best_overall_sc = sc_ls
                    best_overall = a_ls.copy()
                    print(f"[ls] n={nv}, score={sc_ls:.6f} (BEST)")

    # Final focused polish on the overall best
    if best_overall is not None and time.monotonic() < deadline - 10.0:
        remain = deadline - time.monotonic() - 5.0
        if remain > 5.0:
            print(f"[polish] remaining={remain:.1f}s, n={best_overall.size}, score={best_overall_sc:.6f}")

            # Extended coordinate descent
            time_ext_cd = remain * 0.5
            a_p, sc_p = coord_descent(best_overall, time_ext_cd, rng, use_exact=True)
            if sc_p < best_overall_sc:
                best_overall_sc = sc_p
                best_overall = a_p.copy()
                print(f"[polish-cd] improved to {sc_p:.6f}")

            # Final local search
            if time.monotonic() < deadline - 5.0:
                time_ls2 = min(remain * 0.3, deadline - time.monotonic() - 3.0)
                if time_ls2 > 2.0:
                    a_l2, sc_l2 = local_search(best_overall, time_ls2, rng)
                    if sc_l2 < best_overall_sc:
                        best_overall_sc = sc_l2
                        best_overall = a_l2.copy()
                        print(f"[polish-ls] improved to {sc_l2:.6f}")

    # Fallback
    if best_overall is None:
        best_overall = np.ones(128) / 128.0
        best_overall_sc = 2.0

    # Final verification with evaluate_sequence
    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"[final] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")

    return result
