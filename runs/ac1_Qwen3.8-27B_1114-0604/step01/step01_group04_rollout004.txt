Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based fixed-point iteration.
The key idea: given a current sequence f, solve an LP to find the best response g
(maximizing sum(g) subject to conv(f,g) <= max(conv(f,f))), then blend and iterate.
Uses multiple sequence lengths, diverse initializations, and adaptive parameters.

import numpy as np
from scipy.optimize import linprog
import time


def full_conv(a):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()

    return np.fft.irfft(fa * fa, fsize)[:size]

    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def build_lp_matrix(n):
    """Build the constraint matrix for the LP. A[k,j] = 1 if g_j contributes to (f*g)_k."""
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        A[k, j_start:j_end+1] = 1.0
    return A

    n = len(f)
    # (f*g)_k = sum_j f_j * g_{k-j}
    # The constraint matrix is: row k has f[j] at position j where j is in valid range
    # We can build this as: A[k,:] = [f[max(0,k-n+1)], ..., f[min(k,n-1)]] placed at correct positions
    
    # Actually, let's build it efficiently using the structure
    # A_lp[k, j] = f[j] if max(0, k-n+1) <= j <= min(k, n-1), else 0
    
    # For efficiency, precompute the indices
    A_lp = np.zeros((2*n-1, n))
    for j in range(n):
        # g_j contributes to (f*g)_k for k = j, j+1, ..., j+n-1
        # With coefficient f[k-j]
        k_start = j
        k_end = min(j + n - 1, 2*n - 2)
        for k in range(k_start, k_end + 1):
            A_lp[k, j] = f[k - j]
    
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)

    if res.success:
        g = res.x
        s = g.sum()
        if s > 1e-15:
            return g / s
    return None

        return a, M

    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-15:
        return a, M
    a_new /= s
    
    return a_new, M

import time

def solve_lp_response(f, M):
    """Given f (normalized, sum=1) and M = max(conv(f,f)), find g maximizing sum(g)

    n = len(f)

    A = np.zeros((2*n-1, n))
    for j in range(n):
        k_start = j

        A[k_start:k_end+1, j] = f[0:min(n, k_end+1-j)]
