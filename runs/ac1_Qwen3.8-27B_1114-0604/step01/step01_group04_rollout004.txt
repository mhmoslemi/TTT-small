"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using projected gradient descent
with momentum, softmax-smoothed gradients, temperature continuation, adaptive
step sizes, and U-shape (arc-sine family) initializations.
Explores multiple sequence lengths with time-weighted allocation.

import numpy as np
import time


def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def project_simplex(v):
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def smooth_gradient(a, b, tau, n_scales=3):
    """Smoothed subgradient of max(conv(a,a)) using multi-scale softmax."""
    n = a.size
    g = np.zeros(n)
    for s in range(n_scales):
        t = tau * (2.0 ** s)
        shift = b / t
        shift -= shift.max()
        w = np.exp(shift)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(b.size)
            w[int(np.argmax(b))] = 1.0
        else:
            w /= ws
        g += np.correlate(w, a, mode='valid')
    return 2.0 * g / n_scales

    """U-shaped initialization: a[i] proportional to (1/sqrt((i+a)(n-i-a)))^power.
    power=0 gives uniform, power=1 gives arc-sine."""
    i = np.arange(n, dtype=np.float64) + alpha
    base = 1.0 / np.sqrt(i * (n - i))
    a = base ** power
    a += 1e-10
    return a / a.sum()

    inits.append(u_shape_init(n, alpha=0.5, power=1.0))
    inits.append(u_shape_init(n, alpha=0.5, power=0.7))
    inits.append(u_shape_init(n, alpha=1.0, power=0.8))
    inits.append(u_shape_init(n, alpha=0.5, power=0.5))
    inits.append(u_shape_init(n, alpha=1.5, power=1.0))
    # Noisy versions
    inits.append(u_shape_init(n, 0.5, 0.8) + 0.05 * np.abs(rng.standard_normal(n)))
    inits[-1] = inits[-1] / inits[-1].sum()
    # Two-bump
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.2 + 0.1 * rng.random())
    c2 = n * (0.7 + 0.1 * rng.random())
    sigma = n * (0.05 + 0.05 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.01
    inits.append(a / a.sum())
    # Uniform with noise
    a = np.ones(n) + 0.3 * rng.standard_normal(n)
    a = np.abs(a) + 0.01
    inits.append(a / a.sum())
    return inits


def optimize_sequence(a_init, time_budget, rng, n_scales=3):
    """
    Optimize a sequence using projected gradient with momentum and
    adaptive temperature/step size. Returns (best_a, best_score).
    """
    n = a_init.size
    a = project_simplex(a_init.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = score(a)

    tau_frac = 0.06
    eta = 0.5
    beta = 0.7
    vel = np.zeros(n)
    it = 0
    last_improve = 0
    stall_count = 0

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        max_b = float(b.max())
        cur_sc = 2.0 * n * max_b
        tau = max(tau_frac * max_b, 1e-15)

        g = smooth_gradient(a, b, tau, n_scales)
        vel = beta * vel + (1.0 - beta) * g

        # Try a step with backtracking on the smoothed objective
        smoothed_cur = float(np.dot(b / tau, b / tau - (b / tau).max()))  # proxy
        step = eta
        accepted = False
        for backtrack in range(5):
            a_new = project_simplex(a - step * vel)
            b_new = full_conv(a_new)
            new_sc = 2.0 * n * float(b_new.max())
            if new_sc <= cur_sc + 1e-10:
                a = a_new
                accepted = True
                break
            step *= 0.5
        if not accepted:
            a = project_simplex(a - step * vel)
            b_new = full_conv(a)
            new_sc = 2.0 * n * float(b_new.max())

        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = it
            stall_count = 0
        else:
            stall_count += 1

        # Adaptive parameter adjustment
        if stall_count > 200 and stall_count % 100 == 0:
            tau_frac = max(0.005, tau_frac * 0.85)
            eta = 0.2 + 0.6 * rng.random()
            beta = 0.5 + 0.35 * rng.random()

        # Restart from best with perturbation if stuck
        if it - last_improve > 5000:
            a = project_simplex(best_a + 0.02 * rng.standard_normal(n))
            vel[:] = 0.0
            eta = 0.3 + 0.4 * rng.random()
            tau_frac = max(0.01, tau_frac * 1.5)
            last_improve = it
            stall_count = 0

        # Periodic cooling
        if it % 10000 == 0 and it > 0:
            tau_frac = max(0.005, tau_frac * 0.7)

        it += 1

    return best_a, best_sc


def polish_local(a_init, time_budget, rng):
    """Local polish with small random perturbations and exact evaluation."""
    n = a_init.size
    a = project_simplex(a_init.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = score(a)

    t0 = time.monotonic()
    scale = 0.01
    while time.monotonic() - t0 < time_budget and scale > 1e-7:
        improved = False
        n_trials = 20
        for _ in range(n_trials):
            cand = a + scale * rng.standard_normal(n)
            cand = project_simplex(cand)
            sc = score(cand)
            if sc < best_sc:
                best_sc = sc
                best_a = cand.copy()
                a = cand.copy()
                improved = True
        if not improved:
            scale *= 0.7
        else:
            scale *= 0.95

    # Final subgradient polish
    if time.monotonic() - t0 < time_budget * 0.3:
        remain = time_budget - (time.monotonic() - t0)
        if remain > 5:
            a_p, sc_p = optimize_sequence(best_a, remain, rng, n_scales=2)
            if sc_p < best_sc:
                best_sc = sc_p
                best_a = a_p.copy()

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses U-shape initializations, projected gradient with momentum,
    temperature continuation, and multi-scale exploration.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 15.0)

    best_a = None
    best_sc = np.inf

    # Try to load previous best
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size >= 10 and np.all(np.isfinite(h)) and h.sum() > 0.01:
            prev_a = h / h.sum()
            prev_sc = score(prev_a)
            best_a = prev_a.copy()
            best_sc = prev_sc
            print(f"[init] Previous best: n={h.size}, score={prev_sc:.6f}")
    except Exception:
        pass

    # n values to explore
    n_list = [256, 512, 1024, 2048, 4096]
    if best_a is not None:
        pn = best_a.size
        if pn not in n_list and pn >= 128:
            n_list.append(pn)
        n_list = sorted(set(n_list))

    total_time = max(1.0, deadline - time.monotonic())
    print(f"[plan] n_list={n_list}, total_time={total_time:.1f}s, current best={best_sc:.6f}")

    # Time allocation: favor larger n (better theoretical bounds) and previous best n
    raw_w = []
    for nv in n_list:
        w = 1.0 + 0.5 * (nv / 1024.0)
        if best_a is not None and nv == best_a.size:
            w *= 1.5
        raw_w.append(w)
    total_w = sum(raw_w)
    weights = [w / total_w for w in raw_w]

    for idx_n, nv in enumerate(n_list):
        if time.monotonic() > deadline:
            break

        time_n = weights[idx_n] * total_time * 0.85
        inits = init_family(nv, rng)

        # If previous best has this length, insert it first
        if best_a is not None and best_a.size == nv:
            inits.insert(0, best_a.copy())

        n_inits = len(inits)
        time_per_init = time_n / n_inits

        for ii, a_init in enumerate(inits):
            if time.monotonic() > deadline:
                break
            tb = min(time_per_init, deadline - time.monotonic() - 2.0)
            if tb < 5.0:
                break

            a_best, sc_best = optimize_sequence(a_init, tb, rng)

            if sc_best < best_sc:
                best_sc = sc_best
                best_a = a_best.copy()
                print(f"[opt] n={nv}, init={ii}, score={sc_best:.6f} *** NEW BEST ***")
            else:
                print(f"[opt] n={nv}, init={ii}, score={sc_best:.6f}")

        print(f"[phase] n={nv} done. Best so far: {best_sc:.6f}")

    # Final polish phase
    if best_a is not None and time.monotonic() < deadline - 10.0:
        remain = deadline - time.monotonic() - 5.0
        if remain > 20.0:
            print(f"[polish] Starting final polish, remain={remain:.1f}s, n={best_a.size}")
            pol_time1 = min(remain * 0.5, 120.0)
            a_p, sc_p = polish_local(best_a, pol_time1, rng)
            if sc_p < best_sc:
                best_sc = sc_p
                best_a = a_p.copy()
                print(f"[polish] Improved to {sc_p:.6f}")

            # Additional optimization rounds with different params
            if time.monotonic() < deadline - 10.0:
                remain2 = deadline - time.monotonic() - 5.0
                if remain2 > 10.0:
                    a_p2, sc_p2 = optimize_sequence(best_a, remain2, rng, n_scales=2)
                    if sc_p2 < best_sc:
                        best_sc = sc_p2
                        best_a = a_p2.copy()
                        print(f"[polish2] Improved to {sc_p2:.6f}")

    # Fallback
    if best_a is None:
        best_a = np.ones(512) / 512
        best_sc = 2.0

    # Final verification
    result = [float(x) for x in best_a]
    final_sc = float(evaluate_sequence(result))
    print(f"[final] n={len(result)}, internal={best_sc:.6f}, evaluator={final_sc:.6f}")

    return result
