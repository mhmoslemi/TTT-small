def adam_update(seq, conv, topk, m, v, t=None, learning_rate=0.005):

    eps = 1e-7
    
    m[:] = beta1 * m + (1.0 - beta1) * grad
    v[:] = beta2 * v + (1.0 - beta2) * (grad ** 2)
    
    if t == 0:
        # First iteration, use full gradient
        step = -learning_rate * m
    else:
        step = -learning_rate * (m / (1.0 - beta1 ** t)) / (np.sqrt(v / (1.0 - beta1 ** t)) + eps)
        
    new_seq = np.clip(seq + step, 0.0, 1000.0)
    return new_seq, m, v
