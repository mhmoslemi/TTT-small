SART (Semidefinite Alternating Refinement via Linearization) optimizer for
minimizing f(a) = 2n * max(conv(a,a)) / (sum a)^2 over non-negative sequences.

Uses LP-based alternating linearization: fix one side of the convolution, solve
an LP for the optimal update direction, blend, repeat.

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_cross_conv(a, b):
    n = len(a)
    m = len(b)
    M = n + m - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def objective_value(a):
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf

    return 2.0 * n * float(c.max()) / (S * S)


def build_lp_matrices(n, a):
    """Build the LP constraint matrix A where (A @ g)[k] = (a * g)[k]."""
    # Build sparse matrix
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            rows_list.extend(range(k, k + 1) * (j_end - j_start + 1))
            cols_list.extend(range(j_start, j_end + 1))
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    vals = a[rows - cols]
    A_mat = sparse.csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return A_mat


def solve_lp(a, M, timeout=60):
    """Solve: max sum(g) s.t. A(a) @ g <= M, g >= 0."""
    n = len(a)
    A_mat = build_lp_matrices(n, a)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    res = linprog(c_obj, A_ub=A_mat, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'time_limit': timeout})

        return res.x
    return None


def grad_based_step(a, steps=50, lr=0.01, topk=5):
    """Fallback gradient-based optimization step."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a
    a = a.copy()
    for _ in range(steps):
        c = fast_self_conv(a)
        # Top-k peaks
        k_act = min(topk, len(c))
        idx = np.argpartition(c, -k_act)[-k_act:]
        vals = c[idx]
        top = vals.max()
        if top > 1e-12:
            temp = 0.1 * top
            wts = np.exp((vals - top) / temp)
            wts /= wts.sum()
        else:
            wts = np.ones(k_act) / k_act
        # Gradient
        grad = np.zeros(n)
        for i in range(k_act):

            wi = float(wts[i])
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            seg = a[j_start:j_end + 1]
            grad[j_start:j_end + 1] += wi * 2.0 * seg[::-1]

        M_w = float(np.dot(vals, wts))

        a = a - lr * g

"""
SART (Semidefinite Alternating Refinement via Linearization) optimizer for
minimizing f(a) = 2n * max(conv(a,a)) / (sum a)^2 over non-negative sequences.

Key innovations over the previous gradient-based approach:
  1. LP-based alternating linearization: fix one side of convolution, solve LP
     for optimal update direction, blend with line search.
  2. Constraint-subset LP for speed (top-K peaks only).
  3. Hybrid: alternate LP steps with gradient refinement steps.
  4. Sparse matrix construction via vectorized numpy for fast LP setup.
  5. Adaptive restarts with structured perturbations.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse


def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def objective_value(a):
    n = len(a)
    S = a.sum()
    if S < 1e-15:
        return np.inf
    c = fast_self_conv(a)
    return 2.0 * n * float(np.max(c)) / (S * S)


def build_conv_matrix_sparse(n, a, k_indices=None):
    """Build the convolution constraint matrix (sparse).
    If k_indices is given, only build those rows."""
    if k_indices is None:
        k_indices = np.arange(2 * n - 1)
    rows = []
    cols = []
    vals = []
    for k in k_indices:
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            rows.append(np.full(len(js), k, dtype=np.int32))
            cols.append(js.astype(np.int32))
            vals.append(a[js.astype(np.int64) + (k - js).astype(np.int64)] if False else a[k - js])
    if rows:
        rows = np.concatenate(rows)
        cols = np.concatenate(cols)
        vals = np.concatenate(vals)
    else:
        rows = np.array([], dtype=np.int32)
        cols = np.array([], dtype=np.int32)
        vals = np.array([])
    A_mat = sparse.csr_matrix((vals, (rows, cols)), shape=(len(k_indices), n))
    return A_mat


def build_conv_matrix_fast(n, a, k_indices=None):
    """Build convolution matrix using vectorized approach for speed."""
    if k_indices is None:
        k_arr = np.arange(2 * n - 1)
    else:
        k_arr = np.asarray(k_indices)
    
    n_rows = len(k_arr)
    # For each row k, valid j range: max(0, k-n+1) to min(n-1, k)
    # Build using COO format
    row_list = []
    col_list = []
    val_list = []
    
    for i, k in enumerate(k_arr):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            row_list.append(np.full(len(js), i, dtype=np.int32))
            col_list.append(js.astype(np.int32))
            val_list.append(a[k - js])
    
    if row_list:
        rows = np.concatenate(row_list)
        cols = np.concatenate(col_list)
        vals = np.concatenate(val_list)
    else:
        rows = np.array([], dtype=np.int32)
        cols = np.array([], dtype=np.int32)
        vals = np.array([])
    
    A_mat = sparse.csr_matrix((vals, (rows, cols)), shape=(n_rows, n))
    return A_mat


def solve_sart_lp(a, M, k_indices=None, timeout=30):
    """Solve: max 1^T g s.t. K(a) @ g <= M, g >= 0.
    K(a) is the (subset of) convolution matrix rows for k_indices."""
    n = len(a)
    K = build_conv_matrix_fast(n, a, k_indices)
    n_constraints = K.shape[0]
    c_obj = -np.ones(n)
    b_ub = np.full(n_constraints, M)
    
    res = linprog(c_obj, A_ub=K, b_ub=b_ub,
                  bounds=[(0, None)] * n,
                  method='highs',
                  options={'time_limit': timeout})
    if res.success:
        return res.x
    return None


def grad_step(a, n_steps=30, lr=0.02, topk=8, temperature_scale=0.1):
    """Perform gradient-based steps with top-k soft-max smoothing."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return a
    a = a.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    for _ in range(n_steps):
        c = fast_self_conv(a)
        S = a.sum()
        k_act = min(topk, len(c))
        if k_act <= 0:
            break
        idx = np.argpartition(c, -k_act)[-k_act:]
        vals = c[idx]
        order = np.argsort(vals)[::-1]
        idx = idx[order]
        vals = vals[order]
        top = float(vals[0])
        if top > 1e-12:
            temp = temperature_scale * top
            wts = np.exp((vals - top) / temp)
            wts /= wts.sum()
        else:
            wts = np.ones(k_act) / k_act
        
        grad_c = np.zeros(n)
        for i in range(k_act):
            ki = int(idx[i])
            wi = float(wts[i])
            if wi < 1e-15:
                continue
            j_start = max(0, ki - n + 1)
            j_end = min(n - 1, ki)
            if j_start <= j_end:
                seg = a[j_start:j_end + 1]
                grad_c[j_start:j_end + 1] += wi * 2.0 * seg[::-1]
        
        M_w = float(np.dot(vals, wts))
        S2 = S * S
        S3 = S2 * S
        g = 2.0 * n * (grad_c / S2 - 2.0 * M_w / S3)
        
        t += 1
        t1 = b1 ** t
        t2 = b2 ** t
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        mh = m / (1.0 - t1)
        vh = v / (1.0 - t2)
        a = a - lr * mh / (np.sqrt(vh) + eps)
        a = np.clip(a, 0, 1000)
    
    S = a.sum()
    if S > 1e-10:
        a = a / S
    return a
