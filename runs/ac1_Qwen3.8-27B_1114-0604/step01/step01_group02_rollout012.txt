```python
Optimized search for non-negative sequences minimizing 2n*max(autocorr)/(sum)^2.
Uses log-parameterization, exact subgradients with peak tracking, adaptive step sizes,
and multi-scale exploration.
"""

import numpy as np
import time

def fast_self_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(n, c, S):
    """Compute the objective value."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def find_top_peaks(c, k=10, threshold=0.95):
    """Find the top k peaks and all peaks above threshold*max."""
    cmax = float(np.max(c))
    if cmax <= 0:
        return np.array([]), np.array([])
    # All peaks above threshold
    thresh = cmax * threshold
    above = np.where(c >= thresh)[0]
    # Also include top k
    if len(c) > k:
        topk = np.argpartition(c, -k)[-k:]
        combined = np.unique(np.concatenate([above, topk]))
    else:
        combined = above
    return combined, c[combined]

def subgrad_logspace(x, peaks_vals, n):
    """Compute subgradient in log-space.
    x: log of sequence (length n)
    peaks_vals: array of (index, value) for active peaks
    Returns gradient vector of length n.
    """
    a = np.exp(x)
    S = float(np.sum(a))
    M = float(np.max(peaks_vals))
    
    # Weight peaks by proximity to max (soft assignment)
    peak_indices, peak_values = peaks_vals
    # Use weights proportional to value (heavier weight on larger peaks)
    weights = peak_values / peak_values.sum() if peak_values.sum() > 0 else np.ones(len(peak_values))/len(peak_values)
    
    grad = np.zeros(n)
    for i, (ki, vi, wi) in enumerate(zip(peak_indices, peak_values, weights)):
        if wi <= 0:
            continue
        # d(a*a)[ki]/dx_m = a_m * a[ki-m] for valid m
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start > j_end:

        # For m in [j_start, j_end]: contribution is a[m] * a[ki-m]
        m_range = np.arange(j_start, j_end + 1)
        contributions = a[m_range] * a[ki - m_range]
        grad[j_start:j_end+1] += wi * contributions
    
    # score = 2n * M / S^2
    # d(score)/dx_m = (2n/S^2) * grad_M[m] - (4nM/S^3) * S' [m]
    # where S'[m] = a[m]
    grad_M = grad  # This is the weighted sum of d(a*a)[ki]/dx_m
    grad_score = (2.0 * n / (S * S)) * grad_M - (4.0 * n * M / (S ** 3)) * a
    
    return grad_score

        c = fast_self_conv(a)
        S = float(np.sum(a))
        score = compute_score(n, c, S)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_x = x.copy()
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= patience:
            # Restart from best with perturbation
            scale = 0.03 + 0.05 * rng.random()
            x = best_x + scale * rng.standard_normal(n)
            m[:] = 0
            v[:] = 0
            t = 0
            lr = max(0.01, lr * 0.7)
            no_improve = 0
            patience = max(50, patience // 2)

        # Keep the mean of x around 0

    - Backtracking line search
    - Multiple peak sets
    - Periodic re-normalization

            best_a = a.copy()

        if len(peak_idx) == 0:

        a_w = np.exp(x)
        S_w = float(np.sum(a_w))
        M = float(np.max(peak_val))

        weights = peak_val ** 2  # Heavier weight on larger peaks
        w_sum = weights.sum()
        if w_sum > 0:
            weights = weights / w_sum
        else:
            weights = np.ones(len(peak_val)) / len(peak_val)

        for i in range(len(peak_idx)):
            ki = int(peak_idx[i])
            wi = float(weights[i])
            if wi < 1e-12:

            j_start = max(0, ki - n + 1)
            j_end = min(ki, n - 1)
            if j_start > j_end:
                continue
            m_range = np.arange(j_start, j_end + 1)
            grad[j_start:j_end+1] += wi * 2.0 * a_w[m_range] * a_w[ki - m_range]

        grad_score = (2.0 * n / (S_w * S_w)) * grad - (4.0 * n * M / (S_w ** 3)) * a_w

        t += 1
        m[:] = b1 * m + (1 - b1) * grad_score
        v[:] = b2 * v + (1 - b2) * grad_score * grad_score
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        x_new = x - lr * mh / (np.sqrt(vh) + eps)
        
        # Check if new point is valid and better
        a_new = np.exp(x_new)
        a_new = np.clip(a_new, 0, 1000)

        if S_new < 0.01:
            x = x_new

        if np.isfinite(score_new) and score_new < score * 1.05:
            x = x_new
        else:
            # Reduce learning rate
            lr *= 0.9
            x = x_new  # still take the step but with smaller effective lr next time

            x -= np.mean(x)
            # Also try to keep values in a reasonable range

                x -= np.log(500) + np.log(np.mean(a_check[a_check > 0]) if np.any(a_check > 0) else 1)
```

Let me restructure to avoid closures. I'll use a class or just pass around state explicitly.

Let me also reconsider the algorithm. The soft-max/weighting approach for peaks is important. Let me think about what weights to use.

If we have peaks at indices $k_1, k_2, ..., k_m$ with values $v_1 \geq v_2 \geq ... \geq v_m$, then:
- The true gradient of $M = \max v_i$ is in the convex hull of the individual gradients $\nabla v_i$.
- A natural choice is to weight by $v_i / \sum v_j$ (proportional to peak height).
- Another choice is to use a soft-max: $w_i = \exp(v_i / \tau) / \sum \exp(v_j / \tau)$ for small $\tau$.

For our purposes, I think weighting by peak height (or peak height squared) is a good choice because it concentrates the gradient on the most important peaks.

Let me also think about whether to use Adam or a simpler method. Adam can be problematic when the function is not smooth (which ours isn't, due to the max). A simpler approach might be:
- Compute the subgradient
- Use a Polyak step size: $\alpha_t = (f(x_t) - f^*) / \|\nabla f(x_t)\|^2$
- We don't know $f^*$, but we can use the best value so far as a proxy.

Actually, for non-smooth optimization, a more robust approach is:
1. Compute subgradient $g$
2. Take step $x_{t+1} = x_t - \alpha_t g$
3. Choose $\alpha_t$ via backtracking or diminishing schedule

Let me also add a "coordinate perturbation" step: after each gradient step, randomly perturb a few coordinates and keep the perturbation if it improves the score. This can help escape local minima.

```python
"""
Log-parameterized subgradient optimizer for minimizing 2n*max(autocorr)/(sum)^2.
Uses exact peak identification, weighted subgradients, adaptive step sizes,
and multi-phase exploration with periodic restarts.
"""

import numpy as np
import time

# --- Top-level helpers (no closures) ---

def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def find_peaks(c, k_max, thresh_frac):
    """Return array of (peak_index, peak_value) for peaks above thresh_frac*max and top k_max."""
    cmax = float(np.max(c))
    if cmax <= 0:
        return np.array([[]], dtype=object), np.array([], dtype=float)
    thresh = cmax * thresh_frac
    above = np.where(c >= thresh)[0]
    if len(c) > k_max:
        topk = np.argpartition(c, -k_max)[-k_max:]
        combined = np.unique(np.concatenate([above, topk]))
    else:
        combined = above
    order = np.argsort(c[combined])[::-1]
    combined = combined[order]
    return combined, c[combined]

def compute_subgrad(a, peak_idx, peak_val, n):
    """Compute subgradient of score w.r.t. a (not log-space).
    score = 2n * M / S^2 where M = max_k (a*a)[k]
    Subgrad = (2n/S^2) * sum_i w_i * d(a*a)[k_i]/da - (4nM/S^3) * a
    """
    S = float(np.sum(a))
    M = float(np.max(peak_val))
    
    # Weights: proportional to peak value squared (concentrate on top peaks)
    weights = peak_val ** 2
    wsum = float(np.sum(weights))
    if wsum > 0:
        weights = weights / wsum
    else:
        weights = np.ones(len(peak_val)) / len(peak_val)
    
    # Compute weighted gradient of M
    grad_M = np.zeros(n)
    for i in range(len(peak_idx)):
        ki = int(peak_idx[i])
        wi = float(weights[i])
        if wi < 1e-15:
            continue
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start > j_end:
            continue
        # d(a*a)[ki]/da[j] = a[ki-j] + a[j-ki] if both in range, else one of them
        # Actually: (a*a)[ki] = sum_j a[j]*a[ki-j]
        # d/da[m] = a[ki-m] (if 0 <= ki-m < n) + a[m-ki] (if 0 <= m-ki < n)
        # But m-ki < 0 when m < ki, and ki-m < 0 when m > ki
        # For m in [j_start, j_end]: both m and ki-m are in [0, n-1]
        # So d(a*a)[ki]/da[m] = a[ki-m] + (a[m-ki] if m != ki-m else a[ki-m])
        # Wait: if m = ki-m, i.e., m = ki/2, then the term a[m]*a[ki-m] = a[m]^2
        # and d/da[m] = 2*a[m]. Otherwise d/da[m] = a[ki-m].
        # Actually let me be more careful:
        # (a*a)[ki] = sum_{j: 0<=j<n, 0<=ki-j<n} a[j] * a[ki-j]
        # For a specific m: the terms involving a[m] are:
        #   - a[m] * a[ki-m] (when j=m, requiring 0<=ki-m<n)
        #   - a[m'] where m' = ki-m, i.e., a[ki-m]*a[m] (when j=ki-m, requiring 0<=m<n)
        # So if m != ki-m: contribution = 2 * a[ki-m] (from both j=m and j=ki-m)
        # If m == ki-m (i.e., 2m = ki): contribution = 2 * a[m] (from j=m, since a[m]*a[m] has derivative 2a[m])
        # Wait no: if 2m = ki, then the term in the sum is a[m]*a[ki-m] = a[m]*a[m] = a[m]^2
        # d/da[m] of a[m]^2 = 2*a[m]. And this appears only once in the sum (at j=m).
        # If 2m != ki, then a[m] appears in two terms: j=m (giving a[m]*a[ki-m]) and j=ki-m (giving a[ki-m]*a[m])
        # So d/da[m] = a[ki-m] + a[ki-m] = 2*a[ki-m]
        
        # Simplification: for all m in valid range, d(a*a)[ki]/da[m] = 2*a[ki-m] if 2m != ki
        # and = 2*a[m] if 2m == ki (which is the same as 2*a[ki-m] since m=ki-m in that case)
        # So in all cases: d(a*a)[ki]/da[m] = 2 * a[ki-m]
        # Wait, that's only if both m and ki-m are in [0,n-1], which is guaranteed by j_start,j_end.
        # And when m = ki-m, we get 2*a[m] which is correct.
        
        m_range = np.arange(j_start, j_end + 1)
        grad_M[j_start:j_end+1] += wi * 2.0 * a[ki - m_range]
    
    # Full subgradient of score
    grad = (2.0 * n / (S * S)) * grad_M - (4.0 * n * M / (S ** 3)) * a
    return grad

def compute_subgrad_log(a, peak_idx, peak_val, n):
    """Subgradient in log-space: d(score)/dx_j where a_j = exp(x_j).
    d(a_j)/dx_j = a_j, so d(score)/dx_j = d(score)/da_j * a_j
    """
    grad_a = compute_subgrad(a, peak_idx, peak_val, n)
    return grad_a * a  # Chain rule: d/dx = d/da * da/dx = d/da * a

def adam_step(x, grad, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    m[:] = b1 * m + (1 - b1) * grad
    v[:] = b2 * v + (1 - b2) * grad * grad
    mh = m / (1 - b1**t)
    vh = v / (1 - b2**t)
    x -= lr * mh / (np.sqrt(vh) + eps)

def resize_sequence(a, new_n):
```

```python
"""
Log-parameterized subgradient optimizer minimizing 2n*max(autocorr)/(sum)^2.
Uses exact peak identification with power-weighted subgradients, Adam in log-space,
adaptive restarts, multi-length exploration, and coordinate descent polishing.
"""

import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, c, S):
    """Score = 2n * max(c) / S^2"""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def find_peaks(c, k_max, thresh_frac):
    """Return (indices, values) of peaks above threshold and top k_max."""
    cmax = float(np.max(c))
    if cmax <= 0:
        return np.array([], dtype=int), np.array([], dtype=float)
    thresh = cmax * thresh_frac
    above = np.where(c >= thresh)[0]
    if len(c) > k_max:
        topk = np.argpartition(c, -k_max)[-k_max:]
        combined = np.unique(np.concatenate([above, topk]))
    else:
        combined = above
    order = np.argsort(c[combined])[::-1]
    combined = combined[order]
    return combined, c[combined]


def compute_subgrad_log(a, peak_idx, peak_val, n):
    """Subgradient of score w.r.t. log(a) parameters.
    score = 2n*M/S^2, d(score)/dx_j = d(score)/da_j * a_j
    """
    S = float(np.sum(a))
    M = float(np.max(peak_val))
    # Power-weighted peaks (concentrate on largest)
    weights = peak_val ** 2
    wsum = float(np.sum(weights))
    if wsum > 0:
        weights = weights / wsum
    else:
        weights = np.ones(len(peak_val)) / len(peak_val)

    grad_M = np.zeros(n)
    for i in range(len(peak_idx)):
        ki = int(peak_idx[i])
        wi = float(weights[i])
        if wi < 1e-15:
            continue
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        # d(a*a)[ki]/da[m] = 2*a[ki-m] for m in [js, je]
        m_range = np.arange(js, je + 1)
        grad_M[js:je + 1] += wi * 2.0 * a[ki - m_range]

    # d(score)/da = (2n/S^2)*grad_M - (4nM/S^3)*1
    grad_a = (2.0 * n / (S * S)) * grad_M - (4.0 * n * M / (S ** 3)) * 1.0
    # Chain rule to log-space
    return grad_a * a


def adam_update(x, grad, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    """In-place Adam update."""
    m[:] = b1 * m + (1.0 - b1) * grad
    v[:] = b2 * v + (1.0 - b2) * grad * grad
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)
    x -= lr * mh / (np.sqrt(vh) + eps)


def build_initials(n, rng, parent=None):
    """Build diverse log-space initializations."""
    inits = []
    if parent is not None and len(parent) == n:
        x = np.log(np.clip(parent, 1e-10, 1000.0))
        x -= np.mean(x)
        inits.append(x)

    # Uniform
    inits.append(np.zeros(n))
    # Ramps
    xlin = np.linspace(-0.5, 0.5, n)
    for alpha in [0.3, 0.8, 1.5, 3.0]:
        inits.append(alpha * xlin)
    # Cosines
    for fq in [1, 2, 3, 5, 8, 13]:
        for amp in [0.4, 1.0, 2.0]:
            inits.append(amp * np.cos(2.0 * np.pi * fq * np.linspace(0, 1, n)))
    # Random
    for scale in [0.3, 0.8, 1.5, 2.5]:
        inits.append(scale * rng.standard_normal(n))
    # Bimodal
    for split in [0.25, 0.4, 0.5, 0.6, 0.75]:
        x = np.zeros(n)
        x[:int(split * n)] = 0.5
        x[int(split * n):] = -0.5
        inits.append(x)
    # Triangular
    mid = n // 2
    x = np.zeros(n)
    x[:mid] = np.linspace(1.0, 0.0, mid) if mid > 0 else np.zeros(0)
    x[mid:] = np.linspace(0.0, 1.0, n - mid) if n - mid > 0 else np.zeros(0)
    inits.append(0.5 * x)
    # Gapped
    for gap in [3, 5, 7, 11]:
        x = np.full(n, -0.5)
        x[::gap] = 0.5
        inits.append(x)
    return inits


def optimize_log(x0, time_limit, rng, init_lr=0.05, patience=200,
                 k_peaks=12, thresh=0.9, l2_reg=0.0):
    """Optimize in log-space. Returns (best_score, best_a)."""
    n = len(x0)
    x = x0.astype(np.float64).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = init_lr
    no_improve = 0
    cur_patience = patience
    best_score = np.inf
    best_a = None

    t_start = time.monotonic()

    while time.monotonic() - t_start < time_limit:
        a = np.exp(x)
        a = np.clip(a, 0.0, 1000.0)
        S = float(np.sum(a))
        if S < 0.01:
            a = np.ones(n) * (0.01 / n + 1e-6)
            x = np.log(a)
            continue

        c = fast_self_conv(a)
        score = compute_score(n, c, S)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= cur_patience:
            # Adaptive restart: increase perturbation with stall duration
            stall_ratio = float(no_improve) / cur_patience
            scale = 0.02 + 0.04 * stall_ratio * rng.random()
            if best_a is not None:
                x = np.log(np.clip(best_a, 1e-10, 1000.0))
            else:
                x = np.zeros(n)
            x += scale * rng.standard_normal(n)
            x -= np.mean(x)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            lr = max(0.002, lr * 0.75)
            no_improve = 0
            cur_patience = max(20, cur_patience * 3 // 4)
            continue

        peak_idx, peak_val = find_peaks(c, k_peaks, thresh)
        if len(peak_idx) == 0:
            break

        grad = compute_subgrad_log(a, peak_idx, peak_val, n)

        # Add small L2 regularization in log-space for stability
        if l2_reg > 0:
            grad += l2_reg * x

        t += 1
        adam_update(x, grad, m, v, t, lr)

        # Periodic normalization
        if t % 100 == 0:
            x -= np.mean(x)
            mx = np.max(x)
            if mx > 6.5:
                x -= (mx - 6.5) * 0.5
            mn = np.min(x)
            if mn < -8.0:
                x += (mn - (-8.0)) * 0.5

    return best_score, best_a


def coordinate_polish(a, n, time_limit, rng):
    """1D coordinate descent polishing in linear space."""
    a = a.astype(np.float64).copy()
    best_score = evaluate_internal(a, n)
    t_start = time.monotonic()

    while time.monotonic() - t_start < time_limit:
        improved = False
        idx_order = rng.permutation(n)
        for i in range(n):
            if time.monotonic() - t_start > time_limit:
                break
            j = int(idx_order[i])
            old_val = a[j]
            # Try a few candidate values
            candidates = [old_val * 0.5, old_val * 0.75, old_val * 1.25,
                          old_val * 1.5, 0.0, old_val + 0.1, old_val - 0.1]
            candidates = [max(0.0, min(1000.0, cv)) for cv in candidates]
            # Also try values based on the peak structure
            best_cand = old_val
            best_cand_score = best_score
            for cv in candidates:
                a[j] = cv
                sc = evaluate_internal(a, n)
                if sc < best_cand_score:
                    best_cand_score = sc
                    best_cand = cv
            if best_cand != old_val:
                a[j] = best_cand
                if best_cand_score < best_score:
                    best_score = best_cand_score
                    improved = True
            else:
                a[j] = old_val

        if not improved:
            break

    return best_score, a


def evaluate_internal(a, n):
    """Fast internal evaluation."""
    a = np.clip(a, 0.0, 1000.0)
    S = float(np.sum(a))
    if S < 0.01:
        return np.inf
    c = fast_self_conv(a)
    return compute_score(n, c, S)


def resize_sequence(a, new_n):
    """Resize via linear interpolation."""
    old_n = len(a)
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    result = np.interp(new_x, old_x, a)
    return np.clip(result, 0.0, 1000.0)
```
