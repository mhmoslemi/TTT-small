"""
Search algorithm:
- Work in log-space z = log(a), where a = softmax(z) is a probability vector.
- Minimize the peak value of the linear convolution a*a (equivalently the
  scoring function up to the constant normalization).
- Main drivers:
  1. A low-dimensional smooth manifold (Chebyshev / trigonometric / endpoint
     exponential basis) is optimized by L-BFGS-B from several starts.
  2. Full log-space L-BFGS-B polishing is performed from the best low-dim and
     structural starts, annealing the softmax approximation of max down to a
     sharp peak approximation.
  3. A final exact active-set style gradient polish is run with line search,
     small temperature escapes, and periodic reset to the incumbent best.
- The previous sequence height_sequence_1 is used when present, both as a
  fallback and as a starting profile (log-interpolated to the working length).
"""

import numpy as np
import time
from scipy.optimize import minimize

# Global working state.
_MS_N = 0
_MS_NFFT = 0
_MS_L = 0
_MS_TAU = 1e-5
_MS_BOUNDS = None

_LOW_B = None
_LOW_TAU = 1e-4
_LOW_BOUNDS = None

_FULL_BOUND = 35.0
_LOW_BOUND = 20.0

_POLY_DEG = 14
_TRIG_DEG = 28


def _configure_ms(n, poly_deg, trig_deg):
    global _MS_N, _MS_NFFT, _MS_L, _MS_TAU, _MS_BOUNDS, _POLY_DEG, _TRIG_DEG
    _MS_N = int(n)
    _MS_L = 2 * _MS_N - 1
    # N >= 2n - 1 is enough for linear convolution and for the auxiliary
    # correlation used in gradients.
    _MS_NFFT = 1 << ((2 * _MS_N - 1).bit_length())
    _MS_TAU = 1e-5
    _MS_BOUNDS = [(-_FULL_BOUND, _FULL_BOUND)] * _MS_N
    _POLY_DEG = int(poly_deg)
    _TRIG_DEG = int(trig_deg)


def _resample_log(seq, n):
    seq = np.asarray(seq, dtype=np.float64).ravel()
    if seq.size == 0:
        return np.zeros(n, dtype=np.float64)
    seq = np.nan_to_num(seq, nan=0.0, posinf=0.0, neginf=0.0)
    seq = np.clip(seq, 1e-300, 1e6)
    if float(np.max(seq)) <= 0.0:
        return np.zeros(n, dtype=np.float64)
    if seq.size == 1:
        return np.zeros(n, dtype=np.float64)

    z_old = np.log(seq)
    z_old = z_old - float(np.mean(z_old))

    if seq.size == n:
        z = z_old
    else:
        x_old = np.linspace(0.0, 1.0, seq.size, endpoint=True)
        x_new = np.linspace(0.0, 1.0, n, endpoint=True)
        z = np.interp(x_new, x_old, z_old)

    z = np.nan_to_num(z, nan=0.0, posinf=-_FULL_BOUND, neginf=-_FULL_BOUND)
    z = z - float(np.mean(z))
    return np.clip(z, -_FULL_BOUND, _FULL_BOUND)


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.array([], dtype=np.float64)
    z = np.nan_to_num(z, nan=0.0, posinf=_FULL_BOUND, neginf=-_FULL_BOUND)
    z = np.clip(z, -_FULL_BOUND, _FULL_BOUND)
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(z.size, 1.0 / z.size, dtype=np.float64)
    return e / s


def _conv_fft(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n <= 0:
        return np.array([], dtype=np.float64)
    if n == _MS_N:
        N = _MS_NFFT
        L = _MS_L
    else:
        N = 1 << ((2 * n - 1).bit_length())
        L = 2 * n - 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L].real
    c = np.maximum(c, 0.0)
    return c


def _fast_score_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 1e-12:
        return np.inf
    c = _conv_fft(a)
    if c.size == 0:
        return np.inf
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * a.size * m / (s * s))


def _score_seq_generic(seq):
    a = np.asarray(seq, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if not np.isfinite(s) or s < 1e-12:
        return np.inf
    n = a.size
    L = 2 * n - 1
    N = 1 << (L.bit_length())
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L].real
    c = np.maximum(c, 0.0)
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m / (s * s))


def _score_z(z):

def _crosscorr_w(a, w):
    n = _MS_N
    N = _MS_NFFT
    a = np.asarray(a, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    fa = np.fft.rfft(a, N)
    fw = np.fft.rfft(w, N)
    h = np.fft.irfft(fw * np.conjugate(fa), N)[:n].real
    return h


def _full_obj_grad(z):
    n = _MS_N
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n:
        return 1e12, np.zeros(max(1, z.size), dtype=np.float64)

    z = np.nan_to_num(z, nan=0.0, posinf=_FULL_BOUND, neginf=-_FULL_BOUND)
    z = np.clip(z, -_FULL_BOUND, _FULL_BOUND)
    z = z - float(np.mean(z))

    a = _softmax(z)
    c = _conv_fft(a)
    m = float(np.max(c))
    tau = float(_MS_TAU)
    if not np.isfinite(m) or tau <= 0.0:
        return 1e12, np.zeros(n, dtype=np.float64)

    diff = (c - m) / tau
    diff = np.clip(diff, -745.0, 0.0)
    e = np.exp(diff)
    se = float(np.sum(e))
    if se <= 0.0 or not np.isfinite(se):
        return 1e12, np.zeros(n, dtype=np.float64)

    w = e / se
    h = _crosscorr_w(a, w)
    s_h = float(np.dot(a, h))
    grad = 2.0 * a * (h - s_h)

    val = m + tau * float(np.log(se))
    scale = 2.0 * float(n)
    val *= scale
    grad *= scale

    pen = 1e-12
    val += 0.5 * pen * float(np.dot(z, z))
    grad = grad + pen * z

    if not np.isfinite(val):
        val = 1e12
        grad = np.zeros(n, dtype=np.float64)
    return float(val), grad


def _run_full_min(z0, tau, maxfun=400, maxiter=150):

            bounds=_MS_BOUNDS,

        )
        if res is not None and hasattr(res, "x"):

        pass
    return z0


def _build_basis(n):
    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    cols = []

    y = 2.0 * x - 1.0
    t_prev = np.ones(n, dtype=np.float64)
    t_curr = y.copy()
    for m in range(1, _POLY_DEG + 1):
        if m == 1:
            t = t_curr
        else:
            t_next = 2.0 * y * t_curr - t_prev
            t_prev, t_curr = t_curr, t_next
            t = t_curr
        t = t - float(np.mean(t))
        norm = float(np.linalg.norm(t))
        if norm > 1e-12:
            cols.append(t / norm)

    for m in range(1, _TRIG_DEG + 1):
        arg = 2.0 * np.pi * float(m) * x
        c = np.cos(arg)
        d = np.sin(arg)
        c = c - float(np.mean(c))
        d = d - float(np.mean(d))
        nc = float(np.linalg.norm(c))
        nd = float(np.linalg.norm(d))
        if nc > 1e-12:
            cols.append(c / nc)
        if nd > 1e-12:
            cols.append(d / nd)

    # Endpoint/asymmetry helpers, useful for the one-sided convolution problem.
    for sig in (0.03, 0.08, 0.18, 0.40):
        f = np.exp(-x / sig) - np.exp(-1.0 / sig)
        f = f - float(np.mean(f))
        nf = float(np.linalg.norm(f))
        if nf > 1e-12:
            cols.append(f / nf)

        f = np.exp(-(1.0 - x) / sig) - np.exp(-1.0 / sig)
        f = f - float(np.mean(f))
        nf = float(np.linalg.norm(f))
        if nf > 1e-12:
            cols.append(f / nf)

    if not cols:
        xc = x - 0.5
        cols.append(xc - float(np.mean(xc)))

    B = np.column_stack(cols)
    B = B - np.mean(B, axis=0, keepdims=True)

    # Orthonormalize for a better-conditioned low-dimensional parameterization.
    try:
        Q, R = np.linalg.qr(B, mode="reduced")
        diag = np.abs(np.diag(R))
        if diag.size > 0:
            keep = diag > (1e-8 * float(np.max(diag)))
            Q = Q[:, keep]
        B = Q
    except Exception:
        pass

    if B.size == 0 or B.shape[1] == 0:
        xc = x - 0.5
        xc = xc - float(np.mean(xc))
        B = xc.reshape(n, 1)

    return B

        if it % 250 == 0:
            print(
                "polish it={} current_score={:.9f} best={:.9f} time_left={:.0f}s".format(
                    it,
                    2.0 * n * M,
                    2.0 * n * best_M,
                    max(0.0, deadline - time.monotonic()),
                )
            )

    return best_z, best_M


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Returns the best non-negative coefficient list found within budget_s.
    """
    t0 = time.monotonic()
    budget_s = max(60.0, float(budget_s))
    deadline = t0 + budget_s - 8.0
    if deadline <= t0 + 20.0:
        deadline = t0 + max(20.0, budget_s)

    rng = np.random.default_rng(int(seed))

    # Preserve the previously found construction as a hard fallback.
    prev = globals().get("height_sequence_1", None)
    fallback_a = None
    fallback_score = np.inf
    if prev is not None:
        try:
            p = np.asarray(prev, dtype=np.float64).ravel()
            p = np.nan_to_num(p, nan=0.0, posinf=0.0, neginf=0.0)
            p = np.clip(p, 0.0, 1000.0)
            sp = float(np.sum(p))
            if p.size > 0 and np.isfinite(sp) and sp >= 0.01:
                pas = _score_seq_generic(p)
                if np.isfinite(pas):
                    fallback_a = p.copy()
                    fallback_score = float(pas)
        except Exception:
            pass

    # Choose working length. 16k is a good balance between texture, FFT cost,
    # and the official direct convolution cost.
    if budget_s >= 1200.0:
        n = 16384
        poly_deg = 14
        trig_deg = 28
    elif budget_s >= 600.0:
        n = 8192
        poly_deg = 10
        trig_deg = 20
    else:
        n = 4096
        poly_deg = 8
        trig_deg = 14

    _configure_ms(n, poly_deg, trig_deg)

    B = _build_basis(n)
    global _LOW_B
    _LOW_B = B
    global _LOW_BOUNDS
    _LOW_BOUNDS = [(-_LOW_BOUND, _LOW_BOUND)] * B.shape[1]

    # Base log-profile from previous sequence, or zero if unavailable.
    if fallback_a is not None and fallback_a.size >= 2:
        z_base = _resample_log(fallback_a, n)
    else:
        z_base = np.zeros(n, dtype=np.float64)

    x = (np.arange(n, dtype=np.float64) + 0.5) / float(n)
    xc = x - 0.5
    xc2 = xc * xc
    xc2 -= float(np.mean(xc2))

    entries = []
    _add_z_entry(entries, z_base)
    _add_z_entry(entries, z_base[::-1])

    for s in (-3.0, -2.0, -1.0, -0.5, 0.5, 1.0, 2.0, 3.0):
        _add_z_entry(entries, z_base + s * xc)

    for s in (-2.0, -1.0, -0.5, 0.5, 1.0, 2.0):
        _add_z_entry(entries, z_base + s * xc2)

    for f in (1, 2, 3, 5):
        for amp in (0.5, 1.2, 2.0):
            _add_z_entry(entries, z_base + amp * np.sin(2.0 * np.pi * float(f) * x))

    # Random smooth starts in the low-dimensional basis.
    for _ in range(14):
        th = 0.7 * rng.standard_normal(B.shape[1])
        _add_z_entry(entries, B.dot(th))

    entries.sort()

    # Search incumbent.
    search_z = z_base.copy()
    search_a = _softmax(search_z)
    search_score = _fast_score_a(search_a)

    if entries:
        sc0 = entries[0][0]
        if sc0 < search_score:
            search_score = sc0
            search_z = entries[0][2].copy()
            search_a = _softmax(search_z)

    print(
        "start n={} fallback={:.9f} initial_search={:.9f}".format(
            n, fallback_score, search_score
        )
    )

    # Full candidate pool used to seed full-dimension L-BFGS.
    opt_entries = []
    for e in entries[:12]:
        opt_entries.append(e)

    # Low-dimensional optimization.
    theta_entries = []
    _add_theta_entry(theta_entries, _project_theta(z_base))
    _add_theta_entry(theta_entries, _project_theta(z_base[::-1]))
    _add_theta_entry(theta_entries, np.zeros(B.shape[1], dtype=np.float64))
    for _ in range(8):
        _add_theta_entry(theta_entries, 0.5 * rng.standard_normal(B.shape[1]))

    theta_entries.sort()
    k_theta = 5 if budget_s >= 1500.0 else 3

    low_tau_schedule = (2e-4, 1e-4, 5e-5, 2e-5, 1e-5, 4e-6, 1.5e-6)
    for ti in range(min(k_theta, len(theta_entries))):
        if time.monotonic() > deadline - 700.0:
            break
        th_cur = theta_entries[ti][2].copy()
        for tau in low_tau_schedule:
            if time.monotonic() > deadline - 500.0:
                break
            if tau > 5e-4:
                maxfun = 350
            elif tau > 2e-5:
                maxfun = 280
            else:
                maxfun = 220
            th_new = _run_low_min(th_cur, tau, maxfun, max(90, maxfun // 2))
            z_new = B.dot(th_new)
            sc_new = _score_z(z_new)
            if np.isfinite(sc_new):
                th_cur = th_new
                _add_z_entry(opt_entries, z_new)
                if sc_new < search_score - 1e-12:
                    search_score = sc_new
                    search_z = z_new.copy()
                    search_a = _softmax(search_z)
                    print(
                        "lowdim n={} theta_run={} tau={:.2e} score={:.9f}".format(
                            n, ti, tau, search_score
                        )
                    )

    # Full-dimension L-BFGS from promising starts.
    opt_entries.sort()
    k_full = 6 if budget_s >= 1500.0 else 4
    full_tau_schedule = (1e-4, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6)

    for fi in range(min(k_full, len(opt_entries))):
        if time.monotonic() > deadline - 500.0:
            break
        z_cur = opt_entries[fi][2].copy()
        for tau in full_tau_schedule:
            if time.monotonic() > deadline - 300.0:
                break
            if fi == 0:
                if tau > 1e-4:
                    maxfun = 700
                elif tau > 2e-5:
                    maxfun = 900
                else:
                    maxfun = 1200
            else:
                maxfun = 450
            z2 = _run_full_min(z_cur, tau, maxfun, max(120, maxfun // 3))
            z2 = np.nan_to_num(z2, nan=0.0, posinf=_FULL_BOUND, neginf=-_FULL_BOUND)
            z2 = np.clip(z2, -_FULL_BOUND, _FULL_BOUND)
            z2 = z2 - float(np.mean(z2))
            sc2 = _score_z(z2)
            if np.isfinite(sc2):
                z_cur = z2
                _add_z_entry(opt_entries, z_cur)
                if sc2 < search_score - 1e-12:
                    search_score = sc2
                    search_a = _softmax(search_z)
                    print(
                        "fulllbfgs n={} start={} tau={:.2e} score={:.9f}".format(
                            n, fi, tau, search_score
                        )
                    )

    # Final active-set polish.  Allocate the bulk of remaining time to the best
    # candidate, and smaller portions to the next two if available.
    opt_entries.sort()
    polish_list = []
    for e in opt_entries[:5]:
        polish_list.append(e[2])

    for pi, pz in enumerate(polish_list):
        if time.monotonic() > deadline - 20.0:
            break
        if pi == 0:
            pol_deadline = deadline - 2.0
        else:
            rem = max(0.0, deadline - time.monotonic())
            pol_deadline = time.monotonic() + max(10.0, rem * (0.22 if pi == 1 else 0.10))
        try:
            z_best, m_best = _polish_full(pz, pol_deadline)
            sc_best = 2.0 * float(n) * float(m_best)
            if np.isfinite(sc_best) and sc_best < search_score - 1e-12:
                search_score = sc_best
                search_z = z_best.copy()
                search_a = _softmax(search_z)
                print("polished n={} pi={} score={:.9f}".format(n, pi, search_score))
        except Exception:
            pass

    # Choose between our search result and the previous construction fallback.
    if np.isfinite(fallback_score) and (not np.isfinite(search_score) or fallback_score < search_score):
        final_a = fallback_a
        final_score = fallback_score
        final_len = final_a.size if final_a is not None else n
    else:
        final_a = search_a
        final_score = search_score
        final_len = n

    # Defensive normalization.
    final_a = np.asarray(final_a, dtype=np.float64).ravel()
    if final_a.size == 0:
        final_a = np.ones(final_len, dtype=np.float64)
    final_a = np.nan_to_num(final_a, nan=0.0, posinf=0.0, neginf=0.0)
    final_a = np.clip(final_a, 0.0, 1000.0)
    s = float(np.sum(final_a))
    if not np.isfinite(s) or s < 0.01:
        final_a = np.ones(final_len, dtype=np.float64) / float(final_len)
    else:
        final_a = final_a / s

    print("final chosen score={:.9f} length={}".format(final_score, final_a.size))
    return [float(v) for v in final_a.tolist()]
