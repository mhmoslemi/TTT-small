    w = (ratio ** (q - 1)) / len(c)
    w_sum = float(w.sum())
    if w_sum > 0 and np.isfinite(w_sum):
        w = w / w_sum
    smooth = float(np.dot(w, c))
    h = _correlation_valid(w, p)
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.maximum(h, 0.0)
    return smooth, h, c


def _gradient_info(p, mode, param):
    """Unified interface: mode=0 for softmax (param=beta), mode=1 for power mean (param=exponent)."""
    if mode == 0:
        return _smoothed_max_and_grad(p, param)
    else:
        return _power_mean_and_grad(p, param)


def _run_optimize(p0, rng, time_limit, mode_schedule=None, beta_start=20.0, beta_end=2000.0,
                  exponent_start=4.0, exponent_end=60.0, verbose=False, tag=""):
    """Adaptive multiplicative descent with backtracking."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    if time_limit <= 0:
        return p, _exact_score(p)

    n = int(p.size)
    floor = 1e-18 / float(n)
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_exact = _exact_score(p)
    cur_exact = best_exact

    if mode_schedule is None:
        mode_schedule = [(0, beta_start, beta_end), (1, exponent_start, exponent_end)]

    alpha = 0.3
    deadline = time.monotonic() + float(time_limit)
    stall = 0
    it = 0
    last_print = -10000

    # Determine starting mode and parameter
    cur_mode = mode_schedule[0][0]
    cur_param = float(mode_schedule[0][1])
    param_end = float(mode_schedule[0][2])
    phase = 0

    while time.monotonic() < deadline:
        # Switch phases periodically
        if it > 0 and it % 500 == 0 and phase < len(mode_schedule) - 1:
            phase += 1
            cur_mode = mode_schedule[phase][0]
            cur_param = float(mode_schedule[phase][1])
            param_end = float(mode_schedule[phase][2])
            alpha = 0.3

        smooth, h, c = _gradient_info(p, cur_mode, cur_param)
        exact = _exact_score(p)

        if not np.isfinite(h.any()) or np.max(h) < 1e-15:
            p_try = _sanitize(p * np.exp(rng.normal(0.0, 0.05, n)))
            if p_try is not None:
                p_try = np.maximum(p_try, floor)
                p_try = p_try / p_try.sum()
                e = _exact_score(p_try)
                if np.isfinite(e) and e < best_exact * 1.1:
                    p = p_try
                    cur_exact = e
            stall += 1
        else:
            h_max = float(np.max(h))
            h_norm = h / h_max

            # Backtracking line search
            best_p_try = None
            best_smooth = smooth
            best_exact_trial = exact
            a = alpha * 2.0

            for _trial in range(8):
                if a < 1e-5:
                    break
                dec = np.exp(-a * h_norm)
                p_try = p * dec
                p_try = np.maximum(p_try, floor)
                s = float(p_try.sum())
                if s <= 0 or not np.isfinite(s):
                    a *= 0.5
                    continue
                p_try = p_try / s

                e = _exact_score(p_try)
                if np.isfinite(e) and e < best_exact_trial * (1.0 - 1e-8):
                    best_p_try = p_try
                    best_smooth = smooth
                    best_exact_trial = e
                    # Try one more step to see if we can go further
                    a *= 1.5
                    continue
                else:
                    a *= 0.5
                    continue

            if best_p_try is not None:
                p = best_p_try
                cur_exact = best_exact_trial
                stall = 0

                if cur_exact < best_exact:
                    best = p.copy()
                    best_exact = cur_exact
                    if verbose and (it - last_print >= 200):
                        elapsed = time.monotonic() - (deadline - time_limit)
                        print(f"  {tag} n={n} iter={it} best={best_exact:.6f} t={elapsed:.0f}s")
                        last_print = it
            else:
                stall += 1
                alpha = max(1e-5, alpha * 0.7)

                if stall >= 15:
                    std = 0.05 if stall < 50 else min(0.5, 0.05 + 0.01 * stall)
                    p_try = _sanitize(p * np.exp(rng.normal(0.0, std, n)))
                    if p_try is not None:
                        p_try = np.maximum(p_try, floor)
                        p_try = p_try / p_try.sum()
                        e = _exact_score(p_try)
                        if np.isfinite(e) and e < best_exact * 1.1:
                            p = p_try
                            cur_exact = e
                        else:
                            p = best.copy()
                            cur_exact = best_exact
                    stall = 0
                    alpha = 0.3

                if stall > 100:
                    p = best.copy()
                    cur_exact = best_exact
                    alpha = 0.3
                    cur_param = float(mode_schedule[phase][1])
                    stall = 0

        it += 1
        if it % 30 == 0:
            cur_param = min(param_end, cur_param * 1.12)

    return best, best_exact


def _peak_deflation(p, rng, time_limit, tag=""):
    """Explicitly target and reduce the top convolution peaks."""
    p = _sanitize(p)
    if p is None or time_limit <= 0:
        return None, float('inf')

    n = p.size
    floor = 1e-18 / float(n)
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_score = _exact_score(p)
    deadline = time.monotonic() + float(time_limit)

    while time.monotonic() < deadline:
        c = _fft_convolve(p, p)
        c = np.where(np.isfinite(c), c, 0.0)
        cmax = float(np.max(c))
        if cmax < 1e-15:
            break

        # Find top few peak positions
        thresh = cmax * 0.95
        peak_positions = np.where(c >= thresh)[0]

        # For each peak, identify contributing indices and reduce them
        improved = False
        for ppos in peak_positions[:5]:
            # c[ppos] = sum_i a[i] * a[ppos - i] for valid i
            # Contributing pairs: i such that 0<=i<n and 0<=ppos-i<n
            i_min = max(0, ppos - n + 1)
            i_max = min(n - 1, ppos)
            if i_min > i_max:
                continue

            idxs = np.arange(i_min, i_max + 1)
            partners = ppos - idxs

            # Find the largest contributor pair
            contribs = p[idxs] * p[partners]
            if contribs.size == 0:
                continue

            # Try reducing the largest contributors
            top_k = min(3, len(contribs))
            top_idx = np.argsort(contribs)[-top_k:]

            for f in (0.7, 0.5, 0.3):
                cand = p.copy()
                for ti in top_idx:
                    cand[idxs[ti]] *= f
                cand = np.maximum(cand, floor)
                cand = cand / cand.sum()
                sc = _exact_score(cand)
                if sc < best_score - 1e-12:
                    best = cand.copy()
                    best_score = sc
                    p = cand.copy()
                    improved = True
                    break

        if not improved:
            # Try a broader reduction: reduce all values at peak-contributing positions
            all_idxs = set()
            for ppos in peak_positions[:8]:
                i_min = max(0, ppos - n + 1)
                i_max = min(n - 1, ppos)
                for i in range(i_min, i_max + 1):
                    if p[i] > 0.01 * np.max(p):
                        all_idxs.add(i)
            if all_idxs:
                cand = p.copy()
                arr_idxs = np.array(list(all_idxs))
                cand[arr_idxs] *= 0.8
                cand = np.maximum(cand, floor)
                cand = cand / cand.sum()
                sc = _exact_score(cand)
                if sc < best_score - 1e-12:
                    best = cand.copy()
                    best_score = sc
                    p = cand.copy()
                    improved = True

        if not improved:
            # Random perturbation to escape
            p_try = _sanitize(p * np.exp(rng.normal(0.0, 0.03, n)))
            if p_try is not None:
                p_try = np.maximum(p_try, floor)
                p_try = p_try / p_try.sum()
                sc = _exact_score(p_try)
                if sc < best_score * 1.05:
                    p = p_try
                else:
                    p = best.copy()
            break

    return best, best_score


def _coordinate_refine(p, rng, time_limit, tag=""):
    """Gradient-directed coordinate refinement."""
    p = _sanitize(p)
    if p is None or time_limit <= 0:
        return None, float('inf')

    n = p.size
    floor = 1e-18 / float(n)
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_score = _exact_score(p)
    deadline = time.monotonic() + float(time_limit)
    stagnation = 0

    while time.monotonic() < deadline:
        # Get gradient direction
        c = _fft_convolve(p, p)
        c = np.where(np.isfinite(c), c, 0.0)
        c = np.maximum(c, 1e-30)
        # Use moderate power mean for guidance
        q = 10.0
        cq = c ** q
        pm = (float(np.sum(cq)) / len(c)) ** (1.0 / q)
        ratio = c / pm
        w = ratio ** (q - 1)
        w = w / w.sum()

        h = _correlation_valid(w, p)
        h = np.where(np.isfinite(h), h, 0.0)
        h = np.maximum(h, 0.0)

        if np.max(h) < 1e-15:
            break

        # Pick coordinates to modify based on gradient
        h_rank = np.argsort(h)[::-1]

        # Try modifying a random set of top-gradient coordinates
        n_mod = max(1, n // 50)
        candidates = h_rank[:n * 2 // 10]
        mod_idx = rng.choice(candidates, size=min(n_mod, len(candidates)), replace=False)

        improved = False
        for j in mod_idx:
            if time.monotonic() >= deadline:
                break
            # Try different multipliers for coordinate j
            base_val = p[j]
            for mult in (0.4, 0.6, 0.8, 1.2, 1.5, 2.0):
                cand = p.copy()
                cand[j] = base_val * mult
                cand = np.maximum(cand, floor)
                cand = cand / cand.sum()
                sc = _exact_score(cand)
                if sc < best_score - 1e-12:
                    best = cand.copy()
                    best_score = sc
                    p = cand.copy()
                    improved = True
                    break

        if not improved:
            stagnation += 1
            if stagnation > 40:
                p_try = _sanitize(p * np.exp(rng.normal(0.0, 0.04, n)))
                if p_try is not None:
                    p_try = np.maximum(p_try, floor)
                    p_try = p_try / p_try.sum()
                    sc = _exact_score(p_try)
                    if sc < best_score * 1.05:
                        p = p_try
                    else:
                        p = best.copy()
                        best_score = _exact_score(p)
                stagnation = 0
        else:
            stagnation = 0

    return best, best_score


def _initial_beta_pmf(m, rng, a, b):
    """Beta distribution initialization."""
    t = np.linspace(0.01, 0.99, m)
    p = np.power(t, a) * np.power(1.0 - t, b)
    p = p + 1e-12
    return p / p.sum()


def _initial_two_block(m, rng):
    """Two-block initialization."""
    ratio = float(rng.uniform(0.3, 2.0))
    split = int(m * float(rng.uniform(0.3, 0.7)))
    p = np.ones(m, dtype=float)
    p[split:] = ratio
    return p / p.sum()


def _initial_gaussian_bumps(m, rng, K):
    """Gaussian bump initialization."""
    t = (np.arange(m, dtype=float) + 0.5) / float(m)
    p = np.zeros(m, dtype=float)
    centers = np.sort(rng.uniform(0.05, 0.95, K))
    sigmas = rng.uniform(0.03, 0.3, K)
    weights = rng.uniform(0.5, 2.0, K)
    for i in range(K):
        p += weights[i] * np.exp(-((t - centers[i]) / sigmas[i]) ** 2)
    p = p + 1e-12
    return p / p.sum()


def _initial_cosine_mix(m, rng):
    """Cosine series initialization."""
    t = (np.arange(m, dtype=float) + 0.5) / float(m)
    z = np.ones(m, dtype=float)
    for k in range(1, 15):
        amp = float(rng.normal(0, 1.0 / (k + 1)))
        ph = float(rng.uniform(0, 2 * np.pi))
        z += amp * np.cos(2 * np.pi * k * t + ph)
    p = np.log1p(np.exp(np.clip(z, -30, 30)))
    p = p + 1e-12
    return p / p.sum()


def _resample(arr, m):
    """Resample normalized vector to length m via linear interpolation."""
    if m <= 0:
        return None
    a = _sanitize(arr)
    if a is None:
        return None
    n = a.size
    if n == m:
        return a.copy()
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0) + 1e-12
    return _sanitize(out)


def _smooth_array(p, samples=1.0):
    """Gaussian smoothing preserving sum."""
    p = _sanitize(p)
    if p is None or samples <= 0 or p.size < 5:
        return p
    rad = int(max(1, np.ceil(3.0 * samples)))
    k = np.arange(-rad, rad + 1, dtype=float)
    kern = np.exp(-0.5 * (k / samples) ** 2)
    kern /= kern.sum()
    out = np.convolve(p, kern, mode="same")
    return _sanitize(out)


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing the evaluation function.
    Returns a list of non-negative floats.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60.0:
        budget = 60.0

    start_time = time.monotonic()
    deadline = start_time + budget - 20.0  # leave 20s safety margin

    try:
        iseed = int(seed)
    except Exception:
        iseed = 12345
    rng = np.random.default_rng((iseed % 1000000007) + 1)

    # Try to load parent
    parent_arr = None
    try:
        parent_raw = height_sequence_1
        parent_arr = _sanitize(np.asarray(parent_raw, dtype=float))
    except Exception:
        parent_arr = None

    best_score = float('inf')
    best_arr = None

    print(f"Starting search, budget={budget:.0f}s, parent={'yes' if parent_arr is not None else 'no'}")

    # === Phase 1: Quick exploration at moderate size ===
    small_n = 2048
    print(f"Phase 1: Exploration at n={small_n}")

    starts = []
    if parent_arr is not None:
        sp = _resample(parent_arr, small_n)
        if sp is not None:
            sp = _smooth_array(sp, 0.5)
            if sp is not None:
                starts.append(sp)

    starts.append(_initial_two_block(small_n, rng))
    starts.append(_initial_beta_pmf(small_n, rng, 0.3, 0.7))
    starts.append(_initial_beta_pmf(small_n, rng, 1.5, 0.5))
    starts.append(_initial_gaussian_bumps(small_n, rng, 2))
    starts.append(_initial_gaussian_bumps(small_n, rng, 4))
    starts.append(_initial_cosine_mix(small_n, rng))

    # Score and sort starts
    scored = []
    for sp in starts:
        if sp is None:
            continue
        sc = _exact_score(sp)
        if np.isfinite(sc):
            scored.append((sc, sp))
    scored.sort(key=lambda x: x[0])

    # Run quick optimization on top 4 starts
    phase1_time = min(300.0, 0.2 * (deadline - start_time))
    if phase1_time > 60:
        per_start = phase1_time / 4.0
        for i in range(min(4, len(scored))):
            rem = deadline - time.monotonic()
            if rem < 30:
                break
            t_alloc = min(per_start, rem - 10)
            p, sc = _run_optimize(scored[i][1], rng, t_alloc,
                                  mode_schedule=[(0, 20, 300), (1, 4, 30)],
                                  beta_start=20, beta_end=300,
                                  verbose=False, tag=f"P1-{i}")
            if p is not None and sc < best_score:
                best_score = sc
                best_arr = p.copy()
                print(f"  Phase1 start {i}: score={sc:.6f}")

    print(f"  After Phase 1: best={best_score:.6f}, elapsed={time.monotonic()-start_time:.0f}s")

    # === Phase 2: Main optimization at larger size ===
    main_n = 4096
    if deadline - time.monotonic() > 1200:
        main_n = 6144
    if deadline - time.monotonic() > 1800:
        main_n = 8192

    print(f"Phase 2: Main optimization at n={main_n}")

    # Prepare main starts
    main_starts = []

    # Resample best from phase 1
    if best_arr is not None:
        sp = _resample(best_arr, main_n)
        if sp is not None:
            sp = _smooth_array(sp, 1.0)
            if sp is not None:
                main_starts.append(sp)

    # Resample parent
    if parent_arr is not None:
        sp = _resample(parent_arr, main_n)
        if sp is not None:
            sp = _smooth_array(sp, 1.0)
            if sp is not None:
                main_starts.append(sp)

    # Fresh two-block
    main_starts.append(_smooth_array(_initial_two_block(main_n, rng), 2.0))

    # Beta with different params
    main_starts.append(_initial_beta_pmf(main_n, rng, 0.2, 0.8))
    main_starts.append(_initial_beta_pmf(main_n, rng, 0.5, 0.3))

    # Score and select top 3
    main_scored = []
    for sp in main_starts:
        if sp is None:
            continue
        sc = _exact_score(sp)
        if np.isfinite(sc):
            main_scored.append((sc, sp))
    main_scored.sort(key=lambda x: x[0])

    # Run main optimization: best start gets most time
    rem = deadline - time.monotonic()
    if rem > 100:
        n_starts = min(3, len(main_scored))
        fracs = [0.55, 0.25, 0.20]

        for i in range(n_starts):
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            t_alloc = rem * fracs[i]
            t_alloc = max(30.0, t_alloc)

            mode_sched = [(0, 20, 500), (1, 4, 50), (0, 500, 3000)]

            p, sc = _run_optimize(main_scored[i][1], rng, t_alloc,
                                  mode_schedule=mode_sched,
                                  beta_start=20, beta_end=500,
                                  verbose=True, tag=f"P2-{i}")
            if p is not None and sc < best_score:
                best_score = sc
                best_arr = p.copy()
                print(f"  Phase2 start {i}: score={sc:.6f}")

    print(f"  After Phase 2: best={best_score:.6f}, elapsed={time.monotonic()-start_time:.0f}s")

    # === Phase 3: Peak deflation ===
    rem = deadline - time.monotonic()
    if rem > 100 and best_arr is not None:
        t_alloc = min(400.0, 0.25 * rem)
        print(f"Phase 3: Peak deflation, t={t_alloc:.0f}s")
        p, sc = _peak_deflation(best_arr, rng, t_alloc, tag="P3")
        if p is not None and sc < best_score:
            best_score = sc
            best_arr = p.copy()
            print(f"  After deflation: score={sc:.6f}")

    # === Phase 4: Coordinate refinement ===
    rem = deadline - time.monotonic()
    if rem > 80 and best_arr is not None:
        t_alloc = min(300.0, 0.2 * rem)
        print(f"Phase 4: Coordinate refinement, t={t_alloc:.0f}s")
        p, sc = _coordinate_refine(best_arr, rng, t_alloc, tag="P4")
        if p is not None and sc < best_score:
            best_score = sc
            best_arr = p.copy()
            print(f"  After refinement: score={sc:.6f}")

    # === Phase 5: Final high-beta polish ===
    rem = deadline - time.monotonic()
    if rem > 60 and best_arr is not None:
        t_alloc = min(300.0, 0.4 * rem)
        print(f"Phase 5: Final polish, t={t_alloc:.0f}s")
        mode_sched = [(0, 200, 5000), (1, 10, 80)]
        p, sc = _run_optimize(best_arr, rng, t_alloc,
                              mode_schedule=mode_sched,
                              beta_start=200, beta_end=5000,
                              verbose=True, tag="P5")
        if p is not None and sc < best_score:
            best_score = sc
            best_arr = p.copy()
            print(f"  After final polish: score={sc:.6f}")

    # === Phase 6: One more peak deflation ===
    rem = deadline - time.monotonic()
    if rem > 50 and best_arr is not None:
        t_alloc = min(100.0, 0.5 * rem)
        p, sc = _peak_deflation(best_arr, rng, t_alloc, tag="P6")
        if p is not None and sc < best_score:
            best_score = sc
            best_arr = p.copy()
            print(f"  After final deflation: score={sc:.6f}")

    # Final validation
    if best_arr is None:
        best_arr = _initial_beta_pmf(2048, rng, 0.3, 0.7)

    final = _sanitize(best_arr)
    if final is None:
        final = np.ones(2048, dtype=float)

    out = np.asarray(final, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s <= 0 or not np.isfinite(s):
        out = np.ones(2048, dtype=float)
    else:
        out = out / s

    print(f"Final: n={out.size}, score={_exact_score(out):.6f}, "
          f"elapsed={time.monotonic()-start_time:.0f}s")

    return [float(x) for x in out]
