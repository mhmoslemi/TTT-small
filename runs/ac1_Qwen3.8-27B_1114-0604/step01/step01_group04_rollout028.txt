```python
"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using cyclic coordinate descent
with incremental convolution updates and simulated annealing.
"""

import numpy as np
import time

def full_conv(a):
    n = len(a)
    return np.convolve(a, a)

def get_score(n, conv, sum_a):
    return 2.0 * n * float(np.max(conv)) / (sum_a * sum_a)

def coord_descent(a_init, time_budget, rng, verbose=True):
    """
    Cyclic coordinate descent with line search and annealing.
    """
    n = len(a_init)
    a = a_init.astype(np.float64).copy()
    conv = full_conv(a).astype(np.float64)
    sum_a = float(np.sum(a))
    
    best_a = a.copy()
    best_sc = get_score(n, conv, sum_a)
    
    t0 = time.monotonic()
    cycle = 0
    stall_count = 0
    last_improve = 0
    
    # Adaptive temperature
    base_temp = 0.05 / n
    cool_rate = 0.99999
    
    while time.monotonic() - t0 < time_budget:
        cycle += 1
        cycle_improved = False
        
        order = rng.permutation(n)
        
        for i in order:
            t = a[i]
            # Candidate multipliers
            if t > 1e-10:
                multipliers = [0.0, 0.3, 0.6, 0.8, 1.0, 1.2, 1.5, 2.0, 4.0]
            else:
                multipliers = [0.0, 1.0]
                base_val = 1.0 / (4 * n)
                candidates = [base_val * m for m in [0.25, 0.5, 1, 2, 4, 8]]
            
            if t > 1e-10:
                candidates = [t * m for m in multipliers]
            else:
                candidates = [base_val * m for m in [0.25, 0.5, 1, 2, 4, 8]]
            
            cur_sc = get_score(n, conv, sum_a)
            best_t = t
            best_local = cur_sc
            
            for tc in candidates:
                if abs(tc - t) < 1e-14:
                    continue
                delta = tc - t
                new_sum = sum_a + delta
                if new_sum < 1e-6:
                    continue
                
                # Incremental conv update
                k_min = max(0, i)
                k_max = min(2*n - 2, i + n - 1)
                
                # Save and restore (for trying multiple candidates)
                saved = conv[k_min:k_max+1].copy()
                
                for k in range(k_min, k_max + 1):
                    ki = k - i
                    if k == 2 * i:
                        conv[k] += delta * (2.0 * t + delta)
                    else:
                        conv[k] += 2.0 * delta * a[ki]
                
                sc = get_score(n, conv, new_sum)
                
                # Restore
                conv[k_min:k_max+1] = saved
                
                if sc < best_local:
                    best_local = sc
                    best_t = tc
            
            # Apply best
            if best_t != t:
                delta = best_t - t
                k_min = max(0, i)
                k_max = min(2*n - 2, i + n - 1)
                for k in range(k_min, k_max + 1):
                    ki = k - i
                    if k == 2 * i:
                        conv[k] += delta * (2.0 * t + delta)
                    else:
                        conv[k] += 2.0 * delta * a[ki]
                a[i] = best_t
                sum_a += delta
                cycle_improved = True
        
        if cycle_improved:
            # Check if overall best improved
            cur_sc = get_score(n, conv, sum_a)
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a.copy()
                last_improve = cycle
                stall_count = 0
            else:
                stall_count += 1
        else:
            stall_count += 1
        
        # Annealing: periodically perturb
        if stall_count > 50 and (stall_count % 20 == 0):
            # Add random perturbation
            scale = base_temp * rng.random()
            pert = rng.standard_normal(n) * scale
            a_new = np.maximum(a + pert, 0.0)
            # Keep sum roughly the same
            if a_new.sum() > 1e-6:
                a_new = a_new * (sum_a / a_new.sum())
                conv_new = full_conv(a_new)
                sum_new = float(np.sum(a_new))
                sc_new = get_score(n, conv_new, sum_new)
                if sc_new < best_sc * 1.1:  # Accept if not too much worse
                    a = a_new
                    conv = conv_new
                    sum_a = sum_new
            stall_count = 0
        
        # Periodic full recompute to avoid numerical drift
        if cycle % 500 == 0:
            conv = full_conv(a)
            sum_a = float(np.sum(a))
    
    return best_a, best_sc
