import time
import scipy.optimize

def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(a, b):
    la = len(a)
    lb = len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)

def _sanitize_vec(p):
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return p / s

def _exact_score(p):
    p = _sanitize_vec(p)
    if p is None:
        return float("inf")
    m = p.size
    c = _conv_full(p, p)
    if c.size == 0:
        return float("inf")
    c = np.where(np.isfinite(c), c, 0.0)
    return float(2.0 * m * np.max(c))

def _log_param_score(log_params, n, beta):
    """
    Score function for L-BFGS-B optimization in log-parameter space.
    log_params are the log of the sequence values (plus offset).
    """
    a = np.exp(log_params)
    a = a / a.sum()
    c = _conv_full(a, a)
    c = np.where(np.isfinite(c), c, 0.0)
    d = n * c
    # Smooth max
    shift = np.max(d)
    e = np.exp(beta * (d - shift))
    smooth_max = shift + np.log(e.sum()) / beta
    return float(2.0 * n * smooth_max)

# --- Utility functions ---

def _np2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv(a, b):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    if la * lb <= 3_000_000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _np2(la + lb - 1)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:la+lb-1].astype(float, copy=False)

def _san(p):
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 0 or not np.isfinite(s):
        return None
    return p / s

def _score(p):
    p = _san(p)
    if p is None:
        return float('inf')
    m = p.size
    c = _conv(p, p)
    return float(2.0 * m * np.max(c))

def _smooth_obj(p, beta):
    m = p.size
    c = _conv(p, p)
    d = m * c
    sm = np.max(d)
    e = np.exp(beta * (d - sm))
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return float(2.0 * m * sm), d
    return float(2.0 * m * (sm + np.log(s) / beta)), d

def _weights(d, beta):
    sm = np.max(d)
    e = np.exp(beta * (d - sm))
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full(d.size, 1.0 / d.size)
    return e / s

def _grad_dir(p, w):
    """
    Compute the direction h[i] = sum_k w[k] * (p[k-i] + p[i-k])
    which is the gradient of <w, b> with respect to p.
    Uses FFT for efficiency.

    B = np.fft.irfft(Fw * Fp, L)
