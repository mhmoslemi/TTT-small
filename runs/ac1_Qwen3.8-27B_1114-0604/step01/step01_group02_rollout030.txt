def lse_score_grad(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = a.sum()
    if S < 1e-10:
        return 1e10, np.zeros(n)
    c_max = c.max()
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)

    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

L-BFGS-B optimizer with LSE-smoothed max-convolution objective.
Minimizes 2n * max(conv(a,a)) / (sum(a))^2 over non-negative sequences.

Strategy:

  - Multiple structured & random initializations per length
  - Continuation in LSE temperature: beta = 30 -> 100 -> 300 -> 800
  - Local search via targeted random perturbations
  - Projected subgradient polish on the exact max
  - All helper functions top-level, no closures/lambda

import numpy as np
from scipy.optimize import minimize
import time


# ---------------------------------------------------------------------------
# Core numerical helpers
# ---------------------------------------------------------------------------

def fast_conv(a):
    """Linear self-convolution via real FFT. Returns array of length 2n-1."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(a):
    """Exact score: 2n * max(conv) / (sum)^2."""
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_score_grad(a, beta):
    """
    LSE-smoothed score and its gradient.
    LSE(c, beta) = c_max + (1/beta)*log(sum(exp(beta*(c-c_max))))
    Returns (score, grad).
    """
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-12:
        return 1e10, np.zeros(n)
    c_max = float(c.max())
    scaled = beta * (c - c_max)
    exp_scaled = np.exp(scaled)
    log_sum = np.log(exp_scaled.sum())
    L = c_max + log_sum / beta
    score = 2.0 * n * L / (S * S)
    p = exp_scaled / exp_scaled.sum()
    # Gradient: g_j = 2 * sum_m a[m] * p[m+j]  (cross-correlation)
    # g = 2 * conv(a_rev, p)[n-1 : 2n-1]
    rev_a = a[::-1]
    M2 = 3 * n - 2
    M2_fft = 1 << (M2 - 1).bit_length()
    RA = np.fft.rfft(rev_a, M2_fft)
    P = np.fft.rfft(p, M2_fft)
    conv_res = np.fft.irfft(RA * P, M2_fft)[n - 1:2 * n - 1]
    g = 2.0 * conv_res
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (g / S2 - 2.0 * L / S3)
    return score, grad

    """Wrapper for scipy.minimize: returns (value, gradient)."""
    return lse_score_grad(a, beta)

def exact_subgrad_polish(a, total_iters=500):
    """
    Projected subgradient descent on the exact max-conv objective.
    Uses adaptive step size (Polyak-style when possible).
    """
    n = len(a)
    a = a.astype(np.float64).copy()
    best_a = a.copy()
    best_sc = compute_score(a)
    # Polyak step: need a lower bound on optimal value
    # Use 1.0 as a conservative lower bound (score >= 2n/(2n-1) > 1)
    sc_lb = 1.0
    for it in range(total_iters):
        c = fast_conv(a)
        S = float(a.sum())
        if S < 1e-12:
            break
        M = float(c.max())
        k_star = int(np.argmax(c))
        sc = 2.0 * n * M / (S * S)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        # Subgradient of M w.r.t. a
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        sg_M = np.zeros(n)
        if j_end >= j_start:
            seg = a[j_start:j_end + 1]
            sg_M[j_start:j_end + 1] = 2.0 * seg[::-1]
        # Full gradient of score = 2n*(sg_M/S^2 - 2*M/(S^3))
        S2 = S * S
        S3 = S2 * S
        grad = (2.0 * n) * (sg_M / S2 - 2.0 * M / S3)
        gn = float(np.linalg.norm(grad))
        if gn < 1e-15:
            break
        # Polyak step size
        step = (sc - sc_lb) / (gn * gn + 1e-30)
        step = min(step, 0.5)
        a -= step * grad
        a = np.clip(a, 0.0, 1000.0)
        # Decay step for stability
        step_decay = 1.0 / (1.0 + 0.01 * it)
        a -= step_decay * 0.001 * grad / gn
        a = np.clip(a, 0.0, 1000.0)
    return best_sc, best_a

    Perturb the elements that contribute most to the peak convolution lag,
    shifting mass to elements that don't.
    """
    n = len(a)
    c = fast_conv(a)
    k_star = int(np.argmax(c))
    # Find which a[j] contribute to c[k_star]
    j_start = max(0, k_star - n + 1)
    j_end = min(k_star, n - 1)
    # Contribution of a[j] to c[k_star] is a[k_star - j]
    contrib = np.zeros(n)
    if j_end >= j_start:
        contrib[j_start:j_end + 1] = a[j_start:j_end + 1][::-1]
    # Pick the top contributors and reduce them; redistribute to others
    top_count = max(3, n // 50)
    top_idx = np.argsort(contrib)[::-1][:top_count]
    bottom_idx = np.argsort(contrib)[:-top_count] if n > 2 * top_count else np.argsort(contrib)[:top_count]
    a_pert = a.copy()
    total_reduced = 0.0
    for j in top_idx:
        reduce_amt = a_pert[j] * strength
        a_pert[j] -= reduce_amt
        a_pert[j] = max(a_pert[j], 0.0)
        total_reduced += reduce_amt
    if len(bottom_idx) > 0 and total_reduced > 0:
        per = total_reduced / len(bottom_idx)
        for j in bottom_idx:
            a_pert[j] += per
    return np.clip(a_pert, 0.0, 1000.0)

# Main search function
# ---------------------------------------------------------------------------

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing the evaluation function.
    Returns the best sequence found within the time budget.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + max(30.0, float(budget_s)) - 12.0

    # Load parent solution if available
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size >= 64:
                parent = np.clip(p, 0.0, 1000.0)
    except Exception:
        pass

    best_score = np.inf
    best_a = np.ones(1024) * 0.5

    def report(sc, a, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            print(f"  [{label}] NEW BEST: {sc:.8f}  (n={len(a)}, t={time.monotonic()-t_start:.0f}s)")
            return True
        return False

    # Determine length schedule
    lengths = []
    if parent is not None:
        pa = min(max(parent.size, 512), 16384)
        lengths.append(pa)
    for n in [2048, 4096, 8192]:
        if n not in lengths:
            lengths.append(n)
    lengths = sorted(set(lengths))
    # Limit to at most 3 lengths for time management
    lengths = lengths[:3]

    print(f"Lengths to try: {lengths}")
    print(f"Parent available: {parent is not None}")

    # ==================================================================
    # Phase 1: Continuation L-BFGS from multiple initializations
    # ==================================================================
    # Allocate ~60% of time to this phase
    phase1_deadline = t_start + 0.60 * (float(budget_s) - 12.0)

    beta_schedule = [30, 80, 200, 500]

    for n in lengths:
        if time.monotonic() > phase1_deadline:
            break
        print(f"\n--- Phase 1: n = {n} ---")
        inits = build_inits(n, rng, parent)
        # Sort inits: put parent first, then structured ones
        # Try subset of inits if time is tight
        max_inits = max(4, len(inits))
        inits = inits[:max_inits]

        for i, a0 in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            t0 = time.monotonic()
            a_cur = a0.copy()
            for bi, beta in enumerate(beta_schedule):
                if time.monotonic() > phase1_deadline:
                    break
                # Adjust maxiter based on remaining time
                elapsed_sofar = time.monotonic() - t_start
                remaining = phase1_deadline - time.monotonic()
                # Estimate: each L-BFGS call ~2-10s for these sizes
                est_per_call = 5.0
                max_calls_remaining = max(1, int(remaining / est_per_call))
                if max_calls_remaining <= 0:
                    break
                sc, a_opt, res = run_lbfgs(a_cur, beta,
                                           maxiter=min(3000, 500 * max_calls_remaining))
                a_cur = a_opt
                elapsed = time.monotonic() - t0
                report(sc, a_opt, f"n={n}, init={i}, beta={beta}")

    print(f"\nAfter Phase 1: best = {best_score:.8f}")

    # ==================================================================
    # Phase 2: Refinement with higher beta and more iterations
    # ==================================================================
    phase2_deadline = t_start + 0.80 * (float(budget_s) - 12.0)
    print(f"\n--- Phase 2: Refinement ---")

    for beta in [500, 1000, 2000]:
        if time.monotonic() > phase2_deadline:
            break
        sc, a_opt, _ = run_lbfgs(best_a, beta, maxiter=10000, maxfun=200000)
        report(sc, a_opt, f"refine beta={beta}")

    # ==================================================================
    # Phase 3: Local search with perturbations
    # ==================================================================
    print(f"\n--- Phase 3: Local Search ---")
    n_best = len(best_a)

    n_local_trials = 30
    local_deadline = t_start + 0.93 * (float(budget_s) - 12.0)
    for trial in range(n_local_trials):
        if time.monotonic() > local_deadline:
            break
        # Different perturbation strategies
        strategy = trial % 4
        if strategy == 0:
            # Gaussian perturbation
            scale = 0.03 + 0.07 * rng.random()
            a_pert = best_a * (1.0 + scale * rng.standard_normal(n_best))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        elif strategy == 1:
            # Targeted perturbation (reduce peak contributors)
            a_pert = targeted_perturb(best_a, rng, strength=0.03 + 0.05 * rng.random())
        elif strategy == 2:
            # Sinusoidal perturbation
            freq = 1.0 + 5.0 * rng.random()
            phase = 2.0 * np.pi * rng.random()
            amp = 0.02 + 0.05 * rng.random()
            x = np.arange(n_best, dtype=np.float64) / n_best
            a_pert = best_a * (1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase))
            a_pert = np.clip(a_pert, 0.0, 1000.0)
        else:
            # Block swap
            split = n_best // 2 + int(rng.integers(-n_best // 4, n_best // 4 + 1))
            swap_fft_len = np.pi * rng.random()
            a_pert = best_a.copy()
            if 10 < split < n_best - 10:
                seg = a_pert[split - 5:split + 5] * swap_fft_len
                a_pert[split - 5:split + 5] = a_pert[split - 5:split + 5] * (2.0 - swap_fft_len)
            a_pert = np.clip(a_pert, 0.0, 1000.0)

        sc, a_opt, _ = run_lbfgs(a_pert, 500, maxiter=3000, maxfun=60000)
        report(sc, a_opt, f"local #{trial} (strat={strategy})")

    # ==================================================================
    # Phase 4: Exact subgradient polish
    # ==================================================================
    print(f"\n--- Phase 4: Subgradient Polish ---")
    sc, a_polished = exact_subgrad_polish(best_a, total_iters=1000)
    report(sc, a_polished, "subgrad polish")

    # Try a few more L-BFGS rounds from the polished solution
    for beta in [1000, 2000]:
        if time.monotonic() > deadline - 5.0:
            break
        sc, a_opt, _ = run_lbfgs(a_polished, beta, maxiter=5000, maxfun=100000)
        report(sc, a_opt, f"post-polish lbfgs beta={beta}")

    # ==================================================================
    # Final verification and return
    # ==================================================================
    out = [float(max(0.0, min(1000.0, x))) for x in best_a]
    # Ensure sum is at least 0.01
    if sum(out) < 0.01:
        out = [0.5] * 64

    verified = float(evaluate_sequence(out))
    elapsed_total = time.monotonic() - t_start
    print(f"\n{'='*60}")
    print(f"FINAL: internal={best_score:.8f}, verified={verified:.8f}")
    print(f"Length: {len(out)}, Time: {elapsed_total:.1f}s / {budget_s}s")
    print(f"{'='*60}")
    return out
