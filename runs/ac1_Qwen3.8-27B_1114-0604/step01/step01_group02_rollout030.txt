```
def lowpass_noise(n, rng, power=2.0):
    z = np.fft.rfft(rng.standard_normal(n))
    freq = np.fft.rfftfreq(n, d=1.0)
    filt = 1.0 / (1.0 + power * freq)  # or (1/(1+10*freq**power))
    z *= filt
    out = np.fft.irfft(z, n)
    out -= out.mean()
    std = out.std()
    if std > 0: out /= std
    return out
```

```
    v[:] = b2*v + (1-b2)*g*g

    t2 = b2 ** t
```

```
def lowpass_noise(n, rng, power=2.0):
    if n <= 1:
        return np.zeros(n, dtype=np.float64)
    z = np.fft.rfft(rng.standard_normal(n))
    freq = np.fft.rfftfreq(n, d=1.0)
    # freq[0]=0, max=0.5
    filt = 1.0 / (1.0 + 10.0 * np.power(freq + 1e-12, power))
    z = z * filt
    out = np.fft.irfft(z, n)
    out -= out.mean()
    std = float(out.std())
    if std > 1e-15:
        out /= std
    else:
        out = np.zeros(n)
    return out
```

```python
"""
Search for a non-negative coefficient sequence minimizing

    2 * n * max(np.convolve(a, a)) / (sum(a)^2)

using fast FFT convolution, smooth log-sum-exp / top-k surrogates, projected
Adam updates, multi-length multi-start exploration, L-BFGS polishing on a
softmax reparameterization, and final candidate verification with
evaluate_sequence().

The optimizer starts from height_sequence_1 when available, resizes it to
several lengths, adds structured perturbations, and then refines the best
incumbent with progressively sharper objectives and small random restarts.
"""

import numpy as np
import time
import math

try:
    from scipy.optimize import minimize
    SCIPY_OPT = True
except Exception:
    SCIPY_OPT = False


_LB_N = 0
_LB_TAU = 1.0
_LB_RIDGE = 1e-12


def fft_len_for(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def fast_self_conv(a):
    """Linear full self-convolution via FFT. Matches np.convolve(a, a, 'full')."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.zeros(1, dtype=np.float64)
    M = 2 * n - 1
    N = fft_len_for(M)
    A = np.fft.rfft(a, N)
    out = np.fft.irfft(A * A, N)
    return out[:M]


def grad_dense_weights(a, w):
    """
    Gradient of sum_k w_k * conv(a,a)[k] with respect to a.

    For c_k = sum_j a_j a_{k-j}, dc_k/da_i = 2 a_{k-i} when k-i is valid.
    """
    a = np.asarray(a, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = a.size
    M = w.size
    N = fft_len_for(n + M - 1)
    A = np.fft.rfft(a, N)
    W = np.fft.rfft(w, N)
    conv = np.fft.irfft(A * W, N)
    # G_i / 2 = sum_j a_j w_{i+j} = conv( a, w[::-1] )[2n-2-i]
    return 2.0 * conv[n - 1:2 * n - 1][::-1]


def normalize_to_target(a, target):
    """Clip to [0,1000] and rescale to the given non-negative target sum."""
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if (not np.isfinite(s)) or s <= 1e-12:
        return np.full(max(1, a.size), target / float(max(1, a.size)), dtype=np.float64)
    return a * (target / s)


def internal_score(a):
    """Fast objective used during search. Mirrors the security-sanitized evaluator."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = fast_self_conv(a)
    m = float(np.max(c))
    return float(2.0 * a.size * m / (s * s))


def soft_weights_topk(c, tau, topk):
    """
    Return support indices and weights for a top-k soft-max surrogate.

    The returned F is a soft-max value of the top-k peaks (not multiplied by
    any constant; constant factors are irrelevant for Adam).
    """
    c = np.asarray(c, dtype=np.float64)
    M = c.size
    k = int(min(max(1, topk), M))

    if k >= M:
        idx = np.arange(M, dtype=np.int64)
    else:
        idx = np.argpartition(c, -k)[-k:]

    vals = c[idx]
    m = float(np.max(vals))

    if k == 1 or tau <= 1e-300:
        best_local = int(np.argmax(vals))
        return np.array([idx[best_local]], dtype=np.int64), np.array([1.0]), m

    exp_arg = (vals - m) / tau
    exp_arg = np.clip(exp_arg, -745.0, 0.0)
    r = np.exp(exp_arg)
    s = float(r.sum())

    if (not np.isfinite(s)) or s <= 0.0:
        best_local = int(np.argmax(vals))
        return np.array([idx[best_local]], dtype=np.int64), np.array([1.0]), m

    w = r / s
    F = m + tau * math.log(s)
    return idx, w, F


def grad_from_weights(a, idx, wvals):
    """Gradient of the top-k weighted convolution surrogate."""
    a = np.asarray(a, dtype=np.float64)
    idx = np.asarray(idx, dtype=np.int64)
    wvals = np.asarray(wvals, dtype=np.float64)
    n = a.size
    G = np.zeros(n, dtype=np.float64)

    if idx.size == 0:
        return G

    # For moderate sparsity, direct loop is faster than dense FFT.
    if idx.size <= 32 or n <= 512:
        for p in range(idx.size):
            k = int(idx[p])
            wi = float(wvals[p])
            if wi <= 1e-300:
                continue
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_start <= j_end:
                G[j_start:j_end + 1] += wi * 2.0 * a[j_start:j_end + 1][::-1]
        return G

    w = np.zeros(2 * n - 1, dtype=np.float64)
    w[idx] = wvals
    return grad_dense_weights(a, w)


def adam_update(a, g, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    """In-place Adam update."""
    m[:] = b1 * m + (1.0 - b1) * g
    v[:] = b2 * v + (1.0 - b2) * g * g
    t1 = b1 ** t
    t2 = b2 ** t
    mh = m / (1.0 - t1)
    vh = v / (1.0 - t2)
    a -= lr * mh / (np.sqrt(vh) + eps)


def lowpass_noise(n, rng, power=2.0):
    """Mean-zero, unit-roughness smooth random noise."""
    n = int(n)
    if n <= 1:
        return np.zeros(n, dtype=np.float64)
    if rng is None:
        rng = np.random.default_rng(0)

    z = np.fft.rfft(rng.standard_normal(n))
    freq = np.fft.rfftfreq(n, d=1.0)
    filt = 1.0 / (1.0 + 10.0 * np.power(freq + 1e-12, power))
    z = z * filt
    out = np.fft.irfft(z, n)
    out -= float(out.mean())
    std = float(out.std())
    if std > 1e-15:
        out /= std
    else:
        out *= 0.0
    return out


def resize_like(arr, n):
    """Resize a 1D sequence to length n by linear interpolation, clipped nonnegative."""
    arr = np.asarray(arr, dtype=np.float64).ravel()
    n = int(n)
    if arr.size == 0:
        return np.ones(n, dtype=np.float64) / float(max(1, n))
    if arr.size == n:
        out = arr.copy()
    elif arr.size == 1:
        out = np.full(n, float(arr[0]), dtype=np.float64)
    else:
        x_old = np.linspace(0.0, 1.0, arr.size)
        x_new = np.linspace(0.0, 1.0, n)
        out = np.interp(x_new, x_old, arr)
    out = np.clip(out, 0.0, None)
    s = float(out.sum())
    if s <= 0.0:
        out = np.ones(n, dtype=np.float64)
    else:
        out = out / s
    return out


def spectral_init(n, rng):
    """Positive sequence from random phase spectral components."""
    n = int(n)
    if n <= 1:
        return np.ones(n, dtype=np.float64)
    ph = rng.uniform(0.0, 2.0 * np.pi, size=n)
    A = np.exp(1j * ph)
    A[0] = float(n)
    a = np.abs(np.fft.irfft(A, n))
    # Mild sharpening/inversion of spikes; keeps all entries positive.
    a = np.power(a + 1e-8, 0.65)
    s = float(a.sum())
    if s <= 0.0:
        a = np.ones(n, dtype=np.float64)
    else:
        a = a / s
    return a


def x_from_a(a, floor=1e-12):
    """Log-amplitude parameterization for an a >= 0 vector."""
    a = np.clip(np.asarray(a, dtype=np.float64).ravel(), 0.0, None)
    s = float(a.sum())
    n = a.size
    if s <= 0.0 or n == 0:
        return np.zeros(max(1, n), dtype=np.float64)
    a = a / s
    return np.log(a + floor)


def softmax_from_x(x):
    """Stable softmax."""
    x = np.asarray(x, dtype=np.float64).ravel()
    n = x.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)
    xm = float(np.max(x))
    y = np.exp(np.clip(x - xm, -745.0, 0.0))
    s = float(y.sum())
    if (not np.isfinite(s)) or s <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    return y / s


def soft_max_value(c, tau):
    """Stable log-sum-exp soft maximum of c."""
    c = np.asarray(c, dtype=np.float64)
    m = float(np.max(c))
    if tau <= 1e-300:
        return m
    exp_arg = np.clip((c - m) / tau, -745.0, 0.0)
    r = np.exp(exp_arg)
    s = float(r.sum())
    if (not np.isfinite(s)) or s <= 0.0:
        return m
    return m + tau * math.log(s)


def soft_weights_all(c, tau):
    """Weights for dense soft-max over all convolution offsets."""
    c = np.asarray(c, dtype=np.float64)
    M = c.size
    m = float(np.max(c))

    w = np.zeros(M, dtype=np.float64)
    if M == 0:
        return w, 0.0
    if tau <= 1e-300:
        w[int(np.argmax(c))] = 1.0
        return w, m

    exp_arg = np.clip((c - m) / tau, -745.0, 0.0)
    r = np.exp(exp_arg)
    s = float(r.sum())
    if (not np.isfinite(s)) or s <= 0.0:
        w[int(np.argmax(c))] = 1.0
        return w, m

    w = r / s
    return w, m


def lbfgs_x_prepare(x):
    """Prepare x to length _LB_N and remove the flat softmax constant direction."""
    x = np.asarray(x, dtype=np.float64).ravel()
    n = int(_LB_N)
    if x.size < n:
        x = np.concatenate([x, np.zeros(n - x.size, dtype=np.float64)])
    elif x.size > n:
        x = x[:n].copy()
    return x - float(np.mean(x))


def lbfgs_x_fun(x):
    """Objective for L-BFGS on softmax log-amplitude variables."""
    xx = lbfgs_x_prepare(x)
    a = softmax_from_x(xx)
    c = fast_self_conv(a)
    F = 2.0 * float(_LB_N) * soft_max_value(c, _LB_TAU)
    F += _LB_RIDGE * float(np.dot(xx, xx))
    return float(F)


def lbfgs_x_jac(x):
    """Gradient for lbfgs_x_fun."""
    xx = lbfgs_x_prepare(x)
    a = softmax_from_x(xx)
    c = fast_self_conv(a)
    w, _ = soft_weights_all(c, _LB_TAU)
    G = grad_dense_weights(a, w)
    dot = float(np.dot(a, G))
    gx = 2.0 * float(_LB_N) * a * (G - dot)
    # Objective depends on xx = P x, P = I - 11^T/n.
    gx = gx - float(np.mean(gx))
    gx += 2.0 * _LB_RIDGE * xx
    return gx


def lbfgs_local_x(x0, tau_ratio=0.05, maxiter=300):
    """Short L-BFGS-B refinement in x-space. Returns score, x, a."""
    global _LB_N, _LB_TAU, _LB_RIDGE

    if not SCIPY_OPT:
        return np.inf, np.asarray(x0), np.zeros(0)

    x0 = np.asarray(x0, dtype=np.float64).copy()
    n = x0.size
    if n == 0:
        return np.inf, x0, np.zeros(0)

    x0 = x0 - float(np.mean(x0))
    x0 = np.clip(x0, -60.0, 60.0)

    a = softmax_from_x(x0)
    c = fast_self_conv(a)
    mv = float(np.max(c))
    tau = max(1e-12, tau_ratio * mv)

    _LB_N = int(n)
    _LB_TAU = float(tau)
    _LB_RIDGE = 1e-12

    options = {
        "maxiter": int(maxiter),
        "maxfun": int(max(1, 3 * maxiter)),
        "ftol": 1e-12,
        "gtol": 1e-8,
        "tol": 1e-10,
        "maxls": 20
    }

    res = minimize(
        lbfgs_x_fun,
        x0.copy(),
        jac=lbfgs_x_jac,
        method="L-BFGS-B",
        options=options
    )

    x = np.asarray(res.x, dtype=np.float64).copy()
    x = x - float(np.mean(x))
    a = softmax_from_x(x)
    sc = internal_score(a)
    return sc, x, a


def optimize_a_logsum(
    a0,
    deadline,
    rng,
    init_lr=None,
    init_tau_ratio=0.45,
    min_tau_ratio=0.02,
    init_topk=128,
    min_topk=1,
    patience=320,
    tag=""
):
    """
    Local optimizer on a >= 0 with fixed-sum projection.

    The actual scale is irrelevant, but we maintain a convenient target sum
    (at most 1000) so absolute Adam steps are sensible.
    """
    a0 = np.asarray(a0, dtype=np.float64).ravel()
    n = a0.size
    if n == 0:
        return np.inf, np.zeros(0, dtype=np.float64)

    if rng is None:
        rng = np.random.default_rng(0)

    target = min(1000.0, float(n))
    a = normalize_to_target(a0, target)
    mean_val = target / float(n)

    if init_lr is None:
        init_lr = max(1e-7, 0.05 * mean_val)
    lr = float(init_lr)
    min_lr = max(1e-8, 0.05 * init_lr)

    tau_ratio = float(init_tau_ratio)
    topk = int(init_topk)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    t = 0

    best_score = np.inf
    best_a = a.copy()
    no_improve = 0
    patience_cur = int(patience)
    step = 0

    while time.monotonic() < deadline:
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if (not np.isfinite(s)) or s <= 1e-12:
            a = np.full(n, target / float(n), dtype=np.float64)
            s = target
        else:
            a = a * (target / s)

        c = fast_self_conv(a)
        mv = float(np.max(c))
        score = float(2.0 * n * mv / (s * s))

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience_cur:
            # Restart from the current best with smooth random perturbation.
            if best_a.size == n:
                scale = 0.05 + 0.20 * float(rng.random())
                power = 1.0 + 2.0 * float(rng.random())
                noise = lowpass_noise(n, rng, power=power)
                a = best_a + scale * mean_val * noise
            else:
                a = mean_val * (1.0 + 0.3 * rng.standard_normal(n))

            a = normalize_to_target(a, target)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            lr = max(min_lr, init_lr * (0.6 + 0.3 * float(rng.random())))
            tau_ratio = max(min_tau_ratio, tau_ratio * 0.7)
            topk = max(int(min_topk), topk // 2)
            no_improve = 0
            patience_cur = max(60, patience_cur // 2)
            continue

        tau = max(1e-12, tau_ratio * max(mv, 1e-12))
        idx, wv, _ = soft_weights_topk(c, tau, topk)
        G = grad_from_weights(a, idx, wv)

        # Project onto the fixed-sum tangent plane.
        grad = G - float(np.sum(G)) / float(n)

        t += 1
        step += 1

        if no_improve > 50:
            lr = max(min_lr, lr * 0.995)

        adam_update(a, grad, m, v, t, lr)

        # Occasionally exploit reversal symmetry.
        if step % 200 == 0:
            a_try = 0.5 * (a + a[::-1])
            sc_try = internal_score(a_try)
            if np.isfinite(sc_try) and sc_try < score:
                a = a_try

        if no_improve % 20 == 0:
            tau_ratio = max(min_tau_ratio, tau_ratio * 0.995)
            if no_improve > 100:
                topk = max(int(min_topk), topk // 2)

    best_a = np.clip(best_a, 0.0, 1000.0)
    best_score = internal_score(best_a)
    return best_score, best_a


def record_candidate(lst, score, a, n):
    """Keep a small capped pool of candidate sequences."""
    if a is None or a.size == 0:
        return
    val = float(score)
    if not np.isfinite(val):
        return
    arr = np.clip(np.asarray(a, dtype=np.float64), 0.0, 1000.0).copy()
    lst.append((val, arr, int(arr.size)))
    if len(lst) > 30:
        worst = 0
        for i, item in enumerate(lst):
            if item[0] > lst[worst][0]:
                worst = i
        lst.pop(worst)


def choose_ns(parent_size, budget):
    """Choose a few sequence lengths to explore."""
    ns = []
    if parent_size is not None:
        ps = int(parent_size)
        if 32 <= ps <= 16384:
            ns.append(ps)
        elif ps > 16384:
            ns.append(16384)
            ns.append(8192)
        else:
            ns.append(512)

    for val in (512, 1024, 2048, 4096, 8192, 16384):
        if val not in ns:
            ns.append(val)

    ns = sorted(set(v for v in ns if v >= 32))
    if not ns:
        ns = [2048]

    if budget < 900:
        ns = ns[:1]
    elif budget < 1600:
        ns = ns[:2]
    else:
        ns = ns[:4]
    return ns


def build_candidates(ns, parent, rng):
    """Build diverse starting points. Parent-related candidates come first."""
    cands = []
    seen = set()

    if parent is not None:
        parent = np.asarray(parent, dtype=np.float64).ravel()
        pm = parent.size
        if 32 <= pm <= 16384:
            cands.append(("parent", parent.copy(), pm))
            seen.add("parent")

    for n in ns:
        n = int(n)
        if n < 32:
            n = 32

        if parent is not None:
            ap = resize_like(parent, n)
            cands.append(("par%d" % n, ap, n))
            seen.add("par%d" % n)

            if n != pm or "parent" not in seen:
                asym = 0.5 * (ap + ap[::-1])
                cands.append(("psym%d" % n, asym, n))
                seen.add("psym%d" % n)

            noise = lowpass_noise(n, rng, 2.0)
            app = np.clip(ap + 0.08 * (1.0 / n) * noise, 1e-8, None)
            s = float(app.sum())
            if s > 0:
                app = app / s
            cands.append(("parp%d" % n, app, n))
            seen.add("parp%d" % n)

        arnd = lowpass_noise(n, rng, 1.6)
        arnd = np.clip(0.75 + 0.25 * arnd, 1e-8, None)
        sr = float(arnd.sum())
        if sr > 0:
            arnd = arnd / sr
        cands.append(("rand%d" % n, arnd, n))
        seen.add("rand%d" % n)

        aspec = spectral_init(n, rng)
        cands.append(("spec%d" % n, aspec, n))
        seen.add("spec%d" % n)

    out = []
    for tag, a, n in cands:
        if tag in seen:
            a = np.asarray(a, dtype=np.float64).ravel()
            if a.size == int(n) and int(n) >= 32:
                out.append((tag, a, int(n)))
    return out


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Returns a list of non-negative floats.
    """
    start = time.monotonic()
    total = float(budget_s) if budget_s is not None else 2400.0
    if total < 120.0:
        total = 120.0

    reserve = 60.0 if total > 600.0 else max(10.0, 0.08 * total)
    deadline = start + total - reserve

    rng = np.random.default_rng(seed)

    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if arr.size > 0:
            parent = arr
    except Exception:
        parent = None

    print("parent_size=%s budget=%.1fs" % (
        None if parent is None else int(parent.size), total
    ))

    ns = choose_ns(parent.size if parent is not None else None, total)
    cands = build_candidates(ns, parent, rng)
    if not cands:
        cands = [("uni2048", np.ones(2048, dtype=np.float64), 2048)]

    # Keep the parent original first if present.
    cands_sorted = []
    rest = []
    for item in cands:
        tag = item[0]
        if tag.startswith("parent") or tag.startswith("par") or tag.startswith("psym"):
            cands_sorted.append(item)
        else:
            rest.append(item)
    cands = cands_sorted + rest

    top_candidates = []
    best_score = np.inf
    best_a = None
    best_n = 0

    # ------------------------------------------------------------------
    # Phase 1: broad local optimization from diverse starts.
    # ------------------------------------------------------------------
    phase1_end = start + 0.60 * (total - reserve)
    num = len(cands)

    for i, (tag, a0, n) in enumerate(cands):
        if time.monotonic() >= phase1_end:
            break

        now = time.monotonic()
        remaining = max(1.0, phase1_end - now)
        time_each = max(8.0, min(180.0, remaining / float(max(1, num - i))))
        cand_deadline = now + time_each

        if tag.startswith("parent") or tag.startswith("par"):
            init_tau = 0.55
            init_topk = 192
            patience = 320
        else:
            init_tau = 0.50
            init_topk = 128
            patience = 280

        sc, aa = optimize_a_logsum(
            a0,
            cand_deadline,
            rng,
            init_lr=None,
            init_tau_ratio=init_tau,
            min_tau_ratio=0.02,
            init_topk=init_topk,
            min_topk=1,
            patience=patience,
            tag=tag
        )

        record_candidate(top_candidates, sc, aa, n)

        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = aa.copy()
            best_n = int(n)
            print("phase1 [%s] best=%.8f n=%d" % (tag, best_score, best_n))

    if best_a is None:
        n0 = max(64, int(ns[0]))
        best_a = np.ones(n0, dtype=np.float64)
        best_score = internal_score(best_a)
        best_n = n0

    print("phase1_best=%.8f" % best_score)

    # ------------------------------------------------------------------
    # Phase 2: sharpen the current global best, then L-BFGS polish.
    # ------------------------------------------------------------------
    phase2_end = start + 0.85 * (total - reserve)

    if time.monotonic() < phase2_end:
        n = int(best_n)
        target = min(1000.0, float(n))
        mean_val = target / float(n)

        now = time.monotonic()
        t_end = min(phase2_end, now + max(30.0, 0.35 * max(0.0, phase2_end - now)))

        sc, aa = optimize_a_logsum(
            best_a,
            t_end,
            rng,
            init_lr=max(1e-7, 0.02 * mean_val),
            init_tau_ratio=0.12,
            min_tau_ratio=0.008,
            init_topk=32,
            min_topk=1,
            patience=600,
            tag="refine"
        )
        record_candidate(top_candidates, sc, aa, n)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_a = aa.copy()
            best_n = n
            print("phase2 refine=%.8f" % best_score)

        if SCIPY_OPT and time.monotonic() < phase2_end:
            x0 = x_from_a(best_a)
            for tr in (0.18, 0.07, 0.025):
                now = time.monotonic()
                if phase2_end - now < 25.0:
                    break
                if n <= 2048:
                    maxiter = 350
                elif n <= 4096:
                    maxiter = 200
                else:
                    maxiter = 100
                try:
                    sc_l, x_l, a_l = lbfgs_local_x(x0, tau_ratio=tr, maxiter=maxiter)
                except Exception:
                    break

                record_candidate(top_candidates, sc_l, a_l, n)
                if np.isfinite(sc_l) and sc_l < best_score:
                    best_score = sc_l
                    best_a = a_l.copy()
                    best_n = n
                    print("phase2 lbfgs tau=%.3f score=%.8f" % (tr, best_score))
                # Continue from the successful L-BFGS point if it is finite.
                if np.isfinite(sc_l):
                    x0 = x_l

    print("phase2_best=%.8f" % best_score)

    # ------------------------------------------------------------------
    # Phase 3: small restarts around the best solution until final deadline.
    # ------------------------------------------------------------------
    phase3_end = deadline
    if time.monotonic() < phase3_end:
        n = int(best_n)
        target = min(1000.0, float(n))
        mean_val = target / float(n)
        restart_count = 0

        while time.monotonic() < phase3_end - 20.0:
            now = time.monotonic()
            remaining = max(0.0, phase3_end - now)
            if remaining < 25.0:
                break

            time_each = min(60.0, max(5.0, 0.20 * remaining))
            scale = 0.02 + 0.08 * float(rng.random())
            noise = lowpass_noise(n, rng, 2.0)
            a0 = best_a + scale * mean_val * noise

            sc, aa = optimize_a_logsum(
                a0,
                now + time_each,
                rng,
                init_lr=max(1e-8, 0.01 * mean_val),
                init_tau_ratio=0.06,
                min_tau_ratio=0.003,
                init_topk=8,
                min_topk=1,
                patience=250,
                tag="restart%d" % restart_count
            )

            record_candidate(top_candidates, sc, aa, n)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = aa.copy()
                best_n = n
                print("phase3 restart%d score=%.8f" % (restart_count, best_score))

            restart_count += 1
            if restart_count > 300:
                break

    print("final_search_best=%.8f" % best_score)

    # ------------------------------------------------------------------
    # Final candidate selection among internal-score pool and symmetric forms.
    # ------------------------------------------------------------------
    record_candidate(top_candidates, best_score, best_a, best_n)

    final_a = None
    best_internal = np.inf

    for item in top_candidates:
        a = np.asarray(item[1], dtype=np.float64).ravel()
        for cand in (a, 0.5 * (a + a[::-1])):
            sc = internal_score(cand)
            if np.isfinite(sc) and sc < best_internal:
                best_internal = sc
                final_a = cand.copy()

    if final_a is None or final_a.size == 0:
        n_fallback = max(64, int(best_n)) if best_n > 0 else 1024
        final_a = np.ones(n_fallback, dtype=np.float64)

    final_a = np.clip(final_a, 0.0, 1000.0)
    sf = float(final_a.sum())
    if sf < 0.01:
        final_a = np.ones(max(64, final_a.size), dtype=np.float64)

    out = [float(x) for x in final_a]
    sym = [float(x) for x in (0.5 * (final_a + final_a[::-1]))]

    best_out = out
    best_val = np.inf

    try:
        val = float(evaluate_sequence(out))
    except Exception:
        val = np.inf
    best_val = val

    try:
        val_sym = float(evaluate_sequence(sym))
    except Exception:
        val_sym = np.inf
    if np.isfinite(val_sym) and (not np.isfinite(val) or val_sym < val):
        best_out = sym
        best_val = val_sym

    print(
        "final_internal=%.8f final_evaluated=%.8f len=%d time=%.1fs" % (
            best_internal, best_val, len(best_out), time.monotonic() - start
        )
    )

    if not np.isfinite(best_val):
        fallback_len = max(64, min(1024, len(best_out)))
        fallback = [1.0] * fallback_len
        try:
            vf = float(evaluate_sequence(fallback))
        except Exception:
            vf = np.inf
        if np.isfinite(vf):
            return fallback
        return [1.0] * 64

    return best_out
```
