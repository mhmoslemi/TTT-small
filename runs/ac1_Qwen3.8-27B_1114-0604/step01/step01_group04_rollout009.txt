import numpy as np
from scipy.optimize import linprog
import time

def full_conv(a):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def project_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def score_normalized(a):
    """Score assuming sum(a) = 1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))

def lp_direction(a, M):
    """Solve LP: max sum(g) s.t. g >= 0, conv(a,g) <= M."""
    n = len(a)
    # Build constraint matrix
    k = np.arange(2*n-1)
    j = np.arange(n)
    # A[k_idx, j_idx] = a[k_idx - j_idx] if 0 <= k_idx - j_idx < n
    A = np.zeros((2*n-1, n))
    for ki in range(2*n-1):
        for ji in range(max(0, ki-n+1), min(n, ki+1)):
            A[ki, ji] = a[ki - ji]
    
    c = -np.ones(n)
    b = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x

def lp_iterate(a, time_budget, rng, t_init=0.1):
    """Apply LP-based iterations to minimize max(conv(a,a))."""
    n = a.size
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_sc = score_normalized(a)
    
    t0 = time.monotonic()
    t = t_init
    iters = 0
    last_improve = 0
    
    while time.monotonic() - t0 < time_budget:

        M = float(np.max(b))
        
        g = lp_direction(a, M)

                best_a = a_new.copy()
                a = a_new.copy()

Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based directional search.
At each step, solves an LP to find the best direction g satisfying conv(a,g) <= M,
then performs a line search along the direction a -> (1-t)*a + t*g.

import numpy as np
from scipy.optimize import linprog
import time


def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def score_norm(a):
    """Score = 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def build_A(a):
    """Build convolution matrix: A @ g = conv(a, g) (element-wise)."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(n, k + 1)
        for j in range(j_start, j_end):
            A[k, j] = a[k - j]
    return A


def solve_lp(a, M):
    """Solve: max sum(g) s.t. g >= 0, A@g <= M*ones."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x


def lp_optimize(a_init, time_budget, rng, verbose=True):
    """LP-based optimization to minimize max(conv(a,a)) with sum(a)=1, a>=0."""
    n = a_init.size
    a = project_simplex(a_init.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = score_norm(a)
    
    t0 = time.monotonic()
    t = 0.1
    iters = 0
    last_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        M = float(np.max(b))
        
        # Solve LP
        g = solve_lp(a, M)
        if g is None:
            if verbose:
                print(f"  LP failed at iter {iters}")
            break
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Check if g is essentially the same as a (convergence)
        if np.abs(g - a).max() < 1e-8:
            if verbose:
                print(f"  Converged: g ≈ a at iter {iters}")
            break
        
        # Line search
        improved = False
        for tt in [t, t*0.5, t*0.25, t*0.1]:
            a_new = (1 - tt) * a + tt * g
            a_new = project_simplex(a_new)
            sc = score_norm(a_new)
            if sc < best_sc - 1e-10:
                best_sc = sc
                best_a = a_new.copy()
                a = a_new.copy()
                last_improve = iters
                improved = True
                break
        
        if not improved:
            t *= 0.5
            if t < 0.005:
                # Perturb to escape
                a = project_simplex(best_a + 0.02 * rng.standard_normal(n))
                t = 0.05
                last_improve = iters
        
        iters += 1
        if verbose and iters % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"  iter {iters}: score={best_sc:.6f}, t={t:.4f}, elapsed={elapsed:.1f}s")
        
        if iters - last_improve > 200:
            break
    
    return best_a, best_sc

At each iteration, solves an LP to find the best non-negative direction g
satisfying conv(a,g) <= max(conv(a,a)), then performs a line search

    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def score_norm(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def get_conv_indices(n):
    """Precompute indices for building the convolution matrix."""
    if n not in _CONV_INDICES:
        k_idx = []
        j_idx = []
        a_idx = []
        for k in range(2 * n - 1):
            for j in range(max(0, k - n + 1), min(n, k + 1)):
                k_idx.append(k)
                j_idx.append(j)
                a_idx.append(k - j)
        _CONV_INDICES[n] = (
            np.array(k_idx, dtype=np.int32),
            np.array(j_idx, dtype=np.int32),
            np.array(a_idx, dtype=np.int32),
        )
    return _CONV_INDICES[n]


def build_A(a):
    """Build matrix A such that A @ g = conv(a, g). Shape: (2n-1) x n."""
    n = len(a)
    k_idx, j_idx, a_idx = get_conv_indices(n)
    A = np.zeros((2 * n - 1, n))
    A[k_idx, j_idx] = a[a_idx]
    return A


def solve_lp(a, M):
    """Solve: max sum(g) s.t. g >= 0, conv(a,g) <= M (element-wise)."""
    n = len(a)
    A = build_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if res.status != 0:
        return None
    return res.x


def lp_optimize(a_init, time_budget, rng, verbose=False, tag=""):
    """LP-based optimization to minimize max(conv(a,a)) with sum(a)=1, a>=0."""
    n = a_init.size
    a = project_simplex(a_init.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = score_norm(a)
    
    t0 = time.monotonic()
    t = 0.1
    iters = 0
    last_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        M = float(np.max(b))
        
        g = solve_lp(a, M)
        if g is None:
            if verbose:
                print(f"  [{tag}] LP failed at iter {iters}")
            break
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum
        
        # Check convergence: g ≈ a means we're at a fixed point
        if float(np.abs(g - a).max()) < 1e-9:
            if verbose:
                print(f"  [{tag}] Converged at iter {iters}, score={best_sc:.6f}")
            break
        
        # Line search with multiple step sizes
        improved = False
        for tt in [t, t * 0.5, t * 0.25, t * 0.1, 0.5, 0.3, 0.1]:
            a_new = (1.0 - tt) * a + tt * g
            a_new = project_simplex(a_new)
            sc = score_norm(a_new)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_a = a_new.copy()
                a = a_new.copy()
                last_improve = iters
                improved = True
                break
        
        if not improved:
            t *= 0.5
            if t < 0.003:
                # Perturb to escape local optimum
                a = project_simplex(best_a + 0.03 * rng.standard_normal(n))
                t = 0.08
                last_improve = iters
                if verbose:
                    print(f"  [{tag}] Perturbed at iter {iters}")
        
        iters += 1
        if verbose and iters % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  [{tag}] iter {iters}: score={best_sc:.6f}, t={t:.4f}, elapsed={elapsed:.1f}s")
        
        if iters - last_improve > 300:
            if verbose:
                print(f"  [{tag}] Stalled at iter {iters}, breaking")
            break
    
    return best_a, best_sc


def init_uniform(n):
    return np.ones(n) / n


def init_u_shape(n):
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_sided(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / c) ** 2 + 0.05
    return a / a.sum()


def init_two_bump(n, p1, p2, w, rng):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-(((i - n * p1) / (n * w))) ** 2) + np.exp(-(((i - n * p2) / (n * w))) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_cos_bump(n, rng):
    """Cosine-shaped bump in the middle."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = np.cos(np.pi * (i - c) / (n * 0.4)) ** 2
    a = np.maximum(a, 0) + 0.01
    return a / a.sum()


def init_three_bump(n, rng):
    """Three bumps at positions 0.2, 0.5, 0.8."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-(((i - 0.2*n) / (0.08*n))) ** 2) + \
        np.exp(-(((i - 0.5*n) / (0.08*n))) ** 2) + \
        np.exp(-(((i - 0.8*n) / (0.08*n))) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def polish_random(a, time_budget, rng):
    """Local polish via random perturbations with decreasing scale."""
    n = a.size
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_sc = score_norm(a)
    
    t0 = time.monotonic()
    scale = 0.03
    while time.monotonic() - t0 < time_budget and scale > 1e-6:
        improved = False
        for _ in range(30):
            cand = a + scale * rng.standard_normal(n)
            cand = project_simplex(cand)
            sc = score_norm(cand)
            if sc < best_sc:
                best_sc = sc
                best_a = cand.copy()
                a = cand.copy()
                improved = True
        if not improved:
            scale *= 0.6
        else:
            scale *= 0.95
    
    return best_a, best_sc


def subgrad_polish(a, time_budget, rng):
    """Short smoothed-subgradient polish."""
    n = a.size
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_sc = score_norm(a)
    
    eta = 0.05
    tau_frac = 0.02
    vel = np.zeros(n)
    beta = 0.5
    iters = 0
    last_improve = 0
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        M = float(np.max(b))
        tau = max(tau_frac * M, 1e-15)
        
        # Smoothed subgradient
        shift = b / tau
        shift -= shift.max()
        w = np.exp(shift)
        w /= w.sum()
        g = 2.0 * np.correlate(w, a, mode='valid')
        
        vel = beta * vel + (1.0 - beta) * g
        a_new = project_simplex(a - eta * vel)
        sc = score_norm(a_new)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new.copy()
            last_improve = iters
        elif rng.random() < 0.05:
            a = a_new.copy()
        
        iters += 1
        if iters - last_improve > 500:
            a = project_simplex(best_a + 0.02 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.03 + 0.05 * rng.random()
            last_improve = iters
    
    return best_a, best_sc
