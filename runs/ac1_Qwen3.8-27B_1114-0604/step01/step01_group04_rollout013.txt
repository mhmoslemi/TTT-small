```python
"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using:
1. LP-based water-filling iteration (from the SOTA description)
2. Smooth log-sum-exp optimization with scipy
3. Subgradient descent as fallback
4. Multiple n values and diverse initializations

import numpy as np
import time
from scipy.optimize import minimize, linprog

def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))

def build_conv_matrix(a):
    """Build the (2n-1) x n matrix C such that C @ g = conv(a, g)."""
    n = a.size
    C = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        for j in range(n):
            if 0 <= k - j < n:
                C[k, j] = a[k - j]
    return C

def lp_water_fill(f, M, max_sum=None):
    """
    Solve LP: maximize sum(g) subject to g >= 0, (f*g)_k <= M for all k.
    Returns the optimal g (unrescaled) and the optimal sum.
    """
    n = f.size
    C = build_conv_matrix(f)
    
    # Maximize c^T g where c = ones(n)
    # Subject to: C @ g <= M (2n-1 constraints), g >= 0
    c = -np.ones(n)  # Negative for minimization
    A_ub = C
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    
    if result.status != 0:
        return None, None
    
    g = result.x
    return g, float(-result.fun)

def lp_iteration(f, t=0.1, max_rounds=100, tol=1e-8):
    """
    LP-based water-filling iteration.
    f is normalized to sum=1.
    Returns improved f and score.
    """
    n = f.size
    best_f = f.copy()
    best_sc = normed_score(f)
    
    for it in range(max_rounds):
        b = full_conv(f)
        M = float(np.max(b))
        
        # Solve LP
        g, g_sum = lp_water_fill(f, M)
        if g is None or g_sum <= 0:
            break
        
        # Rescale g to sum=1
        g_norm = g / g_sum
        
        # Blend
        f_new = (1 - t) * f + t * g_norm
        f_new = project_simplex(f_new)
        
        new_sc = normed_score(f_new)
        
        if new_sc < best_sc:
            best_sc = new_sc
            best_f = f_new.copy()
        
        if new_sc < best_sc * (1 + tol):
            # Try smaller step
            t = t * 0.9
        else:
            # Reduce t if not improving
            if new_sc > best_sc:
                t = max(t * 0.5, 0.01)
                # Revert
                # Don't update f, just reduce step
                continue
        
        f = f_new
        
        if abs(new_sc - best_sc) < tol:
            break
    
    return best_f, best_sc

def logsumexp_obj(x, n, tau):
    """Objective: 2*n*tau*log(sum(exp(b_k/tau))) where b = conv(x,x)."""
    b = full_conv(x)
    shift = b / tau
    shift -= shift.max()
    lse = shift.max() + np.log(np.sum(np.exp(shift)))
    return 2.0 * n * tau * lse

def subgrad_optimize(a, time_budget, rng):
    """Projected subgradient descent."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    eta = 0.3
    tau = 0.04
    vel = np.zeros(n)
    beta = 0.5
    
    t0 = time.monotonic()
    iters = 0
    last_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))
        M = float(np.max(b))
        
        # Subgradient of max(b): use softmax
        shift = b / (tau * M)
        shift -= shift.max()
        w = np.exp(shift)
        w /= w.sum()
        
        g = 2.0 * np.correlate(w, a, mode='valid')
        vel = beta * vel + (1.0 - beta) * g
        a_new = project_simplex(a - eta * vel)
        
        new_sc = normed_score(a_new)
        if new_sc <= cur_sc or rng.random() < 0.05:
            a = a_new
        
        if new_sc < best_sc:

            last_improve = iters
        
        iters += 1
        if iters % 100 == 0:
            eta *= 0.995
            if iters - last_improve > 2000:

                vel = np.zeros(n)
                eta = 0.2 + 0.2 * rng.random()
                last_improve = iters
    
    return best_a, best_sc

def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.05
    return a / a.sum()

def init_two_cluster(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.15 + 0.15 * rng.random())
    c2 = n * (0.70 + 0.15 * rng.random())
    sigma = n * (0.04 + 0.08 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.02 * np.abs(rng.standard_normal(n))
    return a / a.sum()

def init_u_shape(n):
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i).astype(np.float64) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()

def init_sided(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / c) ** 2 + 0.1
    return a / a.sum()

def init_cosine(n, rng):
    i = np.arange(n, dtype=np.float64)
    freq = 1.0 + int(rng.integers(0, 3))
    a = (1.0 + freq * np.cos(2.0 * np.pi * i / max(n - 1, 1))) ** 2
    a += 1e-6
    return a / a.sum()

def init_noisy_uniform(n, rng):
    a = np.ones(n) + 0.8 * rng.standard_normal(n)
    a = np.abs(a) + 0.01
    return a / a.sum()

def get_inits(n, rng):
```

```python
"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using a combination of:
1. LP-based water-filling iteration (from SOTA description) - solves LP to find
   optimal blending direction that doesn't increase peak autocorrelation
2. Smooth log-sum-exp optimization via scipy SLSQP with adaptive temperature
3. Projected subgradient descent with multi-scale softmax
4. Simulated annealing with structured proposals
5. Coordinate descent and random polish
Uses FFT-based convolution for O(n log n) per iteration.
"""

import numpy as np
import time
from scipy.optimize import minimize, linprog


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix C such that C @ g = conv(a, g)."""
    n = a.size
    m = 2 * n - 1
    C = np.zeros((m, n))
    for j in range(n):
        k_idx = np.arange(j, min(j + n, m))
        C[k_idx, j] = a[:len(k_idx)]
    return C


def lp_water_fill(f, M):
    """
    Solve LP: maximize sum(g) subject to g >= 0, (f*g)_k <= M for all k.
    Returns (g, optimal_sum) or (None, None) on failure.
    """
    n = f.size
    C = build_conv_matrix(f)
    c = -np.ones(n)
    A_ub = C
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if result.status != 0:
            return None, None
        g = result.x
        return g, float(-result.fun)
    except Exception:
        return None, None


def lp_iteration(f, time_budget, rng):
    """
    LP-based water-filling iteration with adaptive step size.
    f is normalized to sum=1. Returns (best_f, best_score).
    """
    n = f.size
    f = project_simplex(f.astype(np.float64).copy())
    best_f = f.copy()
    best_sc = normed_score(f)
    
    t0 = time.monotonic()
    t = 0.15
    iters = 0
    stall = 0
    
    while time.monotonic() - t0 < time_budget:
        b = full_conv(f)
        M = float(np.max(b))
        cur_sc = 2.0 * n * M
        
        g, g_sum = lp_water_fill(f, M)
        if g is None or g_sum <= 1e-10:
            break
        
        g_norm = g / g_sum
        f_new = (1.0 - t) * f + t * g_norm
        f_new = project_simplex(f_new)
        
        new_sc = normed_score(f_new)
        
        if new_sc < best_sc:
            best_sc = new_sc
            best_f = f_new.copy()
            f = f_new
            stall = 0
            t = min(t * 1.1, 0.3)
        else:
            stall += 1
            t = max(t * 0.7, 0.02)
            if stall > 20:
                f = best_f.copy()
                t = 0.1
                stall = 0
                # Try a perturbed version
                if rng.random() < 0.5:
                    f = project_simplex(best_f + 0.02 * rng.standard_normal(n))
        
        iters += 1
        if iters % 50 == 0:
            print(f"  [lp] iter={iters}, sc={best_sc:.6f}, t={t:.4f}")
    
    return best_f, best_sc


def smooth_grad(a, b, tau):
    """Smoothed subgradient using softmax at one temperature."""
    n = a.size
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(b.size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= s
    return 2.0 * np.correlate(w, a, mode='valid')


def multi_scale_grad(a, b, tau_base, n_scales=3):
    """Average subgradients at multiple temperature scales."""
    g_total = np.zeros(a.size)
    for s in range(n_scales):
        tau = tau_base * (2.0 ** s)
        g_total += smooth_grad(a, b, tau)
    return g_total / n_scales

    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    eta = 0.25
    tau_frac = 0.05
    beta = 0.4
    vel = np.zeros(n)
    
    t0 = time.monotonic()
    iters = 0
    last_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))
        M = float(np.max(b))
        tau = tau_frac * M
        if tau < 1e-20:
            tau = 1e-20
        
        g = multi_scale_grad(a, b, tau, 3)
        vel = beta * vel + (1.0 - beta) * g
        a_new = project_simplex(a - eta * vel)
        
        new_sc = normed_score(a_new)
        if new_sc <= cur_sc or rng.random() < 0.04:
            a = a_new
        else:
            a = project_simplex(a - 0.5 * eta * vel)
        
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = iters
        
        iters += 1
        if iters % 200 == 0:
            eta *= 0.99
            tau_frac = 0.03 + 0.04 * rng.random()
        
        if iters - last_improve > 3000:
            a = project_simplex(best_a + 0.04 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.15 + 0.25 * rng.random()
            last_improve = iters
    
    return best_a, best_sc


def logsumexp_objective(x, n, tau):
    """Objective: 2*n*tau*logsumexp(b/tau) where b = conv(x,x)."""
    b = full_conv(x)
    shift = b / tau
    shift -= shift.max()
    return 2.0 * n * tau * (shift.max() + np.log(np.sum(np.exp(shift))))


def logsumexp_grad(x, n, tau):
    """Gradient of the log-sum-exp smoothed objective."""
    b = full_conv(x)
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    w /= w.sum()
    return 2.0 * np.correlate(w, x, mode='valid')


def smooth_scipy_optimize(a, time_budget, rng, tau_init=0.02):
    """Optimize using scipy SLSQP with log-sum-exp smoothing, decreasing tau."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    t0 = time.monotonic()
    tau = tau_init
    
    while time.monotonic() - t0 < time_budget:
        if tau < 1e-7:
            break
        
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 1.0,
                        'jac': lambda x: np.ones(n)}]
        bounds = [(0, None)] * n
        
        # Limit iterations for speed
        max_iter = max(10, int((time_budget - (time.monotonic() - t0)) * 2))
        max_iter = min(max_iter, 100)
        
        try:
            res = minimize(
                lambda x: logsumexp_objective(x, n, tau),
                a.copy(),
                jac=lambda x: logsumexp_grad(x, n, tau),
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': max_iter, 'ftol': 1e-12}
            )
            if res.fun < np.inf:
                x_new = project_simplex(res.x)
                sc = normed_score(x_new)
                if sc < best_sc:
                    best_sc = sc
                    best_a = x_new.copy()
                    a = x_new.copy()
                    print(f"  [smooth] tau={tau:.6f}, sc={sc:.6f}")
        except Exception:
            pass
        
        tau *= 0.75
    
    return best_a, best_sc


def simulated_annealing(a, time_budget, rng):
    """Simulated annealing with structured proposals on the simplex."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    t0 = time.monotonic()
    T = 0.05
    T_end = 0.001
    decay = 0.995
    
    while time.monotonic() - t0 < time_budget and T > T_end:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))
        
        # Proposal: move mass between two random positions
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue
        delta = T * rng.standard_normal()
        # Move delta from i to j
        new_val_i = a[i] - delta
        new_val_j = a[j] + delta
        if new_val_i >= 0 and new_val_j >= 0:
            a[i] = new_val_i
            a[j] = new_val_j
        else:
            # Scale down delta
            if delta > 0:
                delta = min(delta, a[i])
            else:
                delta = max(delta, -a[j])
            a[i] = a[i] - delta
            a[j] = a[j] + delta
        
        new_sc = normed_score(a)
        
        # Metropolis acceptance
        delta_sc = new_sc - cur_sc
        if delta_sc < 0 or rng.random() < np.exp(-delta_sc / max(T, 1e-10)):
            if new_sc < best_sc:
                best_sc = new_sc
                best_a = a.copy()
        else:
            # Revert
            a[i] += delta
            a[j] -= delta
        
        T *= decay
    
    return best_a, best_sc


def coordinate_polish(a, time_budget, rng):
    """Coordinate descent polish: optimize one coordinate at a time."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    t0 = time.monotonic()
    order = list(range(n))
    
    while time.monotonic() - t0 < time_budget:
        np.random.seed(int(rng.integers(0, 2**31)))
        np.random.shuffle(order)
        improved = False
        
        for i in order:
            if time.monotonic() - t0 > time_budget:
                break
            old_val = a[i]
            if old_val < 1e-10:
                # Try setting to small positive
                candidates = [0.0, 0.001, 0.01]
            else:
                # Try scaling the coordinate
                candidates = [0.0, old_val * 0.25, old_val * 0.5, old_val * 0.75,
                             old_val, old_val * 1.25, old_val * 1.5, old_val * 2.0]
            
            best_local = old_val
            best_local_sc = best_sc
            
            for cv in candidates:
                a[i] = cv
                a = project_simplex(a)
                sc = normed_score(a)
                if sc < best_local_sc:
                    best_local_sc = sc
                    best_local = cv
                a[i] = old_val
                a = project_simplex(a)
            
            if best_local != old_val:
                a[i] = best_local
                a = project_simplex(a)
                if best_local_sc < best_sc:
                    best_sc = best_local_sc
                    best_a = a.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_sc


def random_polish(a, time_budget, rng):
    """Random perturbation polish with decreasing scale."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    t0 = time.monotonic()
    scale = 0.03
    
    while time.monotonic() - t0 < time_budget and scale > 1e-6:
        improved = False
        n_tries = max(5, int(2000 * (time_budget - (time.monotonic() - t0)) / max(time_budget, 1)))
        n_tries = min(n_tries, 100)
        
        for _ in range(n_tries):
            if time.monotonic() - t0 > time_budget:
                break
            cand = a + scale * rng.standard_normal(n)
            cand = project_simplex(cand)
            sc = normed_score(cand)
            if sc < best_sc:
                best_sc = sc
                best_a = cand.copy()
                a = cand.copy()
                improved = True
        
        if not improved:
            scale *= 0.6
        else:
            scale *= 0.8
    
    return best_a, best_sc


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.05
    return a / a.sum()


def init_two_cluster(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.15 + 0.15 * rng.random())
    c2 = n * (0.70 + 0.15 * rng.random())
    sigma = n * (0.04 + 0.08 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.02 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_u_shape(n):
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i).astype(np.float64) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_sided(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / c) ** 2 + 0.1
    return a / a.sum()


def init_cosine(n, rng):
    i = np.arange(n, dtype=np.float64)
    freq = 1.0 + int(rng.integers(0, 4))
    a = (1.0 + freq * np.cos(2.0 * np.pi * i / max(n - 1, 1))) ** 2
    a += 1e-6
    return a / a.sum()


def init_noisy_uniform(n, rng):
    a = np.ones(n) + 0.8 * rng.standard_normal(n)
    a = np.abs(a) + 0.01
    return a / a.sum()


def init_peaky(n, rng):
    """Mostly uniform with a few peaks."""
    a = np.ones(n) * 0.1
    n_peaks = rng.integers(2, 5)
    for _ in range(n_peaks):
        pos = rng.integers(0, n)
        width = rng.integers(1, n // 10 + 1)
        a[max(0, pos-width):pos+width+1] += rng.random() * 3.0
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_symmetric(n, rng):
    """Symmetric initialization."""
    half = n // 2
    h = np.abs(rng.standard_normal(half)) + 0.1
    if n % 2 == 0:
        a = np.concatenate([h, h[::-1]])
    else:
        a = np.concatenate([h, [rng.random()], h[::-1]])
    return a / a.sum()


def get_inits(n, rng):
    """Return list of diverse initialization arrays for given n."""
    inits = []
    inits.append(init_two_cluster(n, rng))
    inits.append(init_u_shape(n))
    inits.append(init_sided(n))
    inits.append(init_random(n, rng))
    inits.append(init_noisy_uniform(n, rng))
    inits.append(init_cosine(n, rng))
    inits.append(init_peaky(n, rng))
    inits.append(init_symmetric(n, rng))
    return inits


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Combines LP water-filling, smooth optimization, subgradient descent,
    simulated annealing, and coordinate/polish across multiple lengths.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)
    
    best_overall = None
    best_overall_sc = np.inf
    
    # Try to load previous best
    prev_a = None
    prev_sc = np.inf
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size >= 10 and np.all(np.isfinite(h)) and h.sum() > 0.01:
            prev_a = h / h.sum()
            prev_sc = 2.0 * h.size * float(np.max(full_conv(prev_a)))
            print(f"[init] Previous best: n={h.size}, score={prev_sc:.6f}")
            best_overall = prev_a.copy()
            best_overall_sc = prev_sc
    except Exception:
        pass
    
    # n values to try
    n_list = [96, 128, 160, 192, 256, 320, 384, 512]
    if prev_a is not None:
        np_ = prev_a.size
        if np_ not in n_list and 32 <= np_ <= 1024:
            n_list.append(np_)
        n_list = sorted(set(n_list))
    
    total_time = max(1.0, deadline - time.monotonic())
    print(f"[plan] n_list={n_list}, total_time={total_time:.1f}s")
    
    # Allocate time per n
    raw_w = []
    for nv in n_list:
        if prev_a is not None and nv == prev_a.size:
            raw_w.append(4.0 / (0.3 + nv / 384.0))
        elif nv in [128, 160, 192]:
            raw_w.append(2.0 / (0.3 + nv / 384.0))
        else:
            raw_w.append(1.0 / (0.3 + nv / 384.0))
    total_w = sum(raw_w)
    weights = [w / total_w for w in raw_w]
    
    for idx_n, nv in enumerate(n_list):
        if time.monotonic() > deadline:
            break
        
        time_n = weights[idx_n] * total_time * 0.82
        inits = get_inits(nv, rng)
        
        if prev_a is not None and prev_a.size == nv:
            inits.insert(0, prev_a.copy())
        
        n_inits = len(inits)
        time_per_init = time_n / n_inits
        
        for ii, a_init in enumerate(inits):
            if time.monotonic() > deadline:
                break
            
            remaining_init = min(time_per_init, deadline - time.monotonic() - 2.0)
            if remaining_init < 3.0:
                break
            
            a_cur = project_simplex(a_init.copy())
            sc_cur = normed_score(a_cur)
            label = f"n={nv}, init={ii}"
            
            # Phase 1: LP water-filling (most novel, try first)
            t_lp = remaining_init * 0.30
            if t_lp > 2.0:
                t_lp_actual = min(t_lp, deadline - time.monotonic() - 2.0)
                a_lp, sc_lp = lp_iteration(a_cur, t_lp_actual, rng)
                if sc_lp < sc_cur:
                    sc_cur = sc_lp
                    a_cur = a_lp
                    print(f"  [{label}] LP: {sc_cur:.6f}")
            
            if time.monotonic() > deadline - 2:
                break
            
            # Phase 2: Subgradient
            t_sg = remaining_init * 0.30
            if t_sg > 2.0:
                t_sg_actual = min(t_sg, deadline - time.monotonic() - 2.0)
                a_sg, sc_sg = subgrad_optimize(a_cur, t_sg_actual, rng)
                if sc_sg < sc_cur:
                    sc_cur = sc_sg
                    a_cur = a_sg
                    print(f"  [{label}] SG: {sc_cur:.6f}")
            
            if time.monotonic() > deadline - 2:
                break
            
            # Phase 3: Smooth scipy
            t_sm = remaining_init * 0.25
            if t_sm > 2.0:
                t_sm_actual = min(t_sm, deadline - time.monotonic() - 2.0)
                a_sm, sc_sm = smooth_scipy_optimize(a_cur, t_sm_actual, rng)
                if sc_sm < sc_cur:
                    sc_cur = sc_sm
                    a_cur = a_sm
                    print(f"  [{label}] SM: {sc_cur:.6f}")
            
            if time.monotonic() > deadline - 2:
                break
            
            # Phase 4: Simulated annealing
            t_sa = remaining_init * 0.15
            if t_sa > 2.0:
                t_sa_actual = min(t_sa, deadline - time.monotonic() - 2.0)
                a_sa, sc_sa = simulated_annealing(a_cur, t_sa_actual, rng)
                if sc_sa < sc_cur:
                    sc_cur = sc_sa
                    a_cur = a_sa
                    print(f"  [{label}] SA: {sc_cur:.6f}")
            
            if sc_cur < best_overall_sc:
                best_overall_sc = sc_cur
                best_overall = a_cur.copy()
                print(f"  [{label}] *** NEW BEST: {sc_cur:.6f} ***")
        
        print(f"[done] n={nv} complete. Best so far: {best_overall_sc:.6f}")
    
    # ===== Final Polish Phase =====
    if best_overall is not None and time.monotonic() < deadline - 5.0:
        remain = deadline - time.monotonic() - 3.0
        print(f"[polish] Starting final polish, remaining={remain:.1f}s, n={best_overall.size}")
        
        # 1. Coordinate descent
        t_coord = min(remain * 0.25, 120.0)
        if t_coord > 3.0:
            a_c, sc_c = coordinate_polish(best_overall, t_coord, rng)
            if sc_c < best_overall_sc:
                best_overall_sc = sc_c
                best_overall = a_c.copy()
                print(f"[polish] coord: {sc_c:.6f}")
        
        if time.monotonic() > deadline - 5:
            remain = deadline - time.monotonic() - 3.0
        
        # 2. LP refinement with small step
        t_lp2 = min(remain * 0.25, 60.0)
        if t_lp2 > 3.0:
            a_lp2, sc_lp2 = lp_iteration(best_overall, t_lp2, rng)
            if sc_lp2 < best_overall_sc:
                best_overall_sc = sc_lp2
                best_overall = a_lp2.copy()
                print(f"[polish] LP2: {sc_lp2:.6f}")
        
        if time.monotonic() > deadline - 5:
            remain = deadline - time.monotonic() - 3.0
        
        # 3. Random polish
        t_rand = min(remain * 0.25, 60.0)
        if t_rand > 2.0:
            a_r, sc_r = random_polish(best_overall, t_rand, rng)
            if sc_r < best_overall_sc:
                best_overall_sc = sc_r
                best_overall = a_r.copy()
                print(f"[polish] rand: {sc_r:.6f}")
        
        if time.monotonic() > deadline - 5:
            remain = deadline - time.monotonic() - 3.0
        
        # 4. One more round of LP + coord on the best
        t_final = min(remain * 0.5, 60.0)
        if t_final > 3.0:
            a_f1, sc_f1 = lp_iteration(best_overall, t_final * 0.5, rng)
            if sc_f1 < best_overall_sc:
                best_overall_sc = sc_f1
                best_overall = a_f1.copy()
            if time.monotonic() < deadline - 5:
                t_f2 = min(t_final * 0.5, deadline - time.monotonic() - 4.0)
                if t_f2 > 2.0:
                    a_f2, sc_f2 = coordinate_polish(best_overall, t_f2, rng)
                    if sc_f2 < best_overall_sc:
                        best_overall_sc = sc_f2
                        best_overall = a_f2.copy()
                        print(f"[polish] final: {sc_f2:.6f}")
        
        # 5. Short subgradient burst
        if time.monotonic() < deadline - 4.0:
            t_sg2 = min(deadline - time.monotonic() - 3.0, 30.0)
            if t_sg2 > 2.0:
                a_sg2, sc_sg2 = subgrad_optimize(best_overall, t_sg2, rng)
                if sc_sg2 < best_overall_sc:
                    best_overall_sc = sc_sg2
                    best_overall = a_sg2.copy()
                    print(f"[polish] SG2: {sc_sg2:.6f}")
    
    # Fallback
    if best_overall is None:
        best_overall = np.ones(128) / 128
        best_overall_sc = 2.0
    
    # Final result
    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"[FINAL] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")
    
    return result
```
