def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1

    A = np.fft.rfft(a, M_fft)

    if M_len is None:

Combined LP-guided + Adam optimization for minimizing 2n*max(conv(a,a))/(sum a)^2.
Uses multiple sequence lengths, adaptive top-k smoothing, and periodic LP-based
restarts to escape local minima.

def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def fast_cross_conv(a, g):
    n_a = len(a)
    n_g = len(g)
    M = n_a + n_g - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    G = np.fft.rfft(g, M_fft)
    return np.fft.irfft(A * G, M_fft)[:M]

def score_from_conv(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def topk_indices(c, k):
    k_act = min(k, len(c))
    if k_act <= 0:
        return np.array([], dtype=int), np.array([]), np.array([])
    if k_act == 1:
        idx = np.array([int(np.argmax(c))])
        return idx, c[idx], np.array([1.0])
    idx = np.argpartition(c, -k_act)[-k_act:]
    vals = c[idx]
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]
    top = float(vals[0])
    if top > 1e-12:
        temp = 0.05 * top
        wts = np.exp((vals - top) / temp)
        wts = wts / wts.sum()
    else:
        wts = np.ones(k_act) / k_act
    return idx, vals, wts

def grad_topk(a, c, k, temp_scale=0.05):
    n = len(a)
    S = float(a.sum())
    idx, vals, wts = topk_indices(c, k)
    # Recompute weights with custom temp
    if len(idx) > 1:
        top = float(vals[0])
        if top > 1e-12:
            temp = temp_scale * top
            wts = np.exp((vals - top) / temp)
            wts = wts / wts.sum()
        else:
            wts = np.ones(len(idx)) / len(idx)
    grad = np.zeros(n)
    for i in range(len(idx)):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi <= 1e-15:
            continue
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start > j_end:
            continue
        grad[j_start:j_end+1] += wi * 2.0 * a[j_start:j_end+1][::-1]
    M = float(np.dot(vals, wts))
    S2 = S * S
    S3 = S2 * S
    result = (2.0 * n) * (grad / S2 - 2.0 * M / S3)
    return result, M, S

def adam_step(a, g, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    m[:] = b1 * m + (1.0 - b1) * g
    v[:] = b2 * v + (1.0 - b2) * g * g
    bc1 = 1.0 - b1 ** t
    bc2 = 1.0 - b2 ** t
    a -= lr * (m / bc1) / (np.sqrt(v / bc2) + eps)

def clamp(a):
    np.clip(a, 0.0, 1000.0, out=a)
    if a.sum() < 1e-3:
        a += 1e-3 / a.size
    return a

def lp_kick(a, M_val, n_max_vars=4096):
    """Solve LP to find a direction that could reduce the peak."""
    n = len(a)
    if n > n_max_vars:
        return None
    
    # Build constraint matrix: A[k, j] = a[j] if j in [max(0,k-n+1), min(k,n-1)]
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            if a[j] > 1e-10:  # Skip zeros for sparsity
                rows.append(k)
                cols.append(j)
                vals.append(a[j])
    
    A = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M_val)
    bounds = [(0.0, None)] * n
    
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 5})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None

def build_inits(n, rng):
    inits = []
    inits.append(np.ones(n))
    # Ramp
    for al in [0.5, 1.0, 2.0]:
        x = np.linspace(0, 1, n)
        inits.append(np.clip(1.0 + al*(2*x-1), 0.01, None))
    # Cosine

            x = np.linspace(0, 1, n)

    # Gaussian-like

    inits.append(np.exp(-0.5*x**2))
    # Random

        if len(a) < n:

        inits.append(np.clip(a[:n], 0.01, None))
    return inits

        score = score_from_conv(n, c, S)

            no_improve = 0

                a = (1-alpha) * best_a + alpha * g

"""
Combined gradient-based optimization with LP-kicks and coordinate descent polish
for minimizing 2n*max(conv(a,a))/(sum a)^2 over non-negative sequences.
Uses adaptive top-k soft-max gradients, multiple sequence lengths (4096, 8192),
periodic linear-programming perturbations to escape local minima, and exact
1D coordinate minimization for final polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix


def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def score_from_conv(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def topk_weights(c, k, temp_scale=0.06):
    k_act = min(k, len(c))
    if k_act <= 0:
        return np.array([], dtype=int), np.array([]), np.array([])
    if k_act == 1:
        idx = np.array([int(np.argmax(c))], dtype=int)
        return idx, c[idx], np.array([1.0])
    idx = np.argpartition(c, -k_act)[-k_act:]
    vals = c[idx]
    order = np.argsort(vals)[::-1]
    idx = idx[order].astype(int)
    vals = vals[order]
    top = float(vals[0])
    if top > 1e-12:
        temp = temp_scale * top
        wts = np.exp((vals - top) / temp)
        s = wts.sum()
        if s > 0:
            wts = wts / s
        else:
            wts = np.ones(k_act) / k_act
    else:
        wts = np.ones(k_act) / k_act
    return idx, vals, wts


def grad_topk(a, c, k, temp_scale=0.06):
    n = len(a)
    S = float(a.sum())
    idx, vals, wts = topk_weights(c, k, temp_scale)
    grad = np.zeros(n)
    for i in range(len(idx)):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi <= 1e-15:
            continue
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start > j_end:
            continue
        grad[j_start:j_end + 1] += wi * 2.0 * a[j_start:j_end + 1][::-1]
    M = float(np.dot(vals, wts))
    S2 = S * S
    S3 = S2 * S
    result = (2.0 * n) * (grad / S2 - 2.0 * M / S3)
    return result, M, S


def adam_step(a, g, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    m[:] = b1 * m + (1.0 - b1) * g
    v[:] = b2 * v + (1.0 - b2) * g * g
    bc1 = 1.0 - b1 ** t
    bc2 = 1.0 - b2 ** t
    a -= lr * (m / bc1) / (np.sqrt(v / bc2) + eps)


def clamp(a):
    np.clip(a, 0.0, 1000.0, out=a)
    if a.sum() < 1e-3:
        a += 1e-3 / a.size
    return a


def lp_kick(a, M_val, max_vars=4096, time_limit=8.0):
    n = len(a)
    if n > max_vars:
        return None
    rows = []
    cols = []
    vals = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            aj = a[j]
            if aj > 1e-12:
                rows.append(k)
                cols.append(j)
                vals.append(aj)
    A = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M_val)
    bounds = [(0.0, None)] * n
    try:
        res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                      method='highs',
                      options={'presolve': True, 'time_limit': time_limit})
        if res.success and res.x is not None and np.sum(res.x) > 1e-8:
            return res.x
    except Exception:
        pass
    return None


def build_inits(n, rng):
    inits = []
    inits.append(np.ones(n))
    x = np.linspace(0, 1, n)
    for al in [0.4, 0.8, 1.5, 3.0]:
        inits.append(np.clip(1.0 + al * (2.0 * x - 1.0), 0.01, None))
    x2 = np.linspace(-1, 1, n)
    inits.append(np.exp(-1.5 * x2 * x2))
    inits.append(np.exp(-0.5 * x2 * x2))
    inits.append(np.exp(-3.0 * x2 * x2))
    for fq in [1, 2, 3]:
        for al in [0.4, 0.8]:
            inits.append(np.clip(1.0 + al * np.cos(2.0 * np.pi * fq * x), 0.01, None))
    for nb in [4, 8, 16]:
        per = max(1, n // nb)
        vals = 0.3 + 0.7 * rng.random(nb)
        a = np.repeat(vals, per)
        if len(a) < n:
            a = np.concatenate([a, vals[:n - len(a)]])
        inits.append(np.clip(a[:n], 0.01, None))
    inits.append(np.clip(0.3 + 0.7 * rng.random(n), 0.01, None))
    inits.append(np.clip(1.0 + 0.5 * rng.standard_normal(n), 0.01, None))
    # Gapped
    for gp in [3, 5]:
        a = np.clip(0.5 + 0.5 * rng.random(n), 0.01, None)
        a[::gp] = 0.01
        inits.append(a)
    return inits


def optimize(n, a0, deadline, rng, lr_init, topk_init, topk_max, temp_scale,
             use_lp=True, lp_interval=300, patience_init=400):
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    c = fast_self_conv(a)
    S = a.sum()
    score = score_from_conv(n, c, S)
    if not np.isfinite(score):
        a = np.ones(n)
        c = fast_self_conv(a)
        score = score_from_conv(n, c, a.sum())
    best_score = score
    best_a = a.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = lr_init
    no_improve = 0
    patience = patience_init
    lp_count = 0
    steps = 0

    while time.monotonic() < deadline:
        c = fast_self_conv(a)
        S = a.sum()
        score = score_from_conv(n, c, S)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # LP kick
        if use_lp and no_improve >= lp_interval and n <= 4096:
            M_val = float(np.max(c))
            g = lp_kick(a, M_val)
            if g is not None and g.sum() > 1e-6:
                g = g / g.sum()
                alpha = 0.2 + 0.5 * rng.random()
                a = (1.0 - alpha) * best_a + alpha * g
                a = clamp(a)
                m[:] = 0.0
                v[:] = 0.0
                t = 0
                no_improve = 0
                lp_count += 1
                continue

        if no_improve >= patience:
            scale = 0.05 + 0.15 * rng.random()
            a = best_a * (1.0 + scale * rng.standard_normal(n))
            a = clamp(a)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            lr = max(0.002, lr * 0.65)
            no_improve = 0
            patience = max(80, patience * 3 // 4)
            continue

        # Adaptive top-k and temperature
        progress = min(1.0, no_improve / 2000.0)
        topk = int(topk_init + progress * (topk_max - topk_init))
        temp = temp_scale * (1.0 + 0.5 * progress)

        g, Mw, Sw = grad_topk(a, c, topk, temp)
        t += 1
        steps += 1
        adam_step(a, g, m, v, t, lr)
        a = clamp(a)

        if t % 10000 == 0:
            lr *= 0.92

    return best_score, best_a, steps


def coord_descent_polish(a, deadline, max_sweeps=100):
    """Exact 1D coordinate descent using the closed-form solution."""
    n = len(a)
    a = a.copy()
    c = fast_self_conv(a)
    S = float(a.sum())
    best_score = score_from_conv(n, c, S)
    best_a = a.copy()

    for sweep in range(max_sweeps):
        if time.monotonic() > deadline:
            break
        improved = False
        # Work on a subset of coordinates per sweep for speed
        idx_order = np.arange(n)
        rng_local = np.random.default_rng(sweep * 7 + 3)
        # For large n, only do a fraction of coordinates per inner pass
        if n > 2048:
            n_per_pass = min(n, 2048)
            idx_order = rng_local.choice(n, n_per_pass, replace=False)

        for j in idx_order:
            if time.monotonic() > deadline:
                break
            aj = a[j]
            if aj < 1e-12:
                continue
            # Compute the contribution of a[j] to each convolution term
            # conv[k] = sum_i a[i]*a[k-i]; changing a[j] to t:
            # conv_new[k] = conv[k] - aj*(a[k-j] valid) + t*(a[k-j] valid)
            # More precisely: conv[k] includes terms a[j]*a[k-j] and a[k-j]*a[j]
            # Let's compute D_k = partial conv[k] / partial a[j] (linear coefficient)
            # and C_k = 1 if 2j == k else 0 (quadratic coefficient for a[j]^2)

            # For the dominant peaks only (top 3)
            top3 = np.argpartition(c, -3)[-3:]
            top3 = top3[np.argsort(c[top3])[::-1]]

            best_t = aj
            best_f = np.max(c)

            for ki in top3:
                ki = int(ki)
                # Coefficients for a[j] in conv[ki]:
                # conv[ki] = sum_i a[i] * a[ki - i] (valid i)
                # The terms involving a[j]: a[j]*a[ki-j] (if ki-j valid)
                #   plus a[ki-j]*a[j] (same term, counted once unless ki-j == j)
                # If ki - j == j (i.e., 2j == ki): contribution is a[j]^2, so C=1
                # If ki - j != j but 0 <= ki-j < n: contribution is 2*a[j]*a[ki-j], so B = a[ki-j]
                # If ki - j out of range: no contribution from a[j]

                kj = ki - j
                if 0 <= kj < n and kj != j:
                    B_k = a[kj]
                    C_k = 0.0
                elif 0 <= kj < n and kj == j:
                    B_k = 0.0
                    C_k = 1.0
                else:
                    B_k = 0.0
                    C_k = 0.0

                # Also need the "cross" term: a[j] also appears as the second factor
                # conv[ki] = sum_i a[i]*a[ki-i]. The term a[ki-j]*a[j] is the same as above
                # when we set i = ki-j. And a[j]*a[ki-j] when i = j. These are the same term.
                # So B_k is correct as above.

                # Constant term (independent of t): conv[ki] - 2*B_k*aj - C_k*aj^2
                # But wait, if kj != j: conv[ki] includes a[j]*a[kj] + a[kj]*a[j] = 2*a[j]*a[kj]
                # If kj == j: conv[ki] includes a[j]*a[j] = a[j]^2
                const = c[ki]
                if B_k != 0.0 and C_k == 0.0:
                    const = const - 2.0 * B_k * aj
                elif C_k == 1.0:
                    const = const - aj * aj

                # f(t) = (C_k * t^2 + 2*B_k*t + const) / (S0 + t)^2
                # where S0 = S - aj
                S0 = S - aj
                # Critical point: t* = (const - B_k*S0) / (C_k*S0 - B_k)
                denom = C_k * S0 - B_k
                if abs(denom) > 1e-15:
                    t_star = (const - B_k * S0) / denom
                else:
                    t_star = aj

                # Candidate t values
                candidates = [0.0, 1000.0, aj]
                if 0.0 < t_star < 1000.0:
                    candidates.append(t_star)

                best_cand_t = aj
                best_cand_val = np.inf
                for tc in candidates:
                    if tc < 0 or tc > 1000:
                        continue
                    val = C_k * tc * tc + 2.0 * B_k * tc + const
                    denom_sq = (S0 + tc) ** 2
                    if denom_sq < 1e-12:
                        continue
                    fv = val / denom_sq
                    if fv < best_cand_val:
                        best_cand_val = fv
                        best_cand_t = tc

                if best_cand_val < best_f - 1e-15:
                    best_f = best_cand_val
                    best_t = best_cand_t

            if abs(best_t - aj) > 1e-10:
                a[j] = best_t
                # Update S
                S = S - aj + best_t
                improved = True

        # Recompute convolution at end of coordinate pass
        c = fast_self_conv(a)
        S = float(a.sum())
        score = score_from_conv(n, c, S)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            if not improved:
                break

    return best_score, best_a

        print("Phase2: %.6f" % best_score)

    # Phase 3: Coordinate descent polish (10% of time)
    if best_a is not None and time.monotonic() < deadline:
        cd_end = min(deadline, time.monotonic() + 0.08 * (total - 12.0))
        sc, aa = coord_descent_polish(best_a, cd_end)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print("Phase3 (CD): %.6f" % best_score)

    # Phase 4: Final gradient polish
    if best_a is not None and time.monotonic() < deadline:
        sc, aa, _ = optimize(best_n, best_a, deadline, rng,
                             lr_init=0.01, topk_init=1, topk_max=8,
                             temp_scale=0.02, use_lp=False,
                             patience_init=600)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
        print("Phase4: %.6f" % best_score)

    if best_a is None:
        best_a = np.ones(2048)

    out = [float(x) for x in best_a]
    verified = float(evaluate_sequence(out))
    print("FINAL: internal=%.6f verified=%.6f n=%d time=%.1fs" %
          (best_score, verified, len(out), time.monotonic() - start))
    return out
