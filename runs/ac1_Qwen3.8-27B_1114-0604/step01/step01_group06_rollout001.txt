_NFFT_X = 0  # for cross-correlation (needs to be larger)


def _init_state(n):

    _N = n
    _NFFT = 1 << (2 * n).bit_length()  # for a*a (length 2n-1)
    _NFFT_X = 1 << (3 * n).bit_length()  # for cross-correlation (length 3n-2)


def _fft_conv(a):
    """Linear convolution a*a via FFT."""
    n = _N
    N = _NFFT
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1].real
    return np.maximum(c, 0.0)


def _score_fast(a):
    """Fast score computation (sum(a) assumed to be 1)."""
    n = _N
    c = _fft_conv(a)
    return 2.0 * n * float(np.max(c))


def _subgradient(a, c, top_k=50):
    """
    Compute subgradient of max(c) using top-k active set.
    Returns g of length n where g[j] = 2 * sum_{k in S} a[k-j] / |S|
    (with boundary handling: a[m] = 0 for m outside [0, n-1]).
    """
    n = _N
    L = 2 * n - 1
    # Select top-k lags
    if top_k >= L:
        S_mask = np.ones(L, dtype=np.float64)
    else:

        c_max = float(np.max(c))
        threshold = c_max - 1e-10
        S_mask = (c >= threshold).astype(np.float64)
        if np.sum(S_mask) == 0:

            S_mask = np.zeros(L, dtype=np.float64)

    # Normalize by |S| and multiply by 2
    s_count = float(np.sum(S_mask))
    if s_count > 0:

    return g


def _mirror_descent_step(a, g, eta):
    """

    a_new = a * weights
    s = float(np.sum(a_new))
    if s < 1e-15 or not np.isfinite(s):

    return a_new / s

    e = np.exp(diff)
    se = float(np.sum(e))
    if se <= 0 or not np.isfinite(se):
        return 1e12, np.zeros(n)
    
    p = e / se
    # Gradient of F = c_max + tau * log(sum exp((c-c_max)/tau))
    # dF/dz = tau * sum_k p_k * dc_k/dz
    # dc_k/dz_i = 2 * a_i * (da/dz)_{k-i} ... this is complex
    # Use: dc_k/d a_j = 2 * a_{k-j} (as derived earlier)
    # da_j/dz_i = a_j * (δ_{ji} - a_i)
    # So dc_k/dz_i = sum_j 2*a_{k-j} * a_j * (δ_{ji} - a_i)
    # = 2*a_{k-i} * a_i - 2*a_i * sum_j a_{k-j} * a_j
    # = 2*a_i * a_{k-i} - 2*a_i * (a*a)_k  ... wait that's not right
    
    # Let me redo: dc_k/dz_i = sum_j (dc_k/da_j) * (da_j/dz_i)
    # = sum_j 2*a_{k-j} * a_j * (δ_{ji} - a_i)
    # = 2*a_{k-i} * a_i - 2*a_i * sum_j a_{k-j} * a_j
    # = 2*a_i * a_{k-i} - 2*a_i * c_k
    
    # So dF/dz_i = sum_k p_k * (2*a_i * a_{k-i} - 2*a_i * c_k)
    # = 2*a_i * sum_k p_k * a_{k-i} - 2*a_i * sum_k p_k * c_k
    # = 2*a_i * (p ⋆ a)_i - 2*a_i * (p · c)
    
    # Where (p ⋆ a)_i = sum_k p_k * a_{k-i} (cross-correlation)
    
    fp = np.fft.rfft(p, N)
    fa_conj = np.conj(np.fft.rfft(a, N))
    # p ⋆ a: use ifft(FFT(p) * conj(FFT(a)))[:n]
    h = np.fft.irfft(fp * fa_conj, N)[:n].real
    pc = float(np.dot(p, c))
    
    grad = 2.0 * a * (h - pc)
    val = c_max + tau * np.log(se)
    
    # Scale for numerical stability
    scale = float(n) * float(n)
    val *= scale
    grad *= scale
    
    # Tiny regularization
    pen = 1e-10
    val += 0.5 * pen * float(np.dot(z, z))
    grad += pen * z
    
    return float(val), grad


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - np.max(z)
    e = np.exp(np.clip(z, -50, 50))
    s = float(np.sum(e))
    if s < 1e-15 or not np.isfinite(s):
        return np.full_like(z, 1.0 / z.size)
    return e / s


def _z_from_a(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    a = np.maximum(a, 1e-15)
    z = np.log(a)
    z = z - np.mean(z)
    return z


def _make_starts(seed=42):
    """Create diverse starting sequences."""
    n = _N
    rng = np.random.default_rng(seed)
    starts = []
    
    # From previous best
    try:
        g = globals().get("height_sequence_1")
        if g is not None:
            g = np.asarray(g, dtype=np.float64)
            if g.size > 1:
                g = np.maximum(g, 0.0)
                s = float(np.sum(g))
                if s > 0.01:
                    a = g / s
                    # Resample to n if needed
                    if a.size != n:
                        x_old = np.linspace(0, 1, a.size)
                        x_new = np.linspace(0, 1, n)
                        a = np.interp(x_new, x_old, a)
                        a = np.maximum(a, 0)
                        a /= np.sum(a)
                    starts.append(a.copy())
    except:
        pass
    
    # Uniform
    starts.append(np.ones(n) / n)
    
    # Linear ramp
    x = np.arange(n, dtype=np.float64) / n
    for A in (0.3, 0.5, 0.7, 1.0, 1.5):
        a = 1.0 + A * (2 * x - 1)
        a = np.maximum(a, 0.01)
        a /= a.sum()
        starts.append(a.copy())
    
    # Sigmoid step
    for pos in (0.3, 0.4, 0.5, 0.6, 0.7):
        for width in (0.02, 0.05, 0.1, 0.2):
            a = 1.0 + 0.8 * (1.0 / (1.0 + np.exp(-(x - pos) / width)))
            a = np.maximum(a, 0.01)
            a /= a.sum()
            starts.append(a.copy())
    
    # Cosine window
    for A in (0.3, 0.5, 0.7):
        a = 1.0 + A * np.cos(2 * np.pi * x)
        a = np.maximum(a, 0.01)
        a /= a.sum()
        starts.append(a.copy())
    
    # Parabolic (more mass at edges)
    for A in (0.3, 0.5, 0.7, 1.0):
        a = 1.0 + A * (1.0 - 4.0 * (x - 0.5) ** 2)
        a = np.maximum(a, 0.01)
        a /= a.sum()
        starts.append(a.copy())
    
    # Inverted parabolic (more mass at center) - probably not useful but try
    for A in (0.3, 0.5):
        a = 1.0 - A * (1.0 - 4.0 * (x - 0.5) ** 2)
        a = np.maximum(a, 0.01)
        a /= a.sum()
        starts.append(a.copy())
    
    # Random smooth (low-frequency)
    for _ in range(5):
        coeffs = rng.standard_normal(10) * 0.3
        a = 1.0
        for m, c in enumerate(coeffs, 1):
            a += c * np.cos(2 * np.pi * m * x + rng.standard_normal())
        a = np.maximum(a, 0.01)
        a /= a.sum()
        starts.append(a.copy())
    
    # Two-level step
    half = n // 2
    for ratio in (0.5, 0.7, 1.0, 1.5, 2.0, 3.0):
        a = np.ones(n)
        a[half:] *= ratio
        a = np.maximum(a, 0.01)
        a /= a.sum()
        starts.append(a.copy())
    
    return starts
