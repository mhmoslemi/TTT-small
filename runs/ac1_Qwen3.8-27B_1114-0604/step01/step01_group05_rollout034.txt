        return float('inf')
    b = fft_convolve(a)
    M = float(np.max(b))
    return float(2.0 * n * M / (s * s))


def build_conv_matrix(a):
    """Build sparse matrix A such that A @ g = convolve(a, g). Shape (2n-1, n)."""
    n = len(a)
    row_list = []
    col_list = []
    val_list = []
    for j in range(n):
        row_list.extend(range(j, j + n))
        col_list.extend([j] * n)
        val_list.extend(a)
    A = sp.coo_matrix((val_list, (row_list, col_list)), shape=(2 * n - 1, n))
    return A.tocsr()


def lp_step(a):
    """One LP iteration: find g >= 0, sum(g)=1 minimizing max(a*g)."""
    n = len(a)
    A = build_conv_matrix(a)

    c = np.zeros(n + 1)
    c[-1] = 1.0

    A_dense = A.toarray()
    A_ub = np.empty((2 * n - 1, n + 1), dtype=np.float64)
    A_ub[:, :n] = A_dense
    A_ub[:, n] = -1.0
    b_ub = np.zeros(2 * n - 1, dtype=np.float64)

    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)

    bounds = [(0.0, None)] * (n + 1)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs',
                  options={'presolve': True, 'primal_feasibility_tolerance': 1e-9})

    if res.status != 0:
        return None, None

    g = np.maximum(res.x[:n], 0.0)
    s = g.sum()
    if s < 1e-15:
        return None, None
    g /= s
    M_prime = float(res.x[-1])
    return g, M_prime


def lp_refine(a, max_iters=2000, tol=1e-11, verbose=True, deadline=None):
    """Iterative LP refinement with line search."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    a = a / np.sum(a)
    b = fft_convolve(a)
    M = float(np.max(b))

    for it in range(max_iters):
        if deadline is not None and time.monotonic() >= deadline:
            break

        g, M_prime = lp_step(a)
        if g is None:
            break

        if M_prime >= M - tol:
            if verbose:
                print(f"    LP converged iter {it}: score={2*n*M:.10f}")
            break

        best_t = 0.0
        best_M = M
        for t in [0.5, 0.3, 0.2, 0.1, 0.05, 0.03, 0.01]:
            a_new = (1.0 - t) * a + t * g
            M_new = float(np.max(fft_convolve(a_new)))
            if M_new < best_M - 1e-15:
                best_M = M_new
                best_t = t

        if best_t == 0.0:
            for t in [0.005, 0.002, 0.001]:
                a_new = (1.0 - t) * a + t * g
                M_new = float(np.max(fft_convolve(a_new)))
                if M_new < best_M - 1e-15:
                    best_M = M_new
                    best_t = t

        if best_t == 0.0:
            break

        a = (1.0 - best_t) * a + best_t * g
        M = best_M

        if verbose and (it % 10 == 0 or it == max_iters - 1):
            score_val = 2.0 * n * M
            print(f"    LP iter {it:4d}: score={score_val:.10f}, t={best_t:.4f}")

    a = np.maximum(a, 0.0)
    a = a / np.sum(a)
    return a, M


def smooth_max_weights(b, beta):
    """Compute softmax weights for smooth max approximation."""
    m = float(np.max(b))
    if m <= 0 or beta <= 0:
        w = np.ones_like(b, dtype=np.float64)
        w /= len(b)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -700.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0 or not np.isfinite(Z):
        w = np.zeros_like(b, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
    else:
        w = e / Z
    G = m + float(np.log(max(Z, 1e-300))) / beta
    return G, w


def grad_convolution(a, w):
    """
    Compute gradient of smooth_max(convolve(a,a)) w.r.t. a.
    Returns vector of length n.
    Uses the relation: d(smooth_max(b))/da_i = sum_k w_k * db_k/da_i
    where db_k/da_i = a_{k-i} + a_{k-i} (with special handling at k=2i).
    
    Efficiently: grad_i = 2 * (a * w_reversed)_i where w_reversed is padded.
    """
    n = len(a)
    # b_k = sum_j a_j * a_{k-j}
    # db_k/da_i = a_{k-i} (if 0<=k-i<n) + a_{i} (if 0<=k-i<n)
    # But more carefully: the cross-correlation approach
    # sum_k w_k * db_k/da_i = sum_k w_k * [a_{k-i} + a_{k-i}] (for k != 2i)
    #                        = sum_k w_k * 2*a_{k-i} (approximately, the k=2i case 
    #                          gives 2*a_i which is the same formula)
    # So grad_i = 2 * sum_k w_k * a_{k-i} (with a_{k-i}=0 if out of range)
    # This is 2 * (reverse(w) * a)_i in the convolution sense.
    # More precisely: sum_k w_k * a_{k-i} = (w * a_rev)_i where a_rev = reverse(a)
    # Actually: let's define c_i = sum_k w_k * a_{k-i} for valid k-i in [0,n-1]
    # This is the cross-correlation of w and a: c_i = (w ⊛ a)_i
    # = sum_k w_k * a_{k-i}
    # In convolution form: if we pad w to length 2n-1 (it already is), then
    # c = convolve(w, reverse(a))[n-1 - i + n - 1]... 
    
    # Simpler: use FFT
    # c_i = sum_k w[k] * a[k-i] for 0<=k<2n-1, 0<=k-i<n
    # = sum_{j} w[j+i] * a[j] for valid j and j+i < 2n-1
    # Let's just do it with a direct FFT approach
    size = 1
    while size < 4 * n:
        size *= 2
    W = np.fft.rfft(w, size)
    A = np.fft.rfft(a, size)
    # c_i = sum_k w_k * a_{k-i} = (w * flip(a))_{something}
    # flip(a) = a[::-1], padded to 2n-1 -> let's just use full dot product approach via FFT
    # Actually: sum_k w[k]*a[k-i] = cross-correlation = ifft(fft(w) * conj(fft(a)))
    # But w has length 2n-1 and a has length n. Let me pad a to 2n-1.
    a_padded = np.zeros(2 * n - 1, dtype=np.float64)
    a_padded[:n] = a
    Ap = np.fft.rfft(a_padded, size)
    corr = np.fft.irfft(W * np.conj(Ap), size)
    # corr[i] = sum_k w[k] * a_padded[k+i]... hmm, need to be careful
    # np.fft.irfft(fft(w)*conj(fft(a))) gives circular cross-correlation
    # corr[m] = sum_k w[k] * a[k+m] (circular)
    # We want c_i = sum_k w[k] * a[k-i] = corr[-i] = corr[size-i]
    # But we need linear, not circular. Since we padded to size >= 4n, it should be fine.
    c = np.zeros(n, dtype=np.float64)
    for i in range(n):
        # c_i = sum_k w[k] * a_{k-i}
        # From corr: corr[m] = sum_k w[k]*a[k+m] (circular, but linear due to padding)
        # c_i = corr[-i] = corr[size - i]
        c[i] = corr[(size - i) % size]
    
    grad = 2.0 * c
    return grad


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if target <= 0:
        return np.zeros(n, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(n, target / n, dtype=np.float64)
    
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0 or not np.isfinite(s):
        q = np.full(n, target / n, dtype=np.float64)
    else:
        q *= target / s
    return q


def gradient_refine(a, max_iters=20000, beta=100000.0, deadline=None, verbose=True):
    """Projected gradient refinement."""
    n = len(a)
    a = np.maximum(a, 0.0).astype(np.float64)
    a = a / np.sum(a)
    
    best_a = a.copy()
    best_M = float(np.max(fft_convolve(a)))
    
    step = 1.0
    no_improve = 0
    last_print = -100
    
    for it in range(max_iters):
        if deadline is not None and time.monotonic() >= deadline:
            break
        
        b = fft_convolve(a)
        M = float(np.max(b))
        
        if M < best_M - 1e-15:
            best_M = M
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if it - last_print >= 100:
            if verbose:
                print(f"    Grad iter {it:5d}: score={2*n*M:.10f}, step={step:.3e}")
            last_print = it
        
        if no_improve > 2000:
            break
        
        G, w = smooth_max_weights(b, beta)
        if not np.isfinite(G):
            break
        
        grad = grad_convolution(a, w)
        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-14:
            break
        
        # Project gradient onto tangent space (sum = 0)
        grad = grad - float(np.mean(grad))
        
        accepted = False
        base = step
        for mult in [1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005]:
            alpha = base * mult
            q = project_simplex(a - alpha * grad, 1.0)
            qb = fft_convolve(q)
            M2 = float(np.max(qb))
            if M2 < M - 1e-14:
                a = q
                M = M2
                if M < best_M - 1e-15:
                    best_M = M
                    best_a = a.copy()
                    no_improve = 0
                accepted = True
                break
        
        if not accepted:
            step *= 0.5
            if step < 1e-8:
                break
        else:
            if step < 2.0:
                step = min(2.0, step * 1.1)
    
    return best_a, best_M


def make_initial_candidates(n, rng):
    """Generate diverse initial candidates of length n."""
    candidates = []
    j = np.arange(n, dtype=np.float64)
    
    # Uniform
    candidates.append(np.ones(n, dtype=np.float64) / n)
    
    # Cosine taper (high center)
    a = np.cos(np.pi * j / (2.0 * n + 1)) ** 2
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    candidates.append(a.copy())
    
    # U-shape (high edges)
    a = 1.0 / (1.0 + 0.8 * np.cos(np.pi * j / n) ** 2)
    a = np.maximum(a, 1e-10)
    a /= a.sum()
    candidates.append(a.copy())
    
    # Power law
    for p in [0.5, 1.5, 2.0]:
        a = 1.0 / (j + 1.0) ** p
        a = np.maximum(a, 1e-10)
        a /= a.sum()
        candidates.append(a.copy())
    
    # Symmetric power law
        a = 1.0 / (np.abs(j - n / 2.0) + 1.0) ** p
        a = np.maximum(a, 1e-10)
        a /= a.sum()
        candidates.append(a.copy())
    
    # Sinusoidal modulations
    for k in [1, 2, 3]:
        for eps in [0.1, 0.3, 0.5]:
            a = 1.0 + eps * np.cos(2.0 * np.pi * k * j / n)
            a = np.maximum(a, 1e-10)
            a /= a.sum()
            candidates.append(a.copy())
    
    # Anti-symmetric (odd) modulation
    for k in [1, 2, 3]:
        for eps in [0.5, 0.8]:
            a = 1.0 + eps * np.sin(2.0 * np.pi * k * j / n)
            a = np.maximum(a, 1e-10)
            a /= a.sum()
            candidates.append(a.copy())
    
    # Random
    for _ in range(4):
        a = rng.random(n) + 0.01
        a /= a.sum()
        candidates.append(a.copy())
    
    # Block structures
    for nblocks in [2, 4, 8]:
        if n % nblocks == 0:
            blk = n // nblocks
            for _ in range(3):
                w = rng.random(nblocks) + 0.1
                w /= w.sum()
                a = np.repeat(w, blk).astype(np.float64)
                a /= a.sum()
                candidates.append(a.copy())
    
    return candidates


def try_lp_gradient(a, deadline, verbose=True):
    """Apply LP refinement followed by gradient refinement."""
    n = len(a)
    t0 = time.monotonic()
    
    print(f"  Starting LP+Grad on n={n}")
    
    # Phase 1: LP refinement
    lp_time = max(30.0, 0.5 * (deadline - t0))
    lp_deadline = t0 + lp_time
    a, M = lp_refine(a, max_iters=5000, deadline=lp_deadline, verbose=verbose)
    score_val = 2.0 * n * M
    print(f"  After LP: score={score_val:.10f} (time={time.monotonic()-t0:.1f}s)")
    
    # Phase 2: Gradient refinement on remaining time
    grad_deadline = deadline - 5.0
    if time.monotonic() < grad_deadline:
        a, M = gradient_refine(a, max_iters=50000, deadline=grad_deadline, verbose=verbose)
        score_val = 2.0 * n * M
        print(f"  After Grad: score={score_val:.10f} (time={time.monotonic()-t0:.1f}s)")
    
    a = np.maximum(a, 0.0)
    a = a / np.sum(a)
    return a, 2.0 * n * float(np.max(fft_convolve(a)))


def official_score(seq):
    """Call the authoritative evaluator."""
    try:
        val = evaluate_sequence(seq)
        val = float(val)
    except Exception:
        return float('inf')
    if np.isfinite(val):
        return val
    return float('inf')


def propose_candidate(seed=42, budget_s=2400):
    """
    Search for the best non-negative sequence minimizing the evaluation metric.
    Uses LP-based iterative refinement with multiple initial points and sequence lengths.
    """
    start_time = time.monotonic()
    reserve = 30.0
    deadline = start_time + max(0.0, float(budget_s) - reserve)
    
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    
    rng = np.random.default_rng(seed)
    
    best_a = None
    best_score = float('inf')
    
    # Try multiple sequence lengths
    n_values = [512, 1024, 2048, 4096]
    
    # Prioritize based on available time
    elapsed = time.monotonic() - start_time
    remaining = deadline - time.monotonic()
    
    if remaining < 300:
        n_values = [512, 1024]
    elif remaining < 900:
        n_values = [512, 1024, 2048]
    
    # Also try to use the previous best if available
    try:
        prev = height_sequence_1
        if isinstance(prev, list) and len(prev) > 0:
            prev_arr = np.array([float(max(0, x)) for x in prev], dtype=np.float64)
            if np.all(np.isfinite(prev_arr)) and prev_arr.sum() > 0:
                prev_arr /= prev_arr.sum()
                sp_val = 2.0 * len(prev_arr) * float(np.max(fft_convolve(prev_arr)))
                print(f"Previous sequence: n={len(prev_arr)}, score={sp_val:.8f}")
                if sp_val < best_score:
                    best_score = sp_val
                    best_a = prev_arr.copy()
                # Add to candidates for this length
                n_prev = len(prev_arr)
                if n_prev not in n_values:
                    n_values.append(n_prev)
    except Exception:
        pass
    
    # Sort n_values to try smaller ones first (faster LPs)
    n_values = sorted(set(n_values))
    
    # Time allocation: give more time to larger n if we have budget
    total_time = deadline - time.monotonic()
    n_weights = {}
    for n in n_values:
        # Weight proportional to n but with diminishing returns
        n_weights[n] = min(n, 4096) ** 0.7
    
    total_weight = sum(n_weights.values())
    
    for n in n_values:
        if time.monotonic() >= deadline - 10:
            break
        
        frac = n_weights[n] / total_weight
        n_time = frac * total_time
        n_deadline = max(time.monotonic() + 10, time.monotonic() + n_time)
        
        print(f"\n=== Trying n={n} (time budget: {n_time:.0f}s) ===")
        
        # Generate initial candidates
        candidates = make_initial_candidates(n, rng)
        
        # If we have a previous sequence of this length, include it
        # (already handled above via best_a tracking)
        
        # Evaluate all candidates and pick the top few
        scored = []
        for ca in candidates:
            ca = np.maximum(ca, 0.0).astype(np.float64)
            s = float(np.sum(ca))
            if s > 0:
                ca /= s
            sc = 2.0 * n * float(np.max(fft_convolve(ca)))
            if np.isfinite(sc):
                scored.append((sc, ca))
        
        scored.sort(key=lambda x: x[0])
        
        # Take top 3-5 candidates for LP refinement
        n_top = min(5, len(scored))
        if n_top == 0:
            continue
        
        for idx in range(n_top):
            if time.monotonic() >= n_deadline - 5:
                break
            
            ca = scored[idx][1].copy()
            print(f"  Candidate {idx}: initial score={scored[idx][0]:.8f}")
            
            # Time for this candidate
            cand_dl = max(time.monotonic() + 5, n_deadline - 5 - (n_top - 1 - idx) * 10.0)
            
            a_refined, sc = try_lp_gradient(ca, cand_dl, verbose=True)
            
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_a = a_refined.copy()
                print(f"  NEW BEST: n={n}, score={best_score:.10f}")
        
        total_weight = max(1e-9, total_weight - n_weights[n])
    
    # Final: try a bit more gradient refinement on the best sequence if time remains
    if best_a is not None and time.monotonic() < deadline - 10:
        n = len(best_a)
        print(f"\n=== Final polish on n={n} ===")
        a_polished, M = gradient_refine(best_a, max_iters=100000,
                                         beta=200000.0,
                                         deadline=deadline - 5.0, verbose=True)
        sc = 2.0 * n * M
        if sc < best_score:
            best_score = sc
            best_a = a_polished
            print(f"  Polished score={best_score:.10f}")
    
    if best_a is None:
        # Fallback: uniform
        best_a = np.ones(1024, dtype=np.float64) / 1024
        best_score = 2.0
    
    # Convert to list and verify
    best_a = np.maximum(best_a, 0.0)
    s = float(np.sum(best_a))
    if s > 0 and np.isfinite(s):
        best_a /= s
    
    final_seq = [float(x) for x in best_a]
    
    # Verify with official evaluator
    off = official_score(final_seq)
    print(f"\nFinal: n={len(final_seq)}, fast_score={best_score:.8f}, official={off:.8f}")
    
    if not np.isfinite(off) or off > 2.0 + 1e-9:
        # Fallback to uniform
        fb = [1.0] * 1024
        return fb
    
    return final_seq
