            a[j] = x0 + delta_best
            c[j:j+n] += 2.0 * delta_best * a
            c[2*j] += delta_best * delta_best
            S += delta_best
            updates += 1
    
    return updates, S


def gradient_step_logsumexp(a, c, S, tau, lr):
    """
    One Adam-like gradient step using log-sum-exp surrogate.
    Returns updated a.
    """
    n = len(a)
    # Compute softmax-like weights over conv peaks
    c_clipped = np.clip(c, 0, None)
    max_c = c_clipped.max()
    if max_c < 1e-15:
        return a
    
    exp_c = np.exp((c_clipped - max_c) / tau)
    p = exp_c / exp_c.sum()
    
    # Gradient: d log(sum exp(c_k/tau))/d a_j = (1/tau) * sum_k p_k * 2*a[k-j] (for valid k-j)
    # This is (2/tau) * (p correlated with a)
    # Specifically, for each j: grad_j = (2/tau) * sum_{k: 0<=k-j<n} p_k * a[k-j]
    # = (2/tau) * sum_{m: 0<=m<n} p[j+m] * a[m]
    # This is a cross-correlation of p with a.
    
    # Using FFT:
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    P = np.fft.rfft(p, M_fft)
    A = np.fft.rfft(a, M_fft)
    
    # The correlation (p ⊛ a) _ j = sum_m p[j+m] * a[m]
    # This is the cross-correlation, which can be computed via:
    # IFFT( conj(IFFT(a)) * IFFT(p) ) ... 
    # Actually, corr[j] = sum_m p[j+m] * a[m] for m=0..n-1
    # This is (p * flip(a))[j + n - 1] where * is convolution
    # Or equivalently, it's the cross-correlation.
    
    # Let me just compute it directly for clarity (O(n^2) but n is small enough)
    # Actually for n=2048, O(n^2) = 4M which is fine.
    
    # But let me use the FFT approach:
    # corr[j] = sum_{m=0}^{n-1} p[j+m] * a[m]
    # = (p convolved with reversed a)[j + n - 1] ... 
    
    # Actually, the simplest: the gradient of sum_k p_k * c_k w.r.t. a_j
    # = sum_k p_k * d(c_k)/d(a_j)
    # = sum_k p_k * 2*a[k-j] (for 0 <= k-j < n)
    # = 2 * sum_{m=0}^{n-1} p[j+m] * a[m]
    #   (where m = k-j, so k = j+m, and we need 0 <= j+m < 2n-1, which is always true for j in [0,n-1] and m in [0,n-1])
    
    # So grad_j = (2/tau) * sum_m p[j+m] * a[m] / S ... wait, we also need the chain rule for the 1/S^2 factor.
    
    # The objective is f = 2n * LSE(c, tau) / S^2 (approximately, for small tau)
    # Actually, LSE(c, tau) = tau * log(sum exp(c_k/tau)) ≈ max(c) for small tau.
    # So f ≈ 2n * LSE / S^2.
    
    # df/da_j = (2n/S^2) * dLSE/da_j - (2n * LSE / S^3) * dS/da_j
    # = (2n/S^2) * dLSE/da_j - (2n * LSE / S^3) * 1
    
    # dLSE/da_j = (1/tau) * sum_k p_k * d(c_k)/da_j
    # = (1/tau) * sum_k p_k * 2*a[k-j]
    # = (2/tau) * sum_{m=0}^{n-1} p[j+m] * a[m]
    
    # So: grad_j = (2n/S^2) * (2/tau) * corr_j - (2n * LSE / S^3)
    # where corr_j = sum_m p[j+m] * a[m]
    
    # For the correlation, use FFT:
    # corr = ifft(fft(p, M_fft) * conj(fft(a, M_fft))) ... no, that's not right either.
    
    # Let me think again. corr[j] = sum_{m=0}^{n-1} p[j+m] * a[m]
    # This is the cross-correlation of p and a, evaluated at lag j.
    # In numpy: np.correlate(p, a, mode='full')[n-1+j] ... 
    # Actually: np.correlate(p, a, mode='full')[n-1+j] = sum_m p[j+m] * a[m]
    # Let me verify: np.correlate(a, b, 'full')[k] = sum_i a[i+k] * conj(b[i]) (for the appropriate range)
    # Hmm, numpy's correlate definition:
    # c[k] = sum_n a[n+k] * conj(b[n])  (for 'full' mode)
    # So np.correlate(p, a, 'full')[j] = sum_n p[n+j] * conj(a[n]) = sum_n p[n+j] * a[n] (since a is real)
    # = sum_n p[n+j] * a[n] for n where both are defined.
    # p has length 2n-1, a has length n.
    # For k = j: sum_n p[n+j] * a[n] where n+j < 2n-1 and n < n.
    # = sum_{n=0}^{n-1} p[n+j] * a[n] (since n < n implies n+j < 2n-1 for j < n-1)
    # This is exactly what we want!
    
    # So corr = np.correlate(p, a, 'full')[0:n]
    # But this is O(n * (2n-1)) = O(n^2) which is fine for n=2048.
    
    # Actually, for n=2048, np.correlate would be slow (it's not FFT-based by default).
    # Let me use FFT:
    # corr[j] = sum_m p[j+m] * a[m]
    # = sum_m p[m+j] * a[m]
    # This is the cross-correlation (p ⋆ a)[j]
    # = ifft(fft_p * conj(fft_a))[j] ... let me check.
    
    # ifft(fft_p * conj(fft_a))[k] = sum_n p[n] * a[n-k] (circular, with zero-padding)
    # Hmm, that's not quite right. Let me use the standard formula:
    # (p ⋆ a)[j] = sum_m p[m+j] * a[m]
    # = (p * a_flipped)[j + n - 1]  where a_flipped[m] = a[n-1-m]
    # = ifft(fft(p * a_flipped))[j + n - 1]
    
    # Or more directly:
    # Let's define b[m] = a[n-1-m] (flip a)
    # Then (p * b)[k] = sum_m p[m] * b[k-m] = sum_m p[m] * a[n-1-k+m]
    # For k = j + n - 1: (p * b)[j+n-1] = sum_m p[m] * a[n-1-(j+n-1)+m] = sum_m p[m] * a[m-j]
    # That's sum_m p[m] * a[m-j], which is different from what we want.
    
    # Let me try: (b * p)[k] = sum_m b[m] * p[k-m] = sum_m a[n-1-m] * p[k-m]
    # For k = j + n - 1: sum_m a[n-1-m] * p[j+n-1-m]
    # Let l = n-1-m, so m = n-1-l:
    # = sum_l a[l] * p[j+n-1-(n-1-l)] = sum_l a[l] * p[j+l]
    # = sum_l a[l] * p[j+l] = corr[j] ✓
    
    # So corr[j] = (b * p)[j + n - 1] where b = flip(a)
    # = ifft(fft(b) * fft(p))[j + n - 1]
    
    b = a[::-1].copy()
    B = np.fft.rfft(b, M_fft)
    P = np.fft.rfft(p, M_fft)
    corr_full = np.fft.irfft(B * P, M_fft)
    corr = corr_full[:n]  # corr[j] for j = 0..n-1
    
    # Actually wait, I need to double-check the indexing.
    # Ifft(B*P) gives the linear convolution b * p (zero-padded to M_fft).
    # (b * p)[k] = sum_m b[m] * p[k-m] for valid m, k-m.
    # b has length n, p has length 2n-1.
    # (b*p) has length n + 2n-1 - 1 = 3n-2.
    # We want (b*p)[j+n-1] for j = 0..n-1, i.e., indices n-1 to 2n-2.
    
    corr = np.fft.irfft(B * P, M_fft)[n-1:2*n-1]  # should give corr[0:n]
    
    # Hmm, let me just verify numerically for small n.
    # For now, let me use the direct computation for correctness and switch to FFT later.
    
    # Direct computation (O(n^2)):
    # corr[j] = sum_{m=0}^{n-1} p[j+m] * a[m]
    # We can compute this as: for each m, add p[j+m]*a[m] to corr[j] for all j.
    # That's: corr += a[:,None] * p[None, :] ... no, that's not right.
    # corr[j] = sum_m p[j+m] * a[m]
    # = <p[j:j+n], a>  (dot product)
    
    # This is a sliding dot product, which can be computed as:
    # corr = np.array([np.dot(p[j:j+n], a) for j in range(n)])
    # But that's O(n^2). For n=2048, it's 4M operations, which is fine.
    
    # Actually, let me use the FFT approach and verify:
    # The convolution (b * p) where b = a[::-1]:
    # (b * p)[k] = sum_{m=0}^{n-1} b[m] * p[k-m] = sum_{m=0}^{n-1} a[n-1-m] * p[k-m]
    # Let l = n-1-m: = sum_{l=0}^{n-1} a[l] * p[k-(n-1-l)] = sum_{l=0}^{n-1} a[l] * p[k-n+1+l]
    # For k = j+n-1: = sum_{l=0}^{n-1} a[l] * p[j+l] = corr[j] ✓
    
    # So corr[j] = (b*p)[j+n-1]
    # The full convolution b*p has length (n-1) + (2n-2) + 1 = 3n-2.
    # We need indices n-1 to 2n-2 (inclusive), which is n values. ✓
    
    # Using FFT with zero-padding to M_fft >= 3n-2:
    M_fft = 1 << (3*n-3).bit_length()  # ensure M_fft >= 3n-2
    B = np.fft.rfft(b, M_fft)
    P = np.fft.rfft(p, M_fft)
    conv_bp = np.fft.irfft(B * P, M_fft)
    corr = conv_bp[n-1:2*n-1]
    
    # Gradient
    LSE = tau * np.log(np.sum(np.exp((c_clipped - max_c) / tau)) ) + max_c
    # Actually, LSE = max_c + tau * log(sum exp((c - max_c)/tau))
    
    coeff1 = 2.0 * n / (S * S)
    coeff2 = 2.0 * n * LSE / (S * S * S)
    
    grad = coeff1 * (2.0 / tau) * corr - coeff2
    
    # Clip to non-negative (projected gradient)
    a_new = np.clip(a - lr * grad, 0, 1000.0)
    
    return a_new
