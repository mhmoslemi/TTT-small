Improved low-peak step function search using multi-frequency initialization,
aggressive subgradient optimization with line search, and diverse restarts.

import numpy as np
import time
import sys

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))

def _normalize(a):
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.clip(arr, 0.0, None)
    s = float(arr.sum())
    if s < 1e-15:
        return np.ones_like(arr) / arr.size
    return arr / s

def _project_simplex(v):
    """Euclidean projection onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _subgradient_step(a, c, K=30):
    """Compute the subgradient of max(c) at the top-K lags."""
    n = a.size
    c_len = c.size
    if c_len <= K:
        active = np.arange(c_len)
    else:
        active = np.argpartition(c, -K)[-K:]
    
    grad = np.zeros(n, dtype=np.float64)
    cnt = len(active)
    if cnt == 0:
        return grad
    
    for k in active:
        k = int(k)
        # c[k] = sum_i a[i] * a[k-i]
        # dc[k]/da[i] = 2*a[k-i] if 0 <= k-i <= n-1
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        if i_max >= i_min:
            idx = np.arange(i_min, i_max + 1, dtype=np.int64)
            grad[idx] += 2.0 * a[k - idx]
    
    grad /= float(cnt)
    return grad

def _make_multifreq(n, A1, A2, phi2, A3=0.0, phi3=0.0):

    a = base + sig
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n)
        s = n
    return a / s

    scan_deadline = t0 + min(120.0, total * 0.05)
    scan_a, scan_score = _scan_starting_points(scan_deadline)
    if scan_a is not None:
        scan_score_norm = _internal_score(scan_a)
        if scan_score_norm < best_score:
            best_score = scan_score_norm
            best_overall = scan_a.copy()
            print(f"[phase1] improved to {best_score:.8f}")

    if best_overall is None:
        best_overall = _make_sine_seq(2048, 0.78, 1)
        best_score = _internal_score(best_overall)
        print(f"[phase1] fallback sine score={best_score:.8f}")

    # Phase 2: Main optimization
    # Use the time to optimize at the current best length
    elapsed = time.monotonic() - t0
    main_budget = min(hard_deadline - elapsed - 200.0, total * 0.7)
    main_budget = max(main_budget, 60.0)
    print(f"[phase2] main optimization budget={main_budget:.0f}s, n={best_overall.size}")

    opt_a = _optimize_subgradient(best_overall, main_budget, K_start=20, K_end=60, verbose=True)
    opt_score = _internal_score(opt_a)
    if opt_score < best_score:
        best_score = opt_score
        best_overall = opt_a.copy()
        print(f"[phase2] improved to {best_score:.8f}")
    else:
        print(f"[phase2] no improvement, keeping {best_score:.8f}")

    # Phase 3: Try other lengths
    elapsed = time.monotonic() - t0
    remaining = hard_deadline - elapsed - 30.0
    if remaining > 60:
        print(f"[phase3] trying other lengths, remaining={remaining:.0f}s")
        current_n = best_overall.size
        target_lengths = []
        for nl in [4096, 8192, 16384, 2048]:
            if nl != current_n:
                target_lengths.append(nl)

        per_len_budget = remaining / max(len(target_lengths), 1) - 5.0
        per_len_budget = max(per_len_budget, 20.0)

        for nl in target_lengths:
            if time.monotonic() > hard_deadline - 30:
                break
            rem = hard_deadline - 30 - time.monotonic()
            if rem < 25:
                break
            a_try, sc_try = _try_length(best_overall, nl, min(per_len_budget, rem - 10), K=40, verbose=True)
            if sc_try < best_score:
                best_score = sc_try
                best_overall = a_try.copy()
                print(f"[phase3] new best: n={nl} score={best_score:.8f}")

    # Final: one more short polish
    if time.monotonic() < hard_deadline - 10:
        remaining = hard_deadline - 10 - time.monotonic()
        if remaining > 30:
            print(f"[final] polish budget={remaining:.0f}s")
            opt_a = _optimize_subgradient(best_overall, remaining, K_start=30, K_end=80, verbose=False)
            sc = _internal_score(opt_a)
            if sc < best_score:
                best_score = sc
                best_overall = opt_a.copy()
                print(f"[final] polished to {best_score:.8f}")

    # Return
    final = _normalize(best_overall)
    final_score = _internal_score(final)
    print(f"[result] n={final.size} internal_score={final_score:.8f} total_time={time.monotonic()-t0:.1f}s")

    # Validate
    try:
        real = float(evaluate_sequence([float(x) for x in final]))
        print(f"[validate] real_score={real:.8f}")
        if np.isfinite(real):
            return [float(x) for x in final]
    except Exception as e:
        print(f"[validate] error: {e}")

    # Fallback
    fb = _make_sine_seq(2048, 0.78, 1)
    return [float(x) for x in fb]
