def solve_lp(f, M):

    # Build the convolution matrix T (2n-1) x n

    T = np.zeros((2*n-1, n))

    prob.solve(solver=cp.ECOS, verbose=False)
    if b_var.value is None:
        return f.copy()
    return np.maximum(b_var.value, 0)

"""
Step function optimizer minimizing 2*n*max(f*f)/(sum f)^2.
Multi-stage approach: mirror descent with active-set tracking, simulated annealing
with structured moves, LP-inspired coordinate refinement, and iterative re-optimization.
"""

import numpy as np
from scipy.signal import correlate
import time


def _convolve_fft(f):
    """Compute full linear convolution f*f via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _obj_from_conv(c, n):
    """Compute objective from precomputed convolution."""
    return float(2.0 * n * c.max())


def _objective(f):
    """Compute 2*n*max(f*f)/(sum f)^2 assuming sum(f)=1."""
    n = len(f)
    if f.sum() < 0.01:
        return np.inf
    c = _convolve_fft(f)
    return float(2.0 * n * c.max() / (f.sum() ** 2))


def _mirror_descent(f_init, deadline, rng, label=""):
    """
    Mirror descent (exponentiated gradient) with soft-max active-set tracking.
    Uses adaptive step size and perturbation-based restarts.
    """
    n = len(f_init)
    f = np.maximum(f_init, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f /= s

    c = _convolve_fft(f)
    M = c.max()
    best_f = f.copy()
    best_score = float(2.0 * n * M)

    # Temperature schedule: start broad, narrow to sharp active set
    T_init = M * 0.03
    T_min = M * 1e-9
    T = T_init
    total_iters_estimate = 50000
    cool_factor = (T_min / T_init) ** (1.0 / total_iters_estimate)

    step = 0.8
    min_step = 0.005
    step_decay = 0.99998

    stuck_threshold = 12
    stuck_count = 0
    it = 0

    while time.monotonic() < deadline:
        # Soft-max weights
        cs = c / T
        cs -= cs.max()
        p = np.exp(cs)
        p_sum = p.sum()
        if p_sum < 1e-300:
            p = np.zeros(2 * n - 1)
            p[int(np.argmax(c))] = 1.0
        else:
            p /= p_sum

        # Gradient: 2 * cross-correlation of p and f
        grad = 2.0 * correlate(p, f, mode='valid')

        # Mirror descent step
        gmax, gmin = grad.max(), grad.min()
        span = gmax - gmin
        if span < 1e-15:
            span = 1e-15
        eta = step / span
        log_f = np.log(f + 1e-300) - eta * grad
        log_f -= log_f.max()
        f_new = np.exp(log_f)
        f_sum = f_new.sum()
        if f_sum < 1e-300:
            break
        f_new /= f_sum

        # Evaluate
        c_new = _convolve_fft(f_new)
        M_new = c_new.max()
        score_new = float(2.0 * n * M_new)

        if score_new < best_score - 1e-10:
            best_score = score_new
            best_f = f_new.copy()
            stuck_count = 0
            f = f_new
            c = c_new
            M = M_new
        else:
            stuck_count += 1
            if stuck_count >= stuck_threshold:
                # Tiered perturbation
                if stuck_count < stuck_threshold * 2:
                    mag = 0.03 / np.sqrt(n)
                else:
                    mag = 0.1 / np.sqrt(n)
                f = f + mag * rng.standard_normal(n)
                f = np.maximum(f, 0.0)
                fs = f.sum()
                if fs < 1e-10:
                    f = np.ones(n) / n
                else:
                    f /= fs
                c = _convolve_fft(f)
                M = c.max()
                stuck_count = 0

        # Anneal
        T = max(T * cool_factor, T_min)
        step = max(step * step_decay, min_step)
        it += 1

    return best_f, best_score


def _simulated_annealing(f_init, deadline, rng, label=""):
    """
    Simulated annealing with structured moves:
    - Gaussian perturbation on random subsets
    - Mass transfer between positions
    - Block operations
    - Spectrum-inspired moves
    """
    n = len(f_init)
    f = np.maximum(f_init, 0.0).astype(np.float64).copy()
    f /= f.sum()
    c = _convolve_fft(f)
    M = c.max()
    score = float(2.0 * n * M)
    best_f = f.copy()
    best_score = score

    T = 0.003
    T_min = 5e-6
    cool = 0.99997
    base_scale = 0.05 / np.sqrt(n)
    it = 0

    while time.monotonic() < deadline:
        r = rng.random()
        f_new = f.copy()

        if r < 0.35:
            # Gaussian perturbation on random subset
            ks = int(rng.integers(1, max(2, n // 100)))
            idx = rng.integers(0, n, size=ks)
            f_new[idx] += base_scale * rng.standard_normal(ks)
        elif r < 0.55:
            # Mass transfer: move from high-value to low-value position
            i = int(rng.integers(0, n))
            j = int(rng.integers(0, n))
            if i != j:
                frac = rng.random() * 0.3
                amt = f[i] * frac
                f_new[i] -= amt
                f_new[j] += amt
        elif r < 0.70:
            # Block swap
            blen = int(rng.integers(2, min(n // 20, 200)))
            b1 = int(rng.integers(0, n - blen))
            b2 = int(rng.integers(0, n - blen))
            if b1 != b2:
                tmp = f_new[b1:b1+blen].copy()
                f_new[b1:b1+blen] = f_new[b2:b2+blen]
                f_new[b2:b2+blen] = tmp
        elif r < 0.85:
            # Spectrum-inspired: add a sinusoidal perturbation
            freq = rng.uniform(0.5, 5.0)
            phase = rng.uniform(0, 2 * np.pi)
            pert = base_scale * 0.5 * np.sin(2 * np.pi * freq * np.arange(n) / n + phase)
            f_new += pert
        else:
            # Large restart (rare)
            f_new = f + (base_scale * 3) * rng.standard_normal(n)

        f_new = np.maximum(f_new, 0.0)
        sn = f_new.sum()
        if sn < 1e-10:
            T = max(T * cool, T_min)
            it += 1
            continue
        f_new /= sn

        c_new = _convolve_fft(f_new)
        M_new = c_new.max()
        score_new = float(2.0 * n * M_new)
        delta = score_new - score

        if delta < 0:
            f = f_new
            c = c_new
            M = M_new
            score = score_new
            if score < best_score:
                best_score = score
                best_f = f.copy()
        else:
            accept_prob = np.exp(-delta / max(T, 1e-10))
            if rng.random() < accept_prob:
                f = f_new
                c = c_new
                M = M_new
                score = score_new

        T = max(T * cool, T_min)
        it += 1

    return best_f, best_score

def _lp_inspired_step(f, M, n, rng):
    """
    LP-inspired direction finding:
    For the peak lag k*, find positions where f contributes least,
    and propose increasing them. This is a greedy approximation of
    the LP: max sum(b) s.t. (f*b)[k] <= M, b >= 0.
    """
    c = _convolve_fft(f)
    k_star = int(np.argmax(c))

    # For lag k*, the "cost" of increasing f[j] is f[k_star - j]
    # We want to increase f[j] where this cost is smallest
    costs = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            costs[j] = f[idx]
        else:
            costs[j] = 0.0  # no contribution to this lag

    # Find the position with lowest cost (most "room")
    # Also check a few other peak lags
    all_costs = costs.copy()
    other_peaks = np.where(c >= M * 0.95)[0]
    for k in other_peaks:
        if k == k_star:
            continue
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                all_costs[j] = max(all_costs[j], f[idx])
            else:
                all_costs[j] = max(all_costs[j], 0.0)

    # Sort by cost: lowest cost positions are best to increase
    sorted_positions = np.argsort(all_costs)

    return sorted_positions, all_costs

        print(f"    -> New best: {best_score:.6f}")

    # Phase 3: Second mirror descent round from best
    print(f"\n{'='*50}")
    print(f"Phase 3: Mirror Descent (Round 2)")
    print(f"{'='*50}")
    time_p3 = budget_s * 0.15
    md2_deadline = min(time.monotonic() + time_p3, deadline)
    f_md2, sc_md2 = _mirror_descent(best_f, md2_deadline, rng, label="md2")
    if sc_md2 < best_score:
        best_score = sc_md2
        best_f = f_md2.copy()
        print(f"    -> New best: {best_score:.6f}")

    # Phase 4: LP-inspired polish
    print(f"\n{'='*50}")
    print(f"Phase 4: LP-Inspired Polish")
    print(f"{'='*50}")
    time_p4 = budget_s * 0.12
    lp_deadline = min(time.monotonic() + time_p4, deadline)
    f_lp, sc_lp = _lp_polish(best_f, lp_deadline, rng, label="lp")
    if sc_lp < best_score:
        best_score = sc_lp
        best_f = f_lp.copy()
        print(f"    -> New best: {best_score:.6f}")

    # Phase 5: Final short mirror descent
    print(f"\n{'='*50}")
    print(f"Phase 5: Final Refinement")
    print(f"{'='*50}")
    remaining = max(30.0, deadline - time.monotonic())
    time_p5 = remaining * 0.5
    md3_deadline = min(time.monotonic() + time_p5, deadline)
    f_md3, sc_md3 = _mirror_descent(best_f, md3_deadline, rng, label="md3")
    if sc_md3 < best_score:
        best_score = sc_md3
        best_f = f_md3.copy()
        print(f"    -> New best: {best_score:.6f}")

    # Phase 6: Coordinate polish
    print(f"\n{'='*50}")
    print(f"Phase 6: Coordinate Polish")
    print(f"{'='*50}")
    remaining = max(20.0, deadline - time.monotonic())
    time_p6 = remaining * 0.6
    cp_deadline = min(time.monotonic() + time_p6, deadline)
    f_cp, sc_cp = _coordinate_polish(best_f, cp_deadline, rng, label="cp")
    if sc_cp < best_score:
        best_score = sc_cp
        best_f = f_cp.copy()
        print(f"    -> New best: {best_score:.6f}")

    # Final verification with evaluate_sequence
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = evaluate_sequence(best_f.tolist())
    print(f"\nVerified final score: {final_score:.6f}")

    if final_score < best_score or not np.isfinite(best_score):
        best_score = final_score

    print(f"\n{'='*50}")
    print(f"=== FINAL SCORE: {best_score:.6f} ===")
    print(f"{'='*50}")
    return [float(x) for x in best_f]
