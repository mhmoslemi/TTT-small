```python
from scipy.optimize import linprog

def full_conv(a):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def build_lp_matrix(f, n):
    """Build the (2n-1) x n constraint matrix for the LP."""
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            A[k, j] = f[k - j]
    return A
```

For non-negative sequences, the problem is more constrained. The uniform sequence gives score 2. We need to do better.

I think the LP-based approach is the way to go. Let me implement it carefully.

One more optimization: instead of solving the full LP from scratch each time, I can use **warm starting**. HiGHS supports providing an initial solution. I can pass the previous `g` as the initial solution.

Actually, `scipy.optimize.linprog` doesn't directly support warm starting with HiGHS in a clean way through the Python API. Let me just solve from scratch each time; for n=128-512, it should be fast enough.

Let me also think about **how many iterations the LP method needs**. Each iteration involves:
1. Compute `a*a` (FFT, O(n log n))
2. Build LP matrix (O(n^2) but with simple structure)
3. Solve LP (depends on solver, but typically fast for this size)
4. Blend and normalize (O(n))

For n=256, each iteration might take ~5-10ms. In 2400s, I could do ~240,000 iterations. That's way more than enough for convergence.

For n=1024, each LP might take ~50-200ms. In 2400s, ~12,000-48,000 iterations.

I think the main challenge is not convergence speed but **escaping local optima**. The LP method can get stuck if the current `a` is at a fixed point. So I need good restarts and annealing.

**Annealing strategy:**
- Periodically (every N iterations), increase `M` by a factor (e.g., 1.1x). This allows `g` to be more different from `a`, potentially escaping the local optimum.
- Use a schedule: `M_mult = 1 + alpha * (1 - exp(-it/T))` or just periodic jumps.

"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based iterative
optimization (alternating: fix a, solve LP for g; blend a and g).
Multiple sequence lengths, diverse initializations, annealing.
"""

import numpy as np
from scipy.optimize import linprog
import time


def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def score_of(a):
    """Compute 2*n*max(a*a)/sum(a)^2."""
    n = a.size
    s = a.sum()
    if s < 1e-15:
        return np.inf
    b = full_conv(a)
    return 2.0 * n * float(np.max(b)) / (s * s)


def build_lp_matrix(a):
    """Build (2n-1) x n matrix where A[k,j] = a[k-j] if 0<=k-j<n."""
    n = a.size
    A = np.zeros((2*n-1, n))
    for j in range(n):
        A[j:j+n, j] = a
    return A


def solve_lp(a, M):
    """
    Solve: maximize sum(g) s.t. (a*g)[k] <= M for all k, g >= 0.
    Returns g (the optimal solution) or None if infeasible.
    """
    n = a.size
    A = build_lp_matrix(a)
    b_vec = M * np.ones(2*n-1)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b_vec, bounds=[(0, None)]*n, method='highs')
    if res.status == 0:
        return res.x
    return None


def lp_iterate(a, time_budget, rng, verbose=False):
    """
    Run the LP-based iteration to improve a.
    a is a non-negative vector.
    """
    n = a.size
    a = np.maximum(a, 0.0).copy()
    s = a.sum()
    if s < 1e-15:
        a = np.ones(n) / n
    else:
        a = a / s
    
    best_a = a.copy()
    best_sc = score_of(a)
    
    t0 = time.monotonic()
    blend = 0.3
    M_anneal = 1.0
    last_improve = 0
    it = 0
    stall_count = 0
    
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        M = float(np.max(b))
        
        # Annealing: periodically increase M
        if it % 30 == 0 and it > 0:
            M_anneal = min(1.5, 1.0 + 0.1 * (it // 30))
        M_eff = M * M_anneal
        
        g = solve_lp(a, M_eff)
        if g is None:
            # LP failed, try with larger M
            g = solve_lp(a, M_eff * 1.5)
            if g is None:
                break
        
        sum_g = g.sum()
        if sum_g < 1e-15:
            break
        g = g / sum_g
        
        # Blend
        a_new = (1.0 - blend) * a + blend * g
        a_new = np.maximum(a_new, 0.0)
        s_new = a_new.sum()
        if s_new < 1e-15:
            break
        a_new = a_new / s_new
        
        sc_new = score_of(a_new)
        
        if sc_new < best_sc:
            best_sc = sc_new
            best_a = a_new.copy()
            last_improve = it
            stall_count = 0
            if verbose and it % 100 == 0:
                print(f"  iter {it}: score={best_sc:.6f}")
        else:
            stall_count += 1
        
        a = a_new
        it += 1
        
        # Adaptive: if stalled, restart from best with perturbation
        if stall_count > 100:
            a = best_a + 0.02 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            a = a / a.sum()
            blend = 0.2 + 0.4 * rng.random()
            M_anneal = 1.0
            stall_count = 0
        
        # Decay annealing
        if it % 60 == 0:
            M_anneal = max(1.0, M_anneal * 0.95)
    
    return best_a, best_sc

    return np.ones(n) / n


def init_linear_asc(n):
    a = np.arange(1, n+1, dtype=np.float64)

def init_geometric(n, r=1.5):
    a = np.power(r, np.arange(n, dtype=np.float64))

def init_two_bump(n, rng):

    sig = n * (0.1 + 0.1 * rng.random())
    a = np.exp(-((i-center)/sig)**2)
    a += 0.01

                    break
            
            # Try different values for a[i]
            old_val = a[i]

                old_val * 0.8,
                old_val * 1.2,

                rest_sum / n * 1.5,
                0.0,
                rest_sum / (n - 1) * 2.0,

            if old_val > 0:
                candidates.append(old_val * (1 + 0.1 * rng.standard_normal(len(candidates))))

                    best_a = a.copy()
                    improved_in_pass = True

                a[i] = old_val

            patience += 1
        else:
            patience = 0
    
    return best_a, best_sc
