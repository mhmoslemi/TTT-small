        # Maybe only initialize the two-cluster and one fallback?

        inits_list.append(a_uniform)
        
        # Iterate over each initialization
        for idx, cand in enumerate(inits_list):
            # cand must have been created with non-negativity, sum=1
            # time_budget for this cand: we reduce time_remaining by the outer allocation
            time_used = 0

            global_best_score = current_score

            if current_score < best_score:
                best_score = current_score
                best_candidate = cand.copy()
                print(f"Improvement found (n={n_val}) score={current_score}, global best preserved.")
                
            # Set a local time budget
            local_time_budget = (time_budget_per_n * 0.9) * (1.0 if n_val <= 512 else 1.5)
            local_time_remaining = local_time_budget
            # But we are optimizing n_val; the cost is proportional to n_val^2
            # Since FFT is O(n log n), so for every n_val, the cost is n_val*log(n_val)
            # We'll use 0.5 as a time estimation factor per (n_val*log(n_val) units)
            # This means if local_time_budget is 10 seconds, and n_val is 1024, then iterations = local_time_budget/time_per_iteration ~ 100
            
            # However, to handle all constraints, we do projected gradient steps:
            # Project the sequence onto the simplex and clamp if needed
            cand = project_simplex(cand)
            cand = [min(1000, x) for x in cand]
            cand = cand/cand.sum()
            
            # The score function now has additional clamping, so we may need a final evaluation
            cand_score = float(evaluate_sequence(cand))
            # Compare cand_score with global_best
            
            # We do 200 iterations of projected gradient descent
            updates_momentum = [0.0]*n_val
            learning_rate = 0.01
            for _ in range(200):
                # We check the wall time within each iteration
                if time.time() - start_time > total_time:
                    break
                time_per_step = 0.00005 * n_val    # n_val is 128~8, 1024~80 ns per step
                time_used += time_per_step
                # Compute convolution for current cand
                conv_vals = np.convolve(cand, cand, mode='full')
                max_val = np.max(conv_vals)
                score_val = 2.0 * n_val * max_val / (np.sum(cand)**2)
                
                # Update the direction with momentum
                if n_val <= 512:
                    # With momentum for smaller n
                    updates_momentum = 0.9 * updates_momentum + cand_score.diff_direction  # bull
                else:
                    updates_momentum = cand_score.diff_direction
                
                # Project the update
                velocity = updates_momentum * learning_rate
                new_cand = cand - velocity
                new_cand = project_simplex(new_cand)
                new_cand = [min(1000, x) for x in new_cand]
                new_cand = new_cand/np.sum(new_cand)
                cand_score = evaluate_sequence(new_cand)
                
               # If there's improvement, renew the best candidate

                    global_best_score = cand_score
                    global_best = new_cand.copy()
                
                # Discard the candetter if time is up
                if time_used > local_time_budget:
                    break
                    
            # Post-run momentum cleanup
            updates_momentum = [0.0]*n_val
            
        if clamp to [0,1000] and n_val==1024 and learn_rate high, we have to reheat
        if current_score >=1.5:
            # This means we found a local minimum and scored higher than the target
            # Apply a random restart for n_val=1024
            base_seed = seed + n_val  # derived seed to avoid overlap
            rng_restart = np.random.default_rng(base_seed)

            new_a = clamp_and_normalize(rng_restart.uniform(0, 0.1), n_val)
            local_score = float(evaluate_sequence(new_a))
            global_best_score = local_score
            global_best = new_a.copy()

        else:
            # We have a chance to beat the global best
            pass
            
    # After trying all n_val, if global_best[1] hasn't changed in a while, we might want to restart with no parent
    # But the time budget has been nearly consumed
    if best_candidate is not None and time.time() - start_time < total_time and (best_candidate.sum() < 0.01 or best_score not bounded properly):
        # We still have time to polish with random methods
        # Polishing means introducing stochastic perturbations
        polish_budget = (total_time - time.time()) * 0.9  # but we are inside a loop, cannot
    # Instead, we process polych step before the final export
        
    return best_candidate.tolist()
