**Simpler approach: Gradient descent with projection**

Since we're minimizing max(c) where c = convolve(a, a), and we have the constraint a >= 0, sum(a) = 1:

At each iteration:
1. Compute c = convolve(a, a).
2. Find the active set: lags k where c[k] = max(c) (or c[k] > (1-epsilon)*max(c)).
3. Compute the subgradient: g[i] = 2 * sum_{k in active} a[k-i] (with a[j]=0 for j outside [0,n-1]).
4. Update: a_new = a - eta * g.
5. Project onto {a >= 0, sum(a) = 1}.
6. Normalize.

The projection onto {a >= 0, sum(a) = 1} can be done in O(n log n) or O(n) using the "water-filling" algorithm.

def _project_onto_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).ravel().copy()
    n = v.size
    u = np.sort(v)[::-1]

    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]

Now, the gradient step:
- g[i] = 2 * sum_{k in active} a[(k-i) mod ...]

Wait, for linear convolution, (a*a)[k] = sum_i a[i]*a[k-i] where k-i must be in [0, n-1]. So:
d/d a[i] [(a*a)[k]] = a[k-i] (if 0 <= k-i <= n-1) + a[i-k] (if 0 <= i-k <= n-1)
= a[k-i] + a[i-k] (with the convention that a[j] = 0 for j outside [0, n-1])

For the max over k in the active set S:
g[i] = 2 * sum_{k in S} (a[k-i] + a[i-k]) (with out-of-range = 0)

Wait, I need to be more careful. If the max is achieved at a unique k*, then:
g[i] = 2 * (a[k*-i] + a[i-k*]) where we take a[j]=0 if j is outside [0,n-1].

If the max is achieved at multiple k's, the subgradient is the convex hull of the individual gradients. For simplicity, we can use the average:
g[i] = (2/|S|) * sum_{k in S} (a[k-i] + a[i-k])

Actually, for the update to work well, we should use a weighted combination where the weights are proportional to how close c[k] is to the max. This is essentially the log-sum-exp gradient!

"""

-----------------
We minimize score(a) = 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.
Using a normalization-invariant formulation, we work with sum(a)=1 and minimize 2*n*max(c).

Key innovations over the previous approach:

_OPT_N = 0

def _softmax(z):

    if z.size == 0:
        return np.array([], dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.full_like(z, 1.0 / z.size, dtype=np.float64)
    return e / s


def _fft_conv(a):

    N = max(_OPT_NFFT, 1 << (2 * n - 1).bit_length())
    L = 2 * n - 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L].real
    return np.maximum(c, 0.0)


def _fast_score(a):

    if a.size == 0:
        return np.inf
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    c = _fft_conv(a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * a.size * m / (s * s))


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    v = np.asarray(v, dtype=np.float64).ravel().copy()
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1.0))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _cx_algorithm(a, n_iters=300, damping=0.85, deadline=None):
    """
    Cx algorithm: iteratively reduce peaks in the autocorrelation by
    modifying the power spectrum in the frequency domain.
    """
    n = a.size
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-12:
        return np.ones(n) / n
    a /= s

    best_a = a.copy()

        # Reduce bins that are above the geometric mean.

def _spectral_equalize(a, n_iters=200, strength=0.5, deadline=None):
    """
    Spectral equalization: iteratively flatten |FFT(a)|^2 in the frequency domain.
    This directly targets making the autocorrelation flat.
    """
    n = a.size
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-12:
        return np.ones(n) / n
    a /= s

    N = max(_OPT_NFFT, 1 << (2 * n - 1).bit_length())
    best_a = a.copy()
    best_score = _fast_score(a)

    for it in range(n_iters):
        if deadline is not None and time.monotonic() > deadline:
            break

        fa = np.fft.rfft(a, N)
        power = fa.real ** 2 + fa.imag ** 2

        # Target: flat power spectrum
        p_mean = np.mean(power)
        # Log-domain equalization
        log_p = np.log(power + 1e-30)
        log_target = np.mean(log_p)
        # Reduce excess, boost deficit (with damping)
        log_p_new = log_p - strength * (log_p - log_target)
        # Keep DC and Nyquist reasonable
        power_new = np.exp(log_p_new)
        # Scale to preserve total power (equivalent to preserving sum(a)^2)
        power_new *= (np.sum(power) / (np.sum(power_new) + 1e-30))

        # Reconstruct with original phases
        phase = fa / (np.abs(fa) + 1e-30)
        fa_new = phase * np.sqrt(power_new + 1e-30)
        # Ensure DC is real and non-negative
        fa_new[0] = np.abs(fa_new[0])

        a_new = np.fft.irfft(fa_new, N)[:n].real
        a_new = np.maximum(a_new, 0.0)
        s_new = float(np.sum(a_new))
        if s_new <= 1e-12:
            a = np.ones(n) / n
            continue
        a_new /= s_new

        sc = _fast_score(a_new)
        if sc < best_score:
            best_score = sc
            best_a = a_new.copy()

        a = a_new

    return best_a


def _z_from_f(f):

    if f.size != _OPT_N:
        return None
    f = np.where(np.isfinite(f), f, 1.0)

    z = np.log(f)
    z = z - float(np.mean(z))
    z = np.clip(z, -_OPT_BOUND + 1.0, _OPT_BOUND - 1.0)
    return z


def _obj_grad(z, tau):
    """Objective and gradient for log-sum-exp smoothed max autocorrelation."""
    n = _OPT_N
    N = _OPT_NFFT
    L = 2 * n - 1

    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n:
        return 1e12, np.zeros(max(1, z.size), dtype=np.float64)

    if not np.all(np.isfinite(z)):
        z = np.nan_to_num(z, nan=0.0, posinf=_OPT_BOUND, neginf=-_OPT_BOUND)
    z = np.clip(z, -_OPT_BOUND, _OPT_BOUND)

    a = _softmax(z)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L].real
    c = np.maximum(c, 0.0)

    m = float(np.max(c))
    if not np.isfinite(m):
        return 1e12, np.zeros(n, dtype=np.float64)

    diff = (c - m) / tau
    diff = np.clip(diff, -745.0, 0.0)
    e = np.exp(diff)
    se = float(np.sum(e))
    if se <= 0.0 or not np.isfinite(se):
        return 1e12, np.zeros(n, dtype=np.float64)

    p = e / se
    fp = np.fft.rfft(p, N)
    h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real

    s_h = float(np.dot(h, a))
    g = a * (h - s_h)
    grad = 2.0 * g

    val = m + tau * np.log(se)

    # Scale for numerical stability
    scale = float(n) * float(n)
    val = val * scale
    grad = grad * scale

    # Tiny penalty for gauge invariance
    pen = 1e-10
    val += 0.5 * pen * float(np.dot(z, z))
    grad = grad + pen * z

    if not np.isfinite(val):
        val = 1e12
        grad = np.zeros(n, dtype=np.float64)

    return float(val), grad

    z0 = np.clip(z0 - np.mean(z0), -_OPT_BOUND, _OPT_BOUND)

    def obj(z):
        return _obj_grad(z, tau)

    opts = {
        "maxfun": int(maxfun),
        "maxiter": int(maxiter),
        "ftol": 1e-12,
        "pgtol": 1e-10,
        "disp": False,
    }
    try:
        res = minimize(
            obj,
            z0,
            jac=True,
            method="L-BFGS-B",

            options=opts,
        )
        return res
    except Exception:
        return None


def _multi_stage_gradient(z0, deadline):
    """
    Run L-BFGS with decreasing tau (coarse to fine).
    Returns the best z found and its score.
    """
    taus = [1e-2, 5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5]
    z = z0.copy()
    best_z = z.copy()
    best_score = _fast_score(_softmax(z))

    for tau in taus:
        if time.monotonic() > deadline - 30.0:
            break
        maxfun = 300
        maxiter = 80
        res = _run_l_bfgs(z, tau, maxfun=maxfun, maxiter=maxiter)
        if res is not None and hasattr(res, "x"):
            z_new = np.asarray(res.x, dtype=np.float64).ravel()
            if z_new.size == _OPT_N and np.all(np.isfinite(z_new)):
                a_new = _softmax(np.clip(z_new, -_OPT_BOUND, _OPT_BOUND))
                sc = _fast_score(a_new)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_z = z_new.copy()
                z = z_new

    return best_z, best_score


def _polish(a0, deadline):
    """Active-set gradient polish with random restarts."""
    n = _OPT_N
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-12:
        return a0
    a /= s

    b = np.log(np.maximum(a, 1e-300))
    b = np.clip(b - np.mean(b), -_OPT_BOUND, _OPT_BOUND)

    best = a.copy()
    best_score = _fast_score(a)
    best_maxc = best_score / (2.0 * n)

    rng = np.random.default_rng(424242)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        c = _fft_conv(a)
        F = float(np.max(c))

        if F < best_maxc - 1e-15:
            best = a.copy()
            best_score = 2.0 * n * F
            best_maxc = F

        # Use adaptive tau based on the spread of c values near the peak
        m = F
        sorted_c = np.sort(c)[::-1]
        gap = sorted_c[0] - sorted_c[min(10, len(sorted_c) - 1)]
        tau = max(1e-6, min(1e-3, gap * 0.1))

        diff = (c - m) / tau
        diff = np.clip(diff, -745.0, 0.0)
        e = np.exp(diff)
        se = float(np.sum(e))
        if se <= 0.0 or not np.isfinite(se):
            break
        p = e / se

        N = max(_OPT_NFFT, 1 << (2 * n - 1).bit_length())
        fa = np.fft.rfft(a, N)
        fp = np.fft.rfft(p, N)
        h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real
        s_h = float(np.dot(h, a))
        g = a * (h - s_h)

        g = g * (float(n) * float(n))
        g = np.clip(g, -10.0, 10.0)

        accepted = False
        for eta in (1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 1e-1, 3e-1, 1.0, 3.0):
            b_t = np.clip(b - eta * g, -_OPT_BOUND, _OPT_BOUND)
            a_t = _softmax(b_t)
            F_t = float(np.max(_fft_conv(a_t)))
            if F_t < F - 1e-14:
                b = b_t
                a = a_t
                accepted = True
                break

        if not accepted:
            # Try a few different tau values for escape
            escaped = False
            for escape_tau in (1e-4, 5e-4, 1e-3, 3e-3):
                diff2 = (c - m) / escape_tau
                diff2 = np.clip(diff2, -745.0, 0.0)
                e2 = np.exp(diff2)
                se2 = float(np.sum(e2))
                if se2 <= 0.0 or not np.isfinite(se2):
                    continue
                p2 = e2 / se2
                fp2 = np.fft.rfft(p2, N)
                h2 = np.fft.irfft(fp2 * np.conjugate(fa), N)[:n].real
                s_h2 = float(np.dot(h2, a))
                g2 = a * (h2 - s_h2)
                g2 = g2 * (float(n) * float(n))
                g2 = np.clip(g2, -10.0, 10.0)

                for eta in (1e-3, 1e-2, 3e-2, 1e-1):
                    b_t = np.clip(b - eta * g2, -_OPT_BOUND, _OPT_BOUND)
                    a_t = _softmax(b_t)
                    F_t = float(np.max(_fft_conv(a_t)))
                    if F_t < best_maxc - 1e-14:
                        b = b_t
                        a = a_t
                        escaped = True
                        break
                if escaped:
                    break

            if not escaped:
                # Random perturbation
                for scale in (0.01, 0.05, 0.2, 0.5):
                    b_t = np.clip(b + scale * rng.standard_normal(n), -_OPT_BOUND, _OPT_BOUND)
                    a_t = _softmax(b_t)
                    F_t = float(np.max(_fft_conv(a_t)))
                    if F_t < best_maxc - 1e-14:
                        b = b_t
                        a = a_t
                        escaped = True
                        break

            if escaped:
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 150:
                    break
        else:
            no_improve = 0

        it += 1
        if it % 100 == 0:
            cur_score = 2.0 * n * F
            print(f"  polish it={it} cur={cur_score:.9f} best={best_score:.9f} "
                  f"t_left={max(0, deadline - time.monotonic()):.0f}s")

    return best


def _block_sequence(n):
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = 1.0
    a[half:] = 2.0
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 0:
        a[:] = 1.0 / n
    else:
        a /= s
    return a


def _make_starts(rng, deadline):
    """Generate diverse starting points, pre-optimized with Cx."""
    n = _OPT_N
    starts = []

    u = np.linspace(0.0, 1.0, n, endpoint=False)
    x = 2.0 * u - 1.0

    def add_start(f, apply_cx=True):
        f = np.asarray(f, dtype=np.float64).ravel()
        if f.size != n:
            return
        f = np.where(np.isfinite(f), f, 0.0)
        f = np.maximum(f, 0.0)
        s = float(np.sum(f))
        if s <= 1e-12:
            return
        f /= s

        if apply_cx:
            cx_deadline = min(time.monotonic() + 15.0, deadline - 60.0)
            f = _cx_algorithm(f, n_iters=200, deadline=cx_deadline)

        z = _z_from_f(f)
        if z is not None:
            a = _softmax(z)
            sc = _fast_score(a)
            if np.isfinite(sc):
                starts.append((float(sc), z))

    # 1. Previous best (resampled)
    try:
        g = globals().get("height_sequence_1")
        if g is not None:
            g = np.asarray(g, dtype=np.float64)
            if g.size > 1:
                g = np.clip(g, 1e-300, 1000.0)
                if np.all(np.isfinite(g)) and float(np.sum(g)) > 0:
                    x_old = np.linspace(0.0, 1.0, g.size, endpoint=False)
                    x_new = np.linspace(0.0, 1.0, n, endpoint=False)
                    g_interp = np.interp(x_new, x_old, g)
                    add_start(g_interp)
    except Exception:
        pass

    # 2. Two-level blocks with various ratios and positions
    for ratio in (1.3, 1.5, 1.8, 2.0, 2.5, 3.0, 4.0):
        for frac in (0.3, 0.4, 0.5, 0.6):
            f = np.ones(n)
            n_high = int(n * frac)
            f[:n_high] = ratio
            add_start(f)

    # 3. Smooth profiles
    for A in (0.3, 0.5, 0.7, 1.0, 1.5):
        f = 1.0 + A * x
        f = np.maximum(f, 0.01)
        add_start(f)

    # 4. Sinusoidal
    for m in (1, 2, 3, 4):
        for A in (0.2, 0.4, 0.6, 0.8):
            f = 1.0 + A * np.cos(2.0 * np.pi * m * u)
            f = np.maximum(f, 0.01)
            add_start(f)
            f = 1.0 + A * np.sin(2.0 * np.pi * m * u)
            f = np.maximum(f, 0.01)
            add_start(f)

    # 5. Multi-frequency
    for _ in range(8):
        f = np.ones(n)
        for m in range(1, 8):
            c1 = 0.15 * rng.standard_normal()
            c2 = 0.15 * rng.standard_normal()
            f += c1 * np.sin(2.0 * np.pi * m * u)
            f += c2 * np.cos(2.0 * np.pi * m * u)
        f = np.maximum(f, 0.01)
        add_start(f)

    # 6. "Complementary" pattern: alternating high/low blocks
    for block_size in (16, 32, 64, 128):
        pattern = np.tile([1.0, 1.5], n // block_size * block_size)
        if len(pattern) < n:
            pattern = np.pad(pattern, (0, n - len(pattern)))
        f = pattern[:n]
        add_start(f)

    # 7. Triangular
    f = 1.0 + 0.5 * (1.0 - np.abs(2.0 * u - 1.0))
    add_start(f)
    f = 1.0 - 0.5 * (1.0 - np.abs(2.0 * u - 1.0))
    f = np.maximum(f, 0.01)
    add_start(f)

    # Sort by score and keep top
    starts.sort(key=lambda x: x[0])
    return starts[:20]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative coefficient sequence minimizing
    2*n*max(convolve(a,a)) / (sum(a))^2.
    """
    budget_s = max(60.0, float(budget_s))
    deadline = time.monotonic() + budget_s - 10.0

    # Choose sequence length
    n = 16384 if budget_s >= 2000.0 else 8192
    _set_opt_state(n, tau=1e-4)

    rng = np.random.default_rng(int(seed) + 7)
    t0 = time.monotonic()
    print(f"n={n}, budget={budget_s:.0f}s, starting...")

    # Phase 1: Generate diverse starts (with Cx pre-optimization)
    starts_deadline = min(time.monotonic() + budget_s * 0.25, deadline - budget_s * 0.5)
    starts = _make_starts(rng, starts_deadline)
    print(f"Phase 1 done: {len(starts)} starts, time={time.monotonic()-t0:.0f}s, "
          f"best_start={starts[0][0]:.6f}" if starts else "Phase 1: no starts")

    # Phase 2: Multi-stage gradient optimization on top starts
    best_a = _block_sequence(n)
    best_score = _fast_score(best_a)
    print(f"Initial fallback score={best_score:.9f}")

    candidates = []
    top_k = min(len(starts), 10)

    for idx in range(top_k):
        time_left = deadline - time.monotonic()
        if time_left < 120.0:
            break

        if idx < 3:
            stage_deadline = time.monotonic() + min(time_left - 60.0, budget_s * 0.3)
        elif idx < 6:
            stage_deadline = time.monotonic() + min(time_left - 60.0, budget_s * 0.15)
        else:
            stage_deadline = time.monotonic() + min(time_left - 60.0, budget_s * 0.08)

        z0 = starts[idx][1]
        z_best, sc_best = _multi_stage_gradient(z0, stage_deadline)

        a_best = _softmax(z_best)
        sc = _fast_score(a_best)
        if np.isfinite(sc):
            if sc < best_score:
                best_score = sc
                best_a = a_best.copy()
                print(f"  Stage {idx}: score={sc:.9f} (improved!) t={time.monotonic()-t0:.0f}s")
            candidates.append((sc, a_best.copy()))

    # Also try Cx + spectral equalization on the best candidate
    if candidates:
        candidates.sort(key=lambda x: x[0])
        cx_deadline = min(time.monotonic() + 60.0, deadline - 120.0)
        a_cx = _cx_algorithm(candidates[0][1], n_iters=500, deadline=cx_deadline)
        sc_cx = _fast_score(a_cx)
        if np.isfinite(sc_cx) and sc_cx < best_score:
            best_score = sc_cx
            best_a = a_cx.copy()
            print(f"  Cx refinement: score={sc_cx:.9f} t={time.monotonic()-t0:.0f}s")
        candidates.append((sc_cx, a_cx.copy()))

        # Spectral equalization on best
        se_deadline = min(time.monotonic() + 30.0, deadline - 120.0)
        a_se = _spectral_equalize(candidates[0][1], n_iters=300, deadline=se_deadline)
        sc_se = _fast_score(a_se)
        if np.isfinite(sc_se) and sc_se < best_score:
            best_score = sc_se
            best_a = a_se.copy()
            print(f"  Spectral eq: score={sc_se:.9f} t={time.monotonic()-t0:.0f}s")
        candidates.append((sc_se, a_se.copy()))

    # Phase 3: Active-set polish on top candidates
    if candidates:
        candidates.sort(key=lambda x: x[0])
        # Deduplicate
        unique_cands = []
        for sc, a in candidates:
            if not np.isfinite(sc):
                continue
            key = tuple(np.round(a, 8).tolist())
            dup = False
            for _, a2 in unique_cands:
                if np.allclose(a, a2, atol=1e-6):
                    dup = True
                    break
            if not dup:
                unique_cands.append((sc, a))
        candidates = unique_cands[:5]

        for ci, (sc, a) in enumerate(candidates):
            time_left = deadline - time.monotonic()
            if time_left < 30.0:
                break

            if ci == 0:
                pol_deadline = time.monotonic() + min(time_left - 5.0, budget_s * 0.35)
            elif ci == 1:
                pol_deadline = time.monotonic() + min(time_left - 5.0, budget_s * 0.15)
            else:
                pol_deadline = time.monotonic() + min(time_left - 5.0, budget_s * 0.06)

            print(f"  Polishing candidate {ci} (score={sc:.9f}) for {pol_deadline - time.monotonic():.0f}s")
            try:
                a2 = _polish(a, pol_deadline)
                sc2 = _fast_score(a2)
                if np.isfinite(sc2) and sc2 < best_score:
                    best_score = sc2
                    best_a = np.asarray(a2, dtype=np.float64).copy()
                    print(f"  Polish {ci} improved: score={sc2:.9f} t={time.monotonic()-t0:.0f}s")
            except Exception as e:
                print(f"  Polish {ci} error: {e}")

    # Validate with official evaluator
    try:
        best_a = np.asarray(best_a, dtype=np.float64).ravel()
        best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
        best_a = np.clip(best_a, 0.0, 1000.0)
        s = float(np.sum(best_a))
        if s < 0.01 or not np.isfinite(s):
            best_a = np.ones(n, dtype=np.float64) / n
        else:
            best_a = best_a / s
        exact = evaluate_sequence(best_a.tolist())
        if np.isfinite(exact):
            print(f"Official evaluator: score={float(exact):.9f}")
            if float(exact) < best_score:
                best_score = float(exact)
    except Exception:
        pass

    # Final defensive normalization
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = float(np.sum(best_a))
    if s < 0.01 or not np.isfinite(s):
        best_a = np.ones(n, dtype=np.float64) / n
    else:
        best_a = best_a / s

    print(f"Final: n={n}, score={_fast_score(best_a):.9f}, time={time.monotonic()-t0:.0f}s")
    return [float(v) for v in best_a.tolist()]
