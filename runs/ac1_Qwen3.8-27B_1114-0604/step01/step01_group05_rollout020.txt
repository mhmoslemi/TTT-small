"""
Minimize 2*n*max(a*a)/(sum(a))^2 using adaptive gradient descent with momentum.
Uses smooth max with decreasing beta, momentum, and periodic perturbations.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def fft_convolution(a):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    """Smooth max of b via log-sum-exp."""
    m = float(np.max(b))
    if m <= 0.0:
        w = np.ones_like(b) / b.size
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return G, w


def corr_pw(p, w):
    """s_i = sum_j p_j * w_{i+j} for i=0..n-1 (cross-correlation)."""
    n = p.size
    m = w.size
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n : m][::-1].copy()
    return s


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = v.size
    if n == 0:
        return v
    if target <= 0:

        q = np.full(n, target / n)
    else:
        q *= target / s
    return q

    if s <= 0 or not np.isfinite(s):
        return np.full(p.size, 1.0 / p.size)
    return p / s

def fft_convolution(a):
    """Linear convolution a*a via FFT, clipped to non-negative."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def cross_corr(p, q):
    """s_i = sum_j p_j * q_{i+j} for i=0..len(p)-1."""
    np_ = p.size
    nq = q.size
    if np_ == 0 or nq == 0:
        return np.empty(0, dtype=np.float64)
    need = np_ + nq - 1
    size = 1
    while size < need:

    P = np.fft.rfft(p, size)
    QR = np.fft.rfft(q[::-1], size)
    C = np.fft.irfft(P * QR, size)[:need]
    s = C[nq - np_: nq][::-1].copy()

    if not np.all(np.isfinite(v)):
        return np.full(n, target / n, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - target)
    rho_arr = np.where(cond)[0]
    if len(rho_arr) == 0:
        return np.full(n, target / n, dtype=np.float64)
    rho = rho_arr[-1]
    theta = (cssv[rho] - target) / (rho + 1.0)
    q = np.maximum(v - theta, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        q = np.full(n, target / n, dtype=np.float64)
    else:
        q *= target / s
    return q


def normalize(p):
    """Clip to non-negative and normalize to sum 1."""
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.size, 1.0 / p.size, dtype=np.float64)
    return p / s


def compute_score(p):
    """Compute 2*n*max(a*a) for a sequence with sum=1."""
    b = fft_convolution(p)
    M = float(np.max(b))
    return float(2.0 * p.size * M), b, M


def multi_peak_gradient(p, eps):
    """
    Compute averaged subgradient over all near-peak positions.
    Positions where c_k >= (1-eps)*max(c) are considered peaks.
    Returns gradient (centered to be in simplex tangent space).
    """
    b = fft_convolution(p)
    M = float(np.max(b))
    if M <= 0:
        return np.zeros_like(p), b, M
    threshold = (1.0 - eps) * M
    mask = (b >= threshold).astype(np.float64)
    total = float(np.sum(mask))
    if total <= 0:
        mask[int(np.argmax(b))] = 1.0
        total = 1.0
    mask /= total
    svec = cross_corr(p, mask)
    g = 2.0 * svec
    grad = g - float(np.mean(g))
    return grad, b, M


def optimize_gradient(p0, deadline, eps_start, eps_end, mom_start, mom_end,
                      max_iter=999999, perturb_interval=3000, perturb_scale=0.005):
    """
    Gradient descent with momentum and multi-peak subgradient.
    Anneals eps and momentum over the optimization.
    """
    p = normalize(p0.copy())
    n = p.size
    if n == 0:
        return p, float('inf')

    _, b, M = compute_score(p)
    best_p = p.copy()
    best_score = float(2.0 * n * M)

    vel = np.zeros(n, dtype=np.float64)
    step = 0.5
    it = 0
    no_improve = 0
    rng_local = np.random.default_rng(12345)

    while it < max_iter and time.monotonic() < deadline:
        it += 1
        frac = float(it) / max_iter

        # Anneal eps and momentum
        eps = eps_start + (eps_end - eps_start) * frac
        momentum = mom_start + (mom_end - mom_start) * frac

        # Compute gradient
        grad, b, M = multi_peak_gradient(p, eps)
        cur_score = float(2.0 * n * M)

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > perturb_interval:
            # Perturbation to escape local minimum
            if time.monotonic() < deadline - 5:
                p = normalize(p + perturb_scale * rng_local.standard_normal(n))
                vel *= 0.1
                step = 0.1
                no_improve = 0
            else:
                break

        if it % 2000 == 0:
            print(f"    it={it} eps={eps:.4f} mom={momentum:.3f} score={best_score:.8f} step={step:.4f} ni={no_improve}")

        # Check convergence
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        # Momentum update
        vel = momentum * vel + (1.0 - momentum) * grad

        # Backtracking line search
        cur_score_val = cur_score
        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01):
            alpha = step * mult
            q = project_simplex(p - alpha * vel, 1.0)
            qb = fft_convolution(q)
            qM = float(np.max(qb))
            qscore = float(2.0 * n * qM)
            if qscore < cur_score_val - 1e-13:
                p = q
                accepted = True
                break
            elif qscore < cur_score_val + 1e-10:
                # Accept if roughly equal (helps with flat regions)
                if frac > 0.5:
                    p = q
                    accepted = True
                    break

        if not accepted:
            step *= 0.5
            vel *= 0.5
            if step < 1e-10:
                break
        else:
            if step < 3.0:
                step = min(3.0, step * 1.08)

    return best_p, best_score


def lp_refine(f, M, deadline, n_blends=5):
    """
    LP-based refinement: find g maximizing sum(g) s.t. g>=0, (f*g)_k <= M for all k.
    Then blend f towards g. Returns improved sequence.
    """
    if not HAVE_LINPROG:
        return f

    f = np.asarray(f, dtype=np.float64)
    n = f.size
    num_constr = 2 * n - 1

    # Build constraint matrix A (num_constr x n) where A[k,j]=1 if 0<=k-j<n
    # This is a banded matrix: for row k, columns j are max(0, k-n+1) to min(n-1, k)
    # We can't easily pass this as a dense matrix for n=8192 (would be 16383x8192 ~ 134M entries)
    # Use sparse if available
    try:
        from scipy import sparse
        # Build sparse matrix
        rows = np.repeat(np.arange(num_constr), 1)  # will adjust
        # Actually let's build it properly
        # For each k in 0..2n-2, valid j are max(0, k-n+1) .. min(n-1, k)
        row_list = []
        col_list = []
        for k in range(num_constr):
            j_start = max(0, k - n + 1)
            j_end = min(n - 1, k)
            if j_start <= j_end:
                row_list.extend([k] * (j_end - j_start + 1))
                col_list.extend(range(j_start, j_end + 1))
        A = sparse.csr_matrix((np.ones(len(row_list)), (row_list, col_list)),
                               shape=(num_constr, n))
        c_obj = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
        res = linprog(c_obj, A_ub=A, b_ub=np.full(num_constr, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': max(1, deadline - time.monotonic() - 2)})
        if res.status == 0 and res.x is not None:
            g = np.maximum(res.x, 0)
            g_sum = float(np.sum(g))
            if g_sum > 0:
                g = g / g_sum
            else:
                g = f.copy()
            # Blend
            best_blend = f.copy()
            best_blend_score = 2.0 * n * float(np.max(fft_convolution(f)))
            for t in (0.1, 0.2, 0.3, 0.5, 0.7, 1.0):
                blend = normalize((1 - t) * f + t * g)
                sc = 2.0 * n * float(np.max(fft_convolution(blend)))
                if sc < best_blend_score:
                    best_blend_score = sc
                    best_blend = blend
            return best_blend
    except Exception as e:
        print(f"    LP failed: {e}")

    return f


def try_starting_points(N, rng):
    """Generate diverse starting points and return the best few."""
    starts = []
    # Uniform
    starts.append(np.full(N, 1.0 / N))
    # Linear ramps
    x = np.linspace(0, 1, N)
    for a in (-3.0, -1.5, 1.5, 3.0):
        p = np.exp(a * (x - 0.5))
        starts.append(p / p.sum())
    # U-shaped
    p = 1.0 + 2.0 * (2 * x - 1) ** 2
    starts.append(p / p.sum())
    # Cosine window (arch)
    p = 0.5 * (1 + np.cos(np.pi * (2 * x - 1)))
    p = np.maximum(p, 0.01)
    starts.append(p / p.sum())
    # Random smooth
    for _ in range(3):
        zc = rng.normal(size=N * 2) + 1j * rng.normal(size=N * 2)
        z = np.real(np.fft.ifft(zc))[:N]
        z = np.tanh(1.0 * z)
        p = np.exp(z)
        starts.append(p / p.sum())
    # Score them
    scored = []
    for p in starts:
        sc, _, _ = compute_score(p)
        scored.append((sc, p))
    scored.sort(key=lambda x: x[0])
    return [p for _, p in scored[:4]]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: gradient with multi-peak subgradient, LP refinement, perturbation restarts.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    rng = np.random.default_rng(seed)
    start_time = time.monotonic()
    reserve = 30.0
    deadline = start_time + max(30.0, budget_s - reserve)

    N = 8192

    # Get initial sequence
    best_p = np.full(N, 1.0 / N)
    try:
        h = height_sequence_1
        if isinstance(h, list) and len(h) == N:
            p0 = np.array(h, dtype=np.float64)
            p0 = normalize(p0)
            sc, _, _ = compute_score(p0)
            if np.isfinite(sc) and sc < 2.0:
                best_p = p0.copy()
                print(f"Loaded height_sequence_1: score={sc:.8f}")
    except Exception:
        print("No valid height_sequence_1, starting from scratch")

    # Try multiple starting points and keep the best
    starts = try_starting_points(N, rng)
    for p in starts:
        sc, _, _ = compute_score(p)
        if sc < compute_score(best_p)[0]:
            best_p = p.copy()

    best_score, _, _ = compute_score(best_p)
    print(f"Best starting score: {best_score:.8f}")

    # Phase 1: Broad optimization (40% of budget)
    t1 = start_time + 0.35 * (budget_s - reserve)
    if time.monotonic() < t1:
        print(f"\nPhase 1: Broad gradient (eps 0.15->0.03, mom 0.9->0.6) until t={t1-start_time:.0f}s")
        p1, s1 = optimize_gradient(best_p, t1, eps_start=0.15, eps_end=0.03,
                                   mom_start=0.9, mom_end=0.6, perturb_interval=2000)
        if s1 < best_score:
            best_score = s1
            best_p = p1.copy()
            print(f"  Phase 1 result: {best_score:.8f}")

    # Phase 2: Medium optimization (25% of budget)
    t2 = start_time + 0.60 * (budget_s - reserve)
    if time.monotonic() < t2:
        print(f"\nPhase 2: Medium gradient (eps 0.03->0.005, mom 0.6->0.4) until t={t2-start_time:.0f}s")
        p2, s2 = optimize_gradient(best_p, t2, eps_start=0.03, eps_end=0.005,
                                   mom_start=0.6, mom_end=0.4, perturb_interval=3000)
        if s2 < best_score:
            best_score = s2
            best_p = p2.copy()
            print(f"  Phase 2 result: {best_score:.8f}")

    # Phase 3: Sharp optimization (20% of budget)
    t3 = start_time + 0.80 * (budget_s - reserve)
    if time.monotonic() < t3:
        print(f"\nPhase 3: Sharp gradient (eps 0.005->0.0, mom 0.4->0.2) until t={t3-start_time:.0f}s")
        p3, s3 = optimize_gradient(best_p, t3, eps_start=0.005, eps_end=0.0,
                                   mom_start=0.4, mom_end=0.2, perturb_interval=5000)
        if s3 < best_score:
            best_score = s3
            best_p = p3.copy()
            print(f"  Phase 3 result: {best_score:.8f}")

    # Phase 4: LP refinement + final polish (remaining time)
    if time.monotonic() < deadline - 10 and HAVE_LINPROG:
        print(f"\nPhase 4: LP refinement")
        remaining = deadline - time.monotonic() - 10
        if remaining > 30:
            _, b_curr, M_curr = compute_score(best_p)
            # Try a few LP blends
            for i in range(3):
                if time.monotonic() >= deadline - 10:
                    break
                sub_dl = min(deadline - 10, time.monotonic() + remaining / 3.0)
                p_lp = lp_refine(best_p, float(M_curr), sub_dl)
                sc_lp, _, M_new = compute_score(p_lp)
                if sc_lp < best_score:
                    best_score = sc_lp
                    best_p = p_lp.copy()
                    print(f"  LP iter {i}: {best_score:.8f}")
                    M_curr = M_new
                else:
                    break

    # Phase 5: Final polish with small gradient steps
    if time.monotonic() < deadline - 5:
        print(f"\nPhase 5: Final polish")
        sub_dl = deadline - 5
        p5, s5 = optimize_gradient(best_p, sub_dl, eps_start=0.01, eps_end=0.0,
                                   mom_start=0.3, mom_end=0.1, perturb_interval=10000,
                                   perturb_scale=0.002)
        if s5 < best_score:
            best_score = s5
            best_p = p5.copy()
            print(f"  Final polish: {best_score:.8f}")

    # Final: normalize and return
    final_p = normalize(best_p)
    final_seq = [float(x) for x in final_p]

    # Verify with official evaluator
    try:
        val = float(evaluate_sequence(final_seq))
        if np.isfinite(val):
            print(f"\nOfficial score: {val:.8f}")
            return final_seq
    except Exception:
        pass

    print("Official eval failed, returning best internal")
    return final_seq
