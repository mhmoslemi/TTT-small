"""
Low-peak nonnegative step-function search using multi-mode sinusoidal
parameterization and mirror descent refinement.

Approach:
1. Search over multi-mode sinusoidal shapes: a[i] = (1 + sum A_k sin(k*theta_i))/n
   with non-negativity constraint.
2. Refine using mirror descent (multiplicative weights) with FFT-based
   gradient computation.
3. Try multiple sequence lengths and return the best.
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _score(a):
    """Fast internal score computation."""
    arr = np.asarray(a, dtype=np.float64)
    n = arr.size
    if n == 0:
        return np.inf
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(np.sum(arr))
    if s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    mx = float(np.max(c))
    return float(2.0 * n * mx / (s * s))


def _make_multimode(n, amps, deadline=None):
    """
    Create a multi-mode sinusoidal sequence.
    amps = [A1, A2, A3, ...] for modes 1, 2, 3, ...
    """
    n = int(n)
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    a = np.ones(n, dtype=np.float64)
    for k, Ak in enumerate(amps, 1):
        if abs(Ak) > 1e-10:
            a = a + Ak * np.sin(k * theta)
    a = a / float(n)
    a = np.clip(a, 0.0, 1000.0)
    s = float(np.sum(a))
    if s < 0.01:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def _score_multimode(n, amps):
    """Score a multi-mode shape without storing the full array."""
    a = _make_multimode(n, amps)
    return _score(a)


def _is_nonnegative(n, amps, tol=1e-10):
    """Check if the multi-mode shape is non-negative."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    vals = np.ones(n, dtype=np.float64)
    for k, Ak in enumerate(amps, 1):
        if abs(Ak) > 1e-10:
            vals = vals + Ak * np.sin(k * theta)
    return np.min(vals) >= -tol
