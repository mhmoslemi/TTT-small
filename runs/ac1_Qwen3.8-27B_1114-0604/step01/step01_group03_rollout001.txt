"""
Hybrid search for flat autocorrelation sequences.
Uses improved MWS (Multiplicative Weights for Saddle) with trajectory averaging,
combined with momentum-based multiplicative updates and Frank-Wolfe polish.
"""

import math
import time
import numpy as np

def _normalize(a, eps=1e-300):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a.fill(1.0)
        s = float(a.sum())
    a /= s
    a = np.maximum(a, eps)
    a /= float(a.sum())
    return a

def _fft_len(n):
    m = 2 * n - 1
    L = 1
    while L < m:
        L <<= 1
    return L, m

def _conv_self(a, L, m):
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c

def _score(c, n):
    return 2.0 * n * float(np.max(c))

def _score_a(a, L, m, n):
    return _score(_conv_self(a, L, m), n)

def _gradient_smooth(a, c, n, temp, L, m):
    """Compute gradient of soft max approximation.
    Returns gradient vector of length n.
    """
    v = n * c
    vmax = float(np.max(v))
    z = temp * (v - vmax)
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.full(m, 1.0/m)
    else:
        w = e / s
    
    # gradient_i = 2 * sum_k w_k * a_{k-i} (reversed conv)
    Frev = np.fft.rfft(a[::-1], L)
    W = np.fft.rfft(w, L)
    g = np.fft.irfft(W * Frev, L)[(n-1):m]
    np.maximum(g, 0.0, out=g)
    return 2.0 * g, w

def _mws_improved(f, L, m, n, n_avg, eta_start, eta_end, inner_iter, rng):
    """Improved MWS with trajectory averaging.
    Returns (x_avg, y_avg) - the averaged strategies.
    """
    f = _normalize(f)
    x = _normalize(np.ones(n) * 0.5 + 0.5 * f)  # start close to f
    y = np.full(m, 1.0/m)
    
    x_sum = np.zeros(n)
    y_sum = np.zeros(m)
    count = 0
    
    F = np.fft.rfft(f, L)
    Frev = np.fft.rfft(f[::-1], L)
    
    for t in range(inner_iter):
        frac = t / max(1, inner_iter - 1)
        eta = eta_start + (eta_end - eta_start) * frac
        
        # Update y: best response of maximizer to current x
        X = np.fft.rfft(x, L)
        u = np.fft.irfft(F * X, L)[:m]
        np.maximum(u, 0.0, out=u)
        
        vu = n * u
        vma = float(np.max(vu))
        z = eta * (vu - vma)
        z = np.clip(z, -100.0, 0.0)
        y_new = np.exp(z)
        ys = float(y_new.sum())
        if ys > 0 and np.isfinite(ys):
            y = y_new / ys
        else:
            y.fill(1.0/m)
        
        # Update x: best response of minimizer to current y
        Y = np.fft.rfft(y, L)
        s = np.fft.irfft(Y * Frev, L)[(n-1):m]
        np.maximum(s, 0.0, out=s)
        
        ps = n * s
        sma = float(np.max(ps))
        zx = -eta * (ps - sma)
        zx -= float(np.max(zx))
        zx = np.clip(zx, -100.0, 0.0)
        x_new = np.exp(zx)
        xs = float(x_new.sum())
        if xs > 0 and np.isfinite(xs):
            x = x_new / xs
        else:
            x = f.copy()
        
        # Averaging
        x_sum += x
        y_sum += y
        count += 1
    
    if count > 0:
        x_avg = x_sum / count
        y_avg = y_sum / count
    else:
        x_avg = f.copy()
        y_avg = y
    
    return _normalize(x_avg), _normalize(y_avg)

def _fw_step(a, c, n, L, m, step_size):
    """One Frank-Wolfe step.
    Find the coordinate i that minimizes the subgradient, move towards e_i.
    """
    v = n * c
    # Subgradient direction: for the max, the subgradient at the max point
    # is the reversed conv of the indicator of argmax
    # But for smoothness, use the soft version
    vmax = float(np.max(v))
    # Use a moderate temperature for the FW step
    temp = 500.0
    z = temp * (v - vmax)
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.full(m, 1.0/m)
    else:
        w = e / s
    
    Frev = np.fft.rfft(a[::-1], L)
    W = np.fft.rfft(w, L)
    g = np.fft.irfft(W * Frev, L)[(n-1):m]
    np.maximum(g, 0.0, out=g)
    
    # FW: move towards e_{argmin g}
    i_min = int(np.argmin(g))
    t = step_size  # step size
    a_new = (1.0 - t) * a + t * 0.0
    a_new[i_min] += t
    return _normalize(a_new)

def _random_prior(n, rng):
    """Generate a random smooth prior."""
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0
    z = rng.normal(0.0, 0.3)
    z += rng.normal(0.0, 0.5) * x
    
    kmax = int(rng.integers(2, 10))
    for k in range(1, kmax + 1):
        amp = rng.uniform(0.05, 0.5) / max(1.0, math.sqrt(float(k)))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos(2.0 * np.pi * k * t + ph)
    
    noise = rng.normal(0.0, 0.1, n)
    noise = np.convolve(noise, np.ones(5, dtype=np.float64)/5.0, mode="same")
    z += noise
    
    p = np.exp(z - float(np.max(z)))
    p = np.maximum(p, 1e-300)
    return _normalize(p)

def _momentum_search(a, L, m, n, rng, time_limit):
    """Momentum-based multiplicative weight search."""
    deadline = time.monotonic() + time_limit
    a = _normalize(a)
    c = _conv_self(a, L, m)
    current_score = _score(c, n)
    
    best_a = a.copy()
    best_score = current_score
    
    # Momentum buffer
    v_buf = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    m_buf = np.zeros(n)
    v_buf_sq = np.zeros(n)
    
    temp = 5.0
    eta = 0.05
    no_improve = 0
    it = 0
    t_step = 1.0
    
    while time.monotonic() < deadline:
        c = _conv_self(a, L, m)
        current_score = _score(c, n)
        
        if current_score < best_score - 1e-13:
            best_score = current_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 200:
            # Restart with perturbation
            a = best_a.copy()
            prior = _random_prior(n, rng)
            mix = rng.uniform(0.1, 0.3)
            a = _normalize((1-mix)*a + mix*prior)
            c = _conv_self(a, L, m)
            current_score = _score(c, n)
            no_improve = 0
            temp = 5.0
            eta = 0.05
            m_buf.fill(0)
            v_buf_sq.fill(0)
            continue
        
        # Compute gradient
        g, w = _gradient_smooth(a, c, n, temp, L, m)
        
        # Normalize gradient
        g_max = float(np.max(g))
        if g_max > 0:
            g = g / g_max
        
        # Adam update
        t_step += 1.0
        m_buf = beta1 * m_buf + (1 - beta1) * g
        v_buf_sq = beta2 * v_buf_sq + (1 - beta2) * g * g
        
        m_hat = m_buf / (1 - beta1**t_step)
        v_hat = v_buf_sq / (1 - beta2**t_step)
        
        # Multiplicative update with Adam step
        update = m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = a * np.exp(-eta * update)
        a_new = _normalize(a_new)
        
        c_new = _conv_self(a_new, L, m)
        sc_new = _score(c_new, n)
        
        if sc_new < current_score:
            a = a_new
            eta = min(0.5, eta * 1.05)

            a_half = _normalize(a_half)

                a = a_half
            eta = max(0.01, eta * 0.85)
        
        # Anneal temperature
        if it % 50 == 0:
            temp = min(500.0, temp * 1.3)
        
        it += 1
    
    return best_a, best_score

def _proj_simplex(v):
    """Project v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def _normalize(a, eps=1e-300):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a.fill(1.0)
        s = float(a.sum())
    a /= s
    a = np.maximum(a, eps)
    a /= float(a.sum())
    return a


def _fft_params(n):
    m = 2 * n - 1
    L = 1
    while L < m:
        L <<= 1
    return L, m


def _conv_self(a, L, m):
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _score_c(c, n):
    return 2.0 * n * float(np.max(c))


def _score_a(a, L, m, n):
    return _score_c(_conv_self(a, L, m), n)


def _soft_weight(c, n, temp):
    v = n * c
    vmax = float(np.max(v))
    z = temp * (v - vmax)
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(len(c), 1.0 / len(c))
    return e / s


def _reversed_conv(a, w, L, m, n):
    """Compute g = 2 * (rev(a) * w) truncated to length n."""
    Frev = np.fft.rfft(a[::-1], L)
    W = np.fft.rfft(w, L)
    g = np.fft.irfft(W * Frev, L)[(n - 1):m]
    np.maximum(g, 0.0, out=g)
    return 2.0 * g


def _mws_long(f, L, m, n, inner_iter, eta):
    """Run MWS for inner_iter steps with constant step size eta.
    Returns the averaged x (minimizer strategy).
    """
    f = _normalize(f)
    x = f.copy()
    y = np.full(m, 1.0 / m)

    x_sum = np.zeros(n)
    F = np.fft.rfft(f, L)
    Frev = np.fft.rfft(f[::-1], L)
    count = 0

    for t in range(inner_iter):
        # Update y (maximizer best response to x)
        X = np.fft.rfft(x, L)
        u = np.fft.irfft(F * X, L)[:m]
        np.maximum(u, 0.0, out=u)

        vu = n * u
        vma = float(np.max(vu))
        z = eta * (vu - vma)
        z = np.clip(z, -100.0, 0.0)
        y_new = np.exp(z)
        ys = float(y_new.sum())
        if ys > 0.0 and np.isfinite(ys):
            y = y_new / ys
        else:
            y.fill(1.0 / m)

        # Update x (minimizer best response to y)
        Y = np.fft.rfft(y, L)
        s = np.fft.irfft(Y * Frev, L)[(n - 1):m]
        np.maximum(s, 0.0, out=s)

        ps = n * s
        sma = float(np.max(ps))
        zx = -eta * (ps - sma)
        zx -= float(np.max(zx))
        zx = np.clip(zx, -100.0, 0.0)
        x_new = np.exp(zx)
        xs = float(x_new.sum())
        if xs > 0.0 and np.isfinite(xs):
            x = x_new / xs
        else:
            x = f.copy()

        x_sum += x
        count += 1

    if count > 0:
        return _normalize(x_sum / count)
    return f.copy()


def _mws_schedule(f, L, m, n, inner_iter, eta_start, eta_end):
    """MWS with linearly varying step size."""
    f = _normalize(f)
    x = f.copy()
    y = np.full(m, 1.0 / m)
    x_sum = np.zeros(n)
    F = np.fft.rfft(f, L)
    Frev = np.fft.rfft(f[::-1], L)

    for t in range(inner_iter):
        frac = t / max(1, inner_iter - 1)
        eta = eta_start + (eta_end - eta_start) * frac

        X = np.fft.rfft(x, L)
        u = np.fft.irfft(F * X, L)[:m]
        np.maximum(u, 0.0, out=u)

        vu = n * u
        vma = float(np.max(vu))
        z = eta * (vu - vma)
        z = np.clip(z, -100.0, 0.0)
        y_new = np.exp(z)
        ys = float(y_new.sum())
        if ys > 0.0 and np.isfinite(ys):
            y = y_new / ys

        Y = np.fft.rfft(y, L)
        s = np.fft.irfft(Y * Frev, L)[(n - 1):m]
        np.maximum(s, 0.0, out=s)

        ps = n * s
        sma = float(np.max(ps))
        zx = -eta * (ps - sma)
        zx -= float(np.max(zx))
        zx = np.clip(zx, -100.0, 0.0)
        x_new = np.exp(zx)
        xs = float(x_new.sum())
        if xs > 0.0 and np.isfinite(xs):
            x = x_new / xs
        else:
            x = f.copy()

        x_sum += x

    return _normalize(x_sum / max(1, inner_iter))


def _random_prior(n, rng):
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0
    z = rng.normal(0.0, 0.3)
    z += rng.normal(0.0, 0.5) * x
    kmax = int(rng.integers(2, 10))
    for k in range(1, kmax + 1):
        amp = rng.uniform(0.05, 0.5) / max(1.0, math.sqrt(float(k)))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos(2.0 * np.pi * k * t + ph)
    noise = rng.normal(0.0, 0.1, n)
    noise = np.convolve(noise, np.ones(5, dtype=np.float64) / 5.0, mode="same")
    z += noise
    p = np.exp(z - float(np.max(z)))
    p = np.maximum(p, 1e-300)
    return _normalize(p)

def _proj_simplex(v):
    """Project v onto the probability simplex."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    Cond = u * np.arange(1, n + 1) > (cssv - 1.0)
    rho = int(np.where(Cond)[0][-1])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _nesterov_subgrad(a, L, m, n, time_limit):
    """Nesterov-accelerated projected subgradient method."""
    deadline = time.monotonic() + time_limit
    a = _normalize(a)
    a_prev = a.copy()
    c = _conv_self(a, L, m)
    current = _score_c(c, n)
    best_a = a.copy()
    best_score = current
    no_improve = 0
    alpha = 2.0

    for t in range(1, 100000):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        gamma = (t - 1) / float(t + 2)
        y = a + gamma * (a - a_prev)
        y = _proj_simplex(y)

        # Compute subgradient at y
        c_y = _conv_self(y, L, m)
        # Use soft approximation with high temp for near-exact subgradient
        w = _soft_weight(c_y, n, 5000.0)
        g = _reversed_conv(y, w, L, m, n)
        gmax = float(np.max(g))
        if gmax > 0:
            g /= gmax

        step = alpha / float(t)
        a_new = _proj_simplex(y - step * g)
        a_prev = a
        a = a_new

        c = _conv_self(a, L, m)
        current = _score_c(c, n)
        if current < best_score - 1e-14:
            best_score = current
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            break

    return best_a, best_score

def _line_search(a, d, L, m, n, base_score):
    """Line search along direction d from a."""
    best_a = a.copy()
    best_s = base_score
    for lam in (0.02, 0.05, 0.08, 0.1, 0.15, 0.2, 0.3, 0.4, 0.5, 0.7, 1.0):

def _exact_score(seq):
    try:
        val = evaluate_sequence(seq)
    except NameError:
        val = None
    except Exception:
        return float('inf')
    if val is not None:
        return float(val)
    # Fallback
    if not isinstance(seq, list) or not seq:
        return float('inf')
    for x in seq:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return float('inf')
        if np.isnan(x) or np.isinf(x):
            return float('inf')
    a = [max(0.0, min(1000.0, float(x))) for x in seq]
    nn = len(a)
    s = sum(a)
    if s < 0.01:
        return float('inf')
    b = np.convolve(a, a)
    mb = max(b)
    return float(2.0 * nn * mb / (s * s))


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search: MWS (long inner loops) -> Adam momentum -> Nesterov subgradient -> fine polish.
    """
    try:
        budget = float(budget_s)
    except (ValueError, TypeError):
        budget = 2400.0
    if not np.isfinite(budget) or budget < 10.0:
        budget = 2400.0

    t0 = time.monotonic()
    hard_deadline = t0 + budget - 8.0
    search_deadline = t0 + budget - 90.0
    polish_deadline = t0 + budget - 30.0

    rng = np.random.default_rng(seed)

    # Determine n
    n = 6121
    try:
        base = height_sequence_1
    except NameError:
        base = None
    if base is not None:
        try:
            arr = np.asarray(base, dtype=np.float64).ravel()
            if arr.size > 100 and np.all(np.isfinite(arr)):
                n = arr.size
        except Exception:
            pass

    L, m = _fft_params(n)
    print(f"Phase 0: n={n}, L={L}, m={m}, budget={budget:.0f}s")

    # --- Initialize best candidate ---
    best_a = _normalize(np.ones(n))
    best_score = _score_a(best_a, L, m, n)
    print(f"Uniform baseline: {best_score:.6f}")

    if base is not None:
        try:
            arr = np.asarray(base, dtype=np.float64).ravel()
            if arr.size == n and np.all(np.isfinite(arr)):
                a0 = _normalize(arr)
                s0 = _score_a(a0, L, m, n)
                if s0 < best_score:
                    best_score = s0
                    best_a = a0
                print(f"Base sequence score: {s0:.6f}")
        except Exception:
            pass

    # --- Phase 1: MWS with long inner loops ---
    print(f"\nPhase 1: MWS search (deadline={search_deadline - t0:.0f}s from start)")

    # Try multiple MWS configurations
    mws_configs = [
        # (inner_iter, eta_start, eta_end, use_schedule)
        (500, 5.0, 5.0, False),
        (1000, 3.0, 3.0, False),
        (800, 2.0, 20.0, True),
        (1500, 4.0, 4.0, False),
        (1000, 8.0, 8.0, False),
        (2000, 2.0, 2.0, False),
        (1000, 1.0, 50.0, True),
        (1500, 6.0, 6.0, False),
    ]

    p1_deadline = search_deadline
    for cfg_idx, (inner_it, e1, e2, sched) in enumerate(mws_configs):
        if time.monotonic() > p1_deadline:
            break
        round_time = time.monotonic() - t0
        remaining = p1_deadline - time.monotonic()
        if remaining < 3.0:
            break

        # Try a few different starting points for MWS
        starts = [best_a]
        if cfg_idx < 3:
            starts.append(_random_prior(n, rng))

        for start in starts:
            if time.monotonic() > p1_deadline:
                break
            if sched:
                direction = _mws_schedule(start, L, m, n, inner_it, e1, e2)
            else:
                direction = _mws_long(start, L, m, n, inner_it, e1)

            cand, sc = _line_search(best_a, direction, L, m, n, best_score)
            if sc < best_score - 1e-13:
                best_score = sc
                best_a = cand
                print(f"  MWS cfg {cfg_idx} (it={inner_it}, eta={e1}): {best_score:.6f} [t={time.monotonic()-t0:.0f}s]")

    print(f"After Phase 1: {best_score:.6f}")

    # --- Phase 2: Adam momentum search ---
    p2_deadline = min(polish_deadline - 60.0, search_deadline + 60.0)
    if time.monotonic() < p2_deadline - 5.0:
        lim = min(600.0, p2_deadline - time.monotonic() - 5.0)
        print(f"\nPhase 2: Adam momentum (limit={lim:.0f}s)")
        a2, s2 = _adam_momentum_search(best_a, L, m, n, rng, lim)
        if s2 < best_score - 1e-13:
            best_score = s2
            best_a = a2
            print(f"After Phase 2: {best_score:.6f}")
        if time.monotonic() < p2_deadline - 5.0:
            # Run again from where we left off
            lim2 = min(300.0, p2_deadline - time.monotonic() - 5.0)
            if lim2 > 10.0:
                a2b, s2b = _adam_momentum_search(best_a, L, m, n, rng, lim2)
                if s2b < best_score - 1e-13:
                    best_score = s2b
                    best_a = a2b
                    print(f"After Phase 2b: {best_score:.6f}")

    # --- Phase 3: Nesterov subgradient ---
    if time.monotonic() < polish_deadline - 10.0:
        lim = min(120.0, polish_deadline - time.monotonic() - 10.0)
        print(f"\nPhase 3: Nesterov subgradient (limit={lim:.0f}s)")
        a3, s3 = _nesterov_subgrad(best_a, L, m, n, lim)
        if s3 < best_score - 1e-14:
            best_score = s3
            best_a = a3
            print(f"After Phase 3: {best_score:.6f}")

    # --- Phase 4: Fine polish ---
    if time.monotonic() < hard_deadline - 10.0:
        lim = min(180.0, hard_deadline - time.monotonic() - 10.0)
        print(f"\nPhase 4: Fine polish (limit={lim:.0f}s)")
        a4, s4 = _fine_polish(best_a, L, m, n, lim)
        if s4 < best_score - 1e-14:
            best_score = s4
            best_a = a4
            print(f"After Phase 4: {best_score:.6f}")

        # One more round of fine polish
        if time.monotonic() < hard_deadline - 10.0:
            lim2 = min(90.0, hard_deadline - time.monotonic() - 10.0)
            if lim2 > 5.0:
                a5, s5 = _fine_polish(best_a, L, m, n, lim2)
                if s5 < best_score - 1e-14:
                    best_score = s5
                    best_a = a5
                    print(f"After Phase 4b: {best_score:.6f}")

    # --- Final: try a few MWS restarts if time remains ---
    if time.monotonic() < hard_deadline - 15.0:
        remaining = hard_deadline - time.monotonic() - 15.0
        if remaining > 20.0:
            print(f"\nPhase 5: Final MWS attempts ({remaining:.0f}s remaining)")
            for e in (3.0, 5.0, 10.0):
                if time.monotonic() > hard_deadline - 15.0:
                    break
                direction = _mws_long(best_a, L, m, n, 1000, e)
                cand, sc = _line_search(best_a, direction, L, m, n, best_score)
                if sc < best_score - 1e-13:
                    best_score = sc
                    best_a = cand
                    # Quick polish
                    if time.monotonic() < hard_deadline - 20.0:
                        a_p, s_p = _fine_polish(best_a, L, m, n, min(30.0, hard_deadline - time.monotonic() - 20.0))
                        if s_p < best_score - 1e-14:
                            best_score = s_p
                            best_a = a_p
                    print(f"  MWS eta={e}: {best_score:.6f}")

    # --- Return ---
    seq = [float(x) for x in best_a]
    exact = _exact_score(seq)
    print(f"\nFinal internal score: {best_score:.6f}")
    print(f"Final exact score: {exact:.6f}")
    print(f"Total time: {time.monotonic() - t0:.1f}s")

    return seq
