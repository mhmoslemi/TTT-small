from scipy.optimize import linprog
import time

def full_conv(a):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def normed_score(a):
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))

def solve_lp(f, M):
    """Solve: max sum(g) s.t. f*g <= M, g >= 0"""
    n = f.size
    # Build constraint matrix A (2n-1) x n
    # A[k, i] = f[i] if 0 <= k-i < n, else 0
    # i.e., A[k, i] = f[i] if max(0, k-n+1) <= i <= min(n-1, k)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n, k + 1)
        A[k, i_lo:i_hi] = f[i_lo:i_hi]
    
    c = -np.ones(n)  # minimize -sum(g)
    b = M * np.ones(2*n-1)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    
    if res.status == 0:
        return res.x
    else:
        return None

def lp_optimize(f, time_budget, rng):
    """LP-based optimization."""
    n = f.size
    f = f / f.sum()
    best_f = f.copy()
    best_sc = normed_score(f)
    
    t0 = time.monotonic()
    iters = 0
    last_improve = 0
    
    while time.monotonic() - t0 < time_budget:
        b = full_conv(f)
        M = float(np.max(b))
        
        g = solve_lp(f, M)
        if g is None:
            break
        
        g_sum = float(np.sum(g))
        if g_sum <= 1.0 + 1e-6:
            # Near optimal
            if iters - last_improve > 20:
                break
            # Try with slightly larger M
            M = M * 1.001
            g = solve_lp(f, M)
            if g is None or np.sum(g) <= 1.0 + 1e-6:
                break
        
        g = g / g_sum
        
        # Line search for step size
        best_t = 0.0
        for t in [0.5, 0.3, 0.2, 0.1, 0.05]:
            h = (1-t)*f + t*g
            h_sc = normed_score(h)
            if h_sc < M:  # Actually we want h_sc < current score
                best_t = t
                break
        
        if best_t == 0.0:
            # Try smaller step
            best_t = 0.01
        
        f = (1-best_t)*f + best_t*g
        f = np.maximum(f, 0)  # Ensure non-negative
        f = f / f.sum()
        
        cur_sc = normed_score(f)
        if cur_sc < best_sc:
            best_sc = cur_sc
            best_f = f.copy()
            last_improve = iters
        
        iters += 1
        if iters - last_improve > 100:
            # Restart with perturbation
            f = best_f + 0.01 * rng.standard_normal(n)
            f = np.maximum(f, 0)
            f = f / f.sum()
            last_improve = iters
    
    return best_f, best_sc
