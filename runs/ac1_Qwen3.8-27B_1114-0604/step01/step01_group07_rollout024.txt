import numpy as np
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([])
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _score(a):
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = _fft_convolve(a)
    return 2.0 * len(a) * b.max() / (s * s)

def _grad_max_soft(a, T, K=64):
    """Compute soft subgradient of max(b) w.r.t. a."""
    b = _fft_convolve(a)
    # Soft weights
    b_max = b.max()
    weights = np.exp(T * (b - b_max))
    weights /= weights.sum()
    # Get top K for efficiency
    top_k = np.argsort(weights)[-K:]
    # Gradient of b[k] w.r.t. a is: 2 * a[max(0,k-n+1):k+1] reversed
    n = len(a)
    grad = np.zeros(n)
    for k in top_k:

def _compute_soft_grad(a, T=50, K=64):
    """Compute the soft subgradient of max(b) w.r.t. a."""
    b = _fft_convolve(a)
    n = len(a)
    
    # Soft weights (softmax of T*b)
    b_shifted = b - b.max()
    log_weights = T * b_shifted
    # Clip to avoid underflow/overflow
    log_weights = np.clip(log_weights, -50, 50)
    weights = np.exp(log_weights)
    weights /= weights.sum()
    
    # Get indices of top K weights
    top_idx = np.argsort(weights)[-K:]
    top_weights = weights[top_idx]
    top_weights /= top_weights.sum()  # renormalize
    
    # Compute gradient
    grad = np.zeros(n)
    for k_idx, k in enumerate(top_idx):
        k = int(k)
        w = top_weights[k_idx]
        # grad[j] += 2*w*a[k-j] for j where 0 <= k-j < n
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            m = k - np.arange(j_min, j_max + 1)  # k - j for each j
            grad[j_min:j_max+1] += 2.0 * w * a[m]
    
    return grad

"""
Low-peak nonnegative step-function search using soft-subgradient optimization.

Strategy:
1. Generate diverse initial candidates (sinusoidal, multi-harmonic, block, random).
2. Score all candidates internally using FFT-based convolution.
3. Refine top candidates using soft-subgradient descent on the simplex.
4. Use random restarts to escape local minima.
5. Return the best validated sequence.
"""

import numpy as np
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _internal_score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = _fft_convolve(a)
    if b.size == 0:
        return np.inf
    return 2.0 * a.size * b.max() / (s * s)

def _make_sine(n, amp, mode=1, phase=0.5):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * mode * (i + phase) / n
    a = 1.0 + amp * np.sin(theta)
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / n
    return a / s

def _make_multi_harmonic(n, amps, modes, phases):
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for amp, mode, phase in zip(amps, modes, phases):
        theta = 2.0 * np.pi * mode * (i + phase) / n
        a += amp * np.sin(theta)
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / n
    return a / s

def _make_block(n, frac, gap=0):
    """Two-block structure: high, gap, high (or high, low)."""
    a = np.zeros(n)
    L = int(n * frac)
    if gap == 0:
        # High in first L, high in last L
        a[:L] = 1.0
        a[n-L:] = 1.0
    else:
        # High in first L, low in middle, high in last L
        a[:L] = 1.0
        a[n-L:] = 1.0
        mid_start = L
        mid_end = n - L
        if mid_end > mid_start:
            a[mid_start:mid_end] = 0.1
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / n
    return a / s

def _random_smooth(n, n_freq=5, rng=None):
    """Random smooth sequence from low-frequency Fourier components."""
    if rng is None:
        rng = np.random.default_rng()
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for k in range(1, n_freq + 1):
        amp = rng.uniform(0.1, 0.8) / k
        phase = rng.uniform(0, 2*np.pi)
        a += amp * np.sin(2*np.pi*k*i/n + phase)
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / n
    return a / s

def _compute_soft_grad(a, T=30.0, K=64):
    """Soft subgradient of max(b) w.r.t. a."""
    b = _fft_convolve(a)
    n = len(a)
    
    b_shifted = b - b.max()
    log_w = T * b_shifted
    log_w = np.clip(log_w, -50, 50)
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(b)
        w[b.argmax()] = 1.0
    else:
        w /= w_sum
    
    # Top K by weight
    if K < len(w):
        top_idx = np.argpartition(w, -K)[-K:]
    else:
        top_idx = np.arange(len(w))
    top_w = w[top_idx].copy()
    top_w_sum = top_w.sum()
    if top_w_sum < 1e-30:
        return np.zeros(n)
    top_w /= top_w_sum
    
    grad = np.zeros(n)
    for k_int, wv in zip(top_idx, top_w):
        k = int(k_int)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            m = k - np.arange(j_min, j_max + 1)
            grad[j_min:j_max+1] += 2.0 * wv * a[m]
    
    return grad

    rng = np.random.default_rng(seed)
    start = time.monotonic()

    a = np.clip(a, 0, 1000)
    s = a.sum()
    a /= s
    
    best = a.copy()
    best_score = _internal_score(best)
    
    n = len(a)
    it = 0
    last_improve = 0
    
    while time.monotonic() - start < budget_s:
        it += 1
        
        # Adaptive parameters
        T = 20.0 + 20.0 * min(1.0, it / 500)  # ramp up temperature sharpness
        K = 32 + 32 * min(1.0, it / 200)  # ramp up active set size
        lr = 0.5 + 0.5 * min(1.0, it / 100)  # ramp up learning rate
        
        grad = _compute_soft_grad(a, T, int(K))
        grad -= grad.mean()  # project to simplex tangent
        
        # Line search: try a few step sizes
        best_step_score = np.inf
        best_step_a = None
        for factor in [0.5, 1.0, 2.0, 0.25]:
            step = lr * factor * grad
            step = np.clip(step, -1.5, 1.5)
            a_try = a * np.exp(-step)
            a_try = np.clip(a_try, 0, 1000)
            st = a_try.sum()
            if st < 0.01:
                continue
            a_try /= st
            sc = _internal_score(a_try)
            if sc < best_step_score:
                best_step_score = sc
                best_step_a = a_try
        
        if best_step_a is not None and best_step_score < best_score:
            best = best_step_a.copy()
            best_score = best_step_score
            last_improve = it
            a = best_step_a
        elif best_step_a is not None:
            # Small chance of accepting non-improving step (exploration)
            if rng.random() < 0.1 and it - last_improve < 300:

            else:
                # Rejected

        # Random restart if stuck

            st = a.sum()
            if st > 0.01:

            last_improve = it
        
        # Early stop if over 90% of budget used
        if time.monotonic() - start > budget_s * 0.9:

"""
Low-peak nonnegative step-function search using soft-subgradient optimization
with diverse candidate generation and adaptive local refinement.

Strategy:
1. Generate diverse initial candidates (sinusoidal, multi-harmonic, block, random smooth).
2. Score all candidates using FFT-based convolution.
3. Refine top candidates using soft-subgradient descent on the simplex with
   adaptive temperature, active set size, and random restarts.
4. Return the best validated sequence.
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0, 1000)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    b = _fft_convolve(a)
    if b.size == 0:
        return np.inf
    mx = float(b.max())
    return 2.0 * a.size * mx / (s * s)


def _make_sine(n, amp, mode=1, phase=0.5):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * float(mode) * (i + phase) / float(n)
    a = 1.0 + amp * np.sin(theta)
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / n
    return a / s


def _make_multi_harmonic(n, amps, modes, phases):
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for amp, mode, phase in zip(amps, modes, phases):
        theta = 2.0 * np.pi * float(mode) * (i + phase) / float(n)
        a += amp * np.sin(theta)
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / n
    return a / s


def _make_block(n, frac):
    L = max(1, int(n * frac))
    a = np.zeros(n)
    a[:L] = 1.0
    if n - L > 0:
        a[n - L:] = 1.0
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / n
    return a / s


def _random_smooth(n, n_freq=4, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)

        amp = rng.uniform(0.2, 0.9) / k
        phase = rng.uniform(0, 2 * np.pi)
        a += amp * np.sin(2 * np.pi * k * i / n + phase)
    a = np.clip(a, 0, None)
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / n
    return a / s


def _compute_soft_grad(a, T, K):
    b = _fft_convolve(a)
    n = len(a)
    b_shifted = b - b.max()
    log_w = T * b_shifted
    log_w = np.clip(log_w, -60, 60)
    w = np.exp(log_w)
    w_sum = w.sum()
    if w_sum < 1e-30:
        w = np.zeros_like(b)
        w[int(b.argmax())] = 1.0
    else:
        w /= w_sum
    if K < len(w):
        top_idx = np.argpartition(w, -K)[-K:]
    else:
        top_idx = np.arange(len(w))
    top_w = w[top_idx].copy()
    tws = top_w.sum()
    if tws < 1e-30:
        return np.zeros(n)
    top_w /= tws
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        k = int(top_idx[idx])
        wv = top_w[idx]
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            m = k - np.arange(j_min, j_max + 1)
            grad[j_min:j_max + 1] += 2.0 * wv * a[m]
    return grad

        grad -= grad.mean()

        gnorm = float(np.sqrt(np.mean(grad * grad)))

        grad_normed = grad / gnorm

        best_step_score = np.inf
        best_step_a = None

    # Second round of refinement on the current best
    if time.monotonic() < hard_deadline - 80:
        alloc2 = min(400.0, hard_deadline - time.monotonic() - 40)
        if alloc2 > 40:
            print(f"[refine2] Refining best, budget={alloc2:.0f}s, current_score={best_score:.8f}")
            refined2, ref2_score = _local_optimize(best_a, budget_s=alloc2, seed=seed + 500)
            if ref2_score < best_score:
                best_a = refined2
                best_score = ref2_score
                print(f"[refine2] IMPROVED -> score={ref2_score:.8f}")
            else:
                print(f"[refine2] no improvement (got {ref2_score:.8f})")

            # Third round if still time
            if time.monotonic() < hard_deadline - 50:
                alloc3 = min(200.0, hard_deadline - time.monotonic() - 30)
                if alloc3 > 30:
                    print(f"[refine3] budget={alloc3:.0f}s")
                    refined3, ref3_score = _local_optimize(best_a, budget_s=alloc3, seed=seed + 999)
                    if ref3_score < best_score:
                        best_a = refined3
                        best_score = ref3_score
                        print(f"[refine3] IMPROVED -> score={ref3_score:.8f}")

    final = _sanitize(best_a)
    if final is None:
        final = np.ones(2048) / 2048.0

    try:
        real_score = float(evaluate_sequence(final.tolist()))
        print(f"[final] n={final.size} internal_best={best_score:.8f} actual={real_score:.8f}")
    except Exception as e:
        print(f"[final] eval error: {e}, using fallback")
        real_score = np.inf
        final = np.ones(2048) / 2048.0

    return [float(x) for x in final]
