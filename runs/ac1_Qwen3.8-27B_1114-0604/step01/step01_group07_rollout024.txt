Low-peak nonnegative step-function search using optimized sinusoidal
initialization and aggressive subgradient refinement with multiple restarts.

import numpy as np
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_conv(a, b=None):
    if b is None:
        b = a
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    need = a.size + b.size - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)
    return c[:need]

def _fft_convolve(a):
    return _fft_conv(a, a)

def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)

        return np.inf
    c = _fft_convolve(arr)
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * arr.size * mx / (s * s))

    a = (1.0 + amp * np.sin(mode * theta)) / float(n)
    a = np.clip(a, 0.0)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n) / n
    else:
        a = a / s
    return a

def _compute_subgrad(a, c, K):
    n = a.size
    if c.size <= K:
        active = np.arange(c.size, dtype=np.int64)
    else:
        active = np.argpartition(c, -K)[-K:]
    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(k, n - 1)
        if j_hi >= j_lo:
            grad[j_lo:j_hi+1] += 2.0 * a[k-j_hi:k-j_lo+1][::-1]
    grad /= float(len(active))
    return grad
