```
    # center for numerical
    xc = x - np.mean(x)

    # clean finite
    if not finite: val=1e100; grad=zeros
    return val, grad
```

```
def random_smooth_p(n, rng, alpha=1.0, amp=1.0):
    L=1 << (n-1).bit_length()
    half=L//2+1
    g=np.zeros(half, dtype=np.complex128)
    if L > 1:
       k=np.arange(1,half)
       if k.size:
          a=1.0/np.power(k.astype(float), alpha)
          c=a*(rng.normal(size=k.size)+1j*rng.normal(size=k.size))
          if L % 2 == 0:
             c[-1]=np.real(c[-1])
          g[1:]=c
    x=np.fft.irfft(g, L)[:n]
    x=x - float(np.mean(x))
    x=amp*np.tanh(x)
    p=np.exp(x)
    s=p.sum()
    if s<=0 or not finite: p=np.full(n,1/n)
    else p=p/s
    return p
```

```python
"""
Search for nonnegative sequences minimizing
    2*n*max(conv(a,a)) / sum(a)^2.

The code normalizes sum(a)=1 and uses:
  * Fourier-based fast convolutions and smooth-max surrogate objectives;
  * constrained L-BFGS-B in log-ratio variables (no scale redundancy);
  * multiresolution continuation (larger and larger sequences);
  * a final multiplicative/ratio polish balancing the most active
    convolution coefficients.

The starting point height_sequence_1 is used if it is present.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def _fft_size(need):
    """Smallest power of two >= need (need >= 1)."""
    need = max(1, int(need))
    return 1 << (need - 1).bit_length()


def fft_convolution(a):
    """
    Linear convolution a*a. Tiny negative FFT roundoff is clipped to zero.
    """
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)

    need = 2 * n - 1
    size = _fft_size(need)
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    b = np.ascontiguousarray(b, dtype=np.float64)
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    """
    Smooth maximum of b and normalized softmax weights.
    Returns (G, w) where G ~= max(b) and sum(w)=1.
    """
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float("inf"), np.empty(0, dtype=np.float64)

    m = float(np.max(b))
    if not np.isfinite(m):
        return float("inf"), np.zeros_like(b, dtype=np.float64)

    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b, dtype=np.float64)
        w /= float(b.size)
        return m, w

    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))

    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0

    w = e / Z
    G = m + float(np.log(Z)) / beta
    if not np.isfinite(G):
        return m, w
    return float(G), w


def mixed_top_weights(b, beta, top_frac=0.06):
    """
    Smooth-max weights mixed with a uniform weight over the current top
    coefficients.  Used in the final multiplicative polish.
    """
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return np.empty(0, dtype=np.float64)

    if top_frac <= 0.0:
        _, w = smooth_max_and_weights(b, beta)
        return w

    _, w1 = smooth_max_and_weights(b, beta)

    K = max(16, int(b.size * top_frac))
    if K >= b.size:
        w2 = np.ones_like(b, dtype=np.float64) / float(b.size)
    else:
        # Indices of the largest K values.
        idx = np.argpartition(b, b.size - K - 1)[b.size - K:]
        w2 = np.zeros_like(b, dtype=np.float64)
        w2[idx] = 1.0 / float(K)

    w = 0.5 * w1 + 0.5 * w2
    s = float(np.sum(w))
    if s > 0.0 and np.isfinite(s):
        w /= s
    else:
        w = np.ones_like(b, dtype=np.float64) / float(b.size)
    return w


def conv_derivative(p, w):
    """
    For s_i = sum_j p_j w_{i+j}, i=0..n-1, where p has length n and w has
    length 2*n-1.  This is one half of the gradient of sum_k w_k (p*p)_k.
    """
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    if n == 1:
        return np.array([float(np.dot(p, w[:1]))], dtype=np.float64)

    need = n + m - 1
    size = _fft_size(need)

    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]

    # C has indices 0..n+m-2.  s_i = C[m-1-i] for i=0..n-1.
    s = C[m - n:m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)


def normalize_positive(v, target=1.0):
    """Clip to nonnegative and normalize to target sum."""
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return v
    v[~np.isfinite(v)] = 0.0
    v = np.maximum(v, 0.0)
    s = float(np.sum(v))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(v.shape, target / float(v.size), dtype=np.float64)
    v *= float(target) / s
    return v


def full_score(p):
    """
    Authoritative objective for nonnegative p:
        2*n*max(conv(p,p)) / (sum(p))^2.
    """
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float("inf")

    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float("inf")

    b = fft_convolution(p)
    M = float(np.max(b)) if b.size else 0.0
    if not np.isfinite(M):
        return float("inf")

    return float(2.0 * n * M / (S * S))


def symmetrize(p):
    """Return normalized (p + reverse(p)) / 2."""
    p = np.asarray(p, dtype=np.float64)
    if p.size == 0:
        return p
    q = 0.5 * (p + p[::-1])
    return normalize_positive(q)


def resize_sequence(p, n):
    """Linearly resample p to length n and normalize."""
    p = normalize_positive(p)
    m = p.size
    if m == 0:
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    if m == n:
        return p

    x_old = (np.arange(m, dtype=np.float64) - 0.5) / float(m)
    x_new = (np.arange(n, dtype=np.float64) - 0.5) / float(n)
    y = np.interp(x_new, x_old, p)
    return normalize_positive(y)


def random_smooth_profile(n, rng, alpha=1.0, amp=1.0):
    """
    A positive smooth random profile generated by inverse FFT of
    approximately 1/f^alpha noise.
    """
    n = int(n)
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([1.0], dtype=np.float64)

    L = 1 << (n - 1).bit_length()
    half = L // 2 + 1
    g = np.zeros(half, dtype=np.complex128)

    if L > 1:
        k = np.arange(1, half, dtype=np.float64)
        if k.size:
            a = 1.0 / np.power(k, alpha)
            c = a * (rng.normal(size=k.size) + 1j * rng.normal(size=k.size))
            if (L % 2) == 0:
                c[-1] = np.real(c[-1])
            g[1:] = c

    x = np.fft.irfft(g, L)[:n]
    x = x - float(np.mean(x))
    x = amp * np.tanh(x)

    p = np.exp(x)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        p = np.full(n, 1.0 / float(n), dtype=np.float64)
    else:
        p /= s
    return p


def choose_reference(p):
    """
    Choose an index with reasonably large mass to use as the reference in
    log-ratio parameters.  The 75th percentile is robust to zero endpoints.
    """
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n <= 1:
        return 0

    try:
        q = float(np.quantile(p, 0.75))
    except Exception:
        q = float(np.mean(p))

    avg = float(np.mean(p))
    if not np.isfinite(q) or q <= 1e-300:
        q = 0.5 * avg
    if not np.isfinite(q) or q <= 1e-300:
        return n // 2

    idx = int(np.argmin(np.abs(p - q)))
    return idx


def ratio_to_p(r, n, ref):
    """
    Convert log-ratios r (length n-1) to a probability vector of length n,
    where the reference coordinate is fixed to log-ratio 0.
    """
    r = np.asarray(r, dtype=np.float64).reshape(-1)
    n = int(n)
    ref = int(ref)

    if ref < 0:
        ref = 0
    if ref >= n:
        ref = n - 1

    if r.size != n - 1:
        tmp = np.zeros(n - 1, dtype=np.float64)
        m = min(tmp.size, r.size)
        tmp[:m] = r[:m]
        r = tmp

    r = np.where(np.isfinite(r), r, 0.0)

    rfull = np.empty(n, dtype=np.float64)
    if ref > 0:
        rfull[:ref] = r[:ref]
    if ref + 1 < n:
        rfull[ref + 1:] = r[ref:]
    rfull[ref] = 0.0

    maxz = max(0.0, float(np.max(rfull)))
    z = rfull - maxz
    e = np.exp(z)
    S = float(np.sum(e))

    if S <= 0.0 or not np.isfinite(S):
        return np.full(n, 1.0 / float(n), dtype=np.float64)

    p = e / S
    return p


def ratio_from_p(p, ref, bound=10.0):
    """Convert a probability vector to clipped log-ratios relative to ref."""
    p = normalize_positive(p)
    n = p.size
    ref = int(ref)

    if ref < 0:
        ref = 0
    if ref >= n:
        ref = n - 1

    pr = float(p[ref])
    eps = np.finfo(np.float64).tiny
    if pr <= eps:
        pr = eps

    safe_p = np.maximum(p, eps)
    rel = safe_p / pr

    r = np.empty(n - 1, dtype=np.float64)
    if ref > 0:
        r[:ref] = np.log(rel[:ref])
    if ref + 1 < n:
        r[ref:] = np.log(rel[ref + 1:])

    r = np.where(np.isfinite(r), r, 0.0)
    return np.clip(r, -float(bound), float(bound))


def ratio_objective_jac(r, n, ref, beta):
    """
    Objective and gradient in log-ratio variables.

    Minimize the smooth surrogate
        2*n*G,  G = weighted max of p*p,
    with p determined by r and ref.
    """
    n = int(n)
    ref = int(ref)

    if n <= 1:
        return 0.0, np.zeros(0, dtype=np.float64)

    ref0 = ref
    if ref < 0:
        ref = 0
    if ref >= n:
        ref = n - 1

    p = ratio_to_p(r, n, ref)

    b = fft_convolution(p)
    G, w = smooth_max_and_weights(b, beta)

    if not np.isfinite(G):
        return 1e100, np.zeros(n - 1, dtype=np.float64)

    h = conv_derivative(p, w)
    gp = 2.0 * h
    M = float(np.dot(p, gp))

    val = float(2.0 * n * G)
    if not np.isfinite(val):
        return 1e100, np.zeros(n - 1, dtype=np.float64)

    # Gradient with respect to r_i (i != ref) for the softmax/ratio map:
    #     dF/dr_i = 2n * p_i * (gp_i - sum_j p_j gp_j)
    grad_full = 2.0 * n * p * (gp - M)
    grad_full = np.where(np.isfinite(grad_full), grad_full, 0.0)

    grad = np.empty(n - 1, dtype=np.float64)
    if ref > 0:
        grad[:ref] = grad_full[:ref]
    if ref + 1 < n:
        grad[ref:] = grad_full[ref + 1:]
    grad = np.where(np.isfinite(grad), grad, 0.0)

    # Safety clamp for wildly ill-conditioned steps.
    gmax = float(np.max(np.abs(grad))) if grad.size else 0.0
    if np.isfinite(gmax) and gmax > 1e4:
        grad *= 1e4 / gmax

    return val, grad.astype(np.float64)


def eval_time_estimate(n):
    """Rough seconds per objective+gradient evaluation."""
    n = int(n)
    if n <= 2048:
        return 0.005
    if n <= 4096:
        return 0.010
    if n <= 8192:
        return 0.020
    if n <= 16384:
        return 0.040
    if n <= 32768:
        return 0.090
    if n <= 49152:
        return 0.160
    return 0.250


def lqr_optimize(p0, n, beta_list, iter_list, deadline, bound=10.0):
    """
    Run several L-BFGS-B smooth-surrogate stages in log-ratio variables.
    """
    p = normalize_positive(p0)
    n = p.size
    if n < 4:
        return p, full_score(p)

    best_p = p.copy()
    best_score = full_score(p)
    if not np.isfinite(best_score):
        best_p = np.full(n, 1.0 / float(n), dtype=np.float64)
        best_score = full_score(best_p)

    if not HAVE_SCIPY:
        return best_p, best_score

    work_p = best_p.copy()
    work_score = best_score

    for idx, beta in enumerate(beta_list):
        if time.monotonic() >= deadline - 5.0:
            break

        desired = iter_list[idx] if idx < len(iter_list) else 40
        rem = max(0.0, deadline - time.monotonic())
        per = eval_time_estimate(n)
        cap = max(5, int(rem / max(per, 1e-6) / 6.0))
        maxiter = int(min(desired, cap))
        if maxiter < 5:
            break

        # If the current working point is pathological, fall back to best.
        if not np.all(np.isfinite(work_p)) or float(np.sum(work_p)) <= 0.0:
            work_p = best_p.copy()
            work_score = best_score

        ref = choose_reference(work_p)
        r0 = ratio_from_p(work_p, ref, bound)

        bounds = [(-float(bound), float(bound))] * (n - 1)
        options = {
            "maxiter": maxiter,
            "ftol": 1e-10,
            "gtol": 1e-4,
            "maxls": 40,
            "maxcor": 10,
        }

        try:
            res = minimize(
                ratio_objective_jac,
                r0,
                args=(n, ref, float(beta)),
                method="L-BFGS-B",
                jac=True,
                bounds=bounds,
                options=options,
            )
            r1 = np.asarray(res.x, dtype=np.float64)
        except Exception:
            r1 = r0

        if not np.all(np.isfinite(r1)):
            r1 = r0

        p1 = ratio_to_p(r1, n, ref)
        sc1 = full_score(p1)

        if np.isfinite(sc1) and sc1 < best_score - 1e-12:
            best_score = sc1
            best_p = p1
            work_p = p1.copy()
            work_score = sc1
            print(f"    lqr n={n} beta={int(beta)} score={best_score:.8f}")
        else:
            # Allow the optimizer to move a bit through a slightly worse
            # point if it looks like a different basin for the next beta.
            if np.isfinite(sc1) and sc1 <= max(best_score + 1e-9, best_score * 1.10):
                work_p = p1
                work_score = sc1

        # Symmetric variant of the new point can sometimes reduce the peak.
        ps = symmetrize(p1)
        scs = full_score(ps)
        if np.isfinite(scs) and scs < best_score - 1e-12:
            best_score = scs
            best_p = ps
            work_p = ps.copy()
            work_score = scs
            print(f"    lqr sym n={n} beta={int(beta)} score={best_score:.8f}")

    return best_p, best_score


def ratio_polish(p, deadline, beta=None, max_iter=50000, top_frac=0.06):
    """
    Final multiplicative polish.

    Uses weights on the largest convolution coefficients and a damped update
        q_i = p_i * mean(h_i)/h_i,
    where h is a weighted half-gradient.  Accepts only exact improvements.
    """
    p = normalize_positive(p)
    n = p.size
    if n < 4:
        return p, full_score(p)

    if beta is None:
        beta = 128.0 * float(n)

    best_p = p.copy()
    best_score = full_score(p)
    if not np.isfinite(best_score):
        best_p = np.full(n, 1.0 / float(n), dtype=np.float64)
        best_score = full_score(best_p)

    no_imp = 0
    it = 0

    while it < max_iter and time.monotonic() < deadline:
        it += 1

        b = fft_convolution(p)
        if b.size == 0:
            break

        w = mixed_top_weights(b, float(beta), top_frac)
        h = conv_derivative(p, w)
        hm = float(np.mean(h)) if h.size else 0.0

        if not np.isfinite(hm) or hm <= 1e-300:
            break

        z = p * (hm / np.maximum(h, 1e-300))
        z = normalize_positive(z)

        accepted = False
        for rho in (0.25, 0.125, 0.0625, 0.03125):
            q = (1.0 - rho) * p + rho * z
            q = normalize_positive(q)
            sc = full_score(q)

            if np.isfinite(sc) and sc < best_score - 1e-12:
                p = q
                best_p = q.copy()
                best_score = sc
                accepted = True
                no_imp = 0
                if it % 200 == 0:
                    print(f"    ratio_polish n={n} it={it} score={best_score:.8f}")
                break

        if not accepted:
            no_imp += 1
            # Periodically broaden the smooth max and include more active
            # constraints if the current active set is not moving.
            if no_imp % 25 == 0:
                beta = max(float(n), beta * 0.65)
                top_frac = min(0.18, top_frac * 1.10)

            if no_imp > 100:
                break

    return best_p, best_score


def make_starts(n, rng, parent_p=None, max_count=14):
    """Create a diverse but modest collection of starting sequences."""
    starts = []

    if parent_p is not None and parent_p.size > 0:
        pp = resize_sequence(parent_p, n)
        starts.append(pp)
        starts.append(symmetrize(pp))

    starts.append(np.full(n, 1.0 / float(n), dtype=np.float64))

    x = np.arange(n, dtype=np.float64) / float(n)
    for f, a in ((1, 0.8), (2, 1.0), (3, 0.6), (5, 0.4)):
        prof = np.exp(a * np.cos(2.0 * np.pi * f * (x - 0.5)))
        prof /= prof.sum()
        starts.append(prof)
        starts.append(symmetrize(prof))

    for alpha, amp in ((0.6, 1.2), (1.0, 1.0), (2.0, 0.7)):
        rs = random_smooth_profile(n, rng, alpha, amp)
        starts.append(rs)
        starts.append(symmetrize(rs))

    out = []
    for p in starts:
        p = np.asarray(p, dtype=np.float64)
        if p.size == n and np.all(np.isfinite(p)):
            out.append(normalize_positive(p))
        if len(out) >= max_count:
            break
    return out


def top_by_score(starts, k):
    """Return up to k sequences with lowest full_score, no lambdas used."""
    scored = []
    for p in starts:
        sc = full_score(p)
        if np.isfinite(sc):
            scored.append([sc, p])

    chosen = []
    while scored and len(chosen) < max(0, int(k)):
        bi = 0
        bv = scored[0][0]
        for i in range(1, len(scored)):
            if scored[i][0] < bv:
                bi = i
                bv = scored[i][0]
        chosen.append(scored[bi][1])
        del scored[bi]
    return chosen


def run_phase(starts, n, beta_list, iter_list, deadline, max_starts, tag):
    """Run L-BFGS from several starts and keep the best point."""
    best_p = np.full(n, 1.0 / float(n), dtype=np.float64)
    best_score = full_score(best_p)

    if not starts:
        starts = [best_p]

    top = top_by_score(starts, max_starts)
    if not top:
        top = [best_p]

    # Always keep one potentially exploratory start if there is room.
    if len(top) < max_starts and len(starts) >= len(top) + 1:
        # The last generated start is usually random; use it as exploration.
        extra = starts[-1]
        already = False
        for tp in top:
            if tp is extra:
                already = True
                break
        if not already:
            top.append(extra)

    remaining_its = len(top)
    for p0 in top:
        remaining_its -= 1
        rem_total = max(0.0, deadline - time.monotonic())
        if rem_total < 20.0:
            break

        # Allocate remaining phase time nearly equally among remaining starts.
        share = rem_total / float(max(1, remaining_its + 1))
        sub_deadline = time.monotonic() + max(10.0, share)

        p_fit, sc_fit = lqr_optimize(p0, n, beta_list, iter_list, sub_deadline)
        if np.isfinite(sc_fit) and sc_fit < best_score - 1e-12:
            best_score = sc_fit
            best_p = p_fit.copy()

        ps = symmetrize(p_fit)
        scs = full_score(ps)
        if np.isfinite(scs) and scs < best_score - 1e-12:
            best_score = scs
            best_p = ps.copy()

        print(f"phase {tag} n={n} best={best_score:.8f} time_used={time.monotonic()- (deadline-betaguard):.1f}".replace("betaguard", "0.0"))

    return best_p, best_score


def final_polish(p, deadline):
    """Symmetrization + ratio polish + a short high-beta L-BFGS finish."""
    p = normalize_positive(p)
    n = p.size
    if n < 4:
        return p, full_score(p)

    best_p = p.copy()
    best_score = full_score(p)

    ps = symmetrize(p)
    scs = full_score(ps)
    if np.isfinite(scs) and scs < best_score - 1e-12:
        best_score = scs
        best_p = ps.copy()

    if time.monotonic() < deadline - 30.0:
        p2, sc2 = ratio_polish(best_p, min(deadline, time.monotonic() + 180.0),
                               beta=128.0 * float(n), max_iter=20000,
                               top_frac=0.06)
        if np.isfinite(sc2) and sc2 < best_score - 1e-12:
            best_score = sc2
            best_p = p2.copy()

    if time.monotonic() < deadline - 45.0:
        beta_list = [128.0 * n, 256.0 * n]
        iter_list = [80, 120]
        p3, sc3 = lqr_optimize(best_p, n, beta_list, iter_list, deadline)
        if np.isfinite(sc3) and sc3 < best_score - 1e-12:
            best_score = sc3
            best_p = p3.copy()

        ps3 = symmetrize(p3)
        scs3 = full_score(ps3)
        if np.isfinite(scs3) and scs3 < best_score - 1e-12:
            best_score = scs3
            best_p = ps3.copy()

    return best_p, best_score


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search entry point.

    Returns a list of nonnegative floats minimizing the evaluator.
    """
    start_time = time.monotonic()

    try:
        seed = int(seed)
    except Exception:
        seed = 42

    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 0.0:
        return [1.0]

    if budget_s < 600.0:
        reserve = 20.0
    elif budget_s < 1500.0:
        reserve = 60.0
    else:
        reserve = 120.0

    work_budget = max(1.0, budget_s - reserve)
    deadline = start_time + work_budget
    rng = np.random.default_rng(seed)

    # Load the previously found construction if available.
    parent_p = None
    try:
        parent_obj = globals().get("height_sequence_1")
        if parent_obj is not None:
            parent_arr = np.asarray(parent_obj, dtype=np.float64)
            if parent_arr.size > 0:
                parent_p = normalize_positive(parent_arr)
    except Exception:
        parent_p = None

    # Choose a multiresolution plan.  40960 is a useful compromise between
    # search attack surface and final O(n^2) evaluation cost.
    if budget_s >= 1800.0:
        N1, N2, N3, N4 = 8192, 16384, 32768, 40960
        f1, f2, f3, f4 = 0.12, 0.22, 0.40, 0.26
    elif budget_s >= 900.0:
        N1, N2, N3, N4 = 4096, 8192, 16384, 32768
        f1, f2, f3, f4 = 0.16, 0.26, 0.38, 0.20
    elif budget_s >= 400.0:
        N1, N2, N3, N4 = 2048, 4096, 8192, 16384
        f1, f2, f3, f4 = 0.20, 0.30, 0.30, 0.20
    else:
        N1, N2, N3, N4 = 1024, 2048, 4096, None
        f1, f2, f3, f4 = 0.30, 0.30, 0.40, 0.0

    total_f = f1 + f2 + f3 + f4
    if total_f <= 0.0:
        total_f = 1.0
    f1 /= total_f
    f2 /= total_f
    f3 /= total_f
    f4 /= total_f

    d1 = start_time + work_budget * f1
    d2 = start_time + work_budget * (f1 + f2)
    d3 = start_time + work_budget * (f1 + f2 + f3)
    d4 = deadline if (N4 is not None and f4 > 0.0) else d3

    parent_N1 = resize_sequence(parent_p, N1) if parent_p is not None else parent_p
    starts1 = make_starts(N1, rng, parent_N1, max_count=14)
    p_best, score_best = run_phase(
        starts1, N1,
        [2.0 * N1, 8.0 * N1, 32.0 * N1, 128.0 * N1],
        [20, 35, 55, 80],
        d1,
        5,
        "coarse1"
    )

    parent_N2 = resize_sequence(parent_p, N2) if parent_p is not None else parent_p
    starts2 = [
        p_best,
        symmetrize(p_best),
    ]
    if parent_N2 is not None:
        starts2.append(parent_N2)
        starts2.append(symmetrize(parent_N2))
    starts2.append(random_smooth_profile(N2, rng, 1.0, 1.0))
    p_best, score_best = run_phase(
        starts2, N2,
        [2.0 * N2, 8.0 * N2, 32.0 * N2, 128.0 * N2],
        [30, 45, 70, 100],
        d2,
        4,
        "coarse2"
    )

    parent_N3 = resize_sequence(parent_p, N3) if parent_p is not None else parent_p
    starts3 = [
        p_best,
        symmetrize(p_best),
    ]
    if parent_N3 is not None:
        starts3.append(parent_N3)
    starts3.append(random_smooth_profile(N3, rng, 0.8, 1.1))
    p_best, score_best = run_phase(
        starts3, N3,
        [2.0 * N3, 8.0 * N3, 32.0 * N3, 128.0 * N3, 256.0 * N3],
        [40, 60, 100, 150, 200],
        d3,
        3,
        "fine1"
    )

    if N4 is not None and time.monotonic() < d4 - 30.0:
        # Upsample the best smaller-length sequence and optionally the parent.
        starts4 = [
            resize_sequence(p_best, N4),
            symmetrize(resize_sequence(p_best, N4)),
            random_smooth_profile(N4, rng, 1.0, 1.0),
        ]
        if parent_p is not None:
            starts4.append(resize_sequence(parent_p, N4))

        p4, s4 = run_phase(
            starts4, N4,
            [4.0 * N4, 16.0 * N4, 64.0 * N4, 128.0 * N4, 256.0 * N4],
            [50, 80, 130, 190, 240],
            d4,
            2,
            "fine2"
        )
        if np.isfinite(s4) and s4 < score_best - 1e-12:
            score_best = s4
            p_best = p4.copy()

    # Final polish only if there is enough time to avoid being killed.
    if time.monotonic() < deadline - 10.0:
        pf, sf = final_polish(p_best, min(deadline, time.monotonic() + 120.0))
        if np.isfinite(sf) and sf < score_best - 1e-12:
            score_best = sf
            p_best = pf.copy()

    # Last symmetrization check.
    ps = symmetrize(p_best)
    scs = full_score(ps)
    if np.isfinite(scs) and scs < score_best - 1e-12:
        score_best = scs
        p_best = ps.copy()

    # Sanity normalization.
    try:
        p_best = np.asarray(p_best, dtype=np.float64)
    except Exception:
        p_best = np.array([1.0], dtype=np.float64)

    p_best = np.where(np.isfinite(p_best), p_best, 0.0)
    p_best = np.maximum(p_best, 0.0)
    p_best = np.minimum(p_best, 1000.0)

    s = float(np.sum(p_best))
    if s <= 0.0 or not np.isfinite(s):
        return [1.0]

    p_best *= 1.0 / s

    # If the internal score is unexpectedly bad, fall back to a valid uniform
    # sequence of a safe length.
    if not np.isfinite(score_best) or score_best > 2.0 + 1e-7:
        return [float(1.0) / 8192.0] * 8192

    print(f"final length={int(p_best.size)} internal_score={float(full_score(p_best)):.8f}")
    return [float(x) for x in p_best]
```
