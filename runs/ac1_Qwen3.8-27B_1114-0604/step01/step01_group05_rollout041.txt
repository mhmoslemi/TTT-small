def fft_linear_convolution(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size, n=size)
    b = b[:need]
    b = np.ascontiguousarray(b, dtype=np.float64)
    # remove tiny negative roundoff
    np.maximum(b, 0.0, out=b)
    return b

```python
"""
Search for non-negative step-function heights minimizing
    2*n*max(self-convolution) / (sum(heights))^2

The objective is scale invariant.  This implementation uses:
  1. FFT-constrained mirror/exponentiated-gradient polish in log-space.
  2. L-BFGS-B optimization over softmax-log parameters on the full length.
  3. Multi-resolution block-constant L-BFGS warm starts.
  4. A linearized alternating move (fixed-p convolution subproblem) to escape
     some local basins.
  5. The previously found height_sequence_1, when present, as a strong
     reference/fallback and interpolation source for larger lengths.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


# ----------------------------------------------------------------------
# FFT primitives
# ----------------------------------------------------------------------

def fft_convolve_square(a):
    """Linear convolution a*a by FFT, with small negative roundoff clipped."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)

    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1

    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, n=size)
    b = b[:need]
    b = np.ascontiguousarray(b, dtype=np.float64)
    np.maximum(b, 0.0, out=b)
    return b


def fft_convolve_general(x, y):
    """Linear convolution x*y by FFT, with small negative roundoff clipped."""
    x = np.asarray(x, dtype=np.float64)
    y = np.asarray(y, dtype=np.float64)
    n = x.size
    m = y.size
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)

    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1

    X = np.fft.rfft(x, size)
    Y = np.fft.rfft(y, size)
    c = np.fft.irfft(X * Y, n=size)
    c = c[:need]
    c = np.ascontiguousarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c


def smooth_max_stats(b, beta):
    """
    Return (max_b, smooth_max_b, softmax_weights) for b.
    smooth_max_b uses a log-sum-exp approximation to the maximum.
    """
    b = np.asarray(b, dtype=np.float64)
    m = b.size
    if m == 0:
        return 0.0, 0.0, np.empty(0, dtype=np.float64)

    M = float(np.max(b))
    if not np.isfinite(M):
        return M, M, np.zeros(m, dtype=np.float64)

    if M <= 1e-300 or beta <= 0.0:
        w = np.ones(m, dtype=np.float64) / float(m)
        return M, M, w

    x = beta * (b - M)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))

    if not np.isfinite(Z) or Z <= 0.0:
        w = np.zeros(m, dtype=np.float64)
        idx = int(np.argmax(b))
        w[idx] = 1.0
        G = M
    else:
        G = M + float(np.log(Z)) / float(beta)
        w = e / Z

    if not np.any(w > 0.0):
        w = np.zeros(m, dtype=np.float64)
        idx = int(np.argmax(b))
        w[idx] = 1.0
        G = M

    return M, float(G), w


def correlation_pw(p, w):
    """
    Return s_i = sum_j p_j w_{i+j} for i = 0..len(p)-1.
    This is the subgradient kernel for a weighted self-convolution maximum.
    """
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.empty(0, dtype=np.float64)

    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1

    P = np.fft.rfft(p, size)
    W = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * W, n=size)

    start = m - n
    if start < 0:
        start = 0
    end = m
    if end > need:
        end = need

    s = C[start:end].copy()
    if s.size < n:
        tmp = np.zeros(n, dtype=np.float64)
        tmp[:s.size] = s
        s = tmp
    else:
        s = s[:n]

    s = s[::-1]
    s = np.ascontiguousarray(s, dtype=np.float64)
    np.maximum(s, 0.0, out=s)
    return s


# ----------------------------------------------------------------------
# Normalization / utilities
# ----------------------------------------------------------------------

def normalize_simplex(v):
    """Clip to non-negative and normalize to sum 1."""
    v = np.asarray(v, dtype=np.float64).copy()
    if v.size == 0:
        return v
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = float(np.sum(v))
    if s <= 1e-300 or not np.isfinite(s):
        return np.full(v.shape, 1.0 / float(v.size), dtype=np.float64)
    v *= 1.0 / s
    return v


def add_uniform_floor(p, frac=1e-6):
    """Mix in a tiny uniform component so log-space search stays positive."""
    p = normalize_simplex(p)
    n = p.size
    if n == 0:
        return p
    if frac <= 0.0:
        return p
    q = (1.0 - frac) * p + (frac / float(n)) * np.ones(n, dtype=np.float64)
    return normalize_simplex(q)


def fast_score(p):
    """
    Fast FFT score equivalent to evaluate_sequence for finite nonnegative p.
    """
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float("inf")

    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    S = float(np.sum(p))
    if S <= 1e-300 or not np.isfinite(S):
        return float("inf")

    b = fft_convolve_square(p)
    if b.size == 0:
        return float("inf")

    M = float(np.max(b))
    if not np.isfinite(M):
        return float("inf")

    return float(2.0 * n * M / (S * S))


def interpolate_sequence(seq, N):
    """Interpolate / downsample a reference sequence to length N and normalize."""
    arr = np.asarray(seq, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)

    if N <= 0:
        return np.empty(0, dtype=np.float64)
    if N == 1:
        return np.ones(1, dtype=np.float64)

    s = float(np.sum(arr))
    if arr.size == 0 or s <= 1e-300 or not np.isfinite(s):
        return np.full(N, 1.0 / float(N), dtype=np.float64)

    if arr.size == 1:
        return np.full(N, 1.0 / float(N), dtype=np.float64)

    if arr.size == N:
        return normalize_simplex(arr)

    oldx = np.linspace(0.0, 1.0, arr.size, dtype=np.float64)
    newx = np.linspace(0.0, 1.0, N, dtype=np.float64)
    out = np.interp(newx, oldx, arr / s)
    return normalize_simplex(out)


def load_reference_sequence():
    """Load the previously supplied height_sequence_1 if present."""
    seq = globals().get("height_sequence_1", None)
    if seq is None:
        return None
    try:
        arr = np.asarray(seq, dtype=np.float64)
    except Exception:
        return None

    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    if arr.size > 0 and float(np.sum(arr)) > 1e-300:
        return arr
    return None

def theta_to_p(x, n):
    """Convert x (first n-1 log-entries relative to entry 0) to probabilities."""
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.ones(1, dtype=np.float64)

    theta = np.zeros(n, dtype=np.float64)
    x = np.asarray(x, dtype=np.float64).reshape(-1)
    m = min(n - 1, x.size)
    if m > 0:
        theta[1:1 + m] = x[:m]

    if not np.all(np.isfinite(theta)):
        return np.full(n, 1.0 / float(n), dtype=np.float64)

    theta = np.clip(theta, -50.0, 50.0)
    theta -= float(np.max(theta))
    p = np.exp(theta)
    s = float(np.sum(p))
    if s <= 1e-300 or not np.isfinite(s):
        return np.full(n, 1.0 / float(n), dtype=np.float64)
    p *= 1.0 / s
    return p


def p_to_x(p, n):
    """Convert probabilities to x = log(theta)-log(theta[0]), clipped."""
    if n <= 1:
        return np.empty(0, dtype=np.float64)

    p = normalize_simplex(p)
    floor = 1e-12 / float(n)
    p = np.maximum(p, floor)

    z = np.log(p)
    z = z - z[0]
    x = z[1:]
    return np.clip(x, -35.0, 35.0)


def full_objective(x, n, beta):
    """Objective and gradient for full softmax parameters."""
    if n <= 0:
        return float("inf"), np.empty(0, dtype=np.float64)
    if n == 1:
        p = np.ones(1, dtype=np.float64)
        b = fft_convolve_square(p)
        M, G, _ = smooth_max_stats(b, beta)
        return float(n * G), np.empty(0, dtype=np.float64)

    p = theta_to_p(x, n)
    b = fft_convolve_square(p)
    M, G, w = smooth_max_stats(b, beta)

    if not np.isfinite(G):
        return float("inf"), np.zeros(n - 1, dtype=np.float64)

    s = correlation_pw(p, w)
    grad_p = 2.0 * float(n) * s
    gp = float(np.dot(p, grad_p))
    grad = grad_p - gp
    grad = grad[1:]
    grad = np.where(np.isfinite(grad), grad, 0.0)

    return float(n * G), grad.astype(np.float64)


def run_full_lbfgs(p0, n, beta, maxiter):
    """Run a bounded L-BFGS-B step in the softmax log representation."""
    if not HAVE_SCIPY or n <= 1:
        p0 = normalize_simplex(p0)
        return p0, fast_score(p0)

    p0 = add_uniform_floor(p0, 1e-6)
    x0 = p_to_x(p0, n)
    nvars = n - 1
    if nvars == 0:
        return p0, fast_score(p0)

    x0 = np.clip(x0, -35.0, 35.0)
    bounds = [(-40.0, 40.0)] * nvars
    options = {
        "maxiter": max(1, int(maxiter)),
        "ftol": 1e-12,
        "gtol": 1e-6,
        "maxls": 20,
        "maxcor": 20,
    }

    try:
        res = minimize(
            full_objective,
            x0,
            args=(n, beta),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options=options,
        )
        x = np.asarray(res.x, dtype=np.float64)
    except Exception:
        x = x0

    if x.size != nvars or not np.all(np.isfinite(x)):
        x = x0

    x = np.clip(x, -40.0, 40.0)
    p = theta_to_p(x, n)
    p = normalize_simplex(p)
    sc = fast_score(p)

    if not np.isfinite(sc):
        p = normalize_simplex(p0)
        sc = fast_score(p)
    return p, sc

def block_to_p(n, B, x):
    """Convert block log-weights to a full sequence."""
    if n <= 0:
        return np.empty(0, dtype=np.float64)
    if B <= 1 or n % B != 0:
        return np.full(n, 1.0 / float(n), dtype=np.float64)

    L = n // B
    z = np.zeros(B, dtype=np.float64)
    x = np.asarray(x, dtype=np.float64).reshape(-1)
    if x.size >= B - 1:
        z[:-1] = np.clip(x[:B - 1], -12.0, 12.0)
    else:
        if x.size > 0:
            z[:x.size] = np.clip(x, -12.0, 12.0)

    z = z - float(np.max(z))
    wb = np.exp(z)
    ws = float(np.sum(wb))
    if ws <= 0.0 or not np.isfinite(ws):
        wb = np.ones(B, dtype=np.float64) / float(B)
    else:
        wb /= ws

    H = wb / float(L)
    return np.repeat(H, L)


def p_to_block_x(p, n, B):
    """Derive block log-weights from a full sequence."""
    if B <= 1 or n % B != 0:
        return np.empty(max(0, B - 1), dtype=np.float64)

    p = normalize_simplex(p)
    L = n // B
    arr = np.asarray(p, dtype=np.float64).reshape(B, L)
    probs = arr.sum(axis=1)
    probs = np.maximum(probs, 1e-300)

    x = np.log(probs[:-1] / probs[-1])
    return np.clip(x, -8.0, 8.0)


def block_objective(x, n, B, beta):
    """Objective and gradient for block-constant log weights."""
    if n <= 0:
        return float("inf"), np.empty(max(0, B - 1), dtype=np.float64)
    if B <= 1:
        p = np.full(n, 1.0 / float(n), dtype=np.float64)
        b = fft_convolve_square(p)
        _, G, _ = smooth_max_stats(b, beta)
        return float(n * G), np.empty(0, dtype=np.float64)

    if n % B != 0:
        return float("inf"), np.zeros(B - 1, dtype=np.float64)

    L = n // B
    z = np.zeros(B, dtype=np.float64)
    x = np.asarray(x, dtype=np.float64).reshape(-1)
    if x.size != B - 1:
        tmp = np.zeros(B - 1, dtype=np.float64)
        m = min(B - 1, x.size)
        if m > 0:
            tmp[:m] = x[:m]
        x = tmp
    z[:-1] = np.clip(x, -12.0, 12.0)
    z[-1] = 0.0

    if not np.all(np.isfinite(z)):
        return float("inf"), np.zeros(B - 1, dtype=np.float64)

    z = z - float(np.max(z))
    wb = np.exp(z)
    ws = float(np.sum(wb))
    if ws <= 0.0 or not np.isfinite(ws):
        wb = np.ones(B, dtype=np.float64) / float(B)
    else:
        wb /= ws

    p = np.repeat(wb / float(L), L)
    b = fft_convolve_square(p)
    _, G, w = smooth_max_stats(b, beta)

    if not np.isfinite(G):
        return float("inf"), np.zeros(B - 1, dtype=np.float64)

    s = correlation_pw(p, w)
    g = 2.0 * float(n) * s
    Gsum = g.reshape(B, L).sum(axis=1)
    Wavg = float(np.dot(wb, Gsum))
    gradz = (wb / float(L)) * (Gsum - Wavg)

    grad = gradz[:-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(n * G), grad.astype(np.float64)


def run_block_lbfgs(x0, n, B, beta, maxiter):
    """Run L-BFGS-B on block log weights."""
    if B <= 1:
        return np.empty(0, dtype=np.float64)

    x0 = np.asarray(x0, dtype=np.float64).reshape(-1)
    expected = B - 1
    if x0.size != expected:
        tmp = np.zeros(expected, dtype=np.float64)
        m = min(expected, x0.size)
        if m > 0:
            tmp[:m] = x0[:m]
        x0 = tmp

    x0 = np.clip(x0, -8.0, 8.0)
    if not HAVE_SCIPY:
        return x0

    bounds = [(-8.0, 8.0)] * expected
    options = {
        "maxiter": max(1, int(maxiter)),
        "ftol": 1e-12,
        "gtol": 1e-5,
        "maxls": 20,
        "maxcor": 12,
    }

    try:
        res = minimize(
            block_objective,
            x0,
            args=(n, B, beta),
            method="L-BFGS-B",
            jac=True,
            bounds=bounds,
            options=options,
        )
        x = np.asarray(res.x, dtype=np.float64)
    except Exception:
        x = x0

    if x.size != expected or not np.all(np.isfinite(x)):
        x = x0

    return np.clip(x, -8.0, 8.0)


def block_search(n, deadline, rng, ref_p):
    """
    Multi-resolution piecewise-constant optimization.
    Used primarily to obtain good starting patterns before full-length work.
    """
    best_p = normalize_simplex(ref_p)
    best_p = add_uniform_floor(best_p, 1e-7)
    best_score = fast_score(best_p)

    if not HAVE_SCIPY:
        return best_p, best_score

    B_list = []
    for B in (4, 8, 16, 32, 64, 128, 256, 512, 1024):
        if B < n and n % B == 0:
            B_list.append(B)

    for B in B_list:
        if time.monotonic() >= deadline:
            break

        inits = []
        x_best = p_to_block_x(best_p, n, B)
        inits.append(x_best)

        x_ref = p_to_block_x(ref_p, n, B)
        if x_ref.size == max(0, B - 1):
            inits.append(x_ref)

        # Simple monotone log-ramp directions.
        for a in (-1.0, 1.0):
            z = a * np.linspace(0.0, 1.0, B, dtype=np.float64)
            x = np.clip(z[:-1] - z[-1], -8.0, 8.0)
            inits.append(x)

        if B <= 256:
            inits.append(np.clip(0.8 * rng.normal(0.0, 1.0, B - 1), -8.0, 8.0))

        beta = 80000.0 if n >= 10000 else 50000.0
        maxit = 38 if B > 128 else 55

        for x0 in inits[:4]:
            if time.monotonic() >= deadline:
                break
            x = run_block_lbfgs(x0, n, B, beta, maxit)
            p = block_to_p(n, B, x)
            p = normalize_simplex(p)
            sc = fast_score(p)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_p = p.copy()

        # A short sharpening stage from the current best.
        if time.monotonic() < deadline:
            x = p_to_block_x(best_p, n, B)
            x2 = run_block_lbfgs(x, n, B, 250000.0, 22)
            p = block_to_p(n, B, x2)
            p = normalize_simplex(p)
            sc = fast_score(p)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_p = p.copy()

    return best_p, best_score


# ----------------------------------------------------------------------
# Structured randomized starts
# ----------------------------------------------------------------------

def structured_starts(n, rng, ref_p=None, max_count=30):
    """Create low-cost diverse initial sequences."""
    starts = []
    if n <= 0:
        return starts

    starts.append(np.full(n, 1.0 / float(n), dtype=np.float64))

    if ref_p is not None:
        rp = normalize_simplex(ref_p)
        if rp.size == n:
            starts.append(rp)
            base = np.maximum(rp, 1e-12 / float(n))
            for sig in (0.08, 0.22):
                z = np.log(base) + sig * rng.normal(0.0, 1.0, n)
                z = np.clip(z, -30.0, 30.0)
                z = z - float(np.max(z))
                starts.append(normalize_simplex(np.exp(z)))

    if n > 1:
        x = np.linspace(0.0, 1.0, n, dtype=np.float64)

        # Monotone log-linear profiles.
        for a in (-6.0, -4.0, -2.0, 2.0, 4.0, 6.0):
            z = a * (x - 0.5)
            z = z - float(np.max(z))
            starts.append(normalize_simplex(np.exp(z)))

        # Sinusoidal, nonnegative profiles.
        phases = (0.0, 0.5 * np.pi, np.pi, 1.5 * np.pi)
        for amp in (0.5, 0.9):
            for ph in phases:
                p = 1.0 + amp * np.sin(2.0 * np.pi * x + ph)
                p = np.maximum(p, 1e-8)
                starts.append(normalize_simplex(p))

        # Two-level block profiles.
        half = max(1, n // 2)
        for eps in (0.4, 0.7):
            p = np.ones(n, dtype=np.float64)
            p[:half] = 1.0 + eps
            p[half:] = max(1e-6, 1.0 - eps)
            starts.append(normalize_simplex(p))

            p = np.ones(n, dtype=np.float64)
            p[:half] = max(1e-6, 1.0 - eps)
            p[half:] = 1.0 + eps
            starts.append(normalize_simplex(p))

    # Random smooth Fourier profiles.
    for _ in range(3):
        size = 1
        while size < n:
            size <<= 1
        zc = rng.normal(size=size) + 1j * rng.normal(size=size)
        z = np.real(np.fft.ifft(zc))[:n]
        z = np.where(np.isfinite(z), z, 0.0)
        z -= float(np.mean(z))
        z = 1.5 * np.tanh(z)
        z = z - float(np.max(z))
        starts.append(normalize_simplex(np.exp(z)))

    if len(starts) > max_count:
        starts = starts[:max_count]
    return starts


def choose_top_starts(starts, count):
    """Select up to count starts with lowest fast FFT score."""
    scored = []
    for idx, p in enumerate(starts):
        sc = fast_score(p)
        if np.isfinite(sc):
            scored.append((sc, idx, p))
    scored.sort()

    top = []
    for _, _, p in scored:
        top.append(p)
        if len(top) >= count:
            break
    return top


# ----------------------------------------------------------------------
# Exact-max mirror polish
# ----------------------------------------------------------------------

def mirror_polish_exact(p0, deadline, alpha_start=0.12):
    """
    Multiplicative-weights polish directly aimed at decreasing the exact
    maximum convolution coefficient.
    """
    if p0 is None or p0.size == 0:
        return p0, float("inf")

    p = add_uniform_floor(p0, 1e-8)
    n = p.size
    b = fft_convolve_square(p)
    M = float(np.max(b)) if b.size else float("inf")

    if not np.isfinite(M):
        return p, float("inf")

    best_p = p.copy()
    best_score = 2.0 * float(n) * M
    alpha = alpha_start
    no_improve = 0
    it = 0
    max_iter = 1000000

    while time.monotonic() < deadline and it < max_iter:
        it += 1

        if 2.0 * float(n) * M < best_score - 1e-12:
            best_score = 2.0 * float(n) * M
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 2500:
            break

        tol = max(1e-12, 1e-6 * M)
        top = np.flatnonzero(b >= M - tol)
        if top.size == 0:
            top = np.array([int(np.argmax(b))], dtype=np.int64)

        w = np.zeros(b.size, dtype=np.float64)
        w[top] = 1.0 / float(top.size)

        s = correlation_pw(p, w)
        if M <= 1e-300:
            break

        v = s / M
        v -= float(np.dot(p, v))  # constant shift does not affect normalized multiplicative step
        v = np.clip(v, -50.0, 50.0)

        accepted = False
        tol_accept = max(1e-14, 1e-10 * M)
        for factor in (1.0, 0.7, 0.5, 0.3, 0.2, 0.1):
            a = alpha * factor
            if a <= 1e-9:
                break

            p_new = p * np.exp(-a * v)
            p_new = normalize_simplex(p_new)

            b_new = fft_convolve_square(p_new)
            M_new = float(np.max(b_new)) if b_new.size else float("inf")
            if not np.isfinite(M_new):
                continue

            if M_new < M - tol_accept:
                p = p_new
                b = b_new
                M = M_new
                accepted = True
                break

        if accepted:
            if alpha < 0.35:
                alpha = min(0.35, alpha * 1.12)
        else:
            alpha *= 0.5
            if alpha < 1e-5:
                break

    return best_p, best_score


# ----------------------------------------------------------------------
# Linearized alternating move
# ----------------------------------------------------------------------

def solve_linear_subproblem(p, deadline, max_iter=80, alpha_start=0.20):
    """
    Given fixed p, find q approximating argmin_{q>=0,sum=1} max(p*q).
    This is a convex LP; we solve it with mirrored multiplicative updates.
    """
    n = p.size
    if n <= 0:
        return np.empty(0, dtype=np.float64), float("inf")

    q = add_uniform_floor(p, 1e-7)
    q = normalize_simplex(0.7 * q + 0.3 / float(n))

    c = fft_convolve_general(p, q)
    M = float(np.max(c)) if c.size else float("inf")
    if not np.isfinite(M):
        return q, float("inf")

    alpha = alpha_start
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and it < max_iter:
        it += 1

        tol = max(1e-12, 1e-6 * M)
        top = np.flatnonzero(c >= M - tol)
        if top.size == 0:
            top = np.array([int(np.argmax(c))], dtype=np.int64)

        w = np.zeros(c.size, dtype=np.float64)
        w[top] = 1.0 / float(top.size)

        s = correlation_pw(p, w)
        if M <= 1e-300:
            break

        v = s / M
        v -= float(np.dot(q, v))
        v = np.clip(v, -50.0, 50.0)

        accepted = False
        tol_accept = max(1e-14, 1e-10 * M)
        for factor in (1.0, 0.6, 0.4, 0.25, 0.15, 0.08):
            a = alpha * factor
            if a <= 1e-9:
                break

            q_new = q * np.exp(-a * v)
            q_new = normalize_simplex(q_new)

            c_new = fft_convolve_general(p, q_new)
            M_new = float(np.max(c_new)) if c_new.size else float("inf")
            if not np.isfinite(M_new):
                continue

            if M_new < M - tol_accept:
                q = q_new
                c = c_new
                M = M_new
                accepted = True
                break

        if accepted:
            no_improve = 0
            if alpha < 0.5:
                alpha = min(0.5, alpha * 1.08)
        else:
            no_improve += 1
            alpha *= 0.5
            if alpha < 1e-4 or no_improve > 30:
                break

    return q, M


def alternating_linear_refine(p0, deadline, max_outer=6, inner_iters=80):
    """
    Use the fixed-p linear subproblem as a basin-hopping move.
    """
    if p0 is None or p0.size == 0:
        return p0, float("inf")

    p = add_uniform_floor(p0, 1e-8)
    n = p.size

    b = fft_convolve_square(p)
    M = float(np.max(b)) if b.size else float("inf")
    if not np.isfinite(M):
        return p, float("inf")

    best_p = p.copy()
    best_score = 2.0 * float(n) * M

    outer = 0
    while outer < max_outer and time.monotonic() < deadline:
        outer += 1

        q, _ = solve_linear_subproblem(p, deadline, inner_iters=inner_iters)
        q = normalize_simplex(q)
        sc_q = fast_score(q)

        improved = False

        # Try mixtures; smaller t is safer because q*q may still be high.
        for t in (0.30, 0.50, 0.20, 0.70, 0.10):
            if time.monotonic() >= deadline:
                break
            p_new = normalize_simplex((1.0 - t) * p + t * q)
            sc = fast_score(p_new)
            if np.isfinite(sc) and sc < best_score - 1e-12:
                p = p_new
                best_score = sc
                best_p = p.copy()
                improved = True
                break

        if not improved:
            if np.isfinite(sc_q) and sc_q < best_score - 1e-12:
                p = q.copy()
                best_score = sc_q
                best_p = p.copy()
            else:
                break

    return best_p, best_score


# ----------------------------------------------------------------------
# One-length optimization driver
# ----------------------------------------------------------------------

def optimize_one_length(n, ref_p, deadline, rng, extra_p=None):
    """
    Optimize one length n under an absolute deadline.
    """
    if n <= 0:
        return np.empty(0, dtype=np.float64), float("inf")

    if ref_p is None:
        ref_p = np.full(n, 1.0 / float(n), dtype=np.float64)
    ref_p = normalize_simplex(ref_p)

    best_p = ref_p.copy()
    best_score = fast_score(ref_p)

    if extra_p is not None:
        extra_p = interpolate_sequence(extra_p, n)
        sc_extra = fast_score(extra_p)
        if np.isfinite(sc_extra) and sc_extra < best_score:
            best_score = sc_extra
            best_p = extra_p.copy()

    best_p = add_uniform_floor(best_p, 1e-7)

    t_start = time.monotonic()
    avail = max(0.0, float(deadline - t_start))
    if avail < 25.0 or n < 4:
        return best_p, best_score

    extra_refs = []

    # ---- Block-constant warm start ----
    if HAVE_SCIPY:
        if avail > 300.0:
            block_alloc = min(300.0, 0.20 * avail)
        else:
            block_alloc = 0.28 * avail

        if block_alloc > 15.0:
            block_deadline = t_start + block_alloc
            pb, scb = block_search(n, block_deadline, rng, best_p)
            if np.isfinite(scb) and scb < best_score:
                best_score = scb
                best_p = pb.copy()

    # ---- Optional alternating linearized basin-hopping ----
    if avail > 700.0:
        alt_alloc = min(190.0, 0.11 * avail)
        alt_deadline = time.monotonic() + alt_alloc
        if time.monotonic() < alt_deadline:
            pa, sca = alternating_linear_refine(
                best_p,
                alt_deadline,
                max_outer=5,
                inner_iters=70,
            )
            if np.isfinite(sca):
                extra_refs.append(pa.copy())
                if sca < best_score:
                    best_score = sca
                    best_p = pa.copy()

    # ---- Full-length L-BFGS from several starts ----
    full_deadline = t_start + min(avail - 15.0, 0.78 * avail)
    if HAVE_SCIPY and time.monotonic() < full_deadline:
        starts = [best_p, ref_p]
        starts.extend(extra_refs)
        starts.extend(structured_starts(n, rng, best_p, max_count=28))

        top_count = 4 if n <= 10000 else 3
        top = choose_top_starts(starts, top_count)
        if not top:
            top = [best_p]

        betas = (20000.0, 80000.0, 250000.0)

        if n >= 16000:
            base_iter = 55
        elif n >= 8000:
            base_iter = 75
        else:
            base_iter = 105

        for p_start in top:
            if time.monotonic() >= full_deadline:
                break

            cur_p = normalize_simplex(p_start)
            cur_sc = fast_score(cur_p)
            improved_any = False

            for beta_idx, beta in enumerate(betas):
                if time.monotonic() >= full_deadline:
                    break

                maxit = base_iter
                if beta_idx == 0:
                    maxit = int(maxit * 1.2)
                elif beta_idx == len(betas) - 1:
                    maxit = int(maxit * 0.75)

                # If near the deadline, take a shorter run.
                remaining = max(5.0, float(full_deadline - time.monotonic()))
                maxit = min(maxit, max(12, int(remaining * 0.08)))

                pf, scf = run_full_lbfgs(cur_p, n, beta, maxit)

                if np.isfinite(scf) and scf < cur_sc - 1e-12:
                    cur_p = pf.copy()
                    cur_sc = scf
                    improved_any = True
                elif not improved_any and beta_idx > 0:
                    break

            if np.isfinite(cur_sc) and cur_sc < best_score:
                best_score = cur_sc
                best_p = cur_p.copy()

    # ---- Final exact-max multiplicative polish ----
    if time.monotonic() < deadline - 10.0:
        pm, scm = mirror_polish_exact(best_p, deadline)
        if np.isfinite(scm) and scm < best_score:
            best_score = scm
            best_p = pm.copy()

    return best_p, best_score


# ----------------------------------------------------------------------
# Main search entry point
# ----------------------------------------------------------------------

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best nonnegative sequence under the supplied time budget.
    Always returns a finite list of nonnegative Python floats.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42

    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0

    t0 = time.monotonic()

    ref = load_reference_sequence()
    ref_p_none = None
    ref_score = float("inf")
    ref_list = []

    if ref is not None and ref.size > 0:
        ref_p_none = normalize_simplex(ref)
        ref_score = fast_score(ref_p_none)
        if np.isfinite(ref_score):
            ref_list = [float(x) for x in ref_p_none]

    def uniform_list(n):
        if n <= 0:
            return [1.0]
        return [1.0 / float(n)] * n

    if budget <= 25.0:
        if ref_list:
            return ref_list
        return uniform_list(8192)

    # Leave a conservative margin for finalization; official evaluation is
    # intentionally not invoked here to avoid O(n^2) evaluator cost inside budget.
    if budget > 1200.0:
        reserve = 70.0
    else:
        reserve = 30.0

    total = max(10.0, budget - reserve)
    rng = np.random.default_rng(seed)

    # Lengths to attempt.  Larger lengths usually have more attack surface,
    # but the evaluator's direct convolution cost grows quadratically.
    if budget >= 1500.0:
        if ref is not None and 7000 <= int(ref.size) <= 9000:
            lengths = [8192, 16384]
        else:
            lengths = [16384]
    elif budget >= 700.0:
        lengths = [8192]
    else:
        lengths = [4096]

    best_p = None
    best_score = float("inf")
    best_seq = []

    def consider(p, sc):
        nonlocal best_p, best_score, best_seq
        if p is None or sc is None:
            return
        if not np.isfinite(sc):
            return
        p = normalize_simplex(p)
        sc2 = fast_score(p)
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = sc2
            best_p = p.copy()

    if len(lengths) == 1:
        N = int(lengths[0])
        if ref is not None:
            ref_len = interpolate_sequence(ref, N)
        else:
            ref_len = np.full(N, 1.0 / float(N), dtype=np.float64)

        p, sc = optimize_one_length(N, ref_len, t0 + total, rng)
        consider(p, sc)
    else:
        N1, N2 = int(lengths[0]), int(lengths[1])

        # Stage 1: exact/interpolated reference length.
        if ref is not None:
            ref1 = interpolate_sequence(ref, N1)
        else:
            ref1 = np.full(N1, 1.0 / float(N1), dtype=np.float64)

        t1 = min(450.0, 0.21 * total)
        p1, s1 = optimize_one_length(N1, ref1, t0 + t1, rng)
        consider(p1, s1)

        # Stage 2: larger length, seeded by both the reference and stage-1 best.
        if ref is not None:
            ref2 = interpolate_sequence(ref, N2)
        else:
            ref2 = np.full(N2, 1.0 / float(N2), dtype=np.float64)

        extra2 = None
        if p1 is not None and np.size(p1) > 0:
            extra2 = interpolate_sequence(p1, N2)

        p2, s2 = optimize_one_length(
            N2,
            ref2,
            t0 + total,
            rng,
            extra_p=extra2,
        )
        consider(p2, s2)

    # Prepare final sequence.
    if best_p is not None and np.isfinite(best_score):
        final_p = normalize_simplex(best_p)
        final_p = np.where(np.isfinite(final_p), final_p, 0.0)
        final_p = np.maximum(final_p, 0.0)
        final_p = np.minimum(final_p, 1000.0)
        fs = float(np.sum(final_p))
        if fs > 0.0 and np.isfinite(fs):
            final_p *= 1.0 / fs
        candidate_seq = [float(x) for x in final_p]
        candidate_score = fast_score(final_p)

        # If our search did not beat the previously supplied construction,
        # return the reference rather than a potentially FFT-noise-lesser one.
        if ref_list and np.isfinite(ref_score):
            if not np.isfinite(candidate_score) or ref_score < candidate_score:
                return ref_list
        return candidate_seq

    # Fallbacks.
    if ref_list:
        return ref_list
    return uniform_list(8192)
```
