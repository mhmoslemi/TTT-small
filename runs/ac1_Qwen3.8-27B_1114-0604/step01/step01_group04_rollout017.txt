```python
import cvxpy as cp
import numpy as np
import time

def solve_lp(a, M_target=None):
    """Find g >= 0, sum(g) = 1, minimizing max_k (a*g)[k]."""
    n = len(a)
    g = cp.Variable(n)
    t = cp.Variable()
    
    # conv(a, g) has length 2n-1
    # (a*g)[k] = sum_j a[j] * g[k-j] for valid j
    # We need to express this as a matrix-vector product M @ g
    
    # Build the matrix: row k, col j is a[j] if k-j is in [0, n-1] and j is in [0, n-1]
    # i.e., a[j] * g[k-j] where 0 <= j < n and 0 <= k-j < n
    # So for row k: j ranges from max(0, k-(n-1)) to min(k, n-1)
    
    M_mat = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - (n-1))
        j_max = min(k, n-1)
        for j in range(j_min, j_max+1):
            M_mat[k, j] = a[j]
    
    constraints = [g >= 0, cp.sum(g) == 1]
    if M_target is not None:
        constraints.append(M_mat @ g <= M_target)
    
    objective = cp.Minimize(t)
    constraints.append(M_mat @ g <= t)
    
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS)  # or GLPK, etc.
    
    return g.value, t.value
```

Minimizes 2*n*max(conv(a,a))/sum(a)^2 using LP-based blending and 
smoothed subgradient descent. The LP step finds the optimal blend direction
that minimizes the cross-convolution peak, giving a guaranteed descent.

import time
import scipy.optimize as so

# --- Helper functions ---

def full_conv(a):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def cross_conv(a, b):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
    return np.fft.irfft(fa * fb, fsize)[:size]

def normed_score(a):
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))

def build_conv_matrix(a):
    """M @ g = conv(a, g) where both a and g have length n."""
    n = a.size
    M = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j_lo = max(0, k - (n - 1))
        j_hi = min(k, n - 1)
        M[k, j_lo:j_hi+1] = a[j_lo:j_hi+1]
    return M

def solve_lp(a):
    """LP: min t s.t. conv(a,g) <= t, g>=0, sum(g)=1. Returns (g, t*)."""
    n = a.size
    M_mat = build_conv_matrix(a)
    # Variables: [g (n), t (1)]
    c = np.zeros(n + 1)
    c[n] = 1.0
    # Constraints: M_mat @ g - t <= 0, -g <= 0, sum(g) = 1
    A_ub = np.hstack([M_mat, -np.ones((2*n-1, 1))])
    b_ub = np.zeros(2*n-1)
    A_ub = np.vstack([A_ub, -np.eye(n, n+1)])
    b_ub = np.concatenate([b_ub, np.zeros(n)])
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    res = so.linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=[(0,None)]*n + [(None,None)], method='highs')
    if not res.success:
        return None, None
    g = np.clip(res.x[:n], 0, None)
    g = g / g.sum()
    return g, res.x[n]

def smooth_grad(a, b, tau):
    n = a.size
    s = b / tau
    s -= s.max()
    w = np.exp(s)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros(b.size)
        w[np.argmax(b)] = 1.0
    else:
        w /= wsum
    return 2.0 * np.correlate(w, a, mode='valid')

    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-15:
        return np.ones_like(a) / a.size
    return a / s

    best_sc = normed_score(a)
    t0 = time.monotonic()
    it = 0
    stall = 0
    while time.monotonic() - t0 < budget:
        g, tstar = solve_lp(a)
        if g is None:
            break
        a_new = (1 - alpha) * a + alpha * g
        a_new = project_nonneg_sum1(a_new)
        sc = normed_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best = a_new.copy()
            a = a_new
            stall = 0
        else:
            stall += 1
            if stall > 5:
                alpha = max(0.02, alpha * 0.7)
                if alpha < 0.03:
                    a = project_nonneg_sum1(best + 0.02 * np.random.standard_normal(n))
                    alpha = 0.1
            elif stall > 2:
                alpha = max(0.05, alpha * 0.8)
        a = a_new  # always accept for exploration
        it += 1
        if it % 200 == 0:
            print(f"    [lp] n={n} it={it} best={best_sc:.6f} cur={normed_score(a):.6f}")
    return best, best_sc

def grad_optimize(a, budget, rng):
    """Smoothed subgradient optimization."""
    n = a.size
    a = project_nonneg_sum1(a.astype(np.float64).copy())
    best = a.copy()
    best_sc = normed_score(a)
    eta = 0.3
    tau_frac = 0.05
    beta = 0.5
    vel = np.zeros(n)
    t0 = time.monotonic()
    it = 0
    last_imp = 0
    while time.monotonic() - t0 < budget:
        b = full_conv(a)
        M = float(b.max())
        tau = max(tau_frac * M, 1e-18)
        g = smooth_grad(a, b, tau)
        vel = beta * vel + (1 - beta) * g
        a_new = project_nonneg_sum1(a - eta * vel)
        sc = normed_score(a_new)
        if sc < best_sc:
            best_sc = sc
            best = a_new.copy()
            last_imp = it
        a = a_new
        it += 1
        if it - last_imp > 3000:
            a = project_nonneg_sum1(best + 0.03 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.1 + 0.3 * rng.random()
            tau_frac = 0.03 + 0.05 * rng.random()
            last_imp = it
    return best, best_sc

def init_uniform(n):
    return np.ones(n) / n

def init_u_shape(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = 1.0 / (0.1 + np.abs(i - c) / c)
    return a / a.sum()

def init_edges(n):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-4.0 * (np.minimum(i, n-1-i) / n) ** 2)
    return a / a.sum()

def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()
