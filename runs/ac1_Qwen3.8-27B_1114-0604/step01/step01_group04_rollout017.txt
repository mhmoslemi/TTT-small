```python
from scipy.optimize import linprog, minimize

def full_conv(a):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def normed_score(a):
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))

def project_simplex(v):
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lp_direction(a, M):
    """Find b that maximizes sum(b) subject to b>=0, (a*b)_j <= M for all j."""
    n = a.size
    size = 2 * n - 1
    
    # (a*b)_j = sum_i a_i * b_{j-i} (with appropriate bounds)
    # This is a linear function of b for fixed a
    # (a*b)_j = sum_{i: 0<=i<n, 0<=j-i<n} a_i * b_{j-i}
    # Let k = j-i, so i = j-k
    # (a*b)_j = sum_{k: max(0,j-n+1)<=k<=min(j,n-1)} a_{j-k} * b_k
    
    # Build constraint matrix A where A[j, k] = a[j-k] if 0<=j-k<n and 0<=k<n
    A = np.zeros((size, n))
    for j in range(size):
        for k in range(n):
            i = j - k
            if 0 <= i < n:
                A[j, k] = a[i]
    
    # Maximize sum(b) = c^T b where c = ones
    # Subject to A b <= M (vector of M's, length size)
    # b >= 0
    c = -np.ones(n)  # minimize -sum(b)
    b_ub = M * np.ones(size)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        return res.x
    else:
        return np.ones(n) / n
```

```python
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:

import numpy as np
import time
from scipy.optimize import linprog

def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))

def project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

    n = a.size

    best_sc = normed_score(a)
    best_a = a.copy()
    
    golden = (np.sqrt(5.0) - 1.0) / 2.0
    
    for sweep in range(n_sweeps):
```

This is a non-convex QP (because of a*a). But I can use a penalty method:
minimize M + λ * sum_j max(0, (a*a)_j - M)^2
subject to a >= 0, sum(a) = 1

Or use the softmax trick:
minimize (1/τ) * log(sum_j exp(τ*(a*a)_j))
subject to a >= 0, sum(a) = 1

This is convex in a (for fixed τ)! Wait, is it? (a*a)_j is a quadratic form in a: a^T B_j a where B_j is a fixed matrix. The log-sum-exp of quadratics is not necessarily convex.

Actually, (a*a)_j = sum_i a_i * a_{j-i}. This is a quadratic form: a^T Q_j a where (Q_j)_{i,k} = 1 if i+k=j (with appropriate bounds). The matrix Q_j is symmetric and has entries 0 or 1. It's not positive semidefinite in general (it's a permutation-like matrix).

So the log-sum-exp of these quadratics is not convex. We're back to non-convex optimization.

Looking at this problem, I need to minimize `2*n*max(conv(a,a))/sum(a)^2` for a non-negative sequence. The key insight is that this is scale-invariant, so I can normalize `sum(a)=1` and minimize `2*n*max(conv(a,a))`.

My approach combines:
1. **Pair-wise coordinate descent** - exact 1D subproblem (convex) solved via golden section search
2. **LP-based direction** (from the state of the art) as macro-steps
3. **Smoothed subgradient** for polishing
4. Multiple n values and diverse initializations

The coordinate descent is efficient: for each pair (k,l), fixing all other elements, the max convolution as a function of `a[k]` (with `a[l] = s - a[k]`) is a convex piecewise-linear/quadratic function, solvable by 1D search.

```python
"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Pair-wise coordinate descent (exact convex 1D subproblem per pair)
2. LP-based macro direction (from state-of-the-art formulation)
3. Smoothed subgradient polishing
4. Multiple sequence lengths, diverse initializations, simulated annealing restarts
"""

import numpy as np
import time
from scipy.optimize import linprog


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def project_simplex(v):
    """Project vector v onto {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lp_direction(a, M, max_time=0.5):
    """
    Solve LP: maximize sum(b) s.t. b>=0, (a*b)_j <= M for all j.
    Returns the optimal b (not normalized).
    """
    n = a.size
    size = 2 * n - 1

    # Build constraint matrix: (a*b)_j = sum_i a_i * b_{j-i}
    # A[j, k] = a[j-k] if 0<=j-k<n, else 0
    A = np.zeros((size, n))
    for j in range(size):
        lo_k = max(0, j - n + 1)
        hi_k = min(j, n - 1)
        if lo_k <= hi_k:
            A[j, lo_k:hi_k+1] = a[j - np.arange(lo_k, hi_k+1)]

    c = -np.ones(n)
    b_ub = M * np.ones(size)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': max_time, 'sparse': True})
        if res.success:
            return res.x
    except Exception:
        pass
    return np.ones(n) / n


def pair_cd_optimize(a, time_budget, rng, use_annealing=True, lp_every=500):
    """
    Pair-wise coordinate descent with optional simulated annealing and LP macro-steps.
    a should be normalized to sum=1.
    Returns (best_a, best_score).
    """
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    golden = (np.sqrt(5.0) - 1.0) / 2.0
    t0 = time.monotonic()
    last_improve = time.monotonic()
    temp = 0.01
    pair_count = 0

    a_work = a.copy()

    while time.monotonic() - t0 < time_budget:
        k = int(rng.integers(n))
        l = int(rng.integers(n))
        if k == l:
            continue

        s = a_work[k] + a_work[l]
        if s < 1e-14:
            continue

        # Golden section search: minimize max(full_conv(a_t)) over x in [0, s]
        # where a_t = a_work with a_t[k]=x, a_t[l]=s-x
        lo = 0.0
        hi = s
        d = golden * (hi - lo)
        x1 = lo + d
        x2 = hi - d

        # Eval f1
        a_work[k] = x1
        a_work[l] = s - x1
        f1 = float(np.max(full_conv(a_work)))

        # Eval f2
        a_work[k] = x2
        a_work[l] = s - x2
        f2 = float(np.max(full_conv(a_work)))

        for _ in range(30):
            if f1 < f2:
                hi = x2
                x2 = x1
                f2 = f1
                x1 = lo + golden * (hi - lo)
                a_work[k] = x1
                a_work[l] = s - x1
                f1 = float(np.max(full_conv(a_work)))
            else:
                lo = x1
                x1 = x2
                f1 = f2
                x2 = hi - golden * (hi - lo)
                a_work[k] = x2
                a_work[l] = s - x2
                f2 = float(np.max(full_conv(a_work)))

        x_opt = (lo + hi) / 2.0
        a_work[k] = x_opt
        a_work[l] = s - x_opt

        cur_sc = 2.0 * n * f2 if f2 < f1 else 2.0 * n * f1
        # Recompute properly
        cur_max = float(np.max(full_conv(a_work)))
        cur_sc = 2.0 * n * cur_max

        # Annealing: accept worse solution with probability exp(-delta/temp)
        accept = True
        if use_annealing and cur_sc > best_sc:
            delta = cur_sc - best_sc
            ifrng = rng.random()
            if ifrng > np.exp(-delta / max(temp, 1e-10)):
                accept = False

        if accept:
            if cur_sc < best_sc:
                best_sc = cur_sc
                best_a = a_work.copy()
                last_improve = time.monotonic()

            # LP macro-step every lp_every pairs
            pair_count += 1
            if pair_count % lp_every == 0:
                M_cur = float(np.max(full_conv(a_work)))
                b_lp = lp_direction(a_work, M_cur, max_time=0.3)
                if b_lp is not None and b_lp.sum() > 0:
                    b_norm = b_lp / b_lp.sum()
                    t_step = 0.3
                    a_new = project_simplex((1 - t_step) * a_work + t_step * b_norm)
                    sc_new = normed_score(a_new)
                    if sc_new <= cur_sc or rng.random() < 0.1:
                        a_work = a_new
                        if sc_new < best_sc:
                            best_sc = sc_new
                            best_a = a_work.copy()
                            last_improve = time.monotonic()
        else:
            # Reject: restore
            a_work[k] = x_opt  # keep the optimal for this pair (it's the local best)
            # Actually we should revert to before this pair update
            # But we lost the old values. Let's just keep current (it's still a valid point)
            pass

        # Adaptive temperature
        temp *= 0.9999
        if temp < 1e-6:
            temp = 1e-6

        # Restart if stuck
        if time.monotonic() - last_improve > 30.0:
            a_work = project_simplex(best_a + 0.03 * rng.standard_normal(n))
            temp = 0.02
            last_improve = time.monotonic()

    return best_a, best_sc


def subgrad_polish(a, time_budget, rng):
    """Smoothed subgradient descent for local polishing."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    eta = 0.1
    tau = 0.03
    beta = 0.5
    vel = np.zeros(n)
    t0 = time.monotonic()
    last_improve = 0
    iters = 0

    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_max = float(np.max(b))
        cur_sc = 2.0 * n * cur_max

        tau_val = tau * cur_max
        if tau_val < 1e-18:
            tau_val = 1e-18

        # Smoothed subgradient
        shift = b / tau_val
        shift -= shift.max()
        w = np.exp(shift)
        wsum = w.sum()
        if wsum < 1e-300:
            w = np.zeros(b.size)
            w[int(np.argmax(b))] = 1.0
        else:
            w /= wsum

        g = 2.0 * np.correlate(w, a, mode='valid')
        vel = beta * vel + (1.0 - beta) * g
        a_new = project_simplex(a - eta * vel)

        b_new = full_conv(a_new)
        new_sc = 2.0 * n * float(np.max(b_new))

        if new_sc <= cur_sc or rng.random() < 0.05:
            a = a_new
        else:
            a = project_simplex(a - 0.5 * eta * vel)

        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = iters

        iters += 1
        if iters - last_improve > 2000:
            a = project_simplex(best_a + 0.02 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.05 + 0.1 * rng.random()
            last_improve = iters

    return best_a, best_sc


def init_raised_cosine(n):
    """Raised cosine: good smooth non-negative shape."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + np.cos(2.0 * np.pi * (i + 0.5) / n)
    a += 1e-8
    return a / a.sum()


def init_sine_hump(n):
    """Single sine hump."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a += 1e-8
    return a / a.sum()


def init_gaussian(n):
    """Gaussian centered."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    sigma = n / 4.0
    a = np.exp(-((i - c) / sigma) ** 2)
    a += 1e-8
    return a / a.sum()


def init_u_shape(n):
    """High at edges."""
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i).astype(np.float64) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_two_cluster(n, rng):
    """Two Gaussian clusters."""
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.2 + 0.1 * rng.random())
    c2 = n * (0.7 + 0.1 * rng.random())
    sigma = n * (0.05 + 0.05 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_flat_top(n, rng):
    """Taylor-like flat-top window."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    x = (i - c) / c
    # Taylor window approximation
    a = 1.0 - 0.3*(x**2) + 0.09*(x**4)
    a = np.maximum(a, 0.01)
    return a / a.sum()


def init_noisy_smooth(n, rng):
    """Smooth random initialization."""
    a = np.ones(n) + 0.5 * rng.standard_normal(n)
    # Smooth it
    kernel = np.array([1, 2, 3, 2, 1], dtype=np.float64)
    kernel /= kernel.sum()
    a = np.convolve(a, kernel, mode='same')
    a = np.abs(a) + 0.01
    return a / a.sum()


def get_inits(n, rng):
    """Return diverse initializations."""
    inits = []
    inits.append(init_raised_cosine(n))
    inits.append(init_sine_hump(n))
    inits.append(init_gaussian(n))
    inits.append(init_u_shape(n))
    inits.append(init_two_cluster(n, rng))
    inits.append(init_flat_top(n, rng))
    inits.append(init_noisy_smooth(n, rng))
    # Random
    a = np.abs(rng.standard_normal(n)) + 0.1
    inits.append(a / a.sum())
    return inits
```
