"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using multiplicative mirror descent with adaptive beta, momentum, and
peak-targeted local search. Uses FFT for efficiency and multiple scales.
"""

import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b, L=None):
    la = len(a)
    lb = len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if L is None:
        L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)


def _grad_convolution(w, p, L=None):
    """
    Compute h[k] = sum_j w[j] * d(conv_j)/d(f[k]) for conv = p*p.
    This equals 2 * cross_correlation(w, p) (with boundary handling).
    Result has length len(p).
    """
    lp = len(p)
    if lp == 0:
        return np.array([], dtype=float)
    if L is None:
        L = _next_pow2(max(len(w), 2 * lp))
    fw = np.fft.rfft(w, L)
    fp = np.fft.rfft(p, L)
    # cross-correlation: ifft(F(w) * conj(F(p)))
    out = np.fft.irfft(fw * np.conjugate(fp), L)
    return out[:lp].astype(float, copy=False)

    return p / s

    n = d.size

    if beta < 1e-6:
        return np.full(n, 1.0 / n)
    shift = float(np.max(d))
    e = np.exp(beta * (d - shift))
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return e / s

    n = p.size

    c = _conv_full(p, p, L)
    c = np.where(np.isfinite(c), c, 0.0)
    d = float(n) * c
    exact = float(np.max(d))
    w = _softmax_weights(d, beta)
    smooth = float(np.dot(w, d))
    h = _grad_convolution(w, p, L)
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.maximum(h, 0.0)
    return exact, smooth, h, L

            p = np.ones(n)
            p[idx:] = ratio
            p = _sanitize(p)
            if p is None:
                continue
            sc = _exact_score(p)
            if sc < best_score:
                best_score = sc
                best = p.copy()
    return best


def _resample(arr, m):
    a = _sanitize(arr)
    if a is None or m <= 0:
        return None
    n = a.size
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m) / m
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0) + 1e-12
    return _sanitize(out)


def _run_mmd(p0, rng, time_limit, beta_start=30.0, beta_end=3000.0, verbose=False):
    p = _sanitize(p0)
    if p is None:
        return None, float("inf")
    if time_limit <= 0:
        return p, _exact_score(p)

    n = int(p.size)
    L = _next_pow2(2 * n - 1)
    floor = 1e-18 / float(n)
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_exact = _exact_score(p)
    cur_exact, cur_smooth, h, _ = _compute_grad(p, beta_start, L)

    beta = float(max(1.0, beta_start))
    beta_max = float(max(beta, beta_end))
    alpha = 0.1
    momentum = np.zeros(n)
    mu = 0.8  # momentum coefficient
    deadline = time.monotonic() + float(time_limit)
    stall = 0
    it = 0
    last_print = -10000

    while time.monotonic() < deadline:
        cur_exact, cur_smooth, h, _ = _compute_grad(p, beta, L)

        if not np.isfinite(np.max(h)) or np.max(h) < 1e-15:
            p_try = _sanitize(p * np.exp(rng.normal(0.0, 0.05, n)))
            if p_try is not None:
                p_try = np.maximum(p_try, floor)
                p_try = p_try / p_try.sum()
                e, s2, _, _ = _compute_grad(p_try, beta, L)
                if np.isfinite(e) and e <= best_exact * 1.05:
                    p = p_try
                    cur_exact, cur_smooth = e, s2
            stall += 1
        else:
            h = h / float(np.max(h))
            # Apply momentum
            momentum = mu * momentum + (1 - mu) * h

            tests = []
            for mult in (0.3, 0.5, 1.0, 1.5, 2.0, 3.0, 5.0):
                a_test = alpha * mult
                if 1e-5 <= a_test <= 30.0:
                    tests.append(a_test)
            if not tests:
                tests = [1e-4, 0.05]

            chosen = None
            ch_smooth = cur_smooth
            ch_exact = cur_exact
            ch_a = alpha

            for a_test in tests:
                dec = np.exp(-a_test * momentum)
                p_try = p * dec
                p_try = np.maximum(p_try, floor)
                s_sum = float(p_try.sum())
                if s_sum <= 0 or not np.isfinite(s_sum):
                    continue
                p_try = p_try / s_sum
                if not np.all(np.isfinite(p_try)):
                    continue
                e, s2, _, _ = _compute_grad(p_try, beta, L)
                if not np.isfinite(e) or not np.isfinite(s2):
                    continue
                if s2 + 1e-12 * max(1.0, ch_smooth) < ch_smooth:
                    chosen = p_try
                    ch_smooth = s2
                    ch_exact = e
                    ch_a = a_test

            if chosen is not None:
                p = chosen
                cur_smooth = ch_smooth
                cur_exact = ch_exact
                alpha = min(30.0, ch_a * 1.2)
                stall = 0

                if cur_exact < best_exact:
                    best = p.copy()
                    best_exact = cur_exact
                    if verbose and (it - last_print >= 500):
                        print(f"  mmd n={n} iter={it} best={best_exact:.6f}")

                if cur_exact > best_exact * 1.3 and rng.random() < 0.2:
                    p = best.copy()
                    cur_exact, cur_smooth, h, _ = _compute_grad(p, beta, L)
            else:
                stall += 1
                alpha = max(1e-5, alpha * 0.7)

                if stall >= 10:
                    std = 0.05 if stall < 30 else min(0.5, 0.05 + 0.008 * stall)
                    p_try = _sanitize(p * np.exp(rng.normal(0.0, std, n)))
                    if p_try is not None:
                        p_try = np.maximum(p_try, floor)
                        p_try = p_try / p_try.sum()
                        e, s2, _, _ = _compute_grad(p_try, beta, L)
                        if np.isfinite(e) and e <= best_exact * 1.08:
                            p = p_try
                            cur_exact, cur_smooth = e, s2
                        else:
                            p = best.copy()
                            cur_exact, cur_smooth, h, _ = _compute_grad(p, beta, L)
                    momentum = np.zeros(n)
                    stall = 0

                if stall > 60:
                    p = best.copy()
                    cur_exact, cur_smooth, h, _ = _compute_grad(p, beta, L)
                    alpha = 0.1
                    beta = max(beta, float(beta_start))
                    momentum = np.zeros(n)
                    stall = 0

        it += 1
        if it % 15 == 0:
            beta = min(beta_max, beta * 1.18)

    return best, best_exact


def _peak_chase(p0, rng, time_limit, L=None):
    """
    Targeted optimization: repeatedly identify the top autocorrelation bins
    and reduce the contributions from coordinates that feed into them.
    """
    p = _sanitize(p0)
    if p is None or time_limit <= 0:
        return None, float("inf")
    n = int(p.size)
    if L is None:
        L = _next_pow2(2 * n - 1)
    floor = 1e-18 / float(n)
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_score = _exact_score(p)
    deadline = time.monotonic() + float(time_limit)
    stagnant = 0

    while time.monotonic() < deadline:
        # Compute convolution and find top bins
        c = _conv_full(p, p, L)
        c = np.where(np.isfinite(c), c, 0.0)
        top_k = min(10, c.size)
        top_idxs = np.argsort(c)[-top_k:]
        top_val = c[top_idxs[0]]
        if top_val < 1e-15:
            break

        # For each top bin, find which coordinates contribute most
        # and create a "pressure" vector
        pressure = np.zeros(n)
        for j in top_idxs:
            j = int(j)
            # conv[j] = sum_i p[i]*p[j-i]
            # dconv[j]/dp[k] = p[j-k] (if valid) + p[k-j] (if valid)
            # Simplified: just use p itself shifted
            lo = max(0, j - n + 1)
            hi = min(n - 1, j)
            for i in range(lo, hi + 1):
                idx1 = i
                idx2 = j - i
                # Both idx1 and idx2 contribute
                pressure[idx1] += p[idx2] * 0.5
                pressure[idx2] += p[idx1] * 0.5

        pressure = np.maximum(pressure, 0)
        max_p = float(np.max(pressure))
        if max_p < 1e-15:
            break
        pressure = pressure / max_p

        # Try reducing high-pressure, high-value coordinates
        best_local = p.copy()
        best_local_score = best_score
        factors = (0.3, 0.5, 0.7, 0.85)

        # Pick top 20% of coordinates by pressure * value
        pv = pressure * p
        thresh = np.percentile(pv, 70)
        candidates = np.where(pv > thresh)[0]
        if len(candidates) == 0:
            candidates = np.where(pressure > 0.1)[0]

        # Try random subsets
        for _ in range(min(5, len(candidates))):
            subset = rng.choice(candidates, size=max(1, len(candidates) // 3), replace=False)
            cand = best_local.copy()
            for f in factors:
                cand2 = best_local.copy()
                cand2[subset] *= f
                cand2 = np.maximum(cand2, floor)
                s_sum = float(cand2.sum())
                if s_sum <= 0:
                    continue
                cand2 = cand2 / s_sum
                sc = _exact_score(cand2)
                if sc < best_local_score - 1e-12:
                    best_local_score = sc
                    best_local = cand2

        if best_local_score < best_score - 1e-12:
            p = best_local
            best = p.copy()
            best_score = best_local_score
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 30:
                # Perturb and retry
                p_try = _sanitize(p * np.exp(rng.normal(0.0, 0.03, n)))
                if p_try is not None:
                    p_try = np.maximum(p_try, floor)
                    p_try = p_try / p_try.sum()
                    sc = _exact_score(p_try)
                    if sc < best_score * 1.05:
                        p = p_try
                    else:
                        p = best.copy()
                stagnant = 0

    return best, best_score


def _local_block_search(p0, rng, time_limit):
    """Coordinate block scaling local search."""
    p = _sanitize(p0)
    if p is None or time_limit <= 0:
        return None, float("inf")
    n = int(p.size)
    if n <= 1:
        return p, _exact_score(p)

    L = _next_pow2(2 * n - 1)
    floor = 1e-18 / float(n)
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_score = _exact_score(p)
    deadline = time.monotonic() + float(time_limit)
    stagnant = 0

    while time.monotonic() < deadline:
        r = float(rng.random())
        if r < 0.25:
            w = int(rng.integers(1, max(2, n // 100 + 1)))
        elif r < 0.65:
            w = int(rng.integers(max(2, n // 100), max(3, n // 20 + 1)))
        else:
            low = max(3, n // 20)
            high = max(low + 1, min(n, n // 4 + 1))
            w = int(rng.integers(low, high + 1))
        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))

        base = p.copy()
        current_best = base
        current_score = best_score

        for f in (0.3, 0.5, 0.7, 0.85, 1.15, 1.4, 1.8, 2.5):
            cand = base.copy()
            cand[start:start + w] *= f
            cand = np.maximum(cand, floor)
            s_sum = float(cand.sum())
            if s_sum <= 0:
                continue
            cand = cand / s_sum
            sc = _exact_score(cand)
            if sc < current_score - 1e-12:
                current_score = sc
                current_best = cand

        if current_best is not base:
            p = current_best
            if current_score < best_score:
                best = p.copy()
                best_score = current_score
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 50:
                p_try = _sanitize(p * np.exp(rng.normal(0.0, 0.05, n)))
                if p_try is not None:
                    p_try = np.maximum(p_try, floor)
                    p_try = p_try / p_try.sum()
                    sc = _exact_score(p_try)
                    if sc < best_score * 1.05:
                        p = p_try
                    else:
                        p = best.copy()
                stagnant = 0

    return best, best_score


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search function. Returns a list of non-negative floats minimizing
    2*n*max(conv(a,a))/sum(a)^2.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 300.0

    start_time = time.monotonic()
    deadline = start_time + budget - 30.0

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng(ised = (iseed % 1000000007) + 7)

    # Try to load parent
    parent_arr = None
    try:
        parent_raw = height_sequence_1
        parent_arr = _sanitize(np.asarray(parent_raw, dtype=float))
    except NameError:
        pass
    except Exception:
        pass

    best_arr = None
    best_score = float("inf")

    def _update_best(arr, score):
        nonlocal best_arr, best_score
        if arr is not None and np.isfinite(score) and score < best_score:
            best_arr = _sanitize(arr)
            best_score = float(score)
            print(f"  NEW BEST: {best_score:.6f} (n={best_arr.size}, t={time.monotonic()-start_time:.1f}s)")

    # === Phase 1: Quick exploration at moderate size (200s) ===
    small_n = 2048
    print(f"Phase 1: Exploration at n={small_n}")
    phase1_deadline = min(deadline, start_time + 200.0)

    starts = []

    if parent_arr is not None:
        sp = _resample(parent_arr, small_n)
        if sp is not None:
            starts.append(sp)

    tb = _two_block_best(small_n)
    if tb is not None:
        starts.append(tb)

    for mode in (1, 2, 4, 5, 0, 6):
        p = _initial_sequence(small_n, rng, mode)
        if p is not None:
            starts.append(p)

    # Score all starts
    scored = []
    for s in starts:
        s = _sanitize(s)
        if s is None:
            continue
        sc = _exact_score(s)
        scored.append((sc, s))
    scored.sort(key=lambda x: x[0])

    # Run MMD on top 4 starts
    for sc, s in scored[:4]:
        rem = phase1_deadline - time.monotonic()
        if rem < 20:
            break
        talloc = min(45.0, max(5.0, rem - 5.0))
        p, psc = _run_mmd(s, rng, talloc, beta_start=20.0, beta_end=500.0)
        _update_best(p, psc)

    # === Phase 2: Scale up to main length (80s) ===
    print(f"Phase 2: Scaling to main length")
    main_n = 4096
    phase2_deadline = min(deadline, start_time + 280.0)

    # Build main starts
    main_starts = []

    if best_arr is not None:
        ms = _resample(best_arr, main_n)
        if ms is not None:
            main_starts.append(ms)

    if parent_arr is not None:
        ms = _resample(parent_arr, main_n)
        if ms is not None:
            main_starts.append(ms)

    tb_main = _two_block_best(main_n)
    if tb_main is not None:
        main_starts.append(tb_main)

    for mode in (1, 2, 4, 5):
        ms = _initial_sequence(main_n, rng, mode)
        if ms is not None:
            main_starts.append(ms)

    # Quick warm-up
    for ms in main_starts[:3]:
        rem = phase2_deadline - time.monotonic()
        if rem < 10:
            break
        talloc = min(20.0, max(3.0, rem - 3.0))
        p, psc = _run_mmd(ms, rng, talloc, beta_start=20.0, beta_end=200.0)
        _update_best(p, psc)

    # === Phase 3: Main intensive optimization (1500s) ===
    print(f"Phase 3: Main optimization at n={main_n}")
    phase3_deadline = min(deadline, start_time + 1900.0)

    # Use the best current as starting point
    if best_arr is not None:
        start_p = _resample(best_arr, main_n)
        if start_p is None:
            start_p = main_starts[0] if main_starts else _initial_sequence(main_n, rng, 1)
    else:
        start_p = main_starts[0] if main_starts else _initial_sequence(main_n, rng, 1)

    # Run multiple MMD passes with increasing difficulty
    passes = [
        (30.0, 500.0, 0.4),
        (50.0, 2000.0, 0.35),
        (100.0, 8000.0, 0.25),
    ]

    rem_total = phase3_deadline - time.monotonic()
    for beta_s, beta_e, frac in passes:
        rem = phase3_deadline - time.monotonic()
        if rem < 60:
            break
        talloc = rem * frac
        if best_arr is not None:
            p0 = _resample(best_arr, main_n)
        else:
            p0 = start_p
        if p0 is None:
            p0 = start_p

        p, psc = _run_mmd(p0, rng, max(30.0, talloc),
                          beta_start=beta_s, beta_end=beta_e, verbose=True)
        _update_best(p, psc)

        # Interleave with peak chasing
        rem = phase3_deadline - time.monotonic()
        if rem > 120 and best_arr is not None:
            p_peak = _resample(best_arr, main_n)
            if p_peak is not None:
                p2, psc2 = _peak_chase(p_peak, rng, min(100.0, rem * 0.3),
                                       L=_next_pow2(2 * main_n - 1))
                _update_best(p2, psc2)

    # === Phase 4: Polish (400s) ===
    print(f"Phase 4: Polishing")
    phase4_deadline = min(deadline, start_time + 2300.0)

    if best_arr is not None:
        p0 = _resample(best_arr, main_n)
    else:
        p0 = _initial_sequence(main_n, rng, 1)

    if p0 is not None:
        rem = phase4_deadline - time.monotonic()
        if rem > 100:
            p, psc = _run_mmd(p0, rng, min(200.0, rem * 0.5),
                              beta_start=200.0, beta_end=15000.0, verbose=True)
            _update_best(p, psc)

        rem = phase4_deadline - time.monotonic()
        if rem > 60 and best_arr is not None:
            p0 = _resample(best_arr, main_n)
            p, psc = _local_block_search(p0, rng, min(150.0, rem * 0.6))
            _update_best(p, psc)

        rem = phase4_deadline - time.monotonic()
        if rem > 60 and best_arr is not None:
            p0 = _resample(best_arr, main_n)
            p, psc = _run_mmd(p0, rng, min(100.0, rem * 0.7),
                              beta_start=500.0, beta_end=20000.0, verbose=True)
            _update_best(p, psc)

        rem = phase4_deadline - time.monotonic()
        if rem > 40 and best_arr is not None:
            p0 = _resample(best_arr, main_n)
            p, psc = _peak_chase(p0, rng, min(80.0, rem * 0.5),
                                 L=_next_pow2(2 * main_n - 1))
            _update_best(p, psc)

    # === Phase 5: Try larger n if time permits (100s) ===
    print(f"Phase 5: Trying larger n")
    phase5_deadline = deadline - 5.0

    rem = phase5_deadline - time.monotonic()
    if rem > 200 and best_arr is not None:
        big_n = 8192
        big_p = _resample(best_arr, big_n)
        if big_p is not None:
            talloc = min(150.0, rem * 0.6)
            p, psc = _run_mmd(big_p, rng, talloc,
                              beta_start=50.0, beta_end=5000.0, verbose=True)
            _update_best(p, psc)

            rem = phase5_deadline - time.monotonic()
            if rem > 30 and best_arr is not None:
                big_p = _resample(best_arr, big_n)
                if big_p is not None:
                    p, psc = _local_block_search(big_p, rng, min(50.0, rem * 0.5))
                    _update_best(p, psc)

    # === Final: Verify with evaluate_sequence ===
    print(f"Final: best_score={best_score:.6f}, t={time.monotonic()-start_time:.1f}s")

    if best_arr is None:
        best_arr = _initial_sequence(main_n, rng, 1)
        best_score = _exact_score(best_arr)

    # Build candidate list for final check
    final_candidates = [best_arr]

    # Try to evaluate a few variants
    for arr in final_candidates[:1]:
        a = _sanitize(arr)
        if a is None:
            continue
        lst = [float(x) for x in a]
        try:
            sc = float(evaluate_sequence(lst))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_arr = a
        except Exception:
            pass

    # Ensure output is valid
    out = np.asarray(best_arr, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s <= 0 or not np.isfinite(s):
        out = np.ones(main_n)
    else:
        out = out / s

    print(f"  Final sequence: n={out.size}, score={best_score:.6f}")
    return [float(x) for x in out]
