"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 via adaptive SMC descent,
multi-length search, LP refinement, and block coordinate polishing.
"""

import numpy as np
import time
import scipy.fft
import scipy.sparse

def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _full_conv(a, b):
    nout = len(a) + len(b) - 1
    L = _next_pow2(nout)
    fa = scipy.fft.rfft(a, L)
    fb = scipy.fft.rfft(b, L)
    return scipy.fft.irfft(fa * fb, L)[:nout]

def _valid_corr(c, a):
    """result[j] = sum_i a[i]*c[j+i] = (c cross-correlate a)[j]"""
    L = _next_pow2(max(len(c), len(a)))
    fc = scipy.fft.rfft(c, L)
    fa = scipy.fft.rfft(a, L)
    out = scipy.fft.irfft(fc * np.conjugate(fa), L)
    return out[:len(a)]

def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = p.sum()
    if s <= 0 or not np.isfinite(s):
        return None
    return p / s

def _score(p):
    p = _sanitize(p)
    if p is None:
        return float('inf')
    n = len(p)
    c = _full_conv(p, p)
    c = np.where(np.isfinite(c), c, 0.0)
    return 2.0 * n * c.max()

def _smc_step(p, beta):
    """One SMC step: return (new_p, smooth_val, grad)"""
    n = len(p)
    c = _full_conv(p, p)
    c = np.where(np.isfinite(c), c, 0.0)
    d = n * c  # This makes the objective 2*max(d) = 2n*max(c)
    # Softmax weights
    shift = d.max()
    e = np.exp(beta * (d - shift))
    w = e / e.sum()
    smooth = float(np.dot(w, d))
    # Gradient
    grad = 2.0 * _valid_corr(w, p)
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return smooth, w, grad

def _run_smc(p0, budget, beta0=20.0, beta_growth=1.1, beta_max=5000.0,
             alpha0=0.5, rng=None, verbose=False):
    """Run SMC for given budget (seconds)."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    if budget <= 0:
        return p, _score(p)
    
    if rng is None:
        rng = np.random.default_rng()
    
    n = len(p)
    p = np.maximum(p, 1e-18 / n)
    p = p / p.sum()
    
    best = p.copy()
    best_score = _score(p)
    
    beta = beta0
    alpha = alpha0
    stall = 0
    deadline = time.monotonic() + budget
    it = 0
    
    while time.monotonic() < deadline:
        # Compute smooth value and gradient
        c = _full_conv(p, p)
        c = np.where(np.isfinite(c), c, 0.0)
        d = n * c
        shift = d.max()
        e = np.exp(beta * (d - shift))
        w = e / e.sum()
        smooth = float(np.dot(w, d))
        grad = 2.0 * _valid_corr(w, p)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        grad = np.maximum(grad, 0.0)
        
        if grad.max() < 1e-15:
            # Degenerate: perturb
            p = _sanitize(p * np.exp(rng.normal(0, 0.1, n)))
            p = np.maximum(p, 1e-18/n)
            p = p / p.sum()
            stall += 1
        else:
            # Normalize gradient
            g = grad / grad.max()
            
            # Try different step sizes (backtracking)
            chosen = None
            best_smooth = smooth
            for mult in [0.25, 0.5, 1.0, 2.0, 4.0, 8.0]:
                a = alpha * mult
                p_try = p * np.exp(-a * g)
                p_try = np.maximum(p_try, 1e-18/n)
                p_try = p_try / p_try.sum()
                
                # Evaluate smooth objective
                c_try = _full_conv(p_try, p_try)
                c_try = np.where(np.isfinite(c_try), c_try, 0.0)
                d_try = n * c_try
                shift_try = d_try.max()
                e_try = np.exp(beta * (d_try - shift_try))
                w_try = e_try / e_try.sum()
                smooth_try = float(np.dot(w_try, d_try))
                
                if smooth_try < best_smooth - 1e-14:
                    chosen = p_try
                    best_smooth = smooth_try
            
            if chosen is not None:

                p = chosen

                    best_score = exact
                stall = 0
            else:
                stall += 1
                alpha *= 0.7

                    p = _sanitize(p * np.exp(rng.normal(0, 0.15, n)))
                    if p is None:
                        p = best.copy()
                    p = np.maximum(p, 1e-18/n)
                    p = p / p.sum()
                    alpha = alpha0
                    beta = max(beta0, beta * 0.5)
                    stall = 0
        
        it += 1
        if it % 30 == 0:
            beta = min(beta_max, beta * beta_growth)
        
        if verbose and it % 500 == 0:
            print(f"  SMC: n={n}, it={it}, best={best_score:.6f}, beta={beta:.1f}")
    
    return best, best_score

def _initial_seq(n, rng, mode=0):
    """Generate initial sequence."""
    t = (np.arange(n) + 0.5) / n
    if mode == 0:
        # Gaussian bump
        c = rng.uniform(0.3, 0.7)
        s = rng.uniform(0.15, 0.35)
        p = np.exp(-0.5 * ((t - c) / s) ** 2)
    elif mode == 1:
        # Beta distribution
        a = rng.uniform(0.5, 5.0)
        b = rng.uniform(0.5, 5.0)
        p = t**a * (1-t)**b
    elif mode == 2:
        # Two bumps
        c1, c2 = sorted(rng.uniform(0, 1, 2))
        s1, s2 = rng.uniform(0.1, 0.3, 2)
        p = np.exp(-0.5*((t-c1)/s1)**2) + 0.7*np.exp(-0.5*((t-c2)/s2)**2)
    elif mode == 3:
        # Block function
        k = int(rng.integers(3, 8))
        edges = np.sort(rng.uniform(0.1, 0.9, k-1))
        blocks = np.concatenate([[0], edges, [1]])
        heights = rng.lognormal(0, 1, k)
        p = np.zeros(n)
        for i in range(k):
            s = int(blocks[i]*n)
            e = max(s+1, int(blocks[i+1]*n))
            p[s:np.min(e,n)] = heights[i]
    else:
        # Cosine series
        z = np.ones(n)
        for k in range(1, 10):
            amp = rng.normal(0, 1/(k+1))
            ph = rng.uniform(0, 2*np.pi)
            z += amp * np.cos(2*np.pi*k*t + ph)
        p = np.log1p(np.exp(np.clip(z, -30, 30)))
    
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0) + 1e-10
    return p / p.sum()

def _block_search(p0, budget, rng):
    """Coordinate block scaling search."""
    p = _sanitize(p0)
    if p is None or budget <= 0:
        return None, float('inf')
    n = len(p)
    p = np.maximum(p, 1e-18/n)
    p = p / p.sum()
    best = p.copy()
    best_score = _score(p)
    deadline = time.monotonic() + budget
    stagnant = 0
    factors = [0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 3.0]
    
    while time.monotonic() < deadline:
        # Choose block size
        r = rng.random()
        if r < 0.3:
            w = int(rng.integers(1, max(2, n//100)))
        elif r < 0.7:
            w = int(rng.integers(max(2, n//100), max(3, n//20)))
        else:
            w = int(rng.integers(max(3, n//20), min(n, n//5)))
        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))
        
        cur_best = p.copy()
        cur_score = best_score
        
        for f in factors:
            cand = p.copy()
            cand[start:start+w] *= f
            cand = _sanitize(cand)
            if cand is None:
                continue
            cand = np.maximum(cand, 1e-18/n)
            cand = cand / cand.sum()
            sc = _score(cand)
            if sc < cur_score - 1e-12:
                cur_score = sc
                cur_best = cand
        
        if cur_best is not p:
            p = cur_best
            if cur_score < best_score:
                best = p.copy()
                best_score = cur_score
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 40:
                p_try = _sanitize(p * np.exp(rng.normal(0, 0.1, n)))
                if p_try is not None:
                    p_try = np.maximum(p_try, 1e-18/n)
                    p_try = p_try / p_try.sum()
                    sc = _score(p_try)
                    if sc < best_score * 1.1:
                        p = p_try
                    else:
                        p = best.copy()
                stagnant = 0
    
    return best, best_score

def _resample(arr, n):
    """Resample to length n."""
    a = _sanitize(arr)
    if a is None:
        return None
    if len(a) == n:
        return a.copy()

    x_new = (np.arange(n) + 0.5) / n
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0) + 1e-12
    return _sanitize(out)

Minimize 2*n*max(conv(a,a))/sum(a)^2 using adaptive SMC (Smooth Multiplicative
Costs) descent with backtracking line search, multi-length optimization,
LP-inspired direction finding, and block coordinate polishing.

import numpy as np
import time
import scipy.fft


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _full_conv(a, b):
    nout = len(a) + len(b) - 1
    L = _next_pow2(nout)
    fa = scipy.fft.rfft(a, L)
    fb = scipy.fft.rfft(b, L)
    return scipy.fft.irfft(fa * fb, L)[:nout]


def _valid_corr(c, a):
    """result[j] = sum_{i=0}^{n-1} a[i] * c[j+i] (valid cross-correlation)."""
    L = _next_pow2(max(len(c), len(a)))
    fc = scipy.fft.rfft(c, L)
    fa = scipy.fft.rfft(a, L)
    out = scipy.fft.irfft(fc * np.conjugate(fa), L)
    return out[:len(a)]


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return p / s


def _score(p):
    p = _sanitize(p)
    if p is None:
        return float('inf')
    n = len(p)
    c = _full_conv(p, p)
    c = np.where(np.isfinite(c), c, 0.0)
    return float(2.0 * n * c.max())


def _smooth_max_and_grad(p, beta):
    """Return (smooth_value, weights, gradient) for smoothed max objective."""
    n = len(p)
    c = _full_conv(p, p)
    c = np.where(np.isfinite(c), c, 0.0)
    d = n * c
    shift = float(d.max())
    if not np.isfinite(shift):
        return float('inf'), None, None
    e = np.exp(beta * (d - shift))
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        return float('inf'), None, None
    w = e / s
    smooth = float(np.dot(w, d))
    grad = 2.0 * _valid_corr(w, p)
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return smooth, w, grad


def _eval_smooth(p, beta):
    """Quick smooth evaluation for line search."""
    n = len(p)
    c = _full_conv(p, p)
    c = np.where(np.isfinite(c), c, 0.0)
    d = n * c
    shift = float(d.max())
    if not np.isfinite(shift):
        return float('inf')
    e = np.exp(beta * (d - shift))
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        return float('inf')
    return float(np.dot(e / s, d))

                sc = _score(sp)
                scored3.append((sc, sp))
            scored3.sort(key=lambda x: x[0])

            n_starts3 = min(3, len(scored3))
            for i in range(n_starts3):
                rem = deadline - time.monotonic()
                if rem < 80:
                    break
                t = min(rem * 0.25, 300.0)
                if t < 20:
                    break
                print(f"  Run {i+1}/{n_starts3}: start_score={scored3[i][0]:.6f}, budget={t:.0f}s")
                p, sc = _run_smc(scored3[i][1], t, beta0=20, beta_growth=1.04,
                                 beta_max=4000, alpha0=0.4, rng=rng, verbose=True)
                _try_update(p, sc)

    # === Phase 4: Polish the best ===
    print(f"\nPhase 4: Polishing (current best={best_score:.6f}, n={len(best_arr) if best_arr is not None else 0})")
    rem = deadline - time.monotonic()

    if best_arr is not None and rem > 100:
        t1 = min(rem * 0.35, 350.0)
        print(f"  SMC polish: budget={t1:.0f}s, beta0=80")
        p, sc = _run_smc(best_arr, t1, beta0=80, beta_growth=1.03,
                         beta_max=8000, alpha0=0.3, rng=rng, verbose=True)
        _try_update(p, sc)

    rem = deadline - time.monotonic()
    if best_arr is not None and rem > 80:
        t2 = min(rem * 0.35, 250.0)
        print(f"  Block search: budget={t2:.0f}s")
        p, sc = _block_search(best_arr, t2, rng, verbose=True)
        _try_update(p, sc)

    rem = deadline - time.monotonic()
    if best_arr is not None and rem > 60:
        t3 = min(rem * 0.5, 200.0)
        print(f"  Final SMC: budget={t3:.0f}s, beta0=200")
        p, sc = _run_smc(best_arr, t3, beta0=200, beta_growth=1.02,
                         beta_max=10000, alpha0=0.2, rng=rng, verbose=True)
        _try_update(p, sc)

    # === Phase 5: Up-polish to larger n ===
    rem = deadline - time.monotonic()
    if best_arr is not None and rem > 180:
        cur_n = len(best_arr)
        big_n = min(8192, max(4096, cur_n + 2048))
        if big_n > cur_n:
            print(f"Phase 5: Up-polish to n={big_n}")
            big = _resample(best_arr, big_n)
            if big is not None:
                t4 = min(rem * 0.5, 250.0)
                p, sc = _run_smc(big, t4, beta0=50, beta_growth=1.04,
                                 beta_max=5000, alpha0=0.3, rng=rng, verbose=True)
                _try_update(p, sc)

                rem = deadline - time.monotonic()
                if p is not None and rem > 50:
                    t5 = min(rem * 0.4, 100.0)
                    pl, scl = _block_search(p, t5, rng)
                    _try_update(pl, scl)

    # === Final evaluation ===
    print(f"\nFinal: best_score={best_score:.6f}, n={len(best_arr) if best_arr is not None else 0}")

    if best_arr is None:
        best_arr = np.ones(3072, dtype=float)

    out = _sanitize(best_arr)
    if out is None:
        out = np.ones(3072, dtype=float)

    lst = [float(x) for x in out]

    # Verify with official evaluator
    try:
        sc = float(evaluate_sequence(lst))
        print(f"Official evaluator score: {sc:.6f}")
        if not np.isfinite(sc):
            print("WARNING: Official evaluator returned inf!")
    except Exception as e:
        print(f"Could not call official evaluator: {e}")

    # Time used
    print(f"Total time: {time.monotonic() - start_time:.1f}s / {budget:.0f}s")

    return lst
