    return 2.0 * a.size * float(np.max(b)) / (s * s)

"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based iterative refinement.
Each iteration solves a linear program to find the optimal improvement direction,
then takes a projected step. Uses adaptive step sizes, multiple sequence lengths,
and diverse initializations.
"""

import numpy as np
from scipy.optimize import linprog
import time


def full_conv(a):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def build_A(f):
    """Build constraint matrix A where (f*g)[k] = sum_i A[k,i]*g[i]."""
    n = f.size
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lo = max(0, k - n + 1)
        hi = min(k, n - 1)
        if lo <= hi:
            A[k, lo:hi+1] = f[k-hi:k-lo+1][::-1]
    return A


def solve_lp(f, M):
    """Solve: max sum(g) s.t. g >= 0, conv(f,g) <= M. Returns g or None."""
    n = f.size
    A = build_A(f)
    b_ub = np.full(2 * n - 1, M)
    c = -np.ones(n)
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub,
                      bounds=[(0, None)] * n, method='highs',
                      options={'presolve': True, 'time_limit': 30.0})
        if res.success:
            return np.maximum(res.x, 0)
    except Exception:
        pass
    return None

        g = solve_lp(a, M)
        if g is None or g.sum() < 1e-15:
            break
        g = g / g.sum()
        
        # Line search over step sizes
        best_t = t_step
        best_t_sc = np.inf
        for tc in [t_step, 0.5*t_step, 0.25*t_step, 0.1*t_step, 0.05*t_step]:
            a_new = (1 - tc) * a + tc * g
            a_new = np.maximum(a_new, 0)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc = normed_score(a_new)
            if sc < best_t_sc:
                best_t_sc = sc
                best_t = tc
        
        a = (1 - best_t) * a + best_t * g
        a = np.maximum(a, 0)
        a = a / a.sum()
        
        cur_sc = normed_score(a)
        iter_count += 1
        
        if cur_sc < best_sc - 1e-10:
            best_sc = cur_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                t_step = max(0.05, t_step * 0.7)
                if t_step < 0.06:
                    # Perturb and restart
                    a = best_a + 0.01 * rng.standard_normal(n)
                    a = np.maximum(a, 0)
                    a = a / a.sum()
                    t_step = 0.3
                    stall = 0
        
        if verbose and iter_count % 50 == 0:
            elapsed = time.monotonic() - t0
            print(f"    iter={iter_count}, best_sc={best_sc:.6f}, M={M:.8f}, t={t_step:.4f}, elapsed={elapsed:.1f}s")
    
    return best_a, best_sc


def init_uniform(n):
    return np.ones(n) / n


def init_u_shape(n, power=0.5):
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i + 1, n - i).clip(min=1.0)
    a = 1.0 / (m ** power)
    return a / a.sum()


def init_cosine(n, freq=1):
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + freq * np.cos(2.0 * np.pi * i / n)) ** 2
    a = np.maximum(a, 1e-4)
    return a / a.sum()


def init_exp_edge(n):
    """Exponential edge weights: high at edges."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-i / (n * 0.3)) + np.exp(-(n - 1 - i) / (n * 0.3))
    a += 0.01
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.05
    return a / a.sum()


def get_inits(n, rng):
    inits = [
        init_uniform(n),
        init_u_shape(n, 0.5),
        init_u_shape(n, 1.0),
        init_u_shape(n, 0.3),
        init_cosine(n, 1),
        init_cosine(n, 2),
        init_exp_edge(n),
        init_random(n, rng),
    ]
    return inits
