def full_conv(a):

    n = a.size

def build_conv_matrix(a):
    """Build matrix T where T @ g = convolve(a, g). T is (2n-1) x n."""
    n = a.size
    k_idx = np.arange(2 * n - 1)
    j_idx = np.arange(n)
    diff = k_idx[:, None] - j_idx[None, :]
    mask = (diff >= 0) & (diff < n)
    T = np.zeros((2 * n - 1, n))
    T[mask] = a[diff[mask]]
    return T

def project_simplex(v):
    """Project onto probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based alternating
method (Frank-Wolfe/conditional gradient on autocorrelation constraints),
combined with subgradient polish, spectral-motivated initializations,
and multi-scale exploration. Uses cvxpy for exact LP solves and FFT
for fast convolution.

import numpy as np
import time
import cvxpy as cp


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def build_conv_matrix(a):
    """Build dense matrix T (shape (2n-1, n)) such that T @ g = np.convolve(a, g)."""
    n = a.size
    k_idx = np.arange(2 * n - 1)
    j_idx = np.arange(n)
    diff = k_idx[:, None] - j_idx[None, :]
    mask = (diff >= 0) & (diff < n)
    T = np.zeros((2 * n - 1, n))
    T[mask] = a[diff[mask]]
    return T


def solve_lp(a, M, timeout=10.0):
    """Solve: max sum(g) s.t. conv(a,g)[k] <= M for all k, g >= 0.
    Returns optimal g (or a as fallback)."""
    n = a.size
    g = cp.Variable(n, nonneg=True)
    T = build_conv_matrix(a)
    constraints = [T @ g <= M]
    objective = cp.Maximize(cp.sum(g))
    prob = cp.Problem(objective, constraints)
    
    solvers = [cp.ECOS, cp.CLARABEL, cp.SCS, cp.PDLP]
    for solver in solvers:
        try:
            prob.solve(solver=solver, verbose=False)
            if g.value is not None:
                return np.maximum(np.asarray(g.value, dtype=np.float64), 0.0)
        except Exception:
            continue
    
    # Fallback: return a itself (which is feasible)
    return a.copy()

def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def smooth_grad_single(a, b, tau):
    """Smoothed subgradient using softmax weights."""
    n = a.size
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(b.size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= s
    return 2.0 * np.correlate(w, a, mode='valid')

    m = np.minimum(i, n - 1 - i) + 1.0
    a = 1.0 / np.sqrt(m)
    a += 0.02 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_two_cluster(n, rng):
    """Two Gaussian bumps."""
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.15 + 0.15 * rng.random())
    c2 = n * (0.70 + 0.15 * rng.random())
    sigma = n * (0.04 + 0.08 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.02 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.05
    return a / a.sum()


def init_sided(n, rng):
    """Quadratic: high at edges."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / c) ** 2 + 0.1
    a += 0.02 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_fourier_random(n, rng):
    """Random Fourier features made non-negative."""
    i = np.arange(n, dtype=np.float64)
    sig = np.ones(n)
    n_harm = int(rng.integers(2, 6))
    for _ in range(n_harm):
        f = int(rng.integers(1, max(2, n // 2)))
        amp = rng.uniform(0.2, 0.8)
        phi = 2 * np.pi * rng.random()
        sig += amp * np.cos(2.0 * np.pi * f * i / n + phi)
    a = np.maximum(sig, 0.0) ** 2 + 0.001
    return a / a.sum()


def get_inits(n, rng):
    """Generate diverse initializations."""
    inits = []
    for f in [1, 2, 3, 5]:
        if f < n // 2:
            inits.append(init_cosine_sq(n, f, rng))
    inits.append(init_multi_cosine(n, rng))
    inits.append(init_fourier_random(n, rng))
    inits.append(init_u_shape(n, rng))
    inits.append(init_two_cluster(n, rng))
    inits.append(init_random(n, rng))
    inits.append(init_sided(n, rng))
    return inits


def polish_random(a, time_budget, rng):
    """Local polish via random perturbations with decreasing scale."""
    n = a.size
    a = a.astype(np.float64).copy()
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_sc = normed_score(a)
    
    t0 = time.monotonic()
    scale = 0.02
    stuck = 0
    while time.monotonic() - t0 < time_budget and scale > 1e-6:
        improved = False
        for _ in range(8):
            cand = a + scale * rng.standard_normal(n)
            cand = np.maximum(cand, 0.0)
            if cand.sum() < 1e-10:
                continue
            cand = cand / cand.sum()
            sc = normed_score(cand)
            if sc < best_sc:
                best_sc = sc
                best_a = cand.copy()
                a = cand.copy()
                improved = True
        if not improved:
            stuck += 1
            if stuck > 6:
                break
        else:
            stuck = 0
        scale *= 0.88
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses LP-based alternating method (Frank-Wolfe) as the main optimizer,
    combined with subgradient descent for larger n, diverse initializations,
    and multi-stage polish.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)
    
    # Load previous best
    prev_a = None
    prev_sc = np.inf
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size >= 10 and np.all(np.isfinite(h)) and h.sum() > 0.01:
            prev_a = h / h.sum()
            prev_sc = normed_score(prev_a)
            print(f"[init] Previous best: n={h.size}, score={prev_sc:.6f}")
    except Exception:
        pass
    
    best_overall = None
    best_overall_sc = np.inf
    
    if prev_a is not None and prev_sc < 2.0:
        best_overall = prev_a.copy()
        best_overall_sc = prev_sc
    
    # Phase 1: LP-based optimization for smaller n
    n_lp_list = [128, 256, 512, 1024]
    if prev_a is not None and prev_a.size in [128, 256, 512, 1024]:
        # Ensure prev best length is in the list
        pass
    
    total_time = max(1.0, deadline - time.monotonic())
    phase1_time = total_time * 0.50
    phase2_time = total_time * 0.20
    phase3_time = total_time * 0.15
    polish_time = total_time * 0.15
    
    print(f"[plan] total={total_time:.1f}s | LP={phase1_time:.1f}s SG={phase2_time:.1f}s "
          f"explore={phase3_time:.1f}s polish={polish_time:.1f}s")
    
    # --- Phase 1: LP-based optimization ---
    if prev_a is not None and prev_a.size not in n_lp_list and prev_a.size <= 2048:
        n_lp_list.append(prev_a.size)
        n_lp_list = sorted(n_lp_list)
    
    # Weight: smaller n gets more time (faster LP)
    raw_w = []
    for nv in n_lp_list:
        w = 1.0 / (0.3 + nv / 512.0)
        if prev_a is not None and nv == prev_a.size:
            w *= 2.0
        raw_w.append(w)
    tw = sum(raw_w)
    weights = [w / tw for w in raw_w]
    
    phase1_deadline = time.monotonic() + phase1_time
    for idx_n, nv in enumerate(n_lp_list):
        if time.monotonic() > phase1_deadline:
            break
        
        time_n = weights[idx_n] * phase1_time
        inits = get_inits(nv, rng)
        
        if prev_a is not None and prev_a.size == nv:
            inits.insert(0, prev_a.copy())
        
        inits_per = time_n / len(inits)
        
        for ii, a_init in enumerate(inits):
            if time.monotonic() > phase1_deadline:
                break
            tb = min(inits_per, phase1_deadline - time.monotonic() - 2.0)
            if tb < 2.0:
                break
            
            a_best, sc_best = lp_alternating(a_init, tb, rng)
            
            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = a_best.copy()
                print(f"  [LP] n={nv}, init={ii}, score={sc_best:.6f} *** BEST ***")
            else:
                print(f"  [LP] n={nv}, init={ii}, score={sc_best:.6f}")
    
    print(f"[phase1 done] best={best_overall_sc:.6f} (n={best_overall.size if best_overall is not None else '?'})")
    
    # --- Phase 2: Subgradient for larger n (exploration) ---
    n_sg_list = [2048, 4096]
    if prev_a is not None and prev_a.size > 2048:
        n_sg_list.append(prev_a.size)
    n_sg_list = sorted(set(n_sg_list))
    
    phase2_deadline = time.monotonic() + phase2_time
    sg_weights = [1.0 / (0.5 + nv / 1024.0) for nv in n_sg_list]
    sw = sum(sg_weights)
    sg_weights = [w / sw for w in sg_weights]
    
    for idx_n, nv in enumerate(n_sg_list):
        if time.monotonic() > phase2_deadline:
            break
        
        time_n = sg_weights[idx_n] * phase2_time
        inits = get_inits(nv, rng)
        inits_per = time_n / len(inits)
        
        for ii, a_init in enumerate(inits[:4]):  # limit to 4 inits for large n
            if time.monotonic() > phase2_deadline:
                break
            tb = min(inits_per, phase2_deadline - time.monotonic() - 2.0)
            if tb < 2.0:
                break
            
            a_best, sc_best = subgrad_optimize(a_init, tb, rng, use_momentum=True)
            
            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = a_best.copy()
                print(f"  [SG] n={nv}, init={ii}, score={sc_best:.6f} *** BEST ***")
            else:
                print(f"  [SG] n={nv}, init={ii}, score={sc_best:.6f}")
    
    # --- Phase 3: Second round of LP on the best found (refinement) ---
    if best_overall is not None:
        phase3_deadline = time.monotonic() + phase3_time
        n_best = best_overall.size
        
        # Try LP refinement on the current best with different perturbations
        for trial in range(5):
            if time.monotonic() > phase3_deadline:
                break
            tb = min(30.0, phase3_deadline - time.monotonic() - 2.0)
            if tb < 2.0:
                break
            
            # Perturb the best and re-optimize
            perturbation = 0.03 * rng.standard_normal(n_best)
            a_pert = best_overall + perturbation
            a_pert = np.maximum(a_pert, 0.0)
            a_pert = a_pert / a_pert.sum()
            
            a_best, sc_best = lp_alternating(a_pert, tb, rng)
            
            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = a_best.copy()
                print(f"  [refine] trial={trial}, score={sc_best:.6f} *** BEST ***")
    
    # --- Phase 4: Final polish ---
    if best_overall is not None and time.monotonic() < deadline - 3.0:
        remain = deadline - time.monotonic() - 3.0
        if remain > 5.0:
            n_best = best_overall.size
            print(f"[polish] remaining={remain:.1f}s, n={n_best}")
            
            # LP polish
            lp_pol_time = min(remain * 0.4, 60.0)
            if lp_pol_time > 3.0:
                a_p, sc_p = lp_alternating(best_overall, lp_pol_time, rng)
                if sc_p < best_overall_sc:
                    best_overall_sc = sc_p
                    best_overall = a_p.copy()
                    print(f"  [polish] LP: {sc_p:.6f}")
            
            # Subgradient polish
            sg_pol_time = min(remain * 0.3, 45.0)
            if sg_pol_time > 3.0:
                a_p2, sc_p2 = subgrad_optimize(best_overall, sg_pol_time, rng, use_momentum=True)
                if sc_p2 < best_overall_sc:
                    best_overall_sc = sc_p2
                    best_overall = a_p2.copy()
                    print(f"  [polish] SG: {sc_p2:.6f}")
            
            # Random polish
            if time.monotonic() < deadline - 3.0:
                rp_time = min(remain * 0.2, 30.0)
                if rp_time > 2.0:
                    a_p3, sc_p3 = polish_random(best_overall, rp_time, rng)
                    if sc_p3 < best_overall_sc:
                        best_overall_sc = sc_p3
                        best_overall = a_p3.copy()
                        print(f"  [polish] random: {sc_p3:.6f}")
    
    # Fallback
    if best_overall is None:
        best_overall = np.ones(512) / 512
        best_overall_sc = 2.0
    
    # Ensure non-negative and finite
    best_overall = np.maximum(best_overall, 0.0)
    best_overall = np.nan_to_num(best_overall, nan=0.0, posinf=1000.0, neginf=0.0)
    best_overall = np.minimum(best_overall, 1000.0)
    
    # Final verification
    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"[final] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")
    
    return result
