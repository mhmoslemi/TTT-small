def fft_convolution(a):
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)

def grad_smooth_max(p, w):
    """Gradient of smooth max of p*p w.r.t. p, using softmax weights w."""
    n = p.size
    m = w.size  # 2n-1
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n : m][::-1].copy()
    return 2.0 * s

def softmax_weights(b, beta):
    """Compute softmax weights and smooth max value."""
    m = float(np.max(b))
    if m <= 0 or beta <= 0:
        w = np.ones_like(b) / b.size
        return m, w

def adam_update(params, grads, m_params, v_params, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam update in-place."""
    bc1 = 1 - beta1 ** t
    bc2 = 1 - beta2 ** t

        m_params[i] = beta1 * m_params[i] + (1 - beta1) * grads[i]
        v_params[i] = beta2 * v_params[i] + (1 - beta2) * (grads[i] ** 2)
        m_hat = m_params[i] / bc1
        v_hat = v_params[i] / bc2
        params[i] -= lr * m_hat / (np.sqrt(v_hat) + eps)

"""
Minimize 2*n*max(a*a)/(sum(a))^2 for non-negative sequences.
Strategy: Adam optimizer on log-parameterized sequence with beta annealing,
starting from known good solution, followed by subgradient polish.
"""

import numpy as np
import time

def fft_conv(a):
    """Linear convolution a*a using FFT."""
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)

def smooth_max_weights(b, beta):
    """Return (smooth_max_value, softmax_weights)."""
    m = float(np.max(b))
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b) / b.size
        return m, w
    x = beta * (b - m)
    np.clip(x, -745.0, 0.0, out=x)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z < 1e-30 or not np.isfinite(Z):
        e = np.zeros_like(b)
        e[np.argmax(b)] = 1.0
        Z = 1.0
    w = e / Z
    return m + np.log(Z) / beta, w

def grad_conv(p, w):
    """Gradient of smooth_max(p*p) w.r.t. p, given softmax weights w on p*p."""
    n = p.size
    m = w.size  # 2n-1
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n : m][::-1].copy()
    return 2.0 * s

    """Compute 2*n*max(p*p) for sum-normalized p."""
    n = p.size
    S = float(np.sum(p))

    b = fft_conv(p)

                Fw = F * (1 + 0.2 * rng.standard_normal(N // 2 + 1) * mask)
                p_s = np.real(np.fft.irfft(Fw, N))
            
            p_s = norm1(p_s)
            pr, sr = adam_log(p_s, dl, beta_lo=5000, beta_hi=300000, lr=0.3, max_iter=400000, patience=60000)

                best_score = sr

"""
Minimize 2*n*max(a*a)/(sum(a))^2 for non-negative sequences.

Strategy:
- Start from known good solution (height_sequence_1) if available
- Use Adam optimizer on log-parameterization with beta annealing
- Multiple perturbation restarts for escaping local minima
- Final subgradient polish with exact max
- Time-budgeted multi-phase approach
"""

import numpy as np
import time


def fft_conv(a):
    """Linear convolution a*a using FFT, clipped to non-negative."""
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_weights(b, beta):
    """Return (smooth_max_value, softmax_weights) for b with temperature beta."""
    m = float(np.max(b))
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b, dtype=np.float64) / b.size
        return m, w
    x = beta * (b - m)
    np.clip(x, -745.0, 0.0, out=x)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z < 1e-300 or not np.isfinite(Z):
        e = np.zeros_like(b, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return G, w


def grad_conv(p, w):
    """Gradient of smooth_max(p*p) w.r.t. p, given weights w on p*p."""
    n = p.size
    m = w.size  # 2n-1
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n:m][::-1].copy()
    return 2.0 * s


def score_of(p):
    """Compute 2*n*max(p*p)/(sum(p))^2."""
    n = p.size
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float('inf')
    b = fft_conv(p)
    return 2.0 * n * float(np.max(b)) / (S * S)


def norm1(p):
    """Clip to non-negative and normalize to sum=1."""
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.ones_like(p, dtype=np.float64) / p.size
    return p / s


def adam_optimize(p0, deadline, beta_lo=3000.0, beta_hi=300000.0, lr=0.05,
                  max_iter=1000000, patience=100000, b1=0.9, b2=0.999):
    """
    Adam optimizer on log(p) parameterization with beta annealing.
    Returns (best_p, best_score).
    """
    n = p0.size
    p = norm1(p0.copy())
    logp = np.log(np.maximum(p, 1e-30)).copy()

    mu = np.zeros(n, dtype=np.float64)
    nu = np.zeros(n, dtype=np.float64)

    best_p = p.copy()
    best_score = 2.0 * n * float(np.max(fft_conv(p)))
    no_imp = 0
    t = 0

    while t < max_iter and time.monotonic() < deadline:
        t += 1

        # Logarithmic beta annealing
        frac = min(1.0, t / (0.8 * max_iter))
        beta = beta_lo * (beta_hi / beta_lo) ** frac

        b = fft_conv(p)
        m_cur = float(np.max(b))
        _, w = smooth_max_weights(b, beta)
        g = grad_conv(p, w)
        g -= g.mean()

        # Adam update
        bc1 = 1.0 - b1 ** t
        bc2 = 1.0 - b2 ** t
        mu = b1 * mu + (1.0 - b1) * g
        nu = b2 * nu + (1.0 - b2) * (g * g)
        step = lr * (mu / bc1) / (np.sqrt(nu / bc2) + 1e-8)
        logp -= step

        # Numerical stability and normalization
        logp -= logp.max()
        p = np.exp(logp)
        p = norm1(p)

        # Check improvement (use current b as proxy, recomputing occasionally)
        if t % 50 == 0 or t < 10:
            sc = 2.0 * n * float(np.max(fft_conv(p)))
        else:
            sc = 2.0 * n * m_cur  # approximate (stale)

        if sc < best_score - 1e-13:
            best_score = 2.0 * n * float(np.max(fft_conv(p)))
            best_p = p.copy()
            no_imp = 0
        else:
            no_imp += 1

        if t % 50000 == 0:
            print(f"    adam t={t} beta={beta:.0f} sc={best_score:.8f} no_imp={no_imp}")

        if no_imp > patience:
            break

    return best_p, best_score


def subgrad_polish(p0, deadline, beta=1000000.0, max_iter=500000):
    """
    Projected subgradient descent with backtracking.
    Uses nearly-exact max (high beta) for the gradient direction.
    """
    n = p0.size
    p = norm1(p0.copy())
    best_p = p.copy()
    best_score = 2.0 * n * float(np.max(fft_conv(p)))
    step = 1e-3

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        b = fft_conv(p)
        m = float(np.max(b))
        _, w = smooth_max_weights(b, beta)
        g = grad_conv(p, w)
        g -= g.mean()
        gn = float(np.linalg.norm(g))
        if gn < 1e-18:
            break

        q = p - (step / gn) * g
        q = np.maximum(q, 0.0)
        qs = float(q.sum())
        if qs <= 0.0:
            step *= 0.5
            continue
        q /= qs

        bq = fft_conv(q)
        mq = float(np.max(bq))
        if mq < m - 1e-14:
            p = q
            sc = 2.0 * n * mq
            if sc < best_score:
                best_score = sc
                best_p = p.copy()
            step = min(step * 1.03, 0.5)
        else:
            step *= 0.7
            if step < 1e-12:
                break

        if it % 50000 == 0:
            print(f"    polish it={it} sc={best_score:.8f} step={step:.2e}")

    return best_p, best_score
