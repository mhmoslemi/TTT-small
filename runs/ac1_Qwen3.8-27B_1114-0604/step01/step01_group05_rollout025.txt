```python
"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences.
Uses multi-scale multiplicative weights optimization with diverse initializations.
"""

import numpy as np
import time
from scipy.optimize import minimize

def fft_convolve(a):
    """Linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    if n == 1:
        return np.array([a[0]*a[0]])
    need = 2*n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F*F, size)[:need]
    return np.maximum(b, 0.0)

def score(p):
    """Compute 2n*max(conv)/(sum)^2."""
    S = float(np.sum(p))
    if S <= 0:
        return float('inf')
    b = fft_convolve(p)
    M = float(np.max(b))
    return float(2.0 * len(p) * M / (S*S))

def smooth_max(b, beta):
    """Smooth max and subgradient weights."""
    m = float(np.max(b))
    if m <= 0:
        w = np.ones_like(b) / len(b)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -700, 0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0:
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        w = e / Z
    G = m + np.log(Z) / beta
    return G, w

def corr(p, w):
    """s[i] = sum_j p[j] * w[i+j] for i=0..n-1."""
    n = len(p)
    m = len(w)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m-n:m][::-1].copy()
    return np.maximum(s, 0.0)

def normalize(p):
    """Normalize to sum=1, non-negative."""
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0:
        return np.ones_like(p) / len(p)
    return p / s

def multiplicative_weights(p, deadline, beta, max_iter=500000):
    """Optimize using multiplicative weights (mirror descent)."""
    p = normalize(p.copy())
    n = len(p)
    best_p = p.copy()
    best_score = score(p)
    
    eta = 1.0
    no_improve = 0
    it = 0
    
    while it < max_iter and time.monotonic() < deadline:
        it += 1
        
        b = fft_convolve(p)
        G, w = smooth_max(b, beta)
        
        cur_smooth = n * G
        cur_score = 2.0 * n * float(np.max(b))
        
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 5000:
            break
        
        if it % 500 == 0:
            print(f"  MW it={it} score={best_score:.8f} eta={eta:.4f}")
        
        # Gradient of smooth max
        svec = corr(p, w)
        grad = 2.0 * svec - float(np.mean(2.0 * svec))
        
        # Multiplicative weights update
        p_new = p * np.exp(-eta * grad)
        p_new = normalize(p_new)
        
        # Check if improvement
        b_new = fft_convolve(p_new)
        G_new, _ = smooth_max(b_new, beta)
        
        if n * G_new < cur_smooth * (1 - 1e-12):
            p = p_new
            # Increase eta slightly if improving
            if eta < 10:
                eta = min(eta * 1.02, 10.0)
        else:
            # Decrease eta
            eta *= 0.7
            if eta < 1e-6:
                break
    
    return best_p, best_score
```

"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences.
Uses multi-scale optimization: L-BFGS on block parameters, then full-length
multiplicative weights (mirror descent) with adaptive smooth-max beta,
plus periodic LP-inspired restarts for escaping local minima.

import numpy as np
import time

def fft_convolve(a):
    """Linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def fast_score(p):
    """Compute 2n*max(conv)/(sum)^2."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float('inf')
    S = float(np.sum(p))
    if S <= 0 or not np.isfinite(S):
        return float('inf')
    b = fft_convolve(p)
    M = float(np.max(b))
    if not np.isfinite(M):
        return float('inf')
    return float(2.0 * n * M / (S * S))


def smooth_max_and_weights(b, beta):
    """Smooth max of b and subgradient weights."""
    b = np.asarray(b, dtype=np.float64)
    m = float(np.max(b))
    if m <= 0 or beta <= 0:
        w = np.ones_like(b) / b.size
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -700.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0 or not np.isfinite(Z):
        w = np.zeros_like(b, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
    else:
        w = e / Z
    G = m + np.log(max(Z, 1e-300)) / beta
    return float(G), w


def corr_pw(p, w):
    """s[i] = sum_j p[j] * w[i+j] for i=0..n-1."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n:m][::-1].copy()
    return np.maximum(s, 0.0)


def normalize_simplex(p):
    """Clip non-negative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size), dtype=np.float64)
    return p / s


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    if target <= 0:
        return np.zeros_like(v)
    v = np.where(np.isfinite(v), v, 0.0)
    low = float(np.min(v)) - 1.0
    high = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (low + high)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0 or not np.isfinite(s):
        return np.full(v.shape, target / float(v.size), dtype=np.float64)
    return q * (target / s)


def block_to_full(n, B, c):
    """Convert block log-weights c (length B-1) to full sequence p (length n)."""
    L = n // B
    z = np.zeros(B, dtype=np.float64)
    if B > 1 and c is not None:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        m = min(B - 1, c.size)
        if m > 0:
            z[:m] = c[:m]
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0 or not np.isfinite(s):
        wb = np.ones(B) / B
    else:
        wb = wb / s
    H = wb / L
    return np.repeat(H, L)


def block_obj(c, n, B, beta):
    """Objective and gradient for block-constant representation."""
    L = n // B
    z = np.zeros(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        m = min(B - 1, c.size)
        if m > 0:
            z[:m] = c[:m]
    if not np.all(np.isfinite(z)):
        return float('inf'), np.zeros(max(0, B - 1), dtype=np.float64)
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0 or not np.isfinite(s):
        wb = np.ones(B) / B
        z = np.zeros(B)
    else:
        wb = wb / s
    H = wb / L
    p = np.repeat(H, L)
    b = fft_convolve(p)
    G, w = smooth_max_and_weights(b, beta)
    if not np.isfinite(G):
        return float('inf'), np.zeros(max(0, B - 1), dtype=np.float64)
    svec = corr_pw(p, w)
    g = 2.0 * svec
    Gsum = g.reshape(B, L).sum(axis=1)
    Wavg = float(np.dot(wb, Gsum))
    gradz = (wb / L) * (Gsum - Wavg)
    grad_out = n * gradz[:-1]
    grad_out = np.where(np.isfinite(grad_out), grad_out, 0.0)
    return float(n * G), grad_out.astype(np.float64)


def run_lbfgs(c0, n, B, beta, max_iter=100):
    """Run L-BFGS-B on block parameters."""
    if not HAVE_SCIPY:
        return np.asarray(c0, dtype=np.float64)
    c0 = np.asarray(c0, dtype=np.float64).reshape(-1)
    expected = max(0, B - 1)
    if c0.size != expected:
        tmp = np.zeros(expected)
        m = min(tmp.size, c0.size)
        tmp[:m] = c0[:m]
        c0 = tmp
    c0 = np.clip(c0, -8.0, 8.0)
    bounds = [(-8.0, 8.0)] * expected
    options = {"maxiter": max_iter, "ftol": 1e-12, "gtol": 1e-6, "maxls": 30, "maxcor": 12}
    try:
        res = minimize(block_obj, c0, args=(n, B, beta), method="L-BFGS-B", jac=True,
                       bounds=bounds, options=options)
        c = np.asarray(res.x, dtype=np.float64)
    except Exception:
        c = c0
    if not np.all(np.isfinite(c)):
        c = c0
    return np.clip(c, -8.0, 8.0)


def multiplicative_weights_refine(p0, deadline, beta_start=500.0, beta_end=50000.0,
                                   max_iter=1000000, eta0=1.0):
    """Full-length mirror descent (multiplicative weights) with adaptive beta."""
    p = normalize_simplex(p0.copy())
    n = len(p)
    best_p = p.copy()
    best_score = fast_score(p)

    eta = eta0
    no_improve = 0
    it = 0
    total_iter = max_iter

    # Adaptive beta: start low, increase over time
    n_beta_steps = 30
    beta_schedule = np.linspace(np.log(beta_start), np.log(beta_end), n_beta_steps)

    while it < total_iter and time.monotonic() < deadline:
        it += 1

        # Update beta adaptively
        frac = min(1.0, it / (total_iter * 0.7))
        beta_idx = min(int(frac * n_beta_steps), n_beta_steps - 1)
        beta = np.exp(beta_schedule[beta_idx])

        b = fft_convolve(p)
        G, w = smooth_max_and_weights(b, beta)
        if not np.isfinite(G):
            break

        cur_smooth = float(n) * G
        m = float(np.max(b))
        cur_score = float(2.0 * n * m)

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 1000 == 0:
            print(f"    MW it={it} score={best_score:.9f} beta={beta:.0f} eta={eta:.4f} noimp={no_improve}")

        if no_improve > 8000:
            break

        svec = corr_pw(p, w)
        grad = 2.0 * svec
        grad_centered = grad - float(np.mean(grad))

        gnorm = float(np.linalg.norm(grad_centered))
        if not np.isfinite(gnorm) or gnorm < 1e-15:
            break

        # Multiplicative weights update
        log_ratio = -eta * grad_centered
        log_ratio = np.clip(log_ratio, -50.0, 50.0)
        p_new = p * np.exp(log_ratio)
        p_new = normalize_simplex(p_new)

        # Check improvement
        b_new = fft_convolve(p_new)
        G_new, _ = smooth_max_and_weights(b_new, beta)
        new_smooth = float(n) * G_new

        if new_smooth < cur_smooth - 1e-14:
            p = p_new
            if eta < 5.0:
                eta = min(eta * 1.01, 5.0)
        else:
            eta *= 0.8
            if eta < 1e-8:
                break

    return best_p, best_score
