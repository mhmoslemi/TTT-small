```python
def lp_step(a, M):
    n = len(a)
    # Variables: b[0], ..., b[n-1]
    # Objective: maximize sum(b) -> minimize -sum(b)
    c = -np.ones(n)
    
    # Equality: sum(b) = 1
    A_eq = np.ones((1, n))
    b_eq = [1.0]
    
    # Inequality: conv(a, b)[k] <= M for all k
    # conv(a, b)[k] = sum_{i} a[i] * b[k-i] for valid i
    # For a given k, the valid i range is max(0, k-(n-1)) to min(k, n-1)
    # b index is k-i, so b_j appears in conv[k] when j = k-i, i.e., i = k-j
    # conv[a,b][k] = sum_j a[k-j] * b[j] for valid indices
    # where k-j is in [0, n-1] and j is in [0, n-1]
    
    n_constr = 2 * n - 1
    A_ub = np.zeros((n_constr, n))
    b_ub = np.full(n_constr, M)
    
    for k in range(n_constr):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_ub[k, j] = a[i]
    
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    else:
```

```python
import numpy as np
import time
from scipy.optimize import linprog

def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def conv_ab(a, b):
    """Full linear convolution a*b via FFT."""
    na, nb = a.size, b.size
    size = na + nb - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
    return np.fft.irfft(fa * fb, fsize)[:size]

def score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    x = np.maximum(v, 0.0).copy()
    s = x.sum()
    if s < 1e-30:
        return np.ones(n) / n
    x /= s
    return x

def lp_direction(a, M):
    """
    Solve LP: max sum(b) s.t. b >= 0, conv(a,b)[k] <= M for all k.
    Returns the optimal b (not normalized).
    """
    n = a.size
    n_constr = 2 * n - 1
    
    # Build constraint matrix
    # A_ub[k, j] = a[k-j] if 0 <= k-j < n
    A = np.zeros((n_constr, n))
    for k in range(n_constr):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            A[k, j] = a[k - j]
    
    c = -np.ones(n)  # minimize -sum(b) = maximize sum(b)
    b_ub = np.full(n_constr, M)
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    
    if res.success:
        return res.x
    return None
```

```python
def lp_min_peak(a):
    """
    Solve LP: min M' s.t. b >= 0, sum(b)=1, conv(a,b)[k] <= M' for all k.
    Returns (M', b).
    """
    n = a.size
    n_constr = 2 * n - 1
    
    # Variables: [b[0], ..., b[n-1], M']  -> n+1 variables
    # Objective: minimize M' -> c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [1.0]
    
    # Inequalities: conv(a,b)[k] - M' <= 0 for all k
    A_ub = np.zeros((n_constr, n + 1))
    b_ub = np.zeros(n_constr)
    
    for k in range(n_constr):
        for j in range(n):
            i = k - j
            if 0 <= i < n:
                A_ub[k, j] = a[i]
        A_ub[k, -1] = -1.0  # coefficient of M'
    
    bounds = [(0, None)] * n + [(0, None)]  # b >= 0, M' >= 0
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if res.success:
        return res.x[-1], res.x[:n]  # M', b
    return None, None
```

This is better! Now the algorithm is:
1. Compute M = max(conv(a, a))
2. Solve LP to get M' and b
3. If M' < M (significant improvement possible), update: a <- (1-t)*a + t*(b/sum(b))
4. Repeat

The step t can be found by line search.

```python
"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based iterative optimization.
The key idea: at each step, solve an LP to find the best "direction" b that reduces
the peak of the cross-convolution conv(a,b), then take a step toward b.
Combined with subgradient refinement and multiple sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import linprog


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def conv_cross(a, b):
    """Full linear convolution a*b via FFT."""
    na, nb = a.size, b.size
    size = na + nb - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
    return np.fft.irfft(fa * fb, fsize)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    x = np.maximum(v, 0.0).copy()
    s = x.sum()
    if s < 1e-30:
        return np.ones(n) / n
    x /= s
    return x


def lp_min_peak(a):
    """
    Solve LP: min M' s.t. b >= 0, sum(b)=1, conv(a,b)[k] <= M' for all k.
    Returns (M_prime, b) or (None, None) on failure.
    """
    n = a.size
    n_constr = 2 * n - 1

    # Variables: [b[0], ..., b[n-1], M'] -> n+1 variables
    c = np.zeros(n + 1)
    c[-1] = 1.0

    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    # Inequalities: conv(a,b)[k] - M' <= 0 for all k
    A_ub = np.zeros((n_constr, n + 1))
    b_ub = np.zeros(n_constr)

    for k in range(n_constr):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        js = np.arange(j_min, j_max + 1)
        A_ub[k, js] = a[k - js]
        A_ub[k, -1] = -1.0

    bounds = [(0, None)] * n + [(0, None)]

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x[-1], res.x[:n]
    except Exception:
        pass
    return None, None


def lp_optimize(a, time_budget, rng, verbose=True):
    """
    LP-based iterative optimization.
    a is normalized to sum=1. Returns (best_a, best_score).
    """
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = score_normalized(a)

    t0 = time.monotonic()
    stall = 0
    last_improve = 0
    iters = 0
    step_size = 0.5

    if verbose:
        print(f"  [LP] Start: n={n}, score={best_sc:.6f}")

    while time.monotonic() - t0 < time_budget:
        M_cur = float(np.max(full_conv(a)))

        M_prime, b = lp_min_peak(a)

        if M_prime is None:
            # LP failed, use subgradient step as fallback
            b = a.copy()
            M_prime = M_cur

        if M_prime >= M_cur * 0.999:
            # No significant improvement possible from LP
            stall += 1
            if stall > 50:
                # Perturb and restart
                a = project_simplex(best_a + 0.05 * rng.standard_normal(n))
                stall = 0
                step_size = 0.3 + 0.4 * rng.random()
            iters += 1
            continue

        # b is the LP solution (a probability vector)
        g = b / b.sum() if b.sum() > 1e-15 else a.copy()

        # Line search for step size
        best_t = step_size
        best_s = np.inf
        for t in [0.1, 0.3, step_size, 0.5, 0.7, 0.9]:
            a_try = project_simplex((1.0 - t) * a + t * g)
            s_try = score_normalized(a_try)
            if s_try < best_s:
                best_s = s_try
                best_t = t

        a = project_simplex((1.0 - best_t) * a + best_t * g)
        cur_sc = score_normalized(a)

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            last_improve = iters
            stall = 0
        else:
            stall += 1
            # Revert if got worse
            if cur_sc > best_sc * 1.01:
                a = best_a.copy()

        if iters % 200 == 0 and verbose:
            print(f"  [LP] iter={iters}, score={best_sc:.6f}, M_prime/M={M_prime/M_cur:.4f}")

        iters += 1

        # Adaptive step size
        if stall > 0 and stall % 10 == 0:
            step_size = max(0.1, step_size * 0.8)
        elif stall == 0:
            step_size = min(0.7, step_size * 1.05)

        # Reheat if stuck
        if iters - last_improve > 500:
            a = project_simplex(best_a + 0.03 * rng.standard_normal(n))
            stall = 0
            step_size = 0.4 + 0.3 * rng.random()
            last_improve = iters

    if verbose:
        print(f"  [LP] Done: n={n}, best_score={best_sc:.6f}, iters={iters}")

    return best_a, best_sc


def smoothed_subgrad(a, b, tau):
    """Smoothed subgradient of max(b) using softmax weights."""
    n = a.size
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(b.size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= s
    # Gradient of max(conv(a,a)) w.r.t. a is 2*conv(w, a) (valid part)
    g = 2.0 * np.correlate(w, a, mode='valid')
    return g


def subgrad_optimize(a, time_budget, rng, verbose=True):
    """Projected smoothed-subgradient descent with momentum."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = score_normalized(a)

    eta = 0.15
    tau_frac = 0.05
    beta = 0.5
    vel = np.zeros(n)

    t0 = time.monotonic()
    iters = 0
    last_improve = 0
    stall = 0

    if verbose:
        print(f"  [SG] Start: n={n}, score={best_sc:.6f}")

    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))

        tau = tau_frac * float(np.max(b))
        if tau < 1e-18:
            tau = 1e-18

        g = smoothed_subgrad(a, b, tau)
        vel = beta * vel + (1.0 - beta) * g
        a_new = project_simplex(a - eta * vel)

        b_new = full_conv(a_new)
        new_sc = 2.0 * n * float(np.max(b_new))

        if new_sc <= cur_sc or rng.random() < 0.05:
            a = a_new
        else:
            a = project_simplex(a - 0.3 * eta * vel)

        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = iters
            stall = 0

        iters += 1

        if stall == 0:
            stall = iters - last_improve
        if stall > 1000:
            a = project_simplex(best_a + 0.03 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.05 + 0.15 * rng.random()
            tau_frac = 0.03 + 0.05 * rng.random()
            last_improve = iters
            stall = 0

    if verbose:
        print(f"  [SG] Done: n={n}, best_score={best_sc:.6f}, iters={iters}")

    return best_a, best_sc


def polish(a, time_budget, rng):
    """Local polish via random perturbations with decreasing scale."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = score_normalized(a)

    t0 = time.monotonic()
    scale = 0.01
    stuck = 0

    while time.monotonic() - t0 < time_budget and scale > 1e-7:
        improved = False
        n_tries = max(5, int(2.0 / scale))
        for _ in range(n_tries):
            if time.monotonic() - t0 > time_budget:
                break
            cand = a + scale * rng.standard_normal(n)
            cand = project_simplex(cand)
            sc = score_normalized(cand)
            if sc < best_sc:
                best_sc = sc
                best_a = cand.copy()
                a = cand.copy()
                improved = True
        if not improved:
            stuck += 1
            if stuck > 8:
                break
        else:
            stuck = 0
        scale *= 0.85

    return best_a, best_sc


def init_uniform(n):

def init_two_cluster(n, rng):
    i = np.arange(n, dtype=np.float64)

def init_u_shape(n):

def init_sided(n):
    i = np.arange(n, dtype=np.float64)

def init_random(n, rng):

def init_sine_power(n, rng):
    """Sine wave raised to a power."""
    i = np.arange(n, dtype=np.float64)
    freq = 1.0 + int(rng.integers(0, 3))
    phase = 2.0 * np.pi * rng.random()
    alpha = 0.5 + rng.random()
    a = np.abs(np.sin(2.0 * np.pi * freq * i / n + phase)) ** alpha
    a += 0.01
    return a / a.sum()


def init_linear_slope(n, rng):
    """Linearly increasing (or decreasing)."""
    i = np.arange(n, dtype=np.float64)
    if rng.random() < 0.5:
        a = i / n + 0.1
    else:
        a = (n - 1 - i) / n + 0.1
    a += 0.05 * np.abs(rng.standard_normal(n))
    return a / a.sum()

                best_overall_sc = sc_lp2
                best_overall = a_lp2.copy()
                print(f"[phase3] LP improved to {sc_lp2:.6f}")

    # Phase 4: Local polish
    if best_overall is not None and time.monotonic() < deadline - 10:
        remain = deadline - time.monotonic() - 5.0
        if remain > 5.0:
            pol_time = min(remain * 0.5, 60.0)
            print(f"[phase4] Polish: n={best_overall.size}, time={pol_time:.1f}s")
            a_pol, sc_pol = polish(best_overall, pol_time, rng)
            if sc_pol < best_overall_sc:
                best_overall_sc = sc_pol
                best_overall = a_pol.copy()
                print(f"[phase4] Polish improved to {sc_pol:.6f}")

    # Phase 5: Try a few more LP runs with different inits on the best n
    if best_overall is not None and time.monotonic() < deadline - 15:
        remain = deadline - time.monotonic() - 5.0
        if remain > 20.0:
            nv = best_overall.size
            print(f"[phase5] Extra LP runs: n={nv}, time={remain:.1f}s")
            for attempt in range(5):
                if time.monotonic() > deadline - 5:
                    break
                a_pert = project_simplex(best_overall + 0.1 * rng.standard_normal(nv))
                tb = min(remain * 0.15, 60.0)
                a_try, sc_try = lp_optimize(a_pert, tb, rng, verbose=False)
                if sc_try < best_overall_sc:
                    best_overall_sc = sc_try
                    best_overall = a_try.copy()
                    print(f"  [phase5] attempt {attempt}: NEW BEST {sc_try:.6f}")
                else:
                    print(f"  [phase5] attempt {attempt}: {sc_try:.6f}")

    # Fallback
    if best_overall is None:
        best_overall = np.ones(128) / 128
        best_overall_sc = 2.0

    # Final verification
    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"[final] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")

    return result
```
