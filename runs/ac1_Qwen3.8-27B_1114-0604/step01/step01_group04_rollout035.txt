"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using:
1. Structured initializations (Gaussian, two-Gaussian, Hann, random-sum)

import numpy as np

def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = a.size
    s = a.sum()
    if s < 1e-15:
        return np.inf
    b = full_conv(a)
    return 2.0 * n * float(b.max()) / (s * s)


def init_gaussian(n, rng):
    i = np.arange(n, dtype=np.float64)
    c = n * 0.5
    sigma = n * (0.12 + 0.12 * rng.random())
    a = np.exp(-((i - c) / sigma) ** 2)
    a += 1e-10
    return a / a.sum()


def init_two_gaussian(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.15 + 0.2 * rng.random())
    c2 = n * (0.55 + 0.25 * rng.random())
    sigma = n * (0.06 + 0.14 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 1e-10
    return a / a.sum()


def init_hann(n):

    a = 0.5 * (1.0 - np.cos(2.0 * np.pi * i / max(n - 1, 1)))

    return a / a.sum()

    a = np.where(x < half_width, 1.0, np.exp(-(x - half_width) ** 2 / (2 * edge_width ** 2)))

def coord_descent(a, time_budget, rng):

    Fast coordinate descent minimizing 2n*max(conv(a,a)) with sum(a)=1.
    Uses incremental convolution updates: changing a[i] by delta only
    affects conv[m] for m in [i, i+n-1].
    """
    n = a.size
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-15:
        a = np.ones(n) / n
        return a, 2.0
    a /= a.sum()

    b = full_conv(a)
    best_a = a.copy()
    best_sc = 2.0 * n * float(b.max())

    t0 = time.monotonic()
    pass_count = 0
    stall = 0

    while time.monotonic() - t0 < time_budget:
        pass_count += 1
        order = rng.permutation(n)
        n_improved = 0

        for i in order:
            ai = a[i]
            # Candidate values for a[i]
            cands = [0.0]
            if ai > 0:
                cands.extend([ai * 0.2, ai * 0.5, ai * 0.7, ai * 1.3, ai * 1.7, ai * 2.5])
            cands.extend([1e-4, 5e-4, 2e-3, 1e-2, 5e-2, 0.1])
            # Deduplicate and clip
            cands = sorted(set(max(0.0, min(1.0, round(c, 10))) for c in cands))

            best_t = ai
            best_sc_i = 2.0 * n * float(b.max())  # current (sum=1)

            m_start = i
            m_end = min(i + n, 2 * n - 1)
            mi2 = 2 * i

            for t in cands:
                if abs(t - ai) < 1e-14:
                    continue
                delta = t - ai
                # Save and update b
                old_seg = b[m_start:m_end].copy()
                b[m_start:m_end] += 2.0 * delta * a[m_start:m_end - i]
                if mi2 < 2 * n - 1:
                    b[mi2] += delta * delta

                sum_new = 1.0 + delta
                if sum_new > 1e-15:
                    sc_i = 2.0 * n * float(b.max()) / (sum_new * sum_new)
                else:
                    sc_i = np.inf

                if sc_i < best_sc_i - 1e-12:
                    best_sc_i = sc_i
                    best_t = t
                else:
                    b[m_start:m_end] = old_seg

            if best_t != ai:
                delta = best_t - ai
                b[m_start:m_end] += 2.0 * delta * a[m_start:m_end - i]
                if mi2 < 2 * n - 1:
                    b[mi2] += delta * delta
                a[i] = best_t
                n_improved += 1

        # Renormalize after full pass
        s = a.sum()
        if s > 1e-15:
            a /= s
            b = full_conv(a)
            cur_sc = 2.0 * n * float(b.max())
            if cur_sc < best_sc - 1e-10:
                best_sc = cur_sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        if stall > 15:
            # Perturb from best
            a = best_a + 0.02 * rng.standard_normal(n)
            a = np.maximum(a, 0.0)
            if a.sum() > 1e-15:
                a /= a.sum()
            b = full_conv(a)
            stall = 0

    return best_a, best_sc


def smooth_max_obj_and_grad(x, n, tau):
    """
    Objective and gradient for L-BFGS.
    x: log-parameter, a = softmax(x)
    Returns (f, grad_x)
    """
    # Softmax
    x_shift = x - x.max()
    e = np.exp(x_shift)
    z = e.sum()
    a = e / z  # sum(a) = 1

    b = full_conv(a)

    # Soft max
    bm = b / tau
    bm -= bm.max()
    w = np.exp(bm)
    w /= w.sum()
    sm = tau * (bm.max() + np.log(w.sum()))
    # Actually: soft_max = tau * log(sum(exp(b/tau)))
    # = tau * (bm.max() + log(sum(exp(bm))))
    # But bm already had max subtracted, so:
    sm = tau * (np.log(np.sum(np.exp(bm))))
    # Wait, sm = tau * log(sum(exp(b/tau))) = tau * log(sum(exp(bm + b.max()/tau)))
    # = tau * (b.max()/tau + log(sum(exp(bm))))
    # = b.max() + tau * log(sum(exp(bm)))
    # Let me just compute directly:
    raw = b / tau
    raw_max = raw.max()
    sm = tau * (raw_max + np.log(np.sum(np.exp(raw - raw_max))))

    f = 2.0 * n * sm

    # Gradient
    # grad_a = 2 * np.correlate(w, a, 'valid')
    # where w is the softmax weights over b
    # Recompute w properly
    w = np.exp(raw - raw_max)
    w /= w.sum()

    # grad_a[k] = 2 * sum_{j=0}^{n-1} w[k+j] * a[j]
    # = 2 * np.correlate(w, a, 'valid')[k]
    # np.correlate(w, a, 'valid')[k] = sum_j w[k+j] * a[j]  (length = len(w)-len(a)+1 = n)
    grad_a = 2.0 * np.correlate(w, a, mode='valid')

    # grad_x = 2n * a * (grad_a - dot(a, grad_a))
    dot = np.dot(a, grad_a)
    grad_x = 2.0 * n * a * (grad_a - dot)
    grad_x -= grad_x.mean()  # recenter (invariance to adding constant to x)

    return f, grad_x


def polish_lbfgs(a, time_budget, rng):
    """Polish using L-BFGS with smooth-max approximation."""
    n = a.size
    a = a.copy().astype(np.float64)
    a = np.maximum(a, 0.0)
    a /= a.sum()

    best_a = a.copy()
    best_sc = 2.0 * n * float(full_conv(a).max())

    t0 = time.monotonic()
    x = np.log(a + 1e-30)

    tau_list = [0.02, 0.01, 0.005, 0.002]

    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        max_b = float(b.max())
        for tau_frac in [0.03, 0.01, 0.004]:
            tau = tau_frac * max_b
            if tau < 1e-18:
                tau = 1e-18

            def obj(x_inner):
                f, _ = smooth_max_obj_and_grad(x_inner, n, tau)
                return f

            def grad(x_inner):
                _, g = smooth_max_obj_and_grad(x_inner, n, tau)
                return g

            try:
                res = minimize(obj, x, jac=grad, method='L-BFGS-B',
                               options={'maxiter': 200, 'gtol': 1e-8,
                                        'maxfun': 800, 'disp': False})
                x_new = res.x
                x_s = x_new - x_new.max()
                a_new = np.exp(x_s)
                a_new /= a_new.sum()
                a_new = np.maximum(a_new, 0.0)
                a_new /= a_new.sum()
                sc_new = 2.0 * n * float(full_conv(a_new).max())

                    best_sc = sc_new
                    best_a = a_new.copy()
                    x = x_new.copy()
                    a = a_new.copy()
            except Exception:
                pass

            if time.monotonic() - t0 > time_budget:
                break

    return best_a, best_sc

    print(f"[plan] n_list={n_list}, total_time={total_time:.1f}s")

    # Allocate time: give more time to smaller n (faster iterations) and to prev n
    raw_w = []
    for nv in n_list:
        w = 1.0 / (nv / 128.0) ** 0.5
        if prev_a is not None and nv == prev_a.size:
            w *= 2.0
        raw_w.append(w)
    total_w = sum(raw_w)
    weights = [w / total_w for w in raw_w]

    for idx_n, nv in enumerate(n_list):
        if time.monotonic() > deadline:
            break

        time_n = weights[idx_n] * total_time * 0.80

        # Determine number of inits to try
        inits = get_inits(nv, rng)
        if prev_a is not None and prev_a.size == nv:
            inits.insert(0, prev_a.copy())

        # Also add perturbed versions of previous best
        if prev_a is not None and prev_a.size == nv:
            for _ in range(3):
                pert = prev_a + 0.05 * rng.standard_normal(nv)
                pert = np.maximum(pert, 0.0)
                if pert.sum() > 0:
                    pert /= pert.sum()
                    inits.append(pert)

        n_inits = len(inits)
        time_per_init = time_n / n_inits

        for ii, a_init in enumerate(inits):
            if time.monotonic() > deadline:
                break

            tb = min(time_per_init, deadline - time.monotonic() - 5.0)
            if tb < 5.0:
                break

            # Phase 1: Coordinate descent
            a_cd, sc_cd = coord_descent(a_init, tb * 0.6, rng)

            if sc_cd < best_overall_sc:
                best_overall_sc = sc_cd
                best_overall = a_cd.copy()
                print(f"[cd] n={nv}, init={ii}, score={sc_cd:.6f} (BEST)")
            else:
                print(f"[cd] n={nv}, init={ii}, score={sc_cd:.6f}")

            # Phase 2: L-BFGS polish (if time remains)
            remain_init = min(time_per_init * 0.3, deadline - time.monotonic() - 5.0)
            if remain_init > 3.0:
                a_lp, sc_lp = polish_lbfgs(a_cd, remain_init, rng)
                if sc_lp < best_overall_sc:
                    best_overall_sc = sc_lp
                    best_overall = a_lp.copy()
                    print(f"[lbfgs] n={nv}, init={ii}, score={sc_lp:.6f} (BEST)")

        print(f"[n={nv}] done. Overall best: {best_overall_sc:.6f}")

    # Final polish phase on the best solution
    if best_overall is not None:
        remain = deadline - time.monotonic() - 3.0
        if remain > 10.0:
            n_best = best_overall.size
            print(f"[final polish] n={n_best}, remaining={remain:.1f}s")

            # Coordinate descent polish
            cd_time = min(remain * 0.3, 120.0)
            a_p1, sc_p1 = coord_descent(best_overall, cd_time, rng)
            if sc_p1 < best_overall_sc:
                best_overall_sc = sc_p1
                best_overall = a_p1.copy()
                print(f"[final cd] score={sc_p1:.6f}")

            # L-BFGS polish
            remain2 = deadline - time.monotonic() - 3.0
            if remain2 > 10.0:
                lbfgs_time = min(remain2 * 0.4, 120.0)
                a_p2, sc_p2 = polish_lbfgs(best_overall, lbfgs_time, rng)
                if sc_p2 < best_overall_sc:
                    best_overall_sc = sc_p2
                    best_overall = a_p2.copy()
                    print(f"[final lbfgs] score={sc_p2:.6f}")

            # Block CD polish
            remain3 = deadline - time.monotonic() - 3.0
            if remain3 > 10.0:
                block_time = min(remain3 * 0.3, 60.0)
                a_p3, sc_p3 = block_cd_polish(best_overall, block_time, rng)
                if sc_p3 < best_overall_sc:
                    best_overall_sc = sc_p3
                    best_overall = a_p3.copy()
                    print(f"[final block] score={sc_p3:.6f}")

    # Fallback
    if best_overall is None:
        best_overall = np.ones(128) / 128
        best_overall_sc = 2.0

    # Final: try a few more coordinate descent passes on the very best
    remain4 = deadline - time.monotonic() - 2.0
    if remain4 > 5.0 and best_overall is not None:
        a_final, sc_final = coord_descent(best_overall, remain4, rng)
        if sc_final < best_overall_sc:
            best_overall_sc = sc_final
            best_overall = a_final.copy()

    # Ensure valid output
    result_a = np.maximum(best_overall, 0.0)
    result_a[result_a > 1000.0] = 1000.0
    result = [float(x) for x in result_a]

    final_sc = float(evaluate_sequence(result))
    print(f"[final] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")

    return result
