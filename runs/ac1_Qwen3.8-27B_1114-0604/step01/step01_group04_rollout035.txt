import time

    """Smallest power of 2 >= n."""
    if n <= 0:

    return 1 << (n - 1).bit_length()

    """Full linear convolution a*a via FFT. Returns length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = rfft_size(size)
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def conv_valid(w, a):
    """Compute np.convolve(w, a, 'valid') via FFT.
    w: length 2n-1, a: length n. Returns length n."""
    n = a.size
    full_len = 3 * n - 2
    fsize = rfft_size(full_len)
    fw = np.fft.rfft(w, fsize)
    fa = np.fft.rfft(a, fsize)
    full = np.fft.irfft(fw * fa, fsize)
    return full[n - 1:n - 1 + n]


def project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
