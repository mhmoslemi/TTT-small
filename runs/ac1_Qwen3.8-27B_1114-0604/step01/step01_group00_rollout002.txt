def _conv_fft(f):
    """Compute full linear convolution f*f via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _soft_grad(conv_val, f, T):
    """Compute softmax-smoothed gradient of max(conv)."""
    n = len(f)
    cs = conv_val / T
    cs -= cs.max()
    p = np.exp(cs)
    s = p.sum()
    if s < 1e-300:
        # Degenerate: point mass at argmax
        p = np.zeros(2 * n - 1)
        p[np.argmax(conv_val)] = 1.0
    else:
        p /= s
    # grad[j] = 2 * sum_m p[j+m] * f[m]
    grad = 2.0 * correlate(p, f, mode='valid')
    return grad, p

def _adam_update(log_f, grad, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    """Adam update on log-domain variables."""
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * grad ** 2
    m_hat = m / (1 - b1 ** t)
    v_hat = v / (1 - b2 ** t)
    log_f_new = log_f - lr * m_hat / (np.sqrt(v_hat) + eps)
    return log_f_new, m, v


def _normalize_simplex(f):
    """Project to simplex {x >= 0, sum x = 1}."""
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-30:
        return np.ones(len(f)) / len(f)
    return f / s

    log_f = np.log(f + 1e-300)

def _random_start(n, rng, style):
    """Generate a random starting sequence of length n."""
    j = np.arange(n, dtype=np.float64)
    
    if style == 'uniform':
        return np.ones(n)
    elif style == 'exponential':
        rate = rng.uniform(0.5, 3.0)
        f = np.exp(-rate * j / n)
        if rng.random() < 0.5:
            f = f[::-1]
        return f
    elif style == 'bimodal':
        c1 = rng.uniform(0.1, 0.4)
        c2 = rng.uniform(0.6, 0.9)
        w = rng.uniform(0.01, 0.1)
        f = np.exp(-((j/n - c1)**2) / (2*w**2)) + np.exp(-((j/n - c2)**2) / (2*w**2))
        return f
    elif style == 'ramp':
        f = j.astype(float) + 1
        if rng.random() < 0.5:
            f = f[::-1]
        return f
    elif style == 'sine':
        freq = rng.uniform(1, 5)
        phase = rng.uniform(0, 2*np.pi)
        f = 1.0 + 0.7 * np.sin(2*np.pi*freq*j/n + phase)
        return np.maximum(f, 0.01)
    elif style == 'dirichlet':

    else:
        return rng.random(n) + 0.01
