def _conv_fft(f):
    """Full linear convolution f*f via FFT. Returns array of length 2n-1."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]

def _score_fast(f):
    """Fast score computation without calling evaluate_sequence."""
    n = len(f)
    c = _conv_fft(f)
    max_c = c.max()

    return 2.0 * n * max_c / (s * s)

def _soft_grad(c, f, T):
    """Compute soft-max smoothed gradient of max(conv)."""
    n = len(f)
    cs = c / T
    cs -= cs.max()
    p = np.exp(cs)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros(2 * n - 1)
        p[n - 1] = 1.0
    else:
        p /= s

def _adam_step(f, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam update. Modifies m, v in place. Returns updated f."""
    m[:] = beta1 * m + (1 - beta1) * grad
    v[:] = beta2 * v + (1 - beta2) * (grad * grad)
    m_hat = m / (1.0 - beta1 ** t)
    v_hat = v / (1.0 - beta2 ** t)
    f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
    return f_new
