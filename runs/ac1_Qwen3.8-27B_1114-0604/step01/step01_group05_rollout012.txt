    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    b = np.ascontiguousarray(b, dtype=np.float64)
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    """Smooth max of b and softmax weights."""
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float("inf"), np.empty(0, dtype=np.float64)
    m = float(np.max(b))
    if not np.isfinite(m):
        return float("inf"), np.zeros_like(b, dtype=np.float64)
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b, dtype=np.float64)
        w /= float(b.size)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w


def fast_score(p):
    """Fast objective: 2n * max(conv(p,p)) / sum(p)^2."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float("inf")
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float("inf")
    b = fft_convolution(p)
    if b.size == 0:
        return float("inf")
    M = float(np.max(b))
    if not np.isfinite(M):
        return float("inf")
    return float(2.0 * n * M / (S * S))


def corr_pw(p, w):
    """s_i = sum_j p_j * w_{i+j} for i=0..n-1."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    if n == 1:
        return np.array([float(np.dot(p, w[:1]))], dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n: m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    if target <= 0.0:
        return np.zeros_like(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(v.shape, target / float(v.size), dtype=np.float64)
    low = float(np.min(v)) - 1.0
    high = float(np.max(v)) + 1.0
    for _ in range(50):
        mid = 0.5 * (low + high)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        q = np.full(v.shape, target / float(v.size), dtype=np.float64)
    else:
        q *= target / s
    return q


def normalize_simplex(p):
    """Clip to non-negative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size), dtype=np.float64)
    p *= 1.0 / s
    return p


def gradient_of_smooth_max(p, beta):
    """Compute gradient of n*smooth_max(conv(p,p)) w.r.t. p on simplex."""
    n = p.size
    b = fft_convolution(p)
    G, w = smooth_max_and_weights(b, beta)
    if not np.isfinite(G):
        return None, float("inf"), np.zeros(n)
    svec = corr_pw(p, w)
    svec = np.maximum(svec, 0.0)
    g = 2.0 * svec
    mean_g = float(np.dot(g, p))
    grad = g - mean_g
    return grad, float(n * G), svec


def adam_logspace(p0, deadline, beta_sm=50000.0, lr=0.5, max_iter=20000, rng=None):
    """Adam optimizer in log-space (softmax parameterization)."""
    n = p0.size
    if n == 0:
        return p0, fast_score(p0)
    if rng is None:
        rng = np.random.default_rng(0)

    p = normalize_simplex(p0.copy())
    # Add tiny floor to avoid log(0)
    floor = 1e-20
    p_safe = np.maximum(p, floor)
    p_safe = p_safe / p_safe.sum()
    z = np.log(p_safe)

    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    m_buf = np.zeros(n)
    v_buf = np.zeros(n)
    best_p = p.copy()
    best_score = fast_score(p)
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        # Compute gradient in log-space
        p_curr = np.exp(z - np.max(z))
        p_curr /= p_curr.sum()
        p_curr = np.maximum(p_curr, 1e-30)

        grad_p, Gval, svec = gradient_of_smooth_max(p_curr, beta_sm)
        if grad_p is None:
            break

        # Convert to log-space gradient: dF/dz_i = p_i * (grad_p_i - sum(grad_p * p))
        # Actually grad_p is already (g - mean) so sum(grad_p * p) ~ 0
        # dF/dz_i = p_i * grad_p_i (approximately, since grad_p is already centered)
        grad_z = p_curr * grad_p

        # Adam update
        m_buf = beta1 * m_buf + (1 - beta1) * grad_z
        v_buf = beta2 * v_buf + (1 - beta2) * grad_z ** 2
        m_hat = m_buf / (1 - beta1 ** (it + 1))
        v_hat = v_buf / (1 - beta2 ** (it + 1))
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)

        # Convert back
        p_new = np.exp(z - np.max(z))
        p_new /= p_new.sum()
        p_new = np.maximum(p_new, 0.0)
        s = p_new.sum()
        if s > 0 and np.isfinite(s):
            p_new /= s
        else:
            p_new = p_curr
            break

        sc = fast_score(p_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = p_new.copy()
            no_improve = 0
            if it % 200 == 0:
                print(f"  adam it={it} score={best_score:.8f} lr={lr:.4f}")
        else:
            no_improve += 1

        # Decay learning rate slightly
        if no_improve > 500:
            lr *= 0.9
            no_improve = 0

        if no_improve > 5000:
            break

    return best_p, best_score


def projected_gradient_momentum(p0, deadline, beta_sm=50000.0, max_iter=20000, rng=None):
    """Projected gradient with Nesterov momentum on the simplex."""
    n = p0.size
    if n == 0:
        return p0, fast_score(p0)

    p = normalize_simplex(p0.copy())
    # Break symmetry
    if n > 1 and float(np.max(np.abs(p - 1.0/n))) < 1e-14:
        if rng is not None:
            pert = rng.normal(0, 1e-3, n)
        else:
            pert = 1e-3 * np.sin(np.linspace(0, 6.28, n))
        p = normalize_simplex(p + pert)

    v = np.zeros(n)
    mu = 0.9
    best_p = p.copy()
    best_score = fast_score(p)
    step = 1.0
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        # Nesterov lookahead
        y = p + mu * v
        y = np.maximum(y, 0.0)
        sy = y.sum()
        if sy > 0:
            y /= sy
        else:
            y = p

        grad_p, Gval, svec = gradient_of_smooth_max(y, beta_sm)
        if grad_p is None:
            break

        gnorm = np.linalg.norm(grad_p)
        if not np.isfinite(gnorm) or gnorm < 1e-15:
            break

        # Adaptive step
        alpha = step / (gnorm + 1e-10)

        v_new = mu * v + alpha * grad_p
        p_new = p - alpha * (mu * v + (1 + mu) * grad_p)
        p_new = project_simplex(p_new, 1.0)
        v = v_new

        sc = fast_score(p_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = p_new.copy()
            no_improve = 0
            if it % 500 == 0:
                print(f"  pgm it={it} score={best_score:.8f}")
        else:
            no_improve += 1

        if no_improve > 3000:
            break

        if it % 1000 == 0:
            p = p_new  # sync

        p = p_new

    return best_p, best_score


def lp_best_response(f, M, n_lp=None, max_time=30.0):
    """
    Solve LP: max sum(g) s.t. g >= 0, (f*g)[k] <= M for all k.
    Returns the optimal g (normalized to sum 1) or None if it fails.
    """
    n = f.size
    if n_lp is not None and n_lp < n:
        # Downsample f for the LP
        ratio = n // n_lp
        f_small = f[:n_lp]
    else:
        n_lp = n
        f_small = f

    nlp = n_lp
    n_cons = 2 * nlp - 1

    # Build constraint matrix A: A[k, j] = f_small[k-j] if 0 <= k-j < nlp
    # Use sparse matrix
    t_start = time.monotonic()
    A = lil_matrix((n_cons, nlp), dtype=np.float64)
    for k in range(n_cons):
        j_lo = max(0, k - nlp + 1)
        j_hi = min(k, nlp - 1)
        for j in range(j_lo, j_hi + 1):
            A[k, j] = f_small[k - j]

    # Convert to CSR for efficiency
    A_csr = A.tocsr()

    # LP: max 1^T g s.t. A g <= M, g >= 0
    # scipy linprog: min -1^T g s.t. A g <= M, g >= 0
    c_obj = -np.ones(nlp)
    b_ub = np.full(n_cons, M)

    t_lp = time.monotonic()
    try:
        res = linprog(
            c_obj,
            A_ub=A_csr,
            b_ub=b_ub,
            method='highs',
            options={'time_limit': max_time, 'presolve': True}
        )
    except Exception as e:
        print(f"  LP failed: {e}")
        return None

    if time.monotonic() - t_lp > max_time * 1.5:
        return None

    if res.status != 0 or res.x is None:
        if res.status != 0:
            print(f"  LP status={res.status}")
        return None

    g = np.maximum(res.x, 0.0)
    sg = g.sum()
    if sg <= 0 or not np.isfinite(sg):
        return None
    g /= sg

    # Upsample if needed
    if nlp < n:
        g_full = np.zeros(n)
        # Simple upsampling by interpolation
        idx = np.linspace(0, nlp - 1, n)
        g_full = np.interp(idx, np.arange(nlp), g)
        sg2 = g_full.sum()
        if sg2 > 0:
            g_full /= sg2
        return g_full

    return g


def lp_iteration(f, deadline, n_lp=1024, blend_t=0.3, num_iters=5, max_lp_time=20.0):
    """Apply several LP best-response iterations."""
    p = normalize_simplex(f.copy())
    n = p.size
    best_p = p.copy()
    best_score = fast_score(p)

    for it in range(num_iters):
        if time.monotonic() >= deadline:
            break

        b = fft_convolution(p)
        M = float(np.max(b))
        if not np.isfinite(M) or M <= 0:
            break

        g = lp_best_response(p, M, n_lp=n_lp, max_time=max_lp_time)
        if g is None:
            break

        # Blend
        p_new = (1 - blend_t) * p + blend_t * g
        p_new = normalize_simplex(p_new)

        sc = fast_score(p_new)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = p_new.copy()
            p = p_new
            print(f"  LP iter {it}: score={best_score:.8f}")
        else:
            # Try smaller blend
            p_new2 = (1 - blend_t * 0.3) * p + blend_t * 0.3 * g
            p_new2 = normalize_simplex(p_new2)
            sc2 = fast_score(p_new2)
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_p = p_new2.copy()
                p = p_new2
            else:
                break

    return best_p, best_score


def structured_starts(n, rng, count=12):
    """Generate diverse starting profiles."""
    starts = []

    # Uniform
    starts.append(np.full(n, 1.0 / n))

    # Exponential ramps
    x = np.arange(n, dtype=np.float64) / n
    for a in (2.0, 4.0, 6.0, -2.0, -4.0, -6.0):
        p = np.exp(a * (x - 0.5))
        p /= p.sum()
        starts.append(p)

    # Quadratic
    for a in (3.0, -3.0):
        p = np.exp(a * (x - 0.5) ** 2 * 4)
        p /= p.sum()
        starts.append(p)

    # Cosine
    p = np.abs(np.cos(np.pi * x)) + 0.1
    p /= p.sum()
    starts.append(p)

    # Two-bump
    p = np.exp(-20 * (x - 0.3) ** 2) + np.exp(-20 * (x - 0.7) ** 2) + 0.01
    p /= p.sum()
    starts.append(p)

    # Random smooth
    for _ in range(3):
        k = n // 8 + 1
        z = np.fft.irfft(np.fft.rfft(rng.normal(size=n))[:k] * np.exp(-0.1 * x * n / n))
        z = np.tanh(z)
        p = np.exp(z - z.max())
        p /= p.sum()
        starts.append(p)

    return starts[:count]


def rand_perturb(p, rng, sigma=0.01):
    """Random perturbation projected to simplex."""
    n = p.size
    q = p + sigma * rng.normal(size=n)
    return project_simplex(q, 1.0)


def official_score(seq):
    """Call evaluate_sequence safely."""
    try:
        val = evaluate_sequence(seq)
        val = float(val)
    except Exception:
        return float("inf")
    if np.isfinite(val):
        return val
    return float("inf")


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase search:
    1. Load parent's best as initial.
    2. Try multiple lengths (4096, 8192).
    3. For each: ADAM log-space + projected gradient momentum + LP best-response.
    4. Return best found.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 0:
        return [1.0]

    reserve = 60.0
    start_time = time.monotonic()
    deadline = start_time + max(0.0, budget_s - reserve)

    rng = np.random.default_rng(seed)

    # Get parent's sequence as initial best
    parent_seq = None
    if _global_seq is not None and len(_global_seq) > 0:
        try:
            parent_arr = np.array([float(x) for x in _global_seq], dtype=np.float64)
            parent_arr = np.maximum(parent_arr, 0.0)
            s = parent_arr.sum()
            if s > 0 and np.isfinite(s):
                parent_arr /= s
                parent_seq = parent_arr
        except Exception:
            parent_seq = None

    best_p = None
    best_score = float("inf")
    best_len = 0

    # Determine which lengths to try
    if budget_s >= 2000:
        lengths = [8192, 4096]
    elif budget_s >= 1000:
        lengths = [4096, 8192]
    else:
        lengths = [2048]

    for n in lengths:
        if time.monotonic() >= deadline:
            break

        print(f"\n=== Starting search with n={n} ===")

        if parent_seq is not None and parent_seq.size == n:
            init_p = parent_seq.copy()
        elif parent_seq is not None:
            # Resample parent to length n
            idx = np.linspace(0, parent_seq.size - 1, n)
            init_p = np.interp(idx, np.arange(parent_seq.size), parent_seq)
            init_p = normalize_simplex(init_p)
        else:
            init_p = np.full(n, 1.0 / n)

        # Evaluate initial
        sc = fast_score(init_p)
        print(f"  initial score={sc:.8f}")
        if sc < best_score:
            best_score = sc
            best_p = init_p.copy()
            best_len = n

        # Phase A: Adam log-space optimization
        if time.monotonic() < deadline - 120:
            print("  Phase A: Adam log-space")
            sub_dl = min(deadline - 60, time.monotonic() + (deadline - start_time) * 0.25)
            p_a, sc_a = adam_logspace(init_p, sub_dl, beta_sm=30000.0, lr=1.0,
                                       max_iter=30000, rng=rng)
            if np.isfinite(sc_a) and sc_a < best_score:
                best_score = sc_a
                best_p = p_a.copy()
                best_len = n
                print(f"  Adam result: {best_score:.8f}")

        # Phase B: Projected gradient with momentum from best so far
        if time.monotonic() < deadline - 90:
            print("  Phase B: Projected gradient momentum")
            p_start = best_p.copy() if best_len == n else init_p.copy()
            sub_dl = min(deadline - 30, time.monotonic() + (deadline - start_time) * 0.2)
            p_b, sc_b = projected_gradient_momentum(p_start, sub_dl, beta_sm=50000.0,
                                                     max_iter=30000, rng=rng)
            if np.isfinite(sc_b) and sc_b < best_score:
                best_score = sc_b
                best_p = p_b.copy()
                best_len = n
                print(f"  PGM result: {best_score:.8f}")

        # Phase C: LP best-response iterations
        if time.monotonic() < deadline - 60:
            print("  Phase C: LP best-response")
            p_start = best_p.copy() if best_len == n else init_p.copy()
            n_lp = min(n, 1024)
            sub_dl = min(deadline - 30, time.monotonic() + 180.0)
            p_c, sc_c = lp_iteration(p_start, sub_dl, n_lp=n_lp, blend_t=0.4,
                                      num_iters=8, max_lp_time=15.0)
            if np.isfinite(sc_c) and sc_c < best_score:
                best_score = sc_c
                best_p = p_c.copy()
                best_len = n
                print(f"  LP result: {best_score:.8f}")

            # Phase C2: Gradient refine after LP
            if time.monotonic() < deadline - 30:
                p_c2, sc_c2 = projected_gradient_momentum(p_c, deadline - 10,
                                                           beta_sm=80000.0,
                                                           max_iter=10000, rng=rng)
                if np.isfinite(sc_c2) and sc_c2 < best_score:
                    best_score = sc_c2
                    best_p = p_c2.copy()
                    best_len = n
                    print(f"  Post-LP PGM: {best_score:.8f}")

        # Phase D: More diverse restarts + Adam
        if time.monotonic() < deadline - 60:
            print("  Phase D: Diverse restarts")
            starts = structured_starts(n, rng, count=6)
            # Add perturbed versions of current best
            if best_len == n:
                for _s in range(3):
                    starts.append(rand_perturb(best_p, rng, sigma=0.05))

            # Score all starts and pick best 3
            scored = []
            for pp in starts:
                sct = fast_score(pp)
                if np.isfinite(sct):
                    scored.append((sct, pp))
            scored.sort(key=lambda x: x[0])

            for i in range(min(3, len(scored))):
                if time.monotonic() >= deadline - 30:
                    break
                p_try = scored[i][1]
                # Brief Adam
                sub_dl2 = min(deadline - 10, time.monotonic() + 60.0)
                p_d, sc_d = adam_logspace(p_try, sub_dl2, beta_sm=50000.0, lr=0.8,
                                           max_iter=5000, rng=rng)
                if np.isfinite(sc_d) and sc_d < best_score:
                    best_score = sc_d
                    best_p = p_d.copy()
                    best_len = n
                    print(f"  Restart {i}: {best_score:.8f}")

    # Phase E: Final fine-grained refinement on the overall best
    if time.monotonic() < deadline - 10 and best_p is not None:
        print("\n  Phase E: Final refinement")
        n_final = best_p.size
        sub_dl = deadline - 5
        p_f, sc_f = projected_gradient_momentum(best_p, sub_dl, beta_sm=100000.0,
                                                 max_iter=20000, rng=rng)
        if np.isfinite(sc_f) and sc_f < best_score:
            best_score = sc_f
            best_p = p_f.copy()
            print(f"  Final PGM: {best_score:.8f}")

        # One more Adam pass
        if time.monotonic() < deadline - 5:
            p_f2, sc_f2 = adam_logspace(best_p, deadline - 2, beta_sm=80000.0, lr=0.5,
                                        max_iter=5000, rng=rng)
            if np.isfinite(sc_f2) and sc_f2 < best_score:
                best_score = sc_f2
                best_p = p_f2.copy()
                print(f"  Final Adam: {best_score:.8f}")

    # Prepare final answer
    if best_p is None:
        # Fallback
        n_fb = 8192
        best_p = np.full(n_fb, 1.0 / n_fb, dtype=np.float64)

    final_p = normalize_simplex(best_p)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)

    final_seq = [float(x) for x in final_p]
    print(f"\n  Final internal score: {fast_score(final_p):.8f}")

    # Official evaluation
    off = official_score(final_seq)
    print(f"  Official score: {off:.8f}")

    if not np.isfinite(off) or off > 2.0:
        # Fallback to uniform
        n_fb = 8192
        return [1.0 / n_fb] * n_fb

    return final_seq
