The issue is that np.convolve is NOT the same as autocorrelation in the DSP sense. Let me clarify:

The autocorrelation of a signal a at lag k is: R[k] = sum_i a[i] * a[i+k]

The convolution of a with itself is: (a*a)[n] = sum_i a[i] * a[n-i]

import numpy as np
import time


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _fft_conv(a, b):
    n = len(a) + len(b) - 1
    if n <= 0:
        return np.array([], dtype=float)
    L = _next_pow2(n)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:n]


def _exact_score(a):
    a = np.asarray(a, dtype=float)
    if a.size == 0:
        return float('inf')
    a = np.clip(a, 0, 1e6)
    s = a.sum()
    if s < 1e-10:
        return float('inf')
    n = len(a)
    c = _fft_conv(a, a)
    c = np.where(np.isfinite(c), c, 0.0)
    return float(2.0 * n * np.max(c) / (s * s))


def _topk_objective_grad(a, k):
    n = a.size
    c = _fft_conv(a, a)
    k = min(k, c.size)
    c_sorted = np.sort(c)[::-1]
    gap = c_sorted[k - 1] - c_sorted[k] if k < c.size else 0.0
    if gap > 1e-14:
        top = np.where(c >= c_sorted[k - 1] - 1e-12)[0]
    else:
        top = np.argsort(c)[::-1][:k]
    w = np.zeros(c.size)
    w[top] = 1.0 / len(top)
    obj = float(np.dot(w, c))
    # Gradient: g[j] = sum_k w[k] * (a[k-j] + a[j-k]) with valid index checks
    maxc = c.size - 1
    ar = a[::-1]
    g = np.zeros(n)
    for shift in range(n):
        i_hi = min(maxc, maxc - shift)
        i_lo = max(0, n - 1 - shift)
        seg = w[i_lo:i_hi + 1]
        if seg.sum() > 0:
            g[shift] = np.dot(seg, a[i_hi - (shift - i_lo): i_hi + 1])
    return obj, g

def _resample(a, m):
    a = np.asarray(a, dtype=float)
    a = np.maximum(a, 0)
    n = a.size
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m, dtype=float)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(m) + 0.5) / m
    out = np.interp(x_new, x_old, a)
    return np.maximum(out, 0)


def _init_cosine(n, rng):

    beta = float(rng.uniform(0.5, 2.0))
    a = np.cos(np.pi * t / (2 * n)) ** (2 * beta)
    return np.maximum(a, 0)


def _init_fejer(n, rng):

    p = float(rng.uniform(1.0, 3.0))
    a = np.power(np.maximum(0, 1 - np.abs(t - center) / (center + 1)), p)

def _init_gaussian(n, rng):

    center = (n - 1) / 2.0
    sigma = float(rng.uniform(n * 0.1, n * 0.4))
    a = np.exp(-0.5 * ((t - center) / sigma) ** 2)
    return a


def _init_fourier(n, rng):
    """Construct a with flat-ish |DFT| using random phases constrained to give non-negative signal."""
    L = 2 * n
    num_freqs = L // 2 + 1
    logged = np.log(np.linspace(0.1, 1.0, num_freqs)) + float(rng.uniform(-0.3, 0.3))
    logged[0] = 0.0
    mag = np.exp(logged)
    mag /= mag.sum()
    phase = np.zeros(num_freqs)
    for k in range(1, num_freqs):
        phase[k] = float(rng.uniform(0, 2 * np.pi))
        phase[L - k] = -phase[k]
    X = mag * np.exp(1j * phase)
    X[0] = abs(X[0])
    if L % 2 == 0:
        X[num_freqs - 1] = abs(X[num_freqs - 1])
    sig = np.fft.irfft(X, L)[:n]
    a = np.maximum(sig, 0)
    if a.sum() < 1e-10:
        a = np.ones(n, dtype=float)
    return a
