"""

Uses multiplicative mirror descent with improved initialization (DFT-based,
multi-bump), adaptive smoothing, gradient-guided local search, and mass-transfer

import numpy as np

import math


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b):
    la = len(a)
    lb = len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 3000000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)


def _conv_valid_front(w, p):
    lenw = len(w)
    lenp = len(p)
    if lenp == 0:
        return np.array([], dtype=float)
    L = _next_pow2(max(lenw, 2 * lenp))
    fw = np.fft.rfft(w, L)
    fp = np.fft.rfft(p, L)
    out = np.fft.irfft(fw * np.conjugate(fp), L)
    return out[:lenp].astype(float, copy=False)


def _sanitize_vec(p):
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return p / s


def _exact_score(p):
    p = _sanitize_vec(p)
    if p is None:
        return float("inf")
    m = p.size
    c = _conv_full(p, p)
    if c.size == 0:
        return float("inf")
    c = np.where(np.isfinite(c), c, 0.0)
    return float(2.0 * m * np.max(c))


def _soft_weights(d, beta):
    d = np.asarray(d, dtype=float)
    n = d.size
    if n == 0:
        return np.array([], dtype=float)
    d = np.where(np.isfinite(d), d, 0.0)
    if beta <= 1e-8:
        return np.full(n, 1.0 / n, dtype=float)
    shift = float(np.max(d))
    if not np.isfinite(shift):
        return np.full(n, 1.0 / n, dtype=float)
    e = np.exp(beta * (d - shift))
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n, dtype=float)
    return e / s

    m = p.size

    c = np.where(np.isfinite(c), c, 0.0)

    exact = float(2.0 * max(0.0, np.max(d)))

    maxh = float(np.max(h)) if h.size else 0.0

def _initial_flat_dft(m, rng, num_peaks=3):
    """Create a sequence by starting with flat-magnitude DFT and projecting."""
    # Create a DFT with roughly flat magnitude but random phases
    L = _next_pow2(2 * m)
    phases = rng.uniform(0, 2 * np.pi, L // 2 + 1)
    phases[0] = 0.0
    # Taper the edges to ensure the inverse DFT is more likely non-negative
    taper = np.ones(L // 2 + 1)
    edge = max(1, L // 32)
    for i in range(edge):
        taper[(i + 1)] = taper[2 * (L // 2) - 1 - i] = np.sin(np.pi * (i + 1) / (2 * edge)) ** 2 if edge > 0 else 1.0
    # Actually let's use a simpler approach: cosine-tapered flat spectrum
    kf = np.arange(L // 2 + 1)
    taper = np.cos(np.pi * kf / (2 * (L // 2))) ** 0.3
    taper[0] = 1.0
    spec = taper * np.exp(1j * phases)
    # Make it conjugate symmetric for real output
    full_spec = np.zeros(L, dtype=complex)
    full_spec[:L // 2 + 1] = spec
    if L > 1:
        full_spec[L // 2 + 1:] = np.conjugate(spec[1:L // 2][::-1])
    a = np.fft.irfft(full_spec, L).real[:m]
    a = np.maximum(a + 0.01, 0.0)
    s = a.sum()
    if s > 0:
        a = a / s
    else:
        a = np.ones(m) / m
    return a


def _initial_multi_bump(m, rng, K=None, spread_range=(0.05, 0.3)):

    p = np.maximum(p, 0.0) + 1e-10
    s = p.sum()
    return p / s


def _initial_u_shape(m, rng, gamma=2.0):

    t = np.arange(m, dtype=float) / m

    p = np.power(np.minimum(t, 1 - t) * 2, gamma)
    p = np.maximum(p, 0.0) + 1e-10
    s = p.sum()
    return p / s


def _initial_w_shape(m, rng):

        p -= depth2 * np.exp(-0.5 * ((t - 0.75) / sigma2) ** 2)
    p = np.maximum(p, 0.0) + 1e-10
    s = p.sum()
    return p / s


def _initial_beta_asym(m, rng):
    """Asymmetric beta-like density."""
    a = float(rng.uniform(0.1, 4.0))
    b = float(rng.uniform(0.1, 4.0))

    p = np.power(t, a) * np.power(1 - t, b)
    p = np.maximum(p, 0.0) + 1e-10
    s = p.sum()
    return p / s


def _initial_cosine_series(m, rng, nterms=8):

    t = (np.arange(m, dtype=float) + 0.5) / m

    p = np.log1p(np.exp(np.clip(z, -50, 50)))
    p = np.maximum(p, 0.0) + 1e-10
    s = p.sum()
    return p / s

    K = int(rng.integers(3, 12))
    edges = np.sort(rng.uniform(0, 1, K - 1))
    blocks = np.concatenate([[0], edges, [1]])
    heights = np.exp(rng.normal(0, 1, K))
    p = np.zeros(m)
    for i in range(K):
        s_idx = max(0, int(blocks[i] * m))
        e_idx = max(s_idx + 1, min(m, int(blocks[i + 1] * m)))
        p[s_idx:e_idx] = heights[i]
    p = np.maximum(p, 0.0) + 1e-10
    s = p.sum()
    return p / s


def _resample(arr, m):
    a = _sanitize_vec(arr)
    if a is None or m <= 0:
        return None
    n = a.size
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m, dtype=float)
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(m) + 0.5) / m
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0) + 1e-12
    return _sanitize_vec(out)


def _run_mmd(p0, rng, time_limit, beta_start=40.0, beta_end=2000.0):
    """Multiplicative mirror descent with adaptive parameters."""
    p = _sanitize_vec(p0)
    if p is None:
        return None, float("inf")
    if time_limit <= 0:
        return p, _exact_score(p)

    m = p.size
    floor = 1e-20 / m
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_exact = _exact_score(p)
    cur_exact, cur_smooth, h, maxh = _gradient_metrics(p, beta_start)

    beta = float(beta_start)
    alpha = 0.15
    deadline = time.monotonic() + time_limit
    stall = 0
    it = 0
    best_after = it

    while time.monotonic() < deadline:
        cur_exact, cur_smooth, h, maxh = _gradient_metrics(p, beta)

        if not np.isfinite(maxh) or maxh <= 1e-15:
            p_try = _sanitize_vec(p * np.exp(rng.normal(0, 0.08, m)))
            if p_try is not None:
                p_try = np.maximum(p_try, floor)
                p_try = p_try / p_try.sum()
                e, _, _, _ = _gradient_metrics(p_try, beta)
                if np.isfinite(e) and e <= best_exact * 1.05:
                    p = p_try
            stall += 1
        else:
            h = np.asarray(h, dtype=float)
            h = np.maximum(h, 0.0)
            h = h / maxh

            # Try multiple step sizes
            tests = [alpha * 0.5, alpha, alpha * 2, alpha * 4, alpha * 0.25]
            tests = [t for t in tests if 1e-5 <= t <= 30.0]
            if not tests:
                tests = [0.1]

            chosen = None
            best_smooth = cur_smooth
            best_exact_try = cur_exact

            for a_step in tests:
                dec = np.exp(-a_step * h)
                p_try = p * dec
                p_try = np.maximum(p_try, floor)
                s = p_try.sum()
                if s <= 0 or not np.isfinite(s):
                    continue
                p_try = p_try / s
                e, s2, _, _ = _gradient_metrics(p_try, beta)
                if np.isfinite(s2) and s2 < best_smooth - 1e-12:
                    chosen = p_try
                    best_smooth = s2
                    best_exact_try = e

            if chosen is not None:
                p = chosen.copy()
                cur_smooth = best_smooth
                cur_exact = best_exact_try
                stall = 0

                if cur_exact < best_exact:
                    best = p.copy()
                    best_exact = cur_exact
                    best_after = it
            else:
                stall += 1
                alpha = max(1e-5, alpha * 0.7)
                if stall >= 15:
                    std = min(0.5, 0.08 + 0.01 * stall)
                    p_try = _sanitize_vec(p * np.exp(rng.normal(0, std, m)))
                    if p_try is not None:
                        p_try = np.maximum(p_try, floor)
                        p_try = p_try / p_try.sum()
                        e, _, _, _ = _gradient_metrics(p_try, beta)
                        if np.isfinite(e) and e <= best_exact * 1.1:
                            p = p_try
                            cur_exact = e
                        else:
                            p = best.copy()
                            cur_exact = best_exact
                        cur_exact, cur_smooth, _, _ = _gradient_metrics(p, beta)
                    stall = 0

                if stall > 100:
                    p = best.copy()
                    cur_exact, cur_smooth, _, _ = _gradient_metrics(p, beta)
                    alpha = 0.15
                    stall = 0

        it += 1
        if it % 15 == 0:
            beta = min(beta_end, beta * 1.2)
            # Periodically reset alpha if stuck
            if it - best_after > 200:
                alpha = max(0.05, alpha * 0.5)

    return best, best_exact


def _local_search_gradient(p0, rng, time_limit):
    """Gradient-guided local search: modify blocks in the direction of negative gradient."""
    p = _sanitize_vec(p0)
    if p is None or time_limit <= 0:
        return None, float("inf")

    n = p.size
    floor = 1e-20 / n
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_score = _exact_score(p)
    deadline = time.monotonic() + time_limit
    stagnant = 0
    it = 0

    while time.monotonic() < deadline:
        # Get gradient-like information
        beta = max(50.0, 50.0 + it * 0.5)
        _, _, h, maxh = _gradient_metrics(p, beta)
        if h is None or maxh < 1e-10:
            h = np.zeros(n)
            maxh = 1.0

        h = h / maxh
        # Coordinates with high h should be decreased, low h should be increased
        # Find blocks where h is high (to decrease) or low (to increase)
        r = rng.random()
        if r < 0.6:
            # Decrease a high-h block
            h_sorted = np.argsort(h)
            top_idx = int(h_sorted[int(n * 0.7) + rng.integers(0, n * 3 // 10)])
            w = int(rng.integers(max(1, n // 200), max(2, n // 30 + 1)))
            start = min(max(0, top_idx - w // 2), n - w)
            factor = float(rng.uniform(0.4, 0.85))
        else:
            # Increase a low-h block
            h_sorted = np.argsort(h)
            bot_idx = int(h_sorted[rng.integers(0, n // 3)])
            w = int(rng.integers(max(1, n // 200), max(2, n // 30 + 1)))
            start = min(max(0, bot_idx - w // 2), n - w)
            factor = float(rng.uniform(1.2, 2.5))

        cand = p.copy()
        cand[start:start + w] *= factor
        cand = np.maximum(cand, floor)
        s = cand.sum()
        if s > 0:
            cand = cand / s
        sc = _exact_score(cand)
        if sc < best_score - 1e-12:
            p = cand
            if sc < best_score:
                best = p.copy()
                best_score = sc
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 80:
                # Random perturbation
                p_try = _sanitize_vec(p * np.exp(rng.normal(0, 0.1, n)))
                if p_try is not None:
                    p_try = np.maximum(p_try, floor)
                    p_try = p_try / p_try.sum()
                    sc_try = _exact_score(p_try)
                    if sc_try < best_score * 1.05:
                        p = p_try
                    else:
                        p = best.copy()
                        best_score = _exact_score(p)
                stagnant = 0
            if stagnant > 400:
                p = best.copy()
                stagnant = 0

        it += 1

    return best, best_score


def _local_search_random_block(p0, rng, time_limit):
    """Random block scaling local search."""
    p = _sanitize_vec(p0)
    if p is None or time_limit <= 0:
        return None, float("inf")

    n = p.size
    floor = 1e-20 / n
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_score = _exact_score(p)
    cur_score = best_score
    deadline = time.monotonic() + time_limit
    stagnant = 0

    factors = [0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.7, 2.5]

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.3:
            w = int(rng.integers(1, max(2, n // 150 + 1)))
        elif r < 0.7:
            w = int(rng.integers(max(2, n // 150), max(3, n // 25 + 1)))
        else:
            low = max(3, n // 25)
            high = max(low + 1, min(n, n // 8 + 1))
            w = int(rng.integers(low, high + 1))
        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))

        base = p.copy()
        current_best = base
        current_score = cur_score

        for f in factors:
            cand = base.copy()
            cand[start:start + w] *= f
            cand = np.maximum(cand, floor)
            s = cand.sum()
            if s <= 0:
                continue
            cand = cand / s
            sc = _exact_score(cand)
            if sc < current_score - 1e-12:
                current_score = sc
                current_best = cand

        if current_best is not base:
            p = current_best
            cur_score = current_score
            if cur_score < best_score:
                best = p.copy()
                best_score = cur_score
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 100:
                p_try = _sanitize_vec(p * np.exp(rng.normal(0, 0.06, n)))
                if p_try is not None:
                    p_try = np.maximum(p_try, floor)
                    p_try = p_try / p_try.sum()
                    sc = _exact_score(p_try)
                    if sc < best_score * 1.03:
                        p = p_try
                        cur_score = sc
                    else:
                        p = best.copy()
                        cur_score = best_score
                stagnant = 0

    return best, best_score


def _mass_transfer(p0, rng, time_limit):
    """Mass transfer: move mass from high-contribution areas to low-contribution areas."""
    p = _sanitize_vec(p0)
    if p is None or time_limit <= 0:
        return None, float("inf")

    n = p.size
    floor = 1e-20 / n
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_score = _exact_score(p)
    deadline = time.monotonic() + time_limit
    stagnant = 0
    it = 0

    while time.monotonic() < deadline:
        # Compute the contribution of each element to the max convolution
        c = _conv_full(p, p)
        c = np.where(np.isfinite(c), c, 0.0)
        max_c = np.max(c)
        if max_c < 1e-15:
            break

        # Find which lag has the max
        max_lag = int(np.argmax(c))

        # The elements that contribute to c[max_lag] are pairs (i, max_lag-i)
        # We want to reduce mass at elements that heavily contribute to the max
        contrib = np.zeros(n)
        for i in range(n):
            j = max_lag - i
            if 0 <= j < n:
                contrib[i] += p[i] * p[j]

        # Transfer mass from high-contrib to low-contrib
        hi_thresh = np.percentile(contrib, 70)
        lo_thresh = np.percentile(contrib, 30)
        hi_mask = contrib > hi_thresh
        lo_mask = contrib < lo_thresh

        if hi_mask.sum() < 1 or lo_mask.sum() < 1:
            break

        # Move a small fraction of mass
        frac = float(rng.uniform(0.05, 0.2))
        hi_mass = p[hi_mask].sum() * frac
        lo_total = p[lo_mask].sum()

        cand = p.copy()
        cand[hi_mask] *= (1 - frac)
        # Distribute to low-contrib positions proportional to their current value
        lo_vals = cand[lo_mask]
        if lo_vals.sum() > 0:
            cand[lo_mask] = lo_vals + hi_mass * lo_vals / lo_vals.sum()
        else:
            idx_lo = np.where(lo_mask)[0]
            cand[idx_lo] += hi_mass / len(idx_lo)

        cand = np.maximum(cand, floor)
        s = cand.sum()
        if s > 0:
            cand = cand / s
        sc = _exact_score(cand)
        if sc < best_score - 1e-12:
            p = cand
            if sc < best_score:
                best = p.copy()
                best_score = sc
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 100:
                p = best.copy()
                stagnant = 0

        it += 1

    return best, best_score


def _smooth_array(p, samples=1.0):
    p = _sanitize_vec(p)
    if p is None:
        return None
    s = float(samples)
    if s <= 0 or p.size < 5:
        return p
    rad = int(max(1, np.ceil(3 * s)))
    k = np.arange(-rad, rad + 1, dtype=float)
    kern = np.exp(-0.5 * (k / s) ** 2)
    kern /= kern.sum()
    out = np.convolve(p, kern, mode="same")
    return _sanitize_vec(out)


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Search for the best sequence within budget_s seconds.
    Uses multi-stage optimization: diverse inits -> MMD -> local search -> polish.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget <= 30:
        budget = 60.0

    start_time = time.monotonic()
    deadline = start_time + max(60.0, budget - 25.0)

    try:
        iseed = int(seed)
    except Exception:
        iseed = 12345
    rng = np.random.default_rng((iseed % 1000000007) + 1)

    # Try to use parent
    parent_arr = None
    try:
        parent_raw = height_sequence_1
        parent_arr = _sanitize_vec(np.asarray(parent_raw, dtype=float))
    except (NameError, Exception):
        parent_arr = None

    all_best = []  # list of (score, arr)

    # Stage 1: Quick exploration at moderate length to find good basins
    small_n = 1024
    rem = deadline - time.monotonic()
    if rem > 120:
        small_starts = []

        # Diverse initializations
        small_starts.append(_initial_multi_bump(small_n, rng, K=3, spread_range=(0.08, 0.25)))
        small_starts.append(_initial_multi_bump(small_n, rng, K=4, spread_range=(0.05, 0.2)))
        small_starts.append(_initial_u_shape(small_n, rng, gamma=1.5))
        small_starts.append(_initial_w_shape(small_n, rng))
        small_starts.append(_initial_flat_dft(small_n, rng))
        small_starts.append(_initial_beta_asym(small_n, rng))
        small_starts.append(_initial_cosine_series(small_n, rng, nterms=10))

        if parent_arr is not None:
            sp = _resample(parent_arr, small_n)
            if sp is not None:
                small_starts.append(sp)
                sp2 = _sanitize_vec(sp * np.exp(rng.normal(0, 0.05, small_n)))
                if sp2 is not None:
                    small_starts.append(sp2)

        scored_starts = []
        for sp in small_starts:
            if sp is not None:
                sc = _exact_score(sp)
                if np.isfinite(sc):
                    scored_starts.append((sc, sp))

        scored_starts.sort(key=lambda x: x[0])

        # Run MMD on top 4 starts
        for sc, sp in scored_starts[:4]:
            rem = deadline - time.monotonic()
            if rem < 20:
                break
            talloc = min(25.0, max(5.0, rem - 10.0))
            p, sc2 = _run_mmd(sp, rng, talloc, beta_start=30.0, beta_end=800.0)
            if p is not None and np.isfinite(sc2):
                all_best.append((sc2, p.copy()))

        all_best.sort(key=lambda x: x[0])
        print(f"Stage 1 (n={small_n}): best score = {all_best[0][0]:.6f}" if all_best else "Stage 1: no results")

    # Stage 2: Main optimization at target length
    # Try multiple lengths and pick the best
    rem = deadline - time.monotonic()
    target_lengths = [2048, 3072, 4096]

    # If we have limited time, use fewer lengths
    if rem < 800:
        target_lengths = [2048, 3072]
    if rem < 400:
        target_lengths = [2048]

    for target_n in target_lengths:
        rem = deadline - time.monotonic()
        if rem < 150:
            break

        # Build diverse starts for this length
        final_starts = []

        # From best small-n results (resampled)
        for sc, arr in all_best[:3]:
            sp = _resample(arr, target_n)
            if sp is not None:
                sp = _smooth_array(sp, 1.0)
                if sp is not None:
                    final_starts.append(sp)
                # Also a perturbed version
                sp2 = _sanitize_vec(_resample(arr, target_n) * np.exp(rng.normal(0, 0.03, target_n)))
                if sp2 is not None:
                    final_starts.append(sp2)

        # From parent
        if parent_arr is not None:
            sp = _resample(parent_arr, target_n)
            if sp is not None:
                final_starts.append(sp)
                sp2 = _sanitize_vec(sp * np.exp(rng.normal(0, 0.04, target_n)))
                if sp2 is not None:
                    final_starts.append(sp2)

        # Fresh diverse inits
        final_starts.append(_initial_multi_bump(target_n, rng, K=3, spread_range=(0.06, 0.22)))
        final_starts.append(_initial_multi_bump(target_n, rng, K=4, spread_range=(0.04, 0.18)))
        final_starts.append(_initial_multi_bump(target_n, rng, K=5, spread_range=(0.03, 0.15)))
        final_starts.append(_initial_u_shape(target_n, rng, gamma=1.8))
        final_starts.append(_initial_w_shape(target_n, rng))
        final_starts.append(_initial_flat_dft(target_n, rng))
        final_starts.append(_initial_beta_asym(target_n, rng))

        # Score and sort
        scored_final = []
        for sp in final_starts:
            if sp is not None and sp.size == target_n:
                sc = _exact_score(sp)
                if np.isfinite(sc):
                    scored_final.append((sc, sp))

        scored_final.sort(key=lambda x: x[0])
        print(f"n={target_n}: best start score = {scored_final[0][0]:.6f}" if scored_final else f"n={target_n}: no valid starts")

        if not scored_final:
            continue

        # Allocate time: top 3 starts get most time
        n_start = min(3, len(scored_final))
        rem = deadline - time.monotonic()
        reserve = max(120.0, 0.2 * rem) if rem > 500 else 60.0
        main_time = max(30.0, rem - reserve)
        fracs = [0.45, 0.30, 0.25]

        best_for_n = None
        best_score_for_n = float("inf")

        for idx in range(n_start):
            rem = deadline - time.monotonic()
            if rem < 60:
                break
            frac = fracs[idx] if idx < len(fracs) else 0.15
            talloc = main_time * frac
            talloc = min(talloc, max(10.0, rem - 30.0))
            if talloc < 10:
                talloc = max(5.0, rem - 20.0)

            sp = scored_final[idx][1]
            p, sc = _run_mmd(sp, rng, talloc, beta_start=40.0, beta_end=2000.0)
            if p is not None and np.isfinite(sc):
                all_best.append((sc, p.copy()))
                if sc < best_score_for_n:
                    best_score_for_n = sc
                    best_for_n = p.copy()

        if best_for_n is not None:
            print(f"n={target_n}: MMD best = {best_score_for_n:.6f} (time remaining: {deadline - time.monotonic():.0f}s)")

        # Polish the best for this n
        rem = deadline - time.monotonic()
        if best_for_n is not None and rem > 100:
            # Gradient-guided local search
            t1 = min(0.3 * rem, 300.0)
            p, sc = _local_search_gradient(best_for_n, rng, t1)
            if p is not None and np.isfinite(sc) and sc < best_score_for_n:
                best_score_for_n = sc
                best_for_n = p.copy()
                all_best.append((sc, p.copy()))

            rem = deadline - time.monotonic()
            # Random block search
            if rem > 60:
                t2 = min(0.25 * rem, 250.0)
                p, sc = _local_search_random_block(best_for_n, rng, t2)
                if p is not None and np.isfinite(sc) and sc < best_score_for_n:
                    best_score_for_n = sc
                    best_for_n = p.copy()
                    all_best.append((sc, p.copy()))

            rem = deadline - time.monotonic()
            # Mass transfer
            if rem > 40:
                t3 = min(0.2 * rem, 150.0)
                p, sc = _mass_transfer(best_for_n, rng, t3)
                if p is not None and np.isfinite(sc) and sc < best_score_for_n:
                    best_score_for_n = sc
                    best_for_n = p.copy()
                    all_best.append((sc, p.copy()))

            rem = deadline - time.monotonic()
            # Final MMD polish with high beta
            if rem > 40:
                t4 = min(0.2 * rem, 150.0)
                p, sc = _run_mmd(best_for_n, rng, t4, beta_start=200.0, beta_end=5000.0)
                if p is not None and np.isfinite(sc) and sc < best_score_for_n:
                    best_score_for_n = sc
                    best_for_n = p.copy()
                    all_best.append((sc, p.copy()))

    # Get overall best
    if all_best:
        all_best.sort(key=lambda x: x[0])
        best_score = all_best[0][0]
        best_arr = all_best[0][1]
    else:
        # Fallback: uniform sequence
        best_arr = np.ones(2048) / 2048
        best_score = 2.0

    print(f"Overall best before final polish: {best_score:.6f} (n={best_arr.size})")

    # Final polish on the best candidate
    rem = deadline - time.monotonic()
    if rem > 30:
        # Smooth slightly and re-optimize
        best_arr_s = _smooth_array(best_arr, 0.5)
        if best_arr_s is not None:
            sc_s = _exact_score(best_arr_s)
            if sc_s < best_score:
                best_score = sc_s
                best_arr = best_arr_s

        # Short MMD
        if rem > 20:
            t_final = min(0.3 * rem, 60.0)
            p, sc = _run_mmd(best_arr, rng, t_final, beta_start=500.0, beta_end=10000.0)
            if p is not None and np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_arr = p.copy()

        # Very short local search
        rem = deadline - time.monotonic()
        if rem > 15:
            t_ls = min(0.3 * rem, 30.0)
            p, sc = _local_search_random_block(best_arr, rng, t_ls)
            if p is not None and np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_arr = p.copy()

    # Final validation
    best_arr = _sanitize_vec(best_arr)
    if best_arr is None:
        best_arr = np.ones(2048) / 2048

    # Try to verify with evaluate_sequence
    out_list = [float(x) for x in best_arr]
    try:
        eval_score = float(evaluate_sequence(out_list))
        if np.isfinite(eval_score):
            print(f"evaluate_sequence score: {eval_score:.6f}")
            if eval_score < best_score:
                best_score = eval_score
    except Exception:
        pass

    # Also check a few other candidates
    for sc, arr in all_best[1:4]:
        if time.monotonic() >= deadline - 5:
            break
        a = _sanitize_vec(arr)
        if a is None:
            continue
        try:
            esc = float(evaluate_sequence([float(x) for x in a]))
            if np.isfinite(esc) and esc < best_score:
                best_score = esc
                best_arr = a
                out_list = [float(x) for x in a]
        except Exception:
            pass

    # Final output
    out = np.asarray(out_list if len(out_list) == best_arr.size else best_arr, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s <= 0 or not np.isfinite(s):
        out = np.ones(2048) / 2048
    else:
        out = out / s

    print(f"Final: n={out.size}, internal_score={best_score:.6f}")
    return [float(x) for x in out]
