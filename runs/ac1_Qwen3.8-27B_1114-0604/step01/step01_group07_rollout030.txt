    while n < m:
        n <<= 1

    n = arr.size
    if n == 0:
        return np.array([])
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(arr, N)
    c = np.fft.irfft(fa * fa, N)

    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())

    mx = float(np.max(c))

    i = np.arange(n, dtype=np.float64)
    a = 1.0 + amp * np.sin(2.0 * np.pi * (i + 0.5) / float(n))
    a = np.clip(a, 0.0, 1000.0)
    a /= a.sum()

    best_score = _score(best)

    stall_count = 0

                ki = k - idx
