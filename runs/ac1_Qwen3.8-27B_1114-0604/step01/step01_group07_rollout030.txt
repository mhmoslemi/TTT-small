"""
Low-peak nonnegative step-function search using diverse sinusoidal initializations
and aggressive active-set gradient refinement with adaptive step sizes.
Minimizes 2*n*max(a*a)/(sum(a))^2 for non-negative a.

import numpy as np
import time


def _fft_convolve(a):
    """Compute linear convolution a * a using FFT."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = 1
    while N < need:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _score_internal(a):
    """Fast internal score matching evaluate_sequence."""
    a = np.asarray(a, dtype=np.float64)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    if c.size == 0:
        return np.inf
    mx = float(c.max())
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * len(a) * mx / (s * s))

            if j_end >= j_start:
                g[j_start:j_end + 1] += 2.0 * weights[k] * a[k - j_end:k - j_start + 1][::-1]

        g /= float(n)  # Scale down
        g = g - g.mean()

        # Line search
        best_trial = np.inf
        best_new_a = None
        for trial_lr in [lr, lr * 0.5, lr * 2.0, lr * 0.25]:
            if trial_lr <= 1e-10:
                continue
            exponent = -trial_lr * g * n
            exponent = np.clip(exponent, -3.0, 3.0)
            new_a = a * np.exp(exponent)
            new_a = np.clip(new_a, 0.0, 1000.0)
            ns = float(new_a.sum())
            if ns < 0.01:
                continue
            new_a = new_a / ns
            trial = _score_internal(new_a)
            if trial < best_trial:
                best_trial = trial
                best_new_a = new_a

        if best_new_a is not None and best_trial < best_score - 1e-10:
            a = best_new_a
            best_a = a.copy()
            best_score = best_trial
            stall = 0
            beta = min(beta * 1.1, 500.0)
        else:
            stall += 1
            lr *= 0.7
            if stall > 80:
                break

    return best_a, best_score


def _perturb_and_refine(a, time_budget, rng):
    """Add random perturbation then refine."""
    n = len(a)
    # Multiplicative perturbation
    noise = rng.lognormal(0, 0.1, size=n)
    perturbed = a * noise
    perturbed = np.clip(perturbed, 0.0, 1000.0)
    s = perturbed.sum()
    if s < 0.01:
        perturbed = a.copy()
    else:
        perturbed = perturbed / s
    return _refine_active(perturbed, time_budget)


def _safe_resolve(a):
    """Sanitize sequence for final evaluation."""
    if a is None:
        return None
    arr = np.asarray(a, dtype=np.float64).ravel()
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return None
    arr = arr / s
    return arr


def _optimize_sine_amplitude(n, mode, amps_start):
    """Find optimal amplitude for a sinusoidal family."""
    best_score = np.inf
    best_amp = amps_start[0]
    best_a = _make_sine(n, amps_start[0], mode)

    for amp in amps_start:
        a = _make_sine(n, amp, mode)
        sc = _score_internal(a)
        if sc < best_score:
            best_score = sc
            best_amp = amp
            best_a = a

    # Local refinement
    step = (amps_start[-1] - amps_start[0]) / max(1, len(amps_start) - 1)
    for _ in range(8):
        low = max(0.0, best_amp - step)
        high = min(1.5, best_amp + step)
        if high - low < 1e-8:
            break
        for amp in np.linspace(low, high, 13):
            a = _make_sine(n, amp, mode)
            sc = _score_internal(a)
            if sc < best_score:
                best_score = sc
                best_amp = amp
                best_a = a
        step *= 0.5

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a low-evaluation non-negative sequence.
    Uses diverse initializations and active-set gradient refinement.
    """
    t0 = time.monotonic()
    total_budget = max(30.0, float(budget_s))
    hard_deadline = t0 + total_budget - 25.0

    rng = np.random.default_rng(seed)
    all_candidates = []

    # Phase 1: Generate diverse candidates
    print(f"[phase1] Generating candidates...")

    # Try provided base sequence
    try:
        base = np.asarray(height_sequence_1, dtype=np.float64)
        if base.size > 10 and np.all(np.isfinite(base)):
            base = np.clip(base, 0, 1000)
            s = base.sum()
            if s > 0.01:
                base = base / s
                sc = _score_internal(base)
                all_candidates.append((sc, "base", base))
                print(f"  base: n={len(base)}, score={sc:.8f}")
    except (NameError, Exception):
        pass

    # Sinusoidal candidates with amplitude optimization
    n_options = [2048, 4096, 3072, 1536]
    for n in n_options:
        if time.monotonic() > hard_deadline:
            break
        for mode in [1, 2]:
            amps = np.linspace(0.3, 1.2, 10)
            a, sc = _optimize_sine_amplitude(n, mode, amps)
            all_candidates.append((sc, f"sine_n{n}_m{mode}", a))
            if mode == 1:
                print(f"  sine n={n} mode={mode}: score={sc:.8f}")

        # Squared sinusoidal
        for amp in [0.5, 0.7, 0.9]:
            a = _make_sine_sq(n, amp, 1)
            sc = _score_internal(a)
            all_candidates.append((sc, f"sine2_n{n}_a{amp:.1f}", a))

        # Multi-harmonic
        for a1 in [0.6, 0.8]:
            for a2 in [0.1, 0.3, -0.2]:
                a = _make_multi_sine(n, a1, a2)
                sc = _score_internal(a)
                all_candidates.append((sc, f"multi_n{n}_{a1}_{a2}", a))

    # Sort candidates
    all_candidates.sort(key=lambda x: x[0])
    print(f"[phase1] Best candidate: {all_candidates[0][1]} score={all_candidates[0][0]:.8f}")
    print(f"[phase1] Time: {time.monotonic()-t0:.1f}s")

    # Phase 2: Refine top candidates
    print(f"[phase2] Refining top candidates...")
    best_a = all_candidates[0][2].copy()
    best_score = all_candidates[0][0]

    # Refine the best candidate with most time
    top_indices = list(range(min(3, len(all_candidates))))
    remaining = hard_deadline - time.monotonic()
    n_refine = len(top_indices)

    for idx_i, ci in enumerate(top_indices):
        if time.monotonic() > hard_deadline - 60:
            break
        alloc = (hard_deadline - time.monotonic() - 30) / (n_refine - idx_i)
        alloc = min(alloc, 800.0)
        if alloc < 30:
            break

        sc, name, a = all_candidates[ci]
        print(f"  Refine [{ci}] {name}: initial={sc:.8f}, budget={alloc:.0f}s")

        t_start = time.monotonic()
        refined_a, refined_score = _refine_active(a, alloc, initial_K=16)
        t_elapsed = time.monotonic() - t_start

        if refined_score < best_score:
            best_score = refined_score
            best_a = refined_a.copy()
            print(f"    Improved: {refined_score:.8f} (elapsed {t_elapsed:.0f}s)")
        else:
            print(f"    No improvement: {refined_score:.8f} (elapsed {t_elapsed:.0f}s)")

        # If we still have time, try smooth refinement on the best
        if time.monotonic() < hard_deadline - 100 and idx_i == 0:
            sm_budget = min(200.0, hard_deadline - time.monotonic() - 60)
            if sm_budget > 30:
                print(f"  Smooth refine: budget={sm_budget:.0f}s")
                sm_a, sm_score = _refine_smooth(best_a, sm_budget)
                if sm_score < best_score:
                    best_score = sm_score
                    best_a = sm_a.copy()
                    print(f"    Smooth improved: {sm_score:.8f}")

        # Try perturbation restart if time allows
        if time.monotonic() < hard_deadline - 80 and idx_i < 2:
            pert_budget = min(150.0, hard_deadline - time.monotonic() - 60)
            if pert_budget > 20:
                print(f"  Perturb+refine: budget={pert_budget:.0f}s")
                p_a, p_score = _perturb_and_refine(best_a, pert_budget, rng)
                if p_score < best_score:
                    best_score = p_score
                    best_a = p_a.copy()
                    print(f"    Perturb improved: {p_score:.8f}")

    print(f"[phase2] Best score after refinement: {best_score:.8f}")
    print(f"[phase2] Total time: {time.monotonic()-t0:.1f}s")

    # Phase 3: Validate and return
    print(f"[phase3] Validating...")
    arr = _safe_resolve(best_a)
    if arr is not None:
        try:
            real = float(evaluate_sequence(arr.tolist()))
            print(f"  Real score: {real:.8f}")
            if np.isfinite(real):
                return [float(x) for x in arr]
        except Exception as e:
            print(f"  Error: {e}")

    # Fallback: try other candidates
    for sc, name, cand in all_candidates[:5]:
        arr = _safe_resolve(cand)
        if arr is not None:
            try:
                real = float(evaluate_sequence(arr.tolist()))
                if np.isfinite(real):
                    print(f"  Fallback {name}: real={real:.8f}")
                    return [float(x) for x in arr]
            except Exception:
                pass

    # Ultimate fallback
    fallback = np.ones(256, dtype=np.float64) / 256.0
    print("  Using constant fallback (score=2)")
    return [float(x) for x in fallback]
