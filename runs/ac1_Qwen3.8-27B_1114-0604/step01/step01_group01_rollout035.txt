The objective is F = 2n * LSE(beta * c) / beta.

dLSE/dc_k = exp(beta * c_k) / sum(exp(beta * c)) = w_k (the normalized softmax weight)

So dF/da_i = 2n * (1/beta) * sum_k w_k * beta * dc_k/da_i = 2n * sum_k w_k * dc_k/da_i

And dc_k/da_i = a[i + k - (n-1)] + a[i - (k - (n-1))] (as before)

So dF/da_i = 2n * sum_k w_k * (a[i+k-(n-1)] + a[i-(k-(n-1))])

Let tau = k - (n-1):

By symmetry of w (w[tau+(n-1)] = w[-tau+(n-1)]):
= 2n * 2 * sum_tau w_c[tau] * a[i+tau] (where w_c[tau] = w[tau+(n-1)])

```python
"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS with log-sum-exp smoothing
on a log-softmax parameterization, adaptive beta continuation, multiple diverse
restarts, and exact coordinate block-polish. Different from the previous
multiplicative-descent approach by using quasi-Newton second-order information.
"""

import numpy as np
import time
from scipy.optimize import minimize as sp_min


def _npow2(x):
    x = int(max(1, x))
    return 1 << (x - 1).bit_length()


def _sanitize(a):
    a = np.asarray(a, dtype=float).ravel()
    if a.size == 0:
        return None
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return a / s


def _exact_score(a):
    a = _sanitize(a)
    if a is None:
        return float('inf')
    n = a.size
    L = _npow2(2 * n)
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)[:2 * n - 1]
    c = np.where(np.isfinite(c), c, 0.0)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    return float(2.0 * n * float(np.max(c)) / (s * s))


def _smoothed_obj_grad(b, beta, n, L_conv, L_corr):
    """
    Compute smoothed objective and its gradient w.r.t. log-parameterization b.
    a = softmax(b), objective = 2n * LSE(beta * conv(a,a)) / beta
    """
    b_centered = b - b.mean()
    a = np.exp(b_centered)
    s = a.sum()
    if s <= 0.0 or not np.isfinite(s):
        return 1e10, np.zeros(n)
    a = a / s

    # Full convolution via FFT
    fa = np.fft.rfft(a, L_conv)
    c = np.fft.irfft(fa * fa, L_conv)[:2 * n - 1]
    c = np.maximum(c, 0.0)

    # Log-sum-exp of beta * c
    bc_arr = beta * c
    mx = float(np.max(bc_arr))
    e = np.exp(bc_arr - mx)
    se = float(np.sum(e))
    if se <= 0.0 or not np.isfinite(se):
        return 1e10, np.zeros(n)
    obj = 2.0 * n * (mx + np.log(se)) / beta
    w = e / se

    # Gradient: g_a[i] = 4n * sum_j a[j] * w[j - i + (n-1)]
    # Computed via FFT: v = (a * w_flipped) linear conv, take [n-1 : 2n-1]
    wf = w[::-1]
    Af = np.fft.rfft(a, L_corr)
    Wf = np.fft.rfft(wf, L_corr)
    conv_full = np.fft.irfft(Af * Wf, L_corr)
    v = conv_full[n - 1: 2 * n - 1]
    v = np.where(np.isfinite(v), v, 0.0)

    g_a = 4.0 * n * v
    dot_ga_a = float(np.dot(g_a, a))
    grad_b = a * (g_a - dot_ga_a)
    grad_b = np.where(np.isfinite(grad_b), grad_b, 0.0)

    return float(obj), grad_b


def _make_starts(n, rng, parent=None):
    """Generate diverse non-negative normalized starting points."""
    starts = []
    t = (np.arange(n, dtype=float) + 0.5) / n

    # Parent resampled
    if parent is not None:
        p = _sanitize(parent)
        if p is not None and p.size != n:
            xo = (np.arange(p.size) + 0.5) / p.size
            p = np.interp(t, xo, p)
            p = _sanitize(p)
        if p is not None:
            starts.append(p.copy())

    # Two-block structures
    for ratio in (0.3, 0.5, 0.7):
        for h2 in (1.5, 2.0, 3.0):
            idx = int(ratio * n)
            if idx <= 0 or idx >= n:
                continue
            p = np.ones(n)
            p[idx:] = h2
            p = _sanitize(p)
            if p is not None:
                starts.append(p)

    # Gaussian bumps (various numbers)
    for nb in (2, 3, 5, 8):
        p = np.zeros(n)
        for _ in range(nb):
            c = rng.uniform(0.1, 0.9)
            w = rng.uniform(0.03, 0.25)
            p += np.exp(-((t - c) / w) ** 2)
        p = _sanitize(p)
        if p is not None:
            starts.append(p)

    # Random cosine series
    for _ in range(4):
        p = np.ones(n)
        for k in range(1, 10):
            amp = rng.normal(0, 1.0 / k)
            ph = rng.uniform(0, 2 * np.pi)
            p += amp * np.cos(2 * np.pi * k * t + ph)
        p = np.log1p(np.exp(np.clip(p, -30, 30)))
        p = _sanitize(p)
        if p is not None:
            starts.append(p)

    # Beta-like / asymmetric
    for al in (0.3, 0.7, 2.0, 4.0):
        tt = np.clip(t, 1e-8, 1 - 1e-8)
        p = tt ** al * (1 - tt) ** (al * 0.4 + 0.3)
        p = _sanitize(p)
        if p is not None:
            starts.append(p)

    # Uniform (baseline)
    starts.append(np.ones(n) / n)

    return starts


def _lbfgs_optimize(a0, n, deadline, verbose=False):
    """
    Run L-BFGS with adaptive beta continuation on the log-parameterization.
    Returns (best_a, best_exact_score).
    """
    L_conv = _npow2(2 * n)
    L_corr = _npow2(3 * n)

    b = np.log(a0 + 1e-30)
    b = b - b.mean()

    best_a = a0.copy()
    best_score = _exact_score(a0)

    beta_schedule = [20, 40, 80, 160, 320, 640, 1280, 2560, 5120, 10240, 20480, 40960]

    for bi, beta in enumerate(beta_schedule):
        if time.monotonic() >= deadline:
            break

        time_left = deadline - time.monotonic()
        if time_left < 5.0:
            break

        # Scale max iterations based on remaining time and beta level
        max_iter = 250
        if beta > 4096:
            max_iter = 400

        counter = [0]

        def _fun_wrapper(x):
            counter[0] += 1
            return _smoothed_obj_grad(x, beta, n, L_conv, L_corr)

        try:
            res = sp_min(
                fun=_fun_wrapper,
                x0=b.copy(),
                jac=True,
                method='L-BFGS-B',
                options={
                    'maxiter': max_iter,
                    'maxfun': max_iter + 10,
                    'ftol': 1e-12,
                    'gtol': 1e-8,
                    'maxls': 50
                }
            )
            b = res.x
        except Exception:
            pass

        b = b - b.mean()
        a = np.exp(b)
        a = _sanitize(a)

        if a is not None:
            sc = _exact_score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                if verbose:
                    print(f"    beta={beta}: score={sc:.8f} (iters={counter[0]})")
            elif verbose:
                print(f"  beta={beta}: score={sc:.8f} (iters={counter[0]})")

    return best_a, best_score


def _coord_polish(a0, deadline, rng, verbose=False):
    """
    Exact coordinate block scaling: multiply random contiguous blocks by a factor.
    Accept only if exact score decreases.
    """
    p = _sanitize(a0)
    if p is None:
        return a0, _exact_score(a0)

    n = p.size
    cur = _exact_score(p)
    best = p.copy()
    best_sc = cur
    stagnant = 0

    factors = (0.4, 0.6, 0.75, 0.9, 1.1, 1.25, 1.5, 2.0)

    while time.monotonic() < deadline and stagnant < 300:
        # Choose block width (log-uniform)
        r = rng.random()
        if r < 0.3:
            w = int(rng.integers(1, max(2, n // 128)))
        elif r < 0.7:
            w = int(rng.integers(max(2, n // 128), max(3, n // 16)))
        else:
            w = int(rng.integers(max(3, n // 16), min(n, n // 4)))
        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w))

        improved = False
        for f in factors:
            cand = p.copy()
            cand[start:start + w] *= f
            cand = _sanitize(cand)
            if cand is None:
                continue
            sc = _exact_score(cand)
            if sc < cur - 1e-14:
                p = cand
                cur = sc
                improved = True
                if cur < best_sc:
                    best_sc = cur
                    best = p.copy()
                break

        if not improved:
            stagnant += 1
            if stagnant > 50:
                # Perturb and restart
                p_try = _sanitize(p * np.exp(rng.normal(0, 0.03, n)))
                if p_try is not None:
                    sc_try = _exact_score(p_try)
                    if sc_try <= best_sc * 1.02:
                        p = p_try
                        cur = sc_try
                    else:
                        p = best.copy()
                        cur = best_sc
                stagnant = 0
        else:
            stagnant = 0

    return best, best_sc


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses L-BFGS with log-sum-exp smoothing, adaptive beta continuation,
    diverse restarts, and coordinate block polish.
    """
    t_start = time.monotonic()
    deadline = t_start + max(60.0, float(budget_s) - 40.0)

    try:
        iseed = int(seed)
    except Exception:
        iseed = 12345
    rng = np.random.default_rng((iseed % 10**9) + 1)

    # Try to get parent sequence
    parent = None
    try:
        parent = _sanitize(np.asarray(height_sequence_1, dtype=float))

    # Choose length

    scored = []
    for i, s in enumerate(starts):

    for sc, idx, _ in scored[:5]:
        print(f"  start[{idx}] = {sc:.6f}")

    best_a = scored[0][2].copy() if scored else np.ones(n) / n
    best_score = scored[0][0] if scored else float('inf')

    # L-BFGS optimization on top starts (allocate time proportionally)
```
