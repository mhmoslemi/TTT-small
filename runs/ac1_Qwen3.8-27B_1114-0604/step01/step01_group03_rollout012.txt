"""
LP-based Frank-Wolfe optimization for minimizing the peak of autocorrelation.
Uses scipy's HiGHS solver for the LP direction finding, combined with
soft gradient descent for local refinement.
"""

import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt
import time
import math

def _normalize(a, eps=1e-300):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a.fill(1.0)
        s = float(a.sum())
    a /= s
    a = np.maximum(a, eps)
    a /= float(a.sum())
    return a

def _fft_len(n):
    m = 2 * n - 1
    return 1 << (m - 1).bit_length()

def _conv_self_fft(a, L, m):
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c

def _score(a, L, m, n):
    c = _conv_self_fft(a, L, m)
    return 2.0 * n * float(np.max(c))

def _build_constraint_matrix(f, n):
    """Build the sparse convolution constraint matrix A where (f*g)[k] = (A @ g)[k]."""
    m = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(m):
        # A[k, j] = f[k-j] if 0 <= k-j < n
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = range(j_start, j_end + 1)
            fs = f[k - np.array(js)]
            rows.extend([k] * len(js))
            cols.extend(js)
            vals.extend(fs)
    A = sp.csr_matrix((vals, (rows, cols)), shape=(m, n))
    return A

def _solve_lp_direction(f, A, M, n):
    """Solve the LP: max sum(g) s.t. A@g <= M*1, g >= 0."""
    m = 2 * n - 1

"""
LP-based Frank-Wolfe optimization for minimizing 2*n*max(a*a)/(sum(a))^2.
Combines exact LP direction finding (via HiGHS) with FFT-based soft gradient
descent for local refinement. Uses the corrected subgradient formula:
grad = 2 * (w cross-correlate a) for soft objective, and solves the
max-sum(g) s.t. (f*g)<=M, g>=0 LP for exact FW directions.
"""

import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt
import time
import math


def _normalize(a, eps=1e-300):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 1e-300:
        a.fill(1.0)
        s = float(a.sum())
    a /= s
    a = np.maximum(a, eps)
    a /= float(a.sum())
    return a


def _fft_len(n):
    m = 2 * n - 1
    return 1 << (m - 1).bit_length()


def _conv_self(a, L, m):
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a, L, m, n):
    c = _conv_self(a, L, m)
    return 2.0 * n * float(np.max(c))


def _build_conv_matrix(f, n):
    """Build sparse matrix A where (A @ g)[k] = (f*g)[k] for k=0..2n-2."""
    m = 2 * n - 1
    rows = []
    cols = []
    vals = []
    for k in range(m):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            fs = f[k - js]
            rows.extend([k] * len(js))
            cols.extend(js.tolist())
            vals.extend(fs.tolist())
    A = sp.csr_matrix((vals, (rows, cols)), shape=(m, n))
    return A


def _solve_lp(f, A, M, n):
    """Solve max sum(g) s.t. A@g <= M, g >= 0. Returns normalized g or None."""
    m = 2 * n - 1
    c = -np.ones(n)
    b_ub = np.full(m, M)
    try:
        res = opt.linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=[(0.0, None)] * n, method='highs',
                          options={'disp': False, 'time_limit': 60})
        if res.success and res.x is not None:
            g = np.maximum(res.x, 0.0)
            s = float(g.sum())
            if s > 1e-15:
                g /= s
                return g
    except Exception:
        pass
    return None


def _soft_direction(a, w, L, n):
    """Compute gradient direction: 2*(w cross-corr a) via FFT."""
    W = np.fft.rfft(w, L)
    A = np.fft.rfft(a, L)
    corr = np.fft.irfft(W * np.conj(A), L)[:n]
    np.maximum(corr, 0.0, out=corr)
    return 2.0 * corr


def _soft_step(a, L, m, n, temp, eta):
    """One multiplicative gradient step with soft max."""
    c = _conv_self(a, L, m)
    v = n * c
    vmax = float(v.max())
    temp_eff = min(float(temp), 100.0 / max(vmax, 1e-12))
    if temp_eff <= 0:
        temp_eff = 1e-12
    z = temp_eff * (v - vmax)
    z = np.clip(z, -700.0, 0.0)
    w = np.exp(z)
    ws = float(w.sum())
    if ws <= 0 or not np.isfinite(ws):
        return a
    w /= ws
    d = _soft_direction(a, w, L, n)
    dmax = float(d.max())
    if dmax <= 0:
        return a
    d = d / dmax
    a2 = a * np.exp(-eta * d)
    a2 = _normalize(a2)
    return a2


def _line_search(a, b, L, m, n, best_score):
    """Find best convex combination of a and b."""
    best = a.copy()
    best_sc = best_score
    for lam in (0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0):
        h = _normalize((1.0 - lam) * a + lam * b)
        sc = _score(h, L, m, n)
        if sc < best_sc:
            best_sc = sc
            best = h
    return best, best_sc


def _soft_search(a, L, m, n, time_limit, temp0=1.0, temp1=500.0):
    """Iterative soft gradient descent."""
    deadline = time.monotonic() + float(time_limit)
    a = _normalize(a)
    best_a = a.copy()
    best_sc = _score(a, L, m, n)
    temp = float(temp0)
    eta = 0.08
    no_accept = 0
    it = 0

    while time.monotonic() < deadline:
        a2 = _soft_step(a, L, m, n, temp, eta)
        sc2 = _score(a2, L, m, n)
        if sc2 < best_sc - 1e-12:
            best_a = a2.copy()
            best_sc = sc2
            no_accept = 0
        elif sc2 < _score(a, L, m, n) - 1e-12:
            no_accept = 0
        else:
            no_accept += 1
            if no_accept > 100:
                # Reheat
                rng_local = np.random.default_rng(hash((n, it)) & 0xFFFFFFFF)
                prior = _normalize(rng_local.random(n) + 0.1)
                a = _normalize(0.7 * a + 0.3 * prior)
                no_accept = 0
                temp = max(float(temp0), temp * 0.5)
                eta = 0.08

        a = a2
        it += 1
        if it % 50 == 0:
            temp = min(float(temp1), temp * 1.3)
        if it % 200 == 0:
            if no_accept > 50:
                eta = max(0.02, eta * 0.8)
            else:
                eta = min(0.5, eta * 1.1)

    return best_a, best_sc


def _random_prior(n, rng):
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0
    z = rng.normal(0.0, 0.4)
    z += rng.normal(0.0, 0.6) * x
    z += rng.normal(0.0, 0.5) * (x * x - 1.0 / 3.0)
    kmax = int(rng.integers(2, 10))
    for k in range(1, kmax + 1):
        amp = rng.uniform(0.03, 0.5) / max(1.0, math.sqrt(float(k)))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos((2.0 * np.pi * k) * t + ph)
    noise = rng.normal(0.0, 0.1, n)
    if n > 5:
        noise = np.convolve(noise, np.ones(5) / 5.0, mode="same")
    z += noise
    p = np.exp(z - float(np.max(z)))
    p = np.maximum(p, 1e-300)
    return _normalize(p)


def _shape_candidates(n, rng):
    out = []
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0

    # Outward-heavy shapes (reduce center peak)
    for alpha in (0.3, 0.5, 0.7, 1.0, 1.5, 2.0):
        z = -alpha * (t - 0.5) ** 2
        p = np.exp(z - float(np.max(z)))
        p = np.maximum(p, 1e-300)
        out.append(_normalize(p))

    # Two humps (reduce center)
    for c in (0.2, 0.25, 0.3, 0.35):
        z = -3.0 * (t - c) ** 2 - 3.0 * (t - (1 - c)) ** 2
        p = np.exp(z - float(np.max(z)))
        p = np.maximum(p, 1e-300)
        out.append(_normalize(p))

    # Sine-based
    for amp in (0.5, 0.8, 1.0, 1.5):
        for ph in (0.0, np.pi / 3, 2 * np.pi / 3, np.pi):
            z = amp * np.sin(2 * np.pi * t + ph)
            p = np.exp(z - float(np.max(z)))
            p = np.maximum(p, 1e-300)
            out.append(_normalize(p))

    # Random priors
    for _ in range(8):
        out.append(_random_prior(n, rng))

    # Uniform
    out.append(_normalize(np.ones(n)))

    return out


def _internal_exact(seq):
    if not isinstance(seq, list) or not seq:
        return float(np.inf)
    for x in seq:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return float(np.inf)
        if np.isnan(x) or np.isinf(x):
            return float(np.inf)
    a = [max(0.0, min(1000.0, float(x))) for x in seq]
    n = len(a)
    s = sum(a)
    if s < 0.01:
        return float(np.inf)
    b = np.convolve(a, a)
    mb = max(b)
    return float(2.0 * n * mb / (s * s))


def _exact_score(seq):

        val = evaluate_sequence(seq)

        val = _internal_exact(seq)
