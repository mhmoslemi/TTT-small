    """Identify peak positions and reduce contributing pairs."""

    # Find top positions
    thresh = np.percentile(c, 100 - num_peaks * 100 / m)
    peak_positions = np.where(c >= thresh)[0]

def _normalize(a, eps=1e-300):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a.fill(1.0)
        s = float(a.sum())
    a /= s
    a = np.maximum(a, eps)
    a /= float(a.sum())

def _fft_size(m):
    """Smallest power of 2 >= m."""
    return 1 << (m - 1).bit_length()


def _conv_self(a, L, m):
    """Compute linear self-convolution of a (length n) via FFT of size L."""
    n = len(a)
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    c = np.ascontiguousarray(c)
    np.maximum(c, 0.0, out=c)
    return c


def _compute_grad(a, w, Lg, n):
    """
    Compute gradient direction (w * a_rev)[n-1 : n-1+n] 
    which equals (w cross-correlation a)[0 : n].
    w has length 2n-1, a has length n.
    Need Lg >= (2n-1) + n - 1 = 3n-2 for linear cross-correlation.
    """
    a_rev = a[::-1]
    W = np.fft.rfft(w, Lg)
    A = np.fft.rfft(a_rev, Lg)
    g = np.fft.irfft(W * A, Lg)[n-1 : 2*n-1]
    g = np.ascontiguousarray(g)
    np.maximum(g, 0.0, out=g)
    return g
