import numpy as np
from scipy.signal import correlate
import time

def _autocorr_fft(f):
    n = len(f)
    size = 1 << (2 * n - 2).bit_length()
    F = np.fft.rfft(f, size)
    power = F * np.conj(F)
    return np.fft.irfft(power, size)[:2 * n - 1]

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _subgrad_shift(f, k, n):
    """
    Compute the gradient of conv(f,f)[k] w.r.t. f.
    Result: 2 * f[k-j] for j where 0 <= k-j < n, else 0.
    """
    g = np.zeros(n)
    j_lo = max(0, k - n + 1)
    j_hi = min(n - 1, k)
    if j_lo <= j_hi:
        # g[j] = 2 * f[k-j] for j in [j_lo, j_hi]
        # k-j ranges from k-j_hi to k-j_lo
        g[j_lo:j_hi+1] = 2.0 * f[k-j_hi:k-j_lo+1][::-1]
    return g

def _objective(f):
    """Compute 2*n*max_conv/(sum)^2"""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _autocorr_fft(f)
    return 2.0 * n * c.max() / (s * s)

Step function optimizer using projected subgradient method with spectral
initialization, exact active-set gradients, and adaptive coordinate descent.

import numpy as np
import time

def _autocorr_fft(f):
    n = len(f)
    size = 1 << (2 * n - 2).bit_length()
    F = np.fft.rfft(f, size)
    power = F * np.conj(F)
    c = np.fft.irfft(power, size)
    return c[:2 * n - 1]

def _project_simplex(v):
    n = len(v)
    s = v.sum()
    if s < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _eval(f):
    return evaluate_sequence(f.tolist())

def _spectral_init(n, num_iters, target_appr, rng):
    """
    Spectral equalization initialization.
    target_appr: 'flat', 'tapered', or 'random_phase'
    """
    size = 1 << (2 * n - 2).bit_length()
    
    if target_appr == 'flat':
        # Target: flat spectrum (ideal for flat autocorrelation)
        A = np.fft.rfft(np.ones(size))
        # Set DC component
        A[0] = n  # so that IFFT has sum = n (we'll normalize later)
        # Keep it flat
        for it in range(num_iters):
            f = np.fft.irfft(A, size)[:n]
            f = np.maximum(f, 0)
            s = f.sum()
            if s < 1e-10:
                f = np.ones(n)
            else:
                f /= s
            A = np.fft.rfft(f, size)
            # Modify amplitude to target
            target = np.ones(size // 2 + 1) * np.sqrt(n / (size // 2))
            target[0] = n  # DC
            A = target * np.exp(1j * np.angle(A))
        return f
    elif target_appr == 'tapered':
        # Tapered spectrum: more energy at low frequencies
        A = np.fft.rfft(np.ones(size))
        freqs = np.arange(size // 2 + 1)
        taper = np.exp(-freqs / (size * 0.1))
        taper[0] = 1.0
        target = n * taper / taper.sum() * (size // 2 + 1)
        target[0] = n
        for it in range(num_iters):
            f = np.fft.irfft(A, size)[:n]
            f = np.maximum(f, 0)
            s = f.sum()
            if s < 1e-10:
                f = np.ones(n)
            else:
                f /= s
            A = np.fft.rfft(f, size)
            angles = np.angle(A)
            A = target * np.exp(1j * angles)
        return f
    elif target_appr == 'random_phase':
        # Random phase in frequency domain
        f = np.ones(n) / n
        for it in range(num_iters):
            A = np.fft.rfft(f, size)
            angles = np.angle(A)
            amplitudes = np.abs(A)
            # Add some randomness
            noisy_amp = amplitudes * (1 + 0.1 * rng.standard_normal(size // 2 + 1))
            noisy_amp = np.maximum(noisy_amp, 0)
            A = noisy_amp * np.exp(1j * angles)
            f = np.fft.irfft(A, size)[:n]
            f = np.maximum(f, 0)
            s = f.sum()
            if s < 1e-10:
                f = np.ones(n)
            else:
                f /= s
        return f
    return np.ones(n) / n

def _autocorr_fft(f):
    """Full linear autocorrelation via FFT."""
    n = len(f)
    size = 1 << (2 * n - 2).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * np.conj(F), size)[:2 * n - 1]

def _project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _subgradient(f, c, active_frac=0.05):
    """
    Compute subgradient of max(c) w.r.t. f, averaged over near-max lags.
    c = convolve(f, f), grad_k[j] = 2*f[k-j] if 0<=k-j<n else 0.
    """

    cmax = c.max()
    threshold = cmax * (1 - active_frac)
    active = np.where(c >= threshold)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    # Average subgradients over active lags
    grad = np.zeros(n)
    for k in active:
        k = int(k)
        j_lo = max(0, k - n + 1)
        j_hi = min(n - 1, k)
        if j_lo <= j_hi:

    grad *= 2.0 / len(active)
    return grad

        grad = _subgradient(f, c, active_frac=0.02)

    """

    active_lags = np.where(c >= threshold)[0]

    stagnation = 0
    
    while time.monotonic() < deadline and stagnation < 50:
        c = _autocorr_fft(f)
        cmax = c.max()

        k_star = int(np.argmax(c))

        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        
        if j_lo > j_hi:
            break

        j_idx = j_lo + int(np.argmax(products))
        
        # Find a position to receive mass (low contribution to all active lags)
        contributions = np.zeros(n)
        for k in active_lags[:10]:  # Top 10 active lags
            k = int(k)
            jlo = max(0, k - n + 1)
            jhi = min(n - 1, k)
            if jlo <= jhi:

        contributions[j_idx] = np.inf
        j_recv = int(np.argmin(contributions))

            stagnation += 1
            continue

        best_local = cmax
        best_frac = None
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.5]:
            amt = f[j_idx] * frac
            if amt < 1e-15:
                break
            f_try = f.copy()
            f_try[j_idx] -= amt
            f_try[j_recv] += amt
            f_try /= f_try.sum()
            c_try = _autocorr_fft(f_try)
            if c_try.max() < best_local - 1e-12:
                best_local = c_try.max()
                best_frac = frac

            f[j_idx] -= amt
            f[j_recv] += amt
            f /= f.sum()
            improvements += 1
            stagnation = 0

            stagnation += 1

            cmax = c.max()
            active_lags = np.where(c >= cmax * 0.95)[0]

        score = _objective_val(f)

    momentum = 0.7
    v = np.zeros(n)

    iter_count = 0
    no_improve = 0
    eval_interval = 300
    stuck_threshold = 30

    while time.monotonic() < deadline:
        c = _autocorr_fft(f)
        cmax = c.max()
        cur_obj = 2.0 * n * cmax

        # Compute subgradient over active lags
        grad = _subgradient(f, c, active_frac=0.03)

        # Gradient step with momentum
        v = momentum * v + (1.0 - momentum) * grad
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-14:
            break

        # Adaptive step size: use 1/L as base, adapt based on progress
        eta = min(eta_max, max(eta * 1.01, 0.1 / L))

        # Take step
        f_new = f - eta * v
        f_new = _project_simplex(f_new)

        # Evaluate and decide
        c_new = _autocorr_fft(f_new)
        obj_new = 2.0 * n * c_new.max()

        if obj_new <= cur_obj + 1e-12:
            f = f_new
            c = c_new
            cmax = c.max()
            # Increase step size slightly if making good progress
            if obj_new < cur_obj - 1e-10:
                eta = min(eta * 1.05, eta_max)
        else:
            # Step too large, try smaller
            for _ in range(5):
                eta *= 0.5
                f_new = f - eta * v
                f_new = _project_simplex(f_new)
                c_new = _autocorr_fft(f_new)
                obj_new = 2.0 * n * c_new.max()
                if obj_new <= cur_obj + 1e-12:
                    f = f_new
                    c = c_new
                    cmax = c.max()
                    break
            else:
                # If still increasing, take a tiny step in negative gradient direction
                f = f - 0.01 * eta * v
                f = _project_simplex(f)

        iter_count += 1

        # Periodic evaluation and tracking
        if iter_count % eval_interval == 0:
            score = _obj_from_c(_autocorr_fft(f), n, 1.0)
            if score < best_score - 1e-9:
                best_score = score
                best_f = f.copy()
                no_improve = 0
                if verbose and score < 1.55:
                    print(f"  {label} iter {iter_count}: {score:.8f}")
            else:
                no_improve += 1

            if no_improve >= stuck_threshold:
                # Perturbation to escape local minimum
                scale = 0.02 / np.sqrt(n)
                f = f + scale * rng.standard_normal(n)
                f = np.maximum(f, 0)
                f /= f.sum()
                no_improve = 0
                momentum = min(momentum + 0.05, 0.95)

    # Final check
    score = _objective_val(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()

    return best_f, best_score


def _polish(f, deadline, rng):
    """
    Targeted coordinate descent: transfer mass from peak-contributing
    positions to low-contribution positions.
    """
    n = len(f)
    f = f.copy().astype(np.float64)
    best_score = _objective_val(f)
    best_f = f.copy()

    c = _autocorr_fft(f)
    cmax = c.max()

    stagnation = 0
    max_stagnation = 100

    while time.monotonic() < deadline and stagnation < max_stagnation:
        c = _autocorr_fft(f)
        cmax = c.max()

        # Identify all near-max lags
        threshold = cmax * 0.9
        active_lags = np.where(c >= threshold)[0]

        improvement_found = False

        # For each active lag, find the best donor and try transfer
        for k_star in active_lags[:5]:
            k_star = int(k_star)
            j_lo = max(0, k_star - n + 1)
            j_hi = min(n - 1, k_star)
            if j_lo > j_hi:
                continue

            # Find pair with maximum product for this lag
            contrib = f[j_lo:j_hi + 1] * f[k_star - j_hi:k_star - j_lo + 1][::-1]
            j_donate = j_lo + int(np.argmax(contrib))

            if f[j_donate] < 1e-12:
                continue

            # Compute total contribution of each position to all active lags
            total_contrib = np.zeros(n)
            for k in active_lags[:8]:
                k = int(k)
                jlo = max(0, k - n + 1)
                jhi = min(n - 1, k)
                if jlo <= jhi:
                    total_contrib[jlo:jhi + 1] += f[jlo:jhi + 1] * f[k - jhi:k - jlo + 1][::-1]

            # Find best receiver: low total contribution, not the donor
            total_contrib[j_donate] = np.inf
            j_recv = int(np.argmin(total_contrib))

            if j_recv == j_donate:
                continue

            # Try transfers
            for frac in [0.05, 0.1, 0.2, 0.3, 0.5]:
                amt = f[j_donate] * frac
                if amt < 1e-15:
                    break
                f_try = f.copy()
                f_try[j_donate] -= amt
                f_try[j_recv] += amt
                f_try /= f_try.sum()
                c_try = _autocorr_fft(f_try)
                if c_try.max() < cmax - 1e-13:
                    f = f_try
                    cmax = c_try.max()
                    improvement_found = True
                    break

            if improvement_found:
                stagnation = 0
                break

        if not improvement_found:
            # Try random 2-point moves
            found = False
            for _ in range(20):
                i = int(rng.integers(0, n))
                jx = int(rng.integers(0, n))
                if i == jx:
                    continue
                for frac in [0.05, 0.1, 0.2]:
                    amt = f[i] * frac
                    if amt < 1e-15:
                        break
                    f_try = f.copy()
                    f_try[i] -= amt
                    f_try[jx] += amt
                    f_try /= f_try.sum()
                    c_try = _autocorr_fft(f_try)
                    if c_try.max() < cmax - 1e-13:
                        f = f_try
                        cmax = c_try.max()
                        found = True
                        improvement_found = True
                        break
                if found:
                    break

            if not improvement_found:
                stagnation += 1
                # Perturb
                if stagnation % 20 == 0:
                    idx = int(rng.integers(0, n))
                    jx = int(rng.integers(0, n))
                    if idx != jx:
                        amt = f[idx] * 0.1
                        f[idx] -= amt
                        f[jx] += amt
                        f = np.maximum(f, 0)
                        f /= f.sum()

        # Track best
        if (improvement_found) or (stagnation % 50 == 0):
            score = _obj_from_c(_autocorr_fft(f), n, 1.0)
            if score < best_score:
                best_score = score
                best_f = f.copy()

    # Final
    score = _objective_val(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()

    return best_f, best_score


def _spectral_init(n, size, mode, num_iters=300):
    """Spectral equalization initialization."""
    if mode == 'flat':
        # Target: flat magnitude spectrum
        A = np.fft.rfft(np.ones(n), size)
        for _ in range(num_iters):
            f = np.fft.irfft(A, size)[:n]
            f = np.maximum(f, 0)
            s = f.sum()
            if s < 1e-10:
                return np.ones(n) / n
            f /= s
            A = np.fft.rfft(f, size)
            amp = np.abs(A)
            phase = np.angle(A)
            target = np.full(len(amp), amp.mean())
            target[0] = amp[0]
            A = target * np.exp(1j * phase)
        return f
    elif mode == 'tapered':
        # Target: exponentially tapered spectrum
        A = np.fft.rfft(np.ones(n), size)
        freqs = np.arange(len(A), dtype=np.float64)
        for _ in range(num_iters):
            f = np.fft.irfft(A, size)[:n]
            f = np.maximum(f, 0)
            s = f.sum()
            if s < 1e-10:
                return np.ones(n) / n
            f /= s
            A = np.fft.rfft(f, size)
            amp = np.abs(A)
            phase = np.angle(A)
            target = amp * np.exp(-freqs / (len(A) * 0.2))
            target[0] = amp[0]
            A = target * np.exp(1j * phase)
        return f
    elif mode == 'raised_cosine':
        # Target: raised cosine spectrum
        A = np.fft.rfft(np.ones(n), size)
        freqs = np.arange(len(A), dtype=np.float64)
        fcuts = [0.3, 0.7]
        for _ in range(num_iters):
            f = np.fft.irfft(A, size)[:n]
            f = np.maximum(f, 0)
            s = f.sum()
            if s < 1e-10:
                return np.ones(n) / n
            f /= s
            A = np.fft.rfft(f, size)
            amp = np.abs(A)
            phase = np.angle(A)
            # Raised cosine shaping
            target = np.zeros(len(freqs))
            for i, fc in enumerate(fcuts):
                target += np.exp(-((freqs - fc * len(freqs)) / (len(freqs) * 0.15)) ** 2)
            target += 0.1
            target[0] = amp[0]
            A = target * np.exp(1j * phase)
        return f
    return np.ones(n) / n


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the non-negative sequence minimizing 2*n*max(f*f)/(sum f)^2.
    Uses projected subgradient with exact active-set gradients,
    spectral initialization, and coordinated polish.
    """
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    deadline = t_start + budget_s - 30.0

    f_init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(f_init)
    size = 1 << (2 * n - 2).bit_length()
    print(f"n = {n}, FFT size = {size}")

    # Generate diverse starting points
    starts = {}

    # 1. Given sequence normalized
    f0 = np.maximum(f_init, 0)
    s0 = f0.sum()
    if s0 > 1e-10:
        starts['given'] = f0 / s0
    else:
        starts['given'] = np.ones(n) / n

    # 2. Uniform
    starts['uniform'] = np.ones(n) / n

    # 3-5. Spectral initializations
    starts['flat_spec'] = _spectral_init(n, size, 'flat', 200)
    starts['tapered_spec'] = _spectral_init(n, size, 'tapered', 200)
    starts['raised_cos'] = _spectral_init(n, size, 'raised_cosine', 200)

    # 6. Bimodal
    j = np.arange(n, dtype=np.float64)
    f = np.exp(-((j - 0.25 * n) / (0.1 * n)) ** 2) + np.exp(-((j - 0.75 * n) / (0.1 * n)) ** 2) + 0.01
    starts['bimodal'] = f / f.sum()

    # 7. Triangular (peak at center)
    center = (n - 1) / 2.0
    f = 1.0 - np.abs(j - center) / center
    f = np.maximum(f, 0.01)
    starts['triangular'] = f / f.sum()

    # 8. U-shaped
    f = 0.1 + ((j - center) / center) ** 4
    starts['u_shaped'] = f / f.sum()

    # 9. Sine modulated
    f = 1.0 + 0.5 * np.cos(2 * np.pi * 5 * j / n)
    f = np.maximum(f, 0.05)
    starts['sine5'] = f / f.sum()

    # 10. Random Dirichlet
    starts['dirichlet'] = rng.dirichlet(np.ones(n) * 0.3)

    # 11. Random exponential
    f = rng.exponential(1.0, size=n)
    starts['exponential'] = f / f.sum()

    # Evaluate and sort starts
    start_scores = []
    for name, f0 in starts.items():
        f0 = np.maximum(f0, 0)
        s = f0.sum()
        if s < 1e-10:
            f0 = np.ones(n) / n
        else:
            f0 = f0 / s
        sc = _objective_val(f0)
        start_scores.append((sc, name, f0))

    start_scores.sort(key=lambda x: x[0])
    print("\nStarting point scores:")
    for sc, name, _ in start_scores:
        print(f"  {name}: {sc:.8f}")

    best_score = start_scores[0][0]
    best_f = start_scores[0][2].copy()

    # Phase 1: Optimize from top starts
    # Time allocation for each start (leaving time for polish)
    total_time = budget_s - 60
    opt_time = total_time * 0.65
    polish_time = total_time * 0.25
    final_time = total_time * 0.10

    num_starts = min(5, len(start_scores))
    weights = [0.35, 0.25, 0.20, 0.12, 0.08][:num_starts]
    # Normalize weights
    wsum = sum(weights)
    weights = [w / wsum for w in weights]

    for idx in range(num_starts):
        if time.monotonic() > deadline - polish_time - final_time:
            break
        sc, name, f0 = start_scores[idx]
        t_alloc = opt_time * weights[idx]
        t_deadline = time.monotonic() + t_alloc
        print(f"\nOptimizing from '{name}' (score={sc:.6f}, budget={t_alloc:.0f}s)")

        f_opt, sc_opt = _optimize(f0, t_deadline, rng, verbose=True, label=name)

        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  >>> New best: {best_score:.8f}")

    # Phase 2: Polish the best
    print(f"\nPolishing (score={best_score:.8f})")
    polish_deadline = time.monotonic() + polish_time * 0.8
    polish_deadline = min(polish_deadline, deadline - final_time)
    f_pol, sc_pol = _polish(best_f, polish_deadline, rng)
    if sc_pol < best_score:
        best_score = sc_pol
        best_f = f_pol.copy()
        print(f"  After polish: {best_score:.8f}")

    # Phase 3: Final optimization from polished solution
    if time.monotonic() < deadline - 10:
        print(f"\nFinal optimization (score={best_score:.8f})")
        f_fin, sc_fin = _optimize(best_f, deadline, rng, verbose=True, label="final")
        if sc_fin < best_score:
            best_score = sc_fin
            best_f = f_fin.copy()
            print(f"  Final: {best_score:.8f}")

        # One more polish
        if time.monotonic() < deadline:
            f_pol2, sc_pol2 = _polish(best_f, deadline, rng)
            if sc_pol2 < best_score:
                best_score = sc_pol2
                best_f = f_pol2.copy()
                print(f"  After final polish: {best_score:.8f}")

    # Final validation
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = _eval(best_f)
    if final_score < best_score or not np.isfinite(best_score):
        best_score = final_score

    elapsed = time.monotonic() - t_start
    print(f"\n=== FINAL SCORE: {best_score:.8f} (elapsed: {elapsed:.1f}s) ===")

    return [float(x) for x in best_f]
