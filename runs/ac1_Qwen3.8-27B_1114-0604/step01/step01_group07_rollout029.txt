def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(c.max())
    if not np.isfinite(mx):
        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))


def _make_sine(n, amp, mode, phase=0.5):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / float(n)
    a = 1.0 + float(amp) * np.sin(float(mode) * theta)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-10:
        a = np.ones(n, dtype=np.float64)
        s = n
    return a / s

def _gradient_active(a, c, K):
    """Compute gradient of max(b_k) approximated by average over top K lags."""
    n = a.size
    if c.size <= K:
        active = np.arange(c.size, dtype=np.int64)
    else:
        active = np.argpartition(c, -K)[-K:]
    
    j = np.arange(n, dtype=np.int64)
    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        idx1 = k - j
        mask1 = (idx1 >= 0) & (idx1 < n)
        idx2 = j - k
        mask2 = (idx2 >= 0) & (idx2 < n)
        if mask1.any():
            grad += np.where(mask1, a[np.clip(idx1, 0, n-1)], 0.0)
        if mask2.any():
            grad += np.where(mask2, a[np.clip(idx2, 0, n-1)], 0.0)
    grad /= float(len(active))
    # Center the gradient (project onto tangent space of simplex)
    grad -= grad.mean()
    return grad

"""
Low-peak nonnegative step-function search using mirror descent with adaptive
active-set gradients on the probability simplex.

Algorithm:
1. Initialize from sinusoidal and multi-sinusoidal families, and parent sequence.
2. Optimize using mirror descent (multiplicative updates) with gradient
   computed from top-K autocorrelation lags (active-set approximation).
3. Adaptive learning rate with periodic evaluation and random restarts.
4. Try multiple K values and lengths within the time budget.
"""

import numpy as np
import time


def _next_pow2(m):
    """Smallest power of two >= m."""
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    """Fast evaluation matching evaluate_sequence (assumes valid input)."""
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(c.max())
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))


def _make_sine(n, amp, mode, phase=0.5):
    """Single sinusoidal shape, normalized to sum=1."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase) / float(n)
    a = 1.0 + float(amp) * np.sin(float(mode) * theta)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def _make_multi_sine(n, amps, modes, phases):
    """Multi-sinusoidal shape, normalized to sum=1."""
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for k in range(len(amps)):
        theta = 2.0 * np.pi * (i + phases[k]) / float(n)
        a += float(amps[k]) * np.sin(float(modes[k]) * theta)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 1e-10:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def _gradient_active(arr, c, K, j_cache):
    """Gradient of max(b_k) approx by average over top-K lags."""
    n = arr.size
    if c.size <= K:
        active = np.arange(c.size, dtype=np.int64)
    else:
        active = np.argpartition(c, -K)[-K:]

    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        k = int(k)
        idx1 = k - j_cache
        mask1 = (idx1 >= 0) & (idx1 < n)
        idx2 = j_cache - k
        mask2 = (idx2 >= 0) & (idx2 < n)
        if mask1.any():
            grad[mask1] += arr[idx1[mask1]]
        if mask2.any():
            grad[mask2] += arr[idx2[mask2]]
    grad /= float(len(active))
    grad -= grad.mean()
    return grad


def _optimize_mirror(a, deadline, K=32, eval_period=15, rng=None):
    """Mirror descent on the simplex. Returns (best_sequence, best_score)."""
    if rng is None:
        rng = np.random.RandomState(0)

    n = a.size
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.clip(arr, 0.0, None)
    s = float(arr.sum())
    if s < 1e-10:
        return arr, np.inf
    arr = arr / s

    best = arr.copy()
    best_score = _internal_score(best)

    lr = 0.5
    max_lr = 3.0
    min_lr = 0.02
    it = 0
    last_improve = 0
    patience = 300
    j_cache = np.arange(n, dtype=np.int64)

    while time.monotonic() < deadline:
        it += 1

        c = _fft_convolve(arr)

        if it % eval_period == 0:
            mx = float(c.max())
            s = float(arr.sum())
            sc = 2.0 * n * mx / (s * s)

            if sc < best_score - 1e-12:
                best = arr.copy()
                best_score = sc
                last_improve = it
                lr = min(max_lr, lr * 1.02)
            else:
                if sc > best_score * 1.003:
                    lr = max(min_lr, lr * 0.85)

                if it - last_improve > patience:
                    noise_scale = 0.03 + 0.02 * min(it / 5000.0, 1.0)
                    noise = rng.normal(0, noise_scale, n)
                    arr = arr * np.exp(noise)
                    arr = np.clip(arr, 1e-12, None)
                    arr = arr / arr.sum()
                    lr = max(min_lr, lr * 0.5)
                    last_improve = it

        grad = _gradient_active(arr, c, K, j_cache)

        exponent = -lr * grad
        exponent = np.clip(exponent, -3.0, 3.0)
        arr = arr * np.exp(exponent)
        arr = np.clip(arr, 1e-15, None)
        arr = arr / arr.sum()

    return best, best_score
