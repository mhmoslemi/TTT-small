def full_conv(a):
    """Compute full linear convolution."""
    if len(a) == 0:
        return np.zeros(0, dtype=float)
    n = len(a)
    total_size = 2*n - 1
    conv_size = 2*n - 1
    # FFT-based convolution with minimal padding
    size = 1 << total_size.bit_length()
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)
    return conv[:conv_size]
