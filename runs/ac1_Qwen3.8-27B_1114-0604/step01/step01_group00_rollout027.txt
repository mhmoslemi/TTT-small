"""
Step function optimizer using multi-phase Adam with multi-peak gradient,
coordinate descent, and simulated annealing.
"""

import numpy as np
import time
from scipy.signal import correlate

def _conv_fft(f):
    """Compute linear convolution f*f via FFT."""
    n = len(f)
    size = 1 << (2*n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2*n-1]

def _multi_peak_gradient(f, c, T, top_k=1):
    """
    Compute gradient targeting top_k peaks of the convolution.
    Returns (gradient, peak_weights).
    """
    n = len(f)
    m = len(c)  # 2n-1
    
    if top_k >= m:
        # Uniform weighting
        p = np.ones(m) / m
    else:
        # Select top_k elements and assign softmax weights over them
        # But also include some mass on the rest
        idx = np.argpartition(c, -top_k)[-top_k:]
        p = np.zeros(m)
        # Softmax over top_k
        c_top = c[idx]
        c_top_norm = c_top - c_top.max()
        w = np.exp(c_top_norm / T)
        w /= w.sum()
        p[idx] = w
    
    # Gradient: grad[j] = 2 * sum_k p[k] * f[k-j] (if 0 <= k-j < n)
    # This is 2 * correlate(p, f, mode='valid')
    grad = 2.0 * correlate(p, f, mode='valid')
    return grad, p

def _adam_update(f, grad, state, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer step with simplex projection."""
    t = state['t'] + 1
    state['t'] = t
    
    state['m'] = beta1 * state['m'] + (1 - beta1) * grad
    state['v'] = beta2 * state['v'] + (1 - beta2) * grad**2
    
    m_hat = state['m'] / (1 - beta1**t)
    v_hat = state['v'] / (1 - beta2**t)
    
    f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
    
    # Project onto simplex {x >= 0, sum(x) = 1}
    f_new = _project_simplex(f_new)
    
    return f_new, state

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _eval(f):

def _coordinate_descent(f, max_iter=10000, deadline=None):
    """
    Coordinate descent: try transferring mass between pairs of coordinates.
    Focus on reducing the peak of the convolution.
    """
    n = len(f)
    f = f.copy()
    c = _conv_fft(f)
    peak = c.max()
    best_score = _eval(f)
    best_f = f.copy()
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Find the peak lag
        k_star = int(np.argmax(c))
        
        # For each coordinate j, compute the "contribution" to the peak:
        # If we decrease f[j], the peak changes by -2*f[k_star-j] (if valid)
        # If we increase f[j], the peak changes by +2*f[k_star-j] (if valid)
        
        # We want to decrease f[j] where f[k_star-j] is large
        # and increase f[i] where f[k_star-i] is small
        
        # Compute contribution array
        contrib = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                contrib[j] = f[idx]
        
        # Source: coordinate with highest contribution (to decrease)
        # Destination: coordinate with lowest contribution (to increase)
        
        # Try top few sources and destinations
        sources = np.argsort(contrib)[::-1][:5]
        destinations = np.argsort(contrib)[:5]
        
        improved = False
        for s_idx in sources:
            for d_idx in destinations:
                if s_idx == d_idx:
                    continue
                for frac in [0.01, 0.02, 0.05, 0.1]:
                    amt = f[s_idx] * frac
                    f_try = f.copy()
                    f_try[s_idx] -= amt
                    f_try[d_idx] += amt
                    if f_try[s_idx] < 0:
                        continue
                    f_try /= f_try.sum()  # Should already be 1
                    c_try = _conv_fft(f_try)
                    if c_try.max() < peak - 1e-14:
                        f = f_try
                        c = c_try
                        peak = c.max()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        
        if not improved:
            # Try random perturbations
            for _ in range(20):
                i = int(np.random.randint(n))
                j = int(np.random.randint(n))
                if i == j:
                    continue
                frac = np.random.uniform(0.01, 0.1)
                amt = min(f[i], f[j]) * frac
                f_try = f.copy()
                f_try[i] -= amt
                f_try[j] += amt
                if f_try[i] < 0 or f_try[j] < 0:
                    continue
                f_try /= f_try.sum()
                c_try = _conv_fft(f_try)
                if c_try.max() < peak - 1e-14:
                    f = f_try
                    c = c_try
                    peak = c.max()
                    improved = True
                    break
        
        if not improved:
            # Stuck, break
            break
        
        if it % 1000 == 0:
            score = _eval(f)
            if score < best_score:
                best_score = score
                best_f = f.copy()
    
    score = _eval(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score

def _adam_optimize(f, deadline, lr_init=0.1, lr_end=0.001, top_k_init=1, top_k_end=5, seed=42):
    """
    Adam optimization with adaptive top-k and learning rate.
    """
    rng = np.random.default_rng(seed)
    n = len(f)
    f = f.copy()
    
    state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
    
    c = _conv_fft(f)
    peak0 = c.max()
    T_init = peak0
    T_end = peak0 * 1e-6
    
    best_score = _eval(f)
    best_f = f.copy()
    
    n_iters = 100000
    patience = 0
    max_patience = 2000
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        progress = it / n_iters
        
        # Anneal temperature
        T = T_init * (T_end / T_init) ** progress
        
        # Adaptive top_k: start with 1, increase to top_k_end
        top_k = int(1 + (top_k_end - 1) * progress)
        
        # Learning rate schedule
        lr = lr_init * (lr_end / lr_init) ** (progress ** 0.5)
        
        # Compute gradient
        c = _conv_fft(f)
        grad, p = _multi_peak_gradient(f, c, T, top_k=top_k)
        
        # Adam step
        f, state = _adam_update(f, grad, state, lr)
        
        # Periodic evaluation
        if it % 5000 == 0 or it == n_iters - 1:
            score = _eval(f)
            if score < best_score - 1e-7:
                best_score = score
                best_f = f.copy()
                patience = 0
            else:
                patience += 1
            
            if patience > max_patience:
                # Restart with perturbation
                f = best_f.copy()
                f += 0.01 * rng.standard_normal(n)
                f = np.maximum(f, 0)
                f /= f.sum()
                state = {'t': 0, 'm': np.zeros(n), 'v': np.zeros(n)}
                patience = 0
    
    score = _eval(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score

Actually, I just realized something. Let me re-examine the gradient more carefully.

We want to minimize `max_k c[k]` where `c = conv(f, f)` and we're on the simplex.

The subgradient of `max_k c[k]` at f is `∂c[k*]/∂f` for any k* in argmax.

`∂c[k*]/∂f[j] = 2 * f[k*-j]` (if 0 ≤ k*-j < n, else 0)

So the subgradient is a vector where component j is `2 * f[k*-j]` (or 0).

For the softmax approximation with temperature T:
`grad[j] = 2 * sum_k p(k) * f[k-j]` where p(k) ∝ exp(c[k]/T)

This is what we've been computing. Good.

Now, for Adam to work well, the gradient should be informative. The issue with the pure max (T→0) gradient is that it only depends on f at positions k*-j, which means it's a "permuted" version of f. This can lead to oscillations.

The softmax with moderate T smooths this out and provides a more informative gradient.

Let me think about the optimal T. If T is too small, we're essentially doing subgradient descent with the argmax. If T is too large, we're averaging over all lags and the gradient becomes less focused on the peak.

A good heuristic: T should be on the order of the "peak height" above the average. If the peak is at height H and the average is at height A, then T ≈ (H - A) / ln(K) where K is the number of "significant" lags.

"""
Multi-phase optimizer using Adam with adaptive multi-peak gradient,
coordinate descent, and simulated annealing restarts.
"""

import numpy as np
import time
from scipy.signal import correlate


def _conv_fft(f):
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _grad_from_conv(f, c, T, top_k=1):
    """Compute gradient using softmax over top-k peaks."""
    n = len(f)
    m = len(c)
    
    if top_k >= m:
        p = np.ones(m) / m
    elif top_k <= 1:
        # Pure softmax
        cs = c / T
        cs -= cs.max()
        p = np.exp(cs)
        p /= p.sum()
    else:
        # Hybrid: softmax over top-k, uniform over rest
        idx = np.argpartition(c, -top_k)[-top_k:]
        rest = np.setdiff1d(np.arange(m), idx)
        
        c_top = c[idx]
        cs = c_top / T
        cs -= cs.max()
        w = np.exp(cs)
        w /= w.sum()
        
        p = np.zeros(m)
        p[idx] = w * 0.8  # 80% weight on top-k
        p[rest] = 0.2 / len(rest)  # 20% spread over rest
    
    grad = 2.0 * correlate(p, f, mode='valid')
    return grad


def _project_simplex(v):
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _eval(f):
    return evaluate_sequence(f.tolist())


def _adam_optimize(f, deadline, seed=0, lr_init=0.05, lr_end=0.0005,
                   top_k_init=1, top_k_end=10, T_ratio=1.0, T_end_ratio=1e-6):
    """Adam optimization with adaptive parameters."""
    rng = np.random.default_rng(seed)
    n = len(f)
    f = f.copy()
    
    c = _conv_fft(f)
    peak0 = c.max()
    avg0 = c.mean()
    spread0 = max(peak0 - avg0, 1e-10)
    
    T_init = spread0 * T_ratio
    T_end = spread0 * T_end_ratio
    
    state_m = np.zeros(n)
    state_v = np.zeros(n)
    t = 0
    
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    
    best_score = _eval(f)
    best_f = f.copy()
    patience = 0
    max_patience = 5000
    
    n_max_iters = 200000
    
    for it in range(n_max_iters):
        if time.monotonic() > deadline:
            break
        
        progress = min(1.0, it / n_max_iters)
        
        # Schedule
        T = T_init * (T_end / T_init) ** progress
        lr = lr_init * (lr_end / lr_init) ** (progress ** 0.3)
        top_k = max(1, int(top_k_init + (top_k_end - top_k_init) * progress))
        
        # Compute gradient
        c = _conv_fft(f)
        grad = _grad_from_conv(f, c, T, top_k=top_k)
        
        # Adam update
        t += 1
        state_m = beta1 * state_m + (1 - beta1) * grad
        state_v = beta2 * state_v + (1 - beta2) * grad ** 2
        m_hat = state_m / (1 - beta1 ** t)
        v_hat = state_v / (1 - beta2 ** t)
        
        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = _project_simplex(f_new)
        
        # Periodic evaluation
        if it % 3000 == 0:
            score = _eval(f)
            if score < best_score - 1e-7:
                best_score = score
                best_f = f.copy()
                patience = 0
            else:
                patience += 1
            
            if patience > max_patience:
                # Restart from best with perturbation
                f = best_f + 0.005 * rng.standard_normal(n)
                f = np.maximum(f, 0)
                f /= f.sum()
                state_m = np.zeros(n)
                state_v = np.zeros(n)
                t = 0
                patience = 0
    
    score = _eval(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score


def _coord_descent(f, deadline, rng=None):
    """Coordinate descent: transfer mass to reduce peak."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(f)
    f = f.copy()
    c = _conv_fft(f)
    peak = c.max()
    best_score = _eval(f)
    best_f = f.copy()
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        k_star = int(np.argmax(c))
        
        # Contribution of each coordinate to the peak
        contrib = np.zeros(n)
        valid = (k_star >= 0) & (k_star < 2 * n - 1)
        if valid:
            # f[k_star - j] for valid j
            lo = max(0, k_star - n + 1)
            hi = min(n - 1, k_star)
            for j in range(lo, hi + 1):
                contrib[j] = f[k_star - j]
        
        # Try to move mass from high-contrib to low-contrib
        if contrib.max() > 0:
            sources = np.argsort(contrib)[::-1][:3]
            dests = np.argsort(contrib)[:3]
            
            improved = False
            for si in sources:
                for di in dests:
                    if si == di:
                        continue
                    for frac in [0.02, 0.05, 0.1]:
                        amt = f[si] * frac
                        if amt < 1e-15:
                            continue
                        f_try = f.copy()
                        f_try[si] -= amt
                        f_try[di] += amt
                        c_try = _conv_fft(f_try)
                        if c_try.max() < peak - 1e-14:
                            f = f_try
                            c = c_try
                            peak = c.max()
                            improved = True
                            break
                    if improved:
                        break
                if improved:
                    break
            
            if improved:
                if it % 500 == 0:
                    score = _eval(f)
                    if score < best_score:
                        best_score = score
                        best_f = f.copy()
                continue
        
        # No structured improvement found, try random
        for _ in range(30):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            frac = rng.uniform(0.01, 0.1)
            amt = min(f[i], f[j]) * frac
            if amt < 1e-15:
                continue
            f_try = f.copy()
            f_try[i] -= amt
            f_try[j] += amt
            c_try = _conv_fft(f_try)
            if c_try.max() < peak - 1e-14:
                f = f_try
                c = c_try
                peak = c.max()
                break
        else:
            # Truly stuck
            break
    
    score = _eval(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score


def _sa_optimize(f, deadline, seed=0, T_init=0.01, T_end=1e-5, max_steps=100000):
    """Simulated annealing with structuredPerturbations."""
    rng = np.random.default_rng(seed)
    n = len(f)
    f = f.copy()
    current_score = _eval(f)
    best_score = current_score
    best_f = f.copy()
    
    c = _conv_fft(f)
    peak = c.max()
    
    for it in range(max_steps):
        if time.monotonic() > deadline:
            break
        
        progress = it / max_steps
        T = T_init * (T_end / T_init) ** progress
        
        # Generate perturbation
        pert_type = rng.integers(4)
        f_try = f.copy()
        
        if pert_type == 0:
            # Gaussian noise
            f_try += T * 0.1 * rng.standard_normal(n)
            f_try = np.maximum(f_try, 0)
            f_try /= f_try.sum()
        elif pert_type == 1:
            # Swap mass between two random positions
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                amt = min(f[i], f[j]) * rng.uniform(0.05, 0.3)
                f_try[i] -= amt
                f_try[j] += amt
                f_try = np.maximum(f_try, 0)
                f_try /= f_try.sum()
        elif pert_type == 2:
            # Block perturbation
            block_size = int(rng.integers(10, 100))
            start = int(rng.integers(0, n - block_size))
            f_try[start:start+block_size] += T * 0.1 * rng.standard_normal(block_size)
            f_try = np.maximum(f_try, 0)
            f_try /= f_try.sum()
        else:
            # Peak-directed: move mass away from peak contributors
            k_star = int(np.argmax(c))
            contrib = np.zeros(n)
            lo = max(0, k_star - n + 1)
            hi = min(n - 1, k_star)
            for j in range(lo, hi + 1):
                contrib[j] = f[k_star - j]
            
            # Move from high contrib to low contrib
            si = int(rng.choice(np.argsort(contrib)[::-1][:10]))
            di = int(rng.choice(np.argsort(contrib)[:10]))
            if si != di:
                amt = f[si] * rng.uniform(0.05, 0.2)
                f_try[si] -= amt
                f_try[di] += amt
                f_try = np.maximum(f_try, 0)
                f_try /= f_try.sum()
        
        c_try = _conv_fft(f_try)
        peak_try = c_try.max()
        delta = peak_try - peak
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            f = f_try
            c = c_try
            peak = peak_try
            score = _eval(f)
            if score < best_score:
                best_score = score
                best_f = f.copy()
    
    return best_f, best_score

Hmm, I'm worried about a few things:
1. The coordinate descent might be too slow (copying the array and computing FFT for each trial).
2. The simulated annealing might not be effective enough.
3. The Adam optimizer might still get stuck.

Let me optimize the coordinate descent to be faster. Instead of computing the full FFT for each trial, I can compute the change in the convolution analytically:

If we change f[i] by +δ and f[j] by -δ, the convolution changes by:
`Δc[k] = δ * (f[k-i] + f[k-j])` (with appropriate boundary handling)

Wait, more precisely:
`Δc[k] = δ * f[k-i] + δ * f[k-i]` (from the +δ at position i, appearing twice in the convolution) 
minus `δ * f[k-j] + δ * f[k-j]` (from the -δ at position j)

Wait no. `c[k] = sum_m f[m] * f[k-m]`. If we change f[i] by +δ:
`Δc[k] from f[i] += δ = δ * f[k-i] + f[i+δ] * δ`... no wait.

`c_new[k] = sum_m f'[m] * f'[k-m]` where `f' = f + δ*e_i - δ*e_j`.

`c_new[k] - c[k] = sum_m (δ*δ_{mi} - δ*δ_{mj}) * f[k-m] + f[m] * (δ*δ_{k-m,i} - δ*δ_{k-m,j})`
`= δ*f[k-i] - δ*f[k-j] + δ*f[k-i] - δ*f[k-j]`
`= 2δ*(f[k-i] - f[k-j])`

(where we interpret f[·] = 0 outside [0, n-1]).

So `c_new[k] = c[k] + 2δ*(f[k-i] - f[k-j])` for all k.

This means we can compute the new peak without doing a full FFT! We just need to check:
`max_k (c[k] + 2δ*(f[k-i] - f[k-j]))`

This is O(n) instead of O(n log n) per evaluation. Much faster!

But we still need to update c for the next iteration. We can either:
1. Keep c in memory and update it incrementally (O(n) per step).
2. Recompute c from scratch via FFT every N steps.

Option 1 is better for the coordinate descent.

Let me rewrite the coordinate descent with this optimization:

def _coord_descent_fast(f, deadline, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(f)
    f = f.copy()
    c = _conv_fft(f)  # Initial convolution
    peak = c.max()
    best_score = _eval(f)
    best_f = f.copy()
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        k_star = int(np.argmax(c))
        
        # Compute the "influence" vector: for each coordinate j,
        # if we decrease f[j] by δ, c[k_star] changes by -2δ*f[k_star-j]
        # We want to decrease c[k_star], so we want to decrease f[j] where f[k_star-j] is large.
        
        # Also, we need to increase some f[i] where f[k_star-i] is small.

        infl = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                infl[j] = f[idx]
        
        # Top contributors (to decrease) and least contributors (to increase)
        top_contrib = np.argsort(infl)[::-1][:5]
        low_contrib = np.argsort(infl)[:5]
        
        improved = False
        for si in top_contrib:
            for di in low_contrib:
                if si == di:
                    continue
                # Try different amounts
                for frac in [0.01, 0.03, 0.05]:
                    δ = f[si] * frac
                    if δ < 1e-15:
                        continue
                    # Compute new c at k_star

"""
Multi-phase optimizer for step function coefficients minimizing 2*n*max(conv(f,f))/(sum f)^2.
Uses Adam with adaptive multi-peak softmax gradient, coordinate descent with FFT-based
evaluation, simulated annealing, and multiple diverse starting points.
Key improvements over previous approach:
1. Adam optimizer (adaptive per-coordinate learning rates) vs mirror descent
2. Multi-peak gradient (top-k weighted) for smoother optimization landscape
3. Faster coordinate descent using analytical convolution updates
4. Simulated annealing for global exploration
5. Better time management and restart strategy

import numpy as np
import time
from scipy.signal import correlate


def _conv_fft(f):
    """Compute linear convolution f*f via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _objective(f):
    """Compute 2*n*max(conv(f,f))/(sum f)^2 directly."""
    n = len(f)
    s = f.sum()
    if s < 0.01:
        return np.inf
    c = _conv_fft(f)
    return 2.0 * n * c.max() / (s * s)


def _gradient(f, c, T, top_k=1):
    """Compute multi-peak softmax gradient."""
    n = len(f)
    m = len(c)

    if top_k <= 1:
        cs = c / T
        cs -= cs.max()
        p = np.exp(cs)
        s = p.sum()
        if s < 1e-300:
            p = np.zeros(m)
            p[np.argmax(c)] = 1.0
        else:
            p /= s
    elif top_k >= m:
        p = np.ones(m) / m
    else:
        idx = np.argpartition(c, -top_k)[-top_k:]
        rest_mask = np.ones(m, dtype=bool)
        rest_mask[idx] = False
        rest_count = rest_mask.sum()

        c_top = c[idx]
        cs = c_top / T
        cs -= cs.max()
        w = np.exp(cs)
        w /= w.sum()

        p = np.zeros(m)
        p[idx] = 0.7 * w
        if rest_count > 0:
            p[rest_mask] = 0.3 / rest_count

    grad = 2.0 * correlate(p, f, mode='valid')
    return grad


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.max(np.nonzero(cond)[0])
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _adam_run(f, deadline, seed=0, lr_init=0.05, lr_end=0.0005,
              top_k_init=1, top_k_end=8, T_ratio=1.0, T_end_ratio=1e-5,
              eval_interval=2000, patience_max=3000):
    """Adam optimization with adaptive parameters and restarts."""
    rng = np.random.default_rng(seed)
    n = len(f)
    f = f.copy()

    c = _conv_fft(f)
    peak0 = c.max()
    avg0 = c.mean()
    spread0 = max(peak0 - avg0, 1e-10)

    T_init = spread0 * T_ratio
    T_end = spread0 * T_end_ratio

    state_m = np.zeros(n)
    state_v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8

    best_score = _objective(f)
    best_f = f.copy()
    patience = 0

    n_max = 500000
    last_print = 0

    for it in range(n_max):
        if time.monotonic() > deadline:
            break

        progress = min(1.0, it / n_max)

        # Adaptive schedules
        T = T_init * (T_end / T_init) ** progress
        lr = lr_init * (lr_end / lr_init) ** (progress ** 0.4)
        top_k = max(1, int(top_k_init + (top_k_end - top_k_init) * progress))

        # Compute gradient
        c = _conv_fft(f)
        grad = _gradient(f, c, T, top_k=top_k)

        # Adam update
        t += 1
        state_m = beta1 * state_m + (1 - beta1) * grad
        state_v = beta2 * state_v + (1 - beta2) * grad ** 2
        m_hat = state_m / (1 - beta1 ** t)
        v_hat = state_v / (1 - beta2 ** t)

        f_new = f - lr * m_hat / (np.sqrt(v_hat) + eps)
        f = _project_simplex(f_new)

        # Periodic evaluation
        if it % eval_interval == 0:
            score = _objective(f)
            if score < best_score - 1e-8:
                best_score = score
                best_f = f.copy()
                patience = 0
                if it - last_print > 20000:
                    last_print = it
            else:
                patience += 1

            if patience > patience_max:
                # Restart from best with perturbation
                scale = 0.003 + 0.01 * patience / patience_max
                f = best_f + scale * rng.standard_normal(n)
                f = np.maximum(f, 0.0)
                f /= f.sum()
                state_m = np.zeros(n)
                state_v = np.zeros(n)
                t = 0
                patience = 0

    score = _objective(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()

    return best_f, best_score


def _coord_descent(f, deadline, rng=None, eval_every=500):
    """Coordinate descent: transfer mass between coordinates to reduce peak."""
    if rng is None:
        rng = np.random.default_rng()
    n = len(f)
    f = f.copy()
    c = _conv_fft(f)
    peak = c.max()
    best_score = _objective(f)
    best_f = f.copy()

    it = 0
    stuck = 0
    max_stuck = 200

    while time.monotonic() < deadline:
        it += 1
        k_star = int(np.argmax(c))

        # Compute influence: f[k_star - j] for each j
        infl = np.zeros(n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        for j in range(lo, hi + 1):
            infl[j] = f[k_star - j]

        # Try structured moves: from high influence to low influence
        top_idx = np.argsort(infl)[::-1][:4]
        low_idx = np.argsort(infl)[:4]

        improved = False
        for si in top_idx:
            for di in low_idx:
                if si == di:
                    continue
                for frac in [0.02, 0.05, 0.1]:
                    delta = f[si] * frac
                    if delta < 1e-15:
                        continue
                    f_try = f.copy()
                    f_try[si] -= delta
                    f_try[di] += delta
                    if f_try[si] < 0:
                        continue
                    c_try = _conv_fft(f_try)
                    if c_try.max() < peak - 1e-14:
                        f = f_try
                        c = c_try
                        peak = c.max()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

                di = int(rng.integers(n))

                frac = rng.uniform(0.01, 0.08)

                f_try = f.copy()
                f_try[si] -= delta

        else:

        if it % eval_every == 0:

def _sa_run(f, deadline, seed=0, T_init=0.005, T_end=1e-6):
    """Simulated annealing with structured perturbations."""
    rng = np.random.default_rng(seed)
    n = len(f)
    f = f.copy()
    c = _conv_fft(f)
    peak = c.max()
    best_score = _objective(f)
    best_f = f.copy()

    steps_since_improve = 0

    it = 0
    while time.monotonic() < deadline:
        it += 1
        progress = min(1.0, it / 200000)
        T = T_init * (T_end / T_init) ** progress

        f_try = f.copy()
        pert_type = rng.integers(4)

        if pert_type == 0:
            # Gaussian noise
            noise_scale = T * 0.05
            f_try += noise_scale * rng.standard_normal(n)
            f_try = np.maximum(f_try, 0.0)
            f_try /= f_try.sum()
        elif pert_type == 1:
            # Pair swap
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                amt = min(f[i], f[j]) * rng.uniform(0.05, 0.3)
                f_try[i] -= amt
                f_try[j] += amt
                f_try = np.maximum(f_try, 0.0)
                f_try /= f_try.sum()
        elif pert_type == 2:
            # Block perturbation
            blen = int(rng.integers(20, min(200, n)))
            bstart = int(rng.integers(0, n - blen))
            f_try[bstart:bstart + blen] += T * 0.05 * rng.standard_normal(blen)
            f_try = np.maximum(f_try, 0.0)
            f_try /= f_try.sum()
        else:
            # Peak-directed move
            k_star = int(np.argmax(c))
            infl = np.zeros(n)
            lo = max(0, k_star - n + 1)
            hi = min(n - 1, k_star)
            for j in range(lo, hi + 1):
                infl[j] = f[k_star - j]
            si = int(rng.choice(np.argsort(infl)[::-1][:10]))
            di = int(rng.choice(np.argsort(infl)[:10]))
            if si != di:
                amt = f[si] * rng.uniform(0.05, 0.15)

                f_try[di] += amt
                f_try = np.maximum(f_try, 0.0)
