Combines improved mirror descent with adaptive step sizing,
multi-peak gradient, and LP-based refinement.

import numpy as np
from scipy.signal import correlate
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _autocorr_fft(f):
    """Full linear autocorrelation (convolution) via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]

def _peak_gradient(f, k_star):
    """Exact subgradient of max autocorrelation at peak lag k_star."""
    n = len(f)
    grad = np.zeros(n)
    # grad[j] = 2 * f[k_star - j] if 0 <= k_star - j < n
    valid = (k_star - np.arange(n) >= 0) & (k_star - np.arange(n) < n)
    grad[valid] = 2.0 * f[k_star - np.where(valid)[0]]
    return grad

def _soft_gradient(f, c, T, topK=5):
    """Combined gradient: weighted peaks + soft-max."""
    n = len(f)
    # Soft-max component
    cs = c / T
    cs -= cs.max()
    p = np.exp(cs)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros(2 * n - 1)
        p[n - 1] = 1.0
    else:
        p /= s
    
    # Top-K peak component
    k_max = min(topK, len(c))
    top_indices = np.argsort(c)[::-1][:topK]
    weights = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_indices):
        weights[idx] = 1.0 / (i + 1)
    weights /= weights.sum()
    
    # Combine
    combined = 0.5 * p + 0.5 * weights
    
    # Gradient: 2 * correlate(combined, f, 'valid')
    # correlate(p, f, 'valid') gives the "cross-correlation" 
    # which is what we want for the gradient
    grad = 2.0 * correlate(combined, f, mode='valid')
    return grad

def _mirror_step(f, grad, eta):
    """Mirror descent step on the simplex."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    f_new /= s
    return f_new

def _eval(f):
    return evaluate_sequence(f.tolist())

def _optimize_mirror(f, n, deadline, rng, label=""):
    """Improved mirror descent with adaptive step size and multi-peak gradient."""
    f = np.maximum(f, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f /= s
    
    local_best = _eval(f)
    local_best_f = f.copy()
    
    c = _autocorr_fft(f)
    peak0 = c.max()
    
    # Parameters
    T_ratio_start = 0.1  # T = 0.1 * peak initially
    T_ratio_end = 1e-8   # T = 1e-8 * peak at end
    n_iters = 50000
    stuck_count = 0
    improvement_threshold = 1e-7
    
    for it in range(n_iters):
        if time.monotonic() > deadline:
            break
        
        progress = it / n_iters
        # Temperature: geometric schedule
        T = peak0 * (T_ratio_start * (T_ratio_end / T_ratio_start) ** progress)
        # Step size: geometric from 0.5 to 0.01
        eta = 0.5 * (0.01 / 0.5) ** (progress ** 0.5)
        
        c = _autocorr_fft(f)
        grad = _soft_gradient(f, c, T, topK=5)
        
        f_new = _mirror_step(f, grad, eta)
        
        # Check improvement every 10 steps
        if it % 10 == 0:
            score = _eval(f_new)
            if score < local_best - improvement_threshold:
                f = f_new
                local_best = score
                local_best_f = f.copy()
                stuck_count = 0
            else:
                # Try smaller step
                f_half = _mirror_step(f, grad, eta * 0.5)
                score_half = _eval(f_half)
                if score_half < local_best - improvement_threshold:
                    f = f_half
                    local_best = score_half
                    local_best_f = f.copy()
                    stuck_count = 0
                else:
                    stuck_count += 1
            
            if stuck_count > 10:
                # Perturbation
                mag = 0.01 / np.sqrt(n)
                f = f + mag * rng.standard_normal(n)
                f = np.maximum(f, 0.0)
                f /= f.sum()
                stuck_count = 0
                local_best = _eval(f)
                if local_best < _eval(local_best_f):
                    local_best_f = f.copy()
    
    score = _eval(f)
    if score < local_best:
        local_best = score
        local_best_f = f.copy()
    
    return local_best_f, local_best

"""
Step function optimizer minimizing 2*n*max(f*f)/(sum f)^2.
Uses hybrid approach: mirror descent with multi-peak gradient,
periodic LP-based refinement, and coordinate descent polish.
"""

import numpy as np
from scipy.signal import correlate
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
import time

def _convolution_fft(f):
    """Full linear convolution of f with itself via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]

def _score(f):
    """Compute the objective value (without calling evaluate_sequence)."""
    n = len(f)
    c = _convolution_fft(f)
    max_c = c.max()
    sum_f = f.sum()
    if sum_f < 0.01:
        return np.inf
    return 2.0 * n * max_c / (sum_f ** 2)

def _eval_safe(f):
    """Evaluate using the safe function."""
    return evaluate_sequence(f.tolist())

def _mirror_descent_step(f, grad, eta):
    """Mirror descent (exponentiated gradient) step."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    return f_new / s

def _gradient_multipeak(f, c, T, K):
    """Compute gradient using top-K peaks and soft-max."""
    n = len(f)
    # Soft-max over all lags
    cs = c / T
    cs -= cs.max()
    p = np.exp(cs)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(c)] = 1.0
    else:
        p /= s
    
    # Top-K weighted peaks
    top_idx = np.argsort(c)[::-1][:K]
    peak_w = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_idx):
        peak_w[idx] = 1.0 / (i + 1)
    peak_w /= peak_w.sum()
    
    # Combined weight
    w = 0.4 * p + 0.6 * peak_w
    
    # Gradient: 2 * cross-correlation of w with f
    grad = 2.0 * correlate(w, f, mode='valid')
    return grad

def _solve_lp_reduced(f, M, active_lags):
    """Solve reduced LP: max sum(g) s.t. g>=0, (f*g)[k]<=M for k in active_lags."""
    n = len(f)
    K = len(active_lags)
    if K == 0:
        return None
    
    rows = []
    cols = []
    vals = []
    for i, k in enumerate(active_lags):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(i)
            cols.append(j)
            vals.append(f[k - j])
    
    if len(rows) == 0:
        return None
    
    A = csr_matrix((vals, (rows, cols)), shape=(K, n))
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    b = np.full(K, M)
    
    try:
        result = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                        options={'presolve': True, 'time_limit': 30})
        if result.success:
            return result.x
    except Exception:
        pass
    return None

def _optimize_phase1(f, n, deadline, rng, label=""):

    f = np.maximum(f, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f /= s
    
    local_best = _score(f)
    local_best_f = f.copy()
    
    c = _convolution_fft(f)
    peak0 = c.max()
    
    T_start = peak0 * 0.05
    T_end = peak0 * 1e-10
    eta_start = 0.8
    eta_end = 0.005
    K_peaks = 5
    eval_interval = 5
    stuck_count = 0
    
    it = 0
    max_iters = 100000
    
    while it < max_iters and time.monotonic() < deadline:
        progress = min(1.0, it / 30000.0)
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** (progress ** 0.7)
        
        c = _convolution_fft(f)
        grad = _gradient_multipeak(f, c, T, K_peaks)
        
        f_candidate = _mirror_descent_step(f, grad, eta)
        
        if it % eval_interval == 0:
            sc = _score(f_candidate)
            if sc < local_best - 1e-8:
                f = f_candidate
                local_best = sc
                local_best_f = f.copy()
                stuck_count = 0
            else:
                # Try with half step
                f_half = _mirror_descent_step(f, grad, eta * 0.5)
                sc_half = _score(f_half)
                if sc_half < local_best - 1e-8:
                    f = f_half
                    local_best = sc_half
                    local_best_f = f.copy()
                    stuck_count = 0
                else:
                    stuck_count += 1

                f = f + mag * rng.standard_normal(n)
                f = np.maximum(f, 0.0)
                f /= f.sum()
                stuck_count = 0
                
                sc = _score(f)
                if sc < local_best:
                    local_best = sc
                    local_best_f = f.copy()
        
        it += 1
    
    return local_best_f, local_best

Hybrid approach: mirror descent with multi-peak gradient and adaptive step size,

"""

import numpy as np
from scipy.signal import correlate

def _convolution_fft(f):
    """Full linear convolution of f with itself via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _score(f):
    """Compute the objective value directly."""
    n = len(f)
    c = _convolution_fft(f)
    max_c = c.max()
    sum_f = f.sum()
    if sum_f < 0.01:
        return np.inf
    return 2.0 * n * max_c / (sum_f ** 2)

def _mirror_step(f, grad, eta):
    """Mirror descent (exponentiated gradient) step on the simplex."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy()
    return f_new / s


def _gradient_multipeak(f, c, T, K):
    """Compute gradient using top-K peaks combined with soft-max tail."""
    n = len(f)
    cs = c / T
    cs -= cs.max()
    p = np.exp(cs)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros(2 * n - 1)
        p[np.argmax(c)] = 1.0
    else:
        p /= s

    top_idx = np.argsort(c)[::-1][:K]
    peak_w = np.zeros(2 * n - 1)
    for i, idx in enumerate(top_idx):
        peak_w[idx] = 1.0 / (i + 1)
    peak_w /= peak_w.sum()

    w = 0.3 * p + 0.7 * peak_w
    grad = 2.0 * correlate(w, f, mode='valid')

    """Solve reduced LP: max sum(g) s.t. g>=0, (f*g)[k]<=M for active lags."""
    n = len(f)
    K = len(active_lags)
    if K == 0:
        return None

    rows = []
    cols = []
    vals = []
    for i, k in enumerate(active_lags):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        for j in range(j_start, j_end + 1):
            rows.append(i)
            cols.append(j)
            vals.append(f[k - j])

    if len(rows) == 0:
        return None

    A = csr_matrix((vals, (rows, cols)), shape=(K, n))
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    b = np.full(K, M)

    try:
        result = linprog(c_obj, A_ub=A, b_ub=b, bounds=bounds, method='highs',
                         options={'presolve': True, 'time_limit': 60})
        if result.success:
            return result.x
    except Exception:
        pass
    return None

        f = np.maximum(f0, 0.0)
        s = f.sum()
        if s < 1e-10:
            f = np.ones(n)
        else:
            f = f / s
        sc = _score(f)
        start_results.append((sc, name, f))

    start_results.sort(key=lambda x: x[0])
    print("Starting point scores:")
    for sc, name, _ in start_results:
        print(f"  {name}: {sc:.6f}")

    if start_results[0][0] < best_score:
        best_score = start_results[0][0]
        best_f = start_results[0][2].copy()
        print(f"Best start: {best_score:.6f}")

    # Phase 1: Mirror descent from top 5 starting points
    top_starts = start_results[:5]
    remaining = deadline - time.monotonic()
    time_per_start = remaining * 0.45 / 5.0

    phase1_best = best_score
    for idx, (sc, name, f0) in enumerate(top_starts):
        remaining = deadline - time.monotonic()
        if remaining < 60:
            break
        t_alloc = min(time_per_start * (1.0 - 0.08 * idx), remaining - 150)
        if t_alloc < 30:
            break

        start_dl = time.monotonic() + t_alloc
        print(f"\nPhase 1 [{idx+1}/5]: '{name}' (budget {t_alloc:.0f}s)")

        f_opt, sc_opt = _optimize_mirror(f0, n, start_dl, rng, label=name)

        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            phase1_best = best_score
            print(f"  New best: {best_score:.6f}")

    print(f"\nAfter Phase 1: {best_score:.6f} (elapsed {time.monotonic()-t0:.0f}s)")

    # Phase 2: LP-based refinement
    remaining = deadline - time.monotonic()
    if remaining > 120:
        lp_budget = min(remaining * 0.3, 480.0)
        lp_deadline = time.monotonic() + lp_budget
        print(f"\nPhase 2: LP refinement (budget {lp_budget:.0f}s)")

        f_opt, sc_opt = _optimize_lp(best_f, n, lp_deadline, rng, label="LP")

        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  New best after LP: {best_score:.6f}")

    # Phase 3: Mirror descent again from current best (with fresh parameters)
    remaining = deadline - time.monotonic()
    if remaining > 120:
        md_budget = min(remaining * 0.3, 360.0)
        md_deadline = time.monotonic() + md_budget
        print(f"\nPhase 3: Second mirror descent round (budget {md_budget:.0f}s)")

        f_opt, sc_opt = _optimize_mirror(best_f, n, md_deadline, rng, label="MD2")

        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  New best: {best_score:.6f}")

    # Phase 4: LP again if time permits
    remaining = deadline - time.monotonic()
    if remaining > 60:
        lp_budget = min(remaining * 0.2, 180.0)
        lp_deadline = time.monotonic() + lp_budget
        print(f"\nPhase 4: Second LP refinement (budget {lp_budget:.0f}s)")

        f_opt, sc_opt = _optimize_lp(best_f, n, lp_deadline, rng, label="LP2")

        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  New best: {best_score:.6f}")

    # Phase 5: Polish
    remaining = deadline - time.monotonic()
    if remaining > 30:
        polish_budget = min(remaining - 10, 240.0)
        polish_deadline = time.monotonic() + polish_budget
        print(f"\nPhase 5: Polish (budget {polish_budget:.0f}s)")

        f_opt, sc_opt = _polish(best_f, n, polish_deadline, rng)

        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  Polished to: {sc_opt:.6f}")

    # Final: one more short mirror descent + polish
    remaining = deadline - time.monotonic()
    if remaining > 40:
        final_budget = min(remaining - 10, 120.0)
        final_deadline = time.monotonic() + final_budget
        print(f"\nFinal polish round (budget {final_budget:.0f}s)")
        f_opt, sc_opt = _polish(best_f, n, final_deadline, rng)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  Final: {best_score:.6f}")

    # Ensure valid output
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = _eval_safe(best_f)
    if final_score < best_score or not np.isfinite(best_score):
        best_score = final_score

    print(f"\n=== FINAL SCORE: {best_score:.6f} (elapsed {time.monotonic()-t0:.0f}s) ===")
    return [float(x) for x in best_f]
