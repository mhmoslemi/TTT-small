def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b):
    la = len(a)
    lb = len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 1500000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)


def _valid_corr(w, p):
    """Return v[j] = sum_i w[i] * p[i+j] for j in range(len(p))."""
    lenw = len(w)
    lenp = len(p)
    if lenp == 0:
        return np.array([], dtype=float)
    L = _next_pow2(max(lenw, 2 * lenp))
    fw = np.fft.rfft(w, L)
    fp = np.fft.rfft(p, L)
    out = np.fft.irfft(fw * np.conjugate(fp), L)
    return out[:lenp].astype(float, copy=False)


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:

def _score_and_grad(p, beta, top_k=3):
    """Return (exact_score, smooth_val, gradient) using top-K peak weighting."""
    p = _sanitize(p)
    if p is None:
        return float("inf"), float("inf"), None
    n = p.size
    c = _conv_full(p, p)
    if c.size == 0:
        return float("inf"), float("inf"), None
    c = np.where(np.isfinite(c), c, 0.0)
    d = n * c  # score contribution at each lag
    exact = float(2.0 * np.max(d)) if d.size else float("inf")

    # Top-K weighted softmax
    if d.size <= top_k:
        w = np.full(d.size, 1.0 / d.size)
    else:
        if beta <= 1e-8:
            w = np.full(d.size, 1.0 / d.size)
        else:
            shift = float(np.max(d))

            s = float(e.sum())
            if s <= 0 or not np.isfinite(s):
                w = np.full(d.size, 1.0 / d.size)
            else:
                w = e / s
            # Boost top-K peaks
            idx = np.argsort(d)[-top_k:]
            boost = 1.0 + 2.0 / top_k
            w[idx] *= boost
            w /= w.sum()

    smooth = float(np.dot(w, d))
    h = _valid_corr(w, p)
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.maximum(h, 0.0)
    if h.size and np.max(h) > 0:

def _resample(arr, m):
    a = _sanitize(arr)
    if a is None:
        return None
    n = a.size
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m, dtype=float) / m
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0) + 1e-12
    return _sanitize(out)


def _initial_sequence(m, rng, mode):
    t = (np.arange(m, dtype=float) + 0.5) / float(m)
    p = np.ones(m, dtype=float)

    if mode == 0:
        # Beta-like
        a = float(rng.uniform(0.3, 2.5))
        b = float(rng.uniform(0.3, 2.5))
        tt = np.clip(t, 1e-8, 1.0 - 1e-8)
        p = np.power(tt, a) * np.power(1.0 - tt, b)
    elif mode == 1:
        # Gaussian bumps
        K = int(rng.integers(1, 4))
        p = np.zeros(m)
        centers = np.sort(rng.uniform(0.0, 1.0, K))
        sigmas = rng.uniform(0.05, 0.3, K)
        for ii in range(K):
            p += np.exp(-((t - centers[ii]) / sigmas[ii]) ** 2)
    elif mode == 2:
        # Piecewise linear
        K = int(rng.integers(3, 8))
        knots = np.sort(rng.uniform(0.0, 1.0, K))
        heights = rng.lognormal(0.0, 0.8, K)
        p = np.interp(t, knots, heights)
    elif mode == 3:
        # Two-block
        a = float(rng.uniform(0.3, 0.7))
        idx = int(round(a * m))
        ratio = float(rng.uniform(0.3, 3.0))
        p = np.ones(m)
        p[idx:] = ratio
    elif mode == 4:
        # Cosine series
        z = np.zeros(m)
        for k in range(1, 10):
            amp = float(rng.normal(0, 1.0 / (k + 1)))
            ph = float(rng.uniform(0, 2 * np.pi))
            z += amp * np.cos(2 * np.pi * k * t + ph)
        p = np.log1p(np.exp(np.clip(z, -30, 30)))
    else:
        # Smooth pulse
        c = float(rng.uniform(0.2, 0.8))
        k = float(rng.uniform(5, 25))
        p = 1.0 / (1.0 + np.exp(-np.clip(k * (t - c), -50, 50)))
        p *= np.exp(-((t - 0.5) / 0.3) ** 2)

    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0) + 1e-10
    s = p.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones(m) / m
    return p / s


def _fourier_init(n, rng):
    """Initialize using constant-magnitude frequency domain (CAZAC-inspired)."""
    # Random phases -> constant magnitude in freq domain
    L = 2 * n
    phases = rng.uniform(0, 2 * np.pi, L // 2 + 1)
    # Build a Hermitian-symmetric spectrum with constant magnitude
    spec = np.zeros(L // 2 + 1, dtype=complex)
    spec[0] = 1.0
    spec[-1] = 1.0
    for k in range(1, L // 2):
        spec[k] = np.exp(1j * phases[k])
        spec[L - k] = np.conj(spec[k]) if L - k <= L // 2 else 0
    # Reconstruct
    freq = np.zeros(L, dtype=complex)
    freq[:L // 2 + 1] = spec
    for k in range(1, L // 2):
        freq[L - k] = np.conj(freq[k])
    p = np.real(np.fft.ifft(freq))[:n]
    p = np.maximum(p, 0.0) + 1e-10
    return _sanitize(p)


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 60.0
    start_time = time.monotonic()
    deadline = start_time + budget - 15.0

    rng = np.random.default_rng(int(seed) % 1000000007 + 42)

    # Load parent
    parent = None
    try:
        parent_raw = height_sequence_1
        parent = _sanitize(np.asarray(parent_raw, dtype=float))
    except (NameError, Exception):
        pass

    best_arr = None
    best_score = float("inf")

    def _update_best(arr, sc):
        nonlocal best_arr, best_score
        if arr is not None and np.isfinite(sc) and sc < best_score:
            best_arr = arr.copy()
            best_score = sc

    if parent is not None:
        _update_best(parent, _score(parent))
        print(f"Parent score: {best_score:.6f}, len={parent.size}")

    # Choose target length
    # Try 3072 as primary (matching parent), also explore 4096
    target_n = 3072
    if parent is not None and parent.size > 0:
        target_n = parent.size

    # Phase 1: Quick exploration at small n to find good basins
    p1_deadline = min(start_time + 120.0, deadline - 1200.0)
    small_n = 512
    starts = []

    if parent is not None:
        sp = _resample(parent, small_n)
        if sp is not None:
            starts.append(sp)

    for mode in [3, 0, 1, 4, 2]:
        starts.append(_initial_sequence(small_n, rng, mode))
    starts.append(_fourier_init(small_n, rng))

    for sp in starts[:6]:
        if time.monotonic() >= p1_deadline:
            break
        talloc = min(15.0, max(3.0, p1_deadline - time.monotonic() - 5.0))
        if talloc < 3:
            break
        p, sc = _mmd_search(sp, rng, talloc, beta_start=30, beta_end=400, top_k=3)
        if p is not None:
            _update_best(p, sc)

    # Phase 2: Main optimization at target length
    print(f"Phase 2: main optimization at n={target_n}")
    main_deadline = min(start_time + 1800.0, deadline - 300.0)

    # Build starts at target length
    target_starts = []
    if parent is not None:
        sp = _resample(parent, target_n)
        if sp is not None:
            target_starts.append(sp)
            sp2 = _sanitize(sp * np.exp(rng.normal(0, 0.03, target_n)))
            if sp2 is not None:
                target_starts.append(sp2)

    # Resample best small-n result
    if best_arr is not None and best_arr.size != target_n:
        sp = _resample(best_arr, target_n)
        if sp is not None:
            target_starts.append(sp)

    for mode in [3, 0, 4]:
        target_starts.append(_initial_sequence(target_n, rng, mode))
    target_starts.append(_fourier_init(target_n, rng))

    # Score and sort starts
    scored = []
    for sp in target_starts:
        if sp is None:
            continue
        sc = _score(sp)
        scored.append((sc, sp))
    scored.sort(key=lambda x: x[0])

    print(f"Best start score: {scored[0][0]:.6f}")

    # Run MMD on top starts
    fracs = [0.45, 0.25, 0.15, 0.10, 0.05]
    for idx, (sc, sp) in enumerate(scored[:5]):
        if time.monotonic() >= main_deadline:
            break
        frac = fracs[idx] if idx < len(fracs) else 0.05
        rem = main_deadline - time.monotonic()
        talloc = max(10.0, min(rem - 10.0, 0.35 * rem if idx == 0 else rem * frac))
        if talloc < 10:
            break

        p, sc_r = _mmd_search(sp, rng, talloc, beta_start=40, beta_end=2000, top_k=4)
        _update_best(p, sc_r)
        print(f"  MMD start {idx}: score={sc_r:.6f}")

    # Phase 3: Polishing
    print(f"Phase 3: polishing, current best={best_score:.6f}")
    if best_arr is not None:
        rem = deadline - time.monotonic()
        # Peak coordinate search
        if rem > 120:
            t1 = min(400.0, 0.4 * rem)
            p, sc = _peak_coord_search(best_arr, rng, t1)
            _update_best(p, sc)
            print(f"  Peak coord search: {sc:.6f}")

        # Block search
        rem = deadline - time.monotonic()
        if rem > 100:
            t2 = min(300.0, 0.3 * rem)
            p, sc = _block_search(best_arr, rng, t2)
            _update_best(p, sc)
            print(f"  Block search: {sc:.6f}")

        # Final MMD with high beta
        rem = deadline - time.monotonic()
        if rem > 80:
            t3 = min(250.0, 0.4 * rem)
            p, sc = _mmd_search(best_arr, rng, t3, beta_start=150, beta_end=5000, top_k=2)
            _update_best(p, sc)
            print(f"  Final MMD: {sc:.6f}")

        # Another peak coord pass
        rem = deadline - time.monotonic()
        if rem > 60:
            p, sc = _peak_coord_search(best_arr, rng, min(100.0, 0.5 * rem))
            _update_best(p, sc)
            print(f"  Final peak coord: {sc:.6f}")

    # Phase 4: Try a different length (4096) if time permits
    rem = deadline - time.monotonic()
    if rem > 200 and best_arr is not None:
        print(f"Phase 4: trying n=4096, rem={rem:.0f}s")
        big_n = 4096 if best_arr.size <= 3072 else 4096
        big = _resample(best_arr, big_n)
        if big is not None:
            t_big = min(150.0, 0.5 * rem)
            p, sc = _mmd_search(big, rng, t_big, beta_start=80, beta_end=3000, top_k=4)
            _update_best(p, sc)
            print(f"  n=4096 MMD: {sc:.6f}")

            rem = deadline - time.monotonic()
            if rem > 60:
                p2, sc2 = _peak_coord_search(p if p is not None else big, rng, min(80.0, 0.5 * rem))
                _update_best(p2, sc2)
                print(f"  n=4096 peak: {sc2:.6f}")

    # Final validation
    if best_arr is None:
        best_arr = _initial_sequence(target_n, rng, 3)
        best_score = _score(best_arr)

    best_arr = _sanitize(best_arr)
    if best_arr is None:
        best_arr = np.ones(target_n) / target_n
        best_score = _score(best_arr)

    # Try evaluate_sequence on final candidates
    final_list = [float(x) for x in best_arr]
    try:
        sc_final = evaluate_sequence(final_list)
        print(f"Final evaluate_sequence: {sc_final:.6f}")
    except Exception:
        pass

    # Ensure valid output
    out = np.asarray(final_list, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = np.ones(target_n) / target_n
    else:
        out = out / s

    return [float(x) for x in out]
