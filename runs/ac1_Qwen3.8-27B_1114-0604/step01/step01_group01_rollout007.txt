"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses a hybrid approach: fast multiplicative smoothed-max descent, LP-based
Frank-Wolfe refinement (solving a linear program to find optimal blend direction),
adaptive block/coordinate search, and up-polishing to larger lengths.

import numpy as np
import time
import scipy.sparse
from scipy.optimize import linprog


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b):
    la = len(a)
    lb = len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 1500000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)


def _sanitize_vec(p):
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return p / s


def _exact_score(p):
    p = _sanitize_vec(p)
    if p is None:
        return float("inf")
    m = p.size
    c = _conv_full(p, p)
    if c.size == 0:
        return float("inf")
    c = np.where(np.isfinite(c), c, 0.0)
    return float(2.0 * m * np.max(c))


def _soft_weights(d, beta):
    d = np.asarray(d, dtype=float)
    n = d.size
    if n == 0:
        return np.array([], dtype=float)
    d = np.where(np.isfinite(d), d, 0.0)
    if beta <= 1e-8:
        return np.full(n, 1.0 / n)
    shift = float(np.max(d))
    if not np.isfinite(shift):
        return np.full(n, 1.0 / n)
    e = np.exp(beta * (d - shift))
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return e / s


def _gradient_direction(p, beta):
    """Return (exact_score, smooth_score, gradient_vector, max_grad) for normalized p."""
    p = _sanitize_vec(p)
    if p is None:
        return float("inf"), float("inf"), None, 0.0
    m = p.size
    c = _conv_full(p, p)
    if c.size == 0:
        return float("inf"), float("inf"), None, 0.0
    c = np.where(np.isfinite(c), c, 0.0)
    d = m * c
    exact = float(2.0 * max(0.0, np.max(d))) if d.size else float("inf")
    w = _soft_weights(d, beta)
    # Gradient of sum_j w_j * (m*c)[j] w.r.t. p
    # d/dp_i [m * sum_k w_k * (p*p)[k]] = 2 * m * sum_k w_k * (T_k p)_i
    # where (T_k p)_i = p[k-i]
    # So grad_i = 2*m * sum_k w_k * p[k-i] = 2*m * (w * p_corr)_i
    # where (w * p_corr)_i = sum_k w[k] * p[k-i] for valid k-i
    # This is the "valid" cross-correlation of w and p
    # Using FFT: correlate(w, p) gives sum_k w[k] * conj(p[k-i])... 
    # Actually we need: h[i] = sum_k w[k] * p[k-i] for k-i in [0,m-1]
    # This is the "valid" convolution of w_rev and p, or equivalently the first m
    # elements of the full cross-correlation.
    L = _next_pow2(max(2 * m, 1))
    fw = np.fft.rfft(w, L)
    fp = np.fft.rfft(p, L)
    # We want h[i] = sum_k w[k] * p[k-i]
    # This equals (w * flip(p)) evaluated at position... 
    # Let's use: h = ifft(fft(w) * conj(fft(p))) gives circular cross-correlation
    # h_circ[i] = sum_k w[k] * p[(k-i) mod L]
    # For the "valid" part (no wraparound), we need L >= 2m-1
    h_circ = np.fft.irfft(fw * np.conjugate(fp), L)[:m]
    h = 2.0 * m * h_circ
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.maximum(h, 0.0)
    maxh = float(np.max(h)) if h.size else 0.0
    smooth = float(np.dot(w, d))
    return exact, smooth, h, maxh

def _lp_frank_wolfe(p0, rng, time_limit, verbose=False):
    """LP-based Frank-Wolfe: solve LP to find best blend direction."""
    p = _sanitize_vec(p0)
    if p is None or time_limit <= 0:
        return None, float("inf")

    n = int(p.size)
    if n < 8:
        return p, _exact_score(p)

    floor = 1e-18 / float(n)
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_score = _exact_score(p)
    deadline = time.monotonic() + float(time_limit)
    it = 0
    failed = 0

    # M threshold for active constraints (fraction of max)
    active_frac = 0.80

    while time.monotonic() < deadline:
        c = _conv_full(p, p)
        c = np.where(np.isfinite(c), c, 0.0)
        M = float(np.max(c))
        if M <= 0:
            break

        # Select active constraints
        active_mask = c >= active_frac * M
        n_active = int(np.count_nonzero(active_mask))
        if n_active < 3:
            active_mask = c >= 0.5 * M
            n_active = int(np.count_nonzero(active_mask))
        if n_active < 1:
            active_mask = c >= 0.3 * M
            n_active = int(np.count_nonzero(active_mask))

        active_indices = np.where(active_mask)[0]

        # Build sparse constraint matrix A for active constraints
        # A[k_active, j] = p[k - j] if 0 <= k-j < n
        rows = []
        cols = []
        vals = []
        for ki in active_indices:
            lo = max(0, ki - n + 1)
            hi = min(n, ki + 1)
            for j in range(lo, hi):
                rows.append(len(rows))
                cols.append(j)
                vals.append(p[ki - j])

        if len(rows) == 0:
            break

        A_sp = scipy.sparse.coo_matrix((vals, (rows, cols)), shape=(n_active, n)).tocsr()
        b_vec = np.full(n_active, M)
        c_obj = np.ones(n)

        # Solve: max c_obj^T g s.t. A_sp @ g <= b_vec, g >= 0
        res = linprog(-c_obj, A_ub=A_sp, b_ub=b_vec, bounds=[(0, None)] * n,
                      method='highs', options={'presolve': True, 'time_limit': 30})

        if res is None or not res.success or res.x is None:
            failed += 1
            if failed > 10:
                break
            # Perturb and retry
            p_try = _sanitize_vec(p * np.exp(rng.normal(0.0, 0.02, n)))
            if p_try is not None:
                p_try = np.maximum(p_try, floor)
                p_try = p_try / p_try.sum()
                p = p_try
            continue

        g = np.asarray(res.x, dtype=float)
        g = np.where(np.isfinite(g), g, 0.0)
        g = np.maximum(g, 0.0)
        sum_g = float(g.sum())
        if sum_g < 1e-10 or not np.isfinite(sum_g):
            failed += 1
            if failed > 10:
                break
            continue

        # Normalize g to same sum as p
        g = g / sum_g

        # Frank-Wolfe step: try several step sizes
        cur_score = best_score
        best_local = p.copy()
        best_local_score = cur_score

        for t in (0.05, 0.1, 0.15, 0.2, 0.3, 0.5):
            p_try = (1.0 - t) * p + t * g
            p_try = np.maximum(p_try, floor)
            p_try = p_try / p_try.sum()
            sc = _exact_score(p_try)
            if sc < best_local_score - 1e-12:
                best_local_score = sc
                best_local = p_try.copy()

        if best_local_score < cur_score - 1e-12:
            p = best_local.copy()
            if best_local_score < best_score:
                best = p.copy()
                best_score = best_local_score
                if verbose and it % 10 == 0:
                    print(f"  LP-FW n={n} iter={it} best={best_score:.6f}")
            failed = 0
        else:
            # No improvement; try smaller steps or perturb
            improved = False
            for t in (0.01, 0.02, 0.03):
                p_try = (1.0 - t) * p + t * g
                p_try = np.maximum(p_try, floor)
                p_try = p_try / p_try.sum()
                sc = _exact_score(p_try)
                if sc < cur_score - 1e-12:
                    p = p_try
                    improved = True
                    break
            if not improved:
                # Accept the blend anyway (might help for future iterations)
                t = 0.05
                p_try = (1.0 - t) * p + t * g
                p_try = np.maximum(p_try, floor)
                p_try = p_try / p_try.sum()
                sc = _exact_score(p_try)
                if sc <= best_score * 1.02:
                    p = p_try
                else:
                    p = best.copy()
                failed += 1
                if failed > 15:
                    break

        it += 1
        # Adjust active_frac over time for tighter constraints
        active_frac = min(0.95, 0.80 + 0.005 * it)

    return best, best_score


def _block_search(p0, rng, time_limit):
    """Random contiguous block scaling search."""
    p = _sanitize_vec(p0)
    if p is None or time_limit <= 0:
        return None, float("inf")
    n = int(p.size)
    if n <= 1:
        return p, _exact_score(p)

    floor = 1e-18 / float(n)
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_score = _exact_score(p)
    cur_score = best_score
    deadline = time.monotonic() + float(time_limit)
    stagnant = 0

    factors = (0.4, 0.7, 0.9, 1.1, 1.4, 2.0, 3.0)

    while time.monotonic() < deadline:
        r = float(rng.random())
        if r < 0.25:
            w = int(rng.integers(1, max(2, n // 100 + 1)))
        elif r < 0.65:
            w = int(rng.integers(max(2, n // 100), max(3, n // 30 + 1)))
        elif r < 0.90:
            low = max(3, n // 30)
            high = max(low + 1, min(n, n // 8 + 1))
            w = int(rng.integers(low, high + 1))
        else:
            low = max(5, n // 10)
            high = max(low + 1, min(n, n // 3 + 1))
            w = int(rng.integers(low, high + 1))

        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))

        base = p.copy()
        current_best = base
        current_score = cur_score

        for f in factors:
            cand = base.copy()
            cand[start:start + w] *= float(f)
            cand = _sanitize_vec(cand)
            if cand is None:
                continue
            cand = np.maximum(cand, floor)
            cand = cand / cand.sum()
            sc = _exact_score(cand)
            if sc < current_score - 1e-12:
                current_score = sc
                current_best = cand

        if current_best is not base:
            p = current_best.copy()
            cur_score = current_score
            if cur_score < best_score:
                best = p.copy()
                best_score = cur_score
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 80:
                p_try = _sanitize_vec(p * np.exp(rng.normal(0.0, 0.04, n)))
                if p_try is not None:
                    p_try = np.maximum(p_try, floor)
                    p_try = p_try / p_try.sum()
                    sc = _exact_score(p_try)
                    if np.isfinite(sc) and sc <= best_score * 1.03:
                        p = p_try
                        cur_score = sc
                    else:
                        p = best.copy()
                        cur_score = best_score
                stagnant = 0
            if stagnant > 500:
                p = best.copy()
                cur_score = best_score
                stagnant = 0

    return best, best_score


def _resample(arr, m):
    a = _sanitize_vec(arr)
    if a is None:
        return None
    n = a.size
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m, dtype=float) / m
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0) + 1e-12
    return _sanitize_vec(out)


def _initial_sequence(m, rng, mode=None):
    if m <= 0:
        return None
    t = (np.arange(m, dtype=float) + 0.5) / float(m)
    if mode is None:
        mode = int(rng.integers(0, 7))

    if mode == 0:
        # Cosine series
        z = 1.0
        for k in range(1, 15):
            amp = float(rng.normal(0.0, 1.0 / (k + 1)))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            z = z + amp * np.cos(2.0 * np.pi * k * t + ph)
        p = np.log1p(np.exp(np.clip(z, -50.0, 50.0)))

    elif mode == 1:
        # Beta-like
        a = float(rng.uniform(0.1, 2.5))
        b = float(rng.uniform(0.1, 2.5))
        tt = np.clip(t, 1e-8, 1.0 - 1e-8)
        p = np.power(tt, a) * np.power(1.0 - tt, b)

    elif mode == 2:
        # Gaussian bumps
        K = int(rng.integers(1, 5))
        p = np.zeros(m)
        for ii in range(K):
            c = float(rng.uniform(0.0, 1.0))
            s = float(rng.uniform(0.02, 0.30))
            w = float(rng.uniform(0.5, 2.0))
            p = p + w * np.exp(-((t - c) / max(s, 1e-6)) ** 2)

    elif mode == 3:
        # Piecewise linear
        K = int(rng.integers(3, 12))
        knots = np.sort(rng.uniform(0.0, 1.0, K))
        heights = rng.lognormal(0.0, 1.0, K)
        p = np.interp(t, knots, heights)

    elif mode == 4:
        # Block function
        K = int(rng.integers(3, 20))
        edges = np.sort(rng.uniform(0.0, 1.0, K - 1))
        blocks = np.concatenate([[0.0], edges, [1.0]])
        heights = rng.lognormal(0.0, 1.0, K)
        p = np.zeros(m)
        for ii in range(K):
            s = int(round(blocks[ii] * m))
            e = max(s + 1, int(round(blocks[ii + 1] * m)))
            e = min(m, e)
            p[s:e] = float(heights[ii])

    elif mode == 5:
        # Two-block (asymmetric)
        split = int(rng.integers(int(0.3 * m), int(0.7 * m)))
        ratio = float(rng.choice([0.3, 0.5, 1.0, 2.0, 3.0]))
        p = np.ones(m)
        p[split:] = ratio

    else:
        # Smooth pulse
        c = float(rng.uniform(0.2, 0.8))
        s = float(rng.uniform(0.1, 0.4))
        p = np.exp(-((t - c) / s) ** 2)
        p = p + 0.1 * np.exp(-((t - (1 - c)) / s) ** 2)

    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0) + 1e-10
    s = float(p.sum())
    if s <= 0 or not np.isfinite(s):
        p = np.ones(m) / m
    return p / p.sum()


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses multiplicative descent, LP-based Frank-Wolfe, and block search.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60.0:
        budget = 2400.0

    start_time = time.monotonic()
    deadline = start_time + budget - 20.0  # 20s safety margin

    try:
        iseed = int(seed)
    except Exception:
        iseed = 12345
    rng = np.random.default_rng((iseed % 1000000007) + 7)

    # Get parent if available
    parent_arr = None
    try:
        parent_raw = height_sequence_1
        parent_arr = _sanitize_vec(np.asarray(parent_raw, dtype=float))
    except (NameError, Exception):
        parent_arr = None

    cands = []  # list of (score, array)

    def _add_cand(arr, sc):
        a = _sanitize_vec(arr)
        if a is None:
            return
        if not np.isfinite(sc):
            sc = _exact_score(a)
        cands.append((sc, a.copy()))
        if len(cands) > 20:
            cands.sort(key=lambda x: x[0])
            cands = cands[:15]

    if parent_arr is not None:
        sc = _exact_score(parent_arr)
        _add_cand(parent_arr, sc)
        print(f"Parent score: {sc:.6f}, length: {parent_arr.size}")

    # Choose working length
    n_work = 3072

    # --- Phase 1: Initial exploration at small length ---
    rem = deadline - time.monotonic()
    if rem > 300:
        small_n = 1024
        small_starts = []
        if parent_arr is not None:
            sp = _resample(parent_arr, small_n)
            if sp is not None:
                small_starts.append(sp)
        for mode in (5, 1, 2, 0):
            small_starts.append(_initial_sequence(small_n, rng, mode))

        for p0 in small_starts:
            if time.monotonic() >= deadline - 250:
                break
            p, sc = _run_mmd(p0, rng, 20.0, beta_start=20.0, beta_end=300.0)
            if p is not None:
                _add_cand(p, sc)

        print(f"Phase 1 done. Best so far: {cands[0][0]:.6f}")

    # --- Phase 2: Main multiplicative descent at working length ---
    rem = deadline - time.monotonic()
    if rem > 400:
        final_starts = []
        if parent_arr is not None:
            sp = _resample(parent_arr, n_work)
            if sp is not None:
                sp = _sanitize_vec(sp * np.exp(rng.normal(0.0, 0.03, n_work)))
                final_starts.append(sp)
        for sc_item, arr_item in sorted(cands)[:3]:
            sp = _resample(arr_item, n_work)
            if sp is not None:
                sp = _sanitize_vec(sp * np.exp(rng.normal(0.0, 0.02, n_work)))
                final_starts.append(sp)
        for mode in (5, 1):
            final_starts.append(_initial_sequence(n_work, rng, mode))

        # Score and sort starts
        scored_starts = []
        for sp in final_starts:
            if sp is not None:
                sc = _exact_score(sp)
                scored_starts.append((sc, sp))
        scored_starts.sort(key=lambda x: x[0])

        best_arr = None
        best_score = float("inf")

        # Run MMD on top starts
        top_starts = scored_starts[:3]
        t_alloc = max(60.0, (rem - 200.0) / max(1, len(top_starts)))
        for idx, (sc, sp) in enumerate(top_starts):
            rem = deadline - time.monotonic()
            if rem < 80:
                break
            t = min(t_alloc, rem - 100.0)
            p, sc2 = _run_mmd(sp, rng, max(20.0, t), beta_start=30.0, beta_end=800.0, verbose=True)
            if p is not None and np.isfinite(sc2):
                _add_cand(p, sc2)
                if sc2 < best_score:
                    best_arr = p.copy()
                    best_score = sc2
            print(f"  MMD start {idx}: score={sc2:.6f}")

        if best_arr is None and scored_starts:
            best_arr = scored_starts[0][1].copy()
            best_score = scored_starts[0][0]

        print(f"Phase 2 done. Best: {best_score:.6f}")

    # --- Phase 3: LP-based Frank-Wolfe ---
    rem = deadline - time.monotonic()
    if rem > 300 and best_arr is not None:
        print(f"Phase 3: LP Frank-Wolfe, {rem:.0f}s remaining")
        # Use a slightly smaller length for LP speed if needed
        n_lp = min(n_work, 2048) if n_work > 2048 else n_work
        p_lp = _resample(best_arr, n_lp) if n_lp != n_work else best_arr.copy()
        p_lp = _sanitize_vec(p_lp)
        if p_lp is not None:
            t_lp = max(60.0, min(600.0, 0.4 * rem))
            p, sc = _lp_frank_wolfe(p_lp, rng, t_lp, verbose=True)
            if p is not None and np.isfinite(sc):
                _add_cand(p, sc)
                if sc < best_score:
                    best_arr = p.copy()
                    best_score = sc
            print(f"  LP-FW result: {sc:.6f}")

    # --- Phase 4: Block search fine-tuning ---
    rem = deadline - time.monotonic()
    if rem > 150 and best_arr is not None:
        t_block = max(30.0, min(300.0, 0.3 * rem))
        p, sc = _block_search(best_arr, rng, t_block)
        if p is not None and np.isfinite(sc):
            _add_cand(p, sc)
            if sc < best_score:
                best_arr = p.copy()
                best_score = sc
        print(f"Phase 4 (block): {best_score:.6f}")

    # --- Phase 5: Second MMD pass with high beta ---
    rem = deadline - time.monotonic()
    if rem > 120 and best_arr is not None:
        t = max(30.0, min(250.0, 0.4 * rem))
        p, sc = _run_mmd(best_arr, rng, t, beta_start=100.0, beta_end=3000.0, verbose=True)
        if p is not None and np.isfinite(sc):
            _add_cand(p, sc)
            if sc < best_score:
                best_arr = p.copy()
                best_score = sc
        print(f"Phase 5 (MMD high beta): {best_score:.6f}")

    # --- Phase 6: Up-polish to larger length ---
    rem = deadline - time.monotonic()
    if rem > 300 and best_arr is not None:
        big_n = max(4096, 2 * int(best_arr.size))
        if big_n <= 8192:
            big = _resample(best_arr, big_n)
            if big is not None:
                t_big = max(60.0, min(200.0, 0.35 * rem))
                p, sc = _run_mmd(big, rng, t_big, beta_start=50.0, beta_end=1000.0)
                if p is not None and np.isfinite(sc):
                    _add_cand(p, sc)
                    if sc < best_score:
                        best_arr = p.copy()
                        best_score = sc
                        print(f"  Up-polish n={big_n}: {sc:.6f}")

                rem = deadline - time.monotonic()
                if rem > 100 and best_arr is not None:
                    t_block2 = max(30.0, min(100.0, 0.4 * rem))
                    p, sc = _block_search(best_arr, rng, t_block2)
                    if p is not None and np.isfinite(sc) and sc < best_score:
                        best_arr = p.copy()
                        best_score = sc

    # --- Final: also try LP at the current best length if time remains ---
    rem = deadline - time.monotonic()
    if rem > 200 and best_arr is not None:
        n_cur = int(best_arr.size)
        if n_cur <= 2048:
            t_lp2 = max(60.0, min(150.0, 0.5 * rem))
            p, sc = _lp_frank_wolfe(best_arr, rng, t_lp2)
            if p is not None and np.isfinite(sc) and sc < best_score:
                best_arr = p.copy()
                best_score = sc
                print(f"  Final LP: {sc:.6f}")

    # --- Final evaluation with official evaluator ---
    if best_arr is None and cands:
        best_arr = cands[0][1]
        best_score = cands[0][0]
    if best_arr is None:
        best_arr = np.ones(1024, dtype=float) / 1024
        best_score = _exact_score(best_arr)

    best_arr = _sanitize_vec(best_arr)
    if best_arr is None:
        best_arr = np.ones(1024) / 1024

    # Check with official evaluator
    final_list = [float(x) for x in best_arr]
    try:
        off_sc = float(evaluate_sequence(final_list))
        if np.isfinite(off_sc) and off_sc < best_score:
            best_score = off_sc
        print(f"Official eval: {off_sc:.6f} (internal: {best_score:.6f})")
    except Exception as ex:
        print(f"Eval error: {ex}")

    # Also check a few other candidates
    for sc_c, arr_c in sorted(cands)[:3]:
        if time.monotonic() >= deadline - 5:
            break
        lst = [float(x) for x in arr_c]
        try:
            sc_off = float(evaluate_sequence(lst))
            if np.isfinite(sc_off) and sc_off < best_score:
                best_score = sc_off
                final_list = lst
        except Exception:
            pass

    out = np.asarray(final_list, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s <= 0 or not np.isfinite(s):
        out = np.ones(1024) / 1024
    else:
        out = out / s

    elapsed = time.monotonic() - start_time
    print(f"Total time: {elapsed:.1f}s, final score: {best_score:.6f}, length: {out.size}")

    return [float(x) for x in out]
