from scipy.optimize import linprog, minimize

"""
Algorithm: Hybrid optimization for minimizing peak autocorrelation.
Phase 1: Log-sum-exp smoothed gradient optimization (L-BFGS-B) with adaptive tau.
Phase 2: Sequential Linear Programming (SLP) active-set polishing.
Phase 3: Random perturbation search.

_N = 0
_NFFT = 0

def _setup(n):
    global _N, _NFFT
    _N = int(n)
    _NFFT = 1 << (3 * _N - 2).bit_length()

def _fft_conv(a):
    fa = np.fft.rfft(a, _NFFT)
    L = 2 * _N - 1
    c = np.fft.irfft(fa * fa, _NFFT)[:L].real
    return np.maximum(c, 0.0)

def _score(a):
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    c = _fft_conv(a)
    return float(2 * _N * np.max(c) / (s * s))

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _eval_grad(z, tau):
    """Evaluate the smoothed objective and gradient in log-space."""
    n = _N
    a = _softmax(z)
    c = _fft_conv(a)
    m = float(np.max(c))
    diff = np.clip((c - m) / tau, -745.0, 0.0)
    e = np.exp(diff)
    se = float(e.sum())
    if se <= 0 or not np.isfinite(se):
        return 1e12, np.zeros(n)
    p = e / se
    fa = np.fft.rfft(a, _NFFT)
    fp = np.fft.rfft(p, _NFFT)
    h = np.fft.irfft(fp * np.conjugate(fa), _NFFT)[:n].real
    s_h = float(np.dot(h, a))
    g = 2.0 * a * (h - s_h)
    val = m + tau * np.log(se)
    pen = 1e-10
    val += 0.5 * pen * float(np.dot(z, z))
    g = g + pen * z
    return val, g

    bounds = [(-30.0, 30.0)] * n

        return _eval_grad(z, tau)

                       options={'maxfun': maxfun, 'ftol': 1e-14, 'pgtol': 1e-12, 'disp': False})
        return res.x if res.x is not None else z0
