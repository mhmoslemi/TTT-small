def project_simplex(v):
    v=np.asarray(v,dtype=np.float64).ravel()
    n=v.size
    if n==0: return v
    x=np.sort(v)[::-1]
    cssv=np.cumsum(x)
    idx=np.arange(1,n+1,dtype=np.float64)
    mask = x*idx > (cssv-1.0)
    if not np.any(mask):
       return np.ones(n)/n
    rho=np.where(mask)[0][-1]
    theta=(cssv[rho]-1.0)/(rho+1.0)
    return np.maximum(v-theta,0.0)

"""
Search for non-negative coefficient sequences minimizing
2*n*max(conv(a,a))/sum(a)^2.

The program uses:
- FFT-based full convolution for fast objective evaluation.
- A stochastic smoothed-subgradient / pair-coordinate hybrid local optimizer.
- Evaluation of asymmetric power-law and beta-distribution seeds.
- Interpolation of any previously found sequence to other lengths.
- Several sequence lengths, adaptive time allocation, and final polish.
"""

import time
import numpy as np


# ---------- basic helpers ----------

def normalize_p(a):
    """Normalize a to a non-negative probability vector."""
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        a = np.ones(128, dtype=np.float64)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-15:
        a = np.ones(a.size, dtype=np.float64)
    else:
        a = a / s
    return a


def full_conv(a):
    """Full linear convolution a*a via FFT. Input is assumed non-negative."""
    n = int(a.size)
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    b = np.fft.irfft(fa * fa, fsize)[:size]
    # Avoid returning pathological negative roundoff values that dominate in rare cases.
    # The maximum is always positive for valid inputs, so this is safe.
    return b


def score_unit(a):
    """Assumes sum(a)=1 and returns 2*n*max(a*a)."""
    n = int(a.size)
    if n < 1:
        return np.inf
    b = full_conv(a)
    if b.size == 0:
        return np.inf
    m = float(np.max(b))
    if not np.isfinite(m) or m < 0.0:
        m = 0.0
    return 2.0 * float(n) * m


def project_simplex(v):
    """Project v onto {x>=0, sum(x)=1}."""
    v = np.asarray(v, dtype=np.float64).ravel()
    n = int(v.size)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not np.any(mask):
        return np.ones(n, dtype=np.float64) / float(n)
    rho = int(np.where(mask)[0][-1])
    theta = (cssv[rho] - 1.0) / float(rho + 1)
    return np.maximum(v - theta, 0.0)


def softmax_weights(b, tau):
    """Stable softmax weights over convolution coefficients."""
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return b
    if tau <= 0.0 or not np.isfinite(tau):
        w = np.zeros(b.size, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
        return w
    x = b / tau
    x = x - float(np.max(x))
    e = np.exp(x)
    s = float(np.sum(e))
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros(b.size, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
        return w
    return e / s


def make_weights(b, rng, tau):
    """Choose subgradient weights, either exact peak, top few peaks, or smooth."""
    L = int(b.size)
    w = np.zeros(L, dtype=np.float64)
    r = float(rng.random())
    if r < 0.25:
        w[int(np.argmax(b))] = 1.0
    elif r < 0.55:
        k = int(rng.choice((1, 2, 3, 5, 8, 16)))
        if k >= L:
            w[:] = 1.0 / float(L)
        else:
            idx = np.argpartition(b, -k)[-k:]
            w[idx] = 1.0 / float(len(idx))
    else:
        w = softmax_weights(b, tau)
    s = float(np.sum(w))
    if s <= 0.0 or not np.isfinite(s):
        w = np.zeros(L, dtype=np.float64)
        w[int(np.argmax(b))] = 1.0
    else:
        w = w / s
    return w


def grad_weights(a, w):
    """Subgradient of weighted sum of convolution coefficients."""
    g = np.correlate(w, a, mode="valid")
    return 2.0 * g


def dual_candidate(a, w, blend, rng):
    """
    KKT-inspired jump: solve (K + ridge I)x = 1 where K_ij = w_{i+j},
    then blend x into a. This is optional and only used for moderate n.
    """
    n = int(a.size)
    if n < 2 or n > 320:
        return None
    ii = np.arange(n)[:, None]
    jj = np.arange(n)[None, :]
    K = w[ii + jj]
    tr = float(np.trace(K))
    ridge = (1e-5 + 1e-7 * max(1.0, abs(tr))) * np.eye(n, dtype=np.float64)
    ones = np.ones(n, dtype=np.float64)
    try:
        x = np.linalg.solve(K + ridge, ones)
    except np.linalg.LinAlgError:
        try:
            x = np.linalg.lstsq(K + ridge, ones, rcond=None)[0]
        except Exception:
            return None
    if not np.all(np.isfinite(x)):
        return None
    x = np.maximum(x, 0.0)
    s = float(np.sum(x))
    if s <= 1e-15:
        return None
    p = x / s
    cand = (1.0 - float(blend)) * a + float(blend) * p
    return project_simplex(cand)


# ---------- seeds ----------

def power_seed(n, alpha, beta):
    """Sequence p_i proportional to (i+alpha)^(-beta)."""
    x = np.arange(n, dtype=np.float64) + max(1e-6, float(alpha))
    if abs(float(beta)) < 1e-14:
        p = np.ones(n, dtype=np.float64)
    else:
        p = np.exp(-float(beta) * np.log(x))
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        p = np.ones(n, dtype=np.float64)
        s = float(n)
    return p / s


def power_grid(n, max_secs):
    """Small grid search over power-law seeds."""
    best = None
    best_sc = np.inf
    if n < 2 or max_secs <= 0.0:
        return best, best_sc
    t0 = time.monotonic()
    alphas = np.unique(np.concatenate((
        np.array([0.25, 0.5, 1.0, 2.0, 4.0]),
        np.logspace(0.0, np.log10(4.0 * n + 1.0), 22)
    )))
    betas = np.linspace(0.0, 0.95, 20)
    for beta in betas:
        for alpha in alphas:
            if time.monotonic() - t0 > max_secs:
                break
            p = power_seed(n, alpha, beta)
            sc = score_unit(p)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = p.copy()
        if time.monotonic() - t0 > max_secs:
            break
    return best, best_sc


def beta_seed(n, a, b):
    """Beta-distribution-like seed on [0, n-1]."""
    x = np.arange(n, dtype=np.float64) + 0.5
    xr = n + 0.5 - x
    with np.errstate(divide="ignore", invalid="ignore", over="ignore"):
        p = np.exp((float(a) - 1.0) * np.log(x) + (float(b) - 1.0) * np.log(xr))
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0:
        p = np.ones(n, dtype=np.float64)
        s = float(n)
    return p / s


def beta_grid(n, max_secs):
    """Small grid over beta-like seeds."""
    best = None
    best_sc = np.inf
    if n < 2 or max_secs <= 0.0:
        return best, best_sc
    t0 = time.monotonic()
    shapes = np.array([0.35, 0.5, 0.75, 1.0, 1.25, 1.6, 2.2])
    for a in shapes:
        for b in shapes:
            if time.monotonic() - t0 > max_secs:
                break
            p = beta_seed(n, a, b)
            sc = score_unit(p)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best = p.copy()
        if time.monotonic() - t0 > max_secs:
            break
    return best, best_sc


def interpolate_seed(src, n, rng, noise):
    """Interpolate a previously found sequence to length n."""
    src = np.asarray(src, dtype=np.float64).ravel()
    src = np.where(np.isfinite(src), src, 0.0)
    src = np.maximum(src, 0.0)
    if src.size == n:
        p = src.copy()
    elif src.size == 1:
        p = np.full(n, float(src[0]), dtype=np.float64)
    else:
        old = np.linspace(0.0, 1.0, src.size)
        new = np.linspace(0.0, 1.0, n)
        p = np.interp(new, old, src)
    if noise > 0.0:
        p = p + float(noise) * np.abs(rng.standard_normal(n))
    p = np.maximum(p, 0.0)
    return normalize_p(p)


def random_fourier_seed(n, rng):
    """Random low-frequency non-negative shape."""
    t = np.arange(n, dtype=np.float64)
    base = 0.7 + 0.6 * float(rng.random())
    m = int(rng.integers(2, 6))
    maxfreq = max(2, int(np.sqrt(n)))
    for _ in range(m):
        f = int(rng.integers(1, maxfreq + 1))
        amp = float(rng.normal(0.0, 0.5))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        base = base + amp * np.cos((2.0 * np.pi * f * t) / max(1, n) + ph)
    base = np.abs(base) + 0.05
    base = base + 0.01 * np.abs(rng.standard_normal(n))
    return normalize_p(base)


def random_poly_seed(n, rng):
    """Random positive polynomial shape."""
    t = np.linspace(0.0, 1.0, n)
    deg = int(rng.integers(1, 5))
    c = rng.normal(0.0, 1.0, deg)
    base = np.ones(n, dtype=np.float64)
    for k in range(deg):
        base = base + float(c[k]) * np.power(t, k + 1)
    base = np.abs(base) + 0.05
    base = base + 0.05 * np.abs(rng.standard_normal(n))
    return normalize_p(base)


def generate_seeds(n, prev, rng, grid_secs=2.0, beta_secs=1.0):
    """Return a small list of diverse normalized seeds of length n."""
    cands = []
    if prev is not None:
        cands.append(interpolate_seed(prev, n, rng, 0.0))
    if grid_secs > 0.2:
        p, _ = power_grid(n, grid_secs)
        if p is not None:
            cands.append(p)
    if beta_secs > 0.2:
        p, _ = beta_grid(n, beta_secs)
        if p is not None:
            cands.append(p)
    if prev is not None:
        cands.append(interpolate_seed(prev, n, rng, 0.03))
    cands.append(random_fourier_seed(n, rng))
    cands.append(random_fourier_seed(n, rng))
    cands.append(random_poly_seed(n, rng))
    cands.append(np.ones(n, dtype=np.float64) / float(n))

    out = []
    for p in cands[:8]:
        p = normalize_p(p)
        if np.all(np.isfinite(p)) and np.sum(p) > 0.0:
            out.append(p)
    if not out:
        out.append(np.ones(n, dtype=np.float64) / float(n))
    return out

def hybrid_optimize(a_in, budget, rng):
    """
    Stochastic hybrid local optimizer.

    Alternates between:
    - projected smoothed-subgradient steps with exact line search,
    - simplex pair-move steps with exact line search,
    - occasional KKT-inspired dual jumps and re-randomization.
    """
    a = normalize_p(a_in)
    n = int(a.size)
    if n < 1:
        return np.ones(128, dtype=np.float64) / 128.0, np.inf

    best = a.copy()
    best_sc = score_unit(best)
    if not np.isfinite(best_sc):
        best_sc = np.inf

    d_mom = np.zeros(n, dtype=np.float64)
    beta_m = 0.40
    it = 0
    last_imp = 0
    no_imp = 0
    tidx = 0
    tau_cycle = (0.30, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005, 0.0025)

    t0 = time.monotonic()
    end = t0 + max(1.0, float(budget))

    while time.monotonic() < end:
        it += 1

        b = full_conv(a)
        s = float(np.sum(a))
        if abs(s - 1.0) > 1e-8:
            a = a / s
            b = full_conv(a)
        m = float(np.max(b))
        if not np.isfinite(m) or m <= 0.0:
            break

        cur_sc = 2.0 * float(n) * m
        if cur_sc < best_sc:
            best_sc = float(cur_sc)
            best = a.copy()
            last_imp = it
            no_imp = 0

        tau = m * tau_cycle[tidx % len(tau_cycle)]
        if tau <= 1e-300 or not np.isfinite(tau):
            tau = 1e-300

        w = make_weights(b, rng, tau)
        d = grad_weights(a, w)
        if not np.all(np.isfinite(d)):
            d = np.zeros(n, dtype=np.float64)

        d_mom = beta_m * d_mom + (1.0 - beta_m) * d
        improved = False

        # Projected smoothed-subgradient step with exact line search.
        if it % 2 == 0:
            dirv = d_mom - float(np.mean(d_mom))
            dr = float(np.linalg.norm(dirv))
            if dr > 1e-12:
                base = 0.08 / dr
                for fac in (0.01, 0.05, 0.2, 1.0):
                    alpha = base * fac
                    cand = project_simplex(a - alpha * dirv)
                    sc = score_unit(cand)
                    if np.isfinite(sc) and sc < cur_sc:
                        a = cand
                        cur_sc = float(sc)
                        improved = True
                        if sc < best_sc:
                            best_sc = float(sc)
                            best = a.copy()
                            last_imp = it
                            no_imp = 0
                        break

        # Simplex pair-move step with exact line search.
        if not improved and n > 1:
            idx = np.argsort(d_mom)
            recs = idx[:3]
            dons = idx[-3:]
            r1 = int(rng.integers(0, 3))
            r2 = int(rng.integers(0, 3))
            sidx = int(dons[r1])
            tidx2 = int(recs[r2])
            if sidx == tidx2:
                sidx = int(dons[0])
                tidx2 = int(recs[0])
                if sidx == tidx2:
                    sidx = (sidx + 1) % n

            max_amt = 0.85 * float(a[sidx])
            if max_amt > 0.0:
                best_cand = None
                best_cand_sc = cur_sc
                for fac in (0.02, 0.08, 0.25, 0.60):
                    amt = max_amt * fac
                    if amt > float(a[sidx]):
                        amt = float(a[sidx])
                    cand = a.copy()
                    cand[sidx] -= amt
                    cand[tidx2] += amt
                    sc = score_unit(cand)
                    if np.isfinite(sc) and sc < best_cand_sc:
                        best_cand_sc = float(sc)
                        best_cand = cand.copy()

                if best_cand is not None and best_cand_sc < cur_sc:
                    a = best_cand
                    cur_sc = best_cand_sc
                    improved = True
                    if cur_sc < best_sc:
                        best_sc = float(cur_sc)
                        best = a.copy()
                        last_imp = it
                        no_imp = 0

        # Occasionally try a fully random pair move to escape local basins.
        if not improved and n > 1 and rng.random() < 0.12:
            i1 = int(rng.integers(0, n))
            i2 = int(rng.integers(0, n))
            if i1 != i2 and float(a[i1]) > 0.0:
                amt = 0.10 * float(a[i1])
                cand = a.copy()
                cand[i1] -= amt
                cand[i2] += amt
                sc = score_unit(cand)
                if np.isfinite(sc) and sc < cur_sc:
                    a = cand
                    cur_sc = float(sc)
                    improved = True
                    if sc < best_sc:
                        best_sc = float(sc)
                        best = a.copy()
                        last_imp = it
                        no_imp = 0

        if improved:
            no_imp = 0
        else:
            no_imp += 1
            if no_imp == 70:
                tidx += 1
                beta_m = float(rng.uniform(0.2, 0.7))
                no_imp = 0
            elif no_imp == 250:
                if n <= 320:
                    cand = dual_candidate(a, w, 0.35, rng)
                    if cand is not None:
                        sc = score_unit(cand)
                        if np.isfinite(sc) and sc < cur_sc:
                            a = cand
                            cur_sc = float(sc)
                            improved = True
                            if sc < best_sc:
                                best_sc = float(sc)
                                best = a.copy()
                                last_imp = it
                                no_imp = 0

                sig = 0.02 * min(1.0, max(0.2, 16.0 / max(1.0, float(np.sqrt(n)))))
                a = project_simplex(best + sig * rng.standard_normal(n))
                sc_now = score_unit(a)
                if not np.isfinite(sc_now) or sc_now > best_sc * 1.05 + 1e-6:
                    a = best.copy()
                d_mom = np.zeros(n, dtype=np.float64)
                no_imp = 0
            elif no_imp == 1500:
                sig = 0.05 * min(1.0, max(0.2, 32.0 / max(1.0, float(np.sqrt(n)))))
                a = project_simplex(best + sig * rng.standard_normal(n))

                if not np.isfinite(sc_now) or sc_now > best_sc * 1.10 + 1e-6:
                    a = best.copy()
                d_mom = np.zeros(n, dtype=np.float64)
                no_imp = 0
                last_imp = it

        if it - last_imp > 4000:
            break

    best = normalize_p(best)
    best_sc = score_unit(best)
    if not np.isfinite(best_sc):
        best_sc = np.inf
    return best, best_sc

    diff = abs(int(n) - int(ref_n))
    return 1.0 / (0.5 + float(n) / 192.0 + diff / 256.0)

                print(f"[screen] n={n}, no finite seed, overall={best_sc:.7f}")

    # ---------- Phase 2: refine the most promising lengths ----------
    if time.monotonic() < end_time - 45.0:
        # Ensure the global best is represented in the refinement list.
        if best_a is not None:
            bn = int(best_a.size)
            found = False
            for item in n_results:
                if int(item[0]) == bn:
                    found = True
                    # Make sure it has the global best copy.
                    if best_a.size == item[1].size:
                        item_sc = score_unit(item[1])
                        if best_sc < item_sc:
                            item[1] = best_a.copy()
                            item[2] = float(best_sc)
                    break
            if not found:
                n_results.append((bn, best_a.copy(), float(best_sc)))

        if n_results:
            n_results.sort(key=by_score)
            ref_items = n_results[:5]

            # Allocate decreasing fractions of remaining time.
            if len(ref_items) == 1:
                fracs = (1.0,)
            elif len(ref_items) == 2:
                fracs = (0.65, 0.35)
            elif len(ref_items) == 3:
                fracs = (0.50, 0.30, 0.20)
            elif len(ref_items) == 4:
                fracs = (0.42, 0.26, 0.18, 0.14)
            else:
                fracs = (0.38, 0.22, 0.17, 0.13, 0.10)

            for rank, item in enumerate(ref_items):
                if time.monotonic() >= end_time - 45.0:
                    break

                rem = max(0.0, end_time - time.monotonic())
                if rem <= 0:
                    break
                alloc = rem * fracs[rank]
                # Very long sequences are expensive; cap a bit.
                if int(item[0]) > 512:
                    alloc = min(alloc, 700.0)
                if int(item[0]) > 1024:
                    alloc = min(alloc, 300.0)
                if alloc < 10.0:
                    continue

                n, a0, sc0 = item
                print(f"[refine] rank={rank}, n={n}, start={sc0:.7f}, alloc={alloc:.1f}s")
                a_new, sc_new = hybrid_optimize(a0, alloc, rng)
                if np.isfinite(sc_new) and sc_new < best_sc:
                    best_sc = float(sc_new)
                    best_a = a_new.copy()
                # Update local item for the record.
                item[1] = a_new if np.isfinite(sc_new) else item[1]
                item[2] = sc_new if np.isfinite(sc_new) else item[2]
                print(f"[refine] rank={rank}, n={n}, end={sc_new:.7f}, overall={best_sc:.7f}")

    # ---------- Phase 3: final polish on the global best ----------
    if time.monotonic() < end_time - 15.0 and best_a is not None:
        rem = max(0.0, end_time - time.monotonic())
        if rem > 30.0:
            print(f"[polish] remaining={rem:.1f}s, n={int(best_a.size)}, score={best_sc:.7f}")
            a_new, sc_new = hybrid_optimize(best_a, rem, rng)
            if np.isfinite(sc_new) and sc_new < best_sc:
                best_sc = float(sc_new)
                best_a = a_new.copy()
                print(f"[polish] improved to {best_sc:.7f}")
            else:
                print(f"[polish] no improvement; best remains {best_sc:.7f}")

    # ---------- Fallback / final validation ----------
    if best_a is None or not np.all(np.isfinite(best_a)) or best_a.size < 1:
        # A safe fallback: optimized power-like seed if possible.
        try:
            p, _ = power_grid(128, 2.0)
            if p is None:
                p = np.ones(128, dtype=np.float64)
            best_a = normalize_p(p)
        except Exception:
            best_a = np.ones(128, dtype=np.float64)
        best_sc = score_unit(best_a)

    best_a = normalize_p(best_a)
    internal_score = score_unit(best_a)

    result = [float(x) for x in best_a]
    try:
        eval_score = float(evaluate_sequence(result))
        print(f"[final] n={len(result)}, internal={internal_score:.7f}, evaluator={eval_score:.7f}")
    except Exception:
        print(f"[final] n={len(result)}, internal={internal_score:.7f}, evaluator-unavailable/inf")

    return result
