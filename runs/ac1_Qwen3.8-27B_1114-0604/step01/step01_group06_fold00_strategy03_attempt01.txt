def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---
    if not isinstance(sequence, list):
        return np.inf

    if len(sequence) == 0:
        return np.inf

    total_length = 0
    # Check each element in the list for validity: non-boolean, non-NaN, non-inf
    for x in sequence:
        if isinstance(x, bool):
            return np.inf
        if not isinstance(x, (int, float)):
            return np.inf
        if np.isnan(x) or np.isinf(x):
            return np.inf
        # Accumulate the length
        total_length += 1

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Clip and convert to non-negative
    sequence = [max(0.0, x) for x in sequence]

    # Protect against too large numbers (by stateless clipping at 1000)
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    # Convolve two sequences (non-negative full convolution)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence) if b_sequence.size > 0 else 0.0
    sum_a = np.sum(sequence)

    # If after clipping the sum is still too small, score is np.inf
    if sum_a < 0.01:
        return np.inf

    return float(2.0 * n * max_b / (sum_a ** 2.0))
