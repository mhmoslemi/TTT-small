def _normalize(a, eps=1e-300):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a.fill(1.0)
        s = float(a.sum())
    a /= s
    a = np.maximum(a, eps)
    a /= float(a.sum())
    return a


def _conv_self(a, L, m):
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    c = np.ascontiguousarray(c)
    np.maximum(c, 0.0, out=c)
    return c


def _score_c(c, n):
    return float(2.0 * n * np.max(c))


def _score_a(a, L, m, n):
    return _score_c(_conv_self(a, L, m), n)


def _gradient_direction(a, c, L, m, n, temp):
    """Compute soft-max gradient direction for the self-convolution max."""
    v = n * c
    vmax = float(np.max(v))
    temp_eff = min(float(temp), 200.0 / max(vmax, 1e-12))
    if temp_eff <= 0.0:
        temp_eff = 1e-12
    z = temp_eff * (v - vmax)
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        w = np.ones(m) / m
    else:
        w = e / s
    Frev = np.fft.rfft(a[::-1], L)
    W = np.fft.rfft(w, L)
    g = np.fft.irfft(W * Frev, L][(n - 1):m]
    g = np.ascontiguousarray(g)
    np.maximum(g, 0.0, out=g)
    return g

def _soft_max_search(a, L, m, n, rng, time_limit, temp0=2.0, temp_end=300.0, use_momentum=True):

    deadline = time.monotonic() + float(time_limit)
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score_a(a, L, m, n)
    current_score = best_score

    # Momentum buffer
    mom = np.zeros(n, dtype=np.float64) if use_momentum else None
    mom_beta = 0.7

    temp = float(temp0)
    step = 0.15
    it = 0
    it_since_best = 0
    it_since_reheat = 0
    no_improve = 0

    while time.monotonic() < deadline:
        c = _conv_self(a, L, m)
        current_score = _score_c(c, n)

        if current_score < best_score - 1e-14:
            best_score = current_score
            best_a = a.copy()
            it_since_best = 0
            no_improve = 0
        else:
            it_since_best += 1
            no_improve += 1

        # Reheat strategy
        if it_since_best > 500:
            a = best_a.copy()
            temp = max(float(temp0), temp * 0.3)
            step = 0.15
            if mom is not None:
                mom.fill(0.0)
            it_since_best = 0

        if no_improve > 80:
            prior = _random_prior(n, rng)
            mix = 0.1 + 0.4 * float(rng.random())
            a = _normalize((1.0 - mix) * a + mix * prior)
            c = _conv_self(a, L, m)
            current_score = _score_c(c, n)
            if current_score < best_score:
                best_score = current_score
                best_a = a.copy()
            no_improve = 0
            it_since_best = 0
            temp = max(float(temp0), temp * 0.4)
            if mom is not None:
                mom.fill(0.0)

        # Compute gradient

        gmax = float(g.max())
        if not np.isfinite(gmax) or gmax <= 1e-30:

        # Momentum

def _coordinate_polish(a, L, m, n, rng, time_limit):
    """Coordinate descent polish: try small perturbations to individual coordinates."""
    deadline = time.monotonic() + float(time_limit)
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score_a(a, L, m, n)
    c = _conv_self(a, L, m)
    k_star = int(np.argmax(c))
    current_score = best_score
    delta = 0.05

    while time.monotonic() < deadline:
        # Update k_star
        k_star = int(np.argmax(c))
        improved = False

        # Random coordinate updates
        n_try = min(n, max(50, int(2 * n * 1e-4)))
        idxs = rng.integers(0, n, n_try)

        for j in idxs:
            # Try increasing and decreasing a[j]
            for sign in (1.0, -1.0):
                for d in (delta, 0.5 * delta, 2.0 * delta):
                    a2 = a.copy()
                    a2[j] = max(0.0, a2[j] + sign * d * a2[j])
                    s = a2.sum()
                    if s < 1e-15:
                        continue
                    a2 /= s
                    c2 = _conv_self(a2, L, m)
                    sc2 = _score_c(c2, n)
                    if sc2 < current_score - 1e-15:
                        a = a2
                        c = c2
                        current_score = sc2
                        improved = True
                        if sc2 < best_score:
                            best_score = sc2
                            best_a = a.copy()
                        break

        if not improved:
            delta = max(0.01, delta * 0.8)
            # Occasionally try a random restart
            if rng.random() < 0.1:
                prior = _random_prior(n, rng)
                a = _normalize(0.8 * a + 0.2 * prior)
                c = _conv_self(a, L, m)
                current_score = _score_c(c, n)
                delta = 0.05

    return best_a, best_score


def _mws_search(a, L, m, n, rng, time_limit):
    """Multiplicative weights search for a good mixing direction."""
    deadline = time.monotonic() + float(time_limit)
    best_a = _normalize(a)
    best_score = _score_a(best_a, L, m, n)

    F = np.fft.rfft(best_a, L)
    inner = 20

    while time.monotonic() < deadline:
        prior = _random_prior(n, rng)
        Frev = np.fft.rfft(best_a[::-1], L)
        x = prior.copy()
        avg = np.zeros(n)
        count = 0

        for it in range(inner):
            frac = float(it) / float(inner - 1) if inner > 1 else 0.0
            eta = 3.0 + 30.0 * frac

            X = np.fft.rfft(x, L)
            u = np.fft.irfft(F * X, L)[:m]
            np.maximum(u, 0.0, out=u)
            vu = n * u
            vma = float(np.max(vu))
            z = eta * (vu - vma)
            z = np.clip(z, -50.0, 0.0)
            y = np.exp(z)
            ys = float(y.sum())
            if ys > 0 and np.isfinite(ys):
                y /= ys
            else:
                y = np.ones(m) / m

            avg += x
            count += 1

            Y = np.fft.rfft(y, L)
            s = np.fft.irfft(Y * Frev, L)[(n - 1):m]
            np.maximum(s, 0.0, out=s)
            ps = n * s
            sma = float(np.max(ps))
            zx = -eta * (ps - sma)
            zx -= float(np.max(zx))
            zx = np.clip(zx, -50.0, 0.0)
            x = np.exp(zx)
            xs = float(x.sum())
            if xs > 0 and np.isfinite(xs):
                x /= xs

        if count > 0:
            direction = _normalize(0.5 * (avg / count) + 0.5 * x)
            # Line search mixing
            for lam in (0.1, 0.2, 0.35, 0.5, 0.7):
                cand = _normalize((1.0 - lam) * best_a + lam * direction)
                sc = _score_a(cand, L, m, n)
                if sc < best_score:
                    best_score = sc
                    best_a = cand
                    F = np.fft.rfft(best_a, L)

    return best_a, best_score


def _internal_exact(seq):
    if not isinstance(seq, list) or not seq:
        return float(np.inf)
    for x in seq:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return float(np.inf)
        if np.isnan(x) or np.isinf(x):
            return float(np.inf)
    a = [0.0 if float(x) < 0.0 else min(1000.0, float(x)) for x in seq]
    n = len(a)
    s = sum(a)
    if s < 0.01:
        return float(np.inf)
    b = np.convolve(a, a)
    mb = max(b)
    return float(2.0 * n * mb / (s * s))

            best_a = a.copy()

    if best_a is None:
        best_a = _normalize(np.ones(n, dtype=np.float64))
        best_score = _score_a(best_a, L, m, n)

    elapsed = time.monotonic() - start
    remaining = budget - elapsed

    if remaining < 10:
        return best_a, best_score, n

    # Main soft-max search
    lim1 = min(remaining * 0.5, 600.0)
    if lim1 > 10:
        tmp, sc = _soft_max_search(best_a, L, m, n, rng, lim1, 2.0, 200.0, True)
        if sc < best_score:
            best_score = sc
            best_a = tmp

    elapsed = time.monotonic() - start
    remaining = budget - elapsed

    # MWS phase
    if remaining > 30:
        lim2 = min(remaining * 0.2, 120.0)
        if lim2 > 10:
            tmp, sc = _mws_search(best_a, L, m, n, rng, lim2)
            if sc < best_score:
                best_score = sc
                best_a = tmp

    # Second soft-max with different params
    elapsed = time.monotonic() - start
    remaining = budget - elapsed
    if remaining > 30:
        lim3 = min(remaining * 0.3, 300.0)
        if lim3 > 10:
            rng2 = np.random.default_rng(seed + n)
            tmp, sc = _soft_max_search(best_a, L, m, n, rng2, lim3, 5.0, 400.0, True)
            if sc < best_score:
                best_score = sc
                best_a = tmp

    # Coordinate polish
    elapsed = time.monotonic() - start
    remaining = budget - elapsed
    if remaining > 20:
        lim4 = min(remaining * 0.4, 180.0)
        if lim4 > 5:
            rng3 = np.random.default_rng(seed + n + 1)
            tmp, sc = _coordinate_polish(best_a, L, m, n, rng3, lim4)
            if sc < best_score:
                best_score = sc
                best_a = tmp

    # Final short soft-max
    elapsed = time.monotonic() - start
    remaining = budget - elapsed
    if remaining > 15:
        lim5 = min(remaining - 5, 120.0)
        if lim5 > 5:
            tmp, sc = _soft_max_search(best_a, L, m, n, rng, lim5, 10.0, 500.0, False)
            if sc < best_score:
                best_score = sc
                best_a = tmp

    return best_a, best_score, n


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 1.0:
        budget = max(10.0, budget if np.isfinite(budget) else 2400.0)

    start = time.monotonic()
    hard_deadline = start + max(5.0, budget - 3.0)

    rng = np.random.default_rng(seed)

    # Try multiple n values, allocating time proportionally
    # Primary: 6121 (from parent). Secondary: try a few others.
    n_list = [6121, 5000, 7500]

    # Check if parent is available for n=6121
    has_parent = False
    try:
        base = height_sequence_1
        arr = np.asarray(base, dtype=np.float64).ravel()
        if arr.size > 100 and np.all(np.isfinite(arr)):
            has_parent = True
    except NameError:
        pass

    # If parent available, prioritize its n
    if has_parent:
        arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        n_parent = arr.size
        if n_parent not in n_list:
            n_list.insert(0, n_parent)

    # Allocate time: 60% to primary, 40% split among others
    primary_n = n_list[0]
    n_budget = {primary_n: budget * 0.60}
    if len(n_list) > 1:
        remaining_budget = budget * 0.40
        per_other = remaining_budget / (len(n_list) - 1)
        for i, ni in enumerate(n_list[1:]):
            n_budget[ni] = min(per_other, 300.0)

    all_results = []  # (score, seq, n)

    for ni in n_list:
        if time.monotonic() > hard_deadline - 10:
            break
        b = n_budget.get(ni, 60.0)
        if b < 15:
            continue
        rng_i = np.random.default_rng(seed + ni)
        try:
            best_a, best_sc, used_n = _search_n(ni, rng_i, seed, b)
            seq = [float(x) for x in best_a]
            all_results.append((best_sc, seq, used_n))
            print(f"n={ni}: internal score = {best_sc:.6f}, time used = {time.monotonic()-start:.1f}s")
        except Exception as e:
            print(f"n={ni} failed: {e}")

    if not all_results:
        n = primary_n
        a = _normalize(np.ones(n, dtype=np.float64))
        return [float(x) for x in a]

    # Evaluate all candidates exactly and return the best
    all_results.sort(key=lambda x: x[0])
    best_seq = all_results[0][1]
    best_exact = float(np.inf)

    for sc, seq, ni in all_results:
        if time.monotonic() > hard_deadline - 5:
            break
        val = _exact_score(seq)
        if np.isfinite(val) and val < best_exact:
            best_exact = val
            best_seq = seq
            print(f"Exact score for n={ni}: {val:.6f}")

    if best_exact == float(np.inf):
        best_seq = all_results[0][1]

    total_time = time.monotonic() - start
    print(f"Total time: {total_time:.1f}s, best exact score: {best_exact}")
    return best_seq
