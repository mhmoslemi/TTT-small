import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def full_conv(a):
    n = a.size
    size = 2*n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def score(a):
    n = a.size
    return 2.0 * n * float(np.max(full_conv(a)))

def conv_matrix(a):
    """Build sparse convolution matrix A where (A @ b)[k] = (a*b)[k]."""
    n = a.size
    rows = []
    cols = []
    data = []
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
            data.append(a[k - i])
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))

def lp_minimize_t(a0):
    """
    Solve LP: minimize t s.t. sum(a)=1, a>=0, and linearized (a0*a)(k) + 2*a0*(a-a0)(k) <= t.
    Equivalent: 2*sum_j a0[k-j]*a_j - (a0*a0)(k) <= t for all k.
    """
    n = a0.size
    A = conv_matrix(a0)  # (2n-1) x n, where A[k,j] = a0[k-j]
    
    # Constraint: 2*A @ a - t <= (a0*a0)
    # In standard form: [2A, -I] @ [a; t] <= a0*a0
    
    # But scipy linprog uses A_ub @ x <= b_ub
    # x = [a_0, ..., a_{n-1}, t]
    
    # 2A @ a - t <= a0*a0
    # [2A | -e] @ [a; t] <= a0*a0  where e is a column of 1s
    
    aa0 = full_conv(a0)
    
    # Build the combined matrix
    # Since A is sparse (2n-1 x n), the combined matrix is (2n-1) x (n+1)
    # Last column is -1
    
    import scipy.sparse as sp
    minus_one = sp.csr_matrix(-np.ones((2*n-1, 1)))
    A_combined = sp.hstack([2*A, minus_one], format='csr')
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    # Equality constraint: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(0, None)]  # a_i >= 0, t >= 0
    
    res = linprog(c, A_ub=A_combined, b_ub=aa0, A_eq=A_eq, b_eq=b_eq, 
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x[:n], res.x[n]
    else:
        return a0.copy(), np.inf

def lp_maximize_sum(a, M):
    """
    Solve LP: maximize sum(b) s.t. b >= 0, (a*b)(k) <= M for all k.
    """
    n = a.size
    A = conv_matrix(a)  # (2n-1) x n
    
    c = -np.ones(n)  # minimize -sum(b)
    
    bounds = [(0, None)] * n
    
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=bounds, method='highs')
    
    if res.status == 0:
        return res.x
    else:
        return a.copy()

def project_simplex(v):
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def subgrad_step(a, tau, n_scales=3):
    """Smoothed subgradient step."""
    n = a.size
    b = full_conv(a)
    g_total = np.zeros(n)
    for s in range(n_scales):
        t = tau * (2.0 ** s)
        shift = b / t - b.max() / t
        w = np.exp(shift)
        w /= w.sum()
        g_total += 2.0 * np.correlate(w, a, mode='valid')
    g = g_total / n_scales
    return g, b

def subgrad_optimize(a, time_budget, rng):
    """Subgradient optimization with momentum."""
    n = a.size
    a = project_simplex(a.copy())
    best_a = a.copy()
    best_sc = score(a)

    tau = 0.03 * float(np.max(full_conv(a)))
    vel = np.zeros(n)
    beta = 0.5
    iters = 0
    last_improve = 0
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)

        g, _ = subgrad_step(a, tau)
        vel = beta * vel + (1 - beta) * g
        a_new = project_simplex(a - eta * vel)
        
        b_new = full_conv(a_new)
        new_sc = 2.0 * n * float(np.max(b_new))
        
        if new_sc < best_sc or rng.random() < 0.02:
            a = a_new
        else:
            a = project_simplex(a - 0.5 * eta * vel)
        
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = iters

        if iters - last_improve > 2000:

            vel = np.zeros(n)
            eta = 0.1 + 0.15 * rng.random()
            last_improve = iters
    
    return best_a, best_sc

```python
def make_conv_structure(n):
    """Create the sparse structure for the convolution matrix."""
    rows = []
    cols = []
    for k in range(2*n - 1):
        i_start = max(0, k - n + 1)
        i_end = min(n - 1, k)
        for i in range(i_start, i_end + 1):
            rows.append(k)
            cols.append(i)
    return np.array(rows), np.array(cols)

def make_conv_matrix(a, rows, cols):
    """Fill in values for the convolution matrix."""
    n = a.size
    data = a[rows - cols]  # a[k-i] where k=rows, i=cols
    return csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
```

Wait, `rows - cols` gives `k - i` for each entry. And `a[k-i]` is what we want. But we need to make sure `0 <= k-i < n`, which is guaranteed by the construction.

This is much faster than building the matrix from scratch each time.

`a_0 * a_0 = [0.25, 0.31, 0.28, 0.12, 0.04]` (length 5 = 2*3-1).

The LP: minimize t
s.t. sum(a) = 1, a >= 0
     2*(a_0*a)(k) - (a_0*a_0)(k) <= t for k=0,...,4

For k=0: 2*a_0[0]*a[0] - 0.25 <= t → 1.0*a[0] - 0.25 <= t
For k=1: 2*(a_0[0]*a[1] + a_0[1]*a[0]) - 0.31 <= t → 2*(0.5*a[1] + 0.3*a[0]) - 0.31 <= t
For k=2: 2*(a_0[0]*a[2] + a_0[1]*a[1] + a_0[2]*a[0]) - 0.28 <= t
For k=3: 2*(a_0[1]*a[2] + a_0[2]*a[1]) - 0.12 <= t
For k=4: 2*a_0[2]*a[2] - 0.04 <= t → 0.4*a[2] - 0.04 <= t

This looks correct. The LP will find the `a` that minimizes the maximum of these linear expressions.

Now, the actual `(a*a)(k)` is quadratic, not linear. The LP gives a lower bound on the actual max (since the linearization underestimates the quadratic for `a` in the direction away from `a_0`... actually, it's a tangent plane, so it can be either above or below the true value).

In any case, the LP solution is a good direction to move, and the line search will handle the rest.
