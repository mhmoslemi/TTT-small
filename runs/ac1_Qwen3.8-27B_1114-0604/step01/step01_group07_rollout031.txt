Optimized low-peak nonnegative step-function search.
Uses multi-phase approach: diverse initializations, then subgradient refinement
with large active sets and adaptive step sizing.

import numpy as np
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _score(a):
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * a.size * mx / (s * s))

    a = np.clip(a, 0.0, 1000.0)

    arr = np.asarray(a, dtype=np.float64).copy()

    max_lr = 50.0
