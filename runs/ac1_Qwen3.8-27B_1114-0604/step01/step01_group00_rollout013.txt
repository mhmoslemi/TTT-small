    """Mirror descent (exponentiated gradient) step."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)

    if s < 1e-300:

    """Euclidean projection onto {x >= 0, sum x = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
