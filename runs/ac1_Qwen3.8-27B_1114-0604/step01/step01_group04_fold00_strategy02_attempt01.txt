def evaluate_sequence(sequence):

    try:

    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]

        prev_a_low = pulseogenesis(prev_n)
        prev_a_cos = np.array([1 + 1.5*np.cos(2*np.pi*i/prev_n) for i in range(prev_n)])
        prev_a_sin = np.array([np.sin(0.5*i) for i in range(prev_n)])
        sequence_families = [prev_a_low, prev_a_cos, prev_a_sin]
        prev_score = evaluate_sequence(prev_a_cos)
        # Retrain the best from these if we have time
        best_prev = prev_a_cos
        if prev_score < 1.8:
            best_prev = prev_a_sin
        
        sequence_families.insert(0, best_prev)
    else:
        # Start fresh without parent
        sequence_families = [np.random.rand(1024)] + [np.ones(1024)] + [pulseogenesis(1024)]
    
    candidates = advanced_adaptation(sequence_families, budget)
    best_candidate = harmony_search(len(candidates[0]), budget)
    
    # Final polish
    polished_candidate = best_candidate + 0.01 * rng.standard_normal(len(best_candidate))
    polished_candidate = polished_candidate / np.sum(polished_candidate) * 2 * len(polished_candidate) / np.sqrt(3)
    polished_score = evaluate_sequence(polished_candidate)
    
    # Compare to the best found among families
    if polished_score < evaluate_sequence(best_candidate):
        best_candidate = polished_candidate
        
    # Time for finalization
    final_time = time.time()
    duration = final_time - initial_time
    
    return list(best_candidate)
