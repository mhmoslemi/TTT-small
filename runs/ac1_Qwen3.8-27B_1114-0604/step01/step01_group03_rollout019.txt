def _normalize(a):
    """Project onto probability simplex (assigns to sum-1, non-negative)."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return np.ones_like(a) / len(a)
    return a / s


def _self_conv(a, L):
    """Full self-convolution via FFT. Returns length 2n-1."""
    n = len(a)
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)
    return c[:2 * n - 1]


def _hankel_mv(lam, a, L):
    """Compute (Q @ a) where Q[i,j] = lam[i+j]. Returns length n."""
    # This is the cross-correlation of lam (length 2n-1) with a (length n)
    # Result[m] = sum_j lam[m+j] * a[j] for m = 0, ..., n-1
    n = len(a)
    fl = np.fft.rfft(lam, L)
    fa = np.fft.rfft(a, L)
    # Cross-correlation via FFT
    # (lam_star a)[m] = sum_k lam[k] * a[k-m]
    # We want result[m] = sum_j lam[m+j] * a[j] = sum_k lam[k] * a[k-m] (with k = m+j)
    # This is the cross-correlation at lag m
    result = np.fft.irfft(fl * np.conj(fa), L).real[:n]
    return result


def _simplex_project(a):
