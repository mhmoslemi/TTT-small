"""
Low-peak nonnegative step-function search via smoothed-max gradient optimization.

Uses top-K softmax-weighted gradient of the convolution peak, with multiplicative
updates and adaptive learning rate. Multiple initializations (sinusoidal,
superposition, random anti-palindromic) are explored across several sequence
lengths (2048-16384) to find the best candidate.

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _fft_corr(w, a):
    """Compute g[m] = sum_j a[j] * w[j+m] for m=0..n-1.
    a has length n, w has length 2n-1."""
    n = a.size
    N = _next_pow2(3 * n)
    w_pad = np.zeros(N, dtype=np.float64)
    w_pad[:2 * n - 1] = w
    a_pad = np.zeros(N, dtype=np.float64)
    a_pad[:n] = a
    W = np.fft.rfft(w_pad)
    A = np.fft.rfft(a_pad)
    result = np.fft.irfft(W * np.conj(A), N)
    return result[:n]


def _project_simplex(v):
    """Project v onto probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho_candidates = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho_candidates) == 0:
        return np.ones(n) / n
    rho = rho_candidates[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _score_fast(a):
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    b = _fft_convolve(a)
    if b.size == 0:
        return np.inf
    return float(2.0 * a.size * b.max() / (s * s))


def _make_sine(n, amp, mode=1):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / n
    a = (1.0 + amp * np.sin(mode * theta)) / n
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / n
    return a / s


def _make_superposition(n, amps, modes):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / n
    a = np.ones(n, dtype=np.float64)
    for amp, mode in zip(amps, modes):
        a += amp * np.sin(mode * theta)
    a /= n
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / n
    return a / s


def _make_random_antipal(n, seed=0, num_modes=3):
    rng = np.random.RandomState(seed)
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / n
    modes = rng.randint(1, max(2, n // 4), size=num_modes)
    amps = rng.uniform(0.0, 0.8, size=num_modes)
    a = np.ones(n, dtype=np.float64)
    for j in range(num_modes):
        a += amps[j] * np.sin(modes[j] * theta)
    a /= n
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / n
    return a / s


def _optimize_grad(a, time_budget, verbose_name=""):
    """Optimize a using smoothed-max gradient with multiplicative updates."""
    n = a.size
    a = a.astype(np.float64).copy()
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 0.01:
        return a, np.inf
    a = a / s

    best_a = a.copy()
    best_score = _score_fast(a)
    if not np.isfinite(best_score):
        return best_a, best_score

    lr = 0.3
    tau = 5.0 * n
    K = min(50, 2 * n - 1)
    stuck_count = 0
    last_improve_score = best_score

    start = time.monotonic()
    it = 0
    eval_count = 0

    while time.monotonic() - start < time_budget:
        it += 1

        # Compute gradient
        b = _fft_convolve(a)
        bmax = float(b.max())

        # Adaptive K: increase as optimization progresses
        if it % 200 == 0:
            K = min(K + 10, 2 * n - 1)
            tau = tau * 1.5  # concentrate more
            K = min(K, 2 * n - 1)

        if K < b.size:
            topk_idx = np.argpartition(b, -K)[-K:]
        else:
            topk_idx = np.arange(b.size)
        topk_vals = b[topk_idx]

        # Softmax weights
        w = np.exp(tau * (topk_vals - topk_vals.max()))
        wsum = w.sum()
        if wsum < 1e-30:
            break
        w = w / wsum

        w_full = np.zeros(2 * n - 1, dtype=np.float64)
        w_full[topk_idx] = w

        # Gradient: g[i] = 2 * sum_k w_k * a[k-i]
        g = _fft_corr(w_full, a)
        g *= 2.0

        # Project to tangent space (remove mean for sum constraint)
        g = g - g.mean()

        # Multiplicative update
        exponent = -lr * g
        exponent = np.clip(exponent, -2.0, 2.0)
        a_new = a * np.exp(exponent)
        a_new = np.where(np.isfinite(a_new), a_new, 0.0)
        a_new = np.clip(a_new, 0.0, None)
        ts = a_new.sum()
        if ts < 0.01 or not np.isfinite(ts):
            lr = lr * 0.5
            continue
        a_new = a_new / ts

        # Evaluate (every few iterations to save time)
        if it % 5 == 0 or it <= 10:
            sc = _score_fast(a_new)
            eval_count += 1
            if np.isfinite(sc) and sc < best_score:
                best_a = a_new.copy()
                best_score = sc
                a = a_new
                stuck_count = 0
                # Adaptively increase lr on success

                        lr = 0.5

                        tau = tau * 2.0

                else:

            print(f"[polish] Final polishing with {remaining:.0f}s budget")
            # Alternate between multiplicative and additive
            half = remaining // 2
            pa, ps = _optimize_grad(best_a, half)
            if ps < best_score:
                best_a, best_score = pa, ps
                print(f"  [polish1] score={best_score:.8f}")
            pa2, ps2 = _optimize_grad_additive(best_a, remaining - half - 10)
            if ps2 < best_score:
                best_a, best_score = pa2, ps2
                print(f"  [polish2] score={best_score:.8f}")

    # Validate with evaluate_sequence
    print(f"[validate] Best internal score: {best_score:.8f}")

    # Build list of candidates to validate (best first)
    candidates = []
    if best_a is not None:
        candidates.append(best_a)
    # Add the best init as fallback
    for sc, name, init_a in scored_inits[:2]:
        if init_a is not None and not any(np.array_equal(c, init_a) for c in candidates):
            candidates.append(init_a)
    # Constant fallback
    candidates.append(np.ones(2048, dtype=np.float64) / 2048.0)

    best_actual = np.inf
    best_out = None

    for cand in candidates[:5]:
        if time.monotonic() > hard_deadline:
            break
        arr = np.asarray(cand, dtype=np.float64)
        arr = np.clip(arr, 0.0, 1000.0)
        s = arr.sum()
        if s < 0.01:
            continue
        arr = arr / s
        try:
            real = float(evaluate_sequence(arr.tolist()))
        except Exception:
            real = np.inf
        if np.isfinite(real) and real < best_actual:
            best_actual = real
            best_out = arr
            print(f"[validate] n={arr.size} actual={real:.8f}")
            if best_actual < 1.45:
                break

    if best_out is None:
        best_out = np.ones(2048, dtype=np.float64) / 2048.0
        print("[fallback] constant sequence")

    print(f"[done] Best score: {best_actual:.8f}")
    return [float(x) for x in best_out]
