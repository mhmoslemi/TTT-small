import numpy as np

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _score(a):
    arr = np.asarray(a, dtype=np.float64)

        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return float(2.0 * arr.size * mx / (s * s))

                    # grad[j] += w_k * arr[k-j] for j in [j_min, j_max]
                    # m = k-j ranges from k-j_max to k-j_min
                    m_start = k_int - j_max
                    m_end = k_int - j_min
                    grad[j_min:j_max+1] += w_k * arr[m_start:m_end+1][::-1]
        else:
            # Use FFT-based approach
            # grad[j] = sum_k w_k * arr[k-j]
            # Pad w to length 2n (w_full), pad arr to length 2n (w_full length)
            # conv_full = ifft(fft(w_full, 4n) * fft(arr_padded, 4n))
            # Then grad[j] = conv_full[j + n - 1] ... I need to figure out the offset.
            
            # Actually, let's think of it as:
            # Define W of length 2n-1 (the weight vector over conv lags)
            # Define A of length n (the sequence)
            # We want: grad[j] = sum_{k=0}^{2n-2} W[k] * A[k-j] for 0<=k-j<n
            #          = sum_{k=0}^{2n-2} W[k] * A[k-j] * 1{0<=k-j<n}
            # 
            # Let m = k-j: grad[j] = sum_m W[m+j] * A[m] * 1{0<=m<n, 0<=m+j<=2n-2}
            # For j in [0, n-1] and m in [0, n-1], m+j is in [0, 2n-2], so the indicator is always true.
            # grad[j] = sum_{m=0}^{n-1} W[m+j] * A[m]
            # 
            # This is the cross-correlation of W and A, evaluated at lags 0, 1, ..., n-1.
            # In numpy terms: grad = np.correlate(W, A, mode='valid')? No...
            # 
            # Cross-correlation: (W ⊛ A)[j] = sum_m W[m+j] * conj(A[m])
            # For real: (W ⊛ A)[j] = sum_m W[m+j] * A[m]
            # In numpy: np.correlate(W, A, 'valid') gives sum_m W[m+j]*A[m] for j=0..len(W)-len(A)
            # len(W) = 2n-1, len(A) = n, so we get n-1 values. We want n values.
            # 
            # Let me just pad and use FFT:
            N_fft = _next_pow2(2 * (2*n - 1))
            W_ft = np.fft.rfft(w_full, N_fft)
            A_ft = np.fft.rfft(arr, N_fft)
            # Cross-correlation via FFT: ifft(fft(W) * conj(fft(A)))
            # = ifft(fft(W) * fft(A)) for real
            corr = np.fft.irfft(W_ft * A_ft, N_fft)
            # corr[j] = sum_m W[m+j] * A[m] for j = 0, 1, ..., len(W)-1 (with wraparound)
            # We want j = 0, 1, ..., n-1
            grad = corr[:n].copy()
            # But we need to make sure the wraparound doesn't affect the first n values.
            # Since W has length 2n-1 and A has length n, for j <= n-1, m+j <= n-1+j <= 2n-2,
            # which is within the valid range of W. So no wraparound issue for j < n.
            # Actually, with FFT, there IS wraparound. corr[j] = sum_{m=0}^{N_fft-1} W_padded[m] * A_padded[(m+j) mod N_fft]
            # For j < n and m < n: (m+j) < 2n < N_fft (since N_fft >= 2*(2n-1) > 2n), so no wraparound.
            # And W_padded[m] = 0 for m >= 2n-1, A_padded[m] = 0 for m >= n.
            # So corr[j] = sum_{m=0}^{n-1} W[m] * A_padded[(m+j)] for j < n
            # = sum_{m=0}^{n-1} W[m] * A[(m+j) mod n]... no, A_padded has length N_fft with zeros after n.
            # A_padded[m+j] = A[m+j] if m+j < n, else 0.
            # So corr[j] = sum_{m=0}^{n-1-j} W[m] * A[m+j] for j < n.
            # But we want sum_{m=0}^{n-1} W[m+j] * A[m].
            # These are different!
            
            # Let me redefine. I want: grad[j] = sum_{m=0}^{n-1} W[m+j] * A[m]
            # This is the cross-correlation (A ⊛ W)[j] = sum_m A[m] * W[m+j]
            # In FFT: ifft(fft(W) * conj(fft(A_reversed)))... this is getting confusing.
            
            # Let me just use the direct approach with a limit on K.
            # For K > 100, subsample the active lags.
            if len(active_idx) > 100:
                # Subsample to 100
                perm = np.random.choice(len(active_idx), 100, replace=False)
                active_idx = active_idx[perm]
                conv_active = conv[active_idx]
                w = np.exp(beta * (conv_active - conv_max))
                w_sum = float(w.sum())
                if w_sum < 1e-30:
                    w = np.ones_like(w) / len(w)
                else:
                    w = w / w_sum
            
            grad = np.zeros(n, dtype=np.float64)
            for idx, k in enumerate(active_idx):
                k_int = int(k)
                w_k = float(w[idx])
                j_min = max(0, k_int - n + 1)
                j_max = min(n - 1, k_int)
                if j_max >= j_min:
                    m_start = k_int - j_max
                    m_end = k_int - j_min
                    grad[j_min:j_max+1] += w_k * arr[m_start:m_end+1][::-1]
        
        if not np.any(grad):
            break
        
        # Normalize gradient for stable updates
        grad_max = float(np.max(np.abs(grad)))
        if grad_max < 1e-30:
            break
        grad_norm = grad / grad_max
        
        # Mirror descent update
        alpha = lr
        arr = arr * np.exp(-alpha * grad_norm)
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.maximum(arr, 0.0)
        
        s = float(arr.sum())
        if not np.isfinite(s) or s < 1e-10:
            arr = best.copy()
            lr *= 0.5
            continue
        arr = arr / s
        
        # Check score periodically
        if it % check_interval == 0:
            sc = _score(arr)
            if np.isfinite(sc) and sc < best_score:
                best = arr.copy()
                best_score = sc
                no_improve_count = 0
                # Maybe increase beta (focus more on the max)
                beta = min(beta_max, beta * 1.05)
            else:
                no_improve_count += 1
                if no_improve_count % 10 == 0:
                    # Decrease learning rate
                    lr *= 0.8
                    if lr < 0.001:
                        break
                if no_improve_count > 50:
                    break
        
        # Adaptive beta: increase slowly
        if it % 100 == 0:
            beta = min(beta_max, beta * 1.01)
    
    return best, best_score
