def _mws_direction(f, L, m, n, prior, inner, eta_start=2.0, eta_end=40.0):

    inner = max(1, int(inner))

    F = np.fft.rfft(f, L)
    Frev = np.fft.rfft(f[::-1], L)

    x = prior.copy()
    y = np.full(m, 1.0 / float(m), dtype=np.float64)
    avg = np.zeros_like(x)
    count = 0
    logprior = np.log(np.maximum(prior, 1e-300))

    for it in range(inner):
        frac = 0.0 if inner <= 1 else float(it) / float(inner - 1)
        eta = float(eta_start) + (float(eta_end) - float(eta_start)) * frac

        X = np.fft.rfft(x, L)
        u = np.fft.irfft(F * X, L)[:m]
        np.maximum(u, 0.0, out=u)

        vu = n * u
        vma = float(np.max(vu))
        z = eta * (vu - vma)
        z = np.clip(z, -50.0, 0.0)
        y = np.exp(z)
        ys = float(y.sum())
        if not np.isfinite(ys) or ys <= 0.0:
            y.fill(1.0 / float(m))
        else:
            y /= ys

        avg += x
        count += 1

        Y = np.fft.rfft(y, L)
        s = np.fft.irfft(Y * Frev, L)[(n - 1):m]
        np.maximum(s, 0.0, out=s)

        ps = n * s
        sma = float(np.max(ps))

        zx -= float(np.max(zx))

        xs = float(x.sum())
        if not np.isfinite(xs) or xs <= 0.0:
            x = prior.copy()
        else:
            x /= xs

    if count == 0:
        return x
    return _normalize(0.5 * (avg / float(count)) + 0.5 * x)

def _internal_exact(seq):
    if not isinstance(seq, list) or not seq:
        return float(np.inf)
    for x in seq:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return float(np.inf)
        if np.isnan(x) or np.isinf(x):
            return float(np.inf)
    a = [0.0 if float(x) < 0.0 else min(1000.0, float(x)) for x in seq]
    n = len(a)
    s = sum(a)
    if s < 0.01:
        return float(np.inf)
    b = np.convolve(a, a)
    mb = max(b)
    return float(2.0 * n * mb / (s * s))
