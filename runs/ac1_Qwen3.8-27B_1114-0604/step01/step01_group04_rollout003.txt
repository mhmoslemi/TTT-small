import numpy as np
import time
from scipy.optimize import linprog

def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def conv_with(a, b):
    """Compute conv(a, b) via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
    return np.fft.irfft(fa * fb, fsize)[:size]

def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))

def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def build_constraint_matrix(a):
    """Build the (2n-1) x n matrix A where conv(a,g)_k = (A @ g)_k."""
    n = a.size
    A = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        # conv(a,g)_k = sum_j a[j] * g[k-j] for valid j
        # j must satisfy: 0 <= j < n and 0 <= k-j < n
        # i.e., max(0, k-n+1) <= j <= min(k, n-1)
        jmin = max(0, k - n + 1)
        jmax = min(k, n - 1)
        for j in range(jmin, jmax + 1):
            A[k, j] = a[k - j]
    return A

def solve_lp_direction(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns g."""
    n = a.size
    A = build_constraint_matrix(a)
    c = -np.ones(n)  # minimize -sum(g)
    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n, method='highs')
    if res.status == 0:
        return res.x
    return None

def lp_iterate(a, max_iters=200, tol=1e-10, verbose=False):
    """
    LP-based iterative optimization.
    a is normalized to sum=1.
    Returns (best_a, best_score).
    """

    b = full_conv(a)
    M = float(np.max(b))
    best_a = a.copy()
    best_sc = 2.0 * n * M

    for it in range(max_iters):
        g = solve_lp_direction(a, M)
        if g is None:
            break
        
        g_sum = g.sum()
        if g_sum < 1e-12:
            break
        g = g / g_sum  # normalize to sum=1
        
        # Line search on t
        best_t = 0.0
        best_peak = M
        for t in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05]:
            a_trial = (1 - t) * a + t * g
            a_trial = project_simplex(a_trial)
            b_trial = full_conv(a_trial)
            peak = float(np.max(b_trial))
            if peak < best_peak:
                best_peak = peak
                best_t = t
        
        if best_t == 0.0:
            break
        
        a = (1 - best_t) * a + best_t * g
        a = project_simplex(a)
        b = full_conv(a)
        M = float(np.max(b))
        cur_sc = 2.0 * n * M
        
        if verbose and it % 20 == 0:

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
        
        if M - best_peak < tol:
            break

    return best_a, best_sc

"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using:
1. LP-based iterative refinement (solve LP at each step to find optimal direction)
2. Projected smoothed-subgradient descent with momentum
3. Multiple sequence lengths (128-1024) and diverse initializations
4. Final local polish with adaptive perturbations

import numpy as np
import time
from scipy.optimize import linprog
from scipy.sparse import csr_matrix
from scipy.sparse import coo_matrix

def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def conv_ab(a, b):
    """Compute conv(a, b) via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
    return np.fft.irfft(fa * fb, fsize)[:size]

def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v.copy()
    v = v.astype(np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def compute_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))

def build_sparse_A(a):
    """Build sparse (2n-1) x n matrix where conv(a,g)_k = (A@g)_k."""
    n = a.size
    rows = []
    cols = []
    vals = []
    for j in range(n):
        # A[k, j] = a[k-j] for k-j in [0, n-1], i.e., k in [j, j+n-1]
        kmin = j
        kmax = min(j + n - 1, 2 * n - 2)
        for k in range(kmin, kmax + 1):
            d = k - j
            if 0 <= d < n:
                rows.append(k)
                cols.append(j)
                vals.append(a[d])
    mat = coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    return mat.tocsr()

def solve_lp(a, M):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0. Returns normalized g or None."""
    n = a.size
    A = build_sparse_A(a)
    c = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True, 'time_limit': 30})
        if res.status == 0:
            g = res.x
            s = g.sum()
            if s > 1e-12:
                return g / s
    except Exception:
        pass
    return None

def smooth_grad(a, b, tau):
    """Compute smoothed subgradient of max(conv(a,a)) at one temperature."""
    n = a.size
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(b.size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= s
    # g[j] = 2 * sum_k w[k] * a[k-j] for valid (k-j) in [0, n-1]
    # = 2 * sum_d a[d] * w[j+d] for d in [0,n-1] where j+d < 2n-1
    g = np.zeros(n)

        lo = 0
        hi = min(n, 2 * n - 1 - d)
        if hi > lo:
            g[lo:hi] += 2.0 * a[d] * w[d:d + hi - lo]
    return g

                print(f"  [LP] n={nv}, init={ii}, score={sc_best:.6f} *** BEST ***")
            else:
                print(f"  [LP] n={nv}, init={ii}, score={sc_best:.6f}")

        for ii in range(n_lp_inits, min(n_lp_inits + n_sg_inits, len(inits))):
            if time.monotonic() > deadline - 30:
                break
            tb = min(time_per_sg, deadline - time.monotonic() - 30)
            if tb < 5.0:
                break
            a_init = inits[ii]
            a_best, sc_best = subgrad_optimize(a_init, tb, rng)
            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = a_best.copy()
                print(f"  [SG] n={nv}, init={ii}, score={sc_best:.6f} *** BEST ***")
            else:
                print(f"  [SG] n={nv}, init={ii}, score={sc_best:.6f}")

        # Polish the best found so far for this n
        if time.monotonic() < deadline - 20 and best_overall is not None:
            pass  # Polish at the end globally

    # === Final polish phase ===
    if best_overall is not None and time.monotonic() < deadline - 10:
        remain = deadline - time.monotonic() - 5.0
        if remain > 10:
            print(f"\n[polish] remaining={remain:.1f}s, n={best_overall.size}, score={best_overall_sc:.6f}")

            # Try LP polish on current best
            lp_time = min(remain * 0.4, 120.0)
            if lp_time > 5:
                a_p, sc_p = lp_optimize(best_overall, lp_time)
                if sc_p < best_overall_sc:
                    best_overall_sc = sc_p
                    best_overall = a_p.copy()
                    print(f"  [polish-LP] improved to {sc_p:.6f}")

            # Try subgradient polish
            if time.monotonic() < deadline - 10:
                sg_time = min(remain * 0.3, 90.0)
                if sg_time > 5:
                    a_p, sc_p = polish_subgrad(best_overall, sg_time, rng)
                    if sc_p < best_overall_sc:
                        best_overall_sc = sc_p
                        best_overall = a_p.copy()
                        print(f"  [polish-SG] improved to {sc_p:.6f}")

            # Random perturbation polish
            if time.monotonic() < deadline - 10:
                rp_time = min(remain * 0.3, 60.0)
                if rp_time > 5:
                    a_p, sc_p = polish(best_overall, rp_time, rng)
                    if sc_p < best_overall_sc:
                        best_overall_sc = sc_p
                        best_overall = a_p.copy()
                        print(f"  [polish-RP] improved to {sc_p:.6f}")

            # One more round of LP if time permits
            if time.monotonic() < deadline - 10:
                remain2 = deadline - time.monotonic() - 5.0
                if remain2 > 10:
                    a_p, sc_p = lp_optimize(best_overall, min(remain2, 60.0))
                    if sc_p < best_overall_sc:
                        best_overall_sc = sc_p
                        best_overall = a_p.copy()
                        print(f"  [polish-LP2] improved to {sc_p:.6f}")

    # Fallback
    if best_overall is None:
        best_overall = init_u_shape(128)
        best_overall_sc = compute_score(best_overall)

    # Final verification
    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"\n[final] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")

    return result
