    cs -= cs.max()
    p = np.exp(cs)
    s = p.sum()

    span = grad.max() - grad.min()

    log_f -= log_f.max()

    if s < 1e-300:

    return f_new / s

"""
Step function optimizer minimizing 2*n*max(f*f)/(sum f)^2 for non-negative f.
Uses LP-guided mirror descent with adaptive annealing, line search,
diverse starting points, and coordinate-descent polish.
"""

import numpy as np
from scipy.signal import correlate
from scipy.optimize import linprog
import time


def _fft_conv(a, b):
    """Full linear convolution via FFT."""
    n, m = len(a), len(b)
    size = 1 << (n + m - 1).bit_length()
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b, size), size)[:n + m - 1]


def _autocorr(f):
    """Full autocorrelation (convolution with itself) via FFT."""
    return _fft_conv(f, f)


def _fast_eval(f):
    """Fast evaluation: 2*n*max(conv)/(sum f)^2. Assumes f >= 0, sum f > 0."""
    n = len(f)
    conv = _autocorr(f)
    peak = conv.max()
    s = f.sum()
    if s < 1e-10:
        return np.inf
    return float(2.0 * n * peak / (s * s))


def _softmax_weights(c, T):
    """Compute softmax weights over convolution values."""
    cs = c / T
    cs -= cs.max()
    p = np.exp(cs)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros_like(c)
        p[len(c) // 2] = 1.0
    else:
        p /= s
    return p


def _grad_from_weights(p, f):
    """Compute gradient: 2 * correlate(p, f, 'valid')."""
    n = len(f)
    g = 2.0 * correlate(p, f, mode='valid')
    return g[:n]


def _mirror_step(f, grad, eta):
    """Exponentiated gradient (mirror) descent step on non-negative simplex."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    return f_new / s


def _solve_lp_dir(f, top_k=80, deadline=None):
    """
    Solve reduced LP: minimize t s.t. (f*g)_k <= t for top_k peak lags,
    sum g = 1, g >= 0. Returns (g_star, t_star) or (None, None).
    """
    n = len(f)
    conv = _autocorr(f)
    top_idx = np.argsort(conv)[::-1][:top_k]

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_ub = np.zeros((len(top_idx), n + 1))
    for i, k in enumerate(top_idx):
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        A_ub[i, lo:hi] = f[lo:hi]
        A_ub[i, n] = -1.0

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * (n + 1)

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=np.zeros(len(top_idx)),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs',
                      options={'time_limit': 3.0})
        if res.success:
            return res.x[:n].copy(), res.x[n]
    except Exception:
        pass
    return None, None

    print("\n=== Phase 1: Multi-start optimization ===")
    top_starts = start_results[:4]
    time_per_start = (phase1_deadline - time.monotonic()) / len(top_starts)

    for idx, (sc, name, f0) in enumerate(top_starts):
        if time.monotonic() > phase1_deadline:
            break
        t_left = phase1_deadline - time.monotonic()
        alloc = min(time_per_start, t_left)
        sub_deadline = time.monotonic() + alloc
        print(f"\nOptimizing '{name}' (score={sc:.6f}, time={alloc:.0f}s)")

        f_opt, sc_opt = _optimize_one(f0, n, sub_deadline, rng, label=name)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  New best: {best_score:.6f}")

    print(f"\nAfter Phase 1: best = {best_score:.6f}")

    # Phase 2: LP-guided refinement on the best
    print("\n=== Phase 2: LP-guided refinement ===")
    phase2_deadline = min(time.monotonic() + phase2_budget, deadline)
    f_r, sc_r = _optimize_one(best_f, n, phase2_deadline, rng, label="refine")
    if sc_r < best_score:
        best_score = sc_r
        best_f = f_r.copy()
        print(f"  Refined: {best_score:.6f}")

    # Phase 3: Coordinate descent polish
    print(f"\n=== Phase 3: Polish (score={best_score:.6f}) ===")
    phase3_deadline = min(time.monotonic() + phase3_budget, deadline)
    f_p, sc_p = _polish(best_f, n, phase3_deadline, rng)
    if sc_p < best_score:
        best_score = sc_p
        best_f = f_p.copy()
        print(f"  Polished: {best_score:.6f}")

    # Phase 4: Final short optimization
    if time.monotonic() < deadline - 10:
        print(f"\n=== Phase 4: Final optimization ===")
        phase4_deadline = min(time.monotonic() + phase4_budget, deadline)
        f_f, sc_f = _optimize_one(best_f, n, phase4_deadline, rng, label="final")
        if sc_f < best_score:
            best_score = sc_f
            best_f = f_f.copy()
            print(f"  Final: {best_score:.6f}")

    # Ensure valid output
    best_f = np.clip(best_f, 0.0, 1000.0)
    s = best_f.sum()
    if s < 0.01:
        best_f = np.ones(n) / n
    else:
        best_f = best_f / s * s  # Keep as is (already valid)

    # Final validation
    final_score = evaluate_sequence([float(x) for x in best_f])
    print(f"\n=== FINAL SCORE: {final_score:.6f} ===")

    return [float(x) for x in best_f]
