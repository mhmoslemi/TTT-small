Let me also think about the simplex projection. Given a point `x` in R^n, the projection onto the simplex `{a: sum(a) = 1, a >= 0}` is:
1. Sort `x` in decreasing order: `x_(1) >= x_(2) >= ... >= x_(n)`
2. Find `k* = max{k: x_(k) - (sum_{j<=k} x_(j))/k > 0}`
3. `theta = (sum_{j<=k*} x_(j) - 1) / k*`
4. `a_i = max(x_i - theta, 0)`

This is O(n log n) per projection.

**Full algorithm:**

```
1. Initialize a (from previous best or random)
2. Normalize to simplex
3. For iteration in range(max_iters):

   b. Compute w = softmax(c / tau)
   c. Compute g = grad via FFT
   d. FISTA update:
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences a.
Uses FISTA (accelerated projected gradient descent) on a log-sum-exp
smoothed objective with FFT-based gradient computation.

import numpy as np
import time
import math

def _normalize(a):
    """Normalize to the probability simplex."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        a.fill(1.0)
        s = a.sum()
    a /= s
    return a

def _project_simplex(v):
    """Project vector v onto the probability simplex {x: sum(x)=1, x>=0}."""
    n = len(v)
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(u * np.arange(1, n+1) - (cssv - 1) > 0)
    theta = (cssv[rho] - 1) / (rho + 1)
    return np.maximum(v - theta, 0)

def _fft_conv_self(a, L, m):
    """Compute self-convolution via FFT."""
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c

def _fft_grad(w, a, L, m, n):
    """Compute gradient of log-sum-exp objective.
    g[i] = sum_k w[k] * (a[k-i] + a[i-k])
    = (a * w)[i] + (a * w_flipped)[2n-2-i]
    """
    w_flipped = w[::-1].copy()
    Fa = np.fft.rfft(a, L)
    # (a * w)[i] for i = 0,...,n-1
    Aw = np.fft.irfft(Fa * np.fft.rfft(w, L), L)[:n]
    # (a * w_flipped)[j] for j = n-1,...,2n-2
    Awf = np.fft.irfft(Fa * np.fft.rfft(w_flipped, L), L)
    g1 = Awf[n-1:2*n-1][::-1]
    # Clean up small negatives from FFT
    np.maximum(Aw, 0.0, out=Aw)
    np.maximum(g1, 0.0, out=g1)
    return Aw + g1

def _fista_search(a0, L, m, n, rng, time_limit, tau0, tau_min):
    """Run FISTA on the log-sum-exp smoothed objective."""
    deadline = time.monotonic() + time_limit
    a = _normalize(a0)
    y = a.copy()
    c_prev = a.copy()
    t_k = 1.0
    
    # Initial gradient
    c = _fft_conv_self(y, L, m)
    vmax = c.max()
    tau = max(tau0, 1e-10)
    w = np.exp((c - vmax) / tau)
    w /= w.sum()
    g = _fft_grad(w, y, L, m, n)
    
    # Lipschitz estimate
    L_cfg = max(np.linalg.norm(g, ord=np.inf) * 2.0, 1.0)
    lr = 1.0 / L_cfg
    
    best_a = a.copy()
    best_score = np.inf
    best_max_c = c.max()
    
    it = 0
    while time.monotonic() < deadline:
        # FISTA extrapolation
        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a + (t_k - 1.0) / t_new * (a - c_prev)
        y = np.maximum(y, 0.0)
        y_sum = y.sum()
        if y_sum > 0 and np.isfinite(y_sum):
            y /= y_sum
        else:
            y = a.copy()
        
        # Compute gradient at y
        c_y = _fft_conv_self(y, L, m)
        vmax = c_y.max()
        if vmax > 0:
            w = np.exp((c_y - vmax) / tau)
            w /= w.sum()
        else:
            w = np.ones(m) / m
        g = _fft_grad(w, y, L, m, n)
        
        # Gradient step + projection
        a_new = _project_simplex(y - lr * g)
        
        # Check objective (using exact max for tracking)
        c_new = _fft_conv_self(a_new, L, m)
        max_new = c_new.max()
        if max_new < best_max_c:
            best_max_c = max_new
            best_a = a_new.copy()
        
        c_prev = a
        a = a_new
        t_k = t_new
        
        # Adaptive step size (backtracking)
        if it % 50 == 0:
            g_norm = np.linalg.norm(g, ord=np.inf)
            if g_norm > 0:
                L_cfg = max(g_norm * 1.5, 0.1)
                lr = 1.0 / L_cfg
        
        # Decrease temperature
        if it % 100 == 0:
            tau = max(tau_min, tau * 0.95)
        
        it += 1
    
    # Final exact check
    c_final = _fft_conv_self(best_a, L, m)
    final_max = c_final.max()
    return best_a, final_max
```

```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses FISTA (accelerated projected gradient descent) on a log-sum-exp
smoothed objective with FFT-based gradient computation, geometric cooling,
and multiple sequence length exploration.
"""

import numpy as np
import time
import math


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a.fill(1.0)
    else:
        a /= s
    return a


def _project_simplex(v):
    n = len(v)
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not cond.any():
        theta = cssv[n - 1] / n
    else:
        rho = int(np.max(np.where(cond)[0]))
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _conv_self(a, L, m):
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _grad_logsumexp(w, a, L, m, n):
    w_rev = w[::-1].copy()
    Fa = np.fft.rfft(a, L)
    Fw = np.fft.rfft(w, L)
    Fw_rev = np.fft.rfft(w_rev, L)
    term2 = np.fft.irfft(Fa * Fw, L)[:n]
    full = np.fft.irfft(Fa * Fw_rev, L)
    term1 = full[n - 1:2 * n - 1][::-1].copy()
    np.maximum(term1, 0.0, out=term1)
    np.maximum(term2, 0.0, out=term2)
    return term1 + term2


def _fista_search(a0, L, m, n, rng, time_limit, tau0_factor=0.1, tau_min_factor=1e-5):
    deadline = time.monotonic() + time_limit
    a = _normalize(a0)
    y = a.copy()
    a_prev = a.copy()
    t_k = 1.0

    c = _conv_self(a, L, m)
    max_c = float(c.max())
    if max_c <= 0:
        max_c = 1.0 / n

    best_a = a.copy()
    best_max_c = max_c

    tau = max_c * tau0_factor
    tau_min = max_c * tau_min_factor

    L_cfg = 4.0 / max_c
    lr = 1.0 / L_cfg

    it = 0
    stale = 0
    last_improve = max_c

    while time.monotonic() < deadline:
        t_new = (1.0 + math.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = a + (t_k - 1.0) / t_new * (a - a_prev)
        y = np.maximum(y, 0.0)
        ys = float(y.sum())
        if ys > 1e-300 and np.isfinite(ys):
            y /= ys
        else:
            y = a.copy()

        c_y = _conv_self(y, L, m)
        vmax = float(c_y.max())
        if vmax > 1e-300:
            z = (c_y - vmax) / tau
            z = np.clip(z, -500.0, 0.0)
            e = np.exp(z)
            es = float(e.sum())
            if es > 0 and np.isfinite(es):
                w = e / es
            else:
                w = np.ones(m, dtype=np.float64) / m
        else:
            w = np.ones(m, dtype=np.float64) / m

        g = _grad_logsumexp(w, y, L, m, n)
        gnorm = float(np.linalg.norm(g, ord=np.inf))

        a_new = _project_simplex(y - lr * g)

        # Check improvement
        if it % 10 == 0:
            c_new = _conv_self(a_new, L, m)
            max_new = float(c_new.max())
            if max_new < best_max_c - 1e-15:
                best_max_c = max_new
                best_a = a_new.copy()
                stale = 0
            else:
                stale += 1
        else:
            c_new = c  # skip check
            max_new = last_improve

        a_prev = a
        a = a_new
        t_k = t_new

        # Adaptive step size
        if it % 20 == 0 and gnorm > 0:
            L_cfg = max(2.0 * gnorm / max_c, 1.0 / (max_c * n))
            lr = 1.0 / L_cfg

        # Cool tau
        if it % 50 == 0:
            tau = max(tau_min, tau * 0.92)

        # Restart if stuck
        if stale > 20:
            a = best_a.copy()
            y = a.copy()
            a_prev = a.copy()
            t_k = 1.0
            tau = max(tau_min, tau * 2.0)
            lr = 1.0 / L_cfg
            # Small perturbation
            noise = rng.normal(0, 0.01, n)
            a = _project_simplex(a + noise)
            y = a.copy()
            stale = 0
            last_improve = best_max_c

        it += 1

    return best_a, best_max_c


def _direct_subgrad_polish(a0, L, m, n, rng, time_limit):
    """Polish using exact subgradient of max."""
    deadline = time.monotonic() + time_limit
    a = _normalize(a0)
    best_a = a.copy()
    best_max_c = float(_conv_self(a, L, m).max())

    lr = 1.0 / (4.0 * best_max_c / n * n)
    lr = min(lr, n / (4.0 * best_max_c))

    it = 0
    while time.monotonic() < deadline:
        c = _conv_self(a, L, m)
        max_c = float(c.max())
        if max_c < best_max_c - 1e-15:
            best_max_c = max_c
            best_a = a.copy()

        # Find active set (top few positions)
        threshold = max_c - 1e-10 * max_c
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Uniform subgradient over active set
        g = np.zeros(n, dtype=np.float64)
        for k in active:
            for i in range(n):
                ki = k - i
                if 0 <= ki < n:
                    g[i] += a[ki]
                ik = i - k
                if 0 <= ik < n:
                    g[i] += a[ik]
        g /= len(active)

        a = _project_simplex(a - lr * g)
        it += 1

    return best_a, best_max_c


def _structured_init(n, rng):
    t = np.linspace(0, 1, n)
    x = 2.0 * t - 1.0
    candidates = []

    # U-shaped
    candidates.append(_normalize(np.exp(3.0 * x * x)))
    # V-shaped
    candidates.append(_normalize(np.exp(-3.0 * x * x)))
    # Two peaks
    candidates.append(_normalize(np.exp(-20.0 * (x - 0.5) ** 2) + np.exp(-20.0 * (x + 0.5) ** 2)))
    candidates.append(_normalize(np.exp(-10.0 * (x - 0.3) ** 2) + np.exp(-10.0 * (x + 0.3) ** 2)))
    # Cosine
    candidates.append(_normalize(1.0 + 0.8 * np.cos(np.pi * x)))
    candidates.append(_normalize(1.0 + 0.5 * np.cos(2.0 * np.pi * x)))
    # Power law
    candidates.append(_normalize(np.power(t, 2.0) + np.power(1.0 - t, 2.0) + 0.1))
    candidates.append(_normalize(np.power(t, 0.5) + np.power(1.0 - t, 0.5)))
    # Random
    candidates.append(_normalize(rng.dirichlet(np.ones(n))))
    candidates.append(_normalize(_normalize(np.exp(rng.normal(0, 0.5, n)))))
    # Smoothed random
    r = np.exp(rng.normal(0, 0.7, n))
    if n > 10:
        kernel = np.ones(10) / 10.0
        r = np.convolve(r, kernel, mode='same')
    candidates.append(_normalize(r))

    return candidates


def _internal_score(seq):
    a = np.asarray(seq, dtype=np.float64)
    a = np.maximum(a, 0.0)
    n = len(a)
    s = float(a.sum())
    if s < 0.01:
        return float('inf')
    b = np.convolve(a, a)
    mb = float(b.max())
    return float(2.0 * n * mb / (s * s))
```
