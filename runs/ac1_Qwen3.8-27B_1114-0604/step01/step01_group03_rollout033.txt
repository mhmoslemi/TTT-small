"""
Search for non-negative step heights minimizing

    2 * n * max_k conv(a, a)_k / (sum(a)^2)

The implementation combines:
  - an analytic one-sided power candidate (truncated (1-z)^(-1/2) style),
  - full-vector multiplicative natural-gradient local search using soft max
    weights over the active convolution diagonals,
  - a block-constant coarse optimizer,
  - a multiplicative-weights direction search,
  - candidate mixing/reheating,
  - final exact validation and a final ordering/projection pass.
"""

import math
import time
import numpy as np


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return np.array([1.0], dtype=np.float64)
    if not np.all(np.isfinite(a)):
        a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a.fill(1.0)
        s = float(a.size)
    a /= s
    return a


def _fft_length(n):
    return 1 << max(1, (2 * n - 2).bit_length())


def _conv_self(a, L, m):
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    c = np.ascontiguousarray(c, dtype=np.float64)
    if c.size:
        np.maximum(c, 0.0, out=c)
    return c


def _score_c(c, n):
    if c.size == 0:
        return float("inf")
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return float("inf")
    return float(2.0 * n * mx)


def _score_a(a, L, m, n):
    c = _conv_self(a, L, m)
    return _score_c(c, n)


def _softmax_weights(q, beta):
    q = np.asarray(q, dtype=np.float64)
    if q.size == 0:
        return np.array([], dtype=np.float64)
    beta = float(beta)
    if not np.isfinite(beta) or beta <= 0.0:
        beta = 1.0
    vmax = float(np.max(q))
    z = beta * (q - vmax)
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        e.fill(1.0)
        s = float(e.size)
    return e / s


def _anti_conv_w_a(w, a, L, m, n):
    W = np.fft.rfft(w, L)
    Frev = np.fft.rfft(a[::-1], L)
    g = np.fft.irfft(W * Frev, L)[(n - 1):m]
    g = np.ascontiguousarray(g, dtype=np.float64)
    if g.size:
        np.maximum(g, 0.0, out=g)
    return g


def _direction_valid(d):
    if d is None:
        return False
    d = np.asarray(d, dtype=np.float64)
    if d.size == 0:
        return False
    if not np.all(np.isfinite(d)):
        return False
    mx = float(np.max(d))
    return np.isfinite(mx) and mx > 1e-300


def _smooth_idx(d, w):
    d = np.asarray(d, dtype=np.float64)
    if d.size == 0:
        return d
    w = int(w)
    if w <= 1:
        y = d.copy()
    elif w >= d.size:
        y = d.copy()
    else:
        k = np.ones(w, dtype=np.float64) / float(w)
        y = np.convolve(d, k, mode="same")
    y = np.maximum(y, 0.0)
    mx = float(np.max(y)) if y.size else 0.0
    if np.isfinite(mx) and mx > 0.0:
        y /= mx
    return y


def _smooth_vector(v, w=7):
    v = np.asarray(v, dtype=np.float64)
    if v.size <= int(w):
        return _normalize(v)
    k = np.ones(int(w), dtype=np.float64) / float(int(w))
    y = np.convolve(v, k, mode="same")
    return _normalize(y)


def _random_decreasing(n, rng):
    t = np.linspace(0.0, 1.0, n)
    alpha = float(rng.uniform(0.35, 0.85))
    z = -alpha * np.log(t + 0.05)
    k = int(rng.integers(1, 4))
    ph = float(rng.uniform(0.0, 2.0 * np.pi))
    z += float(rng.uniform(0.05, 0.25)) * np.cos(2.0 * np.pi * k * t + ph)
    noise = rng.normal(0.0, 0.15, n)
    if n > 10:
        noise = np.convolve(noise, np.ones(5, dtype=np.float64) / 5.0, mode="same")
    z += noise
    p = np.exp(z - float(np.max(z)))
    p = np.maximum(p, 1e-300)
    return _normalize(p)


def _random_wave(n, rng):
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0
    z = rng.normal(0.0, 0.45)
    z += rng.normal(0.0, 0.55) * x
    z += rng.normal(0.0, 0.45) * (x * x - 1.0 / 3.0)
    kmax = int(rng.integers(2, 7))
    for k in range(1, kmax + 1):
        amp = float(rng.uniform(0.05, 0.45)) / max(1.0, math.sqrt(float(k)))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        z += amp * np.cos(2.0 * np.pi * k * t + ph)
    noise = rng.normal(0.0, 0.18, n)
    if n > 10:
        noise = np.convolve(noise, np.ones(5, dtype=np.float64) / 5.0, mode="same")
    z += noise
    p = np.exp(z - float(np.max(z)))
    p = np.maximum(p, 1e-300)
    return _normalize(p)


def _sqrt_coeffs(n):
    a = np.empty(n, dtype=np.float64)
    if n <= 0:
        return a
    a[0] = 1.0
    for k in range(1, n):
        a[k] = a[k - 1] * ((2 * k - 1) / (2.0 * k))
    return _normalize(a)


def _add_candidate(arrays, scores, a, sc, max_len=64):
    a = np.asarray(a, dtype=np.float64)
    try:
        sc = float(sc)
    except Exception:
        return
    if not a.size:
        return
    if not np.isfinite(sc):
        return
    if not np.all(np.isfinite(a)):
        return
    if len(scores) < max_len:
        arrays.append(a.copy())
        scores.append(sc)
    else:
        mn = min(scores)
        if sc < mn:
            idx = scores.index(mn)
            arrays[idx] = a.copy()
            scores[idx] = sc


def _register_candidate(arrays, scores, a, sc, max_len, best_a, best_score):
    if a is None:
        return best_a, best_score
    a = np.asarray(a, dtype=np.float64)
    try:
        sc = float(sc)
    except Exception:
        return best_a, best_score
    if not a.size or not np.isfinite(sc):
        return best_a, best_score
    a = _normalize(a)
    _add_candidate(arrays, scores, a, sc, max_len)
    if sc < best_score:
        return a, sc
    return best_a, best_score


def _sorted_desc(a):

def _initial_candidates(n, rng):
    out = []

        if base.size == n and np.all(np.isfinite(base)):

def _fast_soft_search(a, L, m, n, rng, budget, beta_min, beta_max):
    if budget is None or float(budget) <= 0.0:
        a = _normalize(a)
        return a, _score_a(a, L, m, n)

    deadline = time.monotonic() + float(budget)
    a = _normalize(a)
    c = _conv_self(a, L, m)
    score = _score_c(c, n)
    best_a = a.copy()
    best_score = score

    min_beta = float(beta_min) * float(n)
    max_beta = float(beta_max) * float(n)
    if not np.isfinite(min_beta) or min_beta <= 0.0:
        min_beta = float(n)
    if not np.isfinite(max_beta) or max_beta <= min_beta:
        max_beta = max(min_beta * 8.0, float(n))

    beta = min_beta
    eta = 0.15
    no_accept = 0
    it = 0

    while time.monotonic() < deadline:
        c = _conv_self(a, L, m)
        score = _score_c(c, n)
        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_a = a.copy()
            no_accept = 0

        w = _softmax_weights(c, min(max_beta, max(min_beta, beta)))
        g = _anti_conv_w_a(w, a, L, m, n)
        g *= 2.0
        gmax = float(np.max(g)) if g.size else 0.0
        if not np.isfinite(gmax) or gmax <= 0.0:
            break

        d = g / gmax
        dirs = [d]
        if n > 30:
            d3 = _smooth_idx(d, 3)
            if _direction_valid(d3):
                dirs.append(d3)
            d9 = _smooth_idx(d, 9)
            if _direction_valid(d9):
                dirs.append(d9)

        best_step_score = score
        best_step_p = a
        best_lam = 0.0

        for dir_ in dirs:
            if not _direction_valid(dir_):
                continue
            lams = (eta, 0.5 * eta, min(3.0, 2.0 * eta))
            for lam in lams:
                if lam <= 1e-8:
                    continue
                p = a * np.exp(-lam * dir_)
                p = _normalize(p)
                sc = _score_a(p, L, m, n)
                if np.isfinite(sc) and sc < best_step_score - 1e-14:
                    best_step_score = sc
                    best_step_p = p
                    best_lam = lam

        if best_lam > 0.0:
            a = best_step_p
            score = best_step_score
            no_accept = 0
            if best_lam >= 0.9 * eta:
                eta = min(3.0, eta * 1.25)
            else:
                eta = max(0.02, eta * 0.85)
        else:
            no_accept += 1
            eta = max(0.02, eta * 0.75)
            if no_accept > 70:
                prior = _random_decreasing(n, rng)
                mix = 0.10 + 0.40 * float(rng.random())
                a = _normalize((1.0 - mix) * best_a + mix * prior)
                score = _score_a(a, L, m, n)
                if np.isfinite(score) and score < best_score:
                    best_score = score
                    best_a = a.copy()
                no_accept = 0
                eta = 0.15
                beta = max(min_beta, beta * 0.4)

        it += 1
        if it % 50 == 0:
            beta = min(max_beta, beta * 1.35)
        if it % 250 == 0 and no_accept > 30:
            beta = max(min_beta, min(max_beta, beta * 0.5))

    return best_a, best_score


def _mws_direction(f, L, m, n, prior, inner):
    f = _normalize(f)
    prior = _normalize(prior)
    inner = max(1, int(inner))

    F = np.fft.rfft(f, L)
    Frev = np.fft.rfft(f[::-1], L)

    x = prior.copy()
    y = np.full(m, 1.0 / float(m), dtype=np.float64)
    avg = np.zeros_like(x, dtype=np.float64)
    count = 0
    logprior = np.log(np.maximum(prior, 1e-300))

    for it in range(inner):
        frac = 0.0 if inner <= 1 else float(it) / float(inner - 1)
        eta = 2.0 + (40.0 - 2.0) * frac

        X = np.fft.rfft(x, L)
        u = np.fft.irfft(F * X, L)[:m]
        u = np.ascontiguousarray(u, dtype=np.float64)
        if u.size:
            np.maximum(u, 0.0, out=u)

        vu = float(n) * u
        vma = float(np.max(vu)) if vu.size else 0.0
        z = eta * (vu - vma)
        z = np.clip(z, -50.0, 0.0)
        ye = np.exp(z)
        s = float(ye.sum())
        if np.isfinite(s) and s > 0.0:
            y = ye / s
        else:
            y.fill(1.0 / float(m))

        avg += x
        count += 1

        Y = np.fft.rfft(y, L)
        svec = np.fft.irfft(Y * Frev, L)[(n - 1):m]
        svec = np.ascontiguousarray(svec, dtype=np.float64)
        if svec.size:
            np.maximum(svec, 0.0, out=svec)

        ps = float(n) * svec
        sma = float(np.max(ps)) if ps.size else 0.0
        zx = -eta * (ps - sma) + 0.8 * logprior
        zx = zx - float(np.max(zx))
        zx = np.clip(zx, -50.0, 0.0)
        xe = np.exp(zx)
        xs = float(xe.sum())
        if np.isfinite(xs) and xs > 0.0:
            x = xe / xs
        else:
            x = prior.copy()

    if count == 0:
        return x
    return _normalize(0.5 * (avg / float(count)) + 0.5 * x)


def _mws_search(a, L, m, n, rng, budget):
    if budget is None or float(budget) <= 0.0:
        a = _normalize(a)
        return a, _score_a(a, L, m, n)

    deadline = time.monotonic() + float(budget)
    best_a = _normalize(a)
    best_score = _score_a(best_a, L, m, n)

    inner = 16 if float(budget) < 180.0 else 22
    lams = (0.02, 0.05, 0.10, 0.18, 0.30, 0.45, 0.65, 0.85, 1.0)
    it = 0

    while time.monotonic() < deadline:
        prior = _random_decreasing(n, rng)
        x = _mws_direction(best_a, L, m, n, prior, inner)
        dirs = [x, _smooth_vector(x, 7)]
        for d in dirs:
            if not _direction_valid(d):
                continue
            d = _normalize(d)
            for lam in lams:
                h = _normalize((1.0 - lam) * best_a + lam * d)
                sc = _score_a(h, L, m, n)
                if np.isfinite(sc) and sc < best_score - 1e-14:
                    best_score = sc
                    best_a = h.copy()

        it += 1
        if it % 4 == 0 and time.monotonic() > deadline - 2.0:
            break

    return best_a, best_score


def _block_search(a_init, L, m, n, budget, B=150):
    if budget is None or float(budget) <= 0.0:
        a = _normalize(a_init)
        return a, _score_a(a, L, m, n)

    deadline = time.monotonic() + float(budget)
    a_init = _normalize(a_init)
    B = int(max(20, min(int(B), n)))
    if B >= n:
        cuts = np.arange(n + 1, dtype=np.int64)
    else:
        cuts = np.linspace(0, n, B + 1).astype(np.int64)
        for i in range(1, len(cuts)):
            if cuts[i] <= cuts[i - 1]:
                cuts[i] = cuts[i - 1] + 1
        cuts = np.unique(cuts)
        if cuts[0] != 0:
            cuts = np.concatenate((np.array([0], dtype=np.int64), cuts))
        if cuts[-1] != n:
            cuts = np.concatenate((cuts, np.array([n], dtype=np.int64)))
    B = int(len(cuts) - 1)
    if B < 2:
        return a_init, _score_a(a_init, L, m, n)

    sizes_i = np.diff(cuts).astype(np.int64)
    sizes_f = sizes_i.astype(np.float64)

    m = np.add.reduceat(a_init, cuts[:-1]).astype(np.float64)
    m = np.maximum(m, 1e-300)
    m /= float(m.sum())
    logm = np.log(m)

    best_m = m.copy()

    def _expand(bb):
        bb = np.asarray(bb, dtype=np.float64)
        bb = np.maximum(bb, 0.0)
        ss = float(bb.sum())
        if not np.isfinite(ss) or ss <= 0.0:
            bb = np.ones_like(bb, dtype=np.float64)
            ss = float(bb.size)
        bb /= ss
        return np.repeat(bb / sizes_f, sizes_i)

    best_p = _expand(best_m)
    best_score = _score_a(best_p, L, m, n)

    min_beta = float(n)
    max_beta = 64.0 * float(n)
    beta = min_beta
    eta = 0.20
    no_accept = 0
    it = 0

    while time.monotonic() < deadline:
        m_curr = np.maximum(m, 1e-300)
        m_curr /= float(m_curr.sum())
        m = m_curr
        p = _expand(m)
        c = _conv_self(p, L, m)
        score = _score_c(c, n)
        if np.isfinite(score) and score < best_score - 1e-14:
            best_score = score
            best_m = m.copy()
            best_p = p.copy()
            no_accept = 0

        w = _softmax_weights(c, min(max_beta, max(min_beta, beta)))
        g = _anti_conv_w_a(w, p, L, m, n)
        g *= 2.0

        gsum = np.add.reduceat(g, cuts[:-1]).astype(np.float64)
        total_f = float(gsum.sum())
        grad = gsum - m * total_f

        d = grad - float(np.mean(grad))
        d = np.where(np.isfinite(d), d, 0.0)
        scale = float(np.max(np.abs(d)))
        if not np.isfinite(scale) or scale <= 1e-300:
            no_accept += 1
            eta = max(0.02, eta * 0.75)
            if no_accept > 80:
                alpha = float(rng.uniform(0.35, 0.85))
                ts = np.linspace(0.0, 1.0, B)
                u = -alpha * ts + rng.normal(0.0, 0.25, B)
                up = np.exp(u - float(np.max(u)))
                up = np.maximum(up, 1e-300)
                up /= float(up.sum())
                m = np.maximum(0.7 * m + 0.3 * up, 1e-300)
                m /= float(m.sum())
                logm = np.log(m)
                no_accept = 0
                eta = 0.20
                beta = max(min_beta, beta * 0.5)
            it += 1
            continue

        d /= scale

        best_step_score = score
        best_step_m = m
        best_lam = 0.0

        for lam in (eta, 0.5 * eta, min(3.0, 2.0 * eta)):
            if lam <= 1e-8:
                continue
            tmp = logm - lam * d
            tmp -= float(np.max(tmp))
            mm = np.exp(tmp)
            mm = np.maximum(mm, 1e-300)
            mm /= float(mm.sum())
            pp = _expand(mm)
            sc = _score_a(pp, L, m, n)
            if np.isfinite(sc) and sc < best_step_score - 1e-14:
                best_step_score = sc
                best_step_m = mm
                best_lam = lam

        if best_lam > 0.0:
            m = best_step_m
            logm = np.log(np.maximum(m, 1e-300))
            no_accept = 0
            if best_lam >= 0.9 * eta:
                eta = min(3.0, eta * 1.25)
            else:
                eta = max(0.02, eta * 0.85)
        else:
            no_accept += 1
            eta = max(0.02, eta * 0.75)
            if no_accept > 80:
                alpha = float(rng.uniform(0.35, 0.85))
                ts = np.linspace(0.0, 1.0, B)
                u = -alpha * ts + rng.normal(0.0, 0.25, B)
                up = np.exp(u - float(np.max(u)))
                up = np.maximum(up, 1e-300)
                up /= float(up.sum())
                m = np.maximum(0.7 * m + 0.3 * up, 1e-300)
                m /= float(m.sum())
                logm = np.log(m)
                no_accept = 0
                eta = 0.20
                beta = max(min_beta, beta * 0.5)

        it += 1
        if it % 45 == 0:
            beta = min(max_beta, beta * 1.4)

    return best_p, best_score

    if not isinstance(seq, (list, tuple, np.ndarray)) or len(seq) == 0:
        return float("inf")
    a = np.asarray(seq, dtype=np.float64)
    if not np.all(np.isfinite(a)):
        return float("inf")
    a = np.clip(a, 0.0, 1000.0)
    n = int(a.size)
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    b = np.convolve(a, a)
    mb = float(np.max(b)) if b.size else float("inf")
    if not np.isfinite(mb):
        return float("inf")
    return float(2.0 * n * mb / (s * s))


def _exact_score(seq):
    try:
        val = evaluate_sequence(seq)
    except NameError:
        val = _internal_exact(seq)
    except Exception:
        val = _internal_exact(seq)
    try:
        val = float(val)
    except Exception:
        val = float("inf")
    return val


def _make_exact_list(arrays, scores, nseq=16):
    if not arrays:
        return []
    idx = list(range(len(arrays)))
    idx.sort(key=scores.__getitem__)
    out = []
    for i in idx:
        if len(out) >= nseq:
            break
        arr = _normalize(arrays[i])
        out.append(arr)
        if len(out) >= nseq:
            break
        s = np.sort(arr)[::-1]
        out.append(s)
    return out
