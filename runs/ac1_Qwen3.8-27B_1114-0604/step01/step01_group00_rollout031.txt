"""
Optimizes non-negative sequence minimizing 2*n*max(f*f)/(sum f)^2.
Uses mirror descent with annealing, frequency-domain spectral flattening,
targeted coordinate descent, and multi-start strategy.
"""

import numpy as np
from scipy.fft import rfft, irfft
import time


def _autocorr(f):
    """Linear self-convolution via FFT."""
    n = len(f)
    N = 1 << (2 * n - 1).bit_length()
    F = rfft(f, N)
    return irfft(F * F, N)[:2 * n - 1]


def _soft_grad(f, T):
    """Softmax-weighted gradient of peak autocorrelation. Returns (grad, c)."""
    n = len(f)
    c = _autocorr(f)
    cs = c / T
    cs -= cs.max()
    p = np.exp(cs)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros(2 * n - 1)
        p[int(np.argmax(c))] = 1.0
    else:
        p /= s
    grad = 2.0 * np.correlate(p, f, mode='valid')
    return grad, c


def _mirror_step(f, grad, eta):
    """Exponentiated gradient step on simplex."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy()
    return f_new / s


def _freq_flatten(f, alpha, n_iters, deadline):
    """Spectral flattening in frequency domain with non-negativity projection."""
    n = len(f)
    N = 1 << (2 * n - 1).bit_length()
    f_curr = f.copy()
    for i in range(n_iters):
        if time.monotonic() > deadline:
            break
        F = rfft(f_curr, N)
        mag = np.abs(F)
        phase = np.angle(F)
        target = mag.mean()
        mag_new = mag * (1.0 - alpha) + target * alpha
        F_new = mag_new * np.exp(1j * phase)

        if s < 1e-10:
            break
        f_new /= s
        if np.allclose(f_new, f_curr, atol=1e-12):
            break
        f_curr = f_new
    return f_curr


def _targeted_cd(f, n_trials, rng, deadline):
    """Targeted coordinate descent: move mass to reduce peak lag contribution."""
    n = len(f)
    f = f.copy()
    c = _autocorr(f)
    peak = c.max()
    
    for _ in range(n_trials):
        if time.monotonic() > deadline:
            break
        k_star = int(np.argmax(c))
        
        # Contribution of each position j to c_{k_star}: f[j] * f[k_star - j]
        # (valid when 0 <= k_star - j < n)
        lo = max(0, k_star - n + 1)
        hi = min(n - 1, k_star)
        contrib = np.zeros(n)
        if hi >= lo:
            contrib[lo:hi+1] = f[lo:hi+1] * f[k_star - hi:k_star - lo + 1]
        
        # Find best donor (high contribution) and best receiver (low contribution, far from peak)
        j_don = int(np.argmax(contrib))
        # Receiver: prefer positions with low f and low contribution
        score_recv = f + 0.001 * contrib
        j_recv = int(np.argmin(score_recv))
        
        if j_don == j_recv:
            j_recv = int(rng.integers(n))
            if j_recv == j_don:

        amt = f[j_don] * 0.1
        f_try = f.copy()
        f_try[j_don] -= amt
        f_try[j_recv] += amt
        f_try = np.maximum(f_try, 0.0)

        if c_try.max() < peak - 1e-12:
            f = f_try
            peak = c_try.max()
            c = c_try
