    np.maximum(c, 0.0, out=c)

    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)

        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]

import numpy as np
import time
import math


def _fft_len(n):
    """Smallest power of 2 >= n."""
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()


def _conv_self(a, L, m):
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _score(c, n):
    return float(2.0 * n * np.max(c))


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    v = np.asarray(v, dtype=np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1, dtype=np.float64) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n, dtype=np.float64) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _smoothed_obj_grad(a, L, m, n, T):
    """Compute log-sum-exp smoothed objective and gradient w.r.t. a."""
    c = _conv_self(a, L, m)
    Tc = T * c
    Tc_max = float(np.max(Tc))
    exp_Tc = np.exp(Tc - Tc_max)
    sum_exp = float(np.sum(exp_Tc))
    obj = (Tc_max + math.log(sum_exp)) / T
    w = exp_Tc / sum_exp

    conv_result = np.fft.irfft(W * A_rev, L)

    np.maximum(g, 0.0, out=g)
    return obj, g

    """Nesterov-accelerated projected gradient with adaptive temperature."""
    if time_limit <= 1.0:
        a = _project_simplex(a0)
        c = _conv_self(a, L, m)
        return a, _score(c, n)

    deadline = time.monotonic() + time_limit
    a = _project_simplex(a0)
    y = a.copy()
    t_k = 1.0
    best_a = a.copy()
    best_score = _score(_conv_self(a, L, m), n)
    lr = 0.5
    it = 0

    while time.monotonic() < deadline:
        frac = min(1.0, float(it) / max(1, int(time_limit * 50)))
        T = T_start + (T_end - T_start) * frac
        if T < 1.0:
            T = 1.0

        f_y, g_y = _smoothed_obj_grad(y, L, m, n, T)
        step = lr / (t_k * t_k)
        if step > 0.5:
            step = 0.5
        elif step < 1e-10:
            step = 1e-10

        z = _project_simplex(y - step * g_y)

        y = z + ((t_k - 1.0) / t_next) * (z - a)
        a = z
        t_k = t_next

            if best_cand != old_val:
                a[j] = best_cand
                a = a / a.sum()
                improved = True
                c_new = _conv_self(a, L, m)
                sc_new = _score(c_new, n)
                if sc_new < best_score:
                    best_score = sc_new
                    best_a = a.copy()

        pass_count += 1
        if not improved and pass_count > 5:
            break

    return best_a, best_score


def _gen_chirp_init(n, alpha, beta, phase, rng):
    """Generate a chirp-based initialization."""
    t = np.arange(n, dtype=np.float64)
    freq = rng.uniform(0.1, 3.0)
    z = alpha * np.cos(np.pi * t * t * freq / n + phase) + beta * np.cos(2.0 * np.pi * t / n)
    z += rng.normal(0, 0.2, n)
    p = np.exp(z - np.max(z))
    p = np.maximum(p, 1e-15)
    return p / p.sum()


def _gen_mseq_init(n, rng):
    """Generate m-sequence-like initialization (smoothed binary sequence)."""
    # Generate a pseudo-random binary sequence using LFSR-like approach
    m = 1 << int(math.log2(n)) 
    if m < 4:
        m = 4
    # Simple xorshift-based pattern
    state = rng.integers(1, 1 << 30)
    bits = np.zeros(m, dtype=np.float64)
    for i in range(m):
        state ^= (state << 13) & 0xFFFFFFFF
        state ^= (state >> 7)
        state ^= (state << 17) & 0xFFFFFFFF
        bits[i] = 1.0 if (state & 1) else 0.0
    # Resample to length n
    idx = (np.arange(n) * m / n).astype(int) % m
    a = bits[idx]
    # Smooth
    k = max(3, n // 100)
    if k % 2 == 0:
        k += 1
    kernel = np.ones(k) / k
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.01)
    return a / a.sum()


def _gen_smooth_random(n, rng, n_freq=5):
    """Generate smooth random initialization."""
    t = np.linspace(0, 2 * np.pi, n)
    z = np.zeros(n)
    for k in range(1, n_freq + 1):
        amp = rng.uniform(0.3, 1.0) / math.sqrt(k)
        ph = rng.uniform(0, 2 * np.pi)
        z += amp * np.cos(k * t + ph)
    z += rng.normal(0, 0.1, n)
    p = np.exp(z - np.max(z))
    p = np.maximum(p, 1e-15)
    return p / p.sum()


def _gen_bimodal(n, rng):
    """Generate bimodal initialization."""
    c1 = rng.uniform(0.1, 0.4) * n
    c2 = rng.uniform(0.6, 0.9) * n
    sigma1 = rng.uniform(0.05, 0.15) * n
    sigma2 = rng.uniform(0.05, 0.15) * n
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-(t - c1) ** 2 / (2 * sigma1 ** 2)) + np.exp(-(t - c2) ** 2 / (2 * sigma2 ** 2))
    a += 0.01
    return a / a.sum()


def _gen_parent_scaled(n, rng):
    """Try to use parent sequence if available, resampled to length n."""
    try:
        parent = height_sequence_1
        parr = np.asarray(parent, dtype=np.float64).ravel()
        if parr.size == n:
            parr = np.maximum(parr, 0)
            s = parr.sum()
            if s > 0.01:
                return parr / s
        # Resample
        old_n = parr.size
        new_idx = (np.arange(n) * old_n / n).astype(int)
        resampled = parr[new_idx]
        resampled = np.maximum(resampled, 0)
        s = resampled.sum()
        if s > 0.01:
            return resampled / s
    except (NameError, Exception):
        pass
    return None


def _exact_eval(seq):
    """Evaluate using evaluate_sequence if available, else internal."""
    try:
        val = evaluate_sequence(seq)
    except NameError:
        # Internal evaluation
        a = np.asarray(seq, dtype=np.float64)
        a = np.maximum(a, 0)
        a = np.minimum(a, 1000.0)
        n = len(a)
        s = a.sum()
        if s < 0.01:
            return float('inf')
        b = np.convolve(a, a)
        return float(2.0 * n * np.max(b) / (s * s))
    except Exception:
        return float('inf')
    if val is None:
        return float('inf')
    return float(val)
