def _normalize(a):
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a)
    return a / s

def _compute_conv(a, L, m):
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c

def _score_from_conv(c, n):
    return 2.0 * n * c.max()

def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    # Sort in decreasing order
    sorted_v, idx = np.sort(v)[::-1], np.argsort(v)[::-1]
    cumsum = np.cumsum(sorted_v)
    # Find rho: largest k such that sorted_v[k-1] > (cumsum[k-1] - 1) / k
    cs = 1.0 / np.arange(1, n + 1)
    cond = sorted_v > (cumsum - 1.0) * cs
    if not cond.any():
        return np.ones(n) / n
    rho = np.max(np.where(cond)[0]) + 1
    theta = (cumsum[rho - 1] - 1.0) / rho
    return np.maximum(v - theta, 0.0)

def _compute_grad_active(a, c, n, K=20, tol=0.05):
    """Compute subgradient of max(c) using top-K active positions."""
    max_c = c.max()
    # Active set: positions where c_k > max_c * (1 - tol)

def _normalize(a):
    """Normalize to probability simplex {a >= 0, sum(a) = 1}."""
    a = np.maximum(a.astype(np.float64), 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a, dtype=np.float64) / len(a)
    return a / s


def _compute_conv(a, L, m):
    """Compute full auto-convolution using FFT. Returns array of length m."""
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _score(c, n):
    """Score = 2n * max(c), assuming sum(a)=1."""
    return 2.0 * n * float(c.max())


def _project_simplex(v):
    """Exact Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 1:
        return np.array([1.0])
    sorted_v = np.sort(v)[::-1]
    cumsum = np.cumsum(sorted_v)
    rho_candidates = 1.0 / np.arange(1, n + 1)
    cond = sorted_v > (cumsum - 1.0) * rho_candidates
    if not cond.any():
        return np.ones(n, dtype=np.float64) / n
    rho = int(np.max(np.where(cond)[0])) + 1
    theta = (cumsum[rho - 1] - 1.0) / rho
    result = np.maximum(v - theta, 0.0)
    s = result.sum()
    if s > 0:
        result /= s
    return result


def _compute_grad_active(a, c, n, top_k=30, tol_frac=0.03):
    """
    Compute subgradient of max(c) w.r.t. a using active set.
    dc_k/da_i = 2*a_{k-i} for valid indices.
    Returns (grad, max_c, n_active).
    """
    max_c = float(c.max())
    threshold = max_c * (1.0 - tol_frac)
    active = np.where(c >= threshold)[0]
    top_indices = np.argsort(c)[::-1][:top_k]
    active = np.unique(np.concatenate([active, top_indices]))
    
    grad = np.zeros(n, dtype=np.float64)
    for k in active:
        i_lo = max(0, k - n + 1)
        i_hi = min(k, n - 1)
        if i_hi < i_lo:
            continue
        j = np.arange(i_lo, i_hi + 1)
        grad[i_lo:i_hi + 1] += 2.0 * a[k - j]
    
    grad /= float(len(active))
    return grad, max_c, len(active)

def _generate_shapes(n, rng):

        candidates.append(_normalize(np.exp(alpha * (t - 1.0))))
    
    # Gaussian (symmetric, for comparison)

        candidates.append(_normalize(np.exp(-0.5 * ((t - 0.5) / sigma) ** 2)))
    
    # Asymmetric Gaussian
    for sigma in [0.15, 0.2, 0.3]:
        for center in [0.3, 0.4, 0.5, 0.6]:

    # Sigmoid-like
    for k in [5, 10, 20]:

    for _ in range(5):

        # Smooth
        k = np.ones(21) / 21
        x = np.convolve(x, k, mode='same')
        candidates.append(_normalize(np.exp(x)))
    
    return candidates

    t3_end = min(deadline - 0.15 * budget, t2_end + 0.3 * budget)
    t3_lim = t3_end - time.monotonic()
    if t3_lim > 20:
        tmp_a, tmp_sc = _coordinate_descent_v2(best_a, L, m, n, rng, t3_lim)
        if tmp_sc < best_score:
            best_score = tmp_sc
            best_a = tmp_a.copy()
        print(f"Phase 3 (coord desc): score={best_score:.6f} at t={time.monotonic()-_global_start:.0f}s")
    
    # Phase 4: Multi-start with gradient refinement
    t4_end = deadline - 5
    t4_lim = t4_end - time.monotonic()
    if t4_lim > 30:
        n_restarts = max(2, int(t4_lim / 120))
        for r in range(n_restarts):
            if time.monotonic() > t4_end - 10:
                break
            mix = 0.1 + 0.2 * rng.random()
            pert = _normalize(rng.random(n) + 0.05)
            start_a = _normalize((1 - mix) * best_a + mix * pert)
            
            sub_lim = min(100.0, (t4_end - time.monotonic()) - 5)
            if sub_lim < 15:
                break
            
            tmp_a, tmp_sc = _gradient_search(start_a, L, m, n, rng, sub_lim, start_eta=0.01)
            if tmp_sc < best_score:
                best_score = tmp_sc
                best_a = tmp_a.copy()
                print(f"  Restart {r}: improved to {best_score:.6f}")
    
    print(f"After all phases: score={best_score:.6f}")
    
    # Phase 5: Final coordinate descent polish
    if time.monotonic() < deadline - 5:
        polish_lim = deadline - time.monotonic() - 5
        if polish_lim > 10:
            tmp_a, tmp_sc = _coordinate_descent_v2(best_a, L, m, n, rng, min(polish_lim, 60))
            if tmp_sc < best_score:
                best_score = tmp_sc
                best_a = tmp_a.copy()
    
    # Final: normalize and verify
    best_a = _normalize(best_a)
    c_final = _compute_conv(best_a, L, m)
    final_internal = _score(c_final, n)
    print(f"Final internal score: {final_internal:.6f}")
    
    seq = [float(x) for x in best_a]
    try:
        exact = evaluate_sequence(seq)
        print(f"Exact evaluate_sequence score: {exact:.6f}")
    except Exception as e:
        print(f"Exact eval failed: {e}")
        exact = final_internal
    
    return seq
