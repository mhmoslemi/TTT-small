    deadline = time.monotonic() + time_limit
    a = _normalize(a.copy())
    b = _conv_self(a, L, m)
    current_score = _score(b, n)
    best_a = a.copy()
    best_score = current_score

    temp = float(temp0)
    eta = 0.5
    it = 0
    stale = 0
    l2_mix = 0.3

    while time.monotonic() < deadline:
        b = _conv_self(a, L, m)
        current_score = _score(b, n)

        if current_score < best_score - 1e-12:
            best_score = current_score
            best_a = a.copy()
            stale = 0
        else:
            stale += 1

        if stale > 60:
            a = best_a.copy()
            a = _sparsify(a, rng, frac=rng.uniform(0.2, 0.5))
            stale = 0
            temp = max(temp0, temp * 0.4)
            eta = 0.5
            continue

        # Compute gradient
        g_sm = _grad_softmax_obj(a, b, temp, L2, n, m)
        if use_l2:
            g_l2 = _grad_l2_obj(a, b, L2, n, m)
            g_sm_max = float(g_sm.max())
            g_l2_max = float(g_l2.max())
            if g_sm_max > 1e-30 and g_l2_max > 1e-30:
                g = l2_mix * (g_l2 / g_l2_max) + (1.0 - l2_mix) * (g_sm / g_sm_max)
            else:
                g = g_sm
                g_sm_max = g_l2_max
        else:
            g = g_sm
            g_sm_max = float(g.max())

        g_max = float(g.max())
        if g_max <= 1e-30 or not np.isfinite(g_max):
            break
        g_norm = g / g_max

        # Multiplicative update with line search
        accepted = False
        for step_mult in (1.0, 0.5, 0.25, 0.1):
            step = eta * step_mult
            if step <= 1e-15:
                continue
            a_new = a * np.exp(-step * g_norm)
            a_new = _normalize(a_new)
            b_new = _conv_self(a_new, L, m)
            sc_new = _score(b_new, n)
            if sc_new < current_score - 1e-14:
                a = a_new
                accepted = True
                break
            # Also check soft objective improvement
            sm_old = _soft_obj_value(b, temp)
            sm_new = _soft_obj_value(b_new, temp)
            if sm_new < sm_old - 1e-10:
                a = a_new
                accepted = True
                break

        if accepted:
            eta = min(3.0, eta * 1.05)
        else:
            eta = max(0.05, eta * 0.6)
            if eta < 0.1:
                # Reheat
                a = best_a.copy()
                a = _normalize(a + rng.normal(0, 0.02, n))
                eta = 0.5
                temp = max(temp0, temp * 0.7)

        it += 1
        if it % 30 == 0:
            temp = min(temp1, temp * 1.08)
            # Gradually shift weight from L2 to soft-max
            l2_mix = max(0.0, l2_mix * 0.95)

    return best_a, best_score


def _scipy_polish(a_init, L, L2, m, n, time_limit, temp=200):
    """Polish using scipy L-BFGS on softmax parameterization."""
    deadline = time.monotonic() + time_limit
    if time.monotonic() > deadline:
        a_init = _normalize(a_init)
        return a_init, _score(_conv_self(a_init, L, m), n)

    a_init = _normalize(a_init.copy())
    z = np.log(a_init + 1e-300)

    def _obj_grad(z_flat):
        a = _softmax_param(z_flat)
        F = np.fft.rfft(a, L)
        b = np.fft.irfft(F * F, L)[:m]
        np.maximum(b, 0.0, out=b)

        # Soft-max objective value
        tv = temp * b
        tv_max = float(tv.max())
        ev = np.exp(tv - tv_max)
        sv = float(ev.sum())
        if sv <= 0.0 or not np.isfinite(sv):
            obj = float(2.0 * n * tv_max / temp)
            return obj, np.zeros_like(z_flat)
        obj = float(2.0 * n * (tv_max + math.log(sv)) / temp)

        # Gradient
        w = ev / sv
        g_a = _grad_cross(w, a, n, m, L2) * 2.0
        np.maximum(g_a, 0.0, out=g_a)
        mean_g = float(np.dot(g_a, a))
        grad_z = a * (g_a - mean_g)
        return obj, grad_z

    try:
        result = _scipy_minimize(
            _obj_grad, z, method='L-BFGS-B',
            jac=True,
            options={'maxiter': 300, 'maxfun': 500,
                     'gtol': 1e-8, 'ftol': 1e-12, 'iprint': 0}
        )
        z_opt = result.x
        a_opt = _softmax_param(z_opt)
    except Exception:
        a_opt = a_init

    if time.monotonic() > deadline:
        return a_init, _score(_conv_self(a_init, L, m), n)

    b_opt = _conv_self(a_opt, L, m)
    sc = _score(b_opt, n)
    return a_opt, sc


def _internal_exact(seq):
    if not isinstance(seq, list) or not seq:
        return float('inf')
    for x in seq:
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return float('inf')
        if np.isnan(x) or np.isinf(x):
            return float('inf')
    a = [0.0 if float(x) < 0.0 else min(1000.0, float(x)) for x in seq]
    nv = len(a)
    s = sum(a)
    if s < 0.01:
        return float('inf')
    b = np.convolve(a, a)
    mb = max(b)
    return float(2.0 * nv * mb / (s * s))


def _exact_score(seq):
    try:
        val = evaluate_sequence(seq)
    except NameError:
        val = _internal_exact(seq)
    except Exception:
        val = float('inf')
    if val is None:
        return float('inf')
    return float(val)
