"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses multiplicative mirror descent with adaptive softmax smoothing,
structured initializations (geometric, smooth random, two-block),
coordinate shooting local search, and multi-length exploration.
Key improvements: adaptive step sizes, coordinate-wise exact optimization,
and time-adaptive allocation across lengths and initializations.
"""

import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_fft(a, b):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 3000000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout]


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = p.sum()
    if s <= 0 or not np.isfinite(s):
        return None
    return p / s


def _score(p):
    p = _sanitize(p)
    if p is None:
        return float('inf')
    n = p.size
    c = _conv_fft(p, p)
    if c.size == 0:
        return float('inf')
    return float(2.0 * n * np.max(c))


def _softmax_weights(d, beta):
    d = np.asarray(d, dtype=float)
    n = d.size
    if n == 0:
        return np.array([], dtype=float)
    d = np.where(np.isfinite(d), d, 0.0)
    if beta < 1e-6:
        return np.full(n, 1.0/n)
    shift = d.max()
    e = np.exp(beta * (d - shift))
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full(n, 1.0/n)
    return e / s


def _gradient_from_weights(w, p):
    """Compute gradient of sum_j w[j]*conv(p,p)[j] w.r.t. p.
    grad[i] = 2 * sum_j w[j] * p[j-i] (where 0<=j-i<n)
    This is 2 * valid cross-correlation of w and p.
    """
    n = p.size
    nw = w.size
    # We need g[i] = 2 * sum_{j: 0<=j-i<nw'} w[j] * p[j-i]
    # where the sum is over j such that 0 <= j-i < n (p's length)
    # Let k = j-i, so j = k+i. g[i] = 2 * sum_k w[k+i] * p[k] (over valid k, k+i)
    # This is a cross-correlation: (w ⊛ p)[i] = sum_k p[k] * w[k+i]
    # Use FFT: cross-correlation of w and p is IFFT(conj(FFT(p)) * FFT(w))
    # Actually, let's use direct convolution:
    # g[i]/2 = sum_k p[k] * w[k+i] = (p * flip(w))[i + nw - 1 - ... ]
    # Simpler: just compute it via FFT
    L = _next_pow2(n + nw)
    fp = np.fft.rfft(p, L)
    fw = np.fft.rfft(w, L)
    # We want: g[i] = 2 * sum_k p[k] * w[k+i] for i = 0..n-1
    # This is the cross-correlation of w and p at lag i.
    # cross_corr[i] = sum_k p[k] * w[k+i] = IFFT(conj(FFT(p)) * FFT(w))[i] ... no
    # Actually: corr[i] = sum_k p[k+i] * conj(w[k]) ... 
    # Let me just use: g[i] = 2 * sum_{k=0}^{n-1} p[k] * w[k+i] (where k+i < nw)
    # = 2 * (p .* flip(w)) reversed ... 
    # Easiest: g = 2 * valid_convolution approach
    # g[i] = 2 * sum_k p[k] * w[k+i]
    # Let's define v = flip(w), so v[m] = w[nw-1-m]
    # Then g[i] = 2 * sum_k p[k] * w[k+i] = 2 * sum_k p[k] * v[nw-1-k-i]
    # = 2 * conv(p, v)[nw-1-i]
    # where conv is 'full' convolution
    # So g[i] = 2 * conv_full(p, flip(w))[nw-1-i]
    v = w[::-1]
    cw = _conv_fft(p, v)
    # cw has length n + nw - 1
    # g[i] = 2 * cw[nw - 1 - i] for i = 0..n-1
    idx = nw - 1 - np.arange(n)
    g = 2.0 * cw[idx]
    g = np.where(np.isfinite(g), g, 0.0)
    return g


def _mmd_optimize(p0, rng, time_limit, beta_start=50.0, beta_end=2000.0):
    """Multiplicative mirror descent on simplex with adaptive parameters."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    if time_limit <= 0:
        return p, _score(p)

    n = p.size
    floor = 1e-20 / n
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_score = _score(p)
    cur_score = best_score

    beta = beta_start
    step = 0.5
    deadline = time.monotonic() + time_limit
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        c = _conv_fft(p, p)
        d = n * c
        w = _softmax_weights(d, beta)
        smooth_val = float(np.dot(w, d))
        g = _gradient_from_weights(w, p)
        g = np.maximum(g, 0.0)
        maxg = g.max()
        if maxg < 1e-15:
            stall += 1
        else:
            g = g / maxg
            # Try multiple step sizes
            best_trial = None
            best_trial_smooth = smooth_val
            for mult in (0.25, 0.5, 1.0, 2.0, 4.0):
                a = step * mult
                if a < 1e-6 or a > 50:
                    continue
                p_try = p * np.exp(-a * g)
                p_try = np.maximum(p_try, floor)
                s = p_try.sum()
                if s <= 0:
                    continue
                p_try = p_try / s
                c_try = _conv_fft(p_try, p_try)
                d_try = n * c_try
                w_try = _softmax_weights(d_try, beta)
                smooth_try = float(np.dot(w_try, d_try))
                if smooth_try < best_trial_smooth - 1e-12:
                    best_trial_smooth = smooth_try
                    best_trial = p_try

            if best_trial is not None:
                p = best_trial
                cur_score = _score(p)
                stall = 0
                if cur_score < best_score - 1e-10:
                    best = p.copy()
                    best_score = cur_score
                # Adapt step
                it += 1
                if it % 30 == 0:
                    beta = min(beta_end, beta * 1.2)
            else:
                stall += 1
                step = max(1e-5, step * 0.7)
                if stall >= 20:
                    # Perturb
                    sigma = 0.03 + 0.01 * min(stall, 20)
                    p_try = _sanitize(p * np.exp(rng.normal(0, sigma, n)))
                    if p_try is not None:
                        sc = _score(p_try)
                        if sc < best_score * 1.1:
                            p = p_try
                            cur_score = sc
                    else:
                        p = best.copy()
                        cur_score = best_score
                    stall = 0
                    step = 0.5
                if stall > 100:
                    p = best.copy()
                    cur_score = best_score
                    step = 0.5
                    beta = max(beta_start, beta * 0.7)
                    stall = 0

    return best, best_score


def _coord_shoot(p0, rng, time_limit):
    """Coordinate descent: for each coordinate, find optimal value."""
    p = _sanitize(p0)
    if p is None or time_limit <= 0:
        return None, float('inf')
    n = p.size
    best = p.copy()
    best_score = _score(p)
    cur_score = best_score
    deadline = time.monotonic() + time_limit
    stagnant = 0

    while time.monotonic() < deadline:
        improved = False
        # Random order of coordinates
        order = rng.permutation(n)
        for i in order:
            if time.monotonic() >= deadline:
                break
            # Try different values for p[i]
            old_val = p[i]
            sum_rest = 1.0 - old_val
            if sum_rest < 0:
                continue
            # The convolution with p[i] set to x:
            # c_new[j] = c_old[j] - 2*old_val*a[j-i] + 2*x*a[j-i] + old_val^2 - x^2 (for j=2i)
            # Actually let's just try a few multipliers
            best_local = old_val
            best_local_score = cur_score
            for mult in (0.3, 0.6, 0.8, 1.2, 1.5, 2.0, 3.0):
                x = old_val * mult
                if x < 0 or x > 1:
                    continue
                p_try = p.copy()
                p_try[i] = x
                p_try = np.maximum(p_try, 0)
                s = p_try.sum()
                if s < 1e-10:
                    continue
                p_try = p_try / s
                sc = _score(p_try)
                if sc < best_local_score - 1e-12:
                    best_local_score = sc
                    best_local = x
            if best_local != old_val:
                p[i] = best_local
                p = p / p.sum()
                cur_score = best_local_score
                improved = True
                if cur_score < best_score - 1e-10:
                    best = p.copy()
                    best_score = cur_score

        if not improved:
            stagnant += 1
            if stagnant > 30:
                # Perturb
                p_try = _sanitize(p * np.exp(rng.normal(0, 0.05, n)))
                if p_try is not None:
                    sc = _score(p_try)
                    if sc < best_score * 1.05:
                        p = p_try
                        cur_score = sc
                    else:
                        p = best.copy()
                        cur_score = best_score
                stagnant = 0
        else:
            stagnant = 0

        if stagnant > 200:
            break

    return best, best_score


def _block_search(p0, rng, time_limit):
    """Block-based local search: multiply random contiguous blocks."""
    p = _sanitize(p0)
    if p is None or time_limit <= 0:
        return None, float('inf')
    n = p.size
    best = p.copy()
    best_score = _score(p)
    cur_score = best_score
    deadline = time.monotonic() + time_limit
    stagnant = 0
    factors = (0.4, 0.7, 0.85, 1.15, 1.3, 1.7, 2.5)

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.4:
            w = int(rng.integers(1, max(2, n // 100 + 1)))
        elif r < 0.8:
            w = int(rng.integers(max(2, n // 100), max(3, n // 20 + 1)))
        else:
            w = int(rng.integers(max(3, n // 20), max(4, n // 5 + 1)))
        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))

        base = p.copy()
        best_cand = base
        best_sc = cur_score
        for f in factors:
            cand = base.copy()
            cand[start:start+w] *= f
            cand = _sanitize(cand)
            if cand is None:
                continue
            sc = _score(cand)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_cand = cand

        if best_cand is not base:
            p = best_cand
            cur_score = best_sc
            stagnant = 0
            if cur_score < best_score - 1e-10:
                best = p.copy()
                best_score = cur_score
        else:
            stagnant += 1
            if stagnant > 80:
                p_try = _sanitize(p * np.exp(rng.normal(0, 0.05, n)))
                if p_try is not None:
                    sc = _score(p_try)
                    if sc < best_score * 1.05:
                        p = p_try
                        cur_score = sc
                    else:
                        p = best.copy()
                        cur_score = best_score
                stagnant = 0
            if stagnant > 400:
                break

    return best, best_score


def _init_geometric(n, r, rng):
    """Geometric-like sequence with some noise."""
    t = np.arange(n, dtype=float)
    a = np.power(r, t)
    a = a + rng.normal(0, 0.01, n)
    a = np.maximum(a, 0)
    return _sanitize(a)


def _init_smooth_random(n, rng):
    """Smooth random sequence via cosine series."""
    t = np.arange(n, dtype=float) / n
    z = np.zeros(n)
    for k in range(1, 15):
        amp = rng.normal(0, 1.0 / k)
        ph = rng.uniform(0, 2 * np.pi)
        z += amp * np.cos(2 * np.pi * k * t + ph)
    a = np.exp(z)
    return _sanitize(a)


def _init_two_block(n, rng):
    """Two-block sequence."""
    a = np.ones(n)
    split = int(rng.uniform(0.3, 0.7) * n)
    ratio = float(rng.choice([0.3, 0.5, 0.7, 1.5, 2.0, 3.0]))
    a[split:] *= ratio
    return _sanitize(a)


def _init_beta_like(n, rng):
    """Beta-distribution-like sequence."""
    t = np.linspace(0.001, 0.999, n)
    a = float(rng.uniform(0.1, 3.0))
    b = float(rng.uniform(0.1, 3.0))
    p = np.power(t, a) * np.power(1 - t, b)
    return _sanitize(p)


def _resample_to_length(arr, target_n):
    a = _sanitize(arr)
    if a is None:
        return None
    n = a.size
    if n == target_n:
        return a.copy()
    x_old = np.linspace(0.5/n, 1.0 - 0.5/n, n)
    x_new = np.linspace(0.5/target_n, 1.0 - 0.5/target_n, target_n)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0)
    return _sanitize(out)
