def _lp_solve(A, M, n, g_max):
    """Solve LP: max sum(g) s.t. A@g <= M, 0 <= g <= g_max. Returns g or None."""
    c = -np.ones(n)
    b_ub = np.full(2*n-1, M)
    bounds = [(0.0, g_max)] * n
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if res.success and res.x is not None:
        return res.x
    return None

def _coordinate_descent(f, n, deadline, rng):
    """Coordinate descent polish: try moving mass between coordinates."""
    f = f.copy()
    best_score = _eval_local(f)
    best_f = f.copy()
    best_peak = _conv_peak(f)

    c = _conv_full(f)
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        # Find peak lag
        k_star = int(np.argmax(c))

        # Contributions to peak: c[k_star] = sum_i f[i]*f[k_star-i]
        # Gradient of c[k_star] w.r.t. f[j] = f[k_star-j] (if valid)
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = 2.0 * f[idx]

        # Find best coordinate to decrease (high contribution) and increase (low contribution)
        # Decrease where f[j] * grad[j] is high, increase where it's low
        contrib = f * grad

        # Try several pairs
        tried = 0
        while tried < 20:
            # Pick source (high contribution) and destination (low contribution)
            src_candidates = np.argsort(contrib)[::-1][:20]
            dst_candidates = np.argsort(contrib)[:20]

            src = int(src_candidates[tried % len(src_candidates)])
            dst = int(dst_candidates[tried % len(dst_candidates)])
            tried += 1

            if src == dst:
                continue

            for frac in [0.1, 0.05, 0.02]:
                amt = f[src] * frac
                f_try = f.copy()
                f_try[src] -= amt
                f_try[dst] += amt
                if f_try[src] < 0:
                    continue
                f_try /= f_try.sum()
                peak_try = _conv_peak(f_try)
                if peak_try < best_peak - 1e-12:
                    f = f_try
                    best_peak = peak_try
                    c = _conv_full(f)
                    best_score = _eval_local(f)
                    best_f = f.copy()
                    no_improve = 0
                    break

            no_improve += 1
            if no_improve > 50:
                # Random perturbation
                f = f + 0.002 * rng.standard_normal(n)
                f = np.maximum(f, 0.0)
                f /= f.sum()
                best_peak = _conv_peak(f)
                c = _conv_full(f)
                no_improve = 0
                break

        it += 1

    return best_f, best_score
