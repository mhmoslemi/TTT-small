    grad = np.zeros(n)
    count = 0
    for T in Ts:
        if T < 1e-300:
            continue
        cs = c / T
        cs -= cs.max()
        p = np.exp(cs)
        s = p.sum()
        if s < 1e-300:
            continue
        p /= s
        grad += correlate(p, f, mode='valid')
        count += 1
    if count > 0:
        grad *= 2.0 / count
    else:
        grad = _exact_peak_gradient(c, f)[0]
    return grad


def _mirror_step_momentum(f, grad, step, velocity, beta):
    """Mirror descent step with momentum in log-space."""
    velocity = beta * velocity + (1.0 - beta) * grad
    log_f = np.log(f + 1e-300) - step * velocity
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f, velocity
    f_new /= s
    return f_new, velocity


def _optimize_mirror(f, n, deadline, rng, verbose_prefix=""):
    """
    Core optimization: mirror descent with momentum and adaptive temperature.
    Uses exact peak gradient when well-separated, soft-max otherwise.
    """
    f = np.maximum(f, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f = f / s

    local_best = _score_fast(f)
    local_best_f = f.copy()

    c = _autocorr_fft(f)
    peak0 = c.max()
    T0 = peak0 * 0.1  # Start temperature
    T_min = peak0 * 1e-9

    # Momentum parameter
    beta = 0.3

    # Velocity (momentum) in log-space
    velocity = np.zeros(n)

    n_iters = 300000
    stuck = 0
    best_streak = 0

    for it in range(n_iters):
        if time.monotonic() > deadline:
            break

        progress = min(1.0, it / n_iters)

        # Adaptive temperature: geometric annealing with occasional reheating
        T_base = T0 * (T_min / T0) ** (progress ** 0.6)

        # If stuck, temporarily increase temperature
        if stuck > 2:
            T_base = min(T0, T_base * 5.0)

        c = _autocorr_fft(f)
        peak_val = c.max()

        # Check if peak is well-separated from second peak
        c_sorted = np.sort(c)[::-1]
        separation = c_sorted[0] / max(c_sorted[1], 1e-300)

        if separation > 3.0:
            # Use exact peak gradient (more aggressive)
            grad, pv, ks = _exact_peak_gradient(c, f)
            # Step size: larger when well-separated
            step = 0.3 * (0.05) ** (progress ** 0.4)
        else:
            # Use multi-temperature soft-max gradient
            Ts = [T_base * 0.3, T_base, T_base * 3.0]
            grad = _soft_gradient_multi(c, f, Ts)
            step = 0.2 * (0.02) ** (progress ** 0.4)

        f, velocity = _mirror_step_momentum(f, grad, step, velocity, beta)

        # Periodic evaluation
        if it % 3000 == 0 or it == n_iters - 1:
            sc = _score_fast(f)
            if sc < local_best - 1e-7:
                local_best = sc
                local_best_f = f.copy()
                stuck = 0
                best_streak += 1
            else:
                stuck += 1
                best_streak = 0

            # If stuck for a while, apply perturbation
            if stuck > 5:
                # Decay momentum
                velocity *= 0.5
                # Random perturbation
                magnitude = 0.003 / np.sqrt(n)
                f = f + magnitude * rng.standard_normal(n)
                f = np.maximum(f, 0.0)
                f = f / f.sum()
                stuck = 0

            # Decay momentum periodically to allow exploration
            if it % 10000 == 0 and it > 0:
                velocity *= 0.8

    sc = _score_fast(f)
    if sc < local_best:
        local_best = sc
        local_best_f = f.copy()

    return local_best_f, local_best


def _polish_coordinate(f, n, deadline, rng):
    """
    Coordinate descent polish: iteratively try to reduce the peak by
    transferring mass from high-contribution positions to low ones.
    """
    f = f.copy()
    local_best = _score_fast(f)
    local_best_f = f.copy()
    c = _autocorr_fft(f)
    peak = c.max()

    max_passes = 5000
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break

        c = _autocorr_fft(f)
        peak = c.max()
        k_star = int(np.argmax(c))

        # Compute contribution of each coordinate to the peak
        contrib = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                contrib[j] = f[j] * f[idx]

        # Try to reduce the peak by moving mass
        # Donor: highest contribution
        j_donate = int(np.argmax(contrib))

        # Receiver: lowest contribution (or random)
        contrib_sorted = np.argsort(contrib)
        candidates = [contrib_sorted[0], contrib_sorted[1], contrib_sorted[2]]
        
        improved = False
        for j_recv in candidates:
            if j_recv == j_donate:
                continue
            for frac in [0.02, 0.05, 0.1, 0.15, 0.25]:
                f_try = f.copy()
                amt = f_try[j_donate] * frac
                if amt < 1e-15:
                    continue
                f_try[j_donate] -= amt
                f_try[j_recv] += amt
                f_try = np.maximum(f_try, 0.0)
                f_try /= f_try.sum()
                c_try = _autocorr_fft(f_try)
                if c_try.max() < peak - 1e-14:
                    f = f_try
                    peak = c_try.max()
                    improved = True
                    local_best = _score_fast(f)
                    local_best_f = f.copy()
                    break
            if improved:
                break

        # Random coordinate perturbations
        for _ in range(5):
            i = int(rng.integers(0, n))
            jx = int(rng.integers(0, n))
            if i == jx:
                continue
            f_try = f.copy()
            amt = min(f_try[i], f_try[jx]) * 0.03
            direction = 1 if f_try[i] > f_try[jx] else -1
            f_try[i] -= direction * amt
            f_try[jx] += direction * amt
            f_try = np.maximum(f_try, 0.0)
            f_try /= f_try.sum()
            c_try = _autocorr_fft(f_try)
            if c_try.max() < peak - 1e-14:
                f = f_try
                peak = c_try.max()
                local_best = _score_fast(f)
                local_best_f = f.copy()

    return local_best_f, local_best


def _parametric_starts(n, rng):
    """Generate diverse parametric starting points."""
    j = np.arange(n, dtype=np.float64)
    x = j / n  # normalized [0, 1)

    starts = {}

    # Exponential decays (based on analysis showing edge-weighting helps)
    for rate in [0.5, 1.0, 2.0, 4.0, 8.0]:
        starts[f'exp_{rate}'] = np.exp(-rate * x) + 0.05

    # Power law decays
    for p in [0.5, 1.0, 2.0, 3.0]:
        starts[f'pow_{p}'] = (1.0 - x) ** p + 0.05

    # Linear
    starts['linear_down'] = 1.0 + 0.5 * (1.0 - x)
    starts['linear_up'] = 1.0 + 0.5 * x

    # Quadratic / U-shaped
    starts['quad_center'] = 1.0 + 0.5 * (1.0 - 2.0 * x) ** 2
    starts['quad_edge'] = 1.0 + 1.0 * (1.0 - x) ** 2

    # Cosine / sine
    starts['cos_half'] = np.cos(np.pi * x / 2.0) + 0.05
    starts['sine_half'] = np.sin(np.pi * x / 2.0) + 0.05
    starts['sine_full'] = 1.0 + 0.3 * np.sin(np.pi * x)

    # Bimodal (peaks away from center)
    for pos in [0.2, 0.3, 0.7, 0.8]:
        starts[f'bimod_{pos}'] = np.exp(-((x - pos) ** 2) / 0.05) + 0.05

    # Random
    starts['rand_exp'] = rng.exponential(1.0, n)
    starts['rand_dirichlet'] = rng.dirichlet(np.ones(n) * 0.01)
    starts['rand_smooth'] = np.convolve(rng.standard_normal(n), np.ones(50) / 50, 'same')
    starts['rand_smooth'] = np.abs(starts['rand_smooth']) + 0.1

    return starts


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the non-negative sequence minimizing 2*n*max(f*f)/(sum f)^2.

    Strategy:
    1. Rich parametric initializations (exponential, polynomial, random).
    2. Mirror descent with momentum and adaptive multi-temperature gradient.
    3. Exact peak gradient when well-separated, soft-max otherwise.
    4. Coordinate-descent polish on the best solution.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 5.0

    # Load initial sequence
    f_init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(f_init)

    s0 = f_init.sum()
    if s0 > 1e-10:
        f0 = f_init / s0
    else:
        f0 = np.ones(n) / n

    best_score = _score_fast(f0)
    best_f = f0.copy()

    # Generate starting points
    starts = _parametric_starts(n, rng)
    starts['given'] = f0.copy()

    # Evaluate all starts
    scored = []
    for name, f in starts.items():
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s < 1e-10:
            f = np.ones(n) / n
        else:
            f = f / s
        sc = _score_fast(f)
        scored.append((sc, name, f))

    scored.sort(key=lambda x: x[0])

    print(f"n = {n}")
    print("Top starting points:")
    for sc, name, _ in scored[:8]:
        print(f"  {name}: {sc:.6f}")

    if scored[0][0] < best_score:
        best_score = scored[0][0]
        best_f = scored[0][2].copy()

    # Phase 1: Optimize top 5 diverse starts
    top_k = min(5, len(scored))
    time_per_start = (budget_s - 120) / (top_k + 1)  # +1 for final deep opt

    for idx in range(top_k):
        if time.monotonic() > deadline - 120:
            break

        sc, name, f = scored[idx]
        t_alloc = time_per_start * (1.0 - 0.1 * idx)
        start_dl = time.monotonic() + t_alloc

        print(f"\n[{idx+1}/{top_k}] Optimizing '{name}' (init score={sc:.6f}, budget={t_alloc:.0f}s)")
        f_opt, sc_opt = _optimize_mirror(f, n, start_dl, rng)

        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  New best: {best_score:.6f}")
        else:
            print(f"  Best from this start: {sc_opt:.6f} (global best: {best_score:.6f})")

    # Phase 2: Deep optimization from the global best
    if time.monotonic() < deadline - 60:
        remaining = min(300.0, deadline - time.monotonic() - 30)
        print(f"\nDeep optimization (score={best_score:.6f}, budget={remaining:.0f}s)")
        f_deep, sc_deep = _optimize_mirror(best_f, n, time.monotonic() + remaining, rng)
        if sc_deep < best_score:
            best_score = sc_deep
            best_f = f_deep.copy()
            print(f"  Deep result: {best_score:.6f}")

    # Phase 3: Coordinate descent polish
    if time.monotonic() < deadline - 10:
        polish_dl = min(time.monotonic() + 60.0, deadline - 5.0)
        print(f"\nPolishing (score={best_score:.6f})...")
        f_pol, sc_pol = _polish_coordinate(best_f, n, polish_dl, rng)
        if sc_pol < best_score:
            best_score = sc_pol
            best_f = f_pol.copy()
            print(f"  Polished: {best_score:.6f}")

    # Final verification with actual evaluator
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = evaluate_sequence(best_f.tolist())

    # Double-check our fast score matches
    fast_score = _score_fast(best_f)
    if abs(final_score - fast_score) > 0.01:
        print(f"  WARNING: score mismatch! fast={fast_score:.6f}, actual={final_score:.6f}")

    if final_score < best_score or not np.isfinite(best_score):
        best_score = final_score

    print(f"\n=== FINAL SCORE: {best_score:.6f} ===")
    return [float(x) for x in best_f]
