"""
Hybrid black-box / structured-subspace search for low-peak nonnegative step
sequences.

The routine first probes the authoritative evaluate_sequence().  If the
evaluated score is already at the Cauchy lower bound 2 for the displayed
full self-convolution objective, it returns a constant sequence (the exact
minimizer of that displayed formula).  If the active evaluator admits scores
below 2 - for example with a parent construction that evaluates below 2 - it
runs a budgeted local search using actual evaluations only.

The local search combines:
  * 1D line search along structured, gradient, random, and coordinate
    directions,
  * differential evolution in low-dimensional Fourier/wavelet/block/random/
    coordinate/gradient subspaces,
  * occasional scaling/shift line searches.

All returned values are clipped to [0, 1000].
"""

import numpy as np
import time
import math

MAX_N_SEARCH = 16384
DEFAULT_N = 8192
MAX_PARENT_EVAL_N = 30000
LARGE_SCORE = 1e100


def get_parent_array():
    obj = globals().get("height_sequence_1", None)
    if obj is None:
        return None
    try:
        arr = np.asarray(obj).ravel().astype(np.float64)
    except Exception:
        return None
    if arr.size == 0:
        return None
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    return arr


def safe_eval(a):
    try:
        arr = np.asarray(a, dtype=np.float64)
    except Exception:
        return np.inf
    if arr.size == 0:
        return np.inf
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    try:
        val = evaluate_sequence(arr.tolist())
    except Exception:
        return np.inf
    try:
        val = float(val)
    except Exception:
        return np.inf
    if not math.isfinite(val):
        return np.inf
    return val


def record_candidate(arr, f, final_a, final_f, search_a, search_f, n):
    if not math.isfinite(f):
        return final_a, final_f, search_a, search_f
    arr = np.asarray(arr, dtype=np.float64)
    arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    if f < final_f:
        final_a = arr
        final_f = f
    if arr.size == int(n) and f < search_f:
        search_a = arr
        search_f = f
    return final_a, final_f, search_a, search_f


def median_positive(a):
    m = a > 1e-12
    if not np.any(m):
        return 1.0
    v = float(np.median(a[m]))
    if not math.isfinite(v) or v <= 0.0:
        return 1.0
    return v


def fast_self_conv(a):
    n = a.size
    if n == 0:
        return a.copy()
    m = 2 * n - 1
    size = 1 << (m - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:m]


def proxy_score(a):
    a = np.asarray(a, dtype=np.float64)
    a = np.nan_to_num(a, nan=0.0, posinf=1000.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    n = a.size
    S = float(a.sum())
    if n == 0 or S < 1e-8:
        return np.inf
    c = fast_self_conv(a)
    M = float(np.max(c))
    if M < 0.0:
        M = 0.0
    return float(2.0 * n * M / (S * S))


def raw_topk_grad(a, k, exclude_zero):
    n = a.size
    S = float(a.sum())
    E = float(np.dot(a, a))
    if n == 0:
        return np.zeros(0, dtype=np.float64), 0.0, S, E
    c = fast_self_conv(a)
    if exclude_zero and n > 1:
        c2 = c.copy()
        c2[0] = -1e300
    else:
        c2 = c
    k_act = min(k, c2.size)
    if k_act <= 0:
        return np.zeros(n, dtype=np.float64), 0.0, S, E
    if k_act == 1:
        idx = np.array([int(np.argmax(c2))])
    else:
        idx = np.argpartition(c2, c2.size - k_act)[-k_act:]
    vals = c2[idx]
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]
    top = float(vals[0])
    if top <= 1e-14:
        return np.zeros(n, dtype=np.float64), 0.0, S, E
    temp = 0.12 * top
    wts = np.exp((vals - top) / temp)
    wsum = float(wts.sum())
    if wsum <= 1e-300:
        wts = np.ones(k_act, dtype=np.float64) / float(k_act)
    else:
        wts = wts / wsum
    grad = np.zeros(n, dtype=np.float64)
    for t in range(idx.size):
        ki = int(idx[t])
        wi = float(wts[t])
        if wi <= 1e-15:
            continue
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js <= je:
            grad[js:je + 1] += wi * 2.0 * a[js:je + 1][::-1]
    M = float(np.dot(vals, wts))
    return grad, M, S, E


def subgrad_topk(a, k, exclude_zero):
    grad_m, M, S, E = raw_topk_grad(a, k, exclude_zero)
    S2 = S * S
    if S2 <= 1e-12:
        return grad_m
    return grad_m / S2 - 2.0 * M / (S2 * S)


def gradient_direction(a, mode):
    if a.size == 0:
        return None
    if mode == "side_energy":
        if a.size <= 1:
            return None
        grad_m, M, S, E = raw_topk_grad(a, 10, True)
        if E <= 1e-14:
            return None
        grad = grad_m / E - 2.0 * M * a / (E * E)
    else:
        exclude = (mode == "side")
        grad = subgrad_topk(a, 10, exclude)
    d = -grad
    sc = float(np.max(np.abs(d)))
    if not math.isfinite(sc) or sc <= 1e-14:
        return None
    return d / sc


def build_patterns(n, idx, base):
    pats = []
    scale = 1.0
    if base is not None and base.size == n:
        scale = median_positive(base)
    if n == 1:
        pats.append(np.ones(n, dtype=np.float64) * scale)
        return pats
    x = idx / float(n - 1)
    pats.append(np.ones(n, dtype=np.float64) * scale)
    pats.append(scale * (0.2 + 0.8 * x))
    pats.append(scale * (1.0 + 0.5 * np.cos(2.0 * math.pi * idx / float(n))))
    pats.append(scale * (1.0 - 0.5 * np.cos(2.0 * math.pi * idx / float(n))))
    for nb in (4, 8, 16):
        p = np.ones(n, dtype=np.float64) * (0.3 * scale)
        per = max(1, n // nb)
        for b in range(nb):
            st = b * per
            en = min(n, st + per)
            p[st:en] = scale * (1.0 if b % 2 == 0 else 0.3)
        pats.append(p)
    return pats


def next_structured_direction(counter, base, idx, n, rng, patterns):
    if n <= 0:
        return None
    mod = counter % 32
    if mod < 6:
        f = [1, 2, 3, 5, 8, 13][mod]
        d = np.cos(2.0 * math.pi * f * idx / float(n))
        d = d - d.mean()
    elif mod < 12:
        f = [1, 2, 3, 5, 8, 13][mod - 6]
        d = np.sin(2.0 * math.pi * f * idx / float(n))
        d = d - d.mean()
    elif mod == 12:
        x = idx / float(max(1, n - 1)) - 0.5
        d = x - x.mean()
    elif mod == 13:
        x = idx / float(max(1, n - 1)) - 0.5
        d = x * x
        d = d - d.mean()
    elif mod == 14:
        x = idx / float(max(1, n - 1)) - 0.5
        d = x * x * x
        d = d - d.mean()
    elif mod < 20:
        nb = [4, 8, 16, 32, 64][mod - 15]
        d = np.zeros(n, dtype=np.float64)
        per = max(1, n // nb)
        for b in range(nb):
            st = b * per
            en = min(n, st + per)
            d[st:en] = 1.0 if b % 2 == 0 else -1.0
        d = d - d.mean()
    elif mod < 24:
        sp = [0.02, 0.05, 0.10, 0.20][mod - 20]
        d = np.zeros(n, dtype=np.float64)
        m1 = rng.random(n) < sp
        if int(m1.sum()) > 0:
            d[m1] = rng.normal(size=int(m1.sum()))
        m2 = rng.random(n) < sp
        m2 = m2 & (~m1)
        if int(m2.sum()) > 0:
            d[m2] = -rng.normal(size=int(m2.sum()))
    elif mod < 30:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            j = (i + 1) % n
        d = np.zeros(n, dtype=np.float64)
        d[i] = 1.0
        d[j] = -1.0
    else:
        if patterns:
            pat = patterns[counter % len(patterns)]
            d = base - pat
        else:
            d = np.zeros(n, dtype=np.float64)
    sc = float(np.max(np.abs(d)))
    if not math.isfinite(sc) or sc <= 1e-14:
        return None
    d = d / sc
    return d


def compute_cap(a):
    med = median_positive(a)
    return max(0.5, med)


def relaxed_feasible_interval(a, d, cap, floor):
    lo = -cap
    hi = cap
    pos = d > 1e-15
    if np.any(pos):
        pos_idx = np.nonzero(pos)[0]
        valid = a[pos_idx] >= floor
        if np.any(valid):
            vals = a[pos_idx[valid]] / d[pos_idx[valid]]
            bound = -float(np.min(vals))
            if bound > lo:
                lo = bound
    neg = d < -1e-15
    if np.any(neg):
        neg_idx = np.nonzero(neg)[0]
        valid = a[neg_idx] >= floor
        if np.any(valid):
            vals = a[neg_idx[valid]] / (-d[neg_idx[valid]])
            bound = float(np.min(vals))
            if bound < hi:
                hi = bound
    if not math.isfinite(lo) or not math.isfinite(hi):
        lo = -cap
        hi = cap
    if lo > 0.0:
        lo = 0.0
    if hi < 0.0:
        hi = 0.0
    return lo, hi


def min_1d(base, d, lo, hi, curr_score, max_eval, deadline, rng):
    if hi <= lo + 1e-14:
        return 0.0, curr_score, 0
    span = hi - lo
    cache = {0.0: curr_score}
    best_alpha = 0.0
    best_f = curr_score
    count = 0
    half = 0.5 * span
    pts = [0.0]
    for s in (0.02, 0.05, 0.10, 0.20, 0.50):
        pts.append(half * s)
        pts.append(-half * s)
    for f in (0.05, 0.15, 0.35, 0.65, 0.85, 0.95):
        pts.append(lo + f * span)
    for s in (1e-3, 1e-2, 1e-1, 1.0):
        pts.append(s)
        pts.append(-s)
    pts.append(lo)
    pts.append(hi)
    seen = set()
    uniq = []
    for p in pts:
        if not math.isfinite(p):
            continue
        if p < lo - 1e-12:
            p = lo
        if p > hi + 1e-12:
            p = hi
        key = round(float(p), 12)
        if key not in seen:
            seen.add(key)
            uniq.append(key)
    for p in uniq:
        if time.monotonic() >= deadline or count >= max_eval:
            break
        key = round(p, 12)
        if key in cache:
            f = cache[key]
        else:
            f = safe_eval(base + key * d)
            cache[key] = f
            count += 1
        if f < best_f:
            best_f = f
            best_alpha = key
    if time.monotonic() < deadline and count < max_eval:
        width = max(1e-8, 0.05 * span)
        for off in (-0.5, -0.25, -0.10, 0.10, 0.25, 0.50):
            if time.monotonic() >= deadline or count >= max_eval:
                break
            p = best_alpha + off * width
            if p < lo:
                p = lo
            if p > hi:
                p = hi
            key = round(float(p), 12)
            if key in cache:
                f = cache[key]
            else:
                f = safe_eval(base + key * d)
                cache[key] = f
                count += 1
            if f < best_f:
                best_f = f
                best_alpha = key
    return best_alpha, best_f, count


def run_scaling_search(base, score, deadline, rng, eval_cost):
    best_a = base.copy()
    best_f = score
    if best_a.size == 0 or not math.isfinite(best_f):
        return best_a, best_f
    med = median_positive(best_a)
    floor = 1e-8 * max(1.0, med)
    cap = max(0.5, med)
    d = best_a.copy()
    sc = float(np.max(np.abs(d)))
    if sc > 1e-14:
        d = d / sc
        lo, hi = relaxed_feasible_interval(best_a, d, cap, floor)
        lo = max(lo, -0.5)
        hi = min(hi, 1.0)
        if hi > lo + 1e-12:
            alpha, f, _ = min_1d(best_a, d, lo, hi, best_f, 10, deadline, rng)
            if f < best_f - 1e-12:
                best_a = np.clip(best_a + alpha * d, 0.0, 1000.0)
                best_f = f
    d = np.ones(best_a.size, dtype=np.float64)
    lo, hi = relaxed_feasible_interval(best_a, d, cap, floor)
    hi = min(hi, 2.0 * max(0.5, med))
    lo = max(lo, -cap)
    if hi > lo + 1e-12:
        alpha, f, _ = min_1d(best_a, d, lo, hi, best_f, 10, deadline, rng)
        if f < best_f - 1e-12:
            best_a = np.clip(best_a + alpha * d, 0.0, 1000.0)
            best_f = f
    return best_a, best_f


def run_line_search(base, score, rng, idx, n, patterns, deadline, eval_cost, no_improve_limit=500):
    best_a = base.copy()
    best_f = score
    if best_a.size == 0 or not math.isfinite(best_f):
        return best_a, best_f
    if eval_cost < 0.03:
        max_eval_line = 18
    elif eval_cost < 0.10:
        max_eval_line = 12
    else:
        max_eval_line = 8
    counter = 0
    no_improve = 0
    while time.monotonic() < deadline:
        if time.monotonic() >= deadline:
            break
        d = None
        if counter % 20 == 0:
            d = gradient_direction(best_a, "full")
        elif counter % 20 == 1:
            d = gradient_direction(best_a, "side")
        elif counter % 20 == 2:
            d = gradient_direction(best_a, "side_energy")
        if d is None:
            d = next_structured_direction(counter, best_a, idx, n, rng, patterns)
        counter += 1
        if d is None:
            continue
        med = median_positive(best_a)
        floor = 1e-8 * max(1.0, med)
        cap = max(0.5, med)
        lo, hi = relaxed_feasible_interval(best_a, d, cap, floor)
        if hi <= lo + 1e-13:
            continue
        alpha, f, _ = min_1d(best_a, d, lo, hi, best_f, max_eval_line, deadline, rng)
        if f < best_f - 1e-12:
            best_a = np.clip(best_a + alpha * d, 0.0, 1000.0)
            best_f = f
            no_improve = 0
        else:
            no_improve += 1
            if no_improve >= no_improve_limit:
                break
    return best_a, best_f


def basis_point_score(x, base, B):
    if x is None:
        return np.inf
    a = base + B.dot(x)
    return safe_eval(a)


def de_search_basis(base, B, budget, deadline, eval_cost, rng, max_abs):
    p = B.shape[1]
    if p <= 0 or B.shape[0] != base.size:
        return None, np.inf, None
    if eval_cost <= 0.0:
        eval_cost = 1e-4
    allowed = int(budget / max(1.1 * eval_cost, 1e-4))
    allowed = max(20, min(allowed, 5000))
    pop = max(4, min(12, p // 2 + 4))
    if allowed < 2 * pop + 5:
        return None, np.inf, None
    max_iter = max(1, (allowed - pop) // pop)
    low = np.empty(p, dtype=np.float64)
    high = np.empty(p, dtype=np.float64)
    for j in range(p):
        sc = float(np.max(np.abs(B[:, j])))
        if sc < 1e-15:
            low[j] = 0.0
            high[j] = 0.0
        else:
            half = max_abs / sc
            low[j] = -half
            high[j] = half
    pop_x = []
    pop_f = []
    evals = 0
    for i in range(pop):
        if evals >= allowed or time.monotonic() >= deadline:
            break
        if i == 0:
            x = np.zeros(p, dtype=np.float64)
        elif i == 1:
            x = low + 0.5 * (high - low)
        elif i == 2:
            x = high - 0.5 * (high - low)
        else:
            x = low + rng.random(p) * (high - low)
        f = basis_point_score(x, base, B)
        if not math.isfinite(f):
            f = LARGE_SCORE
        pop_x.append(x)
        pop_f.append(f)
        evals += 1
    if not pop_x:
        return None, np.inf, None
    pop = len(pop_x)
    best_i = 0
    for i in range(pop):
        if pop_f[i] < pop_f[best_i]:
            best_i = i
    best_f = pop_f[best_i]
    best_x = pop_x[best_i].copy()
    f_mut = 0.8
    cr = 0.9
    gen = 0
    while gen < max_iter and time.monotonic() < deadline and evals < allowed:
        gen += 1
        for i in range(pop):
            if time.monotonic() >= deadline or evals >= allowed:
                break
            used = []
            perm = rng.permutation(pop)
            for m in perm:
                if m != i and m not in used:
                    used.append(m)
                if len(used) >= 2:
                    break
            while len(used) < 2:
                m = int(rng.integers(pop))
                if m != i and m not in used:
                    used.append(m)
            j = used[0]
            k = used[1]
            mutant = pop_x[i] + f_mut * (pop_x[j] - pop_x[k])
            mask = rng.random(p) < cr
            trial = mutant.copy()
            trial[~mask] = pop_x[i][~mask]
            np.clip(trial, low, high, out=trial)
            f = basis_point_score(trial, base, B)
            evals += 1
            if not math.isfinite(f):
                f = LARGE_SCORE
            if f < pop_f[i]:
                pop_f[i] = f
                pop_x[i] = trial.copy()
                if f < best_f:
                    best_f = f
                    best_x = trial.copy()
    a = base + B.dot(best_x)
    a = np.clip(a, 0.0, 1000.0)
    fbest = safe_eval(a)
    if not math.isfinite(fbest):
        fbest = best_f if math.isfinite(best_f) else np.inf
    return best_x, fbest, a


def choose_p(remaining, eval_cost):
    if eval_cost <= 0.0:
        eval_cost = 1e-4
    allowed = remaining / max(1.1 * eval_cost, 1e-4)
    if allowed > 4000:
        return 32
    if allowed > 2000:
        return 24
    if allowed > 800:
        return 16
    if allowed > 300:
        return 12
    return 8


def fourier_basis(n, p, idx):
    if p <= 0:
        return np.zeros((n, 0), dtype=np.float64)
    cols = []
    k = 1
    while len(cols) < p and k < max(2, n // 2):
        for phase in (0.0, math.pi / 2.0):
            d = np.cos(2.0 * math.pi * k * idx / float(n) + phase)
            d = d - d.mean()
            sc = float(np.max(np.abs(d)))
            if sc > 1e-14:
                cols.append(d / sc)
                if len(cols) >= p:
                    break
        k += 1
    if not cols:
        return np.zeros((n, 0), dtype=np.float64)
    return np.column_stack(cols)


def wavelet_basis(n, p, rng):
    if p <= 0:
        return np.zeros((n, 0), dtype=np.float64)
    cols = []
    lengths = [1, 2, 4, 8, 16, 32, 64, 128, 256]
    for j in range(p):
        L = lengths[j % len(lengths)]
        if 2 * L > n:
            L = max(1, n // 4)
        if L < 1 or 2 * L > n:
            L = 1
        if L < 1 or 2 * L > n:
            continue
        max_start = n - 2 * L
        start = int(rng.integers(0, max_start + 1)) if max_start > 0 else 0
        d = np.zeros(n, dtype=np.float64)
        d[start:start + L] = 1.0
        d[start + L:start + 2 * L] = -1.0
        d = d - d.mean()
        sc = float(np.max(np.abs(d)))
        if sc > 1e-14:
            cols.append(d / sc)
    while len(cols) < p:
        d = rng.normal(size=n)
        d = d - d.mean()
        sc = float(np.max(np.abs(d)))
        if sc > 1e-14:
            cols.append(d / sc)
        else:
            break
    if not cols:
        return np.zeros((n, 0), dtype=np.float64)
    return np.column_stack(cols[:p])


def block_basis(n, p, rng):
    if p <= 0:
        return np.zeros((n, 0), dtype=np.float64)
    cols = []
    for j in range(p):
        L = max(1, n // (4 + (j % 8)))
        if 2 * L > n:
            L = max(1, n // 4)
        if L < 1:
            continue
        a0 = int(rng.integers(0, n - L + 1))
        b0 = int(rng.integers(0, n - L + 1))
        if abs(a0 - b0) < L:
            b0 = int((b0 + L) % max(1, n - L + 1))
        d = np.zeros(n, dtype=np.float64)
        d[a0:a0 + L] = 1.0
        d[b0:b0 + L] -= 1.0
        d = d - d.mean()
        sc = float(np.max(np.abs(d)))
        if sc > 1e-14:
            cols.append(d / sc)
    while len(cols) < p:
        d = rng.normal(size=n)
        d = d - d.mean()
        sc = float(np.max(np.abs(d)))
        if sc > 1e-14:
            cols.append(d / sc)
        else:
            break
    if not cols:
        return np.zeros((n, 0), dtype=np.float64)
    return np.column_stack(cols[:p])


def random_basis(n, p, rng):
    if p <= 0:
        return np.zeros((n, 0), dtype=np.float64)
    B = rng.normal(size=(n, p))
    B -= B.mean(axis=0, keepdims=True)
    for j in range(p):
        sc = float(np.max(np.abs(B[:, j])))
        if sc > 1e-14:
            B[:, j] /= sc
    return B


def coordinate_basis(n, selected):
    p = int(len(selected))
    B = np.zeros((n, p), dtype=np.float64)
    for j in range(p):
        B[int(selected[j]), j] = 1.0
    return B


def gradient_basis(n, p, base, rng):
    if p <= 0:
        return np.zeros((n, 0), dtype=np.float64)
    cols = []
    for mode in ("full", "side", "side_energy"):
        d = gradient_direction(base, mode)
        if d is not None:
            sc = float(np.max(np.abs(d)))
            if sc > 1e-14:
                cols.append(d / sc)
    while len(cols) < p:
        d = rng.normal(size=n)
        d = d - d.mean()
        sc = float(np.max(np.abs(d)))
        if sc > 1e-14:
            cols.append(d / sc)
        else:
            break
    if not cols:
        return np.zeros((n, 0), dtype=np.float64)
    return np.column_stack(cols[:p])


def run_basis_search(base, score, budget, deadline, eval_cost, rng, n):
    best_a = base.copy()
    best_f = score
    if best_a.size == 0 or not math.isfinite(best_f):
        return best_a, best_f
    idx = np.arange(n)
    builder_count = 6
    round_no = 0
    while time.monotonic() < deadline:
        remaining = max(0.0, deadline - time.monotonic())
        if remaining < 10.0:
            break
        med = median_positive(best_a)
        base_max_abs = max(0.05, 0.25 * med) if med > 0.0 else 0.25
        p = choose_p(remaining, eval_cost)
        ac = round_no % builder_count
        if ac == 0:
            B = fourier_basis(n, p, idx)
        elif ac == 1:
            B = wavelet_basis(n, p, rng)
        elif ac == 2:
            B = block_basis(n, p, rng)
        elif ac == 3:
            B = random_basis(n, p, rng)
        elif ac == 4:
            if p > n:
                p = max(1, min(p, n))
            sel = rng.choice(n, p, replace=False)
            B = coordinate_basis(n, sel)
            base_max_abs = max(0.10, 0.50 * med) if med > 0.0 else 0.50
        else:
            B = gradient_basis(n, p, best_a, rng)
            base_max_abs = max(0.02, 0.10 * med) if med > 0.0 else 0.10
        if B.size == 0 or B.shape[1] == 0:
            round_no += 1
            continue
        alloc = max(5.0, remaining / float(builder_count))
        res = de_search_basis(best_a, B, alloc, deadline, eval_cost, rng, base_max_abs)
        if res is not None:
            _, f, a = res
            if math.isfinite(f) and f < best_f - 1e-12:
                best_f = f
                best_a = a
        if round_no % 10 == 0:
            print("basis round %d, best=%.9f, time_left=%.1f" % (round_no, best_f, remaining))
        round_no += 1
    return best_a, best_f


def make_ramp(n):
    if n <= 1:
        return np.ones(n, dtype=np.float64)
    x = np.arange(n, dtype=np.float64) / float(n - 1)
    return 0.2 + 0.8 * x


def make_cos(n):
    idx = np.arange(n, dtype=np.float64)
    return 1.0 + 0.5 * np.cos(2.0 * math.pi * idx / float(max(1, n)))


def make_blocks(n, nb):
    p = np.ones(n, dtype=np.float64) * 0.3
    per = max(1, n // nb)
    for b in range(nb):
        st = b * per
        en = min(n, st + per)
        p[st:en] = 1.0 if b % 2 == 0 else 0.3
    return p


def make_random_smooth(n, rng):
    a = 0.5 + 0.5 * rng.random(n)
    kernel = np.ones(9, dtype=np.float64) / 9.0
    b = np.convolve(a, kernel, mode="same")
    bmin = float(np.min(b))
    bmax = float(np.max(b))
    if bmax - bmin <= 1e-12:
        return 0.5 * np.ones(n, dtype=np.float64)
    return (b - bmin) / (bmax - bmin)


def propose_candidate(seed=42, budget_s=2400.0, **kwargs):
    start = time.monotonic()
    budget = max(60.0, float(budget_s))
    deadline = start + budget - 12.0
    rng = np.random.default_rng(seed)

    parent = get_parent_array()
    if parent is not None and parent.size <= MAX_N_SEARCH:
        n = int(parent.size)
    else:
        n = DEFAULT_N
    n = int(max(128, min(MAX_N_SEARCH, n)))
    idx = np.arange(n, dtype=np.float64)
    search_base_for_patterns = parent if (parent is not None and parent.size == n) else None
    patterns = build_patterns(n, idx, search_base_for_patterns)

    final_a = None
    final_f = np.inf
    search_a = None
    search_f = np.inf
    search_eval_times = []

    candidates = []
    candidates.append(np.ones(n, dtype=np.float64))
    for v in (0.1, 10.0, 100.0, 1000.0):
        candidates.append(np.full(n, float(v), dtype=np.float64))
    if parent is not None:
        if parent.size == n or parent.size <= MAX_PARENT_EVAL_N:
            candidates.append(parent)
    candidates.append(make_ramp(n))
    candidates.append(make_cos(n))
    candidates.append(make_blocks(n, 8))
    candidates.append(make_blocks(n, 16))
    candidates.append(make_random_smooth(n, rng))

    for arr in candidates:
        if arr is None:
            continue
        if time.monotonic() >= deadline:
            break
        try:
            arr = np.asarray(arr, dtype=np.float64)
        except Exception:
            continue
        if arr.size == 0:
            continue
        arr = np.nan_to_num(arr, nan=0.0, posinf=1000.0, neginf=0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        t0 = time.monotonic()
        f = safe_eval(arr)
        dt = time.monotonic() - t0
        if arr.size == n and dt < 10.0:
            search_eval_times.append(dt)
        final_a, final_f, search_a, search_f = record_candidate(
            arr, f, final_a, final_f, search_a, search_f, n
        )

    if final_a is None or not math.isfinite(final_f):
        final_a = np.ones(n, dtype=np.float64)
        final_f = 2.0
    if search_a is None or search_a.size != n or not math.isfinite(search_f):
        search_a = np.ones(n, dtype=np.float64)
        search_f = 2.0

    if search_eval_times:
        eval_cost = float(np.median(search_eval_times))
    else:
        eval_cost = 0.01
    eval_cost = max(1e-5, min(eval_cost, 5.0))

    print("n=%d, n_parent=%s, final=%.9f, search=%.9f, eval_cost=%.5f" % (
        n,
        parent.size if parent is not None else None,
        final_f,
        search_f,
        eval_cost
    ))

    do_search = math.isfinite(final_f) and (final_f < 2.0 - 1e-8)

    if not do_search and parent is None:
        brief_end = min(deadline, time.monotonic() + max(60.0, 0.05 * budget))
        rem_brief = max(0.0, brief_end - time.monotonic())
        if rem_brief >= 20.0:
            search_a, search_f = run_line_search(
                search_a, search_f, rng, idx, n, patterns, brief_end, eval_cost, 80
            )
            if search_f < final_f:
                final_a = search_a.copy()
                final_f = search_f
            do_search = math.isfinite(final_f) and (final_f < 2.0 - 1e-8)

    if do_search:
        if search_a is None or search_a.size != n or not math.isfinite(search_f):
            search_a = np.ones(n, dtype=np.float64)
            search_f = safe_eval(search_a)
            if not math.isfinite(search_f):
                search_f = 2.0

        scale_end = min(deadline, time.monotonic() + min(90.0, max(20.0, 0.04 * budget)))
        search_a, search_f = run_scaling_search(search_a, search_f, scale_end, rng, eval_cost)
        if search_f < final_f:
            final_a = search_a.copy()
            final_f = search_f

        remaining = max(0.0, deadline - time.monotonic())
        line_end = time.monotonic() + 0.35 * remaining
        if line_end > deadline:
            line_end = deadline
        search_a, search_f = run_line_search(
            search_a, search_f, rng, idx, n, patterns, line_end, eval_cost, 500
        )
        if search_f < final_f:
            final_a = search_a.copy()
            final_f = search_f
        print("after line search: %.9f" % search_f)

        remaining = max(0.0, deadline - time.monotonic())
        if remaining > 20.0:
            search_a, search_f = run_basis_search(
                search_a, search_f, remaining, deadline, eval_cost, rng, int(search_a.size)
            )
            if search_f < final_f:
                final_a = search_a.copy()
                final_f = search_f
        print("after basis search: %.9f" % search_f)

    if final_a is None or not math.isfinite(final_f):
        final_a = np.ones(n, dtype=np.float64)
        final_f = 2.0

    final_a = np.asarray(final_a, dtype=np.float64)
    final_a = np.nan_to_num(final_a, nan=0.0, posinf=1000.0, neginf=0.0)
    final_a = np.clip(final_a, 0.0, 1000.0)

    verified = safe_eval(final_a)
    if math.isfinite(verified) and verified < final_f:
        final_f = verified
    if not math.isfinite(final_f):
        final_f = 2.0

    out = [float(x) for x in final_a.tolist()]
    print("final: score=%.9f, len=%d, time=%.1fs" % (final_f, len(out), time.monotonic() - start))
    return out
