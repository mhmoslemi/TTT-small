```python

Step function optimizer using projected subgradient descent on the simplex.
Minimizes 2*n*max(convolve(a,a))/(sum(a))^2 by working with normalized 
probability distributions and using adaptive temperature for peak targeting.
"""

import numpy as np
import time


def _convolve_fft(p):
    """Compute linear convolution p*p via FFT."""
    n = len(p)
    size = 1 << (2 * n - 1).bit_length()
    P = np.fft.rfft(p, size)
    c = np.fft.irfft(P * P, size)
    return c[:2*n-1]


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    if v.sum() <= 0:
        return np.ones(n) / n
    
    # Standard simplex projection
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u * rho > (cssv - 1)
    rho_star = np.max(rho[cond]) if cond.any() else 0
    theta = (cssv[rho_star - 1] - 1) / rho_star if rho_star > 0 else 0
    w = np.maximum(v - theta, 0)
    s = w.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return w / s  # ensure sum is exactly 1


def _compute_gradient(c, p, k_star):
    """
    Compute subgradient of s_{k*} = sum_i p[i]*p[k*-i] w.r.t. p.
    Result: g[j] = 2*p[k*-j] (with p[m]=0 if m not in [0,n-1])
    """
    n = len(p)
    g = np.zeros(n)
    for j in range(n):
        idx = k_star - j
        if 0 <= idx < n:
            g[j] = 2.0 * p[idx]
    return g
```

```python
structured perturbations, and coordinate-descent polishing.

import numpy as np

def _convolve_fft(p):
    """Compute linear convolution p*p via FFT. Returns array of length 2*n-1."""
    n = len(p)
    size = 1 << (2 * n - 1).bit_length()
    P = np.fft.rfft(p, size)
    c = np.fft.irfft(P * P, size)
    return c[:2 * n - 1]


def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if n == 0:
        return v
    v = np.asarray(v, dtype=np.float64)
    if v.sum() <= 1e-300:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1, dtype=np.float64)
    cond = u * rho > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    idx = np.max(np.nonzero(cond)[0])
    theta = (cssv[idx] - 1.0) / (idx + 1.0)
    w = np.maximum(v - theta, 0.0)
    s = w.sum()
    if s < 1e-300:
        return np.ones(n) / n
    return w / s

    p: probability vector (length n)
    Returns: gradient vector (length n)

    return 2.0 * np.correlate(w, p, mode='valid')

    best_p = p.copy()

    c = _convolve_fft(p)
    peak_init = c.max()
    range_init = c.max() - c.min()
    if range_init < 1e-12:
        range_init = peak_init

    # Hyperparameters
    T = peak_init * 0.05
    T_floor = peak_init * 1e-9
    T_decay = 0.9992
    eta = 2.0 * n * 0.005
    eta_decay = 0.9997
    eval_interval = 500
    stuck_limit = 8

    no_improve = 0
    it = 0
    max_iters = 1000000

    while it < max_iters and time.monotonic() < deadline:
        # Compute convolution
        c = _convolve_fft(p)

        # Soft peak weights (softmax over convolution values)
        cs = c / T
        cs -= cs.max()
        w = np.exp(cs)
        w_sum = w.sum()
        if w_sum < 1e-300:
            w = np.zeros(2 * n - 1)
            w[np.argmax(c)] = 1.0
        else:
            w /= w_sum

        # Compute gradient via correlate
        g = _soft_gradient(w, p)

        # Gradient descent step
        p_new = p - eta * g
        p_new = _project_simplex(p_new)

        it += 1

        # Evaluate periodically
        if it % eval_interval == 0:
            score = _eval(p_new)
            if score < best_score - 1e-9:
                best_score = score
                best_p = p_new.copy()
                no_improve = 0
                if it % (eval_interval * 20) == 0:
                    print(f"  {label} iter {it}: score={score:.6f}, T={T:.2e}")
            else:
                no_improve += 1

            # Adaptive temperature cooling
            T = max(T_floor, T * T_decay)

            # Adaptive step size
            eta *= eta_decay

            # Structured perturbation when stuck
            if no_improve >= stuck_limit:
                pert_type = int(rng.integers(0, 4))
                if pert_type == 0:
                    # Random noise
                    noise = (0.02 / n) * rng.standard_normal(n)
                    p_new = p_new + noise
                elif pert_type == 1:
                    # Sinusoidal modulation
                    freq = rng.uniform(1, max(2, n // 100))
                    t_vec = np.arange(n, dtype=np.float64)
                    mod = 1.0 + 0.1 * np.cos(2.0 * np.pi * freq * t_vec / n + rng.uniform(0, 2 * np.pi))
                    p_new = p_new * mod
                elif pert_type == 2:
                    # Anti-symmetric perturbation (break mirror symmetry)
                    t_vec = np.arange(n, dtype=np.float64)
                    center = (n - 1) / 2.0
                    mod = 1.0 + 0.05 * ((t_vec - center) / center)
                    p_new = p_new * mod
                else:
                    # Concentrate on a random sub-interval
                    lo = int(rng.integers(0, max(1, n // 2)))
                    hi = int(rng.integers(lo + max(1, n // 4), n))
                    mask = np.ones(n)
                    mask[lo:hi] = 1.1
                    mask[:lo] = 0.9
                    mask[hi:] = 0.9
                    p_new = p_new * mask

                p_new = _project_simplex(p_new)
                no_improve = 0
                eta = 2.0 * n * 0.005  # Reset step size after perturbation

        p = p_new

    # Final evaluation
    score = _eval(p)
    if score < best_score:
        best_score = score
        best_p = p.copy()

    return best_p, best_score


def _polish_coordinate(p, deadline, rng):
    """
    Coordinate-descent polish: try small transfers between pairs of coordinates.
    Focuses on reducing the maximum convolution value.
    """
    n = len(p)
    p = p.copy()
    best_score = _eval(p)
    best_p = p.copy()

    c = _convolve_fft(p)
    peak = c.max()
    k_star = int(np.argmax(c))

    n_passes = 500
    for passes in range(n_passes):
        if time.monotonic() > deadline:
            break

        # Recompute convolution and find peak
        c = _convolve_fft(p)
        peak = c.max()
        k_star = int(np.argmax(c))

        # Contribution of each position to the peak
        # c[k_star] = sum_i p[i] * p[k_star - i]
        # Gradient of c[k_star] at position j: 2 * p[k_star - j]
        contrib = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                contrib[j] = p[j] * p[idx]

        # Find top contributors (to reduce) and low contributors (to increase)
        top_k = max(5, n // 100)
        donors = np.argsort(contrib)[::-1][:top_k]
        receivers = np.argsort(contrib)[:top_k]

        improved = False
        for d in donors:
            for r in receivers:
                if d == r:
                    continue
                # Try transferring a small amount
                for frac in [0.02, 0.05, 0.1]:
                    amt = p[d] * frac
                    if amt < 1e-12:
                        continue
                    p_try = p.copy()
                    p_try[d] -= amt
                    p_try[r] += amt
                    if p_try[d] < 0:
                        continue
                    p_try = p_try / p_try.sum()
                    c_try = _convolve_fft(p_try)
                    if c_try.max() < peak - 1e-12:
                        p = p_try
                        peak = c_try.max()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if not improved:
            # Try random pairwise transfers
            for _ in range(20):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                for frac in [0.01, 0.03, 0.05]:
                    amt = p[i] * frac
                    if amt < 1e-12:
                        continue
                    p_try = p.copy()
                    p_try[i] -= amt
                    p_try[j] += amt
                    if p_try[i] < 0:
                        continue
                    p_try /= p_try.sum()
                    c_try = _convolve_fft(p_try)
                    if c_try.max() < peak - 1e-12:
                        p = p_try
                        peak = c_try.max()
                        break

        if passes % 50 == 0:
            score = _eval(p)
            if score < best_score:
                best_score = score
                best_p = p.copy()

    score = _eval(p)
    if score < best_score:
        best_score = score
        best_p = p.copy()

    return best_p, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the non-negative sequence minimizing 2*n*max(convolve(a,a))/(sum(a))^2.

    Strategy:
    1. Generate diverse starting points (given, uniform, structured, random).
    2. Optimize each with soft subgradient descent (adaptive temperature, 
       efficient FFT/correlate-based operations).
    3. Polish the best result with coordinate descent.
    4. Run a final optimization round from the polished solution.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0

    # Get initial sequence from global
    f_init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(f_init)
    print(f"Sequence length: {n}")

    # Normalize initial
    s0 = f_init.sum()
    if s0 > 1e-10:
        p_init = f_init / s0
    else:
        p_init = np.ones(n) / n

    # Generate diverse starting points
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0

    starts = {}
    starts['given'] = p_init.copy()

    # Uniform
    starts['uniform'] = np.ones(n) / n

    # U-shaped (high at edges, low at center) - reduces center convolution
    starts['u_shaped'] = 1.0 + 3.0 * ((j - center) / center) ** 2
    starts['u_shaped'] = starts['u_shaped'] / starts['u_shaped'].sum()

    # Strong U-shape
    starts['u_strong'] = 1.0 + 15.0 * ((j - center) / center) ** 2
    starts['u_strong'] = starts['u_strong'] / starts['u_strong'].sum()

    # Asymmetric: high on left, low on right
    starts['asym_l'] = 1.0 + 2.0 * np.exp(-((j - 0.2 * n) / (0.15 * n)) ** 2)
    starts['asym_l'] = starts['asym_l'] / starts['asym_l'].sum()

    # Asymmetric: high on right, low on left
    starts['asym_r'] = 1.0 + 2.0 * np.exp(-((j - 0.8 * n) / (0.15 * n)) ** 2)
    starts['asym_r'] = starts['asym_r'] / starts['asym_r'].sum()

    # Bimodal (two peaks away from center)
    starts['bimodal'] = (np.exp(-((j - 0.25 * n) / (0.08 * n)) ** 2) +
                         np.exp(-((j - 0.75 * n) / (0.08 * n)) ** 2) + 0.1)
    starts['bimodal'] = starts['bimodal'] / starts['bimodal'].sum()

    # Cosine modulated
    starts['cosine'] = 1.0 + 0.5 * np.cos(2.0 * np.pi * j / n)
    starts['cosine'] = starts['cosine'] / starts['cosine'].sum()

    # Random Dirichlet (concentrated)
    starts['dirichlet'] = rng.dirichlet(np.ones(n) * 0.1)

    # Random Dirichlet (spread)
    starts['dirichlet2'] = rng.dirichlet(np.ones(n) * 1.0)

    # Evaluate all starts
    start_scores = []
    for name, p in starts.items():
        p = np.maximum(p, 0.0)
        ps = p.sum()
        if ps < 1e-10:
            p = np.ones(n) / n
        else:
            p = p / ps
        sc = _eval(p)
        start_scores.append((sc, name, p))
        print(f"  Start '{name}': score={sc:.6f}")

    start_scores.sort(key=lambda x: x[0])

    best_score = start_scores[0][0]
    best_p = start_scores[0][2].copy()
    print(f"\nBest initial: '{start_scores[0][1]}' with score {best_score:.6f}")

    # Allocate time across top starts
    time_per_start = min(300.0, (budget_s - 120) / 4.0)

    # Optimize top 4 starts
    for idx, (sc, name, p0) in enumerate(start_scores[:4]):
        if time.monotonic() > deadline - 60:
            break
        t_alloc = time_per_start * (1.0 - 0.15 * idx)
        start_deadline = time.monotonic() + t_alloc
        print(f"\nOptimizing '{name}' (budget {t_alloc:.0f}s)...")

        p_opt, sc_opt = _optimize_soft_subgrad(p0, start_deadline, rng, label=name)

        if sc_opt < best_score:
            best_score = sc_opt
            best_p = p_opt.copy()
            print(f"  New best: {best_score:.6f}")

    # Polish phase
    print(f"\nPolishing (score={best_score:.6f})...")
    polish_deadline = min(time.monotonic() + 180.0, deadline)
    p_polished, sc_polished = _polish_coordinate(best_p, polish_deadline, rng)
    if sc_polished < best_score:
        best_score = sc_polished
        best_p = p_polished.copy()
        print(f"  Polished to: {best_score:.6f}")

    # Final optimization round from polished solution
    if time.monotonic() < deadline - 30:
        print(f"\nFinal optimization round...")
        final_deadline = min(time.monotonic() + 150.0, deadline)
        p_final, sc_final = _optimize_soft_subgrad(best_p, final_deadline, rng, label="final")
        if sc_final < best_score:
            best_score = sc_final
            best_p = p_final.copy()
            print(f"  Final: {best_score:.6f}")

        # One more polish
        if time.monotonic() < deadline:
            p_pol2, sc_pol2 = _polish_coordinate(best_p, min(time.monotonic() + 60, deadline), rng)
            if sc_pol2 < best_score:
                best_score = sc_pol2
                best_p = p_pol2.copy()
                print(f"  Final polish: {best_score:.6f}")

    # Ensure valid output
    best_p = np.clip(best_p, 0.0, 1000.0)
    final_score = _eval(best_p)
    if not np.isfinite(best_score) or final_score < best_score:
        best_score = final_score

    print(f"\n=== FINAL SCORE: {best_score:.6f} ===")
    return [float(x) for x in best_p]
```
