import numpy as np

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    mx = float(np.max(c))
    return float(2.0 * arr.size * mx / (s * s))

def _optimize_single_mode(n, mode, other_amps, deadline):
    """Optimize the amplitude of a single mode, given others fixed."""
    i = np.arange(n, dtype=np.float64)
    # Compute the base (without this mode)
    base = np.ones(n, dtype=np.float64)
    for m, A in other_amps.items():
        if m == mode or A == 0:
            continue
        theta = 2.0 * np.pi * m * (i + 0.5) / float(n)
        base += A * np.sin(theta)
    
    # The sine for this mode
    theta_m = 2.0 * np.pi * mode * (i + 0.5) / float(n)
    sin_m = np.sin(theta_m)
    
    # Non-negativity constraint: base + A * sin_m >= 0
    # For sin_m > 0: A >= -base/sin_m
    # For sin_m < 0: A <= -base/sin_m
    pos = sin_m > 1e-10
    neg = sin_m < -1e-10
    
    A_min = 0.0
    A_max = 2.0  # upper bound
    
    if np.any(pos) and np.any(base[pos] < 0):
        A_min = max(A_min, float(np.max(-base[pos] / sin_m[pos])))
    if np.any(neg) and np.any(base[neg] > 0):
        A_max = min(A_max, float(np.min(-base[neg] / sin_m[neg])))
    
    A_min = max(0.0, A_min)
    if A_max <= A_min:
        # Infeasible, return boundary
        A_max = A_min + 1e-6
    
    # 1D optimization: golden section or grid
    best_A = 0.5 * (A_min + A_max)
    best_score = np.inf
    
    # Coarse grid
    n_grid = 50
    grid = np.linspace(A_min, A_max, n_grid)
    for A in grid:
        if time.monotonic() > deadline:
            break
        a = (base + A * sin_m) / float(n)
        a = np.clip(a, 0.0, 1000.0)
        s = float(a.sum())
        if s < 0.01:
            continue
        c = _fft_convolve(a)
        sc = float(2.0 * n * np.max(c) / (s * s))
        if sc < best_score:
            best_score = sc
            best_A = float(A)
    
    # Fine grid around best
    if np.isfinite(best_score):
        step = (A_max - A_min) / n_grid
        for _ in range(4):
            lo = max(A_min, best_A - step)
            hi = min(A_max, best_A + step)
            if hi - lo < 1e-8:
                break
            grid2 = np.linspace(lo, hi, 21)

                if time.monotonic() > deadline:
