"""
Low-peak nonnegative step-function search using soft-max mirror descent.

Uses a sinusoidal multi-frequency initialization followed by gradient-based
optimization with a log-sum-exp smoothed objective and multiplicative updates.
"""

import numpy as np
import time


def _next_pow2(m):
    """Smallest power of two >= m."""
    if m <= 1:
        return 1
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a, b=None):
    """Linear convolution a * b (or a * a if b is None) using FFT."""
    a = np.asarray(a, dtype=np.float64)
    if b is None:
        b = a
    else:
        b = np.asarray(b, dtype=np.float64)
    n = a.size
    m = b.size
    if n == 0 or m == 0:
        return np.array([], dtype=np.float64)
    need = n + m - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    c = np.fft.irfft(fa * fb, N)
    return c[:need]


def _internal_score(a):
    """Fast internal evaluation matching evaluate_sequence."""
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s_val = float(arr.sum())
    if not np.isfinite(s_val) or s_val < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * arr.size * mx / (s_val * s_val))


def _make_sinusoidal(n, amp=0.78, mode=1, amp2=0.0, mode2=3, phase2=0.0):
    """Create a sinusoidal initialization with optional second mode."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    a = 1.0 + amp * np.sin(float(mode) * theta)
    if amp2 > 0:
        a += amp2 * np.sin(float(mode2) * theta + phase2)
    a = a / float(n)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        a = np.ones(n) / float(n)
    else:
        a = a / s
    return a


def _soft_max_gradient(a, beta):
    """
    Compute the soft-max smoothed gradient of max(a*a) with respect to a.
    Returns (gradient, softmax_weights, convolution).
    """
    n = a.size
    c = _fft_convolve(a)
    if c.size == 0:
        return None, None, None
    
    # Softmax weights
    c_max = np.max(c)
    exp_c = np.exp(beta * (c - c_max))
    total = np.sum(exp_c)
    if total < 1e-300:
        p = np.ones(c.size) / c.size
    else:
        p = exp_c / total
    
    # Compute gradient: (grad)_j = 2 * sum_m a[m] * p[j+m]
    # = 2 * (a * p_rev)[2n-2-j] for j=0..n-1
    p_rev = p[::-1]
    conv = _fft_convolve(a, p_rev)
    # conv has length n + (2n-1) - 1 = 3n-2
    # grad_j = 2 * conv[2n-2-j]
    # So grad = 2 * conv[n-1 : 2n-1][::-1]
    grad = 2.0 * conv[n-1:2*n-1][::-1]
    
    return grad, p, c


def _soft_max_descent(a, deadline, max_iters=2000000):
    """
    Optimize sequence using soft-max mirror descent.
    """
    n = a.size
    arr = np.asarray(a, dtype=np.float64).copy()
    s = float(arr.sum())
    arr = arr / s
    
    best = arr.copy()
    best_score = _internal_score(best)
    
    start_time = time.monotonic()
    beta = 5.0
    eta = float(n) * 0.5  # step size
    best_eta = eta
    last_improve = 0
    it = 0
    stale_count = 0
    
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        
        # Adaptive beta: increase over time
        progress = min(1.0, it / max_iters)
        beta = 5.0 + 200.0 * progress * progress
        
        # Compute gradient
        grad, p, c = _soft_max_gradient(arr, beta)
        if grad is None:
            break
        
        # Project to tangent space of simplex
        g = grad - np.mean(grad)
        g_norm = np.max(np.abs(g))
        if g_norm < 1e-15:
            break
        
        # Normalize gradient for stable step
        g = g / g_norm
        
        # Mirror descent update
        arr_new = arr * np.exp(-eta * g)
        arr_new = np.where(np.isfinite(arr_new), arr_new, 0.0)
        arr_new = np.clip(arr_new, 0.0, None)
        s_new = float(arr_new.sum())
        if s_new < 1e-10:
            # Fallback: add small uniform
            arr_new = arr + 0.01 * np.ones(n)
            arr_new = arr_new / arr_new.sum()
            arr = arr_new
            eta *= 0.5
            continue
        arr_new = arr_new / s_new
        
        # Evaluate
        sc = _internal_score(arr_new)
        
        if np.isfinite(sc) and sc < best_score:
            best = arr_new.copy()
            best_score = sc
            arr = arr_new
            last_improve = it
            stale_count = 0

def _top_k_refine(a, deadline, max_iters=100000):
    """
    Refine using top-K active set gradient (hard max approach).
    """
    n = a.size
    arr = np.asarray(a, dtype=np.float64).copy()
    s = float(arr.sum())
    arr = arr / s
    
    best = arr.copy()
    best_score = _internal_score(best)
    
    start_time = time.monotonic()
    eta = float(n) * 0.3
    K = 16 if n <= 4096 else 32
    last_improve = 0
    it = 0
    stale = 0
    
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        
        c = _fft_convolve(arr)
        if c.size == 0:
            break
        
        # Get top-K lags
        if c.size <= K:
            active = np.arange(c.size)
        else:
            active = np.argpartition(c, -K)[-K:]
        
        # Compute gradient from active set
        grad = np.zeros(n, dtype=np.float64)
        for k in active:
            k = int(k)
            # grad[i] += 2 * a[k-i] for valid i
            i_min = max(0, k - (n - 1))
            i_max = min(n - 1, k)
            if i_max >= i_min:
                idx = np.arange(i_min, i_max + 1)
                grad[idx] += 2.0 * arr[k - idx]
        
        grad /= float(len(active))
        
        # Project
        g = grad - np.mean(grad)
        g_norm = np.max(np.abs(g))
        if g_norm < 1e-15:
            break
        g = g / g_norm
        
        # Update
        arr_new = arr * np.exp(-eta * g)
        arr_new = np.where(np.isfinite(arr_new), arr_new, 0.0)
        arr_new = np.clip(arr_new, 0.0, None)
        s_new = float(arr_new.sum())
        if s_new < 1e-10:
            arr_new = best.copy()
            eta *= 0.5
            continue
        arr_new = arr_new / s_new
        
        sc = _internal_score(arr_new)
        if np.isfinite(sc) and sc < best_score:
            best = arr_new.copy()
            best_score = sc
            arr = arr_new
            last_improve = it
            stale = 0
            eta = min(eta * 1.01, float(n) * 1.5)
        else:
            stale += 1
            eta *= 0.75
            if stale > 300:
                arr = best.copy()
                noise = 0.005 * np.random.randn(n)
                arr = np.clip(arr + arr * noise, 0, None)
                s = arr.sum()
                if s > 1e-10:
                    arr = arr / s
                else:
                    arr = best.copy()
                eta = eta / 0.75  # restore
                eta *= 0.5
                stale = 0
        
        if it % 5000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"  [refine] it={it} eta={eta:.1f} score={best_score:.8f} time={elapsed:.1f}s")
    
    return best, best_score


def _try_initialization(n, deadline_check):
    """Try a sinusoidal initialization and score it."""
    candidates = []
    
    # Single mode
    for amp in [0.6, 0.7, 0.78, 0.85, 0.9, 1.0, 1.1]:

        sc = _internal_score(a)
        candidates.append((sc, a))

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """

    4. Validate with evaluate_sequence.
    """

    t0 = time.monotonic()

    print(f"[phase1] Finding best initialization...")
    
    best_init = None

    phase2_deadline = t0 + 1800.0
    print(f"[phase2] Running soft-max mirror descent...")

    print(f"[phase2] Descent score={desc_score:.8f} time={time.monotonic()-t0:.1f}s")
    
    # Phase 3: Top-K refinement (450s)
    phase3_deadline = t0 + total_budget - 30.0
    print(f"[phase3] Running top-K refinement...")

    # Also try descent from refine (in case refine is better starting point)
    if refine_score < desc_score:

    # Sort by score
    candidates.sort(key=lambda x: x[2])
    
    # Validate top candidates

        arr_clean = np.clip(arr_clean, 0.0, 1000.0)
        s = float(arr_clean.sum())
        if s < 0.01:
            continue
        arr_clean = arr_clean / s
        
        try:
            real_score = float(evaluate_sequence(arr_clean.tolist()))
        except Exception:
            real_score = np.inf
        
        if np.isfinite(real_score) and real_score < best_actual:
            best_actual = real_score
            best_out = arr_clean
            print(f"[validate] {name}: n={arr_clean.size} actual_score={real_score:.8f}")
            if best_actual < 1.45:
                break
    
    if best_out is None:
        best_out = np.ones(256) / 256.0
        print("[validate] Using fallback constant sequence")
    
    print(f"[final] Best score: {best_actual:.8f} total_time={time.monotonic()-t0:.1f}s")
    return [float(x) for x in best_out]
