import numpy as np
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))

def _gradient_refine(a, budget_s=60.0, K=8, lr_init=0.05):
    """Gradient-based refinement on the simplex."""
    start = time.monotonic()
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if s < 0.01:
        return arr, np.inf
    arr = arr / s
    
    n = arr.size
    best = arr.copy()
    best_score = _internal_score(best)
    
    lr = lr_init * n  # scale learning rate with n
    max_lr = 0.5 * n
    
    while time.monotonic() - start < budget_s:
        c = _fft_convolve(arr)
        if c.size == 0:
            break
        maxc = float(np.max(c))
        if not np.isfinite(maxc):
            break
        
        # Find top-K active lags
        if c.size <= K:
            active = np.arange(c.size)
        else:
            active = np.argpartition(c, -K)[-K:]
        
        # Compute gradient: average of db_k/da_i over active lags
        grad = np.zeros(n, dtype=np.float64)
        for k in active:
            k = int(k)
            # db_k/da_i = a_{i-k} + a_{i+k} (where valid)
            # For lag k: b_k = sum_i a_i * a_{i+k} (for i in [0, n-1-k])
            #             + sum_i a_i * a_{i+k} ... 
            # Actually, b_k = sum_{i=0}^{n-1-k} a_i * a_{i+k}
            # db_k/da_i = a_{i+k} (if i+k < n) + a_{i-k} (if i-k >= 0)
            
            for i in range(n):
                val = 0.0
                if i + k < n:
                    val += arr[i + k]
                if i - k >= 0:
                    val += arr[i - k]
                grad[i] += val
        
        grad /= float(len(active))
        
        # Project to simplex tangent: remove mean
        g = grad - np.mean(grad)
        
        if np.max(np.abs(g)) < 1e-15:
            break
        
        # Multiplicative update
        exponent = -lr * g
        exponent = np.clip(exponent, -2.0, 2.0)
        arr_new = arr * np.exp(exponent)
        arr_new = np.clip(arr_new, 0.0, 1000.0)
        s_new = arr_new.sum()
        if s_new < 0.01 or not np.isfinite(s_new):
            arr = best.copy()
            lr *= 0.5
            continue
        arr_new = arr_new / s_new
        
        sc = _internal_score(arr_new)
        if np.isfinite(sc) and sc < best_score:
            best = arr_new.copy()
            best_score = sc
            arr = arr_new
            lr = min(max_lr, lr * 1.02)
        else:
            arr = best.copy()
            lr = max(0.01 * n, lr * 0.7)
    
    return best, best_score

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))

    s = float(a.sum())

def _gradient_refine(a, budget_s=60.0):
    """FFT-accelerated gradient refinement on the simplex."""
    start = time.monotonic()
    arr = np.asarray(a, dtype=np.float64).copy()
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        return arr, np.inf
    arr = arr / s
    
    n = arr.size
    best = arr.copy()
    best_score = _internal_score(best)
    if not np.isfinite(best_score):
        return best, best_score
    
    K = min(16, max(4, n // 256))
    lr = 0.1 * n
    max_lr = 1.0 * n
    min_lr = 0.01 * n
    
    it = 0
    max_iters = 10000
    last_improve = 0
    
    while time.monotonic() - start < budget_s and it < max_iters:
        it += 1
        
        c = _fft_convolve(arr)
        if c.size == 0:
            break
        maxc = float(np.max(c))
        if not np.isfinite(maxc):
            break

        if c.size <= K:
            active = np.arange(c.size, dtype=np.int64)
        else:
            active = np.argpartition(c, -K)[-K:]
        
        # Compute gradient (vectorized)
        grad = np.zeros(n, dtype=np.float64)
        for k in active:
            k = int(k)
            if k == 0:
                grad += 2.0 * arr
            else:
                grad[:n - k] += arr[k:]
                grad[k:] += arr[:n - k]

        # Project to simplex tangent
        g = grad - np.mean(grad)
        g_norm = np.max(np.abs(g))
        if g_norm < 1e-15:

        step = lr * g
        step = np.clip(step, -1.5, 1.5)
        arr_new = arr * np.exp(-step)
        arr_new = np.clip(arr_new, 0.0, 1000.0)
        s_new = float(arr_new.sum())
        if s_new < 0.01 or not np.isfinite(s_new):
            arr = best.copy()
            lr = max(min_lr, lr * 0.5)

            best_score = sc
            arr = arr_new
            last_improve = it

        else:
            arr = best.copy()
            lr = max(min_lr, lr * 0.7)

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))

        modes = list(range(1, len(amps) + 1))
    n = int(n)

    a = a / float(n)
    s = a.sum()
    if s > 0.01:
        a = a / s
    else:
        a = np.ones(n) / float(n)
    return a

"""
Multi-harmonic anti-palindromic step-function search with soft-max mirror descent.

Strategy:
1. Initialize with multi-harmonic sine sequences (anti-palindromic by construction)
2. Refine using mirror descent with soft-max gradient and anti-palindromic projection
3. Try multiple sequence lengths and take the best result
"""

import numpy as np
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))

def _make_multi_sine(n, amps, modes=None):
    if modes is None:
        modes = list(range(1, len(amps) + 1))
    n = int(n)
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for amp, mode in zip(amps, modes):
        a += float(amp) * np.sin(float(mode) * 2.0 * np.pi * (i + 0.5) / float(n))
    a = np.clip(a, 0.0, 1000.0)
    a = a / float(n)
    s = a.sum()
    if s > 0.01:
        a /= s
    else:
        a = np.ones(n) / float(n)
    return a

def _anti_pal_project(a):
    """Project onto anti-palindromic subspace (preserving sum)."""
    n = a.size
    result = a.copy()
    c = 2.0 * a.sum() / n
    half = n // 2
    for i in range(half):

def _soft_max_grad(a, bu, tau, K):

    n = a.size

Multi-harmonic anti-palindromic step search with soft-max mirror descent refinement.

Approach:
- Parameterize sequences as multi-harmonic anti-palindromic sines
  (a_i + a_{n-1-i} = const, ensuring favorable auto-correlation structure)
- Optimize harmonic amplitudes via grid search + local search
- Refine with mirror descent using soft-max gradient and anti-palindromic projection
- Try multiple lengths (2048, 4096, 8192) and return the best

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))


def _make_multi_sine(n, amps, modes=None):
    if modes is None:
        modes = list(range(1, len(amps) + 1))
    n = int(n)
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for amp, mode in zip(amps, modes):
        a += float(amp) * np.sin(float(mode) * 2.0 * np.pi * (i + 0.5) / float(n))
    a = np.clip(a, 0.0, 1000.0)
    a = a / float(n)
    s = float(a.sum())
    if s > 0.01:
        a = a / s
    else:
        a = np.ones(n, dtype=np.float64) / float(n)
    return a


def _anti_pal_project(a):
    n = a.size
    c = 2.0 * a.sum() / n
    half = n // 2
    first = a[:half]
    if n % 2 == 0:
        last = a[half:][::-1]
    else:
        last = a[half+1:][::-1]
    d = first - last
    result = np.empty(n, dtype=np.float64)

    result[n - half:] = (c / 2.0 - d / 2.0)[::-1]
    if n % 2 == 1:
        result[half] = c / 2.0
    result = np.clip(result, 0.0, 1000.0)
    s = result.sum()
    if s > 0.01:
        result = result / s
    else:
        result = np.ones(n, dtype=np.float64) / float(n)
    return result

def _compute_gradient(a, bu, tau, max_active=50):
    """Compute soft-max weighted gradient of max(bu) w.r.t. a."""
    n = a.size
    max_bu = float(np.max(bu))
    
    # Select active lags (those within 5*tau of max)
    threshold = max_bu - 5.0 * tau
    active_mask = bu >= threshold
    active = np.where(active_mask)[0]
    
    if len(active) == 0:
        active = np.array([int(np.argmax(bu))])
    if len(active) > max_active:
        # Keep only the top max_active
        top_idx = np.argpartition(bu, -max_active)[-max_active:]
        active = top_idx
    
    # Softmax weights
    exponents = (bu[active] - max_bu) / tau
    exponents -= np.max(exponents)  # numerical stability
    weights = np.exp(exponents)
    weights /= weights.sum()
    
    # Compute gradient
    g = np.zeros(n, dtype=np.float64)
    for k, w in zip(active, weights):
        k = int(k)
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_max >= j_min:
            idx = np.arange(j_min, j_max + 1)
            g[idx] += 2.0 * w * a[k - idx]
    
    return g
