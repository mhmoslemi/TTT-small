import numpy as np
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return float(2.0 * arr.size * mx / (s * s))

def _make_sine_init(n, amps=None):
    """Create anti-palindromic sinusoidal initialization."""
    if amps is None:
        amps = [0.8]  # single harmonic
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for m, A in enumerate(amps, start=1):
        a += A * np.sin(2.0 * np.pi * m * (i + 0.5) / n)
    a /= n
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
    else:
        a /= s
    return a

def _anti_palindromic_project(a, n):
    """Project a to the anti-palindromic subspace."""
    if n % 2 != 0:
        return a  # skip for odd n
    a = a.copy()
    # For each pair (i, n-1-i), enforce a[i] + a[n-1-i] = 2/n
    # While preserving the difference as much as possible
    first_half = a[:n//2].copy()
    second_half = a[n//2:].copy()[::-1]  # reversed second half
    total = first_half + second_half
    diff = first_half - second_half
    # New values: a_i = 1/n + diff/2, a_{n-1-i} = 1/n - diff/2
    new_first = 1.0/n + diff/2
    new_second = 1.0/n - diff/2
    # Clip to non-negative
    new_first = np.maximum(new_first, 0.0)
    new_second = np.maximum(new_second, 0.0)
    a[:n//2] = new_first
    a[n//2:] = new_second[::-1]
    return a

    a = np.asarray(a, dtype=np.float64).copy()

    s = a.sum()
    if s < 0.01:
        a = np.ones_like(a) / a.size
        s = 1.0
    a = a / s
    n = a.size
    
    best_a = a.copy()
    best_score = _internal_score(a)
    if verbose:
        print(f"  [init] n={n} score={best_score:.8f}")

    tau_list = [n * 0.5, n, 2*n, 5*n, 10*n, 20*n, 50*n, 100*n]

    base_eta = 2.0 * n
    
    it_global = 0
    for tau in tau_list:
        if time.monotonic() > deadline:
            break
        if verbose:
            print(f"  [tau={tau:.0f}] starting, current best={best_score:.8f}")

    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return float(2.0 * arr.size * mx / (s * s))


def _make_sine_init(n, harmonics):
    """
    Create anti-palindromic sinusoidal initialization.
    harmonics: list of (mode, amplitude) tuples.
    """
    i = np.arange(n, dtype=np.float64)
    a = np.zeros(n, dtype=np.float64)
    for mode, amp in harmonics:
        a += amp * np.sin(2.0 * np.pi * float(mode) * (i + 0.5) / float(n))
    a = 1.0 + a
    a /= float(n)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s < 0.01:
        a = np.ones(n) / n
    else:
        a /= s
    return a


def _topk_gradient(c, a, K, n):
    """
    Compute gradient using top-K active lags.
    c: convolution values (length 2n-1)
    a: sequence (length n)
    K: number of top lags to use
    Returns: gradient of length n
    """
    if K >= c.size:
        idx = np.arange(c.size)
    else:
        idx = np.argpartition(c, -K)[-K:]
    
    # For each active lag k, the gradient contribution to a_i is a_{k-i}
    # grad_i = (1/K) * sum_{k in active} a_{k-i}
    # = (1/K) * cross_correlation(indicator_active, a)
    
    # Build indicator vector
    w = np.zeros(c.size, dtype=np.float64)
    w[idx] = 1.0 / K
    
    # Cross-correlation: grad_i = sum_k w_k * a_{k-i}
    # = np.correlate(w, a, 'valid')
    grad = np.correlate(w, a, 'valid')
    return grad


def _softmax_gradient(c, a, tau, n):
    """
    Compute gradient using softmax weights (log-sum-exp gradient).
    """
    tau_c = tau * c
    max_tc = np.max(tau_c)
    exp_tc = np.exp(tau_c - max_tc)
    sum_exp = exp_tc.sum()
    if sum_exp < 1e-300:
        w = np.ones(c.size) / c.size
    else:
        w = exp_tc / sum_exp
    
    grad = np.correlate(w, a, 'valid')
    return grad


def _optimize_sequence(a, deadline, method='softmax', K=100, verbose=True):
    """
    Optimize a sequence to minimize the max convolution score.
    
    method: 'softmax' or 'topk'
    K: number of active lags for topk method
    """
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        a = np.ones(a.size) / a.size
        s = 1.0
    a = a / s
    n = a.size
    
    best_a = a.copy()
    best_score = _internal_score(a)
    
    if verbose:
        print(f"    [opt-init] n={n} score={best_score:.8f}")
    
    # Adaptive learning rate
    eta = float(n)
    
    # For softmax method: tau schedule
    if method == 'softmax':
        tau_schedule = [0.5 * n, n, 2 * n, 5 * n, 10 * n, 20 * n, 50 * n]
    else:
        tau_schedule = [K]  # just K for topk
    
    overall_no_improve = 0
    max_overall_no_improve = 300
    
    for phase, tau in enumerate(tau_schedule):
        if time.monotonic() > deadline:
            break
        if overall_no_improve > max_overall_no_improve:
            break
        
        if verbose:
            label = f"tau={tau:.0f}" if method == 'softmax' else f"K={K}"
            print(f"    [{label}] start, best={best_score:.8f}, time_used={time.monotonic()- (deadline - (2400-30)):.1f}s")
        
        phase_no_improve = 0
        max_phase_no_improve = 200
        
        for it in range(500):
            if time.monotonic() > deadline:
                break
            if phase_no_improve > max_phase_no_improve:
                break
            
            c = _fft_convolve(a)
            
            if method == 'softmax':
                grad = _softmax_gradient(c, a, tau, n)
            else:
                grad = _topk_gradient(c, a, K, n)
            
            # Project to tangent space
            grad = grad - np.mean(grad)
            
            # Check gradient magnitude
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            
            # Adaptive step size: target step magnitude ~ 0.1 in relative terms
            # a is O(1/n), gradient is O(1/n), so eta * grad / a ~ eta * 1/n
            # We want this to be ~ 0.1 to 1
            target_rel_step = 0.3
            step_scale = target_rel_step * n / max(grad_norm, 1e-15)
            # But cap it
            step_scale = min(step_scale, 10.0 * n)
            step_scale = max(step_scale, 0.01 * n)
            
            # Multiplicative update
            delta = step_scale * grad
            delta = np.clip(delta, -3.0, 3.0)
            a_new = a * np.exp(-delta)
            a_new = np.maximum(a_new, 1e-12)
            
            # Renormalize
            s_new = a_new.sum()
            if s_new < 0.01:
                a_new = best_a.copy()
                phase_no_improve += 5
                continue
            a_new = a_new / s_new
            
            # Evaluate
            sc = _internal_score(a_new)
            
            if np.isfinite(sc) and sc < best_score:
                best_a = a_new.copy()
                best_score = sc
                a = a_new
                phase_no_improve = 0
                overall_no_improve = 0
                # Slightly increase step
                target_rel_step = min(target_rel_step * 1.01, 1.0)
            else:
                phase_no_improve += 1
                overall_no_improve += 1
                # Decrease step
                target_rel_step = max(target_rel_step * 0.9, 0.01)
        
        if verbose:
            print(f"    [{tau if method=='softmax' else K}] done, best={best_score:.8f}")
    
    return best_a, best_score


def _try_anti_palindromic(a, n):
    """Try projecting to anti-palindromic and see if it improves."""
    if n % 2 != 0:
        return a, _internal_score(a)
    
    a_ap = np.zeros(n, dtype=np.float64)
    first = a[:n//2]
    second = a[n//2:][::-1]
    diff = first - second
    a_ap[:n//2] = 1.0/n + diff/2
    a_ap[n//2:] = (1.0/n - diff/2)[::-1]
    a_ap = np.maximum(a_ap, 0.0)
    s = a_ap.sum()
    if s < 0.01:
        return a, _internal_score(a)
    a_ap = a_ap / s
    
    sc = _internal_score(a_ap)
    sc_orig = _internal_score(a)
    if sc < sc_orig:
        return a_ap, sc
    return a, sc_orig

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())

    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    return float(2.0 * arr.size * mx / (s * s))
