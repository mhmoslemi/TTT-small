```python
def _lp_step(f, n, deadline):
    """Solve LP: max sum(g) s.t. conv(f,g)(k) <= T for all k, g >= 0."""
    T = _autocorr_fft(f).max()
    
    # Build constraint matrix A where A[k,j] = f[k-j] if 0 <= k-j < n
    # This is a (2n-1) x n matrix
    # A[k,j] = f[k-j] if 0 <= k-j < n, else 0
    
    # For efficiency, build using the banded structure
    # A is actually a "Hankel-like" matrix
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        j_min = max(0, k - n + 1)
        j_max = min(n - 1, k)
        if j_min <= j_max:
            A[k, j_min:j_max+1] = f[k - np.arange(j_min, j_max+1)]
    
    # LP: maximize sum(g) = -min(-sum(g))
    # subject to A @ g <= T, g >= 0
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=A, b_ub=np.full(2*n-1, T), bounds=[(0, None)]*n, method='highs')
    
    if res.status != 0:
        return None
    
    g = res.x
    g = np.maximum(g, 0)
    s = g.sum()
    if s < 1e-10:
        return None
    
    # Blend: f_new = (1-t)*f + t*g/s
    t = 0.3
    f_new = (1 - t) * f + t * g / s
    f_new = np.maximum(f_new, 0)
    f_new /= f_new.sum()
    return f_new
```

def _conv_fft(f):
    """Full linear self-convolution via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _score(f, c=None):
    """Compute score. If c not provided, compute it."""
    n = len(f)
    if c is None:
        c = _conv_fft(f)
    s = f.sum()
    if s < 0.01:
        return np.inf, c
    return 2.0 * n * c.max() / (s * s), c


def _subgrad_single(f, p, n):
    """Subgradient contribution from a single convolution position p."""
    g = np.zeros(n)
    j_min = max(0, p - n + 1)
    j_max = min(n - 1, p)
    if j_min > j_max:
        return g
    j_range = np.arange(j_min, j_max + 1)
    idx = p - j_range
    g[j_range] = 2.0 * f[idx]
    # Fix the diagonal: if p = 2*j for some j in range
    if p % 2 == 0:
        j_mid = p // 2
        if j_min <= j_mid <= j_max:
            offset = j_mid - j_min
            g[j_mid] = f[j_mid]  # halve it (was 2*f, should be f)
    return g


def _subgrad(f, c, n, frac=0.05):
    """Compute subgradient using active set (positions within frac of max)."""
    max_c = c.max()
    if max_c < 1e-15:
        return np.zeros(n)
    threshold = max_c * (1.0 - frac)
    active = np.where(c >= threshold)[0]
    
    g = np.zeros(n)
    for p in active:
        g += _subgrad_single(f, int(p), n)
    return g / len(active)


def _mirror_step(f, g, eta):
    """Mirror descent step on the simplex."""
    exponents = -eta * g
    # Numerical stability
    exponents -= exponents.max()
    f_new = f * np.exp(exponents)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy()
    f_new /= s
    return f_new


def _generate_starts(f_base, n, rng):
    """Generate diverse starting points."""
    j = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    starts = []
    
    starts.append(('base', f_base.copy()))
    starts.append(('uniform', np.ones(n) / n))
    
    # U-shaped
    for a in [1.0, 3.0, 10.0]:
        u = 1.0 + a * ((j - center) / center) ** 2
        u /= u.sum()
        starts.append((f'u_{a}', u))
    
    # Sine-based
    for freq in [2, 4, 8]:
        s = 1.0 + 0.5 * np.sin(freq * 2 * np.pi * j / n)
        s = np.maximum(s, 0.01)
        s /= s.sum()
        starts.append((f'sine_{freq}', s))
    
    # Random
    for trial in range(3):
        r = rng.exponential(1.0, size=n)
        r /= r.sum()
        starts.append((f'rand_{trial}', r))
    
    # Perturbed base
    for trial in range(3):
        p = f_base.copy()
        p += 0.1 * f_base * rng.standard_normal(n)
        p = np.maximum(p, 0)
        p /= p.sum()
        starts.append((f'pert_{trial}', p))
    
    return starts


def _optimize(f, n, rng, deadline, label=""):
    """Main optimization: mirror descent with adaptive step size."""
    f = np.maximum(f, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n) / n
    else:
        f /= s
    
    c = _conv_fft(f)
    current_score = 2.0 * n * c.max() / (f.sum() ** 2)
    best_f = f.copy()
    best_score = current_score
    
    eta = 1.0
    eta_min = 1e-5
    eta_max = 50.0
    frac = 0.05  # Active set fraction
    
    # Anneal frac: start wide, narrow down
    frac_start = 0.2
    frac_end = 0.01
    
    stuck_count = 0
    max_stuck = 10
    
    it = 0
    max_iters = 100000
    
    while time.monotonic() < deadline and it < max_iters:
        it += 1
        progress = min(1.0, it / max_iters)
        frac = frac_start + (frac_end - frac_start) * progress
        
        c = _conv_fft(f)
        max_c = c.max()
        
        g = _subgrad(f, c, n, frac)
        
        # Skip if gradient is too small
        if g.max() < 1e-15:
            break
        
        f_new = _mirror_step(f, g, eta)
        
        c_new = _conv_fft(f_new)
        score_new = 2.0 * n * c_new.max() / (f_new.sum() ** 2)
        
        if score_new < current_score - 1e-12:
            f = f_new
            current_score = score_new
            if score_new < best_score:
                best_score = score_new
                best_f = f.copy()
                stuck_count = 0
            eta = min(eta * 1.02, eta_max)
        else:
            stuck_count += 1
            eta = max(eta * 0.8, eta_min)
            
            if stuck_count >= max_stuck:
                # Perturbation
                magnitude = 0.05 / np.sqrt(n)
                f = f + magnitude * rng.standard_normal(n)
                f = np.maximum(f, 0)
                f /= f.sum()
                stuck_count = 0
                eta = min(eta * 3.0, eta_max)
                c = _conv_fft(f)
                current_score = 2.0 * n * c.max() / (f.sum() ** 2)
    
    # Final check
    c = _conv_fft(f)
    score = 2.0 * n * c.max() / (f.sum() ** 2)
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score


def _polish(f, n, rng, deadline):
    """Coordinate descent polish targeting the peak convolution position."""
    f = np.maximum(f, 0.0).copy()
    f /= f.sum()
    
    c = _conv_fft(f)
    peak_score = 2.0 * n * c.max() / (f.sum() ** 2)
    best_f = f.copy()
    best_score = peak_score
    
    max_passes = int((deadline - time.monotonic()) * 10)  # ~10 passes per second target
    max_passes = max(max_passes, 50)
    max_passes = min(max_passes, 5000)
    
    for p in range(max_passes):
        if time.monotonic() > deadline:
            break
        
        c = _conv_fft(f)
        peak_pos = int(np.argmax(c))
        peak_val = c[peak_pos]
        
        # Find the pair with maximum contribution to the peak
        j_min = max(0, peak_pos - n + 1)
        j_max = min(n - 1, peak_pos)
        
        if j_min > j_max:
            break
        
        contributions = np.zeros(n)
        j_range = np.arange(j_min, j_max + 1)
        idx = peak_pos - j_range
        contributions[j_range] = f[j_range] * f[idx]
        
        # Find best donor (highest contribution)
        j_donate = int(np.argmax(contributions))
        
        # Find best receiver (minimal contribution to peak, and to other peaks)
        # Use a heuristic: positions with low f value and low contribution
        min_contrib = contributions.min()
        # Prefer positions far from the peak interaction
        # Simply pick the position with minimum f that has zero contribution
        zero_contrib = np.where(contributions < 1e-15)[0]
        if len(zero_contrib) > 0:
            # Among zero-contribution positions, pick the one with smallest f
            candidates = zero_contrib
            j_receive = candidates[int(np.argmin(f[candidates]))]
        else:
            # Pick the position with minimum contribution that's not the donor
            contrib_sorted = np.argsort(contributions)
            for candidate in contrib_sorted:
                if candidate != j_donate:
                    j_receive = int(candidate)
                    break
            else:
                j_receive = (j_donate + 1) % n
        
        if j_donate == j_receive:
            continue
        
        # Try transferring mass
        for frac_transfer in [0.5, 0.3, 0.1, 0.05]:
            amt = f[j_donate] * frac_transfer
            f_try = f.copy()
            f_try[j_donate] -= amt
            f_try[j_receive] += amt
            f_try /= f_try.sum()
            
            c_try = _conv_fft(f_try)
            if c_try.max() < peak_val - 1e-14:
                f = f_try
                peak_score = 2.0 * n * c_try.max() / (f.sum() ** 2)
                if peak_score < best_score:
                    best_score = peak_score
                    best_f = f.copy()
                break
        
        # Also try some random moves
        for _ in range(5):
            i = int(rng.integers(0, n))
            jx = int(rng.integers(0, n))
            if i == jx:
                continue
            f_try = f.copy()
            amt = f_try[i] * 0.1
            f_try[i] -= amt
            f_try[jx] += amt
            f_try /= f_try.sum()
            c_try = _conv_fft(f_try)
            if c_try.max() < c.max() - 1e-14:
                f = f_try
    
    return best_f, best_score

def _conv_fft(f):
    """Full linear self-convolution via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _score_from_conv(c, n, s):
    """Compute score from convolution, n, and sum."""
    if s < 0.01:
        return np.inf
    return 2.0 * n * c.max() / (s * s)


def _subgrad_from_position(f, p, n):
    """Subgradient contribution from a single convolution position p.
    Returns vector g where g[j] = 2*f[p-j] for valid j, 0 otherwise.
    """
    g = np.zeros(n)
    j_min = max(0, p - n + 1)
    j_max = min(n - 1, p)
    if j_min > j_max:
        return g
    j_range = np.arange(j_min, j_max + 1)
    idx = p - j_range
    g[j_range] = 2.0 * f[idx]
    return g


def _subgrad_active(f, c, n, frac):
    """Compute averaged subgradient over active set (positions within frac of max)."""
    max_c = c.max()
    if max_c < 1e-15:
        return np.zeros(n)
    threshold = max_c * (1.0 - frac)
    active = np.where(c >= threshold)[0]
    
    g = np.zeros(n)
    for p in active:
        g += _subgrad_from_position(f, int(p), n)
    g /= len(active)
    return g
