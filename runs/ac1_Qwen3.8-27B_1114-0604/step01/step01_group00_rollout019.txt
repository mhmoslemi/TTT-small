    f_new /= s

    starts['sine_abs'] = 1.0 + 0.7 * np.abs(np.sin(np.pi * j / n))
    starts['cos2'] = 1.0 + 0.4 * np.cos(4.0 * np.pi * j / n)
    
    starts['rand_exp'] = rng.exponential(1.0, size=n)
    starts['rand_beta'] = rng.beta(0.5, 0.5, size=n)
    
    # Spectrally-designed start: use inverse DFT of a shaped spectrum
    size = 1 << (2 * n - 1).bit_length()
    G_shape = np.ones(size // 2 + 1, dtype=np.complex128)
    k = np.arange(len(G_shape), dtype=np.float64)
    G_shape *= np.exp(-(k / (n * 0.3)) ** 2)  # Gaussian spectrum
    f_spec = np.fft.irfft(G_shape * n, size)[:n]
    f_spec = np.maximum(f_spec, 0.0)
    if f_spec.sum() > 0:
        starts['spectral_gauss'] = f_spec / f_spec.sum()
    
    # Evaluate all starts
    start_results = []
    for name, f0 in starts.items():
        f = np.maximum(f0, 0.0)
        s = f.sum()
        if s < 1e-10:
            f = np.ones(n)
        else:
            f = f / s
        sc = _eval_local(f)
        start_results.append((sc, name, f.copy()))
        print(f"  Start '{name}': {sc:.6f}")
    
    start_results.sort(key=lambda x: x[0])
    
    if start_results[0][0] < best_score:
        best_score = start_results[0][0]
        best_f = start_results[0][2].copy()
        print(f"Best start: {start_results[0][1]} with score {best_score:.6f}")

    # --- Phase 1: Mirror descent from top starting points ---
    n_starts_to_try = min(4, len(start_results))
    time_per_start = (budget_s * 0.35) / n_starts_to_try
    
    for idx in range(n_starts_to_try):
        if time.monotonic() > deadline - 120:
            break
        
        sc, name, f0 = start_results[idx]
        t_alloc = time_per_start * (1.2 - 0.15 * idx)
        start_deadline = time.monotonic() + t_alloc
        
        print(f"\n[Phase 1] Optimizing from '{name}' (budget {t_alloc:.0f}s, current best {best_score:.6f})")
        
        f_opt, sc_opt = _optimize_mirror(f0, n, start_deadline, rng, label=name)
        
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  New best from '{name}': {best_score:.6f}")

    # --- Phase 2: Spectral equalization on best ---
    print(f"\n[Phase 2] Spectral equalization (current best {best_score:.6f})")
    f_spec = _spectral_flatten(best_f, n, n_passes=3)
    sc_spec = _eval_local(f_spec)
    if sc_spec < best_score:
        best_score = sc_spec
        best_f = f_spec.copy()
        print(f"  Spectral improved to: {best_score:.6f}")
    else:
        print(f"  Spectral gave {sc_spec:.6f} (no improvement)")

    # --- Phase 3: More mirror descent from best ---
    remaining = max(60, deadline - time.monotonic())
    print(f"\n[Phase 3] Further optimization (budget {remaining:.0f}s, current best {best_score:.6f})")
    
    # Multiple rounds with different parameters
    n_rounds = 3
    for r in range(n_rounds):
        if time.monotonic() > deadline - 60:
            break
        round_deadline = time.monotonic() + remaining / (n_rounds - r) * 0.9
        f_opt, sc_opt = _optimize_mirror(best_f, n, round_deadline, rng, label=f"round{r}")
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  Round {r}: {best_score:.6f}")
        
        # Try spectral equalization between rounds
        if r < n_rounds - 1:
            f_eq = _spectral_equalize(best_f, n, strength=0.3)
            sc_eq = _eval_local(f_eq)
            if sc_eq < best_score:
                best_score = sc_eq
                best_f = f_eq.copy()
                print(f"  Spectral after round {r}: {best_score:.6f}")

    # --- Phase 4: Peak-targeted reduction ---
    if time.monotonic() < deadline - 30:
        print(f"\n[Phase 4] Peak reduction (current best {best_score:.6f})")
        peak_deadline = min(time.monotonic() + 60, deadline - 20)
        f_opt, sc_opt = _multi_peak_reduction(best_f, n, peak_deadline, rng)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  Peak reduction: {best_score:.6f}")

    # --- Phase 5: Coordinate descent polish ---
    if time.monotonic() < deadline - 10:
        print(f"\n[Phase 5] Coordinate descent polish (current best {best_score:.6f})")
        cd_deadline = deadline - 5
        f_opt, sc_opt = _coordinate_descent(best_f, n, cd_deadline, rng)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  Polished to: {best_score:.6f}")

    # --- Final verification with the actual evaluator ---
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = evaluate_sequence(best_f.tolist())
    print(f"\nFinal verified score: {final_score:.6f}")
    
    if not np.isfinite(final_score) or final_score > best_score:
        # Fallback: try simpler sequences
        test_seqs = [f_init.copy(), np.ones(n) / n]
        test_seqs.append(_spectral_flatten(f_init, n, n_passes=2))
        for ts in test_seqs:
            ts = np.clip(ts, 0.0, 1000.0)
            ts_score = evaluate_sequence(ts.tolist())
            if np.isfinite(ts_score) and ts_score < final_score:
                final_score = ts_score
                best_f = ts.copy()
    
    elapsed = time.monotonic() - t_start
    print(f"Total time: {elapsed:.1f}s")
    print(f"=== FINAL SCORE: {final_score:.6f} ===")
    
    return [float(x) for x in best_f]
