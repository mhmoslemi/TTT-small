    return np.fft.irfft(A*A, L)[:M]

def objective_p(p):
    n = int(p.size)
    if n <= 0: return np.inf
    c = fft_conv(p)
    c = np.maximum(c,0.0)
    S = float(p.sum())
    if S < 1e-100: return np.inf
    return float(2.0 * n * np.max(c) / (S*S))

def p_from_x(x):
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0: return None
    if not np.all(np.isfinite(x)): return None
    xm = x - np.max(x)
    xm = np.clip(xm, -60.0, 0.0)
    p = np.exp(xm)
    s = float(p.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return None
    p /= s
    p = np.clip(p, 0.0, 1000.0)
    s2 = float(p.sum())
    if s2 <= 1e-300:
        return None
    p /= s2
    return p

def p_from_shape(shape):
    a = np.asarray(shape,dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 1e-300, 1e300)
    s = a.sum()
    if s <= 1e-300: return None
    p = a/s
    return np.clip(p,0,1000)

def resample_log(arr,n):
    a=np.asarray(arr,dtype=np.float64).ravel()
    a=np.where(np.isfinite(a),a,0.0)
    a=np.clip(a,1e-300,1000.0)
    m=a.size
    if m==0:
       return np.ones(n,dtype=np.float64)/n
    if m==n:
       x=np.log(a)
    else:
       xs_old=np.linspace(0.0,1.0,m)
       xs_new=np.linspace(0.0,1.0,n)
       x=np.interp(xs_new,xs_old,np.log(a))
    p=p_from_x(x)
    if p is None:
       p=np.ones(n,dtype=np.float64)/n
    return p

def topk_indices(c,k):
    M=int(c.size)
    k=int(k)
    if k<=0:
       return np.array([],dtype=int), np.array([],dtype=float), np.array([],dtype=float)
    if k>=M:
       k=M
    idx=np.argpartition(c, -k)[-k:]
    vals=np.asarray(c[idx], dtype=np.float64)
    order=np.argsort(vals)[::-1]
    idx=idx[order]
    vals=vals[order]
    top=float(vals[0])
    if top > 1e-300:
       temp=0.08*top
       z=(vals-top)/temp
       wts=np.exp(np.clip(z,-50.0,50.0))
       sw=wts.sum()
       if not math.isfinite(sw) or sw<=0:
          wts=np.ones(k,dtype=np.float64)/k
       else:
          wts/=sw
    else:
       wts=np.ones(k,dtype=np.float64)/k
    return idx, vals, wts

def soft_weights(c, delta, temp_factor):
    M=int(c.size)
    top=float(np.max(c))
    if top <= 1e-300:
       w=np.zeros(M); w[int(np.argmax(c))] =1.0; return w
    thresh=top*(1.0-delta)
    temp=temp_factor*top
    if temp <= 1e-300:
       w=np.zeros(M); w[int(np.argmax(c))]=1.0; return w
    z=(c-thresh)/temp
    w=np.exp(np.clip(z,-50.0,50.0))
    sw=float(w.sum())
    if not math.isfinite(sw) or sw <=0:
       w=np.zeros(M); w[int(np.argmax(c))]=1.0
    else:
       w/=sw
    return w

def gradient_topk(p,c,idx,wts,vals):
    Cw=float(np.dot(wts, vals))
    n=int(p.size)
    grad = -2.0*Cw*p
    for ii in range(idx.size):
       k=int(idx[ii]); wi=float(wts[ii])
       if wi <= 1e-14:
          continue
       j0 = max(0, k - n + 1)
       j1 = min(k, n - 1)
       if j1 >= j0:
          grad[j0:j1+1] += 2.0*wi*p[j0:j1+1][::-1]
    return grad

def gradient_soft_fft(p,c,w,n,L,P):
    Cw=float(np.dot(w,c))
    wr = w[::-1]
    # maybe contiguous
    conv = np.fft.irfft(P * np.fft.rfft(wr, L), L)
    t = conv[n:2*n][::-1]
    grad = 2.0*t - 2.0*Cw*p
    return grad

"""
Search for a non-negative sequence minimizing
    2*n*max(conv(a,a)) / (sum(a))^2.

Strategy:
  * Work in log-softmax coordinates, enforcing positivity and sum=1.
  * Use fast FFT convolution for self-convolution.
  * Approximate the max-of-convolution objective by smoothly weighted
    top peaks or by a soft-threshold weight vector.
  * Use Adam-like updates, restarts, multi-resolution initials, and diverse
    random smooth initializations.
  * Validate and scale the final candidate using the provided evaluator.
"""

import numpy as np
import time
import math


# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------

def fft_conv(a):
    """Linear self-convolution via real FFT."""
    n = int(a.size)
    if n <= 0:
        return np.array([], dtype=np.float64)
    M = 2 * n - 1
    L = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, L)
    return np.fft.irfft(A * A, L)[:M]


def objective_p(p):
    """Objective for a probability vector p (scale invariant)."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n = int(p.size)
    if n <= 0:
        return np.inf
    c = fft_conv(p)
    c = np.maximum(c, 0.0)
    S = float(p.sum())
    if not math.isfinite(S) or S < 1e-100:
        return np.inf
    return float(2.0 * n * np.max(c) / (S * S))


def p_from_x(x):
    """Convert log-probabilities to a normalized non-negative vector."""
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size == 0 or not np.all(np.isfinite(x)):
        return None
    xm = x - np.max(x)
    xm = np.clip(xm, -60.0, 0.0)
    p = np.exp(xm)
    s = float(p.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return None
    p /= s
    p = np.clip(p, 0.0, 1000.0)
    s2 = float(p.sum())
    if not math.isfinite(s2) or s2 <= 1e-300:
        return None
    p /= s2
    return p


def add_to_cands(cands, p):
    """Add a normalized positive candidate if possible."""
    if p is None:
        return
    a = np.asarray(p, dtype=np.float64).ravel()
    if a.size == 0:
        return
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 1e-300, 1000.0)
    s = float(a.sum())
    if not math.isfinite(s) or s <= 1e-300:
        return
    a /= s
    cands.append(a)


def get_parent_array():
    """Fetch the optional global height_sequence_1 if present."""
    try:
        arr = height_sequence_1
    except Exception:
        return None
    try:

    if arr.size == 0:

    arr = np.where(np.isfinite(arr), arr, 0.0)

    return arr


def resample_log(arr, n):
    """Resample a positive array logarithmically to length n."""
    a = np.asarray(arr, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 1e-300, 1000.0)
    m = int(a.size)
    if m == n:
        x = np.log(a)
    else:
        x_old = np.linspace(0.0, 1.0, m)
        x_new = np.linspace(0.0, 1.0, n)
        x = np.interp(x_new, x_old, np.log(a))
    p = p_from_x(x)
    if p is None:
        p = np.ones(n, dtype=np.float64) / float(n)
    return p


def topk_indices(c, k):
    """Top-k convolution peaks with exponential soft-max weights."""
    M = int(c.size)
    k = int(k)
    if k <= 0:
        return np.array([], dtype=int), np.array([], dtype=float), np.array([], dtype=float)
    if k >= M:
        k = M
    idx = np.argpartition(c, -k)[-k:]
    vals = np.asarray(c[idx], dtype=np.float64)
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]
    top = float(vals[0])
    if top > 1e-300:
        temp = 0.08 * top
        z = (vals - top) / temp
        wts = np.exp(np.clip(z, -50.0, 50.0))
        sw = float(wts.sum())
        if not math.isfinite(sw) or sw <= 0.0:
            wts = np.ones(k, dtype=np.float64) / float(k)
        else:
            wts /= sw
    else:
        wts = np.ones(k, dtype=np.float64) / float(k)
    return idx, vals, wts


def soft_weights(c, delta, temp_factor):
    """Soft-threshold weights over all convolution peaks."""
    M = int(c.size)
    top = float(np.max(c))
    if top <= 1e-300:
        w = np.zeros(M, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    thresh = top * (1.0 - delta)
    temp = temp_factor * top
    if temp <= 1e-300:
        w = np.zeros(M, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
        return w
    z = (c - thresh) / temp
    w = np.exp(np.clip(z, -50.0, 50.0))
    sw = float(w.sum())
    if not math.isfinite(sw) or sw <= 0.0:
        w = np.zeros(M, dtype=np.float64)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= sw
    return w


def gradient_topk(p, c, idx, wts, vals):
    """Log-softmax gradient of a weighted combination of peaks."""
    Cw = float(np.dot(wts, vals))
    n = int(p.size)
    grad = -2.0 * Cw * p
    for ii in range(idx.size):
        k = int(idx[ii])
        wi = float(wts[ii])
        if wi <= 1e-14:
            continue
        j0 = max(0, k - n + 1)
        j1 = min(k, n - 1)
        if j1 >= j0:
            grad[j0:j1 + 1] += 2.0 * wi * p[j0:j1 + 1][::-1]
    return grad


def gradient_soft_fft(p, c, w, n, L, P):
    """Log-softmax gradient using an FFT-based correlation term."""
    Cw = float(np.dot(w, c))
    wr = w[::-1]
    conv = np.fft.irfft(P * np.fft.rfft(wr, L), L)
    if n == 1:
        t = conv[:1]
    else:
        t = conv[n:2 * n][::-1]
    return 2.0 * t - 2.0 * Cw * p


def optimize_log(p0, end, rng, style, topk, delta, temp_factor, lr, patience):
    """
    Local optimizer in log-softmax coordinates.

    style = 1: soft-threshold weighted peaks, FFT gradient.
    style = 0: top-k weighted peaks, slice gradient.
    """
    p = np.asarray(p0, dtype=np.float64).ravel()
    p = np.clip(p, 1e-300, 1000.0)
    S = float(p.sum())
    if not math.isfinite(S) or S <= 1e-300:
        p = np.ones(int(p.size), dtype=np.float64)
        S = float(p.size)
    p /= S
    n = int(p.size)

    x = np.log(p)
    x -= np.mean(x)
    x = np.clip(x, -50.0, 50.0)

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    titer = 0

    best_score = objective_p(p)
    best_p = p.copy()
    no_improve = 0
    cur_lr = float(lr)
    cur_patience = int(patience)

    if style == 1:
        L = 1 << max(1, (3 * n - 3).bit_length())
    else:
        L = 0

    b1 = 0.9
    b2 = 0.999
    eps = 1e-8

    while time.monotonic() < end:
        if style == 1:
            P = np.fft.rfft(p, L)
            c = np.fft.irfft(P * P, L)[:2 * n - 1]
            c = np.maximum(c, 0.0)
        else:
            c = fft_conv(p)
            c = np.maximum(c, 0.0)

        if not np.all(np.isfinite(c)):
            p = best_p.copy()
            x = np.log(p)
            x -= np.mean(x)
            x = np.clip(x, -50.0, 50.0)
            m[:] = 0.0
            v[:] = 0.0
            titer = 0
            cur_lr = max(1e-4, cur_lr * 0.5)
            continue

        S = float(p.sum())
        score = float(2.0 * n * np.max(c) / (S * S)) if S > 1e-300 else np.inf

        if math.isfinite(score) and score < best_score:
            best_score = score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= cur_patience:
            q = best_p * (1.0 + (0.05 + 0.15 * rng.random()) * rng.standard_normal(n))
            q = np.clip(q, 1e-300, 1000.0)
            sq = float(q.sum())
            if not math.isfinite(sq) or sq <= 1e-300:
                q = best_p.copy()
            else:
                q /= sq
            p = q
            x = np.log(p)
            x -= np.mean(x)
            x = np.clip(x, -50.0, 50.0)
            m[:] = 0.0
            v[:] = 0.0
            titer = 0
            cur_lr = max(1e-4, cur_lr * 0.7)
            cur_patience = max(40, cur_patience // 2)
            no_improve = 0
            continue

        if style == 1:
            w = soft_weights(c, delta, temp_factor)
            grad = gradient_soft_fft(p, c, w, n, L, P)
        else:
            idx, vals, wts = topk_indices(c, topk)
            grad = gradient_topk(p, c, idx, wts, vals)

        titer += 1
        m[:] = b1 * m + (1.0 - b1) * grad
        v[:] = b2 * v + (1.0 - b2) * (grad * grad)
        bc1 = 1.0 - b1 ** titer
        bc2 = 1.0 - b2 ** titer
        mh = m / bc1
        vh = v / bc2
        step = cur_lr * mh / (np.sqrt(vh) + eps)
        x = x - step

        if not np.all(np.isfinite(x)):
            x = np.log(best_p)
            x -= np.mean(x)
            x = np.clip(x, -50.0, 50.0)
            m[:] = 0.0
            v[:] = 0.0
            titer = 0
            cur_lr = max(1e-4, cur_lr * 0.5)
            continue

        x -= np.mean(x)
        spread = float(np.max(x) - np.min(x))
        if spread > 80.0:
            x = 40.0 * np.tanh(x / 40.0)
            x -= np.mean(x)
        x = np.clip(x, -60.0, 60.0)

        p_new = p_from_x(x)
        if p_new is None:
            p = best_p.copy()
            x = np.log(p)
            x -= np.mean(x)
            x = np.clip(x, -50.0, 50.0)
            m[:] = 0.0
            v[:] = 0.0
            titer = 0
            cur_lr = max(1e-4, cur_lr * 0.5)
        else:
            p = p_new

    return best_score, best_p


def linspace_center(n):
    return (np.arange(n, dtype=np.float64) + 0.5) / float(n)


def random_fourier(n, rng, nmodes, amp):
    """Random smooth positive shape from a low-order Fourier expansion."""
    x = linspace_center(n)
    z = np.zeros(n, dtype=np.float64)
    for m in range(1, nmodes + 1):
        if rng.random() < 0.55:
            w = (2 * m - 1) * math.pi
        else:
            w = 2.0 * math.pi * m
        scale = amp / (0.6 + 0.4 * math.sqrt(float(m)))
        a = float(rng.normal()) * scale
        b = float(rng.normal()) * scale
        z += a * np.cos(w * x) + b * np.sin(w * x)
    z += float(rng.normal()) * 0.3 * x
    z *= float(rng.uniform(0.7, 2.0))
    return p_from_x(z)


def random_poly(n, rng, degree):
    """Random polynomial-shaped log-density."""
    x = linspace_center(n)
    z = np.zeros(n, dtype=np.float64)
    for k in range(1, degree + 1):
        z += float(rng.normal()) * (x ** k) / math.sqrt(float(k))
    z += float(rng.normal()) * x
    z *= float(rng.uniform(0.6, 1.5))
    return p_from_x(z)


def random_smooth_noise(n, rng, sigma_frac):
    """Gaussian-smoothed white noise in log-domain."""
    n = int(n)
    if n <= 0:
        return None
    sigma = max(1.0, float(sigma_frac) * n)
    z = rng.standard_normal(n)
    freq = np.fft.rfftfreq(n) * n
    transfer = np.exp(-2.0 * math.pi * math.pi * sigma * sigma * freq * freq)
    Z = np.fft.rfft(z)
    z = np.fft.irfft(Z * transfer, n)
    z += float(rng.normal()) * 0.2
    return p_from_x(z)


def generate_candidates(n, rng, parent_p, n_rand):
    """A broad set of normalized positive initial candidates."""
    cands = []

    if parent_p is not None:
        add_to_cands(cands, parent_p)
        add_to_cands(cands, parent_p[::-1])
        add_to_cands(cands, np.power(parent_p, 0.6))
        add_to_cands(cands, np.power(parent_p, 1.6))
        for sigma in (0.05, 0.15):
            q = parent_p * (1.0 + sigma * rng.standard_normal(n))
            add_to_cands(cands, q)

    x = linspace_center(n)

    add_to_cands(cands, np.ones(n, dtype=np.float64))
    add_to_cands(cands, 1.0 - x + 1e-3)
    add_to_cands(cands, x + 1e-3)

    for a in (-0.8, -0.5, 0.5, 0.8):
        vals = 1.0 + a * np.cos(math.pi * x)
        add_to_cands(cands, vals)
        vals2 = 1.0 + 0.4 * a * np.cos(2.0 * math.pi * x) + 0.2 * a * np.cos(4.0 * math.pi * x)
        add_to_cands(cands, vals2)

    for _ in range(n_rand):
        nm = int(rng.integers(2, 8))
        amp = float(rng.uniform(0.8, 2.0))
        add_to_cands(cands, random_fourier(n, rng, nm, amp))

    for _ in range(3):
        add_to_cands(cands, random_smooth_noise(n, rng, float(rng.uniform(0.02, 0.08))))

    for _ in range(3):
        add_to_cands(cands, random_poly(n, rng, int(rng.integers(3, 7))))

    return cands


def score_and_sort(cands):
    """Score candidates and return them in ascending score order."""
    scored = []
    scores = []
    default_len = 1024

    for p in cands:
        a = np.asarray(p, dtype=np.float64).ravel()
        if a.size == 0:
            continue
        default_len = int(a.size)
        a = np.where(np.isfinite(a), a, 0.0)
        a = np.clip(a, 1e-300, 1000.0)
        s = float(a.sum())
        if not math.isfinite(s) or s <= 1e-300:
            continue
        a /= s
        sc = objective_p(a)
        if math.isfinite(sc):
            scored.append(a)
            scores.append(sc)

    if not scored:
        return [np.ones(default_len, dtype=np.float64) / float(default_len)], [2.0]

    scores_arr = np.asarray(scores, dtype=np.float64)
    order = np.argsort(scores_arr)
    outp = [scored[i] for i in order]
    outs = [float(scores_arr[i]) for i in order]
    return outp, outs

def propose_candidate(seed=42, budget_s=2400):
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = max(10.0, float(budget_s))
    reserve = 25.0
    usable = max(1.0, total - reserve)
    deadline = start + total - reserve

    parent = get_parent_array()

    if parent is not None and 1024 <= parent.size <= 8192:
        n = int(parent.size)
    else:
        n = 4096
    n = int(min(max(n, 1024), 8192))

    parent_p = resample_log(parent, n) if parent is not None else None

    best_score = np.inf
    best_p = None

    if parent_p is not None:
        sp = objective_p(parent_p)
        if math.isfinite(sp) and sp < best_score:
            best_score = sp
            best_p = parent_p.copy()

    print("starting n=%d parent_score=%s" % (n, best_score if math.isfinite(best_score) else "inf"))

    # ------------------------------------------------------------------
    # Coarse multi-scale phase: find a useful smooth basin quickly.
    # ------------------------------------------------------------------
    coarse_best_p = None
    coarse_best_score = np.inf
    coarse_time = min(90.0, 0.045 * usable)

    if coarse_time > 20.0:
        nc = 256
        parent_c = resample_log(parent, nc) if parent is not None else None
        coarse_cands, coarse_scores = score_and_sort(generate_candidates(nc, rng, parent_c, 8))

        if coarse_cands:
            coarse_best_p = coarse_cands[0].copy()
            coarse_best_score = coarse_scores[0]

            cnt = min(6, len(coarse_cands))
            wt = np.ones(cnt, dtype=np.float64)
            if cnt > 0:
                wt[0] = 2.0
            if cnt > 2:
                wt[1:3] = 1.5
            wsum = float(wt.sum())

            coarse_end = start + coarse_time
            for i in range(cnt):
                t0 = time.monotonic()
                if t0 >= coarse_end:
                    break
                rem = max(0.0, coarse_end - t0)
                if rem <= 2.0:
                    continue
                alloc = rem * float(wt[i]) / wsum
                if alloc < 2.0:
                    continue
                end_i = min(coarse_end, t0 + alloc)
                sc, pp = optimize_log(
                    coarse_cands[i], end_i, rng,
                    style=1, topk=4, delta=0.35, temp_factor=0.15,
                    lr=0.35, patience=60
                )
                if math.isfinite(sc) and sc < coarse_best_score:
                    coarse_best_score = sc
                    coarse_best_p = pp.copy()

            print("coarse best=%s" % coarse_best_score)

    # ------------------------------------------------------------------
    # Main search at target length.
    # ------------------------------------------------------------------
    main_cands = []

    if coarse_best_p is not None:
        cp = resample_log(coarse_best_p, n)
        add_to_cands(main_cands, cp)
        add_to_cands(main_cands, cp[::-1])

    gen_cands = generate_candidates(n, rng, parent_p, 18)
    for pc in gen_cands:
        add_to_cands(main_cands, pc)

    cands, scores = score_and_sort(main_cands)

    if best_p is None and cands:
        best_p = cands[0].copy()
        best_score = scores[0]

    main_end = start + 0.60 * usable

    if cands and main_end > time.monotonic() + 5.0:
        max_rank = min(len(cands), 12)
        for rank in range(max_rank):
            t0 = time.monotonic()
            if t0 >= main_end:
                break
            rem = max(0.0, main_end - t0)
            if rem <= 3.0:
                break

            if rank == 0:
                frac = 0.30
            elif rank == 1:
                frac = 0.18
            elif rank == 2:
                frac = 0.14
            elif rank == 3:
                frac = 0.12
            else:
                leftover = max(1, max_rank - 4)
                frac = 0.26 / float(leftover)

            alloc = frac * rem
            if alloc < 4.0 and rem > 12.0:
                alloc = 4.0
            end_i = min(main_end, t0 + alloc)
            if end_i - t0 < 2.0:
                continue

            if rank < 4:
                style = 1
                topk = 8
                delta = 0.18 if rank == 0 else 0.15
                temp_factor = 0.08 if rank == 0 else 0.06
                lr = 0.15 if rank == 0 else 0.12
                patience = 250 if rank == 0 else 200
            else:
                style = 0
                topk = 12
                delta = 0.0
                temp_factor = 0.0
                lr = 0.08
                patience = 150

            sc, pp = optimize_log(
                cands[rank], end_i, rng,
                style=style, topk=topk, delta=delta,
                temp_factor=temp_factor, lr=lr, patience=patience
            )

            if math.isfinite(sc) and sc < best_score:
                best_score = sc
                best_p = pp.copy()
                print("main rank %d improved to %s" % (rank, best_score))

    print("post-main best=%s" % best_score)

    # ------------------------------------------------------------------
    # Refine the global best.
    # ------------------------------------------------------------------
    phase2_end = start + 0.82 * usable
    if best_p is not None and phase2_end > time.monotonic() + 10.0:
        sc, pp = optimize_log(
            best_p, phase2_end, rng,
            style=0, topk=16, delta=0.0, temp_factor=0.0,
            lr=0.03, patience=350
        )
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = pp.copy()
        print("post-refine best=%s" % best_score)

    # ------------------------------------------------------------------
    # Optional longer-length attempt: upsample best to 2*n.
    # ------------------------------------------------------------------
    base_len = n
    if (best_p is not None and base_len < 8192 and
            time.monotonic() < start + 0.92 * usable - 120.0):
        n2 = min(2 * base_len, 8192)
        p2 = resample_log(best_p, n2)
        n2_end = start + 0.92 * usable

        if n2_end > time.monotonic() + 20.0:
            sc2, pp2 = optimize_log(
                p2, n2_end, rng,
                style=1, topk=16, delta=0.16, temp_factor=0.06,
                lr=0.08, patience=250
            )
            if math.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_p = pp2.copy()
                print("n2=%d improved to %s" % (n2, best_score))
            else:
                print("n2=%d not improved" % n2)

    # ------------------------------------------------------------------
    # Final polish.
    # ------------------------------------------------------------------
    if best_p is not None and deadline > time.monotonic() + 5.0:
        sc, pp = optimize_log(
            best_p, deadline, rng,
            style=0, topk=32, delta=0.0, temp_factor=0.0,
            lr=0.008, patience=600
        )
        if math.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = pp.copy()

    if best_p is None:
        best_p = np.ones(n, dtype=np.float64) / float(n)

    best_p = np.clip(best_p, 0.0, 1000.0)
    s = float(best_p.sum())
    if not math.isfinite(s) or s <= 1e-12:
        best_p = np.ones(n, dtype=np.float64) / float(n)
    else:
        best_p /= s

    # ------------------------------------------------------------------
    # Choose numerically stable scaling by testing a few output scales.
    # ------------------------------------------------------------------
    nm = int(best_p.size)
    scale_a = best_p.copy()
    scale_b = best_p * math.sqrt(2.0 * nm)
    mx = float(np.max(best_p)) if nm > 0 else 0.0
    if mx > 1e-300:
        scale_c = best_p / mx
    else:
        scale_c = best_p

    best_out = None
    best_eval = np.inf

    for arr in (scale_a, scale_b, scale_c):
        arr = np.asarray(arr, dtype=np.float64).ravel()
        arr = np.where(np.isfinite(arr), arr, 0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.size < 1 or float(arr.sum()) < 1e-6:
            continue
        try:
            ev = float(evaluate_sequence([float(v) for v in arr]))
        except Exception:
            ev = np.inf
        if math.isfinite(ev) and ev < best_eval:
            best_eval = ev
            best_out = arr

    if best_out is None:
        best_out = np.ones(nm, dtype=np.float64) * math.sqrt(2.0 * nm)

    best_out = np.where(np.isfinite(best_out), best_out, 0.0)
    best_out = np.clip(best_out, 0.0, 1000.0)

    final_eval = best_eval
    try:
        final_eval = float(evaluate_sequence([float(v) for v in best_out]))
    except Exception:
        pass

    print("final internal=%s final verified=%s len=%d time=%.1f" %
          (best_score, final_eval, int(best_out.size), time.monotonic() - start))

    return [float(v) for v in best_out]
