Optimized step-function search minimizing 2n*max(conv(a,a))/(sum(a))^2.
Uses log-sum-exp smoothed gradients with FFT-accelerated convolution,
adaptive temperature scheduling, and multi-phase optimization.
"""
import numpy as np
import time

def fast_conv(a):
    n = len(a)
    M = 2*n - 1
    M_fft = 1 << (M-1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A*A, M_fft)[:M]

def compute_score(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S*S)

def grad_logsumexp(a, beta):
    n = len(a)
    c = fast_conv(a)
    S = float(a.sum())
    if S < 1e-10:
        return np.zeros(n), np.inf, S, c

    c_max = float(c.max())
    exp_c = np.exp(beta * (c - c_max))
    p = exp_c / exp_c.sum()
    L = (c_max + float(np.log(exp_c.sum()))) / beta
    # Cross-correlation for gradient
    a_rev = a[::-1]
    N = 1 << (3*n - 2 - 1).bit_length()
    A_r = np.fft.rfft(a_rev, N)
    P = np.fft.rfft(p, N)
    conv2 = np.fft.irfft(A_r * P, N)
    g_L = 2.0 * conv2[n-1:2*n-1]
    S2 = S*S
    S3 = S2*S
    grad = 2.0*n*(g_L/S2 - 2.0*L/S3)
    return grad, L, S, c

def adam_step(a, g, m, v, t, lr):
    b1, b2, eps = 0.9, 0.999, 1e-8
    m[:] = b1*m + (1-b1)*g
    v[:] = b2*v + (1-b2)*g*g
    mh = m/(1-b1**t)
    vh = v/(1-b2**t)
    a -= lr * mh/(np.sqrt(vh)+eps)
    np.clip(a, 0, 1000, out=a)

def init_geometric(n, r):
    return np.clip(r**np.arange(n, dtype=float), 0.01, 1000.0)

def init_powerlaw(n, alpha):
    return np.clip((np.arange(n, dtype=float)+1)**alpha, 0.01, 1000.0)

def init_exp_decay(n, alpha):
    return np.clip(np.exp(-alpha*np.arange(n, dtype=float)), 0.001, 1000.0)

def init_gauss_window(n, alpha):
    x = np.arange(n, dtype=float)
    return np.clip(np.exp(alpha*(x/n - 1.0)/0.5 + alpha*(x/n - 1.0)**2), 0.01, 1000.0)

def init_random(n, rng):
    return np.clip(0.1 + rng.random(n), 0.01, 1000.0)

def init_quadratic_residue(n, p):
    """Initialize based on quadratic residues mod p."""
    a = np.zeros(n)
    for j in range(n):
        if j < p:
            a[j] = 1.0 if (j*j % p) < p//2 else 0.3
        else:
            a[j] = 0.3
    return np.clip(a, 0.01, 1000.0)

        # Evaluate actual score periodically
        if t % 100 == 0:
            c = fast_conv(a)
            S = a.sum()
            sc = compute_score(n, c, S)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve >= patience:

                a = best_a * (1 + 0.1 * rng.standard_normal(n))

        best_a = a.copy()
