    Actually, let's just compute sum_k w[k]*(a[k-i] + a[i-k]) for each i."""

    w_pad[:len(w)] = w
    a_pad = np.zeros(L)
    a_pad[:len(a)] = a

```python
"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences a.
Combines FFT-based soft-logsumexp optimization, exact subgradient refinement,
and multi-scale exploration across different sequence lengths.
"""

import time
import math
import numpy as np


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).copy()
    np.nan_to_num(a, copy=False, nan=0.0, posinf=0.0, neginf=0.0)
    np.maximum(a, 0.0, out=a)
    s = float(a.sum())
    if s <= 0:
        a.fill(1.0 / a.size)
        return a
    a /= s
    return a


def _conv_full(a, L):
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)
    return c[:2 * a.size - 1]


def _score(a):
    n = a.size
    s = float(a.sum())
    if s < 1e-15:
        return float('inf')
    L = 1 << (2 * n - 1).bit_length()
    c = _conv_full(a, L)
    return 2.0 * n * float(c.max()) / (s * s)


def _score_L(a, L):
    n = a.size
    s = float(a.sum())
    if s < 1e-15:
        return float('inf')
    c = _conv_full(a, L)
    return 2.0 * n * float(c.max()) / (s * s)


def _fft_gradient(w, a, L):
    """Compute grad[i] = sum_k w[k]*(a[k-i] + a[i-k]) via FFT."""
    n = a.size
    wp = np.zeros(L)
    wp[:w.size] = w
    ap = np.zeros(L)
    ap[:n] = a
    W = np.fft.rfft(wp)
    A = np.fft.rfft(ap)
    corr = np.fft.irfft(W * np.conj(A), L)[:n]
    conv = np.fft.irfft(W * A, L)[:n]
    return corr + conv


def _random_smooth(n, rng, numpoly=5):
    t = np.linspace(0, 1, n)
    x = 2 * t - 1
    z = np.zeros(n)
    for k in range(numpoly):
        amp = rng.uniform(0.1, 0.5) / (k + 1)
        ph = rng.uniform(0, 2 * np.pi)
        z += amp * np.cos(2 * np.pi * (k + 1) * t + ph)
    z += rng.normal(0, 0.3, n)
    # Smooth
    if n > 10:
        k = np.ones(9) / 9
        z = np.convolve(z, k, mode='same')
    p = np.exp(z - z.max())
    p = np.maximum(p, 1e-300)
    return _normalize(p)


def _make_gaussian(n, center, sigma, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-(t - center) ** 2 / (2 * sigma ** 2))
    return _normalize(a)


def _make_bump(n, c1, c2, w1, w2, rng):
    t = np.arange(n, dtype=np.float64)
    a = np.exp(-(t - c1) ** 2 / (2 * w1 ** 2)) + 0.5 * np.exp(-(t - c2) ** 2 / (2 * w2 ** 2))
    return _normalize(a)


def _soft_optimize(a, L, time_limit, rng, beta0=2.0, beta_max=500.0):
    """Main optimization loop using soft-logsumexp."""
    deadline = time.monotonic() + time_limit
    n = a.size
    m = 2 * n - 1
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score_L(a, L)
    
    beta = beta0
    eta = 0.08
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline:
        it += 1
        c = _conv_full(a, L)[:m]
        cur_score = 2.0 * n * float(c.max())
        
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        
        # Soft max weights
        bc = beta * c
        bm = float(bc.max())
        z = bc - bm
        z = np.clip(z, -700.0, 0.0)
        e = np.exp(z)
        s = float(e.sum())
        if s <= 0 or not np.isfinite(s):
            w = np.ones(m) / m
        else:
            w = e / s
        
        # Gradient
        grad = _fft_gradient(w, a, L)
        gmax = float(grad.max())
        if gmax <= 0 or not np.isfinite(gmax):
            break
        grad /= gmax
        
        # Update
        a_new = a * np.exp(-eta * grad)
        a_new = _normalize(a_new)
        
        # Evaluate
        c_new = _conv_full(a_new, L)[:m]
        sc = 2.0 * n * float(c_new.max())
        
        if sc < cur_score * (1 - 1e-10):
            a = a_new
            if sc < best_score:
                best_score = sc
                best_a = a_new.copy()
                no_improve = 0
            eta = min(0.4, eta * 1.03)
        else:
            no_improve += 1
            eta = max(0.01, eta * 0.85)
        
        # Adapt beta
        if no_improve > 30:
            beta = min(beta_max, beta * 1.4)
            no_improve = 0
            eta = 0.08
        elif no_improve > 80:
            # Escape: mix with random
            mix = rng.uniform(0.1, 0.3)
            prior = _random_smooth(n, rng)
            a = _normalize((1 - mix) * best_a + mix * prior)
            beta = 2.0
            eta = 0.08
            no_improve = 0
    
    return best_a, _score_L(best_a, L)


def _subgradient_refine(a, L, time_limit, rng):

    deadline = time.monotonic() + time_limit
    n = a.size
    m = 2 * n - 1
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score_L(a, L)
    
    eta = 0.02
    patience = 0
    
    while time.monotonic() < deadline:
        c = _conv_full(a, L)[:m]
        cur_max = float(c.max())
        cur_score = 2.0 * n * cur_max
        threshold = 0.995 * cur_max
        active = c >= threshold
        
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > 100:
            # Reset
            a = best_a.copy()
            eta = 0.01
            patience = 0
            # Small random perturbation
            noise = rng.normal(0, 0.01, n)
            a = _normalize(a + noise)
            continue
        
        # Subgradient at active set
        # grad[i] = (1/|S|) * sum_{k in S} (a[k-i] + a[i-k])
        # = (1/|S|) * [(w * a)[i] + (w ⊛ a)[i]] where w = indicator of S / |S|
        w = active.astype(np.float64)
        w /= w.sum()
        
        grad = _fft_gradient(w, a, L)
        gmax = float(grad.max())
        if gmax <= 0:
            break
        grad /= gmax
        
        # Update
        a_new = a * np.exp(-eta * grad)
        a_new = _normalize(a_new)
        
        c_new = _conv_full(a_new, L)[:m]
        sc = 2.0 * n * float(c_new.max())
        
        if sc < cur_score:
            a = a_new
            eta = min(0.1, eta * 1.1)
        else:
            eta = max(0
