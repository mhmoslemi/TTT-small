import numpy as np
import time
import math

def _next_pow2(n):
    if n <= 1:
        return 1
    return 1 << (n - 1).bit_length()

class Optimizer:
    def __init__(self, n, L_conv, L_corr):
        self.n = n
        self.L_conv = L_conv  # for self-convolution (>= 2n-1)
        self.L_corr = L_corr  # for gradient (>= 3n-2)
        
    def compute_r(self, a):
        """Compute self-convolution r = a*a, length 2n-1."""
        F = np.fft.rfft(a, self.L_conv)
        r = np.fft.irfft(F * F, self.L_conv)
        return r[:2*self.n-1].copy()
    
    def compute_grad(self, a, w):
        """Compute gradient g[i] = 2 * sum_k w[k] * a[k-i] (valid k)."""
        # g[i] = 2 * cross_correlation(a, w)[i] = 2 * IFFT(FFT(w) * conj(FFT(a)))[i]

                # Restart with perturbation
                a = _perturb_log(best_a, rng, sigma=0.1 + 0.2 * rng.random())
                tau = max(tau0, tau * 0.4)
                eta = 0.3
                no_improve = 0

        it += 1
        if it % 200 == 0:
            # Periodic L2 mixing step
            r = _self_conv(a, L)
            g2 = _grad_corr(a, r, L_corr)
            a_mix = a * np.exp(-0.1 * g2 / max(g2.max(), 1e-30))
            a_mix = _normalize(a_mix)
            sc_mix = _score_fft(a_mix, L)
            if sc_mix < best_score - 1e-14:
                a = a_mix
                best_score = sc_mix
                best_a = a.copy()

    return best_a, best_score


def _polish(a, L, L_corr, rng, time_limit):
    """Targeted polishing: reduce the top few autocorrelation peaks."""
    deadline = time.monotonic() + time_limit
    n = len(a)
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score_fft(a, L)

    it = 0
    while time.monotonic() < deadline:
        r = _self_conv(a, L)
        top_k = min(10, len(r))
        idx_top = np.argsort(r)[::-1][:top_k]

        # For each hot spot, try to reduce it by moving mass
        improved = False
        for ki in idx_top:
            # r[ki] = sum_j a[j] * a[ki-j] (valid j)
            # Positions contributing most to r[ki]:
            lo = max(0, ki - (n - 1))
            hi = min(n - 1, ki)
            if lo >= hi:
                continue
            contrib = a[lo:hi + 1] * a[ki - hi:ki - lo + 1][::-1]
            # Find the position with the most "excess" mass
            target = lo + int(np.argmax(contrib))
            # Try moving a small amount from target to a random low-position
            low_positions = np.argsort(r)[:(n // 4)]
            dest = int(low_positions[rng.integers(len(low_positions))])
            if dest == target:
                continue

            frac = 0.01 + 0.04 * rng.random()
            a_trial = a.copy()
            a_trial[target] *= (1.0 - frac)
            a_trial[dest] += frac * a[target] / max(a[dest], 1e-30)
            a_trial = _normalize(a_trial)
            sc = _score_fft(a_trial, L)
            if sc < best_score - 1e-14:
                a = a_trial
                best_score = sc
                best_a = a.copy()
                improved = True

        it += 1
        if not improved and it % 20 == 0:
            # Small random perturbation
            a = _perturb_log(best_a, rng, sigma=0.05)

    return best_a, best_score


def _exact_eval(seq):
    try:
        val = evaluate_sequence(seq)
    except NameError:
        a = [0.0 if float(x) < 0 else min(1000.0, float(x)) for x in seq]
        n = len(a)
        s = sum(a)
        if s < 0.01:
            return float('inf')
        b = np.convolve(a, a)
        return float(2.0 * n * max(b) / (s * s))
    except Exception:
        return float('inf')
    if val is None:
        return float('inf')
    return float(val)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    
    Strategy:
    1. Initialize from height_sequence_1 if available, plus structured inits.
    2. Run LSE-based multiplicative optimization with adaptive temperature.
    3. Periodically mix with L2 objective.
    4. Final polishing phase targeting top autocorrelation peaks.
    5. Return the best sequence by exact evaluation.
    """
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget < 10.0:
        budget = 2400.0

    start_time = time.monotonic()
    hard_deadline = start_time + max(10.0, budget - 3.0)
    rng = np.random.default_rng(seed)

    n = 6121
    L_conv = _next_pow2(2 * n - 1)
    L_corr = _next_pow2(3 * n - 2)

    # --- Initialization ---
    candidates = []
    scores = []

    # Try to load previous best
    base = None
    try:
        base = height_sequence_1
    except NameError:
        base = None
    except Exception:
        base = None

    if base is not None:
        try:
            arr = np.asarray(base, dtype=np.float64).ravel()
            if arr.size == n and np.all(np.isfinite(arr)):
                arr = _normalize(arr)
                sc = _score_fft(arr, L_conv)
                candidates.append(arr)
                scores.append(sc)
                print(f"Loaded previous best: score={sc:.6f}")
        except Exception as e:
            print(f"Failed to load base: {e}")

    # Structured inits
    for init in _structured_inits(n, rng):
        sc = _score_fft(init, L_conv)
        candidates.append(init)
        scores.append(sc)

    best_idx = int(np.argmin(scores))
    best_a = candidates[best_idx].copy()
    best_score = scores[best_idx]
    print(f"Best initial score: {best_score:.6f}")

    # --- Main optimization ---
    # Phase 1: Initial exploration with multiple starts
    alloc1 = min(300.0, 0.15 * budget)
    for ci in range(min(5, len(candidates))):
        if time.monotonic() > hard_deadline - 60.0:
            break
        remaining = alloc1 + start_time - time.monotonic()
        if remaining < 5.0:
            break
        t_lim = min(60.0, remaining)
        a_cand = candidates[ci].copy()
        a_new, sc_new = _optimize_lse(a_cand, L_conv, L_corr, rng, t_lim,
                                       tau0=30.0, tau1=500.0)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new
            print(f"  Phase1 init {ci}: improved to {best_score:.6f}")

    # Phase 2: Main deep optimization from best
    phase2_deadline = hard_deadline - 120.0  # leave 120s for polishing+eval
    t_lim = phase2_deadline - time.monotonic()
    if t_lim > 30.0:
        print(f"\nPhase 2: deep optimization, time_limit={t_lim:.0f}s")
        a_best = _normalize(best_a)
        # Run multiple rounds with different temperature ranges
        sub_rounds = 4
        sub_time = t_lim / sub_rounds
        for rnd in range(sub_rounds):
            if time.monotonic() > phase2_deadline:
                break
            tau0_r = 20.0 + 40.0 * rnd
            tau1_r = 200.0 + 400.0 * rnd
            a_new, sc_new = _optimize_lse(a_best, L_conv, L_corr, rng, sub_time,
                                           tau0=tau0_r, tau1=tau1_r)
            if sc_new < best_score:
                best_score = sc_new
                best_a = a_new.copy()
                print(f"  Round {rnd}: score={best_score:.6f}")
            else:
                print(f"  Round {rnd}: no improvement (best={best_score:.6f})")
            # Perturb for next round
            if rnd < sub_rounds - 1:
                a_best = _perturb_log(best_a, rng, sigma=0.08)

    # Phase 3: Polishing
    polish_time = min(90.0, hard_deadline - time.monotonic() - 30.0)
    if polish_time > 10.0:
        print(f"\nPhase 3: polishing, time={polish_time:.0f}s")
        a_new, sc_new = _polish(best_a, L_conv, L_corr, rng, polish_time)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            print(f"  Polish improved to: {best_score:.6f}")

    # Phase 4: Final short LSE pass
    final_time = max(5.0, hard_deadline - time.monotonic() - 20.0)
    if final_time > 10.0:
        a_new, sc_new = _optimize_lse(best_a, L_conv, L_corr, rng, final_time,
                                       tau0=200.0, tau1=3000.0)
        if sc_new < best_score:
            best_score = sc_new
            best_a = a_new.copy()
            print(f"  Final pass: {best_score:.6f}")

    # --- Evaluation ---
    print(f"\nEvaluating candidates...")
    best_seq = None
    best_exact = float('inf')

    eval_list = [best_a]
    # Add a few perturbed versions of the best
    for _ in range(3):
        eval_list.append(_perturb_log(best_a, rng, sigma=0.02))

    for arr in eval_list:
        if time.monotonic() > hard_deadline - 2.0:
            break
        seq = [float(max(0.0, x)) for x in arr]
        val = _exact_eval(seq)
        print(f"  Exact score: {val:.6f}")
        if np.isfinite(val) and val < best_exact:
            best_exact = val
            best_seq = seq

    if best_seq is None:
        best_seq = [float(max(0.0, x)) for x in _normalize(best_a)]

    elapsed = time.monotonic() - start_time
    print(f"\nDone in {elapsed:.1f}s. Best exact score: {best_exact:.6f}")
    return best_seq
