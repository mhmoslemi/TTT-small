    """Solve one QP step of the SQP method."""
    n = a_curr.size
    a = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [a >= 0, cp.sum(a) == 1]
    
    for k in range(2*n-1):
        # Linearization of (a*a)[k] around a_curr

        for i in range(max(0, k-n+1), min(k, n-1)+1):

    prob = cp.Problem(cp.Minimize(t), constraints)

Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using:

3. Multi-start with diverse initializations

import numpy as np
import time

def project_simplex(v):
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    cond = u * idx > (cssv - 1.0)
    if not cond.any():
        return np.ones(n) / n
    rho = np.where(cond)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def fft_conv(a):
    """Full linear convolution a*a via FFT. Result length = 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def score_normalized(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    b = fft_conv(a)
    return 2.0 * n * float(np.max(b))


def compute_active_set(a, b, frac=0.95):
    """Return indices k where b[k] >= frac * max(b)."""
    threshold = frac * b.max()
    return np.where(b >= threshold)[0]

    """

    a_curr: current point (sum=1, a>=0)
    delta: trust region radius

    a = cp.Variable(n)
    t = cp.Variable()

    constraints.append(cp.sum_squares(a - a_curr) <= delta ** 2)
    
    # Determine which constraints to include
    b_curr = fft_conv(a_curr)
    max_b = b_curr.max()
    
    if use_all_constraints:
        k_range = range(n)  # By symmetry, k=0..n-1 suffices
    else:
        active = compute_active_set(a_curr, b_curr, frac=0.9)
        # Always include k=0 and k=n-1
        k_set = set(range(n))
        k_set.update(active.tolist())
        k_set.discard(2*(n-1) - (n-1))  # clean up
        k_range = sorted(set(k for k in range(n) if k in k_set))
    
    for k in k_range:
        # (a*a)[k] = sum_{i: 0<=i<n, 0<=k-i<n} a[i]*a[k-i]

        expr = cp.Constant(0.0)
        i_min = max(0, k - (n - 1))

            j = k - i
            expr = expr + a_curr[i] * a[j] + a_curr[j] * a[i] - a_curr[i] * a_curr[j]
        constraints.append(expr <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:

        try:

        except Exception:
            return a_curr, None, False

        return a_curr, None, False

    a_new = a_new / a_new.sum()  # re-normalize to handle solver tolerance

    b_new = fft_conv(a_new)
    t_actual = float(np.max(b_new))
