```python
Step function optimizer minimizing 2*n*max(f*f)/(sum f)^2.
Multi-phase approach: mirror descent with annealing, 
direct peak targeting, and coordinate descent polish.
"""

import numpy as np
import time


def _fft_conv(a):
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]


def _peak_gradient(a, k_star):
    """Compute gradient of b[k_star] w.r.t. a. O(n) time."""
    n = len(a)
    idx = k_star - np.arange(n)
    valid = (idx >= 0) & (idx < n)
    G = np.zeros(n)
    G[valid] = 2.0 * a[idx[valid]]
    return G


def _score(a):
    return evaluate_sequence(a.tolist())


def _mirror_descent(a, deadline, rng):
    """Mirror descent with annealed softmax gradient."""
    n = len(a)
    a = a.copy().astype(np.float64)
    best_score = _score(a)
    best_a = a.copy()
    
    b = _fft_conv(a)
    peak0 = b.max()
    
    T_start = peak0 * 0.05
    T_end = peak0 * 0.0005
    eta_start = 0.3
    eta_end = 0.01
    
    it = 0
    eval_interval = 1000
    perturb_interval = 30000
    
    while time.monotonic() < deadline:
        it += 1
        progress = min(it / 150000.0, 1.0)
        T = T_start * (T_end / T_start) ** progress
        eta = eta_start * (eta_end / eta_start) ** progress
        
        b = _fft_conv(a)
        
        # Softmax
        bs = b / T
        bs -= bs.max()
        w = np.exp(bs)
        ws = w.sum()
        if ws < 1e-300:
            w = np.zeros(len(b))
            w[np.argmax(b)] = 1.0
        else:
            w /= ws
        
        # Gradient: G[i] = 2 * sum_m a[m] * w[m+i]
        G = 2.0 * np.correlate(w, a, mode='valid')
        
        # Mirror descent step
        g_scale = G.max()
        if g_scale < 1e-15:
            break
        a = a * np.exp(-eta * G / g_scale)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-10:
            a = best_a * (1.0 + 0.01 * rng.standard_normal(n))
            a = np.maximum(a, 0.0)
            a /= a.sum()
        else:
            a /= s
        
        if it % eval_interval == 0:
            sc = _score(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            
            if it % perturb_interval == 0:
                # Small perturbation to escape local minima
                pert_scale = 0.003 * (1.0 - progress * 0.5)
                a_p = a * (1.0 + pert_scale * rng.standard_normal(n))
                a_p = np.maximum(a_p, 0.0)
                if a_p.sum() > 1e-10:
                    a_p /= a_p.sum()
                    sc_p = _score(a_p)
                    if sc_p < best_score + 0.002:
                        a = a_p
    
    sc = _score(a)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    
    return best_a, best_score


def _peak_targeting(a, deadline, rng):
    """Direct peak targeting with exponential update."""
    n = len(a)
    a = a.copy()
    best_score = _score(a)
    best_a = a.copy()

    n_peaks = 3

        top_idx = np.argsort(b)[::-1][:n_peaks]

        G = np.zeros(n)

            weight = 1.0 / (rank + 1)
            idx = k - np.arange(n)
```

```python
def _waterfill_step(a, G, target_sum=1.0):
    """Step in direction -G with water-filling to maintain sum and non-negativity."""
    n = len(a)
    # We want: a_new[i] = max(0, a[i] - eta*G[i] - lambda) for some eta, lambda
    # such that sum a_new = target_sum
    # Binary search on lambda for a given eta
    # Or: binary search on both eta and lambda
    
    # Simpler: for a given eta, find lambda by bisection
    def find_lambda(eta):
        # a_new[i] = max(0, a[i] - eta*G[i] - lambda)
        # sum a_new = target_sum
        # Try different lambda values
        # a[i] - eta*G[i] - lambda > 0 iff lambda < a[i] - eta*G[i]
        # So for a given lambda, a_new[i] = max(0, a[i] - eta*G[i] - lambda)
        # sum = sum_i max(0, a[i] - eta*G[i] - lambda)
        
        # The sum is a decreasing function of lambda.
        # At lambda = -inf: sum = sum a[i] (if eta*G[i] can be negative)
        # At lambda = +inf: sum = 0
        
        # Find lambda such that sum = target_sum
        # Use binary search
        lo = -max(a + eta*G)  # very negative
        hi = max(a - eta*G) + 1  # very positive
        
        for _ in range(50):
            mid = (lo + hi) / 2
            vals = np.maximum(0, a - eta*G - mid)
            s = vals.sum()
            if s > target_sum:
                lo = mid
            else:
                hi = mid
        
        lambda_mid = (lo + hi) / 2
        return lambda_mid
    
    # Choose eta: not too large to avoid making many a[i] = 0
    eta = 0.1 / (G.max() + 1e-15)
    
    lam = find_lambda(eta)
    a_new = np.maximum(0, a - eta * G - lam)
    s = a_new.sum()
    if s < 1e-15:
        return a
    a_new /= s  # exact normalization
    return a_new
```
