Multi-strategy search for non-negative sequences minimizing
    2*n*max(conv(a,a)) / sum(a)^2
Combines frequency-domain convolution flattening, multiplicative
gradient descent with adaptive smoothed-max, and block coordinate
refinement. Uses multiple sequence lengths and diverse initializations.

import numpy as np
import time

# --- Utility functions ---

def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(a, b):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.convolve(a, b)
    L = _next_pow2(nout)
    return np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)[:nout]

def _score(p):
    """Compute 2*n*max(conv(p,p)) for normalized p (sum=1)."""
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())

def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return None
    return p / s

def _freq_flatten(p, alpha, L=None):
    """
    Frequency-domain convolution flattening.
    alpha > 1 flattens the convolution.
    Returns a new normalized non-negative sequence.
    """
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)
    
    A = np.fft.rfft(p, L)
    C_freq = A * A
    C = np.fft.irfft(C_freq, L)[:2*n-1]
    
    cmax = C.max()
    if cmax < 1e-15:
        return p.copy()
    
    # Flatten: raise to power 1/alpha (alpha > 1 compresses range)
    C_new = cmax * (C / cmax) ** (1.0 / alpha)
    C_new = np.maximum(C_new, 0)
    
    C_freq_new = np.fft.rfft(C_new, L)
    # Complex square root
    A_new = np.sqrt(C_freq_new.astype(complex))
    a_new = np.fft.irfft(A_new, L)[:n]
    
    a_new = np.maximum(a_new, 0)
    s = a_new.sum()
    if s < 1e-10:
        return None
    return a_new / s

def _soft_weights(d, beta):
    n = d.size
    if n == 0:
        return np.array([])
    d = np.where(np.isfinite(d), d, 0.0)
    if beta < 1e-8:
        return np.ones(n) / n
    shift = d.max()
    e = np.exp(beta * (d - shift))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):
        return np.ones(n) / n
    return e / s

def _gradient_step(p, beta, alpha, L=None):
    """One multiplicative gradient descent step."""
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)
    
    c = _conv_full(p, p)
    d = n * c  # scaled convolution
    exact = 2.0 * d.max()
    
    w = _soft_weights(d, beta)
    # Gradient: d/dp_i sum_j w_j * (p*p)[j] = 2 * (w * p)_i (circular, but we use linear)
    # For linear conv, the gradient of (p*p)[k] w.r.t. p[i] is p[k-i] (if k-i valid)
    # So the overall gradient is: h[i] = sum_k w[k] * p[k-i] (for valid k-i)
    # This is the valid correlation of w with p (reversed).
    
    # h[i] = sum_k w[k] * p[k-i] where k-i must be in [0, n-1]
    # Equivalently, h = valid_convolve(w_flipped, p) where w_flipped[k] = w[n-1-k]
    # Actually, let me think more carefully.
    # (p*p)[k] = sum_i p[i] * p[k-i] (valid range)
    # d(p*p)[k]/dp[i] = p[k-i] (if 0 <= k-i < n)
    # So gradient of sum_k w[k]*(p*p)[k] w.r.t. p[i] = sum_k w[k]*p[k-i] = (w*p)_i (correlation)
    # Which is: h[i] = sum_k w[k]*p[k-i] = sum_j w[i+j]*p[j] (letting j = k-i)
    # = (w_rev * p)_i where w_rev[k] = w[-k] ... 
    
    # Simpler: use FFT
    # h[i] = sum_j w[i+j] * p[j] for j = 0..min(n-1, ...) 
    # This is the "valid" cross-correlation of w and p.
    # h = irfft(rfft(w, L) * conj(rfft(p, L)))[:n]
    
    fp = np.fft.rfft(p, L)
    fw = np.fft.rfft(w, L)
    h = np.fft.irfft(fw * np.conj(fp), L)[:n]
    
    h = np.maximum(h, 0)
    hmax = h.max()
    if hmax < 1e-15:
        return p.copy(), exact, 0.0
    
    h = h / hmax
    dec = np.exp(-alpha * h)
    p_new = p * dec
    p_new = np.maximum(p_new, 1e-18 / n)
    s = p_new.sum()
    if s < 1e-10:
        return p.copy(), exact, 0.0
    p_new = p_new / s
    return p_new, exact, hmax

def _init_gaussian_mixture(n, rng):
    """Sum of Gaussian bumps."""
    K = int(rng.integers(2, 8))
    p = np.zeros(n)
    t = np.arange(n) / n
    for _ in range(K):
        c = rng.uniform(0, 1)

        w = rng.uniform(0.5, 2.0)
        x = (t - c) / s
        # Wrap around
        x = np.minimum(np.abs(x), 1 - np.abs(x))
        p += w * np.exp(-0.5 * x**2)
    p = np.maximum(p, 0)
    return p / p.sum()

def _init_cosine(n, rng):
    """Cosine-based initialization."""
    t = np.arange(n) / n
    p = np.ones(n)
    for k in range(1, 8):
        amp = rng.uniform(-0.5, 0.5) / k
        ph = rng.uniform(0, 2 * np.pi)
        p += amp * np.cos(2 * np.pi * k * t + ph)
    p = np.maximum(p, 0)
    if p.sum() < 1e-10:
        p = np.ones(n)
    return p / p.sum()

        p = pn.copy()

        x_new = (np.arange(n) + 0.5) / n
        p = np.interp(x_new, x_old, pn)
    p = np.maximum(p, 0)
    # Perturb
    p = p * np.exp(rng.normal(0, 0.05, n))

Multi-strategy search for non-negative sequences minimizing
    2*n*max(conv(a,a)) / sum(a)^2
Combines frequency-domain convolution flattening (a novel "convolution square root"
approach), multiplicative gradient descent with adaptive smoothed-max, block
coordinate refinement, and diverse initializations across multiple lengths.

import numpy as np
import time


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b, L=None):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.convolve(a, b).astype(float)
    if L is None:
        L = _next_pow2(nout)
    out = np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)
    return out[:nout].astype(float)


def _score(p):
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10:
        return float('inf')
    p = p / s
    n = len(p)
    c = _conv_full(p, p)
    return float(2.0 * n * c.max())


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s < 1e-10 or not np.isfinite(s):

    return p / s


def _soft_weights(d, beta):
    n = d.size

    d = np.where(np.isfinite(d), d, 0.0)
    if beta < 1e-8:

    shift = d.max()
    e = np.exp(beta * (d - shift))
    s = e.sum()
    if s < 1e-300 or not np.isfinite(s):

    return e / s

def _freq_flatten(p, alpha, L=None):
    """
    Frequency-domain convolution flattening (convolution square root).
    alpha > 1: more flattening of the convolution.
    Returns new normalized non-negative sequence or None.
    """
    n = len(p)
    if L is None:
        L = _next_pow2(2 * n - 1)
    A = np.fft.rfft(p, L)
    C_freq = A * A
    C = np.fft.irfft(C_freq, L)[:2 * n - 1]
    cmax = C.max()
    if cmax < 1e-15:
        return p.copy()
    C_norm = C / cmax
    C_new = cmax * np.power(C_norm, 1.0 / alpha)
    C_new = np.maximum(C_new, 0.0)

    a_new = np.maximum(a_new, 0.0)
    s = a_new.sum()

    if verbose:
        print(f"    Hybrid opt: n={n}, init_score={best_score:.6f}, budget={time_budget:.0f}s")

    while time.monotonic() < deadline:
        # Decide action
        r = rng.random()

        if r < 0.55:
            # Gradient step
            sc, h, max_h = _gradient_info(p, beta, L)
            if max_h < 1e-15:
                stall += 1
                continue
            h_norm = h / max_h

            # Try multiple step sizes
            best_trial = None
            best_trial_sc = cur_score
            for a_try in (alpha * 0.5, alpha, alpha * 2.0):
                a_try = float(np.clip(a_try, 0.005, 5.0))
                p_try = p * np.exp(-a_try * h_norm)
                p_try = np.maximum(p_try, 1e-18 / n)
                s_try = p_try.sum()
                if s_try < 1e-10:
                    continue
                p_try = p_try / s_try
                sc_try = _score(p_try)
                if sc_try < best_trial_sc - 1e-13:
                    best_trial = p_try
                    best_trial_sc = sc_try

            if best_trial is not None:
                p = best_trial
                cur_score = best_trial_sc
                stall = 0
                if cur_score < best_score - 1e-12:
                    best = p.copy()
                    best_score = cur_score
                    if verbose and it % 200 == 0:
                        print(f"      iter={it}, score={best_score:.6f}, beta={beta:.0f}, alpha={alpha:.4f}")
            else:
                stall += 1
                if stall > 8:
                    alpha = max(0.01, alpha * 0.7)
                if stall > 40:
                    p = best.copy()
                    cur_score = best_score
                    alpha = 0.12
                    stall = 10

        elif r < 0.78:
            # Frequency flattening
            alpha_f = float(rng.uniform(1.3, 5.0))
            p_new = _freq_flatten(p, alpha_f, L_fft)
            if p_new is not None:
                sc_new = _score(p_new)
                if sc_new < cur_score - 1e-13:
                    p = p_new
                    cur_score = sc_new
                    stall = 0
                    if sc_new < best_score - 1e-12:
                        best = p.copy()
                        best_score = sc_new
                        if verbose:
                            print(f"      FREQ iter={it}, alpha_f={alpha_f:.2f}, score={best_score:.6f}")
                else:
                    stall += 1
            else:
                stall += 1

        elif r < 0.90:
            # Block scaling
            bs = int(rng.choice([2, 4, 8, 16, 32, 64, 128]))
            bs = min(bs, n)
            start = int(rng.integers(0, n - bs + 1))
            f = float(rng.choice([0.3, 0.5, 0.7, 1.4, 1.8, 2.5]))
            p_try = p.copy()
            p_try[start:start + bs] *= f
            p_try = _sanitize(p_try)
            if p_try is not None:
                sc_try = _score(p_try)
                if sc_try < cur_score - 1e-13:
                    p = p_try
                    cur_score = sc_try
                    stall = 0
                    if sc_try < best_score - 1e-12:
                        best = p.copy()
                        best_score = sc_try
                else:
                    stall += 1
        else:
            # Random perturbation
            if stall > 5:
                std = 0.02 + 0.005 * min(stall, 30)
                p_try = p * np.exp(rng.normal(0.0, std, n))
                p_try = _sanitize(p_try)
                if p_try is not None:
                    sc_try = _score(p_try)
                    if sc_try < best_score * 1.08:
                        p = p_try
                        cur_score = sc_try
                        stall = 0
            else:
                stall += 1

        # Update beta (sharpen focus)
        if it % 10 == 0:
            beta = min(beta_end, beta * 1.08)

        # Periodic pullback
        if cur_score > best_score * 1.15 and best_score < float('inf'):
            if rng.random() < 0.3:
                p = best.copy()
                cur_score = best_score
                stall = 0

        it += 1

    return best, best_score


def _block_refine(p0, rng, time_budget):
    """Block-wise multiplicative refinement with exact scoring."""
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    best = p.copy()
    best_score = _score(p)
    cur_score = best_score
    deadline = time.monotonic() + time_budget
    stall = 0
    factors = (0.4, 0.6, 0.8, 1.25, 1.5, 2.0, 3.0)

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.2:
            bs = int(rng.integers(1, 5))
        elif r < 0.6:
            bs = int(rng.integers(max(2, n // 256), max(3, n // 32)))
        else:
            bs = int(rng.integers(max(8, n // 32), max(16, n // 4)))
        bs = min(bs, n)
        start = int(rng.integers(0, n - bs + 1))

        improved = False
        for f in factors:
            p_try = p.copy()
            p_try[start:start + bs] *= f
            p_try = _sanitize(p_try)
            if p_try is None:
                continue
            sc = _score(p_try)
            if sc < cur_score - 1e-13:
                cur_score = sc
                p = p_try
                improved = True
                break

        if improved:
            if cur_score < best_score - 1e-12:
                best = p.copy()
                best_score = cur_score
            stall = 0
        else:
            stall += 1
            if stall > 50:
                p = best.copy()
                cur_score = best_score
                stall = 0

    return best, best_score


def _init_uniform(n):
    return np.ones(n, dtype=float) / n


def _init_two_level(n, rng):
    k = int(n * rng.uniform(0.3, 0.7))
    ratio = float(rng.uniform(1.5, 5.0))
    p = np.ones(n, dtype=float)
    p[k:] = ratio
    idx = rng.permutation(n)
    p = p[idx]
    p = p * (1.0 + 0.1 * rng.normal(0, 1, n))
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return _init_uniform(n)
    return p / s


def _init_gaussian_mixture(n, rng):
    K = int(rng.integers(2, 8))
    p = np.zeros(n, dtype=float)
    t = np.arange(n, dtype=float) / n
    for _ in range(K):
        c = rng.uniform(0.0, 1.0)
        s = rng.uniform(0.02, 0.3)
        w = rng.uniform(0.5, 2.0)
        x = (t - c) / s
        x = np.minimum(np.abs(x), 1.0 - np.abs(x))
        p += w * np.exp(-0.5 * x * x)
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _init_cosine(n, rng):
    t = np.arange(n, dtype=float) / n
    p = np.ones(n, dtype=float)
    for k in range(1, 10):
        amp = rng.uniform(-0.6, 0.6) / k
        ph = rng.uniform(0, 2 * np.pi)
        p += amp * np.cos(2 * np.pi * k * t + ph)
    p = np.maximum(p, 0)
    s = p.sum()
    if s < 1e-10:
        return _init_uniform(n)
    return p / s


def _init_block_random(n, rng):
    K = int(rng.integers(4, 20))
    p = np.zeros(n, dtype=float)
    edges = np.sort(rng.uniform(0, 1, K - 1))
    blocks = np.concatenate([[0], edges, [1]])
    heights = rng.lognormal(0, 1, K)
    for i in range(K):
        s = int(round(blocks[i] * n))
        e = max(s + 1, int(round(blocks[i + 1] * n)))
        e = min(n, e)
        p[s:e] = heights[i]
    p = np.maximum(p, 1e-10)
    return p / p.sum()


def _resample(p_in, n_new, rng=None):
    p = _sanitize(p_in)
    if p is None:
        return None
    n_old = len(p)
    if n_old == n_new:
        result = p.copy()
    elif n_old == 1:
        result = np.ones(n_new, dtype=float)
    else:
        x_old = (np.arange(n_old) + 0.5) / n_old
        x_new = (np.arange(n_new) + 0.5) / n_new
        result = np.interp(x_new, x_old, p)
    result = np.maximum(result, 0)
    if rng is not None:
        result = result * np.exp(rng.normal(0, 0.03, n_new))
        result = np.maximum(result, 0)
    s = result.sum()
    if s < 1e-10:
        return None
    return result / s


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing the evaluation score.
    Multi-strategy: frequency-domain flattening + gradient descent + block refine.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 2400.0

    start_time = time.monotonic()
    deadline = start_time + budget - 15

    try:
        iseed = int(seed)
    except Exception:
        iseed = 42
    rng = np.random.default_rng(iseed % (2**31 - 1))

    # Get parent
    try:
        parent = _sanitize(np.asarray(height_sequence_1, dtype=float))
    except Exception:
        parent = None

    best_overall = None
    best_score_overall = float('inf')

    def _try_update(p, score):
        nonlocal best_overall, best_score_overall
        if p is not None and np.isfinite(score) and score < best_score_overall:
            best_overall = p.copy()
            best_score_overall = float(score)
            print(f"  *** NEW BEST: {score:.6f} (n={len(p)}) ***")

    # ========== PHASE 1: Quick exploration (first 15% of time) ==========
    print("=" * 60)
    print(f"Phase 1: Quick exploration (budget={budget:.0f}s)")
    t1_end = start_time + 0.15 * budget

    # Evaluate parent
    if parent is not None:
        sc = _score(parent)
        _try_update(parent, sc)
        print(f"  Parent score: {sc:.6f} (n={len(parent)})")

    # Try several initializations at a few lengths
    explore_lengths = [2048, 4096]
    if parent is not None and 512 <= len(parent) <= 8192:
        if len(parent) not in explore_lengths:
            explore_lengths.append(len(parent))
            explore_lengths.sort()

    for n in explore_lengths:
        if time.monotonic() > t1_end:
            break
        inits = [
            _init_uniform(n),
            _init_two_level(n, rng),
            _init_gaussian_mixture(n, rng),
            _init_cosine(n, rng),
            _init_block_random(n, rng),
        ]
        if parent is not None:
            sp = _resample(parent, n, rng)
            if sp is not None:
                inits.append(sp)

        for sp in inits:
            sc = _score(sp)
            _try_update(sp, sc)

    print(f"  Phase 1 done. Best: {best_score_overall:.6f}")

    # ========== PHASE 2: Main optimization (next 45%) ==========
    print("Phase 2: Main optimization")
    t2_end = start_time + 0.60 * budget

    if best_overall is None:
        best_overall = _init_uniform(2048)
        best_score_overall = _score(best_overall)

    # Run multiple optimization passes on the best, with different parameters
    n_main = len(best_overall)

    # Run 1: Standard gradient + freq
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(0.35 * (t2_end - start_time - 0.15 * budget), t_rem * 0.5, 300.0)
        t_alloc = max(20.0, t_alloc)
        p, sc = _optimize_hybrid(best_overall, rng, t_alloc, beta_start=30.0, beta_end=1500.0, verbose=True)
        _try_update(p, sc)

    # Run 2: Higher beta (sharper focus on max)
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(200.0, t_rem * 0.5)
        t_alloc = max(20.0, t_alloc)
        p1 = _resample(best_overall, n_main, rng)
        if p1 is None:
            p1 = best_overall.copy()
        p, sc = _optimize_hybrid(p1, rng, t_alloc, beta_start=100.0, beta_end=5000.0, verbose=True)
        _try_update(p, sc)

    # Run 3: Different length (upsample)
    t_rem = t2_end - time.monotonic()
    if t_rem > 60:
        n_up = min(8192, n_main * 2)
        if n_up != n_main:
            t_alloc = min(150.0, t_rem * 0.5)
            t_alloc = max(20.0, t_alloc)
            p_up = _resample(best_overall, n_up, rng)
            if p_up is not None:
                p, sc = _optimize_hybrid(p_up, rng, t_alloc, beta_start=50.0, beta_end=3000.0, verbose=True)
                _try_update(p, sc)

    # Run 4: Block refinement
    t_rem = t2_end - time.monotonic()
    if t_rem > 30:
        t_alloc = min(120.0, t_rem * 0.6)
        t_alloc = max(15.0, t_alloc)
        p, sc = _block_refine(best_overall, rng, t_alloc)
        _try_update(p, sc)

    print(f"  Phase 2 done. Best: {best_score_overall:.6f} (n={len(best_overall)})")

    # ========== PHASE 3: Diverse restarts (next 15%) ==========
    print("Phase 3: Diverse restarts")
    t3_end = start_time + 0.78 * budget

    t_rem = t3_end - time.monotonic()
    if t_rem > 60:
        # Try different shapes at the best length
        n_best = len(best_overall)
        restart_budget = max(20.0, t_rem * 0.3)

        for mode in range(5):
            if time.monotonic() > t3_end - 10:
                break
            if mode == 0:
                sp = _init_two_level(n_best, rng)
            elif mode == 1:
                sp = _init_gaussian_mixture(n_best, rng)
            elif mode == 2:
                sp = _init_cosine(n_best, rng)
            elif mode == 3:
                sp = _init_block_random(n_best, rng)
            else:
                sp = _resample(best_overall, n_best, rng)
                if sp is None:
                    sp = _init_uniform(n_best)

            t_alloc = min(60.0, restart_budget)
            p, sc = _optimize_hybrid(sp, rng, t_alloc, beta_start=50.0, beta_end=2000.0, verbose=False)
            _try_update(p, sc)

    print(f"  Phase 3 done. Best: {best_score_overall:.6f}")

    # ========== PHASE 4: Final polishing (last 15%) ==========
    print("Phase 4: Final polishing")

    if best_overall is not None:
        # Polish with high-beta gradient
        t_rem = deadline - time.monotonic()
        if t_rem > 30:
            t_alloc = min(0.4 * t_rem, 200.0)
            p, sc = _optimize_hybrid(best_overall, rng, t_alloc, beta_start=200.0, beta_end=8000.0, verbose=True)
            _try_update(p, sc)

        # Block refine
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.3 * t_rem, 100.0)
            p, sc = _block_refine(best_overall, rng, t_alloc)
            _try_update(p, sc)

        # One more gradient pass
        t_rem = deadline - time.monotonic()
        if t_rem > 20:
            t_alloc = min(0.4 * t_rem, 150.0)
            p, sc = _optimize_hybrid(best_overall, rng, t_alloc, beta_start=500.0, beta_end=10000.0, verbose=True)
            _try_update(p, sc)

    # ========== Final output ==========
    if best_overall is None:
        best_overall = _init_uniform(2048)

    # Verify with official evaluator if available
    p_final = _sanitize(best_overall)
    if p_final is None:
        p_final = _init_uniform(2048)

    final_score = _score(p_final)
    try:
        eval_score = float(evaluate_sequence([float(x) for x in p_final]))
        if np.isfinite(eval_score):
            final_score = min(final_score, eval_score)
    except Exception:
        pass

    print(f"\n{'=' * 60}")
    print(f"FINAL: score={final_score:.6f} (n={len(p_final)})")
    print(f"Total time: {time.monotonic() - start_time:.1f}s")

    out = np.where(np.isfinite(p_final), p_final, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = out.sum()
    if s <= 0 or not np.isfinite(s):
        out = _init_uniform(2048)
    else:
        out = out / s

    return [float(x) for x in out]
