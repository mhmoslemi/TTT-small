"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using two-timescale game playing
with FFT-accelerated QP inner loop, adaptive step sizes, and multi-scale
local search.
"""

import numpy as np
import time

def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(a, b):
    """Full convolution using FFT for large inputs."""
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.convolve(a, b)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout]

def _score_normalized(a):
    """Compute score for a normalized (sum=1) non-negative sequence."""
    n = len(a)
    c = _conv_full(a, a)
    if len(c) == 0:
        return float('inf')
    return 2.0 * n * np.max(c)

def _score_raw(a):
    """Compute score for any non-negative sequence (handles normalization)."""
    a = np.asarray(a, dtype=float)
    a = np.maximum(a, 0)
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        return float('inf')
    a = a / s
    return _score_normalized(a)

def _sanitize(a):
    """Ensure a is a valid non-negative normalized vector."""
    a = np.asarray(a, dtype=float)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        return None
    return a / s

def _bmat_vec(p, a, L):
    """Compute (B*p_vec) * a where B[i,j] = p[i+j].
    Uses FFT cross-correlation.
    p: length 2n-1, a: length n, L: FFT length >= 3n-2.
    Returns vector of length n.
    """
    x = np.zeros(L)
    x[:len(a)] = a
    y = np.zeros(L)
    y[:len(p)] = p
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    z = np.fft.irfft(X * np.conj(Y), L)[:len(a)]
    return z

def _game_play_optimize(a0, rng, time_limit, L_cache=None):
    """
    Two-timescale game playing optimization.
    Inner loop: solve QP min a^T B(p) a via multiplicative updates.
    Outer loop: update p = softmax(beta * C) where C = conv(a, a).
    """
    a = _sanitize(a0)
    if a is None:
        return None, float('inf')
    
    n = len(a)
    p_len = 2 * n - 1
    L = _next_pow2(3 * n - 2)  # For cross-correlation
    
    # Initialize p as uniform
    p = np.ones(p_len) / p_len
    
    a = a + 1e-18  # Ensure strictly positive
    a = a / a.sum()
    
    best_a = a.copy()
    best_score = _score_normalized(a)
    
    eta = 1.0  # Step size
    beta = 10.0  # Temperature for p update
    inner_steps = 20
    
    deadline = time.monotonic() + time_limit
    it = 0
    
    while time.monotonic() < deadline:
        # Inner loop: solve QP approximately
        for _ in range(inner_steps):
            z = _bmat_vec(p, a, L)
            z = np.maximum(z, 0)  # z should be >= 0 since p, a >= 0
            if np.max(z) < 1e-30:
                break
            # Adaptive step size
            eta = 1.0 / (np.max(z) + 1e-30)
            # Multiplicative update
            factor = np.exp(-eta * 2 * z)
            a = a * factor
            s = a.sum()
            if s <= 0 or not np.isfinite(s):
                break
            a = a / s
        
        # Outer loop: update p
        C = _conv_full(a, a)
        C = np.maximum(C, 0)
        C = np.where(np.isfinite(C), C, 0)
        
        # Smoothed max update for p
        shift = np.max(C)
        p = np.exp(beta * (C - shift))
        p = np.maximum(p, 1e-30)
        p = p / p.sum()
        
        # Track best
        sc = _score_normalized(a)
        if sc < best_score:
            best_score = sc
            best_a = a.copy()
        
        # An simulate annealing: increase beta over time
        it += 1
        if it % 10 == 0:
            beta = min(beta * 1.1, 500.0)
            inner_steps = min(inner_steps + 1, 50)
        
        # Periodic perturbation
        if it % 50 == 0 and rng.random() < 0.3:
            a_pert = _sanitize(a * np.exp(rng.normal(0, 0.1, n)))
            if a_pert is not None:
                a_pert_sc = _score_normalized(a_pert)
                if a_pert_sc < best_score * 1.1:
                    a = a_pert
    
    return best_a, best_score

I want: z[i] = sum_{j=0}^{n-1} p[i+j] * a[j] for i = 0, ..., n-1.

This is the cross-correlation of a and p: (a ⊛ p)[i] = sum_j a[j] * p[j+i].

Using FFT:
- Zero-pad a to length L: A_pad = [a, 0, 0, ..., 0] (length L)
- Zero-pad p to length L: P_pad = [p, 0, 0, ..., 0] (length L)
- Compute: Z = IFFT(FFT(A_pad) * conj(FFT(P_pad)))
- Then Z[i] = (a ⊛ p)[i] for i = 0, ..., n-1 (no wrap-around if L >= n + (2n-1) - 1 = 3n-2)

Let me verify: the i-th element of IFFT(FFT(x) * conj(FFT(y))) is:
sum_k x[k] * y[(k+i) mod L] (this is the circular cross-correlation)

If L >= n + (2n-1) - 1 = 3n-2, then for i = 0, ..., n-1 and k = 0, ..., n-1:
(k+i) mod L = k+i (since k+i < n+n = 2n < 3n-2 <= L)

And y[k+i] = p[k+i] (since k+i < 2n < 2n-1... wait, k can be up to n-1 and i up to n-1, so k+i up to 2n-2 = len(p)-1. OK so p[k+i] is valid.)
And x[k] = a[k] for k < n, 0 otherwise.

So Z[i] = sum_{k=0}^{n-1} a[k] * p[k+i] = sum_j a[j] * p[j+i] = z[i]. ✓

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using two-timescale fictitious play
with FFT-accelerated convex QP inner loop, adaptive step sizes, periodic
annealing, and multi-type local search. Key improvements over previous:
(1) multiple inner QP steps per outer p-update, (2) adaptive eta,
(3) two-timescale p tracking, (4) diverse perturbation restarts.
"""

import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 3000000:
        return np.convolve(a, b)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout]


def _score_norm(a):
    n = len(a)
    c = _conv_full(a, a)
    if len(c) == 0:
        return float('inf')
    return float(2.0 * n * np.max(c))


def _sanitize(a):
    a = np.asarray(a, dtype=float)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = a.sum()
    if s <= 0 or not np.isfinite(s):
        return None
    return a / s


def _bmat_vec(p, a, L):
    """Compute z[i] = sum_j p[i+j] * a[j] via FFT cross-correlation."""
    x = np.zeros(L)
    x[:len(a)] = a
    y = np.zeros(L)
    y[:len(p)] = p
    X = np.fft.rfft(x, L)
    Y = np.fft.rfft(y, L)
    z = np.fft.irfft(X * np.conj(Y), L)[:len(a)]
    return z


def _init_two_block(n, rng):
    """Asymmetric two-block initialization."""
    a = np.ones(n, dtype=float)
    split = int(n * rng.uniform(0.4, 0.6))
    ratio = float(rng.uniform(0.1, 0.5))
    a[split:] *= ratio
    a = _sanitize(a)
    # Smooth
    rad = max(1, n // 100)
    k = np.arange(-rad, rad + 1, dtype=float)
    kernel = np.exp(-0.5 * (k / (rad * 0.7)) ** 2)
    kernel /= kernel.sum()
    a = np.convolve(a, kernel, mode='same')
    return _sanitize(a)


def _init_bumps(n, rng):
    """Sum of random Gaussian bumps."""
    a = np.zeros(n, dtype=float)
    t = np.arange(n, dtype=float) / n
    K = int(rng.integers(2, 6))
    for _ in range(K):
        c = float(rng.uniform(0.05, 0.95))
        s = float(rng.uniform(0.03, 0.3))
        w = float(rng.uniform(0.5, 2.0))
        a += w * np.exp(-((t - c) / s) ** 2)
    a = _sanitize(a)
    return a


def _init_beta(n, rng):
    """Beta-distribution-like initialization."""
    t = (np.arange(n, dtype=float) + 0.5) / n
    a_p = float(rng.uniform(0.1, 3.0))
    b_p = float(rng.uniform(0.1, 3.0))
    a = t ** a_p * (1.0 - t) ** b_p
    a = _sanitize(a)
    return a


def _game_play(a0, rng, time_limit, beta_start=5.0, beta_end=3000.0):
    """
    Two-timescale fictitious play.
    Inner: multiplicative updates solving min a^T B(p) a.
    Outer: update p = softmax(beta * conv(a,a)).
    """
    a = _sanitize(a0)
    if a is None:
        return None, float('inf')
    if time_limit <= 0:
        return a, _score_norm(a)

    n = len(a)
    p_len = 2 * n - 1
    L = _next_pow2(3 * n - 2)

    a = a + 1e-15
    a = a / a.sum()

    p = np.ones(p_len) / p_len
    beta = float(beta_start)
    inner_steps = 10
    p_update_freq = 5

    best_a = a.copy()
    best_sc = _score_norm(a)

    deadline = time.monotonic() + time_limit
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        # Inner loop: QP gradient steps
        for _ in range(inner_steps):
            z = _bmat_vec(p, a, L)
            z = np.maximum(z, 0.0)
            zmax = np.max(z)
            if zmax < 1e-30:
                break
            eta = 1.0 / (zmax + 1e-30)
            # Try a couple of step sizes
            for mult in (0.5, 1.0, 2.0):
                e = eta * mult
                fac = np.exp(-e * 2.0 * z)
                a_new = a * fac
                s = a_new.sum()
                if s <= 0 or not np.isfinite(s):
                    continue
                a_new = a_new / s
                # Quick acceptance: check if objective decreased
                obj_old = float(np.dot(a, z))
                obj_new = float(np.dot(a_new, z))
                if obj_new < obj_old or mult == 2.0:
                    a = a_new
                    break

        # Outer: update p (every p_update_freq iterations)
        it += 1
        if it % p_update_freq == 0:
            C = _conv_full(a, a)
            C = np.maximum(C, 0.0)
            C = np.where(np.isfinite(C), C, 0.0)
            cmax = np.max(C)
            if cmax > 1e-30:
                p_new = np.exp(beta * (C - cmax))
                p_new = np.maximum(p_new, 1e-30)
                p_new = p_new / p_new.sum()
                # Fast p: use p_new directly; slow p: EMA
                p = 0.5 * p + 0.5 * p_new
                p = p / p.sum()

            # Track best
            sc = _score_norm(a)
            if sc < best_sc:
                best_sc = sc
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1

            # Adaptive beta
            beta = min(beta * 1.08, float(beta_end))
            inner_steps = min(10 + it // 50, 40)

            # Periodic restart with perturbation
            if no_improve > 100 and rng.random() < 0.5:
                std = min(0.3, 0.05 + 0.01 * (no_improve // 50))
                a_pert = _sanitize(a * np.exp(rng.normal(0, std, n)))
                if a_pert is not None:
                    a_pert = a_pert + 1e-15
                    a_pert = a_pert / a_pert.sum()
                    p = np.ones(p_len) / p_len
                    beta = max(beta_start, beta * 0.3)
                    no_improve = 0
                    a = a_pert

            # Hard restart from best
            if no_improve > 300:
                a = best_a + 1e-15
                a = a / a.sum()
                p = np.ones(p_len) / p_len
                beta = float(beta_start)
                no_improve = 0

    return best_a, best_sc


def _local_search(a0, rng, time_limit):
    """Local search with block scaling and pulse moves."""
    a = _sanitize(a0)
    if a is None:
        return None, float('inf')
    if time_limit <= 0:
        return a, _score_norm(a)

    n = len(a)
    best_a = a.copy()
    best_sc = _score_norm(a)
    cur_sc = best_sc

    deadline = time.monotonic() + time_limit
    stagnant = 0
    factors = (0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.7, 2.5)

    while time.monotonic() < deadline:
        r = rng.random()
        if r < 0.5:
            # Block scale
            w = int(rng.integers(1, max(2, min(n, n // 20))))
            start = int(rng.integers(0, n - w))
        elif r < 0.7:
            # Medium block
            w = int(rng.integers(max(2, n // 50), max(3, min(n, n // 10))))
            start = int(rng.integers(0, n - w))
        elif r < 0.85:
            # Pulse: add mass at random position
            w = 1
            start = int(rng.integers(0, n))
        else:
            # Spread: take from one position, add to neighbors
            w = int(rng.integers(2, max(3, n // 20)))
            start = int(rng.integers(0, n - w))

        w = max(1, min(n, w))
        start = max(0, min(n - w, start))

        improved = False
        for f in factors:
            candidate = a.copy()
            candidate[start:start + w] *= float(f)
            if r >= 0.85 and r < 0.95:
                # Pulse: add to one position
                candidate[start] += 0.1 * a[start]
            candidate = _sanitize(candidate)
            if candidate is None:
                continue
            sc = _score_norm(candidate)
            if sc < cur_sc - 1e-12:
                a = candidate
                cur_sc = sc
                improved = True
                if cur_sc < best_sc:
                    best_sc = cur_sc
                    best_a = a.copy()
                break

        if improved:
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 80:
                pert = _sanitize(a * np.exp(rng.normal(0, 0.05, n)))
                if pert is not None:
                    sc_p = _score_norm(pert)
                    if sc_p < best_sc * 1.05:
                        a = pert
                        cur_sc = sc_p
                    else:
                        a = best_a.copy()
                        cur_sc = best_sc
                stagnant = 0
            if stagnant > 300:
                a = best_a.copy()
                cur_sc = best_sc
                stagnant = 0

    return best_a, best_sc


def _resample(a, n_target):
    """Resample a normalized vector to a different length."""
    a = _sanitize(a)
    if a is None:
        return None
    n = len(a)
    if n == n_target:
        return a.copy()
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(n_target) + 0.5) / n_target
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0)
    return _sanitize(out)
