def lp_step(f, M):
    """
    Given f and bound M, solve:
    maximize sum(g)
    subject to (f*g)[k] <= M for k = 0, ..., 2n-2
               g >= 0
    """
    n = len(f)
    # Build the matrix A where A[k, j] = f[k-j] if 0 <= k-j < n, else 0
    # Then A @ g = f * g (the convolution)
    # Actually, (f*g)[k] = sum_j f[k-j] * g[j] (with bounds)
    # So A[k, j] = f[k-j] if 0 <= k-j < n
    
    # A is a (2n-1) x n matrix (Toeplitz-like)
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = f[idx]
    
    # Solve LP: max 1'g s.t. A@g <= M, g >= 0
    from scipy.optimize import linprog
    c_obj = -np.ones(n)  # minimize -sum(g)
    A_ub = A
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return -res.x  # g
    return None

import numpy as np

def _eval(f):
    return evaluate_sequence(f.tolist())


def _normalize(f):
    f = np.maximum(f, 0.0)
    s = f.sum()
    if s < 1e-10:
        return np.ones(len(f)) / len(f)
    return f / s


def _build_conv_structure(n):
    """Pre-compute row/col indices for the (2n-1) x n convolution matrix."""
    rows_list = []
    cols_list = []
    for k in range(2 * n - 1):
        j_start = max(0, k - n + 1)
        j_end = min(n - 1, k)
        if j_start <= j_end:
            rows_list.extend([k] * (j_end - j_start + 1))
            cols_list.extend(range(j_start, j_end + 1))
    rows = np.array(rows_list, dtype=np.int32)
    cols = np.array(cols_list, dtype=np.int32)
    idx = rows - cols  # f index for each entry
    return rows, cols, idx


def _make_sparse_matrix(f, rows, cols, idx):
    """Build CSR matrix A where A[k,j] = f[k-j] if valid."""
    n = len(f)
    data = f[idx].astype(np.float64)
    return scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_solve(f, M, rows, cols, idx, max_constraints=None):
    """Solve LP: max sum(g) s.t. A@f*g <= M, g >= 0."""
    n = len(f)
    A = _make_sparse_matrix(f, rows, cols, idx)
    
    if max_constraints is not None:
        # Only use the tightest constraints
        c = np.convolve(f, f)
        top_k = np.argsort(c)[::-1][:max_constraints]
        A = A[top_k, :]
        b_ub = np.full(len(top_k), M)
    else:
        b_ub = np.full(2 * n - 1, M)
    
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds,
                                  method='highs', options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    return None


def _soft_gradient_step(f, T, alpha):
    """One soft-max gradient descent step."""
    n = len(f)
    c = np.convolve(f, f)
    cs = c / T
    cs -= cs.max()
    p = np.exp(cs)
    ps = p.sum()
    if ps < 1e-300:
        return f, c
    p /= ps
    grad = np.correlate(p, f, mode='valid') / T
    f_new = f - alpha * grad
    f_new = np.maximum(f_new, 0.0)
    s = f_new.sum()
    if s < 1e-10:
        return f, c
    f_new /= s
    return f_new, c


def _spectral_method(f_init, n, n_iters=200):
    """Gerchberg-Saxton style: aim for flat |FFT(f)|."""
    f = f_init.copy()
    size = 1 << (2 * n - 1).bit_length()
    
    for it in range(n_iters):
        F = np.fft.rfft(f, size)
        # Target: flat magnitude proportional to |sinc|
        magnitudes = np.abs(F)
        # Use geometric mean as target
        log_mag = np.log(magnitudes + 1e-30)
        target_mag = np.exp(np.mean(log_mag[magnitudes > 1e-10]))
        # Blend toward target
        blend = 0.3
        new_mag = (1 - blend) * magnitudes + blend * target_mag
        # Keep phase
        phases = F / (magnitudes + 1e-30)
        F_new = new_mag * phases
        f_new = np.fft.irfft(F_new, size)[:n]
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s < 1e-10:
            break
        f_new /= s
        f = f_new
    
    return f


def _gradient_polish(f, deadline, n, label=""):
    """Gradient-based polish with annealing."""
    f = f.copy()
    c = np.convolve(f, f)
    peak = c.max()
    T = peak
    T_min = peak * 1e-10
    
    local_best = _eval(f)
    local_best_f = f.copy()
    
    n_iters = 20000
    start_time = time.monotonic()
    
    for it in range(n_iters):
        elapsed = time.monotonic() - start_time
        if time.monotonic() > deadline:
            break
        if elapsed > 0 and it > 100:
            avg_time = elapsed / it
            if avg_time * (n_iters - it) > (deadline - time.monotonic()) * 1.5:
                break
        
        progress = min(1.0, it / n_iters)
        T = peak * (T_min / peak) ** (progress ** 0.4)
        alpha = 0.05 * (0.001) ** progress
        
        f_new, c = _soft_gradient_step(f, T, alpha)
        c_check = np.convolve(f_new, f_new)
        new_peak = c_check.max()
        
        if new_peak < peak - 1e-14:
            f = f_new
            peak = new_peak
        
        if it % 2000 == 0 or it == n_iters - 1:
            score = _eval(f)
            if score < local_best:
                local_best = score
                local_best_f = f.copy()
                if label:
                    print(f"    {label} polish iter {it}: {score:.6f}")
    
    score = _eval(f)
    if score < local_best:
        local_best = score
        local_best_f = f.copy()
    
    return local_best_f, local_best

                        M = c.max()
                        t_mix = 0.1
                        no_improve = 0
        
        print(f"  LP done: {iter_count} iterations, best={best_score_local:.6f}")
        
        if best_score_local < best_score:
            best_score = best_score_local
            best_f = best_f_local.copy()
    
    # Phase 3: Gradient-based polish
    print(f"\nGradient polish (current best: {best_score:.6f})")
    polish_time = min(300.0, deadline - time.monotonic() - 60)
    if polish_time > 30:
        polish_dl = time.monotonic() + polish_time
        f_polished, sc_polished = _gradient_polish(best_f, polish_dl, n, label="main")
        if sc_polished < best_score:
            best_score = sc_polished
            best_f = f_polished.copy()
            print(f"  After gradient polish: {best_score:.6f}")
    
    # Phase 4: Coordinate polish
    coord_time = min(120.0, deadline - time.monotonic() - 20)
    if coord_time > 20:
        coord_dl = time.monotonic() + coord_time
        print(f"\nCoordinate polish (time={coord_time:.0f}s)")
        f_coord, sc_coord = _coordinate_polish(best_f, coord_dl)
        if sc_coord < best_score:
            best_score = sc_coord
            best_f = f_coord.copy()
            print(f"  After coordinate polish: {best_score:.6f}")
    
    # Phase 5: Final short gradient pass
    final_time = max(30.0, deadline - time.monotonic() - 10)
    if final_time > 30:
        final_dl = time.monotonic() + final_time
        f_final, sc_final = _gradient_polish(best_f, final_dl, n, label="final")
        if sc_final < best_score:
            best_score = sc_final
            best_f = f_final.copy()
            print(f"  After final polish: {best_score:.6f}")
    
    # Ensure valid output
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = _eval(best_f)
    if not np.isfinite(best_score) or final_score < best_score:
        best_score = final_score
    
    print(f"\n{'='*50}")
    print(f"FINAL SCORE: {best_score:.6f}")
    print(f"{'='*50}")
    
    return [float(x) for x in best_f]
