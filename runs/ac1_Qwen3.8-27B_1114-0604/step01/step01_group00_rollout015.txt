"""
Step function optimizer using adaptive mirror descent with hybrid gradient
(exact subgradient + soft-max blending), backtracking step size, and
coordinate-descent polish. Targets score < 1.5027 for n=6121.
"""

import numpy as np
import time

def _autocorr(f):
    """Full linear autocorrelation (convolution of f with itself) via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2*n-1]

def _score(f):
    """Compute the objective value."""
    n = len(f)
    c = _autocorr(f)
    return 2.0 * n * c.max() / (f.sum()**2)

def _gradient_hybrid(c, f, T):
    """
    Hybrid gradient: blend of exact subgradient at max lag
    and soft-max weighted average of all lags.
    Returns gradient vector of shape (n,).
    """
    n = len(f)
    k_max = int(np.argmax(c))
    
    # Exact subgradient at max lag
    # d(c[k_max])/d(f[i]) = 2*f[k_max - i] if 0 <= k_max - i < n, 
    #                        f[i] if 2*i == k_max
    grad_exact = np.zeros(n)
    for i in range(n):
        idx = k_max - i
        if 0 <= idx < n:
            if 2 * i == k_max:
                grad_exact[i] = f[idx]
            else:
                grad_exact[i] = 2.0 * f[idx]
    
    # Soft-max weighted gradient
    cs = c / T
    cs -= cs.max()
    p = np.exp(cs)
    p /= p.sum()
    
    grad_smooth = np.zeros(n)
    # For each lag k with weight p[k], the gradient contribution is:
    #   d(c[k])/d(f[i]) = 2*f[k-i] (or f[k-i] if 2i=k)
    # We can compute this efficiently:
    # grad_smooth[i] = sum_k p[k] * 2*f[k-i] * 1[0<=k-i<n]
    #                 = 2 * (p * f)_reversed valid part
    # More precisely: grad_smooth = 2 * correlate(p, f, 'valid') approximately
    
    # Use convolve: grad_smooth[i] = 2 * sum_k p[k] * f[k-i] * 1[0<=k-i<n]
    # = 2 * (p_flip * f)[n-1-i] ... this is getting complicated.
    # Let's just use the correlate approach.
    
    from scipy.signal import correlate
    grad_smooth = 2.0 * correlate(p, f, mode='valid')
    
    # Blend
    # Find how "peaked" the autocorrelation is
    # If the max is well-separated from the rest, use more exact gradient
    c_max = c[k_max]
    c_rest_max = max(c[:k_max].max() if k_max > 0 else 0, c[k_max+1:].max() if k_max < len(c)-1 else 0)
    gap = (c_max - c_rest_max) / (c_max + 1e-30)
    
    alpha = min(1.0, gap * 5)  # 0 to 1, higher when gap is large
    grad = alpha * grad_exact + (1 - alpha) * grad_smooth
    
    return grad, k_max

def _mirror_step(f, grad, eta):
    """Exponentiated gradient (mirror descent) step."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    f_new /= s
    return f_new

def _project_simplex(v):
    """Euclidean projection onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n+1) > (cssv - 1)
    rho = int(np.max(np.nonzero(cond)[0]))
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def _optimize(f0, n, deadline, rng, label=""):
    """
    Adaptive mirror descent optimization.
    Returns (best_f, best_score).
    """
    f = np.maximum(f0, 0.0).astype(np.float64)
    s = f.sum()
    f = f / s if s > 1e-10 else np.ones(n) / n
    
    c = _autocorr(f)
    peak0 = c.max()
    best_f = f.copy()
    best_score = _score(f)
    
    T = peak0
    T_min = peak0 * 1e-10
    base_eta = 1.0
    
    last_improve = 0
    stall_count = 0
    max_stall = 8
    
    it = 0
    while time.monotonic() < deadline:
        it += 1
        
        c = _autocorr(f)
        peak = c.max()
        
        # Adaptive temperature
        progress = min(1.0, it / 50000.0)
        T = peak0 * (T_min / peak0) ** (progress ** 0.5)
        # Also adapt based on current peak
        T = max(T, peak * 0.1)
        
        # Compute gradient
        grad, k_max = _gradient_hybrid(c, f, T)
        
        # Adaptive step size
        grad_norm = np.max(np.abs(grad))
        if grad_norm < 1e-15:
            break
        eta = base_eta / grad_norm
        
        # Try the step
        f_new = _mirror_step(f, grad, eta)
        c_new = _autocorr(f_new)
        peak_new = c_new.max()
        
        # Backtracking: reduce eta if no improvement
        improved = False
        while peak_new > peak * (1 - 1e-8) and eta > 1e-10:
            eta *= 0.5
            f_new = _mirror_step(f, grad, eta)
            c_new = _autocorr(f_new)
            peak_new = c_new.max()
            if peak_new < peak:
                improved = True
                break
        
        if improved or peak_new < peak:
            f = f_new
            c = c_new
            peak = peak_new
            score = 2.0 * n * peak / (f.sum()**2)
            if score < best_score:
                best_score = score
                best_f = f.copy()
                last_improve = it
                stall_count = 0
        else:
            stall_count += 1
            if stall_count >= max_stall:
                # Perturbation
                # Use structured perturbation based on autocorrelation
                k_star = int(np.argmax(c))
                # Reduce mass at positions that contribute to peak
                # and add to positions that don't
                contrib = np.zeros(n)
                for i in range(n):
                    idx = k_star - i
                    if 0 <= idx < n:
                        contrib[i] = f[i] * f[idx]
                
                # Move mass from high-contrib to low-contrib
                top_k = max(1, n // 100)
                bottom_k = max(1, n // 100)
                top_idx = np.argsort(contrib)[-top_k:]
                bot_idx = np.argsort(contrib)[:bottom_k]
                
                amt = f[top_idx].sum() * 0.05
                f[top_idx] -= f[top_idx] * 0.05
                f[bot_idx] += amt / len(bot_idx)
                f = np.maximum(f, 0)
                f /= f.sum()
                stall_count = 0
        
        # Periodic print
        if it % 10000 == 0:
            score = _score(f)
            print(f"  {label} iter {it}: score={score:.6f}, peak={peak:.8f}")
    
    # Final score
    score = _score(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score

def _polish(f0, n, deadline, rng):
    """
    Coordinate descent polish: iteratively try to reduce the peak autocorrelation
    by moving mass from high-contributing to low-contributing positions.
    """
    f = np.maximum(f0, 0.0).astype(np.float64).copy()
    f /= f.sum()
    c = _autocorr(f)
    peak = c.max()
    best_f = f.copy()
    best_score = _score(f)
    
    while time.monotonic() < deadline:
        c = _autocorr(f)
        peak = c.max()
        k_star = int(np.argmax(c))
        
        # Compute contribution of each position to the peak
        contrib = np.zeros(n)
        for i in range(n):
            idx = k_star - i
            if 0 <= idx < n:
                if 2 * i == k_star:
                    contrib[i] = f[i] * f[idx]
                else:
                    contrib[i] = 2.0 * f[i] * f[idx]
        
        # Find the position to reduce most
        i_donate = int(np.argmax(contrib))
        
        # Find good positions to receive (low contribution AND far from peak)
        # Positions far from the peak lag are safer to add mass
        dist_from_peak = np.abs(np.arange(n) - k_star)
        # Combine: low contribution + far from peak
        recv_score = contrib / (dist_from_peak + 1) + 0.01 * contrib
        i_receive = int(np.argmin(recv_score))
        
        if i_donate == i_receive:
            # Try random
            i_donate = int(np.argmax(contrib))
            i_receive = int(rng.integers(0, n))
            if i_donate == i_receive:
                continue
        
        # Try different transfer amounts
        best_frac = None
        best_peak_new = peak
        for frac in [0.01, 0.02, 0.05, 0.1, 0.2, 0.3, 0.5]:
            amt = f[i_donate] * frac
            f_try = f.copy()
            f_try[i_donate] -= amt
            f_try[i_receive] += amt
            if f_try[i_donate] < -1e-15:
                continue
            f_try = np.maximum(f_try, 0)
            f_try /= f_try.sum()
            c_try = _autocorr(f_try)
            if c_try.max() < best_peak_new:
                best_peak_new = c_try.max()
                best_frac = frac
        
        if best_frac is not None:
            amt = f[i_donate] * best_frac
            f[i_donate] -= amt
            f[i_receive] += amt
            f = np.maximum(f, 0)
            f /= f.sum()
            peak = best_peak_new
            score = 2.0 * n * peak / (f.sum()**2)
            if score < best_score:
                best_score = score
                best_f = f.copy()
        else:
            # No single-coordinate move helps. Try random pairs.
            moved = False
            for _ in range(20):
                i = int(rng.integers(0, n))
                j = int(rng.integers(0, n))
                if i == j:
                    continue
                for frac in [0.01, 0.05, 0.1]:
                    amt = f[i] * frac
                    f_try = f.copy()
                    f_try[i] -= amt
                    f_try[j] += amt
                    if f_try[i] < 0:
                        continue
                    f_try /= f_try.sum()
                    c_try = _autocorr(f_try)
                    if c_try.max() < peak - 1e-12:
                        f = f_try
                        peak = c_try.max()
                        moved = True
                        score = 2.0 * n * peak / (f.sum()**2)
                        if score < best_score:
                            best_score = score
                            best_f = f.copy()
                        break
                if moved:
                    break
        
        if not moved:
            # Stuck, try a larger perturbation
            n_perturb = max(10, n // 100)
            indices = rng.choice(n, size=n_perturb, replace=False)
            f[indices] *= 0.5
            f = np.maximum(f, 0)
            f /= f.sum()
    
    score = _score(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()
    
    return best_f, best_score
