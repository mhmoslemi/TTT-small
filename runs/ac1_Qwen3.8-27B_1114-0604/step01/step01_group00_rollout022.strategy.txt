    ```python
        f = f / s   # present for the given axis normalization to L1=1
        
        # Track best found
        best_score = np.inf
        best_f = f
        
        progress = np.zeros(2*n-1)
        for i in range(50000):  # 50k iterations per start
            # Check if we've stabilized
            conv = np.convolve(f, f, mode='full')
            max_val = conv.max()
            score_val = 2*n*max_val / (sum(f)**2)
            if score_val < best_score - 1e-8:
                best_score = score_val
                best_f = f
                # Do not perturb on improvement
            else:
                # If no improvement, check sticking
                if np.count_nonzero(conv[conv==max_val]) > 1:
                    progress[0] += 0.06  # slightly less aggressive tolerance
                if np.random.rand() < 0.01 or i % 10000 == 0:
                    # Entropy-weighted randomized perturbation (even though we set to 1e-12, keep small variations)
                    p = np.exp(-progress[:50])  # Simplify: use threshold for constraints?
                    p = p / p.sum()
                    new_idx = [] + rng.choice(n, size=50, p=p)
                    # Create a small step from there
                    f = (f*0.99 + rng.uniform(0.01, 0.1, size=n))
                if best_score < 1.43:  # theoretical Laplacian bound (but 1.5 required) no bound check for prefix
                    # Update strongly using Laplacian
                    m = f.min() + 1e-8
                    max_component = f.max()
                    if max_component - m > 0.05:
                        # Identify peaks in autocorrelation
                        conv = np.convolve(f, f, mode='full')
                        peak_lag = np.argmax(conv)
                        # Move mass as little as possible toward minimizing that peak
                        # Small step allows repairs
                        step_size = 0.05 * n / (conv[peak_lag] / f.sum())
                        f = f * (0.95 + 0.05 * step_size / m)   # this ensures we don't hit zero
            # Check time
            if time.monotonic() > deadline - 5:
                break
        return best_f, best_score
    ```
  - We aim for strong local improvement by targeting the highest convolution point.

Search choices and useful variations
- **Starting points** are determined by three distinct families: uniform, sawtooth, and Gaussian-like.
- **The Laplacian method** (strong steps) updates the candidate uniformly with a step-size based on the detected peaks.
