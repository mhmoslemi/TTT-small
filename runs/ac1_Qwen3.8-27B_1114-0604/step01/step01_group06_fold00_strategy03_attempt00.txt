    # Fallback scenario

# Convert the best_z to a complete sequence
best_sequence = np.zeros(n)
for i in range(4):
    start = i * block_length
    end = start + block_length
    # For the last block, if n is not exactly 4*block_length (but 4096 is exact)
    if end > n:
        end = n
    best_sequence[start:end] = best_z[i] * (n / (4*block_length))  # reconstruct the original un-normalized sequence with magnitude determined by the sum

# Defensive normalization after reconstruction
best_sum_val = np.sum(best_sequence)
if best_sum_val < 1e-6:
    best_sequence = np.ones(n, dtype=np.float64)
    best_sequence = best_sequence * (4 * (block_length / n))
    best_sum_val = np.sum(best_sequence)
    best_sequence = best_sequence / best_sum_val

# But note the objective function already handled normalization, and the final sequence we return must be of length n=4096
# We have completed the construction.

# Now return the sequence
return best_sequence.tolist()
