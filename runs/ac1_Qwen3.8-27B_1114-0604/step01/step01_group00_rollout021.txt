def _convolve_fft(f):
    """Compute full linear self-convolution via FFT. O(n log n)."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]

            f_try[i] -= amt
            f_try[jx] += amt
            f_try /= f_try.sum()
            c_try = _convolve_fft(f_try)
            new_peak = c_try.max()
            delta = new_peak - peak
            if delta < -1e-12:
                f = f_try
                peak = new_peak
                c = c_try
                score = _fast_eval(f)
                if score < best_score:
                    best_score = score
                    best_f = f.copy()
            elif sa_T > peak * 1e-8 and rng.random() < np.exp(-max(delta, 0) / sa_T):
                f = f_try
                peak = new_peak
                c = c_try

        # Strategy 3: Gaussian perturbation (occasional)
        if p % 20 == 10:
            mag = 0.002 / np.sqrt(n)
            f_try = f + mag * rng.standard_normal(n)
            f_try = np.maximum(f_try, 0)
            f_try /= f_try.sum()
            c_try = _convolve_fft(f_try)
            if c_try.max() < peak:
                f = f_try
                peak = c_try.max()
                c = c_try
                score = _fast_eval(f)
                if score < best_score:
                    best_score = score
                    best_f = f.copy()

    return best_f, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the non-negative sequence minimizing 2*n*max(f*f)/(sum f)^2.

    Multi-phase strategy:
    1. Generate diverse starts (structured + random).
    2. Fast-screen all starts using FFT-based evaluation.
    3. Run hybrid mirror descent from top starts with time allocation.
    4. Local search (pairwise swaps + SA) on the best result.
    5. Final optimization round from polished solution.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + budget_s - 15.0

    f_init = np.clip(np.asarray(height_sequence_1, dtype=np.float64), 0.0, 1000.0)
    n = len(f_init)
    print(f"Sequence length: {n}")

    # Generate starts
    starts = _generate_starts(n, f_init, rng)

    # Fast-screen all starts
    start_results = []
    for name, f0 in starts.items():
        f0 = np.maximum(f0, 0.0)
        s = f0.sum()
        if s < 0.01:
            continue
        f0 = f0 / s
        sc = _fast_eval(f0)
        start_results.append((sc, name, f0.copy()))

    start_results.sort(key=lambda x: x[0])

    print("\nStarting point scores (top 10):")
    for sc, name, _ in start_results[:10]:
        print(f"  {name}: {sc:.6f}")

    best_score = start_results[0][0]
    best_f = start_results[0][2].copy()

    # Verify with official evaluator
    official_score = _eval(best_f)
    if official_score < best_score:
        best_score = official_score
    print(f"Official score of best start: {best_score:.6f}")

    # Phase 2: Mirror descent from top starts
    top_k = min(5, len(start_results))
    remaining_time = deadline - time.monotonic()
    time_per_opt = remaining_time * 0.55 / top_k

    for idx in range(top_k):
        if time.monotonic() > deadline - 300:
            break

        sc, name, f0 = start_results[idx]
        t_alloc = time_per_opt * (1.0 - 0.08 * idx)
        opt_deadline = time.monotonic() + t_alloc
        print(f"\nOptimizing from '{name}' (score={sc:.6f}, budget={t_alloc:.0f}s)")

        f_opt, sc_opt = _mirror_descent(f0, n, opt_deadline, rng, label=name)

        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  New best: {best_score:.6f}")

    # Phase 3: Local search on best
    remaining = deadline - time.monotonic()
    ls_time = min(remaining * 0.3, 600.0)
    ls_deadline = time.monotonic() + ls_time
    print(f"\nLocal search (budget={ls_time:.0f}s), current best={best_score:.6f}")

    f_ls, sc_ls = _local_search(best_f, n, ls_deadline, rng, label="local")
    if sc_ls < best_score:
        best_score = sc_ls
        best_f = f_ls.copy()
        print(f"  After local search: {best_score:.6f}")

    # Phase 4: One more optimization round from the polished solution
    remaining = deadline - time.monotonic()
    if remaining > 60:
        final_time = min(remaining - 20, 300.0)
        final_deadline = time.monotonic() + final_time
        print(f"\nFinal optimization round (budget={final_time:.0f}s)")

        f_final, sc_final = _mirror_descent(best_f, n, final_deadline, rng, label="final")
        if sc_final < best_score:
            best_score = sc_final
            best_f = f_final.copy()
            print(f"  After final opt: {best_score:.6f}")

        # Quick local search again
        if time.monotonic() < deadline - 30:
            ls2_deadline = min(time.monotonic() + 60, deadline - 20)
            f_ls2, sc_ls2 = _local_search(best_f, n, ls2_deadline, rng, label="local2")
            if sc_ls2 < best_score:
                best_score = sc_ls2
                best_f = f_ls2.copy()
                print(f"  After final local: {best_score:.6f}")

    # Final verification with official evaluator
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = _eval(best_f)
    print(f"\nOfficial final score: {final_score:.6f}")
    if not np.isfinite(final_score) or final_score > best_score:
        # Safety: use the fast-eval best
        best_f = np.clip(best_f, 0.0, 1000.0)
        final_score = _eval(best_f)

    print(f"=== FINAL SCORE: {final_score:.6f} ===")
    return [float(x) for x in best_f]
