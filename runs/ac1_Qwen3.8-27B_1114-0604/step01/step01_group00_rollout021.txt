def _project_simplex(v):

"""
Step function optimizer minimizing 2*n*max(conv(f,f))/(sum f)^2.
Uses Nesterov-accelerated mirror descent with adaptive top-M peak focusing,
temperature annealing, backtracking line search, and a peak-repair local search.
"""

import numpy as np
from scipy.signal import correlate
import time


def _autocorr_fft(f):
    """Full linear autocorrelation via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u >= (cssv - 1.0) / rho
    rho_max = int(np.max(np.where(cond)[0]))
    theta = (cssv[rho_max] - 1.0) / (rho_max + 1.0)
    return np.maximum(v - theta, 0.0)


def _mirror_step(f, grad, eta):
    """Mirror descent step on simplex."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy()
    f_new /= s
    return f_new


def _compute_grad_topM(c, f, n, M, T):
    """Compute gradient with top-M peak focusing."""
    c_shifted = c - c.max()
    
    # Top-M softmax
    top_idx = np.argpartition(c, -M)[-M:]
    c_top = c_shifted[top_idx]
    p_top = np.exp(c_top / T)
    s = p_top.sum()
    if s < 1e-300:
        p_top = np.ones(M) / M
    else:
        p_top /= s
    
    # Background (small uniform component to avoid pure sparsity)
    p = np.full(2 * n - 1, 1.0 / (2 * n - 1))
    # Blend: 80% top-M, 20% uniform
    p[top_idx] = 0.2 * (1.0 / (2 * n - 1)) + 0.8 * p_top
    
    grad = 2.0 * correlate(p, f, mode='valid')
    return grad


def _score(c, n):
    """Score from autocorrelation (assumes sum(f)=1)."""
    return 2.0 * n * c.max()

    v = np.zeros(n)  # velocity

    M_end = 200

"""
Step function optimizer minimizing 2*n*max(conv(f,f))/(sum f)^2.
Uses Nesterov-accelerated mirror descent with adaptive top-M peak focusing,
temperature annealing, backtracking step size, gradient-guided peak repair,
and diverse structured starting points.

import numpy as np
from scipy.signal import correlate
import time


def _autocorr_fft(f):
    """Full linear autocorrelation via FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    if v.sum() < 1e-15:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.arange(1, n + 1)
    cond = u >= (cssv - 1.0) / rho
    rho_max = int(np.max(np.where(cond)[0]))
    theta = (cssv[rho_max] - 1.0) / (rho_max + 1.0)
    return np.maximum(v - theta, 0.0)


def _mirror_step(f, grad, eta):
    """Mirror descent step on the simplex (KL Bregman)."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f.copy()
    f_new /= s
    return f_new


def _grad_topM(c, f, n, M, T):
    """Gradient using top-M peaks with temperature T and uniform background."""
    c_shifted = c - c.max()
    if M >= 2 * n - 1:
        cs = c_shifted / max(T, 1e-15)
        p = np.exp(cs)
        p /= p.sum()
    else:
        top_idx = np.argpartition(c, -M)[-M:]
        c_top = c_shifted[top_idx]
        p_top = np.exp(c_top / max(T, 1e-15))
        s = p_top.sum()
        if s < 1e-300:
            p_top = np.ones(M) / M
        else:
            p_top /= s
        # Blend: 70% top-M focused, 30% uniform background
        p = np.full(2 * n - 1, 0.3 / (2 * n - 1))
        p[top_idx] += 0.7 * p_top
    grad = 2.0 * correlate(p, f, mode='valid')
    return grad

    return 2.0 * n * c.max()
