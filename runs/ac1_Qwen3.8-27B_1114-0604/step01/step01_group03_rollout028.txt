Now let me also think about a **spectral approach** for initialization:

I want to find a sequence a such that conv(a,a) is as flat as possible. In the frequency domain, this means |A(f)|^2 should be flat. So A(f) should have constant magnitude. A continuous signal with flat spectrum and compact support doesn't exist (by uncertainty principle), but we can approximate it. The "ideal" solution would have A(f) = C * exp(i*phi(f)) for some phase phi(f), with |A(f)| = C constant. The constraint that a is zero outside [0,n-1] constrains the phase.

This is related to the theory of "minimum phase" and "linear phase" signals.

    peaks = np.argsort(c)[::-1][:top_k]
    weights = c[peaks]
    wsum = weights.sum()
    g = np.zeros(n, dtype=float)
    for pk, w in zip(peaks, weights):
        g += (w / wsum) * _gradient_at_peak(a, pk, n)
    return g


def _nesterov_step(a, v, grad, eta, beta, n):
    """Take a Nesterov momentum step and project."""
    v_new = beta * v - eta * grad
    a_new = a + v_new
    a_new = _norm(a_new)
    v_new = a_new - a  # Reset velocity to actual displacement
    return a_new, v_new


def _ backtrack_line_search(a, c, eta, grad, n, L, max_iter=20):
    """Simple backtracking line search."""
    current_score = c.max()
    for i in range(max_iter):
        a_new = _norm(a - eta * grad)
        c_new = _conv(a_new, L)
        s_new = c_new.max()
        if s_new < current_score * (1 - 1e-4):
            return a_new, s_new, eta
        eta *= 0.5
    return a, current_score, eta

def _exact_polish(a, L, n, rng, time_limit, eta0=0.05, beta=0.85):

    it = 0

        exp_f = np.exp(np.clip(f_shifted, -50.0, 0.0))

        if s < 1e-300:
            a = _init_uniform(n)
            f = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            continue
        a = exp_f / s

            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 400:
            f = np.log(best_a + 1e-300)
            f -= f.max()
            v = np.zeros(n, dtype=np.float64)
            eta = float(eta0)

    except (TypeError, ValueError):
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 5.0:
        budget = max(10.0, budget if np.isfinite(budget) else 2400.0)

    t_start = time.monotonic()
    hard_deadline = t_start + budget - 3.0
    search_deadline = t_start + budget - 30.0

    rng = np.random.default_rng(seed)

    # Determine n values to try
    base_n = 6121
    try:
        parent_arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if len(parent_arr) > 100:
            base_n = len(parent_arr)
    except (NameError, Exception):
        parent_arr = None

    # Try a few n values
    n_list = [base_n]
    for dn in [-200, -100, 100, 200, 500]:
        nt = base_n + dn
        if 2000 <= nt <= 20000 and nt not in n_list:
            n_list.append(nt)

    # Sort by preference (base first)
    n_list = [base_n] + [x for x in n_list if x != base_n]

    all_results = []  # (score, n, array, L)

    for n in n_list:
        if time.monotonic() > search_deadline:
            break

        L = 1 << (2 * n - 1).bit_length()
        m = 2 * n - 1

        # Gather initial candidates
        inits = []

        # Parent
        if parent_arr is not None:
            pa = _resample(parent_arr, n)
            inits.append(pa)

        # Structured inits
        inits.append(_init_uniform(n))
        inits.append(_init_cosine_taper(n))
        for k in [1, 2, 3, 5]:
            inits.append(_init_two_bump(n, k, 0.7))
            inits.append(_init_two_bump(n, k, 0.95))
        for _ in range(4):
            inits.append(_init_multisine(n, rng, nfreq=rng.integers(2, 5)))
        for _ in range(3):
            inits.append(_init_random(n, rng))

        # Score all inits
        scored = []
        for a in inits:
            c = _fft_conv(a, L)
            sc = _score(c, n)
            scored.append((sc, a))
        scored.sort(key=lambda x: x[0])

        # Take top 5 initializations
        top_inits = [x[1] for x in scored[:5]]

        # Run soft exploration on the best inits
        time_per_n = max(5.0, min(600.0, (search_deadline - time.monotonic()) / max(1, len(n_list) - (n_list.index(n) if n in n_list else 0))))

        best_a = top_inits[0]
        best_sc = scored[0][0]

        # Soft exploration on top 3 inits
        n_explores = min(3, len(top_inits))
        for idx in range(n_explores):
            rem = search_deadline - time.monotonic()
            if rem < 10.0:
                break
            lim = min(120.0, rem / (n_explores - idx) * 0.7)
            if lim < 5.0:
                break
            a_res, sc_res = _soft_explore(top_inits[idx], L, n, rng, lim, temp0=2.0, temp1=150.0)
            if sc_res < best_sc:
                best_sc = sc_res
                best_a = a_res

        # Exact polish
        rem = search_deadline - time.monotonic()
        if rem > 20.0:
            lim = min(180.0, rem * 0.5)
            a_res, sc_res = _exact_polish(best_a, L, n, rng, lim, eta0=0.04, beta=0.85)
            if sc_res < best_sc:
                best_sc = sc_res
                best_a = a_res

        # Top-k polish
        rem = search_deadline - time.monotonic()
        if rem > 20.0:
            lim = min(120.0, rem * 0.5)
            a_res, sc_res = _topk_polish(best_a, L, n, rng, lim, top_k=3, eta0=0.03, beta=0.8)
            if sc_res < best_sc:
                best_sc = sc_res
                best_a = a_res

        all_results.append((best_sc, n, best_a, L))
        print(f"n={n}: best score so far = {best_sc:.6f} (time={time.monotonic()-t_start:.1f}s)")

    # Sort results
    all_results.sort(key=lambda x: x[0])
    global_best_sc = all_results[0][0]
    global_best_n = all_results[0][1]
    global_best_a = all_results[0][2]
    global_best_L = all_results[0][3]

    # Final polish phase on the best result
    rem = hard_deadline - time.monotonic()
    if rem > 30.0:
        # Long polish
        lim = min(rem * 0.6, 600.0)
        a_res, sc_res = _exact_polish(global_best_a, global_best_L, global_best_n, rng, lim, eta0=0.02, beta=0.9)
        if sc_res < global_best_sc:
            global_best_sc = sc_res
            global_best_a = a_res
        print(f"After long polish: {global_best_sc:.6f}")

    # Second method
    rem = hard_deadline - time.monotonic()
    if rem > 30.0:
        lim = min(rem * 0.5, 300.0)
        a_res, sc_res = _topk_polish(global_best_a, global_best_L, global_best_n, rng, lim, top_k=5, eta0=0.02, beta=0.85)
        if sc_res < global_best_sc:
            global_best_sc = sc_res
            global_best_a = a_res
        print(f"After topk polish: {global_best_sc:.6f}")

    # One more soft round for a different angle
    rem = hard_deadline - time.monotonic()
    if rem > 30.0:
        lim = min(rem * 0.6, 200.0)
        a_res, sc_res = _soft_explore(global_best_a, global_best_L, global_best_n, rng, lim, temp0=1.0, temp1=80.0)
        if sc_res < global_best_sc:
            global_best_sc = sc_res
            global_best_a = a_res
        print(f"After final soft: {global_best_sc:.6f}")

    # Final short polish
    rem = hard_deadline - time.monotonic()
    if rem > 15.0:
        lim = min(rem * 0.8, 100.0)
        a_res, sc_res = _exact_polish(global_best_a, global_best_L, global_best_n, rng, lim, eta0=0.01, beta=0.9)
        if sc_res < global_best_sc:
            global_best_sc = sc_res
            global_best_a = a_res

    # Evaluate top candidates with exact method
    candidates = [global_best_a]
    # Add a few diverse candidates
    for sc, nn, aa, LL in all_results[:5]:
        if aa is not None and len(aa) == nn:
            candidates.append(aa)

    best_seq = None
    best_exact = float('inf')

    for arr in candidates:
        if time.monotonic() > hard_deadline - 2.0:
            break
        if arr is None:
            continue
        seq = [float(x) for x in _norm(arr)]
        val = _exact_eval(seq)
        if np.isfinite(val) and val < best_exact:
            best_exact = val
            best_seq = seq
            print(f"Exact eval: {val:.6f}")

    if best_seq is None:
        best_seq = [float(x) for x in _norm(global_best_a)]

    print(f"Final answer: {best_exact:.6f} (n={len(best_seq)})")
    return best_seq
