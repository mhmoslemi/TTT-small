To get max < 1/n, we need to "steal" mass from the center peaks and distribute it to the shoulders.

The center position (k = n-1) has the most "room" for terms: it has n terms in the sum. The edges have only 1 term. So the center is naturally the hotspot.

To reduce the center, we need to make `sum_j a_j * a_{n-1-j}` small. This is the "center product" - the dot product of a with its reverse.

If a is "anti-symmetric" in some sense (high on one side, low on the other), the center product could be small. But since a >= 0, we can't have truly negative correlation.

Wait - `sum_j a_j * a_{n-1-j}` is minimized when a and its reverse are "anti-correlated." The reverse of a = [a0, a1, ..., a_{n-1}] is [a_{n-1}, ..., a1, a0]. The center product is `sum_j a_j * a_{n-1-j}`. - Fix a, solve for t: t = max_k (a*a)_k (trivial)
- Fix t, solve for a: min_a 0 subject to (a*a)_k <= t, sum(a) = 1, a >= 0 (non-convex)

def _fft_conv(a, L):
    """Full self-convolution via FFT, returns array of length 2n-1."""
    n = len(a)
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L, n=2 * n - 1)
    np.maximum(c, 0.0, out=c)
    return c

    """Compute 2*n*max(conv(a,a)) for normalized a."""
    n = len(a)
    return 2.0 * n * _fft_conv(a, L).max()


def _compute_grad(a, w, L):
    """
    Compute subgradient of weighted max-convolution.
    w: weights on the convolution domain (length 2n-1).
    Returns gradient vector of length n.
    g[j] = sum_k w[k] * a[k-j] (for valid k, k-j).
    Computed via: g = IFFT(FFT(w,L) * FFT(a_rev,L))[n-1 : 2n-1]
    """
    n = len(a)
    a_rev = a[::-1]
    W = np.fft.rfft(w, L)
    Ar = np.fft.rfft(a_rev, L)
    g = np.fft.irfft(W * Ar, L)[n - 1:2 * n - 1]
    np.maximum(g, 0.0, out=g)
    return g

    """Hard subgradient using top-k convolution positions."""
    n = len(a)
    m = len(c)

    w = np.zeros(m, dtype=np.float64)
    w[idx] = 1.0 / topk
    return _compute_grad(a, w, L), c[idx].max()

    """Soft subgradient using softmax weights."""
    n = len(a)
    m = len(c)
    vmax = c.max()
    z = beta * (c - vmax)
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = e.sum()
    if s < 1e-30:
        w = np.full(m, 1.0 / m)
    else:
        w = e / s

    while time.monotonic() < deadline:
        # Compute full conv once
        F = np.fft.rfft(a, L)
        c = np.fft.irfft(F * F, L, n=m)
        np.maximum(c, 0.0, out=c)
        current_max = c.max()

        improved = False
        for j in range(n):
            if time.monotonic() > deadline:
                break
            # Conv contribution of a[j] to position k:
            # a[j] appears in conv[k] as a[j]*a[k-j] (i=j) and a[k-j]*a[j] (k-i=j)
            # Change in conv[k] when a[j] -> t:
            #   if k == 2j: t^2 - a[j]^2
            #   elif 0 <= k-j < n: 2*(t-a[j])*a[k-j]
            #   else: 0

            aj = a[j]
            if aj < 1e-15:
                continue

            # Find the current max position
            kmax = int(np.argmax(c))

            # For the optimal t, we want to minimize max_k conv[k](t)
            # conv[k](t) is piecewise linear in t (except k=2j which is quadratic)
            # The optimal t is where the constraint "conv[kmax](t) = conv[second_max](t)"
            # or at a boundary (t=0 or t makes conv[kmax]=0)

            # Simple approach: bisect on t in [0, aj] (we expect to decrease a[j])
            # since reducing mass at hot positions helps.

            # Determine which positions are affected
            # Position k is affected if k-j in [0,n-1] or j-k in [0,n-1]
            # i.e., k in [j, j+n-1] (from i=j term) or k in [j, j+n-1] (from k-i=j term)
            # Actually: i=j gives a[j]*a[k-j], valid for k-j in [0,n-1], so k in [j, j+n-1]
            # k-i=j gives i=k-j, a[k-j]*a[j], valid for k-j in [0,n-1], so k in [j, j+n-1]
            # So affected positions: k in [j, j+n-1] intersect [0, 2n-2]
            # But also when k-i=j: i=k-j must be in [0,n-1], same constraint.
            # And i=j: k-j in [0,n-1], same.
            # Also the "cross" term: when i=j AND k-i=j means k=2j, the term is a[j]^2 (one term, not two)

            # For simplicity, just try a few candidate t values:
            # t = 0, t = aj * 0.9, t = aj * 0.7, t = aj * 0.5, t = aj * 0.3
            # Pick the best one.

            best_t = aj
            best_local_max = current_max

            # The key insight: reducing a[j] by delta reduces conv[k] by roughly
            # delta * (a[k-j] + a[j-k]) for k != 2j, and delta*(2*aj - delta + ...) for k=2j.
            # The position that benefits most is the one where a[j] contributes most.

            # Quick heuristic: try reducing a[j] proportionally to how much
            # it contributes to the peak.
            contrib = 0.0
            if 0 <= kmax - j < n:
                contrib += aj * a[kmax - j]
            if kmax != 2 * j and 0 <= kmax - j < n:
                contrib += a[kmax - j] * aj
            elif kmax == 2 * j:
                contrib = aj * aj

            # Try several reduction factors
            for frac in (0.85, 0.7, 0.5, 0.3):
                t = aj * frac
                # Compute new conv at the peak position
                new_aj = t
                # conv[kmax] change:
                delta = t - aj
                new_conv_kmax = c[kmax]
                if 0 <= kmax - j < n:
                    if kmax == 2 * j:
                        new_conv_kmax += delta * (t + aj)  # t^2 - aj^2
                    else:
                        new_conv_kmax += 2.0 * delta * a[kmax - j]
                # Also check if another position might become the new max
                # (approximation: just check the top few)
                if new_conv_kmax < best_local_max - 1e-15:
                    # This reduction helps at the peak; check it doesn't hurt elsewhere
                    # (rough check: just accept it)
                    best_local_max = new_conv_kmax
                    best_t = t

            if best_t < aj - 1e-15:
                a[j] = best_t
                improved = True
                improved_total += 1

        if not improved:
            break

        # Renormalize
        a = _normalize(a)
        sc = _fft_score(a, L)
        if sc < best_score:
            best_score = sc

    return a, best_score


def _optimize_main(a_init, L, time_budget, seed=42):
    """
    Main FFT-based subgradient optimization with adaptive parameters.
    """
    rng = np.random.default_rng(seed)
    n = len(a_init)
    a = _normalize(a_init.astype(np.float64).copy())

    best_a = a.copy()
    best_score = _fft_score(a, L)
    print(f"    Start score: {best_score:.8f}", flush=True)

    eta = 0.4
    beta_soft = 3.0
    momentum = np.zeros(n, dtype=np.float64)
    mu = 0.6
    it = 0
    stall = 0
    start_time = time.monotonic()
    deadline = start_time + time_budget

    # Schedule parameters
    total_iters_expected = max(1000, int(time_budget / 0.0015))

    while time.monotonic() < deadline:
        c = _fft_conv(a, L)
        sc = 2.0 * n * c.max()

        if sc < best_score:
            best_score = sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        # Reheat logic
        if stall > 400:
            # Perturb from best
            mix = 0.1 + 0.3 * rng.random()
            noise = rng.normal(0.0, 0.2, n)
            a = _normalize((1 - mix) * best_a + mix * np.exp(noise) * best_a)
            beta_soft = 2.0
            eta = 0.4
            momentum *= 0.3
            stall = 0
            continue

        # Progress-based schedule
        progress = min(1.0, float(it) / float(total_iters_expected))

        # Temperature: start soft, get harder
        beta_eff = 2.0 + 80.0 * progress * progress

        # Choose gradient type (cycle through)
        phase = it % 5
        if phase < 2:
            # Hard top-2
            g, _ = _gradient_hard(a, c, 2, L)
        elif phase < 4:
            # Soft gradient
            g, _ = _gradient_soft(a, c, beta_eff, L)
        else:
            # Hard top-5
            g, _ = _gradient_hard(a, c, 5, L)

        # Normalize gradient
        gmax = g.max()
        if gmax > 1e-15:
            g /= gmax
        else:
            break

        # Momentum
        momentum = mu * momentum + (1.0 - mu) * g

        # Line search with a few candidates
        best_local_sc = sc
        best_local_eta = eta
        for try_eta in (eta * 1.2, eta, eta * 0.5, eta * 0.25):
            if try_eta < 1e-6:
                continue
            a_try = a * np.exp(-try_eta * momentum)
            a_try = _normalize(a_try)
            sc_try = _fft_score(a_try, L)
            if sc_try < best_local_sc:
                best_local_sc = sc_try
                best_local_eta = try_eta

        # Apply best step
        a = a * np.exp(-best_local_eta * momentum)
        a = _normalize(a)

        # Adapt step size
        if best_local_sc < sc - 1e-14:
            eta = min(1.5, eta * 1.02)
        else:
            eta = max(0.005, eta * 0.9)

        # Adaptive momentum: increase as we converge
        mu = 0.5 + 0.15 * progress

        it += 1
        if it % 20000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"    iter={it}, score={best_score:.8f}, eta={eta:.4f}, "
                  f"beta={beta_eff:.1f}, elapsed={elapsed:.0f}s", flush=True)

    return best_a, best_score


def _try_different_n(n, L, time_budget, seed):
    """Try optimization with a given n and random init."""
    rng = np.random.default_rng(seed)
    # Shaped random initialization
    t = np.linspace(0, 1, n)
    z = rng.normal(0, 0.4, n)
    # Add some structure
    for k in range(1, 6):
        amp = rng.uniform(0.1, 0.5) / math.sqrt(k)
        ph = rng.uniform(0, 2 * np.pi)
        z += amp * np.cos(2 * np.pi * k * t + ph)
    a = _normalize(np.exp(z - z.max()))
    return _optimize_main(a, L, time_budget, seed)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing the score.
    Returns the best sequence found within the time budget.
    """
    try:
        budget = float(budget_s)
    except (TypeError, ValueError):
        budget = 2400.0
    if not np.isfinite(budget) or budget < 10:
        budget = 2400.0

    t_start = time.monotonic()
    hard_deadline = t_start + max(10.0, budget - 3.0)

    print(f"Budget: {budget}s, starting at {time.monotonic()}", flush=True)

    # --- Phase 1: Get starting point ---
    n_base = 6121
    L_base = 1 << (2 * n_base - 1).bit_length()

    a_base = None
    try:
        base = height_sequence_1
        arr = np.asarray(base, dtype=np.float64).ravel()
        if arr.size == n_base and np.all(np.isfinite(arr)):
            a_base = _normalize(arr)
            print(f"Loaded height_sequence_1, n={n_base}", flush=True)
    except (NameError, Exception):
        pass

    if a_base is None:
        # Generate a decent random start
        rng0 = np.random.default_rng(seed)
        n_base = 6121
        t = np.linspace(0, 1, n_base)
        z = rng0.normal(0, 0.5, n_base)
        for k in range(1, 8):
            z += (0.4 / math.sqrt(k)) * np.cos(2 * np.pi * k * t + rng0.uniform(0, 2 * np.pi))
        a_base = _normalize(np.exp(z - z.max()))
        print(f"Generated random init, n={n_base}", flush=True)

    # Quick score
    L_tmp = 1 << (2 * n_base - 1).bit_length()
    sc0 = _fft_score(a_base, L_tmp)
    print(f"Initial score for n={n_base}: {sc0:.8f}", flush=True)

    # --- Phase 2: Try alternative n values quickly ---
    # Also try n=8192 (same FFT length as 6121 since both need L=16384)
    n_alt = 8192
    L_alt = 1 << (2 * n_alt - 1).bit_length()

    results = []

    # Quick trial on n_alt
    time_for_alt = min(120.0, 0.05 * budget)
    if budget > 300:
        print(f"Trying n={n_alt} for {time_for_alt:.0f}s...", flush=True)
        t_mark = time.monotonic()
        a_alt, sc_alt = _try_different_n(n_alt, L_alt, time_for_alt, seed + 1)
        elapsed_alt = time.monotonic() - t_mark
        print(f"  n={n_alt} result: {sc_alt:.8f} (took {elapsed_alt:.0f}s)", flush=True)
        results.append((sc_alt, n_alt, a_alt, L_alt))

    # Always include the base
    results.insert(0, (sc0, n_base, a_base.copy(), L_tmp))

    # --- Phase 3: Extended optimization on the best candidate ---
    results.sort(key=lambda x: x[0])
    best_sc, best_n, best_a, best_L = results[0]
    print(f"\nBest so far: n={best_n}, score={best_sc:.8f}", flush=True)

    elapsed = time.monotonic() - t_start
    remaining = max(5.0, budget - 60.0 - elapsed)
    print(f"Extended optimization: n={best_n}, budget={remaining:.0f}s", flush=True)

    best_a, best_sc = _optimize_main(best_a, best_L, remaining, seed + 100)
    print(f"After extended: score={best_sc:.8f}", flush=True)

    # --- Phase 4: Coordinate descent polish ---
    elapsed = time.monotonic() - t_start
    polish_time = max(5.0, min(300.0, budget - 30.0 - elapsed))
    if polish_time > 10:
        print(f"Coordinate polish: {polish_time:.0f}s", flush=True)
        best_a, best_sc = _coordinate_polish(best_a, best_L, polish_time)
        print(f"After polish: score={best_sc:.8f}", flush=True)

    # --- Phase 5: Final verification ---
    # Convert to list and verify with evaluate_sequence
    seq = [float(x) for x in best_a]
    try:
        exact = evaluate_sequence(seq)
        print(f"Exact evaluation: {exact:.8f}", flush=True)
        if np.isfinite(exact) and exact < best_sc:
            best_sc = exact
    except (NameError, Exception) as e:
        print(f"Could not verify: {e}", flush=True)

    # Check time
    elapsed = time.monotonic() - t_start
    print(f"Total time used: {elapsed:.1f}s / {budget}s", flush=True)

    # Ensure all values are valid (non-negative, finite)
    for i in range(len(seq)):
        if not np.isfinite(seq[i]) or seq[i] < 0:
            seq[i] = 0.0

    return seq
