def _fft_conv(a, N):
    """Compute linear convolution a*a using FFT."""
    n = len(a)
    L = 2 * n - 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L]
    return np.maximum(c, 0.0)


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

    v_v = beta2 * v_v + (1 - beta2) * g * g
    m_hat = m_v / (1 - beta1 ** t)
    v_hat = v_v / (1 - beta2 ** t)
    z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z, m_v, v_v


def _optimize_start(z0, n, N, deadline, rng, tau_max=1.0, tau_min=0.01,
                    n_steps=20000, lr_init=0.01):
    """Run Adam optimization from a single start with tau annealing."""
    z = z0.copy()
    z = np.clip(z, -30, 30)
    
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    t = 0
    
    best_score = np.inf
    best_z = z.copy()
    
    # Compute initial convolution to set tau
    p = _softmax(z)
    c = _fft_conv(p, N)
    c_max = c.max()
    c_mean = c.mean()
    
    # Tau schedule: geometric from c_max*0.3 to c_max*0.005
    tau_start = max(c_max * 0.3, 1e-6)
    tau_end = max(c_max * 0.005, 1e-8)
    
    # LR schedule: cosine decay
    for step in range(n_steps):
        if time.monotonic() > deadline:
            break
        
        t += 1
        frac = step / max(n_steps - 1, 1)
        
        # Cosine LR decay
        lr = lr_init * 0.5 * (1 + np.cos(np.pi * frac))
        lr = max(lr, 1e-6)
        
        # Tau annealing (geometric)
        tau = tau_start * (tau_end / tau_start) ** frac
        tau = max(tau, 1e-8)
        
        # Compute p and convolution
        p = _softmax(z)
        c = _fft_conv(p, N)
        
        # Smooth max: tau * log(sum(exp((c - m)/tau))) + m
        m = c.max()
        diff = (c - m) / tau
        diff = np.clip(diff, -500, 0)
        e = np.exp(diff)
        se = e.sum()
        if se <= 0 or not np.isfinite(se):
            break
        obj = m + tau * np.log(se)
        
        # Score
        score = 2.0 * n * obj
        if score < best_score:
            best_score = score
            best_z = z.copy()
        
        # Gradient of smooth max w.r.t. c
        w = e / se  # weights on each c[k]
        
        # Gradient of c w.r.t. p: c = conv(p,p), dc/dp[i] = 2*conv Muito
        # Actually: c[k] = sum_j p[j]*p[k-j], so dc[k]/dp[i] = p[k-i] + p[i-k]
        # (with appropriate bounds). This is equivalently 2*conv(p, p_reversed_shifted)
        # More efficiently: the gradient of conv(p,p) w.r.t. p is 2*corr(p, p) 
        # which in FFT is: ifft(fft(w_conv) * conj(fft(p))) where w_conv is padded
        # Actually: grad_p = 2 * ifft(fft(w, N) * conj(fft(p, N)))[:n]
        # Wait, that's not quite right either. Let me think...
        # 
        # c[k] = sum_j p[j] p[k-j]
        # dc[k]/dp[i] = p[k-i] + (1 if i==k-i else 0)*... 
        # Actually no: dc[k]/dp[i] = p[k-i] + p[i-k] (with p[j]=0 for j out of range)
        # = p[k-i] + p[|i-k|]... 
        # 
        # More carefully: c[k] = sum_{j: 0<=j<n, 0<=k-j<n} p[j] p[k-j]
        # dc[k]/dp[i] = p[k-i] (if 0<=k-i<n) + p[i-k] (if 0<=i-k<n, and i != k-i i.e. 2i != k)
        # Hmm, this is the "autocorrelation" gradient.
        # 
        # Actually the simplest way: the gradient of (1/2)*||conv(p,p) - c||_w^2 ...
        # No, let's just use the chain rule properly.
        #
        # We want grad_p of F = tau * log(sum_k exp(c[k]/tau))
        # = sum_k w[k] * dc[k]/dp
        # where w[k] = exp(c[k]/tau) / sum exp(c[j]/tau)
        #
        # dc[k]/dp[i] = p[k-i] + p[i-k] (with p[j]=0 outside [0,n-1])
        # This is the "symmetric correlation" of p with itself, evaluated at lag (k-i).
        #
        # So: grad_p[i] = sum_k w[k] * (p[k-i] + p[i-k])
        # = sum_k w[k] * p[k-i] + sum_k w[k] * p[i-k]
        # = cross_corr(w, p)[i] + cross_corr(w, p)[i]  (with appropriate definitions)
        #
        # In FFT: if W = fft(w), P = fft(p), then
        # cross_corr(w, p)[i] = ifft(W * conj(P))[i] (circular, but we zero-pad)
        #
        # Actually: conv(w, p_reversed)[i] = sum_j w[j] p_reversed[i-j] = sum_j w[j] p[-(i-j)]
        # Hmm, let me just use the fact that:
        # sum_k w[k] p[k-i] = (w convolved with reversed p)[i] ... 
        #
        # Simplest: grad_p = 2 * ifft(fft(w, N) * conj(fft(p, N)))[:n]
        # Let me verify: ifft(fft(w) * conj(fft(p)))[i] = sum_k w[k] p[(k-i) mod N]
        # For zero-padded vectors (N >= 2n-1), this is the circular cross-correlation
        # which equals the linear cross-correlation for |i| < n.
        #
        # So grad_p[i] = sum_k w[k] * (p[k-i] + p[i-k]) 
        # Hmm, p[k-i] for k-i in [0,n-1] and p[i-k] for i-k in [0,n-1]
        # = sum_k w[k] * p[k-i] + sum_k w[k] * p[i-k]
        # First term: (w cross p) at lag i = ifft(fft(w)*conj(fft(p)))[i] (circular)
        # Second term: (w cross p) at lag -i = ifft(fft(w)*conj(fft(p)))[N-i] 
        # But by symmetry of our problem (w and p are real), these are equal!
        # So grad_p[i] = 2 * ifft(fft(w, N) * conj(fft(p, N)))[:n][i]
        
        W = np.fft.rfft(w, N)
        P = np.fft.rfft(p, N)
        grad_p = np.fft.irfft(W * np.conjugate(P), N)[:n].real * 2.0
        
        # Gradient w.r.t. z: p = softmax(z), dp/dz[i] = p[i](delta_ij - p[j])
        # grad_z[j] = sum_i grad_p[i] * p[i] * (delta_ij - p[j])
        # = grad_p[j] * p[j] - p[j] * sum_i grad_p[i] * p[i]
        s = np.dot(grad_p, p)
        grad_z = p * (grad_p - s)
        
        # Gradient clipping
        gnorm = np.linalg.norm(grad_z)
        if gnorm > 1.0:
            grad_z = grad_z / gnorm
        
        # Adam step
        z, m_v, v_v = _adam_step(z, grad_z, m_v, v_v, t, lr)
        z = np.clip(z, -30, 30)
        
        if step % 5000 == 0:
            print(f"  step={step} score={2*n*c.max():.6f} smooth={2*n*obj:.6f} "
                  f"tau={tau:.6f} lr={lr:.6f}")
    
    # Final evaluation with exact max
    p_best = _softmax(best_z)
    c_best = _fft_conv(p_best, N)
    best_score = 2.0 * n * c_best.max()
    
    return p_best, best_score, best_z


def _polish(p0, n, N, deadline):
    """Refine using exact max with active-set gradient and line search."""
    p = p0.copy()
    p = np.maximum(p, 0)
    s = p.sum()
    if s <= 0:
        return p0
    p = p / s
    
    z = np.log(np.maximum(p, 1e-30))
    z = z - z.mean()
    z = np.clip(z, -30, 30)
    
    best_p = p.copy()
    best_score = 2.0 * n * _fft_conv(p, N).max()
    
    rng = np.random.default_rng(12345)
    no_improve = 0
    step = 0
    
    while time.monotonic() < deadline:
        c = _fft_conv(p, N)
        c_max = c.max()
        score = 2.0 * n * c_max
        
        if score < best_score - 1e-12:
            best_score = score
            best_p = p.copy()
            no_improve = 0
        elif no_improve > 500:
            break
        
        # Find active set (top values)
        threshold = c_max * 0.999
        active = c >= threshold
        w = np.zeros_like(c)
        w[active] = 1.0 / active.sum()
        
        # Gradient
        W = np.fft.rfft(w, N)
        P = np.fft.rfft(p, N)
        grad_p = np.fft.irfft(W * np.conjugate(P), N)[:n].real * 2.0
        
        s_g = np.dot(grad_p, p)
        grad_z = p * (grad_p - s_g)
        
        # Line search
        accepted = False
        for eta in [0.001, 0.003, 0.01, 0.03, 0.1, 0.3, 1.0, 3.0]:
            z_t = np.clip(z - eta * grad_z, -30, 30)
            p_t = _softmax(z_t)
            score_t = 2.0 * n * _fft_conv(p_t, N).max()
            if score_t < score - 1e-12:
                z = z_t
                p = p_t
                accepted = True
                break
        
        if not accepted:
            # Small random perturbation
            for scale in [0.01, 0.05, 0.1, 0.3]:
                z_t = np.clip(z + scale * rng.standard_normal(n), -30, 30)
                p_t = _softmax(z_t)
                score_t = 2.0 * n * _fft_conv(p_t, N).max()
                if score_t < best_score - 1e-10:
                    z = z_t
                    p = p_t
                    no_improve = 0
                    accepted = True
                    break
            if not accepted:
                no_improve += 1
        
        step += 1
        if step % 200 == 0:
            print(f"  polish step={step} score={score:.6f} best={best_score:.6f}")
    
    return best_p


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2n*max(conv(a,a))/sum(a)^2.
    Uses softmax parameterization, Adam with tau annealing, multi-start,
    and exact polishing.
    """
    budget_s = max(120.0, float(budget_s))
    t0 = time.monotonic()
    deadline = t0 + budget_s - 10.0
    
    rng = np.random.default_rng(int(seed))
    
    best_score = np.inf
    best_a = None
    
    # Get previous sequence if available
    prev_seq = None
    try:
        g = globals().get('height_sequence_1')
        if g is not None and len(g) > 10:
            prev_seq = np.asarray(g, dtype=np.float64)
            prev_seq = np.clip(prev_seq, 0, None)
            if prev_seq.sum() > 0:
                prev_seq = prev_seq / prev_seq.sum()
            else:
                prev_seq = None
    except Exception:
        prev_seq = None
    
    # Try different sequence lengths
    # Start with moderate size for quick results, then try larger
    configs = [
        (4096, 0.5),   # (n, time_fraction)
        (8192, 0.3),
        (16384, 0.2),
    ]
    
    # If we have a previous sequence, try its length first
    if prev_seq is not None:
        prev_n = len(prev_seq)
        # Find closest power of 2
        configs.insert(0, (min(max(prev_n, 1024), 32768), 0.4))
    
    time_used = 0.0
    total_time = budget_s - 15.0
    
    for n, frac in configs:
        time_alloc = total_time * frac
        alloc_deadline = time.monotonic() + time_alloc
        
        if time.monotonic() > alloc_deadline - 60:
            continue
        
        N = 1 << (3 * n - 2).bit_length()
        N = min(N, 1 << 24)  # Cap FFT size
        
        print(f"\n=== n={n}, N={N}, time_alloc={time_alloc:.0f}s ===")
        
        # Generate initializations
        inits = _make_initializations(n, rng, prev_seq)
        
        # Score all inits and sort
        scored_inits = []
        for z in inits:
            p = _softmax(np.clip(z, -30, 30))
            c = _fft_conv(p, N)
            sc = 2.0 * n * c.max()
            scored_inits.append((sc, z))
        
        scored_inits.sort(key=lambda x: x[0])
        scored_inits = scored_inits[:12]  # Take top 12
        
        # Optimize each start
        n_starts = min(len(scored_inits), 8)
        time_per_start = time_alloc / max(n_starts, 1)
        
        for idx in range(n_starts):
            start_deadline = time.monotonic() + time_per_start - 5
            if time.monotonic() > start_deadline:
                break
            
            sc0, z0 = scored_inits[idx]
            print(f"  Start {idx}: initial_score={sc0:.6f}")
            
            # Determine steps based on time
            est_step_time = 0.005 * (n / 4096)  # rough estimate
            n_steps = min(int((time_per_start - 10) / max(est_step_time, 0.001)), 30000)
            n_steps = max(n_steps, 2000)
            
            try:
                p_opt, sc_opt, z_opt = _optimize_start(
                    z0, n, N, start_deadline, rng,
                    n_steps=n_steps, lr_init=0.02
                )
            except Exception as e:
                print(f"  Error in optimization: {e}")
                continue
            
            if sc_opt < best_score:
                best_score = sc_opt
                best_a = p_opt.copy()
                print(f"  ** New best: score={best_score:.9f} (start {idx}, n={n})")
            
            # Polish the best result
            if time.monotonic() < deadline - 60:
                polish_deadline = min(time.monotonic() + 120, deadline - 30)
                try:
                    p_pol = _polish(p_opt, n, N, polish_deadline)
                    sc_pol = 2.0 * n * _fft_conv(p_pol, N).max()
                    if sc_pol < best_score:
                        best_score = sc_pol
                        best_a = p_pol.copy()
                        print(f"  ** Polished best: score={best_score:.9f}")
                except Exception as e:
                    print(f"  Polish error: {e}")
        
        time_used += time.monotonic() - (t0 + total_time * sum(f for _, f in configs[:configs.index((n, frac))])))
        print(f"  Elapsed: {time.monotonic() - t0:.0f}s, best so far: {best_score:.9f}")
    
    # Final polish on the overall best
    if best_a is not None:
        n_best = len(best_a)
        N_best = 1 << (3 * n_best - 2).bit_length()
        N_best = min(N_best, 1 << 24)
        polish_deadline = deadline
        try:
            p_final = _polish(best_a, n_best, N_best, polish_deadline)
            sc_final = 2.0 * n_best * _fft_conv(p_final, N_best).max()
            if sc_final < best_score:
                best_score = sc_final
                best_a = p_final
                print(f"  ** Final polish: score={best_score:.9f}")
        except Exception:
            pass
    
    # Validate with official evaluator
    if best_a is not None:
        try:
            best_a = np.asarray(best_a, dtype=np.float64)
            best_a = np.clip(best_a, 0, 1000)
            s = best_a.sum()
            if s < 0.01:
                best_a = np.ones(len(best_a)) / len(best_a)
            exact = evaluate_sequence(best_a.tolist())
            print(f"\nOfficial evaluator score: {exact:.9f}")
        except Exception as e:
            print(f"Evaluation error: {e}")
    
    if best_a is None:
        best_a = np.ones(4096) / 4096
    
    # Final cleanup
    best_a = np.asarray(best_a, dtype=np.float64)
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.maximum(best_a, 0.0)
    s = best_a.sum()
    if s < 0.01:
        best_a = np.ones(4096) / 4096
    else:
        best_a = best_a / s
    
    print(f"\nFinal: n={len(best_a)}, score≈{2*len(best_a)*_fft_conv(best_a, 1<<(3*len(best_a)-2).bit_length()).max():.9f}")
    print(f"Total time: {time.monotonic() - t0:.1f}s")
    
    return [float(v) for v in best_a.tolist()]
