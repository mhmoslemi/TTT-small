```python
"""
Minimize 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences a.
Uses a combination of:
  1. Direct subgradient descent on the true max (top-K peaks)
  2. Spectral domain flattening (Fienup-style)
  3. Nesterov momentum acceleration
  4. Adaptive step sizes with backtracking
"""

import numpy as np
import time
import math

def _norm1(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        a = np.ones_like(a) / len(a)
        return a
    return a / s

def _fft_conv(a, L):
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)
    return c[:2*len(a)-1]

def _score_norm(a, L, n):
    """Score assuming sum(a)=1: 2*n*max(conv(a,a))."""
    c = _fft_conv(a, L)
    return float(2.0 * n * np.max(c))

def _topk_peaks(c, K):
    """Return indices of top-K values in c."""
    if K >= len(c):
        return np.arange(len(c))
    return np.argpartition(-c, K)[:K]

def _subgradient_dir(a, L, n, K=5):
    """Compute subgradient direction using top-K peaks."""
    c = _fft_conv(a, L)
    peaks = _topk_peaks(c, K)
    g = np.zeros(n)
    for j in peaks:
        j = int(j)
        # g[k] += a[j-k] for valid k
        lo = max(0, j - n + 1)
        hi = min(n - 1, j)
        for k in range(lo, hi + 1):
            g[k] += a[j - k]
    g /= K
    return g, c

def _subgrad_step(a, L, n, lr, K=5):
    """One subgradient step with line search."""
    a = _norm1(a)
    c = _fft_conv(a, L)
    current_score = float(2.0 * n * np.max(c))
    
    peaks = _topk_peaks(c, K)
    g = np.zeros(n)
    for j in peaks:
        j = int(j)
        lo = max(0, j - n + 1)
        hi = min(n - 1, j)
        for k in range(lo, hi + 1):
            g[k] += a[j - k]
    g /= K
    
    # Normalize gradient
    gn = np.linalg.norm(g)
    if gn < 1e-300:
        return a, current_score
    
    # Try a few step sizes
    best_a = a
    best_score = current_score
    for step in (lr, lr*0.5, lr*0.25, lr*0.1):
        a_new = a - step * g
        a_new = _norm1(a_new)
        c_new = _fft_conv(a_new, L)
        sc = float(2.0 * n * np.max(c_new))
        if sc < best_score:
            best_score = sc
            best_a = a_new
    return best_a, best_score
```

Minimize the peak autocorrelation ratio 2*n*max(conv(a,a))/(sum(a))^2
using a combination of proximal gradient (soft log-sum-exp), Nesterov momentum,
spectral flattening (Fienup), and direct subgradient polish.

import numpy as np
import time
import math

def _norm1(a):
    """Normalize to non-negative, sum=1."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones(len(a)) / len(a)
    return a / s

def _fft_conv(a, L):
    """Linear autocorrelation via FFT."""
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)
    return c[:2*len(a)-1]

def _score_fast(a, c, n):
    """Score from precomputed convolution."""
    return float(2.0 * n * np.max(c))

def _soft_gradient(c, a, L, n, tau):
    """Compute soft gradient using log-sum-exp weights.
    
    Returns (gradient, smoothed_max_value)
    """
    # v = n * c (scaled for temperature)
    v = n * c
    vmax = float(np.max(v))
    z = tau * (v - vmax)
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if s < 1e-300:
        w = np.ones(len(c)) / len(c)
    else:
        w = e / s
    J = float(vmax / n + math.log(s) / (tau * n))  # scaled log-sum-exp
    
    # Gradient: g[k] = sum_m w[m] * a[m-k] (for valid m-k)
    # Use FFT trick: g = ifft(fft(a_reversed) * fft(w))[offsets]
    # a_reversed = a[n-1-i]
    # g[k] = sum_m w[m] * a[m-k] = sum_m w[m] * a_rev[n-1-m+k]
    # Let's use the convolution approach:
    # g = conv(w, a) restricted to valid range
    # Actually: g[k] = sum_m w[m] * a[m-k]
    # This is (w * a_rev)[n-1-k] ... let me just do it directly with FFT

    # = sum_m w[m] * a[m-k]
    # Let b[i] = a[n-1-i] (reversed a)
    # g[k] = sum_m w[m] * b[n-1-m+k]
    # Let j = n-1-m+k, so m = n-1-k+j
    # g[k] = sum_j b[j] * w[n-1-k+j]
    # This is (w * b)[n-1-k] = conv(w, b)[n-1-k]
    
    # Using FFT: conv(w, b) = ifft(fft(w) * fft(b))
    W = np.fft.rfft(w, L)
    B = np.fft.rfft(a[::-1], L)
    g_full = np.fft.irfft(W * B, L)
    # g[k] = g_full[n-1-k] for k in [0, n-1]
    g = g_full[n-1 : n-1-n : -1]  # g_full[n-1], g_full[n-2], ..., g_full[0]
    
    return g, J

def _fienup_step(a, L, n, beta=0.5):

    F = np.fft.rfft(a, L)
    mag = np.abs(F).copy()

    target_mag = np.full_like(mag, math.sqrt(float(np.sum(mag**2)) / L))
    # Geometric interpolation
    new_mag = mag ** (1.0 - beta) * target_mag ** beta
    # Preserve phase
    phase = F / np.maximum(mag, 1e-300)
    new_F = new_mag * phase
    new_a = np.fft.irfft(new_F, L)[:n]
    return _norm1(new_a)

    if K >= len(c):

        k_lo = max(0, j - n + 1)
        k_hi = min(j, n - 1)
        if k_lo <= k_hi:
            ks = np.arange(k_lo, k_hi + 1)
            g[ks] += a[j - ks]
    g /= float(len(peaks))
    g *= 2.0  # factor of 2 from the derivative
    return g
