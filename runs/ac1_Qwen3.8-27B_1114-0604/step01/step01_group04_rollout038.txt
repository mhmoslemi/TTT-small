import numpy as np
import time
from scipy.optimize import minimize_scalar

def full_conv(a):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def score(a):
    n = a.size
    C = full_conv(a)
    s = a.sum()
    if s < 0.01:
        return np.inf
    return 2.0 * n * max(C) / (s * s)

def coordinate_optimize(a, i, C, n_samp=30):
    """
    Optimize a[i] given current a and convolution C.
    Returns (new_x, new_score) where new_x is the optimized a[i] value.
    """
    n = a.size
    x_old = a[i]
    S = a.sum() - x_old  # sum of other coords
    
    def obj(x):
        # Compute C'[k] for all k
        C_new = C.copy()
        # k = 2*i
        k = 2 * i
        if k < 2 * n - 1:
            C_new[k] = C[k] - x_old * x_old + x * x
        # Other k where 0 <= k-i < n and k != 2*i
        k_min = max(0, i)  # k-i >= 0 => k >= i
        k_max = min(2*n-2, i + n - 1)  # k-i <= n-1 => k <= i+n-1
        for k in range(k_min, k_max + 1):
            if k == 2 * i:
                continue
            C_new[k] = C[k] - 2 * x_old * a[k-i] + 2 * x * a[k-i]
        max_C = np.max(C_new)
        sum_a = S + x
        if sum_a < 1e-10:
            return np.inf
        return 2.0 * n * max_C / (sum_a * sum_a)
    
    x_max = max(2 * x_old + 0.01, 1.0)
    res = minimize_scalar(obj, bounds=(0, x_max), method='bounded', 
                          options={'xatol': 1e-8, 'maxiter': 50})
    return res.x, res.fun

def coordinate_optimize_fast(a, i, C, x_old=None):
    n = a.size
    if x_old is None:
        x_old = a[i]
    S = a.sum() - x_old
    
    # Precompute the "base" max (convolution values that don't change)
    # Changed indices: k in [i, i+n-1] (clipped to [0, 2n-2])
    k_lo = max(0, i)
    k_hi = min(2*n-2, i+n-1)
    
    # For k not in [k_lo, k_hi]: C_new[k] = C[k]
    # For k in [k_lo, k_hi]: C_new[k] = C[k] + delta_k(x)
    
    # base_max = max of C[k] for k outside [k_lo, k_hi]
    mask = np.ones(2*n-1, dtype=bool)
    mask[k_lo:k_hi+1] = False
    base_max = np.max(C[mask]) if mask.any() else -np.inf
    
    # For k in [k_lo, k_hi], precompute a[k-i]
    ks = np.arange(k_lo, k_hi+1)
    ai_vals = a[ks - i]  # a[k-i] for k in range
    C_vals = C[ks]
    k_2i = 2*i
    
    def obj(x):
        # Compute C_new for changed indices
        delta = 2.0 * (x - x_old) * ai_vals
        # For k = 2i, the formula is different
        if k_2i >= k_lo and k_2i <= k_hi:
            idx_2i = k_2i - k_lo
            # C_new[2i] = C[2i] + x^2 - x_old^2
            # The delta formula gives: 2*(x-x_old)*a[i] = 2*x*a[i] - 2*x_old*a[i]
            # But we want: x^2 - x_old^2
            # So overwrite:
            C_changed = C_vals + delta
            C_changed[idx_2i] = C_vals[idx_2i] + x*x - x_old*x_old
        else:
            C_changed = C_vals + delta
        
        max_changed = np.max(C_changed)
        max_total = max(base_max, max_changed)
        
        sum_a = S + x
        if sum_a < 1e-10:
            return np.inf
        return 2.0 * n * max_total / (sum_a * sum_a)
    
    x_max = max(2.0 * x_old + 0.01, 1.0)
    res = minimize_scalar(obj, bounds=(0, x_max), method='bounded',

```python
"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using coordinate descent
with exact 1D subproblem solutions, multiple sequence lengths,
diverse initializations, and periodic restarts.
"""

import numpy as np
import time
from scipy.optimize import minimize_scalar


def full_conv(a):
    """Full linear convolution of a with itself via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = a.size
    C = full_conv(a)
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    return 2.0 * n * float(np.max(C)) / (s * s)


def _obj_func(x, data):
    """1D objective function for coordinate optimization."""
    n, S, x_old, C_base_max, C_vals, ai_vals, k_2i_in_range, idx_2i = data
    max_linear = float(np.max(C_vals + 2.0 * (x - x_old) * ai_vals))
    if k_2i_in_range:
        val_2i = C_vals[idx_2i] + x * x - x_old * x_old
        max_changed = max(max_linear, val_2i)
    else:
        max_changed = max_linear
    max_total = max(C_base_max, max_changed)
    sum_a = S + x
    if sum_a < 1e-10:
        return 1e10
    return 2.0 * n * max_total / (sum_a * sum_a)


def optimize_coord(a, i, C):
    """
    Optimize coordinate i of a, given current a and convolution C.
    Returns the new optimal value for a[i].
    """
    n = a.size
    x_old = a[i]
    S = float(a.sum()) - x_old
    
    k_lo = max(0, i)
    k_hi = min(2 * n - 2, i + n - 1)
    k_2i = 2 * i
    k_2i_in_range = (k_lo <= k_2i <= k_hi)
    idx_2i = k_2i - k_lo if k_2i_in_range else -1
    
    total_len = 2 * n - 1
    if k_lo == 0 and k_hi == total_len - 1:
        C_base_max = -1e30
    else:
        # Max of C outside [k_lo, k_hi]
        if k_lo > 0:
            max_left = float(np.max(C[:k_lo]))
        else:
            max_left = -1e30
        if k_hi < total_len - 1:
            max_right = float(np.max(C[k_hi+1:]))
        else:
            max_right = -1e30
        C_base_max = max(max_left, max_right)
    
    C_vals = C[k_lo:k_hi+1]
    ai_vals = a[k_lo-i:k_hi+1-i]
    
    data = (n, S, x_old, C_base_max, C_vals, ai_vals, k_2i_in_range, idx_2i)
    x_max = max(2.0 * x_old + 0.05, 0.5)
    
    res = minimize_scalar(_obj_func, args=(data,), bounds=(0.0, x_max), 
                          method='bounded', options={'xatol': 1e-12, 'maxiter': 200})
    return float(res.x)


def update_conv(C, a, i, x_old, x_new):
    """Incrementally update convolution C after changing a[i] from x_old to x_new."""
    n = a.size
    k_2i = 2 * i
    if k_2i < 2 * n - 1:
        C[k_2i] += x_new * x_new - x_old * x_old
    k_lo = max(0, i)
    k_hi = min(2 * n - 2, i + n - 1)
    for k in range(k_lo, k_hi + 1):
        if k == k_2i:
            continue
        C[k] += 2.0 * (x_new - x_old) * a[k - i]


def coord_descent(a, time_budget, rng, verbose=False):
    """
    Run coordinate descent to minimize the score.
    a should be non-negative. Returns (best_a, best_score).
    """
    n = a.size
    a = a.astype(np.float64).copy()
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-10:
        return a, np.inf
    a /= s
    
    best_a = a.copy()
    best_sc = compute_score(a)
    
    t0 = time.monotonic()
    no_improve = 0
    pass_count = 0
    
    while time.monotonic() - t0 < time_budget:
        order = rng.permutation(n)
        C = full_conv(a)
        
        changed = False
        for i in order:
            if time.monotonic() - t0 > time_budget:
                break
            x_new = optimize_coord(a, i, C)
            x_old = a[i]
            if abs(x_new - x_old) > 1e-12:
                a[i] = x_new
                update_conv(C, a, i, x_old, x_new)
                changed = True
        
        pass_count += 1

```python
"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using coordinate descent with exact
1D subproblem solutions (scipy bounded minimization), multiple sequence
lengths, diverse initializations, incremental convolution updates,
and periodic random restarts to escape local minima.
"""

import numpy as np
import time
from scipy.optimize import minimize_scalar


def full_conv(a):
    """Full linear convolution of a with itself via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def compute_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = a.size
    s = float(a.sum())
    if s < 0.01:
        return np.inf
    C = full_conv(a)
    return 2.0 * n * float(np.max(C)) / (s * s)


def _obj_func(x, data):
    """1D objective for coordinate optimization. x is the new value of a[i]."""
    n, S, x_old, C_base_max, C_vals, ai_vals, k_2i_in_range, idx_2i = data
    max_linear = float(np.max(C_vals + 2.0 * (x - x_old) * ai_vals))
    if k_2i_in_range:
        val_2i = C_vals[idx_2i] + x * x - x_old * x_old
        max_changed = max(max_linear, val_2i)
    else:
        max_changed = max_linear
    max_total = max(C_base_max, max_changed)
    sum_a = S + x
    if sum_a < 1e-10:
        return 1e15
    return 2.0 * n * max_total / (sum_a * sum_a)


def optimize_coord(a, i, C):
    """Optimize coordinate i given current a and convolution C. Returns new a[i]."""
    n = a.size
    x_old = a[i]
    S = float(a.sum()) - x_old

    k_lo = max(0, i)
    k_hi = min(2 * n - 2, i + n - 1)
    k_2i = 2 * i
    k_2i_in_range = (k_lo <= k_2i <= k_hi)
    idx_2i = k_2i - k_lo if k_2i_in_range else -1

    total_len = 2 * n - 1
    if k_lo == 0 and k_hi == total_len - 1:
        C_base_max = -1e30
    else:
        max_left = float(np.max(C[:k_lo])) if k_lo > 0 else -1e30
        max_right = float(np.max(C[k_hi + 1:])) if k_hi < total_len - 1 else -1e30
        C_base_max = max(max_left, max_right)

    C_vals = C[k_lo:k_hi + 1]
    ai_vals = a[k_lo - i:k_hi + 1 - i]

    data = (n, S, x_old, C_base_max, C_vals, ai_vals, k_2i_in_range, idx_2i)
    x_max = max(2.0 * x_old + 0.05, 0.5)

    res = minimize_scalar(_obj_func, args=(data,), bounds=(0.0, x_max),
                          method='bounded', options={'xatol': 1e-12, 'maxiter': 200})
    return float(res.x)


def update_conv(C, a, i, x_old, x_new):
    """Incrementally update convolution after changing a[i] from x_old to x_new."""
    n = a.size
    k_2i = 2 * i
    if k_2i < 2 * n - 1:
        C[k_2i] += x_new * x_new - x_old * x_old
    k_lo = max(0, i)
    k_hi = min(2 * n - 2, i + n - 1)
    for k in range(k_lo, k_hi + 1):
        if k == k_2i:
            continue
        C[k] += 2.0 * (x_new - x_old) * a[k - i]


def coord_descent(a, time_budget, rng, verbose=False):
    """Coordinate descent to minimize the score. Returns (best_a, best_score)."""
    n = a.size
    a = a.astype(np.float64).copy()
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-10:
        return a, np.inf
    a /= s

    best_a = a.copy()
    best_sc = compute_score(a)

    t0 = time.monotonic()
    no_improve = 0
    pass_count = 0

    while time.monotonic() - t0 < time_budget:
        order = rng.permutation(n)
        C = full_conv(a).astype(np.float64)

        for i in order:
            if time.monotonic() - t0 > time_budget:
                break
            x_new = optimize_coord(a, i, C)
            x_old = a[i]
            if abs(x_new - x_old) > 1e-12:
                a[i] = x_new
                update_conv(C, a, i, x_old, x_new)

        pass_count += 1

        C_check = full_conv(a)
        s_check = float(a.sum())
        if s_check > 1e-10:
            sc_candidate = 2.0 * n * float(np.max(C_check)) / (s_check * s_check)
        else:
            sc_candidate = np.inf

        if sc_candidate < best_sc - 1e-12:
            best_sc = sc_candidate
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        a = np.clip(a, 0.0, None)
        s = float(a.sum())
        if s > 1e-10:
            a /= s

        if no_improve > 15:
            if verbose and pass_count % 20 == 0:
                print(f"    restart pass={pass_count} sc={best_sc:.6f} no_imp={no_improve}")
            scale = 0.01 + 0.005 * min(no_improve, 20)
            a = best_a + scale * rng.standard_normal(n)
            a = np.clip(a, 0.0, None)
            s = float(a.sum())
            if s > 1e-10:
                a /= s
            no_improve = 0

        if no_improve > 60:
            if verbose:
                print(f"    aggressive restart pass={pass_count} sc={best_sc:.6f}")
            a = best_a + 0.05 * rng.standard_normal(n)
            a = np.clip(a, 0.0, None)
            s = float(a.sum())
            if s > 1e-10:
                a /= s
            no_improve = 0

    return best_a, best_sc


def init_uniform(n):
    return np.ones(n) / n


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.01
    return a / a.sum()


def init_two_bump(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.1, 0.4)
    c2 = n * rng.uniform(0.6, 0.9)
    sig = n * rng.uniform(0.05, 0.15)
    a = np.exp(-((i - c1) / sig) ** 2) + np.exp(-((i - c2) / sig) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_edges(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / max(c, 1.0)) ** 2 + 0.1
    return a / a.sum()


def init_sine(n, rng):
    i = np.arange(n, dtype=np.float64)
    freq = int(rng.integers(1, 5))
    phase = rng.uniform(0, 2.0 * np.pi)
    a = (1.0 + 0.5 * np.sin(2.0 * np.pi * freq * i / max(n, 1) + phase)) ** 2
    a += 1e-6
    return a / a.sum()


def init_sqrt_decay(n, rng):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a += 0.03 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def resize_sequence(a, new_n):
    """Resize sequence via linear interpolation."""
    old_n = a.size
    if old_n == new_n:
        return a.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    a_new = np.interp(x_new, x_old, a)
    a_new = np.clip(a_new, 0.0, None)
    s = a_new.sum()
    if s > 0:
        a_new /= s
    return a_new
