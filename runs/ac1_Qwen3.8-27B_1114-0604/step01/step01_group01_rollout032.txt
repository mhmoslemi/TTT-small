"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses accelerated multiplicative weights (Nesterov momentum) on the simplex
with adaptive smoothed-max, block coordinate polish, and optional upsample.
"""

import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _fft_conv(a, b):
    la = len(a)
    lb = len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 1000000 and nout <= 4000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)


def _correlate(w, p):
    """Compute r[i] = sum_k w[k] * p[k-i] for i=0..n-1."""
    n = len(p)
    L = _next_pow2(2 * n)
    fw = np.fft.rfft(w, L)
    fp = np.fft.rfft(p, L)
    out = np.fft.irfft(fw * np.conjugate(fp), L)
    return out[:n].astype(float, copy=False)


def _exact_score(p):
    n = p.size
    c = _fft_conv(p, p)
    if c.size == 0:
        return float('inf')
    return float(2.0 * n * np.max(c))


def _smooth_grad(p, beta):
    """Return (exact_score, gradient_of_smoothed_max)."""
    n = p.size
    c = _fft_conv(p, p)
    d = n * c
    d_max = float(np.max(d))
    exact = 2.0 * d_max
    shift = d - d_max
    e = np.exp(beta * shift)
    s = float(e.sum())
    if s <= 0 or not np.isfinite(s):
        w = np.full(c.size, 1.0 / c.size)
    else:
        w = e / s
    r = _correlate(w, p)
    grad = 2.0 * r
    grad = np.maximum(grad, 0.0)
    return exact, grad


def _sanitize(p):
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 1e-10 or not np.isfinite(s):
        return None
    return p / s


def _resample(arr, m):
    if m <= 0:
        return None
    a = _sanitize(arr)
    if a is None:
        return None
    n = a.size
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m, dtype=float)
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0) + 1e-12
    return _sanitize(out)


def _run_mmd(p0, rng, time_limit, beta_start=80.0, beta_end=5000.0, verbose=False):
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    if time_limit <= 0:
        return p, _exact_score(p)

    m = p.size
    floor = 1e-18 / float(m)
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_exact = _exact_score(p)
    cur_exact, _ = _smooth_grad(p, beta_start)

    # Nesterov state
    y = p.copy()  # lookahead point
    beta = float(beta_start)
    alpha = 0.8
    momentum = 0.85
    v = np.zeros(m)
    deadline = time.monotonic() + float(time_limit)
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        # Compute gradient at lookahead point y
        _, grad_y = _smooth_grad(y, beta)
        gmax = float(np.max(grad_y))
        if gmax < 1e-14:
            stall += 10
        else:
            # Momentum update
            v = momentum * v + grad_y
            v = np.maximum(v, 0.0)
            vmax = float(np.max(v))
            if vmax > 1e-14:
                v = v / vmax

            # Multiplicative update with backtracking
            p_try = None
            best_sc = float('inf')
            step = alpha
            for _ in range(4):
                dec = np.exp(-step * v)
                cand = p * dec
                cand = np.maximum(cand, floor)
                cs = float(cand.sum())
                if cs <= 0 or not np.isfinite(cs):
                    step *= 0.5
                    continue
                cand = cand / cs
                sc = _exact_score(cand)
                if sc < best_sc:
                    best_sc = sc
                    p_try = cand
                if sc < cur_exact - 1e-10:
                    step *= 1.4
                else:
                    step *= 0.55

            if p_try is not None and np.isfinite(best_sc):
                # Nesterov: update p as weighted average
                t = 2.0 / (it + 2.5)
                p_new = (1.0 - t) * p + t * p_try
                p_new = np.maximum(p_new, floor)
                p_new = p_new / p_new.sum()
                p = p_new
                cur_exact = best_sc
                # Update lookahead
                y = p * np.exp(-0.5 * alpha * v)
                y = np.maximum(y, floor)
                y = y / y.sum()
                stall = 0
                alpha = min(3.0, alpha * 1.02)

                if cur_exact < best_exact:
                    best = p.copy()
                    best_exact = cur_exact
                    if verbose and it % 100 == 0:
                        print(f"  mmd n={m} it={it} best={best_exact:.7f}")
            else:
                stall += 1
                alpha = max(0.05, alpha * 0.7)

        it += 1
        if it % 40 == 0:
            beta = min(beta_end, beta * 1.25)
            # Periodic exact check
            sc_check = _exact_score(p)
            if sc_check < best_exact:
                best = p.copy()
                best_exact = sc_check

        if stall > 150:
            p = best.copy()
            y = p.copy()
            v = np.zeros(m)
            alpha = 0.5
            beta = max(beta_start * 0.5, beta * 0.4)
            stall = 0
            cur_exact = _exact_score(p)

    return best, best_exact


def _block_polish(p0, rng, time_limit):
    """Block coordinate descent: multiply random blocks by factors, keep best."""
    p = _sanitize(p0)
    if p is None or time_limit <= 0:
        return None, float('inf')
    n = p.size
    if n < 4:
        return p, _exact_score(p)

    floor = 1e-18 / float(n)
    p = np.maximum(p, floor)
    p = p / p.sum()
    best = p.copy()
    best_score = _exact_score(p)
    cur_score = best_score
    deadline = time.monotonic() + float(time_limit)
    stagnant = 0
    factors = (0.4, 0.6, 0.8, 0.9, 1.1, 1.25, 1.5, 2.0, 3.0)

    while time.monotonic() < deadline:
        # Choose block width
        r = float(rng.random())
        if r < 0.25:
            w = int(rng.integers(1, max(2, n // 500 + 1)))
        elif r < 0.60:
            w = int(rng.integers(max(2, n // 100), max(3, n // 30 + 1)))
        elif r < 0.85:
            w = int(rng.integers(max(4, n // 30), max(6, n // 10 + 1)))
        else:
            w = int(rng.integers(max(8, n // 10), max(12, n // 4 + 1)))
        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))

        base = p.copy()
        cur_best = base
        cur_best_sc = cur_score

        for f in factors:
            cand = base.copy()
            cand[start:start + w] *= float(f)
            cand = np.maximum(cand, floor)
            cs = float(cand.sum())
            if cs <= 0:
                continue
            cand = cand / cs
            sc = _exact_score(cand)
            if sc < cur_best_sc - 1e-12:
                cur_best_sc = sc
                cur_best = cand

        if cur_best is not base:
            p = cur_best
            cur_score = cur_best_sc
            if cur_score < best_score:
                best = p.copy()
                best_score = cur_score
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 40:
                # Random kick
                p_try = p * np.exp(rng.normal(0.0, 0.03, n))
                p_try = np.maximum(p_try, floor)
                p_try = p_try / p_try.sum()
                sc = _exact_score(p_try)
                if sc < best_score * 1.02:
                    p = p_try
                    cur_score = sc
                else:
                    p = best.copy()
                    cur_score = best_score
                stagnant = 0

        if stagnant > 400:
            p = best.copy()
            cur_score = best_score
            stagnant = 0

    return best, best_score


def _run_mmd_v2(p0, rng, time_limit, beta_start=100.0, beta_end=8000.0, verbose=False):

    p = _sanitize(p0)

    m = p.size

    p = p / p.sum()

    best = p.copy()
    best_exact = _exact_score(p)
    beta = float(beta_start)
    alpha = 1.0
    deadline = time.monotonic() + float(time_limit)
    stall = 0
    it = 0
    last_best_it = 0

    while time.monotonic() < deadline:
        _, grad = _smooth_grad(p, beta)
        gmax = float(np.max(grad))
        if gmax < 1e-14:
            stall += 5
        else:
            grad_n = grad / gmax

            # Try a range of step sizes (geometric)
            base_step = alpha / gmax
            candidates_steps = [base_step * f for f in (0.25, 0.5, 1.0, 2.0, 4.0)]
            candidates_steps = [s for s in candidates_steps if 1e-6 < s < 10.0]
            if not candidates_steps:
                candidates_steps = [base_step]

            best_cand = None
            best_sc = float('inf')
            for s in sorted(candidates_steps, reverse=True):
                dec = np.exp(-s * grad_n)
                cand = p * dec
                cand = np.maximum(cand, floor)
                cs = float(cand.sum())
                if cs <= 0 or not np.isfinite(cs):
                    continue
                cand = cand / cs
                sc = _exact_score(cand)
                if sc < best_sc:
                    best_sc = sc
                    best_cand = cand
                if sc < best_exact:
                    break  # Got improvement, use this step

            if best_cand is not None and np.isfinite(best_sc):
                p = best_cand
                if best_sc < best_exact:
                    best = p.copy()
                    best_exact = best_sc
                    last_best_it = it
                    stall = 0
                    if verbose and (it - last_best_it >= 80):
                        print(f"  mmdv2 n={m} it={it} best={best_exact:.7f}")
                else:
                    stall += 1
                alpha = min(4.0, alpha * 1.01)
            else:
                stall += 1
                alpha = max(0.1, alpha * 0.7)

        it += 1
        if it % 50 == 0:
            beta = min(beta_end, beta * 1.3)

        if stall > 100 + (it // 200):
            p = best.copy()
            alpha = 0.8
            beta = max(beta_start, beta * 0.6)
            # Small perturbation
            p_try = p * np.exp(rng.normal(0.0, 0.02, m))
            p_try = np.maximum(p_try, floor)
            p_try = p_try / p_try.sum()
            sc = _exact_score(p_try)
            if sc < best_exact * 1.03:
                p = p_try
            stall = 0

    return best, best_exact
