def _mws_direction(a, L, m, n, prior, inner=20, eta_start=2.0, eta_end=50.0):

    a = _normalize(a)
    prior = _normalize(prior)

    F = np.fft.rfft(a, L)
    Frev = np.fft.rfft(a[::-1], L)

    x = prior.copy()
    y = np.full(m, 1.0 / m)
    avg = np.zeros(n)
    logprior = np.log(np.maximum(prior, 1e-300))

    for it in range(inner):
        frac = 0.0 if inner <= 1 else it / (inner - 1)
        eta = eta_start + (eta_end - eta_start) * frac

        X = np.fft.rfft(x, L)
        u = np.fft.irfft(F * X, L)[:m]
        np.maximum(u, 0.0, out=u)

        vu = n * u
        vma = float(np.max(vu))
        z = eta * (vu - vma)
        z = np.clip(z, -40.0, 0.0)
        y = np.exp(z)
        ys = float(y.sum())
        if ys < 1e-300:
            y.fill(1.0 / m)
        else:
            y /= ys

        avg += x

        Y = np.fft.rfft(y, L)
        s = np.fft.irfft(Y * Frev, L)[(n-1):m]
        np.maximum(s, 0.0, out=s)

        ps = n * s
        sma = float(np.max(ps))
        zx = -eta * (ps - sma) + 1.0 * logprior
        zx -= zx.max()
        zx = np.clip(zx, -40.0, 0.0)
        x = np.exp(zx)
        xs = float(x.sum())
        if xs < 1e-300:
            x = prior.copy()
        else:
            x /= xs

    return _normalize(0.5 * (avg / inner) + 0.5 * x)

        # Line search: mix best_a with direction x
        best_mix = a.copy()
        best_sc = best_score
        for lam in (0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0):

        elif best_sc < best_score:

            best_score = best_sc

def _fuglede_regime(a, L, m, n, time_limit, rng):
    """
    Fuglede-style spectral iteration: modify the spectrum to flatten
    the convolution, then project back to non-negative.
    """
    deadline = time.monotonic() + time_limit
    a = _normalize(a)
    best_a = a.copy()
    b = _conv_fft(a, L, m)
    best_score = _score_from_b(b, n)

    A = a.copy()
    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        b = _conv_fft(a, L, m)
        sc = _score_from_b(b, n)

        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Fuglede update: flatten the convolution spectrum
        Af = np.fft.rfft(a, L)
        # Power in convolution domain
        C = Af * Af  # This is FFT of conv(a,a)
        C_real = np.real(C)
        C_imag = np.imag(C)

        # Desired: flat convolution (all b_k equal)
        # Flat spec in convolution domain => all C values equal
        # C_flat = mean(C_real) (real part)
        c_mean = float(np.mean(C_real[:m]))
        c_mean = max(c_mean, 1e-300)

        # Error in convolution domain
        b_current = _conv_fft(a, L, m)
        b_desired = np.full(m, c_mean / m * m)  # uniform
        b_desired = np.ones(m) * (float(np.sum(b_current)) / m)

        err = b_current - b_desired
        # Pad error to length L
        err_full = np.zeros(L)
        err_full[:m] = err

        # FFT of error
        E = np.fft.rfft(err_full)

        # Update: reduce the error component
        # A_new = A - eta * E / (2*A) (gradient step on |A|^2 proxy)
        eta_f = 0.05
        A_upd = Af - eta_f * E / (2.0 * Af + 1e-300)
        # Take real part (since a is real)
        a_new = np.fft.irfft(A_upd, L)[:n]
        # Enforce non-negativity
        np.maximum(a_new, 0.0, out=a_new)
        s_new = float(a_new.sum())
        if s_new < 1e-300:
            a_new = a.copy()
        else:
            a_new /= s_new

        # Add small noise to prevent stagnation
        a_new = a_new + rng.normal(0, 0.001, n)
        a = _normalize(a_new)

        it += 1
        if no_improve > 150:
            # Perturb
            a = _normalize(0.8 * a + 0.2 * _gaussian_prior(n, rng, float(rng.uniform(0.3, 2.0))))
            no_improve = 0

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Searches for non-negative step heights minimizing
    2*n*max(conv(a,a))/sum(a)^2.
    
    Uses a combination of:
    1. Soft active-set subgradient descent (primary)
    2. Fuglede-style spectral iteration (secondary)
    3. MWS minimax direction (tertiary)
    4. Final cold polishing
    With multiple diverse initializations and time allocation.
    """
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget < 10.0:
        budget = 2400.0

    start_time = time.monotonic()
    hard_deadline = start_time + budget - 5.0  # Leave 5s for final evaluation

    rng = np.random.default_rng(seed)
    n = 6121
    L = 1 << (2 * n - 2).bit_length()
    m = 2 * n - 1

    # Get initial candidates
    candidates = []

    # Try to use height_sequence_1 if available
    try:
        base = height_sequence_1
        arr = np.asarray(base, dtype=np.float64).ravel()
        if arr.size == n and np.all(np.isfinite(arr)):
            candidates.append(_normalize(arr))
    except (NameError, Exception):
        pass

    # Add shape-based priors
    candidates.extend(_shape_priors(n, rng))

    # Add a couple random priors
    for _ in range(3):
        alpha = float(rng.uniform(0.3, 2.5))
        candidates.append(_gaussian_prior(n, rng, alpha))

    # Phase 1: Quick screening of initial candidates
    best_a = None
    best_score = float('inf')
    cand_scores = []

    for c in candidates:
        b_c = _conv_fft(c, L, m)
        sc = _score_from_b(b_c, n)
        cand_scores.append(sc)
        if sc < best_score:
            best_score = sc
            best_a = c.copy()

    print(f"Initial best: score={best_score:.6f}, n={n}, L={L}")

    # Phase 2: Main optimization loop
    # Allocate time: 70% to gradient descent, 15% to Fuglede, 15% to MWS/polish
    elapsed = time.monotonic() - start_time
    remaining = hard_deadline - time.monotonic()

    if remaining < 10:
        seq = [float(x) for x in best_a]
        return seq

    # Run gradient descent from best candidate
    t1 = min(remaining * 0.35, 600.0)
    tmp, sc = _gradient_descent(best_a, L, m, n, rng, t1, temp_init=1.0, eta_init=0.08)
    if sc < best_score:
        best_score = sc
        best_a = tmp
    print(f"After GD phase 1: score={best_score:.6f}, time={time.monotonic()-start_time:.1f}s")

    # Run Fuglede from best
    elapsed = time.monotonic() - start_time
    remaining = hard_deadline - time.monotonic()
    if remaining > 60:
        t2 = min(remaining * 0.2, 300.0)
        tmp, sc = _fuglede_regime(best_a, L, m, n, t2, rng)
        if sc < best_score:
            best_score = sc
            best_a = tmp
        print(f"After Fuglede: score={best_score:.6f}, time={time.monotonic()-start_time:.1f}s")

    # Run MWS from best
    remaining = hard_deadline - time.monotonic()
    if remaining > 40:
        t3 = min(remaining * 0.15, 200.0)
        tmp, sc = _mws_loop(best_a, L, m, n, rng, t3)
        if sc < best_score:
            best_score = sc
            best_a = tmp
        print(f"After MWS: score={best_score:.6f}, time={time.monotonic()-start_time:.1f}s")

    # Phase 3: Second round of gradient descent with better initialization
    remaining = hard_deadline - time.monotonic()
    if remaining > 30:
        t4 = min(remaining * 0.3, 500.0)
        tmp, sc = _gradient_descent(best_a, L, m, n, rng, t4, temp_init=0.5, eta_init=0.04)
        if sc < best_score:
            best_score = sc
            best_a = tmp
        print(f"After GD phase 2: score={best_score:.6f}, time={time.monotonic()-start_time:.1f}s")

    # Phase 4: Try a different starting point (diversification)
    remaining = hard_deadline - time.monotonic()
    if remaining > 60:
        # Mix best with a new random prior
        mix_frac = 0.3
        prior = _gaussian_prior(n, rng, float(rng.uniform(0.5, 1.5)))
        alt_start = _normalize((1 - mix_frac) * best_a + mix_frac * prior)
        t5 = min(remaining * 0.25, 300.0)
        tmp, sc = _gradient_descent(alt_start, L, m, n, rng, t5, temp_init=2.0, eta_init=0.1)
        if sc < best_score:
            best_score = sc
            best_a = tmp
            print(f"Alternative start improved: score={best_score:.6f}")

    # Phase 5: Cold polish (fine-tuning)
    remaining = hard_deadline - time.monotonic()
    if remaining > 20:
        t6 = min(remaining * 0.5, 300.0)
        tmp, sc = _gradient_descent(best_a, L, m, n, rng, t6, temp_init=0.1, eta_init=0.01, cold=True)
        if sc < best_score:
            best_score = sc
            best_a = tmp
        print(f"After cold polish: score={best_score:.6f}, time={time.monotonic()-start_time:.1f}s")

    # Final: evaluate with exact function and return best
    seq = [float(x) for x in best_a]

    # Also try a few small perturbations
    remaining = hard_deadline - time.monotonic()
    if remaining > 15:
        for _ in range(5):
            pert = best_a + rng.normal(0, 0.002, n)
            pert = _normalize(pert)
            b_p = _conv_fft(pert, L, m)
            sc_p = _score_from_b(b_p, n)
            if sc_p < best_score:
                best_score = sc_p
                best_a = pert

    seq = [float(x) for x in best_a]

    # Validate with exact evaluator
    exact_val = _exact_score(seq)
    if np.isfinite(exact_val):
        print(f"Final exact score: {exact_val:.6f}")
    else:
        print(f"WARNING: exact score is not finite: {exact_val}")
        # Fallback
        seq = [1.0] * n
        exact_val = _exact_score(seq)
        print(f"Fallback exact score: {exact_val:.6f}")

    print(f"Total time: {time.monotonic()-start_time:.1f}s")
    return seq
