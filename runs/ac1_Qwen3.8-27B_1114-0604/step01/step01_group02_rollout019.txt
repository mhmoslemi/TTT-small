Step function optimization using FFT-based gradient descent with adaptive
top-k peak smoothing, cosine-annealed Adam, sum normalization, and
periodic structured restarts. Multiple sequence lengths are explored.
"""

import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def score_normalized(c, n):
    """Score assuming sum(a) = n (normalized). Score = 2 * max_c / n."""
    return 2.0 * float(np.max(c)) / n


def grad_smooth_max(a, c, topk, temp_scale):
    """Gradient of weighted-smooth-max of c with respect to a.
    Each peak k contributes 2*a[k-j] to d c[k]/d a[j]."""
    n = len(a)
    k_act = min(topk, len(c))
    if k_act == 1:
        idx = np.int64(np.argmax(c))
        wts = np.ones(1)
    else:
        idx = np.argpartition(c, -k_act)[-k_act:]
        order = np.argsort(c[idx])[::-1]
        idx = idx[order]
        vals = c[idx]
        top_val = float(vals[0])
        if top_val > 1e-12:
            temp = temp_scale * top_val
            wts = np.exp((vals - top_val) / temp)
            wts /= wts.sum()
        else:
            wts = np.ones(k_act) / k_act
    
    grad = np.zeros(n)
    for i in range(len(idx)):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-15:
            continue
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        # a[kj - j] for j in [js, je] = a[ki - j] for j in [js, je]
        # This is a reversed slice of a
        partner = a[js:je+1][::-1]
        grad[js:je+1] += wi * 2.0 * partner
    
    return grad

    S = a.sum()
    if S < 1e-6:
        a[:] = 1.0
    else:
        a *= target_sum / S
    a = np.clip(a, 0, 1000.0)
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_score = np.inf
    best_a = a.copy()

    # Each iteration: FFT (O(n log n)) + gradient (O(n * topk)) + Adam (O(n))
    # For n=2048, FFT ~ 0.1ms, gradient ~ 0.5ms, total ~ 1ms per iter
    # In 60 seconds: ~50000 iterations
    # We'll use the time budget to determine total steps
    
    no_improve = 0
    max_patience = patience
    
    time_min = 1e-9
    while time.monotonic() < deadline:
        # Compute progress (0 to 1)
        elapsed = time.monotonic() - (deadline - (deadline))  # not useful
        # Use t as proxy for progress, or just use fixed schedule
        
        # Anneal parameters
        progress = min(1.0, t / max(1, patience * 3))
        topk = max(2, int(topk_start + (topk_end - topk_start) * progress))
        temp = temp_start + (temp_end - temp_start) * progress
        lr = init_lr * (0.5 + 0.5 * np.cos(np.pi * min(progress, 1.0)))
        lr = max(lr, init_lr * 0.05)

        c = fast_self_conv(a)
        max_c = float(np.max(c))
        score = 2.0 * max_c / n  # normalized score

            no_improve = 0

            no_improve += 1

        if no_improve >= max_patience:
            scale = 0.03 + 0.07 * rng.random()
            a = best_a * (1.0 + scale * rng.standard_normal(n))
            a = np.clip(a, 0, 1000.0)
            S = a.sum()
            if S < 1e-6:
                a[:] = 1.0
            else:
                a *= target_sum / S
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            no_improve = 0
            max_patience = max(80, max_patience // 2)
            continue
        
        # Compute gradient
        g = grad_smooth_max(a, c, topk, temp)
        
        # Adam step

        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1**t)

Step function optimization using FFT-based gradient descent with:
- Sum normalization (exploiting scale invariance of the objective)
- Adaptive top-k peak smoothing with cooling temperature schedule
- Cosine-annealed Adam optimizer with restarts
- Multi-length exploration (tries several sequence lengths)
- Structured diverse initializations and restart perturbations

import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution of 1-D array via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def norm_sum(a, target):
    """Scale a in-place-equivalent to have sum = target."""
    S = a.sum()
    if S < 1e-10:
        return np.full_like(a, target / len(a))
    return a * (target / S)


def grad_smooth_max(a, c, topk, temp_scale):
    """Approximate gradient of max(c) w.r.t. a, using top-k softmax weighting."""
    n = len(a)
    k_act = min(topk, len(c))
    if k_act == 1:
        idx = np.int64(np.argmax(c))
        wts = np.ones(1)
    else:
        idx = np.argpartition(c, -k_act)[-k_act:]
        order = np.argsort(c[idx])[::-1]
        idx = idx[order]
        vals = c[idx]
        top_val = float(vals[0])
        if top_val > 1e-12:
            temp = temp_scale * top_val
            wts = np.exp((vals - top_val) / temp)
            wts /= wts.sum()
        else:
            wts = np.ones(k_act) / k_act

    grad = np.zeros(n)
    for i in range(len(idx)):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-15:
            continue
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        partner = a[js:je + 1][::-1]
        grad[js:je + 1] += wi * 2.0 * partner
    return grad

        temp = temp_start + (temp_end - temp_start) * progress
        lr = init_lr * max(0.05, 0.5 + 0.5 * np.cos(np.pi * progress))

        c = fast_self_conv(a)
