def smooth_grad(a, b, tau):
    n = a.size
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()

        w = np.zeros(b.size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= s
    return 2.0 * np.correlate(w, a, mode='valid')

def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

    n = a.size

    n = v.size

        return v.copy()
    v = np.asarray(v, dtype=np.float64)
    v = np.maximum(v, 0)
    s = v.sum()
    if s < 1e-30:
        return np.ones(n) / n
    v = v / s
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp_matrix(a):
    """Build (2n-1) x n matrix where A[k,j] = a[k-j] if 0<=k-j<n else 0."""
    n = a.size
    num_rows = 2 * n - 1
    A = np.zeros((num_rows, n))
    for j in range(n):
        k_end = min(num_rows, j + n)
        A[j:k_end, j] = a[:k_end - j]
    return A


def solve_lp_direction(a, M):
    """Solve LP: max sum(x) s.t. conv(a,x) <= M, x >= 0. Returns normalized direction."""
    n = a.size
    num_rows = 2 * n - 1
    A = build_lp_matrix(a)
    b_ub = np.full(num_rows, M)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.success:
            x = np.maximum(res.x, 0.0)
            s = x.sum()
            if s > 1e-15:
                return x / s
    except Exception:
        pass
    return np.ones(n) / n


def lp_waterfill_optimize(a, time_budget, t_base=0.5, verbose=True):
    """Run LP water-filling optimization with adaptive step and restarts."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    t0 = time.monotonic()
    iters = 0
    last_improve = 0
    t = t_base
    stall_count = 0

    if verbose:
        print(f"  [lp] start n={n}, score={best_sc:.8f}")

    while time.monotonic() - t0 < time_budget:
        if time.monotonic() - t0 > time_budget:
            break

        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))
        M = float(np.max(b))
        if M < 1e-30:
            break

        # Solve LP for direction
        g = solve_lp_direction(a, M)

        # Try main step
        a_new = (1.0 - t) * a + t * g
        a_new = project_simplex(a_new)
        new_sc = normed_score(a_new)

        accepted = False
        if new_sc <= cur_sc + 1e-15:
            a = a_new
            accepted = True
        else:
            # Try half step
            t_half = t * 0.5
            a_half = (1.0 - t_half) * a + t_half * g
            a_half = project_simplex(a_half)
            sc_half = normed_score(a_half)
            if sc_half <= cur_sc + 1e-15:
                a = a_half
                t = t_half
                accepted = True
            else:
                t = t_half

        if new_sc < best_sc or (accepted and sc_half < best_sc if not accepted else new_sc < best_sc):
            cur_best = normed_score(a)
            if cur_best < best_sc:
                best_sc = cur_best
                best_a = a.copy()
                last_improve = iters
                stall_count = 0

        iters += 1

        # Adaptive behavior
        if iters - last_improve > 30:
            stall_count += 1
            if stall_count % 5 == 0:
                t = max(0.05, t * 0.7)
            if stall_count > 20:
                # Restart with perturbation
                pert = 0.1 * rng.standard_normal(n)
                a = project_simplex(best_a + pert)
                t = t_base * (0.5 + 0.5 * rng.random())
                last_improve = iters
                stall_count = 0

        # Periodic re-energization
        if iters % 500 == 0 and iters > 0:
            t = t_base * (0.5 + 0.5 * rng.random())

        if verbose and iters % 200 == 0:
            print(f"    [lp] iter={iters}, best={best_sc:.8f}, t={t:.3f}")

    return best_a, best_sc, iters


def smooth_onesided_grad(a, b, tau):
    """Smoothed subgradient for the max-convolution objective."""
    n = a.size
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(b.size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= s
    return 2.0 * np.correlate(w, a, mode='valid')


def subgrad_optimize(a, time_budget, rng, use_momentum=True):
    """Subgradient descent polishing."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    eta = 0.15
    tau_frac = 0.025
    beta = 0.5
    vel = np.zeros(n)
    iters = 0
    last_improve = 0

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))
        tau = tau_frac * float(np.max(b))
        if tau < 1e-18:
            tau = 1e-18

        g = smooth_onesided_grad(a, b, tau)
        if use_momentum:
            vel = beta * vel + (1.0 - beta) * g
            direction = vel
        else:
            direction = g

        a_new = project_simplex(a - eta * direction)
        new_sc = normed_score(a_new)

        if new_sc <= cur_sc or rng.random() < 0.08:
            a = a_new
        else:
            a = project_simplex(a - 0.4 * eta * direction)

        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = iters

        iters += 1
        if iters - last_improve > 500:
            a = project_simplex(best_a + 0.02 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.05 + 0.15 * rng.random()
            last_improve = iters

    return best_a, best_sc


def init_u_shape(n):
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i).astype(np.float64) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_two_cluster(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.1 + 0.2 * rng.random())
    c2 = n * (0.65 + 0.2 * rng.random())
    sigma = n * (0.03 + 0.07 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.03
    return a / a.sum()


def init_ansatz(n, rng):
    i = np.arange(n, dtype=np.float64)
    method = rng.integers(0, 5)
    if method == 0:
        m = np.minimum(i, n - 1 - i).astype(np.float64) + 1.0
        a = 1.0 / np.sqrt(m)
    elif method == 1:
        freq = 1.0 + rng.random() * 3.0
        a = 0.3 + 0.7 * np.cos(2.0 * np.pi * freq * i / n) ** 2
    elif method == 2:
        c = n * (0.15 + 0.7 * rng.random())
        sigma = n * (0.03 + 0.15 * rng.random())
        a = np.exp(-((i - c) / sigma) ** 2) + 0.3
    elif method == 3:
        a = np.exp(-i / (n * 0.25)) + np.exp(-(n - 1 - i) / (n * 0.25))
    else:
        # Multi-frequency
        f1 = 1.0 + rng.random()
        f2 = 3.0 + rng.random() * 4.0
        a = 0.5 + 0.3 * np.cos(2 * np.pi * f1 * i / n) + 0.2 * np.cos(2 * np.pi * f2 * i / n)
    a = np.abs(a) + 0.005
    return a / a.sum()


def get_inits(n, rng):
    return [
        init_u_shape(n),
        init_two_cluster(n, rng),
        init_random(n, rng),
        init_ansatz(n, rng),
        init_ansatz(n, rng),
    ]


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Primary method: LP-based water-filling (exact direction per step).
    Secondary: subgradient polishing.
    Explores multiple sequence lengths and diverse initializations.
    """
    global rng
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    # Try to load previous best
    prev_a = None
    prev_sc = np.inf
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size >= 10 and np.all(np.isfinite(h)) and h.sum() > 0.01:
            prev_a = h / h.sum()
            prev_sc = 2.0 * h.size * float(np.max(full_conv(prev_a)))
            print(f"[init] Previous best: n={h.size}, score={prev_sc:.8f}")
    except Exception:
        pass

    best_overall = None
    best_overall_sc = np.inf

    if prev_a is not None and prev_sc < 2.0:
        best_overall = prev_a.copy()
        best_overall_sc = prev_sc

    # Sequence lengths to try
    n_list = [64, 128, 256, 512]
    if prev_a is not None:
        np_ = prev_a.size
        if np_ not in n_list and np_ >= 32:
            n_list.append(np_)
        n_list = sorted(set(n_list))

    total_time = deadline - time.monotonic()
    print(f"[plan] n_list={n_list}, total_time={total_time:.1f}s")

    # Allocate time: weight toward smaller n for more LP iterations
    raw_w = [1.0 / (0.4 + nv / 200.0) for nv in n_list]
    total_w = sum(raw_w)
    weights = [w / total_w for w in raw_w]

    for idx_n, nv in enumerate(n_list):
        if time.monotonic() > deadline:
            break

        time_n = weights[idx_n] * total_time * 0.82
        inits = get_inits(nv, rng)

        if prev_a is not None and prev_a.size == nv:
            inits.insert(0, prev_a.copy())

        inits_per = max(3.0, time_n / max(len(inits), 1))

        for ii, a_init in enumerate(inits):
            if time.monotonic() > deadline:
                break

            tb = min(inits_per, deadline - time.monotonic() - 2.0)
            if tb < 2.0:
                break

            print(f"[lp] n={nv}, init={ii}, budget={tb:.1f}s")

            # Phase 1: LP water-filling (70% of time)
            tb_lp = tb * 0.65
            a_best, sc_best, iters = lp_waterfill_optimize(
                a_init.copy(), tb_lp, t_base=0.5, verbose=False)
            print(f"  [lp] done: iter={iters}, score={sc_best:.8f}")

            # Phase 2: Subgradient polish (30% of time)
            tb_sg = tb * 0.30
            if time.monotonic() < deadline - 2.0 and tb_sg > 1.0:
                a_pol, sc_pol = subgrad_optimize(a_best, tb_sg, rng, use_momentum=True)
                if sc_pol < sc_best:
                    sc_best = sc_pol
                    a_best = a_pol
                    print(f"  [sg] polished to {sc_best:.8f}")

            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = a_best.copy()
                print(f"  ** NEW BEST: n={nv}, score={sc_best:.8f}")

        print(f"[summary] n={nv} done. Overall best: {best_overall_sc:.8f}")

    # Final polish phase
    if best_overall is not None and time.monotonic() < deadline - 5.0:
        remain = deadline - time.monotonic() - 3.0
        if remain > 5.0:
            print(f"[polish] remaining={remain:.1f}s, n={best_overall.size}")

            # LP polish with smaller step
            pol_time1 = min(remain * 0.45, 120.0)
            a_p, sc_p, _ = lp_waterfill_optimize(
                best_overall.copy(), pol_time1, t_base=0.25, verbose=False)
            if sc_p < best_overall_sc:
                best_overall_sc = sc_p
                best_overall = a_p.copy()
                print(f"[polish] LP improved to {sc_p:.8f}")

            # Subgradient polish
            if time.monotonic() < deadline - 3.0:
                pol_time2 = min(remain * 0.35, 60.0)
                a_p2, sc_p2 = subgrad_optimize(best_overall, pol_time2, rng, use_momentum=True)
                if sc_p2 < best_overall_sc:
                    best_overall_sc = sc_p2
                    best_overall = a_p2.copy()
                    print(f"[polish] SG improved to {sc_p2:.8f}")

            # One more LP round with very small step
            if time.monotonic() < deadline - 3.0:
                pol_time3 = min(remain * 0.15, 30.0)
                a_p3, sc_p3, _ = lp_waterfill_optimize(
                    best_overall.copy(), pol_time3, t_base=0.1, verbose=False)
                if sc_p3 < best_overall_sc:
                    best_overall_sc = sc_p3
                    best_overall = a_p3.copy()
                    print(f"[polish] fine LP improved to {sc_p3:.8f}")

    # Fallback
    if best_overall is None:
        best_overall = np.ones(128) / 128
        best_overall_sc = 2.0

    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"[final] n={len(result)}, internal={best_overall_sc:.8f}, evaluator={final_sc:.8f}")

    return result
