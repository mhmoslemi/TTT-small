"""
Search for nonnegative step heights minimizing

    2 * n * max(convolve(a, a)) / (sum(a) ** 2).

The search combines several complementary strategies:

  1. FFT-based internal exact scoring.
  2. A spectral "b-flattening" projection method:
       * take current convolution b,
       * project b onto the box b_k <= cap while preserving sum(b)=1,
       * take a square root in the Fourier domain, phase-aligned to current a,
       * project back to the original support and nonnegativity.
  3. L-BFGS-B optimization of a smooth log-sum-exp surrogate in softmax
     coordinates, with beta-annealing and a small sum(b^2) flattening term.
  4. An exact max-subgradient / multiplicative active-set local search,
     using top-lag weights and FFT-correlation gradients.
  5. Multi-start and multi-length exploration, followed by official scoring
     of the best internal candidates.

All helper functions are top level.  The required entry point is
propose_candidate().
"""

import math
import time
import numpy as np

try:
    from scipy.optimize import minimize as _minimize
except Exception:
    _minimize = None

_OPT_BEST_F = 1e100


def _clean(vec):
    """Convert to a 1-D nonnegative finite float array."""
    try:
        arr = np.asarray(vec, dtype=np.float64).ravel()
    except Exception:
        return np.zeros(1, dtype=np.float64)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.maximum(arr, 0.0)
    return arr


def _normalize(a, eps=1e-300):
    """Normalize to sum 1 and keep entries positive."""
    a = _clean(a)
    if a.size == 0:
        return np.ones(0, dtype=np.float64)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a = np.ones_like(a, dtype=np.float64)
        s = float(a.sum())
    a = a / s
    a = np.maximum(a, eps)
    s2 = float(a.sum())
    if not np.isfinite(s2) or s2 <= 0.0:
        a = np.ones_like(a, dtype=np.float64) / float(a.size)
    else:
        a = a / s2
    return a


def _fft_b(a, L, m):
    """Circular convolution a*a on length L, return first m values."""
    A = np.fft.rfft(a, L)
    b = np.fft.irfft(A * A, L)
    if b.size < m:
        b = np.pad(b, (0, m - b.size), mode="constant")
    else:
        b = b[:m]
    b = np.ascontiguousarray(b, dtype=np.float64)
    np.maximum(b, 0.0, out=b)
    return b


def _exact_score_fft(a, L=None):
    """Internal exact score using FFT (matches official for valid inputs)."""
    a = _clean(a)
    n = int(a.size)
    if n == 0:
        return float("inf")
    s = float(a.sum())
    if s < 0.01:
        return float("inf")
    if n == 1:
        return 2.0
    if L is None:
        L = 1 << int((2 * n - 2).bit_length())
    m = 2 * n - 1
    b = _fft_b(a, L, m)
    mb = float(b.max())
    return float(2.0 * n * mb / (s * s))


def _official_score(seq):
    """Call the provided evaluator if available, otherwise use FFT score."""
    try:
        val = evaluate_sequence(seq)
    except Exception:
        val = _exact_score_fft(np.asarray(seq, dtype=np.float64))
    if val is None:
        val = float("inf")
    try:
        val = float(val)
    except Exception:
        val = float("inf")
    return val


def _random_prior(n, rng):
    """Random smooth positive prior from Fourier features plus noise."""
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    t = np.linspace(0.0, 1.0, n)
    z = float(rng.normal(0.0, 0.35))

    kmax = int(min(max(2, n // 512), 28))
    for k in range(1, kmax + 1):
        amp = float(rng.uniform(0.15, 0.65) / math.sqrt(float(k)))
        phc = float(rng.uniform(0.0, 2.0 * np.pi))
        phs = float(rng.uniform(0.0, 2.0 * np.pi))
        z += amp * (0.70 * np.cos(2.0 * np.pi * k * t + phc) +
                    0.70 * np.sin(2.0 * np.pi * k * t + phs))

    noise = rng.normal(0.0, 0.55, n)
    if n > 15:
        kernel = np.ones(15, dtype=np.float64) / 15.0
        noise = np.convolve(noise, kernel, mode="same")
    z += 0.45 * noise

    p = np.exp(z - float(np.max(z)))
    p = np.maximum(p, 1e-300)
    return _normalize(p)


def _sparse_prior(n, rng, kfrac=None):
    """Random sparse positive prior."""
    if n <= 0:
        return np.zeros(0, dtype=np.float64)
    if kfrac is None:
        kfrac = float(rng.uniform(0.08, 0.60))
    kfrac = min(1.0, max(1.0 / float(n), kfrac))
    k = int(max(1, int(round(n * kfrac))))
    a = np.zeros(n, dtype=np.float64)
    if k >= n:
        idx = np.arange(n)
    else:
        mask = rng.random(n)
        idx = np.argpartition(mask, -k)[-k:]
    w = np.exp(rng.normal(0.0, 1.0, idx.size))
    a[idx] = w
    return _normalize(a)


def _shape_candidates(n, rng):
    """A pool of structured and random starting candidates."""
    out = []
    if n <= 0:
        return out

    out.append(_normalize(np.ones(n, dtype=np.float64)))
    for _ in range(5):
        out.append(_random_prior(n, rng))
    for _ in range(3):
        out.append(_sparse_prior(n, rng))

    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0
    zlist = []
    zlist.append(np.zeros(n, dtype=np.float64))
    zlist.append(0.80 * x)
    zlist.append(-0.80 * x)
    zlist.append(0.80 * (x * x - 1.0 / 3.0))
    zlist.append(-0.80 * (x * x - 1.0 / 3.0))

    for k in (1, 2, 3):
        for amp in (0.60, 1.00):
            for ph in (0.0, 0.5 * np.pi, np.pi):
                zlist.append(amp * np.cos(2.0 * np.pi * k * t + ph))

    for c in (0.15, 0.50, 0.85):
        zlist.append(-2.5 * (t - c) ** 2)

    for c1, c2 in ((0.25, 0.75), (0.20, 0.55, 0.80)):
        if len((c1, c2)) == 3:
            zlist.append(-2.5 * (t - c1) ** 2
                         - 2.5 * (t - c2) ** 2
                         - 2.5 * (t - (0.33, 0.60)[1]) ** 2)
        else:
            zlist.append(-2.5 * (t - c1) ** 2 - 2.5 * (t - c2) ** 2)

    for z in zlist:
        z = np.asarray(z, dtype=np.float64)
        sc = float(np.max(np.abs(z)))
        if sc > 1e-12:
            z = z / sc
        p = np.exp(z - float(np.max(z)))
        p = np.maximum(p, 1e-300)
        out.append(_normalize(p))

    return out[:36]


def _resample_length(base, n):
    """Periodic linear resampling of a base sequence to length n."""
    a = _clean(base)
    if a.size == 0:
        return _normalize(np.ones(n, dtype=np.float64))
    if a.size == n:
        return _normalize(a)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        return _normalize(np.ones(n, dtype=np.float64))
    a = a / s
    x0 = np.linspace(0.0, 1.0, a.size, endpoint=False)
    x1 = np.linspace(0.0, 1.0, n, endpoint=False)
    y0 = np.concat(("constant" in dir() and np.zeros(0) or np.array(()), np.array(())))  # no-op placeholder
    yy = np.interp(
        x1,
        np.concatenate((x0, [1.0])),
        np.concatenate((a, [a[0]]))
    )
    return _normalize(yy)


def _soft_p(b, beta):
    """Soft-max distribution over b values."""
    beta = max(1e-300, float(beta))
    vb = beta * b
    vm = float(np.max(vb)) if vb.size else 0.0
    z = vb - vm
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        e = np.ones_like(b, dtype=np.float64)
        s = float(b.size)
    p = e / s
    return p, float(vm + math.log(s))


def _top_p(b, frac):
    """Weight distribution concentrated on top lags."""
    b = np.asarray(b, dtype=np.float64)
    m = int(b.size)
    if m == 0:
        return b
    frac = min(1.0, max(1e-6, float(frac)))
    k = int(max(1, int(round(m * frac))))
    s = float(b.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(m, 1.0 / m, dtype=np.float64)
    if k >= m:
        return b / s
    idx = np.argpartition(b, -k)[-k:]
    th = float(b[idx].min())
    p = np.maximum(b - th, 0.0)
    ps = float(p.sum())
    if not np.isfinite(ps) or ps <= 0.0:
        p = np.zeros_like(b, dtype=np.float64)
        p[idx] = 1.0 / float(idx.size)
    else:
        p = p / ps
    return p


def _corr_rfft(a, p, L, n):
    """
    Compute g_i = sum_k p_k a_{k-i}, the derivative of the smoothed
    convolution objective with respect to a_i.
    """
    A = np.fft.rfft(a, L)
    P = np.fft.rfft(p, L)
    g = np.fft.irfft(A * np.conj(P), L)[:n]
    g = np.ascontiguousarray(g, dtype=np.float64)
    np.maximum(g, 0.0, out=g)
    return g


def _water_cap(b, cap):
    """
    Project a probability vector b onto {x >= 0, sum x = 1, x_k <= cap}.
    Uses binary search on the uniform additive shift lambda:
        x_k = min(b_k + lambda, cap), sum x = 1.
    """
    b = _clean(b)
    m = int(b.size)
    if m == 0:
        return b
    s = float(b.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(m, 1.0 / m, dtype=np.float64)
    b = b / s
    cap = float(cap)
    avg = 1.0 / float(m)
    if not np.isfinite(cap) or cap <= avg:
        return np.full(m, avg, dtype=np.float64)
    peak = float(b.max())
    if peak <= cap:
        return b

    lo = 0.0
    hi = max(cap - float(b.min()) + 1e-12, 1e-12)

    # Ensure f(hi) >= 0.
    for _ in range(8):
        if float(np.sum(np.minimum(b + hi, cap))) - 1.0 >= 0.0:
            break
        hi *= 2.0

    for _ in range(36):
        mid = 0.5 * (lo + hi)
        val = float(np.sum(np.minimum(b + mid, cap))) - 1.0
        if val > 0.0:
            hi = mid
        else:
            lo = mid

    lam = 0.5 * (lo + hi)
    out = np.minimum(b + lam, cap)
    s2 = float(out.sum())
    if not np.isfinite(s2) or s2 <= 0.0:
        out = np.full(m, avg, dtype=np.float64)
    else:
        out = out / s2
    return out


def _aligned_sqrt_rfft(B, A):
    """Complex square root of B, choosing signs to stay close to A."""
    B = np.asarray(B, dtype=np.complex128).copy()
    R = np.sqrt(B)
    d1 = np.abs(R - A)
    d2 = np.abs(-R - A)
    neg = d2 < d1
    R[neg] = -R[neg]

    re0 = float(np.real(B[0]))
    R[0] = np.sqrt(re0 if re0 > 0.0 else 0.0)

    if R.size > 1:
        rel = float(np.real(B[-1]))
        rl = np.sqrt(rel if rel > 0.0 else 0.0)
        if float(np.real(A[-1])) < 0.0:
            rl = -rl
        R[-1] = rl

    return R


def _theta_to_a(x, n):
    """Map theta coordinates (theta_0 fixed to 0) to probability a."""
    x = np.asarray(x, dtype=np.float64).ravel()
    if x.size != max(0, n - 1):
        if n <= 1:
            return np.ones(n, dtype=np.float64) / max(1, n)
        x = np.resize(x, n - 1)
    th = np.empty(n, dtype=np.float64)
    th[0] = 0.0
    th[1:] = np.clip(x, -40.0, 40.0)
    mm = float(np.max(th))
    e = np.exp(th - mm)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.ones(n, dtype=np.float64) / float(n)
    return e / s


def _initial_theta(a, n):
    """Theta coordinates relative to index 0."""
    a = _normalize(a)
    if n <= 1:
        return np.zeros(max(0, n - 1), dtype=np.float64)
    aa = np.maximum(a, 1e-300)
    th = np.log(aa) - float(np.log(aa[0]))
    th = np.clip(th, -40.0, 40.0)
    return th[1:]


def _smooth_loss_grad(x, n, L, m, beta, mu):
    """
    Smooth surrogate:
        F = (1/beta) logsumexp(beta * b) + mu * sum(b^2)
    with b = convolve(a, a), a = softmax([0, x]).
    Returns (F, grad_x).
    """
    a = _theta_to_a(x, n)
    A = np.fft.rfft(a, L)
    b = np.fft.irfft(A * A, L)[:m]
    b = np.maximum(b, 0.0)

    beta = max(1e-300, float(beta))
    mu = float(mu)

    bb = beta * b
    vm = float(np.max(bb)) if bb.size else 0.0
    z = bb - vm
    z = np.clip(z, -700.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        e = np.ones_like(bb, dtype=np.float64)
        s = float(bb.size)
    p = e / s
    lse = float(vm + math.log(s))

    sumb2 = float(np.dot(b, b))
    F = float(lse / beta + mu * sumb2)

    pg = p if mu == 0.0 else p + 2.0 * mu * b
    w = _corr_rfft(a, pg, L, n)
    inner = float(np.dot(w, a))
    grad_all = w - a * inner
    grad = grad_all[1:]
    if not np.all(np.isfinite(grad)):
        grad = np.zeros(max(0, n - 1), dtype=np.float64)
    return F, grad


def _lbfgs_fun(x):
    global _LAST_X, _OPT_BEST_X, _OPT_BEST_F
    if time.monotonic() > _OPT_DEADLINE:
        raise _StopOpt
    x = np.asarray(x, dtype=np.float64).copy()
    _LAST_X = x
    F, g = _smooth_loss_grad(x, _OPT_N, _OPT_L, _OPT_M, float(_OPT_BETA), float(_OPT_MU))
    if np.isfinite(F) and F < _OPT_BEST_F:
        _OPT_BEST_F = float(F)
        _OPT_BEST_X = x.copy()
    if time.monotonic() > _OPT_DEADLINE:
        raise _StopOpt
    return F, g


def _lbfgs_callback(xk):

    _LAST_X = np.asarray(xk, dtype=np.float64).copy()

def _lbfgs_optimize(a, n, L, m, time_limit, rng):
    """Beta-annealed L-BFGS-B on the smooth surrogate."""
    a = _normalize(a)
    base_score = _exact_score_fft(a, L)
    if n <= 1 or time_limit < 15.0 or _minimize is None:
        return a, base_score

    deadline = time.monotonic() + max(5.0, float(time_limit))
    x = _initial_theta(a, n)

    if time_limit < 80.0:
        schedule = [10.0, 100.0, 1000.0, 8000.0]
    elif time_limit < 250.0:
        schedule = [5.0, 20.0, 120.0, 600.0, 3000.0, 15000.0]
    else:
        schedule = [2.0, 10.0, 50.0, 250.0, 1200.0, 6000.0, 25000.0, 80000.0]

    # Drop too many betas if time is very short.
    max_betas = max(2, int(max(5.0, time_limit) / 30.0))
    schedule = schedule[:max_betas]

    a_cur = _theta_to_a(x, n)
    best_a = a_cur.copy()
    best_score = _exact_score_fft(a_cur, L)
    bounds = [(-40.0, 40.0)] * max(0, n - 1)

    time_per = max(8.0, float(time_limit) / max(1, len(schedule)))

    for beta in schedule:
        now = time.monotonic()
        if now >= deadline - 5.0:
            break

        global _OPT_N, _OPT_L, _OPT_M, _OPT_BETA, _OPT_MU, _OPT_DEADLINE
        global _OPT_BEST_X, _OPT_BEST_F

        _OPT_N = int(n)
        _OPT_L = int(L)
        _OPT_M = int(m)
        _OPT_BETA = float(beta)
        if beta < 50.0:
            _OPT_MU = 0.25
        elif beta < 500.0:
            _OPT_MU = 0.10
        elif beta < 5000.0:
            _OPT_MU = 0.03
        else:
            _OPT_MU = 0.00
        _OPT_DEADLINE = min(deadline, now + time_per + 1.0)
        _OPT_BEST_X = x.copy() if x.size else np.zeros(0, dtype=np.float64)
        _OPT_BEST_F = 1e100

        maxiter = 120
        if time_per > 60.0:
            maxiter = 300
        if time_per > 180.0:
            maxiter = 600

        try:
            res = _minimize(
                _lbfgs_fun,
                x,
                jac=True,
                method="L-BFGS-B",
                bounds=bounds,
                options={
                    "maxiter": maxiter,
                    "maxls": 30,
                    "maxcor": 20,
                    "gtol": 1e-5,
                    "ftol": 1e-12,
                    "disp": False,
                },
                callback=_lbfgs_callback,
            )
            if getattr(res, "x", None) is not None:
                x = np.asarray(res.x, dtype=np.float64)
        except _StopOpt:
            pass
        except Exception:
            pass

        if _OPT_BEST_X is not None and np.size(_OPT_BEST_X) == max(0, n - 1):
            x = np.asarray(_OPT_BEST_X, dtype=np.float64).copy()
        elif _LAST_X is not None and np.size(_LAST_X) == max(0, n - 1):
            x = np.asarray(_LAST_X, dtype=np.float64).copy()

        a_cur = _theta_to_a(x, n)
        sc = _exact_score_fft(a_cur, L)
        if np.isfinite(sc) and sc < best_score - 1e-12:
            best_score = sc
            best_a = a_cur.copy()

        # Re-align theta to the current point before next beta.
        x = _initial_theta(a_cur, n)

    return best_a, best_score


def _active_search(a, n, L, m, time_limit, rng):
    """
    Exact max-subgradient multiplicative search.
    Uses top-lag / softmax weights and correct FFT-correlation gradients.
    """
    a = _normalize(a)
    if time_limit <= 0.0:
        return a, _exact_score_fft(a, L)

    cur = a.copy()
    b = _fft_b(cur, L, m)
    cur_score = _exact_score_fft(cur, L)

    best_a = cur.copy()
    best_score = cur_score

    eta = 0.35
    frac = 0.005
    stall = 0
    it = 0
    deadline = time.monotonic() + max(0.0, float(time_limit))

    while time.monotonic() < deadline:
        b = _fft_b(cur, L, m)
        peak = float(b.max())
        if not np.isfinite(peak) or peak <= 0.0:
            break

        beta = max(20.0, 8.0 / max(peak, 1e-300))
        p_soft, _ = _soft_p(b, beta)
        p_top = _top_p(b, frac)
        mix = 0.60 if frac < 0.02 else 0.35
        p = (1.0 - mix) * p_soft + mix * p_top
        ps = float(p.sum())
        if not np.isfinite(ps) or ps <= 0.0:
            p = p_top
        else:
            p = p / ps

        g = _corr_rfft(cur, p, L, n)
        gs = float(g.max())
        if not np.isfinite(gs) or gs <= 1e-300:
            break
        d = g / gs

        lams = (eta, 0.5 * eta, 2.0 * eta, 0.25 * eta)
        best_step = None
        best_sc = cur_score
        best_a2 = None

        for lam in lams:
            if not np.isfinite(lam) or lam <= 1e-12:
                continue
            try:
                a2 = cur * np.exp(-lam * d)
            except Exception:
                continue
            a2 = _normalize(a2)
            b2 = _fft_b(a2, L, m)
            sc = _exact_score_fft(a2, L)
            if np.isfinite(sc) and sc < best_sc:
                best_sc = sc
                best_step = lam
                best_a2 = a2

        if best_a2 is not None:
            cur = best_a2
            cur_score = best_sc
            stall = 0
            if abs(best_step - eta) < 1e-12:
                eta = min(1.60, eta * 1.08)
            elif best_step > eta + 1e-12:
                eta = min(2.20, eta * 1.20)
            else:
                eta = max(0.05, eta * 0.85)
            if best_sc < best_score - 1e-12:
                best_score = best_sc
                best_a = cur.copy()
        else:
            stall += 1
            eta = max(0.03, eta * 0.70)
            frac = min(0.22, frac * 1.10 + 0.001)
            if stall > 25:
                prior = _random_prior(n, rng)
                mixr = 0.20 + 0.35 * float(rng.random())
                cur = _normalize((1.0 - mixr) * best_a + mixr * prior)
                b = _fft_b(cur, L, m)
                cur_score = _exact_score_fft(cur, L)
                if cur_score < best_score - 1e-12:
                    best_score = cur_score
                    best_a = cur.copy()
                eta = 0.35
                frac = 0.005
                stall = 0

        it += 1
        if (it % 200) == 0:
            print(f"    active: it={it} score={best_score:.10f}")

    return best_a, best_score


def _spectral_search(a, n, L, m, time_limit, rng):
    """
    Spectral flattening search.

    Alternates between:
      - projecting b = a*a onto a max-peak box,
      - taking a phase-aligned Fourier square root,
      - projecting back to support and nonnegativity.
    """
    a = _normalize(a)
    if n <= 1 or time_limit <= 0.0:
        return a, _exact_score_fft(a, L)

    a_full = np.zeros(L, dtype=np.float64)
    a_full[:n] = a
    A = np.fft.rfft(a_full, L)

    b = _fft_b(a, L, m)
    cur_score = _exact_score_fft(a, L)

    best_a = a.copy()
    best_score = cur_score
    cur_a = a.copy()

    min_cap = 1.05 / float(m)
    cap = max(min_cap, (cur_score / (2.0 * n)) * 0.985)

    stall = 0
    it = 0
    deadline = time.monotonic() + max(0.0, float(time_limit))

    while time.monotonic() < deadline:
        b = _fft_b(cur_a, L, m)
        peak = float(b.max())
        if not np.isfinite(peak) or peak <= 0.0:
            break

        best_cap = best_score / (2.0 * n)
        desired = min(cap, peak * 0.995)
        if desired > best_cap * 1.01:
            desired = best_cap * 0.995
        if desired >= peak:
            desired = peak * 0.985
        desired = max(min_cap, desired)

        b2 = _water_cap(b, desired)
        b2_full = np.zeros(L, dtype=np.float64)
        b2_full[:m] = b2

        B2 = np.fft.rfft(b2_full, L)
        A2 = _aligned_sqrt_rfft(B2, A)

        # Tiny stochastic sign flip to escape symmetric traps.
        if A2.size > 1 and float(rng.random()) < 0.01:
            idx = int(rng.integers(1, A2.size))
            A2[idx] = -A2[idx]

        a2 = np.fft.irfft(A2, L)[:n]
        a2 = np.maximum(a2, 0.0)
        s2 = float(a2.sum())
        if not np.isfinite(s2) or s2 <= 1e-300:
            a2 = cur_a.copy()
        else:
            a2 = a2 / s2

        score2 = _exact_score_fft(a2, L)
        if not np.isfinite(score2):
            score2 = best_score + 1.0

        if score2 < best_score - 1e-12:
            best_score = score2
            best_a = a2.copy()
            stall = 0
            cap = max(min_cap, (score2 / (2.0 * n)) * 0.997)
        else:
            stall += 1
            if cap > peak * 0.99:
                cap = peak * 0.980
            else:
                cap *= 0.985
            cap = max(min_cap, cap)

            if stall > 60:
                prior = _random_prior(n, rng)
                mixr = 0.15 + 0.30 * float(rng.random())
                a2 = _normalize((1.0 - mixr) * best_a + mixr * prior)
                score2 = _exact_score_fft(a2, L)
                cur_a = a2
                if np.isfinite(score2) and score2 < best_score - 1e-12:
                    best_score = score2
                    best_a = cur_a.copy()
                stall = 0
                cap = max(min_cap, (best_score / (2.0 * n)) * 0.995)

        cur_a = a2
        a_full[:n] = cur_a
        a_full[n:] = 0.0
        A = np.fft.rfft(a_full, L)

        it += 1
        if (it % 200) == 0:
            print(f"    spectral: it={it} score={best_score:.10f}")

    return best_a, best_score


def _add_candidate(arr, score):
    global _CAND_ARRAYS, _CAND_SCORES
    if arr is None:
        return
    try:
        score = float(score)
    except Exception:
        return
    if not np.isfinite(score):
        return
    a = _normalize(arr)
    if a.size == 0:
        return
    _CAND_ARRAYS.append(a)
    _CAND_SCORES.append(score)
    if len(_CAND_ARRAYS) > 256:
        arrs = np.asarray(_CAND_SCORES, dtype=np.float64)
        order = np.argsort(arrs)[:128]
        _CAND_ARRAYS = [_CAND_ARRAYS[int(i)] for i in order]
        _CAND_SCORES = [float(_CAND_SCORES[int(i)]) for i in order]


def _seq_from_arr(arr):
    a = _normalize(arr)
    if a.size == 0:
        a = np.ones(1, dtype=np.float64)
    return [float(max(0.0, x)) for x in a]


def _run_length(n, alloc, rng, parent_arr, search_deadline):
    """Run a complete pipeline for one sequence length."""
    if n <= 1 or alloc <= 5.0:
        return

    n = int(n)
    L = 1 << int((2 * n - 2).bit_length())
    m = 2 * n - 1

    inits = []
    if parent_arr is not None:
        inits.append(_resample_length(parent_arr, n))
    inits.extend(_shape_candidates(n, rng))
    inits = inits[:24]

    scored = []
    for idx, a in enumerate(inits):
        if a is None or a.size != n:
            continue
        sc = _exact_score_fft(a, L)
        if np.isfinite(sc):
            scored.append((sc, idx, a))

    if not scored:
        a = _normalize(np.ones(n, dtype=np.float64))
        sc = _exact_score_fft(a, L)
        scored = [(sc, 0, a)]

    scored.sort()
    for sc, idx, a in scored:
        _add_candidate(a, sc)

    a_work = scored[0][2]
    best_local = scored[0][0]

    # --- Spectral phase: top initial candidate, then a second if available.
    spec_budget = 0.34 * float(alloc)
    spec_deadline = time.monotonic() + spec_budget
    starts = [scored[0][2]]
    if len(scored) > 1:
        starts.append(scored[1][2])

    for start_a in starts:
        rem = min(spec_deadline, search_deadline) - time.monotonic()
        if rem < 5.0:
            break
        lim = min(rem, max(5.0, 0.22 * float(alloc)))
        a_sc, sc = _spectral_search(start_a, n, L, m, lim, rng)
        _add_candidate(a_sc, sc)
        if np.isfinite(sc) and sc < best_local - 1e-12:
            best_local = sc
            a_work = a_sc

    # --- L-BFGS phase
    lbf_budget = 0.30 * float(alloc)
    lim = min(lbf_budget, search_deadline - time.monotonic())
    if lim > 15.0:
        a_sc, sc = _lbfgs_optimize(a_work, n, L, m, lim, rng)
        _add_candidate(a_sc, sc)
        if np.isfinite(sc) and sc < best_local - 1e-12:
            best_local = sc
            a_work = a_sc
        else:
            # Sometimes the L-BFGB path does not improve the exact score;
            # a fresh spectral start from the prior best is cheap.
            if search_deadline - time.monotonic() > 20.0:
                a_sc, sc = _spectral_search(best_local if np.isfinite(sc) else a_work, n, L, m,
                                            min(search_deadline - time.monotonic(), 20.0), rng)
                _add_candidate(a_sc, sc)
                if np.isfinite(sc) and sc < best_local - 1e-12:
                    best_local = sc
                    a_work = a_sc

    # --- Active exact local search
    act_lim = search_deadline - time.monotonic()
    if act_lim > 5.0 and a_work is not None:
        a_sc, sc = _active_search(a_work, n, L, m, act_lim, rng)
        _add_candidate(a_sc, sc)
        if np.isfinite(sc) and sc < best_local - 1e-12:
            best_local = sc
            a_work = a_sc

    # --- Final spectral polish if there is time.
    rem = search_deadline - time.monotonic()
    if rem > 25.0 and a_work is not None:
        lim = min(rem, max(10.0, 0.15 * float(alloc)))
        a_sc, sc = _spectral_search(a_work, n, L, m, lim, rng)
        _add_candidate(a_sc, sc)
        if np.isfinite(sc) and sc < best_local - 1e-12:
            best_local = sc
            a_work = a_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point.  Returns the best nonnegative sequence found.
    """
    global _CAND_ARRAYS, _CAND_SCORES

    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget <= 0.0:
        budget = 2400.0
    budget = max(60.0, budget)

    start_t = time.monotonic()
    reserve = min(240.0, max(30.0, 0.08 * budget))
    search_budget = max(20.0, budget - reserve)
    search_deadline = start_t + search_budget

    _CAND_ARRAYS = []
    _CAND_SCORES = []

    rng = np.random.default_rng(seed)

    parent_arr = None
    parent_len = 0
    try:
        base = height_sequence_1
    except Exception:
        base = None
    if base is not None:
        try:
            b = _clean(base)
            if b.size > 0:
                s = float(b.sum())
                if np.isfinite(s) and s > 0.0:
                    parent_arr = _normalize(b)
                    parent_len = int(parent_arr.size)
        except Exception:
            parent_arr = None
            parent_len = 0

    # Choose lengths to explore.
    lengths = []
    if parent_len > 1:
        n1 = parent_len
    else:
        n1 = 8192
    if n1 > 0:
        lengths.append(n1)

    n2 = 8192
    if n2 not in lengths:
        lengths.append(n2)

    if budget > 1200.0:
        n3 = 10240
        if n3 not in lengths:
            lengths.append(n3)

    # Fractions.  If only one length, use all time.
    if len(lengths) == 1:
        fracs = [1.0]
    elif len(lengths) == 2:
        fracs = [0.60, 0.40]
    else:
        fracs = [0.50, 0.30, 0.20]

    print(f"propose_candidate: budget={budget:.1f}s lengths={lengths} fracs={fracs}")

    cum = 0.0
    for n, frac in zip(lengths, fracs):
        cum += frac
        local_deadline = start_t + cum * search_budget
        alloc = max(0.0, local_deadline - time.monotonic())
        if alloc < 5.0:
            continue
        n_use = int(n)
        if n_use <= 1:
            continue
        try:
            _run_length(n_use, alloc, rng, parent_arr, search_deadline)
        except Exception:
            pass

    # Final official scoring of the best internal candidates.
    best_seq = None
    best_official = float("inf")

    if _CAND_SCORES:
        try:
            scores = np.asarray(_CAND_SCORES, dtype=np.float64)
            order = np.argsort(scores)
        except Exception:
            order = np.arange(len(_CAND_SCORES))

        # Conservative limit to keep official O(n^2) scoring within reserve.
        max_evals = 14
        if _CAND_ARRAYS and _CAND_ARRAYS[0].size > 9000:
            max_evals = 8
        if budget < 600.0:
            max_evals = min(max_evals, 5)

        eval_count = 0
        for idx in order:
            idx = int(idx)
            eval_count += 1
            if eval_count > max_evals:
                break
            if time.monotonic() > start_t + budget - 8.0:
                break
            arr = _CAND_ARRAYS[idx]
            seq = _seq_from_arr(arr)
            val = _official_score(seq)
            if np.isfinite(val) and val < best_official:
                best_official = val
                best_seq = seq
            print(f"  official candidate {idx}: score={val:.10f}")

    if best_seq is None:
        if _CAND_ARRAYS:
            # Pick the best internal candidate by internal score.
            try:
                scores = np.asarray(_CAND_SCORES, dtype=np.float64)
                idx = int(np.argmin(scores))
            except Exception:
                idx = 0
            best_seq = _seq_from_arr(_CAND_ARRAYS[idx])
        else:
            best_seq = _seq_from_arr(parent_arr if parent_arr is not None else np.ones(6121, dtype=np.float64))

    # Safety: ensure a valid simple list of nonnegative floats.
    out = []
    for v in best_seq:
        try:
            f = float(v)
        except Exception:
            f = 0.0
        if not np.isfinite(f) or f < 0.0:
            f = 0.0
        out.append(f)
    s = sum(out)
    if not np.isfinite(s) or s <= 0.01:
        out = _seq_from_arr(np.ones(6121, dtype=np.float64))

    print(f"propose_candidate done: elapsed={time.monotonic() - start_t:.1f}s len={len(out)}")
    return out
