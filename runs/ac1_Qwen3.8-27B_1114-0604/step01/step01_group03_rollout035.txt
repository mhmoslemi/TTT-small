def _binomial_a(n):
    a=np.empty(n, dtype=np.float64)
    if n<=0: return np.array([1.0])
    a[0]=1.0
    for i in range(1,n):
        a[i]=a[i-1] * ((2.0*i - 1.0)/(2.0*i))
    s=a.sum()
    if s<=0: a.fill(1.0)
    else: a/=s
    return a

```python
"""
Search for nonnegative step heights minimizing

    2*n*max(conv(a,a)) / (sum(a)**2)

for variable length n.  The main idea is to start close to the algebraic
family whose infinite product is flat:

    coefficients of (1-x)^(-1/2) = binom(2i,i)/4^i.

This gives an asymptotic score of pi/2.  The code then uses a fast
FFT-based multiplicative projected-gradient search with log-sum-exp
active-set weights, low-dimensional basis coarse-tuning, restarts, and a
final exact/A considered score check.
"""

import math
import time

import numpy as np


def _prepare_a(arr):
    a = np.asarray(arr, dtype=np.float64).ravel()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a.fill(1.0)
        s = float(a.sum())
    a /= s
    return a


def _fft_len(n):
    m = 2 * n - 1
    L = 1
    while L < m:
        L <<= 1
    return L, m


def _conv_self_fa(a, L):
    fa = np.fft.rfft(a, L)
    c = np.fft.irfft(fa * fa, L)
    m = 2 * len(a) - 1
    c = c[:m]
    c = np.where(np.isfinite(c), c, 0.0)
    np.maximum(c, 0.0, out=c)
    return c, fa


def _conv_self(a, L):
    c, _ = _conv_self_fa(a, L)
    return c


def _score_c(c, n):
    if c.size == 0:
        return float("inf")
    mc = float(np.max(c))
    if not np.isfinite(mc) or mc < 0.0:
        mc = 0.0
    return float(2.0 * n * mc)


def _hankel_grad(a, w, L, fa=None):
    n = len(a)
    m = 2 * n - 1
    if fa is None:
        fa = np.fft.rfft(a, L)
    wr = np.ascontiguousarray(w[::-1])
    fw = np.fft.rfft(wr, L)
    conv = np.fft.irfft(fa * fw, L)
    conv = np.where(np.isfinite(conv), conv, 0.0)

    if n == 1:
        return np.array(2.0 * float(a[0]) * float(w[0]), dtype=np.float64)

    off = m - 1
    seg = conv[off:off - n:-1]
    g = 2.0 * seg
    g = np.ascontiguousarray(g, dtype=np.float64)
    g = np.where(np.isfinite(g), g, 0.0)
    np.maximum(g, 0.0, out=g)
    return g


def _weights_lse(c, cm, temp_frac):
    m = len(c)
    if m == 0 or cm <= 0.0 or not np.isfinite(cm):
        return np.full(m, 1.0 / max(1, m), dtype=np.float64)

    T = float(temp_frac) * float(cm)
    if not np.isfinite(T) or T <= 0.0:
        T = max(1e-300, 1e-8 * float(cm))

    z = (c - cm) / T
    np.clip(z, -60.0, 0.0, out=z)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        return np.full(m, 1.0 / m, dtype=np.float64)
    return e / s


def _softmax_v(v):
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.where(np.isfinite(v), v, 0.0)
    x = v - float(np.max(v))
    e = np.exp(x)
    s = float(e.sum())
    n = len(v)
    if not np.isfinite(s) or s <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return e / s


def _prep_v(v, clip=20.0):
    v = np.asarray(v, dtype=np.float64).copy()
    v = np.where(np.isfinite(v), v, 0.0)
    v = v - float(v.mean())
    v = np.clip(v, -clip, clip)
    v = v - float(v.mean())
    return v


def _log_a(a):
    a = np.asarray(a, dtype=np.float64)
    v = np.log(np.maximum(a, 1e-15))
    return _prep_v(v, 30.0)


def _score_v(v, n, L):
    a = _softmax_v(v)
    c = _conv_self(a, L)
    return _score_c(c, n), a


def _binomial_a(n):
    a = np.empty(max(1, n), dtype=np.float64)
    if n <= 0:
        a[0] = 1.0
        return a
    a[0] = 1.0
    for i in range(1, n):
        a[i] = a[i - 1] * ((2.0 * i - 1.0) / (2.0 * i))
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a.fill(1.0)
        s = float(a.sum())
    a /= s
    return a


def _binomial_v(n):
    return _log_a(_binomial_a(n))


def _interp_v(v, n_from, n_to):
    v = np.asarray(v, dtype=np.float64).ravel()
    if n_from == n_to:
        return _prep_v(v, 30.0)
    x0 = np.linspace(0.0, 1.0, n_from)
    x1 = np.linspace(0.0, 1.0, n_to)
    y = np.interp(x1, x0, v)
    return _prep_v(y, 30.0)


def _blend_v(v1, v2, alpha):
    v1 = np.asarray(v1, dtype=np.float64).ravel()
    v2 = np.asarray(v2, dtype=np.float64).ravel()
    v1 = v1 - float(v1.mean())
    v2 = v2 - float(v2.mean())
    v = alpha * v1 + (1.0 - alpha) * v2
    return _prep_v(v, 30.0)


def _add_candidate_v(cands, v):
    cands.append(_prep_v(v, 30.0))


def _expand_initial_v(a, n, L):
    out = []
    a = _prepare_a(a)
    v0 = _log_a(a)
    _add_candidate_v(out, v0)

    asym = 0.5 * (a + a[::-1])
    s = float(asym.sum())
    if s > 0.0:
        asym /= s
    _add_candidate_v(out, _log_a(asym))

    for w in (3, 5):
        k = np.ones(w, dtype=np.float64) / float(w)
        sm = np.convolve(a, k, mode="same")
        sm = np.maximum(sm, 0.0)
        s = float(sm.sum())
        if s > 0.0:
            sm /= s
            _add_candidate_v(out, _log_a(sm))

    for p in (0.8, 0.9, 1.1, 1.2, 1.5):
        b = np.power(np.maximum(a, 1e-15), float(p))
        s = float(b.sum())
        if s > 0.0:
            b /= s
            _add_candidate_v(out, _log_a(b))

    x = np.linspace(0.0, 1.0, n)
    shapes = []
    shapes.append(-(x - 0.5) ** 2)
    shapes.append((x - 0.5) ** 2)
    shapes.append(x - 0.5)
    shapes.append(0.5 - x)
    shapes.append(np.sin(math.pi * x))
    shapes.append(-np.sin(math.pi * x))

    for sh in shapes:
        for sc in (0.05, 0.15):
            _add_candidate_v(out, v0 + sc * sh)

    return out


def _make_basis(n, K=12):
    x = np.linspace(0.0, 1.0, n)
    cols = [
        x - 0.5,
        (x - 0.5) ** 2 - 1.0 / 12.0,
        (x - 0.5) ** 3,
    ]
    k = 1
    while len(cols) < K:
        cols.append(np.sin(math.pi * k * x) / math.sqrt(float(k)))
        cols.append(np.cos(math.pi * k * x) / math.sqrt(float(k)))
        k += 1
    if len(cols) > K:
        cols = cols[:K]
    return np.vstack(cols)


def _basis_coarse(v, n, L, B, max_evals):
    v = _prep_v(v, 30.0)
    a = _softmax_v(v)
    c, _ = _conv_self_fa(a, L)
    best_v = v.copy()
    best_a = a.copy()
    best_sc = _score_c(c, n)

    if B is None or B.size == 0 or max_evals <= 0:
        return best_v, best_a, best_sc

    used = 0
    for pass_idx in range(6):
        improved = False
        for k in range(B.shape[0]):
            for delta in (-0.2, -0.1, -0.03, 0.03, 0.1, 0.2):
                if used >= max_evals:
                    break
                v_try = _prep_v(best_v + float(delta) * B[k], 30.0)
                sc_try, a_try = _score_v(v_try, n, L)
                used += 1
                if sc_try < best_sc - 1e-12:
                    best_sc = sc_try
                    best_v = v_try
                    best_a = a_try
                    improved = True
            if used >= max_evals:
                break
        if not improved or used >= max_evals:
            break

    return best_v, best_a, best_sc


def _make_initial_v(n, L, prev_v, rng, max_cands=70):
    base = _binomial_v(n)
    cands = []
    _add_candidate_v(cands, base)

    prev_use = None
    if prev_v is not None:
        if len(prev_v) == n:
            prev_use = _prep_v(prev_v, 30.0)
            _add_candidate_v(cands, prev_use)
        else:
            vi = _interp_v(prev_v, len(prev_v), n)
            for alpha in (0.5, 0.7, 0.9):
                _add_candidate_v(cands, _blend_v(vi, base, alpha))
            prev_use = _blend_v(vi, base, 0.8)
            _add_candidate_v(cands, prev_use)

    seeds = [base]
    if prev_use is not None:
        seeds.append(prev_use)

    for seed_v in seeds:
        aseed = _softmax_v(seed_v)
        ext = _expand_initial_v(aseed, n, L)
        for v in ext:
            if len(cands) >= max_cands:
                break
            _add_candidate_v(cands, v)
        if len(cands) >= max_cands:
            break

    B = _make_basis(n, 12)
    for _ in range(20):
        if len(cands) >= max_cands:
            break
        beta = rng.normal(0.0, 0.12, 12)
        beta = np.clip(beta, -0.6, 0.6)
        _add_candidate_v(cands, base + B.dot(beta))

    best_v = base.copy()
    best_score, best_a = _score_v(best_v, n, L)

    for v in cands:
        sc, a = _score_v(v, n, L)
        if sc < best_score - 1e-15:
            best_score = sc
            best_v = v
            best_a = a

    return best_v, best_a, best_score

        return None

    rng = np.random.default_rng(seed)

    prev_v = None
    if prev_info is not None:
        prev_v = prev_info[0]

    if budget < 60.0:
        max_cands = 8
    elif budget < 200.0:
        max_cands = 25
    else:
        max_cands = 70

    v0, a0, sc0 = _make_initial_v(n, L, prev_v, rng, max_cands)
    B = _make_basis(n, 12)

    elapsed = time.monotonic() - start

        max_evals = int(min(80, max(20, max(1.0, budget - elapsed) / 20.0)))
        v0, a0, sc0 = _basis_coarse(v0, n, L, B, max_evals)

    elapsed = time.monotonic() - start
    search_budget = max(5.0, budget - elapsed)
    best_a, best_v, best_sc = _run_search(a0, v0, n, L, m, search_budget, seed + 1, label)

        cv, ca, csc = _basis_coarse(best_v, n, L, B, max_evals)
        if csc < best_sc:
            best_a = ca
            best_v = cv
            best_sc = csc

    return best_a, best_sc, best_v, n, L, m

        return float("inf")

def _sort_candidates(candidates, count):
    arr = list(candidates)
    for i in range(len(arr)):
        for j in range(i + 1, len(arr)):
            if arr[j][1] < arr[i][1]:
                arr[i], arr[j] = arr[j], arr[i]
    if len(arr) > count:
        del arr[count:]
    return arr

        scaled = []
        for n, t, w in tasks:
            scaled.append((n, t, float(w) / total_w))
        tasks = scaled

    if budget >= 1500.0:
        reserve = 120.0
    elif budget >= 800.0:
        reserve = 80.0
    elif budget >= 400.0:
        reserve = 40.0
    else:
        reserve = 20.0

    search_end = max(start + 30.0, end_limit - reserve)
    planned_total = max(1.0, search_end - start)

    candidates = []

    for idx, (n, typ, w) in enumerate(tasks):
        now = time.monotonic()
        if now >= search_end:
            break

        task_time = max(15.0, min(float(w) * planned_total, search_end - now))
        if task_time < 8.0:
            continue

        try:
            task_seed = int(rng_parent.integers(0, 2**31 - 1))
            res = _run_task(n, typ, (prev_v, prev_a) if prev_v is not None else None, task_time, task_seed, "task%d-n%d" % (idx, n))
            if res is not None:
                a, sc, v, nn, L, m = res
                if a is not None and np.isfinite(sc):
                    candidates.append((a, float(sc), nn, L, m, v))
                    print("candidate n=%d internal=%.7f" % (nn, float(sc)), flush=True)
        except Exception as e:
            print("task failed n=%d err=%s" % (n, repr(e)), flush=True)

    if not candidates:
        if prev_a is not None:
            fallback_a = prev_a
        else:
            fallback_a = _binomial_a(6121)
        nn = len(fallback_a)
        L, m = _fft_len(nn)
        fallback_v = _log_a(fallback_a)
        fallback_c = _conv_self(fallback_a, L)
        fallback_sc = _score_c(fallback_c, nn)
        candidates.append((fallback_a, fallback_sc, nn, L, m, fallback_v))

    ordered = _sort_candidates(candidates, 3)
    top = ordered[0]
    best_a, best_sc, nn, L, m, best_v = top

    final_polish_time = end_limit - time.monotonic() - 30.0
    if final_polish_time > 20.0:
        final_polish_time = min(120.0, final_polish_time)
        try:
            pa, pv, psc = _run_search(best_a, best_v, nn, L, m, final_polish_time, int(rng_parent.integers(0, 2**31 - 1)), "final")
            if np.isfinite(psc) and psc < best_sc:
                best_a = pa
                best_v = pv
                best_sc = psc
                top = (best_a, best_sc, nn, L, m, best_v)
                ordered[0] = top
        except Exception as e:
            print("final polish failed err=%s" % repr(e), flush=True)

    final_candidates = _sort_candidates(candidates, 3)
    final_seq = None
    final_val = float("inf")

    for idx, (a, sc, nn, L, m, v) in enumerate(final_candidates):
        if time.monotonic() > end_limit - 3.0:
            break
        seq = _final_sequence(a)
        val = _score_seq_exact(seq, sc)
        print("final candidate %d internal=%.7f exact=%.7f n=%d" % (idx, float(sc), float(val), nn), flush=True)
        if np.isfinite(val) and val < final_val:
            final_val = val
            final_seq = seq

    if final_seq is None:
        seq = _final_sequence(best_a)
        final_seq = seq

    return final_seq
```
