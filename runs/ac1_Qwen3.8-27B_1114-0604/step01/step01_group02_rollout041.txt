def mirror_step(p,g,eta_rel):
    gscale = float(np.max(np.abs(g)))

    s = g/gscale

"""
Hybrid minimax active-set search for low peak self-convolution.

The sequence is kept as a probability vector p (sum 1).  The objective is
F(p) = 2 n max_k (p*p)_k.  The main local search uses a linearized minimax
direction: for the most important convolution lags it solves a small LP that
finds a simplex-preserving direction reducing the linearized peak values.
When the LP is not useful, a mirror-gradient / additive adaptive update based
on weighted active peaks is used.  A simple dual-covering weight calculation is
occasionally mixed in.  The routine starts from several diverse seeds, including
the supplied parent when available, and periodically perturbs from the best
incumbent to escape local minima.
"""

_INTERVAL_CACHE = {}


def get_intervals(n):
    if n in _INTERVAL_CACHE:
        return _INTERVAL_CACHE[n]
    L = 2 * n - 1
    starts = np.empty(L, dtype=np.int64)
    ends = np.empty(L, dtype=np.int64)
    for k in range(L):
        s = 0 if k < n else k - n + 1
        e = k if k < n else n - 1
        starts[k] = s
        ends[k] = e
    _INTERVAL_CACHE[n] = (starts, ends)
    return _INTERVAL_CACHE[n]


def normalize_prob(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.array([1.0])
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s <= 1e-300:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    return a / s


def fast_conv(a):
    n = a.size
    M = 2 * n - 1
    P = 1
    while P < M:
        P <<= 1
    A = np.fft.rfft(a, P)
    c = np.fft.irfft(A * A, P)[:M]
    c = np.maximum(c, 0.0)
    return c


def prob_score(p):
    c = fast_conv(p)
    G = float(np.max(c))
    if not np.isfinite(G):
        return np.inf
    return float(2.0 * p.size * G)


def resample_sequence(arr, n):
    arr = np.asarray(arr, dtype=np.float64).ravel()
    arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
    arr = np.clip(arr, 0.0, None)
    if arr.size == 0:
        return np.ones(n, dtype=np.float64) / float(n)
    if arr.size == n:
        return normalize_prob(arr)
    if arr.size < n:
        x = np.linspace(0.0, 1.0, arr.size)
        y = arr
        xq = np.linspace(0.0, 1.0, n)
        res = np.interp(xq, x, y)
        return normalize_prob(res)
    # Downsample by averaging contiguous blocks.
    edges = np.linspace(0, arr.size, n + 1).astype(np.int64)
    res = np.empty(n, dtype=np.float64)
    for i in range(n):
        a = edges[i]
        b = edges[i + 1]
        if b > a:
            res[i] = float(arr[a:b].mean())
        else:
            res[i] = 0.0
    return normalize_prob(res)


def init_uniform(n):

def init_ramp(n, slope):

def init_sine(n, freq, alpha, phase):

def init_random_nonneg(n, rng):

def init_haar(n, rng, levels=4, alpha=0.8):
    a = np.ones(n, dtype=np.float64)
    for ell in range(max(1, int(levels))):
        block = max(2, n // (2 ** ell))
        half = block // 2
        if half < 1:
            continue
        coeff = float(alpha) * (2.0 * rng.random() - 1.0) / float(levels + 1.0)
        for start in range(0, n, block):
            a[start:start + half] += coeff
            a[start + half:min(start + block, n)] -= coeff
    return normalize_prob(np.clip(a, 1e-4, None))


def init_parent_perturb(parent, sigma, rng):

    q = p.astype(np.float64).copy()
    q = q * (1.0 + float(sigma) * rng.standard_normal(q.size))
    q = q + 0.05 * float(sigma) * rng.random(q.size)

def select_active(c, G, K, rng=None, randomize=False):
    L = c.size
    K = int(min(max(1, K), L))
    candK = int(min(2 * K, L))
    if candK >= L:

    vals = c[idx]

def grad_topk_loop(p, idx, w):
    n = p.size
    starts, ends = get_intervals(n)
    g = np.zeros(n, dtype=np.float64)
    for i in range(idx.size):
        k = int(idx[i])
        wi = float(w[i])
        if wi <= 1e-16:
            continue
        s = int(starts[k])
        e = int(ends[k])
        if s <= e:
            seg = p[s:e + 1][::-1]
            g[s:e + 1] += (2.0 * wi) * seg
    return g


def dual_greedy_weights(p):
    n = p.size
    L = 2 * n - 1
    y = np.zeros(L, dtype=np.float64)
    mean_p = float(p.mean()) if n else 1.0
    floor = max(1e-14, mean_p * 1e-6)
    cur = 0.0
    for j in range(n):
        if j > 0:
            cur -= y[j - 1]
        T = 1.0 / max(float(p[j]), floor)
        if cur < T:
            delta = T - cur
            r = j + n - 1
            if 0 <= r < L:
                y[r] += delta
                cur += delta
    s = float(y.sum())
    if s <= 0.0:
        mid = max(0, min(L - 1, n - 1))
        y[mid] = 1.0
    else:
        y /= s
    return y


def lp_direction(p, c, idx, step_max, margin_frac):
    if not HAVE_LINPROG:
        return None
    n = p.size
    K = int(idx.size)
    if K <= 0 or step_max <= 0.0:
        return None

    starts, ends = get_intervals(n)
    A = np.zeros((K, n), dtype=np.float64)
    for r, kk in enumerate(idx):
        k = int(kk)
        s = int(starts[k])
        e = int(ends[k])
        if s <= e:
            A[r, s:e + 1] = 2.0 * p[s:e + 1][::-1]

    b = c[idx].astype(np.float64).copy()
    G = float(np.max(c))
    margin = float(margin_frac) * G
    if margin > 0.0 and b.size > 0:
        margin = min(margin, 0.45 * float(np.min(b)))
    b_margined = b - margin
    b_plain = b.copy()

    A_ub0 = np.empty((K, n + 1), dtype=np.float64)
    A_ub0[:, :n] = A
    A_ub0[:, n] = -1.0

    lower = -np.minimum(step_max, p)
    upper = np.full(n, step_max, dtype=np.float64)
    lb = np.concatenate([lower, np.array([-np.inf])])
    ub = np.concatenate([upper, np.array([np.inf])])
    c_obj = np.zeros(n + 1, dtype=np.float64)
    c_obj[-1] = 1.0
    A_eq = np.zeros((1, n + 1), dtype=np.float64)
    A_eq[0, :n] = 1.0
    b_eq = [0.0]

    def solve_with_b(bv):
        try:
            res = linprog(
                c_obj,
                A_ub=A_ub0,
                b_ub=bv,
                A_eq=A_eq,
                b_eq=b_eq,
                bounds=(lb, ub),
                method="highs",
                options={"primal_feasibility_tol": 1e-9, "dual_feasibility_tol": 1e-9}
            )
            if res is not None and getattr(res, "success", False) and np.all(np.isfinite(res.x)):
                d = res.x[:n].astype(np.float64)
                d = np.clip(d, lower - 1e-12, upper + 1e-12)
                d = d - float(d.mean())
                if float(np.max(np.abs(d))) > 1e-18:
                    return d
        except Exception:
            pass
        return None

    d = solve_with_b(b_margined)
    if d is None:
        d = solve_with_b(b_plain)
    return d


def line_search_lp(p, d, current_F):
    n = p.size
    if d is None or not np.all(np.isfinite(d)):
        return p, current_F
    best = p.copy()
    best_F = current_F
    neg = d < 0.0
    max_step = 2.0
    if np.any(neg):
        ratios = (-p[neg] - 1e-300) / d[neg]
        if ratios.size:
            max_step = min(max_step, float(np.min(ratios)))
    if max_step <= 1e-15:
        return best, best_F

    steps = []
    for s in (1.0, 0.75, 0.5, 0.25, 0.1, 0.03):
        if s <= max_step:
            steps.append(s)
    s_hi = 0.95 * max_step
    if s_hi > 0.0:
        steps.append(s_hi)
    steps = sorted(set(steps), reverse=True)

    for s in steps:
        if s <= 0.0:
            continue
        q = p + s * d
        q = np.clip(q, 0.0, None)
        ss = float(q.sum())
        if ss <= 1e-300:
            continue
        q = q / ss
        Fq = prob_score(q)
        if np.isfinite(Fq) and Fq < best_F:
            best_F = Fq
            best = q
    return best, best_F


def mirror_step(p, g, eta_rel):
    gscale = float(np.max(np.abs(g)))
    if gscale <= 1e-300:
        return p.copy()
    s = g / gscale
    arg = -float(eta_rel) * (s - float(np.max(s)))
    arg = np.clip(arg, -40.0, 40.0)
    z = p * np.exp(arg)
    ss = float(z.sum())
    if ss <= 1e-300:
        return p.copy()
    return z / ss


def additive_step(p, g, lr_mult):
    gscale = float(np.max(np.abs(g)))
    if gscale <= 1e-300:
        return p.copy()
    base = p.mean() / gscale
    lr = float(lr_mult) * base
    q = p - lr * (g / gscale)
    q = np.clip(q, 0.0, None)
    ss = float(q.sum())
    if ss <= 1e-300:
        return p.copy()
    return q / ss

        return best_F, best_p
    return cur_F if np.isfinite(cur_F) else np.inf, p


def try_trim(p, F):
    p = normalize_prob(p)
    n = p.size
    if n <= 64:
        return p, F
    th = max(1e-12, 1e-8 * float(p.mean()))
    nz = np.nonzero(p > th)[0]
    if nz.size == 0:
        return p, F
    first = int(nz[0])
    last = int(nz[-1]) + 1
    if last - first < n:
        q = normalize_prob(p[first:last])
        Fq = prob_score(q)
        if np.isfinite(Fq) and Fq < F:
            return q, Fq
    return p, F


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    start = time.monotonic()
    try:
        total = max(60.0, float(budget_s))
    except Exception:
        total = 2400.0
    deadline = start + total - 12.0
    rng = np.random.default_rng(int(seed) if int(seed) >= 0 else 42)

    n0 = 2048
    parent = None
    try:
        parent = height_sequence_1
    except Exception:
        parent = None

    if parent is not None:
        arr = np.asarray(parent, dtype=np.float64).ravel()
        arr = np.nan_to_num(arr, nan=0.0, posinf=0.0, neginf=0.0)
        arr = np.clip(arr, 0.0, 1000.0)
        if arr.size > 0 and arr.sum() > 0:
            if 512 <= arr.size <= 4096:
                n0 = int(arr.size)
            parent = resample_sequence(arr, n0)
    if parent is None:
        parent = init_uniform(n0)

    n0 = int(parent.size)
    n0 = int(min(max(256, n0), 4096))
    parent = resample_sequence(parent, n0)

    best_p = parent.copy()
    best_F = prob_score(best_p)
    try_trim_out = try_trim(best_p, best_F)
    best_p = try_trim_out[0]
    best_F = try_trim_out[1]
    print("initial best F=%.7f n=%d" % (best_F, best_p.size), flush=True)

    inits = []
    inits.append(("parent", best_p.copy()))
    inits.append(("square", init_anti_square(best_p.size, 1.0 / 3.0)))
    inits.append(("sin1", init_sine(best_p.size, 1, 0.5, 0.0)))
    inits.append(("sin2", init_sine(best_p.size, 2, 0.4, 0.8)))
    inits.append(("haar", init_haar(best_p.size, np.random.default_rng(int(seed) + 1), levels=4, alpha=0.8)))
    inits.append(("rand", init_random_nonneg(best_p.size, np.random.default_rng(int(seed) + 2))))
    inits.append(("rampup", init_ramp(best_p.size, 0.6)))
    inits.append(("rampdown", init_ramp(best_p.size, -0.6)))
    if np.isfinite(best_F):
        inits.append(("parentn1", init_parent_perturb(best_p, 0.05, np.random.default_rng(int(seed) + 3))))
        inits.append(("parentn2", init_parent_perturb(best_p, 0.15, np.random.default_rng(int(seed) + 4))))

    useful = max(30.0, total - 24.0)

    screen_end = start + 0.18 * useful
    for ii, (tag, p0) in enumerate(inits):
        rem = screen_end - time.monotonic()
        if rem < 3.0:
            break
        t_each = min(45.0, max(3.0, rem / max(1, len(inits) - ii)))
        end = min(screen_end, time.monotonic() + t_each)
        sc, pa = optimize_sequence(p0, end, rng, tag, polish=False)
        if np.isfinite(sc) and sc < best_F:
            best_F = sc
            best_p = pa.copy()
            best_p, best_F = try_trim(best_p, best_F)
            print("screen %s -> %.7f (n=%d)" % (tag, best_F, best_p.size), flush=True)

    main_end = min(deadline, start + 0.58 * useful)
    if time.monotonic() < main_end:
        sc, pa = optimize_sequence(best_p, main_end, rng, "main", polish=False)
        if np.isfinite(sc) and sc < best_F:
            best_F = sc
            best_p = pa.copy()
        best_p, best_F = try_trim(best_p, best_F)
        print("after main %.7f n=%d time=%.1f" % (best_F, best_p.size, time.monotonic() - start), flush=True)

    ref_end = min(deadline, start + 0.82 * useful)
    if time.monotonic() < ref_end:
        sc, pa = optimize_sequence(best_p, ref_end, rng, "refine", polish=True)
        if np.isfinite(sc) and sc < best_F:
            best_F = sc
            best_p = pa.copy()
        best_p, best_F = try_trim(best_p, best_F)
        print("after refine %.7f n=%d time=%.1f" % (best_F, best_p.size, time.monotonic() - start), flush=True)

    pol_end = min(deadline, start + 0.96 * useful)
    if time.monotonic() < pol_end:
        sc, pa = optimize_sequence(best_p, pol_end, rng, "polish", polish=True)
        if np.isfinite(sc) and sc < best_F:
            best_F = sc
            best_p = pa.copy()
        best_p, best_F = try_trim(best_p, best_F)
        print("after polish %.7f n=%d time=%.1f" % (best_F, best_p.size, time.monotonic() - start), flush=True)

    if time.monotonic() < deadline - 80 and best_p.size < 4096:
        hi_n = min(4096, 2 * best_p.size)
        if hi_n > best_p.size:
            p_hi = resample_sequence(best_p, hi_n)
            sc, pa = optimize_sequence(p_hi, deadline, rng, "hi", polish=True)
            if np.isfinite(sc) and sc < best_F:
                best_F = sc
                best_p = pa.copy()
            best_p, best_F = try_trim(best_p, best_F)
            print("after hi-res %.7f n=%d time=%.1f" % (best_F, best_p.size, time.monotonic() - start), flush=True)

    if not np.isfinite(best_F) or best_p is None or best_p.size == 0:
        return [1.0 / 2048.0] * 2048

    out = [float(x) for x in best_p]
    if not np.all(np.isfinite(out)):
        out = [1.0 / 2048.0] * 2048

    try:
        verified = float(evaluate_sequence(out))
        print("final internal=%.7f verified=%.7f n=%d time=%.2f" %
              (best_F, verified, len(out), time.monotonic() - start), flush=True)
        if not np.isfinite(verified) or verified > (2.0 if not np.isfinite(best_F) else max(1.0, best_F * 1.25 + 0.05)):
            fb = [1.0 / max(1, len(out)) for _ in out]
            vf = float(evaluate_sequence(fb)) if len(fb) > 0 else np.inf
            if np.isfinite(vf) and vf < (verified if np.isfinite(verified) else np.inf):
                out = fb
    except Exception:
        pass

    return out
