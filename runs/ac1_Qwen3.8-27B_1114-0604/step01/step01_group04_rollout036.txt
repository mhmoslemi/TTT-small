    n = len(a)
    # Build constraint matrix A such that A @ g <= M for all rows
    # Row k: sum_j a_j * g_{k-j} <= M, where g_{k-j} is valid only if 0 <= k-j < n
    A = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        for j in range(n):
            idx = k - j
            if 0 <= idx < n:
                A[k, j] = a[j]
    
    # Maximize sum(g) subject to A@g <= M, g >= 0

    res = linprog(c, A_ub=A, b_ub=np.full(2*n-1, M), bounds=[(0, None)]*n)

    return None

import numpy as np
from scipy.optimize import linprog
import time

def project_simplex(v):
    """Project v onto the simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    v = v.copy()
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def convolve_ab(a, b):
    """Full linear convolution a*b via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
    return np.fft.irfft(fa * fb, fsize)[:size]

    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)

        k_end = j + n - 1
        C[k_start:k_end+1, j] = a[j]
    return C

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding
combined with diverse initializations, subgradient descent for large n,
and local polishing. The LP approach solves the direction-finding problem
from the SOTA method: find g that maximizes sum(g) subject to conv(a,g)<=M.
"""

import numpy as np
from scipy.optimize import linprog
import time


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v.copy()
    v = v.astype(np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def cross_conv(a, b):
    """Full linear convolution a*b via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
    return np.fft.irfft(fa * fb, fsize)[:size]


def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix C where (C @ g)[k] = sum_j a[j]*g[k-j]."""
    n = a.size
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        C[k_start:k_end + 1, j] = a[j]
    return C


def lp_direction(a, M, timeout=10.0):
    """Solve LP: max sum(g) s.t. conv(a,g) <= M, g >= 0, sum(g) <= 3."""
    n = a.size
    C = build_conv_matrix(a)
    c = -np.ones(n)
    A_ub = np.vstack([C, np.ones((1, n))])
    b_ub = np.concatenate([np.full(2 * n - 1, M), [3.0]])
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(a, time_budget, rng, initial_t=0.05):
    """Optimize using LP-based direction finding with line search."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    t0 = time.monotonic()
    iters = 0
    last_improve = 0
    t = initial_t

    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        M = float(np.max(b))
        cur_sc = 2.0 * n * M

        # Solve LP
        g = lp_direction(a, M, timeout=min(30.0, time_budget - (time.monotonic() - t0)))
        if g is None:
            break

        g_sum = float(g.sum())
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Line search
        accepted = False
        for trial_t in [t, 0.5 * t, 0.25 * t, 0.1 * t, 0.05 * t]:
            a_trial = (1.0 - trial_t) * a + trial_t * g
            sc_trial = normed_score(a_trial)
            if sc_trial <= cur_sc * (1.0 + 1e-10):
                a = a_trial
                cur_sc = sc_trial
                accepted = True
                break

        if not accepted:
            # Try a random perturbation to escape
            a = project_simplex(a + 0.01 * rng.standard_normal(n))
            t = max(0.01, 0.5 * t)
            if t < 0.005:
                break
            continue

        if cur_sc < best_sc:
            best_sc = cur_sc
            best_a = a.copy()
            last_improve = iters
            # Slightly increase t if improving
            t = min(0.2, t * 1.02)
        else:
            t = max(0.01, 0.9 * t)

        iters += 1
        if iters - last_improve > 100:
            # Restart with perturbation
            a = project_simplex(best_a + 0.02 * rng.standard_normal(n))
            t = 0.05 + 0.05 * rng.random()
            last_improve = iters

    return best_a, best_sc


def subgrad_optimize(a, time_budget, rng):
    """Projected smoothed-subgradient descent with momentum."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    eta = 0.3
    tau_frac = 0.04
    n_scales = 3
    beta = 0.5
    vel = np.zeros(n)

    t0 = time.monotonic()
    iters = 0
    last_improve = 0

    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))

        tau = tau_frac * float(np.max(b))
        if tau < 1e-18:
            tau = 1e-18

        # Multi-scale smoothed gradient
        g = np.zeros(n)
        for s in range(n_scales):
            tau_s = tau * (2.0 ** s)
            shift = b / tau_s
            shift -= shift.max()
            w = np.exp(shift)
            w_sum = w.sum()
            if w_sum < 1e-300:
                w = np.zeros(b.size)
                w[int(np.argmax(b))] = 1.0
            else:
                w /= w_sum
            g += 2.0 * np.correlate(w, a, mode='valid')
        g /= n_scales

        vel = beta * vel + (1.0 - beta) * g
        a_new = project_simplex(a - eta * vel)

        b_new = full_conv(a_new)
        new_sc = 2.0 * n * float(np.max(b_new))

        if new_sc <= cur_sc or rng.random() < 0.02:
            a = a_new
        else:
            a = project_simplex(a - 0.3 * eta * vel)

        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = iters

        iters += 1
        if iters - last_improve > 3000:
            a = project_simplex(best_a + 0.03 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.2 + 0.3 * rng.random()
            tau_frac = 0.03 + 0.04 * rng.random()
            last_improve = iters

        if iters % 10000 == 0:
            eta = 0.2 + 0.4 * rng.random()
            beta = 0.3 + 0.4 * rng.random()

    return best_a, best_sc


def init_u_shape(n, alpha=0.5):
    """a_i proportional to (min(i, n-1-i) + 1)^(-alpha)."""
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i) + 1.0
    a = m ** (-alpha)
    return a / a.sum()


def init_exponential_edge(n, beta=1.0):
    """High at edges: a_i proportional to exp(beta * (i/n - 0.5)^2)."""
    i = np.arange(n, dtype=np.float64)
    x = i / n - 0.5
    a = np.exp(beta * x * x * 4.0)
    return a / a.sum()


def init_two_cluster(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.15 + 0.15 * rng.random())
    c2 = n * (0.70 + 0.15 * rng.random())
    sigma = n * (0.04 + 0.08 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.02 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.05
    return a / a.sum()


def init_sine(n, freq=1, alpha=2.0, rng=None):
    i = np.arange(n, dtype=np.float64)
    a = np.abs(np.sin(2.0 * np.pi * freq * i / n)) ** alpha
    a += 1e-10
    return a / a.sum()


def init_plateau(n, rng):
    """High in middle, low at edges (opposite of U-shape)."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    width = n * (0.2 + 0.3 * rng.random())
    a = np.exp(-((i - c) / width) ** 2)
    a += 0.05
    return a / a.sum()


def get_inits(n, rng):
    """Return diverse initializations for sequence length n."""
    inits = []
    inits.append(init_u_shape(n, alpha=0.3))
    inits.append(init_u_shape(n, alpha=0.5))
    inits.append(init_u_shape(n, alpha=1.0))
    inits.append(init_exponential_edge(n, beta=1.0))
    inits.append(init_exponential_edge(n, beta=2.0))
    inits.append(init_two_cluster(n, rng))
    inits.append(init_random(n, rng))
    inits.append(init_sine(n, freq=1, alpha=2.0))
    inits.append(init_sine(n, freq=2, alpha=2.0))
    inits.append(init_plateau(n, rng))
    return inits


def polish(a, time_budget, rng):
    """Local polish via random perturbations with decreasing scale."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    t0 = time.monotonic()
    scale = 0.03
    stuck = 0
    while time.monotonic() - t0 < time_budget and scale > 1e-7:
        improved = False
        for _ in range(20):
            cand = project_simplex(a + scale * rng.standard_normal(n))
            sc = normed_score(cand)
            if sc < best_sc:
                best_sc = sc
                best_a = cand.copy()
                a = cand.copy()
                improved = True
        if not improved:
            stuck += 1
            if stuck > 8:
                break
        else:
            stuck = 0
        scale *= 0.85

    return best_a, best_sc


def coordinate_descent(a, time_budget, rng):
    """Coordinate descent: randomly pick a coordinate, optimize it."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        # Pick a random coordinate
        i = rng.integers(n)
        # Try increasing and decreasing a[i]
        best_val = a[i]
        best_sc_local = best_sc
        
        for delta in [0.01, 0.005, 0.001, -0.01, -0.005, -0.001]:
            candidate = a.copy()
            candidate[i] = max(0, candidate[i] + delta)
            # Renormalize
            s = candidate.sum()
            if s < 1e-10:
                continue
            candidate /= s
            sc = normed_score(candidate)
            if sc < best_sc_local:
                best_sc_local = sc
                best_val = candidate[i]
                best_candidate = candidate.copy()
        
        if best_sc_local < best_sc:
            a = best_candidate
            best_sc = best_sc_local
            best_a = a.copy()

    return best_a, best_sc

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding
(SOTA approach: solve max sum(g) s.t. conv(a,g)<=M, g>=0), combined with
diverse initializations, subgradient descent for large n, and local polishing.
The LP gives exact optimal directions; subgradient handles larger n where LP is slow.
"""

import numpy as np
from scipy.optimize import linprog
import time


def project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v.copy()
    v = v.astype(np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def build_conv_matrix(a):
    """Build (2n-1) x n matrix C where (C @ g)[k] = sum_j a[j]*g[k-j]."""
    n = a.size
    C = np.zeros((2 * n - 1, n))
    for j in range(n):
        k_start = j
        k_end = j + n - 1
        C[k_start:k_end + 1, j] = a[j]
    return C


def lp_direction(a, M, timeout=15.0):
    """Solve LP: max sum(g) s.t. C@g <= M, g >= 0, sum(g) <= 3."""
    n = a.size
    C = build_conv_matrix(a)
    c = -np.ones(n)
    A_ub = np.vstack([C, np.ones((1, n))])
    b_ub = np.concatenate([np.full(2 * n - 1, M), [3.0]])
    bounds = [(0, None)] * n
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': timeout, 'presolve': True})
        if res.success and res.x is not None:
            return res.x
    except Exception:
        pass
    return None


def lp_optimize(a, time_budget, rng, initial_t=0.05):
    """Optimize using LP-based direction finding with adaptive line search."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    t0 = time.monotonic()
    iters = 0
    last_improve = 0
    t = initial_t
    stall_count = 0

    while time.monotonic() - t0 < time_budget:
        remaining = time_budget - (time.monotonic() - t0)
        if remaining < 2.0:
            break

        b = full_conv(a)
        M = float(np.max(b))
        cur_sc = 2.0 * n * M

        lp_timeout = min(10.0, max(1.0, remaining * 0.3))
        g = lp_direction(a, M, timeout=lp_timeout)
        if g is None:
            break

        g_sum = float(g.sum())
        if g_sum < 1e-10:
            break
        g = g / g_sum

        # Check if g is too similar to a (stuck)
        sim = float(np.dot(a, g))
        if sim > 0.9999:
            stall_count += 1
            if stall_count > 5:
                break
            a = project_simplex(a + 0.02 * rng.standard_normal(n))
            continue
        stall_count = 0

        # Line search: try decreasing step sizes
        accepted = False
        for trial_t in [t, 0.5 * t, 0.25 * t, 0.1 * t, 0.05 * t]:
            if trial_t < 1e-4:
                break
            a_trial = (1.0 - trial_t) * a + trial_t * g
            a_trial = np.maximum(a_trial, 0.0)
            a_trial = a_trial / a_trial.sum()
            sc_trial = normed_score(a_trial)
            if sc_trial <= cur_sc * (1.0 + 1e-9):
                a = a_trial
                cur_sc = sc_trial
                accepted = True
                break

        if not accepted:
            # Escape: perturb and reduce step
            a = project_simplex(a + 0.01 * rng.standard_normal(n))
            t = max(0.01, 0.7 * t)
            if t < 0.005:
                break
            continue

        if cur_sc < best_sc - 1e-12:
            best_sc = cur_sc
            best_a = a.copy()
            last_improve = iters
            t = min(0.15, t * 1.01)
        else:
            t = max(0.01, 0.95 * t)

        iters += 1
        if it % 50 == 0:

            t = 0.05 + 0.05 * rng.random()
            last_improve = iters

    return best_a, best_sc


def subgrad_optimize(a, time_budget, rng):
    """Projected smoothed-subgradient descent with momentum."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    eta = 0.3
    tau_frac = 0.04
    n_scales = 3
    beta = 0.5
    vel = np.zeros(n)

    t0 = time.monotonic()
    iters = 0
    last_improve = 0

    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        max_b = float(np.max(b))
        cur_sc = 2.0 * n * max_b

        tau = tau_frac * max_b
        if tau < 1e-18:
            tau = 1e-18

        g = np.zeros(n)
        for s in range(n_scales):
            tau_s = tau * (2.0 ** s)
            shift = b / tau_s
            shift -= shift.max()
            w = np.exp(shift)
            w_sum = w.sum()
            if w_sum < 1e-300:
                w = np.zeros(b.size)
                w[int(np.argmax(b))] = 1.0
            else:
                w /= w_sum
            g += 2.0 * np.correlate(w, a, mode='valid')
        g /= n_scales

        vel = beta * vel + (1.0 - beta) * g
        a_new = project_simplex(a - eta * vel)

        b_new = full_conv(a_new)
        new_sc = 2.0 * n * float(np.max(b_new))

        if new_sc <= cur_sc or rng.random() < 0.02:
            a = a_new
        else:
            a = project_simplex(a - 0.3 * eta * vel)

        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = iters

        iters += 1
        if iters - last_improve > 3000:
            a = project_simplex(best_a + 0.03 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.2 + 0.3 * rng.random()
            tau_frac = 0.03 + 0.04 * rng.random()
            last_improve = iters

        if iters % 10000 == 0:
            eta = 0.2 + 0.4 * rng.random()
            beta = 0.3 + 0.4 * rng.random()

    return best_a, best_sc


def init_u_shape(n, alpha=0.5):
    """a_i proportional to (min(i, n-1-i) + 1)^(-alpha)."""
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i) + 1.0
    a = m ** (-alpha)
    return a / a.sum()


def init_exponential_edge(n, beta=1.0):
    """High at edges: a_i proportional to exp(4*beta*(i/n - 0.5)^2)."""
    i = np.arange(n, dtype=np.float64)
    x = i / n - 0.5
    a = np.exp(4.0 * beta * x * x)
    return a / a.sum()


def init_two_cluster(n, rng):
    """Two Gaussian clusters."""
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.15 + 0.15 * rng.random())
    c2 = n * (0.70 + 0.15 * rng.random())
    sigma = n * (0.04 + 0.08 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.02 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.05
    return a / a.sum()

    a += 1e-10

    """Linear ramp with noise."""
    i = np.arange(n, dtype=np.float64)

    a += 0.1 * np.abs(rng.standard_normal(n))

def init_four_cluster(n, rng):

def polish(a, time_budget, rng):
    """Local polish via random perturbations with decreasing scale."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    t0 = time.monotonic()
    scale = 0.02
    stuck = 0
    while time.monotonic() - t0 < time_budget and scale > 1e-8:
        improved = False
        n_trials = 30
        for _ in range(n_trials):
            if time.monotonic() - t0 > time_budget:
                break
            cand = project_simplex(a + scale * rng.standard_normal(n))
            sc = normed_score(cand)
            if sc < best_sc:
                best_sc = sc
                best_a = cand.copy()
                a = cand.copy()
                improved = True
        if not improved:
            stuck += 1
            if stuck > 10:
                break
        else:
            stuck = 0
        scale *= 0.85

    return best_a, best_sc


def coordinate_descent_polish(a, time_budget, rng):

    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    t0 = time.monotonic()

        best_sc_local = best_sc

        for delta in [0.02, 0.01, 0.005, 0.002, -0.02, -0.01, -0.005, -0.002]:
            candidate = a.copy()
            candidate[i] = max(0.0, candidate[i] + delta)

            if s < 1e-10:

        print(f"[fallback] Using U-shape n=128, score={best_overall_sc:.6f}")

    # Final verification with the actual evaluator
    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"[FINAL] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")

    return result
