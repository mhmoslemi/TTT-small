        a >= 0,

Minimizes 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS on a smooth (log-sum-exp)

from scipy.optimize import minimize

# Global best tracker

_best_seq = None

def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = rfft(a, fsize)
    return irfft(fa * fa, fsize)[:size]

def true_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = full_conv(a)
    return 2.0 * n * b.max() / (s * s)

def softmax(x):
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()

def smooth_obj_and_grad(x, tau):
    """
    Compute F_tau(softmax(x)) and its gradient.
    F_tau(a) = 2*n*tau*log(sum(exp(b_k/tau))) where b = conv(a,a).
    """
    a = softmax(x)
    n = a.size
    b = full_conv(a)
    
    # Log-sum-exp of b/tau
    b_scaled = b / tau
    b_max = b_scaled.max()
    log_sum_exp = b_max + np.log(np.sum(np.exp(b_scaled - b_max)))
    obj = 2.0 * n * tau * log_sum_exp
    
    # Softmax weights
    w = np.exp(b_scaled - b_max)
    w /= w.sum()
    
    # Gradient of smooth max w.r.t. a:
    # grad_a[i] = 2 * sum_k w[k] * a[k-i] (for valid k-i)
    # = 2 * (a cross-correlate w) [i]
    # where (a ⋆ w)[i] = sum_j a[j] * w[j+i]
    
    # Compute via FFT:
    # (a ⋆ w)[i] = sum_j a[j] * w[j+i]
    # Let's define: c[m] = sum_j a[j] * w[j+m] for m = 0, 1, ..., 2n-2
    # This is the cross-correlation. Using FFT:
    # Zero-pad a to length >= n + len(w) - 1 = n + 2n-1 - 1 = 3n-2
    # Actually, we only need c[0:n], so let's be careful.
    
    # c[m] = sum_{j=0}^{n-1} a[j] * w[j+m] (where w has length 2n-1)
    # For m=0: sum_j a[j]*w[j]
    # For m=1: sum_j a[j]*w[j+1]
    # ...
    # For m=2n-2: a[0]*w[2n-2] (only j=0 contributes)
    
    # This is like a "sliding window" correlation.
    # We can compute it as: c = ifft(fft(a_padded) * conj(fft(w_padded)))
    # But we need to be careful about the definition.
    
    # Standard cross-correlation: (x ⋆ y)[n] = sum_m x[m+n] * conj(y[m])
    # In our case: c[m] = sum_j a[j] * w[j+m] = sum_j a[j+m-m] * w[j+m] ... 
    # Let's set x = w, y = a, then (x ⋆ y)[m] = sum_j w[j+m] * a[j] = c[m]. ✓
    
    # So c = cross_correlate(w, a) = ifft(conj(fft(a)) * fft(w)) (with zero-padding)
    
    # Padding: a has length n, w has length 2n-1.
    # We need c[m] for m=0..n-1.
    # c has full length n + (2n-1) - 1 = 3n-2.
    # Pad both to length >= 3n-2.
    
    pad_size = 1 << (3 * n - 2 - 1).bit_length()
    fa = rfft(a, pad_size)
    fw = rfft(w, pad_size)
    c = irfft(fa.conj() * fw, pad_size)[:n]
    
    grad_a = 2.0 * c
    
    # Gradient w.r.t. x (softmax parameterization):
    # ∂F/∂x_i = a_i * (grad_a_i - sum_j a_j * grad_a_j)
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    
    return obj, grad_x

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS on a log-sum-exp smoothed
objective with adaptive temperature annealing, diverse initializations across
multiple sequence lengths, and local active-set polishing.
Uses FFT-based convolution for O(n log n) per gradient evaluation.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time

def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = rfft(a, fsize)
    return irfft(fa * fa, fsize)[:size]

def true_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = full_conv(a)
    return 2.0 * n * float(b.max()) / (s * s)

def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()

def smooth_obj_and_grad(x, tau):
    """
    Compute F_tau(softmax(x)) and its gradient.
    F_tau(a) = 2*n*tau*log(sum(exp(b_k/tau))) where b = conv(a,a).
    Returns (value, gradient).
    """
    a = softmax(x)
    n = a.size
    b = full_conv(a)
    
    b_scaled = b / tau
    b_max = b_scaled.max()
    log_sum_exp = b_max + np.log(np.sum(np.exp(b_scaled - b_max)))
    obj = 2.0 * n * tau * log_sum_exp
    
    w = np.exp(b_scaled - b_max)
    w /= w.sum()
    
    # grad_a[i] = 2 * sum_j a[j] * w[j+i]
    # = 2 * (w cross-correlate a)[i]
    pad_size = 1 << (3 * n - 2 - 1).bit_length()
    fa = rfft(a, pad_size)
    fw = rfft(w, pad_size)
    c = irfft(fa.conj() * fw, pad_size)[:n]
    
    grad_a = 2.0 * c
    
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)
    
    return obj, grad_x

def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def lbfgs_anneal(x0, n, tau_start, tau_end, n_temps, max_iter_per_tau):
    """
    Run L-BFGS with geometrically decreasing temperature.
    x0: initial point in log-space (length n)
    Returns: (best_x, best_true_score)
    """
    x = x0.copy()
    best_a = softmax(x).copy()
    best_sc = true_score(best_a)
    
    taus = np.geomspace(tau_start, tau_end, n_temps)
    
    for tau in taus:
        a = softmax(x)
        cur_sc = true_score(a)
        
        # Run L-BFGS at this temperature
        try:
            result = minimize(
                smooth_obj_and_grad, x,
                method='L-BFGS-B',
                jac=True,
                args=(tau,),
                options={'maxiter': max_iter_per_tau, 'ftol': 1e-12, 'gtol': 1e-8}
            )
            x = result.x
        except Exception as e:
            print(f"  L-BFGS error at tau={tau:.2e}: {e}")
            break
        
        a = softmax(x)
        sc = true_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
    
    return best_a, best_sc

def polish_active_set(a, n_iter=2000, tol=1e-10):
    """
    Polish using active-set Newton steps on the dominant quadratic form.
    a is on the simplex (sum=1, a>=0).
    """
    n = a.size
    a = a.copy()
    best_a = a.copy()
    best_sc = true_score(a)
    
    for it in range(n_iter):
        b = full_conv(a)
        k_star = int(np.argmax(b))
        cur_sc = 2.0 * n * b.max()  # since sum(a)=1
        
        if cur_sc < best_sc - tol:
            best_sc = cur_sc
            best_a = a.copy()
        
        # Build the gradient of b_{k*} = sum_j a[j]*a[k*-j]
        # grad[i] = 2*a[k*-i] (if 0<=k*-i<n)
        grad = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                grad[i] = 2.0 * a[j]
        
        # Newton step on the quadratic form a^T J a:
        # The Hessian of b_{k*} w.r.t. a is 2*J_{k*}
        # J_{k*}[i,j] = 1 if i+j = k*
        # Newton: a_new = a - alpha * (2*J a - lambda*1)^+ ... this is complex
        # Let's just do a gradient step with backtracking
        # Since b_{k*} is quadratic, the gradient method with exact line search converges in n steps
        
        # Gradient descent with line search on the simplex
        # Projected gradient: a_new = project(a - lr * grad / max(|grad|, 1))
        lr = 0.01
        for _ in range(20):
            a_new = project_simplex(a - lr * grad)
            b_new = full_conv(a_new)
            sc_new = 2.0 * n * b_new.max()
            if sc_new < cur_sc:
                a = a_new
                break
            lr *= 0.5
        
        if lr < 1e-15:
            break
    
    return best_a, best_sc

def pair_swap_search(a, n_trials=5000, rng=None):
    """
    Local search via random pair swaps on the simplex.
    Move mass from a[i] to a[j] (or vice versa) and keep sum = 1.
    """
    if rng is None:
        rng = np.random.default_rng()
    n = a.size
    a = a.copy()
    best_a = a.copy()
    best_sc = true_score(a)
    
    for t in range(n_trials):
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # Move mass: a[i] decreases by delta, a[j] increases by delta
        delta = 0.01 * rng.random()
        if a[i] < delta:
            delta = a[i] * 0.5
        if delta < 1e-12:
            continue
        
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        sc = true_score(a_new)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
        # else: revert (a already unchanged since we used a_new)
    
    return best_a, best_sc

def make_initializations(n, rng):
    """Generate diverse initializations in log-space."""
    inits = []
    
    # Uniform
    inits.append(np.zeros(n))
    
    # Random
    inits.append(rng.standard_normal(n) * 2)
    
    # U-shaped (high at edges)
    i = np.arange(n, dtype=np.float64) - n / 2
    inits.append(-np.abs(i) / (n * 0.1))
    
    # Cosine window (raised)
    inits.append(0.5 * np.cos(2 * np.pi * i / n) * n * 0.1)
    
    # Two-gaussian
    c1, c2 = n * 0.25, n * 0.75
    sigma = n * 0.1
    inits.append(-((i - c1 + n/2)**2 + (i - c2 + n/2)**2) / (2 * sigma**2))
    
    # Linear ramp
    inits.append(np.linspace(-2, 2, n))
    
    # Sparse peak
    x = np.full(n, -5.0)
    x[n // 4] = 0
    x[3 * n // 4] = 0
    inits.append(x)
    
    # Multi-frequency cosine
    freq = 3
    inits.append(np.cos(2 * np.pi * freq * i / n) * 2 + 1)
    
    return inits

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS on a log-sum-exp smoothed
objective with adaptive temperature annealing, diverse initializations across
multiple sequence lengths, and local active-set/coordinate-descent polishing.
Uses FFT-based convolution for O(n log n) per gradient evaluation.
Key innovation: quasi-Newton (L-BFGS) optimization on the smooth surrogate
converges much faster than first-order subgradient methods, enabling more
restarts and finer temperature schedules within the time budget.
"""

import numpy as np
from scipy.fft import rfft, irfft
from scipy.optimize import minimize
import time


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = rfft(a, fsize)
    return irfft(fa * fa, fsize)[:size]


def true_score(a):
    """Compute 2*n*max(conv(a,a))/sum(a)^2."""
    n = a.size
    s = a.sum()
    if s < 1e-10:
        return np.inf
    b = full_conv(a)
    return 2.0 * n * float(b.max()) / (s * s)


def softmax(x):
    """Numerically stable softmax."""
    x_shifted = x - x.max()
    e = np.exp(x_shifted)
    return e / e.sum()


def smooth_obj_and_grad(x, tau):
    """
    Compute F_tau(softmax(x)) and its gradient w.r.t. x.
    F_tau(a) = 2*n*tau*log(sum_k exp(b_k/tau)) where b = conv(a,a).
    Returns (value, gradient).
    """
    a = softmax(x)
    n = a.size
    b = full_conv(a)

    b_scaled = b / tau
    b_max = b_scaled.max()
    log_sum_exp = b_max + np.log(np.sum(np.exp(b_scaled - b_max)))
    obj = 2.0 * n * tau * log_sum_exp

    w = np.exp(b_scaled - b_max)
    w /= w.sum()

    # grad_a[i] = 2 * sum_j a[j] * w[j+i]  (cross-correlation of a and w)
    # Compute via FFT: ifft(conj(FFT(a)) * FFT(w))
    pad_size = 1 << (3 * n - 3).bit_length()
    fa = rfft(a, pad_size)
    fw = rfft(w, pad_size)
    c = irfft(fa.conj() * fw, pad_size)[:n]

    grad_a = 2.0 * c

    # Softmax Jacobian: dF/dx_i = a_i * (grad_a_i - <a, grad_a>)
    dot = np.dot(a, grad_a)
    grad_x = a * (grad_a - dot)

    return obj, grad_x


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def lbfgs_anneal(x0, tau_start, tau_end, n_temps, max_iter_per_tau, time_limit=None):
    """
    Run L-BFGS with geometrically decreasing temperature.
    x0: initial point in log-space.
    Returns: (best_a, best_true_score)
    """
    x = x0.copy()
    best_a = softmax(x).copy()
    best_sc = true_score(best_a)

    taus = np.geomspace(tau_start, tau_end, n_temps)
    t0 = time.monotonic()

    for ti, tau in enumerate(taus):
        if time_limit is not None and time.monotonic() - t0 > time_limit:
            break

        a = softmax(x)
        cur_sc = true_score(a)

        try:
            result = minimize(
                smooth_obj_and_grad, x,
                method='L-BFGS-B',
                jac=True,
                args=(tau,),
                options={'maxiter': max_iter_per_tau, 'ftol': 1e-14, 'gtol': 1e-10}
            )
            x_new = result.x
            # Verify improvement
            a_new = softmax(x_new)
            sc_new = true_score(a_new)
            if sc_new <= cur_sc * 1.01:
                x = x_new
        except Exception:
            pass

        a = softmax(x)
        sc = true_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()

    return best_a, best_sc


def coordinate_descent_polish(a, n_sweeps=50, time_limit=60.0):

    Cyclic coordinate descent: for each coordinate, find optimal value

                a[i] = candidate
                # Redistribute to other coordinates proportionally
                others_sum = a.sum() - a[i]
                if others_sum < 1e-15:
                    a[i] = a_save
                    continue
                a[1:] *= ((1.0 - a[i]) / others_sum) if i != 0 else 1.0
                # Hmm this is getting complicated. Let me use a simpler approach.
                a[i] = a_save
                break

            # Simpler: just try small perturbations
            a[i] = a_save
            for delta in [0.005, -0.005, 0.01, -0.01]:
                a[i] = a_save + delta
                if a[i] < 0:

                    improved = True

                    break
            a[i] = a_save

        if not improved:
            break

def random_perturb_polish(a, rng, n_trials=20000, time_limit=120.0):
    """
    Random perturbation polish: add small random noise, project, check score.
    """
    n = a.size
    a = a.copy()
    best_a = a.copy()
    best_sc = true_score(a)
    t0 = time.monotonic()
    scale = 0.02

    for t in range(n_trials):

            best_a = cand.copy()
            a = cand.copy()
        elif t % 500 == 0:

    return best_a, best_sc

        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size >= 10 and h.sum() > 0.01:
            prev_a = h / h.sum()
            prev_sc = true_score(prev_a)
            print(f"[init] Previous best: n={h.size}, score={prev_sc:.6f}")
            if prev_sc < best_sc:
                best_sc = prev_sc
                best_a = prev_a.copy()
    except Exception:
        pass

    # n values to try
    n_list = [128, 256, 512, 1024]
    if prev_a is not None:
        np_ = prev_a.size
        if np_ not in n_list and 64 <= np_ <= 4096:
            n_list.append(np_)
        n_list = sorted(set(n_list))

    total_time = deadline - time.monotonic()
    print(f"[plan] n_list={n_list}, total_time={total_time:.1f}s")

    # Time allocation weights
    raw_w = []
    for nv in n_list:
        if nv <= 128:
            raw_w.append(3.0)
        elif nv <= 256:
            raw_w.append(2.8)
        elif nv <= 512:
            raw_w.append(2.5)
        elif nv <= 1024:
            raw_w.append(2.2)
        else:
            raw_w.append(1.5)
    total_w = sum(raw_w)
    weights = [w / total_w for w in raw_w]

    n_inits = 15  # number of initializations per n

    for idx_n, nv in enumerate(n_list):
        if time.monotonic() > deadline:
            break

        time_n = weights[idx_n] * total_time * 0.82
        time_per_init = time_n / n_inits

        print(f"\n[phase] n={nv}, time_budget={time_n:.1f}s, per_init={time_per_init:.1f}s")

        inits = make_initializations(nv, rng)

        # Add previous best if same length
        if prev_a is not None and prev_a.size == nv:
            x_prev = np.log(prev_a + 1e-15)
            inits.insert(0, x_prev)

        for ii in range(min(n_inits, len(inits))):
            if time.monotonic() > deadline:
                break

            x0 = inits[ii].copy()

            # Determine temperature schedule
            a_curr = softmax(x0)
            b_curr = full_conv(a_curr)
            b_max_val = float(b_curr.max())
            tau_start = b_max_val * 3.0
            tau_end = b_max_val * 0.005

            if tau_end < 1e-18:
                tau_end = 1e-18

            n_temps = 10
            max_iter = max(100, min(800, int(time_per_init * 80)))

            tb = min(time_per_init, deadline - time.monotonic() - 2.0)
            if tb < 2.0:
                break

            t0 = time.monotonic()
            a_best, sc_best = lbfgs_anneal(x0, tau_start, tau_end, n_temps, max_iter,
                                            time_limit=tb)
            elapsed = time.monotonic() - t0

            if sc_best < best_sc:
                best_sc = sc_best
                best_a = a_best.copy()
                print(f"  [NEW BEST] n={nv}, init={ii}, score={sc_best:.6f}, t={elapsed:.1f}s")
            else:
                print(f"  n={nv}, init={ii}, score={sc_best:.6f}, t={elapsed:.1f}s")

    print(f"\n[summary] Best after main: n={best_a.size if best_a is not None else '?'}, score={best_sc:.6f}")

    # === Polish Phase ===
    if best_a is not None and time.monotonic() < deadline - 10.0:
        remain = deadline - time.monotonic() - 5.0
        print(f"\n[polish] remaining={remain:.1f}s, n={best_a.size}")

        # 1. L-BFGS with very small tau (fine-tune)
        if time.monotonic() < deadline - 10.0:
            x_start = np.log(best_a + 1e-15)
            b_check = full_conv(best_a)
            tau_final = float(b_check.max()) * 0.003
            tb1 = min(remain * 0.35, 300.0)
            a_p, sc_p = lbfgs_anneal(x_start, tau_final * 5, tau_final, 6, 500,
                                      time_limit=tb1)
            if sc_p < best_sc:
                best_sc = sc_p
                best_a = a_p.copy()
                print(f"  [polish] fine L-BFGS: {sc_p:.6f}")

        # 2. Random perturbation polish
        if time.monotonic() < deadline - 10.0:
            tb2 = min(remain * 0.25, 200.0)
            a_p2, sc_p2 = random_perturb_polish(best_a, rng, n_trials=50000,
                                                 time_limit=tb2)
            if sc_p2 < best_sc:
                best_sc = sc_p2
                best_a = a_p2.copy()
                print(f"  [polish] random perturb: {sc_p2:.6f}")

        # 3. Another round of L-BFGS from perturbed point
        if time.monotonic() < deadline - 10.0:
            x_start2 = np.log(best_a + 1e-15)
            b_check2 = full_conv(best_a)
            tau_f2 = float(b_check2.max()) * 0.002
            tb3 = min(remain * 0.2, 150.0)
            a_p3, sc_p3 = lbfgs_anneal(x_start2, tau_f2 * 3, tau_f2, 5, 400,
                                        time_limit=tb3)
            if sc_p3 < best_sc:
                best_sc = sc_p3
                best_a = a_p3.copy()
                print(f"  [polish] second L-BFGS: {sc_p3:.6f}")

        # 4. Final random perturbation
        if time.monotonic() < deadline - 10.0:
            tb4 = min(remain * 0.1, 60.0)
            a_p4, sc_p4 = random_perturb_polish(best_a, rng, n_trials=20000,
                                                 time_limit=tb4)
            if sc_p4 < best_sc:
                best_sc = sc_p4
                best_a = a_p4.copy()
                print(f"  [polish] final perturb: {sc_p4:.6f}")

    # Fallback
    if best_a is None:
        best_a = np.ones(128) / 128
        best_sc = 2.0

    # Final verification
    result = [float(x) for x in best_a]
    final_sc = float(evaluate_sequence(result))
    print(f"\n[final] n={len(result)}, internal={best_sc:.6f}, evaluator={final_sc:.6f}")

    return result
