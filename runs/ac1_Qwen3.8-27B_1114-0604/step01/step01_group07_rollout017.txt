"""
Improved low-peak nonnegative step-function search.
Uses active-set mirror descent with minimum-norm subgradients,
multiple diverse initializations, and extended optimization time.
Combines sinusoidal starting points with full-shape gradient optimization.

import numpy as np

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    return float(2.0 * arr.size * mx / (s * s))


def _make_sine(n, amp, phase_offset=0.5):
    n = int(n)
    if n <= 0:
        return np.array([], dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + phase_offset) / float(n)
    a = 1.0 + float(amp) * np.sin(theta)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-10:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    return a / s


def _make_multi_sine(n, coeffs):
    n = int(n)
    i = np.arange(n, dtype=np.float64) / float(n)
    a = np.ones(n, dtype=np.float64)
    for k, amp in enumerate(coeffs, start=1):
        a += amp * np.sin(2.0 * np.pi * k * (i + 0.5 / float(n)))
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-10:
        a = np.ones(n, dtype=np.float64)
        s = float(n)
    return a / s


def _make_anti_palindromic(n, shape_func):
    i = np.arange(n, dtype=np.float64)
    a = shape_func(i / float(n))
    a = np.clip(a, 0.0, None)
    a = a - (a + a[::-1]) / 2.0 * 0 + (a - a[::-1]) / 2.0 + (a + a[::-1]) / 2.0 * 0
    # Actually just use the shape directly, ensure non-negative
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 1e-10:
        a = np.ones(n)
        s = float(n)
    return a / s


def _compute_active_set(c, threshold_frac=0.98, max_k=50):
    max_c = float(np.max(c))
    threshold = max_c * threshold_frac
    active = np.where(c >= threshold)[0]
    if len(active) > max_k:
        top = np.argpartition(c, -max_k)[-max_k:]
        active = top[c[top] >= threshold]
        if len(active) == 0:
            active = top
    return active, max_c


def _compute_grad_for_lag(k, a, n):
    g = np.zeros(n, dtype=np.float64)
    k = int(k)
    lo = max(0, k - (n - 1))
    hi = min(n - 1, k)
    if hi >= lo:
        j_vals = np.arange(lo, hi + 1, dtype=np.int64)
        g[j_vals] = 2.0 * a[k - j_vals]
    return g


def _min_norm_combination(grads, m):
    if m == 1:
        return grads[0]
    Q = np.zeros((m, m), dtype=np.float64)
    for i in range(m):
        for j in range(i, m):
            v = float(np.dot(grads[i], grads[j]))
            Q[i, j] = v
            Q[j, i] = v
    # Try interior solution: lambda = Q^{-1} 1 / (1^T Q^{-1} 1)
    try:
        ones = np.ones(m, dtype=np.float64)
        Q_inv_1 = np.linalg.solve(Q, ones)
        denom = float(ones @ Q_inv_1)
        if denom > 1e-10:
            lam = Q_inv_1 / denom
            if np.all(lam >= -1e-10):
                lam = np.clip(lam, 0.0, None)
                lam = lam / lam.sum()
                return lam @ grads
    except np.linalg.LinAlgError:
        pass
    # Fallback: iterative projection onto simplex
    lam = np.ones(m, dtype=np.float64) / m
    for _ in range(200):
        grad_lam = 2.0 * Q @ lam
        # Projected gradient with adaptive step
        step = 1.0 / (np.max(np.abs(np.diag(Q))) + 1e-10)
        new_lam = lam - step * grad_lam
        # Project onto probability simplex
        u = np.sort(new_lam)[::-1]
        cssv = np.cumsum(u)
        idx = np.arange(1, m + 1)
        cond = u * idx > (cssv - 1.0)
        if not np.any(cond):
            rho = m - 1
        else:
            rho = int(np.nonzero(cond)[0][-1])
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
        new_lam = np.maximum(new_lam - theta, 0.0)
        if np.max(np.abs(new_lam - lam)) < 1e-12:
            lam = new_lam
            break
        lam = new_lam
    return lam @ grads

            a2 = _make_multi_sine(n, [amp * 0.8, 0.15, 0.05])
            sc2 = _internal_score(a2)
            candidates.append((sc2, n, amp, a2))

    candidates.sort(key=lambda x: x[0])
    return candidates


def _try_start_sequence():
    try:
        hs = globals().get('height_sequence_1', None)
        if hs is not None:
            arr = np.asarray(hs, dtype=np.float64)
            arr = np.clip(arr, 0.0, 1000.0)
            s = float(arr.sum())
            if s > 0.01:
                return arr / s
    except Exception:
        pass
    return None


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t0 = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    hard_deadline = t0 + total_budget - 30.0

    print(f"[start] budget={total_budget}s deadline_in={total_budget - 30:.0f}s")

    # Phase 1: Quick scan for good starting points (5 min max)
    scan_deadline = t0 + 300.0
    if total_budget < 600:
        scan_deadline = t0 + max(30, total_budget * 0.15)

    print("[phase 1] Quick scan for starting points...")
    candidates = _quick_scan(scan_deadline)

    # Add the provided sequence if available
    hs = _try_start_sequence()
    if hs is not None:
        sc = _internal_score(hs)
        candidates.append((sc, len(hs), -1, hs))
        candidates.sort(key=lambda x: x[0])
        print(f"  Provided sequence: n={len(hs)} score={sc:.8f}")

    # Add some random starts
    rng = np.random.RandomState(seed)
    for n in [2048, 4096]:
        a = rng.dirichlet(np.ones(n) * 0.5)
        sc = _internal_score(a)
        candidates.append((sc, n, -2, a))

    candidates.sort(key=lambda x: x[0])
    print(f"  Top 5 starting points:")
    for i in range(min(5, len(candidates))):
        sc, n, amp, _ = candidates[i]
        print(f"    [{i}] n={n} amp={amp} score={sc:.8f}")

    # Phase 2: Optimize from top starting points
    print("[phase 2] Gradient optimization...")
    best_a = None
    best_score = np.inf
    optimize_deadline = hard_deadline - 60.0

    seen_configs = set()
    n_tried = 0
    max_tries = 12

    for idx in range(min(len(candidates), max_tries)):
        sc, n, amp, a = candidates[idx]
        config_key = (n, round(amp, 2) if amp >= 0 else -1)
        if config_key in seen_configs:
            continue
        seen_configs.add(config_key)
        n_tried += 1

        remaining = optimize_deadline - time.monotonic()
        if remaining < 20:
            break

        # Allocate time based on rank
        if n_tried <= 2:
            budget = min(remaining - 20, 400)
        elif n_tried <= 5:
            budget = min(remaining - 20, 200)
        else:
            budget = min(remaining - 20, 80)

        if budget < 10:
            break

        print(f"  Optimizing init {n_tried}: n={n} init_score={sc:.8f} budget={budget:.0f}s")
        opt_a, opt_score = _optimize_sequence(a.copy(), budget)

        if opt_score < best_score:
            best_score = opt_score
            best_a = opt_a
            print(f"    -> New best: {best_score:.8f}")
        else:
            print(f"    -> No improvement (got {opt_score:.8f})")

    # Phase 3: Extended refinement on the best
    remaining = hard_deadline - time.monotonic()
    if remaining > 30 and best_a is not None:
        print(f"[phase 3] Extended refinement, budget={remaining:.0f}s")
        refined_a, refined_score = _optimize_sequence(best_a.copy(), remaining - 15)
        if refined_score < best_score:
            best_score = refined_score
            best_a = refined_a
            print(f"  Refinement improved to: {best_score:.8f}")
        else:
            print(f"  Refinement no improvement (got {refined_score:.8f})")

    # Fallback
    if best_a is None:
        print("[fallback] Using default sinusoid")
        best_a = _make_sine(2048, 0.78)
        best_score = _internal_score(best_a)

    # Final validation
    arr = np.asarray(best_a, dtype=np.float64)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        arr = _make_sine(2048, 0.78)
        s = float(arr.sum())
    arr = arr / s
    result = [float(x) for x in arr]

    real_score = evaluate_sequence(result)
    print(f"\n[final] n={len(arr)} internal={best_score:.8f} real={real_score:.8f} "
          f"time={time.monotonic()-t0:.1f}s")

    return result
