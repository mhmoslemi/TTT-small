def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

"""
Low-peak nonnegative step-function search using mirror descent with
adaptive active-set size, learning rate, and periodic restarts.
Improves upon the sinusoidal initialization by direct optimization
on the probability simplex.
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx):
        return np.inf
    n = arr.size
    return float(2.0 * n * mx / (s * s))


def _normalize(arr):
    arr = np.asarray(arr, dtype=np.float64).ravel()
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return None
    return arr / s


def _make_sine(n, amp, mode=1, phase=0.0):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * i / float(n) + phase
    a = 1.0 + amp * np.sin(mode * theta)
    return _normalize(a)


def _make_double_sine(n, a1, a2, m1=1, m2=2, p1=0.0, p2=0.0):
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * i / float(n)
    a = 1.0 + a1 * np.sin(m1 * theta + p1) + a2 * np.sin(m2 * theta + p2)
    return _normalize(a)


def _make_uniform(n):
    return np.ones(n, dtype=np.float64) / float(n)


def _compute_gradient(arr, c, K):
    n = arr.size
    c_len = c.size
    if c_len <= K:
        active = np.arange(c_len, dtype=np.int64)
    else:
        active = np.argpartition(c, -K)[-K:]
    
    grad = np.zeros(n, dtype=np.float64)
    cnt = 0
    for kk in active:
        k = int(kk)
        cnt += 1
        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_max >= i_min:
            idx = np.arange(i_min, i_max + 1, dtype=np.int64)
            grad[idx] += 2.0 * arr[k - idx]
    
    if cnt > 0:
        grad /= float(cnt)
    return grad
