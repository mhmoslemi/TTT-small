"""
Minimize 2*n*max(a*a)/(sum(a))^2 for nonnegative sequences.

Implements mirror descent (multiplicative weights) with annealed smooth-max
temperature, starting from the known best solution. Combines:
1. Mirror descent with geometric beta annealing
2. Perturbation + re-optimization to escape local minima
3. Projected gradient polishing at the end
4. Multiple sequence lengths for robustness

import numpy as np
import time


def fft_convolution(a):
    """Linear auto-convolution via FFT, clipped for numerical safety."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def fast_score(p):
    """Compute 2*n*max(p*p)/(sum(p))^2 using FFT."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float("inf")
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float("inf")
    b = fft_convolution(p)
    M = float(np.max(b))
    if not np.isfinite(M):
        return float("inf")
    return float(2.0 * n * M / (S * S))


def smooth_max_and_weights(b, beta):
    """Smooth max and softmax weights."""
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float("inf"), np.empty(0, dtype=np.float64)
    m = float(np.max(b))
    if not np.isfinite(m):
        return float("inf"), np.zeros_like(b, dtype=np.float64)
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b, dtype=np.float64)
        w /= float(b.size)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w


def corr_pw(p, w):
    """Cross-correlation: s_i = sum_j p_j * w_{i+j}."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    if n == 1:
        return np.array([float(np.dot(p, w[:1]))], dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n:m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)


def project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    if target <= 0.0:
        return np.zeros_like(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(v.shape, target / float(v.size), dtype=np.float64)
    low = float(np.min(v)) - 1.0
    high = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (low + high)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            low = mid
        else:
            high = mid
    tau = 0.5 * (low + high)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        q = np.full(v.shape, target / float(v.size), dtype=np.float64)
    else:
        q *= target / s
    return q


def normalize_simplex(p):
    """Clip nonneg and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size), dtype=np.float64)
    return p / s


def compute_grad(p, w):
    """Compute gradient of smooth max: 2 * corr(p, w)."""
    svec = corr_pw(p, w)
    return 2.0 * np.maximum(svec, 0.0)


def mirror_descent_refine(p0, deadline, beta_vals, max_iter_per_beta=200000):
    """
    Mirror descent with annealed beta.
    Uses multiplicative weights update on the simplex.
    """
    p = normalize_simplex(p0)
    n = p.size
    if n == 0:
        return p, float("inf")

    best_p = p.copy()
    best_score = fast_score(p)

    for beta in beta_vals:
        if time.monotonic() >= deadline:
            break

        eta = 1.0
        no_improve = 0
        max_stall = 30000

        for it in range(max_iter_per_beta):
            if time.monotonic() >= deadline:
                break

            b = fft_convolution(p)
            G, w = smooth_max_and_weights(b, beta)
            if not np.isfinite(G):
                break

            m = float(np.max(b))
            cur_score = 2.0 * n * m
            cur_smooth = n * G

            if cur_score < best_score - 1e-13:
                best_score = cur_score
                best_p = p.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > max_stall:
                break

            if it % 5000 == 0:
                print(f"  MD beta={beta:.0f} it={it} score={cur_score:.8f} eta={eta:.4f}")

            grad = compute_grad(p, w)

            # Mirror descent: p_j -> p_j * exp(-eta * grad_j) / Z
            # Use log-domain for numerical stability
            log_p = np.log(np.maximum(p, 1e-300))
            log_update = log_p - eta * grad
            log_update -= float(np.max(log_update))
            new_p = np.exp(log_update)

                    eta = min(eta * 1.05, 50.0)

def projected_gradient_refine(p0, deadline, beta, max_iter=200000):
    """Projected gradient descent on the simplex (Euclidean geometry)."""
    p = normalize_simplex(p0)
    n = p.size
    if n == 0:
        return p, float("inf")

    best_p = p.copy()
    best_score = fast_score(p)

    step = 0.5
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break

        b = fft_convolution(p)
        G, w = smooth_max_and_weights(b, beta)
        if not np.isfinite(G):
            break

        m = float(np.max(b))
        cur_score = 2.0 * n * m
        cur_smooth = n * G

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 15000:
            break

        if it % 5000 == 0:
            print(f"  PGD beta={beta:.0f} it={it} score={cur_score:.8f} step={step:.4f}")

        grad = compute_grad(p, w)
        grad = grad - float(np.mean(grad))
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:
            break

        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.01):
            alpha = step * mult
            q = project_simplex(p - alpha * grad, 1.0)
            qb = fft_convolution(q)
            qG, _ = smooth_max_and_weights(qb, beta)
            if np.isfinite(qG) and n * qG < cur_smooth - 1e-14:
                p = q
                cur_smooth = n * qG
                accepted = True
                if step < 5.0:
                    step = min(step * 1.1, 5.0)
                break

        if not accepted:
            step *= 0.5
            if step < 1e-8:
                break

    return best_p, best_score


def perturb_sequence(p, rng, strength=0.02):
    """Apply a random perturbation while maintaining the simplex."""
    p = np.asarray(p, dtype=np.float64).copy()
    n = p.size
    noise = rng.normal(0.0, strength, n)
    perturbed = p + noise
    return normalize_simplex(perturbed)


def structured_starts(n, rng, count=6):
    """Generate diverse structured starting points."""
    starts = []
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)

    # Uniform
    starts.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Linear ramps
    for a in (-3.0, 3.0, -1.5, 1.5):
        p = np.exp(a * (x - 0.5))
        starts.append(normalize_simplex(p))

    # Bimodal (U-shape)
    for t in (0.5, 1.0, 1.5):
        p = np.exp(t * (2.0 * x - 1.0) ** 2)
        starts.append(normalize_simplex(p))

    # Cosine patterns
    for freq in (2, 3, 5):
        p = np.exp(0.8 * np.cos(2 * np.pi * freq * x))
        starts.append(normalize_simplex(p))

    # Random smooth
    size = 1
    while size < n:
        size <<= 1
    zc = rng.normal(size=size) + 1j * rng.normal(size=size)
    z = np.real(np.fft.ifft(zc))[:n]
    z -= float(np.mean(z))
    z = 1.0 * np.tanh(z)
    p = np.exp(z)
    starts.append(normalize_simplex(p))

    return starts[:count]

        n_fb = 4096
        best_seq = [1.0] * n_fb
        print(f"Using uniform fallback, score should be ~2.0")
    else:
        print(f"\nFinal best score: {best_official_score:.8f} with length {len(best_seq)}")

    return best_seq
