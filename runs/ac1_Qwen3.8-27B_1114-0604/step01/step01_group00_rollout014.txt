```python
"""
Step function optimizer using L-BFGS-B with soft-max annealing and coordinate descent polish.
Minimizes 2*n*max(convolve(f,f))/(sum f)^2 for non-negative f.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def _convolve_fft(f):
    """FFT-based linear convolution of f with itself."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]

def _objective_gradient(f, T):
    """Compute soft-max objective and gradient. f is assumed to be non-negative."""
    n = len(f)
    b = _convolve_fft(f)
    S = np.sum(f)
    
    if S < 1e-15:
        return 1e10, np.zeros(n)
    
    # Soft-max of b with temperature T
    b_scaled = b / T
    b_max_val = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max_val)
    p_sum = exp_b.sum()
    p = exp_b / p_sum
    M_T = T * (np.log(p_sum) + b_max_val)
    
    # Gradient of soft-max wrt f
    grad_soft = 2.0 * correlate(p, f, mode='valid')
    
    # Full gradient
    J = 2.0 * n * M_T / (S * S)
    grad = 2.0 * n * (grad_soft * S - 2.0 * M_T) / (S * S * S)
    
    return J, grad

def _eval(f):
    """Evaluate using the provided function."""
    return evaluate_sequence(f.tolist())

def _lbfgsb_round(f, T, max_time, rng, label=""):
    """Run one round of L-BFGS-B with fixed temperature T."""
    n = len(f)
    
    def fun_g(x):
        x = np.maximum(x, 0.0)  # Ensure non-negativity
        return _objective_gradient(x, T)
    
    # Use a callback to check time
    start_time = time.monotonic()
    call_count = [0]
    
    def callback(xk):
        call_count[0] += 1
        if time.monotonic() - start_time > max_time:
            return True  # Stop
        return None
    
    result = minimize(
        fun_g, f, method='L-BFGS-B',
        jac=True,
        bounds=[(0, None)] * n,
        options={
            'maxiter': 10000,
            'maxfun': 100000,
            'ftol': 1e-15,
            'gtol': 1e-8,
            'maxls': 50,
            'callback': callback,
            'disp': False
        }
    )
    
    f_new = np.maximum(result.x, 0.0)
    score_new = _eval(f_new)
    
    if label:
        print(f"    {label}: T={T:.2e}, score={score_new:.6f}, iters={result.nit}, fun_evals={result.nfev}")
    
    return f_new, score_new

def _polish(f, deadline, rng):
    """Coordinate descent polish: try pairwise mass transfers to reduce peak."""
    n = len(f)
    f = f.copy()
    S = f.sum()
    f = f / S  # Normalize
    
    b = _convolve_fft(f)
    peak = b.max()
    best_score = _eval(f)
    best_f = f.copy()
    
    # Precompute: for each peak lag, the contribution of each position
    n_trials = 0
    max_trials = 500000
    
    while time.monotonic() < deadline and n_trials < max_trials:
        # Find current peak
        k_star = int(np.argmax(b))
        
        # Compute contribution vector for peak lag
        # b[k_star] = sum_i f[i] * f[k_star - i] (valid i)
        # Contribution of position i: f[i] * f[k_star - i] (if valid)
        contrib = np.zeros(n)
        for i in range(n):
            j = k_star - i
            if 0 <= j < n:
                contrib[i] = f[i] * f[j]
        
        # Find best pair to swap: donate from high contrib, receive at low contrib
        # Use top/bottom K by contribution
        K = min(50, n)
        top_idx = np.argsort(contrib)[::-1][:K]
        bot_idx = np.argsort(contrib)[:K]
        
        improved = False
        for i in top_idx:
            for j in bot_idx:
                if i == j:
                    continue
                # Try moving a fraction of f[i] to f[j]
                for frac in [0.01, 0.05, 0.1]:
                    amt = f[i] * frac
                    if amt <= 0:
                        continue
```

```python
"""
Step function optimizer using L-BFGS-B with soft-max annealing and coordinate descent polish.
Minimizes 2*n*max(convolve(f,f))/(sum f)^2 for non-negative f.
Uses FFT-based convolution/correlation for speed, geometric temperature annealing,
and a final pairwise-transfer polish phase.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time


def _convolve_fft(f):
    """FFT-based linear convolution of f with itself."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _objective_gradient(f, T):
    """Compute soft-max objective and gradient for f >= 0."""
    n = len(f)
    b = _convolve_fft(f)
    S = np.sum(f)

    if S < 1e-15:
        return 1e10, np.zeros(n)

    # Soft-max of b with temperature T
    b_scaled = b / T
    b_max_val = b_scaled.max()
    exp_b = np.exp(b_scaled - b_max_val)
    p_sum = exp_b.sum()
    p = exp_b / p_sum
    M_T = T * (np.log(p_sum) + b_max_val)

    # Gradient of soft-max wrt f: 2 * correlate(p, f, 'valid')
    grad_soft = 2.0 * correlate(p, f, mode='valid')

    # Full gradient of J = 2*n*M_T / S^2
    J = 2.0 * n * M_T / (S * S)
    grad = 2.0 * n * (grad_soft * S - 2.0 * M_T) / (S * S * S)

    return J, grad

def _lbfgsb_round(f, T, max_time, label=""):

        val, g = _objective_gradient(x_clipped, T)

        g = np.where(x < 0, 0.0, g)
        return val, g

def _mirror_descent_round(f, T, max_time, rng, label=""):
    """Mirror descent (exponentiated gradient) as fallback/complement to L-BFGS-B."""
    n = len(f)
    f = f.copy()
    start_time = time.monotonic()
    best_local = _eval(f)
    best_local_f = f.copy()
    it = 0
    peak0 = _convolve_fft(f).max()
    T_max = peak0 * 5.0
    T_min = peak0 * 1e-5

    while time.monotonic() - start_time < max_time:
        progress = (time.monotonic() - start_time) / max_time
        T_cur = max(T_min, min(T_max, T * (1.0 - progress * 0.5)))

        b = _convolve_fft(f)
        val, grad = _objective_gradient(f, T_cur)

        # Mirror descent step
        step = 0.5 * (1.0 - progress) + 0.01
        span = grad.max() - grad.min()
        if span > 1e-18:
            eta = step / span
            log_f = np.log(f + 1e-300) - eta * grad
            log_f -= log_f.max()
            f_new = np.exp(log_f)
            s = f_new.sum()
            if s > 1e-300:
                f = f_new / s
            it += 1

        # Evaluate periodically
        if it % 500 == 0:
            sc = _eval(f)
            if sc < best_local:
                best_local = sc
                best_local_f = f.copy()

    sc = _eval(f)
    if sc < best_local:
        best_local = sc
        best_local_f = f.copy()

    if label:
        print(f"    {label}: mirror descent score={best_local:.6f}, iters={it}")
    return best_local_f, best_local

    current_f = f0.copy()
    for i in range(n_rounds):
        if time.monotonic() > deadline - 900:
            break
        T = temperatures[i]
        t_alloc = min(weights[i], deadline - time.monotonic() - 900)
        if t_alloc < 15:
            break
        print(f"  L-BFGS-B round {i+1}/{n_rounds}: T={T:.3e}, alloc={t_alloc:.0f}s")
        f_new, sc_new = _lbfgsb_round(current_f, T, t_alloc, label=f"B{i+1}")
        if sc_new < best_score:
            best_score = sc_new
            best_f = f_new.copy()
            print(f"    -> New best: {best_score:.6f}")
        current_f = f_new

    # Phase 2: Mirror descent with small T (complement to L-BFGS-B)
    if time.monotonic() < deadline - 700:
        T_md = peak0 * 1e-4
        t_md = min(200.0, deadline - time.monotonic() - 700)
        print(f"\n  Mirror descent: T={T_md:.3e}, alloc={t_md:.0f}s")
        f_md, sc_md = _mirror_descent_round(best_f, T_md, t_md, rng, label="MD")
        if sc_md < best_score:
            best_score = sc_md
            best_f = f_md.copy()
            print(f"    -> New best: {best_score:.6f}")

    # Phase 3: Coordinate descent polish
    if time.monotonic() < deadline - 200:
        polish_deadline = min(time.monotonic() + budget_s * 0.25, deadline - 150)
        print(f"\n  Polish phase (deadline in {polish_deadline - time.monotonic():.0f}s)")
        f_pol, sc_pol = _polish(best_f, polish_deadline, rng)
        if sc_pol < best_score:
            best_score = sc_pol
            best_f = f_pol.copy()
            print(f"    -> Polished to: {best_score:.6f}")

    # Phase 4: Final L-BFGS-B polish
    if time.monotonic() < deadline - 30:
        T_fin = peak0 * 1e-5
        t_fin = min(120.0, deadline - time.monotonic() - 10)
        if t_fin > 10:
            print(f"\n  Final L-BFGS-B: T={T_fin:.3e}, alloc={t_fin:.0f}s")
            f_fin, sc_fin = _lbfgsb_round(best_f, T_fin, t_fin, label="Final")
            if sc_fin < best_score:
                best_score = sc_fin
                best_f = f_fin.copy()
                print(f"    -> Final: {best_score:.6f}")

    # Phase 5: One last quick polish
    if time.monotonic() < deadline - 5:
        f_last, sc_last = _polish(best_f, deadline - 5, rng)
        if sc_last < best_score:
            best_score = sc_last
            best_f = f_last.copy()

    # Ensure valid output
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = _eval(best_f)
    if final_score < best_score:
        best_score = final_score

    print(f"\n{'='*50}")
    print(f"=== FINAL SCORE: {best_score:.6f} ===")
    print(f"{'='*50}")
    return [float(x) for x in best_f]
```
