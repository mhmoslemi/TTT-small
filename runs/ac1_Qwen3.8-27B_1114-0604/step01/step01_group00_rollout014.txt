import time

def compute_conv(a):
    """Compute full linear convolution of a with itself."""
    n = len(a)
    size = 1 << (2*n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2*n-1]

    n = len(a)
    conv = compute_conv(a)

    c -= c.max()
    p = np.exp(c)
    p /= p.sum()

    grad_f = (2 * n / S) * (grad_M - 2 * M_smooth / S)
    
    return f, grad_f

from scipy.signal import correlate

def _conv_fft(a):
    """Compute full linear convolution a*a via FFT."""
    n = len(a)
    size = 1 << (2 * n - 1).bit_length()
    A = np.fft.rfft(a, size)
    return np.fft.irfft(A * A, size)[:2 * n - 1]

    """Smoothed objective: 2n * logsumexp(conv/T) / (sum a)^2"""
    n = len(a)
    conv = _conv_fft(a)
    S = a.sum()
    cs = conv / T
    cs -= cs.max()
    M = T * (np.log(np.exp(cs).sum()) + cs.max())
    return 2 * n * M / (S * S)
