```
def adam_update(theta, g, m, v, step, lr, b1=0.9, b2=0.999, eps=1e-8):
```

```python
"""
Search for a non-negative step-function sequence minimizing
    2*n*max(self-convolution) / (sum(sequence))^2.

Key ideas
---------
1. Use a large but tractable length (8192).  The asymptotic optimum is
   accessible, and direct evaluation by np.convolve is still fast.

2. Start from strongly asymmetric trigonometric "flat-convolution"
   initializations.  In particular, densities of the form
       f(x) = (1 - sin(2*pi*(x+phase)))**gamma
   have unusually flat linear self-convolution, far below uniform.

3. Refine with a smooth-max gradient method:
   - keep probabilities normalized to sum 1;
   - approximate max_k c_k by log-sum-exp with annealed concentration;
   - compute the exact smooth gradient by FFT convolution;
   - optimize logits with Adam and periodic randomized restarts.

The function propose_candidate() uses a hard wall-clock budget and always
returns the best finite sequence found.
"""

import numpy as np
import time

TWO_PI = 2.0 * np.pi


def next_power_of_two(m):
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def normalize_positive(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)

    a = np.where(np.isfinite(a) & (a > 0.0), a, 0.0)
    s = float(a.sum())

    if s <= 0.0 or not np.isfinite(s):
        return np.ones(a.size, dtype=np.float64) / float(a.size)

    return a / s


def self_convolve(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    if n == 0:
        return np.zeros(0, dtype=np.float64)

    m = 2 * n - 1
    m_fft = next_power_of_two(m)
    A = np.fft.rfft(a, m_fft)
    return np.fft.irfft(A * A, m_fft)[:m]


def score_probability(p):
    """Score assuming p is non-negative and sums to 1."""
    p = np.asarray(p, dtype=np.float64).ravel()
    n = p.size
    if n == 0:
        return np.inf

    c = self_convolve(p)
    np.maximum(c, 0.0, out=c)

    if c.size == 0:
        return np.inf

    mc = float(c.max())
    if (not np.isfinite(mc)) or mc <= 0.0:
        return np.inf

    return float(2.0 * n * mc)


def make_sine_power(x, amp=1.0, phase=0.0, gamma=1.0):
    x = np.asarray(x, dtype=np.float64)
    val = 1.0 - float(amp) * np.sin(TWO_PI * (x + float(phase)))
    val = np.clip(val, 0.0, None)

    if gamma != 1.0:
        g = float(gamma)
        if g <= 0.0:
            val = np.ones_like(val)
        else:
            val = np.power(val, g)

    return normalize_positive(val)


def make_sine_harm(x, amp=1.0, phase=0.0, gamma=1.0, amp2=0.0, phase2=0.0):
    x = np.asarray(x, dtype=np.float64)
    val = 1.0 - float(amp) * np.sin(TWO_PI * (x + float(phase)))

    if abs(float(amp2)) > 1e-12:
        val = val + float(amp2) * np.sin(4.0 * np.pi * (x + float(phase2)))

    val = np.clip(val, 0.0, None)

    if gamma != 1.0:
        g = float(gamma)
        if g <= 0.0:
            val = np.ones_like(val)
        else:
            val = np.power(val, g)

    return normalize_positive(val)


def make_random_fourier(x, rng):
    x = np.asarray(x, dtype=np.float64)
    val = np.ones_like(x)

    if rng.random() < 0.8:
        a1 = float(rng.uniform(-1.4, -0.3))
    else:
        a1 = float(rng.uniform(0.3, 1.4))

    ph1 = float(rng.random())
    val = val + a1 * np.sin(TWO_PI * (x + ph1))

    for m in (2, 3, 4):
        a = float(rng.uniform(-0.30, 0.30))
        ph = float(rng.random())
        val = val + a * np.sin(TWO_PI * (m * x + ph))

    val = np.clip(val, 0.0, None)
    s = float(val.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.ones_like(x) / float(x.size)
    return val / s


def get_parent_array():
    arr = globals().get("height_sequence_1", None)
    if arr is None:
        return None

    try:
        a = np.asarray(arr, dtype=np.float64).ravel()
    except Exception:
        return None

    if a.size == 0:
        return None

    return np.where(np.isfinite(a) & (a > 0.0), a, 0.0)


def resample_sequence(arr, n):
    if arr is None:
        return None

    a = np.asarray(arr, dtype=np.float64).ravel()
    if a.size == 0:
        return None

    a = np.where(np.isfinite(a) & (a > 0.0), a, 0.0)
    s = float(a.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None

    a = a / s

    if a.size == n:
        return a

    old_x = np.linspace(0.0, 1.0, a.size)
    new_x = np.linspace(0.0, 1.0, n)
    q = np.interp(new_x, old_x, a)

    return normalize_positive(q)


def cand_key(item):
    return item[0]


def build_initial_candidates(n, rng, parent_p):
    cands = []
    x = np.arange(n, dtype=np.float64) / float(n)

    if parent_p is not None:
        sc = score_probability(parent_p)
        if np.isfinite(sc):
            cands.append((sc, parent_p.copy()))

    p_uniform = np.ones(n, dtype=np.float64) / float(n)
    sc = score_probability(p_uniform)
    if np.isfinite(sc):
        cands.append((sc, p_uniform))

    phases = np.linspace(0.0, 1.0, 20, endpoint=False)
    amps = (1.0, 1.3)
    gammas = (0.5, 0.75, 1.0, 1.5, 2.0)

    for ph in phases:
        for amp in amps:
            for gam in gammas:
                p = make_sine_power(x, amp, float(ph), gam)
                sc = score_probability(p)
                if np.isfinite(sc):
                    cands.append((sc, p))

    for _ in range(120):
        ph = float(rng.random())
        amp = float(rng.uniform(0.5, 1.6))
        gam = float(rng.uniform(0.3, 2.5))
        amp2 = float(rng.uniform(-0.35, 0.35))
        ph2 = float(rng.random())

        p = make_sine_harm(x, amp, ph, gam, amp2, ph2)
        sc = score_probability(p)
        if np.isfinite(sc):
            cands.append((sc, p))

    for _ in range(50):
        p = make_random_fourier(x, rng)
        sc = score_probability(p)
        if np.isfinite(sc):
            cands.append((sc, p))

    cands.sort(key=cand_key)

    out = []
    for sc, p in cands:
        if len(out) >= 12:
            break
        out.append((sc, p))

    return out


def softmax_stable(theta):
    theta = np.asarray(theta, dtype=np.float64)
    n = theta.size

    if (not np.all(np.isfinite(theta))) or n == 0:
        return np.ones(n, dtype=np.float64) / float(max(1, n))

    t = theta - np.max(theta)
    e = np.exp(t)
    s = float(e.sum())

    if s <= 0.0 or not np.isfinite(s):
        return np.ones(n, dtype=np.float64) / float(n)

    return e / s


def gradient_by_weights(p, w, fft_len):
    """
    Gradient of sum_k w_k c_k with respect to p, where
    c_k = sum_j p_j p_{k-j}.

    Returns grad_i = 2 * sum_{l=0}^{n-1} w_{i+l} p_l.
    """
    p = np.asarray(p, dtype=np.float64).ravel()
    w = np.asarray(w, dtype=np.float64).ravel()
    n = p.size

    if n == 1:
        return np.array([2.0 * float(w[0]) * float(p[0])], dtype=np.float64)

    pr = p[::-1]
    W = np.fft.rfft(w, fft_len)
    P = np.fft.rfft(pr, fft_len)
    conv = np.fft.irfft(W * P, fft_len)

    return 2.0 * conv[n - 1:2 * n - 1]


def adam_update(theta, g, m, v, step, lr, b1=0.9, b2=0.999, eps=1e-8):
    b1t = b1 ** step
    b2t = b2 ** step

    m *= b1
    m += (1.0 - b1) * g

    v *= b2
    v += (1.0 - b2) * g * g

    mh = m / (1.0 - b1t)
    vh = v / (1.0 - b2t)

    step_arr = mh / (np.sqrt(vh) + eps)

    mean_step = float(np.mean(step_arr))
    if np.isfinite(mean_step):
        step_arr -= mean_step

    theta -= lr * step_arr


def optimize_sequence(p0, limit_time, rng, base_lr, C_start, C_end, restart_gap, tag=""):
    p = normalize_positive(p0)
    n = int(p.size)

    c = self_convolve(p)
    np.maximum(c, 0.0, out=c)
    maxc = float(c.max())
    best_score = 2.0 * n * maxc
    best_p = p.copy()

    theta = np.log(np.maximum(p, 1e-15))
    theta -= float(np.mean(theta))

    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    adam_step = 0

    phase_start = time.monotonic()
    if phase_start >= limit_time:
        return best_score, best_p

    last_improve = phase_start
    last_restart = phase_start
    lr = float(base_lr)

    grad_fft_len = next_power_of_two(3 * n - 2)
    x_for_restart = np.arange(n, dtype=np.float64) / float(n)

    while True:
        now = time.monotonic()
        if now >= limit_time or (limit_time - now) < 0.05:
            break

        c = self_convolve(p)
        np.maximum(c, 0.0, out=c)
        maxc = float(c.max())
        score = 2.0 * n * maxc

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_p = p.copy()
            last_improve = now

        if (now - last_improve > restart_gap) and (now - last_restart > 0.5 * restart_gap):
            if rng.random() < 0.30:
                sp = make_sine_power(
                    x_for_restart,
                    float(rng.uniform(0.9, 1.3)),
                    float(rng.random()),
                    float(rng.uniform(0.5, 2.0)),
                )
                mix = float(rng.uniform(0.30, 0.70))
                p = mix * best_p + (1.0 - mix) * sp
                p = normalize_positive(p)
            else:
                scale = float(0.08 + 0.15 * rng.random())
                theta = np.log(np.maximum(best_p, 1e-15)) + scale * rng.standard_normal(n)
                theta -= float(np.mean(theta))
                p = softmax_stable(theta)

            theta = np.log(np.maximum(p, 1e-15))
            theta -= float(np.mean(theta))
            m.fill(0.0)
            v.fill(0.0)
            adam_step = 0
            lr = max(0.002, lr * 0.75)
            last_restart = now
            last_improve = now
            continue

        if (not np.all(np.isfinite(theta))):
            theta = np.log(np.maximum(p, 1e-15))
            theta -= float(np.mean(theta))
            m.fill(0.0)
            v.fill(0.0)
            adam_step = 0
            continue

        if maxc <= 1e-300:
            break

        frac = (now - phase_start) / max(1e-9, limit_time - phase_start)
        if frac < 0.0:
            frac = 0.0
        if frac > 1.0:
            frac = 1.0

        C = C_start + (C_end - C_start) * frac

        # Reuse c as the log-sum-exp weight array.
        c -= maxc
        c /= maxc
        c *= C
        np.clip(c, -700.0, 0.0, out=c)
        w = c
        np.exp(w, out=w)

        wsum = float(w.sum())
        if (not np.isfinite(wsum)) or wsum <= 0.0:
            w = np.ones_like(c) / float(c.size)
        else:
            w /= wsum

        grad = gradient_by_weights(p, w, grad_fft_len)

        if (not np.all(np.isfinite(grad))):
            continue

        dot = float(np.dot(p, grad))
        g = grad - p * dot

        if (not np.all(np.isfinite(g))):
            continue

        adam_step += 1
        adam_update(theta, g, m, v, adam_step, lr)
        p = softmax_stable(theta)

    return best_score, best_p


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    start = time.monotonic()

    try:
        total = float(budget_s)
    except Exception:
        total = 2400.0

    if (not np.isfinite(total)) or total <= 0.0:
        total = 2400.0

    total = min(max(total, 120.0), 2450.0)

    # Reserve time for final validation and return.
    deadline = start + total - 60.0

    rng = np.random.default_rng(seed)

    # Reasonable fixed length: large enough for near-asymptotic behavior,
    # small enough that direct np.convolve evaluation remains fast.
    n = 8192

    parent = get_parent_array()
    parent_p = resample_sequence(parent, n) if parent is not None else None

    cands = build_initial_candidates(n, rng, parent_p)

    if not cands:
        p_uniform = np.ones(n, dtype=np.float64) / float(n)
        sc = score_probability(p_uniform)
        cands.append((sc, p_uniform))

    cands.sort(key=cand_key)

    best_score = float(cands[0][0])
    best_p = cands[0][1].copy()

    print("initial candidates: %d, best initial score: %.9f" % (len(cands), best_score), flush=True)

    # Short exploration phase: refine the best few initializations.
    short_deadline = start + 0.35 * (total - 60.0)
    short_deadline = min(short_deadline, deadline - 120.0)

    if short_deadline > time.monotonic() + 30.0:
        short_duration = short_deadline - time.monotonic()
        top_count = min(3, len(cands))
        weights = [0.60, 0.25, 0.15]

        for i in range(top_count):
            remaining = short_deadline - time.monotonic()
            if remaining < 15.0:
                break

            if i < top_count - 1:
                alloc = min(remaining, max(15.0, short_duration * weights[i]))
            else:
                alloc = remaining

            limit = time.monotonic() + alloc
            cand_p = cands[i][1]
            cand_score = float(cands[i][0])

            if best_score > 1.30:
                sc, p = optimize_sequence(
                    cand_p,
                    limit,
                    rng,
                    base_lr=0.05,
                    C_start=4.0,
                    C_end=80.0,
                    restart_gap=15.0,
                    tag="short%d" % i,
                )
            else:
                sc, p = optimize_sequence(
                    cand_p,
                    limit,
                    rng,
                    base_lr=0.03,
                    C_start=8.0,
                    C_end=60.0,
                    restart_gap=20.0,
                    tag="short%d" % i,
                )

            if np.isfinite(sc) and sc < best_score:
                best_score = float(sc)
                best_p = p.copy()
                print("short %d improved: %.9f" % (i, best_score), flush=True)
            else:
                # Keep the better candidate if the optimizer fails.
                if float(cand_score) < best_score:
                    best_score = float(cand_score)
                    best_p = cand_p.copy()
                    print("short %d candidate better: %.9f" % (i, best_score), flush=True)

    # Final refinement of the global best.
    if time.monotonic() < deadline - 10.0:
        if best_score > 1.30:
            sc, p = optimize_sequence(
                best_p,
                deadline,
                rng,
                base_lr=0.02,
                C_start=10.0,
                C_end=120.0,
                restart_gap=35.0,
                tag="final",
            )
        else:
            sc, p = optimize_sequence(
                best_p,
                deadline,
                rng,
                base_lr=0.01,
                C_start=20.0,
                C_end=180.0,
                restart_gap=60.0,
                tag="final",
            )

        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_p = p.copy()

    print("final internal score: %.9f" % best_score, flush=True)

    out = []
    for x in best_p:
        out.append(float(x))

    try:
        verified = float(evaluate_sequence(out))
    except Exception as exc:
        print("verification failed: %s" % exc, flush=True)
        verified = np.inf

    if (not np.isfinite(verified)) or verified > 1e6:
        arr = np.asarray(best_p, dtype=np.float64).ravel()
        arr = np.where(np.isfinite(arr) & (arr > 0.0), arr, 0.0)
        s = float(arr.sum())
        if s <= 0.0 or not np.isfinite(s):
            arr = np.ones(n, dtype=np.float64) / float(n)
        else:
            arr = arr / s

        out = []
        for x in arr:
            out.append(float(x))

        try:
            verified = float(evaluate_sequence(out))
        except Exception:
            verified = np.inf

    print("final verified score: %.9f, length: %d" % (verified, len(out)), flush=True)

    return out
```
