    w /= w_sum

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS optimization of smooth
power-mean and log-sum-exp objectives with softmax parameterization,
annealing, m-sequence initializations, multiple sequence lengths,
and subgradient polishing.
"""

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time


def softmax(x):
    x = x - x.max()
    e = np.exp(x)
    return e / e.sum()


def full_conv(a):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = rfft(a, fsize)
    return irfft(fa * fa, fsize)[:size]


def normed_score(a):
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def power_mean_grad(x, p):
    """Compute power-mean objective and its gradient w.r.t. x (log-softmax params)."""
    n = len(x)
    a = softmax(x)
    b = full_conv(a)
    m = b.size  # 2n-1

    f = np.mean(b ** p)

    # df/da_j = (2p/m) * sum_k b_k^(p-1) * a_{k-j} = (2p/m) * c_j
    w = b ** (p - 1)
    c = np.correlate(w, a, mode='valid')
    s = np.dot(w, b)  # sum_k w_k * b_k

    # df/dx_l = (2p/m) * a_l * (c_l - s)
    grad = (2.0 * p / m) * a * (c - s)
    return f, grad


def logsumexp_grad(x, tau):
    """Compute log-sum-exp smoothed max and its gradient w.r.t. x."""
    n = len(x)
    a = softmax(x)
    b = full_conv(a)
    m = b.size

    b_max = b.max()
    shifted = (b - b_max) / tau
    w = np.exp(shifted)
    w_sum = w.sum()
    w /= w_sum
    f = b_max / tau + np.log(w_sum)

    c = np.correlate(w, a, mode='valid')
    s = np.dot(w, b)
    grad = (2.0 / tau) * a * (c - s)
    return f, grad


def lbfgs_powermean(x0, time_budget, p_schedule, maxiter=150):
    """Run L-BFGS with annealing power p. x0 is log-softmax params."""
    n = len(x0)
    best_a = softmax(x0)
    best_score = normed_score(best_a)
    x = x0.copy()
    t0 = time.monotonic()

    for p in p_schedule:
        if time.monotonic() - t0 > time_budget:
            break

        def fg_power(x):
            return power_mean_grad(x, p)

        try:
            result = minimize(fg_power, x, method='L-BFGS-B', jac=True,
                              options={'maxiter': maxiter, 'maxfun': maxiter * 5,
                                       'ftol': 1e-10, 'gtol': 1e-8, 'disp': False})
            x = result.x
        except Exception:
            pass

        a = softmax(x)
        score = normed_score(a)
        if score < best_score:
            best_score = score
            best_a = a.copy()
            print(f"      p={p}: score={score:.6f}")

    return best_a, best_score


def lbfgs_logsumexp(x0, time_budget, tau_schedule, maxiter=150):

    n = len(x0)

    best_score = normed_score(best_a)
    x = x0.copy()

def generate_msequence(n):
    """Generate a binary m-sequence of length n (if n = 2^m - 1) or approximate."""
    # Find m such that 2^m - 1 >= n, or use a primitive polynomial
    # For simplicity, use a LFSR with a known primitive polynomial
    # Degree 7: x^7 + x^3 + 1 (for n=127)
    # Degree 8: x^8 + x^4 + x^3 + x^2 + 1 (for n=255)
    # Degree 9: x^9 + x^5 + x^3 + x^2 + 1 (for n=511)
    # Degree 10: x^10 + x^3 + 1 (for n=1023)

    # Try to find the right degree
    best_m = 7
    best_poly = 0b1001001  # x^7 + x^3 + 1
    for m, poly in [(7, 0b1001001), (8, 0b100111001), (9, 0b1001011001),
                    (10, 0b1000001001), (11, 0b10000101001)]:
        if (1 << m) - 1 >= n:
            best_m = m
            best_poly = poly
            break
    else:
        best_m = 11
        best_poly = 0b10000101001

    # Generate LFSR sequence
    m = best_m
    state = 1  # initial state (any nonzero)
    seq = []
    for _ in range((1 << m) - 1):
        seq.append(state & 1)
        # Shift and compute new bit
        # The primitive polynomial determines the feedback
        # For x^m + ... + 1, the feedback is XOR of specific bits
        # poly bit m is always 1 (the x^m term)
        # Feedback = XOR of bits where poly has 1s (excluding the top bit)
        fb = state & 1
        poly_lower = best_poly & ((1 << (m - 1)) - 1)  # lower m-1 bits
        for bit in range(m - 1):
            if poly_lower & (1 << bit):
                fb ^= (state >> bit) & 1
        state = (state >> 1) | (fb << (m - 1))

    # Convert to non-negative real values with small offset
    a = np.array(seq, dtype=np.float64)
    a = a * 0.7 + 0.15  # map {0,1} to {0.15, 0.85}
    # Pad or truncate to desired length
    if a.size >= n:
        a = a[:n]
    else:
        pad = np.full(n - a.size, 0.15)
        a = np.concatenate([a, pad])
    return a / a.sum()


def get_inits(n, rng, prev_a=None):
    """Generate diverse initializations for sequence of length n."""
    inits = []
    i = np.arange(n, dtype=np.float64)

    # Previous best (resized)
    if prev_a is not None:
        if prev_a.size == n:
            inits.append(prev_a.copy())
        else:
            old_x = np.linspace(0, 1, prev_a.size)
            new_x = np.linspace(0, 1, n)
            from scipy.interpolate import interp1d
            f = interp1d(old_x, prev_a, kind='cubic', fill_value='extrapolate')
            a = np.maximum(f(new_x), 0.0)
            if a.sum() > 0:
                inits.append(a / a.sum())

    # M-sequence based
    inits.append(generate_msequence(n))

    # Uniform
    inits.append(np.ones(n) / n)

    # Two Gaussian bumps
    c1, c2 = n * 0.25, n * 0.75
    sigma = n * 0.07
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.01
    inits.append(a / a.sum())

    # U-shape (high at edges)
    m = np.minimum(i, n - 1 - i).astype(np.float64) + 1
    a = 1.0 / np.sqrt(m)
    inits.append(a / a.sum())

    # Random
    a = np.abs(rng.standard_normal(n)) + 0.1
    inits.append(a / a.sum())

    # Cosine modulated
    for freq in [2, 3, 5]:
        a = (1.0 + 0.5 * np.cos(2 * np.pi * freq * i / n)) ** 2
        a += 0.01
        inits.append(a / a.sum())

    # Sided (quadratic, high at edges)
    c = (n - 1) / 2
    a = ((i - c) / max(c, 1)) ** 2 + 0.1
    inits.append(a / a.sum())

    # Block structure
    a = np.ones(n)
    a[n // 5:2 * n // 5] = 0.3
    a[3 * n // 5:4 * n // 5] = 0.3
    inits.append(a / a.sum())

    # Noise on flat
    a = np.ones(n) * 0.5 + 0.5 * np.abs(rng.standard_normal(n))
    inits.append(a / a.sum())

    return inits


def polish_subgrad(a, time_budget, rng):
    """Subgradient polish with the analytical gradient of the smoothed max."""
    n = a.size
    a = a / a.sum()
    best_a = a.copy()
    best_sc = normed_score(a)

    eta = 0.03
    tau_frac = 0.03
    vel = np.zeros(n)
    beta = 0.6
    iters = 0
    last_improve = 0

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))
        tau = max(tau_frac * float(np.max(b)), 1e-18)

        shifted = (b - b.max()) / tau
        w = np.exp(shifted)
        w /= w.sum()

        c = np.correlate(w, a, mode='valid')
        s = np.dot(w, b)
        g = (2.0 / tau) * a * (c - s)

        vel = beta * vel + (1 - beta) * g
        a_new = a - eta * vel
        a_new = np.maximum(a_new, 0)
        sm = a_new.sum()
        if sm < 1e-10:
            a_new = best_a.copy()
        else:
            a_new = a_new / sm

        new_sc = normed_score(a_new)

        if new_sc <= cur_sc or rng.random() < 0.02:
            a = a_new
        else:
            a = a - 0.5 * eta * vel
            a = np.maximum(a, 0)
            sm = a.sum()
            if sm < 1e-10:
                a = best_a.copy()
            else:
                a = a / sm

        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = iters

        iters += 1

        # Restart if stuck
        if iters - last_improve > 3000:
            a = best_a + 0.005 * rng.standard_normal(n)
            a = np.maximum(a, 0)
            a = a / a.sum()
            vel = np.zeros(n)
            eta = 0.01 + 0.02 * rng.random()
            tau_frac = 0.02 + 0.03 * rng.random()
            last_improve = iters

        # Adaptive step size
        if iters % 5000 == 0:
            eta *= 0.8

    return best_a, best_sc

            b0 = full_conv(a_pert)
            bmax0 = float(b0.max())
            p_schedule = [4, 8, 16, 32]
            a_opt, sc = lbfgs_powermean(x0, tb, p_schedule, maxiter=100)

            if sc < best_score:
                best_score = sc
                best_a = a_opt.copy()
                print(f"    restart {ri}: NEW BEST {sc:.6f}")

    # Final polish phase
    if best_a is not None and time.monotonic() < deadline - 10:
        remain = deadline - time.monotonic() - 5
        print(f"\n[polish] L-BFGS polish with {remain * 0.4:.1f}s")
        a_p, sc_p = polish_lbfgs(best_a, remain * 0.4)
        if sc_p < best_score:
            best_score = sc_p
            best_a = a_p.copy()
            print(f"    L-BFGS polish: {sc_p:.6f}")

        if time.monotonic() < deadline - 10:
            remain2 = deadline - time.monotonic() - 5
            print(f"[polish] Subgradient polish with {remain2:.1f}s")
            a_p2, sc_p2 = polish_subgrad(best_a, remain2, rng)
            if sc_p2 < best_score:
                best_score = sc_p2
                best_a = a_p2.copy()
                print(f"    Subgrad polish: {sc_p2:.6f}")

    # Fallback
    if best_a is None:
        best_a = np.ones(128) / 128
        best_score = 2.0

    result = [float(x) for x in best_a]
    final_sc = float(evaluate_sequence(result))
    print(f"\n[final] n={len(result)}, internal={best_score:.6f}, evaluator={final_sc:.6f}")

    return result
