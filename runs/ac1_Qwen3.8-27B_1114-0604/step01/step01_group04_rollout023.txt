import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time

def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = rfft(a, fsize)
    return irfft(fa * fa, fsize)[:size]

def soft_max_val(b, tau):
    """Stable log-sum-exp approximation of max(b)."""
    bm = b.max()
    return bm + tau * np.log(np.sum(np.exp((b - bm) / tau)))

def soft_max_weights(b, tau):
    """Softmax weights for smooth max."""
    shifted = (b - b.max()) / tau
    w = np.exp(shifted)
    w /= w.sum()
    return w

def compute_grad(a, tau):
    """Gradient of 2*n*soft_max(conv(a,a)) w.r.t. a."""
    n = a.size
    b = full_conv(a)
    w = soft_max_weights(b, tau)
    
    # G[j] = 2 * (sum_k a[k]*w[j+k] + a[j]*sum_k w[j+k])
    # First part: cross-correlation
    corr = np.correlate(a, w, mode='full')
    G1 = 2.0 * corr[2*n-2 : n-1 : -1]
    
    # Second part: a[j] * row_sum[j]
    cumsum_w = np.concatenate([[0.0], np.cumsum(w)])
    S = cumsum_w[n:n+n] - cumsum_w[0:n]
    G2 = 2.0 * a * S
    
    return 2.0 * n * (G1 + G2)

def slsqp_single(a, tau, maxiter=300):
    """Run one round of SLSQP at a given temperature."""
    n = a.size
    
    def obj(a):
        b = full_conv(a)
        return 2.0 * n * soft_max_val(b, tau)
    
    def grad(a):
        return compute_grad(a, tau)
    
    def constraint(a):
        return a.sum() - 1.0
    
    def constraint_jac(a):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': constraint, 'jac': constraint_jac}]
    bounds = [(0.0, None)] * n
    
    result = minimize(obj, a, jac=grad, method='SLSQP', 
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
    
    if result.success or result.fun < obj(a):
        a_new = np.clip(result.x, 0, None)
        a_new = a_new / a_new.sum()
        return a_new, result.fun
    else:
        return a, obj(a)

def anneal_optimize(a_init, n, time_budget, rng):
    """Run SLSQP with geometric temperature annealing."""
    a = a_init.copy()
    a = np.clip(a, 0, None)
    a = a / a.sum()
    
    b = full_conv(a)
    tau = 0.1 * max(b.max(), 1e-10)
    best_a = a.copy()
    best_sc = 2.0 * n * b.max()
    
    t0 = time.monotonic()
    round_num = 0
    while time.monotonic() - t0 < time_budget and tau > 0.005 * b.max():
        elapsed = time.monotonic() - t0
        remaining = time_budget - elapsed
        if remaining < 5:
            break
        
        # Scale maxiter based on remaining time
        maxiter = min(500, max(100, int(remaining * 50)))
        
        a, _ = slsqp_single(a, tau, maxiter)
        a = np.clip(a, 0, None)
        a = a / a.sum()
        
        b = full_conv(a)
        sc = 2.0 * n * b.max()
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
        
        round_num += 1
        tau *= 0.6
        
        if round_num % 3 == 0:
            print(f"    round {round_num}: tau={tau:.6f}, score={sc:.6f}")
    
    return best_a, best_sc

def init_uniform(n):
    return np.ones(n) / n

def init_beta(n, alpha=1.0, rng=None):
    """Beta-distribution-like: a_k ∝ ((k+0.5)*(n-k-0.5))^alpha"""
    k = np.arange(1, n+1, dtype=np.float64)
    a = (k * (n + 1 - k)) ** alpha
    if rng is not None:
        a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()

def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.01
    return a / a.sum()

def init_dirichlet(n, alpha=0.5, rng=None):
    a = rng.dirichlet(np.full(n, alpha))
    return a

def init_two_peak(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.1 + 0.2 * rng.random())
    c2 = n * (0.7 + 0.2 * rng.random())
    sigma = n * (0.05 + 0.1 * rng.random())
    a = np.exp(-((i-c1)/sigma)**2) + np.exp(-((i-c2)/sigma)**2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()

def init_gaussian_mid(n, rng):
    i = np.arange(n, dtype=np.float64)
    sigma = n * (0.1 + 0.15 * rng.random())
    a = np.exp(-((i - n/2) / sigma)**2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()

def init_cosine(n, freq, rng):
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + 0.5 * np.cos(2*np.pi*freq*i/n))**2
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()

    inits.append(init_cosine(n, 2, rng))
