Low-peak nonnegative step-function search using subgradient optimization
with smoothed max objective in the full simplex.

Strategy:
1. Initialize with anti-palindromic sinusoid (known good starting point).
2. Use subgradient/gradient method with log-sum-exp smoothing of the
   max convolution lag.
3. Project onto the simplex (a >= 0, sum(a) = 1) after each step.
4. Use adaptive step size and multiple active lags.
5. Periodically evaluate and keep the best solution.

import numpy as np
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))

def _project_simplex(v, target_sum=1.0):
    """Project v onto {x >= 0, sum(x) = target_sum}."""
    n = v.size
    v = np.asarray(v, dtype=np.float64).copy()
    # Binary search for tau such that sum(max(v - tau, 0)) = target_sum
    lo = float(np.min(v)) - target_sum
    hi = float(np.max(v))
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    result = np.maximum(v - tau, 0.0)
    # Fix numerical issues
    s = float(result.sum())
    if s > 0:
        result = result * (target_sum / s)
    else:
        result = np.full(n, target_sum / n)
    return result

    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())

    Subgradient optimization with smoothed max objective.
    Minimizes logsumexp of top-K convolution lags.
    """
    start_time = time.monotonic()

    rng = np.random.default_rng(seed)

    best_score = _internal_score(best_a)
    best_max_b = float(np.max(_fft_convolve(best_a)))

    patience = 0
    max_patience = 500
    history = []

        it += 1
        if it % 1000 == 0:
            elapsed = time.monotonic() - start_time
            print(f"  [subgrad] it={it} t={elapsed:.1f}s best_score={best_score:.8f} patience={patience}")
        
        # Compute convolution
        b = _fft_convolve(a)
        if b.size == 0:
            break
        
        # Find top-K lags
        if K < b.size:
            top_idx = np.argpartition(b, -K)[-K:]
        else:
            top_idx = np.arange(b.size)
        top_vals = b[top_idx]
        
        # Check for improvement
        current_max = float(np.max(b))
        if current_max < best_max_b - 1e-12:
            best_max_b = current_max
            best_a = a.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            # Restart with perturbation
            a = best_a + rng.normal(0, 0.01 / n, n)
            a = _project_simplex(a)
            patience = 0
            lr = max(0.001, lr * 0.5)
            continue
        
        # Compute gradient of smoothed objective
        # f = (1/K) * sum_{k in top} log(b[k] / (1/K))
        # df/da[i] = (1/K) * sum_{k in top} (1/b[k]) * 2*a[k-i] (with range check)
        grad = np.zeros(n, dtype=np.float64)
        for k in top_idx:
            k = int(k)
            # ∂b[k]/∂a[i] = 2*a[k-i] if k-i in [0, n-1]
            # i ranges from max(0, k-(n-1)) to min(k, n-1)
            i_min = max(0, k - (n - 1))
            i_max = min(k, n - 1)
            if i_max >= i_min:

                vals = a[k - idx]
