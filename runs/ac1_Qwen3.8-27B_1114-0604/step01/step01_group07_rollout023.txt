its maximum. Active-set gradient refinement then further reduces the peak.
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _internal_score(a):
    arr = np.asarray(a, dtype=np.float64)
    if arr.ndim != 1 or arr.size == 0:
        return np.inf
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    if c.size == 0:
        return np.inf
    mx = float(np.max(c))
    if not np.isfinite(mx) or mx < 0.0:
        mx = 0.0
    return float(2.0 * arr.size * mx / (s * s))


def _make_harmonic(n, amps):
    """Create a multi-harmonic anti-palindromic template normalized to sum=1."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / n
    mod = np.zeros(n, dtype=np.float64)
    for m_idx in range(len(amps)):
        m = m_idx + 1
        mod += amps[m_idx] * np.sin(m * theta)
    a = (1.0 + mod) / n
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.ones(n, dtype=np.float64) / n
    return a / s


def _score_harmonic(n, amps):
    """Score a harmonic template without storing the full sequence twice."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / n
    mod = np.zeros(n, dtype=np.float64)
    for m_idx in range(len(amps)):
        m = m_idx + 1
        mod += amps[m_idx] * np.sin(m * theta)
    a = (1.0 + mod) / n
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    c = _fft_convolve(a)
    mx = float(np.max(c))
    return float(2.0 * n * mx / (s * s))


def _optimize_harmonics(n, n_harmonics, budget_s=30.0):
    """Optimize harmonic amplitudes for a given length using grid + local search."""
    start = time.monotonic()
    best_amps = np.zeros(n_harmonics)
    best_score = np.inf

    # Initial guess: fundamental dominates
    base_amps = [0.75] + [0.1] * (n_harmonics - 1)

    # Coarse grid for first harmonic
    for a1 in np.linspace(0.5, 1.2, 15):
        if time.monotonic() - start > budget_s * 0.4:
            break
        amps = np.array([a1] + [0.1] * (n_harmonics - 1))
        sc = _score_harmonic(n, amps)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_amps = amps.copy()

    # Refine with local grid (2D for 2 harmonics, 3D for 3)
    if n_harmonics == 1:
        for a1 in np.linspace(max(0.3, best_amps[0] - 0.2), best_amps[0] + 0.2, 20):
            if time.monotonic() - start > budget_s:
                break
            sc = _score_harmonic(n, np.array([a1]))
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_amps = np.array([a1])
    elif n_harmonics >= 2:
        # 2D refinement on first two
        for _ in range(3):
            if time.monotonic() - start > budget_s * 0.8:
                break
            a1_range = np.linspace(max(0.3, best_amps[0] - 0.15), best_amps[0] + 0.15, 10)
            a2_range = np.linspace(max(0.0, best_amps[1] - 0.1), best_amps[1] + 0.1, 8)
            for a1 in a1_range:
                for a2 in a2_range:
                    if time.monotonic() - start > budget_s * 0.8:
                        break
                    amps = np.array([a1, a2] + list(best_amps[2:]))
                    sc = _score_harmonic(n, amps)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_amps = amps.copy()
        # Refine third if available
        if n_harmonics >= 3:
            a3_range = np.linspace(max(0.0, best_amps[2] - 0.08), best_amps[2] + 0.08, 8)
            for a3 in a3_range:
                if time.monotonic() - start > budget_s:
                    break
                amps = np.array(list(best_amps[:2]) + [a3])
                sc = _score_harmonic(n, amps)
                if np.isfinite(sc) and sc < best_score:
                    best_score = sc
                    best_amps = amps.copy()

    return best_amps, best_score


def _active_refine(arr, budget_s=300.0, K=12):
    """Active-set mirror descent refinement with backtracking line search."""
    start = time.monotonic()
    arr = np.asarray(arr, dtype=np.float64).copy()
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        return arr, np.inf
    arr = arr / s

    n = arr.size
    best = arr.copy()
    best_score = _internal_score(best)
    if not np.isfinite(best_score):
        return best, best_score

    lr = 1.0
    max_lr = float(n) * 0.05
    min_lr = 0.01
    stagnation = 0
    max_stagnation = 200

    it = 0
    while time.monotonic() - start < budget_s:
        it += 1

        c = _fft_convolve(arr)
        if c.size == 0:
            break
        max_c = float(np.max(c))

        # Active set: lags within 2% of max, capped at K
        threshold = 0.98 * max_c
        active = np.where(c >= threshold)[0]
        if len(active) > K:
            active = np.argpartition(c, -K)[-K:]
        elif len(active) == 0:
            active = np.array([int(np.argmax(c))])

        # Compute subgradient
        grad = np.zeros(n, dtype=np.float64)
        for k in active:
            k = int(k)
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            if j_max >= j_min:
                idx = np.arange(j_min, j_max + 1, dtype=np.int64)
                grad[idx] += 2.0 * arr[k - idx]
        grad /= float(len(active))

        # Project to tangent space of simplex
        g = grad - np.mean(grad)
        g_max = np.max(np.abs(g))
        if g_max < 1e-14:
            break
        g = g / g_max

        # Backtracking line search
        alpha = lr
        improved = False
        for _ls in range(8):
            arr_new = arr * np.exp(-alpha * g)
            arr_new = np.where(np.isfinite(arr_new), arr_new, 0.0)
            arr_new = np.clip(arr_new, 0.0, 1000.0)
            s_new = float(arr_new.sum())
            if s_new < 0.01:
                alpha *= 0.5
                continue
            arr_new = arr_new / s_new
            sc_new = _internal_score(arr_new)
            if np.isfinite(sc_new) and sc_new < best_score * (1.0 - 1e-7):
                arr = arr_new
                best = arr.copy()
                best_score = sc_new
                stagnation = 0
                improved = True
                break
            alpha *= 0.5

        if improved:
            lr = min(max_lr, lr * 1.08)
        else:
            lr = max(min_lr, lr * 0.65)
            stagnation += 1
            if stagnation > max_stagnation:
                break
            if stagnation % 20 == 0:
                arr = best.copy()

    return best, best_score


def _active_refine_v2(arr, budget_s=300.0, K=15):
    """Alternative refinement: uses wider active set and gradient clipping."""
    start = time.monotonic()
    arr = np.asarray(arr, dtype=np.float64).copy()
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if s < 0.01:
        return arr, np.inf
    arr = arr / s

    n = arr.size
    best = arr.copy()
    best_score = _internal_score(best)
    if not np.isfinite(best_score):
        return best, best_score

    lr = 2.0
    max_lr = float(n) * 0.03
    min_lr = 0.005
    stagnation = 0

    it = 0
    while time.monotonic() - start < budget_s:
        it += 1

        c = _fft_convolve(arr)
        if c.size == 0:
            break
        max_c = float(np.max(c))

        # Use top K lags with weighted average (weights = c_k / max_c)
        if c.size > K:
            top_idx = np.argpartition(c, -K)[-K:]
        else:
            top_idx = np.arange(c.size)
        weights = c[top_idx] / max_c
        total_w = float(np.sum(weights))
        if total_w < 1e-10:
            break

        grad = np.zeros(n, dtype=np.float64)
        for w, k in zip(weights, top_idx):
            k = int(k)
            w = float(w)
            j_min = max(0, k - n + 1)
            j_max = min(k, n - 1)
            if j_max >= j_min:
                idx = np.arange(j_min, j_max + 1, dtype=np.int64)
                grad[idx] += 2.0 * w * arr[k - idx]
        grad /= total_w

        g = grad - np.mean(grad)
        g_max = np.max(np.abs(g))
        if g_max < 1e-14:
            break
        g = g / g_max

        # Clip gradient to avoid extreme steps
        g = np.clip(g, -2.0, 2.0)

        alpha = lr
        improved = False
        for _ls in range(10):
            arr_new = arr * np.exp(-alpha * g)
            arr_new = np.where(np.isfinite(arr_new), arr_new, 0.0)
            arr_new = np.clip(arr_new, 0.0, 1000.0)
            s_new = float(arr_new.sum())
            if s_new < 0.01:
                alpha *= 0.5
                continue
            arr_new = arr_new / s_new
            sc_new = _internal_score(arr_new)
            if np.isfinite(sc_new) and sc_new < best_score * (1.0 - 1e-8):
                arr = arr_new
                best = arr.copy()
                best_score = sc_new
                stagnation = 0
                improved = True
                break
            alpha *= 0.5

        if improved:
            lr = min(max_lr, lr * 1.05)
        else:
            lr = max(min_lr, lr * 0.7)
            stagnation += 1
            if stagnation > 150:
                break
            if stagnation % 25 == 0:
                arr = best.copy()

    return best, best_score


def _safe_resolve(x):
    """Sanitize a sequence for evaluation."""
    if x is None:
        return None
    arr = np.asarray(x, dtype=np.float64)
    if arr.ndim != 1:
        arr = arr.ravel()
    if arr.size == 0:
        return None
    arr = np.where(np.isfinite(arr), arr, 0.0)
    arr = np.clip(arr, 0.0, 1000.0)
    s = float(arr.sum())
    if not np.isfinite(s) or s < 0.01:
        return None
    arr = arr / s
    return arr


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a low evaluation score and return the best valid sequence found.
    Combines multi-harmonic template search with active-set gradient refinement.
    """
    t0 = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    hard_deadline = t0 + total_budget - 25.0

    candidates = []  # list of (name, array, internal_score)

    # Phase 0: Check height_sequence_1 if available
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            h1 = _safe_resolve(height_sequence_1)
            if h1 is not None:
                sc = _internal_score(h1)
                candidates.append(("height_seq_1", h1, sc))
                print(f"[init] height_sequence_1: n={h1.size}, score={sc:.8f}")
    except Exception:
        pass

    # Phase 1: Multi-harmonic template search
    # Try various lengths and harmonic counts
    lengths_to_try = [8192, 4096, 16384, 2048]
    n_harmonics_options = [1, 2, 3]

    scan_budget = min(400.0, total_budget * 0.2)
    scan_deadline = t0 + scan_budget

    for n in lengths_to_try:
        if time.monotonic() > scan_deadline:
            break
        for nh in n_harmonics_options:
            if time.monotonic() > scan_deadline:
                break
            sub_budget = max(5.0, (scan_deadline - time.monotonic()) / 3.0)
            amps, sc = _optimize_harmonics(n, nh, sub_budget)
            if np.isfinite(sc):
                a = _make_harmonic(n, amps)
                sc = _internal_score(a)
                candidates.append((f"harmonic_n{n}_h{nh}", a, sc))
                print(f"[scan] n={n}, harmonics={nh}, amps={np.array2string(amps, precision=4)}, score={sc:.8f}")

    # Also try the original single-harmonic with refined amplitude
    for n in [4096, 8192]:
        if time.monotonic() > scan_deadline:
            break
        for amp in [0.70, 0.75, 0.78, 0.80, 0.85, 0.90]:
            a = _make_harmonic(n, np.array([amp]))
            sc = _internal_score(a)
            candidates.append((f"sine_n{n}_a{amp}", a, sc))

    # Sort candidates by score
    candidates.sort(key=lambda x: x[2] if np.isfinite(x[2]) else np.inf)

    if not candidates:
        # Fallback
        best_a = np.ones(2048, dtype=np.float64) / 2048
        return [float(x) for x in best_a]

    # Phase 2: Refine the top candidates
    top_candidates = [c for c in candidates if np.isfinite(c[2])][:3]
    
    refine_time_remaining = max(0.0, hard_deadline - time.monotonic() - 60.0)
    
    if refine_time_remaining > 30.0 and top_candidates:
        # Allocate refinement time proportional to how good each candidate is
        # Give more time to the best candidate
        n_refine = min(len(top_candidates), 2)
        for i in range(n_refine):
            if time.monotonic() > hard_deadline - 60.0:
                break
            name, arr, sc = top_candidates[i]
            # Allocate more time to first (best) candidate
            if i == 0:
                b = min(refine_time_remaining * 0.6, total_budget * 0.5)
            else:
                b = min(refine_time_remaining * 0.3, total_budget * 0.25)
            b = max(10.0, b)
            
            print(f"[refine] Starting refinement of {name} (n={arr.size}), budget={b:.0f}s, current_score={sc:.8f}")
            
            # Use both refinement methods
            half_b = b / 2.0
            ref1, sc1 = _active_refine(arr.copy(), half_b, K=12)
            print(f"[refine] Method 1: score={sc1:.8f}")
            ref2, sc2 = _active_refine_v2(ref1.copy(), half_b, K=15)
            print(f"[refine] Method 2: score={sc2:.8f}")
            
            best_ref = ref1 if sc1 <= sc2 else ref2
            best_ref_sc = min(sc1, sc2)
            
            if np.isfinite(best_ref_sc) and best_ref_sc < sc:
                candidates.append((f"refined_{name}", best_ref, best_ref_sc))
                print(f"[refine] Improved! {sc:.8f} -> {best_ref_sc:.8f}")
            else:
                # Keep original if refinement didn't help
                pass

    # Re-sort all candidates
    candidates.sort(key=lambda x: x[2] if np.isfinite(x[2]) else np.inf)

    # Phase 3: Additional exploration with random perturbations if time allows
    if time.monotonic() < hard_deadline - 120.0 and candidates:
        best_name, best_arr, best_sc = candidates[0]
        # Try perturbing the best solution
        rng = np.random.default_rng(seed)
        n_perturb = 3
        perturb_budget = (hard_deadline - time.monotonic() - 60.0) / n_perturb
        perturb_budget = min(perturb_budget, 60.0)
        
        for p in range(n_perturb):
            if time.monotonic() > hard_deadline - 60.0:
                break
            # Small random perturbation
            perturb = rng.standard_normal(best_arr.size) * 0.01
            perturbed = best_arr + perturb
            perturbed = np.clip(perturbed, 0.0, 1000.0)
            s_p = float(perturbed.sum())
            if s_p < 0.01:
                continue
            perturbed = perturbed / s_p
            sc_p = _internal_score(perturbed)
            if np.isfinite(sc_p) and perturb_budget > 5.0:
                ref_p, sc_rp = _active_refine(perturbed.copy(), perturb_budget, K=12)
                if np.isfinite(sc_rp) and sc_rp < best_sc:
                    candidates.append((f"perturb_{p}", ref_p, sc_rp))
                    best_sc = sc_rp
                    best_arr = ref_p.copy()
                    print(f"[perturb] {p}: improved to {sc_rp:.8f}")

    # Phase 4: Validate and return
    candidates.sort(key=lambda x: x[2] if np.isfinite(x[2]) else np.inf)
    
    best_actual_score = np.inf
    best_out = None
    
    # Try validating top candidates (max 4)
    for name, arr, sc in candidates[:4]:
        if time.monotonic() > hard_deadline:
            break
        resolved = _safe_resolve(arr)
        if resolved is None:
            continue
        try:
            real_score = float(evaluate_sequence(resolved.tolist()))
        except Exception:
            real_score = np.inf
        if np.isfinite(real_score) and real_score < best_actual_score:
            best_actual_score = real_score
            best_out = resolved
            print(f"[validate] {name}: n={resolved.size}, actual={real_score:.8f}")
            if best_actual_score < 1.48:
                break

    if best_out is None:
        # Last resort fallback
        best_out = np.ones(4096, dtype=np.float64) / 4096.0
        print("[fallback] using uniform sequence")

    print(f"[done] Best score={best_actual_score:.8f}, elapsed={time.monotonic()-t0:.1f}s")
    return [float(x) for x in best_out]
