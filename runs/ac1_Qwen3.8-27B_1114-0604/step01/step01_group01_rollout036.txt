Let me also think about the initial β. """
Minimize 2*n*max(convolve(a,a))/sum(a)^2 for non-negative sequences.
Uses L-BFGS-B on a soft-max surrogate with temperature annealing,
plus multiplicative polishing and block refinement.
"""

import numpy as np
import time
from scipy.optimize import minimize

def _pow2_ge(x):
    x = int(max(2, x))
    r = 1
    while r < x:
        r <<= 1
    return r

def _conv(a, b):
    n = len(a) + len(b) - 1
    if n <= 0:
        return np.zeros(0)
    L = _pow2_ge(n)
    return np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)[:n].astype(float, copy=False)

def _xcorr(w, a):
    n = len(a)
    L = _pow2_ge(len(w) + n - 1)
    return np.fft.irfft(np.fft.rfft(w, L) * np.conj(np.fft.rfft(a, L)), L)[:n].astype(float, copy=False)

def _soft_max(b, beta):
    m = float(np.max(b))
    e = np.exp(np.clip(beta * (b - m), -700, 700))
    s = float(e.sum())
    if s < 1e-150:
        w = np.full(len(b), 1.0 / len(b))
        return m + np.log(len(b)) / beta, w
    w = e / s
    return m + np.log(s) / beta, w

def _score_exact(a):
    a = np.maximum(np.asarray(a, dtype=float), 0.0)
    n = len(a)
    T = float(a.sum())
    if T < 1e-12:
        return float('inf')
    b = _conv(a, a)
    return float(2.0 * n * np.max(b) / (T * T))

def _make_obj_grad(beta):
    def closure(a):
        a = np.maximum(np.asarray(a, dtype=float), 0.0)
        n = len(a)
        T = float(a.sum())
        if T < 1e-12:
            return 1e10, np.zeros(n)
        b = _conv(a, a)
        S, w = _soft_max(b, beta)
        corr = _xcorr(w, a)
        score = 2.0 * n * S / (T * T)
        grad = 4.0 * n * (T * corr - S) / (T ** 3)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(score), grad
    return closure

def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=float), 0.0)
    s = float(a.sum())
    if s < 1e-15:
        return np.ones(len(a)) / len(a)
    return a / s

def _lbgfsb_run(a0, beta, time_limit, maxiter=2000):
    a0 = np.maximum(np.asarray(a0, dtype=float), 0.0)
    n = len(a0)
    T = a0.sum()
    a0 = a0 / T  # normalize
    
    obj_jac = _make_obj_grad(beta)
    bounds = [(0.0, None)] * n
    
    start = time.monotonic()
    
    def time_check():
        return (time.monotonic() - start) < time_limit
    
    result = minimize(
        obj_jac,
        a0.astype(float),
        method='L-BFGS-B',
        jac=True,
        bounds=bounds,
        options={
            'maxiter': maxiter,
            'maxfun': maxiter * 3,
            'ftol': 1e-12,
            'gtol': 1e-8,
            'maxls': 40,
        }
    )
    
    best = _normalize(result.x)
    return best
