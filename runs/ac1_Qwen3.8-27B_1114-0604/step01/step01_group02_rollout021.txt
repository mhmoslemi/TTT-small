Minimize 2n*max(conv(a,a))/(sum(a))^2 using exact subgradient descent 
with adaptive parameters, LP-based direction finding, and strategic restarts.
"""

import numpy as np

def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def compute_score(n, a, c):
    S = float(a.sum())
    if S < 1e-10:
        return np.inf
    return float(2.0 * n * np.max(c) / (S * S))

def subgradient_max_conv(a, c, eps_frac=0.02):
    """Compute exact subgradient of max(conv(a,a)) with halo."""
    n = len(a)
    M = float(np.max(c))
    eps = eps_frac * M
    # Active: peaks within eps of max
    active_mask = c >= (M - eps)
    # Halo: peaks within 2*eps of max
    halo_mask = c >= (M - 2.0 * eps)
    
    n_active = int(np.sum(active_mask))
    n_halo = int(np.sum(halo_mask)) - n_active
    halo_weight = 0.1  # weight for halo peaks
    
    grad = np.zeros(n)
    
    # Process active peaks
    active_indices = np.where(active_mask)[0]
    for k in active_indices:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            grad[j_start:j_end+1] += a[j_start:j_end+1][::-1]
    if n_active > 0:
        grad /= n_active
    
    # Process halo peaks
    if n_halo > 0:
        halo_indices = np.where(halo_mask & ~active_mask)[0]
        halo_grad = np.zeros(n)
        for k in halo_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_start <= j_end:
                halo_grad[j_start:j_end+1] += a[j_start:j_end+1][::-1]
        halo_grad /= n_halo
        grad += halo_weight * halo_grad
    
    return grad, M

    """Full gradient of F(a) = 2n*max(c)/(sum a)^2."""
    n = len(a)
    S = float(a.sum())
    sg, M = subgradient_max_conv(a, c, eps_frac)
    # dF/da[j] = 2n/S^2 * (sg[j] - M/S)
    grad = (2.0 * n) / (S * S) * (sg - M / S)
    return grad

    eps = 0.01 * M

    if len(active) > n_active_max:

    if len(active) == 0:

    S = len(active)

    A_ub = np.zeros((S, n + 1))

"""
Minimize 2n*max(conv(a,a))/(sum(a))^2 using exact subgradient descent with
adaptive halo smoothing, periodic LP-based direction finding, and strategic
restarts. Key improvements over prior approach: (1) exact subgradient with
adaptive halo instead of fixed soft-max, (2) LP to find directions that
simultaneously decrease all active peaks, (3) backtracking line search,
(4) structured restarts with spectral perturbations.
"""

import numpy as np
import time
from scipy.optimize import linprog


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, a, c):
    """Compute objective 2n*max(c)/(sum(a))^2."""
    S = float(a.sum())
    if S < 1e-10:
        return np.inf
    return float(2.0 * n * np.max(c) / (S * S))


def subgradient_with_halo(a, c, eps_frac=0.02, halo_frac=0.04, halo_weight=0.15):
    """
    Compute subgradient of max(conv(a,a)) with a halo term for smoothing.
    Returns (gradient, M, n_active).
    """
    n = len(a)
    M = float(np.max(c))
    eps = eps_frac * M
    halo_eps = halo_frac * M

    active_mask = c >= (M - eps)
    halo_mask = (c >= (M - halo_eps)) & (~active_mask)

    n_active = int(np.sum(active_mask))
    n_halo = int(np.sum(halo_mask))

    grad = np.zeros(n)

    # Active peaks: equal weight
    active_indices = np.where(active_mask)[0]
    for k in active_indices:
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            grad[j_start:j_end + 1] += a[j_start:j_end + 1][::-1]
    if n_active > 0:
        grad /= n_active

    # Halo peaks: reduced weight (prevents oscillation at peak switches)
    if n_halo > 0:
        halo_indices = np.where(halo_mask)[0]
        hg = np.zeros(n)
        for k in halo_indices:
            k = int(k)
            j_start = max(0, k - n + 1)
            j_end = min(k, n - 1)
            if j_start <= j_end:
                hg[j_start:j_end + 1] += a[j_start:j_end + 1][::-1]
        hg /= n_halo
        grad += halo_weight * hg

    return grad, M, n_active


def full_objective_gradient(a, c, eps_frac=0.02, halo_frac=0.04, halo_weight=0.15):
    """Gradient of F(a) = 2n*max(c)/(sum(a))^2."""
    n = len(a)
    S = float(a.sum())
    sg, M, _ = subgradient_with_halo(a, c, eps_frac, halo_frac, halo_weight)
    grad = (2.0 * n) / (S * S) * (sg - M / S)
    return grad


def lp_find_direction(a, c, max_constraints=40):
    """
    Solve LP to find direction delta minimizing max first-order increase
    at active peaks, subject to sum(delta)=0 and delta >= -a.
    Returns delta or None if LP fails or t >= 0.
    """
    n = len(a)
    M = float(np.max(c))
    eps = 0.005 * M
    active = np.where(c >= M - eps)[0]
    if len(active) == 0:
        return None
    if len(active) > max_constraints:
        # Keep the most critical ones
        active_vals = c[active]
        order = np.argsort(active_vals)[::-1][:max_constraints]
        active = active[order]

    S = len(active)
    # Build A: A[i, j] = a[active[i] - j] for valid j
    A_mat = np.zeros((S, n))
    for i, k in enumerate(active):
        k = int(k)
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        if j_start <= j_end:
            A_mat[i, j_start:j_end + 1] = a[j_start:j_end + 1][::-1]

    # LP: min t s.t. A_mat @ delta - t <= 0, sum(delta) = 0, -a <= delta <= inf
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    A_ub = np.zeros((S, n + 1))
    A_ub[:, :n] = A_mat
    A_ub[:, n] = -1.0
    b_ub = np.zeros(S)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = [(-a[j], None) for j in range(n)]
    bounds.append((None, None))

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={"presolve": True, "time_limit": 5.0})
        if res.success and res.fun < -1e-8:
            delta = res.x[:n]
            if np.linalg.norm(delta) > 1e-10:
                return delta
    except Exception:
        pass
    return None


def spectral_init(n, rng):
    """Initialize with flat-ish spectrum using iteratively retruncated cosine."""
    a = np.ones(n) / n
    # Apply a few iterations of spectral flattening
    for _ in range(3):
        A = np.fft.rfft(a)
        mag = np.abs(A)
        mag[mag > 0] = 1.0 / np.sqrt(mag[mag > 0])
        A = A * (1.0 / mag)
        A[0] = np.sqrt(n)  # Keep DC
        a_real = np.fft.irfft(A, n)
        # Enforce non-negativity via soft clipping
        a = np.maximum(a_real, 0.01)
        a = a / a.sum()
    return a


def build_initializations(n, parent, rng):
    """Build list of initializations to try."""
    inits = []

    inits.append(("uniform", np.ones(n)))
    inits.append(("spectral", spectral_init(n, rng)))
    inits.append(("random", np.clip(0.5 + 0.5 * rng.random(n), 0.01, 1000)))
    x = np.linspace(0, 1, n)
    inits.append(("ramp", 1.0 + 0.6 * x))
    inits.append(("ramp_inv", 1.0 + 0.6 * (1.0 - x)))
    inits.append(("bell", 1.0 + 0.8 * np.exp(-6.0 * (x - 0.5) ** 2)))
    freq = 1.0
    inits.append(("cos1", np.clip(1.0 + 0.4 * np.cos(2 * np.pi * freq * x), 0.01, 1000)))
    inits.append(("cos2", np.clip(1.0 + 0.3 * np.cos(4 * np.pi * freq * x), 0.01, 1000)))
    # Gapped
    g = np.clip(1.0 + 0.3 * rng.random(n), 0.01, 1000)
    g[::4] = 0.1
    inits.append(("gapped4", g))
    return inits


def adam_step(lr, a, grad, m, v, t, b1=0.9, b2=0.999, eps=1e-8):
    """In-place Adam update. Returns updated a."""
    bc1 = 1.0 - b1 ** t
    bc2 = 1.0 - b2 ** t
    m[:] = b1 * m + (1.0 - b1) * grad
    v[:] = b2 * v + (1.0 - b2) * grad * grad
    m_hat = m / bc1
    v_hat = v / bc2
    a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)
    return a_new
