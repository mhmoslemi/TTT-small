Let me also add a "peak shaving" step after the L-BFGS optimization:

Peak shaving idea:
1. Compute C = conv(a, a).
2. Find the current max M and the set of "active" peaks (those within some fraction of M).
3. For each active peak k*, compute the "sensitivity" of each a[j] to C[k*]: ∂C[k*]/∂a[j] = 2*a[k*-j].
4. Find a direction d (with sum(d) = 0, a + t*d >= 0 for small t > 0) that decreases all active peaks simultaneously.
5. Take a step in that direction.

This is essentially gradient descent on the max, but with exact (not smoothed) peak information.

A simpler version: for the single max peak k*, decrease a[j] for j where a[k*-j] is large, and increase a[m] for m where a[k*-m] is small (doesn't contribute much to the peak).

Let me implement a simple version:
- Find k* = argmax(C).
- Compute "contribution" to C[k*] from each a[j]: contrib[j] = a[j] * a[k*-j].
- The "cost" of decreasing a[j] by δ is the increase in other peaks: for peak k, the change is -δ*a[k-j] + δ*a[k-m] where m is where we increase.
- Choose m to minimize the maximum increase across all peaks.

This is a 2D optimization (j, m) which is O(n^2) per step. For n=2048, that's about 4M operations per step, which is fast.

"""
Minimize f(a) = 2n * max_k conv(a,a)[k] / (sum a)^2 using L-BFGS-B 
with a log-sum-exp smoothing and temperature annealing schedule.

The key insight is that the objective is scale-invariant, so we can focus 
on minimizing the smoothed max of the self-convolution. The gradient 
involves a weighted cross-correlation with the softmax weights, computed 
efficiently via FFT.

Strategy:
1. Multiple starting points (parent, uniform, structured, random).
2. Temperature annealing: start with low alpha (smooth) and increase to 
   high alpha (sharp, close to true max).
3. L-BFGS-B with box constraints (a >= 0) at each temperature.
4. Peak-shaving post-processing.
5. Try multiple sequence lengths.
"""

import numpy as np
from scipy.optimize import minimize
import time


def self_convolve(a):
    """Full linear self-convolution a * a via FFT."""
    n = len(a)
    M = 2 * n - 1
    L = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, L)
    return np.fft.irfft(A * A, L)[:M]


def weighted_corr(W, a):
    """Compute h[j] = sum_k W[k] * a[k-j] for j=0,...,n-1.
    W has length 2n-1, a has length n. Uses FFT."""
    n = len(a)
    L = 1 << (3 * n - 3).bit_length()
    FW = np.fft.fft(W, L)
    FA = np.fft.fft(a, L)
    result = np.fft.ifft(FW * np.conjugate(FA)).real
    return result[:n]


def smooth_max_and_gradient(a, alpha):
    """Compute smoothed max of self-convolution and its gradient w.r.t. a.
    Returns (smooth_max, gradient_of_smooth_max)."""
    n = len(a)
    C = self_convolve(a)
    max_C = float(np.max(C))
    shifted = alpha * (C - max_C)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 0)
    exp_shifted = np.exp(shifted)
    sum_exp = float(np.sum(exp_shifted))
    M = max_C + np.log(sum_exp) / alpha
    weights = exp_shifted / sum_exp
    h = weighted_corr(weights, a)
    grad_M = 2.0 * h  # dM/da[j] = sum_k w[k] * 2*a[k-j] = 2*h[j]
    return M, grad_M


def objective_and_grad(a, alpha):
    """Full objective and gradient: f = 2n * M / S^2."""
    n = len(a)
    S = float(np.sum(a))
    if S < 1e-10:
        return 1e10, np.zeros(n)
    M, grad_M = smooth_max_and_gradient(a, alpha)
    obj = 2.0 * n * M / (S * S)
    grad = (2.0 * n / (S * S)) * grad_M - (2.0 * n * M / (S * S * S))
    return obj, grad


def objective_only(a, alpha):
    obj, _ = objective_and_grad(a, alpha)
    return obj


def gradient_only(a, alpha):
    _, g = objective_and_grad(a, alpha)
    return g


def run_lbfgs(a0, alpha, maxiter=300, gtol=1e-8):
    """Run L-BFGS-B with the given alpha."""
    n = a0.size
    bounds = [(0.0, None)] * n
    
    fun = lambda x: objective_only(x, alpha)
    jac = lambda x: gradient_only(x, alpha)
    
    res = minimize(fun, a0, jac=jac, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'gtol': gtol, 'factr': 1e7})
    return res.x, res.fun


def temperature_schedule(a0, n, start, deadline, rng):
    """Run L-BFGS-B with increasing alpha."""
    a = a0.copy()
    alphas = [1.0, 2.0, 5.0, 10.0, 20.0, 50.0, 100.0, 200.0, 500.0, 1000.0]
    
    # How many stages can we afford?
    time_per_stage = 15.0
    elapsed = time.monotonic() - start
    available = deadline - elapsed
    n_stages = max(3, min(len(alphas), int(available / time_per_stage)))
    
    for i in range(n_stages):
        if time.monotonic() > deadline - 30:
            break
        alpha = alphas[i]
        maxiter = 200 if i < 3 else 400
        a, final_val = run_lbfgs(a, alpha, maxiter=maxiter)
        a = np.clip(a, 0, 1000)
        print(f"  alpha={alpha}: obj={final_val:.6f}, sum={a.sum():.4f}")
    
    return a


def peak_shave(a, n_iterations=100):
    """Peak shaving: iteratively reduce the max peak."""
    n = len(a)
    for _ in range(n_iterations):
        C = self_convolve(a)
        k_star = int(np.argmax(C))
        M = C[k_star]
        
        # Find contributing pairs
        # C[k_star] = sum_j a[j] * a[k_star - j]
        j_start = max(0, k_star - n + 1)
        j_end = min(k_star, n - 1)
        
        if j_start > j_end:
            break
        
        # Compute contribution of each a[j] to this peak
        # dC[k_star]/da[j] = 2*a[k_star - j]
        contributions = a[j_start:j_end + 1] * a[k_star - j_end : k_star - j_start + 1][::-1]
        
        # Find the a[j] to decrease (large contribution, large a[k_star-j])
        # and the a[m] to increase (small effect on this peak)
        
        # Simple strategy: decrease the a[j] with the largest a[j]*a[k_star-j]
        # and increase the a[m] with the smallest a[m]*a[k_star-m]
        idx_max = j_start + int(np.argmax(contributions))
        # For the increase, pick m that minimizes a[m]*a[k_star-m] (least effect on this peak)
        # and also minimizes effect on other peaks
        # Simple heuristic: pick m = (k_star + n // 2) % n (far from the peak)
        
        # Actually, let's just do a small gradient step on the exact max
        # Gradient of C[k_star] w.r.t. a:
        grad = np.zeros(n)
        grad[j_start:j_end + 1] = 2.0 * a[k_star - j_end : k_star - j_start + 1][::-1]
        
        # Project onto sum=0 direction: subtract mean
        grad_zero_sum = grad - grad.mean()
        
        # Find step size by line search
        # a_new = a - t * grad_zero_sum (for active peak)
        # Need a_new >= 0
        t_max = np.inf
        mask = grad_zero_sum > 0  # these will decrease
        if mask.any():
            t_max = np.min(a[mask] / grad_zero_sum[mask])
        t = 0.01 * t_max if np.isfinite(t_max) and t_max > 0 else 0.001
        
        a_new = a - t * grad_zero_sum
        a_new = np.clip(a_new, 0, 1000)
        
        # Check if it improved
        C_new = self_convolve(a_new)
        if np.max(C_new) < M:
            a = a_new
        else:
            # Try smaller step
            t *= 0.5
            a_new = np.clip(a - t * grad_zero_sum, 0, 1000)
            C_new = self_convolve(a_new)
            if np.max(C_new) < M:
                a = a_new
            else:
                break
    
    return a
