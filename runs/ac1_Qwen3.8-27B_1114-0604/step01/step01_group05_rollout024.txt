"""
Minimize 2*n*max(a*a)/(sum(a))^2 for non-negative sequences using:
1. Fourier-harmonic parameterization for shape discovery.
2. Block-constant L-BFGS refinement.
3. Momentum-projected gradient with adaptive smooth-max annealing.

import numpy as np
import time

try:
    from scipy.optimize import minimize
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def fft_convolve(a):
    """Full linear auto-convolution via FFT, clipping tiny negatives."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    c = np.fft.irfft(F * F, size)[:need]
    c = np.ascontiguousarray(c, dtype=np.float64)
    return np.maximum(c, 0.0)


def smooth_max_and_weights(b, beta):
    """Smooth max of array b and softmax weights."""
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float("inf"), np.empty(0, dtype=np.float64)
    m = float(np.max(b))
    if not np.isfinite(m):
        return float("inf"), np.zeros_like(b, dtype=np.float64)
    if m <= 1e-300 or beta <= 0:
        w = np.ones_like(b, dtype=np.float64)
        w /= float(b.size)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -700.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w


def fast_score(f):
    """Compute 2*n*max(f*f)/(sum(f))^2 using FFT."""
    f = np.asarray(f, dtype=np.float64)
    n = f.size
    if n == 0:
        return float("inf")
    S = float(np.sum(f))
    if S < 1e-15 or not np.isfinite(S):
        return float("inf")
    c = fft_convolve(f)
    M = float(np.max(c))
    if not np.isfinite(M):
        return float("inf")
    return float(2.0 * n * M / (S * S))


def corr_pw(f, w):
    """Compute s_i = sum_j f[j] * w[i+j] for i=0..n-1 using FFT."""
    f = np.asarray(f, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = f.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(max(n, 1), dtype=np.float64)
    if n == 1:
        return np.array([float(np.dot(f, w[:1]))], dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(f, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(F * WR, size)[:need]
    s = C[m - n:m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)


def project_simplex(v, target=1.0):
    """Project vector onto {x >= 0, sum(x) = target} via bisection."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    if target <= 0:
        return np.zeros(n, dtype=np.float64)
    v = np.where(np.isfinite(v), v, 0.0)
    v = np.maximum(v, 0.0)
    s = float(np.sum(v))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, target / float(n), dtype=np.float64)
    if abs(s - target) < 1e-15:
        return v * (target / s)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s_mid = float(np.sum(np.maximum(v - mid, 0.0)))
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s_q = float(np.sum(q))
    if s_q <= 0.0 or not np.isfinite(s_q):
        return np.full(n, target / float(n), dtype=np.float64)
    return q * (target / s_q)


def normalize_simplex(p):
    """Clip non-negative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size), dtype=np.float64)
    return p * (1.0 / s)


def fourier_to_p(n, K, params, rng):
    """Convert Fourier parameters to a non-negative normalized sequence."""
    i = np.arange(n, dtype=np.float64)
    log_f = np.zeros(n, dtype=np.float64)
    for k in range(min(K, len(params))):
        log_f += params[k] * np.cos(2.0 * np.pi * k * i / n)
    log_f = log_f - float(np.max(log_f))
    log_f = np.clip(log_f, -50.0, 0.0)
    f = np.exp(log_f)
    f = project_simplex(f, 1.0)
    return f


def fourier_objective(params, n, K, beta):
    """Objective and gradient for Fourier-harmonic parameterization."""
    f = fourier_to_p(n, K, params)
    c = fft_convolve(f)
    G, w = smooth_max_and_weights(c, beta)
    if not np.isfinite(G):
        return float("inf"), np.zeros(K, dtype=np.float64)
    obj = n * G
    svec = corr_pw(f, w)
    svec = np.maximum(svec, 0.0)
    g = 2.0 * svec
    # Gradient w.r.t. params: df/dparams_k = f * (-sin(2*pi*k*i/n)) * (i/n) ... 
    # Actually df/dparams_k = f * cos(2*pi*k*i/n) (since f = exp(sum params_k * cos(...)))
    # But f is also projected to simplex, so the gradient is approximate.
    # Use: d_obj/d_params_k ≈ sum_i g[i] * f[i] * cos(2*pi*k*i/n)
    i = np.arange(n, dtype=np.float64)
    wf = g * f
    grad = np.zeros(K, dtype=np.float64)
    for k in range(K):
        grad[k] = float(np.dot(wf, np.cos(2.0 * np.pi * k * i / n)))
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(obj), grad

def block_to_p(n, B, c):
    """Convert block log-weights to full sequence (normalized)."""
    L = n // B
    z = np.zeros(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        m = min(B - 1, c.size)
        if m > 0:
            z[:m] = c[:m]
    z = z - float(np.max(z))
    wb = np.exp(np.clip(z, -50, 0))
    s = float(np.sum(wb))
    if s <= 0 or not np.isfinite(s):
        wb = np.ones(B, dtype=np.float64) / B
    else:
        wb /= s
    H = wb / L
    return np.repeat(H, L)


def block_objective(c, n, B, beta):

    L = n // B
    z = np.zeros(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        m = min(B - 1, c.size)
        if m > 0:
            z[:m] = c[:m]

    wb = np.exp(np.clip(z, -50, 0))

        wb /= s

    Wavg = float(np.dot(wb, Gsum))
    gradz = (wb / L) * (Gsum - Wavg)
    grad = n * gradz[:-1]
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(n * G), grad.astype(np.float64)


def block_search(n, B, deadline, rng):
    """Block-constant L-BFGS optimization."""
    if not HAVE_SCIPY:
        return None, float("inf")
    
    best_c = np.zeros(B - 1, dtype=np.float64)
    best_sc = float("inf")
    bounds = [(-6.0, 6.0)] * (B - 1)
    
    inits = [np.zeros(B - 1, dtype=np.float64)]
    x = np.linspace(0, 1, B - 1)
    for a in (-3, -1.5, 1.5, 3):
        inits.append(a * (x - 0.5))
    for _ in range(4):
        inits.append(rng.normal(0, 1.0, B - 1) * 0.8)
    
    for beta in (5000.0, 50000.0, 300000.0):
        if time.time() > deadline:
            break
        for c0 in inits:
            if time.time() > deadline:
                break
            try:
                res = minimize(
                    block_objective, c0, args=(n, B, beta),
                    method="L-BFGS-B", jac=True, bounds=bounds,
                    options={"maxiter": 100, "ftol": 1e-11, "gtol": 1e-5, "maxls": 30}
                )
                c_f = np.clip(np.asarray(res.x, dtype=np.float64), -6, 6)
                p_f = block_to_p(n, B, c_f)
                sc = fast_score(p_f)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_c = c_f.copy()
            except Exception:
                pass
        inits = [best_c.copy()]
    
    if np.isfinite(best_sc):
        return block_to_p(n, B, best_c), best_sc
    return None, float("inf")


def momentum_pgd(f0, deadline, beta_init=1000.0, beta_final=1000000.0, momentum=0.9):
    """
    Nesterov-accelerated projected gradient with smooth max and beta annealing.
    """
    f = normalize_simplex(f0)
    n = f.size
    if n == 0:
        return f, float("inf")
    
    best_f = f.copy()
    best_score = fast_score(f)
    
    v = np.zeros(n, dtype=np.float64)
    mu = momentum
    step_scale = 1.0 / n
    
    it = 0
    no_improve = 0
    t_start = time.time()
    
    while time.time() < deadline:
        it += 1
        elapsed = time.time() - t_start
        total_est = max(1.0, elapsed + max(1.0, deadline - time.time()))
        frac = min(1.0, elapsed / total_est)
        # Smooth annealing of beta
        beta = beta_init * (beta_final / beta_init) ** (frac ** 0.7)
        
        # Nesterov lookahead
        f_look = project_simplex(f + mu * v, 1.0)
        
        # Compute gradient at lookahead point
        c = fft_convolve(f_look)
        G, w = smooth_max_and_weights(c, beta)
        if not np.isfinite(G):
            break
        svec = corr_pw(f_look, w)
        svec = np.maximum(svec, 0.0)
        grad = 2.0 * svec
        grad = grad - float(np.mean(grad))  # center for simplex
        
        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-15:
            break
        
        # Adaptive step: normalize gradient
        alpha = step_scale / (gnorm + 1e-10)
        v = mu * v - grad * alpha
        f_new = project_simplex(f + v, 1.0)
        
        s_new = fast_score(f_new)
        if np.isfinite(s_new) and s_new < best_score:
            best_score = s_new
            best_f = f_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        f = f_new
        
        if it % 2000 == 0:
            print(f"    PGD it={it} beta={beta:.0f} cur={s_new:.8f} best={best_score:.8f} noimp={no_improve}")
        
        if no_improve > 5000:
            # Restart momentum
            v = np.zeros(n, dtype=np.float64)
            no_improve = 0
            # Small perturbation
            f = project_simplex(f + 0.01 / n * np.random.default_rng(it).normal(size=n), 1.0)
        
        if it % 5000 == 0:
            # Periodically try a larger perturbation
            f_alt = project_simplex(f + 0.05 / n * np.random.default_rng(it + 999).normal(size=n), 1.0)
            s_alt = fast_score(f_alt)
            if np.isfinite(s_alt) and s_alt < best_score:
                best_score = s_alt
                best_f = f_alt.copy()
                f = f_alt.copy()
                v = np.zeros(n, dtype=np.float64)
    
    return best_f, best_score


def lp_dual_update(f, deadline_iters=200):
    """
    LP-inspired update: solve the dual of the LP for g.
    Dual: min M*sum(y) s.t. sliding_window(y, n)[j] >= 1/f[j], y >= 0.
    Use projected subgradient. Returns the corresponding g.
    """
    n = f.size
    M = float(np.max(fft_convolve(f)))
    if M <= 0 or not np.isfinite(M):
        return f.copy()
    
    inv_f = 1.0 / np.maximum(f, 1e-15)
    
    # Initialize y at center
    y = np.zeros(2 * n - 1, dtype=np.float64)
    center = n - 1
    y[center] = float(np.max(inv_f))
    
    # Projected subgradient iterations
    lr0 = 1.0 / (M * n)
    lr = lr0
    k_idx = np.arange(2 * n - 1, dtype=np.int64)
    
    for it in range(deadline_iters):
        if time.time() > deadline_iters:  # won't trigger, just for safety
            break
        # Compute window sums W_j = sum_{k=j}^{j+n-1} y[k]
        cs = np.concatenate(([0.0], np.cumsum(y)))
        W = cs[n:2*n] - cs[:n]
        
        violation = np.maximum(0.0, inv_f - W)
        if float(np.max(violation)) < 1e-12:
            break
        
        # Subgradient: grad[k] = -sum_{j: k in [j, j+n-1]} violation[j]
        # j ranges from max(0, k-n+1) to min(k, n-1)
        lo = np.maximum(0, k_idx - n + 1)
        hi = np.minimum(k_idx, n - 1)
        cs_v = np.concatenate(([0.0], np.cumsum(violation)))
        grad = -(cs_v[hi + 1] - cs_v[lo])
        
        # Step
        y = y - lr * grad
        y = np.maximum(y, 0.0)
        
        # Simple adaptive LR
        if it % 50 == 0:
            cur_obj = M * float(np.sum(y))
            if it > 0 and np.isfinite(prev_obj):
                if cur_obj > prev_obj * 1.01:
                    lr *= 0.7
                else:
                    lr = min(lr * 1.2, lr0)
            prev_obj = cur_obj
    
    # Recover g from dual: g[j] > 0 where dual constraint is tight
    cs = np.concatenate(([0.0], np.cumsum(y)))
    W = cs[n:2*n] - cs[:n]
    tight = (inv_f - W) < 0.05 * np.maximum(inv_f, 1e-10)
    g = np.zeros(n, dtype=np.float64)
    g[tight] = 1.0
    if g.sum() > 0:
        # Weight by how "tight" each constraint is
        weights = np.maximum(0, 1.0 - (inv_f[tight] - W[tight]) / (0.05 * np.maximum(inv_f[tight], 1e-10)))
        g[tight] = weights
        if g.sum() > 0:
            g /= g.sum()
        else:
            g = f.copy()
    else:
        g = f.copy()
    
    return project_simplex(g, 1.0)


def local_refine(f0, deadline, rng):
    """Local random coordinate perturbation refinement."""
    f = normalize_simplex(f0)
    n = f.size
    best_f = f.copy()
    best_score = fast_score(f)
    
    pert_scale = 0.005
    it = 0
    while time.time() < deadline:
        it += 1
        n_pert = min(200, n // 10)
        idx = rng.integers(0, n, n_pert)
        delta = rng.normal(0, pert_scale / np.sqrt(n), n_pert)
        f_try = f.copy()
        f_try[idx] += delta
        f_try = project_simplex(f_try, 1.0)
        s_try = fast_score(f_try)
        if np.isfinite(s_try) and s_try < best_score:
            best_score = s_try
            best_f = f_try.copy()
            f = f_try.copy()
        elif it % 1000 == 0:
            # Anneal perturbation scale down
            pert_scale *= 0.99
        if it % 5000 == 0:
            print(f"    Local it={it} best={best_score:.8f} scale={pert_scale:.4f}")
    
    return best_f, best_score


def official_score(seq):
    """Call the authoritative evaluator with exception safety."""
    try:
        val = float(evaluate_sequence(seq))
        if np.isfinite(val):
            return val
    except Exception:
        pass
    return float("inf")

    """
    Search for optimal non-negative sequence minimizing 2*n*max(a*a)/(sum(a))^2.
    
    Uses multi-phase approach: Fourier parameterization, block L-BFGS,
    momentum projected gradient with adaptive smooth-max, LP dual updates,
    and local stochastic refinement.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0
    
    if budget_s <= 10:
        return [1.0]
    
    start_time = time.time()
    reserve = max(20.0, budget_s * 0.02)

    N = 8192

    if 'height_sequence_1' in globals():
