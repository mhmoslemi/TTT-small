"""
Minimize 2*n*max(a*a)/(sum(a))^2 for non-negative sequences using
Adam optimization in log-space (softmax parameterization) with
adaptive smooth-max and periodic restarts.
"""

import numpy as np
import time
import math

def fft_convolution(a):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)

def smooth_max_and_weights(b, beta):
    """Smooth max of b and softmax weights."""
    m = float(np.max(b))
    if m <= 0.0:
        w = np.ones_like(b) / float(b.size)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0:
        e = np.zeros_like(b)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return G, w

def corr_pw(p, w):
    """s_i = sum_j p_j * w_{i+j} for i=0..n-1."""
    n = p.size
    m = w.size
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n : m][::-1].copy()
    return np.maximum(s, 0.0)

def softmax(z):
    """Stable softmax."""
    z = z - np.max(z)
    e = np.exp(z)
    s = e.sum()
    return e / s

def compute_grad_z(a, beta):
    """Compute gradient of n*smooth_max(conv(a,a)) w.r.t. z (where a=softmax(z))."""
    n = a.size
    b = fft_convolution(a)
    G, w = smooth_max_and_weights(b, beta)
    s = corr_pw(a, w)
    g = 2.0 * s  # gradient w.r.t. a
    # gradient w.r.t. z: a_j * (g_j - <a,g>)
    ag = float(np.dot(a, g))
    grad_z = a * (g - ag)
    return n * G, grad_z, b, float(np.max(b))

def adam_step(z, grad, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    """One Adam step (in-place)."""
    m = b1 * m + (1 - b1) * grad
    v = b2 * v + (1 - b2) * grad * grad
    m_hat = m / (1 - b1**t)
    v_hat = v / (1 - b2**t)
    z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
    return z, m, v

import time

    a = np.ascontiguousarray(a, dtype=np.float64)

    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1

        size <<= 1

    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_weights(b, beta):
    """Return (smooth_max, softmax_weights) of array b."""
    m = float(np.max(b))
    if not np.isfinite(m) or m <= 0.0:
        return m, np.full_like(b, 1.0 / max(b.size, 1))
    x = np.clip(beta * (b - m), -745.0, 0.0)
    e = np.exp(x)
    Z = float(e.sum())
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return G, w

    """Compute s_i = sum_j p_j * w_{i+j} for i = 0..n-1."""
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(max(n, 0), dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1

    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]

    b = fft_convolution(a)
    M = float(np.max(b))

    """Stable softmax."""
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full_like(z, 1.0 / z.size)
    return e / s

import numpy as np

def fft_convolution(a):
    """Linear convolution a*a via FFT, with non-negativity clipping."""
    a = np.ascontiguousarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_weights(b, beta):
    """Return (smooth_max_value, softmax_weights) for array b."""
    m = float(np.max(b))
    if not np.isfinite(m) or m <= 0.0:
        return m, np.full_like(b, 1.0 / max(b.size, 1), dtype=np.float64)
    x = np.clip(beta * (b - m), -745.0, 0.0)
    e = np.exp(x)
    Z = float(e.sum())
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return G, w


def corr_pw(p, w):
    """Compute s_i = sum_j p_j * w_{i+j} for i = 0..n-1 via FFT."""
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(max(n, 0), dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n:m][::-1].copy()
    return np.maximum(s, 0.0)


def score_of(a):
    """Compute the true objective: 2*n*max(a*a)/(sum(a))^2."""
    n = a.size
    S = float(np.sum(a))
    if S <= 0.0 or not np.isfinite(S):
        return float('inf')
    b = fft_convolution(a)
    M = float(np.max(b))
    if not np.isfinite(M):
        return float('inf')
    return float(2.0 * n * M / (S * S))


def softmax_stable(z):
    """Stable softmax producing a probability vector."""
    z = z - float(np.max(z))
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full_like(z, 1.0 / z.size, dtype=np.float64)
    return e / s

            scale *= 0.9995

    if verbose:
        print(f"  [{label}] Local search done: score={best_score:.8f}, accepts={n_accept}, trials={it+1}")

    return best_z, best_score


def get_initial_sequence(N):
    """Get initial sequence from height_sequence_1 or fall back to uniform."""
    try:
        hsq = height_sequence_1
        hsq = np.array([float(x) for x in hsq], dtype=np.float64)
        hsq = np.maximum(hsq, 0.0)
        hs = float(hsq.sum())
        if hs > 0.0 and np.isfinite(hs) and hsq.size > 0:
            if hsq.size == N:
                return hsq / hs
            else:
                # Interpolate to target length
                x_old = np.linspace(0.0, 1.0, hsq.size)
                x_new = np.linspace(0.0, 1.0, N)
                a = np.interp(x_new, x_old, hsq)
                a = np.maximum(a, 0.0)
                s = float(a.sum())
                if s > 0:
                    return a / s
    except Exception:
        pass
    return np.ones(N, dtype=np.float64) / float(N)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing
    2*n*max(a*a)/(sum(a))^2.

    Uses multi-phase Adam optimization in log-space with adaptive
    smooth-max, periodic perturbations, and a final local search.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 0.0:
        return [1.0]

    N = 8192
    start_time = time.monotonic()
    total_time = budget_s - 45.0  # 45s buffer for final checks
    end_time = start_time + total_time

    rng = np.random.default_rng(seed)

    # --- Get initial sequence ---
    a0 = get_initial_sequence(N)
    z0 = np.log(np.maximum(a0, 1e-30)).astype(np.float64)
    best_score = score_of(a0)
    best_z = z0.copy()

    print(f"N={N}, Initial score: {best_score:.8f}")

    # --- Phase 1: Main optimization (50% of time) ---
    # Moderate beta, moderate lr, with perturbations
    p1_deadline = start_time + 0.50 * total_time
    remaining = p1_deadline - time.monotonic()

    if remaining > 10.0:
        p1_iters = int(remaining * 800)  # ~800 iters/sec estimate
        p1_iters = max(10000, min(p1_iters, 500000))

        z1, sc1 = adam_phase(
            best_z, N, p1_deadline, rng,
            beta_start=300.0, beta_end=30000.0,
            lr_start=0.004, lr_end=0.0004,
            max_iter=p1_iters,
            check_every=100,
            restart_every=25000,
            n_perturb=5,
            perturb_base=0.15,
            label="P1-main"
        )
        if np.isfinite(sc1) and sc1 < best_score:
            best_score = sc1
            best_z = z1.copy()
            print(f"After Phase 1: {best_score:.8f}")

    # --- Phase 2: Mid-refinement (25% of time) ---
    # Higher beta, lower lr
    p2_deadline = start_time + 0.75 * total_time
    remaining = p2_deadline - time.monotonic()

    if remaining > 10.0:
        p2_iters = int(remaining * 800)
        p2_iters = max(5000, min(p2_iters, 300000))

        z2, sc2 = adam_phase(
            best_z, N, p2_deadline, rng,
            beta_start=3000.0, beta_end=150000.0,
            lr_start=0.001, lr_end=0.00005,
            max_iter=p2_iters,
            check_every=100,
            restart_every=30000,
            n_perturb=4,
            perturb_base=0.08,
            label="P2-refine"
        )
        if np.isfinite(sc2) and sc2 < best_score:
            best_score = sc2
            best_z = z2.copy()
            print(f"After Phase 2: {best_score:.8f}")

    # --- Phase 3: Fine refinement (15% of time) ---
    # Very high beta, very low lr, no restarts
    p3_deadline = start_time + 0.90 * total_time
    remaining = p3_deadline - time.monotonic()

    if remaining > 5.0:
        p3_iters = int(remaining * 800)
        p3_iters = max(2000, min(p3_iters, 200000))

        z3, sc3 = adam_phase(
            best_z, N, p3_deadline, rng,
            beta_start=30000.0, beta_end=300000.0,
            lr_start=0.0003, lr_end=0.00001,
            max_iter=p3_iters,
            check_every=50,
            restart_every=0,
            n_perturb=0,
            perturb_base=0.0,
            label="P3-fine"
        )
        if np.isfinite(sc3) and sc3 < best_score:
            best_score = sc3
            best_z = z3.copy()
            print(f"After Phase 3: {best_score:.8f}")

    # --- Phase 4: Local search (10% of time) ---
    remaining = end_time - time.monotonic()
    if remaining > 5.0:
        ls_trials = int(remaining * 300)  # local search is cheaper per trial
        ls_trials = max(1000, min(ls_trials, 50000))

        z4, sc4 = local_search(
            best_z, N, end_time, rng,
            n_trials=ls_trials,
            label="P4-local"
        )
        if np.isfinite(sc4) and sc4 < best_score:
            best_score = sc4
            best_z = z4.copy()
            print(f"After Phase 4: {best_score:.8f}")

    # --- Final output ---
    a_final = softmax_stable(best_z)
    a_final = np.maximum(a_final, 0.0)
    a_final = np.minimum(a_final, 1000.0)
    s = float(a_final.sum())
    if s > 0.0 and np.isfinite(s):
        a_final = a_final / s

    final_seq = [float(x) for x in a_final]

    # Verify with official evaluator
    try:
        off = float(evaluate_sequence(final_seq))
        print(f"Official score: {off:.8f}")
        if np.isfinite(off) and off < best_score:
            best_score = off
    except Exception:
        off = float('inf')

    if not np.isfinite(off):
        # Fallback: uniform sequence (score = 2)
        final_seq = [1.0] * N
        try:
            off2 = float(evaluate_sequence(final_seq))
            print(f"Fallback uniform score: {off2:.8f}")
        except Exception:
            pass

    print(f"Final best score: {best_score:.8f}")
    return final_seq
