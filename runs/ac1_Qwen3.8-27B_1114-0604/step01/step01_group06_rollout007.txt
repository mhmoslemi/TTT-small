```python
import numpy as np
import time
from scipy.optimize import minimize

"""
Algorithm: Multi-phase optimization for minimizing 2*n*max(conv(a,a))/(sum(a))^2.

Phase 1: L-BFGS-B on smooth log-sum-exp approximation in log-softmax space.
Phase 2: Exact subgradient method with analytical step-size bounds.
Phase 3: Local coordinate descent + random perturbation restarts.

Key improvements over previous:
- Analytical step-size bounding using first-order convolution approximation
- Wider dynamic range in softmax (less saturating)
- Structured starts inspired by low-autocorrelation sequence theory
- Adaptive active-set tolerance for subgradient
- Multiple sequence length attempts
"""

_N = 0
_NFFT = 0
_TAU = 1e-5
_BOUND = 40.0


def _init_state(n, tau=1e-5):
    global _N, _NFFT, _TAU
    _N = int(n)
    _NFFT = 1 << (2 * _N).bit_length()
    _TAU = float(tau)


def _fft_conv(a):
    N = _NFFT
    L = 2 * _N - 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L].real
    return np.maximum(c, 0.0)


def _score(a):
    s = float(np.sum(a))
    if s < 1e-12:
        return np.inf
    c = _fft_conv(a)
    m = float(np.max(c))
    return float(2.0 * _N * m / (s * s))


def _proj_simplex(v):
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    cond = u * np.arange(1, n + 1) > (cssv - 1)
    if not np.any(cond):
        return np.ones(n) / n
    rho = np.nonzero(cond)[0][-1]
    theta = (cssv[rho] - 1) / (float(rho + 1))
    return np.maximum(v - theta, 0.0)


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - np.max(z)
    e = np.exp(z)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0:
        return np.ones_like(z) / z.size
    return e / s


def _subgradient(a, tol_frac=1e-8):
    """Return (gradient_vec, max_val, active_set) for max(conv(a,a))."""
    n = _N
    N = _NFFT
    c = _fft_conv(a)
    m = float(np.max(c))
    tol = max(tol_frac * m, 1e-15)
    S = np.where(c >= m - tol)[0]
    p = np.zeros(len(c), dtype=np.float64)
    p[S] = 1.0 / float(len(S))

    # grad[m] = 2 * sum_k p[k] * a[k-m]
    Af = np.fft.fft(a, N)
    Pf = np.fft.fft(p, N)
    g = np.fft.ifft(Pf * np.conj(Af)).real[:n]
    return 2.0 * g, m, S


def _smooth_obj_grad(z):
    """Smooth objective and gradient in log-softmax space."""
    n = _N
    N = _NFFT
    tau = _TAU
    L = 2 * n - 1

    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n:
        return 1e12, np.zeros(max(1, n))
    z = np.clip(z, -_BOUND, _BOUND)

    a = _softmax(z)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L].real
    c = np.maximum(c, 0.0)

    m = float(np.max(c))
    if not np.isfinite(m):
        return 1e12, np.zeros(n)

    diff = (c - m) / tau
    diff = np.clip(diff, -700.0, 0.0)
    e = np.exp(diff)
    se = float(np.sum(e))
    if se <= 0 or not np.isfinite(se):
        return 1e12, np.zeros(n)
    p = e / se

    # Gradient
    Pf = np.fft.fft(p, N)
    Af = np.fft.fft(a, N)
    h = np.fft.ifft(Pf * np.conj(Af)).real[:n]
    s_h = float(np.dot(h, a))
    g = 2.0 * a * (h - s_h)

    val = m + tau * np.log(se)
    val = val * float(n)
    g = g * float(n)
    pen = 1e-12
    val += 0.5 * pen * float(np.dot(z, z))
    g = g + pen * z

    if not np.isfinite(val):
        val = 1e12
        g = np.zeros(n)
    return float(val), g


def _run_lbgfs(z0, maxfun=600, maxiter=200, deadline=None):
    z0 = np.asarray(z0, dtype=np.float64).ravel()
    if z0.size != _N:
        return None
    z0 = np.clip(z0 - np.mean(z0), -_BOUND, _BOUND)
    bounds = [(-_BOUND, _BOUND)] * _N
    opts = {"maxfun": int(maxfun), "maxiter": int(maxiter),
            "ftol": 1e-14, "gtol": 1e-10, "disp": False}
    try:
        res = minimize(_smooth_obj_grad, z0, jac=True, method="L-BFGS-B",
                       bounds=bounds, options=opts)
        return res
    except Exception:
        return None


def _make_starts(rng, n):
    """Generate diverse starting points in log-space."""
    starts = []
    u = np.linspace(0.0, 1.0, n, endpoint=True)
    x = 2.0 * u - 1.0

    def add_start(a):
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size != n or not np.all(np.isfinite(a)) or np.any(a < 0):
            return
        s = float(np.sum(a))
        if s < 1e-10:
            return
        a = a / s
        a = np.maximum(a, 1e-15)
        z = np.log(a)
        z = z - np.mean(z)
        z = np.clip(z, -_BOUND + 1, _BOUND - 1)
        starts.append(z)

    # Try to resample height_sequence_1
    try:
        hs = globals().get("height_sequence_1")
        if hs is not None:
            hs = np.asarray(hs, dtype=np.float64)
            if hs.size > 10:
                hs = np.clip(hs, 1e-15, 1000.0)
                if np.all(np.isfinite(hs)) and np.sum(hs) > 0:
                    z_old = np.log(hs / np.sum(hs))
                    z_old = z_old - np.mean(z_old)
                    x_old = np.linspace(0, 1, hs.size)
                    z_new = np.interp(u, x_old, z_old)
                    starts.append(np.clip(z_new, -_BOUND + 1, _BOUND - 1))
    except Exception:
        pass

    # Rudin-Shapiro inspired (soft version)
    def rudin_shapiro_soft(n, sharpness=3.0):
        # Generate RS-like pattern
        rs = np.zeros(n)
        rs[0] = 1.0
        for i in range(1, n):
            # Count trailing zeros in binary representation of i
            j = i
            cnt = 0
            while j % 2 == 0 and cnt < n.bit_length():
                cnt += 1
                j //= 2
            rs[i] = 1.0 if cnt % 2 == 0 else 0.0
        # Soft version: map {0,1} to [min_val, max_val]
        a = 0.5 + sharpness * (rs - 0.5)
        return np.maximum(a, 0.01)

    for sharp in [0.5, 1.0, 1.5, 2.0, 3.0, 4.0]:
        add_start(rudin_shapiro_soft(n, sharp))

    # Two-level blocks
    half = n // 2
    for ratio in [0.5, 0.7, 1.0, 1.3, 1.5, 2.0, 3.0]:
        f = np.empty(n)
        f[:half] = 1.0
        f[half:] = ratio
        add_start(f)

    # Linear profiles
    for A in [0.3, 0.6, 0.9, 1.2, 1.5]:
        f = 1.0 + A * x
        f = np.maximum(f, 0.01)
        add_start(f)

    # Sinusoidal
    for A in [0.3, 0.5, 0.7, 1.0]:
        for m in [1, 2, 3, 4, 5]:
            f = 1.0 + A * np.cos(2 * np.pi * m * u)
            f = np.maximum(f, 0.01)
            add_start(f)
            f = 1.0 + A * np.sin(2 * np.pi * m * u)
            f = np.maximum(f, 0.01)
            add_start(f)

    # Combined low-frequency
    for _ in range(8):
        f = np.ones(n)
        for m in range(1, 12):
            f += 0.15 * rng.standard_normal() * np.sin(2 * np.pi * m * u + rng.standard_normal())
        f = np.maximum(f, 0.01)
        add_start(f)

    # Random smooth
    for _ in range(4):
        f = np.ones(n)
        for m in range(1, 30):
            f += 0.08 * rng.standard_normal() * np.cos(2 * np.pi * m * u)
        f = np.maximum(f, 0.01)
        add_start(f)

    return starts


def _polish_subgrad(a, deadline, verbose=True):
    """
    Exact subgradient polishing with analytical step bounds.
    Works on the simplex (sum(a)=1, a>=0).
    """
    n = _N
    N = _NFFT
    L = 2 * n - 1
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-12:
        return a
    a /= s

    best_a = a.copy()
    best_score = _score(a)
    rng = np.random.default_rng(12345)
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        c = _fft_conv(a)
        m = float(np.max(c))
        cur_score = 2.0 * n * m

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 500:
            # Perturb to escape
            a = a + 0.01 * rng.standard_normal(n)
            a = _proj_simplex(a)
            no_improve = 0
            continue

        # Compute subgradient with adaptive tolerance
        tol_frac = min(1e-6, 1.0 / (50.0 + it * 0.1))
        grad, m_sub, S = _subgradient(a, tol_frac=tol_frac)

        # Compute direction (project gradient to tangent of simplex)
        # Tangent: sum(d) = 0, and we handle a_i=0 constraints via projection
        mean_g = float(np.mean(grad))
        d = grad - mean_g  # This ensures sum(d) = 0

        # Compute (d conv a) for analytical step bound
        Df = np.fft.rfft(d, N)
        Af = np.fft.rfft(a, N)
        da = np.fft.irfft(Df * Af, N)[:L].real

        # Analytical step bound: need c + 2*eta*da <= m for all k
        # eta_max = (m - c[k]) / (2*da[k]) for da[k] > 0
        positive = da > 1e-15
        if np.any(positive):
            eta_analytic = (m - c[positive]) / (2.0 * da[positive])
            eta_max = float(np.max(np.minimum(eta_analytic, 10.0)))
            if eta_max < 1e-15:
                eta_max = 1e-15
        else:
            eta_max = 0.1

        # Try several step sizes
       _candidates_eta = [eta_max * f for f in [0.5, 0.3, 0.2, 0.1, 0.05, 0.01]]
        _candidates_eta = sorted(set(_candidates_eta), reverse=True)

        accepted = False
        for eta in _candidates_eta[:6]:
            a_new = a - eta * d
            a_new = _proj_simplex(a_new)
            sc_new = float(np.max(_fft_conv(a_new)))
            if sc_new < m - 1e-14:
                a = a_new
                accepted = True
                break

        if not accepted:
            # Try smaller steps or random perturbation
            if no_improve > 100:
                scale = 0.005 * (1 + no_improve / 100.0)
                a = a + scale * rng.standard_normal(n)
                a = _proj_simplex(a)
                if _score(a) >= best_score:
                    a = best_a.copy()
                no_improve = min(no_improve, 200)
            else:
                # Try a very small step in negative gradient direction
                eta_tiny = 1e-5
                a_new = _proj_simplex(a - eta_tiny * d)
                if _score(a_new) < _score(a):
                    a = a_new

        it += 1
        if verbose and it % 200 == 0:
            print(f"  subgrad it={it} score={2.0*n*m:.9f} best={best_score:.9f} "
                  f"no_improve={no_improve} time_left={max(0,deadline-time.monotonic()):.0f}s")

    return best_a


def _polish_logspace(a, deadline):
    """Polish in log-softmax space using L-BFGS with decreasing tau."""
    n = _N
    a = np.asarray(a, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 1e-15)
    a /= a.sum()

    best_a = a.copy()
    best_score = _score(a)

    for tau in [1e-4, 5e-5, 2e-5, 1e-5, 5e-6, 2e-6, 1e-6]:
        if time.monotonic() > deadline:
            break
        _init_state(n, tau=tau)
        z0 = np.log(a) - np.mean(np.log(a))
        z0 = np.clip(z0, -_BOUND, _BOUND)

        # Run L-BFGS
        res = _run_lbgfs(z0, maxfun=300, maxiter=100)
        if res is not None and hasattr(res, "x"):
            z_new = np.asarray(res.x, dtype=np.float64).ravel()
            a_new = _softmax(np.clip(z_new, -_BOUND, _BOUND))
            s = _score(a_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_a = a_new.copy()
                a = a_new.copy()
                print(f"  logspace tau={tau:.0e}: score={s:.9f}")

    return best_a


def _optimize_for_n(n, deadline, seed=42):
    """Full optimization pipeline for a given sequence length."""
    _init_state(n, tau=1e-5)
    rng = np.random.default_rng(seed)
    t0 = time.monotonic()

    print(f"\n=== Optimizing n={n} ===")

    # Generate starts
    starts = _make_starts(rng, n)
    print(f"  Generated {len(starts)} starts")

    # Evaluate starts
    scored_starts = []
    for z in starts[:30]:
        a = _softmax(np.clip(z, -_BOUND, _BOUND))
        s = _score(a)
        if np.isfinite(s):
            scored_starts.append((s, z))
    scored_starts.sort(key=lambda x: x[0])
    print(f"  Best start score: {scored_starts[0][0]:.9f}")

    best_a = _softmax(scored_starts[0][1])
    best_score = scored_starts[0][0]

    # Phase 1: L-BFGS on smooth approximation (time budget: 40% of available)
    t_lbgfs_end = time.monotonic() + 0.35 * (deadline - time.monotonic())
    for idx, (sc, z0) in enumerate(scored_starts[:10]):
        if time.monotonic() > t_lbgfs_end:
            break
        _init_state(n, tau=1e-5)
        res = _run_lbgfs(z0, maxfun=800, maxiter=250)
        if res is not None and hasattr(res, "x"):
            z_new = np.asarray(res.x, dtype=np.float64).ravel()
            a_new = _softmax(np.clip(z_new, -_BOUND, _BOUND))
            s = _score(a_new)
            if np.isfinite(s) and s < best_score:
                best_score = s
                best_a = a_new.copy()
                print(f"  L-BFGS start {idx}: score={s:.9f} "
                      f"elapsed={time.monotonic()-t0:.0f}s")

    # Phase 2: Log-space polish with decreasing tau
    t2 = min(deadline - 300, time.monotonic() + 0.25 * (deadline - time.monotonic()))
    best_a = _polish_logspace(best_a, t2)
    best_score = _score(best_a)
    print(f"  After log-space polish: score={best_score:.9f}")

    # Phase 3: Subgradient polish
    t3 = deadline - 30
    best_a = _polish_subgrad(best_a, t3)
    best_score = _score(best_a)
    print(f"  After subgrad polish: score={best_score:.9f}")

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Multi-phase optimization for minimizing the evaluation function.
    Returns the best sequence found.
    """
    budget_s = max(120.0, float(budget_s))
    deadline = time.monotonic() + budget_s - 15.0
    t0 = time.monotonic()

    # Strategy: try n=8192 first (proven to work), then n=12288 or n=16384 if time
    best_a = None
    best_score = np.inf
    best_n = 8192

    # Primary: n = 8192
    n1 = 8192
    alloc1 = 0.55 * (deadline - time.monotonic())
    try:
        a1, s1 = _optimize_for_n(n1, time.monotonic() + alloc1, seed=seed)
        print(f"\n  n={n1} final score: {s1:.9f}")
        if s1 < best_score:
            best_score = s1
            best_a = a1.copy()
            best_n = n1
    except Exception as e:
        print(f"  Error with n={n1}: {e}")

    # Secondary: try a different length if time remains and primary wasn't great
    time_left = deadline - time.monotonic()
    if time_left > 400 and best_score > 1.51:
        n2 = 12288
        try:
            a2, s2 = _optimize_for_n(n2, deadline - 30, seed=seed + 1)
            print(f"\n  n={n2} final score: {s2:.9f}")
            if s2 < best_score:
                best_score = s2
                best_a = a2.copy()
                best_n = n2
        except Exception as e:
            print(f"  Error with n={n2}: {e}")

    # Fallback: if something went wrong, use a simple construction
    if best_a is None or not np.all(np.isfinite(best_a)):
        n = 8192
        best_a = np.ones(n) / n
        best_score = _score(best_a) if _N == n else 2.0

    # Final cleanup and validation
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = float(np.sum(best_a))
    if s < 0.01 or not np.isfinite(s):
        n = best_a.size
        best_a = np.ones(n) / n
    else:
        best_a = best_a / s

    # Validate with official evaluator
    try:
        official = evaluate_sequence(best_a.tolist())
        print(f"\n  Official evaluator score: {float(official):.9f}")
    except Exception as e:
        print(f"  Validation error: {e}")

    total_time = time.monotonic() - t0
    print(f"\n  Total time: {total_time:.1f}s, Final score: {best_score:.9f}, n={best_a.size}")

    return [float(v) for v in best_a.tolist()]
```
