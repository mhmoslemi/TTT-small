         a_new >= 0

    where c_k[j] = a[k-j] if 0 <= k-j < n, else 0

    # x = [a_new, t]

    c_obj = np.zeros(n + 1)

    # 2 * c_k · a_new - t <= 0 for each k in active_set

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: a_new >= 0, t unbounded below (but will be positive)
    bounds = [(0, None)] * n + [(None, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, 
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n], result.x[n]
    else:

```python
"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based active set
method with line search, coordinate descent polish, and multiple sequence
lengths. The LP at each step linearizes the quadratic autocorrelation
constraints and solves a linear program to find the best descent direction.
"""

import numpy as np
from scipy.optimize import linprog
import time


def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def project_simplex(v):
    """Project onto probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def solve_lp_step(a, active_set):
    """
    Solve the LP:
    min t
    s.t. sum(x) = 1, x >= 0
         2 * c_k · x <= t for k in active_set
    where c_k[j] = a[k-j] if 0 <= k-j < n, else 0
    """
    n = a.size
    
    # Build constraint matrix
    # A_ub @ [x, t] <= b_ub
    # 2*c_k · x - t <= 0
    A_ub = np.zeros((len(active_set), n + 1))
    for i, k in enumerate(active_set):
        # c_k[j] = a[k-j] if 0 <= k-j < n
        for j in range(n):
            kj = k - j
            if 0 <= kj < n:
                A_ub[i, j] = 2.0 * a[kj]
        A_ub[i, n] = -1.0
    
    b_ub = np.zeros(len(active_set))
    
    # Equality: sum(x) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:n].astype(np.float64)
    return None


def line_search(a, a_target, b_current, max_b_current):
    """
    Line search: find alpha in [0,1] that minimizes max(conv(a + alpha*(a_target-a), ...))
    """
    n = a.size
    best_alpha = 0.0
    best_b = max_b_current
    
    # Try several values
    for alpha in [1.0, 0.5, 0.25, 0.1, 0.5, 0.75, 0.9]:
        a_trial = (1 - alpha) * a + alpha * a_target
        a_trial = np.maximum(a_trial, 0)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial /= s
        b_trial = full_conv(a_trial)
        mb = float(np.max(b_trial))
        if mb < best_b:
            best_b = mb
            best_alpha = alpha
    
    return best_alpha, best_b


def lp_optimize(a_init, time_budget, rng, delta_frac=0.05):
    """
    LP-based active set optimizer.
    """
    n = a_init.size
    a = project_simplex(a_init.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    t0 = time.monotonic()
    iter_count = 0
    stall = 0
    
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        max_b = float(np.max(b))
        cur_sc = 2.0 * n * max_b
        
        if cur_sc < best_sc - 1e-12:
            best_sc = cur_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Find active set
        threshold = max_b * (1 - delta_frac)
        active_set = list(np.where(b >= threshold)[0])
        
        if len(active_set) == 0:
            active_set = [int(np.argmax(b))]
        
        # Solve LP
        a_target = solve_lp_step(a, active_set)
        
        if a_target is None:
            # Try with just the argmax
            a_target = solve_lp_step(a, [int(np.argmax(b))])
            if a_target is None:
                break
        
        # Line search
        alpha, new_max_b = line_search(a, a_target, b, max_b)
        
        if alpha > 0:
            a = (1 - alpha) * a + alpha * a_target
            a = np.maximum(a, 0)

def coordinate_descent(a, time_budget, rng):
    """
    Coordinate descent polish: swap mass between pairs of coordinates.
    """
    n = a.size
    a = a.copy()
    a /= a.sum()
    best_a = a.copy()
    best_sc = normed_score(a)
    
    t0 = time.monotonic()
    b = full_conv(a)
    max_b = float(np.max(b))
    
    # Precompute the convolution structure
    # For a swap between i and j (move delta from i to j):
    # a[i] -= delta, a[j] += delta
    # Change in b[k]:
    #   From a[i] change: -delta * a[k-i] (if 0<=k-i<n) and -a[k-i]*delta (if 0<=k-i<n and k-i≠i)
    #   From a[j] change: +delta * a[k-j] (if 0<=k-j<n) and +a[k-j]*delta (if 0<=k-j<n and k-j≠j)
    
    # For efficiency, precompute the "influence" of each coordinate
    # influence[i][k] = change in b[k] per unit change in a[i]
    # = a[k-i] + (a[k-i] if k-i≠i and 0<=k-i<n else 0)
    # = a[k-i] * (1 + (1 if 0<=k-i<n and k-i≠i else 0))
    # Actually: db[k]/da[i] = a[k-i] (if 0<=k-i<n) + a[k-i] (if 0<=k-i<n and k-i≠i)
    # = 2*a[k-i] if 0<=k-i<n and k-i≠i
    # = 2*a[i] if k=2i (since k-i=i)
    # = 0 otherwise
    
    # So db[k]/da[i] = 2*a[k-i] if 0<=k-i<n, else 0
    # (This includes the k=2i case where it's 2*a[i])
    
    # Precompute: for each i, the vector of db[k]/da[i] for all k
    # This is just 2*a reversed and shifted.
    
    # For a swap i->j with amount delta:
    # delta_b[k] = delta * (db[k]/da[j] - db[k]/da[i])
    # = delta * (2*a[k-j] - 2*a[k-i]) (with a[m]=0 if m not in [0,n-1])
    
    # To find optimal delta for a given swap (i,j):
    # We want to minimize max_k (b[k] + delta * (2*a[k-j] - 2*a[k-i]))
    # subject to a[i] - delta >= 0 and a[j] + delta >= 0 (i.e., 0 <= delta <= a[i])
    
    # This is a 1D piecewise-linear minimization problem.
    # The max is the upper envelope of lines, and we want the delta where
    # this envelope is minimized.
    
    # For efficiency, I'll use a ternary search or just try a few values.
    
    iterations = 0
    while time.monotonic() - t0 < time_budget:
        improved = False
        
        for i in range(n):
            if time.monotonic() - t0 > time_budget:
                break
            if a[i] < 1e-10:
                continue
            
            # Try moving mass from i to a few random j's
            candidates = rng.choice(n, size=min(10, n), replace=False)
            for j in candidates:
                if j == i:
                    continue
                
                # delta_b[k] = delta * (2*a[k-j] - 2*a[k-i])
                # Find best delta in [0, a[i]]
                
                # The function to minimize is max_k (b[k] + delta * d[k])
                # where d[k] = 2*(a[k-j] - a[k-i])
                
                # We can use the fact that the optimum delta is where two
                # constraints are tight (or at a boundary).
                
                # Simple approach: try a few values of delta
                a_i = a[i]
                best_delta = 0.0
                best_max = float(np.max(b))
                
                for frac in [0.01, 0.05, 0.1, 0.25, 0.5, 0.75, 0.99]:
                    delta = frac * a_i
                    # Compute new b
                    # d[k] = 2*(a[(k-j) % n] - a[(k-i) % n]) ... no, with wraparound = 0
                    d = np.zeros(2 * n - 1)
                    for k in range(2 * n - 1):
                        dk = 0.0
                        if 0 <= k - j < n:
                            dk += 2.0 * a[k - j]
                        if 0 <= k - i < n:
                            dk -= 2.0 * a[k - i]
                        d[k] = dk
                    
                    b_trial = b + delta * d
                    trial_max = float(np.max(b_trial))
                    
                    if trial_max < best_max:
                        best_max = trial_max
                        best_delta = delta
                
                if best_delta > 0:
                    a[i] -= best_delta
                    a[j] += best_delta
                    # Update b
                    d = np.zeros(2 * n - 1)
                    for k in range(2 * n - 1):
                        dk = 0.0
                        if 0 <= k - j < n:
                            dk += 2.0 * a[k - j]
                        if 0 <= k - i < n:
                            dk -= 2.0 * a[k - i]
                        d[k] = dk
                    b += best_delta * d
                    # Fix numerical errors
                    b = np.abs(b)  # shouldn't be needed but safe
                    improved = True
                    
                    cur_sc = 2.0 * n * float(np.max(b))
                    if cur_sc < best_sc:
                        best_sc = cur_sc
                        best_a = a.copy()
        
        if not improved:
            # Try random perturbations
            a_pert = a + 0.005 * rng.standard_normal(n)
            a_pert = np.maximum(a_pert, 0)
            if a_pert.sum() > 0:
                a_pert /= a_pert.sum()
            sc_pert = normed_score(a_pert)
            if sc_pert < best_sc:
                best_sc = sc_pert
                best_a = a_pert.copy()
                a = a_pert.copy()
                b = full_conv(a)
                improved = True
        
        if not improved:
            break
        
        iterations += 1
    
    return best_a, best_sc

```python
"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using an LP-based active set
method with line search, vectorized coordinate descent polish, and multiple
sequence lengths. The key innovation is solving a linear program at each step
that linearizes the quadratic autocorrelation constraints at the current point,
giving a principled descent direction unlike subgradient methods.
"""

import numpy as np
from scipy.optimize import linprog
import time


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def build_lp(a, active_set):
    """
    Build LP: min t s.t. sum(x)=1, x>=0, 2*c_k.x <= t for k in active_set.
    Returns (c_obj, A_ub, b_ub, A_eq, b_eq, bounds).
    """
    n = a.size
    a_rev = a[::-1]
    
    A_ub = np.zeros((len(active_set), n + 1))
    for i, k in enumerate(active_set):
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        src_start = n - 1 - k + j_start
        src_end = n - 1 - k + j_end + 1
        A_ub[i, j_start:j_end + 1] = 2.0 * a_rev[src_start:src_end]
        A_ub[i, n] = -1.0
    
    b_ub = np.zeros(len(active_set))
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    return c_obj, A_ub, b_ub, A_eq, b_eq, bounds


def solve_lp_step(a, active_set):
    """Solve the LP and return x (the new a) or None if infeasible."""
    c_obj, A_ub, b_ub, A_eq, b_eq, bounds = build_lp(a, active_set)
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs',
                     options={'presolve': True})
    if result.success:
        return result.x[:a.size].astype(np.float64)
    return None


def do_line_search(a, a_target, max_b_current):
    """Line search along direction a to a_target. Returns (best_alpha, best_max_b)."""
    best_alpha = 0.0
    best_mb = max_b_current
    direction = a_target - a
    
    # Get current b for incremental update
    b = full_conv(a)
    
    for alpha in [1.0, 0.7, 0.5, 0.3, 0.1, 0.8, 0.9, 0.4]:
        a_trial = a + alpha * direction
        a_trial = np.maximum(a_trial, 0.0)
        s = a_trial.sum()
        if s < 1e-10:
            continue
        a_trial = a_trial / s
        b_trial = full_conv(a_trial)
        mb = float(np.max(b_trial))
        if mb < best_mb:
            best_mb = mb
            best_alpha = alpha
    
    return best_alpha, best_mb


def lp_optimize(a_init, time_budget, rng, delta_frac=0.05):
    """
    LP-based active set optimizer with line search.
    """
    n = a_init.size
    a = project_simplex(a_init.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    t0 = time.monotonic()
    stall = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        max_b = float(np.max(b))
        cur_sc = 2.0 * n * max_b
        
        if cur_sc < best_sc - 1e-12:
            best_sc = cur_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1
        
        # Determine active set
        threshold = max_b * (1.0 - delta_frac)
        active_mask = b >= threshold
        active_set = list(np.where(active_mask)[0])
        
        if len(active_set) == 0:
            active_set = [int(np.argmax(b))]
        
        # Cap active set size for large n
        if len(active_set) > 60:
            top_idx = np.argsort(b)[::-1][:60]
            active_set = list(top_idx)
        
        # Solve LP
        a_target = solve_lp_step(a, active_set)
        
        accepted = False
        if a_target is not None:
            alpha, new_mb = do_line_search(a, a_target, max_b)
            if alpha > 0:
                a = a + alpha * (a_target - a)
                a = np.maximum(a, 0.0)
                s = a.sum()
                if s > 1e-10:
                    a /= s
                accepted = True
        
        if not accepted:
            # Try with just top-5 peaks
            top5 = list(np.argsort(b)[::-1][:5])
            a_target = solve_lp_step(a, top5)
            if a_target is not None:
                alpha, new_mb = do_line_search(a, a_target, max_b)
                if alpha > 0:
                    a = a + alpha * (a_target - a)
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s > 1e-10:
                        a /= s
                    accepted = True
        
        if not accepted:
            # Fallback: smoothed subgradient step
            # Subgradient of max(b) weighted by softmax
            shift = b / (0.02 * max_b + 1e-15)
            shift -= shift.max()
            w = np.exp(shift)
            w /= w.sum()
            
            # Gradient: g[j] = sum_k w[k] * db[k]/da[j] = sum_k w[k] * 2*a[k-j] (if valid)
            g = np.zeros(n)
            for k in range(2 * n - 1):
                if w[k] < 1e-10:
                    continue
                j_start = max(0, k - (n - 1))
                j_end = min(n - 1, k)
                g[j_start:j_end + 1] += 2.0 * w[k] * a[k - j_end:k - j_start + 1:-1]
            
            a = project_simplex(a - 0.05 * g)
        
        iter_count += 1
        
        # Adaptive reset if stalled
        if stall > 30:
            a = project_simplex(best_a + 0.02 * rng.standard_normal(n))
            stall = 0
            delta_frac = 0.03 + 0.08 * rng.random()
        
        # Periodically reheat
        if iter_count % 200 == 0:
            delta_frac = max(0.02, delta_frac * (0.8 + 0.4 * rng.random()))
    
    return best_a, best_sc


def vectorized_cd(a, time_budget, rng):
    """
    Vectorized coordinate descent: try all pairs (i,j) efficiently.
    For swap i->j with amount delta: b changes by delta * d where d[k] = 2*(a[k-j] - a[k-i]).
    Find delta minimizing max(b + delta*d) subject to 0 <= delta <= a[i].
    """
    n = a.size
    a = a.copy()
    a /= a.sum()
    best_a = a.copy()
    best_sc = normed_score(a)
    b = full_conv(a)
    
    t0 = time.monotonic()
    a_rev = a[::-1]
    
    while time.monotonic() - t0 < time_budget:
        improved = False
        max_b = float(np.max(b))
        
        # Process coordinates in random order
        order = rng.permutation(n)
        
        for i in order:
            if time.monotonic() - t0 > time_budget:
                break
            if a[i] < 1e-8:
                continue
            
            # Try a subset of j values (nearest neighbors + random)
            neighbors = [max(0, i - 1), min(n - 1, i + 1), max(0, i - 2), min(n - 1, i + 2)]
            random_js = rng.choice(n, size=min(8, n), replace=False)
            js = list(set(neighbors + list(random_js)))
            js = [j for j in js if j != i]
            
            for j in js:
                # d[k] = 2*(a[k-j] - a[k-i]) for valid k
                d = np.zeros(2 * n - 1)
                
                # Contribution from a[k-j]: valid for k in [j, j+n-1]
                k_start_j = j
                k_end_j = min(2 * n - 2, j + n - 1)
                d[k_start_j:k_end_j + 1] += 2.0 * a[n - 1 - (k_end_j - j):n - 1 - (k_start_j - j) + 1:-1]
                
                # Recompute properly
                d = np.zeros(2 * n - 1)
                # a[k-j] is valid when 0 <= k-j < n, i.e., j <= k <= j+n-1
                for k in range(j, min(2 * n - 1, j + n)):
                    d[k] += 2.0 * a[k - j]
                # a[k-i] is valid when 0 <= k-i < n, i.e., i <= k <= i+n-1
                for k in range(i, min(2 * n - 1, i + n)):
                    d[k] -= 2.0 * a[k - i]
                
                # Find best delta in [0, a[i]] minimizing max(b + delta * d)
                # The function f(delta) = max_k (b[k] + delta * d[k]) is convex in delta
                # (max of linear functions)
                # Optimum is where the "knee" is, or at a boundary.
                
                a_i = a[i]
                # Try several fractions
                best_delta = 0.0
                best_trial_max = max_b
                
                for frac in [0.05, 0.15, 0.3, 0.5, 0.7, 0.9]:
                    delta = frac * a_i
                    trial_max = float(np.max(b + delta * d))
                    if trial_max < best_trial_max:
                        best_trial_max = trial_max
                        best_delta = delta
                
                if best_delta > 1e-10:
                    # Apply the swap
                    a[i] -= best_delta
                    a[j] += best_delta
                    b += best_delta * d
                    # Clip tiny negatives from numerical error
                    b = np.maximum(b, 0.0)
                    improved = True
                    
                    new_sc = 2.0 * n * float(np.max(b))
                    if new_sc < best_sc:
                        best_sc = new_sc
                        best_a = a.copy()
        
        if not improved:
            # Try random perturbation
            a_pert = a + 0.003 * rng.standard_normal(n)
            a_pert = np.maximum(a_pert, 0.0)
            if a_pert.sum() > 0:
                a_pert /= a_pert.sum()
                sc_pert = normed_score(a_pert)
                if sc_pert < best_sc:
                    best_sc = sc_pert
                    best_a = a_pert.copy()
                    a = a_pert.copy()
                    b = full_conv(a)
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_sc


def cd_fast(a, time_budget, rng):
    """
    Faster coordinate descent using precomputed structure.
    Only tries the most promising swaps.
    """
    n = a.size
    a = a.copy()
    a /= a.sum()
    best_a = a.copy()
    best_sc = normed_score(a)
    b = full_conv(a)
    
    t0 = time.monotonic()
    
    # Precompute influence vectors for each coordinate
    # infl[i][k] = db[k]/da[i] = 2*a[k-i] if 0<=k-i<n else 0
    # This is just 2 * a reversed and shifted
    infl = np.zeros((n, 2 * n - 1))
    for i in range(n):
        k_start = i
        k_end = min(2 * n - 2, i + n - 1)
        infl[i, k_start:k_end + 1] = 2.0 * a[n - 1 - (k_end - i): -1 if k_start == 0 else None:-1]
        # Actually let me just do it simply
    # Redo with simple loop (n is at most 2048, 2n-1 at most 4095)
    for i in range(n):
        for k in range(i, min(2 * n - 1, i + n)):
            infl[i, k] = 2.0 * a[k - i]
    
    while time.monotonic() - t0 < time_budget:
        improved = False
        max_b = float(np.max(b))
        
        # Find "hot" source coordinates (large a[i]) and "hot" target coordinates
        # Moving mass from a large a[i] to a small a[j] tends to help
        
        # Sort by a value
        order_desc = np.argsort(a)[::-1]
        order_asc = np.argsort(a)
        
        # Try swapping from top sources to bottom targets
        tried = 0
        for i_idx in range(min(20, n)):
            if time.monotonic() - t0 > time_budget:
                break
            i = order_desc[i_idx]
            if a[i] < 1e-8:
                continue
            
            for j_idx in range(min(10, n)):
                j = order_asc[j_idx]
                if j == i:
                    continue
                
                # d = infl[j] - infl[i]
                d = infl[j] - infl[i]
                
                # Find best delta
                a_i = a[i]
                best_delta = 0.0
                best_trial_max = max_b
                
                for frac in [0.1, 0.3, 0.5, 0.8, 0.95]:
                    delta = frac * a_i
                    trial_max = float(np.max(b + delta * d))
                    if trial_max < best_trial_max:
                        best_trial_max = trial_max
                        best_delta = delta
                
                if best_delta > 1e-10:
                    a[i] -= best_delta
                    a[j] += best_delta
                    b += best_delta * d
                    # Update infl for i and j
                    # infl[i] and infl[j] change because a changed
                    # Actually, infl depends on a, so we need to update all infl rows
                    # This is expensive. Let's just recompute periodically.
                    b = np.maximum(b, 0.0)
                    improved = True
                    
                    new_sc = 2.0 * n * float(np.max(b))
                    if new_sc < best_sc:
                        best_sc = new_sc
                        best_a = a.copy()
                
                tried += 1
                if tried >= 200:
                    break
            
            if time.monotonic() - t0 > time_budget:
                break
        
        # Periodically recompute infl (every few iterations or if a changed significantly)
        if improved and time.monotonic() - t0 < time_budget:
            # Recompute infl for changed coordinates
            pass  # Skip for now, the approximation is good enough for short runs
        
        if not improved:
            # Perturbation
            a_pert = a + 0.005 * rng.standard_normal(n)
            a_pert = np.maximum(a_pert, 0.0)
            if a_pert.sum() > 0:
                a_pert /= a_pert.sum()
                sc_pert = normed_score(a_pert)
                if sc_pert < best_sc:
                    best_sc = sc_pert
                    best_a = a_pert.copy()
                    a = a_pert.copy()
                    b = full_conv(a)
                    # Recompute infl
                    for ii in range(n):
                        infl[ii] = 0.0
                        for k in range(ii, min(2 * n - 1, ii + n)):
                            infl[ii, k] = 2.0 * a[k - ii]
                    improved = True
        
        if not improved:
            break
    
    return best_a, best_sc


def init_uniform(n):
    return np.ones(n) / n


def init_u_shape(n):
    """High at edges, low in middle."""
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i).astype(np.float64) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_bell(n):
    """Gaussian bell in the middle."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    sigma = n * 0.25
    a = np.exp(-((i - c) / sigma) ** 2)
    a += 0.01
    return a / a.sum()


def init_two_bumps(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.25 + 0.1 * rng.random())
    c2 = n * (0.65 + 0.1 * rng.random())
    sigma = n * (0.1 + 0.05 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.01
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.05
    return a / a.sum()


def init_periodic(n, rng):
    i = np.arange(n, dtype=np.float64)
    freq = 1 + int(rng.integers(0, 4))
    phase = rng.random() * 2 * np.pi
    a = (1.0 + 0.7 * np.cos(2 * np.pi * freq * i / n + phase)) ** 2
    a += 0.01
    return a / a.sum()
