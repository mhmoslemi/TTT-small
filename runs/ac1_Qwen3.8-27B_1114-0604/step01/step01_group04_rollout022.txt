import time

    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

    n = len(a)
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # A[k, j] = a[k-j] for valid k
        # k-j ranges from 0 to n-1, so k ranges from j to j+n-1
        # but k must be in [0, 2n-2]

        k_end = min(j + n, 2*n-1)
        A[k_start:k_end, j] = a[0:k_end-j]
    return A

    A = build_conv_matrix(a)
    # Maximize sum(b) subject to A@b <= M, b >= 0
    # In minimization form: minimize -sum(b)
    c_obj = -np.ones(n)
    A_ub = A
    b_ub = M * np.ones(2*n-1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    return None, None

    idx = np.arange(1, n+1, dtype=np.float64)

    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using LP-based iterative refinement.
At each step, solves an LP to find the best direction that reduces the peak
of the autocorrelation, then takes a blended step in that direction.
"""

import numpy as np
from scipy.optimize import linprog
import time

def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def build_conv_matrix(a):
    """Build matrix A such that A@b gives full convolution a*b."""
    n = len(a)
    A = np.zeros((2*n-1, n))
    for i in range(n):
        k_start = i
        k_end = min(i + n, 2*n - 1)
        A[k_start:k_end, i] = a[0:k_end - i]
    return A

def project_simplex(v):
    """Project onto probability simplex."""
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n+1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def normed_score(a):
    """Score assuming sum(a)=1."""
    return 2.0 * a.size * float(np.max(full_conv(a)))

    """Solve LP: max sum(b) s.t. A@b <= M, b >= 0. Returns (sum_b, b)."""
    n = len(a)
    A = build_conv_matrix(a)
    c_obj = -np.ones(n)
    b_ub = M * np.ones(2*n-1)
    bounds = [(0, None)] * n
    res = linprog(c_obj, A_ub=A, b_ub=b_ub, bounds=bounds, method=method)
    if res.success:
        return float(-res.fun), res.x
    return None, None

```python
def lp_min_peak(a, method='highs'):
    """Solve LP: min max((a*b)_k) s.t. sum(b)=1, b>=0. Returns (t*, b*)."""
    n = len(a)
    A = build_conv_matrix(a)  # (2n-1) x n
    # Variables: b (n) + t (1) = n+1 variables
    # Objective: minimize t → c = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n+1)
    c_obj[-1] = 1.0
    
    # Constraints: (a*b)_k <= t for all k
    # A @ b - t <= 0 → [A, -I] @ [b; t] <= 0
    A_ub = np.hstack([A, -np.eye(2*n-1)])
    b_ub = np.zeros(2*n-1)
    
    # Equality: sum(b) = 1
    A_eq = np.hstack([np.ones(n), np.zeros(1)])
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method=method)
    if res.success:
        t_star = float(res.x[-1])
        b_star = res.x[:n]
        return t_star, b_star
    return None, None
```

This LP has (2n-1) inequality constraints, 1 equality constraint, and n+1 variables. For n=512, that's 1023 + 1 constraints and 513 variables. Still very manageable.

The advantage of this formulation: it directly finds the b that minimizes the mixed convolution peak. If t* < M (current self-conv peak), then replacing a with b (or blending) should reduce the peak.

Let me also think about a completely different angle: what if I use the structure of the problem more directly?

The problem is: min max_k (a*a)_k subject to a >= 0, sum(a) = 1.

At the optimum (assuming no degeneracy), the KKT conditions say:
- There exist multipliers λ_k >= 0 for each k where (a*a)_k = max(a*a)_k
- The gradient of the Lagrangian vanishes: 2 * sum_k λ_k * (∂(a*a)_k / ∂a_i) - μ - ν_i = 0
- Where μ is the multiplier for sum(a)=1 and ν_i >= 0 for a_i >= 0

The gradient ∂(a*a)_k / ∂a_i = a_{k-i} + a_{i} (wait, let me be more careful).

(a*a)_k = sum_j a_j * a_{k-j}
∂(a*a)_k / ∂a_i = a_{k-i} + a_i (if both k-i and i are in range, which they are for the full convolution)

Wait, that's only if i ≠ k-i. If i = k-i (i.e., k = 2i), then ∂(a*a)_k / ∂a_i = 2*a_i.

Actually, let me be more careful:
(a*a)_k = sum_j a_j * a_{k-j}

∂(a*a)_k / ∂a_i = sum_j [δ_{j,i} * a_{k-j} + a_j * δ_{k-j,i}]
= a_{k-i} + a_{k-i} (if k-i ≠ i) or a_i (if k-i = i, i.e., k=2i)

Wait no:
= a_{k-i} (from the first term, when j=i) + a_{k-i} (from the second term, when k-j=i, i.e., j=k-i)

But if i = k-i (k=2i), these are the same term, so it's just a_i (the derivative of a_i * a_i is 2*a_i... wait).

Let me redo this. (a*a)_k = sum_j a_j * a_{k-j}.

∂/∂a_i of this = sum_j [δ(j,i) * a_{k-j} + a_j * δ(k-j, i)]
= a_{k-i} + a_{k-i} (when i ≠ k-i)
= 2*a_{k-i} (when i ≠ k-i)
= a_k + a_k... 

Hmm, let me just do it directly. (a*a)_k = a_0*a_k + a_1*a_{k-1} + ... + a_k*a_0 (for 0 <= k < n; for k >= n it's different).

Actually for the full convolution with 0-padding:
(a*a)_k = sum_{j=max(0,k-n+1)}^{min(k,n-1)} a_j * a_{k-j}

This is a quadratic in a. The gradient with respect to a_i is:
∂(a*a)_k / ∂a_i = a_{k-i} + a_{k-i} = 2*a_{k-i} (when i ≠ k-i, i.e., 2i ≠ k)
= 2*a_i (when 2i = k)

Wait, both cases give 2*a_{k-i}! Because when 2i=k, k-i=i, so 2*a_{k-i} = 2*a_i.

So ∂(a*a)_k / ∂a_i = 2*a_{k-i} (with the convention that a_j = 0 for j outside [0,n-1]).

So the KKT condition (for active constraints) is:
sum_{k: (a*a)_k = M} λ_k * 2*a_{k-i} = μ + ν_i

where ν_i >= 0 and ν_i * a_i = 0 (complementary slackness).

For interior points (a_i > 0), ν_i = 0, so:
sum_{k: (a*a)_k = M} λ_k * 2*a_{k-i} = μ

This means: for all i with a_i > 0, the weighted sum of a_{k-i} over peak positions k is constant.

This is a useful characterization! At the optimum, if I define S = {k : (a*a)_k = M} and find weights λ_k on S, then the "deconvolution" of the peak pattern through λ should be constant on the support of a.

This gives an alternative optimization approach:
1. Start with initial a
2. Identify peak positions S
3. Find λ on S such that sum_{k in S} λ_k * a_{k-i} is as constant as possible (on support of a)
4. Update a to push it toward satisfying this condition

```python
"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding
with iterative blending, multiple initializations, and polished refinement.
"""

import numpy as np
from scipy.optimize import linprog
import time


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def build_conv_matrix(a):
    """Build matrix A (shape (2n-1, n)) such that A @ b = full_conv(a, b)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(i + n, 2 * n - 1)
        A[k_start:k_end, i] = a[0:k_end - i]
    return A


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    return 2.0 * a.size * float(np.max(full_conv(a)))


def lp_min_mixed_peak(a, method='highs'):
    """
    Solve LP: min t s.t. (a*b)_k <= t for all k, sum(b)=1, b>=0.
    Returns (t_opt, b_opt).
    """
    n = len(a)
    A = build_conv_matrix(a)
    
    # Variables: [b_0, ..., b_{n-1}, t] (n+1 variables)
    # Minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality: A @ b - t <= 0 for each of (2n-1) rows
    # [A | -I] @ [b; t] <= 0
    A_ub = np.hstack([A, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)
    
    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    bounds = [(0, None)] * n + [(-np.inf, np.inf)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method=method)
    if res.success:
        return float(res.x[-1]), res.x[:n]
    return None, None


def lp_optimize(a0, time_budget, rng, verbose=False):
    """
    Iteratively optimize using LP-based direction finding.
    a0 should be normalized to sum=1.
    """
    n = a0.size
    a = project_simplex(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    t = 0.5
    stuck_count = 0
    max_stuck = 30
    iters = 0
    
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_budget:
        cur_sc = normed_score(a)
        M = cur_sc / (2.0 * n)  # peak of a*a
        
        t_opt, b_opt = lp_min_mixed_peak(a)
        
        if t_opt is None:
            break
        
        # If t_opt >= M, the LP says we can't do better than current peak
        # in terms of mixed convolution. This might mean we're close to optimal.
        if t_opt >= M * (1 - 1e-8):
            stuck_count += 1
            if stuck_count > max_stuck:
                if verbose:
                    print(f"    Stuck after {iters} iters, score={best_sc:.6f}")
                break
            # Perturb and try again
            a = project_simplex(best_a + 0.02 * rng.standard_normal(n))
            continue
        
        # Blend toward b_opt
        a_new = (1 - t) * a + t * b_opt
        a_new = project_simplex(a_new)
        
        new_sc = normed_score(a_new)
        
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a_new.copy()
            stuck_count = 0
            # Increase step if making good progress
            t = min(0.9, t * 1.1)
            if verbose and iters % 5 == 0:
                elapsed = time.monotonic() - t0
                print(f"    iter {iters}: score={new_sc:.6f}, t_opt={t_opt:.6f}, M={M:.6f}, step={t:.3f}, time={elapsed:.1f}s")
        else:
            # Try smaller step
            t = max(0.05, t * 0.5)
            a_new = (1 - t) * a + t * b_opt
            a_new = project_simplex(a_new)
            new_sc = normed_score(a_new)
            if new_sc < best_sc:
                best_sc = new_sc
                best_a = a_new.copy()
                stuck_count = 0
            else:
                stuck_count += 1
        
        a = a_new
        iters += 1
        
        # Periodic restart
        if iters > 0 and iters % 100 == 0 and stuck_count > 10:
            a = project_simplex(best_a + 0.03 * rng.standard_normal(n))
            t = 0.3
            stuck_count = 5
    
    return best_a, best_sc


def init_u_shape(n):
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i).astype(np.float64) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_two_bump(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.2 + 0.1 * rng.random())
    c2 = n * (0.7 + 0.1 * rng.random())
    sigma = n * (0.05 + 0.1 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.01
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def init_cosine(n, rng):
    i = np.arange(n, dtype=np.float64)
    freq = 1 + int(rng.integers(0, 4))
    a = (1.0 + 0.5 * np.cos(2.0 * np.pi * freq * i / n)) ** 2
    a += 1e-6
    return a / a.sum()


def init_sided(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / c) ** 2 + 0.05
    return a / a.sum()


def init_uniform(n):
    return np.ones(n) / n


def gradient_step(a, lr, tau_frac=0.05):
    """One smoothed-subgradient step."""
    n = a.size
    b = full_conv(a)
    M = float(np.max(b))
    tau = tau_frac * M
    
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    w /= w.sum()
    
    g = 2.0 * np.correlate(w, a, mode='valid')
    a_new = project_simplex(a - lr * g)
    return a_new


def gradient_polish(a, time_budget, rng):
    """Polish with gradient steps."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    lr = 0.3
    t0 = time.monotonic()
    iters = 0
    stuck = 0
    
    while time.monotonic() - t0 < time_budget:
        a = gradient_step(a, lr)
        sc = normed_score(a)
        if sc < best_sc:
            best_sc = sc
            best_a = a.copy()
            stuck = 0
        else:
            stuck += 1
            if stuck > 200:
                lr *= 0.7
                stuck = 0
                if lr < 0.01:
                    break
                a = project_simplex(best_a + 0.01 * rng.standard_normal(n))
        
        iters += 1
    
    return best_a, best_sc


def lp_polish(a, time_budget, rng):
    """Polish with small LP steps."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    t = 0.1
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_budget:
        M = float(np.max(full_conv(a)))
        t_opt, b_opt = lp_min_mixed_peak(a)
        if t_opt is None:
            break
        
        if t_opt >= M * 0.999:
            break
        
        a_new = (1 - t) * a + t * b_opt
        a_new = project_simplex(a_new)
        sc = normed_score(a_new)
        
        if sc < best_sc:
            best_sc = sc
            best_a = a_new.copy()
            t = min(0.3, t * 1.05)
        else:
            t = max(0.02, t * 0.7)
        
        a = a_new
    
    return best_a, best_sc

"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding.
At each iteration, solves an LP to find b minimizing the peak of a*b subject
to sum(b)=1, b>=0. Blends toward b to reduce the self-convolution peak.
Uses multiple initializations, sizes, and a two-phase polish.
"""

import numpy as np
from scipy.optimize import linprog
import time


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def full_conv_ab(a, b):
    """Full linear convolution a*b via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    fb = np.fft.rfft(b, fsize)
    return np.fft.irfft(fa * fb, fsize)[:size]


def build_conv_matrix(a):
    """Build matrix A (shape (2n-1, n)) such that A @ b = full_conv(a, b)."""
    n = len(a)
    A = np.zeros((2 * n - 1, n))
    for i in range(n):
        k_start = i
        k_end = min(i + n, 2 * n - 1)
        A[k_start:k_end, i] = a[0:k_end - i]
    return A


def project_simplex(v):
    """Project vector v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    return 2.0 * a.size * float(np.max(full_conv(a)))


def lp_min_mixed_peak(a):
    """
    Solve LP: min t s.t. (a*b)_k <= t for all k, sum(b)=1, b>=0.
    Returns (t_opt, b_opt) or (None, None) on failure.
    """
    n = len(a)
    A = build_conv_matrix(a)

    # Variables: (b_0, ..., b_{n-1}, t)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    # Inequality: [A | -I_{2n-1}] @ [b; t] <= 0
    A_ub = np.hstack([A, -np.eye(2 * n - 1)])
    b_ub = np.zeros(2 * n - 1)

    # Equality: sum(b) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])

    bounds = [(0, None)] * n + [(-np.inf, np.inf)]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.success:
        return float(res.x[-1]), res.x[:n]
    return None, None


def lp_optimize(a0, time_budget, rng, verbose=False):
    """
    Iteratively optimize using LP-based direction finding.
    a0 should be normalized to sum=1.
    """
    n = a0.size
    a = project_simplex(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    t = 0.5
    stuck_count = 0
    max_stuck = 40
    iters = 0

    t0 = time.monotonic()

    while time.monotonic() - t0 < time_budget:
        b_conv = full_conv(a)
        M = float(np.max(b_conv))
        cur_sc = 2.0 * n * M

        t_opt, b_opt = lp_min_mixed_peak(a)

        if t_opt is None:
            if verbose:
                print(f"    LP failed at iter {iters}")
            break

        # Check if we can improve
        if t_opt >= M * (1.0 - 1e-9):
            stuck_count += 1
            if stuck_count > max_stuck:
                if verbose:
                    print(f"    Converged at iter {iters}, score={best_sc:.6f}")
                break
            # Perturb to escape
            a = project_simplex(best_a + 0.01 * rng.standard_normal(n))
            stuck_count = 0
            continue

        a_new = (1.0 - t) * a + t * b_opt
        a_new = project_simplex(a_new)

        new_sc = normed_score(a_new)

        if new_sc < best_sc - 1e-10:
            best_sc = new_sc
            best_a = a_new.copy()
            stuck_count = 0

            stuck_count += 1

            # Retry with smaller step

            a_new = project_simplex(a_new)
            new_sc = normed_score(a_new)

                best_sc = new_sc

def init_u_shape(n):
    """High at both edges, low in middle."""
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i).astype(np.float64) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_two_bump(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.15 + 0.2 * rng.random())
    c2 = n * (0.65 + 0.2 * rng.random())
    sigma = n * (0.04 + 0.1 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.01
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.05
    return a / a.sum()


def init_cosine(n, rng):
    i = np.arange(n, dtype=np.float64)
    freq = 1 + int(rng.integers(0, 4))
    amp = 0.3 + 0.5 * rng.random()
    a = (1.0 + amp * np.cos(2.0 * np.pi * freq * i / n)) ** 2
    a += 1e-6
    return a / a.sum()


def init_sided(n):
    """Quadratic: high at edges."""
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / max(c, 1.0)) ** 2 + 0.05
    return a / a.sum()


def init_half_cos(n, rng):
    """Half-cosine bump centered."""
    i = np.arange(n, dtype=np.float64)
    center = n * (0.3 + 0.4 * rng.random())
    width = n * (0.2 + 0.3 * rng.random())
    a = np.exp(-((i - center) / width) ** 2) + 0.01
    return a / a.sum()


def subgrad_step(a, lr, tau_frac=0.05):
    """One smoothed-subgradient step for min max(a*a)."""
    n = a.size
    b = full_conv(a)
    M = float(np.max(b))
    tau = tau_frac * M
    if tau < 1e-18:
        tau = 1e-18

    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(b.size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= s

    g = 2.0 * np.correlate(w, a, mode='valid')
    a_new = project_simplex(a - lr * g)
    return a_new


def gradient_polish(a, time_budget, rng):
    """Polish with gradient steps and adaptive learning rate."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    lr = 0.5
    t0 = time.monotonic()
    stuck = 0
    best_lr = lr

    while time.monotonic() - t0 < time_budget:
        a_new = subgrad_step(a, lr)
        sc = normed_score(a_new)
        if sc < best_sc - 1e-12:
            best_sc = sc
            best_a = a_new.copy()
            a = a_new
            stuck = 0
            best_lr = lr
            lr = min(1.0, lr * 1.02)
        else:
            stuck += 1
            if stuck > 100:
                lr = best_lr * 0.6
                stuck = 0
                a = project_simplex(best_a + 0.005 * rng.standard_normal(n))
                if lr < 0.01:
                    break

    return best_a, best_sc

                print(f"[phase4] Grad polish improved to {sc_g:.6f}")
            else:
                print(f"[phase4] Grad polish: {sc_g:.6f} (no improvement)")

    # Phase 5: Final LP polish
    if best_a is not None:
        remaining = deadline - time.monotonic()
        if remaining > 10.0:
            pol_time = min(remaining * 0.6, 120.0)
            a_f, sc_f = lp_polish(best_a, pol_time, rng)
            if sc_f < best_sc:
                best_sc = sc_f
                best_a = a_f.copy()
                print(f"[phase5] Final LP improved to {sc_f:.6f}")

    # Fallback
    if best_a is None:
        best_a = np.ones(128) / 128
        best_sc = 2.0

    # Final verification
    result = [float(max(0.0, x)) for x in best_a]
    final_sc = float(evaluate_sequence(result))
    print(f"\n[final] n={len(result)}, internal={best_sc:.6f}, evaluator={final_sc:.6f}")

    return result
```
