def _set_state(n, tau=1e-6):
    global _G_N, _G_NFFT, _G_TAU, _G_BOUND, _G_BOUNDS
    _G_N = int(n)
    _G_NFFT = 1 << (3 * _G_N - 2).bit_length()
    _G_TAU = float(tau)
    _G_BOUND = 35.0
    _G_BOUNDS = [(-_G_BOUND, _G_BOUND)] * _G_N


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.array([], dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.full_like(z, 1.0 / z.size, dtype=np.float64)
    return e / s


def _fft_conv(a):
    n = len(a)
    N = 1 << (3 * n - 2).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1].real
    return np.maximum(c, 0.0)


def _fast_score(a):
    n = len(a)
    s = float(np.sum(a))
    if not np.isfinite(s) or s <= 1e-12:
        return np.inf
    c = _fft_conv(a)
    m = float(np.max(c))
    if not np.isfinite(m):
        return np.inf
    return float(2.0 * n * m / (s * s))


def _obj_grad(z):
    n = _G_N
    N = _G_NFFT
    tau = _G_TAU
    L = 2 * n - 1

    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n:
        return 1e12, np.zeros(max(1, z.size))

    if not np.all(np.isfinite(z)):
        z = np.nan_to_num(z, nan=0.0, posinf=_G_BOUND, neginf=-_G_BOUND)
    z = np.clip(z, -_G_BOUND, _G_BOUND)

    a = _softmax(z)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L].real
    c = np.maximum(c, 0.0)

    m = float(np.max(c))
    if not np.isfinite(m):
        return 1e12, np.zeros(n)

    diff = (c - m) / tau
    diff = np.clip(diff, -745.0, 0.0)
    e = np.exp(diff)
    se = float(np.sum(e))
    if se <= 0.0 or not np.isfinite(se):
        return 1e12, np.zeros(n)

    p = e / se
    fp = np.fft.rfft(p, N)
    h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real

    s_h = float(np.dot(h, a))
    g = a * (h - s_h)
    grad = 2.0 * g

    val = m + tau * np.log(se)
    scale = float(n) * float(n)
    val = val * scale
    grad = grad * scale

    pen = 1e-10
    val += 0.5 * pen * float(np.dot(z, z))
    grad = grad + pen * z

    if not np.isfinite(val):
        val = 1e12
        grad = np.zeros(n)

    return float(val), grad


def _run_lbfgs(z0, maxfun=500, maxiter=150):
    z0 = np.asarray(z0, dtype=np.float64).ravel()
    if z0.size != _G_N:
        return None
    z0 = np.clip(z0 - np.mean(z0), -_G_BOUND, _G_BOUND)
    opts = {"maxfun": int(maxfun), "maxiter": int(maxiter),
            "ftol": 1e-13, "pgtol": 1e-11, "disp": False}
    try:
        res = minimize(_obj_grad, z0, jac=True, method="L-BFGS-B",
                       bounds=_G_BOUNDS, options=opts)
        return res
    except Exception:
        return None


def _z_from_f(f):
    f = np.asarray(f, dtype=np.float64).ravel()
    f = np.where(np.isfinite(f), f, 1.0)
    f = np.maximum(f, 1e-8)
    z = np.log(f)
    z = z - float(np.mean(z))
    z = np.clip(z, -_G_BOUND + 1.0, _G_BOUND - 1.0)
    return z


def _resample_sequence(f, n_target):
    f = np.asarray(f, dtype=np.float64).ravel()
    if f.size == n_target:
        return f.copy()
    x_old = np.linspace(0, 1, f.size)
    x_new = np.linspace(0, 1, n_target)
    return np.interp(x_new, x_old, f)


def _make_structural_starts(n, rng):
    starts = []
    u = np.linspace(0, 1, n)
    x = 2.0 * u - 1.0

    # Monotone increasing profiles (various shapes)
    for A in [0.3, 0.5, 0.7, 1.0, 1.3, 1.6, 2.0]:
        f = 1.0 + A * u
        z = _z_from_f(np.maximum(f, 0.01))
        if z is not None:
            starts.append(z)

    # Smooth step functions
    for A in [0.5, 0.8, 1.2, 1.6]:
        for w in [0.05, 0.1, 0.15, 0.25, 0.4]:
            sig = 1.0 / (1.0 + np.exp(-(u - 0.5) / max(w, 0.01)))
            f = 1.0 + A * (sig - 0.5)
            f = np.maximum(f, 0.01)
            z = _z_from_f(f)
            if z is not None:
                starts.append(z)

    # Exponential ramp
    for rate in [1.0, 2.0, 3.0, 5.0]:
        f = np.exp(rate * (u - 0.5))
        z = _z_from_f(f)
        if z is not None:
            starts.append(z)

    # Two-level blocks at various split points
    for frac in [0.3, 0.4, 0.5, 0.6, 0.7]:
        for ratio in [1.3, 1.5, 2.0, 2.5, 3.0]:
            f = np.empty(n)
            split = int(frac * n)
            f[:split] = 1.0
            f[split:] = ratio
            z = _z_from_f(f)
            if z is not None:
                starts.append(z)

    # Quadratic profiles
    for A in [0.5, 1.0, 1.5]:
        f = 1.0 + A * (u - 0.5) ** 2 * 4.0
        f = np.maximum(f, 0.01)
        z = _z_from_f(f)
        if z is not None:
            starts.append(z)

    # Random smooth
    for _ in range(8):
        f = np.ones(n)
        for m in range(1, 8):
            c1 = 0.15 * rng.standard_normal()
            c2 = 0.15 * rng.standard_normal()
            f += c1 * np.sin(2 * np.pi * m * u)
            f += c2 * np.cos(2 * np.pi * m * u)
        f = np.maximum(f, 0.01)
        z = _z_from_f(f)
        if z is not None:
            starts.append(z)

    return starts


def _lp_update(a, M, n):
    """Solve LP: maximize sum(b) s.t. (a*b)[k] <= M for all k, b >= 0."""
    # Build constraint matrix C where C[k,i] = a[k-i] if 0<=k-i<n else 0
    # C is (2n-1) x n, but we can represent it sparsely
    # For efficiency, we'll use the Toeplitz-like structure
    
    # Actually for large n, building the full sparse matrix might be expensive.
    # Let's only do this for n <= 4096.
    if n > 4096:
        return None
    
    rows = []
    cols = []
    vals = []
    
    for i in range(n):
        # For column i, C[k,i] = a[k-i] for k-i in [0, n-1], i.e., k in [i, i+n-1]
        k_start = i
        k_end = min(i + n - 1, 2 * n - 2)
        ks = range(k_start, k_end + 1)
        ais = a[0:k_end - k_start + 1]
        rows.extend(ks)
        cols.extend([i] * len(ks))
        vals.extend(ais.tolist())
    
    C = csr_matrix((vals, (rows, cols)), shape=(2 * n - 1, n))
    
    c_obj = -np.ones(n)  # minimize -sum(b)
    A_ub = C
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                      method='highs', options={'presolve': True,
                                                'time_limit': 30})
        if res.success:
            b = res.x
            b = np.maximum(b, 0)
            s = float(np.sum(b))
            if s > 1e-12:
                b = b / s  # normalize to sum 1
                return b
    except Exception:
        pass
    return None


def _polish_exact(a0, deadline, n):
    """Polish using exact subgradient at the peak position."""
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s <= 1e-12:
        return a0
    a = a / s
    
    b = np.log(np.maximum(a, 1e-30))
    b = np.clip(b - np.mean(b), -_G_BOUND, _G_BOUND)
    
    best = a.copy()
    best_score = _fast_score(a)
    
    rng = np.random.default_rng(42)
    no_improve = 0
    it = 0
    
    while time.monotonic() < deadline and no_improve < 300:
        c = _fft_conv(a)
        k_star = int(np.argmax(c))
        F = float(c[k_star])
        
        if F < best_score / (2.0 * n) - 1e-18:
            best = a.copy()
            best_score = 2.0 * n * F
        
        # Exact subgradient: d(a*a)[k*]/da[i] = a[k*-i] + a[i-k*] (with bounds)
        grad = np.zeros(n)
        for i in range(n):
            j1 = k_star - i
            j2 = i - k_star
            val = 0.0
            if 0 <= j1 < n:
                val += a[j1]
            if 0 <= j2 < n:
                val += a[j2]
            grad[i] = val
        
        # Gradient in log-space
        s_grad = float(np.dot(grad, a))
        g_log = a * (grad - s_grad)
        
        # Line search
        accepted = False
        for eta in [1e-5, 3e-5, 1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 0.1, 0.3, 1.0]:
            b_t = np.clip(b - eta * g_log * n, -_G_BOUND, _G_BOUND)
            a_t = _softmax(b_t)
            F_t = float(np.max(_fft_conv(a_t)))
            if F_t < F * (1 - 1e-12):
                b = b_t
                a = a_t
                accepted = True
                break
        
        if not accepted:
            # Random perturbation
            for scale in [0.01, 0.05, 0.1, 0.3, 0.8]:
                b_t = np.clip(b + scale * rng.standard_normal(n), -_G_BOUND, _G_BOUND)
                a_t = _softmax(b_t)
                F_t = float(np.max(_fft_conv(a_t)))
                if F_t < best_score / (2.0 * n) - 1e-12:
                    b = b_t
                    a = a_t
                    no_improve = 0
                    break
            else:
                no_improve += 1
        else:
            no_improve = 0
        
        it += 1
        if it % 100 == 0:
            cur = 2.0 * n * F
            print(f"  polish it={it} cur={cur:.9f} best={best_score:.9f} "
                  f"remaining={max(0,deadline-time.monotonic()):.0f}s")
    
    return best


def _adaptive_lbfgs(z0, deadline, n):
    """Run L-BFGS with decreasing tau."""
    best_a = _softmax(z0)
    best_sc = _fast_score(best_a)
    
    taus = [1e-2, 5e-3, 2e-3, 1e-3, 5e-4, 2e-4, 1e-4, 5e-5, 2e-5, 1e-5, 5e-6]
    
    for tau in taus:
        if time.monotonic() > deadline - 30:
            break
        _set_state(n, tau=tau)
        res = _run_lbfgs(z0, maxfun=300, maxiter=100)
        if res is not None and hasattr(res, 'x'):
            z_new = np.asarray(res.x, dtype=np.float64).ravel()
            if z_new.size == n:
                a = _softmax(np.clip(z_new, -_G_BOUND, _G_BOUND))
                sc = _fast_score(a)
                if np.isfinite(sc) and sc < best_sc:
                    best_sc = sc
                    best_a = a.copy()
                    z0 = z_new.copy()
                    print(f"    tau={tau:.0e}: score={sc:.9f}")
    
    return best_a, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative coefficient sequence minimizing 2*n*max(a*a)/(sum a)^2.
    Uses L-BFGS in log-space with soft-max, LP-based updates, and exact polishing.
    """
    budget_s = max(60.0, float(budget_s))
    t_start = time.monotonic()
    deadline = t_start + budget_s - 10.0
    
    rng = np.random.default_rng(int(seed))
    
    # Try multiple dimensions
    best_a = None
    best_score = np.inf
    
    # Load known good sequence if available
    known_seq = None
    try:
        g = globals().get("height_sequence_1")
        if g is not None:
            g = np.asarray(g, dtype=np.float64).ravel()
            if g.size > 10 and np.all(np.isfinite(g)) and np.sum(g) > 0:
                known_seq = np.maximum(g, 0)
                print(f"Loaded known sequence of length {g.size}, "
                      f"score={_fast_score(known_seq):.9f}")
    except Exception:
        pass
    
    # Dimensions to try (larger first since it gives more freedom)
    # But be careful: larger n means slower LP
    n_list = [8192, 4096]
    
    for n in n_list:
        if time.monotonic() > deadline - 120:
            break
        
        _set_state(n, tau=1e-4)
        print(f"\n=== Optimizing with n={n} ===")
        
        # Build starting points
        starts = []
        
        # Primary start: known good sequence resampled
        if known_seq is not None:
            f_resampled = _resample_sequence(known_seq, n)
            z_known = _z_from_f(f_resampled)
            if z_known is not None:
                starts.append(('known', z_known))
        
        # Structural starts
        struct_starts = _make_structural_starts(n, rng)
        for i, z in enumerate(struct_starts):
            starts.append((f"struct{i}", z))
        
        # Score and sort starts
        scored_starts = []
        for name, z in starts:
            a = _softmax(z)
            sc = _fast_score(a)
            if np.isfinite(sc):
                scored_starts.append((sc, name, z))
        
        scored_starts.sort(key=lambda x: x[0])
        print(f"  {len(scored_starts)} starts, best initial score="
              f"{scored_starts[0][0]:.6f}")
        
        # Phase 1: L-BFGS from top starts
        phase1_deadline = min(deadline - 400, t_start + budget_s * 0.35)
        for idx, (sc, name, z) in enumerate(scored_starts[:8]):
            if time.monotonic() > phase1_deadline:
                break
            
            if idx < 3:
                dl = min(phase1_deadline, time.monotonic() + 300)
            elif idx < 6:
                dl = min(phase1_deadline, time.monotonic() + 150)
            else:
                dl = min(phase1_deadline, time.monotonic() + 80)
            
            if time.monotonic() > dl - 10:
                continue
            
            try:
                a_opt, sc_opt = _adaptive_lbfgs(z, dl, n)
                if sc_opt < best_score:
                    best_score = sc_opt
                    best_a = a_opt
                    print(f"  L-BFGS start '{name}': score={sc_opt:.9f} "
                          f"(global best)")
                print(f"  start '{name}' finished: score={sc_opt:.9f} "
                      f"elapsed={time.monotonic()-t_start:.0f}s")
            except Exception as e:
                print(f"  start '{name}' failed: {e}")
        
        # Phase 2: LP-based updates from best
        if best_a is not None and n <= 4096:
            phase2_deadline = min(deadline - 200, time.monotonic() + 300)
            a_l = _resample_sequence(best_a, n) if len(best_a) != n else best_a.copy()
            a_l = np.maximum(a_l, 0)
            s_l = float(np.sum(a_l))
            if s_l > 0:
                a_l = a_l / s_l
            
            print(f"\n  LP phase (n={n}):")
            for lp_it in range(30):
                if time.monotonic() > phase2_deadline:
                    break
                
                c = _fft_conv(a_l)
                M = float(np.max(c))
                b = _lp_update(a_l, M, n)
                if b is None:
                    break
                
                # Try mixing
                improved = False
                for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                    a_mix = (1 - t) * a_l + t * b
                    a_mix = np.maximum(a_mix, 0)
                    s_mix = float(np.sum(a_mix))
                    if s_mix < 0.01:
                        continue
                    a_mix = a_mix / s_mix
                    sc_mix = _fast_score(a_mix)
                    if sc_mix < best_score - 1e-12:
                        a_l = a_mix
                        best_score = sc_mix
                        best_a = a_mix.copy()
                        improved = True
                        print(f"    LP it={lp_it} t={t}: score={sc_mix:.9f}")
                        break
                
                if not improved:
                    # Try smaller steps
                    for t in [0.01, 0.03, 0.05]:
                        a_mix = (1 - t) * a_l + t * b
                        a_mix = np.maximum(a_mix, 0)
                        s_mix = float(np.sum(a_mix))
                        if s_mix < 0.01:
                            continue
                        a_mix = a_mix / s_mix
                        sc_mix = _fast_score(a_mix)
                        if sc_mix < best_score - 1e-12:
                            a_l = a_mix
                            best_score = sc_mix
                            best_a = a_mix.copy()
                            improved = True
                            break
                    if not improved:
                        # Accept a_l as is (change direction for next LP)
                        a_l = np.maximum((1 - 0.5) * a_l + 0.5 * b, 0)
                        s_l = float(np.sum(a_l))
                        if s_l > 0:
                            a_l = a_l / s_l
        
        # Phase 3: Exact polishing
        if best_a is not None:
            phase3_deadline = min(deadline - 5, time.monotonic() + 600)
            print(f"\n  Polishing phase (budget={phase3_deadline - time.monotonic():.0f}s):")
            a_pol = _resample_sequence(best_a, n) if len(best_a) != n else best_a
            a_pol = _polish_exact(a_pol, phase3_deadline, n)
            sc_pol = _fast_score(a_pol)
            if sc_pol < best_score:
                best_score = sc_pol
                best_a = a_pol
                print(f"  Polish improved: score={sc_pol:.9f}")
            else:
                print(f"  Polish: score={sc_pol:.9f} (no improvement)")
        
        # If time remains and n was 8192, don't bother with 4096
        # (the 8192 result should be at least as good)
        if n == 8192:
            # Only try 4096 if we have lots of time left
            if time.monotonic() > deadline - 300:
                n_list = n_list[:1]  # skip 4096
    
    # Final validation
    if best_a is None:
        # Fallback: uniform
        n = 4096
        best_a = np.ones(n)
        best_score = 2.0
        print(f"Fallback to uniform, score={best_score}")
    
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = float(np.sum(best_a))
    if s < 0.01 or not np.isfinite(s):
        best_a = np.ones(4096) / 4096
    else:
        best_a = best_a / s
    
    # Validate with official evaluator
    try:
        exact = evaluate_sequence(best_a.tolist())
        if np.isfinite(exact):
            print(f"\nOfficial evaluator score: {exact:.9f}")
        else:
            print(f"\nOfficial evaluator returned non-finite: {exact}")
            # Fallback
            best_a = np.ones(4096) / 4096
            best_score = 2.0
    except Exception as e:
        print(f"Evaluator error: {e}")
    
    print(f"\nFinal: n={len(best_a)}, score~{best_score:.9f}, "
          f"total time={time.monotonic()-t_start:.1f}s")
    
    return [float(v) for v in best_a.tolist()]
