import time

# Global state
_N = 0
_NFFT = 0

def _set_state(n):
    global _N, _NFFT
    _N = n
    _NFFT = 1 << (3 * n - 1).bit_length()

def _fft_conv(a):
    """Compute full linear convolution of a with itself using FFT."""
    n = _N
    N = _NFFT
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1].real
    return np.maximum(c, 0.0)

def _score(a):
    """Compute the score for a normalized sequence (sum=1)."""
    n = _N
    s = float(np.sum(a))
    if s <= 0:
        return np.inf
    c = _fft_conv(a)
    return float(2.0 * n * np.max(c) / (s*s))

def _project_simplex(v):
    """Project v onto the probability simplex {x >= 0, sum(x) = 1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _grad_smoothed(a, tau):
    """Compute gradient of F_tau(a) = tau * logsumexp(c/tau) where c = conv(a,a)."""
    n = _N
    N = _NFFT
    c = _fft_conv(a)

    max_c = np.max(c)
    exp_c = np.exp((c - max_c) / tau)
    Z = np.sum(exp_c)
    p = exp_c / Z
    
    # Compute gradient: grad[i] = 2 * sum_j a[j] * p[i+j]
    # Using FFT: grad = 2 * convolve(flip(a), p)[n-1 : 2n-1]
    a_flip = a[::-1]
    N2 = 1 << (3 * n - 1).bit_length()
    fa_flip = np.fft.rfft(a_flip, N2)
    fp = np.fft.rfft(p, N2)
    conv = np.fft.irfft(fa_flip * fp, N2)
    grad = 2.0 * conv[n-1:2*n-1]
    
    return grad, float(max_c + tau * np.log(Z) - max_c)  # F_tau value (up to constant)

def _optimize_sgd(a0, n, deadline, seed=0):

    tau_end = 0.001
    total_iters = 2000

            break
        
        # Decay tau
        progress = it / total_iters
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Compute gradient
        grad, F_val = _grad_smoothed(a, tau)
        
        # Adaptive learning rate
        # Scale gradient to have norm ~ 1
        gnorm = np.linalg.norm(grad)
        if gnorm > 0:
            grad = grad / gnorm
        
        # Multiple step sizes
        best_local = a.copy()
        best_local_score = _score(a)
        for eta in (1e-5, 3e-5, 1e-4, 3e-4, 1e-3, 3e-3, 0.01, 0.03):

import numpy as np
import time

"""
Algorithm summary
-----------------
We minimize score = 2*n*max(convolve(a,a)) / (sum(a))^2 over non-negative sequences.
Since the score is scale-invariant, we normalize sum(a)=1 and minimize 2*n*max(conv(a,a)).

Key improvements over previous approach:
1. Direct simplex-constrained optimization (no softmax parameterization issues).
2. Adaptive multi-scale tau schedule for the log-sum-exp smoothing.
3. Coordinate perturbation polish for escaping local minima.
4. Multiple diverse starting points including number-theoretic constructions.
5. Coarse-to-fine optimization for better global exploration.
"""

_N = 0
_NFFT = 0
_NFFT_GRAD = 0


def _set_state(n):
    global _N, _NFFT, _NFFT_GRAD
    _N = int(n)
    _NFFT = 1 << (2 * _N).bit_length()
    _NFFT_GRAD = 1 << (3 * _N - 1).bit_length()


def _fft_conv(a):
    n = _N
    N = _NFFT
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1].real
    return np.maximum(c, 0.0)


def _score(a):
    n = _N
    s = float(np.sum(a))
    if s <= 1e-15:
        return np.inf
    c = _fft_conv(a)
    return float(2.0 * n * float(np.max(c)) / (s * s))


def _score_normalized(a):
    """Score assuming sum(a)=1."""
    n = _N
    c = _fft_conv(a)
    return float(2.0 * n * float(np.max(c)))


def _project_simplex(v):
    """Project onto {x >= 0, sum(x) = 1}."""
    n = len(v)
    v = np.asarray(v, dtype=np.float64).copy()
    if np.all(np.isnan(v)):
        return np.ones(n) / n
    v = np.nan_to_num(v, nan=0.0, posinf=1.0, neginf=-1.0)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def _grad_smoothed(a, tau):
    """
    Compute gradient of F_tau(a) = tau * log(sum_k exp(c[k]/tau))
    where c = convolve(a, a). Returns (grad, F_value).
    grad[i] = 2 * sum_j a[j] * p[i+j]
    """
    n = _N
    c = _fft_conv(a)

    max_c = float(np.max(c))
    shifted = (c - max_c) / tau
    shifted = np.clip(shifted, -700.0, 0.0)
    exp_c = np.exp(shifted)
    Z = float(np.sum(exp_c))
    if Z <= 0 or not np.isfinite(Z):
        return np.zeros(n), max_c
    p = exp_c / Z

    # grad = 2 * convolve(a[::-1], p)[n-1 : 2n-1]
    N = _NFFT_GRAD
    a_rev = a[::-1]
    fa_rev = np.fft.rfft(a_rev, N)
    fp = np.fft.rfft(p, N)
    conv = np.fft.irfft(fa_rev * fp, N)
    grad = 2.0 * conv[n - 1:2 * n - 1]

    F_val = max_c + tau * float(np.log(Z))
    return grad, F_val


def _normalize(a):
    """Normalize to sum=1, non-negative."""
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-15:
        return np.ones_like(a) / len(a)
    return a / s


def _make_start_uniform(n):
    return np.ones(n) / n


def _make_start_trapezoid(n, frac=0.5):
    a = np.zeros(n)
    k = int(n * frac)
    a[:k] = 1.0
    a /= np.sum(a)
    return a


def _make_start_sparse_random(n, m, rng):
    a = np.zeros(n)
    idx = rng.choice(n, size=m, replace=False)
    a[idx] = 1.0
    return a / np.sum(a)


def _make_start_quadratic_residue(n, rng):
    """Sequence based on quadratic residues mod p (next prime >= n)."""
    # Find a prime near n
    p = n
    while not _is_prime(p):
        p += 1
    a = np.zeros(n)
    for i in range(1, p):
        if (i * i) % p == 1 or _is_quadratic_residue(i, p):
            if i < n:
                a[i] = 1.0
    s = np.sum(a)
    if s > 0:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def _is_prime(x):
    if x < 2:
        return False
    if x < 4:
        return True
    if x % 2 == 0 or x % 3 == 0:
        return False
    i = 5
    while i * i <= x:
        if x % i == 0 or x % (i + 2) == 0:
            return False
        i += 6
    return True


def _is_quadratic_residue(a, p):
    return pow(a, (p - 1) // 2, p) == 1


def _make_start_sin_profile(n, A=0.5):
    x = np.linspace(0, 1, n)
    a = 1.0 + A * np.sin(2 * np.pi * x)
    a = np.maximum(a, 0.0)
    a += 0.01
    return a / np.sum(a)


def _make_start_from_provided(n, rng):
    """Resample the provided height_sequence_1 to length n."""
    try:
        g = globals().get("height_sequence_1")
        if g is not None:
            g = np.asarray(g, dtype=np.float64)
            if g.size > 1 and np.all(np.isfinite(g)) and np.sum(g) > 0:
                g = np.maximum(g, 0)
                x_old = np.linspace(0, 1, g.size)
                x_new = np.linspace(0, 1, n)
                # Interpolate in log-space for better handling of small values
                g_safe = np.maximum(g, 1e-300)
                z = np.log(g_safe)
                z_interp = np.interp(x_new, x_old, z)
                a = np.exp(z_interp - np.max(z_interp))
                a += 1e-6
                return a / np.sum(a)
    except Exception:
        pass
    return np.ones(n) / n


def _gradient_descent(a0, n, deadline, seed=0, max_iters=3000):
    """
    Projected gradient descent on the simplex with adaptive tau.
    """
    _set_state(n)
    rng = np.random.default_rng(seed)

    a = _normalize(a0.copy())
    best_a = a.copy()
    best_score = _score_normalized(a)

    # Adaptive tau: start large (smooth), end small (sharp)
    tau_start = 0.1
    tau_end = 0.0005
    n_phases = 3
    iters_per_phase = max_iters // n_phases

    for phase in range(n_phases):
        tau_lo = tau_start * (tau_end / tau_start) ** (phase / n_phases)
        tau_hi = tau_start * (tau_end / tau_start) ** ((phase + 1) / n_phases)

        for it in range(iters_per_phase):
            if time.monotonic() > deadline:
                break

            # Linear tau decay within phase
            t_frac = it / max(1, iters_per_phase)
            tau = tau_hi * (tau_lo / tau_hi) ** t_frac
            tau = max(tau, 1e-6)

            grad, F_val = _grad_smoothed(a, tau)

            if not np.all(np.isfinite(grad)):
                grad = np.zeros(n)

            # Normalize gradient for stable step sizes
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-15:
                break
            grad_normed = grad / gnorm

            # Backtracking line search
            a_old_score = _score_normalized(a)
            found = False
            for eta in (1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 0.1):
                a_try = a - eta * grad_normed * gnorm * 0.1
                a_try = _project_simplex(a_try)
                sc_try = _score_normalized(a_try)
                if sc_try < a_old_score - 1e-12:
                    a = a_try
                    if sc_try < best_score:
                        best_score = sc_try
                        best_a = a.copy()
                    found = True
                    break

            if not found:
                # Try smaller steps
                for eta in (1e-5, 3e-5, 1e-6):
                    a_try = a - eta * grad_normed * gnorm * 0.1
                    a_try = _project_simplex(a_try)
                    sc_try = _score_normalized(a_try)
                    if sc_try < a_old_score - 1e-12:
                        a = a_try
                        if sc_try < best_score:
                            best_score = sc_try
                            best_a = a.copy()
                        break

            # Occasionally add noise to escape local minima
            if it % 200 == 199:
                noise_scale = 0.01 * (1.0 + phase * 0.5)
                a_noisy = a + noise_scale * rng.standard_normal(n)
                a_noisy = _project_simplex(a_noisy)
                sc_noisy = _score_normalized(a_noisy)
                if sc_noisy < best_score:
                    best_score = sc_noisy
                    best_a = a_noisy.copy()
                    a = a_noisy.copy()

        if time.monotonic() > deadline:
            break
        print(f"    Phase {phase}: tau=[{tau_hi:.6f}..{tau_lo:.6f}], "
              f"score={best_score:.9f}")

    return best_a


def _coordinate_polish(a0, n, deadline, seed=1):
    """
    Coordinate descent: try changing one coordinate at a time.
    For each coordinate i, find the optimal value given all others fixed.
    """
    _set_state(n)
    rng = np.random.default_rng(seed)
    a = _normalize(a0.copy())
    best_a = a.copy()
    best_score = _score_normalized(a)

    n_coords = n
    no_improve = 0
    max_no_improve = 50

    order = rng.permutation(n_coords)
    idx = 0

    while time.monotonic() < deadline and no_improve < max_no_improve:
        i = order[idx % n_coords]
        idx += 1

        # Try different values for a[i], keeping sum = 1
        # If we change a[i] by delta, we need to adjust another coordinate.
        # Simpler: try setting a[i] to various fractions of its current value,
        # and redistribute the difference uniformly.
        a_i = a[i]
        if a_i < 1e-15:
            no_improve += 1
            continue

        old_score = best_score
        a_saved = a.copy()

        for frac in (0.3, 0.5, 0.7, 1.3, 1.5, 2.0, 0.1, 0.0):
            new_a = a_saved.copy()
            delta = new_a[i] * (frac - 1.0)
            new_a[i] *= frac
            # Redistribute delta to other elements proportionally
            other_mask = np.ones(n, dtype=bool)
            other_mask[i] = False
            other_sum = np.sum(new_a[other_mask])
            if other_sum > 0:
                new_a[other_mask] -= delta * new_a[other_mask] / other_sum
            new_a = np.maximum(new_a, 0)
            s = np.sum(new_a)
            if s > 0:
                new_a /= s
                sc = _score_normalized(new_a)
                if sc < best_score - 1e-12:
                    best_score = sc
                    best_a = new_a.copy()
                    a = new_a.copy()
                    break

        if best_score < old_score - 1e-12:
            no_improve = 0
        else:
            no_improve += 1

        if idx % 5000 == 0:
            print(f"    Coord polish idx={idx}, score={best_score:.9f}")

    return best_a


def _perturbation_search(a0, n, deadline, seed=2):
    """
    Random perturbation search: make small random changes, accept if better.
    """
    _set_state(n)
    rng = np.random.default_rng(seed)
    a = _normalize(a0.copy())
    best_a = a.copy()
    best_score = _score_normalized(a)

    n_pert = int(np.sqrt(n))  # number of coordinates to perturb
    temp = 0.01
    improvements = 0
    total = 0

    while time.monotonic() < deadline:
        # Create perturbation
        mask = rng.random(n) < (n_pert / n)
        if not np.any(mask):
            mask[rng.integers(n)] = True
        noise = rng.standard_normal(n) * temp * np.sqrt(n_pert)
        noise[~mask] = 0.0

        a_try = a + noise
        a_try = _project_simplex(a_try)
        sc_try = _score_normalized(a_try)

        total += 1
        if sc_try < best_score:
            best_score = sc_try
            best_a = a_try.copy()
            a = a_try.copy()
            improvements += 1
        elif sc_try < best_score + temp * best_score:
            # Accept worse solutions with some probability (simulated annealing)
            if rng.random() < 0.1:
                a = a_try.copy()

        # Cool down
        if total % 100 == 0:
            temp *= 0.95
            if temp < 1e-5:
                break

    return best_a


def _two_scale_optimize(a_coarse, n_fine, n_coarse, deadline, seed=0):
    """
    Take a coarse solution, upsample to fine resolution, and optimize.
    """
    _set_state(n_fine)
    # Upsample by averaging blocks
    ratio = n_fine // n_coarse
    if ratio < 1:
        return _normalize(a_coarse)

    a_fine = np.zeros(n_fine)
    for i in range(n_coarse):
        start = i * ratio
        end = start + ratio
        a_fine[start:end] = a_coarse[i]

    a_fine = _normalize(a_fine)

    # Add small random perturbation to break symmetry
    rng = np.random.default_rng(seed)
    a_fine += 0.01 * rng.standard_normal(n_fine)
    a_fine = _normalize(np.maximum(a_fine, 0))

    # Optimize fine version
    time_left = deadline - time.monotonic()
    max_iters = min(2000, int(max(100, time_left * 3)))
    a_opt = _gradient_descent(a_fine, n_fine, deadline, seed=seed, max_iters=max_iters)
    return a_opt

    print(f"\nRunning perturbation search on best (score={best_score:.9f})")
    pert_deadline = time.monotonic() + min(120.0, max(30.0, deadline - time.monotonic() - 20.0))
    a_pert = _perturbation_search(best_a, n, pert_deadline, seed=seed + 200)
    sc_pert = _score_normalized(a_pert)
    print(f"  After perturbation: score={sc_pert:.9f}")

    if sc_pert < best_score:
        best_score = sc_pert
        best_a = a_pert.copy()

    # Final coordinate polish
    final_deadline = time.monotonic() + min(60.0, max(10.0, deadline - time.monotonic() - 5.0))
    a_final = _coordinate_polish(best_a, n, final_deadline, seed=seed + 300)
    sc_final = _score_normalized(a_final)
    print(f"  After final polish: score={sc_final:.9f}")

    if sc_final < best_score:
        best_score = sc_final
        best_a = a_final.copy()

    # Validate with official evaluator if available
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.maximum(best_a, 0.0)
    best_a = np.minimum(best_a, 1000.0)
    s = float(np.sum(best_a))
    if s < 0.01 or not np.isfinite(s):
        best_a = np.ones(n) / n
    else:
        best_a = best_a / s

    try:
        exact = evaluate_sequence(best_a.tolist())
        print(f"\nOfficial evaluator score: {exact:.9f}")
        if not np.isfinite(exact):
            print("WARNING: Invalid score from evaluator, using fallback")
            best_a = np.ones(n) / n
    except Exception as e:
        print(f"Evaluator check failed: {e}")

    elapsed = time.monotonic() - t0
    print(f"\nTotal time: {elapsed:.1f}s, final internal score: {best_score:.9f}")

    # Final safety
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.maximum(best_a, 0.0)
    best_a = np.minimum(best_a, 1000.0)
    s = float(np.sum(best_a))
    if s < 0.01 or not np.isfinite(s):
        best_a = np.ones(n) / n
    else:
        best_a /= s

    return [float(v) for v in best_a.tolist()]
