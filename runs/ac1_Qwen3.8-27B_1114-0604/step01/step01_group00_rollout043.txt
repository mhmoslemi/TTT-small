probability simplex.  It alternates between:

  * cheap top-active-lag multiplicative updates,
  * occasional full softmax/FFT-smoothed annealing passes,
  * stochastic geometric/skew perturbations to escape local minima,
  * a final exact pairwise-transfer polish.

All internal objective evaluations use FFT-based linear convolution.
"""

import time
import numpy as np


def _pow2_at_least(m):
    """Smallest power of two >= m (m is a Python int)."""
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _normalize(a):
    """Project to the non-negative simplex sum=1."""
    a = np.asarray(a, dtype=np.float64).copy()
    if a.size == 0:
        return np.ones(1, dtype=np.float64)
    a = np.where(a > 0.0, a, 0.0)
    s = a.sum()
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(a.size, dtype=np.float64) / float(a.size)
    a /= s
    return a


def _autocorr(a):
    """Full linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    size = _pow2_at_least(2 * n - 2)
    fa = np.fft.rfft(a, size)
    c = np.fft.irfft(fa * fa, size)[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c


def _local_score(a):
    """Fast local objective; equivalent to evaluate_sequence for valid inputs."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n < 1:
        return np.inf
    a = np.where(a > 0.0, a, 0.0)
    s = a.sum()
    if not np.isfinite(s) or s < 1e-12:
        return np.inf
    a = a / s
    c = _autocorr(a)
    return float(2.0 * n * c.max())


def _score_key(item):
    """Sort key for (score, array) tuples."""
    return item[0]


def _make_starts(n, rng, given_arr=None):
    """Diverse normalized starting sequences of length n."""
    starts = []

    if given_arr is not None:
        g = _normalize(given_arr)
        if g.size == n:
            starts.append(g)

    starts.append(np.ones(n, dtype=np.float64) / float(n))

    t = np.linspace(0.0, 1.0, n, dtype=np.float64)

    # Skewed decays / powers.
    for alpha in (0.5, 1.0, 2.0, 4.0, 8.0):
        for p in (0.5, 1.0, 2.0):
            base = np.exp(-alpha * np.power(t, p)) + 1e-8
            base = np.maximum(base, 0.0)
            s = base.sum()
            if s > 1e-300:
                starts.append(base / s)

    for p in (0.25, 0.5, 1.0, 2.0, 4.0):
        base = np.power(1.0 - t + 1e-8, p) + 1e-8
        base = np.maximum(base, 0.0)
        s = base.sum()
        if s > 1e-300:
            starts.append(base / s)

    # Interior humps.
    for center, width in ((0.3, 0.2), (0.5, 0.3), (0.7, 0.2)):
        base = np.exp(-((t - center) / width) ** 2) + 1e-4
        s = base.sum()
        if s > 1e-300:
            starts.append(base / s)

    # U-shapes.
    u = 2.0 * t - 1.0
    for beta in (0.3, 0.8, 1.5):
        base = 1.0 + beta * u * u + 1e-4
        s = base.sum()
        if s > 1e-300:
            starts.append(base / s)

    # Smooth oscillations.
    for freq in (1, 2, 3):
        for phase in (0.0, 0.7, 1.4):
            base = 1.0 + 0.6 * np.sin(2.0 * np.pi * freq * t + phase) + 1e-4
            base = np.maximum(base, 0.0)
            s = base.sum()
            if s > 1e-300:
                starts.append(base / s)

    # Random starts (only for moderate n to save startup).
    if n <= 12000:
        alphas = (0.05, 0.25, 1.0, 5.0)
        for alpha in alphas:
            try:
                x = rng.dirichlet(np.full(n, alpha, dtype=np.float64))
                starts.append(x)
            except Exception:
                pass

        x = rng.exponential(1.0, size=n)
        x = np.where(x > 0.0, x, 0.0)
        s = x.sum()
        if s > 1e-300:
            starts.append(x / s)

        x = 0.1 + rng.random(n, dtype=np.float64)
        s = x.sum()
        if s > 1e-300:
            starts.append(x / s)

    return starts


def _geometric_pattern(n, rng):
    """Random skewed exponential pattern, normalized."""
    t = np.linspace(0.0, 1.0, n, dtype=np.float64)
    alpha = float(rng.uniform(0.5, 8.0))
    p = float(rng.uniform(0.5, 2.5))
    base = np.exp(-alpha * np.power(t, p)) + 1e-8
    base = np.maximum(base, 0.0)
    s = base.sum()
    if s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    return base / s


def _perturb_array(a, rng, n, sigma, pattern, rho):
    """Multiplicative log-noise perturbation and optional mixture with a pattern."""
    b = np.asarray(a, dtype=np.float64).copy()

    if sigma > 0.0:
        noise = sigma * rng.standard_normal(n, dtype=np.float64)
        noise = np.clip(noise, -8.0, 8.0)
        b = b * np.exp(noise)

    if pattern is not None and rho > 0.0:
        rho = min(max(float(rho), 0.0), 0.75)
        b = (1.0 - rho) * b + rho * pattern

    b = np.maximum(b, 0.0)
    s = b.sum()
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones(n, dtype=np.float64) / float(n)
    b /= s
    return b


def _top_active(c, K):
    """Return active convolution-lag indices and non-negative weights."""
    c = np.asarray(c, dtype=np.float64)
    L = c.size
    K = int(K)

    if K <= 1:
        idx = np.array([int(np.argmax(c))], dtype=np.int64)
        return idx, np.array([1.0], dtype=np.float64)

    if K >= L:
        return np.arange(L, dtype=np.int64), np.full(L, 1.0 / float(L), dtype=np.float64)

    idx = np.argpartition(c, -K)[-K:].astype(np.int64)
    csel = c[idx]
    cmax = float(csel.max())
    cmin = float(csel.min())
    diff = cmax - cmin

    if diff <= 1e-18 * max(1.0, abs(cmax)) or diff <= 0.0:
        return idx, np.full(K, 1.0 / float(K), dtype=np.float64)

    T = diff / 6.0
    z = (csel - cmax) / T
    z = np.clip(z, -50.0, 0.0)
    w = np.exp(z)
    sw = w.sum()
    if not np.isfinite(sw) or sw <= 1e-300:
        w = np.full(K, 1.0 / float(K), dtype=np.float64)
    else:
        w /= sw
    return idx, w


def _gradient_top(a, idx, weights):
    """Exact gradient of sum_k weights_k c_k with respect to a."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    grad = np.zeros(n, dtype=np.float64)

    n1 = n - 1
    for ii in range(idx.size):
        w = float(weights[ii])
        if w <= 1e-14:
            continue
        k = int(idx[ii])

        lo = k - n1
        if lo < 0:
            lo = 0
        hi = k
        if hi > n1:
            hi = n1

        start = k - lo
        stop = k - hi - 1
        if stop < 0:
            seg = a[start::-1]
        else:
            seg = a[start:stop:-1]

        if w < 0.999999999:
            grad[lo:hi + 1] += seg * w
        else:
            grad[lo:hi + 1] += seg

    grad *= 2.0
    return grad


def _softmax_vector(v, T):
    """Numerically stable softmax of v with temperature T."""
    v = np.asarray(v, dtype=np.float64)
    if T is None or not np.isfinite(T) or T <= 1e-300:
        p = np.zeros(v.size, dtype=np.float64)
        p[int(np.argmax(v))] = 1.0
        return p

    z = v / T
    zmax = z.max()
    if not np.isfinite(zmax):
        zmax = 0.0
    z = z - zmax
    p = np.exp(z)
    s = p.sum()
    if not np.isfinite(s) or s <= 1e-300:
        p = np.zeros(v.size, dtype=np.float64)
        p[int(np.argmax(v))] = 1.0
        return p
    p /= s
    return p


def _gradient_weight_fft(a, p, P_rev, size):
    """
    Gradient using dense weights p over all convolution lags.

    P_rev is FFT(padding(reverse(p))).
    """
    a = np.asarray(a, dtype=np.float64)
    p = np.asarray(p, dtype=np.float64)
    n = a.size
    L = p.size

    if n == 1:
        return np.array([2.0 * float(p[0]) * float(a[0])], dtype=np.float64)

    A = np.fft.rfft(a, size)
    conv = np.fft.irfft(A * P_rev, size)

    start = L - 1
    stop = start - n
    if stop >= 0:
        grad = conv[start:stop:-1]
    else:
        grad = conv[start::-1]

    return 2.0 * grad


def _mirror_step(a, grad, lr, floor):
    """
    Multiplicative update on the non-negative simplex.

    Gradient is normalized so the exponent range is approximately [-lr, lr].
    """
    a = np.asarray(a, dtype=np.float64)
    grad = np.asarray(grad, dtype=np.float64)

    gmax = float(grad.max())
    gmin = float(grad.min())
    span = gmax - gmin
    if not np.isfinite(span) or span <= 1e-300:
        return a

    center = 0.5 * (gmax + gmin)
    scale = max(gmax - center, center - gmin, 1e-300)
    expo = -float(lr) * (grad - center) / scale
    expo = np.clip(expo, -50.0, 50.0)

    a = a * np.exp(expo)
    a = np.maximum(a, float(floor))
    s = a.sum()
    if not np.isfinite(s) or s <= 1e-300:
        return np.ones_like(a, dtype=np.float64) / float(a.size)
    a /= s
    return a


def _optimize_window(a0, n, duration, rng):
    """
    Main fast multiplicative search window using top-active-lag gradients.
    """
    a = _normalize(a0)
    n = int(n)
    c = _autocorr(a)
    score = float(2.0 * n * c.max())

    best_a = a.copy()
    best_score = score

    if duration <= 0.0:
        return best_a, best_score

    t0 = time.monotonic()
    end = t0 + float(duration)

    K_max = 96 if n < 3000 else (64 if n < 9000 else 44)
    K_min = 8
    lr_start = 0.85
    lr_end = 0.05
    recomp_start = 25
    recomp_end = 6

    floor = max(1e-18, 1e-12 / float(n))

    idx, weights = _top_active(c, K_max)
    since = 0
    it = 0
    stuck = 0
    grad_ema = None

    while True:
        now = time.monotonic()
        if now >= end:
            break

        progress = min(1.0, max(0.0, (now - t0) / max(1e-6, duration)))
        every = int(recomp_end + (recomp_start - recomp_end) * (1.0 - progress) ** 0.6)
        every = max(4, every)

        if since >= every:
            c = _autocorr(a)
            cm = float(c.max())
            sc = float(2.0 * n * cm)

            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a.copy()
                stuck = 0
            else:
                stuck += 1

            if stuck > 6:
                sigma = 0.02 + 0.08 * (1.0 - progress) + 0.004 * float(stuck)
                use_pattern = (stuck % 2 == 0)
                pat = _geometric_pattern(n, rng) if use_pattern else None
                rho = 0.10 + 0.04 * float(max(0, stuck - 6))
                a = _perturb_array(a, rng, n, sigma, pat, rho)
                stuck = 0
                grad_ema = None

                c = _autocorr(a)
                cm = float(c.max())
                sc = float(2.0 * n * cm)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_a = a.copy()
                stuck = 1

            K = int(K_min + (K_max - K_min) * (1.0 - progress) ** 0.7)
            idx, weights = _top_active(c, K)
            since = -1

        grad = _gradient_top(a, idx, weights)

        if progress < 0.65:
            if grad_ema is None:
                grad_ema = grad.copy()
            else:
                grad_ema *= 0.7
                grad_ema += 0.3 * grad
            use_grad = grad_ema
        else:
            use_grad = grad

        lr = lr_end + (lr_start - lr_end) * (1.0 - progress) ** 0.5
        a = _mirror_step(a, use_grad, lr, floor)

        it += 1
        since += 1

    c = _autocorr(a)
    sc = float(2.0 * n * c.max())
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def _optimize_soft_window(a0, n, duration, rng):
    """
    Softer full-softmax mirror descent window.
    Used only occasionally as an escape mechanism.
    """
    a = _normalize(a0)
    n = int(n)
    c = _autocorr(a)
    score = float(2.0 * n * c.max())

    best_a = a.copy()
    best_score = score

    if duration <= 0.0:
        return best_a, best_score

    t0 = time.monotonic()
    end = t0 + float(duration)

    T_start = 0.35
    T_end = 1e-5
    lr_start = 0.55
    lr_end = 0.04
    recomp_start = 20
    recomp_end = 4
    floor = max(1e-18, 1e-12 / float(n))

    size = _pow2_at_least(3 * n - 2)

    cmax = float(c.max())
    T = T_start * max(cmax, 1e-30)
    p = _softmax_vector(c, T)
    P_rev = np.fft.rfft(p[::-1], size)

    since = 0
    it = 0
    stuck = 0

    while True:
        now = time.monotonic()
        if now >= end:
            break

        progress = min(1.0, max(0.0, (now - t0) / max(1e-6, duration)))
        every = int(recomp_end + (recomp_start - recomp_end) * (1.0 - progress) ** 0.5)
        every = max(3, every)

        if since >= every:
            c = _autocorr(a)
            cmax = float(c.max())
            sc = float(2.0 * n * cmax)

            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a.copy()
                stuck = 0
            else:
                stuck += 1

            if stuck > 10:
                sigma = 0.03 + 0.10 * (1.0 - progress)
                pat = _geometric_pattern(n, rng)
                a = _perturb_array(a, rng, n, sigma, pat, 0.25)
                stuck = 0

                c = _autocorr(a)
                cmax = float(c.max())
                sc = float(2.0 * n * cmax)
                if sc < best_score - 1e-10:
                    best_score = sc
                    best_a = a.copy()
                stuck = 1

            Tfac = T_end + (T_start - T_end) * (1.0 - progress) ** 0.8
            T = max(cmax, 1e-30) * float(Tfac)
            p = _softmax_vector(c, T)
            P_rev = np.fft.rfft(p[::-1], size)
            since = -1

        grad = _gradient_weight_fft(a, p, P_rev, size)
        lr = lr_end + (lr_start - lr_end) * (1.0 - progress) ** 0.5
        a = _mirror_step(a, grad, lr, floor)

        it += 1
        since += 1

    c = _autocorr(a)
    sc = float(2.0 * n * c.max())
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def _transfer_eval(c, a, u, v, eps, n):
    """
    Exact convolution update for moving eps mass from u to v.

    Returns (new_c, new_max).
    """
    c = np.asarray(c, dtype=np.float64)
    a = np.asarray(a, dtype=np.float64)
    L = 2 * n - 1

    c2 = c.copy()
    coef = 2.0 * float(eps)

    c2[v:v + n] += coef * a
    c2[u:u + n] -= coef * a

    e2 = float(eps) * float(eps)
    if 2 * u < L:
        c2[2 * u] += e2
    if 2 * v < L:
        c2[2 * v] += e2
    kv = u + v
    if kv < L:
        c2[kv] -= 2.0 * e2

    return c2, float(c2.max())


def _polish_pairs(a, n, deadline, rng):
    """
    Exact local pair-transfer polish.
    """
    a = _normalize(a)
    n = int(n)
    c = _autocorr(a)
    cmax = float(c.max())
    score = float(2.0 * n * cmax)

    best_a = a.copy()
    best_score = score

    if n < 2:
        return best_a, best_score

    fracs = (0.02, 0.05, 0.12, 0.25)
    activeK = 4 if n <= 10000 else 3
    topD = 3
    topR = 3
    no_improve = 0
    accepted = 0

    while True:
        now = time.monotonic()
        if now >= deadline:
            break
        if deadline - now < 5.0:
            break

        if accepted > 0 and accepted % 50 == 0:
            c = _autocorr(a)
            cmax = float(c.max())
            sc = float(2.0 * n * cmax)
            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a.copy()
                no_improve = 0

        if activeK < c.size:
            idx = np.argpartition(c, -activeK)[-activeK:].astype(np.int64)
        else:
            idx = np.arange(c.size, dtype=np.int64)

        best_newmax = np.inf
        best_cnew = None
        best_u = -1
        best_v = -1
        best_eps = 0.0

        for kk in idx:
            k = int(kk)
            n1 = n - 1
            lo = k - n1
            if lo < 0:
                lo = 0
            hi = k
            if hi > n1:
                hi = n1

            h = np.zeros(n, dtype=np.float64)
            start = k - lo
            stop = k - hi - 1
            if stop < 0:
                h[lo:hi + 1] = a[start::-1]
            else:
                h[lo:hi + 1] = a[start:stop:-1]

            hd = h.copy()
            hd[a <= 1e-20] = -1e300
            if hd.size > topD:
                donors = np.argpartition(hd, -topD)[-topD:]
            else:
                donors = np.argsort(hd)[::-1][:topD]

            hr = h.copy()
            if hr.size > topR:
                receivers = np.argpartition(hr, topR)[:topR]
            else:
                receivers = np.argsort(hr)[:topR]

            for uu in donors:
                u = int(uu)
                au = float(a[u])
                if au <= 1e-20:
                    continue
                for vv in receivers:
                    v = int(vv)
                    if v == u:
                        continue
                    for frac in fracs:
                        eps = au * float(frac)
                        if eps <= 1e-300:
                            continue
                        cnew, newmax = _transfer_eval(c, a, u, v, eps, n)
                        if newmax < best_newmax:
                            best_newmax = newmax
                            best_cnew = cnew
                            best_u = u
                            best_v = v
                            best_eps = eps

        applied = False
        target = best_score - 1e-10
        if best_cnew is not None and (2.0 * n * best_newmax) < target:
            a = a.copy()
            a[best_u] -= best_eps
            a[best_v] += best_eps
            s = a.sum()
            if not np.isfinite(s) or s <= 1e-300:
                a = _normalize(a)
            else:
                a /= s
                c = best_cnew / (s * s)
            cmax = c.max()
            sc = float(2.0 * n * cmax)

            a = np.maximum(a, 0.0)
            s = a.sum()
            if s <= 1e-300:
                a = np.ones(n, dtype=np.float64) / float(n)
                c = _autocorr(a)
                cmax = float(c.max())
                sc = float(2.0 * n * cmax)
            else:
                a /= s

            if sc < best_score - 1e-10:
                best_score = sc
                best_a = a.copy()
            no_improve = 0
            accepted += 1
            applied = True

        if not applied:
            # A few random transfers to escape small plateaus.
            rand_best = None
            for _ in range(6 if n <= 10000 else 3):
                u = int(rng.integers(0, n))
                v = int(rng.integers(0, n))
                if u == v or a[u] <= 1e-20:
                    continue
                fi = int(rng.integers(0, len(fracs)))
                eps = float(a[u]) * float(fracs[fi])
                if eps <= 1e-300:
                    continue
                cnew, newmax = _transfer_eval(c, a, u, v, eps, n)
                if (2.0 * n * newmax) < target:
                    rand_best = (u, v, eps, cnew, newmax)
                    break

            if rand_best is not None:
                u, v, eps, cnew, newmax = rand_best
                a = a.copy()
                a[u] -= eps
                a[v] += eps
                s = a.sum()
                if not np.isfinite(s) or s <= 1e-300:
                    a = _normalize(a)
                else:
                    a /= s
                    c = cnew / (s * s)
                cmax = float(c.max())
                sc = float(2.0 * n * cmax)

                a = np.maximum(a, 0.0)
                s = a.sum()
                if s <= 1e-300:
                    a = np.ones(n, dtype=np.float64) / float(n)
                    c = _autocorr(a)
                    cmax = float(c.max())
                    sc = float(2.0 * n * cmax)
                else:
                    a /= s

                if sc < best_score - 1e-10:
                    best_score = sc
                    best_a = a.copy()
                no_improve = 0
                accepted += 1
                applied = True

        if not applied:
            no_improve += 1
            if no_improve > 80:
                break

    c = _autocorr(a)
    sc = float(2.0 * n * c.max())
    if sc < best_score:
        best_score = sc
        best_a = a.copy()

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400.0, **kwargs):
    """
    Search function invoked by the harness.

    Returns a list of non-negative floats minimizing the provided evaluation.
    """
    rng = np.random.default_rng(int(seed) + 987654321)
    t0 = time.monotonic()

    try:
        total_budget = float(budget_s)
    except Exception:
        total_budget = 2400.0
    if not np.isfinite(total_budget) or total_budget <= 0.0:
        total_budget = 2400.0

    # Keep enough reserve for final polish/validation.
    reserve = 45.0 if total_budget >= 600.0 else min(15.0, 0.05 * total_budget)
    deadline = t0 + max(60.0, total_budget - reserve)

    MAX_RETURN_N = 15000
    MAX_WORK_N = 15000

    # ------------------------------------------------------------------
    # Use the previous known construction if its size is manageable.
    # ------------------------------------------------------------------
    original = None
    original_score = np.inf
    provided = globals().get("height_sequence_1", None)

    if provided is not None:
        try:
            arr = np.asarray(provided, dtype=np.float64)
            if arr.ndim != 1:
                arr = arr.ravel()
            arr = np.where(np.isfinite(arr), arr, 0.0)
            arr = np.clip(arr, 0.0, 1000.0)
            if arr.size >= 2 and arr.size <= 200000:
                sc = _local_score(arr)
                if np.isfinite(sc):
                    original = arr.copy()
                    original_score = float(sc)
        except Exception:
            original = None
            original_score = np.inf

    if original is not None and 512 <= original.size <= MAX_WORK_N:
        n_main = int(original.size)
    else:
        n_main = 6121

    if n_main < 64:
        n_main = 6121
    if n_main > MAX_WORK_N:
        n_main = 6121

    # ------------------------------------------------------------------
    # Build initial pool for the main working length.
    # ------------------------------------------------------------------
    given_same_len = None
    if original is not None and original.size == n_main:
        given_same_len = original

    starts = _make_starts(n_main, rng, given_same_len)

    # If the original sequence has different length, downsample it as a start.
    if original is not None and original.size != n_main and original.size <= 200000:
        step = max(1, int(original.size) // max(1, n_main))
        down = original[::step]
        if down.size >= 16:
            starts.append(_normalize(down))

    scored = []
    max_eval_starts = len(starts) if total_budget > 300.0 else max(8, len(starts) // 4)

    for i, s in enumerate(starts):
        if i >= max_eval_starts:
            break
        try:
            sc = _local_score(s)
        except Exception:
            sc = np.inf
        if np.isfinite(sc):
            scored.append((float(sc), s))

    if scored:
        scored.sort(key=_score_key)

    uniform = np.ones(n_main, dtype=np.float64) / float(n_main)
    work_a = uniform.copy()
    work_score = float(2.0 * n_main)

    global_a = None
    global_score = np.inf

    if original is not None and np.isfinite(original_score):
        global_a = original.copy()
        global_score = float(original_score)

    for sc, arr in scored:
        if sc < work_score:
            work_score = sc
            work_a = arr.copy()
        if np.isfinite(sc) and sc < global_score:
            global_score = sc
            global_a = arr.copy()

    if global_a is None or not np.isfinite(global_score):
        global_a = work_a.copy()
        global_score = work_score

    print(f"n_main={n_main}, work_score={work_score:.8f}, global_score={global_score:.8f}", flush=True)

    # ------------------------------------------------------------------
    # Multi-window basin search.
    # ------------------------------------------------------------------
    attempt = 0
    stagnant = 0
    noise_sched = (0.0, 0.004, 0.008, 0.016, 0.028, 0.045, 0.070, 0.105, 0.150, 0.220)
    mix_sched = (0.0, 0.0, 0.08, 0.14, 0.20, 0.26, 0.32, 0.38, 0.44, 0.50)

    while True:
        now = time.monotonic()
        remaining = deadline - now
        if remaining < 150.0:
            break

        if remaining > 1200.0:
            run_time = min(360.0, remaining * 0.25)
        elif remaining > 650.0:
            run_time = min(250.0, remaining * 0.35)
        else:
            run_time = remaining - 150.0
        run_time = max(30.0, run_time)

        if attempt == 0:
            a0 = work_a.copy()
            sigma = 0.0
            rho = 0.0
            pattern = None
        else:
            si = min(attempt, len(noise_sched) - 1)
            sigma = float(noise_sched[si])
            if stagnant > 0:
                sigma = max(sigma, 0.02)
            if stagnant > 2:
                sigma = min(0.30, sigma + 0.02 * float(stagnant - 2))

            mi = min(attempt, len(mix_sched) - 1)
            rho = float(mix_sched[mi])
            if stagnant > 0:
                rho = min(0.55, rho + 0.03 * float(stagnant))

            # Use a fresh skewed mixture every few attempts.
            if attempt % 4 == 0:
                pattern = _geometric_pattern(n_main, rng)
            elif attempt % 7 == 3 and n_main <= 12000:
                # Random Dirichlet-like start mixed with current best.
                try:
                    pattern = rng.dirichlet(np.full(n_main, 0.25, dtype=np.float64))
                except Exception:
                    pattern = None
            else:
                pattern = None

            a0 = _perturb_array(work_a, rng, n_main, sigma, pattern, rho)

        # Occasional soft window to leave low-dimensional sticky basins.
        use_soft = (attempt > 2) and (stagnant >= 1) and (attempt % 3 == 0) and (remaining > 420.0)

        if use_soft:
            soft_time = min(run_time, 180.0)
            res_a, res_sc = _optimize_soft_window(a0, n_main, soft_time, rng)
            if np.isfinite(res_sc):
                if res_sc < work_score:
                    work_score = res_sc
                    work_a = res_a.copy()
                    stagnant = 0
                elif res_sc <= work_score + 1e-8 and float(rng.random()) < 0.35:
                    work_a = res_a.copy()

                if res_sc < global_score:
                    global_score = res_sc
                    global_a = res_a.copy()
                    stagnant = 0
                    print(f"attempt {attempt}: soft -> global {global_score:.8f}", flush=True)
            if time.monotonic() >= deadline - 120.0:
                break
        else:
            res_a, res_sc = _optimize_window(a0, n_main, run_time, rng)
            if np.isfinite(res_sc):
                if res_sc < work_score:
                    work_score = res_sc
                    work_a = res_a.copy()
                    stagnant = 0
                elif res_sc <= work_score + 1e-8 and float(rng.random()) < 0.35:
                    work_a = res_a.copy()

                if res_sc < global_score:
                    global_score = res_sc
                    global_a = res_a.copy()
                    stagnant = 0
                    print(f"attempt {attempt}: direct -> global {global_score:.8f}", flush=True)
                else:
                    stagnant += 1
            else:
                stagnant += 1

        attempt += 1

    # ------------------------------------------------------------------
    # Final exact pair-transfer polish.
    # ------------------------------------------------------------------
    now = time.monotonic()
    polish_budget = deadline - now
    if polish_budget > 30.0:
        if global_a is not None and global_a.size <= MAX_RETURN_N and np.isfinite(global_score):
            if np.isfinite(work_score) and work_score < global_score:
                polish_a = work_a.copy()
            else:
                polish_a = global_a.copy()
        else:
            polish_a = work_a.copy()

        pdeadline = min(deadline - 3.0, now + max(25.0, min(160.0, polish_budget * 0.65)))
        pa, ps = _polish_pairs(polish_a, len(polish_a), pdeadline, rng)

        if np.isfinite(ps):
            if len(polish_a) == n_main:
                if ps < work_score:
                    work_score = ps
                    work_a = pa.copy()
                elif ps <= work_score + 1e-9:
                    work_a = pa.copy()

            if ps < global_score:
                global_score = ps
                global_a = pa.copy()
                print(f"polish -> global {global_score:.8f}", flush=True)

    # If we still have meaningful time, do one last short polish from work_a.
    now = time.monotonic()
    if deadline - now > 90.0 and work_a is not None:
        pdeadline = min(deadline - 3.0, now + min(80.0, deadline - now - 5.0))
        pa, ps = _polish_pairs(work_a, n_main, pdeadline, rng)
        if np.isfinite(ps):
            if ps < work_score:
                work_score = ps
                work_a = pa.copy()
            if ps < global_score:
                global_score = ps
                global_a = pa.copy()
                print(f"final polish -> global {global_score:.8f}", flush=True)

    # ------------------------------------------------------------------
    # Choose a safe output sequence.
    # ------------------------------------------------------------------
    if global_a is None or not np.isfinite(global_score):
        final_a = work_a.copy()
    elif global_a.size <= MAX_RETURN_N:
        final_a = global_a.copy()
    else:
        final_a = work_a.copy()

    final_a = np.asarray(final_a, dtype=np.float64)
    if final_a.ndim != 1:
        final_a = final_a.ravel()

    if final_a.size < 2 or final_a.size > MAX_RETURN_N:
        final_a = work_a.copy()

    if final_a.size < 2:
        final_a = np.ones(6121, dtype=np.float64) / 6121.0

    final_a = np.where(np.isfinite(final_a), final_a, 0.0)
    final_a = np.maximum(final_a, 0.0)

    s = final_a.sum()
    if not np.isfinite(s) or s < 1e-8:
        final_a = np.ones(n_main, dtype=np.float64) / float(max(1, n_main))
    else:
        final_a = final_a / s

    # Optional exact validation; skip for very large outputs to avoid O(n^2) delay.
    if final_a.size <= 12000 and ("evaluate_sequence" in globals()):
        try:
            vals = [float(x) for x in final_a]
            exact = evaluate_sequence(vals)
            if np.isfinite(exact):
                print(f"final exact score: {float(exact):.8f}", flush=True)
                final_a = np.asarray(vals, dtype=np.float64)
                ss = final_a.sum()
                if ss > 1e-8:
                    final_a = final_a / ss
        except Exception:
            pass

    final_list = [float(x) for x in final_a]
    return final_list
