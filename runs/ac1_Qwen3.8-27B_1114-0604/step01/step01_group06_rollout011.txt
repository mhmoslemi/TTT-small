def _init_state(n):
    global _STATE_N, _STATE_NFFT, _STATE_BOUNDS
    _STATE_N = n
    _STATE_NFFT = 1 << (3*n - 1).bit_length()
    _STATE_BOUNDS = [(-25.0, 25.0)] * n

def _softmax(z):
    z = np.asarray(z, dtype=np.float64)
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(z) / z.size
    return e / s

def _fft_conv(a):
    n = _STATE_N
    N = _STATE_NFFT
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1].real
    return np.maximum(c, 0.0)

def _fast_score(a):
    s = a.sum()
    if s < 1e-12:
        return np.inf
    c = _fft_conv(a)
    return float(2.0 * _STATE_N * c.max() / (s*s))

def _obj_grad(z, tau):
    n = _STATE_N
    N = _STATE_NFFT
    a = _softmax(z)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2*n-1].real
    c = np.maximum(c, 0.0)
    
    m = float(c.max())
    diff = np.clip((c - m) / tau, -700.0, 0.0)
    e = np.exp(diff)
    se = float(e.sum())
    if se < 1e-300:
        return 1e10, np.zeros(n)
    p = e / se
    
    fp = np.fft.rfft(p, N)
    h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real
    
    s_h = float(h @ a)
    grad = 2.0 * a * (h - s_h)
    
    val = (m + tau * np.log(se)) * float(n * n)
    grad = grad * float(n * n)
    
    # Tiny regularization
    val += 1e-10 * float(z @ z)
    grad = grad + 1e-10 * z
    
    if not np.isfinite(val):
        return 1e10, np.zeros(n)
    return float(val), grad

def _make_starts(n, rng):
    """Generate diverse starting points."""
    starts = []
    
    # Resample provided sequence if available
    try:
        g = globals().get('height_sequence_1')
        if g is not None:
            g = np.asarray(g, dtype=np.float64)
            if g.size > 1 and np.all(np.isfinite(g)) and g.sum() > 0:
                g = np.maximum(g, 1e-30)
                z_old = np.log(g) - np.log(g).mean()
                x_old = np.linspace(0, 1, g.size)
                z = np.interp(np.linspace(0, 1, n), x_old, z_old)
                a = _softmax(z)
                sc = _fast_score(a)
                starts.append((sc, z - z.mean()))
    except:
        pass
    
    # Linear ramp
    x = np.linspace(0, 1, n)
    for slope in [0.1, 0.3, 0.5, 0.7, 1.0, 1.5, 2.0]:
        a = 1.0 + slope * x
        a = np.maximum(a, 1e-6)
        z = np.log(a) - np.log(a).mean()
        a_s = _softmax(z)
        sc = _fast_score(a_s)
        starts.append((sc, z))
    
    # Smooth step
    for w in [0.05, 0.1, 0.2, 0.3, 0.5]:
        sig = 1.0 / (1.0 + np.exp(-(x - 0.5) / w))
        a = 0.5 + sig
        a = np.maximum(a, 1e-6)
        z = np.log(a)
