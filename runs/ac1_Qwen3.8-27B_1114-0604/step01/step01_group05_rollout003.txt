def fft_convolution(a):
    """Linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)

        return np.array([a[0] * a[0]])
    need = 2 * n - 1
    size = 1
    while size < need:

    return np.maximum(b, 0.0)

    """Project onto {x >= 0, sum(x) = target}."""

# ============================================================
# Core FFT operations
# ============================================================

def fft_autoconv(a):
    """Linear auto-convolution a*a via FFT, clipped to non-negative."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0)
    if n == 1:
        return np.array([a[0] * a[0]])
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def fft_crossconv(a, b):
    """Linear cross-convolution a*b via FFT."""
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n, m = a.size, b.size
    if n == 0 or m == 0:
        return np.empty(0)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    FA = np.fft.rfft(a, size)
    FB = np.fft.rfft(b, size)
    c = np.fft.irfft(FA * FB, size)[:need]
    return c


def fft_crosscorr(a, b):
    """
    Compute s_i = sum_j a_j * b_{i+j} for i = 0..len(a)-1.
    Uses FFT cross-correlation.
    """
    a = np.asarray(a, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)
    n, m = a.size, b.size
    if n == 0 or m == 0:
        return np.zeros(n)
    # s_i = sum_j a_j * b_{i+j}
    # This is the cross-correlation of a and b
    # corr[i] = sum_j a[j] * b[i+j]
    # Equivalent to: reverse a, then convolve with b, and extract
    a_rev = a[::-1].copy()
    c = fft_crossconv(a_rev, b)
    # c[k] = sum_j a_rev[j] * b[k-j] = sum_j a[n-1-j] * b[k-j]
    # Let i = n-1-j, so j = n-1-i
    # c[k] = sum_i a[i] * b[k-n+1+i]
    # We want s_i = sum_j a[j] * b[i+j]
    # Set k-n+1 = i, so k = i + n - 1
    s = c[n - 1 : n - 1 + n]
    return s[:n]


# ============================================================
# Simplex projection and normalization
# ============================================================

def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64).copy()
    n = v.size
    if n == 0:
        return v
    if target <= 0:
        return np.zeros(n)
    if not np.all(np.isfinite(v)):
        return np.full(n, target / n)
    
    # Efficient projection via sorting
    v_sorted = np.sort(v)[::-1]
    cumsum = np.cumsum(v_sorted)
    rng = np.arange(1, n + 1, dtype=np.float64)
    cond = v_sorted > (cumsum - target) / rng
    if not np.any(cond):
        return np.full(n, target / n)
    rho = np.argmax(cond)
    tau = (cumsum[rho] - target) / (rho + 1)
    return np.maximum(v - tau, 0.0)


def normalize_simplex(p):
    """Clip to non-negative and normalize to sum=1."""
    p = np.asarray(p, dtype=np.float64).copy()
    if p.size == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s <= 0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / p.size)
    p /= s
    return p


# ============================================================
# Score computation
# ============================================================

def fast_score(p):
    """Compute 2*n*max(p*p)/(sum(p))^2 using FFT."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    S = p.sum()
    if S <= 0 or not np.isfinite(S):
        return float('inf')
    b = fft_autoconv(p)
    M = b.max()
    if not np.isfinite(M):
        return float('inf')
    return 2.0 * n * M / (S * S)


# ============================================================
# Top-k max and gradient
# ============================================================

def topk_max_and_weights(b, k):
    """
    Return (topk_smooth_max, weights) where weights are concentrated on top-k elements.
    Uses a sharp softmax over the top-k for gradient computation.
    """
    b = np.asarray(b, dtype=np.float64)
    m = b.size
    if m == 0:
        return float('inf'), np.empty(0)
    k = min(k, m)
    
    if k == m:
        # Just use max
        idx = np.argmax(b)
        w = np.zeros(m)
        w[idx] = 1.0
        return float(b[idx]), w
    
    # Find top-k indices
    part = np.argpartition(b, -k)[-k:]
    max_val = float(b[part].max())
    
    if max_val <= 0:
        w = np.ones(m) / m
        return 0.0, w
    
    # Sharp softmax over top-k
    top_vals = b[part]
    diff = (top_vals - max_val) / max_val
    diff = np.clip(diff, -100, 0)
    exp_diff = np.exp(diff * 50.0)  # Sharp: effectively picks the max
    Z = exp_diff.sum()
    if Z <= 0:
        exp_diff = np.zeros(k)
        exp_diff[np.argmax(top_vals)] = 1.0
        Z = 1.0
    w_top = exp_diff / Z
    
    # Smoothed max estimate
    smooth_max = max_val  # We use the actual max for the score
    
    # Full weight vector
    w_full = np.zeros(m)
    w_full[part] = w_top
    return smooth_max, w_full


# ============================================================
# Block operations
# ============================================================

def block_to_p(n, B, c):
    """Convert block log-weights c (length B-1) to full sequence of length n."""
    L = n // B
    if L == 0:
        return np.full(n, 1.0 / n)
    z = np.zeros(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        m = min(B - 1, c.size)
        z[:m] = c[:m]
    z = z - z.max()
    wb = np.exp(z)
    s = wb.sum()
    if s <= 0 or not np.isfinite(s):
        wb = np.ones(B) / B
    else:
        wb /= s
    H = wb / L
    return np.repeat(H, L)


def p_to_block(p, B):
    """Extract block log-weights from full sequence p."""
    n = p.size
    L = n // B
    if L == 0:
        return np.zeros(max(0, B - 1))
    block_means = p.reshape(B, L).mean(axis=1)
    block_means = np.maximum(block_means, 1e-15)
    log_w = np.log(block_means)
    c = log_w[:-1] - log_w[-1]
    return c


def block_objective(c, n, B):
    """Objective: n * max(block_to_p(n,B,c) * block_to_p(n,B,c))."""
    p = block_to_p(n, B, c)
    b = fft_autoconv(p)
    M = b.max()
    if not np.isfinite(M):
        return float('inf'), np.zeros(max(0, B - 1))
    # Simple gradient: perturb each c_i
    eps = 1e-5
    grad = np.zeros(max(0, B - 1))
    base = n * M
    for i in range(len(grad)):
        c_plus = c.copy()
        c_plus[i] += eps
        sc_plus = n * fft_autoconv(block_to_p(n, B, c_plus)).max()
        c_minus = c.copy()
        c_minus[i] -= eps
        sc_minus = n * fft_autoconv(block_to_p(n, B, c_minus)).max()
        grad[i] = (sc_plus - sc_minus) / (2 * eps)
    return base, grad


def block_objective_fast(c, n, B):
    """Faster objective using analytic gradient via top-1."""
    p = block_to_p(n, B, c)
    b = fft_autoconv(p)
    M = b.max()
    if not np.isfinite(M):
        return float('inf'), np.zeros(max(0, B - 1))
    # Find argmax of b
    k_star = np.argmax(b)
    # d(b_{k_star})/d(p_i) = p_{k_star - i} (for valid indices)
    # Gradient of M w.r.t. p: g_i = p_{k_star - i} (clipped)
    g = np.zeros(n)
    lo = max(0, k_star - n + 1)
    hi = min(n - 1, k_star)
    for i in range(lo, hi + 1):
        g[i] = p[k_star - i]
    # But we need d(M)/d(c_j) = sum over i in block j of g_i * (dp/dc)_j
    # dp/dc is complicated for the softmax block structure.
    # Use numerical gradient instead (faster for small B)
    return float(n * M), None


def run_block_lbfgs(c0, n, B, max_iter=200, time_limit=30.0):
    """Run L-BFGS-B on block representation with numerical gradients."""
    if not HAVE_SCIPY:
        return c0
    c0 = np.asarray(c0, dtype=np.float64).reshape(-1)
    c0 = np.clip(c0, -8.0, 8.0)
    start = time.monotonic()
    
    def obj_and_grad(c):
        if time.monotonic() - start > time_limit:
            return float('inf'), np.zeros_like(c)
        c = np.clip(c, -8.0, 8.0)
        p = block_to_p(n, B, c)
        b = fft_autoconv(p)
        M = b.max()
        if not np.isfinite(M):
            return float('inf'), np.zeros_like(c)
        base = n * M
        # Numerical gradient
        m = c.size
        if m == 0:
            return base, np.zeros(0)
        eps = 0.01
        grad = np.zeros(m)
        for i in range(m):
            if time.monotonic() - start > time_limit:
                return base, np.zeros_like(c)
            cp = c.copy(); cp[i] += eps
            cm = c.copy(); cm[i] -= eps
            sp = n * fft_autoconv(block_to_p(n, B, cp)).max()
            sm = n * fft_autoconv(block_to_p(n, B, cm)).max()
            grad[i] = (sp - sm) / (2 * eps)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return base, grad
    
    bounds = [(-8.0, 8.0)] * c0.size
    try:
        res = minimize(obj_and_grad, c0, method='L-BFGS-B', jac=True,
                       bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-10})
        c = np.asarray(res.x, dtype=np.float64)
    except Exception:
        c = c0
    c = np.clip(c, -8.0, 8.0)
    c = np.where(np.isfinite(c), c, 0.0)
    return c

# Nesterov accelerated projected gradient
# ============================================================

def nesterov_refine(p0, deadline, k_top=50, max_iter=500000):
    """
    Nesterov-accelerated projected gradient on the simplex.
    Uses top-k max for the objective gradient.
    """
    p = normalize_simplex(p0.copy())
    n = p.size
    if n == 0:
        return p, float('inf')
    
    # Break symmetry
    if n > 1 and np.std(p) < 1e-14:
        t = np.arange(n, dtype=np.float64)
        p = normalize_simplex(p * (1.0 + 0.01 * np.sin(0.1 * t) + 0.005 * np.cos(0.3 * t)))
    
    best_p = p.copy()
    best_score = fast_score(p)
    
    # Nesterov momentum
    y = p.copy()
    gamma = 1.0  # momentum parameter: t_k = 2/(k+2) or constant
    step = 5.0
    no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() >= deadline:
            break
        
        # Compute gradient at y
        b = fft_autoconv(y)
        M, w = topk_max_and_weights(b, k_top)
        
        # Gradient of max(b) w.r.t. p: g_i = sum_k w_k * p_{k-i}
        # = cross-correlation of w with p
        g = fft_crosscorr(p, w)
        g = np.maximum(g, 0.0)
        g = 2.0 * g  # factor of 2 from auto-conv
        # Project gradient: subtract mean to stay on simplex
        grad = g - g.mean()
        
        # Nesterov update
        t_k = 2.0 / (it + 2.0)
        y = p + (t_k / (it * t_k + 1.0) - 1.0) * (p - y + p) if it > 0 else p.copy()
        # Standard Nesterov: y = p + alpha*(p - y_prev) where alpha = (t_{k-1}-1)/(t_k)
        # Simplified: use constant momentum
        if it > 0:
            y = p + 0.8 * (p - y)  # momentum 0.8
            y = np.maximum(y, 0)
            ys = y.sum()
            if ys > 0:
                y = y / ys  # Keep on simplex for evaluation
        
        # Recompute gradient at y for Nesterov
        if it > 0:
            b_y = fft_autoconv(y)
            _, w_y = topk_max_and_weights(b_y, k_top)
            g_y = fft_crosscorr(y, w_y)
            g_y = np.maximum(g_y, 0.0)
            grad_y = 2.0 * g_y - (2.0 * g_y).mean()
        else:
            grad_y = grad
        
        # Line search
        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.01):
            alpha = step * mult
            q = project_simplex(p - alpha * grad_y, 1.0)
            sc = fast_score(q)
            if np.isfinite(sc) and sc <= best_score + 0.1 * (fast_score(p) - best_score + 1e-10):
                # Accept if smooth objective improves
                b_q = fft_autoconv(q)
                M_q = b_q.max()
                b_p = fft_autoconv(p)
                M_p = b_p.max()
                if n * M_q <= n * M_p + 1e-12:
                    p = q
                    accepted = True
                    break
        
        if accepted:
            if step < 10.0:
                step = min(step * 1.1, 50.0)
        else:
            step *= 0.5
            if step < 1e-6:
                break
        
        # Track best
        sc = fast_score(p)
        if sc < best_score - 1e-12:
            best_score = sc
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            break
        
        if it % 1000 == 0:
            print(f"  nesterov it={it} best_score={best_score:.8f} step={step:.4f}")
    
    return best_p, best_score


# ============================================================
# Soft LP direction (multiplicative weights)
# ============================================================

def soft_lp_direction(f, M, max_iters=500, eta=0.3):
    """
    Find non-negative b maximizing sum(b) s.t. (f*b)_k <= M.
    Uses multiplicative weights (exponential weights) method.
    Returns normalized b.
    """
    n = f.size
    b = np.ones(n) / n
    f = normalize_simplex(f)
    
    for t in range(max_iters):
        # Compute f * b
        fb = fft_crossconv(f, b)
        # Violations
        viol = np.maximum(fb - M, 0.0)
        if viol.max() < 1e-10:
            break
        # Loss for each b_i: sum_k viol[k] * f[k-i] (cross-correlation)
        loss = fft_crosscorr(b, viol)
        loss = np.maximum(loss, 0.0)
        loss_max = loss.max()
        if loss_max < 1e-15:
            break
        # Multiplicative update
        b = b * np.exp(-eta * loss / loss_max)
        b = np.maximum(b, 1e-15)
        b_sum = b.sum()
        if b_sum <= 0 or not np.isfinite(b_sum):
            b = np.ones(n) / n
        else:
            b /= b_sum
    
    return b

# Asymmetric initializations
# ============================================================

def generate_asymmetric_starts(n, rng, count=12):
    """Generate diverse asymmetric starting points."""
    starts = []
    t = np.linspace(0.0, 1.0, n)
    
    # Linear ramps (various slopes)
    for slope in (-3.0, -2.0, -1.0, 1.0, 2.0, 3.0):
        z = slope * (t - 0.5)
        p = np.exp(z)
        p = normalize_simplex(p)
        starts.append(p)
    
    # One-sided bumps
    for center in (0.3, 0.7):
        for width in (0.1, 0.2, 0.3):
            z = -((t - center) / width) ** 2
            p = np.exp(z)
            p = normalize_simplex(p)
            starts.append(p)
    
    # Sine-based
    for freq in (1.0, 2.0, 3.0):
        z = np.sin(2 * np.pi * freq * t + 0.5) 
        p = np.exp(1.5 * z)
        p = normalize_simplex(p)
        starts.append(p)
    
    # Random smooth
    for _ in range(3):
        size = 1
        while size < n:
            size <<= 1
        zc = rng.normal(size=size) + 1j * rng.normal(size=size)
        z = np.real(np.fft.ifft(zc))[:n]
        z = z - z.mean()
        z = 2.0 * np.tanh(z)
        p = np.exp(z)
        p = normalize_simplex(p)
        starts.append(p)
    
    # Truncated / two-level
    for frac in (0.3, 0.4, 0.6, 0.7):
        p = np.ones(n)
        p[:int(frac * n)] = 1.5
        p[int(frac * n):] = 0.5
        p = normalize_simplex(p)
        starts.append(p)
    
    return starts[:count]

def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing the evaluation function.
    Returns a list of floats.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0
    
    if budget_s <= 0:
        return [1.0]
    
    start_time = time.monotonic()
    reserve = 60.0
    deadline = start_time + max(30.0, budget_s - reserve)
    
    N = 8192
    rng = np.random.default_rng(seed)
    
    print(f"Starting search with N={N}, budget={budget_s:.0f}s, deadline in {deadline-start_time:.0f}s")
    
    # Get previous best if available
    try:
        prev = np.asarray(height_sequence_1, dtype=np.float64)
        if prev.size == N and np.all(prev >= 0) and np.isfinite(prev).all():
            prev = normalize_simplex(prev)
            prev_score = fast_score(prev)
            print(f"Previous best score: {prev_score:.8f}")
        else:
            prev = None
    except Exception:
        prev = None
    
    # ---- Generate diverse starts ----
    all_starts = generate_asymmetric_starts(N, rng, count=16)
    if prev is not None:
        all_starts.insert(0, prev)
    # Also add uniform as safety
    all_starts.append(np.full(N, 1.0 / N))
    
    # Score all starts
    scored = []
    for i, p in enumerate(all_starts):
        sc = fast_score(p)
        if np.isfinite(sc):
            scored.append((sc, i, p))
            print(f"  start {i}: score={sc:.8f}")
    
    scored.sort(key=lambda x: x[0])
    
    best_p = scored[0][2].copy() if scored else np.full(N, 1.0 / N)
    best_score = scored[0][0] if scored else 2.0
    print(f"Best initial: score={best_score:.8f}")
    
    # ---- Phase 1: Block optimization on top starts ----
    phase1_deadline = start_time + 0.35 * budget_s
    
    top_starts = [p for _, _, p in scored[:5]]
    
    for si, p0 in enumerate(top_starts):
        if time.monotonic() >= phase1_deadline:
            break
        remaining = phase1_deadline - time.monotonic()
        if remaining < 30:
            break
        
        print(f"\nPhase 1: Block optimization from start {si} (remaining {remaining:.0f}s)")
        
        # Coarse to fine blocks
        for B in (32, 64, 128, 256):
            if time.monotonic() >= phase1_deadline:
                break
            if not (N % B == 0) or B >= N:
                continue
            
            c0 = p_to_block(p0, B)
            time_per_B = min(30.0, remaining / 3.0)
            c_opt = run_block_lbfgs(c0, N, B, max_iter=100, time_limit=time_per_B)
            p_opt = block_to_p(N, B, c_opt)
            sc = fast_score(p_opt)
            
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_p = p_opt.copy()
                print(f"  B={B}: score={sc:.8f} (improved)")
            
            # Use best so far for next B
            p0 = best_p.copy() if sc < fast_score(p0) else p0
    
    print(f"\nAfter Phase 1: best_score={best_score:.8f}")
    
    # ---- Phase 2: Nesterov full-length refinement ----
    phase2_deadline = start_time + 0.70 * budget_s
    
    # Use top-2 best starts for Nesterov
    nesterov_starts = []
    if best_p is not None:
        nesterov_starts.append(best_p.copy())
    for sc, i, p in scored[:3]:
        if not any(np.allclose(p, ns) for ns in nesterov_starts):
            nesterov_starts.append(p.copy())
    
    for ni, p0 in enumerate(nesterov_starts[:3]):
        if time.monotonic() >= phase2_deadline:
            break
        remaining = phase2_deadline - time.monotonic()
        if remaining < 60:
            break
        
        frac = (0.5, 0.3, 0.2)[ni] if ni < 3 else 0.1
        alloc = remaining * frac
        sub_dl = time.monotonic() + alloc
        
        print(f"\nPhase 2: Nesterov from start {ni} (alloc {alloc:.0f}s)")
        p_fit, sc_fit = nesterov_refine(p0, sub_dl, k_top=30)
        
        if np.isfinite(sc_fit) and sc_fit < best_score:
            best_score = sc_fit
            best_p = p_fit.copy()
            print(f"  Nesterov {ni}: score={sc_fit:.8f} (improved)")
    
    print(f"\nAfter Phase 2: best_score={best_score:.8f}")
    
    # ---- Phase 3: Soft-LP direction + mixing ----
    phase3_deadline = start_time + 0.90 * budget_s
    
    if np.isfinite(best_score):
        print(f"\nPhase 3: Soft-LP direction finding")
        f = normalize_simplex(best_p)
        b_conv = fft_autoconv(f)
        M_val = b_conv.max()
        
        # Try multiple LP directions
        for lp_iter in range(3):
            if time.monotonic() >= phase3_deadline:
                break
            remaining = phase3_deadline - time.monotonic()
            if remaining < 30:
                break
            
            eta = 0.2 * (lp_iter + 1)
            b_dir = soft_lp_direction(f, M_val, max_iters=200, eta=eta)
            
            # Mix: f_new = (1-t)*f + t*b_dir, several t values
            for t in (0.1, 0.2, 0.3, 0.5):
                f_mix = (1 - t) * f + t * b_dir
                f_mix = normalize_simplex(f_mix)
                sc_mix = fast_score(f_mix)
                if np.isfinite(sc_mix) and sc_mix < best_score:
                    best_score = sc_mix
                    best_p = f_mix.copy()
                    print(f"  LP mix t={t}: score={sc_mix:.8f} (improved)")
            
            # Update f for next LP
            f = normalize_simplex(best_p)
            b_conv = fft_autoconv(f)
            M_val = b_conv.max()
        
        # ---- Phase 3b: Short Nesterov polish from LP result ----
        if time.monotonic() < phase3_deadline - 30:
            p_fit, sc_fit = nesterov_refine(best_p, phase3_deadline, k_top=20)
            if np.isfinite(sc_fit) and sc_fit < best_score:
                best_score = sc_fit
                best_p = p_fit.copy()
                print(f"  LP+Nesterov: score={sc_fit:.8f} (improved)")
    
    print(f"\nAfter Phase 3: best_score={best_score:.8f}")
    
    # ---- Phase 4: Final polish ----
    phase4_deadline = deadline
    
    if time.monotonic() < phase4_deadline - 20:
        p_fit, sc_fit = nesterov_refine(best_p, phase4_deadline, k_top=10)
        if np.isfinite(sc_fit) and sc_fit < best_score:
            best_score = sc_fit
            best_p = p_fit.copy()
            print(f"\nFinal polish: score={sc_fit:.8f}")
    
    # ---- Final cleanup and official check ----
    final_p = normalize_simplex(best_p)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_seq = [float(x) for x in final_p]
    
    # Official evaluation
    try:
        off_score = evaluate_sequence(final_seq)
        off_score = float(off_score)
        if np.isfinite(off_score):
            print(f"\nOfficial score: {off_score:.8f}")
            if off_score <= 2.0 + 1e-6:  # Better than or equal to uniform
                return final_seq
    except Exception as e:
        print(f"Evaluation error: {e}")
    
    # Fallback: uniform
    print("Falling back to uniform")
    return [1.0] * N
