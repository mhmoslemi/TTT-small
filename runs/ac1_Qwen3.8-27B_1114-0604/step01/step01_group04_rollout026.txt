"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using LP-based Frank-Wolfe method,
coordinate descent polishing, and multi-scale optimization.
"""

import numpy as np
import time
import scipy.optimize as opt

def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))

def smooth_grad_single(a, b, tau):
    """Smoothed subgradient using softmax weights."""
    n = a.size
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(b.size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= s
    g = 2.0 * np.correlate(w, a, mode='valid')
    return g

def fw_lp_step(f, M):
    """
    Solve the LP: maximize sum(g) subject to g>=0 and (f*g)_k <= M for all k.
    Returns the optimal g (unnormalized).
    """
    n = f.size
    # Build constraint matrix A where A[k,j] = f[k-j] if 0<=k-j<=n-1 else 0
    # (f*g)_k = sum_j A[k,j] * g_j
    # A has shape (2n-1, n)
    
    # For memory efficiency, we could use the structure, but for n up to ~2048 it's fine
    # A[k,j] = f[k-j] for valid (k,j)
    # k ranges 0..2n-2, j ranges 0..n-1
    # k-j ranges k-(n-1) .. k, and we need 0 <= k-j <= n-1
    
    # Build A using the Toeplitz structure
    # Row k: A[k, max(0,k-n+1):min(k,n-1)+1] = f[0:min(k,n-1)+1 - max(0,k-n+1)]
    # This is getting complex. Let me just build it directly.
    
    # Actually, (f*g)_k = sum_{j} f_{k-j} * g_j where 0 <= k-j <= n-1 and 0 <= j <= n-1
    # = sum_{j=max(0,k-n+1)}^{min(k,n-1)} f_{k-j} * g_j
    
    # For the LP, I'll build the matrix using scipy sparse for larger n
    # For now, let's do it directly for n <= 2048
    
    # A is (2n-1) x n
    # Use the fact that it's a convolution Toeplitz matrix
    # Column j of A has f[0:n] shifted down by j positions (in the top 2n-1 rows)
    # Actually: A[k,j] = f[k-j] if 0<=k-j<n, i.e., j<=k<j+n
    
    # Let me just build it.
    A = np.zeros((2*n-1, n))
    for j in range(n):
        # A[k,j] = f[k-j] for k from j to j+n-1
        A[j:j+n, j] = f
    
    # LP: max c^T g s.t. A g <= M (scalar), g >= 0
    # => min -c^T g s.t. A g <= M, -g <= 0
    
    c = -np.ones(n)  # minimize -sum(g)
    A_ub = np.vstack([A, -np.eye(n)])
    b_ub = np.concatenate([np.full(2*n-1, M), np.zeros(n)])
    
    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, method='highs', options={'presolve': True})
    
    if res.success:
        return res.x
    else:
        return None

def fw_optimize(f_init, time_budget, rng, n_fw_steps=None):
    """
    Frank-Wolfe optimization using LP for the linearization step.
    f_init: initial point on simplex (sum=1, non-negative)
    Returns (best_f, best_score).
    """
    n = f_init.size
    f = f_init.astype(np.float64).copy()
    f = project_simplex(f)
    
    best_f = f.copy()
    best_score = normed_score(f)
    
    if n_fw_steps is None:
        n_fw_steps = 10000
    
    t0 = time.monotonic()
    for it in range(n_fw_steps):
        if time.monotonic() - t0 > time_budget:
            break
        
        M = float(np.max(full_conv(f)))
        
        g = fw_lp_step(f, M)
        if g is None:
            # Fallback to subgradient
            b = full_conv(f)
            tau = 0.05 * M
            grad = smooth_grad_single(f, b, tau)
            g = np.maximum(-grad, 0)
        
        g_sum = g.sum()
        if g_sum < 1e-10:
            break
        g = g / g_sum  # normalize to simplex
        
        # FW update: f <- (1-t)*f + t*g
        # Optimal t for FW is 2/(it+2) or we can use line search
        t = 2.0 / (it + 2)
        
        f_new = (1 - t) * f + t * g
        f_new = project_simplex(f_new)  # should already be on simplex
        
        sc = normed_score(f_new)
        f = f_new
        
        if sc < best_score:
            best_score = sc
            best_f = f.copy()
    
    return best_f, best_score

"""

1. LP-based Frank-Wolfe method (novel primary optimization)
2. Averaged FW directions for faster convergence
3. Line search for optimal step size

import numpy as np
import time
from scipy.optimize import linprog

def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    v = v.astype(np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))

def smooth_grad_single(a, b, tau):
    """Smoothed subgradient using softmax weights at one temperature."""
    n = a.size
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(b.size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= s
    g = 2.0 * np.correlate(w, a, mode='valid')
    return g

def multi_scale_grad(a, b, tau_base, n_scales):
    """Average subgradients at multiple temperature scales."""
    g_total = np.zeros(a.size)
    for s in range(n_scales):
        tau = tau_base * (2.0 ** s)
        g_total += smooth_grad_single(a, b, tau)
    return g_total / n_scales

def fw_lp_step(f, M, use_sparse=False):
    """
    Solve LP: maximize sum(g) subject to g>=0 and (f*g)_k <= M for all k.
    Returns optimal g vector (non-negative, unnormalized).
    """
    n = f.size
    # Build convolution constraint matrix: A[k,j] = f[k-j] if 0<=k-j<n else 0
    # Shape: (2n-1) x n
    if use_sparse and n > 512:
        import scipy.sparse as sp
        rows = []
        cols = []
        vals = []
        for j in range(n):
            # Column j: A[j+j_row, j] = f[j_row] for j_row=0..n-1
            row_indices = np.arange(j, j + n)
            rows.extend(row_indices.tolist())
            cols.extend([j] * n)
            vals.extend(f.tolist())
        A = sp.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
        A_ub = A  # just the convolution constraints; handle g>=0 via bounds
    else:
        A = np.zeros((2*n-1, n))
        for j in range(n):
            A[j:j+n, j] = f
    
    c_vec = -np.ones(n)  # minimize -sum(g)
    
    if use_sparse and n > 512:
        import scipy.sparse as sp
        # Use bounds for g >= 0
        bounds = [(0, None)] * n
        res = linprog(c_vec, A_ub=A_ub, b_ub=np.full(2*n-1, M), 
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
    else:
        # Add g >= 0 as -g <= 0
        A_ub_full = np.vstack([A, -np.eye(n)])
        b_ub_full = np.concatenate([np.full(2*n-1, M), np.zeros(n)])
        res = linprog(c_vec, A_ub=A_ub_full, b_ub=b_ub_full,
                      method='highs', options={'presolve': True,
                      'dual_feasibility_tolerance': 1e-9})
    
    if res.success:
        return np.maximum(res.x, 0.0)
    else:
        return None

def fw_line_search(f, g, M, n):
    """
    Find optimal t in [0,1] minimizing score of (1-t)*f + t*g.
    Uses golden section search (score is convex in t).
    """
    def score_at_t(t):
        ft = (1.0 - t) * f + t * g
        b = full_conv(ft)
        return 2.0 * n * float(np.max(b))
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = score_at_t(c)
    fd = score_at_t(d)
    
    for _ in range(30):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = score_at_t(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = score_at_t(d)
    
    t_opt = (a + b) / 2.0
    return t_opt, score_at_t(t_opt)

    Frank-Wolfe optimization using LP for the direction-finding step.
    Uses averaged directions for faster convergence.
    """
    n = f_init.size

    best_f = f.copy()
    best_score = normed_score(f)
    
    use_sparse = n > 512

    last_improve = 0
    stall_counter = 0

        g = fw_lp_step(f, M, use_sparse)
        if g is None:

            b = full_conv(f)
            tau = 0.05 * M
            grad = smooth_grad_single(f, b, tau)
            g = np.maximum(-grad, 0.0)
        
        g_sum = g.sum()
        if g_sum < 1e-12:
            # LP gave trivial solution, try perturbation
            g = f + 0.1 * rng.standard_normal(n)
            g = project_simplex(g)
            g_sum = 1.0
        
        g = g / g_sum  # normalize to simplex
        
        # Update averaged direction
        dir_history.append(g)
        if len(dir_history) > avg_window:
            dir_history.pop(0)
        vel_avg = np.mean(dir_history, axis=0)

        else:

    print(f"[plan] n_list={n_list}, total_time={total_time:.1f}s")
    
    # Time allocation: give more time to promising n values
    # Smaller n is faster per iteration; larger n has more room for improvement
    # Weight: prefer n around 128-512 where FW converges well
    raw_w = []
    for nv in n_list:
        if nv == 128:
            w = 3.0
        elif nv == 256:
            w = 2.5
        elif nv == 512:
            w = 2.0
        elif nv == 1024:
            w = 1.5
        else:
            w = 1.0
        raw_w.append(w)
    total_w = sum(raw_w)
    weights = [w / total_w for w in raw_w]
    
    fw_time_fraction = 0.70  # 70% for FW, 30% for polishing
    
    for idx_n, nv in enumerate(n_list):
        if time.monotonic() > deadline:
            break
        
        time_n = weights[idx_n] * total_time * fw_time_fraction
        if time_n < 5.0:
            continue
        
        elapsed_n = 0.0
        inits = get_inits(nv, rng)
        
        # If we have previous best of this length, use it first
        if prev_a is not None and prev_a.size == nv:
            inits.insert(0, prev_a.copy())
        
        n_inits = len(inits)
        # Allocate time per init, starting with more for first few
        times_per_init = []
        remaining = time_n
        for ii in range(n_inits):
            frac = 0.4 if ii == 0 else (0.3 / max(1, n_inits - 1) if ii == 1 else 
                                        (remaining - 0.3 * remaining / max(1, n_inits - ii)) / max(1, n_inits - ii))
            # Simpler: give 40% to first, split rest equally
            if ii == 0:
                t_alloc = time_n * 0.40
            else:
                t_alloc = time_n * 0.60 / (n_inits - 1)
            times_per_init.append(min(t_alloc, remaining))
            remaining -= t_alloc
        # Adjust last to use remaining
        times_per_init[-1] = max(3.0, times_per_init[-1] + max(0, remaining))
        
        print(f"[plan] n={nv}: {n_inits} inits, time={time_n:.1f}s")
        
        for ii, a_init in enumerate(inits):
            if time.monotonic() > deadline:
                break
            
            tb = min(times_per_init[ii], deadline - time.monotonic() - 2.0)
            if tb < 3.0:
                break
            
            print(f"[fw] n={nv}, init={ii}, time_budget={tb:.1f}s")
            t_start = time.monotonic()
            
            a_best, sc_best = fw_optimize(a_init, tb, rng, avg_window=30)
            
            elapsed_n = time.monotonic() - t_start
            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = a_best.copy()
                print(f"  [fw] n={nv}, init={ii}: score={sc_best:.6f} *** NEW BEST ***")
            else:
                print(f"  [fw] n={nv}, init={ii}: score={sc_best:.6f} (best so far: {best_overall_sc:.6f})")
            
            # If we beat 1.51, try polishing immediately
            if sc_best < 1.51 and time.monotonic() < deadline - 30:
                pol_time = min(60.0, deadline - time.monotonic() - 10.0)
                a_pol, sc_pol = subgrad_polish(a_best, pol_time, rng)
                if sc_pol < best_overall_sc:
                    best_overall_sc = sc_pol
                    best_overall = a_pol.copy()
                    print(f"  [polish] improved to {sc_pol:.6f} *** NEW BEST ***")
    
    # Phase 2: Polish the best result
    if best_overall is not None and time.monotonic() < deadline - 10.0:
        remain = deadline - time.monotonic() - 5.0
        n_best = best_overall.size
        print(f"[polish] n={n_best}, remaining={remain:.1f}s")
        
        # Subgradient polish
        pol_time = min(remain * 0.5, 300.0)
        a_p, sc_p = subgrad_polish(best_overall, pol_time, rng)
        if sc_p < best_overall_sc:
            best_overall_sc = sc_p
            best_overall = a_p.copy()
            print(f"[polish] subgrad: {sc_p:.6f}")
        
        # Another FW round on the polished result
        if time.monotonic() < deadline - 20.0:
            remain2 = deadline - time.monotonic() - 5.0
            fw_time2 = min(remain2 * 0.6, 200.0)
            if fw_time2 > 10.0:
                try:
                    a_fw2, sc_fw2 = fw_optimize(best_overall, fw_time2, rng, avg_window=20)
                    if sc_fw2 < best_overall_sc:
                        best_overall_sc = sc_fw2
                        best_overall = a_fw2.copy()
                        print(f"[polish] FW round 2: {sc_fw2:.6f} *** NEW BEST ***")
                    else:
                        print(f"[polish] FW round 2: {sc_fw2:.6f} (no improvement)")
                except Exception as e:
                    print(f"[polish] FW round 2 error: {e}")
        
        # Final short subgradient polish
        if time.monotonic() < deadline - 5.0:
            remain3 = min(deadline - time.monotonic() - 3.0, 60.0)
            if remain3 > 5.0:
                a_p3, sc_p3 = subgrad_polish(best_overall, remain3, rng)
                if sc_p3 < best_overall_sc:
                    best_overall_sc = sc_p3
                    best_overall = a_p3.copy()
                    print(f"[polish] final: {sc_p3:.6f}")
    
    # Fallback
    if best_overall is None:
        best_overall = np.ones(128) / 128
        best_overall_sc = 2.0
    
    # Final verification
    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"[final] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")
    
    return result
