```python
"""
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS with softmax
parameterization, annealed smooth-max objective, LP direction kicks,
and diverse initializations across multiple sequence lengths.
"""

import numpy as np
import time
from scipy.optimize import minimize
from scipy import sparse
from scipy.optimize import linprog


def full_conv(a):
    """Full linear convolution of a with itself via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Score = 2*n*max(conv(a,a)) when sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def corr_fft(w, a):
    """
    Compute corr_i = sum_k w[k] * a[k-i] for i = 0, ..., n-1.
    w has length 2n-1, a has length n.
    Uses: corr_i = conv(w, a[::-1])[n-1+i]
    """
    n = a.size
    total_len = (2 * n - 1) + n - 1  # = 3n-2
    F = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, F)
    A = np.fft.rfft(a[::-1], F)
    conv_full = np.fft.irfft(W * A, F)[:total_len]
    return conv_full[n - 1: 2 * n - 1]


def grad_autocorr(a, w):
    """
    Gradient of sum_k w[k] * (conv(a,a))_k w.r.t. a.
    w has length 2n-1, a has length n.
    """
    n = a.size
    corr = corr_fft(w, a)
    grad = 2.0 * corr
    # Subtract diagonal correction
    for i in range(n):
        if 2 * i < 2 * n - 1:
            grad[i] -= w[2 * i] * a[i]
    return grad


def softmax(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    return e / e.sum()


def solve_lp(a, M):
    """
    Solve: max sum(g) s.t. conv(a,g)_k <= M for all k, g >= 0.
    Returns the optimal g vector (length n).
    """
    n = a.size
    # Build constraint matrix A where A[k, j] = a[k-j] if 0<=k-j<n else 0
    # A is (2n-1) x n
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b = M * np.ones(2 * n - 1)
    c = -np.ones(n)  # minimize -sum(g) = maximize sum(g)
    
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if res.success:
```

```python
Minimizes 2*n*max(conv(a,a))/sum(a)^2 using L-BFGS with softmax parameterization,

def full_conv(a):
    """Full linear convolution of a with itself via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fspace if False else fsize)[:size]


def _full_conv(a):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Score = 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    b = _full_conv(a)
    return 2.0 * n * float(np.max(b))


def softmax_fn(z):
    """Numerically stable softmax."""
    z = z - z.max()
    e = np.exp(z)
    s = e.sum()
    if s < 1e-300:
        out = np.zeros_like(z)
        out[z.argmax()] = 1.0
        return out
    return e / s


def corr_fft(w, a):
    """
    Compute corr_i = sum_k w[k] * a[k-i] for i = 0, ..., n-1.
    w has length 2n-1, a has length n.
    Uses convolution: corr_i = conv(w, a[::-1])[n-1+i]
    """
    n = a.size
    total_len = (2 * n - 1) + n - 1
    F = 1 << (total_len - 1).bit_length()
    W = np.fft.rfft(w, F)
    A = np.fft.rfft(a[::-1], F)
    conv_full = np.fft.irfft(W * A, F)[:total_len]
    return conv_full[n - 1: 2 * n - 1]


def grad_autocorr(a, w):
    """
    Gradient of sum_k w[k] * (conv(a,a))_k w.r.t. a.
    w has length 2n-1, a has length n.
    """
    n = a.size
    corr = corr_fft(w, a)
    grad = 2.0 * corr
    # Subtract diagonal correction where k=2i
    max_i = min(n, (2 * n - 1 + 1) // 2)
    for i in range(max_i):
        grad[i] -= w[2 * i] * a[i]
    return grad


def _lbgfs_objgrad(zv, T_frac, max_b_ref):
    """
    Top-level objective+gradient for L-BFGS with softmax parameterization.
    Returns (f, grad_z) where f is the smoothed max of conv(a,a).
    """
    av = softmax_fn(zv)
    b = _full_conv(av)
    T_eff = T_frac * max_b_ref
    if T_eff < 1e-15:
        T_eff = 1e-15
    Tb = T_eff * b
    Tb_shift = Tb - Tb.max()
    exp_Tb = np.exp(Tb_shift)
    s = exp_Tb.sum()
    if s < 1e-300:
        f = float(b.max())
        w = np.zeros_like(b)
        w[np.argmax(b)] = 1.0
    else:
        f = float((Tb_shift.max() + np.log(s) + Tb.max()) / T_eff)
        w = exp_Tb / s
    g = grad_autocorr(av, w)
    grad_z = av * (g - np.dot(av, g))
    return f, grad_z

    return partial(_lbgfs_objgrad, T_frac=T_frac, max_b_ref=max_b_ref)


def solve_lp(a, M):
    """
    Solve: max sum(g) s.t. conv(a,g)_k <= M for all k, g >= 0.
    Returns optimal g vector (length n) or None.
    """
    n = a.size
    rows = []
    cols = []
    data = []
    for k in range(2 * n - 1):
        j_min = max(0, k - n + 1)
        j_max = min(k, n - 1)
        for j in range(j_min, j_max + 1):
            rows.append(k)
            cols.append(j)
            data.append(a[k - j])
    A_mat = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_vec = M * np.ones(2 * n - 1)
    c_vec = -np.ones(n)
    bounds = [(0.0, None)] * n
    res = linprog(c_vec, A_ub=A_mat, b_ub=b_vec, bounds=bounds, method='highs',
                  options={'presolve': True})
    if res.success:
        return res.x
    return None

def exact_subgrad_polish(a_init, time_budget, rng):

def init_u_shape(n):
    """U-shaped initialization: high at edges, low in middle."""
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_u_shape_power(n, p):
    """U-shaped with power parameter."""
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i) + 1.0
    a = m ** p  # high at edges for positive p
    return a / a.sum()

        n_list = sorted(set(n_list))

    total_time = max(1.0, deadline - time.monotonic())
    print(f"[plan] n_list={n_list}, total_time={total_time:.1f}s")

    # Time allocation: give more time to n values near the previous best
    raw_w = []
    for nv in n_list:
        w = 1.0 / (0.3 + nv / 256.0)
        if prev_a is not None and nv == prev_a.size:
            w *= 1.5
        raw_w.append(w)
    total_w = sum(raw_w)
    weights = [w / total_w for w in raw_w]

    all_solutions = []  # (score, n, a) for tracking

    for idx_n, nv in enumerate(n_list):
        if time.monotonic() > deadline - 60:
            break

        time_n = weights[idx_n] * total_time * 0.80
        inits = get_inits(nv, rng)

        # Add previous best (resized) if applicable
        if prev_a is not None and prev_a.size != nv:
            # Resize via interpolation
            from numpy import interp
            old_n = prev_a.size
            new_n = nv
            old_x = np.linspace(0, 1, old_n)
            new_x = np.linspace(0, 1, new_n)
            resized = interp(new_x, old_x, prev_a)
            resized = np.maximum(resized, 0)
            resized = resized / resized.sum()
            inits.insert(0, resized)
        elif prev_a is not None and prev_a.size == nv:
            inits.insert(0, prev_a.copy())

        inits_per = max(8.0, time_n / len(inits))
        print(f"[n={nv}] {len(inits)} inits, ~{inits_per:.1f}s each")

        for ii, a_init in enumerate(inits):
            if time.monotonic() > deadline - 60:
                break

            tb = min(inits_per, deadline - time.monotonic() - 60)
            if tb < 5.0:
                break

            t_start = time.monotonic()
            a_best, sc_best = opt_round(a_init, tb, rng)
            elapsed = time.monotonic() - t_start

            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = a_best.copy()
                print(f"  [n={nv}, init={ii}] score={sc_best:.6f} (NEW BEST) [{elapsed:.1f}s]")
            else:
                print(f"  [n={nv}, init={ii}] score={sc_best:.6f} [{elapsed:.1f}s]")

            all_solutions.append((sc_best, nv, a_best.copy()))

        # Also try the LP direction from current best for this n
        if best_overall is not None and best_overall.size == nv:
            if time.monotonic() < deadline - 60:
                b_best = _full_conv(best_overall)
                M = float(np.max(b_best))
                g = solve_lp(best_overall, M)
                if g is not None and g.sum() > 1.001:
                    g_norm = g / g.sum()
                    sc_g = normed_score(g_norm)
                    if sc_g < best_overall_sc:
                        best_overall_sc = sc_g
                        best_overall = g_norm.copy()
                        print(f"  [n={nv}] LP direct: score={sc_g:.6f} (NEW BEST)")

    # Sort solutions by score for potential cross-n transfer
    all_solutions.sort(key=lambda x: x[0])
    print(f"\n[summary] Top 5 solutions:")
    for sc, nv, a in all_solutions[:5]:
        print(f"  n={nv}, score={sc:.6f}")

    # Final polish phase
    if best_overall is not None and time.monotonic() < deadline - 30:
        remain = deadline - time.monotonic() - 15
        if remain > 20:
            print(f"\n[polish] remaining={remain:.1f}s, n={best_overall.size}")

            # Subgradient polish
            pol_time1 = min(remain * 0.5, 120.0)
            a_p, sc_p = exact_subgrad_polish(best_overall, pol_time1, rng)
            if sc_p < best_overall_sc:
                best_overall_sc = sc_p
                best_overall = a_p.copy()
                print(f"  [polish] subgrad: {sc_p:.6f} (NEW BEST)")

            # Another L-BFGS round from polished solution
            if time.monotonic() < deadline - 30:
                remain2 = deadline - time.monotonic() - 15
                if remain2 > 20:
                    pol_time2 = min(remain2 * 0.6, 100.0)
                    a_p2, sc_p2 = opt_round(best_overall, pol_time2, rng)
                    if sc_p2 < best_overall_sc:
                        best_overall_sc = sc_p2
                        best_overall = a_p2.copy()
                        print(f"  [polish] L-BFGS round2: {sc_p2:.6f} (NEW BEST)")

            # Final LP kick
            if time.monotonic() < deadline - 10:
                b_best = _full_conv(best_overall)
                M = float(np.max(b_best))
                g = solve_lp(best_overall, M)
                if g is not None and g.sum() > 1.001:
                    g_norm = g / g.sum()
                    sc_g = normed_score(g_norm)
                    if sc_g < best_overall_sc:
                        best_overall_sc = sc_g
                        best_overall = g_norm.copy()
                        print(f"  [polish] final LP: {sc_g:.6f} (NEW BEST)")

    # Fallback
    if best_overall is None:
        best_overall = np.ones(128) / 128
        best_overall_sc = 2.0
        print("[fallback] Using uniform n=128, score=2.0")

    # Final verification
    result = [float(max(0.0, x)) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"\n[final] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")

    return result
```
