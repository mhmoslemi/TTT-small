```python
def lp_direction(f, M):
    """Find direction g that maximizes sum(g) subject to g>=0 and (f*g)_k <= M."""
    n = len(f)
    # Build the convolution matrix C where (C @ g)[k] = (f*g)[k]
    # C is (2n-1) x n
    # C[k, i] = f[i] if 0 <= k-i < n, else 0
    # i.e., k-i is in [0, n-1], so i is in [max(0, k-n+1), min(k, n-1)]
    
    from scipy.optimize import linprog
    # Maximize sum(g) = minimize -sum(g)
    c = -np.ones(n)
    # Constraints: C @ g <= M (for each of the 2n-1 convolution positions)
    # A_ub = C, b_ub = M * np.ones(2*n-1)
    # g >= 0 (bounds)
    
    # Build C sparsely
    C = np.zeros((2*n-1, n))
    for i in range(n):
        for k in range(i, i+n):  # k-i is in [0, n-1]
            C[k, i] = f[i]
    
    res = linprog(c, A_ub=C, b_ub=M * np.ones(2*n-1), bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None
```

```python
    b = full_conv(a)
    shifted = b / tau
    shifted -= shifted.max()
    exp_b = np.exp(shifted)
    w = exp_b / exp_b.sum()
    M = float(np.dot(w, b))
    grad = compute_gradient(a, w)
    return M, grad, b


def get_flat_target(n):
    """
    Compute the 'flat convolution target' sequence using the recurrence:
    If all (a*a)_k = c, derive a_k recursively.
    """
    c_target = 1.0 / (2 * n - 1)
    a = np.zeros(n)
    a[0] = np.sqrt(c_target)
    for k in range(1, n):
        if k == 1:
            a[k] = c_target / (2.0 * a[0])
        else:
            cross = 0.0
            for i in range(1, k):
                cross += a[i] * a[k - i]
            a[k] = (c_target - 2.0 * cross) / (2.0 * a[0])
            if a[k] < 0:
                a[k] = 0.0
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0:
        a /= s
    else:
        a = np.ones(n) / n
    return a


def init_two_peak(n, rng):
    """Two Gaussian peaks with random parameters."""
    i = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.1, 0.3)
    c2 = n * rng.uniform(0.7, 0.9)
    s1 = n * rng.uniform(0.05, 0.15)
    s2 = n * rng.uniform(0.05, 0.15)
    a = np.exp(-((i - c1) / s1) ** 2) + np.exp(-((i - c2) / s2) ** 2)
    a += 0.01 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_spread_two_peak(n, rng):
    """Two broad peaks that cover most of the domain."""
    i = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.2, 0.4)
    c2 = n * rng.uniform(0.6, 0.8)
    s1 = n * rng.uniform(0.15, 0.3)
    s2 = n * rng.uniform(0.15, 0.3)
    a = np.exp(-((i - c1) / s1) ** 2) + np.exp(-((i - c2) / s2) ** 2)
    a += 0.005 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_u_shape(n):
    """U-shaped: high at edges."""
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i).astype(np.float64) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_edge_heavy(n, rng):
    """Very heavy at edges, light in middle."""
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i).astype(np.float64) + 1.0
    beta = rng.uniform(2, 6)
    a = 1.0 / (m ** beta)
    return a / a.sum()


def init_random_smooth(n, rng):
    """Random smooth sequence via random Fourier features."""
    n_freq = rng.integers(3, 10)
    freqs = rng.uniform(1, 8, n_freq)
    phases = rng.uniform(0, 2 * np.pi, n_freq)
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n)
    for f, p in zip(freqs, phases):
        a += 0.5 * np.cos(2 * np.pi * f * i / n + p)
    a = np.abs(a) + 0.1
    return a / a.sum()


def init_decaying(n, rng):
    """Decaying from one edge."""
    i = np.arange(n, dtype=np.float64)
    rate = rng.uniform(0.5, 2.0)
    a = np.exp(-rate * i / n)
    a += 0.05 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def get_inits(n, rng):
    """Return list of diverse initializations."""
    inits = []
    inits.append(get_flat_target(n))
    inits.append(init_two_peak(n, rng))
    inits.append(init_spread_two_peak(n, rng))
    inits.append(init_u_shape(n))
    inits.append(init_edge_heavy(n, rng))
    inits.append(init_random_smooth(n, rng))
    inits.append(init_decaying(n, rng))
    inits.append(np.ones(n) / n)
    return inits


def optimize_sequence(a_init, time_budget, rng, use_momentum=True, verbose=False):
    """
    Run projected smoothed-gradient descent with momentum on the simplex.
    a_init: initial point on simplex (sum=1, all >= 0).
    Returns (best_a, best_score).
    """
    n = a_init.size
    a = project_simplex(a_init.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    eta = 0.1
    tau_base = 0.03
    beta = 0.7
    vel = np.zeros(n)

    iters = 0
    last_improve = 0
    stall = 0
    t0 = time.monotonic()

    while time.monotonic() - t0 < time_budget:
        M, grad, b = smooth_objective_and_grad(a, tau_base * max(float(np.max(b)), 1e-20))
        cur_sc = 2.0 * n * float(np.max(b))

        if use_momentum:
            vel = beta * vel + (1.0 - beta) * grad
            direction = vel
        else:
            direction = grad

        a_new = project_simplex(a - eta * direction)

        b_new = full_conv(a_new)
        new_sc = 2.0 * n * float(np.max(b_new))

        # Adaptive acceptance
        if new_sc < cur_sc:
            a = a_new
        elif rng.random() < 0.1:
            a = a_new
        else:
            a = project_simplex(a - 0.3 * eta * direction)
            b_new = full_conv(a)
            new_sc = 2.0 * n * float(np.max(b_new))

        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = iters
            stall = 0
            if verbose and iters % 5000 == 0:
                print(f"    iter {iters}: score={best_sc:.6f}")
        elif iters - last_improve > 3000:
            stall += 1

        iters += 1

        # Adaptive parameters
        if stall > 0 and stall % 2 == 0:
            eta *= 0.8
            tau_base *= 1.3

        # Restart if stuck
        if iters - last_improve > 10000:
            a = project_simplex(best_a + 0.03 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.05 + 0.15 * rng.random()
            tau_base = 0.02 + 0.04 * rng.random()
            beta = 0.5 + 0.3 * rng.random()
            last_improve = iters
            stall = 0

    return best_a, best_sc


def polish_subgrad(a, time_budget, rng):
    """Fine-grained subgradient polish with small steps."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    eta = 0.02
    tau = 0.01
    vel = np.zeros(n)
    beta = 0.5
    iters = 0
    last_improve = 0

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))

        M, grad, _ = smooth_objective_and_grad(a, tau * max(float(np.max(b)), 1e-20))
        vel = beta * vel + (1.0 - beta) * grad
        a_new = project_simplex(a - eta * vel)

        b_new = full_conv(a_new)
        new_sc = 2.0 * n * float(np.max(b_new))

        if new_sc <= cur_sc or rng.random() < 0.05:
            a = a_new
        else:
            a = project_simplex(a - 0.4 * eta * vel)

        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = iters

        iters += 1
        if iters - last_improve > 5000:
            a = project_simplex(best_a + 0.01 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.01 + 0.03 * rng.random()
            tau = 0.005 + 0.02 * rng.random()
            last_improve = iters

    return best_a, best_sc


def polish_perturb(a, time_budget, rng):
    """Random perturbation polish with decreasing scale."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    t0 = time.monotonic()
    scale = 0.03
    stuck = 0
    while time.monotonic() - t0 < time_budget and scale > 1e-8:
        improved = False
        n_tries = max(5, int(20 * scale / 0.03))
        for _ in range(n_tries):
            cand = a + scale * rng.standard_normal(n)
            cand = project_simplex(cand)
            sc = normed_score(cand)
            if sc < best_sc:
                best_sc = sc
                best_a = cand.copy()
                a = cand.copy()
                improved = True
        if not improved:
            stuck += 1
            if stuck > 8:
                break
        else:
            stuck = 0
        scale *= 0.85

    return best_a, best_sc


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses projected smoothed-gradient descent with momentum, the 'flat autocorrelation
    target' initialization, multiple diverse initializations, adaptive parameters,
    multiple sequence lengths, and multi-stage polishing.
    """
    rng = np.random.default_rng(seed)
    deadline = time.monotonic() + max(1.0, float(budget_s) - 10.0)

    # Try to load previous best
    prev_a = None
    prev_sc = np.inf
    try:
        h = np.asarray(height_sequence_1, dtype=np.float64)
        h = np.clip(h, 0.0, 1000.0)
        if h.ndim == 1 and h.size >= 10 and np.all(np.isfinite(h)) and h.sum() > 0.01:
            prev_a = h / h.sum()
            prev_sc = 2.0 * h.size * float(np.max(full_conv(prev_a)))
            print(f"[init] Previous best: n={h.size}, score={prev_sc:.6f}")
    except Exception:
        pass

    best_overall = None
    best_overall_sc = np.inf

    if prev_a is not None and prev_sc < 2.0:
        best_overall = prev_a.copy()
        best_overall_sc = prev_sc

    # Sequence lengths to try
    n_list = [64, 128, 256, 512, 1024, 2048]
    if prev_a is not None:
        np_ = prev_a.size
        if np_ not in n_list and np_ >= 32:
            n_list.append(np_)
        n_list = sorted(set(n_list))

    total_time = max(1.0, deadline - time.monotonic())

    # Time allocation: prioritize lengths around the previous best and smaller n
    # (smaller n is faster per iteration, larger n has lower theoretical bound)
    if prev_a is not None:
        prev_n = prev_a.size
        weights = {}
        for nv in n_list:
            # Weight based on proximity to previous best and 1/n (faster for small n)
            prox = 1.0 / (1.0 + abs(nv - prev_n) / prev_n)
            speed = 1.0 / (0.3 + nv / 512.0)
            weights[nv] = prox * speed
    else:
        weights = {nv: 1.0 / (0.3 + nv / 512.0) for nv in n_list}

    total_w = sum(weights.values())
    time_alloc = {nv: weights[nv] / total_w * total_time * 0.8 for nv in n_list}

    print(f"[plan] n_list={n_list}, total_time={total_time:.1f}s")
    for nv in n_list:
        print(f"  n={nv}: allocated {time_alloc[nv]:.1f}s")

    # Main optimization loop
    for nv in sorted(n_list, key=lambda x: -time_alloc[x]):
        if time.monotonic() > deadline - 30:
            break

        time_n = min(time_alloc[nv], deadline - time.monotonic() - 30.0)
        if time_n < 5.0:
            continue

        print(f"\n[optimizing] n={nv}, time={time_n:.1f}s")

        inits = get_inits(nv, rng)

        # If we have previous best of this length, put it first
        if prev_a is not None and prev_a.size == nv:
            inits.insert(0, prev_a.copy())

        # Also add a perturbed version of the flat target
        ft = get_flat_target(nv)
        inits.append(project_simplex(ft + 0.05 * rng.standard_normal(nv)))

        n_inits = len(inits)
        time_per_init = time_n / n_inits

        for ii, a_init in enumerate(inits):
            if time.monotonic() > deadline - 30:
                break

            tb = min(time_per_init, deadline - time.monotonic() - 30.0)
            if tb < 3.0:
                break

            a_best, sc_best = optimize_sequence(
                a_init, tb, rng,
                use_momentum=(nv <= 4096),
                verbose=False
            )

            if sc_best < best_overall_sc:
                best_overall_sc = sc_best
                best_overall = a_best.copy()
                print(f"  [*** NEW BEST ***] n={nv}, init={ii}, score={sc_best:.6f}")
            else:
                print(f"  n={nv}, init={ii}, score={sc_best:.6f} (best={best_overall_sc:.6f})")

    print(f"\n[after main] best_overall_sc={best_overall_sc:.6f}")

    # ===== POLISHING PHASE =====
    if best_overall is not None and time.monotonic() < deadline - 20:
        remain = deadline - time.monotonic() - 10.0
        if remain > 20.0:
            n_best = best_overall.size
            print(f"\n[polish] n={n_best}, remaining={remain:.1f}s")

            # Phase 1: Subgradient polish
            pol1_time = min(remain * 0.4, 300.0)
            if pol1_time > 10.0:
                a_p, sc_p = polish_subgrad(best_overall, pol1_time, rng)
                if sc_p < best_overall_sc:
                    best_overall_sc = sc_p
                    best_overall = a_p.copy()
                    print(f"  [polish1] subgrad: {sc_p:.6f}")
                else:
                    print(f"  [polish1] subgrad: no improvement (={sc_p:.6f})")

            # Phase 2: Perturbation polish
            if time.monotonic() < deadline - 20:
                remain2 = deadline - time.monotonic() - 10.0
                pol2_time = min(remain2 * 0.4, 200.0)
                if pol2_time > 10.0:
                    a_p2, sc_p2 = polish_perturb(best_overall, pol2_time, rng)
                    if sc_p2 < best_overall_sc:
                        best_overall_sc = sc_p2
                        best_overall = a_p2.copy()
                        print(f"  [polish2] perturb: {sc_p2:.6f}")
                    else:
                        print(f"  [polish2] perturb: no improvement (={sc_p2:.6f})")

            # Phase 3: Final fine subgradient
            if time.monotonic() < deadline - 15:
                remain3 = deadline - time.monotonic() - 10.0
                pol3_time = min(remain3 * 0.5, 100.0)
                if pol3_time > 5.0:
                    a_p3, sc_p3 = polish_subgrad(best_overall, pol3_time, rng)
                    if sc_p3 < best_overall_sc:
                        best_overall_sc = sc_p3
                        best_overall = a_p3.copy()
                        print(f"  [polish3] fine subgrad: {sc_p3:.6f}")
                    else:
                        print(f"  [polish3] fine subgrad: no improvement (={sc_p3:.6f})")

    # Fallback
    if best_overall is None:
        print("[fallback] Using uniform sequence")
        best_overall = np.ones(128) / 128
        best_overall_sc = 2.0

    # Final verification
    result = [float(max(0.0, min(1000.0, x))) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"\n[final] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")

    return result
```
