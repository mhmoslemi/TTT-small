def gradient_topk(a, k):
    """Top-k gradient (faster for large tau / few peaks)."""
    n = len(a)
    conv = fast_self_conv(a)
    S = a.sum()
    k_act = min(k, len(conv))
    idx = np.argpartition(conv, -k_act)[-k_act:]
    vals = conv[idx]
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]
    top = float(vals[0])
    if top > 1e-12:
        temp = 0.1 * top
        wts = np.exp((vals - top) / temp)
        wts /= wts.sum()
    else:
        wts = np.ones(k_act) / k_act

    grad_W = np.zeros(n)
    for i in range(idx.size):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-14:
            continue
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start <= j_end:
            grad_W[j_start:j_end + 1] += wi * 2.0 * a[j_start:j_end + 1][::-1]

    M = float(np.dot(vals, wts))
    S2 = S * S
    S3 = S2 * S
    grad_f = 2.0 * n * (grad_W / S2 - 2.0 * M / S3)
    return grad_f


def clamp_nonneg(a):

def adam_step(a, grad, m, v, t, lr):
    """Adam update in-place."""
    b1, b2, eps = 0.9, 0.999, 1e-8
    m[:] = b1 * m + (1.0 - b1) * grad
    v[:] = b2 * v + (1.0 - b2) * grad * grad
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)
    a -= lr * mh / (np.sqrt(vh) + eps)


def init_uniform(n):
    return np.ones(n)


def init_smooth_random(n, rng):
    """Low-pass filtered random (smooth)."""
    r = rng.random(n)
    # Simple moving average as low-pass
    kernel_size = max(3, n // 128)
    kernel = np.ones(kernel_size) / kernel_size
    smooth = np.convolve(r, kernel, mode='same')
    return np.clip(smooth, 0.01, 10.0)


def init_chirp(n, freq, rng):
    """Amplitude-modulated sinusoid (non-negative)."""
    x = np.linspace(0, 1, n)
    base = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x + rng.random() * 2 * np.pi)
    return np.clip(base, 0.01, 10.0)


def init_sparse_support(n, m, rng):
    """Random sparse support with random values."""
    a = np.zeros(n)
    positions = rng.choice(n, size=min(m, n), replace=False)
    a[positions] = 0.5 + rng.random(min(m, n)) * 0.5
    return a


def init_cyclic_structure(n, period, rng):
    """Periodic structure with random phase."""
    a = np.zeros(n)
    for j in range(n):
        phase = (j % period) / period * 2 * np.pi + rng.random()
        a[j] = 0.5 + 0.4 * np.cos(phase)
    return np.clip(a, 0.01, 10.0)


def build_inits(n, rng, parent=None):
    """Build diverse initializations."""
    inits = []
    if parent is not None and len(parent) == n:
        inits.append(("parent", parent.copy()))
    inits.append(("uniform", init_uniform(n)))
    inits.append(("smooth_rand", init_smooth_random(n, rng)))
    for freq in [2, 5, 10, 20]:
        inits.append(("chirp_%d" % freq, init_chirp(n, freq, rng)))
    inits.append(("sparse_64", init_sparse_support(n, 64, rng)))
    inits.append(("sparse_256", init_sparse_support(n, 256, rng)))
    for period in [7, 13, 31]:
        inits.append(("cyclic_%d" % period, init_cyclic_structure(n, period, rng)))
    # Random
    inits.append(("rand", 0.5 + rng.random(n)))
    return inits


def optimize_single(a0, deadline, rng, tau_start, tau_end, lr_start, method="softmax"):
    """Run optimization from a0 until deadline. Returns best score and array."""
    n = a0.size
    a = clamp_nonneg(a0.astype(np.float64).copy())
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_score = score_internal(a)
    best_a = a.copy()
    no_improve = 0
    patience = 300

    # Cosine schedule with warm restarts
    n_restarts = 0
    steps_since_restart = 0
    total_steps_planned = 10000

    while time.monotonic() < deadline:
        # Anneal tau
        progress = min(1.0, t / max(1, total_steps_planned))
        tau = tau_start * (tau_end / tau_start) ** progress
        if method == "topk":
            # Map tau to top-k
            k = max(2, int(50 / (tau * tau_start)))
            k = min(k, 64)
            grad = gradient_topk(a, k)
        else:
            grad = gradient_softmax(a, tau)

        # Learning rate with cosine decay
        steps_since_restart += 1
        progress_r = min(1.0, steps_since_restart / total_steps_planned)
        lr = lr_start * (0.1 + 0.9 * 0.5 * (1 + np.cos(np.pi * progress_r)))

        t += 1
        adam_step(a, grad, m, v, t, lr)
        a = clamp_nonneg(a)

        # Check score periodically (every 10 steps to save time)
        if t % 10 == 0 or t <= 5:
            sc = score_internal(a)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
                no_improve = 0
                steps_since_restart = 0  # reset schedule
            else:
                no_improve += 1

        # Warm restart
        if no_improve >= patience:
            scale = 0.03 + 0.07 * rng.random()
            a = best_a * (1.0 + scale * rng.standard_normal(n))
            a = clamp_nonneg(a)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            no_improve = 0
            patience = max(100, patience // 2)
            n_restarts += 1
            steps_since_restart = 0

    return best_score, best_a


def coordinate_polish(a, deadline, rng):
    """Coordinate descent polish: optimize one coordinate at a time."""
    n = len(a)
    best_score = score_internal(a)
    best_a = a.copy()
    conv = fast_self_conv(a)
    S = a.sum()

    while time.monotonic() < deadline:
        improved = False
        # Random order of coordinates
        order = rng.permutation(n)
        for j in order:
            if time.monotonic() > deadline:
                break
            # Binary search for optimal a[j]
            lo = 0.0
            hi = max(a[j] * 3.0, 1.0)
            best_local = a[j].copy()
            best_local_score = best_score

            # Evaluate at current value
            for _ in range(12):
                mid = (lo + hi) / 2.0
                a[j] = mid
                # Quick score update: recompute conv at affected lags
                # For speed, just recompute full score every 4 iterations
                if _ % 4 == 0 or _ == 11:
                    sc = score_internal(a)
                    if sc < best_local_score - 1e-10:
                        best_local_score = sc
                        best_local = np.array([mid])
                        lo = mid
                    else:
                        hi = mid

            a[j] = float(best_local[0])
            if best_local_score < best_score - 1e-8:
                best_score = best_local_score
                best_a = a.copy()
                improved = True

        if not improved:
            break

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Main search: find non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = max(1.0, float(budget_s))
    deadline_final = start + total - 15.0  # safety margin

    # Determine n from parent or default
    n = 2048
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if arr.size > 0:
            n = int(arr.size)
            parent = np.clip(arr, 0.0, 1000.0)
    except Exception:
        pass
    n = min(max(256, int(n)), 8192)

    print("Starting with n=%d, parent_score=%s" % (
        n,
        "%.6f" % score_internal(parent) if parent is not None else "N/A"))

    best_score = np.inf
    best_a = None

    def consider(sc, arr, label):
        nonlocal best_score, best_a
        if sc < best_score:
            best_score = sc
            best_a = arr.copy()
            print("  [%s] new best: %.8f" % (label, best_score))
            return True
        return False

    # Quick n-selection: try a few n values with random + uniform
    if time.monotonic() < start + 60:
        for n_try in [n // 2, n, n * 2]:
            n_try = min(max(256, n_try), 8192)
            a_test = np.ones(n_try)
            sc_test = score_internal(a_test)
            # Also try random
            a_rand = 0.5 + rng.random(n_try)
            sc_rand = score_internal(a_rand)
            if min(sc_test, sc_rand) < best_score:
                best_a = a_test.copy() if sc_test < sc_rand else a_rand.copy()
                best_score = min(sc_test, sc_rand)
                n = n_try
            print("  n=%d: uniform=%.6f, rand=%.6f" % (n_try, sc_test, sc_rand))
        # Reset to chosen n
        if parent is not None and len(parent) != n:
            parent = None
    print("Chosen n = %d" % n)

    # Phase 1: Multi-start exploration (55% of time)
    phase1_end = start + 0.55 * (total - 15.0)
    inits = build_inits(n, rng, parent)
    time_each = max(5.0, (phase1_end - start) / len(inits))
    print("Phase 1: %d initializations, ~%.0fs each" % (len(inits), time_each))

    for tag, a0 in inits:
        if time.monotonic() > phase1_end:
            break
        a0 = np.clip(a0, 0.0, 1000.0)
        t0 = time.monotonic()
        sc, aa = optimize_single(a0, min(phase1_end, t0 + time_each), rng,
                                 tau_start=1.0 / (3.0 * max(fast_self_conv(a0).max(), 1e-6)),
                                 tau_end=1.0 / (0.1 * max(fast_self_conv(a0).max(), 1e-6)),
                                 lr_start=0.5, method="softmax")
        consider(sc, aa, "P1_" + tag)

    # Phase 2: Refine best (30% of time)
    phase2_end = start + 0.85 * (total - 15.0)
    if best_a is not None and time.monotonic() < phase2_end:
        print("Phase 2: Refining, current best = %.8f" % best_score)
        # Use top-k method for more focus on peaks
        conv_max = fast_self_conv(best_a).max()
        sc, aa = optimize_single(best_a, phase2_end, rng,
                                 tau_start=1.0 / (5.0 * max(conv_max, 1e-6)),
                                 tau_end=1.0 / (0.05 * max(conv_max, 1e-6)),
                                 lr_start=0.1, method="topk")
        consider(sc, aa, "P2_refine")

    # Phase 3: Coordinate polish (12% of time)
    if best_a is not None and time.monotonic() < deadline_final:
        phase3_end = min(deadline_final, time.monotonic() + 0.12 * (total - 15.0))
        print("Phase 3: Coordinate polish, current best = %.8f" % best_score)
        sc, aa = coordinate_polish(best_a, phase3_end, rng)
        consider(sc, aa, "P3_coord")

    # Phase 4: Final micro-polish with very focused top-k
    if best_a is not None and time.monotonic() < deadline_final:
        conv_max = fast_self_conv(best_a).max()
        sc, aa = optimize_single(best_a, deadline_final, rng,
                                 tau_start=1.0 / max(conv_max, 1e-6),
                                 tau_end=1.0 / max(conv_max, 1e-6),
                                 lr_start=0.01, method="topk")
        consider(sc, aa, "P4_final")

    # Safety fallback
    if best_a is None:
        best_a = np.ones(n)

    out = [float(x) for x in best_a]
    verified = float(evaluate_sequence(out))
    print("FINAL: internal=%.8f, verified=%.8f, n=%d, time=%.1fs" % (
        best_score, verified, len(out), time.monotonic() - start))
    return out
