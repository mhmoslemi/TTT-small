"""
Minimize 2*n*max(a*a)/(sum(a))^2 using FISTA (Nesterov accelerated projected gradient)
with adaptive smooth-max approximation on the probability simplex.
"""

import numpy as np
import time
import math

def fft_convolution(a):
    """Linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)

def smooth_max_and_weights(b, beta):
    """Softmax-weighted smooth maximum."""
    m = float(np.max(b))
    if m <= 0.0:
        w = np.ones_like(b) / b.size
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0:
        e = np.zeros_like(b)
        e[np.argmax(b)] = 1.0
        Z = 1.0
    w = e / Z
    G = m + np.log(Z) / beta
    return G, w

def corr_pw(p, w):
    """s_i = sum_j p_j * w_{i+j} for i=0..n-1."""
    n = p.size
    m = w.size
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n : m][::-1].copy()
    return s

def project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        return np.full(n, target / n, dtype=np.float64)
    # Bisection for the threshold
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0:
        q = np.full(n, target / n, dtype=np.float64)
    else:
        q *= target / s
    return q

def fast_score(p):
    """Compute 2*n*max(p*p) for sum(p)=1."""
    b = fft_convolution(p)
    M = float(np.max(b))
    return float(2 * p.size * M)

def fista_optimize(p0, deadline, beta_start=2000, beta_end=200000, max_iter=500000):
    """FISTA with adaptive smooth max beta."""
    p = np.asarray(p0, dtype=np.float64).copy()
    n = p.size
    p = project_simplex(p, 1.0)
    
    y = p.copy()
    t_k = 1.0
    best_p = p.copy()
    best_score = fast_score(p)
    
    step = 1.0
    it = 0
    current_beta = beta_start
    
    while it < max_iter and time.monotonic() < deadline:
        it += 1
        
        # Update beta periodically
        if it % 200 == 0:
            progress = min(1.0, it / (max_iter * 0.8))
            current_beta = beta_start * (beta_end / beta_start) ** progress
        
        # Compute gradient at y
        b = fft_convolution(y)
        G, w = smooth_max_and_weights(b, current_beta)
        
        if not np.isfinite(G):

        svec = np.maximum(svec, 0.0)
        grad = 2.0 * svec
        grad -= float(np.mean(grad))
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-15:

        # Use backtracking
        accepted = False
        base_step = 1.0 / max(n * current_beta, 10.0)  # rough estimate

            # Check decrease (using smooth objective)

            G_new, _ = smooth_max_and_weights(b_new, current_beta)
            
            if np.isfinite(G_new) and G_new <= G + 1e-10:
                p = x_new
                accepted = True
                break
        
        if not accepted:

        momentum = (t_k - 1.0) / t_new
        y = p + momentum * (p - y)
        y = project_simplex(y, 1.0)  # Keep y on the simplex

    # Final evaluation

"""
Minimize 2*n*max(a*a)/(sum(a))^2 using Nesterov-accelerated projected gradient (FISTA)
with adaptive smooth-max approximation. Starts from known good solution height_sequence_1,
uses n=16384 for more attack surface, and employs adaptive beta scheduling with momentum
restarts to escape local minima.
"""

import numpy as np
import time
import math


def fft_convolution(a):
    """Linear convolution a*a using FFT, clipping negative roundoff."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    """Smooth max via log-sum-exp and normalized softmax weights."""
    m = float(np.max(b))
    if m <= 0.0 or beta <= 0:
        w = np.ones_like(b, dtype=np.float64) / b.size
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if not np.isfinite(Z) or Z <= 0.0:
        e = np.zeros_like(b, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + math.log(Z) / beta
    return float(G), w


def corr_pw(p, w):
    """s_i = sum_j p_j * w_{i+j} for i=0..n-1 (cross-correlation)."""
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(0, dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n:m][::-1].copy()
    return s


def project_simplex(v, target=1.0):
    """Project vector onto {x >= 0, sum(x) = target} via bisection."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        return np.full(n, target / n, dtype=np.float64)
    lo = float(np.min(v))
    hi = float(np.max(v))
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        q = np.full(n, target / n, dtype=np.float64)
    else:
        q *= target / s
    return q


def fast_score(p):
    """Compute 2*n*max(p*p) assuming sum(p)=1."""
    b = fft_convolution(p)
    M = float(np.max(b))
    return float(2.0 * p.size * M)


def fista_optimize(p0, deadline, beta_start=5000.0, beta_max=400000.0, verbose=True, tag=""):
    """
    FISTA (Nesterov accelerated projected gradient) with adaptive smooth-max beta.
    Minimize max(p*p) over the probability simplex.
    """
    n = p0.size
    p = project_simplex(np.asarray(p0, dtype=np.float64), 1.0)
    best_p = p.copy()
    best_score = fast_score(p)
    if verbose:
        print(f"  [{tag}] FISTA start: score={best_score:.8f}")

    beta = beta_start
    t_k = 1.0
    y = p.copy()
    it = 0
    no_improve = 0
    last_best = best_score
    step_base = 1.0

    while time.monotonic() < deadline:
        it += 1
        step = step_base / (n * beta)

        # Compute smooth max and gradient at y
        b = fft_convolution(y)
        G, w = smooth_max_and_weights(b, beta)
        if not np.isfinite(G):
            break

        svec = corr_pw(y, w)
        svec = np.maximum(svec, 0.0)
        grad = 2.0 * svec
        grad -= float(np.mean(grad))

        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            t_k = 1.0
            y = p.copy()
            continue

        # FISTA gradient step + projection
        x_new = project_simplex(y - step * grad, 1.0)

        # Nesterov momentum update
        t_new = 0.5 * (1.0 + math.sqrt(1.0 + 4.0 * t_k * t_k))
        momentum = (t_k - 1.0) / t_new
        y = x_new + momentum * (x_new - p)
        y = project_simplex(y, 1.0)
        t_k = t_new
        p = x_new

                best_p = p.copy()

            else:
                no_improve += 30

        if it % 150 == 0:
            beta = min(beta * 1.4, beta_max)
            t_k = 1.0
            y = p.copy()

            t_k = 1.0

def peak_shaving_refinement(p, deadline, iterations=2000, verbose=True, tag=""):
    """
    Direct peak shaving: identify top peaks in the convolution,
    and reduce them by redistributing mass from contributing positions.
    """
    n = p.size
    p = project_simplex(np.asarray(p, dtype=np.float64), 1.0)
    best_p = p.copy()
    best_score = fast_score(p)
    if verbose:
        print(f"  [{tag}] Peak shaving start: score={best_score:.8f}")

    it = 0
    while it < iterations and time.monotonic() < deadline:
        it += 1

        b = fft_convolution(p)
        M = float(np.max(b))
        score = 2.0 * n * M

        if score < best_score:
            best_score = score
            best_p = p.copy()
            if verbose and it % 200 == 0:
                print(f"  [{tag}] peak it={it} score={best_score:.8f} max={M:.2e}")

        # Find top K peak positions
        K = min(5, b.size)
        top_k = np.argsort(b)[-K:]
        top_k = top_k[top_k < 2 * n - 1]

        # For each peak, compute which positions contribute most
        # and try to reduce their mass
        delta = np.zeros(n, dtype=np.float64)
        for pk in top_k:
            pk = int(pk)
            # Contributions: p[i] * p[pk-i] for valid i
            i_min = max(0, pk - (n - 1))
            i_max = min(n - 1, pk)
            if i_max < i_min:
                continue
            idx = np.arange(i_min, i_max + 1)
            contrib = p[idx] * p[pk - idx]
            # Reduce the largest contributors
            top_contrib_idx = idx[np.argsort(contrib)[-min(3, len(idx)):]]
            for ii in top_contrib_idx:
                if p[ii] > 1e-8:
                    delta[ii] -= 1.0

        # Also add mass to positions that don't contribute to peaks
        # Find positions with low total contribution to any peak
        peak_contrib = np.zeros(n, dtype=np.float64)
        for pk in top_k:
            pk = int(pk)
            i_min = max(0, pk - (n - 1))
            i_max = min(n - 1, pk)
            if i_max >= i_min:
                idx = np.arange(i_min, i_max + 1)
                peak_contrib[idx] += p[idx] * p[pk - idx]

        # Add to low-contribution positions
        low_mask = peak_contrib < np.median(peak_contrib)
        if np.any(low_mask):
            n_low = int(np.sum(low_mask))
            total_reduce = max(0.0, -float(np.sum(delta)))
            if total_reduce > 0:
                add_per = total_reduce / n_low
                delta[low_mask] += add_per

        if np.abs(delta).max() < 1e-15:
            break

        # Apply with small step
        lr = 1e-4
        p_new = project_simplex(p + lr * delta, 1.0)

        # Accept if it improves the smooth objective
        score_new = fast_score(p_new)
        if score_new < score + 1e-12:
            p = p_new
        else:
            lr *= 0.5
            p_new = project_simplex(p + lr * delta, 1.0)
            score_new = fast_score(p_new)
            if score_new < score + 1e-12:
                p = p_new

    score = fast_score(p)
    if score < best_score:
        best_score = score
        best_p = p.copy()

    if verbose:
        print(f"  [{tag}] Peak shaving done: it={it} score={best_score:.8f}")
    return best_p, best_score

        p2, s2 = peak_shaving_refinement(best_p, ps_deadline, iterations=3000,
                                         verbose=True, tag="P2")
        if s2 < best_score:
            best_score = s2
            best_p = p2.copy()
            print(f"After Phase 2 (peak shaving): score={best_score:.8f}")

    # Phase 3: Second FISTA pass from the refined point with higher beta
    remaining = deadline - time.monotonic()
    if remaining > 60.0:
        fista2_deadline = time.monotonic() + min(remaining * 0.6, 300.0)
        p3, s3 = fista_optimize(best_p, fista2_deadline, beta_start=20000.0,
                                beta_max=600000.0, verbose=True, tag="P3")
        if s3 < best_score:
            best_score = s3
            best_p = p3.copy()
            print(f"After Phase 3 (FISTA high-beta): score={best_score:.8f}")

    # Phase 4: Try a perturbed restart
    remaining = deadline - time.monotonic()
    if remaining > 90.0:
        # Add small perturbation to escape potential local minimum
        noise_scale = 0.02 / N
        p_pert = best_p + noise_scale * rng.standard_normal(N)
        p_pert = project_simplex(p_pert, 1.0)
        pert_deadline = time.monotonic() + min(remaining * 0.5, 200.0)
        p4, s4 = fista_optimize(p_pert, pert_deadline, beta_start=10000.0,
                                beta_max=400000.0, verbose=True, tag="P4-pert")
        if s4 < best_score:
            best_score = s4
            best_p = p4.copy()
            print(f"After Phase 4 (perturbed restart): score={best_score:.8f}")

    # Final: one more short FISTA polish
    remaining = deadline - time.monotonic()
    if remaining > 30.0:
        polish_deadline = time.monotonic() + min(remaining * 0.7, 120.0)
        p5, s5 = fista_optimize(best_p, polish_deadline, beta_start=50000.0,
                                beta_max=600000.0, verbose=True, tag="P5-polish")
        if s5 < best_score:
            best_score = s5
            best_p = p5.copy()
            print(f"After Phase 5 (polish): score={best_score:.8f}")

    print(f"\nBest internal score: {best_score:.8f}")

    # Final cleanup: ensure valid output
    final_p = np.asarray(best_p, dtype=np.float64)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    s = float(np.sum(final_p))
    if s > 0.0 and np.isfinite(s):
        final_p /= s
    else:
        final_p = np.full(N, 1.0, dtype=np.float64)

    final_seq = [float(x) for x in final_p]

    # Official verification
    try:
        off = float(evaluate_sequence(final_seq))
        if np.isfinite(off):
            print(f"Official score: {off:.8f}")
            if off > best_score + 0.01:
                print("Warning: official score significantly differs from internal!")
        else:
            print("Official score is inf, using fallback")
            return [1.0] * N
    except Exception as e:
        print(f"Evaluation error: {e}, using fallback")
        return [1.0] * N

    # Safety: if score is bad, return uniform
    try:
        off_val = float(evaluate_sequence(final_seq))
        if not np.isfinite(off_val) or off_val > 2.0:
            print("Using uniform fallback")
            return [1.0] * N
    except Exception:
        return [1.0] * N

    elapsed = time.monotonic() - start_time
    print(f"Total time: {elapsed:.1f}s, returning sequence of length {N}")
    return final_seq
