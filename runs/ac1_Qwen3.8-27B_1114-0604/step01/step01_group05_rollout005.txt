"""
Minimize 2*n*max(a*a)/(sum(a))^2 for nonnegative sequences.

Uses Nesterov-accelerated projected gradient descent with smooth max,
warm-started from a previous best solution if available.
Includes block coordinate descent polish and multi-start diversification.
"""

import numpy as np
import time
from scipy.optimize import minimize

def fft_convolution(a):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    """Smooth max via log-sum-exp and softmax weights."""
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float("inf"), np.empty(0, dtype=np.float64)
    m = float(np.max(b))
    if not np.isfinite(m) or m <= 0.0:
        w = np.ones_like(b) / float(b.size)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return G, w


def fast_score(p):
    """Compute 2*n*max(p*p) for sum-normalized p."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float("inf")
    S = float(np.sum(p))
    if S <= 0.0:
        return float("inf")
    b = fft_convolution(p)
    M = float(np.max(b))
    return float(2.0 * n * M / (S * S))


def grad_convolution(p, w):
    """Gradient of smooth max: s_i = sum_j p_j * w_{i-j}, returned as 2*svec."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size  # m = 2n-1
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    if n == 1:
        return np.array([2.0 * float(np.dot(p, w[:1]))], dtype=np.float64)
    
    # s_i = sum_j p_j * w_{i+j} where w has length 2n-1
    # This is a cross-correlation: s = corr(p, w) at specific lags
    # s_i = sum_j p_j * w[i+j] for j=0..n-1
    # = (p * w_reversed)[(2n-2) - i] ... let me think again
    
    # Actually: s_i = sum_{j=0}^{n-1} p_j * w_{i+j}
    # Let's define this as a convolution/correlation.
    # If we reverse p: pr_j = p_{n-1-j}
    # Then (pr * w)_k = sum_j pr_j * w_{k-j} = sum_j p_{n-1-j} * w_{k-j}
    # Let j' = n-1-j: = sum_{j'} p_{j'} * w_{k - n + 1 + j'}
    # So (pr * w)_k = sum_j' p_{j'} * w_{(k-n+1) + j'}
    # We want s_i = sum_j p_j * w_{i+j} = (pr * w)_{i + n - 1}
    
    need = n + m - 1  # = n + (2n-1) - 1 = 3n-2
    size = 1
    while size < need:
        size <<= 1
    
    pr = p[::-1]
    P = np.fft.rfft(pr, size)
    W = np.fft.rfft(w, size)
    C = np.fft.irfft(P * W, size)[:need]
    
    # s_i = C[i + n - 1] for i = 0..n-1
    s = C[n-1 : 2*n-1].copy()
    return 2.0 * s


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if target <= 0:
        return np.zeros_like(v)
    if not np.all(np.isfinite(v)):
        return np.full(n, target / n)
    
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0 or not np.isfinite(s):
        return np.full(n, target / n)
    return q * (target / s)


def normalize(p):
    """Normalize to non-negative, sum=1."""
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / p.size)
    return p / s


def nesterov_refine(p0, deadline, beta_start=5000.0, beta_end=2000000.0, 
                     max_iter=500000, momentum=0.9):

    n = p.size
    if n == 0:
        return p, float("inf")

        t = 0.123 * np.arange(n)
        p = normalize(p * (1.0 + 1e-4 * np.sin(t)))
    
    best_p = p.copy()
    best_score = fast_score(p)
    
    v = np.zeros(n, dtype=np.float64)  # velocity
    p_prev = p.copy()
    
    step = 1.0
    no_improve = 0
    it = 0
    beta = beta_start
    
    # Beta schedule: linear in log space over first 2000 iterations
    beta_schedule_len = 3000
    
    while it < max_iter and time.monotonic() < deadline:
        it += 1
        
        # Update beta
        if it < beta_schedule_len:
            frac = it / beta_schedule_len
            log_beta = np.log(beta_start) + frac * (np.log(beta_end) - np.log(beta_start))
            beta = float(np.exp(log_beta))
        else:
            beta = beta_end
        
        b = fft_convolution(p)
        G, w = smooth_max_and_weights(b, beta)
        if not np.isfinite(G):
            break
        
        cur_smooth = float(n) * G
        m_true = float(np.max(b))
        cur_score = float(2.0 * n * m_true)
        
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if it % 500 == 0:
            print(f"  nesterov it={it} best={best_score:.8f} beta={beta:.0f} step={step:.3e}")
        
        if no_improve > 5000:
            # Try to restart with smaller step
            step = 1.0
            v = np.zeros(n, dtype=np.float64)
            no_improve = 0
            if it > 200000:
                break
        
        # Compute gradient
        svec = grad_convolution(p, w)
        svec = np.maximum(svec, 0.0)
        g = 2.0 * svec
        grad = g - float(np.mean(g))  # Remove component in sum=1 direction
        
        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-15:
            break
        
        # Nesterov momentum
        v = momentum * v + grad
        
        # Line search with backtracking
        accepted = False
        base = step
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001):
            alpha = base * mult
            q = project_simplex(p - alpha * v, 1.0)
            
            qb = fft_convolution(q)
            G2, _ = smooth_max_and_weights(qb, beta)
            if not np.isfinite(G2):
                continue
            
            if float(n) * G2 < cur_smooth - 1e-12:
                p = q
                cur_smooth = float(n) * G2
                cur_score = float(2.0 * n * float(np.max(qb)))
                accepted = True
                if cur_score < best_score - 1e-14:
                    best_score = cur_score
                    best_p = p.copy()
                break
        
        if not accepted:
            step *= 0.5
            v = momentum * v  # dampen velocity
            if step < 1e-8:
                break
        else:
            if step < 3.0:
                step = min(3.0, step * 1.3)
    
    return best_p, best_score


def block_coordinate_polish(p0, deadline, num_sweeps=5):
    """Block coordinate descent: optimize groups of consecutive coordinates."""
    p = normalize(p0)
    n = p.size
    best_p = p.copy()
    best_score = fast_score(p)
    
    # Block size for coordinate descent
    block_sizes = [16, 32, 64]
    
    for bs in block_sizes:
        if time.monotonic() >= deadline:
            break
        
        for sweep in range(num_sweeps):
            if time.monotonic() >= deadline:
                break
            
            num_blocks = n // bs
            for bi in range(num_blocks):
                if time.monotonic() >= deadline:
                    break
                
                start = bi * bs
                end = start + bs

"""
Minimize 2*n*max(a*a)/(sum(a))^2 for nonnegative sequences.

Uses FISTA (accelerated proximal gradient) with smooth max (log-sum-exp),
warm-started from a previous best solution. Includes adaptive beta scheduling,
block coordinate polish, and multi-start diversification with perturbations.
"""

import numpy as np
import time
from scipy.optimize import minimize

def fft_convolution(a):
    """Linear convolution a*a via FFT, clipped to non-negative."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    """Smooth max via log-sum-exp; returns (value, softmax weights)."""
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float("inf"), np.empty(0, dtype=np.float64)
    m = float(np.max(b))
    if not np.isfinite(m) or m <= 0.0:
        w = np.ones_like(b) / float(b.size)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return G, w


def fast_score(p):
    """Compute 2*n*max(p*p)/(sum(p))^2 for non-negative p."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float("inf")
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float("inf")
    b = fft_convolution(p)
    M = float(np.max(b))
    if not np.isfinite(M):
        return float("inf")
    return float(2.0 * n * M / (S * S))


def grad_smooth_max(p, w):
    """
    Gradient of n * smooth_max(p*p, beta) w.r.t. p.
    Returns grad_i = 2 * sum_j p_j * w_{i+j}.
    """
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size  # 2n-1
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    if n == 1:
        return np.array([2.0 * float(p[0] * w[0])], dtype=np.float64)

    # s_i = sum_j p_j * w_{i+j} = (p_rev * w)[i + n - 1]
    need = n + m - 1  # 3n - 2
    size = 1
    while size < need:
        size <<= 1

    p_rev = p[::-1]
    P = np.fft.rfft(p_rev, size)
    W = np.fft.rfft(w, size)
    C = np.fft.irfft(P * W, size)[:need]

    s = C[n - 1: 2 * n - 1].copy()
    return 2.0 * s


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if target <= 0:
        return np.zeros(n, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(n, target / n, dtype=np.float64)

    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, target / n, dtype=np.float64)
    return q * (target / s)


def normalize(p):
    """Clip to non-negative and normalize to sum=1."""
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / float(p.size), dtype=np.float64)
    return p / s

        return p, float("inf")

    # Break symmetry if uniform
    if n > 1 and float(np.max(np.abs(p - p.mean()))) < 1e-14:
        t = 0.173 * np.arange(n, dtype=np.float64)
        p = normalize(p * (1.0 + 2e-4 * np.sin(t) + 1e-4 * np.cos(2 * t)))

    best_p = p.copy()
    best_score = fast_score(p)
    best_smooth = float(n) * smooth_max_and_weights(fft_convolution(p), beta_start)[0]

    y = p.copy()  # FISTA extrapolation point
    t_k = 1.0     # FISTA momentum parameter
    step = 1.0    # base step size
    no_improve = 0
    it = 0
    beta = beta_start

    while it < max_iter and time.monotonic() < deadline:
        it += 1

        # Beta schedule
        if it < beta_schedule_len:
            frac = it / beta_schedule_len
            beta = float(np.exp(np.log(beta_start) + frac * (np.log(beta_end) - np.log(beta_start))))
        else:
            beta = beta_end

        # Evaluate at y
        b_y = fft_convolution(y)
        G_y, w_y = smooth_max_and_weights(b_y, beta)
        if not np.isfinite(G_y):
            break

        obj_y = float(n) * G_y
        true_score_y = float(2.0 * n * float(np.max(b_y)))

        if true_score_y < best_score - 1e-14:
            best_score = true_score_y
            best_p = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 1000 == 0:
            print(f"    FISTA it={it} beta={beta:.0f} obj={obj_y:.8f} best={best_score:.8f} step={step:.4f}")

        if no_improve > restart_every:
            # FISTA restart: reset momentum
            y = p.copy()
            t_k = 1.0
            step = max(step * 0.5, 0.1)
            no_improve = 0
            if it > max_iter * 0.8:
                break

        # Gradient at y
        grad = grad_smooth_max(y, w_y)
        # Remove the component along the all-ones direction (to respect sum=1)
        grad = grad - float(np.mean(grad))

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-16:
            break

        # Step size: use 2/L where L ~ 2*beta (rough estimate)
        # But use backtracking for safety
        alpha = step
        accepted = False
        for _ in range(20):
            p_new = project_simplex(y - alpha * grad, 1.0)

            b_new = fft_convolution(p_new)
            G_new, _ = smooth_max_and_weights(b_new, beta)
            if not np.isfinite(G_new):
                alpha *= 0.5
                continue

            obj_new = float(n) * G_new
            # Armijo condition
            descent = (obj_y - obj_new)
            if descent > 1e-14 * alpha * gnorm * gnorm:
                p = p_new
                accepted = true_score_new = float(2.0 * n * float(np.max(b_new)))
                if true_score_new < best_score - 1e-14:
                    best_score = true_score_new
                    best_p = p.copy()
                    no_improve = 0
                break
            alpha *= 0.5

        if not accepted:
            # Use a very small step in the gradient direction
            p = project_simplex(y - 1e-3 * step * grad, 1.0)
            b_new = fft_convolution(p)
            G_new, _ = smooth_max_and_weights(b_new, beta)
            obj_new = float(n) * G_new if np.isfinite(G_new) else obj_y
            true_score_new = float(2.0 * n * float(np.max(b_new)))
            if true_score_new < best_score - 1e-14:
                best_score = true_score_new
                best_p = p.copy()
                no_improve = 0

        # FISTA update: compute next y
        t_new = (1.0 + float(np.sqrt(1.0 + 4.0 * t_k * t_k))) / 2.0
        theta = (t_k - 1.0) / t_new
        y = p + theta * (p - y)
        y = project_simplex(y, 1.0)  # keep y in feasible set
        t_k = t_new

        # Adapt step size
        if accepted:
            if step < 2.0:
                step = min(2.0, step * 1.05)
        else:
            step = max(step * 0.7, 0.01)

    return best_p, best_score


def heavy_ball_refine(p0, deadline, beta_start=5000.0, beta_end=3000000.0,
                      max_iter=300000, momentum=0.9):
    """
    Heavy-ball momentum projected gradient descent.
    Alternative to FISTA; sometimes escapes different local minima.
    """
    p = normalize(p0)
    n = p.size
    if n == 0:
        return p, float("inf")

    if n > 1 and float(np.max(np.abs(p - p.mean()))) < 1e-14:
        t = 0.211 * np.arange(n, dtype=np.float64)
        p = normalize(p * (1.0 + 1.5e-4 * np.cos(1.5 * t)))

    best_p = p.copy()
    best_score = fast_score(p)

    v = np.zeros(n, dtype=np.float64)
    step = 1.0
    no_improve = 0
    it = 0
    beta = beta_start
    beta_schedule_len = 1500

    while it < max_iter and time.monotonic() < deadline:
        it += 1

        if it < beta_schedule_len:
            frac = it / beta_schedule_len
            beta = float(np.exp(np.log(beta_start) + frac * (np.log(beta_end) - np.log(beta_start))))
        else:
            beta = beta_end

        b = fft_convolution(p)
        G, w = smooth_max_and_weights(b, beta)
        if not np.isfinite(G):
            break

        obj = float(n) * G
        true_score = float(2.0 * n * float(np.max(b)))

        if true_score < best_score - 1e-14:
            best_score = true_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 1000 == 0:
            print(f"    HB it={it} beta={beta:.0f} best={best_score:.8f} step={step:.5f}")

        if no_improve > 6000:
            v = np.zeros(n, dtype=np.float64)
            step = 1.0
            no_improve = 0
            if it > max_iter * 0.85:
                break

        grad = grad_smooth_max(p, w)
        grad = grad - float(np.mean(grad))

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-16:
            break

        v = momentum * v + grad

        accepted = False
        alpha = step
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.01):
            a_try = alpha * mult
            p_new = project_simplex(p - a_try * v, 1.0)
            b_new = fft_convolution(p_new)
            G_new, _ = smooth_max_and_weights(b_new, beta)
            if not np.isfinite(G_new):
                continue
            if float(n) * G_new < obj - 1e-12:
                p = p_new
                true_score_new = float(2.0 * n * float(np.max(b_new)))
                if true_score_new < best_score - 1e-14:
                    best_score = true_score_new
                    best_p = p.copy()
                    no_improve = 0
                accepted = True
                break

        if not accepted:
            step *= 0.5
            v *= 0.5
            if step < 1e-7:
                break
        else:
            if step < 2.0:
                step = min(2.0, step * 1.2)

    return best_p, best_score


def block_optimize(p0, deadline, B_target=256):
    """
    Optimize piecewise-constant block log-weights using L-BFGS-B.
    Provides a different search path than full gradient methods.
    """
    p0 = normalize(p0)
    n = p0.size
    if n < B_target or n % B_target != 0:
        return p0, fast_score(p0)

    L = n // B_target

    def block_to_seq(c):
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        z = np.zeros(B_target, dtype=np.float64)
        m = min(c.size, B_target - 1)
        z[:m] = c[:m]
        z = z - np.max(z)
        wb = np.exp(z)
        s = float(np.sum(wb))
        if s <= 0 or not np.isfinite(s):
            wb = np.ones(B_target) / B_target
        else:
            wb /= s
        H = wb / L
        return np.repeat(H, L)

    def block_obj(c):
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        z = np.zeros(B_target, dtype=np.float64)
        m = min(c.size, B_target - 1)
        z[:m] = c[:m]
        if not np.all(np.isfinite(z)):
            return 1e10, np.zeros(B_target - 1)
        z = z - np.max(z)
        wb = np.exp(z)
        s = float(np.sum(wb))
        if s <= 0:
            wb = np.ones(B_target) / B_target
        else:
            wb /= s
        H = wb / L
        p = np.repeat(H, L)
        b = fft_convolution(p)
        G, w = smooth_max_and_weights(b, 50000.0)
        if not np.isfinite(G):
            return 1e10, np.zeros(B_target - 1)
        svec = grad_smooth_max(p, w)
        g_sum = svec.reshape(B_target, L).sum(axis=1)
        Wavg = float(np.dot(wb, g_sum))
        gz = (wb / L) * (g_sum - Wavg)
        grad = n * gz[:-1]
        grad = np.where(np.isfinite(grad), grad, 0.0)
        return float(n * G), grad

    # Initial block weights from p0
    block_sums = p0.reshape(B_target, L).sum(axis=1)
    block_avg = block_sums / L
    if np.all(block_avg > 0):
        c0 = np.log(block_avg[:-1] / block_avg[-1])
    else:
        c0 = np.zeros(B_target - 1)
    c0 = np.clip(c0, -5.0, 5.0)

    bounds = [(-6.0, 6.0)] * (B_target - 1)
    try:
        res = minimize(block_obj, c0, method='L-BFGS-B', jac=True,
                       bounds=bounds,
                       options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-6})
        c_best = np.clip(res.x, -6.0, 6.0)
    except Exception:
        c_best = c0

    p_best = block_to_seq(c_best)
    return p_best, fast_score(p_best)


def propose_candidate(seed=42, budget_s=2400):
    """
    Main search: FISTA + heavy-ball + block optimization + multi-start.
    Warm-starts from height_sequence_1 if available.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 0:
        return [1.0]

    start_time = time.monotonic()
    reserve = 60.0
    deadline = start_time + max(0.0, budget_s - reserve)
    rng = np.random.default_rng(seed)

    # Get warm start
    warm_start = None
    try:
        hs = height_sequence_1
        if isinstance(hs, list) and len(hs) > 0:
            arr = np.array([float(x) for x in hs])
            if np.all(np.isfinite(arr)) and np.all(arr >= 0) and np.sum(arr) > 0:
                warm_start = normalize(arr)
                print(f"Warm start loaded: len={len(hs)}, score={fast_score(warm_start):.8f}")
    except Exception as e:
        print(f"No warm start: {e}")

    N = 8192
    best_p = None
    best_score = float('inf')
    best_N = N

    def update_best(p, sc):
        nonlocal best_p, best_score, best_N
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_p = p.copy()
            best_N = p.size

    if warm_start is not None:
        if warm_start.size != N:
            # Resample to N
            from scipy.interpolate import interp1d
            idx_old = np.linspace(0, 1, warm_start.size)
            idx_new = np.linspace(0, 1, N)
            arr_new = interp1d(idx_old, warm_start, kind='cubic', fill_value='extrapolate')(idx_new)
            warm_start = normalize(np.maximum(arr_new, 0.0))
        p0 = warm_start.copy()
    else:
        p0 = np.full(N, 1.0 / N, dtype=np.float64)

    init_score = fast_score(p0)
    print(f"Starting N={N}, init score={init_score:.8f}, budget={budget_s:.0f}s")
    update_best(p0, init_score)

    remaining = deadline - time.monotonic()
    if remaining < 30:
        return [float(x) for x in normalize(best_p)]

    # Phase 1: FISTA from warm start (main optimization)
    print("Phase 1: FISTA refinement")
    t1 = time.monotonic()
    alloc1 = min(remaining * 0.45, budget_s * 0.45)
    p1, s1 = fista_refine(p0, t1 + alloc1,
                          beta_start=10000.0, beta_end=5000000.0,
                          max_iter=400000, beta_schedule_len=2000,
                          restart_every=6000)
    update_best(p1, s1)
    print(f"  After FISTA: score={s1:.8f}")

    remaining = deadline - time.monotonic()
    if remaining < 30:
        return [float(x) for x in normalize(best_p)]

    # Phase 2: Heavy-ball from current best (different trajectory)
    print("Phase 2: Heavy-ball momentum")
    t2 = time.monotonic()
    alloc2 = min(remaining * 0.25, 300.0)
    p2, s2 = heavy_ball_refine(best_p, t2 + alloc2,
                               beta_start=20000.0, beta_end=3000000.0,
                               max_iter=200000, momentum=0.9)
    update_best(p2, s2)
    print(f"  After heavy-ball: score={s2:.8f}")

    remaining = deadline - time.monotonic()
    if remaining < 30:
        return [float(x) for x in normalize(best_p)]

    # Phase 3: Block optimization (different parameterization)
    print("Phase 3: Block L-BFGS optimization")
    t3 = time.monotonic()
    alloc3 = min(remaining * 0.2, 600.0)
    p3, s3 = block_optimize(best_p, t3 + alloc3, B_target=256)
    update_best(p3, s3)
    print(f"  After block opt: score={s3:.8f}")

    # Refine block result with FISTA
    if np.isfinite(s3) and s3 < 2.0:
        remaining = deadline - time.monotonic()
        if remaining > 60:
            t3b = time.monotonic()
            p3b, s3b = fista_refine(p3, t3b + min(remaining * 0.5, 300.0),
                                    beta_start=50000.0, beta_end=5000000.0,
                                    max_iter=100000, beta_schedule_len=1000,
                                    restart_every=5000)
            update_best(p3b, s3b)
            print(f"  After FISTA from block: score={s3b:.8f}")

    remaining = deadline - time.monotonic()
    if remaining < 30:
        return [float(x) for x in normalize(best_p)]

    # Phase 4: Multi-start with perturbations
    print("Phase 4: Perturbation multi-start")
    for trial in range(4):
        remaining = deadline - time.monotonic()
        if remaining < 30:
            break
        # Creating perturbation
        eps = 0.005 * (trial + 1)
        pert = eps * rng.standard_normal(best_N)
        # Also try structured perturbations
        if trial == 1:
            t_struct = 2.0 * np.pi * np.arange(best_N) / best_N
            pert = 0.01 * np.cos(3 * t_struct)
        elif trial == 2:
            t_struct = 2.0 * np.pi * np.arange(best_N) / best_N
            pert = 0.008 * np.sin(5 * t_struct)
        elif trial == 3:
            pert = 0.003 * rng.standard_normal(best_N)

        p_pert = normalize(best_p + pert)
        p_pert_score = fast_score(p_pert)
        print(f"  Trial {trial}: perturbed score={p_pert_score:.8f}")

        t4 = time.monotonic()
        alloc4 = min(remaining * 0.3, 200.0)
        p4, s4 = fista_refine(p_pert, t4 + alloc4,
                              beta_start=50000.0, beta_end=5000000.0,
                              max_iter=80000, beta_schedule_len=500,
                              restart_every=4000)
        update_best(p4, s4)
        print(f"  Trial {trial} result: score={s4:.8f}")

    remaining = deadline - time.monotonic()
    if remaining < 20:
        return [float(x) for x in normalize(best_p)]

    # Phase 5: Final FISTA polish with high beta
    print("Phase 5: Final high-beta FISTA polish")
    t5 = time.monotonic()
    p5, s5 = fista_refine(best_p, t5 + min(remaining * 0.7, 600.0),
                          beta_start=500000.0, beta_end=10000000.0,
                          max_iter=100000, beta_schedule_len=500,
                          restart_every=3000)
    update_best(p5, s5)
    print(f"  Final polish: score={s5:.8f}")

    # Phase 6: Try N=4096 if we have time
    remaining = deadline - time.monotonic()
    if remaining > 60:
        N2 = 4096
        if warm_start is not None:
            idx_old = np.linspace(0, 1, warm_start.size)
            idx_new = np.linspace(0, 1, N2)
            p2_0 = normalize(np.maximum(
                np.interp(idx_new, idx_old, warm_start), 0.0))
        else:
            p2_0 = np.full(N2, 1.0 / N2)

        print(f"Phase 6: Trying N={N2}")
        t6 = time.monotonic()
        p6, s6 = fista_refine(p2_0, t6 + min(remaining * 0.6, 300.0),
                              beta_start=10000.0, beta_end=5000000.0,
                              max_iter=100000, beta_schedule_len=1000,
                              restart_every=5000)
        update_best(p6, s6)
        print(f"  N={N2} score={s6:.8f}")

    # Final output
    print(f"\nBest score: {best_score:.8f} at N={best_N}")
    final_p = normalize(best_p)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_seq = [float(x) for x in final_p]

    # Verify
    try:
        off = float(evaluate_sequence(final_seq))
        print(f"Official evaluation: {off:.8f}")
        if not np.isfinite(off) or off > 2.0:
            # Fallback
            fb = [1.0] * best_N
            return fb
    except Exception as e:
        print(f"Eval error: {e}")

    return final_seq
