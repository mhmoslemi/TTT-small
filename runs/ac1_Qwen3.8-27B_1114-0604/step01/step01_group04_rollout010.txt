softmax subgradients, backtracking line search, and multi-scale polish.

import numpy as np
import time

# ---------- Utility functions ----------

def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    d = v.size
    if d == 0:
        return v.copy()
    u = np.sort(v)[::-1]
    css = np.cumsum(u)
    rho = np.arange(1, d + 1, dtype=np.float64)
    mask = u * rho > (css - 1.0)
    if not mask.any():
        return np.ones(d) / d
    r = np.where(mask)[0][-1]
    theta = (css[r] - 1.0) / (r + 1.0)
    return np.maximum(v - theta, 0.0)

def fft_conv(a):
    """Full linear autocorrelation a*a via FFT, result length 2n-1."""
    n = a.size
    out_len = 2 * n - 1
    fsize = 1 << (out_len - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:out_len]

    """Symmetric sequence of length n from p on simplex (sum p = 1)."""
    n = int(n)
    m = n // 2
    a = np.zeros(n)
    if n % 2 == 0:
        a[:m] = 0.5 * p
        a[m:] = 0.5 * p[::-1]
    else:
        a[:m] = 0.5 * p[:m]
        a[m] = p[m]
