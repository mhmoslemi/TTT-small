import numpy as np

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = np.sum(e)
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(z) / len(z)
    return e / s

def _fft_conv(a, N):
    N = 1 << (3 * len(a) - 2).bit_length()
    L = 2 * len(a) - 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L].real
    c = np.maximum(c, 0.0)
    return c

def _compute_h(a, p, N):
    """Compute h[i] = sum_k p[k] * a[k-i] via FFT."""
    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    h = np.fft.irfft(fp * np.conjugate(fa), N)[:len(a)].real
    return h

def _grad_z(z, tau, N):
    """Compute gradient of smoothed objective w.r.t. z."""
    a = _softmax(z)
    n = len(a)
    L = 2 * n - 1
    c = _fft_conv(a, N)
    
    # Compute p = softmax(c / tau)
    c_shifted = c / tau - np.max(c) / tau
    c_shifted = np.clip(c_shifted, -745, 0)
    e = np.exp(c_shifted)
    se = np.sum(e)
    if se <= 0 or not np.isfinite(se):
        return 0.0, np.zeros(n)
    p = e / se
    
    # Compute h
    h = _compute_h(a, p, N)
    
    # Gradient
    s_h = np.dot(h, a)
    grad = a * (h - s_h)
    
    # Value
    val = np.max(c) + tau * np.log(se)
    
    return val, grad

    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1**t)
    v_hat = v / (1 - beta2**t)

import time

"""
Algorithm: Adam-based optimization of smoothed autocorrelation peak in log-space.
Uses adaptive tau schedule, multiple diverse starts, and coordinate descent polish.
"""

def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = float(np.sum(e))
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(z, dtype=np.float64) / z.size
    return e / s

def _fft_conv(a, N):
    n = a.size
    L = 2 * n - 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L].real
    np.maximum(c, 0.0, out=c)
    return c

def _fast_score(a):
    n = a.size
    s = float(np.sum(a))
    if s <= 1e-12:
        return np.inf
    c = _fft_conv(a, _NFFT)
    m = float(np.max(c))
    return 2.0 * n * m / (s * s)

def _grad_and_val(z, tau):
    global _N, _NFFT
    a = _softmax(z)
    n = _N
    c = _fft_conv(a, _NFFT)
    
    # Smoothed max
    cm = float(np.max(c))
    diff = (c - cm) / tau
    diff = np.clip(diff, -745.0, 0.0)
    e = np.exp(diff)
    se = float(np.sum(e))
    if se <= 0 or not np.isfinite(se):
        return cm, np.zeros(n)
    p = e / se
    
    # h[i] = sum_k p[k] * a[k-i]
    fa = np.fft.rfft(a, _NFFT)
    fp = np.fft.rfft(p, _NFFT)
    h = np.fft.irfft(fp * np.conjugate(fa), _NFFT)[:n].real
    
    # Gradient of smoothed max w.r.t. z
    s_h = float(np.dot(h, a))
    grad = a * (h - s_h)
    
    val = cm + tau * np.log(se)
    return val, grad

def _adam_optimize(z0, schedule, deadline):
    """
    Run Adam optimization with a schedule of (tau, n_steps, lr) tuples.
    Returns the best z found.
    """
    global _N
    n = _N
    z = z0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    best_z = z.copy()
    best_val = np.inf
    
    for tau, n_steps, lr in schedule:
        if time.monotonic() > deadline - 30:
            break
        for step in range(n_steps):
            if time.monotonic() > deadline - 30:
                break
            val, grad = _grad_and_val(z, tau)
            
            if np.isfinite(val) and val < best_val:
                best_val = val
                best_z = z.copy()
            
            # Adam step
            t += 1
            grad = np.clip(grad, -100.0, 100.0)
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * grad * grad
            m_hat = m / (1 - 0.9**t)
            v_hat = v / (1 - 0.999**t)
            z = z - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            
            if step % 5000 == 0:
                a = _softmax(z)
                sc = _fast_score(a)
                print(f"  Adam tau={tau:.6f} step={step} score={sc:.9f} best_val={best_val:.9f} time_left={deadline-time.monotonic():.0f}s")
    
    return best_z, best_val

def _coord_descent(a, deadline, max_passes=5):
    """Coordinate descent polish: try adjusting individual elements."""
    global _N, _NFFT
    n = _N
    a = a.copy()
    best_score = _fast_score(a)
    rng = np.random.default_rng(12345)
    
    for pass_num in range(max_passes):
        if time.monotonic() > deadline - 10:
            break
        if pass_num == 0:
            indices = np.arange(n)
        else:
            # Only try elements with large gradient contribution
            c = _fft_conv(a, _NFFT)
            peak = float(np.max(c))
            diff = (c - peak) / 1e-4
            diff = np.clip(diff, -745.0, 0.0)
            e = np.exp(diff)
            se = float(np.sum(e))
            p = e / se
            fa = np.fft.rfft(a, _NFFT)
            fp = np.fft.rfft(p, _NFFT)
            h = np.fft.irfft(fp * np.conjugate(fa), _NFFT)[:n].real
            s_h = float(np.dot(h, a))
            grad_mag = np.abs(a * (h - s_h))
            # Pick top 2000 indices by gradient magnitude
            top_k = min(2000, n)
            indices = np.argsort(-grad_mag)[:top_k]
        
        for i in indices:
            if time.monotonic() > deadline - 10:
                break
            # Try small perturbations

                    a_new[i] = max(0, a[i] + sign * delta * a[i])

                    a_new /= s
                    sc = _fast_score(a_new)
                    if sc < best_score - 1e-12:
                        a = a_new
                        best_score = sc
                        break
    
    return a, best_score

    rng = np.random.default_rng(int(seed))
    starts = []

    # 1. Provided sequence (resampled)
    try:
        g = globals().get("height_sequence_1")
        if g is not None:
            g = np.asarray(g, dtype=np.float64).ravel()
            if g.size > 1 and np.all(np.isfinite(g)):
                g = np.clip(g, 1e-300, 1000.0)
                s_g = float(np.sum(g))
                if s_g > 0:
                    g = g / s_g
                    z = np.log(np.maximum(g, 1e-15))
                    z = z - np.mean(z)
                    x_old = np.linspace(0, 1, g.size)
                    x_new = np.linspace(0, 1, n)
                    z = np.interp(x_new, x_old, z)
                    a = _softmax(z)
                    sc = _fast_score(a)
                    starts.append((sc, z.copy()))
                    print(f"Provided seq (n={g.size} -> {n}): score={sc:.9f}")
    except Exception as ex:
        print(f"Could not use provided sequence: {ex}")

    # 2. Uniform
    z = np.zeros(n)
    a = _softmax(z)
    sc = _fast_score(a)
    starts.append((sc, z.copy()))

    # 3. Two-level blocks
    for ratio in (1.3, 1.6, 2.0, 2.5, 3.0, 4.0, 6.0):
        f = np.ones(n)
        f[:n // 2] = 1.0
        f[n // 2:] = ratio
        z = np.log(f) - np.mean(np.log(f))
        a = _softmax(z)
        sc = _fast_score(a)
        starts.append((sc, z.copy()))

    # 4. Power-law profiles
    for alpha in (0.15, 0.3, 0.5, 0.7, 1.0, 1.5):
        x = np.linspace(1, n, n)
        f = np.power(x, -alpha)
        f = f / f.sum()
        z = np.log(np.maximum(f, 1e-15))
        z = z - np.mean(z)
        a = _softmax(z)
        sc = _fast_score(a)
        starts.append((sc, z.copy()))

    # 5. Gaussian / tent profiles
    x = np.linspace(-1, 1, n)
    for sig in (0.3, 0.5, 0.7, 1.0):
        f = np.exp(-x ** 2 / (2 * sig ** 2))
        z = np.log(f) - np.mean(np.log(f))
        a = _softmax(z)
        sc = _fast_score(a)
        starts.append((sc, z.copy()))

    for peak_h in (0.3, 0.5, 0.7, 0.9):
        f = 1.0 - peak_h * np.abs(x)
        f = np.maximum(f, 0.01)
        z = np.log(f) - np.mean(np.log(f))
        a = _softmax(z)
        sc = _fast_score(a)
        starts.append((sc, z.copy()))

    # 6. Random smooth low-frequency
    u = np.linspace(0, 1, n)
    for _ in range(10):
        f = np.ones(n)
        for m in range(1, 12):
            f += 0.12 * rng.standard_normal() * np.sin(2 * np.pi * m * u)
            f += 0.12 * rng.standard_normal() * np.cos(2 * np.pi * m * u)
        f = np.maximum(f, 1e-6)
        z = np.log(f) - np.mean(np.log(f))
        z = np.clip(z, -30, 30)
        a = _softmax(z)
        sc = _fast_score(a)
        starts.append((sc, z.copy()))

    # Sort by initial score
    starts.sort(key=_start_score_key)
    print(f"\nTotal starts: {len(starts)}, top scores: {[f'{s:.6f}' for s, _ in starts[:5]]}")

    best_score = np.inf
    best_a = np.ones(n) / n
    best_z = np.zeros(n)

    # Allocate time: 50% to start 0, 25% to start 1, 15% to start 2, 10% to rest
    reserve_time = 300.0  # for polish + reheat
    total_opt_time = max(60.0, (deadline - t_start) - reserve_time)
    time_alloc = [0.45, 0.25, 0.15, 0.075, 0.075]
    n_starts_to_try = min(5, len(starts))

    for si in range(n_starts_to_try):
        if time.monotonic() > deadline - reserve_time:
            break
        sc0, z0 = starts[si]
        alloc = time_alloc[si] * total_opt_time
        this_deadline = min(t_start + alloc, deadline - reserve_time)
        if this_deadline <= time.monotonic():
            this_deadline = time.monotonic() + 30

        print(f"\n  Start {si} (init={sc0:.9f}), budget={this_deadline - time.monotonic():.0f}s")
        z_best, sc_best = _run_adam(z0, this_deadline)
        a_best = _softmax(z_best)
        sc_best = _fast_score(a_best)
        print(f"  Start {si} result: {sc_best:.9f} (elapsed={time.monotonic() - t_start:.0f}s)")

        if sc_best < best_score:
            best_score = sc_best
            best_a = a_best.copy()
            best_z = z_best.copy()
            print(f"  *** New best: {best_score:.9f}")

    # Coordinate descent polish
    if time.monotonic() < deadline - 30:
        cd_deadline = deadline - 30
        print(f"\n  Coord descent polish (budget={cd_deadline - time.monotonic():.0f}s)")
        pa, ps = _coord_descent(best_a, cd_deadline)
        if ps < best_score:
            best_score = ps
            best_a = pa.copy()
            print(f"  After CD: {best_score:.9f}")

    # Reheating
    if time.monotonic() < deadline - 60:
        print(f"\n  Reheating (budget={deadline - time.monotonic():.0f}s)")
        for noise in (0.15, 0.08, 0.04, 0.02):
            if time.monotonic() > deadline - 60:
                break
            z_pert = best_z + noise * rng.standard_normal(n)
            z_pert = np.clip(z_pert, -45.0, 45.0)
            reheat_dl = min(deadline - 60, time.monotonic() + 150)
            z_r, sc_r = _run_adam(z_pert, reheat_dl)
            a_r = _softmax(z_r)
            sc_r = _fast_score(a_r)
            if sc_r < best_score:
                best_score = sc_r
                best_a = a_r.copy()
                best_z = z_r.copy()
                print(f"  Reheat noise={noise}: new best {best_score:.9f}")

    # Final validation and safety
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = float(np.sum(best_a))
    if s < 0.01 or not np.isfinite(s):
        best_a = np.ones(n) / n
    else:
        best_a = best_a / s

    try:
        exact = evaluate_sequence(best_a.tolist())
        print(f"\nOfficial evaluator score: {float(exact):.9f}")
    except Exception:
        pass

    print(f"Done. Best={best_score:.9f}, n={n}, time={time.monotonic() - t_start:.0f}s")
    return [float(v) for v in best_a.tolist()]
