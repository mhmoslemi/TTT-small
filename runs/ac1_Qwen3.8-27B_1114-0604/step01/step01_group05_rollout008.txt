def fft_convolution(a):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    """Smooth max and softmax weights."""
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float('inf'), np.empty(0)
    m = float(np.max(b))
    if not np.isfinite(m):
        return float('inf'), np.zeros_like(b)
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b) / b.size
        return m, w
    x = np.clip(beta * (b - m), -700.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return G, w

    if not np.isfinite(M):
        return float('inf')
    return 2.0 * n * M / (S * S)


def correlation(p, w):
    """s_i = sum_j p[j] * w[i+j], computed via FFT."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n:m][::-1].copy()
    return np.maximum(s, 0.0)


def project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if target <= 0.0:
        return np.zeros(n)
    if not np.all(np.isfinite(v)):
        return np.full(n, target / n)
    # Binary search for threshold
    sorted_v = np.sort(v)[::-1]
    csum = np.cumsum(sorted_v)
    rho = np.where(csum - sorted_v * np.arange(1, n + 1) > target)[0]
    if len(rho) == 0:
        return np.full(n, target / n)
    rho = rho[0]
    theta = (csum[rho] - target) / (rho + 1.0)
    q = np.maximum(v - theta, 0.0)
    s = float(np.sum(q))
    if s > 0 and np.isfinite(s):
        q *= target / s
    else:
        q = np.full(n, target / n)
    return q


def normalize_simplex(p):
    """Normalize to sum=1, non-negative."""
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0:
        return np.full(p.shape, 1.0 / p.size)
    return p / s


def block_to_p(n, B, c):
    """Convert block log-weights to sequence."""
    L = n // B
    z = np.zeros(B)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        m = min(B - 1, c.size)
        z[:m] = c[:m]
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0 or not np.isfinite(s):
        wb = np.ones(B) / B
    else:
        wb /= s
    H = wb / L
    return np.repeat(H, L)

def fista(p0, deadline, beta, max_iter=500000):
    """
    FISTA (Nesterov accelerated proximal gradient) on the simplex.
    Minimizes n * smooth_max(p*p) subject to p >= 0, sum(p) = 1.
    """
    p = normalize_simplex(p0)
    n = p.size
    best_p = p.copy()
    best_score = fast_score(p)
    
    # Initialize
    y = p.copy()
    t = 1.0
    step = 1.0  # Will be adapted
    min_step = 1e-8
    max_step = 50.0
    
    no_improve = 0
    it = 0
    
    # Break symmetry
    mean_p = float(np.mean(p))
    if n > 1 and float(np.max(np.abs(p - mean_p))) < 1e-14:
        t_arr = 0.1 * np.arange(n, dtype=np.float64)
        y = normalize_simplex(p * (1.0 + 1e-4 * np.sin(t_arr)))
        p = y.copy()
    
    while it < max_iter and time.monotonic() < deadline:
        it += 1
        
        # Compute gradient at y
        b = fft_convolution(y)
        G, w = smooth_max_and_weights(b, beta)
        if not np.isfinite(G):
            break
        
        svec = correlation(y, w)
        grad = 2.0 * n * svec
        # Project gradient onto the hyperplane sum(x)=1 (remove mean)
        grad = grad - float(np.mean(grad))
        
        # FISTA update
        q = project_simplex(y - step * grad, 1.0)
        
        # Check if step is good (sufficient decrease)
        qb = fft_convolution(q)
        Gq, _ = smooth_max_and_weights(qb, beta)
        
        if Gq > G + 0.5 * step * float(np.dot(grad, q - y)):
            # Step too large, reduce
            step = max(min_step, step * 0.5)
            continue
        
        # Update momentum
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y_new = q + (t - 1.0) / t_new * (q - p)
        y_new = np.maximum(y_new, 0.0)
        # Re-normalize y_new (FISTA extrapolation can go off simplex)
        s = float(np.sum(y_new))
        if s > 0 and np.isfinite(s):
            y_new *= 1.0 / s
        else:
            y_new = q.copy()
        
        p = q
        t = t_new
        
        # Track best
        cur_score = float(2.0 * n * float(np.max(qb)))
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
            # Increase step if improving
            if step < max_step:
                step = min(max_step, step * 1.05)
        else:
            no_improve += 1
        
        # Periodic reports
        if it % 5000 == 0:
            print(f"FISTA it={it} score={best_score:.8f} step={step:.4f}")
        
        if no_improve > 50000:
            # Restart momentum
            y = p.copy()
            t = 1.0
            step = max(min_step, step * 0.5)
            no_improve = 0
            if no_improve > 100000:
                break
    
    return best_p, best_score


def lp_direction(f, M):
    """
    Solve LP: max sum(b) s.t. (f*b)[k] <= M for all k, b >= 0.
    Returns the optimal b (direction) or None if infeasible/too slow.
    """
    n = f.size
    if not HAVE_SCIPY:
        return None
    
    # (f*b)[k] = sum_i f[i] * b[k-i] for k = 0..2n-2
    # This is a matrix A of size (2n-1) x n
    # A[k, i] = f[i] if 0 <= k-i < n, else 0
    
    # For memory efficiency, we'll use the sparse structure
    try:
        from scipy import sparse
        # Build constraint matrix A (2n-1) x n
        # A[k, i] = f[i] if max(0, k-n+1) <= i <= min(n-1, k)
        rows = []
        cols = []
        vals = []
        for k in range(2 * n - 1):
            i_start = max(0, k - n + 1)
            i_end = min(n, k + 1)
            for i in range(i_start, i_end):
                rows.append(k)
                cols.append(i)
                vals.append(f[i])
        A = sparse.coo_matrix((vals, (rows, cols)), shape=(2 * n - 1, n)).tocsc()
        
        # Objective: maximize sum(b) -> minimize -sum(b)
        c_obj = -np.ones(n)
        
        res = linprog(c_obj, A_ub=A, b_ub=np.full(2 * n - 1, M),
                      bounds=[(0, None)] * n, method='highs',
                      options={'time_limit': 30})
        
        if res.success and res.x is not None:
            return np.maximum(res.x, 0.0)
    except Exception as e:
        print(f"LP error: {e}")
    
    return None


def lp_iteration(f, deadline, max_iters=50):
    """
    Iterate the LP-based update:
    1. Compute M = max(f*f)
    2. Solve LP to get direction g
    3. Update f = (1-t)*f + t*g
    """
    n = f.size
    t = 0.1  # Step size for mixing
    it = 0
    
    while it < max_iters and time.monotonic() < deadline:
        it += 1
        f = normalize_simplex(f)
        b = fft_convolution(f)
        M = float(np.max(b))
        cur_score = 2.0 * n * M
        
        g = lp_direction(f, M)
        if g is None:
            break
        
        s = float(np.sum(g))
        if s <= 0:
            break
        
        g = g / s  # Normalize to sum 1
        
        # Update
        f_new = (1.0 - t) * f + t * g
        f_new = normalize_simplex(f_new)
        
        new_score = fast_score(f_new)
        if new_score < cur_score:
            f = f_new
            print(f"LP iter {it}: score {cur_score:.8f} -> {new_score:.8f}")
        else:
            # Try smaller step
            t = max(0.01, t * 0.5)
            f_new = (1.0 - t) * f + t * g
            f_new = normalize_simplex(f_new)
            new_score = fast_score(f_new)
            if new_score < cur_score:
                f = f_new
            else:
                break
    
    return f
