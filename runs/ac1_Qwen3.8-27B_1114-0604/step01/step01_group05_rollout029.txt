import numpy as np
import time

def fft_convolution(a):
    """Compute linear convolution a*a using FFT."""
    n = a.size
    if n == 0:
        return np.empty(0)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)

def score_of(p):
    """Compute the score 2*n*max(p*p)/(sum(p))^2."""
    n = p.size
    S = p.sum()
    if S <= 0:
        return float('inf')
    b = fft_convolution(p)
    M = b.max()
    return 2.0 * n * M / (S * S)

def normalize(p):
    """Normalize to sum=1, non-negative."""
    p = np.maximum(p, 0.0)
    s = p.sum()
    if s <= 0:
        return np.ones_like(p) / p.size
    return p / s

def linear_ramp(n, direction=1):
    """Create a linear ramp sequence normalized to sum 1."""
    a = np.arange(1, n+1, dtype=np.float64)
    if direction == -1:
        a = a[::-1]
    return a / a.sum()

def quadratic_ramp(n, direction=1):
    """Create a quadratic ramp sequence."""
    a = np.arange(1, n+1, dtype=np.float64)**2
    if direction == -1:
        a = a[::-1]
    return a / a.sum()

def exponential_ramp(n, alpha=2.0, direction=1):
    """Create an exponential ramp sequence."""
    x = np.arange(n, dtype=np.float64)
    a = np.exp(alpha * x / n)
    if direction == -1:
        a = a[::-1]
    return a / a.sum()
