"""
Minimize 2*n*max(autocorr)/(sum)^2 using multi-scale Adam optimization
with smooth max continuation, starting from known good sequences.

Strategy:
1. Multi-scale: optimize at increasing lengths (256 -> 512 -> 1024 -> ... -> 8192/16384)

import numpy as np
import time

def fft_convolution(a):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)

def smooth_max(b, beta):
    """Smooth max and gradient weights."""
    m = float(np.max(b))
    if m <= 0.0:
        return m, np.ones_like(b) / b.size
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + np.log(Z) / beta
    return float(G), w

def corr_pw(p, w):
    """s_i = sum_j p_j * w_{i+j} for i=0..n-1 (cross-correlation)."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n : m][::-1].copy()
    return s

def project_simplex(v, target=1.0):
    """Project onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        return np.full(n, target / n, dtype=np.float64)
    lo = -10.0
    hi = 10.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0:
        return np.full(n, target / n, dtype=np.float64)
    return q * (target / s)

def normalize_simplex(p):
    p = np.maximum(np.asarray(p, dtype=np.float64), 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.size, 1.0 / p.size, dtype=np.float64)
    return p / s

def fast_score(p):
    """Compute 2*n*max(b) (assuming sum(p)=1)."""
    b = fft_convolution(p)
    M = float(np.max(b))
    n = p.size
    return 2.0 * n * M

def adam_smooth_max_optimize(p0, n, beta_schedule, lr_schedule, max_iter, deadline, verbose=True):
    """
    Adam optimizer with smooth max on the simplex.
    p0: initial point on simplex (sum=1, non-negative)
    beta_schedule: list of (beta, n_iters) pairs or a single beta
    """
    p = normalize_simplex(p0)
    m = p.size
    v = np.zeros(m, dtype=np.float64)  # first moment
    u = np.zeros(m, dtype=np.float64)  # second moment
    t = 0
    best_p = p.copy()
    best_score = fast_score(p)
    
    if isinstance(beta_schedule, (int, float)):
        beta_schedule = [(beta_schedule, max_iter)]
    if isinstance(lr_schedule, (int, float)):
        lr_schedule = [(lr_schedule, max_iter)]
    
    beta_idx = 0
    lr_idx = 0
    beta_iters_done = 0
    lr_iters_done = 0
    
    while t < max_iter and time.monotonic() < deadline:
        # Update beta and lr if needed
        if beta_idx < len(beta_schedule) and beta_iters_done >= beta_schedule[beta_idx][1]:
            beta_idx += 1
            if beta_idx >= len(beta_schedule):
                beta_idx = len(beta_schedule) - 1
        if lr_idx < len(lr_schedule) and lr_iters_done >= lr_schedule[lr_idx][1]:
            lr_idx += 1
            if lr_idx >= len(lr_schedule):
                lr_idx = len(lr_schedule) - 1
        
        beta = beta_schedule[beta_idx][0]
        base_lr = lr_schedule[lr_idx][0]
        # Decay lr within the current phase
        phase_total = beta_schedule[beta_idx][1]
        frac = beta_iters_done / max(1, phase_total)
        lr = base_lr * (1.0 - 0.5 * frac)  # linear decay to half
        
        b = fft_convolution(p)
        G, w = smooth_max(b, beta)
        
        if not np.isfinite(G):
            break
        
        cur_score = fast_score(p)
        if cur_score < best_score - 1e-10:
            best_score = cur_score
            best_p = p.copy()
        
        # Gradient: d/dp_i of n*G = 2 * sum_j p_j * w_{i+j} (the "smoothing" gradient)
        svec = corr_pw(p, w)
        svec = np.maximum(svec, 0.0)
        g = 2.0 * svec
        # Subtract mean for scale invariance (gradient on simplex)
        g = g - float(np.mean(g))
        
        t += 1
        beta_iters_done += 1
        lr_iters_done += 1
        
        # Adam update
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        v = beta1 * v + (1 - beta1) * g
        u = beta2 * u + (1 - beta2) * (g * g)
        v_hat = v / (1 - beta1 ** t)
        u_hat = u / (1 - beta2 ** t)
        
        p_new = p - lr * v_hat / (np.sqrt(u_hat) + eps)
        p_new = project_simplex(p_new, 1.0)
        
        # Check if step improved the smooth objective
        if t % 10 == 0 or t <= 5:
            b_new = fft_convolution(p_new)
            G_new, _ = smooth_max(b_new, beta)
            if np.isfinite(G_new) and G_new > G * 1.01 and t > 10:
                # Rejected - shrink lr
                p = p  # keep old
                v = np.zeros(m, dtype=np.float64)  # reset momentum
                u = np.zeros(m, dtype=np.float64)
            else:
                p = p_new
        else:
            p = p_new
    
    return best_p, best_score

def upsample(p, factor=2):
    """Upsample by linear interpolation."""
    n = p.size
    new_n = n * factor
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n, dtype=np.float64)
    p_new = np.interp(x_new, x_old, p)
    return normalize_simplex(p_new)

    # Random smooth

def coordinate_descent_polish(p, deadline, verbose=True):
    """
    Coordinate descent: for each coordinate, find the best value
    (given others fixed) by golden-section search or simple grid.
    """
    p = normalize_simplex(p)
    n = p.size
    best_score = fast_score(p)
    
    order = np.random.default_rng(123).permutation(n)
    
    for _ in range(3):  # 3 passes
        if time.monotonic() >= deadline:
            break
        for i in order:
            if time.monotonic() >= deadline:
                break
            # Try a few values for p[i]
            old_val = p[i]
            remaining_sum = 1.0 - old_val
            if remaining_sum <= 0:
                continue
            
            best_val = old_val
            best_sc = fast_score(p)
            
            # Grid of candidate values
            candidates = [0.0, 0.25 * remaining_sum, 0.5 * remaining_sum, 0.75 * remaining_sum, remaining_sum, old_val * 0.5, old_val * 1.5]
            for cv in candidates:
                cv = max(0.0, min(cv, remaining_sum))
                p[i] = cv
                # Renormalize (but we're fixing others, so just set and check)
                sc = fast_score(p)
                if sc < best_sc:
                    best_sc = sc
                    best_val = cv
            p[i] = best_val
    
    p = normalize_simplex(p)
    return p, fast_score(p)

import numpy as np
import time


def fft_convolution(a):
    """Linear convolution a*a via FFT, clipped to non-negative."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_and_grad_weights(b, beta):
    """Return smooth max value and softmax weight vector."""
    m = float(np.max(b))
    if m <= 0.0 or beta <= 0:
        return m, np.ones_like(b) / b.size
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w


def cross_corr(p, w):
    """Compute s_i = sum_j p_j * w_{i+j} for i=0..n-1."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(max(n, 1), dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    # s_i = C[m-1-i] for i=0..n-1
    s = C[m - n:m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target} using bisection."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if not np.all(np.isfinite(v)):
        return np.full(n, target / float(n), dtype=np.float64)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, target / float(n), dtype=np.float64)
    return q * (target / s)


def normalize_simplex(p):
    """Clip to non-negative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(p.size, 1.0 / float(p.size), dtype=np.float64)
    return p / s


def fast_score(p):
    """Compute 2*n*max(b) for normalized p (sum=1)."""
    b = fft_convolution(p)
    return 2.0 * float(np.max(b)) * float(p.size)


def interpolate_sequence(p, new_n):
    """Resample sequence to new length via linear interpolation."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == new_n:
        return p.copy()
    if n == 1:
        return np.full(new_n, p[0], dtype=np.float64)
    x_old = np.arange(n, dtype=np.float64)
    x_new = np.linspace(0, n - 1, new_n, dtype=np.float64)
    p_new = np.interp(x_new, x_old, p)
    return normalize_simplex(p_new)


def momentum_optimize(p0, beta_schedule, lr0, deadline, max_iter, momentum0=0.9):
    """
    Momentum SGD with backtracking line search on the simplex.
    Optimizes n * smooth_max(a*a, beta) with continuation on beta.
    
    beta_schedule: list of (beta, n_iters) tuples
    lr0: initial learning rate
    """
    p = normalize_simplex(p0)
    n = p.size
    v = np.zeros(n, dtype=np.float64)  # momentum
    best_p = p.copy()
    best_score = fast_score(p)
    lr = lr0
    no_improve = 0
    
    # Precompute iteration boundaries
    total_iters = sum(nc for _, nc in beta_schedule)
    if max_iter > 0:
        total_iters = min(total_iters, max_iter)
    
    t = 0
    for beta, n_iters in beta_schedule:
        if t >= total_iters:
            break
        for _ in range(n_iters):
            if t >= total_iters:
                break
            if time.monotonic() >= deadline:
                return best_p, best_score
            t += 1
            
            b = fft_convolution(p)
            G, w = smooth_max_and_grad_weights(b, beta)
            if not np.isfinite(G):
                return best_p, best_score
            
            # True score tracking
            cur_score = fast_score(p)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_p = p.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            if no_improve > 5000 and t > 1000:
                # Try to escape: increase lr temporarily
                lr = min(lr * 2.0, lr0)
                no_improve = 0
                if no_improve > 8000:
                    break
            
            # Gradient of n*G w.r.t. p
            svec = cross_corr(p, w)
            svec = np.maximum(svec, 0.0)
            g = 2.0 * svec
            # Remove mean component (tangent to simplex)
            g = g - float(np.mean(g))
            
            # Momentum update
            v = momentum0 * v - lr * g
            
            # Backtracking line search
            p_new = None
            for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.01):
                q = project_simplex(p + mult * v, 1.0)
                qb = fft_convolution(q)
                Gq, _ = smooth_max_and_grad_weights(qb, beta)
                if np.isfinite(Gq) and Gq <= G + 1e-12:
                    p_new = q
                    break
            
            if p_new is None:
                # All steps rejected, shrink lr
                lr *= 0.5
                v *= 0.5
                if lr < 1e-10:
                    break
            else:
                p = p_new
                # Gradually restore lr
                if lr < lr0:
                    lr = min(lr * 1.1, lr0)
        
        # Between phases, decay momentum slightly for faster adaptation
        v *= 0.5

def diverse_starts(n, rng, count=10):

        print(f"Polish result: {sc_check:.8f}")
        
        if sc_check < best_score_global:
            best_score_global = sc_check
            best_p_global = p_polished.copy()
            
            # Another round with even higher beta
            if time.monotonic() < polish_deadline - 30:
                if n <= 2048:
                    ultra_beta = [(10000000, 500), (50000000, 1500), (200000000, 3000)]
                else:
                    ultra_beta = [(50000000, 500), (200000000, 1500), (1000000000, 3000)]
                ultra_lr = polish_lr * 0.3
                p_ultra, sc_ultra = momentum_optimize(best_p_global, ultra_beta, ultra_lr, polish_deadline, 5000)
                sc_check2 = fast_score(p_ultra)
                print(f"Ultra polish: {sc_check2:.8f}")
                if sc_check2 < best_score_global:
                    best_score_global = sc_check2
                    best_p_global = p_ultra.copy()
    
    if best_p_global is None:
        return [1.0] * 8192
    
    # Convert to output format
    final_p = normalize_simplex(best_p_global)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_s = float(np.sum(final_p))
    if final_s > 0 and np.isfinite(final_s):
        final_p /= final_s
    
    final_seq = [float(x) for x in final_p]
    
    # Official verification
    off = official_score(final_seq)
    print(f"\nFinal official score: {off:.8f} (n={len(final_seq)})")
    
    if not np.isfinite(off) or off > 2.0 + 1e-6:
        print("Falling back to uniform")
        return [1.0] * len(final_seq)
    
    return final_seq
