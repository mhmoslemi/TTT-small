- Adaptive beta schedule for the smooth max approximation

import numpy as np

def fft_convolution(a):
    """Linear convolution a*a using FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def smooth_max_weights(b, beta):
    """Compute smooth max value and softmax weights."""
    m = float(np.max(b))
    if not np.isfinite(m):
        return float('inf'), np.full(b.size, 1.0 / b.size, dtype=np.float64)
    if m <= 0.0:
        return 0.0, np.full(b.size, 1.0 / b.size, dtype=np.float64)
    x = beta * (b - m)
    x = np.clip(x, -800.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return G, w


def correlation_for_grad(a, w):
    """Compute grad[i] = 2 * sum_j w[i+j] * a[j] via FFT."""
    n = a.size
    m = w.size
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    A = np.fft.rfft(a, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(A * WR, size)[:need]
    # s_i = sum_j w[i+j] * a[j] = C[m-1-i] ... let me rederive
    # C[k] = sum_j a[j] * w_rev[k-j] where w_rev = w[::-1]
    # w_rev[k-j] = w[m-1-(k-j)] = w[m-1-k+j]
    # C[k] = sum_j a[j] * w[m-1-k+j]
    # We want s_i = sum_j w[i+j] * a[j]
    # Let k = m-1-i: C[m-1-i] = sum_j a[j] * w[m-1-(m-1-i)+j] = sum_j a[j] * w[i+j]
    # So s_i = C[m-1-i]
    s = C[m - n:m][::-1].copy()
    return 2.0 * s


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = v.size
    if n == 0:
        return v
    if target <= 0:
        return np.zeros_like(v)
    v = np.maximum(v, 0.0)
    s = float(np.sum(v))
    if s <= 0 or not np.isfinite(s):
        return np.full(n, target / n, dtype=np.float64)
    # Bisection for the threshold
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s_mid = float(np.sum(np.maximum(v - mid, 0.0)))
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    sq = float(np.sum(q))
    if sq <= 0 or not np.isfinite(sq):
        return np.full(n, target / n, dtype=np.float64)
    return q * (target / sq)


def normalize_simplex(p):
    """Clip to nonnegative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0 or not np.isfinite(s):
        return np.full(p.shape, 1.0 / p.size, dtype=np.float64)
    return p / s


def fast_score(p):
    """Compute 2*n*max(a*a) with sum(a)=1."""
    n = p.size
    b = fft_convolution(p)
    M = float(np.max(b))
    return float(2.0 * n * M)


def mirror_descent_step(a, grad, lr):
    """KL mirror descent step on the simplex."""
    eps = 1e-30
    log_a = np.log(a + eps)
    # Center gradient to preserve sum approximately
    gc = grad - float(np.mean(grad))
    log_a -= lr * gc
    log_a = log_a - float(np.max(log_a))
    new_a = np.exp(log_a)
    s = float(np.sum(new_a))
    if s <= 0 or not np.isfinite(s):
        return a.copy()
    return new_a / s


def resample_to_length(p, target_n):
    """Resample a sequence to target length using linear interpolation."""
    n = p.size
    if n == target_n:
        return p.copy()
    if n == 0:
        return np.full(target_n, 1.0 / target_n, dtype=np.float64)
    if n == 1:
        return np.full(target_n, 1.0 / target_n, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, n, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, target_n, dtype=np.float64)
    p_new = np.interp(x_new, x_old, p)
    return normalize_simplex(p_new)


def make_smooth_random(n, rng, n_frequencies=5):
    """Generate a random smooth positive sequence."""
    freqs = rng.integers(1, n_frequencies + 1, size=n_frequencies)
    phases = rng.uniform(0, 2 * np.pi, size=n_frequencies)
    amps = rng.uniform(0.3, 1.0, size=n_frequencies)
    x = np.zeros(n, dtype=np.float64)
    for f, ph, am in zip(freqs, phases, amps):
        x += am * np.sin(2.0 * np.pi * f * np.arange(n) / n + ph)
    z = 0.8 * x
    p = np.exp(z)
    return normalize_simplex(p)


def make_structured(n, rng, idx):
    """Generate structured initializations."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    if idx == 0:
        # Sine bump
        z = 1.5 * np.sin(np.pi * x)
        p = np.exp(z)
    elif idx == 1:
        # Cosine
        z = 1.5 * np.cos(2 * np.pi * x)
        p = np.exp(z)
    elif idx == 2:
        # Two bumps
        z = 1.5 * np.sin(2 * np.pi * x) * np.exp(-2.0 * (x - 0.5)**2 / 0.05)
        p = np.exp(z)
    elif idx == 3:
        # Triangle wave
        z = 1.0 * np.abs(2.0 * x - 1.0)
        p = np.exp(z)
    else:
        p = make_smooth_random(n, rng, n_frequencies=3 + idx)
    return normalize_simplex(p)


def run_gradient_phase(a, deadline, initial_beta, final_beta, n_iters_max,
                       use_mirror=False, mirror_lr=0.05):
    """
    Run adaptive smooth-max gradient descent.
    
    Parameters:
    - a: initial sequence (sum=1)
    - deadline: time deadline
    - initial_beta: starting beta for smooth max
    - final_beta: ending beta for smooth max
    - n_iters_max: max iterations
    - use_mirror: if True, use mirror descent instead of Euclidean projection
    - mirror_lr: learning rate for mirror descent
    
    Returns: (best_a, best_score)
    """
    a = normalize_simplex(a.copy())
    n = a.size
    best_a = a.copy()
    best_score = fast_score(a)
    
    step = 1.0
    no_improve = 0
    it = 0
    
    total_iters = max(1, min(n_iters_max, int((deadline - time.monotonic()) / 0.005)))
    
    while it < total_iters and time.monotonic() < deadline:
        it += 1
        # Adaptive beta: geometric interpolation
        t = it / total_iters
        beta = initial_beta * (final_beta / initial_beta) ** t
        
        b = fft_convolution(a)
        G, w = smooth_max_weights(b, beta)
        if not np.isfinite(G):
            break
        
        cur_score = float(2.0 * n * np.max(b))
        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 5000:
            if not use_mirror:
                # Try mirror descent for a bit
                use_mirror = True
                no_improve = 0
            else:
                break
        
        grad = correlation_for_grad(a, w)
        grad = np.maximum(grad, 0.0)
        gc = grad - float(np.mean(grad))
        
        gn = float(np.linalg.norm(gc))
        if not np.isfinite(gn) or gn < 1e-15:
            break
        
        accepted = False
        if use_mirror:
            # Mirror descent with adaptive lr
            lr = mirror_lr
            for _ in range(5):
                new_a = mirror_descent_step(a, gc, lr)
                nb = fft_convolution(new_a)
                nM = float(np.max(nb))
                nG, _ = smooth_max_weights(nb, beta)
                if np.isfinite(nG) and nG < G - 1e-13:
                    a = new_a
                    G = nG
                    accepted = True
                    break
                lr *= 0.5
            if not accepted:
                if mirror_lr < 1e-6:
                    break
                mirror_lr *= 0.5
        else:
            # Euclidean projected gradient with backtracking
            for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.01, 0.005, 0.001):
                alpha = step * mult
                new_a = project_simplex(a - alpha * gc, 1.0)
                nb = fft_convolution(new_a)
                nG, _ = smooth_max_weights(nb, beta)
                if np.isfinite(nG) and nG < G - 1e-13:
                    a = new_a
                    G = nG
                    accepted = True
                    break
            if not accepted:
                step *= 0.5
                if step < 1e-8:
                    break
            else:
                if step < 3.0:
                    step = min(3.0, step * 1.3)
    
    return best_a, best_score


def run_multi_phase(a0, deadline):
    """
    Run multi-phase optimization on a single starting point.
    Phase 1: Low beta, Euclidean (exploration)
    Phase 2: Medium beta, Euclidean (refinement)
    Phase 3: High beta, mirror descent (polish)
    Phase 4: Very high beta, Euclidean with tiny steps (final polish)
    """
    a = normalize_simplex(a0.copy())
    best_a = a.copy()
    best_score = fast_score(a)
    
    now = time.monotonic()
    remaining = deadline - now
    if remaining <= 5.0:
        return best_a, best_score
    
    # Phase 1: Exploration (30% of time)
    p1_deadline = time.monotonic() + 0.30 * remaining
    a, sc = run_gradient_phase(a, p1_deadline, initial_beta=200, final_beta=2000,
                               n_iters_max=100000, use_mirror=False)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    print(f"  Phase 1: score={sc:.8f}, best={best_score:.8f}")
    
    # Phase 2: Refinement (30% of time)
    now = time.monotonic()
    remaining = deadline - now
    if remaining <= 5.0:
        return best_a, best_score
    p2_deadline = time.monotonic() + 0.30 * remaining
    a, sc = run_gradient_phase(a, p2_deadline, initial_beta=2000, final_beta=20000,
                               n_iters_max=100000, use_mirror=False)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    print(f"  Phase 2: score={sc:.8f}, best={best_score:.8f}")
    
    # Phase 3: Mirror descent polish (25% of time)
    now = time.monotonic()
    remaining = deadline - now
    if remaining <= 5.0:
        return best_a, best_score
    p3_deadline = time.monotonic() + 0.25 * remaining
    a, sc = run_gradient_phase(a, p3_deadline, initial_beta=10000, final_beta=100000,
                               n_iters_max=100000, use_mirror=True, mirror_lr=0.02)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    print(f"  Phase 3: score={sc:.8f}, best={best_score:.8f}")
    
    # Phase 4: Final Euclidean polish (15% of time)
    now = time.monotonic()
    remaining = deadline - now
    if remaining <= 5.0:
        return best_a, best_score
    p4_deadline = time.monotonic() + 0.15 * remaining
    a, sc = run_gradient_phase(a, p4_deadline, initial_beta=50000, final_beta=500000,
                               n_iters_max=100000, use_mirror=False)
    if sc < best_score:
        best_score = sc
        best_a = a.copy()
    print(f"  Phase 4: score={sc:.8f}, best={best_score:.8f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence minimizing 2*n*max(conv(a,a))/(sum(a))^2.
    
    Uses multi-phase adaptive smooth-max gradient descent with:
    - N=16384 for good theoretical bounds
    - Multiple diverse initializations
    - Adaptive beta schedule
    - Both Euclidean and mirror descent
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0
    
    if budget_s <= 30.0:
        # Minimal fallback
        N = 1024
        return [1.0 / N] * N
    
    reserve = 120.0 if budget_s > 600.0 else 40.0
    start_time = time.monotonic()
    deadline = start_time + max(10.0, budget_s - reserve)
    
    # Choose N based on budget
    if budget_s >= 1800.0:
        N = 16384
    elif budget_s >= 900.0:
        N = 10240
    else:
        N = 8192
    
    rng = np.random.default_rng(seed)
    
    print(f"Starting search: N={N}, budget={budget_s:.0f}s, deadline in {deadline - start_time:.0f}s")
    
    # Build list of initializations
    inits = []
    
    # 1. Try to use height_sequence_1 if available
    try:
        hs1 = np.array(height_sequence_1, dtype=np.float64)
        if hs1.size > 1 and np.all(np.isfinite(hs1)) and np.sum(hs1) > 0:
            hs1_norm = normalize_simplex(np.maximum(hs1, 0.0))
            hs1_resampled = resample_to_length(hs1_norm, N)
            inits.append(('height_seq', hs1_resampled))
            print(f"Loaded height_sequence_1 (len={hs1.size}), resampled to {N}")
            sc = fast_score(hs1_resampled)
            print(f"  Initial score from height_sequence_1: {sc:.8f}")
    except Exception:
        pass
    
    # 2. Uniform
    inits.append(('uniform', np.full(N, 1.0 / N, dtype=np.float64)))
    
    # 3. Random smooth initializations
    for i in range(4):
        p = make_smooth_random(N, rng, n_frequencies=3 + i)
        inits.append((f'random_{i}', p))
    
    # 4. Structured initializations
    for i in range(4):
        p = make_structured(N, rng, i)
        inits.append((f'struct_{i}', p))
    
    # Score all initializations
    scored_inits = []
    for name, p in inits:
        sc = fast_score(p)
        if np.isfinite(sc):
            scored_inits.append((sc, name, p))
    
    scored_inits.sort(key=lambda x: x[0])
    
    print("Initializations sorted by score:")
    for sc, name, _ in scored_inits[:5]:
        print(f"  {name}: {sc:.8f}")
    
    # Allocate time to top initializations
    # Give 60% to best, 25% to second, 15% to third
    top_inits = scored_inits[:3]
    fractions = [0.55, 0.25, 0.20]
    
    overall_best_a = top_inits[0][2].copy()
    overall_best_score = top_inits[0][0]
    
    for i, (sc, name, p) in enumerate(top_inits):
        remaining = deadline - time.monotonic()
        if remaining <= 30.0:
            break
        
        frac = fractions[i] if i < len(fractions) else 0.1
        alloc = remaining * frac
        sub_deadline = time.monotonic() + alloc
        
        print(f"\n=== Optimizing init '{name}' (score={sc:.8f}), time alloc={alloc:.0f}s ===")
        
        a_best, sc_best = run_multi_phase(p, sub_deadline)
        
        if sc_best < overall_best_score:
            overall_best_score = sc_best
            overall_best_a = a_best.copy()
            print(f"  NEW OVERALL BEST: {sc_best:.8f}")
    
    # Final: try a few more perturbations of the best solution if time remains
    remaining = deadline - time.monotonic()
    if remaining > 60.0:
        print(f"\n=== Perturbation phase: {remaining:.0f}s remaining ===")
        for attempt in range(5):
            if time.monotonic() > deadline - 30:
                break
            # Add small structured perturbation
            x = np.arange(N, dtype=np.float64)
            freq = rng.integers(2, 20)
            phase = rng.uniform(0, 2 * np.pi)
            amp = 0.02 * (1 + attempt * 0.5)
            perturbed = overall_best_a * (1.0 + amp * np.sin(2 * np.pi * freq * x / N + phase))
            perturbed = normalize_simplex(perturbed)
            
            sub_dl = min(deadline - 30, time.monotonic() + remaining * 0.15)
            a_b, sc_b = run_gradient_phase(perturbed, sub_dl,
                                           initial_beta=10000, final_beta=200000,
                                           n_iters_max=50000, use_mirror=True, mirror_lr=0.01)
            if sc_b < overall_best_score:
                overall_best_score = sc_b
                overall_best_a = a_b.copy()
                print(f"  Perturbation {attempt}: NEW BEST {sc_b:.8f}")
            else:
                print(f"  Perturbation {attempt}: {sc_b:.8f} (no improvement)")
    
    # Final normalization
    final_a = normalize_simplex(overall_best_a)
    final_a = np.maximum(final_a, 0.0)
    final_a = np.minimum(final_a, 1000.0)
    
    # Verify with official evaluator
    final_seq = [float(x) for x in final_a]
    try:
        off = evaluate_sequence(final_seq)
        off = float(off)
    except Exception:
        off = float('inf')
    
    if np.isfinite(off):
        print(f"\nFinal official score: {off:.8f}")
    else:
        print("\nOfficial evaluation failed, using fallback")
        final_seq = [1.0 / N] * N
    
    # Safety: if score is worse than uniform (2.0), return uniform
    try:
        if np.isfinite(off) and off > 2.0 + 1e-6:
            final_seq = [1.0 / N] * N
    except Exception:
        pass
    
    elapsed = time.monotonic() - start_time
    print(f"\nTotal time: {elapsed:.1f}s")
    print(f"Returning sequence of length {N}")
    
    return final_seq
