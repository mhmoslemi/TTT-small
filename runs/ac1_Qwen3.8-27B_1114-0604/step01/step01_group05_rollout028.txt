    n = len(a)

    s = float(np.sum(e))

def corr_grad(a, w):
    """Compute s_i = sum_j a_j * w[i+j] for i=0..n-1."""
    n = len(a)

    return s


def score_normalized(a):
    """Score = 2*N*max(conv(a,a)) assuming sum(a)=1."""
    N = len(a)
    if N == 0:
        return float('inf')
    b = fft_conv(a)
    M = float(np.max(b))
    if not np.isfinite(M):
        return float('inf')
    return float(2.0 * N * M)


def compute_grad_smooth(w, beta):
    """Gradient w.r.t. log-weights using smooth-max."""
    N = len(w)
    a = softmax_fn(w)
    b = fft_conv(a)
    G, ws = smax_vals(b, beta)
    if not np.isfinite(G):
        return float('inf'), np.zeros(N)
    s = corr_grad(a, ws)
    sa = float(np.dot(s, a))
    grad = 2.0 * a * (s - sa)
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(2.0 * N * G), grad


def compute_grad_hot(w, hot_indices, K):
    """Gradient w.r.t. log-weights using only hot lags."""
    N = len(w)
    a = softmax_fn(w)
    b = fft_conv(a)
    topK = min(K, len(b))
    idx = np.argsort(b)[::-1][:topK]
    ws = np.zeros(len(b))
    ws[idx] = 1.0 / topK
    s = corr_grad(a, ws)
    sa = float(np.dot(s, a))
    grad = 2.0 * a * (s - sa)
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(2.0 * N * float(np.max(b))), grad


def block_init(N, B, deadline, rng):
    """Coarse block optimization for initialization."""
    L = N // B
    if L < 1:
        w = np.zeros(N)
        return w

    def block_obj(c):
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        z = np.zeros(B)
        if B > 1:
            c_clean = np.clip(c, -8.0, 8.0)
            z[:-1] = c_clean[:B-1]
        z = z - np.max(z)
        wb = np.exp(z)
        wb /= wb.sum()
        H = wb / L
        p = np.repeat(H, L)
        p = np.maximum(p, 1e-15)
        p /= p.sum()
        b = fft_conv(p)
        G, _ = smax_vals(b, 3000.0)
        if not np.isfinite(G):
            return float('inf')
        return float(2.0 * N * G)

    best_c = np.zeros(B - 1)
    best_val = block_obj(best_c)

    if HAVE_SCIPY:
        inits = [np.zeros(B - 1)]
        x = np.linspace(0, 1, B - 1)
        inits.append(0.5 * np.sin(2 * np.pi * x))
        inits.append(0.3 * np.cos(4 * np.pi * x))
        inits.append(0.2 * rng.standard_normal(B - 1))

        bounds = [(-4.0, 4.0)] * (B - 1)
        for c0 in inits:
            if time.monotonic() > deadline:
                break
            try:
                res = minimize(block_obj, c0, method='L-BFGS-B', bounds=bounds,
                              options={'maxiter': 40, 'ftol': 1e-12})
                v = block_obj(res.x)
                if v < best_val:
                    best_val = v
                    best_c = res.x.copy()
            except Exception:
                pass

    z = np.zeros(B)
    if B > 1:
        z[:-1] = best_c
    z = z - np.max(z)
    wb = np.exp(z)
    wb /= wb.sum()
    H = wb / L
    p = np.repeat(H, L)
    p = np.maximum(p, 1e-15)
    p /= p.sum()
    return np.log(p)


def main_optimize(w0, deadline, rng, N):
    """
    Main optimization: adaptive smooth/hot subgradient with backtracking.
    """
    w = w0.copy()
    a = softmax_fn(w)
    b = fft_conv(a)
    best_score = float(2.0 * N * np.max(b))
    best_w = w.copy()
    best_a = a.copy()

    lr = 0.5
    no_improve = 0
    it = 0

    while time.monotonic() < deadline:
        it += 1

        # Adaptive beta schedule
        frac = min(1.0, it / 8000.0)
        if frac < 0.05:
            beta = 500.0
        elif frac < 0.15:
            beta = 2000.0
        elif frac < 0.30:
            beta = 8000.0
        elif frac < 0.50:
            beta = 30000.0
        elif frac < 0.70:
            beta = 80000.0
        elif frac < 0.90:
            beta = 150000.0
        else:
            beta = 300000.0

        # Mixed gradient: smooth most steps, hot every 4th
        if it % 4 == 0:
            _, grad = compute_grad_hot(w, None, K=15)
        else:
            _, grad = compute_grad_smooth(w, beta)

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-16:
            break

        # Backtracking line search
        cur_score = float(2.0 * N * np.max(fft_conv(softmax_fn(w))))
        accepted = False

        for mult in [1.0, 0.5, 0.25, 0.125, 0.0625, 0.03, 0.01]:
            step = lr * mult
            w_trial = w - step * grad
            a_trial = softmax_fn(w_trial)
            b_trial = fft_conv(a_trial)
            score_trial = float(2.0 * N * np.max(b_trial))

            if score_trial < cur_score - 1e-10:
                w = w_trial
                a = a_trial
                cur_score = score_trial
                accepted = True
                break

        if accepted:
            if cur_score < best_score:
                best_score = cur_score
                best_w = w.copy()
                best_a = a.copy()
            no_improve = 0
            if lr < 4.0:
                lr = min(4.0, lr * 1.03)
        else:
            lr *= 0.5
            no_improve += 1
            if lr < 1e-6:
                w = best_w + 0.015 * rng.standard_normal(N)
                lr = 0.5
                no_improve = 0

        if it % 300 == 0:
            print(f"  opt it={it} best={best_score:.8f} beta={beta:.0f} lr={lr:.4f} no_imp={no_improve}")

        if no_improve > 1500:
            w = best_w + 0.04 * rng.standard_normal(N)
            lr = 0.5
            no_improve = 0

    return best_w, best_a, best_score
