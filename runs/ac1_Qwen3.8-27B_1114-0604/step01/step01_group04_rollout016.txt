def coordinate_descent_step(a, c, k, n):
    """Optimize coordinate k of a (given current convolution c). Returns new a_k."""
    a_k = a[k]
    # Find affected m values: m in [k, min(k+n-1, 2n-2)]
    m_start = k
    m_end = min(k + n - 1, 2 * n - 2)
    
    if m_start > m_end:
        return a_k  # no change
    
    # For each affected m, compute the coefficient of x in c'_m
    # c'_m(x) = c_m + L_m * (x - a_k) for m != 2k
    # c'_m(x) = c_m - a_k^2 + x^2 for m = 2k
    affected_m = np.arange(m_start, m_end + 1)
    
    # Compute c_unaffected
    if m_start == 0 and m_end == 2*n-2:
        c_unaff = 0.0
    else:
        mask = np.ones(2*n-1, dtype=bool)
        mask[m_start:m_end+1] = False
        c_unaff = c[mask].max() if mask.any() else 0.0
    
    # For affected m (m != 2k): c'_m = c_m - 2*a_k*a_{m-k} + 2*x*a_{m-k}
    # = (c_m - 2*a_k*a_{m-k}) + 2*a_{m-k}*x
    # So it's linear: intercept + slope * x
    # For m = 2k: c'_m = c_m - a_k^2 + x^2
    
    # Build the list of (intercept, slope) for linear parts, and note the quadratic part
    linear_consts = []
    linear_coeffs = []
    has_quad = False
    quad_const = 0.0  # c_{2k} - a_k^2
    
    for m in affected_m:
        idx = m - k  # this is the index of a in the term a_{m-k}
        if m == 2 * k:
            has_quad = True
            quad_const = c[m] - a_k * a_k
        else:
            # c'_m = c[m] - 2*a_k*a[idx] + 2*x*a[idx]
            linear_consts.append(c[m] - 2.0 * a_k * a[idx])
            linear_coeffs.append(2.0 * a[idx])
    
    linear_consts = np.array(linear_consts) if linear_consts else np.array([0.0])
    linear_coeffs = np.array(linear_coeffs) if linear_coeffs else np.array([0.0])
    
    # Now define f(x) = 2n * max(c_unaff, max(linear_consts + linear_coeffs*x), quad_const + x^2) / (1 - a_k + x)^2
    # Minimize over x >= 0
    
    def f(x):
        vals = np.maximum(linear_consts + linear_coeffs * x, c_unaff)
        if has_quad:
            vals = np.maximum(vals, quad_const + x * x)
        denom = (1.0 - a_k + x) ** 2
        if denom < 1e-30:
            return np.inf
        return 2.0 * n * vals.max() / denom

    result = minimize_scalar(f, bounds=(0, x_max), method='bounded',

"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using:
1. Exact 1D coordinate descent (optimize one coordinate at a time via bounded scalar minimization)
2. LP-based direction finding (maximize sum(b) s.t. a*b <= M, b >= 0)
3. Combined iterative scheme with adaptive restarts
4. Multiple sequence lengths and diverse initializations
Uses FFT for convolution. Incremental convolution updates for speed.
"""

import numpy as np
import time
from scipy.optimize import linprog, minimize_scalar


def full_conv(a):
    """Full linear convolution via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def score(a):
    """Compute the objective score."""
    n = a.size
    s = a.sum()
    if s < 0.01:
        return np.inf
    b = full_conv(a)
    return 2.0 * n * b.max() / (s * s)


def project_simplex(v):
    """Project vector v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def init_uniform(n):
    return np.ones(n) / n


def init_two_cluster(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * (0.15 + 0.15 * rng.random())
    c2 = n * (0.70 + 0.15 * rng.random())
    sigma = n * (0.04 + 0.08 * rng.random())
    a = np.exp(-((i - c1) / sigma) ** 2) + np.exp(-((i - c2) / sigma) ** 2)
    a += 0.02 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_u_shape(n):
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i).astype(np.float64) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_sided(n):
    i = np.arange(n, dtype=np.float64)
    c = (n - 1) / 2.0
    a = ((i - c) / c) ** 2 + 0.1
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.05
    return a / a.sum()


def init_cosine(n, rng):
    i = np.arange(n, dtype=np.float64)
    freq = 1 + int(rng.integers(0, 3))
    a = (1.0 + freq * np.cos(2.0 * np.pi * i / max(n - 1, 1))) ** 2
    a += 1e-6
    return a / a.sum()


def init_noisy_uniform(n, rng):
    a = np.ones(n) + 0.8 * rng.standard_normal(n)
    a = np.abs(a) + 0.01
    return a / a.sum()


def init_sqrt_decay(n, rng):
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a += 0.05 * np.abs(rng.standard_normal(n))
    return a / a.sum()


def init_alternating(n, rng):
    """Alternating high/low pattern."""
    a = np.ones(n)
    a[::2] = 2.0 + 0.5 * rng.random(n // 2)
    a[1::2] = 0.5 + 0.5 * rng.random((n - 1) // 2)
    return a / a.sum()


def get_inits(n, rng):
    inits = []
    inits.append(init_uniform(n))
    inits.append(init_two_cluster(n, rng))
    inits.append(init_u_shape(n))
    inits.append(init_sided(n))
    inits.append(init_random(n, rng))
    inits.append(init_cosine(n, rng))
    inits.append(init_noisy_uniform(n, rng))
    inits.append(init_sqrt_decay(n, rng))
    inits.append(init_alternating(n, rng))
    return inits


def coord_optimize(a, c, k, n):
    """
    Optimize coordinate k of a, given current convolution c.
    Returns (new_a_k, new_partial_max) where new_partial_max is the max of
    the affected portion of the new convolution.
    """
    a_k = a[k]
    if a_k < 1e-15:
        return a_k

    # Affected m values: m in [k, min(k+n-1, 2n-2)]
    m_start = k
    m_end = min(k + n - 1, 2 * n - 2)

    if m_start > m_end:
        return a_k

    # Compute c_unaffected (max of convolution outside affected region)
    if m_start == 0 and m_end == 2 * n - 2:
        c_unaff = 0.0
    else:
        if m_start > 0:
            left_max = c[:m_start].max()
        else:
            left_max = 0.0
        if m_end < 2 * n - 2:
            right_max = c[m_end + 1:].max()
        else:
            right_max = 0.0
        c_unaff = max(left_max, right_max)

    # Build linear and quadratic components for affected region
    affected_m = np.arange(m_start, m_end + 1)
    n_aff = len(affected_m)

    # For m != 2k: c'_m = c[m] - 2*a_k*a[m-k] + 2*x*a[m-k]
    # For m == 2k: c'_m = c[2k] - a_k^2 + x^2
    idx_2k = 2 * k
    has_quad = (m_start <= idx_2k <= m_end)

    # Precompute for linear parts
    # a[m-k] for m in affected_m where m != 2k
    offsets = affected_m - k  # these are the indices into a
    # Filter to valid a indices: 0 <= offset <= n-1
    valid = (offsets >= 0) & (offsets <= n - 1)

    lin_intercepts = []
    lin_coeffs = []
    for ii in range(n_aff):
        m = affected_m[ii]
        if m == idx_2k:
            continue
        off = offsets[ii]
        if 0 <= off < n:
            lin_intercepts.append(c[m] - 2.0 * a_k * a[off])
            lin_coeffs.append(2.0 * a[off])

    if lin_intercepts:
        lin_int = np.array(lin_intercepts)
        lin_coef = np.array(lin_coeffs)
    else:
        lin_int = np.array([0.0])
        lin_coef = np.array([0.0])

    if has_quad:
        quad_const = c[idx_2k] - a_k * a_k
    else:
        quad_const = -np.inf

    a_k_save = a_k
    c_unaff_save = c_unaff
    n_save = n
    li = lin_int
    lc = lin_coef
    qc = quad_const
    hq = has_quad

    def f(x):
        vals = np.maximum(li + lc * x, c_unaff_save)
        if hq:
            vals = np.maximum(vals, qc + x * x)
        denom = (1.0 - a_k_save + x) ** 2
        if denom < 1e-30:
            return 1e30
        return 2.0 * n_save * vals.max() / denom

    # Bounds for x
    x_lo = 0.0
    x_hi = max(2.0 * a_k_save + 0.05, 0.1)

    try:
        res = minimize_scalar(f, bounds=(x_lo, x_hi), method='bounded',
                              options={'xatol': 1e-10, 'maxiter': 100})
        new_x = max(0.0, res.x)
    except Exception:
        new_x = a_k_save

    return new_x


def coordinate_descent(a, time_budget, rng, verbose=False):
    """
    Run coordinate descent on a (normalized to sum=1).
    Does full sweeps, updating convolution incrementally.
    Returns (best_a, best_score).
    """
    n = a.size
    a = np.maximum(a, 0.0).copy()
    a = a / a.sum()
    c = full_conv(a)
    best_a = a.copy()
    best_sc = 2.0 * n * c.max()
    cur_sc = best_sc

    t0 = time.monotonic()
    sweep = 0
    stall = 0
    last_improve = 0
    order = np.arange(n)
    rng.shuffle(order)

    while time.monotonic() - t0 < time_budget:
        for k in order:
            if time.monotonic() - t0 > time_budget:
                break

            old_a_k = a[k]
            new_a_k = coord_optimize(a, c, k, n)

            if abs(new_a_k - old_a_k) < 1e-14:
                continue

            # Update convolution incrementally
            m_start = k
            m_end = min(k + n - 1, 2 * n - 2)
            if m_start <= m_end:
                idx_2k = 2 * k
                for m in range(m_start, m_end + 1):
                    if m == idx_2k:
                        c[m] = c[m] - old_a_k * old_a_k + new_a_k * new_a_k
                    else:
                        off = m - k
                        if 0 <= off < n:
                            c[m] = c[m] - 2.0 * old_a_k * a[off] + 2.0 * new_a_k * a[off]

            a[k] = new_a_k

        # End of sweep
        cur_sc = 2.0 * n * c.max()
        if cur_sc < best_sc - 1e-12:
            best_sc = cur_sc
            best_a = a.copy()
            last_improve = sweep
            stall = 0
        else:
            stall += 1

        sweep += 1

        # Adaptive strategy
        if stall > 5:
            # Restart with perturbation
            a = project_simplex(best_a + 0.03 * rng.standard_normal(n))
            c = full_conv(a)
            cur_sc = 2.0 * n * c.max()
            stall = 0
            order = np.arange(n)
            rng.shuffle(order)
        elif sweep % 10 == 0:
            # Reshuffle order
            order = np.arange(n)
            rng.shuffle(order)

        if verbose and sweep % 100 == 0:
            print(f"  CoordDesc sweep {sweep}: score={best_sc:.8f}")

    # Final check
    final_sc = 2.0 * n * full_conv(a).max()
    if final_sc < best_sc:
        best_sc = final_sc
        best_a = a.copy()

    return best_a, best_sc


def lp_update(a, M):
    """
    Solve LP: max sum(b) s.t. (a*b)_k <= M for all k, b >= 0.
    Returns b or None if infeasible.
    """
    n = a.size
    # Build constraint matrix A_ub: (2n-1) x n
    # (a*b)_k = sum_i a[k-i] * b_i for valid i
    # A_ub[k, i] = a[k-i] if 0 <= k-i <= n-1, else 0

    A_ub = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        i_lo = max(0, k - n + 1)
        i_hi = min(n - 1, k)
        if i_lo <= i_hi:
            # A_ub[k, i] = a[k-i]
            idx = k - np.arange(i_lo, i_hi + 1)
            A_ub[k, i_lo:i_hi + 1] = a[idx]

    b_ub = np.full(2 * n - 1, M)
    c_obj = -np.ones(n)
    bounds = [(0, None)] * n

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')

    if res.success and res.x is not None:
        b = np.maximum(res.x, 0.0)
        s = b.sum()
        if s > 1e-10:
            return b / s
    return None


def lp_iterations(a, time_budget, rng, max_lp_calls=200):
    """
    Iterative LP-based refinement.
    Each iteration: compute M, solve LP for best b, mix.
    Returns (best_a, best_score).
    """
    n = a.size
    a = np.maximum(a, 0.0).copy()
    a = a / a.sum()
    c = full_conv(a)
    best_a = a.copy()
    best_sc = 2.0 * n * c.max()

    t0 = time.monotonic()
    iter_count = 0
    lp_calls = 0
    stall = 0
    t_mix = 0.3

    while time.monotonic() - t0 < time_budget and lp_calls < max_lp_calls:
        c = full_conv(a)
        cur_max = c.max()
        if cur_max < 1e-15:
            break

        M = cur_max
        b = lp_update(a, M)
        lp_calls += 1

        if b is None:
            # Infeasible; try with slightly larger M
            b = lp_update(a, M * 1.01)
            if b is None:
                break

        # Mix
        t = t_mix * (0.9 ** (stall // 3))
        a_new = (1.0 - t) * a + t * b
        a_new = np.maximum(a_new, 0.0)
        a_new = a_new / a_new.sum()

        c_new = full_conv(a_new)
        new_sc = 2.0 * n * c_new.max()

        if new_sc <= 2.0 * n * cur_max:
            a = a_new
            if new_sc < best_sc - 1e-12:
                best_sc = new_sc
                best_a = a.copy()
                stall = 0
            else:
                stall += 1
        else:
            # Try smaller step
            t_half = t * 0.3
            a_half = (1.0 - t_half) * a + t_half * b
            a_half = np.maximum(a_half, 0.0)
            a_half = a_half / a_half.sum()
            sc_half = 2.0 * n * full_conv(a_half).max()
            if sc_half < best_sc:
                a = a_half
                best_sc = sc_half
                best_a = a.copy()
                stall = 0
            else:
                stall += 1

        if stall > 8:
            a = project_simplex(best_a + 0.02 * rng.standard_normal(n))
            stall = 0
            t_mix = 0.2 + 0.2 * rng.random()

        iter_count += 1
        if verbose := (iter_count % 10 == 0):
            print(f"  LP iter {iter_count}: best={best_sc:.8f}, cur={2.0*n*full_conv(a).max():.8f}")

    return best_a, best_sc


def subgrad_optimize(a, time_budget, rng):
    """
    Projected smoothed-subgradient descent (from previous approach, slightly improved).
    """
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = 2.0 * n * full_conv(a).max()

    eta = 0.2
    tau_frac = 0.03
    beta = 0.4
    vel = np.zeros(n)

    t0 = time.monotonic()
    iters = 0
    last_improve = 0
    stall = 0

    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_max = b.max()
        cur_sc = 2.0 * n * cur_max

        tau = tau_frac * cur_max
        if tau < 1e-18:
            tau = 1e-18

        # Smoothed subgradient
        shift = b / tau
        shift -= shift.max()
        w = np.exp(shift)
        s = w.sum()
        if s < 1e-300:
            w = np.zeros(b.size)
            w[int(np.argmax(b))] = 1.0
        else:
            w /= s

        g = 2.0 * np.correlate(w, a, mode='valid')

        vel = beta * vel + (1.0 - beta) * g
        a_new = project_simplex(a - eta * vel)

        b_new = full_conv(a_new)
        new_sc = 2.0 * n * b_new.max()

        if new_sc <= cur_sc or rng.random() < 0.05:
            a = a_new
        else:
            a = project_simplex(a - 0.3 * eta * vel)

        if new_sc < best_sc - 1e-12:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = iters
            stall = 0
        else:
            stall += 1

        iters += 1

        if stall > 3000:
            a = project_simplex(best_a + 0.04 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.1 + 0.2 * rng.random()
            tau_frac = 0.02 + 0.04 * rng.random()
            last_improve = iters
            stall = 0

        if iters % 5000 == 0:
            eta = 0.1 + 0.3 * rng.random()
            beta = 0.3 + 0.3 * rng.random()

    return best_a, best_sc


def polish(a, time_budget, rng):
    """
    Local polish: coordinate descent + small random perturbations.
    """
    n = a.size
    a = np.maximum(a, 0.0).copy()
    a = a / a.sum()
    c = full_conv(a)
    best_a = a.copy()
    best_sc = 2.0 * n * c.max()

    t0 = time.monotonic()
    order = np.arange(n)
    rng.shuffle(order)
    sweep = 0
    stall = 0

    while time.monotonic() - t0 < time_budget:
        improved = False
        for k in order:
            if time.monotonic() - t0 > time_budget:
                break
            old_val = a[k]
            new_val = coord_optimize(a, c, k, n)
            if abs(new_val - old_val) > 1e-14:
                m_start = k
                m_end = min(k + n - 1, 2 * n - 2)
                idx_2k = 2 * k
                for m in range(m_start, m_end + 1):
                    if m == idx_2k:
                        c[m] = c[m] - old_val * old_val + new_val * new_val
                    else:
                        off = m - k
                        if 0 <= off < n:
                            c[m] = c[m] - 2.0 * old_val * a[off] + 2.0 * new_val * a[off]
                a[k] = new_val
                improved = True

        sweep += 1
        cur_sc = 2.0 * n * c.max()
        if cur_sc < best_sc - 1e-12:
            best_sc = cur_sc
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 20:
            # Try small perturbation

            c = full_conv(a)
            stall = 0
            order = np.arange(n)
            rng.shuffle(order)

    return best_a, best_sc

            # Phase 3: Subgradient polish (if time allows)
            remain = deadline - time.monotonic() - 2.0
            if remain > 3.0:
                tb_sg = min(tb_init * 0.15, remain * 0.4)
                a_sg, sc_sg = subgrad_optimize(a_cd, tb_sg, rng)
                if sc_sg < best_overall_sc:
                    best_overall_sc = sc_sg
                    best_overall = a_sg.copy()
                    print(f"  [SG] n={nv}, init={ii}: {sc_sg:.6f} (BEST)")
                a_cd = a_sg

        print(f"[done] n={nv}. Overall best: {best_overall_sc:.6f}")

    # Final polish phase on the best solution found
    if best_overall is not None:
        remain = deadline - time.monotonic() - 3.0
        if remain > 5.0:
            print(f"[polish] remaining={remain:.1f}s, n={best_overall.size}, score={best_overall_sc:.6f}")
            a_p, sc_p = polish(best_overall, min(remain * 0.7, 300.0), rng)
            if sc_p < best_overall_sc:
                best_overall_sc = sc_p
                best_overall = a_p.copy()
                print(f"[polish] improved to {sc_p:.6f}")

            # Final subgradient polish
            remain2 = deadline - time.monotonic() - 3.0
            if remain2 > 5.0:
                a_sg, sc_sg = subgrad_optimize(a_p, min(remain2 * 0.5, 100.0), rng)
                if sc_sg < best_overall_sc:
                    best_overall_sc = sc_sg
                    best_overall = a_sg.copy()
                    print(f"[final SG] improved to {sc_sg:.6f}")

    # Fallback
    if best_overall is None:
        best_overall = np.ones(128) / 128
        best_overall_sc = 2.0

    # Ensure non-negative
    best_overall = np.maximum(best_overall, 0.0)

    # Final verification
    result = [float(x) for x in best_overall]
    final_sc = float(evaluate_sequence(result))
    print(f"[final] n={len(result)}, internal={best_overall_sc:.6f}, evaluator={final_sc:.6f}")

    return result
