import numpy as np
import time
import cvxpy as cp

def full_conv(a):
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def project_simplex(v):
    n = v.size
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def normed_score(a):
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))

def solve_lp(a, M):
    """Solve max sum(g) s.t. (a*g)_k <= M for all k, g >= 0."""
    n = a.size
    g = cp.Variable(n, non_negative=True)
    
    # Compute convolution matrix: C[k, i] = a_{k-i} (0 if out of range)
    # (a * g)_k = sum_i C[k, i] * g_i
    # Build C as a (2n-1) x n matrix
    size = 2 * n - 1
    C = np.zeros((size, n))
    for k in range(size):
        for i in range(n):
            j = k - i
            if 0 <= j < n:
                C[k, i] = a[j]
    
    objective = cp.Maximize(cp.sum(g))
    constraints = [C @ g <= M]
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, verbose=False)
    if g.value is None:
        return a.copy()
    g_val = np.maximum(g.value, 0)
    s = g_val.sum()
    if s < 1e-10:
        return a.copy()
    return g_val / s

def lp_iteration(a, t0):
    """One LP-based iteration. Returns updated a."""
    n = a.size
    a = a / a.sum()
    b = full_conv(a)
    M = float(np.max(b))
    g = solve_lp(a, M)
    a_new = (1 - 0.1) * a + 0.1 * g
    a_new = np.maximum(a_new, 0)
    a_new = a_new / a_new.sum()
    return a_new

"""
LP-based iterative minimization of 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming to find optimal descent directions,
combined with subgradient polish and multi-length search.
"""

import numpy as np
import time
import cvxpy as cp


def full_conv(a):
    """Full linear convolution a*a via FFT."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def project_simplex(v):
    """Project v onto {x >= 0, sum(x) = 1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tmp = u * idx - (cssv - 1.0)
    mask = tmp > 0
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normed_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def build_conv_indices(n):
    """Precompute indices for building convolution matrix."""
    size = 2 * n - 1
    k = np.arange(size)[:, None]
    i = np.arange(n)[None, :]
    j = k - i
    mask = (j >= 0) & (j < n)
    return j, mask


def make_conv_matrix(a, j, mask):
    """Build C where C[k,i] = a[k-i] if valid, else 0."""
    n = a.size
    C = np.zeros((2 * n - 1, n))
    C[mask] = a[j[mask]]
    return C


def solve_lp(a, M, j, mask):
    """Solve max sum(g) s.t. C@ g <= M, g >= 0. Returns normalized g."""
    n = a.size
    C = make_conv_matrix(a, j, mask)
    
    g = cp.Variable(n, non_negative=True)
    objective = cp.Maximize(cp.sum(g))
    constraints = [C @ g <= M]
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=5000, eps=1e-6)
    except Exception:
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except Exception:
            return a.copy()
    
    if g.value is None:
        return a.copy()
    g_val = np.clip(g.value, 0, None)
    s = g_val.sum()
    if s < 1e-10:
        return a.copy()
    return g_val / s


def lp_optimize(a_init, time_budget, verbose=False):
    """LP-based iterative optimization."""
    n = a_init.size
    a = project_simplex(a_init.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    j, mask = build_conv_indices(n)
    
    t0 = time.monotonic()
    iters = 0
    last_improve = 0
    t_step = 0.1
    
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        M = float(np.max(b))
        
        g = solve_lp(a, M, j, mask)
        cur_sc = normed_score(a)
        
        # Line search over step sizes
        best_t = t_step
        best_new_sc = cur_sc
        best_a_new = a.copy()
        
        for t in [t_step, 0.5 * t_step, 0.2 * t_step, 0.05 * t_step]:
            a_cand = (1 - t) * a + t * g
            a_cand = np.maximum(a_cand, 0)
            s = a_cand.sum()
            if s < 1e-10:
                continue
            a_cand = a_cand / s
            sc = normed_score(a_cand)
            if sc < best_new_sc:
                best_new_sc = sc
                best_t = t
                best_a_new = a_cand
        
        a = best_a_new
        iters += 1
        
        if best_new_sc < best_sc - 1e-10:
            best_sc = best_new_sc
            best_a = a.copy()
            last_improve = iters
            if verbose and iters % 50 == 0:
                print(f"    iter {iters}: score={best_sc:.8f}")
        
        # Adaptive
        if iters - last_improve > 100:
            t_step = max(0.01, t_step * 0.8)
            # Restart with perturbation
            if iters - last_improve > 300:
                a = project_simplex(best_a + 0.01 * np.random.default_rng(iters).standard_normal(n))
                t_step = 0.05
                last_improve = iters
    
    return best_a, best_sc


def init_u_shape(n):
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_harmonic(n):
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i + 1, n - i)
    a = 1.0 / m
    return a / a.sum()


def init_cosine(n):
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 + np.cos(2 * np.pi * i / (n - 1)))
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def init_two_bump(n, rng):
    i = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.1, 0.3)
    c2 = n * rng.uniform(0.7, 0.9)
    sig = n * rng.uniform(0.05, 0.15)
    a = np.exp(-((i - c1) / sig) ** 2) + np.exp(-((i - c2) / sig) ** 2)
    a += 1e-6
    return a / a.sum()


def subgrad_polish(a, time_budget, rng):
    """Polish with smoothed subgradient descent."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    eta = 0.05
    tau_frac = 0.02
    beta = 0.5
    vel = np.zeros(n)
    last_improve = 0
    iters = 0
    
    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        cur_sc = 2.0 * n * float(np.max(b))
        tau = tau_frac * float(np.max(b))
        if tau < 1e-18:
            tau = 1e-18
        
        # Smoothed subgradient
        shift = b / tau
        shift -= shift.max()
        w = np.exp(shift)
        s = w.sum()
        if s < 1e-300:
            w = np.zeros(b.size)
            w[int(np.argmax(b))] = 1.0
        else:
            w /= s
        g = 2.0 * np.correlate(w, a, mode='valid')
        
        vel = beta * vel + (1 - beta) * g
        a_new = project_simplex(a - eta * vel)
        new_sc = normed_score(a_new)
        
        if new_sc <= cur_sc or rng.random() < 0.03:
            a = a_new
        else:
            a = project_simplex(a - 0.3 * eta * vel)
        
        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = iters
        
        iters += 1
        if iters - last_improve > 2000:
            a = project_simplex(best_a + 0.02 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.03 + 0.05 * rng.random()
            last_improve = iters
    
    return best_a, best_sc

"""
LP-based iterative minimization of 2*n*max(conv(a,a))/sum(a)^2.
Uses linear programming (HiGHS via scipy) to find optimal descent directions
at each step, combined with multi-length search, diverse initializations,
and subgradient polish. The LP finds the "best response" direction that
maximizes total mass while keeping the cross-convolution bounded.
"""

import numpy as np
import time
from scipy.optimize import linprog


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    tmp = u * idx - (cssv - 1.0)
    mask = tmp > 0
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def normed_score(a):
    """Compute 2*n*max(conv(a,a)) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def build_conv_structure(n):
    """Precompute indices for building the convolution matrix C[k,i]=a[k-i]."""
    size = 2 * n - 1
    k = np.arange(size)[:, None]
    i = np.arange(n)[None, :]
    j = k - i
    mask = (j >= 0) & (j < n)
    rows, cols = np.where(mask)
    a_indices = j[rows, cols]
    return rows, cols, a_indices, size


def solve_lp_dir(a, M, rows, cols, a_indices, size):
    """Solve max sum(g) s.t. C@ g <= M, g >= 0. Returns normalized g."""
    n = a.size
    C = np.zeros((size, n))
    C[rows, cols] = a[a_indices]

    c = -np.ones(n)
    b_ub = np.full(size, M)
    bounds = [(0.0, None)] * n

    try:
        res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'presolve': True})
    except Exception:
        return a.copy()

    if not res.success or res.x is None:
        return a.copy()

    g_val = np.clip(res.x, 0.0, None)
    s = g_val.sum()
    if s < 1e-12:
        return a.copy()
    return g_val / s


def lp_optimize(a_init, time_budget, verbose=False):
    """LP-based iterative optimization to minimize the score."""
    n = a_init.size
    a = project_simplex(a_init.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    rows, cols, a_indices, size = build_conv_structure(n)

    t0 = time.monotonic()
    iters = 0
    last_improve = 0
    t_step = 0.15

    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        M = float(np.max(b))

        g = solve_lp_dir(a, M, rows, cols, a_indices, size)
        cur_sc = normed_score(a)

        # Line search: try several step sizes
        best_t = t_step
        best_new_sc = cur_sc
        best_a_new = a.copy()

        for t in [t_step, 0.5 * t_step, 0.2 * t_step, 0.05 * t_step]:
            a_cand = (1.0 - t) * a + t * g
            a_cand = np.maximum(a_cand, 0.0)
            s = a_cand.sum()
            if s < 1e-12:
                continue
            a_cand = a_cand / s
            sc = normed_score(a_cand)
            if sc < best_new_sc:
                best_new_sc = sc
                best_t = t
                best_a_new = a_cand.copy()

        a = best_a_new
        iters += 1

        if best_new_sc < best_sc - 1e-12:
            best_sc = best_new_sc
            best_a = a.copy()
            last_improve = iters
            if verbose and iters % 100 == 0:
                print(f"      iter {iters}: score={best_sc:.10f}")

        # Adaptive step and restarts
        if iters - last_improve > 200:
            t_step = max(0.03, t_step * 0.8)

        if iters - last_improve > 500:
            rng_local = np.random.default_rng(iters * 7 + 3)
            a = project_simplex(best_a + 0.02 * rng_local.standard_normal(n))
            t_step = 0.1
            last_improve = iters

    return best_a, best_sc


def subgrad_polish(a, time_budget, rng):
    """Polish with smoothed subgradient descent."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)

    eta = 0.03
    tau_frac = 0.015
    beta = 0.4
    vel = np.zeros(n)
    last_improve = 0
    iters = 0

    t0 = time.monotonic()
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        max_b = float(np.max(b))
        cur_sc = 2.0 * n * max_b
        tau = tau_frac * max_b
        if tau < 1e-18:
            tau = 1e-18

        shift = b / tau
        shift -= shift.max()
        w = np.exp(shift)
        s = w.sum()
        if s < 1e-300:
            w = np.zeros(b.size)
            w[int(np.argmax(b))] = 1.0
        else:
            w /= s

        g = 2.0 * np.correlate(w, a, mode='valid')
        vel = beta * vel + (1.0 - beta) * g
        a_new = project_simplex(a - eta * vel)
        new_sc = normed_score(a_new)

        if new_sc <= cur_sc or rng.random() < 0.02:
            a = a_new
        else:
            a = project_simplex(a - 0.4 * eta * vel)

        if new_sc < best_sc:
            best_sc = new_sc
            best_a = a.copy()
            last_improve = iters

        iters += 1
        if iters - last_improve > 3000:
            a = project_simplex(best_a + 0.01 * rng.standard_normal(n))
            vel = np.zeros(n)
            eta = 0.02 + 0.03 * rng.random()
            last_improve = iters

    return best_a, best_sc


def init_u_shape(n):
    """U-shaped: high at edges, low in middle."""
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i, n - 1 - i) + 1.0
    a = 1.0 / np.sqrt(m)
    return a / a.sum()


def init_harmonic(n):
    """Harmonic: even more edge-concentrated."""
    i = np.arange(n, dtype=np.float64)
    m = np.minimum(i + 1.0, n - i).astype(np.float64)
    a = 1.0 / m
    return a / a.sum()


def init_cosine(n):
    """Cosine taper."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(2.0 * np.pi * i / max(n - 1, 1)))
    a = np.maximum(a, 1e-6)
    return a / a.sum()


def init_random(n, rng):
    """Random positive initialization."""
    a = np.abs(rng.standard_normal(n)) + 0.1
    return a / a.sum()


def init_two_bump(n, rng):
    """Two Gaussian bumps at random positions."""
    i = np.arange(n, dtype=np.float64)
    c1 = n * rng.uniform(0.1, 0.35)
    c2 = n * rng.uniform(0.65, 0.9)
    sig = n * rng.uniform(0.05, 0.12)
    a = np.exp(-((i - c1) / sig) ** 2) + np.exp(-((i - c2) / sig) ** 2)
    a += 1e-6
    return a / a.sum()


def init_edge_spike(n, rng):
    """Sharp edges with decay."""
    i = np.arange(n, dtype=np.float64)
    decay = 1.0 + rng.uniform(0.3, 1.0)
    left = (1.0 / (1.0 + decay * i))
    right = (1.0 / (1.0 + decay * (n - 1 - i)))
    a = left + right
    a += 1e-4
    return a / a.sum()


def get_inits(n, rng):
    """Return list of diverse initializations."""
    inits = [
        init_u_shape(n),
        init_harmonic(n),
        init_cosine(n),
        init_random(n, rng),
        init_two_bump(n, rng),
        init_edge_spike(n, rng),
    ]
    return inits
