```python
def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def objective(a):
    n = len(a)
    c = fast_self_conv(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    return 2.0 * n * np.max(c) / (S * S)


def grad_max_conv(a, c, topk=1):
    """Gradient of max(conv(a,a)) using top-k peaks with uniform weights."""
    n = len(a)
    k_act = min(topk, len(c))
    # Get top-k indices
    if k_act == 1:
        idx = np.array([int(np.argmax(c))])
    else:
        idx = np.argpartition(c, -k_act)[-k_act:]
    
    grad = np.zeros(n)
    for ki in idx:
        ki = int(ki)
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start > j_end:
            continue
        grad[j_start:j_end+1] += a[ki - j_end : ki - j_start - 1 : -1]
    
    grad *= 2.0 / k_act
    return grad

    """Gradient of var(conv(a,a)), computed via FFT for efficiency."""
    n = len(a)
    M = 2*n - 1
```

```python
def grad_topk_fft(a, c, topk=1):
    """Efficient gradient of (1/topk) * sum of top-k peaks of conv(a,a)."""
    n = len(a)
    M = 2*n - 1
    k_act = min(topk, M)
    
    # Get top-k indices
    if k_act == 1:
        idx = np.array([int(np.argmax(c))])
    else:
        idx = np.argpartition(c, -k_act)[-k_act:]
    
    # Build impulse signal d
    d = np.zeros(M)
    d[idx] = 1.0
    
    # Compute conv(d, a_reversed) via FFT
    M2 = M + n - 1  # length of linear convolution
    M_fft = 1 << (M2 - 1).bit_length()
    D = np.fft.rfft(d, M_fft)
    A_rev = np.fft.rfft(a[::-1], M_fft)
    conv_result = np.fft.irfft(D * A_rev, M_fft)[:M2]
    
    # grad[m] = (2/k_act) * conv_result[m + n - 1] for m = 0,...,n-1
    grad = (2.0 / k_act) * conv_result[n-1 : 2*n-1]
    return grad
```

```python
- LP-based balancing iterations for strong initial points
- Mirror descent (multiplicative) updates for natural non-negativity
- Adaptive top-k schedule that widens over time for flatter profiles
- Periodic LP restarts to escape local minima
"""

import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def obj_value(a):
    """Compute f(a) = 2n*max(conv(a,a))/(sum a)^2."""
    n = len(a)
    c = fast_self_conv(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    return float(2.0 * n * np.max(c) / (S * S))


def grad_topk_fft(a, c, topk=1):
    """
    Efficient gradient of (1/topk)*sum of top-k peaks of conv(a,a).
    Uses FFT to compute in O(n log n) regardless of topk.
    """
    n = len(a)
    M = 2 * n - 1
    k_act = min(topk, M)
    if k_act <= 0:
        return np.zeros(n)

    if k_act == 1:
        idx = np.array([int(np.argmax(c))])
    else:
        idx = np.argpartition(c, -k_act)[-k_act:]

    d = np.zeros(M)
    d[idx] = 1.0

    M2 = M + n - 1
    M_fft = 1 << (M2 - 1).bit_length()
    D = np.fft.rfft(d, M_fft)
    A_rev = np.fft.rfft(a[::-1], M_fft)
    conv_res = np.fft.irfft(D * A_rev, M_fft)[:M2]

    grad = (2.0 / k_act) * conv_res[n - 1:2 * n - 1]
    return grad


def lp_balance(a, M, max_iter=20):
    """
    LP-based balancing: find b maximizing sum(b) s.t. conv(a,b) <= M, b >= 0.
    Returns the balanced sequence.
    """
    from scipy.optimize import linprog
    from scipy import sparse

    n = len(a)
    M_len = 2 * n - 1

    # Build constraint matrix A_ub where A_ub[k,j] = a[k-j] if 0<=k-j<n else 0
    rows = []
    cols = []
    vals = []
    for k in range(M_len):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            rows.append(k)
            cols.append(j)
            vals.append(a[k - j])

    A_ub = sparse.csr_matrix((vals, (rows, cols)), shape=(M_len, n))
    b_ub = np.full(M_len, M)
    c_obj = -np.ones(n)

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True, 'time_limit': 120})

    if res.status == 0 and res.x is not None:
        b = np.clip(res.x, 0, 1000)
        if b.sum() > 1e-10:
            return b
    return a.copy()


def adam_step(a, grad, m, v, t, lr):
    """Standard Adam update in place."""
    b1, b2, eps = 0.9, 0.999, 1e-8
    m[:] = b1 * m + (1.0 - b1) * grad
    v[:] = b2 * v + (1.0 - b2) * grad * grad
    mh = m / (1.0 - b1 ** t)
    vh = v / (1.0 - b2 ** t)
    a -= lr * mh / (np.sqrt(vh) + eps)


def mirror_step(a, grad, lr):
    """Multiplicative (mirror) update: a *= exp(-lr * grad / scale)."""
    scale = np.max(np.abs(grad))
    if scale < 1e-15:
        return
    a *= np.exp(-lr * grad / scale)
    a = np.clip(a, 0.0, 1000.0)


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    t_start = time.monotonic()
    total = max(1.0, float(budget_s))
    deadline = t_start + total - 15.0

    # Get parent sequence
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if p.size > 0:
                parent = np.clip(p, 0, 1000)
    except Exception:
        pass

    # Choose n
    n = 2048
    if parent is not None and 512 <= parent.size <= 8192:
        n = parent.size
    n = min(max(1024, n), 8192)

    # Initial sequence
    if parent is not None and parent.size == n:
        a = parent.copy()
    else:
        a = np.ones(n)
    a = np.clip(a, 0, 1000)
    if a.sum() < 1e-6:
        a = np.ones(n)

    best_a = a.copy()
    best_score = obj_value(a)
    print(f"Initial: n={n}, score={best_score:.6f}, time=0s")

    # ===================== PHASE 1: LP Balancing =====================
    phase1_end = t_start + 0.20 * total
    print(f"Phase 1 (LP): {phase1_end - t_start:.0f}s budget")

    c = fast_self_conv(a)
    S = a.sum()
    M = float(np.max(c))
    # Work with normalized version for LP
    a_norm = a / S

    lp_count = 0
    while time.monotonic() < phase1_end:
        c_n = fast_self_conv(a_norm)
        M_n = float(np.max(c_n))
        if M_n < 1e-12:
            break
        try:
            b = lp_balance(a_norm, M_n, max_iter=1)
        except Exception:
            break
        Sb = b.sum()
        if Sb < 1e-10:
            break
        b_norm = b / Sb
        t_blend = 0.15
        a_norm = (1.0 - t_blend) * a_norm + t_blend * b_norm
        a_norm = np.clip(a_norm, 0, 1000)
        lp_count += 1

        sc = obj_value(a_norm * S)
        if sc < best_score:
            best_score = sc
            best_a = (a_norm * S).copy()
            print(f"  LP iter {lp_count}: score={sc:.6f}")

    a = a_norm * S
    print(f"Phase 1 done: {lp_count} LP iters, best={best_score:.6f}")

    # ===================== PHASE 2: Gradient Optimization =====================
    phase2_end = t_start + 0.82 * total
    print(f"Phase 2 (Gradient): {phase2_end - time.monotonic():.0f}s budget")

    # Use normalized a for optimization (scale invariant)
    a_opt = a / a.sum()
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    lr = 0.5
    no_improve = 0
    patience = 3000
    topk_current = 3
    topk_target = 80
    steps_done = 0

    while time.monotonic() < phase2_end:
        c = fast_self_conv(a_opt)
        S_opt = a_opt.sum()
        if S_opt < 1e-10:
            a_opt = np.ones(n) / n
            continue
        score = 2.0 * n * np.max(c) / (S_opt * S_opt)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = (a_opt * S).copy()
            no_improve = 0
            # Occasionally print
            if steps_done % 50000 == 0:
                print(f"  Step {steps_done}: score={score:.6f}, topk={topk_current}, lr={lr:.4f}")
        else:
            no_improve += 1

        # Adaptive topk: increase over time
        frac = min(1.0, steps_done / 200000.0)
        topk_current = int(3 + (topk_target - 3) * frac)

        # Compute gradient
        g = grad_topk_fft(a_opt, c, topk_current)

        # Normalize gradient for stability
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm * np.max(np.abs(g))

        # Adam update
        t_adam += 1
        adam_step(a_opt, g, m_state, v_state, t_adam, lr)
        a_opt = np.clip(a_opt, 0.0, 1000.0)
        steps_done += 1

        # Periodic restart
        if no_improve >= patience:
            print(f"  Restart at step {steps_done}, best={best_score:.6f}")
            # Perturb from best
            scale = 0.03 + 0.07 * rng.random()
            a_opt = best_a / S
            a_opt = a_opt * (1.0 + scale * rng.standard_normal(n))
            a_opt = np.clip(a_opt, 0.0, 1000.0)
            m_state[:] = 0.0
            v_state[:] = 0.0
            t_adam = 0
            lr = max(0.05, lr * 0.75)
            no_improve = 0
            patience = max(1000, patience // 2)

        # Learning rate schedule: decay slowly
        if steps_done % 100000 == 0 and steps_done > 100000:
            lr *= 0.85

    print(f"Phase 2 done: {steps_done} steps, best={best_score:.6f}")

    # ===================== PHASE 3: Polish =====================
    print(f"Phase 3 (Polish): {deadline - time.monotonic():.0f}s budget")

    a_opt = best_a / S
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    lr = 0.05
    polish_steps = 0

    while time.monotonic() < deadline:
        c = fast_self_conv(a_opt)
        S_opt = a_opt.sum()
        if S_opt < 1e-10:
            break
        score = 2.0 * n * np.max(c) / (S_opt * S_opt)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = (a_opt * S).copy()

        # Use large topk for polishing (flatten all peaks)
        topk_polish = min(200, 2 * n - 1)
        g = grad_topk_fft(a_opt, c, topk_polish)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            g = g / g_norm * np.max(np.abs(g))

        t_adam += 1
        adam_step(a_opt, g, m_state, v_state, t_adam, lr)
        a_opt = np.clip(a_opt, 0.0, 1000.0)
        polish_steps += 1

        if polish_steps % 100000 == 0:
            print(f"  Polish step {polish_steps}: score={score:.6f}")

    print(f"Phase 3 done: {polish_steps} steps, best={best_score:.6f}")

    # Final verification
    out = [float(x) for x in best_a]
    verified = float(evaluate_sequence(out))
    print(f"FINAL: internal={best_score:.6f}, verified={verified:.6f}, len={len(out)}, "
          f"time={time.monotonic()-t_start:.1f}s")

    if not np.isfinite(verified) or len(out) == 0:
        return [1.0] * n

    return out
```
