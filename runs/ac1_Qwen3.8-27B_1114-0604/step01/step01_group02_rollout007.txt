```python
def fp_based_optimize(f0, n_iter, t):
    """LP-based flat-peak optimization."""
    n = len(f0)
    f = f0.copy()
    M = np.max(np.convolve(f, f))

        # Build constraint matrix A such that A @ b <= M * ones
        # A[k, j] = f[k-j] if 0 <= k-j < n, else 0

        # Actually this is a Toeplitz-like matrix

            f = (1-t)*f + t*g
        M = np.max(np.convolve(f, f))
    
    return f
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def score(a):
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fast_self_conv(a)
    return 2.0 * n * np.max(c) / (S * S)

def lp_direction(f, M):
    """Find the LP direction: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    # Build the convolution matrix F: F[k,j] = f[k-j] if valid
    # F is (2n-1) x n
    # For efficiency, build it using the Toeplitz structure
    # F[k,j] = f[k-j] for k-j in [0, n-1]
    # Column j is f shifted down by j positions
    
    # Actually, let's build it column by column
    # Column j: F[k,j] = f[k-j] for k in [j, j+n-1], else 0
    # So column j is [0]*j + f + [0]*(2n-2-j)
    
    # For n up to ~512, we can build the full matrix
    F = np.zeros((2*n-1, n))
    for j in range(n):
        F[j:j+n, j] = f
    
    # LP: max 1^T b s.t. F b <= M*1, b >= 0
    c_obj = -np.ones(n)
    res = linprog(c_obj, A_ub=F, b_ub=np.full(2*n-1, M),
                  bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None
```

```python
def lp_iterate(f, n_iter, max_t=0.5):
    """LP-based iteration to reduce the peak of conv(f,f)."""
    n = len(f)
    f = f / f.sum()  # normalize to sum 1
    best_f = f.copy()
    best_peak = np.max(fast_self_conv(f))
    
    for it in range(n_iter):
        M = np.max(fast_self_conv(f))
        
        # Solve LP
        b_star = lp_direction(f, M)
        if b_star is None or b_star.sum() < 1e-10:
            break
        
        g = b_star / b_star.sum()
        
        # Compute conv(f,g) and conv(g,g)
        cg = fast_self_conv(f)  # conv(f,f)
        # Need conv(f, g) - this is not the same as fast_self_conv
        # conv(f, g) = FFT(f) * FFT(g) (linear convolution)
        M_fft = 1 << (2*n-2).bit_length()
        Ff = np.fft.rfft(f, M_fft)
        Fg = np.fft.rfft(g, M_fft)
        cg_cross = np.fft.irfft(Ff * Fg, M_fft)[:2*n-1]
        cg_g = np.fft.irfft(Fg * Fg, M_fft)[:2*n-1]
        
        # Minimize peak of (1-t)^2*cg + 2t(1-t)*cg_cross + t^2*cg_g over t in [0, max_t]
        best_t = 0
        best_peak_new = np.max(cg)
        for t_candidate in np.linspace(0, max_t, 50):
            c_new = (1-t_candidate)**2 * cg + 2*t_candidate*(1-t_candidate)*cg_cross + t_candidate**2*cg_g
            peak_new = np.max(c_new)
            if peak_new < best_peak_new:
                best_peak_new = peak_new
                best_t = t_candidate
        
        f = (1-best_t)*f + best_t*g
        f = np.clip(f, 0, None)
        f = f / f.sum()
        
        if np.max(fast_self_conv(f)) < best_peak:
            best_peak = np.max(fast_self_conv(f))
            best_f = f.copy()
    
    return best_f
```

```python
import numpy as np
from scipy.optimize import linprog
import time

def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]

def fast_cross_conv(a, b):
    n1, n2 = len(a), len(b)
    M = n1 + n2 - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]

def score_from_conv(n, c, S):
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)

def conv_matrix(f):
    """Build the convolution matrix F such that (F @ b)[k] = conv(f, b)[k]."""
    n = len(f)
    F = np.zeros((2*n - 1, n))
    for j in range(n):
        F[j:j+n, j] = f
    return F

def lp_solve_direction(f, M):
    """Solve the LP: max sum(b) s.t. conv(f,b) <= M, b >= 0."""
    n = len(f)
    F = conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2*n - 1, M)
    res = linprog(c_obj, A_ub=F, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if res.success:
        return res.x
    return None

def lp_phase(f0, time_limit, verbose=True):
    """Run LP-based iterations until time_limit."""
    f = f0.astype(np.float64).copy()
    if f.sum() < 1e-10:
        f = np.ones(len(f))
    f = f / f.sum()
    
    c = fast_self_conv(f)
    best_peak = float(np.max(c))
    best_f = f.copy()
    start = time.monotonic()
    
    it = 0
    while time.monotonic() - start < time_limit:
        it += 1
        M = float(np.max(c))
        
        b_star = lp_solve_direction(f, M)
        if b_star is None or b_star.sum() < 1e-12:
            break
        
        g = b_star / b_star.sum()
        
        # Compute the cross-convolutions
        cg = c  # conv(f, f)
        cg_cross = fast_cross_conv(f, g)
        cg_g = fast_self_conv(g)
        
        # Line search: minimize peak of quadratic in t
        best_t = 0.0
        best_peak_t = M
        for t in np.linspace(0.01, 0.5, 20):
            c_t = (1-t)**2 * cg + 2*t*(1-t)*cg_cross + t**2 * cg_g
            peak_t = float(np.max(c_t))
            if peak_t < best_peak_t:
                best_peak_t = peak_t
                best_t = t
        
        if best_t < 0.005:
            # Try smaller steps
            for t in np.linspace(0.001, 0.01, 10):
                c_t = (1-t)**2 * cg + 2*t*(1-t)*cg_cross + t**2 * cg_g
                peak_t = float(np.max(c_t))
                if peak_t < best_peak_t:
                    best_peak_t = peak_t
                    best_t = t
        
        f = (1 - best_t) * f + best_t * g
        f = np.clip(f, 0, None)
        f_sum = f.sum()
        if f_sum < 1e-10:
            break
        f = f / f_sum
        
        c = fast_self_conv(f)
        current_peak = float(np.max(c))
        
        if current_peak < best_peak:
            best_peak = current_peak
            best_f = f.copy()
        
        if verbose and it % 100 == 0:
            elapsed = time.monotonic() - start
            n = len(f)
            obj = score_from_conv(n, c, 1.0)
            print(f"  LP iter {it}, peak={best_peak:.8f}, obj={obj:.6f}, t={best_t:.4f}, time={elapsed:.1f}s")
    
    return best_f, best_peak

# Adam-based refinement (similar to previous code but improved)
def topk_indices(c, k):
    k_act = min(k, len(c))
    idx = np.argpartition(c, -k_act)[-k_act:]
    vals = c[idx]
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]
    top = float(vals[0])
    if top > 1e-12:
        temp = 0.05 * top  # sharper weighting than before
        wts = np.exp((vals - top) / temp)
        wts = wts / wts.sum()
    else:
        wts = np.ones(k_act) / k_act
    return idx, vals, wts

def grad_topk(a, c, k):
    n = len(a)
    S = float(a.sum())
    idx, vals, wts = topk_indices(c, k)
    grad_c = np.zeros(n)
    for i in range(idx.size):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi <= 1e-15:
            continue
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start > j_end:
            continue
        grad_c[j_start:j_end + 1] += wi * 2.0 * a[j_start:j_end + 1][::-1]
    M = float(np.dot(vals, wts))
    S2 = S * S
    S3 = S2 * S
    grad = (2.0 * n) * (grad_c / S2 - 2.0 * M / S3)
    return grad

def adam_phase(f0, time_limit, init_lr=0.05, topk=10, patience=300):
    f = f0.astype(np.float64).copy()
    n = len(f)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    lr = init_lr
    best_score = score_from_conv(n, fast_self_conv(f), f.sum())
    best_f = f.copy()
    no_improve = 0
    start = time.monotonic()
    rng = np.random.default_rng(12345)
    
    while time.monotonic() - start < time_limit:
        c = fast_self_conv(f)
        S = f.sum()
        sc = score_from_conv(n, c, S)
        if np.isfinite(sc) and sc < best_score:
            best_score = sc
            best_f = f.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve >= patience:
            scale = 0.03 + 0.05 * rng.random()
            f = best_f * (1.0 + scale * rng.standard_normal(n))
            f = np.clip(f, 0, 1000)
            if f.sum() < 1e-3:
                f = best_f.copy()
            m[:] = 0
            v[:] = 0
            t = 0
            lr = max(0.005, lr * 0.7)
            no_improve = 0
            patience = max(100, patience // 2)
            continue
        
        g = grad_topk(f, c, topk)
        t += 1
        b1, b2, eps = 0.9, 0.999, 1e-8
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        f = f - lr * mh / (np.sqrt(vh) + eps)
        f = np.clip(f, 0, 1000)
    
    return best_f, best_score
```

LP-based + Adam optimizer to minimize 2n*max(conv(a,a))/(sum(a))^2.

Primary method: Linear programming to find optimal update direction.
Finds b >= 0 maximizing sum(b) subject to conv(f,b) <= M (where M is current peak).
Uses line search for step size. Combined with Adam polishing phase.

import numpy as np
from scipy.optimize import linprog
import time


def fast_self_conv(a):
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def fast_cross_conv(a, b):
    n1, n2 = len(a), len(b)
    M = n1 + n2 - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    B = np.fft.rfft(b, M_fft)
    return np.fft.irfft(A * B, M_fft)[:M]


def score_fn(a):
    n = len(a)
    S = float(a.sum())
    if S < 1e-10:
        return np.inf
    c = fast_self_conv(a)
    return 2.0 * n * float(np.max(c)) / (S * S)


def build_conv_matrix(f):
    n = len(f)
    F = np.zeros((2 * n - 1, n))
    for j in range(n):
        F[j:j + n, j] = f
    return F


def solve_lp_direction(f, M):
    n = len(f)
    F = build_conv_matrix(f)
    c_obj = -np.ones(n)
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n
    try:
        res = linprog(c_obj, A_ub=F, b_ub=b_ub, bounds=bounds, method='highs',
                      options={'time_limit': 10})
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def topk_peak_weights(c, k):
    k = min(k, len(c))
    idx = np.argpartition(c, -k)[-k:]
    vals = c[idx]
    order = np.argsort(vals)[::-1]
    idx = idx[order]
    vals = vals[order]
    top = float(vals[0])
    if top > 1e-12:
        temp = 0.08 * top
        wts = np.exp((vals - top) / temp)
        wts /= wts.sum()
    else:
        wts = np.ones(k) / k
    return idx, vals, wts


def compute_gradient(a, c, topk):
    n = len(a)
    S = float(a.sum())
    idx, vals, wts = topk_peak_weights(c, topk)
    grad_c = np.zeros(n)
    for i in range(idx.size):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-15:
            continue
        js = max(0, ki - n + 1)
        je = min(ki, n - 1)
        if js > je:
            continue
        grad_c[js:je + 1] += wi * 2.0 * a[js:je + 1][::-1]
    M = float(np.dot(vals, wts))
    S2 = S * S
    S3 = S2 * S
    return (2.0 * n) * (grad_c / S2 - 2.0 * M / S3)
