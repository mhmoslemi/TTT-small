def _fft_convolve(a, b):
    """Full convolution using FFT."""
    n = len(a) + len(b) - 1

        L *= 2
    fa = np.fft.rfft(a, L)

import numpy as np
import time

# Constants
N_MAIN = 3072

def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _convolve_full(a, b):
    n = len(a) + len(b) - 1
    L = _next_pow2(n)
    return np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)[:n]

def _score(x):
    """2n * max(conv(x,x)) for normalized x (sum=1)."""
    n = len(x)
    c = _convolve_full(x, x)
    return 2.0 * n * float(np.max(c))

def _gradient(x, w):
    """Gradient of sum_m w[m]*(x*x)[m] w.r.t. x.
    w has length 2n-1."""
    n = len(x)
    L = _next_pow2(max(len(w), 2*n))
    W = np.fft.rfft(w, L)
    X = np.fft.rfft(x, L)
    grad = 2.0 * np.fft.irfft(W * np.conjugate(X), L)[:n]
    return grad

def _normalize(x):
    x = np.maximum(x, 0.0)
    x = np.where(np.isfinite(x), x, 0.0)
    s = x.sum()
    if s <= 0 or not np.isfinite(s):
        return None
    return x / s

def _run_descent(x0, rng, time_limit, mode='phase1'):
    """Multiplicative descent with adaptive smoothing."""
    x = _normalize(x0)
    if x is None:
        return None, float('inf')
    n = len(x)
    
    best = x.copy()
    best_score = _score(x)
    cur_score = best_score
    
    # Adaptive parameters
    if mode == 'phase1':
        beta = 20.0
        beta_max = 200.0
        topk = n // 4
    elif mode == 'phase2':
        beta = 100.0
        beta_max = 2000.0
        topk = n // 8
    else:  # phase3 - focused
        beta = 500.0
        beta_max = 5000.0
        topk = 5
    
    alpha = 0.3
    deadline = time.monotonic() + time_limit
    iter_count = 0
    stall = 0
    
    while time.monotonic() < deadline:
        # Compute convolution
        c = _convolve_full(x, x)
        c = np.where(np.isfinite(c), c, 0.0)
        c = np.maximum(c, 0.0)
        
        cur_score = 2.0 * n * float(np.max(c))
        
        if cur_score < best_score:
            best = x.copy()
            best_score = cur_score
            stall = 0
        else:
            stall += 1
            if stall > 100:
                # Perturb and restart
                x = _normalize(x * np.exp(rng.normal(0, 0.1, n)))
                alpha = 0.3
                stall = 0
                continue
        
        # Compute weights (either softmax or top-k)
        if topk < len(c) and topk > 0:
            # Top-k with softmax blending
            sorted_idx = np.argsort(c)[::-1]
            if topk < len(c):
                threshold = c[sorted_idx[topk-1]]
                # Soft top-k: weights proportional to max(0, c - threshold + eps)
                w = np.maximum(c - threshold + 1e-10, 0.0)
                w_sum = w.sum()
                if w_sum > 0:
                    w = w / w_sum
                else:
                    w = np.full(len(c), 1.0/len(c))
            else:
                w = np.full(len(c), 1.0/len(c))
        else:
            # Pure softmax
            beta_eff = min(beta, beta_max)
            c_shift = c - np.max(c)
            w = np.exp(beta_eff * c_shift)
            w_sum = w.sum()
            if w_sum > 0 and np.isfinite(w_sum):
                w = w / w_sum
            else:
                w = np.full(len(c), 1.0/len(c))
        
        # Compute gradient
        grad = _gradient(x, w)
        grad = np.maximum(grad, 0.0)
        grad_max = np.max(grad)
        if grad_max <= 1e-15:
            break
        
        grad_norm = grad / grad_max
        
        # Line search: try several step sizes
        best_alpha = alpha
        best_x = None
        best_s = cur_score

"""
Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2
using adaptive top-k multiplicative descent with multi-phase strategy:
broad softmax exploration, focused top-k descent, and coordinate polishing.
"""

import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _convolve_full(a, b):
    n = len(a) + len(b) - 1
    L = _next_pow2(n)
    return np.fft.irfft(np.fft.rfft(a, L) * np.fft.rfft(b, L), L)[:n]


def _normalize(x):
    x = np.asarray(x, dtype=float)
    x = np.where(np.isfinite(x), x, 0.0)
    x = np.maximum(x, 0.0)
    s = float(x.sum())
    if s <= 1e-30 or not np.isfinite(s):
        return None
    return x / s


def _score(x):
    n = len(x)
    c = _convolve_full(x, x)
    c = np.where(np.isfinite(c), c, 0.0)
    return float(2.0 * n * np.max(c))


def _compute_gradient(x, w):
    """Gradient of sum_m w[m]*(x*x)[m] w.r.t. x. w has length 2n-1."""
    n = len(x)
    L = _next_pow2(max(len(w), 2 * n))
    W = np.fft.rfft(w, L)
    X = np.fft.rfft(x, L)
    grad = 2.0 * np.fft.irfft(W * np.conjugate(X), L)[:n]
    return grad


def _softmax_weights(c, beta):
    c = np.where(np.isfinite(c), c, 0.0)
    c_shift = c - np.max(c)
    w = np.exp(beta * c_shift)
    s = float(w.sum())
    if s <= 0 or not np.isfinite(s):
        return np.full(len(c), 1.0 / len(c))
    return w / s


def _topk_weights(c, k, temperature=1.0):
    c = np.where(np.isfinite(c), c, 0.0)
    k = min(k, len(c))
    if k >= len(c):
        return np.full(len(c), 1.0 / len(c))
    sorted_c = np.sort(c)[::-1]
    threshold = sorted_c[k - 1]
    w = np.maximum(c - threshold + temperature * (sorted_c[-1] - sorted_c[0] + 1e-15), 0.0)
    s = float(w.sum())
    if s <= 0 or not np.isfinite(s):
        w2 = np.zeros(len(c))
        idx = np.argsort(c)[::-1][:k]
        w2[idx] = 1.0 / k
        return w2
    return w / s


def _random_init(n, rng, mode):
    t = np.linspace(0, 1, n)
    if mode == 0:
        c = rng.uniform(0.2, 0.8)
        s = rng.uniform(0.05, 0.2)
        x = np.exp(-((t - c) / s) ** 2)
    elif mode == 1:
        c1, c2 = sorted(rng.uniform(0.1, 0.9, 2))
        s = rng.uniform(0.05, 0.15)
        x = np.exp(-((t - c1) / s) ** 2) + 0.7 * np.exp(-((t - c2) / (s * 1.3)) ** 2)
    elif mode == 2:
        a = rng.uniform(0.5, 3.0)
        b = rng.uniform(0.5, 3.0)
        tt = np.clip(t, 1e-8, 1.0 - 1e-8)
        x = tt ** a * (1.0 - tt) ** b
    elif mode == 3:
        k = int(rng.integers(2, 8))
        x = np.zeros(n)
        bs = n // k
        for i in range(k):
            x[i * bs:(i + 1) * bs] = rng.uniform(0.2, 2.0)
    elif mode == 4:
        x = np.ones(n)
        x += rng.normal(0, 0.3, n)
        kernel = np.exp(-0.5 * (np.arange(-15, 16) / 4.0) ** 2)
        kernel /= kernel.sum()
        x = np.convolve(x, kernel, mode='same')
    elif mode == 5:
        # Multi-bump
        K = int(rng.integers(2, 6))
        centers = np.sort(rng.uniform(0.05, 0.95, K))
        sigmas = rng.uniform(0.03, 0.15, K)

            x += weights[j] * np.exp(-((t - centers[j]) / sigmas[j]) ** 2)
    else:
        x = np.ones(n)
    return _normalize(x)


def _run_descent(x0, rng, time_limit, phase=1):
    """Adaptive multiplicative descent on the probability simplex."""
    x = _normalize(x0)
    if x is None:
        return None, float('inf')
    n = len(x)
    floor = 1e-30 / n

    best = x.copy()
    best_score = _score(x)
    cur_score = best_score

    if phase == 1:
        beta = 15.0
        beta_max = 300.0
        topk_init = max(10, n // 4)
        alpha = 0.5
    elif phase == 2:
        beta = 80.0
        beta_max = 3000.0
        topk_init = max(5, n // 8)
        alpha = 0.4
    else:
        beta = 200.0
        beta_max = 8000.0
        topk_init = 5
        alpha = 0.3

    topk = topk_init
    deadline = time.monotonic() + time_limit
    stall = 0
    it = 0

    while time.monotonic() < deadline:
        c = _convolve_full(x, x)
        c = np.where(np.isfinite(c), c, 0.0)
        c = np.maximum(c, 0.0)

        cur_score = float(2.0 * n * np.max(c))

        if cur_score < best_score - 1e-12:
            best = x.copy()
            best_score = cur_score
            stall = 0
        else:
            stall += 1
            if stall > 80:
                x = _normalize(x * np.exp(rng.normal(0, 0.08, n)))
                if x is None:
                    x = best.copy()
                x = np.maximum(x, floor)
                x = x / x.sum()
                alpha = 0.4
                stall = 0
                continue

        # Compute weights
        span = float(np.max(c) - np.min(c)) + 1e-15
        use_topk = topk < len(c) and (it > 20 or phase >= 3)
        if use_topk:
            w = _topk_weights(c, topk, temperature=0.1 * span)
        else:
            w = _softmax_weights(c, beta)

        # Compute gradient
        grad = _compute_gradient(x, w)
        grad = np.where(np.isfinite(grad), grad, 0.0)
        grad = np.maximum(grad, 0.0)
        gmax = float(np.max(grad))
        if gmax < 1e-15:
            break
        grad = grad / gmax

        # Line search
        best_x = None
        best_s = cur_score
        best_a = alpha

        trial_alphas = []
        for m in [0.25, 0.5, 1.0, 1.5, 2.0, 3.0]:
            a = alpha * m
            if 0.02 <= a <= 25.0:
                trial_alphas.append(a)
        if not trial_alphas:
            trial_alphas = [0.1, 0.5]

        for a in trial_alphas:
            dec = np.exp(-a * grad)
            x_try = x * dec
            x_try = np.maximum(x_try, floor)
            s_try_total = float(x_try.sum())
            if s_try_total < 1e-20 or not np.isfinite(s_try_total):
                continue
            x_try = x_try / s_try_total
            s_try = _score(x_try)
            if s_try < best_s - 1e-14:
                best_s = s_try
                best_x = x_try
                best_a = a

        if best_x is not None:
            x = best_x
            alpha = best_a
            cur_score = best_s
        else:
            alpha = max(0.02, alpha * 0.6)
            if alpha < 0.03:
                alpha = 0.4
                if rng.random() < 0.3:
                    x = _normalize(x * np.exp(rng.normal(0, 0.04, n)))
                    if x is None:
                        x = best.copy()

        it += 1
        # Adapt parameters
        if it % 40 == 0:
            beta = min(beta * 1.4, beta_max)
            topk = max(1, topk // 2)

    return best, best_score

    best_final_score = candidates[0][0]

    for sc, arr in candidates[:5]:
        if time.monotonic() >= deadline - 5:
            break
        lst = [float(v) for v in arr]
        try:
            fs = float(evaluate_sequence(lst))
        except Exception:
            fs = _score(arr)
        if np.isfinite(fs) and fs < best_final_score:
            best_final_score = fs
            best_list = lst

    print(f"Final answer score: {best_final_score:.6f}, length={len(best_list)}")
    return best_list
