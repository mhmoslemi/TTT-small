def fft_convolution(a):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def fast_score(p):
    """Compute 2*n*max(p*p)/(sum(p))^2 using FFT."""
    n = p.size
    if n == 0:
        return float("inf")
    S = float(np.sum(p))
    if S <= 0.0:
        return float("inf")
    b = fft_convolution(p)
    M = float(np.max(b))
    return float(2.0 * n * M / (S * S))


def smooth_max_and_grad(b, beta):
    """
    Smooth max of b and its gradient (softmax weights).
    Returns (smooth_max_value, weights).
    """
    m = float(np.max(b))
    if not np.isfinite(m) or m <= 0:
        w = np.ones_like(b) / len(b)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0 or not np.isfinite(Z):
        e = np.zeros_like(b)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return G, w


def simplex_gradient(p, b, beta):
    """
    Compute gradient of n * smooth_max(b) w.r.t. p, projected onto simplex tangent.
    g_i = 2 * sum_j w_{i+j} * a_j (the smooth subgradient of max(b) w.r.t. a_i)
    """
    n = p.size
    _, w = smooth_max_and_grad(b, beta)
    # s_i = sum_j w_{i+j} * p_j  (cross-correlation of p and w)
    # Use FFT
    need = n + len(w) - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    # s_i = sum_j p_j * w_{i+j} = C[n-1-i]... let me think.
    # C[k] = sum_j p_j * w_rev[k-j] = sum_j p_j * w[n-1-(k-j)] = sum_j p_j * w[n-1-k+j]
    # We want s_i = sum_j p_j * w_{i+j}
    # w_{i+j} = w_rev[n-1-i-j]
    # C[k] = sum_j p_j * w_rev[k-j]
    # Set k-j = n-1-i-j => k = n-1-i
    # C[n-1-i] = sum_j p_j * w_rev[n-1-i-j] = sum_j p_j * w[i+j] ✓
    s = C[n-1-n+1:n+1][::-1].copy()  # C[n-1-(n-1):n-1-0+1] = C[0:n][::-1]
    # Wait: s_i = C[n-1-i] for i=0..n-1, so s = C[n-1], C[n-2], ..., C[0]
    s = C[n-1::-1].copy()
    grad = 2.0 * s
    # Project onto simplex tangent: subtract mean
    grad = grad - float(np.mean(grad))
    return grad


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    n = v.size
    if n == 0:
        return v
    if target <= 0:
        return np.zeros_like(v)
    # Bisection on the threshold
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0 or not np.isfinite(s):
        return np.full(n, target / n)
    q *= target / s
    return q


def normalize_simplex(p):
    """Clip to non-negative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    n = p.size
    if n == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return p / s

    # Asymmetric: mass on left or right
    for frac in (0.3, 0.4, 0.6, 0.7):
        p = np.ones(n)
        p[:int(n*frac)] = 1.0
        p[int(n*frac):] = 0.1
        p = np.maximum(p, 1e-10)
        starts.append(p / p.sum())
    
    # Cosine bell
    p = np.cos(np.pi * x) ** 2
    p = np.maximum(p, 1e-10)
    starts.append(p / p.sum())
    
    # Gaussian centered
    p = np.exp(-20 * (x - 0.5) ** 2)
    p = np.maximum(p, 1e-10)
    starts.append(p / p.sum())
    
    # Gaussian off-center
    for center in (0.3, 0.7):
        p = np.exp(-20 * (x - center) ** 2)
        p = np.maximum(p, 1e-10)
        starts.append(p / p.sum())
    
    # Random
    for _ in range(4):
        p = rng.uniform(0, 1, n)
        p = np.maximum(p, 1e-10)
        starts.append(p / p.sum())
    
    # Chirp-like: modulated by sin
    for freq in (1, 2, 3):
        p = 1.0 + 0.5 * np.sin(2 * np.pi * freq * x)
        p = np.maximum(p, 1e-10)
        starts.append(p / p.sum())
    
    return starts[:count]


def block_optimize(p_init, n, deadline, rng):
    """
    Block-level optimization using L-BFGS-B.
    """
    if not HAVE_SCIPY:
        return p_init, fast_score(p_init)
    
    best_p = p_init.copy()
    best_score = fast_score(best_p)
    
    candidate_Bs = []
    for B in (8, 16, 32, 64, 128, 256, 512):
        if B < n and n % B == 0:
            candidate_Bs.append(B)
    
    for B in candidate_Bs:
        if time.monotonic() >= deadline:
            break
        
        L = n // B
        # Initialize block weights from current best_p
        p_aug = np.concatenate([best_p, np.zeros(n - n % B)])
        blocks = p_aug.reshape(B, L).sum(axis=1)
        blocks = np.maximum(blocks, 1e-10)
        c0 = np.log(blocks[:-1] / blocks[-1])  # relative log-weights
        c0 = np.clip(c0, -4, 4)
        
        bounds = [(-4.0, 4.0)] * (B - 1)
        
        # Objective function
        def block_obj(c):
            c = np.asarray(c, dtype=np.float64).reshape(-1)
            z = np.zeros(B, dtype=np.float64)
            if B > 1:
                z[:-1] = c[:B-1]
            z = z - np.max(z)
            wb = np.exp(z)
            s = wb.sum()
            if s <= 0:
                wb = np.ones(B) / B
            else:
                wb /= s
            H = wb / L
            p = np.repeat(H, L)[:n]
            S = p.sum()
            if S <= 0:
                return 1e10, np.zeros(B-1)
            b = fft_convolution(p)
            G, _ = smooth_max_and_grad(b, 50000.0)
            # Gradient
            _, w = smooth_max_and_grad(b, 50000.0)
            # s_i = sum_j w_{i+j} * p_j
            need2 = n + len(w) - 1
            size2 = 1
            while size2 < need2:
                size2 <<= 1
            P2 = np.fft.rfft(p, size2)
            WR2 = np.fft.rfft(w[::-1], size2)
            C2 = np.fft.irfft(P2 * WR2, size2)[:need2]
            s_vec = C2[n-1::-1]
            g = 2.0 * s_vec
            Gsum = g.reshape(B, L).sum(axis=1)
            Wavg = np.dot(wb, Gsum)
            gradz = (wb / L) * (Gsum - Wavg)
            grad = n * gradz[:-1]
            grad = np.where(np.isfinite(grad), grad, 0.0)
            return float(n * G), grad
        
        try:
            res = minimize(block_obj, c0, method='L-BFGS-B', jac=True,
                          bounds=bounds, options={'maxiter': 60, 'ftol': 1e-10})
            c = res.x
            z = np.zeros(B)
            if B > 1:
                z[:-1] = c[:B-1]
            z = z - np.max(z)
            wb = np.exp(z)
            wb /= wb.sum()
            H = wb / L
            p = np.repeat(H, L)[:n]
            sc = fast_score(p)
            if np.isfinite(sc) and sc < best_score:
                best_score = sc
                best_p = p.copy()
                print(f"  block B={B}: score={sc:.8f}")
        except Exception:
            pass
    
    return best_p, best_score


def gradient_refine(p_init, deadline, beta_start=5000, beta_end=500000):
    """
    Full projected gradient with Nesterov momentum and adaptive beta.
    """
    p = normalize_simplex(p_init)
    n = p.size
    best_p = p.copy()
    best_score = fast_score(best_p)
    
    v = np.zeros(n)  # momentum
    momentum = 0.9
    lr = 1.0
    it = 0
    no_improve = 0
    
    while it < 50000 and time.monotonic() < deadline:
        it += 1
        
        # Adaptive beta
        progress = min(1.0, it / 10000.0)
        beta = beta_start * (beta_end / beta_start) ** progress
        
        b = fft_convolution(p)
        G, w = smooth_max_and_grad(b, beta)
        cur_score = 2.0 * n * float(np.max(b))
        
        if cur_score < best_score - 1e-15:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 2000:
            # Try to escape: reduce momentum, increase lr slightly
            momentum *= 0.8
            if momentum < 0.3:
                break
        
        # Compute gradient
        need = n + len(w) - 1
        size = 1
        while size < need:
            size <<= 1
        P = np.fft.rfft(p, size)
        WR = np.fft.rfft(w[::-1], size)
        C = np.fft.irfft(P * WR, size)[:need]
        s_vec = C[n-1::-1]
        grad = 2.0 * s_vec
        grad = grad - float(np.mean(grad))
        
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        
        # Nesterov step
        v = momentum * v + grad / gnorm
        q = project_simplex(p - lr * v / n, 1.0)
        
        # Check if step improved
        qb = fft_convolution(q)
        q_score = 2.0 * n * float(np.max(qb))
        
        if q_score < cur_score + 1e-12:
            p = q
            if lr < 5.0:
                lr = min(5.0, lr * 1.05)
        else:
            # Reject step, reduce lr
            lr *= 0.7
            v = 0.5 * v
            if lr < 1e-6:
                break
        
        if it % 500 == 0:
            print(f"    grad it={it} beta={beta:.0f} score={best_score:.8f} lr={lr:.4f}")
    
    return best_p, best_score


def coordinate_descent(p_init, deadline, max_rounds=50):
    """
    Cyclic coordinate descent with incremental convolution updates.
    """
    p = normalize_simplex(p_init)
    n = p.size
    b = fft_convolution(p).copy()
    S = float(np.sum(p))
    best_score = 2.0 * n * float(np.max(b)) / (S * S)
    best_p = p.copy()
    
    round_num = 0
    while round_num < max_rounds and time.monotonic() < deadline:
        round_num += 1
        improved = 0
        
        for i in range(n):
            old = p[i]
            if old < 1e-15:
                # Try a small positive value
                trial = np.array([0.0, 1e-6, 5e-6, 1e-5, 5e-5])
            else:
                # Try values around old
                trial = np.array([0.0, old*0.25, old*0.5, old*0.75, old*1.0,
                                  old*1.25, old*1.5, old*2.0])
            
            best_x = old
            best_sc = best_score
            
            # Valid k range: k from max(0, i+0) to min(2n-2, i+n-1)
            # i.e., k-i in [0, n-1], so k in [i, i+n-1]
            k_lo = i
            k_hi = min(2 * n - 2, i + n - 1)
            
            for x in trial:
                x = max(0.0, float(x))
                if x == old:
                    continue
                delta = x - old
                new_S = S - old + x
                if new_S < 1e-10:
                    continue
                
                # Apply update to b
                # For k in [k_lo, k_hi], k != 2i: b[k] += 2*delta*a[k-i]
                # For k = 2i (if in range): b[k] += x^2 - old^2
                ks = np.arange(k_lo, k_hi + 1)
                ai_vals = a_k_minus_i = p[ks - i]  # a[k-i] for each k
                
                if 2 * i >= k_lo and 2 * i <= k_hi:
                    mask = ks != 2 * i
                    b[ks[mask]] += 2 * delta * ai_vals[mask]
                    b[2 * i] += x * x - old * old
                else:
                    b[ks] += 2 * delta * ai_vals
                
                new_max_b = float(np.max(b))
                new_sc = 2.0 * n * new_max_b / (new_S * new_S)
                
                if new_sc < best_sc:
                    best_sc = new_sc
                    best_x = x
                else:
                    # Undo
                    if 2 * i >= k_lo and 2 * i <= k_hi:
                        b[ks[mask]] -= 2 * delta * ai_vals[mask]
                        b[2 * i] -= x * x - old * old
                    else:
                        b[ks] -= 2 * delta * ai_vals
            
            # Apply best update
            if best_x != old:
                delta = best_x - old
                ks = np.arange(k_lo, k_hi + 1)
                ai_vals = p[ks - i]
                if 2 * i >= k_lo and 2 * i <= k_hi:
                    mask = ks != 2 * i
                    b[ks[mask]] += 2 * delta * ai_vals[mask]
                    b[2 * i] += best_x * best_x - old * old
                else:
                    b[ks] += 2 * delta * ai_vals
                p[i] = best_x
                S = S - old + best_x
                improved += 1
        
        new_score = 2.0 * n * float(np.max(b)) / (S * S)
        if new_score < best_score:
            best_score = new_score
            best_p = p.copy()
        
        if round_num % 5 == 0:
            print(f"    coord round={round_num} score={best_score:.8f} improved={improved}")
        
        if improved == 0:
            break
    
    return best_p, best_score
