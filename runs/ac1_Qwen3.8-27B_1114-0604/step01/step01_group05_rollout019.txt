```python
def lp_direction(a, M):
    """Solve LP: max sum(x) s.t. (a*x)[k] <= M for all k, x >= 0"""
    n = len(a)
    # Build sparse matrix A where A[k,j] = a[k-j] if 0 <= k-j < n
    # A is (2n-1) x n
    rows = []
    cols = []
    vals = []
    for k in range(2*n-1):
        for j in range(max(0, k-n+1), min(n, k+1)):
            rows.append(k)
            cols.append(j)
            vals.append(a[k-j])
    
    A = scipy.sparse.csr_matrix((vals, (rows, cols)), shape=(2*n-1, n))
    c = np.ones(n)  # maximize c^T x
    b = M * np.ones(2*n-1)
    
    # scipy linprog minimizes, so negate c
    res = scipy.optimize.linprog(-c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    return res.x
```

```python
import numpy as np
import time
```

```python
    from scipy.optimize import minimize as scipy_minimize

    HAVE_SCIPY = False


def fft_convolution(a):
    """Linear convolution a*a via FFT."""
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    return np.maximum(b, 0.0)


def fft_cross_conv(a, b):
    """Linear convolution a*b via FFT. a has length n, b has length n."""
    n = a.size
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    FA = np.fft.rfft(a, size)
    FB = np.fft.rfft(b, size)
    c = np.fft.irfft(FA * FB, size)[:need]
    return np.maximum(c, 0.0)


def smooth_max_and_weights(b, beta):
    """Smooth max of b and softmax weights."""
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float("inf"), np.empty(0, dtype=np.float64)
    m = float(np.max(b))
    if not np.isfinite(m):
        return float("inf"), np.zeros_like(b)
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b) / b.size
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w


def fast_score(p):
    """Fast objective: 2*n*max(p*p)/sum(p)^2."""
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float("inf")
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float("inf")
    b = fft_convolution(p)
    M = float(np.max(b))
    if not np.isfinite(M):
        return float("inf")
    return float(2.0 * n * M / (S * S))


def corr_pw(p, w):
    """s_i = sum_j p_j w_{i+j} for i=0..n-1 (gradient helper)."""
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    if n == 1:
        return np.array([float(np.dot(p, w[:1]))], dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n:m][::-1].copy()
    return s


def project_simplex(v, target=1.0):
    """Project v onto {x >= 0, sum(x) = target}."""
    v = np.asarray(v, dtype=np.float64)
    n = v.size
    if n == 0:
        return v
    if target <= 0.0:
        return np.zeros(n, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(n, target / n, dtype=np.float64)
    lo = float(np.min(v)) - 1.0
    hi = float(np.max(v)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.maximum(v - mid, 0.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    q = np.maximum(v - tau, 0.0)
    s = float(np.sum(q))
    if s <= 0.0 or not np.isfinite(s):
        q = np.full(n, target / n, dtype=np.float64)
    else:
        q *= target / s
    return q


def normalize_simplex(p):
    """Clip nonnegative and normalize to sum 1."""
    p = np.asarray(p, dtype=np.float64).copy()
    n = p.size
    if n == 0:
        return p
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n, dtype=np.float64)
    p *= 1.0 / s
    return p


def lp_primal_dual(a, M, n, max_iter=3000, warm_start_y=None, warm_start_x=None):
    """
    Solve LP: max 1^T x s.t. (a*x)[k] <= M for all k=0..2n-2, x >= 0
    using primal-dual gradient method.
    Returns optimal x (and the dual variables for warm starting).
    """
    if warm_start_x is not None:
        x = warm_start_x.copy()
    else:
        x = np.ones(n) / n
    if warm_start_y is not None:
        y = warm_start_y.copy()
    else:
        y = np.zeros(2 * n - 1)

    # Adaptive step size based on the magnitude of a
    a_max = float(np.max(np.abs(a)))
    if a_max <= 0:
        a_max = 1.0
    eta = 0.5 / (a_max * n)  # conservative step
    eta = max(eta, 1e-8)
    eta = min(eta, 1.0)

    best_x = x.copy()
    best_obj = float(np.sum(x))

    # Precompute FFT size
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    FA = np.fft.rfft(a, size)

    for t in range(max_iter):
        # Compute A @ x = a * x (cross convolution)
        FX = np.fft.rfft(x, size)
        ax = np.fft.irfft(FA * FX, size)[:need]
        ax = np.maximum(ax, 0.0)

        # Dual update: y += eta * (M - A@x), project to y >= 0
        y = np.maximum(0.0, y + eta * (M - ax))

        # Compute A^T @ y: (A^T y)[j] = sum_m a[m] * y[m+j]
        # This is the cross-correlation of y and a
        FY = np.fft.rfft(y, size)
        # (A^T y)[j] = sum_m a[m] y[m+j]
        # = convolution of a[::-1] with y, evaluated at position j + n - 1
        # (a_rev * y)[k] = sum_i a_rev[i] y[k-i] = sum_i a[n-1-i] y[k-i]
        # Let j' = n-1-i: sum_j' a[j'] y[k-1-j']
        # For k = j + n - 1: sum_j' a[j'] y[j + j'] = (A^T y)[j] ✓
        A_rev = a[::-1]
        FAr = np.fft.rfft(A_rev, size)
        aty_full = np.fft.irfft(FAr * FY, size)[:need]
        aty = aty_full[n - 1:n - 1 + n]

        # Primal update: x += eta * (1 - A^T y), project to x >= 0
        x = np.maximum(0.0, x + eta * (1.0 - aty))

        obj = float(np.sum(x))
        if obj > best_obj:
            best_obj = obj
            best_x = x.copy()

        # Early stopping if converged
        if t > 100 and obj < best_obj * (1 - 1e-8):
            if t % 200 == 0 and (best_obj - obj) < 1e-6:
                break

    return best_x, y, best_obj


def exact_max_subgradient(a, n):
    """
    Compute the subgradient of max_k (a*a)[k] with respect to a.
    Returns a vector sg where sg[j] = sum over argmax positions of 2*a[k*-j].
    Uses a weighted combination of top peaks for stability.
    """
    b = fft_convolution(a)
    M = float(np.max(b))
    if M <= 0:
        return np.zeros(n, dtype=np.float64), M

    # Find all positions where b is close to max (within 1e-6 of max)
    tol = M * 1e-6 + 1e-15
    peaks = np.where(b >= M - tol)[0]

    if len(peaks) == 0:
        peaks = np.array([int(np.argmax(b))])

    # Weight peaks by how close they are to max
    weights = (b[peaks] - M + tol) / tol + 1.0
    weights = np.maximum(weights, 0.0)
    if weights.sum() <= 0:
        weights = np.ones(len(peaks))
    weights /= weights.sum()

    # Subgradient: weighted sum of gradients of b[k] at each peak
    sg = np.zeros(n, dtype=np.float64)
    for w_k, k in zip(weights, peaks):
        # db[k]/da[j] = 2*a[k-j] if 0<=k-j<n else 0
        lo = max(0, k - n + 1)
        hi = min(n, k + 1)
        if lo < hi:
            # a[k-j] for j in [lo, hi): indices k-lo, k-lo+1, ..., k-hi+1
            idx = k - np.arange(lo, hi)
            sg[lo:hi] += 2.0 * w_k * a[idx]

    return sg, M


def gradient_refine(p0, deadline, beta, max_no_improve=5000, rng=None):
    """
    Projected gradient refinement with smooth max.
    Returns (best_p, best_score).
    """
    p = normalize_simplex(p0)
    n = p.size
    if n == 0:
        return p, float("inf")

    best_p = p.copy()
    best_score = fast_score(p)
    step = 1.0
    no_improve = 0
    it = 0

    while it < 10000000 and time.monotonic() < deadline:
        it += 1
        b = fft_convolution(p)
        G, w = smooth_max_and_weights(b, beta)
        if not np.isfinite(G):
            break

        cur_smooth = float(n) * G
        m = float(np.max(b))
        cur_score = float(2.0 * n * m)

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 500 == 0:
            print(f"  grad_refine it={it} best={best_score:.8f} step={step:.2e} no_improve={no_improve}")

        if no_improve > max_no_improve:
            break

        svec = corr_pw(p, w)
        svec = np.maximum(svec, 0.0)
        g = 2.0 * svec
        grad = g - float(np.mean(g))

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-14:
            break

        accepted = False
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001):
            alpha = step * mult
            q = project_simplex(p - alpha * grad, 1.0)

            m2 = float(np.max(qb))
            G2, _ = smooth_max_and_weights(qb, beta)
            if not np.isfinite(G2):
                continue
            if float(n) * G2 < cur_smooth - 1e-12:
                p = q
                cur_smooth = float(n) * G2
                accepted = True
                break

        if not accepted:
            step *= 0.5
            if step < 1e-9:
                break
        else:
            step = min(step * 1.15, 5.0)

    return best_p, best_score

    while it < 5000000 and time.monotonic() < deadline:
        it += 1

        b = fft_convolution(p)
        M = float(np.max(b))
        cur_score = float(2.0 * n * M)

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 200 == 0:
            print(f"  exact_subgrad it={it} best={best_score:.8f} step={step:.2e} no_improve={no_improve}")

        if no_improve > max_no_improve:
            break

        sg, M = exact_max_subgradient(p, n)
        # Center the gradient (remove mean for simplex projection efficiency)
        grad = sg - float(np.mean(sg))

        gnorm = float(np.linalg.norm(grad))
        if not np.isfinite(gnorm) or gnorm < 1e-14:
            break

        accepted = False
        for mult in (1.0, 0.7, 0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005):
            alpha = step * mult
            q = project_simplex(p - alpha * grad, 1.0)
            qb = fft_convolution(q)
            m2 = float(np.max(qb))
            sc2 = float(2.0 * n * m2)
            if sc2 < cur_score - 1e-13:
                p = q
                accepted = True
                break

        if not accepted:
            step *= 0.5
            if step < 1e-10:
                break
        else:
            step = min(step * 1.1, 3.0)

    return best_p, best_score


def lp_escape(p0, deadline, rng, max_outer=500):
    """
    LP-based escape: at each step, solve the LP to find a direction
    that keeps (a*x) bounded, then mix.
    """
    p = normalize_simplex(p0)
    n = p.size
    if n == 0:
        return p, float("inf")

    best_p = p.copy()
    best_score = fast_score(p)
    it = 0
    no_improve = 0

    # Warm start for LP
    x_warm = p.copy()
    y_warm = None

    while it < max_outer and time.monotonic() < deadline:
        it += 1

        b = fft_convolution(p)
        M = float(np.max(b))
        cur_score = float(2.0 * n * M)

        if cur_score < best_score - 1e-14:
            best_score = cur_score
            best_p = p.copy()
            no_improve = 0
        else:
            no_improve += 1

        if it % 10 == 0:
            print(f"  lp_escape it={it} best={best_score:.8f} M={M:.6e} no_improve={no_improve}")

        if no_improve > 100:
            break

        # Solve LP: max sum(x) s.t. (p*x)[k] <= M, x >= 0
        x_dir, y_new, obj = lp_primal_dual(p, M, n, max_iter=2000,
                                            warm_start_y=y_warm, warm_start_x=x_warm)
        x_warm = x_dir
        y_warm = y_new

        if obj <= 1.0 + 1e-6:
            # LP says we can't improve much; break
            if no_improve > 20:
                break
            # Try a smaller M (shrink the constraint)
            M = M * 0.999
            continue

        # Normalize direction
        x_norm = x_dir / float(np.sum(x_dir))

        # Try mixing with different t values
        best_t = None
        best_mix_score = best_score
        for t in (0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.5):
            q = (1.0 - t) * p + t * x_norm
            q = project_simplex(q, 1.0)
            qb = fft_convolution(q)
            m2 = float(np.max(qb))
            sc2 = float(2.0 * n * m2)
            if sc2 < best_mix_score:
                best_mix_score = sc2
                best_t = t

        if best_t is not None:
            p = (1.0 - best_t) * p + best_t * x_norm
            p = project_simplex(p, 1.0)
        else:
            # No improvement from mixing; try small perturbation
            if rng is None:
                rng_p = np.random.default_rng(it)
            else:
                rng_p = rng
            pert = 0.01 * rng_p.standard_normal(n)
            p = project_simplex(p + pert, 1.0)

    return best_p, best_score


def spectral_init(n, freqs, phases, rng):
    """Generate a sequence from spectral components."""
    x = np.zeros(n)
    for f, p in zip(freqs, phases):
        x += np.cos(2 * np.pi * f * np.arange(n) / n + p)
    x = np.maximum(x + 1.0, 0.01)
    return normalize_simplex(x)


def diverse_starts(n, rng):
    """Generate diverse starting points."""
    starts = []

    # Uniform
    starts.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Gaussian bump
    x = np.arange(n)
    for sigma_frac in (0.1, 0.15, 0.2, 0.25, 0.3):
        sigma = sigma_frac * n
        g = np.exp(-0.5 * ((x - n / 2) / sigma) ** 2)
        starts.append(normalize_simplex(g))

    # Two bumps
    for sep in (0.3, 0.4, 0.5):
        x1 = sep * n
        x2 = (1 - sep) * n
        sigma = 0.1 * n
        g = np.exp(-0.5 * ((x - x1) / sigma) ** 2) + np.exp(-0.5 * ((x - x2) / sigma) ** 2)
        starts.append(normalize_simplex(g))

    # Power law
    for alpha in (0.5, 1.0, 2.0, 3.0):
        d = np.abs(x - n / 2) + 1.0
        p = d ** (-alpha)
        starts.append(normalize_simplex(p))

    # Cosine modulated
    for a in (0.1, 0.3, 0.5):
        p = 1.0 + a * np.cos(2 * np.pi * x / n)
        starts.append(normalize_simplex(np.maximum(p, 0.01)))

    # Random smooth (low-pass filtered)
    for _ in range(3):
        z = rng.standard_normal(n)
        # Simple low-pass: boxcar filter
        k = max(3, n // 64)
        kernel = np.ones(k) / k
        z = np.convolve(z, kernel, mode='same')
        z = np.tanh(2.0 * z)
        starts.append(normalize_simplex(np.maximum(z + 1.5, 0.01)))

    # Truncated cosine (sin^2)
    for shift in (0.0, 0.25, 0.5):
        p = np.sin(np.pi * (x / n + shift)) ** 2
        starts.append(normalize_simplex(np.maximum(p, 1e-6)))

    return starts


def official_score(seq):
    """Call evaluate_sequence safely."""
    try:
        val = evaluate_sequence(seq)
        val = float(val)
    except Exception:
        return float("inf")
    if np.isfinite(val):
        return val
    return float("inf")


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Returns a list of nonnegative floats
    that minimizes the evaluation function.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 0:
        return [1.0]

    reserve = 60.0
    start_time = time.monotonic()
    deadline = start_time + max(10.0, budget_s - reserve)

    # Choose sequence length
    if budget_s < 300.0:
        N = 4096
    elif budget_s < 1200.0:
        N = 8192
    else:
        N = 8192

    rng = np.random.default_rng(seed)

    print(f"=== Propose candidate: N={N}, budget={budget_s:.0f}s, seed={seed} ===")

    # --- Initialize from height_sequence_1 if available ---
    best_p = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            hs = np.array(height_sequence_1, dtype=np.float64)
            hs = np.maximum(hs, 0.0)
            hs_sum = float(np.sum(hs))
            if hs_sum > 0 and hs.size == N:
                best_p = hs / hs_sum
                print(f"Loaded height_sequence_1, sum={hs_sum:.4f}")
            elif hs_sum > 0:
                # Resize to N by interpolation or truncation
                if hs.size > N:
                    best_p = hs[:N] / float(np.sum(hs[:N]))
                else:
                    # Upsample
                    idx = np.linspace(0, hs.size - 1, N)
                    best_p = np.interp(idx, np.arange(hs.size), hs)
                    best_p = np.maximum(best_p, 0.0)
                    best_p = best_p / float(np.sum(best_p))
                print(f"Resized height_sequence_1 to N={N}")
    except Exception as e:
        print(f"Could not load height_sequence_1: {e}")

    if best_p is None:
        best_p = np.full(N, 1.0 / N, dtype=np.float64)
        print("Starting from uniform")

    best_score = fast_score(best_p)
    print(f"Initial score: {best_score:.8f}")

    # --- Phase 1: Smooth max gradient refinement (40% of time) ---
    t1 = time.monotonic() + 0.40 * (deadline - start_time)
    print(f"\n--- Phase 1: Smooth max gradient (until t={t1-start_time:.0f}s) ---")

    # First, try a few diverse starts and pick the best
    if time.monotonic() < t1 - 30:
        starts = diverse_starts(N, rng)
        best_start_score = best_score
        for i, s in enumerate(starts):
            if time.monotonic() >= t1 - 30:
                break
            sc = fast_score(s)
            if sc < best_start_score:
                best_start_score = sc
                best_p = s.copy()
                print(f"  Better start {i}: score={sc:.8f}")

        # Incorporate the loaded best as a start too
        sc_loaded = fast_score(best_p)
        if sc_loaded < best_start_score:
            best_start_score = sc_loaded

    # Do gradient refinement
    for beta in (50000.0, 200000.0, 500000.0):
        if time.monotonic() >= t1 - 10:
            break
        sub_deadline = min(t1 - 5, time.monotonic() + 0.3 * (t1 - time.monotonic()))
        p_new, sc_new = gradient_refine(best_p, sub_deadline, beta,
                                         max_no_improve=3000, rng=rng)
        if sc_new < best_score:
            best_score = sc_new
            best_p = p_new.copy()
            print(f"  After beta={beta:.0f}: score={best_score:.8f}")

    # --- Phase 2: LP-based escape (30% of time) ---
    t2 = time.monotonic() + 0.30 * (deadline - t1)
    if time.monotonic() < deadline - 10:
        print(f"\n--- Phase 2: LP escape (until t={t2-start_time:.0f}s) ---")
        p_new, sc_new = lp_escape(best_p, min(t2, deadline - 10), rng)
        if sc_new < best_score:
            best_score = sc_new
            best_p = p_new.copy()
            print(f"  After LP escape: score={best_score:.8f}")

    # --- Phase 3: Exact subgradient refinement + perturbation (20% of time) ---
    remaining = deadline - time.monotonic() - 20
    if remaining > 30:
        print(f"\n--- Phase 3: Exact subgradient + perturb ({remaining:.0f}s) ---")

        # Multiple attempts with perturbations
        for attempt in range(8):
            if time.monotonic() >= deadline - 20:
                break
            alloc = remaining / (8 - attempt)
            sub_dl = time.monotonic() + min(alloc, 200.0)

            # Perturb the best
            if attempt == 0:
                p_start = best_p.copy()
            else:
                pert_size = 0.005 * (1 + attempt * 0.5)
                p_start = project_simplex(best_p + pert_size * rng.standard_normal(N), 1.0)

            p_new, sc_new = exact_subgradient_refine(p_start, sub_dl,
                                                      max_no_improve=1000, rng=rng)
            if sc_new < best_score:
                best_score = sc_new
                best_p = p_new.copy()
                print(f"  Attempt {attempt}: score={best_score:.8f}")

    # --- Phase 4: Final short gradient polish ---
    if time.monotonic() < deadline - 10:
        print(f"\n--- Phase 4: Final polish ---")
        p_new, sc_new = gradient_refine(best_p, deadline - 10, beta=2000000.0,
                                         max_no_improve=2000, rng=rng)
        if sc_new < best_score:
            best_score = sc_new
            best_p = p_new.copy()
            print(f"  After final polish: score={best_score:.8f}")

    # --- Final: one more LP escape if time remains ---
    if time.monotonic() < deadline - 30:
        print(f"\n--- Final LP escape ---")
        p_new, sc_new = lp_escape(best_p, deadline - 10, rng, max_outer=200)
        if sc_new < best_score:
            best_score = sc_new
            best_p = p_new.copy()
            print(f"  After final LP: score={best_score:.8f}")

    # --- Prepare output ---
    final_p = normalize_simplex(best_p)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_seq = [float(x) for x in final_p]

    # Verify with official evaluator
    off = official_score(final_seq)
    print(f"\nOfficial score: {off:.8f}")

    if not np.isfinite(off) or off > 2.0:
        # Fallback to uniform
        final_seq = [1.0 / N] * N
        off2 = official_score(final_seq)
        print(f"Fallback uniform score: {off2:.8f}")
        return final_seq

    return final_seq
```
