But wait, when `j = k-j` (i.e., `2j = k`), the two contributions come from the SAME term `a[j]*a[j]`, and the perturbation is:
`d(a[j]*a[j])/da[j] = 2*a[j]`.
And `2*a[k-j] = 2*a[j]` in this case. So the formula `2*a[k-j]` works in all cases.

Great! So:
`grad_k = 2 * shift(a, k)` where `shift(a, k)[j] = a[k-j]` if `0 <= k-j < n`, else 0.

In other words, `grad_k[j] = 2 * a[k-j]` (with zero-padding for out-of-range).

This is just a reversed and shifted version of `a`, multiplied by 2.

`grad_total[j] = (2/m) * sum_{l=1}^{m} a[k_l - j]` (with zero-padding).

def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.array([], dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(z)
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0.0:
        return np.full_like(z, 1.0 / z.size, dtype=np.float64)
    return e / s


def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    N = 1 << (2 * n - 1).bit_length()
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:2 * n - 1].real
    return np.maximum(c, 0.0)


def _fast_score(a):

        return np.inf

    c = _fft_conv(a)
    m = float(np.max(c))
    if not np.isfinite(m):

    return float(2 * a.size * m / (s * s))

    z0 = np.asarray(z0, dtype=np.float64).ravel()

def _z_from_f(f):

    if f.size != _N:

    f = np.where(np.isfinite(f), f, 1.0)

    z = np.log(f)
    z = z - float(np.mean(z))
    z = np.clip(z, -_BOUND + 1.0, _BOUND - 1.0)

def _upsample(a, n_new):
    """Upsample a sequence to a new length via linear interpolation."""
    a = np.asarray(a, dtype=np.float64).ravel()
    n_old = a.size
    if n_old == n_new:
        return a.copy()
    x_old = np.linspace(0.0, 1.0, n_old)
    x_new = np.linspace(0.0, 1.0, n_new)
    a_new = np.interp(x_new, x_old, a)
    return np.maximum(a_new, 1e-10)

    a /= s
    return a
