```python
import numpy as np

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _score(a):
    """Compute the evaluation score."""
    arr = np.asarray(a, dtype=np.float64)
    n = arr.size
    s = arr.sum()
    if s < 0.01:
        return np.inf

    mx = np.max(c)

        return np.ones(n) / n
    return a / s

    n = a.size

    max_c = np.max(c)

    else:
        # Find top K lags

        # Also include lags within a small tolerance of the max
```

```python
"""
Low-peak nonnegative step-function search using smooth-max gradient descent.

Uses a multi-harmonic sinusoidal initialization followed by active-set
gradient optimization. The objective is 2*n*max(convolve(a,a))/(sum a)^2.
"""

import numpy as np
import time


def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n


def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]


def _score_fast(a):
    """Fast internal score computation."""
    arr = np.asarray(a, dtype=np.float64)
    n = arr.size
    s = float(arr.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    mx = float(np.max(c))
    return float(2.0 * n * mx / (s * s))


def _make_candidate(n, harmonics):
    """
    Create a candidate from harmonic parameters.
    harmonics is a list of (amplitude, frequency, phase) tuples.
    """
    i = np.arange(n, dtype=np.float64)
    a = np.ones(n, dtype=np.float64)
    for (amp, freq, phase) in harmonics:
        a += amp * np.sin(2.0 * np.pi * freq * i / n + phase)
    a = np.clip(a, 0.0, None)
    s = float(a.sum())
    if s < 0.01:
        return np.ones(n) / n
    return a / s
```
