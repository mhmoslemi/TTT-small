```python
"""
Low-peak self-convolution search for non-negative sequences.

The evaluator is scale invariant until the value cap is reached.  Working with
a simplex vector p (sum 1), the score is 2 * max_k (n * (p*p)[k]).  We
optimize p by annealing power objectives

    q -> sum_k (D_k / mean(D))^q,   D_k = n * (p*p)_k,

whose gradients are cheap to compute by FFT.  Low powers flatten D, while
higher powers approximate the min-max objective.  The code uses
multiplicative Adam-style updates, many asymmetric smooth initializations,
and periodic smooth restarts from the incumbent.
"""

import numpy as np
import time

# ---------------------------------------------------------------------------
# Stage plans.  Each entry is (mode, param, eta, time_fraction).
# mode 'pow' uses weights proportional to (D / mean(D))**param.
# mode 'lse' uses softmax(param * (D - mean(D))).
# ---------------------------------------------------------------------------

QUICK_PLAN = [

# Basic helpers

def next_pow2(x):
    x = int(max(1, x))
    return 1 << (x - 1).bit_length()


def clean_sequence(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    np.clip(a, 0.0, 1000.0, out=a)
    return a


def normalize_p(a, floor=None):
    a = clean_sequence(a).astype(np.float64, copy=True)
    if a.size == 0:
        return a
    s = a.sum()
    if not np.isfinite(s) or s <= 0.0:
        a = np.ones(a.size, dtype=np.float64)
    else:
        a = a / s
    if floor is None:
        floor = 1e-15 / max(1, a.size)
    np.maximum(a, floor, out=a)
    s = a.sum()
    if not np.isfinite(s) or s <= 0.0:
        a = np.ones(a.size, dtype=np.float64)
        s = float(a.size)
    a = a / s
    return a


def normalize_inplace(p, floor):
    s = p.sum()
    if not np.isfinite(s) or s < 1e-12:
        p.fill(1.0)
        s = float(p.size)
    else:
        p /= s
    np.maximum(p, floor, out=p)
    s = p.sum()
    if not np.isfinite(s) or s <= 0.0:
        p.fill(1.0)
        s = float(p.size)
    p /= s


# ---------------------------------------------------------------------------
# FFT-based convolution / gradients
# ---------------------------------------------------------------------------

def compute_D(p, n, M):
    """Return D = n * linear_self_convolution(p)."""
    P = np.fft.rfft(p, M)
    c = np.fft.irfft(P * P, M)
    c = c[:2 * n - 1]
    np.maximum(c, 0.0, out=c)
    return c * float(n)


def make_weights(D, mode, param):
    """
    Non-negative weights w over convolution lags.

    For 'pow':  w ~ (D / mean(D))**param.  This is the derivative of a
    super-linear flatness objective and approximates the max as param grows.
    For 'lse':  w = softmax(param * (D - mean(D))).
    """
    N = D.size
    if N == 0:
        return np.zeros(0, dtype=np.float64)
    if not np.all(np.isfinite(D)):
        return np.full(N, 1.0 / N)

    if mode == "pow":
        e = int(param)
        if e <= 1:
            w = D.copy()
        else:
            mean_d = D.mean()
            if not np.isfinite(mean_d) or mean_d <= 1e-300:
                return np.full(N, 1.0 / N)
            w = D / mean_d
            # The optimum is around max/mean ~= 1.5; clipping mainly protects
            # from poor restarts.  20**55 is still safely finite in float64.
            np.clip(w, 0.0, 20.0, out=w)
            w = np.power(w, e)
        s = w.sum()
        if not np.isfinite(s) or s <= 1e-300:
            return np.full(N, 1.0 / N)
        w /= s
        return w

    if mode == "lse":
        gamma = float(param)
        mean_d = D.mean()
        x = gamma * (D - mean_d)
        x -= np.max(x)
        w = np.exp(x)
        s = w.sum()
        if not np.isfinite(s) or s <= 1e-300:
            w = np.zeros(N, dtype=np.float64)
            w[int(np.argmax(D))] = 1.0
        else:
            w /= s
        return w

    return np.full(N, 1.0 / N)


def compute_grad(p, w, n, M):
    """
    Gradient of weighted-conv surrogate sum_k w_k * (p*p)_k.

    Using W and reversed p:
        grad_j = 2*n * conv(w, p[::-1])[j+n-1].
    """
    if p.size != w.size:
        w = np.full(p.size, 1.0 / p.size)
    pr = np.fft.rfft(p[::-1], M)
    wr = np.fft.rfft(w, M)
    corr = np.fft.irfft(wr * pr, M)
    grad = (2.0 * float(n)) * corr[n - 1:2 * n - 1]
    if not np.all(np.isfinite(grad)):
        grad = np.zeros_like(p)
    return grad


def internal_score(a):
    """
    Fast internal score matching evaluate_sequence for non-clipped data.
    """
    a = clean_sequence(a)
    if a.size == 0:
        return np.inf
    s = a.sum()
    if not np.isfinite(s) or s < 0.01:
        return np.inf
    n = a.size
    p = a / s
    M = next_pow2(2 * n - 1)
    D = compute_D(p, n, M)
    return 2.0 * float(np.max(D))


# ---------------------------------------------------------------------------
# Optimization primitives
# ---------------------------------------------------------------------------

def adam_update_p(p, m, v, b1p, b2p, grad, eta, max_step, floor,
                  b1=0.9, b2=0.999, eps=1e-8):
    """
    Multiplicative Adam-style update on a simplex vector.
    Only relative log changes matter, so the common mode is removed.
    """
    b1p = b1p * b1
    b2p = b2p * b2

    centered = grad - grad.mean()

    m *= b1
    m += (1.0 - b1) * centered

    v *= b2
    v += (1.0 - b2) * centered * centered

    step = -eta * m / ((1.0 - b1p) * (np.sqrt(v / (1.0 - b2p)) + eps))
    np.clip(step, -max_step, max_step, out=step)

    p *= np.exp(step)
    normalize_inplace(p, floor)

    return b1p, b2p


def smooth_noise(n, rng, freqs=8):
    """Low-frequency random field, useful for smooth asymmetric restarts."""
    x = rng.standard_normal(n)
    M = next_pow2(n)
    X = np.fft.rfft(x, M)
    keep = 2 + int(freqs)
    if keep < X.size:
        X[keep:] = 0.0
    y = np.fft.irfft(X, M)[:n]
    y = y - y.mean()
    return y

def resample_sequence(a, n):
    """Resample a positive sequence to length n and normalize."""
    a = clean_sequence(a)
    if a.size == 0:
        return None
    s = a.sum()
    if not np.isfinite(s) or s < 0.01:
        return None
    a = a / s
    m = a.size
    if m == n:
        return a.copy()
    if m == 1:
        return np.full(n, a[0], dtype=np.float64)
    x_old = np.arange(m, dtype=np.float64)
    x_new = np.linspace(0.0, m - 1.0, n, dtype=np.float64)
    a_new = np.interp(x_new, x_old, a)
    return normalize_p(a_new)


def init_uniform(n):
    return np.ones(n, dtype=np.float64)


def init_exp_ramp(n, alpha):
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    p = np.exp(alpha * x)
    return normalize_p(p)


def init_smooth_exp(n, rng, freqs, beta):
    z = smooth_noise(n, rng, freqs)
    p = np.exp(beta * z)
    return normalize_p(p)


def init_antisym_random(n, rng, freqs, beta):
    z = smooth_noise(n, rng, freqs)
    a = z - z[::-1]
    a = a - a.mean()
    p = np.exp(beta * a)
    return normalize_p(p)


def init_modulated(n, freq, amp, phase):
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    p = 1.0 + amp * np.cos(2.0 * np.pi * freq * x + phase)
    p = np.maximum(p, 1e-8)
    return normalize_p(p)


def init_beta(n, a, b):
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    x = np.clip(x, 1e-8, 1.0 - 1e-8)
    p = np.power(x, a - 1.0) * np.power(1.0 - x, b - 1.0)
    p = p + 1e-8
    return normalize_p(p)


def build_inits(n, rng, parent):
    inits = []

    if parent is not None:
        par = resample_sequence(parent, n)
        if par is not None:
            inits.append(par)

            mix = smooth_noise(n, rng, 8)
            mix = mix - mix.mean()
            par2 = par * np.exp(0.06 * mix)
            par2 = normalize_p(par2)
            inits.append(par2)

    inits.append(init_uniform(n))
    inits.append(init_exp_ramp(n, 1.5))
    inits.append(init_exp_ramp(n, -1.5))
    inits.append(init_exp_ramp(n, 0.7))
    inits.append(init_exp_ramp(n, -0.7))

    inits.append(init_smooth_exp(n, rng, 4, 0.55))
    inits.append(init_smooth_exp(n, rng, 10, 0.95))

    inits.append(init_antisym_random(n, rng, 5, 0.75))
    inits.append(init_antisym_random(n, rng, 12, 1.15))

    inits.append(init_modulated(n, 1, 0.80, 0.5 * np.pi))
    inits.append(init_modulated(n, 2, 0.50, 0.5 * np.pi))
    inits.append(init_modulated(n, 1, 0.60, 0.0))
    inits.append(init_modulated(n, 3, 0.30, 0.5 * np.pi))

    inits.append(init_beta(n, 1.5, 4.0))
    inits.append(init_beta(n, 4.0, 1.5))

    return inits

# Local optimizer
# ---------------------------------------------------------------------------

def optimize_sequence(p0, t_start, t_end, rng, plan, name, patience=1800):
    """
    Run the staged annealing optimizer from p0 until t_end.
    Returns (best_score, best_p).
    """
    p0 = clean_sequence(p0)
    if p0.size == 0:
        return np.inf, p0

    n = p0.size
    p = normalize_p(p0).copy()
    floor = 1e-15 / max(1, n)

    M2 = next_pow2(2 * n - 1)
    M3 = next_pow2(3 * n - 2)

    D = compute_D(p, n, M2)
    best_score = 2.0 * float(np.max(D))
    best_p = p.copy()

    total = max(0.05, float(t_end) - float(t_start))
    cum_frac = 0.0

    print("  [%s] start score = %.9f" % (name, best_score))

    for mode, param, eta, frac in plan:
        stage_end = float(t_start) + (cum_frac + float(frac)) * total
        cum_frac += float(frac)
        if stage_end > t_end:
            stage_end = float(t_end)

        if time.monotonic() >= stage_end:
            continue

        m = np.zeros(n, dtype=np.float64)
        v = np.zeros(n, dtype=np.float64)
        b1p = 1.0
        b2p = 1.0
        no_improve = 0
        local_patience = max(300, int(patience))
        it = 0
        max_step = 3.0

        while time.monotonic() < stage_end:
            it += 1

            D = compute_D(p, n, M2)
            score = 2.0 * float(np.max(D))

            if score < best_score - 1e-11:
                best_score = score
                best_p = p.copy()
                no_improve = 0
            else:
                no_improve += 1

            if no_improve > local_patience:
                restart_noise = smooth_noise(n, rng, 8)
                restart_noise = restart_noise - restart_noise.mean()
                scale = 0.04 + 0.14 * rng.random()
                p[:] = best_p * np.exp(scale * restart_noise)
                normalize_inplace(p, floor)
                m.fill(0.0)
                v.fill(0.0)
                b1p = 1.0
                b2p = 1.0
                no_improve = 0
                local_patience = max(250, local_patience // 2)
                continue

            w = make_weights(D, mode, param)
            grad = compute_grad(p, w, n, M3)

            if not np.all(np.isfinite(grad)):
                m.fill(0.0)
                v.fill(0.0)
                b1p = 1.0
                b2p = 1.0
                continue

            eta_use = eta * (0.5 if it < 120 else 1.0)
            b1p, b2p = adam_update_p(
                p, m, v, b1p, b2p, grad, eta_use, max_step, floor
            )

            if not np.all(np.isfinite(p)):
                p[:] = best_p
                m.fill(0.0)
                v.fill(0.0)
                b1p = 1.0
                b2p = 1.0

        D = compute_D(p, n, M2)
        cur = 2.0 * float(np.max(D))
        if cur > best_score * (1.0 + 1e-4) + 1e-12:
            p[:] = best_p

    print("  [%s] best score  = %.9f  time=%.1fs" % (
        name, best_score, time.monotonic() - t_start
    ))
    return best_score, best_p

def choose_main_n(parent):
    if parent is not None:
        m = parent.size
        if m > 0:
            if 1024 <= m <= 12000:
                return int(m)
            if m < 1024:
                return 2048
    return 8192


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for a sequence minimizing 2*n*max(conv(a,a)) / (sum(a))^2.
    """
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    budget = max(60.0, float(budget_s))
    work = max(1.0, budget - 25.0)
    deadline = start + budget - 12.0

    parent_raw = None
    try:
        hs = globals().get("height_sequence_1", None)
        if hs is not None:
            arr = np.asarray(hs, dtype=np.float64).ravel()
            if arr.size > 0:
                arr = clean_sequence(arr)
                if arr.size > 0 and arr.sum() > 0.01:
                    parent_raw = arr
    except Exception:
        parent_raw = None

    main_n = choose_main_n(parent_raw)

    # Keep the known previous construction as a hard fallback, if reasonably
    # sized for direct evaluation.
    global_best_score = np.inf
    global_best_a = None

    if parent_raw is not None and parent_raw.size <= 30000:
        ps = internal_score(parent_raw)
        if np.isfinite(ps):
            global_best_score = float(ps)
            global_best_a = parent_raw.copy()

    # Main-space incumbent, always length main_n.
    main_score = np.inf
    main_a = None

    def remember_main(sc, aa):
        nonlocal_return = None  # placeholder to keep style simple
        return None

    # -----------------------------------------------------------------
    # Phase 1: many short asymmetric explorations at main resolution.
    # -----------------------------------------------------------------
    explore_time = min(300.0, 0.10 * work)
    primary_time = 0.72 * work
    polish_time = 0.18 * work
    alloc = explore_time + primary_time + polish_time
    if alloc > work:
        scale = work / alloc
        explore_time *= scale
        primary_time *= scale
        polish_time *= scale

    t = start

    if explore_time > 4.0 and t < deadline:
        inits = build_inits(main_n, rng, parent_raw)
        if not inits:
            inits = [np.ones(main_n, dtype=np.float64)]

        # If the parent was already evaluated, make sure its resampling is
        # among the first candidates.
        n_inits = len(inits)
        time_each = explore_time / float(n_inits)
        exp_end = min(t + explore_time, deadline)

        for i, p0 in enumerate(inits):
            if time.monotonic() >= exp_end:
                break
            t0 = time.monotonic()
            t1 = min(t0 + time_each, exp_end, deadline)
            if t1 <= t0 + 0.2:
                break
            sc, aa = optimize_sequence(p0, t0, t1, rng, QUICK_PLAN,
                                       "explo\n".replace("\n", "%d") % i)
            aa = normalize_p(aa)
            aa_sc = internal_score(aa)
            if not np.isfinite(aa_sc):
                aa_sc = sc
            if aa_sc < main_score:
                main_score = aa_sc
                main_a = aa.copy()
            if aa_sc < global_best_score:
                global_best_score = aa_sc
                global_best_a = aa.copy()
            t = time.monotonic()

        print("after explore: main=%.9f global=%.9f time=%.1fs" % (
            main_score, global_best_score, time.monotonic() - start
        ))

    # Ensure a valid main-space start.
    if main_a is None or main_a.size != main_n or not np.isfinite(main_score):
        if parent_raw is not None:
            par = resample_sequence(parent_raw, main_n)
            if par is not None:
                main_a = par
                main_score = internal_score(main_a)
        if main_a is None or not np.isfinite(main_score):
            main_a = init_uniform(main_n)
            main_score = internal_score(main_a)
        if main_score < global_best_score:
            global_best_score = main_score
            global_best_a = main_a.copy()

    # -----------------------------------------------------------------
    # Phase 2: long primary refinement.
    # -----------------------------------------------------------------
    if primary_time > 4.0 and t < deadline:
        t0 = time.monotonic()
        t1 = min(t0 + primary_time, deadline)
        sc, aa = optimize_sequence(main_a, t0, t1, rng, MAIN_PLAN, "primary")
        aa = normalize_p(aa)
        aa_sc = internal_score(aa)
        if not np.isfinite(aa_sc):
            aa_sc = sc
        if aa_sc < main_score:
            main_score = aa_sc
            main_a = aa.copy()
        if main_score < global_best_score:
            global_best_score = main_score
            global_best_a = main_a.copy()
        t = time.monotonic()

        print("after primary: main=%.9f global=%.9f time=%.1fs" % (
            main_score, global_best_score, t - start
        ))

    # -----------------------------------------------------------------
    # Phase 3: high-power / log-sum-exp polish.
    # -----------------------------------------------------------------
    if polish_time > 4.0 and t < deadline:
        t0 = time.monotonic()
        t1 = min(t0 + polish_time, deadline)
        sc, aa = optimize_sequence(main_a, t0, t1, rng, POLISH_PLAN, "polish")
        aa = normalize_p(aa)
        aa_sc = internal_score(aa)
        if not np.isfinite(aa_sc):
            aa_sc = sc
        if aa_sc < main_score:
            main_score = aa_sc
            main_a = aa.copy()
        if main_score < global_best_score:
            global_best_score = main_score
            global_best_a = main_a.copy()
        t = time.monotonic()

        print("after polish: main=%.9f global=%.9f time=%.1fs" % (
            main_score, global_best_score, t - start
        ))

    # Fallback if everything somehow failed.
    if global_best_a is None or not np.isfinite(global_best_score):
        global_best_a = np.ones(main_n, dtype=np.float64)
        global_best_score = internal_score(global_best_a)

    # Make sure the returned sequence is a Python list of ordinary floats.
    final_a = clean_sequence(global_best_a)
    if final_a.size == 0 or final_a.sum() < 0.01:
        final_a = np.ones(1024, dtype=np.float64)

    out = [float(x) for x in final_a]

    verified = np.inf
    try:
        verified = float(evaluate_sequence(out))  # noqa: F821
    except Exception:
        verified = internal_score(out)

    print("final: len=%d internal=%.9f verified=%.9f time=%.1fs" % (
        len(out), global_best_score, verified, time.monotonic() - start
    ))

    return out
```
