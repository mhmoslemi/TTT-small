import math
import time

def normalize(a):
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s <= 0:
        return np.ones_like(a) / len(a)
    return a / s

def conv_self(a, L, m):
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c

def exact_grad_peakeq(a, k, n):
    """Gradient of conv(a,a)[k] wrt a."""
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo > hi:
        return g
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * a[k - idx]
    return g

def topk_grad(a, c, n, K, decay):
    """Compute weighted gradient from top-K peaks."""
    # Get top-K indices
    top_idx = np.argsort(c)[::-1][:K]
    top_vals = c[top_idx]
    # Exponential decay weights
    if decay > 0:
        weights = np.exp(-decay * (top_vals - top_vals[0]))
    else:
        weights = np.ones(K)
    weights /= weights.sum()
    
    g = np.zeros(n)
    for k, w in zip(top_idx, weights):
        g += w * exact_grad_peakeq(a, k, n)
    return g

    a = normalize(a0.astype(np.float64).copy())
    best_a = a.copy()
    best_score = 2.0 * n * np.max(conv_self(a, L, m))
    
    y = a.copy()  # momentum

    while time.monotonic() < deadline:
        c = conv_self(y, L, m)
        g = topk_grad(y, c, n, topK, decay)

        gmax = np.max(g)
        if gmax <= 0:
            break
        lr = alpha / gmax

        a_new = normalize(a_new)

            best_a = a_new.copy()

        y = (1.0 + momentum) * a - momentum * y
        y = np.maximum(y, 0.0)

        momentum = min(0.9, momentum * 1.001)

  - Final exact evaluation

import numpy as np
import math
import time


def _normalize(a):
    a = np.maximum(np.asarray(a, dtype=np.float64), 0.0)
    s = float(a.sum())
    if s <= 0 or not np.isfinite(s):
        return np.ones_like(a) / a.size
    return a / s


def _fft_len(n):
    m = 2 * n - 1
    return 1 << (m - 1).bit_length()


def _conv_self(a, L, m):
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _score(a, L, m, n):
    c = _conv_self(a, L, m)
    return float(2.0 * n * np.max(c))


def _grad_peakeq(a, k, n):
    g = np.zeros(n)
    lo = max(0, k - n + 1)
    hi = min(n - 1, k)
    if lo > hi:
        return g
    idx = np.arange(lo, hi + 1)
    g[idx] = 2.0 * a[k - idx]
    return g


def _topk_gradient(a, c, n, K, decay):
    if K >= len(c):
        order = np.argsort(c)[::-1]
        vals = c[order]
        if decay > 0 and vals[0] > 0:
            w = np.exp(-decay * (vals - vals[0]))
        else:
            w = np.ones(len(order))
        w /= w.sum()
        g = np.zeros(n)
        for idx, wi in zip(order, w):
            g += wi * _grad_peakeq(a, idx, n)
        return g
    top_idx = np.argsort(c)[::-1][:K]
    top_vals = c[top_idx]
    if decay > 0 and top_vals[0] > 0:
        w = np.exp(-decay * (top_vals - top_vals[0]))
    else:
        w = np.ones(K)
    w /= w.sum()
    g = np.zeros(n)
    for idx, wi in zip(top_idx, w):
        g += wi * _grad_peakeq(a, idx, n)
    return g

            stale = 0
            lr = min(3.0, lr * 1.05)
        else:
            stale += 1
            if stale % 20 == 0:
                lr = max(0.1, lr * 0.85)
            if stale > 200:
                break

        a = a_new
        momentum = min(momentum_max, momentum * 1.002)
        y = (1.0 + momentum) * a - momentum * y
        np.maximum(y, 0.0, out=y)

        n_it += 1
        if n_it % 500 == 0:
            elapsed = time.monotonic() - t_start
            print(f"  {name} it={n_it} score={best_score:.6f} lr={lr:.3f} "
                  f"t={elapsed:.1f}s")

    return best_a, best_score


def _mws_search(a0, L, m, n, time_limit):
    deadline = time.monotonic() + time_limit
    best_a = _normalize(a0.copy())
    best_score = _score(best_a, L, m, n)
    rng = np.random.default_rng(12345)

    F = np.fft.rfft(best_a, L)
    Frev = np.fft.rfft(best_a[::-1], L)

    while time.monotonic() < deadline:
        prior = _normalize(np.ones(n) + rng.random(n) * 0.5)
        inner = 30
        x = prior.copy()
        y = np.full(2 * n - 1, 1.0 / (2 * n - 1))
        avg = np.zeros(n)
        cnt = 0

        for it in range(inner):
            frac = it / max(1, inner - 1)
            eta = 5.0 + 30.0 * frac

            X = np.fft.rfft(x, L)
            u = np.fft.irfft(F * X, L)[:2 * n - 1]
            np.maximum(u, 0.0, out=u)
            vu = n * u
            vma = float(np.max(vu))
            z = eta * (vu - vma)
            np.clip(z, -40.0, 0.0, out=z)
            y = np.exp(z)
            ys = float(y.sum())
            if ys > 0 and np.isfinite(ys):
                y /= ys
            else:
                y.fill(1.0 / (2 * n - 1))

            avg += x
            cnt += 1

            Y = np.fft.rfft(y, L)
            s = np.fft.irfft(Y * Frev, L)[(n - 1):(2 * n - 1)]
            np.maximum(s, 0.0, out=s)
            ps = n * s
            sma = float(np.max(ps))
            lp = np.log(np.maximum(prior, 1e-300))
            zx = -eta * (ps - sma) + 0.5 * lp
            zx -= float(np.max(zx))
            np.clip(zx, -40.0, 0.0, out=zx)
            x = np.exp(zx)
            xs = float(x.sum())
            if xs > 0 and np.isfinite(xs):
                x /= xs
            else:
                x = prior.copy()

        x = _normalize(0.5 * (avg / cnt) + 0.5 * x)
        # Line search mixing best_a with x
        for lam in (0.1, 0.2, 0.3, 0.5, 0.7, 1.0):
            cand = _normalize((1 - lam) * best_a + lam * x)
            sc = _score(cand, L, m, n)
            if sc < best_score:
                best_score = sc
                best_a = cand
                F = np.fft.rfft(best_a, L)
                Frev = np.fft.rfft(best_a[::-1], L)
                break

    return best_a, best_score


def _random_init(n, rng):
    t = np.linspace(0, 1, n)
    x = 2 * t - 1
    z = rng.normal(0, 0.4)
    for k in range(1, 6):
        amp = rng.uniform(0.1, 0.5) / math.sqrt(k)
        ph = rng.uniform(0, 2 * np.pi)
        z += amp * np.cos(2 * np.pi * k * t + ph)
    z += rng.normal(0, 0.2, n)
    p = np.exp(z - np.max(z))
    return _normalize(p)


def _exact_score(seq):
    try:
        val = evaluate_sequence(seq)
    except NameError:
        a = [max(0.0, min(1000.0, float(x))) for x in seq]
        n = len(a)
        s = sum(a)
        if s < 0.01:
            return float('inf')
        b = np.convolve(a, a)
        return float(2.0 * n * max(b) / (s * s))
    except Exception:
        return float('inf')
    return float(val) if val is not None else float('inf')


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    start = time.monotonic()
    total_budget = max(10.0, float(budget_s))
    hard_deadline = start + total_budget - 3.0

    # Get parent sequence
    parent = None
    try:
        arr = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if arr.size > 100 and np.all(np.isfinite(arr)):
            parent = arr
    except (NameError, Exception):
        pass

    best_seq = None
    best_exact = float('inf')

    # Try multiple n values
    n_candidates = [6121]
    if parent is not None and parent.size == 6121:
        pass  # keep 6121 first
    else:
        n_candidates = [4096, 3072, 2048, 6121]

    # Allocate time: 60% to primary n, 40% to others
    time_per_n = {}
    primary_n = n_candidates[0]
    time_per_n[primary_n] = 0.55 * (total_budget - 30)
    remaining_time = 0.35 * (total_budget - 30)
    others = n_candidates[1:]
    if others:
        per_other = remaining_time / len(others)
        for nn in others:
            time_per_n[nn] = per_other

    print(f"Budget: {total_budget:.0f}s, Primary n={primary_n}, "
          f"Others: {others}")
    print(f"Parent available: {parent is not None}, len={parent.size if parent is not None else 0}")

    for n in n_candidates:
        t_rem = hard_deadline - time.monotonic()
        if t_rem < 30:
            break
        alloc = min(time_per_n.get(n, 100.0), t_rem - 30.0)
        if alloc < 15:
            continue

        L = _fft_len(n)
        m = 2 * n - 1
        print(f"\n=== n={n}, L={L}, time_alloc={alloc:.0f}s ===")

        # Build initial candidates
        inits = []
        if parent is not None and parent.size == n:
            inits.append(_normalize(parent.copy()))
            print(f"  Initializing from parent")

        # Add a few random inits
        rng = np.random.default_rng(seed + n)
        for _ in range(3):
            inits.append(_random_init(n, rng))

        # Evaluate initial scores
        best_a = None
        best_score = float('inf')
        for a in inits:
            sc = _score(a, L, m, n)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
            print(f"  Init score: {sc:.6f}")

        if best_a is None:
            best_a = _normalize(np.ones(n))
            best_score = _score(best_a, L, m, n)

        # Phase 1: Mirror descent with K=1 (focus on single worst peak)
        t1 = min(alloc * 0.25, 300.0)
        if t1 > 10:
            print(f"  Phase 1: Mirror descent K=1, t={t1:.0f}s")
            tmp, sc = _mirror_descent(best_a, L, m, n, t1,
                                      topK=1, decay=0.0, alpha=1.0,
                                      momentum0=0.2, momentum_max=0.5,
                                      name="K=1")
            if sc < best_score:
                best_score = sc
                best_a = tmp
                print(f"    Improved to {best_score:.6f}")

        # Phase 2: Mirror descent with K=3
        t2 = min(alloc * 0.30, 400.0)
        if t2 > 10:
            print(f"  Phase 2: Mirror descent K=3, t={t2:.0f}s")
            tmp, sc = _mirror_descent(best_a, L, m, n, t2,
                                      topK=3, decay=3.0, alpha=0.8,
                                      momentum0=0.3, momentum_max=0.6,
                                      name="K=3")
            if sc < best_score:
                best_score = sc
                best_a = tmp
                print(f"    Improved to {best_score:.6f}")

        # Phase 3: Mirror descent with K=5
        t3 = min(alloc * 0.20, 300.0)
        if t3 > 10:
            print(f"  Phase 3: Mirror descent K=5, t={t3:.0f}s")
            tmp, sc = _mirror_descent(best_a, L, m, n, t3,
                                      topK=5, decay=2.0, alpha=0.6,
                                      momentum0=0.4, momentum_max=0.7,
                                      name="K=5")
            if sc < best_score:
                best_score = sc
                best_a = tmp
                print(f"    Improved to {best_score:.6f}")

        # Phase 4: MWS search
        t4 = min(alloc * 0.15, 200.0)
        if t4 > 10:
            print(f"  Phase 4: MWS search, t={t4:.0f}s")
            tmp, sc = _mws_search(best_a, L, m, n, t4)
            if sc < best_score:
                best_score = sc
                best_a = tmp
                print(f"    Improved to {best_score:.6f}")

        # Phase 5: Fine-tuning with K=1, small step
        t5 = min(alloc * 0.10, 150.0)
        if t5 > 5:
            print(f"  Phase 5: Fine-tune K=1, t={t5:.0f}s")
            tmp, sc = _mirror_descent(best_a, L, m, n, t5,
                                      topK=1, decay=0.0, alpha=0.3,
                                      momentum0=0.1, momentum_max=0.3,
                                      name="finetune")
            if sc < best_score:
                best_score = sc
                best_a = tmp
                print(f"    Improved to {best_score:.6f}")

        # Check exact score
        seq = [float(1.0 / n) * float(x) for x in best_a]
        # Scale to have sum >= 0.01 (already is since sum=1)
        exact = _exact_score(seq)
        print(f"  n={n}: internal={best_score:.6f}, exact={exact:.6f}")
        if np.isfinite(exact) and exact < best_exact:
            best_exact = exact
            best_seq = seq

    # Final: try to improve the best with a short direct search
    t_rem = hard_deadline - time.monotonic()
    if t_rem > 30 and best_seq is not None:
        print(f"\nFinal polish: {t_rem:.0f}s remaining")
        n_best = len(best_seq)
        L = _fft_len(n_best)
        m = 2 * n_best - 1
        a = np.asarray(best_seq, dtype=np.float64)
        a = _normalize(a)

        # Quick mirror descent
        tmp, sc = _mirror_descent(a, L, m, n_best, min(t_rem - 5, 60),
                                  topK=1, decay=0.0, alpha=0.3,
                                  momentum0=0.1, momentum_max=0.2,
                                  name="final")
        seq = [float(x) for x in tmp]
        exact = _exact_score(seq)
        print(f"  Final: exact={exact:.6f}")
        if np.isfinite(exact) and exact < best_exact:
            best_exact = exact
            best_seq = seq

    if best_seq is None:
        n = primary_n
        best_seq = [1.0] * n

    print(f"\n=== FINAL RESULT: {best_exact:.6f} ===")
    print(f"Time used: {time.monotonic() - start:.1f}s")
    return best_seq
