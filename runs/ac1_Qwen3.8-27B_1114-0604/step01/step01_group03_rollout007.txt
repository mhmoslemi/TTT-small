import numpy as np

    if s < 1e-300:
        a.fill(1.0 / len(a))

    """Compute 2*n*max(a*a)/(sum(a))^2 using FFT."""
    n = len(a)

    L = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:2 * n - 1]
    c = np.maximum(c, 0)
    return 2.0 * n * float(c.max()) / (s * s)

    B = np.fft.rfft(b, L)
    c = np.fft.irfft(A * B, L)[:na + nb - 1]
    c = np.maximum(c, 0)
    return c
