def coordinate_descent_step(a, c, j, M_inact):
    """Find optimal a[j] given all other coordinates fixed.
    
    c is the current full convolution (length 2n-1).
    M_inact is the max of c[k] for k not affected by a[j].
    """
    n = len(a)
    # Active k values: k in [j, j+n-1]
    k_start = j
    k_end = j + n - 1
    
    # For each active k, c_k = base_k + 2*t*a[k-j] (or +t^2 if k==2*j)
    # base_k = c[k] - 2*a[j]*a[k-j] (or c[k] - a[j]^2 if k==2*j)
    
    t_old = a[j]
    
    # Compute base values for active k's
    # We need to subtract the contribution of a[j] from c
    base = np.zeros(k_end - k_start + 1)
    for idx, k in enumerate(range(k_start, k_end + 1)):
        other = k - j  # a[k-j]
        if k == 2 * j:
            base[idx] = c[k] - t_old * t_old
        else:
            base[idx] = c[k] - 2 * t_old * a[other]
    
    coefs = np.zeros(k_end - k_start + 1)
    for idx, k in enumerate(range(k_start, k_end + 1)):
        other = k - j
        if k == 2 * j:
            coefs[idx] = 0  # will handle t^2 separately
        else:
            coefs[idx] = 2 * a[other]
    
    # c_k(t) = base[idx] + coefs[idx]*t + (t^2 if k==2*j else 0)
    
    # Objective: h(t) = max(M_inact, max_i (base[i] + coefs[i]*t + (t^2 if i corresponds to k=2j else 0)))
    
    # Find optimal t >= 0
    # Use scipy minimize_scalar
    from scipy.optimize import minimize_scalar
    
    def h(t):
        vals = base + coefs * t
        # Add t^2 at the position where k = 2j
        idx_2j = 2 * j - k_start
        if 0 <= idx_2j < len(vals):
            vals[idx_2j] += t * t
        return max(M_inact, np.max(vals))
    
    result = minimize_scalar(h, bounds=(0, 2*t_old + 1), method='bounded', 
                              options={'xatol': 1e-10})
    return result.x

    return np.fft.irfft(A * A, M_fft)[:M]

"""
Coordinate descent optimizer for minimizing f(a) = 2n * max(conv(a,a)) / (sum a)^2.

Key insights:
- The objective is a ratio, so each coordinate update is a 1D unimodal optimization.
- Coordinate descent with exact 1D minimization per coordinate is the main workhorse.
- Periodic perturbations escape local minima.
- Multiple sequence lengths are explored.
- A brief gradient warm-up (LSE surrogate) seeds the coordinate descent.
"""

import numpy as np
import time
from scipy.optimize import minimize_scalar


def fast_self_conv(a):
    """Linear self-convolution via FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def compute_score(n, c, S):
    """Compute the objective value."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def lse_grad_step(a, lr, temp, topk=16):
    """One gradient step using log-sum-exp surrogate for max."""
    n = len(a)
    c = fast_self_conv(a)
    S = a.sum()
    if S < 1e-10:
        return a, np.inf
    # LSE approximation of max
    c_sorted = np.sort(c)[::-1]
    k = min(topk, len(c))
    top_vals = c_sorted[:k]
    lse_sum = np.log(np.sum(np.exp((top_vals - top_vals[0]) / temp))) + top_vals[0] / temp
    # Weighted gradient
    weights = np.exp((top_vals - top_vals[0]) / temp)
    weights = weights / weights.sum()
    top_idx = np.argsort(c)[::-1][:k]
    grad_c = np.zeros(n)
    for i in range(k):
        ki = int(top_idx[i])
        wi = float(weights[i])
        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start <= j_end:
            grad_c[j_start:j_end + 1] += wi * 2.0 * a[j_start:j_end + 1][::-1]
    M_lse = float(lse_sum * temp)
    S2 = S * S
    grad = 2.0 * n * (grad_c / S2 - 2.0 * M_lse / (S2 * S))
    a = a - lr * grad
    a = np.clip(a, 0.0, 1000.0)
    score = compute_score(n, c, S)
    return a, score


def grad_warmup(a, time_limit, rng, init_lr=0.5, init_temp=0.05):
    """Brief gradient-based warmup with adaptive learning rate."""
    n = len(a)
    a = a.copy()
    best_score = np.inf
    best_a = a.copy()

    temp = init_temp
    c = fast_self_conv(a)
    S = a.sum()
    score = compute_score(n, c, S)
    best_score = score
    best_a = a.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_improve = 0
    while time.monotonic() < time_limit:
        a, score_ = lse_grad_step(a, lr, temp)
        c = fast_self_conv(a)
        S = a.sum()
        score = compute_score(n, c, S)
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()

            no_improve = 0

    return best_score, best_a


def coord_optimize_obj(c, a, S, j, n, M_inact):
    """Build the objective function for coordinate j (closure-free via precomputed arrays)."""
    t_old = a[j]
    k_lo = j
    k_hi = min(j + n - 1, 2 * n - 2)
    active_len = k_hi - k_lo + 1

    k_indices = np.arange(k_lo, k_hi + 1)
    other_indices = k_indices - j

    base = c[k_lo:k_hi + 1].copy()
    base -= 2.0 * t_old * a[other_indices]
    coeff = 2.0 * a[other_indices].copy()

    idx_2j = 2 * j
    has_quad = (k_lo <= idx_2j <= k_hi)
    quad_pos = idx_2j - k_lo if has_quad else -1

    S_old = S
    a_j_val = t_old

    return base, coeff, has_quad, quad_pos, M_inact, S_old, a_j_val


def eval_coord_obj(base, coeff, has_quad, quad_pos, M_inact, S_old, a_j_val, n, t):
    """Evaluate the objective for coordinate update."""
    S_new = S_old - a_j_val + t
    if S_new < 1e-10:
        return np.inf
    vals = base + coeff * t
    if has_quad:
        vals = vals.copy()
        vals[quad_pos] += (t - a_j_val) ** 2
    M_new = max(M_inact, float(np.max(vals)))
    return 2.0 * n * M_new / (S_new * S_new)


def update_conv(c, a, j, n, t_new, t_old, base, coeff, has_quad, quad_pos):
    """Update convolution in-place after changing a[j]."""
    k_lo = j
    k_hi = min(j + n - 1, 2 * n - 2)
    a[j] = t_new
    c[k_lo:k_hi + 1] = base + coeff * t_new
    if has_quad:
        c[k_lo + quad_pos] += (t_new - t_old) ** 2


def run_coord_descent(a0, time_limit, rng, max_restarts=50):
    """Main coordinate descent loop with periodic restarts."""
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0.0, 1000.0).copy()
    c = fast_self_conv(a)
    S = float(a.sum())
    score = compute_score(n, c, S)
    best_score = score
    best_a = a.copy()
    best_c = c.copy()
    best_S = S

    no_improve_passes = 0
    patience = 15
    restart_count = 0

    while time.monotonic() < time_limit and restart_count < max_restarts:
        # One full pass of coordinate descent
        order = rng.permutation(n)

        for j in order:
            if time.monotonic() > time_limit:
                break

            t_old = a[j]
            k_lo = j
            k_hi = min(j + n - 1, 2 * n - 2)

            # Compute M_inact (max over inactive positions)
            M_inact = -np.inf
            if k_lo > 0:
                M_inact = max(M_inact, float(c[:k_lo].max()))
            if k_hi < 2 * n - 2:
                M_inact = max(M_inact, float(c[k_hi + 1:].max()))
            if M_inact == -np.inf:
                M_inact = 0.0

            # Build objective components
            params = coord_optimize_obj(c, a, S, j, n, M_inact)
            base, coeff, has_quad, quad_pos, M_inact_p, S_old, a_j_val = params

            # Quick check: if t_old is very small, try t=0
            if t_old < 1e-12:

            # Determine search bounds
            t_upper = max(3.0 * t_old, 1.5 * S / n, 0.1)

            # 1D minimization
            result = minimize_scalar(
                lambda t: eval_coord_obj(base, coeff, has_quad, quad_pos, M_inact_p, S_old, a_j_val, n, t),
                bounds=(0.0, t_upper),
                method='bounded',
                options={'xatol': 1e-14}

            t_new = float(result.x)

            # Check if there's improvement

                S = S - t_old + t_new

        # After full pass, check improvement
        score = compute_score(n, c, S)
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
            best_c = c.copy()
            best_S = S
            no_improve_passes = 0
        else:
            no_improve_passes += 1

        if no_improve_passes >= patience:
            # Restart: perturb from best
            restart_count += 1
            scale = 0.03 + 0.1 * rng.random()
            a = best_a * (1.0 + scale * rng.standard_normal(n))
            a = np.clip(a, 0.0, 1000.0)
            # Also try re-adding some zero coordinates

                    a = np.clip(a, 0.0, 1000.0)
            c = fast_self_conv(a)
            S = float(a.sum())
            no_improve_passes = 0
            patience = max(5, patience // 2)

    # Allocate time proportionally (give more time to parent length if available)
    time_fractions = []
    for n in lengths:
        if parent is not None and n == parent.size:
            time_fractions.append(0.5)
        elif n == 4096:
            time_fractions.append(0.2)
        elif n == 8192:
            time_fractions.append(0.15)
        else:
            time_fractions.append(0.1)
    # Normalize
    tf_sum = sum(time_fractions)
    time_fractions = [f / tf_sum for f in time_fractions]

    time_offset = start
    for i, n in enumerate(lengths):
        t_allocation = time_fractions[i] * (deadline - start)
        t_end_this = time_offset + t_allocation - 3.0
        if t_end_this <= time.monotonic():
            continue

        print(f"Trying n={n}, time budget={t_allocation:.0f}s")

        # Phase A: Gradient warmup (first 20% of allocation)
        t_warm = time_offset + 0.2 * t_allocation
        inits = init_diverse(n, rng, parent)

        best_local_score = np.inf
        best_local_a = None

        time_per_init = max(1.0, (t_warm - time.monotonic()) / len(inits))
        for k, a0 in enumerate(inits):
            a0 = np.clip(a0, 0, 1000)
            if a0.sum() < 0.01:
                a0 = np.ones(n) * 0.1
            t_this_min = min(t_warm, time.monotonic() + time_per_init)
            if time.monotonic() >= t_this_min:
                break
            sc, aa = grad_warmup(a0, t_this_min, rng)
            if sc < best_local_score:
                best_local_score = sc
                best_local_a = aa.copy()

        if best_local_a is None:
            best_local_a = np.ones(n) * 0.5
            best_local_score = compute_score(n, fast_self_conv(best_local_a), best_local_a.sum())

        # Phase B: Coordinate descent (remaining 80%)
        t_cd_end = min(t_end_this, deadline - 10.0)
        sc_cd, aa_cd = run_coord_descent(best_local_a, t_cd_end, rng)
        if sc_cd < best_local_score:
            best_local_score = sc_cd
            best_local_a = aa_cd

        # Update global best
        if best_local_score < best_score:
            best_score = best_local_score
            best_seq = best_local_a.copy()
            print(f"  n={n}: score={best_score:.6f} (improved!)")
        else:
            print(f"  n={n}: score={best_local_score:.6f} (no improvement)")

        time_offset = t_end_this + 3.0

    # Final polish on the best sequence found
    if best_seq is not None and time.monotonic() < deadline - 5:
        n_best = best_seq.size
        sc_final, aa_final = run_coord_descent(best_seq, deadline - 5, rng, max_restarts=100)
        if sc_final < best_score:
            best_score = sc_final
            best_seq = aa_final
            print(f"Final polish: {best_score:.6f}")

    if best_seq is None:
        best_seq = np.ones(2048)

    out = [float(x) for x in best_seq]
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - start
    print(f"DONE: internal={best_score:.6f}, verified={verified:.6f}, len={len(out)}, time={elapsed:.1f}s")
    return out
