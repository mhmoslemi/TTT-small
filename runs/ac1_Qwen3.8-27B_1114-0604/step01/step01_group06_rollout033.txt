Algorithm: Log-space gradient descent with adaptive smoothness (logsumexp)
and exact active-set polishing to minimize 2*n*max(conv(a,a))/sum(a)^2.

Key improvements over previous:

2. Adaptive tau schedule (coarse to fine)
3. Aggressive exact active-set final polish
4. Multi-restart with structured perturbations
5. Start from provided best sequence

def _set_global(n):
    global _G_N, _G_NFFT
    _G_N = int(n)
    _G_NFFT = 1 << (3 * _G_N - 2).bit_length()


def _fft_conv(a):
    """Compute linear autocorrelation (a*a) using FFT."""
    N = _G_NFFT
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N).real[:2 * _G_N - 1]
    return c

    """Score assuming sum(a)=1: 2*n*max(a*a)."""
    c = _fft_conv(a)
    return 2.0 * _G_N * float(np.max(c))

    n = _G_N
    N = _G_NFFT

    z = np.asarray(z, dtype=np.float64).ravel()
    z = z - np.max(z)
    e = np.exp(z)
    s = np.sum(e)
    if s <= 0 or not np.isfinite(s):
        return 1e10, np.zeros(n)
    a = e / s

    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N).real[:2 * n - 1]
    c = np.maximum(c, 0.0)

    m = float(np.max(c))
    diff = np.clip((c - m) / tau, -700.0, 0.0)
    exps = np.exp(diff)
    se = float(np.sum(exps))
    if se <= 0 or not np.isfinite(se):
        return 1e10, np.zeros(n)
    p = exps / se

    # Compute h_i = sum_k p_k * (a_{k-i} + a_{i-k})
    # Term 1: (p*a)_i = sum_k p_k * a_{i-k}
    fp = np.fft.rfft(p, N)
    pa = np.fft.irfft(fp * fa, N).real[:n]

    # Term 2: sum_k p_k * a_{k-i} = (a * p_rev)_{n-1-i} where p_rev[k]=p[n-1-k]
    p_rev = p[::-1].copy()
    fp_rev = np.fft.rfft(p_rev, N)
    term2 = np.fft.irfft(fa * fp_rev, N).real[:n]
    # h_i = pa[i] + term2[n-1-i]
    h = pa + term2[::-1]

    pc = float(np.dot(p, c))
    grad = a * (h - 2.0 * pc)

    val = m + tau * np.log(se)
    return val, grad


def _exact_grad(a):
    """
    Compute an exact subgradient of max_k c_k w.r.t. a.
    Returns gradient vector g where g_i = dc_{k*}/da_i for k* in argmax.
    (Using uniform weighting over the exact argmax set for stability.)
    """
    n = _G_N
    N = _G_NFFT
    c = _fft_conv(a)
    m = float(np.max(c))
    # Soft argmax with very small temperature to avoid numerical issues
    tau = 1e-10
    diff = np.clip((c - m) / tau, -700.0, 0.0)
    exps = np.exp(diff)
    se = float(np.sum(exps))
    if se <= 0:
        return np.zeros(n)
    p = exps / se

    fa = np.fft.rfft(a, N)
    fp = np.fft.rfft(p, N)
    pa = np.fft.irfft(fp * fa, N).real[:n]
    p_rev = p[::-1].copy()
    fp_rev = np.fft.rfft(p_rev, N)
    term2 = np.fft.irfft(fa * fp_rev, N).real[:n]
    h = pa + term2[::-1]

    pc = float(np.dot(p, c))
    grad = a * (h - 2.0 * pc)
    return grad


def _softmax(z):
    z = z - np.max(z)
    e = np.exp(z)
    s = np.sum(e)
    if s <= 0:
        return np.ones_like(z) / len(z)
    return e / s


def _try_polish(a, deadline, max_iters=50000):
    """
    Exact active-set polishing in log-space.
    Uses the true max (via very small tau) for the gradient.
    """
    n = _G_N
    N = _G_NFFT
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.maximum(a, 1e-15)
    a = a / np.sum(a)

    best_a = a.copy()
    best_F = float(np.max(_fft_conv(a)))

    z = np.log(a)
    z = z - np.mean(z)

    rng = np.random.default_rng(12345)
    stall = 0

    for it in range(max_iters):
        if time.monotonic() > deadline - 5.0:
            break

        c = _fft_conv(a)
        F = float(np.max(c))

        if F < best_F - 1e-16:
            best_F = F
            best_a = a.copy()
            stall = 0
        else:
            stall += 1

        if stall > 500:
            # Random escape
            z = z + 0.05 * rng.standard_normal(n)
            z = z - np.mean(z)
            z = np.clip(z, -25.0, 25.0)
            a = _softmax(z)
            stall = 0
            continue

        # Compute smoothed gradient with tiny tau (near-exact)
        tau = max(1e-9, best_F * 1e-6)
        diff = np.clip((c - F) / tau, -700.0, 0.0)
        exps = np.exp(diff)
        se = float(np.sum(exps))
        if se <= 0:
            break
        p = exps / se

        fa = np.fft.rfft(a, N)
        fp = np.fft.rfft(p, N)
        pa = np.fft.irfft(fp * fa, N).real[:n]
        p_rev = p[::-1].copy()
        fp_rev = np.fft.rfft(p_rev, N)
        term2 = np.fft.irfft(fa * fp_rev, N).real[:n]
        h = pa + term2[::-1]
        pc = float(np.dot(p, c))
        grad = a * (h - 2.0 * pc)

        # Project gradient to remove mean component (gauge invariance)
        grad = grad - np.mean(grad)

        # Backtracking line search
        accepted = False
        for eta in (1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 0.1, 0.3, 1.0, 3.0):
            z_new = z - eta * grad
            z_new = z_new - np.mean(z_new)
            z_new = np.clip(z_new, -25.0, 25.0)
            a_new = _softmax(z_new)
            F_new = float(np.max(_fft_conv(a_new)))
            if F_new < F * (1.0 - 1e-14):
                z = z_new
                a = a_new
                accepted = True
                break

        if not accepted:
            if stall > 200:
                # Try a smaller random perturbation
                z = z + 0.01 * rng.standard_normal(n)
                z = z - np.mean(z)
                a = _softmax(z)
                stall = 0
            stall += 1

        if it % 2000 == 0:
            print(f"  polish it={it} F={F:.10f} best={best_F:.10f} "
                  f"score={2*n*best_F:.9f} time_left={deadline-time.monotonic():.0f}s")

    return best_a


def _structured_starts(n, rng):
    """Generate diverse structured starting points."""
    starts = []
    u = np.linspace(0, 1, n)

    # Try to use provided sequence
    try:
        g = globals().get("height_sequence_1")
        if g is not None:
            g = np.asarray(g, dtype=np.float64)
            if g.size > 1:
                g = np.clip(g, 1e-10, 1000.0)
                if np.all(np.isfinite(g)) and np.sum(g) > 0:
                    x_old = np.linspace(0, 1, g.size)
                    g_resampled = np.interp(u, x_old, g)
                    g_resampled = np.maximum(g_resampled, 1e-10)
                    g_resampled /= np.sum(g_resampled)
                    starts.append(g_resampled)
    except Exception:
        pass

    # Two-block sequences
    half = n // 2
    for ratio in (1.1, 1.3, 1.5, 2.0, 2.5, 3.0, 4.0):
        a = np.empty(n)
        a[:half] = 1.0
        a[half:] = ratio
        a = a / np.sum(a)
        starts.append(a)

    # Linear ramps
    for A in (0.2, 0.5, 0.8, 1.2, 1.6, 2.0):
        a = 1.0 + A * (u - 0.5)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
        starts.append(a)

    # Smooth steps (sigmoid-like)
    for A in (0.5, 1.0, 1.5, 2.0):
        for w in (0.02, 0.05, 0.1, 0.2, 0.4):
            sig = 1.0 / (1.0 + np.exp(-(u - 0.5) / w))
            a = 1.0 + A * (sig - 0.5)
            a = np.maximum(a, 0.01)
            a = a / np.sum(a)
            starts.append(a)

    # Cosine windows
    for A in (0.3, 0.6, 1.0, 1.5):
        a = 1.0 + A * np.cos(2.0 * np.pi * u)
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
        starts.append(a)

    # Gaussian
    for sigma in (0.1, 0.15, 0.2, 0.3):
        a = np.exp(-0.5 * ((u - 0.5) / sigma) ** 2)
        a = a / np.sum(a)
        starts.append(a)

    # Random smooth
    for _ in range(5):
        a = np.ones(n)
        for m in range(1, 8):
            a += 0.2 * rng.standard_normal() * np.sin(2 * np.pi * m * u + rng.standard_normal())
        a = np.maximum(a, 0.01)
        a = a / np.sum(a)
        starts.append(a)

    return starts


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Uses log-space gradient with adaptive logsumexp smoothing + exact polish.
    """
    budget_s = max(60.0, float(budget_s))
    deadline = time.monotonic() + budget_s - 8.0
    t0 = time.monotonic()

    n = 8192
    _set_global(n)

    rng = np.random.default_rng(seed)

    # Generate starts
    starts = _structured_starts(n, rng)
    if not starts:
        starts = [np.ones(n) / n]

    # Score all starts
    scored = []
    for a in starts:
        a = np.asarray(a, dtype=np.float64).ravel()
        if a.size != n:
            continue
        a = np.maximum(a, 1e-15)
        a = a / np.sum(a)
        sc = _score_norm(a)
        if np.isfinite(sc):
            scored.append((sc, a.copy()))

    scored.sort(key=lambda x: x[0])
    print(f"Best initial: score={scored[0][0]:.9f}, n={n}")

    best_a = scored[0][1].copy()
    best_score = scored[0][0]

    # Main optimization: L-BFGS in log-space with decreasing tau
    tau_schedule = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001, 0.00005, 0.00002, 0.00001]

    for start_idx in range(min(5, len(scored))):
        if time.monotonic() > deadline - 120:
            break

        sc_init, a_init = scored[start_idx]
        z = np.log(np.maximum(a_init, 1e-30))
        z = z - np.mean(z)
        z = np.clip(z, -25.0, 25.0)

        for tau_base in tau_schedule:
            if time.monotonic() > deadline - 60:
                break

            tau = tau_base / n

            def obj(z_x):
                v, g = _smooth_fg(z_x, tau)
                # Add tiny regularization for numerical stability
                reg = 1e-12
                v += 0.5 * reg * np.dot(z_x, z_x)
                g = g + reg * z_x
                return v, g

            try:
                res = minimize(obj, z.copy(), jac=True, method='L-BFGS-B',
                              options={'maxiter': 300, 'maxfun': 800,
                                       'gtol': 1e-10, 'ftol': 1e-12})
                if res.x is not None:
                    z_new = np.asarray(res.x, dtype=np.float64)
                    z_new = z_new - np.mean(z_new)
                    z_new = np.clip(z_new, -25.0, 25.0)
                    a_new = _softmax(z_new)
                    sc = _score_norm(a_new)
                    if np.isfinite(sc) and sc < best_score:
                        best_score = sc
                        best_a = a_new.copy()
                        z = z_new.copy()
                        print(f"  start={start_idx} tau={tau_base}: "
                              f"score={sc:.9f} elapsed={time.monotonic()-t0:.0f}s")
                    else:
                        # Partial acceptance to allow exploration
                        z = 0.5 * z_new + 0.5 * z
                        z = z - np.mean(z)
            except Exception:
                pass

    print(f"After L-BFGS: best={best_score:.9f} elapsed={time.monotonic()-t0:.0f}s")

    # Exact polishing phase
    if time.monotonic() < deadline - 10:
        print("Starting exact polish...")
        polished = _try_polish(best_a, deadline, max_iters=200000)
        sc_pol = _score_norm(polished)
        if sc_pol < best_score:
            best_score = sc_pol
            best_a = polished.copy()
            print(f"After polish: score={best_score:.9f}")

    # Try a second polish from a slightly perturbed version
    if time.monotonic() < deadline - 30:
        a_pert = best_a * (1.0 + 0.01 * rng.standard_normal(n))
        a_pert = np.maximum(a_pert, 1e-10)
        a_pert = a_pert / np.sum(a_pert)
        polished2 = _try_polish(a_pert, deadline, max_iters=100000)
        sc_pol2 = _score_norm(polished2)
        if sc_pol2 < best_score:
            best_score = sc_pol2
            best_a = polished2.copy()
            print(f"After 2nd polish: score={best_score:.9f}")

    # Final: coordinate-wise improvement attempt
    if time.monotonic() < deadline - 15:
        print("Coordinate improvement...")
        a = best_a.copy()
        n_imp = 0
        for sweep in range(10):
            if time.monotonic() > deadline - 10:
                break
            c = _fft_conv(a)
            cur_F = float(np.max(c))
            improved = False
            # Try random subset of indices
            idx_candidates = rng.choice(n, size=min(200, n), replace=False)
            for idx in idx_candidates:
                if time.monotonic() > deadline - 10:
                    break
                for delta in (0.95, 1.05, 0.9, 1.1, 0.8, 1.2):
                    a_save = a[idx]
                    a[idx] = a_save * delta
                    a[idx] = max(a[idx], 1e-15)
                    # Renormalize (simple: divide by new sum)
                    s = np.sum(a)
                    c_new = _fft_conv(a / s)
                    F_new = float(np.max(c_new))
                    if F_new < cur_F * (1 - 1e-10):
                        a = a / s
                        cur_F = F_new
                        improved = True
                        n_imp += 1
                        break
                    else:
                        a[idx] = a_save
            if not improved and sweep > 2:
                break

        sc_coord = _score_norm(a)
        if sc_coord < best_score:
            best_score = sc_coord
            best_a = a.copy()
            print(f"After coord: score={best_score:.9f} (improvements={n_imp})")

    # Normalize output: scale so sum = sqrt(n) for safety
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.nan_to_num(best_a, nan=0.0, posinf=0.0, neginf=0.0)
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = float(np.sum(best_a))
    if s < 1e-6:
        best_a = np.ones(n) / n
    else:
        best_a = best_a * (np.sqrt(n) / s)

    # Verify with official evaluator
    try:
        result_list = [float(v) for v in best_a.tolist()]
        official = evaluate_sequence(result_list)
        print(f"Official score: {official:.9f}")
    except Exception as ex:
        print(f"Evaluation error: {ex}")
        result_list = [float(v) for v in (best_a / np.sum(best_a)).tolist()]

    print(f"Total time: {time.monotonic()-t0:.1f}s")
    return result_list
