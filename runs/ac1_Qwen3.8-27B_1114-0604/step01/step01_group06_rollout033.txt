"""
Minimizes the score 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences a.
Uses a combination of:
1. Subgradient descent with Nesterov momentum on the simplex
2. Smooth max (log-sum-exp) gradient refinement
3. Coordinate-level local search
4. Multiple sequence lengths and restarts
"""

import numpy as np
import time

# Global state
_G_N = 0
_G_NFFT = 0
_G_DEADLINE = 0.0

def _init(n):
    global _G_N, _G_NFFT
    _G_N = int(n)
    _G_NFFT = 1 << (2 * _G_N - 1).bit_length()

def _fft_conv(a):
    """Linear convolution of a with itself via FFT."""
    N = _G_NFFT
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * _G_N - 1]

def _score_normalized(a):
    """Score for a sequence with sum(a)=1."""
    c = _fft_conv(a)
    return 2.0 * _G_N * float(np.max(c))

def _subgradientFFT(a, active_indices):
    """
    Compute subgradient of max_k (a*a)[k] w.r.t. a.
    g[j] = (1/|S|) * sum_{k in S} a[k-j] (with a[i]=0 outside [0,n-1])
    Uses FFT: g = (s * a_reversed)[n-1 : 2n-1] / |S|
    """
    n = _G_N
    N = _G_NFFT
    s = np.zeros(2 * n - 1)
    s[active_indices] = 1.0
    b = a[::-1]  # reversed
    fs = np.fft.rfft(s, N)
    fb = np.fft.rfft(b, N)
    conv = np.fft.irfft(fs * fb, N)
    g = conv[n - 1:2 * n - 1]
    g /= len(active_indices)
    return g

def _project_simplex(v):
    """Project v onto the probability simplex {a: sum(a)=1, a>=0}."""
    n = len(v)
    # Sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)

def _get_active_set(c, top_frac=0.01):
    """Get indices of the top values in c."""
    n_active = max(1, int(len(c) * top_frac))
    if n_active >= len(c):
        return np.arange(len(c))
    return np.argpartition(c, -n_active)[-n_active:]

def _subgradient_descent(a0, max_iter=5000, deadline=None, verbose=False):
    """
    Subgradient descent with Nesterov momentum on the simplex.
    """
    n = _G_N
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    best_a = a.copy()
    best_score = _score_normalized(a)
    
    # Nesterov momentum
    a_prev = a.copy()
    momentum = 0.9
    
    # Step size schedule
    base_eta = 0.01
    eta_decay = 0.999
    
    if deadline is None:
        deadline = _G_DEADLINE
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Nesterov step: evaluate at y = a + momentum * (a - a_prev)
        y = a + momentum * (a - a_prev)
        y = np.maximum(y, 0)
        s_y = np.sum(y)
        if s_y < 1e-12:
            y = a
        else:
            y = y / s_y
        
        c = _fft_conv(y)
        m = float(np.max(c))
        
        if m * 2.0 * n < best_score:
            best_score = m * 2.0 * n
            best_a = y.copy()
        
        # Get active set
        active = _get_active_set(c, top_frac=0.01)
        g = _subgradientFFT(y, active)
        
        # Update
        eta = base_eta * (eta_decay ** it)
        a_new = y - eta * g
        
        # Project onto simplex
        a_new = _project_simplex(a_new)
        
        # Ensure non-negative and normalized
        a_new = np.maximum(a_new, 0)
        s_new = np.sum(a_new)
        if s_new < 1e-12:
            a_new = best_a.copy()
        else:
            a_new /= s_new
        
        a_prev = a
        a = a_new
        
        if verbose and it % 100 == 0:
            cur = _score_normalized(a)
            print(f"  SGD iter {it}: score={cur:.9f}, best={best_score:.9f}")
    
    return best_a, best_score

def _smooth_max_grad(a, tau=1e-4):
    """
    Compute the gradient of the smooth max approximation.
    F(a) = tau * log(sum(exp(c/tau))) where c = convolve(a,a)
    Returns (F, grad_a)
    """
    n = _G_N
    N = _G_NFFT
    c = _fft_conv(a)
    m = float(np.max(c))
    diff = (c - m) / tau
    diff = np.clip(diff, -700.0, 0.0)
    e = np.exp(diff)
    se = float(np.sum(e))
    if se <= 0 or not np.isfinite(se):
        return m, np.zeros(n)
    p = e / se  # softmax weights over c
    
    # grad F / grad a_j = 2 * sum_k p[k] * a[k-j] (with bounds)
    # This is 2 * (p * a_reversed)[n-1 : 2n-1]
    fs = np.fft.rfft(p, N)
    fb = np.fft.rfft(a[::-1], N)
    conv = np.fft.irfft(fs * fb, N)
    g = conv[n - 1:2 * n - 1] * 2.0
    
    F = m + tau * np.log(se)
    return F, g

def _smooth_max_optimize(a0, max_iter=2000, deadline=None, verbose=False):
    """
    Optimize using the smooth max gradient with Adam-like updates.
    """
    n = _G_N
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    best_a = a.copy()
    best_score = _score_normalized(a)
    
    # Log-space parameterization: a = softmax(z)
    z = np.log(np.maximum(a, 1e-30))
    z -= np.mean(z)
    
    # Adam parameters
    beta1 = 0.8
    beta2 = 0.999
    eps = 1e-8
    lr = 0.001
    m_z = np.zeros(n)
    v_z = np.zeros(n)
    t = 0
    
    tau = 1e-4
    if deadline is None:
        deadline = _G_DEADLINE
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        a = np.exp(z - np.max(z))
        a /= np.sum(a)
        
        F, g_a = _smooth_max_grad(a, tau)
        
        if F * 2.0 * n < best_score:
            best_score = F * 2.0 * n
            best_a = a.copy()
        
        # Convert gradient from a-space to z-space
        # dF/dz_j = sum_i dF/da_i * da_i/dz_j = sum_i g_a_i * a_i * (delta_ij - a_j)
        #         = a_j * (g_a_j - sum_i g_a_i * a_i)
        s_ga = float(np.dot(g_a, a))
        g_z = a * (g_a - s_ga)
        
        # Scale gradient
        g_z *= float(n)  # Scale to get reasonable magnitudes
        
        t += 1
        m_z = beta1 * m_z + (1 - beta1) * g_z
        v_z = beta2 * v_z + (1 - beta2) * g_z ** 2
        m_hat = m_z / (1 - beta1 ** t)
        v_hat = v_z / (1 - beta2 ** t)
        
        z = z - lr * m_hat / (np.sqrt(v_hat) + eps)
        z = np.clip(z, -30, 30)
        z -= np.mean(z)
        
        if verbose and it % 100 == 0:
            cur = _score_normalized(a)
            print(f"  SmoothMax iter {it}: score={cur:.9f}, best={best_score:.9f}")
    
    return best_a, best_score

def _coordinate_refinement(a0, max_iter=500, deadline=None):
    """
    Local coordinate descent: try small perturbations of individual coordinates.
    """
    n = _G_N
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    s = np.sum(a)
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s
    
    best_a = a.copy()
    best_score = _score_normalized(a)
    
    if deadline is None:
        deadline = _G_DEADLINE
    
    rng = np.random.default_rng(12345)
    
    for it in range(max_iter):
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # Pick a random coordinate (or a few)
        n_try = min(10, n)
        indices = rng.choice(n, size=n_try, replace=False)
        
        improved = False
        for j in indices:
            # Try decreasing a[j] (mass moves to others)
            for delta_frac in (0.01, 0.005, 0.001):
                delta = delta_frac * a[j] / n  # small absolute change
                a_new = a.copy()
                a_new[j] -= delta
                # Distribute delta to random other positions (or uniform)
                a_new += delta / n
                # Ensure non-negative
                a_new = np.maximum(a_new, 0)
                s_new = np.sum(a_new)
                if s_new > 1e-12:
                    a_new /= s_new
                    sc = _score_normalized(a_new)
                    if sc < best_score - 1e-15:
                        a = a_new
                        best_a = a.copy()
                        best_score = sc
                        improved = True
                        break
        
        if not improved and it > 100:
            # Try a global random perturbation
            perturbation = 0.01 * rng.standard_normal(n)
            a_new = np.maximum(a + perturbation, 0)
            s_new = np.sum(a_new)
            if s_new > 1e-12:
                a_new /= s_new
                sc = _score_normalized(a_new)
                if sc < best_score - 1e-15:
                    a = a_new
                    best_a = a.copy()
                    best_score = sc
                else:
                    if it % 50 == 0:
                        print(f"  Coord iter {it}: stuck at {best_score:.9f}")
                    if it > 300:
                        break
    
    return best_a, best_score

def _make_initial_candidates(n, seed=42):
    """Generate diverse initial candidates."""
    rng = np.random.default_rng(seed)

                log_g = np.log(g_pos)
                log_interp = np.interp(x_new, x_old, log_g)
                a = np.exp(log_interp - np.max(log_interp))
                a /= np.sum(a)
                candidates.append(a)
    except Exception:

"""
Minimizes score = 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences.
Strategy:
1. Mirror descent (exponential weights / FTLL) in log-space with adaptive step size
2. Soft active-set subgradient via temperature-weighted peaks
3. Smooth max (log-sum-exp) gradient for fine polishing
4. Multiple restarts, sequence lengths, and diverse initializations
5. Coordinate-level local search as final refinement
"""

import numpy as np
import time

_G_N = 0
_G_NFFT = 0
_G_DEADLINE = 0.0


def _init(n):
    global _G_N, _G_NFFT
    _G_N = int(n)
    _G_NFFT = 1 << (2 * _G_N - 1).bit_length()


def _fft_conv(a):
    """Linear convolution a*a via FFT."""
    N = _G_NFFT
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:2 * _G_N - 1]


def _score_norm(a):
    """Score for normalized sequence (sum=1). Returns 2*n*max(c)."""
    c = _fft_conv(a)
    return 2.0 * _G_N * float(np.max(c))


def _subgrad_soft(a, c, tau):
    """
    Smooth subgradient: gradient of tau*logsumexp(c/tau) w.r.t. a.
    g[j] = 2 * sum_k p[k] * a[k-j] where p = softmax(c/tau)
    """
    n = _G_N
    N = _G_NFFT
    m = float(np.max(c))
    diff = (c - m) / tau
    diff = np.clip(diff, -700.0, 0.0)
    e = np.exp(diff)
    se = float(np.sum(e))
    if se <= 0.0 or not np.isfinite(se):
        return np.zeros(n)
    p = e / se
    # g[j] = 2 * sum_k p[k] * a[k-j] (valid indices only)
    # = 2 * (p * a_reversed)[n-1 : 2n-1]
    fs = np.fft.rfft(p, N)
    fb = np.fft.rfft(a[::-1], N)
    conv = np.fft.irfft(fs * fb, N)
    g = conv[n - 1: 2 * n - 1] * 2.0
    return g


def _project_simplex(v):
    """Project onto probability simplex."""
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n + 1) > (cssv - 1))[0]
    if len(rho) == 0:
        return np.ones(n) / n
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    return np.maximum(v - theta, 0)


def _mirror_descent(a0, max_iter=20000, tau_init=0.1, tau_final=0.001,
                    eta_init=0.5, eta_decay=0.9995, deadline=None, verbose=False):
    """
    Mirror descent (exponential weights) on the simplex.
    Updates: a[j] *= exp(-eta * g[j]) then normalize.
    """
    n = _G_N
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 1e-20)
    a /= np.sum(a)

    best_a = a.copy()
    best_score = _score_norm(a)

    if deadline is None:
        deadline = _G_DEADLINE

    log_a = np.log(a)

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_conv(a)
        m = float(np.max(c))
        sc = 2.0 * n * m
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Adaptive tau: start large (smooth), decrease over time
        frac = it / max(1, max_iter)
        tau = tau_init * (tau_final / tau_init) ** frac

        g = _subgrad_soft(a, c, tau)

        # Adaptive eta
        g_norm = float(np.max(np.abs(g)))
        if g_norm < 1e-15:
            break
        eta = eta_init * (eta_decay ** it) / (1.0 + 0.1 * g_norm)

        # Mirror descent update in log-space
        log_a = log_a - eta * g
        # Periodic re-normalization of log_a for numerical stability
        if it % 100 == 0:
            log_a = log_a - np.mean(log_a)

        a = np.exp(log_a - np.max(log_a))
        a = np.maximum(a, 1e-20)
        a /= np.sum(a)

        if verbose and it % 2000 == 0:
            print(f"    MD iter {it}: score={_score_norm(a):.9f}, best={best_score:.9f}, "
                  f"tau={tau:.5f}, eta={eta:.6f}")

    return best_a, best_score


def _smooth_max_adam(a0, max_iter=15000, deadline=None, verbose=False):
    """
    Optimize using smooth max (log-sum-exp) gradient with Adam in log-space.
    """
    n = _G_N
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 1e-20)
    a /= np.sum(a)

    z = np.log(a)
    z -= np.mean(z)

    best_a = a.copy()
    best_score = _score_norm(a)

    beta1, beta2, eps = 0.8, 0.999, 1e-8
    lr = 0.01
    m_z = np.zeros(n)
    v_z = np.zeros(n)
    t = 0

    # Anneal tau from 0.01 to 1e-5
    tau_start = 0.01
    tau_end = 1e-5

    if deadline is None:
        deadline = _G_DEADLINE

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        a = np.exp(z - np.max(z))
        a = np.maximum(a, 1e-20)
        a /= np.sum(a)

        c = _fft_conv(a)
        m = float(np.max(c))
        sc = 2.0 * n * m
        if sc < best_score:
            best_score = sc
            best_a = a.copy()

        # Anneal tau
        frac = min(1.0, it / max(1, max_iter * 0.8))
        tau = tau_start * (tau_end / tau_start) ** frac

        # Compute gradient of smooth max w.r.t. a
        g_a = _subgrad_soft(a, c, tau)

        # Convert to z-space: dF/dz_j = a_j * (g_a_j - sum(g_a * a))
        s_ga = float(np.dot(g_a, a))
        g_z = a * (g_a - s_ga)
        g_z *= float(n)  # Scale for reasonable magnitudes

        t += 1
        m_z = beta1 * m_z + (1 - beta1) * g_z
        v_z = beta2 * v_z + (1 - beta2) * g_z * g_z
        m_hat = m_z / (1 - beta1 ** t)
        v_hat = v_z / (1 - beta2 ** t)

        # Adaptive learning rate
        lr_t = lr / (1.0 + 0.01 * t)
        z = z - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        z = np.clip(z, -40.0, 40.0)
        z -= np.mean(z)

        if verbose and it % 2000 == 0:
            print(f"    SM iter {it}: score={sc:.9f}, best={best_score:.9f}, tau={tau:.6f}")

    return best_a, best_score


def _subgrad_momentum(a0, max_iter=30000, deadline=None, verbose=False):
    """
    Subgradient descent with Nesterov momentum and simplex projection.
    Uses exact (hard) active set for the subgradient.
    """
    n = _G_N
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0.0)
    s = np.sum(a)
    if s < 1e-12:
        a = np.ones(n) / n
    else:
        a /= s

    best_a = a.copy()
    best_score = _score_norm(a)
    v = np.zeros(n)  # velocity for momentum
    mu = 0.85  # momentum
    base_eta = 0.05

    if deadline is None:
        deadline = _G_DEADLINE

    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Nesterov lookahead
        y = a + mu * v
        y = np.maximum(y, 0)
        sy = np.sum(y)
        if sy < 1e-12:
            y = a
        else:
            y = y / sy

        c = _fft_conv(y)
        m = float(np.max(c))
        sc = 2.0 * n * m

        if sc < best_score - 1e-15:
            best_score = sc
            best_a = y.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 5000:
            break

        # Hard active set: top 0.5% of c values
        n_active = max(1, int(len(c) * 0.005))
        top_idx = np.argpartition(c, -n_active)[-n_active:]
        # Compute subgradient via FFT
        s_arr = np.zeros(len(c))
        s_arr[top_idx] = 1.0
        N = _G_NFFT
        fs = np.fft.rfft(s_arr, N)
        fb = np.fft.rfft(y[::-1], N)
        conv = np.fft.irfft(fs * fb, N)
        g = conv[n - 1: 2 * n - 1] / n_active

        # Adaptive step size
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        eta = base_eta / (1.0 + 0.5 * g_norm)

        # Update
        v = mu * v - eta * g
        a_new = y + v
        a_new = _project_simplex(a_new)
        a_new = np.maximum(a_new, 0)
        a_new /= np.sum(a_new)

        a = a_new

        if verbose and it % 5000 == 0:
            print(f"    SubGrad iter {it}: score={sc:.9f}, best={best_score:.9f}")

    return best_a, best_score


def _local_search(a0, max_iter=3000, deadline=None):
    """
    Coordinate-level local search: try small perturbations to reduce the peak.
    """
    n = _G_N
    a = np.asarray(a0, dtype=np.float64).copy()
    a = np.maximum(a, 0)
    a /= np.sum(a)

    best_a = a.copy()
    best_score = _score_norm(a)
    rng = np.random.default_rng(777)

    if deadline is None:
        deadline = _G_DEADLINE

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_conv(a)
        m = float(np.max(c))
        # Focus on coordinates that contribute to peaks
        peak_idx = np.argmax(c)
        # The peak c[peak_idx] = sum a[i]*a[peak_idx-i]
        # Coordinates contributing: i and peak_idx-i for valid i
        lo = max(0, peak_idx - n + 1)
        hi = min(n - 1, peak_idx)
        contributors = np.arange(lo, hi + 1)
        if len(contributors) < 5:
            contributors = np.arange(n)

        # Pick a random contributor and try to reduce it
        j = rng.choice(contributors)
        for delta_frac in (0.02, 0.01, 0.005):
            delta = delta_frac * a[j]
            if delta < 1e-10:
                continue
            a_new = a.copy()
            a_new[j] -= delta
            # Redistribute to random positions (not the peak contributors)
            other_idx = np.setdiff1d(np.arange(n), contributors)
            if len(other_idx) > 0:
                targets = rng.choice(other_idx, size=min(5, len(other_idx)), replace=False)
                for t_idx in targets:
                    a_new[t_idx] += delta / len(targets)
            a_new = np.maximum(a_new, 0)
            a_new /= np.sum(a_new)
            sc = _score_norm(a_new)
            if sc < best_score - 1e-14:
                a = a_new
                best_a = a.copy()
                best_score = sc
                break

    return best_a, best_score


def _make_starts(n, seed=42):
    """Generate diverse initial candidates."""
    rng = np.random.default_rng(seed)
    candidates = []

    # 1. Provided sequence (resampled)

                log_g = np.log(g_pos)
                log_interp = np.interp(x_new, x_old, log_g)

                a = np.maximum(a, 1e-10)

    # 2. Uniform

        half = n // 2
        a = np.empty(n)

        candidates.append(a)

    # 4. Smooth bandlimited random
    for _ in range(8):

        x = np.linspace(0, 1, n)
        a = np.ones(n)
        for k in range(1, n_modes + 1):
            a += 0.15 * rng.standard_normal() * np.cos(2 * np.pi * k * x + rng.standard_normal())
        a = np.maximum(a, 1e-6)
        a /= np.sum(a)
        candidates.append(a)

    # 5. Exponential decay profiles
    for rate in (0.5, 1.0, 2.0, 4.0):
        x = np.linspace(0, 1, n)

    # 6. Sinusoidal

        a = np.maximum(a, 1e-6)
        a /= np.sum(a)

def _optimize_candidate(a0, time_budget):
    """Run multi-phase optimization on a single candidate."""
    best_a = a0.copy()
    best_score = _score_norm(a0)

    # Phase 1: Mirror descent (exploration)
    t1 = time.monotonic() + time_budget * 0.35
    a1, s1 = _mirror_descent(a0, max_iter=20000, deadline=t1)
    if s1 < best_score:
        best_a, best_score = a1, s1

    # Phase 2: Subgradient with momentum
    t2 = time.monotonic() + time_budget * 0.25
    a2, s2 = _subgrad_momentum(best_a, max_iter=30000, deadline=t2)
    if s2 < best_score:
        best_a, best_score = a2, s2

    # Phase 3: Smooth max Adam (exploitation)
    t3 = time.monotonic() + time_budget * 0.30
    a3, s3 = _smooth_max_adam(best_a, max_iter=15000, deadline=t3)
    if s3 < best_score:
        best_a, best_score = a3, s3

    # Phase 4: Local search
    t4 = time.monotonic() + time_budget * 0.10
    a4, s4 = _local_search(best_a, max_iter=3000, deadline=t4)
    if s4 < best_score:
        best_a, best_score = a4, s4

    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main search function. Searches for the best non-negative coefficient sequence
    minimizing 2*n*max(convolve(a,a))/(sum(a))^2.
    Returns the best sequence found as a list of floats.
    """
    global _G_DEADLINE
    budget_s = max(60.0, float(budget_s))
    _G_DEADLINE = time.monotonic() + budget_s - 8.0

    t0 = time.monotonic()
    print(f"=== Search started, budget={budget_s:.0f}s ===")

    # Determine which sequence lengths to try
    if budget_s >= 2000:
        n_options = [8192, 16384]
    elif budget_s >= 1000:
        n_options = [4096, 8192]
    else:
        n_options = [4096]

    best_overall_a = None
    best_overall_score = np.inf
    best_overall_n = 0

    for n in n_options:
        if time.monotonic() > _G_DEADLINE:
            break

        _init(n)
        time_left = _G_DEADLINE - time.monotonic()
        print(f"\n--- n = {n}, time_left = {time_left:.0f}s ---")

        # Generate and score initial candidates
        starts = _make_starts(n, seed=seed)
        scored = []
        for a in starts:
            a = np.asarray(a, dtype=np.float64)
            a = np.maximum(a, 0)
            s = np.sum(a)
            if s < 1e-12:
                continue
            a = a / s
            sc = _score_norm(a)
            if np.isfinite(sc):
                scored.append((sc, a))

        scored.sort(key=lambda x: x[0])
        print(f"  Generated {len(scored)} starts, best initial: {scored[0][0]:.9f}")

        # Allocate time per candidate
        n_cands = min(4, len(scored))
        time_per_cand = (time_left - 30) / max(1, n_cands) * 0.75

        for i in range(n_cands):
            if time.monotonic() > _G_DEADLINE:
                break
            if time.monotonic() + 60 > _G_DEADLINE:
                break

            sc, a = scored[i]
            print(f"  Optimizing candidate {i} (init={sc:.9f}), time_budget={time_per_cand:.0f}s")

            a_opt, sc_opt = _optimize_candidate(a, time_per_cand)
            print(f"  Candidate {i} result: {sc_opt:.9f}")

            if sc_opt < best_overall_score:
                best_overall_score = sc_opt
                best_overall_a = a_opt.copy()
                best_overall_n = n
                print(f"  *** NEW GLOBAL BEST: {sc_opt:.9f} (n={n}) ***")

    if best_overall_a is None:
        # Fallback
        n = 8192
        _init(n)
        best_overall_a = np.ones(n) / n
        best_overall_n = n
        best_overall_score = _score_norm(best_overall_a)

    print(f"\n=== Final: score={best_overall_score:.9f}, n={best_overall_n} ===")

    # Validate
    try:
        a_list = [float(v) for v in best_overall_a]
        val = evaluate_sequence(a_list)
        print(f"  evaluate_sequence verification: {val}")
        if np.isfinite(val) and val < best_overall_score:
            best_overall_score = val
    except Exception as e:
        print(f"  verification error: {e}")

    # Final safety: ensure valid output
    a_out = np.asarray(best_overall_a, dtype=np.float64).ravel()
    a_out = np.nan_to_num(a_out, nan=0.0, posinf=0.0, neginf=0.0)
    a_out = np.clip(a_out, 0.0, 1000.0)
    s = float(np.sum(a_out))
    if s < 0.01:
        a_out = np.ones(best_overall_n) / best_overall_n
    return [float(v) for v in a_out]
