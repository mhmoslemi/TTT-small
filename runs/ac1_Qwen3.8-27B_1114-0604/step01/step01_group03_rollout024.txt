"""
Search for non-negative step heights minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses exact subgradient of the max-convolution (which has the simple structure of 
a reversed/shifted copy of a), adaptive step sizes with line search, top-k peak 
awareness, multiplicative updates on the simplex, structured restarts, and 
polished final candidates via coordinate descent.
"""

import math
import time
import numpy as np


def _next_pow2(m):
    return 1 << (m - 1).bit_length()


def _fft_conv_self(a, L):
    """Compute full self-convolution of a using FFT, returned as length 2*len(a)-1."""
    n = len(a)
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)
    return c[:2 * n - 1]


def _score(conv, n):
    """Compute 2*n*max(conv)."""
    return 2.0 * n * float(np.max(conv))


def _normalize_sum1(a):
    """Normalize to sum 1, non-negative."""
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300:
        a[:] = 1.0 / len(a)
        return a
    a /= s
    return a


def _subgradient_topk(a, conv, k_top=3):
    """
    Compute weighted subgradient using top-k peaks of conv.
    Returns the subgradient vector s of length n.
    s[j] = 2 * sum_t w_t * a[k_t - j] for valid (k_t - j) in [0, n-1].
    """
    n = len(a)
    # Find top-k peaks
    # For efficiency, use argpartition for large conv
    if k_top >= len(conv):
        k_top = len(conv)
    
    if k_top == 1:
        k_star = int(np.argmax(conv))
        # s[j] = 2*a[k_star - j] for 0 <= k_star - j < n
        # i.e., j in [max(0, k_star-n+1), min(n-1, k_star)]
        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        s = np.zeros(n, dtype=np.float64)
        if j_lo <= j_hi:
            indices = np.arange(j_lo, j_hi + 1)
            a_indices = k_star - indices  # = k_star - j for j in indices
            s[indices] = 2.0 * a[a_indices]
        return s
    
    # Top-k with weights
    # Get indices of top-k
    if k_top < len(conv):
        top_idx = np.argpartition(conv, -k_top)[-k_top:]
    else:
        top_idx = np.arange(len(conv))
    
    # Sort by value descending
    top_vals = conv[top_idx]
    order = np.argsort(-top_vals)
    top_idx = top_idx[order]
    top_vals = top_vals[order]
    
    # Weights: proportional to values (clip small values)
    max_val = float(top_vals[0])
    if max_val < 1e-300:
        return np.zeros(n, dtype=np.float64)
    
    # Use exponential weighting for smoothness
    w = top_vals / max_val
    w_sum = float(w.sum())
    w /= w_sum
    
    s = np.zeros(n, dtype=np.float64)
    for t in range(k_top):
        k_t = int(top_idx[t])
        w_t = float(w[t]) * 2.0
        j_lo = max(0, k_t - n + 1)
        j_hi = min(n - 1, k_t)
        if j_lo <= j_hi:
            indices = np.arange(j_lo, j_hi + 1)
            a_indices = k_t - indices
            s[indices] += w_t * a[a_indices]
    
    return s


def _multiplicative_update(a, step, s):
    """Apply multiplicative update: a_new = a * exp(-step * s), normalized."""
    # Clip s to avoid overflow
    s_max = float(np.max(np.abs(s)))
    if s_max < 1e-300:
        return a.copy()
    
    log_update = -step * s / s_max
    # Clip to avoid overflow
    log_update = np.clip(log_update, -50.0, 50.0)
    
    a_new = a * np.exp(log_update)
    return _normalize_sum1(a_new)


def _line_search_multiplicative(a, s, current_score, n, L, max_trials=8):
    """
    Line search for multiplicative update. Try decreasing step sizes.
    Returns (best_a, best_score, best_step).
    """
    best_a = a
    best_score = current_score
    best_step = 0.0
    
    s_norm = float(np.max(s))
    if s_norm < 1e-300:
        return a, current_score, 0.0
    
    # Initial step size
    step = 1.0 / s_norm
    
    for trial in range(max_trials):
        a_new = _multiplicative_update(a, step, s)
        conv_new = _fft_conv_self(a_new, L)
        score_new = _score(conv_new, n)
        
        if score_new < best_score:
            best_score = score_new
            best_a = a_new
            best_step = step
        
        step *= 0.6  # shrink step
    
    return best_a, best_score, best_step


def _structured_perturb(a, rng, alpha=0.3):
    """
    Apply structured perturbation to escape local minima.
    Mix with a random Fourier-based shape.
    """
    n = len(a)
    t = np.linspace(0.0, 1.0, n)
    
    # Random Fourier perturbation
    n_harmonics = int(rng.integers(1, 8))
    pert = np.zeros(n, dtype=np.float64)
    for k in range(1, n_harmonics + 1):
        amp = rng.uniform(0.1, 0.8) / (k ** 0.5)
        ph = rng.uniform(0, 2 * np.pi)
        pert += amp * np.cos(2 * np.pi * k * t + ph)
    
    # Normalize perturbation
    pert = pert - float(np.mean(pert))
    pert_std = float(np.std(pert))
    if pert_std > 1e-10:
        pert = pert / pert_std
    
    # Mix
    new_a = a * (1.0 + alpha * pert)
    return _normalize_sum1(new_a)


def _random_fourier_init(n, rng):
    """Generate a random initialization based on Fourier modes."""
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0
    
    z = np.zeros(n, dtype=np.float64)
    
    # Add random harmonics
    n_harmonics = int(rng.integers(3, 15))
    for k in range(1, n_harmonics + 1):
        amp = rng.uniform(0.1, 1.0) / (k ** rng.uniform(0.3, 1.0))
        ph = rng.uniform(0, 2 * np.pi)
        z += amp * np.cos(2 * np.pi * k * t + ph)
    
    # Add some structure
    shape_choice = rng.integers(0, 5)
    if shape_choice == 0:
        z += 0.5 * x
    elif shape_choice == 1:
        z += 0.5 * (x * x - 1/3)
    elif shape_choice == 2:
        z += 0.3 * np.sin(4 * np.pi * t)
    elif shape_choice == 3:
        z += 0.3 * (1 - 3*(t-0.3)**2) + 0.3 * (1 - 3*(t-0.7)**2)
    else:
        z += 0.3 * np.cos(6 * np.pi * t)
    
    # Convert to positive
    z_shift = z - float(np.min(z)) + 0.1
    z_shift = np.maximum(z_shift, 0.01)
    
    return _normalize_sum1(z_shift)


def _subgradient_descent(a, n, L, rng, time_limit, verbose=False):
    """
    Main local search using subgradient method with multiplicative updates.
    """
    deadline = time.monotonic() + time_limit
    a = _normalize_sum1(a.copy())
    
    conv = _fft_conv_self(a, L)
    current_score = _score(conv, n)
    
    best_a = a.copy()
    best_score = current_score
    
    # Running average
    avg_a = a.copy()
    avg_count = 0
    avg_window = 200  # average over last 200 accepted steps
    
    no_improve = 0
    it = 0
    k_top = 1  # start with single peak
    
    while time.monotonic() < deadline:
        # Compute subgradient
        s = _subgradient_topk(a, conv, k_top)
        
        # Line search
        a_new, score_new, step_used = _line_search_multiplicative(
            a, s, current_score, n, L, max_trials=6
        )
        
        if score_new < current_score - 1e-15:
            a = a_new
            current_score = score_new
            conv = _fft_conv_self(a, L)
            
            # Update running average
            avg_a = avg_a + (a - avg_a) / (avg_count + 1)
            avg_count = min(avg_count + 1, avg_window)
            
            if current_score < best_score - 1e-13:
                best_score = current_score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            # Adaptively increase k_top if peaks are close
            if it % 50 == 0 and it > 0:
                max_val = float(np.max(conv))
                # If second peak is close to max, use top-2
                sorted_conv = np.sort(conv)[::-1]
                if len(sorted_conv) > 1 and sorted_conv[1] > 0.95 * max_val:
                    k_top = min(k_top + 1, 5)
                elif sorted_conv[1] > 0.9 * max_val:
                    k_top = max(1, k_top - 1)

        # Structured restart if stuck
        if no_improve > 100:

            conv = _fft_conv_self(a, L)
            current_score = _score(conv, n)

            avg_a = a.copy()

        if time.monotonic() > deadline - 1.0:

    # Also check the running average
    avg_score = _score(_fft_conv_self(avg_a, L), n)
    if avg_score < best_score:
        best_score = avg_score

    return best_a, best_score


def _coordinate_descent(a, n, L, time_limit, rng, verbose=False):
    """
    Coordinate descent polishing step.
    For each coordinate, find the optimal value given all others fixed.
    """
    deadline = time.monotonic() + time_limit
    a = _normalize_sum1(a.copy())
    n_inner = len(a)
    
    conv = _fft_conv_self(a, L)
    current_score = _score(conv, n_inner)
    best_a = a.copy()
    best_score = current_score
    
    # Process coordinates in random order
    order = rng.permutation(n_inner)
    pos = 0
    
    while time.monotonic() < deadline:
        j = order[pos % n_inner]
        pos += 1
        
        # Try perturbing coordinate j
        a_j = a[j]
        # Try several values
        candidates = [a_j * 0.5, a_j * 0.8, a_j * 1.2, a_j * 1.5,
                      a_j + 0.01 * best_score / n_inner,
                      a_j - 0.01 * best_score / n_inner,
                      0.0, a_j * 2.0]
        
        best_val = a_j
        best_local_score = current_score
        
        for val in candidates:
            if val < 0:
                continue
            a[j] = val
            _normalize_sum1(a)
            conv_test = _fft_conv_self(a, L)
            sc = _score(conv_test, n_inner)
            if sc < best_local_score:
                best_local_score = sc
                best_val = val
        
        a[j] = best_val
        _normalize_sum1(a)
        
        if best_local_score < current_score - 1e-14:
            current_score = best_local_score
            conv = _fft_conv_self(a, L)
            if current_score < best_score - 1e-14:
                best_score = current_score
                best_a = a.copy()
    
    return best_a, best_score

def _exact_score(seq):
    """Call evaluate_sequence if available, else compute internally."""
    try:
        val = evaluate_sequence(seq)
    except NameError:
        # Internal computation
        a = np.array([max(0.0, min(1000.0, float(x))) for x in seq])
        n = len(a)
        s = float(a.sum())
        if s < 0.01:
            return float('inf')
        b = np.convolve(a, a)
        mb = float(np.max(b))
        return float(2.0 * n * mb / (s * s))

    if val is None:
        return float('inf')
    return float(val)

def _next_pow2(m):
    if m <= 1:

def _fft_conv_self(a, L):
    n = len(a)
    f = np.fft.rfft(a, L)
    c = np.fft.irfft(f * f, L)
    return c[:2 * n - 1]


def _score(conv, n):
    return 2.0 * n * float(np.max(conv))


def _normalize_sum1(a):
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 1e-300:
        a[:] = 1.0 / len(a)
        return a
    a /= s
    return a


def _subgradient_topk(a, conv, k_top=3):
    n = len(a)
    m = len(conv)
    k_top = min(k_top, m)
    
    if k_top == 1:
        k_star = int(np.argmax(conv))

        j_lo = max(0, k_star - n + 1)
        j_hi = min(n - 1, k_star)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            a_idx = k_star - idx
            s[idx] = 2.0 * a[a_idx]
        return s
    
    if k_top < m:
        top_idx = np.argpartition(-conv, k_top)[:k_top]
    else:
        top_idx = np.arange(m)

    order = np.argsort(-top_vals)
    top_idx = top_idx[order]
    top_vals = top_vals[order]
    
    max_val = float(top_vals[0])
    if max_val < 1e-300:

            idx = np.arange(j_lo, j_hi + 1)

        return a.copy()

    log_u = np.clip(log_u, -40.0, 40.0)
    a_new = a * np.exp(log_u)
    return _normalize_sum1(a_new)


def _line_search(a, s, current_score, n, L, max_trials=8, step_init=None):

        return a, current_score, 0.0

        print(f"\n=== n={n}, L={L}, time={t_alloc:.0f}s ===")
        
        cands = _generate_candidates(n, L, rng)
        print(f"  Generated {len(cands)} candidates, best init score = {cands[0][0]:.8f}")
        
        top_k = min(3, len(cands))
        cand_times = []
        if top_k >= 3:
            cand_times = [0.50, 0.30, 0.20]
        elif top_k == 2:
            cand_times = [0.60, 0.40]
        else:
            cand_times = [1.0]
        
        for ci in range(top_k):
            if time.monotonic() > search_deadline - 10:
                break
            t_rem = search_deadline - time.monotonic()
            t_ci = min(cand_times[ci] * t_alloc, t_rem - 5.0)
            if t_ci < 5.0:
                break
            
            init_a = cands[ci][1]
            
            if ci == 0:
                a_res, sc_res = _nesterov_subgrad(init_a, n, L, rng, t_ci * 0.45)
                if time.monotonic() < search_deadline - 10:
                    t2 = min(t_ci * 0.35, search_deadline - time.monotonic() - 5)
                    if t2 > 5:
                        a_res2, sc_res2 = _vanilla_subgrad(a_res, n, L, rng, t2)
                        if sc_res2 < sc_res:
                            a_res, sc_res = a_res2, sc_res2
            else:
                a_res, sc_res = _vanilla_subgrad(init_a, n, L, rng, t_ci)
            
            if sc_res < best_score_global:
                best_score_global = sc_res
                best_a_global = a_res.copy()
                best_n_global = n
                print(f"  n={n}, cand {ci+1}: {sc_res:.8f} (NEW GLOBAL BEST)")
            else:
                print(f"  n={n}, cand {ci+1}: {sc_res:.8f}")
    
    if best_a_global is None:
        n = 6121
        best_a_global = np.ones(n, dtype=np.float64) / n
        best_n_global = n
        best_score_global = 2.0
        print("Fallback to uniform sequence.")
    
    print(f"\nBest before polish: n={best_n_global}, score={best_score_global:.8f}")
    
    # Polishing phase
    n = best_n_global
    L = _next_pow2(2 * n - 1)
    time_left = search_deadline - time.monotonic()
    
    if time_left > 30:
        t_polish = min(180.0, time_left * 0.6)
        print(f"\nPolishing (subgradient) for {t_polish:.0f}s...")
        a_p, sc_p = _nesterov_subgrad(best_a_global, n, L, rng, t_polish)
        if sc_p < best_score_global:
            best_score_global = sc_p
            best_a_global = a_p
            print(f"  After Nesterov polish: {sc_p:.8f}")
    
    time_left = search_deadline - time.monotonic()
    if time_left > 20 and n <= 8000:
        t_cd = min(time_left * 0.5, 120.0)
        if t_cd > 10:
            print(f"  Coordinate descent for {t_cd:.0f}s...")
            a_cd, sc_cd = _coord_descent(best_a_global, n, L, t_cd, rng)
            if sc_cd < best_score_global:
                best_score_global = sc_cd
                best_a_global = a_cd
                print(f"  After coord descent: {sc_cd:.8f}")
    
    # Final exact evaluation with multiple scalings
    seq_array = best_a_global.copy()
    best_exact = float('inf')
    best_seq_list = None
    
    for scale in (1.0, 10.0, 100.0, 0.1, 1.0 / float(n)):
        seq_list = [float(x) * scale for x in seq_array]
        sc = _evaluate_local(seq_list)
        if np.isfinite(sc) and sc < best_exact:
            best_exact = sc
            best_seq_list = seq_list
    
    if best_seq_list is None:
        best_seq_list = [float(x) for x in seq_array]
        best_exact = _evaluate_local(best_seq_list)
    
    print(f"\nFinal exact score: {best_exact:.8f}")
    print(f"Sequence length: {len(best_seq_list)}")
    print(f"Total time: {time.monotonic() - t0:.1f}s")
    
    return best_seq_list
