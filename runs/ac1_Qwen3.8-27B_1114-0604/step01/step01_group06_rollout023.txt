        res = minimize(
            _obj_grad_lse,
            z0,
            args=(tau,),
            jac=True,
            method='L-BFGS-B',
            bounds=_BOUNDS,
            options={'maxfun': maxfun, 'maxiter': maxiter, 'ftol': 1e-14, 'pgtol': 1e-12}
        )
        return res.x
    except Exception as e:
        print(f"  L-BFGS error: {e}")
        return None

def _polish(a0, deadline, tau_start=0.1, tau_end=1e-5, rng=None):
    """Intensive gradient polishing with adaptive tau and backtracking."""
    if rng is None:
        rng = np.random.default_rng(12345)
    
    n = a0.size
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0)
    s = float(np.sum(a))
    if s < 1e-12:
        return a0
    a = a / s
    
    z = np.log(np.maximum(a, 1e-300))
    z = z - np.mean(z)
    
    best_a = a.copy()
    best_maxc = float(np.max(_fft_conv(a)))
    best_score = 2.0 * n * best_maxc
    
    total_time = max(60.0, deadline - time.monotonic())
    no_improve = 0
    it = 0
    t0 = time.monotonic()
    
    while time.monotonic() < deadline:
        # Adaptive tau: start large, decrease over time
        progress = min(1.0, (time.monotonic() - t0) / total_time)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Compute current state
        c = _fft_conv(a)
        cur_maxc = float(np.max(c))
        
        if cur_maxc < best_maxc - 1e-18:
            best_a = a.copy()
            best_maxc = cur_maxc
            best_score = 2.0 * n * cur_maxc
            no_improve = 0
        else:
            no_improve += 1
        
        if no_improve > 500:
            # Try to escape
            escaped = False
            for scale in [0.01, 0.05, 0.1, 0.3, 0.5]:
                u = np.linspace(0, 1, n)
                pert = np.zeros(n)
                for k in range(1, 9):
                    amp = scale * rng.standard_normal()
                    pert += amp * np.cos(2 * np.pi * k * u + rng.standard_normal())
                z_pert = np.clip(z + pert, -30.0, 30.0)
                a_pert = _softmax(z_pert)
                maxc_pert = float(np.max(_fft_conv(a_pert)))
                if maxc_pert < best_maxc:
                    z = z_pert
                    a = a_pert
                    escaped = True
                    no_improve = 0
                    break
            if not escaped:
                # Reduce tau further to try different resolution
                tau = max(tau * 0.1, 1e-8)
                if tau < 1e-8:
                    break
        
        # Compute gradient using LSE smoothing
        val, g = _lse_grad(a, tau)
        
        if not np.all(np.isfinite(g)):
            break

        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-15:
            break
        
        g = g / g_norm  # Unit direction
        
        # Negate: we want to decrease the objective
        direction = -g
        
        # Backtracking line search
        accepted = False
        cur_f = 2.0 * n * cur_maxc  # Use exact max, not smoothed
        for eta in [1e-3, 3e-3, 1e-2, 3e-2, 0.1, 0.3, 1.0, 3.0, 10.0]:
            # Move in z-space
            z_new = np.clip(z + eta * direction * n / (2.0 * n), -30.0, 30.0)
            # Actually, let's work directly in a-space with projection
            # a_new = softmax(z + eta * grad_z)
            # But grad is in a-space. Let's convert.
            # In a-space: a_new = a + eta * direction (then project to simplex)
            # In z-space: z_new = z + eta * (direction / a) approximately
            
            # Work in z-space for stability
            # The gradient in z-space is: g_z = a * (g - dot(g, a))
            s_g = float(np.dot(g, a))  # Wait, g is already 2*a*(h-s_h)
            # Actually, let me just work in z-space directly

            break
        
        # Let me redo this more carefully
        # Work in z-space
        # a = softmax(z)
        # dF/dz = a * (g_a - sum(g_a * a)) where g_a is the a-space gradient
        g_a = g  # This is 2*a*(h-s_h) from _lse_grad
        s_g = float(np.dot(g_a, a))
        g_z = a * (g_a - s_g)
        
        # Backtrack in z-space
        accepted = False
        for eta in [1e-4, 3e-4, 1e-3, 3e-3, 1e-2, 3e-2, 0.1, 0.3, 1.0]:
            z_new = np.clip(z - eta * g_z, -30.0, 30.0)
            a_new = _softmax(z_new)
            maxc_new = float(np.max(_fft_conv(a_new)))
            if maxc_new < cur_maxc - 1e-16:
                z = z_new
                a = a_new
                accepted = True
                break
        
        if not accepted:
            # Smaller steps
            for eta in [1e-5, 3e-5, 1e-6, 3e-6]:
                z_new = np.clip(z - eta * g_z, -30.0, 30.0)
                a_new = _softmax(z_new)
                maxc_new = float(np.max(_fft_conv(a_new)))
                if maxc_new < cur_maxc - 1e-16:
                    z = z_new
                    a = a_new
                    accepted = True
                    break
        
        if not accepted:
            no_improve += 3  # Penalty for no progress
        
        it += 1
        if it % 100 == 0:
            elapsed = time.monotonic() - t0
            print(f"  polish it={it} score={2*n*cur_maxc:.9f} best={best_score:.9f} "
                  f"tau={tau:.2e} time={elapsed:.0f}s")
    
    return best_a
