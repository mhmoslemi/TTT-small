"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Parameterized optimization (spline/Gaussian mixtures) for good starts
2. Multiplicative smoothed-max descent (MMD) with adaptive beta
3. Intensive local search (coordinate, pair hit-and-run, block scaling)
4. Multi-length strategy
"""

import numpy as np
import time

def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L

def _conv_full(a, b):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)

def _conv_valid_front(w, p):
    lenw, lenp = len(w), len(p)
    if lenp == 0:
        return np.array([], dtype=float)
    L = _next_pow2(max(lenw, 2 * lenp))
    fw = np.fft.rfft(w, L)
    fp = np.fft.rfft(p, L)
    out = np.fft.irfft(fw * np.conjugate(fp), L)
    return out[:lenp].astype(float, copy=False)

def _sanitize_vec(p):
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return p / s

def _exact_score(p):
    p = _sanitize_vec(p)
    if p is None:
        return float('inf')
    m = p.size
    c = _conv_full(p, p)
    if c.size == 0:
        return float('inf')
    c = np.where(np.isfinite(c), c, 0.0)
    return float(2.0 * m * np.max(c))

def _soft_weights(d, beta):
    d = np.asarray(d, dtype=float)
    n = d.size
    if n == 0:
        return np.array([], dtype=float)
    d = np.where(np.isfinite(d), d, 0.0)
    if beta <= 1e-8:
        return np.full(n, 1.0 / n)
    shift = float(np.max(d))
    if not np.isfinite(shift):
        return np.full(n, 1.0 / n)
    e = np.exp(beta * (d - shift))
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return e / s

    p = _sanitize_vec(p)
    if p is None:
        return float('inf'), float('inf'), None, 0.0
    m = p.size
    c = _conv_full(p, p)
    if c.size == 0:
        return float('inf'), float('inf'), np.array([0.0]), 0.0
    c = np.where(np.isfinite(c), c, 0.0)

def _resample(arr, m):

    a = _sanitize_vec(arr)

    out = np.interp(x_new, x_old, a)

def _param_optimize(n, K, rng, iters=500):

            params[max(1, K//3):] = rng.uniform(0.2, 0.8)
        
        # Use Nelder-Mead or Powell
        from scipy.optimize import minimize
        try:

"""
Minimize 2*n*max(conv(a,a))/sum(a)^2 using:
1. Parameterized spline optimization for diverse high-quality starts
2. Multiplicative smoothed-max descent (MMD) with adaptive beta and momentum
3. Gradient-guided focused local search (active-lag exploitation)
4. Multi-length strategy with careful time allocation
"""

import numpy as np
import time
from scipy.optimize import minimize as scipy_minimize


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b):
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)


def _conv_valid_front(w, p):
    lenw, lenp = len(w), len(p)
    if lenp == 0:
        return np.array([], dtype=float)
    L = _next_pow2(max(lenw, 2 * lenp))
    fw = np.fft.rfft(w, L)
    fp = np.fft.rfft(p, L)
    out = np.fft.irfft(fw * np.conjugate(fp), L)
    return out[:lenp].astype(float, copy=False)


def _sanitize_vec(p):
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return p / s


def _exact_score(p):
    p = _sanitize_vec(p)
    if p is None:
        return float('inf')
    m = p.size
    c = _conv_full(p, p)
    if c.size == 0:
        return float('inf')
    c = np.where(np.isfinite(c), c, 0.0)
    return float(2.0 * m * np.max(c))


def _soft_weights(d, beta):
    d = np.asarray(d, dtype=float)
    n = d.size
    if n == 0:
        return np.array([], dtype=float)
    d = np.where(np.isfinite(d), d, 0.0)
    if beta <= 1e-8:
        return np.full(n, 1.0 / n)
    shift = float(np.max(d))
    if not np.isfinite(shift):
        return np.full(n, 1.0 / n)
    e = np.exp(beta * (d - shift))
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n)
    return e / s


def _mmd_iteration(p, beta, alpha, floor):
    """One MMD iteration: compute gradient, do multiplicative update.
    Returns (new_p, score, smooth_val, grad_h, maxh)."""
    m = p.size
    c = _conv_full(p, p)
    c = np.where(np.isfinite(c), c, 0.0)
    d = m * c
    exact = float(2.0 * max(0.0, np.max(d)))
    w = _soft_weights(d, beta)
    h = _conv_valid_front(w, p)
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.maximum(h, 0.0)
    maxh = float(np.max(h)) if h.size else 0.0
    smooth = float(np.dot(w, d))

    if maxh <= 1e-15:
        return p, exact, smooth, h, maxh

    h = h / maxh
    dec = np.exp(-alpha * h)
    p_new = p * dec
    p_new = np.maximum(p_new, floor)
    s = float(p_new.sum())
    if s <= 0 or not np.isfinite(s):
        return p, exact, smooth, h, maxh
    p_new = p_new / s
    return p_new, exact, smooth, h, maxh


def _mmd_run(p0, rng, time_limit, beta_start=20.0, beta_end=3000.0, use_momentum=True):
    """MMD with adaptive beta and optional momentum."""
    p = _sanitize_vec(p0)
    if p is None:
        return None, float('inf')

    m = int(p.size)
    floor = 1e-18 / m
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_exact = _exact_score(p)
    cur_exact = best_exact

    beta = float(beta_start)
    alpha = 0.15
    deadline = time.monotonic() + float(time_limit)
    stall = 0
    it = 0

    # Momentum
    h_prev = np.zeros(m) if use_momentum else None
    mom = 0.3 if use_momentum else 0.0

    # Adaptive beta tracking
    last_active = -1
    stable_count = 0

    while time.monotonic() < deadline:
        m_size = p.size
        c = _conv_full(p, p)
        c = np.where(np.isfinite(c), c, 0.0)
        d = m_size * c
        exact = float(2.0 * max(0.0, np.max(d)))
        w = _soft_weights(d, beta)
        h = _conv_valid_front(w, p)
        h = np.where(np.isfinite(h), h, 0.0)
        h = np.maximum(h, 0.0)
        maxh = float(np.max(h)) if h.size else 0.0
        smooth = float(np.dot(w, d))

        # Track active lag for adaptive beta
        active_lag = int(np.argmax(w)) if w.size > 0 else 0
        if active_lag == last_active:
            stable_count += 1
        else:
            stable_count = 0
        last_active = active_lag

        if stable_count > 200 and beta < beta_end:
            beta = min(beta_end, beta * 1.3)
            stable_count = 0

        if maxh <= 1e-15:
            # Perturb
            noise = rng.normal(0, 0.02, m)
            p_try = _sanitize_vec(p * np.exp(noise))
            if p_try is not None:
                p_try = np.maximum(p_try, floor)
                p_try = p_try / p_try.sum()
                e_try = _exact_score(p_try)
                if np.isfinite(e_try) and e_try <= best_exact * 1.1:
                    p = p_try
                    cur_exact = e_try
            stall += 1
        else:
            h = h / maxh

            # Apply momentum
            if use_momentum and h_prev is not None:
                h_eff = (1.0 - mom) * h + mom * h_prev
                h_eff = np.maximum(h_eff, 0)
                h_max_eff = float(np.max(h_eff))
                if h_max_eff > 1e-15:
                    h_eff = h_eff / h_max_eff
            else:
                h_eff = h

            h_prev = h.copy() if use_momentum else h_prev

            # Try multiple step sizes
            best_trial = None
            best_trial_smooth = smooth
            for a in (0.3, 0.5, 0.8, 1.2, 1.8):
                dec = np.exp(-a * h_eff)
                p_try = p * dec
                p_try = np.maximum(p_try, floor)
                s_try = float(p_try.sum())
                if s_try <= 0 or not np.isfinite(s_try):
                    continue
                p_try = p_try / s_try
                c_try = _conv_full(p_try, p_try)
                c_try = np.where(np.isfinite(c_try), c_try, 0.0)
                d_try = m_size * c_try
                e_try = float(2.0 * max(0.0, np.max(d_try)))
                w_try = _soft_weights(d_try, beta)
                s2 = float(np.dot(w_try, d_try))
                if s2 + 1e-12 < best_trial_smooth:
                    best_trial = p_try
                    best_trial_smooth = s2
                    best_trial_exact = e_try

            if best_trial is not None:
                p = best_trial
                cur_exact = best_trial_exact
                stall = 0
                if cur_exact < best_exact:
                    best = p.copy()
                    best_exact = cur_exact
            else:
                stall += 1
                alpha = max(0.01, alpha * 0.7)
                if stall > 25:
                    noise = rng.normal(0, min(0.3, 0.02 + 0.01 * (stall // 10)), m)
                    p_try = _sanitize_vec(p * np.exp(noise))
                    if p_try is not None:
                        p_try = np.maximum(p_try, floor)
                        p_try = p_try / p_try.sum()
                        e_try = _exact_score(p_try)
                        if np.isfinite(e_try) and e_try <= best_exact * 1.08:
                            p = p_try
                            cur_exact = e_try
                            alpha = 0.1
                if stall > 100:
                    p = best.copy()
                    cur_exact = best_exact
                    beta = max(beta_start, beta * 0.5)
                    alpha = 0.15
                    stall = 0

        it += 1
        if it % 50 == 0:
            beta = min(beta_end, beta * 1.05)

    return best, best_exact


def _param_search(n, rng, time_limit):
    """Optimize spline parameters to get good starting sequences."""
    t = np.linspace(0, 1, n)
    best_arr = None
    best_score = float('inf')
    deadline = time.monotonic() + float(time_limit)

    for K in (4, 6, 8, 12, 16):
        if time.monotonic() > deadline:
            break
        t_ctrl = np.linspace(0, 1, K)

        def obj(params):
            heights = np.maximum(0, params)
            a = np.interp(np.arange(n), t_ctrl, heights)
            s = a.sum()
            if s < 0.01:
                return 1e10
            a = a / s
            c = _conv_full(a, a)
            return 2.0 * n * float(np.max(c))

        for trial in range(3):
            if time.monotonic() > deadline:
                break
            r = rng.random()
            if r < 0.4:
                params0 = np.ones(K) * 1.0
                params0[:K // 2] *= rng.uniform(2.0, 5.0)
            elif r < 0.7:
                params0 = np.maximum(0.1, rng.normal(1.0, 0.8, K))
            else:
                params0 = np.ones(K)
                half = K // 2
                params0[:half] = rng.uniform(1.5, 4.0, half)
                params0[half:] = rng.uniform(0.2, 1.0, K - half)

            try:
                res = scipy_minimize(obj, params0, method='Nelder-Mead',
                                     options={'maxiter': 300, 'xatol': 1e-7, 'fatol': 1e-9})
                if res.fun < best_score:
                    heights = np.maximum(0, res.x)
                    a = np.interp(np.arange(n), t_ctrl, heights)
                    a = _sanitize_vec(a)
                    if a is not None:
                        best_score = float(res.fun)
                        best_arr = a
            except Exception:
                pass

    return best_arr, best_score


def _focused_local_search(p0, rng, time_limit):
    """Local search exploiting the active (max) lag of the convolution."""
    p = _sanitize_vec(p0)
    if p is None:
        return None, float('inf')

    n = int(p.size)
    floor = 1e-16 / n
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_score = _exact_score(p)
    cur_score = best_score
    deadline = time.monotonic() + float(time_limit)
    stagnant = 0
    it = 0

    while time.monotonic() < deadline:
        # Compute convolution and find active lag
        c = _conv_full(p, p)
        c = np.where(np.isfinite(c), c, 0.0)
        k_star = int(np.argmax(c))

        # Gradient of c[k_star] w.r.t. p_j:
        # (p*p)[k_star] = sum_i p[i] * p[k_star - i]
        # d/dp[j] = p[k_star - j] (if in bounds)
        grad = np.zeros(n)
        for j in range(n):
            idx = k_star - j
            if 0 <= idx < n:
                grad[j] = p[idx]

        # Top contributors (should decrease) and bottom (should increase)
        top_indices = np.argsort(grad)[-min(20, n):][::-1]
        bottom_indices = np.argsort(grad)[:min(20, n)]

        improved = False
        for j in top_indices:
            if time.monotonic() > deadline:
                break
            for l in bottom_indices:
                if j == l:
                    continue
                # Try: p[j] decreases, p[l] increases (keep sum)
                for frac in (0.1, 0.3, 0.5):
                    delta = p[j] * frac
                    if delta < floor:
                        continue
                    cand = p.copy()
                    cand[j] -= delta
                    cand[l] += delta
                    cand = _sanitize_vec(cand)
                    if cand is None:
                        continue
                    sc = _exact_score(cand)
                    if sc < cur_score - 1e-12:
                        p = cand
                        cur_score = sc
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break

        if improved:
            if cur_score < best_score:
                best = p.copy()
                best_score = cur_score
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 30:
                # Random perturbation
                noise = rng.normal(0, 0.01, n)
                p_try = _sanitize_vec(p * np.exp(noise))
                if p_try is not None:
                    p_try = np.maximum(p_try, floor)
                    p_try = p_try / p_try.sum()
                    sc_try = _exact_score(p_try)
                    if sc_try <= best_score * 1.03:
                        p = p_try
                        cur_score = sc_try
                    else:
                        p = best.copy()
                        cur_score = best_score
                stagnant = 0
            if stagnant > 200:
                p = best.copy()
                cur_score = best_score
                stagnant = 0

        it += 1

    return best, best_score


def _coordinate_local_search(p0, rng, time_limit):
    """Block-based coordinate search."""
    p = _sanitize_vec(p0)
    if p is None:
        return None, float('inf')

    n = int(p.size)
    floor = 1e-16 / n
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_score = _exact_score(p)
    cur_score = best_score
    deadline = time.monotonic() + float(time_limit)
    stagnant = 0

    while time.monotonic() < deadline:
        # Random block modification
        r = rng.random()
        if r < 0.3:
            w = int(rng.integers(1, max(2, n // 100 + 1)))
        elif r < 0.7:
            w = int(rng.integers(max(2, n // 100), max(3, n // 20 + 1)))
        else:
            low = max(3, n // 20)
            high = max(low + 1, n // 5)
            w = int(rng.integers(low, high + 1))
        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))

        improved = False
        for f in (0.5, 0.7, 0.85, 1.15, 1.3, 1.5):
            cand = p.copy()
            cand[start:start + w] *= f
            cand = _sanitize_vec(cand)
            if cand is None:
                continue
            sc = _exact_score(cand)
            if sc < cur_score - 1e-12:
                p = cand
                cur_score = sc
                improved = True
                break

        if improved:
            if cur_score < best_score:
                best = p.copy()
                best_score = cur_score
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 80:
                noise = rng.normal(0, 0.03, n)
                p_try = _sanitize_vec(p * np.exp(noise))
                if p_try is not None:
                    p_try = np.maximum(p_try, floor)
                    p_try = p_try / p_try.sum()
                    sc = _exact_score(p_try)
                    if sc <= best_score * 1.05:
                        p = p_try
                        cur_score = sc
                else:
                    p = best.copy()
                    cur_score = best_score
                stagnant = 0
            if stagnant > 400:
                p = best.copy()
                cur_score = best_score
                stagnant = 0

    return best, best_score


def _resample(arr, m):
    if m <= 0:
        return None
    a = _sanitize_vec(arr)
    if a is None:
        return None
    n = a.size
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m, dtype=float)
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0)
    out = out + 1e-12
    return _sanitize_vec(out)


def _two_block_init(n, rng):
    """Generate a good two-block initialization."""
    best = None
    best_sc = float('inf')
    for a in (0.3, 0.4, 0.5, 0.6, 0.7):
        idx = int(round(a * n))
        if idx <= 0 or idx >= n:
            continue
        for ratio in (0.3, 0.5, 0.8, 1.3, 2.0, 3.0):
            p = np.ones(n)
            p[idx:] = ratio
            p = _sanitize_vec(p)
            if p is None:
                continue
            sc = _exact_score(p)
            if sc < best_sc:
                best_sc = sc
                best = p.copy()
    return best


def _initial_sequence(m, rng, mode):
    """Generate a diverse initial sequence."""
    t = (np.arange(m, dtype=float) + 0.5) / float(m)

    if mode == 0:
        z = 1.0
        for k in range(1, 10):
            amp = float(rng.normal(0, 1.0 / (k + 1)))
            ph = float(rng.uniform(0, 2 * np.pi))
            z += amp * np.cos(2 * np.pi * k * t + ph)
        p = np.log1p(np.exp(np.clip(z, -50, 50)))
    elif mode == 1:
        a = float(rng.uniform(0.1, 3.0))
        b = float(rng.uniform(0.1, 3.0))
        tt = np.clip(t, 1e-8, 1 - 1e-8)
        p = tt ** a * (1 - tt) ** b
    elif mode == 2:
        K = int(rng.integers(1, 4))
        p = np.zeros(m)
        centers = np.sort(rng.uniform(0, 1, K))
        sigmas = rng.uniform(0.02, 0.3, K)
        weights = rng.uniform(0.5, 2.0, K)
        for ii in range(K):
            p += weights[ii] * np.exp(-((t - centers[ii]) / sigmas[ii]) ** 2)
    elif mode == 3:
        K = int(rng.integers(3, 8))
        knots = np.sort(rng.uniform(0, 1, K))
        heights = rng.lognormal(0, 1, K)
        p = np.interp(t, knots, heights)
    else:
        c = float(rng.uniform(0.2, 0.8))
        k_val = float(rng.uniform(5, 25))
        p = 1.0 / (1.0 + np.exp(-np.clip(k_val * (t - c), -50, 50)))
        p *= np.exp(-((t - c) / 0.25) ** 2)

    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0, 1000) + 1e-10
    s = p.sum()
    if s <= 0 or not np.isfinite(s):
        p = np.ones(m, dtype=float)
    return p / p.sum()


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Search for a non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
    Returns a list of floats.
    """
    start_time = time.monotonic()
    deadline = start_time + max(60.0, float(budget_s) - 30.0)

    try:
        iseed = int(seed)
    except Exception:
        iseed = 12345
    rng = np.random.default_rng((iseed % 1000000007) + 42)

    all_cands = []  # (score, length, array)

    def add_cand(arr, sc):
        a = _sanitize_vec(arr)
        if a is None or not np.isfinite(sc):
            return
        all_cands.append((float(sc), int(a.size), a.copy()))
        all_cands.sort(key=lambda x: x[0])
        if len(all_cands) > 20:
            all_cands.pop()

    # Get parent
    parent_arr = None
    try:
        parent_raw = height_sequence_1
        parent_arr = _sanitize_vec(np.asarray(parent_raw, dtype=float))
    except (NameError, Exception):
        pass

    if parent_arr is not None:
        sc_p = _exact_score(parent_arr)
        add_cand(parent_arr, sc_p)
        print(f"Parent score: {sc_p:.6f}, length: {len(parent_arr)}")

    # ---- Phase 1: Quick exploration at moderate length (90s) ----
    small_n = 1024
    phase1_pl = parent_arr
    if phase1_pl is None or len(phase1_pl) != small_n:
        tb = _two_block_init(small_n, rng)
        if tb is not None:
            phase1_pl = tb
    if phase1_pl is None:
        phase1_pl = _initial_sequence(small_n, rng, 1)

    rem = deadline - time.monotonic()
    if rem > 120:
        # Parameter search
        print("Phase 1: Parameter search...")
        t_start = time.monotonic()
        p_param, sc_param = _param_search(small_n, rng, min(40, rem - 60))
        if p_param is not None:
            add_cand(p_param, sc_param)
            print(f"  Param search best: {sc_param:.6f}")
        elapsed = time.monotonic() - t_start
        print(f"  Param search took {elapsed:.1f}s")

        # Quick MMD on best starts
        starts = [phase1_pl]
        if p_param is not None:
            starts.append(p_param)
        starts.append(_initial_sequence(small_n, rng, 1))
        starts.append(_initial_sequence(small_n, rng, 2))

        rem = deadline - time.monotonic()
        if rem > 40:
            per_start = max(5, (rem - 20) // len(starts))
            for sp in starts:
                if time.monotonic() > deadline - 20:
                    break
                p, sc = _mmd_run(sp, rng, per_start, beta_start=15, beta_end=500)
                if p is not None:
                    add_cand(p, sc)

    print(f"Phase 1 done. Best so far: {all_cands[0][0]:.6f}" if all_cands else "Phase 1 done.")

    # ---- Phase 2: Main optimization at target length ----
    # Choose target lengths to try
    target_lengths = [2048, 3072, 4096]

    # Build starts for target lengths
    rem = deadline - time.monotonic()
    if rem < 300:
        # Not enough time for full pipeline, just do MMD on best available
        best_start = all_cands[0][2] if all_cands else _initial_sequence(2048, rng, 1)
        target_n = len(best_start)
        p, sc = _mmd_run(best_start, rng, max(30, rem - 40), beta_start=50, beta_end=2000)
        if p is not None:
            add_cand(p, sc)
        final_arr = all_cands[0][2] if all_cands else best_start
    else:
        # Multi-length strategy
        # Allocate time: more time to 3072 (proven good length)
        if rem > 1500:
            time_alloc = {2048: 200, 3072: 1000, 4096: 400}
        elif rem > 1000:
            time_alloc = {2048: 100, 3072: 600, 4096: 200}
        else:
            time_alloc = {3072: max(200, rem - 200)}
            target_lengths = [3072]

        for target_n in target_lengths:
            if target_n not in time_alloc:
                continue
            if time.monotonic() > deadline - 100:
                break

            t_alloc = min(time_alloc[target_n], deadline - time.monotonic() - 60)
            if t_alloc < 60:
                t_alloc = max(30, t_alloc)

            print(f"\nPhase 2: MMD at n={target_n}, time budget={t_alloc:.0f}s")

            # Build diverse starts for this length
            starts_nl = []

            # From parent
            if parent_arr is not None:
                sp = _resample(parent_arr, target_n)
                if sp is not None:
                    sp = _sanitize_vec(sp * np.exp(rng.normal(0, 0.02, target_n)))
                    starts_nl.append(sp)

            # From best small-n candidates
            for item in all_cands[:3]:
                sp = _resample(item[2], target_n)
                if sp is not None:
                    sp = _sanitize_vec(sp * np.exp(rng.normal(0, 0.03, target_n)))
                    starts_nl.append(sp)

            # Two-block
            tb = _two_block_init(target_n, rng)
            if tb is not None:
                starts_nl.append(tb)

            # Random initializations
            for mode in (1, 2, 4):
                starts_nl.append(_initial_sequence(target_n, rng, mode))

            # Score and sort starts
            scored = []
            for sp in starts_nl:
                if sp is None:
                    continue
                sc = _exact_score(sp)
                if np.isfinite(sc):
                    scored.append((sc, sp))
            scored.sort(key=lambda x: x[0])

            print(f"  Best start score: {scored[0][0]:.6f}" if scored else "  No valid starts!")

            # Run MMD on top starts
            if scored:
                n_starts = min(3, len(scored))
                fracs = [0.5, 0.3, 0.2]
                for i in range(n_starts):
                    if time.monotonic() > deadline - 60:
                        break
                    frac = fracs[i] if i < len(fracs) else 0.15
                    t_run = t_alloc * frac
                    t_run = min(t_run, deadline - time.monotonic() - 30)
                    if t_run < 20:
                        break

                    sp = scored[i][1]
                    beta_s = 20.0 + 10.0 * i
                    beta_e = 1500.0 + 500.0 * i

                    p, sc = _mmd_run(sp, rng, t_run, beta_start=beta_s, beta_end=beta_e)
                    if p is not None:
                        add_cand(p, sc)
                        print(f"  MMD start {i}: score={sc:.6f}")

    # ---- Phase 3: Local search on best candidate ----
    if all_cands:
        best_arr = all_cands[0][2]
        best_sc = all_cands[0][0]
        print(f"\nPhase 3: Local search on best (score={best_sc:.6f}, n={len(best_arr)})")

        rem = deadline - time.monotonic()
        if rem > 100:
            # Focused local search (exploits active lag)
            t_focus = min(rem * 0.4, 300)
            p, sc = _focused_local_search(best_arr, rng, t_focus)
            if p is not None:
                add_cand(p, sc)
                if sc < best_sc:
                    best_sc = sc
                    best_arr = p
                    print(f"  Focused search: {sc:.6f}")

            rem = deadline - time.monotonic()
            if rem > 80:
                # Coordinate/local block search
                t_coord = min(rem * 0.5, 250)
                p, sc = _coordinate_local_search(best_arr, rng, t_coord)
                if p is not None:
                    add_cand(p, sc)
                    if sc < best_sc:
                        best_sc = sc
                        best_arr = p
                        print(f"  Coordinate search: {sc:.6f}")

            rem = deadline - time.monotonic()
            if rem > 60:
                # One more MMD pass with high beta
                t_mmd2 = min(rem * 0.6, 200)
                p, sc = _mmd_run(best_arr, rng, t_mmd2, beta_start=200, beta_end=5000)
                if p is not None:
                    add_cand(p, sc)
                    if sc < best_sc:
                        best_sc = sc
                        best_arr = p
                        print(f"  Final MMD: {sc:.6f}")

            rem = deadline - time.monotonic()
            if rem > 60:
                # Another focused search
                t_f2 = min(rem * 0.5, 150)
                p, sc = _focused_local_search(best_arr, rng, t_f2)
                if p is not None:
                    add_cand(p, sc)
                    if sc < best_sc:
                        best_sc = sc
                        best_arr = p

    # ---- Phase 4: Final evaluation ----
    if all_cands:
        best_arr = all_cands[0][2]
        best_sc = all_cands[0][0]
    else:
        best_arr = np.ones(1024, dtype=float)
        best_sc = _exact_score(best_arr)

    # Try a few candidates with the official evaluator
    evaluated = 0
    final_list = [float(x) for x in best_arr]
    for item in all_cands[:5]:
        if time.monotonic() > deadline - 5:
            break
        if evaluated >= 3:
            break
        lst = [float(x) for x in item[2]]
        try:
            sc = float(evaluate_sequence(lst))
        except Exception:
            sc = item[0]
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            final_list = lst
        evaluated += 1

    print(f"\nFinal score: {best_sc:.6f}, length: {len(final_list)}")
    elapsed = time.monotonic() - start_time
    print(f"Total time: {elapsed:.1f}s")

    # Ensure valid output
    out = np.asarray(final_list, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0, 1000)
    s = float(out.sum())
    if s <= 0 or not np.isfinite(s):
        out = np.ones(1024, dtype=float)
    else:
        out = out / s  # Doesn't change the score

    return [float(x) for x in out]
