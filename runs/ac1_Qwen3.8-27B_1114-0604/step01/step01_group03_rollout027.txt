"""
Search for non-negative step heights minimizing 2*n*max(conv(a,a))/(sum(a))^2.
Uses scipy L-BFGS-B with soft-max objective in log-parameterization, combined
with custom peak-shaving and local search refinement.

import time
import math
import numpy as np
import scipy.optimize


def _norm1(a):
    """Normalize to sum=1, non-negative."""
    a = np.asarray(a, dtype=np.float64).copy()
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-300:
        return np.ones_like(a) / len(a)
    return a / s


def _conv_fft(a, L, m):
    """Self-convolution via FFT. a: length n, L: FFT size >= 2n-1, m=2n-1."""
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:m]
    np.maximum(c, 0.0, out=c)
    return c


def _score_internal(a):
    """Compute 2*n*max(conv)/(sum)^2."""
    n = len(a)
    s = a.sum()
    if s < 0.01:
        return float('inf')
    L = 1 << (2 * n - 1).bit_length()
    m = 2 * n - 1
    b = _conv_fft(a, L, m)
    return float(2.0 * n * b.max() / (s * s))


def _exact_score(seq):
    """Call evaluate_sequence if available, else internal."""

    except NameError:

    if val is None:

    return float(val)


def _soft_obj_grad(x, n, L, L2, m, T):
    """
    Soft-max objective and gradient w.r.t. log-parameter x.
    a = softmax(x), obj = 2*n*S(b) where S = soft_max(b; T).
    """
    x_sh = x - x.max()
    a = np.exp(x_sh)
    a_sum = a.sum()
    if a_sum < 1e-300:
        return 0.0, np.zeros(n)
    a /= a_sum

    b = _conv_fft(a, L, m)
    bmax = float(b.max())
    z = (b - bmax) / T
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s_e = float(e.sum())
    if s_e < 1e-300:
        return 2.0 * n * bmax, np.zeros(n)
    w = e / s_e

    obj = 2.0 * n * (bmax + T * math.log(s_e))

    # Gradient: grad_a[i] = 4*n * sum_{k=0}^{n-1} w[i+k] * a[k]
    # = cross-correlation of a and w at lag i
    # Using FFT: ifft(rfft(a, L2) * conj(rfft(w, L2)))
    A = np.fft.rfft(a, L2)
    W = np.fft.rfft(w, L2)
    cross = np.fft.irfft(A * np.conj(W), L2)
    grad_a = 4.0 * n * cross[:n]

    # Chain rule: grad_x = grad_a * a - a * (grad_a . a)
    dot = float(np.dot(grad_a, a))
    grad_x = grad_a * a - a * dot

    return obj, grad_x


def _scipy_phase(a0, n, L, L2, m, T_start, T_end, max_time, rng):
    """Run scipy L-BFGS-B with increasing temperature."""
    deadline = time.monotonic() + max_time

    best_a = a0.copy()

    # Multi-stage temperature schedule

    Ts = np.linspace(T_start, T_end, n_stages)
    
    for si, T in enumerate(Ts):
        stage_time = (deadline - time.monotonic()) / (n_stages - si)
        if stage_time < 1.0:
            break
        
        x_opt = x0
        for trial in range(2):

        # Update x0 to best found so far

def _soft_obj_grad_wrapped(x, n, L, L2, m, T):
    """Wrapper for scipy (returns (f, grad) tuple)."""
    f, g = _soft_obj_grad(x, n, L, L2, m, T)
    if not np.isfinite(f):
        return 1e10, np.zeros(n)
    g = np.where(np.isfinite(g), g, 0.0)
    return f, g


def _softmax(x):
    x = x - x.max()
    e = np.exp(x)
    s = e.sum()
    if s < 1e-300:
        return np.ones_like(x) / len(x)
    return e / s


def _peak_shave(a, n, L, m, max_time, rng):
    """Iteratively reduce peaks in the convolution."""
    deadline = time.monotonic() + max_time
    a = _norm1(a.copy())
    best_a = a.copy()
    best_score = _score_internal(a)
    
    K = 5  # number of peaks to target
    step_decay = 1.0
    
    for it in range(5000):
        if time.monotonic() > deadline - 1.0:
            break
        
        b = _conv_fft(a, L, m)
        score = 2.0 * n * b.max()  # since sum(a)=1
        
        if score < best_score - 1e-12:
            best_score = score
            best_a = a.copy()
        
        # Find top-K peak locations
        top_k = np.argsort(b)[-K:]
        
        # Accumulate subgradient from top peaks
        grad = np.zeros(n)
        for j in top_k:
            # db[j]/da[i] = 2*a[j-i] if 0<=j-i<=n-1
            lo = max(0, j - (n - 1))
            hi = min(n - 1, j)
            if lo <= hi:
                idx = np.arange(lo, hi + 1)
                grad[idx] += 2.0 * a[j - idx]
        
        # Normalize gradient
        gn = np.linalg.norm(grad)
        if gn < 1e-15:
            break
        grad /= gn
        
        # Try multiple step sizes
        improved = False
        for step in [0.01 * step_decay, 0.04 * step_decay, 0.1 * step_decay, 0.2 * step_decay]:
            a_new = a - step * grad
            a_new = _norm1(a_new)
            sc_new = _score_internal(a_new)
            if sc_new < score - 1e-12:
                a = a_new
                improved = True
                step_decay = min(step_decay * 1.3, 5.0)
                break
        
        if not improved:
            step_decay = max(0.1, step_decay * 0.7)
            # Occasionally add random perturbation
            if it % 50 == 49:
                pert = rng.normal(0, 0.01, n)
                a = _norm1(a + pert)
    
    # Final: if best_a is better, use it
    if _score_internal(best_a) < _score_internal(a):
        a = best_a
    return a, _score_internal(a)


def _local_perturb(a, n, L, m, max_time, rng):

    deadline = time.monotonic() + max_time
    a = _norm1(a.copy())
    best_a = a.copy()
    best_score = _score_internal(a)
    
    for it in range(100000):
        if time.monotonic() > deadline - 1.0:

        a_new[idx] *= np.exp(rng.normal(0, 0.3, n_pert))

        if sc < best_score - 1e-12:
            best_score = sc
            best_a = a_new.copy()

        res = scipy.optimize.minimize(
            _soft_obj_grad_wrapped,
            x0,
            args=(n, L, L2, m, T_sharp),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': 5000, 'maxfun': 100000,
                     'ftol': 1e-15, 'gtol': 1e-12, 'iprint': 0}
        )
        a_p4 = _softmax(res.x)
        sc_p4 = _score_internal(a_p4)
        if sc_p4 < best_score:
            best_score = sc_p4
            best_a = a_p4
        print(f"  After sharp scipy: {best_score:.8f}")
    
    return best_a, best_score


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Main entry point. Searches for the best sequence.
    """
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget < 10:
        budget = 2400.0
    
    start_time = time.monotonic()
    hard_deadline = start_time + max(10.0, budget - 3.0)
    
    rng = np.random.default_rng(seed)
    
    print(f"Starting search with budget={budget:.0f}s, seed={seed}")
    
    # Try multiple n values
    # Weight time allocation: more time for promising n values
    n_candidates = [6121, 5000, 7000, 4000, 8000]
    
    # First, quickly evaluate parent if available
    global_best_a = None
    global_best_score = float('inf')
    global_best_n = 0
    
    # Evaluate parent
    try:
        base = height_sequence_1
        arr = np.asarray(base, dtype=np.float64).ravel()
        n_parent = len(arr)
        sc_parent = _score_internal(arr)
        print(f"Parent sequence: n={n_parent}, score={sc_parent:.8f}")
        global_best_a = arr.copy()
        global_best_score = sc_parent
        global_best_n = n_parent
        # Remove parent n from candidates to avoid duplication
        n_candidates = [x for x in n_candidates if x != n_parent]
    except (NameError, Exception) as e:
        print(f"No parent available: {e}")
    
    # Allocate time: give more to parent n, some to others
    # Strategy: spend 60% on parent n (or best n), 40% on others
    if global_best_n > 0:
        main_n = global_best_n
        main_time = (hard_deadline - time.monotonic()) * 0.5
    else:
        main_n = 6121
        main_time = (hard_deadline - time.monotonic()) * 0.4
    
    other_time = (hard_deadline - time.monotonic()) * 0.35
    
    print(f"Main n={main_n}, time={main_time:.0f}s")
    
    # Optimize main n
    a_main, sc_main = _try_n(main_n, rng, main_time)
    if sc_main < global_best_score:
        global_best_score = sc_main
        global_best_a = a_main
        global_best_n = main_n
    
    # Try other n values (less time each)
    remaining = hard_deadline - time.monotonic()
    n_others = [x for x in n_candidates if x != main_n][:3]
    for n_try in n_others:
        t_alloc = min(remaining * 0.25, 200.0)
        if t_alloc < 30:
            break
        print(f"\nTrying n={n_try} with {t_alloc:.0f}s")
        a_try, sc_try = _try_n(n_try, rng, t_alloc)
        if sc_try < global_best_score:
            global_best_score = sc_try
            global_best_a = a_try
            global_best_n = n_try
        remaining = hard_deadline - time.monotonic()
    
    # Final polish on the best
    remaining = hard_deadline - time.monotonic()
    if remaining > 30 and global_best_a is not None:
        print(f"\nFinal polish with {remaining:.0f}s on n={global_best_n}")
        n_best = global_best_n
        m = 2 * n_best - 1
        L = 1 << m.bit_length()
        L2 = 1 << (3 * n_best - 2).bit_length()
        
        # Peak shave
        t_ps = min(remaining * 0.4, 200.0)
        if t_ps > 5:
            a_ps, sc_ps = _peak_shave(global_best_a, n_best, L, m, t_ps, rng)
            if sc_ps < global_best_score:
                global_best_score = sc_ps
                global_best_a = a_ps
        
        # Sharp scipy
        remaining2 = hard_deadline - time.monotonic()
        if remaining2 > 20:
            T_fine = 1.0 / (500.0 * n_best)
            x0 = np.log(np.maximum(global_best_a, 1e-15))
            res = scipy.optimize.minimize(
                _soft_obj_grad_wrapped,
                x0,
                args=(n_best, L, L2, m, T_fine),
                method='L-BFGS-B',
                jac=True,
                options={'maxiter': 10000, 'maxfun': 200000,
                         'ftol': 1e-15, 'gtol': 1e-13, 'iprint': 0}
            )
            a_fine = _softmax(res.x)
            sc_fine = _score_internal(a_fine)
            if sc_fine < global_best_score:
                global_best_score = sc_fine
                global_best_a = a_fine
        
        # Final local perturb
        remaining3 = hard_deadline - time.monotonic()
        if remaining3 > 10:
            a_lp, sc_lp = _local_perturb(global_best_a, n_best, L, m, min(remaining3 - 5, 100.0), rng)
            if sc_lp < global_best_score:
                global_best_score = sc_lp
                global_best_a = a_lp
    
    print(f"\nFinal internal score: {global_best_score:.10f}")
    
    # Convert to list and verify with exact evaluator
    if global_best_a is None:
        seq = [1.0] * 6121
    else:
        seq = [float(max(0.0, x)) for x in global_best_a]
    
    # Verify with exact evaluator
    remaining = hard_deadline - time.monotonic()
    if remaining > 5:
        exact_sc = _exact_score(seq)
        print(f"Exact score: {exact_sc:.10f}")
    
    return seq
