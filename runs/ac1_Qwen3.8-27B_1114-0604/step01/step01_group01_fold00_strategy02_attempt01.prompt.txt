def evaluate_sequence(sequence: list[float]) -> float:
    """
    Evaluates a sequence of coefficients with enhanced security checks.
    Returns np.inf if the input is invalid.
    """
    # --- Security Checks ---

    # Verify that the input is a list
    if not isinstance(sequence, list):
        return np.inf

    # Reject empty lists
    if not sequence:
        return np.inf

    # Check each element in the list for validity
    for x in sequence:
        # Reject boolean types (as they are a subclass of int) and
        # any other non-integer/non-float types (like strings or complex numbers).
        if isinstance(x, bool) or not isinstance(x, (int, float)):
            return np.inf

        # Reject Not-a-Number (NaN) and infinity values.
        if np.isnan(x) or np.isinf(x):
            return np.inf

    # Convert all elements to float for consistency
    sequence = [float(x) for x in sequence]

    # Protect against negative numbers
    sequence = [max(0, x) for x in sequence]

    # Protect against numbers that are too large
    sequence = [min(1000.0, x) for x in sequence]

    n = len(sequence)
    b_sequence = np.convolve(sequence, sequence)
    max_b = max(b_sequence)
    sum_a = np.sum(sequence)

    # Protect against the case where the sum is too close to zero
    if sum_a < 0.01:
        return np.inf

    return float(2 * n * max_b / (sum_a**2))

def _next_pow2(x):
    """Smallest power of two >= x."""
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _conv_full(a, b):
    """Full real convolution, using FFT for larger inputs."""
    la = len(a)
    lb = len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    if la * lb <= 2000000:
        return np.asarray(np.convolve(a, b), dtype=float)
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout].astype(float, copy=False)


def _conv_valid_front(w, p):
    """
    Return v[j] = sum_i p[i] * w[j+i], j=0..len(p)-1.

    This is the 'valid' correlation needed for gradients of
    sum_j w_j * (p*p)[j].
    """
    lenw = len(w)
    lenp = len(p)
    if lenp == 0:
        return np.array([], dtype=float)
    # Need circular length at least lenw.
    L = _next_pow2(max(lenw, 2 * lenp))
    fw = np.fft.rfft(w, L)
    fp = np.fft.rfft(p, L)
    out = np.fft.irfft(fw * np.conjugate(fp), L)
    return out[:lenp].astype(float, copy=False)


def _sanitize_vec(p):
    """Clip finite non-negative vector and normalize to sum 1."""
    p = np.asarray(p, dtype=float)
    if p.ndim != 1 or p.size == 0:
        return None

    p = np.clip(p, 0.0, 1000.0)
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        return None
    return p / s

def _soft_weights(d, beta):
    """Softmax weights used for a differentiable max."""
    d = np.asarray(d, dtype=float)
    n = d.size
    if n == 0:
        return np.array([], dtype=float)
    d = np.where(np.isfinite(d), d, 0.0)
    if beta <= 1e-8:
        return np.full(n, 1.0 / n, dtype=float)
    shift = float(np.max(d))
    if not np.isfinite(shift):
        return np.full(n, 1.0 / n, dtype=float)
    e = np.exp(beta * (d - shift))
    s = float(e.sum())
    if s <= 0.0 or not np.isfinite(s):
        return np.full(n, 1.0 / n, dtype=float)
    return e / s


def _exact_smooth(p, beta):
    """Return (exact_score, smooth_max_d) for normalized p."""
    p = _sanitize_vec(p)
    if p is None:
        return float("inf"), float("inf")
    m = p.size
    c = _conv_full(p, p)
    if c.size == 0:
        return float("inf"), float("inf")
    c = np.where(np.isfinite(c), c, 0.0)
    d = m * c
    exact = float(2.0 * max(0.0, np.max(d))) if d.size else float("inf")
    w = _soft_weights(d, beta)
    smooth = float(np.dot(w, d))
    if not np.isfinite(smooth):
        smooth = exact
    return exact, smooth


def _gradient_metrics(p, beta):
    """Exact score, smooth d-max, gradient h, and max(h)."""
    p = _sanitize_vec(p)
    if p is None:
        return float("inf"), float("inf"), None, 0.0
    m = p.size
    c = _conv_full(p, p)
    if c.size == 0:
        return float("inf"), float("inf"), np.array([0.0]), 0.0
    c = np.where(np.isfinite(c), c, 0.0)
    d = m * c
    exact = float(2.0 * max(0.0, np.max(d)))
    w = _soft_weights(d, beta)
    h = _conv_valid_front(w, p)
    if h.size == 0:
        h = np.array([0.0], dtype=float)
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.maximum(h, 0.0)
    maxh = float(np.max(h)) if h.size else 0.0
    smooth = float(np.dot(w, d))
    return exact, smooth, h, maxh


def _smooth_array(p, samples=1.0):
    """Mild positive Gaussian smoothing, keeping sum 1."""
    p = _sanitize_vec(p)
    if p is None:
        return None
    s = float(samples)
    if s <= 0.0 or p.size < 5:
        return p
    rad = int(max(1, np.ceil(3.0 * s)))
    k = np.arange(-rad, rad + 1, dtype=float)
    kern = np.exp(-0.5 * (k / s) ** 2)
    kern /= kern.sum()
    out = np.convolve(p, kern, mode="same")
    return _sanitize_vec(out)


def _initial_sequence(m, rng, mode=None):
    """Create a diverse non-negative normalized start."""
    if m <= 0:
        return None
    t = (np.arange(m, dtype=float) + 0.5) / float(m)
    if mode is None:
        mode = int(rng.integers(0, 6))

    p = np.ones(m, dtype=float)

    if mode == 0:
        # Soft-plus of a random low-degree cosine series.
        z = 1.0
        for k in range(1, 13):
            amp = float(rng.normal(0.0, 1.0 / (k + 1)))
            ph = float(rng.uniform(0.0, 2.0 * np.pi))
            z = z + amp * np.cos(2.0 * np.pi * k * t + ph)
        p = np.log1p(np.exp(np.clip(z, -50.0, 50.0)))

    elif mode == 1:
        # Asymmetric beta-like density.
        a = float(rng.uniform(0.05, 3.0))
        b = float(rng.uniform(0.05, 3.0))
        tt = np.clip(t, 1e-8, 1.0 - 1e-8)
        p = np.power(tt, a) * np.power(1.0 - tt, b)

    elif mode == 2:
        # Sum of random Gaussian bumps.
        K = int(rng.integers(1, 4))
        p = np.zeros(m, dtype=float)
        centers = np.sort(rng.uniform(0.0, 1.0, K))
        sigmas = rng.uniform(0.02, 0.35, K)
        weights = rng.uniform(0.5, 2.0, K)
        for ii in range(K):
            c = float(centers[ii])
            s = max(1e-6, float(sigmas[ii]))
            w = float(weights[ii])
            p = p + w * np.exp(-((t - c) / s) ** 2)

    elif mode == 3:
        # Random piecewise-linear spline.
        K = int(rng.integers(3, 9))
        knots = np.sort(rng.uniform(0.0, 1.0, K))
        heights = rng.lognormal(0.0, 1.0, K)
        p = np.interp(t, knots, heights)

    elif mode == 4:
        # Random block level function.
        K = int(rng.integers(3, 16))
        edges = np.sort(rng.uniform(0.0, 1.0, K - 1))
        blocks = [0.0]
        blocks.extend(list(edges))
        blocks.append(1.0)
        heights = rng.lognormal(0.0, 1.0, K)
        p = np.zeros(m, dtype=float)
        for ii in range(K):
            s = max(0, int(round(blocks[ii] * m)))
            e = max(s + 1, int(round(blocks[ii + 1] * m)))
            e = min(m, e)
            p[s:e] = float(heights[ii])

    else:
        # Smooth asymmetric pulse / logistic hybrid.
        c = float(rng.uniform(0.2, 0.8))
        k = float(rng.uniform(5.0, 30.0))
        p = 1.0 / (1.0 + np.exp(-np.clip(k * (t - c), -50.0, 50.0)))
        p = p * np.exp(-((t - c) / 0.25) ** 2)

    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0.0, 1000.0)
    p = p + 1e-10
    s = float(p.sum())
    if s <= 0.0 or not np.isfinite(s):
        p = np.ones(m, dtype=float)
        s = float(p.sum())
    return p / s


def _two_block_best(m):
    """
    Quick deterministic 2-block(seed) by grid search.
    This gives an asymmetric low-score starting basin.

    best = None
    best_score = float("inf")

    for a in (0.25, 0.35, 0.45, 0.55, 0.65, 0.75, 0.85):
        idx = int(round(a * m))
        if idx <= 0 or idx >= m:
            continue

            p[idx:] = float(ratio)
            p = _sanitize_vec(p)
            if p is None:
                continue
            sc = _exact_score(p)
            if sc < best_score:
                best_score = sc
                best = p.copy()
    return best


def _resample(arr, m):
    """Resample a normalized vector to length m by linear interpolation."""
    if m <= 0:
        return None
    a = _sanitize_vec(arr)
    if a is None:
        return None
    n = a.size
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m, dtype=float)
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0.0)
    out = out + 1e-12
    return _sanitize_vec(out)

def _run_mmd(p0, rng, time_limit, beta_start=30.0, beta_end=1000.0, verbose=False):
    """
    Multiplicative descent on the probability simplex for a smoothed
    max convolution coefficient.  The exact score is tracked separately.
    """
    p = _sanitize_vec(p0)
    if p is None:
        return None, float("inf")
    if time_limit <= 0:
        return p, _exact_score(p)

    m = int(p.size)
    floor = 1e-18 / float(m)
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_exact = _exact_score(p)
    cur_exact, cur_smooth = _exact_smooth(p, beta_start)

    beta = float(max(1.0, beta_start))
    beta_max = float(max(beta, beta_end))
    alpha = 0.1
    deadline = time.monotonic() + float(time_limit)
    stall = 0
    it = 0
    last_print = -10000

    while time.monotonic() < deadline:
        cur_exact, cur_smooth, h, maxh = _gradient_metrics(p, beta)

        if not np.isfinite(maxh) or maxh <= 1e-15:
            # Degenerate direction: perturb.
            p_try = _sanitize_vec(p * np.exp(rng.normal(0.0, 0.05, m)))
            if p_try is not None:
                p_try = np.maximum(p_try, floor)
                p_try = p_try / p_try.sum()
                e, s2 = _exact_smooth(p_try, beta)
                if np.isfinite(e) and e <= best_exact * 1.05:
                    p = p_try
                    cur_exact, cur_smooth = e, s2
            stall += 1
        else:
            h = np.asarray(h, dtype=float)
            h = np.maximum(h, 0.0)
            h = h / maxh

            tests = []
            for mult in (0.5, 1.0, 2.0, 4.0):
                a = alpha * mult
                if 1e-4 <= a <= 20.0:
                    tests.append(float(a))
            if not tests:
                tests = [1e-4, 0.05]

            chosen = None
            ch_smooth = cur_smooth
            ch_exact = cur_exact
            ch_a = alpha

            for a in tests:
                dec = np.exp(-a * h)
                p_try = p * dec
                p_try = np.maximum(p_try, floor)
                s = float(p_try.sum())
                if s <= 0.0 or not np.isfinite(s):
                    continue
                p_try = p_try / s
                if not np.all(np.isfinite(p_try)):
                    continue
                e, s2 = _exact_smooth(p_try, beta)
                if not np.isfinite(e) or not np.isfinite(s2):
                    continue
                if s2 + 1e-11 * max(1.0, ch_smooth) < ch_smooth:
                    chosen = p_try
                    ch_smooth = s2
                    ch_exact = e
                    ch_a = a

            if chosen is not None:
                p = chosen
                cur_smooth = ch_smooth
                cur_exact = ch_exact
                alpha = ch_a
                stall = 0

                if cur_exact < best_exact:
                    best = p.copy()
                    best_exact = cur_exact
                    if verbose and (it - last_print >= 250):
                        last_print = it
                        print(f"mmd n={m} iter={it} best={best_exact:.6f}")

                # If the working point drifted badly, pull it back.
                if cur_exact > best_exact * 1.25 and rng.random() < 0.25:
                    p = best.copy()
                    cur_exact, cur_smooth = _exact_smooth(p, beta)
            else:
                stall += 1
                alpha = max(1e-4, alpha * 0.6)

                if stall >= 12:
                    std = 0.05 if stall < 40 else min(0.4, 0.05 + 0.005 * stall)
                    p_try = _sanitize_vec(p * np.exp(rng.normal(0.0, std, m)))
                    if p_try is not None:
                        p_try = np.maximum(p_try, floor)
                        p_try = p_try / p_try.sum()
                        e, s2 = _exact_smooth(p_try, beta)
                        if np.isfinite(e) and e <= best_exact * 1.05:
                            p = p_try
                            cur_exact, cur_smooth = e, s2
                        else:
                            p = best.copy()
                            cur_exact, cur_smooth = _exact_smooth(p, beta)
                    stall = 0

                if stall > 80:
                    p = best.copy()
                    cur_exact, cur_smooth = _exact_smooth(p, beta)
                    alpha = 0.1
                    beta = max(beta, float(beta_start))
                    stall = 0

        it += 1
        if it % 20 == 0:
            beta = min(beta_max, beta * 1.15)

    return best, best_exact


def _local_block_search(p0, rng, time_limit):
    """Exact coordinate search: multiply random contiguous blocks."""
    p = _sanitize_vec(p0)
    if p is None or time_limit <= 0:
        return None, float("inf")

    n = int(p.size)
    if n <= 1:
        return p, _exact_score(p)

    floor = 1e-18 / float(n)
    p = np.maximum(p, floor)
    p = p / p.sum()

    best = p.copy()
    best_score = _exact_score(p)
    cur_score = best_score
    deadline = time.monotonic() + float(time_limit)
    stagnant = 0

    factors = (0.35, 0.65, 0.85, 1.15, 1.35, 1.75, 2.5)

    while time.monotonic() < deadline:
        r = float(rng.random())
        if r < 0.30:
            w = int(rng.integers(1, max(2, n // 200 + 1)))
        elif r < 0.75:
            w = int(rng.integers(max(2, n // 200), max(3, n // 40 + 1)))
        else:
            low = max(3, n // 40)
            high = max(low + 1, min(n, n // 5 + 1))
            w = int(rng.integers(low, high + 1))

        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))

        base = p.copy()
        current_best = base
        current_score = cur_score

        for f in factors:
            cand = base.copy()
            cand[start:start + w] *= float(f)
            cand = _sanitize_vec(cand)
            if cand is None:
                continue
            cand = np.maximum(cand, floor)
            cand = cand / cand.sum()
            sc = _exact_score(cand)
            if sc < current_score - 1e-12:
                current_score = sc
                current_best = cand

        if current_best is not base:
            p = current_best.copy()
            cur_score = current_score
            if cur_score < best_score:
                best = p.copy()
                best_score = cur_score
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 60:
                p_try = _sanitize_vec(p * np.exp(rng.normal(0.0, 0.05, n)))
                if p_try is not None:
                    p_try = np.maximum(p_try, floor)
                    p_try = p_try / p_try.sum()
                    sc = _exact_score(p_try)
                    if np.isfinite(sc) and sc <= best_score * 1.05:
                        p = p_try
                        cur_score = sc
                    else:
                        p = best.copy()
                        cur_score = best_score
                stagnant = 0

        if stagnant > 350:
            p = best.copy()
            cur_score = best_score
            stagnant = 0

    return best, best_score
