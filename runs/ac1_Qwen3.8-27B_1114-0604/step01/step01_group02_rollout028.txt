"""
Combined optimizer for minimizing f(a) = 2n*max(conv(a,a))/(sum a)^2.
Uses: (1) log-sum-exp smoothed gradient with Adam, (2) periodic LP-based SCA steps,
(3) GS-inspired spectral initialization, (4) multi-phase time management with restarts.

import numpy as np
import time
import sys

# ----------------------------------------------------------------------------
# Core computational primitives
# ----------------------------------------------------------------------------

def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def cross_corr_weights(w, a):
    """Compute h[j] = sum_k w[k]*a[k-j] for j=0..n-1 (weighted correlation)."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    W = np.fft.rfft(w, M_fft)
    Af = np.fft.rfft(a[::-1], M_fft)
    return np.fft.irfft(W * Af, M_fft)[n - 1:n - 1 + n]


def lse_value_weights(c, temp):
    """Log-sum-exp approximation of max(c) and softmax weights."""
    cmax = c.max()
    shifted = (c - cmax) / temp
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    w = exp_c / s
    return cmax + temp * np.log(s), w


def compute_objective(a):
    """Compute the exact objective value."""
    n = len(a)
    S = a.sum()
    if S < 1e-10:
        return np.inf
    c = fast_self_conv(a)
    return 2.0 * n * float(c.max()) / (S * S), c, S


def compute_grad_lse(a, c, temp):
    """Compute gradient of 2n*LSE(c,temp)/S^2 with respect to a."""
    n = len(a)
    S = a.sum()
    lse_val, w = lse_value_weights(c, temp)
    # Gradient of sum_k w[k]*conv(a,a)[k] w.r.t. a[j] = 2*sum_k w[k]*a[k-j]
    h = cross_corr_weights(w, a)
    grad_c = 2.0 * h
    # Full gradient: d/d a[j] [2n * lse_val / S^2]
    # = 2n * (grad_c[j] * 1/S^2 + lse_val * d(1/S^2)/da[j])
    # d(1/S^2)/da[j] = -2/(S^3)
    grad = (2.0 * n) * (grad_c / (S * S) - 2.0 * lse_val / (S ** 3))
    return grad, lse_val, w


def adam_update(a, grad, state, lr):
    """Adam optimizer step, in-place on state dict."""
    t = state['t'] + 1
    state['t'] = t
    b1, b2, eps = 0.9, 0.999, 1e-8
    state['m'] = b1 * state['m'] + (1.0 - b1) * grad
    state['v'] = b2 * state['v'] + (1.0 - b2) * grad * grad
    bc1 = 1.0 - b1 ** t
    bc2 = 1.0 - b2 ** t
    a -= lr * (state['m'] / bc1) / (np.sqrt(state['v'] / bc2) + eps)
    return a

    return {'m': np.zeros(n), 'v': np.zeros(n), 't': 0}

    S = a.sum()

def sca_lp_step(a, c, K=80):
    """Solve SCA LP with top-K constraints. Returns new a."""
    n = len(a)
    S = a.sum()
    a_norm = a / S  # work with sum=1
    
    # Select top-K lags
    top_idx = np.argpartition(c, -K)[-K:]
    
    # Build constraint matrix: A[k,j] = 2*a_norm[k-j] for valid indices
    # Constraint: 2*(a0*a)[k] - c[k] <= M  for k in top_idx
    # where (a0*a)[k] = sum_j a0[k-j]*a[j]
    
    from scipy.sparse import lil_matrix
    from scipy.optimize import linprog
    
    rows = lil_matrix((K, n))
    rhs = np.zeros(K)
    
    for ii, k in enumerate(top_idx):
        j_start = max(0, k - n + 1)
        j_end = min(k, n - 1)
        for j in range(j_start, j_end + 1):
            val = a_norm[k - j]
            if val > 1e-15:
                rows[ii, j] = 2.0 * val
        rhs[ii] = c[k]  # constraint: 2*(a0*a)[k] - c[k] <= M, so 2*(a0*a)[k] <= M + c[k]
    
    # Variables: [a_0, ..., a_{n-1}, M]
    # Objective: minimize M → c_obj = [0,...,0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = rows.todense()
    A_ub[:, -1] = -1.0  # 2*(a0*a)[k] - M <= c[k]
    b_ub = rhs
    
    # Equality: sum(a) = 1
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([1.0])
    
    # Bounds: a >= 0, M free (but M >= 0 since c >= 0)
    bounds = [(0, None)] * n + [(0, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        a_new = res.x[:n]
        a_new = np.clip(a_new, 0, None)
        a_new = normalize_sum(a_new, S)  # restore original scale
        return a_new
    return a.copy()


# ----------------------------------------------------------------------------
# Initializations
# ----------------------------------------------------------------------------

def init_uniform(n):
    return np.ones(n)


def init_biased_sin(n, freq, rng, amp=0.8):
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    phase = rng.random() * 2 * np.pi
    a = 1.0 + amp * np.sin(2.0 * np.pi * freq * x + phase)
    return np.clip(a, 0.0, None)


def init_cos_sum(n, freqs, rng):
    x = np.linspace(0.0, 1.0, n, endpoint=False)
    a = np.ones(n)
    for freq in freqs:
        phase = rng.random() * 2 * np.pi
        a += 0.4 * np.cos(2.0 * np.pi * freq * x + phase)
    return np.clip(a, 0.0, None)


def init_flat_spec(n, rng):
    """Sequence whose autocorrelation is spread out (random phases)."""
    M_fft = 1 << (2 * n - 2).bit_length()
    phases = rng.random(M_fft // 2 + 1) * 2 * np.pi
    mags = np.ones(M_fft // 2 + 1)
    A = mags * np.exp(1j * phases)
    a = np.fft.irfft(A, M_fft)[:n]
    a = np.abs(a)
    # Sharpen a bit to avoid too much flatness
    a = a ** 1.5
    return np.clip(a, 0.0, None)


def init_ramp(n, alpha, rng):
    x = np.linspace(0.0, 1.0, n)
    a = 1.0 + alpha * (2.0 * x - 1.0)
    return np.clip(a, 0.01, None)


def init_blocks(n, nb, rng):
    per = max(1, n // nb)
    vals = 0.5 + rng.random(nb)
    a = np.repeat(vals, per)
    a = a[:n]
    return np.clip(a, 0.01, None)


def init_fbm(n, rng, power=0.5):
    """Fractional Brownian motion-like initialization."""
    M_fft = 1 << (n - 1).bit_length()
    freqs = np.fft.rfftfreq(M_fft)
    mask = freqs > 0
    mags = np.ones(M_fft // 2 + 1)
    mags[mask] = freqs[mask] ** (-power)
    phases = rng.random(M_fft // 2 + 1) * 2 * np.pi
    A = mags * np.exp(1j * phases)
    a = np.fft.irfft(A, M_fft)[:n]
    a = np.abs(a)
    a = a + 0.5
    return a


def init_parent_scaled(n, parent):
    """Scale and adapt parent sequence to length n."""
    if parent is None or len(parent) == 0:
        return np.ones(n)
    # Interpolate to length n
    parent = np.asarray(parent, dtype=np.float64)
    x_old = np.linspace(0, 1, len(parent))
    x_new = np.linspace(0, 1, n)
    a = np.interp(x_new, x_old, parent)
    return np.clip(a, 0.0, None)


def build_initializations(n, rng, parent=None):
    inits = []
    if parent is not None and len(parent) > 0:
        inits.append(("parent", init_parent_scaled(n, parent)))
    inits.append(("uniform", init_uniform(n)))
    for freq, amp in [(1, 0.7), (2, 0.7), (3, 0.6), (5, 0.5), (7, 0.4)]:
        inits.append(("sin%d_%s" % (freq, amp), init_biased_sin(n, freq, rng, amp)))
    inits.append(("cos12", init_cos_sum(n, [1, 2], rng)))
    inits.append(("cos23", init_cos_sum(n, [2, 3], rng)))
    inits.append(("flat_spec", init_flat_spec(n, rng)))
    inits.append(("ramp_0.5", init_ramp(n, 0.5, rng)))
    inits.append(("ramp_1.0", init_ramp(n, 1.0, rng)))
    inits.append(("blocks_8", init_blocks(n, 8, rng)))
    inits.append(("blocks_16", init_blocks(n, 16, rng)))
    inits.append(("fbm_0.3", init_fbm(n, rng, 0.3)))
    inits.append(("fbm_0.7", init_fbm(n, rng, 0.7)))
    # Random
    inits.append(("rand", 0.3 + 0.7 * rng.random(n)))
    return inits


# ----------------------------------------------------------------------------
# Main optimization driver
# ----------------------------------------------------------------------------

def optimize_sequence(a0, time_limit, init_temp=0.05, lr_init=0.5, 
                      use_lp=True, lp_every=100, lp_K=80, tag=""):
    """Optimize from a0 within time_limit seconds."""
    n = a0.size
    a = np.clip(a0.astype(np.float64), 0, None).copy()
    a = normalize_sum(a, 1.0)
    
    c = fast_self_conv(a)
    score, c, S = compute_objective(a)
    best_score = score
    best_a = a.copy()
    
    state = make_adam_state(n)
    
    # Temperature schedule (geometric decay)
    temp = init_temp
    temp_decay = 0.998
    temp_min = 0.0005
    
    # Learning rate
    lr = lr_init
    
    start = time.monotonic()
    it = 0
    no_improve = 0
    patience = 200
    
    while time.monotonic() - start < time_limit:
        # Check time frequently
        if time.monotonic() - start > time_limit:
            break
        
        it += 1
        
        # Recompute convolution
        c = fast_self_conv(a)
        score, c, S = compute_objective(a)
        
        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Patience-based restart
        if no_improve >= patience:
            # Perturb from best
            scale = 0.03 + 0.07 * np.random.random()
            a = best_a * (1.0 + scale * np.random.randn(n))
            a = np.clip(a, 0, None)
            a = normalize_sum(a, 1.0)
            state = make_adam_state(n)
            lr = max(0.05, lr * 0.6)
            temp = max(temp_min, temp * 2)
            no_improve = 0
            patience = max(50, patience * 3 // 4)
            continue
        
        # Compute gradient with LSE smoothing
        grad, lse_val, w = compute_grad_lse(a, c, temp)
        
        # Adam step
        a = adam_update(a, grad, state, lr)
        a = np.clip(a, 0, None)
        if a.sum() < 1e-8:
            a = best_a * 0.5
            state = make_adam_state(n)
        a = normalize_sum(a, 1.0)
        
        # Periodic LP step for bigger moves
        if use_lp and it % lp_every == 0 and n <= 4000:
            a_lp = sca_lp_step(a, c, K=lp_K)
            # Blend
            blend = 0.3 + 0.4 * (no_improve / patience)
            a = (1 - blend) * a + blend * a_lp
            a = np.clip(a, 0, None)
            a = normalize_sum(a, 1.0)
            state = make_adam_state(n)  # Reset Adam state after big move
        
        # Decay temperature
        temp = max(temp_min, temp * temp_decay)
        
        # Decay learning rate slowly
        lr = max(0.01, lr * 0.9995)
    
    return best_score, best_a

# Entry point

def propose_candidate(seed=123, budget_s=2400, **kwargs):
    """Search for non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = max(1.0, float(budget_s))
    deadline = start + total - 10.0  # Leave 10s margin for final eval
    
    # Get parent sequence if available
    parent = None
    try:
        p = np.asarray(height_sequence_1, dtype=np.float64).ravel()
        if p.size > 0 and np.all(np.isfinite(p)):
            parent = p
    except Exception:
        pass
    
    print("Budget: %.0fs, parent available: %s" % (total, parent is not None))
    
    best_score_global = np.inf
    best_a_global = None
    
    # Try multiple sequence lengths
    lengths_to_try = []
    if parent is not None:
        parent_len = len(parent)
        lengths_to_try.append(parent_len)
    
    # Always try some standard lengths
    for L in [1024, 2048, 4096, 8192]:
        if L not in lengths_to_try:
            lengths_to_try.append(L)
    
    # Sort and filter (cap at 16384 for safety)
    lengths_to_try = [L for L in sorted(set(lengths_to_try)) if 256 <= L <= 16384]
    if not lengths_to_try:
        lengths_to_try = [2048]
    
    # Allocate time: longer sequences get more time if parent matches
    # Strategy: spend 60% on best candidate length, 40% exploring others
    
    # First, quick evaluation of parent if available
    if parent is not None:
        p_eval = evaluate_sequence([float(x) for x in parent])
        print("Parent score: %.6f, len=%d" % (p_eval, len(parent)))
        if np.isfinite(p_eval) and p_eval < best_score_global:
            best_score_global = p_eval
            best_a_global = parent.copy()
    
    # Main optimization loop over different lengths and initializations
    # Phase 1: Explore diverse initializations (55% of time)
    # Phase 2: Refine best (30% of time)  
    # Phase 3: Final polish (15% of time)
    
    phase1_end = start + 0.55 * (total - 10.0)
    phase2_end = start + 0.85 * (total - 10.0)
    
    # Choose primary length (prefer parent length, else 4096)
    if parent is not None and len(parent) in lengths_to_try:
        primary_len = len(parent)
    else:
        primary_len = 4096 if 4096 in lengths_to_try else lengths_to_try[0]
    
    print("Primary length: %d, all lengths: %s" % (primary_len, lengths_to_try))
    
    # Phase 1: Explore
    n = primary_len
    inits = build_initializations(n, rng, parent)
    
    # Time per initialization
    n_inits = len(inits)
    time_per_init = max(3.0, (phase1_end - time.monotonic()) / n_inits)
    time_per_init = min(time_per_init, 120.0)  # Cap at 120s per init
    
    print("Phase 1: %d initializations, ~%.1fs each" % (n_inits, time_per_init))
    
    for tag, a0 in inits:
        if time.monotonic() > phase1_end:
            break
        
        a0 = np.clip(a0.astype(np.float64), 0, None)
        a0 = normalize_sum(a0, 1.0)
        
        t0 = time.monotonic()
        alloc = min(time_per_init, phase1_end - t0)
        if alloc < 2.0:
            break
        
        # Choose hyperparameters based on initialization
        use_lp = (n <= 4000 and tag not in ("uniform",))
        
        sc, aa = optimize_sequence(
            a0, alloc,
            init_temp=0.08,
            lr_init=0.3,
            use_lp=use_lp,
            lp_every=80,
            lp_K=60 if n > 2000 else 100,
            tag=tag
        )
        
        if sc < best_score_global:
            best_score_global = sc
            best_a_global = aa.copy()
            print("  [%s n=%d] NEW BEST: %.6f (t=%.1fs elapsed)" % 
                  (tag, n, sc, time.monotonic() - start))
        else:
            print("  [%s n=%d] score=%.6f (t=%.1fs)" % (tag, n, sc, time.monotonic() - start))
    
    # Also try second-best length briefly
    if time.monotonic() < phase1_end:
        for L in lengths_to_try:
            if L == primary_len:
                continue
            if L > 8192:
                continue
            if time.monotonic() > phase1_end - 5:
                break
            # Quick optimize with uniform init at this length
            a0 = np.ones(L)
            sc, aa = optimize_sequence(a0, min(30.0, phase1_end - time.monotonic()),
                                       init_temp=0.1, lr_init=0.3, use_lp=False)
            if sc < best_score_global:
                best_score_global = sc
                best_a_global = aa.copy()
                print("  [unif n=%d] NEW BEST: %.6f" % (L, sc))
    
    # Phase 2: Refine global best
    if best_a_global is not None and time.monotonic() < phase2_end:
        n_best = len(best_a_global)
        alloc = phase2_end - time.monotonic()
        print("Phase 2: Refining best (n=%d, alloc=%.0fs, score=%.6f)" % 
              (n_best, alloc, best_score_global))
        
        # Try multiple refinement runs with different temperatures
        n_demos = 3
        demo_time = alloc / n_demos
        for demo in range(n_demos):
            if time.monotonic() > phase2_end:
                break
            # Slight perturbation for diversity
            pert_scale = [0.0, 0.02, 0.05][demo]
            if pert_scale > 0:
                a_pert = best_a_global * (1.0 + pert_scale * rng.standard_normal(n_best))
                a_pert = np.clip(a_pert, 0, None)
                a_pert = normalize_sum(a_pert, 1.0)
            else:
                a_pert = best_a_global.copy()
            
            sc, aa = optimize_sequence(
                a_pert, demo_time,
                init_temp=0.03 * (demo + 1),
                lr_init=0.1 / (demo + 1),
                use_lp=(n_best <= 4000),
                lp_every=100,
                lp_K=80
            )
            if sc < best_score_global:
                best_score_global = sc
                best_a_global = aa.copy()
                print("  Phase2 demo%d: %.6f" % (demo, sc))
    
    # Phase 3: Final polish
    if best_a_global is not None and time.monotonic() < deadline:
        n_best = len(best_a_global)
        alloc = deadline - time.monotonic()
        print("Phase 3: Final polish (alloc=%.0fs)" % alloc)
        
        sc, aa = optimize_sequence(
            best_a_global, alloc,
            init_temp=0.005,
            lr_init=0.02,
            use_lp=False,
            lp_every=1000,  # Effectively disable LP
            lp_K=50
        )
        if sc < best_score_global:
            best_score_global = sc
            best_a_global = aa.copy()
    
    # Fallback
    if best_a_global is None:
        best_a_global = np.ones(2048)
    
    # Final verification
    out = [float(x) for x in best_a_global]
    verified = float(evaluate_sequence(out))
    elapsed = time.monotonic() - start
    print("FINAL: internal=%.6f, verified=%.6f, len=%d, time=%.1fs" % 
          (best_score_global, verified, len(out), elapsed))
    
    return out
