            m[:] = 0.0

"""
Minimax autocorrelation sequence optimizer.
Uses recursive flat-convolution initialization, adaptive soft-max gradient
descent, peak-targeting refinement, and multi-scale search over sequence lengths.
"""

import numpy as np
import time


def fast_self_conv(a):
    """Linear self-convolution via real FFT."""
    n = len(a)
    M = 2 * n - 1
    M_fft = 1 << (M - 1).bit_length()
    A = np.fft.rfft(a, M_fft)
    return np.fft.irfft(A * A, M_fft)[:M]


def score_from_conv(n, c, S):
    """Objective: 2n * max(conv) / S^2."""
    if S < 1e-10:
        return np.inf
    return 2.0 * n * float(np.max(c)) / (S * S)


def recursive_flat_init(n):
    """Generate sequence approximating flat left-half convolution."""
    half = (n + 1) // 2
    a = np.zeros(half)
    a[0] = 1.0
    for k in range(1, half):
        if k >= 2:
            s = float(np.dot(a[1:k], a[k - 1:0:-1]))
        else:
            s = 0.0
        val = (1.0 - s) / 2.0
        a[k] = max(0.0, val)
    if n % 2 == 1:
        full = np.concatenate([a, a[-2::-1]])
    else:
        full = np.concatenate([a, a[-1::-1]])
    return full[:n]


def init_raised_cosine(n):
    x = np.linspace(0.0, np.pi, n)
    return np.sin(x) ** 2 + 1e-8


def init_tent(n):
    mid = (n - 1) / 2.0
    return np.maximum(0.01, 1.0 - np.abs(np.arange(n) - mid) / max(mid, 1.0))


def init_u_shape(n, alpha=2.0):
    x = np.linspace(-1.0, 1.0, n)
    return 1.0 + alpha * x ** 2


def init_uniform(n):
    return np.ones(n)


def init_gaussian(n, sigma_frac=0.3):
    x = np.linspace(0.0, 1.0, n)
    center = 0.5
    sigma = sigma_frac
    return np.exp(-0.5 * ((x - center) / sigma) ** 2) + 1e-8


def normalize_seq(a, n):
    """Normalize so sum(a) = sqrt(2n), clip to [0, 1000]."""
    S = a.sum()
    if S > 1e-15:
        a = a * (np.sqrt(2.0 * n) / S)
    return np.clip(a, 0.0, 1000.0)


def compute_softgrad(a, c, n, k, temp):
    """Compute soft-max weighted gradient of the peak objective."""
    S = float(a.sum())
    S2 = S * S
    S3 = S2 * S

    k_act = min(k, len(c))
    if k_act >= len(c):
        idx = np.arange(len(c))
    else:
        idx = np.argpartition(c, -k_act)[-k_act:]
    vals = c[idx]

    top_val = float(vals.max())
    if top_val > 1e-15:
        t_denom = temp * top_val
        if t_denom < 1e-12:
            t_denom = 1e-12
        wts = np.exp((vals - top_val) / t_denom)
        wts = wts / wts.sum()
    else:
        wts = np.ones(k_act) / k_act

    grad = np.zeros(n)
    M_w = 0.0

    for i in range(k_act):
        ki = int(idx[i])
        wi = float(wts[i])
        if wi < 1e-15:
            continue
        M_w += wi * float(c[ki])

        j_start = max(0, ki - n + 1)
        j_end = min(ki, n - 1)
        if j_start > j_end:
            continue

        lo = ki - j_end
        hi = ki - j_start
        partner = a[lo:hi + 1]

        grad[j_start:j_end + 1] += wi * 2.0 * partner

        if ki % 2 == 0:
            mid_j = ki // 2
            if j_start <= mid_j <= j_end:
                grad[mid_j] -= wi * a[mid_j]

    result = (2.0 * n) * (grad / S2 - 2.0 * M_w / S3)
    return result


def adam_step(a, g, m, v, t, lr, b1=0.9, b2=0.999, eps=1e-8):
    """In-place Adam update."""
    t1 = b1 ** t
    t2 = b2 ** t
    m[:] = b1 * m + (1.0 - b1) * g
    v[:] = b2 * v + (1.0 - b2) * g * g
    mh = m / (1.0 - t1)
    vh = v / (1.0 - t2)
    a -= lr * mh / (np.sqrt(vh) + eps)


def peak_reduce(a, c, n, n_reduce=20, factor=0.01):
    """Directly reduce elements contributing most to the peak."""
    k_peak = int(np.argmax(c))
    j_start = max(0, k_peak - n + 1)
    j_end = min(k_peak, n - 1)
    if j_start > j_end:
        return a

    lo = k_peak - j_end
    hi = k_peak - j_start
    contribs = a[j_start:j_end + 1] * a[lo:hi + 1]
    total = contribs.sum()
    if total < 1e-15:
        return a

    n_r = min(n_reduce, j_end - j_start + 1)
    top_local = np.argpartition(contribs, -n_r)[-n_r:]
    for idx in top_local:
        j = j_start + int(idx)
        a[j] *= (1.0 - factor)
    return a


def optimize_from(a0, global_start, end_time, rng, init_lr=0.5,
                  topk_start=3, topk_end=50, temp_start=0.5, temp_end=0.02,
                  patience=400, do_peak_reduce=True):
    """Adaptive gradient optimization with restarts and peak reduction."""
    n = a0.size
    a = a0.astype(np.float64).copy()
    a = np.clip(a, 0.0, 1000.0)
    a = normalize_seq(a, n)

    c = fast_self_conv(a)
    S = a.sum()
    score = score_from_conv(n, c, S)
    best_score = score
    best_a = a.copy()

    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    no_improve = 0
    step_count = 0

    dur = max(end_time - global_start, 1.0)

    while time.monotonic() < end_time:
        frac = (time.monotonic() - global_start) / dur
        frac = min(max(frac, 0.0), 1.0)

        topk = int(topk_start + (topk_end - topk_start) * frac)
        topk = max(2, topk)
        temp = temp_start + (temp_end - temp_start) * frac
        lr = init_lr * (1.0 - 0.8 * frac * frac)
        lr = max(lr, 1e-5)

        c = fast_self_conv(a)
        S = a.sum()
        score = score_from_conv(n, c, S)

        if np.isfinite(score) and score < best_score:
            best_score = score
            best_a = a.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve >= patience:
            scale = 0.02 + 0.08 * rng.random()
            a = best_a * (1.0 + scale * rng.standard_normal(n))
            a = np.clip(a, 0.0, 1000.0)
            a = normalize_seq(a, n)
            m[:] = 0.0
            v[:] = 0.0
            t = 0
            no_improve = 0
            patience = max(80, patience // 2)
            continue

        if do_peak_reduce and step_count > 0 and step_count % 500 == 250:
            a = peak_reduce(a, c, n, n_reduce=10, factor=0.003)
            a = np.clip(a, 0.0, 1000.0)
            a = normalize_seq(a, n)

        g = compute_softgrad(a, c, n, topk, temp)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-10:
            g = g / (g_norm / n)

        t += 1
        step_count += 1
        adam_step(a, g, m, v, t, lr)
        a = np.clip(a, 0.0, 1000.0)
        a = normalize_seq(a, n)

    return best_score, best_a


def optimize_multi_init(n, global_start, end_time, rng, n_inits=5):

    best_score = np.inf
    best_a = None

    inits = []

    # Try to include parent

            if p.size == n:

            elif p.size > 0:
                # Resize parent to n via interpolation
                p_resized = np.interp(
                    np.linspace(0, 1, n),
                    np.linspace(0, 1, p.size),
                    p
                )
                inits.insert(0, ("parent_resized", np.clip(p_resized, 0.0, 1000.0)))
    except Exception:
        pass

    # Limit number of initializations to fit in time
    time_budget = end_time - global_start
    time_each = max(5.0, time_budget / len(inits))

    for tag, a0 in inits:
        if time.monotonic() >= end_time - 5.0:
            break
        t0 = time.monotonic()
        t_end = min(end_time, t0 + time_each)
        sc, aa = optimize_from(a0, global_start, t_end, rng,
                               init_lr=0.4, topk_start=3, topk_end=40,
                               temp_start=0.4, temp_end=0.03, patience=300)
        if sc < best_score:
            best_score = sc
            best_a = aa.copy()
            print("  [%s n=%d] score=%.6f" % (tag, n, best_score))

    return best_score, best_a


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """Search for a non-negative sequence minimizing the evaluation function."""
    rng = np.random.default_rng(seed)
    start = time.monotonic()
    total = max(1.0, float(budget_s))
    deadline = start + total - 12.0

    best_score = np.inf
    best_a = None

    print("=== Starting optimization (budget=%.0fs) ===" % total)

    # Phase allocation:
    # - 40%: explore at n=16384 (multiple inits)
    # - 25%: explore at n=32768 (top inits)
    # - 15%: explore at n=8192 (if parent exists, optimize parent)
    # - 20%: final polish on global best

    # Check parent
    parent = None
    try:
        if 'height_sequence_1' in globals() and height_sequence_1 is not None:
            parent = np.asarray(height_sequence_1, dtype=np.float64).ravel()
            if parent.size > 0:
                parent = np.clip(parent, 0.0, 1000.0)
                print("Parent available, n=%d" % parent.size)
    except Exception:
        pass

    # Phase 1: Explore at n=16384
    n1 = 16384
    t1_end = start + 0.38 * (total - 12.0)
    print("Phase 1: n=%d, until t=%.0fs" % (n1, t1_end - start))
    sc1, aa1 = optimize_multi_init(n1, start, t1_end, rng)
    if sc1 < best_score:
        best_score = sc1
        best_a = aa1.copy()
        print("Phase 1 best: %.6f" % best_score)

    # Phase 2: Explore at n=32768
    n2 = 32768
    t2_end = start + 0.60 * (total - 12.0)
    if time.monotonic() < t2_end - 30:
        print("Phase 2: n=%d, until t=%.0fs" % (n2, t2_end - start))
        sc2, aa2 = optimize_multi_init(n2, start, t2_end, rng)
        if sc2 < best_score:
            best_score = sc2
            best_a = aa2.copy()
            print("Phase 2 best: %.6f" % best_score)
    else:
        print("Skipping Phase 2 (insufficient time)")

    # Phase 3: If parent exists and we haven't used it at its native length
    if parent is not None:
        t3_end = start + 0.78 * (total - 12.0)
        if time.monotonic() < t3_end - 30:
            n_p = parent.size
            print("Phase 3: parent n=%d, until t=%.0fs" % (n_p, t3_end - start))
            sc3, aa3 = optimize_from(parent, start, t3_end, rng,
                                     init_lr=0.3, topk_start=5, topk_end=35,
                                     temp_start=0.3, temp_end=0.03, patience=400)
            if sc3 < best_score:
                best_score = sc3
                best_a = aa3.copy()
                print("Phase 3 best: %.6f" % best_score)

    # Phase 4: Final polish on global best
    if best_a is not None and time.monotonic() < deadline:
        print("Phase 4: final polish, until t=%.0fs" % (deadline - start))
        n_best = best_a.size
        sc4, aa4 = optimize_from(best_a, start, deadline, rng,
                                 init_lr=0.1, topk_start=15, topk_end=80,
                                 temp_start=0.15, temp_end=0.01,
                                 patience=600, do_peak_reduce=True)
        if sc4 < best_score:
            best_score = sc4
            best_a = aa4.copy()
        print("Phase 4 best: %.6f" % best_score)

    # Safety fallback
    if best_a is None:
        best_a = np.ones(1024)
        best_score = np.inf

    out = [float(x) for x in best_a]

    # Verify
    try:
        verified = float(evaluate_sequence(out))
    except Exception:
        verified = np.inf

    print("=== FINAL: internal=%.6f, verified=%.6f, n=%d, time=%.1fs ==="
          % (best_score, verified, len(out), time.monotonic() - start))
    return out
