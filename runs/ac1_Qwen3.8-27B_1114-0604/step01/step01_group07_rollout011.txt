import numpy as np
import time

def _fft_convolve(a):
    """Linear convolution a*a using FFT."""
    n = len(a)
    need = 2 * n - 1
    N = 1
    while N < need:
        N <<= 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _score(a):
    """Compute the objective value."""
    s = np.sum(a)
    if s < 1e-6:
        return np.inf
    c = _fft_convolve(a)
    return 2.0 * len(a) * np.max(c) / (s * s)

def _gradient_topk(a, K=16):
    """Compute the (sub)gradient of max_k (a*a)_k using top-K lags."""
    c = _fft_convolve(a)
    maxc = np.max(c)
    # Get top K lags
    if len(c) <= K:
        active = np.arange(len(c))
        weights = np.ones(len(c))
    else:
        active = np.argpartition(c, -K)[-K:]
        # Weight by proximity to max
        vals = c[active]
        weights = np.exp(-((maxc - vals) * 10.0))  # soft weighting
        weights /= np.sum(weights)
    
    n = len(a)
    g = np.zeros(n)
    for k, w in zip(active, weights):

        i_min = max(0, k - n + 1)
        i_max = min(n - 1, k)
        if i_max >= i_min:

            g[idx] += w * 2.0 * a[k - idx]
    return g, c, maxc

def _project_simplex(v):
    """Project v onto the simplex {x : x >= 0, sum x = 1}."""
    n = len(v)
    # Find rho such that sum(max(v_i - rho, 0)) = 1
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.argmax(1.0 / np.arange(1, n+1) - (cssv - 1.0) > 0)
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)
