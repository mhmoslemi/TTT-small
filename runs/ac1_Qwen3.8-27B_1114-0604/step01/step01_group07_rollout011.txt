```python
"""
Low-peak nonnegative step-function search using mirror descent subgradient
optimization with multi-phase active set strategy.

Algorithm:
1. Initialize with optimal anti-palindromic sinusoidal shape.
2. Phase 1: Coarse mirror descent with top-20 active lags, large steps.
3. Phase 2: Medium mirror descent with top-50 active lags, medium steps.
4. Phase 3: Fine mirror descent with top-200 active lags, small steps.
5. Phase 4: Local coordinate refinement.
6. Validate with evaluate_sequence.
"""

import numpy as np
import time

def _next_pow2(m):
    n = 1
    while n < m:
        n <<= 1
    return n

def _fft_convolve(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.array([], dtype=np.float64)
    need = 2 * n - 1
    N = _next_pow2(need)
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)
    return c[:need]

def _score(a):
    """Fast internal score computation."""
    arr = np.asarray(a, dtype=np.float64)
    n = arr.size
    s = float(arr.sum())
    if s < 0.01:
        return np.inf
    c = _fft_convolve(arr)
    mx = float(np.max(c))
    return float(2.0 * n * mx / (s * s))

def _subgradient(a, c, active_set):
    """
    Compute subgradient of max over active_set of (a*a)[k].
    a: sequence of length n
    c: convolution (length 2n-1)
    active_set: indices into c
    Returns gradient array of length n.
    """
    n = a.size
    grad = np.zeros(n, dtype=np.float64)
    for k in active_set:
        k = int(k)
        # d(a*a)[k]/da[m] = 2*a[k-m] for 0 <= k-m < n
        # m ranges from max(0, k-n+1) to min(k, n-1)
        m_min = max(0, k - n + 1)
        m_max = min(k, n - 1)
        if m_max >= m_min:
            # grad[m] += 2 * a[k-m] for m in [m_min, m_max]
            # a[k-m] for m in [m_min, m_max] is a[m_max] down to a[m_min] reversed
            # i.e., a[k-m_min], a[k-m_min-1], ..., a[k-m_max]
            # = a[k-m_min] : a[k-m_max-1] : -1 = a[k-m_min : k-m_max] : -1
            # Actually: for m = m_min, a[k-m_min]; for m = m_max, a[k-m_max]
            # So the slice is a[k-m_min : k-m_max : -1]
            # Let's verify: k-m_min >= k-m_max, so we go from k-m_min down to k-m_max
            # In numpy: a[k-m_min : k-m_max-1 : -1] gives a[k-m_min], a[k-m_min-1], ..., a[k-m_max]
            # That's (m_max - m_min + 1) elements.
            idx = k - np.arange(m_min, m_max + 1)
            grad[m_min:m_max+1] += 2.0 * a[idx]
    return grad

def _mirror_descent_step(a, grad, eta):
    """
    Multiplicative weights update: a <- a * exp(-eta * grad) / sum(a * exp(-eta * grad))
    """
    exponent = -eta * grad
    exponent = np.clip(exponent, -2.0, 2.0)  # numerical stability
    new_a = a * np.exp(exponent)
    new_a = np.where(np.isfinite(new_a), new_a, 0.0)
    s = np.sum(new_a)
    if s < 1e-10:
        return a.copy()
    return new_a / s

def _make_sine(n, amp, mode=1):
    """Create anti-palindromic sinusoidal sequence normalized to sum=1."""
    i = np.arange(n, dtype=np.float64)
    theta = 2.0 * np.pi * (i + 0.5) / float(n)
    a = (1.0 + amp * np.sin(mode * theta)) / float(n)
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.ones(n) / n
    return a / s

def _optimize_sine(n, deadline):
    """Find optimal amplitude for sinusoidal shape of length n."""
    best_a = None
    best_s = np.inf
    for amp in np.linspace(0.0, 1.0, 21):
        if time.monotonic() > deadline:
            break
        a = _make_sine(n, amp)
        s = _score(a)
        if s < best_s:
            best_s = s
            best_a = a.copy()
    # Refine
    if best_s < np.inf:
        # Find the amplitude that gave best_s
        # Just do a finer scan around it
        amps = np.linspace(0.0, 1.0, 101)
        for amp in amps:
            if time.monotonic() > deadline:
                break
            a = _make_sine(n, amp)
            s = _score(a)
            if s < best_s:
                best_s = s
                best_a = a.copy()
    return best_a, best_s

def _subgradient_optimize(a, budget_s, K=50, eta0=0.01, name=""):
    """
    Mirror descent subgradient optimization.
    """
    start = time.monotonic()
    n = a.size
    best = a.copy()
    best_score = _score(a)
    print(f"[{name}] start score={best_score:.8f} n={n}")
    
    lr = eta0
    it = 0
    last_improve = 0
    patience = 200
    max_iters = 100000
    
    while time.monotonic() - start < budget_s and it < max_iters:
        it += 1
        
        c = _fft_convolve(a)
        if c.size == 0:
            break
        
        mx = float(np.max(c))
        current_score = 2.0 * n * mx / (float(a.sum())**2)
        
        # Check for improvement
        if current_score < best_score:
            best_score = current_score
            best = a.copy()
            last_improve = it
        
        # Patience check
        if it - last_improve > patience:
            # Reset to best and try different step size
            a = best.copy()
            lr = max(0.001, lr * 0.5)
            if lr < 0.002:
                # Try a random perturbation
                a = best.copy()
                noise = np.random.normal(0, 0.1, n) * best
                a = a + noise
                a = np.clip(a, 0, None)
                a = a / a.sum()
                lr = eta0
            patience = 300
            continue
        
        # Get active set: top K lags
        if c.size <= K:
            active = np.arange(c.size)
        else:
            active = np.argpartition(c, -K)[-K:]
        
        # Compute subgradient
        grad = _subgradient(a, c, active)
        
        # Normalize gradient (remove mean for simplex constraint)
        grad = grad - np.mean(grad)
        
        # Scale gradient
        gnorm = np.max(np.abs(grad))
        if gnorm < 1e-15:
            break
        grad = grad / gnorm
        
        # Mirror descent step
        a = _mirror_descent_step(a, grad, lr)
        
        # Adaptive learning rate
        if it % 50 == 0:
            sc = _score(a)
            if sc < best_score:
                lr = min(lr * 1.05, 0.1)
            elif sc > best_score * 1.01:
                lr = max(lr * 0.7, 0.001)
        
        # Cosine annealing every 1000 iters
        if it % 1000 == 0:
            base_lr = eta0 * (1.0 + np.cos(np.pi * it / max_iters)) / 2.0
            lr = max(base_lr, 0.001)
    
    # Final evaluation
    final_score = _score(best)
    print(f"[{name}] done iters={it} best_score={final_score:.8f} time={time.monotonic()-start:.1f}s")
    return best, final_score

def _coordinate_refine(a, budget_s):
    """
    Coordinate-wise refinement: for each coordinate, try nearby values
    and keep the one that reduces the max convolution.
    """
    start = time.monotonic()
    n = a.size
    best = a.copy()
    best_score = _score(a)
    print(f"[coord] start score={best_score:.8f}")
    
    np.random.seed(123)
    it = 0
    while time.monotonic() - start < budget_s:
        it += 1
        # Pick a random coordinate
        i = np.random.randint(n)
        
        # Try a few perturbations
        best_local = a[i]
        for delta in [0.01, 0.005, 0.001, -0.01, -0.005, -0.001]:
            a_new = a.copy()
            a_new[i] = max(0, a[i] + delta)
            if a_new.sum() < 0.01:
                continue
            a_new = a_new / a_new.sum()
            sc = _score(a_new)
            if sc < best_local:
                best_local = sc
        
        if best_local < best_score * (1 - 1e-8):
            # Find the best perturbation
            for delta in [0.01, 0.005, 0.001, -0.01, -0.005, -0.001]:
                a[i] = max(0, a[i] + delta)
                a = a / a.sum()
                sc = _score(a)
                if sc <= best_local * (1 + 1e-10):
                    break
            best_score = _score(a)
            best = a.copy()
    
    print(f"[coord] done iters={it} best_score={best_score:.8f} time={time.monotonic()-start:.1f}s")
    return best, best_score
```

```python
def _smooth_optimize(a, budget_s, beta_start=10, beta_end=1000, name=""):
    """
    Smooth mirror descent optimization.
    """
    start = time.monotonic()
    n = a.size
    best = a.copy()
    best_score = _score(a)
    
    total_iters = 50000
    it = 0
    lr = 1.0 / n  # initial learning rate
    last_improve = 0
    
    while time.monotonic() - start < budget_s and it < total_iters:
        it += 1
        
        # Adaptive beta
        beta = beta_start * (beta_end / beta_start) ** (it / total_iters)
        
        score, grad = _smooth_score_and_grad(a, beta)
        
        if score < best_score:
            best_score = score
            best = a.copy()
            last_improve = it
        
        # Project gradient to simplex tangent plane
        grad = grad - np.mean(grad)
        
        # Multiplicative update
        exponent = -lr * grad
        exponent = np.clip(exponent, -1.0, 1.0)
        a = a * np.exp(exponent)
        a = np.clip(a, 0, None)
        s = a.sum()
        if s < 1e-10:
            a = best.copy()
            break
        a = a / s
        
        # Learning rate scheduling
        if (it + 1) % 100 == 0:
            sc = _score(a)
            if sc < best_score:
                lr = min(lr * 1.1, 10.0 / n)
            else:
                lr = max(lr * 0.8, 0.01 / n)
        
        # Patience
        if it - last_improve > 2000:
            a = best.copy()
            lr = lr * 0.5
            if lr < 0.01 / n:
                # Random perturbation
                a = best * (1 + 0.1 * np.random.randn(n))
                a = np.clip(a, 0, None)
                a = a / a.sum()
                lr = 1.0 / n
                last_improve = it
    
    return best, best_score
```
