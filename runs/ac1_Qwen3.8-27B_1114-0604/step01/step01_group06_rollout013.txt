"""
Minimize 2*n*max(convolve(a,a))/(sum(a))^2 for non-negative sequences a of length n.

Strategy:
- Use log-softmax parameterization for non-negativity and scale invariance.
- Multi-start L-BFGS with adaptive smooth-max (log-sum-exp) approximation.
- Subgradient polish with exact max after L-BFGS converges.
- Random jumps to escape local minima.
- Diverse parametric initializations (power-law, two-block, smooth profiles).
- LP-inspired "mirror" global step for occasional large improvements.

import numpy as np
import time
from scipy.optimize import minimize

# --- Global state ---
_G_N = 0
_G_NFFT = 0
_G_TAU = 1e-6
_G_BOUND = 30.0
_G_BOUNDS = []


def _set_state(n, tau=1e-6):
    global _G_N, _G_NFFT, _G_TAU, _G_BOUND, _G_BOUNDS
    _G_N = int(n)
    _G_NFFT = 1 << (3 * _G_N - 2).bit_length()
    _G_TAU = float(tau)
    _G_BOUND = 30.0
    _G_BOUNDS = [(-_G_BOUND, _G_BOUND)] * _G_N


def _softmax(z):
    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size == 0:
        return np.array([], dtype=np.float64)
    z = z - np.max(z)
    e = np.exp(np.clip(z, -700, 700))
    s = float(np.sum(e))
    if not np.isfinite(s) or s <= 0:
        return np.full(z.size, 1.0 / z.size)
    return e / s


def _fft_conv(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    n = a.size
    N = 1 << (3 * n - 2).bit_length()
    L = 2 * n - 1
    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L].real
    return np.maximum(c, 0.0)


def _fast_score(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return np.inf
    s = float(np.sum(a))
    if s <= 1e-12:
        return np.inf
    c = _fft_conv(a)
    m = float(np.max(c))
    return float(2 * a.size * m / (s * s))


def _obj_grad(z, tau=1e-6):
    n = _G_N
    N = _G_NFFT
    L = 2 * n - 1

    z = np.asarray(z, dtype=np.float64).ravel()
    if z.size != n:
        return 1e12, np.zeros(max(1, z.size))

    z = np.clip(z, -_G_BOUND, _G_BOUND)
    a = _softmax(z)

    fa = np.fft.rfft(a, N)
    c = np.fft.irfft(fa * fa, N)[:L].real
    c = np.maximum(c, 0.0)

    m = float(np.max(c))
    if not np.isfinite(m):
        return 1e12, np.zeros(n)

    diff = (c - m) / tau
    diff = np.clip(diff, -700.0, 0.0)
    e = np.exp(diff)
    se = float(np.sum(e))
    if se <= 0 or not np.isfinite(se):
        return 1e12, np.zeros(n)

    p = e / se
    fp = np.fft.rfft(p, N)
    h = np.fft.irfft(fp * np.conjugate(fa), N)[:n].real
    s_h = float(np.dot(h, a))
    g = a * (h - s_h)
    grad = 2.0 * g

    val = m + tau * np.log(se)
    scale = float(n) * float(n)
    val *= scale
    grad *= scale

    pen = 1e-10
    val += 0.5 * pen * float(np.dot(z, z))
    grad += pen * z

    if not np.isfinite(val):
        return 1e12, np.zeros(n)
    return float(val), grad


def _subgrad_polish(a0, deadline, max_iter=5000):
    """Polish using exact-max subgradient in log-space with adaptive step size."""
    n = _G_N
    a = np.asarray(a0, dtype=np.float64).ravel().copy()
    a = np.maximum(a, 0.0)
    s = float(np.sum(a))
    if s <= 1e-12:
        return a0
    a /= s

    z = np.log(np.maximum(a, 1e-300))
    z -= np.mean(z)
    z = np.clip(z, -_G_BOUND, _G_BOUND)

    best_a = a.copy()
    best_score = _fast_score(a)
    best_maxc = float(np.max(_fft_conv(a)))

    rng = np.random.default_rng(12345)
    eta = 0.5
    no_improve = 0

    for it in range(max_iter):
        if time.monotonic() > deadline:
            break

        c = _fft_conv(a)
        F = float(np.max(c))

        if F < best_maxc - 1e-14:
            best_maxc = F
            best_a = a.copy()
            best_score = 2.0 * n * F
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                # Random jump
                jump_scale = 0.1 * (1 + no_improve / 500.0)
                z = np.clip(z + jump_scale * rng.standard_normal(n), -_G_BOUND, _G_BOUND)
                a = _softmax(z)
                no_improve = 0
                # Re-check
                c = _fft_conv(a)
                F = float(np.max(c))
                if F < best_maxc:
                    best_maxc = F
                    best_a = a.copy()
                    best_score = 2.0 * n * F
                    no_improve = 0

        # Compute subgradient: use top-5 active positions for stability
        top_k = np.argsort(c)[-5:]
        weights = np.zeros(2 * n - 1)
        top_vals = c[top_k]
        top_diffs = (top_vals - top_vals.max()) / 1e-4
        top_diffs = np.clip(top_diffs, -50, 0)
        w_top = np.exp(top_diffs)
        w_top /= w_top.sum()

        # subgrad[j] = sum_{k in active} w_k * a[k-j]
        # Compute via FFT: for each active k, a[k-j] is a shifted version of a
        subgrad = np.zeros(n)
        for idx, k in enumerate(top_k):
            k = int(k)
            for j in range(n):
                m = k - j
                if 0 <= m < n:
                    subgrad[j] += w_top[idx] * a[m]

        # Convert to log-space gradient
        s_sub = float(np.dot(a, subgrad))
        z_grad = a * (subgrad - s_sub)

        # Adaptive step size
        z_new = np.clip(z - eta * z_grad, -_G_BOUND, _G_BOUND)
        a_new = _softmax(z_new)
        F_new = float(np.max(_fft_conv(a_new)))

        if F_new < F - 1e-15:
            z = z_new
            a = a_new
            eta = min(eta * 1.1, 3.0)
        else:
            eta = max(eta * 0.7, 0.01)

        if it % 200 == 0:
            print(f"  subgrad it={it} score={2*n*F:.9f} best={best_score:.9f} eta={eta:.4f}")

    return best_a


def _lp_mirror_step(a, deadline):
    """
    LP-inspired global step: find b >= 0 maximizing sum(b) subject to (a*b)[k] <= M.
    Use projected subgradient on the dual for efficiency.
    """
    n = _G_N
    M = float(np.max(_fft_conv(a)))
    if M <= 0:
        return a

    # Dual: min M * sum(y) s.t. (a⋆y)[j] >= 1, y >= 0
    # (a⋆y)[j] = sum_m a[m] * y[m+j]
    # Use multiplicative weights
    y = np.ones(2 * n - 1) / (2 * n - 1)
    N = _G_NFFT

    fa = np.fft.rfft(a, N)

    for it in range(200):
        if time.monotonic() > deadline - 10:
            break

        # Compute v[j] = sum_m a[m] * y[m+j] for j = 0,...,n-1
        y_rev = y[::-1]
        fy_rev = np.fft.rfft(y_rev, N)
        conv = np.fft.irfft(fa * fy_rev, N)[:2 * n - 1].real
        v = conv[::-1][:n]

        min_v = float(np.min(v))
        if min_v >= 1.0 - 1e-6:
            break

        # Multiplicative weights update
        violations = np.maximum(0.0, 1.0 - v)
        if np.sum(violations) < 1e-12:
            break

        # d[k] = sum_j violations[j] * a[k-j] (valid range)
        fv = np.fft.rfft(violations, N)
        d = np.fft.irfft(fv * fa, N)[:2 * n - 1].real
        d = np.maximum(d, 1e-30)

        eta_mw = 1.0
        y = y * (d ** eta_mw)
        y_total = float(np.sum(y))
        if y_total > 0 and np.isfinite(y_total):
            y /= y_total

    # Extract primal solution: b[j] > 0 where v[j] is tight
    # Use KKT: b[j] proportional to (dual slack) which is 0 where v[j] = 1
    # Simple heuristic: b[j] = max(0, 1 - v[j] + tol) doesn't work
    # Instead, use the fact that optimal b has sum = M * sum(y)
    # and b[j] > 0 only where v[j] <= 1 + epsilon
    tight_mask = v <= 1.0 + 0.1 * (1.0 - np.min(v) + 1e-10)
    b = np.zeros(n)
    b[tight_mask] = 1.0
    # Refine: solve a smaller QP or just normalize
    s = float(np.sum(b))
    if s <= 0:
        b[:] = 1.0 / n
    else:
        b /= s

    # Verify and blend
    score_b = _fast_score(b)
    score_a = _fast_score(a)
    if score_b < score_a:
        return b
    else:
        # Blend
        blend = 0.3 * b + 0.7 * a
        blend /= blend.sum()
        return blend

    # Random smooth (low-frequency Fourier)
    for _ in range(15):
        f = np.ones(n)
        for m in range(1, 12):
            f += 0.15 * rng.standard_normal() * np.sin(2 * np.pi * m * u + rng.uniform(0, 2 * np.pi))
            f += 0.15 * rng.standard_normal() * np.cos(2 * np.pi * m * u + rng.uniform(0, 2 * np.pi))
        f = np.maximum(f, 0.01)
        add_from_f(f)

    # "Anti-peaked" tries: sequences that are high at edges, low in middle
    for A in (0.5, 1.0, 1.5):
        f = 1.0 + A * np.cos(np.pi * u)  # high at edges, low in middle
        f = np.maximum(f, 0.01)

            maxfun = 150
            maxiter = 60

        if time_left < maxfun * 0.05:
            maxfun = max(30, int(time_left * 10))
            maxiter = max(10, maxfun // 3)

        z0 = np.clip(z0 - np.mean(z0), -_G_BOUND, _G_BOUND)

        # Use adaptive tau: start larger, decrease
        tau_start = 0.01 if idx == 0 else 0.001
        tau_end = 1e-7

        def _adaptive_obj(z_inner, tau_ref=[tau_start]):
            # Decrease tau over the course of optimization
            frac = min(1.0, (tau_ref[0] - tau_end) / (tau_start - tau_end + 1e-30))
            tau_ref[0] = tau_start * (tau_end / tau_start) ** frac
            return _obj_grad(z_inner, tau=max(tau_ref[0], tau_end))

        try:
            res = minimize(
                _adaptive_obj,
                z0,
                jac=True,
                method="L-BFGS-B",
                bounds=_G_BOUNDS,
                options={"maxfun": maxfun, "maxiter": maxiter,
                         "ftol": 1e-14, "gtol": 1e-12, "disp": False}
            )
            if hasattr(res, 'x') and res.x.size == n:
                z_new = np.clip(res.x, -_G_BOUND, _G_BOUND)
                a = _softmax(z_new)
                sc = _fast_score(a)
                if np.isfinite(sc):
                    if sc < best_score:
                        best_score = sc
                        best_a = a.copy()
                        print(f"  L-BFGS start {idx}: score={sc:.9f} (t={time.monotonic()-t0:.0f}s)")
                    candidates.append((sc, a.copy(), z_new.copy()))
        except Exception as e:
            pass

        if time.monotonic() - t0 > 10:
            pass  # just for timing awareness

    # Keep top candidates
    candidates.sort(key=lambda x: x[0])
    # Deduplicate (rough)
    unique_cands = []
    for sc, a, z in candidates:
        is_dup = False
        for _, a2, _ in unique_cands:
            if np.max(np.abs(a - a2)) < 1e-6:
                is_dup = True
                break
        if not is_dup:
            unique_cands.append((sc, a, z))
        if len(unique_cands) >= 6:
            break
    candidates = unique_cands

    # Subgradient polish on top candidates
    for ci, (sc, a, z) in enumerate(candidates):
        time_left = deadline - time.monotonic()
        if time_left < 30:
            break

        polish_time = min(time_left - 10, 600 if ci == 0 else 200)
        if polish_time < 10:
            break

        print(f"  Polishing candidate {ci} (score={sc:.9f}) with {polish_time:.0f}s budget")
        try:
            a_polished = _subgrad_polish(a, time.monotonic() + polish_time)
            sc_polished = _fast_score(a_polished)
            if np.isfinite(sc_polished) and sc_polished < best_score:
                best_score = sc_polished
                best_a = a_polished.copy()
                print(f"    Polish improved: {sc_polished:.9f} (t={time.monotonic()-t0:.0f}s)")
        except Exception:
            pass

    # LP mirror step on best
    time_left = deadline - time.monotonic()
    if time_left > 30:
        print(f"  LP mirror step (score={best_score:.9f})")
        try:
            a_mirror = _lp_mirror_step(best_a, deadline)
            sc_mirror = _fast_score(a_mirror)
            if np.isfinite(sc_mirror) and sc_mirror < best_score:
                best_score = sc_mirror
                best_a = a_mirror.copy()
                print(f"    Mirror improved: {sc_mirror:.9f}")
            # Polish the mirror result
            a_m_pol = _subgrad_polish(a_mirror, deadline)
            sc_m_pol = _fast_score(a_m_pol)
            if np.isfinite(sc_m_pol) and sc_m_pol < best_score:
                best_score = sc_m_pol
                best_a = a_m_pol.copy()
                print(f"    Mirror+polish: {sc_m_pol:.9f}")
        except Exception:
            pass

    # Final random restart attempts if time remains
    time_left = deadline - time.monotonic()
    if time_left > 60:
        print(f"  Random restart phase: {time_left:.0f}s remaining")
        for attempt in range(20):
            if time.monotonic() > deadline - 30:
                break
            # Perturb best solution
            z_best = np.log(np.maximum(best_a, 1e-300))
            z_best -= np.mean(z_best)
            z_pert = np.clip(z_best + 0.3 * rng.standard_normal(n), -_G_BOUND, _G_BOUND)
            a_pert = _softmax(z_pert)

            # Quick L-BFGS
            try:
                res = minimize(
                    _obj_grad, z_pert, jac=True,
                    method="L-BFGS-B", bounds=_G_BOUNDS,
                    options={"maxfun": 200, "maxiter": 80,
                             "ftol": 1e-13, "gtol": 1e-11, "disp": False}
                )
                if hasattr(res, 'x') and res.x.size == n:
                    a_r = _softmax(np.clip(res.x, -_G_BOUND, _G_BOUND))
                    sc_r = _fast_score(a_r)
                    if np.isfinite(sc_r) and sc_r < best_score:
                        best_score = sc_r
                        best_a = a_r.copy()
                        print(f"    Restart {attempt}: {sc_r:.9f}")
            except Exception:
                pass

    # Final validation
    best_a = np.asarray(best_a, dtype=np.float64).ravel()
    best_a = np.nan
