            g = np.asarray(g, dtype=np.float64).ravel()
            if g.size > 10:
                g = np.clip(g, 1e-300, 1e6)
                s = g.sum()
                if s > 0 and np.all(np.isfinite(g)):
                    # Resample to length n via interpolation in log-space
                    g = g / s
                    lg = np.log(np.maximum(g, 1e-300))
                    lg = lg - lg.mean()
                    x_old = np.linspace(0.0, 1.0, g.size)
                    z = np.interp(np.linspace(0.0, 1.0, n), x_old, lg)
                    z = np.clip(z, -_BOUNDS, _BOUNDS)
                    a = _softmax(z)
                    sc = _score(a)
                    if np.isfinite(sc):
                        starts.append((sc, z.copy()))
    except Exception:
        pass

    # Two-level blocks with varied split positions and ratios
    for split_frac in (0.3, 0.4, 0.45, 0.5, 0.55, 0.6, 0.7):
        split = int(n * split_frac)
        for ratio in (1.5, 2.0, 3.0, 4.0, 6.0, 10.0):
            a = np.empty(n)
            a[:split] = 1.0
            a[split:] = ratio
            add_from_a(a)

    # Smooth sigmoid profiles
    for center in (0.35, 0.4, 0.45, 0.5, 0.55, 0.6):
        for width in (0.02, 0.05, 0.1, 0.2, 0.3):
            for amp in (0.8, 1.5, 3.0):
                a = 0.1 + amp * (1.0 / (1.0 + np.exp(-(u - center) / max(width, 1e-4))))
                add_from_a(a)

    # Linear ramps
    for A in (0.5, 1.0, 2.0, 4.0):
        for offset in (0.0, 0.3, 0.6):
            a = 0.1 + A * np.clip((u - offset) / (1.0 - offset + 1e-6), 0, 1)
            add_from_a(a)

    # Random smooth (low-frequency Fourier) starts
    for _ in range(8):
        a = np.ones(n)
        for k in range(1, 12):
            a += 0.15 * rng.standard_normal() * np.cos(2 * np.pi * k * u + rng.standard_normal())
        a = np.maximum(a, 0.01)
        add_from_a(a)

    # Trim to top 20 by score
    starts.sort(key=lambda x: x[0])
    return starts[:20]


def _adam_optimize(z0, max_iter, lr0, deadline):
    """Run Adam optimization from z0 with cosine-annealed LR."""
    n = _N
    z = np.asarray(z0, dtype=np.float64).ravel().copy()
    z = np.clip(z, -_BOUNDS, _BOUNDS)
    z = z - z.mean()  # remove gauge freedom

    a = _softmax(z)
    score, grad_z = _score_grad_z(z)
    if not np.isfinite(score):
        return z0, np.inf

    best_z = z.copy()
    best_score = score

    m_adam = np.zeros(n, dtype=np.float64)
    v_adam = np.zeros(n, dtype=np.float64)

    for t in range(max_iter):
        if time.monotonic() > deadline:
            break

        # Recompute gradient (in case z drifted)
        score, grad_z = _score_grad_z(z)
        if not np.isfinite(score) or not np.all(np.isfinite(grad_z)):
            # Revert
            z = best_z.copy()
            score, grad_z = _score_grad_z(z)
            if not np.isfinite(score):
                break
            m_adam = np.zeros(n, dtype=np.float64)
            v_adam = np.zeros(n, dtype=np.float64)
            continue

        if score < best_score:
            best_score = score
            best_z = z.copy()

        z, m_adam, v_adam = _adam_step(z, grad_z, m_adam, v_adam, t, max_iter, lr0)

        # Periodic reheating: small random perturbation
        if t % 2000 == 1999 and t > 1000:
            z = z + 0.3 * np.random.default_rng(t).standard_normal(n)
            z = np.clip(z - z.mean(), -_BOUNDS, _BOUNDS)

        if t % 2000 == 0:
            a = _softmax(z)
            cur_sc = _score(a)
            print(f"  Adam it={t} score={cur_sc:.9f} best={best_score:.9f} "
                  f"time_left={max(0.0, deadline - time.monotonic()):.0f}s")

    return best_z, best_score


def _polish(z0, deadline, rng):
    """Polish with exact active-set gradient + backtracking line search."""
    n = _N
    z = np.asarray(z0, dtype=np.float64).ravel().copy()
    z = np.clip(z - z.mean(), -_BOUNDS, _BOUNDS)

    a = _softmax(z)
    score, grad_z = _score_grad_z(z)
    if not np.isfinite(score):
        return z0

    best_z = z.copy()
    best_score = score
    no_improve = 0
    it = 0

    while time.monotonic() < deadline and no_improve < 300:
        score, grad_z = _score_grad_z(z)
        if not np.isfinite(score) or not np.all(np.isfinite(grad_z)):
            break

        if score < best_score:
            best_score = score
            best_z = z.copy()
            no_improve = 0
        else:
            no_improve += 1

        # Scale gradient for numerical stability
        g_norm = np.linalg.norm(grad_z)
        if g_norm < 1e-14:
            break
        g = grad_z / g_norm

        # Backtracking line search
        step_accepted = False
        for eta in (0.01, 0.03, 0.1, 0.3, 1.0, 3.0, 10.0, 30.0):
            z_t = np.clip(z - eta * grad_z, -_BOUNDS, _BOUNDS)
            a_t = _softmax(z_t)
            sc_t = _score(a_t)
            if np.isfinite(sc_t) and sc_t < score - 1e-12:
                z = z_t
                step_accepted = True
                break

        if not step_accepted:
            # Try random escape
            for scale in (0.5, 1.5, 4.0, 10.0):
                z_t = np.clip(z + scale * rng.standard_normal(n), -_BOUNDS, _BOUNDS)
                z_t = z_t - z_t.mean()
                a_t = _softmax(z_t)
                sc_t = _score(a_t)
                if np.isfinite(sc_t) and sc_t < best_score - 1e-10:
                    z = z_t
                    step_accepted = True
                    no_improve = 0
                    break

        it += 1
        if it % 500 == 0:
            a = _softmax(z)
            cur = _score(a)
            print(f"  Polish it={it} cur={cur:.9f} best={best_score:.9f} "
                  f"time_left={max(0.0, deadline - time.monotonic()):.0f}s")

    return best_z


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for non-negative coefficient vector minimizing 2*n*max_conv/sum^2.
    Uses Adam in log-space with exact active-set gradients, multiple restarts,
    and final polishing phase.
    """
    budget_s = max(60.0, float(budget_s))
    t0 = time.monotonic()
    deadline = t0 + budget_s - 10.0

    n = 16384
    _init_state(n)

    rng = np.random.default_rng(int(seed))
    print(f"Starting search: n={n}, budget={budget_s:.0f}s")

    # Generate diverse starting points
    starts = _make_starts(rng, n)
    if not starts:
        # Fallback: simple step
        a = np.empty(n)
        a[:n // 2] = 1.0
        a[n // 2:] = 3.0
        z0 = np.log(a / a.sum())
        z0 = z0 - z0.mean()
        starts = [(2.0, z0)]

    starts.sort(key=lambda x: x[0])
    print(f"Generated {len(starts)} starts, best initial score={starts[0][0]:.9f}")

    best_z = starts[0][1].copy()
    best_score = starts[0][0]

    # Phase 1: Adam optimization from top starts
    # Allocate time: 60% for Adam, 30% for polish, 10% buffer
    adam_deadline = t0 + 0.55 * (budget_s - 10.0)
    polish_deadline = deadline

    num_optimizations = min(6, len(starts))
    for idx in range(num_optimizations):
        if time.monotonic() > adam_deadline:
            break

        z0 = starts[idx][1]
        # More iterations for better starts
        if idx < 2:
            max_iter = 30000
            lr0 = 0.005
        elif idx < 4:
            max_iter = 20000
            lr0 = 0.004
        else:
            max_iter = 10000
            lr0 = 0.003

        print(f"Adam start {idx} (init score={starts[idx][0]:.9f}), max_iter={max_iter}")
        try:
            z_new, sc_new = _adam_optimize(z0, max_iter, lr0, adam_deadline)
            if np.isfinite(sc_new) and sc_new < best_score:
                best_score = sc_new
                best_z = z_new.copy()
                print(f"  New best: {best_score:.9f}")
        except Exception as e:
            print(f"  Adam start {idx} failed: {e}")

    # Phase 2: Polish the best solution
    if time.monotonic() < deadline - 30:
        print(f"Polishing best (score={best_score:.9f})...")
        try:
            z_polished = _polish(best_z, deadline, rng)
            a_check = _softmax(z_polished)
            sc_check = _score(a_check)
            if np.isfinite(sc_check) and sc_check < best_score:
                best_score = sc_check
                best_z = z_polished.copy()
                print(f"  Polished score: {best_score:.9f}")
        except Exception as e:
            print(f"  Polish failed: {e}")

    # Phase 3: One more Adam pass from the polished solution if time remains
    if time.monotonic() < deadline - 60:
        print("Second Adam pass from polished solution...")
        try:
            z2, sc2 = _adam_optimize(best_z, 15000, 0.002, deadline - 30)
            if np.isfinite(sc2) and sc2 < best_score:
                best_score = sc2
                best_z = z2.copy()
                print(f"  Second pass score: {best_score:.9f}")
        except Exception as e:
            print(f"  Second pass failed: {e}")

    # Convert to final sequence
    a = _softmax(best_z)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if s < 0.01 or not np.isfinite(s):
        a = np.ones(n) / n
    else:
        a = a / s  # normalize sum to 1

    # Final validation
    try:
        exact = evaluate_sequence(a.tolist())
        print(f"Official evaluator score: {exact:.9f}")
        if not np.isfinite(exact):
            # Something went wrong, use a safe fallback
            a = np.ones(n) / n
    except Exception:
        pass

    # Ensure all values are in valid range
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s < 0.01:
        a = np.ones(n) / n

    print(f"Final score estimate: {2.0 * n * float(_fft_conv(a).max()) / (a.sum()**2):.9f}")
    print(f"Total time: {time.monotonic() - t0:.1f}s")

    return [float(v) for v in a.tolist()]
