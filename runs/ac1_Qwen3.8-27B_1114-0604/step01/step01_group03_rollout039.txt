```
def _mws_direction(f, L,m,n, prior, y_init, inner=40):
    f=_normalize(f); prior=_normalize(prior)

       y_init=_normalize(y_init)

    y=y_init.copy()
```

```python
"""
Search for non-negative step heights minimizing
    2*n*max(conv(a, a)) / (sum(a)**2).

The algorithm combines:
  - exact FFT-based active-set gradient descents,
  - sparse LP best-response directions over the currently active peaks,
  - multiplicative-weights game directions for exploration,
  - line searches and adaptive reheating.

The final answer is the best exact candidate found before the time budget
is exhausted.
"""

import time
import numpy as np

try:
    import scipy.sparse as sp
    import scipy.optimize as sco
    _HAS_SCIPY = True
except Exception:
    _HAS_SCIPY = False

_INF = float("inf")


def _get_base():
    try:
        b = height_sequence_1
    except NameError:
        return None
    except Exception:
        return None
    try:
        a = np.asarray(b, dtype=np.float64).ravel()
    except Exception:
        return None
    if a.size == 0:
        return None
    if not np.all(np.isfinite(a)):
        a = np.where(np.isfinite(a), a, 0.0)
    return np.maximum(a, 0.0)


def _normalize(a):
    a = np.asarray(a, dtype=np.float64).ravel()
    if a.size == 0:
        return a
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    s = float(a.sum())
    if not np.isfinite(s) or s <= 0.0:
        a = np.ones_like(a, dtype=np.float64)
        s = float(a.sum())
    a = a / s
    a = np.maximum(a, 0.0)
    s2 = float(a.sum())
    if s2 > 0.0 and np.isfinite(s2):
        a = a / s2
    return a


def _fft_params(n):
    m = 2 * int(n) - 1
    L = 1
    while L < m:
        L <<= 1
    return L, m


def _conv_self(a, L, m):
    F = np.fft.rfft(a, L)
    c = np.fft.irfft(F * F, L)[:m]
    c = np.ascontiguousarray(c, dtype=np.float64)
    np.maximum(c, 0.0, out=c)
    return c


def _score_only(a, L, m, n):
    c = _conv_self(a, L, m)
    if c.size == 0:
        return _INF
    M = float(np.max(c))
    return float(2.0 * float(n) * M)


def _score_components(a, L, m, n):
    c = _conv_self(a, L, m)
    M = float(np.max(c)) if c.size else 0.0
    return c, float(2.0 * float(n) * M)


def _mix(a, b, lam):
    if lam <= 0.0:
        return a.copy()
    if lam >= 1.0:
        return b.copy()
    x = (1.0 - lam) * a + lam * b
    return _normalize(x)


def _line_search(a, d, base_score, L, m, n):
    best_a = a
    best_score = float(base_score)
    lams = (0.05, 0.10, 0.20, 0.35, 0.60, 1.00)
    for lam in lams:
        if lam <= 0.0:
            continue
        h = _mix(a, d, lam)
        sc = _score_only(h, L, m, n)
        if np.isfinite(sc) and sc < best_score:
            best_score = float(sc)
            best_a = h
    return best_a, best_score


def _uniform_vec(n):
    return np.full(n, 1.0 / float(n), dtype=np.float64)


def _random_prior(n, rng):
    t = np.linspace(0.0, 1.0, int(n))
    z = rng.normal(0.0, 0.28, int(n))

    z += rng.uniform(-0.7, 0.7) * t
    z += rng.uniform(-0.7, 0.7) * (1.0 - t)

    kmax = int(rng.integers(2, 7))
    for k in range(1, kmax + 1):
        amp = rng.uniform(0.05, 0.55) / max(1.0, np.sqrt(float(k)))
        ph = rng.uniform(0.0, 2.0 * np.pi)
        z += amp * np.cos(2.0 * np.pi * float(k) * t + ph)

    noise = rng.normal(0.0, 0.12, int(n))
    if n > 5:
        noise = np.convolve(noise, np.ones(5, dtype=np.float64) / 5.0, mode="same")
    z += noise

    z = np.where(np.isfinite(z), z, 0.0)
    p = np.exp(z - float(np.max(z)))
    p = np.maximum(p, 1e-300)
    return _normalize(p)


def _resample(arr, n):
    arr = np.asarray(arr, dtype=np.float64).ravel()
    if arr.size == 0:
        return _uniform_vec(n)
    arr = np.where(np.isfinite(arr), arr, 0.0)
    if arr.size == int(n):
        return _normalize(arr)
    if arr.size == 1:
        return _uniform_vec(n)
    x_old = np.linspace(0.0, 1.0, arr.size)
    x_new = np.linspace(0.0, 1.0, int(n))
    y = np.interp(x_new, x_old, arr)
    return _normalize(y)


def _make_initial_candidates(n, base_seq, rng):
    out = []
    n = int(n)

    if base_seq is not None:
        try:
            out.append(_resample(base_seq, n))
        except Exception:
            pass

    x = np.arange(n, dtype=np.float64)
    left = x + 1.0
    right = np.maximum(n - x, 1.0)

    for alpha in (0.35, 0.45, 0.50, 0.60, 0.75):
        try:
            out.append(_normalize(left ** (-alpha)))
            out.append(_normalize(right ** (-alpha)))
        except Exception:
            pass

    try:
        out.append(_normalize((x + 1.0) ** (-0.5) * right ** (-0.25)))
        out.append(_normalize((x + 1.0) ** (-0.25) * right ** (-0.5)))
        out.append(_normalize((x + 1.0) ** (-0.75) * right ** (-0.75)))
    except Exception:
        pass

    out.append(_uniform_vec(n))

    for _ in range(6):
        try:
            out.append(_random_prior(n, rng))
        except Exception:
            pass

    # Remove obvious duplicates by a lightweight key.
    uniq = []
    seen = set()
    for a in out:
        a = _normalize(a)
        if a.size != n:
            continue
        key = tuple(np.round(a.astype(np.float64), 12)[:8])
        if key in seen:
            continue
        seen.add(key)
        uniq.append(a)
    if not uniq:
        uniq.append(_uniform_vec(n))
    return uniq


def _top_indices(c, K):
    c = np.asarray(c, dtype=np.float64).ravel()
    K = int(K)
    if K <= 0 or c.size == 0:
        return np.array([], dtype=np.int64)
    if K >= c.size:
        return np.arange(c.size, dtype=np.int64)
    return np.argpartition(c, c.size - K)[-K:].astype(np.int64)


def _active_index(c, M, minK, maxK):
    c = np.asarray(c, dtype=np.float64).ravel()
    if c.size == 0 or M <= 0.0:
        return np.arange(c.size, dtype=np.int64)
    thr = 0.90 * float(M)
    idx = np.nonzero(c >= thr)[0].astype(np.int64)
    if idx.size == 0:
        return _top_indices(c, minK)
    if idx.size < int(minK):
        return _top_indices(c, minK)
    if idx.size > int(maxK):
        return _top_indices(c, maxK)
    return idx


def _soft_weights(v, beta):
    v = np.asarray(v, dtype=np.float64).ravel()
    if v.size == 0:
        return v
    vmax = float(np.max(v))
    z = float(beta) * (v - vmax)
    z = np.clip(z, -50.0, 0.0)
    e = np.exp(z)
    s = float(e.sum())
    if not np.isfinite(s) or s <= 0.0:
        e = np.ones_like(v)
        s = float(e.size)
    return e / s


def _cross_y(a, yfull, L, m, n):
    Frev = np.fft.rfft(a[::-1], L)
    Y = np.fft.rfft(yfull, L)
    h = np.fft.irfft(Y * Frev, L)[(int(n) - 1):(int(n) - 1 + int(n))]
    h = np.ascontiguousarray(h, dtype=np.float64)
    np.maximum(h, 0.0, out=h)
    return h


def _soft_block(a, L, m, n, rng, time_limit):
    time_limit = float(time_limit)
    if time_limit <= 0.0:
        a = _normalize(a)
        return a, _score_only(a, L, m, n)

    deadline = time.monotonic() + time_limit
    n = int(n)
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score_only(a, L, m, n)

    beta = 1.0
    eta = 0.25
    stale = 0
    it = 0
    minK = max(32, (m // 32))
    maxK = max(256, (m // 4))
    uniform = _uniform_vec(n)

    while time.monotonic() < deadline - 0.5:
        c = _conv_self(a, L, m)
        M = float(np.max(c)) if c.size else 0.0
        cur_score = float(2.0 * float(n) * M)

        if cur_score < best_score - 1e-13:
            best_score = cur_score
            best_a = a.copy()

        if M <= 0.0:
            break

        idx = _active_index(c, M, minK, maxK)
        if idx.size == 0:
            break

        vact = float(n) * c[idx]
        w = _soft_weights(vact, beta)
        yfull = np.zeros(int(m), dtype=np.float64)
        yfull[idx] = w
        s = float(yfull.sum())
        if not np.isfinite(s) or s <= 0.0:
            yfull.fill(1.0 / float(m))
        else:
            yfull /= s

        h = _cross_y(a, yfull, L, m, n)
        hmax = float(h.max()) if h.size else 0.0
        hmin = float(h.min()) if h.size else 0.0
        if hmax <= 0.0:
            break

        hn = (h - hmin) / (hmax - hmin + 1e-300)
        d = a * np.exp(-eta * hn)
        d = _normalize(0.97 * d + 0.03 * uniform)

        new_a, new_sc = _line_search(a, d, cur_score, L, m, n)
        if np.isfinite(new_sc) and new_sc < cur_score - 1e-14:
            a = new_a
            if new_sc < best_score - 1e-13:
                best_score = float(new_sc)
                best_a = a.copy()
            eta = min(1.8, eta * 1.12)
            stale = 0
        else:
            eta = max(0.03, eta * 0.75)
            stale += 1
            if stale > 70:
                mix = 0.08 + 0.18 * float(rng.random())
                a = _normalize((1.0 - mix) * best_a + mix * _random_prior(n, rng))
                beta = 0.5
                eta = 0.20
                stale = 0

        it += 1
        beta = min(15.0, beta * 1.005)
        if it % 300 == 120:
            beta = max(0.3, beta * 0.30)

    return best_a, best_score


def _active_y_polish(a, yfull, L, m, n, rng, time_limit):
    time_limit = float(time_limit)
    n = int(n)
    a = _normalize(a)
    yfull = _normalize(yfull)
    if float(yfull.sum()) <= 0.0:
        return a, _score_only(a, L, m, n)

    if time_limit <= 0.0:
        return a, _score_only(a, L, m, n)

    deadline = time.monotonic() + time_limit
    Y = np.fft.rfft(yfull, L)
    best_a = a.copy()
    best_score = _score_only(a, L, m, n)
    eta = 0.25
    stale = 0
    uniform = _uniform_vec(n)

    while time.monotonic() < deadline - 0.5:
        Frev = np.fft.rfft(a[::-1], L)
        h = np.fft.irfft(Y * Frev, L)[(n - 1):(n - 1 + n)]
        h = np.ascontiguousarray(h, dtype=np.float64)
        np.maximum(h, 0.0, out=h)
        if h.size == 0 or float(h.max()) <= 0.0:
            break

        cur_score = _score_only(a, L, m, n)
        if cur_score < best_score - 1e-13:
            best_score = float(cur_score)
            best_a = a.copy()

        hmax = float(h.max())
        hmin = float(h.min())
        hn = (h - hmin) / (hmax - hmin + 1e-300)

        d = a * np.exp(-eta * hn)
        d = _normalize(0.97 * d + 0.03 * uniform)

        new_a, new_sc = _line_search(a, d, cur_score, L, m, n)
        if np.isfinite(new_sc) and new_sc < cur_score - 1e-14:
            a = new_a
            if new_sc < best_score - 1e-13:
                best_score = float(new_sc)
                best_a = a.copy()
            eta = min(1.8, eta * 1.15)
            stale = 0
        else:
            eta = max(0.03, eta * 0.70)
            stale += 1
            if stale > 35:
                break

    return best_a, best_score


def _mws_direction(f, L, m, n, prior, y_init, inner):
    f = _normalize(f)
    prior = _normalize(prior)
    inner = max(1, int(inner))
    n = int(n)

    if y_init is None:
        y = np.full(int(m), 1.0 / float(m), dtype=np.float64)
    else:
        y = _normalize(y_init)

    x = prior.copy()
    Ff = np.fft.rfft(f, L)
    Frev = np.fft.rfft(f[::-1], L)

    logp = np.log(np.maximum(prior, 1e-300))
    logp -= float(np.max(logp))

    avg = np.zeros(n, dtype=np.float64)
    eta_start = 1.0
    eta_end = 25.0

    for it in range(inner):
        frac = 0.0 if inner <= 1 else float(it) / float(inner - 1)
        eta = eta_start + (eta_end - eta_start) * frac

        X = np.fft.rfft(x, L)
        u = np.fft.irfft(Ff * X, L)[:m]
        u = np.ascontiguousarray(u, dtype=np.float64)
        np.maximum(u, 0.0, out=u)
        u = float(n) * u
        y = _soft_weights(u, eta)

        Y = np.fft.rfft(y, L)
        s = np.fft.irfft(Y * Frev, L)[(n - 1):(n - 1 + n)]
        s = np.ascontiguousarray(s, dtype=np.float64)
        np.maximum(s, 0.0, out=s)
        s = float(n) * s

        z = -eta * s + 0.7 * logp
        z -= float(np.max(z))
        z = np.clip(z, -50.0, 0.0)
        e = np.exp(z)
        ss = float(e.sum())
        if not np.isfinite(ss) or ss <= 0.0:
            x = prior.copy()
        else:
            x = e / ss

        avg += x

    d = 0.5 * (avg / float(inner)) + 0.5 * x
    return _normalize(d)


def _mws_search(a, L, m, n, rng, time_limit, prior=None):
    time_limit = float(time_limit)
    n = int(n)
    a = _normalize(a)
    if time_limit <= 0.0:
        return a, _score_only(a, L, m, n)

    deadline = time.monotonic() + time_limit
    best_a = a.copy()
    best_score = _score_only(a, L, m, n)

    if prior is None:
        prior = _random_prior(n, rng)
    y_init = None

    fails = 0
    while time.monotonic() < deadline - 1.0:
        try:
            c = _conv_self(a, L, m)
            M = float(np.max(c)) if c.size else 0.0
            if M > 0.0:
                idx = _active_index(c, M, 64, max(256, m // 4))
                if idx.size:
                    y_init = np.zeros(int(m), dtype=np.float64)
                    y_init[idx] = _soft_weights(float(n) * c[idx], 10.0)
                    s = float(y_init.sum())
                    if s > 0.0 and np.isfinite(s):
                        y_init /= s
                    else:
                        y_init = None
        except Exception:
            y_init = None

        try:
            d = _mws_direction(best_a, L, m, n, prior, y_init, 50)
        except Exception:
            d = _random_prior(n, rng)

        new_a, new_sc = _line_search(best_a, d, best_score, L, m, n)
        if np.isfinite(new_sc) and new_sc < best_score - 1e-14:
            best_score = float(new_sc)
            best_a = new_a
            fails = 0
            prior = _normalize(0.7 * best_a + 0.3 * _random_prior(n, rng))
        else:
            fails += 1
            if fails >= 5:
                break
            prior = _normalize(0.6 * best_a + 0.4 * _random_prior(n, rng))

    return best_a, best_score


def _lp_get_y(a, c, K):
    if not _HAS_SCIPY:
        return None
    a = np.asarray(a, dtype=np.float64).ravel()
    c = np.asarray(c, dtype=np.float64).ravel()
    K = int(K)
    if K <= 0 or a.size == 0 or c.size == 0 or float(c.max()) <= 0.0:
        return None

    n = int(a.size)
    m = int(c.size)
    idx = _top_indices(c, K)
    if idx.size < 8:
        return None
    S = idx.astype(np.int64)
    Kc = int(S.size)

    rows = []
    cols = []
    data = []
    scale = float(n)

    for col, k in enumerate(S):
        i0 = max(0, int(k) - (n - 1))
        i1 = min(n - 1, int(k))
        if i1 < i0:
            continue
        length = i1 - i0 + 1
        j_start = int(k) - i0
        j_end = int(k) - i1
        if j_end == 0:
            vals = a[j_start::-1]
        else:
            vals = a[j_start:j_end - 1:-1]
        vals = np.asarray(vals, dtype=np.float64)
        if vals.size != length:
            continue
        rows.append(np.arange(i0, i1 + 1, dtype=np.int64))
        cols.append(np.full(length, col, dtype=np.int64))
        data.append(scale * vals)

    if not rows:
        return None

    row = np.concatenate(rows)
    col = np.concatenate(cols)
    dat = np.concatenate(data)

    B = sp.csr_matrix((dat, (row, col)), shape=(n, Kc))
    A_ones = sp.csr_matrix(np.ones((n, 1), dtype=np.float64))
    A_negB = sp.csr_matrix(-B)
    A_ub = sp.hstack([A_ones, A_negB])

    A_eq = np.zeros((1, Kc + 1), dtype=np.float64)
    A_eq[0, 1:] = 1.0
    b_eq = np.array([1.0], dtype=np.float64)
    b_ub = np.zeros(n, dtype=np.float64)
    c_vec = np.zeros(Kc + 1, dtype=np.float64)
    c_vec[0] = -1.0

    bounds = [(None, None)] + [(0.0, 1.0)] * Kc

    try:
        res = sco.linprog(
            c_vec,
            A_ub=A_ub,
            b_ub=b_ub,
            A_eq=A_eq,
            b_eq=b_eq,
            bounds=bounds,
            method="highs"
        )
    except Exception:
        return None

    if res is None or int(getattr(res, "status", -1)) != 0:
        return None

    y = np.asarray(res.x[1:], dtype=np.float64).copy()
    y = np.clip(y, 0.0, 1.0)
    s = float(y.sum())
    if not np.isfinite(s) or s <= 0.0:
        return None
    y /= s

    h = np.asarray(B.dot(y), dtype=np.float64).ravel()
    if h.size != n or not np.all(np.isfinite(h)):
        return None
    h = np.maximum(h, 0.0)

    yfull = np.zeros(m, dtype=np.float64)
    yfull[S] = y
    tau = float(h.min()) if h.size else 0.0
    return yfull, h, tau


def _lp_polish(a, L, m, n, rng, time_limit):
    time_limit = float(time_limit)
    n = int(n)
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score_only(a, L, m, n)
    if time_limit <= 0.0:
        return best_a, best_score

    deadline = time.monotonic() + time_limit
    K = 128 if n <= 7000 else 96

    try:
        c, _ = _score_components(a, L, m, n)
        lp = _lp_get_y(a, c, K)
    except Exception:
        lp = None

    if lp is not None:
        yfull, h, _ = lp
        try:
            yfull = _normalize(yfull)
            hmax = float(h.max())
            hmin = float(h.min())
            if hmax > 0.0:
                hn = (h - hmin) / (hmax - hmin + 1e-300)
                uniform = _uniform_vec(n)

                d1 = _normalize(0.90 * (a * np.exp(-0.8 * hn)) + 0.10 * uniform)
                new_a, new_sc = _line_search(best_a, d1, best_score, L, m, n)
                if np.isfinite(new_sc) and new_sc < best_score - 1e-14:
                    best_a = new_a
                    best_score = float(new_sc)

                tol = 1e-8 + 0.03 * max(1e-300, hmax - hmin)
                mask = h <= hmin + tol
                if int(mask.sum()) > max(64, n // 4):
                    kk = max(64, min(n // 8, 512))
                    low_idx = np.argpartition(h, kk)[:kk]
                    mask = np.zeros(n, dtype=bool)
                    mask[low_idx] = True
                if int(mask.sum()) > 0:
                    d2 = np.zeros(n, dtype=np.float64)
                    d2[mask] = 1.0
                    d2 = d2 / float(mask.sum())
                    d2 = _normalize(0.70 * d2 + 0.30 * a)
                    new_a, new_sc = _line_search(best_a, d2, best_score, L, m, n)
                    if np.isfinite(new_sc) and new_sc < best_score - 1e-14:
                        best_a = new_a
                        best_score = float(new_sc)
        except Exception:
            pass

    rem = float(max(0.0, deadline - time.monotonic()))
    if rem > 6.0:
        try:
            a2, sc2 = _active_y_polish(best_a, yfull, L, m, n, rng, min(45.0, rem - 2.0))
            if np.isfinite(sc2) and sc2 < best_score - 1e-14:
                best_a = a2
                best_score = float(sc2)
        except Exception:
            pass

    return best_a, best_score


def _optimize_candidate(a, n, rng, time_limit):
    time_limit = float(time_limit)
    n = int(n)
    L, m = _fft_params(n)
    a = _normalize(a)
    best_a = a.copy()
    best_score = _score_only(a, L, m, n)

    if time_limit <= 0.0:
        return best_a, best_score

    deadline = time.monotonic() + time_limit
    cur_a = best_a.copy()
    stale = 0
    cycle = 0

    while time.monotonic() < deadline - 25.0:
        rem = float(max(0.0, deadline - time.monotonic()))
        if rem < 25.0:
            break

        use_lp = (
            _HAS_SCIPY
            and rem > 300.0
            and (cycle % 3 == 1 or stale >= 2)
        )

        if use_lp:
            lim = min(80.0, 0.22 * rem)
            try:
                a2, sc2 = _lp_polish(cur_a, L, m, n, rng, lim)
            except Exception:
                a2, sc2 = cur_a, best_score

            if np.isfinite(sc2) and sc2 < best_score - 1e-12:
                best_a = a2
                best_score = float(sc2)
                cur_a = best_a.copy()
                stale = 0
            else:
                stale += 1
            cycle += 1
            continue

        lim = min(40.0, 0.18 * rem)
        try:
            a2, sc2 = _soft_block(cur_a, L, m, n, rng, lim)
        except Exception:
            a2, sc2 = cur_a, best_score

        if np.isfinite(sc2) and sc2 < best_score - 1e-12:
            best_a = a2
            best_score = float(sc2)
            cur_a = best_a.copy()
            stale = 0
        else:
            stale += 1

            if stale >= 3 and rem > 120.0:
                prior = _random_prior(n, rng)
                mix = 0.12 + 0.25 * float(rng.random())
                probe = _normalize((1.0 - mix) * best_a + mix * prior)
                lim2 = min(70.0, 0.22 * rem)
                try:
                    a3, sc3 = _mws_search(probe, L, m, n, rng, lim2, prior=prior)
                except Exception:
                    a3, sc3 = probe, _score_only(probe, L, m, n)
                if np.isfinite(sc3) and sc3 < best_score - 1e-12:
                    best_a = a3
                    best_score = float(sc3)
                    cur_a = best_a.copy()
                    stale = 1
                else:
                    stale += 1
            elif stale >= 1 and rng.random() < 0.25:
                probe = _normalize(0.92 * best_a + 0.08 * _random_prior(n, rng))
                try:
                    a4, sc4 = _soft_block(probe, L, m, n, rng, min(20.0, 0.12 * rem))
                except Exception:
                    a4, sc4 = probe, _score_only(probe, L, m, n)
                if np.isfinite(sc4) and sc4 < best_score - 1e-12:
                    best_a = a4
                    best_score = float(sc4)
                    cur_a = best_a.copy()
                    stale = 0

        cycle += 1

    rem = float(max(0.0, deadline - time.monotonic()))
    if rem > 12.0:
        try:
            a5, sc5 = _soft_block(best_a, L, m, n, rng, max(5.0, rem - 3.0))
            if np.isfinite(sc5) and sc5 < best_score - 1e-14:
                best_a = a5
                best_score = float(sc5)
        except Exception:
            pass

    return best_a, best_score


def _optimize_n(n, inits, time_limit, rng):
    time_limit = float(time_limit)
    n = int(n)
    if time_limit <= 0.0 or not inits:
        return None

    L, m = _fft_params(n)
    scored = []
    for arr in inits:
        try:
            a = _normalize(arr)
            sc = _score_only(a, L, m, n)
            if np.isfinite(sc):
                scored.append((float(sc), a))
        except Exception:
            pass

    if not scored:
        a = _uniform_vec(n)
        sc = _score_only(a, L, m, n)
        return a, float(sc), a.copy(), float(sc)

    scores = np.array([item[0] for item in scored], dtype=np.float64)
    order = np.argsort(scores)
    top = [scored[int(ii)] for ii in order]

    init_a = top[0][1].copy()
    init_sc = float(top[0][0])
    best_a = init_a.copy()
    best_sc = init_sc

    start0 = time.monotonic()
    deadline0 = start0 + time_limit

    use_two = len(top) >= 2 and float(top[1][0]) <= float(init_sc) + 0.045

    if use_two:
        t1 = max(10.0, min(0.70 * time_limit, time_limit - 30.0))
        rem1 = float(max(0.0, deadline0 - time.monotonic()))
        t1 = min(t1, rem1)
        if t1 > 10.0:
            try:
                a1, sc1 = _optimize_candidate(top[0][1], n, rng, t1)
            except Exception:
                a1, sc1 = top[0][1], init_sc
            if np.isfinite(sc1) and sc1 < best_sc - 1e-14:
                best_a = a1
                best_sc = float(sc1)

        rem = float(max(0.0, deadline0 - time.monotonic()))
        if rem > 90.0 and float(top[1][0]) <= float(best_sc) + 0.08:
            t2 = max(10.0, min(0.30 * time_limit, rem - 20.0))
            try:
                a2, sc2 = _optimize_candidate(top[1][1], n, rng, t2)
            except Exception:
                a2, sc2 = top[1][1], float(top[1][0])
            if np.isfinite(sc2) and sc2 < best_sc - 1e-14:
                best_a = a2
                best_sc = float(sc2)
    else:
        t1 = max(5.0, float(max(0.0, deadline0 - time.monotonic())))
        try:
            a1, sc1 = _optimize_candidate(top[0][1], n, rng, t1)
        except Exception:
            a1, sc1 = top[0][1], init_sc
        if np.isfinite(sc1) and sc1 < best_sc - 1e-14:
            best_a = a1
            best_sc = float(sc1)

    return best_a, float(best_sc), init_a.copy(), float(init_sc)


def _direct_score(arr):
    a = np.asarray(arr, dtype=np.float64).ravel()
    if a.size == 0:
        return _INF
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.maximum(a, 0.0)
    n = int(a.size)
    s = float(a.sum())
    if n == 0 or s < 1e-12:
        return _INF
    try:
        b = np.convolve(a, a)
        mb = float(np.max(b)) if b.size else 0.0
    except Exception:
        return _INF
    if not np.isfinite(mb):
        return _INF
    return float(2.0 * float(n) * mb / (s * s))

def propose_candidate(seed=0, budget_s=2400):

    else:
        search_stop = start + max(30.0, budget - 100.0)
    if search_stop > hard - 30.0:
        search_stop = hard - 30.0

    rng = np.random.default_rng(seed)
    base = _get_base()
    results = []

        if n1 > 12000:
            n1 = 6121

    inits1 = _make_initial_candidates(n1, base, rng)
    total_search = max(1.0, float(search_stop - start))

    t1_alloc = 0.62 * total_search
    t1_alloc = min(t1_alloc, max(10.0, float(search_stop - time.monotonic())))
    try:
        res1 = _optimize_n(n1, inits1, t1_alloc, rng)
        if res1 is not None:
            best_a1, best_sc1, init_a1, init_sc1 = res1
            results.append((best_a1, n1, float(best_sc1)))
            results.append((init_a1, n1, float(init_sc1)))
            print(f"n={n1} initial={init_sc1:.9f} best={best_sc1:.9f} t={time.monotonic()-start:.1f}s", flush=True)
    except Exception:
        pass

    rem = float(max(0.0, search_stop - time.monotonic()))
    n2 = 8192
    inits2 = _make_initial_candidates(n2, base, rng)
    t2_alloc = min(0.34 * total_search, max(0.0, rem - 30.0))
    if t2_alloc > 120.0:
        try:
            res2 = _optimize_n(n2, inits2, t2_alloc, rng)
            if res2 is not None:
                best_a2, best_sc2, init_a2, init_sc2 = res2
                results.append((best_a2, n2, float(best_sc2)))
                results.append((init_a2, n2, float(init_sc2)))
                print(f"n={n2} initial={init_sc2:.9f} best={best_sc2:.9f} t={time.monotonic()-start:.1f}s", flush=True)
        except Exception:
            pass

    rem = float(max(0.0, search_stop - time.monotonic()))
    if rem > 90.0 and results:
        try:
            scores = np.array([r[2] for r in results], dtype=np.float64)
            idx = int(np.argmin(scores))
            cur_a, cur_n, cur_sc = results[idx]
            extra = min(float(rem) - 20.0, 0.25 * total_search)
            if extra > 30.0:
                L, m = _fft_params(cur_n)
                a_p, sc_p = _soft_block(cur_a, L, m, cur_n, rng, extra)
                if np.isfinite(sc_p) and sc_p < float(cur_sc) - 1e-14:
                    results.append((a_p, int(cur_n), float(sc_p)))
                    print(f"polish n={cur_n} score={sc_p:.9f} t={time.monotonic()-start:.1f}s", flush=True)
        except Exception:
            pass

    if not results:
        a = _uniform_vec(6121)
        return [float(x) for x in a]

    scores = np.array([r[2] for r in results], dtype=np.float64)
    finite = np.where(np.isfinite(scores))[0]
    if finite.size == 0:
        idx = 0
    else:
        order = np.argsort(scores[finite])
        idx = int(finite[int(order[0])]) if order.size else 0

    # Exact final checks for a few best proxy candidates.
    finite_idx = np.where(np.isfinite(scores))[0]
    if finite_idx.size:
        finite_idx = finite_idx[np.argsort(scores[finite_idx])]
    else:
        finite_idx = np.array([0], dtype=np.int64)

    max_exact = 4
    if budget > 900.0:
        max_exact = 6
    if budget > 1800.0:
        max_exact = 8
    max_exact = min(max_exact, int(finite_idx.size))

    best_seq = None
    best_val = _INF

    for j in range(max_exact):
        if time.monotonic() > hard - 3.0:
            break
        ii = int(finite_idx[j])
        arr = results[ii][0]
        n = int(results[ii][1])
        if arr is None or n <= 0 or n > 10000:
            continue
        try:
            seq = [float(x) for x in arr]
            val = _final_eval(seq)
        except Exception:
            val = _INF
        if np.isfinite(val) and val < best_val:
            best_val = float(val)
            best_seq = seq

    if best_seq is None:
        # Pick best proxy and sanitize.
        ii = idx
        arr = results[ii][0]
        arr = _normalize(arr)
        best_seq = [float(max(0.0, x)) for x in arr]

    return best_seq
```
