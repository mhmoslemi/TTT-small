import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _fft_conv(a, b):
    """Full linear convolution via FFT."""
    la, lb = len(a), len(b)
    if la == 0 or lb == 0:
        return np.array([], dtype=float)
    nout = la + lb - 1
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    return np.fft.irfft(fa * fb, L)[:nout]


def _cross_corr(u, p):
    """Compute cross_corr[i] = sum_j u[i+j]*p[j] for i=0..len(p)-1.
    Uses FFT: cross_corr = conv(u, p_rev)[n-1 : n-1+n] where p_rev = p[::-1].
    """
    n = len(p)
    nu = len(u)
    if n == 0:
        return np.array([], dtype=float)
    if nu * n <= 2000000:
        out = np.zeros(n, dtype=float)
        for i in range(n):
            hi = min(nu, i + n)
            if i < nu:
                out[i] = np.dot(u[i:hi], p[:hi - i])
        return out
    L = _next_pow2(nu + n - 1)
    fu = np.fft.rfft(u, L)
    p_rev = p[::-1]
    fp_rev = np.fft.rfft(p_rev, L)
    result = np.fft.irfft(fu * fp_rev, L)
    return result[n - 1:n - 1 + n]

    c = _fft_conv(p, p)

    smooth_val = float(np.dot(w, d))
    return smooth_val, w


def _grad_soft_max(p, w):
    """Gradient of soft_max objective: 2 * cross_corr(w, p)."""
    return 2.0 * _cross_corr(w, p)


def _grad_sum_sq(p):
    """Gradient of sum(conv(p,p)^2) = 4 * cross_corr(conv(p,p), p)."""
    c = _fft_conv(p, p)
    return 4.0 * _cross_corr(c, p)


def _sanitize(p):
    """Clip to non-negative, normalize to sum 1."""
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.0)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s <= 1e-30:
        return None
    return p / s


def _mm_descent(p0, time_limit, beta_start=20.0, beta_end=8000.0,
                use_sumsq_warmup=True, verbose=False):
    """
    Multiplicative mirror descent on the probability simplex.
    Phase 1 (optional): minimize sum(c^2) as warmup.
    Phase 2: minimize soft max with increasing beta.
    Returns (best_p, best_score).
    """
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    floor = 1e-30 / n
    p = np.maximum(p, floor)
    p = p / p.sum()

    best_p = p.copy()
    best_score = _score(p)

    deadline = time.monotonic() + time_limit
    stall = 0
    alpha = 0.5

    # Phase 1: Sum-of-squares warmup (if time allows)
    warmup_time = min(30.0, 0.1 * time_limit) if use_sumsq_warmup else 0.0
    warmup_deadline = time.monotonic() + warmup_time

    if warmup_time > 2.0:
        while time.monotonic() < warmup_deadline and time.monotonic() < deadline:
            grad = _grad_sum_sq(p)
            max_g = float(np.max(np.abs(grad)))
            if max_g < 1e-18:
                break
            g = grad / max_g
            g = np.maximum(g, 0.0)
            mx = float(np.max(g))
            if mx > 0:
                g = g / mx

            accepted = False
            for mult in [0.2, 0.5, 1.0, 2.0]:
                a = alpha * mult
                if a < 1e-4 or a > 100:
                    continue
                decay = np.exp(-np.clip(a * g, -50, 50))
                p_new = p * decay
                p_new = np.maximum(p_new, floor)
                s_new = float(p_new.sum())
                if s_new < 1e-30:
                    continue
                p_new = p_new / s_new
                ss_new = _sum_sq_conv(p_new)
                ss_old = _sum_sq_conv(p)
                if ss_new < ss_old - 1e-14:
                    p = p_new
                    sc = _score(p)
                    if sc < best_score:
                        best_score = sc
                        best_p = p.copy()
                    accepted = True
                    break

            if not accepted:
                stall += 1
                alpha = max(0.01, alpha * 0.7)
                if stall > 40:
                    p = best_p.copy()
                    alpha = 0.5
                    stall = 0

    # Phase 2: Soft max descent with increasing beta
    beta = float(beta_start)
    stall = 0
    alpha = 0.5
    last_beta_incr = time.monotonic()

    while time.monotonic() < deadline:
        smooth_val, w = _soft_max_fields(p, beta)
        grad = _grad_soft_max(p, w)

        max_g = float(np.max(np.abs(grad)))
        if max_g < 1e-18:
            break

        g = grad / max_g
        g = np.maximum(g, 0.0)
        mx = float(np.max(g))
        if mx > 0:
            g = g / mx
        else:
            g = np.zeros(n)

        accepted = False
        for mult in [0.25, 0.5, 1.0, 2.0, 4.0]:
            a = alpha * mult
            if a < 1e-4 or a > 200:
                continue
            decay = np.exp(-np.clip(a * g, -50, 50))
            p_new = p * decay
            p_new = np.maximum(p_new, floor)
            s_new = float(p_new.sum())
            if s_new < 1e-30:
                continue
            p_new = p_new / s_new
            sv_new, _ = _soft_max_fields(p_new, beta)
            if sv_new < smooth_val - 1e-12:
                p = p_new
                sc = _score(p)
                if sc < best_score:
                    best_score = sc
                    best_p = p.copy()
                    if verbose:
                        print(f"  n={n} score={best_score:.6f} beta={beta:.0f}")
                accepted = True
                break

        if accepted:
            stall = 0
            # Increase beta periodically
            if time.monotonic() - last_beta_incr > 5.0:
                beta = min(beta_end, beta * 1.3)
                last_beta_incr = time.monotonic()
        else:
            stall += 1
            alpha = max(0.01, alpha * 0.7)
            if stall > 30:
                # Perturbation
                p_try = p * np.exp(np.random.normal(0, 0.1, n))
                p_try = _sanitize(p_try)
                if p_try is not None:
                    p_try = np.maximum(p_try, floor)
                    p_try = p_try / p_try.sum()
                    sc_try = _score(p_try)
                    if sc_try < best_score * 1.02:
                        p = p_try
                    else:
                        p = best_p.copy()
                alpha = 0.5
                beta = max(beta_start, beta * 0.7)
                stall = 0

    return best_p, best_score


def _polish_direct(p0, time_limit):
    """
    Direct polishing: find the peak of conv(p,p), identify which positions
    contribute most to it, and reduce those positions.
    """
    p = _sanitize(p0)
    if p is None:
        return None, float('inf')
    n = len(p)
    floor = 1e-30 / n
    p = np.maximum(p, floor)
    p = p / p.sum()

    best_p = p.copy()
    best_score = _score(p)
    deadline = time.monotonic() + time_limit

    while time.monotonic() < deadline:
        c = _fft_conv(p, p)
        peak_idx = int(np.argmax(c))
        peak_val = c[peak_idx]

        # Find which positions contribute to c[peak_idx]
        # c[peak_idx] = sum_j p[j]*p[peak_idx - j] for valid j
        contrib = np.zeros(n)
        for j in range(n):
            k = peak_idx - j
            if 0 <= k < n and k != j:
                contrib[j] += 2.0 * p[j] * p[k]
            elif k == j:
                contrib[j] += p[j] * p[k]

        # Reduce the top contributors, redistribute to low-contribution positions
        top_k = max(3, n // 200)
        top_positions = np.argsort(contrib)[-top_k:]
        bot_positions = np.argsort(contrib)[:max(3, n // 200)]

        # Transfer a small amount of mass
        transfer = 0.01 * np.mean([p[i] for i in top_positions])
        for i in top_positions:
            p[i] = max(floor, p[i] - transfer / len(top_positions))
        for i in bot_positions:
            p[i] += transfer / len(bot_positions)

        p = _sanitize(p)
        if p is None:
            break
        p = np.maximum(p, floor)
        p = p / p.sum()

        sc = _score(p)
        if sc < best_score:
            best_score = sc
            best_p = p.copy()

    return best_p, best_score


def _initial_uniform(n):
    return np.ones(n, dtype=float) / n


def _initial_two_block(n, ratio=0.7, height=2.0):
    p = np.ones(n, dtype=float)
    split = int(ratio * n)
    p[split:] = height
    return p / p.sum()


def _initial_cosine(n):
    t = np.arange(n, dtype=float) / (n - 1)
    p = np.cos(np.pi * t / 2.0) ** 2
    return p / p.sum()


def _initial_gaussian_mixture(n, rng, K=3):
    p = np.zeros(n)
    t = np.arange(n, dtype=float) / (n - 1)
    for _ in range(K):
        center = rng.uniform(0.1, 0.9)
        width = rng.uniform(0.05, 0.3)
        weight = rng.uniform(0.5, 2.0)
        p += weight * np.exp(-((t - center) / width) ** 2)
    return p / p.sum()


def _initial_block_random(n, rng, K=8):
    p = np.zeros(n)
    edges = np.sort(rng.uniform(0, 1, K - 1))
    blocks = np.concatenate([[0], edges, [1.0]])
    heights = rng.uniform(0.3, 3.0, K)
    for i in range(K):
        s = int(blocks[i] * n)
        e = int(blocks[i + 1] * n)
        e = max(e, s + 1)
        p[s:e] = heights[i]
    return p / p.sum()


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Search for best non-negative sequence minimizing evaluate_sequence score.
    Uses multi-phase optimization: sum-of-squares warmup, soft-max MM descent,
    and direct peak polishing.
    """
    budget = float(budget_s)
    if not np.isfinite(budget) or budget < 60:
        budget = 60.0

    start_time = time.monotonic()
    deadline = start_time + max(30.0, budget - 20.0)

    try:
        iseed = int(seed)
    except Exception:
        iseed = 12345
    rng = np.random.default_rng((iseed % 10**9) + 7)

    # Try to load parent
    parent_arr = None
    try:
        parent_raw = height_sequence_1
        parent_arr = _sanitize(np.asarray(parent_raw, dtype=float))
    except (NameError, Exception):
        pass

    best_overall = None
    best_score_overall = float('inf')

    def _update_best(p, sc):
        nonlocal best_overall, best_score_overall
        if p is not None and np.isfinite(sc) and sc < best_score_overall:
            best_overall = p.copy()
            best_score_overall = sc
            print(f"[{time.monotonic()-start_time:.1f}s] New best: {sc:.6f} (n={len(p)})")

    # Evaluate parent
    if parent_arr is not None:
        sc = _score(parent_arr)
        _update_best(parent_arr, sc)
        print(f"Parent score: {sc:.6f} (n={len(parent_arr)})")

    # Try multiple lengths
    lengths = [2048, 4096]
    # If budget is large, add longer
    if budget > 1200:
        lengths.append(8192)

    time_per_length = (deadline - time.monotonic()) / len(lengths)

    for n in lengths:
        if time.monotonic() >= deadline - 60:
            break

        t_alloc = min(time_per_length, deadline - time.monotonic() - 30)
        if t_alloc < 30:
            break

        print(f"\n=== Length n={n}, time_alloc={t_alloc:.0f}s ===")

        # Generate starting points
        starts = []

        # From parent (resampled)
        if parent_arr is not None and len(parent_arr) != n:
            x_old = (np.arange(len(parent_arr)) + 0.5) / len(parent_arr)
            x_new = (np.arange(n) + 0.5) / n
            p_resamp = np.interp(x_new, x_old, parent_arr)
            p_resamp = _sanitize(p_resamp)
            if p_resamp is not None:
                starts.append(p_resamp)
        elif parent_arr is not None and len(parent_arr) == n:
            starts.append(parent_arr.copy())

        # Standard initializations
        starts.append(_initial_uniform(n))
        starts.append(_initial_two_block(n, 0.7, 2.0))
        starts.append(_initial_two_block(n, 0.8, 3.0))
        starts.append(_initial_cosine(n))
        starts.append(_initial_gaussian_mixture(n, rng, 3))
        starts.append(_initial_block_random(n, rng, 8))

        # Evaluate all starts
        scored_starts = []
        for s in starts:
            p = _sanitize(s)
            if p is None:
                continue
            sc = _score(p)
            scored_starts.append((sc, p))

        scored_starts.sort(key=lambda x: x[0])
        print(f"Best start score: {scored_starts[0][0]:.4f}")

        # Run optimization on top starts
        n_starts = min(4, len(scored_starts))
        for idx in range(n_starts):
            if time.monotonic() >= deadline - 60:
                break

            t_rem = deadline - time.monotonic() - 60
            if t_rem < 15:
                break

            frac = [0.45, 0.30, 0.15, 0.10][idx] if idx < 4 else 0.05
            t_this = max(10.0, min(t_alloc * frac, t_rem * 0.8))

            p0 = scored_starts[idx][1]
            print(f"  Start {idx}: score={scored_starts[idx][0]:.4f}, t={t_this:.0f}s")

            p_best, sc_best = _mm_descent(
                p0, t_this,
                beta_start=20.0,
                beta_end=6000.0,
                use_sumsq_warmup=(t_this > 20),
                verbose=True
            )
            _update_best(p_best, sc_best)

            # Polish if time remains
            t_rem = deadline - time.monotonic() - 60
            if t_rem > 20 and p_best is not None:
                t_polish = min(30.0, t_rem * 0.3)
                pp, scp = _polish_direct(p_best, t_polish)
                _update_best(pp, scp)

        # If best for this length is good, do more optimization on it
        t_rem = deadline - time.monotonic() - 60
        if t_rem > 60 and best_overall is not None and len(best_overall) == n:
            t_extra = min(t_rem * 0.5, 300.0)
            print(f"  Extra optimization: {t_extra:.0f}s")
            p_extra, sc_extra = _mm_descent(
                best_overall, t_extra,
                beta_start=100.0,
                beta_end=10000.0,
                use_sumsq_warmup=False,
                verbose=True
            )
            _update_best(p_extra, sc_extra)

            t_rem = deadline - time.monotonic() - 30
            if t_rem > 15:
                pp, scp = _polish_direct(best_overall, min(t_rem * 0.5, 60.0))
                _update_best(pp, scp)

    # Final: if we have a good candidate, try upscaling
    t_rem = deadline - time.monotonic() - 30
    if t_rem > 60 and best_overall is not None:
        n_big = max(2 * len(best_overall), 4096)
        n_big = min(n_big, 12288)
        if n_big != len(best_overall):
            x_old = (np.arange(len(best_overall)) + 0.5) / len(best_overall)
            x_new = (np.arange(n_big) + 0.5) / n_big
            p_big = np.interp(x_new, x_old, best_overall)
            p_big = _sanitize(p_big)
            if p_big is not None:
                t_up = min(t_rem * 0.4, 120.0)
                print(f"\nUpscale to n={n_big}, t={t_up:.0f}s")
                p_up, sc_up = _mm_descent(
                    p_big, t_up,
                    beta_start=50.0, beta_end=8000.0,
                    use_sumsq_warmup=False, verbose=True
                )
                _update_best(p_up, sc_up)

    # Final validation with evaluate_sequence
    if best_overall is None:
        best_overall = _initial_uniform(2048)
        best_score_overall = _score(best_overall)

    # Check with official evaluator
    try:
        test_list = [float(x) for x in best_overall]
        official_score = float(evaluate_sequence(test_list))
        print(f"\nOfficial score: {official_score:.6f}")
        if not np.isfinite(official_score):
            print("WARNING: Official evaluator returned inf! Falling back.")
            best_overall = _initial_uniform(2048)
    except Exception as e:
        print(f"Could not call evaluate_sequence: {e}")

    # Prepare output
    out = np.asarray(best_overall, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s <= 0:
        out = np.ones(2048) / 2048
    else:
        out = out / s

    print(f"\nFinal: n={len(out)}, internal_score={best_score_overall:.6f}")
    print(f"Time used: {time.monotonic()-start_time:.1f}s / {budget_s}s")

    return [float(x) for x in out]
