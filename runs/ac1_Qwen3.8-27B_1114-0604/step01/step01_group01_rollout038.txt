"""
Search for non-negative sequence minimizing 2*n*max(conv(a,a))/sum(a)^2.
Uses multi-phase approach: diverse initialization, coordinate descent with
exact per-coordinate optimization, smoothed-max descent with adaptive beta,
block-level search, and multi-scale refinement.

import numpy as np
import time


def _next_pow2(x):
    x = max(1, int(x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _fft_autocorr(a):
    """Full convolution a*a via FFT."""
    n = len(a)
    if n == 0:
        return np.array([], dtype=float)
    L = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    return np.fft.irfft(fa * fa, L)[:2 * n - 1]


def _exact_score(a):
    """Compute 2n*max(conv(a,a))/sum(a)^2."""
    n = len(a)
    S = float(a.sum())
    if S <= 1e-12:
        return float('inf')
    conv = _fft_autocorr(a)
    M = float(np.max(conv))
    return 2.0 * n * M / (S * S)


def _sanitize(a):
    """Clip to non-negative finite, normalize to sum 1."""
    a = np.asarray(a, dtype=float)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-15:
        return np.ones(len(a), dtype=float) / len(a)
    return a / s


def _resample(a, m):
    """Resample normalized vector to length m."""
    n = len(a)
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m, dtype=float) / m
    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(m) + 0.5) / float(m)
    out = np.interp(x_new, x_old, a)
    return _sanitize(out)


def _init_cosine(n, rng):
    """Initialize with soft cosine series, clipped to non-negative."""
    t = (np.arange(n, dtype=float) + 0.5) / float(n)
    z = np.ones(n, dtype=float)
    for k in range(1, 20):
        amp = float(rng.normal(0, 1.0 / (k ** 0.7)))
        ph = float(rng.uniform(0, 2 * np.pi))
        z += amp * np.cos(2 * np.pi * k * t + ph)
    p = np.log1p(np.exp(np.clip(z, -30, 30)))
    return _sanitize(p)


def _init_bumps(n, rng):
    """Initialize with sum of Gaussian bumps."""
    t = np.arange(n, dtype=float) / float(n)
    K = int(rng.integers(2, 8))
    p = np.zeros(n, dtype=float)
    for _ in range(K):
        c = float(rng.uniform(0, 1))
        s = float(rng.uniform(0.02, 0.4))
        w = float(rng.uniform(0.5, 3.0))
        p += w * np.exp(-((t - c) / s) ** 2)
    p += 1e-10
    return _sanitize(p)


def _init_blocks(n, rng):
    """Initialize with random block structure."""
    K = int(rng.integers(4, 20))
    heights = np.exp(rng.normal(0, 0.8, K))
    p = np.zeros(n, dtype=float)
    block_size = n // K
    for i in range(K):
        s = i * block_size
        e = s + block_size if i < K - 1 else n
        p[s:e] = heights[i]
    p += 1e-10
    return _sanitize(p)


def _init_two_block(n, rng):
    """Two-block asymmetric initialization."""
    a = float(rng.uniform(0.3, 0.7))
    idx = int(round(a * n))
    ratio = float(rng.choice([0.3, 0.5, 1.0, 2.0, 3.5]))
    p = np.ones(n, dtype=float)
    p[idx:] = ratio

def _init_from_spectrum(n, rng):
    """Initialize from random flat-ish spectrum."""
    L = _next_pow2(2 * n - 1)
    half = L // 2 + 1
    # Create a spectrum with some flatness
    mag = np.ones(half, dtype=float)
    # Add random perturbations
    for i in range(1, half):
        mag[i] = mag[i] * (1.0 + float(rng.normal(0, 0.3)))
        mag[i] = max(0.1, mag[i])
    ph = rng.uniform(0, 2 * np.pi, half)
    spectrum = mag * np.exp(1j * ph)
    # Enforce Hermitian symmetry
    full = np.zeros(L, dtype=complex)
    full[:half] = spectrum
    full[half:] = np.conj(spectrum[1:half][::-1]) if half > 1 else np.array([0.0])
    a = np.fft.irfft(full, L)[:n]
    a = np.abs(a) + 1e-10
    return _sanitize(a)


def _coord_descent(a, rng, time_limit, multipliers=None):
    """
    Coordinate descent: for each coordinate, try several multipliers,
    keep the one that gives the best score.
    Uses efficient O(n) per-multiplier evaluation.
    """
    n = len(a)
    a = _sanitize(a).copy()
    S = float(a.sum())
    conv = _fft_autocorr(a)
    cur_score = 2.0 * n * float(np.max(conv)) / (S * S)
    best = a.copy()
    best_score = cur_score

    if multipliers is None:
        multipliers = np.array([0.05, 0.2, 0.5, 0.75, 1.25, 1.5, 2.0, 4.0, 8.0])

    deadline = time.monotonic() + time_limit
    stall = 0
    passes = 0

    while time.monotonic() < deadline:
        passes += 1
        improved = False
        # Randomize coordinate order for diversity
        order = rng.permutation(n)

        for i in order:
            if time.monotonic() > deadline:
                break
            ai = a[i]
            if ai < 1e-15:
                # Try small positive values
                deltas = [1e-3 / n, 1e-2 / n, 1e-1 / n]
            else:
                deltas = ai * (multipliers - 1.0)

            best_delta = 0.0
            best_local_score = cur_score
            best_new_conv_max = float(np.max(conv))

            lo = i
            hi = min(2 * n - 1, i + n)
            # Precompute outside max (max of conv outside [lo, hi-1])
            if lo > 0:
                outside_left = float(np.max(conv[:lo]))
            else:
                outside_left = -1e30
            if hi < 2 * n - 1:
                outside_right = float(np.max(conv[hi:]))
            else:
                outside_right = -1e30
            outside_max = max(outside_left, outside_right)

            slice_conv = conv[lo:hi]  # length n
            # For each delta, new_conv[lo:hi] = slice_conv + 2*delta*a
            # and new_conv[2*i] += delta^2 (position 2*i - lo = i in the slice)
            base_change = 2.0 * a  # shape (n,)
            pos_2i = 2 * i - lo  # = i, since lo = i

            for d in deltas:
                if abs(d) < 1e-15:
                    continue
                new_S = S + d
                if new_S < 1e-12:
                    continue
                # Compute new max in the slice
                new_slice = slice_conv + d * base_change
                if 0 <= pos_2i < n:
                    new_slice[pos_2i] += d * d
                new_max_slice = float(np.max(new_slice))
                new_max = max(new_max_slice, outside_max)
                new_score = 2.0 * n * new_max / (new_S * new_S)
                if new_score < best_local_score - 1e-12:
                    best_local_score = new_score
                    best_delta = d

            if best_delta != 0.0:
                a[i] = a[i] + best_delta
                # Update conv
                lo = i
                hi = min(2 * n - 1, i + n)
                conv[lo:hi] += 2.0 * best_delta * a
                pos_2i = 2 * i
                if 0 <= pos_2i < 2 * n - 1:
                    conv[pos_2i] += best_delta * best_delta
                S += best_delta
                cur_score = best_local_score
                improved = True
                if cur_score < best_score:
                    best_score = cur_score
                    best = a.copy()

        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                # Perturb to escape local minimum
                a = _sanitize(a * np.exp(rng.normal(0, 0.03, n)))
                S = float(a.sum())
                conv = _fft_autocorr(a)
                cur_score = 2.0 * n * float(np.max(conv)) / (S * S)
                stall = 0
                if time.monotonic() > deadline:
                    break

    return best, best_score

                w = np.full_like(d, 1.0 / len(d))

                    else:

def _block_search(a0, rng, time_limit):

    n = len(a0)
    a = _sanitize(a0).copy()

    deadline = time.monotonic() + time_limit
    stagnant = 0
    factors = [0.3, 0.5, 0.7, 1.4, 1.7, 2.5, 4.0]

    while time.monotonic() < deadline:

        r = float(rng.random())

            w = int(rng.integers(max(2, n // 100), max(3, n // 20 + 1)))

        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))

            a_try = a.copy()

            a_try = _sanitize(a_try)
            sc = _exact_score(a_try)
            if sc < best_sc - 1e-12:
                best_sc = sc
                best_f = f

            a = _sanitize(a)
            cur_score = best_sc

                stagnant = 0
            if stagnant > 300:

                cur_score = best_score
                stagnant = 0

        starts.append(parent.copy())

    for fn in init_funcs:
        if time.monotonic() > deadline - 60:
            break
        try:
            s = fn()
            if s is not None:
                starts.append(s)
        except Exception:
            pass

    # Quick MMD on a few starts
    per_init_budget = max(5.0, init_budget / max(1, len(starts)) * 0.6)
    for s in starts[:6]:
        if time.monotonic() > deadline - 30:
            break
        t = min(per_init_budget, deadline - time.monotonic() - 15)
        if t < 3:
            break
        p, sc = _smoothed_max_descent(s, rng, t, beta_start=30, beta_end=300)
        add_cand(p, sc)
        print(f"  Init cand: score={sc:.6f}")

    print(f"Phase 1 done. Best: {cands[0][0]:.6f}")

    # === Phase 2: Coordinate Descent (600s) ===
    print(f"Phase 2: Coordinate descent (budget: {min(600, deadline - time.monotonic()):.0f}s)")
    cd_budget = min(600.0, deadline - time.monotonic()) * 0.7
    best_a, best_sc = cands[0]
    
    for trial in range(3):
        if time.monotonic() > deadline - 120:
            break
        t = cd_budget / 3.0
        if trial == 0:
            start_a = best_a.copy()
        else:
            start_a = _sanitize(best_a * np.exp(rng.normal(0, 0.05, n)))
        p, sc = _coord_descent(start_a, rng, min(t, deadline - time.monotonic() - 60))
        add_cand(p, sc)
        print(f"  CD trial {trial}: score={sc:.6f}")

    best_a, best_sc = cands[0]
    print(f"Phase 2 done. Best: {best_sc:.6f}")

    # === Phase 3: Smoothed-max refinement (500s) ===
    print(f"Phase 3: Smoothed-max refinement (budget: {min(500, deadline - time.monotonic()):.0f}s)")
    smd_budget = min(500.0, deadline - time.monotonic()) * 0.7
    
    # Run on best candidate
    t = smd_budget * 0.5
    if time.monotonic() < deadline - 60:
        p, sc = _smoothed_max_descent(best_a, rng, min(t, deadline - time.monotonic() - 30),
                                       beta_start=50, beta_end=3000)
        add_cand(p, sc)
        print(f"  SMD best: score={sc:.6f}")

    # Run on second best with different parameters
    if len(cands) > 1 and time.monotonic() < deadline - 60:
        t = smd_budget * 0.3
        p, sc = _smoothed_max_descent(cands[1][1], rng, min(t, deadline - time.monotonic() - 30),
                                       beta_start=100, beta_end=5000)
        add_cand(p, sc)
        print(f"  SMD second: score={sc:.6f}")

    # Run on a perturbed version
    if time.monotonic() < deadline - 60:
        perturbed = _sanitize(best_a * np.exp(rng.normal(0, 0.1, n)))
        t = smd_budget * 0.2
        p, sc = _smoothed_max_descent(perturbed, rng, min(t, deadline - time.monotonic() - 30),
                                       beta_start=30, beta_end=1000)
        add_cand(p, sc)
        print(f"  SMD perturbed: score={sc:.6f}")

    best_a, best_sc = cands[0]
    print(f"Phase 3 done. Best: {best_sc:.6f}")

    # === Phase 4: Block Search (200s) ===
    print(f"Phase 4: Block search (budget: {min(200, deadline - time.monotonic()):.0f}s)")
    bs_budget = min(200.0, deadline - time.monotonic()) * 0.7
    if bs_budget > 20:
        p, sc = _block_search(best_a, rng, min(bs_budget, deadline - time.monotonic() - 30))
        add_cand(p, sc)
        print(f"  Block search: score={sc:.6f}")

    best_a, best_sc = cands[0]
    print(f"Phase 4 done. Best: {best_sc:.6f}")

    # === Phase 5: Final polish (remaining time) ===
    rem = deadline - time.monotonic()
    print(f"Phase 5: Final polish (budget: {rem:.0f}s)")
    
    if rem > 100:
        # Another round of coord descent on best
        t = min(rem * 0.4, 200.0)
        p, sc = _coord_descent(best_a, rng, min(t, rem - 30))
        add_cand(p, sc)
        print(f"  Final CD: score={sc:.6f}")
        best_a, best_sc = cands[0]

    rem = deadline - time.monotonic()
    if rem > 80:
        # SMD with high beta
        t = min(rem * 0.4, 150.0)
        p, sc = _smoothed_max_descent(best_a, rng, min(t, rem - 30),
                                       beta_start=200, beta_end=10000)
        add_cand(p, sc)
        print(f"  Final SMD: score={sc:.6f}")
        best_a, best_sc = cands[0]

    rem = deadline - time.monotonic()
    if rem > 60:
        # Block search on best
        t = min(rem * 0.5, 100.0)
        p, sc = _block_search(best_a, rng, min(t, rem - 20))
        add_cand(p, sc)
        print(f"  Final blocks: score={sc:.6f}")
        best_a, best_sc = cands[0]

    # === Try upscaling if time permits ===
    rem = deadline - time.monotonic()
    if rem > 150 and best_a is not None:
        print(f"Trying upscale to 4096 (remaining: {rem:.0f}s)")
        n_big = 4096
        a_big = _resample(best_a, n_big)
        t = min(90.0, rem * 0.5)
        p_big, sc_big = _smoothed_max_descent(a_big, rng, t, beta_start=50, beta_end=2000)
        add_cand(p_big, sc_big)
        print(f"  Upscaled SMD: score={sc_big:.6f}")
        best_a, best_sc = cands[0]

        rem = deadline - time.monotonic()
        if rem > 40:
            p_big, sc_big = _block_search(best_a, rng, min(30.0, rem - 10))
            add_cand(p_big, sc_big)
            print(f"  Upscaled blocks: score={sc_big:.6f}")
            best_a, best_sc = cands[0]

    # === Final evaluation ===
    print(f"\nFinal candidates (top 5):")
    for i, (sc, a) in enumerate(cands[:5]):
        print(f"  {i}: score={sc:.6f}, len={len(a)}")

    # Evaluate with official evaluator
    final_list = [float(x) for x in best_a]
    best_official = _exact_score(best_a)
    
    for sc, a in cands[:5]:
        if time.monotonic() > deadline:
            break
        lst = [float(x) for x in a]
        try:
            official_sc = float(evaluate_sequence(lst))
        except Exception:
            official_sc = sc
        if np.isfinite(official_sc) and official_sc < best_official:
            best_official = official_sc
            final_list = lst

    print(f"Best internal score: {best_sc:.6f}")
    print(f"Best official score: {best_official:.6f}")

    # Ensure valid output
    out = np.asarray(final_list, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s < 0.01:
        out = np.ones(3072, dtype=float)
        s = float(out.sum())
    
    return [float(x) for x in out]
