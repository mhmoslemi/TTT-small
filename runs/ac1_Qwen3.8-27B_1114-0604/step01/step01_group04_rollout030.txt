```python
"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding.
The key idea is to iteratively find a direction g (via linear programming) that
maximizes mass while keeping conv(f,g) bounded by the current peak M, then
blend the current solution towards g. This is combined with subgradient polishing.
"""

import numpy as np
import time
from scipy.optimize import linprog

def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)

def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]

def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))

def build_conv_matrix(f, n):
    """Build matrix A where A[k,i] = f[k-i] if 0<=k-i<n, else 0.
    Shape: (2n-1, n). Then (f*g)[k] = A[k] @ g."""
    size = 2 * n - 1
    A = np.zeros((size, n))
    # For each output position k, A[k, i] = f[k-i] if 0 <= k-i < n
    for k in range(size):
        i_min = max(0, k - (n - 1))
        i_max = min(k, n - 1)
        for i in range(i_min, i_max + 1):
            A[k, i] = f[k - i]
    return A

def build_conv_matrix_fast(f):
    """Build matrix A efficiently using vectorized ops."""
    n = len(f)
    size = 2 * n - 1
    A = np.zeros((size, n))
    # A[k, i] = f[k-i] if 0 <= k-i < n
    # For fixed i, A[k, i] = f[k-i] for k = i, i+1, ..., i+n-1
    for i in range(n):
        A[i:i+n, i] = f
    return A

def lp_direction(f, M, n):
    """Find g >= 0 maximizing sum(g) subject to conv(f,g) <= M (element-wise)."""
    size = 2 * n - 1
    A = build_conv_matrix_fast(f)
    c = -np.ones(n)  # minimize -sum(g)
    b_ub = np.full(size, M)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, 
                  method='highs')
    if res.success:
        return res.x
    return np.ones(n) / n

def smooth_grad_single(a, b, tau):
    """Smoothed subgradient using softmax weights at one temperature."""
    n = a.size
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(b.size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= s
    g = 2.0 * np.correlate(w, a, mode='valid')
    return g

def subgrad_step(a, eta, tau_frac, n_scales):
    """One subgradient step."""
    n = a.size
    b = full_conv(a)
    cur_sc = 2.0 * n * float(np.max(b))
    tau = tau_frac * float(np.max(b))
    if tau < 1e-18:
        tau = 1e-18
    g = np.zeros(n)
    for s in range(n_scales):
        t = tau * (2.0 ** s)
        g += smooth_grad_single(a, b, t)
    g /= n_scales
    a_new = project_simplex(a - eta * g)
    b_new = full_conv(a_new)
    new_sc = 2.0 * n * float(np.max(b_new))
    return a_new, new_sc, cur_sc

def lp_iterate(a, time_budget, rng, verbose=True):
    """Main LP-iteration loop."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    if verbose:
        print(f"  [lp_iterate] start: n={n}, score={best_sc:.6f}")
    
    t0 = time.monotonic()
    stuck = 0
    
    while time.monotonic() - t0 < time_budget:
        it_time = time_budget - (time.monotonic() - t0)
        
        b = full_conv(a)
        M = float(np.max(b))
        cur_sc = 2.0 * n * M
        
        # Find LP direction
        g = lp_direction(a, M, n)
        
        # Rescale g to sum 1
        g_sum = g.sum()
        if g_sum < 1e-10:
            g = np.ones(n) / n
        else:
            g = g / g_sum
        
        # Try different step sizes
        best_t = 0.5
        best_new_sc = cur_sc
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_candidate = (1 - t) * a + t * g
            a_candidate = project_simplex(a_candidate)
            sc = normed_score(a_candidate)
            if sc < best_new_sc:
                best_new_sc = sc
                best_t = t
        
        a_new = (1 - best_t) * a + best_t * g
        a_new = project_simplex(a_new)
        a_new_sc = normed_score(a_new)
        
        if a_new_sc < best_sc:
            best_sc = a_new_sc
            best_a = a_new.copy()
            a = a_new.copy()
            stuck = 0
            if verbose:
                print(f"  [lp_iterate] improved: score={best_sc:.6f}, t={best_t}")
        else:
            # Try subgradient step as fallback
            a_sg, sc_sg, _ = subgrad_step(a, 0.3, 0.04, 3)
            if sc_sg < best_sc:
                best_sc = sc_sg
                best_a = a_sg.copy()
                a = a_sg.copy()
                stuck = 0
                if verbose:
                    print(f"  [lp_iterate] subgrad improved: score={best_sc:.6f}")
            else:
                stuck += 1
                # Perturb
                if stuck > 5:
                    a = project_simplex(best_a + 0.05 * rng.standard_normal(n))
                    stuck = 0
        
        # Check time
        if time.monotonic() - t0 > time_budget * 0.85:
            break
    
    return best_a, best_sc

def init_random(n, rng):
    a = np.abs(rng.standard_normal(n)) + 0.05
    return a / a.sum()
```

```python
"""
Minimizes score = 2*n*max(conv(a,a))/sum(a)^2 using LP-based direction finding
combined with subgradient polish. The LP finds the direction that maximizes mass
while keeping the cross-convolution bounded, providing principled descent steps
that are more effective than pure subgradient methods.
"""

import numpy as np
import time
from scipy.optimize import linprog


def project_simplex(v):
    """Project vector v onto the probability simplex {x>=0, sum(x)=1}."""
    n = v.size
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    idx = np.arange(1, n + 1, dtype=np.float64)
    mask = u * idx > (cssv - 1.0)
    if not mask.any():
        return np.ones(n) / n
    rho = np.where(mask)[0][-1]
    theta = (cssv[rho] - 1.0) / (rho + 1.0)
    return np.maximum(v - theta, 0.0)


def full_conv(a):
    """Full linear convolution a*a via FFT. Returns array of length 2n-1."""
    n = a.size
    size = 2 * n - 1
    fsize = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(a, fsize)
    return np.fft.irfft(fa * fa, fsize)[:size]


def normed_score(a):
    """Compute 2*n*max(a*a) assuming sum(a)=1."""
    n = a.size
    b = full_conv(a)
    return 2.0 * n * float(np.max(b))


def build_conv_matrix(f):
    """Build matrix A where (f*g)[k] = A[k] @ g for g of same length as f."""
    n = len(f)
    size = 2 * n - 1
    A = np.zeros((size, n))
    for i in range(n):
        A[i:i + n, i] = f
    return A


def lp_direction(f, M):
    """Find g >= 0 maximizing sum(g) subject to conv(f,g) <= M (element-wise).
    
    Returns (g, dual_y) where g is the primal solution and dual_y are dual vars.
    """
    n = len(f)
    size = 2 * n - 1
    A = build_conv_matrix(f)
    c = -np.ones(n)
    b_ub = np.full(size, M)
    
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
    if res.success:
        dual_y = res.ineqlin.marginals if res.ineqlin.marginals is not None else np.zeros(size)
        return res.x, np.abs(dual_y)
    return np.ones(n) / n, np.zeros(size)


def smooth_grad_single(a, b, tau):
    """Smoothed subgradient using softmax weights at one temperature."""
    n = a.size
    shift = b / tau
    shift -= shift.max()
    w = np.exp(shift)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(b.size)
        w[int(np.argmax(b))] = 1.0
    else:
        w /= s
    g = 2.0 * np.correlate(w, a, mode='valid')
    return g


def lp_iterate(a, time_budget, rng, verbose=True):
    """Main LP-iteration loop with line search and dual-guided subgradient fallback."""
    n = a.size
    a = project_simplex(a.astype(np.float64).copy())
    best_a = a.copy()
    best_sc = normed_score(a)
    
    if verbose:
        print(f"  [lp] start: n={n}, score={best_sc:.6f}")
    
    t0 = time.monotonic()
    stuck = 0
    iter_count = 0
    
    while time.monotonic() - t0 < time_budget:
        b = full_conv(a)
        M = float(np.max(b))
        cur_sc = 2.0 * n * M
        
        # Find LP direction
        g, dual_y = lp_direction(a, M)
        g_sum = g.sum()
        if g_sum < 1e-10:
            g = np.ones(n) / n
        else:
            g = g / g_sum
        
        # Line search over step sizes
        best_t = 0.3
        best_new_sc = cur_sc
        for t in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            a_cand = (1 - t) * a + t * g
            a_cand = project_simplex(a_cand)
            sc = normed_score(a_cand)
            if sc < best_new_sc:
                best_new_sc = sc
                best_t = t
        
        a_new = (1 - best_t) * a + best_t * g
        a_new = project_simplex(a_new)
        a_new_sc = normed_score(a_new)
        
        if a_new_sc < best_sc - 1e-10:
            best_sc = a_new_sc
            best_a = a_new.copy()
            a = a_new.copy()
            stuck = 0
            iter_count += 1
            if verbose and iter_count % 10 == 0:
                elapsed = time.monotonic() - t0
                print(f"  [lp] iter={iter_count}, score={best_sc:.6f}, t={best_t}, elapsed={elapsed:.1f}s")
        else:
            # Fallback: dual-guided subgradient
            # Use dual_y to weight the subgradient
            tau = 0.04 * M
            if tau < 1e-18:
                tau = 1e-18
            g_sg = smooth_grad_single(a, b, tau)
            
            # Also try with dual-weighted positions
            # The dual tells us which constraints are tight; use this to focus
            g_dual = dual_y / (dual_y.sum() + 1e-30)
            g_weighted = np.zeros(n)
            for k in range(2 * n - 1):
                if g_dual[k] > 0.01:
                    i_min = max(0, k - (n - 1))
                    i_max = min(k, n - 1)
                    for i in range(i_min, i_max + 1):
                        g_weighted[i] += g_dual[k] * a[k - i]
            
            # Blend subgradient and dual-guided
            g_combined = 0.5 * g_sg + 0.5 * g_weighted / (max(g_weighted.max(), 1e-30))
            
            for eta in [0.2, 0.1, 0.05]:
                a_trial = project_simplex(a - eta * g_combined)
                sc_trial = normed_score(a_trial)
                if sc_trial < best_sc:
                    best_sc = sc_trial
                    best_a = a_trial.copy()
                    a = a_trial.copy()
                    stuck = 0
                    break
            else:
                stuck += 1
                if stuck > 10:
                    a = project_simplex(best_a + 0.03 * rng.standard_normal(n))
                    stuck = 0
    
    return best_a, best_sc
```
