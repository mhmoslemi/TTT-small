    """
    # a = softmax(x)
    x_shift = x - np.max(x)
    exp_x = np.exp(x_shift)
    a = exp_x / exp_x.sum()
    a = np.clip(a, 1e-15, None)  # avoid zeros
    a = a / a.sum()
    
    # Convolution
    c = _conv_full(a, a)  # length 2n-1
    d = n * c
    
    # Smoothed max
    w = _soft_topk(d, beta, k)
    smooth_val = np.dot(w, d)
    f = 2.0 * smooth_val
    
    # Gradient
    h = _conv_valid(w, a)  # length n
    grad_a = 2.0 * h
    
    # Chain rule through softmax
    # da_i/dx_j = a_i * (delta_ij - a_j)
    # grad_x = diag(a) @ grad_a - a * (a^T @ grad_a)
    ax = np.dot(a, grad_a)
    grad_x = a * grad_a - a * ax
    
    return f, grad_x

def _optimize_lbgfsb(a0, n, beta_start, beta_end, k_start, k_end, time_limit, rng):
    """
    Optimize using L-BFGS-B with adaptive beta and k.

    Returns: (best_a, best_score)
    """
    # Initialize x from a0
    a0 = np.clip(a0, 1e-15, None)
    a0 = a0 / a0.sum()

    beta = beta_start

            break

        x_opt = res.x

        a_opt = a_opt / a_opt.sum()
        a_opt = np.clip(a_opt, 1e-15, None)
        a_opt = a_opt / a_opt.sum()
        
        score = _exact_score(a_opt)
        if np.isfinite(score) and score < best_score:
            best_a = a_opt.copy()

        x0 = x_opt

        if time.monotonic() - t_start < 0.5:

            k = max(k_end, int(k * 0.9))

    return best_a, best_score

def _polish_exact(a, n, time_limit, rng):
    """
    Polish by targeting the exact argmax of the convolution.
    Uses multiplicative updates focused on the peak.
    """
    a = _sanitize(a)
    if a is None:
        return a, float('inf')
    
    deadline = time.monotonic() + time_limit
    best_a = a.copy()
    best_score = _exact_score(a)
    
    # Find the peak
    c = _conv_full(a, a)
    peak_idx = np.argmax(c)
    peak_val = c[peak_idx]
    
    # The gradient at the peak: d_c_peak/d_a_j = 2*a[peak_idx - j] (for valid j)
    # So we want to decrease a_j where a[peak_idx - j] is large
    # In other words, we want to reshape a to reduce the overlap at the peak lag
    
    factors = np.array([0.5, 0.7, 0.85, 1.0, 1.15, 1.3, 1.5])
    stagnant = 0
    
    while time.monotonic() < deadline:
        if stagnant > 100:
            break
        
        # Random block modification
        n_blocks = rng.integers(1, 4)
        improved = False

            w = rng.integers(max(2, n//100), max(3, n//10))

                cand[start:start+w] *= f

                if cand is None:
                    continue
                sc = _exact_score(cand)
                if sc < cur_score - 1e-10:
                    cur_score = sc
                    cur_best = cand

                a = cur_best

                best_a = a.copy()
                improved = True
                # Update peak
                c = _conv_full(a, a)
                peak_idx = np.argmax(c)

def _gen_initial(n, mode, rng):
    """Generate diverse initial sequences."""
    t = (np.arange(n) + 0.5) / n
    if mode == 0:
        # Random Fourier series
        z = np.ones(n)
        for k in range(1, 8):
            amp = rng.normal(0, 1.0/(k+1))
            ph = rng.uniform(0, 2*np.pi)
            z += amp * np.cos(2*np.pi*k*t + ph)
        p = np.log1p(np.exp(np.clip(z, -30, 30)))
    elif mode == 1:
        # Beta-like
        a = rng.uniform(0.5, 3.0)
        b = rng.uniform(0.5, 3.0)
        tt = np.clip(t, 1e-8, 1-1e-8)
        p = tt**a * (1-tt)**b
    elif mode == 2:
        # Gaussian bumps
        K = rng.integers(2, 5)
        p = np.zeros(n)
        for _ in range(K):
            c = rng.uniform(0.1, 0.9)
            s = rng.uniform(0.05, 0.3)
            w = rng.uniform(0.5, 2.0)
            p += w * np.exp(-((t-c)/s)**2)
    elif mode == 3:
        # Random blocks
        K = rng.integers(4, 16)

        c = rng.uniform(0.3, 0.7)
        k = rng.uniform(5, 20)
        p = 1/(1+np.exp(-k*(t-c)))
    
    p = np.where(np.isfinite(p), p, 0)
    p = np.clip(p, 0, None)
    p = p + 1e-10

def _resample(arr, n):
    """Resample normalized vector to length n."""
    a = _sanitize(arr)
    if a is None:
        return None
    m = a.size
    if m == n:
        return a.copy()
    x_old = (np.arange(m) + 0.5) / m
    x_new = (np.arange(n) + 0.5) / n
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0)
    return _sanitize(out + 1e-12)

Actually, one more thought: I should consider whether it's better to optimize the p-norm or the softmax-smoothed-max. 

The p-norm `(sum c_j^p)^{1/p}` is:
- Smooth for p > 1
- Equals the max as p → ∞
- For p=2, it's the L2 norm (emphasizes large values)
- For p=4, it's a good approximation to the max when there's one dominant peak

The softmax-smoothed-max `sum w_j c_j` where `w = softmax(beta * c)`:
- Smooth for any finite beta
- Equals the max as beta → ∞
- With large beta, it effectively becomes a "soft top-1"
- More flexible (can use top-k variants)

I think the p-norm is cleaner and has better theoretical properties. Let me use p=4 for the main optimization and p=∞ (exact max with subgradient) for polishing.

2. Regularize with a small L2 penalty on x - simple and effective
