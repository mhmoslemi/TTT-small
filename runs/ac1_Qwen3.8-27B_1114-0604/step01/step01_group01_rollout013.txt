"""

import numpy as np
import time


def _next_pow2(x):
    x = int(max(1, x))
    L = 1
    while L < x:
        L <<= 1
    return L


def _fft_conv_full(a, b):
    """Full linear convolution via FFT."""
    nout = len(a) + len(b) - 1
    L = _next_pow2(nout)
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    out = np.fft.irfft(fa * fb, L)
    return out[:nout]


def _fft_cross_corr(a, w):
    """
    Compute cross-correlation: corr[j] = sum_m a[m] * w[m+j]
    where a has length n and w has length 2n-1.
    Returns array of length n (valid cross-correlation).
    """
    n = len(a)
    nw = len(w)
    nconv = n + nw - 1
    L = _next_pow2(nconv)
    fa = np.fft.rfft(a, L)
    fw = np.fft.rfft(w, L)
    corr = np.fft.irfft(fa * np.conjugate(fw), L)
    return corr[:n]


def _sanitize(a):
    """Clip to valid non-negative finite array, normalize to sum 1."""
    a = np.asarray(a, dtype=float)
    a = np.where(np.isfinite(a), a, 0.0)
    a = np.clip(a, 0.0, 1000.0)
    s = float(a.sum())
    if s <= 1e-15:
        return None
    return a / s


def _exact_score(a):
    """Compute score = 2n*max(conv(a,a))/sum(a)^2 for non-negative a."""
    a = _sanitize(a)
    if a is None:
        return float('inf')
    n = len(a)
    c = _fft_conv_full(a, a)
    mc = float(np.max(c))
    s = float(a.sum())
    if s < 1e-10:
        return float('inf')
    return float(2.0 * n * mc / (s * s))


def _score_normalized(a):
    """Score assuming sum(a)=1, so score = 2n*max(c)."""
    n = len(a)
    if n == 0:
        return float('inf')
    c = _fft_conv_full(a, a)
    return float(2.0 * n * np.max(c))

def _init_smooth_random(n, rng, mode=0):
    t = (np.arange(n, dtype=float) + 0.5) / n
    if mode == 0:
        # Cosine series
        z = np.ones(n)
        for k in range(1, 20):
            amp = rng.normal(0, 1.0 / (k + 1))
            ph = rng.uniform(0, 2 * np.pi)
            z += amp * np.cos(2 * np.pi * k * t + ph)
        p = np.logaddexp(0, z)
    elif mode == 1:
        # Beta distribution
        a, b = rng.uniform(0.5, 4.0), rng.uniform(0.5, 4.0)
        tt = np.clip(t, 1e-6, 1 - 1e-6)
        p = tt ** a * (1 - tt) ** b
    elif mode == 2:
        # Gaussian bumps
        K = int(rng.integers(1, 5))
        p = np.zeros(n)
        for _ in range(K):
            c = rng.uniform(0, 1)
            s = rng.uniform(0.05, 0.4)
            w = rng.uniform(0.5, 2.0)
            p += w * np.exp(-((t - c) / s) ** 2)
    elif mode == 3:
        # Piecewise linear
        K = int(rng.integers(4, 12))
        knots = np.sort(rng.uniform(0, 1, K))
        heights = rng.lognormal(0, 1, K)
        p = np.interp(t, knots, heights)
    elif mode == 4:
        # Two-block with smooth transition
        split = rng.uniform(0.3, 0.7)
        ratio = rng.uniform(0.3, 3.0)
        width = rng.uniform(0.05, 0.2)
        p = np.where(t < split, 1.0, ratio)
        trans = np.exp(-((t - split) / width) ** 2)
        p = p * (1 - 0.5 * trans)
    else:
        # Random phases (squared for non-negativity)
        z = np.ones(n)
        for k in range(1, 15):
            amp = 1.0 / np.sqrt(k)
            ph = rng.uniform(0, 2 * np.pi)
            z += amp * np.cos(2 * np.pi * k * t + ph)
        p = z ** 2 + 0.01

    p = np.where(np.isfinite(p), p, 0.0)
    p = np.clip(p, 0, 1000)
    p = p + 1e-12
    s = p.sum()
    if s <= 0:
        return _init_uniform(n)
    return p / s


def _resample(a, m):
    """Resample normalized vector to length m."""
    n = len(a)
    if n == m:
        return a.copy()
    if n == 1:
        return np.ones(m, dtype=float) / m
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(m) + 0.5) / m
    out = np.interp(x_new, x_old, a)
    out = np.maximum(out, 0)
    out = out + 1e-12
    s = out.sum()
    return out / s


def _topk_gradient(a, k, c=None):
    """
    Compute gradient of sum of top-k convolution values.
    Returns g[j] = 2 * sum_{k' in topk} a[k'-j] (valid cross-correlation).
    """
    n = len(a)
    if c is None:
        c = _fft_conv_full(a, a)
    k = min(k, len(c))
    top_idx = np.argpartition(c, -k)[-k:]
    w = np.zeros(len(c))
    w[top_idx] = 1.0
    g = _fft_cross_corr(a, w)
    g = np.maximum(g, 0.0)
    return g, c, top_idx


def _mmd_topk(a0, rng, time_limit, k_start=1, k_end=256, momentum=0.5,

    Returns (best_a, best_score).
    """

    a = _sanitize(a)
    if a is None:
        return a0, float('inf')

    n = len(a)
    deadline = time.monotonic() + time_limit

    best_a = a.copy()
    best_score = _score_normalized(a)
    cur_score = best_score

    g_mom = np.zeros(n)
    k = k_start
    alpha = alpha_start
    it = 0
    stall = 0
    annealing_T = 5.0

    while time.monotonic() < deadline:
        c = _fft_conv_full(a, a)
        cur_score = float(2.0 * n * np.max(c))

        if cur_score < best_score:
            best_a = a.copy()
            best_score = cur_score
            stall = 0
            if verbose and it % 200 == 0:
                print(f"  MMD n={n} it={it} score={best_score:.6f} k={k} alpha={alpha:.4f}")
        else:
            stall += 1

        # Adaptively increase k
        progress = min(1.0, it / max(1, int(time_limit * 2)))
        k = int(k_start + (k_end - k_start) * progress * 0.7)
        k = max(1, min(k, n // 2))

        # Compute gradient
        g, c, top_idx = _topk_gradient(a, k, c=c)
        g = np.maximum(g, 1e-18)

        # Momentum update
        g_mom = momentum * g_mom + (1 - momentum) * g
        g_use = g_mom
        g_max = float(np.max(g_use))
        if g_max > 1e-15:
            g_use = g_use / g_max

        # Line search
        found = False
        for mult in (0.25, 0.5, 1.0, 2.0, 4.0, 8.0):
            a_try = alpha * mult
            if a_try < 1e-5 or a_try > 50:
                continue
            a_new = a * np.exp(-a_try * g_use)
            a_new = np.maximum(a_new, 1e-30)
            s = a_new.sum()
            if s < 1e-15:
                continue
            a_new = a_new / s
            sc_new = _score_normalized(a_new)
            if sc_new < cur_score - 1e-12:
                a = a_new
                cur_score = sc_new
                alpha = a_try
                found = True
                break

        if not found:
            # Try smaller steps
            for mult in (0.1, 0.05, 0.02, 0.01):
                a_try = alpha * mult
                a_new = a * np.exp(-a_try * g_use)
                a_new = np.maximum(a_new, 1e-30)
                s = a_new.sum()
                if s < 1e-15:
                    continue
                a_new = a_new / s
                sc_new = _score_normalized(a_new)
                if sc_new < cur_score - 1e-12:
                    a = a_new
                    cur_score = sc_new
                    alpha = a_try
                    found = True
                    break

        if not found:
            alpha = max(0.01, alpha * 0.7)
            if stall > 30:
                # Annealing restart
                T_step = annealing_T / (1 + 0.01 * stall)
                noise = np.exp(rng.normal(0, T_step, n))
                a_new = a * noise
                a_new = _sanitize(a_new)
                if a_new is not None:
                    sc_new = _score_normalized(a_new)
                    accept_prob = np.exp(-(sc_new - best_score) / max(0.1, annealing_T))
                    if sc_new < best_score or rng.random() < accept_prob:
                        a = a_new
                        cur_score = sc_new
                    else:
                        a = best_a.copy()
                        cur_score = best_score
                else:
                    a = best_a.copy()
                    cur_score = best_score
                g_mom = np.zeros(n)
                stall = 0
                annealing_T *= 0.95

        it += 1

    # Final exact check
    final_score = _exact_score(best_a)
    return best_a, final_score


def _local_block_search(a0, rng, time_limit):
    """Block-wise exact search: multiply contiguous blocks by factors."""
    a = _sanitize(a0).copy()
    if a is None:
        return a0, float('inf')
    n = len(a)
    deadline = time.monotonic() + time_limit
    best = a.copy()
    best_sc = _score_normalized(a)
    cur_sc = best_sc
    stall = 0

    factors = (0.3, 0.5, 0.7, 0.85, 1.15, 1.3, 1.5, 2.0, 3.0)

    while time.monotonic() < deadline:
        # Choose block width
        r = rng.random()
        if r < 0.2:
            w = int(rng.integers(1, max(2, n // 500 + 1)))
        elif r < 0.6:

        else:

        w = max(1, min(n, w))
        start = int(rng.integers(0, n - w + 1))

        best_f = None
        best_f_sc = base_sc

            cand = _sanitize(cand)

                best_f_sc = sc
                best_f = cand

            a = best_f
            cur_sc = best_f_sc
            if cur_sc < best_sc:
                best = a.copy()
                best_sc = cur_sc
            stall = 0
        else:
            stall += 1
            if stall > 40:
                # Perturb
                noise = np.exp(rng.normal(0, 0.08, n))
                a_new = _sanitize(a * noise)
                if a_new is not None:
                    sc = _score_normalized(a_new)
                    if sc < best_sc * 1.1:
                        a = a_new
                        cur_sc = sc
                    else:
                        a = best.copy()
                        cur_sc = best_sc
                else:
                    a = best.copy()
                    cur_sc = best_sc
                stall = 0

    final_sc = _exact_score(best)
    return best, final_sc


def _coordinate_polish(a0, rng, time_limit):
    """
    Targeted coordinate polish: for each coordinate, try adjusting it
    based on its contribution to peak convolution values.
    """
    a = _sanitize(a0).copy()
    if a is None:
        return a0, float('inf')
    n = len(a)
    deadline = time.monotonic() + time_limit
    best = a.copy()
    best_sc = _score_normalized(a)
    stall = 0

    while time.monotonic() < deadline:
        c = _fft_conv_full(a, a)
        cur_sc = float(2.0 * n * np.max(c))
        peak_pos = int(np.argmax(c))

        # Find coordinates that most contribute to the peak
        lo = max(0, peak_pos - n + 1)
        hi = min(n - 1, peak_pos)
        # Contribution of a[j] to c[peak_pos] is 2*a[peak_pos - j]
        contrib = np.zeros(n)
        contrib[lo:hi + 1] = 2.0 * a[peak_pos - (lo:hi + 1)]

        # Try adjusting the top contributors
        top_contrib_idx = np.argsort(contrib)[-max(1, n // 10):][::-1]

        improved = False
        for j in top_contrib_idx:
            j = int(j)
            if a[j] < 1e-15:
                continue
            for factor in (0.5, 0.7, 0.85, 1.15, 1.3, 1.5):
                cand = a.copy()
                cand[j] *= factor
                # Redistribute the difference to the minimum coordinate
                diff = cand[j] - a[j]
                if abs(diff) > 1e-15:
                    min_idx = int(np.argmin(cand))
                    cand[min_idx] += diff
                cand = _sanitize(cand)
                if cand is None:
                    continue
                sc = _score_normalized(cand)
                if sc < best_sc - 1e-12:
                    a = cand
                    best_sc = sc
                    best = a.copy()
                    improved = True
                    break
            if improved:
                break

        if not improved:
            stall += 1
            if stall > 20:
                # Try random coordinate pairs
                i1 = int(rng.integers(n))
                i2 = int(rng.integers(n))
                if i1 == i2:
                    continue
                for r in (0.3, 0.6, 0.8, 1.2, 1.4, 1.7):
                    cand = a.copy()
                    cand[i1] *= r
                    cand[i2] /= r
                    cand = _sanitize(cand)
                    if cand is None:
                        continue
                    sc = _score_normalized(cand)
                    if sc < best_sc - 1e-12:
                        a = cand
                        best_sc = sc
                        best = a.copy()
                        improved = True
                        break
                if not improved:
                    a = best.copy()
                stall = 0

    final_sc = _exact_score(best)
    return best, final_sc


def propose_candidate(seed=12345, budget_s=2400, **kwargs):
    """
    Search for the best non-negative sequence within budget_s seconds.
    Returns a list of non-negative floats.
    """
    total_budget = float(budget_s)
    if not np.isfinite(total_budget) or total_budget < 60:
        total_budget = 60.0

    t_start = time.monotonic()
    deadline = t_start + total_budget - 15.0  # Reserve 15s for final checks

    rng = np.random.default_rng((int(seed) % 1000000007) + 42)

    # Get parent sequence if available
    parent_arr = None
    try:
        parent_raw = height_sequence_1
        parent_arr = _sanitize(np.asarray(parent_raw, dtype=float))
    except (NameError, Exception):
        pass

    best_arr = None
    best_score = float('inf')

    def update_best(arr, sc):
        nonlocal best_arr, best_score
        if arr is not None and np.isfinite(sc) and sc < best_score:
            best_arr = arr.copy()
            best_score = sc
            print(f"  NEW BEST: score={sc:.6f}, n={len(arr)}")

    # === Phase 1: Quick exploration on small n ===
    rem = deadline - time.monotonic()
    if rem > 120:
        print(f"Phase 1: Quick exploration on n=512, budget={rem:.0f}s")
        small_n = 512
        t_alloc = min(100.0, rem * 0.25)

        starts = []
        if parent_arr is not None:
            sp = _resample(parent_arr, small_n)
            if sp is not None:
                starts.append(sp)
        for mode in range(6):
            starts.append(_init_smooth_random(small_n, rng, mode))

        small_best = None
        small_best_sc = float('inf')
        per_start = t_alloc / len(starts)

        for sp in starts:
            rem2 = deadline - time.monotonic()
            if rem2 < 30:
                break
            p, sc = _mmd_topk(sp, rng, min(per_start, rem2 - 10),
                              k_start=1, k_end=64, momentum=0.3, alpha_start=0.2)
            if p is not None and sc < small_best_sc:
                small_best = p
                small_best_sc = sc

        if small_best is not None:
            update_best(small_best, small_best_sc)

    # === Phase 2: Main optimization at target length ===
    rem = deadline - time.monotonic()
    print(f"Phase 2: Main optimization, remaining={rem:.0f}s")

    # Choose length
    if rem > 1200:
        n_main = 4096
    elif rem > 800:
        n_main = 3072
    else:
        n_main = 2048

    # Build starts for main optimization
    main_starts = []
    if parent_arr is not None:
        sp = _resample(parent_arr, n_main)
        if sp is not None:
            main_starts.append(sp)
    if small_best is not None:
        sp = _resample(small_best, n_main)
        if sp is not None:
            main_starts.append(sp)
            # Also a perturbed version
            sp2 = _sanitize(sp * np.exp(rng.normal(0, 0.05, n_main)))
            if sp2 is not None:
                main_starts.append(sp2)
    for mode in (0, 1, 2, 4, 5):
        main_starts.append(_init_smooth_random(n_main, rng, mode))

    # Score and sort starts
    scored = []
    for sp in main_starts:
        sc = _score_normalized(sp)
        if np.isfinite(sc):
            scored.append((sc, sp))
    scored.sort(key=lambda x: x[0])

    # Run main optimization on top starts
    rem = deadline - time.monotonic()
    n_starts = min(4, len(scored))
    if n_starts == 0:
        n_starts = 1
        scored = [(_score_normalized(_init_uniform(n_main)), _init_uniform(n_main))]

    # Allocate time: best start gets most time
    fracs = [0.40, 0.25, 0.20, 0.15]

    for i in range(n_starts):
        rem = deadline - time.monotonic()
        if rem < 60:
            break
        frac = fracs[i] if i < len(fracs) else 0.1
        t_alloc = rem * frac * 0.85  # Leave some buffer
        t_alloc = max(30.0, t_alloc)

        sc0, sp0 = scored[i]
        print(f"  Main start {i}: n={n_main}, init_score={2*n_main*np.max(_fft_conv_full(sp0,sp0))/sp0.sum()**2:.6f}, time={t_alloc:.0f}s")

        p, sc = _mmd_topk(sp0, rng, t_alloc,
                          k_start=1, k_end=max(32, n_main // 16),
                          momentum=0.5, alpha_start=0.1, verbose=True)
        if p is not None:
            update_best(p, sc)

        # Quick local polish after each main run
        rem = deadline - time.monotonic()
        if rem > 60 and p is not None:
            t_polish = min(60.0, rem * 0.1)
            p2, sc2 = _local_block_search(p, rng, t_polish)
            if p2 is not None:
                update_best(p2, sc2)

    # === Phase 3: Deep polish on the best ===
    rem = deadline - time.monotonic()
    if rem > 100 and best_arr is not None:
        print(f"Phase 3: Deep polish, remaining={rem:.0f}s, current best={best_score:.6f}")

        # Coordinate polish
        t1 = min(120.0, rem * 0.25)
        p, sc = _coordinate_polish(best_arr, rng, t1)
        if p is not None:
            update_best(p, sc)

        # More MMD with higher beta equivalent (larger k)
        rem = deadline - time.monotonic()
        if rem > 80 and best_arr is not None:
            t2 = min(150.0, rem * 0.35)
            p, sc = _mmd_topk(best_arr, rng, t2,
                              k_start=max(8, len(best_arr) // 64),
                              k_end=max(64, len(best_arr) // 8),
                              momentum=0.7, alpha_start=0.05)
            if p is not None:
                update_best(p, sc)

        # Block polish
        rem = deadline - time.monotonic()
        if rem > 60 and best_arr is not None:
            t3 = min(100.0, rem * 0.3)
            p, sc = _local_block_search(best_arr, rng, t3)
            if p is not None:
                update_best(p, sc)

        # Final MMD
        rem = deadline - time.monotonic()
        if rem > 40 and best_arr is not None:
            t4 = min(80.0, rem * 0.5)
            p, sc = _mmd_topk(best_arr, rng, t4,
                              k_start=len(best_arr) // 16,
                              k_end=len(best_arr) // 4,
                              momentum=0.8, alpha_start=0.03)
            if p is not None:
                update_best(p, sc)

    # === Phase 4: Try upscaling ===
    rem = deadline - time.monotonic()
    if rem > 120 and best_arr is not None and len(best_arr) <= 4096:
        print(f"Phase 4: Upscale to n={2*len(best_arr)}, remaining={rem:.0f}s")
        n_big = min(8192, 2 * len(best_arr))
        big = _resample(best_arr, n_big)
        if big is not None:
            t_big = min(100.0, rem * 0.5)
            p, sc = _mmd_topk(big, rng, t_big,
                              k_start=4, k_end=n_big // 16,
                              momentum=0.5, alpha_start=0.1)
            if p is not None:
                update_best(p, sc)

            rem = deadline - time.monotonic()
            if rem > 40:
                p2, sc2 = _local_block_search(p, rng, min(40.0, rem * 0.5))
                if p2 is not None:
                    update_best(p2, sc2)

    # === Final: verify with official evaluator ===
    rem = deadline - time.monotonic()
    if best_arr is None:
        best_arr = _init_uniform(1024)

    final_arr = _sanitize(best_arr)
    if final_arr is None:
        final_arr = _init_uniform(1024)

    # Try a few variations with the official evaluator
    candidates = [final_arr.copy()]
    if parent_arr is not None:
        candidates.append(parent_arr.copy())

    selected = [float(x) for x in final_arr]
    sel_score = best_score

    for cand in candidates:
        if time.monotonic() > deadline - 3:
            break
        lst = [float(x) for x in cand]
        try:
            sc = float(evaluate_sequence(lst))
        except Exception:
            sc = _exact_score(cand)
        if np.isfinite(sc) and sc < sel_score:
            sel_score = sc
            selected = lst

    # Final sanitization
    out = np.asarray(selected, dtype=float)
    out = np.where(np.isfinite(out), out, 0.0)
    out = np.clip(out, 0.0, 1000.0)
    s = float(out.sum())
    if s <= 0 or not np.isfinite(s):
        out = np.ones(1024, dtype=float)
    else:
        # Don't normalize - the evaluator handles any scale
        # But ensure values are reasonable
        out = out  # Keep as-is since score is scale invariant

    print(f"Final: n={len(out)}, internal_score={sel_score:.6f}")
    print(f"Total time: {time.monotonic() - t_start:.1f}s")

    return [float(x) for x in out]
