```python
def project_simplex(v):
    """Project v onto the probability simplex {x: x>=0, sum(x)=1}."""
    n = len(v)
    if np.sum(v) <= 0:
        return np.ones(n) / n
    # Binary search for the threshold
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u * np.arange(1, n+1) > (cssv - 1))[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1.0)
    w = np.maximum(v - theta, 0)
    return w / w.sum()
```

                    stall = 0
                
                if stall > 100:
                    p = best.copy()
                    cur_exact = best_exact
                    _, cur_smooth, _ = _gradient_smooth(p, beta)
                    alpha = 0.3
                    stall = 0
        
        it += 1
        if it % 10 == 0:
            beta = min(beta_end, beta * 1.3)
    
    return best, best_exact


def _block_refine(p0, rng, time_limit):
    """Block coordinate refinement: multiply contiguous blocks."""
    p = _sanitize(p0)
    if p is None or time_limit <= 0:
        return None, float('inf')
    
    n = p.size
    floor = 1e-20 / n
    p = np.maximum(p, floor)
    p = p / p.sum()
    
    best = p.copy()
    best_sc = _score(p)
    deadline = time.monotonic() + time_limit
    stagnant = 0
    
    while time.monotonic() < deadline:
        # Choose block size
        r = rng.random()
        if r < 0.3:
            w = rng.integers(1, max(2, n // 100 + 1))
        elif r < 0.8:
            w = rng.integers(max(2, n // 100), max(3, n // 20 + 1))
        else:
            w = rng.integers(n // 20, n // 5 + 1)
        
        w = max(1, min(n, int(w)))
        start = rng.integers(0, n - w + 1)
        
        base = p.copy()
        cur_best = base
        cur_best_sc = best_sc if p is best else _score(p)
        
        for f in (0.4, 0.6, 0.8, 1.0, 1.25, 1.5, 2.0, 3.0):
            cand = base.copy()
            cand[start:start+w] *= f
            cand = _sanitize(cand)
            if cand is None:
                continue
            cand = np.maximum(cand, floor)
            cand = cand / cand.sum()
            sc = _score(cand)
            if sc < cur_best_sc - 1e-12:
                cur_best_sc = sc
                cur_best = cand
        
        if cur_best is not base:
            p = cur_best
            if _score(p) < best_sc:
                best = p.copy()
                best_sc = _score(best)
            stagnant = 0
        else:
            stagnant += 1
            if stagnant > 40:
                p_try = p * np.exp(rng.normal(0, 0.05, n))
                p_try = np.maximum(p_try, floor)
                p_try = p_try / p_try.sum()
                sc = _score(p_try)
                if np.isfinite(sc) and sc <= best_sc * 1.03:
                    p = p_try
                else:
                    p = best.copy()
                    best_sc = _score(p)
                stagnant = 0
            if stagnant > 200:
                p = best.copy()
                stagnant = 0
    
    return best, _score(best)
