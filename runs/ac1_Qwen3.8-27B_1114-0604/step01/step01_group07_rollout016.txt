def _next_pow2(m):

    n_orig = a.size
    candidates = []
    for n_new in [n_orig * 2, n_orig * 4]:
        if n_new > 16384:
            break
        # Create a new sinusoidal at the larger n with similar amplitude
        # Extract approximate amplitude from current sequence
        mean_val = float(a.mean())
        var = float(np.var(a))
        # Estimate amplitude: for a[i] = (1+A*sin(...))/n, var ≈ A^2/(4n^2)*n = A^2/(4n)
        # So A ≈ 2*sqrt(var*n)
        est_amp = min(1.0, 2.0 * np.sqrt(var * n_orig))
        a_new = _make_sine(n_new, est_amp)
        sc = _internal_score(a_new)
        candidates.append((sc, a_new, n_new))
    return candidates


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    t0 = time.monotonic()
    total_budget = max(60.0, float(budget_s))
    hard_deadline = t0 + total_budget - 30.0

    # Try to use the previous best as a starting point
    start_a = None
    if 'height_sequence_1' in dir() or True:
        try:
            hs = height_sequence_1
            if hs is not None and len(hs) > 10:
                arr = np.array(hs, dtype=np.float64)
                arr = np.clip(arr, 0, 1000)
                s = arr.sum()
                if s > 0.01:
                    arr = arr / s
                    start_a = arr
                    sc = _internal_score(arr)
                    print(f"[init] using height_sequence_1 n={arr.size} score={sc:.8f}")
        except (NameError, TypeError, ValueError):
            pass

    # Phase 1: Parametric search to find a good starting point
    param_deadline = t0 + 120.0
    param_a, param_score = _parametric_search(param_deadline)
    print(f"[phase1] parametric best score={param_score:.8f}")

    # Choose the best starting point
    if start_a is not None:
        start_score = _internal_score(start_a)
    else:
        start_score = np.inf

    if param_score < start_score:
        best_start = param_a
        best_start_score = param_score
        print(f"[phase1] using parametric start (score={param_score:.8f})")
    else:
        best_start = start_a if start_a is not None else param_a
        best_start_score = min(param_score, start_score)
        print(f"[phase1] using height_sequence_1 start (score={best_start_score:.8f})")

    # Phase 2: Local optimization with mirror descent
    remaining = hard_deadline - time.monotonic()
    if remaining > 60.0:
        # Run mirror descent on the best starting point
        md_budget = min(remaining - 30.0, 1200.0)
        md_a, md_score = _mirror_descent(best_start, md_budget, K=500, temp_init=0.04)
        if np.isfinite(md_score) and md_score < best_start_score:
            best_start = md_a
            best_start_score = md_score
            print(f"[phase2] mirror descent improved to {md_score:.8f}")

        # If time remains, try with a different K or restart
        remaining2 = hard_deadline - time.monotonic()
        if remaining2 > 120.0:
            # Try with softer temperature
            md_budget2 = min(remaining2 - 30.0, 600.0)
            md_a2, md_score2 = _mirror_descent(best_start, md_budget2, K=200, temp_init=0.15)
            if np.isfinite(md_score2) and md_score2 < best_start_score:
                best_start = md_a2
                best_start_score = md_score2
                print(f"[phase2b] softer MD improved to {md_score2:.8f}")

        # Try with larger n if we have time
        remaining3 = hard_deadline - time.monotonic()
        if remaining3 > 180.0 and best_start.size <= 4096:
            n_new = best_start.size * 2
            if n_new <= 8192:
                est_amp = min(1.0, 0.75)
                a_larger = _make_sine(n_new, est_amp)
                # Optimize amplitude quickly
                for amp in [0.6, 0.7, 0.75, 0.8, 0.85, 0.9]:
                    a_try = _make_sine(n_new, amp)
                    sc_try = _internal_score(a_try)
                    if sc_try < _internal_score(a_larger):
                        a_larger = a_try
                sc_larger = _internal_score(a_larger)
                if sc_larger < best_start_score:
                    print(f"[phase3] larger n={n_new} parametric score={sc_larger:.8f}")
                    md_budget3 = min(remaining3 - 30.0, 400.0)
                    md_a3, md_score3 = _mirror_descent(a_larger, md_budget3, K=500, temp_init=0.04)
                    if np.isfinite(md_score3) and md_score3 < best_start_score:
                        best_start = md_a3
                        best_start_score = md_score3
                        print(f"[phase3] larger n MD improved to {md_score3:.8f}")

    # Phase 3: Final validation
    print(f"[final] best internal score={best_start_score:.8f} n={best_start.size}")

    # Validate with actual evaluate_sequence
    final_arr = np.asarray(best_start, dtype=np.float64)
    final_arr = np.clip(final_arr, 0.0, 1000.0)
    s = float(final_arr.sum())
    if s < 0.01:
        final_arr = np.ones(2048, dtype=np.float64) / 2048.0

    # Try a few candidates with evaluate_sequence
    candidates = [final_arr]
    # Also try the parametric best
    if param_a is not None and not np.allclose(param_a, final_arr):
        pa = np.clip(np.asarray(param_a, dtype=np.float64), 0, None)
        ps = pa.sum()
        if ps > 0.01:
            candidates.append(pa / ps)
    # Fallback
    fb = _make_sine(2048, 0.7)
    candidates.append(fb)

    best_actual = np.inf
    best_out = None
    for cand in candidates:
        if time.monotonic() > hard_deadline:
            break
        try:
            real_score = float(evaluate_sequence(cand.tolist()))
        except Exception:
            real_score = np.inf
        if np.isfinite(real_score) and real_score < best_actual:
            best_actual = real_score
            best_out = cand
            print(f"[validate] n={cand.size} actual_score={real_score:.8f}")

    if best_out is None:
        best_out = _make_sine(2048, 0.7)
        print("[validate] using fallback")

    return [float(x) for x in best_out]
