    # cross_corr(w,a)_n = sum_k w_{k-n} a_k = sum_j w_j a_{j+n} = sum_j w_j a_{j+n}
    # Hmm, let me just use the direct approach with reversed arrays.
    # 
    # g_i = sum_k w_k a_{i+k} + sum_k w_k a_{i-k}
    # = sum_k w_k a_{i+k} + sum_k w_{2m-2-k} a_{i+k} (substituting k' = 2m-2-k... no)
    #
    # Let me just use: g = ifft(fft_w * fft_a_rev)[n-1:m] + ifft(fft_w * fft_a)[n-1:m]
    # where fft_a_rev = fft(a[::-1] padded)
    # 
    # Actually the cleanest: 
    # term1_i = sum_k w_k a_{i+k} = (w * a)_{i} in "cross-correlation" sense
    # = the i-th element of the sequence where we slide w over a
    # In terms of standard convolution: if we reverse w, then (w_rev * a)_n = sum_k w_rev_k a_{n-k} = sum_k w_{m-1-k} a_{n-k}
    # This is getting confusing. Let me just use the approach from the original code.
    
    # From original code: g = ifft(fft(w) * fft(a_rev))[n-1:m] where a_rev = a[::-1]
    # This gives sum_k w_k * a_rev_{n-1-i-k+...} ... let me just trust the original.
    a_rev = a[::-1]
    Frev = np.fft.rfft(a_rev, L)
    g = np.fft.irfft(W * Frev, L)[(n - 1):m]
    np.maximum(g, 0.0, out=g)
    return g


def _random_init(n, rng):
    """Generate a random initial candidate."""
    t = np.linspace(0.0, 1.0, n)
    x = 2.0 * t - 1.0
    z = np.zeros(n)
    # Add polynomial components
    z += rng.uniform(-0.3, 0.3)
    z += rng.uniform(-0.5, 0.5) * x
    z += rng.uniform(-0.5, 0.5) * (x * x - 1.0 / 3.0)
    # Add sinusoidal components
    for k in range(1, 6):
        amp = rng.uniform(0.1, 0.5) / math.sqrt(k)
        ph = rng.uniform(0, 2 * np.pi)
        z += amp * np.cos(2.0 * np.pi * k * t + ph)
    # Add smoothed noise
    noise = rng.normal(0, 0.2, n)
    if n > 10:
        k = 5
        kernel = np.ones(k) / k
        noise = np.convolve(noise, kernel, mode='same')
    z += noise
    p = np.exp(z - z.max())
    p = np.maximum(p, 1e-20)
    return _norm(p)


def _shape_inits(n, rng):
    """Generate shape-based initial candidates."""
    out = []
    t = np.linspace(0.0, 1.0, n)
    # Raised cosine
    out.append(_norm(0.5 * (1 + np.cos(2 * np.pi * t))))
    # Hanning
    out.append(_norm(0.5 * (1 + np.cos(np.pi * (2 * t - 1)))))
    # Triangle
    out.append(_norm(1 - np.abs(2 * t - 1)))
    # Parabolic
    out.append(_norm(1 - (2 * t - 1)**2))
    # Gaussian-like
    out.append(_norm(np.exp(-2 * (2 * t - 1)**2)))
    # Flat (uniform)
    out.append(_norm(np.ones(n)))
    # Random
    for _ in range(3):
        out.append(_random_init(n, rng))
    return out


def _core_search(a, L, m, n, rng, time_limit):
    """Main optimization: adaptive-temperature soft-max gradient descent."""
    deadline = time.monotonic() + time_limit
    a = _norm(a)
    c = _fft_conv(a, L, m)
    cur_score = _score_c(c, n)
    best_a = a.copy()
    best_score = cur_score
    
    temp = 2.0
    eta = 0.15
    no_improve = 0
    no_accept = 0
    it = 0
    
    while time.monotonic() < deadline:
        if it % 50 == 0:
            c = _fft_conv(a, L, m)
            cur_score = _score_c(c, n)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best_a = a.copy()
                no_improve = 0
            else:
                no_improve += 1
            
            # Adaptive temperature
            if no_improve > 0 and no_improve % 10 == 0:
                temp = min(temp * 1.5, 500.0)
            elif cur_score < best_score:
                temp = max(temp * 0.8, 0.5)
        
        # Compute direction
        c = _fft_conv(a, L, m)
        g = _mws_dir(a, L, m, n, temp)
        gmax = float(g.max())
        if gmax <= 0 or not np.isfinite(gmax):
            break
        d = g / gmax
        
        # Try updates with decreasing step sizes
        accepted = False
        for step in [eta, 0.5 * eta, 0.25 * eta]:
            if step < 1e-6:
                break
            a_new = a * np.exp(-step * d)
            a_new = _norm(a_new)
            c_new = _fft_conv(a_new, L, m)
            sc_new = _score_c(c_new, n)
            if sc_new < cur_score - 1e-14:
                a = a_new
                cur_score = sc_new
                accepted = True
                if step == eta:
                    eta = min(0.5, eta * 1.1)
                else:
                    eta = max(0.02, eta * 0.85)
                no_accept = 0
                break
        if not accepted:
            no_accept += 1
            eta = max(0.02, eta * 0.8)
        
        # Restart if stuck
        if no_accept > 50:
            if rng.random() < 0.3:
                # Random restart
                a = _random_init(n, rng)
            else:
                # Perturbed restart from best
                pert = rng.normal(0, 0.3, n)
                a = _norm(best_a * np.exp(pert))
            c = _fft_conv(a, L, m)
            cur_score = _score_c(c, n)
            temp = max(1.0, temp * 0.5)
            eta = 0.15
            no_accept = 0
        
        # Gradually increase temperature for exploration
        if it % 100 == 0:
            temp = min(temp * 1.1, 500.0)
        
        it += 1
    
    return best_a, best_score


def _coord_polish(a, L, m, n, time_limit, rng):
    """Coordinate descent polish: try improving each coordinate."""
    deadline = time.monotonic() + time_limit
    a = _norm(a)
    c = _fft_conv(a, L, m)
    cur_score = _score_c(c, n)
    best_score = cur_score
    best_a = a.copy()
    
    indices = np.arange(n)
    pass_count = 0
    
    while time.monotonic() < deadline:
        rng.shuffle(indices)
        improved = False
        for i in indices:
            if time.monotonic() > deadline - 2:
                break
            old_val = a[i]
            if old_val < 1e-15:
                continue
            # Try geometric grid around current value
            ratios = [0.3, 0.5, 0.7, 1.0 / 0.7, 1.0 / 0.5, 1.0 / 0.3]
            best_local = old_val
            best_local_score = cur_score
            S = float(a.sum())
            
            for r in ratios:
                new_val = old_val * r
                a[i] = new_val
                S_new = S - old_val + new_val
                # Update affected c values
                # c_k changes by (new-old)*(a_{i+k} + a_{i-k}) for k!=0
                # c_0 changes by new^2 - old^2
                # For efficiency, just recompute via FFT occasionally
                # For now, do a quick check using the formula
                delta = new_val - old_val
                # Change in c_k:
                # For k where i+k < n: c_{i+k} += delta * a[i+k]... 
                # This is O(n) per trial, too slow for many trials.
                # Let's just do FFT every few trials
                if abs(r - 1.0) < 0.3:
                    continue  # skip values close to current
                a[i] = new_val
                c_try = _fft_conv(a, L, m)
                sc_try = _score_c(c_try, n)
                if sc_try < best_local_score:
                    best_local_score = sc_try
                    best_local = new_val
                a[i] = old_val
            
            if best_local != old_val:
                a[i] = best_local
                c = _fft_conv(a, L, m)
                cur_score = _score_c(c, n)
                improved = True
                if cur_score < best_score:
                    best_score = cur_score
                    best_a = a.copy()
        
        pass_count += 1
        if not improved and pass_count > 1:
            break
    
    return best_a, best_score


def _rescale_seq(arr, target_n):
    """Rescale a sequence to a target length using interpolation."""
    n = len(arr)
    if n == target_n:
        return arr.copy()
    x_old = np.linspace(0, 1, n)
    x_new = np.linspace(0, 1, target_n)
    return np.interp(x_new, x_old, arr)


def _exact(seq):
    try:
        val = evaluate_sequence(seq)
        return float(val) if val is not None else float('inf')
    except NameError:
        a = [max(0.0, min(1000.0, float(x))) for x in seq]
        n = len(a)
        s = sum(a)
        if s < 0.01:
            return float('inf')
        b = np.convolve(a, a)
        return float(2.0 * n * max(b) / (s * s))
    except Exception:
        return float('inf')


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    try:
        budget = float(budget_s)
    except Exception:
        budget = 2400.0
    if not np.isfinite(budget) or budget < 10:
        budget = 2400.0
    
    start_time = time.monotonic()
    hard_deadline = start_time + budget - 30.0  # Leave 30s for evaluation
    rng = np.random.default_rng(seed)
    
    # Get parent sequence if available
    parent = None
    try:
        parent = np.asarray(height_sequence_1, dtype=np.float64)
        if parent.size > 0 and np.all(np.isfinite(parent)):
            parent = np.maximum(parent, 0)
            if parent.sum() < 0.01:
                parent = None
    except NameError:
        parent = None
    except Exception:
        parent = None
    
    # Candidate n values
    n_candidates = [4000, 5000, 6121, 7000, 8000]
    if parent is not None:
        # Include parent's n
        pn = len(parent)
        if pn not in n_candidates:
            n_candidates.append(pn)
        n_candidates.sort()
    
    best_overall_score = float('inf')
    best_overall_seq = None
    results = []
    
    # Allocate time proportionally (give more time to parent's n and middle values)
    total_time_available = hard_deadline - time.monotonic()
    # Weight: parent n gets 2x, others get 1x
    weights = []
    if parent is not None:
        for n in n_candidates:
            weights.append(2.0 if len(parent) == n else 1.0)
    else:
        weights = [1.0] * len(n_candidates)
    total_w = sum(weights)
    
    # Limit to at most 4 n values to give each enough time
    if len(n_candidates) > 4:
        # Keep parent's n and the 3 closest
        if parent is not None:
            pn = len(parent)
            others = [n for n in n_candidates if n != pn]
            others.sort(key=lambda x: abs(x - pn))
            n_candidates = [pn] + others[:3]
            weights = [2.0] + [1.0] * 3
        else:
            n_candidates = n_candidates[:4]
            weights = weights[:4]
        total_w = sum(weights)
    
    print(f"Trying n values: {n_candidates}")
    print(f"Total time available: {total_time_available:.1f}s")
    
    for idx, n in enumerate(n_candidates):
        t_remaining = hard_deadline - time.monotonic()
        if t_remaining < 60:
            break
        
        # Allocate time for this n
        fraction = weights[idx] / total_w
        # But don't allocate more than what's remaining
        alloc = min(fraction * total_time_available, t_remaining - 30)
        alloc = max(alloc, 30)  # Minimum 30s
        if idx == len(n_candidates) - 1:
            alloc = t_remaining - 15  # Give last one the rest
        
        print(f"\n--- n={n}, time alloc={alloc:.1f}s ---")
        t0 = time.monotonic()
        
        L = 1 << (2 * n - 2).bit_length()
        m = 2 * n - 1
        
        # Initialize
        inits = []
        if parent is not None:
            p_resized = _rescale_seq(parent, n)
            inits.append(_norm(p_resized))
        inits.extend(_shape_inits(n, rng))
        
        best_a = None
        best_score = float('inf')
        
        # Evaluate initial candidates
        for a in inits:
            c = _fft_conv(a, L, m)
            sc = _score_c(c, n)
            if sc < best_score:
                best_score = sc
                best_a = a.copy()
        
        if best_a is None:
            best_a = _norm(np.ones(n))
            best_score = _score_c(_fft_conv(best_a, L, m), n)
        
        print(f"  Initial best: {best_score:.6f}")
        
        # Phase 1: Core search (70% of allocated time)
        t_phase1 = 0.70 * alloc
        tmp_a, tmp_sc = _core_search(best_a, L, m, n, rng, t_phase1)
        if tmp_sc < best_score:
            best_score = tmp_sc
            best_a = tmp_a
        print(f"  After core search: {best_score:.6f}")
        
        # Phase 2: Coordinate polish (20% of allocated time)
        t_phase2 = 0.20 * alloc
        if time.monotonic() < hard_deadline - 20:
            tmp_a, tmp_sc = _coord_polish(best_a, L, m, n, t_phase2, rng)
            if tmp_sc < best_score:
                best_score = tmp_sc
                best_a = tmp_a
            print(f"  After coord polish: {best_score:.6f}")
        
        # Phase 3: Final core search from polished result (10% of allocated time)
        t_phase3 = 0.10 * alloc
        if time.monotonic() < hard_deadline - 15:
            tmp_a, tmp_sc = _core_search(best_a, L, m, n, rng, t_phase3)
            if tmp_sc < best_score:
                best_score = tmp_sc
                best_a = tmp_a
            print(f"  After final search: {best_score:.6f}")
        
        # Convert to list and evaluate exactly
        seq = [float(x) for x in best_a]
        exact_val = _exact(seq)
        print(f"  Exact evaluation: {exact_val:.6f}")
        
        if exact_val < best_overall_score:
            best_overall_score = exact_val
            best_overall_seq = seq
        
        results.append((n, exact_val))
        elapsed = time.monotonic() - t0
        print(f"  Completed in {elapsed:.1f}s")
        print(f"  Best overall so far: {best_overall_score:.6f}")
    
    # If we have time, do a final polish on the best
    t_remaining = hard_deadline - time.monotonic()
    if t_remaining > 30 and best_overall_seq is not None:
        n_best = len(best_overall_seq)
        L = 1 << (2 * n_best - 2).bit_length()
        m = 2 * n_best - 1
        a_best = _norm(np.asarray(best_overall_seq))
        
        print(f"\n--- Final polish on n={n_best}, {t_remaining:.1f}s available ---")
        tmp_a, tmp_sc = _coord_polish(a_best, L, m, n_best, t_remaining - 10, rng)
        if tmp_sc < best_overall_score:
            seq2 = [float(x) for x in tmp_a]
            ev = _exact(seq2)
            if ev < best_overall_score:
                best_overall_score = ev
                best_overall_seq = seq2
            print(f"  After final polish: {best_overall_score:.6f}")
    
    if best_overall_seq is None:
        # Fallback: uniform
        n_fb = 6121
        best_overall_seq = [1.0 / n_fb] * n_fb
    
    print(f"\nFinal best score: {best_overall_score:.6f} at n={len(best_overall_seq)}")
    print(f"Total time used: {time.monotonic() - start_time:.1f}s")
    
    return best_overall_seq
