"""
Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Strategy: Adaptive top-k soft-max mirror descent with backtracking,
structured perturbations, and frequency-domain diversity in initialization.
"""

import numpy as np
from scipy.signal import correlate
import time


def _convolve_fft(f):
    """Full linear convolution f*f using FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _topk_weights(c, k, T):
    """Compute top-k soft-max probability vector over c."""
    m = len(c)
    if k >= m:
        cs = c / T
        cs -= cs.max()
        p = np.exp(cs)
        s = p.sum()
        if s < 1e-300:
            p = np.zeros(m)
            p[m // 2] = 1.0
        else:
            p /= s
        return p
    top_idx = np.argpartition(c, -k)[-k:]
    c_top = c[top_idx]
    cs = c_top / T
    cs -= cs.max()
    w = np.exp(cs)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(k)
        w[0] = 1.0
    else:
        w /= s
    p = np.zeros(m)
    p[top_idx] = w
    return p


def _mirror_step(f, grad, eta):
    """Exponentiated gradient step on non-negative simplex."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    f_new /= f_new.sum()
    return f_new

def _run_mirrordescent(f0, n, deadline, rng, verbose=False):
    """
    Adaptive top-k soft-max mirror descent.
    
    Key features:
    - Top-k soft-max with k increasing over time
    - Geometric temperature annealing  
    - Backtracking step size
    - Structured perturbations (sinusoidal)
    """
    f = np.maximum(f0, 0.0).astype(np.float64).copy()
    s = f.sum()
    if s < 1e-10:
        f = np.ones(n)
    else:
        f = f / s

    C = _convolve_fft(f)
    C_max = C.max()
    
    best_score = _eval_seq(f)
    best_f = f.copy()
    
    # Hyperparameters
    T = C_max
    T_min = C_max * 1e-10
    k_top = 1
    base_eta = 1.0
    eta_min = 1e-4
    
    eval_every = 1000
    stuck = 0
    max_k = max(2 * n - 1, 1)
    
    it = 0
    while time.monotonic() < deadline:
        # Compute gradient
        p = _topk_weights(C, k_top, T)
        grad = 2.0 * correlate(p, f, mode='valid')
        
        # Adaptive step size with backtracking
        span = grad.max() - grad.min()
        if span < 1e-15:
            # No gradient signal, perturb
            f += rng.standard_normal(n) * 1e-4 / n
            f = np.maximum(f, 0)
            f /= f.sum()
            C = _convolve_fft(f)
            C_max = C.max()
            it += 1
            continue

        improved = False
        for mult in [base_eta, base_eta * 0.5, base_eta * 0.25, base_eta * 0.1, base_eta * 0.05]:

                C = C_new
                C_max = C.max()

Minimizes 2*n*max(conv(a,a))/(sum(a))^2 for non-negative sequences.
Uses adaptive top-k soft-max mirror descent with backtracking step sizes,
structured sinusoidal perturbations, reheating on stagnation, and
peak-equalization coordinate polish.

import numpy as np

import time


def _convolve_fft(f):
    """Full linear convolution f*f using FFT."""
    n = len(f)
    size = 1 << (2 * n - 1).bit_length()
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _topk_weights(c, k, T):
    """Compute top-k soft-max probability vector over c."""
    m = len(c)
    if k >= m:
        cs = c / T
        cs -= cs.max()
        p = np.exp(cs)
        s = p.sum()
        if s < 1e-300:
            p = np.zeros(m)
            p[m // 2] = 1.0
        else:
            p /= s
        return p
    top_idx = np.argpartition(c, -k)[-k:]
    c_top = c[top_idx]
    cs = c_top / T
    cs -= cs.max()
    w = np.exp(cs)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(k)
        w[0] = 1.0
    else:
        w /= s
    p = np.zeros(m)
    p[top_idx] = w
    return p


def _mirror_step(f, grad, eta):
    """Exponentiated gradient (mirror descent) step on non-negative simplex."""
    log_f = np.log(f + 1e-300) - eta * grad
    log_f -= log_f.max()
    f_new = np.exp(log_f)
    f_new /= f_new.sum()
    return f_new

    stuck = 0

def _polish_peaks(f, n, deadline, rng):
    """
    Peak-equalization polish via coordinate descent.
    Targets the top few convolution peaks by moving mass from
    high-contribution positions to low-contribution ones.
    """
    f = f.copy()
    best_score = _eval_seq(f)
    best_f = f.copy()
    C = _convolve_fft(f)
    C_max = C.max()

    it = 0
    no_improve = 0

    while time.monotonic() < deadline:
        # Find top few peak lags
        top_lags = np.argsort(C)[-3:]

        for peak_lag in top_lags:
            peak_lag = int(peak_lag)
            # Contribution of each position j to C[peak_lag]:
            # dC[peak_lag]/df[j] = 2*f[peak_lag-j] if 0<=peak_lag-j<n
            contrib = np.zeros(n)
            lo = max(0, peak_lag - (n - 1))
            hi = min(n - 1, peak_lag)
            for jx in range(lo, hi + 1):
                idx = peak_lag - jx
                if 0 <= idx < n:
                    contrib[jx] = f[idx]

            donor = int(np.argmax(contrib))
            receiver = int(np.argmin(contrib))

            if donor == receiver or contrib[donor] < 1e-15:
                continue

            # Try various transfer amounts
            for frac in [0.03, 0.01, 0.005]:
                amt = f[donor] * frac
                if amt < 1e-15:
                    continue
                f_try = f.copy()
                f_try[donor] -= amt
                f_try[receiver] += amt
                f_try /= f_try.sum()
                C_try = _convolve_fft(f_try)
                if C_try.max() < C_max - 1e-12:
                    f = f_try
                    C = C_try
                    C_max = C.max()
                    no_improve = 0
                    break

        # Random coordinate moves for diversity
        for _ in range(5):
            i = int(rng.integers(0, n))
            jx = int(rng.integers(0, n))
            if i == jx:
                continue
            f_try = f.copy()
            amt = 0.005 / n
            direction = 1 if f[i] > f[jx] else -1
            f_try[i] -= direction * amt
            f_try[jx] += direction * amt
            if f_try[i] < -1e-12 or f_try[jx] < -1e-12:
                continue
            f_try = np.maximum(f_try, 0)
            f_try /= f_try.sum()
            C_try = _convolve_fft(f_try)
            if C_try.max() < C_max - 1e-12:
                f = f_try
                C = C_try
                C_max = C.max()
                no_improve = 0

        it += 1
        if it % 2000 == 0:
            no_improve += 1
            score = _eval_seq(f)
            if score < best_score:
                best_score = score
                best_f = f.copy()
            if no_improve > 5:
                break

    score = _eval_seq(f)
    if score < best_score:
        best_score = score
        best_f = f.copy()

    return best_f, best_score
