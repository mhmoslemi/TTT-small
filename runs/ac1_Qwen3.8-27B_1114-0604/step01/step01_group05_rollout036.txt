"""
Minimize 2*n*max(a*a)/(sum(a))^2 for non-negative sequences.
Uses FISTA (Nesterov-accelerated projected gradient) with adaptive
smooth-max beta annealing, multi-scale optimization, and multiple restarts.
"""

import numpy as np
import time

try:
    from scipy.optimize import minimize
    HAVE_SCIPY = True
except Exception:
    HAVE_SCIPY = False


def fft_convolution(a):
    a = np.asarray(a, dtype=np.float64)
    n = a.size
    if n == 0:
        return np.empty(0, dtype=np.float64)
    if n == 1:
        return np.array([a[0] * a[0]], dtype=np.float64)
    need = 2 * n - 1
    size = 1
    while size < need:
        size <<= 1
    F = np.fft.rfft(a, size)
    b = np.fft.irfft(F * F, size)[:need]
    b = np.ascontiguousarray(b, dtype=np.float64)
    return np.maximum(b, 0.0)


def smooth_max_and_weights(b, beta):
    b = np.asarray(b, dtype=np.float64)
    if b.size == 0:
        return float("inf"), np.empty(0, dtype=np.float64)
    m = float(np.max(b))
    if not np.isfinite(m):
        return float("inf"), np.zeros_like(b, dtype=np.float64)
    if m <= 0.0 or beta <= 0.0:
        w = np.ones_like(b, dtype=np.float64) / float(b.size)
        return m, w
    x = beta * (b - m)
    x = np.clip(x, -745.0, 0.0)
    e = np.exp(x)
    Z = float(np.sum(e))
    if Z <= 0.0 or not np.isfinite(Z):
        e = np.zeros_like(b, dtype=np.float64)
        e[int(np.argmax(b))] = 1.0
        Z = 1.0
    w = e / Z
    G = m + float(np.log(Z)) / beta
    return float(G), w


def fast_score(p):
    p = np.asarray(p, dtype=np.float64)
    n = p.size
    if n == 0:
        return float("inf")
    S = float(np.sum(p))
    if S <= 0.0 or not np.isfinite(S):
        return float("inf")
    b = fft_convolution(p)
    if b.size == 0:
        return float("inf")
    M = float(np.max(b))
    if not np.isfinite(M):
        return float("inf")
    return float(2.0 * n * M / (S * S))


def corr_pw(p, w):
    p = np.asarray(p, dtype=np.float64)
    w = np.asarray(w, dtype=np.float64)
    n = p.size
    m = w.size
    if n == 0 or m == 0:
        return np.zeros(n, dtype=np.float64)
    if n == 1:
        return np.array([float(np.dot(p, w[:1]))], dtype=np.float64)
    need = n + m - 1
    size = 1
    while size < need:
        size <<= 1
    P = np.fft.rfft(p, size)
    WR = np.fft.rfft(w[::-1], size)
    C = np.fft.irfft(P * WR, size)[:need]
    s = C[m - n: m][::-1].copy()
    return np.ascontiguousarray(s, dtype=np.float64)


def project_simplex(v, target=1.0):
    v = np.asarray(v, dtype=np.float64)
    if v.size == 0:
        return v
    if target <= 0:
        return np.zeros_like(v, dtype=np.float64)
    if not np.all(np.isfinite(v)):
        return np.full(v.shape, target / float(v.size), dtype=np.float64)
    # Use the sort-based O(n log n) method
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u * np.arange(1, v.size + 1) > (cssv - target))[0]
    if len(rho) == 0:
        return np.full(v.shape, target / float(v.size), dtype=np.float64)
    cap = rho[-1]
    theta = (cssv[cap] - target) / (cap + 1.0)
    q = np.maximum(v - theta, 0.0)
    s = float(np.sum(q))
    if s <= 0 or not np.isfinite(s):
        q = np.full(v.shape, target / float(v.size), dtype=np.float64)
    else:
        q *= target / s
    return q

def project_simplex(v, target=1.0):

    u = np.sort(v)[::-1]

    ar = np.arange(1, n + 1, dtype=np.float64)

    rho = np.where(cond)[0]

    s = float(np.sum(q))
    if s <= 0 or not np.isfinite(s):
        return np.full(n, target / n)
    q *= target / s
    return q

        if no_improve > restart_interval:
            p = normalize_simplex(p + restart_noise * rng.standard_normal(n))
            y = p.copy()
            t = 1.0
            step = max(step * 0.5, 0.1)
            no_improve = 0

    final_score = fast_score(best_p)
    if final_score < best_score:
        best_score = final_score
    return best_p, best_score


def gen_structured_starts(n, rng, count=10):
    """Generate diverse structured starting points."""
    starts = []
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)

    # Sine perturbations at various frequencies
    for k in (1, 2, 3, 5, 8, 13, 21, 34):
        p = 1.0 + 0.15 * np.sin(2.0 * np.pi * k * x)
        p = np.maximum(p, 1e-6)
        p /= p.sum()
        starts.append(p)

    # Bathtub shapes
    for c in (2.0, 4.0, 8.0):
        p = np.exp(c * (x - 0.5) ** 2)
        p /= p.sum()
        starts.append(p)

    # Bell shapes (inverse)
    for c in (2.0, 4.0, 8.0):
        p = np.exp(-c * (x - 0.5) ** 2)
        p /= p.sum()
        starts.append(p)

    # Quadratic
    p = 1.0 + 0.3 * (x - 0.5) ** 2
    p /= p.sum()
    starts.append(p)

    # Random smooth
    for _ in range(3):
        p = 1.0 + 0.1 * rng.standard_normal(n)
        # Smooth in frequency domain
        size = 1
        while size < n:
            size <<= 1
        P = np.fft.rfft(p, size)
        freqs = np.fft.rfftfreq(size)
        cutoff = 3.0 / n
        P[np.abs(freqs) > cutoff] *= 0.1
        p = np.real(np.fft.irfft(P, size))[:n]
        p = np.maximum(p, 1e-6)
        p /= p.sum()
        starts.append(p)

    return starts[:count]


def upsample(se
q, new_n):
    """Upsample a sequence to a new length by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return np.asarray(seq, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    p = np.interp(x_new, x_old, np.asarray(seq, dtype=np.float64))
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s > 0:
        p /= s
    return p


def downsample(seq, new_n):
    """Downsample a sequence to a new length by averaging blocks."""
    old_n = len(seq)
    if old_n == new_n:
        return np.asarray(seq, dtype=np.float64)
    a = np.asarray(seq, dtype=np.float64)
    x_old = np.linspace(0.0, 1.0, old_n, dtype=np.float64)
    x_new = np.linspace(0.0, 1.0, new_n, dtype=np.float64)
    p = np.interp(x_new, x_old, a)
    p = np.maximum(p, 0.0)
    s = float(np.sum(p))
    if s > 0:
        p /= s
    return p


def block_objective(c, n, B, beta):
    """Objective and gradient for block-constant log-weight parameterization."""
    L = n // B
    z = np.empty(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        tmp = np.zeros(B - 1, dtype=np.float64)
        m = min(tmp.size, c.size)
        tmp[:m] = c[:m]
        c = tmp
        z[:-1] = c
    z[-1] = 0.0

    if not np.all(np.isfinite(z)):
        return float("inf"), np.zeros(max(B - 1, 0), dtype=np.float64)

    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0.0 or not np.isfinite(s):
        wb = np.ones(B, dtype=np.float64) / float(B)
    else:
        wb /= s

    H = wb / float(L)
    p = np.repeat(H, L)
    p = np.maximum(p, 1e-15)
    p /= p.sum()

    b = fft_convolution(p)
    G, w = smooth_max_and_weights(b, beta)
    if not np.isfinite(G):
        return float("inf"), np.zeros(max(B - 1, 0), dtype=np.float64)

    svec = compute_svec(p, w)
    svec = np.maximum(svec, 0.0)
    g = 2.0 * svec

    Gsum = g.reshape(B, L).sum(axis=1)
    Wavg = float(np.dot(wb, Gsum))
    gradz = (wb / float(L)) * (Gsum - Wavg)

    grad = n * gradz[:-1] if B > 1 else np.zeros(0, dtype=np.float64)
    grad = np.where(np.isfinite(grad), grad, 0.0)
    return float(n * G), grad.astype(np.float64)


def block_to_p(n, B, c):
    """Convert block log-weights to full sequence."""
    L = n // B
    z = np.empty(B, dtype=np.float64)
    if B > 1:
        c = np.asarray(c, dtype=np.float64).reshape(-1)
        tmp = np.zeros(B - 1, dtype=np.float64)
        m = min(tmp.size, c.size)
        tmp[:m] = c[:m]
        c = tmp
        z[:-1] = c
    z[-1] = 0.0
    z = z - np.max(z)
    wb = np.exp(z)
    s = float(np.sum(wb))
    if s <= 0.0 or not np.isfinite(s):
        wb = np.ones(B, dtype=np.float64) / float(B)
    else:
        wb /= s
    H = wb / float(L)
    p = np.repeat(H, L)
    p = np.maximum(p, 1e-15)
    p /= p.sum()
    return p


def run_block_opt(n, B, deadline, rng):
    """Run L-BFGS on block parameters from multiple inits."""
    if not HAVE_SCIPY or B < 2 or n < B * 2:
        return None, float("inf")

    m = B - 1
    bounds = [(-5.0, 5.0)] * m
    best_c = np.zeros(m, dtype=np.float64)
    best_sc = float("inf")

    # Initializations
    inits = [np.zeros(m, dtype=np.float64)]
    x = np.linspace(0.0, 1.0, m, dtype=np.float64)
    for a in (-2.0, 2.0, -4.0, 4.0):
        inits.append(a * (x - 0.5))
    t = np.linspace(0.0, 2.0 * np.pi * 3, m, dtype=np.float64)
    inits.append(0.5 * np.cos(t))
    inits.append(0.5 * np.sin(t))
    inits.append(1.0 * rng.standard_normal(m))

    options = {"maxiter": 200, "ftol": 1e-10, "gtol": 1e-6, "maxls": 30}

    for c0 in inits:
        if time.monotonic() >= deadline:
            break
        c0 = np.clip(np.asarray(c0, dtype=np.float64), -5.0, 5.0)
        try:
            res = minimize(block_objective, c0, args=(n, B, 100000),
                           method="L-BFGS-B", jac=True, bounds=bounds, options=options)
            c = np.clip(np.asarray(res.x, dtype=np.float64), -5.0, 5.0)
        except Exception:
            c = c0
        p = block_to_p(n, B, c)
        sc = fast_score(p)
        if np.isfinite(sc) and sc < best_sc:
            best_sc = sc
            best_c = c.copy()

    p = block_to_p(n, B, best_c)
    return p, best_sc


def official_score(seq):
    """Call evaluate_sequence safely."""
    try:
        val = float(evaluate_sequence(seq))
        if np.isfinite(val):
            return val
    except Exception:
        pass
    return float("inf")


def propose_candidate(seed=42, budget_s=2400, **kwargs):
    """
    Search for minimum of 2*n*max(a*a)/(sum(a))^2 over non-negative sequences.
    Uses FISTA with smooth-max at n=16384, multiple restarts, and beta annealing.
    """
    try:
        seed = int(seed)
    except Exception:
        seed = 42
    try:
        budget_s = float(budget_s)
    except Exception:
        budget_s = 2400.0

    if budget_s <= 0:
        return [1.0]

    start_time = time.monotonic()
    reserve = 60.0
    deadline = start_time + max(10.0, budget_s - reserve)

    rng = np.random.default_rng(seed)
    N = 16384

    print(f"Starting optimization, N={N}, budget={budget_s:.0f}s")

    # Try to use height_sequence_1 if available
    initial_starts = []
    try:
        hs = height_sequence_1  # noqa: F821
        if hs is not None and len(hs) > 10:
            up = upsample(hs, N)
            initial_starts.append(up)
            print(f"  Loaded height_sequence_1 (len={len(hs)}), upsampled to N={N}")
    except Exception:
        pass

    # Generate structured starts
    structured = gen_structured_starts(N, rng, count=12)
    initial_starts.extend(structured)

    print(f"  Total initial starts: {len(initial_starts)}")

    # Score all starts
    scored_starts = []
    for p in initial_starts:
        sc = fast_score(p)
        if np.isfinite(sc):
            scored_starts.append((sc, p))

    scored_starts.sort(key=lambda x: x[0])
    print(f"  Best initial score: {scored_starts[0][0]:.8f}")

    best_p = scored_starts[0][1].copy()
    best_score = scored_starts[0][0]

    # Phase 1: Quick FISTA with medium beta (exploration)
    # Allocate 40% of time to exploration
    phase1_deadline = start_time + 0.35 * (budget_s - reserve)
    n_top = min(4, len(scored_starts))

    for i in range(n_top):
        if time.monotonic() >= phase1_deadline:
            break
        p0 = scored_starts[i][1]
        alloc = (phase1_deadline - time.monotonic()) / (n_top - i)
        sub_deadline = time.monotonic() + max(5.0, alloc)
        p_fit, sc_fit = fista_run(p0, sub_deadline, rng, beta=20000.0,
                                  max_iters=200000, step_init=0.5,
                                  restart_interval=20000, restart_noise=0.05)
        if np.isfinite(sc_fit) and sc_fit < best_score:
            best_score = sc_fit
            best_p = p_fit.copy()
            print(f"  Phase1 start {i}: score={best_score:.8f}")

    # Phase 2: FISTA with annealed beta (exploitation)
    # Allocate 55% of time to exploitation
    phase2_deadline = start_time + 0.90 * (budget_s - reserve)
    remaining2 = phase2_deadline - time.monotonic()
    if remaining2 > 30.0:
        print(f"  Phase 2: annealed FISTA, time remaining={remaining2:.0f}s")
        # Use the best solution as starting point
        p0 = best_p.copy()
        p_fit, sc_fit = fista_anneal(p0, phase2_deadline, rng,
                                     beta_start=5000.0, beta_end=500000.0,
                                     total_iters=500000, step_init=0.5)
        if np.isfinite(sc_fit) and sc_fit < best_score:
            best_score = sc_fit
            best_p = p_fit.copy()
            print(f"  Phase2 anneal: score={best_score:.8f}")

    # Phase 3: Final sharp FISTA polish
    remaining3 = deadline - time.monotonic()
    if remaining3 > 20.0:
        print(f"  Phase 3: sharp polish, time remaining={remaining3:.0f}s")
        p0 = best_p.copy()
        p_fit, sc_fit = fista_run(p0, deadline, rng, beta=500000.0,
                                  max_iters=500000, step_init=0.3,
                                  restart_interval=50000, restart_noise=0.01)
        if np.isfinite(sc_fit) and sc_fit < best_score:
            best_score = sc_fit
            best_p = p_fit.copy()
            print(f"  Phase3 polish: score={best_score:.8f}")

    # Final check with a few more restarts if time allows
    if time.monotonic() < deadline - 30.0:
        for trial in range(3):
            remaining = deadline - time.monotonic() - 10.0
            if remaining < 15.0:
                break
            # Perturb best solution
            p0 = normalize_simplex(best_p + 0.02 * rng.standard_normal(N))
            p_fit, sc_fit = fista_run(p0, time.monotonic() + remaining, rng,
                                      beta=500000.0, max_iters=200000,
                                      step_init=0.5, restart_interval=30000,
                                      restart_noise=0.03)
            if np.isfinite(sc_fit) and sc_fit < best_score:
                best_score = sc_fit
                best_p = p_fit.copy()
                print(f"  Final trial {trial}: score={best_score:.8f}")

    print(f"  Best internal score: {best_score:.8f}")

    # Final cleanup
    final_p = normalize_simplex(best_p)
    final_p = np.maximum(final_p, 0.0)
    final_p = np.minimum(final_p, 1000.0)
    final_seq = [float(x) for x in final_p]

    # Official evaluation
    off = official_score(final_seq)
    print(f"  Official score: {off:.8f}")

    if not np.isfinite(off) or off > 2.0:
        # Fallback to uniform
        return [1.0] * N

    return final_seq
