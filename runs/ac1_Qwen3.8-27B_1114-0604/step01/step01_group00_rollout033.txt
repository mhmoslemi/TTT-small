def autcorr_fft(f):
    """Full linear convolution f*f via FFT."""
    n = len(f)
    size = 1
    while size < 2*n - 1:
        size *= 2
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2*n-1]

def soft_gradient(c, f, T, top_k_frac=0.1):
    """Compute soft-max gradient using top-k fraction of lags."""
    n = len(f)
    m = len(c)

    k = max(1, int(m * top_k_frac))
    # Find the k-th largest value
    if k < m:
        # Use partition for efficiency
        threshold = np.partition(c, m-k)[m-k]
    else:
        threshold = c.min()
    
    # Soft-max on all lags (but with higher effective weight on top-k)
    # Actually, let's just do full soft-max with a good temperature
    cs = c / T
    cs -= cs.max()
    p = np.exp(cs)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros(m)
        p[n-1] = 1.0
    else:
        p /= s
    grad = 2.0 * correlate(p, f, mode='valid')
    return grad, p

def soft_gradient_topk(c, f, T, top_k_frac=0.1):
    """Soft-max gradient but only over top-k fraction of lags."""
    n = len(f)
    m = len(c)
    k = max(1, int(m * top_k_frac))
    
    # Get indices of top-k lags
    top_idx = np.argpartition(c, -k)[-k:]

    p_full = np.zeros(m)
    c_top = c[top_idx] / T
    c_top -= c_top.max()
    p_top = np.exp(c_top)
    p_top /= p_top.sum()
    p_full[top_idx] = p_top
    
    grad = 2.0 * correlate(p_full, f, mode='valid')
    return grad, p_full

def mirror_step(f, grad, step):
    """One mirror descent step on the simplex."""
    span = grad.max() - grad.min()
    if span < 1e-15:
        return f

    log_f = np.log(f + 1e-300) - eta * grad

    f_new = np.exp(log_f)
    s = f_new.sum()
    if s < 1e-300:
        return f
    f_new /= s
    return f_new

"""
Step function optimizer: minimize 2*n*max(conv(f,f))/(sum f)^2.
Uses adaptive mirror descent with top-k soft-max gradient, Nesterov momentum,
greedy local search with efficient delta computation, and multi-phase diversification.

import numpy as np
from scipy.signal import correlate
import time


def _autocorr_fft(f):
    """Full linear convolution f*f via FFT."""
    n = len(f)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    F = np.fft.rfft(f, size)
    return np.fft.irfft(F * F, size)[:2 * n - 1]


def _soft_grad_topk(c, f, T, top_k_frac=0.05):
    """Soft-max gradient restricted to top-k fraction of convolution lags."""
    n = len(f)
    m = len(c)
    k = max(1, int(m * top_k_frac))
    top_idx = np.argpartition(c, m - k)[m - k:]
    p = np.zeros(m)
    c_top = c[top_idx] / T
    c_top -= c_top.max()
    p_top = np.exp(c_top)
    s = p_top.sum()
    if s < 1e-300:
        p[top_idx[0]] = 1.0
    else:
        p[top_idx] = p_top / s
    grad = 2.0 * correlate(p, f, mode='valid')
    return grad


def _soft_grad_full(c, f, T):
    """Full soft-max gradient over all lags."""
    n = len(f)
    m = len(c)
    cs = c / T
    cs -= cs.max()
    p = np.exp(cs)
    s = p.sum()
    if s < 1e-300:
        p = np.zeros(m)
        p[n - 1] = 1.0
    else:
        p /= s
    grad = 2.0 * correlate(p, f, mode='valid')
    return grad


def _mirror_step(f, grad, step, f_prev=None, momentum=0.0):
    """Mirror descent step with optional Nesterov momentum."""
    if f_prev is not None and momentum > 0:
        y = f + momentum * (f - f_prev)
        y = np.maximum(y, 1e-15)
    else:
        y = f
    span = grad.max() - grad.min()
    if span < 1e-15:
        return f
    eta = step / (span + 1e-300)
    log_y = np.log(y + 1e-300) - eta * grad
    log_y -= log_y.max()
    f_new = np.exp(log_y)
    s = f_new.sum()
    if s < 1e-300:
        return f
    f_new /= s
    return f_new

    starts['edges'] = np.exp(-2.0 * (j_arr - center) / center) * np.exp(2.0 * (j_arr - center) / center)
    starts['bimodal'] = (np.exp(-((j_arr - 0.22 * n) ** 2) / (0.09 * n) ** 2) +
                         np.exp(-((j_arr - 0.78 * n) ** 2) / (0.09 * n) ** 2) + 0.005)
    starts['trimodal'] = (np.exp(-((j_arr - 0.1 * n) ** 2) / (0.05 * n) ** 2) +
                          np.exp(-((j_arr - 0.5 * n) ** 2) / (0.05 * n) ** 2) +
                          np.exp(-((j_arr - 0.9 * n) ** 2) / (0.05 * n) ** 2) + 0.005)
    starts['comb3'] = 0.5 + 0.5 * np.cos(6.0 * np.pi * j_arr / n) + 0.05
    starts['comb5'] = 0.5 + 0.5 * np.cos(10.0 * np.pi * j_arr / n) + 0.05
    starts['rand_exp'] = rng.exponential(1.0, size=n)
    starts['rand_dir'] = rng.dirichlet(np.ones(n) * 0.05)
    starts['sine_mix'] = 1.0 + 0.4 * np.sin(2 * np.pi * 7 * j_arr / n) + 0.3 * np.sin(2 * np.pi * 13 * j_arr / n) + 0.1

    # Evaluate and sort
    start_list = []
    for name, f0 in starts.items():
        fn = _normalize(f0)
        sc = _eval(fn)
        start_list.append((sc, name, fn))
    start_list.sort(key=lambda x: x[0])
    print("Start scores:")
    for sc, name, _ in start_list:
        print(f"  {name}: {sc:.6f}")

    if start_list[0][0] < best_score:
        best_score = start_list[0][0]
        best_f = start_list[0][2].copy()

    # Phase 1: Mirror descent from top 5 starts
    print(f"\n--- Phase 1: Mirror Descent ---")
    phase1_end = t0 + 900.0
    n_starts = min(5, len(start_list))
    time_per = (phase1_end - time.monotonic()) / n_starts
    for idx in range(n_starts):
        if time.monotonic() > phase1_end - 30:
            break
        sc, name, f0 = start_list[idx]
        dl = time.monotonic() + time_per
        f_opt, sc_opt = _optimize_adaptive(f0, n, dl, rng, label=name)
        if sc_opt < best_score:
            best_score = sc_opt
            best_f = f_opt.copy()
            print(f"  {name} -> {best_score:.6f}")

    print(f"Phase 1 best: {best_score:.6f}")

    # Phase 2: Local search
    if time.monotonic() < deadline - 400:
        print(f"\n--- Phase 2: Local Search ---")
        dl2 = min(time.monotonic() + 500.0, deadline - 400.0)
        f_ls, sc_ls = _local_search(best_f, n, dl2, rng)
        if sc_ls < best_score:
            best_score = sc_ls
            best_f = f_ls.copy()
            print(f"  Local search -> {best_score:.6f}")

    # Phase 3: Mirror descent refinement
    if time.monotonic() < deadline - 200:
        print(f"\n--- Phase 3: Mirror Refinement ---")
        dl3 = min(time.monotonic() + 350.0, deadline - 200.0)
        f_ref, sc_ref = _optimize_adaptive(best_f, n, dl3, rng, label="refine")
        if sc_ref < best_score:
            best_score = sc_ref
            best_f = f_ref.copy()
            print(f"  Refined -> {best_score:.6f}")

    # Phase 4: Pair-swap search
    if time.monotonic() < deadline - 80:
        print(f"\n--- Phase 4: Pair Swap ---")
        dl4 = min(time.monotonic() + 200.0, deadline - 80.0)
        f_ps, sc_ps = _pair_swap_search(best_f, n, dl4, rng)
        if sc_ps < best_score:
            best_score = sc_ps
            best_f = f_ps.copy()
            print(f"  Pair swap -> {best_score:.6f}")

    # Phase 5: Final local search
    if time.monotonic() < deadline - 20:
        print(f"\n--- Phase 5: Final Polish ---")
        dl5 = deadline - 10.0
        f_fin, sc_fin = _local_search(best_f, n, dl5, rng)
        if sc_fin < best_score:
            best_score = sc_fin
            best_f = f_fin.copy()
            print(f"  Final -> {best_score:.6f}")

    # Ensure valid output
    best_f = np.clip(best_f, 0.0, 1000.0)
    final_score = _eval(best_f)
    if final_score < best_score or not np.isfinite(best_score):
        best_score = final_score

    elapsed = time.monotonic() - t0
    print(f"\n=== FINAL: {best_score:.6f} (elapsed: {elapsed:.0f}s) ===")
    return [float(x) for x in best_f]
