import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    
    # Initialize learning rate and best solution
    eta = 0.1
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute overlap integrals for each shift
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_shift = max_idx - (len(h) - 1)
        
        # Compute gradient w.r.t. the maximizing shift
        if max_shift >= 0:
            grad = np.concatenate([np.zeros(max_shift), 1 - h[:-max_shift]])
        else:
            grad = np.concatenate([1 - h[-max_shift:], np.zeros(-max_shift)])
        
        # Update h using gradient descent
        h = h - eta * grad * dx
        
        # Project h back to feasible region
        h = project_h(h, target_sum)
        
        # Evaluate current solution
        current_c5 = compute_c5(h, dx)
        
        # Update best solution if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            eta = min(eta * 1.05, 1.0)  # Increase learning rate if improved
            
        # Reduce learning rate if no improvement
        if iteration % 10 == 0 and iteration > 0:
            eta *= 0.95
            
        # Add noise occasionally to escape local minima
        if iteration % 100 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = project_h(h, target_sum)
            
        iteration += 1
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
