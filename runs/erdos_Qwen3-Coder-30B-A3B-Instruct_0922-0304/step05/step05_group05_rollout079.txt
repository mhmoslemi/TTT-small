import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        # Correlation to compute overlaps
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Initial solution
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Simulated Annealing parameters
    T0 = max(0.01, current_c5 * 0.1)  # Initial temperature
    T = T0
    cooling_rate = 0.999  # Geometric cooling
    min_T = 1e-6
    max_iter = 10000
    iter_count = 0
    
    # Temperature schedule
    def update_temperature(T, cooling_rate, min_T):
        T = T * cooling_rate
        return max(T, min_T)

    # Optimization loop
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Generate neighbor solution
        neighbor_h = current_h + np.random.normal(0, 0.01, n_points)
        neighbor_h = np.clip(neighbor_h, 0, 1)
        # Rescale to satisfy constraint
        neighbor_h = neighbor_h * (n_points / 2) / np.sum(neighbor_h)
        
        # Objective value
        neighbor_c5 = compute_c5(neighbor_h)
        
        # Metropolis acceptance criterion
        if neighbor_c5 < current_c5:
            current_h = neighbor_h.copy()
            current_c5 = neighbor_c5
        else:
            # Accept worse solution with probability
            delta = neighbor_c5 - current_c5
            if np.random.rand() < np.exp(-delta / T):
                current_h = neighbor_h.copy()
                current_c5 = neighbor_c5
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Update temperature
        T = update_temperature(T, cooling_rate, min_T)
        
        # Early exit if very close to target
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1

    # Final check of solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
