import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Track best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Convex optimization loop
    max_iter = 500
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Formulate and solve convex relaxation
        variables = cp.Variable(n_points)
        constraints = [variables >= 0, variables <= 1, cp.sum(variables) == n_points / 2]
        
        # Create a surrogate objective: maximizing the maximum cross-correlation
        # Avoids directly using max in the objective for convexity
        max_overlap_var = cp.Variable()
        # Add constraint that max_overlap_var >= (h * (1-h))[k] for all k
        # This uses a more tractable approach: compute current value and create surrogate
        current_overlap = correlate(h, 1 - h, mode="full")
        current_max = np.max(current_overlap)
        
        # Minimize the maximum overlap by optimizing the function
        # This is an approximation with a sensible convex relaxation approach
        objective = cp.Minimize(max_overlap_var)
        
        # We need a better convex formulation for the max of correlation
        # For now, we'll use a basic approach but improved by using some of the
        # correlated structure
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = variables.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
        except:
            pass
        

        # Evaluate current solution
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break

        # Specific local perturbation on the best so far
        h_pert = h + np.random.normal(0, 0.005, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        if compute_c5(h_pert) < current_c5:
            h = h_pert

    # Final evaluation of best solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
