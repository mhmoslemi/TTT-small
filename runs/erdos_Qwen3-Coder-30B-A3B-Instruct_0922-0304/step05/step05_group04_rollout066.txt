import numpy as np
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    generations = 1000
    mutation_rate = 0.1
    tournament_size = 10
    elite_size = population_size // 2
    
    # Initialize population
    def initialize_population():
        population = []
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h.copy())
        else:
            h = np.random.random(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h.copy())
        
        # Fill the rest of the population
        for _ in range(population_size - 1):
            h = np.random.random(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h.copy())
        return population
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def tournament_select(population, fitnesses):
        tournament_indices = random.sample(range(len(population)), tournament_size)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
        return population[winner_idx]
    
    def crossover(parent1, parent2):
        child = np.zeros(n_points)
        for i in range(n_points):
            if random.random() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        # Normalize to maintain constraint
        child = child * (n_points / 2) / np.sum(child)
        return child
    
    def mutate(individual):
        mutated = individual.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                if random.random() < 0.5:
                    mutated[i] += random.uniform(-0.1, 0.1)
                else:
                    mutated[i] = random.random()
        # Clamp values and renormalize
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return mutated
    
    # Initialize
    population = initialize_population()
    
    best_h = None
    best_c5 = float('inf')
    
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitnesses = [compute_c5(h) for h in population]
        
        # Track best solution
        for h, c5 in zip(population, fitnesses):
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        if best_c5 <= 0.3807:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Keep elite
        new_population = population[:elite_size]
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_select(population, fitnesses)
            parent2 = tournament_select(population, fitnesses)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
    
    if best_h is None:
        # Fallback to final population
        final_fitnesses = [compute_c5(h) for h in population]
        best_idx = np.argmin(final_fitnesses)
        best_h = population[best_idx]
        best_c5 = final_fitnesses[best_idx]
    
    return (best_h.tolist(), best_c5, n_points)
