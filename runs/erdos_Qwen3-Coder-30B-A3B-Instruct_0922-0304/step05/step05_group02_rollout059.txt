import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, target_sum)
    else:
        # Symmetric concentration
        h = np.ones(n_points) * 0.5
        for i in range(n_points):
            h[i] += 0.1 * ((-1) ** i)
        h = project_h(h, target_sum)
    
    # Gradient-based optimization with momentum
    momentum = 0.9
    step_size = 0.01
    max_iter = 50000
    patience = 50
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    prev_c5 = best_c5
    no_improvement_count = 0
    
    # Initialize velocity
    v = np.zeros(n_points)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        overlap = correlate(h, 1 - h, mode="full") * dx
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max]
        
        # Compute gradient (handling shift k_max properly)
        # At k_max, we need gradient w.r.t. h for count of terms in the overlap
        # which is the gradient of (h[i] * (1 - h[i+k_max])) terms
        # For a minimal gradient model, we'll compute approximate component-wise grad
        grad = np.zeros_like(h)
        if k_max < n_points:
            # This is the (len(h) - k_max) terms starting from left
            for j in range(len(h) - k_max):
                grad[j] += 1 - h[j + k_max]
        else:
            # Cases where shift is negative
            shift = k_max - (2*n_points - 1)
            for j in range(n_points + k_max - (2*n_points - 1), n_points):
                grad[j - k_max + (2*n_points - 1)] += 1 - h[j] 
        
        # For a more accurate approach, let's compute directional derivatives
        # by using finite differences in direction of gradient
        grad = 1 - np.roll(h, k_max, axis=0) if k_max < n_points else 1 - np.roll(h, k_max - (2*n_points - 1), axis=0)
        grad = np.clip(grad, 0, 1)
        
        # Update with momentum
        v = momentum * v - step_size * grad
        h_new = h + v
        
        # Project to feasible set
        h_new = project_h(h_new, target_sum)
        c5_new = compute_c5(h_new, dx)
        
        # Adaptive step size
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            # Backtrack if no improvement
            step_size *= 0.9
            
        if no_improvement_count >= patience:
            # Increase step size if not improving much
            step_size = min(0.1, step_size * 1.1)
            no_improvement_count = 0
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
