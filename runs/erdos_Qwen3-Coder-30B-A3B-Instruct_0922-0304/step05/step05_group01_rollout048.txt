import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 20
    max_iter = 1000
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    for _ in range(swarm_size):
        h = h_init + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track best solution
    global_best_h = h_init.copy()
    global_best_c5 = float('inf')
    
    # PSO parameters
    w = 0.7
    c1 = c2 = 1.5
    max_vel = 0.2
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate swarm
        for i, h in enumerate(swarm):
            c5 = compute_c5(h)
            if c5 < global_best_c5:
                global_best_c5 = c5
                global_best_h = h.copy()
                
            # Early termination
            if global_best_c5 <= 0.3807:
                break
                
        # Update particles
        for i in range(swarm_size):
            h = swarm[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            pbest = h.copy()
            if i == 0 or compute_c5(swarm[i-1]) < compute_c5(pbest):
                pbest = swarm[i-1].copy()
                
            v = w * v + c1 * r1 * (pbest - h) + c2 * r2 * (global_best_h - h)
            
            # Clip velocity
            v = np.clip(v, -max_vel, max_vel)
            velocities[i] = v
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            swarm[i] = h_new
            
        # Update inertia weight
        w = max(0.4, w - 0.01)
    
    # Final evaluation
    final_c5 = compute_c5(global_best_h)
    
    return (global_best_h.tolist(), final_c5, n_points)
