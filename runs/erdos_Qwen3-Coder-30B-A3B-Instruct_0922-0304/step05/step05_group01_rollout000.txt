import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx=2.0/70):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def mutate(h, mutation_std, n_points):
    mutated = h + np.random.normal(0, mutation_std, h.shape)
    mutated = np.clip(mutated, 0, 1)
    sum_h = np.sum(mutated)
    mutated = mutated * (n_points / 2) / sum_h
    return mutated

def evaluate_fitness(population):
    for i in range(len(population)):
        population[i] = (population[i], compute_c5(population[i]))

def select_elite(population, elite_fraction):
    population.sort(key=lambda x: x[1])
    return [ind[0] for ind in population[:int(len(population)*elite_fraction)]]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize population
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h
        population = [h.copy()]
        for _ in range(19):
            population.append(mutate(h, 0.01, n_points))
    else:
        # Fallback initialization
        population = []
        for _ in range(20):
            h = np.ones(n_points)
            h[n_points//2:] = 0
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            sum_h = np.sum(h)
            h = h * (n_points / 2) / sum_h
            population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    
    # EA parameters
    mutation_strength = 0.01
    elite_fraction = 0.25
    no_improvement_count = 0
    max_no_improvement = 50
    
    for generation in range(1000):  # Max generations
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        evaluate_fitness([(h, 0) for h in population])
        
        # Find best in generation
        current_best = min(population, key=lambda h: compute_c5(h))
        current_c5 = compute_c5(current_best)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Adaptive mutation strength
        if no_improvement_count > max_no_improvement:
            mutation_strength *= 0.1
            no_improvement_count = 0
            
        # Selection
        elite = select_elite([(h, compute_c5(h)) for h in population], elite_fraction)
        
        # Generate new population
        new_population = []
        for individual in elite:
            new_population.append(mutate(individual, mutation_strength, n_points))
            new_population.append(mutate(individual, mutation_strength, n_points))
            
        # Keep elite and new individuals
        population = new_population[:20]  # Maintain population size

    return (best_h.tolist(), best_c5, n_points)
