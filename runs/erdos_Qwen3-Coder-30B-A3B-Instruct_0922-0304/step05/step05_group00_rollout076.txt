import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum to target_sum."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialization from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Construct a good starting point
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 10000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        # Identify most critical lags (top 5% of contribution to max overlap)
        n_lags = len(overlap)
        critical_indices = np.argsort(overlap)[-int(n_lags * 0.05):]
        
        # Define the local optimization subproblem based on critical lags
        def local_objective(h_local):
            # Recompute overlap with the new h_local
            overlap_local = correlate(h_local, 1 - h_local, mode="full") * dx
            return np.max(overlap_local)
        
        # Define constraints
        def sum_constraint(h_local):
            return np.sum(h_local) - target_sum
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': sum_constraint}
        
        # Optimize using SLSQP with current h as initial guess
        result = minimize(
            local_objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        
        if result.success:
            h_new = result.x
            h_new = project_to_feasible(h_new, target_sum)
            c5_new = compute_c5(h_new, dx)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                
            h = h_new
        else:
            # If optimization fails, perturb and continue
            perturbation = np.random.normal(0, 0.01, n_points)
            h = h + perturbation
            h = project_to_feasible(h, target_sum)
        
        # Early exit if target is reached
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Final check and return
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
