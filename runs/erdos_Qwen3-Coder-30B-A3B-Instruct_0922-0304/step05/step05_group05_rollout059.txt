import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # PSO parameters
    n_particles = 20
    max_iter = 1000
    w_start = 0.8
    w_end = 0.1
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize particles
    particles = [h.copy()]
    for _ in range(n_particles - 1):
        h_pert = h + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        particles.append(h_pert)
    
    # Velocities
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Personal bests
    p_best = [particle.copy() for particle in particles]
    p_best_fitness = [float('inf')] * n_particles
    
    # Global best
    g_best = h.copy()
    g_best_fitness = float('inf')
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Main PSO loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(n_particles):
            # Compute fitness
            fitness = compute_c5(particles[i])
            
            # Update personal best
            if fitness < p_best_fitness[i]:
                p_best_fitness[i] = fitness
                p_best[i] = particles[i].copy()
                
            # Update global best
            if fitness < g_best_fitness:
                g_best_fitness = fitness
                g_best = particles[i].copy()
                
            # Update velocities and positions
            r1, r2 = np.random.rand(2, n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update positions
            particles[i] += velocities[i]
            
            # Enforce constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
        # Early exit
        if g_best_fitness <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(g_best)
    
    return (g_best.tolist(), final_c5, n_points)
