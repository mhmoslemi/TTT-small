import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        # Create a smart initial guess
        h = np.zeros(n_points)
        # Place a peak in the middle
        center = n_points // 2
        h[center-5:center+5] = 1.0
        h = project_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 1000
    patience = 100
    patience_counter = 0
    prev_best = best_c5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the C5 value and identify worst shift
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_shift = max_idx - (len(h) - 1)
        
        # Compute gradient for the active shift
        # shift k corresponds to index k + (n-1) in correlation result
        # For shift k, gradient is (1 - h_shifted)
        gradient = np.zeros_like(h)
        if max_shift >= 0:
            gradient[max_shift:] += (1 - h[:-max_shift]) if max_shift < len(h) else 0
        else:
            gradient[:max_shift] += (1 - h[-max_shift:]) if -max_shift < len(h) else 0
        
        # Update h using gradient descent
        h = h - learning_rate * gradient
        
        # Project to feasible region
        h = project_h(h, target_sum)
        
        # Evaluate and update best
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Early stopping
        if patience_counter >= patience:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
