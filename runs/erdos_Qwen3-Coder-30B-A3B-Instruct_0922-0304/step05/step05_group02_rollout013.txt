import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_feasible(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def compute_gradient(h, dx, k_max):
    """Compute gradient with respect to the worst shift."""
    n = len(h)
    grad = np.zeros(n)
    
    # For the worst shift, compute gradient of overlap
    # Overlap at shift k is sum h[i] * (1 - h[i+k])
    # Gradient = 1 - 2*h[i+k] for h[i] 
    # But we need to be more careful with indexing
    
    # The gradient of (h * (1-h))[k] w.r.t. h[i] 
    # equals (1-h[i+k]) if i is in valid range, and 0 elsewhere
    
    # Compute correlation of h with (1-h) 
    # To compute gradient at shift k:
    # ∂/∂h[i] (h * (1-h))[k] = (1-h[i+k]) if 0 <= i+k < n and 0 <= i < n
    # This corresponds to (1-h) shifted by k, then we need to handle indices
    
    # More direct approach:
    # d(h * (1-h))[k]/dh[i] = (1-h[i+k]) if i+k < n and i >= 0 
    # = (1-h[i-k]) if i-k >= 0 and i < n
        
    # Actually, this is tricky. We need to compute for active shifts.
    # We'll analytically compute a few shifts or focus on average gradient direction
    
    # Let's compute for the k_max shift:
    if k_max >= 0:
        for i in range(n):
            j = i + k_max
            if 0 <= j < n:
                grad[i] = 1 - h[j]
    else:
        # k_max is negative, so we compute for shift -k_max
        k_pos = -k_max
        for i in range(n):
            j = i - k_pos
            if 0 <= j < n:
                grad[i] = 1 - h[j]
    
    return grad

def line_search(h, grad, alpha_min, alpha_max, dx, c5_current, max_backtrack=10):
    """Perform line search for step size."""
    # Simple backtracking
    alpha = alpha_max
    for _ in range(max_backtrack):
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, len(h) / 2)
        c5_new = compute_c5(h_new, dx)
        if c5_new <= c5_current:
            return alpha, h_new
        alpha *= 0.5
    return alpha, h_new

    # Initialize optimization variables
    h = init_h.copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Optimization parameters
    momentum_decay = 0.9
    step_size = 0.1

    momentum = np.zeros(n_points)

        # Compute overlap for all shifts

        # Get maximum overlap and its shift in range [0, 2n_points-2]
