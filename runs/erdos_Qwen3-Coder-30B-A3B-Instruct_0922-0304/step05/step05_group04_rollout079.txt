        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

        overlap = correlate(h, 1 - h, mode="full")

        # Store the best solution
        if max_overlap < best_c5:

            h_plus[i] += eps_val
            h_minus[i] -= eps_val

            h_minus = np.clip(h_minus, 0, 1)

            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
