import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def evaluate_feasibility(h, max_overlap):
        # This function should return True if the given h satisfies the constraint
        # that all overlaps are <= max_overlap
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx <= max_overlap

    # Binary search approach
    left = 0.0
    right = 0.38085  # Current upper bound
    tolerance = 1e-4
    best_c5 = right
    best_h = h.copy()

    while time.time() - start_time < budget_s - 1 and right - left > tolerance:
        mid = (left + right) / 2
        
        # Try to find a feasible solution with this C₅
        def obj_func(h):
            return -compute_c5(h)  # minimize negative of max overlap
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        # We want the actual maximum overlap to be <= mid
        # Let's perform a constrained optimization that focuses on minimizing
        # the maximum overlap, subject to sum constraint
        
        result = minimize(
            obj_func,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )

        if result.success:
            h_test = result.x
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            if evaluate_feasibility(h_test, mid):
                right = mid
                best_c5 = mid
                best_h = h_test.copy()
            else:
                left = mid
        else:
            # If optimization fails, increase left
            left = mid
        
        # Sometimes we should also try perturbed solutions
        if time.time() - start_time > budget_s - 10:
            break
            
        # Random perturbation to escape local minima
        if np.random.random() < 0.1:
            h_perturbed = h + np.random.normal(0, 0.02, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            if evaluate_feasibility(h_perturbed, mid):
                right = mid
                best_c5 = mid
                best_h = h_perturbed.copy()

    # Final refinement with a strongly constrained optimization
    if time.time() - start_time < budget_s - 1:
        # Refine around best found solution
        def obj_func_refine(h):
            return compute_c5(h)
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        result = minimize(
            obj_func_refine,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-9}
        )

        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_c5 = compute_c5(final_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h.copy()

    return (best_h.tolist(), best_c5, n_points)
