import numpy as np
from scipy.signal import correlate
import time

def project_h(h, target_sum):
    """Project h to satisfy sum constraint while keeping in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def line_search(h, grad, dx, max_alpha=1.0, tolerance=1e-4):
    """Perform backtracking line search to find suitable step size."""
    alpha = max_alpha
    current_c5 = compute_c5(h, dx)
    
    while alpha > 1e-8:
        h_new = project_h(h - alpha * grad, len(h) / 2)
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < current_c5 - tolerance * alpha * np.dot(grad, grad):
            return h_new, alpha
        alpha *= 0.5
    
    return h, alpha

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    h = project_h(h, target_sum)
    
    # Optimization parameters
    momentum = 0.95
    prev_grad = np.zeros(n_points)
    prev_h = h.copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    no_improvement_count = 0
    max_no_improvement = 50
    
    # Precompute correlation structure if needed
    while time.time() - start_time < budget_s - 1:
        # Compute all correlations
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_k = max_idx - (n_points - 1)
        
        # Compute gradient for the worst case shift
        grad = np.zeros(n_points)
        if max_k >= 0:
            grad[max_k:] = 1 - h[:-max_k]
        else:
            grad[:max_k] = 1 - h[-max_k:]
        
        # Update with momentum
        grad_momentum = momentum * prev_grad + (1 - momentum) * grad
        prev_grad = grad
        
        # Project and line search
        h_new, alpha = line_search(prev_h, grad_momentum, dx)
        
        # Update h
        h = h_new
        prev_h = h.copy()
        
        # Evaluation
        new_c5 = compute_c5(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count >= max_no_improvement:
            # Perturb solution to escape local minimum
            h = project_h(h + np.random.normal(0, 0.001, n_points), target_sum)
            no_improvement_count = 0
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
