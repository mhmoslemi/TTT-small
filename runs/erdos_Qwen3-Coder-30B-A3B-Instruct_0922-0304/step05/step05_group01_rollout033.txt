import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def validate_and_repair(h, n_points):
    """Validate and repair initial solution"""
    h = np.array(h)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def initialize_coarse(n_coarse=35):
    """Initialize coarse resolution solution"""
    h = np.ones(n_coarse)
    h[n_coarse//2:] = 0
    h += np.random.normal(0, 0.05, n_coarse)
    h = np.clip(h, 0, 1)
    h = h * (n_coarse / 2) / np.sum(h)
    return h

def upscale(h_coarse, n_points):
    """Upscale coarse solution to fine resolution"""
    Interp = np.repeat(h_coarse, 2)
    return Interp

def compute_c5(h, dx=2.0 / len(h)):
    """Compute the C5 value"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def constraint_sum(h, target_sum):
    """Constraint function for sum(h) = target_sum"""
    return np.sum(h) - target_sum

def optimize(h, n_points, max_iter=200, tol=1e-8):
    """Optimize with SLSQP"""
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: constraint_sum(x, target_sum)}]
    
    def objective(h):
        return compute_c5(h, dx)
        
    result = minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': max_iter, 'ftol': tol}
    )
    
    if result.success:
        h_opt = result.x
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        return h_opt
    else:
        # If optimization fails, repair
        h_opt = np.clip(h, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        return h_opt
