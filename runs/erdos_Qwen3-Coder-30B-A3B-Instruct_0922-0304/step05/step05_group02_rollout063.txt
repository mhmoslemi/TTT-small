import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction or create a simple one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Start with a simple piecewise constant function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)

    # Optimization parameters
    momentum_decay = 0.9
    learning_rate = 0.1
    line_search_tolerance = 1e-3
    max_line_search_steps = 20
    
    # Initialize momentum
    m = np.zeros_like(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute current overlap
        c5 = compute_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient by approximating derivative
        # Find worst shift
        overlap = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(overlap)
        
        # Create shifted version to estimate gradient
        h_shifted = np.roll(h, max_shift_idx - len(h) + 1)
        # Compute direction in which to decrease overlap
        # We estimate gradient direction by computing response to small changes
        eps = 1e-4
        h_plus = h + eps
        h_plus = project_to_feasible(h_plus, target_sum)
        c5_plus = compute_c5(h_plus, dx)
        
        h_minus = h - eps
        h_minus = project_to_feasible(h_minus, target_sum)
        c5_minus = compute_c5(h_minus, dx)
        
        # Simple finite difference gradient for selected element
        # Since the overlap is over all shifts, we focus on the structure
        # We can do a more direct approach by computing analytical gradients 
        # or using the difference in overlaps with perturbations
        
        # For simplicity and speed, we'll take a gradient proxy using a batch of random shifts
        # Sample a few shifts to estimate the gradient
        grad_h = np.zeros_like(h)
        n_samples = min(10, len(h))
        sample_indices = np.random.choice(len(h), n_samples, replace=False)
        
        for i in sample_indices:
            # Perturb at position i
            eps = 1e-4
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_feasible(h_minus, target_sum)
            c5_minus = compute_c5(h_minus, dx)
            
            grad_h[i] = (c5_plus - c5_minus) / (2 * eps)
            
        # Apply momentum
        m = momentum_decay * m + (1 - momentum_decay) * grad_h
        
        # Line search for step size
        step_size = learning_rate
        for _ in range(max_line_search_steps):
            h_new = h - step_size * m
            h_new = project_to_feasible(h_new, target_sum)
            c5_new = compute_c5(h_new, dx)
            if c5_new <= c5 - line_search_tolerance * step_size * np.dot(m, m):
                h = h_new
                break
            else:
                step_size *= 0.5
        
        # Apply standard trust-region style update (simplified)
        if iteration % 10 == 0:
            # Periodic refinement using a local optimizer
            def objective(h_vec):
                return compute_c5(h_vec, dx)
            
            def constraint_sum(h_vec):
                return np.sum(h_vec) - target_sum
            
            # Simple local refinement if needed to stay tight
            from scipy.optimize import minimize
            
            bounds = [(0, 1) for _ in range(len(h))]
            constraints = [{'type': 'eq', 'fun': constraint_sum}]
            
            result = minimize(
                objective, 
                h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 5, 'ftol': 1e-6}
            )
            
            if result.success:
                h = result.x
                h = project_to_feasible(h, target_sum)
        
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
