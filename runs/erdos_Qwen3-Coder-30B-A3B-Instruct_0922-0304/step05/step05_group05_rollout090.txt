import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: 3-region piecewise construction
        h = np.zeros(n_points)
        mid1 = n_points // 4
        mid2 = 3 * n_points // 4
        h[:mid1] = 0.5
        h[mid1:mid2] = 0.25
        h[mid2:] = 0.5
        # Perturb slightly
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Track best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Swap-based local search
    n_swaps = 1000
    swap_size = 5
    temperature = 0.1
    
    current_h = h.copy()
    current_c5 = best_c5
    
    for _ in range(n_swaps):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Select two non-overlapping segments
        seg1_start = np.random.randint(0, n_points - swap_size)
        seg2_start = np.random.randint(0, n_points - swap_size)
        
        # Ensure segments don't overlap
        while (seg2_start < seg1_start + swap_size and seg1_start < seg2_start + swap_size):
            seg2_start = np.random.randint(0, n_points - swap_size)
            
        # Make a copy to modify
        new_h = current_h.copy()
        
        # Swap the values in these segments
        seg1_vals = new_h[seg1_start:seg1_start + swap_size]
        seg2_vals = new_h[seg2_start:seg2_start + swap_size]
        
        new_h[seg1_start:seg1_start + swap_size] = seg2_vals
        new_h[seg2_start:seg2_start + swap_size] = seg1_vals
        
        # Adjust to maintain constraint sum(h) = n_points / 2
        old_sum = np.sum(current_h)
        new_sum = np.sum(new_h)
        scale_factor = old_sum / new_sum
        new_h *= scale_factor
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Apply bounds
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Compute new C5
        new_c5 = compute_c5(new_h)
        
        # Accept with probability
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / temperature):
            current_h = new_h
            current_c5 = new_c5
            
        # Track best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break

    # Final polish with SLSQP
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    try:
        result = minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_c5 = compute_c5(final_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h
    except:
        pass  # Keep best_h if polish fails

    return (best_h.tolist(), best_c5, n_points)
