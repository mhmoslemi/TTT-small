import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    
    # Parameters
    eta = 0.1  # Initial learning rate
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute cross-correlation for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_k = max_idx - (len(h) - 1)
        
        # Gradient computation
        # For shift k, the derivative of the overlap is (1 - h_shifted) * dx
        shifted = np.roll(h, max_k)
        gradient = (1 - shifted) * dx
        
        # Gradient descent update
        h = h - eta * gradient
        
        # Project to feasible region
        h = project_h(h, target_sum)
        
        # Compute current C5
        current_c5 = compute_c5(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            # Increase learning rate if improved
            eta = min(eta * 1.01, 1.0)
        else:
            # Decrease learning rate if no improvement
            eta = max(eta * 0.9, 0.001)
        
        # Periodic noise injection to avoid local minima
        if iteration % 100 == 0 and iteration > 0:
            h += np.random.normal(0, 0.005, n_points)
            h = project_h(h, target_sum)
            
        iteration += 1
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
