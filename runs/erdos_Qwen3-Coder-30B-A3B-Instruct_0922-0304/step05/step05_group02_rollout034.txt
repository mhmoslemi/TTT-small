import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():

        h = project_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_h(h, target_sum)

    # Parameters
    momentum = 0.9
    step_size = 0.01
    max_iter = 1000
    patience = 50
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Initialize momentum
    v = np.zeros_like(h)
    
    iter_count = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute all cross-correlations
        overlap = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(overlap)
        max_overlap = overlap[max_shift_idx] * dx
        
        # Update best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > patience:
            break
            
        # Compute gradient direction (active shift)
        active_shift = max_shift_idx - (n_points - 1)
        if active_shift >= 0:
            grad = np.zeros(n_points)
            grad[active_shift:] = 1  # This is not correct: need actual gradient
        else:
            grad = np.zeros(n_points)
            grad[:active_shift] = 1
            
        # Actual gradient: correct derivative
        # Gradient of (h * (1-h)) at shift k is (1 - h_{shifted})
        # We compute: h * (1 - h) for all shifts
        # Then the gradient (d/dh) is computing also all possible shift gradients
        # Let's instead compute how the max value changes with a small perturbation
        # Or just evolve with full gradient of max
        # Compute derivative structure for the max part, not gradient directly
        # Simpler approach: handle shifts directly in gradient computation
        
        # Simple approach via finite difference 
        # Take a direction of update and move along it
        # For each shift we can derive what the gradient w.r.t. h is,
        # which is the derivative of (h ∗ (1-h) = (1-h')_shifted)
        # So ∂C5/∂h_i = -sum_{j: j+k = i} (1 - h_j)  if shift k is maximizing
        # Actually, just implementing a step in the direction that decreases C5
        # This is hard to compute with the full cross correlation.
        # A simpler idea: gradient on approximated evaluation.
        
        # Compute gradient of the maximum correlation
        # sinlge shift corresponds to conv h * (1-h)
        # For each shift k, let y_k = ∫ h(x) (1-h(x+k)) dx
        # Gradient of y_k w.r.t h(x) is: (1 - h(x+k)) 
        # So actually we want to minimize y_{k*} for shift k*
        
        # Heuristic update: gradient vector based on shift k* that gives max
        # Use one of the simpler approaches to compute updates
        
        # Gradient step, simplified method
        # Select k* that maximizes overlap
        grad_h = np.zeros_like(h)
        if active_shift >= 0:
            if active_shift < len(h):
                grad_h[active_shift:] = 1 - h[:-active_shift] if active_shift > 0 else 1 - h
        else:
            if -active_shift < len(h):
                grad_h[:active_shift] = 1 - h[-active_shift:] if active_shift < 0 else 1 - h

        # Since exact jacobian would need all instead of one only, use a directional approach

        # Rather, let's summarize the local search approach

        # Simple linesearch with just one step and check improvement
        h = h_new

    # Final refinement with local optimization
    def objective(x):
        return compute_c5(x, dx)
    
    def constraint_sum(x):
        return np.sum(x) - target_sum
    
    # Use resulting h as start for local search
    try:
        from scipy.optimize import minimize
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraints, 
                          options={'maxiter': 50, 'ftol': 1e-8})
        if result.success:
            h = project_h(result.x, target_sum)
    except:
        pass
    
    final_c5 = compute_c5(h, dx)
    
    return (h.tolist(), final_c5, n_points)
