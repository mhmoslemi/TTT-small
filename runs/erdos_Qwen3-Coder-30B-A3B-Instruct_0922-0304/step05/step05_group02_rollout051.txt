import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have sum equal to target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Use a combination of uniform and random initialization
        h = np.ones(n_points) * 0.5
        # Add some randomness
        h += np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient-based optimization with projection
    max_iter = 5000
    lr_decay = 0.9999
    min_lr = 1e-6
    
    lr = 0.01
    
    iteration = 0
    last_improvement = 0
    patience = 500
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute all overlaps
        overlap_full = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap_full)
        max_overlap = overlap_full[max_idx] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
            if best_c5 <= 0.3807:
                break
            
            last_improvement = iteration
        
        # Find the shift that gives the maximum overlap
        shift = max_idx - (n_points - 1)
        
        # Compute the gradient for this shift
        # For shift k, gradient with respect to h[i] is (1 - h[i-k]) * dx
        # When k is positive: h[i-k] for i in range(k, n_points)
        # When k is negative: h[i+k] for i in range(0, n_points+k)
        grad = np.zeros(n_points)
        if shift >= 0:
            # h[i] gets influence from h[i-shift] 
            grad[shift:] = 1 - h[:n_points-shift]
        else:
            # h[i] gets influence from h[i+shift]  
            grad[:n_points+shift] = 1 - h[-shift:]
        
        # Update h using gradient descent
        h = h - lr * grad
        
        # Project back to feasible region
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Adaptive learning rate
        lr = max(lr * lr_decay, min_lr)
        
        # Early stopping
        if iteration - last_improvement > patience:
            break
            
        iteration += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    # One final refinement step with local optimization
    if time.time() - start_time < budget_s - 2:
        from scipy.optimize import minimize
        
        def objective(h_vec):
            return compute_c5(h_vec, dx)
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - target_sum
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        result = minimize(
            objective, 
            best_h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h = normalize_h(final_h, target_sum)
            final_c5 = compute_c5(final_h, dx)
            best_h = final_h
    
    return (best_h.tolist(), final_c5, n_points)
