import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 50
    max_iterations = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.ones(n_points) * 0.5
    
    # Create swarm
    particles = [h0.copy()]
    velocities = [np.random.uniform(-0.05, 0.05, n_points)]
    
    for _ in range(swarm_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Best positions
    best_positions = [p.copy() for p in particles]
    best_values = [float('inf')] * swarm_size
    
    # Global best
    global_best_h = particles[0].copy()
    global_best_c5 = float('inf')
    
    # PSO parameters
    w_max, w_min = 0.9, 0.4
    phi1, phi2 = 2.0, 2.0
    v_max = 0.1
    
    # Main loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate each particle
        for i in range(swarm_size):
            h = particles[i]
            
            # Compute overlap
            overlap = correlate(h, 1 - h, mode="full")
            c5 = np.max(overlap) * dx
            
            # Update particle best
            if c5 < best_values[i]:
                best_values[i] = c5
                best_positions[i] = h.copy()
                
            # Update global best
            if c5 < global_best_c5:
                global_best_c5 = c5
                global_best_h = h.copy()
                
            # Early exit if target achieved
            if global_best_c5 <= 0.3807:
                break
                
        # Update inertia weight
        w = w_max - (w_max - w_min) * (time.time() - start_time) / budget_s
        
        # Update particles
        for i in range(swarm_size):
            r1, r2 = np.random.random(2)
            v = w * velocities[i] + phi1 * r1 * (best_positions[i] - particles[i]) + phi2 * r2 * (global_best_h - particles[i])
            
            # Clip velocity
            v = np.clip(v, -v_max, v_max)
            
            # Update position
            new_h = particles[i] + v
            
            # Clip and renormalize
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            particles[i] = new_h
            velocities[i] = v
            
        if global_best_c5 <= 0.3807:
            break
    
    return (global_best_h.tolist(), global_best_c5, n_points)
