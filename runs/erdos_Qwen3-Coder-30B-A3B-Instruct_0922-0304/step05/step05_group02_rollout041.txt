import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_h(h, target_sum)
    
    # Optimization parameters
    eta = 0.1
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute correlations for all shifts
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        
        # Extract the specific shift that maximizes overlap
        shift = max_idx - (len(h) - 1)
        
        # Compute gradient for the worst-case shift
        # For h * (1 - h) at shift k, gradient w.r.t. h_i is (1 - h_{i+k})
        # When k is negative, we align it to the right; when positive, to the left
        # We compute the gradient at the specific shift that gave max overlap
        h_shifted = np.roll(1 - h, shift) if shift > 0 else np.roll(1 - h, -shift)
        h_shifted = h_shifted[:len(h)]
        grad = h_shifted * dx
        
        # Update with gradient descent
        h_new = h - eta * grad
        
        # Project h to feasible set
        h = project_h(h_new, target_sum)
        
        # Evaluate and update best solution
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Adjust learning rate
        if iteration % 10 == 0:
            if c5 < best_c5 + 1e-8:
                eta *= 1.01
            else:
                eta *= 0.9
        
        # Periodic perturbation
        if iteration % 100 == 0 and iteration > 0:
            noise = np.random.normal(0, 0.001, n_points)
            h = h + noise
            h = project_h(h, target_sum)
        
        iteration += 1
        
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
