import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 20
    max_generations = 1000
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def is_feasible(h):
        return np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) * dx - 1.0) < 1e-10
    
    def repair(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def mutate(h, mutation_strength):
        mutated = h + np.random.normal(0, mutation_strength, h.shape)
        return repair(mutated)
    
    def select_elite(population, elite_fraction=0.25):
        fitness = [compute_c5(individual) for individual in population]
        indices = np.argsort(fitness)[:int(len(population) * elite_fraction)]
        return [population[i] for i in indices]
    
    # Initialize population
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.ones(n_points) * 0.5
        h0[:n_points//4] = 1.0
        h0[3*n_points//4:] = 0.0
    
    population = [h0]
    for _ in range(population_size - 1):
        population.append(mutate(h0, 0.02))
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # EA main loop
    mutation_strength = 0.01
    no_improvement_count = 0
    max_no_improvement = 50
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = [compute_c5(individual) for individual in population]
        
        # Track best
        min_fitness = min(fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_idx = np.argmin(fitness)
            best_h = population[best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Adjust mutation strength
        if no_improvement_count > max_no_improvement:
            mutation_strength *= 0.9
            no_improvement_count = 0
            
        # Selection and reproduction
        elite = select_elite(population)
        new_population = []
        
        # Add elite directly
        new_population.extend(elite)
        
        # Mutate elite to create offspring
        for individual in elite:
            for _ in range(2):
                if time.time() - start_time > budget_s - 1:
                    break
                offspring = mutate(individual, mutation_strength)
                new_population.append(offspring)
        
        # Trim population to original size
        population = new_population[:population_size]
    
    return (best_h.tolist(), best_c5, n_points)
