import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize to sum to target_sum
    h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative LP approach
    max_iter = 50
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute current overlap
        current_overlap = correlate(h, 1 - h, mode="full")
        # Find top lags contributing to maximum overlap
        top_lags = np.argsort(current_overlap)[-max(10, n_points//10):][::-1]
        
        # LP setup
        h_vars = cp.Variable(n_points)
        c5_var = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == target_sum,
            h_vars >= 0,
            h_vars <= 1,
            c5_var >= 0
        ]

        for k in top_lags:
            # Compute lag k overlap with current h
            # overlap_k ≈ h[i] * (1 - h[i+k]) + (h[i+k] - h[i]) * h[i] * dx
            # For lag k in full correlation, indices go from 0 to n_points-1
            # The lag k contribution to the overlap sum at position k:
            # sum_{i=0}^{n_points-k-1} h[i] * (1 - h[i+k])
            # We approximate with linear terms: 
            # For stability, we use a single linearization around current h.
            # This helps solve the NP-hard overlaps with relaxed convex constraints.
            if k <= n_points - 1:

        try:

                # Project back to feasible region

                # Normalize

                        best_c5 = current_c5

                    h = best_h.copy()
            else:
                h = best_h.copy()
        except Exception as e:
            # Use best previous h if any valid solution failed
            h = best_h.copy()
            if best_c5 < float('inf'):
                break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
