import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint while keeping in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-12:
        return h
    h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def line_search(h, grad, dx, max_alpha=1.0, tolerance=1e-4):
    """Perform line search to find good step size."""
    alpha = max_alpha
    current_c5 = compute_c5(h, dx)
    while alpha > 1e-8:
        h_new = project_h(h - alpha * grad, len(h) / 2)
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < current_c5 - tolerance * alpha * np.dot(grad, grad):
            return h_new, alpha
        alpha *= 0.5
    return h, 0.0

    start_time = time.time()
    
    n_points = 70

    # Algorithm parameters
    momentum = 0.95
    step_size = 0.05
    max_iter = 10000
    patience = 50  # Early stopping
    best_c5 = float('inf')
    best_h = h.copy()
    prev_c5 = float('inf')
    no_improve_count = 0

    # Initialize momentum
    m = np.zeros_like(h)

    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all correlations to find max overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        k = max_idx - (n_points - 1)
        max_overlap = overlap[max_idx]

        # Compute gradient for this shift
        grad = np.zeros_like(h)
        if k >= 0:

        else:
            grad[:k] = 1 - h[-k:] if -k < n_points else 0

        # Add momentum
        m = momentum * m + (1 - momentum) * grad
        # Update with learning rate and project
        h = project_h(h - step_size * m, target_sum)
        
        c5 = compute_c5(h, dx)
        if c5 < best_c5:

            best_h = h.copy()
