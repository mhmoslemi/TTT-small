import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def halton_sequence(n, base=2):
    """Generate n points from 1D Halton sequence with given base."""
    result = np.zeros(n)
    for i in range(n):
        f = 1.0
        result[i] = 0.0
        while i > 0:
            f /= base
            result[i] = (i % base) * f
            i //= base
    return result

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Generate Halton sequence for multi-start
    np.random.seed(seed)
    halton_points = halton_sequence(50, 2)  # 50 points
    halton_points = np.tile(halton_points, (n_points, 1)).T
    
    # Multi-start optimization
    for i in range(50):
        # Create a new starting point using Halton
        h_start = halton_points[i, :]
        
        # Add random noise to diversity
        h_start += np.random.normal(0, 0.01, n_points)
        h_start = normalize_h(h_start, target_sum)
        
        # Local search with SLSQP
        def objective(h):
            return evaluate_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        # SLSQP optimization
        result = minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        if result.success:
            h_new = result.x
            h_new = normalize_h(h_new, target_sum)
            c5_val = evaluate_c5(h_new, dx)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_new.copy()
                
        if time.time() - start_time > budget_s - 1:
            break
            
    return (best_h.tolist(), best_c5, n_points)
