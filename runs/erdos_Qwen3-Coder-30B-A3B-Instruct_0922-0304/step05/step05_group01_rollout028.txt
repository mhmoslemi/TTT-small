import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx=2.0/70):
    """Compute C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def mutate(h, mutation_std, n_points):
    """Mutate h with Gaussian noise and enforce constraints"""
    mutated = h + np.random.normal(0, mutation_std, h.shape)
    mutated = np.clip(mutated, 0, 1)
    sum_h = np.sum(mutated)
    mutated = mutated * (n_points / 2) / sum_h
    return mutated

def evaluate_fitness(population):
    """Evaluate C5 for entire population"""
    return [compute_c5(individual) for individual in population]

def select_elite(population, fitness, selection_fraction=0.25):
    """Select top fraction of population based on fitness"""
    indices = np.argsort(fitness)[:int(len(population) * selection_fraction)]
    return [population[i] for i in indices]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize population
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create random feasible solution
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        
    # Create initial population (20 individuals)
    population = [h.copy()]
    for _ in range(19):
        population.append(mutate(h, 0.01, n_points))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # EA parameters
    max_generations = 1000
    mutation_strength = 0.01
    no_improvement_count = 0
    max_no_improvement = 50
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = evaluate_fitness(population)
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Adapt mutation strength
        if no_improvement_count > max_no_improvement:
            mutation_strength *= 0.1
            no_improvement_count = 0
            
        # Selection
        elite = select_elite(population, fitness)
        
        # Generate new population
        new_population = []
        
        # Keep elite individuals
        for individual in elite:
            new_population.append(individual.copy())
            
        # Create offspring from elite
        for individual in elite:
            for _ in range(2):
                if time.time() - start_time > budget_s - 1:
                    break
                offspring = mutate(individual, mutation_strength, n_points)
                new_population.append(offspring)
                
        # Trim to population size
        population = new_population[:20]
        
        # Add new random individuals occasionally
        if len(population) < 20:
            while len(population) < 20:
                h = np.random.random(n_points)
                h = h * (n_points / 2) / np.sum(h)
                h = np.clip(h, 0, 1)
                population.append(h)
    
    return (best_h.tolist(), best_c5, n_points)
