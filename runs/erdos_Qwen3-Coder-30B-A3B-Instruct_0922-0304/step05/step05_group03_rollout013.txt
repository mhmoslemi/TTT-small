import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx):
    """Compute the gradient of the maximum overlap w.r.t. h."""
    # Compute correlations for all lags
    corr = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    n = len(h)
    
    # Gradient computation via finite differences
    grad = np.zeros(n)
    
    # For simplicity, we can use an approximated gradient
    # by perturbing h[i] slightly and computing change in maximum overlap
    epsilon = 1e-5
    max_overlap = corr[max_idx]
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus = normalize_h(h_plus, np.sum(h))
        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
        max_overlap_plus = np.max(corr_plus)
        
        grad[i] = (max_overlap_plus - max_overlap) / epsilon
    
    return grad
