        adjustment = (n_points / 2) / current_sum
        h = h * adjustment
        h = np.clip(h, 0, 1)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Phase 1: Binary approximation using simple heuristic
    if time.time() - start_time < budget_s / 2 - 1:
        # Use local search in the binary space

        binary_h = np.clip(binary_h, 0, 1)

        for iter_num in range(500):
            if time.time() - start_time > budget_s - 2:
                break
            # Try to improve by flipping one bit
            current_c5 = compute_c5(binary_h)
            best_flip = None
            best_flip_c5 = current_c5
            for i in range(n_points):
                temp_h = binary_h.copy()
                temp_h[i] = 1 - temp_h[i]
                if 0 <= temp_h[i] <= 1:
                    # Renormalize to keep sum = n_points / 2
                    old_val = binary_h[i]
                    temp_h[i] = 1 - old_val
                    temp_sum = np.sum(temp_h)

                    temp_h = np.clip(temp_h, 0, 1)

            new_h = current_h - step_size * g

                    best_h = new_h.copy()
            else:
                step_size *= 0.999  # Reduce step size if no improvement
            
            if best_c5 <= 0.3807:
                break

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
