import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    # Use simulated annealing approach with local refinement
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated annealing parameters
    temp_start = 1.0
    temp_end = 0.001
    cooling_rate = 0.995
    max_iter = 2000
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    def neighbor(h, temp):
        """Generate a neighbor solution."""
        h_new = h.copy()
        i = np.random.randint(0, n_points)
        # Perturb one value with Gaussian noise scaled by temperature
        delta = np.random.normal(0, 0.1 * temp)
        h_new[i] += delta
        return normalize_h(h_new, target_sum)
    
    def acceptance_probability(old_cost, new_cost, temp):
        """Calculate acceptance probability for simulated annealing."""
        if new_cost < old_cost:
            return 1.0
        return np.exp((old_cost - new_cost) / temp)
    
    # Initialize
    current_h = h.copy()
    current_cost = objective(current_h)
    temp = temp_start
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Generate neighbor
        new_h = neighbor(current_h, temp)
        new_cost = objective(new_h)
        
        # Accept or reject
        if acceptance_probability(current_cost, new_cost, temp) > np.random.random():
            current_h = new_h
            current_cost = new_cost
            
        # Update best
        if current_cost < best_c5:
            best_c5 = current_cost
            best_h = current_h.copy()
            
        # Check for early termination
        if best_c5 <= 0.3807:
            break
            
        # Cool down
        temp = max(temp_end, temp * cooling_rate)
    
    # Local refinement with L-BFGS
    def refine_h(h):
        def obj(h_vec):
            h = h_vec.reshape(-1)
            return evaluate_c5(h, dx)
        def constraint_sum(h_vec):
            return np.sum(h_vec.reshape(-1)) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        h_flat = h.flatten()
        result = minimize(
            obj,
            h_flat,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        
        if result.success:
            return normalize_h(result.x.reshape(-1), target_sum)
        return h
    
    # Apply final refinement to the best solution
    best_h = refine_h(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
