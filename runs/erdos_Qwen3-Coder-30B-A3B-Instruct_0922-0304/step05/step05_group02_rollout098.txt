import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum = target_sum and bounds [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, n_points / 2)
    else:
        # Create a simple initial guess
        h = np.ones(n_points) * 0.5
        h = project_to_feasible(h, n_points / 2)
    
    # Track best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 500
    perturb_freq = 50
    noise_magnitude = 0.1
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps for all shifts
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Use a simplified gradient-free approach with alternating optimizations
        # with local projections to avoid getting stuck in poor local minima
        
        # Local optimization using projected gradient descent
        # First, compute gradients approximately via finite differences
        h_original = h.copy()
        eps = 1e-4
        h_plus = h.copy()
        h_minus = h.copy()
        
        # Perform projected gradient step
        gradient = np.zeros_like(h)
        for i in range(n_points):
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_to_feasible(h_plus, n_points / 2)
            h_minus = project_to_feasible(h_minus, n_points / 2)
            c5_plus = compute_c5(h_plus, dx)
            c5_minus = compute_c5(h_minus, dx)
            gradient[i] = (c5_plus - c5_minus) / (2 * eps)
            h_plus[i] = h[i]
            h_minus[i] = h[i]
            
        # Update using a small step size
        step_size = 1e-3
        h = h - step_size * gradient
        h = project_to_feasible(h, n_points / 2)
        
        # Periodically add noise to escape local minima
        if iteration % perturb_freq == 0 and iteration > 0:
            noise_magnitude *= 0.95
            noise = np.random.normal(0, noise_magnitude, n_points)
            h += noise
            h = project_to_feasible(h, n_points / 2)
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
