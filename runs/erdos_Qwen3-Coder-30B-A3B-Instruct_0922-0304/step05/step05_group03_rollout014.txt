import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent with adaptive learning rate
    lr = 0.1
    patience = 0
    max_patience = 100
    last_improvement = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient
        # We need to compute ∂C5/∂h[i] for all i
        # The derivative can be computed via finite differences or by 
        # exploiting the structure of the overlap computation.
        # For now we use numerical differentiation for safety.
        
        # To compute gradient of max correlation w.r.t. h, we approximate:
        # ∂C5/∂h[i] ~ (C5(h+εe_i) - C5(h-εe_i)) / (2ε)
        # But separately finding which correlation peak to differentiate
        # is expensive and unstable.
        
        # Instead, we approximate by simply using a gradient step
        # with a fixed number of gradient steps per iteration.
        # Let's consider modifying h to reduce maximum correlation.
        
        # Compute the full correlation matrix
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]

        window_size = 5

        end_idx = min(max_idx + window_size, len(corr))
        
        # Simple optimization: perturb individual h[i] and see improvement
        best_h_new = h.copy()
        best_delta = 0
        
        # Try small changes to each element
        for i in range(n_points):
            # Try bidirectional changes
            for delta in [-0.01, 0.01]:
                h_trial = h.copy()
                h_trial[i] += delta
                h_trial = normalize_h(h_trial, target_sum)
                
                c5_trial = evaluate_c5(h_trial, dx)
                if c5_trial < current_c5:
                    # Accept this change
                    if c5_trial < best_c5:
                        best_c5 = c5_trial
                        best_h = h_trial.copy()
                    h = h_trial.copy()
                    break  # Accept the first improving variation
            else:

        # Occasionally do larger adjustments to escape local minima
        if iteration % 50 == 0 and patience > 50:
            # Perturb by larger amount

        else:

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
