import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Binary search approach
    C5_low = 0.0
    C5_high = 0.38085
    tolerance = 1e-4
    max_iterations = 40
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Helper to check if a C5 value is feasible
    def is_feasible(C5):
        # Define the overlap constraint for all k
        def overlap_constraint(h):
            overlap = correlate(h, 1 - h, mode="full")
            return C5 * dx - np.max(overlap)  # Should be >= 0 for feasibility
            
        # Try to solve using optimization
        # We want to minimize a constant (to avoid optimization artifacts)
        # and make sure the overlap constraint is satisfied
        constraints = [
            {'type': 'eq', 'fun': constraint_sum},
            {'type': 'ineq', 'fun': overlap_constraint}
        ]
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try several starting points to avoid local optima
        max_attempts = 3
        for _ in range(max_attempts):
            # Random perturbation of h to try
            h_test = h + np.random.normal(0, 0.02, n_points)
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            
            try:
                result = minimize(
                    lambda x: 0,  # Purely a feasibility check
                    h_test,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 100, 'ftol': 1e-6}
                )
                if result.success:
                    # Check the actual overlap value after optimization
                    actual_c5 = compute_c5(result.x)
                    if actual_c5 <= C5:
                        return True, result.x
            except:
                continue
                
        # If all attempts fail, consider infeasible
        return False, None

    # Binary search loop
    iterations = 0
    while C5_high - C5_low > tolerance and iterations < max_iterations:
        if time.time() - start_time > budget_s - 1:
            break
            
        C5_mid = (C5_low + C5_high) / 2.0
        feasible, candidate_h = is_feasible(C5_mid)
        
        if feasible:
            C5_high = C5_mid
            if candidate_h is not None:
                test_c5 = compute_c5(candidate_h)
                if test_c5 < best_c5:
                    best_c5 = test_c5
                    best_h = candidate_h.copy()
        else:
            C5_low = C5_mid
            
        iterations += 1
        
        if best_c5 <= 0.3807:
            break

    # Final refinement using optimization to improve local solution
    # If we have a good starting point from binary search, try to refine it
    if best_c5 > 0.3807:
        # Use a more direct approach to minimize C5 directly
        def objective(h):
            return compute_c5(h)

        def eq_constraint(h):
            return np.sum(h) - n_points / 2

        constraints = [{'type': 'eq', 'fun': eq_constraint}]
        bounds = [(0, 1) for _ in range(n_points)]

        # Try to refine the best found solution if we're not at the target yet
        if iterations < max_iterations:
            try:
                result = minimize(
                    objective, 
                    best_h, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=constraints,
                    options={'maxiter': 100, 'ftol': 1e-8}
                )
                if result.success:
                    refined_c5 = compute_c5(result.x)
                    if refined_c5 < best_c5:
                        best_c5 = refined_c5
                        best_h = result.x.copy()
            except:
                pass

    return (best_h.tolist(), best_c5, n_points)
