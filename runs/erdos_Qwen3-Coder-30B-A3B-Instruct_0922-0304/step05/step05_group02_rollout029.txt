import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        # Create initial feasible solution
        h = np.ones(n_points)
        h[n_points//2:] = 0
        # Add small random noise
        h += np.random.normal(0, 0.05, n_points)
        h = project_h(h, target_sum)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent parameters
    max_iter = 10000
    learning_rate = 0.01
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        overlap_full = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap_full)
        max_overlap = overlap_full[max_k] * dx
        
        # Update best if needed
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient direction
        # This is for compatible shifts only - simplified approach
        # For the maximizing shift k, we need to compute a gradient
        # Approximate this using the full correlation result
        # In practice, we use conceptual gradient: 
        # For shift k, gradient w.r.t h[i] is (1 - h[i-k]) * dx
        
        # However, this requires more precise logic for the exact gradient
        # We will use a fixed-step gradient ascent method with safety
        
        # Clip and project to maintain constraints
        h_new = h.copy()
        # x = np.arange(n_points)  # for indexing
        # Choose a simpler approach with direct gradient estimates
        # Find the element-wise influence
        # We make a gradient step and then project
        # Simplified: gradient estimate from the maximizer k
        k = max_k - (n_points - 1)  # map full index to shift
        # compute the effect (1-h[i+k]) scaled by dx
        grad = np.zeros_like(h_new)
        for i in range(n_points):
            if 0 <= i-k < n_points:
                grad[i] = (1 - h[i-k]) * dx
            else:
                grad[i] = 0
        
        # Apply simple ascent step
        h_new = h_new + learning_rate * grad
        
        # Project to feasible set
        h = project_h(h_new, target_sum)
        
    # Final evaluation of best solution
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
