import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    
    # Perturb the initial construction a bit
    noise = np.random.normal(0, 0.02, n_points)
    base_h = base_h + noise
    base_h = np.clip(base_h, 0, 1)
    base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()

    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute overlap to determine worst contribution
        overlap_full = correlate(best_individual, 1 - best_individual, mode="full")
        # For a 1D signal of length n (here n=70), correlation has length 2*n-1
        # So for indices k in range(2*n-1), the corresponding lag j is k - (n-1)
        # The lag ranges from -n+1 to n-1
        
        # We focus on the lag range that is most critical
        # The worst (highest) value of the overlap array
        k_worst = np.argmax(overlap_full) - (n_points - 1)
        
        # We'll add 3 adjacent lags to make the investigation more robust
        selected_lags = [k_worst - 1, k_worst, k_worst + 1]
        selected_lags = [k for k in selected_lags if -n_points + 1 <= k <= n_points - 1]
        # If we have too few lags, we can do a full optimization
        if len(selected_lags) < 3:
            selected_lags = list(range(-n_points + 1, n_points))
        
        # For constructing the subproblem, we'll sample a rectangular region
        # where we vary a few parameters of the function h
        # We use a global search algorithm
        
        # Perform optimization on the full system with SLSQP
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Optimize a bit more drastically
        result = minimize(objective, best_individual.copy(), 
                          method='SLSQP', bounds=bounds, 
                          constraints=constraint, 
                          options={'maxiter': 100})
        
        if result.success:
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            current_c5 = compute_c5(h_new, dx)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_individual = h_new.copy()
                
                # Early stopping if we've reached the target
                if current_c5 <= 0.3807:
                    break
        else:
            # Fallback to simple adaptive hill climbing
            h_new = best_individual.copy()
            h_new += np.random.normal(0, 0.01, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            
            current_c5 = compute_c5(h_new, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_individual = h_new.copy()
                if current_c5 <= 0.3807:
                    break
        
        iteration += 1
        
    # Final validation
    final_c5 = compute_c5(best_individual, dx)
    
    return (best_individual.tolist(), final_c5, n_points)
