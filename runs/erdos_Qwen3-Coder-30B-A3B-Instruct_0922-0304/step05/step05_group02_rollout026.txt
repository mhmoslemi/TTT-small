import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: clip to [0,1] and normalize."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        # Create smooth initial solution
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.tanh(5 * (x - 1)))
        h = project_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient descent with adaptive step size
    max_iter = 10000
    lr_decay = 0.9999
    min_lr = 1e-6
    lr = 0.01
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap for all shifts
        overlap_full = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap_full)
        
        # Compute gradient numerically for the worst-case shift
        k = max_idx - (n_points - 1)  # actual shift
        grad = np.zeros(n_points)
        
        # Gradient for shift k: d/dh[i] sum_j h[j](1-h[j+k]) = (1-h[j+k]) where j=k+i 
        # downstream: h[j] for j in range(n_points)
        # shifted: h[j] for j in range(max(0,-k), min(n_points, n_points-k))  
        if k >= 0:
            # h[j+k] for j in range(N-k) as such: j in range(N-k), j+k in range(k, N)
            grad[:-k] = (1 - h[k:])
        else:
            # h[j+k] for j in range(-k) as such: j in range(-k), j+k in range(0, k)
            grad[-k:] = (1 - h[:k])
        
        # Update
        h = h - lr * grad
        h = project_h(h, target_sum)
        
        # Check and update best solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Adaptive learning rate
        lr = max(lr * lr_decay, min_lr)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
