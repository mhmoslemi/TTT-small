import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: clip to [0,1] and scale to sum(target_sum)."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)

    # Parameters
    eta = 0.1
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute all correlations
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Compute gradient w.r.t. the max shift
        # For shift k (max_idx), the gradient is (1 - h_shifted) with shift k
        k = max_idx - (n_points - 1)  # the actual shift
        shifted_h = np.roll(h, -k) if k > 0 else np.roll(h, -k)
        gradient = (1 - shifted_h) * dx  # Pointwise gradient

        # Update with gradient descent
        h = h - eta * gradient

        # Project back to feasible region
        h = project_h(h, target_sum)
        
        # Periodic noise perturbation
        if iteration % 100 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = project_h(h, target_sum)
            
        # Compute current C5
        current_c5 = compute_c5(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Adapt learning rate
        if iteration > 0 and iteration % 10 == 0:
            eta = eta * 1.01 if current_c5 < compute_c5(best_h, dx) else eta * 0.9
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
