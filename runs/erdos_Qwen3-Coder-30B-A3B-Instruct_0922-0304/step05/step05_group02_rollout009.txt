import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def line_search(h, grad, dx, max_alpha=1.0, tolerance=1e-4):
    """Perform line search to find optimal step size."""
    alpha = max_alpha
    old_c5 = compute_c5(h, dx)
    for _ in range(20):
        h_new = project_h(h - alpha * grad, len(h) / 2)
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < old_c5 - tolerance * alpha * np.sum(grad**2):
            return alpha, h_new
        alpha *= 0.5
    return alpha, h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum = n_points/2
    h = project_h(h, n_points / 2)
    
    # Optimization parameters
    momentum = 0.9
    step_size = 0.1
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Gradient-based optimization
    v = np.zeros(n_points)
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Compute all correlations and find maximum
        correlations = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(correlations)
        max_corr = correlations[max_k]
        
        # Compute gradient for this shift
        grad = np.zeros(n_points)
        if max_k < n_points:
            # h * (1-h_shifted) for k = max_k
            shift = max_k
            grad[:n_points-shift] += (1 - h[shift:])
        else:
            # h_shifted * (1-h) for k > n_points
            shift = max_k - n_points + 1
            grad[shift:] += (1 - h[:n_points-shift])
        
        # Update with momentum
        v = momentum * v + grad
        # Adaptive step-size
        step_size = 0.1 * (1 + np.abs(np.sum(v)) / (1 + np.sum(np.abs(grad))))
        
        # Perform line search
        alpha, h_new = line_search(h, v, dx, step_size)
        
        h = h_new
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early stopping if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
