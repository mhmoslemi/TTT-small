import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient of overlap at k_max with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    
    # Overlap at lag k_max is sum_{i=0}^{n-1} h[i] * (1 - h[i+k_max])
    # For fixed k_max, the derivative w.r.t. h[i] is:
    # - (1 - h[i+k_max]) if i+k_max < n, else 0
    # - (1 - h[i-k_max]) if i-k_max >= 0, else 0
    
    for i in range(n):
        # Part 1: derivative from h[i] * (1 - h[i+k_max]) for i+k_max < n
        if i + k_max < n:
            grad[i] -= (1 - h[i + k_max])
        
        # Part 2: derivative from h[i] * (1 - h[i-k_max]) for i-k_max >= 0
        if i - k_max >= 0:
            grad[i] -= (1 - h[i - k_max])
            
    return grad * dx

def compute_max_lag_overlap(h, dx):
    """Find the max overlap lag and its value."""
    overlap = correlate(h, 1 - h, mode="full")
    max_lag_idx = np.argmax(overlap)
    max_overlap = overlap[max_lag_idx] * dx
    return max_lag_idx - (len(h) - 1), max_overlap

    start_time = time.time()
    
    # Parameters
    n_points = 70

    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)

    # Parameters for GD
    alpha = 0.1
    decay_factor = 0.5
    decay_interval = 50
    stagnation_count = 0
    max_stagnation = 100
    
    best_c5 = float('inf')
    best_h = h.copy()
    previous_c5 = float('inf')
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Find the max overlap and corresponding lag
        k_max, current_c5 = compute_max_lag_overlap(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
            # Early exit if we have already reached target
            if best_c5 <= 0.3807:
                break
                
        # Random perturbation occasionally to escape local minima
        if np.random.random() < 0.01:
            h += np.random.normal(0, 0.01, n_points)
            h = normalize_h(h, target_sum)
            continue
            
        # Compute gradient of the max overlap
        gradient = compute_gradient(h, dx, k_max)
        
        # Gradient descent update
        h = h - alpha * gradient
        
        # Clip and re-normalize
        h = normalize_h(h, target_sum)
        
        # Adjust learning rate if no improvement
        if current_c5 >= previous_c5:
            stagnation_count += 1
        else:
            stagnation_count = 0
            
        if stagnation_count % decay_interval == 0 and stagnation_count > 0:
            alpha *= decay_factor
            
        previous_c5 = current_c5
            
        # Early stopping check on stagnation
        if stagnation_count > max_stagnation:
            break
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
