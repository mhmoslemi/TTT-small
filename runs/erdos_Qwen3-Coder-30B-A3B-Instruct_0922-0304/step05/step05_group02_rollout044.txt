```python
import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: clip and rescale."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = project_h(h, target_sum)
    
    # Parameters
    eta = 0.1
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute all correlations
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_k = max_idx - (n_points - 1)
        
        # Compute gradient (simplified for maximum shift)
        # When k=0, correlation is h*(1-h) at 0, so gradient is 1-h shifted by 0
        # We compute gradient for actual k
        gradient = np.zeros_like(h)
        if max_k >= 0:
            gradient[max_k:] = (1 - h[:-max_k]) if max_k < n_points else 0
        else:
            gradient[:max_k] = (1 - h[-max_k:]) if -max_k < n_points else 0
            
        # Scale by dx (the correlation is multiplied by dx in the definition)
        gradient = gradient * dx
        
        # Gradient step
        h = h - eta * gradient
            
        # Project to feasible set
        h = project_h(h, target_sum)
        
        # Update best solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Adaptive learning rate
        eta = eta * 1.01 if current_c5 < best_c5 else eta * 0.9
            
        # Periodic noise for escape from local minima
        if iteration % 100 == 0 and iteration > 0:
            noise_scale = 0.01
            h = h + np.random.normal(0, noise_scale, n_points)
            h = project_h(h, target_sum)
            
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
```
