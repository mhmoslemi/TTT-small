import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Adaptive discretization
    max_n_points = 120
    current_n_points = n_points
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Solve convex relaxation
        variables = cp.Variable(current_n_points)
        constraints = [
            variables >= 0,
            variables <= 1,
            cp.sum(variables) == current_n_points / 2
        ]
        
        # For convergence, we approximate the max overlap constraint
        # We'll use a smooth approximation of the max constraint
        # Compute current overlap
        current_overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(current_overlap)
        
        # Objective: minimize max overlap
        objective = cp.Minimize(cp.max(cp.convolve(variables, 1 - variables, 'full')) * dx)
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = variables.value
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (current_n_points / 2) / np.sum(new_h)
                h = new_h
                
                # Check if we improved
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    
                # Early termination
                if best_c5 <= 0.3807:
                    break
                    
                # Increase resolution if we're making progress and time allows
                if iteration % 5 == 0 and current_n_points < max_n_points:
                    current_n_points = min(current_n_points + 10, max_n_points)
                    # Interpolate to new size
                    old_h = h.copy()
                    h = np.interp(np.linspace(0, 2, current_n_points), 
                                np.linspace(0, 2, len(old_h)), old_h)
                    h = np.clip(h, 0, 1)
                    h = h * (current_n_points / 2) / np.sum(h)
        except Exception:
            pass
            
        # Random perturbation if we're stuck
        if iteration % 10 == 0:
            noise = np.random.normal(0, 0.02, current_n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (current_n_points / 2) / np.sum(h)
            
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, current_n_points)
