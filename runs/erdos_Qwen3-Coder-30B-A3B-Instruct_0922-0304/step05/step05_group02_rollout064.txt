import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_feasible(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        scaled = h * (target_sum / current_sum)
        return np.clip(scaled, 0, 1)
    else:
        return np.clip(h, 0, 1)

def compute_gradients(h, dx, worst_shift_idx):
    """Compute gradient w.r.t. the worst contributing shift."""
    # Compute the convolution derivative at the worst shift
    n = len(h)
    grad = np.zeros(n)
    # For h ∗ (1 - h) at shift k, gradient w.r.t. h at position i is:
    # d/dh_i [h_i (1-h_{i+k})] = 1 - h_{i+k} if i+k < n
    # d/dh_i [h_i (1-h_{i+k})] = 1 - h_{i+k-n} if i+k >= n
    if worst_shift_idx >= 0:
        # h * (1-h) at shift worst_shift_idx
        for i in range(n):
            j = i + worst_shift_idx
            if j < n:
                grad[i] += 1 - h[j]
            else:
                j = j - n
                if j < n:
                    grad[i] += 1 - h[j]
    else:
        # Negative shift case - handle appropriately
        shift = -worst_shift_idx 
        for i in range(n):
            j = i - shift
            if 0 <= j < n:
                grad[i] += 1 - h[j]
            else:
                j = j + n
                if 0 <= j < n:
                    grad[i] += 1 - h[j]
    return grad * dx

def line_search(h, grad, max_alpha=1.0, tolerance=1e-3):
    """Perform backtracking line search."""
    alpha = max_alpha
    for _ in range(20):  # Max iterations
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        # Project to satisfy sum constraint
        h_new = project_feasible(h_new, len(h) / 2)
        # Check feasibility
        if np.abs(np.sum(h_new) - len(h) / 2) < 1e-6:
            return h_new, alpha
        alpha *= 0.5
    return h, 0.1

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_feasible(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_feasible(h, target_sum)

    # Optimizer parameters
    momentum_decay = 0.9
    learning_rate = 0.1
    momentum = np.zeros_like(h) 
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute all overlap values
        conv = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(conv)
        max_idx = np.argmax(conv)
        
        # Handle wraparound in max_idx
        max_shift = max_idx - (len(h) - 1)  # Code assumes positive shifts first
        if max_shift < 0:
            max_shift += len(h)
        
        # Compute gradient for worst case shift
        grad = compute_gradients(h, dx, max_shift-len(h)+1)  # convert back to shift index
        
        # Update momentum
        momentum = momentum_decay * momentum + (1 - momentum_decay) * grad
        
        # Apply trust region step
        h_new, actual_alpha = line_search(h, momentum, learning_rate)
        
        # If line search didn't work, try with smaller step
        if actual_alpha < 1e-6:
            h_new, actual_alpha = line_search(h, grad, 0.1)
            
        # Update h with projected version
        h = h_new
        h = project_feasible(h, target_sum)

        # Evaluate and update if better
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1

    return (best_h.tolist(), best_c5, n_points)
