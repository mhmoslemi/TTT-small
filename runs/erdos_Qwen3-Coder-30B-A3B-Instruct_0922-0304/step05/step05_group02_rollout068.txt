import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_h(h, target_sum)

    # Optimization parameters
    eta = 0.1
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute all correlations to find maximum
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_overlap = overlap[max_idx]
        k_shift = max_idx - (n_points - 1)  # Convert to shift index
        
        # Compute gradient w.r.t. k_shift: it's (1 - h_shifted)
        h_shifted = np.roll(h, k_shift)
        grad = (1 - h_shifted) * dx
        
        # Gradient update
        h = h - eta * grad
        
        # Project to feasible region
        h = project_h(h, target_sum)
        
        # Adaptive learning rate
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            eta *= 1.01  # Increase learning rate if improved
        else:
            eta *= 0.9   # Decrease if no improvement
            
        # Add noise periodically to escape local minima
        if iteration % 100 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = project_h(h, target_sum)
        
        # Early exit
        if best_c5 <= 0.3807:
            break

        iteration += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_c5, n_points)
