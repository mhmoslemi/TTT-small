import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initial guess
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Create swarm
    particles = [h.copy()]
    velocities = [np.random.uniform(-0.01, 0.01, n_points)]
    
    for i in range(swarm_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
    
    # PSO parameters
    omega_max = 0.9
    omega_min = 0.4
    phi1 = 2.0
    phi2 = 2.0
    v_max = 0.1
    
    # Track best particles
    p_best = [p.copy() for p in particles]
    p_best_values = [float('inf')] * swarm_size
    g_best = particles[0].copy()
    g_best_value = float('inf')
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Update inertia weight
        omega = omega_max - (omega_max - omega_min) * (time.time() - start_time) / budget_s
        
        # Evaluate all particles
        for i in range(swarm_size):
            h = particles[i]
            
            # Compute C5 for current particle
            c5 = compute_c5(h)
            
            # Update personal best
            if c5 < p_best_values[i]:
                p_best_values[i] = c5
                p_best[i] = h.copy()
            
            # Update global best
            if c5 < g_best_value:
                g_best_value = c5
                g_best = h.copy()
        
        # Check if target achieved
        if g_best_value <= 0.3807:
            break
            
        # Update particles
        for i in range(swarm_size):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (omega * v + 
                 phi1 * r1 * (p_best[i] - h) + 
                 phi2 * r2 * (g_best - h))
            
            # Clamp velocity
            v = np.clip(v, -v_max, v_max)
            
            # Update position
            h = h + v
            
            # Clamp values to [0,1]
            h = np.clip(h, 0, 1)
            
            # Renormalize to satisfy constraint
            h = h * (n_points / 2) / np.sum(h)
            
            # Store updated values
            particles[i] = h
            velocities[i] = v
    
    return (g_best.tolist(), g_best_value, n_points)
