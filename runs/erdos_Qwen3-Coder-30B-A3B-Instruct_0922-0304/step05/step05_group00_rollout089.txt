import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        # Create a reasonable starting point
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        # Add some noise
        base_h += np.random.normal(0, 0.05, n_points)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Local optimization attempts
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        current_c5 = compute_c5(best_individual, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_individual = best_individual.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Random perturbations with lower noise to fine-tune
        h_trial = best_individual.copy()
        noise_level = 0.01 + (0.05 * (1.0 - (time.time() - start_time) / budget_s))
        h_trial += np.random.normal(0, noise_level, n_points)
        
        # Clip and normalize
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = normalize_h(h_trial, target_sum)
        
        # Try to improve with local optimization
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # SLSQP optimization
        result = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
        
        if result.success:
            h_refined = result.x
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = normalize_h(h_refined, target_sum)
            
            # Check if it's better
            refined_c5 = compute_c5(h_refined, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_individual = h_refined.copy()
        else:
            # If optimization failed, just use the trial
            trial_c5 = compute_c5(h_trial, dx)
            if trial_c5 < best_c5:
                best_c5 = trial_c5
                best_individual = h_trial.copy()
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
