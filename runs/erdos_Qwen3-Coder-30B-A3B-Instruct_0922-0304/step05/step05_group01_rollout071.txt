import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize swarm
    n_particles = 20
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        best_h = np.array(initial_h_values)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    else:
        # Fallback initialization
        best_h = np.ones(n_points)
        best_h[n_points//2:] = 0
        best_h += np.random.normal(0, 0.05, n_points)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    best_c5 = float('inf')
    best_h_global = best_h.copy()
    
    # Initialize particles
    for _ in range(n_particles):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update global best
        current_c5 = compute_c5(best_h_global)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_global = best_h_global.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Update particles
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Evaluate current particle
            fitness = compute_c5(h)
            
            # Update personal best
            if fitness < compute_c5(best_h):
                best_h = h.copy()
                
            # PSO velocity update
            r1, r2 = np.random.rand(2)
            v = w * v + c1 * r1 * (best_h - h) + c2 * r2 * (best_h_global - h)
            
            # Clip velocity
            v = np.clip(v, -0.2, 0.2)
            
            # Update position
            h_new = h + v
            h_new = project_to_feasible(h_new)
            
            particles[i] = h_new
            velocities[i] = v
            
            # Update global best for this particle
            if compute_c5(h_new) < compute_c5(best_h_global):
                best_h_global = h_new.copy()
        
        # Decrease inertia
        w = max(0.4, w - 0.001)
    
    # Final evaluation
    final_c5 = compute_c5(best_h_global)
    
    return (best_h_global.tolist(), final_c5, n_points)
