import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have sum equal to target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    scale = target_sum / current_sum
    h_proj = h * scale
    return np.clip(h_proj, 0, 1)

def compute_gradients(h, dx, k_max):
    """Compute gradients for the worst shift."""
    n = len(h)
    # Convert to list of shifts to process
    shifts = list(range(-n+1, n))
    # Get all overlaps
    overlap = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(overlap)
    max_val = overlap[max_idx]
    
    # Find shifts corresponding to maximum value
    worst_k_indices = np.where(overlap == max_val)[0]
    worst_ks = [shifts[i - (n-1)] for i in worst_k_indices]
    
    # Return average gradient for worst shifts
    # For a shift k, the gradient is the derivative of the overlap w.r.t h_i
    # We approximate gradient at the position and signal that contributes to the worst overlap
    
    # We'll focus on the worst shift (we can randomize or take average)
    # Substitute one worst shift (random from the set)
    k = worst_ks[0]
    
    # Analytical gradient calculation for shift k:
    # d/dh_i (h(x) * (1 - h(x+k))) = (1 - h(x+k)) * delta(x-i) - h(x) * delta(x+k-i)
    # Total gradient at i: (1-h[i + k]) - h[i-k] (when i-k >= 0, i+k < n)
    # but since we are doing correlation, much more direct:
    # For h[i], contribution to overlap at shift k is h[i]* (1-h[i+k]) + h[i+k] * (1-h[i])
    
    # Gradient of the overlap = (1 - h[i+k]) if shift is k (but we're looking at gradient with respect to h[i])
    # So we change h[i] to affect (h[i] * h[i+k]) term? Or we bother about derivative of the whole overlap
    
    # So naively use the correlation derivative
    grad = np.zeros(n)
    # Compute gradient d/dh[i] of overlap at position k, using only that term
    # But shifts affect every element to the left by up to k steps
    # So in overlap sum, each h[i] affects the h[i+k] contributions
    
    # Direct way: we look at derivative of overlap for shift k w.r.t. h[i]
    
    # Gradient for shift k where max overlap occurs:
    # The overlap at shift k is sum_i h[i] (1 - h[i+k])
    # d/dh[i] (sum h[i'] (1 - h[i'+k])) = (1 - h[i+k]) for i'=i
    # But we may wrap on boundaries
  
    # Get current state:
    # For the shift k (compute overlap = sum(h[i] * (1 - h[i+k])) for valid indices)
    # The partial df/dh[i] = (1 - h[i+k]) if i+k is valid and k is the shift
    # More accurately for correlation at k: sum_{i=(0 to n-1)} h[i] * (1 - h[i+k]) 
    # Let the derivative w.r.t h[j] = sum_i (d/dh[j] (h[i] * (1 - h[i+k]))) = 0 unless i=j, i=j+k
    # So for a single term at position i, if j = i, term is h[i] * (1 - h[i+k])
    # If j = i+k, term is h[i+k] * (1 - h[i])
    # d/dh[j] is: (1 - h[j+k]) (where vector is h shifted)
    
    # Let's do the gradient way:
    # H[shift_k] = sum_i h[i] * (1 - h[i+k])
    # dH/dh[i] = (1 - h[i+k]) when the indices are valid.
    
    # But that's for only one shift k. We want the maximum over all k.
    # Let's take the derivative of max_k H_k.
    
    # To compute gradient of overlap at position k with respect to h[i]:
    # z[k] = sum_{i} h[i] * (1 - h[i+k]) = dot(h, 1-h_k)
    # d z[k]/d h[i] = (1 - h[i+k]) (if 0 <= i+k < n) and 0 otherwise.
    
    # Since we want gradient of max_k z[k]:
    # It's more complex... Let's just take the derivative of the k where max is achieved.
    
    # We will approximate that the gradient is (1 - h[i+k]) for appropriate k.
    # A// As a rough approximation, we set the gradient for k corresponding to max overlap
    
    # Right way: Make a derivative operator for shift k:
    # grad[k][i] = 1 - h[i+k] if (i+k) is in range [0,n), otherwise 0.
    
    # But for a max, we take the derivative with respect to the actual max k.
    # Let's just take the gradient relative to shift k = k_max
    # i.e. if max overlap is at some k* we compute gradient of overlap at k* w.r.t. h
    
    # Actually let's compute the gradient explicitly by computing field that is relevant:
    # if z[k] = sum_{i=0}^{n-1} h[i] * (1 - h[i+k]) where indices are properly 
    # and we take the derivative w.r.t. h[j]:
    # d/dh[j] z[k] = (1 - h[j+k]) if (j+k) is valid, 0 otherwise
    # This is just a right shift by k of (1-h) 
    # Thus: d/dh[j] z[k]  = (1 - h[j+k]) if j+k < n and j >=0, otherwise 0.
    # This is also the i-th element of (1 - h_shifted_right_by_k)
    
    # All the above can be simplified by a correlation: 
    # If we define u = 1-h, the overlap z[k] = dot(h, u[k]) = h * u_k where u_k = right-shift u by k
    
    # Now d/dh[j] z[k] = (1 - h[j+k]) if j+k < n  and 0 if invalid
    # This gives us a matrix or just computation.
    # Let's compute it the same way:
    # For gradient w.r.t. h at a specific index, we can do it for the maximize k.
    # We are saying w.r.t to max_overlap function,
    # Let it be another implementation:
    # z[k] = sum h[i] * (1- h[i+k])
    # d/dh[j] z[k] = -1 * indicator{ j=k+i } if j+k < n and j >= 0
    # This is the same as convolution integration with [1,1] (because of constraint)
    # So gradient is simply: (1 - 1*h) shifted right by k, but extended only where valid.
    
    # Very good idea:
    # Let us take a fully generic and efficient approach:
    # We extract the correlation value with shift k.
    # If k is worst:
    # we compute gradient like this:
    
    # i = 0 to n-1, then gradient = 1 - h[i+k] if i+k < n and i+k >= 0, 0 otherwise
    # If k < 0 then we can write it as: 
    # i-first, h[i+k] if i+k >= 0 and i+k < n
    # The derivative is 1 - (1-h[i+k]) for any valid and active shift
    
    # Let's use an easier: computation step:
    # For shift k = k_max, compute [1-h], and right shift it by k to create right_shifted(1-h)
    # Let's not bias towards signed shift for efficiency and safety
    
    # Compute the matrix product:
    # let u = 1 - h
    # We compute z[k] = sum h[i] * (1 - h[i+k]) = h * (1-h)[k] using correlation
    # So d/dh_j z[k] is [1-u][k-j]
    # Input: k
    # Let's say our h array is 0 to n-1
    # For derivative wrt h[j], z[k] = sum_i h[i] * (1-h[i+k]) = h * (1-h)_{k}
    # d/dh[j] = (1-h[j+k])  if the indices are valid, 0 otherwise.
    
    grad = np.zeros(n)
    for i in range(n):
        j = i + k
        if j >= 0 and j < n:
            grad[i] = 1 - h[j]
    
    # Now project to feasible region
    grad = np.clip(grad, -1, 1)
    return grad

def line_search(h, grad, dx, max_evals=20):
    """Perform backtracking line search for suitable step size."""
    # Reduce step size until feasibility or acceptable C5 is achieved
    step_size = 1.0
    alpha = 0.5
    c = 1e-4
    initial_c5 = compute_c5(h, dx)
    
    for _ in range(max_evals):
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        if abs(h_new_sum - len(h)/2) < 1e-4:
            h_proj = h_new
        else:
            scale = (len(h)/2) / h_new_sum
            h_proj = h_new * scale
        
        if np.all((h_proj >= 0) & (h_proj <= 1)) and np.abs(np.sum(h_proj) - len(h)/2) < 1e-6:
            new_c5 = compute_c5(h_proj, dx)
            if new_c5 <= initial_c5 - c * step_size * np.dot(grad, grad):
                return step_size, h_proj
        step_size *= alpha
    return step_size, h

def trust_region_update(h, dx, learning_rate=0.1, momentum=0.9):
    """Perform trust region update with momentum term."""
    n = len(h)
    h = np.array(h)
    
    # Get current best overlap
    overlap = correlate(h, 1 - h, mode="full")
    k_max = np.argmax(overlap)
    max_val = overlap[k_max]
    
    # Backcompute the gradient for k_max
    # d/dh[i] (overlap at k_max) = 1 - h[i + k_max] if valid
    
    # Simplify computation for gradient:
    # if we have shift k, the kth shift of h is h[...k:]
    # and we multiply each of h with that.
    # So for k we do:
    # (1 - h) shifted by -k: 1 - h[0-k:] assuming proper indexing
    # If k > 0, then h[0-k:] takes the last k elements and wraps?
    #     We want to compute h[i] * (1 - h[i+k])
    #     so for h[i], the cross dependence is with h[i+k] in 1-h
    #     So:
    #     gradient w.r.t h_j of sum_i h[i] (1 - h[i+k])  
    #     = sum_i (d/dh_j h[i]) * (1 - h[i+k])
    #     = sum_i delta_{ji} (1 - h[i+k])
    #     = 1 - h[j+k] (provided 0 <=j+k < n)
    
    # But if you wrap, i+k might not be within range.
    
    # Let's rebuild.
    # Define vector for differentiation: indices go 0 to n-1
    # So derivative w.r.t h[j] of overlap_k = sum h[i] (1 - h[i+k])
    # = (1 - h[j+k]) if j+k < n & j+k >=0
    # = 0 if j+k >= n or j+k < 0
    
    # To implement in NumPy efficiently:
    # We take vectors: h (length n), and construct (1 - h) shifted by k.
    # Let's compute d/d h[i] (Overlap=k) = 1 - h[i+k] for valid k.
    
    # This means gradient is just the result of shifting (1-h) by k and then shifting it back
    #   Actually it's simpler:
    # gradient[i] = 1 - h[i+k] when i+k < n and i+k >= 0
    
    grad = np.zeros(n)
    for i in range(n):
        j = i + k_max
        if 0 <= j < n:
            grad[i] = 1 - h[j]
    
    # Normalize direction vector
    grad_norm = np.linalg.norm(grad)
    if grad_norm > 1e-12:
        grad = grad / grad_norm
    
    # For momentum acceleration: smoothly track previous gradients
    h_update = learning_rate * grad
    
    # Impose constraints:
    h_new = h - h_update
    h_new = np.clip(h_new, 0, 1)
    h_new = project_to_feasible(h_new, n/2)
    
    return h_new

    # Try to improve with a fast gradient-based method using trust region and momentum
    iter_count = 0
    learning_rate = 0.05
    momentum = 0.9
    
    while time.time() - start_time < budget_s - 1:
        if iter_count % 5 == 0:
            current_c5 = compute_c5(h, dx)
