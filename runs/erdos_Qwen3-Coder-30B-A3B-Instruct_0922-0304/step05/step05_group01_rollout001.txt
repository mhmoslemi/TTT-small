import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 20
    elite_fraction = 0.25
    mutation_strength = 0.01
    
    # Initialize population
    def initialize_population():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            h = np.ones(n_points)
            h[n_points//2:] = 0
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        population = [h.copy()]
        for _ in range(population_size - 1):
            mutant = h + np.random.normal(0, mutation_strength, n_points)
            mutant = np.clip(mutant, 0, 1)
            mutant = mutant * (n_points / 2) / np.sum(mutant)
            population.append(mutant)
        return population
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def mutate(h, strength):
        mutated = h + np.random.normal(0, strength, h.shape)
        mutated = np.clip(mutated, 0, 1)
        sum_h = np.sum(mutated)
        mutated = mutated * (n_points / 2) / sum_h
        return mutated
    
    def select_elite(population, fraction):
        fitness = [compute_c5(h) for h in population]
        indices = np.argsort(fitness)
        elite_indices = indices[:int(len(population) * fraction)]
        return [population[i] for i in elite_indices]
    
    def evaluate_fitness(population):
        return [compute_c5(h) for h in population]
    
    # Initialize population
    population = initialize_population()
    best_h = None
    best_c5 = float('inf')
    
    # Evolution loop
    max_generations = 1000
    no_improvement_count = 0
    max_no_improvement = 50
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current population
        fitness = evaluate_fitness(population)
        
        # Track best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Adjust mutation strength if no improvement
        if no_improvement_count > max_no_improvement:
            mutation_strength *= 0.1
            no_improvement_count = 0
            
        # Select elite
        elite = select_elite(population, elite_fraction)
        
        # Generate new population
        new_population = []
        for individual in elite:
            for _ in range(2):  # Create 2 offspring per elite
                offspring = mutate(individual, mutation_strength)
                new_population.append(offspring)
                
        # Ensure population size
        while len(new_population) < population_size:
            new_population.append(np.random.rand(n_points))
            new_population[-1] = new_population[-1] * (n_points / 2) / np.sum(new_population[-1])
            
        population = new_population[:population_size]
    
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
