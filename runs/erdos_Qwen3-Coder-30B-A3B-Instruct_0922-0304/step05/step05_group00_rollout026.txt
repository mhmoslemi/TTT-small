import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_feasible(h, target_sum):
    """Project h to feasible region [0,1] with target sum."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the parent construction or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)

    best_c5 = compute_c5(h, dx)
    best_h = h.copy()

    # Iterative improvement with local search
    max_iter = 5000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Identify most critical lags
        overlap = correlate(h, 1 - h, mode="full")
        # Select top lags that contribute most to the maximum overlap
        top_lags = np.argsort(overlap)[-int(n_points * 0.05):]  # Top 5%
        max_lag = len(overlap) // 2
        # Focus on lags close to the max lag for better control
        relevant_lags = [lag for lag in top_lags if abs(lag - max_lag) < n_points // 4]
    
        # Create a constrained optimization problem focusing on key overlaps
        def overlap_objective(h):
            result = correlate(h, 1 - h, mode="full")
            return np.max(result) * dx

        # Define bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        def constraint_sum(h):
            return np.sum(h) - target_sum

        # Solve with a more robust approach using trust-constr if available or SLSQP
        try:
            result = minimize(
                overlap_objective,
                h.copy(),
                method='SLSQP',
                bounds=bounds,
                constraints={'type': 'eq', 'fun': constraint_sum},
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            if result.success:
                h_refined = result.x
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = normalize_h(h_refined, target_sum)
                h = h_refined
            else:
                # Fall back to local perturbation
                h = project_feasible(h + np.random.normal(0, 0.005, n_points), target_sum)
        except Exception:
            # Fall back to simple perturbation if optimization fails
            h = project_feasible(h + np.random.normal(0, 0.005, n_points), target_sum)
        
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
