import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up problem parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: create a piecewise constant function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        # Add noise
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        
        # For perturbation, sample a random direction
        epsilon = 0.005
        perturbation = np.random.normal(0, epsilon, n_points)
        
        # Apply perturbation and maintain constraints
        new_h = h + perturbation
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Accept if better or with some probability
        new_c5 = compute_c5(new_h)
        if new_c5 < current_c5 or np.random.rand() < 0.1:
            h = new_h

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
