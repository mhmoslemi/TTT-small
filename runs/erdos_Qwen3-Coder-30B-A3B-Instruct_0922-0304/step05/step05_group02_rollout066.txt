import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

    n_points = 70

    target_sum = n_points / 2

        h = project_h(h, target_sum)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Optimization parameters
    eta = 0.1
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute correlation for all shifts to get max overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_k = max_idx - (len(h) - 1)  # Convert index to shift k
        
        # Compute gradient (simplified as derivative of overlap at max point)
        # Using shift k to compute gradient direction
        h_shifted = np.roll(h, -max_k) if max_k > 0 else np.roll(h, max_k)
        gradient = (1 - h_shifted) * dx
        
        # Gradient descent update
        h = h - eta * gradient
        
        # Project back to feasible region
        h = project_h(h, target_sum)
        
        # Compute current C5
        current_c5 = compute_c5(h, dx)
        
        # Update best if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            # Increase learning rate on improvement
            eta *= 1.01
        else:
            # Decrease learning rate on no improvement
            eta *= 0.9
            
        # Periodic perturbation to escape local minima
        if iteration % 100 == 0:
            h += np.random.normal(0, 0.01, n_points)
