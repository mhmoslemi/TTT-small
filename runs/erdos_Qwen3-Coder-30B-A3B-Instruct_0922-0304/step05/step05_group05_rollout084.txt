import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Initialize PSO parameters
    n_particles = 20
    w_start = 0.8
    w_end = 0.1
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    for i in range(n_particles):
        if i == 0:
            particle = h.copy()
        else:
            particle = h + np.random.normal(0, 0.02, n_points)
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle)
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
        p_best.append(particle.copy())
        p_best_vals.append(compute_c5(particle))

    # Global best
    best_idx = np.argmin(p_best_vals)
    g_best = p_best[best_idx].copy()
    g_best_val = p_best_vals[best_idx]
    
    # PSO main loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (p_best[i] - particles[i]) + 
                            c2 * r2 * (g_best - particles[i]))
            
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply boundary constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate particle
            current_val = compute_c5(particles[i])
            
            # Update personal best
            if current_val < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = current_val
                
                # Update global best
                if current_val < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = current_val
                    
            # Early exit if target reached
            if g_best_val <= 0.3807:
                break
        
        if g_best_val <= 0.3807:
            break
    
    return (g_best.tolist(), g_best_val, n_points)
