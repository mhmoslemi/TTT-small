import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def validate_and_repair(h_values, n_points):
    h = np.array(h_values)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def initialize_coarse(n_coarse=35):
    # Initialize with a simple step function that's already quite good
    h = np.ones(n_coarse)
    h[n_coarse//2:] = 0
    return h

def upscale(h_coarse, n_points):
    # Nearest neighbor upsampling
    factor = n_points // len(h_coarse)
    h_fine = np.repeat(h_coarse, factor)
    return h_fine

def compute_c5(h, dx=2.0 / len(h)):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def optimize(h, n_points, max_iter=200, tol=1e-8):
    dx = 2.0 / n_points
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    for iteration in range(max_iter):
        current_c5 = compute_c5(h, dx)
        result = minimize(
            lambda x: compute_c5(x, dx), 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 5, 'ftol': tol}
        )
        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            break
            
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_coarse = 35
    n_fine = 70
    
    # Validation and initialization
    if 'initial_h_values' in globals():
        h = validate_and_repair(initial_h_values, n_coarse)
    else:
        h = initialize_coarse(n_coarse)
    
    # Coarse optimization
    h_coarse = optimize(h, n_coarse, max_iter=50, tol=1e-6)
    
    # Upsample to fine scale
    h_fine = upscale(h_coarse, n_fine)
    
    # Fine optimization
    h_final = optimize(h_fine, n_fine, max_iter=200, tol=1e-8)
    
    # Final evaluation
    final_c5 = compute_c5(h_final)
    
    return (h_final.tolist(), final_c5, n_fine)
