import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize swarm
    n_particles = 20
    swarm = []
    velocities = []
    
    # Initialize particles
    if 'initial_h_values' in globals():
        best_h = np.array(initial_h_values)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    else:
        best_h = np.ones(n_points)
        best_h[n_points//2:] = 0
        best_h += np.random.normal(0, 0.05, n_points)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    best_c5 = float('inf')
    
    # Initialize swarm around best_h
    for _ in range(n_particles):
        h = best_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_vel = 0.2
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        particle_fitness = []
        for i, h in enumerate(swarm):
            fitness = compute_c5(h)
            particle_fitness.append(fitness)
            
            # Update particle's best
            if fitness < compute_c5(swarm[i]):
                pass  # Already maintaining particles for best positions, but we update global best
            
        # Update global best
        global_best_idx = np.argmin(particle_fitness)
        if particle_fitness[global_best_idx] < best_c5:
            best_c5 = particle_fitness[global_best_idx]
            best_h = swarm[global_best_idx].copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Update particles
        for i in range(n_particles):
            h = swarm[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.random(2)
            pbest = swarm[global_best_idx]  # Best particle in swarm
            v = w * v + c1 * r1 * (pbest - h) + c2 * r2 * (pbest - h)
            
            # Clip velocity
            v = np.clip(v, -max_vel, max_vel)
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Update swarm and velocities
            swarm[i] = h_new
            velocities[i] = v
            
        # Adapt inertia weight
        w = max(0.4, w - 0.001)
    
    return (best_h.tolist(), best_c5, n_points)
