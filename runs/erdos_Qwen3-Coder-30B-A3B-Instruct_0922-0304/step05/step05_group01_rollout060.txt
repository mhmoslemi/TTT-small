from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)

    else:

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def constraint_c5(h, c5_max):
        overlap = correlate(h, 1 - h, mode="full")
        return c5_max - np.max(overlap) * dx

    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Binary search approach
    c5_low = 0.0
    c5_high = 0.38085  # Current upper bound
    tolerance = 1e-4
    max_iter = 40

    # For tracking consecutive failures
    no_improvement_count = 0
    max_no_improvement = 20

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5_mid = (c5_low + c5_high) / 2.0
        
        # Check feasibility using optimization
        con = constraints + [{'type': 'ineq', 'fun': lambda x: constraint_c5(x, c5_mid)}]
        
        # Try multiple starting points to avoid local minima
        best_feasible_h = None
        best_feasible_c5 = float('inf')
        
        for _ in range(3):
            if time.time() - start_time > budget_s - 1:
                break
            # Perturb initialization
            h_try = h + np.random.normal(0, 0.02, n_points)

            h_try = h_try * (n_points / 2) / np.sum(h_try)
            
            try:
                result = minimize(
                    compute_c5, 
                    h_try, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=con,
                    options={'maxiter': 50, 'ftol': 1e-8}
                )
                
                if result.success and result.fun < best_feasible_c5:
                    best_feasible_c5 = result.fun
                    best_feasible_h = result.x.copy()
                    best_feasible_h = np.clip(best_feasible_h, 0, 1)
                    best_feasible_h = best_feasible_h * (n_points / 2) / np.sum(best_feasible_h)
                    
            except:
                continue
        
        if best_feasible_c5 < c5_mid + 1e-6 and best_feasible_c5 < best_c5:
            # Feasible, so try to improve solution further with local optimization
            try:
                result = minimize(
                    compute_c5, 
                    best_feasible_h, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=constraints,
                    options={'maxiter': 100, 'ftol': 1e-10}
                )
                
                if result.success:
                    comp_c5 = compute_c5(result.x)
                    if comp_c5 < best_c5:

                        best_h = result.x.copy()
                        best_h = np.clip(best_h, 0, 1)
                        best_h = best_h * (n_points / 2) / np.sum(best_h)
            except:
                pass
            
            c5_high = c5_mid
            no_improvement_count = 0
        else:

            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improvement_count = 0
            
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
