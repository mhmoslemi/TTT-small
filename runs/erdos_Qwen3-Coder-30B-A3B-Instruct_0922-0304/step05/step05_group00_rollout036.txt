import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base starting solution
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Main optimization loop
    iteration = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap to identify critical lags
        overlap = correlate(h, 1 - h, mode="full")
        # Get indices of top 15 largest overlaps
        top_indices = np.argsort(np.abs(overlap))[-15:][::-1]
        # Use only the highest value overlap to guide local search
        k_best = top_indices[0] - (len(overlap) - 1) // 2
        epsilon = 1e-6
        
        # Objective for local optimization is to minimize C5
        # We can refine solution in the direction of largest overlap
        def objective(h):
            return compute_c5(h, dx)
        
        # Constraint for the sum
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        # Define bounds for each h[i]
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Use SLSQP to refine the solution
        result = minimize(objective, h, method='SLSQP', bounds=bounds,

            h_new = np.clip(h_new, 0, 1)

        # If we're close to target, add random perturbation to escape local minima

        noise = np.random.normal(0, 0.01, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
