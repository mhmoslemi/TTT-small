import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Binary search for minimal C5
    c5_low = 0.0
    c5_high = 0.38085
    tolerance = 1e-4
    max_iterations = 40
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Try to improve starting point
    for _ in range(10):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Random perturbation
        h_pert = h + np.random.normal(0, 0.01, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        
        current_c5 = compute_c5(h_pert)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_pert.copy()
            
    h = best_h.copy()
    
    # Binary search process
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 2:
            break
            
        c5_mid = (c5_low + c5_high) / 2.0
        
        # Define the constraints for feasibility check
        def constraint_max_overlap(h):
            overlap = correlate(h, 1 - h, mode="full") * dx
            return c5_mid - np.max(overlap)
        
        constraints = [
            {'type': 'eq', 'fun': constraint_sum},
            {'type': 'ineq', 'fun': constraint_max_overlap}
        ]
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try to find a feasible solution
        result = minimize(
            lambda x: -np.sum(x),  # Maximize sum to try to push towards better solutions
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            c5_candidate = compute_c5(h_candidate)
            
            if c5_candidate <= c5_mid:
                # This means we found a feasible point with C5 <= c5_mid
                # Try to refine the solution
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
                c5_high = min(c5_mid, best_c5)
            else:
                c5_low = c5_mid
        else:
            # Use smaller step if optimization fails
            c5_low = c5_mid
            
        # Prevent from getting stuck
        if abs(c5_high - c5_low) < tolerance:
            break
    
    # Final local optimization
    def objective(h):
        return compute_c5(h)
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    if time.time() - start_time < budget_s - 2:
        result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        
        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_c5 = compute_c5(final_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h
    
    return (best_h.tolist(), best_c5, n_points)
