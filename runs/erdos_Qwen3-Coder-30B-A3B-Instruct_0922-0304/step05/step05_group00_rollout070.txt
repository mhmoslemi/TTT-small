import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a basic step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize to correct sum
    h = h * (target_sum / np.sum(h))
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def project_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        return h
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    max_iter = 1000
    iter_count = 0
    
    # Focus on the highest overlap lags for linearization
    lag_selection_ratio = 0.1
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Compute all overlaps
        overlap_full = correlate(h, 1 - h, mode="full")
        # The overlap values for lags k=0, 1, ..., 2*n_points-2
        n_full = len(overlap_full)
        # Only consider relevant lags (contribution to C₅)
        involved_lags = np.arange(n_full)
        # Map to actual lags (from -n+1 to n-1)
        actual_lags = involved_lags - (n_full - 1) // 2
        # Filter for meaningful lags (0 to 2)
        valid_lags = np.where((actual_lags >= 0) & (actual_lags <= 2))[0]
        
        # Select the most impacted lags
        lag_values = overlap_full[valid_lags]
        sorted_lags = np.argsort(lag_values)[-max(1, int(lag_selection_ratio * len(valid_lags))):]
        
        # Only consider these high-lag overlaps for linearization
        active_lags = valid_lags[sorted_lags[::-1]]  # Reverse to high to low
        
        # Try to optimize using projected gradient-like updates
        new_h = h.copy()
        # For each selected lag, we approximate the constraint (non-linear overlap)
        # We use the fact that for a fixed h, the overlap sum is linear in h[i+k]
        # So we approximate the full LP-like subproblem
        
        # Simple gradient-like update (projected subgradient)
        # Each lag k contributes to one constraint: 
        # Overlap_k = sum_i h[i] * (1 - h[i+k]) 
        # Gradient: for each h[i], partial_wrt_h[i] = 1 - h[i+k] or -h[i+k] 
        # In practice, to minimize C₅, we look at each lag's contribution.
        # The goal is to reduce the maximum lag overlap, so we decrease values
        # that increase overlap.
        
        # Instead of complex QP solve, we approximate with soft updates
        alpha = 0.05  # Learning rate
        
        # Compute gradient estimate, for each position i, we need gradient
        # of sum over lag = k, for all valid i: h[i] * (1 - h[i+k])
        # When we move h[i], the change in overlap_k terms depends on:
        # partial with respect to h[i] = (1 - h[i + k]) - h[i] if i + k < n_points
        # partial with respect to h[i + k] = - h[i] if i + k < n_points
        #  
        # Simplified: For each pos i, 
        #   d/dh[i] = sum over valid k where i+k < n_points of (1 - h[i + k])
        #   
        # So we get something like the gradient based on local values.
        # Since we want to minimize the max value we process in reverse update
        
        # Estimate local gradient for reducing overlaps
        grad = np.zeros_like(h)
        for k in active_lags:
            # k index corresponds to offset in lag, but since we compute overlap_full
            # it shifts positions. 
            # The actual lag = k - n_points + 1 (negative) to n_points - 1 (positive)
            # So the lag index k maps to lag l = k - (n_full - 1) // 2  
            l = k - (n_full - 1) // 2
            # lag points to a lag l = 0 to 2 here, 
            # We approximate gradient based on overlap contribution 
            for i in range(n_points):
                j = i + l
                if j >= 0 and j < n_points:
                    grad[i] += 1 - h[j]  # approx. gradient wrt h[i] for this lag
                    
        # Apply gradient descent step
        new_h = h - alpha * grad
        # Project step
        new_h = project_feasible(new_h)
        
        # Apply smoothing with a small weight to the current solution
        smooth_weight = 0.01
        h = (1 - smooth_weight) * new_h + smooth_weight * h
        
        iter_count += 1
    
    # Final check of best solution
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
