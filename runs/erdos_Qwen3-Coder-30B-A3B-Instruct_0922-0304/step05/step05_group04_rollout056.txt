import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
    else:
        # Fallback: simple step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
    
    # Initialize other particles
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
    
    # Initialize velocities
    for i in range(swarm_size):
        v = (np.random.rand(n_points) - 0.5) * 0.1
        velocities.append(v.copy())
    
    # PSO parameters
    w_max = 0.9
    w_min = 0.4
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Best positions
    particle_best = [p.copy() for p in particles]
    particle_best_values = [float('inf')] * swarm_size
    global_best = particles[0].copy()
    global_best_value = float('inf')
    
    # Main loop
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Update inertia weight
        w = w_max - (w_max - w_min) * (time.time() - start_time) / budget_s
        
        for i in range(swarm_size):
            h = particles[i]
            
            # Compute overlap
            overlap = correlate(h, 1 - h, mode="full")
            current_c5 = np.max(overlap) * dx
            
            # Update particle best
            if current_c5 < particle_best_values[i]:
                particle_best_values[i] = current_c5
                particle_best[i] = h.copy()
            
            # Update global best
            if current_c5 < global_best_value:
                global_best_value = current_c5
                global_best = h.copy()
            
            # Early exit if target achieved
            if global_best_value <= 0.3807:
                break
                
            # Update velocity and position
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            v = (w * velocities[i] + 
                 c1 * r1 * (particle_best[i] - h) + 
                 c2 * r2 * (global_best - h))
            
            # Clamp velocity
            v = np.clip(v, -v_max, v_max)
            velocities[i] = v
            
            # Update position
            h_new = h + v
            
            # Clamp and renormalize
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[i] = h_new.copy()
            
        if global_best_value <= 0.3807:
            break
    
    # Final evaluation
    overlap_final = correlate(global_best, 1 - global_best, mode="full")
    final_c5 = np.max(overlap_final) * dx
    
    return (global_best.tolist(), final_c5, n_points)
