import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 20
    max_iter = 1000
    
    # Initialize swarm
    def initialize_particle():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
        else:
            h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Initialize swarm
    swarm = [initialize_particle() for _ in range(swarm_size)]
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    
    # Track personal bests
    personal_bests = [swarm[i].copy() for i in range(swarm_size)]
    personal_best_scores = [compute_c5(swarm[i]) for i in range(swarm_size)]
    
    # Track global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_bests[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO parameters
    inertia = 0.7
    cognitive_weight = 1.5
    social_weight = 1.5
    max_velocity = 0.1
    
    # PSO iteration
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia
        inertia = max(0.4, inertia - 0.0001)
        
        for i in range(swarm_size):
            # Evaluate current particle
            current_score = compute_c5(swarm[i])
            
            # Update personal best
            if current_score < personal_best_scores[i]:
                personal_bests[i] = swarm[i].copy()
                personal_best_scores[i] = current_score
                
                # Update global best
                if current_score < global_best_score:
                    global_best = swarm[i].copy()
                    global_best_score = current_score
                    
            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive = cognitive_weight * r1 * (personal_bests[i] - swarm[i])
            social = social_weight * r2 * (global_best - swarm[i])
            velocities[i] = inertia * velocities[i] + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            swarm[i] = swarm[i] + velocities[i]
            
            # Ensure feasibility
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Early termination if target achieved
            if global_best_score <= 0.3807:
                break
    
    return (global_best.tolist(), global_best_score, n_points)
