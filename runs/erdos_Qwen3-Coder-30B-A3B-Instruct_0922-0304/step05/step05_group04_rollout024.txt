from scipy.signal import correlate
import time

        h = h * (n_points / 2) / np.sum(h)

        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    binary_h = np.clip(binary_h, 0, 1)
    binary_h = binary_h * (n_points / 2) / np.sum(binary_h)
    
    # Annealing approach for binary search
    def get_neighbor(h):
        h_new = h.copy()
        pos1 = np.random.randint(0, n_points)
        pos2 = np.random.randint(0, n_points)
        while pos1 == pos2:
            pos2 = np.random.randint(0, n_points)
        h_new[pos1], h_new[pos2] = h_new[pos2], h_new[pos1]
        return h_new

    def binary_cost(h):
        h_binary = (h > 0.5).astype(int)
        # Ensure correct sum
        current_sum = np.sum(h_binary)
        target_sum = n_points // 2
        if current_sum > target_sum:
            # Flip from 1 to 0
            ones_idx = np.where(h_binary == 1)[0]
            flip_indices = np.random.choice(ones_idx, current_sum - target_sum, replace=False)
            for idx in flip_indices:
                h_binary[idx] = 0
        elif current_sum < target_sum:
            # Flip from 0 to 1
            zeros_idx = np.where(h_binary == 0)[0]
            flip_indices = np.random.choice(zeros_idx, target_sum - current_sum, replace=False)
            for idx in flip_indices:
                h_binary[idx] = 1
        return h_binary

    min_temp = 1e-4

        neighbor = get_neighbor(h)

        new_c5 = compute_c5(neighbor)

            grads = []

                h_plus = h.copy()
                h_minus = h.copy()

                h_plus[i] += eps
                h_minus[i] -= eps
                
                # Ensure bounds and normalization
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)

                val_plus = loss_func(h_plus)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
