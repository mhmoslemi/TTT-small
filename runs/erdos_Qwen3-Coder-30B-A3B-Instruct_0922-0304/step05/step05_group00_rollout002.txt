import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint and bounds."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Optimization loop
    iteration = 0
    max_iter = 500  # Prevent excessive iterations
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full")
        # Identify top lags contributing to max overlap
        top_lags = np.argsort(overlap)[-max(1, n_points // 20):]
        
        # Convex relaxation approach
        def objective(h):
            return compute_c5(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        # Bounds for optimization
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Use SLSQP for refined local optimization with constraints
        result = minimize(objective, h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
        
        if result.success:
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
            h = h_new
        else:
            # Fallback: small random perturbation
            noise = np.random.normal(0, 0.005, n_points)
            h = h + noise
            h = project_to_feasible(h, target_sum)
        
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
        
    return (best_h.tolist(), best_c5, n_points)
