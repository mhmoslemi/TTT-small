import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    """Evaluate objective and check constraints."""
    c5 = compute_c5(h, dx)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Starting with a simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Differential evolution inspired search
    population_size = 30
    population = [h.copy()]
    for _ in range(population_size - 1):
        candidate = h + np.random.normal(0, 0.05, n_points)
        candidate = np.clip(candidate, 0, 1)
        candidate = normalize_h(candidate, target_sum)
        population.append(candidate)
    
    # Parameters
    step_sizes = [0.1] * population_size
    step_decay = 0.98
    max_iter = 10000
    
    for iteration in range(max_iter):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Evaluate all population members
        fitness = []
        for h_pop in population:
            fit = compute_c5(h_pop, dx)
            fitness.append(fit)
        
        # Track best
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_h = population[min_fit_idx].copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Generate new population
        new_population = []
        
        # Keep best
        new_population.append(best_h.copy())
        
        # Differential evolution style mutation
        for i in range(1, population_size):
            # Select three different individuals
            indices = np.random.choice(population_size, 3, replace=False)
            a, b, c = population[indices[0]], population[indices[1]], population[indices[2]]
            
            # Mutation
            mutant = a + 0.8 * (b - c)
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            crossover_prob = 0.7
            mask = np.random.random(n_points) < crossover_prob
            trial = np.where(mask, mutant, population[i])
            
            # Repair
            trial = normalize_h(trial, target_sum)
            new_population.append(trial)
        
        # Add some gradient-based local search
        for i in range(1, min(5, len(new_population))):
            h_candidate = new_population[i].copy()
            
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h):
                return compute_c5(h, dx)
            
            # Small gradient-based local search
            result = minimize(objective, h_candidate, method='SLSQP', 
                            bounds=bounds, constraints=constraint, 
                            options={'maxiter': 20, 'ftol': 1e-6, 'gtol': 1e-6})
            if result.success:
                h_refined = result.x
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = normalize_h(h_refined, target_sum)
                new_population[i] = h_refined
        
        # Add some random mutations
        for _ in range(3):
            rand_h = np.random.random(n_points)
            rand_h = np.clip(rand_h, 0, 1)
            rand_h = normalize_h(rand_h, target_sum)
            new_population.append(rand_h)
        
        # Keep only the best ones
        fitness = [compute_c5(h, dx) for h in new_population]
        sorted_indices = np.argsort(fitness)
        population = [new_population[i] for i in sorted_indices[:population_size]]
        
        # Update step sizes
        step_sizes = [s * step_decay for s in step_sizes]
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
