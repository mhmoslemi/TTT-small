import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    """Evaluate objective and check constraints."""
    c5 = compute_c5(h, dx)
    return c5

def local_refine(h, dx, target_sum):
    """Refine solution using local optimization."""
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    def objective(h):
        return compute_c5(h, dx)
        
    bounds = [(0, 1) for _ in range(len(h))]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
    if result.success:
        h_refined = result.x
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = normalize_h(h_refined, target_sum)
        return h_refined
    else:
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize with best known construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a symmetric initial guess
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        # Evaluate current solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Do local refinement to improve current solution
        h = local_refine(h, dx, target_sum)
        
        # Add noise to escape local optima
        noise = np.random.normal(0, 0.005, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Occasionally do more exploratory moves
        if iteration % 10 == 0:
            # Random perturbation
            perturb_indices = np.random.choice(n_points, size=n_points//10, replace=False)
            h[perturb_indices] += np.random.normal(0, 0.02, len(perturb_indices))
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)

    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
