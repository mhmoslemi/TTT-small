import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def repair_feasibility(h, target_sum):
    """Repair h to satisfy constraints."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def gradient_descent_step(h, dx, target_sum, step_size=0.01):
    """Perform gradient descent step on h."""
    # Compute gradient approximation using finite differences
    n = len(h)
    grad = np.zeros(n)
    eps = 1e-4
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = repair_feasibility(h_plus, target_sum)
        c5_plus = compute_c5(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = repair_feasibility(h_minus, target_sum)
        c5_minus = compute_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * eps)
    
    # Move in descent direction
    h_new = h - step_size * grad
    h_new = repair_feasibility(h_new, target_sum)
    return h_new

def differential_evolution_step(population, target_sum, dx, F=0.8, Cr=0.9):
    """Generate a new candidate via differential evolution."""
    a, b, c = np.random.choice(len(population), 3, replace=False)
    mutant = population[a] + F * (population[b] - population[c])
    trial = np.where(np.random.random(len(mutant)) < Cr, mutant, population[a])
    trial = repair_feasibility(trial, target_sum)
    return trial

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 30
    population = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    
    # Initialize population with diverse samples
    for _ in range(population_size):
        h = base_h.copy()
        noise_level = 0.02 + 0.03 * np.random.random()
        h += np.random.normal(0, noise_level, len(h))
        h = repair_feasibility(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_individual = None
    step_size = 0.01
    iteration = 0
    
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate entire population
        fitness = [compute_c5(h, dx) for h in population]
        
        # Track best individual so far
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_individual = population[min_fit_idx].copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Generate new population
        new_population = []
        
        # Elitism: keep the best individual
        new_population.append(best_individual.copy())
        
        # Selection based on fitness (tournament selection)
        tournament_size = 5
        selected_indices = []
        for _ in range(population_size - 1):
            tournament = np.random.choice(len(population), tournament_size)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected_indices.append(winner)
        
        selected_pop = [population[i] for i in selected_indices]
        
        # Create offspring
        for _ in range(population_size - 1):
            choice = np.random.random()
            if choice < 0.4:  # Differential evolution
                new_h = differential_evolution_step(selected_pop, target_sum, dx)
            elif choice < 0.7:  # Gradient descent
                new_h = gradient_descent_step(selected_pop[np.random.randint(len(selected_pop))], dx, target_sum, step_size)
            elif choice < 0.9:  # Random mutation of base
                new_h = base_h.copy()
                new_h += np.random.normal(0, 0.01, len(new_h))
                new_h = repair_feasibility(new_h, target_sum)
            else:  # Mutation of best individual
                new_h = best_individual.copy()
                noise_level = 0.01 + 0.02 * np.random.random()
                new_h += np.random.normal(0, noise_level, len(new_h))
                new_h = repair_feasibility(new_h, target_sum)
            
            new_population.append(new_h)
            
        # Update population
        population = new_population[:population_size]
            
        # Adaptive step size adjustment
        if iteration % 10 == 0:
            step_size *= 0.98
        
        iteration += 1
    
    final_h = best_individual
    final_c5 = compute_c5(final_h, dx)
    return (final_h.tolist(), final_c5, n_points)
