import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
    
    # Normalize to correct sum
    current_sum = np.sum(h)
    h = h * (target_sum / current_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Grid search with refinement
    # Try different structures: uniform, step, and smooth variations
    candidates = []
    candidates.append(h.copy())
    
    # Add some variations
    for i in range(10):
        h_perturbed = h + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = np.sum(h_perturbed)
        h_perturbed = h_perturbed * (target_sum / current_sum)
        candidates.append(h_perturbed.copy())
    
    # Block optimization with local search
    def objective(h):
        return evaluate_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    # Local optimization on all candidates
    for c in candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        from scipy.optimize import minimize
        result = minimize(
            objective,
            c,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            current_sum = np.sum(h_opt)
            h_opt = h_opt * (target_sum / current_sum)
            
            c5 = objective(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                
        # Early exit if good enough
        if best_c5 <= 0.3807:
            break
        
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
