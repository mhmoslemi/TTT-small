import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        h = np.clip(base_h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use a more systematic local search
    iteration = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(np.abs(overlap))
        max_k = max_overlap_idx - (len(overlap) // 2)
        
        # Identify critical lags near maximum
        critical_lags = []
        if max_k > 0:
            # Include max_k and nearby lags
            for k in range(max_k - 3, max_k + 4):
                if k != 0 and -n_points < k < n_points:
                    critical_lags.append(k)
        else:
            for k in range(max_k - 3, max_k + 4):
                if k != 0 and -n_points < k < n_points:
                    critical_lags.append(k)

        # Optimization bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        # Define the objective: minimize C5
        def objective(h):
            return compute_c5(h, dx)
            
        # Apply local optimization with constraints
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Use SLSQP for local refinement
        result = minimize(objective, h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
        
        if result.success:
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            h = h_new.copy()

        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        # Perturb slightly for escape from local minima
        if iteration % 10 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
