```python
import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: clip to [0,1] and scale to sum=target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_h(h, target_sum)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Parameters
    eta = 0.1
    decay_factor = 0.99
    noise_std = 0.01
    noise_interval = 100
    
    t = 0
    while time.time() - start_time < budget_s - 1:
        # Compute cross-correlation for all shifts
        cross_corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(cross_corr)
        max_k = max_idx - (len(h) - 1)
        
        # Compute gradient for the worst overlap
        # For shift k, gradient w.r.t h_i = (1 - h_{i+k}) * dx
        # Build gradient vector
        grad = np.zeros(len(h))
        if max_k >= 0:
            grad[max_k:] = -1 * (1 - h[:-max_k])
        else:
            grad[:max_k] = -1 * (1 - h[-max_k:])
        
        # Gradient descent update
        h = h - eta * grad
        
        # Project back to feasible region
        h = project_h(h, target_sum)
        
        # Periodic noise perturbation
        if t % noise_interval == 0:
            noise = np.random.normal(0, noise_std, len(h))
            h = h + noise
            h = project_h(h, target_sum)
        
        # Compute current C5
        current_c5 = compute_c5(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            # Adjust learning rate
            eta = min(1.0, eta * 1.01)
        else:
            # Reduce learning rate if no improvement
            eta *= decay_factor
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        t += 1
    
    return (best_h.tolist(), best_c5, n_points)
```
