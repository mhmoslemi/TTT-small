import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have sum equal to target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from provided construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
    else:
        # Start with a simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, n_points / 2)

    # Optimization parameters
    momentum_decay = 0.9
    learning_rate = 0.1
    line_search_tolerance = 1e-3
    max_line_search_iterations = 20
    min_improvement = 1e-8
    
    # Initialize momentum
    m = np.zeros_like(h)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute all correlations and find max
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_val = overlap[max_idx]
        
        # Find the corresponding shift k
        k = max_idx - (n_points - 1)
        
        # Compute gradient of the overlap w.r.t. h
        # The maximum overlap occurs at shift k
        # Gradient w.r.t h[i] is: contribution from 1-h[i+k] for i+k in valid range
        # But we need to carefully compute the effect of small perturbations
        # For now, we'll compute gradient approximately by finite difference
        epsilon = 1e-5
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_pert = h.copy()
            h_pert[i] += epsilon
            h_pert = normalize_h(h_pert, n_points / 2)
            overlap_pert = correlate(h_pert, 1 - h_pert, mode="full")
            grad[i] = (overlap_pert[max_idx] - max_val) / epsilon
        
        # Update momentum
        m = momentum_decay * m + (1 - momentum_decay) * grad
        
        # Perform gradient descent with momentum
        h_new = h - learning_rate * m
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, n_points / 2)
        
        # Line search to ensure improvement
        step_size = learning_rate
        for _ in range(max_line_search_iterations):
            h_candidate = h - step_size * m
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = normalize_h(h_candidate, n_points / 2)
            c5_candidate = compute_c5(h_candidate, dx)
            
            if c5_candidate <= best_c5 - min_improvement or step_size < 1e-8:
                h = h_candidate
                break
            step_size *= 0.5
        
        # Update best solution
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Adaptive learning rate based on progress
        if iteration % 10 == 0:
            if abs(c5 - best_c5) < min_improvement:
                learning_rate *= 0.9
            else:
                learning_rate = min(learning_rate * 1.05, 1.0)
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
