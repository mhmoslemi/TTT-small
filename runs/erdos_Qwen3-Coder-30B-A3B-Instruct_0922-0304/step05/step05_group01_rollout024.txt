import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 20
    max_iter = 1000
    
    # Initialize swarm
    def init_particle():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
        else:
            h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    swarm = [init_particle() for _ in range(swarm_size)]
    velocities = [np.random.random(n_points) * 0.1 for _ in range(swarm_size)]
    personal_best = [p.copy() for p in swarm]
    personal_best_scores = [float('inf')] * swarm_size
    
    # Global best
    global_best = init_particle()
    global_best_score = float('inf')
    
    # PSO parameters
    w = 0.7
    c1 = c2 = 1.5
    max_velocity = 0.5
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update swarm
        for i in range(swarm_size):
            # Evaluate fitness
            fitness = compute_c5(swarm[i])
            
            # Update personal best
            if fitness < personal_best_scores[i]:
                personal_best_scores[i] = fitness
                personal_best[i] = swarm[i].copy()
            
            # Update global best
            if fitness < global_best_score:
                global_best_score = fitness
                global_best = swarm[i].copy()
                
            # Early exit
            if global_best_score <= 0.3807:
                break
                
            # Update velocity
            r1 = np.random.random(n_points)
            r2 = np.random.random(n_points)
            cognitive = c1 * r1 * (personal_best[i] - swarm[i])
            social = c2 * r2 * (global_best - swarm[i])
            
            velocities[i] = w * velocities[i] + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            swarm[i] += velocities[i]
            
            # Ensure feasibility
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
        # Update inertia weight
        w = max(0.4, w - 0.001)
        
        # Early exit if target achieved
        if global_best_score <= 0.3807:
            break
    
    return (global_best.tolist(), global_best_score, n_points)
