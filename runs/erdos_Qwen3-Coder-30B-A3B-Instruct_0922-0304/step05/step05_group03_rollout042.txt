import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

    start_time = time.time()

    target_sum = n_points / 2

    def compute_c5(h):
        return evaluate_c5(h, dx)
    
    # Initialize best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Use a hybrid approach: perturb + local optimization
    max_iter = 10000
    iter_count = 0
    
    # Adaptive perturbation and optimization
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        # 1. Perturb
        h_perturbed = h + np.random.normal(0, 0.02, n_points)
        h_perturbed = normalize_h(h_perturbed, target_sum)
        
        # 2. Local optimization on perturbed version

            return compute_c5(h_local)
            
        def constraint_sum(h_local):
            return np.sum(h_local) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        result = minimize(
            objective,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,

        if result.success:

    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
