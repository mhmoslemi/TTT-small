import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        # Random initialization
        h = np.random.rand(n_points)
        h = project_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 1000
    patience = 50
    patience_counter = 0
    last_c5 = float('inf')
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Update best
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient
        # For shift k, gradient is (1 - h_{k+j}) at position j
        # This is a complex operation; we'll compute it by shifting properly
        # For efficiency, we compute gradient at the position associated with max overlap
        if max_k >= n_points:
            # This means that the maximum overlap is from a positive shift
            # overlap[k] corresponds to h shifted by k - n_points
            shift = max_k - n_points + 1  # shift amount
            # Element-wise gradient: we want (1 - h_shifted)
            gradient = np.zeros_like(h)
            if shift == 0:
                gradient = 1 - h
            elif shift > 0:
                # h shifted right
                gradient[shift:] = 1 - h[:-shift]
                gradient[:shift] = 1  # arbitrary, outside support
            else:
                # shift < 0, h shifted left
                gradient[:shift] = 1 - h[-shift:]
                gradient[shift:] = 1  # arbitrary, outside support
        else:
            # Negative shift, overlap in h[] * (1 - h_{k+j}) for k negative 
            # This is mathematically more complex
            # For now using approximate gradient: just the max overlap gradient
            # which should be: (1 - h_{k+j}) for all j in the support  
            # Let's align it back and follow main flow:
            shift = max_k - n_points + 1  # shift such that h[j] * (1 - h[j+shift])
            
            gradient = np.zeros_like(h)
            gradient[:min(n_points, max_k+1)] = 1  # approximate
            
            # Compute real shifts
            if shift >= 0:  # rightshift
                grad_idx = np.arange(shift, min(n_points, n_points-shift))
                gradient[grad_idx] = 1 - h[grad_idx-shift]
            else:  # leftshift
                grad_idx = np.arange(-shift, n_points)
                gradient[grad_idx] = 1 - h[grad_idx+shift]
        
        # Update h
        h_new = h - learning_rate * gradient
        
        # Project
        h = project_h(h_new, target_sum)
        
        # Check convergence
        if abs(max_overlap - last_c5) < 1e-8:
            patience_counter += 1
            if patience_counter >= patience:
                break
        else:
            patience_counter = 0
        last_c5 = max_overlap
        
        # Adaptive learning rate
        learning_rate = max(0.001, learning_rate * 0.999)
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
