```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def local_search(h, dx, max_iter=100):
    """Refine solution using local search (trust-constr)."""
    def objective(h_vec):
        return compute_c5(h_vec, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - len(h_vec) / 2
    
    bounds = [(0, 1) for _ in range(len(h))]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    result = minimize(
        objective, 
        h, 
        method='trust-constr', 
        bounds=bounds, 
        constraints=constraints,
        options={'maxiter': max_iter, 'gtol': 1e-8, 'xtol': 1e-8}
    )
    
    if result.success:
        h_new = result.x
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_feasible(h_new, len(h_new) / 2)
        return h_new
    else:
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h = project_to_feasible(h, n_points / 2)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iter_count = 0
    perturbation_magnitude = 0.1
    
    while time.time() - start_time < budget_s - 1:
        # Local search to improve current solution
        h = local_search(h, dx, max_iter=100)
        
        # Compute C5
        c5 = compute_c5(h, dx)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Periodic perturbation
        if iter_count % 50 == 0 and iter_count > 0:
            noise = np.random.normal(0, perturbation_magnitude, n_points)
            h = h + noise
            h = project_to_feasible(h, n_points / 2)
            perturbation_magnitude *= 0.95
            
        iter_count += 1
    
    # Final evaluation of best solution
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
```
