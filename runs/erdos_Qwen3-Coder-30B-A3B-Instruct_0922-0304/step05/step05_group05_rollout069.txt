import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: piecewise function with 3 regions
        h = np.ones(n_points)
        mid1 = n_points // 4
        mid2 = 3 * n_points // 4
        h[mid1:mid2] = 0.25
        h[mid2:] = 0.5
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def rescale_h(h):
        return h * (n_points / 2) / np.sum(h)
    
    def accept_swap( old_c5, new_c5, temperature ):
        if new_c5 < old_c5:
            return True
        return np.random.rand() < np.exp((old_c5 - new_c5) / temperature)

    # Diffusion-based local search
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    # Try to improve with swap-based local search
    n_swaps = 1000
    swap_size = 5
    temperature = 0.1
    
    for _ in range(n_swaps):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Create a copy of current h
        h_trial = best_h.copy()
        
        # Select two non-overlapping segments
        pos1 = np.random.randint(0, n_points - swap_size)
        pos2 = np.random.randint(0, n_points - swap_size)
        while abs(pos1 - pos2) < swap_size:
            pos2 = np.random.randint(0, n_points - swap_size)
        
        # Swap entire segments
        segment1 = h_trial[pos1:pos1+swap_size].copy()
        segment2 = h_trial[pos2:pos2+swap_size].copy()
        
        h_trial[pos1:pos1+swap_size] = segment2
        h_trial[pos2:pos2+swap_size] = segment1
        
        # Rescale to ensure integral constraint
        h_trial = rescale_h(h_trial)
        
        # Compute new overlap
        new_c5 = compute_c5(h_trial)
        
        # Accept or reject
        if accept_swap(best_c5, new_c5, temperature):
            best_h = h_trial
            best_c5 = new_c5
        
        # Early exit if optimal is found
        if best_c5 <= 0.3807:
            break

    # Final polishing with local optimization
    # Though SLSQP is more expensive, we'll do a final check
    from scipy.optimize import minimize
    def objective(h):
        return compute_c5(h)
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    
    try:
        result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h = rescale_h(final_h)
            final_c5 = compute_c5(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass  # local optimization failed, continue with best

    return (best_h.tolist(), best_c5, n_points)
