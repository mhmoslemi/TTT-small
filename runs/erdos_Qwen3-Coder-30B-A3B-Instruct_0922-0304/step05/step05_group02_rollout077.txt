import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have sum equal to target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create initial guess
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)

    # Track best solution
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Gradient-based optimization
    max_iter = 10000
    step_size = 0.01
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_overlap = overlap[max_idx] * dx
        
        # Find shift that gives maximum overlap
        k = max_idx - (len(h) - 1)  # Convert from full correlation index to shift
        
        # Compute gradient w.r.t. shift k
        grad = np.zeros_like(h)
        if k >= 0:
            grad[k:] += (1 - h[:-k]) * dx
        else:
            grad[:k] += (1 - h[-k:]) * dx
            
        # Update h with gradient descent
        h = h - step_size * grad
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Compute current C5
        current_c5 = compute_c5(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early stopping if target achieved
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
