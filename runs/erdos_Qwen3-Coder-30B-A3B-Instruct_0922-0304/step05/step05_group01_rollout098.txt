import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 20
    max_iter = 1000
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    for _ in range(population_size):
        h = h_init.copy()
        # Add variation
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h)
        velocities.append(np.zeros(n_points))
    
    # Track best solution
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    # PSO parameters
    w_start = 0.7
    w_end = 0.4
    c1 = c2 = 1.5
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness and update personal bests
        for i in range(population_size):
            h = swarm[i]
            fitness = compute_c5(h)
            
            if fitness < best_c5:
                best_c5 = fitness
                best_h = h.copy()
                
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
                
        # Update swarm
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(population_size):
            h = swarm[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.random(2)
            pbest = h_init if i == 0 else swarm[0]  # placeholder pbest - we don't maintain it explicitly
            v = w * v + c1 * r1 * (pbest - h) + c2 * r2 * (best_h - h)
            
            # Clip velocity
            max_vel = 0.1
            v = np.clip(v, -max_vel, max_vel)
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            swarm[i] = h_new
            velocities[i] = v
            
        # Early exit check
        if best_c5 <= 0.3807:
            break
            
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
