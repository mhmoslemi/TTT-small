import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    max_generations = 1000
    mutation_rate = 0.1
    
    # Initialize population
    def initialize_population():
        population = []
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h.copy())
        else:
            # Initialize with random individuals
            for _ in range(population_size):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                population.append(h.copy())
        return population
    
    def compute_c5(h):
        # Compute overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap) * dx
        return max_overlap
    
    def tournament_selection(population, fitnesses, k=10):
        """Select two individuals via tournament selection"""
        selected = []
        for _ in range(2):
            tournament_indices = np.random.choice(len(population), k)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
            selected.append(population[winner_idx].copy())
        return selected[0], selected[1]
    
    def uniform_crossover(parent1, parent2):
        """Perform uniform crossover between two parents"""
        mask = np.random.rand(n_points) > 0.5
        child = np.where(mask, parent1, parent2)
        return child
    
    def mutate(individual):
        """Apply mutation to an individual"""
        mutated = individual.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                # Randomly decide mutation type
                if np.random.rand() < 0.5:
                    # Add perturbation
                    mutated[i] += np.random.normal(0, 0.1)
                else:
                    # Set to extreme value
                    mutated[i] = np.random.choice([0.0, 1.0])
        # Clamp and renormalize
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return mutated
    
    # Initialize
    population = initialize_population()
    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    
    # Main loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = [compute_c5(individual) for individual in population]
        
        # Track best solution
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Selection and reproduction
        new_population = []
        
        # Keep top performers
        top_indices = np.argsort(fitnesses)[:population_size // 2]
        for idx in top_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1, parent2 = tournament_selection(population, fitnesses)
            child = uniform_crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
            
        # Ensure correct size
        population = new_population[:population_size]
    
    return (best_h.tolist(), best_c5, n_points)
