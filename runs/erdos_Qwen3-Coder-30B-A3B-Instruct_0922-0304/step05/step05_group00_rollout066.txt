import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    """Evaluate objective and check constraints."""
    c5 = compute_c5(h, dx)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        best_h = np.array(initial_h_values)
        best_h = np.clip(best_h, 0, 1)
        best_h = normalize_h(best_h, target_sum)
    else:
        # Create a simple initial guess
        best_h = np.ones(n_points)
        best_h[n_points//2:] = 0
        best_h = np.clip(best_h, 0, 1)
        best_h = normalize_h(best_h, target_sum)
    
    best_c5 = compute_c5(best_h, dx)
    
    # Differential evolution parameters
    population_size = 30
    mutation_rate = 0.8
    crossover_rate = 0.7
    max_generations = 200
    
    # Initialize population
    population = [best_h.copy()]
    for _ in range(population_size - 1):
        h = best_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    step_size = 0.02
    step_decay = 0.98
    improvement_count = 0
    
    for generation in range(max_generations):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Evaluate population
        fitness = [evaluate_individual(h, dx) for h in population]
        
        # Update best solution
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_h = population[min_fit_idx].copy()
            improvement_count = 0
        else:
            improvement_count += 1
            if improvement_count > 10:
                step_size *= step_decay
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        # Create new population
        new_population = [best_h.copy()]
        
        # Differential evolution
        for _ in range(population_size - 1):
            # Select three random individuals
            idxs = np.random.choice(len(population), 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            mutant = a + mutation_rate * (b - c)
            mutant = np.clip(mutant, 0, 1)
            
            # Crossover
            cross_points = np.random.rand(n_points) < crossover_rate
            if not np.any(cross_points):
                cross_points[np.random.randint(0, n_points)] = True
            trial = np.where(cross_points, mutant, population[np.random.randint(len(population))])
            
            # Repair
            trial = normalize_h(trial, target_sum)
            
            # Selection
            trial_fitness = evaluate_individual(trial, dx)
            old_fitness = fitness[np.random.randint(len(population))]
            
            if trial_fitness < old_fitness:
                new_population.append(trial)
            else:
                new_population.append(population[np.random.randint(len(population))])
        
        # Add some gradient-based improvements
        for _ in range(population_size // 3):
            h = np.random.choice(population)
            # Simple gradient descent step
            h_new = h + np.random.normal(0, step_size, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            new_population.append(h_new)
        
        population = new_population[:population_size]
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
