import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to satisfy sum(h) = target_sum while clipping to [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def compute_gradients(h, dx, k_max):
    """Compute gradients for the worst shift k_max"""
    n = len(h)
    # Compute cross-correlation for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # We only compute gradient for the specific shift
    grad = np.zeros(n)
    # Compute gradient for shift k_max in forward direction
    for i in range(n):
        if i + k_max < n:
            grad[i] += -1.0  # derivative of h[i] with respect to (1-h[i+k_max])
    return grad * dx

def line_search(h, grad, max_step=0.1, c1=1e-4, max_iter=10):
    """Backtracking line search for step size selection"""
    alpha = max_step
    for _ in range(max_iter):
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, len(h) / 2)
        if np.sum(h_new) > 0:
            break
        alpha *= 0.5
    return alpha
