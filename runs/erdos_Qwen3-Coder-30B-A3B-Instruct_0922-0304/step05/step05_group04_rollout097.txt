import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initialize with alternating pattern
        h = np.zeros(n_points)
        h[::2] = 1
        h = h * (n_points / 2) / np.sum(h)

    # Binary search phase
    def binary_search():
        best_h = h.copy()
        best_c5 = compute_c5(best_h, dx)
        temp = 1.0
        cooling_rate = 0.99
        max_inner_iter = 1000
        
        while temp > 1e-4 and time.time() - start_time < budget_s - 10:
            for _ in range(max_inner_iter):
                if time.time() - start_time > budget_s - 10:
                    break
                # Random neighbor swap
                idx1, idx2 = np.random.choice(n_points, 2, replace=False)
                new_h = best_h.copy()
                new_h[idx1], new_h[idx2] = new_h[idx2], new_h[idx1]
                
                # Normalize
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                
                new_c5 = compute_c5(new_h, dx)
                
                if new_c5 < best_c5 or np.random.rand() < np.exp(-(new_c5 - best_c5) / temp):
                    best_h = new_h
                    best_c5 = new_c5
                    
                if best_c5 <= 0.3807:
                    return best_h, best_c5
                    
            temp *= cooling_rate
            
        return best_h, best_c5
    
    # Perform binary search
    h, c5 = binary_search()
    
    # Continuous relaxation phase
    def continuous_relaxation():
        # Convert to binary approximation
        binary_h = (h > 0.5).astype(float)
        binary_h = binary_h * (n_points / 2) / np.sum(binary_h)
        c5_val = compute_c5(binary_h, dx)
        h = binary_h
        
        # Use CVXPY for continuous optimization
        h_var = cp.Variable(n_points)
        
        # Set up constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Objective: minimize maximum overlap
        # We'll compute the constraint based on current h
        # For simplicity, we'll solve a decoupled version first
        
        # Create and solve problem using ECOS
        try:
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                return h_new, compute_c5(h_new, dx)
        except:
            pass
            
        return h, c5_val
    
    # Attempt continuous relaxation
    if time.time() - start_time < budget_s - 10:
        h, c5 = continuous_relaxation()
    
    # Final refinement using gradient approach
    def gradient_refinement():
        max_iter_local = 500
        learning_rate = 0.01
        
        h_ref = h.copy()
        for _ in range(max_iter_local):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute gradient approximation 
            g = np.zeros(n_points)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h_ref.copy()
                h_minus = h_ref.copy()
                
                h_plus[i] += eps
                h_minus[i] -= eps
                
                # Clamp and normalize
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
                overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps) * dx
                
            # Update
            new_h = h_ref - learning_rate * g
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            # Accept if improvement
            if compute_c5(new_h, dx) < compute_c5(h_ref, dx):
                h_ref = new_h
                
        return h_ref, compute_c5(h_ref, dx)
    
    if time.time() - start_time < budget_s - 5:
        try:
            h, c5 = gradient_refinement()
        except:
            pass
            
    return (h.tolist(), c5, n_points)
