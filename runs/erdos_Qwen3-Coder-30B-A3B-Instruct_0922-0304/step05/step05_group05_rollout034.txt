import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # PSO parameters
    n_particles = 20
    max_iter = 1000
    w_start = 0.9
    w_end = 0.3
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize particles and velocities
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    # Create initial particles
    for i in range(n_particles):
        if i == 0:
            # First particle is the initial solution
            h_particle = h.copy()
        else:
            # Perturb the initial solution
            h_particle = h + np.random.normal(0, 0.02, n_points)
            h_particle = np.clip(h_particle, 0, 1)
            h_particle = h_particle * (n_points / 2) / np.sum(h_particle)
        
        particles.append(h_particle.copy())
        velocities.append(np.zeros(n_points))
        p_best.append(h_particle.copy())
        p_best_vals.append(compute_c5(h_particle))

    # Global best
    best_idx = np.argmin(p_best_vals)
    g_best = p_best[best_idx].copy()
    g_best_val = p_best_vals[best_idx]
    
    best_h = g_best.copy()
    best_c5 = g_best_val
    
    # PSO loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * (iter_count / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            cognitive = c1 * r1 * (p_best[i] - particles[i])
            social = c2 * r2 * (g_best - particles[i])
            velocities[i] = w * velocities[i] + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate fitness
            fitness = compute_c5(particles[i])
            
            # Update personal best
            if fitness < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = fitness
                
                # Update global best
                if fitness < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = fitness
                    
                    # Update best solution found
                    if g_best_val < best_c5:
                        best_c5 = g_best_val
                        best_h = g_best.copy()
            
            # Early exit
            if g_best_val <= 0.3807:
                break
    
    return (best_h.tolist(), best_c5, n_points)
