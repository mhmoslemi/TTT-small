import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def validate_and_repair(h_values, n_points):
    h = np.array(h_values)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def initialize_coarse(n_coarse=35):
    # Initialize with a simple step function
    h = np.ones(n_coarse)
    h[n_coarse//2:] = 0
    return h

def upscale(h_coarse, n_points):
    # Upsample using nearest neighbor interpolation
    factor = n_points // len(h_coarse)
    h_fine = np.repeat(h_coarse, factor)
    return h_fine

def compute_c5(h, dx=2.0/len(h)):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def optimize(h, n_points, max_iter=200, tol=1e-8):
    dx = 2.0 / n_points
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    def objective(h):
        return compute_c5(h, dx)
    
    result = minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': max_iter, 'ftol': tol}
    )
    
    if result.success:
        h = result.x
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

    # Coarse optimization
    n_coarse = 35
    h_coarse = h[::2]  # Take every other point to get coarse resolution
    h_coarse = validate_and_repair(h_coarse, n_coarse)
    
    h_coarse = optimize(h_coarse, n_coarse, max_iter=50, tol=1e-6)
    
    # Upsample to fine resolution
    h_fine = upscale(h_coarse, n_points)
    
    # Fine optimization
    h_final = optimize(h_fine, n_points, max_iter=200, tol=1e-8)
    
    # Final evaluation
    final_c5 = compute_c5(h_final, dx)
    
    # Ensure it meets constraints
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    return (h_final.tolist(), final_c5, n_points)
