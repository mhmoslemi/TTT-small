import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing with adaptive cooling
    temperature = 1.0
    min_temperature = 1e-6
    cooling_rate = 0.999
    iteration = 0
    max_iter = 50000
    
    # Track recent good solutions
    recent_solutions = []
    max_recent = 10
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        # Perturb current solution
        h_new = h + np.random.normal(0, 0.03, n_points)
        h_new = normalize_h(h_new, target_sum)
        
        # Accept or reject with simulated annealing probability
        c5_old = compute_c5(h)
        c5_new = compute_c5(h_new)
        
        if c5_new < c5_old or np.random.random() < np.exp(-(c5_new - c5_old) / temperature):
            h = h_new.copy()
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                recent_solutions.append((h.copy(), c5_new))
                if len(recent_solutions) > max_recent:
                    recent_solutions.pop(0)
        
        # Adaptive cooling
        temperature *= cooling_rate
        if temperature < min_temperature:
            temperature = min_temperature
            
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    # Final local optimization on best solution
    def objective(h):
        return compute_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    result = minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 50, 'ftol': 1e-10}
    )
    
    if result.success:
        best_h = normalize_h(result.x, target_sum)
        best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
