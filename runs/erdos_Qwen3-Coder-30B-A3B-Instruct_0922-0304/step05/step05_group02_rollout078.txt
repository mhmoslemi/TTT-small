import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        # Initialize with a simple piecewise constant function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_h(h, target_sum)
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 1000
    patience = 50
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(overlap)
        max_overlap = overlap[max_overlap_idx] * dx
        
        # Update best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break

        # Compute gradient (simplified finite difference approach)
        # For a shift k where overlap is maximum,
        # gradient = (1 - h_shifted) * dx
        # Shift indices in correlation result: from -n+1 to n-1
        shift = max_overlap_idx - (n_points - 1)
        if shift >= 0:
            # h_shifted = h[shift:]
            h_shifted = np.concatenate([h[shift:], np.zeros(shift)])
        else:
            # h_shifted = h[:shift]
            h_shifted = np.concatenate([np.zeros(-shift), h[:shift]])
            
        # Gradient
        gradient = (1 - h_shifted) * dx
        
        # Gradient step
        h_new = h - learning_rate * gradient
        
        # Project to feasible region
        h = project_h(h_new, target_sum)
        
        # Adaptive learning rate
        if iteration > 0 and iteration % 100 == 0:
            learning_rate *= 0.95
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
