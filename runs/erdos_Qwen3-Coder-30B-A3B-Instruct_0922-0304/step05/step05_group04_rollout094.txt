import numpy as np
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    mutation_rate = 0.1
    tournament_size = 10
    max_generations = 1000
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Fill the rest of the population
    for _ in range(population_size - len(population)):
        h = np.random.random(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    def compute_c5(h):
        # Compute all overlaps at once
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_h = None
    best_c5 = float('inf')
    
    # Main genetic algorithm loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            c5_val = compute_c5(h)
            fitness.append(c5_val)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
        
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection using tournament
        selected = []
        for _ in range(population_size):
            tournament = random.sample(list(zip(population, fitness)), tournament_size)
            winner = min(tournament, key=lambda x: x[1])
            selected.append(winner[0])
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < population_size else selected[0]
            
            # Uniform crossover
            child1 = np.copy(parent1)
            child2 = np.copy(parent2)
            
            for j in range(n_points):
                if np.random.random() < 0.5:
                    child1[j] = parent2[j]
                    child2[j] = parent1[j]
            
            # Mutation with repair
            for j in range(n_points):
                if np.random.random() < mutation_rate:
                    if np.random.random() < 0.5:
                        child1[j] += np.random.normal(0, 0.1)
                    else:
                        child1[j] = np.random.random()
                    child1[j] = max(0, min(1, child1[j]))
                    
                if np.random.random() < mutation_rate:
                    if np.random.random() < 0.5:
                        child2[j] += np.random.normal(0, 0.1)
                    else:
                        child2[j] = np.random.random()
                    child2[j] = max(0, min(1, child2[j]))
            
            # Repair to sum constraint
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
    
    # Final computation with best solution
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
