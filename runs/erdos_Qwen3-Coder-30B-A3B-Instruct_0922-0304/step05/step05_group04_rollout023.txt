from scipy.signal import correlate
import time

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # First phase: Binary search with simulated annealing
    def gen_neighbor(h):
        h_new = h.copy()
        i, j = np.random.choice(n_points, 2, replace=False)
        h_new[i], h_new[j] = h_new[j], h_new[i]
        return h_new
    
    def binary_search():
        nonlocal h
        T = 1.0
        cooling = 0.99
        max_iter = 2000
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 2:
                break
            h_new = gen_neighbor(h)
            # Ensure h_new is binary
            h_new = np.clip(h_new, 0, 1)
            # Renormalize 
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_old = compute_c5(h)
            c5_new = compute_c5(h_new)
            if c5_new < c5_old or np.random.rand() < np.exp(-(c5_new - c5_old) / T):
                h = h_new
            T *= cooling
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            if best_c5 <= 0.3807:
                break
        return best_c5, best_h
    
    # Run binary search
    binary_search()
    
    # Second phase: Continuous relaxation with CVXPY
    # This will minimize C5 subject to h in [0,1] and sum(h) = n_points/2
    if time.time() - start_time < budget_s - 2:
        try:
            h_var = cp.Variable(n_points)
            # Minimize C5 - we'll handle the maximization differently
            # We compute the max overlap and try to reduce it
            constraints = [
                cp.sum(h_var) == n_points / 2,
                h_var >= 0,
                h_var <= 1
            ]
            # Use the optimized binary solution as warm start
            h_start = best_h
            prob = cp.Problem(cp.Minimize(0), constraints)
            # Actually we compute C5 from the solution at each step
            # So we'll do iterative filtering approach for relaxation
            
            # Direct gradient-like optimization based on approximated gradients
            # We use finite differences to estimate gradient of C5 with respect to h
            step_size = 0.01
            iter_count = 0
            while iter_count < 1000 and time.time() - start_time < budget_s - 1:
                iter_count += 1
                # Evaluate current objective
                current_c5 = compute_c5(best_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = best_h.copy()
                if best_c5 <= 0.3807:
                    break
                    
                # Gradient estimation using finite differences
                grad = np.zeros(n_points)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = best_h.copy()
                    h_minus = best_h.copy()
                    h_plus[i] += eps
                    h_minus[i] -= eps
                    # Clamp and renormalize
                    h_plus = np.clip(h_plus, 0, 1)
                    h_minus = np.clip(h_minus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                    overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
                    grad[i] = (np.max(overlap_plus) - np.max(overlap_minus)) / (2 * eps) * dx
                
                # Gradient descent step
                new_h = best_h - step_size * grad
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                best_h = new_h.copy()
                
            # Final evaluation of best solution
            current_c5 = compute_c5(best_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
            
        except Exception as e:
            pass  # Fall back to best_h if optimization fails

    return (best_h.tolist(), best_c5, n_points)
