import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def validate_and_repair(h):
    h = np.array(h)
    h = np.clip(h, 0, 1)
    h = h * (70 / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_coarse = validate_and_repair(initial_h_values)
    else:
        # Create a reasonable initial guess
        h_coarse = np.ones(35)
        h_coarse[17:] = 0
        h_coarse = h_coarse * (35 / 2) / np.sum(h_coarse)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * (2.0 / len(h))

    def constraint_sum(h):
        return np.sum(h) - len(h) / 2

    # Define constraints and bounds
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(len(h_coarse))]

    # Coarse optimization
    best_c5_coarse = float('inf')
    best_h_coarse = h_coarse.copy()
    
    max_iter_coarse = 50
    for iteration in range(max_iter_coarse):
        if time.time() - start_time > budget_s * 0.2:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h_coarse)
        if current_c5 < best_c5_coarse:
            best_c5_coarse = current_c5
            best_h_coarse = h_coarse.copy()
        
        # Optimize using SLSQP
        result = minimize(
            compute_c5, 
            h_coarse, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 10, 'ftol': 1e-6}
        )
        
        if result.success:
            h_coarse = result.x
            h_coarse = np.clip(h_coarse, 0, 1)
            h_coarse = h_coarse * (35 / 2) / np.sum(h_coarse)
        else:
            h_coarse += np.random.normal(0, 0.01, len(h_coarse))
            h_coarse = np.clip(h_coarse, 0, 1)
            h_coarse = h_coarse * (35 / 2) / np.sum(h_coarse)
    
    # Upsample to fine scale (70 points)
    h_fine = np.repeat(best_h_coarse, 2)
    
    # Refine with fine optimization
    bounds_fine = [(0, 1) for _ in range(70)]
    constraints_fine = [{'type': 'eq', 'fun': lambda x: np.sum(x) - 35.0}]
    
    best_c5_fine = float('inf')
    best_h_fine = h_fine.copy()
    
    max_iter_fine = 200
    for iteration in range(max_iter_fine):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h_fine)
        if current_c5 < best_c5_fine:
            best_c5_fine = current_c5
            best_h_fine = h_fine.copy()
            
        if best_c5_fine <= 0.3807:
            break
            
        # Optimize using SLSQP
        result = minimize(
            compute_c5, 
            h_fine, 
            method='SLSQP', 
            bounds=bounds_fine, 
            constraints=constraints_fine,
            options={'maxiter': 5, 'ftol': 1e-8}
        )
        
        if result.success:
            h_fine = result.x
            h_fine = np.clip(h_fine, 0, 1)
            h_fine = h_fine * (70 / 2) / np.sum(h_fine)
        else:
            h_fine += np.random.normal(0, 0.005, 70)
            h_fine = np.clip(h_fine, 0, 1)
            h_fine = h_fine * (70 / 2) / np.sum(h_fine)
    
    # Return best solution
    final_h = best_h_fine
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, 70)
