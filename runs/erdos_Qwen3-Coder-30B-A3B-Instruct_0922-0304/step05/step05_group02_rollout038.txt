import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_sum(h, target_sum):
    """Project h to have sum equal to target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_sum(h, target_sum)
    else:
        # Default initialization
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = project_sum(h, target_sum)
    
    # Optimization parameters
    momentum_decay = 0.9
    learning_rate = 0.1
    max_line_search = 10
    tolerance = 1e-6
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Momentum cache
    m = np.zeros_like(h)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute the current C5
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break

        # Compute correlation for all shifts
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Compute gradient of the worst overlap
        # For shift k, gradient of h[i] = -1 if 0 <= i-k < n
        # Expressed as: h_{i} = - (1 - h_{i+k})
        # Gradient of maximum overlap term w.r.t. h
        grad = np.zeros_like(h)
        # This is a simplified gradient approximation
        # A full gradient computation would require phased derivative
        # coefficients for each shift k, but we'll focus on the max k only

        # We compute an estimated gradient based on the shift that gives max overlap
        actual_k = max_idx - (len(h) - 1)  # Correct the shift

        # The gradient for the coefficient corresponding to shifts
        # We apply a simple gradient approximation:
        # d/dh[i] { (h ∗ (1 - h))(k) } = -h[i-k] if k = max_overlap_k (simplified)
        # However, it is non-trivial to get an exact gradient
        
        # As a simpler approach, use finite differences in context
        # Small perturbation approach
        eps = 1e-4
        grad = np.zeros_like(h)
        for i in range(len(h)):
            # Estimate gradient for h[i]
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = project_sum(h_plus, target_sum)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = project_sum(h_minus, target_sum)
            
            c5_plus = compute_c5(h_plus, dx)
            c5_minus = compute_c5(h_minus, dx)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)

        # Accumulate momentum: m = momentum_decay * m + (1 - momentum_decay) * grad
        m = momentum_decay * m + (1 - momentum_decay) * grad

        # Perform line search
        step_size = learning_rate
        for _ in range(max_line_search):
            h_new = h - step_size * m
            h_new = np.clip(h_new, 0, 1)
            h_new = project_sum(h_new, target_sum) 
            
            new_c5 = compute_c5(h_new, dx)
            
            if new_c5 < current_c5 or step_size < 1e-8:
                h = h_new
                break
            else:
                step_size *= 0.5

        # Update learning rate based on progress
        if iteration % 10 == 0:
            if abs(current_c5 - new_c5) < tolerance:
                learning_rate *= 0.9
            else:
                learning_rate = min(learning_rate * 1.1, 1.0)
                
        iteration += 1
    
    # Final optimization step
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
