import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 100000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization 
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def perturb(h, temperature):
        # Create a perturbed version with Gaussian noise
        h_new = h + np.random.normal(0, temperature * 0.1, n_points)
        h_new = np.clip(h_new, 0, 1)
        # Rescale to maintain the constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new

    # Simulated Annealing
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Initial temperature based on typical energy differences
    T0 = max(0.1, current_c5 * 0.1)
    T = T0
    cooling_rate = 0.9995
    
    iteration = 0
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        # Generate new candidate solution
        new_h = perturb(current_h, T)
        new_c5 = compute_c5(new_h)
        
        # Accept or reject the new solution
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
            current_h = new_h
            current_c5 = new_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
        # Cooling schedule
        T *= cooling_rate
        
        iteration += 1
        
        # Early termination if we've found a good enough solution
        if best_c5 <= 0.3807:
            break

    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
