import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 20
    elite_fraction = 0.25
    mutation_strength = 0.01
    max_generations = 1000
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def is_feasible(h):
        return np.all((h >= 0) & (h <= 1)) and abs(np.sum(h) - n_points / 2) < 1e-10
    
    def repair_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if not is_feasible(h):
            h = repair_h(h)
        population.append(h.copy())
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        population.append(h.copy())
    
    # Fill the rest with random feasible solutions
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = repair_h(h)
        population.append(h.copy())
    
    # Track best solution
    best_c5 = float('inf')
    best_h = population[0].copy()
    
    # Iteration loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = compute_c5(h)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Selection and mutation
        elite_count = max(1, int(population_size * elite_fraction))
        elite = population[:elite_count]
        
        new_population = []
        for h in elite:
            for _ in range(2):
                mutated = h + np.random.normal(0, mutation_strength, n_points)
                mutated = repair_h(mutated)
                new_population.append(mutated)
        
        # Keep population size fixed
        population = new_population[:population_size]
    
    # Final check
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
