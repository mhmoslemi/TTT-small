import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Projected gradient descent
    max_iter = 1000
    learning_rate = 0.01
    patience = 0
    patience_limit = 50
    decay_every = 100
    
    # Gradient computation function
    def compute_gradient(h, max_lags):
        gradient = np.zeros(n_points)
        for k in max_lags:
            # For each lag k, compute gradient contribution
            # Part 1: h[i] contributes (1 - h[i+k]) when i+k < n
            if k >= 0:
                for i in range(n_points - k):
                    gradient[i] += (1 - h[i + k])
            # Part 2: h[i] contributes -h[i-k] when i-k >= 0
            else:
                for i in range(-k, n_points):
                    gradient[i] -= h[i + k]
        return gradient / len(max_lags) if max_lags else np.zeros(n_points)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        overlaps = correlate(h, 1 - h, mode="full")
        max_val = overlaps.max()
        max_lags = [i - (n_points - 1) for i, val in enumerate(overlaps) if val == max_val]
        
        # Compute gradient
        grad = compute_gradient(h, max_lags)
        
        # Update with gradient and project
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = compute_c5(h_new)
        
        # Update if improvement
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Update h
        h = h_new
        
        # Adaptive learning rate
        if iter_num % decay_every == 0 and iter_num > 0:
            learning_rate *= 0.9
            
        if patience >= patience_limit:
            # Reset with better solution if possible
            h = best_h.copy()
            learning_rate = max(0.001, learning_rate * 0.5)
            patience = 0

    # Local refinement with SLSQP
    try:
        result = minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
            refined_c5 = compute_c5(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
        
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
