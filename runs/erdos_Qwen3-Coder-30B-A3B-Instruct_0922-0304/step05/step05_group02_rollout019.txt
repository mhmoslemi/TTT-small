```python
import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: clip to [0,1] and rescale to sum to target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_h(h, target_sum)

    # Gradient descent parameters
    eta = 0.1
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute cross-correlation for all shifts
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_shift = max_idx - (len(h) - 1)
        
        # Compute gradient based on the maximum overlap shift
        h_shifted = np.roll(h, -max_shift)
        gradient = (1 - h_shifted) * dx
        
        # Update h with gradient descent
        h = h - eta * gradient
        
        # Project h to feasible region
        h = project_h(h, target_sum)
        
        # Compute current C5 value
        current_c5 = compute_c5(h, dx)
        
        # Update best solution if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Adaptive learning rate
        if current_c5 < best_c5 + 1e-10:
            eta *= 1.01
        else:
            eta *= 0.9
            
        # Periodic perturbation
        if iteration % 100 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h = h + noise
            h = project_h(h, target_sum)
            
        iteration += 1
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
```
