import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
    
    # Start with a refined version of the best solution from the previous run
    # Try to rotate and refine the current solution with local optimizers
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Try multiple strategies to escape local minima
    strategies = [
        ("GradientDescent", lambda x: minimize(
            lambda h: compute_c5(h),
            x,
            method='SLSQP',
            bounds=Bounds([0]*len(x), [1]*len(x)),
            constraints=[{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}],
            options={'maxiter': 50, 'ftol': 1e-8}
        )),
        ("SimplePerturbation", lambda x: x + np.random.normal(0, 0.01, len(x))),
        ("RandomSearch", lambda x: np.random.rand(len(x))),
        ("PatternRefinement", lambda x: refine_pattern(x))
    ]
    
    for strategy_name, strategy in strategies:
        if time.time() - start_time > budget_s - 2:
            break
            
        # Try 3 refinements per strategy with 200s budget
        for iter_idx in range(3):
            if time.time() - start_time > budget_s - 2:
                break
                
            try:
                result = strategy(h.copy())
                
                if strategy_name == "GradientDescent" and result.success:
                    h_new = result.x
                else:
                    h_new = result
                
                h_new = normalize_h(h_new, target_sum)
                c5_eval = compute_c5(h_new)
                
                if c5_eval < best_c5:
                    best_c5 = c5_eval
                    best_h = h_new.copy()
                
                if best_c5 <= 0.3807:
                    return (best_h.tolist(), best_c5, n_points)
            except:
                pass
            
            # Perform local iterations with small perturbations
            for _ in range(30):
                if time.time() - start_time > budget_s - 2:
                    break
                h_perturb = h + np.random.normal(0, 0.02, n_points)
                h_perturb = normalize_h(h_perturb, target_sum)
                c5_test = compute_c5(h_perturb)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h_perturb.copy()
                if best_c5 <= 0.3807:
                    return (best_h.tolist(), best_c5, n_points)
    
    # Refine using direct optimization
    def objective(h):
        return compute_c5(h)
            
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Local search around current solution
    while time.time() - start_time < budget_s - 1:
        result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        if result.success:
            h_refined = result.x
            h_refined = normalize_h(h_refined, target_sum)
            current_c5 = compute_c5(h_refined)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_refined
        else:
            break
            
        # Small random perturbations 
        h_perturb = best_h + np.random.normal(0, 0.005, n_points)
        h_perturb = normalize_h(h_perturb, target_sum)
        c5_perturb = compute_c5(h_perturb)
        if c5_perturb < best_c5:
            best_c5 = c5_perturb
            best_h = h_perturb
            
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)

def refine_pattern(h):
    """Try to refine h by adjusting edges and smoothing"""
    h_new = h.copy()
    # Simple smoothing
    h_new = np.convolve(h_new, np.ones(5)/5, mode='same')
    # Renormalize
    h_new = normalize_h(h_new, len(h_new) / 2)
    return h_new
