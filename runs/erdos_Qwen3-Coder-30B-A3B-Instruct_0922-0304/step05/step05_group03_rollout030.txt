import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, k_max, dx, n_points):
    """Compute gradient of overlap at k_max with respect to h."""
    grad = np.zeros(n_points)
    for i in range(n_points):
        # For overlap at lag k_max, the contribution to h[i] is from h[i+k_max] term
        # But in the correlation we do h[i] * (1-h[i+k_max]) 
        # So gradient of overlap w.r.t h[i] = (1-h[i+k_max]) if i+k_max < n_points
        # But we should use the correlation formula more carefully
        # Actually, for i-th element, the effect on overlap at lag k_max 
        # is 1 - h[i+k_max] if k_max >= 0 and i+k_max < n_points
        if i + k_max < n_points and i + k_max >= 0:
            grad[i] = 1 - h[i + k_max]
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent parameters
    alpha = 0.1
    patience = 0
    patience_limit = 20
    decay_rate = 0.5
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Compute overlap and find maximum lag
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        
        # Compute gradient
        grad = compute_gradient(h, k_max - (n_points - 1), dx, n_points)
        
        # Update h
        h = h - alpha * grad
        h = normalize_h(h, target_sum)
        
        # Adapt learning rate
        if iteration % 50 == 0:
            if compute_c5(h, dx) > current_c5:
                alpha *= decay_rate
                patience += 1
                if patience > patience_limit:
                    h = best_h.copy()
                    alpha = 0.1
                    patience = 0
            else:
                patience = 0

    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
