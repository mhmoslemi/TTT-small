import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.95
    min_T = 1e-6
    max_iter = 10000
    
    # Initialize
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    T = T0
    iteration = 0
    
    while T > min_T and iteration < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb current solution
        new_h = current_h + np.random.normal(0, 0.02, n_points)
        new_h = np.clip(new_h, 0, 1)
        # Rescale to meet constraint
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Compute new objective
        new_c5 = compute_c5(new_h)
        
        # Accept or reject
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = new_h.copy()
            current_c5 = new_c5
            
        # Update best
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Early exit if target met
        if best_c5 <= 0.3807:
            break
            
        # Cooling schedule
        T *= alpha
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
