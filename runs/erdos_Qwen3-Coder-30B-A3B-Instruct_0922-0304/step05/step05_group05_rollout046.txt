import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def compute_gradient(h, max_lags):
        gradient = np.zeros(n_points)
        for k in max_lags:
            # Contribution from h[i] * (1 - h[i+k]) 
            if k >= 0:
                # For i+k < n_points, gradient[i] += (1 - h[i+k])
                for i in range(n_points - k):
                    gradient[i] += (1 - h[i + k])
                # For i-k >= 0, gradient[i] -= h[i-k] 
                for i in range(k, n_points):
                    gradient[i] -= h[i - k]
            else:
                # k < 0, so we have h[i+k] term
                k_abs = -k
                # For i+k >= 0, gradient[i] += (1 - h[i+k]) 
                for i in range(k_abs, n_points):
                    gradient[i] += (1 - h[i - k_abs])
                # For i-k < n_points, gradient[i] -= h[i-k]
                for i in range(n_points):
                    if i - k_abs < n_points:
                        gradient[i] -= h[i - k_abs]
        return gradient / len(max_lags) if max_lags else np.zeros(n_points)
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 1000
    patience = 50
    patience_counter = 0
    prev_best_c5 = best_c5
    learning_rate_decay = 0.9
    decay_interval = 100
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap and find max lags
        overlaps = correlate(h, 1 - h, mode="full")
        max_val = overlaps.max()
        max_lags = [i - (n_points - 1) for i, val in enumerate(overlaps) if val == max_val]
        
        # Compute gradient
        gradient = compute_gradient(h, max_lags)
        
        # Update
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate new solution
        current_c5 = compute_c5(h)
        
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Adaptive learning rate
        if it % decay_interval == 0 and it > 0:
            learning_rate *= learning_rate_decay
            
        # Early stopping on no improvement
        if patience_counter >= patience:
            # Try SLSQP refinement
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
            try:
                result = minimize(
                    compute_c5,
                    best_h.copy(),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 30, 'ftol': 1e-8}
                )
                if result.success:
                    refined_h = result.x
                    refined_h = np.clip(refined_h, 0, 1)
                    refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
                    refined_c5 = compute_c5(refined_h)
                    if refined_c5 < best_c5:
                        best_c5 = refined_c5
                        best_h = refined_h.copy()
            except:
                pass
            patience_counter = 0
            
    # Final SLSQP refinement
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    try:
        result = minimize(
            compute_c5,
            best_h.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-10}
        )
        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_c5 = compute_c5(final_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h.copy()
    except:
        pass
        
    return (best_h.tolist(), best_c5, n_points)
