import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (n_points / 2) / current_sum
    else:
        h = np.ones(n_points) * 0.5
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (n_points / 2) / current_sum

    best_h = h.copy()
    best_c5 = np.max(correlate(h, 1 - h, mode="full")) * dx
    
    # Early exit if already good
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Interior-point optimization parameters
    max_iter = 500
    perturbation_scale = 0.1
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current correlations
        corr = correlate(h, 1 - h, mode="full")
        c_values = corr * dx
        
        # Find max correlation for upper bound
        c5_upper = np.max(c_values)
        
        # Use convex relaxation approach
        h_var = Variable(n_points)
        c_bound = Variable()
        
        # Constraints
        constraints = [
            sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1,
            c_bound >= c5_upper  # Over-approximate, we'll tighten this
        ]
        
        # Check if we can optimize directly with all correlation constraints
        # This is a simplified version of a direct convex optimization approach
        # Create a relaxed optimization problem with c5_upper as a variable
        c_relaxed = Variable()
        h_new = Variable(n_points)
        
        # One constraint for the maximum shift
        max_shift = len(c_values) // 2
        constraints = [
            sum(h_new) == n_points / 2,
            h_new >= 0,
            h_new <= 1,
            c_relaxed >= max(c_values)  # We hardcode the current value
        ]
        
        # Formulate and solve
        objective = Minimize(c_relaxed)
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=CVXOPT, verbose=False, max_iters=200)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new_sol = h_new.value
                if h_new_sol is not None:
                    h = np.clip(h_new_sol, 0, 1)
                    current_sum = np.sum(h)
                    if current_sum > 0:
                        h = h * (n_points / 2) / current_sum
        except Exception:
            pass  # Fall back to local search if optimization fails
            
        # Local search using gradient-based method
        def objective(h_vec):
            return np.max(correlate(h_vec, 1-h_vec, mode="full")) * dx

        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2

        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        # Try gradient-based improvement
        try:
            from scipy.optimize import minimize
            result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                              constraints=constraints, options={'maxiter': 20, 'ftol': 1e-8})
            if result.success:
                h_new = np.clip(result.x, 0, 1)
                h_sum = np.sum(h_new)
                if h_sum > 0:
                    h_new = h_new * (n_points / 2) / h_sum
                h = h_new
        except Exception:
            pass
            
        # Evaluate the current solution
        c5_current = np.max(correlate(h, 1 - h, mode="full")) * dx
        
        # Update best solution if improved
        if c5_current < best_c5:
            best_h = h.copy()
            best_c5 = c5_current
            
        # Early exit if target is reached
        if best_c5 <= 0.3807:
            break
        
        # Apply perturbation to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            noise_scale = max(0.01, perturbation_scale * (0.9 ** (iteration // 50)))
            noise = np.random.normal(0, noise_scale, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            if current_sum > 0:
                h = h * (n_points / 2) / current_sum
                
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
