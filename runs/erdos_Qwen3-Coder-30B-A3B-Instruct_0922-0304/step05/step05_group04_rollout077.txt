from scipy.signal import correlate
import time

    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - create alternating structure
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if i % 2 == 0 else 0
        # Perturb slightly
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative tightening approach
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Current overlap computation
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Try different optimization strategies
        if iteration % 3 == 0:
            # Use gradient-based refinement with small steps
            g = np.zeros(n_points)
            eps_val = 1e-5
            
            # Compute gradient by finite differences
            for i in range(n_points):
                h_plus = h.copy()
                h_minus = h.copy()
                
                h_plus[i] += eps_val
                h_minus[i] -= eps_val
                
                # Clamp and renormalize
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
                overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
                g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps_val) * dx

            # Apply gradient descent step
            h = h - 0.01 * g
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        else:
            # Local random search to diversify
            pert = np.random.normal(0, 0.01, n_points)
            h = h + pert
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
