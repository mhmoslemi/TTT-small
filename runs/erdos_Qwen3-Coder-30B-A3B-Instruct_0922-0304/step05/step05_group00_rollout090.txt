import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    """Evaluate objective and check constraints."""
    c5 = compute_c5(h, dx)
    return c5

def repair_feasibility(h, target_sum):
    """Repair h to maintain feasibility."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Normalize sum
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    population_size = 30
    max_iter = 500
    
    # Initialize
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
    else:
        # Default construction
        h_base = np.zeros(n_points)
        h_base[:n_points//2] = 1.0
    
    h_base = repair_feasibility(h_base, target_sum)
    
    # Population generation
    population = []
    for _ in range(population_size):
        h = h_base + np.random.normal(0, 0.02, n_points)
        h = repair_feasibility(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_individual = h_base.copy()
    
    iter_count = 0
    step_size = 0.05
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute fitness
        fitness = [evaluate_individual(h, dx) for h in population]
        
        # Track best
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_individual = population[min_fit_idx].copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Generate new population
        new_population = []
        
        # Keep best individual
        new_population.append(best_individual.copy())
        
        # Differential evolution (40%)
        for _ in range(int(0.4 * (population_size - 1))):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            parent_a, parent_b, parent_c = population[a], population[b], population[c]
            child = parent_a + 0.8 * (parent_b - parent_c)
            child = repair_feasibility(child, target_sum)
            new_population.append(child)
        
        # Gradient-based (30%)
        for _ in range(int(0.3 * (population_size - 1))):
            idx = np.random.randint(len(population))
            h = population[idx].copy()
            # Simple gradient ascent using finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = repair_feasibility(h_plus, target_sum)
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = repair_feasibility(h_minus, target_sum)
                grad[i] = (evaluate_individual(h_plus, dx) - evaluate_individual(h_minus, dx)) / (2 * eps)
            
            # Update using gradient
            h = h - step_size * grad
            h = repair_feasibility(h, target_sum)
            new_population.append(h)
        
        # Random sampling (20%)
        for _ in range(int(0.2 * (population_size - 1))):
            h = np.random.rand(n_points) * 2
            h = repair_feasibility(h, target_sum)
            new_population.append(h)
        
        # Mutation of best individual (10%)
        for _ in range(int(0.1 * (population_size - 1))):
            h = best_individual + np.random.normal(0, 0.02, n_points)
            h = repair_feasibility(h, target_sum)
            new_population.append(h)
        
        # Fill up
        while len(new_population) < population_size:
            h = np.random.rand(n_points) * 2
            h = repair_feasibility(h, target_sum)
            new_population.append(h)
        
        # Update population
        population = new_population[:population_size]
        
        # Adjust step size
        step_size *= 0.98
        
        iter_count += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_individual, dx)
    
    return (best_individual.tolist(), final_c5, n_points)
