import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Create swarm
    particles = [h.copy() for _ in range(swarm_size)]
    velocities = [np.random.uniform(-0.05, 0.05, n_points) for _ in range(swarm_size)]
    
    # Best positions
    particle_best = [p.copy() for p in particles]
    particle_best_vals = [float('inf')] * swarm_size
    global_best = h.copy()
    global_best_val = float('inf')
    
    # PSO parameters
    omega_max = 0.9
    omega_min = 0.4
    phi1 = 2.0
    phi2 = 2.0
    v_max = 0.1
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        omega = omega_max - (omega_max - omega_min) * (time.time() - start_time) / budget_s
        
        for i in range(swarm_size):
            h = particles[i]
            v = velocities[i]
            
            # Compute current overlap
            current_val = compute_c5(h)
            
            # Update particle best
            if current_val < particle_best_vals[i]:
                particle_best_vals[i] = current_val
                particle_best[i] = h.copy()
            
            # Update global best
            if current_val < global_best_val:
                global_best_val = current_val
                global_best = h.copy()
            
            # Early exit if target achieved
            if global_best_val <= 0.3807:
                break
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = phi1 * r1 * (particle_best[i] - h)
            social = phi2 * r2 * (global_best - h)
            v = omega * v + cognitive + social
            
            # Clamp velocity
            v = np.clip(v, -v_max, v_max)
            
            # Update position
            h = h + v
            
            # Enforce bounds
            h = np.clip(h, 0, 1)
            
            # Renormalize
            h = h * (n_points / 2) / np.sum(h)
            
            # Update
            particles[i] = h
            velocities[i] = v
            
        if global_best_val <= 0.3807:
            break
    
    return (global_best.tolist(), global_best_val, n_points)
