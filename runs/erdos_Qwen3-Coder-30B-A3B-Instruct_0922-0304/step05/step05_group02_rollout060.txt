import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have sum equal to target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values)
        init_h = np.clip(init_h, 0, 1)
        init_h = normalize_h(init_h, n_points / 2)
    else:
        # Use a smoother initial guess
        init_h = np.ones(n_points)
        init_h[:n_points//2] = 0.5
        init_h[n_points//2:] = 0
        init_h = normalize_h(init_h, n_points / 2)

    h = init_h.copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()

    # Trust-region parameters
    momentum_decay = 0.9
    learning_rate = 0.1
    line_search_decay = 0.5
    min_step_size = 1e-6
    max_line_search_steps = 10
    
    # Momentum cache
    m = np.zeros(n_points)
    
    # Gradient-based optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 1000:
        # Compute all overlaps to find maximum
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_overlap = overlap[max_idx] * dx

        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient
        # Derivative of max_overlap w.r.t. h at the worst shift k
        # For shift k, the derivative is 1 - 2*h[i] where i is at the shifted position
        # We compute a sampled gradient at a few extreme points
        gradient = np.zeros(n_points)
        # Get the indices that contribute to the maximum overlap
        k_idx = max_idx - (n_points - 1)  # convert to shift index (k)
        # Compute the effective gradient from the correlation
        # We compute gradient based on worst contributing shift
        shift_indices = np.arange(n_points)
        shifted_indices = (shift_indices - k_idx) % n_points
        for i in range(n_points):
            j = shifted_indices[i]
            if j < n_points:
                gradient[i] += (1 - 2 * h[j]) * dx
        
        # Apply momentum
        m = momentum_decay * m + (1 - momentum_decay) * gradient
        # Apply trust-region update
        step = learning_rate * m
        
        # Perform line search
        step_size = 1.0
        h_trial = h - step_size * step
        
        # Validate feasibility (clip)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = normalize_h(h_trial, n_points / 2)
        
        # Check if this improved the objective function
        trial_c5 = compute_c5(h_trial, dx)
        if trial_c5 < best_c5:
            h = h_trial
            best_c5 = trial_c5
            best_h = h.copy()
        elif step_size > min_step_size:
            # Try smaller steps
            for _ in range(max_line_search_steps):
                step_size *= line_search_decay
                h_trial = h - step_size * step
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = normalize_h(h_trial, n_points / 2)
                trial_c5 = compute_c5(h_trial, dx)
                if trial_c5 < best_c5:
                    h = h_trial
                    best_c5 = trial_c5
                    best_h = h.copy()
                    break

        # Update learning rate based on progress
        # This is simplified but retains some adaptivity
        iteration += 1
        
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
