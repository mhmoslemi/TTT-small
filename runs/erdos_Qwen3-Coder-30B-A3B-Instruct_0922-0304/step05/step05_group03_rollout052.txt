import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def halton_sequence(n, base=2):
    """Generate n points using base-2 Halton sequence."""
    result = np.zeros(n)
    for i in range(n):
        f = 1
        result[i] = 0
        while f < 1:
            result[i] += (i % base) * f
            i //= base
            f /= base
    return result

    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available

    def compute_c5(h):
        return evaluate_c5(h, dx)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start with Halton sequence
    n_starts = 50

        h_start = halton_points[i] * np.ones(n_points) + np.random.normal(0, 0.05, n_points)

        def objective(h):
            return compute_c5(h)
