import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Objective function
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Simulated Annealing
    def simulated_annealing():
        current_h = h.copy()
        current_c5 = compute_c5(current_h)
        best_h = current_h.copy()
        best_c5 = current_c5
        
        # Initial temperature based on the expected energy difference
        # We'll estimate using a few random perturbations
        T0 = 0.01
        T = T0
        cooling_rate = 0.95
        min_T = 1e-6
        max_iter = int(1e4)
        
        # Track iterations without improvement
        no_improvement_count = 0
        max_no_improvement = 500
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Generate neighbor by perturbation
            neighbor_h = current_h + np.random.normal(0, 0.02, n_points)
            neighbor_h = np.clip(neighbor_h, 0, 1)
            neighbor_h = neighbor_h * (n_points / 2) / np.sum(neighbor_h)
            
            # Compute neighbor cost
            neighbor_c5 = compute_c5(neighbor_h)
            
            # Accept or reject
            if neighbor_c5 < current_c5:
                current_h = neighbor_h
                current_c5 = neighbor_c5
                if neighbor_c5 < best_c5:
                    best_h = neighbor_h
                    best_c5 = neighbor_c5
                no_improvement_count = 0
            else:
                # Accept with probability
                delta = neighbor_c5 - current_c5
                if np.random.rand() < np.exp(-delta / T):
                    current_h = neighbor_h
                    current_c5 = neighbor_c5
                no_improvement_count += 1
                
            # Cool down
            T = max(min_T, T * cooling_rate)
            
            # Early exit if we've found a good solution
            if best_c5 <= 0.3807:
                break
                
            # Stop if no improvement for a while
            if no_improvement_count > max_no_improvement:
                T = max(min_T, T * 0.9)
                no_improvement_count = 0
                
        return best_h, best_c5
    
    # Run SA
    best_h, best_c5 = simulated_annealing()
    
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
