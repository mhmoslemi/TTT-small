import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_swaps = 1000
    swap_size = 5
    temperature = 0.1
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: 3-region piecewise function
        h = np.ones(n_points)
        mid_point = n_points // 2
        h[mid_point:] = 0
        # Add some randomness to avoid local minima
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def satisfy_sum_constraint(h):
        return h * (n_points / 2) / np.sum(h)
    
    def swap_segments(h, i1, i2, size):
        h_new = h.copy()
        # Swap segments
        segment1 = h_new[i1:i1+size].copy()
        segment2 = h_new[i2:i2+size].copy()
        h_new[i1:i1+size] = segment2
        h_new[i2:i2+size] = segment1
        # Rescale to maintain sum constraint
        return satisfy_sum_constraint(h_new)
    
    # Initialize
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Main swap loop
    for swap_iter in range(max_swaps):
        if time.time() - start_time > budget_s - 1:
            break
                
        # Select two non-overlapping segments
        i1 = np.random.randint(0, n_points - swap_size)
        # Ensure segments don't overlap
        max_i2 = n_points - swap_size
        min_i2 = i1 + swap_size
        if min_i2 < max_i2:
            i2 = np.random.randint(min_i2, max_i2)
        else:
            i2 = np.random.randint(0, i1)
            
        # Perform swap
        h_new = swap_segments(best_h, i1, i2, swap_size)
        
        # Evaluate
        new_c5 = compute_c5(h_new)
        
        # Accept or reject
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        else:
            # Probabilistically accept, decreasing temperature over time
            prob = np.exp(-(new_c5 - best_c5) / temperature)
            if np.random.rand() < prob:
                best_c5 = new_c5
                best_h = h_new.copy()
                
        # Early exit if we reached target
        if best_c5 <= 0.3807:
            break

    # Final polishing with SLSQP
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    try:
        result = minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h = satisfy_sum_constraint(final_h)
            final_c5 = compute_c5(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass

    return (best_h.tolist(), best_c5, n_points)
