import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx):
    """Compute gradient of C5 with respect to h."""
    n = len(h)
    # Compute correlation of h with (1-h) and its reverse
    corr = correlate(h, 1 - h, mode="full")
    # Find the index of maximum correlation
    max_lag_idx = np.argmax(corr)
    # The gradient computation requires gradient of correlation
    # We use finite differences on the correlation with respect to h
    # This is a simplified gradient estimation
    grad = np.zeros(n)
    eps = 1e-6
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = normalize_h(h_plus, np.sum(h))
        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
        grad[i] = (corr_plus[max_lag_idx] - corr[max_lag_idx]) / eps
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
        
    # Add small random noise to initial state
    h += np.random.normal(0, 0.01, n_points)
    h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Hyperparameters
    learning_rate = 0.1
    max_iter = 10000
    patience = 100
    patience_counter = 0
    improvement_threshold = 1e-6
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter > patience:
            # Reduce learning rate
            learning_rate *= 0.5
            patience_counter = 0

        # Early exit if we found a good enough solution
        if best_c5 <= 0.3807:
            break
            
        # Gradient step (note: this is an approximation)
        try:
            grad = compute_gradient(h, dx)
            h = h - learning_rate * grad
            h = normalize_h(h, target_sum)
        except:
            # If gradient computation fails, fall back to a perturbation
            h = np.clip(h + np.random.normal(0, 0.001, n_points), 0, 1)
            h = normalize_h(h, target_sum)
            
        # Coordinate-wise improvement
        if iteration % 10 == 0:
            for i in range(n_points):
                h_test = h.copy()
                h_test[i] += np.random.normal(0, 0.005)
                h_test = normalize_h(h_test, target_sum)
                test_c5 = evaluate_c5(h_test, dx)
                if test_c5 < current_c5:
                    h = h_test.copy()
    
    # Final refinement with local search
    if time.time() - start_time < budget_s - 5:

        def objective(h):
            return evaluate_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        result = minimize(
            objective,

            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            best_h = result.x
            best_h = normalize_h(best_h, target_sum)
            best_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
