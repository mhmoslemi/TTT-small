import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Helper to compute C5
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Initialize population
    population_size = 20
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Fill population with mutated versions
    for _ in range(population_size - 1):
        mutant = h + np.random.normal(0, 0.02, n_points)
        mutant = np.clip(mutant, 0, 1)
        mutant = mutant * (n_points / 2) / np.sum(mutant)
        population.append(mutant.copy())
    
    def mutate(h, mutation_std):
        mutated = h + np.random.normal(0, mutation_std, h.shape)
        mutated = np.clip(mutated, 0, 1)
        sum_h = np.sum(mutated)
        mutated = mutated * (n_points / 2) / sum_h
        return mutated
    
    def evaluate_fitness(population):
        fitness = []
        for h in population:
            fitness.append(compute_c5(h))
        return np.array(fitness)
    
    def select_elite(population, fitness, elite_fraction=0.25):
        idxs = np.argsort(fitness)
        elite_count = max(1, int(len(population) * elite_fraction))
        return [population[i] for i in idxs[:elite_count]]
    
    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    
    max_generations = 1000
    mutation_strength = 0.01
    no_improvement_generations = 0
    max_no_improvement = 50
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        fitness = evaluate_fitness(population)
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            no_improvement_generations = 0
        else:
            no_improvement_generations += 1
            
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_generations > max_no_improvement:
            mutation_strength *= 0.1
            no_improvement_generations = 0
            
        elite = select_elite(population, fitness)
        new_population = []
        
        for individual in elite:
            for _ in range(2):
                new_pop = mutate(individual, mutation_strength)
                new_population.append(new_pop)
        
        # Keep some random members for diversity
        while len(new_population) < population_size:
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            new_population.append(h)
        
        population = new_population[:population_size]
    
    return (best_h.tolist(), best_c5, n_points)
