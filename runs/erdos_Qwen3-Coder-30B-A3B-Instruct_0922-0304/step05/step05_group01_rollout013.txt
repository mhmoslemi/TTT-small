import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def objective(h):
        return compute_c5(h)

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Binary search bounds
    c5_low = 0.0
    c5_high = 0.38085  # Current upper bound
    tolerance = 1e-4
    max_iterations = 40

    # Track performance
    iteration = 0
    last_improvement = 0

    while iteration < max_iterations and time.time() - start_time < budget_s - 1:
        mid_c5 = (c5_low + c5_high) / 2.0
        
        # Create objective function with additional constraint
        def constrained_objective(h):
            return compute_c5(h)
        
        # Create constraint that max overlap must be <= mid_c5 * dx
        def max_overlap_constraint(h):
            overlap = correlate(h, 1 - h, mode="full")
            return mid_c5 * dx - np.max(overlap) * dx
        
        # Update constraints
        constrained_constraints = [
            {'type': 'eq', 'fun': constraint_sum},
            {'type': 'ineq', 'fun': max_overlap_constraint}
        ]

        # Try to find a feasible solution
        result = minimize(
            constrained_objective, 
            h.copy(), 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constrained_constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            current_c5 = compute_c5(h_opt)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt.copy()
                last_improvement = iteration
            
            if current_c5 <= mid_c5:
                c5_high = mid_c5
            else:
                c5_low = mid_c5
        else:
            # If optimization fails, try occasionally with the previous solution
            if iteration - last_improvement > 5:
                h += np.random.normal(0, 0.01, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Try to recover with gradient descent on constraints
                result = minimize(
                    constrained_objective,
                    h.copy(),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 20, 'ftol': 1e-6}
                )
                if result.success:
                    h = result.x
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                else:
                    # Random perturbation if no improvement
                    h += np.random.normal(0, 0.02, n_points)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
            
            # For monitoring purposes, go to the next iteration with current solution
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                last_improvement = iteration

        iteration += 1

        # Early exit check
        if best_c5 <= 0.3807:
            break

        # Update tolerance based on the progress
        if abs(c5_high - c5_low) < tolerance:
            tolerance = abs(c5_high - c5_low) / 10.0

    # Final evaluation for best solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
