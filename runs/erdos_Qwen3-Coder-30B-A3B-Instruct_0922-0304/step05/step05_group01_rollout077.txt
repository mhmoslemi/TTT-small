import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 20
    max_generations = 1000
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def is_feasible(h):
        return np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) - n_points / 2) < 1e-10
    
    def repair(h):
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (n_points / 2) / sum_h
        return h
    
    # Initialize population
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        sum_h0 = np.sum(h0)
        h0 = h0 * (n_points / 2) / sum_h0
        population = [h0]
    else:
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 = repair(h0)
        population = [h0]
    
    # Fill population
    while len(population) < population_size:
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = repair(h)
        population.append(h)
    
    def evaluate_fitness(population):
        fitness = []
        for h in population:
            c5 = compute_c5(h)
            fitness.append(c5)
        return np.array(fitness)
    
    def select_elite(population, fitness, elite_fraction=0.25):
        indices = np.argsort(fitness)
        elite_indices = indices[:int(len(population)*elite_fraction)]
        return [population[i] for i in elite_indices]
    
    def mutate(h, mutation_strength):
        mutated = h + np.random.normal(0, mutation_strength, h.shape)
        mutated = np.clip(mutated, 0, 1)
        sum_h = np.sum(mutated)
        if sum_h > 0:
            mutated = mutated * (n_points / 2) / sum_h
        return mutated
    
    best_c5 = float('inf')
    best_h = None
    
    # Main evolutionary loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        fitness = evaluate_fitness(population)
        
        # Track best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
        
        elite = select_elite(population, fitness)
        
        # Generate new population from elite members
        new_population = []
        for individual in elite:
            for _ in range(2):  # Each elite produces 2 offspring
                offspring = mutate(individual, 0.01)
                new_population.append(offspring)
        
        # Keep some diversity
        while len(new_population) < population_size:
            h = np.random.rand(n_points)
            h = repair(h)
            new_population.append(h)
            
        population = new_population[:population_size]
    
    # Final check and return
    if best_h is None:
        # If no improvement occurred, use initial
        best_h = h0
        best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
