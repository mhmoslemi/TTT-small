import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have sum equal to target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    scale = target_sum / current_sum
    return h * scale

def compute_gradient(h, dx, k_best):
    """Compute gradient with respect to shift k_best."""
    n = len(h)
    grad = np.zeros(n)
    
    # For correlation h * (1 - h) at shift k_best
    # This is approximately the derivative of overlap at that shift
    # We'll use finite differences or compute gradients directly
    
    # For efficient gradient computation, we'll compute element-wise
    # We have a contribution from each h[i] to the overlap at shift k_best
    # This happens when we compute: sum_i h[i] * (1 - h[i + k_best])
    
    # Compute derivative symbolically:
    # f = h[i] * (1 - h[i + k_best])
    # df/dh[i] = 1 - h[i + k_best]
    # df/dh[i + k_best] = -h[i]
    
    # However, since we have a convolution with 1-h, we can compute:

    # We want to compute gradient of sum_{i,k} h[i] * (1 - h[i+k]) = sum_{i,k} h[i] - h[i]*h[i+k]
    # This is sum_i h[i] * len(h) - sum_{i,k} h[i]*h[i+k]
    # The symmetric case: sum_{i,k} h[i]*h[i+k] = sum_{i,k} h[i]*h[i+k] = \sum_k \sum_{i} h[i]* h[i+k]
    # If we're looking at derivative w.r.t h[j]:
    # The summand h[i]*h[i+k] contributes to partial derivative when i=j or i+k=j.
    # When i=j: we get 1 * h[j+k] (from h[i+k])
    # When i+k=j: we get h[j-k] (from h[i])
    
    # So for the derivative of overlap w.r.t h[j]:
    # - it's derived from h[i]*h[i+k]
    # - When i=j: contributes h[i+k] 
    # - When i+k=j: contributes h[i] -> h[j-k]
    
    # For each fixed k_best, we want gradient w.r.t h for the overlap at this k.
    # When i + k_best = j, i = j - k_best
    # The chosen k_best can be used to compute:
    # Gradient: [0, 0, ..., h[j-k_best], 0, ..., h[j+k_best], 0, ...]
    # BUT this is tricky since it's a different term for *each* j.

    # Alternative (simpler):
    # Just compute for the shift k_best which gives the max overlap.
    # The max overlap is a single number. 
    # So we want to differentiate that single maximum.
    # This requires the structure of how the max occurs.
    
    # However, since we are just doing local descent, 
    # it's easier to assume that it is approximately:
    #Gradient w.r.t h[i] = - sum_{k: i+k = argmax} 1[-k]  = -h[i-k_best]
    # which is incorrect in general.

    # Simpler approach:
    # For a fixed shift k,  f = sum h[i] * (1 - h[i+k])
    # df/dh[i] = 1 - h[i+k] 

    # There are h[i+k] terms which are part of our sum, and therefore:
    # df/dh[i] = 1 - h[i+k_best] 
    # To make this code tractable in continuous time, we just use it for one shift.
    # Better: Differentiate the maximum value across all shifts.
    # If we have a candidate shift where full overlaps have maximum value,
    # Then we compute the gradient with respect to the maximum shift and apply that.
    
    # Assume we are looking at the gradient of h with respect to the maximum overlap shift.
    # Let i be index of h, and we have a specific k_best.
    # Overlap_{k_best} = sum h[i] * (1 - h[i+k_best]) over i
    # d Overlap_i / d h[i] = 1 - h[i+k_best]
    
    # But since we sum over all i, we get derivatives for ALL h[i].
    # For i: d/dh[i] sum_j h[j] * (1 - h[j+k_best]) = 1 - h[i+k_best] if i+k_best within range, else 0

    # We compute gradient for the max overlap shift.
    # In order to differentiate the maximum, we can still apply a method like:
    # Let alpha = argmax_k overlap_k -> derivative of max_k overlap_k at alpha.
    # For that, we approximate:
    # gradient(w.r.t. h) of overlap_alpha ≈ each term contributing to it (1-h[i+k_alpha])
    for i in range(n):
        if i + k_best < n:
            grad[i] = 1 - h[i + k_best]
        else:
            grad[i] = 0
    return grad

def line_search_step(h, grad, step_size, target_sum):
    """Perform a line search step."""
    h_trial = h - step_size * grad
    h_trial = project_to_feasible(h_trial, target_sum)
    return h_trial

def trust_region_update(h, grad, target_sum, max_iter=20):
    """Update h using trust region method."""
    h = h.copy()
    for _ in range(max_iter):
        # Try step sizes
        step_sizes = [0.001, 0.01, 0.1, 0.5, 1.0]
        best_h = h
        best_c5 = float('inf')
        
        for step_size in step_sizes:
            h_new = line_search_step(h, grad, step_size, target_sum)
            if np.all(h_new >= 0) and np.all(h_new <= 1):
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new
            # Stop early if small change
            if abs(step_size) < 1e-5:
                break
        if best_c5 < compute_c5(h, dx):
            h = best_h
        else:
            break
    return h

        x = np.linspace(0, 2, n_points)

        init_h = normalize_h(init_h, target_sum)
    
    h = init_h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Momentum parameters
    momentum = np.zeros(n_points)
    momentum_decay = 0.9
    step_size = 0.05
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Find the most problematic shift
        cross_corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(cross_corr)
        c5 = cross_corr[k_max] * dx
        grad = compute_gradient(h, dx, k_max)
        
        # Compute momentum
        momentum = momentum_decay * momentum + (1 - momentum_decay) * grad
        
        # Trust region update
        h = h - step_size * momentum
        h = project_to_feasible(h, target_sum)
        
        # Evaluate
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
