import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: piecewise function
        h = np.ones(n_points)
        h[n_points//4:3*n_points//4] = 0
        h[3*n_points//4:] = 1
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Core optimization parameters
    n_swaps = 1000
    swap_size = 5
    temperature = 0.1
    
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()
    
    # Swap-based local search
    for iteration in range(n_swaps):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Make a swap
        idx1 = np.random.randint(0, n_points - swap_size)
        idx2 = np.random.randint(0, n_points - swap_size)
        
        # Ensure non-overlapping
        while abs(idx1 - idx2) < swap_size:
            idx2 = np.random.randint(0, n_points - swap_size)
        
        # Swap values within segments
        h_new = current_h.copy()
        seg1 = h_new[idx1:idx1+swap_size]
        seg2 = h_new[idx2:idx2+swap_size]
        
        h_new[idx1:idx1+swap_size] = seg2
        h_new[idx2:idx2+swap_size] = seg1
        
        # Adjust to maintain integral constraint
        total_sum = np.sum(h_new)
        scale_factor = (n_points / 2) / total_sum
        h_new = h_new * scale_factor
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate
        new_c5 = compute_c5(h_new)
        
        # Accept or reject based on temperature
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / temperature):
            current_h = h_new
            current_c5 = new_c5
        
        # Track best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    # Final polishing with SLSQP
    try:
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        result = minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_c5 = compute_c5(final_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h
    except:
        pass

    return (best_h.tolist(), best_c5, n_points)
