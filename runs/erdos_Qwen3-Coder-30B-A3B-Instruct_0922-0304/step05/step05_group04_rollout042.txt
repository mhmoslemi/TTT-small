import numpy as np
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    mutation_rate = 0.1
    tournament_size = 10
    max_generations = 1000
    
    # Initialize population
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Fill rest of population with random individuals
    while len(population) < population_size:
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_h = None
    best_c5 = float('inf')
    
    # Evolution loop
    generation = 0
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        generation += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            c5 = compute_c5(h)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Selection (tournament)
        new_population = []
        
        # Keep best individuals
        sorted_indices = np.argsort(fitness)
        for i in range(min(5, len(population))):
            new_population.append(population[sorted_indices[i]].copy())
            
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            tournament_indices = random.sample(range(len(population)), tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            parent1_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            tournament_indices = random.sample(range(len(population)), tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            parent2_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            # Crossover (uniform)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            child = np.zeros(n_points)
            
            for i in range(n_points):
                if random.random() < 0.5:
                    child[i] = parent1[i]
                else:
                    child[i] = parent2[i]
                    
            # Mutation
            for i in range(n_points):
                if random.random() < mutation_rate:
                    if random.random() < 0.5:
                        child[i] += random.uniform(-0.1, 0.1)
                    else:
                        child[i] = random.choice([0, 1])
                        
            # Repair
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
            
        population = new_population
    
    # Final evaluation of best solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
