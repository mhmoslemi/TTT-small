import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_and_normalize(h, target_sum):
    """Project h to feasible region and normalize."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with available construction or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    max_iter = 1000
    iteration = 0
    
    # Each iteration can do local optimization + global refinement
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_lag_idx = np.argmax(overlap)
        # Consider a few top lags to emphasize
        top_lags = np.argpartition(overlap, -min(10, len(overlap)))[:-11:-1]
        
        # Use a target-based subgradient approach
        # Compute a loss based on top lags
        def objective(h):
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            return compute_c5(h, dx)

        # Coordinate descent with local search
        for i in range(n_points):
            if time.time() - start_time > budget_s - 2:
                break
            # Small perturbations in coordinate i
            h_base = h.copy()
            delta = 0.005
            initial_val = h[i]
            best_h_i = initial_val
            best_c5_i = np.inf
            
            for step in [-1, 0, 1]:
                h[i] = initial_val + step * delta
                h[i] = max(0, min(1, h[i]))
                new_c5 = objective(h)
                if new_c5 < best_c5_i:
                    best_c5_i = new_c5
                    best_h_i = h[i]
            
            h[i] = best_h_i
            h = project_and_normalize(h, target_sum)
            
        # Refine using scipy minimize with SLSQP only once per iteration
        h_refined = h.copy()
        
        def constraint_sum(h):
            return np.sum(h) - target_sum

        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective, h_refined, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 20})
        if result.success:
            h_refined = result.x
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = normalize_h(h_refined, target_sum)
        else:
            h_refined = h.copy()
            
        h = h_refined
        c5 = compute_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
