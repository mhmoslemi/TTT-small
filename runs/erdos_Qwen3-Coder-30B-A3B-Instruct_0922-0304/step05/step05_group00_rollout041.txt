import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def repair_feasibility(h, target_sum):
    """Repair h to be feasible: clipping and normalization."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from imported construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Differential evolution inspired search
    population_size = 30
    population = [h.copy()]
    for _ in range(population_size - 1):
        h_mut = h + np.random.normal(0, 0.02, n_points)
        h_mut = repair_feasibility(h_mut, target_sum)
        population.append(h_mut)
    
    step_size = 0.05
    step_decay = 0.98
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 5000:
        # Evaluate all population
        fitness = [compute_c5(ind, dx) for ind in population]
        
        # Update best
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_h = population[min_fit_idx].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Generate new population
        new_population = []
        
        # Keep best
        new_population.append(population[min_fit_idx].copy())
        
        # Differential evolution combinations
        for _ in range(12):  # 40% of population
            idxs = np.random.choice(len(population), 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            mutant = a + 0.8 * (b - c)
            mutant = repair_feasibility(mutant, target_sum)
            new_population.append(mutant)
            
        # Gradient descent steps (simplified)
        for _ in range(9):  # 30% of population
            ind = population[np.random.randint(len(population))]
            # Simple gradient-like update
            grad = np.zeros_like(ind)
            for i in range(n_points):
                # Approximate gradient via finite differences
                h_plus = ind.copy()
                h_plus[i] = min(1.0, h_plus[i] + 0.01)
                h_plus = repair_feasibility(h_plus, target_sum)
                h_minus = ind.copy()
                h_minus[i] = max(0.0, h_minus[i] - 0.01)
                h_minus = repair_feasibility(h_minus, target_sum)
                grad[i] = (compute_c5(h_plus, dx) - compute_c5(h_minus, dx)) / 0.02
            new_ind = ind - step_size * grad
            new_ind = repair_feasibility(new_ind, target_sum)
            new_population.append(new_ind)
            
        # Random mutations
        for _ in range(6):  # 20% of population
            idx = np.random.randint(len(population))
            ind = population[idx].copy()
            noise_idx = np.random.randint(n_points)
            ind[noise_idx] += np.random.normal(0, 0.03)
            ind = repair_feasibility(ind, target_sum)
            new_population.append(ind)
            
        # Mutation of best
        for _ in range(3):  # 10% of population
            ind = best_h.copy()
            noise_idx = np.random.randint(n_points)
            ind[noise_idx] += np.random.normal(0, 0.04)
            ind = repair_feasibility(ind, target_sum)
            new_population.append(ind)
            
        # Replace worst with new ones
        fitness = [compute_c5(ind, dx) for ind in new_population]
        sorted_indices = np.argsort(fitness)
        population = [new_population[i] for i in sorted_indices[:population_size]]
        
        step_size *= step_decay
        iteration += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
