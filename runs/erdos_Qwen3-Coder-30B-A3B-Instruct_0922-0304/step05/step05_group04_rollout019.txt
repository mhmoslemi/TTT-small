import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    mutation_rate = 0.1
    tournament_size = 10
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Fill population with random individuals
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def tournament_selection(population, tournament_size):
        tournament = np.random.choice(len(population), tournament_size)
        tournament_fitness = [compute_c5(population[i]) for i in tournament]
        winner_idx = tournament[np.argmin(tournament_fitness)]
        return population[winner_idx]

    def uniform_crossover(parent1, parent2):
        mask = np.random.rand(n_points) < 0.5
        child = np.where(mask, parent1, parent2)
        # Renormalize
        child = child * (n_points / 2) / np.sum(child)
        return child

    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                # Apply mutation
                if np.random.rand() < 0.5:
                    h[i] += np.random.normal(0, 0.1)
                else:
                    h[i] = np.random.rand()
                # Clamp and renormalize
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        return h

    best_c5 = float('inf')
    best_h = None

    gen = 0
    while time.time() - start_time < budget_s - 1:
        gen += 1

        # Evaluate population
        fitness = [compute_c5(h) for h in population]
        
        # Track best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break

        # Create new population
        new_population = []
        
        # Keep top individuals
        top_indices = np.argsort(fitness)[:population_size//2]
        for idx in top_indices:
            new_population.append(population[idx].copy())

        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, tournament_size)
            parent2 = tournament_selection(population, tournament_size)
            child = uniform_crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
            
        population = new_population
    
    # Final validation
    if best_h is not None:
        final_c5 = compute_c5(best_h)
    else:
        final_c5 = float('inf')
        best_h = population[0] if population else np.ones(n_points) * 0.5
        
    return (best_h.tolist(), final_c5, n_points)
