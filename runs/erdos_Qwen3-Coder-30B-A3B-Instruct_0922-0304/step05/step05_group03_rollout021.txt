import numpy as np
from scipy.signal import correlate
import time
from collections import defaultdict

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
    
    # Gradient-based local search
    def local_search(h):
        def objective(h):
            return compute_c5(h)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        # L-BFGS optimization
        from scipy.optimize import minimize
        result = minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        
        if result.success:
            h_new = result.x
            h_new = normalize_h(h_new, target_sum)
            return h_new
        return h
    
    # Coarse grid search with local refinement
    # Create a set of candidate solutions
    num_candidates = 100
    candidates = []
    
    # Start with an initial solution
    candidates.append(h.copy())
    
    # Generate diverse candidates using different strategies
    for _ in range(num_candidates - 1):
        # Random perturbation with some structure
        h_perturbed = h + np.random.normal(0, 0.05, n_points)
        h_perturbed = normalize_h(h_perturbed, target_sum)
        candidates.append(h_perturbed)
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Evaluate candidates
        fitness = []
        for candidate in candidates:
            c5 = compute_c5(candidate)
            fitness.append(c5)
            
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = candidates[min_fitness_idx].copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Local search on top candidates
        top_candidates = np.argsort(fitness)[:min(20, len(candidates))]
        for idx in top_candidates:
            h_local = local_search(candidates[idx])
            candidates[idx] = h_local
        
        # Perturb best solution to generate new candidates
        if iteration % 5 == 0:
            # Add new candidates by small random perturbations of best
            for _ in range(10):
                h_new = best_h + np.random.normal(0, 0.02, n_points)
                h_new = normalize_h(h_new, target_sum)
                candidates.append(h_new)
                
        # Remove worst solutions to keep population small
        if len(candidates) > num_candidates:
            worst_indices = np.argsort(fitness)[-20:]
            candidates = [candidates[i] for i in range(len(candidates)) if i not in worst_indices]
    
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
