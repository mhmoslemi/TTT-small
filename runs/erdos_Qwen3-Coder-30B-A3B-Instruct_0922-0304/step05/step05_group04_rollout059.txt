import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    best_positions = []
    best_values = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: create a simple piecewise constant function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (n_points / 2) / np.sum(h)
    
    # Initialize swarm
    particles.append(h.copy())
    best_positions.append(h.copy())
    best_values.append(float('inf'))
    
    # Generate additional random particles
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        best_positions.append(h.copy())
        best_values.append(float('inf'))
    
    # Initialize velocities
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(swarm_size)]
    
    # PSO parameters
    w_max = 0.9
    w_min = 0.4
    c1 = 2.0
    c2 = 2.0
    max_velocity = 0.1
    
    # Global best
    g_best_value = float('inf')
    g_best_position = None
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Update inertia weight
        w = w_max - (w_max - w_min) * (time.time() - start_time) / budget_s
        
        for i in range(swarm_size):
            h = particles[i]
            v = velocities[i]
            p_best = best_positions[i]
            
            # Compute current overlap
            overlap = correlate(h, 1 - h, mode="full")
            current_c5 = np.max(overlap) * dx
            
            # Update particle's personal best
            if current_c5 < best_values[i]:
                best_values[i] = current_c5
                best_positions[i] = h.copy()
            
            # Update global best
            if current_c5 < g_best_value:
                g_best_value = current_c5
                g_best_position = h.copy()
            
            # Early exit if target achieved
            if g_best_value <= 0.3807:
                break
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (w * v + 
                 c1 * r1 * (p_best - h) + 
                 c2 * r2 * (g_best_position - h))
            
            # Limit velocity
            v = np.clip(v, -max_velocity, max_velocity)
            velocities[i] = v
            
            # Update position
            h_new = h + v
            
            # Clamp
            h_new = np.clip(h_new, 0, 1)
            
            # Renormalize
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[i] = h_new
            
        if g_best_value <= 0.3807:
            break
    
    # Final computation
    if g_best_position is not None:
        overlap = correlate(g_best_position, 1 - g_best_position, mode="full")
        final_c5 = np.max(overlap) * dx
        return (g_best_position.tolist(), final_c5, n_points)
    else:
        # Fallback to best found in loop
        best_h = best_positions[np.argmin(best_values)]
        overlap = correlate(best_h, 1 - best_h, mode="full")
        final_c5 = np.max(overlap) * dx
        return (best_h.tolist(), final_c5, n_points)
