import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available

    else:

        h = np.ones(n_points)
        h[n_points//2:] = 0

    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient ascent with constraints optimization
    def optimize_step(h, dx, target_sum):
        # Project onto feasible set and then local optimization
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
            
        # Local optimization using projected gradient descent
        def objective(h):
            return evaluate_c5(h, dx)
            
        def gradient(h):
            # Simplified gradient estimation
            eps = 1e-6
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus_sum = np.sum(h_plus)
                if h_plus_sum > 0:
                    h_plus = h_plus * (target_sum / h_plus_sum)
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus_sum = np.sum(h_minus)
                if h_minus_sum > 0:
                    h_minus = h_minus * (target_sum / h_minus_sum)
                grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
            return grad
            
        # Simple gradient descent with projection
        learning_rate = 0.01
        for _ in range(50):
            grad = gradient(h)
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            if current_sum > 0:

        h_new = h + np.random.normal(0, strength, n_points)
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
        return h_new
    
    temp = 1.0
    cooling_rate = 0.995
    min_temp = 1e-5
    
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h, dx)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()

        # Generate neighbor solution
        neighbor_h = perturb(current_h, strength=0.03 * temp)
        neighbor_c5 = evaluate_c5(neighbor_h, dx)

        if neighbor_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - neighbor_c5) / temp):

        temp *= cooling_rate
    
    # Refine best solution with local optimization
    refined_h = optimize_step(best_h, dx, target_sum)
    refined_c5 = evaluate_c5(refined_h, dx)
    
    if refined_c5 < best_c5:
        best_c5 = refined_c5
        best_h = refined_h
    
    return (best_h.tolist(), best_c5, n_points)
