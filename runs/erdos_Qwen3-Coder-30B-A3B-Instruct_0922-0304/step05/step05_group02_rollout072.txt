import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum(h) = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_feasible(h, target_sum)
    else:
        h = np.ones(n_points) * 0.5
        h = project_feasible(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Define optimization variables
    h_var = Variable(n_points)
    C = Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, sum(h_var) == target_sum, C >= 0]
    
    # Solve optimization problem
    prob = Problem(Minimize(C), constraints)
    
    iteration = 0
    perturbation_magnitude = 0.1
    
    while time.time() - start_time < budget_s - 1:
        # Compute overlaps
        overlap_vals = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap_vals) * dx
        
        # Update constraint
        constraints = [h_var >= 0, h_var <= 1, sum(h_var) == target_sum, C >= max_overlap]
        prob = Problem(Minimize(C), constraints)
        
        try:
            prob.solve(solver=SCS, verbose=False, max_iters=1000)
            if prob.status == "optimal":
                h_new = h_var.value
                if h_new is not None:
                    h_new = np.array(h_new)
                    h_new = np.clip(h_new, 0, 1)
                    # Re-scale to target sum
                    current_sum = np.sum(h_new)
                    if current_sum > 0:
                        h_new = h_new * (target_sum / current_sum)
                        h_new = np.clip(h_new, 0, 1)
                    
                    # Evaluate new solution
                    new_c5 = compute_c5(h_new, dx)
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h_new.copy()
                        
                        if best_c5 <= 0.3807:
                            break
                        
        except:
            pass
        
        # Perturb and reproject
        if iteration % 10 == 0 and iteration > 0:
            noise = np.random.normal(0, perturbation_magnitude, n_points)
            h = h + noise
            h = project_feasible(h, target_sum)
            perturbation_magnitude *= 0.95
        
        h = project_feasible(h, target_sum)
        iteration += 1
        
    return (best_h.tolist(), best_c5, n_points)
