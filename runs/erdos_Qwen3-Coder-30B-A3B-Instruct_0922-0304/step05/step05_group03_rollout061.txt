import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx):
    """Compute the gradient of C5 with respect to h."""
    # Simpler approach: perturb each element and estimate gradient
    n = len(h)
    grad = np.zeros(n)
    epsilon = 1e-5
    base_c5 = evaluate_c5(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus = normalize_h(h_plus, np.sum(h))
        c5_plus = evaluate_c5(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / epsilon
    return grad

    start_time = time.time()

    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available

    def compute_c5(h):
        return evaluate_c5(h, dx)
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Learning rate and decay
    lr = 0.1
    decay_rate = 0.5
    patience = 0
    patience_limit = 50
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update best solution
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:

        # Early exit
        if best_c5 <= 0.3807:

        # Gradient-based update
        grad = compute_gradient(h, dx)
        h_new = h - lr * grad
        h_new = normalize_h(h_new, target_sum)
        
        # Coordinate-wise refinement
        for i in range(n_points):
            h_trial = h.copy()
            # Small random perturbation
            perturbation = np.random.uniform(-0.01, 0.01)
            h_trial[i] += perturbation
            h_trial = normalize_h(h_trial, target_sum)
            
            c5_trial = compute_c5(h_trial)
            if c5_trial < c5:

                c5 = c5_trial
        
        # Update h
        h = h_new
        
        # Adaptive learning rate
        if patience > patience_limit:
            lr *= decay_rate
            patience = 0
            
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
