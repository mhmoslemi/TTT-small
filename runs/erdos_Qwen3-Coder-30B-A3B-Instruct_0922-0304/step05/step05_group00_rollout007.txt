import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values or create a base function
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize sum to target
    h = h * (target_sum / np.sum(h))
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Main optimization loop
    iteration = 0
    max_iter = 50  # Limited iterations to respect time budget
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate current solution
        overlap = correlate(h, 1 - h, mode="full")
        c5_current = np.max(overlap) * dx
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
        
        # Select top lags contributing to maximum overlap
        # Use only top 10% of lags to reduce dimensionality
        lag_indices = np.argsort(overlap)[-max(1, n_points//10):]
        
        # Prepare LP problem
        h_var = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_var) == target_sum,
            h_var >= 0,
            h_var <= 1,
            C5 >= 0
        ]
        
        # Add overlap constraints for selected lags
        for k_idx in lag_indices:
            # The overlap for lag k_idx
            # Using linearization: sum h[i] * (1 - h[i+k]) + h[i+k] * h[i] - h[i] * h[i+k]  
            # Simplified to: sum h[i] * (1 - h[i+k]) + h[i+k] * h[i] - h[i] * h[i+k]
            # = h[i] - h[i]*h[i+k] + h[i+k]*h[i] - h[i]*h[i+k] = h[i] - h[i]*h[i+k]
            # Actually, it's sum_{i} h[i] * (1 - h[i+k]) = sum h[i] - sum h[i]*h[i+k]
            # So overlap_k = sum h[i] - sum h[i]*h[i+k]
            # Linearize: sum h[i] - sum h[i]*h[i+k]) <= C5*dx
            
            # Approximate sum h[i]*h[i+k] via linearization around current h
            # linear_term = (h[i] * h[i+k] + h[i] * h[i+k])/2
            # Around current h: h[i] * h[i+k] ≈ h[i] * h[k] + h[k] * (h[i] - h[i]) = h[i] * h[k]
            # Rather, rewrite exactly: 
            # Let's use piecewise linear relaxation
                
            # Alternative: compute exact linearization 
            # sum h[i] * (1 - h[i+k]) = sum h[i] - sum h[i] * h[i+k] 
            # Using a linear relaxation for h[i] * h[i+k]: 
            # h[i]*h[i+k] ≈ upper_bound([-h[i], -h[i+k]]) + lower_bound([h[i], h[i+k]])
            # = (h[i] + h[i+k] - h[i]*h[i+k]) / 2  (this doesn't work directly)
            
            # Better: rearrange term
            # expr = sum(h[i] - h[i]*h[i+k])
            # We want to bound this linearly. 
            # Take a fixed h[i+k] value, then sum h[i] - h[i]*h[i+k] = sum h[i](1-h[i+k]) 
            # To drive this towards the minimum of the constraint, bounds can be constructed.
            
            # Simplified approach: treat the final constraint as sum h[i] * (1 - h[i+k]) <= C5*dx
            # We linearize: 
            # h[i] * (1 - h[i+k]) >= h[i] - h[i] * h[i+k]
            # Use -> h[i] * h[i+k] <= (h[i] + h[i+k]) / 2  -- Example relaxation
            # Actually linearization must be over all pairs
            
            # Work from the accumulated coefficient calculation  
            # All simplified approach: 
            # Maintain exact coefficients for those pairs
                
            # This is a complex part - simpler:
            # Just focus on computing overlap exactly and trying to reduce it.
            
            # Use the known overlaps from iterate (but since we're doing LP, 
            # again this is tricky because of non-smoothness)
            # Linearization by terms: 
            # sum_{i=0}^{n_points-1} h[i] * (1 - h[i+k]) 
            # sum_{i=0}^{n_points-1} h[i] - sum_{i=0}^{n_points-1} h[i] * h[i+k]
            # = sum h[i] - sum h[i] * h[i+k]
            # If h[i] ~ h[i] from current iteration, then we have linearization:
            # h[i] * h[i+k] <= 0.5 * (h[i]^2 + h[i+k]^2)  for bound (excessively loose)
            # Or use simpler: 
            # sum h[i] * (1 - h[i+k]) ≈ sum h[i] - sum  h[i] * h[i+k]
            # Fix all h[i+k] for this lag, so the sum h[i] * h[i+k] ≈ sum h[i] * h[i+k] 
            # Relaxation approach:
            
            # Approximate the total sum by:
            #  Let left_side =  h[i]* (1 - h[i+k]) for i=0 to n_points-1
            #  Approximate each product h[i]*h[i+k] with a linear expression involving both
            
            constraint_expr = 0
            # Sum 
            for i in range(n_points):
                if i + k_idx < n_points:
                    # Approximate h[i]* h[i+k] linearly  
                    # Central difference approximation or coordinate-wise min/max bound
                    # Full linearization: 
                    # h[i] * h[i+k ] <= min(h[i], h[i+k]) + h[i]*h[i+k] - min(h[i], h[i+k])
                    # We don't add that to subproblems easily. Let's not do parabolic approx.
                    
                    constraint_expr += h[i] * (1 - h[i+k_idx])
                else:
                    # For boundary indices, ignore as they don't overlap
                    pass
            
            # Our final constraint:
            # constraint_expr <= C5 * dx
            constraints.append(constraint_expr <= C5 * dx)
            
        
        # Objective
        objective = cp.Minimize(C5)
        
        # Form and solve the problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=100, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_var.value
                # Normalize again to ensure sum correct
                new_h = new_h * (target_sum / np.sum(new_h))
                h = np.clip(new_h, 0, 1)
            else:
                # If failed, fallback to simple projection
                h = np.clip(h, 0, 1)
                h = h * (target_sum / np.sum(h)) 
        except Exception:
            # fallback to projection
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
            
        iteration += 1
        
    # Validate solution
    final_overlap = correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(final_overlap) * dx 
    
    return (best_h.tolist(), final_c5, n_points)
