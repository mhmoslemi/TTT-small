import numpy as np
from scipy.signal import correlate
import time
from collections import defaultdict
import math

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 1000
    
    # Check for initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
    else:
        # Fallback to triangle function
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(mid):
            h[i] = i / mid
            h[n_points - 1 - i] = i / mid
        h[mid] = 1.0
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Local search with pattern mutations
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Generate neighbor by perturbing with pattern
        h_neighbor = h.copy()
        
        # Select a pattern type
        pattern_type = np.random.choice(['smooth', 'step', 'oscillate'])
        
        if pattern_type == 'smooth':
            # Apply smooth perturbation
            perturb = np.random.normal(0, 0.02, n_points)
            h_neighbor += perturb
        elif pattern_type == 'step':
            # Modify a portion with step function
            start = np.random.randint(0, n_points - 10)
            end = min(start + 10 + np.random.randint(0, 15), n_points)
            step_height = np.random.normal(0, 0.1)
            h_neighbor[start:end] += step_height
        else:  # oscillate
            # Apply oscillatory perturbation
            freq = np.random.randint(1, 5)
            phase = np.random.random() * 2 * np.pi
            oscillation = np.sin(np.arange(n_points) * freq * 0.5 + phase) * 0.05
            h_neighbor += oscillation
            
        # Ensure bounds
        h_neighbor = np.clip(h_neighbor, 0, 1)
        # Ensure sum constraint
        current_sum = np.sum(h_neighbor)
        if current_sum > 0:
            h_neighbor = h_neighbor * (target_sum / current_sum)
        
        # Accept if improvement or with probability
        new_c5 = compute_c5(h_neighbor)
        if new_c5 < current_c5 or np.random.random() < 0.1:
            h = h_neighbor
            
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
