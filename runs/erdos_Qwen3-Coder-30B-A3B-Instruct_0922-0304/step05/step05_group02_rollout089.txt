import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

    # so gradient components are: 1 - 2*h[i+shift_idx] for i < n-shift_idx
    # and -h[i-shift_idx] for i >= shift_idx (due to 1-h backwards)
    if shift_idx == 0:
        # This would require careful handling, so we skip it for this approach
        return np.zeros(n)
    elif shift_idx < n:
        # Gradient at position i contributes (1 - 2*h[i+shift_idx]) if we look at
        # the mixing trick more carefully, but it's difficult to do a proper
        # explicit gradient w.r.t. to overlap for a single shift at once.
        # So, we combine two related shifted versions as an approximation.

        # For a concrete direction, we compute gradient via finite differences
