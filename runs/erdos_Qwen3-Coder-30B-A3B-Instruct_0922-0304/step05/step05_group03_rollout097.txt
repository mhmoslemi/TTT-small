import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    # Simulated Annealing approach
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Parameters for simulated annealing
    temp = 1.0
    min_temp = 1e-8
    cooling_rate = 0.999
    iterations_per_temp = 100
    
    # Track best solution
    current_h = h.copy()
    current_c5 = best_c5
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Perturb solution
        new_h = current_h.copy()
        idx = np.random.randint(0, n_points)
        # Small random change
        delta = np.random.normal(0, 0.02)
        new_h[idx] += delta
        # Ensure within bounds and normalize
        new_h = normalize_h(new_h, target_sum)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(new_h, dx)
        
        # Accept or reject based on simulated annealing
        if new_c5 < current_c5 or np.random.random() < np.exp((current_c5 - new_c5) / temp):
            current_h = new_h
            current_c5 = new_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                
                # Early exit if target reached
                if best_c5 <= 0.3807:
                    break
        
        #降温
        if iteration % iterations_per_temp == 0:
            temp *= cooling_rate
            if temp < min_temp:
                temp = min_temp
                
    return (best_h.tolist(), best_c5, n_points)
