import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def repair_feasibility(h, target_sum):
    """Repair h to ensure it's in [0,1] and sums to target_sum."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize using initial_h_values or a simple construction
    if 'initial_h_values' in globals():

    else:
        # Simple step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Differential evolution inspired search
    population_size = 30
    population = [h.copy()]
    for _ in range(population_size - 1):
        perturbed = h + np.random.normal(0, 0.02, n_points)
        perturbed = repair_feasibility(perturbed, target_sum)
        population.append(perturbed)
        
    # Step size for local search
    step_size = 0.01
    step_decay = 0.98
    max_iter = 5000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = [compute_c5(ind, dx) for ind in population]
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_h = population[min_fit_idx].copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Generate new population
        new_population = [best_h.copy()]  # Elitism
        
        # 40%: Differential evolution
        for _ in range(int(0.4 * (population_size - 1))):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            v = population[a] + 0.8 * (population[b] - population[c])
            v = repair_feasibility(v, target_sum)
            new_population.append(v)
            
        # 30%: Gradient-based (limited)
        for _ in range(int(0.3 * (population_size - 1))):
            ind = population[np.random.randint(population_size)].copy()
            # Simple gradient step towards reducing C5
            # We perturb in random directions
            pert = np.random.normal(0, step_size, n_points) 
            ind += pert
            ind = repair_feasibility(ind, target_sum)
            new_population.append(ind)
            
        # 20%: Random samples
        for _ in range(int(0.2 * (population_size - 1))):
            rand_h = np.random.rand(n_points)
            rand_h = repair_feasibility(rand_h, target_sum)
            new_population.append(rand_h)
            
        # 10%: Mutate best
        for _ in range(int(0.1 * (population_size - 1))):
            ind = best_h + np.random.normal(0, step_size, n_points)
            ind = repair_feasibility(ind, target_sum)
            new_population.append(ind)
            
        # Ensure we have the right population size
        while len(new_population) < population_size:
            new_population.append(repair_feasibility(np.random.rand(n_points), target_sum))
            
        population = new_population[:population_size]
        
        # Adjust step size
        step_size *= step_decay
    
    # Final evaluation of best
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
