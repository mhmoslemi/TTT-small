import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    generations = 1000
    mutation_rate = 0.1
    tournament_size = 10
    
    # Initialize population
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    else:
        # Generate random individuals
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Ensure we have at least one individual
    if not population:
        h = np.ones(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Fitness function
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_h = None
    best_c5 = float('inf')
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for entire population
        fitness_scores = [compute_c5(h) for h in population]
        
        # Track best solution
        for i, fitness in enumerate(fitness_scores):
            if fitness < best_c5:
                best_c5 = fitness
                best_h = population[i].copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Selection using tournament
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_index].copy())
            
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < population_size else selected[0]
            
            # Uniform crossover
            mask = np.random.rand(n_points) < 0.5
            child1 = np.where(mask, parent1, parent2)
            child2 = np.where(mask, parent2, parent1)
            
            # Mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    if np.random.rand() < 0.5:
                        child1[j] += np.random.normal(0, 0.1)
                    else:
                        child1[j] = np.random.rand() * 2 - 1  # Random value in [-1,1]
                    
                    if np.random.rand() < 0.5:
                        child2[j] += np.random.normal(0, 0.1)
                    else:
                        child2[j] = np.random.rand() * 2 - 1  # Random value in [-1,1]
            
            # Repair and renormalize
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.extend([child1, child2])
            
        # Trim to exact population size
        population = new_population[:population_size]
        
        # Keep updating the best with best individuals from past generations
        if gen > 0 and gen % 10 == 0:
            # Add some diversity
            for _ in range(5):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                population.append(h)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
