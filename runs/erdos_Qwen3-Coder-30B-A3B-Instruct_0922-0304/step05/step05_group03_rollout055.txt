import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

        # Since h[i] is in numerator of lag k_max correlation, and
        # it's 1-h[i+k_max], we have: 
        # d/(dh[i]) \int h(x)(1-h(x+k_max)) dx
        # If k_max >= 0 then h[i] contributes to h[i]*(1-h[i+k_max]) 
        # So gradient is 1-h[i+k_max] (when i+k_max < n_points)
        if i + k_max < n_points:
            grad[i] = 1 - h[i + k_max]

        # For the circular nature, we still should consider the wraps

    return grad
