import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Binary search parameters
    c5_low = 0.0
    c5_high = 0.38085  # current upper bound
    tolerance = 1e-4
    max_iterations = 40
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5_mid = (c5_low + c5_high) / 2.0
        
        # Define constraints for feasibility check
        def constraint_max_overlap(h):
            overlap = correlate(h, 1 - h, mode="full") * dx
            return c5_mid - np.max(overlap)
        
        constraints = [
            {'type': 'eq', 'fun': constraint_sum},
            {'type': 'ineq', 'fun': constraint_max_overlap}
        ]
        
        # Define bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try to find a feasible solution
        try:
            result = minimize(
                lambda x: -np.sum(x),  # minimize negative sum (to keep h balanced)
                h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            if result.success:
                h_candidate = result.x
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                current_c5 = compute_c5(h_candidate)
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_candidate.copy()
                
                if current_c5 <= c5_mid:
                    c5_high = c5_mid
                else:
                    c5_low = c5_mid
                    
                if abs(c5_high - c5_low) < tolerance:
                    break
            else:
                c5_low = c5_mid
                
        except:
            c5_low = c5_mid
            
        # Perturbation strategy
        if time.time() - start_time > budget_s - 1:
            break
            
        # Introduce random perturbation
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Final check of current solution if it's better
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
    # Final refinement
    def objective(h):
        return compute_c5(h)

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    try:
        result = minimize(
            objective,
            best_h.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-10}
        )
        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_c5 = compute_c5(final_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h
    except:
        pass
        
    return (best_h.tolist(), best_c5, n_points)
