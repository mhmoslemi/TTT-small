import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum(h) = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, target_sum)
    else:
        # Random initialization
        h = np.random.rand(n_points)
        h = project_h(h, target_sum)
    
    # Optimization parameters
    lr = 0.01
    max_iter = 1000
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_overlap = overlap[max_idx] * dx
        
        # Update best if improved
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early stopping for target
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient direction for the worst case shift
        # The gradient is (1 - h_shifted) for the shift that gives the max overlap
        shift = max_idx - (len(h) - 1)
        if shift >= 0:
            grad = np.zeros(len(h))
            grad[shift:] = 1 - h[:-shift]
        else:
            grad = np.zeros(len(h))
            grad[:shift] = 1 - h[-shift:]
        
        # Gradient descent step
        h = h - lr * grad
        
        # Project to feasible region
        h = project_h(h, target_sum)
        
        # Adaptive learning rate
        lr = max(0.001, lr * 0.999)
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
