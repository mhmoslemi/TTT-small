import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Core optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Solve convex relaxation
        variables = cp.Variable(n_points)
        constraints = [variables >= 0, variables <= 1, cp.sum(variables) == n_points / 2]
        
        # Objective: minimize max of correlation
        # This needs approximation because cp.max is not allowed directly
        # We'll use a heuristic approach with iterative refinement
        h_current = h.copy()
        overlap = correlate(h_current, 1 - h_current, mode="full")
        max_overlap = np.max(overlap)
        target = max_overlap * dx
        
        # Approximate the optimization via a quadratic surrogate
        # Minimize sum of squares of overlap deviation from target
        objective = cp.Minimize(cp.sum_squares(cp.convolve(variables, 1 - variables) - target))
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h = variables.value
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # fallback to a slight perturbation if solver fails
                h = h + np.random.normal(0, 0.005, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            h = h + np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit check
        if best_c5 <= 0.3807:
            break

    # Final evaluation with potentially better solution
    if best_h is None:
        best_h = h.copy()
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
