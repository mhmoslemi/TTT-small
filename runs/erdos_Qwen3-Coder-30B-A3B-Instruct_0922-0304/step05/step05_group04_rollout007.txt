import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 50
    max_iterations = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 += np.random.normal(0, 0.05, n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Create initial swarm
    particles.append(h0.copy())
    velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Add random particles
    for _ in range(swarm_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        
    # Track best solutions
    best_particle = particles[0].copy()
    best_c5 = float('inf')
    global_best_c5 = float('inf')
    
    # PSO parameters
    omega_max = 0.9
    omega_min = 0.4
    phi1 = 2.0
    phi2 = 2.0
    v_max = 0.1
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Main loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        omega = omega_max - (omega_max - omega_min) * (time.time() - start_time) / budget_s
        
        for i in range(swarm_size):
            h = particles[i]
            v = velocities[i]
            
            # Compute current C5
            current_c5 = compute_c5(h)
            
            # Update personal best
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_particle = h.copy()
                
            # Update global best
            if current_c5 < global_best_c5:
                global_best_c5 = current_c5
                
            # Early termination
            if global_best_c5 <= 0.3807:
                break
                
            # Update velocity
            r1, r2 = np.random.rand(2)
            pbest = best_particle
            gbest = best_particle
            
            new_v = (omega * v + 
                     phi1 * r1 * (pbest - h) + 
                     phi2 * r2 * (gbest - h))
            
            # Clamp velocity
            new_v = np.clip(new_v, -v_max, v_max)
            
            # Update position
            new_h = h + new_v
            
            # Clamp and renormalize
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            # Store updated values
            particles[i] = new_h
            velocities[i] = new_v
            
        if global_best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_particle)
    
    return (best_particle.tolist(), final_c5, n_points)
