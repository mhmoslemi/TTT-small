import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_feasible(h, target_sum):
    """Project h to feasible region with sum = target_sum and 0 ≤ h_i ≤ 1."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        # Clip again in case scaling made it invalid
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values)
        init_h = np.clip(init_h, 0, 1)
        # Normalize to have correct sum
        init_h = project_feasible(init_h, target_sum)
    else:
        # Start with a uniform distribution scaled to sum
        init_h = np.ones(n_points) * 0.5
        init_h = project_feasible(init_h, target_sum)
    
    best_h = init_h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Early exit if target already met
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    iteration = 0
    perturbation_magnitude = 0.1
    
    while time.time() - start_time < budget_s - 1:
        # Occasionally do small random perturbations
        if iteration % 50 == 0 and iteration > 0:
            perturbation_magnitude *= 0.9
            noise = np.random.normal(0, perturbation_magnitude, n_points)
            new_h = best_h + noise
            best_h = project_feasible(new_h, target_sum)
            current_c5 = compute_c5(best_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
        
        # Compute current overlaps
        # We want to minimize: max_k (h * (1-h))(k) * dx
        # which is equivalent to minimizing max_k of cross-correlation terms
        # We'll compute the correlation terms for all shifts and set up a convex optimization
        
        # Compute cross-correlations for all shifts using full mode
        h_arr = best_h
        corr = correlate(h_arr, 1 - h_arr, mode="full")
        # corr is length 2*n_points - 1, center is at n_points-1
        
        # Compute upper bounds on each correlation term
        corr_full = corr * dx  # Scale by dx
        
        # Try to solve a convex relaxation:
        # min C
        # s.t. C >= corr_full[k] for all k
        #      sum(h) = target_sum
        #      0 <= h[i] <= 1
        
        # Construct symmetric problem using CVXPY
        
        # Variables
        h_var = cp.Variable(n_points)
        C_var = cp.Variable()
        
        # Objective
        objective = cp.Minimize(C_var)
        
        # Constraints
        constraints = [
            C_var >= corr_full[k] for k in range(len(corr_full))
        ]
        constraints.append(cp.sum(h_var) == target_sum)
        constraints.append(h_var >= 0)
        constraints.append(h_var <= 1)
        
        # Solve optimization
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=5000)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                # Re-scale to target sum
                h_new = project_feasible(h_new, target_sum)
                current_c5 = compute_c5(h_new, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_new.copy()
            else:
                # Fallback to simple perturbation if optimization fails
                noise = np.random.normal(0, 0.01, n_points)
                h_new = best_h + noise
                h_new = project_feasible(h_new, target_sum)
                current_c5 = compute_c5(h_new, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_new.copy()
        except:
            # If optimization fails, fall back to simple change
            noise = np.random.normal(0, 0.01, n_points)
            h_new = best_h + noise
            h_new = project_feasible(h_new, target_sum)
            current_c5 = compute_c5(h_new, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_new.copy()
        
        iteration += 1
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
