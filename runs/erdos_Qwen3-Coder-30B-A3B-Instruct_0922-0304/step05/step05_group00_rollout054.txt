```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative improvement with local optimization
    max_iter = 1000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Local optimization using SLSQP
        def objective(h):
            return compute_c5(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        # Define bounds
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Optimization step
        result = minimize(
            objective, 
            h.copy(), 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraint, 
            options={'maxiter': 50}
        )
        
        if result.success:
            h = np.clip(result.x, 0, 1)
            h = normalize_h(h, target_sum)
        
        # Evaluate current solution
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early exit on target
        if best_c5 <= 0.3807:
            break
            
        # Perturb the solution to escape local optima
        perturbation_magnitude = 0.01
        h = h + np.random.normal(0, perturbation_magnitude, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
```
