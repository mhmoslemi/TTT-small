from scipy.signal import correlate
from scipy.optimize import minimize
import time

    start_time = time.time()

    dx = 2.0 / n_points

        # Repair if necessary
        h = np.clip(h, 0, 1)

    else:

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-resolution approach
    n_coarse = n_points // 2
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.zeros(n_coarse)
    
    # Initialize coarse solution
    if 'initial_h_values' in globals():
        h_coarse = np.array(initial_h_values[:n_coarse])
        h_coarse = np.clip(h_coarse, 0, 1)
        h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    else:
        h_coarse = np.ones(n_coarse)
        h_coarse[n_coarse//2:] = 0
        h_coarse += np.random.normal(0, 0.05, n_coarse)
        h_coarse = np.clip(h_coarse, 0, 1)
        h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    
    # Coarse optimization
    def compute_c5_coarse(h_coarse):
        h_fine = np.repeat(h_coarse, 2)
        overlap = correlate(h_fine, 1 - h_fine, mode="full")
        return np.max(overlap) * dx_coarse

    def constraint_sum_coarse(h_coarse):
        return np.sum(h_coarse) - n_coarse / 2

    constraints_coarse = [{'type': 'eq', 'fun': constraint_sum_coarse}]
    bounds_coarse = [(0, 1) for _ in range(n_coarse)]

    # First coarse optimization
    result_coarse = minimize(
        compute_c5_coarse, 
        h_coarse, 
        method='SLSQP', 
        bounds=bounds_coarse, 
        constraints=constraints_coarse,
        options={'maxiter': 50, 'ftol': 1e-6}
    )

    if result_coarse.success:
        h_coarse = result_coarse.x
        h_coarse = np.clip(h_coarse, 0, 1)
        h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)

    # Upsample coarse solution to fine grid
    h_fine = np.repeat(h_coarse, 2)
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)

    # Fine optimization with refined starting point
    def compute_c5_fine(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Hyperparameters for optimization
    max_iter = 200
    no_improvement_count = 0

        current_c5 = compute_c5_fine(h_fine)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_fine.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:

        if no_improvement_count > 10:

        result = minimize(
            compute_c5_fine, 
            h_fine, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        
        if result.success:
            h_fine = result.x
            # Ensure feasibility
            h_fine = np.clip(h_fine, 0, 1)
            h_fine = h_fine * (n_points / 2) / np.sum(h_fine)

            h_fine += np.random.normal(0, 0.005, n_points)
            h_fine = np.clip(h_fine, 0, 1)
            h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    
    # Final evaluation
    final_c5 = compute_c5_fine(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
