import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    """Evaluate objective and check constraints."""
    c5 = compute_c5(h, dx)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a simple initial construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Differential evolution inspired search
    population_size = 30
    population = []
    
    # Initialize population
    for _ in range(population_size):
        h_temp = h.copy() + np.random.normal(0, 0.05, n_points)
        h_temp = np.clip(h_temp, 0, 1)
        h_temp = normalize_h(h_temp, target_sum)
        population.append(h_temp)
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitness = [evaluate_individual(ind, dx) for ind in population]
        
        # Update best solution
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_h = population[min_fit_idx].copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Create new population
        new_population = []
        
        # Elitism: keep best individual
        new_population.append(best_h.copy())
        
        # Generate offspring via different mechanisms
        for _ in range(population_size - 1):
            method = np.random.choice([
                'de',          # Differential evolution
                'gd',          # Gradient descent
                'random',      # Random mutation
                'best_mutation' # Mutate best solution
            ], p=[0.4, 0.3, 0.2, 0.1])
            
            if method == 'de':
                # Differential evolution
                a, b, c = np.random.choice(len(population), 3, replace=False)
                mutant = population[a] + 0.8 * (population[b] - population[c])
                mutant = np.clip(mutant, 0, 1)
                mutant = normalize_h(mutant, target_sum)
                new_population.append(mutant)
                
            elif method == 'gd':
                # Gradient descent on a local neighborhood
                idx = np.random.randint(len(population))
                h_current = population[idx].copy()
                
                def objective(h):
                    return compute_c5(h, dx)
                
                # Simple gradient descent with different step sizes
                step_size = 0.02
                for _ in range(20):
                    h_new = h_current - step_size * (h_current - 0.5)  # Simple gradient
                    h_new = np.clip(h_new, 0, 1)
                    h_new = normalize_h(h_new, target_sum)
                    if compute_c5(h_new, dx) < compute_c5(h_current, dx):
                        h_current = h_new
                    else:
                        break
                        
                new_population.append(h_current)
                
            elif method == 'random':
                # Random mutation
                h_temp = h.copy() + np.random.normal(0, 0.08, n_points)
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = normalize_h(h_temp, target_sum)
                new_population.append(h_temp)
                
            else:  # best_mutation
                # Mutate best solution
                h_temp = best_h.copy() + np.random.normal(0, 0.05, n_points)
                h_temp = np.clip(h_temp, 0, 1)
                h_temp = normalize_h(h_temp, target_sum)
                new_population.append(h_temp)
        
        # Replace population
        population = new_population
        
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
