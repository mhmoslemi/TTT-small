import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # PSO parameters
    n_particles = 20
    max_iter = 1000
    w_start = 0.8
    w_end = 0.1
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    for i in range(n_particles):
        if i == 0 and 'initial_h_values' in globals():
            particle = h.copy()
        else:
            particle = h + np.random.normal(0, 0.02, n_points)
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle)
        velocities.append(np.random.uniform(-0.02, 0.02, n_points))
        p_best.append(particle.copy())
        p_best_vals.append(compute_c5(particle))
    
    # Global best
    global_best_idx = np.argmin(p_best_vals)
    g_best = p_best[global_best_idx].copy()
    g_best_val = p_best_vals[global_best_idx]
    
    best_h = g_best.copy()
    best_c5 = g_best_val
    
    # PSO loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start - (w_start - w_end) * (iter_num / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel_cognitive = c1 * r1 * (p_best[i] - particles[i])
            vel_social = c2 * r2 * (g_best - particles[i])
            velocities[i] = w * velocities[i] + vel_cognitive + vel_social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds and normalization
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            c5 = compute_c5(particles[i])
            
            # Update personal best
            if c5 < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = c5
                
                # Update global best
                if c5 < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = c5
                    best_h = g_best.copy()
                    best_c5 = g_best_val
                    
            # Early stopping
            if best_c5 <= 0.3807:
                break
                
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
