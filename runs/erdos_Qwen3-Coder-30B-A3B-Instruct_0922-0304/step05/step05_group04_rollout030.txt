import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    best_positions = []
    best_values = []
    
    # Use initial_h_values if available as first particle
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Fallback initialization
        h_init = np.random.uniform(0, 1, n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    particles.append(h_init.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    best_positions.append(h_init.copy())
    best_values.append(float('inf'))
    
    # Add more random particles
    for _ in range(swarm_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        best_positions.append(h.copy())
        best_values.append(float('inf'))
    
    # PSO parameters
    w_max = 0.9
    w_min = 0.4
    phi1 = 2.0
    phi2 = 2.0
    v_max = 0.1
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_max - (w_max - w_min) * (time.time() - start_time) / budget_s
        
        for i in range(swarm_size):
            h = particles[i]
            
            # Evaluate current particle
            overlap = correlate(h, 1 - h, mode="full")
            current_c5 = np.max(overlap) * dx
            
            # Update personal best
            if current_c5 < best_values[i]:
                best_values[i] = current_c5
                best_positions[i] = h.copy()
            
            # Find global best
            global_best_idx = np.argmin(best_values)
            global_best_h = best_positions[global_best_idx]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           phi1 * r1 * (best_positions[i] - h) + 
                           phi2 * r2 * (global_best_h - h))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            new_h = h + velocities[i]
            
            # Clamp to [0,1]
            new_h = np.clip(new_h, 0, 1)
            
            # Renormalize
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            particles[i] = new_h
            
            # Early exit if target achieved
            if best_values[global_best_idx] <= 0.3807:
                break
                
        if best_values[global_best_idx] <= 0.3807:
            break
    
    # Return best solution
    final_h = best_positions[global_best_idx]
    final_c5 = best_values[global_best_idx]
    
    return (final_h.tolist(), final_c5, n_points)
