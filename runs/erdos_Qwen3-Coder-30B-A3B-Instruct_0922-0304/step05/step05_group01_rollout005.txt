import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def validate_and_repair(h, n_points):
    """Validate and repair initial solution"""
    h = np.array(h)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def initialize_coarse(n_coarse=35):
    """Initialize coarse solution"""
    # Start with a simple construction
    h = np.ones(n_coarse)
    h[n_coarse//2:] = 0
    h = h * (n_coarse / 2) / np.sum(h)
    return h

def upscale(h_coarse, n_points):
    """Upsample coarse solution to fine grid"""
    # Upsample using nearest neighbor interpolation
    h_fine = np.repeat(h_coarse, 2)
    # Ensure proper sum
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    return h_fine

def compute_c5(h, dx=2.0/len(h)):
    """Compute the C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def optimize(h, n_points, max_iter=200, tol=1e-8):
    """Optimize h with constraints"""
    dx = 2.0 / n_points
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    def obj(h):
        return compute_c5(h, dx)
        
    result = minimize(obj, h, method='SLSQP', bounds=bounds, 
                      constraints=constraints, options={'maxiter': max_iter, 'ftol': tol})
    
    if result.success:
        h = result.x
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h
