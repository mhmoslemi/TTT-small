import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize the solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base solution with a Gaussian-like shape
        indices = np.arange(n_points)
        mu = n_points / 2
        sigma = n_points / 6
        base = np.exp(-0.5 * ((indices - mu) / sigma) ** 2)
        base = np.clip(base, 0, 1)
        h = normalize_h(base, target_sum)
        
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Main optimization loop
    iter_count = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Apply small random perturbations
        noise = np.random.normal(0, 0.015, n_points)
        h_pert = h + noise
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = normalize_h(h_pert, target_sum)
        
        # Local optimization using SLSQP
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return compute_c5(h, dx)
        
        result = minimize(objective, h_pert, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
        if result.success:
            h_local = result.x
            h_local = np.clip(h_local, 0, 1)
            h_local = normalize_h(h_local, target_sum)
            c5_local = compute_c5(h_local, dx)
            if c5_local < best_c5:
                best_c5 = c5_local
                best_h = h_local.copy()
        
        # Refinement: also try optimizing the top k lags where overlap is largest
        overlap = correlate(h, 1 - h, mode="full")
        k_indices = np.argsort(overlap)[-10:]  # Top 10 lag positions
        k_indices = k_indices[overlap[k_indices] > 0]  # Only positive overlaps
        
        # Try a targeted local search on top contributing lags
        num_lags = min(len(k_indices), 15)
        if num_lags > 0:
            # Compute lags to sample
            lags = k_indices[-num_lags:]
            # Refine using SLSQP with fewer variables
            h_refined = h.copy()
            # We will try to reduce overlap on selected lags
            for i in range(20):
                if time.time() - start_time > budget_s - 1:
                    break
                idx = np.random.choice(lags)
                # Directly modify h to reduce the overlap at this lag
                h_trial = h_refined.copy()
                # Add a bit of backwards shift in replacement
                start_idx = idx
                end_idx = min(start_idx + 10, len(h_trial))
                for j in range(start_idx, end_idx):
                    if j < len(h_trial):
                        h_trial[j] = max(0, h_trial[j] - 0.02 * (1 if h_trial[j] > 0 else 0))
                h_trial = normalize_h(h_trial, target_sum)
                c5_trial = compute_c5(h_trial, dx)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                if c5_trial < compute_c5(h_refined, dx):
                    h_refined = h_trial
        
        h = best_h.copy()
        iter_count += 1

        if best_c5 <= 0.3807:
            break

    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
