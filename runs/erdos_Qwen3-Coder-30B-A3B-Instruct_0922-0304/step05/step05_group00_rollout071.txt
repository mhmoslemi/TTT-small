import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    """Evaluate objective and check constraints."""
    c5 = compute_c5(h, dx)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Start with initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Optimizer parameters
    max_iter = 1000
    step_size = 0.01
    smoothing = 0.005
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:

        # Compute current overlaps
        overlapping = correlate(h, 1 - h, mode="full")
        max_overlaps = np.max(overlapping)
        
        # Try gradient descent
        gradient = np.zeros_like(h)
        
        # Compute gradient estimation through finite differences
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = normalize_h(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = normalize_h(h_minus, target_sum)
            c5_minus = compute_c5(h_minus, dx)
            
            gradient[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update with gradient descent
        h_new = h - step_size * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        # Add some smoothing
        h_new = h_new * (1 - smoothing) + h * smoothing
        
        # Test new solution
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            h = h_new.copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
        
    return (best_h.tolist(), best_c5, n_points)
