```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a base step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        
    # Normalize to meet sum constraint
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps to find significant lags
        overlap = correlate(h, 1 - h, mode="full")
        # Only consider non-negative lags
        overlap = overlap[len(overlap)//2:]
        
        # Get top 10% of lags
        num_top_lags = max(1, len(overlap) // 10)
        top_lags = np.argsort(overlap)[-num_top_lags:][::-1]
        
        # Create LP model
        h_vars = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == target_sum,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add overlap constraints
        for k in top_lags:
            # Linear approximation of overlap_k
            # overlap_k = sum(h_vars[i] * (1 - h_vars[i + k]) - h[i] * (h_vars[i + k] - h[i]))
            # But the current h is fixed, so treating it as constant:
            a = 0.0
            b = 0.0
            for i in range(n_points - k):
                a += h[i] * (1 - h[i + k])
                b += h[i + k] * h[i]
            c = a - b
            
            # Linear approximation loss:
            # overlap_k ≈ sum(h_vars[i] * (1 - h[i + k]) + h[i] * (h[i + k] - h_vars[i + k]))
            # Simplifies to: sum(h_vars[i] * (1 - h[i + k]) - h[i] * h[i + k] + h[i] * h_vars[i + k])
            # Which is sum(h_vars[i] * (1 - h[i + k]) - h[i] * h[i + k] + h[i] * h_vars[i + k])
            # = sum(h_vars[i] * (1 - h[i + k]) - h[i] * h[i + k]) + sum(h[i] * h_vars[i + k])
            
            # Actually we just want to constrain:
            # sum(h_vars[i] * (1 - h[i + k])) <= C5 + sum(h[i] * h[i + k]) - sum(h[i]) 
            # Slightly expensive way: constraint: sum(h_vars[i] * (1 - h[i + k]) - h[i] * (\h[i + k])) <= C5 
            # Generate a vector of coefficients for h_vars
            coeff = np.zeros(n_points)
            for i in range(n_points - k):
                coeff[i] += (1 - h[i + k])
                coeff[i + k] -= h[i]
            
            constraints.append(cp.sum(cp.multiply(coeff, h_vars)) <= C5 + c)
        
        # Objective
        prob = cp.Problem(cp.Minimize(C5), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = np.array(h_vars.value).flatten()
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (target_sum / np.sum(h_new))  # normalize
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new.copy()
        except Exception:
            pass
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
```
