import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
    
    # Normalize
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    # Reflection-based local search
    def reflect(h, idx, delta):
        h_new = h.copy()
        h_new[idx] = np.clip(h[idx] + delta, 0, 1)
        return h_new
    
    def optimize_local(h, max_iter=50):
        current_h = h.copy()
        current_c5 = evaluate_c5(current_h, dx)
        best_h_local = current_h.copy()
        best_c5_local = current_c5
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            improved = False
            for i in range(n_points):
                # Try small perturbations
                for delta in [-0.02, -0.01, 0.01, 0.02]:
                    h_test = reflect(current_h, i, delta)
                    # Ensure the constraint is satisfied
                    h_test = np.clip(h_test, 0, 1)
                    current_sum = np.sum(h_test)
                    if current_sum > 0:
                        h_test = h_test * (target_sum / current_sum)
                    
                    c5_test = evaluate_c5(h_test, dx)
                    if c5_test < current_c5:
                        current_h = h_test
                        current_c5 = c5_test
                        improved = True
                        if c5_test < best_c5_local:
                            best_c5_local = c5_test
                            best_h_local = h_test.copy()
            
            if not improved:
                break
        
        return best_h_local, best_c5_local
    
    # Perturbation search
    max_iterations = 500
    for iter_num in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Start with current best
        h_current = best_h.copy()
        
        # Create random perturbations
        h_perturbed = h_current + np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = np.sum(h_perturbed)
        if current_sum > 0:
            h_perturbed = h_perturbed * (target_sum / current_sum)
        
        # Local optimization on perturbation
        h_opt, c5_opt = optimize_local(h_perturbed)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            
            if best_c5 <= 0.3807:
                break
    
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
