import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, k_max):
    """Compute the gradient of the overlap at lag k_max with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    # For correlation at lag k_max, we compute the gradient
    # with respect to h[i]: it's (1 - h[i + k_max]) if i + k_max < n
    for i in range(n):
        j = i + k_max
        if j < n:
            grad[i] += -1.0
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    alpha = 0.1
    no_improvement_count = 0
    max_no_improvement = 100
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5 and the lag with maximal overlap
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        current_c5 = overlap[k_max] * dx
        
        # Track best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Termination condition
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count >= max_no_improvement:
            # If no improvement for a while, reduce learning rate
            alpha *= 0.5
            no_improvement_count = 0
            if alpha < 1e-6:
                break
                
        # Compute gradient
        grad = compute_gradient(h, dx, k_max)
        
        # Update h
        h_new = h - alpha * grad
        
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        # Occasionally add small random perturbation
        if iteration % 50 == 0 and np.random.random() < 0.3:
            perturbation = np.random.normal(0, 0.01, n_points)
            h_new = h_new + perturbation
            h_new = normalize_h(h_new, target_sum)
            
        h = h_new
        
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
