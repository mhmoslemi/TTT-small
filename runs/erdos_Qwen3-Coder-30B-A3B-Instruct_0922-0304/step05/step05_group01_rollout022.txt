import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize swarm
    n_particles = 20
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 += np.random.normal(0, 0.05, n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Initialize particles
    for i in range(n_particles):
        if i == 0:
            h = h0.copy()
        else:
            h = np.random.random(n_points) * 0.5
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_values.append(float('inf'))
    
    # Global best
    gbest = h0.copy()
    gbest_value = float('inf')
    
    # PSO parameters
    w = 0.7
    c1 = c2 = 1.5
    max_velocity = 0.2
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Evaluate fitness
            fitness = compute_c5(particles[i])
            
            # Update particle best
            if fitness < pbest_values[i]:
                pbest_values[i] = fitness
                pbest[i] = particles[i].copy()
            
            # Update global best
            if fitness < gbest_value:
                gbest_value = fitness
                gbest = particles[i].copy()
            
            # Early termination
            if gbest_value <= 0.3807:
                break
        
        # Update velocities and positions if not terminated early
        if gbest_value <= 0.3807:
            break
            
        # Update inertia weight
        w = max(0.4, w - 0.01)
        
        for i in range(n_particles):
            r1, r2 = np.random.random(2)
            # Update velocity
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            
            # Ensure feasibility
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Final result
    final_c5 = compute_c5(gbest)
    
    return (gbest.tolist(), final_c5, n_points)
