import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    # Memoization of states for dynamic programming
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Repeated local optimization
    max_iterations = 500
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h, dx)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()
    
    # Alternate search strategies to escape local optima
    strategies = [
        "gradient_descent",
        "random_perturbation", 
        "pattern_search",
        "simulated_annealing"
    ]
    
    for iteration in range(max_iterations):
        if time.time() - start_time >= budget_s - 1:
            break
            
        strategy = strategies[iteration % len(strategies)]
        
        if strategy == "gradient_descent":
            # Gradient-based local search
            def objective(h):
                return evaluate_c5(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = [{'type': 'eq', 'fun': constraint_sum}]
            
            from scipy.optimize import minimize
            result = minimize(
                objective,
                current_h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 30, 'ftol': 1e-8}
            )
            
            if result.success:
                current_h = normalize_h(result.x, target_sum)
                current_c5 = evaluate_c5(current_h, dx)
            
        elif strategy == "random_perturbation":
            # Random small perturbations
            perturbation = np.random.normal(0, 0.02, n_points)
            current_h += perturbation
            current_h = normalize_h(current_h, target_sum)
            current_c5 = evaluate_c5(current_h, dx)
            
        elif strategy == "pattern_search":
            # Pattern-based improvement
            # Introduce a pattern that reduces overlap
            pattern = np.zeros(n_points)
            pattern[:n_points//4] = 1  # Step up
            pattern[n_points//4:3*n_points//4] = 0  # Flat
            pattern[3*n_points//4:] = 1  # Step down
            
            variation = np.random.uniform(0.8, 1.2, n_points)
            trial_h = np.clip(current_h * variation, 0, 1)
            trial_h = normalize_h(trial_h, target_sum)
            trial_c5 = evaluate_c5(trial_h, dx)
            
            if trial_c5 < current_c5:
                current_h = trial_h
                current_c5 = trial_c5
                
        elif strategy == "simulated_annealing":
            # Simple simulated annealing steps
            current_h_copy = current_h.copy()
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.05)
            current_h_copy[idx] += delta
            current_h_copy = normalize_h(current_h_copy, target_sum)
            new_c5 = evaluate_c5(current_h_copy, dx)
            
            # Accept with probability based on temperature
            if new_c5 < current_c5:

                current_c5 = new_c5
            elif np.random.random() < np.exp((current_c5 - new_c5) / (0.01 * (1.0 - iteration/max_iterations))):
                current_h = current_h_copy
                current_c5 = new_c5
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
