import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize solution with initial_h_values or fallback to a simple construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Gradient-based local refinement with bounds
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Try optimizing multiple times with local search
    max_iterations = 500
    for iteration in range(max_iterations):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Use local optimization to refine the solution
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
        if result.success:
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            c5_new = compute_c5(h_new, dx)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                if best_c5 <= 0.3807:
                    break
            h = h_new
        else:
            # If optimization fails, do a small random perturbation and try again
            h = h + np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Occasionally perturb the current solution to escape local minima
        if iteration % 100 == 0:
            h = h + np.random.normal(0, 0.002, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
    
    return (best_h.tolist(), best_c5, n_points)
