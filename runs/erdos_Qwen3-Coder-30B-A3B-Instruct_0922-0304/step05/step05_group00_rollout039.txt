import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    iteration = 0
    max_iter = 5000
    
    # Use a more focused optimization strategy
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute overlap for current h
        overlap = correlate(h, 1 - h, mode="full")
        # Get top contributing indices
        top_k_indices = np.argsort(np.abs(overlap))[-20:]  # Top 20 overlaps
        # Focus on a subset of these indices to reduce complexity
        focus_indices = top_k_indices[np.argsort(overlap[top_k_indices])[-10:]]
        
        # Constraint satisfaction with scaled bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use gradient-based local optimization
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        def objective(h):
            return compute_c5(h, dx)
        
        # Perform local optimization
        result = minimize(objective, h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 50, 'ftol': 1e-8})
        
        if result.success:
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            h = h_new.copy()
        
        # Evaluate and update best
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
