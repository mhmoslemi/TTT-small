from scipy.signal import correlate
from scipy.optimize import minimize
import time

    start_time = time.time()

    dx = 2.0 / n_points

        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Binary search for optimal C5
    c5_low = 0.0
    c5_high = 0.38085
    tolerance = 1e-4
    max_iterations = 40
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5_mid = (c5_low + c5_high) / 2.0
        
        # Define the constraint for the overlap
        def overlap_constraint(h):
            overlap = correlate(h, 1 - h, mode="full")
            return c5_mid * dx - np.max(overlap) * dx
        
        # Constraints for feasibility check
        feasibility_constraints = [
            {'type': 'eq', 'fun': constraint_sum},
            {'type': 'ineq', 'fun': overlap_constraint}
        ]

        # Start from current solution for better warm start
        h_try = h.copy()
        
        try:
            result = minimize(
                lambda x: -np.max(correlate(x, 1 - x, mode="full") * dx),

                bounds=bounds,
                constraints=feasibility_constraints,

            )

                current_c5 = compute_c5(h)

                    # This is a valid solution
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h.copy()
                    
                    # Try to reduce high bound
                    c5_high = c5_mid
                else:
                    c5_low = c5_mid
            else:
                c5_low = c5_mid

            c5_low = c5_mid
            
        # Early exit if target achieved

    if time.time() - start_time < budget_s - 10:

            return compute_c5(h)

        final_constraints = [{'type': 'eq', 'fun': constraint_sum}]
