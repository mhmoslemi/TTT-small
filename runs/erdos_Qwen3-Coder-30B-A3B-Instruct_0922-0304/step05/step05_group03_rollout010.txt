import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx):
    """Compute gradient of max overlap with respect to h."""
    # This is an approximation of gradients using finite differences
    # For each element, we compute how much changing it affects the maximum overlap
    n = len(h)
    grad = np.zeros(n)
    max_overlap = evaluate_c5(h, dx)
    
    eps = 1e-6
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = normalize_h(h_plus, np.sum(h))
        overlap_plus = evaluate_c5(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = normalize_h(h_minus, np.sum(h))
        overlap_minus = evaluate_c5(h_minus, dx)
        
        grad[i] = (overlap_plus - overlap_minus) / (2 * eps)
    
    return grad

    # Optimization parameters
    lr = 0.1
    patience = 10
    patience_counter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1

        current_c5 = evaluate_c5(h, dx)

        # Adaptive learning rate
        if patience_counter > patience:
            lr *= 0.5
            patience_counter = 0
            
        # Gradient step
        grad = compute_gradient(h, dx)
        h = h - lr * grad
        
        # Project back to feasible region
        h = normalize_h(h, target_sum)
        
        # Coordinate-wise improvement
        if iteration % 5 == 0:
