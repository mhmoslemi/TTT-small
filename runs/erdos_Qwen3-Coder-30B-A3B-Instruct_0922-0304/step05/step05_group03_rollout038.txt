import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradients(h, dx):
    """Compute gradients of C5 with respect to h."""
    # Get the full correlation
    corr = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    # For each element, compute influence on max correlation
    # We'll compute rough influence by finite differences
    grad = np.zeros_like(h)
    eps = 1e-5
    max_corr = corr[max_idx]
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = normalize_h(h_plus, np.sum(h))
        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
        grad[i] = (corr_plus[max_idx] - max_corr) / eps
    return grad

    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70

    # Initialize from initial_h_values if available

    # Initial evaluation
    current_c5 = evaluate_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Optimization parameters
    learning_rate = 0.1
    max_iter = 1000
    patience = 50
    patience_counter = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Compute gradient ascent direction to minimize C5
        grad = compute_gradients(h, dx)
        
        # Update h
        h_new = h - learning_rate * grad
        h_new = normalize_h(h_new, target_sum)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept improvement or revert
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                patience_counter = 0
            else:
                patience_counter += 1
        else:
            patience_counter += 1
            
        # Reduce learning rate if no improvement
        if patience_counter > patience:
            learning_rate *= 0.5
            patience_counter = 0
            
        # Check for early exit
        if best_c5 <= 0.3807:
            break
    
    # Coordinate-wise local search
    for _ in range(500):
        if time.time() - start_time > budget_s - 1:
            break
        
        improved = False
        for i in range(n_points):
            # Try small perturbations
            perturbations = [0.01, -0.01]
            for delta in perturbations:
                h_test = h.copy()
                h_test[i] += delta
                h_test = normalize_h(h_test, target_sum)
                test_c5 = evaluate_c5(h_test, dx)
                if test_c5 < current_c5:
                    h = h_test
                    current_c5 = test_c5
                    improved = True
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h.copy()
        
        if not improved:
            break
    
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
