import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have sum equal to target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a simple initial guess
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)
    
    # For gradient-based optimization
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Adaptive learning rate
    lr = 0.01
    momentum = 0.9
    velocity = np.zeros(n_points)
    
    iters = 0
    while time.time() - start_time < budget_s - 1:
        # Compute cross-correlations for all shifts
        # We compute overlap for all possible shifts
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Find the shift that produces the maximum overlap
        shift = max_idx - (n_points - 1)
        
        # Compute gradient
        # For shift k, gradient is (1 - h_{i+k}) 
        # We compute it directly using correlation
        h_shifted = np.roll(1 - h, shift)
        grad = h_shifted
        grad = grad * dx  # Scale by dx
        
        # Update with momentum
        velocity = momentum * velocity - lr * grad
        h = h + velocity
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Evaluate current solution
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Adaptive learning rate - reduce if no improvement
        lr *= 0.999
        iters += 1
        
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
