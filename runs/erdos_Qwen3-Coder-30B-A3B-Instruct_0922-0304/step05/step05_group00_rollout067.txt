import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def repair_feasibility(h, target_sum):
    """Repair h to be feasible: [0,1] and sum=target_sum."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
    h = repair_feasibility(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Population-based search
    population_size = 30
    population = [h.copy()]
    for _ in range(population_size - 1):
        h_mut = h + np.random.normal(0, 0.02, n_points)
        h_mut = repair_feasibility(h_mut, target_sum)
        population.append(h_mut)
    
    step_size = 0.05
    step_decay = 0.98
    
    iteration = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate population
        fitness = [compute_c5(p, dx) for p in population]
        
        # Track best
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_h = population[min_fit_idx].copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection - keep top 10%
        sorted_indices = np.argsort(fitness)[:population_size//10]
        selected = [population[i] for i in sorted_indices]
        
        # Generate new population
        new_population = []
        
        # 1. Differential evolution (40%)
        for _ in range(population_size * 2 // 5):
            if len(selected) >= 3:
                a, b, c = np.random.choice(len(selected), 3, replace=False)
                mutant = selected[a] + 0.8 * (selected[b] - selected[c])
                mutant = repair_feasibility(mutant, target_sum)
                new_population.append(mutant)
            else:
                h_mut = h + np.random.normal(0, step_size, n_points)
                h_mut = repair_feasibility(h_mut, target_sum)
                new_population.append(h_mut)
        
        # 2. Gradient descent with random steps (30%)
        for _ in range(population_size * 3 // 10):
            idx = np.random.randint(0, len(selected))
            h_ref = selected[idx].copy()
            # Slight gradient estimate
            grad = np.zeros(n_points)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_ref.copy()
                h_plus[i] += eps
                h_plus = repair_feasibility(h_plus, target_sum)
                c_plus = compute_c5(h_plus, dx)
                
                h_minus = h_ref.copy()
                h_minus[i] -= eps
                h_minus = repair_feasibility(h_minus, target_sum)
                c_minus = compute_c5(h_minus, dx)
                
                grad[i] = (c_plus - c_minus) / (2 * eps)
            
            # Update with negative gradient
            new_h = h_ref - step_size * grad
            new_h = repair_feasibility(new_h, target_sum)
            new_population.append(new_h)
        
        # 3. Random sampling (20%)
        for _ in range(population_size // 5):
            h_rand = np.random.rand(n_points)
            h_rand = repair_feasibility(h_rand, target_sum)
            new_population.append(h_rand)
        
        # 4. Mutation of best (10%)
        for _ in range(population_size // 10):
            h_best = best_h.copy()
            h_best += np.random.normal(0, step_size*0.5, n_points)
            h_best = repair_feasibility(h_best, target_sum)
            new_population.append(h_best)
            
        # Keep all candidates
        population = new_population.copy()
        
        # Repair all candidates
        for i in range(len(population)):
            population[i] = repair_feasibility(population[i], target_sum)
            
        # Replace worst if needed
        if len(population) > population_size:
            pop_fitness = [compute_c5(p, dx) for p in population]
            worst_indices = np.argsort(pop_fitness)[-10:]
            for i in sorted(worst_indices, reverse=True):
                population.pop(i)
            assert len(population) == population_size
            
        # Adjustment
        step_size *= step_decay
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
