import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def halton_sequence(n, base=2):
    """Generate n points using Halton sequence with given base."""
    result = np.zeros(n)
    for i in range(n):
        f = 1.0
        result[i] = 0.0
        while i > 0:
            f /= base
            result[i] = (result[i] + (i % base) * f)
            i //= base
    return result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
    
    # Use Halton sequence for multi-start
    num_starts = 50
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Generate Halton sequence for diverse starting points
    halton_points = halton_sequence(num_starts, 2)
    halton_points = halton_points.reshape(-1, 1)
    
    for i in range(num_starts):
        # Create randomized initial guess
        h_init = h.copy()
        # Vary the shape using Halton points
        noise = np.random.normal(0, 0.1, n_points)
        h_init += noise * (1 - halton_points[i])
        h_init = normalize_h(h_init, target_sum)
        
        def objective(h):
            return compute_c5(h)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        # Run SLSQP optimization
        result = minimize(
            objective,
            h_init,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        if result.success:
            h_opt = result.x
            h_opt = normalize_h(h_opt, target_sum)
            c5_opt = compute_c5(h_opt)
            
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final refinement with local search
    def objective(h):
        return compute_c5(h)
        
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    result = minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 50, 'ftol': 1e-9}
    )
    
    if result.success:
        best_h = result.x
        best_h = normalize_h(best_h, target_sum)
