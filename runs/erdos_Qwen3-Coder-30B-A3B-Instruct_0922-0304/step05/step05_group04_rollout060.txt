import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - step function with slight randomization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        # Add small noise to escape local optima
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Local search with simulated annealing-like approach
    temperature = 0.1
    cooling_rate = 0.9995
    iteration = 0
    
    # Track good solutions to avoid getting stuck
    recent_improvements = []
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            recent_improvements.append((current_c5, h.copy()))
            if len(recent_improvements) > 5:
                recent_improvements.pop(0)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Create neighbor solution
        new_h = h.copy()
        
        # Perturb randomly with varying magnitude
        if np.random.random() < 0.7:
            # Small change
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.01)
            new_h[idx] += delta
        else:
            # Larger change in a few locations
            num_changes = np.random.randint(1, 5)
            for _ in range(num_changes):
                idx = np.random.randint(0, n_points)
                delta = np.random.normal(0, 0.05)
                new_h[idx] += delta
        
        # Clip and renormalize
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Accept or reject based on Metropolis criterion
        new_c5 = compute_c5(new_h)
        if new_c5 < current_c5 or np.random.random() < np.exp((current_c5 - new_c5) / temperature):
            h = new_h
            # Adjust temperature based on performance
            if current_c5 > new_c5:
                temperature *= 0.999
            else:
                temperature *= cooling_rate
        else:
            # If rejected and no improvement, slightly perturb current solution
            if iteration % 100 == 0:
                idx = np.random.randint(0, n_points)
                delta = np.random.normal(0, 0.005)
                h[idx] += delta
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        
        # Occasionally go back to a recent good solution to escape local minima
        if len(recent_improvements) > 2 and np.random.random() < 0.01:
            _, sample_h = recent_improvements[np.random.randint(0, len(recent_improvements))]
            h = sample_h.copy()
        
        # Cool down the temperature
        temperature = max(0.001, temperature * cooling_rate)
    
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
