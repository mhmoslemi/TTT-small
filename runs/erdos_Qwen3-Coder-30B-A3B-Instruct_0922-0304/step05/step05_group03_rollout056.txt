import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, k_max, dx):
    """Compute gradient of overlap at k_max with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    # For overlap at lag k_max, we have:
    # overlap[k_max] = sum h[i] * (1 - h[i + k_max]) for valid i
    # Gradient for h[i] w.r.t overlap[k_max] is:
    # d/dh[i] overlap[k_max] = (1 - h[i + k_max]) if i+k_max < n else 0
    # But also d/dh[i+k_max] overlap[k_max] = -h[i] if i < n-k_max
    for i in range(n):
        if i + k_max < n:
            grad[i] += (1 - h[i + k_max])
        if i - k_max >= 0:
            grad[i - k_max] -= h[i]
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    # Initialize
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent settings
    learning_rate = 0.1
    min_lr = 1e-6
    patience = 10
    patience_counter = 0
    c5_prev = float('inf')
    
    # Main optimization loop
    max_iter = int(budget_s * 10) # Rough estimate, adjust based on time
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap to find maximum
        overlap_full = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap_full)
        max_overlap = overlap_full[max_k] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if we're close enough
        if best_c5 <= 0.3807:
            break
        
        # Compute gradient at max overlap
        grad = compute_gradient(h, max_k, dx)
        
        # Update h
        h_new = h - learning_rate * grad
        
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        # Check if improvement
        new_c5 = evaluate_c5(h_new, dx)
        
        if new_c5 < max_overlap:
            h = h_new
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter >= patience:
                learning_rate *= 0.5
                patience_counter = 0
                if learning_rate < min_lr:
                    break
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
