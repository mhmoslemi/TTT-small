```python
import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: clip to [0,1] and normalize sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        # Create a reasonable starting point
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient descent with projected updates
    max_iter = 50000
    lr = 0.01
    decay = 0.999
    patience = 1000
    patience_counter = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_k = max_idx - (n_points - 1)  # Convert index to shift
        
        # Compute gradient for active shift
        # Using numerical gradient approximation
        h_shifted = np.roll(1 - h, max_k)
        grad = h_shifted
        
        # Update with gradient descent
        h = h - lr * grad
        
        # Project to feasible region
        h = project_h(h, target_sum)
        
        # Compute new C5
        current_c5 = compute_c5(h, dx)
        
        # Update best if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Decay learning rate
        lr *= decay
        
        # Early stopping if no improvement
        if patience_counter > patience:
            break
            
    return (best_h.tolist(), best_c5, n_points)
```
