import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create random binary vector (approximate solution)
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        np.random.shuffle(h)
        h = h * (n_points / 2) / np.sum(h)
    
    # Binary search with simulated annealing
    T = 1.0
    alpha = 0.995
    max_iter = 20000
    iteration = 0
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def compute_c5_fast(h):
        # Use FFT for fast correlation
        h_pad = np.pad(h, (0, n_points-1), 'constant')
        one_minus_h_pad = np.pad(1 - h, (0, n_points-1), 'constant')
        corr = np.real(np.fft.ifft(np.fft.fft(h_pad) * np.fft.fft(one_minus_h_pad).conj()))
        return np.max(corr) * dx
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Generate neighbor by swapping two random indices
        i, j = np.random.choice(n_points, 2, replace=False)
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # Renormalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h_new = np.clip(h_new, 0, 1)
        
        c5_old = compute_c5(h)
        c5_new = compute_c5(h_new)
        
        if c5_new < c5_old or np.random.rand() < np.exp(-(c5_new - c5_old) / T):
            h = h_new
            c5 = c5_new
            
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Annealing
        if T > 1e-4:
            T *= alpha
    
    # Continuous relaxation with CVXPY
    if time.time() - start_time < budget_s - 1:
        try:
            h_var = cp.Variable(n_points)
            prob = cp.Problem(cp.Minimize(cp.maximum(cp.matmul(
                cp.Variable((2*n_points-1, n_points), 
                           value=np.tril(np.ones((2*n_points-1, n_points)), k=0)),
                h_var) * dx)), 
                             [cp.sum(h_var) == n_points / 2, 
                              h_var >= 0, h_var <= 1])
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_relaxed = h_var.value
                c5_relaxed = compute_c5(h_relaxed)
                if c5_relaxed < best_c5:
                    best_c5 = c5_relaxed
                    best_h = h_relaxed.copy()
        except:
            pass
    
    return (best_h.tolist(), best_c5, n_points)
