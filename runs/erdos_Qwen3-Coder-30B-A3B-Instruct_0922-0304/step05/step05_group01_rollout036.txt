import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Binary search for optimal C5
    c5_low = 0.0
    c5_high = 0.38085  # Current upper bound
    tolerance = 1e-4
    max_iter = 40
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5_mid = (c5_low + c5_high) / 2.0
        
        # Optimization to check feasibility of c5_mid
        def objective(h):
            return compute_c5(h)
            
        def constraint_max_overlap(h):
            overlap = correlate(h, 1 - h, mode="full")
            return c5_mid * dx - np.max(overlap) * dx

        # Define bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Add constraints
        constraints = [
            {'type': 'eq', 'fun': constraint_sum},
            {'type': 'ineq', 'fun': constraint_max_overlap}
        ]
        
        # Try multiple initializations
        best_feasible_h = None
        best_feasible_c5 = float('inf')
        for _ in range(3):
            # Perturb the current h a bit
            h_init = h + np.random.normal(0, 0.01, n_points)
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            
            try:
                result = minimize(
                    objective, 
                    h_init, 
                    method='SLSQP', 
                    bounds=bounds, 
                    constraints=constraints,
                    options={'maxiter': 50, 'ftol': 1e-8}
                )
                
                if result.success:
                    current_c5 = compute_c5(result.x)
                    if current_c5 < best_feasible_c5:
                        best_feasible_c5 = current_c5
                        best_feasible_h = result.x.copy()
            except:
                continue
        
        if best_feasible_h is not None:
            h = best_feasible_h
            if best_feasible_c5 < best_c5:
                best_c5 = best_feasible_c5
                best_h = best_feasible_h.copy()
            c5_high = c5_mid
        else:
            c5_low = c5_mid
            
        if abs(c5_high - c5_low) < tolerance:
            break
            
        # If no improvement for a while, perturb
        if iteration % 10 == 0:
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation with refinement
    if time.time() - start_time < budget_s - 1:
        def objective(h):
            return compute_c5(h)
            
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            result = minimize(
                objective, 
                best_h, 
                method='SLSQP', 
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-10}
            )
            if result.success:
                final_c5 = compute_c5(result.x)
                best_c5 = min(best_c5, final_c5)
                best_h = result.x
        except:
            pass
    
    return (best_h.tolist(), best_c5, n_points)
