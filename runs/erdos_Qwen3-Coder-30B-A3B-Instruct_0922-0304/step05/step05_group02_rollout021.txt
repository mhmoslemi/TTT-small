```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        h = np.ones(n_points) * 0.5
        h = project_to_feasible(h, target_sum)

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Early exit if already good enough
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    iteration = 0
    perturbation_magnitude = 0.1
    
    while time.time() - start_time < budget_s - 1:
        if iteration % 50 == 0 and iteration > 0:
            perturbation_magnitude *= 0.9
        
        # Compute all cross-correlations to get upper bounds
        # For a sliding window approach that can be combined
        corr_full = correlate(h, 1 - h, mode="full")
        
        # Compute C_k for all k
        c_values = corr_full * dx
        
        # Set up convex optimization problem
        h_var = cp.Variable(n_points)
        max_c = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_var) == target_sum,
            h_var >= 0,
            h_var <= 1,
        ]
        
        # Add constraint for each shift
        for i in range(len(c_values)):
            # c_values[i] corresponds to a correlation at shift i
            # Need to compute the correlation with correponding shifted values correctly
            # indices shifted relative to the full convolution mode
            shift = i - (len(h) - 1)  # Midpoint shift for each correlation element
            # Vectorized version
            corr_value = c_values[i]
            constraints.append(max_c >= corr_value)
        
        # Objective
        objective = cp.Minimize(max_c)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                h_new = project_to_feasible(h_new, target_sum)
                h = h_new
        except:
            pass
        
        # Local perturbation every 50 iterations
        if iteration % 50 == 0:
            noise = np.random.normal(0, perturbation_magnitude, n_points)
            h = h + noise
            h = project_to_feasible(h, target_sum)
        
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            return (best_h.tolist(), best_c5, n_points)
            
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
```
