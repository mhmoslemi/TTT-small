```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] with sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Start with a step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = project_to_feasible(h, target_sum)
        
    # Initial evaluation
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Optimization loop
    iteration = 0
    perturbation_factor = 0.1
    max_iter = 500
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # For large problems, we approximate using a convex relaxation:
        # minimize C
        # subject to C >= (h * (1-h))[k] * dx for all k
        #          sum(h) = target_sum
        #          0 <= h[i] <= 1
        
        # Approximate by numerical computation of all correlations
        # We create a convex optimization problem
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints for all correlation terms (simplified for computational efficiency)
        # This is a coarse approximation but we can efficiently update the solution
        # Let's use a gradient-based approach using the known gradient at each point
        # since the full correlation is expensive.
        
        # Instead: let's solve a relaxed problem that approximates a good local minimum
        # by approximating based on the current h
        
        # Compute current overlap function values
        h_flat = h.copy()
        overlap_terms = []
        for k in range(2 * n_points - 1):
            # Compute (h * (1-h))[k]
            shifted_h = np.concatenate([h_flat[k:], np.zeros(k)])
            current_contribution = np.sum(h_flat * (1 - shifted_h[:n_points]))
            overlap_terms.append(current_contribution * dx)
        
        max_over = np.max(overlap_terms)
        
        # Formulate the convex optimization problem
        obj = cp.Minimize(C)
        constraints = [C >= t for t in overlap_terms]
        constraints += [cp.sum(h_var) == target_sum]
        constraints += [h_var >= 0]
        constraints += [h_var <= 1]
        
        # Solve quadratic program with convex solver
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            h_new = h_var.value
            h_new = np.array(h_new)
            h_new = project_to_feasible(h_new, target_sum)
            if h_new is not None:
                h = h_new.copy()
                c5 = compute_c5(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                
                # Early termination check
                if best_c5 <= 0.3807:
                    break
        except Exception:
            pass
            
        # Perturb solution occasionally to escape local minima
        if iteration % 50 == 0:
            noise = np.random.normal(0, perturbation_factor, n_points)
            h = h + noise
            h = project_to_feasible(h, target_sum)
            perturbation_factor *= 0.9
            if perturbation_factor < 1e-6:
                perturbation_factor = 1e-6
        
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
```
