import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_feasible(h, target_sum):
    """Project h to feasible region [0,1] and normalize."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Identify critical lags that contribute most to overlap
        overlap = correlate(h, 1 - h, mode="full")

        # Constraints
        def constraint_sum(h_new):
            return np.sum(h_new) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}

        # Early stopping
