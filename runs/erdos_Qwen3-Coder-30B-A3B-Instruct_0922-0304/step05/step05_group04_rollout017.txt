import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    mutation_rate = 0.1
    tournament_size = 10
    max_generations = 1000
    
    # Initialize population
    def initialize_population():
        population = []
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
        
        # Fill up the rest with random individuals
        while len(population) < population_size:
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
        
        return population
    
    # Compute C5 for an individual
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Tournament selection
    def tournament_selection(population, fitnesses):
        selected = []
        for _ in range(population_size):
            # Randomly sample tournament participants
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
            selected.append(population[winner_idx].copy())
        return selected
    
    # Uniform crossover
    def crossover(parent1, parent2):
        # Uniform crossover
        mask = np.random.rand(n_points) > 0.5
        child = np.where(mask, parent1, parent2)
        
        # Repair: keep sum = n_points / 2
        child = child * (n_points / 2) / np.sum(child)
        return np.clip(child, 0, 1)
    
    # Mutation with repair
    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                if np.random.rand() < 0.5:
                    # Add/subtract small random value
                    h[i] += np.random.normal(0, 0.1)
                else:
                    # Set to 0 or 1
                    h[i] = np.random.choice([0, 1])
        
        # Repair: keep sum = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    # Main GA loop
    population = initialize_population()
    best_h = None
    best_c5 = float('inf')
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = [compute_c5(ind) for ind in population]
        
        # Track best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Create new population
        new_population = []
        # Keep best individual
        new_population.append(population[min_fitness_idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = compute_c5(best_h)
    else:
        final_c5 = float('inf')
    
    return (best_h.tolist(), final_c5, n_points)
