import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
        
    # Initialize population
    population_size = 50
    population = []
    population.append(h.copy())
    for _ in range(population_size - 1):
        h_perturbed = h + np.random.normal(0, 0.05, n_points)
        h_perturbed = normalize_h(h_perturbed, target_sum)
        population.append(h_perturbed)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # GA parameters
    tournament_size = 10
    mutation_rate = 0.1
    elite_size = 5
    local_search_top = 10
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Evaluate population
        fitness = []
        for ind in population:
            c5 = compute_c5(ind)
            fitness.append(c5)
            
        # Track best individual
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Selection (tournament)
        selected = []
        for _ in range(population_size):
            candidates = np.random.choice(population_size, tournament_size)
            best_idx = candidates[np.argmin([fitness[i] for i in candidates])]
            selected.append(population[best_idx].copy())
        
        # Create new population
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitness)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Selection for crossover
            parent1_idx, parent2_idx = np.random.choice(population_size, 2, replace=False)
            parent1 = selected[parent1_idx]
            parent2 = selected[parent2_idx]
            
            # Single point crossover
            crossover_point = np.random.randint(1, n_points)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Mutation
            if np.random.random() < mutation_rate:
                # Add noise with direction-dependent scaling
                noise = np.random.normal(0, 0.02, n_points)
                child1 += noise
                child1 = normalize_h(child1, target_sum)
            if np.random.random() < mutation_rate:
                noise = np.random.normal(0, 0.02, n_points)
                child2 += noise
                child2 = normalize_h(child2, target_sum)
                
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        # Local search on top candidates
        top_indices = np.argsort(fitness)[:local_search_top]
        for idx in top_indices:
            # Apply gradient-based local search with adaptive steps
            h = population[idx]
            
            def objective(h):
                return compute_c5(h)
                
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = [{'type': 'eq', 'fun': constraint_sum}]
            
            # Try different optimizers for better results
            try:
                result = minimize(
                    objective,
                    h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 50, 'ftol': 1e-9}
                )
                if result.success:
                    h_new = result.x
                    h_new = normalize_h(h_new, target_sum)
                    new_population[idx] = h_new
            except:
                # Fallback to simple gradient descent
                step_size = 0.01
                for _ in range(20):
                    gradient = np.zeros_like(h)
                    for i in range(n_points):
                        h_perturb = h.copy()
                        h_perturb[i] += 1e-5
                        h_perturb = normalize_h(h_perturb, target_sum)
                        val_plus = compute_c5(h_perturb)
                        
                        h_perturb = h.copy()
                        h_perturb[i] -= 1e-5
                        h_perturb = normalize_h(h_perturb, target_sum)
                        val_minus = compute_c5(h_perturb)
                        
                        gradient[i] = (val_plus - val_minus) / (2e-5)
                    
                    h -= step_size * gradient
                    h = normalize_h(h, target_sum)
                
                new_population[idx] = h
            
        population = new_population
        
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
