import numpy as np
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 20
    max_iter = 1000
    
    # Initialize swarm
    def initialize_particle():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
        else:
            h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    swarm = [initialize_particle() for _ in range(swarm_size)]
    particle_best = [p.copy() for p in swarm]
    particle_best_fitness = [float('inf')] * swarm_size
    
    # Global best
    global_best = initialize_particle()
    global_best_fitness = float('inf')
    
    # PSO parameters
    w = 0.7
    c1 = c2 = 1.5
    max_velocity = 0.1
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update swarm
        for i in range(swarm_size):
            # Evaluate current particle
            fitness = compute_c5(swarm[i])
            
            # Update particle best
            if fitness < particle_best_fitness[i]:
                particle_best_fitness[i] = fitness
                particle_best[i] = swarm[i].copy()
            
            # Update global best
            if fitness < global_best_fitness:
                global_best_fitness = fitness
                global_best = swarm[i].copy()
            
            # Check for early termination
            if global_best_fitness <= 0.3807:
                break
                
        # Early termination check
        if global_best_fitness <= 0.3807:
            break
            
        # Update particles
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.random(2)
            if i == 0:  # Use global best for all particles
                p_best = global_best
            else:
                p_best = particle_best[i]
                
            velocity = (w * swarm[i] + 
                       c1 * r1 * (p_best - swarm[i]) + 
                       c2 * r2 * (global_best - swarm[i]))
            
            # Clamp velocity
            velocity = np.clip(velocity, -max_velocity, max_velocity)
            
            # Update position
            swarm[i] += velocity
            swarm[i] = np.clip(swarm[i], 0, 1)
            
            # Normalize
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
        # Update inertia weight
        w = max(0.4, w - 0.0003)
    
    # Final evaluation
    final_c5 = compute_c5(global_best)
    
    return (global_best.tolist(), final_c5, n_points)
