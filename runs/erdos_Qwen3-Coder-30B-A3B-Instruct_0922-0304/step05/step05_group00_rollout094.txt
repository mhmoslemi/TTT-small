import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def repair_feasible(h, target_sum):
    """Repair h to be feasible: within [0,1] and sum to target_sum."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Parameters
    pop_size = 30
    max_iter = 1000
    step_size = 0.01
    decay = 0.98
    
    # Track population of diverse solutions
    population = [h.copy()]
    for _ in range(pop_size - 1):
        h_mut = h + np.random.normal(0, 0.01, n_points)
        h_mut = repair_feasible(h_mut, target_sum)
        population.append(h_mut)
    
    # Iteration loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for ind in population:
            c5 = compute_c5(ind, dx)
            fitness.append(c5)
            
        # Update best
        min_idx = np.argmin(fitness)
        if fitness[min_idx] < best_c5:
            best_c5 = fitness[min_idx]
            best_h = population[min_idx].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Select parents (tournament)
        parents = []
        tournament_size = 5
        for _ in range(pop_size):
            idxs = np.random.choice(len(population), tournament_size)
            winner_idx = idxs[np.argmin([fitness[i] for i in idxs])]
            parents.append(population[winner_idx].copy())
        
        # Generate new population
        new_pop = []
        
        # 1. Differential evolution
        for _ in range(12):
            [a, b, c] = np.random.choice(len(parents), 3, replace=False)
            mutant = parents[a] + 0.8 * (parents[b] - parents[c])
            mutant = repair_feasible(mutant, target_sum)
            new_pop.append(mutant)
            
        # 2. Gradient descent from best
        for _ in range(5):
            current = best_h.copy()
            for _ in range(5):
                # Simple gradient descent step
                overlaps = correlate(current, 1 - current, mode="full")
                grad = np.zeros_like(current)
                for i in range(len(current)):
                    # Approximate gradient using finite differences
                    eps = 1e-5
                    current_plus = current.copy()
                    current_plus[i] += eps
                    current_plus = repair_feasible(current_plus, target_sum)
                    overlap_plus = correlate(current_plus, 1 - current_plus, mode="full")
                    grad[i] = (np.max(overlap_plus) - np.max(overlaps)) / eps
                # Apply gradient step
                current = current - 0.001 * grad
                current = repair_feasible(current, target_sum)
            new_pop.append(current)
            
        # 3. Random samples
        for _ in range(5):
            r = np.random.random(n_points)
            r = repair_feasible(r, target_sum)
            new_pop.append(r)
            
        # 4. Mutate best
        for _ in range(3):
            h_mut = best_h + np.random.normal(0, 0.01, n_points) 
            h_mut = repair_feasible(h_mut, target_sum)
            new_pop.append(h_mut)
        
        # Add elites
        new_pop.append(best_h.copy())
        new_pop.append(parents[min_idx].copy())
        
        # Keep only pop_size individuals
        # Order by fitness
        fitness = [compute_c5(ind, dx) for ind in new_pop]
        sorted_idx = np.argsort(fitness)
        population = [new_pop[i] for i in sorted_idx[:pop_size]]
        
        step_size *= decay
        
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
