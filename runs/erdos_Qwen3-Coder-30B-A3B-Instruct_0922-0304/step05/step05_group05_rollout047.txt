import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def compute_gradient(h, max_lags):
        gradient = np.zeros(n_points)
        for k in max_lags:
            # Forward difference for positive lags
            for i in range(n_points):
                if i + k < n_points:
                    gradient[i] += (1 - h[i + k])
                # Backward difference for negative lags  
                if i - k >= 0:
                    gradient[i] -= h[i - k]
        return gradient / len(max_lags)
    
    # Constraint function
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Initialize tracking variables
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 1000
    patience = 50
    patience_counter = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        overlaps = correlate(h, 1 - h, mode="full")
        max_val = overlaps.max()
        max_lags = [i - (n_points - 1) for i, val in enumerate(overlaps) if val == max_val]
        
        # Compute gradient
        grad = compute_gradient(h, max_lags)
        
        # Update step
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        # Update h
        h = h_new
            
        # Adaptive learning rate
        if i % 100 == 0 and i > 0:
            learning_rate *= 0.9
            
        if patience_counter >= patience:
            # Reduce learning rate if no improvement for patience iterations
            learning_rate *= 0.9
            patience_counter = 0
    
    # Post-processing: local optimization
    try:
        result = minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_c5 = compute_c5(final_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
