import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    generations = 1000
    mutation_rate = 0.1
    
    # Initialize population
    def create_individual():
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values)]
        population.extend([create_individual() for _ in range(population_size - 1)])
    else:
        population = [create_individual() for _ in range(population_size)]
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_h = None
    best_c5 = float('inf')
    
    # Tournament selection
    def tournament_selection(pop, k=10):
        selected = []
        for _ in range(len(pop) // 2):
            tournament = np.random.choice(pop, k)
            winner = min(tournament, key=compute_c5)
            selected.append(winner)
        return selected
    
    # Crossover
    def crossover(parent1, parent2):
        mask = np.random.random(n_points) > 0.5
        child = np.where(mask, parent1, parent2)
        # Renormalize
        child = child * (n_points / 2) / np.sum(child)
        return child
    
    # Mutation
    def mutate(individual):
        mutated = individual.copy()
        for i in range(n_points):
            if np.random.random() < mutation_rate:
                if np.random.random() < 0.5:
                    mutated[i] += np.random.normal(0, 0.05)
                else:
                    mutated[i] = np.random.random()
        # Clamp and renormalize
        mutated = np.clip(mutated, 0, 1)
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return mutated
    
    # Main GA loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = [compute_c5(ind) for ind in population]
        
        # Track best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Selection
        selected = tournament_selection(population)
        
        # Create new population
        new_population = []
        # Keep top 50%
        top_individuals = [population[i] for i in np.argsort(fitness)[:population_size//2]]
        new_population.extend(top_individuals)
        
        # Crossover and mutation
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = new_population
        
        # Occasionally reinitialize population to maintain diversity
        if gen % 50 == 0 and gen > 0:
            for i in range(len(population)):
                if np.random.random() < 0.2:
                    population[i] = create_individual()
    
    final_c5 = compute_c5(best_h) if best_h is not None else float('inf')
    return (best_h.tolist(), final_c5, n_points)
