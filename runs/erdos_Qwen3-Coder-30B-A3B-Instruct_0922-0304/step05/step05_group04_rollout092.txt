import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    mutation_rate = 0.1
    tournament_size = 10
    max_generations = 1000
    
    # Initialize population
    def initialize_population():
        population = []
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
        else:
            h = np.ones(n_points)
            h[n_points//2:] = 0
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
        
        # Fill the rest with random individuals
        for _ in range(population_size - 1):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
        return population
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def selection(population, fitnesses):
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
            selected.append(population[winner_idx].copy())
        return selected

    def crossover(parent1, parent2):
        mask = np.random.rand(n_points) < 0.5
        child = np.where(mask, parent1, parent2)
        # Normalize to maintain constraint
        child = child * (n_points / 2) / np.sum(child)
        return child
    
    def mutate(individual):
        mutated = individual.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                if np.random.rand() < 0.5:
                    mutated[i] += np.random.normal(0, 0.1)
                else:
                    mutated[i] = np.random.rand()
        # Clamp values
        mutated = np.clip(mutated, 0, 1)
        # Renormalize
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return mutated

    population = initialize_population()
    best_c5 = float('inf')
    best_h = None
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = [compute_c5(h) for h in population]
        
        # Track best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected = selection(population, fitnesses)
        
        # Create new population
        new_population = []
        # Keep the best individual
        new_population.append(population[min_fitness_idx].copy())
        
        # Generate offspring
        for i in range(0, population_size - 1, 2):
            parent1 = selected[i]
            parent2 = selected[i + 1]
            child1 = crossover(parent1, parent2)
            child2 = crossover(parent2, parent1)
            new_population.append(mutate(child1))
            new_population.append(mutate(child2))
        
        # Trim to exact size
        population = new_population[:population_size]
    
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
