import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradients(h, dx):
    """Compute gradients for each h[i] with respect to the maximum overlap."""
    n = len(h)
    # Compute correlations
    corr = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    max_val = corr[max_idx]
    
    # Compute gradient
    grad = np.zeros(n)
    # For each position, compute the effect on the maximum overlap
    for i in range(n):
        # Simulate small change
        h_pert = h.copy()
        h_pert[i] += 1e-6
        h_pert = normalize_h(h_pert, np.sum(h))
        corr_pert = correlate(h_pert, 1 - h_pert, mode="full")
        grad[i] = (np.max(corr_pert) - max_val) / 1e-6
    
    return grad
