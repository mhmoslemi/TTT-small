import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop
    max_iter = 1000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Convex optimization step
        variables = cp.Variable(n_points)
        constraints = [
            variables >= 0,
            variables <= 1,
            cp.sum(variables) == n_points / 2
        ]
        
        # Compute correlation objective
        # We use a direct approach: proximal gradient descent on a convex relaxation
        try:
            # Simple method: projected gradient descent
            h_new = h.copy()
            step_size = 0.01
            for _ in range(10):
                # Compute gradient approximation
                corr = correlate(h_new, 1 - h_new, mode="full")
                max_idx = np.argmax(corr)
                grad = np.zeros_like(h_new)
                # Simple gradient ascent approximation
                # Rather than compute exact gradient, we use a simple perturbation
                eps = 1e-5
                for j in range(n_points):
                    h_pert = h_new.copy()
                    h_pert[j] += eps
                    h_pert = np.clip(h_pert, 0, 1)
                    h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
                    corr_pert = correlate(h_pert, 1 - h_pert, mode="full")
                    grad[j] = (corr_pert[max_idx] - corr[max_idx]) / eps
                    
                h_new = h_new + step_size * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
            h = h_new
            current_c5 = compute_c5(h)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            if best_c5 <= 0.3807:
                break
                
        except Exception as e:
            # Fallback to simple local search if cvxpy fails
            try:
                # Run simple local optimization
                h = h + np.random.normal(0, 0.01, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                current_c5 = compute_c5(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            except:
                pass
                
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
