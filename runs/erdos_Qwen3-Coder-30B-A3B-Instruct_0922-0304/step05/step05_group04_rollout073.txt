import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization - binary search with alternating pattern
        h = np.zeros(n_points)
        h[::2] = 1
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()

    # Phase 1: Binary search using simulated annealing
    temperature = 1.0
    cooling_rate = 0.99
    max_sa_iter = 3000
    
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    # Track best from SA phase
    sa_best_h = current_h.copy()
    sa_best_c5 = current_c5
    
    iteration = 0
    while time.time() - start_time < budget_s * 0.6 and iteration < max_sa_iter:
        iteration += 1
        
        # Create neighbor by swapping two random positions
        i, j = np.random.choice(n_points, 2, replace=False)
        h_new = current_h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # Normalize to exact constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        new_c5 = compute_c5(h_new)
        
        # Accept or reject
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
            current_h = h_new
            current_c5 = new_c5
        
        # Track the best
        if current_c5 < sa_best_c5:
            sa_best_c5 = current_c5
            sa_best_h = current_h.copy()
        
        # Cooling
        temperature *= cooling_rate
        
        # Early exit if target achieved
        if sa_best_c5 <= 0.3807:
            break
            
    # Phase 2: Continuous relaxation with CVXPY
    # Use the best binary solution as starting point
    h_start = sa_best_h.copy()
    
    # Use CVXPY to solve continuous optimization
    h_var = cp.Variable(n_points)
    objective = cp.Minimize(cp.max(cp.conv(h_var, 1 - h_var, mode='full')) * dx)
    
    constraints = [
        cp.sum(h_var) == n_points / 2,
        h_var >= 0,
        h_var <= 1
    ]
    
    # Assemble problem and solve
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        h_opt = h_var.value
        h_opt = np.clip(h_opt, 0, 1)
        # Renormalize
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        final_c5 = compute_c5(h_opt)
    except:
        # Fallback to original
        final_c5 = compute_c5(h_start)
        h_opt = h_start
    
    # Final check for best overall
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h_opt.copy()
        
    # Ensure final solution meets constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
