import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize swarm
    n_particles = 20
    swarm = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Fallback initialization  
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize particles
    for _ in range(n_particles):
        h = h_init.copy()
        # Add small random noise
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track best solutions
    global_best_h = h_init.copy()
    global_best_c5 = float('inf')
    
    # PSO parameters
    w_start = 0.7
    w_end = 0.4
    c1 = c2 = 1.5
    max_vel = 0.2
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(n_particles):
            h = swarm[i]
            v = velocities[i]
            
            # Evaluate current particle
            current_c5 = compute_c5(h)
            
            # Update personal best
            if current_c5 < global_best_c5:
                global_best_c5 = current_c5
                global_best_h = h.copy()
            
            # PSO velocity update
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            cognitive = c1 * r1 * (h - h)  # personal best is h itself in this case
            social = c2 * r2 * (global_best_h - h)
            
            v = w * v + cognitive + social
            v = np.clip(v, -max_vel, max_vel)
            velocities[i] = v
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            swarm[i] = h_new
            
        # Early exit if target achieved
        if global_best_c5 <= 0.3807:
            break
    
    return (global_best_h.tolist(), global_best_c5, n_points)
