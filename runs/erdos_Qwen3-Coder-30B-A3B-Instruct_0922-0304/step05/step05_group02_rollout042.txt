import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: clip to [0,1] and scale to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    
    # Parameters
    eta = 0.1
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute all correlations
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        # Compute gradient: the shifted version of (1-h) corresponding to max_k
        # max_k is the index in the full correlation output
        # correlate returns full correlation of length 2*n_points-1
        # we want to find the shift k that maximizes the overlap
        # k = max_k - (n_points-1) 
        k = max_k - (n_points - 1)
        if k >= 0:
            grad = np.concatenate([np.zeros(k), 1 - h[:-k]])
        else:
            grad = np.concatenate([1 - h[-k:], np.zeros(-k)])
        
        # Gradient step
        h_new = h - eta * grad * dx
        
        # Project to feasible region
        h_new = project_h(h_new, target_sum)
        
        # Evaluate new solution
        new_c5 = compute_c5(h_new, dx)
        
        # Update if improved
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            eta = eta * 1.01  # Increase learning rate
        else:
            eta = eta * 0.9   # Decrease learning rate
            
        h = h_new
        iter_count += 1
        
        # Add noise every 100 iterations to escape local optima
        if iter_count % 100 == 0:
            noise = np.random.normal(0, 0.001, n_points)
            h = h + noise
            h = project_h(h, target_sum)
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
