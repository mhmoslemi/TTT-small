import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
    
    # Initialize with a simpler approach: gradient descent with constraints
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simple optimization: perform gradient ascent on the negative of C5
    current_h = h.copy()
    step_size = 0.01
    max_iter = 10000
    patience = 200
    patience_counter = 0
    prev_c5 = float('inf')
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5_val = compute_c5(current_h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = current_h.copy()
            
        if c5_val <= 0.3807:
            break
            
        # Gradient approximation using finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            if h_plus_sum > 0:
                h_plus = h_plus * (target_sum / h_plus_sum)
                
            c5_plus = evaluate_c5(h_plus, dx)
            
            grad[j] = (c5_plus - c5_val) / eps
            
        # Update step
        new_h = current_h - step_size * grad
        
        # Project back to feasible set
        new_h = np.clip(new_h, 0, 1)
        new_h_sum = np.sum(new_h)
        if new_h_sum > 0:
            new_h = new_h * (target_sum / new_h_sum)
            
        # Check for improvement
        if abs(c5_val - prev_c5) < 1e-8:
            patience_counter += 1
            if patience_counter >= patience:
                # Reduce step size
                step_size *= 0.9
                patience_counter = 0
        else:
            patience_counter = 0
            
        prev_c5 = c5_val
        current_h = new_h
            
    return (best_h.tolist(), best_c5, n_points)
