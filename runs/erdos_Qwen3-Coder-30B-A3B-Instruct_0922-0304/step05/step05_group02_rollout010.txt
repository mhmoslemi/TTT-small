import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to satisfy sum(h) = target_sum with bounds [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = target_sum / current_sum
        h = h * scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Create a reasonable starting solution
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    
    # Trust region optimization
    # Momentum parameters
    momentum_decay = 0.9
    learning_rate = 0.1
    line_search_decay = 0.5
    line_search_tolerance = 1e-3
    max_line_search_steps = 10
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Initialize momentum
    m = np.zeros_like(h)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute cross-correlation to find worst shift
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Find which shift caused this maximum
        k = max_idx - (len(h) - 1)
        if k < 0:
            # This corresponds to h[i] * (1 - h[i-k]) for i-k >= 0
            # Gradient w.r.t. h[i] will be (1 - h[i-k]) at positions where i-k>=0
            # We're differentiating (h * (1-h))[k] = sum(h[i] * (1 - h[i-k]))
            # Gradient w.r.t. h[i] = (1 - h[i-k]) at valid positions
            grad = np.zeros_like(h)
            for i in range(len(h)):
                if i - k >= 0:
                    grad[i] = 1 - h[i-k]
        else:
            # This corresponds to h[i] * (1 - h[i-k]) for i-k < 0
            # Gradient w.r.t. h[i] will be (1 - h[i-k]) at positions where i-k < len(h)
            grad = np.zeros_like(h)
            for i in range(len(h)):
                if i - k < len(h):
                    grad[i] = 1 - h[i-k]
        
        # Normalize gradient (optional but helps step size)
        norm_grad = np.linalg.norm(grad)
        if norm_grad > 0:
            grad = grad / norm_grad
        
        # Accelerate with momentum
        m = momentum_decay * m + (1 - momentum_decay) * grad
        # Apply momentum to update direction
        update_direction = m
        
        # Perform line search
        step_size = learning_rate
        current_c5 = compute_c5(h, dx)
        
        for _ in range(max_line_search_steps):
            # Try a step in the update direction
            test_h = h - step_size * update_direction
            test_h = project_to_feasible(test_h, target_sum)
            test_c5 = compute_c5(test_h, dx)
            
            if test_c5 < current_c5:
                h = test_h
                break
            else:
                step_size *= line_search_decay
        
        # Update best solution found
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
