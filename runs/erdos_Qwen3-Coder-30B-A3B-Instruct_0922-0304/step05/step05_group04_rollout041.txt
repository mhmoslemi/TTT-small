import numpy as np
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    mutation_rate = 0.1
    tournament_size = 10
    max_generations = 10000
    
    # Initialize population
    def initialize_population():
        population = []
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
        else:
            # Initialize with random individuals
            for _ in range(population_size - len(population)):
                h = np.random.random(n_points)
                h = h * (n_points / 2) / np.sum(h)
                population.append(h)
        return population
    
    # Fitness function
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Check feasibility
    def is_feasible(h):
        return (np.all(h >= 0) and np.all(h <= 1) and 
                abs(np.sum(h) - n_points / 2) < 1e-6)
    
    # Selection using tournament
    def tournament_selection(population, fitnesses, k=tournament_size):
        selected = []
        for _ in range(len(population)):
            tournament_indices = random.sample(range(len(population)), k)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
            selected.append(population[winner_idx].copy())
        return selected
    
    # Uniform crossover
    def crossover(parent1, parent2):
        child1 = np.empty(n_points)
        child2 = np.empty(n_points)
        for i in range(n_points):
            if random.random() < 0.5:
                child1[i] = parent1[i]
                child2[i] = parent2[i]
            else:
                child1[i] = parent2[i]
                child2[i] = parent1[i]
        return child1, child2
    
    # Mutation
    def mutate(h):
        new_h = h.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                if random.random() < 0.5:
                    new_h[i] += random.uniform(-0.1, 0.1)
                else:
                    new_h[i] = random.choice([0, 1])
                new_h[i] = max(0, min(1, new_h[i]))
        return new_h
    
    # Repair individual to maintain constraints
    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        scale_factor = (n_points / 2) / current_sum
        h = h * scale_factor
        return h
    
    # Main genetic algorithm
    population = initialize_population()
    best_h = None
    best_c5 = float('inf')
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = compute_c5(h)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Crossover and mutation
        new_population = []
        for i in range(0, len(selected), 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % len(selected)]
            child1, child2 = crossover(parent1, parent2)
            child1 = repair(mutate(child1))
            child2 = repair(mutate(child2))
            new_population.extend([child1, child2])
        
        # Keep the best individuals
        population = new_population[:population_size]
        
        # Ensure population size
        while len(population) < population_size:
            h = np.random.random(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Final validation
    assert is_feasible(best_h), "Best solution is not feasible"
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
