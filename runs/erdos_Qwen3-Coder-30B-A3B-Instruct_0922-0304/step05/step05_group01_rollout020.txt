import numpy as np
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 20
    
    # Initialize swarm
    swarm = []
    best_h = None
    best_c5 = float('inf')
    
    # Initialize particles
    for _ in range(swarm_size):
        if 'initial_h_values' in globals():

        else:
            h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
    
    # Initialize personal bests
    personal_best_h = [h.copy() for h in swarm]
    personal_best_c5 = [float('inf')] * swarm_size
    
    # Global best tracking
    global_best_h = None
    global_best_c5 = float('inf')
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # PSO parameters
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    max_iter = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia
        inertia = max(0.4, 0.7 - 0.01 * iteration)
        
        for i in range(swarm_size):
            h = swarm[i]
            
            # Evaluate fitness
            c5 = compute_c5(h)
            
            # Update personal best
            if c5 < personal_best_c5[i]:
                personal_best_c5[i] = c5
                personal_best_h[i] = h.copy()
                
            # Update global best
            if c5 < global_best_c5:
                global_best_c5 = c5
                global_best_h = h.copy()
                
            # Early exit if target achieved
            if global_best_c5 <= 0.3807:
                break
            
            # Update swarm position
            r1, r2 = np.random.rand(2)
            if i == 0 or global_best_h is None:
                # Fallback to random movement for first particle or no global best
                new_h = h + 0.01 * (np.random.rand(n_points) - 0.5)
            else:
                # PSO velocity update
                cognitive_velocity = cognitive * r1 * (personal_best_h[i] - h)
                social_velocity = social * r2 * (global_best_h - h)
                velocity = inertia * (cognitive_velocity + social_velocity)
                new_h = h + velocity
                
            # Ensure feasibility
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            swarm[i] = new_h
            
        if global_best_c5 <= 0.3807:
            break
    
    # Return final solution
    if global_best_h is not None:
        final_c5 = global_best_c5
        final_h = global_best_h.tolist()
    else:
        # Fallback to best particle
        final_h = personal_best_h[np.argmin(personal_best_c5)].tolist()
        final_c5 = min(personal_best_c5)
    
    return (final_h, final_c5, n_points)
