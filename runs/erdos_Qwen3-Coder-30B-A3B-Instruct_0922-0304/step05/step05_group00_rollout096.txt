import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def linearize_constraints(h_prev, k, dx, n_points):
    """Linearize the constraint for lag k."""
    # h[i] * (1 - h[i+k]) term
    # Linearized as: h[i] - h[i]*h_prev[i+k] - h_prev[i]*h[i+k] + h_prev[i]*h_prev[i+k]
    # Constraint: h[i] - h[i]*h_prev[i+k] - h_prev[i]*h[i+k] + h_prev[i]*h_prev[i+k] <= C5*dx
    # Simplifies to: h[i] - h_prev[i]*h[i+k] <= C5*dx + h[i]*h_prev[i+k] - h_prev[i]*h_prev[i+k]
    # But we rewrite as: h[i] - h_prev[i] - h[i]*h_prev[i+k] + h_prev[i]*h_prev[i+k] <= C5*dx - h_prev[i]*h[i+k]
    # Final form we compute: a_k * h + b_k <= C5*dx
    # But easier approach: to minimize the max overlap, we linearize at h_prev,
    # and express overlap_k as dot product of h with a known coefficient vector.
    # For fixed h_prev:
    # overlap_k = sum_{i} h[i] * (1 - h_prev[i+k]) = sum_{i} h[i] - sum_{i} h[i]*h_prev[i+k]
    
    # Vector for linearization: row vector for h, except some shifted part
    a_k = np.zeros(n_points)
    # Set contribution from h[i]
    a_k[:] = 1.0
    # Subtract contribution from h_prev[i+k] 
    for i in range(n_points - k):
        a_k[i] -= h_prev[i+k]
    b_k = np.sum(h_prev[:n_points-k] * h_prev[k:])
    # So constraint becomes a_k * h + b_k <= C5 * dx
    return a_k, b_k

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 70
    dx = 2.0 / n_points  
    target_sum = n_points / 2

    # Read initial construction if available, otherwise a step function
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_to_feasible(h, target_sum)

    # Objective: minimize C5
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)

    if best_c5 <= 0.3807:
        return (h.tolist(), best_c5, n_points)

    max_iter = 1000
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        if c5 <= 0.3807:
            break
            
        # Identify most constraining lags (those with largest overlap) 
        overlap = correlate(h, 1 - h, mode="full")
        # Consider top 10% overlapping lags, skip first and last (they're dangling)
        # The middle lags are most informative
        n_lags = len(overlap) 
        lag_indices = np.arange(n_lags)
        # Only consider valid lags where both operands are meaningful
        # Correlate full computes lags from -n+1 to n-1
        center_lag_indices = lag_indices[n_lags//2 - n_points//2 : n_lags//2 + n_points//2]
        sorted_lag_indices = center_lag_indices[np.argsort(-overlap[center_lag_indices])]
        # Take top 10% or max 10
        top_k = min(10, max(1, n_points // 10))
        selected_lags = sorted_lag_indices[:top_k]
        
        # Create linearized subproblem
        A = np.zeros((len(selected_lags), n_points))  
        b = np.zeros(len(selected_lags))
        
        # Build linearized constraints
        for i, k in enumerate(selected_lags):
            # k is offset, translate to full array index
            full_idx = k + n_points - 1
            # Calculates a_k, b_k for constraint h[i] * (1 - h[i+k]) <= C5*dx
            # Or more accurately, we linearize h[i] - h[i] * h_prev[i+k] <= C5*dx + h_prev[i]*h[i+k] - h_prev[i]*h_prev[i+k]
            # The linearized form at h_prev is: a_k * h + b_k <= C5 * dx
            # a_k[i] = 1.0, a_k[i+k] = -h_prev[i]  
            a_k = np.zeros(n_points)
            a_k[:] = 1.0
            for j in range(n_points - k):
                a_k[j] -= h[j + k]  # We are linearizing around h
            b_k = np.sum(h[:n_points - k] * h[k:])  # The constant part
            A[i] = a_k
            b[i] = b_k
            
        # Now solve convex relaxation:
        # minimize C5
        # subject to A @ h + b <= C5 * dx
        #         sum(h) = target_sum
        #         0 <= h[i] <= 1
        
        # Convert to standard form: solve min C5 s.t. linear_block * [h; C5] <= rhs
        # But let's reformulate directly into h-space with reparametrized constraints
        
        # Use simpler method: optimize h directly with constraints
        # Solve:
        #   minimize f(h) = max_k overlap_k(h)  
        #   subject to   sum(h) = target_sum, 0 <= h[i] <= 1
        
        # Instead of directly solving, use projected gradient method or simpler optimization
        # Keep it simple and do projected update in direction of steepest descent of max overlap
        
        # Find the max overlap lag and focus on it
        max_lag = np.argmax(overlap) - (n_points - 1)
        # We know that f(x) = sum h[i] * (1 - h[i + k]) is the current overlap  
        # To minimize it we compute the gradient: d/dh_i of h[i] * (1 - h[i + k])
        #        = 1 - h[i + k] - h[i] * 0 = 1 - h[i + k] 
        # So for each h[i], we compute the gradient contribution from overlaps that include it
        
        # For gradient of overlap_k, systematic approach:
        # Gradient w.r.t. h[i]: if i+k is valid for i+k < n, we have term h[i] * (1-h[i+k])  
        #    => 1 - h[i+k]  if i+k < n
        #    => 0           if i+k >= n
        # Since overlap_k = sum_{valid i} h[i] * (1 - h[i+k]), used for our test
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + max_lag < n_points:
                grad[i] += 1 - h[i + max_lag]
            if i - max_lag >= 0 and max_lag > 0:
                grad[i] += 1 - h[i - max_lag]
                
        # Gradient update with decaying step size
        step = 0.01
        h_new = h - step * grad
        h_new = project_to_feasible(h_new, target_sum)
        
        # Now, save current h as reference and also execute a more focused update strategy
        
        # For efficiency and avoiding blow-up, do some smoothing: 
        # smooth the change by interpolation
        h = 0.95 * h + 0.05 * h_new  # blend
        h = project_to_feasible(h, target_sum)

    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
