import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # PSO parameters
    num_particles = 20
    max_iter = 1000
    w_start = 0.8
    w_end = 0.1
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize particles
    particles = [h.copy()]
    for _ in range(num_particles - 1):
        h_pert = h + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        particles.append(h_pert)
    
    # Initialize velocities and personal bests
    velocities = [np.zeros(n_points) for _ in range(num_particles)]
    p_best = [p.copy() for p in particles]
    p_best_fitness = [compute_c5(p) for p in particles]
    
    # Global best tracking
    best_idx = np.argmin(p_best_fitness)
    g_best = p_best[best_idx].copy()
    g_best_fitness = p_best_fitness[best_idx]
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Ensure constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate new position
            fitness = compute_c5(particles[i])
            
            # Update personal best
            if fitness < p_best_fitness[i]:
                p_best[i] = particles[i].copy()
                p_best_fitness[i] = fitness
                
                # Update global best
                if fitness < g_best_fitness:
                    g_best = particles[i].copy()
                    g_best_fitness = fitness
                    
        # Early exit if target reached
        if g_best_fitness <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(g_best)
    
    return (g_best.tolist(), final_c5, n_points)
