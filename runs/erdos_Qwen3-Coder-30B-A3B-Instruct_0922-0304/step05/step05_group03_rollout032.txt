import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, k_max, dx, n_points):
    """Compute gradient of overlap at k_max with respect to h."""
    # For overlap at lag k_max, we have:
    # overlap[k_max] = sum(h[i] * (1 - h[i + k_max])) for valid i
    # gradient[i] = - (1 - h[i + k_max]) if i + k_max < n_points else 0
    # gradient[i] = - h[i - k_max] if i - k_max >= 0 else 0
    
    gradient = np.zeros(n_points)
    for i in range(n_points):
        # Contribution from h[i] * (1 - h[i + k_max])
        if i + k_max < n_points:
            gradient[i] -= (1 - h[i + k_max])
        # Contribution from h[i] * (1 - h[i - k_max])
        if i - k_max >= 0:
            gradient[i] -= h[i - k_max]
    
    return gradient * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    # Set learning parameters
    learning_rate = 0.1
    patience = 0
    patience_limit = 10
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        c5 = evaluate_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if best_c5 <= 0.3807:
            break
            
        # Find the maximum overlap lag
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        
        # Compute gradient
        gradient = compute_gradient(h, k_max, dx, n_points)
        
        # Update h
        h = h - learning_rate * gradient
        
        # Project to feasible region
        h = normalize_h(h, target_sum)
        
        # Reduce learning rate if no progress
        if patience > patience_limit:
            learning_rate *= 0.5
            patience = 0
            
        # Occasionally add small random perturbations
        if iteration % 50 == 0:
            perturbation = np.random.normal(0, 0.005, n_points)
            h = h + perturbation
            h = normalize_h(h, target_sum)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
