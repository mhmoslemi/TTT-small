import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
    
    # Gradient-based optimization with constraints
    from scipy.optimize import minimize
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Perturb the initial solution
    initial_perturbed = h + np.random.normal(0, 0.01, n_points)
    initial_perturbed = normalize_h(initial_perturbed, target_sum)
    
    # Callback to track progress
    iteration = 0
    def callback(x):
        nonlocal iteration, best_c5, best_h
        iteration += 1
        c5 = compute_c5(x)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x.copy()
        if time.time() - start_time > budget_s - 1:
            return True
        return False
    
    # Define constraints
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    # Set bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Start with the initial solution
    x0 = initial_perturbed.copy()
    
    # Use multiple local minimizations with different starting points
    n_starts = 10
    for i in range(n_starts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbation to avoid local minima
        x_start = h + np.random.normal(0, 0.02, n_points)
        x_start = normalize_h(x_start, target_sum)
        
        # Local optimization
        try:
            result = minimize(
                compute_c5,
                x_start,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                callback=callback,
                options={'maxiter': 50, 'ftol': 1e-10}
            )
            if result.success:
                c5 = compute_c5(result.x)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = result.x.copy()
            if best_c5 <= 0.3807:
                break
        except:
            continue
    
    # Final check
    final_c5 = compute_c5(best_h)
    
    # Make sure we never exceed the budget
    while time.time() - start_time < budget_s - 1 and final_c5 > 0.3807:
        # Try perturbing solution further to try to improve it
        h_perturbed = best_h + np.random.normal(0, 0.005, n_points)
        h_perturbed = normalize_h(h_perturbed, target_sum)
        
        # Local optimization on this perturbed version
        try:
            result = minimize(
                compute_c5,
                h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,

        if best_c5 <= 0.3807:
