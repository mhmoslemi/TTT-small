import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def halton_sequence(n, base=2):
    """Generate n points of 1D Halton sequence."""
    result = np.zeros(n)
    for i in range(n):
        f = 1.0
        result[i] = 0.0
        while f >= 1.0:
            result[i] += (i % base) * f
            i //= base
            f /= base
    return result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
    
    # Multi-start optimization with Halton sequences
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Generate 50 Halton-based initial points
    num_starts = 50
    initial_points = []
    for i in range(num_starts):
        # Use Halton to create diverse initializations
        h_init = halton_sequence(n_points, base=2)
        # Scale and shift to [0,1]
        h_init = (h_init - h_init.min()) / (h_init.max() - h_init.min())
        h_init = normalize_h(h_init, target_sum)
        initial_points.append(h_init)
    
    # Add the current solution as one of the starting points
    initial_points.append(h.copy())
    
    # Define constraint for optimization
    def constraint_sum(h):
        return np.sum(h) - target_sum

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    iterations = 0
    for h_start in initial_points:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Run local optimization on this initial point
        result = minimize(
            compute_c5,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        
        if result.success:
            h_opt = result.x
            h_opt = normalize_h(h_opt, target_sum)
            c5_opt = compute_c5(h_opt)
            
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                
        iterations += 1
        
        # Early exit if we're close to target
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
