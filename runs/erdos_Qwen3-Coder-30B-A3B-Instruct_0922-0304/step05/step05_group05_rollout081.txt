import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # PSO parameters
    n_particles = 20
    max_iter = 1000
    inertia_weight_start = 0.8
    inertia_weight_end = 0.1
    cognitive_coeff = 2.0
    social_coeff = 2.0
    velocity_max = 0.1

    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_scores = []
    
    # Create initial population around the parent or fallback
    for i in range(n_particles):
        if i == 0:
            particle = h.copy()
        else:
            # Create variation by adding noise
            particle = h + np.random.normal(0, 0.02, n_points)
            particle = np.clip(particle, 0, 1)
        # Normalize to sum to n_points/2
        particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        p_best.append(particle.copy())
        p_best_scores.append(compute_c5(particle))

    # Global best
    best_idx = np.argmin(p_best_scores)
    g_best = p_best[best_idx].copy()
    g_best_score = p_best_scores[best_idx]

    # PSO main loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = inertia_weight_start - (inertia_weight_start - inertia_weight_end) * iter_num / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = cognitive_coeff * r1 * (p_best[i] - particles[i])
            social = social_coeff * r2 * (g_best - particles[i])
            velocities[i] = w * velocities[i] + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce constraints
            particles[i] = np.clip(particles[i], 0, 1)
            # Rescale to correct sum
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Compute new score
            score = compute_c5(particles[i])
            
            # Update personal best
            if score < p_best_scores[i]:
                p_best[i] = particles[i].copy()
                p_best_scores[i] = score
                
                # Update global best if needed
                if score < g_best_score:
                    g_best = particles[i].copy()
                    g_best_score = score
                    
        # Early exit
        if g_best_score <= 0.3807:
            break

    return (g_best.tolist(), g_best_score, n_points)
