import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use a hybrid approach: local search with simulated annealing ideas
    # Parameters for simulated annealing
    temp = 1.0
    temp_decay = 0.999
    min_temp = 1e-6
    perturbation_magnitude = 0.1
    max_iterations = 1000000
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        
        # Annealing temperature decay
        if temp > min_temp:
            temp *= temp_decay
        
        # Generate a candidate solution with perturbation
        h_new = h + np.random.normal(0, perturbation_magnitude, n_points)
        h_new = normalize_h(h_new, target_sum)
        
        # Evaluate both solutions
        current_c5 = evaluate_c5(h, dx)
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject the new solution
        if new_c5 < current_c5 or np.random.random() < np.exp(-(new_c5 - current_c5) / temp):
            h = h_new
            current_c5 = new_c5
        
        # Track best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    # Fine-tune the best solution with local optimization
    def objective(h):
        return evaluate_c5(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Refine with SLSQP
    result = minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-10}
    )
    
    if result.success:
        final_h = result.x
        final_h = normalize_h(final_h, target_sum)
        final_c5 = evaluate_c5(final_h, dx)
        best_c5 = min(best_c5, final_c5)
        best_h = final_h
    
    return (best_h.tolist(), best_c5, n_points)
