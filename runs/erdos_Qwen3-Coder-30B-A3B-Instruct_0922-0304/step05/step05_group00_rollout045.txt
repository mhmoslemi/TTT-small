import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum"""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from pre-loaded construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Fall back to a simple step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Iterate with constraint linearization
    max_iter = 1000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Focus on high overlap lags to improve
        # Just go with 10 most significant lags
        selected_lags = np.argsort(overlap)[-10:] 
        selected_lags = selected_lags[selected_lags >= 0]  # Keep only non-negative
        
        # Use a trust region approach
        # Simple: gradient-free min step
        step_size = 0.01
        h_new = h.copy() + np.random.uniform(-step_size, step_size, n_points)
        h_new = project_to_feasible(h_new, target_sum)
        
        # Also attempt a local descent step on current problem
        # What if we estimate the gradient and move?
        # Since we don't have an easy gradient, we use a random perturbation
                
        h_trial = h_new.copy()
        c5_trial = compute_c5(h_trial, dx)
        
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_trial.copy()
        
        # Small random exploration
        h = h_trial
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        iter_count += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
