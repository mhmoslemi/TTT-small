```python
import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize from existing construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        # Default initialization
        h = np.ones(n_points) * 0.5
        h[0:n_points//4] = 1.0
        h[3*n_points//4:n_points] = 0.0
        h = project_h(h, target_sum)

    # Algorithm parameters
    momentum = 0.95
    alpha = 0.05
    max_iter = 1000
    patience = 50
    best_c5 = float('inf')
    best_h = h.copy()
    prev_c5 = float('inf')
    patience_counter = 0

    v = np.zeros_like(h)
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k]
        
        # Compute gradient (corresponding to the shift with max overlap)
        # For shift k, we get (1 - h) convolved with h at position k
        # This corresponds to the gradient of overlap w.r.t. h
        grad = np.zeros_like(h)
        if max_k < n_points:
            # h convolved with (1-h) at shift k
            shifted_h = np.roll(1 - h, max_k)
            grad = shifted_h
        else:
            # The indices when convolving a sequence of length n with another of length n
            # gives n*2-1 results
            cache_len = 2 * n_points - 1
            k = max_k - (cache_len - n_points)
            shifted_h = np.roll(h, k)
            grad = 1 - shifted_h
            
        # Simple gradient step
        v = momentum * v + grad
        h_new = h - alpha * v
        
        # Project to feasible set
        h_new = project_h(h_new, target_sum)
        
        # Evaluate new solution
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Early stopping if no improvement for a while
        if patience_counter >= patience:
            break
            
        h = h_new.copy()
        
        # Occasionally do fine tuning
        if iteration % 10 == 0:
            h = project_h(h, target_sum)

    # Final validation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
```
