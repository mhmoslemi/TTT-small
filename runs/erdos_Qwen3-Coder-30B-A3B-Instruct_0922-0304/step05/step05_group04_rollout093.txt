import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    generations = 1000
    mutation_rate = 0.1
    tournament_size = 10
    
    # Initialize population
    def initialize_population():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population = [h.copy()]
            for _ in range(population_size - 1):
                h_mut = h + np.random.normal(0, 0.05, n_points)
                h_mut = np.clip(h_mut, 0, 1)
                h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
                population.append(h_mut)
        else:
            population = []
            for _ in range(population_size):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                population.append(h)
        return population
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Tournament selection
    def tournament_selection(population, fitnesses):
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
            selected.append(population[winner_idx].copy())
        return selected
    
    # Uniform crossover
    def crossover(parent1, parent2):
        mask = np.random.rand(n_points) > 0.5
        child = np.where(mask, parent1, parent2)
        return child
    
    # Mutation with repair
    def mutate(h):
        h_mut = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                if np.random.rand() < 0.5:
                    h_mut[i] += np.random.normal(0, 0.1)
                else:
                    h_mut[i] = np.random.rand()
        # Repair
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
        return h_mut
    
    # Main genetic algorithm
    population = initialize_population()
    best_fitness = float('inf')
    best_individual = None
    
    for generation in range(generations):
        # Compute fitness
        fitnesses = [compute_c5(ind) for ind in population]
        
        # Track best individual
        gen_best_idx = np.argmin(fitnesses)
        if fitnesses[gen_best_idx] < best_fitness:
            best_fitness = fitnesses[gen_best_idx]
            best_individual = population[gen_best_idx].copy()
            
        # Early termination
        if best_fitness <= 0.3807:
            break
            
        # Time check
        if time.time() - start_time > budget_s - 1:
            break
            
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < population_size else selected[0]
            
            child1 = crossover(parent1, parent2)
            child2 = crossover(parent2, parent1)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
            
        population = new_population[:population_size]
        
        # Print progress
        if generation % 50 == 0:
            print(f"Gen {generation}: Best C5 = {best_fitness:.6f}")
    
    # Final evaluation
    final_c5 = compute_c5(best_individual)
    
    return (best_individual.tolist(), final_c5, n_points)
