import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iteration loop
    max_iter = 50
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps
        overlap = correlate(h, 1 - h, mode="full")
        # Get toplags contributing most to C5
        top_k_indices = np.argsort(-overlap)[:max(5, n_points // 10)]
        
        # Formulate linear programming relaxation
        h_var = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        constraints = []
        # Non-negativity and upper bounds
        constraints += [h_var >= 0, h_var <= 1]
        # Sum constraint
        constraints += [cp.sum(h_var) == target_sum]
        # Linearized overlap constraints
        for k in top_k_indices:
            # For correlation at lag k
            h_shifted = np.roll(h, -k)
            # Approximate h[i] * (1 - h[i+k]) - h[i+k] * h[i]  
            # as: h[i] * (1 - h[i+k]) - h[i+k] * h[i]
            # By linearizing we sum terms of form: h[i] * (1-h[i+k]) for i = 0..N-k-1
            # and h[i+k] * h[i] for i = 0..N-k-1
            # But we want: h[i] * (1 - h[i+k]) + h[i+k] * (1 - h[i]) 
            #           = h[i] - h[i]*h[i+k] + h[i+k] - h[i]*h[i+k]
            #           = h[i] + h[i+k] - 2*h[i]*h[i+k]
            # But that's not what we are dealing with in the overlap:
            # sum(h[i]*(1 - h[i+k]))  
            # sum(h[i]*1 - h[i]*h[i+k]) 
            # sum(h[i] - h[i]*h[i+k]) 
            # So overlap_k = sum(h[i] - h[i] * h[i+k]) 
            
            # The actual overlap term is
            # overlap_k = sum(h[i] * (1 - h[i+k]), i = 0 to n-k-1)
            # This is a bilinear term
            
            # Our linear approximation will be 
            # let h_current = h[i], h_prev = h[i+k]  
            # Then overlap_k = sum(h_current * (1 - h_prev))  
            # = sum(h_current - h_current*h_prev)
            # = sum(h_current) - sum(h_current * h_prev) 
            
            # A common linearization: We use the current estimate h
            # for h_prev, and define 
            # L = sum(h_current * (1 - h_prev_current))  -- linearized overlap  
            # We'll add constraint: L <= C5_var
            
            # So the linear term for overlap_k is:
            # sum(h[i] * (1 - h[i+k])) - sum(h[i+k] * h[i])
            # Experimentally we will just consider the full set of pairwise products
            # and linearize using the current h
            
            # The terms would be something like h[i]*1 - h[i]*h[i+k]
            # But what we really compute is overlap_k = sum_{i=0}^{n-k-1} h[i] * (1-h[i+k])
            # So for each such product h[i]*h[i+k], we'll linearize as:
            # h[i]*(1-h[i+k]) ≈ h[i] - h[i] * h[i+k]
            # But we use the first-order Taylor expansion at current h:
            # h[i]*(1 - h[i+k]) <= h[i] - h[i+k]*h[i] 
            # But we can also do:
            # h[i]*(1 - h[i+k]) ≈ h[i] - h[i]*h[i+k]
            # And we do linearization around current h:
            # Let h0 be h[i], h1 be h[i+k]  
            # Linear approximation:
            # h0*(1-h1) ≈ h0*(1-h1_current) + (1-h1_current) * (h0-h0_current)  
            # No, that won't work since we don't know h0 and h1 independently.
            # Simpler: h[i]*(1 - h[i+k]) ≈ h[i] - h[i]*h[i+k]  
            # Linear in h[i] but constant w.r.t. h[i+k] (or vice versa) if we use current values
            
            # Let's linearize about h:
            # For term i, h[i]*(1 - h[i+k]) 
            # The linearization about current values is:
            # h[i]*(1 - h[i+k]) ≈ h[i]*(1 - h[i+k]) + (1-h[i+k])*(h[i] - h[i]) 
            # = h[i]*(1 - h[i+k]) + 0 = h[i]*(1 - h[i+k])
            # That doesn't help. Let's try:
            # We can use 
            # h[i]*(1-h[i+k]) ≤ 1*h[i] 
            # But that's weak. 
            
            # Try by treating the actual overlap definition:
            # overlap_k = sum(h[i] * (1 - h[i+k]))
            # = sum(h[i]) - sum(h[i]*h[i+k])
            # 
            # For each i, h[i]*h[i+k] can be linearized via  
            # z[i,k] = h[i]*h[i+k]
            # z[i,k] ≤ h[i] (since h[i+k] ∈ [0,1])
            # z[i,k] ≤ h[i+k] (since h[i] ∈ [0,1])
            # z[i,k] ≥ h[i] + h[i+k] - 1 (since for binary variables it'd be true)
            # But since h[i,k] are continuous, we treat:  
            # h[i]*h[i+k] = sum of products. So linearize by ignoring h[i+k] term
            # And approxmate by h[i] = h[i] (since h[i+k] is constant in that 
            # context, its current value). So:
            # sum(h[i]*(1-h[i+k])) ≈ sum(h[i]) - sum_approx(h[i]*h[i+k])
            
            # A better approach:
            # For linearization of sum(h[i] * (1 - h[i+k])):
            # Let f(h) = sum(h[i] * (1 - h[i+k])) for fixed k.   
            # Gradient would be f'(h) = (1 - h[i+k], ..., 1-h[i+k], ...) 
            # We'll use a simple version:
            # Using current h values, we approximate sum(h[i] * (1 - h'[i+k]))
            # We are trying to bound sum(h[i]*(1-h[i+k])) <= C5 
            # We make this an inequality constraint via linearization:
            # The term h[i] * (1 - h[i+k])  
            # When h[i] is fixed, we face:
            # constant term - h[i] * h[i+k]. So:
            # sum(h[i] * (1 - h[i+k])) <= sum(h[i]) - sum(h[i] * h[i+k]) 
            
            # Linearize the full term:
            # f_k(h) = sum(h[i] * (1 - h[i+k]))  
            # We could write it as h[i] - h[i]*h[i+k]
            # We can compute linearization around h as:  
            # = sum( h[i] * (1 - h0[i+k]) + (1-h0[i+k])*(h[i]-h0[i]) )
            # Omit second order term (h[i]-h0[i])*(h[i+k]-h0[i+k]) 
            # = sum( h[i] * (1-h0[i+k]) + (1-h0[i+k])*(h[i]-h0[i]) )
            # = sum(h[i] - h[i]*h0[i+k] + (1-h0[i+k])*(h[i]-h0[i]))
            # = sum(h[i] - h[i]*h0[i+k] + h[i] - h0[i] - h[i]h0[i+k] + h0[i]h0[i+k])
            # = sum(2 h[i] - h0[i] - 2 h[i] h0[i+k] + h0[i]h0[i+k] )
            # = sum((2 h[i] - h0[i]) - h[i] (2 h0[i+k] - h0[i+k]) )   
            # = sum(2h[i] - h0[i] - h[i] * h0[i+k] )  
            # Simplest is to linearize h[i] * (1-h[i+k]) as: h[i] - h[i]*h[i+k]  
            # Use the value h[i+k] from current solution, treating it fixed.
            # Let h_fixed_k = h[i+k] (or equivalently constant term also here).                 
            # We're approximating sum(h[i]*(1-h[i+k])) 
            # as sum(h[i] - h[i]*h_fixed[i+k])  = sum(h[i]) - sum(h[i]*h_fixed[i+k])
            
            # Therefore:
            # linear approximation:
            # sum(h[i] * (1 - h0[i+k]))  ≤ C5_var
            # where h0 is current h values 
            
            # Let's directly solve LP constraints:
            # We're trying to bound each 
            # overlap_k = sum_{i=0}^{n-k-1} h[i]*(1 - h[i+k]) <= C5_var
            
            # If we linearize, we replace h[i+k] by h[i+k] (the current value)
            # This gives us a linear constraint:
            # sum(h[i](1 - h[i+k])) ≤ C5_var  
            # With current values of h[i+k] fixed, this is a linear function of h[i]
            # For each i: h[i] - h[i]*h[i+k] 
            # Linearization around h:
            # = h[i] - h[i] * h[i+k]  
            # = h[i] * (1 - h[i+k])  
            # So: h[i] * (1 - h[i+k]) 
            # We need    
            # sum over i from 0 to n-k-1 of (h[i] * (1 - h[i+k])) <= C5_var 
            
            # Formulate exact LP constraint for this:
            # we know terms:
            # so the linear term is sum(h[i] * (1 - h[i+k])) 
            # where h[i+k] = h[i+k] (current).
            # 
            # So the constraint is:
            # sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k]) <= C5_var  
            # with terms like 1 * h[i] - h[i] * h[i+k]  
            # = sum h[i] - sum h[i] * h[i+k] 
            # Let's write:
            # v = h[i] * (1 - h[i+k])  
            # where i = 0 to n-k-1.  
            # For each i, v = h[i] - h[i]*h[i+k]  
            # When h[i+k] is fixed (current), we can write constraint:
            # sum(v) <= C5_var  
            # With  where v[i] =  h[i] - h[i]*h[i+k] 
            # = sum(h[i] - h[i]*h[i+k]) 
            #
            # If we knew h[i+k], we can compute:
            # l = h[i] - h[i]*h[i+k] 
            # we bound Sum_l <= C5_var
            # If h[i,k] = h[i+k] is a specific number, this simplifies
            # So for fixed current h we get:
            # ¿constrain?  
            # We assume h[i+k] is fixed in evaluating lag k 
            # (up to need to keep shifting)
            # 
            # h[i] * (1 - h[i+k])  
            # We linearize around current h as:
            # h[i]*(1-h[i+k]) ≈ h[i]*(1 - h[i+k]) + 1*(h[i] - h[i])  
            # No, wrong.    
            # Let h[i] = h[i],  h[i+k] = h[i+k] 
            # Simple idea:
            # h[i]*(1 - h[i+k])  
            # Fix h[i+k] = h[i+k], then linear in h[i] 
            # We linearize it w.r.t. current h[i]:  
            # h[i] + 0*(h[i] - current_h[i]) = h[i] 
            # We can approximate the term:
            # Let us consider (1 - h[i+k])  
            # What is h[i+k]? It's h[i+k]

            # h[i]*(1-h[i+k]) = x[i]* (1 - y[i])
            # You can linearize it, for 0 <= x,y <= 1:
            # x[i] (1 - y[i]) ≤ 1 * x[i]  
            #          ≤ x[i] + (1 - y[i]) - 1 = x[i] - y[i]

            # L(x,y) ≈ x[0] * (1 - y[0]) + (1 - y[0]) * (x - x[0]) + (x[0] - x[0]) 
            # = x * (1 - y[0])  
            # 
            # Better approach we use already computed terms:
            # We form a linear approximation of
            # overlap_k = sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k])
            # Known portions, fixed h:
            # So for term i, let h[i+k] be the known constant at current h 
            # h[i] * (1 - h[i+k]) 
            # Become: h[i] - h[i]*h[i+k]  
            # Re-index so it's sum_{i=0}^{n-k-1} (h[i] - h[i]*h[i+k])  
            # Thus: 
            # sum(h[i]) - sum(h[i]*h[i+k]) ≤ C5_var 
            # Prepare this constraint
            left = h[:n_points-k]
            right = h[k:]
            # Now compute sum(h[i] * (1 - h[i+k]))  
            # = sum(h[i]) - sum(h[i]*h[i+k])  
            zero_sum = np.sum(h[:n_points-k])
            hprod = np.sum(h[:n_points-k] * h[k:])
            
            # So linearized overlap_k = zero_sum - hprod  
            # We check whether this is correct:
            # Actually, let's redo it:
            # overlap_k = sum_{i} h[i]*(1-h[i+k]) 
            # Here the sum runs to n-k-1 = n_points-k-1  
            # Let's directly compute it and linearize it:
            # For k = 0, it is sum(h[i] * (1 - h[i]))  
            # So it's not aligned with our LP - let's adjust.
            # The index for lag k: 0 to n_points-k-1
            # So sum(h[i] * (1 - h[i+k])) for i from 0 to n_points-k-1  
            # = sum(h[i]) - sum(h[i]*h[i+k]) where i from 0 to n_points-k-1  
            # i is from 0 to n_points-k-1, but we can reduce sum to n-k of h[i]*h[i+k] 
            
            # Let's compute a true linear constraint:
            # sum for i=0:n_points-k-1 of h[i] * (1-h[i+k])  
            # = sum of h[i] - sum of h[i]*h[i+k]  
            # = sum(h[0] to h[n_points-k-1]) - sum of h[i]*h[i+k] for i=0 to n_points-k-1  
            # We approximate this using h[i+k] from current solution:
            # So 1st term = sum(h[:n_points-k])  
            # 2nd term = sum(h[:n_points-k]*h[k:])  
            # = result = (h[i] - h[i]*h[i+k])  
            # The affine approximation is via  
            # h[i] - h[i] * h[i+k]
            # = h[i] * (1-h[i+k])  at current point  
            # The linearization about h is just the same because terms are bilinear but all terms use specified 
            # h[i+k] and we know how to compute it.  
            # So we have coefficient vectors:
            # Constraint coefficient matrix for the linear approximation:
            # Let's build the problem correctly  
            coeffs = np.zeros(n_points)  
            coeffs[:n_points-k] = 1.0  # from h[i] terms  
            # For h[i]*h[i+k], we approximate with linearization:
            # h[i] - h[i]*h[i+k] = h[i] - (h[i]*h[i+k])
            # So we modify coeffs by subtracting h[i+k] for indices i:
            # So for each i we have:
            # coeffs[i] = 1
            # then subtract h[i+k] for the h[i]*h[i+k] term:
            if k > 0:
                for i in range(n_points-k):
                    coeffs[i] -= h[i+k] 
            # The linearization gives: coeffs.T * h_var <= C5_var  
            my_sum = np.sum(coeffs * h_var)  # yyyyyyy 
            constraints += [my_sum <= C5_var]
        
        # Objective function
        objective = cp.Minimize(C5_var)
        
        # Solve the program
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INFEASIBLE]:
                h_new = h_var.value
                # Peel the constraint violation
                h_new = np.clip(h_new, 0, 1) 
                # Scale to meet requirement
                h_new = h_new * (target_sum / np.sum(h_new))
                # Compute C5
                proposed_c5 = compute_c5(h_new, dx)  
                
                if proposed_c5 < best_c5:
                    best_c5 = proposed_c5
                    best_h = h_new.copy()
                
                h = h_new.copy()   
                
                # Early exit if target achieved
                if best_c5 <= 0.3807:
                    break
        except:
            pass
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
