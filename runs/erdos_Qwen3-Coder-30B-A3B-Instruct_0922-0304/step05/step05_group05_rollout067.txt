import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: 3-region piecewise function
        h = np.ones(n_points)
        mid = n_points // 2
        h[mid:] = 0
        # Add some randomness to break symmetry
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def rescale_to_sum(h, target_sum):
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * target_sum / current_sum
        return h
    
    def swap_segments(h, i1, i2, size):
        h_new = h.copy()
        # Swap segments
        temp = h_new[i1:i1+size].copy()
        h_new[i1:i1+size] = h_new[i2:i2+size]
        h_new[i2:i2+size] = temp
        return h_new
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Swap-based local search hyperparameters
    n_swaps = 1000
    swap_size = 5
    temperature = 0.1
    
    # Track time
    for _ in range(n_swaps):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Select two random non-overlapping segments
        segment1_start = np.random.randint(0, n_points - swap_size)
        segment2_start = np.random.randint(0, n_points - swap_size)
        
        # Ensure they don't overlap
        while (abs(segment1_start - segment2_start) < swap_size):
            segment2_start = np.random.randint(0, n_points - swap_size)
            
        # Perform swap
        h_new = swap_segments(h, segment1_start, segment2_start, swap_size)
        
        # Ensure values stay in [0,1] and rescale to constraint
        h_new = np.clip(h_new, 0, 1)
        h_new = rescale_to_sum(h_new, n_points / 2)
        
        # Compute new C5
        new_c5 = compute_c5(h_new)
        
        # Accept or reject
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h_new.copy()
        elif np.random.rand() < np.exp(-(new_c5 - best_c5) / temperature):
            h = h_new
        else:
            pass  # Reject the swap
    
    # Final polishing with SLSQP if time allows
    if time.time() - start_time < budget_s - 1:
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        
        try:
            result = minimize(
                compute_c5,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 30, 'ftol': 1e-8}
            )
            if result.success:
                final_h = result.x
                final_h = np.clip(final_h, 0, 1)
                final_h = final_h * (n_points / 2) / np.sum(final_h)
                final_c5 = compute_c5(final_h)
                if final_c5 < best_c5:
                    best_h = final_h
                    best_c5 = final_c5
        except:
            pass  # If SLSQP fails, keep current best
    
    return (best_h.tolist(), best_c5, n_points)
