import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.95
    min_T = 1e-8
    max_iter = 5000
    iter_per_T = 100
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    T = T0
    current_h = h.copy()
    current_c5 = best_c5
    
    iter_count = 0
    time_limit = budget_s - 1
    
    while T > min_T and time.time() - start_time < time_limit:
        for _ in range(iter_per_T):
            if time.time() - start_time > time_limit:
                break
                
            # Perturb solution
            h_new = current_h + np.random.normal(0, T * 0.05, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Compute new objective
            c5_new = compute_c5(h_new)
            
            # Accept or reject
            delta = c5_new - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                current_h = h_new
                current_c5 = c5_new
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
            
            iter_count += 1
            
            # Early exit
            if best_c5 <= 0.3807:
                break
        
        T *= alpha
        
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
