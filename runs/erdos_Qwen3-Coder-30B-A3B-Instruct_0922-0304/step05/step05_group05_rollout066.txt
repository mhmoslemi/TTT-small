import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_swaps = 1000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: three regions
        h = np.zeros(n_points)
        mid1 = n_points // 4
        mid2 = 3 * n_points // 4
        h[:mid1] = 0.5
        h[mid1:mid2] = 0.25
        h[mid2:] = 0.5
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def swap_regions(h, i1, i2, size):
        h_new = h.copy()
        # Swap segments
        h_new[i1:i1+size], h_new[i2:i2+size] = h_new[i2:i2+size], h_new[i1:i1+size]
        
        # Rescale to maintain sum constraint
        current_sum = np.sum(h_new)
        h_new = h_new * (n_points / 2) / current_sum
        return h_new

    best_c5 = compute_c5(h)
    best_h = h.copy()
    current_h = h.copy()
    
    # Swap-based local search
    swapped = 0
    temperature = 0.1
    
    while swapped < max_swaps and time.time() - start_time < budget_s - 1:
        # Randomly select two non-overlapping segments to swap
        size = 5
        i1 = np.random.randint(0, n_points - size)
        i2 = np.random.randint(0, n_points - size)
        
        # Ensure segments don't overlap
        while abs(i1 - i2) < size:
            i2 = np.random.randint(0, n_points - size)
        
        # Create new solution by swapping
        h_new = swap_regions(current_h, i1, i2, size)
        
        # Evaluate new solution
        new_c5 = compute_c5(h_new)
        
        # Accept with probability based on temperature
        if new_c5 < best_c5 or np.random.rand() < np.exp(-(new_c5 - best_c5) / temperature):
            current_h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        
        swapped += 1
        
        # Gradually decrease temperature
        temperature = max(0.001, temperature * 0.999)
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break

    # Final polishing with SLSQP
    from scipy.optimize import minimize
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        result = minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            best_c5 = compute_c5(best_h)
    except:
        pass  # Ignore error if local search fails
    
    return (best_h.tolist(), best_c5, n_points)
