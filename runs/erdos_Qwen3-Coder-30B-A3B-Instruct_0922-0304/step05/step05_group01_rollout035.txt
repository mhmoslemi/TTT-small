import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: piecewise constant function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Binary search for optimal C5
    c5_low = 0.0
    c5_high = 0.38085  # Current upper bound
    tolerance = 1e-5
    best_c5 = c5_high
    best_h = h.copy()

    max_iter = 50
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5_mid = (c5_low + c5_high) / 2.0
        
        # Define bounds for optimization
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        # Constraint that all overlaps are <= c5_mid * dx
        def max_overlap_constraint(h):
            overlap = correlate(h, 1 - h, mode="full")
            return c5_mid * dx - np.max(overlap) * dx
        
        constraints.append({'type': 'ineq', 'fun': max_overlap_constraint})
        
        # Optimization to see if we can satisfy this C5
        try:
            result = minimize(
                lambda x: -np.max(correlate(x, 1 - x, mode="full")) * dx,  # We want to minimize the maximum overlap
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            if result.success and max_overlap_constraint(result.x) >= 0:
                # Feasible solution found
                c5_high = c5_mid
                best_c5 = c5_mid
                best_h = result.x.copy()
                # Update h for next iteration
                h = result.x.copy()
            else:
                # Infeasible
                c5_low = c5_mid
        except Exception:
            c5_low = c5_mid
            
        if abs(c5_low - c5_high) < tolerance:
            break

    # Final validation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
