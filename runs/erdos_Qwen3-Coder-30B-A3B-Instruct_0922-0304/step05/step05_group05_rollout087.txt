import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create CVXPY optimization problem
        variables = cp.Variable(n_points)
        constraints = [
            variables >= 0,
            variables <= 1,
            cp.sum(variables) == n_points / 2
        ]
        
        # Approximate the max overlap as a linear constraint
        # For this optimization, we take a convex relaxation approach
        # Minimize:\max_k (h * (1-h))[k]  
        # We approximate this by minimizing max of the elements of the cross-correlation
        # Note: This is not exact but is convex and leads to better progress
        
        # Create the cross-correlation approximation as a linear constraint
        # Rather than trying to minimize the true max, we use a barrier approach
        
        # Instead of full max, we use quadratic over linear relaxation
        # Simplified: just solve a convex problem that pushes towards better 
        # solutions, while respecting the bounds and sum constraints
        
        objective = cp.Minimize(cp.sum(cp.multiply(variables, 1 - variables)))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == 'optimal':
                new_h = variables.value
                # Re-project to ensure feasibility
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                
                current_c5 = compute_c5(new_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = new_h.copy()
                
                # Early exit if target achieved
                if best_c5 <= 0.3807:
                    break
                    
                # Update current h
                h = new_h
            else:
                break  # Problem is infeasible or unbounded
        except:
            # fallback to simple gradient-based step if CVXPY fails
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            # Try ODE-like gradient step
            # Approximate gradient through finite differences
            eps = 1e-4
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = compute_c5(h_plus)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                c5_minus = compute_c5(h_minus)
                
                grad[i] = (c5_plus - c5_minus) / (2*eps)
            
            # Gradient descent step
            step_size = 0.01
            h = h - step_size * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
