import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
    
    # Add more random particles
    for _ in range(swarm_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
    
    # Track best positions
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [float('inf')] * swarm_size
    global_best = particles[0].copy()
    global_best_score = float('inf')
    
    # PSO parameters
    omega_start = 0.9
    omega_end = 0.4
    phi1 = 2.0
    phi2 = 2.0
    velocity_limit = 0.1
    
    def compute_c5(h):
        # Compute the overlap integral
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Main PSO loop
    for iteration in range(max_iter):
        current_time = time.time()
        if current_time - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        omega = omega_start - (omega_start - omega_end) * (current_time - start_time) / budget_s
        
        for i in range(swarm_size):
            # Compute current score
            score = compute_c5(particles[i])
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = particles[i].copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best = particles[i].copy()
            
            # Early exit if target achieved
            if global_best_score <= 0.3807:
                break
                
            # Update velocity
            r1, r2 = np.random.random(2)
            new_velocities = []
            for j in range(n_points):
                cognITIVE = phi1 * r1 * (personal_best[i][j] - particles[i][j])
                social = phi2 * r2 * (global_best[j] - particles[i][j])
                new_velocities.append(omega * velocities[i][j] + cognITIVE + social)
            
            # Apply velocity limits
            new_velocities = np.clip(new_velocities, -velocity_limit, velocity_limit)
            velocities[i] = np.array(new_velocities)
            
            # Update position
            new_positions = particles[i] + velocities[i]
            
            # Clamp to [0,1]
            new_positions = np.clip(new_positions, 0, 1)
            
            # Renormalize to sum to n_points / 2
            new_positions = new_positions * (n_points / 2) / np.sum(new_positions)
            
            particles[i] = new_positions
        
    # Final evaluation
    best_c5 = compute_c5(global_best)
    
    return (global_best.tolist(), best_c5, n_points)
