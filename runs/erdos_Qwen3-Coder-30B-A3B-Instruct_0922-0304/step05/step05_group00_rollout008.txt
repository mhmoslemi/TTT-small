import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    best_h = h.copy()
    best_c5 = np.max(correlate(h, 1 - h, mode="full")) * dx
    
    # Early exit if target achieved
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
        
    iteration = 0
    max_iter = 2000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Identify top lags contributing to C5
        overlap = correlate(h, 1 - h, mode="full")
        top_lags = np.argsort(overlap)[-max(10, n_points//10):][::-1]
        
        # Build LP model
        h_var = cp.Variable(n_points)
        C5 = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target_sum,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add linearized overlap constraints for top lags
        for k in top_lags:
            if k < n_points:
                # For lag k, overlap_k = sum(h[i] * (1 - h[i+k]) for i in [0, n_points-k-1])
                # Linearization around current h: 
                # New expr: sum(h[i]*(1-h[i+k]) + (1-2*h[i])*(h[i+k] - h[i+k]))
                # Simplified: sum(h[i] - h[i]*h[i+k] - h[i+k] + h[i]*h[i+k]) = sum(h[i] - h[i+k])
                # Use at current h proper linearization:
                # Overlap_k ≈ sum(h[i]*(1-h[i+k]) + (1-h[i])*(h[i+k] - h[i*k])) = 
                # = sum(h[i] - h[i]*h[i+k] + h[i+k] - h[i]*h[i+k] - h[i]*h[i+k])
                # Too complex, use simpler split:
                # h[i]*(1-h[i+k]) term linearized in-paper: h[i] + (1-2*h[i]) * h[i+k] - h[i+k]^2
                # But e.g., we can use equality vs. constraint explicitly.
                # Or use Schwarz inequality-based approach.
                # Do direct computation of linearized term for now.
                # Linearized coefficient for h[i] term of overlap_k:
                # = 1 - h[i+k] 
                # For h[i+k] term: = - h[i]
                # i = 0..n_points-k-1
                expr = 0
                for i in range(n_points - k):
                    expr += h_var[i] * (1 - h[i+k]) + (1 - 2 * h[i]) * h[i+k] - h[i+k]**2
                constraints.append(expr * dx <= C5)
            else:
                # Lag exceeds length, sum over reversed subset
                expr = 0
                lag = k - n_points + 1
                for i in range(n_points - lag):
                    expr += h_var[i] * (1 - h[i+lag])  
                constraints.append(expr * dx <= C5)
        
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (target_sum / np.sum(h_new))
                
                new_c5 = np.max(correlate(h_new, 1 - h_new, mode="full")) * dx
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
                    
                if best_c5 <= 0.3807:
                    return (best_h.tolist(), best_c5, n_points)
                
                h = h_new
            else:
                # Fallback to simple local search
                h = h + np.random.normal(0, 0.01, n_points)
                h = np.clip(h, 0, 1)
                h = h * (target_sum / np.sum(h))
        except:
            # Fallback to simple local search
            h = h + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))

        iteration += 1
    
    # Final evaluation
    final_c5 = np.max(correlate(best_h, 1 - best_h, mode="full")) * dx
    return (best_h.tolist(), final_c5, n_points)
