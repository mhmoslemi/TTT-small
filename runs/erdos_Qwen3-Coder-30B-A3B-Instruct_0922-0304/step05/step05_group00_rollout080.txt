    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

                # In consequence just treat linear part directly in 1D:
                # And plot; this is a discrete sum of a quadratic, with linear correction.
                # Since we don't know whether both terms can be emphasized or not, we build a convex relaxation.

                # Use some stabilization:
                # Full path: We define overlap_k constraint as:
                #   overlap_k = sum i from 0 to N-1-k: h[i] - h[i]*h[i+k] + h[i+k]*h[i] - h[i+k]^2

                # h[i+k]*h[i] - h[i]*h[i+k] = 0

                # No: f(i) = h[i] - h[i+k]^2 = h[i] - h[i+k]^2  This leads to the approach.  Let’s be careful.

                # Take the coarse view: approximate

                # = (N-1-k)*h_mean - sum_{j=k}^{N-1} h[j]^2

                # The most efficient path for constraints is to think of these in sparsity form.
                # With CVXPY runEquation:
                # Let’s backoff, just define it as a conventional bring-in-block 
                # We solve the general case.
                
                # What's a robust CVXPY approach:
                
                # Way to go, compute total at h:
                # Directionally, let us numerically linearize linear parts directly over subset of terms (blocks).
                # For clarity, we start simple by looping over all lags and compute:
                # The sum h[i] (1 - h[i+k]) + h[i+k] (h[i] - h[i+k])  
                # Perform linearization via first order Taylor around h:
                # Term to linearize = h[i] (1 - h[i+k]) + h[i+k] (h[i] - h[i+k]) 
                #  = h[i] - h[i]*h[i+k] + h[i+k]*h[i] - h[i+k]^2  
                # = h[i] - h[i+k]^2  (in the contradiction of simplification)
                # Fact check. 
                # No.  
                # Try a different nomal method.
                # Define a quadratic view model just around h point.
                # Let’s intepret overlap_k at h as given:
                # ⇒ h[i] - h[i]*h[i+k] + h[i+k]*h[i] - h[i+k]^2  
                # The gradient of the element is:
                # d/d_h[i] = 1 - h[i+k]  
                # d/d_h[i+k] = -h[i] + h[i]  = 0  
                # 
                # So that doesn’t go anywhere.
                # What if we still write:
                # Contribution for i:
                # 1. h[i] - (h[i]*h[i+k]) + 
                # 2. h[i+k] * h[i] − h[i+k]^2  
                # From 1 element:  
                # Far term = h[i] hence: 1 = gradient wrt h[i]
                # h[i]*h[i+k] ≈ h[i] h0[i+k] + h0[i] h[i+k],  
                # So derivative = h0[i+k]  
                # From 2 element: 
                # h[i+k] h[i] ≈ h0[i+k] h[i] + h[i+k] h0[i]  
                # Simplest it’s h0[i]  
                # So deriv w.r.t h[i] from expr 2 = h0[i+k]  
                # ✔ derivatives of second element wrt h[i] = h0[i+k]
                # From 2 derivative wrt h[i+k] = h[i]
                # ↑ prefix mismatch from index-wise.
                # In summarizing: x = h[i] (1 - h[i+k]) + h[i+k] (h[i] - h[i+k]) for a fixed element. 
                # = h[i] - h[i]*h[i+k] + h[i+k]*h[i] - h[i+k]^2 
                # = h[i] - (h[i]*h[i+k]) + (h[i+k]*h[i]) - h[i+k]^2
                # ∂x/∂h[i] = 1 - h0[i+k] + h0[i+k] = 1
                # ∂x/∂h[i+k] = - h0[i] + h0[i] - 2 h0[i+k] = - 2 h0[i+k]  
                # So in our primal constraint we’ll absorb the point derivatives 
                # For our all-bounds defined Lag non-linear function:
                # Linear version for given h (as in usage for entire k total):
                #  ∑ h[i] (1 - h[i+k]) + h[i+k] (h[i] - h[i+k])  
                # = ∑ of h[i] + h[i+k]*h[i] - h[i]*h[i+k] - h[i+k]^2  
                # ∑ { h[i] - h[i]*h[i+k] + h[i+k]*h[i] - h[i+k]^2 }
                # We make a linear function for that that is valid over the h space.
                # At h0 = current solution, it's value:
                # f(x) on h  ≈ value_of_f_at_h0 + sum_i 1*(h[i] - h0[i]) + sum_j -2*h0[j]*(h[j] - h0[j])

                # Now we take a mostly uniform, very clean approach in our LP:

                # For every k ∈ lag_indices:
                #  \sum_{i=0}^{N-1-k} [ (1 - h0[i+k]) * h[i] + (-2 h0[i+k]) * h[i+k] ] ≤ C₅

                #    h[i]    h[i+k]  
                #     1     -2 * h0[i+k]

                # Now, finally, proceed to build CVXPY model for current step:
                # i: from 0 to N-1-k, the valid indices for it.

                    # Exactly iterate over valid i indices.
                    coeff_i = (1 - h[iidx + k])
                    coeff_k = -2 * h[iidx + k] 
                    constraints.append(coeff_i * h_var[iidx] + coeff_k * h_var[iidx + k] <= c5)

            # Solve the LP relaxation for minimal C5 value
            objective = cp.Minimize(c5)
            prob = cp.Problem(objective, constraints)
            try:
                prob.solve(solver=cp.SCS)
                if prob.status == cp.OPTIMAL:
                    current_c5 = c5.value
                    # Check if improvement
                    if current_c5 < best_c5 - 1e-6:
                        best_c5 = current_c5
                        best_h = h_var.value
                        # Project onto feasible set if needed
                        projected_h = np.clip(h_var.value, 0, 1)
                        projected_h = projected_h * (target_sum / np.sum(projected_h))
                        if np.abs(np.sum(projected_h) - target_sum) < 1e-6:
                            best_h = projected_h.copy()
                else:
                    # Adjust h slightly for robustness
                    h = h + np.random.normal(0, 0.01)
                    h = np.clip(h, 0, 1)
                    h = h * (target_sum / np.sum(h))
            except Exception as e:
                # On failure, step back to previous best or use small perturbation
                h = h + np.random.normal(0, 0.01)
                h = np.clip(h, 0, 1)
                h = h * (target_sum / np.sum(h))
        iteration += 1
        
        # Refinement step
        # Simple descent optimization of top few lags
        if iteration % 5 == 0:
            h = best_h
            
    return (best_h.tolist(), best_c5, n_points)
