import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

    best_c5 = float('inf')

    max_iter = 2000
    step_size = 0.01

            h_plus = normalize_h(h_plus, target_sum)
            c5_plus = evaluate_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = normalize_h(h_minus, target_sum)
            c5_minus = evaluate_c5(h_minus, dx)

    final_c5 = evaluate_c5(best_h, dx)
