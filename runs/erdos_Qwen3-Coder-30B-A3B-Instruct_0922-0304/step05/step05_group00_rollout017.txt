import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def repair_feasibility(h, target_sum):
    """Repair h to be feasible: clip to [0,1] and normalize."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with initial_h_values or with a simple construction
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
    else:
        # Simple step function
        h_base = np.ones(n_points)
        h_base[n_points//2:] = 0
    
    h_base = repair_feasibility(h_base, target_sum)
    
    # Start with the base solution
    best_h = h_base.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Population size
    population_size = 30
    population = [best_h.copy()]
    
    # Generate initial population
    for _ in range(population_size - 1):
        h_new = best_h + np.random.normal(0, 0.05, n_points)
        h_new = repair_feasibility(h_new, target_sum)
        population.append(h_new)
    
    # Step size for differential evolution
    step_size = 0.5
    step_decay = 0.98
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 500:
        # Evaluate all candidates
        fitness = [compute_c5(h, dx) for h in population]
        
        # Update best solution
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_h = population[min_fit_idx].copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Create offspring using different methods
        new_population = []
        
        # Keep best solution (elitism)
        new_population.append(best_h.copy())
        
        # Differential evolution
        for _ in range(12):  # ~40% of population
            if len(population) >= 3:
                a, b, c = np.random.choice(len(population), 3, replace=False)
                mutant = population[a] + step_size * (population[b] - population[c])
                # Crossover
                trial = np.where(np.random.rand(n_points) < 0.9, mutant, population[np.random.randint(len(population))])
                trial = repair_feasibility(trial, target_sum)
                new_population.append(trial)
        
        # Local search using gradient descent
        for _ in range(8):  # ~30% of population
            idx = np.random.randint(len(population))
            h = population[idx].copy()
            
            # Constraint: sum to target_sum
            def constraint_sum(h):
                return np.sum(h) - target_sum
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h):
                return compute_c5(h, dx)
            
            try:
                result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 20})
                if result.success:
                    h_refined = result.x
                    h_refined = repair_feasibility(h_refined, target_sum)
                    new_population.append(h_refined)
            except:
                pass
        
        # Random sampling
        for _ in range(4):  # ~20% of population
            h_random = np.random.rand(n_points)
            h_random = repair_feasibility(h_random, target_sum)
            new_population.append(h_random)
        
        # Mutation of best solution
        for _ in range(2):  # ~10% of population
            h_mut = best_h + np.random.normal(0, 0.02, n_points)
            h_mut = repair_feasibility(h_mut, target_sum)
            new_population.append(h_mut)
        
        # Add more diversity if needed
        while len(new_population) < population_size:
            h_new = np.random.rand(n_points)
            h_new = repair_feasibility(h_new, target_sum)
            new_population.append(h_new)
        
        # Keep only the best
        combined = list(zip(population, fitness)) + list(zip(new_population, [compute_c5(h, dx) for h in new_population]))
        combined.sort(key=lambda x: x[1])
        population = [x[0] for x in combined[:population_size]]
        
        # Decrease step size
        step_size *= step_decay
        
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
