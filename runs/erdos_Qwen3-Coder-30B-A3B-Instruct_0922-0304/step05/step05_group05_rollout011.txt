        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Convex optimization approach
    for iteration in range(500):  # Limit iterations to stay within budget
        if time.time() - start_time > budget_s - 1:

        try:

            variables = cp.Variable(n_points)
            
            # Constraints
            constraints = [
                variables >= 0,
                variables <= 1,

            ]

            current_overlap = correlate(h, 1 - h, mode="full")
            max_overlap_val = np.max(current_overlap)
            
            # Constraint that overlap shouldn't exceed current value
            # This is a simplification - we're not truly solving convex relaxation properly

            old_h = h.copy()
            old_c5 = compute_c5(old_h)
