import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_coarse = 35
    
    # Function to compute C5
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Constraint function
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Coarse optimization
    h_coarse = h[::2] + h[1::2]  # Downsample by averaging pairs
    h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    
    # Coarse bounds and constraints
    coarse_bounds = [(0, 1) for _ in range(n_coarse)]
    coarse_constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_coarse / 2}]  
    
    # Optimize coarse
    coarse_result = minimize(
        lambda x: compute_c5(x), 
        h_coarse, 
        method='SLSQP', 
        bounds=coarse_bounds, 
        constraints=coarse_constraints,
        options={'maxiter': 50, 'ftol': 1e-6}
    )
    
    if coarse_result.success:
        h_coarse = coarse_result.x
        h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    
    # Upsample to fine scale
    h_upsampled = np.repeat(h_coarse, 2)
    h_upsampled = h_upsampled * (n_points / 2) / np.sum(h_upsampled)
    
    # Fine optimization with better initialization
    h = h_upsampled.copy()
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 200
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Optimize
        result = minimize(
            compute_c5, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 10, 'ftol': 1e-8}
        )
        
        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            # Perturbation if optimization fails
            h += np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
