import numpy as np
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 20
    max_iter = 1000
    
    # Initialize swarm
    def initialize_particle():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
        else:
            h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Initialize particles
    particles = [initialize_particle() for _ in range(swarm_size)]
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(swarm_size)]
    
    # Track personal bests
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [float('inf')] * swarm_size
    
    # Track global best
    global_best = particles[0].copy()
    global_best_score = float('inf')
    
    # PSO parameters
    w = 0.7
    c1 = c2 = 1.5
    max_velocity = 0.2
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(swarm_size):
            # Evaluate current particle
            current_score = compute_c5(particles[i])
            
            # Update personal best
            if current_score < personal_best_scores[i]:
                personal_best_scores[i] = current_score
                personal_best[i] = particles[i].copy()
            
            # Update global best
            if current_score < global_best_score:
                global_best_score = current_score
                global_best = particles[i].copy()
            
            # Update velocity
            r1 = np.random.random(n_points)
            r2 = np.random.random(n_points)
            
            vel_cognitive = c1 * r1 * (personal_best[i] - particles[i])
            vel_social = c2 * r2 * (global_best - particles[i])
            
            velocities[i] = w * velocities[i] + vel_cognitive + vel_social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            
            # Ensure constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        
        # Update inertia weight
        w = max(0.4, w - 0.001)
    
    # Final evaluation
    final_c5 = compute_c5(global_best)
    
    return (global_best.tolist(), final_c5, n_points)
