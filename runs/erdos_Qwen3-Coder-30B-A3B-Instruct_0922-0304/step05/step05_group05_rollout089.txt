import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Convex optimization approach to find improved solution
    best_c5 = float('inf')
    best_h = h.copy()
    best_iter = -1
    
    # Use CVXPY with convex relaxation
    for iteration in range(1000):  # Limit iterations
        if time.time() - start_time > budget_s - 1:
            break
            
        # Formulate convex problem
        variables = cp.Variable(n_points)
        constraints = [variables >= 0, variables <= 1, cp.sum(variables) == n_points / 2]
        
        # Compute current h for linearization
        h_current = h.copy()
        # Create a quadratic approx of the objective by linearizing around best solution so far
        overlap = correlate(h_current, 1 - h_current, mode="full")
        max_overlap = np.max(overlap)
        
        # Use relaxed objective to avoid non-convexity
        objective = cp.Minimize(max_overlap)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            # Fallback to gradient-based optimization if convex solver fails
            pass

        if prob.status == cp.OPTIMAL:
            new_h = variables.value
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            c5 = compute_c5(new_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = new_h.copy()
                best_iter = iteration
        else:
            # Fallback to gradient-free optimization
            new_h = h_current + np.random.normal(0, 0.01, n_points)
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            c5 = compute_c5(new_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = new_h.copy()
                best_iter = iteration
        
        # Early exit condition
        if best_c5 <= 0.3807:
            break
        
        # Annealing step: perturb the current solution gently
        h = best_h.copy()
        h = h + np.random.normal(0, 0.005, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    # Ensure final solution meets constraints
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
