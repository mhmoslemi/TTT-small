import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    generations = 1000
    mutation_rate = 0.1
    tournament_size = 10
    
    # Initialize population
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Fill remaining population with random individuals
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h.copy())
    
    best_h = None
    best_c5 = float('inf')
    
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection with tournament
        new_population = []
        
        # Keep top individuals
        sorted_indices = np.argsort(fitness)
        for i in range(population_size // 2):
            new_population.append(population[sorted_indices[i]].copy())
            
        # Generate rest through crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            parent1_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            parent2_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Uniform crossover
            child = np.where(np.random.rand(n_points) < 0.5, parent1, parent2)
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    # Perturb randomly or set to boundary
                    if np.random.rand() < 0.5:
                        child[i] += np.random.normal(0, 0.05)
                    else:
                        child[i] = np.random.rand()
                        
            # Keep within bounds and normalize
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
            
        population = new_population[:population_size]
    
    # Final evaluation of best solution
    final_c5 = evaluate_h(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
