import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def repair_feasibility(h, target_sum):
    """Repair h to ensure it's feasible."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def evaluate_individual(h, dx):
    """Evaluate objective and check constraints."""
    c5 = compute_c5(h, dx)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Differential evolution parameters
    population_size = 30
    mutation_factor = 0.8
    crossover_rate = 0.7
    step_size = 0.05
    
    # Population initialization
    population = [repair_feasibility(h + np.random.normal(0, 0.02, n_points), target_sum) 
                  for _ in range(population_size)]
    
    best_c5 = float('inf')
    best_individual = h.copy()
    
    iteration = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate population
        fitness = [evaluate_individual(ind, dx) for ind in population]
        
        # Update best solution
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_individual = population[min_fit_idx].copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Create new population using differential evolution
        new_population = []
        
        # Keep best individual
        new_population.append(best_individual.copy())
        
        # Generate offspring using DE
        for i in range(1, population_size):
            # Select three different individuals
            candidates = list(range(population_size))
            candidates.remove(i)
            a, b, c = np.random.choice(candidates, 3, replace=False)
            
            # Mutation
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            
            # Crossover
            crossover_mask = np.random.rand(n_points) < crossover_rate
            if not np.any(crossover_mask):
                crossover_mask[np.random.randint(n_points)] = True
            trial = np.where(crossover_mask, mutant, population[i])
            
            # Repair
            trial = repair_feasibility(trial, target_sum)
            
            # Selection
            trial_fitness = evaluate_individual(trial, dx)
            old_fitness = fitness[i]
            
            if trial_fitness < old_fitness:
                new_population.append(trial)
            else:
                new_population.append(population[i])
        
        # Add some gradient descent to refine top individuals
        top_indices = np.argsort(fitness)[:5]
        for idx in top_indices:
            # Simple local search: gradient descent with constraints
            h_local = population[idx].copy()
            for _ in range(20):
                # Compute gradient approximation using finite differences
                grad = np.zeros(n_points)
                eps = 1e-4
                base_c5 = evaluate_individual(h_local, dx)
                
                for j in range(n_points):
                    h_plus = h_local.copy()
                    h_plus[j] += eps
                    h_plus = repair_feasibility(h_plus, target_sum)
                    c5_plus = evaluate_individual(h_plus, dx)
                    grad[j] = (c5_plus - base_c5) / eps
                
                # Update with gradient
                h_local -= step_size * grad
                h_local = repair_feasibility(h_local, target_sum)
            
            new_population.append(h_local)
        
        # Fill up population
        while len(new_population) < population_size:
            noise = np.random.normal(0, 0.02, n_points)
            mut = repair_feasibility(h + noise, target_sum)
            new_population.append(mut)
            
        population = new_population[:population_size]
        
        # Reduce step size with time
        step_size *= 0.995
        iteration += 1
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
