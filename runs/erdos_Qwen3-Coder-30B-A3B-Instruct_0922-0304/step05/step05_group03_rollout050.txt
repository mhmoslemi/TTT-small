import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_halton_sequence(n_points, n_dim, seed=42):
    """Generate Halton sequence for diversified sampling."""
    np.random.seed(seed)
    # Simple Halton-like sequence for 1D
    result = np.zeros((n_points, n_dim))
    for i in range(n_points):
        for j in range(n_dim):
            result[i, j] = (i+1) / (j+2) % 1
    return result

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start optimization with Halton sequences
    n_starts = 50
    halton_seq = generate_halton_sequence(n_starts, n_points, seed)
    
    # Run optimization for each Halton sample
    for i in range(n_starts):
        if time.time() - start_time >= budget_s - 1:
            break
            
        h_start = normalize_h(halton_seq[i] * 2, target_sum)
        
        def objective(h):
            return evaluate_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        # SLSQP optimization
        result = minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        
        if result.success:
            h_opt = normalize_h(result.x, target_sum)
            c5_opt = evaluate_c5(h_opt, dx)
            
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                
        # Early exit
        if best_c5 <= 0.3807:
            break
            
    # Final refinement with more iterations
    if time.time() - start_time < budget_s - 1:
        def objective(h):
            return evaluate_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 200, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        
        if result.success:
            h_opt = normalize_h(result.x, target_sum)
            c5_opt = evaluate_c5(h_opt, dx)
            
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    
    return (best_h.tolist(), best_c5, n_points)
