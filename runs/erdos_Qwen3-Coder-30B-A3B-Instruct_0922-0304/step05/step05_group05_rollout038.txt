import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Stochastic gradient descent with adaptive step size and constraint projection
    max_iter = 3000
    lr = 0.01
    momentum = 0.9
    v = np.zeros_like(h)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient approximation
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
        
        # Sample a few directions for gradient estimation
        grad_approx = np.zeros_like(h)
        eps = 1e-3
        for _ in range(5):
            pert = np.random.normal(0, eps, n_points)
            pert = np.clip(pert, -0.1, 0.1)
            h_plus = np.clip(h + pert, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = np.clip(h - pert, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            c5_plus = compute_c5(h_plus)
            c5_minus = compute_c5(h_minus)
            grad_approx += (c5_plus - c5_minus) / (2 * eps) * pert
            
        # Update using momentum
        v = momentum * v - lr * grad_approx
        h = h + v
        
        # Project to feasible set
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
