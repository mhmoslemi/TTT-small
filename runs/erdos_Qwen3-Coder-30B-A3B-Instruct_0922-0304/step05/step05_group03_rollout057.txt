import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient of overlap at k_max with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    
    # Compute gradient for the overlap at k_max
    # overlap[k] = sum_{i=0}^{n-1} h[i] * (1 - h[i+k]) for -n+1 <= k <= n-1
    # For k = k_max, gradient w.r.t. h[i] is -(1 - h[i+k_max]) when i+k_max is valid
    
    for i in range(n):
        idx = i + k_max
        if 0 <= idx < n:
            grad[i] = -(1 - h[idx])
    
    return grad * dx

    # Gradient descent parameters
    alpha = 0.1
    max_iter = int(budget_s * 500)  # Rough estimate of iterations
    patience = 0
    patience_limit = 50
    improvement_threshold = 1e-6
    
    best_c5 = float('inf')
    best_h = h.copy()
    current_c5 = evaluate_c5(h, dx)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:

        k_max = np.argmax(overlap)

        # Update h
        h_new = h - alpha * grad
        
        # Ensure h_new stays within bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize new h
        h_new = normalize_h(h_new, target_sum)
        
        # Compute new C5
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept the update if it improves C5
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
