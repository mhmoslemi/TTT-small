import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: binary with exactly n_points/2 ones
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        np.random.shuffle(h)
    
    best_c5 = float('inf')
    best_h = h.copy()

    def compute_c5(h):
        # Compute correlation for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Binary search with simulated annealing for initial configuration
    max_iter = 10000
    temperature = 1.0
    cooling_rate = 0.995
    iter_count = 0

    while time.time() - start_time < budget_s * 0.6 and iter_count < max_iter:
        iter_count += 1
        
        # Create neighbor by swapping two random elements
        h_new = h.copy()
        i, j = np.random.choice(n_points, 2, replace=False)
        h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # Ensure it's binary and sum is correct
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute overlap for new h
        c5_new = compute_c5(h_new)
        c5_old = compute_c5(h)
        
        # Accept or reject based on Metropolis criterion
        if c5_new < c5_old or np.random.rand() < np.exp(-(c5_new - c5_old) / temperature):
            h = h_new
        
        # Update best if improved
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
        # Cooling
        temperature *= cooling_rate
        
        # Early exit
        if best_c5 <= 0.3807:
            break

    # Continuous optimization using CVXPY
    if time.time() - start_time < budget_s * 0.9:
        # Refine the continuous function using CVXPY
        try:
            h_var = cp.Variable(n_points)
            
            # Objective function - we minimize a placeholder
            obj = cp.Minimize(0)
            
            # Constraints
            constraints = [
                cp.sum(h_var) == n_points / 2,
                h_var >= 0,
                h_var <= 1
            ]
            
            # Create a temporary problem to check feasibility
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                refined_h = h_var.value
                
                # Refine further with gradient-based method
                def get_overlap(h_val):
                    return np.max(correlate(h_val, 1 - h_val, mode="full")) * dx
                
                # Gradient descent with projection
                lr = 0.01
                max_steps = 1000
                current_h = refined_h.copy()
                current_c5 = get_overlap(current_h)
                
                for _ in range(max_steps):
                    if time.time() - start_time > budget_s * 0.95:
                        break
                        
                    # Gradient estimation (finite difference)
                    grad = np.zeros(n_points)
                    eps_val = 1e-4
                    for i in range(n_points):
                        h_plus = current_h.copy()
                        h_minus = current_h.copy()
                        h_plus[i] += eps_val
                        h_minus[i] -= eps_val
                        h_plus = np.clip(h_plus, 0, 1)
                        h_minus = np.clip(h_minus, 0, 1)
                        # Renormalize
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                        
                        c5_plus = get_overlap(h_plus)
                        c5_minus = get_overlap(h_minus)
                        grad[i] = (c5_plus - c5_minus) / (2 * eps_val)
                    
                    # Update step
                    new_h = current_h - lr * grad
                    # Project back to feasible set
                    new_h = np.clip(new_h, 0, 1)
                    new_h = new_h * (n_points / 2) / np.sum(new_h)
                    
                    new_c5 = get_overlap(new_h)
                    
                    if new_c5 < current_c5:
                        current_h = new_h
                        current_c5 = new_c5
                    else:
                        lr *= 0.99  # Reduce step size
            
                if current_c5 < best_c5:
                    best_h = current_h
                    best_c5 = current_c5
        except:
            pass
    
    # Final computation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
