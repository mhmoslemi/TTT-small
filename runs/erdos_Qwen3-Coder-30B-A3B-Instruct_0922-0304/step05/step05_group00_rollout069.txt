import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def repair_feasibility(h, target_sum):
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Differential evolution parameters
    population_size = 30
    population = []
    for _ in range(population_size):
        candidate = h + np.random.normal(0, 0.05, n_points)
        candidate = repair_feasibility(candidate, target_sum)
        population.append(candidate)
    
    # Steps for diversity
    step_size = 0.05
    step_decay = 0.98
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 1000:
        # Evaluate
        fitness = [compute_c5(ind, dx) for ind in population]
        
        # Update best
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_h = population[min_fit_idx].copy()
            
        if best_c5 <= 0.3807:
            break
        
        # Selection
        selected_indices = []
        for _ in range(population_size):
            tournament = np.random.choice(len(population), 3)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected_indices.append(winner)
        
        # Generate new population
        new_population = []
        
        # Elitism
        new_population.append(best_h.copy())
        
        # Differential evolution
        for _ in range(population_size - 1):
            a, b, c = np.random.choice(len(population), 3, replace=False)
            mutant = population[a] + 0.8 * (population[b] - population[c])
            mutant = repair_feasibility(mutant, target_sum)
            
            # Crossover
            crossover_mask = np.random.rand(n_points) < 0.7
            if not np.any(crossover_mask):
                crossover_mask[np.random.randint(n_points)] = True
            trial = np.where(crossover_mask, mutant, population[selected_indices[-1]])
            
            new_population.append(trial)
        
        # Local optimization for some individuals
        for i in range(0, len(new_population), 3):
            if i < len(new_population):
                h_local = new_population[i].copy()
                
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                def objective(h):
                    return compute_c5(h, dx)
                
                try:
                    result = minimize(objective, h_local, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 30})
                    if result.success:
                        h_local = result.x
                        h_local = np.clip(h_local, 0, 1)
                        h_local = normalize_h(h_local, target_sum)
                        new_population[i] = h_local
                except:
                    pass
        
        # Repair all to ensure feasibility
        for i in range(len(new_population)):
            new_population[i] = repair_feasibility(new_population[i], target_sum)
        
        # Update population
        population = new_population
        
        # Adjust step size
        if iteration % 5 == 0:
            step_size *= step_decay
            
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
