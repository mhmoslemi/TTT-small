import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        h = np.ones(n_points) * 0.5
        h = project_h(h, target_sum)

    # Parameters
    eta = 0.1
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute all cross-correlations
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        # Compute gradient - it's the shifted version of (1-h) at the max_k
        # This is a simplified version of gradient computation
        # For k=0, we want the derivative of (h * (1-h))(0)
        # But since we take the max over all k, we update based on max_k
        
        # The update direction is along the cross-correlation at the maximizing shift
        k_offset = max_k - (n_points - 1)
        if k_offset >= 0:
            grad = np.concatenate([np.zeros(k_offset), 1 - h[:-k_offset]])
        else:
            grad = np.concatenate([1 - h[-k_offset:], np.zeros(-k_offset)])
            
        # Apply gradient
        h = h - eta * grad * dx
        
        # Project back to feasible set
        h = project_h(h, target_sum)
        
        # Periodically perturb with noise to escape local optima
        if iteration % 100 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h = h + noise
            h = project_h(h, target_sum)
        
        # Update best solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Adjust learning rate
        if iteration % 10 == 0:
            eta = eta * 1.01 if current_c5 < best_c5 + 1e-10 else eta * 0.9
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
