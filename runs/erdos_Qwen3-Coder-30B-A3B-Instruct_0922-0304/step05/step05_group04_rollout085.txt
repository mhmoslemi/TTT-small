import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Annealing-based optimization
    iteration = 0
    temp = 0.1
    cooling_rate = 0.999
    
    # Large neighborhood search
    max_neighbors = 20
    neighbor_size = max(1, n_points // 20)
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        current_c5 = overlap[max_k] * dx
        
        # Store best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Create neighbor solution using local perturbations
        new_h = h.copy()
        
        # Try several random changes
        for _ in range(np.random.randint(1, max_neighbors)):
            # Select random indices to modify
            indices = np.random.choice(n_points, size=min(neighbor_size, n_points//2), replace=False)
            
            # Modify each selected index
            for i in indices:
                # Add small random perturbation
                perturbation = np.random.normal(0, 0.01)
                new_h[i] += perturbation
                
            # Clamp and normalize
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
        # Accept or reject based on Metropolis criterion
        new_overlap = correlate(new_h, 1 - new_h, mode="full")
        new_c5 = new_overlap[np.argmax(new_overlap)] * dx
        
        if new_c5 < current_c5:
            # Always accept improving changes
            h = new_h
        else:
            # Accept with probability based on temperature
            if np.random.uniform() < np.exp(-(new_c5 - current_c5) / temp):
                h = new_h
                
        # Decrease temperature
        temp *= cooling_rate
        
        # Occasionally perform large jumps
        if np.random.random() < 0.05:
            # Reset with a new random solution that satisfies constraints
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final check with evalution
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
