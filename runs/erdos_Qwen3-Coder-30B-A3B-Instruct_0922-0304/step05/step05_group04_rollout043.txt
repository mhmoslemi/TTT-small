import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    mutation_rate = 0.1
    tournament_size = 10
    max_generations = 1000
    
    # Initialize population
    def initialize_population():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population = [h.copy()]
        else:
            population = []
        
        # Fill rest with random individuals
        while len(population) < population_size:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
        
        return population
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def fitness(h):
        return compute_c5(h)

    def tournament_selection(population, fitnesses):
        selected = []
        for _ in range(2):
            tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
            selected.append(population[winner_index])
        return selected

    def uniform_crossover(parent1, parent2):
        mask = np.random.rand(n_points) < 0.5
        child = np.where(mask, parent1, parent2)
        return child

    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                if np.random.rand() < 0.5:
                    h[i] += np.random.normal(0, 0.1)
                else:
                    h[i] = np.random.rand()
        # Clamp and renormalize
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def evaluate_population(population):
        fitnesses = [fitness(h) for h in population]
        return fitnesses

    # Main algorithm
    population = initialize_population()
    best_h = None
    best_c5 = float('inf')
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        fitnesses = evaluate_population(population)
        
        # Track best individual
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Selection and breeding
        new_population = []
        
        # Keep top individuals
        top_indices = np.argsort(fitnesses)[:population_size//2]
        for idx in top_indices:
            new_population.append(population[idx])
        
        # Generate rest through crossover and mutation
        while len(new_population) < population_size:
            parents = tournament_selection(population, fitnesses)
            child = uniform_crossover(parents[0], parents[1])
            child = mutate(child)
            new_population.append(child)
        
        population = new_population

    # Final evaluation of best individual
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
