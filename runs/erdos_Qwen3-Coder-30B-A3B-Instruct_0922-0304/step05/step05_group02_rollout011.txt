```python
import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_sum(h, target_sum):
    """Project h to have sum equal to target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_sum(h, n_points / 2)
    else:
        # Create a starting point
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points) 
        h = np.clip(h, 0, 1)
        h = project_sum(h, n_points / 2)
    
    # Parameters
    momentum_decay = 0.9
    learning_rate = 0.1
    min_learning_rate = 1e-6
    max_iterations = 10000
    tolerance = 1e-6
    
    # For gradient computation
    h_prev = h.copy()
    momentum = np.zeros_like(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute C5
        current_c5 = compute_c5(h, dx)
        
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Compute gradient direction
        # We compute gradients for all relevant shifts
        # and focus on the worst shift (i.e. highest overlap)
        overlap = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(overlap)
        
        # These are the indices in the full cross-correlation
        # For a vector of length n, the full cross-correlation has length 2*n-1
        # Shifts go from -n+1 to n-1
        shift = max_shift_idx - (n_points - 1)
        # Get the derivative for the worst shift using finite difference
        # To get gradient for a shift k, we look at derivative of 
        # integral h(x) * (1 - h(x + k)) wrt h_i
        # This involves propagating derivative through convolution:
        # Differentiate with respect to h_i: Δ_i = 1_{i+k >= 0} * 1_{i+k < n} 
        # But we only need the indices for the relevant convolution pair.
        # Let's compute derivatives for just one top shift to save effort.
        # For shift k = i - j, gradient wrt h_i is - (1_{j + k < n} - 1_{j + k >= n})
        # But it is easier to just compute numerical gradient if needed.
        
        # Using approximate gradient computation:
        # We can compute the derivative of the integral involving h(x+k) and h(x)
        # using the definition of cross-correlation derivative.
        # Derivative for shift k of sum h(x)(1 - h(x+k)) wrt h[i]: 
        # -> - (1 - h[i+k]) if i+k is in valid range, and 0 otherwise
        # But we must account for that the derivative normalization and presume these
        # are discrete approximations.
        # So we'll simply set the gradient based on local influence
        
        # For each position i, our gradient should be:
        # sum over j of  (d/dh_i) h[j] * (1 - h[j + i - k]) 
        # = 0 if i != j, else - (1 - h[i + k]) 
        # But we consider only non-zero terms.
        # The actual derivative for a shift k is simply:
        # For each i, the derivative of h[i] (from the form h[i](1 - h[i+k]))
        # should be: - (1 - h[i + k]) if k is such that i + k is valid.
        
        # Simplified approach:
        # Compute gradient as - (1 - h[shift + i]), but perform only forward shifts
        # and backwards indices only for valid elements.
        gradient = np.zeros_like(h)
        
        # Compute the simple gradient for a single worst overlapping shift:
        if shift <= 0: # right shift (k positive)
            # h_i*(1 - h_{i+k})
            for i in range(len(h)):
                if i + shift >= 0 and i + shift < len(h):
                    gradient[i] += -1.0 * (1 - h[i + shift]) * dx
        else: # left shift (k negative)
            for i in range(len(h)):
                if i - shift >= 0 and i - shift < len(h):
                    gradient[i] += -1.0 * (1 - h[i - shift]) * dx
                    
        # Update momentum
        momentum = momentum_decay * momentum + gradient
        # Normalize momentum
        norm_momentum = np.linalg.norm(momentum)
        if norm_momentum > 0:
            momentum = momentum / norm_momentum
        
        # Perform step size line search
        step_size = learning_rate
        best_h_trial = None
        best_trial_c5 = float('inf')
        
        # Try sufficient descent
        for attempt in range(10):
            h_trial = h - step_size * momentum
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = project_sum(h_trial, n_points / 2)
            trial_c5 = compute_c5(h_trial, dx)
            
            if trial_c5 < best_trial_c5:
                best_trial_c5 = trial_c5
                best_h_trial = h_trial.copy()
            
            if trial_c5 < current_c5:
                break
            step_size *= 0.5  # backtrack
        
        # Apply best update  
        h = best_h_trial if best_h_trial is not None else h.copy()
        
        # Adaptive learning rate
        if abs(best_trial_c5 - current_c5) < tolerance:
            learning_rate = max(min_learning_rate, learning_rate * 0.9)
        else:
            learning_rate = min(1.0, learning_rate * 1.1)
            
    return (best_h.tolist(), best_c5, n_points)
```
