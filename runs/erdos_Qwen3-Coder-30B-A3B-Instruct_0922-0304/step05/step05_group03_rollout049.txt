import numpy as np
from scipy.signal import correlate
import time
from collections import defaultdict

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback: sparse step function
        h = np.zeros(n_points)
        h[:n_points//3] = 1.0
        h = normalize_h(h, target_sum)
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
    
    # Initialize with best known solution
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    # Use gradient-based local search with restarts
    max_restarts = 20
    max_iter_per_restart = 50
    max_inner_iter = 10
    
    current_h = best_h.copy()
    current_c5 = best_c5
    
    restart_count = 0
    
    while time.time() - start_time < budget_s - 1 and restart_count < max_restarts:
        # Local optimization with SLSQP
        from scipy.optimize import minimize
        
        # Define objective function
        def objective(h_vec):
            return compute_c5(h_vec)
        
        # Define constraint function for sum
        def constraint_sum(h_vec):
            return np.sum(h_vec) - target_sum
        
        # Create bounds
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        # Run optimization
        result = minimize(
            objective,
            current_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': max_inner_iter, 'ftol': 1e-8}
        )
        
        if result.success:
            new_h = np.clip(result.x, 0, 1)
            new_h = normalize_h(new_h, target_sum)
            new_c5 = compute_c5(new_h)
            
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
                
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
                    
                    # Early exit if target met
                    if best_c5 <= 0.3807:
                        break
        
        # Add small random perturbation for escape from local optima
        perturbation_magnitude = 0.01
        current_h = normalize_h(current_h + np.random.normal(0, perturbation_magnitude, n_points), target_sum)
        current_c5 = compute_c5(current_h)
        
        restart_count += 1
    
    # Ensure final solution is normalized
    final_h = normalize_h(best_h, target_sum)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
