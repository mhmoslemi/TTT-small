import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Simulated Annealing parameters
    initial_temp = 1.0
    cooling_rate = 0.995
    min_temp = 1e-8
    max_iter = 3000
    iter_per_temp = 20
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Current state
    current_h = h.copy()
    current_c5 = best_c5
    temperature = initial_temp
    
    iteration = 0
    temp_iter = 0
    
    while temperature > min_temp and time.time() - start_time < budget_s - 1:
        for _ in range(iter_per_temp):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Mutate current solution
            h_new = current_h + np.random.normal(0, temperature * 0.05, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Compute objective for new solution
            c5_new = compute_c5(h_new)
            
            # Accept or reject
            if c5_new < current_c5 or np.random.rand() < np.exp(-(c5_new - current_c5) / temperature):
                current_h = h_new
                current_c5 = c5_new
                
                # Update best if improved
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
            
            iteration += 1
            temp_iter += 1
            
        # Cool down the temperature
        temperature *= cooling_rate
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
    
    # Final confirmation of best solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
