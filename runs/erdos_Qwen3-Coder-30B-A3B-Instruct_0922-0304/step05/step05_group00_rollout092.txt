import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def repair_feasibility(h, target_sum):
    """Repair h to ensure it's in [0,1] and sums to target_sum."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():

    else:
        # Create a simple starting point
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    h = repair_feasibility(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Differential evolution inspired search
    population_size = 30
    population = []
    
    # Generate initial population
    for i in range(population_size):
        if i == 0:
            individual = h.copy()
        else:
            individual = h + np.random.normal(0, 0.05, n_points)
        individual = repair_feasibility(individual, target_sum)
        population.append(individual)
    
    step_size = 0.1
    max_iter = 10000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate all individuals
        fitness = [compute_c5(individual, dx) for individual in population]
        
        # Track best
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_h = population[min_fit_idx].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Create new population
        new_population = []
        
        # Keep the best
        new_population.append(population[min_fit_idx].copy())
        
        # Generate new individuals
        while len(new_population) < population_size:
            choice = np.random.random()
            
            if choice < 0.4:  # Differential evolution
                a, b, c = np.random.choice(population_size, 3, replace=False)
                mutant = population[a] + 0.8 * (population[b] - population[c])
                mutant = repair_feasibility(mutant, target_sum)
                new_population.append(mutant)
                
            elif choice < 0.7:  # Gradient descent
                idx = np.random.randint(population_size)
                individual = population[idx].copy()
                
                # Simple gradient estimation
                def obj(h):
                    return compute_c5(h, dx)
                
                # Simple gradient via finite differences
                eps = 1e-4
                grad = np.zeros_like(individual)
                for i in range(len(individual)):
                    h_plus = individual.copy()
                    h_plus[i] += eps
                    h_plus = repair_feasibility(h_plus, target_sum)
                    h_minus = individual.copy()
                    h_minus[i] -= eps
                    h_minus = repair_feasibility(h_minus, target_sum)
                    grad[i] = (obj(h_plus) - obj(h_minus)) / (2 * eps)
                
                # Update with gradient
                individual = individual - step_size * grad
                individual = repair_feasibility(individual, target_sum)
                new_population.append(individual)
                
            elif choice < 0.9:  # Random mutation
                individual = np.random.rand(n_points) * 2
                individual = repair_feasibility(individual, target_sum)
                new_population.append(individual)
                
            else:  # Clone best and mutate slightly
                individual = best_h + np.random.normal(0, 0.02, n_points)
                individual = repair_feasibility(individual, target_sum)
                new_population.append(individual)
        
        # Replace population
        population = new_population[:population_size]
        
        # Decay step size
        step_size *= 0.98
        iteration += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
