import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def compute_gradient(h, max_lags):
        gradient = np.zeros(n_points)
        for k in max_lags:
            # Positive contribution
            if k >= 0:
                for i in range(n_points - k):
                    gradient[i] += (1 - h[i + k])
            else:
                # Negative contribution (k < 0)
                for i in range(-k, n_points):
                    gradient[i] += (-h[i + k])
        return gradient / len(max_lags)

    def project(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 1000
    patience = 50
    best_c5 = float('inf')
    best_h = h.copy()
    
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap values
        overlaps = correlate(h, 1 - h, mode="full")
        max_val = overlaps.max()
        max_lags = [i - (n_points - 1) for i, val in enumerate(overlaps) if val == max_val]
        
        # Compute gradient
        gradient = compute_gradient(h, max_lags)
        
        # Update h
        h -= learning_rate * gradient
        h = project(h)
        
        # Evaluate objective
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Early termination 
        if best_c5 <= 0.3807:
            break
            
        # Adaptive learning rate
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= 0.9
            
        if iteration - last_improvement > patience:
            break
    
    # Local refinement
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        result = minimize(
            compute_c5,
            best_h.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
            refined_c5 = compute_c5(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
