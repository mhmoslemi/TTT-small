import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradients(h, dx):
    """Compute gradients of the maximum overlap with respect to h."""
    n = len(h)
    # Compute correlation of h with (1-h)
    corr_h_1h = correlate(h, 1 - h, mode="full")
    # Find maximum lag
    max_lag_idx = np.argmax(corr_h_1h)
    # Compute gradient by finite differences
    grad = np.zeros(n)
    eps = 1e-6
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = normalize_h(h_plus, np.sum(h))
        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
        grad[i] = (corr_plus[max_lag_idx] - corr_h_1h[max_lag_idx]) / eps
    return grad

    start_time = time.time()

    n_points = 70

    # Optimization parameters
    max_iter = 2000
    learning_rate = 0.1
    patience = 50
    patience_count = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_count = 0
        else:
            patience_count += 1
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradients
        grad = compute_gradients(h, dx)
        
        # Gradient step
        h += learning_rate * grad
        h = normalize_h(h, target_sum)
        
        # Coordinate-wise improvement
        for i in range(n_points):
            if time.time() - start_time > budget_s - 1:
                break
            orig_h = h[i]
            h[i] = orig_h + 0.01
            h = normalize_h(h, target_sum)
            new_c5 = evaluate_c5(h, dx)
            if new_c5 < current_c5:

            else:
                h[i] = orig_h
                h[i] = orig_h - 0.01
                h = normalize_h(h, target_sum)
                new_c5 = evaluate_c5(h, dx)
                if new_c5 < current_c5:
                    continue
                else:
                    h[i] = orig_h
        
        # Adjust learning rate if no improvement
        if patience_count > patience:
            learning_rate *= 0.5
            patience_count = 0

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
