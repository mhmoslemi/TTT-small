import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import math

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    # For faster gradient-based optimization
    def compute_c5(h):
        return evaluate_c5(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient-based local search with restarts
    max_iterations = 1000
    local_search_iters = 100
    restarts = 10
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        
        # Try multiple restarts
        for _ in range(restarts):
            # Option 1: refined initial h
            h_temp = h.copy()
            
            # Add small noise to explore
            noise_magnitude = 0.01
            noise = np.random.normal(0, noise_magnitude, n_points)
            h_temp += noise
            h_temp = normalize_h(h_temp, target_sum)
            
            # Local search using SLSQP
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = [{'type': 'eq', 'fun': constraint_sum}]
            
            def objective(h):
                return compute_c5(h)
                
            result = minimize(
                objective,
                h_temp,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': local_search_iters, 'ftol': 1e-8}
            )
            
            if result.success:
                h_new = result.x
                h_new = normalize_h(h_new, target_sum)
                current_c5 = compute_c5(h_new)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_new.copy()
            
            if best_c5 <= 0.3807:
                break
                
        # Try some evolutionary operators for diversity
        if iteration % 5 == 0:
            # Create variation
            h_var = h.copy()
            # Randomly perturb some values
            mask = np.random.random(n_points) < 0.2
            perturbation = np.random.normal(0, 0.02, n_points)
            h_var[mask] += perturbation[mask]
            h_var = normalize_h(h_var, target_sum)
            
            # Local optimization
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = [{'type': 'eq', 'fun': constraint_sum}]
            result = minimize(
                compute_c5,
                h_var,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            if result.success:
                h_new = result.x
                h_new = normalize_h(h_new, target_sum)
                current_c5 = compute_c5(h_new)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_new.copy()
        
        if best_c5 <= 0.3807:
            break

    # Final optimization of best solution
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    result = minimize(
        compute_c5,
        best_h.copy(),
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-10}
    )
    
    if result.success:
        best_h = normalize_h(result.x, target_sum)
        best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
