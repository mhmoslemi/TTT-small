import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def constraint_c5(h, c5_max):
        overlap = correlate(h, 1 - h, mode="full")
        return c5_max - np.max(overlap) * dx

    # Define constraints for feasibility check
    def feasibility_constraints(c5_estimate):
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        constraints.append({'type': 'ineq', 'fun': lambda h: constraint_c5(h, c5_estimate)})
        return constraints

    # Binary search on C5
    c5_low = 0.0
    c5_high = 0.38085  # Current known upper bound
    tolerance = 1e-4
    max_iterations = 40
    iteration = 0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Optimizer parameters
    bounds = [(0, 1) for _ in range(n_points)]
    max_iter_opt = 50
    
    # Refinement process: binary search
    while iteration < max_iterations and (c5_high - c5_low) > tolerance:
        if time.time() - start_time > budget_s - 1:
            break
            
        c5_mid = (c5_low + c5_high) / 2.0
        
        # Try to find feasible solution for this c5 value
        constraints = feasibility_constraints(c5_mid)
        
        # Start optimization from current best solution
        h_guess = best_h.copy()
        result = minimize(
            compute_c5,
            h_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': max_iter_opt, 'ftol': 1e-8}
        )
        
        if result.success:
            # Evaluate the solution
            new_c5 = compute_c5(result.x)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = result.x.copy()
                
            # Solution is feasible for this c5 value
            if new_c5 <= c5_mid:
                c5_high = c5_mid
            else:
                c5_low = c5_mid
        else:
            # Infeasible; try another approach
            # Try random perturbation from best solution
            h_pert = best_h + np.random.normal(0, 0.01, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            
            result2 = minimize(
                compute_c5,
                h_pert,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': max_iter_opt, 'ftol': 1e-8}
            )
            
            if result2.success:
                new_c5 = compute_c5(result2.x)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = result2.x.copy()
                if new_c5 <= c5_mid:
                    c5_high = c5_mid
                else:
                    c5_low = c5_mid
            else:
                c5_low = c5_mid  # Indicate infeasible with this c5
        
        iteration += 1
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

    # Final optimization from best solution
    if time.time() - start_time < budget_s - 1:
        constraints_final = [{'type': 'eq', 'fun': constraint_sum}]
        result_final = minimize(
            compute_c5,
            best_h.copy(),
            method='SLSQP',
            bounds=bounds,
            constraints=constraints_final,
            options={'maxiter': 100, 'ftol': 1e-10}
        )
        
        if result_final.success:
            final_c5 = compute_c5(result_final.x)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = result_final.x.copy()

    return (best_h.tolist(), best_c5, n_points)
