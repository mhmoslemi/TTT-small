import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        # Compute overlap integral using correlation
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing Parameters
    T0 = 1.0  # Initial temperature
    alpha = 0.95  # Cooling rate
    n_iter = 1000  # Max iterations
    
    # Temperature schedule
    T = T0
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()

    for i in range(n_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate neighbor solution
        new_h = current_h.copy()
        # Add Gaussian noise
        noise = np.random.normal(0, 0.02, n_points)
        new_h += noise
        # Clip to bounds
        new_h = np.clip(new_h, 0, 1)
        # Rescale to maintain constraint
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Compute new objective
        new_c5 = compute_c5(new_h)
        
        # Accept or reject based on Metropolis criterion
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = new_h
            current_c5 = new_c5
            
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Cool down temperature
        T = T * alpha
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
