from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):

        # Shuffle to get random binary solution
        np.random.shuffle(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def binary_search():
        """Optimize binary solution using simulated annealing"""
        # Start with current h and convert to binary
        # Use binary search with simulated annealing
        
        # Generate initial binary with exactly n_points//2 ones
        h_bin = h.copy()
        idxs = np.arange(n_points)
        np.random.shuffle(idxs)
        h_bin[idxs[:n_points//2]] = 1.0
        h_bin[idxs[n_points//2:]] = 0.0
        
        best_h = h_bin.copy()
        best_c5 = compute_c5(best_h)
        
        # Simulated annealing parameters
        temp = 1.0
        cool_rate = 0.995
        max_iter = 5000
        min_temp = 1e-6
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Generate neighbor by swapping two random positions
            i, j = np.random.choice(n_points, 2, replace=False)
            h_new = h_bin.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
            
            # Compute overlap for the new configuration
            c5_new = compute_c5(h_new)
            
            # Accept or reject based on simulated annealing
            if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / temp):
                h_bin = h_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_bin.copy()
            
            temp *= cool_rate
            if temp < min_temp:
                temp = min_temp
                
            if best_c5 <= 0.3807:
                break
                
        return best_h, best_c5
        
    def continuous_relaxation(h_init):
        """Use CVXPY to refine solution to continuous variables"""
        h_var = cp.Variable(n_points)
        
        # Objective is to minimize maximum overlap
        # We'll bound the overlap by introducing an auxiliary variable
        max_overlap = cp.Variable()

        # We compute overlap values explicitly and add constraints
        # This is an approximation of the full optimization problem
        
        # Create a two-way constraint: fm(k) <= max_overlap for all k
        # For computational handling, use a large enough sample
        
        # Instead, we'll minimize using gradient descent on computed overlaps
        # Reuse the original h as starting point
        h_sol = h_init.copy()
        
        # Local refinement using gradient descent on the computed overlap
        current_c5 = compute_c5(h_sol)
        
        # Simple gradient ascent step
        lr = 0.1
        for _ in range(2000):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Gradient approximation using finite differences
            grad = np.zeros(n_points)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h_sol.copy()
                h_minus = h_sol.copy()
                
                h_plus[i] += eps
                h_minus[i] -= eps
                
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                
                # Renormalize
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                # Compute overlap changes
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
                overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps) * dx
                
            # Update with negative gradient to minimize
            h_sol = h_sol - lr * grad
            h_sol = np.clip(h_sol, 0, 1)
            h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
            
            # Update current c5
            current_c5 = compute_c5(h_sol)
            
            if current_c5 <= 0.3807:

        return h_sol, current_c5
        
    # Run binary search step
    h_bin, c5_bin = binary_search()
    
    # Run continuous refinement step
    h_final, c5_final = continuous_relaxation(h_bin)
    
    # Return best final configuration
    final_c5 = compute_c5(h_final)
    
    return (h_final.tolist(), final_c5, n_points)
