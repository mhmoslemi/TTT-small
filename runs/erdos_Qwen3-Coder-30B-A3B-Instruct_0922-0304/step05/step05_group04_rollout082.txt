import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Create swarm
    particles = [h.copy() for _ in range(swarm_size)]
    velocities = [np.random.uniform(-0.05, 0.05, n_points) for _ in range(swarm_size)]
    
    # Best positions for each particle and global best
    personal_best = [p.copy() for p in particles]
    personal_best_values = [float('inf')] * swarm_size
    global_best = particles[0].copy()
    global_best_value = float('inf')
    
    # PSO parameters
    omega_max = 0.9
    omega_min = 0.4
    phi1 = 2.0
    phi2 = 2.0
    v_max = 0.1
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Update inertia weight
        omega = omega_max - (omega_max - omega_min) * (time.time() - start_time) / budget_s
        
        for i in range(swarm_size):
            h = particles[i]
            v = velocities[i]
            
            # Compute current C5 value
            current_c5 = compute_c5(h)
            
            # Update personal best
            if current_c5 < personal_best_values[i]:
                personal_best[i] = h.copy()
                personal_best_values[i] = current_c5
            
            # Update global best
            if current_c5 < global_best_value:
                global_best = h.copy()
                global_best_value = current_c5
                
            # Early termination
            if global_best_value <= 0.3807:
                break
                
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (omega * v + 
                 phi1 * r1 * (personal_best[i] - h) + 
                 phi2 * r2 * (global_best - h))
            
            # Limit velocity
            v = np.clip(v, -v_max, v_max)
            velocities[i] = v
            
            # Update position
            new_h = h + v
            
            # Keep within bounds [0,1]
            new_h = np.clip(new_h, 0, 1)
            
            # Renormalize
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            particles[i] = new_h
            
        if global_best_value <= 0.3807:
            break
    
    final_c5 = compute_c5(global_best)
    return (global_best.tolist(), final_c5, n_points)
