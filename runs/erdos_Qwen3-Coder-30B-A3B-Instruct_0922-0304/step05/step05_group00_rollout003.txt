import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to [0,1] and normalize to target sum."""
    h = np.clip(h, 0, 1)
    return normalize_h(h, target_sum)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default construction: a simple step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    iteration = 0
    max_iter = 5000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current C5
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Sample random perturbations and evaluate
        candidates = []
        num_candidates = 20
        
        for _ in range(num_candidates):
            # Generate a slightly perturbed version
            perturbation = np.random.normal(0, 0.02, n_points)
            h_pert = h + perturbation
            h_pert = project_to_feasible(h_pert, target_sum)
            candidates.append(h_pert)
        
        # Evaluate all candidates
        candidate_scores = [compute_c5(c, dx) for c in candidates]
        best_candidate_idx = np.argmin(candidate_scores)
        
        # Accept best candidate
        h = candidates[best_candidate_idx]
        
        iteration += 1
    
    # Final check and return
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
