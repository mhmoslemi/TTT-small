import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-resolution approach
    # Coarse optimization first
    n_coarse = n_points // 2
    dx_coarse = 2.0 / n_coarse
    h_coarse = h[::2]  # Downsample by taking every other element
    h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    
    # Coarse optimization
    def compute_c5_coarse(h_coarse):
        h_full = np.repeat(h_coarse, 2)
        overlap = correlate(h_full, 1 - h_full, mode="full")
        return np.max(overlap) * dx_coarse

    def constraint_sum_coarse(h_coarse):
        return np.sum(h_coarse) - n_coarse / 2

    constraints_coarse = [{'type': 'eq', 'fun': constraint_sum_coarse}]
    bounds_coarse = [(0, 1) for _ in range(n_coarse)]

    coarse_start = time.time()
    result = minimize(
        compute_c5_coarse, 
        h_coarse, 
        method='SLSQP', 
        bounds=bounds_coarse, 
        constraints=constraints_coarse,
        options={'maxiter': 50, 'ftol': 1e-6}
    )
    
    if result.success:
        h_coarse = result.x
        h_coarse = np.clip(h_coarse, 0, 1)
        h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
        
    # Upsample to full resolution
    h_fine = np.repeat(h_coarse, 2)
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    
    # Fine optimization
    h = h_fine

    # Main optimization loop with multi-start
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Optimize using SLSQP with restarts
        result = minimize(
            compute_c5, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        if result.success:
            h = result.x
            # Ensure feasibility
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            # If optimization fails, try different strategy
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            
        # Occasionally, restart with a different initialization
        if iteration % 100 == 0 and iteration > 0:
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
