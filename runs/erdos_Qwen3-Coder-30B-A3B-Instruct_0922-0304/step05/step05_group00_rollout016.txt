import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    """Evaluate objective and check constraints."""
    c5 = compute_c5(h, dx)
    return c5

def mutate_individual(h, noise_level=0.01):
    """Apply small random mutation to an individual."""
    h = h + np.random.normal(0, noise_level, len(h))
    h = np.clip(h, 0, 1)
    return h

def differential_evolution_step(population, idx, F=0.8, CR=0.7):
    """Perform differential evolution step for individual at idx."""
    p = population[idx]
    candidates = list(range(len(population)))
    candidates.remove(idx)
    a, b, c = np.random.choice(candidates, 3, replace=False)
    
    mutant = population[a] + F * (population[b] - population[c])
    mutant = np.clip(mutant, 0, 1)
    
    trial = np.where(np.random.random(len(p)) < CR, mutant, p)
    return trial

def gradient_step(h, dx, learning_rate=0.1):
    """Perform a gradient-based step."""
    # Simple gradient approximation
    # For efficiency, we'll directly compute gradient of a single overlap term
    # This is a rough approximation and could be improved by computing full Jacobian
    
    n = len(h)
    # Compute current overlaps
    overlap = correlate(h, 1 - h, mode="full") * dx
    
    # Approximate gradient with respect to h
    # For now we use finite differences to estimate gradient
    eps = 1e-5
    gradient = np.zeros(n)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        overlap_plus = correlate(h_plus, 1 - h_plus, mode="full") * dx
        gradient[i] = (np.max(overlap_plus) - np.max(overlap)) / eps
    
    h_new = h - learning_rate * gradient
    h_new = np.clip(h_new, 0, 1)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use n_points = 70 as in previous work
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 30
    population = []
    
    # Initialize from available parent construction, if it exists
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        # Simple construction to start
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    
    # Create initial population
    for _ in range(population_size):
        h = mutate_individual(base_h.copy(), 0.02)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_individual = None
    step_size = 0.1
    
    iteration = 0
    max_iter = 300
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate entire population
        fitness = [evaluate_individual(h, dx) for h in population]
        
        # Track best individual so far
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_individual = population[min_fit_idx].copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Generate new candidates with mixed strategies
        new_population = []
        
        # Differential evolution: 40%
        for _ in range(int(0.4 * population_size)):
            idx = np.random.randint(0, len(population))
            new_individual = differential_evolution_step(population, idx)
            new_individual = np.clip(new_individual, 0, 1)
            new_individual = normalize_h(new_individual, target_sum)
            new_population.append(new_individual)
        
        # Gradient descent: 30%
        for _ in range(int(0.3 * population_size)):
            idx = np.random.randint(0, len(population))
            new_individual = gradient_step(population[idx], dx, step_size)
            new_individual = normalize_h(new_individual, target_sum)
            new_population.append(new_individual)
        
        # Random sampling: 20%
        for _ in range(int(0.2 * population_size)):
            new_individual = np.random.random(n_points)
            new_individual = np.clip(new_individual, 0, 1)
            new_individual = normalize_h(new_individual, target_sum)
            new_population.append(new_individual)
        
        # Mutation of best solution: 10%
        for _ in range(int(0.1 * population_size)):
            new_individual = mutate_individual(best_individual.copy(), 0.01)
            new_individual = normalize_h(new_individual, target_sum)
            new_population.append(new_individual)
        
        # Elitism
        new_population.append(best_individual.copy())
        
        # Ensure correct population size
        new_population = new_population[:population_size]
        
        # Replace worst individuals
        worst_indices = np.argsort(fitness)[-len(new_population):]
        for i, idx in enumerate(worst_indices[:len(new_population)]):
            population[idx] = new_population[i]
        
        # Adjust step size
        if iteration > 0 and iteration % 10 == 0:
            step_size *= 0.98
            
        iteration += 1
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
