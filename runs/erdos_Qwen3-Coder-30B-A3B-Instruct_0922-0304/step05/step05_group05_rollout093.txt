import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters  
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: 3-region piecewise function
        h = np.ones(n_points)
        split1 = n_points // 2
        split2 = 3 * n_points // 4
        h[split1:split2] = 0.25
        h[split2:] = 0.5
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def rescale(h):
        return h * (n_points / 2) / np.sum(h)
    
    def swap_regions(h, i1, i2, size):
        h_new = h.copy()
        # Swap values in two non-overlapping regions
        h_new[i1:i1+size], h_new[i2:i2+size] = h_new[i2:i2+size], h_new[i1:i1+size]
        return rescale(h_new)
    
    # Simulated Annealing with swaps
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    # Parameters
    n_swaps = 1000
    swap_size = 5
    temperature = 0.1
    
    for _ in range(n_swaps):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Select two random non-overlapping regions
        i1 = np.random.randint(0, n_points - swap_size)
        i2 = np.random.randint(0, n_points - swap_size)
        
        # Ensure they don't overlap
        while abs(i1 - i2) < swap_size:
            i2 = np.random.randint(0, n_points - swap_size)
        
        # Perform swap
        h_new = swap_regions(best_h, i1, i2, swap_size)
        
        # Evaluate
        new_c5 = compute_c5(h_new)
        
        # Accept with probability based on temperature
        if new_c5 < best_c5 or np.random.rand() < np.exp(-(new_c5 - best_c5) / temperature):
            best_h = h_new
            best_c5 = new_c5
            
        # Cool down
        temperature *= 0.999
        
        if best_c5 <= 0.3807:
            break
    
    # Final polish with SLSQP
    from scipy.optimize import minimize
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    try:
        result = minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h = rescale(final_h)
            final_c5 = compute_c5(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
