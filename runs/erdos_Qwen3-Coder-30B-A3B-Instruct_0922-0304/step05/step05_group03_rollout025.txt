import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    dx = 2.0 / n_points
    target_sum = n_points / 2

    def compute_c5(h):
        return evaluate_c5(h, dx)
        
    # Use Halton sequence for initialization
    def halton_sequence(n, base=2):
        result = []
        for i in range(n):
            value = 0
            f = 1.0 / base
            k = i + 1
            while k > 0:
                result.append((k % base) * f)
                k //= base
                f /= base
        return np.array(result)
    
    # Generate multiple initial points using Halton sequence
    population_size = 50
    population = []
    population.append(h.copy())
    
    # Generate Halton-based initial points
    halton_seq = halton_sequence(population_size * n_points, base=2)
    halton_seq = halton_seq.reshape(population_size, n_points)
    
    for i in range(1, population_size):
        h_init = halton_seq[i]
        h_init = normalize_h(h_init, target_sum)
        population.append(h_init)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start optimization with SLSQP
    for ind in population:
        # Apply SLSQP optimization
        def objective(h):
            return compute_c5(h)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]

        result = minimize(

                best_c5 = c5
                best_h = h_new.copy()
                
        # Check early termination
        if time.time() - start_time > budget_s - 1:
            break
            
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
