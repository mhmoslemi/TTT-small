import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints and bounds
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Projected Gradient Descent
    max_iter = 1000
    learning_rate = 0.01
    stagnation_count = 0
    best_c5_stagnant = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        overlap_full = correlate(h, 1 - h, mode="full")
        max_val = overlap_full.max()
        
        # Find all lags where maximum overlap occurs
        max_lags = []
        for i in range(len(overlap_full)):
            if overlap_full[i] == max_val:
                k = i - (n_points - 1)
                max_lags.append(k)
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for k in max_lags:
            for i in range(n_points):
                if i + k < n_points:
                    gradient[i] += (1 - h[i + k])
                if i - k >= 0:
                    gradient[i] -= h[i - k]
        
        if max_lags:
            gradient /= len(max_lags)
        
        # Update h using gradient descent
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute C5 for current h
        current_c5 = compute_c5(h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            stagnation_count = 0
            best_c5_stagnant = current_c5
        else:
            stagnation_count += 1
            
        # Adaptive learning rate
        if stagnation_count > 50 or abs(current_c5 - best_c5_stagnant) < 1e-8:
            learning_rate *= 0.9
            stagnation_count = 0
            best_c5_stagnant = current_c5
        
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    # Final SLSQP refinement
    try:
        result = minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_c5 = compute_c5(final_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h
    except:
        pass  # If local search fails, keep best solution so far

    return (best_h.tolist(), best_c5, n_points)
