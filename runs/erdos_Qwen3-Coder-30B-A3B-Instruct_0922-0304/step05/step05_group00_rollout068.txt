import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def evaluate_individual(h, dx):
    """Evaluate objective and check constraints."""
    c5 = compute_c5(h, dx)
    return c5

def repair_feasibility(h, target_sum):
    """Repair h to satisfy sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = repair_feasibility(h, target_sum)
    else:
        # Create a simple step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = repair_feasibility(h, target_sum)
    
    # Track best
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Differential evolution inspired search
    population_size = 30
    population = [h.copy()]
    for i in range(1, population_size):
        h_new = h + np.random.normal(0, 0.05, n_points)
        h_new = repair_feasibility(h_new, target_sum)
        population.append(h_new)
    
    # Evolution parameters
    step_size = 0.05
    step_decay = 0.98
    iterations = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        # Evaluate all individuals
        fitness = [evaluate_individual(p, dx) for p in population]
        
        # Update best
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_h = population[min_fit_idx].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Generate new population
        new_population = [best_h.copy()]  # Elite
        
        # Differential evolution: 40% of population
        for _ in range(int(0.4 * (population_size - 1))):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            donor = population[a] + 0.8 * (population[b] - population[c])
            donor = repair_feasibility(donor, target_sum)
            new_population.append(donor)
            
        # Gradient descent: 30% of population
        for _ in range(int(0.3 * (population_size - 1))):
            idx = np.random.randint(len(population))
            h_current = population[idx].copy()
            # Simple gradient: small step in direction of decrease
            grad = np.zeros_like(h_current)
            # Approximate gradient by finite differences
            eps = 1e-4
            base_c5 = compute_c5(h_current, dx)
            for i in range(n_points):
                h_test = h_current.copy()
                h_test[i] += eps
                h_test = repair_feasibility(h_test, target_sum)
                c5_test = compute_c5(h_test, dx)
                grad[i] = (c5_test - base_c5) / eps
            # Update with gradient step
            new_h = h_current - step_size * grad
            new_h = repair_feasibility(new_h, target_sum)
            new_population.append(new_h)
            
        # Random: 20%
        for _ in range(int(0.2 * (population_size - 1))):
            new_h = np.random.uniform(0, 1, n_points)
            new_h = repair_feasibility(new_h, target_sum)
            new_population.append(new_h)
            
        # Mimic best: 10%
        for _ in range(int(0.1 * (population_size - 1))):
            h_ref = best_h + np.random.normal(0, 0.02, n_points)
            h_ref = repair_feasibility(h_ref, target_sum)
            new_population.append(h_ref)
            
        # Fill up to population size
        while len(new_population) < population_size:
            h_new = h + np.random.normal(0, 0.05, n_points)
            h_new = repair_feasibility(h_new, target_sum)
            new_population.append(h_new)
        
        # Replace old population
        population = new_population[:population_size]
        
        # Decrement step size
        step_size *= step_decay
        iterations += 1
    
    return (best_h.tolist(), best_c5, n_points)
