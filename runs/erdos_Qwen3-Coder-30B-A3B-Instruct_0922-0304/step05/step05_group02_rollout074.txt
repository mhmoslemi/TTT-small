```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Problem, Minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum = target_sum and 0 <= h <= 1."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        h = np.ones(n_points) * 0.5
        h = project_to_feasible(h, target_sum)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute correlations for all shifts
        corrs = correlate(h, 1 - h, mode="full")
        max_corrs = np.max(corrs) * dx
        
        # If we found a better solution, keep it
        if max_corrs < best_c5:
            best_c5 = max_corrs
            best_h = h.copy()
            
        # Early exit if target is met
        if best_c5 <= 0.3807:
            break
            
        # Use a convex relaxation approach
        # This will involve solving a convex optimization problem
        # Set up variables
        h_var = cp.Variable(n_points)
        C = cp.Variable(1)
        
        # Objective: minimize C
        objective = Minimize(C)
        
        # Constraints
        constraints = []
        
        # Every correlation should be bounded by C
        for k in range(len(corrs)):
            # Compute the correlation at shift k 
            corr_val = corrs[k] * dx
            constraints.append(C >= corr_val)
            
        # Sum constraint
        constraints.append(cp.sum(h_var) == target_sum)
        
        # Bound constraints
        constraints.append(h_var >= 0)
        constraints.append(h_var <= 1)
        
        # Create and solve problem
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            
            if prob.status == 'optimal':
                # Update h
                new_h = h_var.value
                h = project_to_feasible(new_h, target_sum)
        except:
            pass
            
        # Periodic perturbations to escape local minima
        if iteration % 50 == 0:
            noise_scale = 0.05 * np.exp(-iteration / 100.0)
            noise = np.random.normal(0, noise_scale, n_points)
            h = h + noise
            h = project_to_feasible(h, target_sum)
            
    # Final evaluation of the best solution
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
```
