import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: 3-region piecewise function
        h = np.ones(n_points)
        mid = n_points // 2
        h[mid:] = 0
        h[mid//2:mid] = 0.5
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def rescale(h):
        """Ensure sum(h) = n_points/2"""
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)

    def swap_segments(h, i1, i2, size):
        """Swap segments of h starting at i1 and i2"""
        h_new = h.copy()
        temp = h_new[i1:i1+size].copy()
        h_new[i1:i1+size] = h_new[i2:i2+size]
        h_new[i2:i2+size] = temp
        return rescale(h_new)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Swap-based local search
    n_swaps = 2000
    swap_size = 5
    temperature = 0.1
    
    for _ in range(n_swaps):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Select two non-overlapping segments
        i1 = np.random.randint(0, n_points - swap_size)
        i2 = np.random.randint(0, n_points - swap_size)
        while abs(i1 - i2) < swap_size:
            i2 = np.random.randint(0, n_points - swap_size)
            
        # Perform swap
        h_new = swap_segments(best_h, i1, i2, swap_size)
        c5_new = compute_c5(h_new)
        
        # Accept with probability
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / temperature):
            best_h = h_new
            best_c5 = c5_new
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
    # Final polish with SLSQP
    from scipy.optimize import minimize
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
    
    try:
        result = minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            best_c5 = compute_c5(best_h)
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
