import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

    def halton_sequence(dim, n_points):
        # Simple Halton sequence generator for [0,1]^dim
        result = np.zeros((n_points, dim))
        for i in range(dim):
            base = 2 + i
            for j in range(n_points):
                result[j, i] = 0
