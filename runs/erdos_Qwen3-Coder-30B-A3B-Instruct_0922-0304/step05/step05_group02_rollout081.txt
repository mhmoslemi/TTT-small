import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def line_search(h, grad, dx, max_alpha=1.0, tol=1e-4):
    """Perform line search to find optimal step size."""
    def eval_c5(alpha):
        h_new = project_h(h - alpha * grad, len(h) / 2)
        return compute_c5(h_new, dx)
    
    alpha = max_alpha
    c5_prev = eval_c5(0.0)
    c5_new = eval_c5(alpha)
    
    # Backtracking line search
    while c5_new >= c5_prev and alpha > 1e-8:
        alpha *= 0.5
        c5_new = eval_c5(alpha)
    
    return alpha

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from existing construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        # Create initial guess
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_h(h, target_sum)
    
    # Parameters
    momentum = 0.9
    initial_lr = 0.05
    min_lr = 1e-6
    use_momentum = True
    adaptive_lr = True
    
    # For tracking
    best_c5 = float('inf')
    best_h = h.copy()
    prev_grad_norm = 1.0
    no_improvement = 0
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute all correlations
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        k_opt = max_idx - (len(h) - 1)
        
        # Compute gradient with respect to shift k_opt
        # Use central difference to compute gradient-like update
        h_shifted = np.roll(h, k_opt)
        grad = 1 - h_shifted
        
        # Apply momentum if needed
        if use_momentum and iteration > 1:
            grad = momentum * grad + (1 - momentum) * grad
        
        # Line search for optimal step size
        lr = initial_lr
        if adaptive_lr and prev_grad_norm > 1e-8:
            lr = min(initial_lr, 0.1 / prev_grad_norm)
        
        lr = max(lr, min_lr)
        
        # Perform line search
        alpha = line_search(h, grad, dx, lr)
        
        # Update h
        h_new = h - alpha * grad
        h_new = project_h(h_new, target_sum)
        
        # Update gradient norm
        grad_norm = np.linalg.norm(grad)
        prev_grad_norm = grad_norm
        
        # Evaluate
        current_c5 = compute_c5(h_new, dx)
        
        # Update best if better
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        # Check if stuck
        if no_improvement > 50:
            # Random perturbation
            noise = np.random.normal(0, 0.01, n_points)
            h_new = h + noise
            h_new = project_h(h_new, target_sum)
            no_improvement = 0
            
        h = h_new
    
    # Final validation    
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
