import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, k_max):
    """Compute the gradient of the overlap at lag k_max with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    # For correlation at lag k_max, we get contribution from h[i] to overlap at k_max
    # overlap[k_max] = sum h[i] * (1 - h[i + k_max]) for valid i
    # So the gradient with respect to h[i] is (1 - h[i + k_max]) if i + k_max < n else 0
    for i in range(n):
        if i + k_max < n:
            grad[i] = 1 - h[i + k_max]
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
        
    # Initialize
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Learning rate and decay
    alpha = 0.1
    alpha_decay = 0.5
    patience = 0
    patience_limit = 50
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        current_c5 = evaluate_c5(h, dx)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if patience >= patience_limit:
            alpha *= alpha_decay
            patience = 0

        # Find k that maximizes overlap
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        
        # Compute gradient
        gradient = compute_gradient(h, dx, k_max)
        
        # Update h
        h = h - alpha * gradient
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Occasionally perturb to escape local maxima
        if iteration % 100 == 0:
            h += np.random.normal(0, 0.001, n_points)
            h = normalize_h(h, target_sum)
            
        # Early exit if we have a good enough result
        if best_c5 <= 0.3807:
            break
            
    final_c5 = evaluate_c5(best_h, dx)

    return (best_h.tolist(), final_c5, n_points)
