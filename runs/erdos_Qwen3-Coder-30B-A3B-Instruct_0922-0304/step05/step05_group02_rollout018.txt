import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        # Basic initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    
    # Parameters
    eta = 0.1  # Initial learning rate
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute all correlations
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        # Compute gradient for the worst case (max overlap)
        # Correlation at k = max_k involves shifting h left by k positions
        # For correlation h * (1-h) at shift k: 
        # We have sum_{i} h[i] * (1-h[i+k]) 
        # Gradient wrt h[i]: (1 - h[i+k]) for i+k < n, else 0
        # Though this is a bit tricky, we approximate as:
        # when we're looking at the maximizer shift, we know:
        # at this shift, h[i] * (1-h[i+k]) was contributing to the max
        
        # Simple gradient descent approach:
        # Compute the gradient of the worst-case overlap with respect to h
        # The derivative ∂/∂h[i] of the overlap at shift k is (1 - h[i-k]) 
        # We construct a gradient vector
        gradient = np.zeros(n_points)
        # We know that at shift k, overlap = sum over i of h[i] * (1 - h[i+k])
        # So deriv w.r.t. h[i] is 1 - h[i+k] when i+k < n_points, else 0
        # Need to be careful about indexing
        
        # Method: construct a matrix of shifted (1-h) values
        shifted_ones_minus_h = np.zeros((n_points, n_points))
        for i in range(n_points):
            shifted_ones_minus_h[i, :] = np.roll(1 - h, i)
        
        # Compute gradient vector based on max k
        # gradient[i] = - (1 - h[i-k]) where it's non-zero
        # For shift k = max_k, corresponding to correlation index max_k,
        # which is at the 'middle' of full convolution, i.e. index n_points-1 + k where k <= n_points-1
        # shift k means index i shifts to i+k (or i-k)
        # Let's be consistent
        # shift k means h[i] * (1 - h[i+k]) so gradient w.r.t. h[i] = (1 - h[i+k])
        #corr_ind = max_k  # This is the index in full correlation  
        # Since full convolution of h of length n has length 2n-1
        # The shift k ranges 0 to n-1, where k corresponds to shift in 0 to n-1  
        # The shift k = corr_ind - (n-1) where corr_ind ranges from 0 to 2n-2
        # if corr_ind >= n-1, then shift k = corr_ind - (n-1) 
        # if corr_ind < n-1, the shift is leading negative.
        shift_k = max_k - (n_points - 1) 
        if shift_k < 0:
            # No overlap will contribute to this shift
            pass  # no change to gradient, which is 0
        else:
            # shift_k = k >= 0
            # Gradient is (1 - h[i+k]) where i is varied  
            # This is directly: 
            for i in range(n_points - shift_k):
                if i + shift_k < n_points:
                    gradient[i] += 1 - h[i + shift_k]
        
        # Apply gradient descent
        h = h - eta * gradient
        h = project_h(h, target_sum)
        
        # Adaptive learning rate
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            eta *= 1.01
        else:
            eta *= 0.9
            
        if best_c5 <= 0.3807:
            break
            
        # Occasionally add noise to avoid local minimum
        if iteration % 100 == 0:
            h += np.random.normal(0, 0.005, n_points)
            h = project_h(h, target_sum)
        iteration += 1
    
    # Final check for best solution
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
