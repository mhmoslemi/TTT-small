import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def repair_feasibility(h, target_sum):
    """Repair h to satisfy constraints."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Parameters
    population_size = 30
    max_iter = 10000
    step_size = 0.01
    step_decay = 0.98
    
    # Main loop
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Generate new population
        new_population = [best_h.copy()]
        
        # Differential evolution mutation (40%)
        for _ in range(int(0.4 * population_size) - 1):
            a, b, c = np.random.choice(len(new_population), 3, replace=False)
            mut = new_population[a] + 0.8 * (new_population[b] - new_population[c])
            mut = repair_feasibility(mut, target_sum)
            new_population.append(mut)
        
        # Gradient descent (30%)
        for _ in range(int(0.3 * population_size)):
            # Simple gradient approximation
            h_trial = h + np.random.normal(0, step_size, n_points)
            h_trial = repair_feasibility(h_trial, target_sum)
            c5_trial = compute_c5(h_trial, dx)
            if c5_trial < best_c5:
                h = h_trial.copy()
                best_c5 = c5_trial
                best_h = h.copy()
        
        # Random sampling (20%)
        for _ in range(int(0.2 * population_size)):
            h_rand = np.random.rand(n_points)
            h_rand = repair_feasibility(h_rand, target_sum)
            new_population.append(h_rand)
        
        # Mutation of best (10%)
        for _ in range(int(0.1 * population_size)):
            h_mut = best_h + np.random.normal(0, step_size * 0.5, n_points)
            h_mut = repair_feasibility(h_mut, target_sum)
            new_population.append(h_mut)
        
        # Evaluate and update best
        evaluated = [compute_c5(h, dx) for h in new_population]
        best_idx = np.argmin(evaluated)
        if evaluated[best_idx] < best_c5:
            best_c5 = evaluated[best_idx]
            best_h = new_population[best_idx].copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Update step size
        step_size *= step_decay
        
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
