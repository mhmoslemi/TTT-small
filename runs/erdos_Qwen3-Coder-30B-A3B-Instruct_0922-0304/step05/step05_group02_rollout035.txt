import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to satisfy sum(h) = target_sum while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = target_sum / current_sum
        h = h * scale
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient w.r.t. h for the k-th shift where k achieves max overlap."""
    grad = np.zeros_like(h)
    # Compute convolution of h with (1-h) and find max
    conv = correlate(h, 1-h, mode="full")
    k_best = np.argmax(conv)
    # The gradient is the derivative of convolution w.r.t. h
    # For k-position, d/dh_i of (h * (1-h))(k) involves terms:
    # If k = i+j, then d/dh_i h[j] * (1-h[k-i]) and d/dh_i (1-h[j]) * h[k-i] 
    # So d/dh_i (h[j] * (1-h[k-i])) = h[j] * d/dh_i (1-h[k-i]) 
    # d/dh_i (1-h[k-i]) = -1 if i = k-i else 0 => h[j] * (-1 if it is at position) 
    # or 
    # d/dh_i (h[j] * (1-h[k-i])) = (1-h[k-i]) * d/dh_i h[j] if i==j and 0 otherwise
    # Actually simpler approach: d/dh_i [h_i * (1-h_{i+k})] = (1-h_{i+k}) - h_i * d/dh_i h_{i+k}
    # No wait, just use numerical gradient
    
    # Refining approach: 
    # We directly compute gradient of the overlap part directly
    # The convolution results are such that for overlaps at k position = sum(h[i] * (1-h[i+k]))
    # Taking gradient of this w.r.t. h_j:
    # If j != i+k: derivative = 0
    # If j = i+k: derivative = -h_i
    # If j = i: derivative = 1-h_{i+k}  
    # 
    # This suggests for the fixed position k with a particular element h[i],  the derivative of 
    # sum over i of h[i] * (1 - h[i+k]) with respect to h_j depends on whether j = i or j = i+k
    
    # Re-approach: convolution is conv[i] = sum_{j} h[j] * (1 - h[i-j]) 
    # Mark j such that h[j] is participating in (1 - h[i-j]) for a fixed i.
    # That is: if conv[i] includes h[j] * (1 - h[i-j]), we are deriving w.r.t. h[j]
    
    # Start from der[i] of sum(j) h[j] * (1-h[i-j])
    # This is the derivative of h[j] * (1-h[i-j]) w.r.t. h[j] = (1-h[i-j]) + h[j] * (-d/dh[j](h[i-j]))
    # But d/dh[j](h[i-j]) = 0 unless j = i-j or i = 2j, which means j = i/2. This is confusing.
    # Let us use finite difference for precise numerical ones: we compute the derivative of 
    # the max overlap shift for given h using numerical finite diff
    
    # So compute the zero-order derivative directly from dual interpretation
    # We'll use finite differences on the objective function itself, much more reliable.
    
    # x = h is our decision variable
    # Gradient of C5 w.r.t. h_i = lim_{ε -> 0} [C5(h+εe_i) - C5(h)] / ε
    # This is computationally simpler with finite differences for any reasonable ε
    
    eps = 1e-5
    grad = np.zeros_like(h)
    h_current = h.copy()
    c5_current = compute_c5(h_current, dx)
    
    # For each i, compute finite difference of the maximum overlap w.r.t. h[i]
    for i in range(len(h)):
        h_plus = h_current.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h) / 2)
        c5_plus = compute_c5(h_plus, dx)
        grad[i] = (c5_plus - c5_current) / eps
    return grad

def compute_gradient_active_shift(h, dx, k_max):
    """Compute gradient w.r.t. h for some active shift contributing to max overlap."""
    # Compute maximum overlap shift
    conv = correlate(h, 1 - h, mode="full")
    k_best = np.argmax(conv)
    # For this particular overlap (k_best), compute gradient
    # Convolution at position k is sum of h[j] * (1 - h[j-k]) over all j where j-k is valid
    # Derivative w.r.t. h[i] = sum_{j: j-k==i} (1 - h[j]) + sum_{j: j==i} (-h[j-k])  
    # This is more systematically done with finite diff of intermediate values
    
    # Instead, give approximate gradient by finite diff:
    eps = 1e-5
    grad = np.zeros_like(h)
    h_current = h.copy()
    c5_current = compute_c5(h_current, dx)
    for i in range(len(h)):
        h_plus = h_current.copy()
        h_plus[i] += eps
        h_plus = project_to_feasible(h_plus, len(h) / 2)
        c5_plus = compute_c5(h_plus, dx)
        grad[i] = (c5_plus - c5_current) / eps
    return grad

def line_search(h_current, grad, dx, num_backtracks=10, step_size=0.1):
    """Perform line search to find an appropriate step size."""
    hproj = h_current.copy()
    initial_c5 = compute_c5(hproj, dx)
    for i in range(num_backtracks):
        step = -step_size * grad
        h_new = hproj + step
        h_new = project_to_feasible(h_new, len(h_new) / 2)
        c5_new = compute_c5(h_new, dx)
        reduction = initial_c5 - c5_new
        if reduction > 1e-8:
            return h_new, step_size
        else:
            step_size *= 0.5
    return h_current, step_size

        init_h = project_to_feasible(init_h, n_points / 2)

    # Initialize
    h = init_h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Momentum
    momentum = np.zeros_like(h)
    momentum_coeff = 0.9
    learning_rate = 0.1
    
    # Main optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        if iter_count % 10 == 0:
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if best_c5 <= 0.3807:
                break
        
        # Compute gradient
        grad = compute_gradient_active_shift(h, dx, n_points)
        
        # Update momentum
        momentum = momentum_coeff * momentum + (1 - momentum_coeff) * grad
        
        # Line search
        h, _ = line_search(h, momentum, dx)
        
        # Update learning rate based on improvement
        # Momentum already incorporates spatial gradient, reducing local dependence
        # Do some refinement at intervals
        
        # Occasionally restart with something better
        if iter_count % 50 == 0:
            # Do local gradient descent step to guide momentum towards good directions
            h = project_to_feasible(h, n_points/2)
            grad = compute_gradient_active_shift(h, dx, n_points)

            momentum = momentum_coeff * momentum + (1 - momentum_coeff) * grad
            h, _ = line_search(h, momentum, dx)
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
