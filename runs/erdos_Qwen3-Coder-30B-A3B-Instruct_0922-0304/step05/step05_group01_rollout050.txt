import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 20
    elite_fraction = 0.25
    mutation_strength = 0.01
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def is_feasible(h):
        return np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) - n_points/2) < 1e-10

    def repair_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def mutate(h, strength):
        mutated = h + np.random.normal(0, strength, h.shape)
        return repair_feasible(mutated)

    def select_elite(population, fraction):
        fitnesses = [compute_c5(ind) for ind in population]
        elite_indices = np.argsort(fitnesses)[:int(len(population) * fraction)]
        return [population[i] for i in elite_indices]

    # Initialize population
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if not is_feasible(h0):
            h0 = repair_feasible(h0)
    else:
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 = repair_feasible(h0)

    population = [h0]
    for _ in range(population_size - 1):
        population.append(mutate(h0, mutation_strength))

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    no_improvement_count = 0
    max_no_improvement = 50
    
    max_generations = 1000
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = [compute_c5(ind) for ind in population]
        
        # Update best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Reduce mutation strength if no improvement
        if no_improvement_count > max_no_improvement:
            mutation_strength *= 0.5
            no_improvement_count = 0
            
        # Select elite
        elite = select_elite(population, elite_fraction)
        
        # Generate new population
        new_population = []
        for individual in elite:
            for _ in range(2):  # Two offspring per elite
                new_individual = mutate(individual, mutation_strength)
                new_population.append(new_individual)
                
        # Keep some elite individuals
        for i in range(min(len(elite), len(new_population) // 2)):
            new_population[i] = elite[i].copy()
            
        population = new_population
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
