import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with existing construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create base function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 10000
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current C5
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Identify top contributing lags
        overlap = correlate(h, 1 - h, mode="full")
        lag_indices = np.argsort(np.abs(overlap))[::-1][:15]
        
        # Reconstruct effective subproblem indices
        causal_indices = []
        for lag in lag_indices:
            if lag < len(overlap) // 2:
                causal_indices.append(lag)
            else:
                causal_indices.append(lag - len(overlap) + 1)
                
        # Local optimization around top lags
        def local_objective(h_local):
            # Compute overlap for top lags only
            h_extended = np.concatenate([h_local, np.zeros(n_points-1)])
            over = correlate(h_local, 1 - h_local, mode="full")
            
            # Use historical information to build objective
            max_val = 0
            for i, lag in enumerate(lag_indices):
                if lag <= len(overlap) // 2:
                    val = over[lag]
                else:
                    val = over[lag - len(overlap) + 1]
                max_val = max(max_val, val)
            
            return max_val * dx

        # Use SLSQP for local refinement
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
        
        # Perturb solution to avoid stagnation
        h_perturbed = h + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = normalize_h(h_perturbed, target_sum)
        
        result = minimize(local_objective, h_perturbed, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
        if result.success:
            h = result.x.copy()
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
        else:
            # Fall back to random perturbation
            h = h + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
