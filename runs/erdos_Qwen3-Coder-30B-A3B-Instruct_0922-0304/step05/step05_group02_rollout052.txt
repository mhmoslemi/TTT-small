```python
import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have sum equal to target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)

    # Best solution tracking
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient descent with projection
    learning_rate = 0.01
    max_iter = 1000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute all overlaps
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        
        # Gradient: for the k that maximizes overlap
        # We compute grad = (1 - h_shifted) * dx where shift = k
        # The gradient for each position j is (1 - h[j-k]) if k >= 0 or (1 - h[j+k]) if k < 0
        # But since it's correlation, we're looking at h[i] * (1 - h[i+k])
        # So we have ∇[h[i]] = (1 - h[i+k]) for all positions
        
        # Create shifted h to compute the gradient more directly
        grad = np.zeros(n_points)
        for i in range(n_points):
            # For correlation at shift k, we consider h[i] * (1 - h[i+k])
            # The gradient of h[i] * (1 - h[i+k]) w.r.t h[i] is (1 - h[i+k])
            # Thus, we propagate the contribution of h[i] to h[i+k] in the gradient
            if i + max_k - (n_points - 1) < n_points:
                idx = i + max_k - (n_points - 1)
                if 0 <= idx < n_points:
                    grad[i] = 1 - h[idx]
        
        # Update
        h = h - learning_rate * grad
        # Project to feasible set: [0,1] and sum = n_points/2
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Check and update best solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1
        
    return (best_h.tolist(), best_c5, n_points)
```
