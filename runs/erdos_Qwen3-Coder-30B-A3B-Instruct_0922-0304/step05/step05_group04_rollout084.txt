import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Create swarm
    particles = [h.copy() for _ in range(swarm_size)]
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(swarm_size)]
    
    # Best positions
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [float('inf')] * swarm_size
    global_best = h.copy()
    global_best_score = float('inf')
    
    # PSO parameters
    w_max = 0.9
    w_min = 0.4
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.1
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_max - (w_max - w_min) * (time.time() - start_time) / budget_s
        
        for i in range(swarm_size):
            # Evaluate current particle
            current_score = compute_c5(particles[i])
            
            # Update personal best
            if current_score < personal_best_scores[i]:
                personal_best_scores[i] = current_score
                personal_best[i] = particles[i].copy()
            
            # Update global best
            if current_score < global_best_score:
                global_best_score = current_score
                global_best = particles[i].copy()
                
            # Early exit
            if global_best_score <= 0.3807:
                break
                
            # Update velocity
            r1, r2 = np.random.random(2)
            new_vel = (w * velocities[i] + 
                      c1 * r1 * (personal_best[i] - particles[i]) + 
                      c2 * r2 * (global_best - particles[i]))
            
            # Clip velocity
            new_vel = np.clip(new_vel, -max_vel, max_vel)
            
            # Update position
            new_pos = particles[i] + new_vel
            
            # Apply bounds
            new_pos = np.clip(new_pos, 0, 1)
            
            # Renormalize
            new_pos = new_pos * (n_points / 2) / np.sum(new_pos)
            
            # Update particle
            particles[i] = new_pos
            velocities[i] = new_vel
            
        if global_best_score <= 0.3807:
            break
    
    # Return final result
    final_c5 = compute_c5(global_best)
    return (global_best.tolist(), final_c5, n_points)
