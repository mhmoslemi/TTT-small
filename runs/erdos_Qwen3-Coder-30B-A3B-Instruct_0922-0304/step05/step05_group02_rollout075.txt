```python
import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have sum equal to target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a reasonable starting point
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)
        
        # Add some noise
        noise = np.random.normal(0, 0.05, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)

    # Track best solution
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Gradient-based optimization
    max_iter = 10000
    lr = 0.01
    
    for i in range(max_iter):
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
            
        # Get current C5 value
        current_c5 = compute_c5(h, dx)
        
        # Update best if current is better
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early stopping if target is met
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient direction
        # Find the shift that produces maximum overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(overlap)
        
        # Compute gradient analytically for the max shift
        # The gradient of score with respect to h for shift k is (1 - h_{k+1})
        # But we're using full convolution so we derive it properly
        # Gradient is (1 - h_shifted)
        h_shifted = np.roll(1 - h, max_shift_idx - n_points + 1)  
        grad = h_shifted[:]
        
        # Update h using gradient descent step
        h = h - lr * grad
        
        # Project to feasible region: [0,1] and sum to target_sum
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Adaptive learning rate
        lr = max(0.001, lr * 0.9999)
        
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
```
