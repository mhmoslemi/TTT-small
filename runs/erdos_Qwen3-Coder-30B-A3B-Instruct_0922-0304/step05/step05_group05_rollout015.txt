from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        # Rescale to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: 3-region piecewise construction
        h = np.ones(n_points)
        mid = n_points // 2
        h[mid:] = 0
        # Add some randomness for better exploration
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        # Compute correlation and find max
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def swap_regions(h_copy, i1, i2, size):
        # Ensure no overlap between regions
        if i2 <= i1 + size or i1 <= i2 + size:
            # Overlapping, pick non-overlapping positions
            return h_copy.copy(), False
        
        h_new = h_copy.copy()
        
        # Swap regions
        tmp = h_new[i1:i1+size].copy()
        h_new[i1:i1+size] = h_new[i2:i2+size]
        h_new[i2:i2+size] = tmp
        
        # Rescale to keep sum constraint
        old_sum = np.sum(h_new)
        h_new = h_new * (n_points / 2) / old_sum
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        return h_new, True

    # Core optimization loop
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    
    # Tweakable parameters
    n_swaps = 1000
    swap_size = 5
    temperature = 0.1
    
    # Simulated annealing with swaps
    for i in range(n_swaps):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Choose random locations for swapping
        i1 = np.random.randint(0, n_points - swap_size)
        i2 = np.random.randint(0, n_points - swap_size)
        
        h_new, success = swap_regions(h_best, i1, i2, swap_size)
        
        if not success:
            continue
            
        c5_new = compute_c5(h_new)
        
        # Accept or reject based on simulated annealing
        if c5_new < c5_best or np.random.rand() < np.exp(-(c5_new - c5_best) / temperature):
            h_best = h_new
            c5_best = c5_new
            
            if c5_best <= 0.3807:
                break
                
        # Cool down temperature
        temperature *= 0.999

    # Final polishing with SLSQP
    from scipy.optimize import minimize
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]

    try:
        result = minimize(
            compute_c5,
            h_best,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            h_best = result.x
            h_best = np.clip(h_best, 0, 1)
            h_best = h_best * (n_points / 2) / np.sum(h_best)
            c5_best = compute_c5(h_best)
    except:
        pass

    return (h_best.tolist(), c5_best, n_points)
