import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from scipy.stats import qmc

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
        
    # Generate diverse initial points using Halton sequence
    sampler = qmc.Halton(d=n_points, seed=seed)
    points = sampler.random(n=50)
    initial_points = []
    for i in range(50):
        h_init = points[i] * 2  # scale to [0,2]
        h_init = normalize_h(h_init, target_sum)
        initial_points.append(h_init)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Add the provided initialization as the first point
    initial_points[0] = h.copy()
    
    # Multi-start optimization
    for i, h_start in enumerate(initial_points):
        # Local optimization with SLSQP for each start
        h_opt = h_start.copy()
        
        def objective(h):
            return compute_c5(h)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        try:
            result = minimize(
                objective,
                h_opt,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            if result.success:
                h_new = result.x
                h_new = normalize_h(h_new, target_sum)
                c5 = compute_c5(h_new)
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                    
        except:
            pass
            
        # Early exit if we've found a good enough solution
        if best_c5 <= 0.3807:
            break
        
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation of best solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
