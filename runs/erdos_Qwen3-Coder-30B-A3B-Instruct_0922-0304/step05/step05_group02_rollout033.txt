import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint while maintaining bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, k_max, dx):
    """Compute gradient for the k_max shift."""
    n = len(h)
    grad = np.zeros(n)
    # The gradient with respect to h_k for shift k_max is 1 - h_{k_max-k} if k_max >= k, else 0
    # But since we compute cross-correlation with (1-h), gradient at position i is the sum of 1-h at positions where h contributes
    # The jacobian for h[i] of (h * (1-h))[k_max] = sum over j of 1 if j-i=k_max else 0, i.e. 1 if i = j-k_max, 0 otherwise.
    # So for k_max we compute: sum_j h[j] * delta_{k_max, j-i} * (-1) = - h[i-k_max] if i>=k_max, else 0
    # But this contradicts the actual thing - we compute: (h * (1-h))[k] = sum_j h[j] * (1 - h[j+k])
    # This makes the computation complicated, so let's just compute the full gradient wrt each element from the derivative
    # for the right shift k_max: for each j, we have (h * (1-h))[k] = h[j] * (1-h[j+k]) 
    # The derivative w.r.t h[i] for the shift k:
    # - if i = j, term = (1-h[j+k])
    # - if i = j+k, term = -h[j]
    # - otherwise 0
    
    # Actually more directly: we know that the gradient of h[k] w.r.t (h * (1-h))[k_max] is 1 - h[k_max-k] if k_max >= k, else 0,
    # But for the cross-correlation, it's clearer to denote shifting convention: 
    # (h * (1-h))[k_max] = sum_j h[j] * (1 - h[j+k_max]) = sum_j h[j] - h[j] * h[j+k_max]
    # So the gradient w.r.t h[m] is:
    # - If m = j for some j, then d/dh[m] (sum_j h[j] - h[j] * h[j+k_max]) = 1 - h[m+k_max] 
    # - If m = j+k_max for some j, then += -h[j] = -h[m-k_max]
    # - Otherwise 0
    
    # Let's compute it this way  
    for i in range(n):
        if i < n - k_max:
            grad[i] -= (1 - h[i + k_max])
        if i >= k_max:
            grad[i] -= h[i - k_max]
    return grad * dx

def line_search(h, grad, k_max, dx, step_size, max_iter=10):
    """Perform line search to find the best step size."""
    # Objective value with one step of gradient
    initial_c5 = compute_c5(h, dx)
    for _ in range(max_iter):
        h_new = project_h(h - step_size * grad, len(h) / 2)
        if compute_c5(h_new, dx) < initial_c5 * 0.99999:
            return step_size
        step_size *= 0.5
    return step_size

    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from available construction or create new one
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
    else:
        # Construct initial solution
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        
    h = project_h(h, n_points / 2)
    
    # Optimization parameters
    learning_rate = 0.01
    momentum = 0.99
    prev_grad = np.zeros(n_points)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    no_improvement = 0
    
    # Main optimization loop
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement < 50:
        # Compute all correlations
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        max_val = corr[k_max]
        
        # Compute gradient
        grad = compute_gradient(h, k_max, dx)
        
        # Apply momentum
        grad_momentum = momentum * prev_grad + (1 - momentum) * grad
        
        # Line search
        lr = line_search(h, grad_momentum, k_max, dx, learning_rate)
        
        # Update
        h_new = project_h(h - lr * grad_momentum, n_points / 2)
        
        # Update momentum
        prev_grad = grad_momentum
        
        # Evaluate
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement = 0
        else:
            no_improvement += 1
            
        if c5_new < 0.3807:
            break
            
        h = h_new
        iteration += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
