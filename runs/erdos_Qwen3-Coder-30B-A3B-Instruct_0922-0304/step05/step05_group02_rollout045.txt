```python
import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum(h) = target_sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_feasible(h, target_sum)
    else:
        # Start with a simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * target_sum / np.sum(h)

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    # Early exit if target already achieved
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    iteration = 0
    perturb_freq = 50
    noise_level = 0.1
    
    while time.time() - start_time < budget_s - 1:
        # Compute all cross-correlations
        corr_full = correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr_full) * dx
        
        # If we've improved, update best
        if max_corr < best_c5:
            best_c5 = max_corr
            best_h = h.copy()
            
            if best_c5 <= 0.3807:
                break
        
        # Solve convex relaxation to improve
        h_var = Variable(n_points)
        C = Variable()
        
        # Constraints: each shift
        constraints = [C >= corr_full[k] * dx for k in range(len(corr_full))]
        constraints.append(sum(h_var) == target_sum)
        constraints.append(0 <= h_var)
        constraints.append(h_var <= 1)
        
        # Objective
        prob = Problem(Minimize(C), constraints)
        try:
            prob.solve(solver=OSQP, eps_abs=1e-5, eps_rel=1e-5, verbose=False)
            new_h = h_var.value
            if new_h is not None:
                h = new_h.copy()
                h = np.clip(h, 0, 1)
                h = project_feasible(h, target_sum)
        except:
            # Fallback to gradient-like updates
            pass
        
        # Apply perturbation periodically
        if iteration % perturb_freq == 0 and iteration > 0:
            noise = np.random.normal(0, noise_level, n_points)
            h = h + noise
            h = project_feasible(h, target_sum)
            noise_level *= 0.9
        
        iteration += 1

    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
