import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Repair if necessary
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-resolution approach: coarse then fine
    # Coarse resolution: 35 points
    n_coarse = 35
    dx_coarse = 2.0 / n_coarse
    
    # Initialize coarse h
    if 'initial_h_values' in globals():
        h_coarse = np.array(initial_h_values)
        h_coarse = np.clip(h_coarse, 0, 1)
        h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    else:
        h_coarse = np.ones(n_coarse)
        h_coarse[n_coarse//2:] = 0
        h_coarse += np.random.normal(0, 0.05, n_coarse)
        h_coarse = np.clip(h_coarse, 0, 1)
        h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)

    def compute_c5_coarse(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx_coarse

    def constraint_sum_coarse(h):
        return np.sum(h) - n_coarse / 2

    constraints_coarse = [{'type': 'eq', 'fun': constraint_sum_coarse}]
    bounds_coarse = [(0, 1) for _ in range(n_coarse)]

    # Coarse optimization
    max_iter_coarse = 50
    for iteration in range(max_iter_coarse):
        if time.time() - start_time > budget_s * 0.2:
            break
            
        current_c5 = compute_c5_coarse(h_coarse)
        if current_c5 < best_c5:
            best_c5 = current_c5
            # Upsample to fine grid for best_h
            h_fine_upsampled = np.repeat(h_coarse, 2)
            best_h = h_fine_upsampled * (n_points / 2) / np.sum(h_fine_upsampled)
            
        result = minimize(
            compute_c5_coarse, 
            h_coarse, 
            method='SLSQP', 
            bounds=bounds_coarse, 
            constraints=constraints_coarse,
            options={'maxiter': 10, 'ftol': 1e-6}
        )
        
        if result.success:
            h_coarse = result.x
            h_coarse = np.clip(h_coarse, 0, 1)
            h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
        else:
            # Perturb if optimization fails
            h_coarse += np.random.normal(0, 0.01, n_coarse)
            h_coarse = np.clip(h_coarse, 0, 1)
            h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)

    # Now refine at fine scale with the upsampled solution
    h_fine = np.repeat(h_coarse, 2)
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)

    # Fine optimization
    max_iter_fine = 200
    for iteration in range(max_iter_fine):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5(h_fine)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_fine.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        result = minimize(
            compute_c5, 
            h_fine, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 5, 'ftol': 1e-8}
        )
        
        if result.success:
            h_fine = result.x
            h_fine = np.clip(h_fine, 0, 1)
            h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        else:
            # Perturb if optimization fails
            h_fine += np.random.normal(0, 0.005, n_points)
            h_fine = np.clip(h_fine, 0, 1)
            h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
