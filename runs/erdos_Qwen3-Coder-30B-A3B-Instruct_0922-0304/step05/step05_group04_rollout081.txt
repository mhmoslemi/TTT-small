import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Algorithm parameters
    n_points = 70
    swarm_size = 50
    max_iterations = 10000
    omega_max = 0.9
    omega_min = 0.4
    phi1 = 2.0
    phi2 = 2.0
    velocity_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Create swarm with diverse starting positions
    particles.append(h.copy())
    velocities.append(np.random.uniform(-0.01, 0.01, n_points))
    
    # Add random particles
    for _ in range(swarm_size - 1):
        h_new = np.random.rand(n_points)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        particles.append(h_new)
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
    
    # Track personal bests
    personal_bests = [p.copy() for p in particles]
    personal_best_scores = [float('inf')] * swarm_size
    
    # Global best tracking
    global_best = particles[0].copy()
    global_best_score = float('inf')
    
    # Main loop
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        
        # Update inertia weight
        omega = omega_max - (omega_max - omega_min) * (time.time() - start_time) / budget_s
        
        # Evaluate all particles
        for i in range(swarm_size):
            h = particles[i]
            
            # Compute correlation and max overlap
            overlap = correlate(h, 1 - h, mode="full")
            max_overlap = np.max(overlap) * (2.0 / n_points)
            
            # Update personal best
            if max_overlap < personal_best_scores[i]:
                personal_best_scores[i] = max_overlap
                personal_bests[i] = h.copy()
            
            # Update global best
            if max_overlap < global_best_score:
                global_best_score = max_overlap
                global_best = h.copy()
                
                # Early termination if target achieved
                if global_best_score <= 0.3807:
                    break
        
        # Update particles
        for i in range(swarm_size):
            h = particles[i]
            v = velocities[i]
            
            # Compute random factors
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            # Update velocity
            v = (omega * v 
                 + phi1 * r1 * (personal_bests[i] - h) 
                 + phi2 * r2 * (global_best - h))
            
            # Apply velocity limits
            v = np.clip(v, -velocity_max, velocity_max)
            
            # Update position
            h = h + v
            
            # Ensure bounds
            h = np.clip(h, 0, 1)
            
            # Renormalize to sum to n/2
            h = h * (n_points / 2) / np.sum(h)
            
            # Store updated values
            particles[i] = h
            velocities[i] = v
            
            # Check if solution is valid
            if abs(np.sum(h) - n_points / 2) > 1e-6:
                # Renormalize again if needed
                h = h * (n_points / 2) / np.sum(h)
                particles[i] = h
        
        # Early exit if target reached
        if global_best_score <= 0.3807:
            break
    
    # Final evaluation of global best
    final_overlap = correlate(global_best, 1 - global_best, mode="full")
    final_c5 = np.max(final_overlap) * (2.0 / n_points)
    
    # Ensure result is valid
    final_h = np.clip(global_best, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    
    return (final_h.tolist(), float(final_c5), n_points)
