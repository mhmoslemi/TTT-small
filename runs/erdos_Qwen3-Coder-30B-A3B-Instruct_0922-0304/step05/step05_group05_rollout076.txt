import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        # Compute overlap integral using correlation
        overlap = correlate(h, 1 - h, mode="full")
        # The maximum corresponds to the peak of the correlation
        return np.max(overlap) * dx

    def perturb_h(h, temp):
        # Create a perturbed version of h with Gaussian noise scaled by temperature
        h_new = h + np.random.normal(0, temp * 0.1, n_points)
        h_new = np.clip(h_new, 0, 1)
        # Rescale to meet constraint sum(h) = n_points / 2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new

    def acceptance_probability(old_c5, new_c5, temp):
        if new_c5 < old_c5:
            return 1.0
        else:
            # Accept worse solution with probability based on temperature
            return np.exp(-(new_c5 - old_c5) / temp)

    # Simulated Annealing parameters
    initial_temp = 1.0
    cooling_rate = 0.999
    min_temp = 1e-8
    
    # Initialize
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    best_h = current_h.copy()
    best_c5 = current_c5
    
    temp = initial_temp
    
    iterations = 0
    no_improvement_count = 0
    max_no_improvement = 2000
    
    while time.time() - start_time < budget_s - 1 and temp > min_temp:
        # Generate new candidate solution
        new_h = perturb_h(current_h, temp)
        new_c5 = compute_c5(new_h)
        
        # Accept or reject the new solution
        if acceptance_probability(current_c5, new_c5, temp) > np.random.rand():
            current_h = new_h
            current_c5 = new_c5
            
            # Update best solution if improved
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Cooling schedule
        temp *= cooling_rate
        
        iterations += 1
        
        # Early stopping if no improvement for a while
        if no_improvement_count > max_no_improvement:
            break

    # Final validation
    final_c5 = compute_c5(best_h)
    
    # Ensure final solution meets constraints exactly
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
