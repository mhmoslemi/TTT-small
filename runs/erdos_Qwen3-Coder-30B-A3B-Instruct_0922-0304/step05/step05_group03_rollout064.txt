import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradients(h, dx):
    """Compute gradients for each h[i] with respect to C5."""
    # Compute the full correlation
    corr = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    
    # Compute gradient using finite differences for each position
    grad = np.zeros_like(h)
    eps = 1e-6
    
    for i in range(len(h)):
        h_plus = h.copy()
        h_minus = h.copy()
        h_plus[i] += eps
        h_minus[i] -= eps
        
        h_plus = normalize_h(h_plus, np.sum(h))
        h_minus = normalize_h(h_minus, np.sum(h))
        
        corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
        corr_minus = correlate(h_minus, 1 - h_minus, mode="full")
        
        grad[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2 * eps)
    
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Set initial learning rate
    lr = 0.1
    patience = 0
    patience_limit = 10
    lr_decay = 0.5
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        current_c5 = evaluate_c5(h, dx)
        
        # Track best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if best_c5 <= 0.3807:
            break
            
        if patience > patience_limit:
            lr *= lr_decay
            patience = 0
        
        # Compute gradient
        grad = compute_gradients(h, dx)
        
        # Update with gradient descent
        h_new = h - lr * grad
        h_new = normalize_h(h_new, target_sum)
        
        # Coordinate-wise local search
        for i in range(n_points):
            old_val = h[i]
            h[i] = max(0, min(1, old_val + np.random.normal(0, 0.01)))
            h = normalize_h(h, target_sum)
            new_c5 = evaluate_c5(h, dx)
            if new_c5 >= current_c5:
                h[i] = old_val
                h = normalize_h(h, target_sum)
        
        # Apply gradient update
        h = h_new
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
