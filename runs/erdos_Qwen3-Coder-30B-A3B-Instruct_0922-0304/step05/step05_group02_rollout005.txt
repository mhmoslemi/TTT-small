import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to maintain sum constraint and box constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def line_search(h, grad, dx, max_alpha=1.0, tolerance=1e-4):
    """Perform line search along gradient direction."""
    alpha = max_alpha
    current_c5 = compute_c5(h, dx)
    while alpha > 1e-8:
        h_new = project_to_feasible(h - alpha * grad, len(h) / 2)
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < current_c5 - tolerance * alpha * np.dot(grad, grad):
            return h_new, alpha
        alpha *= 0.5
    return h, alpha

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
    
    h = project_to_feasible(h, n_points / 2)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Proximal gradient parameters
    momentum = 0.95
    step_size = 0.05
    prev_h = h.copy()
    prev_grad = np.zeros(n_points)
    
    iteration = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Compute all correlations
        overlap = correlate(h, 1 - h, mode="full")
        
        # Find index k with max overlap
        k_max = np.argmax(overlap)
        shift = k_max - (n_points - 1)
        
        # Compute gradient (simplified, using just the correlation at k_max)
        grad = np.zeros(n_points)
        if shift >= 0:
            grad[shift:] = 1 - h[:-shift]
        else:
            grad[:shift] = 1 - h[-shift:]
        
        # Apply momentum
        desc_dir = momentum * prev_grad + (1 - momentum) * grad
        
        # Line search
        h_new, step_size = line_search(h, desc_dir, dx, step_size)
        
        # Update
        h = h_new
        prev_grad = desc_dir
        prev_h = h.copy()
        
        # Check improvement
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        if no_improvement_count > 50:
            step_size *= 0.8
            no_improvement_count = 0
    
    return (best_h.tolist(), best_c5, n_points)
