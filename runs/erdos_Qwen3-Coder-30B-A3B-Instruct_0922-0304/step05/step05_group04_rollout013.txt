import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: Alternating pattern
        h = np.ones(n_points)
        h[::2] = 0
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use a gradient-based optimization approach with proper constraints
    max_iter = 10000
    iteration = 0
    lr = 0.01  # learning rate
    momentum = 0.9  # momentum
    velocity = np.zeros(n_points)
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap) * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient approximations
        grad = np.zeros(n_points)
        eps_val = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            
            h_plus[i] += eps_val
            h_minus[i] -= eps_val
            
            # Clamp values
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            
            # Renormalize
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps_val) * dx

        # Gradient descent update with momentum
        velocity = momentum * velocity - lr * grad
        h = h + velocity
        
        # Project to feasible set [0,1] and renormalize
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Small random perturbation to escape local minima occasionally
        if iteration % 100 == 0:
            noise = np.random.normal(0, 0.001, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
