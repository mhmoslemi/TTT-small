import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Add initial particle
    particles.append(h.copy())
    velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Add random particles
    for _ in range(swarm_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # PSO parameters
    omega_start = 0.9
    omega_end = 0.4
    phi1 = 2.0
    phi2 = 2.0
    vel_max = 0.1
    
    # Track best positions
    particle_best = [p.copy() for p in particles]
    particle_best_scores = [float('inf')] * swarm_size
    global_best = particles[0].copy()
    global_best_score = float('inf')
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        omega = omega_start - (omega_start - omega_end) * (time.time() - start_time) / budget_s
        
        for i in range(swarm_size):
            # Evaluate current particle
            overlap = correlate(particles[i], 1 - particles[i], mode="full")
            score = np.max(overlap) * dx
            
            # Update particle best
            if score < particle_best_scores[i]:
                particle_best_scores[i] = score
                particle_best[i] = particles[i].copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best = particles[i].copy()
                
                # Early exit if target achieved
                if global_best_score <= 0.3807:
                    break
            
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            velocities[i] = (omega * velocities[i] + 
                           phi1 * r1 * (particle_best[i] - particles[i]) + 
                           phi2 * r2 * (global_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -vel_max, vel_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Clamp values to [0,1]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Renormalize to sum to n_points/2
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Final evaluation
    overlap_final = correlate(global_best, 1 - global_best, mode="full")
    final_c5 = np.max(overlap_final) * dx
    
    return (global_best.tolist(), final_c5, n_points)
