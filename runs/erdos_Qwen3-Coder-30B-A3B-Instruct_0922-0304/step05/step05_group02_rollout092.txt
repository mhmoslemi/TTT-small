import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: clip to [0,1] and scale to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient descent parameters
    eta = 0.1
    decay_factor = 0.99
    noise_std = 0.01
    noise_freq = 100
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute cross-correlation for all shifts
        full_correlation = correlate(h, 1 - h, mode="full")
        
        # Find the shift with maximum overlap
        max_idx = np.argmax(full_correlation)
        max_val = full_correlation[max_idx]
        
        # Identify corresponding k value
        k = max_idx - (n_points - 1)  # k ranges from -(n_points-1) to (n_points-1)
        
        # Compute the gradient for this k
        # h[i] contributes to (h * (1 - h))(i + k) if k >= 0,
        # and (h * (1 - h))(i - k) if k < 0
        # we build (1-h) shifted by k
        if k >= 0:
            shifted_h = np.concatenate([np.zeros(k), 1 - h[:-k]])
        else:
            shifted_h = np.concatenate([1 - h[abs(k):], np.zeros(abs(k))])
        
        # The gradient of C w.r.t h is 1 - h_shifted (scaled by dx)
        grad = 1 - shifted_h
        
        # Perform gradient step
        h = h - eta * grad * dx
        
        # Project to feasible region
        h = project_h(h, target_sum)
        
        # Update best solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Adaptive learning rate
        if iteration % 10 == 0:
            eta = max(0.001, eta * decay_factor)
        
        # Add noise periodically to escape local minima
        if iteration % noise_freq == 0:
            h += np.random.normal(0, noise_std, n_points)
            h = project_h(h, target_sum)
        
        iteration += 1
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
