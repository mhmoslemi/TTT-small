import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        # Initialize with a piecewise constant function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        # Add some noise to break symmetry
        h += np.random.normal(0, 0.05, n_points)
        h = project_h(h, target_sum)
    
    # Parameters
    eta = 0.1  # learning rate
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute all correlations
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Update best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient: for shift k that maximizes overlap
        # The gradient w.r.t. h for overlap at k is (1 - h_shifted) * dx
        # But we compute the derivative with respect to each element of h
        # Letting shift = k - (n_points-1) = k - 69 (since we start from 0)
        shift = max_k - (n_points - 1)
        
        # Create shifted version of 1-h  
        one_minus_h_shifted = np.zeros(n_points)
        if shift >= 0:
            one_minus_h_shifted[shift:] = 1 - h[:-shift]
        else:
            one_minus_h_shifted[:shift] = 1 - h[-shift:]
        
        # Gradient is one_minus_h_shifted * dx
        grad = one_minus_h_shifted * dx
        
        # Update
        h = h - eta * grad
        
        # Project back to feasible set
        h = project_h(h, target_sum)
        
        # Adapt learning rate based on improvement
        if iteration % 10 == 0:
            new_c5 = compute_c5(h, dx)
            if new_c5 < max_overlap:
                eta *= 1.05  # Increase learning rate
            else:
                eta *= 0.9   # Decrease learning rate
                
        # Add small noise every 100 iterations for escape
        if iteration % 100 == 0:
            noise = np.random.normal(0, 0.001, n_points)
            h = h + noise
            h = project_h(h, target_sum)
        
        iteration += 1
    
    # Final computation of best C5
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
