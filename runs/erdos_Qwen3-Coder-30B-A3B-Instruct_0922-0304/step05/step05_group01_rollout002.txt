import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h):
    """Compute the C5 value for a given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * (2.0 / len(h))

def mutate(h, mutation_std):
    """Mutate h with Gaussian noise and re-normalize."""
    mutated = h + np.random.normal(0, mutation_std, h.shape)
    mutated = np.clip(mutated, 0, 1)
    sum_h = np.sum(mutated)
    if sum_h > 0:
        mutated = mutated * (len(h) / 2) / sum_h
    return mutated

def evaluate_fitness(population):
    """Evaluate fitness for all individuals in population."""
    return [compute_c5(ind) for ind in population]

def select_elite(population, fitness, elite_frac=0.25):
    """Select elite individuals based on fitness."""
    indices = np.argsort(fitness)
    elite_indices = indices[:int(len(population) * elite_frac)]
    return [population[i] for i in elite_indices]

def update_best(population, fitness, best_h, best_c5):
    """Update the best solution found."""
    min_idx = np.argmin(fitness)
    if fitness[min_idx] < best_c5:
        best_h[:] = population[min_idx]
        best_c5 = fitness[min_idx]
    return best_h, best_c5
