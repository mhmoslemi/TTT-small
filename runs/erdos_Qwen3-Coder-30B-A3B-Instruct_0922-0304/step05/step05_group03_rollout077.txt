import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Load from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
    
    # Generate diverse initial guesses using a low-discrepancy sequence
    # Using Halton-like sequence for a more uniform distribution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Mini-Batch Gradient Ascent with different strategies
    num_starts = 50
    for i in range(num_starts):
        # Generate a new start point using pseudo-random sequence with Halton-like approach
        h_start = np.random.rand(n_points)
        h_start = normalize_h(h_start, target_sum)
        
        def objective(h):
            return compute_c5(h)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        # SLSQP optimization
        result = minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        if result.success:
            h_new = result.x
            h_new = normalize_h(h_new, target_sum)
            c5_val = compute_c5(h_new)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_new.copy()
        
        if time.time() - start_time > budget_s - 1:
            break
                
        # Also try a simple local gradient-based search with perturbations
        h_perturbed = h_start + np.random.normal(0, 0.02, n_points)
        h_perturbed = normalize_h(h_perturbed, target_sum)
        
        # Optimization with perturbed start
        result = minimize(
            objective,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        if result.success:
            h_new = result.x
            h_new = normalize_h(h_new, target_sum)
            c5_val = compute_c5(h_new)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_new.copy()
        
        if time.time() - start_time > budget_s - 1:
            break

    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
