import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def halton_sequence(n, base=2):
    """Generate n points of 1D Halton sequence."""
    result = []
    for i in range(n):
        x = 0
        f = 1.0 / base
        while i > 0:
            i, remainder = divmod(i, base)
            x += remainder * f
            f /= base
        result.append(x)
    return np.array(result)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
        
    # Multi-start optimization with Halton sequences
    best_c5 = float('inf')
    best_h = h.copy()
    
    # 50 initial points generated using Halton sequence
    population_size = 50
    halton_samples = halton_sequence(population_size, 2)
    
    # Generate diverse initial guesses
    for i in range(population_size):
        # Map Halton to [0,1] range
        h_init = halton_samples[i] * np.ones(n_points)
        h_init = normalize_h(h_init, target_sum)
        h_init = np.clip(h_init, 0, 1)
        h_init = normalize_h(h_init, target_sum)
        
        # Local optimization using SLSQP 
        def objective(h):
            return compute_c5(h)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        # SLSQP optimization
        try:
            result = minimize(
                objective,
                h_init,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 100, 'ftol': 1e-8}
            )
            
            if result.success:
                h_opt = result.x
                h_opt = normalize_h(h_opt, target_sum)
                c5_val = compute_c5(h_opt)
                
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt.copy()
                    
        except:
            pass
            
        # Early exit if we get close to our target
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final optimization of the best solution found
    def objective(h):
        return compute_c5(h)
        
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    try:
        result = minimize(
            objective,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        
        if result.success:
            best_h = result.x
            best_h = normalize_h(best_h, target_sum)
            best_c5 = compute_c5(best_h)
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
