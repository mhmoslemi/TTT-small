from scipy.signal import correlate

        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Convex optimization approach
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative convex optimization
    max_iter = 200
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Solve relaxed convex problem
        try:
            variables = cp.Variable(n_points)
            constraints = [variables >= 0, variables <= 1, cp.sum(variables) == n_points / 2]
            
            # Create variable for the overlapping term
            # We use a simplified approach where we minimize the maximum cross-correlation
            # Linearize objective: use current gradient at h
            
            # This step typically uses advanced convex modeling, 
            # here we directly compute directly the overlap function
            # For this implementation, we simplify and use gradient-based update
            
            # Small update step in descent direction
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            if best_c5 <= 0.3807:
                break
                
            # Simple gradient-based update: 
            # This is a crude approximation - in practice could compute actual gradients
            # For now, we do a small random perturbation
            perturbation = np.random.normal(0, 0.005, n_points)
            h_new = h + perturbation
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new

                    return compute_c5(h_vec)

            # fallback
            h += np.random.normal(0, 0.001, n_points)

    # Final validation
    final_c5 = compute_c5(best_h)
