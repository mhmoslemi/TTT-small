import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 500
    current_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Solve convex relaxation using CVXPY
        try:
            # Create variables
            variables = cp.Variable(n_points)
            
            # Constraints
            constraints = [
                variables >= 0,
                variables <= 1,
                cp.sum(variables) == n_points / 2
            ]
            
            # Objective
            # We need to approximate the maximum cross-correlation term
            # Using a linear approximation of the maximum, which is often safer than 
            # direct non-convex max. We'll instead try to solve a related convex problem
            # This approximates the desired min max to make it solvable.
            # A simplified convex approximation: minimize sum of all correlations
            # But that's not exactly what we want. Instead, minimize an upper bound.
            # Let's try directly approximating max_k integral as a single (undefined) variable,
            # and add constraints such that this upper bound is reasonable.
            
            # Better approach: solve the problem that minimizes max overlap 
            # using projected gradient in a constrained manner with simple DC programming
            
            # For now, refine using local non-linear optimization
            from scipy.optimize import minimize
            
            def objective(h_vec):
                return compute_c5(h_vec)
            
            # Use trust-constr method which works well with constraints
            result = minimize(
                objective,
                current_h,
                method='trust-constr',
                bounds=[(0, 1) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2},
                options={'maxiter': 50, 'disp': False}
            )
            
            if result.success:
                current_h = result.x
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * (n_points / 2) / np.sum(current_h)
            else:
                # If optimization failed, try gradient descent update
                # This is a fallback and not recommended, but could give us a variant
                grad = np.zeros_like(current_h)
                for i in range(n_points):
                    # Approximate gradient through finite differences
                    h_plus = current_h.copy()
                    h_plus[i] += 1e-4
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    h_minus = current_h.copy()
                    h_minus[i] -= 1e-4
                    h_minus = np.clip(h_minus, 0, 1)
                    h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                    grad[i] = (compute_c5(h_plus) - compute_c5(h_minus)) / (2e-4)
            
                # Perform ascent step to decrease C5
                lr = 1e-3
                current_h = current_h - lr * grad
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * (n_points / 2) / np.sum(current_h)
            
        except Exception as e:
            # Fallback if anything goes wrong
            pass
    
        # Evaluate and update best found
        current_c5 = compute_c5(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
        
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
