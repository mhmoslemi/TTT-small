import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 50
    max_iterations = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
        particles.append(h0.copy())
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
    else:
        h0 = np.ones(n_points) * 0.5
        h0 = h0 * (n_points / 2) / np.sum(h0)
        particles.append(h0.copy())
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
    
    # Initialize other particles
    for i in range(swarm_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
    
    # Track best positions
    personal_best = [p.copy() for p in particles]
    personal_best_values = [float('inf')] * swarm_size
    global_best = particles[0].copy()
    global_best_value = float('inf')
    
    # PSO parameters
    omega_max = 0.9
    omega_min = 0.4
    phi1 = 2.0
    phi2 = 2.0
    max_velocity = 0.1
    
    def compute_c5(h):
        # Compute correlation
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Main loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        omega = omega_max - (omega_max - omega_min) * (time.time() - start_time) / budget_s
        
        for i in range(swarm_size):
            h = particles[i]
            
            # Compute current overlap
            current_c5 = compute_c5(h)
            
            # Update personal best
            if current_c5 < personal_best_values[i]:
                personal_best_values[i] = current_c5
                personal_best[i] = h.copy()
            
            # Update global best
            if current_c5 < global_best_value:
                global_best_value = current_c5
                global_best = h.copy()
                
            # Early termination
            if global_best_value <= 0.3807:
                break
                
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            v = omega * velocities[i] + \
                phi1 * r1 * (personal_best[i] - h) + \
                phi2 * r2 * (global_best - h)
            
            # Clip velocity
            v = np.clip(v, -max_velocity, max_velocity)
            
            # Update position
            h_new = h + v
            
            # Clamp to [0, 1]
            h_new = np.clip(h_new, 0, 1)
            
            # Renormalize to sum to n_points / 2
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Update particle
            particles[i] = h_new
            velocities[i] = v
            
        if global_best_value <= 0.3807:
            break
    
    return (global_best.tolist(), global_best_value, n_points)
