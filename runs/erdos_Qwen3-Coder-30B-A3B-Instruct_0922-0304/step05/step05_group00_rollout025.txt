import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the known good construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default construction - step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Number of main iterations
    max_iter = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Identify most critical lags
        overlap = correlate(h, 1-h, mode="full")
        # Get indices of top 10% of lags (excluding trivial zeros)
        top_lag_indices = np.argsort(overlap)[-max(1, n_points//10):]
        
        # Create a convex optimization problem targeting these lags
        # Use the fact that we can optimize over the slack variables
        
        # For computational efficiency, we use direct projection and local search
        # We'll sample small random perturbations on a few indices
        
        # Sample candidates by small local variations
        candidates = [h.copy()]
        
        # Add some mutations to explore
        for _ in range(5):
            new_h = h.copy()
            # Randomly perturb a few points
            indices = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
            for i in indices:
                new_h[i] += np.random.normal(0, 0.01)
            new_h = np.clip(new_h, 0, 1)
            new_h = normalize_h(new_h, target_sum)
            candidates.append(new_h)
        
        # Evaluate all candidates and keep the best
        for candidate in candidates:
            candidate_c5 = compute_c5(candidate, dx)
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = candidate.copy()
                
            # Early stopping if we've found a good enough solution
            if best_c5 <= 0.3807:
                break
                
        # Update current solution to best found so far
        h = best_h.copy()
        
        # Occasionally apply SQP refinement on the current best
        if iteration % 10 == 0:
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h):
                return compute_c5(h, dx)
                
            result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                            constraints=constraint, options={'maxiter': 20})
            if result.success:
                h = np.clip(result.x, 0, 1)
                h = normalize_h(h, target_sum)
                current_c5 = compute_c5(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    
        # Occasionally also project and re-normalize to ensure feasibility
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
    return (best_h.tolist(), best_c5, n_points)
