import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, k_max, dx, n_points):
    """Compute gradient of overlap at k_max with respect to h."""
    grad = np.zeros(n_points)
    # For correlation h * (1-h) at lag k_max
    # The contribution to overlap at k_max from h[i] is (1-h[i+k_max]) if i+k_max < n_points
    for i in range(n_points):
        if i + k_max < n_points:
            grad[i] += 1.0
    # Normalize by dx since we're computing a derivative
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available

        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    # Gradient descent parameters
    alpha = 0.1
    patience = 0
    patience_limit = 50
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        c5 = evaluate_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > patience_limit:
                alpha *= 0.5
                patience = 0
                if alpha < 1e-6:
                    break
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Find the lag with maximum overlap
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        overlap_max = overlap[k_max]
        
        # Compute the gradient and update
        grad = compute_gradient(h, k_max, dx, n_points)
        h = h - alpha * grad
        
        # Clip and normalize
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
