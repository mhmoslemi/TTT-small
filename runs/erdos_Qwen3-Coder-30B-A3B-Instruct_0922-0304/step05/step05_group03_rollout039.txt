import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, top_lag_indices):
    """Compute gradient of C5 with respect to h using top lag indices."""
    n = len(h)
    grad = np.zeros(n)
    # Compute correlation for all lags
    corr = correlate(h, 1-h, mode="full")
    # Gradient computation via finite differences for top lags
    for lag in top_lag_indices:
        # Compute analytical gradient for fixed lag
        if lag < n:
            # h[i] contributes to overlap at position lag via h[i] * (1 - h[i+lag])
            # We sum the contributions for this lag
            for i in range(n):
                if i + lag < n:
                    # grad wrt h[i] 
                    grad[i] += (1 - 2*h[i] - h[i+lag]) * dx
                else:
                    # lag > n, so we look at the start of second array
                    # In the correlation sense, lag should be treated specially
                    # but for our modeling, we'll approximate using the main region
                    continue
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Parameters
    learning_rate = 0.1
    patience = 100
    patience_counter = 0
    prev_c5 = float('inf')
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter > patience or best_c5 <= 0.3807:
            break

        # Get indices of top lags with highest correlation
        corr = correlate(h, 1-h, mode="full")
        # Get indices to look at for gradient
        top_lags = np.argsort(corr)[-10:]  # top 10 lags by absolute correlation
        top_lag_indices = []
        for lag in top_lags:
            # Map full correlation index back to actual lag
            if lag >= n_points - 1:
                actual_lag = lag - (n_points - 1)
                top_lag_indices.append(actual_lag)
            else:
                top_lag_indices.append(lag)

        # Compute approximate gradient
        gradient = compute_gradient(h, dx, top_lag_indices)
        
        # Gradient descent step
        h_new = h - learning_rate * gradient
        h_new = normalize_h(h_new, target_sum)
        
        # Coordinate-wise local search
        for i in range(n_points):
            h_trial = h.copy()
            # Small change around current value
            perturbation = np.random.uniform(-0.01, 0.01)
            h_trial[i] += perturbation
            h_trial = normalize_h(h_trial, target_sum)
            new_c5 = evaluate_c5(h_trial, dx)
            if new_c5 < current_c5:
                h = h_trial.copy()
        
        # Set h to new value after gradient descent
        h = h_new.copy()
        
        # Adaptive learning rate
        if abs(prev_c5 - current_c5) < 1e-6:
            learning_rate *= 0.5
            if learning_rate < 1e-5:
                learning_rate = 1e-5
        prev_c5 = current_c5
        
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
