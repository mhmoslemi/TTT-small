import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def repair_feasibility(h, target_sum):
    """Repair h to be feasible (clamp and normalize)."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with provided construction or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a simple initial construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Evolution parameters
    population_size = 30
    population = [h.copy()]
    for _ in range(population_size - 1):
        h_mut = h + np.random.normal(0, 0.02, n_points)
        h_mut = repair_feasibility(h_mut, target_sum)
        population.append(h_mut)
    
    # Step size for different search methods
    de_step = 0.5
    gd_step = 0.01
    mutation_rate = 0.2
    step_decay = 0.98
    
    iteration = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate population
        fitness = [evaluate_c5(ind, dx) for ind in population]
        
        # Track best
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_h = population[min_fit_idx].copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Generate new population
        new_population = []
        
        # 1. Keep the best individual (elitism)
        new_population.append(population[min_fit_idx].copy())
        
        # 2. Differential evolution (40%)
        for _ in range(population_size // 2):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[a] + de_step * (population[b] - population[c])
            mutant = repair_feasibility(mutant, target_sum)
            
            # Crossover with current individual
            r = np.random.rand(n_points)
            trial = np.where(r < 0.7, mutant, population[min_fit_idx])
            new_population.append(trial)
        
        # 3. Gradient descent (30%)
        for _ in range(population_size // 3):
            idx = np.random.randint(population_size)
            current_h = population[idx].copy()
            
            def objective(h):
                return evaluate_c5(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                result = minimize(objective, current_h, method='SLSQP', 
                                bounds=bounds, constraints=constraint,
                                options={'maxiter': 30, 'ftol': 1e-6})
                if result.success:
                    new_h = result.x
                    new_h = np.clip(new_h, 0, 1)
                    new_h = normalize_h(new_h, target_sum)
                    new_population.append(new_h)
                else:
                    h_m = current_h + np.random.normal(0, 0.01, n_points)
                    new_population.append(repair_feasibility(h_m, target_sum))
            except:
                h_m = current_h + np.random.normal(0, 0.01, n_points)
                new_population.append(repair_feasibility(h_m, target_sum))
        
        # 4. Random mutation (20%)
        for _ in range(population_size // 5):
            idx = np.random.randint(population_size)
            h_m = population[idx] + np.random.normal(0, 0.02, n_points)
            new_population.append(repair_feasibility(h_m, target_sum))
        
        # 5. Mutation of best (10%)
        for _ in range(population_size // 10):
            h_m = best_h + np.random.normal(0, 0.01, n_points)
            new_population.append(repair_feasibility(h_m, target_sum))
        
        # Ensure we have enough individuals
        while len(new_population) < population_size:
            new_h = np.random.rand(n_points)
            new_h = repair_feasibility(new_h, target_sum)
            new_population.append(new_h)
        
        population = new_population[:population_size]
        
        # Adjust step sizes
        de_step *= step_decay
        gd_step *= step_decay
        
        iteration += 1
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
