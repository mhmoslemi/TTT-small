import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
    h = project_to_feasible(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    max_iter = 1000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        # Compute overlaps for all lags
        overlap = correlate(h, 1 - h, mode="full")
        # Get indices of largest overlaps
        max_indices = np.argsort(overlap)[-10:]  # Top 10 lags
        max_indices = max_indices[max_indices >= n_points-1]  # Only consider lags >= 0

        # Create simple convex relaxation
        A = np.zeros((len(max_indices), n_points))
        b = np.zeros(len(max_indices))
        for i, idx in enumerate(max_indices):
            # Compute overlap at lag (idx - n_points + 1)
            lag = idx - n_points + 1
            # For i-th lag, constraints are: h[i] * (1 - h[i+lag]) term
            # Approximate: linearize around current values
            if lag >= 0:
                for j in range(n_points - lag):
                    A[i, j] = 1 - h[j + lag]
                    A[i, j + lag] = -h[j]
                b[i] = overlap[idx] - np.sum(h[:n_points-lag] * h[lag:]) - np.sum(h[:n_points-lag] * (1 - h[lag:]))
            else:
                lag = abs(lag)
                for j in range(n_points - lag):
                    A[i, j + lag] = 1 - h[j]
                    A[i, j] = -h[j + lag]
                b[i] = overlap[idx] - np.sum(h[lag:] * h[:n_points-lag]) - np.sum(h[lag:] * (1 - h[:n_points-lag]))
        
        # Simplified problem solving
        # Instead of solving the system, do simple gradient descent with mirrored x
        delta = 0.001
        h_new = h.copy()
        for j in range(n_points):
            # Try small perturbation
            h_test = h.copy()
            h_test[j] += delta
            if h_test[j] <= 1:
                c5_new = compute_c5(h_test, dx)
                if c5_new < best_c5:
                    h_new[j] = h_test[j]
                    best_c5 = c5_new
                    best_h = h_test.copy()
                    
        # Project to feasible region
        h = project_to_feasible(h_new, target_sum)
        
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
