import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        np.random.shuffle(h)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def compute_gradient(h, eps=1e-4):
        g = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps) * dx
        return g

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing Phase
    T = 1.0
    alpha = 0.99
    iter_count = 0
    
    while time.time() - start_time < budget_s * 0.7 and iter_count < max_iter:
        iter_count += 1
        # Generate neighbor by swapping two random positions
        i, j = np.random.choice(n_points, 2, replace=False)
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        
        # Renormalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute C5
        c5_new = compute_c5(h_new)
        
        c5_old = compute_c5(h)
        
        # Accept or reject
        if c5_new < c5_old or np.random.rand() < np.exp(-(c5_new - c5_old) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                
        # Cooling
        T *= alpha
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
    # Continuous relaxation with CVXPY
    if time.time() - start_time < budget_s - 1:
        # Use best binary solution to initialize
        h_init = best_h.copy()
        
        # Formulate and solve the convex problem
        h_var = cp.Variable(n_points)
        constraints = [cp.sum(h_var) == n_points / 2,
                      h_var >= 0,
                      h_var <= 1]
        
        # Simple approach: use existing solution and do local optimization
        # Let's perform gradient-based descent with projection
        max_iter_cont = 5000
        step_size = 0.001
        for i in range(max_iter_cont):
            if time.time() - start_time > budget_s - 1:
                break
                
            g = compute_gradient(h)
            h = h - step_size * g
            
            # Projection onto feasible set
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Update best if improved
            c5_current = compute_c5(h)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h.copy()
                
        # Final check
        best_c5 = compute_c5(best_h)

    return (best_h.tolist(), best_c5, n_points)
