import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: piecewise function
        h = np.ones(n_points)
        h[n_points//4:3*n_points//4] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def perturb_h(h):
        h_pert = h + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        return h_pert

    def swap_regions(h, size=5):
        h_new = h.copy()
        i1 = np.random.randint(0, n_points - size)
        i2 = np.random.randint(0, n_points - size)
        while abs(i1 - i2) < size:
            i2 = np.random.randint(0, n_points - size)
        
        # Swap regions
        h_new[i1:i1+size], h_new[i2:i2+size] = h_new[i2:i2+size], h_new[i1:i1+size]
        
        # Rescale to maintain constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new

    best_c5 = float('inf')
    best_h = h.copy()

    # Swap-based local search
    n_swaps = 1000
    swap_size = 5
    temperature = 0.1
    
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    for _ in range(n_swaps):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate neighbor by swapping
        neighbor_h = swap_regions(current_h, swap_size)
        neighbor_c5 = compute_c5(neighbor_h)
        
        # Accept or reject
        if neighbor_c5 < current_c5:
            current_h = neighbor_h
            current_c5 = neighbor_c5
        else:
            # Sometimes accept worse solution with probability
            prob = np.exp(- (neighbor_c5 - current_c5) / temperature)
            if np.random.rand() < prob:
                current_h = neighbor_h
                current_c5 = neighbor_c5
                
        # Track best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Early exit if close to target
        if best_c5 <= 0.3807:
            break

    # Final polishing with local optimization
    from scipy.optimize import minimize
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    try:
        result = minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            best_c5 = compute_c5(best_h)
    except:
        pass

    return (best_h.tolist(), best_c5, n_points)
