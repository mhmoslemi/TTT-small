import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop with different strategies
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Get current best C5
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Strategy 1: Gradient-based local search
        # Compute finite difference gradient approximation
        g = np.zeros(n_points)
        eps_val = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps_val
            h_minus[i] -= eps_val
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps_val) * dx

        # Update with gradient descent
        step_size = 0.01
        new_h = h - step_size * g
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Strategy 2: Random search in feasible region
        # Occasionally add random perturbations
        if np.random.random() < 0.1:
            pert = np.random.normal(0, 0.01, n_points)
            test_h = h + pert
            test_h = np.clip(test_h, 0, 1)
            test_h = test_h * (n_points / 2) / np.sum(test_h)
            test_c5 = compute_c5(test_h)
            if test_c5 < current_c5:
                new_h = test_h
                
        # Update h
        h = new_h
        
        # Sometimes try alternating block initialization
        if np.random.random() < 0.05:
            # Try a structured approach
            block_size = max(1, n_points // 8)
            h = np.zeros(n_points)
            for i in range(0, n_points, 2*block_size):
                h[i:i+block_size] = 1
            h = h * (n_points / 2) / np.sum(h)

    # Final validation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
