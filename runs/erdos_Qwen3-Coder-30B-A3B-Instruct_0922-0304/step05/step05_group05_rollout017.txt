import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Optimization parameters
    n_swaps = 1000
    swap_size = 5
    temperature = 0.1
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()
    
    # Swap-based optimization
    for i in range(n_swaps):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Select two non-overlapping regions
        start1 = np.random.randint(0, n_points - swap_size)
        start2 = np.random.randint(0, n_points - swap_size)
        
        # Ensure they don't overlap
        while (start2 <= start1 + swap_size and start2 + swap_size >= start1):
            start2 = np.random.randint(0, n_points - swap_size)
        
        # Create candidate h
        candidate_h = current_h.copy()
        
        # Swap values
        temp_values = candidate_h[start1:start1+swap_size].copy()
        candidate_h[start1:start1+swap_size] = candidate_h[start2:start2+swap_size]
        candidate_h[start2:start2+swap_size] = temp_values
        
        # Rescale to keep sum constraint
        total_sum = np.sum(candidate_h)
        scale_factor = (n_points / 2) / total_sum
        candidate_h *= scale_factor
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Compute new C5
        new_c5 = compute_c5(candidate_h)
        
        # Accept or reject
        if new_c5 < current_c5:
            current_h = candidate_h
            current_c5 = new_c5
        else:
            # Accept with probability based on temperature
            if np.random.rand() < np.exp(-(new_c5 - current_c5) / temperature):
                current_h = candidate_h
                current_c5 = new_c5
        
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    # Final polishing with SLSQP
    from scipy.optimize import minimize
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    try:
        result = minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            best_c5 = compute_c5(best_h)
    except:
        pass
    
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
