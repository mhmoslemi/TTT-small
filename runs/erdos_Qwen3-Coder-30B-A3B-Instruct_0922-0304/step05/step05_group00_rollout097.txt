import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def evaluate_individual(h, dx):
    """Evaluate objective and check constraints."""
    c5 = compute_c5(h, dx)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from given construction or create a simple one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a simple step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Normalize to target sum
    h = h * (target_sum / np.sum(h))
    
    # Initialize best solution
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Iterative improvement with linearization
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # For each lag, we compute approximate overlap with linearization
        # Find high-overlap lags
        overlap = correlate(h, 1 - h, mode="full")
        # Identify the max lag contributions
        max_lag = np.argmax(overlap)
        # Only focus on top 20% of lags
        selected_lags = np.argsort(overlap)[-int(0.2 * len(overlap)):]
        
        # We now solve a linearized problem:
        # minimize C5 subject to:
        #   C5 >= sum(h[i] * (1 - h_prev[i+k]))  for k in selected_lags
        
        # Use simple gradient descent approach to improve h directly
        step_size = 0.001
        h_old = h.copy()
        
        # Simple local search - perturb one element at a time
        for i in range(n_points):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Try a small perturbation
            h_trial = h.copy()
            perturbation = np.random.normal(0, 0.01)
            h_trial[i] += perturbation
            h_trial = np.clip(h_trial, 0, 1)
            
            # Re-scale to maintain sum
            h_trial = h_trial * (target_sum / np.sum(h_trial))
            
            # Check improvement
            c5_trial = compute_c5(h_trial, dx)
            if c5_trial < best_c5:
                h = h_trial
                best_c5 = c5_trial
                if best_c5 <= 0.3807:
                    break
        
        # Periodic full recomputation to avoid local minima
        if iteration % 10 == 0:
            # Move towards a solution with fewer large bumps
            # This introduces slight smoothing
            h_filtered = np.convolve(h, np.ones(3)/3, mode='same')
            h_filtered = np.clip(h_filtered, 0, 1)
            # Re-normalize
            h_filtered = h_filtered * (target_sum / np.sum(h_filtered))
            h = h_filtered
        
        # Another simple optimization approach: 
        # try to find a stable point by simple descent
        h_final = h.copy()
        best_h_current = h.copy()
        c5_current = compute_c5(h, dx)
        
        # Evaluate a modification step
        for _ in range(5):
            # Randomly modify one region
            idx = np.random.randint(n_points)
            h_test = h.copy()
            pert = np.random.normal(0, 0.01)
            h_test[idx] += pert
            h_test = np.clip(h_test, 0, 1)
            # Re-normalize
            h_test = h_test * (target_sum / np.sum(h_test))
            
            c5_test = compute_c5(h_test, dx)
            if c5_test < c5_current:
                h_final = h_test

        h = h_final

            best_c5 = c5_current
            best_h = h.copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
