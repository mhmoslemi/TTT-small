import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: binary with ~n_points/2 ones
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        np.random.shuffle(h)
        h = h * (n_points / 2) / np.sum(h)
        
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Phase 1: Binary search with simulated annealing
    def binary_search():
        nonlocal h, best_c5, best_h
        T = 1.0
        alpha = 0.99
        max_iter = 1000
        
        current_c5 = compute_c5(h)
        best_c5 = current_c5
        best_h = h.copy()
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Try to improve solution
            candidate = h.copy()
            i, j = np.random.randint(0, n_points, 2)
            candidate[i], candidate[j] = candidate[j], candidate[i]
            
            # Maintain sum constraint
            candidate_sum = np.sum(candidate)
            candidate = candidate * (n_points / 2) / candidate_sum
            
            candidate_c5 = compute_c5(candidate)
            
            # Accept or reject
            if candidate_c5 < current_c5 or np.random.rand() < np.exp(-(candidate_c5 - current_c5) / T):
                h = candidate
                current_c5 = candidate_c5
                
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            # Cooling schedule
            T *= alpha
            
            # Early exit
            if best_c5 <= 0.3807:
                break
    
    # Phase 2: Continuous relaxation using CVXPY
    def continuous_refinement():
        nonlocal h, best_c5, best_h
        # Convex relaxation using CVXPY
        h_var = cp.Variable(n_points)
        
        # Maximize the overlap
        constraints = [cp.sum(h_var) == n_points / 2,
                      h_var >= 0,
                      h_var <= 1]
        
        # Create a constraint that makes the maximum overlap equal to our current value
        # For simplicity, we use a minimax formulation where we try to minimize the worst-case overlap
        
        objective = cp.Minimize(cp.max(cp.multiply(h_var, 1 - h_var)))
        
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                refined_h = h_var.value
                refined_h = np.clip(refined_h, 0, 1)
                refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
                refined_c5 = compute_c5(refined_h)
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = refined_h.copy()
        except:
            pass  # Skip if optimization fails
    
    # Run both phases
    binary_search()
    
    # Refinement step
    if time.time() - start_time < budget_s - 1:
        continuous_refinement()

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    # Output best known solution
    return (best_h.tolist(), final_c5, n_points)
