```python
import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have sum equal to target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Start with a simple construction
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)
        # Add some noise to break symmetry
        h += np.random.normal(0, 0.05, n_points)
        h = project_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Gradient-based optimization with adaptive learning rate
    max_iter = 10000
    learning_rate = 0.01
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient for the maximum overlap case
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_k = max_idx - (n_points - 1)
        
        # Compute gradient: d/dh_i of max_k overlap term
        # The maximum overlap occurs when we look at h[j]*(1-h[j+k]) for all j
        gradient = np.zeros(n_points)
        for j in range(n_points):
            if 0 <= j + max_k < n_points:
                gradient[j] = -1 * (1 - h[j + max_k]) 
            if 0 <= j - max_k < n_points and max_k != 0:
                gradient[j] += -1 * (1 - h[j - max_k])
        
        # Update step with learning rate
        h_new = h - learning_rate * gradient
        h_new = project_h(h_new, target_sum)
        
        # Evaluate new solution
        new_c5 = compute_c5(h_new, dx)
        
        # Accept if improvement
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            
        h = h_new
        
        # Adaptive learning rate
        if i > 0 and i % 100 == 0:
            learning_rate *= 0.98
    
    return (best_h.tolist(), best_c5, n_points)
```
