import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Create initial swarm
    particles = [h.copy()]
    velocities = [np.random.uniform(-0.05, 0.05, n_points)]
    
    # Add random particles to swarm
    for _ in range(swarm_size - 1):
        h_new = np.random.uniform(0, 1, n_points)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        particles.append(h_new)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Best positions tracking
    best_particles = [p.copy() for p in particles]
    best_values = [float('inf')] * swarm_size
    
    # Global best
    global_best_h = particles[0].copy()
    global_best_c5 = float('inf')
    
    # PSO parameters
    w_max = 0.9
    w_min = 0.4
    phi1 = 2.0
    phi2 = 2.0
    max_vel = 0.1
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Update inertia weight
        w = w_max - (w_max - w_min) * (time.time() - start_time) / budget_s
        
        for i in range(swarm_size):
            h = particles[i]
            
            # Compute fitness
            c5 = compute_c5(h)
            
            # Update particle's best
            if c5 < best_values[i]:
                best_values[i] = c5
                best_particles[i] = h.copy()
                
                # Update global best
                if c5 < global_best_c5:
                    global_best_c5 = c5
                    global_best_h = h.copy()
            
            # Early exit
            if global_best_c5 <= 0.3807:
                break
                
            # Update velocity
            r1, r2 = np.random.random(2)
            v = w * velocities[i] + \
                phi1 * r1 * (best_particles[i] - h) + \
                phi2 * r2 * (global_best_h - h)
            
            # Clip velocity
            v = np.clip(v, -max_vel, max_vel)
            
            # Update position
            h_new = h + v
            
            # Ensure feasibility
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Update particle
            particles[i] = h_new
            velocities[i] = v
            
        # Early exit
        if global_best_c5 <= 0.3807:
            break

    return (global_best_h.tolist(), global_best_c5, n_points)
