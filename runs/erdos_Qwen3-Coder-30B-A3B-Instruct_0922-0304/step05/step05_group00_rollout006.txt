import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from initial_h_values or create a base
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a simple step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # LP approach with iterative linearization
    max_iter = 1000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlaps
        overlap = correlate(h, 1 - h, mode="full")
        
        # Identify top lags contributing to C5
        top_lags = np.argsort(overlap)[-max(10, n_points//10):][::-1]
        
        # Build LP
        h_var = cp.Variable(n_points)
        C5 = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == target_sum,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add linearized constraints for significant lags
        for k in top_lags:
            # Adjust lag index for correlate's full mode
            # correlate gives a result of length 2*n_points - 1
            # The lag k corresponds to index k + n_points - 1
            lag_idx = k + n_points - 1
            
            # Linear approximation of overlap_k
            # overlap_k ≈ sum(h[i]*(1-h[i+k]) + h[i+k]*(h[i]-h[i+k])) = sum(h[i] - h[i]*h[i+k] + h[i+k]*h[i] - h[i+k]^2)
            # This simplifies to sum(h[i] - h[i+k]^2) = sum(h[i]) - sum(h[i+k]^2)
            # But we need to consider the actual computation:
            # overlap_k = sum(h[i] * (1 - h[i+k]) + h[i+k] * (h[i] - h[i+k]))
            #           = sum(h[i] - h[i]*h[i+k] + h[i+k]*h[i] - h[i+k]^2)
            #           = sum(h[i] - h[i+k]^2) = sum(h[i]) - sum(h[i+k]^2)
            
            # The actual formula from correlate when sliding:
            # overlap_k = sum(h[i] * (1 - h[i+k]) for i = 0..n_points-1-k)
            # Let's compute this more carefully using linear approximation:
            # For small deformations, we can use first-order approximation:
            # h[i+k] ≈ h_old[i+k] + dh[i+k]
            # So h[i] * (1-h[i+k]) ≈ h[i] * (1-h_old[i+k]) - h[i] * dh[i+k]
            # The total linearization for overlap_k:
            # sum(h[i] * (1-h_old[i+k]) - h[i] * dh[i+k]) = sum(h[i]) * (1-h_old[i+k]) - sum(h[i] * dh[i+k])
            # This needs to be rewritten more carefully
            
            # Simpler linearization approach:
            # Let's manually construct the constraint carefully using full overlap
            # overlap_k ≈ sum(h[i] * (1-h[i+k])) = sum(h[i]) - sum(h[i]*h[i+k])
            
            # Set linearization around current h
            linearized_overlap = 0
            for i in range(n_points - k):
                linearized_overlap += h[i] * (1 - h[i+k]) + h[i+k] * (h[i] - h[i+k])
            
            # Simple linearization: 
            # Let's compute sum(h[i] * (1-h_old[i+k]) - h[i] * (h[i+k]-h_old[i+k]))
            # Which is sum(h[i] * (1-h_old[i+k]) - h[i] * h[i+k] + h[i] * h_old[i+k])
            # = sum(h[i]) * (1-h_old[i+k]) - sum(h[i]*h[i+k]) + sum(h[i]*h_old[i+k]) 
            # Let y = h[i+k] for i fixed
            # overlap_k ≈ sum(h[i] * (1-h[i+k])) ≈ \sum_{i=0}^{n-k-1} h[i] - \sum_{i=0}^{n-k-1} h[i]*h[i+k]
            # We linearize the second term:
            # Linearizing h[i]*h[i+k] around current values:
            # h[i]*h[i+k] ≈ h[i]*h_old[i+k] + h_old[i]*h[i+k] - h_old[i]*h_old[i+k]
            
            # Final correct linearization approach:
            # We have: sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k]) = sum h[i] - sum h[i] * h[i+k]  
            # Linearizing h[i] * h[i+k] around current h:
            # h[i] * h[i+k] ≈ h[i] * h_old[i+k] + h_old[i] * h[i+k] - h_old[i] * h_old[i+k]
            # So the linearized sum is:
            # sum h[i] - (sum h[i] * h_old[i+k] + sum h_old[i] * h[i+k] - sum h_old[i] * h_old[i+k])
            # = sum h[i] - sum h[i] * h_old[i+k] - sum h_old[i] * h[i+k] + sum h_old[i] * h_old[i+k]
            # = sum (h[i] - h[i] * h_old[i+k]) - sum h_old[i] * h[i+k] + sum h_old[i] * h_old[i+k]
            
            # Using approximate cross terms (fix one, approximate the other):
            # overlap_k_manual = 0
            # for i in range(n_points - k):
            #     overlap_k_manual += h[i] * (1 - h[i+k])
            # Linearize around h:
            # This is linearization at h:
            # overlap_k ≈ sum_{i} h[i] * (1 - h[i+k]) ≈ sum h[i] - sum (h[i] * h[i+k])
            # Linearization of h[i]*h[i+k] via:
            # h[i]*h[i+k] ≈ h[i]*h[i+k]^{old} + h[i+k]^{old} * h[i] - h[i+k]^{old} * h[i+k]^{old}
            # = h[i]*h[i+k]^{old} + h[i+k]^{old} * h[i] - h[i+k]^{old2}
            
            # Simpler approach:
            # For i = 0 to n_points-k-1:
            # overlap_k = sum(h[i] * (1 - h[i+k])) = sum(h[i]) - sum(h[i]*h[i+k])
            # To linearize h[i]*h[i+k], approximate using h_old:
            # h[i] * h[i+k] ≈ h[i] * h_old[i+k] + h_old[i] * h[i+k] - h_old[i] * h_old[i+k]
            # So we can write:
            # sum(h[i]) - sum(h[i]*h[i+k]) ≈ sum(h[i]) - sum(h[i]*h_old[i+k]) - sum(h_old[i]*h[i+k]) + sum(h_old[i]*h_old[i+k])
            # Hence: overlap_k ≈ 1 * sum(h[i]) - sum(h[i]*h_old[i+k]) - sum(h_old[i]*h[i+k]) + sum(h_old[i]*h_old[i+k])
            
            # Simple, crude linear approximation instead for now:
            # Summary of problem:
            # It's better to compute actual constraints for the overlap terms one by one
            
            # Instead of linearizing, compute actual positions in the overlap sum
            # overlap_k = sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k])
            # When we linearize this around current values z:
            # overlap_k ≈ sum_{i=0}^{n-k-1} z[i] * (1 - h[i+k]) + sum_{i=0}^{n-k-1} h[i] * (1 - z[i+k])
            #              - z[i] * z[i+k] + h[i] * h[i+k]
            # This simplifies to:
            # overlap_k ≈ sum h[i] * (1 - h[i+k]) + sum z[i] * (1 - z[i+k]) - h[i] * h[i+k] + z[i] * z[i+k]
            # This is complex to do correctly.
            # Let's directly use the constraint: sum(h[i] * (1 - h[i+k])) <= C5 * dx
            
            # Instead, here we use an even simpler direct approach.
            # my_linear_cons = sum(h_var[i] * (1 - h_var[i+k])) - C5 <= 0
            # But this is bilinear. For fixed h_var[i] and h_var[i+k], we compute approx according to:
            
            # Just compute the constraint directly as a set of linear inequalities that represent
            # the current overlap_k being bounded by a linear expression in h_var
            
            const_val = 0.0
            if k < n_points:
                for i in range(n_points - k):
                    const_val += h[i] * (1 - h[i+k])
            
            # Exact form for Lag k:
            # We will now create a linear constraint for overlap_k <= C5 * dx
            # Define the linear part from current h.
            # Define offset = const_val - C5*dx. Then,
            # sum(h[i] * (1 - h[i+k])) <= C5 * dx, 
            # set linear part as sum(h[i]) + sum(h[i] * (1 - h[i+k])) - sum(h[i] * h[i+k]):
            # sum(h[i] * (1 - h_old[i+k])) = sum(h[i]) - sum(h[i] * h_old[i+k])
            # and for every position, h[i+k] is estimated to be h[i+k] in h_new
            
            # Linearize the constraint_overlap_k <= C5 * dx
            # overlap_k = sum(h[i] * (1-h[i+k]) for i in range(n-k)) 
            # Linearizing around current h:
            # h[i] * (1 - h[i+k]) ≈ h[i] * (1 - h_old[i+k]) - h_old[i] * (h[i+k] - h_old[i+k]) 
            # = h[i] * (1 - h_old[i+k]) - h_old[i] * (h_new[i+k] - h[i+k])
            
            # This is still complex to do correctly; for a faster pass: 
            # Just add constraint:     sum(h[i] * (1 - h[i+k])) - C5 * dx <= 0
            # and linearise this using current solution
            
            # BUT actually that makes it very direct:
            # Actually we are minimizing C5 and adding constraints sum(h[i] * (1 - h[i+k])) <= C5 * dx
            # So define new variables one_by_one, solve afterward, this gives:
            
            # Let's restart with feasibility first. 
            # Use the computed overlap to create one by one linear inequalities based on 
            # current h.
            # overlap_k = sum(h[i] * (1 - h[i+k]) for i = 0 to n-k-1)
            # Inverse of right side: C5 * dx >= overlap_k  
            # At any point, we know all overlap terms. 
            # Rather than linearizing every individual, just directly impose exact inequality:
            # This still needs discretization to see values of overlaps
            
            # Realisticly, we'll solve a few iterations of hardcoded small LP.
            
            # We're just solving:
            # minimize C5 subject to: 
            #   sum h[i] = target_sum
            #   0 <= h[i] <= 1
            #   sum(h[i] * (1 - h[i+k])) <= C5 * dx for all top_lags k
            
            # Approximate the constraints as linearized.
            constraint_expr = 0
            for i in range(n_points - k):
                if i+k < n_points:
                    constraint_expr += h_var[i] * (1 - h[i+k])
            
            constraints.append(constraint_expr <= C5 * dx)
        
        # Solve
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GLPK, max_iters=1000, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                if h_new is not None:
                    h = np.clip(h_new, 0, 1)
                    # renormalize to meet sum
                    h = h * (target_sum / np.sum(h))
                    c5 = compute_c5(h, dx)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                    
                    if best_c5 <= 0.3807:
                        break
        except Exception as e:
            pass

        iteration += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
