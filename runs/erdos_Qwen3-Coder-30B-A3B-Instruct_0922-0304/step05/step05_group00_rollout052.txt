import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_and_normalize(h, target_sum):
    """Project h to feasible region and normalize."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Fallback: create a simple step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Iterative refinement using constrained optimization
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Identify key lags that contribute most to overlap
        overlap = correlate(h, 1 - h, mode="full")
        # Get top 5% oflags with highest contribution
        top_lags = np.argsort(overlap)[-max(1, n_points // 20):]  # Top 5% of lags
        
        # Try optimizing using these specific lags as constraints
        # We'll do local optimization around current h with more precise constraints
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Try reducing overlap at all lags with index > 0 (the real non-trivial ones)
        def objective(h):
            return compute_c5(h, dx)
        
        # For local search, use SLSQP
        constraint_sum = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint_sum, options={'maxiter': 50})
        
        if result.success:
            h_new = result.x
            h_new = project_and_normalize(h_new, target_sum)
            c5_new = compute_c5(h_new, dx)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                
        # Add small random perturbation to escape local minima
        h = project_and_normalize(h + np.random.normal(0, 0.005, n_points), target_sum)
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
