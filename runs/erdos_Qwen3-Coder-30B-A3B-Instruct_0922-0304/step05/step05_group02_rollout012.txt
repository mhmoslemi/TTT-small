import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to satisfy sum(h) = target_sum while keeping in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        scale = target_sum / current_sum
        h = h * scale
    return h

def line_search(h, grad, c5_old, dx, alpha_max=1.0, tolerance=1e-6, max_iter=20):
    """Perform backtracking line search to find step size."""
    alpha = alpha_max
    c5_new = c5_old + 1.0  # Initialize to ensure first iteration enters the loop
    for _ in range(max_iter):
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, len(h) / 2)
        c5_new = compute_c5(h_new, dx)
        if c5_new <= c5_old - tolerance * alpha * np.dot(grad, grad):
            return alpha, h_new, c5_new
        alpha *= 0.5
    return alpha, h, c5_old

def trust_region_update(h, grad, c5_old, dx, momentum=None, mu=0.9, alpha_init=0.1):
    """Perform a trust-region update with momentum."""
    if momentum is not None:
        grad = (1 - mu) * grad + mu * momentum
    
    alpha, h_new, c5_new = line_search(h, grad, c5_old, dx, alpha_max=alpha_init)
    
    if momentum is not None:
        momentum = grad.copy()
    else:
        momentum = grad.copy()
        
    return h_new, c5_new, momentum

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, n_points / 2)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, n_points / 2)

    # Tracks
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    momentum = None
    alpha = 0.1
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        c5_old = compute_c5(h, dx)
        
        # Find the worst shift (shift that maximizes the overlap term)
        full_correlation = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(full_correlation)
        
        # Compute gradient for the worst shift
        # We only need gradient for the specific shift that contributes most
        grad = np.zeros(n_points)
        # The gradient w.r.t. h[i] is the convolution derivative
        # Specifically, for shift k: d/dh[i] (h[i] * (1-h[i+k])) 
        # = 1 - 2*h[i] if there is a contribution at shift k
        # This is included in the convolution result. 
        # Instead, we compute the actual gradient numerically
        
        # Actually, we compute a numerical approximation of gradient for the current best shift
        # Target shift idx is max_shift_idx - (n_points - 1) 
        # This gives us the shifted index of h
        # For overlap h(i) * (1 - h(i + k)), we identify k
        k = max_shift_idx - (n_points - 1)
        # Gradient is approximately (1 - 2 * h[i]) for entries where shifts occur
        # i = j + k for j = 0..n_points-1, so for a fixed k, we know where h[i] appears
        # For j from 0 to n_points - 1, h[j] contributes to shift k if j + k is within bounds
        
        # Let's compute the gradient direction
        # Approximate gradient for shift k (for consistency with math)
        # We suppose the integration over shift k contributes the most.
        # Gradient of overlap at i w.r.t. h[i] is 1 - 2 * h[i+k] when there's a contribution.
        # Yet practical computation of such a gradient is more complex.
        # We compute it via finite differences:
        
        epsilon = 1e-5
        grad_approx = np.zeros(n_points)
        for i in range(n_points):
            h_eps = h.copy()
            h_eps[i] += epsilon
            h_eps = project_to_feasible(h_eps, n_points / 2)
            c5_plus = compute_c5(h_eps, dx)
            
            c5_minus = c5_old  # Corrected to be an approximation via central differences
            h_eps[i] -= 2 * epsilon
            h_eps = project_to_feasible(h_eps, n_points / 2)
            c5_minus = compute_c5(h_eps, dx)
            
            grad_approx[i] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # Update
        h_new, c5_new, momentum = trust_region_update(h, grad_approx, c5_old, dx, 
                                                      momentum=momentum, alpha_init=alpha)
        
        h = h_new
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        alpha = min(alpha * 1.02, 0.5)  # Adaptive learning rate
            
    return (best_h.tolist(), best_c5, n_points)
