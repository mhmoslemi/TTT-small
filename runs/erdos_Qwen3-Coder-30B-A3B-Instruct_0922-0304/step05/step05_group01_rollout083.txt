import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def validate_and_repair(h, n_points):
    h = np.array(h)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def initialize_coarse(n_coarse=35):
    h = np.ones(n_coarse)
    h[n_coarse//2:] = 0
    return h

def upscale(h_coarse, n_points):
    # Nearest neighbor upsampling
    h_fine = np.repeat(h_coarse, 2)
    if len(h_fine) > n_points:
        h_fine = h_fine[:n_points]
    elif len(h_fine) < n_points:
        h_fine = np.pad(h_fine, (0, n_points - len(h_fine)), constant_values=0)
    return h_fine

def compute_c5(h, dx=2.0 / len(h)):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def optimize(h, n_points, max_iter=200, tol=1e-8):
    dx = 2.0 / n_points
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimize using SLSQP
    result = minimize(
        lambda x: compute_c5(x, dx),
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': max_iter, 'ftol': tol}
    )
    
    if result.success:
        h = result.x
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    n_coarse = 35
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_coarse = validate_and_repair(initial_h_values, n_coarse)
    else:
        h_coarse = initialize_coarse(n_coarse)
    
    # Coarse optimization
    h_coarse = optimize(h_coarse, n_coarse, max_iter=50, tol=1e-6)
    
    # Upscale to fine resolution
    h_fine = upscale(h_coarse, n_points)
    
    # Fine optimization with stricter tolerances
    h_fine = optimize(h_fine, n_points, max_iter=200, tol=1e-8)
    
    # Evaluate final solution
    final_c5 = compute_c5(h_fine)
    
    # Early exit if target achieved
    if final_c5 <= 0.3807:
        return (h_fine.tolist(), final_c5, n_points)
    
    # Additional refinement if needed
    best_h = h_fine.copy()
    best_c5 = final_c5
    
    # Try additional optimizations
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        np.random.seed(int(time.time() * 1000) % (2**32))
        perturbation = np.random.normal(0, 0.02, n_points)
        h_try = h_fine + perturbation
        h_try = validate_and_repair(h_try, n_points)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    
    return (best_h.tolist(), best_c5, n_points)
