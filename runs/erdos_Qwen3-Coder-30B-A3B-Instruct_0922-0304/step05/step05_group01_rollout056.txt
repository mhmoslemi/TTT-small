import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def validate_and_repair(h_values, n_points):
    """Validate and repair initial solution to meet constraints."""
    h = np.array(h_values)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def initialize_coarse(n_coarse=35):
    """Initialize coarse solution."""
    h = np.ones(n_coarse)
    h[n_coarse//2:] = 0
    return h

def upscale(h_coarse, n_points):
    """Upscale coarse solution to fine grid."""
    h_fine = np.repeat(h_coarse, 2)
    # Adjust to maintain sum constraint
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    return h_fine

def compute_c5(h, dx=2.0/len(h)):
    """Compute C5 value."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def optimize(h, n_points, max_iter=200, tol=1e-8):
    """Optimize h with constraints."""
    dx = 2.0 / n_points
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    def objective(h):
        return compute_c5(h, dx)

    result = minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': max_iter, 'ftol': tol}
    )
    
    if result.success:
        h = result.x
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    n_coarse = 35
    dx = 2.0 / n_points
    
    # Initialize coarse solution
    if 'initial_h_values' in globals():
        h_coarse = validate_and_repair(initial_h_values, n_coarse)
        # Ensure correct size
        if len(h_coarse) != n_coarse:
            h_coarse = initialize_coarse(n_coarse)
    else:
        h_coarse = initialize_coarse(n_coarse)

    # Coarse optimization (smaller budget)
    best_c5_coarse = float('inf')
    best_h_coarse = h_coarse.copy()
    
    # Reduce computational budget for coarse optimization
    coarse_budget = min(0.2 * budget_s, 100)
    coarse_start = time.time()
    
    for iteration in range(50):
        if time.time() - coarse_start > coarse_budget - 1:
            break
            
        h_coarse = optimize(h_coarse, n_coarse, max_iter=50, tol=1e-6)
        current_c5 = compute_c5(h_coarse, 2.0/n_coarse)
        if current_c5 < best_c5_coarse:
            best_c5_coarse = current_c5
            best_h_coarse = h_coarse.copy()
    
    # Upsample to fine scale
    h_fine = upscale(best_h_coarse, n_points)
    
    # Fine optimization with remaining budget
    fine_budget = budget_s - (time.time() - start_time)
    if fine_budget > 10:
        best_c5_fine = float('inf')
        best_h_fine = h_fine.copy()
        
        # Track best overall
        best_overall_c5 = float('inf')
        best_overall_h = None
        
        # Allow more iterations for fine tune
        fine_iter = 0
        while time.time() - start_time < budget_s - 2 and fine_iter < 200:
            h_fine = optimize(h_fine, n_points, max_iter=20, tol=1e-8)
            current_c5 = compute_c5(h_fine, dx)
            if current_c5 < best_c5_fine:
                best_c5_fine = current_c5
                best_h_fine = h_fine.copy()
                
            if current_c5 < best_overall_c5:
                best_overall_c5 = current_c5
                best_overall_h = h_fine.copy()
                
            fine_iter += 1
            
            # Early exit if target achieved
            if current_c5 <= 0.3807:
                break
                
        if best_overall_h is not None:
            final_c5 = best_overall_c5
            final_h = best_overall_h
        else:
            final_c5 = best_c5_fine
            final_h = best_h_fine
    else:
        final_c5 = compute_c5(h_fine, dx)
        final_h = h_fine

    return (final_h.tolist(), final_c5, n_points)
