import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize to target sum
    h = h * (target_sum / np.sum(h))
    h = np.clip(h, 0, 1)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 50  # More iterations with LP
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Re-evaluate C5
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Identify top contributing lags
        overlap = correlate(h, 1 - h, mode='full')
        # Get top 10 percent of lags by magnitude
        top_lags = np.argsort(np.abs(overlap))[-max(1, n_points//10):][::-1]
        top_lags = top_lags[overlap[top_lags] > 0]  # positive overlaps only
        
        # Define LP variables
        h_var = Variable(n_points)
        C5_var = Variable(1)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            sum(h_var) == target_sum,
            C5_var >= 0
        ]
        
        # For each top lag, add linearized constraint
        for k in top_lags:
            # k represents the lag offset
            # For lag k: sum h[i] * (1 - h[i + k]) + h[i + k] * (h[i] - h[i + k])
            # This simplifies to sum h[i] - h[i] * h[i + k] + h[i + k] * h[i] - h[i + k]^2
            # Which is sum h[i] - h[i + k]^2
            
            # To linearize, we use first-order Taylor around current h
            # h[i] * (1 - h[i + k]) ~ h[i] * (1 - h[i + k]_old) + h[i + k]_old * (h[i] - h[i]_old) - h[i + k]_old^2
            
            # Actually, let's do a simpler linearization:
            # For the term h[i]* (1-h[i+k]) in the overlap sum,
            # we linearize as: h[i] * (1 - h[i+k]) ~ 
            # h[i] * (1 - h[i+k]_prev) + (1 - h[i+k]_prev) * (h[i] - h[i]_prev) = 
            # h[i] - h[i]*h[i+k]_prev + h[i] - h[i]_prev - h[i+k]_prev * h[i] + h[i]_prev * h[i+k]_prev
            
            # Simpler: 
            # Compute linearization for overlap term based on current h
            if k < n_points:
                # Not a full convolution channel; cutoffs
                row = np.zeros(n_points-k)
                for i in range(n_points-k):
                    # overlap_k corresponds to sum_{i=0}^{n-k-1}
                    # h[i] * (1 - h[i+k]) + h[i+k] * (h[i] - h[i+k])
                    # This can be rewritten as
                    # h[i] - h[i]*h[i+k] + h[i]*h[i] - h[i+k]^2
                    # which leads to cancellation
                    # Let's recompute the actual term:
                    # The overlap k term is sum_{i=0}^{n-k-1} h[i] * (1 - h[i+k]) 
                    #         = sum h[i] - sum h[i] * h[i+k]
                    # We can approximate the cross term using first-order Taylor:
                    row[i] = h[i] * (1 - h[i+k]) + h[i+k] * (h[i] - h[i+k]) 
                    # Actually, this is better:
                    # -h[i] * h[i+k] + h[i]*h[i] - h[i+k]^2
                    # h[i] - h[i+k]^2
                    # Let's actually compute and linearize properly:
                    # h[i] * (1 - h[i+k]) = h[i] - h[i] * h[i+k]
                    # Contribution to overlap_k = h[i] - h[i] * h[i+k]
                    # Linearization around h_prev:
                    # h[i] - h[i] * h[i+k] ~ h[i] - h[i] * h_prev[i+k]
                    # So the constraint becomes: 
                    # sum_{i=0}^{n-k-1} (h[i] - h[i] * h_prev[i+k]) <= C5
                    # or: sum_{i=0}^{n-k-1} h[i] - sum_{i=0}^{n-k-1} h[i] * h_prev[i+k]  <= C5
                    # i.e., sum h[i] - sum h[i] * h_prev[i+k] <= C5
                
                # This is probably the right approach:
                # Our goal is to minimize C5 such that for each lag k
                # sum_i h[i] * (1 - h[i+k]) <= C5
                # This sums to h[i] - h[i]*h[i+k]  for i: i+k < n_points
                # For the linearization at current h:
                # For i: i+k < n_points, sum up to sum(h[i]) - sum(h[i]*h[i+k])
                # Use h[i] - h[i]*h_prev[i+k] = h[i]*(1 - h_prev[i+k])
                # So the constraint becomes: sum(h[i]*(1-h_prev[i+k])) <= C5
                # This transcription of the correction term is incorrect
                # what we want is h[i]*(1-h[i+k]) = h[i] - h[i]*h[i+k]
                # Linearization around h_prev:
                # h[i]*(1-h[i+k]) = h[i] - h[i]*h[i+k] => 
                # Linearized as h[i] - h[i]*h_prev[i+k] 
                # Total: sum(h[i]*(1 - h_prev[i+k]))
                # But we need to impose that sum_{i} h[i]*(1-h[i+k]) <= C5
                
                # As an approximation, we can do:
                # Instead of sum h[i] * (1 - h[i+k]) <= C5
                # Constraint approximates as:
                # h[i] * (1 - h_prev[i+k]) (each term)
                
                linearized_term = 0
                for i in range(n_points - k):
                    linearized_term += h_var[i] * (1 - h[i+k]) 
                
                # Or linearize as sum(h[i] - h[i]*h_prev[i+k])
                linearized_sum = sum(h_var[i] * (1 - h[i+k]) for i in range(n_points - k))
                
                # But we want sum(h[i]* (1 - h[i+k])) <= C5
                # So we approx. as: 
                # twice linearization, still, centers around current guess:
                # or reuse the same h for both:
                # sum(h[i] - h[i]*h[i+k]) <= C5.
                # linear term: sum(h[i]) 
                # cross term: sum(h[i]*h[i+k]) 
                # At current h:
                # We are looking to bound:
                # sum_{i=0}^{n-k-1} h[i] *(1 - h[i+k])  
                # = sum h[i] - sum h[i]*h[i+k]
                # etc. 
                # Linear constraints: 
                # sum_{i=0}^{n-k-1} (h[i] - h[i] h_prev[i+k]) <= C5[i+k] 
                # which is: 
                # sum_{i=0}^{n-k-1} h[i] - sum_{i=0}^{n-k-1} h[i] h_prev[i+k] <= C5
                # The constraint can be written as:
                # 1. sum(h[i] - h[i]*h_prev[i+k]) <= C5 for each valid k
                # We approximate sum h[i] - sum h[i] h_prev[i+k] <= C5
                # Thus sum(h[i]) - sum(h[i] h_prev[i+k]) <= C5
                # The constraint is fixed due to previous h and is equal to:
                # sum_{i} h[i]*(1-h_prev[i+k])
                # or equivalently sum h[i] - sum h[i] h_prev[i+k]
                
                # Simplification:
                # Populate into constraint
                # Actually, let's do direct approach for LP:
                # We want: 
                # sum h[i] - sum h[i] * h[i+k] <= C5
                # subsum = sum_{i} h[i] * h[i+k] 
                # So sum h[i] - subsum <= C5
                # The challenge is that lags k may need special form
                
                subsum = sum(h_var[i] * h[i+k] for i in range(n_points - k))
                constraints.append(sum(h_var[i] for i in range(n_points - k)) - subsum <= C5_var)
            else:
                # Out of bounds; fully computed including padding  
                # This is actually more complex
                pass
                
        # Solve with CVXPY
        objective = Minimize(C5_var)
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=SCS, verbose=False, max_iters=1000)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_new = np.array([float(h_var[i].value) for i in range(n_points)])
                h = np.clip(h_new, 0, 1)
                # Normalize
                h = h * (target_sum / np.sum(h))
            else:
                # Fallback: keep old worse value if LP fails
                pass
            
        except:
            pass
            
        iteration += 1
        
    # Re-evaluate final result
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
