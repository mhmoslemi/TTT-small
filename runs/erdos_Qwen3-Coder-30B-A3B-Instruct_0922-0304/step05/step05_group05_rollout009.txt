import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO Parameters
    n_particles = 20
    max_iter = 1000
    w_start = 0.8
    w_end = 0.1
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    # Create initial particles
    if 'initial_h_values' in globals():
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.02, 0.02, n_points))
        p_best.append(h.copy())
        p_best_vals.append(compute_c5(h))
    else:
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.02, 0.02, n_points))
        p_best.append(h.copy())
        p_best_vals.append(compute_c5(h))
    
    for _ in range(n_particles - 1):
        h_pert = h + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        particles.append(h_pert.copy())
        velocities.append(np.random.uniform(-0.02, 0.02, n_points))
        p_best.append(h_pert.copy())
        p_best_vals.append(compute_c5(h_pert))
    
    g_best_idx = np.argmin(p_best_vals)
    g_best = p_best[g_best_idx].copy()
    g_best_val = p_best_vals[g_best_idx]
    
    # PSO loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start - (w_start - w_end) * (iter_num / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) +
                           c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds and constraint
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate fitness
            fitness = compute_c5(particles[i])
            
            # Update personal best
            if fitness < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = fitness
                
                # Update global best
                if fitness < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = fitness
                    
                    if g_best_val <= 0.3807:
                        break
                        
        # Early exit
        if g_best_val <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(g_best)
    
    return (g_best.tolist(), final_c5, n_points)
