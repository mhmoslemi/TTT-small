import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have sum equal to target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

def project_sum(h, target_sum):
    """Project h to have the target sum using L2 projection."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    scale = target_sum / current_sum
    return h * scale

        # Add small noise
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Store best
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient-based optimization
    max_iter = 10000  # Conservative limit
    momentum = 0.0
    momentum_decay = 0.9
    learning_rate = 0.1
    line_search_tol = 0.1
    min_improvement = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        if best_c5 <= 0.3807:
            break
    
        # Compute gradient with respect to the worst shift
        # This is a simplified gradient estimation via finite difference
        # For actual implementation, one could compute the gradient by hand
        # For now, we'll manually sample and compute the gradient for optimization
        # Gradients by finite differences
        eps = 1e-4
        worst_shift_idx = None
        max_overlap_val = -1
        
        full_correlation = correlate(h, 1 - h, mode='full')

        # Compute gradient approximation using finite differences
        grad = np.zeros_like(h)

            h_plus = h.copy()

            h_plus = np.clip(h_plus, 0, 1)
            h_plus = normalize_h(h_plus, target_sum)
            
            c5_plus = compute_c5(h_plus, dx)
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Add momentum term
        momentum = momentum_decay * momentum + (1 - momentum_decay) * grad
        
        # Synchronize: make sure we respect bounds and sum
        # Apply step
        h_new = h - learning_rate * momentum
        
        # Clip and project to constraints
        h_new = np.clip(h_new, 0, 1)

        step_size = learning_rate
