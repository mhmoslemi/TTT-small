import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import math

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
        
    # Optimization using gradient-based method with constraints
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Store best solution so far
    best_solution = (h.copy(), compute_c5(h), n_points)
    
    # Try different evolutionary strategies
    max_generations = 500
    population_size = 20
    
    # Initial population based on refinement of best solution
    population = [h.copy()]
    for _ in range(population_size-1):
        # Add small perturbations
        pert = np.random.normal(0, 0.03, n_points)
        h_pert = h + pert
        h_pert = normalize_h(h_pert, target_sum)
        population.append(h_pert)
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for individual in population:
            c5 = compute_c5(individual)
            fitness.append(c5)
            
        # Track best individual
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
            best_solution = (best_h.copy(), best_c5, n_points)
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Sorting for selection
        sorted_indices = np.argsort(fitness)
        selected = [population[i] for i in sorted_indices[:population_size//2]]
        
        # Create offspring through crossover and mutation
        new_population = selected[:]
        
        while len(new_population) < population_size:
            if len(selected) >= 2:
                # Tournament selection
                idx1 = np.random.choice(len(selected))
                idx2 = np.random.choice(len(selected))
                parent1 = selected[idx1]
                parent2 = selected[idx2]
                
                # Crossover (uniform)
                mask = np.random.rand(n_points) < 0.5
                child = np.where(mask, parent1, parent2)
                
                # Mutation: add noise
                noise = np.random.normal(0, 0.02, n_points)
                child = child + noise
                child = normalize_h(child, target_sum)
                
                new_population.append(child)
            else:
                # Random re-initialization
                child = np.random.rand(n_points)
                child = normalize_h(child, target_sum)
                new_population.append(child)
        
        # Local search for top individuals
        local_search_top = min(5, len(selected))
        for i in range(local_search_top):
            idx = sorted_indices[i]
            current_h = population[idx]
            
            # Call scipy minimize with constraints
            def objective(h):
                return compute_c5(h)
                
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraints = [{'type': 'eq', 'fun': constraint_sum}]
            
            try:
                result = minimize(
                    objective,
                    current_h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 10, 'ftol': 1e-6}
                )
                
                if result.success:
                    h_new = result.x
                    h_new = normalize_h(h_new, target_sum)
                    new_population[idx % len(new_population)] = h_new
            except:
                pass  # Fallback to top individuals
                
        population = new_population.copy()
    
    # Final check of best solution
    final_c5 = compute_c5(best_h)
    
    # Return the best solution found
    return (best_h.tolist(), final_c5, n_points)
