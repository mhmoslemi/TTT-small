import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def evaluate_h(h, dx):
    """Evaluate single h vector."""
    return compute_c5(h, dx)

def line_search(h, grad, dx, alpha_max=1.0, tol=1e-4):
    """Simple backtracking line search."""
    alpha = alpha_max
    current_val = evaluate_h(h, dx)
    while alpha > 1e-8:
        h_new = project_h(h - alpha * grad, len(h) / 2)
        new_val = evaluate_h(h_new, dx)
        if new_val <= current_val - tol * alpha * np.dot(grad, grad):
            return h_new, alpha
        alpha *= 0.5
    return h, alpha

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from parent construction or create baseline
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, n_points / 2)
    else:
        # Symmetric construction for exploration
        h = np.ones(n_points) * 0.5
        for i in range(n_points):
            h[i] = 0.5 + 0.2 * np.sin(2 * np.pi * i / n_points)
        h = project_h(h, n_points / 2)
    
    # Optimization parameters
    momentum = 0.9
    alpha_init = 0.05
    beta = 0.95
    min_improvement = 1e-6
    patience = 50
    
    # Tracking
    best_c5 = float('inf')
    best_h = h.copy()
    previous_c5 = float('inf')
    stagnant = 0
    
    # Main optimization loop
    last_time = start_time
    while time.time() - start_time < budget_s - 1:
        # Compute all correlations
        overlap_full = correlate(h, 1 - h, mode="full")
        kmax = np.argmax(overlap_full)
        max_overlap = overlap_full[kmax] * dx
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break

        # Compute gradient for max overlap shift
        # In cross-correlation, the gradient is 1-h shifted by the max overlap
        # Left half (first n_points elements) is for indices 0 to n_points-1
        # Right half (next n_points-1 elements) is for indices -n_points+1 to -1
        if kmax < n_points:
            # Positive shift
            shift = kmax
            grad = np.zeros(n_points)
            grad[shift:] = 1 - h[:-shift] if shift > 0 else 1 - h
        else:
            # Negative shift
            shift = kmax - n_points + 1
            grad = np.zeros(n_points)
            grad[:-shift] = 1 - h[shift:] if shift > 0 else 1 - h

        # Gradient descent with momentum
        if 'velocity' not in locals():
            velocity = np.zeros(n_points)
        
        velocity = momentum * velocity - alpha_init * grad
        h_new = project_h(h + velocity, n_points / 2)
        
        # Line search
        h_new, alpha = line_search(h, grad, dx, alpha_init)
        h = h_new
        alpha_init = alpha_init * beta
        
        # Monitor stagnation
        if abs(previous_c5 - max_overlap) < min_improvement:
            stagnant += 1
        else:
            stagnant = 0
        previous_c5 = max_overlap
        
        if stagnant > patience:
            # Reinitialize with small Gaussian noise
            h = project_h(h + np.random.normal(0, 0.01, n_points), n_points / 2)
            stagnant = 0
            alpha_init = 0.05
    
    return (best_h.tolist(), best_c5, n_points)
