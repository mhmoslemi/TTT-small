import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def optimize_step(h, c5_bound):
        # Objective: minimize max overlap constrained to be <= c5_bound
        def objective(h):
            return compute_c5(h)
            
        def constraint_overlap(h):
            overlap = correlate(h, 1 - h, mode="full")
            return c5_bound * dx - np.max(overlap) * dx
            
        constraints = [
            {'type': 'eq', 'fun': constraint_sum},
            {'type': 'ineq', 'fun': constraint_overlap}
        ]
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = minimize(
            objective, 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            return h_new
        else:
            return h
    
    # Binary search on C5
    c5_low = 0.0
    c5_high = 0.38085
    tolerance = 1e-4
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    iteration = 0
    while c5_high - c5_low > tolerance and time.time() - start_time < budget_s - 1:
        if iteration > 40:
            break
        c5_mid = (c5_low + c5_high) / 2.0
        h_new = optimize_step(h, c5_mid)
        c5_new = compute_c5(h_new)
        if c5_new <= c5_mid:
            c5_high = c5_mid
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
        else:
            c5_low = c5_mid
            
        iteration += 1
    
    # Fine tune the best solution found so far
    if time.time() - start_time < budget_s - 1:
        # Run more optimization on the best found solution
        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
            h_new = optimize_step(best_h, best_c5)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                if best_c5 <= 0.3807:
                    break

    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
