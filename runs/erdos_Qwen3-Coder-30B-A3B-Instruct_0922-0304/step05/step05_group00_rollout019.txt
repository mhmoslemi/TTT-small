import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def repair_feasibility(h, target_sum):
    """Repair h to be feasible: clip and normalize."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def evaluate_individual(h, dx):
    """Evaluate objective and check constraints."""
    c5 = compute_c5(h, dx)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Parameters
    population_size = 30
    stepsizes = [0.02, 0.01, 0.005, 0.0025]  # Different step sizes for different strategies
    stepsize_decay = 0.98
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Prepare population
    population = []
    for _ in range(population_size):
        population.append(h.copy())
    
    iteration = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate all individuals
        fitness = []
        for ind in population:
            c5 = compute_c5(ind, dx)
            fitness.append(c5)
        
        # Update best solution
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_h = population[min_fit_idx].copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Create new population
        new_population = []
        
        # 1. Differential evolution (40%)
        for _ in range(int(0.4 * population_size)):
            a, b, c = np.random.choice(population_size, 3, replace=False)
            mutant = population[a] + 0.8 * (population[b] - population[c])
            # Crossover
            crossover_prob = 0.7
            trial = np.where(np.random.random(n_points) < crossover_prob, mutant, population[a])
            trial = repair_feasibility(trial, target_sum)
            new_population.append(trial)
        
        # 2. Gradient descent (30%)
        for _ in range(int(0.3 * population_size)):
            idx = np.random.randint(0, population_size)
            h_refined = population[idx].copy()
            # Simple gradient step
            stepsize = stepsizes[np.random.randint(len(stepsizes))]
            # Slightly random gradient to ensure exploration
            gradient = np.random.normal(0, stepsize, n_points) * 0.1
            h_refined += gradient
            h_refined = repair_feasibility(h_refined, target_sum)
            new_population.append(h_refined)
        
        # 3. Random sampling (20%)
        for _ in range(int(0.2 * population_size)):
            new_h = np.random.rand(n_points)
            new_h = repair_feasibility(new_h, target_sum)
            new_population.append(new_h)
        
        # 4. Mutation of best (10%)
        for _ in range(int(0.1 * population_size)):
            new_h = best_h + np.random.normal(0, 0.01, n_points)
            new_h = repair_feasibility(new_h, target_sum)
            new_population.append(new_h)
        
        # Fill up
        while len(new_population) < population_size:
            h_sample = np.random.rand(n_points)
            h_sample = repair_feasibility(h_sample, target_sum)
            new_population.append(h_sample)
        
        # Update population
        population = new_population[:population_size]
        
        # Adjust step sizes
        stepsizes = [s * stepsize_decay for s in stepsizes]
        
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
