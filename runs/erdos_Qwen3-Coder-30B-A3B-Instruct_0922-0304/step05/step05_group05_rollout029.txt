import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Simulated Annealing parameters
    T0 = 0.1  # Initial temperature
    alpha = 0.95  # Cooling rate
    max_iter = 10000
    min_temp = 1e-8
    
    # Track best solution
    best_h = h.copy()
    current_c5 = compute_c5(h, dx)
    best_c5 = current_c5
    
    # Temperature
    T = T0
    
    # Main SA loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate neighbor by perturbing current solution
        h_new = h + np.random.normal(0, T * 0.1, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        new_c5 = compute_c5(h_new, dx)
        
        # Accept or reject
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            current_c5 = new_c5
            
            # Update best solution if improved
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                
        # Cool down
        T = max(T * alpha, min_temp)
        
        # Early termination if target reached
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
