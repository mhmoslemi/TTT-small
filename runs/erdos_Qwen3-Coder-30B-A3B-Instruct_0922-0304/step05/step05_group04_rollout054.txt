import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def compute_gradients(h):
        g = np.zeros(n_points)
        eps_val = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps_val
            h_minus[i] -= eps_val
            h_plus = project_to_feasible(h_plus)
            h_minus = project_to_feasible(h_minus)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps_val) * dx
        return g

    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap
        current_c5 = compute_c5(h)
        
        # Store the best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient
        g = compute_gradients(h)
        
        # Gradient step with adaptive learning rate
        lr = 0.01
        new_h = h - lr * g
        new_h = project_to_feasible(new_h)
        
        # Local search refinement - try multiple small steps with momentum
        momentum_h = h.copy()
        momentum = 0.9
        for _ in range(5):
            g_local = compute_gradients(momentum_h)
            momentum_h = momentum_h - 0.001 * g_local
            momentum_h = project_to_feasible(momentum_h)
            
        # Choose the better among current, updated, and momentum solutions
        candidates = [h, new_h, momentum_h]
        candidate_scores = [compute_c5(c) for c in candidates]
        best_candidate_idx = np.argmin(candidate_scores)
        h = candidates[best_candidate_idx].copy()
        
        # Occasionally do randomized perturbation to escape local optima
        if iteration % 10 == 0:
            perturbation = np.random.normal(0, 0.01, n_points)
            h = h + perturbation
            h = project_to_feasible(h)
    
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
