import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization with binary pattern
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        np.random.shuffle(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Binary search via simulated annealing
    def binary_search():
        best_h = h.copy()
        best_c5 = compute_c5(best_h)
        temp = 1.0
        alpha = 0.99
        max_iter = 10000
        
        iter_count = 0
        while temp > 1e-4 and iter_count < max_iter and time.time() - start_time < budget_s - 1:
            iter_count += 1
            # Generate neighbor by swapping two random positions
            i, j = np.random.choice(n_points, 2, replace=False)
            h_new = best_h.copy()
            h_new[i], h_new[j] = h_new[j], h_new[i]
            
            # Check feasibility (maintain sum = n_points/2)
            # For binary case, we just need to maintain integral constraint
            # We'll work with a slightly different convex approach
            
            new_c5 = compute_c5(h_new)
            delta = new_c5 - best_c5
            
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                best_h = h_new
                best_c5 = new_c5
                
            if best_c5 <= 0.3807:
                break
                
            temp *= alpha
            
        return best_h, best_c5
    
    # Continuous relaxation
    def continuous_relaxation(h_binary):
        # Use CVXPY for continuous optimization
        h_var = cp.Variable(n_points)
        
        # Simplified approach - minimize the maximum overlap using constraints
        # We'll derive the max value and use a penalty approach
        
        # Start from binary solution
        h_start = h_binary
        
        # Create convex optimization problem
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # The objective is to minimize max overlap
        # This is inherently non-convex; we will think of a surrogate
        
        # For consistency with the prior's approach, we try using cvxpy 
        # for actual optimization of continuous h with linear constraints
        
        # Solve using ECOS
        prob = cp.Problem(cp.Minimize(cp.sum(cp.multiply(h_var, 1-h_var))), constraints)
        
        # Try solving, but since the problem is not convex exactly,
        # we will try a heuristic improvement approach
        
        # This will try a few iterations of steepest descent type for small adjustments
        h_opt = h_start.copy()
        h_prev = h_opt.copy()
        
        for _ in range(50):  # Limited iterations for speed
            if time.time() - start_time > budget_s - 1:
                break
            
            # Compute overlap and gradient approximation
            overlap = correlate(h_opt, 1 - h_opt, mode="full")
            k_max_idx = np.argmax(overlap)
            max_overlap = overlap[k_max_idx] * dx
            
            # Simple local tangent-based descent step
            # Approximate gradient directly on the correlation sum
            grad = np.zeros(n_points)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h_opt.copy()
                h_minus = h_opt.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                o_plus = correlate(h_plus, 1 - h_plus, mode="full")
                o_minus = correlate(h_minus, 1 - h_minus, mode="full")
                grad[i] = (o_plus[k_max_idx] - o_minus[k_max_idx]) / (2 * eps) * dx
            
            # Update step with decay
            h_new = h_opt - 0.01 * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Accept
            h_opt = h_new
            
        return h_opt
    
    # Main loop
    h_best = h.copy()
    c5_best = compute_c5(h_best)
    
    # Try binary search first
    if time.time() - start_time < budget_s - 5:
        h_binary, c5_bin = binary_search()
        if c5_bin < c5_best:
            h_best = h_binary
            c5_best = c5_bin
    
    # Then try continuous refinement
    if time.time() - start_time < budget_s - 5:
        h_cont = continuous_relaxation(h_best)
        c5_cont = compute_c5(h_cont)
        if c5_cont < c5_best:
            h_best = h_cont
            c5_best = c5_cont
            
    return (h_best.tolist(), c5_best, n_points)
