import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 10000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Perturb current best solution
        h_new = h + np.random.normal(0, 0.02, n_points)
        h_new = normalize_h(h_new, target_sum)
        
        # Local optimization around the new candidate
        def objective(h_candidate):
            return compute_c5(h_candidate)
        
        def constraint_sum(h_candidate):
            return np.sum(h_candidate) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        # Use L-BFGS for local refinement
        result = minimize(
            objective,
            h_new,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            h_candidate = result.x
            h_candidate = normalize_h(h_candidate, target_sum)
            c5_candidate = compute_c5(h_candidate)
            
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
                
                # Early exit if target reached
                if best_c5 <= 0.3807:
                    break
        
        # Occasionally add a more drastic change
        if iteration % 100 == 0:
            h = best_h.copy()
            # Add a small random perturbation plus structured modification
            perturb = np.random.normal(0, 0.01, n_points)
            # Use a sigmoid shape to encourage structures that help with overlap
            x = np.linspace(0, 2, n_points)
            sigmoid_mod = 1 / (1 + np.exp(-10 * (x - 1)))
            perturb += 0.005 * sigmoid_mod
            h = h + perturb
            h = normalize_h(h, target_sum)
        
        # Keep current best
        h = best_h.copy()
    
    # Final computation of the best solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
