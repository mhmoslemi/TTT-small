import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def compute_c5_fast(h):
        # Use the already computed overlap values
        # Since we're just optimizing, we can use the same
        # optimization logic but more efficiently. But for now,
        # we keep the same correlation-based method for consistency
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def perturb_h(h, temp):
        # Create a new candidate solution with temperature-dependent perturbation
        h_new = h + np.random.normal(0, temp, n_points)
        h_new = np.clip(h_new, 0, 1)
        # Rescale to meet constraint sum(h) = n_points / 2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # SA parameters
    T0 = 0.1  # Initial temperature
    alpha = 0.99  # Cooling rate
    max_iter = 10000
    min_temp = 1e-8
    temp = T0
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    # Iteration tracking
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while temp > min_temp and iteration < max_iter and time.time() - start_time < budget_s - 1:
        # Perturb current solution
        new_h = perturb_h(current_h, temp)
        new_c5 = compute_c5(new_h)
        
        # Accept or reject the new solution
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            current_h = new_h
            current_c5 = new_c5
            
            # Update best solution if improved
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Cooling
        temp *= alpha
        iteration += 1
        
        # Early exit if we've found a solution below the target
        if best_c5 <= 0.3807:
            break
            
        # Check for convergence
        if no_improvement_count > max_no_improvement:
            temp *= 0.9  # Cool further
            no_improvement_count = 0
    
    # Final check of best solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
