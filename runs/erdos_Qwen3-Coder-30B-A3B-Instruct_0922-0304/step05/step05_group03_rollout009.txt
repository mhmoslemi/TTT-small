import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, k_max):
    n = len(h)
    grad = np.zeros(n)
    # For the correlation at lag k_max, we need to compute contribution of each h[i]
    # in the overlap sum. For h[i] at position i, it contributes to overlap at k_max
    # when we compare with (1-h) at position i+k_max
    # The contribution to the overlap at k_max is h[i] * (1-h[i+k_max])
    # we're taking derivative: d/dh[i] of h[i] * (1-h[i+k_max]) = 1 - h[i+k_max]
    for i in range(n):
        j = i + k_max
        if 0 <= j < n:
            grad[i] = 1 - h[j]
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent parameters
    alpha = 0.1
    patience = 0
    patience_limit = 20
    alpha_decay = 0.5
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate current solution
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute overlaps to find max lag
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        
        # Compute gradient
        grad = compute_gradient(h, dx, k_max)
        
        # Update h
        h_new = h - alpha * grad
        
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
            
        # Check improvement
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < c5:
            h = h_new.copy()
            patience = 0
        else:
            patience += 1
            if patience >= patience_limit:
                alpha *= alpha_decay
                patience = 0
                
        # Periodic random perturbation to escape local minima
        if iteration % 50 == 0 and alpha > 1e-6:
            noise = np.random.normal(0, 0.005, n_points)
            h = h + noise
            h = normalize_h(h, target_sum)
    
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
