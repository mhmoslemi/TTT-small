import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint while keeping in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, k_max, dx):
    """Compute gradient for given shift k."""
    n = len(h)
    grad = np.zeros_like(h)
    # This computes the gradient w.r.t. the maximum overlap shift
    if k_max >= 0:
        # h[i] contributes to (h * (1 - h))[i + k_max]
        # For each i, the effect of changing h[i] on (h * (1 - h))[i + k_max]
        # is - (1 - h[i + k_max]) if i + k_max is valid
        for i in range(n):
            j = i + k_max
            if 0 <= j < n:
                grad[i] = - (1 - h[j])
    else:
        # For negative k_max, h[i] contributes to (h * (1 - h))[i + k_max]
        # which means corresponding to h[i + k_max] in h
        for i in range(n):
            j = i + k_max
            if 0 <= j < n:
                grad[i] = - (1 - h[j])
    return grad * dx

def line_search(h, grad, alpha_max, dx, max_iter=20):
    """Perform line search for optimal step size."""
    # Just simple backtracking line search
    alpha = alpha_max
    c1 = 1e-4
    rho = 0.5
    current_c5 = compute_c5(h, dx)
    
    for _ in range(max_iter):
        h_new = project_h(h - alpha * grad, len(h) / 2)
        new_c5 = compute_c5(h_new, dx)
        if new_c5 <= current_c5 - c1 * alpha * np.dot(grad, grad):
            return alpha, h_new
        alpha *= rho
    return alpha, h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    
    start_time = time.time()
    
    # Initialize from provided construction or make a heuristic one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Simple start: some randomization with structure
        h = np.ones(n_points) * 0.5
        # Perturb a bit
        h += np.random.normal(0, 0.05, n_points)
        # Ensure feasible sum
        h = project_h(h, n_points / 2)
    
    # Parameters
    max_iter = 10000
    momentum = 0.9
    lr = 0.1
    tol = 1e-8
    
    h_history = []
    c5_history = []
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    prev_grad = np.zeros_like(h)
    v = np.zeros_like(h)  # momentum term
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all correlations to find max overlap
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max]
        
        # Compute gradient for this k_max
        grad = compute_gradient(h, k_max - (n_points - 1), dx)
        
        # Update with momentum
        v = momentum * v + grad
        # Effective gradient with momentum
        grad_eff = v
        
        # Line search
        alpha, h_new = line_search(h, grad_eff, lr, dx)
        
        # Update h
        h = h_new
        
        if i % 100 == 0:
            current_c5 = compute_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
                if best_c5 <= 0.3807:
                    break
                
                # Early exit
                if i > 1000 and current_c5 > 1.01 * best_c5:
                    continue
                
                h_history.append(h.copy())
                c5_history.append(current_c5)
    
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
