import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: three regions
        h = np.ones(n_points)
        split1 = n_points // 4
        split2 = 3 * n_points // 4
        h[split1:split2] = 0.2
        h[split2:] = 0.8
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def swap_segments(h, i1, i2, size):
        h_new = h.copy()
        # Swap segments
        temp = h_new[i1:i1+size].copy()
        h_new[i1:i1+size] = h_new[i2:i2+size]
        h_new[i2:i2+size] = temp
        
        # Rescale to maintain integral constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h_new = np.clip(h_new, 0, 1)
        return h_new

    # Main optimization loop
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    n_swaps = 1000
    swap_size = 5
    temperature = 0.1
    
    for _ in range(n_swaps):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Select two non-overlapping segments
        i1 = np.random.randint(0, n_points - swap_size)
        i2 = np.random.randint(0, n_points - swap_size)
        while abs(i1 - i2) < swap_size:
            i2 = np.random.randint(0, n_points - swap_size)
        
        # Perform swap
        h_try = swap_segments(h, i1, i2, swap_size)
        c5_try = compute_c5(h_try)
        
        # Accept or reject based on temperature
        if c5_try < best_c5 or np.random.rand() < np.exp((best_c5 - c5_try) / temperature):
            h = h_try
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h.copy()
                
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    # Final polish with SLSQP
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    try:
        result = minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            best_c5 = compute_c5(best_h)
    except:
        pass  # If local search fails, keep original

    return (best_h.tolist(), best_c5, n_points)
