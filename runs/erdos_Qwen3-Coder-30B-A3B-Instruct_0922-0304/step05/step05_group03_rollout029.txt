import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from scipy.stats import halton

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available, else use a simple step function
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback: piecewise constant step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)

    def compute_c5(h):
        return evaluate_c5(h, dx)
        
    # Generate multiple initial seeds using Halton sequence for diversity
    halton_seq = halton.Halton(n_points)
    seeds = halton_seq.random(50)
    seeds = np.clip(seeds, 0, 1)  # Ensure in [0,1]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start optimization with SLSQP
    for i in range(50):
        # Use Halton-sequence based perturbation
        h_start = seeds[i] * 0.8 + 0.1  # Scale to reasonable range
        h_start = normalize_h(h_start, target_sum)
        
        def objective(h):
            return compute_c5(h)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        # Run SLSQP optimization with constraints
        result = minimize(
            objective,
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            h_opt = result.x
            h_opt = normalize_h(h_opt, target_sum)
            c5_val = compute_c5(h_opt)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
        
        # Check if we have time to spare before running out
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
