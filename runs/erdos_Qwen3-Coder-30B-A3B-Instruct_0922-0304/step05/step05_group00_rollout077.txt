import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_h(h, target_sum):
    """Project h to feasible region [0,1] and normalize."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from best known construction or simple step function
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Allow for more iterations with time control
    max_iter = 1000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Identify critical lags that contribute to max overlap
        overlap = correlate(h, 1 - h, mode="full")
        # Focus on top 10% of the lags
        top_lag_indices = np.argsort(overlap)[-max(1, n_points//10):]
        
        # Apply gradient descent with constraints on selected lags
        def objective(h):
            return evaluate_c5(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Use SLSQP with local search
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, 
                          options={'maxiter': 100, 'ftol': 1e-9, 'gtol': 1e-9})
        
        if result.success:
            h_new = result.x
            h_new = project_h(h_new, target_sum)
            c5_new = evaluate_c5(h_new, dx)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                
            # Early exit if we've reached target
            if best_c5 <= 0.3807:
                break
                
            h = h_new
        else:
            # If optimization failed, try small random perturbations
            h = h + np.random.normal(0, 0.005, n_points)
            h = project_h(h, target_sum)
            
        iteration += 1
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
