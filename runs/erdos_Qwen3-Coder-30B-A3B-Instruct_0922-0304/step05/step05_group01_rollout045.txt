import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize particles
    n_particles = 20
    particles = []
    velocities = []
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Create a good starting guess
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize each particle
    for _ in range(n_particles):
        h = h_init.copy()
        # Add some noise
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Best particle tracking
    best_particle = h_init.copy()
    best_c5 = float('inf')
    
    # PSO parameters
    inertia_w = 0.7
    cognitive_c = 1.5
    social_c = 1.5
    max_velocity = 0.2
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        for i, particle in enumerate(particles):
            # Compute C5
            overlap = correlate(particle, 1 - particle, mode="full")
            c5 = np.max(overlap) * dx
            
            # Update personal best
            if c5 < best_c5:
                best_c5 = c5
                best_particle = particle.copy()
            
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
                
        # Update particles
        for i in range(n_particles):
            particle = particles[i]
            velocity = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive = cognitive_c * r1 * (particle - particle)
            social = social_c * r2 * (best_particle - particle)
            velocity = inertia_w * velocity + cognitive + social
            
            # Clip velocity
            velocity = np.clip(velocity, -max_velocity, max_velocity)
            
            # Update position
            particle += velocity
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            # Update velocity for next iteration
            velocities[i] = velocity
            
        # Decay inertia weight
        inertia_w = max(0.4, inertia_w - 0.001)
            
    # Final evaluation
    overlap = correlate(best_particle, 1 - best_particle, mode="full")
    final_c5 = np.max(overlap) * dx
    
    return (best_particle.tolist(), final_c5, n_points)
