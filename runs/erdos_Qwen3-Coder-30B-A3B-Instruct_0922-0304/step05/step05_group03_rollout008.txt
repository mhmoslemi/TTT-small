import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, k_max):
    """Compute the gradient of the overlap at k_max with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    # For overlap at lag k_max, we look at correlation sum_{i} h[i] * (1 - h[i+k_max])
    # Gradient wrt h[i] = (1 - h[i+k_max]) if i+k_max < n else 0
    for i in range(n):
        if i + k_max < n:
            grad[i] = 1 - h[i + k_max]
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    # Gradient descent parameters
    alpha = 0.1
    patience = 0
    patience_limit = 100
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > patience_limit:
                alpha *= 0.5
                patience = 0
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Find the lag with maximum overlap
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        
        # Compute gradient and update
        grad = compute_gradient(h, dx, k_max)
        h = h - alpha * grad
        h = normalize_h(h, target_sum)
        
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
