import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have sum equal to target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

def project_to_feasible(h, n_points):
    """Project h to feasible region with constraints sum(h) = n_points/2 and 0 <= h_i <= 1."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scale = (n_points/2) / current_sum
        h = h * scale
        h = np.clip(h, 0, 1)
    return h

def solve_convex_relaxation(h, n_points, dx):
    """Solve convex relaxation to improve h."""
    # Create a vector to hold variables
    h_vars = cp.Variable(n_points)
    
    # Create an expression to hold the constraints
    # We'll compute using the current h to form the constraints
    
    # Precompute the convolution values for current h
    conv_vals = correlate(h, 1 - h, mode="full")
    max_k = len(conv_vals) - 1
    n_k = max_k + 1
    C_bounds = conv_vals * dx
    
    # Additional variables for the maximum
    C = cp.Variable()
    
    # Constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == n_points / 2,
        C >= C_bounds[k] for k in range(n_k)
    ]
    
    # Objective: minimize C
    objective = cp.Minimize(C)
    
    # Problem setup
    prob = cp.Problem(objective, constraints)
    
    try:
        # Solve the convex problem
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status == cp.OPTIMAL:
            return h_vars.value
    except:
        pass
    return h

def perturb_solution(h, n_points, iteration):
    """Add small perturbations to escape local optima."""
    # Decrease perturbation strength over time
    noise_level = 0.1 * (0.9 ** (iteration // 50))
    noise = np.random.normal(0, noise_level, n_points)
    h_new = h + noise
    h_new = np.clip(h_new, 0, 1)
    h_new = normalize_h(h_new, n_points / 2)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    n_iter = 0
    best_c5 = float('inf')
    best_h = None
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values)
        init_h = project_to_feasible(init_h, n_points)
    else:
        # Create a random initial solution
        init_h = np.random.rand(n_points)
        init_h = normalize_h(init_h, n_points / 2)
    
    h = init_h.copy()
    best_h = h.copy()
    current_c5 = compute_c5(h, dx)
    best_c5 = current_c5
    
    while time.time() - start_time < budget_s - 1:
        # Every few iterations, try to improve with convex relaxation
        if n_iter % 20 == 0:
            h_relaxed = solve_convex_relaxation(h, n_points, dx)
            h_relaxed = project_to_feasible(h_relaxed, n_points)
            c5_relaxed = compute_c5(h_relaxed, dx)
            
            if c5_relaxed < current_c5:
                h = h_relaxed
                current_c5 = c5_relaxed
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        
        # Perturb occasionally to escape local minima
        if n_iter % 50 == 0:
            h = perturb_solution(h, n_points, n_iter)
            current_c5 = compute_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # Minor local improvement using slight perturbations
            h_new = perturb_solution(h, n_points, n_iter)
            c5_new = compute_c5(h_new, dx)
            if c5_new < current_c5:
                h = h_new
                current_c5 = c5_new
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()

        # Early exit if target is achieved
        if best_c5 <= 0.3807:
            break

        n_iter += 1

    # Final evaluation of best solution
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
