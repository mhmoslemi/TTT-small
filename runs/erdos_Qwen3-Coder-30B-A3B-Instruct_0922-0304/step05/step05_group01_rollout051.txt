import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 20
    elite_fraction = 0.25
    mutation_strength = 0.01
    
    # Initialize population
    def initialize_population():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
        
        population = [h.copy()]
        for _ in range(population_size - 1):
            h_mut = h + np.random.normal(0, mutation_strength, n_points)
            h_mut = np.clip(h_mut, 0, 1)
            h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
            population.append(h_mut)
        return population
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Initialize population
    population = initialize_population()
    
    # Track best solution
    best_c5 = float('inf')
    best_h = None
    
    # Main EA loop
    max_generations = 1000
    no_improvement = 0
    max_no_improvement = 100
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = [compute_c5(h) for h in population]
        
        # Update best solution
        min_fitness = min(fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitness.index(min_fitness)].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Check for stagnation
        if no_improvement > max_no_improvement:
            # Restart with new random population
            population = initialize_population()
            no_improvement = 0
            continue
            
        # Selection: top 25%
        elite_indices = np.argsort(fitness)[:int(population_size * elite_fraction)]
        elite = [population[i] for i in elite_indices]
        
        # Mutation: generate offspring
        new_population = []
        for individual in elite:
            # Create two mutated offspring
            for _ in range(2):
                mutated = individual + np.random.normal(0, mutation_strength, n_points)
                mutated = np.clip(mutated, 0, 1)
                mutated = mutated * (n_points / 2) / np.sum(mutated)
                new_population.append(mutated)
                
        # Ensure population size maintained
        while len(new_population) < population_size:
            new_population.append(np.random.rand(n_points))
            new_population[-1] = new_population[-1] * (n_points / 2) / np.sum(new_population[-1])
            
        population = new_population[:population_size]
    
    # Final evaluation
    if best_h is None:
        # Fallback: evaluate all members of the final population
        fitness = [compute_c5(h) for h in population]
        best_c5 = min(fitness)
        best_h = population[fitness.index(best_c5)].copy()
    
    return (best_h.tolist(), best_c5, n_points)
