import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to maintain sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        # Default initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    
    # Parameters
    learning_rate = 0.1
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        # Compute all correlations
        corr_full = correlate(h, 1 - h, mode="full")
        # Find the index of maximum correlation
        max_idx = np.argmax(corr_full)
        # Compute the actual shift k
        k = max_idx - (n_points - 1)
        
        # Compute gradient for this shift
        # h * (1 - h_shifted) where shift k means we take h[i] * (1 - h[i-k])
        # The gradient direction is (1 - h_shifted) at active position
        # For shift k, we compute (1 - h_shifted) where h_shifted[i] = h[i-k]
        h_shifted = np.roll(h, k)
        gradient = (1 - h_shifted) * dx
        
        # Update using gradient ascent (minimize C5)
        h = h - learning_rate * gradient
        
        # Project back to feasibility
        h = project_h(h, target_sum)
        
        # Evaluate current solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Adaptive learning rate
        if iter_count % 100 == 0:
            # Add noise
            h += np.random.normal(0, 0.005, n_points)
            h = project_h(h, target_sum)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        iter_count += 1
        
        # Adjust learning rate
        if iter_count % 10 == 0:
            if current_c5 < best_c5:
                learning_rate *= 1.01
            else:
                learning_rate *= 0.9
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
