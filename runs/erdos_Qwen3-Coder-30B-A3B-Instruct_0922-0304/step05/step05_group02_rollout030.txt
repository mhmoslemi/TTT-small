import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, target_sum)
    else:
        # Create a good starting point
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = project_h(h, target_sum)
    
    # Optimization parameters
    momentum = 0.95
    step_size = 0.05
    max_iterations = 10000
    patience = 50
    best_c5 = float('inf')
    best_h = h.copy()
    prev_c5 = float('inf')
    no_improvement_count = 0
    
    # Initialize momentum
    m = np.zeros(n_points)
    
    for i in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        
        # Compute gradient for the maximum shift
        h_shifted = np.zeros_like(h)
        if k_max < n_points:
            h_shifted[k_max:] = h[:-k_max]
        else:
            h_shifted[:n_points - k_max] = h[k_max:]
            
        # Gradient direction
        grad = 1 - h_shifted
        grad = np.clip(grad, -1, 1)  # Clip gradient for stability
        
        # Momentum update
        m = momentum * m + (1 - momentum) * grad
        
        # Update with proximal operator
        h_new = h - step_size * m
        h_new = project_h(h_new, target_sum)
        
        # Evaluate new solution
        new_c5 = compute_c5(h_new, dx)
        
        # Accept improvement
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Adaptive step size based on convergence
        if new_c5 < prev_c5:
            step_size *= 1.05
        else:
            step_size *= 0.9
            
        step_size = max(1e-6, min(0.5, step_size))
        
        # Update h
        h = h_new
        prev_c5 = new_c5
        
        if no_improvement_count > patience:
            # Reset with a slightly perturbed solution
            h = best_h.copy() + np.random.normal(0, 0.01, n_points)
            h = project_h(h, target_sum)
            no_improvement_count = 0
            step_size = 0.05
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
