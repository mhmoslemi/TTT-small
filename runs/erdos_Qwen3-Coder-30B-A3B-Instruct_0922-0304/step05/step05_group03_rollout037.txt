import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, top_lag_idx):
    """Compute gradient of overlap with respect to h."""
    n = len(h)
    gradient = np.zeros(n)
    
    # Compute correlation of h with 1-h for all lags
    corr = correlate(h, 1 - h, mode="full")
    
    # The gradient is the derivative of correlation w.r.t. h
    # At lag k, the derivative of h[i] is -1 if i = k or i = k + n - 1
    # This needs to be computed carefully for the full correlation
    # Let's compute the actual gradient at the position of max overlap
    for i in range(n):
        # Gradient w.r.t. h[i]
        grad = 0
        for k in range(-n + 1, n):
            # The contribution of h[i] to the correlation at lag k
            if 0 <= i - k < n:
                grad += 1 - h[i - k]
            # The contribution of (1-h[i]) to the correlation at lag k
            if 0 <= i + k < n:
                grad += h[i + k]
        gradient[i] = grad * dx
    return gradient

    best_c5 = float('inf')

        # For gradient descent, approximately compute contribution of each h[i]

        grad = np.zeros(n_points)

            # Compute approximate gradient w.r.t. h[i]
            temp_h = h.copy()

            temp_h[i] += epsilon
            temp_h = normalize_h(temp_h, target_sum)
            new_c5 = evaluate_c5(temp_h, dx)
