import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Simulated Annealing
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    # Initial temperature based on expected energy difference
    T0 = 0.05
    T = T0
    cooling_rate = 0.999
    min_temp = 1e-8
    max_iter = 50000
    
    # Perturbation parameters
    sigma = 0.05
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate neighbor by adding Gaussian noise
        h_new = h + np.random.normal(0, sigma, n_points)
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to maintain constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new objective
        c5_new = compute_c5(h_new)
        
        # Accept or reject
        delta = c5_new - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        # Cooling schedule
        T = max(min_temp, T * cooling_rate)
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
