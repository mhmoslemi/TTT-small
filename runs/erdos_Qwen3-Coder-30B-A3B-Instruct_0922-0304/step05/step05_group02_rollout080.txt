import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def compute_gradient(h, k_max, n_points):
    """Compute gradient for the maximum overlap shift."""
    # Create shifted version of h
    h_shifted = np.zeros(n_points)
    if k_max >= 0:
        h_shifted[k_max:] = h[:-k_max] if k_max < n_points else np.zeros(0)
    else:
        h_shifted[:k_max] = h[-k_max:] if -k_max < n_points else np.zeros(0)
    
    # Gradient is (1 - h_shifted)
    return 1 - h_shifted

def line_search(h, grad, step_size, dx, max_iter=20):
    """Perform line search to find good step size."""
    alpha = step_size
    current_c5 = compute_c5(h, dx)
    
    for _ in range(max_iter):
        h_new = project_h(h - alpha * grad, len(h) / 2)
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < current_c5:
            return h_new, alpha
        alpha *= 0.5
    return h, step_size

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, n_points / 2)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = project_h(h, n_points / 2)
    
    # Optimization parameters
    momentum = 0.99
    step_size = 0.1
    beta = 1.0
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Compute correlations
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        max_overlap = corr[k_max]
        
        # Update best solution
        current_c5 = max_overlap * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iter_count
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient (direct computation of gradient for max overlap)
        grad = compute_gradient(h, k_max - (n_points - 1), n_points)
        
        # Accelerated gradient update
        if iter_count == 1:
            v = grad
        else:
            v = momentum * v + grad
            
        # Line search
        h, step_size = line_search(h, v, step_size, dx)
        
        # Adaptive learning rate
        if iter_count % 10 == 0:
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 1e-6:
                step_size = min(0.1, 1.0 / grad_norm)
        
        # Check for stagnation
        if iter_count - last_improvement > 50:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
