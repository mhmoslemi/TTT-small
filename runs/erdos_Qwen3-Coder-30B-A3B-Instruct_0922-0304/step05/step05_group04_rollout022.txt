import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        # Add some noise
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap)
        return max_overlap * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Binary search phase
    def create_binary_vector(n, num_ones):
        indices = np.random.choice(n, num_ones, replace=False)
        h_bin = np.zeros(n)
        h_bin[indices] = 1
        return h_bin
    
    def mutate_binary(h_bin):
        # Swap two random positions
        i, j = np.random.choice(len(h_bin), 2, replace=False)
        h_new = h_bin.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        return h_new
    
    # Simulated Annealing for binary search
    T = 1.0
    alpha = 0.99
    max_iter = 10000
    num_ones = n_points // 2
    
    binary_h = create_binary_vector(n_points, num_ones)
    current_c5 = compute_c5(binary_h)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = binary_h.copy()
    
    # Annealing process
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_candidate = mutate_binary(binary_h)
        candidate_c5 = compute_c5(h_candidate)
        
        # Acceptance probability
        if candidate_c5 < current_c5 or np.random.rand() < np.exp(-(candidate_c5 - current_c5) / T):
            binary_h = h_candidate
            current_c5 = candidate_c5
            
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = binary_h.copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        T *= alpha
        if T < 1e-4:
            break

    # Continuous relaxation with CVXPY
    # Use ECOS solver
    h_var = cp.Variable(n_points)
    objective = cp.Minimize(cp.max(cp.conv(h_var, cp.reverse(1 - h_var))))
    
    # Constraints
    constraints = [
        cp.sum(h_var) == n_points / 2,
        h_var >= 0,
        h_var <= 1
    ]
    
    # Formulate problem
    prob = cp.Problem(objective, constraints)
    
    # Initial guess from best binary solution
    h_init = best_h.copy()
    h_init = np.clip(h_init, 0, 1)
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Try to solve
    try:
        prob.solve(solver=cp.ECOS, warm_start=True)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_continuous = h_var.value
            h_continuous = np.clip(h_continuous, 0, 1)
            h_continuous = h_continuous * (n_points / 2) / np.sum(h_continuous)
            continue_c5 = compute_c5(h_continuous)
            if continue_c5 < best_c5:
                best_c5 = continue_c5
                best_h = h_continuous.copy()
    except Exception:
        # If CVXPY fails, proceed with best so far
        pass

    # Final evaluation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
