import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 50
    mutation_rate = 0.1
    tournament_size = 10
    max_generations = 1000
    
    # Initialize population
    def initialize_population():
        if 'initial_h_values' in globals():
            # Use the provided initial construction
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population = [h.copy()]
        else:
            population = []
        
        # Add random individuals
        for _ in range(pop_size - len(population)):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
        return population
    
    # Compute C5 for a given h
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Evaluate individuals and return fitness
    def evaluate_fitness(population):
        fitness = []
        for h in population:
            c5 = compute_c5(h)
            fitness.append(c5)
        return fitness
    
    # Tournament selection
    def tournament_selection(population, fitness):
        selected = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_index].copy())
        return selected
    
    # Uniform crossover
    def uniform_crossover(parent1, parent2):
        mask = np.random.rand(n_points) < 0.5
        child = np.where(mask, parent1, parent2)
        
        # Repair to maintain sum constraint
        child = child * (n_points / 2) / np.sum(child)
        return np.clip(child, 0, 1)
    
    # Mutation with repair
    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                if np.random.rand() < 0.5:
                    # Add or subtract small value
                    delta = np.random.rand() * 0.2 - 0.1
                    h[i] += delta
                else:
                    # Set to 0 or 1
                    h[i] = np.random.choice([0, 1])
        # Repair to sum constraint
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    # Main GA loop
    population = initialize_population()
    best_h = None
    best_c5 = float('inf')
    
    generation = 0
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        generation += 1
        
        # Evaluate fitness
        fitness = evaluate_fitness(population)
        
        # Track best solution
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected = tournament_selection(population, fitness)
        
        # Crossover and mutation
        new_population = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i+1) % pop_size]
            child1 = uniform_crossover(parent1, parent2)
            child2 = uniform_crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        # Maintain population size
        population = new_population[:pop_size]
    
    # final computation of best solution
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
