import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum=h."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient descent parameters
    eta = 0.1
    noise_std = 0.01
    noise_freq = 100
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute all cross-correlations
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        # Compute gradient from the maximal shift
        if max_k < n_points:
            # h * (1 - h_shifted) where h_shifted is h shifted right by (n_points - max_k)
            shift = n_points - max_k
            grad_h = np.concatenate([np.zeros(shift), 1 - h[:-shift]]) * dx
        else:
            # Tail case: h_shifted is h shifted left by (max_k - n_points + 1)
            shift = max_k - n_points + 1
            grad_h = np.concatenate([1 - h[shift:], np.zeros(shift)]) * dx
        
        # Gradient descent step
        h = h - eta * grad_h
        
        # Project to feasible region
        h = project_h(h, target_sum)
        
        # Periodic noise injection
        if iteration % noise_freq == 0:
            noise = np.random.normal(0, noise_std, n_points)
            h = h + noise
            h = project_h(h, target_sum)
        
        # Compute current C5
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Adapt learning rate
        eta = eta * 1.01 if current_c5 < best_c5 else eta * 0.9
        
        iteration += 1
        
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
