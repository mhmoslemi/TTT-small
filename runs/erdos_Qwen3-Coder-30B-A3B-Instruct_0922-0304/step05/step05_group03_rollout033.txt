import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient of overlap at k_max with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    # The correlation at lag k_max is sum_{i=0}^{n-1} h[i] * (1 - h[i+k_max])
    # So gradient w.r.t. h[i] is -1 * (1 - h[i+k_max]) if k_max >= 0 and i+k_max < n
    # And 1 * h[i-k_max] if k_max <= 0 and i-k_max >= 0
    
    # This is not exact, but a reasonable approximation for gradient
    # Let's compute it with correlation dx
    
    # For each i, we compute how much it contributes to the overlap at k_max
    for i in range(n):
        if i + k_max >= 0 and i + k_max < n:
            grad[i] = -(1 - h[i + k_max])
        if i - k_max >= 0 and i - k_max < n and k_max > 0:
            grad[i] += h[i - k_max]
    
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent parameters
    learning_rate = 0.1
    max_iterations = int(budget_s * 200)  # Roughly 200 iterations per second
    patience = 100
    patience_counter = 0
    
    # Precompute indices for faster access
    indices = np.arange(n_points)
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Early exit if we're close enough
        if best_c5 <= 0.3807:
            break
            
        if patience_counter >= patience:
            learning_rate *= 0.5
            patience_counter = 0
            
        # Find the lag that maximizes overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(overlap)
        # Convert from full correlation indices to actual lag
        k_max = max_overlap_idx - (n_points - 1)
        
        # Compute gradient for that k_max
        grad = compute_gradient(h, dx, k_max)
        
        # Update h
        h = h - learning_rate * grad
        
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Re-normalize
        h = normalize_h(h, target_sum)
        
        # Small random perturbation occasionally to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            noise = np.random.normal(0, 0.001, n_points)
            h = h + noise
            h = normalize_h(h, target_sum)

    # Final check
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
