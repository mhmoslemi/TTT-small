import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, k_max, dx):
    """Compute gradient of overlap at k_max with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    
    # For overlap at lag k_max, the gradient is:
    # ∂/∂h[i] ∫ h(x) (1 - h(x + k_max)) dx = (1 - h[i + k_max]) if i + k_max < n else 0
    # This is assuming the correlation is computed correctly considering the full mode.
    # However, we want the gradient of the overlap directly at k_max.
    
    # To compute ∂ overlap[k_max]/∂h[i]
    # overlap[k_max] = sum_{x} h[x] * (1 - h[x + k_max]) 
    # If we index differently (0-based):
    # overlap[full_idx] = sum_{x} h[x] * (1 - h[x + k_max]) 
    # where full_idx = k_max + n_points - 1 (in scipy's convention for full overlap)
    # or we take the correlation with mode="full" and look at the appropriate index.
    
    # Let's directly compute the gradient via finite difference on overlap for clarity,
    # but correct approach would be to look at the definition:
    # overlap[(k_max + n - 1) % (2 * n - 1)] for scipy's indexing
    # OR compute the derivative manually as a sum over valid i such that overlap_{k_max} includes h[i]
    
    # However, for exact gradients we have:
    # ∂/(∂h[i]) sum_j h[j] (1 - h[i+j])
    # = (1 - h[i+k_max]) if i+k_max < n  
    # = 0 otherwise.

    # Go through the correlation indices
    if k_max >= 0:
        for i in range(n):
            j = i + k_max
            if j < n:
                # Overlap contribution as sum over h[i] and h[j]
                # But derivative is with respect to h[i], so if j = i+k_max,
                # ∂(h[i]*h[j])/∂h[i] = h[j]  = h[i+k_max]
                # But we want derivative of overlap = sum h[i] (1 - h[i + k_max])
                # So ∂/∂h[i] = - (1 - h[i + k_max]) = h[i + k_max] - 1
                
                # Actually, wait the appropriate term from the correlation:
                # overlap[i]  = sum_{j} h[j] * h[i-j] : incorrect interpretation
                # For correlate(h, 1-h, mode='full') we compute:
                # C[k] = sum_x h[x] * (1 - h[x + k])
                # Since corr(a,b) computes sum_x a[x] * b[x + shift]
                # our relevant correlation at shift k is:
                # C_k = sum_x h[x] * (1 - h[x + k]) 
                # derivative d C_k / d h[i] = - (1 - h[i + k]) if i+k exists within valid range.
                # = -1 + h[i + k] if i+k < n else 0
                grad[i] = h[i + k_max] - 1 if i + k_max < n else 0
            else:
                grad[i] = 0
    else:
        for i in range(n):
            j = i + k_max # this is negative, so will be out of bounds for j >= n
            if j >= 0:
                grad[i] = h[j] - 1 if j < n else 0
            else:
                grad[i] = 0
    
    return grad * dx  # Scale by dx since correlation is scaled by dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    # Optimization parameters
    learning_rate = 0.1
    patience = 50
    patience_count = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    prev_c5 = float('inf')
    
    # Iteration loop
    max_iter = int(budget_s * 10)  # Roughly scale by safety factor
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute overlap to find k_max
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap) - (n_points - 1)  # Convert to actual lag value
        
        # Compute gradient
        grad = compute_gradient(h, k_max, dx)
        
        # Update h
        h_new = h - learning_rate * grad
        
        # Wrap around the boundary of the correlation for gradient calculation of wrap-around lands
        # But the mode is full, which simulates circular shift or accumulates extended conv.
        # So we compute h[i] - lr * grad[i] and then bring back to [0,1] and normalize to sum
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize
        h_new = normalize_h(h_new, target_sum)
        
        h = h_new
        
        # Monitor convergence
        if current_c5 >= prev_c5:
            patience_count += 1
            if patience_count >= patience:
                learning_rate *= 0.5
                patience_count = 0
        else:
            patience_count = 0
            
        prev_c5 = current_c5
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
