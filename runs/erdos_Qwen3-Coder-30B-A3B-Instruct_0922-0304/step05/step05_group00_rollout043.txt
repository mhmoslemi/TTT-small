import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def repair_feasibility(h, target_sum):
    """Repair h to be within bounds and sum to target_sum."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a simple initial guess
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = normalize_h(h, target_sum)

    # Initialize
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Hyperparameters
    population_size = 30
    max_iter = 10000
    step_size_de = 0.5
    step_size_gd = 0.01
    step_size_mutation = 0.05
    step_decay = 0.98
    elite_size = 3
    
    # Population initialization with some diversity
    population = [h.copy()]
    for _ in range(population_size - 1):
        h_m = h + np.random.normal(0, step_size_mutation, n_points)
        h_m = repair_feasibility(h_m, target_sum)
        population.append(h_m)
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate population
        fitness = [compute_c5(ind, dx) for ind in population]
        
        # Update best solution
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_h = population[min_fit_idx].copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
        
        # Sort population
        sorted_indices = np.argsort(fitness)
        elite = [population[i] for i in sorted_indices[:elite_size]]
        
        # Generate new population
        new_population = elite.copy()
        
        # Differential evolution
        for i in range(population_size - elite_size):
            a, b, c = np.random.choice(len(elite), 3, replace=False)
            mutant = elite[a] + step_size_de * (elite[b] - elite[c])
            mutant = repair_feasibility(mutant, target_sum)
            
            # Crossover
            crossover_rate = 0.7
            j_rand = np.random.randint(n_points)
            for j in range(n_points):
                if np.random.rand() > crossover_rate and j != j_rand:
                    mutant[j] = population[np.random.randint(len(population))][j]
            
            new_population.append(mutant)
        
        # Gradient descent on best individuals
        for i in range(elite_size):
            if i == 0:  # Use mutation on the best
                h_new = best_h + np.random.normal(0, step_size_mutation, n_points)
            else:  # Use mutation on elite members
                h_new = elite[i] + np.random.normal(0, step_size_mutation, n_points)
            
            h_new = repair_feasibility(h_new, target_sum)
            # Simple gradient ascent step (approximate with finite differences)
            h_plus = h_new.copy() + 1e-4
            h_plus = repair_feasibility(h_plus, target_sum)
            h_minus = h_new.copy() - 1e-4
            h_minus = repair_feasibility(h_minus, target_sum)
            
            grad = (compute_c5(h_plus, dx) - compute_c5(h_minus, dx)) / (2e-4)
            h_new = h_new - step_size_gd * grad * (h_new - best_h)
            h_new = repair_feasibility(h_new, target_sum)
            
            new_population.append(h_new)
            
        # Update population
        population = new_population[:population_size]
        
        # Adjust step sizes
        step_size_de *= step_decay
        step_size_gd *= step_decay
        step_size_mutation *= step_decay
        
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
