import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradients(h, dx, k_max):
    """Compute gradients of C5 with respect to h."""
    # Compute the overlap at all lags
    overlap = correlate(h, 1 - h, mode="full")
    # Find the lag with maximum overlap
    max_idx = np.argmax(overlap)
    max_lag = max_idx - (len(h) - 1)
    
    # Initialize gradient vector
    grad = np.zeros_like(h)
    
    # Compute gradient for maximum lag
    if max_lag >= 0:
        # h[i] contributes to overlap at positions 0, 1, ..., k_max
        for i in range(len(h)):
            for k in range(min(k_max, len(h) - i)):
                if i + k < len(h):
                    grad[i] += (1 - h[i + k]) * dx
    else:
        # h[i] contributes to overlap in negative lags
        for i in range(len(h)):
            for k in range(min(len(h), i + 1)):
                if i - k >= 0:
                    grad[i] += h[i - k] * dx
    
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = normalize_h(h, target_sum)
        if np.isnan(h).any():
            h = np.ones(n_points)
            h = normalize_h(h, target_sum)
    else:
        # Fallback: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    # Factor to adjust learning rate
    lr_factor = 1.0
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # For 1000 iterations or until budget is almost used
    iteration = 0
    max_iter = 10000
    
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if we've reached our target
        if best_c5 <= 0.3807:
            break
            
        # Alternate between gradient descent and coordinate-wise updates
        if iteration % 2 == 0:
            # Gradient descent step
            grad = compute_gradients(h, dx, n_points)
            
            # Adjust learning rate
            lr = 0.1 * lr_factor
            h_new = h - lr * grad
            h_new = normalize_h(h_new, target_sum)
            
            # Evaluate move
            new_c5 = evaluate_c5(h_new, dx)
            if new_c5 < current_c5:
                h = h_new
            else:
                # Reduce learning rate
                lr_factor *= 0.5
                
        else:
            # Coordinate-wise local search
            h_backup = h.copy()
            for i in range(n_points):
                # Try small positive and negative changes
                step = 0.01
                test_h = h.copy()
                test_h[i] += step
                test_h = normalize_h(test_h, target_sum)
                c5_test = evaluate_c5(test_h, dx)
                
                if c5_test < current_c5:
                    h = test_h
                    current_c5 = c5_test
                else:
                    # Try negative step
                    test_h = h.copy()
                    test_h[i] -= step
                    test_h = normalize_h(test_h, target_sum)
                    c5_test = evaluate_c5(test_h, dx)
                    if c5_test < current_c5:
                        h = test_h
                        current_c5 = c5_test
                        
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
