import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h):
    """Compute the C5 overlap integral for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * (2.0 / len(h))

def mutate(h, mutation_std, n_points):
    """Mutate h with Gaussian noise and ensure constraints."""
    mutated = h + np.random.normal(0, mutation_std, h.shape)
    mutated = np.clip(mutated, 0, 1)
    sum_h = np.sum(mutated)
    mutated = mutated * (n_points / 2) / sum_h
    return mutated

def evaluate_fitness(population):
    """Evaluate C5 for all individuals in population."""
    return [compute_c5(individual) for individual in population]

def select_elite(population, fitnesses, elite_frac=0.25):
    """Select top individuals based on fitness."""
    indices = np.argsort(fitnesses)
    elite_count = max(1, int(len(population) * elite_frac))
    return [population[i] for i in indices[:elite_count]]

def update_best(population, fitnesses, best_h, best_c5):
    """Update the best solution found."""
    min_idx = np.argmin(fitnesses)
    if fitnesses[min_idx] < best_c5:
        best_c5 = fitnesses[min_idx]
        best_h = population[min_idx].copy()
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize population
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum

    population_size = 20
    population = [h.copy()]
    for _ in range(population_size - 1):
        mutated = mutate(h, 0.01, n_points)
        population.append(mutated)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Evolutionary algorithm parameters
    max_generations = 1000
    mutation_strength = 0.01
    no_improvement_count = 0
    max_no_improvement = 50
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for all population members
        fitnesses = evaluate_fitness(population)
        
        # Update best solution
        best_h, best_c5 = update_best(population, fitnesses, best_h, best_c5)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Select elite
        elite = select_elite(population, fitnesses)
        
        # Create new population through mutation
        new_population = []
        for individual in elite:
            for _ in range(2):  # Create two offspring per elite
                mutated = mutate(individual, mutation_strength, n_points)
                new_population.append(mutated)
                
        # Ensure population size
        while len(new_population) < population_size:
            new_population.append(mutate(h, mutation_strength, n_points))
        
        population = new_population[:population_size]
        
        # Adaptive mutation
        if no_improvement_count > max_no_improvement:
            mutation_strength *= 0.1
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
    # Final evaluation
    final_fitness = compute_c5(best_h)
    
    return (best_h.tolist(), final_fitness, n_points)
