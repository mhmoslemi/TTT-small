import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def validate_and_repair(h):
    h = np.array(h)
    h = np.clip(h, 0, 1)
    h = h * (70 / 2) / np.sum(h)
    return h

def initialize_coarse():
    h_coarse = np.ones(35)
    h_coarse[17:] = 0
    return h_coarse

def upscale(h_coarse, n_points):
    h_fine = np.repeat(h_coarse, 2)
    h_fine = h_fine[:n_points]
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    return h_fine

def compute_c5(h):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * (2.0 / len(h))

def constraint_sum(h):
    return np.sum(h) - 70 / 2
