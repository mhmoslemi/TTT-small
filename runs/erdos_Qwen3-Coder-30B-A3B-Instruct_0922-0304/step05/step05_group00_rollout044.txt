import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def repair_feasibility(h, target_sum):
    """Repair h to maintain feasibility constraints."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def evaluate_individual(h, dx):
    """Evaluate objective and check constraints."""
    c5 = compute_c5(h, dx)
    return c5

def diff_evo_step(population, fitness, F=0.8, CR=0.9):
    """Differential evolution step."""
    new_pop = []
    for i in range(len(population)):
        idxs = np.random.choice(len(population), 3, replace=False)
        a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
        mutant = a + F * (b - c)
        trial = np.where(np.random.rand(len(mutant)) < CR, mutant, population[i])
        new_pop.append(trial)
    return new_pop

def local_descent_step(h, dx, target_sum, max_iter=50):
    """Local descent step using SLSQP."""
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    def objective(h):
        return compute_c5(h, dx)
    
    bounds = [(0, 1) for _ in range(len(h))]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': max_iter})
    if result.success:
        return np.clip(result.x, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 30
    population = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    
    # Generate initial population
    for _ in range(population_size):
        h = base_h + np.random.normal(0, 0.02, n_points)
        h = repair_feasibility(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_individual = None
    
    step_size = 0.05
    iteration = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate entire population
        fitness = [evaluate_individual(h, dx) for h in population]
        
        # Track best individual so far
        min_fit_idx = np.argmin(fitness)
        if fitness[min_fit_idx] < best_c5:
            best_c5 = fitness[min_fit_idx]
            best_individual = population[min_fit_idx].copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Generate new population
        new_population = []
        
        # Elitism: keep the best individual
        new_population.append(best_individual.copy())
        
        # Differential evolution (40%)
        for _ in range(int(0.4 * (population_size - 1))):
            new_pop = diff_evo_step(population, fitness)
            for h in new_pop:
                h = repair_feasibility(h, target_sum)
                new_population.append(h)
                
        # Local descent (30%)
        for _ in range(int(0.3 * (population_size - 1))):
            idx = np.random.randint(len(population))
            h = local_descent_step(population[idx], dx, target_sum)
            new_population.append(h)
            
        # Random sampling (20%)
        for _ in range(int(0.2 * (population_size - 1))):
            h = np.random.uniform(0, 1, n_points)
            h = repair_feasibility(h, target_sum)
            new_population.append(h)
            
        # Mutation of best (10%)
        for _ in range(int(0.1 * (population_size - 1))):
            h = best_individual + np.random.normal(0, 0.01 * step_size, n_points)
            h = repair_feasibility(h, target_sum)
            new_population.append(h)
            
        # Fill up to population_size
        while len(new_population) < population_size:
            h = base_h + np.random.normal(0, 0.02, n_points)
            h = repair_feasibility(h, target_sum)
            new_population.append(h)
            
        # Replace population
        population = new_population[:population_size]
        
        # Adjust step size
        step_size *= 0.98
        iteration += 1
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
