```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have sum equal to target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

def project_to_feasible(h, n_points):
    """Project h to feasible region: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        scale = (n_points/2) / current_sum
        h = h * scale
    return h

def solve_convex_subproblem(h, dx, n_points):
    """Solve the convex subproblem using CVXPY."""
    # Create optimization variables
    h_var = Variable(n_points)
    C = Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, sum(h_var) == n_points/2]
    
    # Compute correlations with current h
    corrs = correlate(h, 1-h, mode="full")
    # Now constraint C >= max(corrs) * dx
    for i in range(len(corrs)):
        constraints.append(C >= corrs[i] * dx)
    
    # Objective
    objective = Minimize(C)
    
    # Solve
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=CPLEX, verbose=False, max_iters=1000)
        if prob.status in (OPTIMAL, OPTIMAL_INACCURATE):
            new_h = h_var.value
            # Project to feasible set
            new_h = project_to_feasible(new_h, n_points)
            return new_h
    except:
        pass
    
    return h

def perturb_solution(h, iteration, n_points, max_perturb=0.1):
    """Apply controlled perturbation."""
    if iteration % 50 == 0 and iteration > 0:
        decay = 0.9 ** (iteration // 50)
        perturb = max_perturb * decay
        noise = np.random.normal(0, perturb, n_points)
        h = h + noise
        h = project_to_feasible(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values)
        init_h = np.clip(init_h, 0, 1)
        init_h = normalize_h(init_h, n_points / 2)
    else:
        init_h = np.ones(n_points)
        init_h[n_points//2:] = 0
        init_h += np.random.normal(0, 0.05, n_points)
        init_h = np.clip(init_h, 0, 1)
        init_h = normalize_h(init_h, n_points / 2)

    # Initialize best
    best_h = init_h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Main loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Solve convex subproblem
        new_h = solve_convex_subproblem(best_h, dx, n_points)
        
        # Evaluate new solution
        new_c5 = compute_c5(new_h, dx)
        
        # Update best if improved
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
        
        # Apply perturbations
        best_h = perturb_solution(best_h, iteration, n_points)
        
        iteration += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
```
