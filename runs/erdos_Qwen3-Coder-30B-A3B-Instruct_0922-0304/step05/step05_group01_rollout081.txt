import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def validate_and_repair(h_init, n_points):
    h = np.array(h_init)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def initialize_coarse():
    # Initialize with a reasonable construction
    n_coarse = 35
    h = np.ones(n_coarse)
    h[n_coarse//2:] = 0
    h = h * (n_coarse / 2) / np.sum(h)
    return h

def upscale(h_coarse, n_points):
    # Nearest neighbor upsampling
    h_fine = np.repeat(h_coarse, 2)
    h_fine = h_fine[:n_points]
    # Normalize to match sum constraint
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    return h_fine

def compute_c5(h, dx=2.0/len(h)):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def optimize(h_start, n_points, max_iter=200, tol=1e-8):
    dx = 2.0 / n_points
    h = h_start.copy()
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    for iteration in range(max_iter):
        current_c5 = compute_c5(h, dx)
        result = minimize(
            lambda x: compute_c5(x, dx),
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 10, 'ftol': tol}
        )
        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            # Perturb slightly if optimization fails
            h += np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available
    if 'initial_h_values' in globals():
        h_coarse = validate_and_repair(initial_h_values, 70)
        # Downsample to coarse
        h_coarse = h_coarse[::2]
        h_coarse = h_coarse * (35 / 2) / np.sum(h_coarse)
    else:
        h_coarse = initialize_coarse()
        
    # Coarse optimization
    h_coarse = optimize(h_coarse, 35, max_iter=50, tol=1e-6)
    
    # Upsample to fine scale
    h_fine = upscale(h_coarse, 70)
    
    # Fine optimization
    h_final = optimize(h_fine, 70, max_iter=200, tol=1e-8)
    
    final_c5 = compute_c5(h_final)
    
    return (h_final.tolist(), final_c5, 70)
