import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def repair_feasibility(h, target_sum, dx):
    """Repair h to satisfy constraints."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Main iterative optimization loop
    iteration = 0
    max_iter = 1000
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlaps and identify problematic lags
        overlap_full = correlate(h, 1 - h, mode="full")
        overlap_values = overlap_full[len(overlap_full)//2:]
        # Look at top lags that contribute most to overlap
        top_lags = np.argsort(overlap_values)[-int(n_points*0.1):]
        
        # Assemble and solve a linearized convex subproblem
        # We'll minimize a bound on C5 based on current gradients
        # and keep the overlapping constraints
        # New model:  
        # minimize C5
        # subject to: C5 >= sum(h_i * (1 - h_{i+k})) * dx for k in top_lags
        #             0 <= h_i <= 1 for all i
        #             sum(h_i) = target_sum
        
        # Create constraints matrix and rhs
        # Linearize sum(h_i * (1 - h_{i+k})) ≈ sum(h_i) - sum(h_i * h_prev_{i+k}) - sum(h_prev_i * h_{i+k}) + sum(h_prev_i * h_prev_{i+k})
        # So we get sum(h_i * (1 - h_{i+k})) ≈ 1*sum(h_i) - sum(h_i * h_prev_{i+k}) - h_prev_i*1*sum(h_{i+k}) + sum(h_prev_i * h_prev_{i+k})
        
        A_eq = np.zeros((1, n_points))
        A_eq[0] = 1.0
        b_eq = target_sum
        
        A_ineq = []
        b_ineq = []
        
        for k in top_lags:
            if k > 0:
                # Lag k corresponds to i -> i+k
                row = np.zeros(n_points)
                row[0:n_points-k] = 1.0
                A_ineq.append(row)
                # For the right-hand side we use the current optimization value
                b_ineq.append(overlap_values[k])
                # The constraint is h^T * (1 - h_{k}) >= overlap_k
                # Approximating with linearization:
                # sum(h_i) - sum(h_{i} * h_{i+k}^{prev}) >= overlap_k 
                # This gives: sum(h_i) - sum(h_{i} * h_{i+k}^{prev}) >= overlap_k
            else:
                # k = 0 then h_{i+0} = h_i, so just: sum(h_i * (1 - h_i)) >= overlap_k
                # Linearized around current h: h_i * (1 - h_i^{prev}) + h_i^{prev} * (1 - h_i) >= overlap_k
                # = h_i * (1 - h_i^{prev}) + h_i^{prev} - h_i^{prev} * h_i >= overlap_k
                # = h_i*(1 - h_i^{prev})  + h_i^{prev} - h_i^{prev}*h_i >= overlap_k
                # = h_i * (1 - h_i^{prev} - h_i^{prev}) + h_i^{prev} >= overlap_k
                # = h_i * (1 - 2 * h_i^{prev}) + h_i^{prev} >= overlap_k
                
        # Simplifying approach: normalize h and optimize using more direct method
        # Use gradient descent or local improvements on the entire system
        
        # Apply a small gradient descent step: h = h - step_size * gradient
        step_size = 0.001
        # Gradient estimated from top overlaps, simplify: gradient computed by changing h[i]
        # Worst case gradients: small change in h[i] affects many lag overlaps
        gradient = np.zeros(n_points)
        
        # Linear approximation of overlap constraint
        # evaluate ∇ sum(h_i * (1 - h_{i+k}))
        # if k > 0 we have:
        # sum(h_i * (1 - h_{i+k})) = sum(h_i - h_i * h_{i+k}) = sum(h_i) - sum(h_i * h_{i+k})
        # so ∇ = (1 - h_{i+k} - h_i) = 1 - h_{i+k} if h_i is variable
        # but it's 1 - h_{i+k} [from h_i side]; and if h_{j+k} is a variable, then it's -h_j [from h_{j+k} side]
        
        for k in top_lags:
            if k > 0:
                if k < n_points:
                    for i in range(n_points - k):
                        # ∇[sum(h_i * (1 - h_{i+k}))] wrt h_i = 1 - h_{i+k}
                        # ∇[sum(h_i * (1 - h_{i+k}))] wrt h_{i+k} = -h_i
                        gradient[i] += 1 - h[i+k]
                        # assert for i+k < n_points
                        if i+k < n_points:
                            gradient[i+k] -= h[i]
        
        # Update using gradient descent, but falling back to constrained optimization  
        h = h - step_size * gradient
        
        # Repair feasibility step
        h = repair_feasibility(h, target_sum, dx)
        
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
