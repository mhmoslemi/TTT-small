import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Projected gradient descent
    max_iter = 1000
    learning_rate = 0.01
    patience = 0
    patience_limit = 50
    
    # Precompute correlation indices
    lag_indices = np.arange(-n_points + 1, n_points)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps first
        overlaps = correlate(h, 1 - h, mode="full")
        max_val = overlaps.max()
        max_lags = lag_indices[overlaps == max_val]
        
        # Compute gradient with respect to the max overlap(s)
        gradient = np.zeros(n_points)
        for k in max_lags:
            # Gradient component from h[i] affecting overlap at lag k
            if k >= 0:  # h[i] * (1 - h[i+k])
                for i in range(n_points - k):
                    gradient[i] += (1 - h[i + k])
            else:  # h[i] * (-h[i-k])
                for i in range(-k, n_points):
                    gradient[i] += (-h[i + k])
        
        gradient /= len(max_lags)
        
        # Update with gradient
        h_new = h - learning_rate * gradient
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
            
        h = h_new
        
        # Adaptive learning rate
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= 0.9
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        if patience >= patience_limit:
            if learning_rate < 1e-6:
                break
            learning_rate *= 0.5
            patience = 0
    
    # Post-processing with SLSQP
    try:
        result = minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            post_h = result.x
            post_h = np.clip(post_h, 0, 1)
            post_h = post_h * (n_points / 2) / np.sum(post_h)
            post_c5 = compute_c5(post_h)
            if post_c5 < best_c5:
                best_h = post_h
                best_c5 = post_c5
    except:
        pass  # If local search fails, keep original
        
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
