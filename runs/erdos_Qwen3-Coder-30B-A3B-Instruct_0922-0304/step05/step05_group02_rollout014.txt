import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def line_search(h, grad, step_size, dx, max_iter=20):
    """Perform backtracking line search."""
    alpha = step_size
    for _ in range(max_iter):
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, len(h) / 2)
        c5_new = compute_c5(h_new, dx)
        if c5_new < compute_c5(h, dx) or alpha < 1e-8:
            return h_new, alpha
        alpha *= 0.5
    return h, alpha

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from parent if available
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values)
        init_h = np.clip(init_h, 0, 1)
        init_h = project_to_feasible(init_h, n_points / 2)
    else:
        # Create initial guess
        init_h = np.ones(n_points)
        init_h[n_points//2:] = 0
        init_h += np.random.normal(0, 0.05, n_points)
        init_h = np.clip(init_h, 0, 1)
        init_h = project_to_feasible(init_h, n_points / 2)
    
    h = init_h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Optimizer parameters
    momentum = 0.0
    decay = 0.9
    step_size = 0.1
    momentum_cache = np.zeros(n_points)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute current correlation
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Compute gradient for max overlap shift
        shift = max_idx - (n_points - 1)
        if shift < 0:
            grad = h[shift:] - h[:n_points+shift]
        else:
            grad = h[:n_points-shift] - h[shift:]
        grad = np.concatenate([grad, np.zeros(shift if shift > 0 else -shift)])
        gradient = grad * dx
        
        # Apply momentum
        momentum_cache = decay * momentum_cache + (1 - decay) * gradient
        direction = momentum_cache
        
        # Line search
        h_new, _ = line_search(h, direction, step_size, dx)
        
        # Update
        h = h_new
        
        # Update best solution
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
