import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize swarm
    n_particles = 20
    particles = []
    velocities = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        best_h = np.array(initial_h_values)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    else:
        # Fallback initialization
        best_h = np.ones(n_points)
        best_h[n_points//2:] = 0
        best_h += np.random.normal(0, 0.05, n_points)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    best_c5 = float('inf')
    
    # Initialize particles
    for _ in range(n_particles):
        h = best_h.copy() + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_vel = 0.2
    
    # Store personal bests
    personal_bests = [p.copy() for p in particles]
    personal_best_c5 = [compute_c5(p) for p in particles]
    
    # Global best tracking
    global_best_idx = np.argmin(personal_best_c5)
    global_best = personal_bests[global_best_idx].copy()
    global_best_c5 = personal_best_c5[global_best_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update personal and global bests
        for i in range(n_particles):
            current_c5 = compute_c5(particles[i])
            if current_c5 < personal_best_c5[i]:
                personal_bests[i] = particles[i].copy()
                personal_best_c5[i] = current_c5
                if current_c5 < global_best_c5:
                    global_best = particles[i].copy()
                    global_best_c5 = current_c5
                    if global_best_c5 <= 0.3807:
                        return (global_best.tolist(), global_best_c5, n_points)
        
        # Update particles
        for i in range(n_particles):
            r1, r2 = np.random.random(2), np.random.random(2)
            # Update velocity
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_bests[i] - particles[i]) +
                           c2 * r2 * (global_best - particles[i]))
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            # Update position
            particles[i] += velocities[i]
            # Ensure feasibility
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        
        # Update inertia weight
        if w > 0.4:
            w -= 0.001
    
    # Final evaluation
    final_c5 = compute_c5(global_best)
    
    return (global_best.tolist(), final_c5, n_points)
