import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Computes the C5 value for a given step function h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Projects h to satisfy sum constraint and box constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def compute_gradient(h, shift_idx, dx):
    """Compute gradient w.r.t. shift_idx in cross-correlation."""
    n = len(h)
    grad = np.zeros_like(h)
    # Compute gradient of the specific shift
    for i in range(n):
        if (i + shift_idx) < n:
            # There is overlap in this direction
            grad[i] = 1 - h[i + shift_idx]
        else:
            # Per convolution logic
            pass
    # Since we're working with cross-correlation h * (1-h) effectively the gradient 
    # for k-th shift is (1-h) at position that would create overlap.
    # Manual indexing approach. 
    # For simplicity, relabel the shift in convolution context.
    # Cross-correlation k_shift gives contribution from h[i] to (1-h[j]) where j = i - k_shift
    # That is for shift k the gradient w.r.t. h[i] is 1 - h[i+k] (for valid i+k)
    # since (h * (1-h))[k] = sum over i of h[i] * (1 - h[i+k])
    # So gradient w.r.t. h[i] is sum over k where i+k is valid, (1-h[i+k])
    # But we want single shift gradient for dynamic update.
    # So we compute directional gradient for active shift.
    # Cleaning up with direct gradient definition for specific shift.
    # Rewritten so (y * h)[k] = sum_i y[i] h[i-k] so for fixed shift k:
    # gradient[i] w.r.t. (sum_i h[i] (1-h[i+k])) = (1 - h[i+k])
    # That leads to a boxed gradient shaped like: 
    # Example: for shift t=0 the output values impact h[0], h[1], ..., h[n-1] 
    # but when you look at h[i], you have contribution to (1-h[i+0]) = h[i] 
    #      from (h[0] * (1-h[-1]) if that can happen)
    # For direct crosscorr: (h * (1-h))[k] = \sum_i h[i] * (1 - h[i+k])
    # gradient[i] = sum over k of d/dh[i] (h[i] * (1-h[i+k])) = sum of (1-h[i+k]) 
    # for valid k where i+k is within domain.
    return grad

def line_search(h, grad, max_iter=100, alpha_max=1.0, tol=1e-4):
    """Perform line search for optimal step size."""
    alpha = alpha_max
    baseline = compute_c5(h, dx)
    for _ in range(max_iter):
        h_new = project_h(h - alpha * grad, target_sum)
        new_c5 = compute_c5(h_new, dx)
        if new_c5 <= baseline:
            return alpha, h_new
        alpha *= 0.5
    return alpha, h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from pre-defined initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, target_sum)
    else:
        # Initialize with symmetric height
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 + 0.1 * (-1)**i
        h = project_h(h, target_sum)
    
    # Parameters
    max_iter = 10000
    momentum = 0.9
    step_size = 0.1
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Precompute intial correlation matrix to find maximum shift efficiently
    prev_h = h.copy()
    vel = np.zeros_like(h)
    last_improvement = 0
    improvement_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlations
        cross_corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(cross_corr)
        
        # Create gradient vector for max_k shift in convolution
        # Goal: minimize max_k overlap.
        # If x[k] = sum_i h[i] * (1-h[i+k]) then dx/dh[i] = (1-h[i+k]) when i+k valid
        
        # Since we're dealing with cross-correlation y[i] * x[i] = sum_{i+k=t} y[i] x[i-k]
        # For h * (1-h) at k-th shift: output[t] = sum_i h[i] (1-h[i+k])
        # Gradient w.r.t. h[i] is sum over t where the derivative contributes.
        # Which is the sum over t of (1-h[i+k]) when t += i+k (hence t-k =i).
        # This means gradient for h[i] = sum_k of (1-h[i+k]) over k
        # We know: (h * (1-h))[k] = sum_i h[i] (1 - h[i+k])
        # So for shift k, d/dh[i] (h * (1-h))[k] = (1 - h[i+k]) when ih neutrally sliding.
        # We take the gradient for the maximum spatial shift!

        # Naively get gradient w.r.t. current max shift k
        grad = np.zeros(n_points)
        for j in range(n_points):
            if (j + max_k) < n_points:
                grad[j] = 1 - h[j + max_k]
            # But since crosscorr normalized, we take specific shift contribution
            # So the shift is where max overlap was found.
            # Meaning for direct gradient update: 
            # (h * (1 - h))[shift] = sum_i h[i] * (1 - h[i + shift])
            # For gradient w.r.t. h[j], derivative is (1 - h[j+shift]), assuming j+shift < n
            
        # Instead: 
        # grad[j] = sum_k of (1 - h[j + shift_k]) 
        # This is over all k; but only one is actively minimized.
        # Let’s simply define: for shift k at maximum overlap, 
        # grad[i] = (1 - h[i+k]) if i+k < n.
        h_shifted = np.roll(h, max_k)
        grad = 1 - h_shifted
        
        # If we're not at zero (spike ) we can subtract one from non-zero contribution.
        # TODO: subtle handling here -- redefine more carefully.
        # For now just compute gradient as shifted vector flipped
        grad = np.zeros_like(h)
        shifted = np.roll(h, max_k)
        for j in range(n_points):
            if (j + max_k) < len(h):
                grad[j] = 1 - h[j + max_k]
        
        # Apply gradient descent with momentum
        vel = momentum * vel - step_size * grad
        h = h + vel
        
        # Project to feasible region
        h = project_h(h, target_sum)
        
        # Update best solution
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            improvement_count = 0
        else:
            improvement_count += 1
            
        # Early exit if target hit
        if best_c5 <= 0.3807:
            break
            
        # Convergence condition
        if improvement_count > 50:
            # Reduce step size slightly
            step_size *= 0.9
            improvement_count = 0
            
    # Final check
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
