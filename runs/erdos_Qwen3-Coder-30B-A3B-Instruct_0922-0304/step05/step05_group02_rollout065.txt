import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_h(h, target_sum)

    # Gradient descent parameters
    eta = 0.1
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute all correlations
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        max_val = corr[k_max]

        # Get the shift that causes max overlap
        # The full correlation has length 2*n_points - 1
        # k ranges from 0 to 2*n_points - 2
        if k_max < n_points:
            # Positive shift: shift by k_max
            shift = k_max
            # Compute gradient (same as 1-h shifted by shift)
            gradient = np.zeros_like(h)
            if shift == 0:
                gradient = 1 - h
            else:
                gradient[shift:] = 1 - h[:-shift]
        else:
            # Negative shift: shift by (k_max - n_points + 1)
            shift = k_max - n_points + 1
            gradient = np.zeros_like(h)
            if shift == 0:
                gradient = 1 - h
            else:
                gradient[:-shift] = 1 - h[shift:]
        
        # Update h using gradient descent
        h = h - eta * gradient * dx
        
        # Project h back to feasible region
        h = project_h(h, target_sum)
        
        # Update best solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Adaptive learning rate
        if iteration % 10 == 0:
            eta *= 1.01 if current_c5 < best_c5 else 0.9
        
        # Add noise occasionally
        if iteration % 100 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = project_h(h, target_sum)
            
        iteration += 1
        
        # Early stopping if target reached
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
