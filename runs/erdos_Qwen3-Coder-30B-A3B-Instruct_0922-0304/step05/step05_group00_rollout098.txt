import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def repair_feasibility(h, target_sum):
    """Repair h to be feasible: clip to [0,1] and normalize."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)

    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 1000

        # Compute current overlaps
        overlaps = correlate(h, 1 - h, mode="full")

        k_vals = np.arange(-n_points + 1, n_points)
        sorted_indices = np.argsort(overlaps)[::-1]
        
        # Take top 10% lags that contribute to maximum overlap

        # Determine the current maximum overlap value

        # Prepare convex constraints

                local_best_h = h_trial.copy()

        if local_best_c5 < current_c5:
