import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: piecewise 3-region function
        h = np.ones(n_points)
        split1 = n_points // 2
        split2 = 3 * n_points // 4
        h[split1:] = 0.2
        h[split2:] = 0.8
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def swap_segments(h, i1, i2, size):
        h_new = h.copy()
        # Swap segments
        temp = h_new[i1:i1+size].copy()
        h_new[i1:i1+size] = h_new[i2:i2+size]
        h_new[i2:i2+size] = temp
        
        # Rescale to maintain sum constraint
        total_sum = np.sum(h_new)
        h_new = h_new * (n_points / 2) / total_sum
        return h_new

    def solve_step(h, temperature=0.1):
        current_c5 = compute_c5(h)
        best_h = h.copy()
        best_c5 = current_c5
        
        # Parameters
        n_swaps = 1000
        swap_size = 5
        
        for _ in range(n_swaps):
            if time.time() - start_time > budget_s - 1:
                break
            
            # Choose two non-overlapping segments
            idx1 = np.random.randint(0, n_points - swap_size)
            idx2 = np.random.randint(0, n_points - swap_size)
            while abs(idx1 - idx2) < swap_size:
                idx2 = np.random.randint(0, n_points - swap_size)
                
            h_trial = swap_segments(h, idx1, idx2, swap_size)
            c5_trial = compute_c5(h_trial)
            
            if c5_trial < best_c5:
                best_h = h_trial.copy()
                best_c5 = c5_trial
            elif np.random.rand() < np.exp(-(c5_trial - current_c5) / temperature):
                h = h_trial
                current_c5 = c5_trial
        
        return best_h, best_c5
    
    # Main loop
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Multiple rounds of optimization
    for i in range(200):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Anneal temperature
        temp = 0.1 * (1 - i / 200.0)
        if temp < 0.01:
            temp = 0.01
            
        h_new, c5_new = solve_step(best_h, temperature=temp)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        if best_c5 <= 0.3807:
            break

    # Final polishing
    from scipy.optimize import minimize
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        result = minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            final_c5 = compute_c5(best_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
    except:
        pass

    return (best_h.tolist(), best_c5, n_points)
