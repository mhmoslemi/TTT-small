import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, k_max, dx, n_points):
    """Compute gradient of overlap at lag k_max with respect to h."""
    grad = np.zeros(n_points)
    # For correlation h * (1-h) at lag k_max, 
    # the gradient at h[i] is the contribution from the overlapping part
    # that involves h[i] in the correlation
    for i in range(n_points):
        j = i + k_max
        if 0 <= j < n_points:
            grad[i] += 1.0  # contribution of h[i] to overlap at k_max
        
    # Adjust by dx factor for the integral approximation 
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    # Track best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Algorithm settings
    alpha = 0.1
    patience = 0
    max_patience = 100
    learning_rate_decay = 0.5
    perturbation_freq = 20
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5 and all overlaps
        c5 = evaluate_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Calculate all overlaps to find the one with maximum value
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        
        # Compute gradient
        grad = compute_gradient(h, k_max, dx, n_points)
        
        # Update h
        h_new = h - alpha * grad
        
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        # Calculate new overlap
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept improvement or adjust learning rate
        if new_c5 < c5:
            h = h_new.copy()
            patience = 0
        else:
            patience += 1
            if patience >= max_patience:
                alpha *= learning_rate_decay
                patience = 0
                
        # Occasionally perturb to escape local minima
        if iteration % perturbation_freq == 0:
            perturb = np.random.normal(0, 0.01, n_points)
            h = h + perturb
            h = normalize_h(h, target_sum)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
