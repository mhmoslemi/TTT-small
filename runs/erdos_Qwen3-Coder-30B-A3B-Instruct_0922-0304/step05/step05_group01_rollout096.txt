import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize swarm
    n_particles = 20
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
        best_c5 = np.max(correlate(h_init, 1 - h_init, mode="full")) * dx
    else:
        best_h = np.ones(n_points)
        best_h[n_points//2:] = 0
        best_h += np.random.normal(0, 0.05, n_points)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_c5 = np.max(correlate(best_h, 1 - best_h, mode="full")) * dx

    # Initialize particles
    for i in range(n_particles):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    omega = 0.7
    phi_p = 1.5
    phi_g = 1.5
    max_vel = 0.1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate particles
        for i in range(n_particles):
            h = particles[i]
            c5 = np.max(correlate(h, 1 - h, mode="full")) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if best_c5 <= 0.3807:
                break
        
        # Update particles
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            pbest = h.copy()
            gbest = best_h.copy()
            
            v = omega * v + phi_p * r1 * (pbest - h) + phi_g * r2 * (gbest - h)
            
            # Clip velocity
            v = np.clip(v, -max_vel, max_vel)
            
            # Update position
            h = h + v
            h = np.clip(h, 0, 1)
            
            # Normalize
            h = h * (n_points / 2) / np.sum(h)
            
            particles[i] = h
            velocities[i] = v
            
        # Update omega
        omega = max(0.4, omega - 0.0001)
    
    return (best_h.tolist(), best_c5, n_points)
