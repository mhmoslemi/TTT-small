import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have sum equal to target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)

    # Optimization parameters
    momentum_decay = 0.9
    learning_rate = 0.1
    max_iterations = 1000
    min_improvement = 1e-6
    min_c5_target = 0.3807

    # Initialize momentum
    momentum = np.zeros_like(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Main optimization loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5 and all overlaps
        c5 = compute_c5(h, dx)
        overlap = correlate(h, 1 - h, mode="full")
        
        # Find maximum overlap (and indices)
        max_val = np.max(overlap)
        max_indices = np.where(overlap == max_val)[0]
        # Take the first one as reference
        max_k = max_indices[0] - (len(h) - 1)
        
        # Compute gradient (approximate) using finite differences
        # Perturb to get gradient w.r.t. h at the worst lag
        eps = 1e-4
        grad = np.zeros_like(h)
        
        h_perturbed = h.copy()
        # Compute gradient for a few key shifts
        for i in range(len(h)):
            h_perturbed[i] = min(1.0, max(0.0, h[i] + eps))
            c5_perturbed = compute_c5(h_perturbed, dx)
            grad[i] = (c5_perturbed - c5) / eps
            h_perturbed[i] = h[i]  # Reset
      
        # Apply momentum
        momentum = momentum_decay * momentum + (1 - momentum_decay) * grad
        
        # Perform line search
        step_size = learning_rate
        best_h_trial = h.copy()
        best_trial_c5 = c5
        for _ in range(10):  # Max 10 line search steps
            h_trial = h - step_size * momentum
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = normalize_h(h_trial, target_sum)
            trial_c5 = compute_c5(h_trial, dx)
            
            if trial_c5 < best_trial_c5:
                best_trial_c5 = trial_c5
                best_h_trial = h_trial.copy()
            else:
                step_size *= 0.5  # Backtrack
                if step_size < 1e-8:
                    break
                
        # Update h
        h = best_h_trial.copy()
        c5 = best_trial_c5
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Early exit if target achieved

            break

        previous_c5 = best_c5
            
    return (best_h.tolist(), best_c5, n_points)
