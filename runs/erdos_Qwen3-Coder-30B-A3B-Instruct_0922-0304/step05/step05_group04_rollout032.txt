import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Add the initial solution as first particle
    particles.append(h.copy())
    velocities.append(np.random.uniform(-0.01, 0.01, n_points))
    
    # Add random particles
    for _ in range(swarm_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
    
    # Track best solutions
    best_particle_idx = 0
    best_c5 = float('inf')
    
    # PSO hyperparameters
    omega_max = 0.9
    omega_min = 0.4
    phi1 = 2.0
    phi2 = 2.0
    vel_max = 0.1
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update omega (decreasing inertia)
        omega = omega_max - (omega_max - omega_min) * (time.time() - start_time) / budget_s
        
        # Evaluate all particles
        particle_c5s = []
        for i, h in enumerate(particles):
            # Compute correlation
            overlap = correlate(h, 1 - h, mode="full")
            c5 = np.max(overlap) * dx
            particle_c5s.append(c5)
            
            # Update personal best
            if c5 < particle_c5s[best_particle_idx]:
                best_particle_idx = i
                best_c5 = c5
                
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Update global best if needed
        current_best_idx = np.argmin(particle_c5s)
        if particle_c5s[current_best_idx] < best_c5:
            best_c5 = particle_c5s[current_best_idx]
            best_particle_idx = current_best_idx
            
        # Update each particle
        for i in range(swarm_size):
            # Get current particle and velocity
            h = particles[i]
            v = velocities[i]
            
            # Set personal best and global best
            p_best = particles[best_particle_idx]
            
            # Update velocity
            r1 = np.random.random(n_points)
            r2 = np.random.random(n_points)
            
            v = (omega * v + 
                 phi1 * r1 * (p_best - h) + 
                 phi2 * r2 * (p_best - h))
            
            # Clip velocity
            v = np.clip(v, -vel_max, vel_max)
            
            # Update position
            h_new = h + v
            
            # Clamp values to [0, 1]
            h_new = np.clip(h_new, 0, 1)
            
            # Renormalize to satisfy sum(h) = n/2
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Update particle and velocity
            particles[i] = h_new.copy()
            velocities[i] = v.copy()
    
    # Return best solution
    return (particles[best_particle_idx].tolist(), best_c5, n_points)
