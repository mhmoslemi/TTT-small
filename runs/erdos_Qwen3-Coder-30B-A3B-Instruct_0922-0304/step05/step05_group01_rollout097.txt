import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 20
    max_iter = 1000
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * (n_points / 2) / np.sum(h_base)
    else:
        h_base = np.ones(n_points)
        h_base[n_points//2:] = 0
        h_base += np.random.normal(0, 0.05, n_points)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * (n_points / 2) / np.sum(h_base)
    
    # Initialize swarm with random perturbations around base
    for _ in range(swarm_size):
        h = h_base + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # Track personal bests
    personal_bests = [h.copy() for h in swarm]
    personal_best_scores = [float('inf')] * swarm_size
    
    # Track global best
    global_best_h = h_base.copy()
    global_best_score = float('inf')
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_velocity = 0.1
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate swarm
        for i, h in enumerate(swarm):
            score = compute_c5(h)
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_bests[i] = h.copy()
            
            if score < global_best_score:
                global_best_score = score
                global_best_h = h.copy()
        
        # Early termination
        if global_best_score <= 0.3807:
            break
            
        # Update particles
        for i in range(swarm_size):
            r1, r2 = np.random.random(2)
            
            # Update velocity
            v = w * velocities[i]
            v += c1 * r1 * (personal_bests[i] - swarm[i])
            v += c2 * r2 * (global_best_h - swarm[i])
            
            # Clip velocity
            v = np.clip(v, -max_velocity, max_velocity)
            
            # Update position
            swarm[i] += v
            swarm[i] = np.clip(swarm[i], 0, 1)
            
            # Maintain sum constraint
            swarm[i] = swarm[i] * (n_points / 2) / np.sum(swarm[i])
            
            # Update velocity
            velocities[i] = v
        
        # Update inertia weight
        if w > 0.4:
            w -= 0.001
    
    # Final evaluation of global best
    final_c5 = compute_c5(global_best_h)
    
    return (global_best_h.tolist(), final_c5, n_points)
