import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize swarm
    n_particles = 20
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize particles
    for _ in range(n_particles):
        h = h_init.copy()
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal bests
    personal_bests = [p.copy() for p in particles]
    personal_best_scores = [float('inf')] * n_particles
    
    # Track global best
    global_best = h_init.copy()
    global_best_score = float('inf')
    
    # PSO parameters
    omega = 0.7
    phi_p = 1.5
    phi_g = 1.5
    max_velocity = 0.2
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        for i, particle in enumerate(particles):
            score = compute_c5(particle)
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_bests[i] = particle.copy()
                
            if score < global_best_score:
                global_best_score = score
                global_best = particle.copy()
        
        # Update particles
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive = phi_p * r1 * (personal_bests[i] - particles[i])
            social = phi_g * r2 * (global_best - particles[i])
            velocities[i] = omega * velocities[i] + cognitive + social
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particles[i] += velocities[i]
            
            # Ensure feasibility
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
        # Adaptively decrease inertia
        omega = max(0.4, omega - 0.0005)
        
        # Early exit if target achieved
        if global_best_score <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(global_best)
    
    return (global_best.tolist(), final_c5, n_points)
