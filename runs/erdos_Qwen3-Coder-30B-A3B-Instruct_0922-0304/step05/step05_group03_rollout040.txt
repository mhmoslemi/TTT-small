import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use adaptive simulated annealing
    temp = 1.0
    min_temp = 1e-6
    cooling_rate = 0.999
    patience = 0
    max_patience = 100
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
    
    # Initial evaluation
    current_c5 = compute_c5(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    while time.time() - start_time < budget_s - 1 and temp > min_temp:
        # Generate neighbor solution
        h_new = h.copy()
        # Perturb a random subset of indices
        indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        for i in indices:
            # Add small random change
            delta = np.random.normal(0, 0.05)
            h_new[i] += delta
        
        # Project to feasible region
        h_new = normalize_h(h_new, target_sum)
        
        new_c5 = compute_c5(h_new)
        
        # Accept or reject based on Metropolis criterion
        if new_c5 < current_c5 or np.random.random() < np.exp(-(new_c5 - current_c5) / temp):
            h = h_new
            current_c5 = new_c5

                best_h = h.copy()
                patience = 0
            else:
                patience += 1
        else:
            patience += 1
        
        # Adaptive cooling
        if patience > max_patience:
            temp *= cooling_rate
            patience = 0
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
