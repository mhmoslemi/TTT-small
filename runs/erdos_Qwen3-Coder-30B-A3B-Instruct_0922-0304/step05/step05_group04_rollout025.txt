import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization with alternating blocks
        h = np.zeros(n_points)
        h[::2] = 1
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def fast_local_search(h, max_iter=50):
        best_h = h.copy()
        best_c5 = compute_c5(h)
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Try small perturbations
            for i in range(n_points):
                if time.time() - start_time > budget_s - 1:
                    break
                    
                h_new = h.copy()
                # Try a small random change
                delta = np.random.normal(0, 0.01)
                h_new[i] += delta
                h_new = project_to_feasible(h_new)
                
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    h = h_new.copy()
                    
        return best_h, best_c5

    best_c5 = float('inf')
    best_h = h.copy()

    iteration = 0
    max_iter = 20000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        current_c5 = compute_c5(h)
        
        # Store the best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Gradient-based update with larger step size and perturbation
        # Approximate gradient by finite difference in a few directions
        grad = np.zeros(n_points)
        eps = 1e-4
        
        for i in range(n_points):
            # Perturb in positive and negative directions
            h_plus = h.copy()
            h_minus = h.copy()
            
            h_plus[i] += eps
            h_minus[i] -= eps
            
            # Clamp and normalize
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            # Compute finite difference
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps) * dx
        
        # Update h with negative gradient (to minimize overlap)
        step_size = 0.02
        h_new = h - step_size * grad
        
        # Perturb with random noise to avoid local minima
        noise_amp = 0.005
        h_new += np.random.normal(0, noise_amp, n_points)
        
        h = project_to_feasible(h_new)
        
        # Fast local search after significant update
        if iteration % 100 == 0:
            h, _ = fast_local_search(h, max_iter=20)
            
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
