import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    def solve_feasibility(c5_max):
        # Try to solve for given C₅_max
        # We want to minimize a slack variable subject to overlap constraints
        def objective_slack(slack_vars):
            # Minimize slack variable
            return slack_vars[0]
        
        # Overlap constraints
        def overlap_constraint(slack_vars):
            h_vars = slack_vars[1:]
            h = np.array(h_vars)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h) if np.sum(h) > 0 else h
            
            overlap = correlate(h, 1 - h, mode="full")
            max_overlap = np.max(overlap) * dx
            return max_overlap - c5_max
            
        # Weight constraint
        def sum_constraint(slack_vars):
            h_vars = slack_vars[1:]
            return np.sum(h_vars) - n_points / 2

        # Define bounds: first variable is slack, others are h
        bounds = [(0, None)] + [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'ineq', 'fun': overlap_constraint},
                       {'type': 'eq', 'fun': sum_constraint}]

        # Starting guess
        x0 = [0.5] + list(h)
        try:
            result = minimize(objective_slack, 
                              x0, 
                              bounds=bounds,
                              constraints=constraints,
                              method='SLSQP',
                              options={'maxiter': 50, 'ftol': 1e-8})
            if result.success:
                h_sol = np.clip(result.x[1:], 0, 1)
                h_sol = h_sol * (n_points / 2) / np.sum(h_sol) if np.sum(h_sol) > 0 else h_sol
                return True, h_sol
        except:
            pass

        return False, h

    # Binary search
    c5_low = 0.0
    c5_high = 0.38085  # initial upper bound
    tolerance = 1e-4

    best_c5 = c5_high
    best_h = h.copy()
    
    iterations = 0
    max_iterations = 40
    
    while iterations < max_iterations and (c5_high - c5_low) > tolerance:
        if time.time() - start_time > budget_s - 1:
            break
            
        c5_mid = (c5_low + c5_high) / 2.0
        feasible, h_candidate = solve_feasibility(c5_mid)
        
        if feasible:
            c5_high = c5_mid
            best_c5 = c5_mid
            best_h = h_candidate.copy()
        else:
            c5_low = c5_mid
            
        iterations += 1
        
        # If we're getting closer to target, refine tolerance
        if c5_high - c5_low < 1e-3:
            tolerance = 1e-5

    # Validate final solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
