import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def validate_and_repair(h_vals, n_points):
    """Validate and repair initial solution"""
    h = np.array(h_vals)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def initialize_coarse(n_coarse=35):
    """Initialize coarse solution"""
    h = np.ones(n_coarse)
    h[n_coarse//2:] = 0
    return h

def upscale(h_coarse, n_points):
    """Upscale coarse solution to fine grid"""
    # Nearest neighbor upsampling
    upscale_factor = n_points // len(h_coarse)
    h_fine = np.repeat(h_coarse, upscale_factor)
    # Ensure exact sum constraint
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    return h_fine

def compute_c5(h, dx=2.0):
    """Compute C5 value"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx / 2.0

def optimize(h, n_points, max_iter=200, tol=1e-8):
    """Optimize with constraints"""
    dx = 2.0 / n_points
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    def objective(h):
        return compute_c5(h, dx)

    result = minimize(
        objective,
        h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': max_iter, 'ftol': tol}
    )
    
    if result.success:
        h_opt = result.x
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        return h_opt
    else:
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from parent or fallback
    if 'initial_h_values' in globals():

    else:
        h = initialize_coarse(35)
        h = upscale(h, n_points)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Coarse optimization
    n_coarse = 35
    h_coarse = h[::2]  # Extract coarse values
    h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    
    h_coarse = optimize(h_coarse, n_coarse, max_iter=50, tol=1e-6)
    
    # Upsample
    h = upscale(h_coarse, n_points)
    
    # Fine optimization
    h = optimize(h, n_points, max_iter=200, tol=1e-8)
    
    # Final evaluation
    final_c5 = compute_c5(h, dx)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    
    # Early exit
    if best_c5 <= 0.3807:
        # Perform final refinement
        h_final = optimize(best_h, n_points, max_iter=100, tol=1e-9)
        final_c5 = compute_c5(h_final, dx)
        return (h_final.tolist(), final_c5, n_points)
    
    return (best_h.tolist(), best_c5, n_points)
