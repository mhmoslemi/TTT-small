import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def compute_gradient_and_shift(h):
    """Compute gradient and the shift that maximizes overlap."""
    corr = correlate(h, 1 - h, mode="full")
    max_k = np.argmax(corr)
    max_val = corr[max_k]
    # Gradient based on the shift that gives maximum overlap
    # shift is k - (n-1) where k is from full correlation
    shift = max_k - (len(h) - 1)
    return shift, max_val

def line_search(h, grad, step_size, dx, max_iter=20):
    """Perform line search to find good step size."""
    # Try to find a good step size using backtracking
    for _ in range(max_iter):
        h_new = h - step_size * grad
        h_new = project_h(h_new, len(h) / 2)
        c5_new = compute_c5(h_new, dx)
        if c5_new < compute_c5(h, dx):
            return h_new, step_size
        step_size *= 0.5
    return h, step_size

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points / 2)
    else:
        # Start with a coarse symmetric construction
        h = np.zeros(n_points)
        mid = n_points // 2
        h[mid-10:mid+10] = 0.5
        h = project_h(h, n_points / 2)
    
    # Optimization parameters
    learning_rate = 0.1
    momentum = 0.9
    velocity = np.zeros_like(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iter_count = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Compute gradient direction
        shift, _ = compute_gradient_and_shift(h)
        # Construct gradient based on shift (simplified)
        gradient = np.zeros_like(h)
        # We compute the gradient as gradient of correlation at k=shift
        # More precise: for the overlap we focus on the shift that maximizes overlap
        if shift < 0:
            gradient[-shift:] = 1
        else:
            gradient[:-shift] = 1
            
        gradient = np.clip(gradient, 0, 1)
        
        # Update velocity with momentum
        velocity = momentum * velocity + (1 - momentum) * gradient
        
        # Update with velocity step
        h_new = h - learning_rate * velocity
        h_new = project_h(h_new, n_points / 2)
        
        # Evaluate and update if better
        current_c5 = compute_c5(h_new, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Line search refinement
        h_new, lr = line_search(h, velocity, learning_rate, dx)
        h = h_new
        
        # If no improvement for a while, reduce learning rate
        if no_improvement_count > 20:
            learning_rate *= 0.7
            no_improvement_count = 0
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
