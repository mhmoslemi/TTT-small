import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have sum equal to target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

def project_to_feasible(h, target_sum):
    """Project h to satisfy the sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    scale = target_sum / current_sum
    return h * scale

def compute_gradient(h, k_worst, dx):
    """Compute gradient w.r.t. the worst shift."""
    n = len(h)
    grad = np.zeros(n)
    for i in range(n):
        if i + k_worst < n:
            grad[i] += 1  # Derivative of h[i] * (1 - h[i + k_worst]) w.r.t. h[i]
        if i - k_worst >= 0:
            grad[i] += 1  # Derivative of h[i - k_worst] * (1 - h[i]) w.r.t. h[i]
    return grad * dx

def line_search(h, grad, learning_rate, dx, target_sum, max_backtrack=10):
    """Perform line search to find a suitable step size."""
    alpha = learning_rate
    for _ in range(max_backtrack):
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new, dx)
        if c5_new < compute_c5(h, dx):
            return h_new, alpha
        alpha *= 0.5
    return h, alpha

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize

        h = np.clip(h, 0, 1)
    else:
        # Construct a simple piecewise linear function
        h = np.ones(n_points)
        h[n_points//2:] = 0
    h = normalize_h(h, target_sum)
    
    # Optimization parameters
    momentum_decay = 0.9
    learning_rate = 0.1
    momentum = np.zeros(n_points)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Main loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute all overlaps
        overlaps = correlate(h, 1 - h, mode="full")
        worst_k = np.argmax(overlaps)
        max_overlap = overlaps[worst_k]
        c5 = max_overlap * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient at worst shift
        # This is a simplified gradient that takes the full derivative
        grad = np.zeros(n_points)
        n = len(h)
        k = worst_k - (n - 1)  # Convert to shift index
        for i in range(n):
            if i + k < n and i + k >= 0:
                grad[i] += 1  # Gradient contribution from h[i] * (1 - h[i+k]) term
            if i - k < n and i - k >= 0:
                grad[i] += 1  # Gradient contribution from h[i-k] * (1 - h[i]) term
        
        # Accelerate with momentum
        momentum = momentum_decay * momentum + (1 - momentum_decay) * grad
        grad_momentum = momentum
        
        # Line search
        h, alpha = line_search(h, grad_momentum, learning_rate, dx, target_sum)
        
        # Update learning rate adaptively if our steps are too small
        if alpha < 1e-6:
            learning_rate *= 0.9
        elif alpha > 0.05:  # If we take large steps
            learning_rate = min(learning_rate * 1.1, 1.0)
        
        iteration += 1
    
    # Final refinement
    h_refined = best_h.copy()
    final_c5 = compute_c5(h_refined, dx)
    
    # If we didn't converge fast enough, do a final gradient pass
    for _ in range(100):
        overlaps = correlate(h_refined, 1 - h_refined, mode="full")
        worst_k = np.argmax(overlaps)
        max_overlap = overlaps[worst_k]
        c5 = max_overlap * dx
        if c5 > final_c5:
            break
        grad = np.zeros(n_points)
        n = len(h_refined)
        k = worst_k - (n - 1)
        for i in range(n):
            if i + k < n and i + k >= 0:
                grad[i] += 1  
            if i - k < n and i - k >= 0:
                grad[i] += 1  
        h_refined = h_refined - 0.5 * grad
        h_refined = project_to_feasible(h_refined, target_sum)
        final_c5 = compute_c5(h_refined, dx)
        
    return (h_refined.tolist(), final_c5, n_points)
