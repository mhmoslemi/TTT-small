import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
    
    # Gradient-based optimization with trust region
    from scipy.optimize import minimize
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Parameters for local search
    max_iter = 500
    tol = 1e-8
    
    # Continue optimizing until time expires or target reached
    while time.time() - start_time < budget_s - 1:
        # Local optimization using trust-constr
        def objective(h):
            return compute_c5(h)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        result = minimize(
            objective,
            h,
            method='trust-constr',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': min(200, max_iter), 'gtol': tol, 'ftol': tol}
        )
        
        if result.success:
            h = result.x.copy()
            h = normalize_h(h, target_sum)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Make perturbations to escape local minima
        if np.random.random() < 0.5:
            perturbation = np.random.normal(0, 0.02, n_points)
            h = h + perturbation
            h = normalize_h(h, target_sum)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
