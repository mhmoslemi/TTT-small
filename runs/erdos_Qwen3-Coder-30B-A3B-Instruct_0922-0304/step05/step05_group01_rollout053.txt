import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 20
    generations = 1000
    elite_fraction = 0.25
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Initialize population with diverse variants
    population.append(h.copy())
    for _ in range(population_size - 1):
        h_mut = h + np.random.normal(0, 0.02, n_points)
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
        population.append(h_mut.copy())
    
    best_h = population[0].copy()
    best_c5 = compute_c5(best_h)
    
    # Evolutionary algorithm
    mutation_strength = 0.01
    no_improvement_count = 0
    
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for all population members
        fitness = []
        for h in population:
            c5 = compute_c5(h)
            fitness.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Select elite
        elite_size = max(1, int(population_size * elite_fraction))
        elite = population[:elite_size]
        
        # Generate new population
        new_population = elite.copy()
        
        # Create offspring through mutation
        for individual in elite:
            for _ in range(2):
                mutated = individual + np.random.normal(0, mutation_strength, n_points)
                mutated = np.clip(mutated, 0, 1)
                mutated = mutated * (n_points / 2) / np.sum(mutated)
                new_population.append(mutated)
                
        # Trim to population size
        population = new_population[:population_size]
        
        # Adapt mutation strength
        if no_improvement_count > 50:
            mutation_strength *= 0.9
            no_improvement_count = 0
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
