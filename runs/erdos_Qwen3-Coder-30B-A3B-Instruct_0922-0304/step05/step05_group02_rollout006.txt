import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def compute_gradient(h, k_max):
    """Compute gradient based on the maximum overlap shift k_max."""
    n = len(h)
    grad = np.zeros_like(h)
    # Build the shifted version (1-h) for the specific k
    # The position of the shift k in the cross-correlation index is: k + n - 1
    # So we work with indexing - make sure not to use the wrong indices
    shifted = np.zeros(2*n - 1)
    shifted[n-1-k_max:n-1-k_max+len(h)] = 1 - h
    # Replace with direct indexing
    h_shifted = np.zeros(n)
    if k_max >= 0:
        h_shifted[k_max:] = h[:-k_max] if k_max > 0 else h[:]
    else:
        h_shifted[:k_max] = h[-k_max:] if k_max < 0 else h[:]
    grad = -h_shifted
    return grad

def line_search(h, grad, k_max, dx, alpha_max=1.0, c1=1e-4):
    """Perform line search for step size alpha."""
    # Compute test points along gradient direction: h - alpha * grad
    # but project back after.
    # stop when decrease occurs: f(x - alpha * grad) <= f(x) - c1 * alpha * grad^T * grad
    phi0 = compute_c5(h, dx)
    g0 = grad.T @ grad
    if g0 <= 1e-10:
        return 0.0
    
    # Try different alpha values
    alphas = np.logspace(-3, 0, 20)
    best_alpha = 0.0
    best_c5 = phi0
    for alpha in alphas:
        h_alpha = h - alpha * grad
        h_alpha = project_h(h_alpha, len(h) / 2)
        c5_alpha = compute_c5(h_alpha, dx)
        # Backtrack
        if c5_alpha <= phi0 - c1 * alpha * g0:
            best_alpha = alpha
            best_c5 = c5_alpha
            break
                
    return best_alpha

    # Initialize
    h = project_h(init_h, n_points / 2)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient descent setup
    momentum = 0.95
    alpha = 0.1
    prev_grad = np.zeros_like(h)
    prev_h = h.copy()
    
    iter_count = 0
    stall_count = 0
    max_stall = 50
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Compute all correlations
        corr = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr)
        
        # Compute gradient 
        grad = compute_gradient_direct(h, k_max)
        
        # Compute descent direction with momentum
        descent = momentum * prev_grad + grad
        
        # Line search for step size
        alpha = line_search(h, descent, k_max, dx, alpha_max=1.0)
        
        # Update
        h = h - alpha * descent
        h = project_h(h, n_points / 2)
        
        # Update tracking
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            stall_count = 0
        else:
            stall_count += 1
            
        # Check early exit condition
        if best_c5 <= 0.3807:
            break
            
        # Early stopping
        if stall_count > max_stall:
            break
            
        # Track prev values
        prev_grad = descent
        prev_h = h.copy()
    
    # Final evaluation of best solution
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
