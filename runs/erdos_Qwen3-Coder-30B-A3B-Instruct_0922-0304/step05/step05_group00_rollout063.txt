import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the available parent construction if exists
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base function by adjusting a simple step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        # Add some noise to avoid local minima trap
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    iteration = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Local search with SLSQP focusing on top contributing lags
        # Identify lags that contribute most to the overlap
        overlap = correlate(h, 1 - h, mode="full")
        # We're looking for the maximum among all lags
        max_idx = np.argmax(np.abs(overlap))
        # Select several surrounding lags to constrain more precisely
        lag_window = 10
        start_lag = max(0, max_idx - lag_window)
        end_lag = min(len(overlap), max_idx + lag_window + 1)
        relevant_lags = np.arange(start_lag, end_lag)
        
        # Objective is to reduce the maximum overlap
        # For optimization, it's better to first try minimizing the max overlap directly with bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Constraint: sum equals target
        def constraint_sum(h):
            return np.sum(h) - target_sum

        # Objective to minimize C5
        def objective(h):
            return compute_c5(h, dx)

        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Minimize C5 using SLSQP with bounds and constraint
        result = minimize(objective, h.copy(), method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
        
        if result.success:
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
            
            h = h_new
        
        iteration += 1
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
