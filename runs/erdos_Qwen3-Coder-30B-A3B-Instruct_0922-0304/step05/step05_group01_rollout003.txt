import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def mutate(h, mutation_strength):
        mutated = h + np.random.normal(0, mutation_strength, h.shape)
        mutated = np.clip(mutated, 0, 1)
        sum_h = np.sum(mutated)
        mutated = mutated * (n_points / 2) / sum_h
        return mutated

    # Evolutionary algorithm parameters
    population_size = 20
    elite_fraction = 0.25
    mutation_strength = 0.01
    max_generations = 1000
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        population.append(mutate(h, mutation_strength))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Track improvement
    last_improvement_gen = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Main evolutionary loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for all population members
        fitness = []
        for candidate in population:
            c5 = compute_c5(candidate)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = candidate.copy()
                last_improvement_gen = generation

        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elite_indices = sorted_indices[:int(elite_fraction * population_size)]
        elite = [population[i] for i in elite_indices]
        
        # Create new population
        new_population = elite.copy()
        
        # Generate offspring through mutation
        for parent in elite:
            for _ in range(2):
                if time.time() - start_time > budget_s - 1:
                    break
                child = mutate(parent, mutation_strength)
                new_population.append(child)
        
        # Trim to exact population size
        if len(new_population) > population_size:
            new_population = new_population[:population_size]
            
        # Update population
        population = new_population
        
        # Adjust mutation strength based on progress
        no_improvement_count = generation - last_improvement_gen
        if no_improvement_count > 50:
            mutation_strength *= 0.9
            last_improvement_gen = generation

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
