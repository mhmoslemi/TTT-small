import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def compute_c5_exact(h):
        # Use numpy correlate exactly
        h = np.array(h)
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Better optimization with alternating blocks
    def compute_gradient(h, k):
        """Compute approximate gradient at point k"""
        eps = 1e-4
        g = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] = min(1.0, h_plus[i] + eps)
            h_minus[i] = max(0.0, h_minus[i] - eps)
            # Renormalize
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            # Compute gradient approximation
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps) * dx
        return g

    def project_and_renormalize(h):
        """Project h to [0,1] and renormalize"""
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Early stopping if target achieved
    max_iter = min(10000, int(budget_s * 50))  # Rough estimate
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1

        # Compute current C5
        current_c5 = compute_c5_exact(h)
        
        # Track best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Try a different approach: optimize with projected gradient
        # Compute approximate gradient at position that maximizes overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_k_idx = np.argmax(overlap)
        max_k = (max_k_idx - (n_points - 1)) * dx  # k shift in [0, 2]
        
        # Gradient-based update with projection
        g = compute_gradient(h, max_k_idx)
        
        # Simplified update: perturb towards lower overlap
        step_size = 0.01
        h_new = h - step_size * g
        
        # Project and renormalize
        h_new = project_and_renormalize(h_new)
        
        # Basic convergence check using line search: try smaller steps
        h_temp = h.copy()
        step = step_size
        for _ in range(10):  # Limit line search steps
            h_trial = h - step * g
            h_trial = project_and_renormalize(h_trial)
            trial_c5 = compute_c5_exact(h_trial)
            if trial_c5 < current_c5:
                h_temp = h_trial
                step *= 0.8  # Reduce step size
            else:
                break
        
        h = h_temp

        # Occasionally use small random perturbations
        if iteration % 100 == 0:
            pert = np.random.normal(0, 0.005, n_points)
            h = h + pert
            h = project_and_renormalize(h)

    # Final refinement
    current_c5 = compute_c5_exact(best_h)
    return (best_h.tolist(), current_c5, n_points)
