import numpy as np
from scipy.optimize import line_search
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

def clip_and_project(h, target_sum):
    h_clipped = np.clip(h, 0, 1)
    scale = target_sum / np.sum(h_clipped)
    return h_clipped * scale + (1 - scale) * (h - h_clipped)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points

    else:

        init_h += np.random.normal(0, 0.05, n_points)

        init_h = normalize_h(init_h, n_points / 2)

    h = init_h.copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    momentum = np.zeros_like(h)
    learn_rate = 0.1
    mu = 0.9
    
    while time.time() - start_time < budget_s - 1:
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        k = max_idx - (n_points - 1)
        
        shift = int(k)
        if shift > 0:
            h_shifted = np.zeros_like(h)
            h_shifted[:-shift] = h[shift:]
        else:
            h_shifted = np.zeros_like(h)
            h_shifted[-shift:] = h[:-shift]
        
        gradient = (1 - h_shifted) * dx
        gradient = gradient[:len(h)]
        
        gradient = gradient - np.mean(gradient) * np.ones_like(gradient)
        
        gradient = gradient / np.linalg.norm(gradient) if np.linalg.norm(gradient) > 1e-8 else gradient

        momentum = descent_dir
