import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Problem, Minimize, Variable, norm, upperbound

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region ensuring sum(h) = target_sum and 0≤h≤1."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def solve_convex_subproblem(h, n_points, dx, target_sum):
    """Solve convex relaxation subproblem to update h."""
    # Create variables
    h_var = Variable(n_points)
    C = Variable()

    # Compute all correlations
    correlations = []
    for k in range(2 * n_points - 1):
        start_idx = max(0, k - n_points + 1)
        end_idx = min(n_points, k + 1)
        h_slice = h[start_idx:end_idx]
        one_minus_h_slice = 1 - h[max(0, n_points - 1 - k):min(2 * n_points - 2 - k, n_points)]
        if len(h_slice) == len(one_minus_h_slice):
            corr = np.dot(h_slice, one_minus_h_slice)
            correlations.append(corr)
        else:
            corr = 0
            correlations.append(corr)

    constraints = [C >= corr * dx for corr in correlations]
    constraints.append(cp.sum(h_var) == target_sum)
    constraints.append(0 <= h_var)
    constraints.append(h_var <= 1)
    
    # Solve using CVXPY
    objective = Minimize(C)
    prob = Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            new_h = h_var.value
        else:
            new_h = h.copy()
    except:
        new_h = h.copy()
        
    return np.clip(new_h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        h = np.ones(n_points) * 0.5
        h = project_to_feasible(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Solve convex relaxation
        new_h = solve_convex_subproblem(h, n_points, dx, target_sum)
        
        # Project to feasible region
        new_h = project_to_feasible(new_h, target_sum)
        
        # Evaluate new solution
        new_c5 = compute_c5(new_h, dx)
        
        # Accept improvement or occasional worse solution
        if new_c5 <= best_c5 or np.random.rand() < 0.05:
            h = new_h
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                
        # Perturb occasionally to escape local minima
        if iteration % 50 == 0:
            noise_magnitude = 0.1 * np.exp(-iteration/200)
            h += np.random.normal(0, noise_magnitude, n_points)
            h = project_to_feasible(h, target_sum)
            
        iteration += 1
        
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
