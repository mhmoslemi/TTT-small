import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def mutate(h, mutation_std):
        mutated = h + np.random.normal(0, mutation_std, h.shape)
        mutated = np.clip(mutated, 0, 1)
        sum_h = np.sum(mutated)
        mutated = mutated * (n_points / 2) / sum_h
        return mutated

    def evaluate_fitness(population):
        fitness = []
        for h in population:
            fitness.append(compute_c5(h))
        return np.array(fitness)

    def select_elite(population, fitness, elite_fraction=0.25):
        indices = np.argsort(fitness)
        elite_count = max(1, int(len(population) * elite_fraction))
        return [population[i] for i in indices[:elite_count]]

    def initialize_population():
        if 'initial_h_values' in globals():
            h0 = np.array(initial_h_values)
            h0 = np.clip(h0, 0, 1)
            sum_h0 = np.sum(h0)
            h0 = h0 * (n_points / 2) / sum_h0
        else:
            h0 = np.ones(n_points)
            h0[n_points//2:] = 0
            h0 += np.random.normal(0, 0.05, n_points)
            h0 = np.clip(h0, 0, 1)
            h0 = h0 * (n_points / 2) / np.sum(h0)
            
        population = [h0]
        for _ in range(19):
            h_mut = mutate(h0, 0.01)
            population.append(h_mut)
        return population

    # Main algorithm
    population = initialize_population()
    best_c5 = float('inf')
    best_h = None
    
    # Evolutionary algorithm parameters
    mutation_strength = 0.01
    max_generations = 500
    no_improvement_count = 0
    max_no_improvement = 50
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = evaluate_fitness(population)
        
        # Track best solution
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        current_best_h = population[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Adapt mutation strength
        if no_improvement_count > max_no_improvement:
            mutation_strength *= 0.1
            no_improvement_count = 0
            
        # Selection and reproduction
        elite = select_elite(population, fitness)
        new_population = []
        
        # Keep elite individuals
        new_population.extend(elite)
        
        # Generate offspring from elite
        for individual in elite:
            for _ in range(2):
                child = mutate(individual, mutation_strength)
                new_population.append(child)
                
        # Keep population size consistent
        population = new_population[:20]
        
    # Final validation
    if best_h is None:
        best_h = population[0]
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
