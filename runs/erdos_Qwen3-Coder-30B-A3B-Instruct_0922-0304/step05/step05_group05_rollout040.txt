import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: piecewise function with 3 regions
        h = np.ones(n_points)
        split1 = n_points // 4
        split2 = 3 * n_points // 4
        h[split1:split2] = 0.2
        h[split2:] = 0.8
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Hyperparameters
    n_swaps = 1000
    swap_size = 5
    temperature = 0.1

    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    for iteration in range(n_swaps):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Make a swap
        i1 = np.random.randint(0, n_points - swap_size)
        i2 = np.random.randint(0, n_points - swap_size)
        # Ensure non-overlapping segments
        while abs(i1 - i2) < swap_size:
            i2 = np.random.randint(0, n_points - swap_size)
            
        # Create new h
        new_h = current_h.copy()
        tmp = new_h[i1:i1+swap_size].copy()
        new_h[i1:i1+swap_size] = new_h[i2:i2+swap_size]
        new_h[i2:i2+swap_size] = tmp
        
        # Maintain sum constraint
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        new_c5 = compute_c5(new_h)
        
        # Accept with simulated annealing probability
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / temperature):
            current_h = new_h
            current_c5 = new_c5
            
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        if best_c5 <= 0.3807:
            break

    # Final polish with local optimization
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    try:
        result = minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            polished_h = result.x
            polished_h = np.clip(polished_h, 0, 1)
            polished_h = polished_h * (n_points / 2) / np.sum(polished_h)
            polished_c5 = compute_c5(polished_h)
            if polished_c5 < best_c5:
                best_h = polished_h
                best_c5 = polished_c5
    except:
        pass

    return (best_h.tolist(), best_c5, n_points)
