from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # compute current overlap
        c5 = evaluate_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience_count = 0
        else:
            patience_count += 1
        
        # Early exit check
        if best_c5 <= 0.3807:
            break

            # decay learning rate
            eta *= decay
            patience_count = 0

        grad = compute_gradients(h, dx)

        # Update h using gradient
        h = h - eta * grad

        h = normalize_h(h, target_sum)

        # Coordinate-wise random perturbation in some frequency to escape local minima

                    pert = np.random.normal(0, 0.01)
                    h[i] += pert
                    h = normalize_h(h, target_sum)

    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
