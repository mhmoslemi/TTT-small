import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Fallback initialization
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init += np.random.normal(0, 0.05, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Create swarm
    particles.append(h_init.copy())
    velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Add random particles
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Track best solutions
    particle_best = [p.copy() for p in particles]
    particle_best_values = [float('inf')] * swarm_size
    global_best = particles[0].copy()
    global_best_value = float('inf')
    
    # PSO parameters
    w_max = 0.9
    w_min = 0.4
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    tolerance = 1e-6
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Update inertial weight
        w = w_max - (w_max - w_min) * (time.time() - start_time) / budget_s
        
        for i in range(swarm_size):
            h = particles[i]
            
            # Compute current C5
            c5_current = compute_c5(h)
            
            # Update personal best
            if c5_current < particle_best_values[i]:
                particle_best_values[i] = c5_current
                particle_best[i] = h.copy()
            
            # Update global best
            if c5_current < global_best_value:
                global_best_value = c5_current
                global_best = h.copy()
            
            # Early exit condition
            if global_best_value <= 0.3807:
                break
                
            # Update velocity
            r1, r2 = np.random.rand(2)
            if i == 0:  # First particle is the initial one
                p_best = particle_best[i]
                g_best = global_best
            else:
                p_best = particle_best[i]
                g_best = global_best
            
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best - h) + 
                           c2 * r2 * (g_best - h))
            
            # Apply velocity limits
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] = h + velocities[i]
            
            # Clamp to [0, 1]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Renormalize to sum to n_points / 2
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Ensure we return the best result found
    final_c5 = computed_value = compute_c5(global_best)
    
    return (global_best.tolist(), final_c5, n_points)
