import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = target_sum."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Normalize
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)

    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 1000
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps for maximum
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max] * dx
        
        # If we found a better solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break

        # Compute gradient for the worst-case shift
        # The gradient of overlap at shift k is (1 - h_{k:]) for h[:n-k]
        # But we need to be more careful about indexing in correlation
        
        # Let's compute it properly:
        # For overlap at shift k, the part contributing is h[i] * (1 - h[i+k])
        # The gradient w.r.t. h[i] is (1 - h[i+k]) for i < n-k, 0 otherwise
        # And for h[i+k] the gradient is -h[i] for i < n-k, 0 otherwise
        # So the total gradient component for h[i] is sum_{k} (1 - h[i+k]) for valid k
        
        # Actually, let's compute the gradient more carefully:
        # hand = h; h_shifted = h[shift:] where shift is taken from the max overlap
        # Compute analytical gradient of the maximum overlap
        
        # Recomputing all overlaps to get the right indices:
        k_max_overlap = np.argmax(correlate(h, 1-h, mode="full"))
        
        # The maximum overlap is at shift k_max
        # We compute gradient wrt to the maximum contribution
        # For shift k_max, we compute gradient components
        grad = np.zeros(n_points)
        
        # We compute it for the shifted version that is actually used
        # It's hard to compute the analytical gradient correctly due to mode="full"
        # Let's use a multi-step approach
        if k_max >= n_points:
            # k_max is positive, comes from right shift
            k_offset = k_max - n_points + 1  # Positive offset in the correlation
            # Shift of k_offset in h for correlation 
            # h[i] * (1 - h[i + k_offset])
            # Gradient wrt h[i]: 1 - h[i + k_offset]
            for i in range(n_points - k_offset):
                grad[i] += 1 - h[i + k_offset]
        else:
            # k_max < n_points, negative shift to the left
            k_offset = n_points - 1 - k_max  # place in shifted array
            # h[i + k_offset] * (1 - h[i]) 
            # Gradient wrt h[i] is -h[i + k_offset]
            for i in range(n_points - k_offset):
                grad[i] -= h[i + k_offset]
        
        # Update h with gradient descent
        h = h - learning_rate * grad
        h = project_to_feasible(h, target_sum)
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            learning_rate *= 0.9

    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
