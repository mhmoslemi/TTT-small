import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize swarm
    n_particles = 20
    particles = []
    velocities = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        # Create a reasonable initial guess
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 += np.random.normal(0, 0.05, n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)

    # Initialize particles and velocities
    best_h = h0.copy()
    best_c5 = float('inf')
    
    for _ in range(n_particles):
        # Initialize each particle near the best solution
        h = h0 + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track global best
    global_best_h = best_h.copy()
    global_best_c5 = best_c5
    
    # PSO parameters
    w_start = 0.7
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    v_max = 0.1
    
    def compute_c5(h):
        # Compute all overlaps at once using correlation
        overlap = correlate(h, 1 - h, mode="full")
        # Take maximum across all shifts
        return np.max(overlap) * dx
    
    def ensure_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Main optimization loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(n_particles):
            # Evaluate current particle
            current_c5 = compute_c5(particles[i])
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = particles[i].copy()
                
            if current_c5 < global_best_c5:
                global_best_c5 = current_c5
                global_best_h = particles[i].copy()
                
            # Early exit if target achieved
            if global_best_c5 <= 0.3807:
                break
                
            # Update velocity and position
            r1, r2 = np.random.random(2)
            # Update velocity
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (best_h - particles[i]) + 
                           c2 * r2 * (global_best_h - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = ensure_feasible(particles[i])
        
        # Early exit if target achieved
        if global_best_c5 <= 0.3807:
            break
    
    # Final evaluation of the best solution
    final_c5 = compute_c5(global_best_h)
    
    return (global_best_h.tolist(), final_c5, n_points)
