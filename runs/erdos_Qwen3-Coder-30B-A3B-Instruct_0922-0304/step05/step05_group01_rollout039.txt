from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Binary search on C_5
    C5_low = 0.0
    C5_high = 0.38085
    tolerance = 1e-4
    max_iterations = 40
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Solver with tight tolerances
    def solve_feasibility(h0, C5_max):
        def obj(h):
            return -compute_c5(h)  # We maximize by minimizing negative
        
        def constraint_overlap(h):
            overlap = correlate(h, 1 - h, mode="full")
            return C5_max * dx - np.max(overlap) * dx
            
        constraints = [
            {'type': 'eq', 'fun': constraint_sum},
            {'type': 'ineq', 'fun': constraint_overlap}
        ]
        
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(
            obj, 
            h0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        
        return result.success and result.fun < -C5_max * dx + 1e-8, result.x if result.success else h0
    
    # Binary search loop
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        C5_mid = (C5_low + C5_high) / 2.0
        feasible, new_h = solve_feasibility(h.copy(), C5_mid)
        
        if feasible:
            C5_high = C5_mid
            h = new_h
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            C5_low = C5_mid
            
        # Early exit if target achieved
        if C5_high <= 0.3807:
            break

    if time.time() - start_time < budget_s - 1:
        # Try gradient-based refinement on the best result
        def refine_optimization(h0, max_iter=30):

                return compute_c5(h)

                )
                if result.success:
                    current_h = result.x
                    current_h = np.clip(current_h, 0, 1)
                    current_h = current_h * (n_points / 2) / np.sum(current_h)
                    current_c5 = compute_c5(current_h)
                    if current_c5 < best_c5:

                        best_h = current_h.copy()
                else:

            return best_h, best_c5
