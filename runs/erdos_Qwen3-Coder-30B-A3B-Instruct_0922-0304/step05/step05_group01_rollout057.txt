import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def validate_and_repair(h_values, n_points):
    h = np.array(h_values)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def initialize_coarse():
    # Create a reasonable initial coarse solution
    n_coarse = 35
    h = np.ones(n_coarse)
    h[n_coarse//2:] = 0
    # Add some noise for diversity
    h += np.random.normal(0, 0.05, n_coarse)
    h = np.clip(h, 0, 1)
    h = h * (n_coarse / 2) / np.sum(h)
    return h

def upscale(h_coarse, n_points):
    # Upsample using nearest neighbor interpolation
    h_fine = np.repeat(h_coarse, 2)
    # Adjust to make sum correct
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    return h_fine

def compute_c5(h, dx=2.0/len(h)):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def optimize(h_initial, n_points, max_iter=200, tolerance=1e-8):
    dx = 2.0 / n_points
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    h = h_initial.copy()
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Optimize using SLSQP
        result = minimize(
            lambda x: compute_c5(x, dx), 
            h, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraints,
            options={'maxiter': 5, 'ftol': tolerance}
        )
        
        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            # If optimization fails, use a small perturbation
            h += np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return best_h, best_c5

    np.random.seed(seed)

    # Parameters
    n_points = 70
    n_coarse = 35
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available

    else:
        h = initialize_coarse()
    
    # Coarse optimization
    h_coarse = h[:n_coarse]
    h_coarse, c5_coarse = optimize(h_coarse, n_coarse, max_iter=50, tolerance=1e-6)
    
    # Upsample to fine scale
    h_fine = upscale(h_coarse, n_points)
    
    # Fine optimization
    h_final, final_c5 = optimize(h_fine, n_points, max_iter=200, tolerance=1e-8)
    
    # Final evaluation
    final_c5 = compute_c5(h_final, dx)
    
    return (h_final.tolist(), final_c5, n_points)
