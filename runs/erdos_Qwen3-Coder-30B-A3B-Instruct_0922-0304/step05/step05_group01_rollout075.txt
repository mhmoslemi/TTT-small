import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h):
    """Compute C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * (2.0 / len(h))

def mutate(h, mutation_std, n_points):
    """Mutate h with Gaussian noise, then enforce constraints"""
    mutated = h + np.random.normal(0, mutation_std, h.shape)
    mutated = np.clip(mutated, 0, 1)
    sum_h = np.sum(mutated)
    mutated = mutated * (n_points / 2) / sum_h
    return mutated

def evaluate_fitness(population):
    """Evaluate C5 for all individuals and sort"""
    fitness = [compute_c5(ind) for ind in population]
    indices = np.argsort(fitness)
    return [population[i] for i in indices], [fitness[i] for i in indices]

def select_elite(population, elite_fraction=0.25):
    """Select top fraction of population"""
    n_elite = max(1, int(len(population) * elite_fraction))
    return population[:n_elite]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Population size
    pop_size = 20
    
    # Initialize population
    if 'initial_h_values' in globals():

        h0 = np.clip(h0, 0, 1)
        sum_h0 = np.sum(h0)
        h0 = h0 * (n_points / 2) / sum_h0
    else:
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 += np.random.normal(0, 0.05, n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Create initial population
    population = [h0]
    for _ in range(pop_size - 1):
        h = mutate(h0, 0.01, n_points)
        population.append(h)
    
    # Track best
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # Evolutionary algorithm
    mutation_strength = 0.01
    no_improvement_count = 0
    max_no_improvement = 50
    
    max_generations = 500
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        population, fitness = evaluate_fitness(population)
        
        # Check if we found a better solution
        current_best_c5 = fitness[0]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Adapt mutation strength
        if no_improvement_count > max_no_improvement:
            mutation_strength *= 0.9
            no_improvement_count = 0
            
        # Select elite
        elite = select_elite(population, 0.25)
        
        # Generate new population through mutation
        new_population = []
        for individual in elite:
            # Create two offspring per elite
            for _ in range(2):
                child = mutate(individual, mutation_strength, n_points)
                new_population.append(child)
                
        # Keep some diversity by adding random individuals
        while len(new_population) < pop_size:
            h = np.random.random(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            new_population.append(h)
            
        population = new_population[:pop_size]
    
    return (best_h.tolist(), best_c5, n_points)
