import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with random perturbation
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Simulated Annealing
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Annealing parameters
    temp = 0.1 * best_c5  # Initial temperature
    cooling_rate = 0.999
    min_temp = 1e-8
    max_iterations = 10000
    iteration = 0
    
    # Main SA loop
    while temp > min_temp and iteration < max_iterations:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate neighbor by perturbing current solution
        h_new = h.copy()
        # Apply perturbation
        perturbation = np.random.normal(0, 0.02, n_points)
        h_new += perturbation
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = compute_c5(h_new)
        
        # Accept or reject
        delta = c5_new - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h = h_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new

        # Cool down
        temp *= cooling_rate
        iteration += 1
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
