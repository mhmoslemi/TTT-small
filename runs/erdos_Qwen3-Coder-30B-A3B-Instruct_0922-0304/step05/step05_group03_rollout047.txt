import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
    
    # Normalize to sum to target_sum
    h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Start with a gradient-based optimization
    from scipy.optimize import minimize
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Use SLSQP for the main optimization
    if time.time() - start_time < budget_s - 1:
        result = minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-9}
        )
        
        if result.success:
            h = result.x
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Try to refine with different search strategies
    # Perturb the current best and try PSO-style updates
    if time.time() - start_time < budget_s - 1:
        # Create perturbed individuals
        for _ in range(20):
            if time.time() - start_time > budget_s - 1:
                break
                
            h_perturbed = best_h + np.random.normal(0, 0.02, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (target_sum / np.sum(h_perturbed))
            
            result = minimize(
                objective,
                h_perturbed,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 50, 'ftol': 1e-8}
            )
            
            if result.success:
                h_test = result.x
                c5_test = evaluate_c5(h_test, dx)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h_test.copy()
                    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
