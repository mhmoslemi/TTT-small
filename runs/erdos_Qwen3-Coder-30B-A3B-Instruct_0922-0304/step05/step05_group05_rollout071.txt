import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def compute_gradient(h, max_lags):
        gradient = np.zeros(n_points)
        for k in max_lags:
            # Gradient contribution from forward lags
            for j in range(n_points):
                if j + k < n_points:
                    gradient[j] += (1 - h[j + k])
            # Gradient contribution from backward lags
            for j in range(n_points):
                if j - k >= 0:
                    gradient[j] -= h[j - k]
        return gradient / len(max_lags) if max_lags else np.zeros(n_points)

    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Initialize variables
    best_c5 = float('inf')
    best_h = h.copy()
    learning_rate = 0.01
    stagnation_count = 0
    
    # Main optimization loop
    for iteration in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap and find maximum
        overlaps = correlate(h, 1 - h, mode="full")
        max_val = overlaps.max()
        max_lags = [i - (n_points - 1) for i, val in enumerate(overlaps) if val == max_val]
        
        # Compute gradient
        gradient = compute_gradient(h, max_lags)
        
        # Update with projected gradient descent
        h -= learning_rate * gradient
        h = project_h(h)
        
        # Evaluate
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Adjust learning rate
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= 0.9
        if stagnation_count > 50:
            learning_rate *= 0.9
            stagnation_count = 0

    # Final local optimization with SLSQP
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        result = minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_c5 = compute_c5(final_h)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h
    except:
        pass  # If local search fails, keep previous best

    return (best_h.tolist(), best_c5, n_points)
