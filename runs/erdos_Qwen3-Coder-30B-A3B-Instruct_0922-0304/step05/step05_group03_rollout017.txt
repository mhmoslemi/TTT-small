import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    # Local search parameters
    max_iter = 100
    step_size = 0.01
    patience = 10
    improvement_threshold = 1e-6
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
        
    # Simulated Annealing approach
    current_c5 = compute_c5(h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    # Temperature schedule
    temp_init = 0.1
    temp_decay = 0.995
    min_temp = 1e-6
    
    temp = temp_init
    stable_count = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        if temp < min_temp:
            temp = min_temp
            
        # Make a small perturbation
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        perturbation = np.random.normal(0, step_size * temp)
        h_new[idx] += perturbation
        
        # Project to feasible region
        h_new = normalize_h(h_new, target_sum)
        
        new_c5 = compute_c5(h_new)
        delta_c5 = new_c5 - current_c5
        
        # Accept or reject
        if delta_c5 < 0 or np.random.random() < np.exp(-delta_c5 / temp):
            h = h_new
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                stable_count = 0
            else:
                stable_count += 1
        else:
            stable_count += 1
            
        # Reduce temperature
        temp *= temp_decay
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Check for convergence
        if stable_count > patience:
            temp *= 0.5
            stable_count = 0
            
    # Final refinement with gradient-based optimization
    def objective(h):
        return compute_c5(h)
        
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Try several local minima
    for _ in range(5):
        h_refined = best_h + np.random.normal(0, 0.005, n_points)
        h_refined = normalize_h(h_refined, target_sum)
        
        result = minimize(
            objective,
            h_refined,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        
        if result.success:
            h_test = result.x
            h_test = normalize_h(h_test, target_sum)
            c5_test = compute_c5(h_test)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_test.copy()
                
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
