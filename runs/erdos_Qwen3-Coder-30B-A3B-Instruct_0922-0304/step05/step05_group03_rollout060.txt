import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradients(h, dx, k_max):
    """Compute gradient of C_5 with respect to h."""
    # Compute correlation
    corr = correlate(h, 1 - h, mode="full")
    # Find the maximum value and corresponding lag
    max_idx = np.argmax(corr)
    max_lag = max_idx - (len(h) - 1)
    
    # Gradient computation
    grad = np.zeros_like(h)
    for i in range(len(h)):
        # Compute the derivative of max correlation term
        # Overlap at lag k_max is sum over all j: h[j] * (1 - h[j+k_max])
        # For h[i], the derivative = -1 if h[i+k_max] is involved
        # For k_max = max_lag (the optimal lag)
        if i + max_lag >= 0 and i + max_lag < len(h):
            grad[i] = -1.0
    
    # Apply small smoothing to avoid zero gradient
    grad = grad + 1e-8 * np.random.randn(len(h))
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Track history for adaptive step size
    last_improvement = 0
    iteration = 0
    lr = 0.1
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        c5 = evaluate_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient
        gradients = compute_gradients(h, dx, np.argmax(correlate(h, 1-h, mode="full")) - (n_points-1))
        
        # Update with gradient descent
        h = h - lr * gradients
        
        # Project to feasible region
        h = normalize_h(h, target_sum)
        
        # Coordinate-wise improvement
        for i in range(n_points):
            # Try perturbation
            old_val = h[i]
            h[i] = old_val + np.random.uniform(-0.01, 0.01)
            h = normalize_h(h, target_sum)
            
            new_c5 = evaluate_c5(h, dx)
            if new_c5 < c5:
                c5 = new_c5
            else:
                h[i] = old_val  # Revert if no improvement
                
        # Adaptive learning rate
        if iteration - last_improvement > 50:
            lr *= 0.5
            last_improvement = iteration
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
