import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region [0,1] with correct sum."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Construct initial solution
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Early exit if already good enough
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    iteration = 0
    max_iter = 500
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full")
        
        # Select top lags contributing to C5
        # Find indices around the maximum
        max_idx = np.argmax(overlap)
        half_window = max(1, n_points // 10)
        lag_indices = np.arange(max(0, max_idx - half_window), min(len(overlap), max_idx + half_window))
        
        # Pick top contributing lags (we focus on a few key lags)
        top_lags = np.argsort(overlap[lag_indices])[-5:][::-1]
        top_lags = lag_indices[top_lags]
        
        # Formulate refined optimization problem
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        # Local search using SLSQP
        result = minimize(objective, h.copy(), method='SLSQP', bounds=bounds, 
                         constraints=constraints, options={'maxiter': 50, 'ftol': 1e-6})
        
        if result.success:
            h_refined = project_to_feasible(result.x, target_sum)
            c5_refined = compute_c5(h_refined, dx)
            
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = h_refined.copy()
        
        # Local random perturbation
        h_perturbed = h + np.random.normal(0, 0.01, n_points)
        h_perturbed = project_to_feasible(h_perturbed, target_sum)
        c5_perturbed = compute_c5(h_perturbed, dx)
        
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed.copy()
        
        # Update current solution
        h = project_to_feasible(h_perturbed, target_sum)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
