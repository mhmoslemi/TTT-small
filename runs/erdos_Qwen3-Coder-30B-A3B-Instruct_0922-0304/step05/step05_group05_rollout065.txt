import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: piecewise function with 3 regions
        h = np.zeros(n_points)
        mid1 = n_points // 4
        mid2 = 3 * n_points // 4
        h[:mid1] = 0.5
        h[mid1:mid2] = 0.25
        h[mid2:] = 0.5
        # Add small random noise
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_to_constraint(h):
        """Ensure sum(h) = n_points / 2 and h[i] in [0,1]"""
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        else:
            h = np.ones_like(h) * (n_points / 2) / n_points
        return h
    
    # Global best
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    # Swap-based local search
    n_swaps = 1000
    swap_size = 5
    temperature = 0.1
    
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    for _ in range(n_swaps):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Select two non-overlapping segments
        seg1_start = np.random.randint(0, n_points - swap_size)
        seg2_start = np.random.randint(0, n_points - swap_size)
        
        # Ensure segments don't overlap
        while abs(seg1_start - seg2_start) < swap_size:
            seg2_start = np.random.randint(0, n_points - swap_size)
            
        # Make copies for swapping
        h_temp = current_h.copy()
        seg1_vals = h_temp[seg1_start:seg1_start+swap_size]
        seg2_vals = h_temp[seg2_start:seg2_start+swap_size]
        
        # Swap the segments
        h_temp[seg1_start:seg1_start+swap_size] = seg2_vals
        h_temp[seg2_start:seg2_start+swap_size] = seg1_vals
        
        # Project to constraints
        h_temp = project_to_constraint(h_temp)
        
        # Compute new overlap
        new_c5 = compute_c5(h_temp)
        
        # Accept with simulated annealing probability
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / temperature):
            current_h = h_temp
            current_c5 = new_c5
            
        # Update global best
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
    
    # Final polish with local optimization
    from scipy.optimize import minimize
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    try:
        result = minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            best_h = result.x
            best_h = project_to_constraint(best_h)
            best_c5 = compute_c5(best_h)
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
