import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Annealing with local search approach
    temp = 1.0
    cooling_rate = 0.9995
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Current overlap computation
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Attempt local search with temperature-based perturbation
        new_h = h.copy()
        
        # Random perturbation
        perturbation = np.random.normal(0, 0.01 * temp, n_points)
        new_h += perturbation
        new_h = np.clip(new_h, 0, 1)
        # Renormalize
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Evaluate new solution
        new_overlap = correlate(new_h, 1 - new_h, mode="full")
        new_max_overlap = new_overlap[np.argmax(new_overlap)] * dx
        
        # Accept if better, or with probability based on temperature
        if new_max_overlap < max_overlap or np.random.rand() < np.exp((max_overlap - new_max_overlap) / temp):
            h = new_h
            max_overlap = new_max_overlap
        
        # Cool down
        temp *= cooling_rate
        
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
