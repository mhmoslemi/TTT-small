import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def evaluate_individual(h, dx):
    """Evaluate objective and check constraints."""
    c5 = compute_c5(h, dx)
    return c5

    if best_c5 <= 0.3807:

        if max_overlap * dx < best_c5:

        # Gradient-like update
        # We shift the function to reduce the maximum overlap
        # Approximate gradient using finite differences
        h_new = h.copy()

        # Analyze which indices to adjust based on max overlap
