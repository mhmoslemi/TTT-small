import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Initialize swarm
    particles.append(h.copy())
    velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Add random particles
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Track best solutions
    best_particle_idx = 0
    best_c5 = float('inf')
    global_best = None
    global_best_c5 = float('inf')
    
    # PSO parameters
    w_max = 0.9
    w_min = 0.4
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.1
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        w = w_max - (w_max - w_min) * (time.time() - start_time) / budget_s
        
        for i in range(swarm_size):
            # Evaluate current particle
            c5 = compute_c5(particles[i])
            
            # Update personal best
            if c5 < best_c5:
                best_c5 = c5
                best_particle_idx = i
                global_best = particles[i].copy()
                global_best_c5 = c5
                
            # Early exit if target achieved
            if global_best_c5 <= 0.3807:
                break
                
            # Update global best
            if c5 < global_best_c5:
                global_best_c5 = c5
                global_best = particles[i].copy()
            
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (particles[best_particle_idx] - particles[i]) +
                           c2 * r2 * (global_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] += velocities[i]
            
            # Keep within bounds [0,1]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize to correct sum
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
        if global_best_c5 <= 0.3807:
            break
    
    if global_best is None:
        # Fallback to the last particle if nothing was found
        global_best = particles[-1].copy()
        global_best_c5 = compute_c5(global_best)
    
    return (global_best.tolist(), global_best_c5, n_points)
