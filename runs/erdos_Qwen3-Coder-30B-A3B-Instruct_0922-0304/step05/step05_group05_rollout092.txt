import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_lags = 2 * n_points - 1
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: piecewise function
        h = np.ones(n_points)
        h[n_points//4:3*n_points//4] = 0
        h[n_points//2:] = 0.5
        # Add random noise
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def rescale(h):
        return h * (n_points / 2) / np.sum(h)

    def swap_segments(h, i1, i2, size):
        h_new = h.copy()
        # Extract segments
        seg1 = h_new[i1:i1+size].copy()
        seg2 = h_new[i2:i2+size].copy()
        # Swap
        h_new[i1:i1+size] = seg2
        h_new[i2:i2+size] = seg1
        # Rescale to meet sum constraint
        h_new = rescale(h_new)
        return h_new

    # Simulated Annealing with swap-based moves
    best_h = h.copy()
    best_c5 = compute_c5(best_h)

    # Early exit condition
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)

    # Parameters
    max_swaps = 5000
    swap_size = 5
    # Start with high temperature
    temp = 0.2
    
    # Simulated Annealing
    for it in range(max_swaps):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Select two non-overlapping segments to swap
        i1 = np.random.randint(0, n_points - swap_size)
        i2 = np.random.randint(0, n_points - swap_size)
        
        # Avoid overlapping regions
        while abs(i1 - i2) < swap_size:
            i2 = np.random.randint(0, n_points - swap_size)
            
        # Perform swap
        h_new = swap_segments(best_h, i1, i2, swap_size)
        
        # Compute new C5
        new_c5 = compute_c5(h_new)
        
        # Accept or reject
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
        else:
            # Probabilistic acceptance
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / temp):
                best_h = h_new
                best_c5 = new_c5
                
        # Cool down temperature
        temp = max(1e-6, temp * 0.999)
        
        if best_c5 <= 0.3807:
            break

    # Final polish with local optimization
    try:
        # Define constraint and bounds
        def constraint_sum(h):
            return np.sum(h) - n_points / 2

        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        result = minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8}
        )
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = rescale(best_h)
            best_c5 = compute_c5(best_h)
    except:
        pass
        
    return (best_h.tolist(), best_c5, n_points)
