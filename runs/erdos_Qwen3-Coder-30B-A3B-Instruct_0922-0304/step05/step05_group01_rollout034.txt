import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def validate_and_repair(h, n_points):
    """Ensure h is feasible: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = validate_and_repair(np.array(initial_h_values), n_points)
    else:
        # Fallback: simple binary step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = validate_and_repair(h, n_points)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Phase 1: Coarse optimization (n=35)
    n_coarse = 35
    dx_coarse = 2.0 / n_coarse
    
    # Initialize coarse h
    if 'initial_h_values' in globals():
        h_coarse = validate_and_repair(np.array(initial_h_values), n_points)
        # Downsample to coarse
        h_coarse = np.array([np.mean(h_coarse[i*2:(i+1)*2]) for i in range(n_coarse)])
    else:
        h_coarse = np.ones(n_coarse)
        h_coarse[n_coarse//2:] = 0
        h_coarse = validate_and_repair(h_coarse, n_coarse)

    def compute_c5_coarse(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx_coarse
    
    def constraint_sum_coarse(h):
        return np.sum(h) - n_coarse / 2
        
    constraints_coarse = [{'type': 'eq', 'fun': constraint_sum_coarse}]
    bounds_coarse = [(0, 1) for _ in range(n_coarse)]
    
    # Optimize coarse
    if time.time() - start_time < budget_s * 0.2:
        result_coarse = minimize(
            compute_c5_coarse,
            h_coarse,
            method='SLSQP',
            bounds=bounds_coarse,
            constraints=constraints_coarse,
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        if result_coarse.success:
            h_coarse = result_coarse.x
            h_coarse = validate_and_repair(h_coarse, n_coarse)
    
    # Phase 2: Upsample and refine
    # Upsample coarse to fine grid via nearest neighbor
    h_fine = np.repeat(h_coarse, 2)
    h_fine = validate_and_repair(h_fine, n_points)
    
    # Fine optimization
    max_iter = 200
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5(h_fine)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_fine.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Optimization step
        result = minimize(
            compute_c5,
            h_fine,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 20, 'ftol': 1e-8}
        )
        
        if result.success:
            h_fine = result.x
            h_fine = validate_and_repair(h_fine, n_points)
        else:
            # Perturb if optimization fails
            h_fine += np.random.normal(0, 0.01, n_points)
            h_fine = validate_and_repair(h_fine, n_points)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
