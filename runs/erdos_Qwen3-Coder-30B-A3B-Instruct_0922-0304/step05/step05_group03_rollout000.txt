import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def halton_sequence(start, n_points, dimensions):
    """Generate Halton sequence for better sampling."""
    result = np.zeros((n_points, dimensions))
    for i in range(dimensions):
        base = 2 + i  # Use different bases for each dimension
        for j in range(n_points):
            result[j, i] = 0
            k = j + 1
            base_k = 1
            while k > 0:
                base_k *= base
                result[j, i] += (k % base) / base_k
                k //= base
    return result
