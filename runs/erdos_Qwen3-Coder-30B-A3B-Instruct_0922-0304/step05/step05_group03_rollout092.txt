import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
import random

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
        
    # Simple local optimization with simulated annealing approach
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Parameters for simulated annealing
    temp = 1.0
    min_temp = 1e-6
    alpha = 0.995
    max_iter = 5000
    
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()
    
    # Annealing iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate neighbor solution
        neighbor_h = current_h.copy()
        idx = random.randint(0, n_points - 1)
        delta = random.uniform(-0.1, 0.1)
        neighbor_h[idx] += delta
        
        # Project to feasible set
        neighbor_h = normalize_h(neighbor_h, target_sum)
        
        neighbor_c5 = compute_c5(neighbor_h)
        
        # Accept or reject
        if neighbor_c5 < current_c5:
            current_h = neighbor_h
            current_c5 = neighbor_c5
        else:
            # Accept with probability based on temperature
            if random.random() < np.exp(-(neighbor_c5 - current_c5) / temp):
                current_h = neighbor_h
                current_c5 = neighbor_c5
                
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Cool down
        temp *= alpha
        
        if best_c5 <= 0.3807:
            break
    
    # Final local refinement with SLSQP
    def objective(h):
        return compute_c5(h)
        
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    result = minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    if result.success:
        final_h = result.x
        final_h = normalize_h(final_h, target_sum)
        final_c5 = compute_c5(final_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h
            
    return (best_h.tolist(), best_c5, n_points)
