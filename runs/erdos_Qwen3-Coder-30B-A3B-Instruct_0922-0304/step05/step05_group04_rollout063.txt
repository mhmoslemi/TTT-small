import numpy as np
from scipy.signal import correlate
import time

from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 5000

    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use alternating pattern (good for overlap studies)
    h_alternating = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            h_alternating[i] = 1.0
    h_alternating = h_alternating * (n_points / 2) / np.sum(h_alternating)

    # Create a smoothed version of the starting point
    h_smooth = np.convolve(h, np.ones(5)/5, mode='same')
    h_smooth = h_smooth * (n_points / 2) / np.sum(h_smooth)
    h_smooth = np.clip(h_smooth, 0, 1)

    # Start with the best of the initialization approaches
    h = h_smooth.copy()
    
    iteration = 0
    
    # Main optimization loop with multiple strategies
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

        # Strategy 1: Gradient-based descent with projected updates
        # Approximate gradient using finite differences
        g = np.zeros(n_points)
        eps = 1e-4

        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            # Compute overlap
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps) * dx

        # Update using gradient descent
        learning_rate = 0.005
        new_h = h - learning_rate * g
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Accept the better of the two (current or updated) depending on overlap
        h = new_h.copy()
        
        # Occasionally revert to a known good solution (if gap is large)
        if iteration % 100 == 0:
            # Perturb more significantly
            pert = np.random.normal(0, 0.01, n_points)
            pert = pert * (np.sum(h) > n_points / 2) - pert * (np.sum(h) <= n_points / 2)
            
            new_h = h + pert
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            # Keep new_h only if it's an improvement
            overlap_new = correlate(new_h, 1 - new_h, mode="full").max() * dx
            if overlap_new < max_overlap:
                h = new_h.copy()

        # Random local search with perturbation
        if iteration % 50 == 0 and np.random.rand() < 0.3:
            # Use random more significant modification with smaller chance
            idx = np.random.randint(0, n_points)
            h[idx] += np.random.normal(0, 0.02)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    # Final evaluation of best solution
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
