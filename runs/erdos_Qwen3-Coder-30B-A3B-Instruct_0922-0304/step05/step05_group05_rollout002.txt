import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with slight randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        # Compute overlap integral
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Track best solution
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T0 = 1.0
    alpha = 0.995
    T_min = 1e-8
    max_iter = 50000
    iter_per_temp = 50
    
    T = T0
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    # Main SA loop
    iterations = 0
    for temp_iter in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for _ in range(iter_per_temp):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Generate neighbor by perturbing current solution
            h_new = current_h + np.random.normal(0, 0.02, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate neighbor
            c5_new = compute_c5(h_new)
            
            # Accept or reject based on SA criteria
            delta = c5_new - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                current_h = h_new
                current_c5 = c5_new
                
                # Update best solution
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
            
            iterations += 1
        
        # Cooling schedule
        T = max(T * alpha, T_min)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final validation and evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
