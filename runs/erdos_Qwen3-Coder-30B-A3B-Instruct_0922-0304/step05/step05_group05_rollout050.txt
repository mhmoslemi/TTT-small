import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T0 = 1.0
    T_min = 1e-8
    alpha = 0.99
    max_iter = 10000
    current_T = T0
    
    # Track improvements
    last_improvement = 0
    patience = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create neighbor solution
        h_new = h + np.random.normal(0, current_T * 0.1, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = compute_c5(h_new)
        
        # Accept or reject the new solution
        delta = c5_new - best_c5
        if delta < 0 or np.random.random() < np.exp(-delta / current_T):
            h = h_new
            c5 = c5_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = iteration
        
        # Adapt temperature
        current_T = max(T_min, T0 * (alpha ** iteration))
        
        # Early exit if we've improved significantly
        if best_c5 <= 0.3807:
            break
            
        # Early stopping if no improvement in patience iterations
        if iteration - last_improvement > patience:
            break
    
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
