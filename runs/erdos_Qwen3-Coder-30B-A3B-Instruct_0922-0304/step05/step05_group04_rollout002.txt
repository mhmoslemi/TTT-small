import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initialize with alternating blocks
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if i % 2 == 0 else 0
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 10000
    iteration = 0
    
    # Use a more direct optimization approach with gradient-based updates
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        current_c5 = overlap[max_k] * dx
        
        # Store best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient estimate using finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            
            h_plus[i] += eps
            h_minus[i] -= eps
            
            # Clamp and renormalize
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            # Compute overlap
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (overlap_plus[max_k] - overlap_minus[max_k]) / (2 * eps) * dx
        
        # Update using gradient descent
        step_size = 0.01
        new_h = h - step_size * grad
        
        # Clamp and renormalize
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        h = new_h
    
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
