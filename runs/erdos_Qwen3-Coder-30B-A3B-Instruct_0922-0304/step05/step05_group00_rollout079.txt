import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_feasible(h):
    """Project h to feasible region [0,1] and normalize."""
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from existing construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_feasible(h)
        h = normalize_h(h, target_sum)
    else:
        # Create a simple piecewise constant construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_feasible(h)
        h = normalize_h(h, target_sum)

    best_c5 = compute_c5(h, dx)
    best_h = h.copy()

    # Optimization loop
    max_iter = 2000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate current solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Perform local optimization
        def objective(h):
            return compute_c5(h, dx)
        
        # Constraint function for sum
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        # bounds
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Apply sequential quadratic programming with bounds
        result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                         constraints=constraint, options={'maxiter': 100})
        
        if result.success:
            h = result.x
        else:
            # Fall back to gradient descent in feasible direction
            grad = np.zeros_like(h)
            # Approximate gradient using finite differences (simplified)
            eps = 1e-4
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] = min(1.0, h[i] + eps)
                h_minus = h.copy()
                h_minus[i] = max(0.0, h[i] - eps)
                grad[i] = (compute_c5(h_plus, dx) - compute_c5(h_minus, dx)) / (2 * eps)
            
            # Update step (adaptive learning rate)
            lr = 0.1 / (1 + iteration / 100)
            h = h - lr * grad
            h = np.clip(h, 0, 1)
            
        # Normalize to correct sum
        h = normalize_h(h, target_sum)
        
        # Add small random jitter to escape local minima
        noise_factor = max(0.001, 0.01 - iteration * 1e-5)
        h += np.random.normal(0, noise_factor, n_points)
        h = project_feasible(h)
        h = normalize_h(h, target_sum)
        
        iteration += 1
        
    # Final check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
