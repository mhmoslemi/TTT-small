import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_feasible(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_feasible(h, target_sum)
    else:
        # Default initialization
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = project_feasible(h, target_sum)
    
    # Optimization parameters
    learning_rate = 0.1
    momentum_decay = 0.9
    max_iterations = 10000
    momentum = np.zeros(n_points)
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    for i in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        current_c5 = compute_c5(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Find the maximum overlap and corresponding shift
        overlap = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(overlap)
        max_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient for the worst shift
        gradient = np.zeros(n_points)
        if max_shift >= 0:
            # h[i] * (1 - h[i + max_shift]) for i in range(n_points - max_shift)
            gradient[:n_points - max_shift] += (1 - h[max_shift:])
            gradient[max_shift:] -= h[:n_points - max_shift]
        else:
            # h[i] * (1 - h[i - max_shift]) for i in range(-max_shift, n_points)
            neg_shift = -max_shift
            gradient[neg_shift:] += (1 - h[:n_points - neg_shift])
            gradient[:n_points - neg_shift] -= h[neg_shift:]
        
        # Apply momentum
        momentum = momentum_decay * momentum + (1 - momentum_decay) * gradient
        
        # Line search for step size
        step_size = learning_rate
        for _ in range(10):
            h_trial = h - step_size * momentum
            h_trial = project_feasible(h_trial, target_sum)
            trial_c5 = compute_c5(h_trial, dx)
            
            if trial_c5 < current_c5:
                h = h_trial
                break
            else:
                step_size *= 0.5
                
        # Adapt learning rate
        if i % 100 == 0 and i > 0:
            # Reduce learning rate if no improvement
            learning_rate *= 0.95
            if learning_rate < 1e-6:
                learning_rate = 1e-6
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
