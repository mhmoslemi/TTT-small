import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def line_search(h, grad, dx, max_alpha=1.0, tolerance=1e-4):
    """Perform line search to find optimal step size."""
    alpha = max_alpha
    c5_current = compute_c5(h, dx)
    
    # Try various step sizes
    for _ in range(20):
        h_new = project_h(h - alpha * grad, len(h) / 2)
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < c5_current:
            return alpha, h_new
        
        alpha *= 0.5
    
    return alpha, h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, n_points / 2)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = project_h(h, n_points / 2)
    
    # Optimizer parameters
    momentum = 0.9
    step_size = 0.1
    prev_grad = np.zeros(n_points)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    iter_count = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Compute all correlations
        corr = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        
        # Compute gradient at max_k
        grad = np.zeros(n_points)
        if max_k < n_points:
            # Positive shift case
            grad[:n_points - max_k] += (1 - h[max_k:])
        else:
            # Negative shift case
            shift = max_k - n_points + 1
            grad[shift:] += (1 - h[:n_points - shift])
        
        # Apply momentum
        descent = momentum * prev_grad + grad
        prev_grad = descent
        
        # Line search
        step_size, h_new = line_search(h, descent, dx, step_size)
        
        # Update h
        h = h_new
        
        # Evaluate current solution
        c5 = compute_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adjust step size
        if no_improvement_count > 20:
            step_size *= 0.9
            no_improvement_count = 0
            
        # Early termination
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
