import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 500
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Solve convex relaxation using CVXPY
        try:
            variables = cp.Variable(n_points)
            constraints = [
                variables >= 0,
                variables <= 1,
                cp.sum(variables) == n_points / 2
            ]
            
            # For simplicity, we approximate the objective as a linear constraint
            # This is a rough approximation; the actual constraint would require
            # careful handling of the max operator
            
            # For now, let's approximately minimize the objective: max_k (h * (1-h))[k]
            # We use our current h to estimate a linear approximation around that point
            
            # The full correlation is the main constraint
            # This approach is approximate -- more precise convex formulations
            # involving epigraph variables would be better but more complex
            
            # Simplify by trying direct optimization on the convex problem
            # This will still be a rough approximation but better than naive methods
            
            # Try a reformulation using QCQP or other relaxation if necessary
            # Here we simplify by using a feasibility restoration approach after optimization
            
            # Define a simple objective to reduce overlap (in a convex way)
            # This is simplified and not fully mathematically equivalent
            # But it provides a better convergence than just using gradient based
            # methods without suitable convexity
            objective = cp.Minimize(cp.sum(variables * (1 - variables)))
            prob = cp.Problem(objective, constraints)
            
            try:
                prob.solve(solver=cp.SCS, verbose=False)
                if prob.status == 'optimal':
                    h_new = variables.value
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                else:
                    # Fallback
                    h_new = h.copy()
            except:
                # Fallback to current h
                h_new = h.copy()

        except Exception as e:
            # Fallback
            h_new = h.copy()
            
        h = h_new
        
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        if best_c5 <= 0.3807:
            break
            
        # Periodic re-initialization with a small random component 
        if iteration % 5 == 0:
            # Add slight perturbation to prevent stagnation
            h += np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
