import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradients(h, k_max, dx):
    """Compute gradients of overlap at k_max with respect to h[i]"""
    n = len(h)
    grad = np.zeros(n)
    # derivative of overlap at k_max with respect to h[i]
    for i in range(n):
        # at lag k_max: overlap = sum h[j] * (1 - h[j + k_max])
        # d/dh[i] = 0 unless i = j or i = j + k_max
        # contribution from h[j] term: (1 - h[j + k_max]) if j + k_max = i (i.e. j = i - k_max)
        # contribution from (1 - h[j + k_max]) term: -h[j] if j = i (i.e. j + k_max = i + k_max)
        # Only specific pairs contribute:
        # Term h[j] * (1 - h[j + k_max]): d/dh[i] = (1 - h[i + k_max]) if j = i
        # Term (1 - h[j]) * h[j + k_max]: d/dh[i] = -h[i + k_max] if j + k_max = i  
        # i = j + k_max -> j = i - k_max
        if i >= k_max:
            grad[i] += (1 - h[i - k_max])  # contribution of h[j] term where j = i - k_max
        if i + k_max < n:
            grad[i] -= h[i + k_max]  # contribution of -h[j+k_max] term where j = i
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent parameters
    alpha = 0.1
    patience = 0
    patience_limit = 10
    decay_factor = 0.5
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current overlap
        overlap_full = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap_full)
        current_c5 = overlap_full[k_max] * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient
        grad = compute_gradients(h, k_max, dx)
        
        # Update h
        h_new = h - alpha * grad
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Re-normalize
        h_new = normalize_h(h_new, target_sum)
        
        # Check if this improved our solution
        new_overlap = correlate(h_new, 1 - h_new, mode="full")
        new_c5 = new_overlap[k_max] * dx
        
        if new_c5 < current_c5:
            h = h_new
            patience = 0
        else:
            patience += 1
            if patience >= patience_limit:
                alpha *= decay_factor
                patience = 0

    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
