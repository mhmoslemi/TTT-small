import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    n_particles = 20
    w_start = 0.8
    w_end = 0.1
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Add initial solution as first particle
    particles.append(h.copy())
    velocities.append(np.zeros(n_points))
    
    # Add random perturbations
    for _ in range(n_particles - 1):
        h_pert = h + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        particles.append(h_pert)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))

    # Personal bests
    p_best_positions = [p.copy() for p in particles]
    p_best_values = [compute_c5(p) for p in particles]
    
    # Global best
    g_best_idx = np.argmin(p_best_values)
    g_best_position = p_best_positions[g_best_idx].copy()
    g_best_value = p_best_values[g_best_idx]
    
    # PSO iteration
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel_update = (
                w * velocities[i] +
                c1 * r1 * (p_best_positions[i] - particles[i]) +
                c2 * r2 * (g_best_position - particles[i])
            )
            
            # Clip velocity
            vel_update = np.clip(vel_update, -v_max, v_max)
            velocities[i] = vel_update
            
            # Update position
            new_pos = particles[i] + velocities[i]
            
            # Ensure bounds and constraint
            new_pos = np.clip(new_pos, 0, 1)
            new_pos = new_pos * (n_points / 2) / np.sum(new_pos)
            
            # Evaluate new position
            new_c5 = compute_c5(new_pos)
            
            # Update personal best
            if new_c5 < p_best_values[i]:
                p_best_values[i] = new_c5
                p_best_positions[i] = new_pos.copy()
                
                # Update global best
                if new_c5 < g_best_value:
                    g_best_value = new_c5
                    g_best_position = new_pos.copy()
            
            # Update particle position
            particles[i] = new_pos.copy()
            
            # Early termination
            if g_best_value <= 0.3807:
                break
        
        if g_best_value <= 0.3807:
            break
    
    # Final return values
    final_c5 = g_best_value
    return (g_best_position.tolist(), final_c5, n_points)
