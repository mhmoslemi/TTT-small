import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: piecewise function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid//2] = 1.0
        h[mid//2:mid] = 0.0
        h[mid:3*mid//2] = 1.0
        h[3*mid//2:] = 0.0
        # Add small noise to break symmetry
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Swap-based local search parameters
    n_swaps = 1000
    swap_size = 5
    
    # Initialize with current best
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()
    
    # Simulated annealing style swap search
    temperature = 0.1
    
    for _ in range(n_swaps):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create candidate by swapping
        h_new = current_h.copy()
        
        # Select two non-overlapping segments
        seg1_start = np.random.randint(0, n_points - swap_size)
        seg2_start = np.random.randint(0, n_points - swap_size)
        
        # Ensure they don't overlap
        while (seg2_start < seg1_start + swap_size and seg1_start < seg2_start + swap_size):
            seg2_start = np.random.randint(0, n_points - swap_size)
        
        # Swap values
        h_new[seg1_start:seg1_start+swap_size], h_new[seg2_start:seg2_start+swap_size] = \
            h_new[seg2_start:seg2_start+swap_size].copy(), h_new[seg1_start:seg1_start+swap_size].copy()
        
        # Rescale to maintain sum constraint
        sum_old = np.sum(h_new)
        h_new = h_new * (n_points / 2) / sum_old
        h_new = np.clip(h_new, 0, 1)
        
        # Compute new overlap
        c5_new = compute_c5(h_new)
        
        # Accept or reject based on temperature
        if c5_new < current_c5 or np.random.rand() < np.exp((current_c5 - c5_new) / temperature):
            current_h = h_new
            current_c5 = c5_new
            
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Early exit if target is achieved
        if best_c5 <= 0.3807:
            break
    
    # Final polish with SLSQP if time allows
    if time.time() - start_time < budget_s - 1:
        try:
            result = minimize(
                compute_c5,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 30, 'ftol': 1e-8}
            )
            if result.success:
                best_h = result.x
                best_h = np.clip(best_h, 0, 1)
                best_h = best_h * (n_points / 2) / np.sum(best_h)
                best_c5 = compute_c5(best_h)
        except:
            pass
    
    return (best_h.tolist(), best_c5, n_points)
