import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent parameters
    max_iter = 1000
    learning_rate = 0.01
    patience = 0
    patience_limit = 50
    
    # Gradient descent loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute cross-correlation and find maximum
        overlaps = correlate(h, 1 - h, mode="full")
        max_val = overlaps.max()
        max_lags = [i - (n_points - 1) for i, val in enumerate(overlaps) if val == max_val]
        
        # Compute gradient for maximum lag(s)
        gradient = np.zeros(n_points)
        for k in max_lags:
            for j in range(n_points):
                if j + k < n_points:
                    gradient[j] += (1 - h[j + k])
                if j - k >= 0:
                    gradient[j] -= h[j - k]
        
        gradient /= len(max_lags)
        
        # Update step
        h -= learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = compute_c5(h)
        
        # Track best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Adaptive learning rate
        if i % 100 == 0 and i > 0:
            learning_rate *= 0.9
            
        if patience >= patience_limit:
            # Try SLSQP local search for fine-tuning
            try:
                result = minimize(
                    compute_c5,
                    h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraints,
                    options={'maxiter': 30, 'ftol': 1e-8}
                )
                if result.success:
                    h = result.x
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
            except:
                pass
            
            patience = 0  # Reset patience after local search
    
    # Final local optimization
    try:
        result = minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h = final_h * (n_points / 2) / np.sum(final_h)
            final_c5 = compute_c5(final_h)
            if final_c5 < best_c5:
                best_h = final_h
                best_c5 = final_c5
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
