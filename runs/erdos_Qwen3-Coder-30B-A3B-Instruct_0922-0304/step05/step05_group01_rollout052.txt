import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 20
    elite_fraction = 0.25
    
    # Initialize population
    def initialize_population():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
        population = [h]
        for _ in range(population_size - 1):
            mutated = h + np.random.normal(0, 0.01, n_points)
            mutated = np.clip(mutated, 0, 1)
            mutated = mutated * (n_points / 2) / np.sum(mutated)
            population.append(mutated)
        return population
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def mutate(h, mutation_std):
        mutated = h + np.random.normal(0, mutation_std, h.shape)
        mutated = np.clip(mutated, 0, 1)
        sum_h = np.sum(mutated)
        mutated = mutated * (n_points / 2) / sum_h
        return mutated
    
    def select_elite(population, fitnesses):
        indices = np.argsort(fitnesses)
        elite_count = max(1, int(len(population) * elite_fraction))
        return [population[i] for i in indices[:elite_count]]
    
    # Initialize
    population = initialize_population()
    best_c5 = float('inf')
    best_h = None
    
    # Evolutionary algorithm
    max_generations = 1000
    mutation_strength = 0.01
    no_improvement_count = 0
    max_no_improvement = 50
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = [compute_c5(h) for h in population]
        
        # Update best
        min_fitness = min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[np.argmin(fitnesses)].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Adapt mutation strength based on no improvement
        if no_improvement_count > max_no_improvement:
            mutation_strength *= 0.1
            no_improvement_count = 0
            
        # Selection
        elite = select_elite(population, fitnesses)
        
        # Generate new population
        new_population = []
        for individual in elite:
            new_population.append(mutate(individual, mutation_strength))
            new_population.append(mutate(individual, mutation_strength))
            
        # Fill up population
        while len(new_population) < population_size:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            new_population.append(h)
            
        population = new_population[:population_size]
    
    # Final evaluation
    if best_h is None:
        # Fallback to the best of last population
        fitnesses = [compute_c5(h) for h in population]
        best_c5 = min(fitnesses)
        best_h = population[np.argmin(fitnesses)].copy()
        
    return (best_h.tolist(), best_c5, n_points)
