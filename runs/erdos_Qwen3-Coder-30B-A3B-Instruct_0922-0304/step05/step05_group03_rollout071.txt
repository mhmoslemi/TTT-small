import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
    
    # Normalize 
    current_sum = np.sum(h)
    h = h * (target_sum / current_sum)
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
    
    # Helper for projection onto constraints
    def project_h(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
        return h
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    # Gradient-based local search using L-BFGS
    from scipy.optimize import minimize
    
    def objective(h):
        return compute_c5(h)
        
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Optimization step
        result = minimize(
            objective,
            best_h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 20, 'ftol': 1e-8}
        )
        
        if result.success:
            h_new = result.x
            h_new = project_h(h_new)
            new_c5 = compute_c5(h_new)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Perturb current solution to escape local optima
        noise = np.random.normal(0, 0.01, n_points)
        h_perturbed = best_h + noise
        h_perturbed = project_h(h_perturbed)
        perturbed_c5 = compute_c5(h_perturbed)
        
        if perturbed_c5 < best_c5:
            best_c5 = perturbed_c5
            best_h = h_perturbed.copy()
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
