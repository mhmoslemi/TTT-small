import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, k_max, dx, n_points):
    """Compute the gradient of overlap at k_max with respect to h."""
    grad = np.zeros(n_points)
    # For lag k_max, compute derivative of h[i] * (1 - h[i+k_max]) 
    # with respect to h[i]
    for i in range(n_points):
        if i + k_max < n_points:
            grad[i] += 1 - h[i + k_max]
        if i - k_max >= 0:
            grad[i] += 1 - h[i - k_max]
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    # Gradient descent parameters
    alpha = 0.1
    patience = 0
    patience_limit = 100
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        c5 = overlap[k_max] * dx
        
        # Track best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Check patience
        if patience > patience_limit:
            alpha *= 0.5
            patience = 0
            
        # Compute gradient
        grad = compute_gradient(h, k_max, dx, n_points)
        
        # Update step
        h_new = h - alpha * grad
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to sum constraint
        h_new = normalize_h(h_new, target_sum)
        
        # Perturb occasionally to escape local minima
        if iteration % 50 == 0 and np.random.random() < 0.2:
            h_new += np.random.normal(0, 0.02, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
        
        h = h_new
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
