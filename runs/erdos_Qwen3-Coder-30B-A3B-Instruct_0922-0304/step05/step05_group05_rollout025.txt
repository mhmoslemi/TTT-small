import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with small perturbations
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Simulated Annealing
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Initial temperature calculation
    temp = best_c5 * 0.1  # Initial temperature
    
    # Cooling schedule
    cooling_rate = 0.995
    min_temp = 1e-6
    
    # Maximum iterations
    max_iter = 10000
    
    # For tracking improvement
    last_improvement = 0
    patience = 100
    
    # Main SA loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate candidate solution
        h_new = h + np.random.normal(0, temp * 0.1, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute objective
        c5_new = compute_c5(h_new)
        
        # Accept or reject
        delta = c5_new - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h = h_new.copy()
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                last_improvement = iteration
                
        # Cool down
        temp *= cooling_rate
        if temp < min_temp:
            temp = min_temp
            
        # Early exit if found target bound
        if best_c5 <= 0.3807:
            break
            
        # Early termination if no improvement
        if iteration - last_improvement > patience:
            break
    
    return (best_h.tolist(), best_c5, n_points)
