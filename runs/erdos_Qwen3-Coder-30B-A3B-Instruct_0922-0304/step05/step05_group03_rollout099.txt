import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = normalize_h(h, target_sum)
    else:

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    def compute_c5(h):
        return evaluate_c5(h, dx)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Hill climbing with multiple restarts
    n_restarts = 100
    n_steps_per_restart = 1000
    max_iter = n_restarts * n_steps_per_restart
    
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()
    
    # Add noise to make a better starting point
    new_h = current_h + np.random.normal(0, 0.01, n_points)
    new_h = normalize_h(new_h, target_sum)
    new_c5 = compute_c5(new_h)
    if new_c5 < best_c5:
        best_c5 = new_c5
        best_h = new_h.copy()
        current_h = new_h.copy()
        current_c5 = new_c5
        
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s - 1:

        # Start with random perturbation of current best
        current_h = best_h + np.random.normal(0, 0.01, n_points)

            if new_c5 <= current_c5:

                current_c5 = new_c5
                
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                
            # Early exit if target achieved
            if best_c5 <= 0.3807:

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
