import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (target_sum / h_sum)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Iterative LP refinement
    max_iter = 50
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps to determine key lags
        overlap = correlate(h, 1 - h, mode="full")
        top_lags_idx = np.argsort(overlap)[-max(10, n_points//10):]  # Top 10% or at least 10
        top_lags = np.array(top_lags_idx) - (n_points - 1)  # Convert to lag indices
        
        # Formulate LP
        h_var = cp.Variable(n_points)
        C5 = cp.Variable()
        
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
        
        # Linearize overlap constraints using current h
        for k in top_lags:
            if k == 0:
                continue  # Skip zero lag for linearization
            # Compute linear approximation for lag k
            # overlap_k ≈ sum(h[i] * (1 - h[i+k]) for i in range(n_points-k)) * dx
            # Let's simplify to get:
            # For i = 0 to n_points - k - 1:
            # sum(h_var[i] * (1 - h[i+k]) + h[i] * (h_var[i+k] - h[i+k])) * dx <= C5
            
            left_terms = []
            right_terms = []
            for i in range(n_points - abs(k)):
                if k > 0:
                    # h_var[i] * (1 - h[i+k]) + h[i] * (h_var[i+k] - h[i+k])
                    left_terms.append(h_var[i] * (1 - h[i+k]))
                    right_terms.append(h[i] * (h_var[i+k] - h[i+k]))
                else:  # k < 0
                    k_abs = -k
                    left_terms.append(h_var[i+k_abs] * (1 - h[i]))
                    right_terms.append(h[i+k_abs] * (h_var[i] - h[i]))
            
            # Total linear term: sum(left_terms + right_terms)
            linear_term = cp.sum(left_terms) + cp.sum(right_terms)
            # Add constraint
            constraints.append(linear_term * dx <= C5)
        
        # Objective
        objective = cp.Minimize(C5)
        
        # Solve
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                if h_new is not None:
                    # Project onto feasibility
                    h_new = np.clip(h_new, 0, 1)
                    h_new_sum = np.sum(h_new)
                    if h_new_sum > 0:
                        h_new = h_new * (target_sum / h_new_sum)
                        
                    # Compute new C5
                    c5_new = np.max(correlate(h_new, 1 - h_new, mode="full")) * dx
                    
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        
                    h = h_new.copy()
                    
                    if best_c5 <= 0.3807:
                        break
        except:
            pass
    
    # Final evaluation
    final_c5 = np.max(correlate(best_h, 1 - best_h, mode="full")) * dx
    
    return (best_h.tolist(), final_c5, n_points)
