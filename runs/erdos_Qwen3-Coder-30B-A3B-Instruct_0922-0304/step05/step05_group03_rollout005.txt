import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradients(h, k_max, dx, n_points):
    """Compute gradients for the overlap at the maximum lag k_max."""
    # Create kernel 1-h for correlation
    kernel = 1 - h
    # Compute the gradient of the overlap at k_max
    grad = np.zeros(n_points)
    # For the overlap at k_max: sum h[i] * (1 - h[i+k_max]) 
    # i = 0..n_points-k_max-1
    for i in range(n_points - k_max):
        grad[i] += 1 - h[i + k_max]
    # But we also need to account for the derivative of h[i+k_max]
    # ... which will have +h[i] in d/dh[i] of the overlap
    for i in range(k_max, n_points):
        grad[i - k_max] += h[i]
    return grad * dx
