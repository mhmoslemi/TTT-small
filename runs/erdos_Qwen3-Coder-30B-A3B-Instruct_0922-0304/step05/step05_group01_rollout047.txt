import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 20
    max_iter = 1000
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 += np.random.normal(0, 0.05, n_points)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Create initial swarm
    for _ in range(swarm_size):
        h = h0.copy()
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        swarm.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Best positions
    personal_best = [h.copy() for h in swarm]
    personal_best_fitness = [float('inf')] * swarm_size
    global_best = h0.copy()
    global_best_fitness = float('inf')
    
    # PSO parameters
    w_start = 0.7
    w_end = 0.4
    c1 = c2 = 1.5
    max_velocity = 0.2
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(swarm_size):
            h = swarm[i]
            
            # Evaluate current particle
            fitness = compute_c5(h)
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best_fitness[i] = fitness
                personal_best[i] = h.copy()
            
            # Update global best
            if fitness < global_best_fitness:
                global_best_fitness = fitness
                global_best = h.copy()
            
            # Update velocity and position
            r1, r2 = np.random.random(2)
            vel = (w * velocities[i] + 
                   c1 * r1 * (personal_best[i] - h) + 
                   c2 * r2 * (global_best - h))
            
            # Clip velocity
            vel = np.clip(vel, -max_velocity, max_velocity)
            
            # Update position
            new_h = h + vel
            
            # Ensure feasibility
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            # Update swarm and velocity
            swarm[i] = new_h
            velocities[i] = vel
            
            # Early exit if target achieved
            if global_best_fitness <= 0.3807:
                break
        
        if global_best_fitness <= 0.3807:
            break
    
    # Return best solution found
    return (global_best.tolist(), global_best_fitness, n_points)
