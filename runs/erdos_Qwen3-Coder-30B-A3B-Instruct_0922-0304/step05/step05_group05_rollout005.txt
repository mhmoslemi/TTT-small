import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: step function with random perturbation
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # PSO parameters
    n_particles = 20
    max_iter = 1000
    w_max, w_min = 0.8, 0.1
    c1, c2 = 2.0, 2.0
    v_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    # Create initial population
    particles.append(h.copy())
    velocities.append(np.zeros(n_points))
    p_best.append(h.copy())
    p_best_vals.append(compute_c5(h))
    
    for _ in range(n_particles - 1):
        h_pert = h + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        particles.append(h_pert)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        p_best.append(h_pert.copy())
        p_best_vals.append(compute_c5(h_pert))
    
    # Global best tracking
    best_idx = np.argmin(p_best_vals)
    g_best = p_best[best_idx].copy()
    g_best_val = p_best_vals[best_idx]
    
    best_h = g_best.copy()
    best_c5 = g_best_val
    
    # PSO iterations
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_max - (w_max - w_min) * iter_num / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            vel_cog = c1 * r1 * (p_best[i] - particles[i])
            vel_soc = c2 * r2 * (g_best - particles[i])
            velocities[i] = w * velocities[i] + vel_cog + vel_soc
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Rescale to satisfy constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate new position
            new_val = compute_c5(particles[i])
            
            # Update personal best
            if new_val < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = new_val
            
            # Update global best
            if new_val < g_best_val:
                g_best = particles[i].copy()
                g_best_val = new_val
            
            # Early stopping
            if g_best_val <= 0.3807:
                break
        
        # Update best solution
        if g_best_val < best_c5:
            best_c5 = g_best_val
            best_h = g_best.copy()
    
    # Final validation of best solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
