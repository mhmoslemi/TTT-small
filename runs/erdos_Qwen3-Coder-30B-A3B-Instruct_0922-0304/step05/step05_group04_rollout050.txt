import numpy as np
from scipy.signal import correlate
import time
from cvxpy import Variable, Minimize, Problem, norm, sum as cp_sum, multiply, abs as cp_abs
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Start with binary heuristic to get a starting configuration
    def binary_search_heuristic(h0, max_iter=100):
        h_binary = h0.copy()
        h_binary = (h_binary > 0.5).astype(float)
        h_binary = h_binary * (n_points / 2) / np.sum(h_binary)
        return h_binary
    
    # Convert to continuous and optimize
    def relax_to_continuous(h0, max_iter=50):
        h = h0.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Fast approximate gradient
            g = np.zeros(n_points)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_minus = h.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
                overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
                g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps) * dx

            # Gradient descent step
            step_size = 0.005
            h_new = h - step_size * g
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            
        return h

    # Main optimization loop
    max_iter = 1000
    iteration = 0

    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current C5
        current_c5 = compute_c5(h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

        # Refine
        h = relax_to_continuous(h, max_iter=20)
        
        # Sometimes add a random perturbation to escape local minima
        if iteration % 5 == 0:
            pert = np.random.normal(0, 0.002, n_points)
            h = h + pert
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    # Final computation of best solution
    if best_h is not None:
        final_c5 = compute_c5(best_h)
        return (best_h.tolist(), final_c5, n_points)
    else:
        return (h.tolist(), compute_c5(h), n_points)
