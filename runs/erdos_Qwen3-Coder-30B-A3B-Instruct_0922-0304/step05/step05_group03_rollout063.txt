import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Hyperparameters
    learning_rate = 0.1
    patience = 10
    patience_counter = 0
    last_improvement = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Compute current C5
        current_c5 = evaluate_c5(h, dx)
        
        # Track best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
            last_improvement = iteration
        else:
            patience_counter += 1
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
        # Adaptive learning rate
        if iteration - last_improvement > 50:
            learning_rate *= 0.5
            last_improvement = iteration
            
        if patience_counter > patience:
            # Reinitialize with a better guess
            h = np.ones(n_points)
            h[n_points//4:3*n_points//4] = 0.5
            h = normalize_h(h, target_sum)
            patience_counter = 0
            learning_rate = 0.1
        
        # Gradient-based update
        # Compute overlap to find max lag
        overlap = correlate(h, 1 - h, mode="full")
        max_lag_idx = np.argmax(overlap)
        max_lag = max_lag_idx - (len(h) - 1)
        
        # Compute gradients for the correlation at max lag
        grad = np.zeros_like(h)
        # Approximate gradient using finite differences for the relevant correlation
        for i in range(len(h)):
            # Compute partial derivative of overlap w.r.t h[i]
            h_plus = h.copy()
            h_plus[i] += 1e-6
            h_plus = normalize_h(h_plus, target_sum)
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            
            h_minus = h.copy()
            h_minus[i] -= 1e-6
            h_minus = normalize_h(h_minus, target_sum)
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            grad[i] = (overlap_plus[max_lag_idx] - overlap_minus[max_lag_idx]) / (2e-6)
        
        # Update with gradient step
        h += learning_rate * grad
        h = normalize_h(h, target_sum)
        
        # Coordinate-wise local search
        if iteration % 3 == 0:
            h_backup = h.copy()
            for i in range(n_points):
                # Try small perturbations
                perturbation = np.random.uniform(-0.01, 0.01)
                h[i] += perturbation

                new_c5 = evaluate_c5(h, dx)
                if new_c5 < current_c5:

                else:
                    h = h_backup.copy()
    
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
