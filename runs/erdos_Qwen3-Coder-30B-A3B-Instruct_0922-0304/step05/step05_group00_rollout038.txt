import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    iteration = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute overlaps
        overlap = correlate(best_individual, 1 - best_individual, mode="full")
        overlap_values = overlap * dx

        # Select top contributing lags (maximize overlap)
        top_lags = np.argsort(np.abs(overlap_values))[-10:]
        
        # Local search on top lags
        def objective(h_local):
            overlap_local = correlate(h_local, 1 - h_local, mode="full")
            return np.max(overlap_local) * dx

        # Define bounds and constraint
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h_local):
            return np.sum(h_local) - target_sum
        
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Use SLSQP with constraints
        result = minimize(objective, best_individual.copy(), method='SLSQP', 
                          bounds=bounds, constraints=constraint, options={'maxiter': 50,
                          'ftol': 1e-6, 'gtol': 1e-6})
        
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = normalize_h(refined_h, target_sum)
            
            new_c5 = compute_c5(refined_h, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_individual = refined_h.copy()
                
        if best_c5 <= 0.3807:
            break
            
        # Perturb
        perturbation = np.random.normal(0, 0.01, n_points)
        perturbed_h = best_individual + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = normalize_h(perturbed_h, target_sum)

        new_c5 = compute_c5(perturbed_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_individual = perturbed_h.copy()
            
        iteration += 1
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
