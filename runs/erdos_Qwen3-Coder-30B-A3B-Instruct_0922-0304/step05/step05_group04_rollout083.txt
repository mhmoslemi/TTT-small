import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 10000
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
    
    # Initialize rest of the swarm
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
    
    # Initialize velocities
    for i in range(swarm_size):
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Initialize best positions
    particle_best = [p.copy() for p in particles]
    particle_best_scores = [float('inf')] * swarm_size
    global_best = particles[0].copy()
    global_best_score = float('inf')
    
    # PSO parameters
    w_max = 0.9
    w_min = 0.4
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    def compute_C5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Update inertia weight
        w = w_max - (w_max - w_min) * (time.time() - start_time) / budget_s
        
        for i in range(swarm_size):
            h = particles[i]
            
            # Compute current score
            score = compute_C5(h)
            
            # Update particle best
            if score < particle_best_scores[i]:
                particle_best_scores[i] = score
                particle_best[i] = h.copy()
                
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best = h.copy()
                
            # Early exit if target achieved
            if global_best_score <= 0.3807:
                break
                
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            v = (w * velocities[i] + 
                 c1 * r1 * (particle_best[i] - h) + 
                 c2 * r2 * (global_best - h))
            
            # Limit velocity
            v = np.clip(v, -v_max, v_max)
            
            # Update position
            h_new = h + v
            
            # Ensure bounds and normalization
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Update particle and velocity
            particles[i] = h_new
            velocities[i] = v
            
        # Early exit check
        if global_best_score <= 0.3807:
            break
    
    # Final evaluation
    final_score = compute_C5(global_best)
    
    return (global_best.tolist(), final_score, n_points)
