import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import Problem, Minimize, Variable, norm, sum as cp_sum

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have sum equal to target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

def project_to_feasible(h, n_points):
    """Project h to satisfy sum(h) = n_points / 2 and 0 <= h <= 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        scale_factor = (n_points / 2) / current_sum
        h = h * scale_factor
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Generate a default starting point
        h = np.full(n_points, 0.5)
        h = project_to_feasible(h, n_points)
    
    # Best solution tracking
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute cross-correlation values
        c_values = []
        for k in range(2 * n_points - 1):
            # Compute convolution for shift k
            if k < n_points:
                # Overlap of h[0:n-k] with (1-h[k:n])
                h1 = h[:n_points-k]
                h2 = 1 - h[k:]
            else:
                # Overlap of h[k-n+1:n] with (1-h[0:n-k+n-1])
                h1 = h[k-n+1:]
                h2 = 1 - h[:n_points-k+n-1]
            c = np.sum(h1 * h2) * dx
            c_values.append(c)
        c_values = np.array(c_values)
        max_c = np.max(c_values)
        
        # If this is a new best, update
        if max_c < best_c5:
            best_c5 = max_c
            best_h = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

        # Use a convex optimization approach
        # Minimize max overlap subject to constraints
        h_var = Variable(n_points)
        
        # BBB: Add a safety net to prevent negative variance
        constraint_sum = cp_sum(h_var) == target_sum
        constraints = [h_var >= 0, h_var <= 1, constraint_sum]
        
        # Create overlapping term constraints (simplified approach)
        # Try to tighten the convex relaxation
        # For each k, compute h ∗ (1-h) where h has fixed size
        
        # Find upper bound C using CVXOPT
        C = Variable()
        # For all valid shifts k (total n_points shifts)
        for k in range(n_points):
            # Snap to range that avoids all bord error
            start1 = 0
            end1 = n_points - k
            start2 = k
            end2 = n_points
            
            if end1 <= 0:
                continue
                
            h1 = h_var[start1:end1]
            h2 = 1 - h_var[start2:end2]
            
            # Match length by truncating to minimum
            min_len = min(len(h1), len(h2))
            if min_len > 0:
                c_conv = cp_sum(h1[:min_len] * h2[:min_len]) * dx
                constraints.append(C >= c_conv)
        
        # Formulate and solve the polyhedral relaxation
        try:
            prob = Problem(Minimize(C), constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=100)
            if prob.status == 'optimal':
                h_new = np.array(h_var.value).flatten()
                h_new = project_to_feasible(h_new, n_points)
                h = h_new
        except:
            # Fall back to simple local search with small random perturbations
            h += np.random.normal(0, 0.005, n_points)
            h = project_to_feasible(h, n_points)
        
        # Periodic random perturbation
        if iteration % 50 == 0 and iteration > 0:
            perturbation = np.random.normal(0, 0.01, n_points)
            h += perturbation
            h = project_to_feasible(h, n_points)
        
        iteration += 1
    
    # Final computation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
