import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Define constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Projected gradient descent
    max_iter = 1000
    learning_rate = 0.01
    patience = 0
    patience_limit = 50
    
    # Initialize tracking variables
    prev_c5 = float('inf')
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps and find max
        overlaps = correlate(h, 1 - h, mode="full")
        max_val = overlaps.max()
        max_lags = [i - (n_points - 1) for i, val in enumerate(overlaps) if val == max_val]
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for k in max_lags:
            for j in range(n_points):
                if j + k < n_points:
                    gradient[j] += (1 - h[j + k])
                if j - k >= 0:
                    gradient[j] -= h[j - k]
        gradient /= len(max_lags)
        
        # Update h with gradient descent
        h -= learning_rate * gradient
        
        # Project into feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Update learning rate adaptively
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > patience_limit:
                learning_rate *= 0.9
                patience = 0
                
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Adaptive reduction of learning rate
        if iter_count > 0 and iter_count % 100 == 0:
            learning_rate *= 0.9

    # Apply local refinement with SLSQP
    result = minimize(
        compute_c5,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 30, 'ftol': 1e-8}
    )
    if result.success:
        final_h = result.x
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (n_points / 2) / np.sum(final_h)
        final_c5 = compute_c5(final_h)
        if final_c5 < best_c5:
            best_h = final_h
            best_c5 = final_c5

    return (best_h.tolist(), best_c5, n_points)
