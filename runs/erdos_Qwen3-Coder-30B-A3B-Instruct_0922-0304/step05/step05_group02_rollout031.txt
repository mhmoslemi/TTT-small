import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from `initial_h_values` if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        # Symmetric initialization
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 + 0.1 * ((-1)**i)
        h = project_h(h, target_sum)
    
    # Optimization parameters
    momentum = 0.95
    learning_rate = 0.05
    max_iter = 1000
    patience = 50
    best_c5 = float('inf')
    best_h = h.copy()
    prev_c5 = float('inf')
    no_improvement_count = 0
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all correlations
        overlap_full = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap_full)
        max_overlap = overlap_full[max_k]
        c5 = max_overlap * dx
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= patience:
            break
            
        # Gradient estimation: we compute gradient for the direction of maximum overlap
        # The gradient at point i for the max overlap is 1 - h[i - k] if that's where the max occurs
        # We approximate gradient direction at active k
        if max_k < n_points:
            # The overlapping elements are from the second part
            k = max_k
            grad = np.zeros(n_points)
            for i in range(n_points):
                if i - k >= 0:
                    grad[i] = 1 - h[i - k]
                else:
                    grad[i] = 0
        else:
            # Fall back to simple finite difference-based gradient estimation
            grad = np.zeros_like(h)
            for i in range(n_points):
                if i + 1 < n_points:
                    grad[i] = 0.5 * (h[i+1] - h[i-1]) if i > 0 else h[i+1] - h[i]
                else:
                    grad[i] = 0.5 * (h[i-1] - h[i-2]) if i > 1 else h[i] - h[i-1]
        
        # Update using momentum
        h = h + learning_rate * grad
        h = project_h(h, target_sum)
        
        # Adaptive learning rate
        if iteration % 10 == 0:
            if abs(c5 - prev_c5) < 1e-8:
                learning_rate *= 0.9
            prev_c5 = c5
            
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
