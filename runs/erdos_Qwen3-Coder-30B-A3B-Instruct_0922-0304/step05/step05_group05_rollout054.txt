import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 50000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with noise
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Simulated Annealing
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Initial temperature based on problem scale
    T0 = current_c5 * 0.1
    T = T0
    cooling_rate = 0.9995
    min_T = 1e-8
    
    # Annealing parameters
    iter_per_T = 20
    
    # Main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb the current solution
        h_new = current_h + np.random.normal(0, T * 0.1, n_points)
        h_new = np.clip(h_new, 0, 1)
        # Rescale to meet constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = compute_c5(h_new)
        
        # Accept or reject
        if c5_new < current_c5:
            current_h = h_new
            current_c5 = c5_new
        else:
            # Accept with probability
            delta = c5_new - current_c5
            if np.random.rand() < np.exp(-delta / T):
                current_h = h_new
                current_c5 = c5_new
                
        # Update best
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Cooling schedule
        if iteration % iter_per_T == 0:
            T = max(T * cooling_rate, min_T)
    
    return (best_h.tolist(), best_c5, n_points)
