import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, k_max, dx, n_points):
    """Compute the gradient of the overlap at lag k_max with respect to h."""
    grad = np.zeros(n_points)
    
    # For correlation h * (1-h) at lag k_max
    for i in range(n_points):
        if i + k_max < n_points:
            # Contribution to overlap at lag k_max
            grad[i] = 1 - h[i + k_max]
    
    return grad * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    # Hyperparameters
    alpha = 0.1
    patience = 50
    patience_counter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5 and overlap
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter >= patience:
                alpha *= 0.5
                patience_counter = 0
        
        # If we've reached our target, early exit
        if best_c5 <= 0.3807:
            break
            
        # Compute all overlaps to find max
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        
        # Compute gradient
        gradient = compute_gradient(h, k_max, dx, n_points)
        
        # Update h
        h = h - alpha * gradient
        
        # Ensure feasibility
        h = normalize_h(h, target_sum)
        
        # Occasionally add small random perturbations
        if iteration % 100 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h = h + noise
            h = normalize_h(h, target_sum)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
