```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_and_normalize(h, target_sum):
    """Project h to [0,1] and normalize it."""
    h = np.clip(h, 0, 1)
    return normalize_h(h, target_sum)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Simple construction as fallback
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)
    
    # Track best solution
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Subproblem formulation parameters
    lag_ratio = 0.05  # Top 5% lags
    
    iteration = 0
    max_iter = 500  # limit iterations for time budget
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlaps
        overlap = correlate(h, 1 - h, mode="full")
        # We need to consider only the relevant part of overlap
        # The full overlap is of length 2*n_points - 1 but we care about the lags
        # that cause the maximum conflict
        
        # Get top indices contributing to max overlap
        top_lags = np.argsort(overlap)[-int(n_points * lag_ratio):]
        
        # Create weight matrix based on importance of lags
        weights = np.zeros(2 * n_points - 1)
        weights[top_lags] = 1.0
        
        # Objective function
        def objective(h):
            # Compute overlap at the weighted top_lags
            result = correlate(h, 1 - h, mode="full") * dx
            # Consider only top lags for gradient estimation
            total = np.sum(result * weights)
            return total
        
        # Constraint for sum
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        # Define bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Optimize using constrained SQP
        try:
            result = minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraint,
                options={'maxiter': 200}
            )
            if result.success:
                h_new = result.x
                # Keep within bounds and re-normalize
                h_new = project_and_normalize(h_new, target_sum)
                # Evaluate new solution
                new_c5 = compute_c5(h_new, dx)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
                
                if best_c5 <= 0.3807:
                    break
                    
                # Update h for next iteration
                h = h_new
        except Exception:
            # Fallback to simple projection
            h = project_and_normalize(h, target_sum)
        
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
```
