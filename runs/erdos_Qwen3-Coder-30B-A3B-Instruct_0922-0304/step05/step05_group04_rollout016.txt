import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    mutation_rate = 0.1
    tournament_size = 10
    max_generations = 1000
    
    # Initialize population
    def create_individual():
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values)]
        population.extend([create_individual() for _ in range(population_size - 1)])
    else:
        population = [create_individual() for _ in range(population_size)]
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_h = None
    best_c5 = float('inf')
    
    # Main genetic algorithm loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = compute_c5(h)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection using tournament
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size)
            winner_idx = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < population_size else selected[0]
            
            # Uniform crossover
            mask = np.random.rand(n_points) < 0.5
            child1 = np.where(mask, parent1, parent2)
            child2 = np.where(mask, parent2, parent1)
            
            # Mutation
            if np.random.rand() < mutation_rate:
                for j in range(n_points):
                    if np.random.rand() < 0.1:
                        child1[j] += np.random.choice([-0.1, 0.1])
                    if np.random.rand() < 0.2:
                        child1[j] = np.random.choice([0, 1])
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points / 2) / np.sum(child1)
                    
            if np.random.rand() < mutation_rate:
                for j in range(n_points):
                    if np.random.rand() < 0.1:
                        child2[j] += np.random.choice([-0.1, 0.1])
                    if np.random.rand() < 0.2:
                        child2[j] = np.random.choice([0, 1])
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points / 2) / np.sum(child2)
                
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
                
        population = new_population
    
    return (best_h.tolist(), best_c5, n_points)
