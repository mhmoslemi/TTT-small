import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def mutate(h, mutation_std, n_points):
    """Mutate h with Gaussian noise and ensure constraints"""
    mutated = h + np.random.normal(0, mutation_std, h.shape)
    mutated = np.clip(mutated, 0, 1)
    sum_h = np.sum(mutated)
    mutated = mutated * (n_points / 2) / sum_h
    return mutated

def evaluate_fitness(population, dx):
    """Evaluate fitness for all individuals in population"""
    return [compute_c5(individual, dx) for individual in population]

def select_elite(population, fitnesses, elite_fraction=0.25):
    """Select elite individuals based on fitness"""
    indices = np.argsort(fitnesses)
    elite_count = max(1, int(len(population) * elite_fraction))
    return [population[i] for i in indices[:elite_count]]

def update_best(population, fitnesses, best_h, best_c5):
    """Update the best solution found so far"""
    min_index = np.argmin(fitnesses)
    if fitnesses[min_index] < best_c5:
        best_c5 = fitnesses[min_index]
        best_h = population[min_index].copy()
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 20
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        mutated = mutate(h, 0.01, n_points)
        population.append(mutated)
        
    # Evolutionary algorithm
    max_generations = 1000
    mutation_strength = 0.01
    best_c5 = float('inf')
    best_h = None
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for all individuals
        fitnesses = evaluate_fitness(population, dx)
        
        # Update best solution
        best_h, best_c5 = update_best(population, fitnesses, best_h, best_c5)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection
        elite = select_elite(population, fitnesses)
        
        # Generate new population
        new_population = []
        for individual in elite:
            # Create two offspring for each elite individual
            offspring1 = mutate(individual, mutation_strength, n_points)
            offspring2 = mutate(individual, mutation_strength, n_points)
            new_population.append(offspring1)
            new_population.append(offspring2)
            
        # Keep population size consistent
        population = new_population[:population_size]
        
        # Decrease mutation strength if no improvement for a while
        if generation % 50 == 0 and generation > 0:
            mutation_strength *= 0.9

    # Final evaluation
    if best_h is None:
        # Fallback to evaluating initial population
        fitnesses = evaluate_fitness(population, dx)
        min_index = np.argmin(fitnesses)
        best_h = population[min_index]
        best_c5 = fitnesses[min_index]
        
    return (best_h.tolist(), best_c5, n_points)
