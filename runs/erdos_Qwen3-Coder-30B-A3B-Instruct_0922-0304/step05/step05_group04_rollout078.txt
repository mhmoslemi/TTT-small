import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization with alternating blocks
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 if i % 2 == 0 else 0
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current C5
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Try local search with perturbations
        # Random small perturbations
        perturbation = np.random.normal(0, 0.005, n_points)
        h_new = h + perturbation
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Track if better solution found
        new_c5 = compute_c5(h_new)
        if new_c5 < current_c5:
            h = h_new.copy()
        else:
            # Try alternating pattern approach
            h_alt = np.zeros(n_points)
            for i in range(n_points):
                h_alt[i] = 1 if i % 2 == 0 else 0
            h_alt = h_alt * (n_points / 2) / np.sum(h_alt)
            
            alt_c5 = compute_c5(h_alt)
            if alt_c5 < current_c5:
                h = h_alt.copy()
            
        # Occasionally try gradient-like adjustments
        if iteration % 50 == 0:
            # Gradient approximation
            g = np.zeros(n_points)
            eps_val = 1e-4
            for i in range(n_points):
                h_plus = h.copy()
                h_minus = h.copy()
                h_plus[i] += eps_val
                h_minus[i] -= eps_val
                # Clamp values
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                # Renormalize
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
                overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
                g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps_val) * dx
                
            # Update using gradient descent with step size
            step_size = 0.005
            h = h - step_size * g
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
