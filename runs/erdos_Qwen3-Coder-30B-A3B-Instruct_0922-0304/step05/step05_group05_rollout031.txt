import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # PSO parameters
    n_particles = 20
    max_iter = 1000
    w_start = 0.9
    w_end = 0.3
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize particles
    particles = [h.copy()]
    for _ in range(n_particles - 1):
        h_pert = h + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        particles.append(h_pert)
    
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Track personal bests
    p_best = [particle.copy() for particle in particles]
    p_best_values = [compute_c5(particle) for particle in particles]
    
    # Initialize global best
    global_best_idx = np.argmin(p_best_values)
    g_best = p_best[global_best_idx].copy()
    g_best_value = p_best_values[global_best_idx]
    
    best_h = g_best.copy()
    best_c5 = g_best_value
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Rescale to correct integral
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            c5 = compute_c5(particles[i])
            
            # Update personal best
            if c5 < p_best_values[i]:
                p_best[i] = particles[i].copy()
                p_best_values[i] = c5
                
                # Update global best
                if c5 < g_best_value:
                    g_best = particles[i].copy()
                    g_best_value = c5
                    if g_best_value <= 0.3807:
                        break
                        
        # Track best found so far
        if g_best_value < best_c5:
            best_c5 = g_best_value
            best_h = g_best.copy()
    
    return (best_h.tolist(), best_c5, n_points)
