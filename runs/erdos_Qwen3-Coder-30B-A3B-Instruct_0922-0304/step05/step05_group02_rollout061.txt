import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have sum equal to target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        return h * (target_sum / current_sum)
    else:
        return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum"""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def compute_gradient(h, dx, k_max):
    """Compute gradient with respect to the maximum overlap shift."""
    # Compute correlation values
    corr = correlate(h, 1 - h, mode="full")
    max_k_idx = np.argmax(corr)
    # Get the gradient for this specific shift
    # When computing the derivative of h(k) * (1 - h(k + shift))
    # We need to consider that both parts depend on h, so we need to compute both directions
    # The derivative of the correlation at shift k_max with respect to h[i]
    # is 1 - h[i + k_max] if i + k_max < len(h) and -h[i - k_max] if i - k_max >= 0
    # We simplify this to compute approximately the gradient of correlation at k_max
    # Create a template for gradient
    grad = np.zeros_like(h)
    # We assume this is a linear approximation over small regions
    # Approximate gradient of the correlation with respect to h[shift]
    # and extend it to all positions using local influence
    shift = k_max - (len(h) - 1)  # Convert to positive shift (assuming full mode)
    for i in range(len(h)):
        if i + shift < len(h) and i + shift >= 0:
            grad[i] += -1.0
        if i - shift >= 0 and i - shift < len(h):
            grad[i] += 1.0
    return grad * dx

def line_search(h, grad, learning_rate, dx):
    """Perform line search to find an appropriate step."""
    step = learning_rate * grad
    trial_h = h - step
    trial_h = project_to_feasible(trial_h, len(h) / 2)
    new_c5 = compute_c5(trial_h, dx)
    return trial_h, new_c5

        h[n_points//2:] = 0
        h = normalize_h(h, n_points / 2)
        # Add small noise
        noise = np.random.normal(0, 0.02, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points / 2)
    
    # Optimizer parameters
    momentum_decay = 0.9
    learning_rate = 0.1
    tolerance = 1e-6
    max_iter = 10000
    
    # Initialize momentum
    momentum = np.zeros_like(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early exit condition
        if best_c5 < 0.3807:
            break
            
        # Compute correlations and find maximum overlap
        corr = correlate(h, 1 - h, mode="full")
        max_k_idx = np.argmax(corr)
        k_max = max_k_idx - (len(h) - 1)
        
        # Compute gradient (approximate)
        gradient = compute_gradient(h, dx, k_max)
        
        # Apply momentum to gradient
        momentum = momentum_decay * momentum + (1 - momentum_decay) * gradient

        step_size = learning_rate

        h = trial_h

    final_c5 = compute_c5(best_h, dx)
