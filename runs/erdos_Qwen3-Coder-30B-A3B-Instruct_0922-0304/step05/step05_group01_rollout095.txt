import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    swarm_size = 20
    max_iter = 1000
    
    # Initialize swarm
    def initialize_particle():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
        else:
            h = np.random.random(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Initialize swarm particles
    swarm = [initialize_particle() for _ in range(swarm_size)]
    swarm_velocities = [np.random.random(n_points) * 0.1 for _ in range(swarm_size)]
    particle_best = [p.copy() for p in swarm]
    particle_best_fitness = [float('inf')] * swarm_size
    global_best = None
    global_best_fitness = float('inf')
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # PSO parameters
    inertia_w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_vel = 0.1
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        for i in range(swarm_size):
            particle = swarm[i]
            fitness = compute_c5(particle)
            
            # Update particle best
            if fitness < particle_best_fitness[i]:
                particle_best_fitness[i] = fitness
                particle_best[i] = particle.copy()
            
            # Update global best
            if fitness < global_best_fitness:
                global_best_fitness = fitness
                global_best = particle.copy()
        
        # Early termination if target achieved
        if global_best_fitness <= 0.3807:
            break
            
        # Update particles
        inertia_w = max(0.4, inertia_w - 0.0001)
        
        for i in range(swarm_size):
            r1, r2 = np.random.random(2)
            particle = swarm[i]
            vel = swarm_velocities[i]
            
            # Update velocity
            vel = (inertia_w * vel + 
                   c1 * r1 * (particle_best[i] - particle) + 
                   c2 * r2 * (global_best - particle))
            
            # Clip velocity
            vel = np.clip(vel, -max_vel, max_vel)
            
            # Update position
            particle += vel
            particle = np.clip(particle, 0, 1)
            
            # Normalize to sum to n_points/2
            particle = particle * (n_points / 2) / np.sum(particle)
            
            # Update velocity
            swarm_velocities[i] = vel
            
            # Keep feasible
            swarm[i] = particle
    
    # Final evaluation of global best
    final_c5 = compute_c5(global_best)
    
    return (global_best.tolist(), final_c5, n_points)
