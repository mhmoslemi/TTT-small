import numpy as np
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    generations = 1000
    mutation_rate = 0.1
    
    # Initialize population
    def initialize_population():
        if 'initial_h_values' in globals():
            pop = [np.array(initial_h_values)]
        else:
            pop = []
        
        # Fill up with random individuals
        while len(pop) < population_size:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            pop.append(h)
        
        return pop
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def is_feasible(h):
        return (abs(np.sum(h) - n_points / 2) < 1e-6 and 
                np.all(h >= 0) and np.all(h <= 1))
    
    def repair(h):
        # Ensure sum constraint
        h = h * (n_points / 2) / np.sum(h)
        # Clamp values
        h = np.clip(h, 0, 1)
        return h
    
    def tournament_selection(population, fitnesses, k=10):
        tournament_indices = random.sample(range(len(population)), k)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
        return population[winner_idx]
    
    def uniform_crossover(parent1, parent2):
        mask = np.random.rand(n_points) < 0.5
        child = np.where(mask, parent1, parent2)
        return child
    
    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if random.random() < mutation_rate:
                if random.random() < 0.5:
                    h[i] += random.uniform(-0.1, 0.1)
                else:
                    h[i] = random.choice([0.0, 1.0])
        return h
    
    # Main algorithm
    population = initialize_population()
    best_h = None
    best_c5 = float('inf')
    
    for gen in range(generations):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = [compute_c5(ind) for ind in population]
        
        # Track best solution
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Selection
        new_population = []
        
        # Keep top 50% as elites
        elite_indices = np.argsort(fitnesses)[:population_size//2]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            p1 = tournament_selection(population, fitnesses)
            p2 = tournament_selection(population, fitnesses)
            
            child = uniform_crossover(p1, p2)
            child = mutate(child)
            child = repair(child)
            
            new_population.append(child)
        
        population = new_population
    
    # Final validation and recomputation
    if best_h is not None:
        final_c5 = compute_c5(best_h)
    else:
        # Fallback if no valid solution found
        final_c5 = float('inf')
    
    return (best_h.tolist(), final_c5, n_points)
