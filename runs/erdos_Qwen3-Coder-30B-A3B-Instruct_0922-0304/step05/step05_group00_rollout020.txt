import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def process_solution(h, target_sum):
    """Process and repair solution to ensure feasibility."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Gradient descent style approach with local refinement
    max_iter = 5000
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        if iteration % 50 == 0:
            # Run local optimization on a small subset
            h_local = h.copy()
            indices = np.random.choice(n_points, size=max(5, n_points//10), replace=False)
            def objective(h_sub):
                h_full = h.copy()
                h_full[indices] = h_sub
                h_full = process_solution(h_full, target_sum)
                return compute_c5(h_full, dx)
                
            bounds = [(0, 1) for _ in indices]
            h_sub = h_local[indices]
            if len(indices) > 0:
                from scipy.optimize import minimize
                try:
                    result = minimize(objective, h_sub, bounds=bounds, method='L-BFGS-B', options={'maxiter': 50})
                    if result.success:
                        h_local[indices] = result.x
                        h_local = process_solution(h_local, target_sum)
                        c5_new = compute_c5(h_local, dx)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_local.copy()
                        h = h_local
                        if best_c5 <= 0.3807:
                            break
                except:
                    pass
                        
        # Random mutation with repair
        h_new = h.copy()
        for i in range(n_points):
            if np.random.rand() < 0.1:
                h_new[i] += np.random.normal(0, 0.02)
        
        h_new = process_solution(h_new, target_sum)
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            if best_c5 <= 0.3807:
                break
        h = h_new
        
        iteration += 1
        
    return (best_h.tolist(), best_c5, n_points)
