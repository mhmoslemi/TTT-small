from scipy.signal import correlate
import time

        ones_indices = np.where(h == 1)[0]
        if len(ones_indices) > n_points // 2:
            # Remove extras
            extra_indices = np.random.choice(ones_indices, len(ones_indices) - n_points // 2, replace=False)
            h[extra_indices] = 0
        elif len(ones_indices) < n_points // 2:
            # Add missing
            zeros_indices = np.where(h == 0)[0]
            missing_indices = np.random.choice(zeros_indices, n_points // 2 - len(ones_indices), replace=False)
            h[missing_indices] = 1
        # Normalize to sum to n_points/2
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def binary_search(h, max_iter=10000):
        # Simulated annealing for binary function optimization
        best_c5 = float('inf')
        best_h = h.copy()
        
        T = 1.0
        alpha = 0.99
        min_T = 1e-4
        
        iteration = 0
        
        while T > min_T and iteration < max_iter:
            iteration += 1
            
            # Create neighbor by swapping two random positions
            idx1, idx2 = np.random.choice(n_points, 2, replace=False)
            h_new = h.copy()
            h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
            
            # Compute overlap for new solution
            overlap = correlate(h_new, 1 - h_new, mode="full")
            new_c5 = np.max(overlap) * dx
            
            # Accept or reject
            if new_c5 < best_c5 or np.random.rand() < np.exp(-(new_c5 - best_c5) / T):
                h = h_new
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            
            # Cool down
            T *= alpha
            
            # Early exit if target is achieved
            if best_c5 <= 0.3807:
                break
                
        return best_h, best_c5
    
    def continuous_relaxation(h):
        # Use CVXPY for continuous optimization
        h_var = cp.Variable(n_points)
        
        # Maximize overlap as a proxy for minimization
        # This is a bit indirect via constraint satisfaction
        
        # Objective is to minimize maximum overlap
        # We'll make this constraint-based using a relaxation approach
        
        # Key insight: we're trying to minimize max_k integral, so
        # we'll do slice-wise relaxations as needed or directly use
        # gradient-based methods for closest local minimum
        
        # Use a local gradient update approach due to budget constraints
        
        max_iter = 5000
        step_size = 0.005
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute gradient-like update via finite differences
            current_c5 = compute_c5(h)
            
            # Small perturbations to estimate gradient
            grad = np.zeros_like(h)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = h.copy()
                h_minus = h.copy()
                
                h_plus[j] = min(1.0, h_plus[j] + eps)
                h_minus[j] = max(0.0, h_minus[j] - eps)
                
                # Renormalize both
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                # Compute finite difference of overlap
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
                overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                grad[j] = (np.max(overlap_plus) - np.max(overlap_minus)) / (2 * eps) * dx
            
            # Update step
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)

            h = h_new
            
        return h
    
    # Phase 1: Binary search

    # Phase 2: Continuous refinement

    final_c5 = compute_c5(h_refined)
    
    # Return best solution seen
    if c5_binary < final_c5:
        result_h = h_binary
        result_c5 = c5_binary
    else:
        result_h = h_refined
        result_c5 = final_c5
    
    return (result_h.tolist(), result_c5, n_points)
