import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Ensure h sums to target_sum and is in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    # Use local optimization on best solution if still within budget
    def compute_c5(h):
        return evaluate_c5(h, dx)
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Try gradient-based local search to refine the solution
    from scipy.optimize import minimize
    
    # Constraint function for sum
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    # Bounds for the optimization
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # SLSQP optimization with randomness
    if time.time() - start_time < budget_s - 1:
        # Perturb the initial solution
        h_perturbed = h + np.random.normal(0, 0.01, n_points)
        h_perturbed = normalize_h(h_perturbed, target_sum)
        
        # Optimize using SLSQP
        result = minimize(
            compute_c5,
            h_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-9}
        )
        
        if result.success:
            h_new = result.x
            h_new = normalize_h(h_new, target_sum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    # If we still have time, run a few more local searches with different restarts
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 1.0:
        num_restarts = min(5, int(remaining_time / 0.5))
        for _ in range(num_restarts):
            if time.time() - start_time > budget_s - 0.5:
                break
            # Random restart
            h_restart = np.random.rand(n_points)
            h_restart = normalize_h(h_restart, target_sum)
            
            result = minimize(
                compute_c5,
                h_restart,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 30, 'ftol': 1e-8}
            )
            
            if result.success:
                h_new = result.x
                h_new = normalize_h(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
    
    # Final check
    best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
