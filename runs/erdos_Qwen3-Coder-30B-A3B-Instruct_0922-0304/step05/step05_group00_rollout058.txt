        # and form Jacobian for products h[i]*h[i+k]):
        # g = h[i] + h[i+k] - 2 * h[i]*h[i+k] 
        # of course we linearize jacobian at current h...
        # matrix indices slightly messy
        # But do direct small scale LP, because:
        # 1. We have matrix LPing for sum, zero-one bounds.  
        # 2. Then handle term of overlap contribution to a moderately k and then maximize over k.
        # But behave serverasking structure: 
        # Given fixed h, enumerate top terms in overlap, and enforce constraints individually
        # Use Lagrange multipliers.  
        
        # Clear New:
        # So approach formal:
        # 主要过程:
        # 1. Gain current h  
        # 2. Compute correlation (note miss computation before)
        # 3. Identify which k terms are important  
        # 4. Linearize constraint for gradient of extent: 
        #     h[i] + h[i+k] - h[i]*h[i+k]  
        # For LP, use easy solvers:
        # The key strength of this is dual formulation.
        # F = ∑ (h[i] + h[i+k] - h[i]*h[i+k]) - C5 
        # let F[i,k] the terms.  
        # Over all such occur, consider Lagrangian:  
        # L = ∑ F[i,k] - λ (C5)  
        # Set ∂L/∂C5 = 1 - λ  
        # So λ = 1    for strictness = 1 
        # That just implies one constraint: W ( ∑ F[i,k] <= C5 )
        # Since it’s equal across k, not less.  
        
        # That's done with equality!  
        # so choose a subset of full terms.
        # So consider max_k with respect to h
        
        # Direct tutorial approach with full correlation now:
        
        # Approach: discrete maximum; à la gradient-free search but not architecture based  
        # For simple update use:
        overlap = correlate(h, 1-h, mode='full')
        top_k_indices = np.argsort(overlap)[-n_points//5:]
        top_k_indices = np.flip(top_k_indices)
        
        # Try solve single LP now:
        h_vars = cp.Variable(n_points)
        c5 = cp.Variable()
        
        # Add constraints for each top max
        # e.g., each contribution of <h_i + h_{i+k} - 2 h_i h_{i+k}>
        linear_constraints = []
        for k in top_k_indices[:max(1, n_points//5)]: # take top lags
            # sum_{i = 0 to n-k-1} [h[i] + h[i+k] - 2 h[i] h[i+k]] <= C5
            sum_term = cp.sum(h_vars) + cp.sum(h_vars[k:]) + cp.sum(h_vars[:n_points-k]) - 2*cp.sum(cp.multiply(h_vars[:n_points-k], h_vars[k:]))
            
            # Better approach: when we know summing directly is correct aver range since fixed.
            # Since lag=0,... n-1, we compute equally and bounded—we just index from the overlap array
            
            # Use fast version:
            # INDEXING: get each subtuple of overlap, mutual corr,
            # for k in range: valid start indices, and valid pairs
            # lag_k = [i | i+k < n]
            # sum_{i=0}^{n-1-k} h[i] + h[i+k] - 2 h[i]*h[i+k]
            
            if k == 0:
                # Special sum: directly use h[0] - h[0]*h[0]  
                # Overwrite - duplicated sum
                const_pairs = []
                for i in range(n_points):
                    const_pairs.append(h_vars[i] + h_vars[i] - 2 * h_vars[i] * h_vars[i]) 
                linear_constraints.append(cp.sum(const_pairs) <= c5) 
            else:
                # For new worker enforce logic on sub-arrays  
                
                # First, generate subset for valid i:
                left_sum = cp.sum(h_vars[:n_points-k])
                right_sum = cp.sum(h_vars[k:])  
                prod_sum = cp.sum(cp.multiply(h_vars[:n_points-k], h_vars[k:]))  
                linear_constraints.append(left_sum + right_sum - 2*prod_sum <= c5)
        
        # Add extra constraints:
        linear_constraints.append(cp.sum(h_vars) == target_sum)
        linear_constraints.append(h_vars >= 0)
        linear_constraints.append(h_vars <= 1)
        
        objective = cp.Minimize(c5)
        prob = cp.Problem(objective, linear_constraints)
        
        try:
            prob.solve(solver='ECOS', verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                new_h = h_vars.value
                if new_h is not None:
                    new_c5 = c5.value
                    if new_c5 is not None:
                        h = np.clip(new_h, 0, 1)
                        h = h * (target_sum / np.sum(h))  # normalize
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
                        
                        if best_c5 <= 0.3807:
                            break
        except:
            pass

    # Final eval
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), float(final_c5), n_points)
