from scipy.signal import correlate

    dx = 2.0 / n_points
    max_iter = 5000

        h = np.zeros(n_points)

            h[i] = 1 if (i // 10) % 2 == 0 else 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop with adaptive step sizes and early stopping
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Update best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            last_improvement = iter_count
            
        # Stop if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Adaptive step size based on convergence
        step_size = max(1e-4, 1e-3 * np.exp(-iter_count / 1000))
        
        # Gradient approximation with finite differences
        g = np.zeros(n_points)
        eps_val = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps_val
            h_minus[i] -= eps_val
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps_val) * dx

        # Update with gradient descent
        new_h = h - step_size * g
        
        # Project to feasible set (clamp and renormalize)
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        h = new_h
        
        # Occasionally add random noise to escape local optima
        if iter_count % 50 == 0 and np.random.random() < 0.3:
            noise = np.random.normal(0, 0.005, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
