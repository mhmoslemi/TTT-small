import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_swaps = 1000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: 3-region construction
        h = np.ones(n_points)
        mid = n_points // 2
        h[mid:] = 0
        # Add some randomness
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Swap-based local search
    for swap_iter in range(max_swaps):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Select two non-overlapping segments
        segment_size = 5
        start1 = np.random.randint(0, n_points - segment_size)
        start2 = np.random.randint(0, n_points - segment_size)
        
        # Ensure they don't overlap
        while (start2 < start1 + segment_size and start1 < start2 + segment_size):
            start2 = np.random.randint(0, n_points - segment_size)
        
        # Copy and swap
        h_new = h.copy()
        temp_vals = h_new[start1:start1+segment_size].copy()
        h_new[start1:start1+segment_size] = h_new[start2:start2+segment_size]
        h_new[start2:start2+segment_size] = temp_vals
        
        # Rescale to maintain constraint
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        c5_new = compute_c5(h_new)
        
        # Accept with probability
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
        elif np.random.rand() < 0.05:  # 5% chance to accept worse
            h = h_new
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
    # Final polishing with scipy.optimize
    from scipy.optimize import minimize
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    try:
        result = minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            best_c5 = compute_c5(best_h)
    except:
        pass  # If optimization fails, keep previous best
    
    return (best_h.tolist(), best_c5, n_points)
