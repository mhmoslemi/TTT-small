import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Fallback initialization: piecewise constant
        h = np.ones(n_points)
        h[n_points//2:] = 0
    
    # Normalize to correct sum
    h = h * (target_sum / np.sum(h))
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Simulated Annealing with local refinement
    def perturb(h, scale=0.05):
        h_new = h + np.random.normal(0, scale, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        return h_new
    
    def adaptive_perturb(h, temp):
        scale = 0.05 * temp
        return perturb(h, scale)
    
    T0 = 1.0
    T_min = 1e-6
    alpha = 0.95
    max_iter = 1000
    
    T = T0
    current_h = h.copy()
    current_c5 = best_c5
    
    # Local search parameters
    local_iter = 10
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and T > T_min:
        iteration += 1
        # Generate neighbor
        new_h = adaptive_perturb(current_h, T)
        new_c5 = compute_c5(new_h)
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
            current_h = new_h
            current_c5 = new_c5
            
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Local refinement
        for _ in range(local_iter):
            if time.time() - start_time > budget_s - 1:
                break
            new_h = perturb(current_h, 0.01)
            new_c5 = compute_c5(new_h)
            if new_c5 < current_c5:
                current_h = new_h
                current_c5 = new_c5
                
        # Reduce temperature
        T = T * alpha
        
    # Final refinement with optimization
    from scipy.optimize import minimize
    
    def objective(h):
        return compute_c5(h)
        
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    result = minimize(
        objective,
        best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 50, 'ftol': 1e-8}
    )
    
    if result.success:
        best_h = result.x.copy()
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (target_sum / np.sum(best_h))
        best_c5 = compute_c5(best_h)
        
    # Final evaluation to be sure
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
