import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h):
    """Compute C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * (2.0 / len(h))

def mutate(h, mutation_std):
    """Apply Gaussian mutation to h"""
    mutated = h + np.random.normal(0, mutation_std, h.shape)
    mutated = np.clip(mutated, 0, 1)
    sum_h = np.sum(mutated)
    if sum_h > 0:
        mutated = mutated * (len(h) / 2) / sum_h
    return mutated

def evaluate_fitness(population):
    """Evaluate fitness for all individuals in population"""
    return [compute_c5(ind) for ind in population]

def select_elite(population, fitness, elite_frac):
    """Select elite individuals"""
    indices = np.argsort(fitness)
    elite_count = max(1, int(len(population) * elite_frac))
    return [population[i] for i in indices[:elite_count]]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 20
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (n_points / 2) / sum_h
        population.append(h)
    else:
        # Initialize with a basic step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (n_points / 2) / sum_h
        population.append(h)
    
    # Fill population with mutated versions
    for _ in range(population_size - 1):
        if len(population) > 0:
            parent = population[0]
        else:
            parent = np.ones(n_points)
            parent[n_points//2:] = 0
            parent = np.clip(parent, 0, 1)
            sum_h = np.sum(parent)
            if sum_h > 0:
                parent = parent * (n_points / 2) / sum_h
        child = mutate(parent, 0.01)
        population.append(child)
    
    # EA parameters
    elite_fraction = 0.25
    mutation_strength = 0.01
    max_generations = 500
    no_improvement_limit = 50
    no_improvement_count = 0
    
    # Track best solution
    best_c5 = float('inf')
    best_h = population[0].copy()
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = evaluate_fitness(population)
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Adjust mutation strength if no improvement
        if no_improvement_count > no_improvement_limit:
            mutation_strength *= 0.9
            no_improvement_count = 0
            
        # Selection
        elite = select_elite(population, fitness, elite_fraction)
        
        # Create new population through mutation
        new_population = []
        for individual in elite:
            # Create 2 offspring per elite
            offspring1 = mutate(individual, mutation_strength)
            offspring2 = mutate(individual, mutation_strength)
            new_population.append(offspring1)
            new_population.append(offspring2)
            
        # Keep population size
        population = new_population[:population_size]
        
        # Add a random individual occasionally to maintain diversity
        if len(population) < population_size:
            random_h = np.ones(n_points)
            random_h[n_points//2:] = 0
            random_h = np.clip(random_h, 0, 1)
            sum_h = np.sum(random_h)
            if sum_h > 0:
                random_h = random_h * (n_points / 2) / sum_h
            population.append(random_h)
            
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
