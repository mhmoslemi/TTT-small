import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        # Create a feasible initial solution
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = project_h(h, target_sum)
        
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 1000
    patience = 50
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        
        # Compute gradient for the max overlap shift
        # For shift k, the gradient of overlap w.r.t. h[i] is (1 - h[i+k])
        # We compute it for the specific k that achieves max overlap
        # Simplified: use a finite difference approximation or analytical gradient
        # For shifting, use correlation again with the shifted version
        
        grad = np.zeros(n_points)
        # This is the gradient for the max_k shift
        # Each h[i] contributes to -(1-h[i+k]) in overlap with shift k
        # For gradient w.r.t. h[i] we get: -shifted_h[i-k] = -h[i-k] for i-k >=0
        # And also due to (1-h[i]), we get -1 in certain positions
        
        # Compute gradient more carefully:
        # d/dh[i] ∫ h(x) (1 - h(x+k)) dx  
        # = ∫ (1-h(x+k)) δ(x-i) dx  = (1-h[i+k]) 
        
        # We use h[0:n-k] with shift k to compute gradient terms at h[0:n-k]
        # and at the end we use a different approach:
        # optimize by computing the empirically estimated gradient for the max overlap
        # For each i, the contribution to max overlap when i is shifted by k is:
        #   ∇ = (1 - h[i+k]) (so this is -1 * derivative of 1-h[i+k])
        # We want to reduce this.
        
        # Compute gradient for shift where overlap is maximized
        k = max_k - n_points + 1  # correct offset
        if k < 0:
            # t = 1 - h - shifted h
            #   > t[i] = h[i] - h[i+k]
            # gradient of overlap w.r.t. h[i] = (1 - h[i+k])  
            # Actually, overlap at shift k is: sum over i of h[i] * (1 - h[i+k])
            # grad wrt h[i] = (1 - h[i+k]) if i+k < len(h) else 0
            # gradient of h[i] = - (1-h[i+k]) if i+k < n
            grad = np.zeros_like(h)
            if k >= 0:
                grad[k:] -= (1-h[:-k])
            else:
                grad[:k] -= (1-h[-k:])

        # Update h
        h = h - learning_rate * grad
        h = project_h(h, target_sum)
        
        # Evaluate and update best
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
