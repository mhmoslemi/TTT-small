import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        # Create a piecewise constant function
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Minimize C5 using a combination of local search and structured perturbation
    max_iter = 2000
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Random perturbation
        h = base_h + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Local refinement with constraints
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return compute_c5(h, dx)
        
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, 
                          options={'maxiter': 50, 'ftol': 1e-6})
        
        # Update if successful
        if result.success:
            h_refined = result.x
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = normalize_h(h_refined, target_sum)
            current_c5 = compute_c5(h_refined, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_individual = h_refined.copy()
                last_improvement = iteration
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Adjust perturbation based on progress
        if iteration - last_improvement > 50:
            # Increase perturbation if stagnated
            perturbation = min(0.1, 0.02 * (1 + (iteration - last_improvement) / 100))
            h = base_h + np.random.normal(0, perturbation, n_points)
        else:
            h = base_h + np.random.normal(0, 0.02, n_points)
        
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_individual = h.copy()
            last_improvement = iteration
        
        iteration += 1
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
