import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_h(h, target_sum):
    """Project h to feasible region: [0,1] and sum to target_sum."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize using the provided initial_h_values or default
    if 'initial_h_values' in globals():

    else:
        # Simple initial construction, step function with one peak at center
        h = np.zeros(n_points)
        h[n_points//2-5:n_points//2+5] = 1.0
        h = np.clip(h, 0, 1)
        
    h = normalize_h(h, target_sum)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Use sequential quadratic programming for refinement
    max_iter = 1000
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # First get current overlap values for constraint identification
        overlap = correlate(h, 1 - h, mode="full")
        k_indices = np.argsort(overlap)[-int(n_points*0.05):]  # Top 5% lags
        
        # Define objective with selected constraints
        def objective(h):
            return compute_c5(h, dx)
        
        # Subproblem using constrained optimization
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Constraint specific to max overlap at critical lags
        # Just optimize directly with SLSQP, using projection for bounds
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Optimization step
        result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                          constraints=constraint, options={'maxiter': 100})
        
        if result.success:
            h = result.x.copy()
            
        # Project to improve feasibility
        h = project_h(h, target_sum)
        
        # Update best solution
        current_c5 = compute_c5(h, dx)

            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
