import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum to target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, target_sum)
    else:
        # Create a simple initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_to_feasible(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Keep track of best feasible solutions
    best_solutions = [(h.copy(), best_c5)]
    
    max_iter = 1000
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full")
        # Use top 5% of lags contributing to maximum
        num_lags = len(overlap)
        top_lags = np.argsort(overlap)[-max(1, num_lags // 20):][::-1]
        
        # Objective function to minimize using SQP on a restricted subproblem
        def objective(h_sub):
            # For the selected lags, compute approximation of the contribution to C5
            h_full = h.copy()
            # We'll run this on the whole vector but only focus on the computed lags
            # working with sum( h[i] * (1 - h[i+k]) )
            # This is a convex quadratic in h with a lag k structure
            # We don't directly minimize C5 since it's not convex, but we can move in 
            # directions that locally decrease it and keep pushing.
            return compute_c5(h_sub, dx)
            
        # Define bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h_sub):
            return np.sum(h_sub) - target_sum
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # SQP approach on a convex subproblem approximation
        # For this, we optimize the whole h with the constraint
        # but adjust the weights toward directions that are strongly contributing to C5.
        # This mimics a line-search/warm start refinement.
        
        try:
            # Set up result for minimization
            result = minimize(objective, h.copy(), method='SLSQP', bounds=bounds, 
                              constraints=constraint, options={'maxiter': 50})
            
            if result.success:
                h_new = result.x
                h_new = project_to_feasible(h_new, target_sum)
                current_c5 = compute_c5(h_new, dx)
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_new.copy()
                    
                # Store some solutions
                if len(best_solutions) < 10:
                    best_solutions.append((h_new.copy(), current_c5))
                else:
                    # Replace worst if this is better
                    worst_idx = np.argmax([s[1] for s in best_solutions])
                    if current_c5 < best_solutions[worst_idx][1]:
                        best_solutions[worst_idx] = (h_new.copy(), current_c5)
                        
                h = h_new
                
        except:
            pass
            
        # Perturb a bit if needed to escape local minima
        if iter_count % 10 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h = h + noise
            h = project_to_feasible(h, target_sum)
            
        # Early termination
        if best_c5 < 0.3807:
            break
    
    # Ensure last best solution is feasible
    final_h = project_to_feasible(best_h, target_sum)
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
