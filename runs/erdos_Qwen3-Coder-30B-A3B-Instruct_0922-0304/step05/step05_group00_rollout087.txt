import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    """Evaluate objective and check constraints."""
    c5 = compute_c5(h, dx)
    return c5

def mutate_individual(h, noise_level=0.01):
    """Apply small random mutation to an individual."""
    h = h + np.random.normal(0, noise_level, len(h))
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        # Create a simple piecewise constant function with some noise
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        base_h += np.random.normal(0, 0.02, n_points)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Main optimization loop
    max_iter = 15000
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Local search with SLSQP on full problem
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return compute_c5(h, dx)
        
        result = minimize(objective, best_individual.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
        if result.success:
            h_refined = result.x
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = normalize_h(h_refined, target_sum)
            current_c5 = compute_c5(h_refined, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_individual = h_refined.copy()
                last_improvement = iteration
        
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        # Mutate best to explore
        if iteration % 10 == 0:
            mutated = mutate_individual(best_individual, 0.015)
            mutated = normalize_h(mutated, target_sum)
            current_c5 = compute_c5(mutated, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_individual = mutated.copy()
                last_improvement = iteration
        
        iteration += 1
        
        # Break if no improvement in a while
        if iteration - last_improvement > 500:
            break
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
