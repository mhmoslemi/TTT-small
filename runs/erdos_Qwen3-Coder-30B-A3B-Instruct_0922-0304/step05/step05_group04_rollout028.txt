import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: block structure
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    patience = 0
    min_patience = 100
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Current overlap computation
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        if patience > min_patience:
            # Try random perturbations to escape local optima
            h = np.clip(h + np.random.normal(0, 0.02, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            patience = 0
            continue
            
        # Local optimization using gradient approximation 
        # Create a proxy metric to update h
        new_h = h.copy()
        g = np.zeros(n_points)
        
        # Compute gradient approximation (finite differences)
        for i in range(n_points):
            eps_val = 1e-4
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps_val
            h_minus[i] -= eps_val
            # Clamp values
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            # Renormalize
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps_val) * dx

        # Gradient descent step
        step_size = 0.01
        new_h = h - step_size * g
        # Project to feasible set
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Use the step that improves the objective the most
        # But make sure that we still get improvements
        if np.any(new_h != h):
            h = new_h
            
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
