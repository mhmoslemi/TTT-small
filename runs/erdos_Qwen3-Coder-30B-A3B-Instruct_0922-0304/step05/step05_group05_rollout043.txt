import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Swap-based local search parameters
    n_swaps = 1000
    swap_size = 5
    temperature = 0.1
    
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()

    # Main swap loop
    for i in range(n_swaps):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Select two non-overlapping regions
        pos1 = np.random.randint(0, n_points - swap_size)
        pos2 = np.random.randint(0, n_points - swap_size)
        
        # Avoid overlapping regions
        while abs(pos1 - pos2) < swap_size:
            pos2 = np.random.randint(0, n_points - swap_size)
        
        # Create a copy and make swap
        new_h = current_h.copy()
        temp_values = new_h[pos1:pos1+swap_size].copy()
        new_h[pos1:pos1+swap_size] = new_h[pos2:pos2+swap_size]
        new_h[pos2:pos2+swap_size] = temp_values
        
        # Ensure sum constraint is maintained
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        new_h = np.clip(new_h, 0, 1)
        
        # Compute new overlap
        new_c5 = compute_c5(new_h)
        
        # Accept or reject
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
        else:
            # Accept with probability based on temperature
            delta = new_c5 - current_c5
            if np.random.rand() < np.exp(-delta / temperature):
                current_h = new_h
                current_c5 = new_c5
        
        # Update best if better
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break

    # Final polishing with SLSQP if time allows
    if time.time() - start_time < budget_s - 2:
        from scipy.optimize import minimize
        
        def objective(h):
            return compute_c5(h)
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
            
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            result = minimize(
                objective,
                best_h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 30, 'ftol': 1e-8}
            )
            if result.success:
                polished_h = result.x
                polished_h = np.clip(polished_h, 0, 1)
                polished_h = polished_h * (n_points / 2) / np.sum(polished_h)
                polished_c5 = compute_c5(polished_h)
                if polished_c5 < best_c5:
                    best_c5 = polished_c5
                    best_h = polished_h
        except:
            pass

    return (best_h.tolist(), best_c5, n_points)
