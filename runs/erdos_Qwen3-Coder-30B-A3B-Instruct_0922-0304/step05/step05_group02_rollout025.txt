```python
import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: clip to [0,1] and normalize to target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    
    # Best solution tracking
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient-based optimization
    learning_rate = 0.01
    max_iter = 1000
    iter_no_improve = 0
    max_iter_no_improve = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all cross-correlations
        overlap_full = correlate(h, 1 - h, mode="full")
        
        # Find maximum overlap and corresponding shift
        max_overlap = np.max(overlap_full)
        max_shift_idx = np.argmax(overlap_full)
        
        # Compute gradient for this shift
        # For shift k, gradient is (1 - h_{k+offset}) 
        # We only need gradient for active shift
        shift = max_shift_idx - (n_points - 1)
        grad = np.zeros(n_points)
        if shift >= 0:
            grad[shift:] = 1 - h[:-shift]
        else:
            grad[:shift] = 1 - h[-shift:]
        
        # Gradient descent step
        h_new = h - learning_rate * grad
        
        # Project to feasible region
        h_new = project_h(h_new, target_sum)
        
        # Evaluate new solution
        new_c5 = compute_c5(h_new, dx)
        
        # Accept improvement
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            iter_no_improve = 0
        else:
            iter_no_improve += 1
        
        # Early stopping if no improvement
        if iter_no_improve >= max_iter_no_improve:
            break
            
        h = h_new
        
        # Adaptive learning rate
        if iteration > 0 and iteration % 10 == 0:
            learning_rate *= 0.98
    
    return (best_h.tolist(), best_c5, n_points)
```
