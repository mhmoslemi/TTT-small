import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: clip to [0,1] and normalize to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        # Random initialization
        h = np.random.uniform(0, 1, n_points)
        h = project_h(h, target_sum)
    
    # Track best solution
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = 1000
    inner_iter = 10
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_shift = max_idx - (n_points - 1)
        
        # Compute gradient w.r.t. the maximizer shift
        # For shift k, derivative of overlap w.r.t. h[i] is (1 - h[i+k])
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_shift
            if 0 <= j < n_points:
                grad[i] = 1 - h[j]
        
        # Gradient step
        h_new = h - learning_rate * grad
        h_new = project_h(h_new, target_sum)
        
        # Local refinement using gradient descent
        for _ in range(inner_iter):
            overlap = correlate(h_new, 1 - h_new, mode="full")
            max_idx = np.argmax(overlap)
            max_shift = max_idx - (n_points - 1)
            
            grad = np.zeros(n_points)
            for i in range(n_points):
                j = i + max_shift
                if 0 <= j < n_points:
                    grad[i] = 1 - h_new[j]
            
            h_new = h_new - learning_rate * grad
            h_new = project_h(h_new, target_sum)
        
        # Evaluate and update best
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # Update h
        h = h_new
        
        # Early stopping if target achieved
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
