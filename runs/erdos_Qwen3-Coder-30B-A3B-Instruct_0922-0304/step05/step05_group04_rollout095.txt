import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: alternating pattern
        h = np.ones(n_points)
        h[n_points//2:] = 0
        # Random shuffle to avoid symmetry
        indices = np.random.permutation(n_points)
        h = h[indices]
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Phase 1: Binary search with simulated annealing
    def binary_search():
        nonlocal h, best_h, best_c5
        
        # Generate binary vector with exactly n_points/2 ones
        def random_binary():
            idx = np.random.permutation(n_points)[:n_points//2]
            binary = np.zeros(n_points)
            binary[idx] = 1
            return binary
        
        # If no initial, generate random binary
        if 'initial_h_values' not in globals():
            h = random_binary()
            h = h * (n_points / 2) / np.sum(h)
        
        # Simulated annealing
        T = 1.0
        alpha = 0.99
        max_iter = 10000
        
        while time.time() - start_time < budget_s - 5 and T > 1e-4:
            # Make a neighbor
            h_new = h.copy()
            i1, i2 = np.random.choice(n_points, 2, replace=False)
            h_new[i1], h_new[i2] = h_new[i2], h_new[i1]
            
            # Ensure sum constraint is maintained (scale back to sum = n_points/2)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Compute overlap
            overlap = correlate(h_new, 1 - h_new, mode="full")
            new_c5 = np.max(overlap) * dx
            
            # Accept with probability
            delta = new_c5 - compute_c5(h)
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h = h_new
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    
            # Cooling schedule
            T *= alpha
            
            # Early exit
            if best_c5 <= 0.3807:
                break
    
    # Phase 2: Continuous refinement with CVXPY
    def continuous_refinement():
        nonlocal h, best_h, best_c5
        
        # Start with current h
        # Use CVXPY to solve continuous problem
        h_var = cp.Variable(n_points)
        
        # Add constraint that sum equals n_points/2
        constraints = [cp.sum(h_var) == n_points / 2, 
                       h_var >= 0, 
                       h_var <= 1]
        
        # Create the objective by building constraints
        # This approach: approach problem with large-scale method
        # No optimization as CVXPROB is too slow
        
        # Instead, use a simple optimization step:
        # gradient descent with projection
        lr = 0.01
        max_iters = 1000
        
        for i in range(max_iters):
            if time.time() - start_time >= budget_s - 5:
                break
                
            # Compute gradient: numerical approximation
            grad = np.zeros(n_points)
            eps = 1e-4
            base_c5 = compute_c5(h)
            for j in range(n_points):
                h_plus = h.copy()
                h_minus = h.copy()
                h_plus[j] += eps
                h_minus[j] -= eps
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                over_plus = correlate(h_plus, 1-h_plus, mode="full")
                over_minus = correlate(h_minus, 1-h_minus, mode="full")
                grad[j] = (over_plus.max() - over_minus.max()) / (2 * eps) * dx
                
            # Gradient descent step
            new_h = h - lr * grad
            
            # Project back to feasible set [0,1] and maintain sum = n_points/2
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            h = new_h
            
            # Update best if improved
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            if best_c5 <= 0.3807:
                break
    
    # Run the phases
    binary_search()
    continuous_refinement()
    
    return (best_h.tolist(), best_c5, n_points)
