import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Convex relaxation approach
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try to solve a convex relaxation
        try:
            # Define the CVXPY variables and constraints
            variables = cp.Variable(n_points)
            constraints = [
                variables >= 0,
                variables <= 1,
                cp.sum(variables) == n_points / 2
            ]
            
            # Create a simple convex quadratic approximation of the overlap
            # This approximates |h * (1 - h)| with a simpler expression
            # But directly minimizing the max of convolution is nonconvex
            
            # Instead, let's try to solve directly with a convex approximation approach
            # Reinitialize the problem each time
            objective = cp.Minimize(cp.max(cp.convolve(variables, 1 - variables, method='direct')))
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.SCS, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = variables.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = compute_c5(h_new)
                
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            
            # Early exit condition
            if best_c5 <= 0.3807:
                break
                
        except Exception:
            pass  # If SCS fails, continue with current h
        
        # Local optimization using scipy
        try:
            from scipy.optimize import minimize
            def objective(h_vec):
                return compute_c5(h_vec)
                
            def constraint_sum(h_vec):
                return np.sum(h_vec) - n_points / 2
                
            constraints = [{'type': 'eq', 'fun': constraint_sum}]
            bounds = [(0, 1) for _ in range(n_points)]
            
            result = minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 10, 'ftol': 1e-8}
            )
            
            if result.success:
                h_local = result.x
                h_local = np.clip(h_local, 0, 1)
                h_local = h_local * (n_points / 2) / np.sum(h_local)
                c5_local = compute_c5(h_local)
                
                if c5_local < best_c5:
                    best_c5 = c5_local
                    best_h = h_local.copy()
                    
        except Exception:
            pass
        
        # Adaptive mutation
        noise_scale = max(0.01, 0.1 * (1.0 - iteration / max_iter))
        h += np.random.normal(0, noise_scale, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Check if current h is better
        c5_current = compute_c5(h)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    # Final validation of best solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
