```python
import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: clip to [0,1] and normalize to target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Gradient-based optimization
    max_iter = 10000
    lr_decay = 0.9995
    lr = 0.01
    patience = 200
    patience_counter = 0
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Compute gradient
        # This is a bit tricky: we need to find which shifted version contributes to max overlap
        # For shift k, gradient wrt h[i] is (1 - h[i+k]) * dx (for i+k < n)
        # We'll compute this directly from the signal processing
        
        # Compute derivative numerically, but we can compute it directly
        # For shift k, the gradient affecting overlap is approximately 
        # ∇C₅ = (1 - h_shifted_k) * dx when max overlap occurs at shift k
        # In fact, overlapping area i contributes to overlap at shift k iff ks + i = k
        
        # We can think of it as a local gradient update for each shift k where max overlap occurs
        # But we compute the shifted version just once and compute the gradient
        
        # Construct h_shifted_k  (for the shift that achieves max overlap)
        h_shifted = np.zeros_like(h)
        if max_k < n_points:
            h_shifted[max_k:] = h[:n_points - max_k]
        else:
            h_shifted[:n_points - max_k] = h[max_k:]
            
        grad = (1 - h_shifted) * dx
        
        # Update with gradient descent
        h_new = h - lr * grad
        
        # Project back to feasible set
        h_new = project_h(h_new, target_sum)
        
        # Evaluate new solution
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Update h
        h = h_new
        
        # Adaptive learning rate
        if patience_counter > 100:
            lr *= lr_decay
            
        # Early exit if we are very close to target
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation of best solution
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
```
