import numpy as np
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    mutation_rate = 0.1
    tournament_size = 10
    max_generations = 1000
    
    # Initialize population
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    else:
        # Initialize with random feasible vectors
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h.copy())
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Evaluate initial population
    fitness_scores = [compute_c5(h) for h in population]
    
    best_idx = np.argmin(fitness_scores)
    best_h = population[best_idx].copy()
    best_c5 = fitness_scores[best_idx]
    
    # Main genetic algorithm loop
    generation = 0
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        generation += 1
        
        # Selection: Tournament
        new_population = []
        for _ in range(population_size):
            # Tournament selection
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            new_population.append(population[winner_idx].copy())
        
        # Crossover and Mutation
        offspring = []
        for i in range(0, population_size, 2):
            parent1 = new_population[i]
            parent2 = new_population[i+1] if i+1 < population_size else new_population[0]
            
            # Uniform crossover
            mask = np.random.rand(n_points) > 0.5
            child1 = np.where(mask, parent1, parent2)
            child2 = np.where(mask, parent2, parent1)
            
            # Mutation
            for j in range(n_points):
                if random.random() < mutation_rate:
                    mutation_type = random.random()
                    if mutation_type < 0.5:
                        # Add small random perturbation
                        child1[j] += np.random.normal(0, 0.05)
                    elif mutation_type < 0.75:
                        # Set to 0 or 1
                        child1[j] = random.choice([0, 1])
                    else:
                        # Set to random value
                        child1[j] = random.random()
            
            # Repair to maintain constraints
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            
            offspring.append(child1)
            
            if len(offspring) < population_size:
                for j in range(n_points):
                    if random.random() < mutation_rate:
                        mutation_type = random.random()
                        if mutation_type < 0.5:
                            child2[j] += np.random.normal(0, 0.05)
                        elif mutation_type < 0.75:
                            child2[j] = random.choice([0, 1])
                        else:
                            child2[j] = random.random()
                
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points / 2) / np.sum(child2)
                
                offspring.append(child2)
        
        # Evaluate offspring
        offspring_fitness = []
        for h in offspring:
            fitness = compute_c5(h)
            offspring_fitness.append(fitness)
        
        # Replace worst with best offspring
        worst_indices = np.argsort(fitness_scores)[-len(offspring):]
        for i, worst_idx in enumerate(worst_indices):
            if i < len(offspring):
                population[worst_idx] = offspring[i].copy()
                fitness_scores[worst_idx] = offspring_fitness[i]
        
        # Update best solution
        best_idx = np.argmin(fitness_scores)
        if fitness_scores[best_idx] < best_c5:
            best_c5 = fitness_scores[best_idx]
            best_h = population[best_idx].copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
