```python
import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to feasible region: clip and rescale to sum to target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_h(h, target_sum)

    # Parameters
    eta = 0.1
    best_c5 = float('inf')
    best_h = h.copy()
    iter_count = 0

    # Main loop
    while time.time() - start_time < budget_s - 1:
        # Compute all correlations and find maximum
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_val = overlap[max_k] * dx

        # Update best if improved
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()

        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

        # Compute gradient for the maximum overlap shift
        # The gradient is (1 - h_shifted) * dx where h_shifted corresponds to the max_k
        if max_k >= n_points:
            # We are in the right-side tail
            shift = max_k - n_points + 1
            grad = np.zeros(n_points)
            grad[shift:] = 1 - h[:-shift]
            grad = grad * dx
        else:
            # We are in the left-side tail
            shift = n_points - 1 - max_k
            grad = np.zeros(n_points)
            grad[:n_points-shift] = 1 - h[shift:]
            grad = grad * dx
        
        # Gradient descent step
        h = h - eta * grad

        # Project back to feasibility
        h = project_h(h, target_sum)

        # Adaptive learning rate
        if iter_count % 10 == 0:
            eta = eta * 1.01 if max_val < best_c5 else eta * 0.9

        # Add noise periodically to escape local minima
        if iter_count % 100 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = project_h(h, target_sum)

        iter_count += 1

    # Final computation
    final_c5 = compute_c5(best_h, dx)

    return (best_h.tolist(), final_c5, n_points)
```
