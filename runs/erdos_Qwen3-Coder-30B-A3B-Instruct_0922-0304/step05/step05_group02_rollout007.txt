import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def compute_gradient(h, k_max):
    """Compute gradient for the maximum overlap shift."""
    n = len(h)
    grad = np.zeros(n)
    # For shift k_max, gradient points to elements where (1-h) is large, i.e., h is small
    # Simpler approximation: contribution from the current element to shift k_max
    for i in range(n):
        # This would be ∂/∂h_i of (h * (1-h))[i+k_max]
        # We use simplified first-order approximation of gradient
        if i + k_max < n:
            grad[i] = -1.0  # Approximate gradient for correlation
    return grad

def line_search(h, grad, dx, step_size, max_iter=10):
    """Perform simple line search to find good step size."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    for _ in range(max_iter):
        h_new = h - step_size * grad
        h_new = project_h(h_new, len(h) / 2)
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        step_size *= 0.5
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals() and initial_h_values:
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_h(h, n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0

    # Parameters
    momentum = 0.9
    step_size = 0.05
    best_c5 = float('inf')
    best_h = h.copy()
    no_improvement = 0
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 1:
        # Compute overlaps for all shifts
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max]
        
        # Compute approximate gradient
        grad = np.zeros(n_points)
        # Gradient is negative because we are minimizing C5
        # Approximate the gradient w.r.t. h
        # For shift k_max, the gradient at position i is approximately (1-h[i-k_max])
        for i in range(n_points):
            j = i - k_max + n_points - 1
            if 0 <= j < n_points:
                grad[i] = -1.0 * (1 - h[j])
        
        # Apply momentum
        if 'prev_grad' in locals():
            grad = momentum * grad + (1 - momentum) * prev_grad
        prev_grad = grad.copy()
        
        # Line search
        h_new, c5_new = line_search(h, grad, dx, step_size)
        
        # Check improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Adjust learning rate adaptively
        if no_improvement > 20:
            step_size *= 0.8
            no_improvement = 0
        if step_size < 1e-6:
            break
            
        h = h_new
    
    # Final validation
    best_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
