import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    """Evaluate objective and check constraints."""
    c5 = compute_c5(h, dx)
    return c5

    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize solution from initial_h_values or create a good starting point
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize initial solution
    h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative refinement using local optimization
    max_iter = 1000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Gradient-based local optimization
        def objective(h):
            return compute_c5(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Try multiple random perturbations for diversity
        num_trials = 5
        trial_results = []
        
        for _ in range(num_trials):
            # Add small random noise
            h_perturbed = h + np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = normalize_h(h_perturbed, target_sum)
            
            # Optimize this point
            result = minimize(objective, h_perturbed, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
            
            if result.success:
                trial_results.append((result.x, result.fun))
        
        if trial_results:
            # Pick best from trial results
            best_trial_h, best_trial_c5 = min(trial_results, key=lambda x: x[1])
            
            if best_trial_c5 < best_c5:
                best_c5 = best_trial_c5
                best_h = best_trial_h.copy()
            
            # Accept the best trial result as current
            h = best_trial_h.copy()
            
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
        
        # Occasionally do a structured mutation
        if iteration % 10 == 0:
            # Flip some bits or modify parts
            num_changes = max(1, n_points // 20)
            indices = np.random.choice(n_points, num_changes, replace=False)
            for i in indices:
                h[i] = 1 - h[i]  # Simple flip
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
    
    # Final refinement
    def final_objective(h):
        return compute_c5(h, dx)
    
    final_result = minimize(final_objective, best_h, method='SLSQP', bounds=[(0, 1) for _ in range(n_points)], constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}, options={'maxiter': 100})
