import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 50
    mutation_rate = 0.1
    tournament_size = 10
    max_generations = 1000
    
    # Initialize population
    def create_individual():
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values)]
        population.extend([create_individual() for _ in range(population_size - 1)])
    else:
        population = [create_individual() for _ in range(population_size)]
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_h = None
    best_c5 = float('inf')
    
    # Genetic Algorithm
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = [compute_c5(ind) for ind in population]
        
        # Track best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Selection (tournament)
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Crossover and Mutation
        new_population = []
        
        # Elitism: keep best individuals
        elite_indices = np.argsort(fitness)[:population_size//4]
        for i in elite_indices:
            new_population.append(population[i].copy())
            
        # Generate rest through crossover
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            
            # Uniform crossover
            mask = np.random.random(n_points) < 0.5
            child = np.where(mask, parent1, parent2)
            
            # Mutation
            mut_mask = np.random.random(n_points) < mutation_rate
            for i in range(n_points):
                if mut_mask[i]:
                    if np.random.random() < 0.5:
                        child[i] += np.random.normal(0, 0.1)
                    else:
                        child[i] = np.random.choice([0, 1])
            
            # Repair
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population[:population_size]
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
