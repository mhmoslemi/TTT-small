import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize to correct sum
    h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative LP optimization
    max_iter = 50
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap to identify critical lags
        overlap = correlate(h, 1 - h, mode="full")
        # Find top 10% of lags (approximately)
        top_lags = np.argsort(overlap)[-max(1, n_points//10):][::-1]
        
        # Build LP problem
        h_var = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Constraints: h in [0,1] and sum = target_sum
        constrs = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum, C5 >= 0]
        
        # Add linearized overlap constraints for top lags
        for k in top_lags:
            # Align indices for convolution-like computation
            # For lag k: overlap_k = sum h[i]*(1-h[i+k]) 
            # Linearize using current h values
            if k < n_points:
                # Index i runs from 0 to (n_points-k-1) 
                # because h[i+k] must be valid
                idxs = np.arange(n_points - k)
                # Linear approximation:
                # h[i]*(1-h[i+k]) ≈ h[i] - h[i]*h[i+k]
                # But h[i+k] is fixed in this linearization step
                # If we substitute h[i+k] = h[i+k], and we linearize h[i]*h[i+k]
                # We approximate it as h[i+k]*h[i] ≈ h[i+k]*h[i]
                # Approximate:
                # Linearization around h values, (sum h[i]*(1-h[i+k])) ≈ 
                # sum (h[i] - h[i]*h[i+k]) = sum h[i] - sum h[i]*h[i+k]
                # LHS = linearization around h = sum h[i] - sum h[i]*h[i+k] <= C5*dx
                # But that's not valid. 
                # Let's try: 
                # For h[i](1-h[i+k]) = h[i] - h[i]*h[i+k]  
                # Linearize it around current h with fixed h[i+k]:
                h_const = h[k:]
                h_var_slice = h_var[:len(h_const)]
                if k == 0:
                    # C5 >= sum(h[i]*(1-h[i])) = sum(h[i] - h[i]^2)
                    linear_term = h_var - h_var**2
                    constrs.append(cp.sum(linear_term) <= C5 * dx)
                else:
                    # C5 >= sum_{i=0}^{n-k-1} h[i]*(1 - h[i+k])
                    # = sum h[i] - sum h[i]*h[i+k]
                    # Linearize h[i]*h[i+k] around h[i] and h[i+k]
                    # Or: use h[i+k] = h_old[i+k] and linearize h[i]*h[i+k] with respect to h[i]
                    # We are linearizing term h[i] * h[i+k]  
                    # Linearization about h[i]^{old}: h[i]*h[i+k] ≈ h[i+k] * h[i]^{old} + h[i]^{old} * (h[i] - h[i]^{old})
                    # = h[i+k] * h[i]^{old} + h[i]^{old} * h[i] - h[i]^{old} * h[i]^{old} 
                    # = h[i+k] * h[i]^{old} + h[i]^{old} * (h[i] - h[i]^{old}) 
                    # = h[i]^{old} * h[i+k] + h[i]^{old} * h[i] - h[i]^{old} * h[i]^{old}
                    # Approximate: Use h[i] * h_old[i+k] ≈ h_old[i+k] * h[i] = h[i] * h_old[i+k]
                    # Since we want LHS >= overlap:
                    # C5*dx >= sum h[i] - sum h[i]^2 - sum h[i]*h[i+k]
                    # results in:  C5*dx <= sum h[i] - sum h[i]^2 - [sum h[i] * h_old[i+k] - h_old[i+k]*sum h[i]]  
                    # Concretely:  
                    # For lag k, the constraint is
                    # sum_{i=0}^{n-k-1} h[i](1-h[i+k]) = sum_{i=0}^{n-k-1} h[i] - sum_{i=0}^{n-k-1} h[i]*h[i+k] 
                    # Let's align summation indices:
                    # All terms where i runs from 0 to n-k-1
                    # Linearization:
                    # h[i]*h_old[i+k]  
                    # Then  sum_{i=0}^{n-k-1} (h[i] - h[i]*h[i+k]) 
                    # = sum h[i] - sum h[i]*h[i+k] 
                    # Using h[i+k] fixed:
                    # sum h[i] - sum h[i]*h_old[i+k] <= C5*dx -> rearrange: 
                    # sum h[i] - sum h[i]*h_old[i+k] <= C5*dx  
                    # i.e:  sum(h[i]) - sum(h[i]*h_old[i+k]) <= C5*dx  
                    # So we get terms like: h[i] - h[i]*h_old[i+k] for i in 0 to n-k-1
                    # We optimize: sum h[i] - sum h[i]*h_old[i+k] <= C5*dx  
                    # So each linear term is {h[i] - h[i]*h_old[i+k]}  
                    c1 = cp.sum(h_var_slice) - cp.sum(h_var_slice * h_const)
                    constrs.append(c1 <= C5 * dx)  
            
        # Objective: minimize C5
        prob = cp.Problem(cp.Minimize(C5), constrs)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                if C5.value < best_c5:
                    best_c5 = C5.value
                    best_h = h_var.value
                    if best_c5 <= 0.3807:
                        break
        except Exception as e:
            pass
            
        # Project back to feasible set
        h = np.clip(h_var.value, 0, 1)
        h = h * (target_sum / np.sum(h))
        
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
