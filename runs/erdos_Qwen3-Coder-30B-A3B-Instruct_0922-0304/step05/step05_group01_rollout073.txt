import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize particles
    n_particles = 20
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        best_h = np.array(initial_h_values)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    else:
        best_h = np.ones(n_points)
        best_h[n_points//2:] = 0
        best_h += np.random.normal(0, 0.05, n_points)
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    best_c5 = float('inf')
    
    # Initialize swarm
    for _ in range(n_particles):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    w_start = 0.7
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    v_max = 0.2
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        # Evaluate each particle and update personal bests
        for i in range(n_particles):
            h = particles[i]
            current_c5 = compute_c5(h)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
                
        # Update velocities and positions
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Velocity update
            r1, r2 = np.random.rand(2)
            pbest = h.copy()  # In this simple version, we just use current particle as pbest
            v = w * v + c1 * r1 * (pbest - h) + c2 * r2 * (best_h - h)
            
            # Clamp velocity
            v = np.clip(v, -v_max, v_max)
            
            # Update position
            h_new = h + v
            
            # Ensure feasibility
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[i] = h_new.copy()
            velocities[i] = v.copy()
            
    return (best_h.tolist(), best_c5, n_points)
