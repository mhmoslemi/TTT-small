import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: piecewise function
        h = np.ones(n_points)
        mid = n_points // 2
        h[mid:] = 0
        # Add some random perturbation
        h += np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def move_swap(h, swap_size):
        h_new = h.copy()
        # Select two non-overlapping segments
        start1 = np.random.randint(0, n_points - swap_size)
        start2 = np.random.randint(0, n_points - swap_size)
        # Ensure they don't overlap
        while (start2 < start1 + swap_size and start1 < start2 + swap_size):
            start2 = np.random.randint(0, n_points - swap_size)
        
        # Perform the swap
        temp = h_new[start1:start1+swap_size].copy()
        h_new[start1:start1+swap_size] = h_new[start2:start2+swap_size]
        h_new[start2:start2+swap_size] = temp
        
        # Rescale to maintain sum constraint
        total_sum = np.sum(h_new)
        h_new = h_new * (n_points / 2) / total_sum
        h_new = np.clip(h_new, 0, 1)
        
        return h_new

    def accept_probability(delta, temperature):
        if delta < 0:
            return 1.0
        return np.exp(-delta / temperature)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    temperature = 0.1
    n_swaps = 1000
    swap_size = 5
    
    for i in range(n_swaps):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_trial = move_swap(best_h, swap_size)
        c5_trial = compute_c5(h_trial)
        
        delta = c5_trial - best_c5
        if accept_probability(delta, temperature) > np.random.rand():
            best_h = h_trial
            best_c5 = c5_trial
            
        # Decrease temperature
        temperature = max(0.001, temperature * 0.999)
        
        if best_c5 <= 0.3807:
            break

    # Final polish with SLSQP
    from scipy.optimize import minimize
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    try:
        result = minimize(
            compute_c5,
            best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 30, 'ftol': 1e-8}
        )
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            best_c5 = compute_c5(best_h)
    except:
        pass  # If local search fails, keep original
        
    return (best_h.tolist(), best_c5, n_points)
